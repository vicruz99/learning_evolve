

<strategy>
To improve the packing sum of radii past 2.636, we will decouple the position and radius optimization. For any fixed set of circle centers, finding the optimal radii that maximize the sum is a **Linear Programming (LP)** problem. This approach is mathematically exact and avoids the non-convex coupling that often traps joint optimizers like SLSQP in poor local minima.

**Key Improvements:**
1. **LP-based Radius Computation:** Given centers, we solve `max sum(r_i)` subject to `r_i + r_j <= dist(i,j)` and `r_i <= dist_to_boundary(i)`. This guarantees that for any position configuration, we immediately get the best possible radii.
2. **Global Position Optimization:** We optimize the 52 center coordinates using `scipy.optimize.minimize` with the `'Powell'` method. Powell is derivative-free and handles the non-smooth objective landscape (caused by active constraints switching) much better than gradient-based methods.
3. **Diverse Initialization & Perturbation:** We generate multiple starting configurations (hexagonal lattices with varying row counts, and force-repulsion spread layouts). After finding the best configuration, we run an iterative perturbation loop to escape local basins and refine the geometry further.
4. **Efficient LP Solver:** We precompute the constraint matrix structure and use the highly optimized `highs` method in `linprog`, making each objective evaluation take ~0.2ms.
5. **Strict Feasibility Repair:** A final deterministic pass shrinks radii proportionally to resolve any numerical overlaps or boundary violations, guaranteeing the validation function passes.

```python
import numpy as np
from scipy.optimize import minimize, linprog

N = 26

# Precompute constant structure of the LP constraint matrix
# Constraints: r_i + r_j <= D_ij  (pairwise)
#              r_i   <= B_i       (boundary)
# Matrix shape: (N*(N-1)//2 + N, N)
A_CONSTRAINTS = np.zeros((N * (N - 1) // 2 + N, N))
_idx = 0
for i in range(N):
    for j in range(i + 1, N):
        A_CONSTRAINTS[_idx, i] = 1.0
        A_CONSTRAINTS[_idx, j] = 1.0
        _idx += 1
for i in range(N):
    A_CONSTRAINTS[_idx, i] = 1.0
    _idx += 1

# Right-hand side array to be updated per evaluation
b_ub_buffer = np.zeros(A_CONSTRAINTS.shape[0])

def compute_max_radii_sum(centers):
    """
    Solves LP to find optimal radii for fixed centers.
    Returns (sum_radii, radii_array).
    """
    n = centers.shape[0]
    
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    xb = np.minimum(centers[:, 0], 1.0 - centers[:, 0])
    yb = np.minimum(centers[:, 1], 1.0 - centers[:, 1])
    b_bound = np.minimum(xb, yb)
    b_bound = np.maximum(b_bound, 1e-9)  # Prevent infeasibility from exact boundary hits
    
    # Pairwise distances
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    # Fill b_ub buffer
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            b_ub_buffer[idx] = dists[i, j]
            idx += 1
    b_ub_buffer[idx:] = b_bound
    
    # Solve LP: min -sum(r) s.t. A r <= b, r >= 0
    res = linprog(-np.ones(n), A_ub=A_CONSTRAINTS, b_ub=b_ub_buffer, 
                  bounds=[(0.0, None)] * n, method='highs')
    
    if res.success:
        return -res.fun, res.x
    return 0.0, np.zeros(n)

def objective_function(p):
    """Objective for Powell: minimize negative sum of optimal radii."""
    centers = p.reshape(-1, 2)
    # Keep centers strictly inside to avoid numerical singularities in LP
    centers = np.clip(centers, 1e-6, 1.0 - 1e-6)
    s, _ = compute_max_radii_sum(centers)
    return -s

def generate_hex_init(row_counts, r0, seed):
    """Generates a perturbed hexagonal lattice initialization."""
    rng = np.random.default_rng(seed)
    centers = []
    y = r0
    for row_idx, count in enumerate(row_counts):
        offset = r0 if row_idx % 2 == 1 else 0.0
        x = r0 + offset
        for _ in range(count):
            centers.append([x + rng.normal(0, 0.005), y + rng.normal(0, 0.005)])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
    centers = np.array(centers[:N])
    return np.clip(centers, 0.05, 0.95)

def force_repulsion_init(seed):
    """Generates well-spread points using simple force-directed repulsion."""
    rng = np.random.default_rng(seed)
    c = rng.uniform(0.2, 0.8, (N, 2))
    # Push corners explicitly
    corners = np.array([[0.1, 0.1], [0.9, 0.1], [0.1, 0.9], [0.9, 0.9]])
    c[:4] = corners
    
    for _ in range(500):
        forces = np.zeros_like(c)
        for i in range(N):
            for j in range(i + 1, N):
                d_vec = c[i] - c[j]
                dist = np.linalg.norm(d_vec)
                if dist < 0.25 and dist > 1e-5:
                    push = (0.25 - dist) * 0.5
                    f = (d_vec / dist) * push
                    forces[i] += f
                    forces[j] -= f
        c += forces * 0.02
        c = np.clip(c, 0.02, 0.98)
    return c

def repair_solution(centers, radii):
    """Deterministically resolves numerical overlaps and boundary violations."""
    for _ in range(50):
        changed = False
        # Fix pairwise overlaps
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                req = radii[i] + radii[j]
                if d < req - 1e-10:
                    shrink = (req - d) / 2.0 + 1e-10
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        # Fix boundary violations
        for i in range(N):
            x, y = centers[i]
            max_r = min(x, 1.0 - x, y, 1.0 - y)
            if radii[i] > max_r + 1e-10:
                radii[i] = max_r
                changed = True
        if not changed:
            break
    return np.maximum(radii, 0.0)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize sum of radii."""
    rng = np.random.default_rng(42)
    bounds = [(1e-6, 1.0 - 1e-6)] * (2 * N)
    
    best_p = None
    best_sum = -np.inf
    
    # Phase 1: Generate diverse initial configurations
    starts = []
    
    # Hexagonal patterns with varying row structures
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 5, 5, 6], 
        [4, 6, 6, 6, 4], [6, 4, 6, 6, 4], [5, 5, 5, 6, 5]
    ]
    for i, pat in enumerate(patterns):
        for r0 in [0.09, 0.10, 0.105]:
            starts.append(generate_hex_init(pat, r0, i * 100 + int(r0 * 1000)).flatten())
            
    # Force-repulsion starts
    for i in range(8):
        starts.append(force_repulsion_init(i).flatten())
        
    # Phase 2: Multi-start Powell Optimization
    for i, p0 in enumerate(starts):
        try:
            res = minimize(objective_function, p0, method='Powell', bounds=bounds,
                           options={'maxiter': 2000, 'ftol': 1e-13, 'xtol': 1e-13})
            if -res.fun > best_sum:
                best_sum = -res.fun
                best_p = res.x.copy()
        except Exception:
            continue
            
    # Phase 3: Iterative Perturbation to escape local minima
    if best_p is not None:
        for step in range(40):
            noise_scale = 0.004 * (0.85 ** (step // 10))
            p_trial = best_p + rng.normal(0, noise_scale, best_p.shape)
            p_trial = np.clip(p_trial, 1e-6, 1.0 - 1e-6)
            try:
                res = minimize(objective_function, p_trial, method='Powell', bounds=bounds,
                               options={'maxiter': 1500, 'ftol': 1e-13})
                if -res.fun > best_sum:
                    best_sum = -res.fun
                    best_p = res.x.copy()
            except Exception:
                continue
                
    # Phase 4: Extract final configuration and strict repair
    centers = best_p.reshape(-1, 2)
    # Compute exact optimal radii for the final centers
    _, radii = compute_max_radii_sum(centers)
    radii = repair_solution(centers, radii)
    
    final_sum = float(np.sum(radii))
    return centers, radii, final_sum
```
