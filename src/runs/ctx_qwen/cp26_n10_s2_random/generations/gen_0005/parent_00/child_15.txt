

<strategy>
The key to improving the packing lies in decoupling the radius optimization from the center optimization. Given fixed centers, finding the optimal radii that maximize the sum is a convex Linear Programming problem. This allows us to:
1. Formulate an exact, fast LP solver for radii given any center configuration.
2. Optimize only the 52 center coordinates using a robust, gradient-free method (`Powell`) that can handle the non-smooth objective landscape created by the LP bottleneck.
3. Use diverse initial configurations (hexagonal lattices with varying row structures, force-directed repulsion) combined with adaptive perturbation to escape local minima.
4. Precompute the constant structure of the LP constraint matrix to drastically speed up the inner optimization loop.
5. Apply a strict numerical repair pass to guarantee the validator's `1e-12` tolerance is met without sacrificing the optimized sum.

This hybrid LP + global/local search approach is mathematically sound and typically pushes packings past the thresholds reached by joint constrained optimization, which often struggles with the high dimensionality and non-convex constraints.
</strategy>

```python
import numpy as np
from scipy.optimize import minimize, linprog

N = 26

# Precompute constant parts of LP constraint matrix for speed
# Constraints: r_i + r_j <= dist(i,j) for all i < j
#              r_i <= dist_to_boundary(i) for all i
pair_idx = np.triu_indices(N, 1)
A_pair = np.zeros((N*(N-1)//2, N))
A_pair[np.arange(len(pair_idx[0])), pair_idx[0]] = 1.0
A_pair[np.arange(len(pair_idx[0])), pair_idx[1]] = 1.0
A_bound = np.eye(N)
A_ub_const = np.vstack([A_pair, A_bound])

def compute_lp_radii(centers_flat):
    """Solves LP for optimal radii given fixed centers. Returns (sum_radii, radii_array)."""
    c = centers_flat.reshape(N, 2)
    c = np.clip(c, 1e-6, 1.0 - 1e-6)
    
    # Boundary limits
    ub = np.minimum(np.minimum(c[:, 0], 1.0 - c[:, 0]),
                    np.minimum(c[:, 1], 1.0 - c[:, 1]))
    ub = np.maximum(ub, 0.0)
    
    # Pairwise distances
    diff = c[:, None, :] - c[None, :, :]
    dists = np.linalg.norm(diff, axis=2)
    np.fill_diagonal(dists, np.inf)
    
    # RHS vector for LP
    b_ub = np.concatenate([dists[pair_idx], ub])
    
    # Solve: max sum(r) s.t. A_ub r <= b_ub, r >= 0
    try:
        res = linprog(-np.ones(N), A_ub=A_ub_const, b_ub=b_ub, 
                      bounds=[(0, None)]*N, method='highs')
        if res.success:
            return -res.fun, res.x
    except Exception:
        pass
    return 0.0, np.zeros(N)

def objective_centers(v):
    """Objective for center optimization: minimize negative sum of radii."""
    s, _ = compute_lp_radii(v)
    return -s

def generate_inits():
    """Generates diverse initial center configurations."""
    rng = np.random.default_rng(42)
    inits = []
    
    # Hexagonal lattice patterns with varying row distributions
    patterns = [[5,6,5,6,4], [6,5,6,5,4], [5,5,5,5,6], [6,6,5,5,4], 
                [5,5,5,6,5], [4,6,6,5,5], [5,4,6,5,6], [6,5,5,5,5]]
    
    for pat in patterns:
        for r_est in [0.09, 0.098, 0.105]:
            c = []
            y = r_est
            for r_idx, cnt in enumerate(pat):
                shift = r_est if r_idx % 2 == 1 else 0.0
                x = r_est + shift
                for _ in range(cnt):
                    c.append([x, y])
                    x += 2.0 * r_est
                y += r_est * np.sqrt(3)
            c = np.array(c[:N]) + rng.normal(0, 0.003, (N, 2))
            inits.append(np.clip(c.flatten(), 0.02, 0.98))
            
    # Force-directed repulsion starts to find non-lattice optima
    for _ in range(15):
        c = rng.uniform(0.2, 0.8, (N, 2))
        for _ in range(60):
            diff = c[:, None, :] - c[None, :, :]
            dists = np.linalg.norm(diff, axis=2)
            np.fill_diagonal(dists, 1.0)
            forces = diff / (dists**2 + 1e-5)[:, :, None]
            c += 0.006 * np.sum(forces, axis=1)
            c = np.clip(c, 0.05, 0.95)
        inits.append(c.flatten())
        
    return inits

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Packs 26 circles in a unit square to maximize the sum of radii."""
    inits = generate_inits()
    best_val = np.inf
    best_v = None
    
    bounds = [(0.02, 0.98)] * (2*N)
    
    # Phase 1: Local optimization of centers using Powell
    # Powell efficiently navigates the non-smooth landscape created by the LP bottleneck
    for v0 in inits:
        try:
            res = minimize(objective_centers, v0, method='Powell', bounds=bounds,
                           options={'maxiter': 5000, 'ftol': 1e-13, 'xtol': 1e-13})
            if res.fun < best_val:
                best_val = res.fun
                best_v = res.x.copy()
        except Exception:
            continue
            
    # Phase 2: Adaptive perturbation to escape local minima
    if best_v is not None:
        curr = best_v.copy()
        rng = np.random.default_rng(123)
        for step in range(40):
            noise = 0.006 * (0.85 ** step)
            v_pert = curr + rng.normal(0, noise, curr.shape)
            v_pert = np.clip(v_pert, 0.02, 0.98)
            try:
                res = minimize(objective_centers, v_pert, method='Powell', bounds=bounds,
                               options={'maxiter': 2000, 'ftol': 1e-13, 'xtol': 1e-13})
                if res.fun < best_val:
                    best_val = res.fun
                    best_v = res.x.copy()
                    curr = best_v.copy()
            except Exception:
                continue
                
    # Extract optimal centers and compute exact radii via LP
    final_centers = best_v.reshape(N, 2)
    _, final_radii = compute_lp_radii(final_centers)
    
    # Phase 3: Strict numerical repair to guarantee validator passes
    for _ in range(100):
        changed = False
        # Fix pairwise overlaps
        for i in range(N):
            for j in range(i+1, N):
                d = np.linalg.norm(final_centers[i] - final_centers[j])
                req = final_radii[i] + final_radii[j]
                if d < req - 1e-12:
                    shrink = (req - d) / 2.0 + 1e-9
                    final_radii[i] -= shrink
                    final_radii[j] -= shrink
                    changed = True
        # Fix boundary violations
        for i in range(N):
            x, y, r = final_centers[i, 0], final_centers[i, 1], final_radii[i]
            mr = min(x, 1.0-x, y, 1.0-y)
            if r > mr + 1e-12:
                final_radii[i] = mr
                changed = True
        if not changed:
            break
            
    final_radii = np.maximum(final_radii, 0.0)
    return final_centers, final_radii, float(np.sum(final_radii))
```
