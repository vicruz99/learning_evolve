import numpy as np
from scipy.optimize import minimize, linprog

N = 26
idx_i, idx_j = np.triu_indices(N, 1)
A_ub = np.zeros((len(idx_i), N))
A_ub[np.arange(len(idx_i)), idx_i] = 1.0
A_ub[np.arange(len(idx_i)), idx_j] = 1.0

def compute_lp_sum(centers):
    """Solves LP to find maximum sum of radii for fixed centers."""
    ub = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    ub = np.maximum(ub, 1e-9)
    
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    b_ub = dists[idx_i, idx_j]
    
    res = linprog(c=-np.ones(N), A_ub=A_ub, b_ub=b_ub, 
                  bounds=[(0, u) for u in ub], method='highs')
    if res.success:
        return -res.fun, res.x
    return 0.0, np.zeros(N)

def objective(centers_flat):
    """Objective function for center optimization: minimize negative LP sum."""
    centers = centers_flat.reshape(N, 2)
    centers = np.clip(centers, 1e-4, 1.0 - 1e-4)
    val, _ = compute_lp_sum(centers)
    return -val

def generate_starts():
    """Generates diverse initial center configurations."""
    np.random.seed(42)
    starts = []
    
    # 1. Hexagonal lattice patterns
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 5, 5, 6],
        [6, 4, 6, 5, 5], [4, 6, 6, 6, 4], [5, 4, 6, 6, 5],
        [6, 6, 5, 5, 4], [5, 5, 6, 5, 5], [4, 5, 6, 5, 6],
        [5, 5, 5, 6, 5], [6, 5, 5, 5, 5], [5, 6, 6, 5, 4]
    ]
    
    for pat in patterns:
        for r_est in np.linspace(0.085, 0.105, 4):
            c = []
            y = r_est
            for r_idx, cnt in enumerate(pat):
                shift = r_est if r_idx % 2 == 1 else 0.0
                x = r_est + shift
                for _ in range(cnt):
                    c.append([x, y])
                    x += 2.0 * r_est
                y += r_est * np.sqrt(3)
            c = np.array(c[:N])
            c += np.random.normal(0, 0.003, c.shape)
            c = np.clip(c, 0.02, 0.98)
            starts.append(c.flatten())
            
    # 2. Force-repulsed random starts
    for _ in range(25):
        c = np.random.rand(N, 2) * 0.7 + 0.15
        for _ in range(120):
            forces = np.zeros_like(c)
            for i in range(N):
                for j in range(i + 1, N):
                    d_vec = c[i] - c[j]
                    dist = np.linalg.norm(d_vec)
                    if dist < 0.2 and dist > 1e-6:
                        push = (0.2 - dist) * 15.0 / dist
                        f = d_vec * push
                        forces[i] += f
                        forces[j] -= f
            c += forces * 0.008
            c = np.clip(c, 0.05, 0.95)
        starts.append(c.flatten())
        
    # 3. Structured grid subsets
    for gx in [5, 6, 7]:
        for gy in [4, 5, 6]:
            xs = np.linspace(0.1, 0.9, gx)
            ys = np.linspace(0.1, 0.9, gy)
            grid = np.array(np.meshgrid(xs, ys)).T.reshape(-1, 2)
            if len(grid) >= N:
                idx = np.random.choice(len(grid), N, replace=False)
                c = grid[idx].copy()
                c += np.random.normal(0, 0.008, c.shape)
                c = np.clip(c, 0.02, 0.98)
                starts.append(c.flatten())
                
    # 4. Corner-focused starts
    for _ in range(10):
        c = np.random.rand(N, 2) * 0.6 + 0.2
        corners = [[0.08, 0.08], [0.92, 0.08], [0.08, 0.92], [0.92, 0.92]]
        c[:4] = corners
        c += np.random.normal(0, 0.005, c.shape)
        c = np.clip(c, 0.02, 0.98)
        starts.append(c.flatten())
        
    return starts

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    starts = generate_starts()
    best_centers = None
    best_radii = None
    best_sum = -1.0
    
    # Phase 1: Multi-start local optimization
    for x0 in starts:
        try:
            res = minimize(objective, x0, method='Powell',

            if -res.fun > best_sum:
                best_sum = -res.fun
                best_centers = res.x.reshape(N, 2)

        except Exception:
            pass

    if best_centers is not None:

            try:

    if best_centers is not None:

            # Fix overlaps
            for i in range(N):

                    req = best_radii[i] + best_radii[j]

                        shrink = (req - d) / 2.0 + 1e-9
                        best_radii[i] -= shrink
                        best_radii[j] -= shrink
                        changed = True
            # Fix boundary violations
            for i in range(N):
                x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
                max_r = min(x, 1.0 - x, y, 1.0 - y)
                if r > max_r + 1e-12:
                    best_radii[i] = max_r
                    changed = True
            if not changed:
                break
                
        best_radii = np.maximum(best_radii, 0.0)
        
    return best_centers, best_radii, float(np.sum(best_radii))
