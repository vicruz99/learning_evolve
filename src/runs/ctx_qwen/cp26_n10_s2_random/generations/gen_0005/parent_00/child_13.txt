import numpy as np
from scipy.optimize import minimize, linprog
from itertools import combinations

N = 26

def compute_constraints_joint(v):
    """Computes all boundary and non-overlap constraints for joint optimization."""
    c = v[:2*N].reshape(N, 2)
    r = v[2*N:]
    
    con = []
    con.append(c[:, 0] - r)
    con.append(1.0 - c[:, 0] - r)
    con.append(c[:, 1] - r)
    con.append(1.0 - c[:, 1] - r)
    
    idx = np.triu_indices(N, 1)
    d = np.linalg.norm(c[idx[0]] - c[idx[1]], axis=1)
    con.append(d - (r[idx[0]] + r[idx[1]]))
    
    return np.concatenate(con)

def obj_joint(v):
    """Objective for joint optimization: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def get_max_radii_lp(centers):
    """Solves LP to find exact maximum sum of radii for fixed centers."""
    n = centers.shape[0]
    ub = np.minimum(
        np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
        np.minimum(centers[:, 1], 1.0 - centers[:, 1])
    )
    ub = np.maximum(ub, 0.0)
    
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    idx_pairs = list(combinations(range(n), 2))
    A_ub = np.zeros((len(idx_pairs), n))
    for k, (i, j) in enumerate(idx_pairs):
        A_ub[k, i] = 1.0
        A_ub[k, j] = 1.0
    b_ub = np.array([dists[i, j] for i, j in idx_pairs])
    
    res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=[(0, u) for u in ub], method='highs')
    if res.success:
        return res.x
    return np.full(n, 0.01)

def obj_centers(centers_flat):
    """Objective for center optimization: evaluates exact LP radii sum."""
    c = centers_flat.reshape(N, 2)
    c = np.clip(c, 1e-6, 1.0 - 1e-6)
    r = get_max_radii_lp(c)
    return -np.sum(r)

def generate_inits(rng):
    """Generates diverse initial configurations for multi-start optimization."""
    inits = []
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 5, 5, 6],
        [6, 4, 6, 5, 5], [4, 6, 6, 6, 4], [5, 4, 6, 6, 5],
        [6, 6, 5, 5, 4], [5, 5, 6, 5, 5], [4, 5, 6, 5, 6],
        [5, 5, 5, 6, 5], [5, 5, 5, 5, 5, 1]
    ]
    
    for pat in patterns:
        for r_est in [0.092, 0.098, 0.102, 0.108]:
            c = []
            y = r_est
            for r_idx, cnt in enumerate(pat):
                shift = r_est if r_idx % 2 == 1 else 0.0
                x = r_est + shift
                for _ in range(cnt):
                    c.append([x, y])
                    x += 2.0 * r_est
                y += r_est * np.sqrt(3)
            c = np.array(c[:N])
            c += rng.normal(0, 0.005, c.shape)
            c = np.clip(c, 0.05, 0.95)
            r_init = np.full(N, 0.09)
            inits.append(np.concatenate([c.flatten(), r_init]))
            
    for _ in range(15):
        c = rng.uniform(0.15, 0.85, (N, 2))
        for _ in range(100):
            for i in range(N):
                for j in range(i+1, N):
                    d_vec = c[i] - c[j]
                    dist = np.linalg.norm(d_vec)
                    if dist < 0.15:
                        push = (0.15 - dist) * 0.5
                        c[i] += d_vec / dist * push
                        c[j] -= d_vec / dist * push
        c = np.clip(c, 0.1, 0.9)
        r_init = np.full(N, 0.08)
        inits.append(np.concatenate([c.flatten(), r_init]))
        
    return inits

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    best_v = None
    best_sum = -1.0
    
    bounds = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': compute_constraints_joint}
    
    inits = generate_inits(rng)
    
    # Phase 1: Joint SLSQP Optimization
    for v0 in inits:
        try:
            res = minimize(obj_joint, v0, method='SLSQP', bounds=bounds, 
                          constraints=cons, options={'maxiter': 3000, 'ftol': 1e-12})
            s = np.sum(res.x[2*N:])
            if s > best_sum:
                best_sum = s
                best_v = res.x.copy()
        except Exception:
            pass
            
    if best_v is None:
        best_v = inits[0]
        
    # Phase 2: Center-only optimization using LP for exact radii
    best_centers = best_v[:2*N].reshape(N, 2)
    best_c_sum = -1.0
    best_c_flat = best_centers.flatten()
    
    # Generate perturbation candidates around the best found configuration
    cands = [best_c_flat]
    for i in range(12):
        pert = best_c_flat + rng.normal(0, 0.006, best_c_flat.shape)
        cands.append(np.clip(pert, 0.02, 0.98))
        
    for x0 in cands:
        try:
            res = minimize(obj_centers, x0, method='Powell',
                          options={'maxiter': 1200, 'ftol': 1e-12})
            if -res.fun > best_c_sum:
                best_c_sum = -res.fun
                best_c_flat = res.x.copy()
        except Exception:
            pass
            
    # Extract final centers and compute optimal radii via LP
    final_centers = best_c_flat.reshape(N, 2)
    final_radii = get_max_radii_lp(final_centers)
    
    # Strict numerical repair to guarantee validation passes
    for _ in range(50):
        changed = False
        for i in range(N):
            for j in range(i+1, N):
                d = np.linalg.norm(final_centers[i] - final_centers[j])
                req = final_radii[i] + final_radii[j]
                if d < req - 1e-12:
                    shrink = (req - d) / 2.0 + 1e-9
                    final_radii[i] -= shrink
                    final_radii[j] -= shrink
                    changed = True
        for i in range(N):
            x, y, r = final_centers[i, 0], final_centers[i, 1], final_radii[i]
            max_r = min(x, 1.0-x, y, 1.0-y)
            if r > max_r + 1e-12:
                final_radii[i] = max_r
                changed = True
        if not changed:
            break
            
    final_radii = np.maximum(final_radii, 0.0)
    final_sum = float(np.sum(final_radii))
    
    return final_centers, final_radii, final_sum
