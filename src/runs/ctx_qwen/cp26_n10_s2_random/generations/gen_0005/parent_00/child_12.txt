

<strategy>
To maximize the sum of radii for 26 circles in a unit square, we decouple the problem into optimizing centers first, then solving for radii exactly via Linear Programming (LP). This avoids the suboptimal radius choices common in joint non-linear optimization.

1. **LP-based Objective for Centers**: For any fixed set of centers, the maximum sum of radii is found by solving `max sum(r_i)` subject to `r_i + r_j <= dist(i,j)` and boundary constraints. This LP is solved instantly using `scipy.optimize.linprog` with the `highs` solver.
2. **Derivative-Free Center Optimization**: We use `Powell` method to maximize the LP objective over center coordinates. Powell handles the non-smooth nature of the LP objective (kinks when active constraints change) effectively.
3. **Diverse Initializations**: We generate hexagonal lattice patterns with varying row counts and densities, plus force-directed repulsion configurations to cover both lattice and non-lattice topologies.
4. **Escape Local Minima**: An adaptive perturbation phase progressively adds noise to the best found centers and re-optimizes, allowing the solver to jump out of local basins.
5. **Joint Refinement**: After finding optimal centers, we run a constrained SLSQP optimization on joint `(x,y,r)` variables to fine-tune positions and radii simultaneously, smoothing out minor numerical discrepancies.
6. **Strict Numerical Repair**: A final pass shrinks radii proportionally to resolve any microscopic overlaps or boundary violations introduced by floating-point arithmetic, guaranteeing the validator passes.
</strategy>

```python
import numpy as np
from scipy.optimize import minimize, linprog

N = 26

# Precompute constant structural parts for the LP constraints
PAIRS_I, PAIRS_J = np.triu_indices(N, 1)
A_LP = np.zeros((len(PAIRS_I), N))
A_LP[np.arange(len(PAIRS_I)), PAIRS_I] = 1.0
A_LP[np.arange(len(PAIRS_I)), PAIRS_J] = 1.0
ONES_N = np.ones(N)

def compute_max_radii_lp(centers):
    """Solves LP to find optimal radii for fixed centers."""
    n = centers.shape[0]
    # Max radius limited by distance to square boundaries
    ub = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    ub = np.maximum(ub, 1e-9)
    
    # Pairwise Euclidean distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    b_ub = dists[PAIRS_I, PAIRS_J]
    bounds = [(0.0, u) for u in ub]
    res = linprog(-ONES_N, A_ub=A_LP, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return np.full(n, 1e-5)

def objective_lp(x_flat):
    """Objective for center optimization: negative sum of optimal radii."""
    centers = np.clip(x_flat.reshape(N, 2), 1e-4, 1.0 - 1e-4)
    radii = compute_max_radii_lp(centers)
    return -np.sum(radii)

def generate_hex_init(row_counts, r_est, rng):
    """Generates a hexagonal lattice initialization with specified row counts."""
    c = []
    y = r_est
    for r_idx, cnt in enumerate(row_counts):
        shift = r_est if r_idx % 2 == 1 else 0.0
        x = r_est + shift
        for _ in range(cnt):
            c.append([x, y])
            x += 2.0 * r_est
        y += r_est * np.sqrt(3)
    c = np.array(c[:N])
    c += rng.normal(0, 0.003, c.shape)
    return np.clip(c, 0.05, 0.95)

def joint_obj(v):
    """Objective for joint SLSQP: minimize negative sum of radii."""
    return -np.sum(v[2*N:])
        
def joint_cons(v):
    """Computes boundary and non-overlap constraints for joint optimization."""
    c = v[:2*N].reshape(N, 2)
    r = v[2*N:]
    con = []
    con.append(c[:, 0] - r)
    con.append(1.0 - c[:, 0] - r)
    con.append(c[:, 1] - r)
    con.append(1.0 - c[:, 1] - r)
    d = np.linalg.norm(c[PAIRS_I] - c[PAIRS_J], axis=1)
    con.append(d - (r[PAIRS_I] + r[PAIRS_J]))
    return np.concatenate(con)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Packs 26 circles in a unit square to maximize the sum of radii."""
    rng = np.random.default_rng(42)
    best_val = np.inf
    best_centers = None
    
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 5, 5, 6],
        [6, 4, 6, 5, 5], [4, 6, 6, 6, 4], [5, 4, 6, 6, 5],
        [6, 6, 5, 5, 4], [5, 5, 6, 5, 5], [4, 5, 6, 5, 6],
        [5, 5, 5, 6, 5], [5, 5, 5, 5, 5, 1], [4, 5, 5, 6, 6]
    ]
    
    candidates = []
    for pat in patterns:
        for r_est in [0.085, 0.092, 0.098, 0.105]:
            c = generate_hex_init(pat, r_est, rng)
            candidates.append(c.flatten())
            
    for _ in range(30):
        c = rng.uniform(0.1, 0.9, (N, 2))
        for _ in range(40):
            diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
            dists = np.linalg.norm(diff, axis=2)
            np.fill_diagonal(dists, 1.0)
            forces = diff / (dists**2 + 1e-5)[:, :, np.newaxis]
            c += 0.005 * np.sum(forces, axis=1)
        c = np.clip(c, 0.08, 0.92)
        candidates.append(c.flatten())

    # Phase 1: Powell optimization on centers using exact LP radii
    for x0 in candidates:
        try:
            res = minimize(objective_lp, x0, method='Powell',
                           options={'maxiter': 600, 'ftol': 1e-12, 'xtol': 1e-12})
            if res.fun < best_val:
                best_val = res.fun
                best_centers = res.x.reshape(N, 2)
        except Exception:
            continue

    if best_centers is None:
        best_centers = candidates[0].reshape(N, 2)

    # Phase 2: Adaptive perturbation to escape local minima
    curr = best_centers.copy()
    for i in range(25):
        sigma = 0.008 * (0.85 ** i)
        pert = curr + rng.normal(0, sigma, curr.shape)
        pert = np.clip(pert, 0.05, 0.95)
        try:
            res = minimize(objective_lp, pert.flatten(), method='Powell',
                           options={'maxiter': 500, 'ftol': 1e-12})
            if res.fun < best_val:
                best_val = res.fun
                best_centers = res.x.reshape(N, 2)
                curr = best_centers
        except Exception:
            continue

    # Phase 3: Joint refinement with SLSQP
    centers = best_centers.copy()
    radii = compute_max_radii_lp(centers)
    
    v0 = np.concatenate([centers.flatten(), radii])
    bounds = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': joint_cons}
    
    try:
        res = minimize(joint_obj, v0, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 2000, 'ftol': 1e-14})
        if np.min(joint_cons(res.x)) >= -1e-7:
            centers = res.x[:2*N].reshape(N, 2)
            radii = res.x[2*N:]
    except Exception:
        pass

    # Phase 4: Strict numerical repair
    for _ in range(50):
        changed = False
        for i in range(N):
            for j in range(i+1, N):
                d = np.linalg.norm(centers[i] - centers[j])
                req = radii[i] + radii[j]
                if d < req - 1e-12:
                    shrink = (req - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        for i in range(N):
            x, y, r = centers[i, 0], centers[i, 1], radii[i]
            max_r = min(x, 1.0-x, y, 1.0-y)
            if r > max_r + 1e-12:
                radii[i] = max_r
                changed = True
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    return centers, radii, float(np.sum(radii))
```
