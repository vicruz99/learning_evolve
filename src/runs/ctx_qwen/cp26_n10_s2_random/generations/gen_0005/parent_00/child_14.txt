import numpy as np
from scipy.optimize import minimize, linprog

N = 26

# Precompute constant pairwise constraint matrix for LP: r_i + r_j <= dist_ij
PAIR_INDICES = []
for i in range(N):
    for j in range(i + 1, N):
        PAIR_INDICES.append((i, j))

A_ub_pairs = np.zeros((len(PAIR_INDICES), N))
for idx, (i, j) in enumerate(PAIR_INDICES):
    A_ub_pairs[idx, i] = 1.0
    A_ub_pairs[idx, j] = 1.0

def solve_radii_lp(centers):
    """Solves LP to find max sum of radii for fixed centers."""
    c_obj = -np.ones(N)
    
    # Boundary limits for each circle
    ub = np.minimum(
        np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
        np.minimum(centers[:, 1], 1.0 - centers[:, 1])
    )
    ub = np.maximum(ub, 1e-9)
    
    # Pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    # Build RHS for pairwise constraints
    b_ub = np.zeros(len(PAIR_INDICES))
    for idx, (i, j) in enumerate(PAIR_INDICES):
        b_ub[idx] = dists[i, j]
        
    bounds = [(0.0, u) for u in ub]
    res = linprog(c_obj, A_ub=A_ub_pairs, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return -res.fun, res.x
    return 0.0, np.zeros(N)

def objective_centers(centers_flat):
    """Objective for center optimization: maximize sum of radii via LP."""
    centers = centers_flat.reshape(N, 2)
    centers = np.clip(centers, 1e-5, 1.0 - 1e-5)
    s, _ = solve_radii_lp(centers)
    return -s

def generate_hex_centers(row_counts, r_est, rng):
    """Generates a hexagonal lattice initialization."""
    c = []
    y = r_est
    for r_idx, cnt in enumerate(row_counts):
        shift = r_est if r_idx % 2 == 1 else 0.0
        x = r_est + shift
        for _ in range(cnt):
            c.append([x, y])
            x += 2.0 * r_est
        y += r_est * np.sqrt(3)
    c = np.array(c[:N])
    c += rng.normal(0, 0.003, c.shape)
    return np.clip(c, 0.05, 0.95)

def compute_constraints_joint(v):
    """Joint boundary and non-overlap constraints for SLSQP."""
    c = v[:2*N].reshape(N, 2)
    r = v[2*N:]
    con = []
    con.append(c[:, 0] - r)
    con.append(1.0 - c[:, 0] - r)
    con.append(c[:, 1] - r)
    con.append(1.0 - c[:, 1] - r)
    idx = np.triu_indices(N, 1)
    d = np.linalg.norm(c[idx[0]] - c[idx[1]], axis=1)
    con.append(d - (r[idx[0]] + r[idx[1]]))
    return np.concatenate(con)

def obj_joint(v):
    """Joint objective: maximize sum of radii."""
    return -np.sum(v[2*N:])

def run_packing() -> tuple:
    rng = np.random.default_rng(42)
    best_sum = -1.0
    best_centers = None
    
    # High-quality hexagonal patterns summing to 26
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 5, 5, 6],
        [6, 4, 6, 5, 5], [4, 6, 6, 6, 4], [5, 5, 6, 5, 5],
        [6, 6, 5, 5, 4], [5, 5, 5, 6, 5]
    ]
    
    starts = []
    for pat in patterns:
        for r_est in [0.092, 0.098, 0.104]:
            starts.append(generate_hex_centers(pat, r_est, rng).flatten())
    for _ in range(4):
        starts.append(rng.uniform(0.15, 0.85, (N, 2)).flatten())
        
    bounds_c = [(1e-4, 1.0 - 1e-4)] * (2 * N)
    
    # Phase 1: Optimize centers using Powell with LP evaluation
    for s in starts:
        try:
            res = minimize(objective_centers, s, method='Powell', bounds=bounds_c,
                           options={'maxiter': 500, 'ftol': 1e-12, 'xtol': 1e-12})
            curr_sum = -res.fun
            if curr_sum > best_sum:
                best_sum = curr_sum
                best_centers = res.x.reshape(N, 2)
        except Exception:
            continue
            
    if best_centers is None:
        best_centers = rng.uniform(0.2, 0.8, (N, 2))
        
    # Phase 2: Compute exact radii via LP for the best centers found
    _, best_radii = solve_radii_lp(best_centers)
    best_sum = float(np.sum(best_radii))
    
    # Phase 3: Joint SLSQP polish to fine-tune both centers and radii
    v0 = np.concatenate([best_centers.flatten(), best_radii])
    bounds_joint = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': compute_constraints_joint}
    
    try:
        res = minimize(obj_joint, v0, method='SLSQP', bounds=bounds_joint,
                       constraints=cons, options={'maxiter': 2000, 'ftol': 1e-13})
        if -res.fun > best_sum - 1e-5:
            final_centers = res.x[:2*N].reshape(N, 2)
            final_radii = res.x[2*N:]
        else:
            final_centers = best_centers
            final_radii = best_radii
    except Exception:
        final_centers = best_centers
        final_radii = best_radii
        
    # Phase 4: Strict numerical repair to guarantee validation passes
    for _ in range(60):
        changed = False
        # Fix overlaps
        for i in range(N):
            for j in range(i + 1, N):
                d = np.linalg.norm(final_centers[i] - final_centers[j])
                req = final_radii[i] + final_radii[j]
                if d < req - 1e-12:
                    shrink = (req - d) / 2.0 + 1e-9
                    final_radii[i] -= shrink
                    final_radii[j] -= shrink
                    changed = True
        # Fix boundary violations
        for i in range(N):
            x, y, r = final_centers[i, 0], final_centers[i, 1], final_radii[i]
            max_r = min(x, 1.0 - x, y, 1.0 - y)
            if r > max_r + 1e-12:
                final_radii[i] = max_r
                changed = True
        if not changed:
            break
            
    final_radii = np.maximum(final_radii, 0.0)
    final_sum = float(np.sum(final_radii))
    
    return final_centers, final_radii, final_sum
