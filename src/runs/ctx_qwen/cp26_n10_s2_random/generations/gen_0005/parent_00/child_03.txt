import numpy as np
from scipy.optimize import minimize, differential_evolution, linprog

N = 26

def solve_lp_radii(centers):
    """
    Given fixed centers, solves an LP to find radii that maximize sum of radii
    subject to non-overlap and boundary constraints.
    """
    n = centers.shape[0]
    # Boundary distances: max possible radius for each circle
    xb = np.minimum(centers[:, 0], 1.0 - centers[:, 0])
    yb = np.minimum(centers[:, 1], 1.0 - centers[:, 1])
    ub = np.minimum(xb, yb)
    
    # Pairwise Euclidean distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    # LP constraints: r_i + r_j <= dists[i,j] for all i < j
    i_idx, j_idx = np.triu_indices(n, 1)
    n_pairwise = len(i_idx)
    
    # Build A_ub and b_ub for linprog: A_ub @ r <= b_ub
    A_ub = np.zeros((n_pairwise + n, n))
    A_ub[:n_pairwise, i_idx] = 1.0
    A_ub[:n_pairwise, j_idx] = 1.0
    A_ub[n_pairwise:, np.arange(n)] = 1.0
    
    b_ub = np.concatenate([dists[i_idx, j_idx], ub])
    
    # Objective: maximize sum(r) => minimize -sum(r)
    c_obj = -np.ones(n)
    bounds = [(0.0, u) for u in ub]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return -res.fun, res.x
    except Exception:
        pass
        
    # Fallback direct computation if LP fails
    rp = 0.5 * np.min(dists, axis=1)
    radii = np.minimum(ub, rp)
    return np.sum(radii), radii

def de_objective(x_flat):
    """Objective for Differential Evolution: maximize sum of radii via LP."""
    centers = x_flat.reshape(N, 2)
    centers = np.clip(centers, 1e-4, 1.0 - 1e-4)
    s, _ = solve_lp_radii(centers)
    return -s

def obj_joint(v):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(v[2*N:])

def con_joint(v):
    """Computes all boundary and non-overlap constraints for SLSQP."""
    c = v[:2*N].reshape(N, 2)
    r = v[2*N:]
    
    con = []
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    con.append(c[:, 0] - r)
    con.append(1.0 - c[:, 0] - r)
    con.append(c[:, 1] - r)
    con.append(1.0 - c[:, 1] - r)
    
    # Overlap constraints: dist(i,j) >= r_i + r_j
    idx = np.triu_indices(N, 1)
    d = np.linalg.norm(c[idx[0]] - c[idx[1]], axis=1)
    con.append(d - (r[idx[0]] + r[idx[1]]))
    
    return np.concatenate(con)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)
    
    # Phase 1: Global search on centers using Differential Evolution
    bounds_de = [(0.08, 0.92)] * (2 * N)
    try:
        de_res = differential_evolution(
            de_objective, bounds_de, seed=42, 
            maxiter=120, popsize=15, tol=1e-7, 
            mutation=(0.5, 1.0), recombination=0.9, 
            polish=False
        )
        best_c = de_res.x.reshape(N, 2)
        best_s, best_r = solve_lp_radii(best_c)
    except Exception:
        best_c = rng.uniform(0.15, 0.85, (N, 2))
        best_s, best_r = solve_lp_radii(best_c)
        
    best_sum = best_s
    best_centers = best_c.copy()
    best_radii = best_r.copy()
    
    bounds_joint = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N
    cons_joint = {'type': 'ineq', 'fun': con_joint}
    
    # Phase 2: Joint SLSQP refinement
    v0 = np.concatenate([best_centers.flatten(), best_radii])
    try:
        res = minimize(
            obj_joint, v0, method='SLSQP', 
            bounds=bounds_joint, constraints=cons_joint, 
            options={'maxiter': 5000, 'ftol': 1e-14, 'disp': False}
        )
        if np.sum(res.x[2*N:]) > best_sum:
            best_sum = np.sum(res.x[2*N:])
            best_centers = res.x[:2*N].reshape(N, 2)
            best_radii = res.x[2*N:]
    except Exception:
        pass
        
    # Phase 3: Adaptive perturbation search to escape local minima
    curr_c = best_centers.copy()
    for step in range(60):
        sigma = 0.006 * (0.88 ** step)
        pert_c = curr_c + rng.normal(0, sigma, (N, 2))
        pert_c = np.clip(pert_c, 0.02, 0.98)
        
        # Fast LP evaluation for perturbed centers
        _, pert_r = solve_lp_radii(pert_c)
        v0_pert = np.concatenate([pert_c.flatten(), pert_r])
        
        try:
            res = minimize(
                obj_joint, v0_pert, method='SLSQP', 
                bounds=bounds_joint, constraints=cons_joint, 
                options={'maxiter': 3000, 'ftol': 1e-14, 'disp': False}
            )
            if np.sum(res.x[2*N:]) > best_sum:
                best_sum = np.sum(res.x[2*N:])
                best_centers = res.x[:2*N].reshape(N, 2)
                best_radii = res.x[2*N:]
                curr_c = best_centers
        except Exception:
            continue
            
    # Phase 4: Final LP on best centers to guarantee radii are optimal for positions
    final_s, final_r = solve_lp_radii(best_centers)
    best_radii = final_r
    best_sum = final_s
    
    # Phase 5: Strict numerical repair to satisfy validator tolerances
    for _ in range(100):
        changed = False
        # Fix overlaps
        for i in range(N):
            for j in range(i+1, N):
                d = np.linalg.norm(best_centers[i] - best_centers[j])
                req = best_radii[i] + best_radii[j]
                if d < req - 1e-12:
                    shrink = (req - d) / 2.0 + 1e-9
                    best_radii[i] -= shrink
                    best_radii[j] -= shrink
                    changed = True
        # Fix boundary violations
        for i in range(N):
            x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
            max_r = min(x, 1.0-x, y, 1.0-y)
            if r > max_r + 1e-12:
                best_radii[i] = max_r
                changed = True
        if not changed:
            break
            
    best_radii = np.maximum(best_radii, 0.0)
    return best_centers, best_radii, float(np.sum(best_radii))
