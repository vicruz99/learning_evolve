import numpy as np
from scipy.optimize import minimize, linprog

N = 26

# Precompute constant parts of LP constraint matrix for speed
# Constraints: r_i + r_j <= dist_ij  and  r_i <= boundary_dist_i
IDX_TRIU = np.triu_indices(N, 1)
M_PAIR = N * (N - 1) // 2
A_PAIR = np.zeros((M_PAIR, N))
A_PAIR[np.arange(M_PAIR), IDX_TRIU[0]] = 1.0
A_PAIR[np.arange(M_PAIR), IDX_TRIU[1]] = 1.0
A_BOUND = np.eye(N)
A_LP = np.vstack([A_PAIR, A_BOUND])

def compute_lp_radii(centers):
    """Computes optimal radii and their sum for fixed centers via Linear Programming."""
    ub = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    ub = np.maximum(ub, 0.0)
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    # RHS of constraints: pairwise distances and boundary limits
    b_lp = np.concatenate([dists[IDX_TRIU], ub])
    c_obj = -np.ones(N)
    bounds = [(0.0, None)] * N
    
    res = linprog(c_obj, A_ub=A_LP, b_ub=b_lp, bounds=bounds, method='highs')
    if res.success:
        return res.x, -res.fun
    return np.zeros(N), 0.0

def lp_objective(v):
    """Objective for center optimization: maximize sum of LP-optimal radii."""
    centers = np.clip(v.reshape(N, 2), 0.01, 0.99)
    _, s = compute_lp_radii(centers)
    return -s

def generate_initial_centers(n, method='hex', rng=None):
    """Generates diverse initial center configurations."""
    if rng is None:
        rng = np.random.default_rng()
    if method == 'hex':
        r = rng.uniform(0.08, 0.115)
        pts = []
        y = r
        row = 0
        while len(pts) < n:
            shift = r if row % 2 == 1 else 0.0
            x = r + shift
            while x + r <= 1.0 and len(pts) < n:
                pts.append([x, y])
                x += 2.0 * r
            y += r * np.sqrt(3)
            row += 1
        c = np.array(pts[:n])
        c += rng.normal(0, 0.004, c.shape)
        return np.clip(c, 0.02, 0.98)
    elif method == 'repulse':
        c = rng.uniform(0.1, 0.9, (n, 2))
        for _ in range(600):
            forces = np.zeros_like(c)
            for i in range(n):
                for j in range(i + 1, n):
                    d_vec = c[i] - c[j]
                    dist = np.linalg.norm(d_vec)
                    if dist < 0.22:
                        f = (0.22 - dist) / (dist + 1e-5)
                        forces[i] += d_vec * f
                        forces[j] -= d_vec * f
            c += 0.006 * forces
            c = np.clip(c, 0.05, 0.95)
        return c
    else:
        return rng.uniform(0.1, 0.9, (n, 2))

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Packs 26 circles in a unit square to maximize the sum of radii."""
    rng = np.random.default_rng(42)
    best_centers = None
    best_radii = None
    best_sum = 0.0
    
    inits = []
    for _ in range(12):
        inits.append(generate_initial_centers(N, 'hex', rng))
    for _ in range(8):
        inits.append(generate_initial_centers(N, 'repulse', rng))
        
    # Phase 1: Optimize centers using Powell on LP objective
    for c0 in inits:
        v0 = c0.flatten()
        try:
            res = minimize(lp_objective, v0, method='Powell',
                           options={'maxiter': 1500, 'ftol': 1e-12, 'xtol': 1e-12})
            c_opt = res.x.reshape(N, 2)
            r_opt, s_opt = compute_lp_radii(c_opt)
            if s_opt > best_sum:
                best_sum = s_opt
                best_centers = c_opt
                best_radii = r_opt
        except Exception:
            pass
            
    # Phase 2: Joint refinement with SLSQP
    v0_joint = np.concatenate([best_centers.flatten(), best_radii])
    bounds = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    
    def joint_obj(v):
        return -np.sum(v[2*N:])
        
    def joint_con(v):
        c = v[:2*N].reshape(N, 2)
        r = v[2*N:]
        cons = [c[:, 0] - r, 1.0 - c[:, 0] - r, c[:, 1] - r, 1.0 - c[:, 1] - r]
        d = np.linalg.norm(c[IDX_TRIU[0]] - c[IDX_TRIU[1]], axis=1)
        cons.append(d - (r[IDX_TRIU[0]] + r[IDX_TRIU[1]]))
        return np.concatenate(cons)
        
    cons_dict = {'type': 'ineq', 'fun': joint_con}
    
    try:
        res = minimize(joint_obj, v0_joint, method='SLSQP', bounds=bounds,
                       constraints=cons_dict, options={'maxiter': 12000, 'ftol': 1e-14})
        if np.sum(res.x[2*N:]) > best_sum:
            best_sum = np.sum(res.x[2*N:])
            best_centers = res.x[:2*N].reshape(N, 2)
            best_radii = res.x[2*N:]
    except Exception:
        pass
        
    # Phase 3: Perturbation & Local Search to escape local minima
    curr_c = best_centers.copy()
    for step in range(50):
        noise = 0.005 * (0.88 ** step)
        c_pert = curr_c + rng.normal(0, noise, curr_c.shape)
        c_pert = np.clip(c_pert, 0.01, 0.99)
        
        r_pert, s_pert = compute_lp_radii(c_pert)
        if s_pert > best_sum:
            best_sum = s_pert
            best_centers = c_pert
            best_radii = r_pert
            
            # Quick SLSQP refinement from perturbation
            v_p = np.concatenate([c_pert.flatten(), r_pert])
            try:
                res = minimize(joint_obj, v_p, method='SLSQP', bounds=bounds,
                               constraints=cons_dict, options={'maxiter': 4000, 'ftol': 1e-14})
                s_new = np.sum(res.x[2*N:])
                if s_new > best_sum:
                    best_sum = s_new
                    best_centers = res.x[:2*N].reshape(N, 2)
                    best_radii = res.x[2*N:]
            except Exception:
                pass
        curr_c = best_centers.copy()
        
    # Phase 4: Strict numerical repair to guarantee validation passes
    centers = best_centers
    radii = best_radii
    for _ in range(100):
        changed = False
        for i in range(N):
            for j in range(i + 1, N):
                d = np.linalg.norm(centers[i] - centers[j])
                if d < radii[i] + radii[j] - 1e-12:
                    shrink = (radii[i] + radii[j] - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        for i in range(N):
            x, y, r = centers[i, 0], centers[i, 1], radii[i]
            mx = min(x, 1.0 - x, y, 1.0 - y)
            if r > mx + 1e-12:
                radii[i] = mx
                changed = True
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    final_sum = float(np.sum(radii))
    return centers, radii, final_sum
