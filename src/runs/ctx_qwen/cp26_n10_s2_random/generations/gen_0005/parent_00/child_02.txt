import numpy as np
from scipy.optimize import minimize, linprog

N = 26
N_PAIRS = N*(N-1)//2

# Precompute the constant structure of the LP constraint matrix
# Constraints: r_i + r_j <= dist(i,j) for all i<j, and r_i <= boundary_dist(i)
A_ub = np.zeros((N_PAIRS + N, N))
idx = 0
for i in range(N):
    for j in range(i+1, N):
        A_ub[idx, i] = 1.0
        A_ub[idx, j] = 1.0
        idx += 1
np.fill_diagonal(A_ub[N_PAIRS:], 1.0)

# Precompute indices for fast upper-triangle extraction
TRIU_INDICES = np.triu_indices(N, 1)

def get_optimal_radii_lp(centers):
    """Solves LP to find maximum sum of radii for fixed centers."""
    # Distance to boundaries
    b = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                   np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    b = np.maximum(b, 0.0)
    
    # Pairwise distances
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    # Vectorized RHS construction
    b_ub = np.zeros(N_PAIRS + N)
    b_ub[:N_PAIRS] = dists[TRIU_INDICES]
    b_ub[N_PAIRS:] = b
    
    # LP: minimize -sum(r) s.t. A_ub @ r <= b_ub, r >= 0
    res = linprog(-np.ones(N), A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*N, method='highs')
    if res.success:
        return res.x, np.sum(res.x)
    return np.zeros(N), 0.0

def obj(c_flat):
    """Objective for center optimization: maximize LP-derived sum of radii."""
    # Clip to avoid degenerate distances or boundary violations during search
    C = np.clip(c_flat.reshape(N, 2), 1e-4, 1.0 - 1e-4)
    _, s = get_optimal_radii_lp(C)
    return -s

def generate_starts(rng):
    """Generates diverse initial center configurations."""
    starts = []
    
    # Hexagonal lattice patterns with varying row structures and scales
    patterns = [
        [5,6,5,6,4], [6,5,6,5,4], [5,5,5,5,6], [4,6,6,6,4], 
        [6,4,6,5,5], [5,4,6,6,5], [6,6,5,5,4], [5,5,6,5,5]
    ]
    for pat in patterns:
        for r in [0.092, 0.098, 0.102]:
            c = []
            y = r
            for r_idx, cnt in enumerate(pat):
                shift = r if r_idx % 2 == 1 else 0.0
                x = r + shift
                for _ in range(cnt):
                    c.append([x, y])
                    x += 2.0 * r
                y += r * np.sqrt(3)
            c = np.array(c[:N])
            c += rng.normal(0, 0.003, c.shape)
            starts.append(c.flatten())
            
    # Force-directed repulsion starts
    for _ in range(12):
        c = rng.uniform(0.1, 0.9, (N, 2))
        for _ in range(80):
            for i in range(N):
                for j in range(i+1, N):
                    d = c[i] - c[j]
                    dist = np.linalg.norm(d)
                    if dist < 0.16:
                        push = (0.16 - dist) * 0.4
                        c[i] += d / dist * push
                        c[j] -= d / dist * push
        c = np.clip(c, 0.05, 0.95)
        starts.append(c.flatten())
        
    # Corner-anchored starts to encourage boundary utilization
    for _ in range(8):
        c = rng.uniform(0.2, 0.8, (N, 2))
        c[:4] = [[0.06, 0.06], [0.94, 0.06], [0.06, 0.94], [0.94, 0.94]]
        starts.append(c.flatten())
        
    return starts

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Packs 26 circles in a unit square to maximize the sum of radii."""
    rng = np.random.default_rng(42)
    starts = generate_starts(rng)
    best_c = None
    best_s = -1.0
    
    # Phase 1: Multi-start local optimization
    for s in starts:
        try:
            res = minimize(obj, s, method='Powell',
                           options={'maxiter': 3000, 'ftol': 1e-10, 'xtol': 1e-10})
            if -res.fun > best_s:
                best_s = -res.fun
                best_c = res.x.reshape(N, 2)
        except Exception:
            pass
            
    if best_c is None:
        best_c = starts[0].reshape(N, 2)
        
    # Phase 2: Adaptive perturbation to escape local minima
    curr_c = best_c.copy()
    for step in range(30):
        noise = 0.006 * (0.82 ** step)
        pert = curr_c + rng.normal(0, noise, curr_c.shape)
        pert = np.clip(pert, 0.001, 0.999)
        try:
            res = minimize(obj, pert.flatten(), method='Powell',
                           options={'maxiter': 2500, 'ftol': 1e-10, 'xtol': 1e-10})
            if -res.fun > best_s:
                best_s = -res.fun
                best_c = res.x.reshape(N, 2)
                curr_c = best_c.copy()
        except Exception:
            pass
            
    # Final LP evaluation on best centers
    final_centers = np.clip(best_c, 1e-4, 1.0 - 1e-4)
    radii, final_sum = get_optimal_radii_lp(final_centers)
    
    # Phase 3: Strict numerical repair to guarantee validator tolerance
    centers = final_centers
    for _ in range(50):
        changed = False
        # Fix pairwise overlaps
        for i in range(N):
            for j in range(i+1, N):
                d = np.linalg.norm(centers[i] - centers[j])
                req = radii[i] + radii[j]
                if d < req - 1e-12:
                    shrink = (req - d) / 2.0 + 1e-9
                    radii[i] -= shrink
                    radii[j] -= shrink
                    changed = True
        # Fix boundary violations
        for i in range(N):
            x, y = centers[i]
            max_r = min(x, 1.0-x, y, 1.0-y)
            if radii[i] > max_r + 1e-12:
                radii[i] = max_r
                changed = True
        if not changed:
            break
            
    radii = np.maximum(radii, 0.0)
    final_sum = float(np.sum(radii))
    return centers, radii, final_sum
