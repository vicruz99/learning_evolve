import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def get_lp_radii(centers):
    """Computes the exact maximum feasible radii for a given set of centers using LP."""
    n = centers.shape[0]
    c = np.clip(centers, 1e-8, 1.0 - 1e-8)
    
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    ub = np.minimum(np.minimum(c[:, 0], 1.0 - c[:, 0]), 
                    np.minimum(c[:, 1], 1.0 - c[:, 1]))
    ub = np.maximum(ub, 0.0)
    
    # Pairwise constraints: r_i + r_j <= dist(i,j)
    diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    A = np.zeros((n*(n-1)//2, n))
    b = np.zeros(n*(n-1)//2)
    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = dists[i, j]
            idx += 1
            
    # Maximize sum(r) <=> Minimize -sum(r)
    res = linprog(-np.ones(n), A_ub=A, b_ub=b, bounds=[(0.0, u) for u in ub], method='highs')
    if res.success:
        return res.x
    return np.zeros(n)
