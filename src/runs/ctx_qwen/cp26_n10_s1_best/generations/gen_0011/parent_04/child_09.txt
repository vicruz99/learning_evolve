2. **Aggressive Topology Search via Deflation & Perturbation**: Optimal packings are defined by their contact graphs. Rigid contact networks trap gradient-based optimizers. We iteratively shrink the radii of a random subset of circles, perturb their centers, and re-optimize. 3. **Hybrid SLSQP + LP Snapping**: Use SLSQP for joint center-radius optimization with polynomial constraints (for smooth gradients), followed immediately by a fast Linear Programming (LP) solve to optimally assign radii for the fixed centers. This decouples the non-convex geometric constraints from the linear radius assignment, significantly stabilizing convergence. import numpy as np
from scipy.optimize import minimize, linprog
import warnings
warnings.filterwarnings('ignore')

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Returns all inequality constraints >= 0 (vectorized, polynomial form for stability)."""
    xs, ys, rs = x[0::3], x[1::3], x[2::3]
    # Boundary constraints
    c = np.concatenate([xs - rs, 1.0 - xs - rs, ys - rs, 1.0 - ys - rs])
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    c = np.concatenate([c, dx[TRIL_IDX]**2 + dy[TRIL_IDX]**2 - dr[TRIL_IDX]**2])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-7, 0.5]."""
    return [(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)] * N

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist)
            idx += 1
            
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=[(0.0, None)] * n, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-8)
    except Exception:
        pass
    return np.full(n, 0.05)

def project(x):
    """Project variables to strictly satisfy bounds."""
    for i in range(N):
        r = max(1e-6, x[3 * i + 2])
        x[3 * i] = np.clip(x[3 * i], r, 1.0 - r)
        x[3 * i + 1] = np.clip(x[3 * i + 1], r, 1.0 - r)
        x[3 * i + 2] = r
    return x

def make_hex_init(r0, ang, scale=1.0):
    """Generates a hexagonal lattice initialization with optional rotation."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 5:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0 and len(pts) < N + 5:
            pts.append([x * scale, y * scale])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
    pts = np.array(pts[:N + 5])
    if ang != 0.0:
        c, s = np.cos(ang), np.sin(ang)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    while len(pts) < N:
        pts = np.vstack([pts, [0.5, 0.5]])
    return pts[:N]

def make_force_init(seed):
    """Force-directed layout to spread points evenly and push to boundaries."""
    np.random.seed(seed)
    pts = np.random.uniform(0.15, 0.85, (N, 2))
    rs = np.full(N, 0.07)
    for step in range(250):
        f = np.zeros_like(pts)
        for i in range(N):
            for j in range(i + 1, N):
                dx = pts[j] - pts[i]
                d = np.hypot(dx[0], dx[1])
                target = rs[i] + rs[j] + 0.03
                if d < target and d > 1e-5:
                    rep = 0.015 / (d**2 + 0.001)
                    f[i] -= dx * rep / d
                    f[j] += dx * rep / d
            for dim in range(2):
                if pts[i, dim] < rs[i] + 0.02: f[i, dim] += 0.06
                elif pts[i, dim] > 1.0 - rs[i] - 0.02: f[i, dim] -= 0.06
        lr = 0.03 * (1.0 - step / 250.0)
        pts += f * lr
        pts = np.clip(pts, 0.02, 0.98)
    return pts

def optimize_start(pts, bounds, cons):
    """Run SLSQP optimization from a given starting configuration."""
    x0 = np.zeros(3 * N)
    x0[0::3] = pts[:, 0]
    x0[1::3] = pts[:, 1]
    r_lp = solve_lp_radii(pts)
    x0[2::3] = r_lp * 0.995
    x0 = project(x0)
    
    try:
        res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 25000, 'ftol': 1e-13, 'disp': False})
        if not np.isnan(res.fun):
            c_tmp = np.column_stack((res.x[0::3], res.x[1::3]))
            r_snap = solve_lp_radii(c_tmp)
            if r_snap is not None:
                return np.sum(r_snap), np.column_stack((c_tmp[:,0], c_tmp[:,1], r_snap))
    except Exception:
        pass
    return -1.0, None

def refine_config(centers, radii, bounds, cons, steps=30):
    """Escape local minima via deflation, perturbation, and re-optimization."""
    best_c = centers.copy()
    best_r = radii.copy()
    best_sum = np.sum(radii)
    
    for step in range(steps):
        noise_scale = 0.004 * (1.0 - step / steps)
        c_p = best_c + np.random.normal(0, noise_scale, best_c.shape)
        c_p = np.clip(c_p, 0.01, 0.99)
        
        # Deflate a random subset to break rigid contact networks
        subset = np.random.choice(N, size=max(3, N // 3), replace=False)
        r_p = best_r.copy()
        r_p[subset] *= 0.85
        
        x0 = np.zeros(3 * N)
        x0[0::3] = c_p[:, 0]
        x0[1::3] = c_p[:, 1]
        x0[2::3] = r_p
        x0 = project(x0)
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 18000, 'ftol': 1e-13, 'disp': False})
            if not np.isnan(res.fun):
                c_new = np.column_stack((res.x[0::3], res.x[1::3]))
                r_new = solve_lp_radii(c_new)
                if r_new is not None:
                    s_new = np.sum(r_new)
                    if s_new > best_sum:
                        best_sum = s_new
                        best_c = c_new
                        best_r = r_new
        except Exception:
            pass
    return best_c, best_r, best_sum

def run_packing():
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -1.0
    best_data = None
    
    # Phase 1: Generate Diverse Initial Configurations
    inits = []
    for r0 in [0.08, 0.085, 0.09, 0.095, 0.10, 0.105]:
        for ang in np.linspace(-0.4, 0.4, 15):
            inits.append(make_hex_init(r0, ang))
    for s in range(15):
        inits.append(make_force_init(s))
    for s in range(10):
        np.random.seed(s + 1000)
        inits.append(np.random.uniform(0.1, 0.9, (N, 2)))
        
    # Phase 2: Multi-start Optimization
    for pts in inits:
        s, data = optimize_start(pts, bounds, cons)
        if s > best_sum:
            best_sum = s
            best_data = data
            
    # Phase 3: Aggressive Refinement & Topology Search
    if best_data is not None:
        c_best, r_best = best_data[:,:2], best_data[:,2]
        c_best, r_best, best_sum = refine_config(c_best, r_best, bounds, cons, steps=40)
        
        # Rotational search on the best configuration to exploit boundary asymmetry
        for ang_deg in np.linspace(-15, 15, 7):
            ang = np.deg2rad(ang_deg)
            c_rot, s_rot = np.cos(ang), np.sin(ang)
            c_new = (c_best - 0.5) @ np.array([[c_rot, -s_rot], [s_rot, c_rot]]) + 0.5
            c_new = np.clip(c_new, 0.01, 0.99)
            r_new = solve_lp_radii(c_new)
            if r_new is not None and np.sum(r_new) > best_sum:
                best_sum = np.sum(r_new)
                c_best, r_best = c_new, r_new
                c_best, r_best, best_sum = refine_config(c_best, r_best, bounds, cons, steps=20)

    # Fallback initialization
    if best_data is None:
        best_data = np.zeros((N, 3))
        best_data[:, 0] = np.tile(np.linspace(0.1, 0.9, 5), 6)[:N]
        best_data[:, 1] = np.repeat(np.linspace(0.1, 0.9, 6), 5)[:N]
        best_data[:, 2] = 0.06
        best_sum = np.sum(best_data[:, 2])
    else:
        best_data = np.column_stack((c_best, r_best))

    centers = best_data[:, :2]
    radii = best_data[:, 2].copy()
    
    # Final LP squeeze to maximize radii for the best centers found
    final_r = solve_lp_radii(centers)
    if np.sum(final_r) > np.sum(radii) - 1e-7:
        radii = final_r.copy()

    # Final strict validation repair against 1e-12 tolerance
    for _ in range(100):
        valid = True
        for i in range(N):
            if radii[i] < 0 or centers[i, 0] < radii[i] - 1e-9 or centers[i, 0] > 1.0 - radii[i] + 1e-9 or \
               centers[i, 1] < radii[i] - 1e-9 or centers[i, 1] > 1.0 - radii[i] + 1e-9:
                valid = False; break
        if valid:
            for i in range(N):
                for j in range(i + 1, N):
                    d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                    if d < radii[i] + radii[j] - 1e-9:
                        valid = False; break
                if not valid: break
        if valid: break
            
        # Minimal shrinkage to recover strict feasibility
        radii *= 0.9995
        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)

    return centers, radii, float(np.sum(radii))
