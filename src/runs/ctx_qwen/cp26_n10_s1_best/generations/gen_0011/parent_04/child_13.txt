import numpy as np
from scipy.optimize import minimize, linprog
import warnings
warnings.filterwarnings('ignore')

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Returns all inequality constraints >= 0 (vectorized)."""
    xs, ys, rs = x[0::3], x[1::3], x[2::3]
    # Boundary constraints
    c = np.concatenate([xs - rs, 1.0 - xs - rs, ys - rs, 1.0 - ys - rs])
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    c = np.concatenate([c, dx[TRIL_IDX]**2 + dy[TRIL_IDX]**2 - dr[TRIL_IDX]**2])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-7, 0.5]."""
    return [(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)] * N

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist)
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-8)
    except Exception:
        pass
    return np.full(n, 0.05)

def project(x):
    """Project variables to strictly satisfy bounds."""
    for i in range(N):
        r = max(1e-6, x[3 * i + 2])
        x[3 * i] = np.clip(x[3 * i], r, 1.0 - r)
        x[3 * i + 1] = np.clip(x[3 * i + 1], r, 1.0 - r)
        x[3 * i + 2] = r
    return x

def make_hex_init(r0, ang):
    """Generates a hexagonal lattice initialization with optional rotation."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0 and len(pts) < N + 10:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
    pts = np.array(pts[:N + 10])
    if ang != 0.0:
        c, s = np.cos(ang), np.sin(ang)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    while len(pts) < N:
        pts = np.vstack([pts, [np.random.uniform(0.1, 0.9), np.random.uniform(0.1, 0.9)]])
    return pts[:N]

def make_corner_edge_init():
    """Generates an initialization focused on corners and edges."""
    pts = np.zeros((N, 2))
    r = 0.09
    pts[0] = [r, r]
    pts[1] = [1-r, r]
    pts[2] = [r, 1-r]
    pts[3] = [1-r, 1-r]
    pts[4] = [0.5, r]
    pts[5] = [0.5, 1-r]
    pts[6] = [r, 0.5]
    pts[7] = [1-r, 0.5]
    pts[8] = [0.5, 0.5]
    idx = 9
    y = r + 0.05
    row = 0
    while idx < N and y + r <= 1.0:
        x = r + 0.05 if row % 2 == 0 else r + 0.18

            pts[idx] = [x, y]
            idx += 1
            x += 0.22
        y += 0.15
        row += 1
    for i in range(idx, N):
        pts[i] = np.random.uniform(0.1, 0.9, 2)
    pts += np.random.uniform(-0.005, 0.005, pts.shape)
    return pts

def make_force_init(seed):
    """Force-directed layout to spread points evenly and push to boundaries."""
    np.random.seed(seed)
    pts = np.random.uniform(0.15, 0.85, (N, 2))
    rs = np.full(N, 0.07)
    for step in range(250):
        f = np.zeros_like(pts)
        lr = 0.03 * (1.0 - step / 250.0)
        for i in range(N):
            for j in range(i + 1, N):
                dx = pts[j] - pts[i]
                d = np.hypot(dx[0], dx[1])
                if d < 0.3 and d > 1e-5:
                    rep = 0.015 / (d**2 + 0.001)
                    f[i] -= dx * rep / d
                    f[j] += dx * rep / d
            for dim in range(2):
                if pts[i, dim] < rs[i] + 0.02: f[i, dim] += 0.06
                elif pts[i, dim] > 1.0 - rs[i] - 0.02: f[i, dim] -= 0.06
        pts += f * lr
        pts = np.clip(pts, 0.02, 0.98)
    return pts

def local_search_lp(centers, radii, max_iter=4000):
    """Stochastic local search optimizing centers via exact LP radius evaluation."""
    best_sum = np.sum(radii)
    best_centers = centers.copy()
    step_size = 0.025
    for _ in range(max_iter):
        k = 1 if np.random.rand() < 0.6 else 2
        idxs = np.random.choice(N, size=k, replace=False)
        new_centers = best_centers.copy()
        moves = np.random.uniform(-step_size, step_size, (k, 2))
        new_centers[idxs] += moves
        new_centers = np.clip(new_centers, 0.001, 0.999)
        new_radii = solve_lp_radii(new_centers)
        s = np.sum(new_radii)
        if s > best_sum + 1e-9:
            best_sum = s
            best_centers = new_centers.copy()
            step_size *= 1.03
        else:
            step_size *= 0.97
        step_size = max(step_size, 1e-4)
    return best_centers, solve_lp_radii(best_centers), best_sum
