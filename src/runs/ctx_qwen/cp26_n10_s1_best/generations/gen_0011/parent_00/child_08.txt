import numpy as np
from scipy.optimize import minimize, linprog
import warnings
warnings.filterwarnings('ignore')

N = 26

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Returns all inequality constraints >= 0 (vectorized)."""
    xs, ys, rs = x[0::3], x[1::3], x[2::3]
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    c = np.concatenate([
        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs
    ])
    
    # Overlap constraints: dist >= r_i + r_j
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    i, j = np.tril_indices(N, -1)
    c = np.concatenate([c, np.hypot(dx[i, j], dy[i, j]) - dr[i, j]])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-7, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)])
    return b

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist)
            idx += 1
            
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=[(0.0, None)] * n, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-7)
    except Exception:
        pass
    return np.full(n, 0.05)

def jam_packing(centers, steps=200):
    """Vectorized growth-separation algorithm to find good center configurations."""
    pts = centers.copy()
    n = pts.shape[0]
    rs = np.full(n, 0.001)
    ixs, jxs = np.tril_indices(n, -1)
    
    for _ in range(steps):
        # Compute clearances to walls
        cw = np.minimum(
            np.minimum(pts[:, 0], 1.0 - pts[:, 0]),
            np.minimum(pts[:, 1], 1.0 - pts[:, 1])
        )
        clear = cw - rs
        
        # Compute clearances to neighbors
        dx = pts[ixs, 0] - pts[jxs, 0]
        dy = pts[ixs, 1] - pts[jxs, 1]
        dists = np.hypot(dx, dy)
        pc = dists - rs[ixs] - rs[jxs]
        if len(pc) > 0:
            clear = np.minimum(clear, pc.min())
            
        if clear <= 0.0:
            clear = 0.0
            
        # Grow radii
        rs += clear * 0.4
        
        # Compute repulsive forces from overlaps and walls
        f = np.zeros_like(pts)
        
        # Wall forces
        for i in range(n):
            if pts[i, 0] < rs[i]: f[i, 0] += 0.15 * (rs[i] - pts[i, 0])
            if pts[i, 0] > 1.0 - rs[i]: f[i, 0] -= 0.15 * (pts[i, 0] - (1.0 - rs[i]))
            if pts[i, 1] < rs[i]: f[i, 1] += 0.15 * (rs[i] - pts[i, 1])
            if pts[i, 1] > 1.0 - rs[i]: f[i, 1] -= 0.15 * (pts[i, 1] - (1.0 - rs[i]))
            
        # Overlap forces
        overlaps = np.maximum(rs[ixs] + rs[jxs] - dists, 0.0)
        if len(overlaps) > 0:
            inv_d = 1.0 / np.where(dists > 1e-8, dists, 1e-8)
            fx = dx * inv_d * overlaps
            fy = dy * inv_d * overlaps
            np.add.at(f, ixs, np.column_stack([fx, fy]))
            np.add.at(f, jxs, -np.column_stack([fx, fy]))
            
        pts += f * 0.04
        pts = np.clip(pts, 1e-5, 1.0 - 1e-5)
        rs = np.maximum(rs, 1e-5)
        
    return pts, rs

def make_init(strategy, seed):
    """Generate diverse initial center configurations."""
    np.random.seed(seed)
    n = N
    pts = np.zeros((n, 2))
    
    if strategy == 'hex':
        r = 0.09
        y = r
        row = 0
        idx = 0
        while idx < n and y + r <= 1.0:
            x = r if row % 2 == 0 else 2.0 * r
            while idx < n and x + r <= 1.0:
                pts[idx] = [x, y]
                idx += 1
                x += 2.0 * r
            y += np.sqrt(3.0) * r
            row += 1
        while idx < n:
            pts[idx] = np.random.uniform(0.2, 0.8, 2)
            idx += 1
        # Random rotation
        ang = np.random.uniform(-0.25, 0.25)
        c, s = np.cos(ang), np.sin(ang)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
    elif strategy == 'grid':
        idx = 0
        for r_idx in range(6):
            for c_idx in range(5):
                if idx < n:
                    pts[idx] = [0.08 + c_idx * 0.18, 0.08 + r_idx * 0.16]
                    idx += 1
        pts += np.random.uniform(-0.01, 0.01, pts.shape)
    elif strategy == 'rand':
        pts = np.random.uniform(0.15, 0.85, (n, 2))
        
    return np.clip(pts, 0.02, 0.98)

def project_to_feasible(x):
    """Ensure variables strictly satisfy bounds."""
    x = x.copy()
    for i in range(N):
        r = max(1e-6, x[3 * i + 2])
        x[3 * i] = np.clip(x[3 * i], r, 1.0 - r)
        x[3 * i + 1] = np.clip(x[3 * i + 1], r, 1.0 - r)
        x[3 * i + 2] = r
    return x

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    bounds_obj = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -1.0
    best_x = None
    
    # Phase 1: Multi-start with Jam Packing -> LP -> SLSQP
    starts = []
    for s in range(12): starts.append(('hex', s))
    for s in range(8): starts.append(('grid', s))
    for s in range(10): starts.append(('rand', s))
    
    for strat, seed in starts:
        pts = make_init(strat, seed)
        pts_j, _ = jam_packing(pts, steps=150)
        
        r_lp = solve_lp_radii(pts_j)
        
        x0 = np.zeros(3 * N)
        x0[0::3] = pts_j[:, 0]
        x0[1::3] = pts_j[:, 1]
        x0[2::3] = r_lp
        x0 = project_to_feasible(x0)
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds_obj, constraints=cons,
                           options={'maxiter': 15000, 'ftol': 1e-13, 'disp': False})
            if not np.isnan(res.fun):
                curr_sum = -res.fun
                c_opt = np.column_stack((res.x[0::3], res.x[1::3]))
                r_opt = solve_lp_radii(c_opt)
                if np.sum(r_opt) > curr_sum:
                    curr_sum = np.sum(r_opt)
                    res.x[2::3] = r_opt
                    
                if curr_sum > best_sum and np.min(constraints(res.x)) >= -1e-6:
                    best_sum = curr_sum
                    best_x = res.x.copy()
        except Exception:
            pass

    # Phase 2: Topology Search & Refinement
    if best_x is not None:
        for cyc in range(60):
            x_p = best_x.copy()
            noise = 0.0025 * (0.93 ** cyc)
            x_p += np.random.normal(0, noise, 3 * N)
            
            # Deflate random subset to break rigid contacts
            k = np.random.randint(3, 7)
            subset = np.random.choice(N, size=k, replace=False)
            x_p[subset * 3 + 2] *= 0.85
            x_p = project_to_feasible(x_p)
            
            try:
                res = minimize(objective, x_p, method='SLSQP', bounds=bounds_obj, constraints=cons,
                               options={'maxiter': 8000, 'ftol': 1e-13, 'disp': False})
                if not np.isnan(res.fun):
                    c_tmp = np.column_stack((res.x[0::3], res.x[1::3]))
                    r_tmp = solve_lp_radii(c_tmp)
                    s_val = np.sum(r_tmp)
                    if s_val > best_sum and np.min(constraints(res.x)) >= -1e-6:
                        best_sum = s_val
                        best_x = res.x.copy()
                        best_x[2::3] = r_tmp
            except Exception:
                pass
                
            # Swap heuristic: exchange positions of two circles and re-optimize
            i, j = np.random.choice(N, 2, replace=False)
            x_swap = best_x.copy()
            x_swap[3*i], x_swap[3*j] = x_swap[3*j], x_swap[3*i]
            x_swap[3*i+1], x_swap[3*j+1] = x_swap[3*j+1], x_swap[3*i+1]
            x_swap[3*i:3*i+2] += np.random.normal(0, 0.0015, 2)
            x_swap[3*j:3*j+2] += np.random.normal(0, 0.0015, 2)
            x_swap = project_to_feasible(x_swap)
            
            try:
                res = minimize(objective, x_swap, method='SLSQP', bounds=bounds_obj, constraints=cons,
                               options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
                if not np.isnan(res.fun):
                    c_tmp = np.column_stack((res.x[0::3], res.x[1::3]))
                    r_tmp = solve_lp_radii(c_tmp)
                    s_val = np.sum(r_tmp)
                    if s_val > best_sum and np.min(constraints(res.x)) >= -1e-6:
                        best_sum = s_val
                        best_x = res.x.copy()
                        best_x[2::3] = r_tmp
            except Exception:
                pass

    # Fallback
    if best_x is None:
        best_x = np.zeros(3 * N)
        best_x[0::3] = np.tile(np.linspace(0.1, 0.9, 5), 6)[:N]
        best_x[1::3] = np.repeat(np.linspace(0.1, 0.9, 6), 5)[:N]
        best_x[2::3] = 0.06
        best_sum = np.sum(best_x[2::3])

    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3].copy()

    # Final strict validity repair against 1e-12 tolerance
    for _ in range(100):
        valid = True
        for i in range(N):
            if radii[i] < 0 or centers[i, 0] < radii[i] - 1e-9 or centers[i, 0] > 1.0 - radii[i] + 1e-9 or \
               centers[i, 1] < radii[i] - 1e-9 or centers[i, 1] > 1.0 - radii[i] + 1e-9:
                valid = False
                break
        if valid:
            for i in range(N):
                for j in range(i + 1, N):
                    d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                    if d < radii[i] + radii[j] - 1e-9:
                        valid = False
                        break
                if not valid:
                    break
        if valid:
            break
            
        radii *= 0.9999
        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)

    return centers, radii, float(np.sum(radii))
