3. I need to combine multiple strategies: LP-based position optimization, coordinate descent fine-tuning, and aggressive topology exploration

My approach:
1. Use LP-based position optimization: for fixed contact graph, positions are determined by solving distance constraints - this can be formulated as a small LP
2. Implement a "growing circles" simulated annealing approach: start with small circles, grow them, repel when they overlap
3. Use LP to optimally compute radii for any given center configuration (this is a convex problem)
4. Coordinate descent on centers: for each circle, try moving it to positions that maximize LP-computed radii sum
5. Systematically try different topologies by deflating and re-optimizing
6. Multiple diverse initializations: hexagonal lattices with many rotations, force-directed layouts, corner-focused layouts
7. Post-processing refinement using coordinate descent with multiple step sizes

import numpy as np
from scipy.optimize import minimize, linprog
import warnings
warnings.filterwarnings('ignore')

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def objective(x):
    return -np.sum(x[2::3])

def constraints(x):
    xs, ys, rs = x[0::3], x[1::3], x[2::3]
    c = np.concatenate([xs - rs, 1.0 - xs - rs, ys - rs, 1.0 - ys - rs])
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    c = np.concatenate([c, dx[TRIL_IDX]**2 + dy[TRIL_IDX]**2 - dr[TRIL_IDX]**2])
    return c

def get_bounds():
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)])
    return b

def solve_lp_radii(centers):
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
    
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist)
            idx += 1
    
    bounds_lp = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_lp, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-7)
    except Exception:
        pass
    return np.full(n, 0.05)

def project_to_feasible(x):
    x = x.copy()
    for i in range(N):
        r = max(1e-6, x[3 * i + 2])
        x[3 * i] = np.clip(x[3 * i], r, 1.0 - r)
        x[3 * i + 1] = np.clip(x[3 * i + 1], r, 1.0 - r)
        x[3 * i + 2] = r
    return x

def make_hex_init(r0, angle):
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0 and len(pts) < N + 10:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
    
    pts = np.array(pts[:N + 10])
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
    
    mask = (pts[:, 0] >= 0.01) & (pts[:, 0] <= 0.99) & (pts[:, 1] >= 0.01) & (pts[:, 1] <= 0.99)
    pts = pts[mask]
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.1, 0.9, (1, 2))])
    return pts[:N]

def force_directed_init(seed, steps=600):
    np.random.seed(seed)
    pts = np.random.uniform(0.1, 0.9, (N, 2))
    for step in range(steps):
        f = np.zeros_like(pts)
        lr = 0.015 * max(0.05, 1.0 - step / steps)
        for i in range(N):
            for j in range(i + 1, N):
                dx = pts[j] - pts[i]
                d = np.hypot(dx[0], dx[1])
                if d < 0.4 and d > 1e-6:
                    rep = 0.006 / (d**2 + 0.002)
                    f[i] -= dx * rep / d
                    f[j] += dx * rep / d
            for dim in range(2):
                if pts[i, dim] < 0.08:
                    f[i, dim] += 0.05
                elif pts[i, dim] > 0.92:
                    f[i, dim] -= 0.05
        pts += f * lr
        pts = np.clip(pts, 0.01, 0.99)
    return pts

def corner_init(seed):
    np.random.seed(seed)
    pts = np.zeros((N, 2))
    r = 0.085
    pts[0] = [r + 0.02, r + 0.02]
    pts[1] = [1 - r - 0.02, r + 0.02]
    pts[2] = [r + 0.02, 1 - r - 0.02]
    pts[3] = [1 - r - 0.02, 1 - r - 0.02]
    pts[4] = [0.5, r + 0.01]
    pts[5] = [0.5, 1 - r - 0.01]
    pts[6] = [r + 0.01, 0.5]
    pts[7] = [1 - r - 0.01, 0.5]
    idx = 8
    
    y = r + 0.05
    row = 0
    while idx < N and y + r <= 1.0 - 0.02:
        x_start = r + 0.05 if row % 2 == 0 else r + 0.15
        x = x_start
        while x + r <= 1.0 - 0.02 and idx < N:
            pts[idx] = [x, y]
            idx += 1
            x += 0.2
        y += 0.14
        row += 1
    
    for i in range(idx, N):
        pts[i] = np.random.uniform(0.15, 0.85, 2)
    
    pts += np.random.uniform(-0.005, 0.005, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def grow_circles_init(seed):
    np.random.seed(seed)
    pts = np.random.uniform(0.15, 0.85, (N, 2))
    rs = np.full(N, 0.005)
    
    for grow_step in range(800):
        growth_rate = 0.0003 * max(0.1, 1.0 - grow_step / 800)
        f = np.zeros_like(pts)
        
        for i in range(N):
            rs[i] = min(rs[i] + growth_rate, 
                       min(pts[i, 0], 1.0 - pts[i, 0], pts[i, 1], 1.0 - pts[i, 1]))
        
        for i in range(N):
            for j in range(i + 1, N):
                dx = pts[j] - pts[i]
                d = np.hypot(dx[0], dx[1])
                needed = rs[i] + rs[j]
                if d < needed and d > 1e-6:
                    repel = 0.01 * (needed - d) / d
                    f[i] -= dx * repel
                    f[j] += dx * repel
        
        for i in range(N):
            for dim in range(2):
                if pts[i, dim] < rs[i] + 0.005:
                    f[i, dim] += 0.02
                elif pts[i, dim] > 1.0 - rs[i] - 0.005:
                    f[i, dim] -= 0.02
        
        pts += f * 0.3
        pts = np.clip(pts, 0.005, 0.995)
        for i in range(N):
            rs[i] = min(rs[i], pts[i, 0], 1.0 - pts[i, 0], pts[i, 1], 1.0 - pts[i, 1])
    
    return pts

def coordinate_descent_refine(centers, radii, n_passes=5):
    for p in range(n_passes):
        improved = True
        passes = 0
        while improved and passes < 20:
            improved = False
            passes += 1
            step_sizes = [0.008, 0.004, 0.002, 0.001, 0.0005]
            for step in step_sizes:
                for i in range(N):
                    best_sum = np.sum(radii)
                    best_dx, best_dy = 0.0, 0.0
                    
                    for dx in [-step, 0.0, step]:
                        for dy in [-step, 0.0, step]:
                            if dx == 0.0 and dy == 0.0:
                                continue
                            new_x = np.clip(centers[i, 0] + dx, radii[i] + 1e-8, 1.0 - radii[i] - 1e-8)
                            new_y = np.clip(centers[i, 1] + dy, radii[i] + 1e-8, 1.0 - radii[i] - 1e-8)
                            if new_x != centers[i, 0] or new_y != centers[i, 1]:
                                new_centers = centers.copy()
                                new_centers[i] = [new_x, new_y]
                                r_new = solve_lp_radii(new_centers)
                                s_new = np.sum(r_new)
                                if s_new > best_sum + 1e-12:
                                    best_sum = s_new
                                    best_dx, best_dy = dx, dy
                                    improved = True
                    
                    if best_dx != 0.0 or best_dy != 0.0:
                        centers[i, 0] = np.clip(centers[i, 0] + best_dx, radii[i], 1.0 - radii[i])
                        centers[i, 1] = np.clip(centers[i, 1] + best_dy, radii[i], 1.0 - radii[i])
                        radii = solve_lp_radii(centers)
                        improved = True
                        break
                if improved:
                    break
            if not improved:
                break
    return centers, radii

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    bounds_obj = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    inits = []
    
    for r0 in [0.082, 0.087, 0.092, 0.097, 0.102, 0.107]:
        for ang in np.linspace(-0.45, 0.45, 21):
            inits.append(make_hex_init(r0, ang))
    
    for s in range(30):
        inits.append(force_directed_init(s, steps=600))
    
    for s in range(15):
        inits.append(corner_init(s))
    
    for s in range(10):
        inits.append(grow_circles_init(s))
    
    for pts in inits:
        r_lp = solve_lp_radii(pts)
        x0 = np.zeros(3 * N)
        x0[0::3] = pts[:, 0]
        x0[1::3] = pts[:, 1]
        x0[2::3] = r_lp * 0.993
        x0 = project_to_feasible(x0)
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds_obj, constraints=cons,
                           options={'maxiter': 25000, 'ftol': 1e-14, 'disp': False})
            if not np.isnan(res.fun):
                c_opt = np.column_stack((res.x[0::3], res.x[1::3]))
                r_opt = solve_lp_radii(c_opt)
                if r_opt is not None:
                    curr_sum = np.sum(r_opt)
                    if curr_sum > best_sum:
                        best_sum = curr_sum
                        best_centers = c_opt.copy()
                        best_radii = r_opt.copy()
        except Exception:
            pass
    
    if best_centers is not None:
        centers_c = best_centers.copy()
        radii_c = best_radii.copy()
        
        for cyc in range(250):
            x0 = np.zeros(3 * N)
            x0[0::3] = centers_c[:, 0] + np.random.normal(0, 0.0025 * (0.93 ** cyc), N)
            x0[1::3] = centers_c[:, 1] + np.random.normal(0, 0.0025 * (0.93 ** cyc), N)
            x0[2::3] = radii_c.copy() * (0.82 + 0.18 * (cyc / 250.0))
            
            k = np.random.randint(2, 10)
            subset = np.random.choice(N, size=k, replace=False)
            x0[subset * 3 + 2] *= 0.70
            
            x0 = project_to_feasible(x0)
            
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds_obj, constraints=cons,
                               options={'maxiter': 15000, 'ftol': 1e-14, 'disp': False})
                if not np.isnan(res.fun):
                    c_opt = np.column_stack((res.x[0::3], res.x[1::3]))
                    r_opt = solve_lp_radii(c_opt)
                    if r_opt is not None:
                        curr_sum = np.sum(r_opt)
                        if curr_sum > best_sum:
                            best_sum = curr_sum
                            best_centers = c_opt.copy()
                            best_radii = r_opt.copy()
                            centers_c = c_opt.copy()
                            radii_c = r_opt.copy()
            except Exception:
                pass
        
        centers_c = best_centers.copy()
        radii_c = best_radii.copy()
        
        centers_refined, radii_refined = coordinate_descent_refine(centers_c, radii_c, n_passes=8)
        refined_sum = np.sum(radii_refined)
        if refined_sum > best_sum:
            best_sum = refined_sum
            best_centers = centers_refined
            best_radii = radii_refined
    
    if best_centers is None:
        best_centers = make_hex_init(0.09, 0.0)
        best_radii = solve_lp_radii(best_centers)
        best_sum = np.sum(best_radii)
    
    centers = best_centers.copy()
    radii = best_radii.copy()
    
    for _ in range(200):
        valid = True
        for i in range(N):
            if radii[i] < 0 or centers[i, 0] < radii[i] - 1e-9 or centers[i, 0] > 1.0 - radii[i] + 1e-9 or \
               centers[i, 1] < radii[i] - 1e-9 or centers[i, 1] > 1.0 - radii[i] + 1e-9:
                valid = False
                break
        if valid:
            for i in range(N):
                for j in range(i + 1, N):
                    d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                    if d < radii[i] + radii[j] - 1e-9:
                        valid = False
                        break
                if not valid:
                    break
        if valid:
            break
        radii *= 0.9997
        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)
    
    return centers, radii, float(np.sum(radii))
