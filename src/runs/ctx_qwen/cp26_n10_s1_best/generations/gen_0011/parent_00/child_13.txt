import numpy as np
from scipy.optimize import minimize, linprog
import warnings
warnings.filterwarnings('ignore')

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Returns all inequality constraints >= 0 (vectorized)."""
    xs, ys, rs = x[0::3], x[1::3], x[2::3]
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    c = np.concatenate([
        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs
    ])
    
    # Overlap constraints: dist >= r_i + r_j
    # Using np.hypot provides stable, non-vanishing gradients at contact points
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    c = np.concatenate([c, np.hypot(dx[TRIL_IDX], dy[TRIL_IDX]) - dr[TRIL_IDX]])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-7, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)])
    return b

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist)
            idx += 1
            
    bounds_lp = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_lp, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-8)
    except Exception:
        pass
    return np.full(n, 0.04)

def make_hex_init(r0, angle, scale=1.0):
    """Generates a rotated hexagonal lattice initialization."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0 and len(pts) < N + 10:
            pts.append([x * scale, y * scale])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        
    pts = np.array(pts[:N + 10])
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
        
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.1, 0.9, (1, 2))])
    return pts[:N]

def make_spiral_init():
    """Generates a golden-angle spiral initialization."""
    pts = []
    golden_angle = np.pi * (3 - np.sqrt(5))
    for i in range(N):
        r = 0.45 * np.sqrt(i / N)
        theta = i * golden_angle
        x = 0.5 + r * np.cos(theta)
        y = 0.5 + r * np.sin(theta)
        pts.append([x, y])
    return np.array(pts)

def project_to_feasible(x):
    """Project variables to strictly satisfy bounds."""
    x = x.copy()
    for i in range(N):
        r = max(1e-6, x[3 * i + 2])
        x[3 * i] = np.clip(x[3 * i], r, 1.0 - r)
        x[3 * i + 1] = np.clip(x[3 * i + 1], r, 1.0 - r)
        x[3 * i + 2] = r
    return x

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    bounds_obj = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -1.0
    best_x = None
    
    # Phase 1: Diverse Initial Configurations -> SLSQP
    inits = []
    for scale in [0.90, 0.95, 1.0, 1.05, 1.1]:
        r0 = 0.09 * scale
        for ang in np.linspace(-0.4, 0.4, 13):
            inits.append(make_hex_init(r0, ang))
            
    inits.append(make_spiral_init())
    for _ in range(8):
        inits.append(np.random.uniform(0.15, 0.85, (N, 2)))
        
    for c_init in inits:
        x0 = np.zeros(3 * N)
        x0[0::3] = c_init[:, 0]
        x0[1::3] = c_init[:, 1]
        r_lp = solve_lp_radii(c_init)
        x0[2::3] = r_lp
        x0 = project_to_feasible(x0)
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds_obj, constraints=cons,
                           options={'maxiter': 15000, 'ftol': 1e-13, 'disp': False})
            if not np.isnan(res.fun):
                x0 = res.x.copy()
                curr_sum = np.sum(x0[2::3])
                if curr_sum > best_sum and np.min(constraints(x0)) >= -1e-6:
                    best_sum = curr_sum
                    best_x = x0.copy()
        except Exception:
            pass
            
    # Phase 2: Basin Hopping / Simulated Annealing on Centers
    if best_x is not None:
        temp = 0.015
        for epoch in range(120):
            # Perturb centers
            pert_centers = best_x[0::3].reshape(N, 2) + np.random.normal(0, temp, (N, 2))
            pert_centers = np.clip(pert_centers, 0.01, 0.99)
            
            # Random rotation to explore orientation space
            ang = np.random.uniform(-0.25, 0.25)
            c, s = np.cos(ang), np.sin(ang)
            pert_centers = (pert_centers - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
            pert_centers = np.clip(pert_centers, 0.01, 0.99)
            
            # Random swap of a subset to break contact graphs
            k = np.random.randint(2, 6)
            idx_swap = np.random.choice(N, k, replace=False)

            # Compute LP radii for perturbed centers
            r_pert = solve_lp_radii(pert_centers)
            new_sum = np.sum(r_pert)

            accept = False
            if new_sum > best_sum:
                accept = True
            else:
                if temp > 1e-6:
                    prob = np.exp((new_sum - best_sum) / temp)
                    if np.random.rand() < prob:
                        accept = True
                        
            if accept:
                best_sum = new_sum
                best_x = np.zeros(3 * N)
                best_x[0::3] = pert_centers[:, 0]
                best_x[1::3] = pert_centers[:, 1]
                best_x[2::3] = r_pert
                
                # Refine with SLSQP from new position
                best_x = project_to_feasible(best_x)
                try:
                    res = minimize(objective, best_x, method='SLSQP', bounds=bounds_obj, constraints=cons,
                                   options={'maxiter': 10000, 'ftol': 1e-13, 'disp': False})
                    if not np.isnan(res.fun):
                        best_x = res.x.copy()
                        best_sum = np.sum(best_x[2::3])
                except Exception:
                    pass
                    
            # Cool temperature
            temp *= 0.965
            
    # Phase 3: Fine-grained local search on centers (Coordinate Descent)
    if best_x is not None:
        curr_centers = np.column_stack((best_x[0::3], best_x[1::3]))
        for _ in range(4):
            improved = True
            while improved:
                improved = False
                best_local_sum = np.sum(best_x[2::3])
                for i in range(N):
                    for dx in [-0.004, -0.0015, -0.0005, 0.0005, 0.0015, 0.004]:
                        for dy in [-0.004, -0.0015, -0.0005, 0.0005, 0.0015, 0.004]:
                            if dx == 0 and dy == 0: continue
                            new_centers = curr_centers.copy()
                            new_centers[i, 0] = np.clip(curr_centers[i, 0] + dx, 0.01, 0.99)
                            new_centers[i, 1] = np.clip(curr_centers[i, 1] + dy, 0.01, 0.99)
                            
                            r_new = solve_lp_radii(new_centers)
                            if r_new is not None:
                                s_new = np.sum(r_new)
                                if s_new > best_local_sum + 1e-7:
                                    curr_centers = new_centers
                                    best_local_sum = s_new
                                    best_x[0::3] = curr_centers[:, 0]
                                    best_x[1::3] = curr_centers[:, 1]
                                    best_x[2::3] = r_new
                                    best_sum = s_new
                                    improved = True
                                    break
                        if improved: break

    # Fallback (should not be reached given robust starts)

        best_x = np.zeros(3 * N)
        best_x[0::3] = np.tile(np.linspace(0.1, 0.9, 5), 6)[:N]
        best_x[1::3] = np.repeat(np.linspace(0.1, 0.9, 6), 5)[:N]
        best_x[2::3] = 0.06
        best_sum = np.sum(best_x[2::3])

    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3].copy()

    for _ in range(100):

                valid = False
                break

                for j in range(i + 1, N):
                    d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])

                        break
                if not valid:

            break
            
        # Minimal shrinkage to recover strict feasibility
        radii *= 0.9999
        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)

    return centers, radii, float(np.sum(radii))
