import numpy as np
from scipy.optimize import minimize, linprog
import math
import warnings

warnings.filterwarnings('ignore')

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Returns all inequality constraints >= 0 (vectorized)."""
    xs, ys, rs = x[0::3], x[1::3], x[2::3]
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    c = np.concatenate([
        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs
    ])
    
    # Overlap constraints: dist >= r_i + r_j
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    c = np.concatenate([c, np.hypot(dx[TRIL_IDX], dy[TRIL_IDX]) - dr[TRIL_IDX]])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-7, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)])
    return b

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist)
            idx += 1
            
    bounds_lp = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_lp, method='highs')
        if res.success and np.min(res.x) >= -1e-9:
            return np.maximum(res.x, 1e-7)
    except Exception:
        pass
    # Fallback safe radii
    safe_r = np.full(n, 0.02)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        for j in range(n):
            if i != j:
                d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                if d < lim:
                    lim = d
        safe_r[i] = max(1e-7, lim * 0.95)
    return safe_r

def make_hex_init(r0, angle):
    """Generates a rotated hexagonal lattice initialization."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0 and len(pts) < N + 10:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        
    pts = np.array(pts[:N + 10])
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
        
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.1, 0.9, (1, 2))])
    return pts[:N]

def make_corner_init():
    """Generates an initialization focused on corners and edges."""
    pts = np.zeros((N, 2))
    r = 0.09
    pts[0] = [r, r]
    pts[1] = [1-r, r]
    pts[2] = [r, 1-r]
    pts[3] = [1-r, 1-r]
    pts[4] = [0.5, r]
    pts[5] = [0.5, 1-r]
    pts[6] = [r, 0.5]
    pts[7] = [1-r, 0.5]
    pts[8] = [0.5, 0.5]
    idx = 9
    
    y = r + 0.04
    row = 0
    while idx < N and y + r <= 1.0:
        x = r + 0.04 if row % 2 == 0 else r + 0.16
        while idx < N and x + r <= 1.0:
            pts[idx] = [x, y]
            idx += 1
            x += 0.20
        y += 0.13
        row += 1
        
    for i in range(idx, N):
        pts[i] = np.random.uniform(0.1, 0.9, 2)
    pts += np.random.uniform(-0.002, 0.002, pts.shape)
    return pts

def force_spread_init(seed):
    """Force-directed layout to spread points evenly."""
    np.random.seed(seed)
    pts = np.random.uniform(0.15, 0.85, (N, 2))
    for _ in range(200):
        f = np.zeros_like(pts)
        for i in range(N):
            for j in range(i + 1, N):
                dx = pts[j] - pts[i]
                d = np.hypot(dx[0], dx[1])
                if d < 0.35 and d > 1e-5:
                    rep = 0.01 / (d**2 + 0.001)
                    f[i] -= dx * rep / d
                    f[j] += dx * rep / d
            for dim in range(2):
                if pts[i, dim] < 0.08: f[i, dim] += 0.05
                elif pts[i, dim] > 0.92: f[i, dim] -= 0.05
        pts += f * 0.05
        pts = np.clip(pts, 0.02, 0.98)
    return pts

def project_to_feasible(x):
    """Project variables to strictly satisfy bounds."""
    x = x.copy()
    for i in range(N):
        r = max(1e-6, x[3 * i + 2])
        x[3 * i] = np.clip(x[3 * i], r, 1.0 - r)
        x[3 * i + 1] = np.clip(x[3 * i + 1], r, 1.0 - r)
        x[3 * i + 2] = r
    return x

def local_search_lp(centers, radii, iters=1500):
    """LP-driven derivative-free local search on centers."""
    best_c = centers.copy()
    best_r = radii.copy()
    best_s = np.sum(radii)
    step = 0.015
    
    for i in range(iters):
        # Adaptive step size
        current_step = step * (1.0 - 0.8 * (i / iters))
        
        # Perturb 1 to 3 circles
        k = np.random.randint(1, 4)
        idx = np.random.choice(N, k, replace=False)
        new_c = best_c.copy()
        new_c[idx] += np.random.normal(0, current_step, (k, 2))
        new_c = np.clip(new_c, 0.01, 0.99)
        
        # Random swap to break symmetry (probability decreases over time)
        if np.random.rand() < 0.4 * (1.0 - i / iters):
            i1, i2 = np.random.choice(N, 2, replace=False)
            new_c[[i1, i2]] = new_c[[i2, i1]]
            
        # Evaluate via LP
        r_new = solve_lp_radii(new_c)
        s_new = np.sum(r_new)
        
        # Accept if better
        if s_new > best_s:
            best_c = new_c
            best_r = r_new
            best_s = s_new
            step = min(0.02, step * 1.1)
        else:
            step *= 0.97
            
    return best_c, best_r, best_s

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    bounds_obj = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -1.0
    best_x = None
    
    # Phase 1: Diverse Initial Configurations
    inits = []
    for ang in np.linspace(-0.4, 0.4, 11):
        inits.append(make_hex_init(0.095, ang))
        inits.append(make_hex_init(0.10, ang))
        
    for _ in range(4):
        inits.append(make_corner_init())
        
    for s in range(8):
        inits.append(force_spread_init(s))

    # Phase 2: Multi-start SLSQP Optimization
    for c_init in inits:
        x0 = np.zeros(3 * N)
        x0[0::3] = c_init[:, 0]
        x0[1::3] = c_init[:, 1]
        
        r_lp = solve_lp_radii(c_init)
        x0[2::3] = r_lp
        x0 = project_to_feasible(x0)

        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds_obj, constraints=cons,
                           options={'maxiter': 15000, 'ftol': 1e-13, 'disp': False})
            if not np.isnan(res.fun):
                x0 = res.x.copy()
        except Exception:
            continue

        curr_sum = np.sum(x0[2::3])
        if curr_sum > best_sum and np.min(constraints(x0)) >= -1e-6:
            best_sum = curr_sum
            best_x = x0.copy()

    # Phase 3: Aggressive LP-Based Local Search & Symmetry Breaking
    if best_x is not None:
        curr_c = np.column_stack((best_x[0::3], best_x[1::3]))
        curr_r = best_x[2::3].copy()
        curr_s = best_sum
        
        # Run multiple rounds of local search followed by SLSQP polish
        for round_num in range(4):
            # LP-driven local search
            new_c, new_r, new_s = local_search_lp(curr_c, curr_r, iters=1200)
            
            if new_s > curr_s:
                curr_c, curr_r, curr_s = new_c, new_r, new_s
                
                # Polish with SLSQP
                x_pol = np.zeros(3 * N)
                x_pol[0::3] = curr_c[:, 0]
                x_pol[1::3] = curr_c[:, 1]
                x_pol[2::3] = curr_r
                x_pol = project_to_feasible(x_pol)

                except Exception:
                    pass
                
            best_sum = curr_s

    centers = np.column_stack((best_x[0::3], best_x[1::3]))

    # Final strict validity adjustment against 1e-12 tolerance
    for _ in range(100):

                valid = False

    return centers, radii, float(np.sum(radii))
