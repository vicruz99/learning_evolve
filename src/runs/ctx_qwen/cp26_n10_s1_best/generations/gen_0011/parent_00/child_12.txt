

<strategy>
To break past the ~2.634 barrier, we need a more robust global search strategy that decouples center placement from radius optimization. Local optimizers like SLSQP struggle with the highly non-convex contact graph of circle packings. 

Strategy:
1. **LP-Decoupled Objective**: Treat center coordinates as the primary variables and compute optimal radii via Linear Programming (`scipy.optimize.linprog`). This creates a smoother, albeit non-differentiable, objective landscape that guides centers to maximally spaced configurations.
2. **Differential Evolution + Basin Hopping**: Use `scipy.optimize.differential_evolution` for broad global search over center coordinates, followed by `basinhopping` for local escape. The objective function is `-sum(LP_radii)`.
3. **Hybrid Local Refinement**: After finding promising center layouts, run SLSQP on the full (centers + radii) space to polish the solution to machine precision.
4. **Targeted Perturbations**: Implement a contact-graph-aware perturbation that temporarily shrinks tightly coupled circles to allow topological rearrangements, then re-optimizes.
5. **Strict Validation & Repair**: Ensure all outputs strictly satisfy the 1e-12 tolerance required by the validator.

This approach separates the difficult combinatorial topology search from the continuous radius optimization, significantly improving the chance of finding the global optimum or a much better local optimum.
</strategy>

import numpy as np
from scipy.optimize import minimize, linprog, differential_evolution, basinhopping
import warnings
warnings.filterwarnings('ignore')

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist)
            idx += 1
            
    bounds_lp = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_lp, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-8)
    except Exception:
        pass
    return np.full(n, 1e-8)

def objective_de(centers_flat):
    """Objective for DE/Basinhopping: minimize negative sum of LP radii."""
    centers = centers_flat.reshape(N, 2)
    centers = np.clip(centers, 0.005, 0.995)
    radii = solve_lp_radii(centers)
    return -np.sum(radii)

def make_hex_init(r0, angle, scale=1.0):
    """Generates a hexagonal lattice initialization."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0 and len(pts) < N + 10:
            pts.append([x * scale, y * scale])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        
    pts = np.array(pts[:N + 10])
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
        
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.1, 0.9, (1, 2))])
    return pts[:N]

def objective_slscp(x):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints_slscp(x):
    """Vectorized inequality constraints g(x) >= 0."""
    xs, ys, rs = x[0::3], x[1::3], x[2::3]
    c = np.concatenate([
        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs
    ])
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    c = np.concatenate([c, np.hypot(dx[TRIL_IDX], dy[TRIL_IDX]) - dr[TRIL_IDX]])
    return c

def get_bounds_slscp():
    """Variable bounds for SLSQP."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)])
    return b

def project_to_feasible(x):
    """Project variables to strictly satisfy bounds."""
    x = x.copy()
    for i in range(N):
        r = max(1e-6, x[3 * i + 2])
        x[3 * i] = np.clip(x[3 * i], r, 1.0 - r)
        x[3 * i + 1] = np.clip(x[3 * i + 1], r, 1.0 - r)
        x[3 * i + 2] = r
    return x

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Phase 1: Differential Evolution for robust global search on centers
    print("Phase 1: Global search with DE...")
    bounds_de = [(0.01, 0.99)] * (N * 2)
    try:
        de_res = differential_evolution(objective_de, bounds_de, popsize=15, maxiter=80, 
                                        mutation=(0.5, 1.5), recombination=0.9, seed=42, 
                                        polish=False, disp=False)
        c_de = de_res.x.reshape(N, 2)
        r_de = solve_lp_radii(c_de)
        if np.sum(r_de) > best_sum:
            best_sum = np.sum(r_de)
            best_centers = c_de.copy()
            best_radii = r_de.copy()
    except Exception as e:
        print(f"DE failed: {e}")

    # Phase 2: Basin Hopping on top of DE result for local escape
    print("Phase 2: Basin hopping refinement...")
    if best_centers is not None:
        def kick(x):
            noise = np.random.normal(0, 0.01, x.shape)
            return x + noise
        
        try:
            bh_minimizer_kwargs = {'method': 'Nelder-Mead', 'options': {'maxiter': 500, 'xatol': 1e-5, 'fatol': 1e-5}}
            bh_res = basinhopping(objective_de, best_centers.flatten(), kick=kick, 
                                  niter=40, minimizer_kwargs=bh_minimizer_kwargs, 
                                  stepsize=0.02, accept_test=None, callback=None)
            c_bh = bh_res.x.reshape(N, 2)
            r_bh = solve_lp_radii(c_bh)
            if np.sum(r_bh) > best_sum:
                best_sum = np.sum(r_bh)
                best_centers = c_bh.copy()
                best_radii = r_bh.copy()
        except Exception as e:
            print(f"Basin hopping failed: {e}")

    # Phase 3: Diverse Initial Starts + SLSQP Local Optimization
    print("Phase 3: Multi-start SLSQP refinement...")
    bounds_obj = get_bounds_slscp()
    cons = {'type': 'ineq', 'fun': constraints_slscp}
    
    inits = []
    # Generate varied hexagonal lattices
    for scale in [0.95, 1.0, 1.05]:
        for ang in np.linspace(-0.3, 0.3, 7):
            inits.append(make_hex_init(0.09, ang, scale))
    # Add best centers from global search
    if best_centers is not None:
        inits.append(best_centers)
        
    for c_init in inits:
        x0 = np.zeros(3 * N)
        x0[0::3] = c_init[:, 0]
        x0[1::3] = c_init[:, 1]
        r_lp = solve_lp_radii(c_init)
        x0[2::3] = r_lp
        x0 = project_to_feasible(x0)
        
        try:
            res = minimize(objective_slscp, x0, method='SLSQP', bounds=bounds_obj, constraints=cons,
                           options={'maxiter': 15000, 'ftol': 1e-14, 'disp': False})
            if not np.isnan(res.fun):
                c_tmp = np.column_stack((res.x[0::3], res.x[1::3]))
                r_new = solve_lp_radii(c_tmp)
                s_new = np.sum(r_new)
                if s_new > best_sum and np.min(constraints_slscp(res.x)) >= -1e-6:
                    best_sum = s_new
                    best_centers = c_tmp.copy()
                    best_radii = r_new.copy()
        except Exception:
            continue

    # Phase 4: Aggressive Topology Search (Deflation & Repositioning)
    print("Phase 4: Topology escape search...")
    if best_centers is not None:
        for cyc in range(60):
            x0 = np.zeros(3 * N)
            x0[0::3] = best_centers[:, 0]
            x0[1::3] = best_centers[:, 1]
            x0[2::3] = best_radii
            
            # Perturb centers
            noise = 0.0025 * (0.92 ** cyc)
            x0[0::3] += np.random.normal(0, noise, N)
            x0[1::3] += np.random.normal(0, noise, N)
            
            # Deflate a subset to break rigid contacts
            subset = np.random.choice(N, size=max(4, N // 3), replace=False)
            x0[subset * 3 + 2] *= 0.85
            
            x0 = project_to_feasible(x0)
            
            try:
                res = minimize(objective_slscp, x0, method='SLSQP', bounds=bounds_obj, constraints=cons,
                               options={'maxiter': 10000, 'ftol': 1e-13, 'disp': False})
                if not np.isnan(res.fun):
                    c_p = np.column_stack((res.x[0::3], res.x[1::3]))
                    r_p = solve_lp_radii(c_p)
                    s_p = np.sum(r_p)
                    if s_p > best_sum and np.min(constraints_slscp(res.x)) >= -1e-5:
                        best_sum = s_p
                        best_centers = c_p.copy()
                        best_radii = r_p.copy()
            except Exception:
                pass

    # Fallback
    if best_centers is None:
        best_centers = make_hex_init(0.09, 0.0)
        best_radii = solve_lp_radii(best_centers)
        best_sum = np.sum(best_radii)

    centers = best_centers.copy()
    radii = best_radii.copy()

    # Final strict validity adjustment against 1e-12 tolerance
    for _ in range(100):
        valid = True
        for i in range(N):
            if radii[i] < 0 or centers[i, 0] < radii[i] - 1e-9 or centers[i, 0] > 1.0 - radii[i] + 1e-9 or \
               centers[i, 1] < radii[i] - 1e-9 or centers[i, 1] > 1.0 - radii[i] + 1e-9:
                valid = False
                break
        if valid:
            for i in range(N):
                for j in range(i + 1, N):
                    d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                    if d < radii[i] + radii[j] - 1e-9:
                        valid = False
                        break
                if not valid:
                    break
        if valid:
            break
            
        # Minimal shrinkage to recover strict feasibility
        radii *= 0.9998
        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)

    return centers, radii, float(np.sum(radii))
