4. **SLSQP Polish**: Finally, run a standard SLSQP optimization on the best configuration found to precisely tighten contacts and satisfy the `1e-12` validation tolerance exactly.

import numpy as np
from scipy.optimize import minimize, linprog
import warnings
warnings.filterwarnings('ignore')

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-6, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)])
    return b

def compute_violations_sq(x):
    """Compute sum of squared constraint violations."""
    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]
    v = 0.0
    # Boundary violations
    v += np.sum(np.maximum(0.0, rs - xs)**2)
    v += np.sum(np.maximum(0.0, rs - (1.0 - xs))**2)
    v += np.sum(np.maximum(0.0, rs - ys)**2)
    v += np.sum(np.maximum(0.0, rs - (1.0 - ys))**2)
    # Overlap violations
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dist = np.hypot(dx, dy)
    dr = rs[:, None] + rs[None, :]
    ov = np.maximum(0.0, dr - dist)
    v += np.sum(ov[TRIL_IDX]**2)
    return v

def penalty_obj(x, penalty):
    """Unconstrained objective: maximize sum of radii with penalty for violations."""
    return -np.sum(x[2::3]) + penalty * compute_violations_sq(x)

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist)
            idx += 1
            
    bounds_lp = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_lp, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-7)
    except Exception:
        pass
    return np.full(n, 0.05)

def make_hex_init(r0, angle):
    """Generates a rotated hexagonal lattice initialization."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0 and len(pts) < N + 10:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
    pts = np.array(pts[:N + 10])
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.1, 0.9, (1, 2))])
    return pts[:N]

def constraints_slsqp(x):
    """Vectorized inequality constraints for SLSQP polish: g(x) >= 0."""
    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]
    c = np.concatenate([
        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs
    ])
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    c = np.concatenate([c, dx[TRIL_IDX]**2 + dy[TRIL_IDX]**2 - dr[TRIL_IDX]**2])
    return c

def objective_slsqp(x):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(x[2::3])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints_slsqp}
    
    best_sum = -1.0
    best_x = None
    
    # Phase 1: Diverse Initial Configurations
    inits = []
    for ang in np.linspace(-0.45, 0.45, 13):
        inits.append(make_hex_init(0.088, ang))
    for ang in np.linspace(-0.45, 0.45, 13):
        inits.append(make_hex_init(0.095, ang))
    for _ in range(12):
        pts = np.random.uniform(0.15, 0.85, (N, 2))
        inits.append(pts)

    # Phase 2: Penalty Method + LP Alternation
    for pts in inits:
        x0 = np.zeros(3 * N)
        x0[0::3] = pts[:, 0]
        x0[1::3] = pts[:, 1]
        x0[2::3] = 0.085
        
        x_curr = x0.copy()
        penalty = 8.0
        
        # Anneal penalty
        for _ in range(28):
            res = minimize(penalty_obj, x_curr, args=(penalty,), method='L-BFGS-B', 
                           bounds=bounds, options={'maxiter': 250, 'ftol': 1e-13, 'gtol': 1e-9})
            if not np.isnan(res.fun):
                x_curr = res.x.copy()
                # LP refinement to snap radii to exact limits
                centers_tmp = np.column_stack((x_curr[0::3], x_curr[1::3]))
                r_lp = solve_lp_radii(centers_tmp)
                x_curr[2::3] = r_lp
            penalty *= 1.35
            
        if compute_violations_sq(x_curr) < 1e-5:
            curr_sum = np.sum(x_curr[2::3])
            if curr_sum > best_sum:
                best_sum = curr_sum
                best_x = x_curr.copy()

    # Phase 3: Topology Search via Deflation & Perturbation
    if best_x is not None:
        for cyc in range(60):
            x_p = best_x.copy()
            noise = 0.004 * (0.88 ** cyc)
            x_p += np.random.normal(0, noise, 3 * N)
            
            # Deflate random subset to break rigid contacts
            k = np.random.randint(2, 6)
            subset = np.random.choice(N, size=k, replace=False)
            x_p[subset * 3 + 2] *= 0.70
            
            x_p[0::3] = np.clip(x_p[0::3], 1e-5, 1.0 - 1e-5)
            x_p[1::3] = np.clip(x_p[1::3], 1e-5, 1.0 - 1e-5)
            x_p[2::3] = np.maximum(x_p[2::3], 1e-5)
            
            penalty = 30.0
            for _ in range(18):
                res = minimize(penalty_obj, x_p, args=(penalty,), method='L-BFGS-B',
                               bounds=bounds, options={'maxiter': 150, 'ftol': 1e-13, 'gtol': 1e-9})
                if not np.isnan(res.fun):
                    x_p = res.x.copy()
                    centers_tmp = np.column_stack((x_p[0::3], x_p[1::3]))
                    r_lp = solve_lp_radii(centers_tmp)
                    x_p[2::3] = r_lp
                penalty *= 1.4
                
            if compute_violations_sq(x_p) < 1e-5:
                curr_sum = np.sum(x_p[2::3])
                if curr_sum > best_sum:
                    best_sum = curr_sum
                    best_x = x_p.copy()

    # Phase 4: SLSQP Precise Polish
    if best_x is not None:
        try:
            res = minimize(objective_slsqp, best_x, method='SLSQP', bounds=bounds, 
                           constraints=cons, options={'maxiter': 20000, 'ftol': 1e-14, 'disp': False})
            if not np.isnan(res.fun) and np.min(constraints_slsqp(res.x)) >= -1e-7:
                curr_sum = -res.fun
                if curr_sum > best_sum:
                    best_sum = curr_sum
                    best_x = res.x.copy()
        except Exception:
            pass

    # Fallback
    if best_x is None:
        best_x = np.zeros(3 * N)
        best_x[0::3] = np.tile(np.linspace(0.1, 0.9, 5), 6)[:N]
        best_x[1::3] = np.repeat(np.linspace(0.1, 0.9, 6), 5)[:N]
        best_x[2::3] = 0.06
        
    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3].copy()
    
    # Final strict validation repair
    for _ in range(100):
        valid = True
        for i in range(N):
            if radii[i] < 0 or centers[i, 0] < radii[i] - 1e-9 or centers[i, 0] > 1.0 - radii[i] + 1e-9 or \
               centers[i, 1] < radii[i] - 1e-9 or centers[i, 1] > 1.0 - radii[i] + 1e-9:
                valid = False
                break
        if valid:
            for i in range(N):
                for j in range(i + 1, N):
                    d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                    if d < radii[i] + radii[j] - 1e-9:
                        valid = False
                        break
                if not valid:
                    break
        if valid:
            break
            
        radii *= 0.9998
        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)
        
    return centers, radii, float(np.sum(radii))
