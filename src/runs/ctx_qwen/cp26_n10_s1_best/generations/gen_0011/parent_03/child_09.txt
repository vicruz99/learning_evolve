import numpy as np
from scipy.optimize import minimize, linprog
import warnings
warnings.filterwarnings('ignore')

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist)
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-7)
    except Exception:
        pass
    return np.full(n, 0.01)

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-7, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)])
    return b

def constraints_slsqp(x):
    """Vectorized inequality constraints for SLSQP: g(x) >= 0."""
    xs, ys, rs = x[0::3], x[1::3], x[2::3]
    c = np.concatenate([
        xs - rs, 1.0 - xs - rs,
        ys - rs, 1.0 - ys - rs
    ])
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    c = np.concatenate([c, np.hypot(dx[TRIL_IDX], dy[TRIL_IDX]) - dr[TRIL_IDX]])
    return c

def objective_slsqp(x):
    """Objective to minimize: negative sum of radii."""
    return -np.sum(x[2::3])

def project_x(x):
    """Project variables to strictly satisfy bounds."""
    x = x.copy()
    for i in range(N):
        r = max(1e-7, x[3 * i + 2])
        x[3 * i] = np.clip(x[3 * i], r, 1.0 - r)
        x[3 * i + 1] = np.clip(x[3 * i + 1], r, 1.0 - r)
        x[3 * i + 2] = r
    return x

def make_hex_init(r0, angle=0.0):
    """Generates a hexagonal lattice initialization with optional rotation."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
    pts = np.array(pts[:N + 10])
    if angle != 0.0:
        c_a, s_a = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c_a, -s_a], [s_a, c_a]]) + 0.5
    mask = (pts[:, 0] >= 0.01) & (pts[:, 0] <= 0.99) & (pts[:, 1] >= 0.01) & (pts[:, 1] <= 0.99)
    pts = pts[mask]
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.1, 0.9, (1, 2))])
    return pts[:N]

def make_force_init(seed):
    """Force-directed layout to spread points evenly and push to boundaries."""
    np.random.seed(seed)
    pts = np.random.uniform(0.15, 0.85, (N, 2))
    for step in range(300):
        f = np.zeros_like(pts)
        lr = 0.02 * (1.0 - step / 300)
        for i in range(N):
            for j in range(i + 1, N):
                dx = pts[j] - pts[i]
                d = np.linalg.norm(dx)
                if d < 0.3 and d > 1e-6:
                    rep = 0.01 / (d**2 + 0.001)
                    f[i] -= dx * rep / d
                    f[j] += dx * rep / d
            for dim in range(2):
                if pts[i, dim] < 0.1: f[i, dim] += 0.05
                elif pts[i, dim] > 0.9: f[i, dim] -= 0.05
        pts += f * lr
        pts = np.clip(pts, 0.02, 0.98)
    return pts

def local_search_lp(centers, radii, iterations=40):
    """Coordinate descent local search: move one circle at a time to maximize LP sum of radii."""
    curr_sum = np.sum(radii)
    best_centers = centers.copy()
    best_radii = radii.copy()
    
    for t in range(iterations):
        improved = False
        order = np.random.permutation(N)
        for i in order:
            best_local_sum = curr_sum
            best_local_centers = best_centers.copy()
            
            move_scale = 0.004 / (1.0 + t * 0.05)
            for _ in range(20):
                nc = best_centers.copy()
                nc[i] += np.random.uniform(-move_scale, move_scale, 2)
                nc[i] = np.clip(nc[i], 0.001, 0.999)
                
                r_tmp = solve_lp_radii(nc)
                s_tmp = np.sum(r_tmp)
                if s_tmp > best_local_sum:
                    best_local_sum = s_tmp
                    best_local_centers = nc.copy()
                    
            if best_local_sum > curr_sum + 1e-9:
                best_centers = best_local_centers
                best_radii = solve_lp_radii(best_centers)
                curr_sum = best_local_sum
                improved = True
        if not improved:
            break
            
    return best_centers, best_radii, curr_sum

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints_slsqp}
    
    best_sum = -1.0
    best_x = None
    
    # Phase 1: Diverse Initial Configurations
    inits = []
    for r0 in [0.08, 0.085, 0.09, 0.095, 0.10, 0.105, 0.11]:
        for ang in np.linspace(-0.4, 0.4, 13):
            inits.append(make_hex_init(r0, ang))
    for s in range(12):
        inits.append(make_force_init(s))
        
    # Phase 2: SLSQP Optimization on each start
    for pts in inits:
        x0 = np.zeros(3 * N)
        x0[0::3] = pts[:, 0]
        x0[1::3] = pts[:, 1]
        r_lp = solve_lp_radii(pts)
        x0[2::3] = np.maximum(r_lp * 0.995, 1e-5)
        x0 = project_x(x0)
        
        try:
            res = minimize(objective_slsqp, x0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 8000, 'ftol': 1e-13, 'disp': False})
            if not np.isnan(res.fun):
                c_tmp = np.column_stack((res.x[0::3], res.x[1::3]))
                r_new = solve_lp_radii(c_tmp)
                s_val = np.sum(r_new)
                if s_val > best_sum:
                    best_sum = s_val
                    best_x = res.x.copy()
                    best_x[2::3] = r_new
        except Exception:
            pass
            
    # Phase 3: LP-based Local Search & Refinement
    if best_x is not None:
        centers = np.column_stack((best_x[0::3], best_x[1::3]))
        radii = best_x[2::3]
        curr_sum = best_sum
        
        # Iterative coordinate descent with LP
        for cycle in range(50):
            centers, radii, curr_sum = local_search_lp(centers, radii, iterations=30)
            if curr_sum > best_sum:
                best_sum = curr_sum
                best_x[0::3] = centers[:, 0]
                best_x[1::3] = centers[:, 1]
                best_x[2::3] = radii
                
        # Phase 4: Deflation & SLSQP restart to escape local minima
        for cyc in range(35):
            x0 = best_x.copy()
            shrink = 0.70 + 0.30 * (cyc / 35.0)
            x0[2::3] *= shrink
            noise = 0.003 / (cyc + 1)
            x0[0::3] += np.random.normal(0, noise, N)
            x0[1::3] += np.random.normal(0, noise, N)
            x0 = project_x(x0)
            
            try:
                res = minimize(objective_slsqp, x0, method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 8000, 'ftol': 1e-13, 'disp': False})
                if not np.isnan(res.fun):

    # Fallback initialization

        best_x[0::3] = np.tile(np.linspace(0.1, 0.9, 5), 6)[:N]
        best_x[1::3] = np.repeat(np.linspace(0.1, 0.9, 6), 5)[:N]

    radii = best_x[2::3].copy()
    
    # Final LP squeeze to maximize radii for the best centers found
    final_r = solve_lp_radii(centers)

        radii *= 0.9995
