import numpy as np
from scipy.optimize import minimize, linprog
import warnings
warnings.filterwarnings('ignore')

N = 26

def obj_func(x):
    """Objective: maximize sum of radii."""
    return -np.sum(x[2::3])

def con_func(x):
    """Inequality constraints >= 0."""
    xs, ys, rs = x[0::3], x[1::3], x[2::3]
    
    # Boundary constraints
    c = np.concatenate([
        xs - rs, 1.0 - xs - rs,
        ys - rs, 1.0 - ys - rs
    ])
    
    # Overlap constraints using Euclidean distance for stable gradients
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    idx = np.tril_indices(N, -1)
    c = np.concatenate([c, np.hypot(dx[idx], dy[idx]) - dr[idx]])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-7, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)])
    return b

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        x, y = centers[i]
        lim = min(x, 1.0 - x, y, 1.0 - y)
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, d)
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-7)
    except Exception:
        pass
    return np.full(n, 0.02)

def project_x(x):
    """Project variables to strictly satisfy bounds."""
    x = x.copy()
    for i in range(N):
        r = max(1e-7, x[3*i+2])
        x[3*i] = np.clip(x[3*i], r, 1.0-r)
        x[3*i+1] = np.clip(x[3*i+1], r, 1.0-r)
        x[3*i+2] = r
    return x

def make_hex_start(r0, angle):
    """Generates a rotated hexagonal lattice initialization."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0 and len(pts) < N + 10:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
    pts = np.array(pts[:N+10])
    c = np.array([0.5, 0.5])
    ca, sa = np.cos(angle), np.sin(angle)
    pts = (pts - c) @ np.array([[ca, -sa], [sa, ca]]) + c
    mask = (pts[:,0]>=0.02) & (pts[:,0]<=0.98) & (pts[:,1]>=0.02) & (pts[:,1]<=0.98)
    pts = pts[mask]
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.2, 0.8, (1,2))])
    return pts[:N]

def make_grid_start(seed):
    """Generates a perturbed grid initialization."""
    np.random.seed(seed)
    pts = []
    for i in range(5):
        for j in range(6):
            pts.append([0.08 + i*0.18, 0.08 + j*0.15])
    np.random.shuffle(pts)
    return np.array(pts[:N])

def force_spread(centers, radii, steps=100):
    """Force-directed layout to spread points evenly and push to boundaries."""
    pts = centers.copy()
    r = radii.copy()
    for step in range(steps):
        f = np.zeros_like(pts)
        lr = 0.012 * (1.0 - step/steps)
        for i in range(N):
            for j in range(i+1, N):
                dx = pts[j] - pts[i]
                d = np.hypot(dx[0], dx[1])
                if d < r[i] + r[j] + 0.04 and d > 1e-5:
                    rep = 0.015 / (d**2 + 0.001)
                    f[i] -= dx * rep / d
                    f[j] += dx * rep / d
            for dim in range(2):
                if pts[i, dim] < r[i] + 0.01: f[i, dim] += 0.04
                elif pts[i, dim] > 1.0 - r[i] - 0.01: f[i, dim] -= 0.04
        pts += f * lr
        pts = np.clip(pts, 0.005, 0.995)
    return pts

def coord_search(centers, radii, iterations=20):
    """Local search: move one circle at a time to maximize LP sum of radii."""
    improved = True
    while improved and iterations > 0:
        improved = False
        iterations -= 1
        order = np.random.permutation(N)
        for i in order:
            best_pos = centers[i].copy()
            best_sum = np.sum(radii)
            for _ in range(15):
                dx = np.random.uniform(-0.012, 0.012)
                dy = np.random.uniform(-0.012, 0.012)
                nc = centers.copy()
                nc[i] = np.clip(nc[i] + [dx, dy], 1e-4, 1.0-1e-4)
                nr = solve_lp_radii(nc)
                s = np.sum(nr)
                if s > best_sum + 1e-7:
                    best_sum = s
                    best_pos = nc[i].copy()
                    improved = True
            centers[i] = best_pos
            radii = solve_lp_radii(centers)
    return centers, radii

def run_packing():

    for pts in starts:

        x0[1::3] = pts[:,1]

        try:

                          bounds=bounds_obj, constraints=cons,

            if res.success and not np.isnan(res.fun):
                c_tmp = np.column_stack((res.x[0::3], res.x[1::3]))
                r_tmp = solve_lp_radii(c_tmp)
                s_val = np.sum(r_tmp)
                if s_val > best_sum:
                    best_sum = s_val
                    best_centers = c_tmp.copy()
