def objective(x):

def constraints(x):
    """Returns all inequality constraints g(x) >= 0 (vectorized)."""

        xs - rs, 1.0 - xs - rs,
        ys - rs, 1.0 - ys - rs
    ])

    dy = ys[:, None] - ys[None, :]

        idx += 1
