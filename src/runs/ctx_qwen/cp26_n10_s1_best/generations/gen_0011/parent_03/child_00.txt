3. **SLSQP Refinement**: Use SLSQP on the best configurations found by the local search to simultaneously tune centers and radii, handling the non-linear constraints precisely.

import numpy as np
from scipy.optimize import minimize, linprog
import warnings

warnings.filterwarnings('ignore')

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist)
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-8)
    except Exception:
        pass
    return np.full(n, 0.01)

def force_layout(centers, steps=80):
    """Force-directed relaxation to spread points evenly and push to boundaries."""
    pts = centers.copy()
    n = pts.shape[0]
    for step in range(steps):
        f = np.zeros_like(pts)
        lr = 0.025 * (1.0 - step / steps)
        for i in range(n):
            for j in range(i + 1, n):
                dx = pts[j] - pts[i]
                d = np.hypot(dx[0], dx[1])
                if d < 0.3 and d > 1e-6:
                    rep = 0.006 / (d**2 + 0.0005)
                    f[i] -= dx * rep / d
                    f[j] += dx * rep / d
            for dim in range(2):
                if pts[i, dim] < 0.08: f[i, dim] += 0.06
                elif pts[i, dim] > 0.92: f[i, dim] -= 0.06
        pts += f * lr
        pts = np.clip(pts, 0.02, 0.98)
    return pts

def local_search_centers(centers, cycles=6):
    """Coordinate descent: move one circle at a time to maximize LP sum of radii."""
    pts = centers.copy()
    best_r = solve_lp_radii(pts)
    best_sum = np.sum(best_r)
    
    for _ in range(cycles):
        improved = False
        for i in np.random.permutation(N):
            curr_sum = best_sum
            for _ in range(10):
                dx = np.random.uniform(-0.012, 0.012)
                dy = np.random.uniform(-0.012, 0.012)
                new_pts = pts.copy()
                new_pts[i] = np.clip(pts[i] + [dx, dy], 1e-4, 1.0 - 1e-4)
                r_new = solve_lp_radii(new_pts)
                s = np.sum(r_new)
                if s > curr_sum + 1e-9:
                    curr_sum = s
                    pts[i] = new_pts[i].copy()
            if curr_sum > best_sum + 1e-9:
                best_sum = curr_sum
                improved = True
        if not improved:
            break
    return pts, solve_lp_radii(pts), best_sum

def make_hex_init(r0, angle=0.0):
    """Generates a rotated hexagonal lattice initialization."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
    pts = np.array(pts[:N + 10])
    if angle != 0.0:
        c_a, s_a = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c_a, -s_a], [s_a, c_a]]) + 0.5
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.15, 0.85, (1, 2))])
    return pts[:N]

def make_grid_init():
    """Generates a staggered grid initialization."""
    pts = np.zeros((N, 2))
    idx = 0
    rows = [5, 6, 5, 6, 4]
    y = 0.1
    for r_count in rows:
        x_step = 0.8 / (r_count - 1) if r_count > 1 else 0.5
        for k in range(r_count):
            if idx < N:
                x = 0.1 + k * x_step
                pts[idx] = [x, y]
                idx += 1
        y += 0.2
    return pts

def constraints(x):
    """Returns all inequality constraints g(x) >= 0."""
    xs, ys, rs = x[0::3], x[1::3], x[2::3]
    c = np.concatenate([xs - rs, 1.0 - xs - rs, ys - rs, 1.0 - ys - rs])
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    c = np.concatenate([c, np.hypot(dx[TRIL_IDX], dy[TRIL_IDX]) - dr[TRIL_IDX]])
    return c

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-7, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)])
    return b

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    best_sum = -1.0
    best_centers = None
    best_radii = None
    bounds_obj = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    
    # Phase 1: Diverse Initial Configurations
    inits = []
    for r0 in [0.085, 0.09, 0.095, 0.10, 0.105]:
        for ang in np.linspace(-0.35, 0.35, 7):
            inits.append(make_hex_init(r0, ang))
    for _ in range(5):
        g = make_grid_init()
        g += np.random.uniform(-0.01, 0.01, g.shape)
        inits.append(g)
    for s in range(8):
        np.random.seed(s + 100)
        pts = np.random.uniform(0.15, 0.85, (N, 2))
        inits.append(force_layout(pts, steps=60))
        
    # Phase 2: Local Search on Centers to find promising topologies
    for pts in inits:
        pts, r, s = local_search_centers(pts, cycles=5)
        if s > best_sum:
            best_sum = s
            best_centers = pts.copy()
            best_radii = r.copy()
            
    # Phase 3: SLSQP Refinement on best configuration
    x0 = np.zeros(3 * N)
    x0[0::3] = best_centers[:, 0]
    x0[1::3] = best_centers[:, 1]
    x0[2::3] = best_radii * 0.995
    
    try:
        res = minimize(objective, x0, method='SLSQP', bounds=bounds_obj, constraints=cons,
                       options={'maxiter': 15000, 'ftol': 1e-13, 'disp': False})
        if not np.isnan(res.fun):
            c_opt = np.column_stack((res.x[0::3], res.x[1::3]))
            r_opt = solve_lp_radii(c_opt)
            if np.sum(r_opt) > best_sum:
                best_sum = np.sum(r_opt)
                best_centers = c_opt
                best_radii = r_opt
    except Exception:
        pass
        
    # Phase 4: Deflation & Perturbation to escape local minima
    centers = best_centers.copy()
    radii = best_radii.copy()
    for cyc in range(50):
        x0 = np.zeros(3 * N)
        noise = 0.0025 * (0.94 ** cyc)
        x0[0::3] = centers[:, 0] + np.random.normal(0, noise, N)
        x0[1::3] = centers[:, 1] + np.random.normal(0, noise, N)
        
        # Shrink a random subset to break rigid contacts
        subset = np.random.choice(N, size=max(3, N // 4), replace=False)
        rad_shrunk = radii.copy()
        rad_shrunk[subset] *= 0.80
        x0[2::3] = rad_shrunk
        
        # Project to feasible bounds
        for i in range(N):
            r = max(1e-7, x0[3*i+2])
            x0[3*i] = np.clip(x0[3*i], r, 1.0-r)
            x0[3*i+1] = np.clip(x0[3*i+1], r, 1.0-r)
            x0[3*i+2] = r
            
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds_obj, constraints=cons,
                           options={'maxiter': 8000, 'ftol': 1e-13, 'disp': False})
            if not np.isnan(res.fun):
                c_tmp = np.column_stack((res.x[0::3], res.x[1::3]))
                r_new = solve_lp_radii(c_tmp)
                if np.sum(r_new) > best_sum:
                    best_sum = np.sum(r_new)
                    centers = c_tmp
                    radii = r_new
        except Exception:
            pass
            
    # Phase 5: Final Local Search Polish
    centers, radii, final_sum = local_search_centers(centers, cycles=8)
    if final_sum > best_sum:
        best_sum = final_sum
        best_centers = centers
        best_radii = radii
        
    # Final strict validation repair
    radii = best_radii
    centers = best_centers
    for _ in range(100):
        valid = True
        for i in range(N):
            if radii[i] < 0 or centers[i, 0] < radii[i] - 1e-9 or centers[i, 0] > 1.0 - radii[i] + 1e-9 or \
               centers[i, 1] < radii[i] - 1e-9 or centers[i, 1] > 1.0 - radii[i] + 1e-9:
                valid = False; break
        if valid:
            for i in range(N):
                for j in range(i + 1, N):
                    d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                    if d < radii[i] + radii[j] - 1e-9:
                        valid = False; break
                if not valid: break
        if valid: break
        radii *= 0.9998
        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)
        
    return centers, radii, float(np.sum(radii))
