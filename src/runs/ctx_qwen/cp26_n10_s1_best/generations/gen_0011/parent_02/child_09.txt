import numpy as np
from scipy.optimize import minimize, linprog
import warnings
warnings.filterwarnings('ignore')

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Returns all inequality constraints >= 0 (vectorized)."""
    xs, ys, rs = x[0::3], x[1::3], x[2::3]
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    c = np.concatenate([
        xs - rs, 1.0 - xs - rs,
        ys - rs, 1.0 - ys - rs
    ])
    
    # Overlap constraints: dist >= r_i + r_j
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    # hypot provides stable gradients at contact points
    c = np.concatenate([c, np.hypot(dx[TRIL_IDX], dy[TRIL_IDX]) - dr[TRIL_IDX]])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-7, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)])
    return b

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist)
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-7)
    except Exception:
        pass
    return np.full(n, 0.01)

def project_x(x0):
    """Project variables to strictly satisfy bounds."""
    x0 = x0.copy()
    for i in range(N):
        r = max(1e-7, x0[3 * i + 2])
        x0[3 * i] = np.clip(x0[3 * i], r, 1.0 - r)
        x0[3 * i + 1] = np.clip(x0[3 * i + 1], r, 1.0 - r)
        x0[3 * i + 2] = r
    return x0

def make_hex_init(rows, angle=0.0, scale_factor=1.0):
    """Generates a hexagonal lattice initialization with specific row counts."""
    pts = []
    y = 0.1
    row_idx = 0
    base_r = 0.09 * scale_factor
    
    for n_pts in rows:
        x = base_r if row_idx % 2 == 0 else 2.0 * base_r
        for _ in range(n_pts):
            if x + base_r <= 1.0:
                pts.append([x, y])
                x += 2.0 * base_r
        y += np.sqrt(3.0) * base_r
        row_idx += 1
        if len(pts) >= N:
            break
            
    while len(pts) < N:
        pts.append([np.random.uniform(0.1, 0.9), np.random.uniform(0.1, 0.9)])
        
    pts = np.array(pts[:N])
    
    if angle != 0.0:
        c_a, s_a = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c_a, -s_a], [s_a, c_a]]) + 0.5
        
    pts += np.random.uniform(-0.002, 0.002, pts.shape)
    pts = np.clip(pts, 0.02, 0.98)
    return pts

def make_force_init(seed):
    """Force-directed layout to spread points evenly and push to boundaries."""
    np.random.seed(seed)
    pts = np.random.uniform(0.15, 0.85, (N, 2))
    for step in range(250):
        f = np.zeros_like(pts)
        lr = 0.02 * (1.0 - step / 250.0)
        for i in range(N):
            for j in range(i + 1, N):
                dx = pts[j] - pts[i]
                d = np.linalg.norm(dx)
                if d < 0.3 and d > 1e-6:
                    rep = 0.005 / (d**2 + 0.001)
                    f[i] -= dx * rep / d
                    f[j] += dx * rep / d
            for dim in range(2):
                if pts[i, dim] < 0.1: f[i, dim] += 0.05
                elif pts[i, dim] > 0.9: f[i, dim] -= 0.05
        pts += f * lr
        pts = np.clip(pts, 0.02, 0.98)
    return pts

def polish_coordinate_descent(centers, radii, bounds, cons):
    """Fine-grained local search to squeeze out extra precision."""
    improved = True
    steps = 0
    while improved and steps < 50:
        improved = False
        steps += 1
        current_sum = np.sum(radii)
        for i in range(N):
            best_local_sum = current_sum
            best_local_c = centers[i].copy()
            best_local_r = radii.copy()
            
            # Try displacements in 8 directions + random
            deltas = np.array([
                [0.002, 0.0], [-0.002, 0.0], [0.0, 0.002], [0.0, -0.002],
                [0.001, 0.001], [-0.001, 0.001], [0.001, -0.001], [-0.001, -0.001],
                np.random.uniform(-0.003, 0.003, 2)
            ])
            
            for dx, dy in deltas:
                new_c = centers.copy()
                new_c[i] = np.clip(new_c[i] + [dx, dy], 0.005, 0.995)
                r_new = solve_lp_radii(new_c)
                s_new = np.sum(r_new)
                if s_new > best_local_sum + 1e-9:
                    best_local_sum = s_new
                    best_local_c = new_c[i].copy()
                    best_local_r = r_new.copy()
            
            if best_local_sum > current_sum + 1e-9:
                centers[i] = best_local_c
                radii = best_local_r.copy()
                current_sum = best_local_sum
                improved = True
    return centers, radii

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

                    best_sum = s_val
                    best_x = res.x.copy()

            pass
