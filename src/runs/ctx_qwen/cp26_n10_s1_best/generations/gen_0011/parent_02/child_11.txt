import warnings
warnings.filterwarnings('ignore')

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Returns all inequality constraints g(x) >= 0 (vectorized)."""
    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]

    c = np.concatenate([xs - rs, 1.0 - xs - rs, ys - rs, 1.0 - ys - rs])
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    c = np.concatenate([c, dx[TRIL_IDX]**2 + dy[TRIL_IDX]**2 - dr[TRIL_IDX]**2])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-7, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)])
    return b

def solve_lp_radii(centers):

    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A = np.zeros((num_ineq, n))
    b = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        x_val, y_val = centers[i]
        lim = min(x_val, 1.0 - x_val, y_val, 1.0 - y_val)
        A[idx, i] = 1.0
        b[idx] = max(0.0, lim)
        idx += 1

    for i in range(n):

            idx += 1

    except Exception:
        pass
    return np.full(n, 0.05)

def project_x(x0):
    """Project variables to strictly satisfy bounds."""
    x0 = x0.copy()
    for i in range(N):
        r = max(1e-7, x0[3 * i + 2])
        x0[3 * i] = np.clip(x0[3 * i], r, 1.0 - r)
        x0[3 * i + 1] = np.clip(x0[3 * i + 1], r, 1.0 - r)
        x0[3 * i + 2] = r
    return x0

def make_hex_init(rows, angle=0.0):
    """Generates a hexagonal lattice initialization with specific row counts."""
    pts = []
    y = 0.0
    row_idx = 0
    for n_pts in rows:
        x = 0.0 if row_idx % 2 == 0 else 1.0
        for _ in range(n_pts):
            pts.append([x, y])
            x += 2.0
        y += np.sqrt(3.0)
        row_idx += 1
    pts = np.array(pts[:N])

    w = np.max(pts[:, 0]) - np.min(pts[:, 0])
    h = np.max(pts[:, 1]) - np.min(pts[:, 1])
    scale = 0.8 / max(w, h)
    pts *= scale
    pts += 0.5 - pts.mean(axis=0)
    
    # Rotate around center

    # Add tiny noise to break exact symmetries
    pts += np.random.uniform(-0.0015, 0.0015, pts.shape)
    pts = np.clip(pts, 0.02, 0.98)
    return pts

def make_force_init(seed):

    return pts
