from scipy.optimize import minimize, linprog
import warnings

warnings.filterwarnings('ignore')

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Returns all inequality constraints >= 0 (vectorized)."""
    xs, ys, rs = x[0::3], x[1::3], x[2::3]
    
    # Boundary constraints
    c = np.concatenate([
        xs - rs, 1.0 - xs - rs,
        ys - rs, 1.0 - ys - rs
    ])
    
    # Overlap constraints: Euclidean distance >= sum of radii
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    c = np.concatenate([c, np.hypot(dx[TRIL_IDX], dy[TRIL_IDX]) - dr[TRIL_IDX]])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-7, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)])
    return b

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist)
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-8)
    except Exception:
        pass
    return np.full(n, 0.02)

def force_layout(centers, steps=400):
    """Force-directed relaxation to spread points evenly and push to boundaries."""
    pts = centers.copy()
    n = pts.shape[0]
    for step in range(steps):
        f = np.zeros_like(pts)
        lr = 0.015 * (1.0 - step / steps)
        for i in range(n):
            for j in range(i + 1, n):
                dx = pts[j] - pts[i]
                d = np.hypot(dx[0], dx[1])
                if d < 0.3 and d > 1e-5:
                    rep = 0.02 / (d**2 + 0.001)
                    f[i] -= dx * rep / d
                    f[j] += dx * rep / d
            for dim in range(2):
                if pts[i, dim] < 0.08: f[i, dim] += 0.08
                elif pts[i, dim] > 0.92: f[i, dim] -= 0.08
        pts += f * lr
        pts = np.clip(pts, 0.01, 0.99)
    return pts

def make_hex_init(r0, angle):
    """Generates a rotated hexagonal lattice initialization."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0

            x += 2.0 * r0

        row += 1

    pts = pts[mask]
    
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.1, 0.9, (1, 2))])
    return pts[:N]

def make_staggered_init():
    """Generates a row-based initialization adapted to square boundaries."""
    pts = []
    rows_config = [6, 5, 6, 5, 4]
    y_step = 0.19
    y = 0.105
    for r_count in rows_config:
        x_step = 0.9 / (r_count - 1) if r_count > 1 else 0.5
        for k in range(r_count):
            x = 0.05 + k * x_step
            pts.append([x, y])
            if len(pts) >= N:
                break
        y += y_step
        if len(pts) >= N:
            break
            
    while len(pts) < N:
        pts.append(np.random.uniform(0.1, 0.9, 2))
    return np.array(pts[:N])

def run_sa_refine(centers, iters=3000, init_temp=0.008):
    """Simulated Annealing on centers with LP radii evaluation."""
    pts = centers.copy()
    radii = solve_lp_radii(pts)
    best_sum = np.sum(radii)
    best_pts = pts.copy()
    best_radii = radii.copy()
    
    T = init_temp
    decay = 0.995
    step_size = 0.004
    
    for _ in range(iters):
        # Pick a random circle to perturb
        i = np.random.randint(N)
        new_pts = pts.copy()
        new_pts[i] += np.random.normal(0, step_size * (0.1 + T/0.001), 2)
        new_pts[i] = np.clip(new_pts[i], 0.02, 0.98)
        
        # Evaluate via LP
        new_radii = solve_lp_radii(new_pts)
        new_sum = np.sum(new_radii)

        # Accept if better, or with probability exp(delta/T)

            pts = new_pts
            radii = new_radii
            if new_sum > best_sum:
                best_sum = new_sum
                best_pts = pts.copy()
                best_radii = radii.copy()
                
        T *= decay
        step_size *= 0.999
        
    return best_pts, best_radii, best_sum
