import numpy as np
from scipy.optimize import minimize, linprog
import warnings
warnings.filterwarnings('ignore')

N = 26
TRIL_IDX = np.tril_indices(N, -1)
NUM_OVERLAP = N * (N - 1) // 2
NUM_CONSTRAINTS = 4 * N + NUM_OVERLAP

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def get_constraints(x):
    """Returns all inequality constraints >= 0 (vectorized)."""
    xs, ys, rs = x[0::3], x[1::3], x[2::3]
    c = np.empty(NUM_CONSTRAINTS)
    c[:N] = xs - rs
    c[N:2*N] = 1.0 - xs - rs
    c[2*N:3*N] = ys - rs
    c[3*N:4*N] = 1.0 - ys - rs
    
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dist = np.hypot(dx, dy)
    dr = rs[:, None] + rs[None, :]
    c[4*N:] = dist[TRIL_IDX] - dr[TRIL_IDX]
    return c

def get_jacobian(x):
    """Returns the Jacobian matrix of constraints."""
    xs, ys, rs = x[0::3], x[1::3], x[2::3]
    jac = np.zeros((NUM_CONSTRAINTS, 3*N))
    
    # Boundary constraints Jacobian
    jac[:N, 0::3] = 1.0
    jac[:N, 2::3] = -1.0
    jac[N:2*N, 0::3] = -1.0
    jac[N:2*N, 2::3] = -1.0
    jac[2*N:3*N, 1::3] = 1.0
    jac[2*N:3*N, 2::3] = -1.0
    jac[3*N:4*N, 1::3] = -1.0
    jac[3*N:4*N, 2::3] = -1.0
    
    # Overlap constraints Jacobian
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dist = np.hypot(dx, dy)
    
    idx = 4*N
    i_arr, j_arr = TRIL_IDX
    for i, j in zip(i_arr, j_arr):
        d = dist[i, j]
        if d > 1e-12:
            fx = (xs[i] - xs[j]) / d
            fy = (ys[i] - ys[j]) / d
            jac[idx, 3*i] = fx
            jac[idx, 3*i+1] = fy
            jac[idx, 3*i+2] = -1.0
            jac[idx, 3*j] = -fx
            jac[idx, 3*j+1] = -fy
            jac[idx, 3*j+2] = -1.0
        idx += 1
    return jac

cons_fun = {'type': 'ineq', 'fun': get_constraints, 'jac': get_jacobian}

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-7, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)])
    return b

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist)
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-7)
    except Exception:
        pass
    return np.full(n, 0.02)

def project_to_feasible(x):
    """Project variables to strictly satisfy bounds."""
    x = x.copy()
    for i in range(N):
        r = max(1e-7, x[3 * i + 2])
        x[3 * i] = np.clip(x[3 * i], r, 1.0 - r)
        x[3 * i + 1] = np.clip(x[3 * i + 1], r, 1.0 - r)
        x[3 * i + 2] = r
    return x

def force_layout(centers, steps=300):
    """Force-directed relaxation to spread points evenly and push to boundaries."""
    pts = centers.copy()
    n = pts.shape[0]
    for step in range(steps):
        f = np.zeros_like(pts)
        lr = 0.02 * (1.0 - step / steps)
        for i in range(n):
            for j in range(i + 1, n):
                dx = pts[j] - pts[i]
                d = np.hypot(dx[0], dx[1])
                if d < 0.3 and d > 1e-5:
                    rep = 0.015 / (d**2 + 0.001)
                    f[i] -= dx * rep / d
                    f[j] += dx * rep / d
            for dim in range(2):
                if pts[i, dim] < 0.08: f[i, dim] += 0.1
                elif pts[i, dim] > 0.92: f[i, dim] -= 0.1
        pts += f * lr
        pts = np.clip(pts, 0.01, 0.99)
    return pts

def make_hex_init(r0, angle):
    """Generates a rotated hexagonal lattice initialization."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0 and len(pts) < N + 10:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
    pts = np.array(pts[:N + 10])
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.1, 0.9, (1, 2))])
    return pts[:N]

def make_corner_init():
    """Generates an initialization focused on corners and edges."""
    pts = np.zeros((N, 2))
    r = 0.09
    pts[0] = [r, r]
    pts[1] = [1-r, r]
    pts[2] = [r, 1-r]
    pts[3] = [1-r, 1-r]
    pts[4] = [0.5, r]
    pts[5] = [0.5, 1-r]
    pts[6] = [r, 0.5]
    pts[7] = [1-r, 0.5]
    pts[8] = [0.5, 0.5]
    idx = 9
    y = r + 0.04
    row = 0
    while idx < N and y + r <= 1.0:
        x = r + 0.04 if row % 2 == 0 else r + 0.16
        while idx < N and x + r <= 1.0:
            pts[idx] = [x, y]
            idx += 1
            x += 0.20
        y += 0.13
        row += 1
    for i in range(idx, N):
        pts[i] = np.random.uniform(0.1, 0.9, 2)
    pts += np.random.uniform(-0.002, 0.002, pts.shape)
    return pts

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    bounds_obj = get_bounds()
    
    best_sum = -1.0
    best_x = None
    
    inits = []
    for ang in np.linspace(-0.35, 0.35, 13):
        inits.append(make_hex_init(0.09, ang))
    for _ in range(8):
        inits.append(make_corner_init())
    for s in range(12):
        np.random.seed(s + 200)
        pts = np.random.uniform(0.15, 0.85, (N, 2))
        inits.append(force_layout(pts, steps=300))
        
    for c_init in inits:
        x0 = np.zeros(3 * N)
        x0[0::3] = c_init[:, 0]
        x0[1::3] = c_init[:, 1]
        r_lp = solve_lp_radii(c_init)
        x0[2::3] = r_lp
        x0 = project_to_feasible(x0)
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds_obj,
                           constraints=cons_fun, options={'maxiter': 15000, 'ftol': 1e-13, 'disp': False})
            if not np.isnan(res.fun):
                c_tmp = np.column_stack((res.x[0::3], res.x[1::3]))
                r_new = solve_lp_radii(c_tmp)

                    curr_sum = np.sum(r_new)
                    x_p = res.x.copy()
                    x_p[2::3] = r_new
                    if curr_sum > best_sum and np.min(get_constraints(x_p)) >= -1e-6:
                        best_sum = curr_sum
                        best_x = x_p.copy()
        except Exception:
