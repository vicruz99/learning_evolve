N = 26
TRIL_IDX = np.tril_indices(N, -1)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Returns all inequality constraints >= 0 (vectorized)."""
    xs, ys, rs = x[0::3], x[1::3], x[2::3]
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    c = np.concatenate([
        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs
    ])
    
    # Overlap constraints: dist >= r_i + r_j
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    c = np.concatenate([c, np.hypot(dx[TRIL_IDX], dy[TRIL_IDX]) - dr[TRIL_IDX]])
    return c

def get_bounds():

def solve_lp_radii(centers, margin=0.0):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros((num_ineq, n))
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y) - margin
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim - margin)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j) - margin
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist - margin)
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-7)
    except Exception:
        pass
    return np.full(n, 0.05)

def project_to_feasible(x):

    return x

def force_layout_init(seed, steps=500):
    """Force-directed layout with adaptive repulsion."""
    np.random.seed(seed)
    pts = np.random.uniform(0.15, 0.85, (N, 2))
    
    for step in range(steps):
        f = np.zeros_like(pts)
        lr = 0.02 * (1.0 - step / steps) ** 0.5
        
        for i in range(N):
            for j in range(i + 1, N):
                dx = pts[j] - pts[i]
                d = np.hypot(dx[0], dx[1])
                if d < 0.35 and d > 1e-6:
                    target = 0.25
                    rep = 0.015 / (d * d + 0.0005) * (target - d) / target
                    f[i] -= dx * rep / d
                    f[j] += dx * rep / d
                    
            # Push toward boundaries
            for dim in range(2):
                if pts[i, dim] < 0.12:
                    f[i, dim] += 0.06
                elif pts[i, dim] > 0.88:
                    f[i, dim] -= 0.06
                else:
                    # Gentle push to boundaries
                    if pts[i, dim] < 0.5:
                        f[i, dim] += 0.01
                    else:
                        f[i, dim] -= 0.01
        
        pts += f * lr
        pts = np.clip(pts, 0.01, 0.99)
    
    return pts
