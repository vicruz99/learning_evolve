import numpy as np
from scipy.optimize import minimize, linprog, differential_evolution
import warnings
warnings.filterwarnings('ignore')

N = 26

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist)
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-9)
    except Exception:
        pass
    return np.full(n, 0.05)

def eval_sum_radii(c_flat):
    """Objective for center optimization: negative sum of LP-optimal radii."""
    centers = np.clip(c_flat.reshape(N, 2), 0.0, 1.0)
    radii = solve_lp_radii(centers)
    return -np.sum(radii)

def make_hex_init(r0, angle=0.0):
    """Generates a hexagonal lattice initialization with optional rotation."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 5:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0 and len(pts) < N + 5:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
    pts = np.array(pts[:N + 5])
    
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
        
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.1, 0.9, (1, 2))])
    return pts[:N]

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    bounds_centers = [(0.0, 1.0)] * (2 * N)
    
    best_c_flat = None
    best_score = np.inf
    
    # Phase 1: Structured Hexagonal Starts with Local Refinement
    # Hex lattices are dense packings; slight rotations often improve boundary fitting.
    for r0 in [0.08, 0.085, 0.09, 0.095, 0.10]:
        for ang in np.linspace(-0.25, 0.25, 11):
            pts = make_hex_init(r0, ang)
            c0 = pts.flatten()
            try:
                res = minimize(eval_sum_radii, c0, method='Powell', 
                               bounds=bounds_centers, options={'maxiter': 1500, 'ftol': 1e-12})
                if res.fun < best_score:
                    best_score = res.fun
                    best_c_flat = res.x.copy()
            except Exception:
                pass
                
    # Phase 2: Global Search via Differential Evolution
    # Robustly explores the 52D center space, escaping local minima.
    for de_seed in [42, 123, 456]:
        try:
            de_res = differential_evolution(eval_sum_radii, bounds_centers, 
                                            popsize=20, maxiter=120, tol=1e-9, 
                                            mutation=(0.5, 1.5), recombination=0.9, 
                                            seed=de_seed, polish=False)
            if de_res.fun < best_score:
                best_score = de_res.fun
                best_c_flat = de_res.x.copy()
        except Exception:
            pass
            
    # Phase 3: Final Local Polish on Best Centers
    if best_c_flat is not None:
        try:
            res_fin = minimize(eval_sum_radii, best_c_flat, method='Powell',
                               bounds=bounds_centers, options={'maxiter': 3000, 'ftol': 1e-13})
            best_c_flat = res_fin.x
        except Exception:
            pass
            
    best_centers = np.clip(best_c_flat.reshape(N, 2), 0.0, 1.0)
    best_radii = solve_lp_radii(best_centers)
    
    # Phase 4: Joint SLSQP Refinement
    # Tightens constraints and handles non-smoothness of LP objective at contact points
    x0 = np.zeros(3 * N)
    x0[0::3] = best_centers[:, 0]
    x0[1::3] = best_centers[:, 1]
    x0[2::3] = best_radii
    
    def obj_full(x):
        return -np.sum(x[2::3])
        
    def cons_full(x):
        xs, ys, rs = x[0::3], x[1::3], x[2::3]
        c = np.concatenate([xs - rs, 1.0 - xs - rs, ys - rs, 1.0 - ys - rs])
        dx = xs[:, None] - xs[None, :]
        dy = ys[:, None] - ys[None, :]
        dr = rs[:, None] + rs[None, :]
        i, j = np.tril_indices(N, -1)
        c = np.concatenate([c, np.hypot(dx[i, j], dy[i, j]) - dr[i, j]])
        return c

    try:
        res_slsqp = minimize(obj_full, x0, method='SLSQP', 
                             bounds=[(0,1), (0,1), (1e-9, 0.5)]*N,
                             constraints={'type': 'ineq', 'fun': cons_full},
                             options={'maxiter': 15000, 'ftol': 1e-14})
        if res_slsqp.success:
            best_centers = np.column_stack((res_slsqp.x[0::3], res_slsqp.x[1::3]))
            best_radii = res_slsqp.x[2::3]
    except Exception:
        pass
        
    # Validation & Minimal Repair
    centers = best_centers
    radii = best_radii
    for _ in range(50):
        valid = True
        for i in range(N):
            if radii[i] < 0 or centers[i,0] < radii[i] - 1e-9 or centers[i,0] > 1 - radii[i] + 1e-9 or \
               centers[i,1] < radii[i] - 1e-9 or centers[i,1] > 1 - radii[i] + 1e-9:
                valid = False
                break
        if valid:
            for i in range(N):
                for j in range(i+1, N):
                    if np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1]) < radii[i]+radii[j] - 1e-9:
                        valid = False
                        break
                if not valid:
                    break
        if valid:
            break
            
        radii *= 0.9998
        centers[:,0] = np.clip(centers[:,0], radii, 1-radii)
        centers[:,1] = np.clip(centers[:,1], radii, 1-radii)
        
    return centers, radii, float(np.sum(radii))
