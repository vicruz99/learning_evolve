The dual variables (shadow prices) of the overlap and boundary constraints give the exact gradient direction for moving centers to increase the sum of radii.
2. **Gradient Ascent on Centers**: We use the extracted duals to compute a force vector for each circle. 3. **Adaptive Line Search & Deflation**: Each gradient step is accepted only if it improves the LP-optimal sum of radii. If not, the learning rate decays. import numpy as np
from scipy.optimize import linprog, minimize
import warnings

warnings.filterwarnings('ignore')

N = 26

def solve_lp_radii_and_duals(centers):
    """Solves LP for radii given fixed centers and returns radii and dual variables."""
    n = N
    c_obj = -np.ones(n)
    n_ineq = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((n_ineq, n))
    b_ub = np.zeros(n_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= x, r_i <= 1-x, r_i <= y, r_i <= 1-y
    for i in range(n):
        A_ub[idx, i] = 1.0; b_ub[idx] = centers[i, 0]; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - centers[i, 0]; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = centers[i, 1]; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - centers[i, 1]; idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i,j)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = d
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            # Duals for min c^T x s.t. Ax <= b are typically <= 0. 
            # We take -duals to get positive "pressure" values for active constraints.
            duals = np.maximum(-res.ineqlin.marginals, 0.0)
            return res.x, duals
    except Exception:
        pass
    return None, None

def compute_gradients(centers, duals):
    """Computes gradient of sum(radii) w.r.t centers using LP duals."""
    n = N
    grad = np.zeros((n, 2))
    if duals is None: return grad
    
    # Boundary contributions: push away from active walls
    for i in range(n):
        grad[i, 0] += duals[i] - duals[n + i]
        grad[i, 1] += duals[2*n + i] - duals[3*n + i]
        
    # Overlap contributions: push apart constrained pairs
    idx = 4 * n
    for i in range(n):
        for j in range(i + 1, n):
            lam = duals[idx]
            if lam > 1e-9:
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                d = np.hypot(dx, dy)
                if d > 1e-8:
                    inv_d = 1.0 / d
                    grad[i, 0] += lam * dx * inv_d
                    grad[i, 1] += lam * dy * inv_d
                    grad[j, 0] -= lam * dx * inv_d
                    grad[j, 1] -= lam * dy * inv_d
            idx += 1
    return grad

def gradient_ascent_optimize(centers_init, max_iter=600, init_lr=0.08):
    """Runs gradient ascent on centers using LP dual sensitivities."""
    centers = centers_init.copy()
    r_curr, _ = solve_lp_radii_and_duals(centers)
    if r_curr is None: return centers, 0.0
    best_sum = np.sum(r_curr)
    
    lr = init_lr
    for _ in range(max_iter):
        r, duals = solve_lp_radii_and_duals(centers)
        if r is None: break
        curr_sum = np.sum(r)
        if curr_sum > best_sum: best_sum = curr_sum
        
        grad = compute_gradients(centers, duals)
        if np.linalg.norm(grad) < 1e-8: break
        
        step = lr * grad
        new_c = centers + step
        new_c = np.clip(new_c, 1e-5, 1.0 - 1e-5)
        
        r_new, _ = solve_lp_radii_and_duals(new_c)
        if r_new is not None:
            new_sum = np.sum(r_new)
            if new_sum > best_sum:
                centers = new_c
                best_sum = new_sum
            else:
                lr *= 0.85
        else:
            lr *= 0.85
            
        if lr < 1e-6: break
    return centers

def make_hex_init(r0, angle):
    """Generates a rotated hexagonal lattice initialization."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0 and len(pts) < N + 10:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
    pts = np.array(pts[:N + 10])
    
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
        
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    if len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.15, 0.85, (N - len(pts), 2))])
    pts += np.random.uniform(-0.002, 0.002, pts.shape)
    return pts[:N]

def force_spread_init(seed):
    """Generates a force-spread initialization."""
    np.random.seed(seed + 1000)
    pts = np.random.uniform(0.15, 0.85, (N, 2))
    for _ in range(300):
        f = np.zeros_like(pts)
        for i in range(N):
            for j in range(i + 1, N):
                dx = pts[j] - pts[i]
                d = np.hypot(dx[0], dx[1])
                if d < 0.35 and d > 1e-5:
                    rep = 0.01 / (d**2 + 0.001)
                    f[i] -= dx * rep / d
                    f[j] += dx * rep / d
            for dim in range(2):
                if pts[i, dim] < 0.1: f[i, dim] += 0.05
                elif pts[i, dim] > 0.9: f[i, dim] -= 0.05
        pts += f * 0.04
        pts = np.clip(pts, 0.02, 0.98)
    return pts

def objective_slsqp(x):
    return -np.sum(x[2::3])

def constraints_slsqp(x):
    xs, ys, rs = x[0::3], x[1::3], x[2::3]
    c = np.concatenate([
        xs - rs, 1.0 - xs - rs, ys - rs, 1.0 - ys - rs
    ])
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    i, j = np.tril_indices(N, -1)
    c = np.concatenate([c, np.hypot(dx[i,j], dy[i,j]) - dr[i,j]])
    return c

def get_bounds_slsqp():
    return [(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)] * N

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    inits = []
    # Hexagonal lattices with varied parameters
    for r0 in [0.085, 0.090, 0.095, 0.100, 0.105]:
        for ang in np.linspace(-0.35, 0.35, 9):
            inits.append(make_hex_init(r0, ang))
    # Force-spread random starts
    for s in range(8):
        inits.append(force_spread_init(s))
    # Corner/Edge biased
    for s in range(5):
        np.random.seed(s + 2000)
        pts = np.random.uniform(0.05, 0.95, (N, 2))
        # Push some to corners
        pts[0] = [0.1, 0.1]; pts[1] = [0.9, 0.1]
        pts[2] = [0.1, 0.9]; pts[3] = [0.9, 0.9]
        inits.append(pts)

    # Phase 1: Gradient Ascent Optimization
    for c_init in inits:
        # Run gradient ascent
        c_opt = gradient_ascent_optimize(c_init, max_iter=500, init_lr=0.06)
        r_opt, _ = solve_lp_radii_and_duals(c_opt)
        if r_opt is not None:
            s = np.sum(r_opt)
            if s > best_sum:
                best_sum = s
                best_centers = c_opt.copy()
                best_radii = r_opt.copy()
                
    # Phase 2: Deflation & Restart to Escape Local Minima
    if best_centers is not None:
        for cyc in range(40):
            # Shrink towards center to break rigid contacts
            pert_c = 0.96 * best_centers + 0.04 * np.array([0.5, 0.5])
            pert_c += np.random.normal(0, 0.002 * (0.9 ** cyc), pert_c.shape)
            pert_c = np.clip(pert_c, 0.02, 0.98)
            
            c_new = gradient_ascent_optimize(pert_c, max_iter=300, init_lr=0.04)
            r_new, _ = solve_lp_radii_and_duals(c_new)
            if r_new is not None:
                s = np.sum(r_new)
                if s > best_sum:
                    best_sum = s
                    best_centers = c_new.copy()
                    best_radii = r_new.copy()

    # Phase 3: SLSQP Polish
    if best_centers is not None:
        x0 = np.zeros(3 * N)
        x0[0::3] = best_centers[:, 0]
        x0[1::3] = best_centers[:, 1]
        x0[2::3] = best_radii
        
        cons = {'type': 'ineq', 'fun': constraints_slsqp}
        bounds = get_bounds_slsqp()
        try:
            res = minimize(objective_slsqp, x0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 15000, 'ftol': 1e-14, 'disp': False})
            if not np.isnan(res.fun):
                curr_sum = -res.fun
                # Final LP snap for strict feasibility
                c_tmp = np.column_stack((res.x[0::3], res.x[1::3]))
                r_tmp, _ = solve_lp_radii_and_duals(c_tmp)
                if r_tmp is not None:
                    curr_sum = np.sum(r_tmp)
                    if curr_sum > best_sum:
                        best_sum = curr_sum
                        best_centers = c_tmp.copy()
                        best_radii = r_tmp.copy()
                elif curr_sum > best_sum:
                    best_sum = curr_sum
                    best_centers = c_tmp.copy()
                    best_radii = res.x[2::3].copy()
        except Exception:
            pass

    # Fallback
    if best_centers is None:
        best_centers = np.tile(np.linspace(0.1, 0.9, 5), 6)[:N, None]
        best_centers = np.hstack([best_centers, np.repeat(np.linspace(0.1, 0.9, 6), 5)[:N, None]])
        best_radii = np.full(N, 0.06)
        best_sum = np.sum(best_radii)

    # Final strict validation repair
    for _ in range(100):
        valid = True
        for i in range(N):
            if best_radii[i] < 0: valid = False; break
            if best_centers[i, 0] - best_radii[i] < -1e-9 or best_centers[i, 0] + best_radii[i] > 1.0 + 1e-9: valid = False; break
            if best_centers[i, 1] - best_radii[i] < -1e-9 or best_centers[i, 1] + best_radii[i] > 1.0 + 1e-9: valid = False; break
        if valid:
            for i in range(N):
                for j in range(i + 1, N):
                    d = np.hypot(best_centers[i, 0] - best_centers[j, 0], best_centers[i, 1] - best_centers[j, 1])
                    if d < best_radii[i] + best_radii[j] - 1e-9: valid = False; break
                if not valid: break
        if valid: break
        best_radii *= 0.9998
        best_centers[:, 0] = np.clip(best_centers[:, 0], best_radii, 1.0 - best_radii)
        best_centers[:, 1] = np.clip(best_centers[:, 1], best_radii, 1.0 - best_radii)
        
    return best_centers, best_radii, float(np.sum(best_radii))
