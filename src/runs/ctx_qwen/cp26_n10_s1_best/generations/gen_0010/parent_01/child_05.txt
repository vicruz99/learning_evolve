To improve upon the current sum of radii (~2.634), we need to break out of the local optimum that hexagonal-based initializations consistently fall into. The strategy involves:

1. **Constraint Formulation**: Use squared distance constraints `(dx^2 + dy^2 >= (r_i + r_j)^2)` instead of `hypot`. This removes the `1/distance` singularity in gradients when circles are nearly touching, leading to more stable Newton/SLSQP steps.
2. **Topology Breaking**: The current packing is likely stuck in a rigid distorted hexagonal lattice. We will aggressively break this by:
   - Randomly selecting a subset of circles and shrinking them significantly (deflation), allowing the optimizer to push them into different gaps.
   - Applying random global rotations and shears to realign circles with the square boundaries.
   - Using a "coordinate-wise" polish step: iterate through circles and optimize their `(x, y, r)` while freezing others, which can squeeze out microscopic improvements.
3. **Hybrid Optimization Loop**: Alternate between SLSQP (for global constraint satisfaction) and Linear Programming (for exact radius expansion given fixed centers). This decouples the difficult non-linear geometry from the linear radius scaling.
4. **Diverse Initializations**: Generate not just hex grids, but also specific row-count patterns (e.g., 6-5-5-5-5, 5-6-6-5-4) and force-directed layouts that naturally push circles to boundaries.
5. **Robust Validation**: A strict final repair loop to guarantee compliance with the `1e-12` tolerance in the validator.

This approach systematically explores different contact networks and boundary alignments, targeting the theoretical limit for 26 circles.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Returns all inequality constraints >= 0 (vectorized)."""
    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]
    
    # Boundary constraints
    c = np.concatenate([
        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs
    ])
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    # Squared form avoids sqrt singularity in gradients near contact
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    c = np.concatenate([c, dx[TRIL_IDX]**2 + dy[TRIL_IDX]**2 - dr[TRIL_IDX]**2])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-7, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)])
    return b

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist)
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-7)
    except Exception:
        pass
    return np.full(n, 0.05)

def project_to_feasible(x):
    """Project variables to strictly satisfy bounds."""
    x = x.copy()
    for i in range(N):
        r = max(1e-6, x[3 * i + 2])
        x[3 * i] = np.clip(x[3 * i], r, 1.0 - r)
        x[3 * i + 1] = np.clip(x[3 * i + 1], r, 1.0 - r)
        x[3 * i + 2] = r
    return x

def make_hex_init(r0, angle=0.0):
    """Generates a hexagonal lattice initialization with optional rotation."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0 and len(pts) < N + 10:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        
    pts = np.array(pts[:N + 10])
    
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
        
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.2, 0.8, (1, 2))])
    return pts[:N]

def make_row_init(row_counts, angle=0.0):
    """Generates a row-based initialization with specified circle counts per row."""
    pts = []
    r_est = 0.09
    y = r_est
    for rc in row_counts:
        x_start = r_est
        for _ in range(rc):
            if len(pts) >= N: break
            pts.append([x_start, y])
            x_start += 2.0 * r_est
        y += np.sqrt(3.0) * r_est * 0.95 # Slightly compressed rows
        
    while len(pts) < N:
        pts.append([np.random.uniform(0.1, 0.9), np.random.uniform(0.1, 0.9)])
        
    pts = np.array(pts[:N])
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
    return pts

def force_layout(centers, steps=150):
    """Force-directed layout to spread points and push to boundaries."""
    pts = centers.copy()
    rs = np.full(N, 0.07)
    for step in range(steps):
        f = np.zeros_like(pts)
        lr = 0.015 * (1.0 - step / steps)
        for i in range(N):
            for j in range(i + 1, N):
                dx = pts[j] - pts[i]
                d = np.hypot(dx[0], dx[1])
                if d < 0.3 and d > 1e-5:
                    rep = 0.012 / (d**2 + 0.001)
                    f[i] -= dx * rep / d
                    f[j] += dx * rep / d
            for dim in range(2):
                if pts[i, dim] < rs[i] + 0.02: f[i, dim] += 0.06
                elif pts[i, dim] > 1.0 - rs[i] - 0.02: f[i, dim] -= 0.06
        pts += f * lr
        pts = np.clip(pts, 0.01, 0.99)
    return pts

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -1.0
    best_x = None
    
    # Phase 1: Diverse Initial Configurations
    inits = []
    # Hexagonal grids
    for r0 in [0.085, 0.090, 0.095, 0.100]:
        for ang in np.linspace(-0.35, 0.35, 13):
            inits.append(make_hex_init(r0, ang))
            
    # Specific row patterns to break symmetry
    for pattern in [[6,5,5,5,5], [5,6,5,5,5], [5,5,6,5,5], [6,6,5,5,4], [5,6,6,5,4]]:
        for ang in np.linspace(-0.2, 0.2, 7):
            inits.append(make_row_init(pattern, ang))
            
    # Force-directed random layouts
    for s in range(12):
        np.random.seed(s + 1000)
        pts = np.random.uniform(0.2, 0.8, (N, 2))
        inits.append(force_layout(pts))

    # Phase 2: Multi-start Optimization
    for pts in inits:
        x0 = np.zeros(3 * N)
        x0[0::3] = pts[:, 0]
        x0[1::3] = pts[:, 1]
        
        # Initialize radii via LP for maximum head-start
        r_lp = solve_lp_radii(pts)
        x0[2::3] = np.maximum(r_lp * 0.99, 0.01)
        x0 = project_to_feasible(x0)
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 20000, 'ftol': 1e-14, 'disp': False})
            if not np.isnan(res.fun):
                c_tmp = np.column_stack((res.x[0::3], res.x[1::3]))
                r_new = solve_lp_radii(c_tmp)
                if r_new is not None:
                    curr_sum = np.sum(r_new)
                    if curr_sum > best_sum:
                        best_sum = curr_sum
                        best_x = np.zeros(3 * N)
                        best_x[0::3] = c_tmp[:, 0]
                        best_x[1::3] = c_tmp[:, 1]
                        best_x[2::3] = r_new
        except Exception:
            pass

    # Phase 3: Aggressive Topology Search & Deflation
    if best_x is not None:
        for cyc in range(120):
            x0 = best_x.copy()
            
            # Cooling perturbation
            noise_scale = 0.0025 * (0.93 ** cyc)
            x0[0::3] += np.random.normal(0, noise_scale, N)
            x0[1::3] += np.random.normal(0, noise_scale, N)
            
            # Deflate radii to create slack for repositioning
            shrink_factor = 0.80 + 0.15 * (cyc / 120.0)
            x0[2::3] *= shrink_factor
            
            # Aggressively deflate a random subset to break rigid contact networks
            subset = np.random.choice(N, size=max(5, N // 3), replace=False)
            x0[subset * 3 + 2] *= 0.65
            
            # Optional: random global rotation to find better boundary alignments
            if cyc % 10 == 0:
                ang = np.random.uniform(-0.15, 0.15)
                c, s = np.cos(ang), np.sin(ang)
                centers_rot = np.column_stack((x0[0::3], x0[1::3]))
                centers_rot = (centers_rot - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
                x0[0::3] = centers_rot[:, 0]
                x0[1::3] = centers_rot[:, 1]
            
            x0 = project_to_feasible(x0)
            
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 15000, 'ftol': 1e-14, 'disp': False})
                if not np.isnan(res.fun):
                    c_tmp = np.column_stack((res.x[0::3], res.x[1::3]))
                    r_new = solve_lp_radii(c_tmp)
                    if r_new is not None:
                        x_p = res.x.copy()
                        x_p[2::3] = r_new * 0.999
                        x_p = project_to_feasible(x_p)
                        
                        s_val = np.sum(x_p[2::3])
                        vals = constraints(x_p)
                        if np.min(vals) >= -1e-5 and s_val > best_sum:
                            best_sum = s_val
                            best_x = x_p.copy()
            except Exception:
                pass
                
        # Phase 4: Coordinate-wise Polish
        # Optimize each circle's position/radius sequentially while others are fixed
        x_pol = best_x.copy()
        for _ in range(5): # Multiple passes
            for i in range(N):
                # Freeze others
                fixed_x = x_pol.copy()
                # Define local objective and constraints
                def local_obj(vars):
                    tmp = fixed_x.copy()
                    tmp[3*i] = vars[0]
                    tmp[3*i+1] = vars[1]
                    tmp[3*i+2] = vars[2]
                    return -vars[2] # Maximize this circle's radius
                
                def local_cons(vars):
                    tmp = fixed_x.copy()
                    tmp[3*i] = vars[0]
                    tmp[3*i+1] = vars[1]
                    tmp[3*i+2] = vars[2]
                    return constraints(tmp)
                
                local_bounds = [(fixed_x[3*i], fixed_x[3*i]), 
                                (fixed_x[3*i+1], fixed_x[3*i+1]),
                                (fixed_x[3*i+1], fixed_x[3*i+1])] # Placeholder, corrected below
                # Actually, just optimize x, y, r for circle i
                lb = max(1e-6, fixed_x[3*i+2] * 0.5)
                local_bounds = [(lb, 1.0-lb), (lb, 1.0-lb), (lb, 0.5)]
                
                try:
                    res = minimize(local_obj, [x_pol[3*i], x_pol[3*i+1], x_pol[3*i+2]], 
                                   method='SLSQP', bounds=local_bounds, 
                                   constraints={'type': 'ineq', 'fun': local_cons},
                                   options={'maxiter': 2000, 'ftol': 1e-13})
                    if not np.isnan(res.fun):
                        x_pol[3*i] = res.x[0]
                        x_pol[3*i+1] = res.x[1]
                        x_pol[3*i+2] = res.x[2]
                except Exception:
                    pass
            # LP snap after coordinate-wise pass
            c_tmp = np.column_stack((x_pol[0::3], x_pol[1::3]))
            r_new = solve_lp_radii(c_tmp)
            if r_new is not None:
                x_pol[2::3] = r_new
                x_pol = project_to_feasible(x_pol)

    radii = best_x[2::3].copy()

    for _ in range(100):
        valid = True
        for i in range(N):
            if radii[i] < 0 or centers[i, 0] < radii[i] - 1e-9 or centers[i, 0] > 1.0 - radii[i] + 1e-9 or \

                        break

        # Minimal shrinkage to recover strict feasibility
        radii *= 0.9998
        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)
        
    return centers, radii, float(np.sum(radii))
