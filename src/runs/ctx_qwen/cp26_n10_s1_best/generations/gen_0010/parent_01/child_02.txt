To push the sum of radii beyond the current plateau of ~2.634, we need a more robust and diverse optimization strategy. The key insights are:
1. **Constraint Formulation**: Using squared distance constraints (`dist² ≥ (r_i + r_j)²`) avoids square-root singularities in gradients and is numerically more stable for SLSQP.
2. **Alternating LP & Nonlinear Optimization**: Fixing centers and solving for optimal radii via Linear Programming (LP) provides exact radii for any configuration. Alternating this with SLSQP (which optimizes centers and radii simultaneously) creates a powerful hybrid solver.
3. **Diverse Topological Starts**: Hexagonal lattices are optimal for infinite planes, but boundary effects for N=26 favor specific row patterns (e.g., 6-5-6-5-4, 5-6-5-6-4). We generate these explicitly with varied rotations, scales, and shifts to cover different contact networks.
4. **Deflation-Perturbation Refinement**: To escape local minima, we systematically deflate a random subset of radii, perturb centers with cooling noise, and re-optimize. This breaks rigid contact graphs and allows circles to "roll" into deeper energy wells.
5. **Strict Feasibility Projection**: After every optimization step, we project variables to strictly satisfy boundary constraints, preventing SLSQP from stalling on infeasible boundaries.
6. **High-Precision Polish**: The final configuration undergoes multiple high-iteration SLSQP runs with tight tolerances and a final LP squeeze to extract every fraction of a radius.

import numpy as np
from scipy.optimize import minimize, linprog
import warnings

warnings.filterwarnings('ignore')

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Returns all inequality constraints g(x) >= 0."""
    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    c = np.concatenate([
        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs
    ])
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    c = np.concatenate([c, dx[TRIL_IDX]**2 + dy[TRIL_IDX]**2 - dr[TRIL_IDX]**2])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-7, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)])
    return b

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A = np.zeros((num_ineq, n))
    b = np.zeros(num_ineq)
    
    idx = 0
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A[idx, i] = 1.0
        b[idx] = max(0.0, lim)
        idx += 1
        
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = max(0.0, d)
            idx += 1
            
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0.0, None)] * n, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-7)
    except Exception:
        pass
    return np.full(n, 0.05)

def project_to_feasible(x):
    """Project variables to strictly satisfy bounds."""
    x = x.copy()
    for i in range(N):
        r = max(1e-6, x[3 * i + 2])
        x[3 * i] = np.clip(x[3 * i], r, 1.0 - r)
        x[3 * i + 1] = np.clip(x[3 * i + 1], r, 1.0 - r)
        x[3 * i + 2] = r
    return x

def make_hex_rows(rows, angle=0.0, scale=1.0):
    """Generates a hexagonal lattice with specific row counts."""
    pts = []
    r0 = 0.09 * scale
    y = r0
    for r_idx, cnt in enumerate(rows):
        x = r0 if r_idx % 2 == 0 else 2.0 * r0
        for _ in range(cnt):
            if len(pts) >= N:
                break
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        
    pts = np.array(pts[:N])
    
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
        
    pts = np.clip(pts, 0.02, 0.98)
    if len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.15, 0.85, (N - len(pts), 2))])
        
    pts += np.random.uniform(-0.002, 0.002, pts.shape)
    return pts[:N]

def make_corner_fill():
    """Generates an initialization focused on corners and edges."""
    pts = []
    r = 0.088
    pts.append([r, r])
    pts.append([1-r, r])
    pts.append([r, 1-r])
    pts.append([1-r, 1-r])
    
    y = r
    for i in range(4, N):
        x = r + (i-4) * 0.14
        if x > 1-r:
            y += 0.14
            x = r
        pts.append([x, y])
        
    pts = np.array(pts[:N])
    pts += np.random.uniform(-0.003, 0.003, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -1.0
    best_x = None
    
    # Phase 1: Generate highly diverse initial configurations
    starts = []
    row_patterns = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [6, 6, 5, 5, 4],
        [4, 5, 6, 6, 5], [7, 6, 6, 7], [5, 7, 5, 6, 3]
    ]
    for pat in row_patterns:
        total = sum(pat)
        if total < N:
            pat = pat + [N - total]
        elif total > N:
            pat = pat[:N]
        for ang in np.linspace(-0.35, 0.35, 9):
            for sc in [0.92, 0.96, 1.00, 1.04]:
                starts.append(make_hex_rows(pat, ang, sc))
                
    for _ in range(8):
        starts.append(make_corner_fill())
        
    for s in range(12):
        np.random.seed(s + 800)
        pts = np.random.uniform(0.15, 0.85, (N, 2))
        starts.append(pts)

    # Phase 2: Multi-start Optimization with LP-SLSQP Alternation
    for pts in starts:
        x0 = np.zeros(3 * N)
        x0[0::3] = pts[:, 0]
        x0[1::3] = pts[:, 1]
        
        r_lp = solve_lp_radii(pts)
        x0[2::3] = np.maximum(r_lp * 0.992, 0.01)
        x0 = project_to_feasible(x0)
        x0 += np.random.normal(0, 5e-4, x0.shape)
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 25000, 'ftol': 1e-14, 'disp': False})
            if not np.isnan(res.fun):
                c_tmp = np.column_stack((res.x[0::3], res.x[1::3]))
                r_new = solve_lp_radii(c_tmp)
                if r_new is not None:
                    s_val = np.sum(r_new)
                    if s_val > best_sum:
                        best_sum = s_val
                        best_x = np.zeros(3 * N)
                        best_x[0::3] = c_tmp[:, 0]
                        best_x[1::3] = c_tmp[:, 1]
                        best_x[2::3] = r_new
        except Exception:
            pass

    # Phase 3: Iterative Deflation & Perturbation to Escape Local Minima
    if best_x is not None:
        for cyc in range(70):
            x0 = best_x.copy()
            
            # Gradual radius recovery
            shr = 0.78 + 0.22 * (cyc / 70.0)
            x0[2::3] *= shr
            
            # Cooling noise
            noise = 0.0025 * (0.91 ** cyc)
            x0[0::3] += np.random.normal(0, noise, N)
            x0[1::3] += np.random.normal(0, noise, N)
            
            # Break rigid contact networks by deflating a subset
            subset = np.random.choice(N, size=max(5, N // 3), replace=False)
            x0[subset * 3 + 2] *= 0.65
            
            x0 = project_to_feasible(x0)
            
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 18000, 'ftol': 1e-14, 'disp': False})
                if not np.isnan(res.fun):
                    c_tmp = np.column_stack((res.x[0::3], res.x[1::3]))
                    r_new = solve_lp_radii(c_tmp)
                    if r_new is not None:
                        x_p = res.x.copy()
                        x_p[2::3] = np.maximum(r_new * 0.999, 0.01)
                        x_p = project_to_feasible(x_p)
                        
                        s_val = np.sum(x_p[2::3])
                        if s_val > best_sum:
                            best_sum = s_val
                            best_x = x_p.copy()
            except Exception:
                pass

    # Phase 4: High-Precision Polish
    if best_x is not None:
        for _ in range(6):
            x_pol = best_x.copy()
            x_pol[2::3] *= 0.995
            x_pol = project_to_feasible(x_pol)
            try:
                res = minimize(objective, x_pol, method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 30000, 'ftol': 1e-15, 'disp': False})
                if not np.isnan(res.fun):
                    c_pol = np.column_stack((res.x[0::3], res.x[1::3]))
                    r_pol = solve_lp_radii(c_pol)
                    if r_pol is not None:
                        s_pol = np.sum(r_pol)
                        if s_pol > best_sum:
                            best_sum = s_pol
                            best_x = np.zeros(3 * N)
                            best_x[0::3] = c_pol[:, 0]
                            best_x[1::3] = c_pol[:, 1]
                            best_x[2::3] = r_pol
            except Exception:
                pass

    # Fallback (should not be reached)
    if best_x is None:
        best_x = np.zeros(3 * N)
        best_x[0::3] = np.tile(np.linspace(0.1, 0.9, 5), 6)[:N]
        best_x[1::3] = np.repeat(np.linspace(0.1, 0.9, 6), 5)[:N]
        best_x[2::3] = 0.06
        best_sum = np.sum(best_x[2::3])
        
    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3].copy()
    
    # Final strict validation repair against 1e-12 tolerance
    for _ in range(100):
        valid = True
        for i in range(N):
            if radii[i] < 0: valid = False; break
            if centers[i, 0] < radii[i] - 1e-9 or centers[i, 0] > 1.0 - radii[i] + 1e-9: valid = False; break
            if centers[i, 1] < radii[i] - 1e-9 or centers[i, 1] > 1.0 - radii[i] + 1e-9: valid = False; break
        if valid:
            for i in range(N):
                for j in range(i + 1, N):
                    d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                    if d < radii[i] + radii[j] - 1e-9: valid = False; break
                if not valid: break
        if valid: break
            
        radii *= 0.9997
        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)
        
    return centers, radii, float(np.sum(radii))
