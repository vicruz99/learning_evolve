import numpy as np
from scipy.optimize import minimize, linprog
import warnings
warnings.filterwarnings('ignore')

N = 26

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist)
            idx += 1
            
    bounds = [(1e-9, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return np.full(n, 0.01)

def get_lp_sum(centers):
    """Returns the sum of radii for a given center configuration."""
    return np.sum(solve_lp_radii(centers))

def lp_objective(c_flat):
    """Objective function for center optimization: negative sum of LP radii."""
    c = c_flat.reshape(N, 2)
    return -get_lp_sum(c)

def objective_joint(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints_joint(x):
    """Returns all inequality constraints >= 0."""
    xs, ys, rs = x[0::3], x[1::3], x[2::3]
    c = np.concatenate([xs - rs, 1.0 - xs - rs, ys - rs, 1.0 - ys - rs])
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    idx = np.tril_indices(N, -1)
    c = np.concatenate([c, np.hypot(dx[idx], dy[idx]) - dr[idx]])
    return c

def get_bounds_joint():
    """Variable bounds: x,y in [0,1], r in [1e-9, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-9, 0.5)])
    return b

def make_hex_init(r0, angle):
    """Generates a rotated hexagonal lattice initialization."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0 and len(pts) < N + 10:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
    pts = np.array(pts[:N + 10])
    center = np.array([0.5, 0.5])
    pts -= center
    c, s = np.cos(angle), np.sin(angle)
    pts = pts @ np.array([[c, -s], [s, c]])
    pts += center
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    if len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.2, 0.8, (N - len(pts), 2))])
    pts += np.random.uniform(-0.002, 0.002, pts.shape)
    return pts[:N]

def force_spread(centers, steps=200):
    """Force-directed relaxation to spread points evenly."""
    pts = centers.copy()
    for step in range(steps):
        f = np.zeros_like(pts)
        lr = 0.02 * (1.0 - step / steps)
        for i in range(N):
            for j in range(i + 1, N):
                dx = pts[j] - pts[i]
                d = np.hypot(dx[0], dx[1])
                if d < 0.35 and d > 1e-5:
                    rep = 0.008 / (d**2 + 0.0005)
                    f[i] -= dx * rep / d
                    f[j] += dx * rep / d
            for dim in range(2):
                if pts[i, dim] < 0.08: f[i, dim] += 0.06
                elif pts[i, dim] > 0.92: f[i, dim] -= 0.06
        pts += f * lr
        pts = np.clip(pts, 0.02, 0.98)
    return pts

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    best_sum = -1.0
    best_centers = None
    best_radii = None

    # Phase 1: Generate Diverse Initial Configurations
    inits = []
    for r0 in [0.085, 0.09, 0.095, 0.10, 0.105]:
        for ang in np.linspace(-0.3, 0.3, 9):
            inits.append(make_hex_init(r0, ang))
    for s in range(10):
        np.random.seed(s + 1000)
        pts = np.random.uniform(0.15, 0.85, (N, 2))
        inits.append(force_spread(pts, steps=200))

    bounds_opt = [(0.01, 0.99)] * (2 * N)

    # Phase 2: Optimize centers by maximizing LP radius sum
    for init_c in inits:
        try:
            res = minimize(lp_objective, init_c.ravel(), method='L-BFGS-B', bounds=bounds_opt,
                           options={'maxiter': 2000, 'ftol': 1e-15, 'gtol': 1e-12})
            c_opt = res.x.reshape(N, 2)
            r_opt = solve_lp_radii(c_opt)
            curr_sum = np.sum(r_opt)
            
            if curr_sum > best_sum:
                best_sum = curr_sum
                best_centers = c_opt.copy()
                best_radii = r_opt.copy()
        except Exception:
            continue

    # Phase 3: Joint SLSQP Refinement with Perturbation
    if best_centers is not None:
        x0 = np.zeros(3 * N)
        x0[0::3] = best_centers[:, 0]
        x0[1::3] = best_centers[:, 1]
        x0[2::3] = best_radii
        
        for _ in range(6):
            # Perturb to escape local minima
            x0[0::3] += np.random.normal(0, 0.0015, N)
            x0[1::3] += np.random.normal(0, 0.0015, N)
            x0[2::3] *= 0.97
            
            # Project to feasible bounds
            for i in range(N):
                r = x0[3*i+2]
                x0[3*i] = np.clip(x0[3*i], r, 1.0-r)
                x0[3*i+1] = np.clip(x0[3*i+1], r, 1.0-r)
                
            try:
                res = minimize(objective_joint, x0, method='SLSQP', bounds=get_bounds_joint(),
                               constraints={'type': 'ineq', 'fun': constraints_joint},
                               options={'maxiter': 8000, 'ftol': 1e-13})
                if not np.isnan(res.fun):
                    c_tmp = np.column_stack((res.x[0::3], res.x[1::3]))
                    r_tmp = solve_lp_radii(c_tmp)
                    if np.sum(r_tmp) > best_sum:
                        best_sum = np.sum(r_tmp)
                        best_centers = c_tmp.copy()
                        best_radii = r_tmp.copy()
            except Exception:
                pass

    # Phase 4: Final strict validation repair against 1e-12 tolerance
    for _ in range(100):
        valid = True
        for i in range(N):
            if best_radii[i] < 0: valid = False; break
            if best_centers[i, 0] < best_radii[i] - 1e-9 or best_centers[i, 0] > 1.0 - best_radii[i] + 1e-9: valid = False; break
            if best_centers[i, 1] < best_radii[i] - 1e-9 or best_centers[i, 1] > 1.0 - best_radii[i] + 1e-9: valid = False; break
        if valid:
            for i in range(N):
                for j in range(i + 1, N):
                    d = np.hypot(best_centers[i, 0] - best_centers[j, 0], best_centers[i, 1] - best_centers[j, 1])
                    if d < best_radii[i] + best_radii[j] - 1e-9: valid = False; break
                if not valid: break
        if valid: break
        best_radii *= 0.9995
        best_centers[:, 0] = np.clip(best_centers[:, 0], best_radii, 1.0 - best_radii)
        best_centers[:, 1] = np.clip(best_centers[:, 1], best_radii, 1.0 - best_radii)

    return best_centers, best_radii, float(best_sum)
