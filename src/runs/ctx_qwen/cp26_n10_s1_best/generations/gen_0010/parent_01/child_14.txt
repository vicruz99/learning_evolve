To maximize the sum of radii for 26 circles in a unit square, we recognize that the problem is highly non-convex with many local optima corresponding to different contact graphs. Previous solutions using SLSQP on (x,y,r) variables converge to ~2.634292 but struggle to escape this local optimum.

Key improvements in this strategy:
1. **Decoupled Optimization**: We separate center positions from radii. For any fixed set of centers, the maximum sum of radii is a Linear Programming (LP) problem. LP is solved instantly using `scipy.optimize.linprog` with the `highs` solver. This provides a highly accurate and smooth objective function landscape over the center coordinates.
2. **Simulated Annealing (SA) on Centers**: We run SA directly on the 52 center coordinates, using the LP sum of radii as the evaluation function. This allows global exploration of different contact topologies that SLSQP gets stuck on. The SA uses an adaptive cooling schedule and occasional random jumps to escape local basins.
3. **Precomputed LP Structure**: The constraint matrix for the LP is constant. We precompute it once, drastically reducing the overhead per LP call. This enables tens of thousands of rapid evaluations during SA.
4. **Hybrid Refinement**: After SA finds a promising configuration, we switch back to SLSQP on the full (x,y,r) space for local gradient-based polishing. We also run multiple SA chains from diverse initial layouts (hexagonal, perturbed, random) to cover the search space thoroughly.
5. **Strict Validation Repair**: A final safety loop ensures the returned configuration strictly satisfies all boundary and overlap constraints within the required tolerance, shrinking radii minimally if numerical drift occurs.

This combination of fast LP evaluation, global SA search, and local SLSQP refinement is designed to break the symmetry of the hexagonal lattice and discover packings with boundary contacts and varied radii that push the sum beyond 2.635.

import numpy as np
from scipy.optimize import linprog, minimize
import warnings

warnings.filterwarnings('ignore')

N = 26
TRIL_IDX = np.tril_indices(N, -1)
N_PAIRS = N * (N - 1) // 2

# Precompute constant parts of LP and SLSQP
A_overlap = np.zeros((N_PAIRS, N))
idx = 0
for i in range(N):
    for j in range(i + 1, N):
        A_overlap[idx, i] = 1.0
        A_overlap[idx, j] = 1.0
        idx += 1

A_FULL = np.vstack([np.eye(N), A_overlap])
BOUNDS_LP = [(0.0, None)] * N
OBJ_LP = -np.ones(N)
ZEROS_N = np.zeros(N)

def lp_sum_radii(centers):
    """Compute max sum of radii for fixed centers using fast LP."""
    ub = np.minimum(np.minimum(centers[:,0], 1.0 - centers[:,0]), 
                    np.minimum(centers[:,1], 1.0 - centers[:,1]))
    np.maximum(ub, 1e-9, out=ub)
    
    dx = centers[:, None, 0] - centers[None, :, 0]
    dy = centers[:, None, 1] - centers[None, :, 1]
    D = np.sqrt(dx*dx + dy*dy)
    D = D[TRIL_IDX]
    
    b_ub = np.empty(N + N_PAIRS)
    b_ub[:N] = ub
    b_ub[N:] = D
    
    res = linprog(OBJ_LP, A_ub=A_FULL, b_ub=b_ub, bounds=BOUNDS_LP, method='highs')
    if res.success:
        return np.sum(res.x), res.x
    return 0.0, ZEROS_N

def slsqp_polish(centers):
    """Refine centers and radii using SLSQP."""
    _, r_init = lp_sum_radii(centers)
    x0 = np.zeros(3*N)
    x0[0::3] = centers[:,0]
    x0[1::3] = centers[:,1]
    x0[2::3] = r_init
    
    def obj(x):
        return -np.sum(x[2::3])
        
    def cons(x):
        cx = x[0::3]
        cy = x[1::3]
        cr = x[2::3]
        c = np.concatenate([cx - cr, 1.0 - cx - cr, cy - cr, 1.0 - cy - cr])
        dx = cx[:, None] - cx[None, :]
        dy = cy[:, None] - cy[None, :]
        dr = cr[:, None] + cr[None, :]
        c = np.concatenate([c, dx[TRIL_IDX]**2 + dy[TRIL_IDX]**2 - dr[TRIL_IDX]**2])
        return c
        
    bounds = [(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)] * N
    try:
        res = minimize(obj, x0, method='SLSQP', bounds=bounds, 
                       constraints={'type': 'ineq', 'fun': cons},
                       options={'maxiter': 5000, 'ftol': 1e-12, 'disp': False})
        if not np.isnan(res.fun):
            c_opt = np.column_stack((res.x[0::3], res.x[1::3]))
            s, r_opt = lp_sum_radii(c_opt)
            return c_opt, r_opt, s
    except Exception:
        pass
    return centers, r_init, np.sum(r_init)

def make_hex_init(r0, angle=0.0):
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0 and len(pts) < N + 10:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        
    pts = np.array(pts[:N + 10])
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
        
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.1, 0.9, (1, 2))])
    return pts[:N]

def run_sa(start_centers, steps=15000):
    np.random.seed(np.random.randint(0, 100000))
    cur_c = start_centers.copy()
    cur_s, cur_r = lp_sum_radii(cur_c)
    best_c, best_r, best_s = cur_c.copy(), cur_r.copy(), cur_s
    
    T = 0.008
    for i in range(steps):
        sigma = max(0.0003, T * (0.5 + 0.5 * np.cos(np.pi * i / steps)) + 0.001)
        new_c = cur_c + np.random.normal(0, sigma, cur_c.shape)
        new_c = np.clip(new_c, 0.005, 0.995)
        
        s, r = lp_sum_radii(new_c)
        
        delta = s - cur_s
        if delta > 0 or np.random.rand() < np.exp(delta / max(T, 1e-6)):
            cur_c, cur_r, cur_s = new_c.copy(), r.copy(), s
            if s > best_s:
                best_c, best_r, best_s = cur_c.copy(), cur_r.copy(), s
                
        # Occasionally inject larger noise to escape
        if i % 2000 == 0 and i > 0:
            cur_c += np.random.normal(0, 0.02, cur_c.shape)
            cur_c = np.clip(cur_c, 0.005, 0.995)
            cur_s, cur_r = lp_sum_radii(cur_c)
            if cur_s > best_s:
                best_c, best_r, best_s = cur_c.copy(), cur_r.copy(), cur_s
                
        T *= 0.9999
    return best_c, best_r, best_s

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Phase 1: SA Search from diverse starts
    starts = [
        make_hex_init(0.09, 0.0),
        make_hex_init(0.09, 0.15),
        make_hex_init(0.085, -0.1),
        np.random.uniform(0.1, 0.9, (N, 2))
    ]
    
    for s_init in starts:
        c, r, s = run_sa(s_init, steps=12000)
        if s > best_sum:
            best_sum = s
            best_centers = c
            best_radii = r
            
    # Phase 2: SLSQP Polish
    best_centers, best_radii, best_sum = slsqp_polish(best_centers)
    
    # Phase 3: Deflation & Re-search to escape final local minima
    for _ in range(5):
        pert_c = best_centers.copy()
        subset = np.random.choice(N, size=N//3, replace=False)
        pert_c[subset] += np.random.normal(0, 0.015, (len(subset), 2))
        pert_c = np.clip(pert_c, 0.01, 0.99)
        c, r, s = run_sa(pert_c, steps=5000)
        if s > best_sum:
            best_sum = s
            best_centers = c
            best_radii = r
            best_centers, best_radii, best_sum = slsqp_polish(best_centers)
            
    # Final strict validation repair
    centers = best_centers.copy()
    radii = best_radii.copy()
    
    for _ in range(100):
        valid = True
        for i in range(N):
            if radii[i] < 0: valid = False; break
            if centers[i, 0] < radii[i] - 1e-9 or centers[i, 0] > 1.0 - radii[i] + 1e-9: valid = False; break
            if centers[i, 1] < radii[i] - 1e-9 or centers[i, 1] > 1.0 - radii[i] + 1e-9: valid = False; break
        if valid:
            for i in range(N):
                for j in range(i + 1, N):
                    d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                    if d < radii[i] + radii[j] - 1e-9: valid = False; break
                if not valid: break
        if valid: break
            
        radii *= 0.9998
        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)
        
    return centers, radii, float(np.sum(radii))
