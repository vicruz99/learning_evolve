import numpy as np
from scipy.optimize import minimize, linprog
import warnings

warnings.filterwarnings('ignore')

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def solve_lp(centers):
    """Solve for optimal radii given fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A = np.zeros((num_ineq, n))
    b = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A[idx, i] = 1.0
        b[idx] = max(0.0, lim)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = max(0.0, d)
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success and res.status == 0:
            return np.maximum(res.x, 1e-9)
    except Exception:
        pass
    return np.full(n, 1e-9)

def objective_joint(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints_joint(x):
    """Vectorized inequality constraints for SLSQP."""
    xs, ys, rs = x[0::3], x[1::3], x[2::3]
    c = np.concatenate([
        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs
    ])
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    i, j = TRIL_IDX
    c = np.concatenate([c, dx[i, j]**2 + dy[i, j]**2 - dr[i, j]**2])
    return c

def get_bounds_joint():
    """Variable bounds for joint optimization."""
    return [(0.0, 1.0), (0.0, 1.0), (1e-9, 0.5)] * N

def force_init(seed):
    """Generate initial configuration via repulsive force simulation."""
    np.random.seed(seed)
    pts = np.random.uniform(0.1, 0.9, (N, 2))
    for step in range(300):
        f = np.zeros_like(pts)
        lr = 0.02 * (1.0 - step / 300.0)
        for i in range(N):
            for j in range(i + 1, N):
                dx = pts[j, 0] - pts[i, 0]
                dy = pts[j, 1] - pts[i, 1]
                d = np.hypot(dx, dy)
                if d < 0.25 and d > 1e-5:
                    rep = 0.01 / (d**2 + 0.001)
                    f[i, 0] -= (dx / d) * rep
                    f[i, 1] -= (dy / d) * rep
                    f[j, 0] += (dx / d) * rep
                    f[j, 1] += (dy / d) * rep
            for dim in range(2):
                if pts[i, dim] < 0.05:
                    f[i, dim] += 0.05
                elif pts[i, dim] > 0.95:
                    f[i, dim] -= 0.05
        pts += f * lr
        pts = np.clip(pts, 0.005, 0.995)
    return pts

def hex_init(angle=0.0):
    """Generate a rotated hexagonal lattice initialization."""
    pts = []
    r0 = 0.09
    y = r0
    row = 0
    while len(pts) < N + 5:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
    pts = np.array(pts[:N + 5])
    
    if angle != 0.0:
        c = np.array([0.5, 0.5])
        cos_a, sin_a = np.cos(angle), np.sin(angle)
        pts = (pts - c) @ np.array([[cos_a, -sin_a], [sin_a, cos_a]]) + c
        
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.1, 0.9, (1, 2))])
    return pts[:N]

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    for pts in inits:
        centers = pts.copy()
        radii = solve_lp(centers)
        curr_sum = np.sum(radii)
        
        # Hill climbing on centers using LP for exact radius evaluation
        step = 0.02
        for it in range(500):
            k = max(2, N // 4)

            old_centers = centers[idx].copy()

            new_radii = solve_lp(centers)
            new_sum = np.sum(new_radii)
            
            if new_sum > curr_sum:
                radii = new_radii
                curr_sum = new_sum
            else:
                centers[idx] = old_centers
                
            if it > 0 and it % 60 == 0:
                step *= 0.82

                    break
                    
        if curr_sum > best_sum:

            best_centers = centers.copy()
            best_radii = radii.copy()

            pass

    centers = best_centers
    radii = best_radii

                break

                        break

        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)
        
    return centers, radii, float(np.sum(radii))
