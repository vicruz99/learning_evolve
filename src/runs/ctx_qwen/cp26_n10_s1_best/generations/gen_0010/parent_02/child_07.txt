import numpy as np
from scipy.optimize import minimize, linprog
import warnings
warnings.filterwarnings('ignore')

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints_sq(x):
    """Squared distance constraints: dist^2 >= (r_i + r_j)^2."""
    xs, ys, rs = x[0::3], x[1::3], x[2::3]
    c = np.concatenate([
        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs
    ])
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    c = np.concatenate([c, dx[TRIL_IDX]**2 + dy[TRIL_IDX]**2 - dr[TRIL_IDX]**2])
    return c

def constraints_hypot(x):
    """Direct distance constraints: dist >= r_i + r_j."""
    xs, ys, rs = x[0::3], x[1::3], x[2::3]
    c = np.concatenate([
        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs
    ])
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    c = np.concatenate([c, np.hypot(dx[TRIL_IDX], dy[TRIL_IDX]) - dr[TRIL_IDX]])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-7, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)])
    return b

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    idx = 0
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist)
            idx += 1
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-7)
    except Exception:
        pass
    return np.full(n, 0.05)

def project_to_feasible(x):
    """Project variables to strictly satisfy bounds."""
    x = x.copy()
    for i in range(N):
        r = max(1e-6, x[3 * i + 2])
        x[3 * i] = np.clip(x[3 * i], r, 1.0 - r)
        x[3 * i + 1] = np.clip(x[3 * i + 1], r, 1.0 - r)
        x[3 * i + 2] = r
    return x

def make_hex_init(r0, angle=0.0):
    """Generates a rotated hexagonal lattice initialization."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        if y + r0 > 1.0:
            break
    pts = np.array(pts[:N + 10])
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
    mask = (pts[:, 0] >= 0.01) & (pts[:, 0] <= 0.99) & (pts[:, 1] >= 0.01) & (pts[:, 1] <= 0.99)
    pts = pts[mask]
    while len(pts) < N:
        pts = np.vstack([pts, [np.random.uniform(0.1, 0.9), np.random.uniform(0.1, 0.9)]])
    return pts[:N]

def make_corner_hex_init(r0, angle=0.0):
    """Hex lattice biased towards corners."""
    pts = []
    y = r0 * 0.8
    row = 0
    while len(pts) < N + 10:
        x = r0 * 0.8 if row % 2 == 0 else r0 * 1.6
        while x <= 1.0 - r0 * 0.8:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0 * 0.95
        row += 1
        if y + r0 > 1.0:
            break
    pts = np.array(pts[:N + 10])
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
    mask = (pts[:, 0] >= 0.01) & (pts[:, 0] <= 0.99) & (pts[:, 1] >= 0.01) & (pts[:, 1] <= 0.99)
    pts = pts[mask]
    while len(pts) < N:
        pts = np.vstack([pts, [np.random.uniform(0.1, 0.9), np.random.uniform(0.1, 0.9)]])
    return pts[:N]

def make_edge_focused_init(seed, r0=0.09):
    """Initialize with circles near edges and corners."""
    np.random.seed(seed)
    pts = []
    r = r0
    # Corners
    pts.append([r, r])
    pts.append([1 - r, r])
    pts.append([r, 1 - r])
    pts.append([1 - r, 1 - r])
    # Edge centers
    pts.append([0.5, r])
    pts.append([0.5, 1 - r])
    pts.append([r, 0.5])
    pts.append([1 - r, 0.5])
    # Center
    pts.append([0.5, 0.5])
    # Hex fill
    y = r + 0.03
    row = 0
    while len(pts) < N:
        x = r + 0.03 if row % 2 == 0 else 2.0 * r + 0.01
        while x <= 1.0 - r - 0.01 and len(pts) < N:
            pts.append([x, y])
            x += 2.0 * r + 0.02
        y += np.sqrt(3.0) * r * 0.9
        row += 1
        if y + r > 1.0:
            break
    pts = np.array(pts[:N])
    pts += np.random.uniform(-0.005, 0.005, pts.shape)
    pts = np.clip(pts, 0.05, 0.95)
    return pts

def make_staggered_rows_init(rows_pattern, angle=0.0):
    """Generates a staggered row initialization with custom pattern."""
    pts = []
    r = 0.09
    y = r
    for i, count in enumerate(rows_pattern):
        x_start = r if i % 2 == 0 else 2.0 * r
        for _ in range(count):
            if len(pts) >= N:
                break
            x = x_start
            for _ in range(10):
                if len(pts) >= N:
                    break
                pts.append([x, y])
                x += 2.0 * r
            if x > 1.0 - r:
                break
        y += np.sqrt(3.0) * r
        if y + r > 1.0:
            break
    pts = np.array(pts[:N + 5])
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
    mask = (pts[:, 0] >= 0.01) & (pts[:, 0] <= 0.99) & (pts[:, 1] >= 0.01) & (pts[:, 1] <= 0.99)
    pts = pts[mask]
    while len(pts) < N:
        pts = np.vstack([pts, [np.random.uniform(0.1, 0.9), np.random.uniform(0.1, 0.9)]])
    return pts[:N]

def force_layout(seed, steps=250):
    """Force-directed layout to spread points evenly."""
    np.random.seed(seed)
    pts = np.random.uniform(0.15, 0.85, (N, 2))
    rs = np.full(N, 0.05)
    for step in range(steps):
        f = np.zeros_like(pts)
        lr = 0.02 * (1.0 - step / steps)
        for i in range(N):
            for j in range(i + 1, N):
                dx = pts[j] - pts[i]
                d = np.hypot(dx[0], dx[1])
                if d < 0.3 and d > 1e-5:
                    rep = 0.012 / (d**2 + 0.001)
                    f[i] -= dx * rep / d
                    f[j] += dx * rep / d
            for dim in range(2):
                if pts[i, dim] < rs[i] + 0.02:
                    f[i, dim] += 0.06
                elif pts[i, dim] > 1.0 - rs[i] - 0.02:
                    f[i, dim] -= 0.06
        pts += f * lr
        pts = np.clip(pts, 0.02, 0.98)
    return pts

def try_optimize(x0, bounds_obj, cons, maxiter=15000):
    """Run SLSQP optimization and return result."""
    try:
        res = minimize(objective, x0, method='SLSQP', bounds=bounds_obj, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'disp': False})
        if not np.isnan(res.fun):
            return res
    except Exception:
        pass
    return None
