1. **Constraint Formulation**: Switch from squared-distance constraints to Euclidean distance constraints (`dist - (r_i + r_j) >= 0`). Squared constraints have vanishing gradients at contact points, which hampers SLSQP's ability to fine-tune tight packings. Euclidean distance provides consistent gradients.
2. **Diverse Initializations**: Combine rotated hexagonal lattices, force-directed layouts, and a "grow circles" algorithm. The grow algorithm naturally finds dense topologies by expanding radii while resolving overlaps geometrically.
3. **Topology-Breaking Perturbations**: Add a phase that swaps positions of random circle pairs and deflates subsets before re-optimizing. This helps escape rigid local minima where circles are locked in a suboptimal contact network.
4. **LP-SLSQP Alternation**: Use Linear Programming to compute optimal radii for fixed centers, then feed back into SLSQP. This decouples the non-convex geometry from the radius scaling, speeding up convergence and improving precision.
5. **High-Precision Polish**: Final phase with tight tolerances and minimal deflation to squeeze out numerical slack.

import numpy as np
from scipy.optimize import minimize, linprog
import warnings
warnings.filterwarnings('ignore')

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Returns all inequality constraints >= 0 using Euclidean distances."""
    xs, ys, rs = x[0::3], x[1::3], x[2::3]
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    c = np.concatenate([
        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs
    ])
    
    # Overlap constraints: dist >= r_i + r_j
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    # Euclidean distance provides stable gradients at contact points
    dists = np.hypot(dx[TRIL_IDX], dy[TRIL_IDX])
    c = np.concatenate([c, dists - dr[TRIL_IDX]])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-7, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)])
    return b

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist)
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-6)
    except Exception:
        pass
    return np.full(n, 0.05)

def project_to_feasible(x):
    """Project variables to strictly satisfy bounds."""
    x = x.copy()
    for i in range(N):
        r = max(1e-6, x[3 * i + 2])
        x[3 * i] = np.clip(x[3 * i], r, 1.0 - r)
        x[3 * i + 1] = np.clip(x[3 * i + 1], r, 1.0 - r)
        x[3 * i + 2] = r
    return x

def make_hex_starts():
    """Generates rotated hexagonal lattice initializations."""
    starts = []
    for ang in np.linspace(-0.35, 0.35, 15):
        pts = []
        r0 = 0.09
        y = r0
        row = 0
        while len(pts) < N + 10:
            x = r0 if row % 2 == 0 else 2.0 * r0
            while x <= 1.0 - r0:
                pts.append([x, y])
                x += 2.0 * r0
            y += np.sqrt(3.0) * r0
            row += 1
        pts = np.array(pts[:N + 10])
        
        c = np.array([0.5, 0.5])
        pts = (pts - c) @ np.array([[np.cos(ang), -np.sin(ang)], [np.sin(ang), np.cos(ang)]]) + c
        
        mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
        pts = pts[mask]
        while len(pts) < N:
            pts = np.vstack([pts, [np.random.uniform(0.1, 0.9), np.random.uniform(0.1, 0.9)]])
        pts = pts[:N]
        starts.append(pts)
    return starts

def make_force_starts(num_starts):
    """Generates initial configurations via repulsive force simulation."""
    starts = []
    for s in range(num_starts):
        np.random.seed(s + 500)
        pts = np.random.uniform(0.15, 0.85, (N, 2))
        rs = np.full(N, 0.05)
        for step in range(300):
            f = np.zeros_like(pts)
            lr = 0.02 * (1.0 - step / 300.0)
            for i in range(N):
                for j in range(i + 1, N):
                    dx = pts[j] - pts[i]
                    d = np.hypot(dx[0], dx[1])
                    if d < 0.35 and d > 1e-5:
                        rep = 0.015 / (d**2 + 0.001)
                        f[i] -= dx * rep / d
                        f[j] += dx * rep / d
                for dim in range(2):
                    if pts[i, dim] < rs[i] + 0.02: f[i, dim] += 0.06
                    elif pts[i, dim] > 1.0 - rs[i] - 0.02: f[i, dim] -= 0.06
            pts += f * lr
            pts = np.clip(pts, 0.01, 0.99)
        starts.append(pts)
    return starts

def make_grow_starts(num_starts):
    """Generates initial configurations by growing circles from random positions."""
    starts = []
    for s in range(num_starts):
        np.random.seed(s + 1000)
        pts = np.random.uniform(0.1, 0.9, (N, 2))
        r = 0.01
        for step in range(600):
            r += 0.0005
            for i in range(N):
                for j in range(i + 1, N):
                    dx = pts[j, 0] - pts[i, 0]
                    dy = pts[j, 1] - pts[i, 1]
                    d = np.hypot(dx, dy)
                    if d < 2.0 * r and d > 1e-6:
                        overlap = 2.0 * r - d
                        ux, uy = dx / d, dy / d
                        pts[i, 0] -= ux * overlap * 0.5
                        pts[i, 1] -= uy * overlap * 0.5
                        pts[j, 0] += ux * overlap * 0.5
                        pts[j, 1] += uy * overlap * 0.5
            for i in range(N):
                if pts[i, 0] < r: pts[i, 0] = r
                if pts[i, 0] > 1.0 - r: pts[i, 0] = 1.0 - r
                if pts[i, 1] < r: pts[i, 1] = r
                if pts[i, 1] > 1.0 - r: pts[i, 1] = 1.0 - r
        starts.append(pts)
    return starts

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    bounds_obj = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -1.0
    best_x = None
    
    # Phase 1: Generate Diverse Initial Configurations
    inits = []
    inits.extend(make_hex_starts())
    inits.extend(make_force_starts(15))
    inits.extend(make_grow_starts(10))
    
    # Phase 2: Multi-start Optimization with LP Initialization
    for pts in inits:
        x0 = np.zeros(3 * N)
        x0[0::3] = pts[:, 0]
        x0[1::3] = pts[:, 1]
        
        r_lp = solve_lp_radii(pts)
        x0[2::3] = np.maximum(r_lp * 0.99, 0.005)
        x0 = project_to_feasible(x0)
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds_obj, constraints=cons,
                           options={'maxiter': 25000, 'ftol': 1e-14, 'disp': False})
            if not np.isnan(res.fun):
                c_opt = np.column_stack((res.x[0::3], res.x[1::3]))
                r_opt = solve_lp_radii(c_opt)
                if r_opt is not None:
                    curr_sum = np.sum(r_opt)
                    if curr_sum > best_sum:
                        best_sum = curr_sum
                        best_x = np.zeros(3 * N)
                        best_x[0::3] = c_opt[:, 0]
                        best_x[1::3] = c_opt[:, 1]
                        best_x[2::3] = r_opt
        except Exception:
            pass
            
    # Phase 3: Topology-Breaking Perturbations (Deflation, Swapping, Rotation)
    if best_x is not None:
        for cyc in range(80):
            x0 = best_x.copy()
            
            # Cooling perturbation
            noise_scale = 0.002 * (0.90 ** cyc)
            x0[0::3] += np.random.normal(0, noise_scale, N)
            x0[1::3] += np.random.normal(0, noise_scale, N)
            
            # Deflate radii to create slack
            x0[2::3] *= 0.94
            
            # Randomly swap positions of 2-4 pairs to change contact graph
            swap_idx = np.random.choice(N, size=min(8, N), replace=False)
            if len(swap_idx) > 1:
                np.random.shuffle(swap_idx)
                for k in range(0, len(swap_idx)-1, 2):
                    i, j = swap_idx[k], swap_idx[k+1]
                    x0[3*i], x0[3*j] = x0[3*j], x0[3*i]
                    x0[3*i+1], x0[3*j+1] = x0[3*j+1], x0[3*i+1]
            
            # Aggressively deflate a random subset
            subset = np.random.choice(N, size=max(3, N // 4), replace=False)
            x0[subset * 3 + 2] *= 0.70
            
            x0 = project_to_feasible(x0)
            
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds_obj, constraints=cons,
                               options={'maxiter': 12000, 'ftol': 1e-13, 'disp': False})
                if not np.isnan(res.fun):
                    c_tmp = np.column_stack((res.x[0::3], res.x[1::3]))
                    r_new = solve_lp_radii(c_tmp)
                    if r_new is not None:
                        x_p = res.x.copy()
                        x_p[2::3] = r_new * 0.999
                        x_p = project_to_feasible(x_p)
                        
                        s_val = np.sum(x_p[2::3])
                        vals = constraints(x_p)
                        if np.min(vals) >= -1e-5 and s_val > best_sum:
                            best_sum = s_val
                            best_x = x_p.copy()
            except Exception:
                pass
                
        # Phase 4: Rotation Search from Best
        best_centers = np.column_stack((best_x[0::3], best_x[1::3]))
        for ang_deg in np.linspace(-12, 12, 13):
            ang = np.deg2rad(ang_deg)
            c, s = np.cos(ang), np.sin(ang)
            rot_centers = (best_centers - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
            rot_centers = np.clip(rot_centers, 0.01, 0.99)
            
            r_rot = solve_lp_radii(rot_centers)
            if r_rot is not None:
                x_rot = np.zeros(3 * N)
                x_rot[0::3] = rot_centers[:, 0]
                x_rot[1::3] = rot_centers[:, 1]
                x_rot[2::3] = r_rot * 0.998
                x_rot = project_to_feasible(x_rot)
                
                try:
                    res = minimize(objective, x_rot, method='SLSQP', bounds=bounds_obj, constraints=cons,
                                   options={'maxiter': 8000, 'ftol': 1e-14, 'disp': False})
                    if not np.isnan(res.fun):
                        c_pol = np.column_stack((res.x[0::3], res.x[1::3]))
                        r_pol = solve_lp_radii(c_pol)
                        if r_pol is not None:
                            s_pol = np.sum(r_pol)
                            if s_pol > best_sum:
                                best_sum = s_pol
                                best_x = np.zeros(3 * N)
                                best_x[0::3] = c_pol[:, 0]
                                best_x[1::3] = c_pol[:, 1]
                                best_x[2::3] = r_pol
                except Exception:
                    pass

        # Phase 5: High-Precision Polish
        for _ in range(10):
            x_pol = best_x.copy()
            x_pol[2::3] *= 0.995
            x_pol = project_to_feasible(x_pol)
            try:
                res = minimize(objective, x_pol, method='SLSQP', bounds=bounds_obj, constraints=cons,
                               options={'maxiter': 30000, 'ftol': 1e-15, 'disp': False})
                if not np.isnan(res.fun):
                    c_pol = np.column_stack((res.x[0::3], res.x[1::3]))
                    r_pol = solve_lp_radii(c_pol)
                    if r_pol is not None:
                        s_pol = np.sum(r_pol)
                        if s_pol > best_sum:
                            best_sum = s_pol
                            best_x = np.zeros(3 * N)
                            best_x[0::3] = c_pol[:, 0]
                            best_x[1::3] = c_pol[:, 1]
                            best_x[2::3] = r_pol
            except Exception:
                pass

    # Fallback
    if best_x is None:
        best_x = np.zeros(3 * N)
        best_x[0::3] = np.tile(np.linspace(0.1, 0.9, 5), 6)[:N]
        best_x[1::3] = np.repeat(np.linspace(0.1, 0.9, 6), 5)[:N]
        best_x[2::3] = 0.06
        best_sum = np.sum(best_x[2::3])
        
    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3].copy()
    
    # Final strict validation repair against 1e-12 tolerance
    for _ in range(100):
        valid = True
        for i in range(N):
            if radii[i] < 0 or centers[i, 0] < radii[i] - 1e-9 or centers[i, 0] > 1.0 - radii[i] + 1e-9 or \
               centers[i, 1] < radii[i] - 1e-9 or centers[i, 1] > 1.0 - radii[i] + 1e-9:
                valid = False
                break
        if valid:
            for i in range(N):
                for j in range(i + 1, N):
                    d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                    if d < radii[i] + radii[j] - 1e-9:
                        valid = False
                        break
                if not valid:
                    break
        if valid:
            break
            
        radii *= 0.9995
        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)
        
    return centers, radii, float(np.sum(radii))
