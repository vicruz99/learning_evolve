Looking at the problem, the current best sum of radii is 2.634292 and we need to reach 2.636. The key bottleneck is escaping local optima in the circle packing configuration space.

Key insights for improvement:
1. **Gradient conditioning**: Using `np.hypot` for distance constraints gives much better gradients near contact points compared to squared distance constraints. The squared constraint has zero gradient when circles touch, making SLSQP stall.
2. **Topology exploration**: The contact graph (which circles touch which) is the key determinant. We need aggressive perturbation strategies to find better topologies.
3. **Dual formulation**: Alternate between optimizing centers (SLSQP) and radii (LP). The LP gives exact optimal radii for fixed centers, which is fast.
4. **More diverse inits**: Use corner-focused, rotated hexagonal, force-directed, and perturbed grid initializations.
5. **Large deflation**: Sometimes we need to shrink circles dramatically to allow the optimizer to reconfigure the topology, then let them grow back.

Strategy:
- Generate 100+ diverse initial configurations
- Use hypot-based constraints for primary optimization
- Run aggressive deflation with large random perturbations (100 cycles)
- Add "swap" operations (randomly reposition subsets of circles)
- Add rotation operations from the best configuration
- High-precision final polish with both constraint formulations
- LP snap after every SLSQP run to maximize radii for achieved centers

import numpy as np
from scipy.optimize import minimize, linprog
import warnings

warnings.filterwarnings('ignore')

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints_hypot(x):
    """Uses np.hypot for better gradient conditioning near contact."""
    xs, ys, rs = x[0::3], x[1::3], x[2::3]
    
    c = np.concatenate([
        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs
    ])
    
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    c = np.concatenate([c, np.hypot(dx[TRIL_IDX], dy[TRIL_IDX]) - dr[TRIL_IDX]])
    return c

def constraints_sq(x):
    """Squared distance constraints (alternative formulation)."""
    xs, ys, rs = x[0::3], x[1::3], x[2::3]
    
    c = np.concatenate([
        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs
    ])
    
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    c = np.concatenate([c, dx[TRIL_IDX]**2 + dy[TRIL_IDX]**2 - dr[TRIL_IDX]**2])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-7, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)])
    return b

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist)
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-6)
    except Exception:
        pass
    return np.full(n, 0.05)

def project_to_feasible(x):
    """Project variables to strictly satisfy bounds."""
    x = x.copy()
    for i in range(N):
        r = max(1e-6, x[3 * i + 2])
        x[3 * i] = np.clip(x[3 * i], r, 1.0 - r)
        x[3 * i + 1] = np.clip(x[3 * i + 1], r, 1.0 - r)
        x[3 * i + 2] = r
    return x

def make_hex_init(r0, angle):
    """Generates a rotated hexagonal lattice initialization."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 15:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        
    pts = np.array(pts[:N + 15])
    c = np.array([0.5, 0.5])
    
    if angle != 0.0:
        pts = (pts - c) @ np.array([[np.cos(angle), -np.sin(angle)], [np.sin(angle), np.cos(angle)]]) + c
    
    mask = (pts[:, 0] >= 0.01) & (pts[:, 0] <= 0.99) & (pts[:, 1] >= 0.01) & (pts[:, 1] <= 0.99)
    pts = pts[mask]
    while len(pts) < N:
        pts = np.vstack([pts, [np.random.uniform(0.1, 0.9), np.random.uniform(0.1, 0.9)]])
    return pts[:N]

def force_layout(seed, steps=300):
    """Force-directed layout to spread points evenly."""
    np.random.seed(seed)
    pts = np.random.uniform(0.15, 0.85, (N, 2))
    rs = np.full(N, 0.05)
    
    for step in range(steps):
        f = np.zeros_like(pts)
        lr = 0.025 * (1.0 - step / steps)
        
        for i in range(N):
            for j in range(i + 1, N):
                dx = pts[j] - pts[i]
                d = np.hypot(dx[0], dx[1])
                if d < 0.35 and d > 1e-5:
                    rep = 0.018 / (d**2 + 0.001)
                    f[i] -= dx * rep / d
                    f[j] += dx * rep / d
            
            for dim in range(2):
                if pts[i, dim] < rs[i] + 0.02:
                    f[i, dim] += 0.1
                elif pts[i, dim] > 1.0 - rs[i] - 0.02:
                    f[i, dim] -= 0.1
        
        pts += f * lr
        pts = np.clip(pts, 0.01, 0.99)
    
    return pts

def make_corner_init(seed):
    """Corner-focused initialization."""
    np.random.seed(seed)
    pts = np.zeros((N, 2))
    r = 0.09
    
    pts[0] = [r, r]
    pts[1] = [1-r, r]
    pts[2] = [r, 1-r]
    pts[3] = [1-r, 1-r]
    pts[4] = [0.5, r]
    pts[5] = [0.5, 1-r]
    pts[6] = [r, 0.5]
    pts[7] = [1-r, 0.5]
    pts[8] = [0.5, 0.5]
    
    idx = 9
    y = r + 0.06
    row = 0
    while idx < N and y + r <= 1.0:
        x = r + 0.06 if row % 2 == 0 else r + 0.18
        while idx < N and x + r <= 1.0:
            pts[idx] = [x, y]
            idx += 1
            x += 2.0 * r
        y += np.sqrt(3.0) * r * 0.9
        row += 1
    
    for i in range(idx, N):
        pts[i] = np.random.uniform(0.15, 0.85, 2)
    
    pts += np.random.uniform(-0.005, 0.005, pts.shape)
    pts = np.clip(pts, 0.02, 0.98)
    return pts

def make_staggered_init(seed, cols, rows, angle=0.0):
    """Staggered grid initialization with optional rotation."""
    np.random.seed(seed)
    pts = []
    x_step = 1.0 / (cols + 0.5)
    y_step = 1.0 / (rows + 0.5)
    
    for r in range(rows + 1):
        for c in range(cols + 1):
            x = (c + 0.5) * x_step + (0.5 if r % 2 == 1 else 0) * x_step * 0.5
            y = (r + 0.5) * y_step
            if x <= 1.0 and y <= 1.0:
                pts.append([x, y])
            if len(pts) >= N:
                break
        if len(pts) >= N:
            break
    
    pts = np.array(pts[:N])
    
    if angle != 0.0:
        c = np.array([0.5, 0.5])
        pts = (pts - c) @ np.array([[np.cos(angle), -np.sin(angle)], [np.sin(angle), np.cos(angle)]]) + c
        pts = np.clip(pts, 0.02, 0.98)
    
    pts += np.random.uniform(-0.008, 0.008, pts.shape)
    pts = np.clip(pts, 0.02, 0.98)
    
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.1, 0.9, (1, 2))])
    
    return pts[:N]

def run_slsqp_optimize(x0, cons_func, bounds_obj, max_iter=15000):
    """Run SLSQP optimization and return result with LP refinement."""
    try:
        res = minimize(objective, x0, method='SLSQP', bounds=bounds_obj,
                       constraints={'type': 'ineq', 'fun': cons_func},
                       options={'maxiter': max_iter, 'ftol': 1e-14, 'disp': False})
        if not np.isnan(res.fun):
            c_opt = np.column_stack((res.x[0::3], res.x[1::3]))
            r_opt = solve_lp_radii(c_opt)
            if r_opt is not None:
                return res.x, c_opt, r_opt, np.sum(r_opt)
    except Exception:
        pass
    return None, None, None, -1.0

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
