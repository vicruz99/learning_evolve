

Looking at the past solutions, I see they're converging to ~2.634-2.635. The key issues are:
1. Getting stuck in similar local optima despite various perturbations
2. The constraint formulation (squared vs hypot) affects gradient quality near contacts
3. Need more aggressive topology exploration to escape local minima

My strategy:
- Use `np.hypot` in constraints for better conditioning at contact points
- More aggressive deflation with wider random subset targeting
- Additional initialization strategies (corner-focused, edge-focused)
- More polishing iterations with tighter tolerances
- Better random seed exploration to find diverse starting points
- Try swapping circle indices to break rigid contact networks

```python
import numpy as np
from scipy.optimize import minimize, linprog
import warnings
warnings.filterwarnings('ignore')

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Returns all inequality constraints >= 0 using hypot for better conditioning."""
    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]
    
    # Boundary constraints
    c = np.concatenate([
        xs - rs, 1.0 - xs - rs,
        ys - rs, 1.0 - ys - rs
    ])
    
    # Overlap constraints using hypot for stable gradients at contact
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    c = np.concatenate([c, np.hypot(dx[TRIL_IDX], dy[TRIL_IDX]) - dr[TRIL_IDX]])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-7, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)])
    return b

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist)
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-7)
    except Exception:
        pass
    return np.full(n, 0.01)

def project_x(x0):
    """Project variables to strictly satisfy bounds."""
    x0 = x0.copy()
    for i in range(N):
        r = max(1e-7, x0[3 * i + 2])
        x0[3 * i] = np.clip(x0[3 * i], r, 1.0 - r)
        x0[3 * i + 1] = np.clip(x0[3 * i + 1], r, 1.0 - r)
        x0[3 * i + 2] = r
    return x0

def make_hex_init(r0, angle=0.0):
    """Generates a hexagonal lattice initialization with optional rotation."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 15:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        
    pts = np.array(pts[:N + 15])
    
    if angle != 0.0:
        c_a, s_a = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c_a, -s_a], [s_a, c_a]]) + 0.5
        
    mask = (pts[:, 0] >= 0.01) & (pts[:, 0] <= 0.99) & (pts[:, 1] >= 0.01) & (pts[:, 1] <= 0.99)
    pts = pts[mask]
    
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.1, 0.9, (1, 2))])
    return pts[:N]

def make_force_init(seed, steps=250):
    """Force-directed layout to spread points evenly."""
    np.random.seed(seed)
    pts = np.random.uniform(0.12, 0.88, (N, 2))
    for step in range(steps):
        f = np.zeros_like(pts)
        lr = 0.015 * (1.0 - step / steps)
        for i in range(N):
            for j in range(i + 1, N):
                dx = pts[j] - pts[i]
                d = np.linalg.norm(dx)
                if d < 0.28 and d > 1e-6:
                    rep = 0.010 / (d**2 + 0.0008)
                    f[i] -= dx * rep / d
                    f[j] += dx * rep / d
            for dim in range(2):
                if pts[i, dim] < 0.08: f[i, dim] += 0.06
                elif pts[i, dim] > 0.92: f[i, dim] -= 0.06
        pts += f * lr
        pts = np.clip(pts, 0.02, 0.98)
    return pts

def make_corner_edge_init(seed):
    """Initialization focused on corners and edges."""
    np.random.seed(seed)
    pts = np.zeros((N, 2))
    r = 0.09 + np.random.uniform(-0.005, 0.005)
    
    # 4 corners
    pts[0] = [r, r]
    pts[1] = [1 - r, r]
    pts[2] = [r, 1 - r]
    pts[3] = [1 - r, 1 - r]
    
    # 4 edge midpoints
    pts[4] = [0.5, r]
    pts[5] = [0.5, 1 - r]
    pts[6] = [r, 0.5]
    pts[7] = [1 - r, 0.5]
    
    # Center
    pts[8] = [0.5, 0.5]
    idx = 9
    
    # Fill with staggered hexagonal pattern
    y = r + 0.045
    row = 0
    while idx < N and y + r <= 1.0:
        x_start = r + 0.06 if row % 2 == 0 else r + 0.14
        x = x_start
        while x + r <= 1.0 and idx < N:
            pts[idx] = [x, y]
            idx += 1
            x += 0.20
        y += 0.14
        row += 1
    
    for i in range(idx, N):
        pts[i] = np.random.uniform(0.15, 0.85, 2)
    
    pts += np.random.uniform(-0.003, 0.003, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def make_staggered_init(r0):
    """Creates a staggered grid pattern."""
    pts = []
    spacing_x = 2.0 * r0
    spacing_y = np.sqrt(3.0) * r0
    y = r0
    row = 0
    while len(pts) < N + 5:
        x_offset = r0 if row % 2 == 0 else 2.0 * r0
        x = x_offset
        while x <= 1.0 - r0:
            pts.append([x, y])
            x += spacing_x
        y += spacing_y
        row += 1
    return np.array(pts[:N])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    best_sum = -1.0
    best_x = None
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    
    # Phase 1: Generate diverse initial configurations
    inits = []
    
    # Extensive hexagonal lattice exploration with many rotations and scales
    for r0 in [0.082, 0.085, 0.088, 0.090, 0.093, 0.095, 0.098, 0.100, 0.103, 0.105, 0.108, 0.110]:
        for ang in np.linspace(-0.5, 0.5, 21):
            inits.append(make_hex_init(r0, ang))
    
    # Force-directed layouts with different seeds
    for s in range(25):
        inits.append(make_force_init(s))
    
    # Corner/edge focused
    for s in range(15):
        inits.append(make_corner_edge_init(s))
    
    # Staggered patterns
    for r0 in [0.088, 0.092, 0.096, 0.100]:
        pts = make_staggered_init(r0)
        if len(pts) >= N:
            inits.append(pts[:N])

    # Phase 2: Initial optimization sweep
    for pts in inits:
        x0 = np.zeros(3 * N)
        x0[0::3] = pts[:, 0]
        x0[1::3] = pts[:, 1]
        r_lp = solve_lp_radii(pts)
        x0[2::3] = r_lp * 0.995
        x0 = project_x(x0)
        
        # Add tiny noise to break exact symmetries
        x0 += np.random.normal(0, 1e-5, x0.shape)
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 20000, 'ftol': 1e-13, 'disp': False})
            if not np.isnan(res.fun):
                c_tmp = np.column_stack((res.x[0::3], res.x[1::3]))
                r_new = solve_lp_radii(c_tmp)
                s_val = np.sum(r_new)
                if s_val > best_sum:
                    best_sum = s_val
                    best_x = np.zeros(3 * N)
                    best_x[0::3] = c_tmp[:, 0]
                    best_x[1::3] = c_tmp[:, 1]
                    best_x[2::3] = r_new
        except Exception:
            pass

    # Phase 3: Aggressive topology search with deflation
    if best_x is not None:
        for cycle in range(120):
            x0 = best_x.copy()
            
            # Progressive shrink - starts aggressive, becomes gentler
            shrink = 0.55 + 0.45 * (cycle / 120.0)
            x0[2::3] *= shrink
            
            # Decaying noise
            noise_scale = 0.006 / (cycle + 1)
            x0[0::3] += np.random.normal(0, noise_scale, N)
            x0[1::3] += np.random.normal(0, noise_scale, N)
            
            # Random subset deflation to break rigid contact networks
            subset_size = max(4, N // 3)
            subset = np.random.choice(N, size=subset_size, replace=False)
            x0[2::3][subset] *= 0.35 + 0.15 * (cycle / 120.0)
            
            # Random circle swaps to break symmetries
            if cycle % 4 == 0:
                for _ in range(3):
                    i_sw, j_sw = np.random.choice(N, 2, replace=False)
                    x0[3*i_sw], x0[3*j_sw] = x0[3*j_sw], x0[3*i_sw]
                    x0[3*i_sw+1], x0[3*j_sw+1] = x0[3*j_sw+1], x0[3*i_sw+1]
                    x0[3*i_sw+2], x0[3*j_sw+2] = x0[3*j_sw+2], x0[3*i_sw+2]
            
            # Random translation of a subset
            if cycle % 6 == 0:
                trans_subset = np.random.choice(N, size=5, replace=False)
                trans_vec = np.random.normal(0, 0.005, (5, 2))
                for idx_pos, i in enumerate(trans_subset):
                    x0[3*i] += trans_vec[idx_pos, 0]
                    x0[3*i+1] += trans_vec[idx_pos, 1]
            
            x0 = project_x(x0)
            
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 12000, 'ftol': 1e-13, 'disp': False})
                if not np.isnan(res.fun):
                    c_tmp = np.column_stack((res.x[0::3], res.x[1::3]))
                    r_new = solve_lp_radii(c_tmp)
                    
                    x_p = np.zeros(3 * N)
                    x_p[0::3] = c_tmp[:, 0]
                    x_p[1::3] = c_tmp[:, 1]
                    x_p[2::3] = r_new * 0.997
                    x_p = project_x(x_p)
                    
                    s_val = np.sum(x_p[2::3])
                    if s_val > best_sum:
                        best_sum = s_val
                        best_x = x_p.copy()
                        
                        # Secondary polish after improvement
                        x0_sec = best_x.copy()
                        x0_sec[2::3] *= 0.990
                        x0_sec = project_x(x0_sec)
                        try:
                            res2 = minimize(objective, x0_sec, method='SLSQP', bounds=bounds,
                                            constraints=cons, options={'maxiter': 10000, 'ftol': 1e-13, 'disp': False})
                            if not np.isnan(res2.fun):
                                c_sec = np.column_stack((res2.x[0::3], res2.x[1::3]))
                                r_sec = solve_lp_radii(c_sec)
                                x_sec = np.zeros(3 * N)
                                x_sec[0::3] = c_sec[:, 0]
                                x_sec[1::3] = c_sec[:, 1]
                                x_sec[2::3] = r_sec
                                x_sec = project_x(x_sec)
                                s_sec = np.sum(x_sec[2::3])
                                if s_sec > best_sum:
                                    best_sum = s_sec
                                    best_x = x_sec.copy()
                        except Exception:
                            pass
            except Exception:
                pass

    # Phase 4: Rotation-based exploration from best solution
    if best_x is not None:
        best_centers = np.column_stack((best_x[0::3], best_x[1::3]))
        for ang_deg in np.linspace(-15, 15, 19):
            ang = np.deg2rad(ang_deg)
            c_a, s_a = np.cos(ang), np.sin(ang)
            rot_centers = (best_centers - 0.5) @ np.array([[c_a, -s_a], [s_a, c_a]]) + 0.5
            rot_centers = np.clip(rot_centers, 0.015, 0.985)
            
            r_rot = solve_lp_radii(rot_centers)
            if r_rot is not None:
                x_rot = np.zeros(3 * N)
                x_rot[0::3] = rot_centers[:, 0]
                x_rot[1::3] = rot_centers[:, 1]
                x_rot[2::3] = r_rot * 0.995
                x_rot = project_x(x_rot)
                
                try:
                    res = minimize(objective, x_rot, method='SLSQP', bounds=bounds, constraints=cons,
                                   options={'maxiter': 10000, 'ftol': 1e-13, 'disp': False})
                    if not np.isnan(res.fun):
                        c_rot = np.column_stack((res.x[0::3], res.x[1::3]))
                        r_final = solve_lp_radii(c_rot)
                        s_rot = np.sum(r_final)
                        if s_rot > best_sum:
                            best_sum = s_rot
                            best_x = np.zeros(3 * N)
                            best_x[0::3] = c_rot[:, 0]
                            best_x[1::3] = c_rot[:, 1]
                            best_x[2::3] = r_final
                except Exception:
                    pass

    # Phase 5: High-precision polish on best solution
    if best_x is not None:
        for polish_iter in range(15):
            x_polish = best_x.copy()
            x_polish[2::3] *= 0.995
            x_polish = project_x(x_polish)
            try:
                res = minimize(objective, x_polish, method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 25000, 'ftol': 1e-15, 'disp': False})
                if not np.isnan(res.fun):
                    c_pol = np.column_stack((res.x[0::3], res.x[1::3]))
                    r_pol = solve_lp_radii(c_pol)
                    x_pol = np.zeros(3 * N)
                    x_pol[0::3] = c_pol[:, 0]
                    x_pol[1::3] = c_pol[:, 1]
                    x_pol[2::3] = r_pol
                    x_pol = project_x(x_pol)
                    s_pol = np.sum(x_pol[2::3])
                    if s_pol > best_sum:
                        best_sum = s_pol
                        best_x = x_pol.copy()
            except Exception:
                pass

    # Fallback
    if best_x is None:
        best_x = np.zeros(3 * N)
        best_x[0::3] = np.tile(np.linspace(0.1, 0.9, 5), 6)[:N]
        best_x[1::3] = np.repeat(np.linspace(0.1, 0.9, 6), 5)[:N]
        best_x[2::3] = 0.06
        best_sum = np.sum(best_x[2::3])

    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3].copy()
    
    # Final LP squeeze to maximize radii for best centers
    final_r = solve_lp_radii(centers)
    if np.sum(final_r) > np.sum(radii) - 1e-7:
        radii = final_r.copy()

    # Final strict validation repair
    for _ in range(200):
        valid = True
        for i in range(N):
            if radii[i] < 0:
                valid = False
                break
            if centers[i, 0] < radii[i] - 1e-9 or centers[i, 0] > 1.0 - radii[i] + 1e-9:
                valid = False
                break
            if centers[i, 1] < radii[i] - 1e-9 or centers[i, 1] > 1.0 - radii[i] + 1e-9:
                valid = False
                break
        if valid:
            for i in range(N):
                for j in range(i + 1, N):
                    d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                    if d < radii[i] + radii[j] - 1e-9:
                        valid = False
                        break
                if not valid:
                    break
        if valid:
            break
        radii *= 0.9995
        for i in range(N):
            centers[i, 0] = np.clip(centers[i, 0], radii[i], 1.0 - radii[i])
            centers[i, 1] = np.clip(centers[i, 1], radii[i], 1.0 - radii[i])

    return centers, radii, float(np.sum(radii))
```
