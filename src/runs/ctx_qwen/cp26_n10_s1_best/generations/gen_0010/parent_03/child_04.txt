import numpy as np
from scipy.optimize import minimize, linprog
import warnings
warnings.filterwarnings('ignore')

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A = np.zeros((num_ineq, n))
    b = np.zeros(num_ineq)
    
    idx = 0
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A[idx, i] = 1.0
        b[idx] = max(0.0, lim)
        idx += 1
        
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = max(0.0, dist)
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-7)
    except Exception:
        pass
    return np.full(n, 0.01)

def center_objective_and_grad(vars, radii):
    """Objective: minimize overlap and boundary penalties. Returns (val, grad)."""
    n = len(radii)
    xs = vars[0::2]
    ys = vars[1::2]
    
    val = 0.0
    grad = np.zeros_like(vars)
    
    for i in range(n):
        for j in range(i + 1, n):
            dx = xs[i] - xs[j]
            dy = ys[i] - ys[j]
            dist = np.hypot(dx, dy) + 1e-9
            sum_r = radii[i] + radii[j]
            gap = sum_r - dist
            if gap > 0.0:
                val += gap**2
                term = -2.0 * gap * (dx / dist)
                term_y = -2.0 * gap * (dy / dist)
                grad[2*i] += term
                grad[2*i+1] += term_y
                grad[2*j] -= term
                grad[2*j+1] -= term_y
                
    for i in range(n):
        r = radii[i]

            gap = r - xs[i]
            val += gap**2
            grad[2*i] -= 2.0 * gap

            gap = xs[i] - (1.0 - r)
            val += gap**2
            grad[2*i] += 2.0 * gap

            val += gap**2
            grad[2*i+1] -= 2.0 * gap
        if ys[i] > 1.0 - r:
            gap = ys[i] - (1.0 - r)
            val += gap**2
            grad[2*i+1] += 2.0 * gap
            
    return val, grad

def optimize_centers(radii, centers_init, steps=200):
    """Optimize centers to minimize overlaps for fixed radii."""
    vars0 = np.zeros(2 * N)
    vars0[0::2] = centers_init[:, 0]
    vars0[1::2] = centers_init[:, 1]
    
    bounds = []
    for r in radii:
        lb = max(0.0, r)
        ub = min(1.0, 1.0 - r)
        bounds.extend([(lb, ub), (lb, ub)])
        
    res = minimize(center_objective_and_grad, vars0, args=(radii,),
                   method='L-BFGS-B', jac=True, bounds=bounds,
                   options={'maxiter': steps, 'ftol': 1e-15, 'maxcor': 10})
    if res.success:
        return np.column_stack((res.x[0::2], res.x[1::2]))
    return centers_init

def make_hex_init(r0, angle=0.0):
    """Generates a hexagonal lattice initialization with optional rotation."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        
    pts = np.array(pts[:N + 10])
    
    if angle != 0.0:
        c_a, s_a = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c_a, -s_a], [s_a, c_a]]) + 0.5
        
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.1, 0.9, (1, 2))])
    return pts[:N]

def make_force_init(seed):
    """Force-directed layout to spread points evenly."""
    np.random.seed(seed)
    pts = np.random.uniform(0.15, 0.85, (N, 2))
    for _ in range(150):
        f = np.zeros_like(pts)
        for i in range(N):
            for j in range(i + 1, N):
                dx = pts[j] - pts[i]
                d = np.linalg.norm(dx)
                if d < 0.25 and d > 1e-6:
                    rep = 0.005 / (d**2 + 0.001)
                    f[i] -= dx * rep / d
                    f[j] += dx * rep / d
            for dim in range(2):
                if pts[i, dim] < 0.12: f[i, dim] += 0.03
                elif pts[i, dim] > 0.88: f[i, dim] -= 0.03
        pts += f * 0.03
        pts = np.clip(pts, 0.02, 0.98)
    return pts
