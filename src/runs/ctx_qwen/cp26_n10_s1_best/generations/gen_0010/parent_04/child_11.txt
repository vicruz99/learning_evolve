import numpy as np
from scipy.optimize import minimize, linprog

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Returns all inequality constraints >= 0 (vectorized, using hypot for stability)."""
    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    c = np.concatenate([
        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs
    ])
    
    # Overlap constraints: dist >= r_i + r_j (using hypot for stable gradients)
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    c = np.concatenate([c, np.hypot(dx[TRIL_IDX], dy[TRIL_IDX]) - dr[TRIL_IDX]])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-7, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)])
    return b

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist)
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-8)
    except Exception:
        pass
    return np.full(n, 0.05)

def project(x):
    """Project variables to strictly satisfy bounds."""
    x = x.copy()
    for i in range(N):
        r = max(1e-6, x[3 * i + 2])
        x[3 * i] = np.clip(x[3 * i], r, 1.0 - r)
        x[3 * i + 1] = np.clip(x[3 * i + 1], r, 1.0 - r)
        x[3 * i + 2] = r
    return x

def make_hex_init(r0, ang):
    """Generates a hexagonal lattice initialization with optional rotation."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0 and len(pts) < N + 10:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
    pts = np.array(pts[:N + 10])
    
    if ang != 0.0:
        c, s = np.cos(ang), np.sin(ang)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
        
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.1, 0.9, (1, 2))])
    return pts[:N]

def make_force_init(seed):
    """Force-directed layout to spread points evenly and push to boundaries."""
    np.random.seed(seed)
    pts = np.random.uniform(0.15, 0.85, (N, 2))
    for step in range(300):
        f = np.zeros_like(pts)
        for i in range(N):
            for j in range(i + 1, N):
                dx = pts[j] - pts[i]
                d = np.hypot(dx[0], dx[1])
                if d < 0.25 and d > 1e-5:
                    rep = 0.008 / (d**2 + 0.001)
                    f[i] -= dx * rep / d
                    f[j] += dx * rep / d
            for dim in range(2):
                if pts[i, dim] < 0.12: f[i, dim] += 0.05
                elif pts[i, dim] > 0.88: f[i, dim] -= 0.05
        lr = 0.04 * (1.0 - step / 300.0)
        pts += f * lr
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def make_corner_init():
    """Generates initialization with circles in corners and edges."""
    pts = np.zeros((N, 2))
    r = 0.095
    # Four corners
    pts[0] = [r, r]
    pts[1] = [1.0 - r, r]
    pts[2] = [r, 1.0 - r]
    pts[3] = [1.0 - r, 1.0 - r]
    # Edge midpoints
    pts[4] = [0.5, r]
    pts[5] = [0.5, 1.0 - r]
    pts[6] = [r, 0.5]
    pts[7] = [1.0 - r, 0.5]
    # Center
    pts[8] = [0.5, 0.5]
    idx = 9
    
    y = r + 0.04
    row = 0
    while idx < N and y + r <= 1.0 - 0.01:
        x = r + 0.04 if row % 2 == 0 else 1.0 - r - 0.04
        while x <= 1.0 - r and idx < N:
            pts[idx] = [x, y]
            idx += 1
            x += 0.22
        y += 0.14
        row += 1
    
    for i in range(idx, N):
        pts[i] = np.random.uniform(0.1, 0.9, 2)
    pts += np.random.uniform(-0.005, 0.005, pts.shape)
    return np.clip(pts, 0.03, 0.97)

def make_grid_init(seed):
    """Generates perturbed grid initialization."""
    np.random.seed(seed)
    pts = np.zeros((N, 2))
    idx = 0
    for i in range(6):
        for j in range(5):
            if idx < N:
                pts[idx] = [0.1 + j * 0.18, 0.1 + i * 0.16]
                idx += 1
    pts += np.random.uniform(-0.012, 0.012, pts.shape)
    return np.clip(pts, 0.04, 0.96)

def run_slsqp_then_lp(x0, bounds, max_iter=15000):
    """Run SLSQP followed by LP radii optimization, return best result."""
    x_curr = x0.copy()
    
    try:
        res = minimize(objective, x_curr, method='SLSQP', bounds=bounds,
                       constraints={'type': 'ineq', 'fun': constraints},
                       options={'maxiter': max_iter, 'ftol': 1e-13, 'disp': False})
        if not np.isnan(res.fun):
            centers_tmp = np.column_stack((res.x[0::3], res.x[1::3]))
            r_new = solve_lp_radii(centers_tmp)
            if r_new is not None:
                x_curr = res.x.copy()
                x_curr[2::3] = r_new
                x_curr = project(x_curr)
    except Exception:
        pass
    return x_curr

def run_packing() -> tuple:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    bounds = get_bounds()
    
    best_sum = -1.0
    best_x = None
    
    # Phase 1: Generate diverse initial configurations
    inits = []
    
    # Hexagonal lattices with varied scales and rotations
    for r0 in [0.085, 0.088, 0.090, 0.092, 0.095, 0.098, 0.100, 0.103, 0.106]:
        for ang in np.linspace(-0.35, 0.35, 17):
            inits.append(make_hex_init(r0, ang))
    
    # Force-directed layouts
    for s in range(20):
        inits.append(make_force_init(s))
    
    # Corner-focused patterns
    for _ in range(8):
        inits.append(make_corner_init())
    
    # Grid patterns
    for s in range(6):
        inits.append(make_grid_init(s))
    
    # Phase 2: Multi-start optimization with LP refinement
    for pts in inits:
        x0 = np.zeros(3 * N)
        x0[0::3] = pts[:, 0]
        x0[1::3] = pts[:, 1]
        r_lp = solve_lp_radii(pts)
        x0[2::3] = np.maximum(r_lp * 0.995, 1e-6)
        x0 = project(x0)
        
        x_opt = run_slsqp_then_lp(x0, bounds, max_iter=18000)
        
        s_val = np.sum(x_opt[2::3])
        c_check = constraints(x_opt)
        if s_val > best_sum and np.min(c_check) >= -1e-5:
            best_sum = s_val
            best_x = x_opt.copy()
    
    # Phase 3: Iterative Deflation & Refinement to Escape Local Minima
    if best_x is not None:
        for cycle in range(100):
            x0 = best_x.copy()
            
            # Gradual recovery of radii with noise
            shrink = 0.80 + 0.20 * (cycle / 100.0)
            x0[2::3] *= shrink
            
            # Cooling noise on centers
            noise = 0.003 * (0.95 ** (cycle / 10.0))
            x0[0::3] += np.random.normal(0, noise, N)
            x0[1::3] += np.random.normal(0, noise, N)
            
            # Aggressively deflate random subset to break rigid contact networks
            k = np.random.randint(4, 10)
            subset = np.random.choice(N, size=k, replace=False)
            x0[subset * 3 + 2] *= 0.70
            
            x0 = project(x0)
            
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                               constraints={'type': 'ineq', 'fun': constraints},
                               options={'maxiter': 15000, 'ftol': 1e-13, 'disp': False})
                if not np.isnan(res.fun):
                    c_tmp = np.column_stack((res.x[0::3], res.x[1::3]))
                    r_new = solve_lp_radii(c_tmp)
                    if r_new is not None:
                        x0 = res.x.copy()
                        x0[2::3] = r_new * 0.999
                        x0 = project(x0)
                        s_val = np.sum(x0[2::3])
                        c_check = constraints(x0)
                        if s_val > best_sum and np.min(c_check) >= -1e-5:
                            best_sum = s_val
                            best_x = x0.copy()
            except Exception:
                pass
    
    # Phase 4: High-precision polish
    if best_x is not None:

            x_pol = best_x.copy()

            try:

                               constraints={'type': 'ineq', 'fun': constraints},

                if not np.isnan(res.fun):

                    r_pol = solve_lp_radii(c_pol)
                    if r_pol is not None:
                        x_pol = res.x.copy()
                        x_pol[2::3] = r_pol
                        x_pol = project(x_pol)
                        s_pol = np.sum(x_pol[2::3])

                            best_x = x_pol.copy()
            except Exception:
                pass
    
    # Fallback initialization
    if best_x is None:
        best_x = np.zeros(3 * N)
        best_x[0::3] = np.tile(np.linspace(0.1, 0.9, 5), 6)[:N]
        best_x[1::3] = np.repeat(np.linspace(0.1, 0.9, 6), 5)[:N]
        best_x[2::3] = 0.06
    
    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3].copy()
    
    # Final LP squeeze to maximize radii for the best centers found
    final_r = solve_lp_radii(centers)
    if final_r is not None and np.sum(final_r) > np.sum(radii) - 1e-8:
        radii = final_r.copy()

    # Final strict validation repair against 1e-12 tolerance
    for _ in range(100):
        valid = True
        for i in range(N):
            if radii[i] < 0 or centers[i, 0] < radii[i] - 1e-9 or centers[i, 0] > 1.0 - radii[i] + 1e-9 or \

                valid = False
                break
        if valid:
            for i in range(N):
                for j in range(i + 1, N):
                    d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                    if d < radii[i] + radii[j] - 1e-9:
                        valid = False
                        break

                    break
        if valid:
            break
            
        radii *= 0.9998
        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)

    return centers, radii, float(np.sum(radii))
