import numpy as np
from scipy.optimize import minimize, linprog
import warnings
warnings.filterwarnings('ignore')

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Returns all inequality constraints >= 0 (vectorized)."""
    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    c = np.concatenate([
        xs - rs, 1.0 - xs - rs,
        ys - rs, 1.0 - ys - rs
    ])
    
    # Overlap constraints: dist >= r_i + r_j
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    # np.hypot provides stable, non-zero gradients at contact points
    c = np.concatenate([c, np.hypot(dx[TRIL_IDX], dy[TRIL_IDX]) - dr[TRIL_IDX]])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-7, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)])
    return b

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist)
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-8)
    except Exception:
        pass
    return np.full(n, 0.05)

def force_optimize_centers(centers, radii, steps=100):
    """Force-directed relaxation to push centers apart and resolve overlaps."""
    pts = centers.copy()
    r = radii.copy()
    for step in range(steps):
        f = np.zeros_like(pts)
        lr = 0.008 * (1.0 - step / steps)
        for i in range(N):
            for j in range(i + 1, N):
                dx = pts[j, 0] - pts[i, 0]
                dy = pts[j, 1] - pts[i, 1]
                d = np.hypot(dx, dy)
                if d < 1e-6:
                    d = 1e-6
                    dx, dy = np.random.randn(2) * d
                req = r[i] + r[j]
                if d < req:
                    # Strong repulsion proportional to overlap squared
                    push = (req - d) ** 2 / (d + 1e-6)
                    f[i, 0] -= push * dx / d
                    f[i, 1] -= push * dy / d
                    f[j, 0] += push * dx / d
                    f[j, 1] += push * dy / d
                    
            # Boundary repulsion
            for dim in range(2):
                val = pts[i, dim]
                if val < r[i]:
                    f[i, dim] += (r[i] - val)**2 * 50.0
                elif val > 1.0 - r[i]:
                    f[i, dim] -= (val - (1.0 - r[i]))**2 * 50.0
                    
        pts += f * lr
        pts = np.clip(pts, 1e-5, 1.0 - 1e-5)
    return pts

def project_x(x0):
    """Project variables to strictly satisfy bounds."""
    x0 = x0.copy()
    for i in range(N):
        r = max(1e-7, x0[3 * i + 2])
        x0[3 * i] = np.clip(x0[3 * i], r, 1.0 - r)
        x0[3 * i + 1] = np.clip(x0[3 * i + 1], r, 1.0 - r)
        x0[3 * i + 2] = r
    return x0

def make_init_grid():
    """Generates a uniform grid initialization adapted to the square."""
    pts = []
    for i in range(5):
        for j in range(5):
            pts.append([0.1 + j * 0.2, 0.1 + i * 0.2])
    pts.append([0.5, 0.5])
    return np.array(pts[:N]) + np.random.uniform(-0.01, 0.01, (N, 2))

def make_init_hex(angle=0.0):
    """Generates a rotated hexagonal lattice initialization."""
    pts = []
    r0 = 0.095
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0 and len(pts) < N + 10:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
    pts = np.array(pts[:N + 10])
    
    if angle != 0.0:
        c_a, s_a = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c_a, -s_a], [s_a, c_a]]) + 0.5
        
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.2, 0.8, (1, 2))])
    return np.clip(pts[:N] + np.random.uniform(-0.002, 0.002, (N, 2)), 0.02, 0.98)

    for pts in inits:

        radii = solve_lp_radii(centers)

        x0 = np.zeros(3 * N)

        x0 = project_x(x0)
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 20000, 'ftol': 1e-14, 'disp': False})
            if not np.isnan(res.fun):
                c_tmp = np.column_stack((res.x[0::3], res.x[1::3]))
                r_new = solve_lp_radii(c_tmp)
                s_val = np.sum(r_new)

            pass
