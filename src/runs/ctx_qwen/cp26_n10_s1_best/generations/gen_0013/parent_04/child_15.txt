

<strategy>
1. **Initialization Diversity**: Generate multiple rotated hexagonal lattices, force-directed layouts, and corner-focused layouts. This covers different topological starts.
2. **Joint Optimization (SLSQP)**: Optimize centers and radii simultaneously using SLSQP with smooth constraints (`hypot` for distances). Initialize radii via LP for a feasible head-start.
3. **LP Radius Solving**: After each optimization step, fix centers and solve the LP exactly to maximize radii. This is crucial as LP gives the true maximum radii for any fixed center configuration.
4. **Aggressive Deflation & Perturbation**: To escape local minima, systematically shrink radii (deflation), add Gaussian noise to centers, and occasionally swap circle positions to break rigid contact networks. Run this in a cooling schedule.
5. **Rotation Search**: The square boundary favors specific lattice rotations. Test rotations of the best configuration around the center to find boundary-optimal alignments.
6. **Coordinate Descent Polish**: Fine-tune by moving each circle individually in multiple directions and checking if the LP-solved radii sum improves. This exploits the non-smooth nature of the LP objective landscape.
7. **Strict Validation & Repair**: Finally, ensure all constraints are met within `1e-12` tolerance by minimally shrinking radii if necessary.
</strategy>
```python
import numpy as np
from scipy.optimize import minimize, linprog
import warnings
warnings.filterwarnings('ignore')

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Returns all inequality constraints >= 0 (vectorized)."""
    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    c = np.concatenate([
        xs - rs, 1.0 - xs - rs,
        ys - rs, 1.0 - ys - rs
    ])
    
    # Overlap constraints: dist >= r_i + r_j
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    # np.hypot provides stable, non-zero gradients at contact points
    c = np.concatenate([c, np.hypot(dx[TRIL_IDX], dy[TRIL_IDX]) - dr[TRIL_IDX]])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-7, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)])
    return b

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist)
            idx += 1
            
    bounds_lp = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_lp, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-8)
    except Exception:
        pass
    return np.full(n, 0.04)

def project_x(x0):
    """Project variables to strictly satisfy bounds."""
    x0 = x0.copy()
    for i in range(N):
        r = max(1e-7, x0[3 * i + 2])
        x0[3 * i] = np.clip(x0[3 * i], r, 1.0 - r)
        x0[3 * i + 1] = np.clip(x0[3 * i + 1], r, 1.0 - r)
        x0[3 * i + 2] = r
    return x0

def make_hex_init(angle=0.0, scale=1.0):
    """Generates a rotated and scaled hexagonal lattice initialization."""
    pts = []
    r0 = 0.092 * scale
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0 and len(pts) < N + 10:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
    pts = np.array(pts[:N + 10])
    
    if angle != 0.0:
        c_a, s_a = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c_a, -s_a], [s_a, c_a]]) + 0.5
        
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.2, 0.8, (1, 2))])
        
    return np.clip(pts[:N] + np.random.uniform(-0.003, 0.003, (N, 2)), 0.02, 0.98)

def make_force_init(seed):
    """Force-directed layout to spread points evenly and push to boundaries."""
    np.random.seed(seed)
    pts = np.random.uniform(0.15, 0.85, (N, 2))
    for step in range(300):
        f = np.zeros_like(pts)
        lr = 0.015 * (1.0 - step / 300.0)
        for i in range(N):
            for j in range(i + 1, N):
                dx = pts[j] - pts[i]
                d = np.linalg.norm(dx)
                if d < 0.28 and d > 1e-6:
                    rep = 0.006 / (d**2 + 0.0005)
                    f[i] -= dx * rep / d
                    f[j] += dx * rep / d
            for dim in range(2):
                if pts[i, dim] < 0.12: f[i, dim] += 0.05
                elif pts[i, dim] > 0.88: f[i, dim] -= 0.05
        pts += f * lr
        pts = np.clip(pts, 0.02, 0.98)
    return pts

def make_corner_init(seed):
    """Generates an initialization focused on corners and edges."""
    np.random.seed(seed)
    pts = np.zeros((N, 2))
    r = 0.09
    pts[0] = [r, r]
    pts[1] = [1-r, r]
    pts[2] = [r, 1-r]
    pts[3] = [1-r, 1-r]
    pts[4] = [0.5, r]
    pts[5] = [0.5, 1-r]
    pts[6] = [r, 0.5]
    pts[7] = [1-r, 0.5]
    idx = 8
    y = r + 0.06
    while idx < N and y + r <= 1.0:
        x = r + 0.06
        while x + r <= 1.0 and idx < N:
            pts[idx] = [x, y]
            idx += 1
            x += 2 * r
        y += np.sqrt(3) * r
    while idx < N:
        pts[idx] = np.random.uniform(0.15, 0.85, 2)
        idx += 1
    pts += np.random.uniform(-0.005, 0.005, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def rotate_config(centers, angle):
    """Rotate a configuration around the center."""
    c_a, s_a = np.cos(angle), np.sin(angle)
    rot = (centers - 0.5) @ np.array([[c_a, -s_a], [s_a, c_a]]) + 0.5
    return np.clip(rot, 0.01, 0.99)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    best_sum = -1.0
    best_x = None
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    
    # Phase 1: Diverse Initial Configurations
    inits = []
    # Rotated hexagonal lattices with various scales
    for scale in [0.95, 1.0, 1.05, 1.1]:
        for ang in np.linspace(-0.35, 0.35, 13):
            inits.append(make_hex_init(ang, scale))
    # Force-directed random spreads
    for s in range(12):
        inits.append(make_force_init(s))
    # Corner/Edge focused
    for s in range(8):
        inits.append(make_corner_init(s))

    # Phase 2: Multi-start Optimization with LP Refinement
    for pts in inits:
        x0 = np.zeros(3 * N)
        x0[0::3] = pts[:, 0]
        x0[1::3] = pts[:, 1]
        
        r_lp = solve_lp_radii(pts)
        x0[2::3] = r_lp * 0.95
        x0 = project_x(x0)
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 20000, 'ftol': 1e-14, 'disp': False})
            if not np.isnan(res.fun):
                c_tmp = np.column_stack((res.x[0::3], res.x[1::3]))
                r_new = solve_lp_radii(c_tmp)
                s_val = np.sum(r_new)
                if s_val > best_sum:
                    best_sum = s_val
                    best_x = res.x.copy()
                    best_x[2::3] = r_new
        except Exception:
            pass

    # Phase 3: Aggressive Topology Search with Deflation & Swaps
    if best_x is not None:
        for cyc in range(150):
            x0 = best_x.copy()
            
            # Cooling perturbation & deflation
            noise_scale = 0.004 * (1.0 - cyc / 150.0)
            shrink = 0.65 + 0.25 * (cyc / 150.0)
            
            subset_size = max(4, N // 3)
            subset = np.random.choice(N, size=subset_size, replace=False)
            
            x0[subset * 3] += np.random.normal(0, noise_scale, subset_size)
            x0[subset * 3 + 1] += np.random.normal(0, noise_scale, subset_size)
            x0[subset * 3 + 2] *= shrink
            
            # Position swap to break rigid contact networks
            if cyc % 3 == 0 and cyc > 0:
                i, j = np.random.choice(N, 2, replace=False)
                x0[[i*3, j*3]] = x0[[j*3, i*3]]
                x0[[i*3+1, j*3+1]] = x0[[j*3+1, i*3+1]]
                
            x0 = project_x(x0)
            
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 10000, 'ftol': 1e-14, 'disp': False})
                if not np.isnan(res.fun):
                    c_tmp = np.column_stack((res.x[0::3], res.x[1::3]))
                    r_new = solve_lp_radii(c_tmp)
                    
                    x_p = res.x.copy()
                    x_p[2::3] = r_new * 0.999
                    x_p = project_x(x_p)
                    
                    s_val = np.sum(x_p[2::3])
                    c_vals = constraints(x_p)
                    if np.min(c_vals) >= -1e-6 and s_val > best_sum:
                        best_sum = s_val
                        best_x = x_p.copy()
            except Exception:
                pass

    # Phase 4: Rotation Search from best
    if best_x is not None:
        best_centers = np.column_stack((best_x[0::3], best_x[1::3]))
        for ang_deg in np.linspace(-15, 15, 15):
            ang = np.deg2rad(ang_deg)
            rot_centers = rotate_config(best_centers, ang)
            
            x_rot = np.zeros(3 * N)
            x_rot[0::3] = rot_centers[:, 0]
            x_rot[1::3] = rot_centers[:, 1]
            r_rot = solve_lp_radii(rot_centers)
            x_rot[2::3] = r_rot * 0.98
            x_rot = project_x(x_rot)
            
            try:
                res = minimize(objective, x_rot, method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 8000, 'ftol': 1e-14, 'disp': False})
                if not np.isnan(res.fun):
                    c_rot = np.column_stack((res.x[0::3], res.x[1::3]))
                    r_final = solve_lp_radii(c_rot)
                    s_val = np.sum(r_final)
                    if s_val > best_sum:
                        best_sum = s_val
                        best_x = res.x.copy()
                        best_x[2::3] = r_final
            except Exception:
                pass

    # Phase 5: Coordinate Descent Fine-Tuning
    if best_x is not None:
        curr_centers = np.column_stack((best_x[0::3], best_x[1::3]))
        improved = True
        while improved:
            improved = False
            for i in range(N):
                best_local_sum = np.sum(best_x[2::3])
                # Try displacements in 8 directions + variations
                for dx in [-0.006, -0.003, -0.001, 0.001, 0.003, 0.006]:
                    for dy in [-0.006, -0.003, -0.001, 0.001, 0.003, 0.006]:
                        if dx == 0 and dy == 0:
                            continue
                        new_centers = curr_centers.copy()
                        new_centers[i, 0] = np.clip(curr_centers[i, 0] + dx, 0.01, 0.99)
                        new_centers[i, 1] = np.clip(curr_centers[i, 1] + dy, 0.01, 0.99)
                        
                        r_new = solve_lp_radii(new_centers)
                        s_new = np.sum(r_new)
                        if s_new > best_local_sum + 1e-8:
                            curr_centers = new_centers
                            best_local_sum = s_new
                            best_x[0::3] = curr_centers[:, 0]
                            best_x[1::3] = curr_centers[:, 1]
                            best_x[2::3] = r_new
                            best_sum = s_new
                            improved = True
                            break
                    if improved:
                        break

    # Fallback initialization
    if best_x is None:
        best_x = np.zeros(3 * N)
        best_x[0::3] = np.tile(np.linspace(0.1, 0.9, 5), 6)[:N]
        best_x[1::3] = np.repeat(np.linspace(0.1, 0.9, 6), 5)[:N]
        best_x[2::3] = 0.06
        best_sum = np.sum(best_x[2::3])

    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3].copy()
    
    # Final LP squeeze to maximize radii for the best centers found
    final_r = solve_lp_radii(centers)
    if np.sum(final_r) > np.sum(radii) - 1e-7:
        radii = final_r.copy()

    # Final strict validation repair against 1e-12 tolerance
    for _ in range(100):
        valid = True
        for i in range(N):
            if radii[i] < 0 or centers[i, 0] < radii[i] - 1e-9 or centers[i, 0] > 1.0 - radii[i] + 1e-9 or \
               centers[i, 1] < radii[i] - 1e-9 or centers[i, 1] > 1.0 - radii[i] + 1e-9:
                valid = False
                break
        if valid:
            for i in range(N):
                for j in range(i + 1, N):
                    d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                    if d < radii[i] + radii[j] - 1e-9:
                        valid = False
                        break
                if not valid:
                    break
        if valid:
            break
            
        radii *= 0.9995
        for i in range(N):
            centers[i, 0] = np.clip(centers[i, 0], radii[i], 1.0 - radii[i])
            centers[i, 1] = np.clip(centers[i, 1], radii[i], 1.0 - radii[i])

    return centers, radii, float(np.sum(radii))
```
