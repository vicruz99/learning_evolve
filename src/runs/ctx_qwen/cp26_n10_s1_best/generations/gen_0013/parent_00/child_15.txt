import numpy as np
from scipy.optimize import minimize, linprog
import warnings
warnings.filterwarnings('ignore')

N = 26
TRIL_IDX = np.tril_indices(N, -1)
TRIU_IDX = np.triu_indices(N, 1)

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-7, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)])
    return b

def constraints_slsqp(x):
    """Vectorized inequality constraints g(x) >= 0 using squared distances for smooth gradients."""
    xs, ys, rs = x[0::3], x[1::3], x[2::3]
    
    # Boundary constraints
    c = np.concatenate([
        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs
    ])
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    c = np.concatenate([c, dx[TRIL_IDX]**2 + dy[TRIL_IDX]**2 - dr[TRIL_IDX]**2])
    return c

def objective_slsqp(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist)
            idx += 1
            
    bounds_lp = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_lp, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-8)
    except Exception:
        pass
    return np.full(n, 0.05)

def center_penalty(c_flat, radii):
    """Smooth penalty function for centers given fixed radii."""
    c = c_flat.reshape(-1, 2)
    x, y, r = c[:, 0], c[:, 1], radii
    p = 0.0
    # Boundary penalties
    p += np.sum(np.maximum(r - x, 0.0)**2)
    p += np.sum(np.maximum(x - (1.0 - r), 0.0)**2)
    p += np.sum(np.maximum(r - y, 0.0)**2)
    p += np.sum(np.maximum(y - (1.0 - r), 0.0)**2)
    
    # Overlap penalties
    dx = c[:, 0, None] - c[:, 0]
    dy = c[:, 1, None] - c[:, 1]
    d = np.hypot(dx, dy)
    gap = d - r[:, None] - r[None, :]
    p += np.sum(np.maximum(-gap[TRIU_IDX], 0.0)**2)
    return p

def opt_centers_lbfgs(centers, radii):
    """Optimize centers for fixed radii using L-BFGS-B."""
    bounds = [(0.001, 0.999)] * (2 * N)
    res = minimize(center_penalty, centers.flatten(), args=(radii,),
                   method='L-BFGS-B', bounds=bounds, options={'maxiter': 3000, 'ftol': 1e-13, 'gtol': 1e-12})
    return res.x.reshape(-1, 2)

def make_hex_init(r0, angle):
    """Generates a rotated hexagonal lattice initialization."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0 and len(pts) < N + 10:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        
    pts = np.array(pts[:N + 10])
    
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
        
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    
    # Pad with random if needed
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.1, 0.9, (1, 2))])
        
    # Break exact symmetries
    pts += np.random.uniform(-0.001, 0.001, pts.shape)
    return pts[:N]

def make_force_init(seed):
    """Force-directed layout to spread points evenly and push to boundaries."""
    np.random.seed(seed)
    pts = np.random.uniform(0.15, 0.85, (N, 2))
    for step in range(200):
        f = np.zeros_like(pts)
        lr = 0.04 * (1.0 - step / 200.0)
        for i in range(N):
            for j in range(i + 1, N):
                dx = pts[j] - pts[i]
                d = np.hypot(dx[0], dx[1])
                if d < 0.25 and d > 1e-5:
                    rep = 0.012 / (d**2 + 0.001)
                    f[i] -= dx * rep / d
                    f[j] += dx * rep / d
            for dim in range(2):
                if pts[i, dim] < 0.12: f[i, dim] += 0.05
                elif pts[i, dim] > 0.88: f[i, dim] -= 0.05
        pts += f * lr
        pts = np.clip(pts, 0.03, 0.97)
    return pts

def project_to_feasible(x):
    """Project variables to strictly satisfy bounds."""
    x = x.copy()
    for i in range(N):
        r = max(1e-6, x[3 * i + 2])
        x[3 * i] = np.clip(x[3 * i], r, 1.0 - r)
        x[3 * i + 1] = np.clip(x[3 * i + 1], r, 1.0 - r)
        x[3 * i + 2] = r
    return x

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    np.random.seed(42)
    bounds_obj = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints_slsqp}
    
    best_sum = -1.0
    best_x = None
    
    # Phase 1: Diverse Initial Configurations
    inits = []
    # Hexagonal lattices with varied base radii and rotations
    for r0 in [0.085, 0.090, 0.095, 0.100]:
        for ang in np.linspace(-0.4, 0.4, 13):
            inits.append(make_hex_init(r0, ang))
    # Force-directed spreads
    for s in range(15):
        inits.append(make_force_init(s))
    # Pure random starts
    for s in range(10):
        np.random.seed(s + 500)
        inits.append(np.random.uniform(0.15, 0.85, (N, 2)))

    # Phase 2: Alternating LP + L-BFGS-B Optimization
    for c_init in inits:
        c = c_init.copy()
        r = solve_lp_radii(c)
        
        # Alternating loop
        for _ in range(12):
            r = solve_lp_radii(c)
            c = opt_centers_lbfgs(c, r)
            
        # SLSQP Polish
        x0 = np.zeros(3 * N)
        x0[0::3] = c[:, 0]
        x0[1::3] = c[:, 1]
        x0[2::3] = np.maximum(r * 0.995, 1e-6)
        x0 = project_to_feasible(x0)
        
        try:
            res = minimize(objective_slsqp, x0, method='SLSQP', bounds=bounds_obj, constraints=cons,
                           options={'maxiter': 12000, 'ftol': 1e-14, 'disp': False})
            if not np.isnan(res.fun):
                c_p = np.column_stack((res.x[0::3], res.x[1::3]))
                r_p = solve_lp_radii(c_p)
                if r_p is not None:
                    s_val = np.sum(r_p)
                    if s_val > best_sum:
                        best_sum = s_val
                        best_x = res.x.copy()
                        best_x[2::3] = r_p
        except Exception:
            pass

    # Phase 3: Deflation & Perturbation to Escape Local Minima
    if best_x is not None:
        for cycle in range(60):
            x0 = best_x.copy()
            
            # Cooling perturbation
            noise = 0.0025 * (0.91 ** cycle)
            x0[0::3] += np.random.normal(0, noise, N)
            x0[1::3] += np.random.normal(0, noise, N)
            
            # Deflate a random subset to break rigid contact networks
            k = np.random.randint(3, 8)
            subset = np.random.choice(N, size=k, replace=False)
            x0[subset * 3 + 2] *= 0.85
            
            x0 = project_to_feasible(x0)
            
            # Quick alternating polish
            c = np.column_stack((x0[0::3], x0[1::3]))
            r = x0[2::3]

                c = opt_centers_lbfgs(c, r)
                
            x_p = np.zeros(3 * N)
            x_p[0::3] = c[:, 0]
            x_p[1::3] = c[:, 1]

                    r_p = solve_lp_radii(c_p)
                    if r_p is not None:

                            best_x[2::3] = r_p
            except Exception:
                pass

    # Phase 4: Micro Coordinate Descent Fine-Tuning
    if best_x is not None:

        improved = True
        while improved:
            improved = False
            for i in range(N):
                best_local_sum = np.sum(best_x[2::3])

                        if s_new > best_local_sum + 1e-9:
                            curr_centers = new_centers
                            best_local_sum = s_new
                            best_x[0::3] = curr_centers[:, 0]
                            best_x[1::3] = curr_centers[:, 1]
                            best_x[2::3] = r_new
                            best_sum = s_new
                            improved = True
                            break
                    if improved: break

    # Fallback (should not be reached)
    if best_x is None:
        best_x = np.zeros(3 * N)
        best_x[0::3] = np.tile(np.linspace(0.1, 0.9, 5), 6)[:N]
        best_x[1::3] = np.repeat(np.linspace(0.1, 0.9, 6), 5)[:N]

    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3].copy()

    for _ in range(100):

                valid = False

                for j in range(i + 1, N):
                    d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                    if d < radii[i] + radii[j] - 1e-9:
                        valid = False
                        break
                if not valid:
                    break
        if valid:
            break
            
        # Minimal shrinkage to recover strict feasibility
        radii *= 0.9999
        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)

    return centers, radii, float(np.sum(radii))
