import numpy as np
from scipy.optimize import minimize, linprog
import warnings
warnings.filterwarnings('ignore')

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-7, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)])
    return b

def constraints(x):
    """Returns all inequality constraints >= 0 (vectorized)."""
    xs, ys, rs = x[0::3], x[1::3], x[2::3]
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    c = np.concatenate([
        xs - rs, 1.0 - xs - rs, ys - rs, 1.0 - ys - rs
    ])
    
    # Overlap constraints: dist >= r_i + r_j
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    # Using np.hypot provides stable gradients near contact points
    c = np.concatenate([c, np.hypot(dx[TRIL_IDX], dy[TRIL_IDX]) - dr[TRIL_IDX]])
    return c

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist)
            idx += 1
            
    bounds_lp = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_lp, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-8)
    except Exception:
        pass
    return np.full(n, 0.05)

def center_penalty(c_flat, radii):
    """Smooth penalty function for centers given fixed radii."""
    c = c_flat.reshape(-1, 2)
    p = 0.0
    x, y = c[:, 0], c[:, 1]
    
    # Boundary penalties
    p += np.sum(np.maximum(radii - x, 0.0)**2)
    p += np.sum(np.maximum(x - (1.0 - radii), 0.0)**2)
    p += np.sum(np.maximum(radii - y, 0.0)**2)
    p += np.sum(np.maximum(y - (1.0 - radii), 0.0)**2)
    
    # Overlap penalties
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    d = np.hypot(dx, dy)
    gap = d - radii[:, None] - radii[None, :]
    p += np.sum(np.maximum(-gap[TRIL_IDX], 0.0)**2)
    return p

def opt_centers_lbfgs(centers, radii, max_iter=2000):
    """Optimize centers for fixed radii using L-BFGS-B."""
    bounds = [(0.001, 0.999)] * (2 * N)
    res = minimize(center_penalty, centers.flatten(), args=(radii,),
                   method='L-BFGS-B', bounds=bounds, options={'maxiter': max_iter, 'ftol': 1e-14})
    return res.x.reshape(-1, 2)

def make_hex_init(r0, angle, scale=1.0):
    """Generates a rotated hexagonal lattice initialization."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0 and len(pts) < N + 10:
            pts.append([x * scale, y * scale])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        
    pts = np.array(pts[:N + 10])
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
        
    mask = (pts[:, 0] >= 0.01) & (pts[:, 0] <= 0.99) & (pts[:, 1] >= 0.01) & (pts[:, 1] <= 0.99)
    pts = pts[mask]
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.1, 0.9, (1, 2))])
    return pts[:N]

def make_force_init(seed):
    """Force-directed layout to spread points evenly and push to boundaries."""
    np.random.seed(seed)
    pts = np.random.uniform(0.15, 0.85, (N, 2))
    for _ in range(150):
        f = np.zeros_like(pts)
        for i in range(N):
            for j in range(i + 1, N):
                dx = pts[j] - pts[i]
                d = np.hypot(dx[0], dx[1])
                if d < 0.25 and d > 1e-5:
                    rep = 0.01 / (d**2 + 0.001)
                    f[i] -= dx * rep / d
                    f[j] += dx * rep / d
            for dim in range(2):
                if pts[i, dim] < 0.1: f[i, dim] += 0.05
                elif pts[i, dim] > 0.9: f[i, dim] -= 0.05
        pts += f * 0.04
        pts = np.clip(pts, 0.03, 0.97)
    return pts

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    bounds_obj = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -1.0
    best_x = None
    
    # Phase 1: Generate Diverse Initial Configurations
    inits = []
    # Hexagonal lattices with varied scales and rotations
    for scale in [0.9, 1.0, 1.1]:
        for ang in np.linspace(-0.4, 0.4, 11):
            inits.append(make_hex_init(0.092 * scale, ang))
            
    # Force-directed spreads
    for s in range(12):
        inits.append(make_force_init(s))
        
    # Random starts
    for s in range(10):
        np.random.seed(s + 100)
        inits.append(np.random.uniform(0.15, 0.85, (N, 2)))

    # Phase 2: Alternating Optimization + SLSQP Polish
    for c_init in inits:
        c_curr = c_init.copy()
        r_curr = solve_lp_radii(c_curr)
        
        # Alternating: LP radii -> L-BFGS-B centers
        for _ in range(20):
            r_curr = solve_lp_radii(c_curr)
            c_curr = opt_centers_lbfgs(c_curr, r_curr, max_iter=1500)
            c_curr = np.clip(c_curr, r_curr, 1.0 - r_curr)
            
        x0 = np.zeros(3 * N)
        x0[0::3] = c_curr[:, 0]
        x0[1::3] = c_curr[:, 1]
        x0[2::3] = np.maximum(r_curr * 0.995, 1e-7)
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds_obj, constraints=cons,
                           options={'maxiter': 12000, 'ftol': 1e-13, 'disp': False})
            if not np.isnan(res.fun):
                c_tmp = np.column_stack((res.x[0::3], res.x[1::3]))
                r_new = solve_lp_radii(c_tmp)
                curr_sum = np.sum(r_new)
                if curr_sum > best_sum and np.min(constraints(res.x)) >= -1e-5:
                    best_sum = curr_sum
                    best_x = res.x.copy()
        except Exception:
            pass

    # Phase 3: Aggressive Topology Search (Deflation & Repositioning)
    if best_x is not None:
        for cyc in range(80):
            x0 = best_x.copy()
            
            # Cooling perturbation
            noise_scale = 0.002 * (0.90 ** cyc)
            x0[0::3] += np.random.normal(0, noise_scale, N)
            x0[1::3] += np.random.normal(0, noise_scale, N)
            
            # Deflate a random subset to break rigid contact networks
            k = np.random.randint(3, 7)
            subset = np.random.choice(N, size=k, replace=False)
            x0[subset * 3 + 2] *= 0.80
            
            r_pert = np.maximum(x0[2::3], 1e-7)
            c_pert = np.column_stack((x0[0::3], x0[1::3]))
            c_pert = np.clip(c_pert, r_pert, 1.0 - r_pert)
            
            # Quick alternating refinement
            for _ in range(15):
                r_pert = solve_lp_radii(c_pert)
                c_pert = opt_centers_lbfgs(c_pert, r_pert, max_iter=1000)
                c_pert = np.clip(c_pert, r_pert, 1.0 - r_pert)
                
            x0[0::3] = c_pert[:, 0]
            x0[1::3] = c_pert[:, 1]
            x0[2::3] = np.maximum(r_pert * 0.998, 1e-7)
            
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds_obj, constraints=cons,
                               options={'maxiter': 10000, 'ftol': 1e-13, 'disp': False})
                if not np.isnan(res.fun):
                    c_tmp = np.column_stack((res.x[0::3], res.x[1::3]))
                    r_new = solve_lp_radii(c_tmp)
                    curr_sum = np.sum(r_new)
                    if curr_sum > best_sum and np.min(constraints(res.x)) >= -1e-5:
                        best_sum = curr_sum
                        best_x = res.x.copy()
            except Exception:
                pass

    # Fallback initialization (should not be reached)
    if best_x is None:
        best_x = np.zeros(3 * N)
        best_x[0::3] = np.tile(np.linspace(0.1, 0.9, 5), 6)[:N]
        best_x[1::3] = np.repeat(np.linspace(0.1, 0.9, 6), 5)[:N]
        best_x[2::3] = 0.06
        best_sum = np.sum(best_x[2::3])

    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3].copy()

    # Final strict validation adjustment against 1e-12 tolerance
    for _ in range(100):
        valid = True
        for i in range(N):
            if radii[i] < 0 or centers[i, 0] < radii[i] - 1e-9 or centers[i, 0] > 1.0 - radii[i] + 1e-9 or \
               centers[i, 1] < radii[i] - 1e-9 or centers[i, 1] > 1.0 - radii[i] + 1e-9:
                valid = False
                break
        if valid:
            for i in range(N):
                for j in range(i + 1, N):
                    d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                    if d < radii[i] + radii[j] - 1e-9:
                        valid = False
                        break
                if not valid:
                    break
        if valid:
            break
            
        # Minimal shrinkage to recover strict feasibility
        radii *= 0.9999
        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)

    return centers, radii, float(np.sum(radii))
