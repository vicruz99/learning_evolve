To push the sum of radii past 2.636, we need a robust optimization strategy that efficiently explores the non-convex landscape of unequal circle packing. The key insights are:
1. **Alternating Optimization**: For fixed centers, radii optimization is a Linear Program (LP). For fixed radii, center optimization is a smooth non-linear problem. Alternating between them (LP for radii, SLSQP for joint refinement) converges rapidly to local optima.
2. **Deflation & Perturbation**: Local optima in packing are often rigid contact graphs. Shrinking a random subset of radii and perturbing centers breaks these contacts, allowing the optimizer to settle into a denser configuration.
3. **Coordinate Descent**: Fine-tuning individual circle positions while recomputing optimal radii for the new configuration is highly effective for squeezing out marginal gains. We test discrete directional moves for each circle.
4. **Diverse Initialization**: Hexagonal lattices with various rotations and force-spread random points provide good starting topologies.
5. **Strict Feasibility Repair**: Final numerical drift is corrected by minimal uniform shrinking to guarantee validation passes.

The implementation combines these into a multi-phase solver: initialization -> multi-start SLSQP -> iterative deflation/perturbation -> coordinate descent polish -> validation repair.

import numpy as np
from scipy.optimize import minimize, linprog
import warnings
warnings.filterwarnings('ignore')

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def obj(x):
    """Objective: maximize sum of radii."""
    return -np.sum(x[2::3])

def cons(x):
    """Inequality constraints: boundary and non-overlap."""
    xs, ys, rs = x[0::3], x[1::3], x[2::3]
    # Boundary: x >= r, 1-x >= r, y >= r, 1-y >= r
    c = np.concatenate([xs - rs, 1.0 - xs - rs, ys - rs, 1.0 - ys - rs])
    # Overlap: dist >= r_i + r_j
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    c = np.concatenate([c, np.hypot(dx[TRIL_IDX], dy[TRIL_IDX]) - dr[TRIL_IDX]])
    return c

def lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    n_ineq = n + n * (n - 1) // 2
    A = np.zeros((n_ineq, n))
    b = np.zeros(n_ineq)
    idx = 0
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A[idx, i] = 1.0
        b[idx] = max(0.0, lim)
        idx += 1
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = max(0.0, d)
            idx += 1
    try:
        res = linprog(-np.ones(n), A_ub=A, b_ub=b, bounds=[(0.0, None)] * n, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-8)
    except Exception:
        pass
    return np.full(n, 0.05)

def proj(x):
    """Project variables to strictly satisfy bounds."""
    x = x.copy()
    for i in range(N):
        r = max(1e-7, x[3 * i + 2])
        x[3 * i] = np.clip(x[3 * i], r, 1.0 - r)
        x[3 * i + 1] = np.clip(x[3 * i + 1], r, 1.0 - r)
        x[3 * i + 2] = r
    return x

def hex_init(r0, ang):
    """Generates a rotated hexagonal lattice initialization."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0 and len(pts) < N + 10:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
    pts = np.array(pts[:N + 10])
    if ang != 0.0:
        c, s = np.cos(ang), np.sin(ang)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.15, 0.85, (1, 2))])
    return pts[:N]

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    bnds = [(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)] * N
    cns = {'type': 'ineq', 'fun': cons}
    best_sum = -1.0
    best_x = None

    # Phase 1: Diverse Initial Configurations
    inits = []
    for ang in np.linspace(-0.5, 0.5, 21):
        inits.append(hex_init(0.095, ang))
    for s in range(15):
        np.random.seed(s)
        pts = np.random.uniform(0.15, 0.85, (N, 2))
        # Force spread to boundaries
        for _ in range(100):
            f = np.zeros_like(pts)
            for i in range(N):
                for j in range(i + 1, N):
                    dx = pts[j] - pts[i]
                    d = np.hypot(dx[0], dx[1])
                    if d < 0.25 and d > 1e-5:
                        rep = 0.01 / (d ** 2 + 0.001)
                        f[i] -= dx * rep / d
                        f[j] += dx * rep / d
                for dim in range(2):
                    if pts[i, dim] < 0.1:
                        f[i, dim] += 0.04
                    elif pts[i, dim] > 0.9:
                        f[i, dim] -= 0.04
            pts += f * 0.05
            pts = np.clip(pts, 0.03, 0.97)
        inits.append(pts)

    # Phase 2: Multi-start SLSQP
    for pts in inits:
        x0 = np.zeros(3 * N)
        x0[0::3] = pts[:, 0]
        x0[1::3] = pts[:, 1]
        x0[2::3] = lp_radii(pts)
        x0 = proj(x0)
        try:
            res = minimize(obj, x0, method='SLSQP', bounds=bnds, constraints=cns,
                           options={'maxiter': 15000, 'ftol': 1e-14})
            if not np.isnan(res.fun):
                s = -res.fun
                if s > best_sum and np.min(cons(res.x)) >= -1e-6:
                    best_sum = s
                    best_x = res.x.copy()
        except Exception:
            pass

    # Phase 3: Deflation & Perturbation to escape local minima
    if best_x is not None:
        for cyc in range(150):
            x0 = best_x.copy()
            noise = 0.003 * (0.92 ** cyc)
            x0[0::3] += np.random.normal(0, noise, N)
            x0[1::3] += np.random.normal(0, noise, N)
            # Shrink random subset to break rigid contacts
            k = np.random.randint(3, 8)
            sub = np.random.choice(N, k, replace=False)
            x0[sub * 3 + 2] *= 0.85
            x0 = proj(x0)
            try:
                res = minimize(obj, x0, method='SLSQP', bounds=bnds, constraints=cns,
                               options={'maxiter': 10000, 'ftol': 1e-13})
                if not np.isnan(res.fun):
                    c_p = np.column_stack((res.x[0::3], res.x[1::3]))
                    r_p = lp_radii(c_p)
                    s_p = np.sum(r_p)
                    if s_p > best_sum:
                        x_p = res.x.copy()
                        x_p[2::3] = r_p
                        x_p = proj(x_p)
                        if np.min(cons(x_p)) >= -1e-6:
                            best_sum = s_p
                            best_x = x_p.copy()
            except Exception:
                pass

    # Phase 4: Coordinate Descent Local Search
    if best_x is not None:
        for _ in range(4):
            improved = True
            while improved:
                improved = False
                cc = np.column_stack((best_x[0::3], best_x[1::3]))
                cr = best_x[2::3]
                bls = np.sum(cr)
                for i in range(N):
                    for dx in [-0.005, -0.002, -0.001, 0.001, 0.002, 0.005]:
                        for dy in [-0.005, -0.002, -0.001, 0.001, 0.002, 0.005]:
                            if dx == 0 and dy == 0:
                                continue
                            nc = cc.copy()
                            nc[i] = np.clip(cc[i] + np.array([dx, dy]), 0.01, 0.99)
                            nr = lp_radii(nc)
                            sn = np.sum(nr)
                            if sn > bls + 1e-8:
                                cc = nc
                                cr = nr
                                bls = sn
                                improved = True
                                break
                        if improved:
                            break
                    if improved:
                        break
                if improved:
                    xn = np.zeros(3 * N)
                    xn[0::3] = cc[:, 0]
                    xn[1::3] = cc[:, 1]
                    xn[2::3] = cr
                    xn = proj(xn)
                    try:
                        res = minimize(obj, xn, method='SLSQP', bounds=bnds, constraints=cns,
                                       options={'maxiter': 8000, 'ftol': 1e-13})
                        if not np.isnan(res.fun):
                            cp = np.column_stack((res.x[0::3], res.x[1::3]))
                            rp = lp_radii(cp)
                            if np.sum(rp) > best_sum:
                                best_sum = np.sum(rp)
                                best_x = res.x.copy()
                                best_x[2::3] = rp
                    except Exception:
                        pass

    # Fallback (should not be reached)
    if best_x is None:
        best_x = np.zeros(3 * N)
        best_x[0::3] = np.tile(np.linspace(0.1, 0.9, 5), 6)[:N]
        best_x[1::3] = np.repeat(np.linspace(0.1, 0.9, 6), 5)[:N]
        best_x[2::3] = 0.06
        best_sum = np.sum(best_x[2::3])

    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3].copy()

    # Final strict validity adjustment against 1e-12 tolerance
    for _ in range(100):
        valid = True
        for i in range(N):
            if radii[i] < 0 or centers[i, 0] < radii[i] - 1e-9 or centers[i, 0] > 1.0 - radii[i] + 1e-9 or \
               centers[i, 1] < radii[i] - 1e-9 or centers[i, 1] > 1.0 - radii[i] + 1e-9:
                valid = False
                break
        if valid:
            for i in range(N):
                for j in range(i + 1, N):
                    if np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1]) < radii[i] + radii[j] - 1e-9:
                        valid = False
                        break
                if not valid:
                    break
        if valid:
            break
        # Minimal shrinkage to recover strict feasibility
        radii *= 0.9999
        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)

    return centers, radii, float(np.sum(radii))
