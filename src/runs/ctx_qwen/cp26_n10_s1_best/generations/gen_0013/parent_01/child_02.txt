import numpy as np
from scipy.optimize import minimize, linprog
import warnings
warnings.filterwarnings('ignore')

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def objective_func(x):
    """Objective: maximize sum of radii."""
    return -np.sum(x[2::3])

def constraints_func(x):
    """Inequality constraints for SLSQP: g(x) >= 0."""
    xs, ys, rs = x[0::3], x[1::3], x[2::3]
    c = np.concatenate([xs - rs, 1.0 - xs - rs, ys - rs, 1.0 - ys - rs])
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    c = np.concatenate([c, np.hypot(dx[TRIL_IDX], dy[TRIL_IDX]) - dr[TRIL_IDX]])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-8, 0.5]."""
    return [(0.0, 1.0), (0.0, 1.0), (1e-8, 0.5)] * N

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    num_ineq = n + n * (n - 1) // 2
    A = np.zeros((num_ineq, n))
    b = np.zeros(num_ineq)
    idx = 0
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A[idx, i] = 1.0
        b[idx] = max(0.0, lim)
        idx += 1
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = max(0.0, d)
            idx += 1
    bounds = [(0.0, None)] * n
    try:
        res = linprog(-np.ones(n), A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-9)
    except Exception:
        pass
    return np.full(n, 0.05)

def center_penalty(c_flat, radii):
    """Smooth penalty function for centers given fixed radii."""
    c = c_flat.reshape(-1, 2)
    x, y, r = c[:, 0], c[:, 1], radii
    p = 0.0
    # Boundary penalties
    p += 1e5 * np.sum(np.maximum(r - x, 0.0)**2)
    p += 1e5 * np.sum(np.maximum(x - (1.0 - r), 0.0)**2)
    p += 1e5 * np.sum(np.maximum(r - y, 0.0)**2)
    p += 1e5 * np.sum(np.maximum(y - (1.0 - r), 0.0)**2)
    # Overlap penalties
    dx = c[:, 0, None] - c[:, 0]
    dy = c[:, 1, None] - c[:, 1]
    d = np.hypot(dx, dy)
    gap = d - r[:, None] - r[None, :]
    violations = np.maximum(-gap[TRIL_IDX], 0.0)
    p += 1e4 * np.sum(violations**2)
    return p

def opt_centers_lbfgs(centers, radii):
    """Optimize centers for fixed radii using L-BFGS-B."""
    bounds = [(0.0001, 0.9999)] * (2 * N)
    res = minimize(center_penalty, centers.flatten(), args=(radii,),
                   method='L-BFGS-B', bounds=bounds, options={'maxiter': 3000, 'ftol': 1e-16})
    return res.x.reshape(-1, 2)

def make_hex_init(r0, angle):
    """Generates a rotated hexagonal lattice initialization."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0 and len(pts) < N + 10:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
    pts = np.array(pts[:N + 10])
    if angle != 0.0:
        ca, sa = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[ca, -sa], [sa, ca]]) + 0.5
    mask = (pts[:, 0] >= 0.01) & (pts[:, 0] <= 0.99) & (pts[:, 1] >= 0.01) & (pts[:, 1] <= 0.99)
    pts = pts[mask]
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.1, 0.9, (1, 2))])
    return pts[:N]

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    np.random.seed(42)
    bounds_obj = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints_func}
    best_sum = -1.0
    best_x = None
    best_c = None
    best_r = None

    # Phase 1: Diverse Initial Configurations
    inits = []
    for ang in np.linspace(-0.5, 0.5, 19):
        inits.append(make_hex_init(0.095, ang))

        np.random.seed(s)
        pts = np.random.uniform(0.15, 0.85, (N, 2))
        # Quick force spread to separate points
        for _ in range(80):
            f = np.zeros_like(pts)
            for i in range(N):
                for j in range(i + 1, N):
                    dx = pts[j] - pts[i]
                    d = np.hypot(dx[0], dx[1])
                    if d < 0.3 and d > 1e-5:
                        rep = 0.01 / (d**2 + 0.001)
                        f[i] -= dx * rep / d
                        f[j] += dx * rep / d
                for dim in range(2):
                    if pts[i, dim] < 0.08: f[i, dim] += 0.06
                    elif pts[i, dim] > 0.92: f[i, dim] -= 0.06
            pts += f * 0.05
            pts = np.clip(pts, 0.02, 0.98)
        inits.append(pts)

    # Phase 2: Alternating Optimization + SLSQP Polish
    for pts in inits:
        c = pts.copy()
        r = solve_lp_radii(c)
        for _ in range(12):
            r = solve_lp_radii(c)
            c = opt_centers_lbfgs(c, r)
        x0 = np.zeros(3 * N)
        x0[0::3] = c[:, 0]
        x0[1::3] = c[:, 1]
        x0[2::3] = np.maximum(r * 0.998, 1e-7)
        try:
            res = minimize(objective_func, x0, method='SLSQP', bounds=bounds_obj, constraints=cons,
                           options={'maxiter': 20000, 'ftol': 1e-14})
            if not np.isnan(res.fun):
                curr_sum = -res.fun
                if curr_sum > best_sum:
                    best_sum = curr_sum
                    best_x = res.x.copy()
                    best_c = np.column_stack((best_x[0::3], best_x[1::3]))
                    best_r = best_x[2::3].copy()
        except Exception:
            pass

    # Phase 3: Perturbation & Deflation Search
    if best_x is not None:
        for cyc in range(60):
            x0 = best_x.copy()
            noise = 0.0025 * (0.91 ** cyc)

                x0[3 * i] = np.clip(x0[3 * i], r, 1.0 - r)
                x0[3 * i + 1] = np.clip(x0[3 * i + 1], r, 1.0 - r)

                    c_p = np.column_stack((res.x[0::3], res.x[1::3]))
                    r_p = solve_lp_radii(c_p)
                    if np.sum(r_p) > best_sum:
                        best_sum = np.sum(r_p)
                        best_x = res.x.copy()
                        best_x[2::3] = r_p
                        best_c = c_p
                        best_r = r_p
            except Exception:
                pass

    # Phase 4: Radius Swap Heuristic to escape topological local minima
    if best_c is not None:

            br_swap[i], br_swap[j] = br_swap[j], br_swap[i]
            c_new = opt_centers_lbfgs(bc_swap, br_swap)
            r_new = solve_lp_radii(c_new)
            if np.sum(r_new) > best_sum + 1e-9:
                best_sum = np.sum(r_new)
                best_c = c_new
                best_r = r_new
                best_x = np.zeros(3 * N)
                best_x[0::3] = best_c[:, 0]
                best_x[1::3] = best_c[:, 1]
                best_x[2::3] = best_r

    # Phase 5: Coordinate Descent Micro-tuning
    if best_c is not None:
        for _ in range(2):
            improved = True
            while improved:
                improved = False
                for i in range(N):
                    best_local = best_sum
                    steps = [-0.003, -0.0015, -0.0005, 0.0005, 0.0015, 0.003]
                    for dx in steps:

                            if dx == 0 and dy == 0: continue
                            c_try = best_c.copy()
                            c_try[i, 0] = np.clip(c_try[i, 0] + dx, 0.001, 0.999)
                            c_try[i, 1] = np.clip(c_try[i, 1] + dy, 0.001, 0.999)
                            r_try = solve_lp_radii(c_try)
                            s_try = np.sum(r_try)
                            if s_try > best_local + 1e-9:
                                best_local = s_try
                                best_c = c_try
                                best_r = r_try
                                improved = True
                                break
                        if improved: break

                best_sum = best_local

    # Phase 6: Final SLSQP Polish & Uniform Expansion Test
    if best_c is not None:
        x0 = np.zeros(3 * N)
        x0[0::3] = best_c[:, 0]
        x0[1::3] = best_c[:, 1]
        x0[2::3] = np.maximum(best_r * 0.999, 1e-8)
        try:
            res = minimize(objective_func, x0, method='SLSQP', bounds=bounds_obj, constraints=cons,
                           options={'maxiter': 30000, 'ftol': 1e-15})
            if not np.isnan(res.fun):
                c_final = np.column_stack((res.x[0::3], res.x[1::3]))
                r_final = solve_lp_radii(c_final)
                if np.sum(r_final) > best_sum:
                    best_sum = np.sum(r_final)
                    best_c = c_final
                    best_r = r_final
        except Exception:
            pass

            c_exp = opt_centers_lbfgs(best_c, r_exp)
            r_exp_new = solve_lp_radii(c_exp)
            if np.sum(r_exp_new) > best_sum + 1e-9:
                best_sum = np.sum(r_exp_new)
                best_c = c_exp
                best_r = r_exp_new

    radii = best_r

        for i in range(N):

        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)

    return centers, radii, float(np.sum(radii))
