import warnings

warnings.filterwarnings('ignore')

N = 26
TRIU_IDX = np.triu_indices(N, 1)

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    num_ineq = n + n * (n - 1) // 2
    A = np.zeros((num_ineq, n))
    b = np.zeros(num_ineq)
    
    idx = 0
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A[idx, i] = 1.0
        b[idx] = max(1e-9, lim)
        idx += 1

        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = max(1e-9, d)
            idx += 1
            
    bounds = [(1e-9, None)] * n
    try:
        res = linprog(-np.ones(n), A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return np.full(n, 0.05)

def center_penalty(c_flat, radii):

    dy = c[:, 1, None] - c[:, 1]

    gap = d - r[:, None] - r[None, :]
    p += 10.0 * np.sum(np.maximum(-gap[TRIU_IDX], 0.0)**2)
    
    return p

def opt_centers_lbfgs(centers, radii, maxiter=400):
    """Optimize centers for fixed radii using L-BFGS-B."""

                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return res.x.reshape(-1, 2)
    except Exception:
        return centers

def make_hex_init(r0, angle):

    y = r0
    row = 0

        x = r0 if row % 2 == 0 else 2.0 * r0

            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
    pts = np.array(pts[:N])
    if angle != 0.0:
        ca, sa = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[ca, -sa], [sa, ca]]) + 0.5
    return np.clip(pts, 0.01, 0.99)

def make_staggered_init():
    """Generates a staggered row initialization adapted to square boundaries."""
    pts = []
    rows_config = [6, 5, 6, 5, 4]

    y = 0.10

                break

            pts.append([x, y])
        y += y_step
        if len(pts) >= N:
            break
    while len(pts) < N:
        pts.append(np.random.uniform(0.15, 0.85, 2))
    return np.array(pts[:N])
