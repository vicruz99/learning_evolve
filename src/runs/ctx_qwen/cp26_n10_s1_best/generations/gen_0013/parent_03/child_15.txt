import numpy as np
from scipy.optimize import minimize, linprog
import warnings
warnings.filterwarnings('ignore')

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Returns all inequality constraints >= 0 (vectorized)."""
    xs, ys, rs = x[0::3], x[1::3], x[2::3]
    c = np.concatenate([xs - rs, 1.0 - xs - rs, ys - rs, 1.0 - ys - rs])
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    c = np.concatenate([c, np.hypot(dx[TRIL_IDX], dy[TRIL_IDX]) - dr[TRIL_IDX]])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-7, 0.5]."""
    return [(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)] * N

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A = np.zeros((num_ineq, n))
    b = np.zeros(num_ineq)
    idx = 0
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A[idx, i] = 1.0
        b[idx] = max(0.0, lim)
        idx += 1
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = max(0.0, d)
            idx += 1
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0.0, None)] * n, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-8)
    except Exception:
        pass
    return np.full(n, 0.05)

def center_penalty(c_flat, radii):
    """Smooth penalty function for centers given fixed radii."""
    c = c_flat.reshape(-1, 2)
    p = 0.0
    x, y, r = c[:, 0], c[:, 1], radii
    p += 100.0 * (np.sum(np.maximum(r - x, 0.0)**2) +
                   np.sum(np.maximum(x - (1.0 - r), 0.0)**2) +
                   np.sum(np.maximum(r - y, 0.0)**2) +
                   np.sum(np.maximum(y - (1.0 - r), 0.0)**2))
    dx = c[:, 0, None] - c[:, 0]
    dy = c[:, 1, None] - c[:, 1]
    d = np.hypot(dx, dy)
    gap = d - r[:, None] - r[None, :]
    p += 100.0 * np.sum(np.maximum(-gap[TRIL_IDX], 0.0)**2)
    return p

def opt_centers_penalty(centers, radii):
    """Optimize centers for fixed radii using L-BFGS-B."""
    bounds = [(0.001, 0.999)] * (2 * N)
    res = minimize(center_penalty, centers.flatten(), args=(radii,), method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': 2000, 'ftol': 1e-12})
    return res.x.reshape(-1, 2)

def project(x):
    """Project variables to strictly satisfy bounds."""
    x = x.copy()
    for i in range(N):
        r = max(1e-6, x[3 * i + 2])
        x[3 * i] = np.clip(x[3 * i], r, 1.0 - r)
        x[3 * i + 1] = np.clip(x[3 * i + 1], r, 1.0 - r)
        x[3 * i + 2] = r
    return x

def make_hex_init(r0, angle):
    """Generates a rotated hexagonal lattice initialization."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0 and len(pts) < N + 10:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
    pts = np.array(pts[:N + 10])
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.15, 0.85, (1, 2))])
    return pts[:N]

def make_force_init(seed):
    """Force-directed layout to spread points evenly and push to boundaries."""
    np.random.seed(seed)
    pts = np.random.uniform(0.1, 0.9, (N, 2))
    for step in range(300):
        f = np.zeros_like(pts)
        lr = 0.02 * (1.0 - step / 300.0)
        for i in range(N):
            for j in range(i + 1, N):
                dx = pts[j] - pts[i]
                d = np.hypot(dx[0], dx[1])
                if d < 0.3 and d > 1e-5:
                    rep = 0.01 / (d**2 + 0.001)
                    f[i] -= dx * rep / d
                    f[j] += dx * rep / d
            for dim in range(2):
                if pts[i, dim] < 0.1: f[i, dim] += 0.06
                elif pts[i, dim] > 0.9: f[i, dim] -= 0.06
        pts += f * lr
        pts = np.clip(pts, 0.02, 0.98)
    return pts

def run_packing():

    best_x = None

            inits.append(make_hex_init(r0, ang))

        inits.append(make_force_init(s))

    for pts in inits:
        x0 = np.zeros(3 * N)
        x0[0::3] = pts[:, 0]
        x0[1::3] = pts[:, 1]
        r_lp = solve_lp_radii(pts)
        x0[2::3] = r_lp * 0.995
        x0 = project(x0)

        # Alternate LP radii and center optimization
        c_curr = pts.copy()
        r_curr = r_lp
        for _ in range(3):
            r_curr = solve_lp_radii(c_curr)
            c_curr = opt_centers_penalty(c_curr, r_curr)

        x0[0::3] = c_curr[:, 0]
        x0[1::3] = c_curr[:, 1]
        x0[2::3] = r_curr * 0.998
        x0 = project(x0)

        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds_obj, constraints=cons,
                           options={'maxiter': 25000, 'ftol': 1e-13, 'disp': False})
            if not np.isnan(res.fun):
                c_tmp = np.column_stack((res.x[0::3], res.x[1::3]))
                r_new = solve_lp_radii(c_tmp)
                if r_new is not None:
                    curr_sum = np.sum(r_new)
                    if curr_sum > best_sum:
                        best_sum = curr_sum
                        best_x = res.x.copy()
                        best_x[2::3] = r_new
        except Exception:
            pass

    # Phase 2: Iterative Deflation & Refinement to Escape Local Minima
    if best_x is not None:
        for cyc in range(80):
            x0 = best_x.copy()
            shrink = 0.75 + 0.25 * (cyc / 80.0)
            x0[2::3] *= shrink
            noise = 0.003 * (0.92 ** cyc)
            x0[0::3] += np.random.normal(0, noise, N)
            x0[1::3] += np.random.normal(0, noise, N)
            x0 = project(x0)

            try:

                    if r_new is not None:

                        x_p[2::3] = r_new
                        x_p = project(x_p)
                        curr_sum = np.sum(x_p[2::3])
                        if curr_sum > best_sum and np.min(constraints(x_p)) >= -1e-5:

                            best_x = x_p.copy()
            except Exception:
                pass

        c_best = np.column_stack((best_x[0::3], best_x[1::3]))

                            for dy in steps:
                                if dx == 0.0 and dy == 0.0:
                                    continue
                                new_c = c_best.copy()
                                new_c[i, 0] = np.clip(c_best[i, 0] + dx, 0.005, 0.995)
                                new_c[i, 1] = np.clip(c_best[i, 1] + dy, 0.005, 0.995)
                                r_new = solve_lp_radii(new_c)
                                s_new = np.sum(r_new)
                                if s_new > best_local + 1e-8:
                                    best_local = s_new
                                    c_try = new_c[i].copy()
                    if best_local > best_lp_sum + 1e-8:
                        c_best[i] = c_try
                        best_lp_sum = best_local
                        r_best = solve_lp_radii(c_best)
                        best_x[0::3] = c_best[:, 0]
                        best_x[1::3] = c_best[:, 1]
                        best_x[2::3] = r_best
                        best_sum = best_lp_sum

    centers = np.column_stack((best_x[0::3], best_x[1::3]))

    for _ in range(100):
        valid = True
        for i in range(N):

        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)

    return centers, radii, float(np.sum(radii))
