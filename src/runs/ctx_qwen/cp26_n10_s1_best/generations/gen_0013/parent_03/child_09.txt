import numpy as np
from scipy.optimize import minimize, linprog

N = 26
TRIL_IDX = np.tril_indices(N, -1)
TRIU_IDX = np.triu_indices(N, 1)

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A = np.zeros((num_ineq, n))
    b = np.zeros(num_ineq)
    
    idx = 0
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A[idx, i] = 1.0
        b[idx] = max(0.0, lim)
        idx += 1
        
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = max(0.0, d)
            idx += 1
            
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0.0, None)] * n, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-9)
    except Exception:
        pass
    return np.full(n, 0.05)

def optimize_centers_penalty(centers, radii):
    """Optimize centers for fixed radii using L-BFGS-B with a penalty function."""
    def penalty(c_flat):
        c = c_flat.reshape(-1, 2)
        p = 0.0
        x, y, r = c[:, 0], c[:, 1], radii
        
        # Boundary penalties
        p += np.sum(np.maximum(r - x, 0.0)**2)
        p += np.sum(np.maximum(x - (1.0 - r), 0.0)**2)
        p += np.sum(np.maximum(r - y, 0.0)**2)
        p += np.sum(np.maximum(y - (1.0 - r), 0.0)**2)
        
        # Overlap penalties
        dx = c[:, 0, None] - c[:, 0]
        dy = c[:, 1, None] - c[:, 1]
        d = np.hypot(dx, dy)
        gap = d - r[:, None] - r[None, :]
        p += np.sum(np.maximum(-gap[TRIU_IDX], 0.0)**2)
        return p

    res = minimize(penalty, centers.flatten(), method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * (2 * N), options={'maxiter': 10000, 'ftol': 1e-14})
    return res.x.reshape(-1, 2)

def alternating_opt(centers, radii, iters=20):
    """Alternate between LP radii optimization and penalty-based center optimization."""
    c = centers.copy()
    r = radii.copy()
    for _ in range(iters):
        r = solve_lp_radii(c)
        c = optimize_centers_penalty(c, r)
    return c, r

def make_hex_init(r0, angle):
    """Generates a rotated hexagonal lattice initialization."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 5:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0 and len(pts) < N + 5:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
    pts = np.array(pts[:N + 5])
    if angle != 0.0:
        ca, sa = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[ca, -sa], [sa, ca]]) + 0.5
    mask = (pts[:, 0] >= 0.01) & (pts[:, 0] <= 0.99) & (pts[:, 1] >= 0.01) & (pts[:, 1] <= 0.99)
    pts = pts[mask]
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.1, 0.9, (1, 2))])
    return pts[:N]

def make_staggered_init():
    """Generates a row-based initialization adapted to square boundaries."""
    pts = []
    rows = [6, 5, 6, 5, 4]
    y = 0.08
    for r_count in rows:
        x_step = 0.9 / (r_count - 1) if r_count > 1 else 0.5
        for k in range(r_count):
            x = 0.05 + k * x_step
            pts.append([x, y])
            if len(pts) >= N:
                break
        y += 0.18
        if len(pts) >= N:
            break
    while len(pts) < N:
        pts.append(np.random.uniform(0.1, 0.9, 2))
    return np.array(pts[:N])

def make_force_init(seed):
    """Force-directed layout to spread points evenly and push to boundaries."""
    np.random.seed(seed)
    pts = np.random.uniform(0.15, 0.85, (N, 2))
    for step in range(300):
        f = np.zeros_like(pts)
        lr = 0.02 * (1.0 - step / 300.0)
        for i in range(N):
            for j in range(i + 1, N):
                dx = pts[j] - pts[i]
                d = np.hypot(dx[0], dx[1])
                if d < 0.3 and d > 1e-5:
                    rep = 0.015 / (d**2 + 0.001)
                    f[i] -= dx * rep / d
                    f[j] += dx * rep / d
            for dim in range(2):
                if pts[i, dim] < 0.1: f[i, dim] += 0.06
                elif pts[i, dim] > 0.9: f[i, dim] -= 0.06
        pts += f * lr
        pts = np.clip(pts, 0.02, 0.98)
    return pts

def project_feasible(x):
    """Project variables to strictly satisfy bounds."""
    x = x.copy()
    for i in range(N):
        r = max(1e-6, x[3 * i + 2])
        x[3 * i] = np.clip(x[3 * i], r, 1.0 - r)
        x[3 * i + 1] = np.clip(x[3 * i + 1], r, 1.0 - r)
        x[3 * i + 2] = r
    return x

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    best_x = None

            best_sum = s_val
            best_x = np.zeros(3 * N)
            best_x[0::3] = c[:, 0]
            best_x[1::3] = c[:, 1]
            best_x[2::3] = r

            x0 = project_feasible(x0)

            if s_val > best_sum:
                best_sum = s_val
                best_x = np.zeros(3 * N)
                best_x[0::3] = c[:, 0]
                best_x[1::3] = c[:, 1]
                best_x[2::3] = r
