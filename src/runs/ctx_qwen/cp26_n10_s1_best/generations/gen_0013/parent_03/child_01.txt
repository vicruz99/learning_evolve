import numpy as np
from scipy.optimize import minimize, linprog
import warnings
warnings.filterwarnings('ignore')

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A = np.zeros((num_ineq, n))
    b = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A[idx, i] = 1.0
        b[idx] = max(0.0, lim)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = max(0.0, d)
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-9)
    except:
        pass
    return np.full(n, 0.01)

def lp_hill_climb(centers, max_iter=4000, temp_init=0.025):
    """Hill-climb on centers using LP radii solver as fitness oracle."""
    best_c = centers.copy()
    best_score = -1.0
    
    for step in range(max_iter):
        i = np.random.randint(N)
        # Cooling schedule for step size
        temp = temp_init * (0.997 ** (step // N))
        
        new_c = best_c.copy()
        new_c[i] += np.random.uniform(-temp, temp, 2)
        new_c[i] = np.clip(new_c[i], 0.005, 0.995)
        
        new_r = solve_lp_radii(new_c)
        new_score = np.sum(new_r)
        
        if new_score > best_score:
            best_c = new_c
            best_score = new_score
            
    return best_c, solve_lp_radii(best_c), best_score

def make_hex_init(r0, angle):
    """Generates a rotated hexagonal lattice initialization."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 5:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0 and len(pts) < N + 5:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
    pts = np.array(pts[:N + 5])
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
    mask = (pts[:, 0] >= 0.01) & (pts[:, 0] <= 0.99) & (pts[:, 1] >= 0.01) & (pts[:, 1] <= 0.99)
    pts = pts[mask]
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.1, 0.9, (1, 2))])
    return pts[:N]

def make_force_init(seed):
    """Force-directed layout to spread points evenly and push to boundaries."""
    np.random.seed(seed)
    pts = np.random.uniform(0.15, 0.85, (N, 2))
    for step in range(150):
        f = np.zeros_like(pts)
        lr = 0.012 * (1.0 - step / 150.0)
        for i in range(N):
            for j in range(i + 1, N):
                dx = pts[j] - pts[i]
                d = np.hypot(dx[0], dx[1])
                if d < 0.26 and d > 1e-5:
                    rep = 0.008 / (d**2 + 0.001)
                    f[i] -= dx * rep / d
                    f[j] += dx * rep / d
            for dim in range(2):
                if pts[i, dim] < 0.1: f[i, dim] += 0.06
                elif pts[i, dim] > 0.9: f[i, dim] -= 0.06
        pts += f * lr
        pts = np.clip(pts, 0.02, 0.98)
    return pts

def run_packing():
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    best_score = -1.0
    best_centers = None
    best_radii = None
    
    # Phase 1: Generate Diverse Initial Configurations
    inits = []
    for r0 in [0.085, 0.090, 0.095, 0.100]:
        for ang in np.linspace(-0.35, 0.35, 13):
            inits.append(make_hex_init(r0, ang))
    for s in range(12):
        inits.append(make_force_init(s))
    for s in range(8):
        np.random.seed(s + 200)
        inits.append(np.random.uniform(0.1, 0.9, (N, 2)))
        
    # Phase 2: LP-Guided Local Search + Topology Escape
    for idx, pts in enumerate(inits):
        c, r, s = lp_hill_climb(pts, max_iter=4000)
        if s > best_score:
            best_score = s
            best_centers = c
            best_radii = r
            
        # Escape rigid contact networks by randomizing a subset of centers
        if idx % 2 == 0:
            c_esc = best_centers.copy()
            subset = np.random.choice(N, size=max(5, N // 4), replace=False)
            c_esc[subset] = np.random.uniform(0.15, 0.85, (len(subset), 2))
            c2, r2, s2 = lp_hill_climb(c_esc, max_iter=3000, temp_init=0.03)
            if s2 > best_score:
                best_score = s2
                best_centers = c2
                best_radii = r2

    # Phase 3: SLSQP Joint Optimization Polish
    x0 = np.zeros(3 * N)
    x0[0::3] = best_centers[:, 0]
    x0[1::3] = best_centers[:, 1]
    x0[2::3] = best_radii
    
    def obj(x): return -np.sum(x[2::3])
    def cons(x):
        xs, ys, rs = x[0::3], x[1::3], x[2::3]
        c = np.concatenate([xs - rs, 1.0 - xs - rs, ys - rs, 1.0 - ys - rs])
        dx = xs[:, None] - xs[None, :]
        dy = ys[:, None] - ys[None, :]
        dr = rs[:, None] + rs[None, :]
        c = np.concatenate([c, np.hypot(dx[TRIL_IDX], dy[TRIL_IDX]) - dr[TRIL_IDX]])
        return c
        
    bounds = [(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)] * N
    try:
        res = minimize(obj, x0, method='SLSQP', bounds=bounds, constraints={'type': 'ineq', 'fun': cons},
                       options={'maxiter': 20000, 'ftol': 1e-13})
        if not np.isnan(res.fun):
            c_sl = np.column_stack((res.x[0::3], res.x[1::3]))
            r_sl = solve_lp_radii(c_sl)
            s_sl = np.sum(r_sl)
            if s_sl > best_score:
                best_score = s_sl
                best_centers = c_sl
                best_radii = r_sl
    except:
        pass
        
    # Phase 4: Final Fine-Grained LP Hill-Climb
    c_final, r_final, s_final = lp_hill_climb(best_centers, max_iter=3000, temp_init=0.015)
    if s_final > best_score:
        best_score = s_final
        best_centers = c_final
        best_radii = r_final

    # Final strict validity adjustment against 1e-12 tolerance
    for _ in range(100):
        valid = True
        for i in range(N):
            if best_radii[i] < 0 or best_centers[i, 0] < best_radii[i] - 1e-9 or best_centers[i, 0] > 1.0 - best_radii[i] + 1e-9 or \
               best_centers[i, 1] < best_radii[i] - 1e-9 or best_centers[i, 1] > 1.0 - best_radii[i] + 1e-9:
                valid = False; break
        if valid:
            for i in range(N):
                for j in range(i + 1, N):
                    d = np.hypot(best_centers[i, 0] - best_centers[j, 0], best_centers[i, 1] - best_centers[j, 1])
                    if d < best_radii[i] + best_radii[j] - 1e-9:
                        valid = False; break
                if not valid: break
        if valid: break
        best_radii *= 0.9995
        best_centers[:, 0] = np.clip(best_centers[:, 0], best_radii, 1.0 - best_radii)
        best_centers[:, 1] = np.clip(best_centers[:, 1], best_radii, 1.0 - best_radii)

    return best_centers, best_radii, float(best_score)
