

<strategy>
To push past the local optima that previously capped the sum of radii at ~2.636, we need a more robust optimization strategy that separates the non-smooth contact constraints from the continuous geometric search. 

Key improvements:
1. **Squared Distance Constraints in SLSQP**: Using `dist^2 >= (r_i + r_j)^2` instead of `hypot` avoids gradient singularities at contact points, giving the solver smoother landscapes to navigate.
2. **Simulated Annealing on Centers with LP Black-Box**: Instead of relying solely on coupled SLSQP updates, we optimize centers directly. For any proposed center configuration, we solve a fast Linear Program to find the exact maximum radii. This turns the problem into a smooth(er) landscape for centers only. Simulated annealing allows controlled escapes from rigid contact networks.
3. **Deflation & Restarts**: Periodically shrinking radii, perturbing centers, and re-optimizing breaks symmetries and stuck contact graphs that trap gradient-based methods.
4. **Coordinate Descent Fine-Tuning**: After global search, we perform precise single-circle repositioning guided by LP radii re-evaluation to squeeze out micro-improvements.
5. **Robust Initialization**: Combines rotated hexagonal lattices (varying scales) with random starts to cover diverse topological basins.

This hybrid approach leverages LP for exact radial bounds under fixed geometry, while using global and local search techniques to navigate the highly non-convex center landscape.

```python
import numpy as np
from scipy.optimize import minimize, linprog
import warnings
warnings.filterwarnings('ignore')

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Returns all inequality constraints >= 0 (vectorized)."""
    xs, ys, rs = x[0::3], x[1::3], x[2::3]
    c = np.concatenate([xs - rs, 1.0 - xs - rs, ys - rs, 1.0 - ys - rs])
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    # Squared distance constraints provide smooth, polynomial gradients
    c = np.concatenate([c, dx[TRIL_IDX]**2 + dy[TRIL_IDX]**2 - dr[TRIL_IDX]**2])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-7, 0.5]."""
    return [(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)] * N

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A = np.zeros((num_ineq, n))
    b = np.zeros(num_ineq)
    
    idx = 0
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A[idx, i] = 1.0
        b[idx] = max(0.0, lim)
        idx += 1
        
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = max(0.0, d)
            idx += 1
            
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0.0, None)] * n, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-9)
    except Exception:
        pass
    return np.full(n, 0.02)

def project(x):
    """Project variables to strictly satisfy bounds."""
    x = x.copy()
    for i in range(N):
        r = max(1e-6, x[3 * i + 2])
        x[3 * i] = np.clip(x[3 * i], r, 1.0 - r)
        x[3 * i + 1] = np.clip(x[3 * i + 1], r, 1.0 - r)
        x[3 * i + 2] = r
    return x

def make_hex_init(r0, angle, scale=1.0):
    """Generates a hexagonal lattice initialization with optional rotation and scaling."""
    pts = []
    y = r0 * scale
    row = 0
    while len(pts) < N + 10:
        x = r0 * scale if row % 2 == 0 else 2.0 * r0 * scale
        while x <= 1.0 - r0 * scale and len(pts) < N + 10:
            pts.append([x, y])
            x += 2.0 * r0 * scale
        y += np.sqrt(3.0) * r0 * scale
        row += 1
    pts = np.array(pts[:N + 10])
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
    mask = (pts[:, 0] >= 0.01) & (pts[:, 0] <= 0.99) & (pts[:, 1] >= 0.01) & (pts[:, 1] <= 0.99)
    pts = pts[mask]
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.1, 0.9, (1, 2))])
    return pts[:N]

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -1.0
    best_x = None

    # Phase 1: Diverse Multi-start SLSQP
    inits = []
    for s in [0.95, 1.0, 1.05]:
        for r0 in [0.085, 0.095]:
            for ang in np.linspace(-0.3, 0.3, 11):
                inits.append(make_hex_init(r0, ang, s))
    for _ in range(10):
        inits.append(np.random.uniform(0.15, 0.85, (N, 2)))
        
    for pts in inits:
        x0 = np.zeros(3 * N)
        x0[0::3] = pts[:, 0]
        x0[1::3] = pts[:, 1]
        r_lp = solve_lp_radii(pts)
        x0[2::3] = r_lp * 0.99
        x0 = project(x0)
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 20000, 'ftol': 1e-13, 'disp': False})
            if not np.isnan(res.fun):
                s_val = -res.fun
                if s_val > best_sum and np.min(constraints(res.x)) >= -1e-5:
                    best_sum = s_val
                    best_x = res.x.copy()
        except Exception:
            pass
            
    # Phase 2: Simulated Annealing on Centers with LP Black-Box Objective
    # This separates center topology search from radial constraint handling
    if best_x is not None:
        curr_c = np.column_stack((best_x[0::3], best_x[1::3]))
        curr_r = solve_lp_radii(curr_c)
        best_lp_sum = np.sum(curr_r)
        best_lp_c = curr_c.copy()
        T = 0.005  # Initial temperature for acceptance of worse configurations
        
        for step in range(500):
            T = max(0.0005, T * 0.99)
            noise_scale = 0.002 * (1.0 - step / 500.0) + 0.0005
            new_c = curr_c + np.random.normal(0, noise_scale, (N, 2))
            new_c = np.clip(new_c, 0.005, 0.995)
            
            new_r = solve_lp_radii(new_c)
            new_sum = np.sum(new_r)
            delta = new_sum - best_lp_sum
            
            # Accept if better, or with probability if worse
            if delta > 0 or np.random.rand() < min(1.0, np.exp(delta / (T + 1e-9))):
                best_lp_sum = new_sum
                best_lp_c = new_c
                curr_c = new_c

        if best_lp_sum > best_sum:
            best_sum = best_lp_sum
            x_temp = np.zeros(3 * N)
            x_temp[0::3] = best_lp_c[:, 0]
            x_temp[1::3] = best_lp_c[:, 1]
            x_temp[2::3] = solve_lp_radii(best_lp_c)
            x_temp = project(x_temp)
            best_x = x_temp

    # Phase 3: Deflation & SLSQP Restarts to Break Rigid Contact Networks
    if best_x is not None:
        for cyc in range(30):
            x0 = best_x.copy()
            shrink = 0.80 + 0.20 * (cyc / 30.0)
            x0[2::3] *= shrink
            noise = 0.002 / (cyc + 1)
            x0[0::3] += np.random.normal(0, noise, N)
            x0[1::3] += np.random.normal(0, noise, N)
            x0 = project(x0)
            
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 12000, 'ftol': 1e-13, 'disp': False})
                if not np.isnan(res.fun):
                    c_tmp = np.column_stack((res.x[0::3], res.x[1::3]))
                    r_new = solve_lp_radii(c_tmp)
                    x0[2::3] = r_new * 0.998
                    x0 = project(x0)
                    s_val = np.sum(x0[2::3])
                    if s_val > best_sum and np.min(constraints(x0)) >= -1e-5:
                        best_sum = s_val
                        best_x = x0.copy()
            except Exception:
                pass

        # Phase 4: Coordinate Descent Fine-Tuning on Centers
        c_best = np.column_stack((best_x[0::3], best_x[1::3]))
        r_best = best_x[2::3]
        best_lp_sum = np.sum(r_best)
        
        for _ in range(3):
            improved = True
            while improved:
                improved = False
                for i in range(N):
                    best_local = best_lp_sum
                    steps = [-0.005, -0.002, 0.002, 0.005]
                    for dx in steps:
                        for dy in steps:
                            if dx == 0.0 and dy == 0.0:
                                continue
                            new_c = c_best.copy()
                            new_c[i, 0] = np.clip(c_best[i, 0] + dx, 0.005, 0.995)
                            new_c[i, 1] = np.clip(c_best[i, 1] + dy, 0.005, 0.995)
                            r_new = solve_lp_radii(new_c)
                            s_new = np.sum(r_new)
                            if s_new > best_local + 1e-8:
                                best_local = s_new
                                c_best[i] = new_c[i]
                    if best_local > best_lp_sum + 1e-8:
                        best_lp_sum = best_local
                        r_best = solve_lp_radii(c_best)
                        best_x[0::3] = c_best[:, 0]
                        best_x[1::3] = c_best[:, 1]
                        best_x[2::3] = r_best
                        best_sum = best_lp_sum
                        improved = True

    # Fallback (should not be reached)
    if best_x is None:
        best_x = np.zeros(3 * N)
        best_x[2::3] = 0.06
        best_x[0::3] = np.tile(np.linspace(0.1, 0.9, 5), 6)[:N]
        best_x[1::3] = np.repeat(np.linspace(0.1, 0.9, 6), 5)[:N]
        best_sum = np.sum(best_x[2::3])
        
    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3].copy()
    
    # Final LP squeeze to guarantee maximal radii for final centers
    final_r = solve_lp_radii(centers)
    if np.sum(final_r) > np.sum(radii) - 1e-7:
        radii = final_r.copy()

    # Final strict validation repair against 1e-12 tolerance
    for _ in range(100):
        valid = True
        for i in range(N):
            if radii[i] < 0 or centers[i, 0] < radii[i] - 1e-9 or centers[i, 0] > 1.0 - radii[i] + 1e-9 or \
               centers[i, 1] < radii[i] - 1e-9 or centers[i, 1] > 1.0 - radii[i] + 1e-9:
                valid = False
                break
        if valid:
            for i in range(N):
                for j in range(i + 1, N):
                    if np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1]) < radii[i] + radii[j] - 1e-9:
                        valid = False
                        break
                if not valid:
                    break
        if valid:
            break
            
        radii *= 0.9995
        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)

    return centers, radii, float(np.sum(radii))
```
