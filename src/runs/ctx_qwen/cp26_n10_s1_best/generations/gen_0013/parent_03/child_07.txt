import numpy as np
from scipy.optimize import minimize, linprog
import warnings
warnings.filterwarnings('ignore')

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A = np.zeros((num_ineq, n))
    b = np.zeros(num_ineq)
    
    idx = 0
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A[idx, i] = 1.0
        b[idx] = max(0.0, lim)
        idx += 1
        
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = max(0.0, d)
            idx += 1
            
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0.0, None)] * n, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-9)
    except Exception:
        pass
    return np.full(n, 0.05)

def center_penalty_and_grad(centers_flat, radii):
    """Smooth penalty function and its gradient for center optimization."""
    centers = centers_flat.reshape(-1, 2)
    n = centers.shape[0]
    p = 0.0
    grad = np.zeros_like(centers)
    
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            d = np.hypot(dx, dy)
            overlap = radii[i] + radii[j] - d
            if overlap > 0.0:
                p += overlap**2
                f = 2.0 * overlap / (d + 1e-10)
                grad[i, 0] -= f * dx
                grad[i, 1] -= f * dy
                grad[j, 0] += f * dx
                grad[j, 1] += f * dy
                
    for i in range(n):
        r = radii[i]
        x, y = centers[i]
        if x < r:
            ov = r - x
            p += ov**2
            grad[i, 0] += 2.0 * ov
        if x > 1.0 - r:
            ov = x - (1.0 - r)
            p += ov**2
            grad[i, 0] -= 2.0 * ov
        if y < r:
            ov = r - y
            p += ov**2
            grad[i, 1] += 2.0 * ov
        if y > 1.0 - r:
            ov = y - (1.0 - r)
            p += ov**2
            grad[i, 1] -= 2.0 * ov
            
    return p, grad.flatten()

def opt_centers_lbfgs(centers, radii, max_iter=1000):
    """Optimize centers to minimize overlap/boundary penalties."""
    bounds = [(0.0, 1.0)] * (2 * N)
    try:
        res = minimize(center_penalty_and_grad, centers.flatten(), args=(radii,),
                       jac=True, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-12})
        return res.x.reshape(-1, 2)
    except Exception:
        return centers.copy()

def constraints_slsqp(x):
    """Inequality constraints for SLSQP: g(x) >= 0."""
    xs, ys, rs = x[0::3], x[1::3], x[2::3]
    c = np.concatenate([xs - rs, 1.0 - xs - rs, ys - rs, 1.0 - ys - rs])
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    c = np.concatenate([c, np.hypot(dx[TRIL_IDX], dy[TRIL_IDX]) - dr[TRIL_IDX]])
    return c

def objective_slsqp(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def get_bounds_slsqp():
    """Variable bounds for SLSQP."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)])
    return b

def make_hex_init(r0, angle):
    """Generates a rotated hexagonal lattice initialization."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 5:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0 and len(pts) < N + 5:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
    pts = np.array(pts[:N+5])
    if angle != 0.0:
        ca, sa = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[ca, -sa], [sa, ca]]) + 0.5
    mask = (pts[:, 0] >= 0.01) & (pts[:, 0] <= 0.99) & (pts[:, 1] >= 0.01) & (pts[:, 1] <= 0.99)
    pts = pts[mask]
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.1, 0.9, (1, 2))])
    return pts[:N]

def make_force_init(seed):
    """Force-directed layout to spread points evenly and push to boundaries."""
    np.random.seed(seed)
    c = np.random.uniform(0.1, 0.9, (N, 2))
    for step in range(200):
        f = np.zeros_like(c)
        lr = 0.05 * (1.0 - step / 200.0)
        for i in range(N):
            for j in range(i+1, N):
                dx = c[j] - c[i]
                d = np.hypot(dx[0], dx[1])
                if d < 0.3 and d > 1e-5:
                    rep = 0.01 / (d**2 + 0.001)
                    f[i] -= dx * rep / d
                    f[j] += dx * rep / d
            for dim in range(2):
                if c[i, dim] < 0.1: f[i, dim] += 0.06
                elif c[i, dim] > 0.9: f[i, dim] -= 0.06
        c += f * lr
        c = np.clip(c, 0.02, 0.98)
    return c

def run_packing():

    np.random.seed(42)
    best_sum = -1.0
    best_centers = None
    best_radii = None

    # Phase 1: Generate Diverse Initial Configurations
    inits = []
    for r0 in [0.085, 0.090, 0.095, 0.100, 0.105]:
        for ang in np.linspace(-0.35, 0.35, 13):
            inits.append(make_hex_init(r0, ang))
    for s in range(12):
        inits.append(make_force_init(s))

    # Phase 2: Expansion-Optimization Alternating Loop
    for c_init in inits:
        c = c_init.copy()
        r = solve_lp_radii(c)
        
        for _ in range(25):
            r = solve_lp_radii(c)
            # Expand radii slightly to force topology changes
            r_exp = r * 1.015
            c = opt_centers_lbfgs(c, r_exp, max_iter=800)
            r = solve_lp_radii(c)
            
        s = np.sum(r)
        if s > best_sum:
            best_sum = s
            best_centers = c.copy()
            best_radii = r.copy()
            
    # Phase 3: Deflation & Perturbation to Escape Local Minima
    if best_centers is not None:
        for step in range(70):
            c = best_centers.copy()
            r = best_radii.copy()
            
            # Deflate a random subset to break rigid contacts
            k = np.random.randint(3, 8)
            subset = np.random.choice(N, size=k, replace=False)
            r[subset] *= 0.80
            
            # Cooling perturbation
            noise = 0.0025 * (0.94 ** step)
            c += np.random.normal(0, noise, c.shape)
            c = np.clip(c, 0.01, 0.99)
            
            # Resolve configuration for deflated radii
            c = opt_centers_lbfgs(c, r, max_iter=500)
            r_new = solve_lp_radii(c)
            s_new = np.sum(r_new)
            
            if s_new > best_sum:
                best_sum = s_new
                best_centers = c.copy()
                best_radii = r_new.copy()
                
                # Deep refine from new best
                for _ in range(12):
                    best_radii = solve_lp_radii(best_centers)
                    best_centers = opt_centers_lbfgs(best_centers, best_radii * 1.012, max_iter=600)
                    best_radii = solve_lp_radii(best_centers)
                    best_sum = np.sum(best_radii)

    # Phase 4: Final SLSQP Polish
    if best_centers is not None:
        x0 = np.zeros(3 * N)
        x0[0::3] = best_centers[:, 0]
        x0[1::3] = best_centers[:, 1]
        x0[2::3] = best_radii
        
        bounds = get_bounds_slsqp()
        cons = {'type': 'ineq', 'fun': constraints_slsqp}
        
        try:
            res = minimize(objective_slsqp, x0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 15000, 'ftol': 1e-13, 'disp': False})

                c_s = np.column_stack((res.x[0::3], res.x[1::3]))

                break

                    d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                    if d < radii[i] + radii[j] - 1e-9:
                        valid = False
                        break

                    break

            break
            
        radii *= 0.9995
        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)

    return centers, radii, float(np.sum(radii))
