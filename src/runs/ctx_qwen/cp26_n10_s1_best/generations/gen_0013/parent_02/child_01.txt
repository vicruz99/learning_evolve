import numpy as np
from scipy.optimize import minimize, linprog
import warnings
warnings.filterwarnings('ignore')

N = 26
TRIL_IDX = np.tril_indices(N, -1)
TRIU_IDX = np.triu_indices(N, 1)

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    num_ineq = n + n * (n - 1) // 2
    A = np.zeros((num_ineq, n))
    b = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A[idx, i] = 1.0
        b[idx] = max(0.0, lim)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = max(0.0, d)
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(-np.ones(n), A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-9)
    except Exception:
        pass
    return np.full(n, 0.05)

def overlap_penalty(centers_flat, radii):
    """Vectorized smooth penalty function for center optimization."""
    c = centers_flat.reshape(-1, 2)
    p = 0.0
    
    # Boundary penalties
    diff_x = np.clip(radii - c[:, 0], None, 0.0)
    diff_x_r = np.clip(c[:, 0] - (1.0 - radii), None, 0.0)
    diff_y = np.clip(radii - c[:, 1], None, 0.0)
    diff_y_t = np.clip(c[:, 1] - (1.0 - radii), None, 0.0)
    p += np.sum(diff_x**2) + np.sum(diff_x_r**2) + np.sum(diff_y**2) + np.sum(diff_y_t**2)
    
    # Pairwise overlap penalties
    dx = c[:, 0, None] - c[:, 0]
    dy = c[:, 1, None] - c[:, 1]
    d = np.hypot(dx, dy)
    gap = d - radii[:, None] - radii[None, :]
    p += np.sum(np.clip(-gap[TRIU_IDX], None, 0.0)**2)
    
    return p

def opt_centers_fixed_radii(centers, radii):
    """Optimize centers for fixed radii using L-BFGS-B."""
    bounds = [(0.001, 0.999)] * (2 * N)
    res = minimize(overlap_penalty, centers.flatten(), args=(radii,),
                   method='L-BFGS-B', bounds=bounds, options={'maxiter': 800, 'ftol': 1e-15})
    return res.x.reshape(-1, 2)

def make_hex_init(r0, angle):
    """Generates a rotated hexagonal lattice initialization."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0 and len(pts) < N + 10:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        
    pts = np.array(pts[:N + 10])
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
        
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.2, 0.8, (1, 2))])
    return np.clip(pts[:N], 0.05, 0.95)

def make_force_init(seed):
    """Force-directed layout to spread points evenly."""
    np.random.seed(seed)
    pts = np.random.uniform(0.15, 0.85, (N, 2))
    for _ in range(200):
        f = np.zeros_like(pts)
        for i in range(N):
            for j in range(i + 1, N):
                dx = pts[j] - pts[i]
                d = np.hypot(dx[0], dx[1])
                if d < 0.25 and d > 1e-5:
                    rep = 0.015 / (d**2 + 0.001)
                    f[i] -= dx * rep / d
                    f[j] += dx * rep / d
            for dim in range(2):
                if pts[i, dim] < 0.12: f[i, dim] += 0.06
                elif pts[i, dim] > 0.88: f[i, dim] -= 0.06
        pts += f * 0.04
        pts = np.clip(pts, 0.03, 0.97)
    return pts

def run_alternating(init_centers, steps=25):
    """Alternating LP radii and L-BFGS-B center optimization."""
    c = init_centers.copy()
    r = solve_lp_radii(c) * 0.98  # Shrink slightly to ensure strict feasibility
    
    for _ in range(steps):
        r = solve_lp_radii(c)
        c = opt_centers_fixed_radii(c, r)
        # If fully feasible, we might have converged
        if overlap_penalty(c.flatten(), r) < 1e-8:
            break
    return c, solve_lp_radii(c)

def joint_constraints(x):
    """Inequality constraints for SLSQP: g(x) >= 0"""
    xs, ys, rs = x[0::3], x[1::3], x[2::3]
    c = np.concatenate([
        xs - rs, 1.0 - xs - rs, ys - rs, 1.0 - ys - rs
    ])
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    c = np.concatenate([c, dx[TRIL_IDX]**2 + dy[TRIL_IDX]**2 - dr[TRIL_IDX]**2])
    return c

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    np.random.seed(42)

    for s in range(15):

    # Random starts
    for s in range(10):
        np.random.seed(s + 500)
        inits.append(np.random.uniform(0.15, 0.85, (N, 2)))
        
    # Phase 2: Alternating Optimization + SLSQP Polish
    for c0 in inits:
        c_alt, r_alt = run_alternating(c0, steps=20)
        curr_sum = np.sum(r_alt)
        
        # Prepare for SLSQP
        x0 = np.zeros(3 * N)
        x0[0::3] = c_alt[:, 0]
        x0[1::3] = c_alt[:, 1]
        x0[2::3] = r_alt * 0.995
        
        try:
            res = minimize(lambda x: -np.sum(x[2::3]), x0, method='SLSQP', 
                           bounds=bounds_obj, constraints=cons,
                           options={'maxiter': 8000, 'ftol': 1e-14, 'disp': False})
            if not np.isnan(res.fun):
                x_tmp = res.x.copy()
                c_tmp = np.column_stack((x_tmp[0::3], x_tmp[1::3]))
                r_lp = solve_lp_radii(c_tmp)
                x_tmp[2::3] = r_lp
                s_val = np.sum(r_lp)
                if s_val > best_sum and np.min(joint_constraints(x_tmp)) >= -1e-5:
                    best_sum = s_val
                    best_x = x_tmp.copy()
        except Exception:
            pass
            
    # Phase 3: Deflation & Perturbation to Escape Local Minima
