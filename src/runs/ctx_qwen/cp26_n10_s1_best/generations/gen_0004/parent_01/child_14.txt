import numpy as np
from scipy.optimize import minimize, linprog

N_CIRCLES = 26

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraint_func(x):
    """Vectorized inequality constraints: g(x) >= 0."""
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    # Boundary constraints: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    b = np.concatenate([
        cx - r,
        1.0 - cx - r,
        cy - r,
        1.0 - cy - r
    ])
    
    # Overlap constraints: dist^2 - (r1+r2)^2 >= 0
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dr = r[:, None] + r[None, :]
    
    i, j = np.tril_indices(N_CIRCLES, -1)
    o = dx[i, j]**2 + dy[i, j]**2 - dr[i, j]**2
    
    return np.concatenate([b, o])

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-7, 0.5]."""
    b = []
    for _ in range(N_CIRCLES):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)])
    return b

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        b_ub[i] = max(0.0, lim)
        A_ub[i, i] = 1.0
        idx += 1
        
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist)
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def generate_hex_points(n, r0, angle_deg, seed=0):
    """Generate hexagonal lattice points, optionally rotated and perturbed."""
    np.random.seed(seed)
    pts = []
    y = r0
    row = 0
    while len(pts) < n + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        if y > 1.0 + r0:
            break
            
    pts = np.array(pts[:n + 10])
    
    if angle_deg != 0:
        rad = np.deg2rad(angle_deg)
        c, s = np.cos(rad), np.sin(rad)
        center = np.array([0.5, 0.5])
        pts = (pts - center) @ np.array([[c, -s], [s, c]]) + center
        
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    
    while len(pts) < n:
        pts = np.vstack([pts, [np.random.uniform(0.1, 0.9), np.random.uniform(0.1, 0.9)]])
        
    return pts[:n] + np.random.normal(0, 0.002, (n, 2))

def generate_grid_points(n, seed=0):
    """Generate perturbed grid points."""
    np.random.seed(seed)
    pts = []
    for r in range(6):
        for c in range(5):
            if len(pts) >= n: break
            pts.append([0.08 + c * 0.18, 0.08 + r * 0.16])
    return np.array(pts[:n]) + np.random.normal(0, 0.01, (n, 2))

def generate_corner_points(n, seed=0):
    """Generate points clustered near corners and edges to test boundary limits."""
    np.random.seed(seed)
    pts = []
    corners = [[0.1, 0.1], [0.9, 0.1], [0.1, 0.9], [0.9, 0.9]]
    for cx, cy in corners:
        for _ in range(5):
            pts.append([cx + np.random.normal(0, 0.05), cy + np.random.normal(0, 0.05)])
            
    while len(pts) < n:
        edge = np.random.choice([0, 1, 2, 3])
        p = [np.random.uniform(0.05, 0.95), np.random.uniform(0.05, 0.95)]
        if edge == 0: p[0] = np.random.uniform(0.05, 0.25)
        elif edge == 1: p[0] = np.random.uniform(0.75, 0.95)
        elif edge == 2: p[1] = np.random.uniform(0.05, 0.25)
        else: p[1] = np.random.uniform(0.75, 0.95)
        pts.append(p)
        
    return np.clip(np.array(pts[:n]), 0.05, 0.95)

def project_to_bounds(x, n):
    """Ensure variables respect x >= r, 1-x >= r bounds."""
    for i in range(n):
        r = max(1e-6, x[3*i+2])
        x[3*i] = np.clip(x[3*i], r, 1.0 - r)
        x[3*i+1] = np.clip(x[3*i+1], r, 1.0 - r)
        x[3*i+2] = r
    return x

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    n = N_CIRCLES
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraint_func}
    
    best_sum = -1.0
    best_x = None
    
    # Phase 1: Diverse Initial Configurations
    inits = []
    for s in range(20):
        inits.append(generate_hex_points(n, 0.09, angle_deg=s * 3.0, seed=s))
    for s in range(10):
        inits.append(generate_grid_points(n, seed=s))
    for s in range(10):
        inits.append(generate_corner_points(n, seed=s))
        
    for c_init in inits:
        r_init = solve_lp_radii(c_init)
        if r_init is None:
            r_init = np.full(n, 0.06)
            
        x0 = np.zeros(3 * n)
        x0[0::3] = c_init[:, 0]
        x0[1::3] = c_init[:, 1]
        x0[2::3] = r_init
        x0 = project_to_bounds(x0, n)
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 12000, 'ftol': 1e-14, 'disp': False})
            if not np.isnan(res.fun) and -res.fun > best_sum:
                best_sum = -res.fun
                best_x = res.x.copy()
        except Exception:
            continue

    # Phase 2: Basin-Hopping Refinement with LP Guidance
    if best_x is not None:
        for epoch in range(120):
            # Adaptive noise decay
            noise_scale = 0.006 * (0.96 ** epoch)
            x_pert = best_x.copy()
            
            # Perturb centers
            x_pert[0::3] += np.random.normal(0, noise_scale, n)
            x_pert[1::3] += np.random.normal(0, noise_scale, n)
            
            # Shrink radii to free up space for topological changes
            x_pert[2::3] *= 0.985
            
            x_pert = project_to_bounds(x_pert, n)
            
            # LP step: snap radii to optimal values for the new perturbed centers
            c_pert = np.column_stack((x_pert[0::3], x_pert[1::3]))
            r_lp = solve_lp_radii(c_pert)
            if r_lp is not None:
                x_pert[2::3] = np.maximum(r_lp, 1e-6)
                x_pert = project_to_bounds(x_pert, n)
                
            try:
                res = minimize(objective, x_pert, method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 10000, 'ftol': 1e-14, 'disp': False})
                if not np.isnan(res.fun):
                    curr_sum = -res.fun
                    c_vals = constraint_func(res.x)
                    if np.min(c_vals) >= -1e-6 and curr_sum > best_sum:
                        best_sum = curr_sum
                        best_x = res.x.copy()
            except Exception:
                pass

    # Extract results
    if best_x is None:
        best_x = np.zeros(3 * n)
        best_x[2::3] = 0.06
        best_x[0::3] = np.tile(np.linspace(0.1, 0.9, 5), 6)[:n]
        best_x[1::3] = np.repeat(np.linspace(0.1, 0.9, 6), 5)[:n]
        
    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3].copy()
    
    # Final strict validation and minimal numerical repair
    for _ in range(200):
        valid = True
        for i in range(n):
            if radii[i] < 0: valid = False; break
            if centers[i, 0] - radii[i] < -1e-12 or centers[i, 0] + radii[i] > 1.0 + 1e-12: valid = False; break
            if centers[i, 1] - radii[i] < -1e-12 or centers[i, 1] + radii[i] > 1.0 + 1e-12: valid = False; break
        if valid:
            for i in range(n):
                for j in range(i + 1, n):
                    d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                    if d < radii[i] + radii[j] - 1e-12:
                        valid = False; break
                if not valid: break
        if valid: break
            
        # Minimal shrinkage to recover strict feasibility
        radii *= 0.9998
        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)
        
    return centers, radii, float(np.sum(radii))
