

<strategy>
To push the sum of radii beyond 2.636, we need to escape the local minima that trap standard hexagonal packings. The strategy focuses on:
1. **Corner & Edge Exploitation**: Hexagonal lattices waste space near boundaries. We will explicitly generate initial configurations that force circles into corners and along edges, then fill the interior.
2. **Rigidity Breaking**: Stagnant hexagonal clusters prevent further expansion. We implement a force-directed "shaker" that applies repulsive forces and random rotations to sub-clusters to break symmetry before re-optimization.
3. **Alternating LP & NLP**: We alternate between Linear Programming (fast, exact radii for fixed centers) and Non-Linear Programming (SLSQP/trust-constr for joint center/radius optimization). LP quickly snaps radii to their theoretical maximum for a given topology, while NLP shifts centers to create more room.
4. **Deflation-Reinflation Cycles**: We periodically shrink radii, perturb positions heavily, and let the optimizer re-expand them. This helps jump over energy barriers in the constraint landscape.
5. **Robust Validation**: A strict final check with minimal shrinkage guarantees compliance with the `1e-12` tolerance without sacrificing significant objective value.
</strategy>

import numpy as np
from scipy.optimize import minimize, linprog
import math

N_CIRCLES = 26
TRIL_IDX = np.tril_indices(N_CIRCLES, -1)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Returns all inequality constraints g(x) >= 0."""
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    # Boundary constraints
    c = np.concatenate([
        cx - r,
        1.0 - cx - r,
        cy - r,
        1.0 - cy - r
    ])
    
    # Overlap constraints: dist^2 - (r_i + r_j)^2 >= 0
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dr = r[:, None] + r[None, :]
    
    c = np.concatenate([c, dx[TRIL_IDX]**2 + dy[TRIL_IDX]**2 - dr[TRIL_IDX]**2])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-7, 0.5]."""
    b = []
    for _ in range(N_CIRCLES):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)])
    return b

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        b_ub[idx] = max(0.0, lim)
        A_ub[idx, i] = 1.0
        idx += 1
        
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist)
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def force_directed_spread(centers, steps=400, temp=0.05):
    """Aggressive force-directed layout to break rigid clusters and push to boundaries."""
    pts = centers.copy()
    n = pts.shape[0]
    for step in range(steps):
        forces = np.zeros_like(pts)
        decay = 1.0 - step / steps
        for i in range(n):
            for j in range(i + 1, n):
                dx = pts[j, 0] - pts[i, 0]
                dy = pts[j, 1] - pts[i, 1]
                dist = np.hypot(dx, dy)
                if dist < 0.35 and dist > 1e-6:
                    f = 0.025 / (dist**2 + 0.002)
                    fx, fy = f * dx, f * dy
                    forces[i] -= [fx, fy]
                    forces[j] += [fx, fy]
            # Boundary repulsion
            for dim in range(2):
                if pts[i, dim] < 0.08: forces[i, dim] += 0.06 * decay
                elif pts[i, dim] > 0.92: forces[i, dim] -= 0.06 * decay
                
        pts += forces * 0.08 * decay
        pts = np.clip(pts, 0.01, 0.99)
    return pts

def generate_corner_hex(n, r0, angle=0.0):
    """Hexagonal lattice explicitly anchored to corners and edges."""
    pts = []
    # Place 4 in corners
    margin = r0 + 0.01
    pts.append([margin, margin])
    pts.append([1.0 - margin, margin])
    pts.append([margin, 1.0 - margin])
    pts.append([1.0 - margin, 1.0 - margin])
    
    y = r0 * 1.5
    row = 0
    while len(pts) < n:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0 and len(pts) < n:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        if y > 1.0 - r0: break
        
    pts = np.array(pts[:n])
    if angle != 0.0:
        rad = math.radians(angle)
        c, s = np.cos(rad), np.sin(rad)
        center = np.array([0.5, 0.5])
        pts = (pts - center) @ np.array([[c, -s], [s, c]]) + center
        
    return np.clip(pts, 0.02, 0.98)

def generate_grid_init(n, spacing=0.18):
    """Perturbed grid initialization."""
    pts = []
    for r in range(7):
        for c in range(6):
            if len(pts) >= n: break
            x = 0.08 + c * spacing
            y = 0.08 + r * spacing
            pts.append([x, y])
    pts = np.array(pts[:n])
    pts += np.random.uniform(-0.015, 0.015, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(7)
    n = N_CIRCLES
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Phase 1: Diverse Initializations
    inits = []
    # Corner-anchored hex grids
    for r0 in [0.085, 0.095, 0.105]:
        for ang in [0, 15, 30, 45, 60]:
            inits.append(generate_corner_hex(n, r0, ang))
    # Grid starts
    for _ in range(5):
        inits.append(generate_grid_init(n, spacing=0.17 + np.random.uniform(-0.02, 0.02)))
    # Random + Force spread
    for _ in range(4):
        rand_pts = np.random.uniform(0.1, 0.9, (n, 2))
        inits.append(force_directed_spread(rand_pts, steps=200))
        
    for pts in inits:
        radii = np.full(n, 0.07)
        # Safety margin radii
        for i in range(n):
            d_min = min(pts[i,0], 1.0-pts[i,0], pts[i,1], 1.0-pts[i,1])
            for j in range(n):
                if i != j:
                    d = np.hypot(pts[i,0]-pts[j,0], pts[i,1]-pts[j,1])
                    if d < d_min: d_min = d
            radii[i] = d_min * 0.4
            
        x0 = np.zeros(3 * n)
        x0[0::3] = pts[:, 0]
        x0[1::3] = pts[:, 1]
        x0[2::3] = radii
        
        # SLSQP Optimization
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 8000, 'ftol': 1e-14, 'disp': False})
            if not np.isnan(res.fun):
                c_opt = np.column_stack((res.x[0::3], res.x[1::3]))
                r_opt = res.x[2::3]
                curr_sum = -res.fun
                if curr_sum > best_sum:
                    best_sum = curr_sum
                    best_centers = c_opt.copy()
                    best_radii = r_opt.copy()
        except Exception:
            continue

    # Phase 2: Alternating LP + Center Optimization + Deflation
    if best_centers is not None:
        for round_idx in range(12):
            # LP refinement
            lp_r = solve_lp_radii(best_centers)
            if lp_r is not None:
                best_radii = lp_r
                best_sum = np.sum(best_radii)
                
            # Deflation & Perturbation to escape local minima
            shrink_factor = 0.96 + 0.03 * np.random.random()
            pert_r = best_radii * shrink_factor
            pert_c = best_centers + np.random.normal(0, 0.004, best_centers.shape)
            pert_c = np.clip(pert_c, 0.02, 0.98)
            
            x0 = np.zeros(3 * n)
            x0[0::3] = pert_c[:, 0]
            x0[1::3] = pert_c[:, 1]
            x0[2::3] = pert_r
            
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 6000, 'ftol': 1e-14, 'disp': False})
                if not np.isnan(res.fun):
                    c_new = np.column_stack((res.x[0::3], res.x[1::3]))
                    r_new = res.x[2::3]
                    
                    # LP snap
                    lp_r_new = solve_lp_radii(c_new)
                    if lp_r_new is not None:
                        r_new = lp_r_new
                        s_new = np.sum(r_new)
                    else:
                        s_new = -res.fun
                        
                    if s_new > best_sum:
                        best_sum = s_new
                        best_centers = c_new.copy()
                        best_radii = r_new.copy()
            except Exception:
                pass
                
            # Rigidity breaker: force spread then re-optimize
            if round_idx % 3 == 0:
                broken_c = force_directed_spread(best_centers, steps=150)
                broken_r = solve_lp_radii(broken_c)
                if broken_r is not None:
                    x0_b = np.zeros(3*n)
                    x0_b[0::3] = broken_c[:,0]
                    x0_b[1::3] = broken_c[:,1]
                    x0_b[2::3] = broken_r
                    try:
                        res_b = minimize(objective, x0_b, method='SLSQP', bounds=bounds,
                                         constraints=cons, options={'maxiter': 5000, 'ftol': 1e-14, 'disp': False})
                        if not np.isnan(res_b.fun):
                            s_b = -res_b.fun
                            if s_b > best_sum:
                                best_sum = s_b
                                best_centers = np.column_stack((res_b.x[0::3], res_b.x[1::3]))
                                best_radii = res_b.x[2::3]
                    except Exception:
                        pass

    # Fallback
    if best_centers is None:
        best_centers = generate_corner_hex(n, 0.09)
        best_radii = np.full(n, 0.08)
        best_sum = np.sum(best_radii)
        
    # Phase 3: Strict Validation & Minimal Numerical Repair
    for _ in range(100):
        valid = True
        for i in range(n):
            if best_radii[i] < 1e-12:
                valid = False; break
            if best_centers[i,0] - best_radii[i] < -1e-13 or best_centers[i,0] + best_radii[i] > 1.0 + 1e-13:
                valid = False; break
            if best_centers[i,1] - best_radii[i] < -1e-13 or best_centers[i,1] + best_radii[i] > 1.0 + 1e-13:
                valid = False; break
        if valid:
            for i in range(n):
                for j in range(i + 1, n):
                    d = np.hypot(best_centers[i,0] - best_centers[j,0], best_centers[i,1] - best_centers[j,1])
                    if d < best_radii[i] + best_radii[j] - 1e-13:
                        valid = False; break
                if not valid: break
        if valid: break
        
        # Gentle shrinkage to recover strict feasibility
        best_radii *= 0.9995
        best_centers[:, 0] = np.clip(best_centers[:, 0], best_radii, 1.0 - best_radii)
        best_centers[:, 1] = np.clip(best_centers[:, 1], best_radii, 1.0 - best_radii)
        
    return best_centers, best_radii, float(np.sum(best_radii))
