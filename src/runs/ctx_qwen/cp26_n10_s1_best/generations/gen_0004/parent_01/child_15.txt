

<strategy>
The core difficulty in circle packing is the non-convexity of the distance constraints. Direct non-linear optimization (like SLSQP on squared distances) often gets trapped in local minima where circles are "jammed" in suboptimal configurations.

To break through the 2.634 barrier, we will use a **LP-Duality Guided Gradient Ascent** strategy:
1. **Decoupling**: Treat radii and centers separately. For any fixed set of centers, finding the optimal radii is a *Linear Programming* problem: maximize $\sum r_i$ subject to $r_i + r_j \le d_{ij}$ and $r_i \le \text{boundary}_i$.
2. **Exact Gradients via Duality**: The optimal LP solution provides dual variables (Lagrange multipliers) for each constraint. By sensitivity analysis, the gradient of the total sum of radii with respect to the center positions is exactly determined by these dual variables. Specifically, moving centers apart along the line connecting them increases the clearance $d_{ij}$, and the rate of improvement is given by the dual variable $\lambda_{ij}$. This gives a mathematically exact direction to push circles apart to maximize the sum of radii.
3. **Projected Gradient Ascent**: We perform gradient ascent on the centers using the dual-derived gradients, projecting back to $[0,1]^2$ at each step.
4. **Stagnation Escape & Multi-Start**: If progress stalls, we add a controlled random perturbation to escape local minima. We run this from diverse initial configurations (hexagonal lattices, force-spread random points, rotated grids).
5. **Polishing & Validation**: After convergence, we compute the final optimal radii via LP, apply a minimal safety shrink to guarantee strict compliance with the $10^{-12}$ validator tolerance, and return the result.

This approach is highly robust, avoids the numerical pitfalls of squared-distance constraints near contact, and directly targets the objective function's geometry.
</strategy>

import numpy as np
from scipy.optimize import linprog

N_CIRCLES = 26

def get_lp_and_grad(centers):
    """
    Solve LP for optimal radii given fixed centers.
    Return radii, sum_of_radii, and exact gradient of sum_of_radii w.r.t centers 
    computed via LP dual variables.
    """
    n = centers.shape[0]
    c_obj = -np.ones(n)  # Minimize -sum(r) <=> Maximize sum(r)
    num_ineq = 4 * n + n * (n - 1) // 2
    
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    pair_map = []
    
    # Boundary constraints: r_i <= x, 1-x, y, 1-y
    for i in range(n):
        x, y = centers[i]
        vals = [x, 1.0 - x, y, 1.0 - y]
        for v in vals:
            A_ub[idx, i] = 1.0
            b_ub[idx] = max(0.0, v)
            idx += 1
            
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            d = np.hypot(dx, dy)
            if d < 1e-9: d = 1e-9
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = d
            pair_map.append((i, j))
            idx += 1
            
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if not res.success:
        return None, None, None
        
    radii = res.x
    duals = np.asarray(res.ineqlin.marginals)
    grad = np.zeros_like(centers)
    
    # Boundary gradients: d(sum_r)/d(x_i) = lambda(x_i) - lambda(1-x_i)
    bd_duals = duals[:4*n]
    for i in range(n):
        grad[i, 0] += bd_duals[4*i] - bd_duals[4*i + 1]
        grad[i, 1] += bd_duals[4*i + 2] - bd_duals[4*i + 3]
        
    # Overlap gradients: d(sum_r)/d(c_i) = sum_j lambda_ij * (c_i - c_j) / d_ij
    ov_duals = duals[4*n:]
    for k, (i, j) in enumerate(pair_map):
        lam = ov_duals[k]
        if lam > 1e-8:
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            d = np.hypot(dx, dy)
            if d > 1e-9:
                inv_d = 1.0 / d
                grad[i, 0] += lam * dx * inv_d
                grad[i, 1] += lam * dy * inv_d
                grad[j, 0] -= lam * dx * inv_d
                grad[j, 1] -= lam * dy * inv_d
                
    return radii, np.sum(radii), grad

def run_gradient_ascent(centers, steps=300, max_step=0.04):
    """Optimize centers using LP-duality guided gradient ascent."""
    best_sum = -1.0
    best_centers = centers.copy()
    no_improve = 0
    threshold = 1e-8
    
    for step in range(steps):
        radii, curr_sum, grad = get_lp_and_grad(centers)
        if grad is None:
            break
            
        if curr_sum > best_sum + threshold:
            best_sum = curr_sum
            best_centers = centers.copy()
            no_improve = 0
        else:
            no_improve += 1
            
        # Escape local minima if stagnant
        if no_improve > 40:
            centers += np.random.normal(0, 0.004, centers.shape)
            centers = np.clip(centers, 1e-5, 1.0 - 1e-5)
            no_improve = 0
            continue
            
        lr = 0.05 * (0.96 ** (step // 40))
        update = lr * grad
        step_norm = np.linalg.norm(update)
        if step_norm > max_step:
            update *= max_step / step_norm
            
        centers += update
        centers = np.clip(centers, 1e-5, 1.0 - 1e-5)
        
    # Final exact radii computation
    radii, best_sum, _ = get_lp_and_grad(best_centers)
    return best_centers, radii, best_sum

def generate_starts(n, num_random=10):
    """Generate diverse initial center configurations."""
    starts = []
    np.random.seed(42)
    
    # Force-spread random starts
    for s in range(num_random):
        c = np.random.uniform(0.15, 0.85, (n, 2))
        for _ in range(150):
            forces = np.zeros_like(c)
            for i in range(n):
                for j in range(i + 1, n):
                    d = np.hypot(c[i,0]-c[j,0], c[i,1]-c[j,1])
                    if d < 0.35 and d > 1e-5:
                        f = 0.015 / (d**2 + 0.001)
                        dx = c[i,0]-c[j,0]
                        dy = c[i,1]-c[j,1]
                        forces[i] += f * np.array([dx, dy])
                        forces[j] -= f * np.array([dx, dy])
            c += forces * 0.04
            c = np.clip(c, 0.05, 0.95)
        starts.append(c.copy())
        
    # Hexagonal lattice starts with rotations
    for r0 in [0.082, 0.090, 0.098]:
        for ang in [0.0, 0.12, 0.25, 0.40]:
            pts = []
            y = r0
            row = 0
            while len(pts) < n + 6:
                x = r0 if row % 2 == 0 else 2.0 * r0
                while x <= 1.0 - r0:
                    pts.append([x, y])
                    x += 2.0 * r0
                y += np.sqrt(3.0) * r0
                row += 1
                if y > 1.0 + r0: break
            pts = np.array(pts[:n])
            if ang != 0.0:
                ca, sa = np.cos(ang), np.sin(ang)
                c0 = np.array([0.5, 0.5])
                pts = (pts - c0) @ np.array([[ca, -sa], [sa, ca]]) + c0
            pts += np.random.uniform(-0.004, 0.004, pts.shape)
            pts = np.clip(pts, 0.05, 0.95)
            starts.append(pts.copy())
            
    return starts

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    n = N_CIRCLES
    starts = generate_starts(n, num_random=12)
    
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Phase 1: Multi-start gradient ascent
    for c0 in starts:
        c_opt, r_opt, s_opt = run_gradient_ascent(c0, steps=250, max_step=0.035)
        if s_opt > best_sum:
            best_sum = s_opt
            best_centers = c_opt
            best_radii = r_opt
            
    # Phase 2: Fine perturbation refinement on the best found configuration
    if best_centers is not None:
        for _ in range(8):
            c_pert = best_centers + np.random.normal(0, 0.0015, best_centers.shape)
            c_pert = np.clip(c_pert, 1e-5, 1.0 - 1e-5)
            c_opt, r_opt, s_opt = run_gradient_ascent(c_pert, steps=150, max_step=0.025)
            if s_opt > best_sum:
                best_sum = s_opt
                best_centers = c_opt
                best_radii = r_opt
                
    # Fallback (should not be reached)
    if best_centers is None:
        best_centers = np.random.uniform(0.1, 0.9, (n, 2))
        _ , best_radii, best_sum = get_lp_and_grad(best_centers)
        
    # Final strict validation & minimal numerical repair to guarantee 1e-12 compliance
    for _ in range(100):
        valid = True
        for i in range(n):
            if best_radii[i] < 0: valid=False; break
            if best_centers[i,0] - best_radii[i] < -1e-12 or best_centers[i,0] + best_radii[i] > 1 + 1e-12: valid=False; break
            if best_centers[i,1] - best_radii[i] < -1e-12 or best_centers[i,1] + best_radii[i] > 1 + 1e-12: valid=False; break
        if valid:
            for i in range(n):
                for j in range(i+1, n):
                    d = np.hypot(best_centers[i,0]-best_centers[j,0], best_centers[i,1]-best_centers[j,1])
                    if d < best_radii[i] + best_radii[j] - 1e-12:
                        valid = False; break
                if not valid: break
        if valid: break
        best_radii *= 0.99995
        
    return best_centers, best_radii, float(best_sum)
