from scipy.optimize import minimize, linprog

N_CIRCLES = 26

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def objective_jac(x):
    """Jacobian of the objective function."""
    jac = np.zeros(3 * N_CIRCLES)
    jac[2::3] = -1.0
    return jac

def c_fun(x):
    """Inequality constraints g(x) >= 0."""
    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]
    
    # Boundary constraints
    c = np.concatenate([xs - rs, 1.0 - xs - rs, ys - rs, 1.0 - ys - rs])
    
    # Overlap constraints (squared distance - sum_radii^2 >= 0)
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    idx = np.tril_indices(N_CIRCLES, -1)
    c = np.concatenate([c, dx[idx]**2 + dy[idx]**2 - dr[idx]**2])
    return c

def c_jac(x):
    """Jacobian of the constraint function."""
    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]
    
    m = 4 * N_CIRCLES + N_CIRCLES * (N_CIRCLES - 1) // 2
    jac = np.zeros((m, 3 * N_CIRCLES))
    
    # Boundary constraints jacobian
    for i in range(N_CIRCLES):
        jac[i, 3*i] = 1.0; jac[i, 3*i+2] = -1.0
        jac[N_CIRCLES+i, 3*i] = -1.0; jac[N_CIRCLES+i, 3*i+2] = -1.0
        jac[2*N_CIRCLES+i, 3*i+1] = 1.0; jac[2*N_CIRCLES+i, 3*i+2] = -1.0
        jac[3*N_CIRCLES+i, 3*i+1] = -1.0; jac[3*N_CIRCLES+i, 3*i+2] = -1.0
        
    # Overlap constraints jacobian
    base = 4 * N_CIRCLES
    idx = np.tril_indices(N_CIRCLES, -1)
    for k, (i, j) in enumerate(zip(idx[0], idx[1])):
        r = base + k
        dxi = xs[i] - xs[j]
        dyi = ys[i] - ys[j]
        dri = rs[i] + rs[j]
        
        jac[r, 3*i] = 2.0 * dxi; jac[r, 3*j] = -2.0 * dxi
        jac[r, 3*i+1] = 2.0 * dyi; jac[r, 3*j+1] = -2.0 * dyi
        jac[r, 3*i+2] = -2.0 * dri; jac[r, 3*j+2] = -2.0 * dri
        
    return jac

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist)
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def make_init_hex(seed, angle=0.0, r0=0.09):
    """Generate hexagonal lattice initialization with optional rotation."""
    np.random.seed(seed)
    pts = []
    y = r0
    row = 0
    while len(pts) < N_CIRCLES + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        if y > 1.0 + r0:
            break
            
    pts = np.array(pts[:N_CIRCLES + 10])
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        center = np.array([0.5, 0.5])
        pts = (pts - center) @ np.array([[c, -s], [s, c]]) + center
        
    mask = (pts[:,0]>=0.02) & (pts[:,0]<=0.98) & (pts[:,1]>=0.02) & (pts[:,1]<=0.98)
    pts = pts[mask]
    while len(pts) < N_CIRCLES:
        pts = np.vstack([pts, np.random.uniform(0.1, 0.9, 2)])
    pts = pts[:N_CIRCLES] + np.random.uniform(-0.005, 0.005, pts.shape)
    return pts

def make_init_force(seed):
    """Force-directed layout pushing points to boundaries."""
    np.random.seed(seed)
    pts = np.random.uniform(0.2, 0.8, (N_CIRCLES, 2))
    for step in range(800):
        f = np.zeros_like(pts)
        for i in range(N_CIRCLES):
            for j in range(i+1, N_CIRCLES):
                dx = pts[j] - pts[i]
                d = np.linalg.norm(dx)
                if d < 0.25 and d > 1e-6:
                    val = 0.03 / (d**2)
                    f[i] -= val * dx
                    f[j] += val * dx
            for dim in range(2):
                if pts[i, dim] < 0.15: f[i, dim] += 0.05
                elif pts[i, dim] > 0.85: f[i, dim] -= 0.05
        lr = 0.05 * (1.0 - step/800.0)
        pts += f * lr
        pts = np.clip(pts, 0.05, 0.95)
    return pts
