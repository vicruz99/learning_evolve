import numpy as np
from scipy.optimize import differential_evolution, linprog, minimize
import warnings
warnings.filterwarnings('ignore')

N_CIRCLES = 26

def constraint_func(vars):
    """Vectorized inequality constraints: g(vars) >= 0."""
    n = N_CIRCLES
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    # Boundary constraints
    b = np.concatenate([
        x - r,
        1.0 - x - r,
        y - r,
        1.0 - y - r
    ])
    
    # Overlap constraints: dist^2 - (r1+r2)^2 >= 0
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dr = r[:, None] + r[None, :]
    
    i, j = np.tril_indices(n, -1)
    o = dx[i, j]**2 + dy[i, j]**2 - dr[i, j]**2
    
    return np.concatenate([b, o])

def objective_slsqp(vars):
    """Objective to minimize: negative sum of radii."""
    return -np.sum(vars[2::3])

def get_lp_radii(centers):
    """Solve for optimal radii given fixed centers using Linear Programming."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    # Boundary limits: r_i <= dist_to_wall
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], 
                  centers[i, 1], 1.0 - centers[i, 1])
        b_ub[idx] = max(0.0, lim)
        A_ub[idx, i] = 1.0
        idx += 1
        
    # Overlap limits: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], 
                         centers[i, 1] - centers[j, 1])
            b_ub[idx] = d
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return np.full(n, 0.04)

def de_objective(x):
    """Objective for Differential Evolution: minimize negative sum of LP-optimal radii."""
    centers = x.reshape(N_CIRCLES, 2)
    centers = np.clip(centers, 1e-6, 1.0 - 1e-6)
    radii = get_lp_radii(centers)
    return -np.sum(radii)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    bounds_de = [(0.0, 1.0)] * (2 * N_CIRCLES)
    
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Phase 1: Global search over centers using Differential Evolution
    # LP guarantees optimal radii for any given center configuration
    for seed in range(4):
        try:
            res = differential_evolution(
                de_objective, bounds_de, maxiter=1000, popsize=10, 
                tol=1e-9, atol=1e-9, seed=seed, polish=False
            )
            if -res.fun > best_sum:
                best_sum = -res.fun
                best_centers = res.x.reshape(N_CIRCLES, 2).copy()
                best_radii = get_lp_radii(best_centers)
        except Exception:
            continue
            
    # Phase 2: Local refinement using SLSQP on full joint variables
    if best_centers is not None:
        x0 = np.zeros(3 * N_CIRCLES)
        x0[0::3] = best_centers[:, 0]
        x0[1::3] = best_centers[:, 1]
        x0[2::3] = best_radii
        
        bounds_opt = [(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)] * N_CIRCLES
        cons = {'type': 'ineq', 'fun': constraint_func}
        
        # Multiple perturbed restarts to escape local minima
        for pert_seed in range(5):
            np.random.seed(pert_seed)
            x_start = x0.copy()
            x_start += np.random.normal(0, 1e-4, x_start.shape)
            
            # Project to ensure strict initial feasibility for SLSQP
            for i in range(N_CIRCLES):
                r = max(1e-7, x_start[3 * i + 2])
                x_start[3 * i] = np.clip(x_start[3 * i], r, 1.0 - r)
                x_start[3 * i + 1] = np.clip(x_start[3 * i + 1], r, 1.0 - r)
                x_start[3 * i + 2] = r
                
            try:
                res = minimize(
                    objective_slsqp, x_start, method='SLSQP',
                    bounds=bounds_opt, constraints=cons,
                    options={'maxiter': 30000, 'ftol': 1e-14, 'disp': False}
                )
                if not np.isnan(res.fun) and -res.fun > best_sum:
                    best_sum = -res.fun
                    best_centers = np.column_stack((res.x[0::3], res.x[1::3]))
                    best_radii = res.x[2::3]
            except Exception:
                continue
                
        # Final LP snap to ensure radii are theoretically maximal for found centers
        lp_r = get_lp_radii(best_centers)
        if np.sum(lp_r) > best_sum:
            best_radii = lp_r
            best_sum = np.sum(lp_r)

    # Fallback initialization (hexagonal lattice)
    if best_centers is None:
        best_centers = np.zeros((N_CIRCLES, 2))
        idx = 0
        y = 0.09
        row = 0
        while idx < N_CIRCLES and y + 0.09 <= 1.0:
            x = 0.09 if row % 2 == 0 else 0.18
            while idx < N_CIRCLES and x + 0.09 <= 1.0:
                best_centers[idx] = [x, y]
                idx += 1
                x += 0.18
            y += np.sqrt(3.0) * 0.09
            row += 1
        best_radii = get_lp_radii(best_centers)
        best_sum = np.sum(best_radii)

    # Phase 3: Strict validation and minimal numerical repair

            if best_radii[i] < 1e-8:
                valid = False
                break
            if best_centers[i, 0] < best_radii[i] - 1e-9 or best_centers[i, 0] > 1 - best_radii[i] + 1e-9:
                valid = False
                break
            if best_centers[i, 1] < best_radii[i] - 1e-9 or best_centers[i, 1] > 1 - best_radii[i] + 1e-9:
                valid = False
                break
        if valid:
            for i in range(N_CIRCLES):
                for j in range(i + 1, N_CIRCLES):
                    d = np.hypot(best_centers[i, 0] - best_centers[j, 0], 
                                 best_centers[i, 1] - best_centers[j, 1])
                    if d < best_radii[i] + best_radii[j] - 1e-9:
                        valid = False
                        break
                if not valid:
                    break
        if valid:
            break
        best_radii *= 0.999
        
    return best_centers, best_radii, float(np.sum(best_radii))
