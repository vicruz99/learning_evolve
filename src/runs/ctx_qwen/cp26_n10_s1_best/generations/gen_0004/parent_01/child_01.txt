import numpy as np
from scipy.optimize import minimize, linprog
import math

N_CIRCLES = 26

def objective(vars):
    """Minimize negative sum of radii."""
    return -np.sum(vars[2::3])

def constraints(vars):
    """Vectorized inequality constraints: g(vars) >= 0."""
    n = N_CIRCLES
    cx = vars[0::3]
    cy = vars[1::3]
    cr = vars[2::3]
    
    # Boundary constraints
    b = np.concatenate([
        cx - cr,
        1.0 - cx - cr,
        cy - cr,
        1.0 - cy - cr
    ])
    
    # Overlap constraints: dist^2 - (r1+r2)^2 >= 0
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dr = cr[:, None] + cr[None, :]
    
    i, j = np.tril_indices(n, -1)
    o = dx[i, j]**2 + dy[i, j]**2 - dr[i, j]**2
    
    return np.concatenate([b, o])

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-7, 0.5]."""
    b = []
    for _ in range(N_CIRCLES):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)])
    return b

def solve_lp_radii(centers):
    """Solve for optimal radii given fixed centers using Linear Programming."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        b_ub[i] = max(0.0, lim)
        A_ub[i, i] = 1.0
        idx += 1
        
    for i in range(n):
        for j in range(i + 1, n):
            dist = math.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist)
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def push_centers_force(centers, radii, iterations=100):
    """Force-directed relaxation to resolve overlaps and spread circles."""
    pts = centers.copy()
    rs = radii.copy()
    dt = 0.5
    
    for _ in range(iterations):
        f = np.zeros_like(pts)
        for i in range(N_CIRCLES):
            for j in range(i + 1, N_CIRCLES):
                dx = pts[j, 0] - pts[i, 0]
                dy = pts[j, 1] - pts[i, 1]
                d = math.hypot(dx, dy)
                if d < rs[i] + rs[j] and d > 1e-8:
                    overlap = rs[i] + rs[j] - d
                    force = overlap * 5.0  # Strong repulsion for overlaps
                    fx = force * dx / d
                    fy = force * dy / d
                    f[i] -= [fx, fy]
                    f[j] += [fx, fy]
                    
            # Wall repulsion
            for dim in range(2):
                if pts[i, dim] < rs[i] + 1e-6:
                    f[i, dim] += 10.0 * (rs[i] + 1e-6 - pts[i, dim])
                if pts[i, dim] > 1.0 - rs[i] - 1e-6:
                    f[i, dim] -= 10.0 * (pts[i, dim] - (1.0 - rs[i] - 1e-6))
                    
        pts += f * dt
        pts = np.clip(pts, 1e-7, 1.0 - 1e-7)
        
        # Gradually reduce step size for stability
        if _ % 20 == 0 and _ > 0:
            dt *= 0.95
            if dt < 0.01: dt = 0.01
            
    return pts

def generate_diverse_inits():
    """Generate a wide variety of initial configurations."""
    inits = []
    
    # 1. Hexagonal lattices with varied parameters
    for r0 in [0.075, 0.085, 0.095, 0.105]:
        for angle in [0, 10, 20, 30, 45, 60]:
            pts = []
            y = r0
            row = 0
            while len(pts) < N_CIRCLES + 10:
                x = r0 if row % 2 == 0 else 2.0 * r0
                while x <= 1.0 - r0:
                    pts.append([x, y])
                    x += 2.0 * r0
                y += math.sqrt(3.0) * r0
                row += 1
                if y > 1.0 + r0: break
                
            pts = np.array(pts[:N_CIRCLES + 10])
            if angle != 0:
                rad = math.radians(angle)
                c, s = math.cos(rad), math.sin(rad)
                pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
                
            mask = (pts[:, 0] >= 0.01) & (pts[:, 0] <= 0.99) & (pts[:, 1] >= 0.01) & (pts[:, 1] <= 0.99)
            pts = pts[mask]
            while len(pts) < N_CIRCLES:
                pts = np.vstack([pts, np.random.uniform(0.2, 0.8, 2)])
            pts = np.array(pts[:N_CIRCLES]) + np.random.uniform(-0.003, 0.003, (N_CIRCLES, 2))
            inits.append(pts)
            
    # 2. Staggered grids
    for _ in range(5):
        pts = []
        for r in range(7):
            for c in range(5):
                if len(pts) >= N_CIRCLES: break
                x = 0.08 + c * 0.18
                y = 0.08 + r * 0.15
                x += (r % 2) * 0.09
                pts.append([x, y])
        pts = np.array(pts[:N_CIRCLES]) + np.random.uniform(-0.015, 0.015, (N_CIRCLES, 2))
        inits.append(pts)
        
    # 3. Random starts
    for _ in range(5):
        pts = np.random.uniform(0.1, 0.9, (N_CIRCLES, 2))
        inits.append(pts)
        
    return inits

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    n = N_CIRCLES
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -1.0
    best_c = None
    best_r = None
    
    inits = generate_diverse_inits()
    
    # Phase 1: Multi-start with LP + Force + SLSQP
    for pts in inits:
        c_cur = pts.copy()
        r_cur = np.full(n, 0.06)
        
        # Alternating LP and Force to find a tight configuration
        for _ in range(4):
            r_lp = solve_lp_radii(c_cur)
            if r_lp is not None:
                r_cur = r_lp
            c_cur = push_centers_force(c_cur, r_cur, 60)
            
        x0 = np.zeros(3 * n)
        x0[0::3] = c_cur[:, 0]
        x0[1::3] = c_cur[:, 1]
        x0[2::3] = r_cur
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 8000, 'ftol': 1e-14, 'disp': False})
            
            if not np.isnan(res.fun):
                c_opt = np.column_stack((res.x[0::3], res.x[1::3]))
                r_opt = res.x[2::3]
                
                # Snap radii to theoretical max for these centers
                r_snap = solve_lp_radii(c_opt)
                if r_snap is not None:
                    r_opt = r_snap
                    curr_sum = np.sum(r_opt)
                else:
                    curr_sum = -res.fun
                    
                # Strict feasibility check
                valid = True
                for i in range(n):
                    if r_opt[i] < 1e-8 or c_opt[i, 0] < r_opt[i] - 1e-9 or c_opt[i, 0] > 1 - r_opt[i] + 1e-9 or \
                       c_opt[i, 1] < r_opt[i] - 1e-9 or c_opt[i, 1] > 1 - r_opt[i] + 1e-9:
                        valid = False
                        break
                if valid:
                    for i in range(n):
                        for j in range(i + 1, n):
                            d = math.hypot(c_opt[i, 0] - c_opt[j, 0], c_opt[i, 1] - c_opt[j, 1])
                            if d < r_opt[i] + r_opt[j] - 1e-9:
                                valid = False
                                break
                        if not valid: break
                        
                if valid and curr_sum > best_sum:
                    best_sum = curr_sum
                    best_c = c_opt.copy()
                    best_r = r_opt.copy()
        except Exception:
            continue
            
    # Phase 2: Basin hopping / Refinement to escape local minima
    if best_c is not None:
        for step in range(20):
            noise = 0.003 * (0.85 ** step)
            p_c = best_c + np.random.normal(0, noise, best_c.shape)
            p_c = np.clip(p_c, 0.01, 0.99)
            p_r = best_r * 0.985
            
            x0 = np.zeros(3 * n)
            x0[0::3] = p_c[:, 0]
            x0[1::3] = p_c[:, 1]
            x0[2::3] = p_r
            
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 5000, 'ftol': 1e-14, 'disp': False})
                if not np.isnan(res.fun):
                    c_o = np.column_stack((res.x[0::3], res.x[1::3]))
                    r_o = res.x[2::3]
                    
                    r_s = solve_lp_radii(c_o)
                    if r_s is not None:
                        r_o = r_s
                        curr = np.sum(r_o)
                    else:
                        curr = -res.fun
                        
                    if curr > best_sum:
                        # Quick validation
                        valid = True
                        for i in range(n):
                            if r_o[i]<1e-8 or c_o[i,0]<r_o[i]-1e-9 or c_o[i,0]>1-r_o[i]+1e-9 or \
                               c_o[i,1]<r_o[i]-1e-9 or c_o[i,1]>1-r_o[i]+1e-9:
                                valid=False; break
                        if valid:
                            for i in range(n):
                                for j in range(i+1,n):
                                    d = math.hypot(c_o[i,0]-c_o[j,0], c_o[i,1]-c_o[j,1])
                                    if d < r_o[i]+r_o[j]-1e-9:
                                        valid=False; break
                                if not valid: break
                        if valid:
                            best_sum = curr
                            best_c = c_o
                            best_r = r_o
            except Exception:
                pass
                
    # Fallback
    if best_c is None:
        best_c = inits[0]
        best_r = np.full(n, 0.08)
        best_sum = np.sum(best_r)
        
    # Final strict safety shrink to guarantee compliance with 1e-12 tolerance
    for _ in range(100):
        valid = True
        for i in range(n):
            if best_r[i] < 0: valid = False; break
            if best_c[i, 0] - best_r[i] < -1e-12 or best_c[i, 0] + best_r[i] > 1 + 1e-12: valid = False; break
            if best_c[i, 1] - best_r[i] < -1e-12 or best_c[i, 1] + best_r[i] > 1 + 1e-12: valid = False; break
        if valid:
            for i in range(n):
                for j in range(i + 1, n):
                    d = math.hypot(best_c[i, 0] - best_c[j, 0], best_c[i, 1] - best_c[j, 1])
                    if d < best_r[i] + best_r[j] - 1e-12:
                        valid = False
                        break
                if not valid: break
        if valid: break
        best_r *= 0.9999
        best_c[:, 0] = np.clip(best_c[:, 0], best_r, 1.0 - best_r)
        best_c[:, 1] = np.clip(best_c[:, 1], best_r, 1.0 - best_r)
        
    return best_c, best_r, float(best_sum)
