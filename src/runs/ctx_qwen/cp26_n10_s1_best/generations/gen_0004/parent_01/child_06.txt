import numpy as np
from scipy.optimize import minimize, linprog

N_CIRCLES = 26
TRIL_IDX = np.tril_indices(N_CIRCLES, -1)

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-6, 0.5]."""
    b = []
    for _ in range(N_CIRCLES):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)])
    return b

def constraints(x):
    """Vectorized inequality constraints: g(x) >= 0."""
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    # Boundary constraints
    c = np.concatenate([
        cx - r,
        1.0 - cx - r,
        cy - r,
        1.0 - cy - r
    ])
    
    # Overlap constraints: dist^2 - (r_i + r_j)^2 >= 0
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dr = r[:, None] + r[None, :]
    
    c = np.concatenate([c, dx[TRIL_IDX]**2 + dy[TRIL_IDX]**2 - dr[TRIL_IDX]**2])
    return c

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def solve_lp_radii(centers):
    """
    Optimally compute radii for fixed centers using Linear Programming.
    Maximizes sum(r_i) subject to r_i + r_j <= dist(i,j) and boundary limits.
    """
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], 
                  centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], 
                            centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist)
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-7)
    except Exception:
        pass
    return np.full(n, 0.05)

def generate_hex_init(angle_deg):
    """Generates a rotated hexagonal lattice initialization."""
    rad = np.deg2rad(angle_deg)
    pts = []
    y = 0.08
    row = 0
    while len(pts) < 30:
        x = 0.08 if row % 2 == 0 else 0.16
        while x <= 0.92:
            pts.append([x, y])
            x += 0.16
        y += 0.08 * np.sqrt(3.0)
        row += 1
        
    pts = np.array(pts[:N_CIRCLES])
    c, s = np.cos(rad), np.sin(rad)
    center = np.array([0.5, 0.5])
    pts = (pts - center) @ np.array([[c, -s], [s, c]]) + center
    return pts

def generate_force_init(seed):
    """Force-directed layout to spread points evenly and utilize boundaries."""
    np.random.seed(seed)
    pts = np.random.uniform(0.15, 0.85, (N_CIRCLES, 2))
    for _ in range(300):
        forces = np.zeros_like(pts)
        for i in range(N_CIRCLES):
            for j in range(i + 1, N_CIRCLES):
                dx = pts[j, 0] - pts[i, 0]
                dy = pts[j, 1] - pts[i, 1]
                d = np.hypot(dx, dy)
                if d < 0.28 and d > 1e-6:
                    f = 0.012 / (d**2 + 0.001)
                    forces[i] -= f * np.array([dx, dy])
                    forces[j] += f * np.array([dx, dy])
            # Boundary repulsion
            for dim in range(2):
                if pts[i, dim] < 0.10: forces[i, dim] += 0.04
                elif pts[i, dim] > 0.90: forces[i, dim] -= 0.04
                
        pts += forces * 0.06
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    n = N_CIRCLES
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Phase 1: Diverse Initializations
    inits = []
    # Rotated hexagonal lattices
    for ang in range(-25, 26, 5):
        inits.append(generate_hex_init(ang))
    # Force-directed layouts
    for s in range(8):
        inits.append(generate_force_init(s))
    # Perturbed grids
    for _ in range(4):
        pts = []
        for r in range(6):
            for c in range(5):
                if len(pts) >= n: break
                pts.append([0.09 + c * 0.18, 0.09 + r * 0.16])
        pts = np.array(pts[:n]) + np.random.uniform(-0.015, 0.015, (n, 2))
        inits.append(pts)
        
    # Phase 2: Multi-start Optimization with LP Refinement
    for pts in inits:
        r0 = solve_lp_radii(pts) * 0.999  # Slight shrinkage for strict feasibility
        x0 = np.zeros(3 * n)
        x0[0::3] = pts[:, 0]
        x0[1::3] = pts[:, 1]
        x0[2::3] = r0
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 15000, 'ftol': 1e-14, 'disp': False})
            
            if not np.isnan(res.fun):
                c_opt = np.column_stack((res.x[0::3], res.x[1::3]))
                r_opt = res.x[2::3]
                
                # LP refinement: snap radii to theoretical max for these centers
                r_lp = solve_lp_radii(c_opt) * 0.9999
                curr_sum = np.sum(r_lp)
                
                if curr_sum > best_sum:
                    best_sum = curr_sum
                    best_centers = c_opt.copy()
                    best_radii = r_lp.copy()
        except Exception:
            continue
            
    # Phase 3: Deflation & Perturbation Refinement
    if best_centers is not None:
        for step in range(25):
            noise_scale = 0.0025 * (0.92 ** step)
            pert_c = best_centers + np.random.normal(0, noise_scale, best_centers.shape)
            pert_c = np.clip(pert_c, 0.02, 0.98)
            
            # Deflate radii to free space for repositioning
            pert_r = best_radii * 0.985
            
            x0 = np.zeros(3 * n)
            x0[0::3] = pert_c[:, 0]
            x0[1::3] = pert_c[:, 1]
            x0[2::3] = pert_r
            
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 10000, 'ftol': 1e-14, 'disp': False})
                
                if not np.isnan(res.fun):
                    c_opt = np.column_stack((res.x[0::3], res.x[1::3]))
                    r_lp = solve_lp_radii(c_opt) * 0.9999
                    curr_sum = np.sum(r_lp)
                    
                    if curr_sum > best_sum:
                        best_sum = curr_sum
                        best_centers = c_opt.copy()
                        best_radii = r_lp.copy()
            except Exception:
                pass

    # Fallback (should not be reached)
    if best_centers is None:
        best_centers = generate_hex_init(0)
        best_radii = solve_lp_radii(best_centers) * 0.99
        best_sum = np.sum(best_radii)
        
    # Phase 4: Strict Validation & Minimal Numerical Repair
    for _ in range(50):
        valid = True
        for i in range(n):
            if best_radii[i] < 0:
                valid = False; break
            if best_centers[i,0] - best_radii[i] < -1e-12 or best_centers[i,0] + best_radii[i] > 1 + 1e-12:
                valid = False; break
            if best_centers[i,1] - best_radii[i] < -1e-12 or best_centers[i,1] + best_radii[i] > 1 + 1e-12:
                valid = False; break
                
        if valid:
            for i in range(n):
                for j in range(i + 1, n):
                    d = np.hypot(best_centers[i,0] - best_centers[j,0], 
                                 best_centers[i,1] - best_centers[j,1])
                    if d < best_radii[i] + best_radii[j] - 1e-12:
                        valid = False; break
                if not valid: break
                
        if valid: break
        
        # Minimal shrinkage to recover validity
        best_radii *= 0.9998
        for i in range(n):
            best_centers[i,0] = np.clip(best_centers[i,0], best_radii[i], 1.0 - best_radii[i])
            best_centers[i,1] = np.clip(best_centers[i,1], best_radii[i], 1.0 - best_radii[i])
            
    return best_centers, best_radii, float(np.sum(best_radii))
