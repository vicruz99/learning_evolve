import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Vectorized inequality constraints: g(x) >= 0."""
    cx = x[0::3]
    cy = x[1::3]
    cr = x[2::3]
    
    # Boundary constraints
    c = np.concatenate([
        cx - cr,
        1.0 - cx - cr,
        cy - cr,
        1.0 - cy - cr
    ])
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dr = cr[:, None] + cr[None, :]
    
    i, j = np.tril_indices(N, -1)
    c = np.concatenate([c, dx[i, j]**2 + dy[i, j]**2 - dr[i, j]**2])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [0, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
    return b

def solve_lp_radii(centers):
    """Solve for optimal radii given fixed centers using Linear Programming."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist)
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def force_layout(centers, radii, steps=400):
    """Force-directed relaxation to spread points and push to boundaries."""
    pts = centers.copy()
    r = radii.copy()
    for step in range(steps):
        forces = np.zeros_like(pts)
        lr = 0.04 * (1.0 - step / steps)
        for i in range(N):
            for j in range(i + 1, N):
                dx = pts[j, 0] - pts[i, 0]
                dy = pts[j, 1] - pts[i, 1]
                d = np.hypot(dx, dy)
                if d < r[i] + r[j] and d > 1e-6:
                    rep = (r[i] + r[j] - d) * 0.5 / d
                    fx, fy = rep * dx, rep * dy
                    forces[i] -= [fx, fy]
                    forces[j] += [fx, fy]
            # Boundary repulsion
            for dim in range(2):
                if pts[i, dim] < r[i] + 0.03:
                    forces[i, dim] += 0.06
                elif pts[i, dim] > 1.0 - r[i] - 0.03:
                    forces[i, dim] -= 0.06
        pts += forces * lr
        pts = np.clip(pts, 0.001, 0.999)
    return pts

def init_hex(r0, angle=0.0):
    """Generate hexagonal lattice points, optionally rotated."""
    np.random.seed(0)
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 15:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        if y > 1.0 + r0:
            break
    pts = np.array(pts[:N+15])
    if angle != 0:
        rad = np.deg2rad(angle)
        c, s = np.cos(rad), np.sin(rad)
        center = [0.5, 0.5]
        pts = (pts - center) @ np.array([[c, -s], [s, c]]) + center
    mask = (pts[:,0]>=0.01) & (pts[:,0]<=0.99) & (pts[:,1]>=0.01) & (pts[:,1]<=0.99)
    pts = pts[mask]
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.1, 0.9, 2)])
    return pts[:N]

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -1.0
    best_x = None
    
    # Phase 1: Diverse Initializations
    starts = []
    for r0 in [0.085, 0.090, 0.095, 0.100]:
        for ang in [0, 15, 30, 45, 60]:
            pts = init_hex(r0, ang)
            pts += np.random.uniform(-0.005, 0.005, pts.shape)
            starts.append(pts.copy())
            
    for _ in range(6):
        pts = np.random.uniform(0.1, 0.9, (N, 2))
        starts.append(pts.copy())
        
    best_centers = None
    best_radii = None
    
    for pts in starts:
        radii = np.full(N, 0.05)
        # Pre-optimization: alternating LP radii and force layout
        for _ in range(4):
            lp_r = solve_lp_radii(pts)
            if lp_r is not None:
                radii = lp_r
            pts = force_layout(pts, radii, steps=250)
            
        x0 = np.zeros(3 * N)
        x0[0::3] = pts[:, 0]
        x0[1::3] = pts[:, 1]
        x0[2::3] = radii
        
        # Project to strict bounds
        for i in range(N):
            r = max(1e-5, x0[3*i+2])
            x0[3*i] = np.clip(x0[3*i], r, 1.0 - r)
            x0[3*i+1] = np.clip(x0[3*i+1], r, 1.0 - r)
            x0[3*i+2] = r
            
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 20000, 'ftol': 1e-14, 'disp': False})
            if not np.isnan(res.fun) and -res.fun > best_sum:
                best_sum = -res.fun
                best_x = res.x.copy()
        except Exception:
            pass
            
    if best_x is None:
        best_x = np.zeros(3*N)
        best_x[0::3] = np.tile(np.linspace(0.1, 0.9, 5), 6)[:N]
        best_x[1::3] = np.repeat(np.linspace(0.1, 0.9, 6), 5)[:N]
        best_x[2::3] = 0.05
        best_sum = -objective(best_x)

    # Phase 2: Intensive Deflation & Refinement
    best_centers = best_x.reshape(N, 3)[:, :2]
    best_radii = best_x[2::3]
    
    for rnd in range(25):

        pert_r = best_radii * 0.96
        # Perturb centers

        pert_c = np.clip(pert_c, 0.01, 0.99)
        
        x0 = np.zeros(3 * N)
        x0[0::3] = pert_c[:, 0]
        x0[1::3] = pert_c[:, 1]
        x0[2::3] = pert_r

            x0[3*i+2] = r
            
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 15000, 'ftol': 1e-14, 'disp': False})
            if not np.isnan(res.fun):
                c_opt = np.column_stack((res.x[0::3], res.x[1::3]))
                r_opt = res.x[2::3]
                
                # LP polish to maximize radii for the new centers
                lp_r = solve_lp_radii(c_opt)
                if lp_r is not None:
                    r_opt = lp_r
                    curr_sum = np.sum(r_opt)
                else:
                    curr_sum = -res.fun
                    
                if curr_sum > best_sum:
                    # Strict feasibility validation
                    valid = True
                    for i in range(N):
                        if r_opt[i] < 1e-8 or c_opt[i,0] < r_opt[i] - 1e-9 or c_opt[i,0] > 1 - r_opt[i] + 1e-9 or \
                           c_opt[i,1] < r_opt[i] - 1e-9 or c_opt[i,1] > 1 - r_opt[i] + 1e-9:

                            for j in range(i+1, N):
                                d = np.hypot(c_opt[i,0]-c_opt[j,0], c_opt[i,1]-c_opt[j,1])
                                if d < r_opt[i] + r_opt[j] - 1e-9:

            best_centers[i,1] = np.clip(best_centers[i,1], best_radii[i], 1.0 - best_radii[i])
            
    return best_centers, best_radii, float(np.sum(best_radii))
