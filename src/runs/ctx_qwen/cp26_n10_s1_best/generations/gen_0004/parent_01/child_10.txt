import numpy as np
from scipy.optimize import minimize, linprog

N_CIRCLES = 26

def solve_lp_radii(centers):
    """Compute optimal radii for fixed centers using Linear Programming."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist)
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def max_slack_centers(centers, radii, maxiter=3000):
    """Optimize centers to maximize minimum clearance (slack) for fixed radii."""
    n = N_CIRCLES
    v0 = np.zeros(2*n + 1)
    v0[:2*n:2] = centers[:, 0]
    v0[1:2*n:2] = centers[:, 1]
    v0[-1] = 0.0
    
    bounds = [(0.0, 1.0)]*(2*n) + [(-0.1, 0.3)]
    
    def obj(v):
        return -v[-1]
        
    def constr(v):
        x = v[:2*n:2]
        y = v[1:2*n:2]
        eps = v[-1]
        c = np.empty(4*n + n*(n-1)//2)
        idx = 0
        for i in range(n):
            c[idx] = x[i] - radii[i] - eps; idx += 1
            c[idx] = 1.0 - x[i] - radii[i] - eps; idx += 1
            c[idx] = y[i] - radii[i] - eps; idx += 1
            c[idx] = 1.0 - y[i] - radii[i] - eps; idx += 1
        for i in range(n):
            for j in range(i+1, n):
                dx = x[i] - x[j]
                dy = y[i] - y[j]
                c[idx] = dx*dx + dy*dy - (radii[i] + radii[j] + eps)**2
                idx += 1
        return c

    try:
        res = minimize(obj, v0, method='SLSQP', bounds=bounds,
                       constraints={'type': 'ineq', 'fun': constr},
                       options={'maxiter': maxiter, 'ftol': 1e-12, 'disp': False})
        if res.success and not np.isnan(res.fun):
            return np.column_stack((res.x[:2*n:2], res.x[1:2*n:2])), res.x[-1]
    except Exception:
        pass
    return centers, 0.0

def joint_optimize(centers, radii, maxiter=8000):
    """Jointly optimize centers and radii using SLSQP."""
    n = N_CIRCLES
    v0 = np.zeros(3*n)
    v0[0::3] = centers[:, 0]
    v0[1::3] = centers[:, 1]
    v0[2::3] = radii
    
    bounds = [(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)] * n
    
    def obj(v):
        return -np.sum(v[2::3])
        
    def constr(v):
        x = v[0::3]
        y = v[1::3]
        r = v[2::3]
        c = np.empty(4*n + n*(n-1)//2)
        idx = 0
        c[idx:idx+n] = x - r; idx += n
        c[idx:idx+n] = 1.0 - x - r; idx += n
        c[idx:idx+n] = y - r; idx += n
        c[idx:idx+n] = 1.0 - y - r; idx += n
        
        dx = x[:, None] - x[None, :]
        dy = y[:, None] - y[None, :]
        dr = r[:, None] + r[None, :]
        i_idx, j_idx = np.tril_indices(n, -1)
        c[idx:] = dx[i_idx, j_idx]**2 + dy[i_idx, j_idx]**2 - dr[i_idx, j_idx]**2
        return c

    try:
        res = minimize(obj, v0, method='SLSQP', bounds=bounds,
                       constraints={'type': 'ineq', 'fun': constr},
                       options={'maxiter': maxiter, 'ftol': 1e-13, 'disp': False})
        if not np.isnan(res.fun):
            return np.column_stack((res.x[0::3], res.x[1::3])), res.x[2::3]
    except Exception:
        pass
    return centers, radii

def generate_inits():

    inits = []

    # Hexagonal lattices with rotations
    for r0 in [0.085, 0.095, 0.105]:

        pts = np.array(pts[:n])
        
        inits.append(pts.copy())

            mask = (p_rot[:, 0] > 0.02) & (p_rot[:, 0] < 0.98) & (p_rot[:, 1] > 0.02) & (p_rot[:, 1] < 0.98)
            p_filt = p_rot[mask]
