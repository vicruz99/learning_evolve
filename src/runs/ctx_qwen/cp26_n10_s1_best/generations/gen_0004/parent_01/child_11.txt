N_CIRCLES = 26

    return -np.sum(x[2::3])

    c = np.concatenate([
        cx - r,
        1.0 - cx - r,
        cy - r,
        1.0 - cy - r
    ])

    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dr = r[:, None] + r[None, :]

    c = np.concatenate([c, dx[i_idx, j_idx]**2 + dy[i_idx, j_idx]**2 - dr[i_idx, j_idx]**2])

        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1

        pass
    return None

    dy = centers[:, 1] - cy
