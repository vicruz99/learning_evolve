import numpy as np
from scipy.optimize import minimize, linprog

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Returns all inequality constraints g(x) >= 0."""
    cx, cy, r = x[0::3], x[1::3], x[2::3]
    c = np.concatenate([
        cx - r, 1.0 - cx - r,
        cy - r, 1.0 - cy - r
    ])
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dr = r[:, None] + r[None, :]
    c = np.concatenate([c, dx[TRIL_IDX]**2 + dy[TRIL_IDX]**2 - dr[TRIL_IDX]**2])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-7, 0.5]."""
    return [(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)] * N

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    A_ub = np.zeros((n + n*(n-1)//2, n))
    b_ub = np.zeros(n + n*(n-1)//2)
    idx = 0
    for i in range(n):
        lim = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
    for i in range(n):
        for j in range(i+1, n):
            d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = d
            idx += 1
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=[(0.0, None)]*n, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def hex_init(rows, r0, angle=0.0):
    """Generates a hexagonal lattice initialization with optional rotation."""
    pts = []
    y = r0
    for i, cnt in enumerate(rows):
        x_start = r0 + (r0 if i%2==1 else 0.0)
        for k in range(cnt):
            pts.append([x_start + k*2.0*r0, y])
        y += np.sqrt(3.0)*r0
    pts = np.array(pts[:N])
    while len(pts) < N:
        pts = np.vstack([pts, [np.random.uniform(0.2, 0.8), np.random.uniform(0.2, 0.8)]])
    if angle != 0.0:
        c = np.array([0.5, 0.5])
        ca, sa = np.cos(angle), np.sin(angle)
        pts = (pts - c) @ np.array([[ca, -sa], [sa, ca]]) + c
    return pts

def force_init(seed):
    """Force-directed layout to spread points evenly."""
    np.random.seed(seed)
    pts = np.random.uniform(0.1, 0.9, (N, 2))
    for _ in range(400):
        f = np.zeros_like(pts)
        for i in range(N):
            for j in range(i+1, N):
                dx = pts[j]-pts[i]
                d = np.linalg.norm(dx)
                if d < 0.3 and d > 1e-4:
                    ff = 0.012/(d**2+0.001)
                    f[i] -= ff*dx
                    f[j] += ff*dx
            for dim in range(2):
                if pts[i,dim] < 0.12: f[i,dim] += 0.04
                elif pts[i,dim] > 0.88: f[i,dim] -= 0.04
        pts += f*0.05
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    best_sum = -1.0
    best_x = None

    inits = []
    patterns = [[5,6,5,6,4], [6,5,6,5,4], [5,6,6,5,4], [4,6,6,6,4], [5,5,5,5,6], [6,6,6,4,4], [5,7,5,6,3], [4,5,6,5,6]]
    for pat in patterns:
        for ang in np.linspace(-0.4, 0.4, 13):
            inits.append(hex_init(pat, 0.092, ang))

        inits.append(force_init(s))

        x0 = np.zeros(3*N)
        x0[0::3] = c0[:,0]
        x0[1::3] = c0[:,1]
        x0[2::3] = r0

            if not np.isnan(res.fun):
                c_vals = constraints(res.x)
                if np.min(c_vals) >= -1e-6 and -res.fun > best_sum:
                    best_sum = -res.fun
                    best_x = res.x.copy()
        except Exception:
            continue

    if best_x is None:

                        valid = False; break
                if not valid: break
        if valid: break
        radii *= 0.9999
        centers[:,0] = np.clip(centers[:,0], radii, 1.0-radii)
        centers[:,1] = np.clip(centers[:,1], radii, 1.0-radii)

    return centers, radii, float(np.sum(radii))
