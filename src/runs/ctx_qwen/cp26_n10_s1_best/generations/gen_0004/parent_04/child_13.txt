import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Returns all inequality constraints g(x) >= 0."""
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    # Boundary constraints
    c = np.concatenate([
        cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r
    ])
    
    # Overlap constraints using squared distances
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dr = r[:, None] + r[None, :]
    
    idx = np.tril_indices(N, -1)
    c = np.concatenate([c, dx[idx]**2 + dy[idx]**2 - dr[idx]**2])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-7, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)])
    return b

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A = np.zeros((num_ineq, n))
    b = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A[idx, i] = 1.0
        b[idx] = max(1e-10, lim)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = max(1e-10, dist)
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-7)
    except Exception:
        pass
    return np.full(n, 0.01)

def force_spread(centers, radii, steps=100):
    """Force-directed relaxation to spread centers apart and push to boundaries."""
    pts = centers.copy()
    rs = radii.copy()
    
    for step in range(steps):
        f = np.zeros_like(pts)
        lr = 0.04 * (1.0 - step / steps)
        
        for i in range(N):
            for j in range(i + 1, N):
                dx = pts[j, 0] - pts[i, 0]
                dy = pts[j, 1] - pts[i, 1]
                d = np.hypot(dx, dy)
                target = rs[i] + rs[j] + 0.005
                if d < target and d > 1e-6:
                    rep = (target - d) * 0.5 / d
                    fx, fy = rep * dx, rep * dy
                    f[i] -= [fx, fy]
                    f[j] += [fx, fy]
                    
            for dim in range(2):
                if pts[i, dim] < rs[i] + 0.005:
                    f[i, dim] += 0.03
                elif pts[i, dim] > 1.0 - rs[i] - 0.005:
                    f[i, dim] -= 0.03
                    
        pts += f * lr
        pts = np.clip(pts, 0.001, 0.999)
    return pts

def make_init_hex(r0, angle=0.0, seed=0):
    """Generates a hexagonal lattice initialization with rotation."""
    np.random.seed(seed)
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 5:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        if y > 1.0 + r0:
            break
            
    pts = np.array(pts[:N + 5])
    if angle != 0.0:
        ca, sa = np.cos(angle), np.sin(angle)
        dx = pts[:, 0] - 0.5
        dy = pts[:, 1] - 0.5
        pts[:, 0] = dx * ca - dy * sa + 0.5
        pts[:, 1] = dx * sa + dy * ca + 0.5
        
    pts = pts[:N]
    pts += np.random.uniform(-0.003, 0.003, pts.shape)
    return np.clip(pts, 0.05, 0.95)

def is_strictly_valid(centers, radii):
    """Check strict feasibility against validator tolerances."""
    for i in range(N):
        if radii[i] < 0:
            return False
        if centers[i, 0] - radii[i] < -1e-12 or centers[i, 0] + radii[i] > 1.0 + 1e-12:
            return False
        if centers[i, 1] - radii[i] < -1e-12 or centers[i, 1] + radii[i] > 1.0 + 1e-12:
            return False
    for i in range(N):
        for j in range(i + 1, N):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            if d < radii[i] + radii[j] - 1e-12:
                return False
    return True

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Phase 1: Diverse Initializations
    inits = []
    # Hexagonal lattices with varied densities and rotations
    for r0 in [0.082, 0.090, 0.102]:
        for ang in np.linspace(0.0, 1.4, 9):
            inits.append(make_init_hex(r0, ang, seed=int(ang * 1000)))
            
    # Random starts
    for _ in range(12):
        pts = np.random.uniform(0.15, 0.85, (N, 2))
        inits.append(pts)
        
    for c_init in inits:
        r_init = solve_lp_radii(c_init)
        # Pre-optimization loop: LP + Force
        for _ in range(4):
            r_init = solve_lp_radii(c_init)
            c_init = force_spread(c_init, r_init, steps=60)
            
        # Prepare SLSQP vector
        x0 = np.zeros(3 * N)
        x0[0::3] = c_init[:, 0]
        x0[1::3] = c_init[:, 1]
        x0[2::3] = r_init * 0.995  # Slight shrink ensures strict feasibility
        
        # Project centers to respect boundary constraints
        for i in range(N):
            r = x0[3 * i + 2]
            x0[3 * i] = np.clip(x0[3 * i], r, 1.0 - r)
            x0[3 * i + 1] = np.clip(x0[3 * i + 1], r, 1.0 - r)
            
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 15000, 'ftol': 1e-13, 'disp': False})
            
            if not np.isnan(res.fun):
                c_opt = np.column_stack((res.x[0::3], res.x[1::3]))
                r_opt = solve_lp_radii(c_opt)
                curr_sum = np.sum(r_opt)
                
                if curr_sum > best_sum:
                    best_sum = curr_sum
                    best_centers = c_opt.copy()
                    best_radii = r_opt.copy()
        except Exception:
            continue

    # Phase 2: Deflation & Perturbation Refinement to escape local minima
    if best_centers is not None:
        for step in range(12):
            noise = 0.004 / (step + 1)
            c_pert = best_centers + np.random.normal(0, noise, best_centers.shape)
            c_pert = np.clip(c_pert, 0.02, 0.98)
            r_pert = best_radii * 0.985
            
            # Quick refinement loop
            for _ in range(3):
                r_pert = solve_lp_radii(c_pert)
                c_pert = force_spread(c_pert, r_pert, steps=40)
                
            x0 = np.zeros(3 * N)
            x0[0::3] = c_pert[:, 0]
            x0[1::3] = c_pert[:, 1]
            x0[2::3] = r_pert * 0.995
            
            for i in range(N):
                r = x0[3 * i + 2]
                x0[3 * i] = np.clip(x0[3 * i], r, 1.0 - r)
                x0[3 * i + 1] = np.clip(x0[3 * i + 1], r, 1.0 - r)
                
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 10000, 'ftol': 1e-13, 'disp': False})
                
                if not np.isnan(res.fun):
                    c_opt = np.column_stack((res.x[0::3], res.x[1::3]))
                    r_opt = solve_lp_radii(c_opt)
                    s_opt = np.sum(r_opt)
                    
                    if s_opt > best_sum:
                        best_sum = s_opt
                        best_centers = c_opt.copy()
                        best_radii = r_opt.copy()
            except Exception:
                pass

    # Fallback (should not be reached)
    if best_centers is None:
        best_centers = make_init_hex(0.09)
        best_radii = solve_lp_radii(best_centers)
        best_sum = np.sum(best_radii)
        
    # Final safety correction to guarantee strict compliance with 1e-12 tolerance
    for _ in range(50):
        if is_strictly_valid(best_centers, best_radii):
            break
        # Minimal uniform shrinkage to recover validity
        best_radii *= 0.9999
        for i in range(N):
            best_centers[i, 0] = np.clip(best_centers[i, 0], best_radii[i], 1.0 - best_radii[i])
            best_centers[i, 1] = np.clip(best_centers[i, 1], best_radii[i], 1.0 - best_radii[i])
            
    return best_centers, best_radii, float(best_sum)
