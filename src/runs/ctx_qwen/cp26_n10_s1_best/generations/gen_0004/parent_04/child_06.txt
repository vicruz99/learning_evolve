import numpy as np
from scipy.optimize import minimize, linprog

N_CIRCLES = 26
TRIL_IDX = np.tril_indices(N_CIRCLES, -1)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Returns all inequality constraints g(x) >= 0."""
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    c = np.concatenate([
        cx - r,
        1.0 - cx - r,
        cy - r,
        1.0 - cy - r
    ])
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dr = r[:, None] + r[None, :]
    
    c = np.concatenate([c, dx[TRIL_IDX]**2 + dy[TRIL_IDX]**2 - dr[TRIL_IDX]**2])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-7, 0.5]."""
    b = []
    for _ in range(N_CIRCLES):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)])
    return b

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N_CIRCLES
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist)
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return np.full(n, 0.05)

def force_spread(centers, radii, steps=150):
    """Force-directed relaxation to spread points and push to boundaries."""
    pts = centers.copy()
    rs = radii.copy()
    for step in range(steps):
        f = np.zeros_like(pts)
        decay = 1.0 - step / steps
        for i in range(N_CIRCLES):
            for j in range(i + 1, N_CIRCLES):
                dx = pts[j, 0] - pts[i, 0]
                dy = pts[j, 1] - pts[i, 1]
                d = np.hypot(dx, dy)
                if d < rs[i] + rs[j] + 0.06 and d > 1e-6:
                    rep = 0.025 / (d**2 + 0.001)
                    fx, fy = rep * dx / d, rep * dy / d
                    f[i] -= [fx, fy]
                    f[j] += [fx, fy]
            # Boundary repulsion to utilize square edges
            for dim in range(2):
                if pts[i, dim] < rs[i] + 0.02:
                    f[i, dim] += 0.06 * decay
                elif pts[i, dim] > 1.0 - rs[i] - 0.02:
                    f[i, dim] -= 0.06 * decay
        pts += f * 0.05 * decay
        pts = np.clip(pts, 0.001, 0.999)
    return pts

def make_hex_init(seed, r0, angle=0.0):
    """Generate a hexagonal lattice initialization with optional rotation."""
    np.random.seed(seed)
    pts = []
    y = r0
    row = 0
    while len(pts) < N_CIRCLES + 5:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        if y > 1.0 + r0:
            break
            
    pts = np.array(pts[:N_CIRCLES + 5])
    
    if angle != 0.0:
        cx, cy = 0.5, 0.5
        ca, sa = np.cos(angle), np.sin(angle)
        dx = pts[:, 0] - cx
        dy = pts[:, 1] - cy
        pts[:, 0] = dx * ca - dy * sa + cx
        pts[:, 1] = dx * sa + dy * ca + cy
        
    mask = (pts[:,0]>=0.05) & (pts[:,0]<=0.95) & (pts[:,1]>=0.05) & (pts[:,1]<=0.95)
    pts = pts[mask]
    while len(pts) < N_CIRCLES:
        pts = np.vstack([pts, np.random.uniform(0.2, 0.8, (N_CIRCLES-len(pts), 2))])
    pts = pts[:N_CIRCLES]
    return pts

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -1.0
    best_x = None
    
    # Generate diverse initial configurations
    inits = []
    # Rotated hexagonal lattices with varying densities
    for r0 in np.linspace(0.085, 0.105, 5):
        for ang in np.linspace(0.0, 0.55, 9):
            inits.append(make_hex_init(0, r0, ang))
            
    # Force-directed layouts to break symmetry
    for s in range(8):
        pts = np.random.uniform(0.15, 0.85, (N_CIRCLES, 2))
        inits.append(force_spread(pts, np.full(N_CIRCLES, 0.05), steps=200))
        
    # Phase 1: Multi-start optimization with LP/SLSQP alternation
    for c_init in inits:
        r_lp = solve_lp_radii(c_init)
        x0 = np.zeros(3 * N_CIRCLES)
        x0[0::3] = c_init[:, 0]
        x0[1::3] = c_init[:, 1]
        x0[2::3] = r_lp
        
        # SLSQP optimization
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 8000, 'ftol': 1e-12, 'disp': False})
            if not np.isnan(res.fun):
                curr_sum = -res.fun
                c_vals = constraints(res.x)
                if np.min(c_vals) >= -1e-7 and curr_sum > best_sum:
                    best_sum = curr_sum
                    best_x = res.x.copy()
        except Exception:
            continue
            
        # Local refinement loop on this start
        x_curr = best_x.copy() if best_x is not None and curr_sum == best_sum else x0
        for _ in range(2):
            x_pert = x_curr + np.random.normal(0, 1e-4, 3 * N_CIRCLES)
            x_pert[2::3] = np.maximum(x_pert[2::3] * 0.99, 0.001)
            r_p = x_pert[2::3]
            x_pert[0::3] = np.clip(x_pert[0::3], r_p, 1.0 - r_p)
            x_pert[1::3] = np.clip(x_pert[1::3], r_p, 1.0 - r_p)
            
            try:
                res = minimize(objective, x_pert, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 6000, 'ftol': 1e-12, 'disp': False})
                if not np.isnan(res.fun):
                    curr = -res.fun
                    c_vals = constraints(res.x)
                    if np.min(c_vals) >= -1e-7 and curr > best_sum:
                        best_sum = curr
                        best_x = res.x.copy()
                        x_curr = res.x.copy()
                        
                    # LP radii snap
                    if best_x is not None:
                        c_best = np.column_stack((best_x[0::3], best_x[1::3]))
                        r_lp = solve_lp_radii(c_best)
                        if np.sum(r_lp) > best_sum:
                            best_x[2::3] = r_lp
                            best_sum = np.sum(r_lp)
                            x_curr = best_x.copy()
            except Exception:
                pass

    # Phase 2: Global deflation & perturbation refinement
    if best_x is not None:
        for step in range(20):
            noise = 0.0025 / (step + 1)
            x_pert = best_x + np.random.normal(0, noise, best_x.shape)
            r_pert = np.maximum(x_pert[2::3] * 0.983, 0.005)
            x_pert[2::3] = r_pert
            x_pert[0::3] = np.clip(x_pert[0::3], r_pert, 1.0 - r_pert)
            x_pert[1::3] = np.clip(x_pert[1::3], r_pert, 1.0 - r_pert)
            
            try:
                res = minimize(objective, x_pert, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 10000, 'ftol': 1e-12, 'disp': False})
                if not np.isnan(res.fun):
                    curr = -res.fun
                    c_vals = constraints(res.x)
                    if np.min(c_vals) >= -1e-7 and curr > best_sum:
                        best_sum = curr
                        best_x = res.x.copy()
                        
                    # LP check
                    c_opt = np.column_stack((res.x[0::3], res.x[1::3]))
                    r_lp = solve_lp_radii(c_opt)
                    if np.sum(r_lp) > best_sum:
                        best_x[2::3] = r_lp
                        best_sum = np.sum(r_lp)
            except Exception:
                pass

    # Fallback (should not be reached)
    if best_x is None:
        best_x = np.zeros(3 * N_CIRCLES)
        best_x[0::3] = np.tile(np.linspace(0.15, 0.85, 6), 5)[:N_CIRCLES]
        best_x[1::3] = np.repeat(np.linspace(0.15, 0.85, 5), 6)[:N_CIRCLES]
        best_x[2::3] = 0.08
        best_sum = -objective(best_x)

    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3].copy()
    
    # Final strict validity adjustment against 1e-12 tolerance
    for _ in range(100):
        valid = True
        for i in range(N_CIRCLES):
            if radii[i] < 0: valid=False; break
            if centers[i,0] - radii[i] < -1e-12 or centers[i,0] + radii[i] > 1.0 + 1e-12: valid=False; break
            if centers[i,1] - radii[i] < -1e-12 or centers[i,1] + radii[i] > 1.0 + 1e-12: valid=False; break
        if valid:
            for i in range(N_CIRCLES):
                for j in range(i+1, N_CIRCLES):
                    d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
                    if d < radii[i] + radii[j] - 1e-12: valid=False; break
                if not valid: break
        if valid: break
        # Minimal shrinkage to recover validity
        radii *= 0.9999
        centers[:,0] = np.clip(centers[:,0], radii, 1.0 - radii)
        centers[:,1] = np.clip(centers[:,1], radii, 1.0 - radii)
        
    return centers, radii, float(np.sum(radii))
