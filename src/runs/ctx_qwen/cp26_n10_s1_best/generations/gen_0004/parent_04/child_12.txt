import numpy as np
from scipy.optimize import minimize, linprog

N_CIRCLES = 26
TRIL_IDX = np.tril_indices(N_CIRCLES, -1)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Returns all inequality constraints g(x) >= 0."""
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    # Boundary constraints
    c = np.concatenate([
        cx - r,
        1.0 - cx - r,
        cy - r,
        1.0 - cy - r
    ])
    
    # Overlap constraints using squared distances for better gradient behavior
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dr = r[:, None] + r[None, :]
    
    c = np.concatenate([c, dx[TRIL_IDX]**2 + dy[TRIL_IDX]**2 - dr[TRIL_IDX]**2])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-7, 0.5]."""
    b = []
    for _ in range(N_CIRCLES):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)])
    return b

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N_CIRCLES
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist)
            idx += 1
            
    bounds_lp = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_lp, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def generate_hex_init(pattern, r0, angle_deg):
    """Generates a hexagonal lattice initialization with optional rotation."""
    centers = np.zeros((N_CIRCLES, 2))
    idx = 0
    y = r0
    y_step = np.sqrt(3.0) * r0
    for i, cnt in enumerate(pattern):
        x_start = r0 + (r0 if i % 2 == 1 else 0.0)
        for k in range(cnt):
            if idx < N_CIRCLES:
                centers[idx, 0] = x_start + k * 2.0 * r0
                centers[idx, 1] = y
                idx += 1
        y += y_step
    while idx < N_CIRCLES:
        centers[idx, 0] = np.random.uniform(0.1, 0.9)
        centers[idx, 1] = np.random.uniform(0.1, 0.9)
        idx += 1
        
    if angle_deg != 0.0:
        ang = np.deg2rad(angle_deg)
        c, s = np.cos(ang), np.sin(ang)
        centers -= 0.5
        centers = centers @ np.array([[c, -s], [s, c]])
        centers += 0.5
        centers = np.clip(centers, 0.01, 0.99)
    return centers

def force_directed_init(seed):
    """Force-directed layout to spread points evenly."""
    np.random.seed(seed)
    centers = np.random.uniform(0.15, 0.85, (N_CIRCLES, 2))
    for _ in range(800):
        forces = np.zeros_like(centers)
        for i in range(N_CIRCLES):
            for j in range(i + 1, N_CIRCLES):
                dx = centers[j, 0] - centers[i, 0]
                dy = centers[j, 1] - centers[i, 1]
                dist = np.hypot(dx, dy)
                if dist < 0.25 and dist > 1e-6:
                    f = 0.01 / (dist**2 + 0.001)
                    fx, fy = f * dx, f * dy
                    forces[i] -= [fx, fy]
                    forces[j] += [fx, fy]
            for dim in range(2):
                if centers[i, dim] < 0.1: forces[i, dim] += 0.02
                elif centers[i, dim] > 0.9: forces[i, dim] -= 0.02
        centers += forces * 0.05
        centers = np.clip(centers, 0.02, 0.98)
    return centers

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    best_sum = -1.0
    best_x = None
    
    patterns = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [6, 6, 5, 5, 4],
        [5, 5, 6, 6, 4], [6, 6, 6, 4, 4], [4, 6, 6, 6, 4],
        [5, 6, 6, 5, 4], [6, 5, 5, 6, 4], [5, 5, 5, 5, 6]
    ]
    
    configs = []
    # Rotated hexagonal lattices
    for pat in patterns:
        for ang in np.linspace(-4.0, 4.0, 17):
            c = generate_hex_init(pat, 0.085, ang)
            configs.append(c)
            
    # Force-directed starts
    for s in range(12):
        configs.append(force_directed_init(s))
        
    # Grid start
    c_grid = np.zeros((N_CIRCLES, 2))
    idx = 0
    for i in range(6):
        for j in range(5):
            if idx < N_CIRCLES:
                c_grid[idx] = [0.1 + 0.18*j, 0.1 + 0.16*i]
                idx += 1
    configs.append(c_grid)
    
    # Phase 1: Multi-start optimization with LP refinement
    for c_init in configs:
        # Initialize radii via LP for exact feasibility
        r_lp = solve_lp_radii(c_init)
        if r_lp is None:
            r_lp = np.full(N_CIRCLES, 0.06)
            
        x0 = np.zeros(3 * N_CIRCLES)
        x0[0::3] = c_init[:, 0]
        x0[1::3] = c_init[:, 1]
        x0[2::3] = r_lp * 0.9995  # Slight shrink to ensure strict initial feasibility
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 20000, 'ftol': 1e-13, 'disp': False})
            if not np.isnan(res.fun):
                c_opt = np.column_stack((res.x[0::3], res.x[1::3]))
                
                # LP refinement to extract absolute maximum radii for optimized centers
                r_lp_ref = solve_lp_radii(c_opt)
                if r_lp_ref is not None:
                    curr_sum = np.sum(r_lp_ref)
                    if curr_sum > best_sum:
                        best_sum = curr_sum
                        best_x = np.zeros(3 * N_CIRCLES)
                        best_x[0::3] = c_opt[:, 0]
                        best_x[1::3] = c_opt[:, 1]
                        best_x[2::3] = r_lp_ref
                else:
                    curr_sum = -res.fun
                    c_vals = constraints(res.x)
                    if np.min(c_vals) >= -1e-7 and curr_sum > best_sum:
                        best_sum = curr_sum
                        best_x = res.x.copy()
        except Exception:
            pass

    if best_x is not None:

            x_pert = best_x + np.random.normal(0, noise_scale, best_x.shape)
            r_pert = np.maximum(x_pert[2::3] * 0.982, 0.01)
            x_pert[2::3] = r_pert
            x_pert[0::3] = np.clip(x_pert[0::3], r_pert, 1.0 - r_pert)
            x_pert[1::3] = np.clip(x_pert[1::3], r_pert, 1.0 - r_pert)
            
            try:
                res = minimize(objective, x_pert, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 15000, 'ftol': 1e-13, 'disp': False})
                if not np.isnan(res.fun):
                    c_opt = np.column_stack((res.x[0::3], res.x[1::3]))
                    r_lp_ref = solve_lp_radii(c_opt)
                    if r_lp_ref is not None:
                        curr_sum = np.sum(r_lp_ref)
                        if curr_sum > best_sum:
                            best_sum = curr_sum
                            best_x = np.zeros(3 * N_CIRCLES)
                            best_x[0::3] = c_opt[:, 0]
                            best_x[1::3] = c_opt[:, 1]
                            best_x[2::3] = r_lp_ref
            except Exception:
                pass
                
    # Fallback (should not be reached)
    if best_x is None:
        best_x = np.zeros(3 * N_CIRCLES)
        best_x[0::3] = np.tile(np.linspace(0.1, 0.9, 5), 6)[:N_CIRCLES]
        best_x[1::3] = np.repeat(np.linspace(0.1, 0.9, 6), 5)[:N_CIRCLES]
        best_x[2::3] = 0.08
        
    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3]
    
    # Final strict validity adjustment against 1e-12 tolerance
    for _ in range(100):
        valid = True
        for i in range(N_CIRCLES):
            if radii[i] < 0 or centers[i,0] < radii[i]-1e-10 or centers[i,0] > 1-radii[i]+1e-10 or \
               centers[i,1] < radii[i]-1e-10 or centers[i,1] > 1-radii[i]+1e-10:

        if valid:

                for j in range(i+1, N_CIRCLES):

        if valid: break

        radii *= 0.9995
        centers[:,0] = np.clip(centers[:,0], radii, 1.0-radii)
        centers[:,1] = np.clip(centers[:,1], radii, 1.0-radii)
        
    return centers, radii, float(np.sum(radii))
