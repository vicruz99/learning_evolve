To improve upon the current packing sum of radii (~2.630), we need a more robust optimization strategy that escapes local minima and tightly couples center positioning with radius expansion. 

Key improvements in this approach:
1. **Alternating Optimization (SLSQP ↔ LP)**: Instead of optimizing centers and radii simultaneously in one long SLSQP run, we alternate between optimizing positions (SLSQP) and optimally computing radii for fixed centers (Linear Programming). This forces the packing to be as tight as possible at each step, activating constraints properly and guiding the gradient search effectively.
2. **Constraint-Driven Refinement**: After convergence, we intentionally "deflate" radii slightly, perturb centers, and re-optimize. This breaks rigid local configurations and allows circles to slide into better topological arrangements before expanding again.
3. **Aggressive Inflation Phase**: Once a stable configuration is found, we uniformly scale radii up by ~1.5% and re-run optimization. This pushes the solver to find configurations that can structurally support larger circles.
4. **Diverse, Physics-Informed Initializations**: We generate hexagonal lattices with varied row counts, apply rotations to break symmetry, and use force-directed layouts that explicitly repel circles while keeping them away from edges until necessary. Safe initial radii are computed to guarantee strict feasibility before optimization begins.
5. **Strict Feasibility Management**: All intermediate steps project variables back to feasible bounds. A final safety loop guarantees compliance with the `1e-12` tolerance used in validation, shrinking radii minimally only if necessary.

import numpy as np
from scipy.optimize import minimize, linprog

N_CIRCLES = 26

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Returns all inequality constraints g(x) >= 0."""
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    # Boundary constraints
    c = np.concatenate([
        cx - r,
        1.0 - cx - r,
        cy - r,
        1.0 - cy - r
    ])
    
    # Overlap constraints (squared distance - sum_radii^2 >= 0)
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dr = r[:, None] + r[None, :]
    
    i, j = np.tril_indices(N_CIRCLES, -1)
    c = np.concatenate([c, dx[i, j]**2 + dy[i, j]**2 - dr[i, j]**2])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-6, 0.5]."""
    b = []
    for _ in range(N_CIRCLES):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)])
    return b

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N_CIRCLES
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A = np.zeros((num_ineq, n))
    b = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], 
                  centers[i, 1], 1.0 - centers[i, 1])
        A[idx, i] = 1.0
        b[idx] = max(0.0, lim)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], 
                            centers[i, 1] - centers[j, 1])
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = max(0.0, dist)
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def generate_inits():
    """Generate diverse initial configurations."""
    inits = []
    patterns = [
        [5,6,5,6,4], [6,5,6,5,4], [5,6,6,5,4],
        [4,6,6,6,4], [5,5,5,5,6], [6,6,6,4,4],
        [5,7,5,6,3], [6,6,5,5,4], [7,5,6,5,3],
        [4,5,6,5,6], [5,5,6,5,5], [6,5,5,5,5]
    ]
    
    # Hexagonal lattices
    for pat in patterns:
        centers = np.zeros((N_CIRCLES, 2))
        idx = 0
        y = 0.09
        y_step = np.sqrt(3.0) * 0.09
        for i, cnt in enumerate(pat):
            x_start = 0.09 + (0.09 if i % 2 == 1 else 0.0)
            for k in range(cnt):
                if idx < N_CIRCLES:
                    centers[idx, 0] = x_start + k * 0.18
                    centers[idx, 1] = y
                    idx += 1
            y += y_step
        while idx < N_CIRCLES:
            centers[idx] = np.random.uniform(0.15, 0.85, 2)
            idx += 1
        inits.append(centers)
        
    # Rotated hexagonal lattices
    base = inits[0]
    for ang in [0.12, -0.12, 0.25, -0.25, 0.4, 0.6]:
        cx, cy = 0.5, 0.5
        ca, sa = np.cos(ang), np.sin(ang)
        dx = base[:, 0] - cx
        dy = base[:, 1] - cy
        c_rot = np.column_stack([dx*ca - dy*sa + cx, dx*sa + dy*ca + cy])
        inits.append(c_rot)
        
    # Force-directed layouts
    for s in range(10):
        np.random.seed(s)
        c = np.random.uniform(0.15, 0.85, (N_CIRCLES, 2))
        for _ in range(500):
            f = np.zeros_like(c)
            for i in range(N_CIRCLES):
                for j in range(i + 1, N_CIRCLES):
                    d = c[j] - c[i]
                    dist = np.linalg.norm(d)
                    if dist < 0.28 and dist > 1e-5:
                        ff = 0.012 / (dist**2 + 0.001)
                        f[i] -= ff * d
                        f[j] += ff * d
                # Boundary repulsion
                if c[i, 0] < 0.12: f[i, 0] += 0.025
                elif c[i, 0] > 0.88: f[i, 0] -= 0.025
                if c[i, 1] < 0.12: f[i, 1] += 0.025
                elif c[i, 1] > 0.88: f[i, 1] -= 0.025
            c += f * 0.05
            c = np.clip(c, 0.05, 0.95)
        inits.append(c)
        
    return inits

def compute_safe_radii(centers):
    """Compute strictly feasible initial radii."""
    n = N_CIRCLES
    radii = np.zeros(n)
    for i in range(n):
        min_d = min(centers[i, 0], 1.0 - centers[i, 0],
                    centers[i, 1], 1.0 - centers[i, 1])
        for j in range(n):
            if i != j:
                d = np.hypot(centers[i, 0] - centers[j, 0],
                             centers[i, 1] - centers[j, 1])
                if d < min_d:
                    min_d = d
        radii[i] = min_d * 0.4
    return radii

def project_feasible(x):
    """Project optimization vector to satisfy bounds and basic feasibility."""
    x = x.copy()
    for i in range(N_CIRCLES):
        r = max(1e-6, x[3 * i + 2])
        x[3 * i] = np.clip(x[3 * i], r, 1.0 - r)
        x[3 * i + 1] = np.clip(x[3 * i + 1], r, 1.0 - r)
        x[3 * i + 2] = r
    return x

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -1.0
    best_x = None
    inits = generate_inits()
    
    # Phase 1: Multi-start with alternating SLSQP & LP refinement
    for c_init in inits:
        x0 = np.zeros(3 * N_CIRCLES)
        x0[0::3] = c_init[:, 0]
        x0[1::3] = c_init[:, 1]
        x0[2::3] = compute_safe_radii(c_init)
        x0 += np.random.normal(0, 1e-4, x0.shape)
        x0 = project_feasible(x0)
        
        # Alternate optimization for tighter packing
        for _ in range(3):
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 15000, 'ftol': 1e-13, 'disp': False})
                if np.isnan(res.fun):
                    break
                c_opt = np.column_stack((res.x[0::3], res.x[1::3]))
                r_lp = solve_lp_radii(c_opt)
                if r_lp is not None:
                    x0[0::3] = c_opt[:, 0]
                    x0[1::3] = c_opt[:, 1]
                    x0[2::3] = r_lp * 0.9999  # Slight shrink to maintain strict feasibility
                    x0 = project_feasible(x0)
            except Exception:
                break
                
        curr_sum = np.sum(x0[2::3])
        if curr_sum > best_sum:
            best_sum = curr_sum
            best_x = x0.copy()
            
    # Phase 2: Deflation & Perturbation to escape local minima
    if best_x is not None:
        for step in range(25):
            x0 = best_x.copy()
            x0[2::3] *= 0.980  # Deflate to free space
            x0[0::3] += np.random.normal(0, 0.002, N_CIRCLES)
            x0[1::3] += np.random.normal(0, 0.002, N_CIRCLES)
            x0 = project_feasible(x0)
            
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 12000, 'ftol': 1e-13, 'disp': False})
                if not np.isnan(res.fun):
                    c_opt = np.column_stack((res.x[0::3], res.x[1::3]))
                    r_lp = solve_lp_radii(c_opt)
                    if r_lp is not None:
                        curr_sum = np.sum(r_lp)
                        if curr_sum > best_sum:
                            best_sum = curr_sum
                            best_x = np.zeros(3 * N_CIRCLES)
                            best_x[0::3] = c_opt[:, 0]
                            best_x[1::3] = c_opt[:, 1]
                            best_x[2::3] = r_lp
            except Exception:
                pass
                
    # Phase 3: Aggressive Inflation Push
    if best_x is not None:
        x0 = best_x.copy()
        x0[2::3] *= 1.025
        x0 = project_feasible(x0)
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 20000, 'ftol': 1e-14, 'disp': False})
            if not np.isnan(res.fun):
                c_opt = np.column_stack((res.x[0::3], res.x[1::3]))
                r_lp = solve_lp_radii(c_opt)
                if r_lp is not None and np.sum(r_lp) > best_sum:
                    best_sum = np.sum(r_lp)
                    best_x = np.zeros(3 * N_CIRCLES)
                    best_x[0::3] = c_opt[:, 0]
                    best_x[1::3] = c_opt[:, 1]
                    best_x[2::3] = r_lp
        except Exception:
            pass

    # Fallback
    if best_x is None:
        c_f = generate_inits()[0]
        r_f = compute_safe_radii(c_f)
        best_x = np.zeros(3 * N_CIRCLES)
        best_x[0::3] = c_f[:, 0]
        best_x[1::3] = c_f[:, 1]
        best_x[2::3] = r_f
        
    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3]
    
    # Final strict validity adjustment against 1e-12 tolerance
    for _ in range(100):
        valid = True
        for i in range(N_CIRCLES):
            if radii[i] < 0: valid = False; break
            if centers[i, 0] - radii[i] < -1e-10 or centers[i, 0] + radii[i] > 1.0 + 1e-10: valid = False; break
            if centers[i, 1] - radii[i] < -1e-10 or centers[i, 1] + radii[i] > 1.0 + 1e-10: valid = False; break
        if valid:
            for i in range(N_CIRCLES):
                for j in range(i + 1, N_CIRCLES):
                    d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                    if d < radii[i] + radii[j] - 1e-10:
                        valid = False; break
                if not valid: break
        if valid: break
        radii *= 0.9998
        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)
        
    return centers, radii, float(np.sum(radii))
