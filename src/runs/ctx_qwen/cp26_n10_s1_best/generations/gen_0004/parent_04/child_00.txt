import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Returns all inequality constraints g(x) >= 0."""
    cx = x[0::3]
    cy = x[1::3]
    cr = x[2::3]
    
    c = np.empty(4*N + N*(N-1)//2)
    c[:N] = cx - cr
    c[N:2*N] = 1.0 - cx - cr
    c[2*N:3*N] = cy - cr
    c[3*N:4*N] = 1.0 - cy - cr
    
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dr = cr[:, None] + cr[None, :]
    
    i_idx, j_idx = np.triu_indices(N, 1)
    c[4*N:] = dx[i_idx, j_idx]**2 + dy[i_idx, j_idx]**2 - dr[i_idx, j_idx]**2
    return c

def constraint_jac(x):
    """Analytic Jacobian of constraints."""
    cx = x[0::3]
    cy = x[1::3]
    cr = x[2::3]
    
    jac = np.zeros((4*N + N*(N-1)//2, 3*N))
    
    # Boundary constraints derivatives
    for i in range(N):
        jac[i, 3*i] = 1.0; jac[i, 3*i+2] = -1.0
        jac[N+i, 3*i] = -1.0; jac[N+i, 3*i+2] = -1.0
        jac[2*N+i, 3*i+1] = 1.0; jac[2*N+i, 3*i+2] = -1.0
        jac[3*N+i, 3*i+1] = -1.0; jac[3*N+i, 3*i+2] = -1.0
        
    # Overlap constraints derivatives
    i_idx, j_idx = np.triu_indices(N, 1)
    k = 4*N
    for i, j in zip(i_idx, j_idx):
        dx = cx[i] - cx[j]
        dy = cy[i] - cy[j]
        dr = cr[i] + cr[j]
        
        jac[k, 3*i] = 2.0*dx; jac[k, 3*j] = -2.0*dx
        jac[k, 3*i+1] = 2.0*dy; jac[k, 3*j+1] = -2.0*dy
        jac[k, 3*i+2] = -2.0*dr; jac[k, 3*j+2] = -2.0*dr
        k += 1
    return jac

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-6, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)])
    return b

def solve_lp_radii(centers):
    """Optimally scale radii for fixed centers using Linear Programming."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_ineq = n + n*(n-1)//2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    for i in range(n):
        lim = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    for i in range(n):
        for j in range(i+1, n):
            d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, d)
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def force_directed_init(n, seed, steps=400):
    """Force-directed layout to spread points evenly."""
    np.random.seed(seed)
    pts = np.random.uniform(0.12, 0.88, (n, 2))
    for _ in range(steps):
        forces = np.zeros_like(pts)
        for i in range(n):
            for j in range(i+1, n):
                diff = pts[j] - pts[i]
                dist = np.hypot(diff[0], diff[1])
                if dist < 0.32 and dist > 1e-5:
                    f = 0.012 / (dist**2 + 0.001)
                    forces[i] -= f * diff
                    forces[j] += f * diff
        for i in range(n):
            if pts[i,0] < 0.14: forces[i,0] += 0.06
            elif pts[i,0] > 0.86: forces[i,0] -= 0.06
            if pts[i,1] < 0.14: forces[i,1] += 0.06
            elif pts[i,1] > 0.86: forces[i,1] -= 0.06
        pts += forces * 0.06
        pts = np.clip(pts, 0.05, 0.95)
    return pts

        dy = pts[:,1] - cy
        pts[:,0] = dx*ca - dy*sa + cx
        pts[:,1] = dx*sa + dy*ca + cy
        inits.append(pts)
        
    # 2. Force-directed spreads
    for s in range(8):
        inits.append(force_directed_init(N, seed=s))

    for s in range(5):

        pts = np.zeros((N, 2))
        idx = 0
        for i in range(6):
            for j in range(5):
                if idx < N:
                    pts[idx] = [0.12 + 0.155*j + np.random.uniform(-0.02, 0.02),
                                0.12 + 0.155*i + np.random.uniform(-0.02, 0.02)]
                    idx += 1
        inits.append(pts)
        
    # Phase 1: Multi-start optimization + LP refinement
    for c_init in inits:
        r_init = np.full(N, 0.05)
        x0 = np.zeros(3*N)
        x0[0::3] = c_init[:,0]
        x0[1::3] = c_init[:,1]
