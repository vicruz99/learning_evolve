The key insight is that for fixed centers, the optimal radii can be found exactly and instantly via Linear Programming (LP). This allows us to decouple the problem:
1. **LP Radius Solver**: Given any set of centers, `linprog` computes the strictly optimal radii satisfying all boundary and non-overlap constraints, maximizing the sum.
2. **Alternating Optimization**: We alternate between optimizing centers (to create more space) and solving LP for radii (to exploit the space). Centers are optimized using `SLSQP` on the full nonlinear formulation, initialized from LP-refined states.
3. **Diverse Initialization**: We generate high-quality starting configurations using rotated hexagonal lattices (varying density and angle), perturbed grids, and force-directed layouts. This covers different topological basins.
4. **Perturbation & Basin Hopping**: After each successful optimization, we perturb the best configuration with decaying noise and re-optimize. This simulates a temperature reduction to escape shallow local optima.
5. **Strict Validation**: A final pass ensures all constraints are satisfied within the `1e-12` tolerance of the validator, applying minimal radius shrinkage only if absolutely necessary.

import numpy as np
from scipy.optimize import minimize, linprog

N_CIRCLES = 26
TRIL_IDX = np.tril_indices(N_CIRCLES, -1)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Returns all inequality constraints g(x) >= 0."""
    cx, cy, r = x[0::3], x[1::3], x[2::3]
    
    # Boundary constraints
    c = np.concatenate([
        cx - r,
        1.0 - cx - r,
        cy - r,
        1.0 - cy - r
    ])
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dr = r[:, None] + r[None, :]
    
    return np.concatenate([c, dx[TRIL_IDX]**2 + dy[TRIL_IDX]**2 - dr[TRIL_IDX]**2])

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-7, 0.5]."""
    b = []
    for _ in range(N_CIRCLES):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)])
    return b

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0],
                  centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0],
                         centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, d)
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def make_x0(centers, radii):
    """Flatten centers and radii into optimization vector."""
    x = np.zeros(3 * N_CIRCLES)
    x[0::3] = centers[:, 0]
    x[1::3] = centers[:, 1]
    x[2::3] = radii
    return x

def generate_initial_centers():
    """Generate diverse high-quality initial center configurations."""
    inits = []
    
    # 1. Rotated Hexagonal Lattices with varying densities
    for r0 in [0.082, 0.088, 0.094, 0.100]:
        for ang in [0.0, 0.12, 0.25, -0.15, 0.38, 0.52]:
            c = np.zeros((N_CIRCLES, 2))
            idx = 0
            y = r0
            row = 0
            while idx < N_CIRCLES and y + r0 <= 1.0:
                x_start = r0 if row % 2 == 0 else 2.0 * r0
                x = x_start
                while idx < N_CIRCLES and x + r0 <= 1.0:
                    c[idx] = [x, y]
                    idx += 1
                    x += 2.0 * r0
                y += np.sqrt(3.0) * r0
                row += 1
            while idx < N_CIRCLES:
                c[idx] = np.random.uniform(0.1, 0.9, 2)
                idx += 1
                
            if ang != 0.0:
                cx, cy = 0.5, 0.5
                c -= [cx, cy]
                rot = np.array([[np.cos(ang), -np.sin(ang)], 
                               [np.sin(ang), np.cos(ang)]])
                c = (c @ rot.T) + [cx, cy]
            inits.append(c)
            
    # 2. Force-directed layouts
    for s in range(6):
        np.random.seed(s + 100)
        c = np.random.uniform(0.15, 0.85, (N_CIRCLES, 2))
        for _ in range(400):
            forces = np.zeros_like(c)
            for i in range(N_CIRCLES):
                for j in range(i + 1, N_CIRCLES):
                    dx = c[j] - c[i]
                    d = np.hypot(dx[0], dx[1])
                    if d < 0.25 and d > 1e-5:
                        f = 0.008 / (d**2 + 0.001)
                        forces[i] -= f * dx
                        forces[j] += f * dx
                for k in range(2):
                    if c[i, k] < 0.12: forces[i, k] += 0.06
                    elif c[i, k] > 0.88: forces[i, k] -= 0.06
            c += forces * 0.06
            c = np.clip(c, 0.05, 0.95)
        inits.append(c)
        
    # 3. Perturbed Grid
    c_grid = np.zeros((N_CIRCLES, 2))
    idx = 0
    for i in range(6):
        for j in range(5):
            if idx < N_CIRCLES:
                c_grid[idx] = [0.09 + 0.18 * j, 0.09 + 0.16 * i]
                idx += 1
    c_grid += np.random.uniform(-0.01, 0.01, c_grid.shape)
    inits.append(c_grid)
    
    return inits

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    inits = generate_initial_centers()
    
    # Phase 1: Multi-start optimization with LP refinement
    for c_init in inits:
        r_init = np.full(N_CIRCLES, 0.08)
        x0 = make_x0(c_init, r_init)
        x0 += np.random.normal(0, 1e-5, x0.shape)
        # Project to feasible bounds
        r_proj = np.maximum(x0[2::3], 0.01)
        x0[2::3] = r_proj
        x0[0::3] = np.clip(x0[0::3], r_proj, 1.0 - r_proj)
        x0[1::3] = np.clip(x0[1::3], r_proj, 1.0 - r_proj)
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 25000, 
                                                      'ftol': 1e-13, 'disp': False})
            
            if not np.isnan(res.fun):
                c_opt = np.column_stack((res.x[0::3], res.x[1::3]))
                r_lp = solve_lp_radii(c_opt)
                
                if r_lp is not None:
                    curr_sum = np.sum(r_lp)
                    if curr_sum > best_sum:
                        best_sum = curr_sum
                        best_centers = c_opt.copy()
                        best_radii = r_lp.copy()
        except Exception:
            continue
            
        # Phase 2: Perturbation & Refinement Loop (Simulated Annealing style)
        if best_centers is not None:
            for step in range(12):
                noise_scale = 0.0025 / (step + 1)
                c_pert = best_centers + np.random.normal(0, noise_scale, best_centers.shape)
                r_pert = best_radii * 0.988
                
                # Project to feasible region
                r_pert = np.maximum(r_pert, 0.005)
                c_pert = np.clip(c_pert, r_pert, 1.0 - r_pert)
                
                x0_pert = make_x0(c_pert, r_pert)
                
                try:
                    res = minimize(objective, x0_pert, method='SLSQP', bounds=bounds,
                                   constraints=cons, options={'maxiter': 15000, 
                                                              'ftol': 1e-13, 'disp': False})
                    if not np.isnan(res.fun):
                        c_opt = np.column_stack((res.x[0::3], res.x[1::3]))
                        r_lp = solve_lp_radii(c_opt)
                        
                        if r_lp is not None:
                            curr_sum = np.sum(r_lp)
                            if curr_sum > best_sum:
                                best_sum = curr_sum
                                best_centers = c_opt.copy()
                                best_radii = r_lp.copy()
                except Exception:
                    pass
                    
    # Fallback (should not be reached)
    if best_centers is None:
        best_centers = inits[0]
        best_radii = np.full(N_CIRCLES, 0.06)
        best_sum = np.sum(best_radii)
        
    # Phase 3: Strict Validation & Minimal Numerical Repair
    # Ensures strict compliance with the 1e-12 tolerance in the validator
    for _ in range(100):
        valid = True
        for i in range(N_CIRCLES):
            if (best_radii[i] < 0 or 
                best_centers[i, 0] - best_radii[i] < -1e-12 or 
                best_centers[i, 0] + best_radii[i] > 1.0 + 1e-12 or
                best_centers[i, 1] - best_radii[i] < -1e-12 or 
                best_centers[i, 1] + best_radii[i] > 1.0 + 1e-12):
                valid = False
                break
        if valid:
            for i in range(N_CIRCLES):
                for j in range(i + 1, N_CIRCLES):
                    d = np.hypot(best_centers[i, 0] - best_centers[j, 0],
                                 best_centers[i, 1] - best_centers[j, 1])
                    if d < best_radii[i] + best_radii[j] - 1e-12:
                        valid = False
                        break
                if not valid:
                    break
        if valid:
            break
            
        # Gentle shrinkage to recover strict feasibility
        best_radii *= 0.9999
        best_centers[:, 0] = np.clip(best_centers[:, 0], best_radii, 1.0 - best_radii)
        best_centers[:, 1] = np.clip(best_centers[:, 1], best_radii, 1.0 - best_radii)
        
    return best_centers, best_radii, float(best_sum)
