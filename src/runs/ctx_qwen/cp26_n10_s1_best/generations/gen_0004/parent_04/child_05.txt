from scipy.optimize import minimize, linprog

N_CIRCLES = 26

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Returns all inequality constraints g(x) >= 0."""
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    # Boundary constraints
    c = np.concatenate([
        cx - r,
        1.0 - cx - r,
        cy - r,
        1.0 - cy - r
    ])
    
    # Overlap constraints: dist^2 >= (ri + rj)^2
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dr = r[:, None] + r[None, :]
    
    i_idx, j_idx = np.tril_indices(N_CIRCLES, -1)
    c = np.concatenate([c, dx[i_idx, j_idx]**2 + dy[i_idx, j_idx]**2 - dr[i_idx, j_idx]**2])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [0, 0.5]."""
    b = []
    for _ in range(N_CIRCLES):
        b.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
    return b

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N_CIRCLES
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(dist to walls)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], 
                  centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], 
                            centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dist
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def force_directed_init(seed, steps=1000):
    """Generate initial configuration using repulsive force simulation."""
    np.random.seed(seed)
    centers = np.random.uniform(0.15, 0.85, (N_CIRCLES, 2))
    
    for step in range(steps):
        forces = np.zeros_like(centers)
        for i in range(N_CIRCLES):
            for j in range(i + 1, N_CIRCLES):
                dx = centers[j, 0] - centers[i, 0]
                dy = centers[j, 1] - centers[i, 1]
                dist = np.hypot(dx, dy)
                if dist < 0.28 and dist > 1e-6:
                    f = 0.012 / (dist**2 + 0.001)
                    fx, fy = f * dx, f * dy
                    forces[i] -= [fx, fy]
                    forces[j] += [fx, fy]
            # Boundary repulsion
            for dim in range(2):
                if centers[i, dim] < 0.1: forces[i, dim] += 0.025
                elif centers[i, dim] > 0.9: forces[i, dim] -= 0.025
                
        lr = 0.05 * (1.0 - step / steps)
        centers += forces * lr
        centers = np.clip(centers, 0.02, 0.98)
    return centers

def hex_lattice_init(r0):
    """Generate a hexagonal lattice initialization."""
    centers = np.zeros((N_CIRCLES, 2))
    idx = 0
    y = r0
    row = 0
    while idx < N_CIRCLES:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while idx < N_CIRCLES and x + r0 <= 1.0:
            centers[idx] = [x, y]
            idx += 1
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
    while idx < N_CIRCLES:
        centers[idx] = np.random.uniform(0.2, 0.8, 2)
        idx += 1
    return centers

def make_x0(centers, r_arr):
    """Flatten centers and radii into optimization vector."""
    x0 = np.zeros(3 * N_CIRCLES)
    x0[0::3] = centers[:, 0]
    x0[1::3] = centers[:, 1]
    x0[2::3] = r_arr
    return x0

def check_validity(c_arr, r_arr):
    """Strictly check feasibility against validator tolerances."""
    n = len(r_arr)
    for i in range(n):
        if r_arr[i] < 0: return False
        if c_arr[i,0] < r_arr[i] - 1e-9 or c_arr[i,0] > 1.0 - r_arr[i] + 1e-9: return False
        if c_arr[i,1] < r_arr[i] - 1e-9 or c_arr[i,1] > 1.0 - r_arr[i] + 1e-9: return False
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(c_arr[i,0]-c_arr[j,0], c_arr[i,1]-c_arr[j,1])
            if d < r_arr[i] + r_arr[j] - 1e-9: return False
    return True
