def objective(x):

    return -np.sum(x[2::3])

def constraints(x):

    cy = x[1::3]
    r = x[2::3]

    c = np.concatenate([
        cx - r,
        1.0 - cx - r,
        cy - r,
        1.0 - cy - r
    ])
    
    # Overlap constraints using squared distances
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dr = r[:, None] + r[None, :]

    c = np.concatenate([c, dx[i_idx, j_idx]**2 + dy[i_idx, j_idx]**2 - dr[i_idx, j_idx]**2])

def solve_lp_radii(centers):

    n = N_CIRCLES
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)

        b_ub[idx] = max(0.0, lim)
        A_ub[idx, i] = 1.0
        idx += 1

            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])

            idx += 1

    except Exception:
        pass
    return None

    r = radii.copy()
    n = N_CIRCLES
    
    for step in range(steps):
        forces = np.zeros_like(pts)

                dy = pts[j, 1] - pts[i, 1]

                    overlap = r[i] + r[j] - d
                    f = overlap / d * 0.5
                    forces[i] -= [f * dx, f * dy]
                    forces[j] += [f * dx, f * dy]

        pts = np.clip(pts, 1e-6, 1.0 - 1e-6)

def make_hex_init(row_counts, r0, angle=0.0):

    y = r0

                idx += 1

        centers[idx, 0] = np.random.uniform(r0, 1.0 - r0)

    centers = np.clip(centers, r0, 1.0 - r0)
    return centers
