import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def solve_lp_radii(centers):
    """Compute optimal radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        bound = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = bound
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dist
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return np.full(n, 0.0)

def objective(x):
    """Objective to minimize: negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Vectorized inequality constraints: g(x) >= 0."""
    n = N
    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]
    
    # Boundary constraints
    c = np.concatenate([
        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs
    ])
    
    # Overlap constraints (squared)
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    i_idx, j_idx = np.tril_indices(n, -1)
    c = np.concatenate([c, dx[i_idx, j_idx]**2 + dy[i_idx, j_idx]**2 - dr[i_idx, j_idx]**2])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [0, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
    return b

def make_hex_init(angle=0.0):
    """Generate a hexagonal lattice initialization."""
    centers = np.zeros((N, 2))
    r0 = 0.095
    y = r0
    row = 0
    idx = 0
    while idx < N and y + r0 <= 1.0:
        start_x = r0 if row % 2 == 0 else 2.0 * r0
        cx = start_x
        while idx < N and cx + r0 <= 1.0:
            centers[idx] = [cx, y]
            idx += 1
            cx += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
    while idx < N:
        centers[idx] = [0.5, 0.5]
        idx += 1
        
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        centers = (centers - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
    return centers

def run_packing():
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_centers = None
    best_radii = None
    best_sum = -1.0
    
    np.random.seed(42)
    
    # Phase 1: Multi-start Hill Climbing + SLSQP Refinement
    for seed in range(12):
        # Diverse hexagonal starts with rotation and perturbation
        centers = make_hex_init(angle=seed * 0.15)
        centers += np.random.uniform(-0.006, 0.006, centers.shape)
        centers = np.clip(centers, 0.01, 0.99)
        
        radii = solve_lp_radii(centers)
        curr_sum = np.sum(radii)
        
        # Phase 1A: Hill Climbing on Centers (evaluated via LP)
        step_size = 0.012
        for step in range(900):
            i = np.random.randint(N)
            old_pos = centers[i].copy()
            
            # Standard step
            new_pos = old_pos + np.random.uniform(-step_size, step_size, 2)
            new_pos = np.clip(new_pos, 0.005, 0.995)
            
            # Occasional larger jump to escape local minima
            if np.random.rand() < 0.025:
                new_pos = old_pos + np.random.uniform(-0.04, 0.04, 2)
                new_pos = np.clip(new_pos, 0.005, 0.995)
                
            centers[i] = new_pos
            new_radii = solve_lp_radii(centers)
            new_sum = np.sum(new_radii)
            
            if new_sum > curr_sum:
                radii = new_radii
                curr_sum = new_sum
            else:
                centers[i] = old_pos
                
            step_size *= 0.9985
            
        # Phase 1B: Smooth NLP Refinement
        x0 = np.zeros(3 * N)
        x0[0::3] = centers[:, 0]
        x0[1::3] = centers[:, 1]
        x0[2::3] = radii
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 6000, 'ftol': 1e-13, 'disp': False})
            if -res.fun > curr_sum:
                curr_sum = -res.fun
                centers = res.x[:2*N].reshape(N, 2)
                radii = res.x[2*N:]
        except Exception:
            pass
            
        if curr_sum > best_sum:
            best_sum = curr_sum
            best_centers = centers.copy()
            best_radii = radii.copy()
            
    # Phase 2: Final strict validation & minimal repair
    centers = best_centers
    radii = best_radii
    
    for _ in range(60):
        valid = True
        for i in range(N):
            if (radii[i] < 0 or centers[i,0] < radii[i]-1e-9 or centers[i,0] > 1-radii[i]+1e-9 or 
                centers[i,1] < radii[i]-1e-9 or centers[i,1] > 1-radii[i]+1e-9):
                valid = False
                break
        if valid:
            for i in range(N):
                for j in range(i+1, N):
                    if np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1]) < radii[i]+radii[j]-1e-9:
                        valid = False
                        break
                if not valid:
                    break
        if valid:
            break
            
        # Gentle shrinkage to guarantee strict compliance
        radii *= 0.9996
        for i in range(N):
            centers[i,0] = np.clip(centers[i,0], radii[i], 1-radii[i])
            centers[i,1] = np.clip(centers[i,1], radii[i], 1-radii[i])
            
    return centers, radii, float(np.sum(radii))
