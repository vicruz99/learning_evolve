from scipy.optimize import minimize, linprog

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Returns all inequality constraints g(x) >= 0 (vectorized)."""
    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]
    
    # Boundary constraints: 4*N
    c = np.concatenate([
        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs
    ])
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [0, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0

        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, d)
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def force_directed_init(n, seed, steps=500):

    np.random.seed(seed)
    centers = np.random.uniform(0.15, 0.85, (n, 2))

        forces = np.zeros_like(centers)

                    forces[j] += f * dx

                    forces[i, dim] += 0.05

                    forces[i, dim] -= 0.05
                    
        lr = 0.05 * (1.0 - step / steps)
        centers += forces * lr
        centers = np.clip(centers, 0.02, 0.98)
        
    return centers

def generate_hex_init(rows, r0):
    """Generates a hexagonal lattice initialization."""
    centers = np.zeros((N, 2))
    idx = 0
    y = r0
    y_step = np.sqrt(3.0) * r0
    for i, cnt in enumerate(rows):
        x_start = r0 + (r0 if i % 2 == 1 else 0.0)
        for k in range(cnt):
            if idx < N:
                centers[idx, 0] = x_start + k * 2.0 * r0

                idx += 1
        y += y_step
    while idx < N:
        centers[idx] = np.random.uniform(0.1, 0.9, 2)
        idx += 1
    return centers

def rotate_grid(centers, angle):
    """Rotates a grid of centers around the square center."""
    c, s = np.cos(angle), np.sin(angle)
    pts = centers - 0.5
    new_pts = np.column_stack((pts[:, 0]*c - pts[:, 1]*s, pts[:, 0]*s + pts[:, 1]*c)) + 0.5
    return new_pts
