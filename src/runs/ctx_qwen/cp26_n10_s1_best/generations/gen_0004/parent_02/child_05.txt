import numpy as np
from scipy.optimize import minimize, linprog

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Returns all inequality constraints >= 0 (vectorized)."""
    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]
    
    # Boundary constraints
    c = np.concatenate([
        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs
    ])
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    c = np.concatenate([c, dx[TRIL_IDX]**2 + dy[TRIL_IDX]**2 - dr[TRIL_IDX]**2])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-7, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)])
    return b

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A = np.zeros((num_ineq, n))
    b = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A[idx, i] = 1.0
        b[idx] = max(0.0, lim)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = max(0.0, dist)
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def make_x0(centers, radii):
    """Flatten centers and radii into optimization vector."""
    x = np.zeros(3 * N)
    x[0::3] = centers[:, 0]
    x[1::3] = centers[:, 1]
    x[2::3] = radii
    return x

def project_x(x):
    """Project variables to strictly satisfy bounds and non-negative radii."""
    x = x.copy()
    for i in range(N):
        r = max(1e-7, x[3*i+2])
        x[3*i] = np.clip(x[3*i], r, 1.0 - r)
        x[3*i+1] = np.clip(x[3*i+1], r, 1.0 - r)
        x[3*i+2] = r
    return x

def hex_init(r0, angle, seed=0):
    """Generate hexagonal lattice initialization with rotation."""
    rng = np.random.RandomState(seed)
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 15:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        if y > 1.0 + r0:
            break
            
    pts = np.array(pts[:N+15])
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
        
    mask = (pts[:, 0] >= 0) & (pts[:, 0] <= 1) & (pts[:, 1] >= 0) & (pts[:, 1] <= 1)
    pts = pts[mask]
    while len(pts) < N:
        pts = np.vstack([pts, rng.uniform(0.15, 0.85, 2)])
        
    return pts[:N] + rng.uniform(-0.001, 0.001, (N, 2))

def force_init(seed, steps=400):
    """Generate initial configuration using repulsive force simulation."""
    rng = np.random.RandomState(seed)
    pts = rng.uniform(0.2, 0.8, (N, 2))
    
    for _ in range(steps):
        forces = np.zeros_like(pts)
        for i in range(N):
            for j in range(i + 1, N):
                dx = pts[j] - pts[i]
                d = np.hypot(dx[0], dx[1])
                if d < 0.3 and d > 1e-6:
                    f = 0.008 / (d**2 + 0.001)
                    forces[i] -= f * dx
                    forces[j] += f * dx
                    
            for dim in range(2):
                if pts[i, dim] < 0.12:
                    forces[i, dim] += 0.03
                elif pts[i, dim] > 0.88:
                    forces[i, dim] -= 0.03
                    
        pts += forces * 0.04
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -1.0
    best_x = None
    
    # Phase 1: Generate diverse initial configurations
    inits = []
    for r0 in np.linspace(0.085, 0.105, 7):
        for ang in np.linspace(0.0, np.pi / 6, 9):
            inits.append(hex_init(r0, ang))
            
    for s in range(12):
        inits.append(force_init(s))
        
    # Specific row patterns known to perform well in squares
    patterns = [[5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 6, 4], [4, 6, 6, 6, 4]]
    for pat in patterns:
        pts = []
        y = 0.08
        r0 = 0.09
        for row, cnt in enumerate(pat):
            x_start = 0.5 - (cnt - 1) * r0
            x = x_start
            for _ in range(cnt):
                if len(pts) < N:
                    pts.append([x, y])
                    x += 2.0 * r0
            y += np.sqrt(3.0) * r0
        inits.append(np.array(pts[:N]))
        
    np.random.seed(42)
    for c_init in inits:
        # Snap radii to LP optimum for these centers
        r_lp = solve_lp_radii(c_init)
        if r_lp is None:
            r_lp = np.full(N, 0.08)
            
        x0 = project_x(make_x0(c_init, r_lp))
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 12000, 'ftol': 1e-14, 'disp': False})
            if -res.fun > best_sum:
                vals = constraints(res.x)
                if np.min(vals) >= -1e-6:
                    best_sum = -res.fun
                    best_x = res.x.copy()
        except Exception:
            continue
            
    # Phase 2: Deflation & Perturbation Refinement
    if best_x is not None:
        for step in range(25):
            x0 = best_x.copy()
            # Shrink radii to free up space for topological rearrangement
            x0[2::3] *= 0.982
            # Perturb centers
            noise = 0.0025 * (0.92 ** step)
            x0 += np.random.normal(0, noise, 3 * N)
            x0 = project_x(x0)
            
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 8000, 'ftol': 1e-14, 'disp': False})
                if -res.fun > best_sum:
                    vals = constraints(res.x)
                    if np.min(vals) >= -1e-6:
                        best_sum = -res.fun
                        best_x = res.x.copy()
                        
                # LP snap to maximize radii for new centers
                centers_cur = best_x.reshape(N, 3)[:, :2]
                r_new = solve_lp_radii(centers_cur)
                if r_new is not None:
                    best_x[2::3] = r_new
                    if np.sum(r_new) > best_sum:
                        best_sum = np.sum(r_new)
            except Exception:
                pass
                
        # Aggressive scaling attempts to push boundaries
        for _ in range(8):
            x0 = best_x.copy()
            x0[2::3] *= 1.005
            x0 = project_x(x0)
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 6000, 'ftol': 1e-14, 'disp': False})
                if -res.fun > best_sum:
                    vals = constraints(res.x)
                    if np.min(vals) >= -1e-6:
                        best_sum = -res.fun
                        best_x = res.x.copy()
            except Exception:
                pass

    # Fallback safety
    if best_x is None:
        best_x = project_x(make_x0(hex_init(0.09, 0.0), np.full(N, 0.08)))
        best_sum = -objective(best_x)
        
    centers = best_x.reshape(N, 3)[:, :2]
    radii = best_x[2::3]
    
    # Final strict validation & minimal repair
    for _ in range(100):
        valid = True
        for i in range(N):
            if radii[i] < 0 or centers[i, 0] < radii[i] - 1e-9 or centers[i, 0] > 1.0 - radii[i] + 1e-9 or \
               centers[i, 1] < radii[i] - 1e-9 or centers[i, 1] > 1.0 - radii[i] + 1e-9:
                valid = False
                break
        if valid:
            for i in range(N):
                for j in range(i + 1, N):
                    d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                    if d < radii[i] + radii[j] - 1e-9:
                        valid = False
                        break
                if not valid:
                    break
        if valid:
            break
            
        # Gentle shrinkage to guarantee strict compliance
        radii *= 0.9995
        for i in range(N):
            centers[i, 0] = np.clip(centers[i, 0], radii[i], 1.0 - radii[i])
            centers[i, 1] = np.clip(centers[i, 1], radii[i], 1.0 - radii[i])
            
    return centers, radii, float(np.sum(radii))
