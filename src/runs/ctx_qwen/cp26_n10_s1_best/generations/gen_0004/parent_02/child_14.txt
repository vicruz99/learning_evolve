1. **Constraint Conditioning**: The squared distance constraint `dist^2 - (r_i + r_j)^2 >= 0` has a gradient that vanishes when circles touch, making it hard for gradient-based optimizers to push them further apart. Switching to the direct distance constraint `dist - (r_i + r_j) >= 0` provides a constant gradient magnitude near contact, significantly improving SLSQP's ability to expand radii.

2. **LP-Optimized Radii**: For any fixed set of centers, the optimal radii that maximize the sum can be found exactly via Linear Programming. This avoids the nonlinearity of radius optimization and guarantees maximal radii for the current topology.

3. **Iterative Topology Search**: We alternate between:

   - Perturbing centers (deflating radii slightly to create clearance) and re-optimizing jointly with SLSQP.

4. **Diverse Initialization**: We generate starts from rotated hexagonal lattices (covering many dense topologies) and force-spread random clusters anchored to corners/edges. This ensures we explore both interior-dense and boundary-utilizing arrangements.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def get_lp_radii(centers):
    """Solve for optimal radii given fixed centers using Linear Programming."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], 
                  centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], 
                         centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = d
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def constraints_vec(x):
    """Vectorized inequality constraints: g(x) >= 0."""
    cx = x[0::3]
    cy = x[1::3]
    cr = x[2::3]
    
    c = np.concatenate([
        cx - cr, 1.0 - cx - cr,
        cy - cr, 1.0 - cy - cr
    ])
    
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dr = cr[:, None] + cr[None, :]
    dist = np.hypot(dx, dy)
    
    i_idx, j_idx = np.tril_indices(N, -1)
    c = np.concatenate([c, dist[i_idx, j_idx] - dr[i_idx, j_idx]])
    return c

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [0, 0.5]."""
    return [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N

def make_hex_init(r0, angle, seed):
    """Generate a rotated hexagonal lattice initialization."""
    np.random.seed(seed)
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 15:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0 + 0.02:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        if y > 1.0 + r0:
            break
            
    pts = np.array(pts[:N+15])
    center = np.array([0.5, 0.5])
    pts -= center
    ca, sa = np.cos(angle), np.sin(angle)
    pts = pts @ np.array([[ca, -sa], [sa, ca]]) + center
    
    mask = (pts[:,0]>=0.02) & (pts[:,0]<=0.98) & (pts[:,1]>=0.02) & (pts[:,1]<=0.98)
    pts = pts[mask]
    
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.2, 0.8, (1, 2))])
    pts = pts[:N]
    pts += np.random.uniform(-0.003, 0.003, pts.shape)
    return pts

def make_force_init(seed):
    """Force-directed spread initialization with boundary anchoring."""
    np.random.seed(seed)
    pts = np.random.uniform(0.2, 0.8, (N, 2))
    for _ in range(400):
        f = np.zeros_like(pts)
        for i in range(N):
            for j in range(i+1, N):
                d = np.hypot(pts[i,0]-pts[j,0], pts[i,1]-pts[j,1])
                if d < 0.25 and d > 1e-6:
                    v = pts[i] - pts[j]
                    f[i] += v/d
                    f[j] -= v/d
            for dim in range(2):
                if pts[i,dim] < 0.15: f[i,dim] += 0.04
                elif pts[i,dim] > 0.85: f[i,dim] -= 0.04
        pts += f * 0.03
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints_vec}
    
    best_sum = -1.0
    best_x = None
    
    # Phase 1: Diverse Initializations
    starts = []
    for r0 in [0.085, 0.090, 0.095, 0.100]:
        for ang in [0.0, 0.15, 0.30, 0.50, 0.70, 1.00]:
            starts.append(make_hex_init(r0, ang, seed=42))
            
    for s in range(8):
        starts.append(make_force_init(seed=s))

    for c_init in starts:
        r_lp = get_lp_radii(c_init)
        if r_lp is None:
            r_lp = np.full(N, 0.06)
            
        x0 = np.zeros(3 * N)
        x0[0::3] = c_init[:, 0]
        x0[1::3] = c_init[:, 1]
        x0[2::3] = r_lp
        
        x0 += np.random.normal(0, 1e-4, 3 * N)
        
        for i in range(N):
            r = max(0.0, x0[3*i+2])
            x0[3*i] = np.clip(x0[3*i], r, 1.0 - r)
            x0[3*i+1] = np.clip(x0[3*i+1], r, 1.0 - r)
            x0[3*i+2] = r
            
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, 
                           constraints=cons, options={'maxiter': 20000, 'ftol': 1e-13, 'disp': False})
            
            curr_sum = -res.fun
            c_vals = constraints_vec(res.x)
            if np.min(c_vals) >= -1e-6 and curr_sum > best_sum:
                best_sum = curr_sum
                best_x = res.x.copy()
        except Exception:
            pass

    # Phase 2: Iterative Deflation-Perturbation-Refinement
    if best_x is not None:
        for round_idx in range(40):
            x0 = best_x.copy()
            x0[2::3] *= 0.982
            
            noise = 0.0012 * (0.94 ** round_idx)
            x0[0::3] += np.random.normal(0, noise, N)
            x0[1::3] += np.random.normal(0, noise, N)
            
            for i in range(N):
                r = max(0.005, x0[3*i+2])
                x0[3*i] = np.clip(x0[3*i], r, 1.0 - r)
                x0[3*i+1] = np.clip(x0[3*i+1], r, 1.0 - r)
                x0[3*i+2] = r
                
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 15000, 'ftol': 1e-13, 'disp': False})
                
                if -res.fun > best_sum:
                    best_x = res.x.copy()
                    best_sum = -res.fun
                    
                c_opt = best_x.reshape(N, 3)[:, :2]
                r_lp = get_lp_radii(c_opt)
                if r_lp is not None:
                    lp_sum = np.sum(r_lp)
                    if lp_sum > best_sum:
                        best_x[2::3] = r_lp
                        best_sum = lp_sum
            except Exception:
                pass
                
        # Final LP sweep
        c_final = best_x.reshape(N, 3)[:, :2]
        r_final = get_lp_radii(c_final)
        if r_final is not None:
            best_x[2::3] = r_final
            best_sum = np.sum(r_final)

    if best_x is None:
        # Fallback
        c_fallback = make_hex_init(0.09, 0.0, seed=0)
        r_fallback = get_lp_radii(c_fallback) or np.full(N, 0.08)
        best_x = np.zeros(3*N)
        best_x[0::3] = c_fallback[:,0]
        best_x[1::3] = c_fallback[:,1]
        best_x[2::3] = r_fallback
        best_sum = np.sum(r_fallback)

    centers = best_x.reshape(N, 3)[:, :2]
    radii = best_x[2::3]
    
    # Strict numerical repair
    for _ in range(100):
        valid = True
        for i in range(N):
            if radii[i] < 0 or centers[i,0] < radii[i]-1e-9 or centers[i,0] > 1-radii[i]+1e-9 or \
               centers[i,1] < radii[i]-1e-9 or centers[i,1] > 1-radii[i]+1e-9:
                valid = False; break
        if valid:
            for i in range(N):
                for j in range(i+1, N):
                    if np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1]) < radii[i]+radii[j]-1e-9:
                        valid = False; break
                if not valid: break
        if valid: break
        
        radii *= 0.9998
        for i in range(N):
            centers[i,0] = np.clip(centers[i,0], radii[i], 1-radii[i])
            centers[i,1] = np.clip(centers[i,1], radii[i], 1-radii[i])
            
    return centers, radii, float(np.sum(radii))
