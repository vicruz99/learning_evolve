

Looking at the problem, I need to maximize the sum of radii for 26 circles in a unit square. The current best is 2.634292, and I'm targeting 2.636.

My strategy:
1. Generate far more diverse initial configurations with systematic rotation angles and varied hexagonal lattice patterns
2. Use LP to optimally compute radii for any fixed centers (mathematically exact)
3. Implement alternating optimization: SLSQP for positions → LP for radii → repeat
4. Aggressive multi-phase refinement with decreasing perturbation scales
5. Better force-directed layouts with stronger boundary pushing

```python
import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Returns all inequality constraints >= 0 (vectorized)."""
    n = N
    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]
    
    c = np.concatenate([
        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs
    ])
    
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    i_idx, j_idx = np.tril_indices(n, -1)
    dist_sq = dx[i_idx, j_idx]**2 + dy[i_idx, j_idx]**2
    r_sum_sq = dr[i_idx, j_idx]**2
    
    c = np.concatenate([c, dist_sq - r_sum_sq])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [0, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
    return b

def solve_lp_radii(centers):
    """Optimally scale radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    for i in range(n):
        x_val, y_val = centers[i]
        bound = min(x_val, 1.0 - x_val, y_val, 1.0 - y_val)
        A_ub[idx, i] = 1.0
        b_ub[idx] = bound
        idx += 1
        
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dist
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def generate_hex_centers(r0, row_counts):
    """Generate hexagonal lattice centers with specified row counts."""
    centers = np.zeros((N, 2))
    idx = 0
    y = r0
    y_step = np.sqrt(3.0) * r0
    for row_idx, count in enumerate(row_counts):
        x_start = r0 + (r0 if row_idx % 2 == 1 else 0.0)
        for k in range(count):
            if idx < N:
                centers[idx, 0] = x_start + k * 2.0 * r0
                centers[idx, 1] = y
                idx += 1
        y += y_step
    while idx < N:
        centers[idx, 0] = np.random.uniform(0.1, 0.9)
        centers[idx, 1] = np.random.uniform(0.1, 0.9)
        idx += 1
    return centers

def rotate_centers(centers, angle_deg):
    """Rotate centers around (0.5, 0.5)."""
    ang = np.deg2rad(angle_deg)
    c, s = np.cos(ang), np.sin(ang)
    cx, cy = 0.5, 0.5
    dx = centers[:, 0] - cx
    dy = centers[:, 1] - cy
    new_centers = np.zeros_like(centers)
    new_centers[:, 0] = dx * c - dy * s + cx
    new_centers[:, 1] = dx * s + dy * c + cy
    return new_centers

def force_directed_init(n, seed, steps=800):
    """Force-directed relaxation to spread points evenly and push to boundaries."""
    np.random.seed(seed)
    centers = np.random.uniform(0.12, 0.88, (n, 2))
    
    for step in range(steps):
        forces = np.zeros_like(centers)
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[j] - centers[i]
                dist = np.linalg.norm(dx)
                if dist < 0.22 and dist > 1e-6:
                    f = 0.02 / (dist**2 + 0.0005)
                    forces[i] -= f * dx
                    forces[j] += f * dx
                    
            for dim in range(2):
                if centers[i, dim] < 0.08:
                    forces[i, dim] += 0.15 * (0.08 - centers[i, dim])
                elif centers[i, dim] > 0.92:
                    forces[i, dim] -= 0.15 * (centers[i, dim] - 0.92)
                    
        learning_rate = 0.04 * max(0.1, 1.0 - step / steps)
        centers += forces * learning_rate
        centers = np.clip(centers, 0.03, 0.97)
        
    return centers

def run_packing():
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -1.0
    best_x = None
    
    # Phase 1: Generate diverse initial configurations
    inits = []
    
    # Hexagonal lattice patterns with various row counts
    patterns = [
        [5, 6, 5, 6, 4],
        [6, 5, 6, 5, 4],
        [5, 6, 6, 5, 4],
        [4, 6, 6, 6, 4],
        [5, 5, 6, 5, 5],
        [6, 6, 5, 5, 4],
        [4, 5, 6, 6, 5],
        [5, 7, 5, 6, 3],
        [3, 6, 6, 6, 5],
        [6, 6, 6, 4, 4],
        [5, 6, 4, 6, 5],
        [4, 6, 5, 6, 5],
        [5, 5, 5, 6, 5],
        [6, 4, 6, 6, 4],
        [4, 5, 5, 6, 6],
    ]
    
    for pat in patterns:
        for r0 in [0.082, 0.088, 0.094, 0.100]:
            c = generate_hex_centers(r0, pat)
            for ang in [0.0, 5.0, 10.0, 15.0, 20.0, 25.0, 30.0, 35.0, 40.0, 45.0]:
                c_rot = rotate_centers(c, ang)
                mask = (c_rot[:, 0] >= 0.02) & (c_rot[:, 0] <= 0.98) & \
                       (c_rot[:, 1] >= 0.02) & (c_rot[:, 1] <= 0.98)
                if np.sum(mask) >= N:
                    inits.append(c_rot[mask][:N].copy())
                else:
                    c_rot2 = c_rot.copy()
                    bad_idx = np.where(~mask)[0]
                    for idx in bad_idx:
                        c_rot2[idx, 0] = np.random.uniform(0.15, 0.85)
                        c_rot2[idx, 1] = np.random.uniform(0.15, 0.85)
                    inits.append(c_rot2[:N].copy())
    
    # Force-directed initializations
    for s in range(12):
        inits.append(force_directed_init(N, seed=s))
    
    # Optimize each initialization
    for c_init in inits:
        # LP for initial radii
        r_lp = solve_lp_radii(c_init)
        if r_lp is None:
            r_lp = np.full(N, 0.08)
        
        x0 = np.zeros(3 * N)
        x0[0::3] = c_init[:, 0]
        x0[1::3] = c_init[:, 1]
        x0[2::3] = r_lp
        
        # Ensure feasible starting point
        for i in range(N):
            r = max(0.001, x0[3*i+2])
            x0[3*i] = np.clip(x0[3*i], r, 1.0 - r)
            x0[3*i+1] = np.clip(x0[3*i+1], r, 1.0 - r)
            x0[3*i+2] = r
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, 
                           constraints=cons, 
                           options={'maxiter': 12000, 'ftol': 1e-14, 'disp': False})
            
            curr_sum = -res.fun
            vals = constraints(res.x)
            if np.min(vals) >= -1e-6 and curr_sum > best_sum:
                best_sum = curr_sum
                best_x = res.x.copy()
        except Exception:
            pass
    
    # Phase 2: Alternating SLSQP and LP refinement
    if best_x is not None:
        for round_idx in range(20):
            centers_cur = np.column_stack((best_x[0::3], best_x[1::3]))
            r_lp = solve_lp_radii(centers_cur)
            if r_lp is not None:
                best_x[2::3] = r_lp
                curr_sum = np.sum(r_lp)
                if curr_sum > best_sum:
                    best_sum = curr_sum
            
            # SLSQP re-optimization
            x0 = best_x.copy()
            x0 += np.random.normal(0, 5e-6, 3 * N)
            for i in range(N):
                r = max(0.001, x0[3*i+2])
                x0[3*i] = np.clip(x0[3*i], r, 1.0 - r)
                x0[3*i+1] = np.clip(x0[3*i+1], r, 1.0 - r)
                x0[3*i+2] = r
            
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                               constraints=cons, 
                               options={'maxiter': 6000, 'ftol': 1e-14, 'disp': False})
                if -res.fun > best_sum:
                    best_x = res.x.copy()
                    best_sum = -res.fun
            except Exception:
                pass
    
    # Phase 3: Deflation + Perturbation + Re-optimization (escape local minima)
    if best_x is not None:
        for round_idx in range(40):
            x0 = best_x.copy()
            # Deflate radii slightly
            x0[2::3] *= 0.99
            
            # Perturb with decreasing noise
            noise_scale = 0.003 * (0.90 ** round_idx)
            x0 += np.random.normal(0, noise_scale, 3 * N)
            
            for i in range(N):
                r = max(0.005, x0[3*i+2])
                x0[3*i] = np.clip(x0[3*i], r, 1.0 - r)
                x0[3*i+1] = np.clip(x0[3*i+1], r, 1.0 - r)
                x0[3*i+2] = r
            
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                               constraints=cons, 
                               options={'maxiter': 10000, 'ftol': 1e-14, 'disp': False})
                
                if -res.fun > best_sum:
                    best_x = res.x.copy()
                    best_sum = -res.fun
                    
                    # LP refinement after finding better solution
                    centers_cur = np.column_stack((best_x[0::3], best_x[1::3]))
                    r_lp = solve_lp_radii(centers_cur)
                    if r_lp is not None:
                        best_x[2::3] = r_lp
                        curr_sum = np.sum(r_lp)
                        if curr_sum > best_sum:
                            best_sum = curr_sum
            except Exception:
                pass
    
    # Phase 4: Ultra-fine refinement
    if best_x is not None:
        for _ in range(15):
            x0 = best_x.copy()
            x0 += np.random.normal(0, 1e-6, 3 * N)
            for i in range(N):
                r = max(0.001, x0[3*i+2])
                x0[3*i] = np.clip(x0[3*i], r, 1.0 - r)
                x0[3*i+1] = np.clip(x0[3*i+1], r, 1.0 - r)
                x0[3*i+2] = r
            
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                               constraints=cons, 
                               options={'maxiter': 5000, 'ftol': 1e-15, 'disp': False})
                if -res.fun > best_sum:
                    best_x = res.x.copy()
                    best_sum = -res.fun
            except Exception:
                pass
    
    # Extract results
    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3].copy()
    
    # Final safety adjustment to guarantee strict compliance
    for _ in range(200):
        valid = True
        for i in range(N):
            if radii[i] < 0 or centers[i,0] < radii[i]-1e-9 or centers[i,0] > 1-radii[i]+1e-9 or \
               centers[i,1] < radii[i]-1e-9 or centers[i,1] > 1-radii[i]+1e-9:
                valid = False
                break
        if valid:
            for i in range(N):
                for j in range(i+1, N):
                    if np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1]) < radii[i]+radii[j]-1e-9:
                        valid = False
                        break
                if not valid:
                    break
        if valid:
            break
            
        radii *= 0.9995
        for i in range(N):
            centers[i,0] = np.clip(centers[i,0], radii[i], 1-radii[i])
            centers[i,1] = np.clip(centers[i,1], radii[i], 1-radii[i])
            
    return centers, radii, float(np.sum(radii))
```
