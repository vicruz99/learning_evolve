import numpy as np
from scipy.optimize import linprog, minimize
import random

N = 26

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    for i in range(n):
        x, y = centers[i]
        bound = min(x, 1.0 - x, y, 1.0 - y)
        b_ub[idx] = max(0.0, bound)
        A_ub[idx, i] = 1.0
        idx += 1
        
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = d
            idx += 1
            
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None

def run_sa(centers, steps=20000):
    """Simulated Annealing over center positions, optimizing radii via LP at each step."""
    curr_radii = solve_lp_radii(centers)
    if curr_radii is None:
        return centers, 0.0, None
    curr_sum = np.sum(curr_radii)
    
    best_centers = centers.copy()
    best_sum = curr_sum
    best_radii = curr_radii.copy()
    
    temp = 0.02
    for step in range(steps):
        i = np.random.randint(N)
        move_scale = 0.007 * np.sqrt(temp)
        
        new_centers = centers.copy()
        new_centers[i, 0] += np.random.uniform(-move_scale, move_scale)
        new_centers[i, 1] += np.random.uniform(-move_scale, move_scale)
        new_centers[i] = np.clip(new_centers[i], 0.005, 0.995)
        
        new_radii = solve_lp_radii(new_centers)
        if new_radii is not None:
            new_sum = np.sum(new_radii)
            delta = new_sum - curr_sum
            
            if delta > 0 or random.random() < np.exp(delta / temp):
                centers = new_centers
                curr_sum = new_sum
                curr_radii = new_radii
                
                if curr_sum > best_sum:
                    best_centers = centers.copy()
                    best_sum = curr_sum
                    best_radii = curr_radii.copy()
                    
        temp *= 0.9991
        if temp < 2e-5:
            break
            
    return best_centers, best_sum, best_radii

def generate_hex(r0, angle=0.0):
    """Generate hexagonal lattice points, optionally rotated."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        if y > 1.0 + r0:
            break
            
    pts = np.array(pts[:N + 10])
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        center = np.array([0.5, 0.5])
        pts = (pts - center) @ np.array([[c, -s], [s, c]]) + center
        
    mask = (pts[:, 0] >= 0.01) & (pts[:, 0] <= 0.99) & (pts[:, 1] >= 0.01) & (pts[:, 1] <= 0.99)
    pts = pts[mask]
    
    while len(pts) < N:
        pts = np.vstack([pts, [np.random.uniform(0.2, 0.8), np.random.uniform(0.2, 0.8)]])
        
    return np.clip(pts[:N], 0.02, 0.98)

def run_packing():
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    random.seed(42)
    
    best_c = None
    best_r = None
    best_sum = -1.0
    
    # Diverse initial configurations
    starts = []
    for r0 in [0.085, 0.090, 0.095, 0.100]:
        for ang in [0.0, 0.1, 0.2, 0.35, 0.5]:
            starts.append(generate_hex(r0, ang))
    for _ in range(5):
        starts.append(np.random.uniform(0.1, 0.9, (N, 2)))
        
    # Run SA + LP on each start
    for init_c in starts:
        c, s, r = run_sa(init_c, steps=15000)
        if s > best_sum:
            best_sum = s
            best_c = c
            best_r = r
            
    # Final SLSQP polish to satisfy constraints precisely
    def objective(x):
        return -np.sum(x[2::3])
        
    def constraints(x):
        cx, cy, cr = x[0::3], x[1::3], x[2::3]
        c = np.concatenate([cx - cr, 1.0 - cx - cr, cy - cr, 1.0 - cy - cr])
        dx = cx[:, None] - cx[None, :]
        dy = cy[:, None] - cy[None, :]
        dr = cr[:, None] + cr[None, :]
        i, j = np.tril_indices(N, -1)
        c = np.concatenate([c, dx[i, j]**2 + dy[i, j]**2 - dr[i, j]**2])
        return c
        
    if best_c is not None:
        x0 = np.zeros(3 * N)
        x0[0::3] = best_c[:, 0]
        x0[1::3] = best_c[:, 1]
        x0[2::3] = best_r
        
        try:
            res = minimize(objective, x0, method='SLSQP',
                           constraints={'type': 'ineq', 'fun': constraints},
                           bounds=[(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N,
                           options={'maxiter': 15000, 'ftol': 1e-14, 'disp': False})
            if not np.isnan(res.fun) and -res.fun > best_sum:
                best_c = np.column_stack((res.x[0::3], res.x[1::3]))
                best_r = res.x[2::3]
                best_sum = -res.fun
        except Exception:
            pass
            
    # Strict validation and minimal numerical repair
    for _ in range(100):
        valid = True
        for i in range(N):
            if (best_r[i] < 0 or best_c[i, 0] < best_r[i] - 1e-9 or best_c[i, 0] > 1.0 - best_r[i] + 1e-9 or
                best_c[i, 1] < best_r[i] - 1e-9 or best_c[i, 1] > 1.0 - best_r[i] + 1e-9):
                valid = False
                break
        if valid:
            for i in range(N):
                for j in range(i + 1, N):
                    d = np.hypot(best_c[i, 0] - best_c[j, 0], best_c[i, 1] - best_c[j, 1])
                    if d < best_r[i] + best_r[j] - 1e-9:
                        valid = False
                        break
                if not valid:
                    break
        if valid:
            break
            
        best_r *= 0.9995
        best_c[:, 0] = np.clip(best_c[:, 0], best_r, 1.0 - best_r)
        best_c[:, 1] = np.clip(best_c[:, 1], best_r, 1.0 - best_r)
        
    return best_c, best_r, float(np.sum(best_r))
