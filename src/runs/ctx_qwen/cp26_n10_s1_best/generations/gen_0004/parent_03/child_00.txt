import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Compute all boundary and non-overlap constraints as a vector >= 0."""
    cx = x[0::3]
    cy = x[1::3]
    cr = x[2::3]
    
    # Boundary constraints: 4 * N
    c = np.concatenate([
        cx - cr, 1.0 - cx - cr,
        cy - cr, 1.0 - cy - cr
    ])
    
    # Overlap constraints: N*(N-1)/2 using squared distances
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dr = cr[:, None] + cr[None, :]
    
    idx = np.tril_indices(N, -1)
    c = np.concatenate([c, dx[idx]**2 + dy[idx]**2 - dr[idx]**2])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [0, 0.5]."""
    return [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)  # Clamp to avoid infeasibility from numerical drift
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dist
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def make_hex_init(pattern, r0, angle=0.0, seed=0):
    """Generate a hexagonal lattice initialization with specified row counts."""
    np.random.seed(seed)
    centers = []
    y = r0
    for i, cnt in enumerate(pattern):
        # Stagger odd rows by r0
        shift = r0 if i % 2 == 1 else 0.0
        # Center the row horizontally
        row_width = (cnt - 1) * 2.0 * r0
        start_x = (1.0 - row_width) / 2.0 + shift
        
        for k in range(cnt):
            x = start_x + k * 2.0 * r0
            centers.append([x, y])
        y += np.sqrt(3.0) * r0
        
    centers = np.array(centers)
    if len(centers) < N:
        pad = N - len(centers)
        centers = np.vstack([centers, np.random.uniform(0.1, 0.9, (pad, 2))])
    centers = centers[:N]
    
    if angle != 0.0:
        c = np.array([0.5, 0.5])
        centers -= c
        ca, sa = np.cos(angle), np.sin(angle)
        centers = centers @ np.array([[ca, -sa], [sa, ca]])
        centers += c
        
    # Small perturbation to break exact symmetry
    centers += np.random.uniform(-0.004, 0.004, centers.shape)
    centers = np.clip(centers, 0.02, 0.98)
    
    radii = np.full(N, r0 * 0.94)
    return centers, radii
