import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def get_lp_radii(centers):
    """Compute optimal radii for fixed centers using Linear Programming."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A = np.zeros((num_ineq, n))
    b = np.zeros(num_ineq)
    idx = 0
    
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A[idx, i] = 1.0
        b[idx] = max(0.0, lim)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = d
            idx += 1
            
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0.0, None)] * n, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return np.full(n, 0.01)

def optimize_clearance(centers, radii):
    """Maximize minimum clearance (slack) for fixed radii using Nonlinear Optimization."""
    n = len(radii)
    
    def obj(vars):
        # We maximize slack s => minimize -s
        return -vars[2 * n]

    def constr(vars):
        cx = vars[:n]
        cy = vars[n:2 * n]
        s = vars[2 * n]
        
        # Boundary slacks >= s
        c = np.concatenate([
            cx - radii - s,
            1.0 - cx - radii - s,
            cy - radii - s,
            1.0 - cy - radii - s
        ])
        
        # Overlap slacks >= s
        dx = cx[:, None] - cx[None, :]
        dy = cy[:, None] - cy[None, :]
        d = np.sqrt(dx ** 2 + dy ** 2)
        dr = radii[:, None] + radii[None, :]
        i_idx, j_idx = np.tril_indices(n, -1)
        o = d[i_idx, j_idx] - dr[i_idx, j_idx] - s
        
        return np.concatenate([c, o])

    x0 = np.zeros(2 * n + 1)
    x0[:n] = centers[:, 0]
    x0[n:2 * n] = centers[:, 1]
    x0[2 * n] = -0.01  # Start with slightly negative slack to guarantee feasibility
    
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)] * n + [(-0.5, 0.5)]
    cons = {'type': 'ineq', 'fun': constr}
    
    try:
        res = minimize(obj, x0, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 10000, 'ftol': 1e-13, 'disp': False})
        if not np.isnan(res.fun):
            return np.column_stack((res.x[:n], res.x[n:2 * n]))
    except Exception:
        pass
    return centers

def generate_hex_init(seed, density=0.09, angle=0.0):
    """Generate a rotated hexagonal lattice initialization."""
    np.random.seed(seed)
    pts = []
    y = density
    row = 0
    while len(pts) < N + 15:
        x = density + (row % 2) * density
        while x <= 1.0 - density:
            pts.append([x, y])
            x += 2.0 * density
        y += np.sqrt(3.0) * density
        row += 1
        
    pts = np.array(pts[:N + 15])
    center = np.array([0.5, 0.5])
    pts -= center
    ang = np.deg2rad(angle)
    rot = np.array([[np.cos(ang), -np.sin(ang)], [np.sin(ang), np.cos(ang)]])
    pts = pts @ rot.T + center
    
    # Filter points safely inside the square
    mask = (pts[:, 0] >= 0.03) & (pts[:, 0] <= 0.97) & (pts[:, 1] >= 0.03) & (pts[:, 1] <= 0.97)
    pts = pts[mask]
    
    if len(pts) < N:
        pad = N - len(pts)
        pts = np.vstack([pts, np.random.uniform(0.15, 0.85, (pad, 2))])
    pts = pts[:N]
    
    # Break exact symmetry
    pts += np.random.uniform(-0.002, 0.002, pts.shape)
    return pts

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    best_sum = -1.0
    best_c = None
    best_r = None

    # Phase 1: Alternating LP/NL optimization with diverse starts
    configs = []
    for s in range(12):
        configs.append(generate_hex_init(s, density=0.095, angle=0.0))
    for s in range(10):
        configs.append(generate_hex_init(s, density=0.090, angle=20.0))
    for s in range(8):
        configs.append(generate_hex_init(s, density=0.100, angle=45.0))
    for s in range(5):
        configs.append(generate_hex_init(s, density=0.085, angle=10.0))

    for init_c in configs:
        c_curr = init_c.copy()
        r_curr = np.full(N, 0.08)

        # Alternate: LP finds max radii, NLO pushes centers apart
        for _ in range(12):
            r_curr = get_lp_radii(c_curr)
            c_curr = optimize_clearance(c_curr, r_curr)
            c_curr = np.clip(c_curr, 0.0, 1.0)

        curr_sum = np.sum(r_curr)
        if curr_sum > best_sum:
            best_sum = curr_sum
            best_c = c_curr.copy()
            best_r = r_curr.copy()

    # Phase 2: Perturbation & Refinement to escape local minima
    if best_c is not None:
        for it in range(18):
            noise = 0.0045 * (0.88 ** it)
            c_pert = best_c + np.random.normal(0, noise, best_c.shape)
            c_pert = np.clip(c_pert, 0.02, 0.98)
            r_pert = best_r * 0.982

            for _ in range(10):
                r_pert = get_lp_radii(c_pert)
                c_pert = optimize_clearance(c_pert, r_pert)
                c_pert = np.clip(c_pert, 0.0, 1.0)

            if np.sum(r_pert) > best_sum:
                best_sum = np.sum(r_pert)
                best_c = c_pert.copy()
                best_r = r_pert.copy()

    # Phase 3: Final Joint SLSQP Polish to handle coupling effects
    def obj_joint(x):
        return -np.sum(x[2::3])

    def cons_joint(x):
        cx = x[0::3]
        cy = x[1::3]
        cr = x[2::3]
        c = np.concatenate([
            cx - cr, 1.0 - cx - cr, cy - cr, 1.0 - cy - cr
        ])
        dx = cx[:, None] - cx[None, :]
        dy = cy[:, None] - cy[None, :]
        dr = cr[:, None] + cr[None, :]
        i_idx, j_idx = np.tril_indices(N, -1)
        c = np.concatenate([c, dx[i_idx, j_idx]**2 + dy[i_idx, j_idx]**2 - dr[i_idx, j_idx]**2])
        return c

    x0 = np.zeros(3 * N)
    x0[0::3] = best_c[:, 0]
    x0[1::3] = best_c[:, 1]
    x0[2::3] = best_r

    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': cons_joint}

    try:
        res = minimize(obj_joint, x0, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 25000, 'ftol': 1e-14, 'disp': False})
        if not np.isnan(res.fun):
            c_vals = cons_joint(res.x)
            if np.min(c_vals) >= -1e-8:
                best_c = np.column_stack((res.x[0::3], res.x[1::3]))
                best_r = res.x[2::3].copy()
                best_sum = np.sum(best_r)
    except Exception:
        pass

    # Phase 4: Strict Validation & Minimal Numerical Repair
    centers = best_c
    radii = best_r

    for _ in range(100):
        valid = True
        for i in range(N):
            if radii[i] < 0:
                valid = False; break
            if centers[i, 0] < radii[i] - 1e-10 or centers[i, 0] > 1.0 - radii[i] + 1e-10 or \
               centers[i, 1] < radii[i] - 1e-10 or centers[i, 1] > 1.0 - radii[i] + 1e-10:
                valid = False; break
        if valid:
            for i in range(N):
                for j in range(i + 1, N):
                    d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                    if d < radii[i] + radii[j] - 1e-10:
                        valid = False; break
                if not valid: break
        if valid: break

        # Gentle shrinkage to guarantee strict compliance
        radii *= 0.9995
        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)

    return centers, radii, float(np.sum(radii))
