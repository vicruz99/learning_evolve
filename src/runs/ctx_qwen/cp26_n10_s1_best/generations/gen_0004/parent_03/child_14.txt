import numpy as np
from scipy.optimize import minimize, linprog

N = 26

TRIL_IDX = np.tril_indices(N, -1)


def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])


def constraints(x):
    """Compute all boundary and non-overlap constraints as a vector >= 0."""
    cx = x[0::3]
    cy = x[1::3]
    cr = x[2::3]

    # Boundary constraints: 4 * N
    c = np.concatenate([cx - cr, 1.0 - cx - cr, cy - cr, 1.0 - cy - cr])

    # Overlap constraints (squared distances): N*(N-1)/2
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dr = cr[:, None] + cr[None, :]

    c = np.concatenate([c, dx[TRIL_IDX]**2 + dy[TRIL_IDX]**2 - dr[TRIL_IDX]**2])
    return c


def get_bounds():
    """Variable bounds: x,y in [0,1], r in [0, 0.5]."""
    return [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N


def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)

    idx = 0
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist)
            idx += 1

    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def force_spread(centers, radii, steps=200):
    """Force-directed relaxation with radius-aware repulsion."""
    pts = centers.copy()
    r = radii.copy()
    n = pts.shape[0]
    for step in range(steps):
        forces = np.zeros_like(pts)
        for i in range(n):
            for j in range(i + 1, n):
                dx = pts[j] - pts[i]
                dist = np.linalg.norm(dx)
                desired = r[i] + r[j] + 0.03
                if dist < desired and dist > 1e-6:
                    overlap = max(0, desired - dist)
                    f = overlap * 3.0 / (dist + 0.01)
                    forces[i] -= f * dx / dist
                    forces[j] += f * dx / dist
            # Boundary repulsion
            for dim in range(2):
                if pts[i, dim] < r[i] + 0.01:
                    forces[i, dim] += 0.15
                elif pts[i, dim] > 1.0 - r[i] - 0.01:
                    forces[i, dim] -= 0.15

        lr = 0.12 * (1.0 - step / steps)
        pts += forces * lr
        pts = np.clip(pts, 0.001, 0.999)
    return pts


def make_hex_init(r_base, angle_deg, seed):
    """Generate a rotated hexagonal lattice initialization with safe feasible radii."""
    np.random.seed(seed)
    pts = []
    y = 0.0
    row = 0
    while len(pts) < N + 25:
        x = (row % 2) * r_base
        while x <= 1.0 + r_base:
            pts.append([x, y])
            x += 2.0 * r_base
        y += np.sqrt(3.0) * r_base
        row += 1

    pts = np.array(pts[:N + 25])

    # Rotate and center around (0.5, 0.5)
    center = np.array([0.5, 0.5])
    pts -= center
    ang = np.deg2rad(angle_deg)
    rot = np.array([[np.cos(ang), -np.sin(ang)], [np.sin(ang), np.cos(ang)]])
    pts = pts @ rot.T + center

    # Filter points strictly inside the square
    mask = (pts[:, 0] >= 0.05) & (pts[:, 0] <= 0.95) & (pts[:, 1] >= 0.05) & (pts[:, 1] <= 0.95)
    pts = pts[mask]

    # Pad if rotation/filtering removed too many points
    if len(pts) < N:
        pad = N - len(pts)
        pts = np.vstack([pts, np.random.uniform(0.2, 0.8, (pad, 2))])
    pts = pts[:N]

    # Add small perturbation
    pts += np.random.uniform(-0.002, 0.002, pts.shape)

    # Compute safe initial radii
    radii = np.zeros(N)
    for i in range(N):
        min_d = min(pts[i, 0], 1.0 - pts[i, 0], pts[i, 1], 1.0 - pts[i, 1])
        for j in range(N):
            if i != j:
                d = np.hypot(pts[i, 0] - pts[j, 0], pts[i, 1] - pts[j, 1])
                if d < min_d:
                    min_d = d
        radii[i] = min_d * 0.45

    return pts, radii


def make_random_init(seed):
    """Generate a random initialization with force-directed spreading."""
    np.random.seed(seed)
    pts = np.random.uniform(0.15, 0.85, (N, 2))
    radii = np.full(N, 0.06)
    pts = force_spread(pts, radii, steps=200)
    return pts, radii


def make_grid_init(seed):
    """Generate a perturbed grid initialization."""
    np.random.seed(seed)
    pts = np.zeros((N, 2))
    idx = 0
    for row in range(6):
        for col in range(5):
            if idx >= N:
                break
            x = 0.08 + col * 0.18 + np.random.uniform(-0.008, 0.008)
            y = 0.08 + row * 0.16 + np.random.uniform(-0.008, 0.008)
            pts[idx] = [x, y]
            idx += 1
    while idx < N:
        pts[idx] = np.random.uniform(0.15, 0.85, 2)
        idx += 1
    radii = np.full(N, 0.075)
    return pts, radii


def optimize_from_init(c_init, r_init, bounds, cons, maxiter=15000):
    """Full optimization pipeline from an initial configuration."""
    # Step 1: LP for optimal radii given centers
    r_lp = solve_lp_radii(c_init)
    if r_lp is not None:
        r_init = r_lp

    # Step 2: Force spread with radii-aware repulsion
    c_init = force_spread(c_init, r_init, steps=150)

    # Step 3: LP again after force spread
    r_lp2 = solve_lp_radii(c_init)
    if r_lp2 is not None:
        r_init = r_lp2

    # Build initial vector
    x0 = np.zeros(3 * N)
    x0[0::3] = c_init[:, 0]
    x0[1::3] = c_init[:, 1]
    x0[2::3] = r_init

    # Ensure feasibility
    for i in range(N):
        r = max(0.001, x0[3 * i + 2])
        x0[3 * i] = np.clip(x0[3 * i], r, 1.0 - r)
        x0[3 * i + 1] = np.clip(x0[3 * i + 1], r, 1.0 - r)
        x0[3 * i + 2] = r

    try:
        res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-13, 'disp': False})

        if np.isnan(res.fun):
            return None, None, 0.0

        c_opt = np.column_stack((res.x[0::3], res.x[1::3]))
        r_opt = solve_lp_radii(c_opt)

        if r_opt is not None:
            return c_opt, r_opt, np.sum(r_opt)
        else:
            return c_opt, res.x[2::3], -res.fun
    except Exception:
        pass

    return c_init, r_init, 0.0
