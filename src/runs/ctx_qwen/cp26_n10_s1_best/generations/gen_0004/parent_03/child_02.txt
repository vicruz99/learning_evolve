1. Use a hybrid optimization approach: alternate between LP-based radius optimization (for fixed centers) and force-directed center spreading (for fixed radii)
2. Generate diverse initial configurations: hexagonal lattices with various rotations, random + force-directed, corner-focused patterns
3. Use multiple refinement strategies: SLSQP for fine-tuning, Nelder-Mead with penalty for escaping local minima, deflation-based re-optimization
4. Run many more starts (30-50) with varied parameters

import numpy as np
from scipy.optimize import minimize, linprog

N_CIRCLES = 26

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Returns all inequality constraints g(x) >= 0."""
    n = N_CIRCLES
    cx = x[0::3]
    cy = x[1::3]
    cr = x[2::3]
    
    # Boundary constraints
    c = np.concatenate([
        cx - cr,
        1.0 - cx - cr,
        cy - cr,
        1.0 - cy - cr
    ])
    
    # Overlap constraints using squared distances
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dr = cr[:, None] + cr[None, :]
    
    idx = np.tril_indices(n, -1)
    c = np.concatenate([c, dx[idx]**2 + dy[idx]**2 - dr[idx]**2])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [0, 0.5]."""
    return [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N_CIRCLES

def compute_safe_radii(centers, shrink=0.3):
    """Compute strictly feasible radii for given centers."""
    n = centers.shape[0]
    radii = np.zeros(n)
    for i in range(n):
        min_d = min(centers[i, 0], 1.0 - centers[i, 0],
                    centers[i, 1], 1.0 - centers[i, 1])
        for j in range(n):
            if i != j:
                d = np.hypot(centers[i, 0] - centers[j, 0],
                             centers[i, 1] - centers[j, 1])
                if d < min_d:
                    min_d = d
        radii[i] = min_d * shrink
    return radii

def solve_lp_radii(centers):
    """Solve for optimal radii given fixed centers using Linear Programming."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    for i in range(n):
        x_val, y_val = centers[i]
        bound = min(x_val, 1.0 - x_val, y_val, 1.0 - y_val)
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, bound)
        idx += 1
    
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0],
                           centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist)
            idx += 1
    
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-10)
    except Exception:
        pass
    return None

def force_directed_spread(centers, radii, steps=500, lr_init=0.02):
    """Force-directed layout to spread points evenly."""
    pts = centers.copy()
    rs = radii.copy()
    n = pts.shape[0]
    
    for step in range(steps):
        forces = np.zeros_like(pts)
        lr = lr_init * (1.0 - step / steps)
        
        for i in range(n):
            for j in range(i + 1, n):
                dx = pts[j, 0] - pts[i, 0]
                dy = pts[j, 1] - pts[i, 1]
                dist = np.hypot(dx, dy)
                if dist < rs[i] + rs[j] + 0.15 and dist > 1e-6:
                    target = rs[i] + rs[j] + 0.08
                    if dist < target:
                        f = (target - dist) * 0.15 / (dist + 0.01)
                        forces[i, 0] -= f * dx / dist
                        forces[i, 1] -= f * dy / dist
                        forces[j, 0] += f * dx / dist
                        forces[j, 1] += f * dy / dist
            
            # Boundary repulsion
            for dim in range(2):
                if pts[i, dim] < rs[i] + 0.02:
                    forces[i, dim] += 0.03 * (rs[i] + 0.02 - pts[i, dim])
                elif pts[i, dim] > 1.0 - rs[i] - 0.02:
                    forces[i, dim] -= 0.03 * (pts[i, dim] - (1.0 - rs[i] - 0.02))
        
        pts += forces * lr
        pts = np.clip(pts, 0.01, 0.99)
    
    return pts

def generate_hex_init(seed, r_base=0.09, angle_deg=0.0):
    """Generate a rotated hexagonal lattice initialization."""
    np.random.seed(seed)
    pts = []
    y = r_base
    row = 0
    while len(pts) < N_CIRCLES + 15:
        x = r_base if row % 2 == 0 else 2.0 * r_base
        while x <= 1.0 - r_base:
            pts.append([x, y])
            x += 2.0 * r_base
        y += np.sqrt(3.0) * r_base
        row += 1
        if y > 1.0 + r_base:
            break
    
    pts = np.array(pts[:N_CIRCLES + 10])
    
    # Rotate around center
    if angle_deg != 0.0:
        center = np.array([0.5, 0.5])
        ang = np.deg2rad(angle_deg)
        c, s = np.cos(ang), np.sin(ang)
        dx = pts[:, 0] - center[0]
        dy = pts[:, 1] - center[1]
        pts[:, 0] = dx * c - dy * s + center[0]
        pts[:, 1] = dx * s + dy * c + center[1]
    
    # Filter valid points and pad
    mask = (pts[:, 0] >= 0.05) & (pts[:, 0] <= 0.95) & (pts[:, 1] >= 0.05) & (pts[:, 1] <= 0.95)
    pts = pts[mask]
    while len(pts) < N_CIRCLES:
        pts = np.vstack([pts, np.random.uniform(0.2, 0.8, 2)])
    pts = pts[:N_CIRCLES]
    
    # Perturb
    pts += np.random.uniform(-0.003, 0.003, pts.shape)
    pts = np.clip(pts, 0.05, 0.95)
    
    radii = compute_safe_radii(pts, shrink=0.35)
    return pts, radii

def generate_random_force_init(seed):
    """Generate random points then spread with force-directed method."""
    np.random.seed(seed)
    centers = np.random.uniform(0.15, 0.85, (N_CIRCLES, 2))
    radii = np.full(N_CIRCLES, 0.06)
    centers = force_directed_spread(centers, radii, steps=300, lr_init=0.03)
    radii = compute_safe_radii(centers, shrink=0.35)
    return centers, radii

def generate_corner_init(seed):
    """Generate initialization that places circles near corners."""
    np.random.seed(seed)
    centers = np.zeros((N_CIRCLES, 2))
    # Place 4 circles near corners
    centers[0] = [0.12, 0.12]
    centers[1] = [0.88, 0.12]
    centers[2] = [0.12, 0.88]
    centers[3] = [0.88, 0.88]
    # Fill rest with grid
    idx = 4
    for i in range(6):
        for j in range(5):
            if idx >= N_CIRCLES:
                break
            centers[idx] = [0.2 + j * 0.14, 0.2 + i * 0.14]
            idx += 1
    while idx < N_CIRCLES:
        centers[idx] = np.random.uniform(0.15, 0.85, 2)
        idx += 1
    
    centers += np.random.uniform(-0.01, 0.01, centers.shape)
    centers = np.clip(centers, 0.05, 0.95)
    radii = compute_safe_radii(centers, shrink=0.35)
    return centers, radii

def alternating_optimize(centers, radii, n_cycles=12):
    """Alternate between LP for radii and force-directed for centers."""
    pts = centers.copy()
    rs = radii.copy()
    
    for _ in range(n_cycles):
        # Step 1: LP to get optimal radii for current centers
        lp_rs = solve_lp_radii(pts)
        if lp_rs is not None:
            rs = lp_rs
        
        # Step 2: Force-directed to spread centers for current radii
        pts = force_directed_spread(pts, rs, steps=150, lr_init=0.015)
        
        # Step 3: Recompute safe radii (might increase)
        new_rs = solve_lp_radii(pts)
        if new_rs is not None:
            rs = new_rs
    
    return pts, rs

def nelder_mead_penalty(x0, penalty_weight=100.0):
    """Nelder-Mead with penalty for constraint violations."""
    n = N_CIRCLES
    def obj(x):
        cx = x[0::3]
        cy = x[1::3]
        cr = x[2::3]
        
        penalty = 0.0
        # Boundary violations
        for i in range(n):
            if cx[i] - cr[i] < 0:
                penalty += penalty_weight * (cx[i] - cr[i]) ** 2
            if 1.0 - cx[i] - cr[i] < 0:
                penalty += penalty_weight * (1.0 - cx[i] - cr[i]) ** 2
            if cy[i] - cr[i] < 0:
                penalty += penalty_weight * (cy[i] - cr[i]) ** 2
            if 1.0 - cy[i] - cr[i] < 0:
                penalty += penalty_weight * (1.0 - cy[i] - cr[i]) ** 2
        
        # Overlap violations
        for i in range(n):
            for j in range(i + 1, n):
                d = np.hypot(cx[i] - cx[j], cy[i] - cy[j])
                if d < cr[i] + cr[j]:
                    penalty += penalty_weight * (d - cr[i] - cr[j]) ** 2
        
        return -np.sum(cr) + penalty
    
    try:
        res = minimize(obj, x0, method='Nelder-Mead',
                       options={'maxiter': 5000, 'xatol': 1e-10, 'fatol': 1e-12})
        return res.x, res.fun
    except Exception:
        return x0, obj(x0)

def ensure_feasibility(centers, radii):
    """Clip centers to be within bounds given radii."""
    for i in range(N_CIRCLES):
        r = max(0.001, radii[i])
        centers[i, 0] = np.clip(centers[i, 0], r, 1.0 - r)
        centers[i, 1] = np.clip(centers[i, 1], r, 1.0 - r)
    return centers

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Phase 1: Generate diverse initial configurations
    configs = []
    
    # Hexagonal lattices with various rotations and densities
    for r_base in [0.085, 0.09, 0.095, 0.10, 0.105]:
        for angle in [0.0, 10.0, 15.0, 20.0, 30.0, 45.0, 60.0]:
            for seed in range(3):
                c, r = generate_hex_init(seed * 100 + int(angle), r_base, angle)
                configs.append((c.copy(), r.copy()))
    
    # Force-directed random starts
    for seed in range(12):
        c, r = generate_random_force_init(seed)
        configs.append((c.copy(), r.copy()))
    
    # Corner-focused starts
    for seed in range(8):
        c, r = generate_corner_init(seed)
        configs.append((c.copy(), r.copy()))
    
    # Phase 2: Optimize each configuration with alternating LP + forces + SLSQP
    for c_init, r_init in configs:
        # Alternating optimization
        c_opt, r_opt = alternating_optimize(c_init, r_init, n_cycles=8)
        
        # Build optimization vector
        x0 = np.zeros(3 * N_CIRCLES)
        x0[0::3] = c_opt[:, 0]
        x0[1::3] = c_opt[:, 1]
        x0[2::3] = r_opt
        
        # SLSQP refinement
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': 20000, 'ftol': 1e-13, 'disp': False})
            
            if not np.isnan(res.fun):
                curr_sum = -res.fun
                c_vals = constraints(res.x)
                if np.min(c_vals) >= -1e-6 and curr_sum > best_sum:
                    best_sum = curr_sum
                    best_centers = np.column_stack((res.x[0::3], res.x[1::3]))
                    best_radii = res.x[2::3].copy()
        except Exception:
            pass
        
        # Also try Nelder-Mead penalty approach
        x0_nm = x0.copy()
        x0_nm += np.random.normal(0, 0.002, 3 * N_CIRCLES)
        for i in range(N_CIRCLES):
            r = max(0.005, x0_nm[3*i+2])
            x0_nm[3*i] = np.clip(x0_nm[3*i], r, 1.0-r)
            x0_nm[3*i+1] = np.clip(x0_nm[3*i+1], r, 1.0-r)
        
        try:
            x_nm, _ = nelder_mead_penalty(x0_nm, penalty_weight=50.0)
            cx = x_nm[0::3]
            cy = x_nm[1::3]
            cr = x_nm[2::3]
            
            # Check feasibility
            valid = True
            for i in range(N_CIRCLES):
                if cr[i] < 0 or cx[i] < cr[i] - 1e-9 or cx[i] > 1.0 - cr[i] + 1e-9 or \
                   cy[i] < cr[i] - 1e-9 or cy[i] > 1.0 - cr[i] + 1e-9:
                    valid = False
                    break
            if valid:
                for i in range(N_CIRCLES):
                    for j in range(i + 1, N_CIRCLES):
                        d = np.hypot(cx[i] - cx[j], cy[i] - cy[j])
                        if d < cr[i] + cr[j] - 1e-9:
                            valid = False
                            break
                    if not valid:
                        break
            
            if valid:
                curr_sum = np.sum(cr)
                if curr_sum > best_sum:
                    best_sum = curr_sum
                    best_centers = np.column_stack((cx, cy))
                    best_radii = cr.copy()
        except Exception:
            pass
    
    # Phase 3: Aggressive refinement of best solution
    if best_centers is not None:

        for cycle in range(20):

            pert_c = best_centers + np.random.normal(0, 0.002 * (0.95 ** cycle), best_centers.shape)
            pert_c = np.clip(pert_c, 0.02, 0.98)
            pert_r = best_radii * (0.97 + 0.01 * cycle)

            c_opt, r_opt = alternating_optimize(pert_c, pert_r, n_cycles=6)
            
            x0 = np.zeros(3 * N_CIRCLES)
            x0[0::3] = c_opt[:, 0]
            x0[1::3] = c_opt[:, 1]

                              constraints=cons, options={'maxiter': 15000, 'ftol': 1e-13, 'disp': False})
                
                if not np.isnan(res.fun):
                    curr_sum = -res.fun
                    c_vals = constraints(res.x)
                    if np.min(c_vals) >= -1e-6 and curr_sum > best_sum:
                        best_sum = curr_sum
                        best_centers = np.column_stack((res.x[0::3], res.x[1::3]))
                        best_radii = res.x[2::3].copy()
            except Exception:
                pass
            
            # Try Nelder-Mead from current best with slight perturbation
            x_nm = np.zeros(3 * N_CIRCLES)
            x_nm[0::3] = best_centers[:, 0] + np.random.normal(0, 0.001, N_CIRCLES)
            x_nm[1::3] = best_centers[:, 1] + np.random.normal(0, 0.001, N_CIRCLES)
            x_nm[2::3] = best_radii * 0.985
            
            try:

                cx = x_res[0::3]
                cy = x_res[1::3]
                cr = x_res[2::3]
                
                # Validate

                for i in range(N_CIRCLES):

                       cy[i] < cr[i] - 1e-9 or cy[i] > 1.0 - cr[i] + 1e-9:
                        valid = False
                        break
                if valid:
                    for i in range(N_CIRCLES):
                        for j in range(i + 1, N_CIRCLES):
                            d = np.hypot(cx[i] - cx[j], cy[i] - cy[j])
                            if d < cr[i] + cr[j] - 1e-9:
