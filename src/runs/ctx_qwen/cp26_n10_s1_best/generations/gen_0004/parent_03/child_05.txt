N = 26

def objective(x):

    return -np.sum(x[2::3])

def constraints(x):
    """Compute all boundary and non-overlap constraints as a vector >= 0."""

    cy = x[1::3]
    cr = x[2::3]

    c = np.concatenate([cx - cr, 1.0 - cx - cr, cy - cr, 1.0 - cy - cr])

    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dr = cr[:, None] + cr[None, :]

    return c

def solve_lp_radii(centers):

    n = N
    c_obj = -np.ones(n)

    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], 
                  centers[i, 1], 1.0 - centers[i, 1])

        b_ub[idx] = max(0.0, lim)
        idx += 1

        pass
    return None

            c[:, dim] += push

        c = np.clip(c, r + 1e-9, 1.0 - r - 1e-9)
        
    return c
