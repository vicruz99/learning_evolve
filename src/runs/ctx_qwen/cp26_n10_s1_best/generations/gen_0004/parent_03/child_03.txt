To break past the local minima that have capped previous solutions around 2.634, we implement a robust alternating optimization strategy:
1. **LP-Based Radius Calculation**: For any fixed set of centers, computing the maximal radii is a Linear Programming problem. This is solved exactly and rapidly using `scipy.optimize.linprog`.
2. **Alternating Coordinate Ascent**: We alternate between fixing centers and solving for radii via LP, then fixing radii (slightly shrunk to allow movement) and optimizing centers via SLSQP. This decouples the non-convex joint problem into two much easier subproblems.
3. **Decaying Noise & Deflation**: During the refinement phase, we add controlled Gaussian noise to centers and slightly deflate radii in each iteration. The noise scale and deflation factor decay over time, allowing the optimizer to escape shallow local minima early and converge precisely later.
4. **Diverse Initialization**: We generate multiple hexagonal lattice initializations with varying rotations (0°, 15°, 30°, 45°, 60°) and base radii. Hexagonal patterns naturally approximate dense circle packings, giving SLSQP a strong starting topology.
5. **Strict Feasibility Repair**: A final safety loop minimally shrinks radii and clips centers to guarantee strict compliance with the validation tolerance (`1e-12`), preventing false failures from numerical drift.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Compute all boundary and non-overlap constraints as a vector >= 0."""
    cx, cy, cr = x[0::3], x[1::3], x[2::3]
    c = np.concatenate([cx - cr, 1.0 - cx - cr, cy - cr, 1.0 - cy - cr])
    
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dr = cr[:, None] + cr[None, :]
    
    idx = np.tril_indices(N, -1)
    c = np.concatenate([c, dx[idx]**2 + dy[idx]**2 - dr[idx]**2])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-7, 0.5]."""
    return [(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)] * N

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    k = 0
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[k, i] = 1.0
        b_ub[k] = max(0.0, lim)
        k += 1
        
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = max(0.0, d)
            k += 1
            
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=[(0.0, None)] * n, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def hex_init(seed, angle, r0):
    """Generate a rotated hexagonal lattice initialization."""
    np.random.seed(seed)
    pts = []
    y = 0.0
    row = 0
    while len(pts) < N + 15:
        x = (row % 2) * r0
        while x <= 1.0 + r0:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        
    pts = np.array(pts[:N + 15])
    center = np.array([0.5, 0.5])
    pts = pts - center
    ang = np.deg2rad(angle)
    rot = np.array([[np.cos(ang), -np.sin(ang)], [np.sin(ang), np.cos(ang)]])
    pts = pts @ rot.T + center
    
    mask = (pts[:, 0] >= 0.05) & (pts[:, 0] <= 0.95) & (pts[:, 1] >= 0.05) & (pts[:, 1] <= 0.95)
    pts = pts[mask]
    
    if len(pts) < N:
        pad = N - len(pts)
        pts = np.vstack([pts, np.random.uniform(0.2, 0.8, (pad, 2))])
    pts = pts[:N]
    pts += np.random.uniform(-0.001, 0.001, pts.shape)
    return pts

def run_packing():
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -1.0
    best_x = None
    
    # Phase 1: Diverse Initializations
    inits = []
    for s in range(12):
        inits.append(hex_init(s, 0.0, 0.095))
    for s in range(10):
        inits.append(hex_init(s, 15.0, 0.093))
    for s in range(10):
        inits.append(hex_init(s, 30.0, 0.097))
    for s in range(8):
        inits.append(hex_init(s, 45.0, 0.091))
    for s in range(6):
        inits.append(hex_init(s, 60.0, 0.094))
        
    for pts in inits:
        radii_lp = solve_lp_radii(pts)
        if radii_lp is None:
            radii_lp = np.full(N, 0.08)
            
        x0 = np.zeros(3 * N)
        x0[0::3] = pts[:, 0]
        x0[1::3] = pts[:, 1]
        x0[2::3] = radii_lp * 0.995
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 25000, 'ftol': 1e-13, 'disp': False})
            if not np.isnan(res.fun):
                curr_sum = -res.fun
                c_vals = constraints(res.x)
                if np.min(c_vals) >= -1e-6 and curr_sum > best_sum:
                    best_sum = curr_sum
                    best_x = res.x.copy()
        except Exception:
            pass
            
    # Phase 2: Alternating LP-SLSQP Refinement with Decaying Noise
    if best_x is not None:
        for it in range(25):
            centers = np.column_stack((best_x[0::3], best_x[1::3]))
            radii_lp = solve_lp_radii(centers)
            if radii_lp is None:
                break
                
            x0 = best_x.copy()
            # Gradually reduce deflation to converge tightly
            shrink = 0.96 + 0.04 * (it / 25.0)
            x0[2::3] = radii_lp * shrink
            
            # Decaying perturbation to escape local minima
            noise = 0.0025 * (0.92 ** it)
            x0[0::3] += np.random.normal(0, noise, N)
            x0[1::3] += np.random.normal(0, noise, N)
            
            # Project to strict bounds
            for i in range(N):
                r = max(1e-7, x0[3 * i + 2])
                x0[3 * i] = np.clip(x0[3 * i], r, 1.0 - r)
                x0[3 * i + 1] = np.clip(x0[3 * i + 1], r, 1.0 - r)
                x0[3 * i + 2] = r
                
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 20000, 'ftol': 1e-13, 'disp': False})
                if not np.isnan(res.fun):
                    curr_sum = -res.fun
                    if curr_sum > best_sum:
                        best_x = res.x.copy()
                        best_sum = curr_sum
            except Exception:
                pass
                
    # Fallback
    if best_x is None:
        best_x = np.zeros(3 * N)
        pts = hex_init(0, 0.0, 0.09)
        best_x[0::3] = pts[:, 0]
        best_x[1::3] = pts[:, 1]
        best_x[2::3] = 0.08
        best_sum = -objective(best_x)
        
    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3].copy()
    
    # Phase 3: Strict Validation & Minimal Numerical Repair
    for _ in range(100):
        valid = True
        for i in range(N):
            if radii[i] < 0 or centers[i, 0] < radii[i] - 1e-10 or centers[i, 0] > 1.0 - radii[i] + 1e-10 or \
               centers[i, 1] < radii[i] - 1e-10 or centers[i, 1] > 1.0 - radii[i] + 1e-10:
                valid = False
                break
        if valid:
            for i in range(N):
                for j in range(i + 1, N):
                    d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                    if d < radii[i] + radii[j] - 1e-10:
                        valid = False
                        break
                if not valid:
                    break
        if valid:
            break
            
        radii *= 0.9998
        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)
        
    return centers, radii, float(np.sum(radii))
