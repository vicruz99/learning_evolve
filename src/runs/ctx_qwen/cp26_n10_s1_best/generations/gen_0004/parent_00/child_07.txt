from scipy.optimize import minimize, linprog

N = 26

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [0, 0.5]."""
    return [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N

def constraints(x):
    """Vectorized inequality constraints g(x) >= 0 using squared distances."""
    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]
    
    # Boundary constraints: 4 * N
    c = np.concatenate([
        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs
    ])
    
    # Overlap constraints: dist^2 >= (ri + rj)^2
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    idx = np.triu_indices(N, 1)
    c = np.concatenate([c, dx[idx]**2 + dy[idx]**2 - dr[idx]**2])
    return c

def objective(x):
    """Objective to minimize: negative sum of radii."""
    return -np.sum(x[2::3])

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A = np.zeros((num_ineq, n))
    b = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        x_val, y_val = centers[i]
        bound = min(x_val, 1.0 - x_val, y_val, 1.0 - y_val)
        A[idx, i] = 1.0
        b[idx] = max(0.0, bound)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = max(0.0, dist)
            idx += 1
            
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0.0, None)] * n, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return np.full(n, 0.06)

def force_spread(centers, radii, steps=80):

        f = np.zeros_like(pts)

            # Pairwise repulsion

                    f[j] += [fx, fy]

                    f[i, dim] += 0.03

                    f[i, dim] -= 0.03
                    
        pts += f * 0.05
        pts = np.clip(pts, 0.001, 0.999)
    return pts

def make_hex_init(row_counts, r_base, angle):
    """Generate a hexagonal lattice initialization with specified row counts."""
    pts = []
    y = r_base
    y_step = np.sqrt(3.0) * r_base
    for i, cnt in enumerate(row_counts):
        x_start = r_base + (r_base if i % 2 == 1 else 0.0)
        for k in range(cnt):
            x = x_start + k * 2.0 * r_base
            pts.append([x, y])
        y += y_step
        
    # Fill or trim to exactly N
    pts = np.array(pts)
    if len(pts) < N:
        pad = N - len(pts)
        pts = np.vstack([pts, np.random.uniform(0.1, 0.9, (pad, 2))])

        dy = pts[:, 1] - cy
        pts[:, 0] = dx * ca - dy * sa + cx
        pts[:, 1] = dx * sa + dy * ca + cy
        
    pts += np.random.uniform(-0.003, 0.003, pts.shape)
    return np.clip(pts, 0.05, 0.95), np.full(N, r_base * 0.85)
