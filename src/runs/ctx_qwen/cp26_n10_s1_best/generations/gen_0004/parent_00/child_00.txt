from scipy.optimize import minimize, linprog

N_CIRCLES = 26

def objective(x):
    """Objective to minimize: negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):

    ys = x[1::3]
    rs = x[2::3]

    c = np.concatenate([
        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs
    ])
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    i_idx, j_idx = np.triu_indices(N_CIRCLES, 1)
    c = np.concatenate([c, dx[i_idx, j_idx]**2 + dy[i_idx, j_idx]**2 - dr[i_idx, j_idx]**2])

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [0, 0.5]."""
    b = []
    for _ in range(N_CIRCLES):
        b.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
    return b

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N_CIRCLES
    c_obj = -np.ones(n)  # Maximize sum of radii
    num_ineq = n + n * (n - 1) // 2
    A = np.zeros((num_ineq, n))
    b = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        bound = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A[idx, i] = 1.0
        b[idx] = bound
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = dist
            idx += 1

        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0.0, None)] * n, method='highs')

            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return np.full(n, 0.06)

def init_hex(pattern, r0, angle=0.0):
    """Generate a hexagonal lattice initialization."""
    pts = []
    y = r0
    for i, cnt in enumerate(pattern):
        x_start = r0 + (r0 if i % 2 == 1 else 0.0)
        for k in range(cnt):
            x = x_start + k * 2.0 * r0
            pts.append([x, y])
        y += np.sqrt(3.0) * r0
    pts = np.array(pts[:N_CIRCLES])
    
    if angle != 0.0:
        cx, cy = 0.5, 0.5
        ca, sa = np.cos(angle), np.sin(angle)
        dx = pts[:, 0] - cx
        dy = pts[:, 1] - cy
        pts[:, 0] = dx * ca - dy * sa + cx
        pts[:, 1] = dx * sa + dy * ca + cy
    return pts

def force_spread_init(seed, steps=200):
    """Force-directed layout to spread points and push to boundaries."""
    np.random.seed(seed)
    c = np.random.uniform(0.15, 0.85, (N_CIRCLES, 2))
    for _ in range(steps):
        f = np.zeros_like(c)
        for i in range(N_CIRCLES):
            for j in range(i + 1, N_CIRCLES):
                d = np.hypot(c[i, 0] - c[j, 0], c[i, 1] - c[j, 1])

        c += f * 0.05
        c = np.clip(c, 0.02, 0.98)
    return c
