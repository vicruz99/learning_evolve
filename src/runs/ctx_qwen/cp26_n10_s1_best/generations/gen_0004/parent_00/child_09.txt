import numpy as np
from scipy.optimize import minimize, linprog

N_CIRCLES = 26

def get_lp_structure():
    """Precompute the constraint matrix structure for LP radius optimization."""
    n = N_CIRCLES
    n_cons = n + n * (n - 1) // 2
    A = np.zeros((n_cons, n))
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        A[idx, i] = 1.0
        idx += 1
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            idx += 1
    return A, n_cons

A_LP, N_LP_CONS = get_lp_structure()

def solve_lp_radii(centers):
    """Compute optimal radii for fixed centers using Linear Programming."""
    b = np.zeros(N_LP_CONS)
    idx = 0
    for i in range(N_CIRCLES):
        x, y = centers[i]
        b[idx] = min(x, 1.0 - x, y, 1.0 - y)
        idx += 1
    for i in range(N_CIRCLES):
        for j in range(i + 1, N_CIRCLES):
            b[idx] = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            idx += 1
            
    try:
        res = linprog(-np.ones(N_CIRCLES), A_ub=A_LP, b_ub=b, 
                      bounds=[(0.0, None)] * N_CIRCLES, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return np.full(N_CIRCLES, 0.02)

def force_directed_init(seed, steps=400):

    np.random.seed(seed)

    for _ in range(steps):
        forces = np.zeros_like(centers)

                    forces[i, dim] -= 0.03
                    
        centers += forces * 0.05
        centers = np.clip(centers, 0.05, 0.95)
    return centers

    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]
    
    c = np.concatenate([
        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs
    ])
    
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    idx = np.tril_indices(N_CIRCLES, -1)
    c = np.concatenate([c, dx[idx]**2 + dy[idx]**2 - dr[idx]**2])
    return c

def objective_slsqp(x):
    """Objective for SLSQP: minimize negative sum of radii."""
    return -np.sum(x[2::3])
