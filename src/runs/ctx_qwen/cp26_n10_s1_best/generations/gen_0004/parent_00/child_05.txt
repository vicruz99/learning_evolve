def constraints(x):

    ys = x[1::3]
    rs = x[2::3]

    c = np.concatenate([
        xs - rs, 1.0 - xs - rs, ys - rs, 1.0 - ys - rs
    ])

    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]

    c = np.concatenate([c, dx[i, j]**2 + dy[i, j]**2 - dr[i, j]**2])

def solve_lp_radii(centers):
