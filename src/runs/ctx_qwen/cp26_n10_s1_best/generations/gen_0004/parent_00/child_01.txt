from scipy.optimize import minimize, linprog

N = 26

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [0, 0.5]."""
    return [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N

def constraints(x):
    """Vectorized inequality constraints g(x) >= 0."""

    ys = x[1::3]
    rs = x[2::3]
    
    # Boundary constraints
    c = np.concatenate([
        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs
    ])
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    i, j = np.triu_indices(N, 1)
    c = np.concatenate([c, dx[i,j]**2 + dy[i,j]**2 - dr[i,j]**2])

def objective(x):
    """Objective to minimize: negative sum of radii."""
    return -np.sum(x[2::3])

def solve_lp_radii(centers):

    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A = np.zeros((num_ineq, n))
    b = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        x_val, y_val = centers[i]
        bound = min(x_val, 1.0 - x_val, y_val, 1.0 - y_val)
        A[idx, i] = 1.0
        b[idx] = bound
        idx += 1

        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])

            A[idx, j] = 1.0

            idx += 1

        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0.0, None)] * n, method='highs')

            return res.x
    except Exception:
        pass
    return np.full(n, 0.06)

def make_hex_init(pat, r0, angle=0.0):

    pts = []
    y = r0
    row = 0
    for cnt in pat:
        x_start = r0 + (r0 if row % 2 == 1 else 0.0)
        for k in range(cnt):
            pts.append([x_start + k * 2.0 * r0, y])
        y += np.sqrt(3.0) * r0
        row += 1

        dy = pts[:, 1] - cy
        pts[:, 0] = dx * ca - dy * sa + cx
        pts[:, 1] = dx * sa + dy * ca + cy
        
    return pts, np.full(N, r0 * 0.75)
