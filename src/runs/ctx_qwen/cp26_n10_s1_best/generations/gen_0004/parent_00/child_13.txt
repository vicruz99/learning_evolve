import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [0, 0.5]."""
    return [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N

def constraints(x):
    """Vectorized inequality constraints g(x) >= 0 using squared distances."""
    cx, cy, cr = x[0::3], x[1::3], x[2::3]
    
    # Boundary constraints
    c = np.concatenate([
        cx - cr,
        1.0 - cx - cr,
        cy - cr,
        1.0 - cy - cr
    ])
    
    # Overlap constraints: dist^2 >= (ri + rj)^2
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dr = cr[:, None] + cr[None, :]
    
    idx = np.triu_indices(N, 1)
    c = np.concatenate([c, dx[idx]**2 + dy[idx]**2 - dr[idx]**2])
    return c

def objective(x):
    """Objective to minimize: negative sum of radii."""
    return -np.sum(x[2::3])

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A = np.zeros((num_ineq, n))
    b = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A[idx, i] = 1.0
        b[idx] = max(0.0, lim)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = max(0.0, d)
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return np.full(n, 0.06)

def force_spread(centers, radii, steps=150):
    """Force-directed relaxation to push centers apart and maximize clearance."""
    pts = centers.copy()
    rs = radii.copy()
    
    for step in range(steps):
        f = np.zeros_like(pts)
        damp = 0.5 * (1.0 - step / steps)
        
        for i in range(N):
            # Pairwise repulsion
            for j in range(i + 1, N):
                dx = pts[j] - pts[i]
                d = np.hypot(dx[0], dx[1])
                if d < rs[i] + rs[j] and d > 1e-6:
                    force = (rs[i] + rs[j] - d) * 0.5 / d
                    f[i] -= force * dx / d
                    f[j] += force * dx / d
                    
            # Boundary repulsion
            for dim in range(2):
                if pts[i, dim] < rs[i]:
                    f[i, dim] += 0.1 * (rs[i] - pts[i, dim])
                if pts[i, dim] > 1.0 - rs[i]:
                    f[i, dim] -= 0.1 * (pts[i, dim] - (1.0 - rs[i]))
                    
        pts += f * 0.05 * damp
        pts = np.clip(pts, 0.001, 0.999)
    return pts

def make_hex_init(r0, angle=0.0, seed=0):
    """Generate a hexagonal lattice initialization with optional rotation."""
    np.random.seed(seed)
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        if y > 1.0 + r0:
            break
            
    pts = np.array(pts[:N + 10])
    
    # Rotate around center

        dx, dy = pts[:, 0] - cx, pts[:, 1] - cy
        pts[:, 0] = dx * c - dy * s + cx
        pts[:, 1] = dx * s + dy * c + cy
        
    # Filter out of bounds points
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    
    # Pad if needed
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.2, 0.8, (N - len(pts), 2))])
        
    return pts[:N] + np.random.uniform(-0.002, 0.002, (N, 2))
