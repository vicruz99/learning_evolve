import numpy as np
from scipy.optimize import minimize, linprog

N_CIRCLES = 26
TRIU_IDX = np.triu_indices(N_CIRCLES, 1)

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [0, 0.5]."""
    return [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N_CIRCLES

def constraints(x):
    """Vectorized inequality constraints g(x) >= 0 using squared distances."""
    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]
    
    # Boundary constraints
    c = np.concatenate([
        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs
    ])
    
    # Overlap constraints: dist^2 >= (ri + rj)^2
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    c = np.concatenate([
        c,
        dx[TRIU_IDX]**2 + dy[TRIU_IDX]**2 - dr[TRIU_IDX]**2
    ])
    return c

def objective(x):
    """Objective to minimize: negative sum of radii."""
    return -np.sum(x[2::3])

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N_CIRCLES
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A = np.zeros((num_ineq, n))
    b = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        x_val, y_val = centers[i]
        bound = min(x_val, 1.0 - x_val, y_val, 1.0 - y_val)
        A[idx, i] = 1.0
        b[idx] = max(0.0, bound)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = max(0.0, dist)
            idx += 1
            
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0.0, None)] * n, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return np.full(n, 0.06)

def force_spread(centers, radii, steps=120):
    """Force-directed relaxation to push centers apart and maximize clearance."""
    pts = centers.copy()
    rs = radii.copy()
    for step in range(steps):
        f = np.zeros_like(pts)
        for i in range(N_CIRCLES):
            for j in range(i + 1, N_CIRCLES):
                dx = pts[j, 0] - pts[i, 0]
                dy = pts[j, 1] - pts[i, 1]
                d = np.hypot(dx, dy)
                if d < 0.35 and d > 1e-6:
                    rep = 0.015 / (d**2 + 0.001)
                    fx = rep * dx / d
                    fy = rep * dy / d
                    f[i] -= [fx, fy]
                    f[j] += [fx, fy]
            for dim in range(2):
                if pts[i, dim] < rs[i] + 0.015:
                    f[i, dim] += 0.04
                if pts[i, dim] > 1.0 - rs[i] - 0.015:
                    f[i, dim] -= 0.04
                    
        pts += f * 0.05 * (1.0 - step / steps)
        pts = np.clip(pts, 0.001, 0.999)
    return pts

def make_init_hex(seed, angle=0.0, r_base=0.09):
    """Generate a hexagonal lattice initialization with optional rotation."""
    np.random.seed(seed)
    pts = []
    y = r_base
    row = 0
    while len(pts) < N_CIRCLES + 5:
        x = r_base if row % 2 == 0 else 2.0 * r_base
        while x <= 1.0 - r_base:
            pts.append([x, y])
            x += 2.0 * r_base
        y += np.sqrt(3.0) * r_base
        row += 1
        if y > 1.0 + r_base:
            break
            
    pts = np.array(pts[:N_CIRCLES + 5])
    
    if angle != 0.0:
        cx, cy = 0.5, 0.5
        ca, sa = np.cos(angle), np.sin(angle)
        dx = pts[:, 0] - cx
        dy = pts[:, 1] - cy
        pts[:, 0] = dx * ca - dy * sa + cx
        pts[:, 1] = dx * sa + dy * ca + cy
        
    pts = pts[:N_CIRCLES]
    pts += np.random.uniform(-0.005, 0.005, pts.shape)
    return np.clip(pts, 0.05, 0.95), np.full(N_CIRCLES, r_base)

        c = np.random.uniform(0.15, 0.85, (N_CIRCLES, 2))

        c = force_spread(c, r, steps=150)
        r = solve_lp_radii(c)
        inits.append((c, r))
    # Grid start
    c_grid = np.zeros((N_CIRCLES, 2))
    idx = 0

        x0[1::3] = np.clip(x0[1::3], 0.005, 0.995)
        x0[2::3] = np.clip(x0[2::3], 0.01, 0.3)
