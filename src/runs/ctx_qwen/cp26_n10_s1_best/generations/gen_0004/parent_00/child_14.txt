import numpy as np
from scipy.optimize import minimize, linprog

N_CIRCLES = 26
TRIU_IDX = np.triu_indices(N_CIRCLES, 1)

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [0, 0.5]."""
    return [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N_CIRCLES

def constraints(x):
    """Vectorized inequality constraints g(x) >= 0 using squared distances."""
    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]
    
    c = np.concatenate([
        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs
    ])
    
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    c = np.concatenate([
        c,
        dx[TRIU_IDX]**2 + dy[TRIU_IDX]**2 - dr[TRIU_IDX]**2
    ])
    return c

def objective(x):
    """Objective to minimize: negative sum of radii."""
    return -np.sum(x[2::3])

def setup_lp_matrix():
    """Precompute the constant constraint matrix for Linear Programming."""
    n = N_CIRCLES
    num_ineq = n + n * (n - 1) // 2
    A = np.zeros((num_ineq, n))
    idx = 0
    # Boundary constraints: r_i <= bound_i
    for i in range(n):
        A[idx, i] = 1.0
        idx += 1
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            idx += 1
    return A

def solve_lp_radii(centers, A):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N_CIRCLES
    num_ineq = n + n * (n - 1) // 2
    b = np.zeros(num_ineq)
    idx = 0
    
    for i in range(n):
        x_val, y_val = centers[i]
        b[idx] = min(x_val, 1.0 - x_val, y_val, 1.0 - y_val)
        idx += 1
        
    for i in range(n):
        for j in range(i + 1, n):
            b[idx] = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            idx += 1
            
    try:
        res = linprog(-np.ones(n), A_ub=A, b_ub=b, bounds=[(0.0, None)] * n, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return np.full(n, 0.01)

def get_max_sum_radii(centers, A):
    """Evaluate the objective function using LP."""
    r = solve_lp_radii(centers, A)
    return np.sum(r)

def sa_refine(centers, A, iters=12000):
    """Simulated Annealing on centers evaluated by LP radius optimization."""
    best_centers = centers.copy()
    best_sum = get_max_sum_radii(best_centers, A)
    temp = 0.025
    step = 0.012
    
    for k in range(iters):
        idx = np.random.randint(N_CIRCLES)
        new_centers = best_centers.copy()
        new_centers[idx] += np.random.normal(0, step, 2)
        new_centers[idx] = np.clip(new_centers[idx], 0.002, 0.998)
        
        curr_sum = get_max_sum_radii(new_centers, A)
        delta = curr_sum - best_sum

            best_centers = new_centers
            best_sum = curr_sum
            
        temp *= 0.9996
        step *= 0.9995
        
    return best_centers

def make_init_hex(seed, angle=0.0):
    """Generate a hexagonal lattice initialization with optional rotation."""
    np.random.seed(seed)
    pts = []
    r = 0.09
    y = r
    row = 0
    while len(pts) < N_CIRCLES + 5:
        x = r if row % 2 == 0 else 2.0 * r
        while x <= 1.0 - r:
            pts.append([x, y])
            x += 2.0 * r
        y += np.sqrt(3.0) * r
        row += 1
        if y > 1.0 + r:
            break
            
    pts = np.array(pts[:N_CIRCLES + 5])
    
    if angle != 0.0:
        cx, cy = 0.5, 0.5
        ca, sa = np.cos(angle), np.sin(angle)
        dx = pts[:, 0] - cx
        dy = pts[:, 1] - cy
        pts[:, 0] = dx * ca - dy * sa + cx
        pts[:, 1] = dx * sa + dy * ca + cy
        
    pts = pts[:N_CIRCLES]
    pts += np.random.uniform(-0.004, 0.004, pts.shape)
    return np.clip(pts, 0.05, 0.95), np.full(N_CIRCLES, 0.085)
