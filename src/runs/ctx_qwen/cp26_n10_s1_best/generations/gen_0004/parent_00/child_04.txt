import numpy as np
from scipy.optimize import minimize, linprog

N = 26
TRIU_IDX = np.triu_indices(N, 1)

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [0, 0.5]."""
    return [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N

def constraints(x):
    """Vectorized inequality constraints g(x) >= 0 using squared distances."""
    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]
    
    # Boundary constraints
    c = np.concatenate([
        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs
    ])
    
    # Overlap constraints: dist^2 >= (ri + rj)^2
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    c = np.concatenate([
        c,
        dx[TRIU_IDX]**2 + dy[TRIU_IDX]**2 - dr[TRIU_IDX]**2
    ])
    return c

def objective(x):
    """Objective to minimize: negative sum of radii."""
    return -np.sum(x[2::3])

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A = np.zeros((num_ineq, n))
    b = np.zeros(num_ineq)
    
    idx = 0
    for i in range(n):
        x_val, y_val = centers[i]
        bound = min(x_val, 1.0 - x_val, y_val, 1.0 - y_val)
        A[idx, i] = 1.0
        b[idx] = bound
        idx += 1
        
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = dist
            idx += 1
            
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0.0, None)] * n, method='highs')
        if res.success:
            return res.x * 0.9998  # Slight shrinkage for strict feasibility
    except Exception:
        pass
    return np.full(n, 0.06)

def force_spread(centers, radii, steps=80):
    """Force-directed relaxation to push centers apart and maximize clearance."""
    pts = centers.copy()
    for step in range(steps):
        f = np.zeros_like(pts)
        for i in range(N):
            for j in range(i + 1, N):
                dx = pts[j, 0] - pts[i, 0]
                dy = pts[j, 1] - pts[i, 1]
                d = np.hypot(dx, dy)
                if d < 0.35 and d > 1e-6:
                    rep = 0.012 / (d**2 + 0.001)
                    fx = rep * dx / d
                    fy = rep * dy / d
                    f[i] -= [fx, fy]
                    f[j] += [fx, fy]
            for dim in range(2):
                if pts[i, dim] < radii[i] + 0.015:
                    f[i, dim] += 0.03
                if pts[i, dim] > 1.0 - radii[i] - 0.015:
                    f[i, dim] -= 0.03
                    
        pts += f * 0.04 * (1.0 - step / steps)
        pts = np.clip(pts, 0.005, 0.995)
    return pts

def make_init_hex(seed, row_counts, angle=0.0):
    """Generate a hexagonal lattice initialization with specified row counts."""
    np.random.seed(seed)
    pts = []
    r = 0.092
    y = r
    for i, cnt in enumerate(row_counts):
        x_start = r + (r if i % 2 == 1 else 0.0)
        for k in range(cnt):
            pts.append([x_start + k * 2.0 * r, y])
        y += np.sqrt(3.0) * r
        
    pts = np.array(pts[:N])
    if len(pts) < N:
        pad = N - len(pts)
        pts = np.vstack([pts, np.random.uniform(0.2, 0.8, (pad, 2))])
        
    if angle != 0.0:
        cx, cy = 0.5, 0.5
        ca, sa = np.cos(angle), np.sin(angle)
        dx = pts[:, 0] - cx
        dy = pts[:, 1] - cy
        pts[:, 0] = dx * ca - dy * sa + cx
        pts[:, 1] = dx * sa + dy * ca + cy
        
    pts += np.random.uniform(-0.004, 0.004, pts.shape)
    return np.clip(pts, 0.05, 0.95), np.full(N, 0.08)

def make_init_force(seed):
    """Generate initial configuration using repulsive force simulation."""
    np.random.seed(seed)
    centers = np.random.uniform(0.15, 0.85, (N, 2))
    radii = np.full(N, 0.06)
    for _ in range(150):
        f = np.zeros_like(centers)
        for i in range(N):
            for j in range(i + 1, N):
                dx = centers[j, 0] - centers[i, 0]
                dy = centers[j, 1] - centers[i, 1]
                d = np.hypot(dx, dy)
                if d < 0.3 and d > 1e-6:
                    rep = 0.01 / (d**2 + 0.001)
                    fx, fy = rep * dx / d, rep * dy / d
                    f[i] -= [fx, fy]
                    f[j] += [fx, fy]
            for dim in range(2):
                if centers[i, dim] < 0.12: f[i, dim] += 0.02
                elif centers[i, dim] > 0.88: f[i, dim] -= 0.02
        centers += f * 0.05
        centers = np.clip(centers, 0.05, 0.95)
    return centers, radii

def project_feasible(centers, radii):
    """Project centers and radii to a strictly feasible state."""
    radii = solve_lp_radii(centers) * 0.995
    for i in range(N):
        r = radii[i]
        centers[i, 0] = np.clip(centers[i, 0], r + 1e-6, 1.0 - r - 1e-6)
        centers[i, 1] = np.clip(centers[i, 1], r + 1e-6, 1.0 - r - 1e-6)
    return centers, radii

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -1.0
    best_x = None
    
    # Diverse row patterns for hexagonal lattices
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 5, 5, 6],
        [6, 6, 6, 4, 4], [7, 6, 6, 7], [5, 7, 5, 6, 3],
        [4, 6, 6, 6, 4], [6, 6, 5, 5, 4], [5, 6, 6, 5, 4]
    ]
    
    inits = []
    # Hexagonal lattices with rotations
    for pat in patterns:
        for ang in [0.0, 0.1, 0.2, 0.35, 0.5, 0.7]:
            c, r = make_init_hex(0, pat, angle=ang)
            inits.append((c, r))
            
    # Force-directed starts
    for s in range(8):
        inits.append(make_init_force(s))
        
    # Primary optimization loop
    for c0, r0 in inits:
        c_curr, r_curr = c0.copy(), r0.copy()
        
        # Hybrid refinement: alternate LP radii and force-spreading centers
        for _ in range(4):
            r_curr = solve_lp_radii(c_curr)
            c_curr = force_spread(c_curr, r_curr, steps=60)
            
        c_curr, r_curr = project_feasible(c_curr, r_curr)
        
        x0 = np.zeros(3 * N)
        x0[0::3] = c_curr[:, 0]
        x0[1::3] = c_curr[:, 1]
        x0[2::3] = r_curr
        
        # Try SLSQP
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 15000, 'ftol': 1e-14, 'disp': False})
            if not np.isnan(res.fun) and -res.fun > best_sum:
                if np.min(constraints(res.x)) >= -1e-7:
                    best_sum = -res.fun
                    best_x = res.x.copy()
        except Exception:
            pass
            
        # Try trust-constr (often escapes different basins)
        from scipy.optimize import NonlinearConstraint
        try:
            nl_con = NonlinearConstraint(constraints, 0, np.inf)
            res2 = minimize(objective, x0, method='trust-constr', bounds=bounds,
                            constraints=[nl_con], options={'maxiter': 10000, 'verbose': 0})
            if not np.isnan(res2.fun) and -res2.fun > best_sum:
                if np.min(constraints(res2.x)) >= -1e-7:
                    best_sum = -res2.fun
                    best_x = res2.x.copy()
        except Exception:
            pass

    # Secondary refinement: perturb best solution and re-optimize
    if best_x is not None:
        for step in range(12):
            noise_scale = 0.0025 / (step + 1)
            x_pert = best_x + np.random.normal(0, noise_scale, best_x.shape)
            
            # Deflate radii slightly to allow center movement
            r_pert = np.maximum(x_pert[2::3] * 0.980, 0.005)
            x_pert[2::3] = r_pert
            x_pert[0::3] = np.clip(x_pert[0::3], r_pert, 1.0 - r_pert)
            x_pert[1::3] = np.clip(x_pert[1::3], r_pert, 1.0 - r_pert)
            
            try:
                res = minimize(objective, x_pert, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 12000, 'ftol': 1e-14, 'disp': False})
                if not np.isnan(res.fun):
                    if np.min(constraints(res.x)) >= -1e-7 and -res.fun > best_sum:
                        best_sum = -res.fun
                        best_x = res.x.copy()
            except Exception:
                pass
                
    # Fallback
    if best_x is None:
        best_x = np.zeros(3 * N)
        best_x[2::3] = 0.06
        best_x[0::3] = np.tile(np.linspace(0.1, 0.9, 5), 6)[:N]
        best_x[1::3] = np.repeat(np.linspace(0.1, 0.9, 6), 5)[:N]
        
    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3]
    
    # Final strict validity adjustment
    for _ in range(100):
        valid = True
        for i in range(N):
            if radii[i] < 0:
                valid = False; break
            if centers[i, 0] - radii[i] < -1e-10 or centers[i, 0] + radii[i] > 1.0 + 1e-10:
                valid = False; break
            if centers[i, 1] - radii[i] < -1e-10 or centers[i, 1] + radii[i] > 1.0 + 1e-10:
                valid = False; break
        if valid:
            for i in range(N):
                for j in range(i + 1, N):
                    d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                    if d < radii[i] + radii[j] - 1e-10:
                        valid = False; break
                if not valid: break
        if valid: break
        # Minimal shrinkage to recover validity
        radii *= 0.9998
        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)
        
    return centers, radii, float(np.sum(radii))
