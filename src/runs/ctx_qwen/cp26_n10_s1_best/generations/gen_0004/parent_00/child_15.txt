

<strategy>
To push the sum of radii beyond 2.634, we need a more robust exploration of the non-convex optimization landscape. The primary bottlenecks in previous attempts were:
1. **Local Minima in Radius Space**: SLSQP often settles on suboptimal radii while keeping centers fixed, or vice versa.
2. **Topology Traps**: Small perturbations aren't enough to change the contact graph (which circles touch which).

**Improvement Strategy:**
- **Hybrid LP-SLSQP Alternation**: After optimizing centers with SLSQP, we solve a Linear Program (LP) to find the *exact* optimal radii for those fixed centers. LP solves this subproblem globally and instantly, providing a powerful "kick" to escape radius-local-minima. We then feed these optimal radii back into SLSQP to refine centers.
- **Deflation Perturbation**: We deliberately shrink radii by ~1.5% and add targeted noise to centers. This temporarily frees up space, allowing the optimizer to find new contact topologies that yield larger radii when expanded again.
- **Diverse Structured Initializations**: We generate rotated hexagonal lattices, force-spread random points, and explicit grid patterns. Rotation breaks symmetries that trap gradient-based methods.
- **Strict Feasibility Enforcement**: A final validation loop applies minimal shrinkage only if necessary, guaranteeing compliance with the `1e-12` tolerance in the validator without sacrificing the optimized sum.

This combination of exact subproblem solving, topology exploration via deflation, and rigorous initialization typically pushes packings past plateau barriers.

import numpy as np
from scipy.optimize import minimize, linprog

N_CIRCLES = 26
TRIL_IDX = np.tril_indices(N_CIRCLES, -1)

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [0, 0.5]."""
    return [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N_CIRCLES

def get_constraints(x):
    """Vectorized inequality constraints g(x) >= 0."""
    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]
    
    # Boundary constraints
    c = np.concatenate([
        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs
    ])
    
    # Overlap constraints: dist^2 >= (ri + rj)^2
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    c = np.concatenate([c, dx[TRIL_IDX]**2 + dy[TRIL_IDX]**2 - dr[TRIL_IDX]**2])
    return c

def objective(x):
    """Objective to minimize: negative sum of radii."""
    return -np.sum(x[2::3])

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N_CIRCLES
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A = np.zeros((num_ineq, n))
    b = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A[idx, i] = 1.0
        b[idx] = max(0.0, lim)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = max(0.0, dist)
            idx += 1
            
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0.0, None)] * n, method='highs')
    if res.success:
        return res.x
    return np.full(n, 0.06)

def make_hex_init(seed, angle=0.0, r_base=0.095):
    """Generate a hexagonal lattice initialization with optional rotation."""
    np.random.seed(seed)
    pts = []
    y = r_base
    row = 0
    while len(pts) < N_CIRCLES + 10:
        x = r_base if row % 2 == 0 else 2.0 * r_base
        while x <= 1.0 - r_base + 1e-9:
            pts.append([x, y])
            x += 2.0 * r_base
        y += np.sqrt(3.0) * r_base
        row += 1
        if y > 1.0 + r_base:
            break
            
    pts = np.array(pts[:N_CIRCLES + 10])
    
    # Rotate around center
    if angle != 0.0:
        cx, cy = 0.5, 0.5
        ca, sa = np.cos(angle), np.sin(angle)
        dx = pts[:, 0] - cx
        dy = pts[:, 1] - cy
        pts[:, 0] = dx * ca - dy * sa + cx
        pts[:, 1] = dx * sa + dy * ca + cy
        
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    while len(pts) < N_CIRCLES:
        pts = np.vstack([pts, np.random.uniform(0.1, 0.9, (1, 2))])
    pts = pts[:N_CIRCLES]
    pts += np.random.uniform(-0.003, 0.003, pts.shape)
    return np.clip(pts, 0.05, 0.95)

def force_spread_init(seed):
    """Force-directed layout to spread points evenly and push to boundaries."""
    np.random.seed(seed)
    pts = np.random.uniform(0.15, 0.85, (N_CIRCLES, 2))
    for _ in range(300):
        f = np.zeros_like(pts)
        for i in range(N_CIRCLES):
            for j in range(i + 1, N_CIRCLES):
                diff = pts[j] - pts[i]
                d = np.hypot(diff[0], diff[1])
                if d < 0.25 and d > 1e-5:
                    rep = 0.02 / (d**2 + 0.001)
                    f[i] -= rep * diff / d
                    f[j] += rep * diff / d
            for dim in range(2):
                if pts[i, dim] < 0.1: f[i, dim] += 0.05
                elif pts[i, dim] > 0.9: f[i, dim] -= 0.05
        pts += f * 0.04
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': get_constraints}
    
    best_sum = 0.0
    best_x = None
    
    # Generate diverse initial configurations
    inits = []
    for s in range(25):
        inits.append(make_hex_init(s, angle=s * 0.1))
    for s in range(15):
        inits.append(force_spread_init(s))
        
    # Phase 1: Primary optimization with LP-initialized radii
    for pts in inits:
        r_lp = solve_lp_radii(pts)
        x0 = np.zeros(3 * N_CIRCLES)
        x0[0::3] = pts[:, 0]
        x0[1::3] = pts[:, 1]
        x0[2::3] = r_lp
        
        # Project to strict feasible bounds for optimizer stability
        for i in range(N_CIRCLES):
            r = max(0.01, x0[3*i+2])
            x0[3*i] = np.clip(x0[3*i], r, 1.0 - r)
            x0[3*i+1] = np.clip(x0[3*i+1], r, 1.0 - r)
            x0[3*i+2] = r
            
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 25000, 'ftol': 1e-14, 'disp': False})
            if not np.isnan(res.fun):
                val = get_constraints(res.x)
                if np.min(val) >= -1e-6 and -res.fun > best_sum:
                    best_sum = -res.fun
                    best_x = res.x.copy()
        except Exception:
            pass
            
    # Phase 2: Deflation & Perturbation to escape local topological minima
    if best_x is not None:
        for step in range(12):
            x0 = best_x.copy()
            # Shrink radii to free up configuration space
            x0[2::3] *= 0.985
            # Perturb centers to explore new contact graphs
            x0[0::3] += np.random.normal(0, 0.0015, N_CIRCLES)
            x0[1::3] += np.random.normal(0, 0.0015, N_CIRCLES)
            
            for i in range(N_CIRCLES):
                r = max(0.01, x0[3*i+2])
                x0[3*i] = np.clip(x0[3*i], r, 1.0 - r)
                x0[3*i+1] = np.clip(x0[3*i+1], r, 1.0 - r)
                x0[3*i+2] = r
                
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 15000, 'ftol': 1e-14, 'disp': False})
                if not np.isnan(res.fun):
                    val = get_constraints(res.x)
                    if np.min(val) >= -1e-6 and -res.fun > best_sum:
                        best_sum = -res.fun
                        best_x = res.x.copy()
            except Exception:
                pass
                
    # Phase 3: Final LP refinement on the best centers found
    if best_x is not None:
        centers = np.column_stack((best_x[0::3], best_x[1::3]))
        radii = solve_lp_radii(centers)
        curr_sum = np.sum(radii)
        
        # Validate LP result against strict geometric constraints
        valid = True
        for i in range(N_CIRCLES):
            if (centers[i,0] < radii[i] - 1e-9 or centers[i,0] > 1.0 - radii[i] + 1e-9 or 
                centers[i,1] < radii[i] - 1e-9 or centers[i,1] > 1.0 - radii[i] + 1e-9):
                valid = False
                break
        if valid:
            for i in range(N_CIRCLES):
                for j in range(i + 1, N_CIRCLES):
                    d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
                    if d < radii[i] + radii[j] - 1e-9:
                        valid = False
                        break
                if not valid:
                    break
                    
        if valid and curr_sum > best_sum:
            best_x = np.zeros(3 * N_CIRCLES)
            best_x[0::3] = centers[:, 0]
            best_x[1::3] = centers[:, 1]
            best_x[2::3] = radii
            best_sum = curr_sum
        elif not valid:
            # Fallback shrink if LP boundary numerical drift occurred
            radii *= 0.999
            centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
            centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)
            best_x[0::3] = centers[:, 0]
            best_x[1::3] = centers[:, 1]
            best_x[2::3] = radii
            best_sum = np.sum(radii)
            
    # Extract final result
    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3]
    
    # Safety validation loop against 1e-12 validator tolerance
    for _ in range(100):
        valid = True
        for i in range(N_CIRCLES):
            if (radii[i] < 0 or centers[i,0] < radii[i] - 1e-12 or centers[i,0] > 1.0 - radii[i] + 1e-12 or 
                centers[i,1] < radii[i] - 1e-12 or centers[i,1] > 1.0 - radii[i] + 1e-12):
                valid = False
                break
        if valid:
            for i in range(N_CIRCLES):
                for j in range(i + 1, N_CIRCLES):
                    d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
                    if d < radii[i] + radii[j] - 1e-12:
                        valid = False
                        break
                if not valid:
                    break
        if valid:
            break
        radii *= 0.9998
        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)
        
    return centers, radii, float(np.sum(radii))
