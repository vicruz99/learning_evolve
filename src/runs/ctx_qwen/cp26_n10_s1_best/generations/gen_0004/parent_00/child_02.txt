The previous approaches relied heavily on SLSQP from various starts, which can be slow and prone to getting stuck in suboptimal symmetric configurations.

Key improvements in this strategy:
1. **Hybrid Alternating Optimization**: Instead of relying solely on SLSQP, we alternate between Linear Programming (LP) for radii and a force-directed relaxation for centers. LP optimally assigns radii for fixed centers in milliseconds, while the force method efficiently pushes centers apart to create more space. This combination is highly effective for dense packings.
2. **Focused Multi-Start**: We generate ~100 diverse initial configurations (rotated hexagonal lattices with varying densities and angles, plus random starts). We run the fast LP+Force cycle on all of them, then select only the top 15 performers for the expensive SLSQP refinement. This allocates computational budget where it matters most.
3. **Deflation & Reinflation**: After finding a local optimum, we deliberately shrink radii by ~2%, add controlled noise to centers, and re-optimize. This "deflation" breaks binding constraints, allowing the optimizer to explore new topologies and often escape local minima to find a globally better arrangement.
4. **Strict Feasibility & Minimal Shrinkage**: A final validation loop ensures all constraints are met within the validator's tolerance. If micro-violations exist due to numerical drift, we apply a minimal uniform shrinkage to guarantee validity without sacrificing the score significantly.
5. **Vectorized & Efficient**: All helper functions are top-level, closures are avoided, and matrix operations are used where possible to keep execution fast and stable.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [0, 0.5]."""
    return [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N

def constraint_func(x):
    """Vectorized inequality constraints g(x) >= 0 using squared distances."""
    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]
    
    # Boundary constraints
    c = np.concatenate([
        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs
    ])
    
    # Overlap constraints: dist^2 >= (ri + rj)^2
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    c = np.concatenate([
        c,
        dx[TRIL_IDX]**2 + dy[TRIL_IDX]**2 - dr[TRIL_IDX]**2
    ])
    return c

def objective(x):
    """Objective to minimize: negative sum of radii."""
    return -np.sum(x[2::3])

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A = np.zeros((num_ineq, n))
    b = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        x_val, y_val = centers[i]
        bound = min(x_val, 1.0 - x_val, y_val, 1.0 - y_val)
        A[idx, i] = 1.0
        b[idx] = max(0.0, bound)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = dist
            idx += 1
            
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0.0, None)] * n, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return np.full(n, 0.06)

def force_spread(centers, radii, steps=100):
    """Force-directed relaxation to push centers apart and maximize clearance."""
    pts = centers.copy()
    for _ in range(steps):
        f = np.zeros_like(pts)
        for i in range(N):
            # Pairwise repulsion
            for j in range(i + 1, N):
                dx = pts[j, 0] - pts[i, 0]
                dy = pts[j, 1] - pts[i, 1]
                d = np.hypot(dx, dy)
                if d < radii[i] + radii[j] + 0.03 and d > 1e-6:
                    rep = 0.02 / (d**2 + 0.0001)
                    fx = rep * dx / d
                    fy = rep * dy / d
                    f[i] -= [fx, fy]
                    f[j] += [fx, fy]
            # Boundary repulsion
            for dim in range(2):
                if pts[i, dim] < radii[i] + 0.01:
                    f[i, dim] += 0.05
                if pts[i, dim] > 1.0 - radii[i] - 0.01:
                    f[i, dim] -= 0.05
                    
        pts += f * 0.05
        pts = np.clip(pts, 0.005, 0.995)
    return pts

def make_hex_init(rows, r, ang, seed):
    """Generate a hexagonal lattice initialization with optional rotation."""
    np.random.seed(seed)
    pts = []
    y = r
    for i, cnt in enumerate(rows):
        x_start = r + (r if i % 2 == 1 else 0.0)
        for k in range(cnt):
            pts.append([x_start + k * 2 * r, y])
        y += np.sqrt(3.0) * r
        
    pts = np.array(pts[:N])
    if len(pts) < N:
        pad = N - len(pts)
        pts = np.vstack([pts, np.random.uniform(0.2, 0.8, (pad, 2))])
        
    if ang != 0.0:
        cx, cy = 0.5, 0.5
        ca, sa = np.cos(ang), np.sin(ang)
        dx = pts[:, 0] - cx
        dy = pts[:, 1] - cy
        pts[:, 0] = dx * ca - dy * sa + cx
        pts[:, 1] = dx * sa + dy * ca + cy
        
    pts += np.random.uniform(-0.003, 0.003, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraint_func}
    
    best_sum = -1.0
    best_x = None
    candidates = []
    
    # Diverse initial configurations
    patterns = [[5,6,5,6,4], [6,5,6,5,4], [5,6,6,5,4], [4,6,6,6,4], [6,6,5,5,4]]
    configs = []
    for pat in patterns:
        for r in [0.09, 0.095, 0.10, 0.105]:
            for ang in np.linspace(0, 0.6, 13):
                configs.append((make_hex_init(pat, r, ang, seed=0), r))
    for s in range(20):
        configs.append((np.random.uniform(0.15, 0.85, (N, 2)), 0.06))
        
    # Phase 1: Fast Hybrid LP + Force Optimization
    for c0, r0 in configs:
        c_curr, r_curr = c0.copy(), np.full(N, r0)
        # Alternate LP radii and force-spreading centers
        for _ in range(5):
            r_curr = solve_lp_radii(c_curr)
            c_curr = force_spread(c_curr, r_curr, steps=80)
            
        candidates.append((c_curr.copy(), r_curr.copy(), np.sum(r_curr)))
        
    # Sort candidates by sum of radii descending
    candidates.sort(key=lambda x: x[2], reverse=True)
    
    # Phase 2: Precise SLSQP Refinement on top performers
    for c_best, r_best, _ in candidates[:15]:
        x0 = np.zeros(3 * N)
        x0[0::3] = c_best[:, 0]
        x0[1::3] = c_best[:, 1]
        x0[2::3] = r_best
        
        # Project to strictly feasible bounds for SLSQP stability
        for i in range(N):
            r = max(0.01, x0[3*i+2])
            x0[3*i] = np.clip(x0[3*i], r, 1.0 - r)
            x0[3*i+1] = np.clip(x0[3*i+1], r, 1.0 - r)
            
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 15000, 'ftol': 1e-13, 'disp': False})
            
            if not np.isnan(res.fun):
                vals = constraint_func(res.x)
                if np.min(vals) >= -1e-7 and -res.fun > best_sum:
                    best_sum = -res.fun
                    best_x = res.x.copy()
        except Exception:
            continue
            
    # Phase 3: Deflation & Perturbation to escape local minima
    if best_x is not None:
        for step in range(10):
            noise_scale = 0.002 / (step + 1)
            x_pert = best_x + np.random.normal(0, noise_scale, 3 * N)
            # Shrink radii slightly to free up configuration space
            r_pert = np.maximum(x_pert[2::3] * 0.982, 0.005)
            x_pert[0::3] = np.clip(x_pert[0::3], r_pert, 1.0 - r_pert)
            x_pert[1::3] = np.clip(x_pert[1::3], r_pert, 1.0 - r_pert)
            x_pert[2::3] = r_pert
            
            try:
                res = minimize(objective, x_pert, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 10000, 'ftol': 1e-13, 'disp': False})
                if not np.isnan(res.fun):
                    vals = constraint_func(res.x)
                    if np.min(vals) >= -1e-7 and -res.fun > best_sum:
                        best_sum = -res.fun
                        best_x = res.x.copy()
            except Exception:
                pass
                
    # Fallback (should not be reached)
    if best_x is None:
        c_f = make_hex_init([5,6,5,6,4], 0.09, 0.0, 0)
        r_f = solve_lp_radii(c_f)
        best_x = np.zeros(3 * N)
        best_x[0::3] = c_f[:, 0]
        best_x[1::3] = c_f[:, 1]
        best_x[2::3] = r_f
        
    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3]
    
    # Final strict validity adjustment against 1e-12 tolerance
    for _ in range(100):
        valid = True
        for i in range(N):
            if radii[i] < 0:
                valid = False; break
            if centers[i, 0] - radii[i] < -1e-9 or centers[i, 0] + radii[i] > 1.0 + 1e-9:
                valid = False; break
            if centers[i, 1] - radii[i] < -1e-9 or centers[i, 1] + radii[i] > 1.0 + 1e-9:
                valid = False; break
        if valid:
            for i in range(N):
                for j in range(i + 1, N):
                    d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                    if d < radii[i] + radii[j] - 1e-9:
                        valid = False; break
                if not valid: break
        if valid: break
        # Minimal shrinkage to guarantee strict compliance
        radii *= 0.9999
        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)
        
    return centers, radii, float(np.sum(radii))
