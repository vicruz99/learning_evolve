N_CIRCLES = 26
TRIU_IDX = np.triu_indices(N_CIRCLES, 1)

def get_bounds():

    return [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N_CIRCLES

def objective(x):
    """Objective to minimize: negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):

    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]
    
    # Boundary constraints
    c = np.concatenate([
        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs
    ])

    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dist = np.sqrt(dx**2 + dy**2)
    dr = rs[:, None] + rs[None, :]

def compute_safe_radii(centers):

    n = centers.shape[0]
    r = np.full(n, 0.2)

        x, y = centers[i]

                lim = min(lim, d * 0.45)

    return r

    return pts[:N_CIRCLES]

def project_to_bounds(x, eps=1e-5):
    """Project variables to satisfy bounds strictly."""
    x_out = x.copy()
    for i in range(N_CIRCLES):
        r = max(eps, x_out[3*i + 2])

        x_out[3*i + 1] = np.clip(x_out[3*i + 1], r, 1.0 - r)
        x_out[3*i + 2] = r
    return x_out
