import numpy as np
from scipy.optimize import minimize

def _compute_penalty_and_gradient(x, n, r):
    """
    Computes the penalty function value and its gradient for circle packing constraints.
    x: flat array of shape (2*n,) containing center coordinates
    n: number of circles
    r: target radius
    Returns: (penalty_value, gradient_array)
    """
    centers = x.reshape(-1, 2)
    pen = 0.0
    grad = np.zeros_like(x)
    two_r = 2.0 * r
    four_r2 = 4.0 * r * r
    
    for i in range(n):
        ci = centers[i]
        # Boundary constraints
        for d in range(2):
            v = ci[d]
            if v < r:
                err = r - v
                pen += err * err
                grad[2*i + d] -= 2.0 * err
            elif v > 1.0 - r:
                err = v - (1.0 - r)
                pen += err * err
                grad[2*i + d] += 2.0 * err
                
        # Pairwise non-overlap constraints
        for j in range(i + 1, n):
            cj = centers[j]
            dx = ci[0] - cj[0]
            dy = ci[1] - cj[1]
            d2 = dx*dx + dy*dy
            if d2 < four_r2 and d2 > 1e-12:
                dist = np.sqrt(d2)
                overlap = two_r - dist
                pen += overlap * overlap
                factor = -2.0 * overlap / dist
                grad[2*i] += factor * dx
                grad[2*i+1] += factor * dy
                grad[2*j] -= factor * dx
                grad[2*j+1] -= factor * dy
    return pen, grad

def run_packing():
    n = 26
    
    # Initialize centers on a hexagonal-like grid
    init_centers = np.zeros((n, 2))
    idx = 0
    for row in range(5):
        for col in range(6):
            if idx >= n:
                break
            x = 0.05 + col * 0.15 + (row % 2) * 0.075
            y = 0.05 + row * 0.18
            init_centers[idx] = [x, y]
            idx += 1
    init_centers = np.clip(init_centers, 0.01, 0.99)
    
    x0 = init_centers.flatten()
    bounds = [(0.0, 1.0)] * (2 * n)
    
    # Phase 1: Gradual radius expansion
    r = 0.01
    best_r = r
    x_best = x0.copy()
    
    while r <= 0.105:
        res = minimize(
            _compute_penalty_and_gradient,
            x0=x_best,
            args=(n, r),
            method='L-BFGS-B',
            jac=True,
            bounds=bounds,
            options={'maxiter': 3000, 'ftol': 1e-12, 'gtol': 1e-8}
        )
        if res.fun < 1e-4:
            best_r = r
            x_best = res.x.copy()
            r += 0.0005
        else:
            break
            
    # Phase 2: Binary search refinement
    r_lo = best_r - 0.001
    r_hi = best_r + 0.003
    for _ in range(30):
        r_mid = (r_lo + r_hi) * 0.5
        res = minimize(
            _compute_penalty_and_gradient,
            x0=x_best,
            args=(n, r_mid),
            method='L-BFGS-B',
            jac=True,
            bounds=bounds,
            options={'maxiter': 3000, 'ftol': 1e-12}
        )
        if res.fun < 1e-6:
            r_lo = r_mid
            x_best = res.x.copy()
        else:
            r_hi = r_mid
            
    final_r = r_lo
    centers = x_best.reshape(-1, 2)
    radii = np.full(n, final_r)
    sum_radii = final_r * n
    
    return centers, radii, sum_radii
