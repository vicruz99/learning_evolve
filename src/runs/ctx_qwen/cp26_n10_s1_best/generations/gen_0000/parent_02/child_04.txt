2. Non-overlap constraints: $\sqrt{(x_i-x_j)^2 + (y_i-y_j)^2} \ge r_i + r_j$ for all $i < j$.

- **Constraints**: All pairwise non-overlap conditions are enforced as inequality constraints. SLSQP handles vector-valued constraint functions efficiently.
- **Robustness**: Multiple random restarts (5 iterations) are performed, keeping the configuration with the highest sum of radii. A final post-processing step slightly scales down radii if numerical precision causes marginal constraint violations, ensuring strict validity against the checker.

import numpy as np
from scipy.optimize import minimize

def compute_constraints_26(vars):
    """
    Computes non-overlap constraints for 26 circles.
    Returns an array where each element is (distance - r_i - r_j).
    Constraint is satisfied if all elements >= 0.
    """
    n = 26
    num_constraints = n * (n - 1) // 2
    dists = np.empty(num_constraints)
    k = 0
    for i in range(n):
        xi, yi, ri = vars[3*i], vars[3*i+1], vars[3*i+2]
        for j in range(i + 1, n):
            xj, yj, rj = vars[3*j], vars[3*j+1], vars[3*j+2]
            dists[k] = np.hypot(xi - xj, yi - yj) - (ri + rj)
            k += 1
    return dists

def compute_objective(vars):
    """
    Objective function to maximize: sum of radii.
    Scipy minimizes, so we return negative sum.
    """
    return -np.sum(vars[2::3])

def run_packing():
    n = 26
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * n
    cons = {'type': 'ineq', 'fun': compute_constraints_26}
    
    best_result = None
    best_sum = -1.0
    
    np.random.seed(42)
    num_restarts = 6
    
    for _ in range(num_restarts):
        # Initialize with 5x5 grid + 1 circle
        centers = np.zeros((n, 2))
        radii = np.full(n, 0.1)
        
        idx = 0
        for i in range(5):
            for j in range(5):
                centers[idx] = [(i + 0.5) * 0.2, (j + 0.5) * 0.2]
                idx += 1
        centers[25] = [0.5, 0.5]
        radii[25] = 0.05
        
        # Add random perturbation to escape grid symmetry and local minima
        centers += np.random.uniform(-0.025, 0.025, centers.shape)
        radii += np.random.uniform(-0.01, 0.01, radii.shape)
        radii = np.maximum(radii, 0.01)
        centers = np.clip(centers, 0.01, 0.99)
        
        # Flatten to optimization vector: [x0, y0, r0, x1, y1, r1, ...]
        x0 = np.zeros(3 * n)
        for i in range(n):
            x0[3*i] = centers[i, 0]
            x0[3*i+1] = centers[i, 1]
            x0[3*i+2] = radii[i]
            
        # Run SLSQP optimizer
        res = minimize(compute_objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 800, 'ftol': 1e-9, 'disp': False})
        
        current_sum = -res.fun
        if current_sum > best_sum:
            best_sum = current_sum
            best_result = res
            
    if best_result is not None:
        x_opt = best_result.x
        centers_opt = np.zeros((n, 2))
        radii_opt = np.zeros(n)
        for i in range(n):
            centers_opt[i, 0] = x_opt[3*i]
            centers_opt[i, 1] = x_opt[3*i+1]
            radii_opt[i] = x_opt[3*i+2]
    else:
        # Fallback to safe grid
        centers_opt = np.zeros((n, 2))
        radii_opt = np.full(n, 0.05)
        for i in range(n):
            centers_opt[i] = [(i % 5 + 0.5) * 0.2, (i // 5 + 0.5) * 0.2]
            
    # Post-processing: ensure strict validity against checker tolerance
    # If any pair is too close, uniformly scale radii down slightly
    needs_scale = False
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers_opt[i, 0] - centers_opt[j, 0], 
                            centers_opt[i, 1] - centers_opt[j, 1])
            if dist < radii_opt[i] + radii_opt[j] - 1e-10:
                needs_scale = True
                break
        if needs_scale:
            break
            
    if needs_scale:
        # Scale down by 0.995 to guarantee clearance
        radii_opt *= 0.995
        
    return centers_opt, radii_opt, np.sum(radii_opt)
