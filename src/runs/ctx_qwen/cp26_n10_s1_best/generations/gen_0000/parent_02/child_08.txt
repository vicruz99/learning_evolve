import numpy as np
from scipy.optimize import minimize

def get_initial_guess(n, seed):
    """
    Generate a valid initial guess using a perturbed hexagonal grid.
    """
    rng = np.random.default_rng(seed)
    
    # Determine grid dimensions to fit n circles
    # Rough estimate for hexagonal packing
    # Spacing factor
    spacing = 0.25 
    
    # Try to fit in a grid
    cols = int(np.ceil(np.sqrt(n * 2 / np.sqrt(3))))
    rows = int(np.ceil(n / cols))
    
    points = []
    count = 0
    
    for r in range(rows):
        y = 0.1 + r * spacing * np.sqrt(3) / 2
        # Shift odd rows
        x_offset = (0.5 * spacing) if r % 2 == 1 else 0
        
        for c in range(cols):
            x = 0.1 + c * spacing + x_offset
            
            # Check bounds
            if 0 <= x <= 1 and 0 <= y <= 1:
                points.append((x, y))
                count += 1
                if count >= n:
                    break
        if count >= n:
            break
    
    # If grid didn't produce enough points, fill with random
    while len(points) < n:
        x = rng.uniform(0.1, 0.9)
        y = rng.uniform(0.1, 0.9)
        points.append((x, y))
    
    # Trim to exactly n
    points = points[:n]
    
    centers = np.array(points)
    # Add small random perturbation to break symmetry and help optimization
    centers += rng.normal(0, 0.02, centers.shape)
    
    # Clip to safe bounds [0.1, 0.9] initially to ensure validity with small radii
    centers = np.clip(centers, 0.1, 0.9)
    
    # Initial small radii
    radii = np.full(n, 0.05)
    
    return centers, radii

def encode(centers, radii):
    """
    Encode centers and radii into a single 1D vector.
    Format: [x1, y1, r1, x2, y2, r2, ...]
    """
    n = centers.shape[0]
    vec = np.zeros(3 * n)
    for i in range(n):
        vec[3*i] = centers[i, 0]
        vec[3*i+1] = centers[i, 1]
        vec[3*i+2] = radii[i]
    return vec

def decode(vec, n):
    """
    Decode vector back to centers and radii.
    """
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        centers[i, 0] = vec[3*i]
        centers[i, 1] = vec[3*i+1]
        radii[i] = vec[3*i+2]
    return centers, radii

# Objective function: Maximize sum of radii => Minimize -sum(radii)
def objective(vec):
    n = len(vec) // 3
    radii = vec[2::3] # Slicing is efficient: r1, r2, ...
    return -np.sum(radii)

# Constraint helper functions (Top level, no closures)

def con_x_min(vec, i):
    # x_i - r_i >= 0
    return vec[3*i] - vec[3*i+2]

def con_x_max(vec, i):
    # 1 - x_i - r_i >= 0
    return 1.0 - vec[3*i] - vec[3*i+2]

def con_y_min(vec, i):
    # y_i - r_i >= 0
    return vec[3*i+1] - vec[3*i+2]

def con_y_max(vec, i):
    # 1 - y_i - r_i >= 0
    return 1.0 - vec[3*i+1] - vec[3*i+2]

def con_overlap(vec, i, j):
    # (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
    xi, yi, ri = vec[3*i], vec[3*i+1], vec[3*i+2]
    xj, yj, rj = vec[3*j], vec[3*j+1], vec[3*j+2]
    
    dist_sq = (xi - xj)**2 + (yi - yj)**2
    sum_r_sq = (ri + rj)**2
    
    return dist_sq - sum_r_sq

                objective,
                x0,
