def _init_hexagonal_grid(n):

    # Add small random noise to break symmetry
    centers += np.random.normal(0, 0.01, centers.shape)
    # Clip to valid range
    centers = np.clip(centers, 0.05, 0.95)
    
    return centers

def _optimize_centers_fixed_r(init_centers, r):
    """
    Minimize penalty for centers given fixed radius r.
    """
    def penalty(x_flat):
        x = x_flat.reshape(-1, 2)
        # Wall penalties
        # x >= r => r - x <= 0
        # x <= 1-r => x + r - 1 <= 0
        viol_x_left = np.maximum(0, r - x[:, 0])**2
        viol_x_right = np.maximum(0, x[:, 0] + r - 1)**2
        viol_y_bottom = np.maximum(0, r - x[:, 1])**2
        viol_y_top = np.maximum(0, x[:, 1] + r - 1)**2
        
        wall_pen = np.sum(viol_x_left + viol_x_right + viol_y_bottom + viol_y_top)
        
        # Overlap penalties
        # dist >= 2r => 2r - dist <= 0
        dists = np.linalg.norm(x[:, np.newaxis, :] - x[np.newaxis, :, :], axis=2)
        np.fill_diagonal(dists, 1.0) # Avoid self
        overlap_pen = np.sum(np.maximum(0, 2 * r - dists)**2)
        
        return wall_pen + overlap_pen

    res = minimize(penalty, init_centers.flatten(), method='L-BFGS-B')
    return res

def _optimize_variable_radii(init_centers, init_radii):
    """
    Optimize both centers and radii to maximize sum of radii.
    Objective: Maximize sum(radii) <=> Minimize -sum(radii) + lambda * penalty
    """
    n = len(init_radii)
    x0 = np.concatenate([init_centers.flatten(), init_radii])
    
    # Large lambda to enforce constraints strictly
    LAMBDA = 1000.0 
    
    def objective(vars_flat):
        c = vars_flat[:2*n].reshape(-1, 2)
        r = vars_flat[2*n:]
        
        # Wall penalties
        v_x_l = np.maximum(0, r - c[:, 0])**2
        v_x_r = np.maximum(0, c[:, 0] + r - 1)**2
        v_y_b = np.maximum(0, r - c[:, 1])**2
        v_y_t = np.maximum(0, c[:, 1] + r - 1)**2
        wall_p = np.sum(v_x_l + v_x_r + v_y_b + v_y_t)
        
        # Overlap penalties
        dists = np.linalg.norm(c[:, np.newaxis, :] - c[np.newaxis, :, :], axis=2)
        r_sum = r[:, np.newaxis] + r[np.newaxis, :]
        np.fill_diagonal(dists, 1.0)
        overlap_p = np.sum(np.maximum(0, r_sum - dists)**2)
        
        total_penalty = wall_p + overlap_p
        
        # We want to maximize sum(r), so minimize -sum(r)
        # Add penalty
        return -np.sum(r) + LAMBDA * total_penalty

    # Bounds: radii >= 0, centers in [0,1] (loose bounds for optimizer)
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
    
    res = minimize(objective, x0, method='L-BFGS-B', bounds=bounds, options={'maxiter': 500})
    
    best_c = res.x[:2*n].reshape(-1, 2)
    best_r = res.x[2*n:]
    
    return {'centers': best_c, 'radii': best_r}
