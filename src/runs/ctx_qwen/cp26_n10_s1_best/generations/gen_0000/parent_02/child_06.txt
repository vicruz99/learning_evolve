import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    best_centers = None
    best_radii = None
    best_sum = 0.0

    # Helper to check if a packing is valid with a specific radius
    def check_packing(r):
        # Penalty function for a fixed radius r
        def objective(positions):
            centers = positions.reshape(n, 2)
            penalty = 0.0
            
            # Boundary penalties
            for i in range(n):
                x, y = centers[i]
                # Push away from walls: x must be >= r, <= 1-r, etc.
                if x < r:
                    penalty += (r - x)**2
                if x > 1 - r:
                    penalty += (x - (1 - r))**2
                if y < r:
                    penalty += (r - y)**2
                if y > 1 - r:
                    penalty += (y - (1 - r))**2
            
            # Overlap penalties
            for i in range(n):
                for j in range(i + 1, n):
                    dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                    min_dist = 2 * r
                    if dist < min_dist:
                        penalty += (min_dist - dist)**2
            return penalty

        # Try multiple random restarts to avoid local minima
        max_success = False
        final_positions = None
        
        for _ in range(3): # 3 restarts per radius check
            # Initialize positions randomly within valid bounds for r
            # Add a small buffer to ensure initial valid state if possible
            margin = 0.0
            if r > 0: margin = r + 1e-6
            else: margin = 1e-6
            
            # Ensure range is valid
            if 1 - 2*r < 0: 
                return False, None

            init_centers = np.random.uniform(margin, 1 - margin, (n, 2))
            init_positions = init_centers.flatten()
            
            # Optimization
            # Using L-BFGS-B for bounds handling is tricky with flattened array, 
            # so we rely on penalty function and bounds on position?
            # Actually, simple bounds on positions [0,1]
            bounds = [(0, 1)] * (2 * n)
            
            res = minimize(objective, init_positions, method='L-BFGS-B', bounds=bounds, 
                           options={'maxiter': 500, 'ftol': 1e-8})
            
            if res.fun < 1e-6:
                max_success = True
                final_positions = res.x
                break
        
        if max_success:
            return True, final_positions.reshape(n, 2)
        else:
            return False, None

    # Binary Search for r
    low = 0.0
    high = 0.5 # Max possible radius
    best_r = 0.0
    
    # Optimization iterations
    for _ in range(20):
        mid = (low + high) / 2.0
        feasible, centers = check_packing(mid)
        if feasible:
            low = mid
            best_r = mid
            best_centers = centers
        else:
            high = mid

    # If binary search didn't find a valid one or we want to refine sum of radii
    # We can try to optimize radii directly now, but equal radii is strong.
    # Let's just output the best found equal radii packing.
    if best_centers is not None:
        radii = np.full(n, best_r)
        sum_radii = np.sum(radii)
        return best_centers, radii, sum_radii
    
    # Fallback: Should not happen
    centers = np.random.rand(n, 2) * 0.9 + 0.05
    radii = np.full(n, 0.1)
    return centers, radii, 2.6
