import numpy as np
from scipy.optimize import minimize

def obj_displacement(x, ref_centers):
    """Objective: minimize displacement from reference centers during expansion phase."""
    n = len(ref_centers)
    c = x.reshape((n, 2))
    return np.sum((c - ref_centers)**2)

def constr_wall(x, radii):
    """Boundary constraints: circles must stay inside [0,1]^2."""
    xs = x[::2]
    ys = x[1::2]
    return np.concatenate([xs - radii, 1.0 - (xs + radii), ys - radii, 1.0 - (ys + radii)])

def constr_pairs(x, radii):
    """Pairwise non-overlap constraints: dist_sq >= (r1 + r2)^2"""
    xs = x[::2]
    ys = x[1::2]
    n = len(radii)
    vals = np.empty(n * (n - 1) // 2)
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            dx = xs[i] - xs[j]
            dy = ys[i] - ys[j]
            vals[idx] = dx * dx + dy * dy - (radii[i] + radii[j])**2
            idx += 1
    return vals

def final_obj(x):
    """Objective: maximize sum of radii (minimize negative sum)."""
    return -np.sum(x[2::3])

def final_wall(x):
    """Boundary constraints for joint optimization variables [x1,y1,r1, x2,y2,r2, ...]"""
    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]
    return np.concatenate([xs - rs, 1.0 - (xs + rs), ys - rs, 1.0 - (ys + rs)])

def final_pairs(x):
    """Pairwise constraints for joint optimization variables."""
    n = len(x) // 3
    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]
    vals = np.empty(n * (n - 1) // 2)
    idx = 0

        for j in range(i + 1, n):
            dx = xs[i] - xs[j]
            dy = ys[i] - ys[j]

    return vals

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    # Initial configuration: staggered hexagonal layout
    rows = [6, 5, 6, 5, 4]
    centers = np.zeros((n, 2))
    idx = 0
    r_base = 0.075
    dy = r_base * np.sqrt(3)
    
    for i, count in enumerate(rows):
        y = r_base + i * dy
        total_w = (count - 1) * 2 * r_base
        x_start = (1.0 - total_w) / 2.0
        if i % 2 == 1:
            x_start += r_base
        for j in range(count):
            x = x_start + j * 2 * r_base
            centers[idx] = [x, y]
            idx += 1

    for _ in range(120):

            {'type': 'ineq', 'fun': constr_pairs, 'args': (radii,)}
        ]
        bounds = [(0.0, 1.0) for _ in range(2 * n)]
        
        res = minimize(obj_displacement, x0, args=(centers,), method='SLSQP',
                       bounds=bounds, constraints=cons, options={'maxiter': 80, 'ftol': 1e-9})
                       
        if res.success:
            new_centers = res.x.reshape((n, 2))
            # Verify strict feasibility before accepting
            valid = True

                for j in range(i + 1, n):
                    d = np.linalg.norm(new_centers[i] - new_centers[j])

                        valid = False
                        break
                if not valid:
                    break
                    
            if valid:
                centers = new_centers
            else:
                radii /= 1.003
                break

            break

    return opt_centers, opt_radii, float(np.sum(opt_radii))
