import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Pack 26 circles in a unit square [0,1]x[0,1] to maximize the sum of radii.
    
    Returns:
        centers: np.array of shape (26, 2) with (x, y) coordinates
        radii: np.array of shape (26) with radius of each circle
        sum_radii: float sum of radii
    """
    n = 26
    np.random.seed(42)

    # Step 1: Generate initial centers using a hexagonal lattice
    # 5 rows of 5 and one row of 1, shifted to allow a slightly larger radius than a strict grid
    initial_positions = []
    
    # Create a base grid of 5x5
    rows = 5
    cols = 5
    # Add an offset for hexagonal packing
    # We will try to fit 26 points. 5x5=25, so we add one.
    # A common perturbation for N=26 is to shift rows.
    
    # Initialize a 5x5 grid
    for i in range(rows):
        for j in range(cols):
            x = (j + 0.5) / 5.0
            y = (i + 0.5) / 5.0
            # Add hexagonal shift
            if i % 2 == 1:
                x += 0.1 # Shift odd rows
            initial_positions.append([x, y])
            
    # We need 26 points. We have 25. 
    # Insert the 26th point in a gap or replace a point if needed.
    # For simplicity in the solver, we can just use 26 random points 
    # or a specific pattern. Let's try to add a point in the center 
    # or a corner gap if space permits. 
    # Actually, a randomized start near a grid is robust.
    
    # Let's restart with a more robust initial guess: 
    # 26 points distributed somewhat evenly.
    # We can use a hexagonal packing for 26 points.
    # 6 rows: 5, 4, 5, 4, 5, 3 (sum 26)
    
    centers_init = []
    row_counts = [5, 4, 5, 4, 5, 3]
    y_step = 1.0 / (len(row_counts) + 1) # 6 rows -> 7 slots? 
    # Let's just space them by 1/7
    
    current_y = 1.0 / (len(row_counts) + 1)
    for i, count in enumerate(row_counts):
        # x positions
        x_step = 1.0 / (count + 1)
        for j in range(count):
            x = x_step * (j + 1)
            # Shift odd rows for hex packing
            if i % 2 == 1:
                x += x_step / 2.0
            centers_init.append([x, current_y])
        current_y += y_step

    centers_init = np.array(centers_init[:26]) # Ensure we have exactly 26

    # Step 2: Define the optimization problem
    # We want to maximize r such that all circles fit.
    # This is equivalent to maximizing the minimum distance between centers (d_ij)
    # and the minimum distance to the boundary.
    # Let f(x) = min( min_i dist(center_i, boundary), min_{i!=j} dist(center_i, center_j)/2 )
    # We want to maximize f(x). So we minimize -f(x).

    def objective(params):
        centers = params.reshape(-1, 2)
        min_dist = 1.0 # Initialize with a large value

        # Distance to boundaries
        for i in range(n):
            x, y = centers[i]
            dist_boundary = min(x, 1-x, y, 1-y)
            if dist_boundary < min_dist:
                min_dist = dist_boundary

        # Distance between circles
        # We only need the radius r. If min_dist_between_centers < 2r, they overlap.
        # We are optimizing centers. The radius is determined by the tightest constraint.
        # Let's assume equal radii r. Then r <= dist_boundary and r <= dist_center_center / 2.
        
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                if dist < min_dist:
                    min_dist = dist

        # The maximum equal radius we can fit is min(min_dist_boundary, min_dist_center/2)
        # Wait, min_dist calculated above includes both boundary and center-center distances?
        # No, in the loop above, I compared dist (center-center) with min_dist (which holds boundary dist).
        # But center-center dist must be >= 2r. Boundary dist must be >= r.
        # So if we define min_val = min(all boundary dists, all center-center dists / 2),
        # then r = min_val.
        
        # Let's recalculate properly.
        val = 1.0
        
        # Check boundaries
        for i in range(n):
            x, y = centers[i]
            d_bound = min(x, 1-x, y, 1-y)
            if d_bound < val:
                val = d_bound
        
        # Check centers
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                # Constraint: dist >= 2r => r <= dist/2
                r_limit = dist / 2.0
                if r_limit < val:
                    val = r_limit
                    
        return -val # We want to maximize val

    # Initial guess for parameters (flattened centers)
    x0 = centers_init.flatten()

    # Use a derivative-free optimizer like Nelder-Mead or Powell
    # L-BFGS-B is good but requires gradients (approximated). Nelder-Mead is robust for this.
    # Given the non-convex nature, we might want to try multiple starts or a specific method.
    # Let's use Powell or Nelder-Mead.
    
    # Bounds for centers: [0, 1] for each coordinate.
    bounds = [(0, 1) for _ in range(2 * n)]

    # Run optimization
    # Nelder-Mead doesn't support bounds directly, but we can penalize or clamp.
    # L-BFGS-B supports bounds.
    
    res = minimize(objective, x0, method='L-BFGS-B', bounds=bounds, options={'maxiter': 1000, 'ftol': 1e-12})

    best_centers = res.x.reshape(-1, 2)
    best_r = -res.fun

    # Step 3: Refine radii
    # The objective function maximized the minimum radius r assuming equal radii.
    # However, we can potentially have unequal radii to increase the sum?
    # Usually, for this problem, equal radii is near optimal.
    # Let's verify the radius constraint and set radii.
    
    radii = np.full(n, best_r)
    
    # Verify and adjust if necessary (though optimizer should have handled it)
    # Recalculate max possible radius for each circle based on final positions
    # If we allow unequal radii, we could potentially squeeze more sum.
    # But let's stick to the equal radius solution first as it's robust.
    # The objective returned -r, so r = -res.fun.
    
    # Let's double check the radius calculation from the centers
    # r_i = min( min_j (dist_ij / 2), min_wall )
    # Since we optimized for equal r, all r_i should be roughly best_r.
    
    # Calculate actual max radii for each circle to ensure validity
    final_radii = np.zeros(n)
    for i in range(n):
        r_min = 1.0
        # Wall distance
        x, y = best_centers[i]
        r_wall = min(x, 1-x, y, 1-y)
        if r_wall < r_min:
            r_min = r_wall
        
        # Circle distances
        for j in range(n):
            if i == j: continue
            dist = np.sqrt(np.sum((best_centers[i] - best_centers[j])**2))
            r_circ = dist / 2.0
            if r_circ < r_min:
                r_min = r_circ
        final_radii[i] = r_min

    # If the optimization resulted in unequal clearances, final_radii will capture it.
    # However, L-BFGS-B on the equal-r objective forces them to be equal.
    # Let's use the final_radii calculated above.
    
    # Re-run a local optimizer on radii to maximize sum?
    # If centers are fixed, r_i is fixed by geometry.
    # So we can't change radii independently without moving centers.
    
    sum_radii = np.sum(final_radii)
    
    return best_centers, final_radii, sum_radii

# Example usage
if __name__ == "__main__":
    c, r, s = run_packing()
    print(f"Sum of radii: {s}")
    print(f"Min radius: {np.min(r)}")
    print(f"Max radius: {np.max(r)}")
