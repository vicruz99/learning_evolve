import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Pack 26 circles in a unit square to maximize the sum of radii.
    
    Returns:
        centers: np.array of shape (26, 2)
        radii: np.array of shape (26,)
        sum_radii: float
    """
    n_circles = 26
    # Optimization variables: [x1, y1, r1, x2, y2, r2, ..., x26, y26, r26]
    # Total variables = 26 * 3 = 78
    n_vars = n_circles * 3
    
    # ---------------------------------------------------------
    # 1. Generate Initial Guess (Hexagonal Grid)
    # ---------------------------------------------------------
    centers = np.zeros((n_circles, 2))
    radii = np.zeros(n_circles)
    
    # We try to pack circles with radius ~0.1 initially.
    # Hexagonal spacing: vertical distance sqrt(3)*r, horizontal 2r.
    r_init = 0.09 
    # Scale factor for grid generation to fit roughly in [0.1, 0.9]
    # We'll generate points and select valid ones.
    
    pts = []
    # Grid parameters
    # We iterate over rows and columns to generate hexagonal points
    # y = r + k * sqrt(3)*r
    # x = r + m * 2r  (shifted by r for odd rows)
    
    # Heuristic bounds for grid generation
    max_k = 10
    max_m = 10
    
    for k in range(max_k):
        y = r_init + k * np.sqrt(3) * r_init
        if y > 1 - r_init:
            break
        for m in range(max_m):
            # Shift for odd rows to create hexagonal pattern
            shift = r_init if k % 2 == 1 else 0
            x = r_init + shift + m * 2 * r_init
            if x > 1 - r_init:
                break
            pts.append((x, y))
            
    # If we don't have enough points in the strict grid, we can add some or just use what we have
    # But we need exactly 26. The grid with r=0.09 should have enough.
    # Let's verify and trim/pad if necessary (though padding with small circles is safe)
    
    # Shuffle or just take first 26. Taking first 26 from top-left is fine.
    if len(pts) < n_circles:
        # Fallback: create a grid if hex failed or didn't generate enough
        # Simple grid
        pts = []
        step = 0.2
        for x in np.arange(0.1, 1.0, step):
            for y in np.arange(0.1, 1.0, step):
                pts.append((x, y))
    
    # Select 26 points
    selected_pts = pts[:n_circles]
    
    for i, (x, y) in enumerate(selected_pts):
        centers[i] = [x, y]
        radii[i] = r_init

    # Concatenate into optimization vector
    x0 = np.zeros(n_vars)
    for i in range(n_circles):
        x0[3*i] = centers[i, 0]
        x0[3*i+1] = centers[i, 1]
        x0[3*i+2] = radii[i]

    # ---------------------------------------------------------
    # 2. Define Objective Function
    # ---------------------------------------------------------
    def objective(x):
        # x is [x1, y1, r1, x2, y2, r2, ...]
        # We want to maximize sum(radii), so minimize -sum(radii)
        r = x[2::3] # Radii are at indices 2, 5, 8, ...
        return -np.sum(r)

    # ---------------------------------------------------------
    # 3. Define Constraints
    # ---------------------------------------------------------
    # Inequality constraints g(x) >= 0
    
    def constraints(x):
        # x is vector of size 78
        # Unpack
        xs = x[0::3]
        ys = x[1::3]
        rs = x[2::3]
        
        cons = []
        
        # Boundary constraints:
        # x_i >= r_i  => x_i - r_i >= 0
        # 1 - x_i >= r_i => 1 - x_i - r_i >= 0
        # same for y
        for i in range(n_circles):
            xi, yi, ri = xs[i], ys[i], rs[i]
            cons.append(xi - ri)          # Left wall
            cons.append(1.0 - xi - ri)    # Right wall
            cons.append(yi - ri)          # Bottom wall
            cons.append(1.0 - yi - ri)    # Top wall
            
        # Overlap constraints:
        # dist(i, j) >= ri + rj
        # dist - ri - rj >= 0
        # We compute squared distance to avoid sqrt in loop, but constraint is linear in r?
        # No, dist is sqrt. We must use sqrt.
        # However, checking all pairs is O(N^2).
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                dx = xs[i] - xs[j]
                dy = ys[i] - ys[j]
                dist = np.sqrt(dx*dx + dy*dy)
                cons.append(dist - rs[i] - rs[j])
                
        return np.array(cons)

    # Define constraint dict for scipy
    cons = {'type': 'ineq', 'fun': constraints}

    # ---------------------------------------------------------
    # 4. Bounds
    # ---------------------------------------------------------
    # x, y in [0, 1], r >= 0
    # Actually, x, y bounds can be tighter [0,1], r bounds [0, 0.5]
    bounds = []
    for i in range(n_circles):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 0.5)) # r (radius cannot exceed 0.5)

    # ---------------------------------------------------------
    # 5. Run Optimization
    # ---------------------------------------------------------
    # Using SLSQP
    # To avoid getting stuck in poor local minima, we might run multiple starts,
    # but with a good hexagonal start, one run might suffice.
    # We'll try one robust run.
    
    # Increase max iterations to allow convergence
    result = minimize(
        objective, 
        x0, 
        method='SLSQP', 
        bounds=bounds, 
        constraints=cons,
        options={'maxiter': 2000, 'ftol': 1e-12, 'disp': False}
    )
    
    # Extract result
    x_opt = result.x
    centers_opt = np.zeros((n_circles, 2))
    radii_opt = np.zeros(n_circles)
    
    for i in range(n_circles):
        centers_opt[i] = [x_opt[3*i], x_opt[3*i+1]]
        radii_opt[i] = x_opt[3*i+2]
        
    sum_r = np.sum(radii_opt)
    
    # Final validation check (optional but good for debugging)
    # The validation function is provided in the prompt, we assume it works.
    # We just return the values.
    
    return centers_opt, radii_opt, float(sum_r)
