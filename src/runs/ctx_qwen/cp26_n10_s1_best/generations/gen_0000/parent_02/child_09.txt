import numpy as np
from scipy.optimize import minimize
import random

def run_packing():
    """
    Pack 26 circles in a unit square to maximize the sum of radii.
    Returns (centers, radii, sum_radii)
    """
    n_circles = 26
    best_sum_radii = -1.0
    best_centers = None
    best_radii = None

    # Hex grid spacing
    s = 0.2
    pts = []
    # Generate points in a grid

        for col in range(7):
            x = col * s + (row % 2) * s / 2.0
            y = row * s * np.sqrt(3) / 2.0
            if 0 <= x <= 1 and 0 <= y <= 1:
                pts.append((x, y))
    
    # Select 26 points from the generated grid
    pts = pts[:n_circles]
    base_centers = np.array(pts)

    for _ in range(50):
        # Perturb initial centers slightly
        centers = base_centers + np.random.normal(0, 0.02, base_centers.shape)

        centers = np.clip(centers, 0.01, 0.99)
        
        # Initial radii: small but non-zero
        radii = np.full(n_circles, 0.05)
        
        # Combine into a single optimization vector: [x0, y0, r0, x1, y1, r1, ...]
        x0 = np.hstack([centers.flatten(), radii])
        
        # Define objective function: Minimize -sum(radii) + Penalty
        def objective(x):
            cs = x[:2 * n_circles].reshape(n_circles, 2)
            rs = x[2 * n_circles:]
            
            # 1. Boundary constraints
            penalty = 0.0
            for i in range(n_circles):
                x_c, y_c = cs[i]
                r = rs[i]
                # Penalty for going outside [0, 1]
                if x_c - r < 0: penalty += (r - x_c)**2
                if x_c + r > 1: penalty += (x_c + r - 1)**2
                if y_c - r < 0: penalty += (r - y_c)**2
                if y_c + r > 1: penalty += (y_c + r - 1)**2
            
            # 2. Overlap constraints
            for i in range(n_circles):
                for j in range(i + 1, n_circles):
                    dist = np.sqrt(np.sum((cs[i] - cs[j])**2))
                    if dist < rs[i] + rs[j]:
                        penalty += (rs[i] + rs[j] - dist)**2
            
            # Objective: Maximize sum of radii => Minimize -sum
            return -np.sum(rs) + 1000.0 * penalty
        
        # Bounds: centers in [0, 1], radii in [0, 0.5]
        bounds = [(0.0, 1.0) for _ in range(2 * n_circles)] + [(0.0, 0.5) for _ in range(n_circles)]
        
        # Optimize
        res = minimize(objective, x0, method='L-BFGS-B', bounds=bounds, options={'maxiter': 1000, 'ftol': 1e-9})
        
        # Extract results
        opt_centers = res.x[:2 * n_circles].reshape(n_circles, 2)

        current_sum = np.sum(opt_radii)
        if current_sum > best_sum_radii:
            best_sum_radii = current_sum
            best_centers = opt_centers
            best_radii = opt_radii

    for i in range(n_circles):

        r = best_radii[i]
        best_centers[i, 0] = np.clip(x, r, 1 - r)
        best_centers[i, 1] = np.clip(y, r, 1 - r)
        
    # Recalculate sum after potential clamping
    final_sum = np.sum(best_radii)
    
    return (best_centers, best_radii, final_sum)
