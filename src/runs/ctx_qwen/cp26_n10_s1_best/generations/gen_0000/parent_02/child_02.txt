import numpy as np


def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Pack 26 circles in a unit square to maximize sum of radii.
    Uses force-based optimization with multiple restarts.
    """
    n = 26

    def create_hexagonal_grid(n, seed=0, perturbation=0.04):
        """Create a hexagonal grid of n points with random perturbation."""
        np.random.seed(seed)
        centers = np.zeros((n, 2))
        idx = 0

        # Try different spacings and pick the best fit
        best_centers = None
        best_score = -1

        for spacing_mult in np.linspace(0.85, 1.15, 7):
            cur_centers = np.zeros((n, 2))
            cur_idx = 0

            base_spacing = 0.21 * spacing_mult
            spacing_y = base_spacing * np.sqrt(3) / 2

            y = 0.12
            row = 0
            while cur_idx < n and y < 0.9:
                x = 0.10 if row % 2 == 0 else 0.10 + base_spacing / 2
                while x < 0.90 and cur_idx < n:
                    cur_centers[cur_idx] = [x, y]
                    cur_idx += 1
                    x += base_spacing
                y += spacing_y
                row += 1

            if cur_idx >= n:
                cur_centers = cur_centers[:n]
                # Add perturbation
                perturbed = cur_centers + np.random.uniform(
                    -perturbation, perturbation, size=(n, 2)
                )
                perturbed = np.clip(perturbed, 0.08, 0.92)

                # Score: sum of minimum distances (prefer well-separated)
                min_dists = np.zeros(n)
                for i in range(n):
                    for j in range(n):
                        if i != j:
                            d = np.sqrt(np.sum((perturbed[i] - perturbed[j]) ** 2))
                            min_dists[i] = max(min_dists[i], d)
                score = np.sum(min_dists)

                if score > best_score:
                    best_score = score
                    best_centers = perturbed.copy()

        return best_centers

    def optimize_force_based(centers, radii, n, max_iter=100000, base_grow=1.5e-6):
        """
        Force-based optimization that grows circles while maintaining feasibility.
        """
        alpha = 0.12  # Step size for position updates
        grow_rate = base_grow

        for iteration in range(max_iter):
            forces = np.zeros_like(centers)

            # Inter-circle repulsion forces
            for i in range(n):
                for j in range(i + 1, n):
                    diff = centers[i] - centers[j]
                    dist_sq = np.sum(diff ** 2)
                    dist = np.sqrt(max(dist_sq, 1e-20))

                    min_dist = radii[i] + radii[j]
                    overlap = min_dist - dist

                    if overlap > 1e-12:
                        # Strong repulsion when overlapping
                        force_mag = overlap * 250.0
                        unit_dir = diff / dist
                        forces[i] += unit_dir * force_mag
                        forces[j] -= unit_dir * force_mag
                    elif dist < min_dist + 0.02:
                        # Mild repulsion to maintain some buffer
                        force_mag = 0.5 / (dist + 0.01)
                        unit_dir = diff / dist
                        forces[i] += unit_dir * force_mag
                        forces[j] -= unit_dir * force_mag

            # Boundary forces
            margin = 1e-7
            for i in range(n):
                x, y = centers[i]
                r = radii[i]

                if x - r < margin:
                    forces[i, 0] += (margin - (x - r)) * 600.0
                if x + r > 1.0 - margin:
                    forces[i, 0] -= (x + r - (1.0 - margin)) * 600.0
                if y - r < margin:
                    forces[i, 1] += (margin - (y - r)) * 600.0
                if y + r > 1.0 - margin:
                    forces[i, 1] -= (y + r - (1.0 - margin)) * 600.0

            # Apply forces
            centers = centers + alpha * forces
            centers = np.clip(centers, margin, 1.0 - margin)

            # Grow radii
            radii = radii + grow_rate

            # Decay parameters
            alpha = alpha * 0.99997
            if iteration % 15000 == 0 and iteration > 0:
                grow_rate = grow_rate * 0.6

        return centers, radii

    def validate_and_get_sum(centers, radii, n):
        """Validate packing and return sum of radii if valid, else 0."""
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if r < 0:
                return 0.0
            if x - r < -1e-11 or x + r > 1.0 + 1e-11:
                return 0.0
            if y - r < -1e-11 or y + r > 1.0 + 1e-11:
                return 0.0

        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist < radii[i] + radii[j] - 1e-11:
                    return 0.0

        return np.sum(radii)

    def try_configuration(seed, perturbation=0.04, initial_radius=0.035):
        """Try a single configuration and return (centers, radii, sum)."""
        centers = create_hexagonal_grid(n, seed=seed, perturbation=perturbation)
        radii = np.ones(n) * initial_radius

        centers_opt, radii_opt = optimize_force_based(
            centers.copy(), radii.copy(), n
        )

        s = validate_and_get_sum(centers_opt, radii_opt, n)
        return centers_opt, radii_opt, s

    # Phase 1: Broad search with different configurations
    best_sum = 0.0
    best_centers = None
    best_radii = None

    for trial in range(300):
        seed = trial * 137 + 42
        perturbation = 0.03 + (trial % 5) * 0.01
        initial_radius = 0.03 + (trial % 3) * 0.005

        centers_opt, radii_opt, s = try_configuration(
            seed, perturbation, initial_radius
        )

        if s > best_sum:
            best_sum = s
            best_centers = centers_opt.copy()
            best_radii = radii_opt.copy()

    # Phase 2: Fine-tune the best configuration with more iterations
    if best_centers is not None:
        for refinement in range(20):
            seed = 1000 + refinement * 71
            centers_perturbed = best_centers.copy()
            centers_perturbed = centers_perturbed + np.random.uniform(
                -0.005, 0.005, size=(n, 2)
            )
            centers_perturbed = np.clip(centers_perturbed, 0.05, 0.95)

            radii_perturbed = best_radii.copy()
            radii_perturbed = radii_perturbed * (0.95 + refinement * 0.002)

            centers_opt, radii_opt = optimize_force_based(
                centers_perturbed, radii_perturbed, n, max_iter=120000, base_grow=1e-6
            )

            s = validate_and_get_sum(centers_opt, radii_opt, n)
            if s > best_sum:
                best_sum = s
                best_centers = centers_opt.copy()
                best_radii = radii_opt.copy()

    # Phase 3: Try corner-optimized configurations
    for trial in range(50):
        np.random.seed(2000 + trial)

        # Place circles with emphasis on corners and edges
        centers = np.zeros((n, 2))
        radii = np.zeros(n)

        # 4 corner circles (larger)
        corner_r = 0.11 + np.random.uniform(-0.01, 0.01)
        centers[0] = [corner_r, corner_r]
        radii[0] = corner_r
        centers[1] = [1 - corner_r, corner_r]
        radii[1] = corner_r
        centers[2] = [corner_r, 1 - corner_r]
        radii[2] = corner_r
        centers[3] = [1 - corner_r, 1 - corner_r]
        radii[3] = corner_r

        # Fill remaining with hexagonal pattern
        idx = 4
        spacing = 0.18
        for row in range(8):
            y = 0.15 + row * spacing * np.sqrt(3) / 2
            x_off = spacing / 2 if row % 2 == 0 else 0
            x = 0.15 + x_off
            while x < 0.85 and idx < n:
                centers[idx] = [x, y]
                radii[idx] = 0.06
                idx += 1
                x += spacing
            if idx >= n:
                break

        centers += np.random.uniform(-0.02, 0.02, size=(n, 2))
        centers = np.clip(centers, 0.05, 0.95)
        radii = np.clip(radii, 0.03, 0.15)

        centers_opt, radii_opt = optimize_force_based(
            centers.copy(), radii.copy(), n, max_iter=80000, base_grow=2e-6
        )

        s = validate_and_get_sum(centers_opt, radii_opt, n)
        if s > best_sum:
            best_sum = s
            best_centers = centers_opt.copy()
            best_radii = radii_opt.copy()

    # Phase 4: Additional fine-tuning with very slow growth
    if best_centers is not None:
        for final_trial in range(30):
            np.random.seed(3000 + final_trial)

            centers_perturbed = best_centers.copy()
            centers_perturbed = centers_perturbed + np.random.uniform(
                -0.003, 0.003, size=(n, 2)
            )
            centers_perturbed = np.clip(centers_perturbed, 0.02, 0.98)

            # Start with slightly smaller radii and grow very slowly
            radii_start = best_radii * 0.90

            centers_opt, radii_opt = optimize_force_based(
                centers_perturbed, radii_start, n, max_iter=150000, base_grow=8e-7
            )

            s = validate_and_get_sum(centers_opt, radii_opt, n)
            if s > best_sum:
                best_sum = s
                best_centers = centers_opt.copy()
                best_radii = radii_opt.copy()

    return best_centers, best_radii, best_sum
