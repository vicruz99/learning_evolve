import numpy as np
from scipy.optimize import minimize

def compute_overlap_con(params, n, i, j):
    """Compute overlap constraint: dist(i,j) - r_i - r_j >= 0"""
    xi, yi = params[2 * i], params[2 * i + 1]
    xj, yj = params[2 * j], params[2 * j + 1]
    ri, rj = params[2 * n + i], params[2 * n + j]
    dx = xi - xj
    dy = yi - yj
    dist = np.sqrt(dx * dx + dy * dy)
    return dist - ri - rj

def compute_bnd_xmin(params, n, i):
    """Boundary constraint: x_i - r_i >= 0"""
    return params[2 * i] - params[2 * n + i]

def compute_bnd_xmax(params, n, i):
    """Boundary constraint: 1 - x_i - r_i >= 0"""
    return 1.0 - params[2 * i] - params[2 * n + i]

def compute_bnd_ymin(params, n, i):
    """Boundary constraint: y_i - r_i >= 0"""
    return params[2 * i + 1] - params[2 * n + i]

def compute_bnd_ymax(params, n, i):
    """Boundary constraint: 1 - y_i - r_i >= 0"""
    return 1.0 - params[2 * i + 1] - params[2 * n + i]

def obj_func(params, n):
    """Negative sum of radii (to maximize)"""
    return -np.sum(params[2 * n:])

def penalty_obj(params, n, weight=1000.0):
    """Objective with penalty for constraint violations."""
    centers = params[:2 * n].reshape(n, 2)

        total_penalty += max(0.0, r - (1.0 - x)) ** 2
        total_penalty += max(0.0, r - y) ** 2
        total_penalty += max(0.0, r - (1.0 - y)) ** 2

            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)

    return -np.sum(radii) + weight * total_penalty

def optimize_with_slsqp(x0, n):
    """Run SLSQP optimization."""
    bounds = []
    for i in range(n):
        bounds.extend([(0.0, 1.0), (0.0, 1.0)])
        bounds.append((0.0, 0.5))
    
    constraints = []
    for i in range(n):
        constraints.append({'type': 'ineq', 'fun': lambda p, ii=i: compute_bnd_xmin(p, n, ii)})
        constraints.append({'type': 'ineq', 'fun': lambda p, ii=i: compute_bnd_xmax(p, n, ii)})
        constraints.append({'type': 'ineq', 'fun': lambda p, ii=i: compute_bnd_ymin(p, n, ii)})
        constraints.append({'type': 'ineq', 'fun': lambda p, ii=i: compute_bnd_ymax(p, n, ii)})
    
    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({'type': 'ineq', 'fun': lambda p, ii=i, jj=j: compute_overlap_con(p, n, ii, jj)})
    
    result = minimize(
        lambda p: obj_func(p, n),
        x0,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 5000, 'ftol': 1e-14, 'disp': False}
    )
    return result
