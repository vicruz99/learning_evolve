import numpy as np


def compute_forces_and_update(centers, radii, n, dt, growth_rate):
    """Compute forces and update positions/radii for circle packing."""
    forces = np.zeros((n, 2))
    
    for i in range(n):
        for j in range(i + 1, n):
            diff = centers[i] - centers[j]
            dist_sq = np.sum(diff ** 2)
            dist = np.sqrt(dist_sq + 1e-12)
            min_dist = radii[i] + radii[j]
            
            if dist < min_dist:
                overlap = min_dist - dist
                force_mag = overlap / (dist + 1e-12) * 80.0
                direction = diff / (dist + 1e-12)
                forces[i] += force_mag * direction * 0.5
                forces[j] -= force_mag * direction * 0.5
        
        # Wall repulsion forces
        r = radii[i]
        if centers[i, 0] < r + 1e-8:
            forces[i, 0] += (r - centers[i, 0] + 1e-8) * 100.0
        if centers[i, 0] > 1.0 - r - 1e-8:
            forces[i, 0] -= (centers[i, 0] - (1.0 - r) + 1e-8) * 100.0
        if centers[i, 1] < r + 1e-8:
            forces[i, 1] += (r - centers[i, 1] + 1e-8) * 100.0
        if centers[i, 1] > 1.0 - r - 1e-8:
            forces[i, 1] -= (centers[i, 1] - (1.0 - r) + 1e-8) * 100.0
    
    # Update positions
    centers = centers + dt * forces
    centers = np.clip(centers, 1e-9, 1.0 - 1e-9)
    
    # Update radii: grow toward maximum possible
    for i in range(n):
        max_r = float('inf')
        for j in range(n):

                available = d - radii[j]
                if available < max_r:
                    max_r = available
        
        wall_d = min(centers[i, 0], 1.0 - centers[i, 0],
                     centers[i, 1], 1.0 - centers[i, 1])
        if wall_d < max_r:
            max_r = wall_d
        
        # Smooth growth
        target = max_r * 0.9995
        if target > radii[i]:
            radii[i] = radii[i] + growth_rate * (target - radii[i])
        else:
            radii[i] = max(radii[i] * 0.999, 1e-9)
    
    return centers, radii


def run_single_packing(centers_init, radii_init, n, seed=None):
    """Run a single packing simulation from given initial conditions."""
    if seed is not None:
        np.random.seed(seed)
    
    centers = centers_init.copy()
    radii = radii_init.copy()
    
    n_steps = 40000
    
    for step in range(n_steps):
        progress = step / n_steps
        
        # Cooling schedule
        dt = 0.015 * (1.0 - progress * 0.6)
        growth_rate = 0.003 * (1.0 - progress * 0.2)
        
        centers, radii = compute_forces_and_update(centers, radii, n, dt, growth_rate)
    
    # Final cleanup: ensure strict validity
    for i in range(n):
        r = radii[i]
        x, y = centers[i]
        max_valid_r = min(x - 1e-10, 1.0 - x - 1e-10,
                         y - 1e-10, 1.0 - y - 1e-10)
        radii[i] = min(r, max_valid_r)
        radii[i] = max(radii[i], 0.0)
    
    return centers, radii
