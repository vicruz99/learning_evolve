import numpy as np
import cvxpy as cp
from scipy.optimize import differential_evolution

def solve_lp_for_radii(centers):
    """
    Given fixed centers, solve the LP to find radii that maximize sum of radii.
    
    Args:
        centers: np.array of shape (n, 2)
        
    Returns:
        optimal_sum_radii: float
        optimal_radii: np.array of shape (n,)
    """
    n = centers.shape[0]
    
    # Variables: radii r_0, ..., r_{n-1}
    r = cp.Variable(n)
    
    # Objective: Maximize sum of radii
    objective = cp.Maximize(cp.sum(r))
    
    constraints = []
    
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    # Which implies r_i <= min(x_i, 1-x_i, y_i, 1-y_i)
    for i in range(n):
        x, y = centers[i]
        constraints.append(r[i] <= x)
        constraints.append(r[i] <= 1 - x)
        constraints.append(r[i] <= y)
        constraints.append(r[i] <= 1 - y)
        constraints.append(r[i] >= 0) # Radii must be non-negative
        
    # Non-overlap constraints: r_i + r_j <= distance(c_i, c_j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            constraints.append(r[i] + r[j] <= dist)
            
    problem = cp.Problem(objective, constraints)
    
    try:
        # Solve with ECOS solver which is usually reliable for LP
        problem.solve(solver=cp.ECOS, verbose=False)
        
        if problem.status in ["optimal", "optimal_inaccurate"]:
            return problem.value, np.array(r.value).flatten()
        else:
            # If LP fails, return 0 and small radii to prevent crashing
            # This shouldn't happen often with valid centers
            return 0.0, np.zeros(n)
    except Exception:
        return 0.0, np.zeros(n)
