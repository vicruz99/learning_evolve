import numpy as np
from scipy.optimize import minimize

def objective_and_penalty(v, penalty_weight):
    """
    Computes the objective function: negative sum of radii plus penalty for constraint violations.
    v: flattened array of (x, y, r) for 26 circles. Shape (78,)
    penalty_weight: weight for the constraint violations.
    """
    n = 26
    C = v.reshape(n, 3)
    centers = C[:, :2]
    radii = C[:, 2]

    x = centers[:, 0]
    y = centers[:, 1]
    r = radii

    # Boundary constraint violations: circle must be inside [0,1]x[0,1]
    # Violations are positive when constraints are breached, 0 otherwise.
    v_left   = np.maximum(0, r - x)
    v_right  = np.maximum(0, r - (1.0 - x))
    v_bottom = np.maximum(0, r - y)
    v_top    = np.maximum(0, r - (1.0 - y))
    bound_pen = np.sum(v_left**2) + np.sum(v_right**2) + np.sum(v_bottom**2) + np.sum(v_top**2)

    # Overlap constraint violations: dist(c_i, c_j) >= r_i + r_j
    diff = centers[:, None, :] - centers[None, :, :]
    dist = np.sqrt(np.sum(diff**2, axis=2) + 1e-12)
    sum_r = r[:, None] + r[None, :]
    overlap = np.maximum(0, sum_r - dist)
    
    # Only count each pair once (upper triangle)
    triu_overlap = np.triu(overlap, k=1)
    overlap_pen = np.sum(triu_overlap**2)

    total_penalty = bound_pen + overlap_pen
    obj = -np.sum(r) + penalty_weight * total_penalty
    return obj

                y = 0.1 + i * 0.14 + 0.001 * (n - idx)
                init_centers.append([x, y, 0.05])
                idx += 1
    x0 = np.array(init_centers).flatten()

    # Bounds: x, y in [0, 1], r in [1e-7, 0.5]
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)] * n + [(1e-7, 0.5)] * n

    penalty = 50.0
    best_v = x0

    # Iteratively increase penalty to enforce constraints strictly
    for step in range(12):
        res = minimize(objective_and_penalty, best_v, args=(penalty,), method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': 8000, 'ftol': 1e-10, 'gtol': 1e-6})
        best_v = res.x
        penalty *= 2.0
        if res.success or res.nit < 50:
            break

    C = best_v.reshape(n, 3)
    centers_opt = C[:, :2]
    radii_opt = C[:, 2]
    
    # Ensure non-negative radii and clip to bounds just in case of numerical drift
    radii_opt = np.maximum(radii_opt, 0.0)
    centers_opt = np.clip(centers_opt, 0.0, 1.0)
    
    sum_radii = np.sum(radii_opt)
    return centers_opt, radii_opt, sum_radii
