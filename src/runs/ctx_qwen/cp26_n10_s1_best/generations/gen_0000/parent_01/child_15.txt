import numpy as np
from scipy.optimize import minimize


def compute_energy(params, penalty_weight):
    """Compute objective with penalty for constraint violations.
    
    Objective is negative sum of radii (minimize) plus penalties.
    """
    n = 26
    centers = params[:52].reshape(n, 2)
    radii = params[52:]

    energy = -np.sum(radii)

    # Boundary penalties
    for i in range(n):
        x = centers[i, 0]
        y = centers[i, 1]
        r = radii[i]
        if r < 0:
            energy += penalty_weight * r * r
        else:
            if x < r:
                energy += penalty_weight * (r - x) ** 2
            if x + r > 1.0:
                energy += penalty_weight * (x + r - 1.0) ** 2
            if y < r:
                energy += penalty_weight * (r - y) ** 2
            if y + r > 1.0:
                energy += penalty_weight * (y + r - 1.0) ** 2

    # Overlap penalties
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            sum_r = radii[i] + radii[j]
            if dist < sum_r:
                energy += penalty_weight * (sum_r - dist) ** 2

    return energy


def init_hexagonal(seed):
    """Initialize with hexagonal pattern for 26 circles."""
    np.random.seed(seed)
    n = 26
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.095

    idx = 0
    for row in range(6):
        # Alternate row lengths for hexagonal packing
        ncols = 5 + (row % 2)
        for col in range(ncols):
            if idx < n:
                x = 0.06 + col * 0.17 + (0.085 if row % 2 == 1 else 0.0)
                y = 0.06 + row * 0.175
                centers[idx] = [x, y]
                idx += 1
            else:
                break
        if idx >= n:
            break

    # Add small random perturbation
    centers += np.random.randn(n, 2) * 0.012
    radii = np.maximum(radii + np.random.randn(n) * 0.008, 0.02)

    return np.concatenate([centers.flatten(), radii])


def init_random(seed):
    """Initialize with random positions and radii."""
    np.random.seed(seed + 10000)
    n = 26
    centers = np.random.uniform(0.12, 0.88, (n, 2))
    radii = np.random.uniform(0.04, 0.11, n)
    return np.concatenate([centers.flatten(), radii])


def init_grid(seed):
    """Initialize with a perturbed grid pattern."""
    np.random.seed(seed + 20000)
    n = 26
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.08

    idx = 0
    for row in range(6):
        for col in range(5):
            if idx < n:
                x = 0.12 + col * 0.17
                y = 0.08 + row * 0.17
                centers[idx] = [x, y]
                idx += 1
            else:
                break
        if idx >= n:
            break

    centers += np.random.randn(n, 2) * 0.02
    radii = np.maximum(radii + np.random.randn(n) * 0.01, 0.02)
    return np.concatenate([centers.flatten(), radii])


def refine_params(params):
    """Refine parameters using progressive penalty optimization."""
    bounds = [(0.001, 0.999)] * 52 + [(0.001, 0.5)] * 26

    # Progressive penalty: start loose, then tighten
    for pw in [200, 1000, 5000, 20000, 80000, 200000]:
        result = minimize(compute_energy, params, method='L-BFGS-B',
                         bounds=bounds, args=(pw,),
                         options={'maxiter': 15000, 'ftol': 1e-16})
        params = result.x

    return params


def check_valid(params):
    """Check if solution is valid according to constraints."""
    n = 26
    centers = params[:52].reshape(n, 2)
    radii = params[52:]

    for i in range(n):
        x = centers[i, 0]
        y = centers[i, 1]
        r = radii[i]
        if r <= 0:
            return False
        if x < r - 1e-10 or x > 1.0 - r + 1e-10:
            return False
        if y < r - 1e-10 or y > 1.0 - r + 1e-10:
            return False

    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            if dist < radii[i] + radii[j] - 1e-10:
                return False

    return True


def run_packing():
    """Main packing function that returns optimal configuration."""
    best_params = None
    best_sum = 0.0

    init_functions = [init_hexagonal, init_random, init_grid]

    # Try many different seeds and initialization methods
    for seed in range(80):
        for init_func in init_functions:
            params = init_func(seed)
            params = refine_params(params)
            
            current_sum = np.sum(params[52:])
            
            if check_valid(params) and current_sum > best_sum:
                best_sum = current_sum
                best_params = params.copy()

    # Fallback if no valid solution found
    if best_params is None:
        best_params = refine_params(init_hexagonal(0))
        best_sum = np.sum(best_params[52:])

    centers = best_params[:52].reshape(26, 2)
    radii = best_params[52:]

    return centers, radii, float(best_sum)
