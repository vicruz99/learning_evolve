import numpy as np
from scipy.optimize import minimize

N = 26

def objective_func(vars):
    """Objective: maximize sum of radii (minimize negative sum)"""
    return -np.sum(vars[N*2:])

def constraint_func(vars):
    """Vectorized constraint function for SLSQP"""
    centers = vars[:N*2].reshape(N, 2)
    radii = vars[N*2:]
    
    c = []
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    c.append(centers[:, 0] - radii)
    c.append((1.0 - centers[:, 0]) - radii)
    c.append(centers[:, 1] - radii)
    c.append((1.0 - centers[:, 1]) - radii)
    
    # Non-overlap constraints: dist(i,j) >= r_i + r_j
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist = np.sqrt(np.sum(diff**2, axis=2))
    r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    overlap = dist - r_sum
    
    # Extract upper triangle to avoid duplicates and self-constraints
    idx = np.triu_indices(N, k=1)
    c.append(overlap[idx])
    
    return np.concatenate(c)

def constraint_wrapper(vars):
    """Top-level wrapper for constraint function to avoid closures/lambdas"""
    return constraint_func(vars)

def create_initial_guess(seed):
    """Generate initial configuration using hexagonal lattice + random fallback"""
    np.random.seed(seed)
    r0 = 0.09
    centers = np.zeros((N, 2))
    radii = np.full(N, r0)
    
    idx = 0
    row = 0
    y = r0 + 0.02
    while idx < N and y < 1.0 - r0 - 0.02:
        x = r0 + 0.02
        if row % 2 == 1:
            x += r0
        while idx < N and x < 1.0 - r0 - 0.02:
            centers[idx] = [x, y]
            idx += 1
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        
    # Fill remaining positions if hex lattice wasn't enough
    if idx < N:
        while idx < N:
            pt = np.random.uniform(r0+0.01, 1.0-r0-0.01, 2)
            if idx > 0:
                d = np.min(np.sqrt(np.sum((centers[:idx] - pt)**2, axis=1)))
                if d < 2.0*r0 + 0.01:
                    continue
            centers[idx] = pt
            idx += 1
            
    return np.concatenate([centers.flatten(), radii])

def validate_local(centers, radii):
    """Strict validation matching the provided validation function"""
    n = centers.shape[0]
    if np.isnan(centers).any() or np.isnan(radii).any():
        return False
    for i in range(n):
        if radii[i] < -1e-12:
            return False
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            return False
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False
    return True

def run_packing():
    best_vars = None
    best_obj = -np.inf
    
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraint_wrapper}
    
    # Multiple restarts to escape local minima
    for seed in range(12):
        x0 = create_initial_guess(seed)
        try:
            res = minimize(objective_func, x0, method='SLSQP', 
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 300, 'ftol': 1e-8, 'disp': False})
            if res.success:
                curr_obj = -res.fun
                if curr_obj > best_obj:
                    c = res.x[:N*2].reshape(N, 2)
                    r = res.x[N*2:]
                    if validate_local(c, r):
                        best_vars = res.x
                        best_obj = curr_obj
        except Exception:
            continue
            
    # Refine the best solution found
    if best_vars is not None:
        res = minimize(objective_func, best_vars, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 500, 'ftol': 1e-9, 'disp': False})
        best_vars = res.x
        
    centers = best_vars[:N*2].reshape(N, 2)
    radii = best_vars[N*2:]
    
    return centers, radii, np.sum(radii)
