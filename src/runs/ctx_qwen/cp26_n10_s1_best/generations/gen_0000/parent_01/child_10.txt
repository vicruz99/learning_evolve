import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
import math

def compute_objective(params, n=26):
    """Negative sum of radii (we minimize)"""
    centers = params[:2*n].reshape(n, 2)
    radii = params[2*n:]
    return -np.sum(radii)

def compute_constraints(params, n=26):
    """
    Returns array of constraint values. Each constraint should be >= 0.
    Constraints:
    - For each pair (i,j): dist(i,j) - r_i - r_j >= 0
    - For each circle i: x_i - r_i >= 0, 1 - x_i - r_i >= 0, y_i - r_i >= 0, 1 - y_i - r_i >= 0
    - For each circle i: r_i >= 0 (we handle this with bounds)
    """
    centers = params[:2*n].reshape(n, 2)
    radii = params[2*n:]
    
    constraints_list = []
    
    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        constraints_list.append(x - r)  # >= 0
        constraints_list.append(1 - x - r)  # >= 0
        constraints_list.append(y - r)  # >= 0
        constraints_list.append(1 - y - r)  # >= 0
    
    # Non-overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            constraints_list.append(dist - radii[i] - radii[j])
    
    return np.array(constraints_list)

def init_hexagonal(n=26):
    """Initialize circles in a hexagonal pattern"""
    centers = np.zeros((n, 2))
    
    # Calculate approximate spacing
    # For hexagonal packing, rows are offset
    rows = int(math.ceil(math.sqrt(n * 2 / math.sqrt(3))))
    
    idx = 0
    row = 0
    while idx < n and row < rows:
        if row % 2 == 0:
            num_in_row = math.ceil(math.sqrt(n))
        else:
            num_in_row = math.ceil(math.sqrt(n)) - 1
        
        y = (row + 0.5) * 0.2
        for col in range(num_in_row):
            if idx >= n:
                break
            x = (col + 0.5 + (row % 2) * 0.5) * 0.22
            centers[idx] = [x, y]
            idx += 1
        row += 1
    
    # Scale to fit in [0,1]^2
    if centers.size > 0:
        centers[:, 0] = (centers[:, 0] - centers[:, 0].min()) / (centers[:, 0].max() - centers[:, 0].min() + 1e-12) * 0.8 + 0.1
        centers[:, 1] = (centers[:, 1] - centers[:, 1].min()) / (centers[:, 1].max() - centers[:, 1].min() + 1e-12) * 0.8 + 0.1
    
    return centers

def init_random(n=26):
    """Initialize circles randomly"""
    centers = np.random.rand(n, 2)
    return centers

def init_grid(n=26):
    """Initialize circles in a grid pattern"""
    centers = np.zeros((n, 2))
    cols = 5
    rows = 6
    idx = 0
    for r in range(rows):
        for c in range(cols):
            if idx >= n:
                break
            centers[idx] = [(c + 0.5) / cols, (r + 0.5) / rows]
            idx += 1
            if idx >= n:
                break
        if idx >= n:
            break
    return centers

def optimize_packing(centers_init, n=26, r_init=0.02):
    """Optimize packing starting from given centers"""
    params = np.concatenate([centers_init.flatten(), np.full(n, r_init)])
    
    # Bounds
    bounds = []
    for i in range(2 * n):
        bounds.append((0, 1))  # x and y coordinates
    for i in range(n):
        bounds.append((0, 0.5))  # radii
    
    # Constraints
    cons = {'type': 'ineq', 'fun': compute_constraints}
    
    # Run optimization
    result = minimize(
        compute_objective,
        params,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 1000, 'ftol': 1e-12}
    )
    
    return result

def force_based_optimization(centers, radii, n=26, iterations=5000, lr=1e-4):
    """Force-based optimization to increase radii"""
    for it in range(iterations):
        # Compute forces between circles
        forces = np.zeros((n, 2))
        
        for i in range(n):
            for j in range(i + 1, n):
                diff = centers[i] - centers[j]
                dist = np.sqrt(np.sum(diff ** 2)) + 1e-12
                desired_dist = radii[i] + radii[j]
                
                if dist < desired_dist:
                    # Repulsive force
                    force_magnitude = (desired_dist - dist) * 100 / dist
                    forces[i] += diff / dist * force_magnitude
                    forces[j] -= diff / dist * force_magnitude
        
        # Boundary forces
        for i in range(n):
            if centers[i][0] - radii[i] < 0:
                forces[i][0] += 100 * (radii[i] - centers[i][0])
            if centers[i][0] + radii[i] > 1:
                forces[i][0] -= 100 * (centers[i][0] + radii[i] - 1)
            if centers[i][1] - radii[i] < 0:
                forces[i][1] += 100 * (radii[i] - centers[i][1])
            if centers[i][1] + radii[i] > 1:
                forces[i][1] -= 100 * (centers[i][1] + radii[i] - 1)
        
        # Update positions
        centers += lr * forces
        
        # Gradually increase radii
        for i in range(n):
            radii[i] += 1e-6
        
        # Clip to bounds
        for i in range(n):
            centers[i][0] = max(radii[i], min(1 - radii[i], centers[i][0]))
            centers[i][1] = max(radii[i], min(1 - radii[i], centers[i][1]))
        
        if it % 1000 == 0:
            print(f"Iteration {it}, sum_radii={np.sum(radii):.6f}")
    
    return centers, radii
