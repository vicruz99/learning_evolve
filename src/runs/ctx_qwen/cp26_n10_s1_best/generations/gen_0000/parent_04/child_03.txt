import numpy as np
from scipy.optimize import minimize
from functools import partial
import math

def constraint_x_lower(x, i, n):
    return x[2 * i] - x[2 * n + i]

def constraint_x_upper(x, i, n):
    return 1.0 - x[2 * i] - x[2 * n + i]

def constraint_y_lower(x, i, n):
    return x[2 * i + 1] - x[2 * n + i]

def constraint_y_upper(x, i, n):
    return 1.0 - x[2 * i + 1] - x[2 * n + i]

def constraint_radius_nonneg(x, i, n):
    return x[2 * n + i]

def constraint_non_overlap(x, i, j, n):
    dx = x[2 * i] - x[2 * j]
    dy = x[2 * i + 1] - x[2 * j + 1]
    dist_sq = dx * dx + dy * dy
    if dist_sq < 1e-20:
        return -x[2 * n + i] - x[2 * n + j]
    dist = math.sqrt(dist_sq)
    return dist - x[2 * n + i] - x[2 * n + j]

def objective_func(x, n):
    return -np.sum(x[2 * n:])

def hexagonal_init(n):
    centers = []
    radii = []

    cols = 6
    rows = 5
    x_spacing = 0.165
    y_spacing = x_spacing * math.sqrt(3) / 2
    margin = 0.09

    count = 0
    for row in range(rows):
        for col in range(cols):
            if count >= n:
                break
            x = margin + col * x_spacing + (x_spacing / 2 if row % 2 == 1 else 0)
            y = margin + row * y_spacing

            if x <= 1 - margin and y <= 1 - margin:
                centers.append([x, y])
                radii.append(0.055)
                count += 1

    return np.array(centers[:n]), np.array(radii[:n])

def random_init(n, seed_val):
    np.random.seed(seed_val)
    centers = np.random.uniform(0.15, 0.85, (n, 2))
    radii = np.ones(n) * 0.03
    return centers, radii

def grid_init(n):
    cols = int(math.sqrt(n))
    rows = math.ceil(n / cols)

    if cols <= 1:
        cols = 5
    if rows <= 1:
        rows = 6

    x_spacing = 0.8 / (cols - 1) if cols > 1 else 0
    y_spacing = 0.8 / (rows - 1) if rows > 1 else 0

    centers = []
    radii = []

    count = 0
    for i in range(rows):
        for j in range(cols):
            if count >= n:
                break
            x = 0.1 + j * x_spacing
            y = 0.1 + i * y_spacing
            centers.append([x, y])
            radii.append(0.04)
            count += 1

    return np.array(centers[:n]), np.array(radii[:n])

def dense_hex_init(n):
    centers = []
    radii = []

    count = 0
    row = 0
    col = 0

    x_base = 0.12
    y_base = 0.12
    x_spacing = 0.155
    y_spacing = x_spacing * math.sqrt(3) / 2

    while count < n:
        x = x_base + col * x_spacing + (x_spacing / 2 if row % 2 == 1 else 0)
        y = y_base + row * y_spacing

        if x <= 1 - 0.12 and y <= 1 - 0.12:
            centers.append([x, y])
            radii.append(0.06)
            count += 1
            col += 1
        else:
            col = 0
            row += 1

    return np.array(centers[:n]), np.array(radii[:n])

def physics_optimize(centers_init, radii_init, n, num_iterations=30000):
    centers = centers_init.copy()
    radii = radii_init.copy()

    learning_rate = 0.008
    growth_rate = 0.00004
    damping = 0.95

    velocities = np.zeros((n, 2))

    for iteration in range(num_iterations):
        forces = np.zeros((n, 2))

        for i in range(n):
            for j in range(i + 1, n):
                diff = centers[i] - centers[j]
                dist_sq = np.sum(diff ** 2)
                dist = math.sqrt(dist_sq)
                min_dist = radii[i] + radii[j]

                if dist < min_dist * 1.3 and dist > 1e-10:
                    overlap = max(0, min_dist - dist)
                    strength = overlap / (dist + 1e-10) * 2.0
                    force = strength * diff
                    forces[i] += force
                    forces[j] -= force

            for dim in range(2):
                if centers[i][dim] < radii[i] + 0.005:
                    forces[i][dim] += (radii[i] - centers[i][dim] + 0.005) * 8.0
                if centers[i][dim] > 1 - radii[i] - 0.005:
                    forces[i][dim] -= (centers[i][dim] - (1 - radii[i]) + 0.005) * 8.0

        velocities = damping * velocities + learning_rate * forces

        centers += velocities

        for i in range(n):
            centers[i][0] = max(radii[i], min(1 - radii[i], centers[i][0]))
            centers[i][1] = max(radii[i], min(1 - radii[i], centers[i][1]))

        current_growth = growth_rate / (1 + iteration * 0.0002)
        radii += current_growth

    return centers, radii

def build_constraints(n):
    constraints = []

    for i in range(n):
        constraints.append({
            'type': 'ineq',
            'fun': partial(constraint_x_lower, i=i, n=n)
        })
        constraints.append({
            'type': 'ineq',
            'fun': partial(constraint_x_upper, i=i, n=n)
        })
        constraints.append({
            'type': 'ineq',
            'fun': partial(constraint_y_lower, i=i, n=n)
        })
        constraints.append({
            'type': 'ineq',
            'fun': partial(constraint_y_upper, i=i, n=n)
        })
        constraints.append({
            'type': 'ineq',
            'fun': partial(constraint_radius_nonneg, i=i, n=n)
        })

    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({
                'type': 'ineq',
                'fun': partial(constraint_non_overlap, i=i, j=j, n=n)
            })

    return constraints

def build_bounds(n):
    bounds = []
    for _ in range(2 * n):
        bounds.append((0.0, 1.0))
    for _ in range(n):
        bounds.append((0.0, 0.5))
    return bounds

def run_packing():
    n = 26

    best_sum = -float('inf')
    best_centers = None
    best_radii = None

    initial_configs = [
        hexagonal_init(n),
        dense_hex_init(n),
        grid_init(n),
        random_init(n, 0),
        random_init(n, 42),
        random_init(n, 123),
        random_init(n, 999),
    ]

    constraints = build_constraints(n)
    bounds = build_bounds(n)

    for centers_init, radii_init in initial_configs:
        centers_phys, radii_phys = physics_optimize(centers_init, radii_init, n, 25000)

        x0 = np.concatenate([centers_phys.flatten(), radii_phys])

        try:
            result = minimize(
                objective_func,
                x0,
                method='SLSQP',
                constraints=constraints,
                bounds=bounds,
                args=(n,),
                options={'maxiter': 1500, 'ftol': 1e-14}
            )

            current_sum = np.sum(result.x[2 * n:])
            centers_opt = result.x[:2 * n].reshape(n, 2)
            radii_opt = result.x[2 * n:]

            valid = True
            for i in range(n):
                if centers_opt[i][0] < radii_opt[i] - 1e-9:
                    valid = False
                    break
                if centers_opt[i][0] > 1 - radii_opt[i] + 1e-9:
                    valid = False
                    break
                if centers_opt[i][1] < radii_opt[i] - 1e-9:
                    valid = False
                    break
                if centers_opt[i][1] > 1 - radii_opt[i] + 1e-9:
                    valid = False
                    break
                if radii_opt[i] < 0:
                    valid = False
                    break

            if valid and current_sum > best_sum:
                best_sum = current_sum
                best_centers = centers_opt.copy()
                best_radii = radii_opt.copy()
        except Exception as e:
            pass

    if best_centers is None:
        centers, radii = hexagonal_init(n)
        best_sum = np.sum(radii)
        best_centers = centers
        best_radii = radii

    return best_centers, best_radii, best_sum
