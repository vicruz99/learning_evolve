def compute_violations(centers, radii):

    Compute the total violation of constraints (overlaps and boundaries).
    Returns a scalar value representing the sum of squared violations.
    """
    n = centers.shape[0]
    violation = 0.0

    # x - r >= 0  => r - x <= 0
    # x + r <= 1  => x + r - 1 <= 0
    # y - r >= 0  => r - y <= 0
    # y + r <= 1  => y + r - 1 <= 0
    # We penalize max(0, violation)

        r = radii[i]

        v = r - x

            violation += v**2

            violation += v**2

    # dist >= r_i + r_j  => r_i + r_j - dist <= 0
    for i in range(n):

            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))

    return violation
