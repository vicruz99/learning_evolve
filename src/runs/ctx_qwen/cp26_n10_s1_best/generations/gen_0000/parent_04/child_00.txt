import numpy as np
from scipy.optimize import minimize

def compute_penalty(params, n):
    """Computes the sum of squared constraint violations."""
    centers = params[:2*n].reshape(n, 2)
    radii = params[2*n:]
    
    # Boundary penalties: circle must be inside [0,1]x[0,1]
    # x >= r  => r - x <= 0
    pen_bound = np.sum(np.maximum(0, radii - centers[:, 0])**2)
    # x <= 1-r => r - (1-x) <= 0
    pen_bound += np.sum(np.maximum(0, radii - (1 - centers[:, 0]))**2)
    # y >= r
    pen_bound += np.sum(np.maximum(0, radii - centers[:, 1])**2)
    # y <= 1-r
    pen_bound += np.sum(np.maximum(0, radii - (1 - centers[:, 1]))**2)
    
    # Pairwise penalties: circles must not overlap
    # dist >= r_i + r_j  => r_i + r_j - dist <= 0
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    sum_r = radii[:, np.newaxis] + radii[np.newaxis, :]
    violations = np.maximum(0, sum_r - dists)
    np.fill_diagonal(violations, 0)
    
    pen_pairs = np.sum(violations**2)
    
    return pen_bound + pen_pairs

def objective(params, n, lam):
    """Objective function: maximize sum of radii (minimize negative sum) subject to penalties."""
    radii = params[2*n:]
    return -np.sum(radii) + lam * compute_penalty(params, n)

def get_bounds(n):
    """Returns bounds for the optimization variables."""
    bounds = []
    for _ in range(n):
        bounds.append((0.0, 1.0)) # x coordinates
        bounds.append((0.0, 1.0)) # y coordinates
    for _ in range(n):
        bounds.append((0.0, 0.5)) # radii
    return bounds

def generate_hex_start(n):
    """Generates a hexagonal lattice initialization for better packing density."""
    centers = []
    y = 0.15
    row = 0
    spacing = 0.18
    while len(centers) < n:
        x = 0.15
        if row % 2 == 1:
            x = 0.15 + spacing/2
        while x < 0.9 and len(centers) < n:
            centers.append([x, y])
            x += spacing
        y += spacing * np.sqrt(3)/2
        row += 1
    centers = np.array(centers[:n])
    radii = np.full(n, 0.08)
    return np.concatenate([centers.flatten(), radii])

def generate_starts(n, rng):
    """Generates multiple starting configurations."""
    starts = []
    
    # Random starts
    for _ in range(5):
        centers = rng.random((n, 2))
        radii = np.full(n, 0.02)
        starts.append(np.concatenate([centers.flatten(), radii]))
        
    # Grid start
    k = int(np.ceil(np.sqrt(n)))
    step = 0.9 / k
    offset = (1 - (k-1)*step)/2
    pts = []
    for i in range(k):
        for j in range(k):
            if len(pts) < n:
                pts.append([offset + i*step, offset + j*step])
    centers = np.array(pts)
    radii = np.full(n, 0.05)
    starts.append(np.concatenate([centers.flatten(), radii]))
    
    # Hexagonal start
    starts.append(generate_hex_start(n))
    
    return starts

def run_packing():
    n = 26
    bounds = get_bounds(n)
    best_centers = None
    best_radii = None
    best_sum = -np.inf
    
    rng = np.random.default_rng(42)
    starts = generate_starts(n, rng)
    
    for x0 in starts:
        lam = 10.0
        curr_x = x0.copy()
        
        # Annealing schedule for penalty parameter
        for step in range(4):
            lam *= 5
            res = minimize(
                objective,
                curr_x,
                args=(n, lam),
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 3000, 'ftol': 1e-12}
            )
            curr_x = res.x
            
            pen = compute_penalty(curr_x, n)
            if pen < 1e-10:
                break
        
        pen = compute_penalty(curr_x, n)
        if pen < 1e-6:
            radii = curr_x[2*n:]
            current_sum = np.sum(radii)
            if current_sum > best_sum:
                best_sum = current_sum
                best_centers = curr_x[:2*n].reshape(n, 2).copy()
                best_radii = radii.copy()
                
    if best_centers is None:
        # Fallback to a valid small packing
        best_centers = rng.random((n, 2))
        best_radii = np.full(n, 0.01)
        best_sum = np.sum(best_radii)
        
    return best_centers, best_radii, best_sum
