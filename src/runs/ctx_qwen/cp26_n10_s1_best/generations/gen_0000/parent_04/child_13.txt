import numpy as np
from scipy.optimize import minimize
import warnings
warnings.filterwarnings('ignore')

def compute_penalty(params, n):
    """Compute squared penalty for constraint violations."""
    r = params[:n]
    cx = params[n:2*n]
    cy = params[2*n:3*n]
    
    p = 0.0
    
    # Pairwise non-overlap constraints
    cx_diff = cx[:, None] - cx[None, :]
    cy_diff = cy[:, None] - cy[None, :]
    dist = np.sqrt(cx_diff**2 + cy_diff**2 + 1e-12)
    req = r[:, None] + r[None, :]
    violations = np.maximum(0.0, req - dist)
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    p += np.sum(violations[mask]**2)
    
    # Boundary constraints
    p += np.sum(np.maximum(0.0, r - cx)**2)
    p += np.sum(np.maximum(0.0, r - (1.0 - cx))**2)
    p += np.sum(np.maximum(0.0, r - cy)**2)
    p += np.sum(np.maximum(0.0, r - (1.0 - cy))**2)
    
    return p

def compute_loss_lam(params, n, lam):
    """Objective function: maximize sum of radii with penalty constraints."""
    return -np.sum(params[:n]) + lam * compute_penalty(params, n)

def get_initial_config():
    """Generate a hexagonal lattice initialization for 26 circles."""
    n = 26
    rows = [5, 6, 5, 6, 4]
    r_est = 0.09
    w = 2.0 * r_est
    h = np.sqrt(3) * r_est
    
    points = []
    row_y = r_est
    for i, count in enumerate(rows):
        row_x = r_est + (i % 2) * (w / 2.0)
        for j in range(count):
            points.append((row_x + j * w, row_y))
        row_y += h
        
    xs = [p[0] for p in points]
    ys = [p[1] for p in points]
    min_x, max_x = min(xs), max(xs)
    min_y, max_y = min(ys), max(ys)
    
    # Scale to fit comfortably within [0,1] with margin
    scale_x = 0.92 / (max_x - min_x)
    scale_y = 0.92 / (max_y - min_y)
    scale = min(scale_x, scale_y)
    
    cx = np.array([(p[0] - min_x) * scale + (1.0 - 0.92 * scale) / 2.0 for p in points])
    cy = np.array([(p[1] - min_y) * scale + (1.0 - 0.92 * scale) / 2.0 for p in points])
    r = np.full(n, r_est)
    
    return cx, cy, r

def run_packing():
    np.random.seed(42)
    n = 26
    bnds = [(1e-6, 0.5)] * n + [(1e-6, 1.0 - 1e-6)] * (2 * n)
    
    best_params = None
    best_sum = -np.inf
    
    # Multiple restarts to escape local minima
    for restart in range(15):
        cx, cy, r = get_initial_config()
        if restart > 0:
            cx += np.random.uniform(-0.015, 0.015, n)
            cy += np.random.uniform(-0.015, 0.015, n)
            cx = np.clip(cx, 1e-6, 1.0-1e-6)
            cy = np.clip(cy, 1e-6, 1.0-1e-6)
            
        params = np.concatenate([r, cx, cy])
        
        # Continuation method: gradually increase penalty weight
        lam = 50.0
        for step in range(25):
            try:
                res = minimize(compute_loss_lam, params, args=(n, lam), method='L-BFGS-B', 
                              bounds=bnds, options={'maxiter': 200, 'ftol': 1e-14})
                params = res.x
            except Exception:
                break
            lam *= 1.5
            
        pen = compute_penalty(params, n)
        # Accept solution if effectively feasible
        if pen < 1e-9:
            curr_sum = np.sum(params[:n])
            if curr_sum > best_sum:
                best_sum = curr_sum
                best_params = params.copy()
                
    # Fallback if optimization fails (unlikely)
    if best_params is None:
        cx, cy, r = get_initial_config()
        best_params = np.concatenate([r, cx, cy])
        best_sum = np.sum(r)
        
    centers = np.column_stack((best_params[n:2*n], best_params[2*n:3*n]))
    radii = best_params[:n]
    return centers, radii, best_sum
