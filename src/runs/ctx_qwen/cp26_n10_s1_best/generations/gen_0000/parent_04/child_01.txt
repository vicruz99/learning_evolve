            x = x_start + j * 2 * r_init

        x0[3*i+2] = radii[i]

    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)] * n + [(0.0, 0.5)] * n

        # We want to maximize sum(r_i), so we minimize -sum(r_i)
        # We use a penalty method for constraints
        
        x = vars[0::3]
        y = vars[1::3]
        r = vars[2::3]

        val = -np.sum(r)
        
        # Penalty parameters
        # Weights need to be high enough to enforce constraints
        w_boundary = 1000.0
        w_overlap = 10000.0

        # Circle i must be inside [0,1]x[0,1]
        # r <= x <= 1-r  =>  x - r >= 0  AND  x + r <= 1
        # r <= y <= 1-r  =>  y - r >= 0  AND  y + r <= 1
        
        # Check x bounds
        left_violation = np.maximum(0.0, r - x)
        right_violation = np.maximum(0.0, x + r - 1.0)

        # dist(i, j) >= r_i + r_j
        # dist^2 >= (r_i + r_j)^2
        # We check dist >= r_i + r_j directly

        # 26 circles -> 325 pairs. Loop is fine.

            for j in range(i + 1, n):
                xj, yj, rj = x[j], y[j], r[j]
                
                # Distance
                dx = xi - xj
                dy = yi - yj
                dist = np.sqrt(dx*dx + dy*dy)
                
                # Overlap amount
                overlap = ri + rj - dist
                
                if overlap > 0:
                    penalty += w_overlap * (overlap ** 2)
                    
        return val + penalty

    # Run optimization
    # L-BFGS-B is efficient for bound-constrained problems
    # We rely on the penalty to handle inequality constraints
    res = minimize(objective, x0, method='L-BFGS-B', bounds=bounds,

import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Packs 26 circles in a unit square [0,1]x[0,1] to maximize the sum of radii.
    
    Strategy:
    1. Initialize circles in a dense hexagonal pattern to ensure a valid starting configuration.
    2. Use scipy.optimize.minimize with L-BFGS-B method to maximize the sum of radii.
    3. Use a penalty function to handle non-linear constraints (non-overlap and boundary containment).
    """
    n = 26
    
    # 1. Initialization: Hexagonal Packing
    # We arrange 26 circles in 5 rows with counts [6, 5, 6, 5, 4].
    # This mimics a triangular lattice which is optimal for density.
    rows = [6, 5, 6, 5, 4]
    centers = []
    
    # Initial radius. A radius of 0.08 fits comfortably within the unit square
    # for this arrangement without overlaps, providing a valid start.
    r_init = 0.08
    
    # Vertical distance between rows in a hexagonal packing is r * sqrt(3)
    dy = r_init * np.sqrt(3)
    
    # Center the entire configuration vertically within [0, 1]
    # With 5 rows, the centers span a height of 4 * dy.
    y_base = 0.5 - 2 * dy
    
    for i, count in enumerate(rows):
        y = y_base + i * dy
        
        # Horizontal positioning
        # The width occupied by 'count' circles of radius r is count * 2r.
        width = count * 2 * r_init
        
        # Center the row horizontally within [0, 1]
        x_start = (1.0 - width) / 2 + r_init
        
        for j in range(count):
            x = x_start + j * 2 * r_init
            # In hexagonal packing, every other row is shifted horizontally by r
            # to nestle circles into the gaps of the previous row.
            if i % 2 == 1:
                x += r_init
            centers.append([x, y])
            
    centers = np.array(centers)
    radii = np.full(n, r_init)
    
    # Flatten variables for the optimizer: [x1, y1, r1, x2, y2, r2, ..., x26, y26, r26]
    x0 = np.zeros(3 * n)
    for i in range(n):
        x0[3*i] = centers[i, 0]
        x0[3*i+1] = centers[i, 1]
        x0[3*i+2] = radii[i]
        
    # Define bounds for variables
    # x, y coordinates must be in [0, 1]
    # radii must be non-negative and <= 0.5 (since diameter <= 1)
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)] * n + [(0.0, 0.5)] * n
    
    def objective(vars):
        # We want to maximize sum(radii), so we minimize -sum(radii).
        # We incorporate constraints into the objective via a penalty method.
        
        x = vars[0::3]
        y = vars[1::3]
        r = vars[2::3]
        
        # Primary objective term
        val = -np.sum(r)
        
        # Penalty weights
        # High weights ensure constraints are strictly satisfied
        w_boundary = 5000.0
        w_overlap = 10000.0
        
        penalty = 0.0
        
        # 1. Boundary Constraints: Circles must be inside [0, 1] x [0, 1]
        # This implies: r <= x <= 1-r  and  r <= y <= 1-r
        # Violations: x < r, x > 1-r, y < r, y > 1-r
        
        # Left boundary violation (x < r)
        left_v = np.maximum(0.0, r - x)
        # Right boundary violation (x > 1-r => x + r > 1)
        right_v = np.maximum(0.0, x + r - 1.0)
        # Bottom boundary violation (y < r)
        bottom_v = np.maximum(0.0, r - y)
        # Top boundary violation (y > 1-r => y + r > 1)
        top_v = np.maximum(0.0, y + r - 1.0)
        
        penalty += w_boundary * (np.sum(left_v**2) + np.sum(right_v**2) + 
                                 np.sum(bottom_v**2) + np.sum(top_v**2))

            for j in range(i + 1, n):

    # Calculate the final sum of radii
    sum_radii = float(np.sum(best_radii))
    
    return best_centers, best_radii, sum_radii
