                x += d
            y += d * np.sqrt(3) / 2
            row += 1
        return np.array(points[:n_points])

    def constraints(vars, n):
        cx = vars[:n]
        cy = vars[n:2*n]
        cr = vars[2*n:3*n]

        c_list.extend(1 - cx - cr)
        c_list.extend(cy - cr)
        c_list.extend(1 - cy - cr)
