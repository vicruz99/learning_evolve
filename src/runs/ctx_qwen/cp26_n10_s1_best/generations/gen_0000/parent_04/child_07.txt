import numpy as np
from scipy.optimize import minimize, linprog
import itertools

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    best_sum_r = -1.0
    best_centers = None
    best_radii = None

    def get_initial_configs():
        configs = []
        
        # 1. Rectangular Grid (6x5)
        x = np.linspace(0, 1, 6)
        y = np.linspace(0, 1, 5)
        cx, cy = np.meshgrid(x, y)
        centers = np.column_stack([cx.flatten(), cy.flatten()])[:n]
        configs.append(centers)

        # 2. Hexagonal Grid
        hex_centers = []
        r_start = 0.06 # Small initial radius to fit many
        for row in range(6):
            y_pos = r_start + row * np.sqrt(3) * r_start
            x_start = r_start if row % 2 == 0 else r_start * 1.5
            x_val = x_start
            while x_val <= 1 - r_start and len(hex_centers) < n:
                hex_centers.append([x_val, y_pos])
                x_val += 2 * r_start
        configs.append(np.array(hex_centers[:n]))

        # 3. Random
        np.random.seed(42)
        configs.append(np.random.uniform(0.05, 0.95, (n, 2)))

        return configs

    def linprog_pack(centers):
        """Refine radii for fixed centers using LP to maximize sum of radii."""
        n_c = centers.shape[0]
        c_obj = np.ones(n_c) * -1
        
        A_ub = []
        b_ub = []

        # Constraints: r_i + r_j <= dist_ij
        for i, j in itertools.combinations(range(n_c), 2):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            row = np.zeros(n_c)
            row[i] = 1; row[j] = 1
            A_ub.append(row)
            b_ub.append(dist)

        # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, etc.
        for i in range(n_c):
            x, y = centers[i]
            bounds_list = [x, 1 - x, y, 1 - y]
            for b in bounds_list:
                row = np.zeros(n_c)
                row[i] = 1
                A_ub.append(row)
                b_ub.append(b)

        A_ub = np.array(A_ub)
        b_ub = np.array(b_ub)
        bounds = [(0, None) for _ in range(n_c)]

        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
            if res.success:
                return res.x
        except Exception:
            pass
        return None

    def objective(vars, n):
        r = vars[2::3]
        return -np.sum(r)

    def constraint_boundary(vars, n):
        constraints = []
        for i in range(n):
            x, y, r = vars[3*i], vars[3*i+1], vars[3*i+2]
            constraints.extend([x - r, 1 - x - r, y - r, 1 - y - r])
        return constraints

    def constraint_overlap(vars, n):
        constraints = []
        for i in range(n):
            for j in range(i + 1, n):
                xi, yi, ri = vars[3*i], vars[3*i+1], vars[3*i+2]
                xj, yj, rj = vars[3*j], vars[3*j+1], vars[3*j+2]
                dx, dy = xi - xj, yi - yj
                dist_sq = dx**2 + dy**2
                constraints.append(dist_sq - (ri + rj)**2)
        return constraints

    def run_opt(centers_init):
        # Step 1: Refine radii for initial centers
        r_refined = linprog_pack(centers_init)
        if r_refined is None:
            r_refined = np.full(n, 0.05)
        
        # Ensure no zero radii to help solver
        r_refined = np.maximum(r_refined, 0.001)
        
        # Initial variables
        x0 = np.zeros(3 * n)
        for i in range(n):
            x0[3*i] = centers_init[i, 0]
            x0[3*i+1] = centers_init[i, 1]
            x0[3*i+2] = r_refined[i]

        # Define constraints for scipy
        cons = []
        # Boundary constraints
        for i in range(n):
            cons.append({'type': 'ineq', 'fun': lambda v, i=i: v[3*i] - v[3*i+2]})
            cons.append({'type': 'ineq', 'fun': lambda v, i=i: 1 - v[3*i] - v[3*i+2]})
            cons.append({'type': 'ineq', 'fun': lambda v, i=i: v[3*i+1] - v[3*i+2]})
            cons.append({'type': 'ineq', 'fun': lambda v, i=i: 1 - v[3*i+1] - v[3*i+2]})
        
        # Overlap constraints
        for i in range(n):
            for j in range(i + 1, n):
                cons.append({'type': 'ineq', 'fun': lambda v, i=i, j=j: 
                             (v[3*i]-v[3*j])**2 + (v[3*i+1]-v[3*j+1])**2 - (v[3*i+2]+v[3*j+2])**2})

        bounds = []
        for i in range(n):
            bounds.extend([(0, 1), (0, 1), (0, 0.5)]) # x, y, r bounds

        try:
            res = minimize(objective, x0, args=(n,), method='SLSQP', bounds=bounds, 
                           constraints=cons, options={'maxiter': 500, 'ftol': 1e-9})
            
            if res.success or res.fun < -0.1: # Check if we found a valid positive sum
                centers = np.column_stack([res.x[::3], res.x[1::3]])
                radii = res.x[2::3]
                return centers, radii, -res.fun
        except Exception:
            pass
        return None, None, 0.0

    configs = get_initial_configs()
    for centers in configs:
        c, r, s = run_opt(centers)
        if c is not None and s > best_sum_r:
            best_sum_r = s
            best_centers = c
            best_radii = r

    return best_centers, best_radii, best_sum_r
