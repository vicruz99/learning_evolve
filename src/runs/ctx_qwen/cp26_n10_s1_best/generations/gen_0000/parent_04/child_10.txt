import numpy as np
from scipy.optimize import minimize

def calculate_repulsive_energy(points, p_power=4):
    """
    Calculates the repulsive energy between points and boundaries.
    High energy indicates overlaps or points too close to boundaries.
    """
    n = points.shape[0]
    energy = 0.0
    
    # Inter-point repulsion (1/distance^p)
    # Using broadcasting for efficiency
    diffs = points[:, np.newaxis, :] - points[np.newaxis, :, :]
    dists_sq = np.sum(diffs**2, axis=2)
    
    # Mask out diagonal (self-distance) to avoid division by zero
    mask = ~np.eye(n, dtype=bool)
    dists_sq = dists_sq[mask]
    
    # Avoid division by zero for extremely close points by adding a small epsilon
    dists_sq = np.maximum(dists_sq, 1e-12)
    energy += np.sum(dists_sq ** (-p_power / 2.0))
    
    # Boundary repulsion (1/distance^p)
    # We penalize distance to x=0, x=1, y=0, y=1
    boundary_dists = np.concatenate([
        points[:, 0],           # distance to x=0
        1.0 - points[:, 0],     # distance to x=1
        points[:, 1],           # distance to y=0
        1.0 - points[:, 1]      # distance to y=1
    ])
    
    boundary_dists = np.maximum(boundary_dists, 1e-12)
    # Weight boundary repulsion slightly higher to ensure circles don't touch edges too closely
    energy += 2.0 * np.sum(boundary_dists ** (-p_power / 2.0))
    
    return energy
