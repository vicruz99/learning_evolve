import numpy as np
from scipy.optimize import linprog
import math

def compute_radii_lp(centers, n):
    """Compute optimal radii given fixed centers using linear programming.
    
    Maximizes sum(r_i) subject to:
    - r_i + r_j <= dist(i,j) for all pairs
    - r_i <= distance to boundary for each circle
    - r_i >= 0
    """
    c_obj = -np.ones(n)
    
    A_ub_rows = []
    b_ub_vals = []
    
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = math.sqrt(dx * dx + dy * dy)
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub_rows.append(row)
            b_ub_vals.append(dist)
        
        mx = centers[i, 0]
        my = centers[i, 1]
        max_r = min(mx, 1.0 - mx, my, 1.0 - my)
        row = np.zeros(n)
        row[i] = 1.0
        A_ub_rows.append(row)
        b_ub_vals.append(max(0.0, max_r))
    
    A_ub = np.array(A_ub_rows)
    b_ub = np.array(b_ub_vals)
    bounds = [(0.0, None)] * n
    
    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            return result.x
    except Exception:
        pass
    
    return np.ones(n) * 0.02


def initialize_hexagonal(n):
    """Create initial placement using hexagonal grid patterns."""
    best_centers = None
    best_score = -1.0
    
    for spacing in [0.18, 0.19, 0.20, 0.22, 0.24]:
        centers_try = np.zeros((n, 2))
        idx = 0
        
        for row in range(12):
            for col in range(12):
                if idx >= n:
                    break
                offset_x = spacing * 0.5 if row % 2 == 1 else 0.0
                x = col * spacing + offset_x + spacing * 0.5
                y = row * spacing * math.sqrt(3) / 2.0 + spacing * 0.5
                
                if x < 1.0 - spacing * 0.5 and y < 1.0 - spacing * 0.5:
                    centers_try[idx] = [x, y]
                    idx += 1
            if idx >= n:
                break
        
        if idx == n:
            radii = compute_radii_lp(centers_try, n)
            score = np.sum(radii)
            if score > best_score:
                best_score = score
                best_centers = centers_try.copy()
    
    if best_centers is None:
        best_centers = np.random.rand(n, 2) * 0.8 + 0.1
    
    return best_centers


def force_based_optimize(centers, radii, n, num_iters=150):
    """Push circles apart using repulsive forces."""
    c = centers.copy()
    r = radii.copy()
    
    for it in range(num_iters):
        forces = np.zeros((n, 2))
        cooling = 1.0 - (it / num_iters) * 0.5
        
        for i in range(n):
            for j in range(i + 1, n):
                dx = c[i, 0] - c[j, 0]
                dy = c[i, 1] - c[j, 1]
                dist_sq = dx * dx + dy * dy
                dist = math.sqrt(dist_sq) if dist_sq > 1e-12 else 1e-10
                min_dist = r[i] + r[j]
                
                if dist < min_dist + 0.005:
                    strength = (min_dist + 0.005 - dist) / dist * cooling
                    fx = dx * strength * 0.15
                    fy = dy * strength * 0.15
                    forces[i, 0] += fx
                    forces[i, 1] += fy
                    forces[j, 0] -= fx
                    forces[j, 1] -= fy
            
            x, y = c[i]
            ri = r[i]
            margin = 0.001
            if x - ri < margin:
                forces[i, 0] += 0.08 * cooling
            if x + ri > 1.0 - margin:
                forces[i, 0] -= 0.08 * cooling
            if y - ri < margin:
                forces[i, 1] += 0.08 * cooling
            if y + ri > 1.0 - margin:
                forces[i, 1] -= 0.08 * cooling
        
        c += forces
        c = np.clip(c, 0.002, 0.998)
        
        if it % 10 == 0:
            r = compute_radii_lp(c, n)
    
    r = compute_radii_lp(c, n)
    return c, r


def coordinate_ascent_refine(centers, radii, n):
    """Refine each circle's position one at a time."""
    c = centers.copy()
    r = radii.copy()
    
    for sweep in range(3):
        for i in range(n):
            best_sum = np.sum(r)
            best_pos = c[i].copy()
            
            for step_size in [0.02, 0.01, 0.005, 0.002]:
                directions = [
                    [step_size, 0.0], [-step_size, 0.0],
                    [0.0, step_size], [0.0, -step_size],
                    [step_size, step_size], [-step_size, step_size],
                    [step_size, -step_size], [-step_size, -step_size],
                ]
                
                improved = False
                for dx, dy in directions:
                    new_c = c.copy()
                    new_c[i] = c[i] + np.array([dx, dy])
                    new_c[i] = np.clip(new_c[i], 0.005, 0.995)
                    
                    new_r = compute_radii_lp(new_c, n)
                    new_sum = np.sum(new_r)
                    
                    if new_sum > best_sum + 1e-10:
                        best_sum = new_sum
                        best_pos = new_c[i].copy()
                        improved = True
                
                if not improved:
                    break
            
            c[i] = best_pos
        
        r = compute_radii_lp(c, n)
    
    return c, r


def run_packing():
    n = 26
    
    best_centers = None
    best_radii = None
    best_sum = 0.0
    
    for trial in range(25):
        centers = initialize_hexagonal(n)
        
        if trial > 0:
            centers += np.random.randn(n, 2) * 0.04
            centers = np.clip(centers, 0.05, 0.95)
        
        radii = compute_radii_lp(centers, n)
        
        centers, radii = force_based_optimize(centers, radii, n, num_iters=200)
        
        centers, radii = coordinate_ascent_refine(centers, radii, n)
        
        centers, radii = force_based_optimize(centers, radii, n, num_iters=100)
        
        centers, radii = coordinate_ascent_refine(centers, radii, n)
        
        s = np.sum(radii)
        if s > best_sum:
            best_sum = s
            best_centers = centers.copy()
            best_radii = radii.copy()
    
    return best_centers, best_radii, best_sum
