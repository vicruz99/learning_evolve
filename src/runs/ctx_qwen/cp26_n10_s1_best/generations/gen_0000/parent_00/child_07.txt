import numpy as np
from scipy.optimize import minimize
import math

def run_packing():
    """
    Pack 26 circles in a unit square [0,1]x[0,1] to maximize the sum of radii.
    Returns (centers, radii, sum_radii).
    """
    n = 26
    
    def get_radii(centers):
        """
        Computes the maximum possible radius for each circle given the centers,
        ensuring they fit in the unit square and do not overlap.
        """
        radii = np.full(n, 1.0)
        # Clip centers to [0, 1] to handle any out-of-bound values safely during optimization
        safe_centers = np.clip(centers, 0.0, 1.0)
        
        for i in range(n):
            x, y = safe_centers[i]
            # Distance to boundaries
            r_bound = min(x, 1.0 - x, y, 1.0 - y)
            radii[i] = r_bound
            
            # Distance to other centers (radius is half the distance to nearest neighbor)
            for j in range(n):
                if i != j:
                    dx = safe_centers[i, 0] - safe_centers[j, 0]
                    dy = safe_centers[i, 1] - safe_centers[j, 1]
                    dist = math.sqrt(dx*dx + dy*dy)
                    if dist / 2.0 < radii[i]:
                        radii[i] = dist / 2.0
        return radii

    def objective(x_flat):
        """
        Objective function to minimize: negative sum of radii.
        """
        centers = x_flat.reshape((n, 2))
        radii = get_radii(centers)
        return -np.sum(radii)

    # Initialization strategies
    inits = []
    
    # 1. 5x5 Grid + 1 point in a gap
    # 5x5 grid with spacing 0.2 gives radius 0.1. Sum = 2.5.
    # We add a point to try to improve.
    grid_x = np.linspace(0.1, 0.9, 5)
    grid_y = np.linspace(0.1, 0.9, 5)
    gx, gy = np.meshgrid(grid_x, grid_y)
    c1 = np.column_stack([gx.ravel(), gy.ravel()])
    # Add a point in a gap, e.g., (0.2, 0.2) is distance 0.141 from (0.1, 0.1)
    c1 = np.vstack([c1, [0.2, 0.2]])
    inits.append(c1)
    
    # 2. Random initialization
    np.random.seed(42)
    inits.append(np.random.rand(n, 2))
    
    # 3. Hexagonal packing initialization
    # Generates a hexagonal lattice and selects 26 points
    hex_pts = []
    s = 0.2  # Initial spacing estimate
    for row in range(10):
        for col in range(10):
            y = row * s * math.sqrt(3) / 2
            x = col * s + (row % 2) * s / 2
            if 0 <= x <= 1 and 0 <= y <= 1:
                hex_pts.append([x, y])
    
    if len(hex_pts) >= n:
        # Select a subset of points. Random selection is fine for initial guess.
        indices = np.random.choice(len(hex_pts), n, replace=False)
        inits.append(np.array([hex_pts[i] for i in indices]))
    else:
        # Fallback if not enough points generated
        inits.append(np.array(hex_pts + [np.random.rand(2) for _ in range(n - len(hex_pts))]))

    best_score = -np.inf
    best_res = None
    bounds = [(0, 1)] * (n * 2)
    
    # Optimization Loop
    for init_centers in inits:
        try:
            # Primary optimization with L-BFGS-B (supports bounds)
            res = minimize(objective, init_centers.flatten(), method='L-BFGS-B', bounds=bounds, 
                           options={'maxiter': 5000, 'ftol': 1e-12, 'gtol': 1e-8})
            
            centers_opt = res.x.reshape((n, 2))
            radii_opt = get_radii(centers_opt)
            score = np.sum(radii_opt)
            
            if score > best_score:
                best_score = score
                best_res = (centers_opt, radii_opt)
            
            # Secondary refinement with Powell (derivative-free, good for non-smooth objectives)
            # Powell does not support bounds, so we clip the result afterwards
            try:
                res2 = minimize(objective, res.x, method='Powell', 
                                options={'maxiter': 1000, 'ftol': 1e-12})
                centers_opt2 = res2.x.reshape((n, 2))
                centers_opt2 = np.clip(centers_opt2, 0.0, 1.0)
                radii_opt2 = get_radii(centers_opt2)
                score2 = np.sum(radii_opt2)
                
                if score2 > best_score:
                    best_score = score2
                    best_res = (centers_opt2, radii_opt2)
            except Exception:
                pass
                
        except Exception:
            continue

    if best_res:
        return best_res[0], best_res[1], best_score
    else:
        # Fallback to zeros if optimization fails completely
        return np.zeros((n, 2)), np.zeros(n), 0.0
