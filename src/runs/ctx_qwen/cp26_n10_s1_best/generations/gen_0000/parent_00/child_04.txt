We arrange 26 circles in alternating rows of 5 and 4 circles with initial radius 0.07, ensuring no overlaps and plenty of room for expansion.
2. **Optimization Formulation**: 
   - Variables: A 1D array of length $3 \times 26 = 78$ representing $(x_i, y_i, r_i)$ for each circle.
   - Objective: Minimize $-\sum r_i$ (equivalent to maximizing sum of radii).
   - Constraints: 
     - Boundary: $x_i - r_i \ge 0$, $1 - x_i - r_i \ge 0$, $y_i - r_i \ge 0$, $1 - y_i - r_i \ge 0$ for all $i$.

   - Bounds: $0 \le x_i, y_i \le 1$, $0 \le r_i \le 0.5$.
3. **Solver**: Use `scipy.optimize.minimize` with the `SLSQP` method, which handles inequality constraints and bounds efficiently. Provide an analytical gradient for the objective to accelerate convergence.

import numpy as np
from scipy.optimize import minimize

def boundary_constraints(vars):
    n = 26
    c = np.empty(4 * n)
    for i in range(n):
        idx = 4 * i
        c[idx] = vars[3*i] - vars[3*i+2]
        c[idx+1] = 1.0 - vars[3*i] - vars[3*i+2]
        c[idx+2] = vars[3*i+1] - vars[3*i+2]
        c[idx+3] = 1.0 - vars[3*i+1] - vars[3*i+2]
    return c

def pairwise_constraints(vars):
    n = 26
    num_cons = n * (n - 1) // 2
    c = np.empty(num_cons)
    idx = 0
    for i in range(n):
        xi, yi, ri = vars[3*i], vars[3*i+1], vars[3*i+2]
        for j in range(i + 1, n):
            xj, yj, rj = vars[3*j], vars[3*j+1], vars[3*j+2]
            c[idx] = (xi - xj)**2 + (yi - yj)**2 - (ri + rj)**2
            idx += 1
    return c

def objective(vars):
    return -np.sum(vars[2::3])

def objective_grad(vars):
    grad = np.zeros_like(vars)
    grad[2::3] = -1.0
    return grad

def run_packing():
    n = 26
    # Hexagonal initial layout for better convergence
    x0, y0, r0 = np.zeros(n), np.zeros(n), np.full(n, 0.07)
    idx = 0
    dx = 0.2
    dy = dx * np.sqrt(3) / 2.0
    
    for row in range(6):
        y_base = 0.1 + row * dy
        x_start = 0.1 if row % 2 == 0 else 0.2
        count = 5 if row % 2 == 0 else 4
        for col in range(count):
            if idx < n:
                x0[idx] = x_start + col * dx
                y0[idx] = y_base
                idx += 1
                
    # Fill any remaining slots (should be 0 for n=26)
    for k in range(idx, n):
        x0[k] = 0.5
        y0[k] = 0.5 + 0.02 * (k - idx)
        
    vars0 = np.zeros(3 * n)
    vars0[0::3] = x0
    vars0[1::3] = y0
    vars0[2::3] = r0
    
    bounds = [(0, 1)] * (2 * n) + [(0, 0.5)] * n
    cons = [
        {'type': 'ineq', 'fun': boundary_constraints},
        {'type': 'ineq', 'fun': pairwise_constraints}
    ]
    
    res = minimize(objective, vars0, method='SLSQP', jac=objective_grad,
                   bounds=bounds, constraints=cons, 
                   options={'maxiter': 800, 'ftol': 1e-12, 'disp': False})
                   
    vars_opt = res.x if res.success else vars0
        
    centers = np.zeros((n, 2))
    centers[:, 0] = vars_opt[0::3]
    centers[:, 1] = vars_opt[1::3]
    radii = np.maximum(vars_opt[2::3], 0.0)
    
    return centers, radii, np.sum(radii)
