import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    """
    n_circles = 26
    
    # 1. Generate Initial Configuration
    # We use a hexagonal-like staggered arrangement.
    # Target sum ~2.636 implies average r ~ 0.101.
    # We start with r = 0.09 to ensure validity, then optimize.
    
    centers = np.zeros((n_circles, 2))
    radii = np.full(n_circles, 0.09)
    
    # Layout: Staggered rows
    # Row configuration: 5, 5, 5, 5, 4, 2 (Total 26)
    # This allows tighter vertical packing than a square grid.
    
    row_counts = [5, 5, 5, 5, 4, 2]
    y_pos = 0.09 # Start near bottom boundary
    idx = 0
    
    # Estimate spacing based on initial radius
    dx = 2 * radii[0]
    dy = np.sqrt(3) * radii[0]
    
    for row_idx, count in enumerate(row_counts):
        # Stagger odd rows (0-indexed) or even rows depending on preference
        # Let's stagger rows 1, 3, 5... by shifting x by r
        offset = (radii[0] if row_idx % 2 == 1 else 0)
        
        # Calculate x positions for this row to center them
        # Total width occupied by 'count' circles with spacing dx is (count-1)*dx
        # We want to center this span in [0, 1]
        # But boundaries must be respected. 
        # Simple approach: space them evenly within [radius, 1-radius]
        
        # Refined positioning:
        # We want to fit 'count' circles. 
        # Let's just place them linearly with spacing dx, centered.
        total_width = (count - 1) * dx
        start_x = (1.0 - total_width) / 2.0
        
        for j in range(count):
            centers[idx, 0] = start_x + j * dx + offset
            centers[idx, 1] = y_pos
            idx += 1
            
        y_pos += dy
        
    # 2. Optimization
    # We flatten the variables: [x0, y0, r0, x1, y1, r1, ...]
    x0 = np.concatenate([centers.ravel(), radii])
    
    # Bounds for variables
    # x, y in [0, 1], r in [0, 0.5]
    bounds = []
    for i in range(n_circles):
        bounds.extend([(0, 1), (0, 1), (0, 0.5)])
        
    # Non-linear constraints
    # 1. Non-overlap: dist(i, j) >= r_i + r_j
    #    (x_i - x_j)^2 + (y_i - y_j)^2 >= (r_i + r_j)^2
    #    -> (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
    cons = []
    for i in range(n_circles):
        for j in range(i + 1, n_circles):
            idx_i_x, idx_i_y, idx_i_r = 3*i, 3*i+1, 3*i+2
            idx_j_x, idx_j_y, idx_j_r = 3*j, 3*j+1, 3*j+2
            
            def constraint_func(vars, i=i, j=j, 
                                ii_x=idx_i_x, ii_y=idx_i_y, ii_r=idx_i_r,
                                ij_x=idx_j_x, ij_y=idx_j_y, ij_r=idx_j_r):
                xi, yi, ri = vars[ii_x], vars[ii_y], vars[ii_r]
                xj, yj, rj = vars[ij_x], vars[ij_y], vars[ij_r]
                dist_sq = (xi - xj)**2 + (yi - yj)**2
                sum_r_sq = (ri + rj)**2
                return dist_sq - sum_r_sq

            cons.append({'type': 'ineq', 'fun': constraint_func})
            
    # 2. Boundary constraints: 
    #    x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
    #    Actually, these are easier handled by bounds if we fix r? 
    #    But r is variable. So we need nonlinear constraints or bounds dependent on r.
    #    SLSQP bounds are static. So we must add constraints.
    
    for i in range(n_circles):
        ii_x, ii_y, ii_r = 3*i, 3*i+1, 3*i+2
        
        def left(vars, i=i, ix=ii_x, ir=ii_r):
            return vars[ix] - vars[ir]
        def right(vars, i=i, ix=ii_x, ir=ii_r):
            return 1.0 - vars[ix] - vars[ir]
        def bottom(vars, i=i, iy=ii_y, ir=ii_r):
            return vars[iy] - vars[ir]
        def top(vars, i=i, iy=ii_y, ir=ii_r):
            return 1.0 - vars[iy] - vars[ir]
            
        cons.append({'type': 'ineq', 'fun': left})
        cons.append({'type': 'ineq', 'fun': right})
        cons.append({'type': 'ineq', 'fun': bottom})
        cons.append({'type': 'ineq', 'fun': top})

    # Objective function: maximize sum of radii -> minimize -sum(radii)
    def objective(vars):
        s = 0
        for i in range(n_circles):
            s += vars[3*i + 2]
        return -s

    # Run optimization
    # Using SLSQP
    result = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, 
                      options={'maxiter': 1000, 'ftol': 1e-9, 'disp': False})
    
    # Extract result
    if result.success:
        final_vars = result.x
    else:
        # Fallback to initial if failed, though unlikely
        final_vars = x0
        
    final_centers = final_vars[:2*n_circles].reshape((n_circles, 2))
    final_radii = final_vars[2*n_circles:]
    
    # Ensure non-negative radii
    final_radii = np.maximum(final_radii, 0.0)
    
    # Center circles if they are slightly out due to numerical error (clamp)
    # The constraints should handle this, but just in case
    for i in range(n_circles):
        r = final_radii[i]
        cx, cy = final_centers[i]
        final_centers[i, 0] = np.clip(cx, r, 1 - r)
        final_centers[i, 1] = np.clip(cy, r, 1 - r)

    sum_radii = np.sum(final_radii)
    
    return final_centers, final_radii, sum_radii
