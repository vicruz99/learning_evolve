import numpy as np
from scipy.optimize import linprog

def run_packing():
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    Uses an iterative approach: solve for optimal radii (LP) given centers,
    then update centers based on constraint violations (repulsive forces).
    """
    n = 26
    
    # Precompute constraint matrix A for LP: A_ub * r <= b_ub
    # Constraints types:
    # 1. r_i <= x_i        (Distance to left wall)
    # 2. r_i <= 1 - x_i    (Distance to right wall)
    # 3. r_i <= y_i        (Distance to bottom wall)
    # 4. r_i <= 1 - y_i    (Distance to top wall)
    # 5. r_i + r_j <= dist_ij (Non-overlap)
    
    num_boundary = 4 * n
    num_pairs = n * (n - 1) // 2
    num_constraints = num_boundary + num_pairs
    
    # Initialize A matrix
    A = np.zeros((num_constraints, n))
    
    # Set up boundary constraint rows
    # Rows 0 to n-1: r_i <= x_i
    # Rows n to 2n-1: r_i <= 1-x_i
    # Rows 2n to 3n-1: r_i <= y_i
    # Rows 3n to 4n-1: r_i <= 1-y_i
    for i in range(n):
        A[i, i] = 1.0
        A[n + i, i] = 1.0
        A[2*n + i, i] = 1.0
        A[3*n + i, i] = 1.0
        
    # Set up pairwise constraint rows
    # r_i + r_j <= dist
    row_idx = num_boundary
    for i in range(n):
        for j in range(i + 1, n):
            A[row_idx, i] = 1.0
            A[row_idx, j] = 1.0
            row_idx += 1
            
    # Objective: Maximize sum(r_i) <=> Minimize -sum(r_i)
    c = -np.ones(n)
    
    # Bounds for radii: r_i >= 0
    bounds = [(0, None)] * n
    
    # --- Initialization ---
    # Create an initial grid-like distribution of centers
    # 6 rows, approx 4-5 cols
    centers = np.zeros((n, 2))
    
    # Using a hexagonal lattice pattern initialization for better packing
    # Approximate side length for 26 circles in unit square
    # Density ~ 0.9, Area ~ 1. 26 * pi * r^2 * 0.9 ~ 1 => r ~ 0.11
    # Diameter ~ 0.22. Spacing ~ 0.22.
    # 1 / 0.22 ~ 4.5 rows/cols.
    
    rng = np.random.default_rng(42) # Fixed seed for reproducibility
    # Perturb a grid to avoid symmetry issues
    grid_x = np.linspace(0.15, 0.85, 6)
    grid_y = np.linspace(0.15, 0.85, 5)
    
    pts = []
    for y in grid_y:
        for x in grid_x:
            pts.append([x, y])
            
    # We have 30 points, take 26. 
    # Or just random initialization which is often robust with optimization.
    # Let's use random initialization for diversity.
    centers = rng.uniform(0.1, 0.9, size=(n, 2))
    
    # --- Optimization Loop ---
    max_iter = 300
    tol = 1e-7
    step_size = 0.02
    prev_sum = 0.0
    
    radii = np.zeros(n) # Placeholder
    
    for iteration in range(max_iter):
        # 1. Construct b vector based on current centers
        b = np.zeros(num_constraints)
        
        # Boundary constraints RHS
        for i in range(n):
            x, y = centers[i]
            b[i] = x
            b[n + i] = 1.0 - x
            b[2*n + i] = y
            b[3*n + i] = 1.0 - y
            
        # Pairwise constraints RHS (distances)
        dists = np.sqrt(np.sum((centers[:, np.newaxis, :] - centers[np.newaxis, :, :])**2, axis=2))
        row_idx = num_boundary
        for i in range(n):
            for j in range(i + 1, n):
                b[row_idx] = dists[i, j]
                row_idx += 1
        
        # 2. Solve LP
        # Using 'highs' method which is fast and robust
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        
        if res.success:
            radii = res.x
            current_sum = -res.fun
        else:
            # Fallback if LP fails (should not happen with r>=0 feasible)
            break
            
        # Check convergence
        if iteration > 0 and abs(current_sum - prev_sum) < 1e-9:
            break
        prev_sum = current_sum
        
        # 3. Calculate Forces
        forces = np.zeros_like(centers)
        
        # Boundary Forces
        # If constraint is active (slack < eps), push away from wall
        slack_tol = 1e-5
        
        # Left Wall: r_i <= x_i. Active if x_i - r_i ~ 0. Push Right (+x)
        slack_left = centers[:, 0] - radii
        mask_left = slack_left < slack_tol
        forces[mask_left, 0] += 1.0
        
        # Right Wall: r_i <= 1-x_i. Active if 1-x_i - r_i ~ 0. Push Left (-x)
        slack_right = (1.0 - centers[:, 0]) - radii
        mask_right = slack_right < slack_tol
        forces[mask_right, 0] -= 1.0
        
        # Bottom Wall: r_i <= y_i. Active if y_i - r_i ~ 0. Push Up (+y)
        slack_bottom = centers[:, 1] - radii
        mask_bottom = slack_bottom < slack_tol
        forces[mask_bottom, 1] += 1.0
        
        # Top Wall: r_i <= 1-y_i. Active if 1-y_i - r_i ~ 0. Push Down (-y)
        slack_top = (1.0 - centers[:, 1]) - radii
        mask_top = slack_top < slack_tol
        forces[mask_top, 1] -= 1.0
        
        # Pairwise Forces
        # If r_i + r_j <= dist_ij is active, push apart
        # Vector from j to i is (c_i - c_j)
        # Force on i is +direction, on j is -direction
        
        # We can vectorize this slightly or loop. Loop is clearer for N=26.
        row_idx = num_boundary
        for i in range(n):
            for j in range(i + 1, n):
                # Slack = dist - (r_i + r_j)
                slack = dists[i, j] - (radii[i] + radii[j])
                
                if slack < slack_tol:
                    # Calculate unit vector from j to i
                    vec = centers[i] - centers[j]
                    d = np.linalg.norm(vec)
                    if d > 1e-9:
                        direction = vec / d
                        forces[i] += direction
                        forces[j] -= direction
                row_idx += 1
                
        # 4. Update Centers
        centers += step_size * forces
        
        # Clip centers to valid range [epsilon, 1-epsilon] to keep valid radii > 0
        # Although LP handles 0, keeping centers strictly inside helps stability
        centers = np.clip(centers, 1e-4, 1.0 - 1e-4)
        
        # Decay step size
        step_size *= 0.995
        
    return centers, radii, sum(radii)
