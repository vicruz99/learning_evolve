import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def compute_penalty(centers, r):
    """Compute smooth penalty for boundary violations and overlaps."""
    N = centers.shape[0]
    p = 0.0
    # Boundary penalties
    for i in range(N):
        x, y = centers[i]
        if x < r: 
            p += (x - r)**2
        if x > 1 - r: 
            p += (x - (1 - r))**2
        if y < r: 
            p += (y - r)**2
        if y > 1 - r: 
            p += (y - (1 - r))**2
    # Overlap penalties
    for i in range(N):
        for j in range(i + 1, N):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx*dx + dy*dy)
            if dist < 2 * r:
                p += (2 * r - dist)**2
    return p

def objective_function(vars, N):

    centers = vars[:N*2].reshape(-1, 2)
    r = vars[-1]
    if r <= 1e-7:
        return 1e9
    # Large coefficient ensures constraints are respected
    return -r + 1000.0 * compute_penalty(centers, r)

def generate_initial_centers(N):

            y = r_idx * 0.86602540378
            centers[idx, 0] = x
            centers[idx, 1] = y
            idx += 1
    # Normalize to [0, 1] and center in square
    mx = centers.max()
    centers = centers / mx
    centers = 0.4 * centers + 0.3
    return centers
