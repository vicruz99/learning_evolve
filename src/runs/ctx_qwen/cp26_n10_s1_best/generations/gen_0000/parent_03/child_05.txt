I'll use an alternating optimization approach:
1. **Initialize** circles in a hexagonal pattern (efficient packing geometry)
2. **Alternating optimization loop:**
   - Fix radii, optimize positions using scipy.optimize (minimize overlap/boundary violations)
   - Fix positions, optimize radii using Linear Programming (maximize sum of radii subject to linear constraints)
3. **Multiple restarts** to avoid poor local optima
4. **LP for radii**: Given positions, finding optimal radii is a linear program: maximize sum(r_i) subject to r_i + r_j ≤ dist(i,j) and boundary constraints
5. **Force-based refinement** as a final polishing step

The key insight is that computing optimal radii for fixed positions is an LP problem, which is efficient and exact. Then optimizing positions for fixed radii uses smooth optimization. This alternating scheme converges to locally optimal solutions.

For the hexagonal initialization, I'll use 5 rows with counts [5, 6, 5, 6, 4] = 26 circles, which is a natural hexagonal arrangement.

import numpy as np
from scipy.optimize import minimize, linprog


def hexagonal_init(n=26):
    """Initialize circles in a hexagonal pattern."""
    row_counts = [5, 6, 5, 6, 4]
    r_init = 0.09
    centers = np.zeros((n, 2))
    
    y = r_init + 0.03
    y_spacing = np.sqrt(3) * r_init * 1.5
    idx = 0
    
    for row_idx, count in enumerate(row_counts):
        x_spacing = 2 * r_init
        row_width = (count - 1) * x_spacing
        x_start = (1 - row_width) / 2
        
        if row_idx % 2 == 1:
            x_start += r_init
        
        for col in range(count):
            if idx < n:
                centers[idx, 0] = x_start + col * x_spacing
                centers[idx, 1] = y
                idx += 1
        
        y += y_spacing
    
    return centers


def random_scatter_init(n=26, seed=42):
    """Initialize circles randomly scattered."""
    rng = np.random.RandomState(seed)
    centers = np.zeros((n, 2))
    for i in range(n):
        centers[i, 0] = rng.uniform(0.1, 0.9)
        centers[i, 1] = rng.uniform(0.1, 0.9)
    return centers


def grid_init(n=26):
    """Initialize circles in a grid pattern."""
    centers = np.zeros((n, 2))
    # 6x5 grid with 1 extra
    cols = 6
    rows = 5
    x_step = 0.8 / (cols - 1)
    y_step = 0.8 / (rows - 1)
    idx = 0
    for r in range(rows):
        for c in range(cols):
            if idx < n:
                centers[idx, 0] = 0.1 + c * x_step
                centers[idx, 1] = 0.1 + r * y_step
                idx += 1
    return centers


def position_penalty(centers_flat, radii, n=26):
    """Compute total constraint violation penalty for given positions and radii."""
    centers = centers_flat.reshape(n, 2)
    penalty = 0.0
    
    # Overlap penalties
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            overlap = radii[i] + radii[j] - dist
            if overlap > 0:
                penalty += overlap ** 2
    
    # Boundary penalties
    for i in range(n):
        x = centers[i, 0]
        y = centers[i, 1]
        r = radii[i]
        if x < r:
            penalty += (r - x) ** 2
        if x > 1 - r:
            penalty += (x + r - 1) ** 2
        if y < r:
            penalty += (r - y) ** 2
        if y > 1 - r:
            penalty += (y + r - 1) ** 2
    
    return penalty


def compute_optimal_radii_lp(centers, n=26):
    """Given positions, find radii maximizing sum using Linear Programming."""
    c = -np.ones(n)
    
    # Upper bounds from boundary constraints
    ub = np.zeros(n)
    for i in range(n):
        ub[i] = min(centers[i, 0], 1 - centers[i, 0],
                    centers[i, 1], 1 - centers[i, 1])
        ub[i] = max(0, ub[i])
    
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    num_constraints = n * (n - 1) // 2
    A_ub = np.zeros((num_constraints, n))
    b_ub = np.zeros(num_constraints)
    
    constraint_idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            A_ub[constraint_idx, i] = 1
            A_ub[constraint_idx, j] = 1
            b_ub[constraint_idx] = dist
            constraint_idx += 1
    
    bounds = [(0, ub[i]) for i in range(n)]
    
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success and result.fun is not None:
            return result.x
    except Exception:
        pass
    
    # Fallback: compute radii greedily
    return greedy_radii(centers, n)


def greedy_radii(centers, n=26):
    """Compute radii greedily for fixed positions."""
    radii = np.zeros(n)
    
    for _ in range(300):
        new_radii = np.zeros(n)
        for i in range(n):
            r = min(centers[i, 0], 1 - centers[i, 0],
                    centers[i, 1], 1 - centers[i, 1])
            for j in range(n):
                if i != j:
                    dx = centers[i, 0] - centers[j, 0]
                    dy = centers[i, 1] - centers[j, 1]
                    dist = np.sqrt(dx * dx + dy * dy)
                    r = min(r, dist - radii[j])
            new_radii[i] = max(0, r)
        
        if np.allclose(radii, new_radii, atol=1e-12):
            break
        radii = new_radii
    
    return radii


def optimize_positions_for_radii(centers, radii, n=26, num_restarts=3):
    """Given radii, optimize positions to minimize constraint violations."""
    best_centers = centers.copy()
    best_penalty = position_penalty(centers.flatten(), radii, n)
    
    for restart in range(num_restarts):
        if restart == 0:
            x0 = centers.flatten().copy()
        else:
            # Perturbed copy
            x0 = centers.flatten().copy()
            x0 += np.random.randn(2 * n) * 0.005
            x0 = np.clip(x0, 0.001, 0.999)
        
        result = minimize(
            position_penalty,
            x0,
            args=(radii, n),
            method='BFGS',
            options={'maxiter': 3000, 'gtol': 1e-10, 'disp': False}
        )
        
        if result.fun < best_penalty:
            best_penalty = result.fun
            best_centers = result.x.reshape(n, 2)
    
    return best_centers


def force_refine(centers, radii, n=26, num_steps=500):
    """Refine positions using force-directed method."""
    centers = centers.copy()
    
    for step in range(num_steps):
        forces = np.zeros_like(centers)
        
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                
                if dist < 1e-10:
                    dx = np.random.randn() * 1e-6
                    dy = np.random.randn() * 1e-6
                    dist = 1e-6
                
                overlap = radii[i] + radii[j] - dist
                if overlap > 0:
                    force_mag = overlap * 50.0 / dist
                    forces[i, 0] += force_mag * dx
                    forces[i, 1] += force_mag * dy
                    forces[j, 0] -= force_mag * dx
                    forces[j, 1] -= force_mag * dy
            
            # Boundary repulsion
            for d in range(2):
                if centers[i, d] < radii[i]:
                    forces[i, d] += (radii[i] - centers[i, d]) * 50.0
                if centers[i, d] > 1 - radii[i]:
                    forces[i, d] -= (centers[i, d] + radii[i] - 1) * 50.0
        
        lr = 0.002 / (1 + step * 0.005)
        centers += lr * forces
        
        # Clamp to valid region
        for i in range(n):
            centers[i, 0] = max(radii[i], min(1 - radii[i], centers[i, 0]))
            centers[i, 1] = max(radii[i], min(1 - radii[i], centers[i, 1]))
    
    return centers


def alternating_optimization(initial_centers, n=26, num_iterations=30):
    """Run alternating optimization: optimize positions, then radii."""
    centers = initial_centers.copy()
    radii = np.ones(n) * 0.03
    
    best_sum = 0
    best_centers = centers.copy()
    best_radii = radii.copy()
    
    for iteration in range(num_iterations):
        # Optimize positions for current radii
        centers = optimize_positions_for_radii(centers, radii, n, num_restarts=2)
        
        # Optimize radii for current positions using LP
        radii = compute_optimal_radii_lp(centers, n)
        
        current_sum = np.sum(radii)
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers.copy()
            best_radii = radii.copy()
        
        # Force refine to help escape local optima
        centers = force_refine(centers, radii, n, num_steps=100)
        radii = compute_optimal_radii_lp(centers, n)
        
        current_sum = np.sum(radii)
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers.copy()
            best_radii = radii.copy()
    
    return best_centers, best_radii


def run_packing():
