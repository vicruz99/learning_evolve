def calculate_objective_and_penalty(params, n):
    """
    Calculates the negative sum of radii and a penalty for constraint violations.
    
    Args:
        params: Flattened array of [x1, y1, r1, x2, y2, r2, ...]
        n: Number of circles
        
    Returns:
        objective_val: Negative sum of radii + penalty
    """
    # Reshape params
    # Shape (n, 3) -> [x, y, r]
    centers_radii = params.reshape((n, 3))
    centers = centers_radii[:, :2]
    radii = centers_radii[:, 2]
    
    # Objective: Maximize sum of radii -> Minimize -sum(radii)
    obj = -np.sum(radii)
    
    penalty = 0.0
    
    # Boundary Constraints
    # x >= r  => r - x <= 0
    # x <= 1 - r => r + x - 1 <= 0
    # y >= r
    # y <= 1 - r
    
    # Vectorized boundary checks
    # r - x
    pen_x_min = np.maximum(0, radii - centers[:, 0])**2
    # r + x - 1
    pen_x_max = np.maximum(0, radii + centers[:, 0] - 1.0)**2
    # r - y
    pen_y_min = np.maximum(0, radii - centers[:, 1])**2
    # r + y - 1
    pen_y_max = np.maximum(0, radii + centers[:, 1] - 1.0)**2
    
    boundary_penalty = np.sum(pen_x_min + pen_x_max + pen_y_min + pen_y_max)
    
    # Overlap Constraints
    # dist(i, j) >= r_i + r_j
    # (dist(i, j) - (r_i + r_j)) >= 0
    
    # Compute pairwise distances
    # centers shape (n, 2)
    # diff shape (n, n, 2)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    # Radii sum matrix shape (n, n)
    r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    
    # Overlap amount: positive if overlapping (dist < r_sum)
    # We want dist >= r_sum, so r_sum - dist <= 0
    # Violation is max(0, r_sum - dist)
    
    # We only care about upper triangle (i < j) to avoid double counting and self-check
    # But for penalty, summing all pairs and dividing by 2 is fine, or just upper triangle
    
    # Mask for upper triangle
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    
    violation = np.maximum(0, r_sum - dists)
    overlap_penalty = np.sum(violation[mask]**2)
    
    total_penalty = boundary_penalty + overlap_penalty
    
    return obj + 1000.0 * total_penalty, radii
