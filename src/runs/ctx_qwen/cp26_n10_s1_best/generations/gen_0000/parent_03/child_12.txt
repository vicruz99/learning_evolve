def make_hexagonal_grid(n, row_counts, r):
    centers = np.zeros((n, 2))
    idx = 0
    
    # Center the arrangement
    max_cols = max(row_counts)
    total_width = (max_cols - 1) * 2 * r + 2 * r
    x_offset = (1 - total_width) / 2
    
    total_height_est = (len(row_counts) - 1) * r * np.sqrt(3) + 2 * r
    y_offset = max(0, (1 - total_height_est) / 2)
    
    y = r + y_offset
    for row_idx, count in enumerate(row_counts):
        offset = r * (row_idx % 2)
        x_start = x_offset + offset
        for j in range(count):
            x = x_start + j * 2 * r
            if idx < n:
                centers[idx] = [x, y]
                idx += 1
        y += r * np.sqrt(3)
    
    return centers


def make_rectangular_grid(n, rows, cols, remainder, r):
    centers = np.zeros((n, 2))
    radii = np.full(n, r)
    idx = 0
    
    x_spacing = 1 / (cols + 1)
    y_spacing = 1 / (rows + 1)
    
    for i in range(rows):
        n_cols = cols + (1 if i < remainder else 0)
        for j in range(n_cols):
            if idx < n:
                centers[idx] = [x_spacing * (j + 1), y_spacing * (i + 1)]
                idx += 1
    
    return centers, radii

def optimize_positions(centers, radii, n, max_steps=1500):
    lr = 0.008
    for step in range(max_steps):
        forces = np.zeros((n, 2))
        
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                d2 = dx * dx + dy * dy
                d = np.sqrt(d2)
                req_dist = radii[i] + radii[j]
                if d < req_dist and d > 1e-12:
                    overlap = req_dist - d
                    fx = (dx / d) * overlap
                    fy = (dy / d) * overlap
                    forces[i, 0] += fx
                    forces[i, 1] += fy
                    forces[j, 0] -= fx
                    forces[j, 1] -= fy
        
        # Add repulsion from boundaries
        for i in range(n):
            if centers[i, 0] < radii[i] + 0.001:
                forces[i, 0] += 0.01 * (radii[i] + 0.001 - centers[i, 0])
            if centers[i, 0] > 1 - radii[i] - 0.001:
                forces[i, 0] -= 0.01 * (centers[i, 0] - (1 - radii[i] - 0.001))
            if centers[i, 1] < radii[i] + 0.001:
                forces[i, 1] += 0.01 * (radii[i] + 0.001 - centers[i, 1])
            if centers[i, 1] > 1 - radii[i] - 0.001:
                forces[i, 1] -= 0.01 * (centers[i, 1] - (1 - radii[i] - 0.001))
        
        centers += lr * forces
        
        # Clip to valid region
        for i in range(n):
            centers[i, 0] = np.clip(centers[i, 0], radii[i], 1 - radii[i])
            centers[i, 1] = np.clip(centers[i, 1], radii[i], 1 - radii[i])
        
        lr *= 0.9995
        
        # Early termination
        if np.max(np.abs(forces)) < 1e-10:
            break
    
    return centers


def compute_max_radii(centers, n, max_iter=500):
    # Initialize from boundary constraints
    radii = np.zeros(n)
    for i in range(n):
        radii[i] = min(centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])
    
    # Iteratively resolve mutual constraints
    for iteration in range(max_iter):
        changed = False
        for i in range(n):
            max_r = min(centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])
            for j in range(n):
                if i != j:
                    dx = centers[i, 0] - centers[j, 0]
                    dy = centers[i, 1] - centers[j, 1]
                    dist = np.sqrt(dx * dx + dy * dy)
                    max_r = min(max_r, dist - radii[j])
            max_r = max(0.0, max_r)
            if abs(max_r - radii[i]) > 1e-12:
                changed = True
            radii[i] = max_r
        if not changed:
            break
    
    return radii


def check_validity(centers, radii, n):
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if r < -1e-12:
            return False
        if x - r < -1e-9 or x + r > 1 + 1e-9:
            return False
        if y - r < -1e-9 or y + r > 1 + 1e-9:
            return False
    
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            if dist < radii[i] + radii[j] - 1e-9:
                return False
    
    return True


def repair_packing(centers, radii, n):
    # Reduce radii to ensure validity
    radii = compute_max_radii(centers, n)
    
    # Re-optimize positions once more
    centers = optimize_positions(centers, radii, n, max_steps=500)
    radii = compute_max_radii(centers, n)
    
    return centers, radii
