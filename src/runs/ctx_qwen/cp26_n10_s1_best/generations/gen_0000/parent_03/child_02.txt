import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    """
    n_circles = 26
    
    # Objective: Maximize sum of radii -> Minimize negative sum
    def objective(vars_flat):
        radii = vars_flat[2::3]
        return -np.sum(radii)

    # Constraints helper
    def create_constraints(n):
        constraints = []
        
        # 1. Boundary and Non-negative Radius Constraints
        # 0 <= x - r, x + r <= 1  => r <= x <= 1 - r
        # Same for y
        for i in range(n):
            idx = i * 3
            # r <= x
            constraints.append({'type': 'ineq', 'fun': lambda v, i=i: v[idx] - v[idx+2]})
            # x <= 1 - r  => x + r <= 1
            constraints.append({'type': 'ineq', 'fun': lambda v, i=i: 1.0 - v[idx] - v[idx+2]})
            
            # r <= y
            constraints.append({'type': 'ineq', 'fun': lambda v, i=i: v[idx+1] - v[idx+2]})
            # y <= 1 - r
            constraints.append({'type': 'ineq', 'fun': lambda v, i=i: 1.0 - v[idx+1] - v[idx+2]})
            
            # r >= 0
            constraints.append({'type': 'ineq', 'fun': lambda v, i=i: v[idx+2]})

        # 2. Non-overlap Constraints
        # dist(i,j) >= r_i + r_j  => dist(i,j) - (r_i + r_j) >= 0
        for i in range(n):
            for j in range(i + 1, n):
                idx_i = i * 3
                idx_j = j * 3
                
                def overlap_con(v, i=i, j=j):
                    xi, yi, ri = v[idx_i], v[idx_i+1], v[idx_i+2]
                    xj, yj, rj = v[idx_j], v[idx_j+1], v[idx_j+2]
                    dist = np.sqrt((xi - xj)**2 + (yi - yj)**2)
                    return dist - (ri + rj)
                
                constraints.append({'type': 'ineq', 'fun': overlap_con})

        return constraints

    constraints = create_constraints(n_circles)

    # Bounds: x, y in [0, 1], r in [0, 0.5]
    bounds = [(0, 1), (0, 1), (0, 0.5)] * n_circles

    # Strategy: Multi-start optimization from a 5x5 grid + 1
    best_result = None
    best_sum_radii = -1.0

    # Define a base 5x5 grid configuration
    def generate_initial_guess(perturbation_scale=0.0):
        x = np.linspace(0.1, 0.9, 5)
        y = np.linspace(0.1, 0.9, 5)
        xx, yy = np.meshgrid(x, y)
        centers = np.column_stack([xx.ravel(), yy.ravel()]) # 25 points
        
        # Add 26th point in a central gap (0.2, 0.2)
        centers = np.vstack([centers, [0.2, 0.2]])
        
        radii = np.full(26, 0.1)
        # Adjust radii slightly if perturbed
        
        vars_flat = np.empty(26 * 3)
        for i in range(26):
            vars_flat[3*i] = centers[i, 0]
            vars_flat[3*i+1] = centers[i, 1]
            vars_flat[3*i+2] = radii[i]
            
        if perturbation_scale > 0:
            noise = np.random.normal(0, perturbation_scale, 26*3)
            vars_flat += noise
            
        return vars_flat

    # Run optimizer multiple times
    for trial in range(5):
        pert_scale = 0.005 if trial > 0 else 0.0
        x0 = generate_initial_guess(pert_scale)
        
        # Clip initial radii to be feasible to help solver start
        # (x0 radii might be too large if noise makes center move too close to edge)
        # We enforce bounds in the solver, but good x0 helps
        
        res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints,
                       options={'maxiter': 1000, 'ftol': 1e-9})
        
        if res.success or res.fun < best_sum_radii * -1: # Check if we improved (since fun is negative sum)
            # We need to be careful, res.fun is the objective value (negative sum)
            # So a lower res.fun (more negative) is better.
            current_sum = -res.fun
            if current_sum > best_sum_radii:
                best_sum_radii = current_sum
                best_result = res
    
    if best_result is None:
        # Fallback to simple grid if optimization fails (unlikely)
        x0 = generate_initial_guess()
        best_result = {'x': x0}

    vars_opt = best_result.x
    centers = np.array([[vars_opt[3*i], vars_opt[3*i+1]] for i in range(n_circles)])
    radii = np.array([vars_opt[3*i+2] for i in range(n_circles)])
    
    return centers, radii, np.sum(radii)
