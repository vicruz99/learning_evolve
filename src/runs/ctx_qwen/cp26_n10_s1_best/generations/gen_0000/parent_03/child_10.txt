import numpy as np
from scipy.optimize import minimize


def create_initial_hex(n=26):
    """Create hexagonal initial configuration"""
    centers = []
    count = 0
    row = 0
    
    while count < n:
        cols = 6 if row % 2 == 0 else 5
        remaining = n - count
        cols = min(cols, remaining)
        
        y = 0.065 + row * 0.175
        for c in range(cols):
            if cols > 1:
                x = 0.05 + c * 0.9 / (cols - 1)
            else:
                x = 0.5
            centers.append([x, y])
            count += 1
        row += 1
    
    return np.array(centers), np.ones(n) * 0.085


def create_initial_grid(n=26):
    """Create square grid initial configuration"""
    nc = int(np.ceil(np.sqrt(n)))
    centers = []
    for i in range(nc):
        for j in range(nc):
            if len(centers) < n:
                x = 0.05 + j * 0.9 / max(nc - 1, 1)
                y = 0.05 + i * 0.9 / max(nc - 1, 1)
                centers.append([x, y])
    return np.array(centers), np.ones(n) * 0.08


def create_initial_staggered(n=26):
    """Create staggered pattern with varying row sizes"""
    centers = []
    rows_pattern = [6, 5, 6, 5, 4]
    row_y = [0.07, 0.21, 0.35, 0.49, 0.63]
    
    for idx, (num_in_row, y) in enumerate(zip(rows_pattern, row_y)):
        for c in range(num_in_row):
            if len(centers) < n:
                if num_in_row > 1:
                    x = 0.05 + c * 0.9 / (num_in_row - 1)
                else:
                    x = 0.5
                centers.append([x, y])
    
    return np.array(centers), np.ones(n) * 0.085


def optimize_from_start(n, centers_init, radii_init, seed=None):
    """Run optimization from a given starting point"""
    if seed is not None:
        np.random.seed(seed)
    
    x0 = np.concatenate([centers_init.flatten(), radii_init])
    
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
    
    def objective(x):
        return -np.sum(x[2 * n:])
    
    constraints = []
    
    for i in range(n):
        for j in range(i + 1, n):
            def no_overlap(x, ii=i, jj=j):
                xi, yi = x[2 * ii], x[2 * ii + 1]
                xj, yj = x[2 * jj], x[2 * jj + 1]
                ri, rj = x[2 * n + ii], x[2 * n + jj]
                return (xi - xj) ** 2 + (yi - yj) ** 2 - (ri + rj) ** 2
            constraints.append({'type': 'ineq', 'fun': no_overlap})
    
    for i in range(n):
        def left_bc(x, ii=i):
            return x[2 * ii] - x[2 * n + ii]
        constraints.append({'type': 'ineq', 'fun': left_bc})
        
        def right_bc(x, ii=i):
            return 1.0 - x[2 * ii] - x[2 * n + ii]
        constraints.append({'type': 'ineq', 'fun': right_bc})
        
        def bottom_bc(x, ii=i):
            return x[2 * ii + 1] - x[2 * n + ii]
        constraints.append({'type': 'ineq', 'fun': bottom_bc})
        
        def top_bc(x, ii=i):
            return 1.0 - x[2 * ii + 1] - x[2 * n + ii]
        constraints.append({'type': 'ineq', 'fun': top_bc})
    
    try:
        result = minimize(
            objective, x0, method='SLSQP',
            bounds=bounds, constraints=constraints,
            options={
                'maxiter': 3000,
                'ftol': 1e-14,
                'disp': False
            }
        )
        return result.x, -result.fun, result.success
    except Exception:
        return x0, np.sum(radii_init), False


def perturb_config(centers, radii, scale=0.02):
    """Create a perturbed version of the configuration"""
    centers_p = centers + np.random.randn(*centers.shape) * scale
    centers_p = np.clip(centers_p, 0.01, 0.99)
    radii_p = np.clip(radii * (1 + np.random.randn(len(radii)) * 0.1), 0.01, 0.4)
    return centers_p, radii_p


def run_packing() -> tuple:
    n = 26
    best_x = None
    best_obj = float('-inf')
    
    # Generate multiple starting configurations
    configs = []
    
    configs.append(create_initial_hex(n))
    configs.append(create_initial_grid(n))
    configs.append(create_initial_staggered(n))
    
    # Add perturbed versions
    for _ in range(5):
        c, r = create_initial_hex(n)
        configs.append(perturb_config(c, r, scale=0.03))
    
    # Add some random configurations
    for _ in range(3):
        centers = np.random.rand(n, 2) * 0.8 + 0.1
        radii = np.random.rand(n) * 0.05 + 0.05
        configs.append((centers, radii))
    
    # Optimize from each starting point
    for idx, (centers, radii) in enumerate(configs):
        result, obj, success = optimize_from_start(n, centers, radii)
        if obj > best_obj:
            best_obj = obj
            best_x = result
    
    # If we have a good result, try a few more perturbations around it
    if best_x is not None and best_obj > 2.0:
        centers_opt = best_x[:2 * n].reshape(n, 2)
        radii_opt = best_x[2 * n:]
        
        for _ in range(8):
            centers_p, radii_p = perturb_config(centers_opt, radii_opt, scale=0.015)
            result, obj, success = optimize_from_start(n, centers_p, radii_p)
            if obj > best_obj:
                best_obj = obj
                best_x = result
                centers_opt = best_x[:2 * n].reshape(n, 2)
                radii_opt = best_x[2 * n:]
    
    # Extract final results
    if best_x is not None:
        centers = best_x[:2 * n].reshape(n, 2)
        radii = best_x[2 * n:]
        
        # Clean up any tiny negative radii from numerical issues
        radii = np.maximum(radii, 0.0)
        
        sum_r = np.sum(radii)
    else:
        centers, radii = create_initial_hex(n)
        sum_r = np.sum(radii)
    
    # Handle NaN cases
    if np.isnan(centers).any() or np.isnan(radii).any():
        centers, radii = create_initial_hex(n)
        sum_r = np.sum(radii)
    
    # Ensure centers are within bounds (numerical cleanup)
    centers = np.clip(centers, 0.0, 1.0)
    
    return centers, radii, sum_r
