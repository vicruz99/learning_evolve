import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def objective(vars):
    """Objective function: minimize negative sum of radii."""
    s = 0.0
    for i in range(N_CIRCLES):
        s += vars[3*i + 2]
    return -s

def create_constraints():
    """Create list of constraint dictionaries for SLSQP."""
    cons = []
    # Boundary constraints for each circle
    for i in range(N_CIRCLES):
        def c_x_min(v, idx=i): return v[3*idx] - v[3*idx+2]
        def c_x_max(v, idx=i): return 1.0 - v[3*idx] - v[3*idx+2]
        def c_y_min(v, idx=i): return v[3*idx+1] - v[3*idx+2]
        def c_y_max(v, idx=i): return 1.0 - v[3*idx+1] - v[3*idx+2]
        def c_r_pos(v, idx=i): return v[3*idx+2] - 1e-4

        cons.append({'type': 'ineq', 'fun': c_x_min})
        cons.append({'type': 'ineq', 'fun': c_x_max})
        cons.append({'type': 'ineq', 'fun': c_y_min})
        cons.append({'type': 'ineq', 'fun': c_y_max})
        cons.append({'type': 'ineq', 'fun': c_r_pos})

    # Pairwise non-overlap constraints
    for i in range(N_CIRCLES):
        for j in range(i + 1, N_CIRCLES):
            def c_pair(v, i_idx=i, j_idx=j):
                xi, yi, ri = v[3*i_idx], v[3*i_idx+1], v[3*i_idx+2]
                xj, yj, rj = v[3*j_idx], v[3*j_idx+1], v[3*j_idx+2]
                return (xi - xj)**2 + (yi - yj)**2 - (ri + rj)**2
            cons.append({'type': 'ineq', 'fun': c_pair})
    return cons

def run_packing():
    np.random.seed(42)
    n = N_CIRCLES
    
    # Initial configuration: Hexagonal-ish grid layout
    centers = np.zeros((n, 2))
    idx = 0
    # Distribute 26 circles in 5 rows with slight staggering
    rows = [5, 6, 5, 6, 4] 
    y_coords = np.linspace(0.15, 0.85, 5)
    
    for r_idx, count in enumerate(rows):
        y = y_coords[r_idx]
        shift = 0.06 if r_idx % 2 == 1 else 0.0
        x_coords = np.linspace(0.05 + shift, 0.95 + shift, count)
        for x in x_coords:
            if idx < n:
                centers[idx] = [x, y]
                idx += 1
                
    radii_init = np.full(n, 0.085)
    
    # Flatten to optimization variables: [x1, y1, r1, x2, y2, r2, ...]
    x0 = np.zeros(3*n)
    for i in range(n):
        x0[3*i] = centers[i, 0]
        x0[3*i+1] = centers[i, 1]
        x0[3*i+2] = radii_init[i]
        
    bounds = [(0.0, 1.0)] * (2*n) + [(0.001, 0.5)] * n
    constraints = create_constraints()
    
    best_sum = -1.0
    best_x = x0.copy()
    
    # Run optimization with multiple restarts to avoid local minima
    for run in range(4):
        x_start = x0.copy()
        # Perturb centers slightly
        noise = np.random.normal(0, 0.015, 2*n)
        x_start[:2*n] += noise
        x_start[:2*n] = np.clip(x_start[:2*n], 0.01, 0.99)
        
        try:
            res = minimize(objective, x_start, method='SLSQP', bounds=bounds, 
                          constraints=constraints, options={'maxiter': 1200, 'ftol': 1e-9, 'disp': False})
            curr_sum = -res.fun
            if curr_sum > best_sum:
                best_sum = curr_sum
                best_x = res.x.copy()
        except Exception:
            pass
            
    final_centers = best_x[:2*n].reshape(n, 2)
    final_radii = best_x[2*n:]
    
    return final_centers, final_radii, np.sum(final_radii)
