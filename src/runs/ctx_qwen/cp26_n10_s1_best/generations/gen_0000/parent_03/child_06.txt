import numpy as np
from scipy.optimize import minimize

def run_packing():
    n = 26
    
    # 1. Generate Initial Configuration
    # We create a grid of potential centers. 
    # A 6x5 grid gives 30 points, we take the first 26.
    # We space them out to ensure initial validity with small radii.
    
    # X coordinates: 6 points spread in [0.1, 0.9]
    # Y coordinates: 5 points spread in [0.1, 0.9]
    # Actually, let's just make a dense grid in [0,1] and pick points.
    # To be safe with boundaries, let's keep centers away from edges initially.
    
    # Let's use a 6x5 grid spacing.
    # Width 1.0. 6 columns -> 5 gaps. Gap = 1.0 / 6 approx? 
    # Let's place centers at 1/12, 3/12, ... to leave margin.
    
    x_vals = np.linspace(0.1, 0.9, 6)
    y_vals = np.linspace(0.1, 0.9, 5)
    
    centers_init = []
    count = 0
    for y in y_vals:
        for x in x_vals:
            if count < n:
                centers_init.append([x, y])
                count += 1
            else:
                break
        if count >= n:
            break
            
    centers_init = np.array(centers_init)
    radii_init = np.full(n, 0.05) # Small initial radius, safe
    
    # 2. Define Optimization Variables and Bounds
    # Vector: [x0, y0, r0, x1, y1, r1, ..., x25, y25, r25]
    # Or structured: [x_all, y_all, r_all] for easier vectorization
    # Let's use [x_all, y_all, r_all]
    
    x0 = np.concatenate([centers_init[:, 0], centers_init[:, 1], radii_init])
    
    # Bounds: x in [0,1], y in [0,1], r in [0, 0.5]
    bounds = []
    for _ in range(n):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 0.5)) # r
    
    # 3. Objective Function
    def objective(vars):
        # vars: [x0..x25, y0..y25, r0..r25]
        r = vars[2*n:]
        return -np.sum(r) # Maximize sum of radii
    
    # 4. Constraints
    # We return a dictionary of constraints for SLSQP
    # Or a list of dicts.
    # Constraints are of type 'ineq' meaning func(vars) >= 0
    
    def constraints(vars):
        # Unpack
        x = vars[:n]
        y = vars[n:2*n]
        r = vars[2*n:]
        
        cons = []
        
        # Boundary constraints
        # x - r >= 0
        cons.extend(x - r)
        # (1 - x) - r >= 0
        cons.extend(1.0 - x - r)
        # y - r >= 0
        cons.extend(y - r)
        # (1 - y) - r >= 0
        cons.extend(1.0 - y - r)
        
        # Non-overlap constraints
        # For all i < j: dist^2 - (ri + rj)^2 >= 0
        # Vectorized computation
        # x is (n,), y is (n,), r is (n,)
        
        # Compute pairwise squared distances
        # X_diff = x[:, None] - x[None, :]  -> shape (n, n)
        # But we only need upper triangle
        
        # Efficiently compute all pairwise distances squared
        # Using broadcasting
        X = x[:, np.newaxis] - x[np.newaxis, :] # (n, n)
        Y = y[:, np.newaxis] - y[np.newaxis, :] # (n, n)
        D2 = X**2 + Y**2
        
        R_sum = r[:, np.newaxis] + r[np.newaxis, :] # (n, n)
        R_sum2 = R_sum**2
        
        # Constraint matrix: D2 - R_sum2 >= 0
        # We only need upper triangle (i < j) to avoid duplicates and self
        # Mask upper triangle
        mask = np.triu(np.ones((n, n), dtype=bool), k=1)
        constraints_vals = (D2 - R_sum2)[mask]
        
        cons.extend(constraints_vals)
        
        return np.array(cons)

    cons_dict = {'type': 'ineq', 'fun': constraints}
    
    # 5. Run Optimization
    # SLSQP is suitable for non-linear constraints
    try:
        res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons_dict, 
                       options={'maxiter': 2000, 'ftol': 1e-9, 'disp': False})
        
        if res.success:
            final_x = res.x[:n]
            final_y = res.x[n:2*n]
            final_r = res.x[2*n:]
        else:
            # Fallback to initial if failed (though unlikely with valid start)
            # Or try to use the result anyway if it's feasible?
            # SLSQP might fail to converge but still improve.
            final_x = res.x[:n]
            final_y = res.x[n:2*n]
            final_r = res.x[2*n:]
            
    except Exception as e:
        # In case of error, return initial guess (which is valid)
        final_x = centers_init[:, 0]
        final_y = centers_init[:, 1]
        final_r = radii_init

    centers = np.column_stack((final_x, final_y))
    radii = final_r
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
