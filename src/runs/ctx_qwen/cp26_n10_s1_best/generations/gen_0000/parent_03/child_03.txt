```python
import numpy as np
import scipy.optimize as opt

def compute_penalty_and_obj(vars_flat, mu):
    """
    Computes the objective function (negative sum of radii) and the penalty term
    for constraint violations.
    
    Args:
        vars_flat: numpy array of shape (3*N) containing [x1, y1, r1, x2, y2, r2, ...]
        mu: penalty multiplier (unused in calculation but kept for signature compatibility)
    
    Returns:
        obj: float, the objective value
        penalty: float, the accumulated penalty value
    """
    N = len(vars_flat) // 3
    x = vars_flat[0::3]
    y = vars_flat[1::3]
    r = vars_flat[2::3]
    
    obj = -np.sum(r)
    penalty = 0.0
    
    # Boundary penalties
    # Constraint: r <= x  => violation if r > x
    penalty += np.sum(np.maximum(0, r - x)**2)
    # Constraint: x <= 1 - r  => violation if x + r > 1
    penalty += np.sum(np.maximum(0, x + r - 1.0)**2)
    # Constraint: r <= y  => violation if r > y
    penalty += np.sum(np.maximum(0, r - y)**2)
    # Constraint: y <= 1 - r  => violation if y + r > 1
    penalty += np.sum(np.maximum(0, y + r - 1.0)**2)
    
    # Overlap penalties
    # Constraint: dist(i, j) >= r[i] + r[j]
    # We use vectorized operations for efficiency
    
    # Compute pairwise distances matrix
    # x[:, None] has shape (N, 1), x[None, :] has shape (1, N)
    dx = x[:, None] - x[None, :] 
    dy = y[:, None] - y[None, :] 
    dist = np.sqrt(dx**2 + dy**2)
    
    # Compute pairwise sum of radii matrix
    rad_sum = r[:, None] + r[None, :]
    
    # Calculate violations: positive value if circles overlap
    violations = np.maximum(0, rad_sum - dist)
    
    # We only need to count each pair once (i < j). 
    # The diagonal (i=i) is always a violation (2r > 0), so we ignore it.
    # Using triu with k=1 extracts the upper triangle above the main diagonal.
    upper_violations = np.triu(violations, k=1)
    penalty += np.sum(upper_violations**2)
    
    return obj, penalty

def objective_with_penalty(vars_flat, mu):
    """
    Wrapper function that computes the penalized objective for the optimizer.
    """
    obj, penalty = compute_penalty_and_obj(vars_flat, mu)
    return obj + mu * penalty

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Pack 26 circles in a unit square to maximize the sum of radii.
    """
    N = 26
    
    # --- Initialization: Hexagonal Lattice ---
    # Generating points on a hexagonal lattice provides a dense initial configuration.
    points = []
    range_val = 10
    for i in range(-range_val, range_val + 1):
        for j in range(-range_val, range_val + 1):
            # Hexagonal lattice basis vectors
            x = i + 0.5 * j
            y = (np.sqrt(3) / 2) * j
            points.append([x, y])
    
    points = np.array(points)
    
    # Select the 26 points closest to the origin to form a compact cluster
    dists_to_origin = np.linalg.norm(points, axis=1)
    indices = np.argsort(dists_to_origin)[:N]
    initial_points = points[indices]
    
    # Normalize and scale the cluster to fit within the unit square [0, 1]x[0, 1]
    # We leave some margin (scale to 0.8 size) to allow for radii expansion
    min_x, min_y = np.min(initial_points, axis=0)
    max_x, max_y = np.max(initial_points, axis=0)
    width = max_x - min_x
    height = max_y - min_y
    scale_factor = 0.8 / max(width, height)
    
    centered_points = initial_points - np.mean(initial_points, axis=0)
    scaled_points = centered_points * scale_factor + 0.5
    
    # Initialize radii to a small value
    initial_radii = np.full(N, 0.05)
    
    # Flatten variables for optimizer: [x1, y1, r1, x2, y2, r2, ...]
    x0 = np.zeros(3 * N)
    for i in range(N):
        x0[3*i] = scaled_points[i, 0]
        x0[3*i+1] = scaled_points[i, 1]
        x0[3*i+2] = initial_radii[i]
        
    # --- Optimization ---
    # We use a penalty method with L-BFGS-B.
    # The penalty term penalizes constraint violations (overlaps and boundary crossings).
    # We iteratively increase the penalty multiplier 'mu' to enforce constraints strictly.
    
    current_vars = x0
    mu = 100.0
    
    # Bounds for variables: x, y in [0, 1], r in [0, 0.5]
    bounds_lbgfs = [(0, 1), (0, 1), (0, 0.5)] * N
    
    # Iterative optimization with increasing penalty
    for _ in range(15):
        res = opt.minimize(objective_with_penalty, current_vars, method='L-BFGS-B', 
                           bounds=bounds_lbgfs, args=(mu,),
                           options={'maxiter': 1000, 'ftol': 1e-15, 'gtol': 1e-12})
        current_vars = res.x
        mu *= 5
        
        # Check if penalty is sufficiently small to stop early
        _, pen = compute_penalty_and_obj(current_vars, mu)
        if pen < 1e-10:
            break
            
    # Extract centers and radii from the optimized variables
    centers = np.array([[current_vars[3*i], current_vars[3*i+1]] for i in range(N)])
    radii = np.array([current_vars[3*i+2] for i in range(N)])
    
    # Ensure radii are non-negative (bounds should handle this, but just in case)
    radii = np.maximum(radii, 0.0)
    
    sum_radii = float(np.sum(radii))
    
    return centers, radii, sum_radii
```
