import numpy as np
from scipy.optimize import minimize, linprog
import warnings
warnings.filterwarnings('ignore')

N = 26
TRIL_IDX = np.tril_indices(N, -1)
TRIU_IDX = np.triu_indices(N, 1)

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    num_ineq = n + n * (n - 1) // 2
    A = np.zeros((num_ineq, n))
    b = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A[idx, i] = 1.0
        b[idx] = max(0.0, lim)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = max(0.0, d)
            idx += 1
            
    bounds = [(0.0, 0.5)] * n
    try:
        res = linprog(-np.ones(n), A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-8)
    except Exception:
        pass
    return np.full(n, 0.02)

def force_optimize_centers(centers, radii, steps=200):
    """Force-directed relaxation to push centers apart according to radii."""
    c = centers.copy()
    r = radii.copy()
    n = len(c)
    
    for step in range(steps):
        f = np.zeros_like(c)
        lr = 0.008 * (1.0 - step / steps)**0.5
        
        for i in range(n):
            for j in range(i + 1, n):
                dx = c[j] - c[i]
                d = np.hypot(dx[0], dx[1])
                # Repel if close to touching
                if d < r[i] + r[j] + 0.035:
                    rep = 8.0 / (d**2 + 1e-3)
                    f[i] -= rep * dx / d
                    f[j] += rep * dx / d
                    
            # Boundary repulsion
            for dim in range(2):
                if c[i, dim] < r[i] + 1e-3:
                    f[i, dim] += 40.0 * (r[i] - c[i, dim] + 1e-3)
                elif c[i, dim] > 1.0 - r[i] - 1e-3:
                    f[i, dim] -= 40.0 * (c[i, dim] - (1.0 - r[i]) - 1e-3)
                    
        c += f * lr
        c = np.clip(c, 1e-4, 1.0 - 1e-4)
        
    return c

def make_hex_init(r0, angle, scale=1.0):
    """Generates a rotated/scaled hexagonal lattice initialization."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0 and len(pts) < N + 10:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        
    pts = np.array(pts[:N + 10])
    
    # Scale around center
    pts = (pts - 0.5) * scale + 0.5
    
    if angle != 0.0:
        ca, sa = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[ca, -sa], [sa, ca]]) + 0.5
        
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.15, 0.85, (1, 2))])
        
    # Break exact symmetries
    pts += np.random.uniform(-0.001, 0.001, pts.shape)
    return pts[:N]

def constraints_slsqp(x):
    """Inequality constraints for SLSQP: g(x) >= 0."""
    xs, ys, rs = x[0::3], x[1::3], x[2::3]
    c = np.concatenate([xs - rs, 1.0 - xs - rs, ys - rs, 1.0 - ys - rs])
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    c = np.concatenate([c, np.hypot(dx[TRIL_IDX], dy[TRIL_IDX]) - dr[TRIL_IDX]])
    return c

def objective_func(x):
    """Objective: maximize sum of radii."""
    return -np.sum(x[2::3])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    np.random.seed(42)

    best_sum = -1.0
    best_c = None
    best_r = None

    inits = []

            inits.append(make_hex_init(0.092, ang, sc))
    for s in range(15):
        np.random.seed(s + 50)
        inits.append(np.random.uniform(0.1, 0.9, (N, 2)))
        
    for c0 in inits:
        c = c0.copy()
        r = solve_lp_radii(c)
        
        # Alternating optimization
        for _ in range(8):
            r = solve_lp_radii(c)
            c = force_optimize_centers(c, r, steps=150)
            
        r = solve_lp_radii(c)
        s = np.sum(r)
        if s > best_sum:
            best_sum = s
            best_c = c.copy()
            best_r = r.copy()
            
    # Phase 2: Perturbation & Alternating from best configuration
    for cyc in range(70):
        noise = 0.004 * (0.91 ** cyc)

        if s_p > best_sum:
            best_sum = s_p
            best_c = c_p.copy()
            best_r = r_p.copy()

        ang = np.deg2rad(ang_deg)

        pass
        
    c_final = np.column_stack((x0[0::3], x0[1::3]))
    r_final = solve_lp_radii(c_final)
    
    # Final strict validity adjustment against 1e-12 tolerance

                        break

        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)
        
    return centers, radii, float(np.sum(radii))
