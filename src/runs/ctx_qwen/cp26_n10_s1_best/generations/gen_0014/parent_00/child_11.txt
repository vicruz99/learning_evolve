import numpy as np
from scipy.optimize import minimize, linprog
import warnings

warnings.filterwarnings('ignore')

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    num_ineq = n + n * (n - 1) // 2
    A = np.zeros((num_ineq, n))
    b = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A[idx, i] = 1.0
        b[idx] = max(0.0, lim)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = max(0.0, d)
            idx += 1
            
    bounds = [(0.0, 0.5)] * n
    try:
        res = linprog(-np.ones(n), A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-8)
    except Exception:
        pass
    return np.full(n, 0.02)

def get_score(centers):
    """Returns sum of radii and the radii array for a given center configuration."""
    radii = solve_lp_radii(centers)
    return np.sum(radii), radii

def make_hex_init(r0, angle, scale=1.0):
    """Generates a rotated hexagonal lattice initialization."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0 and len(pts) < N + 10:
            pts.append([x * scale, y * scale])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
    pts = np.array(pts[:N + 10])
    
    if angle != 0.0:
        ca, sa = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[ca, -sa], [sa, ca]]) + 0.5
        
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.15, 0.85, (1, 2))])
    return pts[:N]

def perturb_centers(c, step_size, move_type):
    """Applies a perturbation to centers based on move type."""
    c = c.copy()
    if move_type == 'shift':
        idx = np.random.choice(N, np.random.randint(1, 4), replace=False)
        c[idx] += np.random.normal(0, step_size, (len(idx), 2))
    elif move_type == 'rotate_pair':
        i, j = np.random.choice(N, 2, replace=False)
        mid = (c[i] + c[j]) / 2.0
        ang = np.random.uniform(-step_size, step_size)
        ca, sa = np.cos(ang), np.sin(ang)
        diff = np.array([c[i] - mid, c[j] - mid])
        rot_diff = diff @ np.array([[ca, -sa], [sa, ca]])
        c[i], c[j] = mid + rot_diff[0], mid + rot_diff[1]
    elif move_type == 'rotate_subset':
        k = np.random.randint(3, 7)
        idx = np.random.choice(N, k, replace=False)
        pivot = np.mean(c[idx], axis=0)
        ang = np.random.uniform(-step_size, step_size)
        ca, sa = np.cos(ang), np.sin(ang)
        c[idx] = (c[idx] - pivot) @ np.array([[ca, -sa], [sa, ca]]) + pivot
        
    return np.clip(c, 0.001, 0.999)

def simulated_annealing(c0, max_iter=6000):
    """Simulated annealing on centers using LP radii evaluation."""
    c = c0.copy()
    score, _ = get_score(c)
    best_c = c.copy()
    best_score = score
    T = 0.03
    
    for i in range(max_iter):
        # Adaptive step size: starts larger, cools down
        step = 0.006 * (0.9992 ** i) + 0.0004
        move = np.random.choice(['shift', 'shift', 'shift', 'rotate_pair', 'rotate_subset'])
        c_new = perturb_centers(c, step, move)
        
        score_new, _ = get_score(c_new)
        
        delta = score_new - score
        if delta > 0 or (T > 1e-6 and np.random.rand() < np.exp(delta / T)):
            c = c_new
            score = score_new
            if score > best_score:
                best_score = score
                best_c = c.copy()
                
        T *= 0.9994
        
    return best_c, best_score

def slsqp_polish(centers):
    """Polish configuration using SLSQP joint optimization."""
    radii = solve_lp_radii(centers)
    x = np.zeros(3 * N)
    x[0::3] = centers[:, 0]
    x[1::3] = centers[:, 1]
    x[2::3] = radii * 0.9995
    
    def obj(xv):
        return -np.sum(xv[2::3])
        
    def cons(xv):
        xs, ys, rs = xv[0::3], xv[1::3], xv[2::3]
        c = np.concatenate([xs - rs, 1.0 - xs - rs, ys - rs, 1.0 - ys - rs])
        dx = xs[:, None] - xs[None, :]
        dy = ys[:, None] - ys[None, :]
        dr = rs[:, None] + rs[None, :]
        c = np.concatenate([c, np.hypot(dx[TRIL_IDX], dy[TRIL_IDX]) - dr[TRIL_IDX]])
        return c
        
    bnds = [(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)] * N
    try:
        res = minimize(obj, x, method='SLSQP', bounds=bnds, 
                       constraints={'type': 'ineq', 'fun': cons},
                       options={'maxiter': 12000, 'ftol': 1e-14, 'disp': False})
        if not np.isnan(res.fun):
            nc = np.column_stack((res.x[0::3], res.x[1::3]))
            nr = solve_lp_radii(nc)
            return nc, nr, np.sum(nr)
    except Exception:
        pass
    return centers, radii, np.sum(radii)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    np.random.seed(42)

    best_r = None

        starts.append(np.random.uniform(0.15, 0.85, (N, 2)))
        
    for pts in starts:

        c_pol, r_pol, s_pol = slsqp_polish(c_sa)
        if s_pol > best_sum:
            best_sum = s_pol
            best_c, best_r = c_pol, r_pol
            
    # Phase 2: Deflation & Recovery to escape local minima
    if best_c is not None:
        for cycle in range(25):
            c_def = best_c.copy()
            r_def = best_r.copy()
            
            # Aggressively shrink random subset to break rigid contact networks
            k = np.random.randint(4, 10)
            idx = np.random.choice(N, k, replace=False)
            r_def[idx] *= 0.65
            c_def[idx] += np.random.normal(0, 0.025, (k, 2))
            c_def = np.clip(c_def, 0.01, 0.99)
            
            c_new, s_new = simulated_annealing(c_def, max_iter=4000)
            c_pol, r_pol, s_pol = slsqp_polish(c_new)
            if s_pol > best_sum:
                best_sum = s_pol
                best_c, best_r = c_pol, r_pol
                
    # Phase 3: Fine-grained SA + Polish on current best
    if best_c is not None:
        c_fin, s_fin = simulated_annealing(best_c, max_iter=7000)
        c_pol, r_pol, s_pol = slsqp_polish(c_fin)
        if s_pol > best_sum:
            best_sum = s_pol
            best_c, best_r = c_pol, r_pol

    # Fallback (should not be reached)
    if best_c is None:
        best_c = np.tile(np.linspace(0.1, 0.9, 5), 6).reshape(5, 6)[:N].T
        best_r = np.full(N, 0.06)
        best_sum = np.sum(best_r)

    # Final strict validation repair against 1e-12 tolerance
    centers = best_c.copy()
    radii = best_r.copy()

        valid = True
        for i in range(N):

    return centers, radii, float(np.sum(radii))
