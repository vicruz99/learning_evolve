import numpy as np
from scipy.optimize import minimize, linprog
import warnings

warnings.filterwarnings('ignore')

N = 26

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    num_ineq = n + n * (n - 1) // 2
    A = np.zeros((num_ineq, n))
    b = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A[idx, i] = 1.0
        b[idx] = max(0.0, lim)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = max(0.0, d)
            idx += 1
            
    bounds = [(0.0, 0.5)] * n
    try:
        res = linprog(-np.ones(n), A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-9)
    except Exception:
        pass
    return np.full(n, 0.01)

def objective_slsqp(x):
    """Objective: maximize sum of radii."""
    return -np.sum(x[2::3])

def constraints_slsqp(x):
    """Inequality constraints for SLSQP: g(x) >= 0."""
    xs, ys, rs = x[0::3], x[1::3], x[2::3]
    c = np.concatenate([xs - rs, 1.0 - xs - rs, ys - rs, 1.0 - ys - rs])
    
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    # Use squared distance constraints for smoother gradients away from contact,
    # but switch to Euclidean for stability near contact.
    c = np.concatenate([c, np.hypot(dx[np.tril_indices(N, -1)], dy[np.tril_indices(N, -1)]) - dr[np.tril_indices(N, -1)]])
    return c

def run_sa(centers, steps=2500, T0=0.06):
    """Simulated Annealing on centers using LP radius evaluation."""
    c = centers.copy()
    r = solve_lp_radii(c)
    best_c, best_r = c.copy(), r.copy()
    best_sum = np.sum(r)
    current_sum = best_sum
    T = T0
    
    for _ in range(steps):
        c_new = c.copy()
        move = np.random.randint(4)
        
        if move == 0:  # Displace random circle
            idx = np.random.randint(N)
            c_new[idx] += np.random.normal(0, 0.004, 2)
        elif move == 1:  # Push to nearest boundary
            idx = np.random.randint(N)
            d = np.array([c_new[idx, 0], 1.0-c_new[idx, 0], c_new[idx, 1], 1.0-c_new[idx, 1]])
            dirs = [[-1, 0], [1, 0], [0, -1], [0, 1]]
            c_new[idx] += dirs[np.argmin(d)] * np.random.uniform(0.002, 0.008)
        elif move == 2:  # Rotate pair around midpoint
            idx = np.random.choice(N, 2, replace=False)
            ang = np.random.uniform(-0.15, 0.15)
            ca, sa = np.cos(ang), np.sin(ang)
            mid = (c_new[idx[0]] + c_new[idx[1]]) / 2.0
            p1, p2 = c_new[idx[0]] - mid, c_new[idx[1]] - mid
            c_new[idx[0]] = mid + np.array([p1[0]*ca - p1[1]*sa, p1[0]*sa + p1[1]*ca])
            c_new[idx[1]] = mid + np.array([p2[0]*ca - p2[1]*sa, p2[0]*sa + p2[1]*ca])
        else:  # Swap two circles
            idx = np.random.choice(N, 2, replace=False)
            c_new[idx[0]], c_new[idx[1]] = c_new[idx[1]], c_new[idx[0]]
            
        c_new = np.clip(c_new, 1e-5, 1.0 - 1e-5)
        r_new = solve_lp_radii(c_new)
        s_new = np.sum(r_new)
        
        if s_new > current_sum or (T > 1e-7 and np.random.rand() < np.exp((s_new - current_sum) / T)):
            c, r = c_new, r_new
            current_sum = s_new
            if s_new > best_sum:
                best_sum = s_new
                best_c, best_r = c_new.copy(), r_new.copy()
                
        T *= 0.994
    return best_c, best_r, best_sum

def make_hex_init(r0, angle):
    """Generates a rotated hexagonal lattice initialization."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0 and len(pts) < N + 10:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
    pts = np.array(pts[:N + 10])
    
    if angle != 0.0:
        ca, sa = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[ca, -sa], [sa, ca]]) + 0.5
        
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.15, 0.85, (1, 2))])
    return pts[:N] + np.random.uniform(-0.001, 0.001, (N, 2))

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    bounds_obj = [(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints_slsqp}
    
    best_sum = -1.0
    best_x = None
    
    # Phase 1: Generate diverse initial configurations
    inits = []

        pts = np.random.uniform(0.15, 0.85, (N, 2))
        # Quick force spread
        for _ in range(100):
            f = np.zeros_like(pts)
            for i in range(N):
                for j in range(i+1, N):
                    dx = pts[j] - pts[i]
                    d = np.hypot(dx[0], dx[1])
                    if d < 0.3 and d > 1e-5:
                        rep = 0.01 / (d**2 + 0.001)
                        f[i] -= dx * rep / d
                        f[j] += dx * rep / d
                for dim in range(2):
                    if pts[i, dim] < 0.12: f[i, dim] += 0.05
                    elif pts[i, dim] > 0.88: f[i, dim] -= 0.05
            pts += f * 0.03
            pts = np.clip(pts, 0.02, 0.98)
        inits.append(pts)

    # Phase 2: SA Optimization on each start
    for c_init in inits:

            best_sum = s_sa
            best_x = np.zeros(3 * N)
            best_x[0::3] = c_sa[:, 0]
            best_x[1::3] = c_sa[:, 1]
            best_x[2::3] = r_sa

    # Phase 3: SLSQP Polish
    if best_x is not None:
        try:
            res = minimize(objective_slsqp, best_x, method='SLSQP', bounds=bounds_obj, constraints=cons,
                           options={'maxiter': 15000, 'ftol': 1e-14, 'disp': False})
            if not np.isnan(res.fun) and -res.fun > best_sum:
                c_tmp = np.column_stack((res.x[0::3], res.x[1::3]))
                r_new = solve_lp_radii(c_tmp)
                s_new = np.sum(r_new)

                    best_x[1::3] = c_tmp[:, 1]
                    best_x[2::3] = r_new
        except Exception:
            pass

            # Shrink radii to break rigid contacts

            x_def[2::3] *= shrink
            # Perturb centers

            # Project to feasible bounds
            for i in range(N):

                x_def[3*i+1] = np.clip(x_def[3*i+1], r, 1.0-r)
                
            c_def = np.column_stack((x_def[0::3], x_def[1::3]))
            c_new, r_new, s_new = run_sa(c_def, steps=1500, T0=0.04)

                best_x = np.zeros(3 * N)

                    pass

        best_x[1::3] = np.repeat(np.linspace(0.1, 0.9, 6), 5)[:N]
        best_x[2::3] = 0.06
        best_sum = np.sum(best_x[2::3])

    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3].copy()

    # Final strict validity adjustment against 1e-12 tolerance
    for _ in range(100):

        for i in range(N):
            if radii[i] < 0 or centers[i, 0] < radii[i] - 1e-9 or centers[i, 0] > 1.0 - radii[i] + 1e-9 or \
               centers[i, 1] < radii[i] - 1e-9 or centers[i, 1] > 1.0 - radii[i] + 1e-9:

        if valid:

                    if np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1]) < radii[i] + radii[j] - 1e-9:

    return centers, radii, float(np.sum(radii))
