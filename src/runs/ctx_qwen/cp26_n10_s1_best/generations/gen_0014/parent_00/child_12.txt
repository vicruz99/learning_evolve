import numpy as np
from scipy.optimize import minimize, linprog
import warnings

warnings.filterwarnings('ignore')

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    num_ineq = n + n * (n - 1) // 2
    A = np.zeros((num_ineq, n))
    b = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A[idx, i] = 1.0
        b[idx] = max(0.0, lim)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = max(0.0, d)
            idx += 1
            
    bounds = [(0.0, 0.5)] * n
    try:
        res = linprog(-np.ones(n), A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-9)
    except Exception:
        pass
    return np.full(n, 1e-9)

def objective_func(x):
    """Objective to minimize: negative sum of radii."""
    return -np.sum(x[2::3])

def constraints_slsqp(x):
    """Inequality constraints for SLSQP: g(x) >= 0."""
    xs, ys, rs = x[0::3], x[1::3], x[2::3]
    c = np.concatenate([xs - rs, 1.0 - xs - rs, ys - rs, 1.0 - ys - rs])
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    # np.hypot provides stable gradients at contact points
    c = np.concatenate([c, np.hypot(dx[TRIL_IDX], dy[TRIL_IDX]) - dr[TRIL_IDX]])
    return c

def make_hex_init(r0, angle):
    """Generates a rotated hexagonal lattice initialization."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0 and len(pts) < N + 10:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
    pts = np.array(pts[:N + 10])
    if angle != 0.0:
        ca, sa = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[ca, -sa], [sa, ca]]) + 0.5
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.15, 0.85, (1, 2))])
    # Break exact symmetries to help optimization
    pts += np.random.normal(0, 0.001, pts.shape)
    return pts[:N]

def sa_optimize(c0, steps=3000, T_init=0.025):
    """Simulated annealing on center coordinates using LP radii evaluation."""
    c = c0.copy()
    curr_sum = np.sum(solve_lp_radii(c))
    best_sum = curr_sum
    best_c = c.copy()
    
    T = T_init
    for step in range(steps):
        c_new = c.copy()
        if np.random.rand() < 0.7:
            idx = np.random.choice(N, np.random.randint(3, 12), replace=False)
            c_new[idx] += np.random.normal(0, T * 0.4, (len(idx), 2))
        elif np.random.rand() < 0.5:
            ang = np.random.uniform(-T, T)
            ca, sa_val = np.cos(ang), np.sin(ang)
            c_new = (c - 0.5) @ np.array([[ca, -sa_val], [sa_val, ca]]) + 0.5
        else:
            s = np.random.uniform(1 - T*0.15, 1 + T*0.15)
            c_new = (c - 0.5) * s + 0.5
            
        c_new = np.clip(c_new, 0.005, 0.995)
        new_sum = np.sum(solve_lp_radii(c_new))
        
        if new_sum > curr_sum or (T > 1e-5 and np.random.rand() < np.exp((new_sum - curr_sum) / T)):
            c = c_new
            curr_sum = new_sum
            if new_sum > best_sum:
                best_sum = new_sum
                best_c = c.copy()
        T *= 0.9985
        
    r = solve_lp_radii(best_c)
    return best_c, r, best_sum

def coordinate_descent(centers, radii, iterations=40):
    """Systematic local search nudging centers to increase LP radii sum."""
    c = centers.copy()
    r = radii.copy()
    curr_sum = np.sum(r)
    step_sizes = [0.005, 0.002, 0.001, 0.0005]
    
    for _ in range(iterations):
        improved = False
        for i in range(N):
            for step in step_sizes:
                for dx in [-step, 0, step]:
                    for dy in [-step, 0, step]:
                        if dx == 0 and dy == 0:
                            continue
                        c_new = c.copy()
                        c_new[i, 0] = np.clip(c[i, 0] + dx, 0.001, 0.999)
                        c_new[i, 1] = np.clip(c[i, 1] + dy, 0.001, 0.999)
                        r_new = solve_lp_radii(c_new)
                        s_new = np.sum(r_new)
                        if s_new > curr_sum + 1e-11:
                            c = c_new
                            r = r_new
                            curr_sum = s_new
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            break
    return c, r, curr_sum

def run_packing() -> tuple:

            best_r = r.copy()
            
    if best_c is not None:

            if not np.isnan(res.fun) and -res.fun > best_sa_sum:
