import numpy as np
from scipy.optimize import minimize, linprog
import warnings
warnings.filterwarnings('ignore')

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    num_ineq = n + n * (n - 1) // 2
    A = np.zeros((num_ineq, n))
    b = np.zeros(num_ineq)
    idx = 0
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A[idx, i] = 1.0
        b[idx] = max(0.0, lim)
        idx += 1
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = max(0.0, d)
            idx += 1
    bounds = [(0.0, 0.5)] * n
    try:
        res = linprog(-np.ones(n), A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-9)
    except Exception:
        pass
    return np.full(n, 0.02)

def constraints_func(x):
    """Inequality constraints for SLSQP: g(x) >= 0."""
    xs, ys, rs = x[0::3], x[1::3], x[2::3]
    c = np.concatenate([
        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs
    ])
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    # Squared distance avoids sqrt singularity at contact points
    c = np.concatenate([c, dx[TRIL_IDX]**2 + dy[TRIL_IDX]**2 - dr[TRIL_IDX]**2])
    return c

def objective_func(x):
    """Objective: maximize sum of radii."""
    return -np.sum(x[2::3])

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-7, 0.5]."""
    return [(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)] * N

def make_hex_init(r0, angle):
    """Generates a rotated hexagonal lattice initialization."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0 and len(pts) < N + 10:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
    pts = np.array(pts[:N + 10])
    if angle != 0.0:
        ca, sa = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[ca, -sa], [sa, ca]]) + 0.5
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.15, 0.85, (1, 2))])
    return pts[:N]

def make_force_init(seed):
    """Force-directed layout to spread points evenly and push to boundaries."""
    np.random.seed(seed)
    pts = np.random.uniform(0.15, 0.85, (N, 2))
    for _ in range(300):
        f = np.zeros_like(pts)
        for i in range(N):
            for j in range(i + 1, N):
                dx = pts[j] - pts[i]
                d = np.hypot(dx[0], dx[1])
                if d < 0.25 and d > 1e-5:
                    rep = 0.01 / (d**2 + 0.001)
                    f[i] -= dx * rep / d
                    f[j] += dx * rep / d

                if pts[i, dim] < 0.1: f[i, dim] += 0.05
                elif pts[i, dim] > 0.9: f[i, dim] -= 0.05
        pts += f * 0.05
        pts = np.clip(pts, 0.03, 0.97)
    return pts

def coordinate_descent_refine(centers, radii, max_iter=300):
    """Local refinement by moving one circle at a time to boundaries/corners."""
    c = centers.copy()
    r = radii.copy()
    best_sum = np.sum(r)
    for _ in range(max_iter):
        improved = False
        for i in range(N):
            current_r_sum = np.sum(r)
            candidates = [c[i].copy()]
            # Boundaries
            candidates.extend([[r[i], c[i, 1]], [1.0 - r[i], c[i, 1]], [c[i, 0], r[i]], [c[i, 0], 1.0 - r[i]]])
            # Corners
            candidates.extend([[r[i], r[i]], [1.0 - r[i], r[i]], [r[i], 1.0 - r[i]], [1.0 - r[i], 1.0 - r[i]]])
            
            best_cand = c[i]
            best_cand_sum = current_r_sum
            
            for nc in candidates:
                if not (0.005 <= nc[0] <= 0.995 and 0.005 <= nc[1] <= 0.995):
                    continue
                c_tmp = c.copy()
                c_tmp[i] = nc
                r_tmp = solve_lp_radii(c_tmp)
                s_tmp = np.sum(r_tmp)
                if s_tmp > best_cand_sum + 1e-9:
                    best_cand_sum = s_tmp
                    best_cand = nc
            
            if best_cand_sum > current_r_sum + 1e-9:
                c[i] = best_cand
                r = solve_lp_radii(c)
                if np.sum(r) > best_sum + 1e-9:
                    best_sum = np.sum(r)
                improved = True
        if not improved:
            break
    return c, r, best_sum
