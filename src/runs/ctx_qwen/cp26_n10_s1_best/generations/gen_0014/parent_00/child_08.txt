import numpy as np
from scipy.optimize import minimize, linprog
import warnings
warnings.filterwarnings('ignore')

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    num_ineq = n + n * (n - 1) // 2
    A = np.zeros((num_ineq, n))
    b = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A[idx, i] = 1.0
        b[idx] = max(0.0, lim)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = max(0.0, d)
            idx += 1
            
    bounds_lp = [(0.0, 0.5)] * n
    try:
        res = linprog(-np.ones(n), A_ub=A, b_ub=b, bounds=bounds_lp, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-8)
    except Exception:
        pass
    return np.full(n, 0.02)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Inequality constraints for SLSQP: g(x) >= 0."""
    xs, ys, rs = x[0::3], x[1::3], x[2::3]
    c = np.concatenate([xs - rs, 1.0 - xs - rs, ys - rs, 1.0 - ys - rs])
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    c = np.concatenate([c, np.hypot(dx[TRIL_IDX], dy[TRIL_IDX]) - dr[TRIL_IDX]])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-7, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)])
    return b

def make_hex_init(r0, angle):
    """Generates a rotated hexagonal lattice initialization."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0 and len(pts) < N + 10:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
    pts = np.array(pts[:N + 10])
    if angle != 0.0:
        ca, sa = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[ca, -sa], [sa, ca]]) + 0.5
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.15, 0.85, (1, 2))])
    return pts[:N]

def make_force_init(seed):
    """Force-directed layout to spread points evenly and push to boundaries."""
    np.random.seed(seed)
    pts = np.random.uniform(0.15, 0.85, (N, 2))
    for _ in range(150):
        f = np.zeros_like(pts)
        for i in range(N):
            for j in range(i + 1, N):
                dx = pts[j] - pts[i]
                d = np.hypot(dx[0], dx[1])
                if d < 0.3 and d > 1e-5:
                    rep = 0.01 / (d**2 + 0.001)
                    f[i] -= dx * rep / d
                    f[j] += dx * rep / d
            for dim in range(2):
                if pts[i, dim] < 0.12: f[i, dim] += 0.04
                elif pts[i, dim] > 0.88: f[i, dim] -= 0.04
        pts += f * 0.03
        pts = np.clip(pts, 0.02, 0.98)
    return pts

def refine_centers_lp(centers):
    """Coordinate descent on centers using LP oracle for exact radius evaluation."""
    c = centers.copy()
    improved = True
    max_iter = 15
    while improved and max_iter > 0:
        improved = False
        max_iter -= 1
        r_curr = solve_lp_radii(c)
        curr_sum = np.sum(r_curr)
        
        for i in range(N):
            best_s = curr_sum
            best_c_i = c[i].copy()
            # Fine grid search around current position
            steps = np.array([-0.0015, -0.0005, 0.0005, 0.0015])
            for dx in steps:
                for dy in steps:
                    if dx == 0 and dy == 0:
                        continue
                    c_try = c.copy()
                    c_try[i, 0] = np.clip(c_try[i, 0] + dx, 0.001, 0.999)
                    c_try[i, 1] = np.clip(c_try[i, 1] + dy, 0.001, 0.999)
                    r_try = solve_lp_radii(c_try)
                    s_try = np.sum(r_try)
                    if s_try > best_s + 1e-9:
                        best_s = s_try
                        best_c_i = c_try[i].copy()
            if best_s > curr_sum + 1e-9:
                c[i] = best_c_i
                curr_sum = best_s
                improved = True
    r_final = solve_lp_radii(c)
    return c, r_final, np.sum(r_final)

def basin_hop(c0, r0, steps=500):
    """Simulated annealing on centers using LP radii evaluation."""
    c = c0.copy()
    r = r0.copy()
    best_sum = np.sum(r)
    best_c = c.copy()
    best_r = r.copy()
    T = 0.025
    
    for _ in range(steps):
        c_new = c.copy()
        choice = np.random.rand()
        if choice < 0.5:
            idx = np.random.choice(N, np.random.randint(2, 8), replace=False)
            c_new[idx] += np.random.normal(0, 0.004, (len(idx), 2))
        elif choice < 0.75:
            ang = np.random.uniform(-0.03, 0.03)
            ca, sa = np.cos(ang), np.sin(ang)
            c_new = (c - 0.5) @ np.array([[ca, -sa], [sa, ca]]) + 0.5
        else:
            c_new += np.random.uniform(-0.008, 0.008, c.shape)
            
        c_new = np.clip(c_new, 0.005, 0.995)
        r_new = solve_lp_radii(c_new)
        s_new = np.sum(r_new)
        
        if s_new > best_sum or (T > 1e-6 and np.random.rand() < np.exp((s_new - best_sum) / T)):
            c, r = c_new, r_new
            if s_new > best_sum:
                best_sum, best_c, best_r = s_new, c_new.copy(), r_new.copy()
        T *= 0.993
    return best_c, best_r, best_sum

            if not np.isnan(res.fun):
