import numpy as np
from scipy.optimize import minimize, linprog
import warnings
warnings.filterwarnings('ignore')

N = 26
TRIL_IDX = np.tril_indices(N, -1)
TRIU_IDX = np.triu_indices(N, 1)

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    num_ineq = n + n * (n - 1) // 2
    A = np.zeros((num_ineq, n))
    b = np.zeros(num_ineq)
    
    idx = 0
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A[idx, i] = 1.0
        b[idx] = max(0.0, lim)
        idx += 1
        
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = max(0.0, d)
            idx += 1
            
    bounds = [(0.0, 0.5)] * n
    try:
        res = linprog(-np.ones(n), A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-8)
    except Exception:
        pass
    return np.full(n, 0.02)

def center_penalty(c_flat, radii):
    """Smooth penalty function for centers given fixed radii."""
    c = c_flat.reshape(-1, 2)
    x, y = c[:, 0], c[:, 1]
    r = radii
    
    p = 0.0
    p += 1000.0 * np.sum(np.maximum(r - x, 0.0)**2)
    p += 1000.0 * np.sum(np.maximum(x - (1.0 - r), 0.0)**2)
    p += 1000.0 * np.sum(np.maximum(r - y, 0.0)**2)
    p += 1000.0 * np.sum(np.maximum(y - (1.0 - r), 0.0)**2)
    
    dx = c[:, 0, None] - c[:, 0]
    dy = c[:, 1, None] - c[:, 1]
    d = np.hypot(dx, dy)
    gap = d - r[:, None] - r[None, :]
    p += 5000.0 * np.sum(np.maximum(-gap[TRIU_IDX], 0.0)**2)
    return p

def opt_centers_lbfgs(centers, radii):
    """Optimize centers for fixed radii using L-BFGS-B."""
    bounds = [(0.001, 0.999)] * (2 * N)
    res = minimize(center_penalty, centers.flatten(), args=(radii,),
                   method='L-BFGS-B', bounds=bounds, options={'maxiter': 3000, 'ftol': 1e-16})
    return res.x.reshape(-1, 2)

def make_hex_init(r0, angle):
    """Generates a rotated hexagonal lattice initialization."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 15:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0 and len(pts) < N + 15:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
    pts = np.array(pts[:N + 15])
    if angle != 0.0:
        ca, sa = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[ca, -sa], [sa, ca]]) + 0.5
    mask = (pts[:, 0] >= 0.01) & (pts[:, 0] <= 0.99) & (pts[:, 1] >= 0.01) & (pts[:, 1] <= 0.99)
    pts = pts[mask]
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.15, 0.85, (1, 2))])
    return pts[:N]

def make_force_init(seed):
    """Force-directed layout to spread points evenly and push to boundaries."""
    np.random.seed(seed)
    pts = np.random.uniform(0.15, 0.85, (N, 2))
    for step in range(300):
        f = np.zeros_like(pts)

                elif pts[i, dim] > 0.9: f[i, dim] -= 0.04
        pts += f * lr
        pts = np.clip(pts, 0.02, 0.98)
    return pts

def constraints_slsqp(x):
    """Inequality constraints for SLSQP: g(x) >= 0."""
    xs, ys, rs = x[0::3], x[1::3], x[2::3]
    c = np.concatenate([xs - rs, 1.0 - xs - rs, ys - rs, 1.0 - ys - rs])
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    # Squared distance constraints are polynomial and smooth
    c = np.concatenate([c, dx[TRIL_IDX]**2 + dy[TRIL_IDX]**2 - dr[TRIL_IDX]**2])
    return c

def objective_func(x):
    """Objective: maximize sum of radii."""
    return -np.sum(x[2::3])
