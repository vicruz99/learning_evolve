import numpy as np
from scipy.optimize import minimize, linprog
import warnings
warnings.filterwarnings('ignore')

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-8, 0.5]."""
    return [(0.0, 1.0), (0.0, 1.0), (1e-8, 0.5)] * N

def constraints_slsqp(x):
    """Returns all inequality constraints >= 0."""
    xs, ys, rs = x[0::3], x[1::3], x[2::3]
    c = np.concatenate([
        xs - rs, 1.0 - xs - rs,
        ys - rs, 1.0 - ys - rs
    ])
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    c = np.concatenate([c, np.hypot(dx[TRIL_IDX], dy[TRIL_IDX]) - dr[TRIL_IDX]])
    return c

def objective_func(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    num_ineq = n + n * (n - 1) // 2
    A = np.zeros((num_ineq, n))
    b = np.zeros(num_ineq)
    idx = 0
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A[idx, i] = 1.0
        b[idx] = max(0.0, lim)
        idx += 1
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = max(0.0, d)
            idx += 1
    bounds = [(0.0, None)] * n
    try:
        res = linprog(-np.ones(n), A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-9)
    except Exception:
        pass
    return np.full(n, 0.02)

def center_penalty(c_flat, radii):
    """Smooth penalty function for centers given fixed radii."""
    c = c_flat.reshape(-1, 2)
    x, y, r = c[:, 0], c[:, 1], radii
    p = np.sum(np.maximum(r - x, 0.0)**2)
    p += np.sum(np.maximum(x - (1.0 - r), 0.0)**2)
    p += np.sum(np.maximum(r - y, 0.0)**2)
    p += np.sum(np.maximum(y - (1.0 - r), 0.0)**2)
    dx = c[:, 0, None] - c[:, 0]
    dy = c[:, 1, None] - c[:, 1]
    d = np.hypot(dx, dy)
    gap = d - r[:, None] - r[None, :]
    p += np.sum(np.maximum(-gap[TRIL_IDX], 0.0)**2)
    return p

def opt_centers_lbfgs(centers, radii):
    """Optimize centers for fixed radii using L-BFGS-B."""
    bounds = [(1e-4, 0.999)] * (2 * N)
    res = minimize(center_penalty, centers.flatten(), args=(radii,),
                   method='L-BFGS-B', bounds=bounds, options={'maxiter': 2000, 'ftol': 1e-15})
    return res.x.reshape(-1, 2)

def make_hex_init(r0, angle):
    """Generates a rotated hexagonal lattice initialization."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0 and len(pts) < N + 10:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
    pts = np.array(pts[:N + 10])
    if angle != 0.0:
        ca, sa = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[ca, -sa], [sa, ca]]) + 0.5
    mask = (pts[:, 0] >= 0.01) & (pts[:, 0] <= 0.99) & (pts[:, 1] >= 0.01) & (pts[:, 1] <= 0.99)
    pts = pts[mask]
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.2, 0.8, (1, 2))])
    return np.clip(pts[:N], 0.02, 0.98)

def make_force_init(seed):
    """Force-directed layout to spread points evenly."""
    np.random.seed(seed)
    pts = np.random.uniform(0.1, 0.9, (N, 2))
    for _ in range(200):
        f = np.zeros_like(pts)
        for i in range(N):
            for j in range(i + 1, N):
                dx = pts[j] - pts[i]
                d = np.hypot(dx[0], dx[1])
                if d < 0.3 and d > 1e-5:
                    rep = 0.01 / (d**2 + 0.001)
                    f[i] -= dx * rep / d
                    f[j] += dx * rep / d
            for dim in range(2):
                if pts[i, dim] < 0.1: f[i, dim] += 0.05
                elif pts[i, dim] > 0.9: f[i, dim] -= 0.05
        pts += f * 0.04
        pts = np.clip(pts, 0.02, 0.98)
    return pts

def project_to_feasible(x):
    """Project variables to strictly satisfy bounds."""
    x = x.copy()
    for i in range(N):
        r = max(1e-8, x[3 * i + 2])
        x[3 * i] = np.clip(x[3 * i], r, 1.0 - r)
        x[3 * i + 1] = np.clip(x[3 * i + 1], r, 1.0 - r)
        x[3 * i + 2] = r
    return x

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints_slsqp}
    
    best_sum = -1.0
    best_x = None
    
    # Phase 1: Diverse Initial Configurations
    inits = []
    for ang in np.linspace(-0.35, 0.35, 15):
        inits.append(make_hex_init(0.092, ang))
    for s in range(12):
        inits.append(make_force_init(s))
    for s in range(10):
        np.random.seed(s + 100)
        inits.append(np.random.uniform(0.15, 0.85, (N, 2)))

    # Phase 2: Alternating LP & L-BFGS-B + SLSQP Polish
    for c0 in inits:
        c = c0.copy()
        r = solve_lp_radii(c)
        for _ in range(12):
            r = solve_lp_radii(c)
            c = opt_centers_lbfgs(c, r)
        x0 = np.zeros(3 * N)
        x0[0::3] = c[:, 0]
        x0[1::3] = c[:, 1]
        x0[2::3] = np.maximum(r * 0.998, 1e-7)
        try:
            res = minimize(objective_func, x0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 15000, 'ftol': 1e-14, 'disp': False})
            if not np.isnan(res.fun):
                c_r = np.column_stack((res.x[0::3], res.x[1::3]))
                r_lp = solve_lp_radii(c_r)
                s_val = np.sum(r_lp)
                if s_val > best_sum and np.min(constraints_slsqp(res.x)) >= -1e-5:
                    best_sum = s_val
                    best_x = np.zeros(3 * N)
                    best_x[0::3] = c_r[:, 0]
                    best_x[1::3] = c_r[:, 1]
                    best_x[2::3] = r_lp
        except Exception:
            pass

    # Phase 3: Simulated Annealing on Centers with LP Evaluation
    if best_x is not None:
        c_curr = np.column_stack((best_x[0::3], best_x[1::3]))
        best_c = c_curr.copy()
        best_v = best_sum
        T = 0.035
        for step in range(5000):
            T *= 0.998
            k = np.random.randint(2, 8)
            idx = np.random.choice(N, k, replace=False)
            c_new = c_curr.copy()
            c_new[idx] += np.random.normal(0, T, (k, 2))
            c_new = np.clip(c_new, 1e-4, 1.0 - 1e-4)
            v_new = np.sum(solve_lp_radii(c_new))
            delta = v_new - best_v
            if delta > 0 or (T > 1e-11 and np.random.rand() < np.exp(delta / (T + 1e-12))):
                c_curr = c_new
                if v_new > best_v:
                    best_v = v_new
                    best_c = c_new.copy()
        if best_v > best_sum:
            best_sum = best_v
            best_x = np.zeros(3 * N)
            best_x[0::3] = best_c[:, 0]
            best_x[1::3] = best_c[:, 1]
            best_x[2::3] = solve_lp_radii(best_c)

    # Phase 4: Deflation & Repositioning to Escape Local Minima
    if best_x is not None:
        for cyc in range(60):
            x0 = best_x.copy()
            noise = 0.002 * (0.90 ** cyc)
            x0[0::3] += np.random.normal(0, noise, N)
            x0[1::3] += np.random.normal(0, noise, N)
            subset = np.random.choice(N, size=max(4, N // 3), replace=False)
            x0[subset * 3 + 2] *= 0.80
            x0 = project_to_feasible(x0)
            try:
                res = minimize(objective_func, x0, method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 10000, 'ftol': 1e-14, 'disp': False})
                if not np.isnan(res.fun):
                    c_p = np.column_stack((res.x[0::3], res.x[1::3]))
                    r_p = solve_lp_radii(c_p)
                    x_p = res.x.copy()
                    x_p[2::3] = r_p
                    x_p = project_to_feasible(x_p)
                    s_val = np.sum(r_p)
                    if s_val > best_sum and np.min(constraints_slsqp(x_p)) >= -1e-5:
                        best_sum = s_val
                        best_x = x_p.copy()
            except Exception:
                pass

    # Phase 5: Global Rotation Search
    if best_x is not None:
        bc = np.column_stack((best_x[0::3], best_x[1::3]))
        for ang_deg in np.linspace(-12, 12, 13):
            ang = np.deg2rad(ang_deg)
            ca, sa = np.cos(ang), np.sin(ang)
            rot_c = (bc - 0.5) @ np.array([[ca, -sa], [sa, ca]]) + 0.5
            rot_c = np.clip(rot_c, 0.01, 0.99)
            r_rot = solve_lp_radii(rot_c)
            x_rot = np.zeros(3 * N)
            x_rot[0::3] = rot_c[:, 0]
            x_rot[1::3] = rot_c[:, 1]
            x_rot[2::3] = r_rot * 0.99
            try:
                res = minimize(objective_func, x_rot, method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 8000, 'ftol': 1e-14, 'disp': False})
                if not np.isnan(res.fun):
                    c_r = np.column_stack((res.x[0::3], res.x[1::3]))
                    r_r = solve_lp_radii(c_r)
                    if np.sum(r_r) > best_sum:
                        best_sum = np.sum(r_r)
                        best_x = np.zeros(3 * N)
                        best_x[0::3] = c_r[:, 0]
                        best_x[1::3] = c_r[:, 1]
                        best_x[2::3] = r_r
            except Exception:
                pass

    # Phase 6: Coordinate Descent Fine-Tuning
    if best_x is not None:
        c_c = np.column_stack((best_x[0::3], best_x[1::3]))
        improved = True
        while improved:
            improved = False
            for i in range(N):
                base_s = np.sum(best_x[2::3])
                for dx in [-0.003, -0.001, 0.001, 0.003]:
                    for dy in [-0.003, -0.001, 0.001, 0.003]:
                        if dx == 0 and dy == 0:
                            continue
                        nc = c_c.copy()
                        nc[i, 0] = np.clip(c_c[i, 0] + dx, 0.01, 0.99)
                        nc[i, 1] = np.clip(c_c[i, 1] + dy, 0.01, 0.99)
                        rn = solve_lp_radii(nc)
                        sn = np.sum(rn)
                        if sn > base_s + 1e-8:
                            c_c = nc
                            best_x[0::3] = c_c[:, 0]
                            best_x[1::3] = c_c[:, 1]
                            best_x[2::3] = rn
                            best_sum = sn
                            improved = True
                            break
                    if improved:
                        break

    # Fallback
    if best_x is None:
        best_x = np.zeros(3 * N)
        best_x[0::3] = np.tile(np.linspace(0.1, 0.9, 5), 6)[:N]
        best_x[1::3] = np.repeat(np.linspace(0.1, 0.9, 6), 5)[:N]
        best_x[2::3] = 0.06
        best_sum = np.sum(best_x[2::3])

    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3].copy()

    # Final strict validity adjustment against 1e-12 tolerance
    for _ in range(100):
        valid = True
        for i in range(N):
            if radii[i] < 0 or centers[i, 0] < radii[i] - 1e-9 or centers[i, 0] > 1.0 - radii[i] + 1e-9 or \
               centers[i, 1] < radii[i] - 1e-9 or centers[i, 1] > 1.0 - radii[i] + 1e-9:
                valid = False
                break
        if valid:
            for i in range(N):
                for j in range(i + 1, N):
                    d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                    if d < radii[i] + radii[j] - 1e-9:
                        valid = False
                        break
                if not valid:
                    break
        if valid:
            break
        radii *= 0.9999
        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)

    return centers, radii, float(np.sum(radii))
