3. **Simulated Annealing on Centers**: Optimize centers directly by evaluating the LP sum of radii as the objective. This avoids non-convex constraint handling during center search and allows efficient exploration.
4. **SLSQP Joint Polish**: After SA finds a good center configuration, use SLSQP to jointly optimize centers and radii with strict inequality constraints for final precision.
5. **Deflation & Coordinate Descent**: Aggressively break rigid contact networks by shrinking random subsets of radii, re-optimizing, and fine-tuning individual circle positions.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
TRIL_IDX = np.tril_indices(N, -1)
NUM_PAIRS = N * (N - 1) // 2
NUM_CONS = N + NUM_PAIRS

# Precompute constant LP constraint matrix for speed
A_LP = np.zeros((NUM_CONS, N))
for i in range(N):
    A_LP[i, i] = 1.0
idx = N
for i in range(N):
    for j in range(i + 1, N):
        A_LP[idx, i] = 1.0
        A_LP[idx, j] = 1.0
        idx += 1

def objective_func(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraint_func(x):
    """Returns all inequality constraints >= 0."""
    xs, ys, rs = x[0::3], x[1::3], x[2::3]
    c = np.concatenate([
        xs - rs, 1.0 - xs - rs,
        ys - rs, 1.0 - ys - rs
    ])
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    c = np.concatenate([c, np.hypot(dx[TRIL_IDX], dy[TRIL_IDX]) - dr[TRIL_IDX]])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-8, 0.5]."""
    return [(0.0, 1.0), (0.0, 1.0), (1e-8, 0.5)] * N

def solve_lp_radii_fast(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    b = np.zeros(NUM_CONS)
    for i in range(N):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        b[i] = max(0.0, lim)
        
    for i in range(N):
        for j in range(i + 1, N):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            idx = N + (i * (2 * N - 1 - i)) // 2 + j - i - 1
            b[idx] = max(0.0, d)
            
    try:
        res = linprog(-np.ones(N), A_ub=A_LP, b_ub=b, bounds=[(0.0, None)] * N, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-9)
    except Exception:
        pass
    return np.full(N, 0.05)

def make_hex_init(r0, angle, scale=1.0):
    """Generates a rotated hexagonal lattice initialization."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0 and len(pts) < N + 10:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        
    pts = np.array(pts[:N + 10])
    pts = (pts - 0.5) * scale + 0.5
    
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
        
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.1, 0.9, 2)])
    return np.clip(pts[:N], 0.05, 0.95)

def make_pattern_init():
    """Generates explicit row-based patterns tailored for N=26."""
    patterns = [
        [6, 5, 6, 5, 4],
        [5, 6, 5, 6, 4],
        [7, 5, 5, 5, 4],
        [6, 6, 5, 5, 4],
        [8, 6, 6, 4, 2]
    ]
    inits = []
    for rows in patterns:
        pts = []
        y_step = 0.85 / (len(rows) - 1 + 1.2)
        curr_y = 0.05 + y_step
        for r_idx, count in enumerate(rows):
            if count == 1:
                pts.append([0.5, curr_y])
            else:
                x_step = 0.9 / (count - 1)
                offset = 0.05 + (r_idx % 2) * (x_step / 2.0) * 0.6
                for k in range(count):
                    pts.append([offset + k * x_step, curr_y])
            curr_y += y_step
        pts = np.array(pts[:N])
        pts += np.random.uniform(-0.005, 0.005, pts.shape)
        inits.append(np.clip(pts, 0.02, 0.98))
    return inits

def project_to_feasible(x):
    """Project variables to strictly satisfy bounds."""
    x = x.copy()
    for i in range(N):
        r = max(1e-8, x[3 * i + 2])
        x[3 * i] = np.clip(x[3 * i], r, 1.0 - r)
        x[3 * i + 1] = np.clip(x[3 * i + 1], r, 1.0 - r)
        x[3 * i + 2] = r
    return x

def sa_optimize(centers, n_steps=600, temp_init=0.012):
    """Simulated annealing on centers evaluated by LP radii sum."""
    curr_c = centers.copy()
    best_c = curr_c.copy()
    best_val = np.sum(solve_lp_radii_fast(curr_c))
    temp = temp_init
    
    for _ in range(n_steps):
        temp *= 0.997
        k = np.random.randint(1, 5)
        idx = np.random.choice(N, k, replace=False)
        new_c = curr_c.copy()
        new_c[idx] += np.random.normal(0, temp, (k, 2))
        new_c = np.clip(new_c, 1e-5, 1.0 - 1e-5)
        
        val = np.sum(solve_lp_radii_fast(new_c))
        delta = val - best_val
        if delta > 0 or (temp > 1e-12 and np.random.rand() < np.exp(delta / (temp + 1e-12))):
            curr_c = new_c
            if val > best_val:
                best_val = val
                best_c = curr_c.copy()
    return best_c, best_val

def run_packing() -> tuple:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraint_func}
    
    best_sum = -1.0
    best_x = None
    
    # Phase 1: Generate Diverse Initial Configurations
    inits = []
    for scale in np.linspace(0.9, 1.2, 7):
        for ang in np.linspace(-0.4, 0.4, 7):
            inits.append(make_hex_init(0.09 * scale, ang, scale))
    inits.extend(make_pattern_init())
    
    for s in range(12):
        np.random.seed(s * 13 + 7)
        pts = np.random.uniform(0.15, 0.85, (N, 2))
        for _ in range(100):
            f = np.zeros_like(pts)
            for i in range(N):
                for j in range(i + 1, N):
                    d = np.hypot(pts[j, 0] - pts[i, 0], pts[j, 1] - pts[i, 1])
                    if d < 0.3 and d > 1e-4:
                        rep = 0.01 / (d**2 + 0.001)
                        dx = pts[j] - pts[i]
                        f[i] -= dx * rep / d
                        f[j] += dx * rep / d
                if pts[i, 0] < 0.1: f[i, 0] += 0.05
                elif pts[i, 0] > 0.9: f[i, 0] -= 0.05
                if pts[i, 1] < 0.1: f[i, 1] += 0.05
                elif pts[i, 1] > 0.9: f[i, 1] -= 0.05
            pts += f * 0.04
            pts = np.clip(pts, 0.05, 0.95)
        inits.append(pts)
        
    # Phase 2: SA + SLSQP Pipeline on each init
    for c0 in inits:
        x0 = np.zeros(3 * N)
        x0[0::3] = c0[:, 0]
        x0[1::3] = c0[:, 1]
        r0 = solve_lp_radii_fast(c0)
        x0[2::3] = r0 * 0.99
        x0 = project_to_feasible(x0)
        
        # SA on centers
        c_sa, v_sa = sa_optimize(c0, n_steps=500, temp_init=0.01)
        if v_sa > np.sum(r0):
            x0[0::3] = c_sa[:, 0]
            x0[1::3] = c_sa[:, 1]
            x0[2::3] = solve_lp_radii_fast(c_sa) * 0.995
        
        x0 = project_to_feasible(x0)
        try:
            res = minimize(objective_func, x0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 15000, 'ftol': 1e-14, 'disp': False})
            if not np.isnan(res.fun):
                x_tmp = res.x.copy()
                c_tmp = np.column_stack((x_tmp[0::3], x_tmp[1::3]))
                r_lp = solve_lp_radii_fast(c_tmp)
                x_tmp[2::3] = r_lp
                x_tmp = project_to_feasible(x_tmp)
                s_val = np.sum(r_lp)
                if s_val > best_sum and np.min(constraint_func(x_tmp)) >= -1e-5:
                    best_sum = s_val
                    best_x = x_tmp.copy()
        except Exception:

    # Phase 3: Deflation & Repositioning to Escape Local Minima

            subset = np.random.choice(N, size=max(3, N // 3), replace=False)

                    s_val = np.sum(r_p)
