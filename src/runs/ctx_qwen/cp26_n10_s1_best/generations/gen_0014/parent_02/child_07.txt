import numpy as np
from scipy.optimize import minimize, linprog
import warnings

warnings.filterwarnings('ignore')

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    num_ineq = n + n * (n - 1) // 2
    A = np.zeros((num_ineq, n))
    b = np.zeros(num_ineq)
    
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    A[:n, np.arange(n)] = 1.0
    b[:n] = np.minimum(
        np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
        np.minimum(centers[:, 1], 1.0 - centers[:, 1])
    )
    
    # Overlap constraints: r_i + r_j <= dist(i, j)
    ixs, jxs = TRIL_IDX
    A[n:, ixs] = 1.0
    A[n:, jxs] = 1.0
    dx = centers[ixs, 0] - centers[jxs, 0]
    dy = centers[ixs, 1] - centers[jxs, 1]
    b[n:] = np.hypot(dx, dy)
    
    bounds = [(0.0, None)] * n
    try:
        res = linprog(-np.ones(n), A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-9)
    except Exception:
        pass
    return np.full(n, 0.05)

def constraint_func(x):
    """Returns all inequality constraints >= 0 for SLSQP."""
    xs, ys, rs = x[0::3], x[1::3], x[2::3]
    
    # Boundary constraints
    c = np.concatenate([
        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs
    ])
    
    # Overlap constraints: dist >= r_i + r_j
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    c = np.concatenate([c, np.hypot(dx[TRIL_IDX], dy[TRIL_IDX]) - dr[TRIL_IDX]])
    return c

def objective_func(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-8, 0.5]."""
    return [(0.0, 1.0), (0.0, 1.0), (1e-8, 0.5)] * N

def make_hex_init(r0, angle):
    """Generates a rotated hexagonal lattice initialization."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0 and len(pts) < N + 10:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        
    pts = np.array(pts[:N + 10])
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
        
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.2, 0.8, 2)])
    return np.clip(pts[:N], 0.05, 0.95)

def make_staggered_init(rows_counts):
    """Generates a staggered row initialization."""
    pts = []
    y = 0.10
    for idx, cnt in enumerate(rows_counts):
        x_step = 0.80 / (cnt - 1) if cnt > 1 else 0.0
        x_start = 0.10 + (idx % 2) * (x_step * 0.5)
        for k in range(cnt):
            x = x_start + k * x_step
            pts.append([x, y])
        y += 0.17
    return np.array(pts[:N])

def force_spread(centers, steps=150):
    """Force-directed relaxation to spread points evenly."""
    pts = centers.copy()
    for _ in range(steps):
        f = np.zeros_like(pts)
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(pts[i, 0] - pts[j, 0], pts[i, 1] - pts[j, 1])
                if d < 0.35 and d > 1e-4:
                    rep = 0.015 / (d**2 + 0.001)
                    dx = pts[j] - pts[i]
                    f[i] -= dx * rep / d
                    f[j] += dx * rep / d
            for dim in range(2):
                if pts[i, dim] < 0.08: f[i, dim] += 0.06
                elif pts[i, dim] > 0.92: f[i, dim] -= 0.06
        pts += f * 0.05
        pts = np.clip(pts, 0.02, 0.98)
    return pts

def center_penalty(c_flat, radii):
    """Smooth penalty function for centers given fixed radii."""
    c = c_flat.reshape(-1, 2)
    p = 0.0
    x, y, r = c[:, 0], c[:, 1], radii
    
    p += 100.0 * np.sum(np.maximum(r - x, 0.0)**2)
    p += 100.0 * np.sum(np.maximum(x - (1.0 - r), 0.0)**2)
    p += 100.0 * np.sum(np.maximum(r - y, 0.0)**2)
    p += 100.0 * np.sum(np.maximum(y - (1.0 - r), 0.0)**2)
    
    dx = c[:, 0, None] - c[:, 0]
    dy = c[:, 1, None] - c[:, 1]
    d = np.hypot(dx, dy)
    gap = d - r[:, None] - r[None, :]
    p += 500.0 * np.sum(np.maximum(-gap[TRIL_IDX], 0.0)**2)
    return p

def opt_centers_lbfgs(centers, radii):
    """Optimize centers for fixed radii using L-BFGS-B."""
    bounds = [(0.001, 0.999)] * (2 * N)
    res = minimize(center_penalty, centers.flatten(), args=(radii,),
                   method='L-BFGS-B', bounds=bounds, options={'maxiter': 2500, 'ftol': 1e-15})
    return res.x.reshape(-1, 2)

def project_to_feasible(x):
    """Project variables to strictly satisfy bounds."""
    x = x.copy()
    for i in range(N):
        r = max(1e-8, x[3 * i + 2])
        x[3 * i] = np.clip(x[3 * i], r, 1.0 - r)
        x[3 * i + 1] = np.clip(x[3 * i + 1], r, 1.0 - r)
        x[3 * i + 2] = r
    return x

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    bounds = get_bounds()

    best_sum = -1.0
    best_x = None

    inits = []

    row_patterns = [[6,5,6,5,4], [5,6,5,6,4], [6,6,6,4,4], [7,6,7,6], [8,6,6,6]]
    for pat in row_patterns:
        inits.append(make_staggered_init(pat))
        
    # Random starts with force-directed spread
    for s in range(12):
        np.random.seed(s * 17 + 7)
        pts = np.random.uniform(0.15, 0.85, (N, 2))
        inits.append(force_spread(pts, steps=120))

    # Phase 2: Alternating Optimization + SLSQP Polish
    for c_init in inits:
        c = c_init.copy()
        r = solve_lp_radii(c)
        
        # Alternate LP radii and L-BFGS-B centers
        for _ in range(5):
            r = solve_lp_radii(c)
            c = opt_centers_lbfgs(c, r)
            
        x0 = np.zeros(3 * N)
        x0[0::3] = c[:, 0]
        x0[1::3] = c[:, 1]
        x0[2::3] = np.maximum(r * 0.995, 1e-6)
        x0 = project_to_feasible(x0)
        
        try:
            res = minimize(objective_func, x0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 15000, 'ftol': 1e-14, 'disp': False})
            if not np.isnan(res.fun):
                x_tmp = res.x.copy()
                c_tmp = np.column_stack((x_tmp[0::3], x_tmp[1::3]))
                r_lp = solve_lp_radii(c_tmp)
                x_tmp[2::3] = r_lp
                x_tmp = project_to_feasible(x_tmp)
                s_val = np.sum(r_lp)
                if s_val > best_sum and np.min(constraint_func(x_tmp)) >= -1e-5:
                    best_sum = s_val
                    best_x = x_tmp.copy()
        except Exception:
            pass

    # Phase 3: Simulated Annealing on Centers with LP Evaluation
    if best_x is not None:
        curr_c = np.column_stack((best_x[0::3], best_x[1::3]))
        best_c = curr_c.copy()
        best_v = best_sum

                    best_c = new_c.copy()

            best_x[0::3] = best_c[:, 0]
            best_x[1::3] = best_c[:, 1]
            best_x[2::3] = solve_lp_radii(best_c)

            # Shrink random subset to break rigid contact networks

                    r_p = solve_lp_radii(c_p)
                    x_p = res.x.copy()
                    x_p[2::3] = r_p
                    x_p = project_to_feasible(x_p)
                    s_val = np.sum(r_p)

    centers = np.column_stack((best_x[0::3], best_x[1::3]))

    # Final strict validity adjustment against 1e-12 tolerance
    for _ in range(100):
