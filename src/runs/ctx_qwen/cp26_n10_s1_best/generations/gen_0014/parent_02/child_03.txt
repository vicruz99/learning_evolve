import numpy as np
from scipy.optimize import minimize, linprog
import warnings

warnings.filterwarnings('ignore')

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def get_lp_matrices():
    """Precompute the constant A matrix for the LP radius solver."""
    n = N
    num_ineq = n + n * (n - 1) // 2
    A = np.zeros((num_ineq, n))
    idx = 0
    for i in range(n):
        A[idx, i] = 1.0
        idx += 1
    for i in range(n):
        for j in range(i + 1, n):
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            idx += 1
    return A

A_LP = get_lp_matrices()

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    b = np.zeros(n + n * (n - 1) // 2)
    idx = 0
    for i in range(n):
        x, y = centers[i]
        lim = min(x, 1.0 - x, y, 1.0 - y)
        b[idx] = max(0.0, lim)
        idx += 1
        
    dx = centers[:, 0, None] - centers[:, 0]
    dy = centers[:, 1, None] - centers[:, 1]
    dists = np.hypot(dx[TRIL_IDX], dy[TRIL_IDX])
    b[idx:] = dists
    
    try:
        res = linprog(-np.ones(n), A_ub=A_LP, b_ub=b, bounds=[(0.0, None)] * n, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-9)
    except Exception:
        pass
    return np.full(n, 0.02)

def force_layout(pts, steps=200):
    """Force-directed relaxation to spread points evenly and push to boundaries."""
    pts = pts.copy()
    for _ in range(steps):
        f = np.zeros_like(pts)
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(pts[i,0]-pts[j,0], pts[i,1]-pts[j,1])
                if d < 0.25 and d > 1e-5:
                    rep = 0.015 / (d**2 + 0.001)
                    f[i] -= (pts[i]-pts[j]) * rep / d
                    f[j] += (pts[i]-pts[j]) * rep / d
            for dim in range(2):
                if pts[i, dim] < 0.1: f[i, dim] += 0.06
                elif pts[i, dim] > 0.9: f[i, dim] -= 0.06
        pts += f * 0.03
        pts = np.clip(pts, 0.02, 0.98)
    return pts

def make_hex_init(r0, angle, scale=1.0):
    """Generates a rotated hexagonal lattice initialization."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0 and len(pts) < N + 10:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
    pts = np.array(pts[:N + 10])
    pts = (pts - 0.5) * scale + 0.5
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.2, 0.8, 2)])
    return np.clip(pts[:N], 0.05, 0.95)

def sa_optimize_centers(c0, steps=2000):
    """Simulated Annealing on centers to maximize sum of LP-optimal radii."""
    c = c0.copy()
    best_c = c.copy()
    best_v = np.sum(solve_lp_radii(c))
    temp = 0.03
    
    for step in range(steps):
        temp *= 0.9985
        k = np.random.randint(3, 8)
        idx = np.random.choice(N, k, replace=False)
        new_c = c.copy()
        new_c[idx] += np.random.normal(0, temp, (k, 2))
        new_c = np.clip(new_c, 1e-4, 1.0 - 1e-4)
        
        v_new = np.sum(solve_lp_radii(new_c))
        delta = v_new - best_v
        
        if delta > 0 or (temp > 1e-10 and np.random.rand() < np.exp(delta / (temp + 1e-12))):
            c = new_c
            if v_new > best_v:
                best_v = v_new
                best_c = c.copy()
                
    return best_c, best_v

def coord_descent(c0):
    """Deterministic coordinate descent to fine-tune centers."""
    c = c0.copy()
    best_v = np.sum(solve_lp_radii(c))
    improved = True
    while improved:
        improved = False
        for i in range(N):
            for dx in [-0.004, -0.002, -0.001, 0.001, 0.002, 0.004]:
                for dy in [-0.004, -0.002, -0.001, 0.001, 0.002, 0.004]:
                    if dx == 0 and dy == 0: continue
                    nc = c.copy()
                    nc[i, 0] = np.clip(nc[i, 0] + dx, 1e-4, 1.0-1e-4)
                    nc[i, 1] = np.clip(nc[i, 1] + dy, 1e-4, 1.0-1e-4)
                    v = np.sum(solve_lp_radii(nc))
                    if v > best_v + 1e-7:
                        c = nc
                        best_v = v
                        improved = True
                        break
                if improved: break
            if improved: break
    return c, best_v

def objective(x):
    """Objective for joint optimization: maximize sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Inequality constraints for joint optimization: g(x) >= 0."""
    xs, ys, rs = x[0::3], x[1::3], x[2::3]
    c = np.concatenate([xs - rs, 1.0 - xs - rs, ys - rs, 1.0 - ys - rs])
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    c = np.concatenate([c, np.hypot(dx[TRIL_IDX], dy[TRIL_IDX]) - dr[TRIL_IDX]])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-8, 0.5]."""
    return [(0.0, 1.0), (0.0, 1.0), (1e-8, 0.5)] * N

def slsqp_polish(x0):
    """Joint SLSQP optimization."""
    try:
        res = minimize(objective, x0, method='SLSQP', bounds=get_bounds(), 
                       constraints={'type': 'ineq', 'fun': constraints},
                       options={'maxiter': 10000, 'ftol': 1e-14, 'disp': False})
        if res.success and not np.isnan(res.fun):
            return res.x
    except Exception:
        pass
    return None

def project(x):
    """Project variables to strictly satisfy bounds."""
    x = x.copy()
    for i in range(N):
        r = max(1e-8, x[3*i+2])
        x[3*i] = np.clip(x[3*i], r, 1.0-r)
        x[3*i+1] = np.clip(x[3*i+1], r, 1.0-r)
        x[3*i+2] = r
    return x

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    best_c = None
    best_v = -1.0
    best_x = None
    
    # Phase 1: Generate diverse initial configurations
    inits = []
    for ang in np.linspace(-0.4, 0.4, 11):
        for sc in [0.95, 1.0, 1.05]:
            inits.append(make_hex_init(0.09, ang, sc))
    for s in range(10):
        np.random.seed(s*7+1)
        pts = np.random.uniform(0.15, 0.85, (N, 2))
        inits.append(force_layout(pts, steps=150))
        
    # Phase 2: Simulated Annealing to find strong basins
    for c_init in inits:
        c_sa, v_sa = sa_optimize_centers(c_init, steps=1200)
        if v_sa > best_v:
            best_v = v_sa
            best_c = c_sa.copy()
            
    # Run longer SA on the best found so far
    best_c, best_v = sa_optimize_centers(best_c, steps=2500)
    
    # Phase 3: Coordinate Descent Fine-Tuning
    best_c, best_v = coord_descent(best_c)
    
    # Phase 4: SLSQP Joint Polish
    r_lp = solve_lp_radii(best_c)
    x0 = np.zeros(3 * N)
    x0[0::3] = best_c[:, 0]
    x0[1::3] = best_c[:, 1]
    x0[2::3] = r_lp * 0.99
    x0 = project(x0)
    
    res_x = slsqp_polish(x0)
    if res_x is not None:
        c_tmp = np.column_stack((res_x[0::3], res_x[1::3]))
        r_tmp = solve_lp_radii(c_tmp)
        if np.sum(r_tmp) > best_v:
            best_v = np.sum(r_tmp)
            best_c = c_tmp.copy()
            best_x = res_x.copy()
            best_x[2::3] = r_tmp
            
    # Phase 5: Deflation & Recovery to escape deep local minima
    if best_x is not None:
        for cyc in range(50):
            x0 = best_x.copy()
            noise = 0.0025 * (0.92 ** cyc)
            x0[0::3] += np.random.normal(0, noise, N)
            x0[1::3] += np.random.normal(0, noise, N)
            subset = np.random.choice(N, size=max(4, N//3), replace=False)
            x0[subset*3+2] *= 0.80
            x0 = project(x0)
            
            res_x = slsqp_polish(x0)
            if res_x is not None:
                c_p = np.column_stack((res_x[0::3], res_x[1::3]))
                r_p = solve_lp_radii(c_p)
                v_p = np.sum(r_p)
                if v_p > best_v + 1e-7:

                    best_c = c_p.copy()
                    best_x = res_x.copy()
                    best_x[2::3] = r_p

                    best_c, best_v = coord_descent(best_c)
                    best_x[0::3] = best_c[:, 0]
                    best_x[1::3] = best_c[:, 1]
                    best_x[2::3] = solve_lp_radii(best_c)

    radii = solve_lp_radii(centers)
    
    # Phase 6: Strict validation repair against 1e-12 tolerance
    for _ in range(100):
        valid = True
        for i in range(N):

                break

                    d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
                    if d < radii[i] + radii[j] - 1e-9:
                        valid = False
                        break
                if not valid: break
        if valid: break
        radii *= 0.9998
        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0-radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0-radii)
        
    return centers, radii, float(np.sum(radii))
