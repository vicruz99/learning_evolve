1. **Diverse Initial Configurations**: We generate starting points using rotated/scaled hexagonal lattices and force-directed layouts. Hexagonal packing is theoretically dense, while force-directed spreading ensures points are well-distributed before optimization begins.
2. **Joint Constrained Optimization (SLSQP)**: We use SLSQP to jointly optimize centers and radii with hard boundary and non-overlap constraints. After each SLSQP run, we re-solve for radii using Linear Programming (LP) to find the exact maximal radii for the optimized centers, ensuring no slack is left.
3. **Alternating LP / L-BFGS-B**: SLSQP can sometimes stagnate when many constraints are active. We alternate between fixing centers to solve LP for radii, and fixing radii to optimize centers using L-BFGS-B with a smooth penalty function. This alternating scheme is highly robust for circle packing.
4. **Deflation & Perturbation**: To break rigid contact networks that trap the optimizer, we periodically shrink a random subset of radii, add controlled noise to centers, and re-optimize. This allows the contact graph topology to change.
5. **Rigid Body Subset Rotation**: We randomly select subsets of circles, rotate them around their centroid, and accept the move if it increases the total radius sum. This explores new contact configurations without destroying the overall structure.
6. **Coordinate Descent Fine-Tuning**: Finally, we perform a systematic grid search around each circle's position to squeeze out the last fraction of radius sum.
7. **Strict Validation & Repair**: A final pass ensures all constraints are satisfied within numerical tolerance, applying minimal shrinkage only if necessary.

import numpy as np
from scipy.optimize import minimize, linprog
import warnings
warnings.filterwarnings('ignore')

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-8, 0.5]."""
    return [(0.0, 1.0), (0.0, 1.0), (1e-8, 0.5)] * N

def constraints(x):
    """Returns all inequality constraints >= 0."""
    xs, ys, rs = x[0::3], x[1::3], x[2::3]
    c = np.concatenate([
        xs - rs, 1.0 - xs - rs,
        ys - rs, 1.0 - ys - rs
    ])
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    c = np.concatenate([c, np.hypot(dx[TRIL_IDX], dy[TRIL_IDX]) - dr[TRIL_IDX]])
    return c

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    num_ineq = n + n*(n-1)//2
    A = np.zeros((num_ineq, n))
    b = np.zeros(num_ineq)
    idx = 0
    for i in range(n):
        lim = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
        A[idx, i] = 1.0
        b[idx] = max(0.0, lim)
        idx += 1
    for i in range(n):
        for j in range(i+1, n):
            d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = max(0.0, d)
            idx += 1
    try:
        res = linprog(-np.ones(n), A_ub=A, b_ub=b, bounds=[(0.0, None)]*n, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-9)
    except Exception:
        pass
    return np.full(n, 0.05)

def project(x):
    """Project variables to strictly satisfy bounds."""
    x = x.copy()
    for i in range(N):
        r = max(1e-8, x[3*i+2])
        x[3*i] = np.clip(x[3*i], r, 1.0-r)
        x[3*i+1] = np.clip(x[3*i+1], r, 1.0-r)
        x[3*i+2] = r
    return x

def center_penalty(c_flat, radii):
    """Smooth penalty function for centers given fixed radii."""
    c = c_flat.reshape(-1, 2)
    p = 0.0
    x, y, r = c[:, 0], c[:, 1], radii
    p += np.sum(np.maximum(r - x, 0.0)**2)
    p += np.sum(np.maximum(x - (1.0 - r), 0.0)**2)
    p += np.sum(np.maximum(r - y, 0.0)**2)
    p += np.sum(np.maximum(y - (1.0 - r), 0.0)**2)
    dx = c[:, 0, None] - c[:, 0]
    dy = c[:, 1, None] - c[:, 1]
    d = np.hypot(dx, dy)
    gap = d - r[:, None] - r[None, :]
    p += 10.0 * np.sum(np.maximum(-gap[TRIL_IDX], 0.0)**2)
    return p

def opt_centers_lbfgs(centers, radii):
    """Optimize centers for fixed radii using L-BFGS-B."""
    bounds_c = [(0.001, 0.999)] * (2 * N)
    res = minimize(center_penalty, centers.flatten(), args=(radii,),
                   method='L-BFGS-B', bounds=bounds_c, options={'maxiter': 5000, 'ftol': 1e-15})
    return res.x.reshape(-1, 2)

def make_hex(r0, angle, scale=1.0):
    """Generates a rotated/scaled hexagonal lattice initialization."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row%2==0 else 2.0*r0
        while x <= 1.0-r0 and len(pts) < N+10:
            pts.append([x, y])
            x += 2.0*r0
        y += np.sqrt(3.0)*r0
        row += 1
    pts = np.array(pts[:N+10])
    pts = (pts - 0.5)*scale + 0.5
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
    mask = (pts[:,0]>=0.02) & (pts[:,0]<=0.98) & (pts[:,1]>=0.02) & (pts[:,1]<=0.98)
    pts = pts[mask]
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.2, 0.8, 2)])
    return np.clip(pts[:N], 0.05, 0.95)

def force_spread(pts, steps=200):
    """Force-directed relaxation to spread points evenly and push to boundaries."""
    for _ in range(steps):
        f = np.zeros_like(pts)
        for i in range(N):
            for j in range(i+1, N):
                d = np.hypot(pts[j,0]-pts[i,0], pts[j,1]-pts[i,1])
                if d < 0.35 and d > 1e-4:
                    rep = 0.01 / (d**2 + 0.001)
                    dx = pts[j] - pts[i]
                    f[i] -= dx * rep / d
                    f[j] += dx * rep / d
            for dim in range(2):
                if pts[i, dim] < 0.08: f[i, dim] += 0.04
                elif pts[i, dim] > 0.92: f[i, dim] -= 0.04
        pts += f * 0.03
        pts = np.clip(pts, 0.02, 0.98)
    return pts

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    best_sum = -1.0
    best_x = None

    # Phase 1: Diverse Initial Configurations
    inits = []
    for s in [0.95, 1.0, 1.05, 1.1]:
        for ang in np.linspace(-0.4, 0.4, 9):
            inits.append(make_hex(0.09*s, ang, s))
    for s in range(12):
        np.random.seed(s*13)
        pts = np.random.uniform(0.15, 0.85, (N, 2))
        inits.append(force_spread(pts))

    # Phase 2: SLSQP Joint Optimization on each init
    for c0 in inits:
        x0 = np.zeros(3*N)
        x0[0::3] = c0[:,0]; x0[1::3] = c0[:,1]
        x0[2::3] = solve_lp_radii(c0) * 0.99
        x0 = project(x0)
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 30000, 'ftol': 1e-14})
            if not np.isnan(res.fun):
                x_tmp = res.x.copy()
                c_tmp = np.column_stack((x_tmp[0::3], x_tmp[1::3]))
                r_lp = solve_lp_radii(c_tmp)
                x_tmp[2::3] = r_lp
                x_tmp = project(x_tmp)
                s_val = np.sum(r_lp)
                if s_val > best_sum and np.min(constraints(x_tmp)) >= -1e-5:
                    best_sum = s_val
                    best_x = x_tmp.copy()
        except Exception:
            pass

    # Phase 3: Alternating LP / L-BFGS-B Refinement
    if best_x is not None:
        curr_c = np.column_stack((best_x[0::3], best_x[1::3]))
        curr_r = best_x[2::3].copy()
        for _ in range(20):
            curr_r = solve_lp_radii(curr_c)
            curr_c = opt_centers_lbfgs(curr_c, curr_r)
        s_alt = np.sum(solve_lp_radii(curr_c))
        if s_alt > best_sum:
            best_sum = s_alt
            best_x = np.zeros(3*N)
            best_x[0::3] = curr_c[:,0]
            best_x[1::3] = curr_c[:,1]
            best_x[2::3] = solve_lp_radii(curr_c)

    # Phase 4: Deflation & Perturbation to Escape Local Minima
    if best_x is not None:
        for cyc in range(80):
            x0 = best_x.copy()
            noise = 0.003 * (0.91 ** cyc)
            x0[0::3] += np.random.normal(0, noise, N)
            x0[1::3] += np.random.normal(0, noise, N)
            sub = np.random.choice(N, size=max(3, N//3), replace=False)
            x0[sub*3+2] *= 0.85
            x0 = project(x0)
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 15000, 'ftol': 1e-14})
                if not np.isnan(res.fun):
                    c_p = np.column_stack((res.x[0::3], res.x[1::3]))
                    r_p = solve_lp_radii(c_p)
                    x_p = res.x.copy()
                    x_p[2::3] = r_p
                    x_p = project(x_p)
                    s_val = np.sum(r_p)
                    if s_val > best_sum and np.min(constraints(x_p)) >= -1e-5:
                        best_sum = s_val
                        best_x = x_p.copy()
            except Exception:
                pass

    # Phase 5: Rigid Rotation of Subsets
    if best_x is not None:
        curr_c = np.column_stack((best_x[0::3], best_x[1::3]))
        for _ in range(60):
            k = np.random.randint(3, 10)
            idx = np.random.choice(N, k, replace=False)
            pivot = curr_c[idx].mean(axis=0)
            ang = np.random.uniform(-0.15, 0.15)
            c, s = np.cos(ang), np.sin(ang)
            new_c = curr_c.copy()
            vec = curr_c[idx] - pivot
            new_c[idx] = pivot + np.column_stack((vec[:,0]*c - vec[:,1]*s, vec[:,0]*s + vec[:,1]*c))
            new_c = np.clip(new_c, 0.01, 0.99)
            rn = solve_lp_radii(new_c)
            sn = np.sum(rn)
            if sn > best_sum:
                best_sum = sn
                best_x = np.zeros(3*N)
                best_x[0::3] = new_c[:,0]
                best_x[1::3] = new_c[:,1]
                best_x[2::3] = rn
                curr_c = new_c

    # Phase 6: Coordinate Descent Fine-Tuning
    if best_x is not None:
        curr_c = np.column_stack((best_x[0::3], best_x[1::3]))
        improved = True
        while improved:
            improved = False
            for i in range(N):
                base_s = best_sum
                for dx in [-0.005, -0.002, -0.001, 0.001, 0.002, 0.005]:
                    for dy in [-0.005, -0.002, -0.001, 0.001, 0.002, 0.005]:
                        if dx==0 and dy==0: continue
                        nc = curr_c.copy()
                        nc[i,0] = np.clip(curr_c[i,0]+dx, 0.01, 0.99)
                        nc[i,1] = np.clip(curr_c[i,1]+dy, 0.01, 0.99)
                        rn = solve_lp_radii(nc)
                        sn = np.sum(rn)
                        if sn > base_s + 1e-7:
                            curr_c = nc
                            best_x[0::3] = curr_c[:,0]
                            best_x[1::3] = curr_c[:,1]
                            best_x[2::3] = rn
                            best_sum = sn
                            improved = True
                            break
                    if improved: break

    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3].copy()

    # Final strict validation and minimal repair
    for _ in range(100):
        valid = True
        for i in range(N):
            if radii[i] < 0 or centers[i,0] < radii[i]-1e-9 or centers[i,0] > 1.0-radii[i]+1e-9 or \
               centers[i,1] < radii[i]-1e-9 or centers[i,1] > 1.0-radii[i]+1e-9:
                valid = False; break
        if valid:
            for i in range(N):
                for j in range(i+1, N):
                    d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
                    if d < radii[i]+radii[j]-1e-9: valid=False; break
                if not valid: break
        if valid: break
            
        # Minimal shrinkage to recover strict feasibility
        radii *= 0.9999
        centers[:,0] = np.clip(centers[:,0], radii, 1.0-radii)
        centers[:,1] = np.clip(centers[:,1], radii, 1.0-radii)

    return centers, radii, float(np.sum(radii))
