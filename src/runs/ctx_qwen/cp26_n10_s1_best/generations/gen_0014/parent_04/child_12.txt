2. **Diverse Initialization**: Hexagonal lattices with varied rotations, translations, and force-directed spreads provide strong starting points. Optimal packings often resemble slightly distorted hexagonal patterns aligned with the square boundaries.
3. **SLSQP Joint Optimization**: Using squared distance constraints `dx^2 + dy^2 >= (r_i + r_j)^2` provides smooth gradients near contact points, allowing SLSQP to efficiently push circles apart and expand radii simultaneously.

import numpy as np
from scipy.optimize import minimize, linprog
import warnings

warnings.filterwarnings('ignore')
np.seterr(all='ignore')

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    m = n + n * (n - 1) // 2
    A = np.zeros((m, n))
    b = np.zeros(m)
    idx = 0
    for i in range(n):
        x, y = centers[i]
        lim = min(x, 1.0 - x, y, 1.0 - y)
        A[idx, i] = 1.0
        b[idx] = max(0.0, lim)
        idx += 1
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = max(0.0, d)
            idx += 1
    try:
        res = linprog(-np.ones(n), A_ub=A, b_ub=b, bounds=[(0.0, None)] * n, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-9)
    except Exception:
        pass
    return np.full(n, 1e-4)

def get_lp_sum(centers):
    """Returns the sum of optimally computed radii for given centers."""
    return np.sum(solve_lp_radii(centers))

def constraints_joint(x):
    """Inequality constraints for joint SLSQP optimization: g(x) >= 0."""
    xs, ys, rs = x[0::3], x[1::3], x[2::3]
    c = np.concatenate([
        xs - rs, 1.0 - xs - rs,
        ys - rs, 1.0 - ys - rs
    ])
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    c = np.concatenate([c, dx[TRIL_IDX]**2 + dy[TRIL_IDX]**2 - dr[TRIL_IDX]**2])
    return c

def objective_joint(x):
    """Objective for joint SLSQP optimization: minimize negative sum of radii."""
    return -np.sum(x[2::3])

def make_hex_init(r0, angle, shift_x=0.0, shift_y=0.0):
    """Generates a rotated hexagonal lattice initialization."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0 and len(pts) < N + 10:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
    pts = np.array(pts[:N + 10])
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
    pts += [shift_x, shift_y]
    mask = (pts[:, 0] >= 0.001) & (pts[:, 0] <= 0.999) & (pts[:, 1] >= 0.001) & (pts[:, 1] <= 0.999)
    pts = pts[mask]
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.1, 0.9, (1, 2))])
    return pts[:N]

def make_force_init(seed):
    """Force-directed layout to spread points evenly and push to boundaries."""
    np.random.seed(seed)
    pts = np.random.uniform(0.15, 0.85, (N, 2))
    for _ in range(150):
        f = np.zeros_like(pts)
        for i in range(N):
            for j in range(i + 1, N):
                dx = pts[j] - pts[i]
                d = np.hypot(dx[0], dx[1])
                if d < 0.25 and d > 1e-5:
                    rep = 0.01 / (d**2 + 0.001)
                    f[i] -= dx * rep / d
                    f[j] += dx * rep / d
            for dim in range(2):
                if pts[i, dim] < 0.1: f[i, dim] += 0.05
                elif pts[i, dim] > 0.9: f[i, dim] -= 0.05
        pts += f * 0.02
        pts = np.clip(pts, 0.02, 0.98)
    return pts

def project_to_feasible(x):
    """Project variables to strictly satisfy bounds."""
    x = x.copy()
    for i in range(N):
        r = max(1e-6, x[3 * i + 2])
        x[3 * i] = np.clip(x[3 * i], r, 1.0 - r)
        x[3 * i + 1] = np.clip(x[3 * i + 1], r, 1.0 - r)
        x[3 * i + 2] = r
    return x

def run_packing():
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    bounds_obj = [(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints_joint}

    best_sum = -1.0
    best_x = None

    # Phase 1: Diverse Initial Configurations
    inits = []
    # Hexagonal lattices with varied rotations and scales
    for ang in np.linspace(-0.4, 0.4, 11):
        inits.append(make_hex_init(0.090, ang))
        inits.append(make_hex_init(0.095, ang))
    # Shifted hex lattices
    for sx in np.linspace(-0.04, 0.04, 5):
        for sy in np.linspace(-0.04, 0.04, 5):
            inits.append(make_hex_init(0.092, 0.0, sx, sy))
    # Force-directed spreads
    for s in range(6):
        inits.append(make_force_init(s))

    # Phase 2: Multi-start SLSQP Optimization
    for c_init in inits:
        x0 = np.zeros(3 * N)
        x0[0::3] = c_init[:, 0]
        x0[1::3] = c_init[:, 1]
        r_lp = solve_lp_radii(c_init)
        x0[2::3] = np.maximum(r_lp * 0.995, 1e-6)
        x0 = project_to_feasible(x0)

        try:
            res = minimize(objective_joint, x0, method='SLSQP', bounds=bounds_obj, constraints=cons,
                           options={'maxiter': 12000, 'ftol': 1e-13, 'disp': False})
            if not np.isnan(res.fun):
                c_tmp = np.column_stack((res.x[0::3], res.x[1::3]))
                r_new = solve_lp_radii(c_tmp)
                curr_sum = np.sum(r_new)
                if curr_sum > best_sum:
                    best_sum = curr_sum
                    best_x = res.x.copy()
                    best_x[2::3] = r_new
        except Exception:
            continue

    # Phase 3: Deflation & Perturbation Loop to Escape Local Minima
    if best_x is not None:
        for cyc in range(50):
            x0 = best_x.copy()
            noise = 0.002 * (0.92 ** cyc)
            x0[0::3] += np.random.normal(0, noise, N)
            x0[1::3] += np.random.normal(0, noise, N)
            
            # Deflate a random subset to break rigid contact networks
            k = np.random.randint(2, 6)
            subset = np.random.choice(N, size=k, replace=False)
            x0[subset * 3 + 2] *= 0.82
            
            x0 = project_to_feasible(x0)

            try:
                res = minimize(objective_joint, x0, method='SLSQP', bounds=bounds_obj, constraints=cons,
                               options={'maxiter': 10000, 'ftol': 1e-13, 'disp': False})
                if not np.isnan(res.fun):
                    c_tmp = np.column_stack((res.x[0::3], res.x[1::3]))
                    r_new = solve_lp_radii(c_tmp)
                    if np.sum(r_new) > best_sum:
                        best_sum = np.sum(r_new)
                        best_x = res.x.copy()
                        best_x[2::3] = r_new
            except Exception:
                pass

    # Phase 4: Coordinate Ascent & Pairwise Refinement
    if best_x is not None:
        curr_c = np.column_stack((best_x[0::3], best_x[1::3]))
        
        # Single-circle grid ascent
        for _ in range(3):
            improved = True
            while improved:
                improved = False
                for i in range(N):
                    best_c = curr_c[i].copy()
                    best_s = get_lp_sum(curr_c)
                    for dx in [-0.003, 0.003]:
                        for dy in [-0.003, 0.003]:
                            tc = curr_c.copy()
                            tc[i] = np.clip(curr_c[i] + [dx, dy], 0.001, 0.999)
                            ts = get_lp_sum(tc)
                            if ts > best_s + 1e-8:
                                best_s = ts
                                best_c = tc[i].copy()
                    if not np.array_equal(best_c, curr_c[i]):
                        curr_c[i] = best_c
                        improved = True
                curr_r = solve_lp_radii(curr_c)
                curr_sum = np.sum(curr_r)
                if curr_sum > best_sum:
                    best_sum = curr_sum
                    best_x[0::3] = curr_c[:, 0]
                    best_x[1::3] = curr_c[:, 1]
                    best_x[2::3] = curr_r

        # Pairwise grid ascent
        for _ in range(2):
            improved = True
            while improved:
                improved = False
                for i in range(N):
                    for j in range(i + 1, N):
                        best_s = get_lp_sum(curr_c)
                        for dxi in [-0.002, 0.002]:
                            for dyi in [-0.002, 0.002]:
                                for dxj in [-0.002, 0.002]:
                                    for dyj in [-0.002, 0.002]:
                                        tc = curr_c.copy()
                                        tc[i] = np.clip(curr_c[i] + [dxi, dyi], 0.001, 0.999)
                                        tc[j] = np.clip(curr_c[j] + [dxj, dyj], 0.001, 0.999)
                                        ts = get_lp_sum(tc)
                                        if ts > best_s + 1e-8:
                                            best_s = ts
                                            curr_c[i] = tc[i].copy()
                                            curr_c[j] = tc[j].copy()
                                            improved = True
                curr_r = solve_lp_radii(curr_c)
                curr_sum = np.sum(curr_r)
                if curr_sum > best_sum:
                    best_sum = curr_sum
                    best_x[0::3] = curr_c[:, 0]
                    best_x[1::3] = curr_c[:, 1]
                    best_x[2::3] = curr_r

    # Phase 5: Final High-Precision SLSQP Polish
    if best_x is not None:
        x0 = best_x.copy()
        try:
            res = minimize(objective_joint, x0, method='SLSQP', bounds=bounds_obj, constraints=cons,
                           options={'maxiter': 20000, 'ftol': 1e-13, 'disp': False})
            if not np.isnan(res.fun):
                c_tmp = np.column_stack((res.x[0::3], res.x[1::3]))
                r_new = solve_lp_radii(c_tmp)
                if np.sum(r_new) > best_sum:
                    best_sum = np.sum(r_new)
                    best_x = res.x.copy()
                    best_x[2::3] = r_new
        except Exception:
            pass

    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3].copy()

    # Final strict validity adjustment against 1e-12 tolerance
    for _ in range(100):
        valid = True
        for i in range(N):
            if radii[i] < 0 or centers[i, 0] < radii[i] - 1e-9 or centers[i, 0] > 1.0 - radii[i] + 1e-9 or \
               centers[i, 1] < radii[i] - 1e-9 or centers[i, 1] > 1.0 - radii[i] + 1e-9:

                break
        if valid:
            for i in range(N):
                for j in range(i + 1, N):
                    if np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1]) < radii[i] + radii[j] - 1e-9:
                        valid = False
                        break
                if not valid:
                    break
        if valid:
            break
        radii *= 0.9999
        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)

    return centers, radii, float(np.sum(radii))
