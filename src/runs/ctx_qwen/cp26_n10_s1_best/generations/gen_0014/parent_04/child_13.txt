import numpy as np
from scipy.optimize import minimize, linprog
import warnings

warnings.filterwarnings('ignore')

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    m = n + n * (n - 1) // 2
    A = np.zeros((m, n))
    b = np.zeros(m)
    idx = 0
    
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A[idx, i] = 1.0
        b[idx] = max(1e-9, lim)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = max(1e-9, d)
            idx += 1
            
    try:
        res = linprog(-np.ones(n), A_ub=A, b_ub=b, bounds=[(0.0, None)] * n, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-9)
    except Exception:
        pass
    return np.full(n, 1e-4)

def constraints_slsqp(x):
    """Inequality constraints for SLSQP: g(x) >= 0."""
    xs, ys, rs = x[0::3], x[1::3], x[2::3]
    c = np.concatenate([
        xs - rs, 1.0 - xs - rs,
        ys - rs, 1.0 - ys - rs
    ])
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    c = np.concatenate([c, np.hypot(dx[TRIL_IDX], dy[TRIL_IDX]) - dr[TRIL_IDX]])
    return c

def objective_slsqp(x):
    """Objective: maximize sum of radii."""
    return -np.sum(x[2::3])

def init_hex(angle=0.0, scale=1.0):
    """Generates a rotated hexagonal lattice initialization."""
    pts = []
    r0 = 0.095 * scale
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0 and len(pts) < N + 10:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        
    pts = np.array(pts[:N + 10])
    if angle != 0.0:
        ca, sa = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[ca, -sa], [sa, ca]]) + 0.5
        
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.1, 0.9, (1, 2))])
    return pts[:N]

def init_force(seed):
    """Force-directed layout to spread points evenly and push to boundaries."""
    np.random.seed(seed)
    pts = np.random.uniform(0.15, 0.85, (N, 2))
    for _ in range(200):
        f = np.zeros_like(pts)
        for i in range(N):
            for j in range(i + 1, N):
                dx = pts[j] - pts[i]
                d = np.hypot(dx[0], dx[1])
                if d < 0.25 and d > 1e-5:
                    rep = 0.012 / (d**2 + 0.001)
                    f[i] -= dx * rep / d
                    f[j] += dx * rep / d
            for dim in range(2):
                if pts[i, dim] < 0.12: f[i, dim] += 0.05
                elif pts[i, dim] > 0.88: f[i, dim] -= 0.05
        pts += f * 0.04
        pts = np.clip(pts, 0.02, 0.98)
    return pts

def init_corner():
    """Initialization focused on corners and edges."""
    pts = np.zeros((N, 2))
    r = 0.095
    pts[0] = [r, r]; pts[1] = [1 - r, r]; pts[2] = [r, 1 - r]; pts[3] = [1 - r, 1 - r]
    pts[4] = [0.5, r]; pts[5] = [0.5, 1 - r]; pts[6] = [r, 0.5]; pts[7] = [1 - r, 0.5]; pts[8] = [0.5, 0.5]
    idx = 9
    y = r + 0.04
    row = 0
    while idx < N and y + r <= 1.0:
        x = r + 0.04 if row % 2 == 0 else r + 0.15
        while idx < N and x + r <= 1.0:
            pts[idx] = [x, y]
            idx += 1
            x += 0.22
        y += 0.14
        row += 1
    while idx < N:
        pts[idx] = np.random.uniform(0.1, 0.9, 2)
        idx += 1
    return pts + np.random.uniform(-0.005, 0.005, (N, 2))

def basin_hopping_search(c0, steps=1200):
    """Simulated annealing on centers using LP radii evaluation."""
    c = c0.copy()
    r = solve_lp_radii(c)
    best_c, best_r, best_sum = c.copy(), r.copy(), np.sum(r)
    T = 0.035
    
    for _ in range(steps):
        cn = c.copy()
        choice = np.random.rand()
        if choice < 0.6:
            idx = np.random.choice(N, np.random.randint(3, 10), replace=False)
            cn[idx] += np.random.normal(0, 0.006 * (T / 0.035), (len(idx), 2))
        elif choice < 0.8:
            ang = np.random.uniform(-0.04, 0.04)
            ca, sa = np.cos(ang), np.sin(ang)
            cn = (cn - 0.5) @ np.array([[ca, -sa], [sa, ca]]) + 0.5
        else:
            s = np.random.uniform(0.96, 1.04)
            cn = (cn - 0.5) * s + 0.5
            
        cn = np.clip(cn, 0.005, 0.995)
        rn = solve_lp_radii(cn)
        sn = np.sum(rn)
        
        if sn > best_sum or (T > 1e-5 and np.random.rand() < np.exp((sn - best_sum) / T)):
            c, r = cn, rn
            if sn > best_sum:
                best_c, best_r, best_sum = cn.copy(), rn.copy(), sn
        T *= 0.993
        
    return best_c, best_r, best_sum

def slsqp_polish(c, r):
    """Joint SLSQP optimization for centers and radii."""
    x0 = np.zeros(3 * N)
    x0[0::3] = c[:, 0]
    x0[1::3] = c[:, 1]
    x0[2::3] = r
    
    # Project to strictly feasible bounds
    for i in range(N):
        ri = max(1e-6, x0[3 * i + 2])
        x0[3 * i] = np.clip(x0[3 * i], ri, 1.0 - ri)
        x0[3 * i + 1] = np.clip(x0[3 * i + 1], ri, 1.0 - ri)
        x0[3 * i + 2] = ri
        
    bounds = [(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints_slsqp}
    
    try:
        res = minimize(objective_slsqp, x0, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 12000, 'ftol': 1e-14, 'disp': False})
        if not np.isnan(res.fun):
            return res.x
    except Exception:
        pass
    return x0
