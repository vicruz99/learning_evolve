import numpy as np
from scipy.optimize import minimize, linprog

N = 26
TRIL_IDX = np.tril_indices(N, -1)
TRIU_IDX = np.triu_indices(N, 1)

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    m = n + n * (n - 1) // 2
    A = np.zeros((m, n))
    b = np.zeros(m)
    
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    edges = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]), 
                       np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    A[:n, np.arange(n)] = 1.0
    b[:n] = np.maximum(edges, 0.0)
    
    # Pairwise distance constraints: r_i + r_j <= dist(i, j)
    dx = centers[:, 0, None] - centers[:, 0]
    dy = centers[:, 1, None] - centers[:, 1]
    dists = np.hypot(dx, dy)
    dists_triu = dists[TRIU_IDX]
    
    row_offset = n
    for i, (r, c) in enumerate(zip(TRIU_IDX[0], TRIU_IDX[1])):
        A[row_offset + i, r] = 1.0
        A[row_offset + i, c] = 1.0
        b[row_offset + i] = max(0.0, dists_triu[i])
        
    try:
        res = linprog(-np.ones(n), A_ub=A, b_ub=b, bounds=[(0.0, None)] * n, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-9)
    except Exception:
        pass
    return np.full(n, 1e-4)

def get_lp_sum(centers):
    """Returns the sum of optimally computed radii for given centers."""
    return np.sum(solve_lp_radii(centers))

def project_to_feasible(x):
    """Project variables to strictly satisfy bounds."""
    x = x.copy()
    for i in range(N):
        r = max(1e-7, x[3 * i + 2])
        x[3 * i] = np.clip(x[3 * i], r, 1.0 - r)
        x[3 * i + 1] = np.clip(x[3 * i + 1], r, 1.0 - r)
        x[3 * i + 2] = r
    return x

def constraints_slsqp(x):
    """Inequality constraints for joint SLSQP optimization: g(x) >= 0."""
    xs, ys, rs = x[0::3], x[1::3], x[2::3]
    c = np.concatenate([xs - rs, 1.0 - xs - rs, ys - rs, 1.0 - ys - rs])
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    c = np.concatenate([c, np.hypot(dx[TRIL_IDX], dy[TRIL_IDX]) - dr[TRIL_IDX]])
    return c

def objective_slsqp(x):
    """Objective for joint SLSQP optimization: maximize sum of radii."""
    return -np.sum(x[2::3])

def make_hex_init(r0, angle):
    """Generates a rotated hexagonal lattice initialization."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0 and len(pts) < N + 10:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
    pts = np.array(pts[:N + 10])
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.1, 0.9, (1, 2))])
    return pts[:N]

def make_corner_init():
    """Initialization focused on corners and edges."""
    pts = np.zeros((N, 2))
    r = 0.095
    pts[0] = [r, r]
    pts[1] = [1 - r, r]
    pts[2] = [r, 1 - r]
    pts[3] = [1 - r, 1 - r]
    pts[4] = [0.5, r]
    pts[5] = [0.5, 1 - r]
    pts[6] = [r, 0.5]
    pts[7] = [1 - r, 0.5]
    idx = 8
    y = r + 0.04
    row = 0
    while idx < N and y + r <= 1.0:
        x = r + 0.04 if row % 2 == 0 else r + 0.15
        while idx < N and x + r <= 1.0:
            pts[idx] = [x, y]
            idx += 1
            x += 0.22
        y += 0.14
        row += 1
    while idx < N:
        pts[idx] = np.random.uniform(0.1, 0.9, 2)
        idx += 1
    return pts + np.random.uniform(-0.005, 0.005, (N, 2))

def sa_search(c0, steps=3000, T0=0.05):
    """Simulated Annealing on centers to maximize LP sum of radii."""
    best_c = c0.copy()
    best_sum = get_lp_sum(c0)
    curr_c = c0.copy()
    curr_sum = best_sum
    T = T0
    
    for _ in range(steps):
        r = np.random.rand()
        if r < 0.6:
            idx = np.random.choice(N, np.random.randint(1, 6), replace=False)
            curr_c[idx] += np.random.normal(0, 0.006, (len(idx), 2))
        elif r < 0.85:
            ang = np.random.uniform(-0.03, 0.03)
            c, s = np.cos(ang), np.sin(ang)
            curr_c = (curr_c - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
        else:
            sc = np.random.uniform(0.97, 1.03)
            curr_c = (curr_c - 0.5) * sc + 0.5
            
        curr_c = np.clip(curr_c, 0.005, 0.995)
        
        new_sum = get_lp_sum(curr_c)
        
        if new_sum > curr_sum or (T > 1e-6 and np.random.rand() < np.exp((new_sum - curr_sum) / T)):
            curr_sum = new_sum
            if new_sum > best_sum:
                best_sum = new_sum
                best_c = curr_c.copy()
        T *= 0.994
    return best_c, best_sum

def coord_descent(centers, iters=2):
    """Coordinate descent refinement: optimize one center at a time."""
    c = centers.copy()
    for _ in range(iters):
        for i in range(N):
            best_c_i = c[i].copy()
            best_sum_i = get_lp_sum(c)
            for dx in [-0.002, 0.002]:
                for dy in [-0.002, 0.002]:
                    nc = c.copy()
                    nc[i] = np.clip(c[i] + [dx, dy], 0.001, 0.999)
                    ns = get_lp_sum(nc)
                    if ns > best_sum_i:
                        best_sum_i = ns
                        best_c_i = nc[i].copy()
            c[i] = best_c_i
    return c
