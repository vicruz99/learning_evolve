import numpy as np
from scipy.optimize import minimize, linprog
import warnings

warnings.filterwarnings('ignore')

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A = np.zeros((num_ineq, n))
    b = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        x, y = centers[i]
        lim = min(x, 1.0 - x, y, 1.0 - y)
        A[idx, i] = 1.0
        b[idx] = max(0.0, lim)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = max(0.0, d)
            idx += 1
            
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0.0, None)] * n, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-9)
    except Exception:
        pass
    return np.full(n, 1e-4)

def get_lp_sum(centers):
    """Returns the sum of optimally computed radii for given centers."""
    return np.sum(solve_lp_radii(centers))

def center_penalty(c_flat, radii):
    """Smooth penalty function for centers given fixed radii."""
    c = c_flat.reshape(-1, 2)
    p = 0.0
    x, y = c[:, 0], c[:, 1]
    r = radii
    
    # Boundary penalties
    p += 200.0 * np.sum(np.maximum(r - x, 0.0)**2)
    p += 200.0 * np.sum(np.maximum(x - (1.0 - r), 0.0)**2)
    p += 200.0 * np.sum(np.maximum(r - y, 0.0)**2)
    p += 200.0 * np.sum(np.maximum(y - (1.0 - r), 0.0)**2)
    
    # Overlap penalties
    dx = c[:, 0, None] - c[:, 0]
    dy = c[:, 1, None] - c[:, 1]
    d = np.hypot(dx, dy)
    gap = d - r[:, None] - r[None, :]
    p += 500.0 * np.sum(np.maximum(-gap[TRIL_IDX], 0.0)**2)
    return p

def opt_centers_lbfgs(centers, radii):
    """Optimize centers for fixed radii using L-BFGS-B."""
    bounds = [(0.001, 0.999)] * (2 * N)
    res = minimize(center_penalty, centers.flatten(), args=(radii,),
                   method='L-BFGS-B', bounds=bounds, options={'maxiter': 2000, 'ftol': 1e-14})
    return res.x.reshape(-1, 2)

def constraints_joint(x):
    """Inequality constraints for joint SLSQP optimization."""
    xs, ys, rs = x[0::3], x[1::3], x[2::3]
    c = np.concatenate([
        xs - rs, 1.0 - xs - rs,
        ys - rs, 1.0 - ys - rs
    ])
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    # Squared distance constraints for smooth gradients
    c = np.concatenate([c, dx[TRIL_IDX]**2 + dy[TRIL_IDX]**2 - dr[TRIL_IDX]**2])
    return c

def objective_joint(x):
    """Objective for joint SLSQP optimization."""
    return -np.sum(x[2::3])

def make_hex_init(r0, angle, scale=1.0):
    """Generates a rotated hexagonal lattice initialization."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0 and len(pts) < N + 10:
            pts.append([x * scale, y * scale])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        
    pts = np.array(pts[:N + 10])
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
        
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.1, 0.9, (1, 2))])
    return pts[:N]

def make_force_init(seed):
    """Force-directed layout to spread points evenly."""
    np.random.seed(seed)
    pts = np.random.uniform(0.15, 0.85, (N, 2))
    for step in range(200):
        f = np.zeros_like(pts)
        lr = 0.02 * (1.0 - step / 200.0)
        for i in range(N):
            for j in range(i + 1, N):
                dx = pts[j] - pts[i]
                d = np.hypot(dx[0], dx[1])
                if d < 0.3 and d > 1e-5:
                    rep = 0.015 / (d**2 + 0.001)
                    f[i] -= dx * rep / d
                    f[j] += dx * rep / d
            for dim in range(2):
                if pts[i, dim] < 0.1: f[i, dim] += 0.05
                elif pts[i, dim] > 0.9: f[i, dim] -= 0.05
        pts += f * lr
        pts = np.clip(pts, 0.02, 0.98)
    return pts

def make_corner_init():
    """Initialization focused on corners and edges."""
    pts = np.zeros((N, 2))
    r = 0.095
    pts[0] = [r, r]
    pts[1] = [1 - r, r]
    pts[2] = [r, 1 - r]
    pts[3] = [1 - r, 1 - r]
    pts[4] = [0.5, r]
    pts[5] = [0.5, 1 - r]

    while idx < N and y + r <= 1.0:
        x = r + 0.04 if row % 2 == 0 else r + 0.15

            pts[idx] = [x, y]
            idx += 1
            x += 0.22
        y += 0.14
        row += 1
    while idx < N:
        pts[idx] = np.random.uniform(0.1, 0.9, 2)
        idx += 1
    return pts + np.random.uniform(-0.005, 0.005, (N, 2))

def project_to_feasible(x):
    """Project variables to strictly satisfy bounds."""
    x = x.copy()
    for i in range(N):
        r = max(1e-6, x[3 * i + 2])
        x[3 * i] = np.clip(x[3 * i], r, 1.0 - r)
        x[3 * i + 1] = np.clip(x[3 * i + 1], r, 1.0 - r)
        x[3 * i + 2] = r
    return x

def opt_single_center_nelder_mead(centers, idx):
    """Optimize one circle position to maximize LP sum using Nelder-Mead."""
    def neg_lp_sum(p):
        tmp = centers.copy()
        tmp[idx] = p
        return -get_lp_sum(tmp)
        
    res = minimize(neg_lp_sum, centers[idx], 
                   method='Nelder-Mead', 
                   options={'xatol': 1e-6, 'fatol': 1e-7, 'maxiter': 200})
    return np.clip(res.x, 0.0, 1.0)
