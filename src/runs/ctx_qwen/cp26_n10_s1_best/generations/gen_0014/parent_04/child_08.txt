import numpy as np
from scipy.optimize import minimize, linprog
import warnings
warnings.filterwarnings('ignore')

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    m = n + n * (n - 1) // 2
    A = np.zeros((m, n))
    b = np.zeros(m)
    idx = 0
    for i in range(n):
        x, y = centers[i]
        lim = min(x, 1.0 - x, y, 1.0 - y)
        A[idx, i] = 1.0
        b[idx] = max(0.0, lim)
        idx += 1
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = max(0.0, d)
            idx += 1
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0.0, None)] * n, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-9)
    except Exception:
        pass
    return np.full(n, 1e-4)

def get_lp_sum_and_radii(centers):
    """Returns the sum of optimally computed radii and the radii array."""
    r = solve_lp_radii(centers)
    return np.sum(r), r

def constraints_joint(x):
    """Inequality constraints for joint SLSQP optimization."""
    xs, ys, rs = x[0::3], x[1::3], x[2::3]
    c = np.concatenate([xs - rs, 1.0 - xs - rs, ys - rs, 1.0 - ys - rs])
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    c = np.concatenate([c, dx[TRIL_IDX]**2 + dy[TRIL_IDX]**2 - dr[TRIL_IDX]**2])
    return c

def objective_joint(x):
    """Objective for joint SLSQP optimization."""
    return -np.sum(x[2::3])

def project_to_feasible(x):
    """Project variables to strictly satisfy bounds."""
    x = x.copy()
    for i in range(N):
        r = max(1e-6, x[3 * i + 2])
        x[3 * i] = np.clip(x[3 * i], r, 1.0 - r)
        x[3 * i + 1] = np.clip(x[3 * i + 1], r, 1.0 - r)
        x[3 * i + 2] = r
    return x

def make_hex_init(r0, angle, shift=np.array([0.0, 0.0])):
    """Generates a rotated hexagonal lattice initialization."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0 and len(pts) < N + 10:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
    pts = np.array(pts[:N + 10])
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
    pts += shift
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.1, 0.9, (1, 2))])
    return pts[:N]

def make_force_init(seed):
    """Force-directed layout to spread points evenly."""
    np.random.seed(seed)
    pts = np.random.uniform(0.15, 0.85, (N, 2))
    for _ in range(200):
        f = np.zeros_like(pts)
        for i in range(N):
            for j in range(i + 1, N):
                dx = pts[j] - pts[i]
                d = np.hypot(dx[0], dx[1])
                if d < 0.25 and d > 1e-5:
                    rep = 0.012 / (d**2 + 0.001)
                    f[i] -= dx * rep / d
                    f[j] += dx * rep / d
            for dim in range(2):
                if pts[i, dim] < 0.12: f[i, dim] += 0.05
                elif pts[i, dim] > 0.88: f[i, dim] -= 0.05
        pts += f * 0.04
        pts = np.clip(pts, 0.03, 0.97)
    return pts

def run_packing():
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    bounds_obj = [(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints_joint}
    
    best_sum = -1.0
    best_c = None
    best_r = None

    # Phase 1: Generate diverse initial configurations
    inits = []
    for ang in np.linspace(-0.3, 0.3, 11):
        for sx in np.linspace(-0.05, 0.05, 3):
            for sy in np.linspace(-0.05, 0.05, 3):
                inits.append(make_hex_init(0.092, ang, np.array([sx, sy])))
    for s in range(8):
        inits.append(make_force_init(s))
    for s in range(10):
        np.random.seed(s + 1000)
        inits.append(np.random.uniform(0.1, 0.9, (N, 2)))

    # Phase 2: Basin Hopping / Simulated Annealing on centers
    for c_init in inits:
        c = c_init.copy()
        s, r = get_lp_sum_and_radii(c)
        
        best_local_sum = s
        best_local_c = c.copy()
        best_local_r = r.copy()
        
        T = 0.05
        for step in range(900):
            choice = np.random.rand()
            c_new = c.copy()
            if choice < 0.6:
                k = np.random.randint(3, 8)
                idx = np.random.choice(N, k, replace=False)
                c_new[idx] += np.random.normal(0, 0.004, (k, 2))
            elif choice < 0.8:
                ang = np.random.uniform(-0.03, 0.03)
                ca, sa = np.cos(ang), np.sin(ang)
                c_new = (c - 0.5) @ np.array([[ca, -sa], [sa, ca]]) + 0.5
            else:
                sc = np.random.uniform(0.97, 1.03)
                c_new = (c - 0.5) * sc + 0.5
                c_new += np.random.uniform(-0.01, 0.01, c.shape)
                
            c_new = np.clip(c_new, 0.005, 0.995)
            s_new, r_new = get_lp_sum_and_radii(c_new)
            
            delta = s_new - best_local_sum
            if delta > 0 or (T > 1e-5 and np.random.rand() < np.exp(delta / T)):
                c, s = c_new, s_new
                if s > best_local_sum:
                    best_local_sum = s
                    best_local_c = c.copy()
                    best_local_r = r_new.copy()
            T *= 0.995
            
        if best_local_sum > best_sum:
            best_sum = best_local_sum
            best_c = best_local_c
            best_r = best_local_r

    # Phase 3: Coordinate Descent Refinement
    if best_c is not None:

            for i in range(N):

                s_best_i = get_lp_sum_and_radii(c_curr)[0]

                            continue
                        c_try = c_curr.copy()
                        c_try[i] = np.clip(c_try[i] + [dx, dy], 0.0, 1.0)
                        s_try, _ = get_lp_sum_and_radii(c_try)
                        if s_try > s_best_i + 1e-8:
                            s_best_i = s_try
                            c_best_i = c_try[i].copy()
                c_curr[i] = c_best_i
                if not np.array_equal(c_curr[i], best_c[i]):
                    improved_any = True
                    
            s_new, r_new = get_lp_sum_and_radii(c_curr)
            if s_new > best_sum:
                best_sum = s_new
                best_c = c_curr.copy()
                best_r = r_new.copy()
            if not improved_any:
                break

    # Phase 4: Deflation & Relocation Strategy
    if best_c is not None:
        for cyc in range(25):
            c_def = best_c.copy()
            r_def = best_r.copy()
            k = np.random.randint(4, 9)
            idx_def = np.random.choice(N, k, replace=False)
            
            # Shrink selected circles significantly to break contacts
            r_def[idx_def] *= 0.65
            
            # Perturb their positions to allow reorganization
            c_def[idx_def] += np.random.normal(0, 0.015, (k, 2))
            c_def = np.clip(c_def, 0.02, 0.98)
            
            # Quick local optimization on deflated centers
            for _ in range(2):
                for i in range(N):
                    c_best_i = c_def[i].copy()
                    s_best_i = get_lp_sum_and_radii(c_def)[0]
                    for dx in np.linspace(-0.003, 0.003, 3):
                        for dy in np.linspace(-0.003, 0.003, 3):
                            if dx == 0.0 and dy == 0.0: continue
                            c_try = c_def.copy()
                            c_try[i] = np.clip(c_try[i] + [dx, dy], 0.0, 1.0)
                            s_try, _ = get_lp_sum_and_radii(c_try)
                            if s_try > s_best_i + 1e-8:
                                s_best_i = s_try
                                c_best_i = c_try[i].copy()
                    c_def[i] = c_best_i
                    
            s_new, r_new = get_lp_sum_and_radii(c_def)
            if s_new > best_sum:
                best_sum = s_new
                best_c = c_def.copy()
                best_r = r_new.copy()

    # Phase 5: SLSQP Joint Polish
    if best_c is not None:
        x0 = np.zeros(3 * N)
        x0[0::3] = best_c[:, 0]
        x0[1::3] = best_c[:, 1]
        x0[2::3] = best_r * 0.999
        x0 = project_to_feasible(x0)
        
        try:
            res = minimize(objective_joint, x0, method='SLSQP', bounds=bounds_obj, constraints=cons,
                           options={'maxiter': 25000, 'ftol': 1e-14, 'disp': False})
            if not np.isnan(res.fun):
                c_tmp = np.column_stack((res.x[0::3], res.x[1::3]))
                r_new = solve_lp_radii(c_tmp)
                if np.sum(r_new) > best_sum:
                    best_sum = np.sum(r_new)
                    best_c = c_tmp
                    best_r = r_new
        except Exception:
            pass

    centers = best_c
    radii = best_r

    # Final strict validity adjustment against 1e-12 tolerance
    for _ in range(100):
        valid = True
        for i in range(N):
            if radii[i] < 0 or centers[i, 0] < radii[i] - 1e-9 or centers[i, 0] > 1.0 - radii[i] + 1e-9 or \
               centers[i, 1] < radii[i] - 1e-9 or centers[i, 1] > 1.0 - radii[i] + 1e-9:

                break
        if valid:
            for i in range(N):
                for j in range(i + 1, N):
                    if np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1]) < radii[i] + radii[j] - 1e-9:
                        valid = False
                        break
                if not valid:
                    break
        if valid:
            break
        radii *= 0.9999
        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)

    return centers, radii, float(np.sum(radii))
