import numpy as np
from scipy.optimize import minimize, linprog
import warnings

warnings.filterwarnings('ignore')

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Returns all inequality constraints >= 0 (vectorized)."""
    xs, ys, rs = x[0::3], x[1::3], x[2::3]
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    c = np.concatenate([
        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs
    ])
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    c = np.concatenate([c, dx[TRIL_IDX]**2 + dy[TRIL_IDX]**2 - dr[TRIL_IDX]**2])
    return c

def get_bounds():
    """Variable bounds: x,y in [0.01, 0.99], r in [1e-6, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.01, 0.99), (0.01, 0.99), (1e-6, 0.5)])
    return b

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    centers = np.clip(centers, 0.005, 0.995)
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A = np.zeros((num_ineq, n))
    b = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A[idx, i] = 1.0
        b[idx] = max(0.0, lim)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = max(0.0, dist)
            idx += 1
            
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0.0, None)] * n, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-9)
    except Exception:
        pass
    return np.full(n, 0.05)

def project_to_feasible(x):
    """Project variables to strictly satisfy bounds."""
    x = x.copy()
    for i in range(N):
        r = max(1e-6, x[3 * i + 2])
        x[3 * i] = np.clip(x[3 * i], r + 1e-7, 1.0 - r - 1e-7)
        x[3 * i + 1] = np.clip(x[3 * i + 1], r + 1e-7, 1.0 - r - 1e-7)
        x[3 * i + 2] = r
    return x

def make_hex_init(r0, angle, scale=1.0):
    """Generates a rotated/scaled hexagonal lattice initialization."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 5:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0 and len(pts) < N + 5:
            pts.append([x * scale, y * scale])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        
    pts = np.array(pts[:N + 5])
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
        
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.1, 0.9, (1, 2))])
    return pts[:N]

def make_force_init(seed):
    """Force-directed layout to spread points evenly and push to boundaries."""
    np.random.seed(seed)
    pts = np.random.uniform(0.15, 0.85, (N, 2))
    for step in range(150):
        f = np.zeros_like(pts)
        lr = 0.01 * (1.0 - step / 150.0)
        for i in range(N):
            for j in range(i + 1, N):
                dx = pts[j] - pts[i]
                d = np.hypot(dx[0], dx[1])
                if d < 0.25 and d > 1e-5:
                    rep = 0.008 / (d**2 + 0.001)
                    f[i] -= dx * rep / d
                    f[j] += dx * rep / d
            for dim in range(2):
                if pts[i, dim] < 0.1: f[i, dim] += 0.03
                elif pts[i, dim] > 0.9: f[i, dim] -= 0.03
        pts += f * lr
        pts = np.clip(pts, 0.02, 0.98)
    return pts

def simulated_annealing(c0, steps=5000, T0=0.04):
    """Simulated annealing on centers with LP radii evaluation."""
    c = c0.copy()
    r = solve_lp_radii(c)
    best_sum = np.sum(r)
    best_c = c.copy()
    T = T0
    
    for step in range(steps):
        c_new = c.copy()
        # Perturb a random subset of centers
        k = np.random.randint(1, 5)
        idx = np.random.choice(N, k, replace=False)
        c_new[idx] += np.random.normal(0, 0.008, (k, 2))
        
        # Occasional global rotation to escape rotational minima
        if np.random.rand() < 0.02:
            ang = np.random.uniform(-0.03, 0.03)
            ca, sa = np.cos(ang), np.sin(ang)
            c_new = (c - 0.5) @ np.array([[ca, -sa], [sa, ca]]) + 0.5
            
        c_new = np.clip(c_new, 0.02, 0.98)
        
        r_new = solve_lp_radii(c_new)
        s_new = np.sum(r_new)
        
        if s_new > best_sum or np.random.rand() < np.exp((s_new - best_sum) / max(T, 1e-6)):
            c, r = c_new, r_new
            if s_new > best_sum:
                best_sum, best_c = s_new, c.copy()
                
        T *= 0.9992
        
    return best_c, best_sum

def run_packing():
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    bounds_obj = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -1.0
    best_x = None

    # Phase 1: Diverse Initial Configurations
    inits = []
    for r0 in [0.085, 0.090, 0.095, 0.100]:
        for ang in np.linspace(-0.35, 0.35, 13):
            inits.append(make_hex_init(r0, ang))
            
    for s in range(8):
        inits.append(make_force_init(s))
        
    for s in range(5):
        np.random.seed(s + 200)
        inits.append(np.random.uniform(0.15, 0.85, (N, 2)))

    # Phase 2: Multi-start Optimization with LP Initialization
    for c_init in inits:
        x0 = np.zeros(3 * N)
        x0[0::3] = c_init[:, 0]
        x0[1::3] = c_init[:, 1]
        
        r_lp = solve_lp_radii(c_init)
        x0[2::3] = np.maximum(r_lp * 0.99, 1e-5)
        x0 = project_to_feasible(x0)

        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds_obj, constraints=cons,
                           options={'maxiter': 15000, 'ftol': 1e-13, 'disp': False})
            if not np.isnan(res.fun):
                c_tmp = np.column_stack((res.x[0::3], res.x[1::3]))
                r_new = solve_lp_radii(c_tmp)
                curr_sum = np.sum(r_new)
                
                if curr_sum > best_sum and np.min(constraints(res.x)) >= -1e-5:
                    best_sum = curr_sum
                    best_x = res.x.copy()
                    best_x[2::3] = r_new
        except Exception:
            continue

    # Phase 3 & 4: SA and Deflation Repositioning
    if best_x is not None:
        bc = np.column_stack((best_x[0::3], best_x[1::3]))
        
        # Simulated Annealing
        sa_c, sa_sum = simulated_annealing(bc, steps=5000, T0=0.035)
        if sa_sum > best_sum:
            best_sum = sa_sum
            best_x[0::3] = sa_c[:, 0]
            best_x[1::3] = sa_c[:, 1]
            best_x[2::3] = solve_lp_radii(sa_c)
            
        # Deflation & Repositioning to break rigid contact networks
        for cyc in range(60):
            x0 = best_x.copy()
            noise = 0.002 * (0.91 ** cyc)
            x0[0::3] += np.random.normal(0, noise, N)
            x0[1::3] += np.random.normal(0, noise, N)
            
            # Deflate a subset significantly to allow topology changes
            k = np.random.randint(2, 7)
            subset = np.random.choice(N, size=k, replace=False)
            x0[subset * 3 + 2] *= 0.78
            
            x0 = project_to_feasible(x0)
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds_obj, constraints=cons,
                               options={'maxiter': 10000, 'ftol': 1e-13, 'disp': False})
                if not np.isnan(res.fun):

                    r_p = solve_lp_radii(c_p)
                    curr_sum = np.sum(r_p)
                    if curr_sum > best_sum and np.min(constraints(res.x)) >= -1e-5:

                        best_x = res.x.copy()
                        best_x[2::3] = r_p
            except Exception:
                pass

        # Phase 5: Coordinate Descent Fine-Tuning
        curr_centers = np.column_stack((best_x[0::3], best_x[1::3]))
        curr_radii = best_x[2::3]
        best_local_sum = np.sum(curr_radii)
        
        for _ in range(5):
            improved = True
            while improved:
                improved = False
                for i in range(N):
                    best_step_sum = best_local_sum
                    best_step_c = curr_centers[i].copy()
                    
                    # Multi-scale steps
                    for dx in [-0.005, -0.002, -0.001, 0.001, 0.002, 0.005]:
                        for dy in [-0.005, -0.002, -0.001, 0.001, 0.002, 0.005]:
                            nc = curr_centers.copy()
                            nc[i, 0] = np.clip(curr_centers[i, 0] + dx, 0.02, 0.98)
                            nc[i, 1] = np.clip(curr_centers[i, 1] + dy, 0.02, 0.98)
                            
                            rn = solve_lp_radii(nc)
                            sn = np.sum(rn)
                            if sn > best_step_sum + 1e-8:
                                best_step_sum = sn
                                best_step_c = nc[i].copy()
                                
                    if best_step_sum > best_local_sum + 1e-8:
                        curr_centers[i] = best_step_c
                        best_local_sum = best_step_sum
                        curr_radii = solve_lp_radii(curr_centers)
                        best_x[0::3] = curr_centers[:, 0]
                        best_x[1::3] = curr_centers[:, 1]
                        best_x[2::3] = curr_radii
                        best_sum = best_local_sum
                        improved = True

        # Phase 6: Fine Rotation Search
        best_centers = np.column_stack((best_x[0::3], best_x[1::3]))

            r_rot = solve_lp_radii(rot_c)
            
            if np.sum(r_rot) > best_sum:
                best_sum = np.sum(r_rot)
                best_x[0::3] = rot_c[:, 0]

        best_sum = np.sum(best_x[2::3])

    centers = np.column_stack((best_x[0::3], best_x[1::3]))

    # Final strict validity adjustment against 1e-12 tolerance
    for _ in range(100):

        for i in range(N):

                if not valid:

            break
            
        # Minimal shrinkage to recover strict feasibility
        radii *= 0.9999
        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)

    return centers, radii, float(np.sum(radii))
