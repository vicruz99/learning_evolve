3. **Hybrid SLSQP Polish**: Use the SA+LocalSearch result as a high-quality starting point for SLSQP to achieve precise numerical convergence.

import numpy as np
from scipy.optimize import minimize, linprog
import warnings

warnings.filterwarnings('ignore')

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist)
            idx += 1
            
    bounds_lp = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_lp, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-9)
    except Exception:
        pass
    return np.full(n, 0.05)

def local_search_centers(centers, passes=3):
    """Greedy coordinate descent to maximize LP sum of radii."""
    c = centers.copy()
    for _ in range(passes):
        for i in range(N):
            best_c = c.copy()
            best_r = solve_lp_radii(c)
            best_s = np.sum(best_r)
            
            # Try fine displacements
            for dx in [-0.004, -0.002, -0.001, 0.001, 0.002, 0.004]:
                for dy in [-0.004, -0.002, -0.001, 0.001, 0.002, 0.004]:
                    nc = c.copy()
                    nc[i, 0] = np.clip(c[i, 0] + dx, 0.01, 0.99)
                    nc[i, 1] = np.clip(c[i, 1] + dy, 0.01, 0.99)
                    nr = solve_lp_radii(nc)
                    ns = np.sum(nr)
                    if ns > best_s + 1e-9:
                        best_s, best_r, best_c = ns, nr, nc
            c[i] = best_c[i]
    return c, solve_lp_radii(c), np.sum(solve_lp_radii(c))

def sa_optimize(init_centers, n_steps=1500):
    """Simulated Annealing directly optimizing the LP sum of radii."""
    np.random.seed(42)
    c = init_centers.copy()
    r = solve_lp_radii(c)
    best_c, best_r, best_s = c.copy(), r.copy(), np.sum(r)
    T = 0.04
    
    for step in range(n_steps):
        c_new = c.copy()
        m = np.random.rand()
        if m < 0.6:
            # Local displacement
            k = np.random.randint(1, 4)
            idx = np.random.choice(N, k, replace=False)
            c_new[idx] += np.random.normal(0, 0.007, (k, 2))
        elif m < 0.8:
            # Global rotation
            ang = np.random.uniform(-0.02, 0.02)
            ca, sa = np.cos(ang), np.sin(ang)
            c_new = (c - 0.5) @ np.array([[ca, -sa], [sa, ca]]) + 0.5
        else:
            # Deflation jump to break rigid contacts
            k = np.random.randint(2, 5)
            idx = np.random.choice(N, k, replace=False)
            c_new[idx] += np.random.normal(0, 0.035, (k, 2))
            
        c_new = np.clip(c_new, 0.02, 0.98)
        r_new = solve_lp_radii(c_new)
        s_new = np.sum(r_new)
        
        delta = s_new - best_s
        if delta > 0 or (T > 1e-5 and np.random.rand() < np.exp(delta / T)):
            c = c_new
            r = r_new
            if s_new > best_s:
                best_c, best_r, best_s = c_new.copy(), r_new.copy(), s_new
        T *= 0.996
    return best_c, best_r, best_s

def make_hex_init(r0, angle, scale=1.0):
    """Generates a rotated/scaled hexagonal lattice initialization."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 5:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0 and len(pts) < N + 5:
            pts.append([x * scale, y * scale])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        
    pts = np.array(pts[:N + 5])
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
        
    mask = (pts[:, 0] >= 0.01) & (pts[:, 0] <= 0.99) & (pts[:, 1] >= 0.01) & (pts[:, 1] <= 0.99)
    pts = pts[mask]
    
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.1, 0.9, (1, 2))])
    return pts[:N]

def make_force_init(seed):
    """Force-directed layout to spread points evenly."""
    np.random.seed(seed)
    pts = np.random.uniform(0.15, 0.85, (N, 2))
    for step in range(200):
        f = np.zeros_like(pts)
        lr = 0.015 * (1.0 - step / 200.0)
        for i in range(N):
            for j in range(i + 1, N):
                dx = pts[j] - pts[i]
                d = np.hypot(dx[0], dx[1])
                if d < 0.25 and d > 1e-5:
                    rep = 0.01 / (d**2 + 0.001)
                    f[i] -= dx * rep / d
                    f[j] += dx * rep / d
            for dim in range(2):
                if pts[i, dim] < 0.1: f[i, dim] += 0.04
                elif pts[i, dim] > 0.9: f[i, dim] -= 0.04
        pts += f * lr
        pts = np.clip(pts, 0.02, 0.98)
    return pts

def constraints(x):
    """Vectorized inequality constraints for SLSQP: g(x) >= 0."""
    xs, ys, rs = x[0::3], x[1::3], x[2::3]
    c = np.concatenate([
        xs - rs, 1.0 - xs - rs,
        ys - rs, 1.0 - ys - rs
    ])
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    c = np.concatenate([c, dx[TRIL_IDX]**2 + dy[TRIL_IDX]**2 - dr[TRIL_IDX]**2])
    return c

def objective(x):
    return -np.sum(x[2::3])

def get_bounds():
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)])
    return b

def project_to_feasible(x):
    x = x.copy()
    for i in range(N):
        r = max(1e-6, x[3 * i + 2])
        x[3 * i] = np.clip(x[3 * i], r, 1.0 - r)
        x[3 * i + 1] = np.clip(x[3 * i + 1], r, 1.0 - r)
        x[3 * i + 2] = r
    return x

def run_packing():
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    bounds_obj = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -1.0
    best_x = None

    # Phase 1: Generate Diverse Initial Configurations
    inits = []
    for r0 in [0.085, 0.090, 0.095, 0.100]:
        for ang in np.linspace(-0.35, 0.35, 11):
            inits.append(make_hex_init(r0, ang))
    for s in range(8):
        inits.append(make_force_init(s))
    for s in range(6):
        np.random.seed(s + 500)
        inits.append(np.random.uniform(0.15, 0.85, (N, 2)))

    # Phase 2: SA + Local Search Pipeline
    for c_init in inits:
        # SA exploration
        c_sa, r_sa, s_sa = sa_optimize(c_init, n_steps=1200)
        
        # Local coordinate descent polish
        c_ls, r_ls, s_ls = local_search_centers(c_sa, passes=3)
        
        if s_ls > best_sum:
            best_sum = s_ls
            best_c = c_ls.copy()
            best_r = r_ls.copy()

    if best_x is None and best_c is not None:
        # Phase 3: SLSQP Precision Polish
        x0 = np.zeros(3 * N)
        x0[0::3] = best_c[:, 0]
        x0[1::3] = best_c[:, 1]
        x0[2::3] = np.maximum(best_r * 0.998, 1e-6)
        x0 = project_to_feasible(x0)
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds_obj, constraints=cons,
                           options={'maxiter': 25000, 'ftol': 1e-14, 'disp': False})
            if not np.isnan(res.fun):
                c_tmp = np.column_stack((res.x[0::3], res.x[1::3]))
                r_new = solve_lp_radii(c_tmp)
                curr_sum = np.sum(r_new)
                if curr_sum > best_sum and np.min(constraints(res.x)) >= -1e-6:
                    best_sum = curr_sum
                    best_x = res.x.copy()
                    best_x[2::3] = r_new
        except Exception:
            pass

    # Phase 4: Deflation & Perturbation Cycles
    if best_x is not None:
        for cycle in range(50):
            x0 = best_x.copy()
            noise_scale = 0.002 * (0.92 ** cycle)
            x0[0::3] += np.random.normal(0, noise_scale, N)
            x0[1::3] += np.random.normal(0, noise_scale, N)
            
            # Deflate subset to break rigidity
            k = np.random.randint(3, 7)
            subset = np.random.choice(N, size=k, replace=False)
            x0[subset * 3 + 2] *= 0.85
            
            x0 = project_to_feasible(x0)

            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds_obj, constraints=cons,
                               options={'maxiter': 15000, 'ftol': 1e-13, 'disp': False})
                if not np.isnan(res.fun):
                    centers_p = np.column_stack((res.x[0::3], res.x[1::3]))
                    r_p = solve_lp_radii(centers_p)
                    if r_p is not None:
                        curr_sum = np.sum(r_p)
                        if curr_sum > best_sum and np.min(constraints(res.x)) >= -1e-6:
                            best_sum = curr_sum
                            best_x = res.x.copy()
                            best_x[2::3] = r_p
            except Exception:
                pass

        # Phase 5: Final Rotation Scan & Uniform Expansion
        best_centers = np.column_stack((best_x[0::3], best_x[1::3]))
        for ang_deg in np.linspace(-12, 12, 13):
            ang = np.deg2rad(ang_deg)
            ca, sa = np.cos(ang), np.sin(ang)
            rot_c = (best_centers - 0.5) @ np.array([[ca, -sa], [sa, ca]]) + 0.5
            rot_c = np.clip(rot_c, 0.01, 0.99)
            r_rot = solve_lp_radii(rot_c)
            if np.sum(r_rot) > best_sum:
                best_sum = np.sum(r_rot)
                best_x[0::3] = rot_c[:, 0]
                best_x[1::3] = rot_c[:, 1]
                best_x[2::3] = r_rot

    # Fallback
    if best_x is None:
        best_x = np.zeros(3 * N)
        best_x[0::3] = np.tile(np.linspace(0.1, 0.9, 5), 6)[:N]
        best_x[1::3] = np.repeat(np.linspace(0.1, 0.9, 6), 5)[:N]
        best_x[2::3] = 0.06
        best_sum = np.sum(best_x[2::3])

    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3].copy()

    # Final strict validity adjustment
    for _ in range(100):
        valid = True
        for i in range(N):
            if radii[i] < 0 or centers[i, 0] < radii[i] - 1e-9 or centers[i, 0] > 1.0 - radii[i] + 1e-9 or \
               centers[i, 1] < radii[i] - 1e-9 or centers[i, 1] > 1.0 - radii[i] + 1e-9:
                valid = False
                break
        if valid:
            for i in range(N):
                for j in range(i + 1, N):
                    d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                    if d < radii[i] + radii[j] - 1e-9:
                        valid = False
                        break
                if not valid:
                    break
        if valid:
            break
            
        radii *= 0.9999
        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)

    return centers, radii, float(np.sum(radii))
