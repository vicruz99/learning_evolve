import numpy as np
from scipy.optimize import minimize, linprog
import warnings

warnings.filterwarnings('ignore')

N = 26
TRIL_IDX = np.tril_indices(N, -1)
TRIU_IDX = np.triu_indices(N, 1)

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist)
            idx += 1
            
    bounds_lp = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_lp, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-9)
    except Exception:
        pass
    return np.full(n, 0.05)

def center_penalty(c_flat, radii):
    """Smooth penalty function for centers given fixed radii."""
    c = c_flat.reshape(-1, 2)
    p = 0.0
    x, y, r = c[:, 0], c[:, 1], radii
    
    # Boundary penalties
    p += 1000.0 * np.sum(np.maximum(r - x, 0.0)**2)
    p += 1000.0 * np.sum(np.maximum(x - (1.0 - r), 0.0)**2)
    p += 1000.0 * np.sum(np.maximum(r - y, 0.0)**2)
    p += 1000.0 * np.sum(np.maximum(y - (1.0 - r), 0.0)**2)
    
    # Overlap penalties
    dx = c[:, 0, None] - c[:, 0]
    dy = c[:, 1, None] - c[:, 1]
    d = np.hypot(dx, dy)
    gap = d - r[:, None] - r[None, :]
    p += 5000.0 * np.sum(np.maximum(-gap[TRIU_IDX], 0.0)**2)
    return p

def opt_centers_lbfgs(centers, radii):
    """Optimize centers for fixed radii using L-BFGS-B."""
    bounds = [(0.001, 0.999)] * (2 * N)
    res = minimize(center_penalty, centers.flatten(), args=(radii,),
                   method='L-BFGS-B', bounds=bounds, options={'maxiter': 5000, 'ftol': 1e-15})
    return res.x.reshape(-1, 2)

def make_hex_init(r0, angle):
    """Generates a rotated hexagonal lattice initialization."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0 and len(pts) < N + 10:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
    pts = np.array(pts[:N + 10])
    if angle != 0.0:
        ca, sa = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[ca, -sa], [sa, ca]]) + 0.5
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.15, 0.85, (1, 2))])
    return pts[:N]

def make_force_init(seed):
    """Force-directed layout to spread points evenly and push to boundaries."""
    np.random.seed(seed)
    pts = np.random.uniform(0.15, 0.85, (N, 2))
    for step in range(200):
        f = np.zeros_like(pts)
        lr = 0.015 * (1.0 - step / 200.0)
        for i in range(N):
            for j in range(i + 1, N):
                dx = pts[j] - pts[i]
                d = np.hypot(dx[0], dx[1])
                if d < 0.25 and d > 1e-5:
                    rep = 0.012 / (d**2 + 0.001)
                    f[i] -= dx * rep / d
                    f[j] += dx * rep / d
            for dim in range(2):
                if pts[i, dim] < 0.12: f[i, dim] += 0.05
                elif pts[i, dim] > 0.88: f[i, dim] -= 0.05
        pts += f * lr
        pts = np.clip(pts, 0.03, 0.97)
    return pts

def run_packing():
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    best_sum = -1.0
    best_c = None
    best_r = None

    # Phase 1: Diverse Initial Configurations
    inits = []
    for ang in np.linspace(-0.4, 0.4, 15):
        inits.append(make_hex_init(0.092, ang))
    for s in range(10):
        inits.append(make_force_init(s))
    for s in range(10):
        np.random.seed(s + 1000)
        inits.append(np.random.uniform(0.15, 0.85, (N, 2)))

    # Initial optimization from diverse starts
    for c0 in inits:
        c = c0.copy()
        r = solve_lp_radii(c)
        for _ in range(8):
            r = solve_lp_radii(c)
            c = opt_centers_lbfgs(c, r)
        s_val = np.sum(r)
        if s_val > best_sum:
            best_sum = s_val
            best_c = c.copy()
            best_r = r.copy()

    # Phase 2: Basin Hopping with Deflation to escape local minima
    if best_c is not None:
        c_curr, r_curr = best_c.copy(), best_r.copy()
        T = 0.025
        for step in range(600):
            c_new = c_curr.copy()
            r_new = r_curr.copy()
            
            choice = np.random.rand()
            if choice < 0.65:
                # Jitter subset
                k = np.random.randint(3, 12)
                idx = np.random.choice(N, k, replace=False)
                c_new[idx] += np.random.normal(0, 0.006, (k, 2))
                r_new[idx] *= np.random.uniform(0.75, 0.95)
            elif choice < 0.85:
                # Rotate cluster
                ang = np.random.uniform(-0.05, 0.05)
                ca, sa = np.cos(ang), np.sin(ang)
                c_new = (c_curr - 0.5) @ np.array([[ca, -sa], [sa, ca]]) + 0.5
            else:
                # Scale & jitter
                s = np.random.uniform(0.97, 1.03)
                c_new = (c_curr - 0.5) * s + 0.5
                c_new += np.random.uniform(-0.008, 0.008, c_curr.shape)
                
            c_new = np.clip(c_new, 0.005, 0.995)
            
            # Optimize radii and centers for the perturbed state
            r_new = solve_lp_radii(c_new)
            for _ in range(6):
                r_new = solve_lp_radii(c_new)
                c_new = opt_centers_lbfgs(c_new, r_new)
                
            s_new = np.sum(r_new)
            
            # Acceptance criterion
            if s_new > best_sum or (T > 1e-6 and np.random.rand() < np.exp((s_new - best_sum) / T)):
                c_curr, r_curr = c_new, r_new
                if s_new > best_sum:
                    best_sum = s_new
                    best_c = c_curr.copy()
                    best_r = r_curr.copy()
            T *= 0.994

    # Phase 3: SLSQP Constrained Polish
    def objective_slsqp(x):
        return -np.sum(x[2::3])

    def constraints_slsqp(x):
        xs, ys, rs = x[0::3], x[1::3], x[2::3]
        c = np.concatenate([xs - rs, 1.0 - xs - rs, ys - rs, 1.0 - ys - rs])
        dx = xs[:, None] - xs[None, :]
        dy = ys[:, None] - ys[None, :]
        dr = rs[:, None] + rs[None, :]
        c = np.concatenate([c, np.hypot(dx[TRIL_IDX], dy[TRIL_IDX]) - dr[TRIL_IDX]])
        return c

    x0 = np.zeros(3 * N)
    x0[0::3] = best_c[:, 0]
    x0[1::3] = best_c[:, 1]
    x0[2::3] = best_r * 0.995
    
    bounds_obj = [(0.0, 1.0), (0.0, 1.0), (1e-8, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints_slsqp}
    
    try:
        res = minimize(objective_slsqp, x0, method='SLSQP', bounds=bounds_obj, constraints=cons,
                       options={'maxiter': 20000, 'ftol': 1e-14})
        if not np.isnan(res.fun):
            c_tmp = np.column_stack((res.x[0::3], res.x[1::3]))
            r_tmp = solve_lp_radii(c_tmp)
            if np.sum(r_tmp) > best_sum:
                best_sum = np.sum(r_tmp)
                best_c = c_tmp
                best_r = r_tmp
    except Exception:
        pass

    # Phase 4: Coordinate Descent Micro-Adjustments
    curr_centers = best_c.copy()
    curr_radii = best_r.copy()
    best_local_sum = np.sum(curr_radii)
    
    for _ in range(4):
        improved = True
        while improved:
            improved = False
            for i in range(N):
                best_step_sum = best_local_sum
                best_step_c = curr_centers[i].copy()
                
                for dx in [-0.005, -0.002, -0.001, 0.001, 0.002, 0.005]:
                    for dy in [-0.005, -0.002, -0.001, 0.001, 0.002, 0.005]:
                        new_centers = curr_centers.copy()
                        new_centers[i, 0] = np.clip(curr_centers[i, 0] + dx, 0.005, 0.995)
                        new_centers[i, 1] = np.clip(curr_centers[i, 1] + dy, 0.005, 0.995)
                        
                        r_new = solve_lp_radii(new_centers)
                        s_new = np.sum(r_new)
                        if s_new > best_step_sum + 1e-8:
                            best_step_sum = s_new
                            best_step_c = new_centers[i].copy()
                            
                if best_step_sum > best_local_sum + 1e-8:
                    curr_centers[i] = best_step_c
                    best_local_sum = best_step_sum
                    curr_radii = solve_lp_radii(curr_centers)
                    best_c = curr_centers.copy()
                    best_r = curr_radii.copy()
                    best_sum = best_local_sum
                    improved = True

    # Phase 5: Global Rotation Search
    for ang_deg in np.linspace(-15, 15, 17):
        ang = np.deg2rad(ang_deg)
        ca, sa = np.cos(ang), np.sin(ang)
        rot_c = (best_c - 0.5) @ np.array([[ca, -sa], [sa, ca]]) + 0.5
        rot_c = np.clip(rot_c, 0.01, 0.99)
        r_rot = solve_lp_radii(rot_c)
        
        if np.sum(r_rot) > best_sum:
            best_sum = np.sum(r_rot)
            best_c = rot_c
            best_r = r_rot

    # Final strict validity adjustment against 1e-12 tolerance
    centers = best_c.copy()
    radii = best_r.copy()
    for _ in range(100):
        valid = True
        for i in range(N):
            if radii[i] < 0 or centers[i, 0] < radii[i] - 1e-9 or centers[i, 0] > 1.0 - radii[i] + 1e-9 or \
               centers[i, 1] < radii[i] - 1e-9 or centers[i, 1] > 1.0 - radii[i] + 1e-9:
                valid = False
                break
        if valid:
            for i in range(N):
                for j in range(i + 1, N):
                    d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                    if d < radii[i] + radii[j] - 1e-9:
                        valid = False
                        break
                if not valid:
                    break
        if valid:
            break
            
        radii *= 0.9999
        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)

    return centers, radii, float(np.sum(radii))
