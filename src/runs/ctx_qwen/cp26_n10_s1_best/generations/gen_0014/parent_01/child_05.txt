To break through the plateau around 2.6359, we need a more aggressive topology search. Standard gradient-based methods (like SLSQP) struggle with the non-smooth, combinatorial nature of circle packing contact networks. 

Strategy:
1. **LP-Driven Evaluation**: Use Linear Programming (`solve_lp_radii`) to compute the exact optimal radii for any fixed set of centers. This converts the problem into maximizing a non-smooth objective over centers.
2. **Simulated Annealing (SA) on Centers**: Run a custom SA that perturbs centers using diverse moves (single displacement, cluster rotation, reflection, boundary pushing) and accepts/rejects based on the LP-computed sum of radii. This bypasses gradient singularities at contact points.
3. **Deflation & Regrowth**: Periodically shrink a random subset of circles, perturb their positions, and let the LP naturally regrow them into newly freed gaps. This effectively breaks rigid contact graphs.
4. **Multi-Start & Polish**: Initialize SA from diverse hexagonal, force-directed, and random layouts. After SA converges, polish the result with SLSQP and coordinate descent to extract every last bit of precision.
5. **Robust Validation**: Final strict clipping and minimal shrinkage to guarantee `1e-12` tolerance compliance.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
TRIL_IDX = np.tril_indices(N, -1)
TRIU_IDX = np.triu_indices(N, 1)

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist)
            idx += 1
            
    bounds_lp = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_lp, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-9)
    except Exception:
        pass
    return np.full(n, 0.05)

def center_penalty(c_flat, radii):
    """Smooth penalty function for centers given fixed radii."""
    c = c_flat.reshape(-1, 2)
    p = 0.0
    x, y, r = c[:, 0], c[:, 1], radii
    
    # Boundary penalties
    p += 100.0 * np.sum(np.maximum(r - x, 0.0)**2)
    p += 100.0 * np.sum(np.maximum(x - (1.0 - r), 0.0)**2)
    p += 100.0 * np.sum(np.maximum(r - y, 0.0)**2)
    p += 100.0 * np.sum(np.maximum(y - (1.0 - r), 0.0)**2)
    
    # Overlap penalties
    dx = c[:, 0, None] - c[:, 0]
    dy = c[:, 1, None] - c[:, 1]
    d = np.hypot(dx, dy)
    gap = d - r[:, None] - r[None, :]
    p += 500.0 * np.sum(np.maximum(-gap[TRIU_IDX], 0.0)**2)
    return p

def opt_centers_lbfgs(centers, radii):
    """Optimize centers for fixed radii using L-BFGS-B."""
    bounds = [(0.001, 0.999)] * (2 * N)
    res = minimize(center_penalty, centers.flatten(), args=(radii,),
                   method='L-BFGS-B', bounds=bounds, options={'maxiter': 3000, 'ftol': 1e-15})
    return res.x.reshape(-1, 2)

def make_hex_init(r0, angle):
    """Generates a rotated hexagonal lattice initialization."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0 and len(pts) < N + 10:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        
    pts = np.array(pts[:N + 10])
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
        
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.15, 0.85, (1, 2))])
    return pts[:N]

def make_force_init(seed):
    """Force-directed layout to spread points evenly and push to boundaries."""
    np.random.seed(seed)
    pts = np.random.uniform(0.15, 0.85, (N, 2))
    for _ in range(200):
        f = np.zeros_like(pts)
        for i in range(N):
            for j in range(i + 1, N):
                dx = pts[j] - pts[i]
                d = np.hypot(dx[0], dx[1])
                if d < 0.25 and d > 1e-5:
                    rep = 0.012 / (d**2 + 0.001)
                    f[i] -= dx * rep / d
                    f[j] += dx * rep / d
            for dim in range(2):
                if pts[i, dim] < 0.12: f[i, dim] += 0.05
                elif pts[i, dim] > 0.88: f[i, dim] -= 0.05
        pts += f * 0.04
        pts = np.clip(pts, 0.03, 0.97)
    return pts

def sa_optimize(initial_centers, n_steps=1200):
    """Simulated Annealing on centers using LP radii evaluation."""
    c = initial_centers.copy()
    r = solve_lp_radii(c)
    best_sum = np.sum(r)
    best_c = c.copy()
    best_r = r.copy()
    T = 0.03
    
    for step in range(n_steps):
        c_new = c.copy()
        move_type = np.random.choice(['perturb', 'cluster_rotate', 'reflect', 'boundary_push', 'deflate_regrow'])
        
        if move_type == 'perturb':
            idx = np.random.randint(N)
            c_new[idx] += np.random.normal(0, 0.006, 2)
        elif move_type == 'cluster_rotate':
            center_idx = np.random.randint(N)
            dists = np.linalg.norm(c - c[center_idx], axis=1)
            neighbors = np.where(dists < 0.25)[0]
            if len(neighbors) > 1:
                ang = np.random.uniform(-0.06, 0.06)
                ca, sa = np.cos(ang), np.sin(ang)
                centroid = c[neighbors].mean(axis=0)
                c_new[neighbors] = (c[neighbors] - centroid) @ np.array([[ca, -sa], [sa, ca]]) + centroid
        elif move_type == 'reflect':
            i, j = np.random.choice(N, 2, replace=False)
            c_new[i] = 2 * c[j] - c[i]
        elif move_type == 'boundary_push':
            idx = np.random.randint(N)
            for dim in range(2):
                if c[idx, dim] < 0.5: c_new[idx, dim] -= np.random.uniform(0.003, 0.012)
                else: c_new[idx, dim] += np.random.uniform(0.003, 0.012)
        elif move_type == 'deflate_regrow':
            k = np.random.randint(3, 7)
            subset = np.random.choice(N, size=k, replace=False)
            c_new[subset] += np.random.normal(0, 0.008, (k, 2))
            
        c_new = np.clip(c_new, 0.005, 0.995)
        
        r_new = solve_lp_radii(c_new)
        s_new = np.sum(r_new)
        
        delta = s_new - best_sum
        if delta > 0 or (T > 1e-6 and np.random.rand() < np.exp(delta / T)):
            c, r = c_new, r_new
            if s_new > best_sum:
                best_sum, best_c, best_r = s_new, c_new.copy(), r_new.copy()
        T *= 0.996
        
    return best_c, best_r, best_sum

def constraints_slsqp(x):
    """Inequality constraints for SLSQP: g(x) >= 0."""
    xs, ys, rs = x[0::3], x[1::3], x[2::3]
    c = np.concatenate([xs - rs, 1.0 - xs - rs, ys - rs, 1.0 - ys - rs])
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    c = np.concatenate([c, np.hypot(dx[TRIL_IDX], dy[TRIL_IDX]) - dr[TRIL_IDX]])
    return c

def objective_func(x):
    """Objective: maximize sum of radii."""
    return -np.sum(x[2::3])

def run_packing():
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    bounds_obj = [(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints_slsqp}
    
    best_sum = -1.0
    best_x = None
    
    # Phase 1: Diverse Initial Configurations
    inits = []
    for ang in np.linspace(-0.35, 0.35, 15):
        inits.append(make_hex_init(0.092, ang))
    for s in range(10):
        inits.append(make_force_init(s))
    for s in range(8):
        np.random.seed(s + 1000)
        inits.append(np.random.uniform(0.15, 0.85, (N, 2)))

    # Phase 2: Alternating Optimization + SLSQP Polish
    for pts in inits:
        c = pts.copy()
        r = solve_lp_radii(c)
        
        for _ in range(15):
            r = solve_lp_radii(c)
            c = opt_centers_lbfgs(c, r)
            
        x0 = np.zeros(3 * N)
        x0[0::3] = c[:, 0]
        x0[1::3] = c[:, 1]
        x0[2::3] = np.maximum(r * 0.995, 1e-6)
        
        try:
            res = minimize(objective_func, x0, method='SLSQP', bounds=bounds_obj, constraints=cons,
                           options={'maxiter': 15000, 'ftol': 1e-14})
            if not np.isnan(res.fun):
                curr_sum = -res.fun
                if curr_sum > best_sum and np.min(constraints_slsqp(res.x)) >= -1e-6:
                    best_sum = curr_sum
                    best_x = res.x.copy()
        except Exception:
            pass

    # Phase 3: Basin Hopping (SA on Centers) to escape local minima

        bc = np.column_stack((best_x[0::3], best_x[1::3]))

                best_x = np.zeros(3 * N)

                best_x[1::3] = bc[:, 1]
                best_x[2::3] = br
                
        # Final SLSQP polish after SA
        x0 = best_x.copy()
        x0[2::3] *= 0.998
        for i in range(N):
            r = x0[3 * i + 2]
            x0[3 * i] = np.clip(x0[3 * i], r, 1.0 - r)

        best_x[1::3] = np.repeat(np.linspace(0.1, 0.9, 6), 5)[:N]
