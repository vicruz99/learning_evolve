import warnings

N = 26
TRIL_IDX = np.tril_indices(N, -1)
TRIU_IDX = np.triu_indices(N, 1)

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    dx = centers[:, 0][:, None] - centers[:, 0]
    dy = centers[:, 1][:, None] - centers[:, 1]
    dists = np.hypot(dx, dy)
    
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    # Boundary constraints
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[i, i] = 1.0
        b_ub[i] = max(0.0, lim)
        
    # Overlap constraints
    k = n
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = max(0.0, dists[i, j])
            k += 1
            
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=[(0.0, None)]*n, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-9)
    except Exception:
        pass
    return np.full(n, 0.01)

def constraints(x):
    """Inequality constraints for SLSQP: g(x) >= 0."""
    xs, ys, rs = x[0::3], x[1::3], x[2::3]
    c = np.concatenate([xs - rs, 1.0 - xs - rs, ys - rs, 1.0 - ys - rs])
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    c = np.concatenate([c, np.hypot(dx[TRIL_IDX], dy[TRIL_IDX]) - dr[TRIL_IDX]])
    return c

def objective(x):

    return -np.sum(x[2::3])

def make_hex_init(r0, angle):

    pts = []
    y = r0
    row = 0
    while len(pts) < N + 10:

            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
    pts = np.array(pts[:N + 10])

    pts = pts[mask]

        pts = np.vstack([pts, np.random.uniform(0.15, 0.85, (1, 2))])
    return pts[:N]

def make_force_init(seed):
    """Force-directed layout to spread points evenly and push to boundaries."""
    np.random.seed(seed)
    pts = np.random.uniform(0.15, 0.85, (N, 2))
    for _ in range(150):
        f = np.zeros_like(pts)
        for i in range(N):
            for j in range(i + 1, N):
                dx = pts[j] - pts[i]
                d = np.hypot(dx[0], dx[1])
                if d < 0.25 and d > 1e-5:
                    rep = 0.012 / (d**2 + 0.001)
                    f[i] -= dx * rep / d
                    f[j] += dx * rep / d

                elif pts[i, dim] > 0.88: f[i, dim] -= 0.05
        pts += f * 0.04
        pts = np.clip(pts, 0.03, 0.97)
    return pts

def basin_hopping(centers0, steps=700):

    c = centers0.copy()
    r = solve_lp_radii(c)
    best_sum = np.sum(r)
    best_c = c.copy()
    T = 0.006

            if s_new > best_sum:
                best_sum, best_c = s_new, c_new.copy()
        T *= 0.994
        
    return best_c, best_sum
