1. **LP Radii Step**: For any fixed set of centers, LP globally maximizes the sum of radii subject to non-overlap and boundary constraints. This avoids the local minima traps that plague gradient-based radius optimization.
2. **Force-Directed Centers Step**: After radii are expanded, the force simulation pushes overlapping or tightly packed centers apart, creating clearance for the next LP step to grow radii further. This natural feedback loop drives the system toward denser configurations.
3. **SLSQP Polish**: Once the alternating loop converges, a precise gradient-based constrained optimizer (SLSQP) refines the exact boundary and contact points.
4. **Deflation & Perturbation**: To escape local optima, the best solution is periodically shrunk and perturbed, then re-optimized. This mimics simulated annealing and helps jump to higher-sum basins.
5. **Diverse Initialization**: Multiple rotated hexagonal lattices, corner-focused layouts, and random starts ensure broad coverage of the solution space.

import numpy as np
from scipy.optimize import linprog, minimize

N_CIRCLES = 26

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [0, 0.5]."""
    return [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N_CIRCLES

def sgs_objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def sgs_constraints(x):
    """Compute all boundary and non-overlap constraints as a vector >= 0."""
    cx = x[0::3]
    cy = x[1::3]
    cr = x[2::3]
    # Boundary constraints
    c = np.concatenate([cx - cr, 1.0 - cx - cr, cy - cr, 1.0 - cy - cr])
    # Overlap constraints (squared distances)
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dr = cr[:, None] + cr[None, :]
    idx = np.tril_indices(N_CIRCLES, -1)
    c = np.concatenate([c, dx[idx]**2 + dy[idx]**2 - dr[idx]**2])
    return c

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        row = np.zeros(n)
        row[i] = 1.0
        A_ub.append(row)
        b_ub.append(max(0.0, lim))
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dist)
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-9)
    except Exception:
        pass
    return np.full(n, 0.05)

def force_optimize_centers(centers, radii, steps=150):
    """Push centers apart using repulsive forces to create clearance for larger radii."""
    pts = centers.copy()
    rs = radii.copy()
    lr = 0.02
    for step in range(steps):
        forces = np.zeros_like(pts)
        for i in range(N_CIRCLES):
            for j in range(i + 1, N_CIRCLES):
                dx = pts[j, 0] - pts[i, 0]
                dy = pts[j, 1] - pts[i, 1]
                d = np.hypot(dx, dy)
                target = rs[i] + rs[j] + 0.002
                if d < target and d > 1e-6:
                    f = (target - d) * 0.5
                    fx = f * dx / d
                    fy = f * dy / d
                    forces[i] -= [fx, fy]
                    forces[j] += [fx, fy]
            # Boundary repulsion
            for dim in range(2):
                if pts[i, dim] < rs[i] + 0.005:
                    forces[i, dim] += 0.1
                elif pts[i, dim] > 1.0 - rs[i] - 0.005:
                    forces[i, dim] -= 0.1
        pts += forces * lr * (1.0 - step / steps)
        pts = np.clip(pts, 1e-4, 1.0 - 1e-4)
    return pts

def generate_hex_init(r0, angle, seed, pad=5):
    """Generate a rotated hexagonal lattice initialization."""
    np.random.seed(seed)
    pts = []
    y = r0
    row = 0
    while len(pts) < N_CIRCLES + pad:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0 and len(pts) < N_CIRCLES + pad:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
    pts = np.array(pts[:N_CIRCLES + pad])
    
    if angle != 0.0:
        c = np.array([0.5, 0.5])
        ca, sa = np.cos(angle), np.sin(angle)
        pts = ((pts - c) @ np.array([[ca, -sa], [sa, ca]]) + c)
        
    mask = (pts[:, 0] >= 0.01) & (pts[:, 0] <= 0.99) & (pts[:, 1] >= 0.01) & (pts[:, 1] <= 0.99)
    pts = pts[mask]
    while len(pts) < N_CIRCLES:
        pts = np.vstack([pts, [np.random.uniform(0.15, 0.85), np.random.uniform(0.15, 0.85)]])
    return pts[:N_CIRCLES]

def generate_corner_init(seed):
    """Generate a configuration biased towards corners and edges."""
    np.random.seed(seed)
    pts = np.array([
        [0.1, 0.1], [0.9, 0.1], [0.1, 0.9], [0.9, 0.9],
        [0.0, 0.5], [1.0, 0.5], [0.5, 0.0], [0.5, 1.0],
        [0.5, 0.5]
    ])
    while len(pts) < N_CIRCLES:
        pts = np.vstack([pts, [np.random.uniform(0.1, 0.9), np.random.uniform(0.1, 0.9)]])
    pts += np.random.normal(0, 0.02, pts.shape)
    return np.clip(pts[:N_CIRCLES], 0.05, 0.95)

def run_packing():
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    best_sum = -1.0
    best_centers = None
    best_radii = None

    # Phase 1: Diverse initial configurations
    inits = []
    for s in range(20):
        r0 = 0.085 + s * 0.001
        inits.append(generate_hex_init(r0, angle=s * 0.05, seed=s))
    for s in range(10):
        inits.append(generate_corner_init(seed=s * 10 + 7))
    for s in range(15):
        np.random.seed(s)
        inits.append(np.random.uniform(0.1, 0.9, (N_CIRCLES, 2)))

    # Phase 2: Alternating LP & Force optimization
    for c_init in inits:
        centers = c_init.copy()
        for cycle in range(25):
            radii = solve_lp_radii(centers)
            centers = force_optimize_centers(centers, radii, steps=100)

        # Phase 3: SLSQP polish
        x0 = np.zeros(3 * N_CIRCLES)
        x0[0::3] = centers[:, 0]
        x0[1::3] = centers[:, 1]
        x0[2::3] = radii

        try:
            res = minimize(sgs_objective, x0, method='SLSQP', bounds=get_bounds(),
                           constraints={'type': 'ineq', 'fun': sgs_constraints},
                           options={'maxiter': 6000, 'ftol': 1e-12, 'disp': False})
            if not np.isnan(res.fun):
                curr_sum = -res.fun
                if curr_sum > best_sum:
                    best_sum = curr_sum
                    best_centers = res.x[0::3].reshape(N_CIRCLES, 2)
                    best_radii = res.x[2::3].copy()
        except Exception:
            pass

    # Phase 4: Deflation & Perturbation to escape local minima
    if best_centers is not None:
        for i in range(15):
            x0 = np.zeros(3 * N_CIRCLES)
            x0[0::3] = best_centers[:, 0]
            x0[1::3] = best_centers[:, 1]
            x0[2::3] = best_radii * 0.985
            noise = 0.002 * (0.8 ** i)
            x0[0::3] += np.random.normal(0, noise, N_CIRCLES)
            x0[1::3] += np.random.normal(0, noise, N_CIRCLES)
            for k in range(N_CIRCLES):
                r = max(0.01, x0[3 * k + 2])
                x0[3 * k] = np.clip(x0[3 * k], r, 1.0 - r)
                x0[3 * k + 1] = np.clip(x0[3 * k + 1], r, 1.0 - r)
                x0[3 * k + 2] = r
            try:
                res = minimize(sgs_objective, x0, method='SLSQP', bounds=get_bounds(),
                               constraints={'type': 'ineq', 'fun': sgs_constraints},
                               options={'maxiter': 8000, 'ftol': 1e-12, 'disp': False})
                if not np.isnan(res.fun) and -res.fun > best_sum:
                    best_sum = -res.fun
                    best_centers = res.x[0::3].reshape(N_CIRCLES, 2)
                    best_radii = res.x[2::3].copy()
            except Exception:
                pass

    # Fallback (should not be reached)
    if best_centers is None:
        best_centers = generate_hex_init(0.09, 0.0, 0)
        best_radii = solve_lp_radii(best_centers)
        best_sum = np.sum(best_radii)

    # Phase 5: Strict validation and minimal numerical repair
    for _ in range(50):
        valid = True
        for i in range(N_CIRCLES):
            if best_radii[i] < 0:
                valid = False
                break
            if best_centers[i, 0] < best_radii[i] - 1e-9 or best_centers[i, 0] > 1.0 - best_radii[i] + 1e-9:
                valid = False
                break
            if best_centers[i, 1] < best_radii[i] - 1e-9 or best_centers[i, 1] > 1.0 - best_radii[i] + 1e-9:
                valid = False
                break
        if valid:
            for i in range(N_CIRCLES):
                for j in range(i + 1, N_CIRCLES):
                    d = np.hypot(best_centers[i, 0] - best_centers[j, 0], best_centers[i, 1] - best_centers[j, 1])
                    if d < best_radii[i] + best_radii[j] - 1e-9:
                        valid = False
                        break
                if not valid:
                    break
        if valid:
            break
        best_radii *= 0.999
        best_centers[:, 0] = np.clip(best_centers[:, 0], best_radii, 1.0 - best_radii)
        best_centers[:, 1] = np.clip(best_centers[:, 1], best_radii, 1.0 - best_radii)

    return best_centers, best_radii, float(np.sum(best_radii))
