import numpy as np
from scipy.optimize import minimize, linprog

N_CIRCLES = 26

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Compute all boundary and non-overlap constraints as a vector >= 0."""
    cx = x[0::3]
    cy = x[1::3]
    cr = x[2::3]
    
    # Boundary constraints: 4 * N
    c = np.concatenate([cx - cr, 1.0 - cx - cr, cy - cr, 1.0 - cy - cr])
    
    # Overlap constraints: N*(N-1)/2
    idx = np.tril_indices(N_CIRCLES, -1)
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dr = cr[:, None] + cr[None, :]
    c = np.concatenate([c, dx[idx]**2 + dy[idx]**2 - dr[idx]**2])
    
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [0, 0.5]."""
    return [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N_CIRCLES

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist)
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def make_hex_init(r0, angle, seed):
    """Generate a perturbed hexagonal lattice initialization."""
    np.random.seed(seed)
    n = N_CIRCLES
    centers = []
    y = r0
    row = 0
    while len(centers) < n + 5:
        x_start = r0 + (r0 if row % 2 == 1 else 0.0)
        x = x_start
        while x <= 1.0 - r0 and len(centers) < n + 5:
            centers.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        
    centers = np.array(centers[:n + 5])
    
    # Rotate around center
    if angle != 0.0:
        c = np.array([0.5, 0.5])
        ca, sa = np.cos(angle), np.sin(angle)
        centers = ((centers - c) @ np.array([[ca, -sa], [sa, ca]]) + c)
        
    # Filter points strictly inside the square
    mask = (centers[:, 0] >= 0.01) & (centers[:, 0] <= 0.99) & \
           (centers[:, 1] >= 0.01) & (centers[:, 1] <= 0.99)
    centers = centers[mask]
    
    # Pad if rotation/filtering removed points
    while len(centers) < n:
        centers = np.vstack([centers, [np.random.uniform(0.2, 0.8), np.random.uniform(0.2, 0.8)]])
    centers = centers[:n]
    
    # Perturb and ensure initial feasibility relative to r0
    centers += np.random.normal(0, 0.001, centers.shape)
    centers = np.clip(centers, r0 + 0.001, 1.0 - r0 - 0.001)
    
    x0 = np.zeros(3 * n)
    x0[0::3] = centers[:, 0]
    x0[1::3] = centers[:, 1]
    x0[2::3] = r0
    return x0

def make_force_init(seed):
    """Generate a dense initial configuration using repulsive force simulation."""
    np.random.seed(seed)
    n = N_CIRCLES
    cx = np.random.uniform(0.15, 0.85, n)
    cy = np.random.uniform(0.15, 0.85, n)
    cr = np.full(n, 0.08)
    
    for step in range(2000):
        dx = cx[:, None] - cx[None, :]
        dy = cy[:, None] - cy[None, :]
        d2 = dx**2 + dy**2
        d = np.sqrt(d2)
        
        mask = (d < 0.28) & (d > 1e-6)
        f = np.zeros_like(d2)
        f[mask] = 0.015 / (d2[mask] + 0.0005)
        
        fx = np.sum(f * dx, axis=1)
        fy = np.sum(f * dy, axis=1)
        
        # Wall repulsion
        w = 0.08
        fx += np.where(cx < cr + 0.02, w, 0.0) - np.where(cx > 1.0 - cr - 0.02, w, 0.0)
        fy += np.where(cy < cr + 0.02, w, 0.0) - np.where(cy > 1.0 - cr - 0.02, w, 0.0)
        
        alpha = 0.03 * (1.0 - step / 2000.0)
        cx += fx * alpha
        cy += fy * alpha
        
        cx = np.clip(cx, cr, 1.0 - cr)
        cy = np.clip(cy, cr, 1.0 - cr)
        
    x = np.zeros(3 * n)
    x[0::3] = cx
    x[1::3] = cy
    x[2::3] = cr
    return x

def make_grid_init(seed):
    """Generate a grid-based initialization with perturbation."""
    np.random.seed(seed)
    n = N_CIRCLES
    centers = []
    # 5 columns x 6 rows = 30, take 26
    for r in range(6):
        for c in range(5):
            if len(centers) >= n:
                break
            x = 0.08 + c * 0.18
            y = 0.08 + r * 0.16
            centers.append([x, y])
    
    centers = np.array(centers[:n])
    centers += np.random.uniform(-0.008, 0.008, centers.shape)
    centers = np.clip(centers, 0.05, 0.95)
    
    x0 = np.zeros(3 * n)
    x0[0::3] = centers[:, 0]
    x0[1::3] = centers[:, 1]
    x0[2::3] = 0.085
    return x0

def safe_projection(x0, n):
    """Project variables to strictly feasible bounds."""
    result = x0.copy()
    for i in range(n):
        r = max(0.005, result[3*i + 2])
        result[3*i] = np.clip(result[3*i], r, 1.0 - r)
        result[3*i + 1] = np.clip(result[3*i + 1], r, 1.0 - r)
        result[3*i + 2] = r
    return result

def check_validity(centers, radii, n):
    """Check if configuration is valid within tolerance."""
    for i in range(n):
        if radii[i] < 0:
            return False
        if centers[i, 0] < radii[i] - 1e-9 or centers[i, 0] > 1.0 - radii[i] + 1e-9:
            return False
        if centers[i, 1] < radii[i] - 1e-9 or centers[i, 1] > 1.0 - radii[i] + 1e-9:
            return False
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            if d < radii[i] + radii[j] - 1e-9:
                return False
    return True

def repair_config(centers, radii, n):
    """Repair invalid configuration by minimal shrinkage."""
    for _ in range(100):
        if check_validity(centers, radii, n):
            return centers, radii
        radii *= 0.999
        for i in range(n):
            centers[i, 0] = np.clip(centers[i, 0], radii[i], 1.0 - radii[i])
            centers[i, 1] = np.clip(centers[i, 1], radii[i], 1.0 - radii[i])
    return centers, radii

def run_packing():
    n = N_CIRCLES
    bnds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -1.0
    best_x = None
    
    # Phase 1: Diverse multi-start optimization
    inits = []
    
    # Varied hexagonal starts with many rotations
    for s in range(30):
        r0 = 0.085 + s * 0.0015
        angle = s * 0.06
        inits.append(make_hex_init(r0, angle=angle, seed=s * 100))
    
    # Additional hexagonal starts with larger rotations
    for angle_deg in [15, 30, 45, 60, 75]:
        angle = np.deg2rad(angle_deg)
        for s in range(5):
            r0 = 0.088 + s * 0.002
            inits.append(make_hex_init(r0, angle=angle, seed=s * 1000 + angle_deg))
    
    # Force-directed starts
    for s in range(15):
        inits.append(make_force_init(seed=s))
    
    # Grid-based starts
    for s in range(10):
        inits.append(make_grid_init(seed=s))
        
    for x0 in inits:
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bnds, constraints=cons,
                           options={'maxiter': 12000, 'ftol': 1e-13, 'disp': False})
            curr_sum = -res.fun
            c_vals = constraints(res.x)
            if np.min(c_vals) >= -1e-7 and curr_sum > best_sum:
                best_sum = curr_sum
                best_x = res.x.copy()
        except Exception:
            pass
            
    # Phase 2: LP + SLSQP alternating refinement
    if best_x is not None:
        for cycle in range(20):
            # LP refinement: compute optimal radii for current centers

            x0[0::3] += np.random.normal(0, noise_scale, n)
            x0[1::3] += np.random.normal(0, noise_scale, n)
            x0 = safe_projection(x0, n)
            
            try:

                c_vals = constraints(res.x)
