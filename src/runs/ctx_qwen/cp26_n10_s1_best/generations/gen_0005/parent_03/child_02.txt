2. **LP Radius Snap**: For any fixed set of centers, the optimal radii form a linear programming problem. Solving this exactly (`linprog`) rapidly pushes radii to their theoretical maximums for the current geometry.
3. **Deflation & Perturbation**: After LP snaps radii, slightly deflate them and add controlled noise to centers. This creates "slack" allowing SLSQP to reposition circles into denser configurations without immediately hitting constraint boundaries.
4. **Force-Relaxation Pre-conditioning**: Apply a fast repulsive force simulation to resolve severe overlaps in initial configurations before handing them to SLSQP, improving convergence stability.
5. **Strict Validation & Repair**: A final safety loop guarantees strict compliance with the `1e-12` tolerance in the validator by minimally shrinking radii and clipping centers only if necessary.

import numpy as np
from scipy.optimize import minimize, linprog

N_CIRCLES = 26

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Compute all boundary and non-overlap constraints as a vector >= 0."""
    cx = x[0::3]
    cy = x[1::3]
    cr = x[2::3]
    
    # Boundary constraints: 4 * N
    c = np.concatenate([cx - cr, 1.0 - cx - cr, cy - cr, 1.0 - cy - cr])
    
    # Overlap constraints: N*(N-1)/2
    idx = np.tril_indices(N_CIRCLES, -1)
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dr = cr[:, None] + cr[None, :]
    c = np.concatenate([c, dx[idx]**2 + dy[idx]**2 - dr[idx]**2])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [0, 0.5]."""
    return [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N_CIRCLES

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, d)
            idx += 1
            
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=[(0.0, None)] * n, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-7)
    except Exception:
        pass
    return np.full(n, 0.05)

def force_relax(centers, radii, steps=50):
    """Fast force-directed relaxation to resolve overlaps and push to boundaries."""
    pts = centers.copy()
    rs = radii.copy()
    for step in range(steps):
        fx, fy = np.zeros(N_CIRCLES), np.zeros(N_CIRCLES)
        for i in range(N_CIRCLES):
            for j in range(i + 1, N_CIRCLES):
                dx = pts[j, 0] - pts[i, 0]
                dy = pts[j, 1] - pts[i, 1]
                d = np.hypot(dx, dy)
                if d < rs[i] + rs[j] and d > 1e-6:
                    f = (rs[i] + rs[j] - d) * 2.0 / d
                    fx[i] -= f * dx
                    fy[i] -= f * dy
                    fx[j] += f * dx
                    fy[j] += f * dy
            # Wall repulsion
            if pts[i, 0] < rs[i] + 0.001: fx[i] += 0.1
            elif pts[i, 0] > 1.0 - rs[i] - 0.001: fx[i] -= 0.1
            if pts[i, 1] < rs[i] + 0.001: fy[i] += 0.1
            elif pts[i, 1] > 1.0 - rs[i] - 0.001: fy[i] -= 0.1
            
        alpha = 0.01 * (1.0 - step / steps)
        pts[:, 0] += fx * alpha
        pts[:, 1] += fy * alpha
        pts[:, 0] = np.clip(pts[:, 0], rs, 1.0 - rs)
        pts[:, 1] = np.clip(pts[:, 1], rs, 1.0 - rs)
    return pts

def make_hex_init(seed, angle, r0):
    """Generate a perturbed hexagonal lattice initialization."""
    np.random.seed(seed)
    pts = []
    y = r0
    row = 0
    while len(pts) < N_CIRCLES + 10:
        x_start = r0 if row % 2 == 0 else 2.0 * r0
        x = x_start
        while x <= 1.0 - r0 and len(pts) < N_CIRCLES + 10:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        
    pts = np.array(pts[:N_CIRCLES + 10])
    if angle != 0.0:
        c = np.array([0.5, 0.5])
        ca, sa = np.cos(angle), np.sin(angle)
        pts = ((pts - c) @ np.array([[ca, -sa], [sa, ca]])) + c
        
    mask = (pts[:, 0] >= 0.01) & (pts[:, 0] <= 0.99) & (pts[:, 1] >= 0.01) & (pts[:, 1] <= 0.99)
    pts = pts[mask]
    while len(pts) < N_CIRCLES:
        pts = np.vstack([pts, np.random.uniform(0.2, 0.8, 2)])
    pts = pts[:N_CIRCLES]
    pts += np.random.normal(0, 0.002, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    n = N_CIRCLES
    bnds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -1.0
    best_x = None
    
    # Phase 1: Diverse multi-start optimization
    starts = []
    # Varied hexagonal starts
    for s in range(25):
        r0 = 0.085 + s * 0.0015
        starts.append(make_hex_init(s, angle=s * 0.06, r0=r0))
    for s in range(15):
        starts.append(make_hex_init(s, angle=0.0, r0=0.092))
    # Random starts
    for s in range(10):
        starts.append(np.random.uniform(0.15, 0.85, (n, 2)))
        
    for pts in starts:
        r = solve_lp_radii(pts)
        x0 = np.zeros(3 * n)
        x0[0::3] = pts[:, 0]
        x0[1::3] = pts[:, 1]
        x0[2::3] = r
        
        # Ensure strict feasibility for SLSQP
        for i in range(n):
            ri = max(1e-4, x0[3*i+2])
            x0[3*i] = np.clip(x0[3*i], ri, 1.0 - ri)
            x0[3*i+1] = np.clip(x0[3*i+1], ri, 1.0 - ri)
            
        # Pre-condition with force relaxation to fix bad overlaps
        c_relaxed = force_relax(x0[0::3].reshape(n, 2), x0[2::3], steps=50)
        x0[0::3] = c_relaxed[:, 0]
        x0[1::3] = c_relaxed[:, 1]
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bnds, constraints=cons,
                           options={'maxiter': 15000, 'ftol': 1e-14, 'disp': False})
            if not np.isnan(res.fun):
                cs = constraints(res.x)
                if np.min(cs) >= -1e-7 and -res.fun > best_sum:
                    best_sum = -res.fun
                    best_x = res.x.copy()
        except Exception:
            pass
            
    # Phase 2: Alternating LP (radii) + SLSQP (centers) refinement
    if best_x is not None:
        for rnd in range(30):
            centers = np.column_stack((best_x[0::3], best_x[1::3]))
            radii = solve_lp_radii(centers)
            best_x[2::3] = radii
            
            # Deflate radii slightly to create slack for center movement
            x0 = best_x.copy()
            x0[2::3] *= 0.980
            noise = 0.0015 * np.random.randn(3 * n)
            x0 += noise
            
            # Project back to strict bounds
            for i in range(n):
                ri = max(0.005, x0[3*i+2])
                x0[3*i] = np.clip(x0[3*i], ri, 1.0 - ri)
                x0[3*i+1] = np.clip(x0[3*i+1], ri, 1.0 - ri)
                
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bnds, constraints=cons,
                               options={'maxiter': 12000, 'ftol': 1e-14, 'disp': False})
                if not np.isnan(res.fun):
                    cs = constraints(res.x)
                    if np.min(cs) >= -1e-7 and -res.fun > best_sum:
                        best_sum = -res.fun
                        best_x = res.x.copy()
            except Exception:
                pass

    # Fallback (should not be reached)
    if best_x is None:
        pts = make_hex_init(0, 0.0, 0.09)
        best_x = np.zeros(3 * n)
        best_x[0::3] = pts[:, 0]
        best_x[1::3] = pts[:, 1]
        best_x[2::3] = 0.085
        best_sum = -objective(best_x)
        
    # Extract centers and radii
    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3]
    
    # Phase 3: Strict validation and minimal numerical repair
    for _ in range(50):
        valid = True
        for i in range(n):
            if (radii[i] < 0 or centers[i, 0] < radii[i] - 1e-9 or centers[i, 0] > 1.0 - radii[i] + 1e-9 or 
                centers[i, 1] < radii[i] - 1e-9 or centers[i, 1] > 1.0 - radii[i] + 1e-9):
                valid = False
                break
        if valid:
            for i in range(n):
                for j in range(i + 1, n):
                    d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                    if d < radii[i] + radii[j] - 1e-9:
                        valid = False
                        break
                if not valid:
                    break
        if valid:
            break
            
        # Minimal shrinkage to guarantee strict compliance
        radii *= 0.9995
        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)
        
    return centers, radii, float(np.sum(radii))
