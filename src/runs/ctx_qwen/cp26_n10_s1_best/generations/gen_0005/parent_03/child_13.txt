import numpy as np
from scipy.optimize import minimize, linprog, differential_evolution

N_CIRCLES = 26

def solve_lp_radii(centers):
    """Compute optimal radii for fixed centers using Linear Programming."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], 
                  centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], 
                            centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist)
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return np.full(n, 0.05)

def de_objective(x_flat):
    """Objective for Differential Evolution: maximize LP-computed sum of radii."""
    centers = x_flat.reshape(-1, 2)
    radii = solve_lp_radii(centers)
    return -np.sum(radii)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Compute all boundary and non-overlap constraints as a vector >= 0."""
    cx = x[0::3]
    cy = x[1::3]
    cr = x[2::3]
    
    c = np.concatenate([cx - cr, 1.0 - cx - cr, cy - cr, 1.0 - cy - cr])
    
    idx = np.tril_indices(N_CIRCLES, -1)
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dr = cr[:, None] + cr[None, :]
    c = np.concatenate([c, dx[idx]**2 + dy[idx]**2 - dr[idx]**2])
    
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [0, 0.5]."""
    return [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N_CIRCLES

def make_hex_init(r0, angle, seed):
    """Generate a perturbed hexagonal lattice initialization."""
    np.random.seed(seed)
    pts = []
    y = r0
    row = 0
    while len(pts) < N_CIRCLES + 10:
        x_start = r0 + (r0 if row % 2 == 1 else 0.0)
        x = x_start
        while x <= 1.0 - r0 and len(pts) < N_CIRCLES + 10:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        
    pts = np.array(pts[:N_CIRCLES + 10])
    
    if angle != 0.0:
        c = np.array([0.5, 0.5])
        ca, sa = np.cos(angle), np.sin(angle)
        pts = ((pts - c) @ np.array([[ca, -sa], [sa, ca]])) + c
        
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & \
           (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    
    while len(pts) < N_CIRCLES:
        pts = np.vstack([pts, [np.random.uniform(0.2, 0.8), np.random.uniform(0.2, 0.8)]])
    pts = pts[:N_CIRCLES]
    
    pts += np.random.normal(0, 0.001, pts.shape)
    return pts

def run_packing():
    np.random.seed(42)
    n = N_CIRCLES
    bnds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -1.0
    best_x = None
    
    # Phase 1: Global Search via Differential Evolution on centers
    # LP handles radius constraints automatically, creating a smoother objective landscape
    try:
        de_bounds = [(0.02, 0.98)] * (2 * n)
        de_res = differential_evolution(de_objective, de_bounds, popsize=10, maxiter=400, 
                                        seed=42, polish=False, workers=1)
        best_centers_de = de_res.x.reshape(n, 2)
        best_radii_de = solve_lp_radii(best_centers_de)
    except Exception:
        best_centers_de = np.random.uniform(0.15, 0.85, (n, 2))
        best_radii_de = np.full(n, 0.07)
        
    # Phase 2: Diverse Hexagonal Starts
    inits = []
    for s in range(15):
        r0 = 0.085 + s * 0.0025
        pts = make_hex_init(r0, angle=s * 0.06, seed=s)
        r_lp = solve_lp_radii(pts)
        inits.append((pts, r_lp))
        
    # Add DE result to initializations
    inits.append((best_centers_de, best_radii_de))
    
    for c_init, r_init in inits:
        x0 = np.zeros(3 * n)
        x0[0::3] = c_init[:, 0]
        x0[1::3] = c_init[:, 1]
        x0[2::3] = r_init
        
        # Project to strict bounds for SLSQP stability
        for i in range(n):
            r = max(1e-6, x0[3*i + 2])
            x0[3*i] = np.clip(x0[3*i], r, 1.0 - r)
            x0[3*i + 1] = np.clip(x0[3*i + 1], r, 1.0 - r)
            x0[3*i + 2] = r
            
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bnds, constraints=cons,
                           options={'maxiter': 20000, 'ftol': 1e-14, 'disp': False})
            if not np.isnan(res.fun):
                curr_sum = -res.fun
                c_vals = constraints(res.x)
                if np.min(c_vals) >= -1e-7 and curr_sum > best_sum:
                    best_sum = curr_sum
                    best_x = res.x.copy()
        except Exception:
            continue
            
    # Phase 3: Deflation-Perturbation Refinement to escape local minima
    if best_x is not None:
        for step in range(25):
            x0 = best_x.copy()
            # Shrink radii to allow centers to move
            x0[2::3] *= 0.975
            
            # Perturb centers
            noise_scale = 0.0025 * (0.85 ** step)
            x0[0::3] += np.random.normal(0, noise_scale, n)
            x0[1::3] += np.random.normal(0, noise_scale, n)
            
            for i in range(n):
                r = max(1e-5, x0[3*i + 2])
                x0[3*i] = np.clip(x0[3*i], r, 1.0 - r)
                x0[3*i + 1] = np.clip(x0[3*i + 1], r, 1.0 - r)
                x0[3*i + 2] = r
                
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bnds, constraints=cons,
                               options={'maxiter': 15000, 'ftol': 1e-14, 'disp': False})
                if not np.isnan(res.fun):
                    curr_sum = -res.fun
                    c_vals = constraints(res.x)
                    if np.min(c_vals) >= -1e-7 and curr_sum > best_sum:
                        best_sum = curr_sum
                        best_x = res.x.copy()
            except Exception:
                pass
                
            # LP Refinement on current best centers
            centers_cur = best_x.reshape(n, 3)[:, :2]
            r_lp = solve_lp_radii(centers_cur)
            if np.sum(r_lp) > best_sum - 1e-8:
                best_x[2::3] = r_lp
                best_sum = np.sum(r_lp)
                
    # Extract results
    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3].copy()
    
    # Phase 4: Strict validity check and minimal numerical repair
    for _ in range(100):
        valid = True
        for i in range(n):
            if (radii[i] < 0 or 
                centers[i, 0] < radii[i] - 1e-10 or centers[i, 0] > 1.0 - radii[i] + 1e-10 or 
                centers[i, 1] < radii[i] - 1e-10 or centers[i, 1] > 1.0 - radii[i] + 1e-10):
                valid = False
                break
        if valid:
            for i in range(n):
                for j in range(i + 1, n):
                    d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                    if d < radii[i] + radii[j] - 1e-10:
                        valid = False
                        break
                if not valid:
                    break
                    
        if valid:
            break
            
        # Minimal shrinkage to guarantee strict compliance
        radii *= 0.999
        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)
        
    return centers, radii, float(np.sum(radii))
