import numpy as np
from scipy.optimize import minimize, linprog, basinhopping

N = 26

def get_bounds():
    """Variable bounds: x,y in [0.02, 0.98], r in [0, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.02, 0.98), (0.02, 0.98), (0.0, 0.5)])
    return b

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist)
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.fun * -1, res.x
    except Exception:
        pass
    return 0.0, np.zeros(n)

def lp_objective(centers_flat):
    """Objective for center optimization: maximize LP radii sum."""
    val, _ = solve_lp_radii(centers_flat.reshape(N, 2))
    return -val

def make_hex_centers(r0, angle_deg=0.0):
    """Generate a rotated hexagonal lattice initialization."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 20:
        x = (row % 2) * r0
        while x <= 1.0 - r0:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        
    pts = np.array(pts[:N + 20])
    center = np.array([0.5, 0.5])
    pts -= center
    ang = np.deg2rad(angle_deg)
    rot = np.array([[np.cos(ang), -np.sin(ang)], [np.sin(ang), np.cos(ang)]])
    pts = pts @ rot.T + center
    
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    if len(pts) < N:
        pad = N - len(pts)
        pts = np.vstack([pts, np.random.uniform(0.2, 0.8, (pad, 2))])
    return pts[:N]

def force_spread_centers(centers, steps=300):
    """Force-directed layout to push points apart and towards boundaries."""
    pts = centers.copy()
    for step in range(steps):
        f = np.zeros_like(pts)
        for i in range(N):
            for j in range(i + 1, N):
                dx = pts[j] - pts[i]
                d = np.hypot(dx[0], dx[1])
                if d < 0.35 and d > 1e-6:
                    rep = 0.012 / (d**2 + 1e-4)
                    f[i] -= rep * dx
                    f[j] += rep * dx
            for dim in range(2):
                if pts[i, dim] < 0.06: f[i, dim] += 0.06
                elif pts[i, dim] > 0.94: f[i, dim] -= 0.06
        lr = 0.04 * (1.0 - step / steps)
        pts += f * lr
        pts = np.clip(pts, 0.02, 0.98)
    return pts

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    
    # Phase 1: Optimize centers by maximizing the LP radii sum
    best_lp_sum = 0.0
    best_centers_flat = None
    
    # Diverse starting configurations
    starts = []
    for ang in [0, 10, 20, 30, 40, 50, 60]:
        starts.append(make_hex_centers(0.09, ang).flatten())
    for _ in range(6):
        starts.append(force_spread_centers(np.random.uniform(0.15, 0.85, (N, 2))).flatten())
        
    min_kwargs = {'method': 'Nelder-Mead', 'options': {'maxiter': 2500, 'fatol': 1e-11}}
    
    for x0 in starts:
        try:
            bp = basinhopping(lp_objective, x0, niter=40, minimizer_kwargs=min_kwargs, 
                              stepsize=0.015, interval=50, seed=42)
            if -bp.fun > best_lp_sum:
                best_lp_sum = -bp.fun
                best_centers_flat = bp.x.copy()
        except Exception:
            continue
            
    centers = best_centers_flat.reshape(N, 2)
    _, radii = solve_lp_radii(centers)
    
    # Phase 2: Joint optimization with SLSQP
    def objective_joint(x):
        return -np.sum(x[2::3])

    def constraints_joint(x):
        cx = x[0::3]
        cy = x[1::3]
        cr = x[2::3]
        c = np.concatenate([cx - cr, 1.0 - cx - cr, cy - cr, 1.0 - cy - cr])
        dx = cx[:, None] - cx[None, :]
        dy = cy[:, None] - cy[None, :]
        dr = cr[:, None] + cr[None, :]
        i_idx, j_idx = np.tril_indices(N, -1)
        c = np.concatenate([c, dx[i_idx, j_idx]**2 + dy[i_idx, j_idx]**2 - dr[i_idx, j_idx]**2])
        return c

    bounds_joint = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints_joint}
    
    x0_joint = np.zeros(3 * N)
    x0_joint[0::3] = centers[:, 0]
    x0_joint[1::3] = centers[:, 1]
    x0_joint[2::3] = radii * 0.98
    
    best_x = None
    best_sum = -1.0
    
    # Run SLSQP with perturbations to escape local minima
    for trial in range(12):
        curr_x0 = x0_joint.copy()
        if trial > 0:
            noise = 0.0025 * (1.0 - trial / 15.0)
            curr_x0[0::3] += np.random.normal(0, noise, N)
            curr_x0[1::3] += np.random.normal(0, noise, N)
            curr_x0[2::3] *= 0.96
            
        # Project to strict bounds
        for i in range(N):
            r = max(0.005, curr_x0[3*i+2])
            curr_x0[3*i] = np.clip(curr_x0[3*i], r, 1.0-r)
            curr_x0[3*i+1] = np.clip(curr_x0[3*i+1], r, 1.0-r)
            curr_x0[3*i+2] = r
            
        try:
            res = minimize(objective_joint, curr_x0, method='SLSQP', bounds=bounds_joint,
                           constraints=cons, options={'maxiter': 15000, 'ftol': 1e-13, 'disp': False})
            if not np.isnan(res.fun):
                curr_sum = -res.fun
                c_vals = constraints_joint(res.x)
                if np.min(c_vals) >= -1e-5 and curr_sum > best_sum:
                    best_sum = curr_sum
                    best_x = res.x.copy()
                    
                    # LP refinement on new centers to snap radii to theoretical max
                    c_mat = np.column_stack((res.x[0::3], res.x[1::3]))
                    lp_sum, lp_rad = solve_lp_radii(c_mat)
                    if lp_sum > best_sum:
                        best_x[2::3] = lp_rad
                        best_sum = lp_sum
        except Exception:
            pass
            
    # Fallback if optimization fails
    if best_x is None:
        best_x = x0_joint.copy()
        best_sum = np.sum(radii)
        
    centers_opt = best_x[0::3]
    cy_opt = best_x[1::3]
    radii_opt = best_x[2::3]
    centers_mat = np.column_stack((centers_opt, cy_opt))
    
    # Phase 3: Strict Validation & Minimal Numerical Repair
    valid = True
    for i in range(N):
        if radii_opt[i] < 0: valid = False; break
        if centers_mat[i, 0] < radii_opt[i] - 1e-9 or centers_mat[i, 0] > 1.0 - radii_opt[i] + 1e-9: valid = False; break
        if centers_mat[i, 1] < radii_opt[i] - 1e-9 or centers_mat[i, 1] > 1.0 - radii_opt[i] + 1e-9: valid = False; break
    if valid:
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers_mat[i, 0] - centers_mat[j, 0], centers_mat[i, 1] - centers_mat[j, 1])
                if d < radii_opt[i] + radii_opt[j] - 1e-9: valid = False; break
            if not valid: break
            
    if not valid:
        for _ in range(100):
            radii_opt *= 0.999
            centers_mat[:, 0] = np.clip(centers_mat[:, 0], radii_opt, 1.0 - radii_opt)
            centers_mat[:, 1] = np.clip(centers_mat[:, 1], radii_opt, 1.0 - radii_opt)
            is_ok = True
            for i in range(N):
                for j in range(i + 1, N):
                    d = np.hypot(centers_mat[i, 0] - centers_mat[j, 0], centers_mat[i, 1] - centers_mat[j, 1])
                    if d < radii_opt[i] + radii_opt[j] - 1e-9:
                        is_ok = False; break
                if not is_ok: break
            if is_ok: break
            
    return centers_mat, radii_opt, float(np.sum(radii_opt))
