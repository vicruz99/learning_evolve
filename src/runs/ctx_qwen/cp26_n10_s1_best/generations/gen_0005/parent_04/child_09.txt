import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    
    num_ineq = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    for i in range(n):
        # r_i <= x_i
        A_ub[idx, i] = 1.0; b_ub[idx] = centers[i, 0]; idx += 1
        # r_i <= 1 - x_i
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - centers[i, 0]; idx += 1
        # r_i <= y_i
        A_ub[idx, i] = 1.0; b_ub[idx] = centers[i, 1]; idx += 1
        # r_i <= 1 - y_i
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - centers[i, 1]; idx += 1
        
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dist
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return np.full(n, 1e-5)

def lp_objective(centers_flat):
    """Objective for center-only optimization: minimize negative sum of LP radii."""
    centers = centers_flat.reshape(-1, 2)
    radii = solve_lp_radii(centers)
    return -np.sum(radii)

def make_hex_init(angle_deg=0.0, r0=0.095):
    """Generate a rotated hexagonal lattice initialization."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 20:
        x = (row % 2) * r0
        while x <= 1.0 - r0:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        
    pts = np.array(pts[:N + 20])
    center = np.array([0.5, 0.5])
    pts -= center
    ang = np.deg2rad(angle_deg)
    rot = np.array([[np.cos(ang), -np.sin(ang)], [np.sin(ang), np.cos(ang)]])
    pts = pts @ rot.T + center
    
    mask = (pts[:, 0] >= 0.01) & (pts[:, 0] <= 0.99) & (pts[:, 1] >= 0.01) & (pts[:, 1] <= 0.99)
    pts = pts[mask]
    if len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.2, 0.8, (N - len(pts), 2))])
    pts = pts[:N]
    return pts

def force_spread_init(seed):
    """Generate initial configuration using repulsive force simulation."""
    np.random.seed(seed)
    pts = np.random.uniform(0.15, 0.85, (N, 2))
    for _ in range(150):
        f = np.zeros_like(pts)
        for i in range(N):
            for j in range(i + 1, N):
                dx = pts[j, 0] - pts[i, 0]
                dy = pts[j, 1] - pts[i, 1]
                d = np.hypot(dx, dy)
                if d < 0.3 and d > 1e-6:
                    ff = 0.015 / (d**2 + 0.001)
                    f[i] -= ff * [dx, dy]
                    f[j] += ff * [dx, dy]
        pts += f * 0.05
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    
    best_centers = None
    best_radii = None
    best_sum = -1.0
    
    # Generate diverse initial configurations
    starts = []
    for ang in np.linspace(-60, 60, 11):
        starts.append(make_hex_init(ang, 0.095))
        starts.append(make_hex_init(ang, 0.105))
        
    for s in range(5):
        starts.append(force_spread_init(s))
        
    for pts in starts:
        # Phase 1: Optimize centers using LP objective (derivative-free)
        try:
            res = minimize(lp_objective, pts.flatten(), method='Powell', 
                           options={'maxiter': 250, 'xtol': 1e-8, 'ftol': 1e-10})
            curr_centers = res.x.reshape(-1, 2)
            curr_radii = solve_lp_radii(curr_centers)
            curr_sum = np.sum(curr_radii)
            
            if curr_sum > best_sum:
                best_sum = curr_sum
                best_centers = curr_centers.copy()
                best_radii = curr_radii.copy()
        except Exception:
            continue
            
        # Phase 2: Refine with SLSQP on full variables
        x0 = np.zeros(3 * N)
        x0[0::3] = best_centers[:, 0]
        x0[1::3] = best_centers[:, 1]
        x0[2::3] = best_radii * 0.995 
        
        def obj_full(x): 
            return -np.sum(x[2::3])
        
        def cons_full(x):
            cx, cy, cr = x[0::3], x[1::3], x[2::3]
            dx = cx[:, None] - cx[None, :]
            dy = cy[:, None] - cy[None, :]
            dr = cr[:, None] + cr[None, :]
            i, j = np.tril_indices(N, -1)
            overlaps = dx[i,j]**2 + dy[i,j]**2 - dr[i,j]**2
            bounds_c = np.concatenate([cx - cr, 1.0 - cx - cr, cy - cr, 1.0 - cy - cr])
            return np.concatenate([bounds_c, overlaps])
            
        bnds = [(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)] * N
        
        try:
            res = minimize(obj_full, x0, method='SLSQP', bounds=bnds,
                           constraints={'type': 'ineq', 'fun': cons_full},
                           options={'maxiter': 5000, 'ftol': 1e-12, 'disp': False})

                if np.min(c_vals) >= -1e-6:
                    curr_sum = -res.fun
                    if curr_sum > best_sum:
                        best_centers = np.column_stack((res.x[0::3], res.x[1::3]))
                        best_radii = res.x[2::3]
                        best_sum = curr_sum
        except Exception:
            pass

    # Fallback initialization (should not be reached with valid inits)
    if best_centers is None:
        best_centers = make_hex_init(0.0, 0.095)
        best_radii = solve_lp_radii(best_centers)
        best_sum = np.sum(best_radii)

    # Final safety & formatting
    radii = best_radii.copy()
    centers = best_centers.copy()

        valid = True
        for i in range(N):
            if radii[i] < 0: valid = False; break
            if centers[i,0] < radii[i] - 1e-9 or centers[i,0] > 1 - radii[i] + 1e-9: valid = False; break
            if centers[i,1] < radii[i] - 1e-9 or centers[i,1] > 1 - radii[i] + 1e-9: valid = False; break
        if valid:
            for i in range(N):
                for j in range(i+1, N):
                    d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
                    if d < radii[i] + radii[j] - 1e-9: valid = False; break
                if not valid: break
        if valid: break
        radii *= 0.9995
        centers[:,0] = np.clip(centers[:,0], radii, 1-radii)
        centers[:,1] = np.clip(centers[:,1], radii, 1-radii)
        
    return centers, radii, float(np.sum(radii))
