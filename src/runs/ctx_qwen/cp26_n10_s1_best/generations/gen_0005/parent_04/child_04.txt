import numpy as np
from scipy.optimize import linprog, minimize

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-6, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)])
    return b

def objective_slsqp(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints_slsqp(x):
    """Vectorized inequality constraints for SLSQP: g(x) >= 0."""
    cx = x[0::3]
    cy = x[1::3]
    cr = x[2::3]
    
    # Boundary constraints
    c = np.concatenate([cx - cr, 1.0 - cx - cr, cy - cr, 1.0 - cy - cr])
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dr = cr[:, None] + cr[None, :]
    
    c = np.concatenate([c, dx[TRIL_IDX]**2 + dy[TRIL_IDX]**2 - dr[TRIL_IDX]**2])
    return c

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist)
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def force_resolve(centers, radii, steps=100, dt=0.05):
    """Force-directed layout to resolve overlaps and push to boundaries."""
    pts = centers.copy()
    n = len(pts)
    pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
    
    for step in range(steps):
        forces = np.zeros_like(pts)
        for i, j in pairs:
            dx = pts[j, 0] - pts[i, 0]
            dy = pts[j, 1] - pts[i, 1]
            d = np.hypot(dx, dy)
            if d < 1e-7:
                dx, dy = 1e-4, 0.0
                d = 1e-4
                
            min_d = radii[i] + radii[j]
            if d < min_d:
                overlap = min_d - d
                f = overlap * 0.5 / d
                forces[i, 0] -= f * dx
                forces[i, 1] -= f * dy
                forces[j, 0] += f * dx
                forces[j, 1] += f * dy
                
        # Boundary forces
        for i in range(n):
            for k in range(2):
                if pts[i, k] < radii[i]:
                    forces[i, k] += (radii[i] - pts[i, k]) * 3.0
                elif pts[i, k] > 1.0 - radii[i]:
                    forces[i, k] -= (pts[i, k] - (1.0 - radii[i])) * 3.0
                    
        # Damping
        damping = 1.0 - step / (steps + 1)
        pts += forces * dt * damping
        pts = np.clip(pts, 1e-5, 1.0 - 1e-5)
        
    return pts

def make_hex_init(r0, angle_deg=0.0):
    """Generate a rotated hexagonal lattice initialization."""
    pts = []
    y = r0
    row = 0
    # Generate slightly more than N points to allow rotation/filtering loss
    while len(pts) < N + 20:
        x = (row % 2) * r0
        while x <= 1.0 - r0:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        
    pts = np.array(pts[:N + 20])
    
    # Rotate and center around (0.5, 0.5)
    center = np.array([0.5, 0.5])
    pts -= center
    ang = np.deg2rad(angle_deg)
    rot = np.array([[np.cos(ang), -np.sin(ang)], [np.sin(ang), np.cos(ang)]])
    pts = pts @ rot.T + center
    
    # Filter points strictly inside the square
    mask = (pts[:, 0] >= 0.01) & (pts[:, 0] <= 0.99) & (pts[:, 1] >= 0.01) & (pts[:, 1] <= 0.99)
    pts = pts[mask]
    
    # Pad if rotation/filtering removed too many points
    if len(pts) < N:
        pad = N - len(pts)
        pts = np.vstack([pts, np.random.uniform(0.2, 0.8, (pad, 2))])
    pts = pts[:N]
    
    # Add small perturbation to break symmetry
    pts += np.random.uniform(-0.002, 0.002, pts.shape)
    return pts

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Phase 1: Generate diverse initial configurations
    inits = []
    for ang in np.linspace(-30, 30, 13):
        for r0 in [0.09, 0.095, 0.10, 0.105]:
            inits.append(make_hex_init(r0, ang))
            
    for _ in range(10):
        pts = np.random.uniform(0.15, 0.85, (N, 2))
        inits.append(pts)
        
    # Phase 2: Alternating optimization (Inflate -> Force Resolve -> LP)
    for c_init in inits:
        # Initial radii estimation
        r_init = np.full(N, 0.08)
        c_curr = force_resolve(c_init, r_init, steps=100)
        r_curr = solve_lp_radii(c_curr)
        if r_curr is None:
            continue
            
        # Alternating loop to push radii up
        for iter_count in range(25):
            # Perturb slightly to escape local minima
            if iter_count % 5 == 0:
                c_curr += np.random.normal(0, 0.0015, c_curr.shape)
                c_curr = np.clip(c_curr, 0.01, 0.99)
                
            # Inflate radii to push boundaries
            r_curr *= 1.0025
            # Resolve center positions to accommodate inflated radii
            c_curr = force_resolve(c_curr, r_curr, steps=80, dt=0.06)
            # Recompute optimal radii for new centers
            r_new = solve_lp_radii(c_curr)
            if r_new is not None:
                r_curr = r_new
                
        if np.sum(r_curr) > best_sum:
            best_sum = np.sum(r_curr)
            best_centers = c_curr.copy()
            best_radii = r_curr.copy()
            
    # Phase 3: Final polish with SLSQP to handle tight constraints
    if best_centers is not None:
        x0 = np.zeros(3 * N)
        x0[0::3] = best_centers[:, 0]
        x0[1::3] = best_centers[:, 1]
        x0[2::3] = best_radii
        
        try:
            res = minimize(objective_slsqp, x0, method='SLSQP', bounds=get_bounds(),
                           constraints={'type': 'ineq', 'fun': constraints_slsqp},
                           options={'maxiter': 8000, 'ftol': 1e-12, 'disp': False})
            if not np.isnan(res.fun):
                c_val = constraints_slsqp(res.x)
                if np.min(c_val) >= -1e-7:
                    curr_sum = -res.fun
                    if curr_sum > best_sum:
                        best_sum = curr_sum
                        best_centers = np.column_stack((res.x[0::3], res.x[1::3]))
                        best_radii = res.x[2::3]
        except Exception:
            pass
            
    # Phase 4: Strict validation & minimal numerical repair
    radii = best_radii.copy()
    centers = best_centers.copy()
    for _ in range(100):
        valid = True
        for i in range(N):
            if radii[i] < 0 or centers[i,0] < radii[i]-1e-10 or centers[i,0] > 1.0-radii[i]+1e-10 or \
               centers[i,1] < radii[i]-1e-10 or centers[i,1] > 1.0-radii[i]+1e-10:
                valid = False; break
        if valid:
            for i in range(N):
                for j in range(i+1, N):
                    d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
                    if d < radii[i]+radii[j]-1e-10:
                        valid = False; break
                if not valid: break
        if valid: break
            
        # Gentle shrinkage to guarantee strict compliance without sacrificing much sum
        radii *= 0.9995
        centers[:,0] = np.clip(centers[:,0], radii, 1.0-radii)
        centers[:,1] = np.clip(centers[:,1], radii, 1.0-radii)
        
    return centers, radii, float(np.sum(radii))
