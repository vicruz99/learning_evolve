import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Compute all boundary and non-overlap constraints as a vector >= 0."""
    cx = x[0::3]
    cy = x[1::3]
    cr = x[2::3]
    
    # Boundary constraints: 4 * N
    c = np.concatenate([cx - cr, 1.0 - cx - cr, cy - cr, 1.0 - cy - cr])
    
    # Overlap constraints: N*(N-1)/2 using squared distances
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dr = cr[:, None] + cr[None, :]
    
    idx = np.tril_indices(N, -1)
    c = np.concatenate([c, dx[idx]**2 + dy[idx]**2 - dr[idx]**2])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-7, 0.5]."""
    return [(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)] * N

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist)
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def project_to_feasible(x):
    """Project variables to strictly satisfy bounds."""
    cx, cy, cr = x[0::3].copy(), x[1::3].copy(), x[2::3].copy()
    for i in range(N):
        r = max(1e-7, cr[i])
        cx[i] = np.clip(cx[i], r, 1.0 - r)
        cy[i] = np.clip(cy[i], r, 1.0 - r)
        cr[i] = r
    x[0::3], x[1::3], x[2::3] = cx, cy, cr
    return x

def make_hex_init(rows_cfg, r0, angle_deg=0.0):
    """Generate a rotated hexagonal lattice initialization based on row configuration."""
    pts = []
    y = r0
    for i, count in enumerate(rows_cfg):
        row_width = (count - 1) * 2.0 * r0
        x_offset = (1.0 - row_width) / 2.0
        for j in range(count):
            x = x_offset + j * 2.0 * r0
            pts.append([x, y])
        y += np.sqrt(3.0) * r0
        
    pts = np.array(pts)
    if angle_deg != 0.0:
        ang = np.deg2rad(angle_deg)
        c, s = np.cos(ang), np.sin(ang)
        center = np.array([0.5, 0.5])
        pts = (pts - center) @ np.array([[c, -s], [s, c]]) + center
    return pts[:N]

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

        [5, 5, 5, 5, 5, 1], [1, 5, 5, 5, 5, 5], [4, 5, 5, 5, 5, 2], [2, 5, 5, 5, 5, 4]
    ]
    
    inits = []
    for topo in topologies:
        if sum(topo) != N: 
            continue
        for ang in np.linspace(-15, 15, 13):

                inits.append(make_hex_init(topo, r0, ang))

        try:

        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)
        
    return centers, radii, float(np.sum(radii))
