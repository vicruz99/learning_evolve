This turns the problem into a black-box optimization over centers only, which SA handles very robustly.
2. **LP Radii Evaluation**: For each center configuration evaluated by SA, `linprog` solves a small LP (26 variables, ~351 constraints) to get the exact maximum radii sum. This is computationally cheap and avoids the gradient inaccuracies of SLSQP's squared-distance constraints.
3. **Diverse Initializations**: I will seed SA with rotated hexagonal lattices (varying angle and scale) and force-directed layouts to ensure coverage of the configuration space.
4. **Numerical Safety**: All constraints in the LP will include a small `1e-9` slack to guarantee strict compliance with the validator's `1e-12` tolerance. A final repair loop will clip any residual drift.

import numpy as np
from scipy.optimize import linprog

N = 26

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim - 1e-9)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist - 1e-9)
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def make_hex_init(seed, angle, scale):
    """Generate a rotated and scaled hexagonal lattice initialization."""
    np.random.seed(seed)
    pts = []
    r = 0.1 * scale
    y = r
    row = 0
    while len(pts) < N + 10:
        x = r if row % 2 == 0 else 2.0 * r
        while x <= 1.0 + r:
            pts.append([x, y])
            x += 2.0 * r
        y += np.sqrt(3.0) * r
        row += 1
    pts = np.array(pts[:N + 10])
    
    center = np.array([0.5, 0.5])
    pts -= center
    rot = np.array([[np.cos(angle), -np.sin(angle)], [np.sin(angle), np.cos(angle)]])
    pts = pts @ rot.T + center
    
    mask = (pts[:, 0] >= 0.05) & (pts[:, 0] <= 0.95) & (pts[:, 1] >= 0.05) & (pts[:, 1] <= 0.95)
    pts = pts[mask]
    if len(pts) < N:
        pad = N - len(pts)
        pts = np.vstack([pts, np.random.uniform(0.1, 0.9, (pad, 2))])
    pts = pts[:N]
    pts += np.random.uniform(-0.002, 0.002, pts.shape)
    return np.clip(pts, 0.02, 0.98)

def make_force_init(seed):
    """Generate initial configuration using repulsive force simulation."""
    np.random.seed(seed)
    pts = np.random.uniform(0.15, 0.85, (N, 2))
    for _ in range(150):
        f = np.zeros_like(pts)
        for i in range(N):
            for j in range(i + 1, N):
                dx = pts[j] - pts[i]
                d = np.hypot(dx[0], dx[1])
                if d < 0.25 and d > 1e-6:
                    ff = 0.01 / (d**2 + 0.001)
                    f[i] -= ff * dx
                    f[j] += ff * dx
            if pts[i, 0] < 0.1: f[i, 0] += 0.02
            elif pts[i, 0] > 0.9: f[i, 0] -= 0.02
            if pts[i, 1] < 0.1: f[i, 1] += 0.02
            elif pts[i, 1] > 0.9: f[i, 1] -= 0.02
        pts += f * 0.05
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def run_sa(centers, steps=1200):
    """Simulated Annealing on centers, using LP for exact radii evaluation."""
    n = centers.shape[0]
    best_c = centers.copy()
    r = solve_lp_radii(best_c)
    if r is None:
        r = np.full(n, 0.05)
        for i in range(n):
            best_c[i, 0] = np.clip(best_c[i, 0], r[i], 1-r[i])
            best_c[i, 1] = np.clip(best_c[i, 1], r[i], 1-r[i])
    best_sum = np.sum(r)
    
    curr_c = best_c.copy()
    curr_sum = best_sum
    
    for t in range(steps):
        temp = 0.02 * (0.0001 / 0.02) ** (t / max(steps - 1, 1))
        temp = max(temp, 1e-5)
        
        # Occasionally shift multiple circles to escape local minima
        if np.random.rand() < 0.15:
            idxs = np.random.choice(n, 3, replace=False)
        else:
            idxs = [np.random.randint(n)]
            
        new_c = curr_c.copy()
        for idx in idxs:
            shift = np.random.normal(0, temp, 2)
            new_c[idx] += shift
            new_c[idx] = np.clip(new_c[idx], 0.02, 0.98)
            
        r_new = solve_lp_radii(new_c)
        if r_new is not None:
            new_sum = np.sum(r_new)
            delta = new_sum - curr_sum
            if delta > 0 or np.random.rand() < np.exp(delta / (temp + 1e-9)):
                curr_c = new_c
                curr_sum = new_sum
                if curr_sum > best_sum:
                    best_sum = curr_sum
                    best_c = curr_c.copy()
    return best_c, best_sum

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    inits = []
    # Diverse hexagonal lattices
    for seed in range(2):
        for ang in [0.0, 0.15, 0.3, 0.45]:
            for sc in [0.9, 1.05]:
                inits.append(make_hex_init(seed, ang, sc))
    # Force-directed layouts
    for seed in range(3):
        inits.append(make_force_init(seed))
        
    best_centers = None
    best_sum = -1.0
    
    # Run SA on each initialization
    for c_init in inits:
        c_opt, s_opt = run_sa(c_init, steps=1200)
        if s_opt > best_sum:
            best_sum = s_opt
            best_centers = c_opt
            
    # Final exact LP radii computation
    radii = solve_lp_radii(best_centers)
    if radii is None:
        radii = np.full(N, 0.08)
        best_centers[:, 0] = np.clip(best_centers[:, 0], radii, 1-radii)
        best_centers[:, 1] = np.clip(best_centers[:, 1], radii, 1-radii)
        
    # Final strict validation & minimal numerical repair
    for _ in range(100):
        valid = True
        for i in range(N):
            if radii[i] < 0 or best_centers[i, 0] < radii[i] - 1e-10 or best_centers[i, 0] > 1.0 - radii[i] + 1e-10 or \
               best_centers[i, 1] < radii[i] - 1e-10 or best_centers[i, 1] > 1.0 - radii[i] + 1e-10:
                valid = False
                break
        if valid:
            for i in range(N):
                for j in range(i + 1, N):
                    if np.hypot(best_centers[i, 0] - best_centers[j, 0], best_centers[i, 1] - best_centers[j, 1]) < radii[i] + radii[j] - 1e-10:
                        valid = False
                        break
                if not valid:
                    break
        if valid:
            break
        radii *= 0.9995
        best_centers[:, 0] = np.clip(best_centers[:, 0], radii, 1.0 - radii)
        best_centers[:, 1] = np.clip(best_centers[:, 1], radii, 1.0 - radii)
        
    return best_centers, radii, float(np.sum(radii))
