N = 26
TRIL_IDX = np.tril_indices(N, -1)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Compute all boundary and non-overlap constraints as a vector >= 0."""
    cx = x[0::3]
    cy = x[1::3]
    cr = x[2::3]
    
    c = np.concatenate([cx - cr, 1.0 - cx - cr, cy - cr, 1.0 - cy - cr])
    
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dr = cr[:, None] + cr[None, :]
    
    c = np.concatenate([c, dx[TRIL_IDX]**2 + dy[TRIL_IDX]**2 - dr[TRIL_IDX]**2])
    return c

def get_bounds():

def solve_lp_radii(centers):

    n = N
    c_obj = -np.ones(n)

    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])

            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist)
            idx += 1

        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_lp, method='highs')

    y = r0

        x = x_start
        for _ in range(cnt):
            if x + r0 <= 1.0 + 1e-12:
                pts.append([x, y])
            x += 2.0 * r0

    pts = np.clip(pts, 0.02, 0.98)
    return pts

                dx = pts[j] - pts[i]
                d = np.hypot(dx[0], dx[1])
                if d < 0.3 and d > 1e-6:

                    f[i] -= ff * dx
                    f[j] += ff * dx

                    f[i, dim] -= 0.03
        lr = 0.04 * max(0.1, 1.0 - step / steps)
        pts += f * lr
        pts = np.clip(pts, 0.02, 0.98)
    return pts

def project_feasible(x, n):
    """Project variables to feasible region."""
    for i in range(n):
        r = max(0.005, x[3 * i + 2])
        x[3 * i] = np.clip(x[3 * i], r, 1.0 - r)
        x[3 * i + 1] = np.clip(x[3 * i + 1], r, 1.0 - r)

    return x
