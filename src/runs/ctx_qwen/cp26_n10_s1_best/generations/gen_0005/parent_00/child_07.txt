from scipy.optimize import minimize, linprog

N_CIRCLES = 26

def get_bounds():
    b = []
    for _ in range(N_CIRCLES):
        b.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
    return b

def objective(vars):
    """Minimize negative sum of radii."""
    return -np.sum(vars[2::3])

def constraints(vars):
    """Compute all boundary and non-overlap constraints as a vector >= 0."""
    cx = vars[0::3]
    cy = vars[1::3]
    cr = vars[2::3]
    
    # Boundary constraints
    c = np.concatenate([cx - cr, 1.0 - cx - cr, cy - cr, 1.0 - cy - cr])
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dr = cr[:, None] + cr[None, :]
    
    i, j = np.tril_indices(N_CIRCLES, -1)
    c = np.concatenate([c, dx[i, j]**2 + dy[i, j]**2 - dr[i, j]**2])
    return c

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N_CIRCLES
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = lim
        idx += 1
        
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = d
            idx += 1
            
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=[(0.0, None)] * n, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-8)
    except Exception:
        pass
    return np.full(n, 0.05)

def make_hex_rows(rows, r0, angle=0.0):
    """Generate a hexagonal lattice based on specified row counts."""
    pts = []
    y = r0
    for i, cnt in enumerate(rows):
        x_start = r0 + (r0 if i % 2 == 1 else 0.0)
        for k in range(cnt):
            pts.append([x_start + k * 2.0 * r0, y])
        y += np.sqrt(3.0) * r0
    pts = np.array(pts[:N_CIRCLES])
    
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        center = [0.5, 0.5]
        dx = pts[:, 0] - center[0]
        dy = pts[:, 1] - center[1]
        pts[:, 0] = dx * c - dy * s + center[0]
        pts[:, 1] = dx * s + dy * c + center[1]
        
    pts += np.random.uniform(-0.005, 0.005, pts.shape)
    pts = np.clip(pts, 0.05, 0.95)
    return pts

def force_spread(centers, steps=150):
    """Simple force-directed layout to spread points evenly."""
    pts = centers.copy()
    for _ in range(steps):
        f = np.zeros_like(pts)
        for i in range(N_CIRCLES):
            for j in range(i + 1, N_CIRCLES):
                dx = pts[j, 0] - pts[i, 0]
                dy = pts[j, 1] - pts[i, 1]
                d = np.hypot(dx, dy)
                if d < 0.25 and d > 1e-6:
                    rep = 0.01 / (d**2 + 0.001)
                    fx, fy = rep * dx / d, rep * dy / d
                    f[i] -= [fx, fy]
                    f[j] += [fx, fy]
            if pts[i, 0] < 0.15: f[i, 0] += 0.02
            elif pts[i, 0] > 0.85: f[i, 0] -= 0.02
            if pts[i, 1] < 0.15: f[i, 1] += 0.02
            elif pts[i, 1] > 0.85: f[i, 1] -= 0.02
        pts += f * 0.05
        pts = np.clip(pts, 0.05, 0.95)
    return pts
