

<strategy>
The previous solutions relied on `SLSQP` to jointly optimize centers and radii, which creates a highly non-convex QCQP with ~430 constraints. This often leads to getting trapped in local minima despite multiple restarts.

To break through the 2.634 bottleneck, we decouple the problem:
1. **Inner Problem (Linear Programming):** For any fixed set of center positions, maximizing the sum of radii is a pure Linear Program. Constraints are simply `r_i + r_j <= dist(i,j)` and `r_i <= min(x, 1-x, y, 1-y)`. This can be solved optimally and instantly using `scipy.optimize.linprog`.
2. **Outer Problem (Global Optimization):** We only optimize the 52 center coordinates. The objective becomes `minimize -sum(LP_optimal_radii(centers))`. This transforms the problem into a smooth(er) landscape over `[0,1]^52`.
3. **Algorithm:** 
   - Use `Differential Evolution` for robust global search over center positions.
   - Refine the best found configuration with `Powell` (derivative-free local optimizer, excellent for piecewise-linear/non-smooth objectives like LP-value functions).
   - Supplement with structured hexagonal lattice starts to ensure we explore the topologically optimal packing arrangement.
   - This decoupling drastically improves the chance of finding the global optimum while avoiding constraint-violation pitfalls of joint non-linear solvers.
</strategy>

```python
import numpy as np
from scipy.optimize import minimize, linprog, differential_evolution
import warnings
warnings.filterwarnings('ignore')

N = 26

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist)
            idx += 1
            
    bounds_lp = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_lp, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-9)
    except Exception:
        pass
    return np.full(n, 0.01)

def objective_centers(centers_flat):
    """Objective to minimize: negative sum of optimally packed radii."""
    centers = centers_flat.reshape(N, 2)
    radii = solve_lp_radii(centers)
    return -np.sum(radii)

def generate_hex_centers(n, r0=0.09, angle=0.0):
    """Generate a rotated hexagonal lattice initialization."""
    pts = []
    y = r0
    row = 0
    while len(pts) < n + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        if y > 1.0 + r0:
            break
            
    pts = np.array(pts[:n + 10])
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts -= 0.5
        pts = pts @ np.array([[c, -s], [s, c]])
        pts += 0.5
        
    mask = (pts[:,0]>=0.0) & (pts[:,0]<=1.0) & (pts[:,1]>=0.0) & (pts[:,1]<=1.0)
    pts = pts[mask]
    while len(pts) < n:
        pts = np.vstack([pts, np.random.uniform(0.1, 0.9, (1, 2))])
    return pts[:n]

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    
    best_sum = -1.0
    best_centers = None
    
    # Phase 1: Global search via Differential Evolution over center positions
    bounds = [(0.0, 1.0)] * (2 * N)
    try:
        de_res = differential_evolution(
            objective_centers, bounds, 
            popsize=10, maxiter=600,
            mutation=(0.5, 1.5), recombination=0.7, 
            seed=42, polish=False, disp=False
        )
        # Local refinement with Powell (robust to the non-smooth LP objective)
        res = minimize(
            objective_centers, de_res.x, method='Powell',
            options={'maxiter': 5000, 'xtol': 1e-8, 'ftol': 1e-10}
        )
        if -res.fun > best_sum:
            best_sum = -res.fun
            best_centers = res.x.reshape(N, 2)
    except Exception:
        pass
        
    # Phase 2: Structured Hexagonal Starts (topological diversity)
    hex_configs = [
        (0.085, 0.0), (0.090, 0.15), (0.095, 0.30), (0.100, 0.0),
        (0.088, 0.20), (0.098, 0.10), (0.092, 0.40)
    ]
    for r0, ang in hex_configs:
        init_c = generate_hex_centers(N, r0=r0, angle=ang)
        init_c += np.random.uniform(-0.003, 0.003, init_c.shape)
        res = minimize(
            objective_centers, init_c.flatten(), method='Powell',
            options={'maxiter': 3000, 'xtol': 1e-8, 'ftol': 1e-10}
        )
        if -res.fun > best_sum:
            best_sum = -res.fun
            best_centers = res.x.reshape(N, 2)
            
    # Fallback (should not be reached)
    if best_centers is None:
        best_centers = generate_hex_centers(N, 0.09, 0.0)
        best_sum = -objective_centers(best_centers.flatten())
        
    # Phase 3: Final LP extraction and strict validation
    radii = solve_lp_radii(best_centers)
    
    # Minimal numerical repair to guarantee validator compliance
    for _ in range(50):
        valid = True
        for i in range(N):
            x, y = best_centers[i]
            r = radii[i]
            if x - r < -1e-10 or x + r > 1 + 1e-10 or y - r < -1e-10 or y + r > 1 + 1e-10:
                valid = False
                break
        if valid:
            for i in range(N):
                for j in range(i + 1, N):
                    d = np.hypot(best_centers[i,0]-best_centers[j,0], best_centers[i,1]-best_centers[j,1])
                    if d < radii[i] + radii[j] - 1e-10:
                        valid = False
                        break
                if not valid:
                    break
        if valid:
            break
            
        # Gentle shrinkage to recover strict feasibility
        radii *= 0.9995
        best_centers[:, 0] = np.clip(best_centers[:, 0], radii, 1.0 - radii)
        best_centers[:, 1] = np.clip(best_centers[:, 1], radii, 1.0 - radii)
        
    return best_centers, radii, float(np.sum(radii))
```
