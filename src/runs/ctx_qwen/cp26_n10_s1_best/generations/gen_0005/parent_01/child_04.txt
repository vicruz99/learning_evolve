import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def objective_penalty(vars, lam):
    """Penalty-based objective for circle packing."""
    xs = vars[0:N]
    ys = vars[N:2*N]
    rs = vars[2*N:3*N]
    
    # Base objective: maximize sum of radii
    base_obj = -np.sum(rs)
    
    # Overlap penalty (squared violations)
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dist = np.sqrt(dx**2 + dy**2)
    dist[np.diag_indices_from(dist)] = 1.0  # Ignore self-distance
    
    dr = rs[:, None] + rs[None, :]
    overlap = np.maximum(0.0, dr - dist)
    triu_idx = np.triu_indices(N, 1)
    overlap_pen = np.sum(overlap[triu_idx]**2)
    
    # Boundary penalty (squared violations)
    bound_pen = np.sum(np.maximum(0.0, rs - xs)**2)
    bound_pen += np.sum(np.maximum(0.0, rs - (1.0 - xs))**2)
    bound_pen += np.sum(np.maximum(0.0, rs - ys)**2)
    bound_pen += np.sum(np.maximum(0.0, rs - (1.0 - ys))**2)
    
    return base_obj + lam * (overlap_pen + bound_pen)

def get_bounds_penalty():
    """Variable bounds for penalty optimization."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
    return b

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, d)
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def generate_hex_init(r0):
    """Generates a hexagonal lattice initialization."""
    centers = np.zeros((N, 2))
    idx = 0
    y = r0
    row = 0
    while idx < N and y + r0 <= 1.0:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while idx < N and x + r0 <= 1.0:
            centers[idx] = [x, y]
            idx += 1
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
    # Fill remaining spots if any
    while idx < N:
        centers[idx] = np.random.uniform(0.1, 0.9, 2)
        idx += 1
    return centers

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    
    best_centers = None
    best_radii = None
    best_sum = -1.0
    
    # Phase 1: Penalty-based optimization with continuation
    # Helps find a good topology and expand radii smoothly without hard constraints
    for seed in range(12):
        np.random.seed(seed)
        c_init = generate_hex_init(0.10)
        c_init += np.random.normal(0, 0.004, c_init.shape)
        c_init = np.clip(c_init, 0.05, 0.95)
        
        r_init = np.full(N, 0.09)
        
        x0 = np.zeros(3 * N)
        x0[:N] = c_init[:, 0]
        x0[N:2*N] = c_init[:, 1]
        x0[2*N:] = r_init
        
        bounds_p = get_bounds_penalty()
        
        # Continuation: gradually increase penalty weight to tighten feasibility
        for lam in [5.0, 20.0, 100.0, 500.0, 2000.0]:
            try:
                res = minimize(objective_penalty, x0, args=(lam,), method='L-BFGS-B',
                               bounds=bounds_p, options={'maxiter': 3000, 'ftol': 1e-14})
                x0 = res.x.copy()
            except Exception:
                pass
                
        # Extract configuration
        c_pen = np.column_stack((x0[:N], x0[N:2*N]))
        
        # Phase 2: LP refinement on penalty result to snap radii to theoretical max
        r_lp = solve_lp_radii(c_pen)
        if r_lp is not None:
            curr_sum = np.sum(r_lp)
            if curr_sum > best_sum:
                best_sum = curr_sum
                best_centers = c_pen.copy()
                best_radii = r_lp.copy()
                
    # Phase 3: SLSQP fine-tuning with exact constraints
    if best_centers is not None:
        bounds_s = []
        for _ in range(N):
            bounds_s.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
            
        def cons_slsqp(vars):
            xs = vars[0::3]
            ys = vars[1::3]
            rs = vars[2::3]
            # Boundary constraints
            c = np.concatenate([
                xs - rs, 1.0 - xs - rs, ys - rs, 1.0 - ys - rs
            ])
            # Overlap constraints: dist^2 >= (ri+rj)^2
            dx = xs[:, None] - xs[None, :]
            dy = ys[:, None] - ys[None, :]
            dr = rs[:, None] + rs[None, :]
            triu = np.triu_indices(N, 1)
            c = np.concatenate([c, dx[triu]**2 + dy[triu]**2 - dr[triu]**2])
            return c
            
        def obj_slsqp(vars):
            return -np.sum(vars[2::3])
            
        for rnd in range(20):
            x0 = np.zeros(3 * N)
            # Perturb current best to escape local minima
            noise_scale = 0.002 * (0.88 ** rnd)
            x0[0::3] = best_centers[:, 0] + np.random.normal(0, noise_scale, N)
            x0[1::3] = best_centers[:, 1] + np.random.normal(0, noise_scale, N)
            x0[2::3] = best_radii * 0.992  # Deflate slightly to allow positional movement
            
            # Project variables to strictly feasible bounds for SLSQP stability
            for i in range(N):
                r = max(0.005, x0[3*i+2])
                x0[3*i] = np.clip(x0[3*i], r, 1.0-r)
                x0[3*i+1] = np.clip(x0[3*i+1], r, 1.0-r)
                x0[3*i+2] = r
                
            try:
                res = minimize(obj_slsqp, x0, method='SLSQP', bounds=bounds_s,
                               constraints={'type': 'ineq', 'fun': cons_slsqp},
                               options={'maxiter': 25000, 'ftol': 1e-14, 'disp': False})
                if not np.isnan(res.fun):
                    c_opt = np.column_stack((res.x[0::3], res.x[1::3]))
                    # LP check to see if radii can be improved for these centers
                    r_lp = solve_lp_radii(c_opt)
                    if r_lp is not None:
                        s = np.sum(r_lp)
                        if s > best_sum:
                            best_sum = s
                            best_centers = c_opt.copy()
                            best_radii = r_lp.copy()
            except Exception:
                pass
                
    # Fallback initialization
    if best_centers is None:
        best_centers = generate_hex_init(0.09)
        best_radii = np.full(N, 0.08)
        best_sum = np.sum(best_radii)
        
    centers = best_centers.copy()
    radii = best_radii.copy()
    
    # Final strict compliance fix against 1e-12 validator tolerance
    for _ in range(100):
        valid = True
        for i in range(N):
            if radii[i] < 0 or centers[i, 0] < radii[i] - 1e-11 or centers[i, 0] > 1.0 - radii[i] + 1e-11 or \
               centers[i, 1] < radii[i] - 1e-11 or centers[i, 1] > 1.0 - radii[i] + 1e-11:
                valid = False; break
        if valid:
            for i in range(N):
                for j in range(i + 1, N):
                    d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                    if d < radii[i] + radii[j] - 1e-11:
                        valid = False; break
                if not valid: break
        if valid: break
        # Minimal shrinkage to recover strict feasibility
        radii *= 0.9995
        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)
        
    return centers, radii, float(np.sum(radii))
