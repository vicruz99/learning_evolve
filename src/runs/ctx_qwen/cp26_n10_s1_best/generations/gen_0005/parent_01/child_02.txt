import numpy as np
from scipy.optimize import minimize, linprog

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Returns all inequality constraints g(x) >= 0 (vectorized)."""
    xs, ys, rs = x[0::3], x[1::3], x[2::3]
    
    # Boundary constraints
    c = np.concatenate([
        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs
    ])
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    c = np.concatenate([c, dx[TRIL_IDX]**2 + dy[TRIL_IDX]**2 - dr[TRIL_IDX]**2])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [0, 0.5]."""
    return [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, d)
            idx += 1
            
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=[(0.0, None)]*n, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def make_hex(rows, r0):
    """Generate hexagonal lattice points for given row counts."""
    centers = []
    y = r0
    for i, cnt in enumerate(rows):
        x_start = r0 + (r0 if i % 2 == 1 else 0.0)
        for k in range(cnt):
            centers.append([x_start + k * 2.0 * r0, y])
        y += np.sqrt(3.0) * r0
    centers = np.array(centers[:N])
    while len(centers) < N:
        centers = np.vstack([centers, [np.random.uniform(0.2, 0.8), np.random.uniform(0.2, 0.8)]])
    return centers

def fit_and_rotate(centers, angle):
    """Rotate and scale configuration to fit tightly in unit square."""
    c, s = np.cos(angle), np.sin(angle)
    pts = centers - 0.5
    rot = pts @ np.array([[c, -s], [s, c]]) + 0.5
    
    min_x, max_x = rot[:, 0].min(), rot[:, 0].max()
    min_y, max_y = rot[:, 1].min(), rot[:, 1].max()
    
    scale_x = 0.998 / (max_x - min_x) if max_x > min_x else 1.0
    scale_y = 0.998 / (max_y - min_y) if max_y > min_y else 1.0
    scale = min(scale_x, scale_y)
    
    center = (rot.min(axis=0) + rot.max(axis=0)) / 2.0
    rot = (rot - center) * scale + 0.5
    return np.clip(rot, 0.002, 0.998)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -1.0
    best_x = None
    
    # Diverse hexagonal patterns summing to 26
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 6, 4], [6, 6, 6, 4, 4],
        [4, 6, 5, 6, 5], [5, 7, 5, 5, 4], [6, 4, 6, 6, 4], [4, 5, 6, 6, 5],
        [5, 6, 6, 5, 4], [6, 5, 5, 6, 4], [5, 5, 5, 5, 6]
    ]
    
    configs = []
    for pat in patterns:
        base = make_hex(pat, r0=0.095)
        # Try multiple rotations to find optimal boundary fit
        for ang in np.linspace(0, np.pi/2, 12):
            configs.append(fit_and_rotate(base, ang))
            
    # Add random perturbed hex grids
    for _ in range(8):
        base = make_hex([5, 6, 5, 6, 4], r0=0.095)
        base += np.random.normal(0, 0.005, base.shape)
        configs.append(fit_and_rotate(base, np.random.uniform(0, 1.0)))

    # Phase 1: Multi-start Optimization
    for c_init in configs:
        r_init = solve_lp_radii(c_init)
        if r_init is None:
            r_init = np.full(N, 0.09)
            
        x0 = np.zeros(3 * N)
        x0[0::3] = c_init[:, 0]
        x0[1::3] = c_init[:, 1]
        x0[2::3] = r_init
        
        # Project to feasible bounds
        for i in range(N):
            r = max(0.01, x0[3*i+2])
            x0[3*i] = np.clip(x0[3*i], r, 1.0 - r)
            x0[3*i+1] = np.clip(x0[3*i+1], r, 1.0 - r)
            
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 20000, 'ftol': 1e-12, 'disp': False})
            
            if not np.isnan(res.fun):
                curr_sum = -res.fun
                c_vals = constraints(res.x)
                if np.min(c_vals) >= -1e-5 and curr_sum > best_sum:
                    best_sum = curr_sum
                    best_x = res.x.copy()
        except Exception:
            pass
            
    # Phase 2: Iterative Refinement (LP <-> SLSQP with deflation)

                best_sum = np.sum(r_lp)

            x0 += np.random.normal(0, noise, 3 * N)
            
            for i in range(N):
                r = max(0.01, x0[3*i+2])
                x0[3*i] = np.clip(x0[3*i], r, 1.0 - r)
                x0[3*i+1] = np.clip(x0[3*i+1], r, 1.0 - r)
                
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 15000, 'ftol': 1e-12, 'disp': False})
                
                if not np.isnan(res.fun):
                    curr_sum = -res.fun
                    c_vals = constraints(res.x)
                    if np.min(c_vals) >= -1e-5 and curr_sum > best_sum:
                        best_x = res.x.copy()
                        best_sum = curr_sum
            except Exception:
                pass
                
    # Phase 3: Aggressive Uniform Expansion & Validation
    centers = best_x.reshape(N, 3)[:, :2]
    radii = best_x[2::3]
    
    # Try to uniformly scale up radii until constrained, then shrink slightly for safety
    for _ in range(50):
        valid = True
        for i in range(N):
            if radii[i] < 0 or centers[i, 0] < radii[i] - 1e-10 or centers[i, 0] > 1.0 - radii[i] + 1e-10 or \
               centers[i, 1] < radii[i] - 1e-10 or centers[i, 1] > 1.0 - radii[i] + 1e-10:
                valid = False
                break
        if valid:
            for i in range(N):
                for j in range(i + 1, N):
                    if np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1]) < radii[i] + radii[j] - 1e-10:
                        valid = False
                        break
                if not valid: break
                
        if not valid:
            break
            
        # Slightly increase radii
        radii *= 1.0005
        # Re-project centers to maintain boundary constraints
        for i in range(N):
            centers[i, 0] = np.clip(centers[i, 0], radii[i], 1.0 - radii[i])
            centers[i, 1] = np.clip(centers[i, 1], radii[i], 1.0 - radii[i])
            
    # Final strict validation & minimal numerical repair
    for _ in range(100):
        valid = True
        for i in range(N):
            if radii[i] < 0 or centers[i, 0] < radii[i] - 1e-11 or centers[i, 0] > 1.0 - radii[i] + 1e-11 or \
               centers[i, 1] < radii[i] - 1e-11 or centers[i, 1] > 1.0 - radii[i] + 1e-11:

            centers[i, 1] = np.clip(centers[i, 1], radii[i], 1.0 - radii[i])
            
    return centers, radii, float(np.sum(radii))
