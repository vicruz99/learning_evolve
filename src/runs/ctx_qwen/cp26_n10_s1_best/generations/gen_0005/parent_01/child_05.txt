

<strategy>
To improve the circle packing sum of radii beyond the stagnation point of ~2.634, we need to address three key aspects:
1. **Constraint Conditioning**: Switch from squared distance constraints `dist^2 >= (r_i + r_j)^2` to direct distance constraints `dist >= r_i + r_j`. This improves gradient conditioning near contact points, allowing SLSQP to converge more accurately.
2. **Alternating LP/SLSQP with Expansion**: The LP step finds optimal radii for fixed centers, but SLSQP can get trapped in topologies where radii are suboptimally distributed. Adding an "expansion phase" that slightly scales up radii and re-optimizes centers helps push the configuration into higher-sum regions.
3. **Diverse Topological Seeds**: Generate multiple hexagonal lattice patterns with varying row counts and rotations, combined with force-directed layouts. This increases the chance of landing in the basin of attraction for the global optimum. Deflation + perturbation during refinement ensures we escape local minima.

The implementation vectorizes constraints for speed, uses `linprog` for exact radius optimization given centers, and includes strict post-processing to guarantee compliance with the `1e-12` validation tolerance.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Returns all inequality constraints g(x) >= 0."""
    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]
    
    # Boundary constraints: 4*N
    c = np.concatenate([
        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs
    ])
    
    # Overlap constraints: dist >= r_i + r_j
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dist = np.hypot(dx, dy)
    r_sum = rs[:, None] + rs[None, :]
    
    i_idx, j_idx = np.tril_indices(N, -1)
    c = np.concatenate([c, dist[i_idx, j_idx] - r_sum[i_idx, j_idx]])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [0, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
    return b

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, d)
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def force_directed_init(n, seed, steps=400):
    """Force-directed relaxation to spread points evenly and push to boundaries."""
    np.random.seed(seed)
    centers = np.random.uniform(0.15, 0.85, (n, 2))
    radii = np.full(n, 0.05)
    
    for step in range(steps):
        forces = np.zeros_like(centers)
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[j] - centers[i]
                dist = np.linalg.norm(dx)
                if dist < 0.25 and dist > 1e-6:
                    f = 0.012 / (dist**2 + 0.001)
                    forces[i] -= f * dx
                    forces[j] += f * dx
                    
            for dim in range(2):
                if centers[i, dim] < radii[i] + 0.08:
                    forces[i, dim] += 0.06
                elif centers[i, dim] > 1.0 - radii[i] - 0.08:
                    forces[i, dim] -= 0.06
                    
        lr = 0.05 * (1.0 - step / steps)
        centers += forces * lr
        centers = np.clip(centers, 0.02, 0.98)
        
    return centers

def generate_hex_init(r0, rows_pattern, seed=0):
    """Generates a hexagonal lattice initialization with specified row counts."""
    np.random.seed(seed)
    centers = np.zeros((N, 2))
    idx = 0
    y = r0
    y_step = np.sqrt(3.0) * r0
    
    for i, cnt in enumerate(rows_pattern):
        x_start = r0 + (r0 if i % 2 == 1 else 0.0)
        for k in range(cnt):
            if idx < N:
                centers[idx, 0] = x_start + k * 2.0 * r0
                centers[idx, 1] = y
                idx += 1
        y += y_step
        
    while idx < N:
        centers[idx] = np.random.uniform(0.1, 0.9, 2)
        idx += 1
        
    centers += np.random.uniform(-0.002, 0.002, centers.shape)
    return centers

def rotate_grid(centers, angle):
    """Rotates a grid of centers around the square center."""
    c, s = np.cos(angle), np.sin(angle)
    pts = centers - 0.5
    new_pts = np.column_stack((pts[:, 0]*c - pts[:, 1]*s, pts[:, 0]*s + pts[:, 1]*c)) + 0.5
    return new_pts

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -1.0
    best_x = None
    
    # Phase 1: Diverse Initializations
    configs = []
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 6, 6, 4], [6, 6, 6, 4, 4],
        [5, 7, 5, 5, 4], [4, 6, 6, 6, 4], [6, 4, 6, 6, 4], [5, 6, 6, 5, 4],
        [5, 5, 5, 5, 6], [6, 6, 4, 5, 5]
    ]
    
    for pat in patterns:
        c = generate_hex_init(0.09, pat)
        configs.append(c)
        for ang in [0.05, 0.1, 0.2, 0.35, 0.5, 0.7]:
            configs.append(rotate_grid(c, ang))
            
    for s in range(15):
        configs.append(force_directed_init(N, seed=s))
        
    # Optimize each initialization
    for c_init in configs:
        r_lp = solve_lp_radii(c_init)
        if r_lp is None:
            r_lp = np.full(N, 0.08)
            
        x0 = np.zeros(3 * N)
        x0[0::3] = c_init[:, 0]
        x0[1::3] = c_init[:, 1]
        x0[2::3] = r_lp * 0.995  # Slight shrinkage for strict feasibility
        
        # Project to strict bounds
        for i in range(N):
            r = max(0.0, x0[3*i+2])
            x0[3*i] = np.clip(x0[3*i], r, 1.0 - r)
            x0[3*i+1] = np.clip(x0[3*i+1], r, 1.0 - r)
            x0[3*i+2] = r
            
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 25000, 'ftol': 1e-14, 'disp': False})
            
            if not np.isnan(res.fun):
                curr_sum = -res.fun
                c_vals = constraints(res.x)
                if np.min(c_vals) >= -1e-6 and curr_sum > best_sum:
                    best_sum = curr_sum
                    best_x = res.x.copy()
        except Exception:
            pass
            
    # Phase 2: Alternating LP / SLSQP Refinement with Deflation & Expansion
    if best_x is not None:
        for step in range(30):
            # LP step: maximize radii for current centers
            c_curr = best_x.reshape(N, 3)[:, :2]
            r_lp = solve_lp_radii(c_curr)
            if r_lp is not None:
                best_x[2::3] = r_lp
                best_sum = np.sum(r_lp)
                
            # Deflate and perturb to escape local minima
            x0 = best_x.copy()
            x0[2::3] *= 0.980
            noise = 0.003 * (0.88 ** step)
            x0[0::3] += np.random.normal(0, noise, N)
            x0[1::3] += np.random.normal(0, noise, N)
            
            for i in range(N):
                r = max(0.01, x0[3*i+2])
                x0[3*i] = np.clip(x0[3*i], r, 1.0 - r)
                x0[3*i+1] = np.clip(x0[3*i+1], r, 1.0 - r)
                x0[3*i+2] = r
                
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 20000, 'ftol': 1e-14, 'disp': False})
                
                if not np.isnan(res.fun):
                    curr_sum = -res.fun
                    c_vals = constraints(res.x)
                    if np.min(c_vals) >= -1e-6 and curr_sum > best_sum:
                        best_x = res.x.copy()
                        best_sum = curr_sum
            except Exception:
                pass
                
            # Expansion phase: try to uniformly scale up radii and re-optimize positions
            if step % 3 == 0:
                x0_exp = best_x.copy()
                x0_exp[2::3] *= 1.008
                for i in range(N):
                    r = max(0.01, x0_exp[3*i+2])
                    x0_exp[3*i] = np.clip(x0_exp[3*i], r, 1.0 - r)
                    x0_exp[3*i+1] = np.clip(x0_exp[3*i+1], r, 1.0 - r)
                    x0_exp[3*i+2] = r
                    
                try:
                    res_exp = minimize(objective, x0_exp, method='SLSQP', bounds=bounds, constraints=cons,
                                       options={'maxiter': 15000, 'ftol': 1e-14, 'disp': False})
                    if not np.isnan(res_exp.fun):
                        curr_sum_exp = -res_exp.fun
                        c_vals_exp = constraints(res_exp.x)
                        if np.min(c_vals_exp) >= -1e-6 and curr_sum_exp > best_sum:
                            best_x = res_exp.x.copy()
                            best_sum = curr_sum_exp
                except Exception:
                    pass
                    
    # Extract results
    centers = best_x.reshape(N, 3)[:, :2]
    radii = best_x[2::3]
    
    # Final strict validity adjustment against 1e-12 tolerance
    for _ in range(100):
        valid = True
        for i in range(N):
            if radii[i] < 0 or centers[i, 0] < radii[i] - 1e-11 or centers[i, 0] > 1.0 - radii[i] + 1e-11 or \
               centers[i, 1] < radii[i] - 1e-11 or centers[i, 1] > 1.0 - radii[i] + 1e-11:
                valid = False
                break
        if valid:
            for i in range(N):
                for j in range(i + 1, N):
                    if np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1]) < radii[i] + radii[j] - 1e-11:
                        valid = False
                        break
                if not valid:
                    break
        if valid:
            break
            
        # Minimal shrinkage to recover strict feasibility
        radii *= 0.9998
        for i in range(N):
            centers[i, 0] = np.clip(centers[i, 0], radii[i], 1.0 - radii[i])
            centers[i, 1] = np.clip(centers[i, 1], radii[i], 1.0 - radii[i])
            
    return centers, radii, float(np.sum(radii))
