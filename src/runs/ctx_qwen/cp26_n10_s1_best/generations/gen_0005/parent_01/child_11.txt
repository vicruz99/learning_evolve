import numpy as np
from scipy.optimize import minimize, linprog

N_CIRCLES = 26

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N_CIRCLES
    c_obj = -np.ones(n)  # Maximize sum(r)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], 
                  centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], 
                         centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, d)
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-9)
    except Exception:
        pass
    return np.full(n, 0.08)

def force_optimize_centers(centers, radii, steps=1500):
    """Force-directed relaxation to push centers apart and towards boundaries."""
    pts = centers.copy()
    rs = np.maximum(radii, 1e-4)
    
    for step in range(steps):
        forces = np.zeros_like(pts)
        lr = 0.015 * (1.0 - step / steps) ** 1.5
        
        # Pairwise repulsion
        for i in range(N_CIRCLES):
            for j in range(i + 1, N_CIRCLES):
                dx = pts[j, 0] - pts[i, 0]
                dy = pts[j, 1] - pts[i, 1]
                dist = np.hypot(dx, dy)
                if dist < rs[i] + rs[j] and dist > 1e-6:
                    # Quadratic penalty force for smooth convergence
                    overlap = rs[i] + rs[j] - dist
                    f = overlap**2 / (dist + 1e-6) * 0.5
                    fx, fy = f * dx / dist, f * dy / dist
                    forces[i, 0] -= fx
                    forces[i, 1] -= fy
                    forces[j, 0] += fx
                    forces[j, 1] += fy
                    
        # Boundary repulsion
        for i in range(N_CIRCLES):
            for dim in range(2):
                if pts[i, dim] < rs[i]:
                    forces[i, dim] += (rs[i] - pts[i, dim])**2 * 2.0
                elif pts[i, dim] > 1.0 - rs[i]:
                    forces[i, dim] -= (pts[i, dim] - (1.0 - rs[i]))**2 * 2.0
                    
        pts += forces * lr
        pts = np.clip(pts, 1e-6, 1.0 - 1e-6)
        
    return pts

def generate_hex_configs():
    """Generate diverse hexagonal lattice initializations."""
    configs = []
    # Base hexagonal pattern
    for rot in np.linspace(0, np.pi/2, 7):
        centers = np.zeros((N_CIRCLES, 2))
        idx = 0
        r0 = 0.095
        y = r0
        row = 0
        while idx < N_CIRCLES and y + r0 <= 1.0:
            x_start = r0 + (r0 if row % 2 == 1 else 0.0)
            x = x_start
            while idx < N_CIRCLES and x + r0 <= 1.0:
                centers[idx] = [x, y]
                idx += 1
                x += 2.0 * r0
            y += np.sqrt(3.0) * r0
            row += 1
        while idx < N_CIRCLES:
            centers[idx] = np.random.uniform(0.1, 0.9, 2)
            idx += 1
            
        # Rotate around center
        if rot > 1e-5:
            c, s = np.cos(rot), np.sin(rot)
            pts = centers - 0.5
            centers = np.column_stack((pts[:, 0]*c - pts[:, 1]*s, 
                                       pts[:, 0]*s + pts[:, 1]*c)) + 0.5
        configs.append(centers.copy())
        
    # Grid fallbacks
    for _ in range(4):
        g = np.array([[0.1 + 0.18*c, 0.1 + 0.16*r] for r in range(6) for c in range(5)])[[:N_CIRCLES]]
        g += np.random.uniform(-0.01, 0.01, g.shape)
        configs.append(np.clip(g, 0.05, 0.95))
        
    return configs

def alternating_opt(centers_init, radii_init, max_rounds=12):
    """Run alternating LP radii and force-directed center optimization."""
    pts = centers_init.copy()
    rs = radii_init.copy()
    
    for rnd in range(max_rounds):
        # 1. LP step: maximize radii for current centers
        rs = solve_lp_radii(pts)
        
        # 2. Force step: optimize centers for current radii
        pts = force_optimize_centers(pts, rs, steps=600 - rnd*30)
        
        # Small perturbation to escape flat regions
        pts += np.random.normal(0, 0.0005 / (rnd + 1), pts.shape)
        pts = np.clip(pts, 0.005, 0.995)
        
    return pts, rs

def objective(x):
    return -np.sum(x[2::3])

def constraints(x):
    xs, ys, rs = x[0::3], x[1::3], x[2::3]
    c = np.concatenate([xs - rs, 1.0 - xs - rs, ys - rs, 1.0 - ys - rs])
    dx, dy = xs[:, None] - xs[None, :], ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    idx = np.tril_indices(N_CIRCLES, -1)
    c = np.concatenate([c, dx[idx]**2 + dy[idx]**2 - dr[idx]**2])
    return c

def get_bounds():
    return [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N_CIRCLES

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -1.0
    best_x = None

        x0 = np.zeros(3 * N_CIRCLES)
        x0[0::3] = pts[:, 0]
        x0[1::3] = pts[:, 1]
        x0[2::3] = rs
        
        # Project to strict bounds
        for i in range(N_CIRCLES):
            r = max(0.005, x0[3*i+2])
            x0[3*i] = np.clip(x0[3*i], r, 1.0 - r)
            x0[3*i+1] = np.clip(x0[3*i+1], r, 1.0 - r)
            x0[3*i+2] = r
            
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 15000, 'ftol': 1e-13, 'disp': False})
            if not np.isnan(res.fun):
                curr = -res.fun
                if np.min(constraints(res.x)) >= -1e-7 and curr > best_sum:
                    best_sum = curr
                    best_x = res.x.copy()
        except Exception:
            pass
            
    # Phase 2: Basin Hopping / Perturbation Refinement
    if best_x is not None:
        for hop in range(15):
            x0 = best_x.copy()
            x0[2::3] *= 0.98
            noise = 0.002 * (0.8 ** hop)
            x0[0::3] += np.random.normal(0, noise, N_CIRCLES)
            x0[1::3] += np.random.normal(0, noise, N_CIRCLES)
            
            for i in range(N_CIRCLES):
                r = max(0.01, x0[3*i+2])
                x0[3*i] = np.clip(x0[3*i], r, 1.0 - r)
                x0[3*i+1] = np.clip(x0[3*i+1], r, 1.0 - r)
                x0[3*i+2] = r
                
            try:

                               options={'maxiter': 12000, 'ftol': 1e-13, 'disp': False})
                if not np.isnan(res.fun):
                    curr = -res.fun
                    if np.min(constraints(res.x)) >= -1e-7 and curr > best_sum:
                        best_sum = curr

            except Exception:
                pass
                
    # Fallback
    if best_x is None:
        best_x = np.zeros(3 * N_CIRCLES)
        best_x[0::3] = np.linspace(0.15, 0.85, N_CIRCLES)
        best_x[1::3] = 0.5
        best_x[2::3] = 0.08
        best_sum = -objective(best_x)
        
    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3].copy()
    
    # Final strict validation & minimal repair
    for _ in range(100):
        valid = True
        for i in range(N_CIRCLES):

                        valid = False; break

        if valid: break
        
        radii *= 0.9998
        for i in range(N_CIRCLES):
            centers[i, 0] = np.clip(centers[i, 0], radii[i], 1.0 - radii[i])
            centers[i, 1] = np.clip(centers[i, 1], radii[i], 1.0 - radii[i])
            
    return centers, radii, float(np.sum(radii))
