

<strategy>
To improve upon the previous best sum of radii (~2.634), we need to explore the non-convex landscape more thoroughly. Circle packing is highly sensitive to topology (how circles are arranged relative to each other and boundaries). Standard gradient-based optimizers like SLSQP easily get trapped in local minima where circles are "jammed" in a suboptimal configuration.

Key improvements in this strategy:
1. **Enhanced Initialization**: Generate a wider variety of starting topologies, including densely packed hexagonal lattices at multiple scales, rotated lattices, force-directed repulsive layouts, and structured grids. This increases the chance of starting near a better global basin.
2. **Alternating LP + SLSQP Refinement**: For fixed centers, the optimal radii can be found exactly and instantaneously via Linear Programming. By alternating between optimizing centers (SLSQP) and snapping radii to their theoretical maximum (LP), we maintain strict feasibility while aggressively pushing the objective.
3. **Deflation & Perturbation**: Periodically shrink radii slightly to create "slack", then perturb centers to allow the optimizer to reorganize the packing topology and escape local jamming.
4. **Topology Swaps**: Randomly swap the positions of two circles, deflate radii, and re-optimize. This explicitly changes the connectivity/topology of the packing, which is crucial for escaping deep local minima that simple perturbations cannot overcome.
5. **Robust Projection & Validation**: Strictly project variables to feasible bounds after every perturbation and apply a final numerical repair loop to guarantee compliance with the `validate_packing` tolerance.

These combined techniques systematically explore configuration space while maintaining mathematical rigor in constraint handling.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Returns all inequality constraints g(x) >= 0."""
    cx, cy, r = x[0::3], x[1::3], x[2::3]
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    c = np.concatenate([
        cx - r,
        1.0 - cx - r,
        cy - r,
        1.0 - cy - r
    ])
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dr = r[:, None] + r[None, :]
    
    c = np.concatenate([c, dx[TRIL_IDX]**2 + dy[TRIL_IDX]**2 - dr[TRIL_IDX]**2])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-6, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)])
    return b

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A = np.zeros((num_ineq, n))
    b = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A[idx, i] = 1.0
        b[idx] = max(1e-9, lim)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = max(1e-9, d)
            idx += 1
            
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0.0, None)] * n, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-9)
    except Exception:
        pass
    return None

def make_hex_init(r0, angle, seed=0):
    """Generates a hexagonal lattice initialization with optional rotation."""
    np.random.seed(seed)
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        if y > 1.0 + r0: break
            
    pts = np.array(pts[:N + 10])
    
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
        
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.2, 0.8, (1, 2))])
    pts = pts[:N]
    pts += np.random.uniform(-0.002, 0.002, pts.shape)
    return np.clip(pts, 0.01, 0.99)

def make_force_init(seed, steps=600):
    """Force-directed layout to spread points evenly and push to boundaries."""
    np.random.seed(seed)
    pts = np.random.uniform(0.15, 0.85, (N, 2))
    for step in range(steps):
        forces = np.zeros_like(pts)
        for i in range(N):
            for j in range(i + 1, N):
                dx = pts[j] - pts[i]
                d = np.linalg.norm(dx)
                if d < 0.25 and d > 1e-5:
                    f = 0.01 / (d**2 + 0.001)
                    forces[i] -= f * dx
                    forces[j] += f * dx
            for dim in range(2):
                if pts[i, dim] < 0.1: forces[i, dim] += 0.05
                elif pts[i, dim] > 0.9: forces[i, dim] -= 0.05
        lr = 0.05 * (1.0 - step / steps)
        pts += forces * lr
        pts = np.clip(pts, 0.02, 0.98)
    return pts

def project_bounds(x):
    """Ensure optimization vector strictly respects bounds."""
    for i in range(N):
        r = max(1e-5, x[3 * i + 2])
        x[3 * i] = np.clip(x[3 * i], r, 1.0 - r)
        x[3 * i + 1] = np.clip(x[3 * i + 1], r, 1.0 - r)
        x[3 * i + 2] = r
    return x

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -1.0
    best_x = None
    
    # Phase 1: Diverse initializations
    configs = []
    for r0 in [0.085, 0.09, 0.095, 0.10, 0.105]:
        configs.append(make_hex_init(r0, 0.0, 0))
        for ang in [0.15, -0.15, 0.3, 0.5, 0.8]:
            configs.append(make_hex_init(r0, ang, 0))
    for s in range(10):
        configs.append(make_force_init(s))
        
    # Grid init
    c_grid = np.zeros((N, 2))
    idx = 0
    for i in range(6):
        for j in range(5):
            if idx < N:
                c_grid[idx] = [0.1 + 0.18 * j, 0.1 + 0.16 * i]
                idx += 1
    configs.append(c_grid)

    # Phase 1: Multi-start with Alternating LP + SLSQP
    for c_init in configs:
        x0 = np.zeros(3 * N)
        x0[0::3] = c_init[:, 0]
        x0[1::3] = c_init[:, 1]
        
        # Initial LP to snap radii to feasible maximum for these centers
        r_lp = solve_lp_radii(c_init)
        if r_lp is None:
            r_lp = np.full(N, 0.05)
        x0[2::3] = r_lp
        x0 = project_bounds(x0)

        # Alternating refinement cycles
        for _ in range(4):
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 15000, 'ftol': 1e-13, 'disp': False})
                if not np.isnan(res.fun):
                    x0 = res.x.copy()
                    # LP snap on new centers to instantly maximize radii
                    centers_new = np.column_stack((x0[0::3], x0[1::3]))
                    r_new = solve_lp_radii(centers_new)
                    if r_new is not None:
                        x0[2::3] = r_new
                        x0 = project_bounds(x0)
            except Exception:
                break

        curr_sum = np.sum(x0[2::3])
        c_vals = constraints(x0)
        if np.min(c_vals) >= -1e-6 and curr_sum > best_sum:
            best_sum = curr_sum
            best_x = x0.copy()

    # Phase 2: Deflation & Perturbation to escape local minima
    if best_x is not None:
        for step in range(40):
            noise = 0.002 / (step + 1)
            x_pert = best_x + np.random.normal(0, noise, 3 * N)
            # Shrink radii slightly to create slack for repositioning
            r_pert = np.maximum(x_pert[2::3] * 0.985, 0.005)
            x_pert[2::3] = r_pert
            x_pert = project_bounds(x_pert)

            try:
                res = minimize(objective, x_pert, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 10000, 'ftol': 1e-13, 'disp': False})
                if not np.isnan(res.fun):
                    centers_pert = np.column_stack((res.x[0::3], res.x[1::3]))
                    r_pert_lp = solve_lp_radii(centers_pert)
                    if r_pert_lp is not None:
                        x_pert[2::3] = r_pert_lp
                        x_pert = project_bounds(x_pert)
                        curr_sum = np.sum(x_pert[2::3])
                        c_vals = constraints(x_pert)
                        if np.min(c_vals) >= -1e-6 and curr_sum > best_sum:
                            best_sum = curr_sum
                            best_x = x_pert.copy()
            except Exception:
                pass

    # Phase 3: Topology Swap Refinement (changes connectivity)
    if best_x is not None:
        for _ in range(30):
            x_swap = best_x.copy()
            i, j = np.random.choice(N, 2, replace=False)
            # Swap centers of two circles
            x_swap[[3*i, 3*i+1]], x_swap[[3*j, 3*j+1]] = x_swap[[3*j, 3*j+1]].copy(), x_swap[[3*i, 3*i+1]].copy()
            x_swap[2::3] *= 0.97
            x_swap = project_bounds(x_swap)
            
            try:
                res = minimize(objective, x_swap, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 12000, 'ftol': 1e-13, 'disp': False})
                if not np.isnan(res.fun):
                    centers_swap = np.column_stack((res.x[0::3], res.x[1::3]))
                    r_swap_lp = solve_lp_radii(centers_swap)
                    if r_swap_lp is not None:
                        x_swap[2::3] = r_swap_lp
                        x_swap = project_bounds(x_swap)
                        curr_sum = np.sum(x_swap[2::3])
                        c_vals = constraints(x_swap)
                        if np.min(c_vals) >= -1e-6 and curr_sum > best_sum:
                            best_sum = curr_sum
                            best_x = x_swap.copy()
            except Exception:
                pass

    # Fallback (should not be reached)
    if best_x is None:
        best_x = np.zeros(3 * N)
        best_x[0::3] = np.tile(np.linspace(0.1, 0.9, 5), 6)[:N]
        best_x[1::3] = np.repeat(np.linspace(0.1, 0.9, 6), 5)[:N]
        best_x[2::3] = 0.06

    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3].copy()

    # Final strict validity adjustment against 1e-12 tolerance
    for _ in range(200):
        valid = True
        for i in range(N):
            if radii[i] < 0 or centers[i, 0] < radii[i] - 1e-9 or centers[i, 0] > 1.0 - radii[i] + 1e-9 or \
               centers[i, 1] < radii[i] - 1e-9 or centers[i, 1] > 1.0 - radii[i] + 1e-9:
                valid = False
                break
        if valid:
            for i in range(N):
                for j in range(i + 1, N):
                    d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                    if d < radii[i] + radii[j] - 1e-9:
                        valid = False
                        break
                if not valid: break
        if valid: break
        
        # Minimal shrinkage to recover strict feasibility
        radii *= 0.9999
        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)

    return centers, radii, float(np.sum(radii))
