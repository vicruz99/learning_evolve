import numpy as np
from scipy.optimize import minimize, linprog

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Returns all inequality constraints g(x) >= 0."""
    cx, cy, r = x[0::3], x[1::3], x[2::3]
    
    # Boundary constraints
    c = np.concatenate([
        cx - r, 1.0 - cx - r,
        cy - r, 1.0 - cy - r
    ])
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dr = r[:, None] + r[None, :]
    
    c = np.concatenate([c, dx[TRIL_IDX]**2 + dy[TRIL_IDX]**2 - dr[TRIL_IDX]**2])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-6, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)])
    return b

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A = np.zeros((num_ineq, n))
    b = np.zeros(num_ineq)
    
    idx = 0
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A[idx, i] = 1.0
        b[idx] = max(0.0, lim)
        idx += 1
        
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = max(0.0, d)
            idx += 1
            
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0.0, None)] * n, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-7)
    except Exception:
        pass
    return None

def hex_init(r0, angle=0.0):
    """Generates a hexagonal lattice initialization with optional rotation."""
    pts = []
    y = r0
    row = 0
    counts = [5, 6, 5, 6, 4]  # 26 circles total
    idx = 0
    
    for c in counts:
        x = r0 if row % 2 == 0 else 2.0 * r0
        for _ in range(c):
            if idx < N and x + r0 <= 1.0:
                pts.append([x, y])
                idx += 1
                x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        if idx >= N:
            break
            
    pts = np.array(pts)
    
    if angle != 0.0:
        c_val, s_val = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c_val, -s_val], [s_val, c_val]]) + 0.5
        
    pts = np.clip(pts, 0.02, 0.98)
    return pts[:N]

def force_push(centers, radii, steps=300):
    """Force-directed relaxation to push circles apart and resolve overlaps."""
    pts = centers.copy()
    r = radii.copy()
    for step in range(steps):
        forces = np.zeros_like(pts)
        for i in range(N):
            for j in range(i + 1, N):
                dx = pts[j, 0] - pts[i, 0]
                dy = pts[j, 1] - pts[i, 1]
                dist = np.hypot(dx, dy)
                if dist < r[i] + r[j] and dist > 1e-6:
                    overlap = (r[i] + r[j] - dist) * 0.5 / dist
                    forces[i, 0] -= overlap * dx
                    forces[i, 1] -= overlap * dy
                    forces[j, 0] += overlap * dx
                    forces[j, 1] += overlap * dy
                    
            for dim in range(2):
                if pts[i, dim] < r[i] + 1e-5:
                    forces[i, dim] += 0.05
                elif pts[i, dim] > 1.0 - r[i] - 1e-5:
                    forces[i, dim] -= 0.05
                    
        pts += forces * 0.05 * (1.0 - step / steps)
        pts = np.clip(pts, 1e-5, 1.0 - 1e-5)
    return pts

def project_to_bounds(x0):
    """Ensure optimization vector strictly respects bounds."""
    for i in range(N):
        r = max(1e-6, x0[3 * i + 2])
        x0[3 * i] = np.clip(x0[3 * i], r, 1.0 - r)
        x0[3 * i + 1] = np.clip(x0[3 * i + 1], r, 1.0 - r)
        x0[3 * i + 2] = r
    return x0

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    np.random.seed(42)
    best_sum = -1.0
    best_x = None
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}

    # Generate diverse initial configurations
    configs = []

        configs.append(hex_init(r0))

            configs.append(hex_init(r0, ang))
            
    # Phase 1: Multi-start with Force-Push + LP + SLSQP
    for c_init in configs:
        r_init = np.full(N, 0.08)
        c_init = force_push(c_init, r_init, steps=150)
        
        r_lp = solve_lp_radii(c_init)
        if r_lp is None:
            r_lp = np.full(N, 0.06)
            
        x0 = np.zeros(3 * N)
        x0[0::3] = c_init[:, 0]
        x0[1::3] = c_init[:, 1]
        x0[2::3] = r_lp
        x0 = project_to_bounds(x0)
        
        # Alternating refinement
        for _ in range(2):
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 12000, 'ftol': 1e-13, 'disp': False})
                if not np.isnan(res.fun):
                    x0 = res.x.copy()
                    centers_new = np.column_stack((x0[0::3], x0[1::3]))
                    r_new = solve_lp_radii(centers_new)
                    if r_new is not None:
                        x0[2::3] = r_new
                        x0 = project_to_bounds(x0)
            except Exception:
                break
                
        curr_sum = np.sum(x0[2::3])
        c_vals = constraints(x0)
        if np.min(c_vals) >= -1e-5 and curr_sum > best_sum:
            best_sum = curr_sum
            best_x = x0.copy()

    # Phase 2: Deflation & Perturbation to escape local minima
    if best_x is not None:
        for step in range(40):

            x_pert = project_to_bounds(x_pert)

            r_p = x_pert[2::3]

            x_pert[0::3] = c_p[:, 0]

                        res.x = project_to_bounds(res.x)
                        curr_sum = np.sum(res.x[2::3])
                        c_vals = constraints(res.x)
                        if np.min(c_vals) >= -1e-5 and curr_sum > best_sum:
                            best_sum = curr_sum
                            best_x = res.x.copy()
            except Exception:
                pass

    # Fallback
    if best_x is None:
        best_x = np.zeros(3 * N)
        best_x[0::3] = np.tile(np.linspace(0.1, 0.9, 5), 6)[:N]
        best_x[1::3] = np.repeat(np.linspace(0.1, 0.9, 6), 5)[:N]
        best_x[2::3] = 0.06

    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3].copy()

    # Final strict validity adjustment against 1e-12 tolerance
    for _ in range(150):
        valid = True
        for i in range(N):
            if radii[i] < 0: valid = False; break
            if centers[i, 0] - radii[i] < -1e-9 or centers[i, 0] + radii[i] > 1.0 + 1e-9: valid = False; break
            if centers[i, 1] - radii[i] < -1e-9 or centers[i, 1] + radii[i] > 1.0 + 1e-9: valid = False; break
        if valid:

        # Minimal shrinkage to recover strict feasibility
        radii *= 0.9999
        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)

    return centers, radii, float(np.sum(radii))
