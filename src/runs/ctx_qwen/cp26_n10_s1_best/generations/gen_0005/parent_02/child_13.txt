

<strategy>
1. **Diverse Initialization**: Generate multiple starting configurations including hexagonal lattices (various rotations/scales), square grids, corner-focused layouts, and force-directed random layouts. This covers different topological basins.
2. **Alternating LP + SLSQP Refinement**: For each configuration, run an iterative loop:
   - Fix centers, compute maximum feasible radii via Linear Programming (fast, exact, convex).
   - Use these radii to initialize SLSQP, which jointly optimizes centers and radii while enforcing non-overlap and boundary constraints.
   - Repeat to converge to a strong local optimum.
3. **Basin Hopping Escape**: To break past local minima (~2.630), implement a structured perturbation scheme:
   - Randomly select a subset of circles (10-40%), relocate them to random valid positions, and shrink radii to guarantee feasibility.
   - Re-run the refinement loop. Accept if the new sum of radii improves.
   - Vary perturbation intensity and subset size across iterations to explore both local tweaks and global rearrangements.
4. **Block Coordinate Descent**: Periodically optimize only a subset of circles while fixing others. This reduces dimensionality temporarily, helping the optimizer navigate complex constraint landscapes and escape saddle points.
5. **Strict Validation & Repair**: After optimization, rigorously check against the `validate_packing` tolerance (1e-12). Apply minimal uniform shrinking and center clipping only if necessary to guarantee strict compliance without sacrificing meaningful sum gains.
6. **Efficiency & Robustness**: Vectorize constraint evaluations, use `highs` method for LP, and wrap optimization calls in `try-except` blocks to ensure stability. Limit total iterations to balance exploration depth with runtime.
</strategy>

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Returns all inequality constraints g(x) >= 0."""
    cx, cy, r = x[0::3], x[1::3], x[2::3]
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    c = np.concatenate([
        cx - r,
        1.0 - cx - r,
        cy - r,
        1.0 - cy - r
    ])
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dr = r[:, None] + r[None, :]
    
    c = np.concatenate([c, dx[TRIL_IDX]**2 + dy[TRIL_IDX]**2 - dr[TRIL_IDX]**2])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-6, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)])
    return b

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A = np.zeros((num_ineq, n))
    b = np.zeros(num_ineq)
    
    idx = 0
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A[idx, i] = 1.0
        b[idx] = max(0.0, lim)
        idx += 1
        
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = max(0.0, d)
            idx += 1
            
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0.0, None)] * n, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def project_to_bounds(x0):
    """Ensure optimization vector strictly respects bounds."""
    x0 = x0.copy()
    for i in range(N):
        r = max(1e-6, x0[3 * i + 2])
        x0[3 * i] = np.clip(x0[3 * i], r, 1.0 - r)
        x0[3 * i + 1] = np.clip(x0[3 * i + 1], r, 1.0 - r)
        x0[3 * i + 2] = r
    return x0

def refine_packing(x0, n_cycles=6):
    """Alternating LP + SLSQP refinement."""
    current_x = x0.copy()
    for _ in range(n_cycles):
        try:
            # SLSQP to adjust centers and radii together
            res = minimize(objective, current_x, method='SLSQP', bounds=get_bounds(),
                           constraints={'type': 'ineq', 'fun': constraints},
                           options={'maxiter': 8000, 'ftol': 1e-12, 'disp': False})
            if not np.isnan(res.fun):
                current_x = res.x.copy()
                
            # LP snap to maximize radii for current centers
            centers = np.column_stack((current_x[0::3], current_x[1::3]))
            r_lp = solve_lp_radii(centers)
            if r_lp is not None:
                current_x[2::3] = r_lp
                current_x = project_to_bounds(current_x)
        except Exception:
            break
    return current_x

def generate_hex_init(r0, angle=0.0):
    """Generates a hexagonal lattice initialization with optional rotation."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 5:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        if y > 1.0 + r0: break
            
    pts = np.array(pts[:N + 5])
    
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
        
    mask = (pts[:, 0] >= 0.05) & (pts[:, 0] <= 0.95) & (pts[:, 1] >= 0.05) & (pts[:, 1] <= 0.95)
    pts = pts[mask]
    
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.2, 0.8, (1, 2))])
    return pts[:N]

def generate_force_init(seed, steps=300):
    """Force-directed layout to spread points evenly."""
    np.random.seed(seed)
    pts = np.random.uniform(0.15, 0.85, (N, 2))
    for _ in range(steps):
        forces = np.zeros_like(pts)
        for i in range(N):
            for j in range(i + 1, N):
                dx = pts[j] - pts[i]
                d = np.linalg.norm(dx)
                if d < 0.25 and d > 1e-5:
                    f = 0.01 / (d**2 + 0.001)
                    forces[i] -= f * dx
                    forces[j] += f * dx
            for dim in range(2):
                if pts[i, dim] < 0.15: forces[i, dim] += 0.02
                elif pts[i, dim] > 0.85: forces[i, dim] -= 0.02
        pts += forces * 0.05
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    best_sum = -1.0
    best_x = None
    bounds = get_bounds()

    # Phase 1: Diverse Initializations
    configs = []
    for r0 in [0.085, 0.09, 0.095, 0.10]:
        configs.append(generate_hex_init(r0))
        for ang in [0.1, -0.1, 0.2, 0.3, 0.5]:
            configs.append(generate_hex_init(r0, ang))
    for s in range(8):
        configs.append(generate_force_init(s))
        
    # Grid variant
    c_grid = np.zeros((N, 2))
    idx = 0
    for i in range(6):
        for j in range(5):
            if idx < N:
                c_grid[idx] = [0.1 + 0.18 * j, 0.1 + 0.16 * i]
                idx += 1
    configs.append(c_grid)

    # Optimize each initialization
    for c_init in configs:
        x0 = np.zeros(3 * N)
        x0[0::3] = c_init[:, 0]
        x0[1::3] = c_init[:, 1]
        
        r_lp = solve_lp_radii(c_init)
        if r_lp is None:
            r_lp = np.full(N, 0.05)
        x0[2::3] = r_lp
        x0 = project_to_bounds(x0)

        refined_x = refine_packing(x0, n_cycles=6)
        curr_sum = np.sum(refined_x[2::3])
        
        c_vals = constraints(refined_x)
        if np.min(c_vals) >= -1e-6 and curr_sum > best_sum:
            best_sum = curr_sum
            best_x = refined_x.copy()

    # Phase 2: Basin Hopping & Perturbation to escape local minima
    if best_x is not None:
        for step in range(40):
            # Determine perturbation strategy based on step
            if step % 4 == 0:
                # Large structural change: relocate 30% of circles
                n_move = int(0.3 * N)
                indices = np.random.choice(N, n_move, replace=False)
                x_pert = best_x.copy()
                r_shrink = np.maximum(x_pert[2::3] * 0.85, 0.01)
                x_pert[2::3] = r_shrink
                
                for idx in indices:
                    new_r = np.random.uniform(0.01, 0.08)
                    x_pert[3 * idx] = np.random.uniform(new_r, 1.0 - new_r)
                    x_pert[3 * idx + 1] = np.random.uniform(new_r, 1.0 - new_r)
                    x_pert[3 * idx + 2] = new_r
                    
                x_pert = project_to_bounds(x_pert)
                
            elif step % 4 == 1:
                # Gaussian noise + deflation
                noise_scale = 0.002 * (1.0 + step / 10.0)
                x_pert = best_x + np.random.normal(0, noise_scale, 3 * N)
                x_pert[2::3] = np.maximum(x_pert[2::3] * 0.95, 0.005)
                x_pert = project_to_bounds(x_pert)
                
            elif step % 4 == 2:
                # Block coordinate style: perturb half, fix half
                x_pert = best_x.copy()
                half_idx = N // 2
                swap_idx = np.random.permutation(N)[:half_idx]
                for idx in swap_idx:
                    new_r = best_x[3 * idx + 2] * 0.9
                    x_pert[3 * idx] = np.random.uniform(new_r, 1.0 - new_r)
                    x_pert[3 * idx + 1] = np.random.uniform(new_r, 1.0 - new_r)
                    x_pert[3 * idx + 2] = new_r
                x_pert = project_to_bounds(x_pert)
                
            else:
                # Gentle nudge
                x_pert = best_x + np.random.normal(0, 0.0005, 3 * N)
                x_pert = project_to_bounds(x_pert)

            refined_pert = refine_packing(x_pert, n_cycles=5)
            curr_sum = np.sum(refined_pert[2::3])
            
            c_vals = constraints(refined_pert)
            if np.min(c_vals) >= -1e-6 and curr_sum > best_sum:
                best_sum = curr_sum
                best_x = refined_pert.copy()

    # Fallback (should not be reached)
    if best_x is None:
        best_x = np.zeros(3 * N)
        best_x[0::3] = np.tile(np.linspace(0.1, 0.9, 5), 6)[:N]
        best_x[1::3] = np.repeat(np.linspace(0.1, 0.9, 6), 5)[:N]
        best_x[2::3] = 0.06

    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3].copy()

    # Final strict validity adjustment against 1e-12 tolerance
    for _ in range(100):
        valid = True
        for i in range(N):
            if radii[i] < 0: valid = False; break
            if centers[i, 0] - radii[i] < -1e-9 or centers[i, 0] + radii[i] > 1.0 + 1e-9: valid = False; break
            if centers[i, 1] - radii[i] < -1e-9 or centers[i, 1] + radii[i] > 1.0 + 1e-9: valid = False; break
        if valid:
            for i in range(N):
                for j in range(i + 1, N):
                    d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                    if d < radii[i] + radii[j] - 1e-9: valid = False; break
                if not valid: break
        if valid: break
        
        # Minimal shrinkage to recover strict feasibility
        radii *= 0.9999
        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)

    return centers, radii, float(np.sum(radii))
