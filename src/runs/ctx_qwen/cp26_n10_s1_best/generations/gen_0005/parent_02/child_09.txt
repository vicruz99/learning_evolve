import numpy as np
from scipy.optimize import minimize

N = 26

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-6, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)])
    return b

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def get_constraints_and_jac(x):
    """Returns constraint values and analytical Jacobian matrix."""
    n = N
    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]
    
    n_bnd = 4 * n
    n_ovl = n * (n - 1) // 2
    n_c = n_bnd + n_ovl
    
    c = np.empty(n_c)
    J = np.empty((n_c, 3 * n))
    J[:] = 0.0
    
    # Boundary constraints: 4 per circle
    for i in range(n):
        idx = 3 * i
        # x - r >= 0
        c[2*i] = xs[i] - rs[i]
        J[2*i, idx] = 1.0
        J[2*i, idx+2] = -1.0
        # 1 - x - r >= 0
        c[2*i+1] = 1.0 - xs[i] - rs[i]
        J[2*i+1, idx] = -1.0
        J[2*i+1, idx+2] = -1.0
        # y - r >= 0
        c[n_bnd + 2*i] = ys[i] - rs[i]
        J[n_bnd + 2*i, idx+1] = 1.0
        J[n_bnd + 2*i, idx+2] = -1.0
        # 1 - y - r >= 0
        c[n_bnd + 2*i + 1] = 1.0 - ys[i] - rs[i]
        J[n_bnd + 2*i + 1, idx+1] = -1.0
        J[n_bnd + 2*i + 1, idx+2] = -1.0
        
    # Overlap constraints: dist^2 - (r_i + r_j)^2 >= 0
    k = n_bnd
    for i in range(n):
        for j in range(i + 1, n):
            idx_i = 3 * i
            idx_j = 3 * j
            dx = xs[i] - xs[j]
            dy = ys[i] - ys[j]
            dr = rs[i] + rs[j]
            
            c[k] = dx*dx + dy*dy - dr*dr
            J[k, idx_i] = 2.0 * dx
            J[k, idx_j] = -2.0 * dx
            J[k, idx_i+1] = 2.0 * dy
            J[k, idx_j+1] = -2.0 * dy
            J[k, idx_i+2] = -2.0 * dr
            J[k, idx_j+2] = -2.0 * dr
            k += 1
            
    return c, J

def constraint_func(x):
    return get_constraints_and_jac(x)[0]

def jac_func(x):
    return get_constraints_and_jac(x)[1]

def project_to_bounds(x):
    """Project variables to strictly satisfy boundary and non-negative radius constraints."""
    for i in range(N):
        r = max(1e-6, x[3 * i + 2])
        x[3 * i] = np.clip(x[3 * i], r, 1.0 - r)
        x[3 * i + 1] = np.clip(x[3 * i + 1], r, 1.0 - r)
        x[3 * i + 2] = r
    return x

def make_hex_init(r0, angle, seed):
    """Generate a rotated hexagonal lattice initialization."""
    np.random.seed(seed)
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 5:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        if y > 1.0 + r0: break
            
    pts = np.array(pts[:N + 5])
    
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
        
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.2, 0.8, (1, 2))])
    return pts[:N] + np.random.uniform(-0.002, 0.002, (N, 2))

def force_init(seed, steps=300):
    """Force-directed layout to spread points evenly."""
    np.random.seed(seed)
    pts = np.random.uniform(0.15, 0.85, (N, 2))
    for _ in range(steps):
        forces = np.zeros_like(pts)
        for i in range(N):
            for j in range(i + 1, N):
                dx = pts[j] - pts[i]
                d = np.linalg.norm(dx)
                if d < 0.25 and d > 1e-5:
                    f = 0.01 / (d**2 + 0.001)
                    forces[i] -= f * dx
                    forces[j] += f * dx
            for dim in range(2):
                if pts[i, dim] < 0.15: forces[i, dim] += 0.02
                elif pts[i, dim] > 0.85: forces[i, dim] -= 0.02
        pts += forces * 0.05
        pts = np.clip(pts, 0.05, 0.95)
    return pts

            if not np.isnan(res.fun):

                    best_sum = curr_sum
                    best_x = res.x.copy()
        except Exception:
            continue

    # Phase 2: Homotopy Expansion (Iteratively push radii up)
