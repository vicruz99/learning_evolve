import numpy as np
from scipy.optimize import minimize, linprog

N = 26
TRIL = np.tril_indices(N, -1)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Returns all inequality constraints g(x) >= 0."""
    cx, cy, r = x[0::3], x[1::3], x[2::3]
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    c = np.concatenate([
        cx - r,
        1.0 - cx - r,
        cy - r,
        1.0 - cy - r
    ])
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dr = r[:, None] + r[None, :]
    
    c = np.concatenate([c, dx[TRIL]**2 + dy[TRIL]**2 - dr[TRIL]**2])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-6, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)])
    return b

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(1e-9, lim)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(1e-9, d)
            idx += 1
            
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=[(0.0, None)] * n, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-6)
    except Exception:
        pass
    return None

def hex_init(r0, angle, shift_x=0.0, shift_y=0.0):
    """Generates a hexagonal lattice initialization with optional rotation and shift."""
    pts = []
    y = r0 + shift_y
    row = 0
    while len(pts) < N + 8:
        x = r0 + shift_x if row % 2 == 0 else 2.0 * r0 + shift_x
        while x <= 1.0 - r0:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        if y > 1.0 + r0: break
            
    pts = np.array(pts[:N + 8])
    
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
        
    mask = (pts[:, 0] >= 0.03) & (pts[:, 0] <= 0.97) & (pts[:, 1] >= 0.03) & (pts[:, 1] <= 0.97)
    pts = pts[mask]
    
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.2, 0.8, (1, 2))])
    return pts[:N]

def force_relax(centers, radii, steps=150):
    """Simple force-directed relaxation to spread points and resolve overlaps."""
    pts = centers.copy()
    rs = radii.copy()
    for _ in range(steps):
        forces = np.zeros_like(pts)
        for i in range(N):
            for j in range(i + 1, N):
                dx = pts[j] - pts[i]
                d = np.linalg.norm(dx)
                if d < rs[i] + rs[j] + 0.08 and d > 1e-6:
                    f = 0.015 / (d**2 + 0.001)
                    forces[i] -= f * dx
                    forces[j] += f * dx
            for dim in range(2):
                if pts[i, dim] < rs[i] + 0.02: forces[i, dim] += 0.03
                elif pts[i, dim] > 1.0 - rs[i] - 0.02: forces[i, dim] -= 0.03
        pts += forces * 0.03
        pts = np.clip(pts, 0.02, 0.98)
    return pts

def project_to_bounds(x0):
    """Ensure optimization vector strictly respects bounds."""
    for i in range(N):
        r = max(1e-5, x0[3 * i + 2])
        x0[3 * i] = np.clip(x0[3 * i], r, 1.0 - r)
        x0[3 * i + 1] = np.clip(x0[3 * i + 1], r, 1.0 - r)
        x0[3 * i + 2] = r
    return x0

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    best_sum = -1.0
    best_x = None
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}

    # Phase 1: Generate diverse initial configurations
    configs = []
    for r0 in [0.08, 0.085, 0.09, 0.095, 0.10]:
        configs.append(hex_init(r0, 0.0))
        for ang in [0.1, -0.1, 0.2, 0.3, 0.5, 0.8]:
            configs.append(hex_init(r0, ang))
        for sx in [0.0, 0.02, -0.02]:
            for sy in [0.0, 0.02, -0.02]:
                configs.append(hex_init(r0, 0.0, sx, sy))
                
    for s in range(8):
        c = np.random.uniform(0.15, 0.85, (N, 2))
        r = np.full(N, 0.06)
        configs.append(force_relax(c, r, steps=200))
        
    # Phase 2: Multi-start optimization with LP-SLSQP alternating
    for c_init in configs:
        x0 = np.zeros(3 * N)
        x0[0::3] = c_init[:, 0]
        x0[1::3] = c_init[:, 1]
        
        # Initial LP to snap radii to feasible maximum
        r_lp = solve_lp_radii(c_init)
        if r_lp is None:
            r_lp = np.full(N, 0.05)
        x0[2::3] = r_lp
        x0 = project_to_bounds(x0)

        # Alternating refinement: SLSQP -> LP -> SLSQP
        for _ in range(3):
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 12000, 'ftol': 1e-13, 'disp': False})
                if not np.isnan(res.fun):
                    x0 = res.x.copy()
                    centers_new = np.column_stack((x0[0::3], x0[1::3]))
                    r_new = solve_lp_radii(centers_new)
                    if r_new is not None:
                        x0[2::3] = r_new
                        x0 = project_to_bounds(x0)
            except Exception:
                break

        curr_sum = np.sum(x0[2::3])
        if curr_sum > best_sum:
            c_vals = constraints(x0)
            if np.min(c_vals) >= -1e-6:
                best_sum = curr_sum
                best_x = x0.copy()

    # Phase 3: Deflation & Perturbation to escape local minima
    if best_x is not None:
        for step in range(35):
            noise = 0.0018 / (step + 1)
            x_pert = best_x + np.random.normal(0, noise, 3 * N)
            # Shrink radii to create slack for topological rearrangement
            r_pert = np.maximum(x_pert[2::3] * 0.975, 0.005)
            x_pert[2::3] = r_pert
            x_pert = project_to_bounds(x_pert)

            try:
                res = minimize(objective, x_pert, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 9000, 'ftol': 1e-13, 'disp': False})
                if not np.isnan(res.fun):
                    centers_pert = np.column_stack((res.x[0::3], res.x[1::3]))
                    r_pert_lp = solve_lp_radii(centers_pert)
                    if r_pert_lp is not None:
                        x_pert[2::3] = r_pert_lp
                        x_pert = project_to_bounds(x_pert)
                        curr_sum = np.sum(x_pert[2::3])
                        c_vals = constraints(x_pert)
                        if np.min(c_vals) >= -1e-6 and curr_sum > best_sum:
                            best_sum = curr_sum
                            best_x = x_pert.copy()
            except Exception:
                pass

    # Fallback (should not be reached)
    if best_x is None:
        best_x = np.zeros(3 * N)
        best_x[0::3] = np.tile(np.linspace(0.1, 0.9, 5), 6)[:N]
        best_x[1::3] = np.repeat(np.linspace(0.1, 0.9, 6), 5)[:N]
        best_x[2::3] = 0.06

    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3].copy()

    # Phase 4: Strict Validation & Minimal Numerical Repair
    for _ in range(100):
        valid = True
        for i in range(N):
            if radii[i] < 0: valid = False; break
            if centers[i, 0] - radii[i] < -1e-9 or centers[i, 0] + radii[i] > 1.0 + 1e-9: valid = False; break
            if centers[i, 1] - radii[i] < -1e-9 or centers[i, 1] + radii[i] > 1.0 + 1e-9: valid = False; break

        radii *= 0.9999
        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)

    return centers, radii, float(np.sum(radii))
