import numpy as np
from scipy.optimize import minimize

N = 26
TRI_IND = np.tril_indices(N, -1)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Compute all inequality constraints g(x) >= 0."""
    xi = x[0::3]
    yi = x[1::3]
    ri = x[2::3]
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    b = np.concatenate([xi - ri, 1.0 - xi - ri, yi - ri, 1.0 - yi - ri])
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = xi[:, None] - xi[None, :]
    dy = yi[:, None] - yi[None, :]
    dr = ri[:, None] + ri[None, :]
    
    o = dx[TRI_IND]**2 + dy[TRI_IND]**2 - dr[TRI_IND]**2
    
    return np.concatenate([b, o])

def get_init(seed):
    """Generate diverse initial configurations."""
    np.random.seed(seed)
    x = np.zeros(3*N)
    
    pat_type = seed % 5
    if pat_type < 2:
        # Hexagonal lattice patterns
        rows = np.random.choice([[6,5,6,5,4], [5,6,5,6,4], [6,6,5,5,4], [4,5,6,6,5]])
        r0 = 0.095 + np.random.uniform(-0.008, 0.008)
        y = r0
        idx = 0
        dy = np.sqrt(3) * r0
        for i, cnt in enumerate(rows):
            shift = r0 if i % 2 == 1 else 0.0
            for j in range(cnt):
                if idx < N:
                    cx = r0 + shift + j * 2 * r0 + np.random.uniform(-0.005, 0.005)
                    cy = y + np.random.uniform(-0.005, 0.005)
                    x[3*idx] = cx
                    x[3*idx+1] = cy
                    x[3*idx+2] = r0
                    idx += 1
            y += dy
    else:
        # Grid-like or random feasible patterns
        r0 = 0.085 + np.random.uniform(0, 0.015)
        idx = 0
        if pat_type == 2:
            for row in range(6):
                for col in range(5):
                    if idx < N:
                        cx = 0.1 + col * 0.18 + (row%2)*0.09 + np.random.uniform(-0.01, 0.01)
                        cy = 0.1 + row * 0.16 + np.random.uniform(-0.01, 0.01)
                        x[3*idx] = cx
                        x[3*idx+1] = cy
                        x[3*idx+2] = r0
                        idx += 1
        else:
            for i in range(N):
                r0_i = r0 + np.random.uniform(-0.01, 0.01)
                x[3*i+2] = max(0.02, r0_i)
                x[3*i] = np.random.uniform(x[3*i+2], 1-x[3*i+2])
                x[3*i+1] = np.random.uniform(x[3*i+2], 1-x[3*i+2])
                
    # Project to boundary bounds
    for i in range(N):
        r = x[3*i+2]
        x[3*i] = np.clip(x[3*i], r, 1-r)
        x[3*i+1] = np.clip(x[3*i+1], r, 1-r)
        
    return x

def run_packing():
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_val = np.inf
    best_x = None
    
    # Phase 1: Multi-start optimization
    for seed in range(60):

        # Break exact symmetries
        x0 += np.random.normal(0, 1e-5, 3*N)
        x0[0::3] = np.clip(x0[0::3], x0[2::3], 1-x0[2::3])
        x0[1::3] = np.clip(x0[1::3], x0[2::3], 1-x0[2::3])
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 20000, 'ftol': 1e-13, 'disp': False})

                best_val = res.fun
                best_x = res.x.copy()

            continue

                        break

    return centers, radii, float(sum_r)
