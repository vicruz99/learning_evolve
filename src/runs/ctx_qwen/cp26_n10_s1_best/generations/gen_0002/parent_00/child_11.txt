import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def objective(vars):
    """Minimize negative sum of radii."""
    return -np.sum(vars[2::3])

def constraints(vars, n):
    """Compute all inequality constraints g(vars) >= 0."""
    xs = vars[0::3]
    ys = vars[1::3]
    rs = vars[2::3]
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    b = np.concatenate([xs - rs, 1.0 - xs - rs, ys - rs, 1.0 - ys - rs])
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    triu = np.triu_indices(n, k=1)
    o = dx[triu]**2 + dy[triu]**2 - dr[triu]**2
    
    return np.concatenate([b, o])

def get_bounds(n):
    """Variable bounds: x,y in [0,1], r in [0, 0.5]."""
    return [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * n

def force_relax_init(n, seed, steps=2500):
    """Generate initial configuration using repulsive force simulation."""
    np.random.seed(seed)
    # Start with a dense hexagonal pattern
    r0 = 0.095
    centers = np.zeros((n, 2))
    radii = np.full(n, r0)
    
    idx = 0
    y = r0
    while idx < n and y + r0 <= 1.0:
        x = r0
        # Stagger odd rows
        row_idx = int(y / r0)
        shift = r0 if (row_idx % 2 == 1) else 0.0
        
        while idx < n and x + r0 <= 1.0:
            centers[idx] = [x + shift, y]
            idx += 1
            x += 2 * r0
        y += np.sqrt(3) * r0
        
    # Fill any remaining spots randomly if grid was sparse
    while idx < n:
        centers[idx] = np.random.uniform(r0, 1.0 - r0, 2)
        idx += 1
        
    centers = centers[:n]
    # Add initial noise to break exact symmetry
    centers += np.random.normal(0, 0.02, centers.shape)
    centers = np.clip(centers, 0.05, 0.95)
    
    dt = 0.008
    for _ in range(steps):
        forces = np.zeros_like(centers)
        
        # Vectorized repulsion between circles
        diff = centers[:, None, :] - centers[None, :, :]  # (n, n, 2)
        dists = np.sqrt(np.sum(diff**2, axis=2))
        min_dists = radii[:, None] + radii[None, :]
        
        # Only compute forces for pairs that are too close
        overlap_mask = (dists < min_dists + 0.03) & (dists > 1e-6)
        force_mag = (min_dists[overlap_mask] + 0.03 - dists[overlap_mask]) * 0.8
        
        # Apply forces
        for i in range(n):
            for j in range(i + 1, n):
                if overlap_mask[i, j]:
                    f = force_mag[np.triu_indices(n, k=1)[0] == np.triu_indices(n, k=1)[0]] # inefficient fallback, better to use mask directly
                    # Simplified pairwise loop for clarity and speed in Python
                    pass
        
        # Faster pairwise loop for forces
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[j, 0] - centers[i, 0]
                dy = centers[j, 1] - centers[i, 1]
                d = np.hypot(dx, dy)
                req = radii[i] + radii[j]
                if d < req + 0.04 and d > 1e-7:
                    f = (req + 0.04 - d) * 1.5 / d
                    forces[i, 0] -= f * dx
                    forces[i, 1] -= f * dy
                    forces[j, 0] += f * dx
                    forces[j, 1] += f * dy
                    
        # Wall repulsion
        for i in range(n):
            r = radii[i]
            if centers[i, 0] < r + 0.02: forces[i, 0] += (r + 0.02 - centers[i, 0]) * 15.0
            elif centers[i, 0] > 1.0 - r - 0.02: forces[i, 0] -= (1.0 - r - 0.02 - centers[i, 0]) * 15.0
            if centers[i, 1] < r + 0.02: forces[i, 1] += (r + 0.02 - centers[i, 1]) * 15.0
            elif centers[i, 1] > 1.0 - r - 0.02: forces[i, 1] -= (1.0 - r - 0.02 - centers[i, 1]) * 15.0
            
        centers += forces * dt
        centers = np.clip(centers, 0.001, 0.999)
        
    return centers, radii

def generate_hex_init(n, seed, rot_angle=0.0):
    """Generate a rotated hexagonal lattice initialization."""
    np.random.seed(seed)
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.092)
    idx = 0
    r0 = 0.092
    y = r0
    
    while idx < n and y + r0 <= 1.0:
        x = r0
        row_idx = int(y / r0)
        shift = r0 if (row_idx % 2 == 1) else 0.0
        
        while idx < n and x + r0 <= 1.0:
            centers[idx] = [x + shift, y]
            idx += 1
            x += 2 * r0
        y += np.sqrt(3) * r0
        
    while idx < n:
        centers[idx] = np.random.uniform(r0, 1.0 - r0, 2)
        idx += 1
        
    centers = centers[:n]
    # Rotate around center
    if rot_angle != 0.0:
        c = 0.5
        c_pts = centers - c
        cos_a, sin_a = np.cos(rot_angle), np.sin(rot_angle)
        centers[:, 0] = c_pts[:, 0] * cos_a - c_pts[:, 1] * sin_a + c
        centers[:, 1] = c_pts[:, 0] * sin_a + c_pts[:, 1] * cos_a + c
        centers = np.clip(centers, 0.02, 0.98)
        
    return centers, radii

def ensure_feasibility(vars, n):
    """Project variables to satisfy bound constraints roughly."""
    for i in range(n):
        r = max(0.01, vars[3*i + 2])
        vars[3*i + 2] = r
        vars[3*i] = np.clip(vars[3*i], r, 1.0 - r)
        vars[3*i + 1] = np.clip(vars[3*i + 1], r, 1.0 - r)
    return vars

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    bounds = get_bounds(n)
    cons = {'type': 'ineq', 'fun': constraints, 'args': (n,)}
    
    best_sum = -1.0

    # Mix of force-relaxed, rotated hex grids, and standard hex grids
    inits = []

        inits.append((c, r))

        inits.append((c, r))
        
    for c_init, r_init in inits:

        x0[1::3] = c_init[:, 1]
        x0[2::3] = r_init
        x0 = ensure_feasibility(x0, n)
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,

            if not np.isnan(current_sum) and current_sum > best_sum:
                # Quick feasibility check
                vals = constraints(res.x, n)
                if np.min(vals) >= -1e-8:
                    best_sum = current_sum
                    best_vars = res.x.copy()

            continue

    if best_vars is not None:

                try:
                    res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,

                    break
                    
    # Fallback safety
    if best_vars is None:
        c, r = generate_hex_init(n, seed=0)
        best_vars = np.zeros(3 * n)
        best_vars[0::3] = c[:, 0]
        best_vars[1::3] = c[:, 1]
        best_vars[2::3] = r

    centers = np.zeros((n, 2))

        radii[i] = best_vars[3*i+2]

    valid = True
    for i in range(n):
        if radii[i] < 0: valid = False; break
        if centers[i,0] < radii[i] - 1e-10 or centers[i,0] > 1.0 - radii[i] + 1e-10: valid = False; break
        if centers[i,1] < radii[i] - 1e-10 or centers[i,1] > 1.0 - radii[i] + 1e-10: valid = False; break
