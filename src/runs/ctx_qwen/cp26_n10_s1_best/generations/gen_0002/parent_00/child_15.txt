        res.append(x - r)

        res.append(y - r)
        res.append(1.0 - y - r)

            dy = yi - yj

        for i in range(n):
            r = radii[i]
            for axis in range(2):
                if centers[i, axis] < r + 0.003:
                    forces[i, axis] += 8 * (r + 0.003 - centers[i, axis])
                elif centers[i, axis] > 1 - r - 0.003:
                    forces[i, axis] -= 8 * (centers[i, axis] - (1 - r - 0.003))
            for j in range(i + 1, n):
                dx = centers[j, 0] - centers[i, 0]
                dy = centers[j, 1] - centers[i, 1]
                d = np.sqrt(dx*dx + dy*dy)
                margin = radii[i] + radii[j] + 0.003
                if d < margin and d > 1e-10:
                    f = 15 * (margin - d) / d
                    forces[i, 0] -= f * dx
                    forces[i, 1] -= f * dy
                    forces[j, 0] += f * dx
                    forces[j, 1] += f * dy
        progress = step / steps
