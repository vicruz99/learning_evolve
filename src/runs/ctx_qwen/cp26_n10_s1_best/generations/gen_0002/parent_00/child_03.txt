from scipy.optimize import minimize

N_CIRCLES = 26

def objective(vars):
    """Minimize negative sum of radii."""
    return -np.sum(vars[2::3])

def constraints(vars):
    """Returns all inequality constraints >= 0 (vectorized)."""
    n = N_CIRCLES
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    num_wall = n * 4
    num_overlap = n * (n - 1) // 2
    res = np.empty(num_wall + num_overlap)
    
    # Boundary constraints
    res[:n] = x - r
    res[n:2*n] = 1.0 - x - r
    res[2*n:3*n] = y - r
    res[3*n:4*n] = 1.0 - y - r
    
    # Overlap constraints: dist^2 - (r_i + r_j)^2 >= 0
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dr = r[:, None] + r[None, :]
    
    i_idx, j_idx = np.tril_indices(n, -1)
    res[4*n:] = dx[i_idx, j_idx]**2 + dy[i_idx, j_idx]**2 - dr[i_idx, j_idx]**2
    
    return res
