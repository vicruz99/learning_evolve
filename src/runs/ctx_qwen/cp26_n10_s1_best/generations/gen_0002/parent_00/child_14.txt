N_CIRCLES = 26

def objective(vars):
    """Objective to minimize: negative sum of radii."""
    return -np.sum(vars[2::3])

def constraints(vars):
    """Vectorized inequality constraints: g(vars) >= 0."""
    n = N_CIRCLES
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    # Boundary constraints: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    b = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    
    # Overlap constraints: dist^2 - (r_i + r_j)^2 >= 0
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dr = r[:, None] + r[None, :]

    o = dx[i, j]**2 + dy[i, j]**2 - dr[i, j]**2

    return [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N_CIRCLES

    np.random.seed(seed)
    centers = []
    y = r_base
    row = 0

        x = r_base
        shift = r_base if row % 2 == 1 else 0.0

            centers.append([x + shift, y])
            x += 2.0 * r_base
        y += r_base * np.sqrt(3)
        row += 1

    if angle != 0.0:
        cx, cy = 0.5, 0.5
