import numpy as np
from scipy.optimize import minimize

N = 26

def objective(vars):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(vars[2::3])

def constraints(vars):
    """
    Vectorized inequality constraints g(vars) >= 0.
    Computes boundary and non-overlap constraints efficiently.
    """
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    # Preallocate constraint array
    c = np.empty(N * 4 + N*(N-1)//2)
    
    # Boundary constraints: circle inside [0,1]x[0,1]
    c[:N] = x - r
    c[N:2*N] = 1.0 - x - r
    c[2*N:3*N] = y - r
    c[3*N:4*N] = 1.0 - y - r
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    iu, ju = np.tril_indices(N, -1)
    dx = x[iu] - x[ju]
    dy = y[iu] - y[ju]
    dr = r[iu] + r[ju]
    c[4*N:] = dx**2 + dy**2 - dr**2
    
    return c

def hex_init(seed, rows_counts, r0=0.09):
    """Generates a hexagonal lattice initialization with slight noise."""
    np.random.seed(seed)
    vars = np.zeros(3*N)
    idx = 0
    y = r0
    for i, cnt in enumerate(rows_counts):
        x_start = r0 + (r0 if i % 2 == 1 else 0.0)
        for k in range(cnt):
            if idx < N:
                vars[3*idx] = x_start + k * 2*r0 + np.random.uniform(-0.002, 0.002)
                vars[3*idx+1] = y + np.random.uniform(-0.002, 0.002)
                vars[3*idx+2] = r0 + np.random.uniform(-0.002, 0.002)
                idx += 1
        y += np.sqrt(3) * r0
        
    # Fallback for any remaining circles
    while idx < N:
        vars[3*idx] = np.random.uniform(0.2, 0.8)
        vars[3*idx+1] = np.random.uniform(0.2, 0.8)
        vars[3*idx+2] = 0.06
        idx += 1
    return vars

def expand_init(seed):
    """Force-directed expansion initialization to find dense local packings."""
    np.random.seed(seed)
    cx = np.random.uniform(0.15, 0.85, N)
    cy = np.random.uniform(0.15, 0.85, N)
    r = np.full(N, 0.04)
    
    iu, ju = np.tril_indices(N, -1)
    for _ in range(2000):
        r += 0.00001
        dx = cx[iu] - cx[ju]
        dy = cy[iu] - cy[ju]
        dist = np.sqrt(dx**2 + dy**2)
        min_d = r[iu] + r[ju]
        overlap = np.maximum(min_d - dist, 0.0)
        
        # Resolve overlaps vectorized where possible, loop for precision
        mask = overlap > 1e-8
        if np.any(mask):
            iu_m, ju_m = iu[mask], ju[mask]
            dx_m, dy_m = dx[mask], dy[mask]
            dist_m, olap_m = dist[mask], overlap[mask]
            for k in range(len(iu_m)):
                d = dist_m[k]
                if d > 1e-8:
                    nx, ny = dx_m[k]/d, dy_m[k]/d
                    shift = olap_m[k] * 0.5
                    cx[iu_m[k]] -= shift * nx
                    cy[iu_m[k]] -= shift * ny
                    cx[ju_m[k]] += shift * nx
                    cy[ju_m[k]] += shift * ny
                    
        # Enforce wall boundaries
        cx = np.clip(cx, r, 1.0 - r)
        cy = np.clip(cy, r, 1.0 - r)
        
    res = np.zeros(3*N)
    res[0::3] = cx
    res[1::3] = cy
    res[2::3] = r
    return res

    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -1.0
