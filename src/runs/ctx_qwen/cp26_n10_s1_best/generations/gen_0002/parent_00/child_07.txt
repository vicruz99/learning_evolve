import numpy as np
from scipy.optimize import minimize

N = 26

def objective(vars):
    """Minimize negative sum of radii."""
    return -np.sum(vars[2::3])

def constraints_dist(vars):
    """Distance-based constraints (better gradients for radii)."""
    n = N
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    num_bc = 4 * n
    num_oc = n * (n - 1) // 2
    cons = np.empty(num_bc + num_oc)
    
    idx = 0
    for i in range(n):
        cons[idx] = x[i] - r[i]
        cons[idx + 1] = 1.0 - x[i] - r[i]
        cons[idx + 2] = y[i] - r[i]
        cons[idx + 3] = 1.0 - y[i] - r[i]
        idx += 4
    
    i, j = np.tril_indices(n, -1)
    dx = x[i] - x[j]
    dy = y[i] - y[j]
    dist = np.sqrt(dx * dx + dy * dy + 1e-16)
    cons[num_bc:] = dist - r[i] - r[j]
    
    return cons

def constraints_sq(vars):
    """Squared distance constraints (smoother near contact)."""
    n = N
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    num_bc = 4 * n
    num_oc = n * (n - 1) // 2
    cons = np.empty(num_bc + num_oc)
    
    idx = 0
    for i in range(n):
        cons[idx] = x[i] - r[i]
        cons[idx + 1] = 1.0 - x[i] - r[i]
        cons[idx + 2] = y[i] - r[i]
        cons[idx + 3] = 1.0 - y[i] - r[i]
        idx += 4
    
    i, j = np.tril_indices(n, -1)
    dx = x[i] - x[j]
    dy = y[i] - y[j]
    dr = r[i] + r[j]
    cons[num_bc:] = dx * dx + dy * dy - dr * dr
    
    return cons

def make_vars(centers, radii):
    """Convert centers and radii to flat optimization vector."""
    n = len(radii)
    vars = np.zeros(3 * n)
    vars[0::3] = centers[:, 0]
    vars[1::3] = centers[:, 1]
    vars[2::3] = radii
    return vars

def hex_init(n, seed, r_base=0.095, row_counts=None):
    """Generate hexagonal lattice initialization."""
    np.random.seed(seed)
    if row_counts is None:
        row_counts = [6, 5, 6, 5, 4]
    
    centers = np.zeros((n, 2))
    radii = np.full(n, r_base)
    
    idx = 0
    y = r_base
    for row_idx, count in enumerate(row_counts):
        if idx >= n:
            break
        x_start = r_base if row_idx % 2 == 0 else 2.0 * r_base
        for k in range(count):
            if idx >= n:
                break
            x = x_start + k * 2.0 * r_base
            centers[idx, 0] = x + np.random.uniform(-0.003, 0.003)
            centers[idx, 1] = y + np.random.uniform(-0.003, 0.003)
            idx += 1
        y += np.sqrt(3.0) * r_base
    
    while idx < n:
        centers[idx, 0] = np.random.uniform(r_base, 1.0 - r_base)
        centers[idx, 1] = np.random.uniform(r_base, 1.0 - r_base)
        idx += 1
    
    return centers, radii

def force_init(n, seed):
    """Generate force-directed random initialization."""
    np.random.seed(seed)
    centers = np.random.uniform(0.12, 0.88, (n, 2))
    
    for _ in range(2500):
        forces = np.zeros((n, 2))
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[j, 0] - centers[i, 0]
                dy = centers[j, 1] - centers[i, 1]
                dist = np.hypot(dx, dy)
                if dist < 0.18 and dist > 1e-10:
                    f = 0.004 / (dist * dist + 1e-8)
                    fx = f * dx / dist
                    fy = f * dy / dist
                    forces[i] -= [fx, fy]
                    forces[j] += [fx, fy]
        
        for i in range(n):
            if centers[i, 0] < 0.08:
                forces[i, 0] += 0.015
            elif centers[i, 0] > 0.92:
                forces[i, 0] -= 0.015
            if centers[i, 1] < 0.08:
                forces[i, 1] += 0.015
            elif centers[i, 1] > 0.92:
                forces[i, 1] -= 0.015
        
        centers += forces * 0.04
        centers = np.clip(centers, 0.05, 0.95)
    
    radii = np.zeros(n)
    for i in range(n):
        min_d = min(centers[i, 0], 1.0 - centers[i, 0],
                    centers[i, 1], 1.0 - centers[i, 1])
        for j in range(n):
            if i != j:
                d = np.hypot(centers[i, 0] - centers[j, 0],
                            centers[i, 1] - centers[j, 1])
                if d < min_d:
                    min_d = d
        radii[i] = min_d / 2.0 * 0.92
    
    return centers, radii

def grid_init(n, seed):

    np.random.seed(seed)
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    pts = []
    for row in range(7):

            y = 0.1 + row * 0.135
            pts.append([x, y])
    
    for i in range(n):
        if i < len(pts):
            centers[i, 0] = pts[i][0] + np.random.uniform(-0.01, 0.01)
            centers[i, 1] = pts[i][1] + np.random.uniform(-0.01, 0.01)
            radii[i] = 0.07 + np.random.uniform(-0.005, 0.005)
        else:
            centers[i, 0] = np.random.uniform(0.1, 0.9)
            centers[i, 1] = np.random.uniform(0.1, 0.9)
            radii[i] = 0.06
    
    return centers, radii

def ensure_feasible(vars, n):
    """Ensure variables satisfy bounds."""
    for i in range(n):
        r = max(0.01, vars[3 * i + 2])
        vars[3 * i + 2] = r
        vars[3 * i] = np.clip(vars[3 * i], r, 1.0 - r)
        vars[3 * i + 1] = np.clip(vars[3 * i + 1], r, 1.0 - r)
    return vars

def check_validity(centers, radii, n, tol=1e-9):
    """Check if packing is valid."""
    for i in range(n):
        x, y, r = centers[i, 0], centers[i, 1], radii[i]
        if x - r < -tol or x + r > 1 + tol or y - r < -tol or y + r > 1 + tol:
            return False
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.hypot(dx, dy)
            if dist < radii[i] + radii[j] - tol:
                return False
    return True

def shrink_to_valid(centers, radii, n):
    """Shrink radii until valid."""
    for _ in range(200):
        if check_validity(centers, radii, n, 1e-12):
            return True
        radii *= 0.998
    return False
