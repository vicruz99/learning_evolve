import numpy as np
from scipy.optimize import minimize

N = 26

def objective(vars):
    """Objective to minimize: negative sum of radii."""
    return -np.sum(vars[2::3])

def constraints(vars):
    """Returns all inequality constraints >= 0."""
    xs = vars[0::3]
    ys = vars[1::3]
    rs = vars[2::3]
    
    # Boundary constraints
    b_cons = np.concatenate([
        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs
    ])
    
    # Overlap constraints (vectorized)
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    i, j = np.tril_indices(N, -1)
    o_cons = dx[i, j]**2 + dy[i, j]**2 - dr[i, j]**2
    
    return np.concatenate([b_cons, o_cons])

def get_bounds():
    """Variable bounds."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
    return b

def affine_hex_init(seed, scale=1.0, rot=0.0, tx=0.0, ty=0.0):
    """Generates a rotated/scaled hexagonal lattice initialization."""
    rng = np.random.RandomState(seed)
    r0 = 0.085 * scale
    centers = []
    y = r0
    row = 0
    while len(centers) < N:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while len(centers) < N:
            centers.append([x, y])
            x += 2.0 * r0
            if x + r0 > 1.0:
                break
        y += np.sqrt(3.0) * r0
        row += 1
        if y + r0 > 1.0:
            break
            
    centers = np.array(centers[:N])
    # Apply affine transformation
    c = centers - 0.5
    rot_mat = np.array([[np.cos(rot), -np.sin(rot)], [np.sin(rot), np.cos(rot)]])
    c = c @ rot_mat * scale + 0.5 + np.array([tx, ty])
    
    # Clip to valid range
    c = np.clip(c, r0, 1.0 - r0)
    radii = np.full(N, r0)
    
    vars = np.zeros(3 * N)
    vars[0::3] = c[:, 0]
    vars[1::3] = c[:, 1]
    vars[2::3] = radii
    return vars

def force_init(seed):
    """Generates initial configuration using repulsive force simulation."""
    rng = np.random.RandomState(seed)
    c = rng.uniform(0.2, 0.8, (N, 2))
    r = np.full(N, 0.04)
    
    for _ in range(1500):
        f = np.zeros((N, 2))
        for i in range(N):
            for j in range(i + 1, N):
                dx, dy = c[j] - c[i]
                d = np.hypot(dx, dy)
                if d < 0.25 and d > 1e-5:
                    push = 0.015 / (d * d)
                    fx, fy = push * dx, push * dy
                    f[i] -= [fx, fy]
                    f[j] += [fx, fy]
            # Wall repulsion
            for k in range(2):
                if c[i, k] < r[i] + 0.06:
                    f[i, k] += 0.03
                elif c[i, k] > 1.0 - r[i] - 0.06:
                    f[i, k] -= 0.03
        c += f * 0.015
        c = np.clip(c, 0.05, 0.95)
        
    r = np.full(N, 0.06)
    vars = np.zeros(3 * N)
    vars[0::3] = c[:, 0]
    vars[1::3] = c[:, 1]
    vars[2::3] = r
    return vars

def optimize_start(x0, bounds, maxit=8000):
    """Runs SLSQP optimization on a given start vector."""
    try:
        res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                       constraints={'type': 'ineq', 'fun': constraints},
                       options={'maxiter': maxit, 'ftol': 1e-12, 'disp': False})
        return res
    except Exception:
        return None

def project_to_bounds(vars, n):
    """Ensures centers respect radius bounds."""
    for i in range(n):
        r = vars[3 * i + 2]
        if r < 0.001: r = 0.001
        vars[3 * i] = np.clip(vars[3 * i], r, 1.0 - r)
        vars[3 * i + 1] = np.clip(vars[3 * i + 1], r, 1.0 - r)
        vars[3 * i + 2] = r
    return vars

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    bounds = get_bounds()
    best_sum = -1.0
    best_vars = None
    
    # Phase 1: Affine Hexagonal Lattices
    for seed in range(35):
        scale = 0.92 + np.random.uniform(0, 0.12)
        rot = np.random.uniform(-0.4, 0.4)
        tx = np.random.uniform(-0.04, 0.04)
        ty = np.random.uniform(-0.04, 0.04)
        x0 = affine_hex_init(seed, scale, rot, tx, ty)
        x0 += np.random.normal(0, 0.002, 3 * N)
        res = optimize_start(x0, bounds, maxit=10000)
        if res is not None and -res.fun > best_sum:
            best_sum = -res.fun
            best_vars = res.x.copy()
            
    # Phase 2: Force-Directed Layouts
    for seed in range(25):
        x0 = force_init(seed)
        res = optimize_start(x0, bounds, maxit=10000)
        if res is not None and -res.fun > best_sum:
            best_sum = -res.fun
            best_vars = res.x.copy()
            
    # Phase 3: Iterative Radius Expansion (Pressure Method)
    if best_vars is not None:
        for step in range(8):
            x0 = best_vars.copy()
            x0[2::3] *= 1.012  # Grow radii
            x0 = project_to_bounds(x0, N)
            # Add tiny perturbation to avoid stalling
            x0 += np.random.normal(0, 5e-5, 3 * N)
            res = optimize_start(x0, bounds, maxit=15000)
            if res is not None and -res.fun > best_sum:
                best_sum = -res.fun
                best_vars = res.x.copy()
                
    # Phase 4: High-Precision Final Polish
    if best_vars is not None:
        for _ in range(3):
            x0 = best_vars + np.random.normal(0, 1e-5, 3 * N)
            x0 = project_to_bounds(x0, N)
            res = optimize_start(x0, bounds, maxit=20000)
            if res is not None and -res.fun > best_sum:
                best_sum = -res.fun
                best_vars = res.x.copy()
                
    # Extract results
    centers = np.zeros((N, 2))
    radii = np.zeros(N)
    centers[:, 0] = best_vars[0::3]
    centers[:, 1] = best_vars[1::3]
    radii[:] = best_vars[2::3]
    
    # Safety validation & minimal shrinkage to guarantee strict feasibility
    for _ in range(150):
        valid = True
        for i in range(N):
            x, y, r = centers[i, 0], centers[i, 1], radii[i]
            if x - r < -1e-11 or x + r > 1 + 1e-11 or y - r < -1e-11 or y + r > 1 + 1e-11:
                valid = False
                break
        if valid:
            for i in range(N):
                for j in range(i + 1, N):
                    dx = centers[i, 0] - centers[j, 0]
                    dy = centers[i, 1] - centers[j, 1]
                    dist = np.hypot(dx, dy)
                    if dist < radii[i] + radii[j] - 1e-11:
                        valid = False
                        break
                if not valid:
                    break
        if valid:
            break
            
        # Shrink minimally to recover validity
        radii *= 0.998
        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)
        
    return centers, radii, float(np.sum(radii))
