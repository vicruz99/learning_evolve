import numpy as np
from scipy.optimize import minimize, linprog

N_CIRCLES = 26

def unpack(vars):
    """Extracts centers and radii from flat optimization vector."""
    return vars[0::3], vars[1::3], vars[2::3]

def objective(vars):
    """Objective to minimize: negative sum of radii."""
    return -np.sum(vars[2::3])

def get_constraints(vars):
    """Computes all boundary and non-overlap constraints as a vector >= 0."""
    x, y, r = unpack(vars)
    n = N_CIRCLES
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    b = np.concatenate([
        x - r,
        1.0 - x - r,
        y - r,
        1.0 - y - r
    ])
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dr = r[:, None] + r[None, :]
    idx = np.tril_indices(n, -1)
    o = dx[idx]**2 + dy[idx]**2 - dr[idx]**2
    
    return np.concatenate([b, o])

def get_bounds():
    """Returns variable bounds: x,y in [0,1], r in [0, 0.5]."""
    b = []
    for _ in range(N_CIRCLES):
        b.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
    return b

def get_optimal_radii_lp(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    c = -np.ones(n)  # Minimize negative sum -> maximize sum
    n_walls = n * 4
    n_pairs = n * (n - 1) // 2
    A_ub = np.empty((n_walls + n_pairs, n))
    b_ub = np.empty(n_walls + n_pairs)
    
    row_idx = 0
    for i in range(n):
        x, y = centers[i]
        for val in [x, 1.0 - x, y, 1.0 - y]:
            A_ub[row_idx, i] = 1.0
            b_ub[row_idx] = val
            row_idx += 1
            
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[row_idx, i] = 1.0
            A_ub[row_idx, j] = 1.0
            b_ub[row_idx] = dist
            row_idx += 1
            
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(0.0, 0.5), method='highs')
    if res.success:
        return res.x
    return np.full(n, 0.05)

def create_init_centers(seed, method='hex'):
    """Generates diverse initial center layouts."""
    np.random.seed(seed)
    centers = np.zeros((N_CIRCLES, 2))
    
    if method == 'hex':
        r = 0.09
        y = r
        row = 0
        idx = 0
        while idx < N_CIRCLES:
            x = r
            shift = r if row % 2 == 1 else 0.0
            while idx < N_CIRCLES:
                cx = x + shift
                if cx + r > 1.0:
                    break
                centers[idx] = [cx, y]
                idx += 1
                x += 2.0 * r
            y += np.sqrt(3.0) * r
            row += 1
            if y + r > 1.0:
                break
    elif method == 'grid':
        idx = 0
        for i in range(6):
            for j in range(5):
                if idx < N_CIRCLES:
                    centers[idx] = [0.08 + j * 0.18, 0.08 + i * 0.16]
                    idx += 1
    else:  # rand
        for i in range(N_CIRCLES):
            centers[i] = np.random.uniform(0.15, 0.85, 2)
            
    # Add controlled noise to break symmetries
    centers += np.random.normal(0, 0.01, centers.shape)
    centers = np.clip(centers, 0.01, 0.99)
    return centers

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': get_constraints}
    
    best_sum = -1.0
    best_vars = None
    
    # Phase 1: LP-guided initialization & joint optimization
    for seed in range(40):
        methods = ['hex', 'hex', 'grid', 'rand']
        method = methods[seed % len(methods)]
        centers = create_init_centers(seed, method)
        
        # Step 1: Find optimal radii for these fixed centers via LP
        radii = get_optimal_radii_lp(centers)
        
        # Step 2: Assemble initial vector for joint optimization
        x0 = np.zeros(3 * N_CIRCLES)
        x0[0::3] = centers[:, 0]
        x0[1::3] = centers[:, 1]
        x0[2::3] = radii
        
        # Ensure strict feasibility for optimizer start
        for i in range(N_CIRCLES):
            r = max(0.01, min(0.49, x0[3 * i + 2]))
            x0[3 * i] = np.clip(x0[3 * i], r, 1.0 - r)
            x0[3 * i + 1] = np.clip(x0[3 * i + 1], r, 1.0 - r)
            x0[3 * i + 2] = r
            
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 15000, 'ftol': 1e-13, 'disp': False})
            if not np.isnan(res.fun) and -res.fun > best_sum:
                best_sum = -res.fun
                best_vars = res.x.copy()
        except Exception:
            continue
            
    # Phase 2: Intensive local search and perturbation from best

                x0[3 * i] = np.clip(x0[3 * i], r, 1.0 - r)

    radii = np.zeros(N_CIRCLES)
    for i in range(N_CIRCLES):
        centers[i, 0] = best_vars[3 * i]

        radii[i] = best_vars[3 * i + 2]

           centers[i, 1] < radii[i] - 1e-9 or centers[i, 1] + radii[i] > 1.0 + 1e-9:
            valid = False
            break
    if valid:

            for j in range(i + 1, N_CIRCLES):
                d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])

                    valid = False
                    break
            if not valid:
                break

            radii *= 0.995
            centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
            centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)
            ok = True
            for i in range(N_CIRCLES):

                        ok = False
                        break
                if not ok:
                    break
            if ok:
                break
                
    return centers, radii, float(np.sum(radii))
