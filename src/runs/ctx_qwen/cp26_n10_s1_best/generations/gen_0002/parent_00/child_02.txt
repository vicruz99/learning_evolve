from scipy.optimize import minimize
import math

N_CIRCLES = 26

def unpack(vars):
    centers = np.zeros((N_CIRCLES, 2))
    radii = np.zeros(N_CIRCLES)
    centers[:, 0] = vars[0::3]
    centers[:, 1] = vars[1::3]
    radii[:] = vars[2::3]
    return centers, radii

def objective(vars):
    """Minimize negative sum of radii."""
    return -np.sum(vars[2::3])

def constraints(vars):
    """Returns all inequality constraints g(vars) >= 0."""
    n = N_CIRCLES
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    # Boundary constraints
    b = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    
    # Overlap constraints (vectorized)
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dr = r[:, None] + r[None, :]
    tri = np.tril_indices(n, -1)
    o = dx[tri]**2 + dy[tri]**2 - dr[tri]**2
    
    return np.concatenate([b, o])

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [0, 0.5]."""
    return [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N_CIRCLES

def make_init_hex(seed, r=0.09):

    y = r
    row = 0

        idx += 1
    return vars

def make_init_random(seed):

        r = np.random.uniform(0.05, 0.12)
        vars[3*i] = np.random.uniform(r, 1-r)
        vars[3*i+1] = np.random.uniform(r, 1-r)
        vars[3*i+2] = r
    return vars

def relax(centers, radii, steps=600):
    """Force-directed relaxation with gradual radius growth."""
    n = len(radii)
    dx = centers[:, None, 0] - centers[None, :, 0]
    dy = centers[:, None, 1] - centers[None, :, 1]
    dist = np.hypot(dx, dy)
    dist = np.where(dist < 1e-6, 1e-6, dist)
    tri = np.tril_indices(n, -1)
    
    for _ in range(steps):
        forces = np.zeros((n, 2))
        # Pairwise repulsion
        for i in range(n):
            for j in range(i+1, n):
                d = dist[i, j]
                if d < 0.4:
                    f = 0.02 / (d*d)
                    fx, fy = f*dx[i,j]/d, f*dy[i,j]/d
                    forces[i] -= [fx, fy]
                    forces[j] += [fx, fy]
                    
        # Boundary repulsion
        for i in range(n):
            r = radii[i]

    return centers, radii
