import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26
TRIL_IDX = np.tril_indices(N_CIRCLES, -1)

def objective(vars):
    """Objective to minimize: negative sum of radii."""
    return -np.sum(vars[2::3])

def constraints(vars):
    """Vectorized inequality constraints: g(vars) >= 0."""
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    # Boundary constraints: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    b = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    
    # Overlap constraints: dist^2 - (r1+r2)^2 >= 0
    dx = x[:, None] - x
    dy = y[:, None] - y
    dr = r[:, None] + r
    
    i, j = TRIL_IDX
    o = dx[i, j]**2 + dy[i, j]**2 - dr[i, j]**2
    
    return np.concatenate([b, o])

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-6, 0.5]."""
    return [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N_CIRCLES

def is_valid(centers, radii):
    """Check strict feasibility of a packing configuration."""
    n = centers.shape[0]
    for i in range(n):
        if radii[i] < 0:
            return False
        if centers[i, 0] < radii[i] - 1e-9 or centers[i, 0] > 1.0 - radii[i] + 1e-9:
            return False
        if centers[i, 1] < radii[i] - 1e-9 or centers[i, 1] > 1.0 - radii[i] + 1e-9:
            return False
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            if d < radii[i] + radii[j] - 1e-9:
                return False
    return True

    """Generate hexagonal lattice initialization."""
    np.random.seed(seed)

    radii = np.full(N_CIRCLES, r0)
    idx = 0
    y = r0
    row = 0

                break

            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        if y + r0 > 1.0:
            break
            
    # Perturb to break exact symmetry
    centers += np.random.uniform(-0.003, 0.003, centers.shape)
    centers = np.clip(centers, r0, 1.0 - r0)
    return centers, radii

def random_init(seed):

    centers = np.random.uniform(0.15, 0.85, (N_CIRCLES, 2))
    radii = np.full(N_CIRCLES, 0.04)
    return centers, radii

def force_init(seed):

    centers = np.random.uniform(0.1, 0.9, (N_CIRCLES, 2))
    radii = np.full(N_CIRCLES, 0.04)

        forces = np.zeros((N_CIRCLES, 2))

            for j in range(i + 1, N_CIRCLES):

                dy = centers[j, 1] - centers[i, 1]
