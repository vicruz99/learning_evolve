from scipy.optimize import minimize

N = 26

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Vectorized inequality constraints g(x) >= 0."""
    n = N
    xc = x[0::3]
    yc = x[1::3]
    rc = x[2::3]
    
    num_overlap = n * (n - 1) // 2
    c = np.empty(4 * n + num_overlap)
    
    # Boundary constraints
    c[:n] = xc - rc
    c[n:2*n] = 1.0 - xc - rc
    c[2*n:3*n] = yc - rc
    c[3*n:4*n] = 1.0 - yc - rc
    
    # Overlap constraints using vectorized lower triangle indices
    i_idx, j_idx = np.tril_indices(n, -1)
    dx = xc[i_idx] - xc[j_idx]
    dy = yc[i_idx] - yc[j_idx]
    dr = rc[i_idx] + rc[j_idx]
    c[4*n:] = dx**2 + dy**2 - dr**2
    
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-6, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)])
    return b

def project_to_bounds(x, n):
    """Project variables to satisfy box constraints and basic boundary constraints."""
    for i in range(n):
        r = max(1e-6, x[3*i+2])
        x[3*i] = np.clip(x[3*i], r, 1.0 - r)
        x[3*i+1] = np.clip(x[3*i+1], r, 1.0 - r)
        x[3*i+2] = r
    return x

                    idx += 1

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
