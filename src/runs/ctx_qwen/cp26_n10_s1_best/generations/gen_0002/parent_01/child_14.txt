import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraint_func(x):
    """Vectorized inequality constraints: g(x) >= 0."""
    n = N_CIRCLES
    xp = x[0::3]
    yp = x[1::3]
    r = x[2::3]
    
    # Boundary constraints: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    cons = np.concatenate([
        xp - r, 
        1.0 - xp - r, 
        yp - r, 
        1.0 - yp - r
    ])
    
    # Overlap constraints: dist^2 - (r_i + r_j)^2 >= 0
    # Vectorized pairwise differences
    dx = xp[:, None] - xp[None, :]
    dy = yp[:, None] - yp[None, :]
    dr = r[:, None] + r[None, :]
    
    # Extract strictly lower triangle to avoid duplicates and self-comparisons
    tri = np.tril_indices(n, -1)
    cons = np.concatenate([cons, dx[tri]**2 + dy[tri]**2 - dr[tri]**2])
    
    return cons

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [0, 0.5]."""
    b = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)]
    return b * N_CIRCLES

def make_hex_init(seed, rotation=0.0):
    """Generate a feasible hexagonal lattice initialization with rotation and perturbation."""
    np.random.seed(seed)
    centers = np.zeros((N_CIRCLES, 2))
    r_init = 0.095
    idx = 0
    y = r_init
    row = 0
    
    # Fill hexagonal pattern
    while idx < N_CIRCLES and y + r_init <= 1.0:
        x = r_init if row % 2 == 0 else 2 * r_init
        while idx < N_CIRCLES and x + r_init <= 1.0:
            centers[idx] = [x, y]
            idx += 1
            x += 2 * r_init
        y += np.sqrt(3) * r_init
        row += 1
        
    # Fallback for remaining circles (rare)
    while idx < N_CIRCLES:
        centers[idx] = np.random.uniform(0.1, 0.9, 2)
        idx += 1
        
    # Apply rotation around center
    if rotation != 0.0:
        cx, cy = 0.5, 0.5
        cos_a, sin_a = np.cos(rotation), np.sin(rotation)
        centers -= [cx, cy]
        rot = np.array([[cos_a, -sin_a], [sin_a, cos_a]])
        centers = centers @ rot.T
        centers += [cx, cy]
        
    # Add controlled noise to break symmetry
    centers += np.random.normal(0, 0.008, centers.shape)
    centers = np.clip(centers, 0.02, 0.98)
    
    # Compute conservative feasible radii for each circle
    radii = np.full(N_CIRCLES, r_init)
    for i in range(N_CIRCLES):
        d_wall = min(centers[i,0], 1.0 - centers[i,0], centers[i,1], 1.0 - centers[i,1])
        d_min = d_wall
        for j in range(N_CIRCLES):
            if i != j:
                d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
                if d < d_min:
                    d_min = d
        radii[i] = d_min / 2.0 * 0.94  # 94% of max possible ensures strict initial feasibility
        
    x0 = np.zeros(3 * N_CIRCLES)
    x0[0::3] = centers[:, 0]
    x0[1::3] = centers[:, 1]
    x0[2::3] = radii
    return x0

def shrink_to_valid(centers, radii):
    """Minimally shrink radii to guarantee strict validity against numerical drift."""
    n = centers.shape[0]
    for _ in range(150):
        valid = True
        for i in range(n):
            if radii[i] < 0:
                valid = False; break
            if centers[i,0] < radii[i] - 1e-11 or centers[i,0] > 1.0 - radii[i] + 1e-11:
                valid = False; break
            if centers[i,1] < radii[i] - 1e-11 or centers[i,1] > 1.0 - radii[i] + 1e-11:
                valid = False; break
        if valid:
            for i in range(n):
                for j in range(i + 1, n):
                    d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
                    if d < radii[i] + radii[j] - 1e-11:
                        valid = False; break
                if not valid:
                    break
        if valid:
            return centers, radii
        # Shrink radii slightly to resolve violations
        radii *= 0.999
    return centers, radii

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    bounds = get_bounds()

        try:

                           constraints=cons, options={'maxiter': 20000, 'ftol': 1e-12, 'disp': False})
            
            if not np.isnan(res.fun):
                curr_sum = -res.fun

                c_vals = constraint_func(res.x)

                pass

    # Fallback (should not be reached)

        best_x = make_hex_init(0, 0.0)
        best_sum = -objective(best_x)

    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    centers[:, 0] = best_x[0::3]
    centers[:, 1] = best_x[1::3]
    radii[:] = best_x[2::3]

    final_sum = float(np.sum(radii))
    
    return centers, radii, final_sum
