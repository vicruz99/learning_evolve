import math

N_CIRCLES = 26

    return -np.sum(x[2::3])

def constraints(x):
    """Vectorized inequality constraints: g(x) >= 0."""

    xi = x[0::3]
    yi = x[1::3]
    ri = x[2::3]

    b = np.concatenate([
        xi - ri,
        1.0 - xi - ri,
        yi - ri,
        1.0 - yi - ri
    ])
    
    # Overlap constraints: dist^2 - (r1+r2)^2 >= 0
    dx = xi[:, None] - xi[None, :]
    dy = yi[:, None] - yi[None, :]
    dr = ri[:, None] + ri[None, :]

    o = dx[i, j]**2 + dy[i, j]**2 - dr[i, j]**2
