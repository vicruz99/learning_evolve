import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def objective(vars):
    """Minimize negative sum of radii."""
    return -np.sum(vars[2::3])

def constraint_func(vars):
    """Vectorized inequality constraints: g(vars) >= 0."""
    n = N_CIRCLES
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    # Boundary constraints: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    bcons = np.concatenate([
        x - r,
        1.0 - x - r,
        y - r,
        1.0 - y - r
    ])
    
    # Overlap constraints: dist^2 - (r1+r2)^2 >= 0
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dr = r[:, None] + r[None, :]
    
    # Extract strictly lower triangle indices (avoids self and duplicates)
    i, j = np.tril_indices(n, -1)
    ocons = dx[i, j]**2 + dy[i, j]**2 - dr[i, j]**2
    
    return np.concatenate([bcons, ocons])

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [0, 0.5]."""
    bounds = []
    for _ in range(N_CIRCLES):
        bounds.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
    return bounds

def generate_init(seed):
    """Generates a diverse, strictly feasible initial configuration."""
    np.random.seed(seed)
    centers = np.zeros((N_CIRCLES, 2))
    
    if seed % 3 == 0:
        # Hexagonal lattice
        r0 = 0.085
        y = r0
        row = 0
        idx = 0
        while idx < N_CIRCLES and y + r0 <= 1.0:
            x_start = r0 + (row % 2) * r0
            x = x_start
            while idx < N_CIRCLES and x + r0 <= 1.0:
                centers[idx] = [x, y]
                idx += 1
                x += 2.0 * r0
            y += np.sqrt(3) * r0
            row += 1
        while idx < N_CIRCLES:
            centers[idx] = np.random.uniform(0.2, 0.8, 2)
            idx += 1
    elif seed % 3 == 1:
        # Staggered grid
        idx = 0
        for i in range(6):
            for j in range(5):
                if idx < N_CIRCLES:
                    x = 0.1 + j * 0.18 + (i % 2) * 0.09
                    y = 0.1 + i * 0.15
                    centers[idx] = [x, y]
                    idx += 1
    else:
        # Random placement
        centers = np.random.uniform(0.15, 0.85, (N_CIRCLES, 2))
        
    # Perturb to break symmetries
    centers += np.random.normal(0, 0.004, centers.shape)
    
    # Ensure strict initial feasibility
    r_init = 0.08
    centers = np.clip(centers, r_init + 1e-6, 1.0 - r_init - 1e-6)
    
    x0 = np.zeros(3 * N_CIRCLES)
    x0[0::3] = centers[:, 0]
    x0[1::3] = centers[:, 1]
    x0[2::3] = r_init
    return x0

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    n = N_CIRCLES
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraint_func}
    
    best_val = -np.inf
    best_x = None
    
    # Phase 1: Multi-start optimization from diverse geometries
    for seed in range(80):
        x0 = generate_init(seed)
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, 
                           constraints=cons, options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
            
            if not np.isnan(res.fun) and -res.fun > best_val:
                best_val = -res.fun
                best_x = res.x.copy()
        except Exception:
            continue

    # Phase 2: Local refinement to escape shallow local minima
    if best_x is not None:
        for _ in range(10):
            noisy = best_x + np.random.normal(0, 1e-4, 3 * n)
            for i in range(n):
                r = max(1e-5, noisy[3*i+2])
                noisy[3*i] = np.clip(noisy[3*i], r, 1.0 - r)
                noisy[3*i+1] = np.clip(noisy[3*i+1], r, 1.0 - r)
                noisy[3*i+2] = r
                
            try:
                res = minimize(objective, noisy, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 3000, 'ftol': 1e-13, 'disp': False})
                if not np.isnan(res.fun) and -res.fun > best_val:
                    best_val = -res.fun
                    best_x = res.x.copy()
            except Exception:
                pass

    # Extract results
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        centers[i, 0] = best_x[3*i]
        centers[i, 1] = best_x[3*i+1]
        radii[i] = max(0.0, best_x[3*i+2])
        
    # Phase 3: Strict validity check and numerical repair
    for _ in range(100):
        valid = True
        for i in range(n):
            if radii[i] < 1e-9:
                radii[i] = 1e-9
            centers[i, 0] = np.clip(centers[i, 0], radii[i], 1.0 - radii[i])
            centers[i, 1] = np.clip(centers[i, 1], radii[i], 1.0 - radii[i])
            
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                if dist < radii[i] + radii[j] - 1e-12:
                    valid = False
                    break
            if not valid:
                break
                
        if valid:
            break
        # Minimally shrink radii to resolve micro-overlaps from numerical drift
        radii *= 0.999
        
    return centers, radii, float(np.sum(radii))
