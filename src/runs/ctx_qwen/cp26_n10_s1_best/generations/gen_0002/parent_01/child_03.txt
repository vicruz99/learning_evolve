1. **Diverse High-Quality Initializations**: Instead of random starts, we generate structured initial packings using rotated hexagonal lattices and a physics-based force-directed simulation with uniformly growing radii. This places circles in dense, feasible configurations close to optimal boundaries.
2. **Multi-Start SLSQP Optimization**: We run `scipy.optimize.minimize` with the SLSQP method from each initialization. SLSQP handles non-linear constraints well, and multiple starts increase the probability of finding the global optimum.
3. **Constraint Formulation**: We use squared distance constraints `dx^2 + dy^2 - (ri + rj)^2 >= 0` for smoothness, combined with boundary constraints `x >= r`, `1-x >= r`, etc.
4. **Systematic Shaking/Refinement**: After finding the best solution, we perturb it with decreasing noise scales and re-optimize. This helps escape shallow local minima by exploring the surrounding basin of attraction.
5. **Numerical Safety**: A strict validation check ensures circles don't overlap or exit the square. If numerical drift causes micro-violations, radii are minimally scaled down and centers clipped to guarantee strict compliance with the validator's `1e-12` tolerance.

import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraint_func(x):
    """Vectorized inequality constraints: g(x) >= 0."""
    n = N_CIRCLES
    cs = x.reshape(n, 3)
    cx, cy, cr = cs[:, 0], cs[:, 1], cs[:, 2]
    
    # Boundary constraints: cx-cr >= 0, 1-cx-cr >= 0, cy-cr >= 0, 1-cy-cr >= 0
    bcons = np.concatenate([
        cx - cr,
        1.0 - cx - cr,
        cy - cr,
        1.0 - cy - cr
    ])
    
    # Overlap constraints: dist^2 - (ri+rj)^2 >= 0
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dr = cr[:, None] + cr[None, :]
    
    # Strictly lower triangle indices to avoid duplicates and self-comparison
    i, j = np.tril_indices(n, -1)
    ocons = dx[i, j]**2 + dy[i, j]**2 - dr[i, j]**2
    
    return np.concatenate([bcons, ocons])

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [0, 0.5]."""
    return [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N_CIRCLES

def check_validity(centers, radii, tol=1e-11):
    """Check strict feasibility of a packing configuration."""
    n = centers.shape[0]
    for i in range(n):
        if radii[i] < 0: return False
        if centers[i, 0] < radii[i] - tol or centers[i, 0] > 1 - radii[i] + tol: return False
        if centers[i, 1] < radii[i] - tol or centers[i, 1] > 1 - radii[i] + tol: return False
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            if d < radii[i] + radii[j] - tol: return False
    return True

def generate_hex_init(n, seed, angle=0.0):
    """Generate a rotated hexagonal lattice initialization."""
    np.random.seed(seed)
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.092)
    idx = 0
    r = 0.09
    y = r
    row = 0
    while idx < n and y + r <= 1.0:
        x = r if row % 2 == 0 else 2.0 * r
        while idx < n and x + r <= 1.0:
            centers[idx] = [x, y]
            idx += 1
            x += 2.0 * r
        y += np.sqrt(3) * r
        row += 1
    while idx < n:
        centers[idx] = np.random.uniform(0.15, 0.85, 2)
        idx += 1
        
    # Rotate and perturb
    cx, cy = 0.5, 0.5
    ca, sa = np.cos(angle), np.sin(angle)
    for i in range(n):
        dx = centers[i, 0] - cx
        dy = centers[i, 1] - cy
        centers[i, 0] = dx * ca - dy * sa + cx + np.random.uniform(-0.012, 0.012)
        centers[i, 1] = dx * sa + dy * ca + cy + np.random.uniform(-0.012, 0.012)
        centers[i, 0] = np.clip(centers[i, 0], radii[i], 1.0 - radii[i])
        centers[i, 1] = np.clip(centers[i, 1], radii[i], 1.0 - radii[i])
    return centers, radii

def generate_force_init(n, seed):
    """Generate initial configuration using repulsive force simulation with growing radius."""
    np.random.seed(seed)
    centers = np.random.uniform(0.2, 0.8, (n, 2))
    r = 0.015
    for _ in range(2500):
        r += 0.000025
        forces = np.zeros_like(centers)
        for i in range(n):
            # Wall repulsion
            if centers[i,0] < r: forces[i,0] += 30*(r - centers[i,0])
            elif centers[i,0] > 1-r: forces[i,0] -= 30*(centers[i,0] - (1-r))
            if centers[i,1] < r: forces[i,1] += 30*(r - centers[i,1])
            elif centers[i,1] > 1-r: forces[i,1] -= 30*(centers[i,1] - (1-r))
            
            # Pair repulsion
            for j in range(i+1, n):
                dx = centers[i,0] - centers[j,0]
                dy = centers[i,1] - centers[j,1]
                dist = np.hypot(dx, dy)
                min_d = 2*r
                if dist < min_d and dist > 1e-8:
                    f = 20 * (min_d - dist) / dist
                    forces[i,0] += f * dx
                    forces[i,1] += f * dy
                    forces[j,0] -= f * dx
                    forces[j,1] -= f * dy
                    
        centers += forces * 0.012 + np.random.normal(0, 0.0004, centers.shape)
        centers = np.clip(centers, r, 1-r)
    return centers, np.full(n, r)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    n = N_CIRCLES
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraint_func}
    
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Prepare diverse initial configurations
    inits = []
    for s in range(6):
        inits.append(generate_hex_init(n, s, angle=0.0))
        inits.append(generate_hex_init(n, s, angle=0.15))
        inits.append(generate_force_init(n, s))
        
    for c_init, r_init in inits:
        x0 = np.zeros(3 * n)
        x0[0::3] = c_init[:, 0]
        x0[1::3] = c_init[:, 1]
        x0[2::3] = r_init
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, 
                           constraints=cons, options={'maxiter': 12000, 'ftol': 1e-13, 'disp': False})
            
            if not np.isnan(res.fun):
                curr = -res.fun
                if curr > best_sum:
                    c_opt = res.x[:2*n].reshape(n, 2)
                    r_opt = res.x[2*n:]
                    # Quick validity check
                    if check_validity(c_opt, r_opt):
                        best_sum = curr
                        best_centers = c_opt.copy()
                        best_radii = r_opt.copy()
        except Exception:
            continue

    # Refinement phase: systematic shaking to escape local minima
    if best_centers is not None:
        for k in range(10):
            scale = 0.008 / (k + 1)
            c_pert = best_centers + np.random.normal(0, scale, best_centers.shape)
            r_pert = best_radii * 0.985
            c_pert = np.clip(c_pert, r_pert, 1.0 - r_pert)
            x0 = np.zeros(3 * n)
            x0[0::3] = c_pert[:, 0]
            x0[1::3] = c_pert[:, 1]
            x0[2::3] = r_pert
            
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 8000, 'ftol': 1e-13, 'disp': False})
                if not np.isnan(res.fun):
                    curr = -res.fun
                    if curr > best_sum:
                        c_opt = res.x[:2*n].reshape(n, 2)
                        r_opt = res.x[2*n:]
                        if check_validity(c_opt, r_opt):
                            best_sum = curr
                            best_centers = c_opt.copy()
                            best_radii = r_opt.copy()
            except Exception:
                continue

    # Final safety correction for numerical drift
    if best_centers is not None:
        for _ in range(30):
            if check_validity(best_centers, best_radii, tol=1e-12):
                break
            best_radii *= 0.9995
            best_centers[:, 0] = np.clip(best_centers[:, 0], best_radii, 1.0 - best_radii)
            best_centers[:, 1] = np.clip(best_centers[:, 1], best_radii, 1.0 - best_radii)
            
    # Fallback guarantee
    if best_centers is None:
        best_centers, best_radii = generate_hex_init(n, 0)
        best_sum = np.sum(best_radii)
        
    return best_centers, best_radii, float(np.sum(best_radii))
