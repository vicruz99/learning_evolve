from scipy.optimize import minimize

N_CIRCLES = 26

def objective(vars):
    """Minimize negative sum of radii."""
    return -np.sum(vars[2::3])

def constraint_func(vars):
    """Vectorized inequality constraints: g(vars) >= 0."""
    n = N_CIRCLES
    cs = vars.reshape(n, 3)
    x, y, r = cs[:, 0], cs[:, 1], cs[:, 2]
    
    # Boundary constraints: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    bcons = np.concatenate([
        x - r,
        1.0 - x - r,
        y - r,
        1.0 - y - r
    ])
    
    # Overlap constraints: dist^2 - (r1+r2)^2 >= 0
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dr = r[:, None] + r[None, :]
    
    # Extract strictly lower triangle indices to avoid duplicates and self-comparison
    i, j = np.tril_indices(n, -1)
    ocons = dx[i, j]**2 + dy[i, j]**2 - dr[i, j]**2
    
    return np.concatenate([bcons, ocons])

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-6, 0.5]."""
    bounds = []
    for _ in range(N_CIRCLES):
        bounds.extend([(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)])
    return bounds

def generate_hex_init(r_start, rotation, noise_scale):
    """Generate a dense hexagonal lattice initialization with rotation and noise."""
    centers = []
    y = r_start
    row = 0
    while len(centers) < N_CIRCLES and y + r_start <= 1.0:
        x_start = r_start if row % 2 == 0 else 2 * r_start
        while len(centers) < N_CIRCLES:
            if x_start + r_start > 1.0:
                break
            centers.append([x_start, y])
            x_start += 2 * r_start
        y += r_start * np.sqrt(3)
        row += 1
        
    centers = np.array(centers[:N_CIRCLES])
    radii = np.full(N_CIRCLES, r_start)
    
    # Apply rotation around square center
    if rotation != 0.0:

            dx, dy = centers[k, 0] - cx, centers[k, 1] - cy
            centers[k, 0] = dx * ca - dy * sa + cx
            centers[k, 1] = dx * sa + dy * ca + cy

    radii += np.random.normal(0, noise_scale / 2, N_CIRCLES)

    x0 = np.zeros(3 * N_CIRCLES)
    x0[0::3] = centers[:, 0]
    x0[1::3] = centers[:, 1]
    x0[2::3] = radii
    return x0
