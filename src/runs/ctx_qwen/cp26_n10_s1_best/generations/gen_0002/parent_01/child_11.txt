import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def objective(vars):
    """Minimize negative sum of radii."""
    return -np.sum(vars[2::3])

def constraints(vars):
    """Vectorized inequality constraints: g(vars) >= 0."""
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    # Boundary constraints: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    b = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = x[:, None] - x
    dy = y[:, None] - y
    dr = r[:, None] + r
    
    idx = np.tril_indices(N_CIRCLES, -1)
    o = dx[idx]**2 + dy[idx]**2 - dr[idx]**2
    
    return np.concatenate([b, o])

def get_bounds():
    bounds = []
    for _ in range(N_CIRCLES):
        bounds.extend([(0.0, 1.0), (0.0, 1.0), (1e-5, 0.5)])
    return bounds

def make_hex_init(seed, angle=0.0):
    np.random.seed(seed)
    r = 0.092
    centers = []
    y = r
    row = 0
    while len(centers) < N_CIRCLES + 5 and y <= 1.0 - r:
        x = r + (row % 2) * r
        while x <= 1.0 - r:
            centers.append([x, y])
            x += 2 * r
        y += np.sqrt(3) * r
        row += 1
        
    centers = np.array(centers[:N_CIRCLES])
    centers += np.random.normal(0, 0.006, centers.shape)
    
    if angle != 0.0:
        cx, cy = 0.5, 0.5
        c, s = np.cos(angle), np.sin(angle)
        dx = centers[:, 0] - cx
        dy = centers[:, 1] - cy
        centers[:, 0] = dx * c - dy * s + cx
        centers[:, 1] = dx * s + dy * c + cy
        
    centers = np.clip(centers, r, 1.0 - r)
    return centers, np.full(N_CIRCLES, r)

def make_force_init(seed):

        centers = np.clip(centers, 0.05, 0.95)

        mind = min(centers[i,0], 1.0 - centers[i,0], centers[i,1], 1.0 - centers[i,1])

            if i != j:
                d = np.hypot(centers[i,0] - centers[j,0], centers[i,1] - centers[j,1])
                if d < mind: mind = d
        radii[i] = mind / 2.0 * 0.96
    return centers, radii
