import numpy as np
from scipy.optimize import minimize

N = 26

def objective(vars):
    """Objective: minimize negative sum of radii."""
    return -np.sum(vars[2::3])

def constraint_func(vars):
    """
    Computes all inequality constraints g(vars) >= 0.
    Returns 1D array of shape (429,) containing boundary and overlap constraints.
    """
    n = N
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    b = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dr = r[:, None] + r[None, :]
    
    i_idx, j_idx = np.tril_indices(n, -1)
    o = dx[i_idx, j_idx]**2 + dy[i_idx, j_idx]**2 - dr[i_idx, j_idx]**2
    
    return np.concatenate([b, o])

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-6, 0.5]"""
    return [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N

def make_hex_init(seed, r_start):
    """Generates a perturbed hexagonal lattice initialization."""
    np.random.seed(seed)
    centers = []
    y = r_start
    row = 0
    while len(centers) < N and y + r_start <= 1.0:
        x = r_start + (row % 2) * r_start
        while len(centers) < N and x + r_start <= 1.0:
            centers.append([x, y])
            x += 2.0 * r_start
        y += np.sqrt(3.0) * r_start
        row += 1
        
    while len(centers) < N:
        centers.append([np.random.uniform(0.2, 0.8), np.random.uniform(0.2, 0.8)])
        
    centers = np.array(centers[:N]) + np.random.normal(0, 1e-4, centers.shape)
    centers = np.clip(centers, r_start + 1e-6, 1.0 - r_start - 1e-6)
    radii = np.full(N, r_start)
    return centers, radii

def make_force_init(seed):
    """Generates initialization via force-directed repulsion."""
    np.random.seed(seed)
    centers = np.random.uniform(0.2, 0.8, (N, 2))
    radii = np.full(N, 0.05)

        forces = np.zeros_like(centers)
        for i in range(N):
            for j in range(i + 1, N):
                diff = centers[j] - centers[i]
                dist = np.linalg.norm(diff)

                    repulse = 0.02 / (dist**2)
                    unit = diff / dist

                    forces[j] += unit * repulse
        centers += forces * 0.05
        centers = np.clip(centers, 0.05, 0.95)
        
    for i in range(N):
        min_d = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        for j in range(N):
            if i != j:
                d = np.linalg.norm(centers[i] - centers[j])
                if d < min_d:
                    min_d = d
        radii[i] = min_d / 2.0 * 0.85
        
    return centers, radii

def run_packing():
    n = N
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraint_func}
    
    best_val = -np.inf
    best_vars = None

    inits = []
    for s in range(12):

        x0[2::3] = radii
