from scipy.optimize import minimize

def objective(vars):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(vars[2::3])

def constraints(vars):
    """
    Compute all inequality constraints g(vars) >= 0.
    Returns a 1D array containing:
    - Boundary constraints (4*n)
    - Squared overlap constraints (n*(n-1)/2)
    """
    n = 26
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    # Boundary constraints: circle inside [0,1]x[0,1]
    c = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    # Vectorized computation for efficiency
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dr = r[:, None] + r[None, :]
    
    # Extract strictly lower triangle indices
    i_idx, j_idx = np.tril_indices(n, -1)
    co = dx[i_idx, j_idx]**2 + dy[i_idx, j_idx]**2 - dr[i_idx, j_idx]**2
    
    return np.concatenate([c, co])
