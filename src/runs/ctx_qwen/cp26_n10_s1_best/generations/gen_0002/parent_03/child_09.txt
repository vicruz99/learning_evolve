N = 26

def objective(vars):
    """Minimize negative sum of radii."""
    return -np.sum(vars[2::3])

def constraint_func(vars):
    """Compute all boundary and non-overlap constraints g(vars) >= 0."""
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    num_overlaps = N * (N - 1) // 2
    c = np.empty(4 * N + num_overlaps)

    c[:N] = x - r
    c[N:2*N] = 1.0 - x - r
    c[2*N:3*N] = y - r
    c[3*N:4*N] = 1.0 - y - r

    i_idx, j_idx = np.tril_indices(N, -1)
    dx = x[i_idx] - x[j_idx]
    dy = y[i_idx] - y[j_idx]
    dr = r[i_idx] + r[j_idx]
    c[4*N:] = dx*dx + dy*dy - dr*dr
    
    return c

def get_bounds():
