import numpy as np

N = 26

def objective(x, n):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraint_vec(x, n):
    """Vectorized inequality constraints: g(x) >= 0."""
    cs = x.reshape(n, 3)
    xc = cs[:, 0]
    yc = cs[:, 1]
    rc = cs[:, 2]
    
    # Boundary constraints
    b = np.concatenate([
        xc - rc,
        1.0 - xc - rc,
        yc - rc,
        1.0 - yc - rc
    ])
    
    # Overlap constraints (squared form)
    dx = xc[:, None] - xc[None, :]
    dy = yc[:, None] - yc[None, :]
    dr = rc[:, None] + rc[None, :]
    
    i_idx, j_idx = np.tril_indices(n, -1)
    o = dx[i_idx, j_idx]**2 + dy[i_idx, j_idx]**2 - dr[i_idx, j_idx]**2
    
    return np.concatenate([b, o])

def get_bounds(n):
    b = []
    for _ in range(n):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-8, 0.5)])
    return b

def hex_pattern_init(n, pattern, r_start, noise_scale=0.005):
    """Generate hexagonal initialization with given row pattern."""
    centers = np.zeros((n, 2))
    radii = np.full(n, r_start)
    
    dy = np.sqrt(3) * r_start
    y = r_start
    idx = 0
    
    for row_i, count in enumerate(pattern):
        x_start = r_start + (r_start if row_i % 2 == 1 else 0.0)
        for col in range(count):
            if idx < n:
                x = x_start + col * 2 * r_start
                centers[idx, 0] = x
                centers[idx, 1] = y
                idx += 1
        y += dy
        if y + r_start > 1.0:
            break
    
    while idx < n:
        centers[idx, 0] = r_start + (idx % 5) * 0.2
        centers[idx, 1] = r_start + (idx // 5) * 0.2
        radii[idx] = r_start * 0.7
        idx += 1
    
    centers += np.random.normal(0, noise_scale, centers.shape)
    centers = np.clip(centers, r_start + 1e-6, 1.0 - r_start - 1e-6)
    
    x0 = np.zeros(3 * n)
    x0[0::3] = centers[:, 0]
    x0[1::3] = centers[:, 1]
    x0[2::3] = radii
    return x0

def force_directed_init(n, seed, steps=3000):
    """Generate initialization via force-directed relaxation."""
    np.random.seed(seed)
    centers = np.random.uniform(0.15, 0.85, (n, 2))
    radii = np.full(n, 0.04)
    
    for step in range(steps):
        forces = np.zeros((n, 2))
        alpha = 0.02 * (1.0 - step / steps)  # cooling schedule
        
        for i in range(n):
            # Boundary repulsion
            for dim in range(2):
                if centers[i, dim] - radii[i] < 0.02:
                    forces[i, dim] += 0.01 * (0.02 - (centers[i, dim] - radii[i]))
                if centers[i, dim] + radii[i] > 0.98:
                    forces[i, dim] -= 0.01 * ((centers[i, dim] + radii[i]) - 0.98)
            
            for j in range(i + 1, n):
                dx = centers[j, 0] - centers[i, 0]
                dy = centers[j, 1] - centers[i, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                min_dist = radii[i] + radii[j]
                
                if dist < min_dist * 1.5 and dist > 1e-8:
                    overlap = min_dist - dist
                    if overlap > 0:
                        f = overlap * 2.0
                        fx, fy = f * dx / dist, f * dy / dist
                        forces[i] -= [fx, fy]
                        forces[j] += [fx, fy]
        
        centers += forces * alpha
        centers = np.clip(centers, 0.02, 0.98)
        
        # Gradually grow radii
        if step % 100 == 0 and step > 0:
            for i in range(n):
                max_r = min(centers[i, 0], 1.0 - centers[i, 0], 
                            centers[i, 1], 1.0 - centers[i, 1])
                for j in range(n):
                    if i != j:
                        d = np.sqrt((centers[i,0]-centers[j,0])**2 + (centers[i,1]-centers[j,1])**2)
                        max_r = min(max_r, (d - radii[j]) / 2.0)
                if max_r > 0:
                    radii[i] = max(radii[i], min(max_r * 0.95, radii[i] * 1.05 + 0.001))
    
    x0 = np.zeros(3 * n)
    x0[0::3] = centers[:, 0]
    x0[1::3] = centers[:, 1]
    x0[2::3] = radii
    return x0

def grid_init(n, r_start, noise_scale=0.003):
    """Generate grid initialization."""
    centers = np.zeros((n, 2))
    radii = np.full(n, r_start)
    
    idx = 0
    for row in range(6):
        for col in range(5):
            if idx < n:
                centers[idx, 0] = r_start + col * 2 * r_start
                centers[idx, 1] = r_start + row * 2 * r_start
                idx += 1
    
    while idx < n:
        centers[idx] = np.random.uniform(r_start, 1-r_start, 2)
        radii[idx] = r_start * 0.6
        idx += 1
    
    centers += np.random.normal(0, noise_scale, centers.shape)
    centers = np.clip(centers, r_start + 1e-6, 1.0 - r_start - 1e-6)
    
    x0 = np.zeros(3 * n)
    x0[0::3] = centers[:, 0]
    x0[1::3] = centers[:, 1]
    x0[2::3] = radii
    return x0

def rotated_hex_init(n, pattern, r_start, angle, noise_scale=0.003):
    """Generate rotated hexagonal initialization."""
    x0 = hex_pattern_init(n, pattern, r_start, noise_scale=0.0)
    
    centers = np.zeros((n, 2))
    centers[:, 0] = x0[0::3]
    centers[:, 1] = x0[1::3]
    radii = x0[2::3].copy()
    
    cx, cy = 0.5, 0.5
    ca, sa = np.cos(angle), np.sin(angle)
    for i in range(n):
        dx = centers[i, 0] - cx
        dy = centers[i, 1] - cy
        centers[i, 0] = dx * ca - dy * sa + cx
        centers[i, 1] = dx * sa + dy * ca + cy
    
    centers = np.clip(centers, radii + 1e-6, 1.0 - radii - 1e-6)
    
    x0[0::3] = centers[:, 0]
    x0[1::3] = centers[:, 1]
    return x0

def corner_anchored_init(n, seed):
    """Place larger circles in corners and fill interior."""
    np.random.seed(seed)
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Corner circles
    corner_r = 0.14
    corners = [[corner_r, corner_r], [1-corner_r, corner_r], 
               [corner_r, 1-corner_r], [1-corner_r, 1-corner_r]]
    for i, c in enumerate(corners):
        centers[i] = c
        radii[i] = corner_r
    
    # Fill interior with smaller circles in hexagonal pattern
    interior_r = 0.075
    idx = 4
    y = interior_r + corner_r + 0.01
    row = 0
    while idx < n and y + interior_r < 1.0 - corner_r - 0.01:
        x_start = interior_r + corner_r + 0.01 + (interior_r if row % 2 == 1 else 0.0)
        x = x_start
        while idx < n and x + interior_r < 1.0 - corner_r - 0.01:
            centers[idx, 0] = x
            centers[idx, 1] = y
            radii[idx] = interior_r
            idx += 1
            x += 2 * interior_r
        y += np.sqrt(3) * interior_r
        row += 1
    
    # Fill remaining

        cx, cy = np.random.uniform(0.15, 0.85, 2)
        r = 0.04
        centers[idx] = [cx, cy]
        radii[idx] = r
        idx += 1
    
    centers += np.random.normal(0, 0.002, centers.shape)
    for i in range(n):
        centers[i, 0] = np.clip(centers[i, 0], radii[i], 1.0 - radii[i])
        centers[i, 1] = np.clip(centers[i, 1], radii[i], 1.0 - radii[i])
    
    x0 = np.zeros(3 * n)
    x0[0::3] = centers[:, 0]
    x0[1::3] = centers[:, 1]
    x0[2::3] = radii
    return x0

def project_feasible(x, n):
    """Project variables to feasible region."""
    x = x.copy()
    for i in range(n):
        r = max(x[3*i+2], 1e-6)
        x[3*i+2] = r
        x[3*i] = np.clip(x[3*i], r, 1.0 - r)
        x[3*i+1] = np.clip(x[3*i+1], r, 1.0 - r)
    return x

def run_optimize(x0, n, bounds, cons, maxiter=15000, ftol=1e-13):
    """Run SLSQP optimization with error handling."""
    try:
        res = minimize(objective, x0, args=(n,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol, 'disp': False})
        return res
    except Exception:
        return None

def check_validity(centers, radii, n):
    """Check if packing is valid."""
    for i in range(n):
        if radii[i] < 0:
            return False
        if centers[i,0] - radii[i] < -1e-12 or centers[i,0] + radii[i] > 1.0 + 1e-12:
            return False
        if centers[i,1] - radii[i] < -1e-12 or centers[i,1] + radii[i] > 1.0 + 1e-12:
            return False
    
    for i in range(n):
        for j in range(i+1, n):
            d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
            if d < radii[i] + radii[j] - 1e-12:
                return False
    return True
