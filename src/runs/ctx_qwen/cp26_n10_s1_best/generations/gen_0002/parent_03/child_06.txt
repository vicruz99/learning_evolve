import numpy as np
from scipy.optimize import minimize

def objective(vars):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(vars[2::3])

def constraints(vars, n):
    """Inequality constraints: boundary and non-overlap."""
    xi = vars[0::3]
    yi = vars[1::3]
    ri = vars[2::3]
    
    num_ineq = n * 4 + n * (n - 1) // 2
    res = np.empty(num_ineq)
    
    # Boundary constraints: circle inside [0,1]x[0,1]
    res[:n] = xi - ri
    res[n:2*n] = 1.0 - xi - ri
    res[2*n:3*n] = yi - ri
    res[3*n:4*n] = 1.0 - yi - ri
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = xi[:, None] - xi[None, :]
    dy = yi[:, None] - yi[None, :]
    dr = ri[:, None] + ri[None, :]
    
    i_idx, j_idx = np.tril_indices(n, -1)
    res[4*n:] = dx[i_idx, j_idx]**2 + dy[i_idx, j_idx]**2 - dr[i_idx, j_idx]**2
    
    return res

def generate_force_init(n, seed):
    """Generate a valid, dense initial configuration using repulsive forces."""
    np.random.seed(seed)
    centers = np.random.uniform(0.2, 0.8, (n, 2))
    radii = np.full(n, 0.07)
    dt = 0.015
    
    for step in range(500):
        radii[:] = np.clip(radii + 0.00012, 0, 0.4)
        
        diff = centers[:, None, :] - centers[None, :, :]
        dists = np.linalg.norm(diff, axis=2)
        np.fill_diagonal(dists, np.inf)
        
        min_dists = radii[:, None] + radii[None, :]
        overlaps = np.maximum(min_dists - dists, 0.0)
        
        safe_dists = np.where(dists > 1e-6, dists, 1.0)
        mask = dists > 1e-6
        
        forces = np.zeros_like(centers)
        f_mag = np.zeros_like(dists)
        f_mag[mask] = overlaps[mask] / safe_dists[mask]
        forces += np.sum(f_mag[:, :, None] * diff / safe_dists[:, :, None], axis=1) * 8.0
        
        # Wall repulsion
        for i in range(n):
            if centers[i, 0] < radii[i]: forces[i, 0] += (radii[i] - centers[i, 0]) * 15.0
            if centers[i, 0] > 1 - radii[i]: forces[i, 0] -= (centers[i, 0] - (1 - radii[i])) * 15.0
            if centers[i, 1] < radii[i]: forces[i, 1] += (radii[i] - centers[i, 1]) * 15.0
            if centers[i, 1] > 1 - radii[i]: forces[i, 1] -= (centers[i, 1] - (1 - radii[i])) * 15.0
            
        centers += forces * dt
        centers = np.clip(centers, radii + 1e-5, 1.0 - radii - 1e-5)
        
    return centers, radii

def generate_hex_init(n, seed):
    """Generate a hexagonal lattice initialization with perturbation."""
    np.random.seed(seed)
    r0 = 0.09 + np.random.uniform(-0.005, 0.005)
    centers = []
    y = r0
    row = 0
    while len(centers) < n and y + r0 <= 1.0:
        x = r0 + (row % 2) * r0
        while len(centers) < n and x + r0 <= 1.0:
            centers.append([x, y])
            x += 2 * r0
        y += r0 * np.sqrt(3)
        row += 1
        
    while len(centers) < n:
        centers.append([np.random.uniform(0.15, 0.85), np.random.uniform(0.15, 0.85)])
        
    centers = np.array(centers[:n])
    centers += np.random.normal(0, 0.003, centers.shape)
    centers = np.clip(centers, r0 + 1e-5, 1.0 - r0 - 1e-5)
    
    return centers, np.full(n, r0)
