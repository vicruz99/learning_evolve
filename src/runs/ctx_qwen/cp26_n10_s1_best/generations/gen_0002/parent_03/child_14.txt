import numpy as np
from scipy.optimize import minimize

N = 26

def objective(vars):
    """Minimize negative sum of radii."""
    return -np.sum(vars[2::3])

def constraints(vars):
    """Compute all inequality constraints g(vars) >= 0."""
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    # Boundary constraints: circle inside [0,1]x[0,1]
    c = [x - r, 1.0 - x - r, y - r, 1.0 - y - r]
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = x[:, None] - x
    dy = y[:, None] - y
    dr = r[:, None] + r
    
    # Strictly lower triangle to avoid duplicates and self-comparison
    i_idx, j_idx = np.tril_indices(N, -1)
    c.append(dx[i_idx, j_idx]**2 + dy[i_idx, j_idx]**2 - dr[i_idx, j_idx]**2)
    
    return np.concatenate(c)

def project(vars):
    """Project variables to strictly satisfy boundary and non-negativity constraints."""
    vars = vars.copy()
    r = np.maximum(vars[2::3], 1e-6)
    vars[2::3] = r
    vars[0::3] = np.clip(vars[0::3], r, 1.0 - r)
    vars[1::3] = np.clip(vars[1::3], r, 1.0 - r)
    return vars

def generate_hex(pat, angle, shift, r0):
    """Generate a rotated and shifted hexagonal lattice initialization."""
    centers = np.zeros((N, 2))
    idx = 0
    y = r0
    for i, cnt in enumerate(pat):
        x_start = r0 + (i % 2) * r0
        for k in range(cnt):
            if idx >= N: break
            cx = x_start + k * 2 * r0
            cy = y
            cx += shift[0]
            cy += shift[1]
            
            # Apply rotation around center
            dx, dy = cx - 0.5, cy - 0.5
            cos_a, sin_a = np.cos(angle), np.sin(angle)
            centers[idx, 0] = dx*cos_a - dy*sin_a + 0.5
            centers[idx, 1] = dx*sin_a + dy*cos_a + 0.5
            idx += 1
        y += r0 * np.sqrt(3)
        
    # Fill remaining spots randomly if pattern sums to < N (rare)
    while idx < N:
        centers[idx] = np.random.uniform(0.1, 0.9, 2)
        idx += 1
    return centers

def run_packing():
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_val = -np.inf
    best_vars = None
    
    # Diverse row count patterns that sum to 26
    patterns = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [6, 6, 5, 5, 4],
        [5, 5, 5, 5, 6], [6, 5, 5, 5, 5], [4, 6, 6, 6, 4]
    ]
    
    for seed in range(100):
        np.random.seed(seed)
        pat = patterns[seed % len(patterns)]
        angle = np.random.uniform(-0.2, 0.2)
        shift = np.random.uniform(-0.08, 0.08, 2)
        r0 = 0.09 + np.random.uniform(-0.015, 0.015)
        
        centers = generate_hex(pat, angle, shift, r0)
        x0 = np.zeros(3 * N)
        x0[0::3] = centers[:, 0]
        x0[1::3] = centers[:, 1]
        x0[2::3] = r0
        x0 = project(x0)
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 15000, 'ftol': 1e-14, 'disp': False})
            val = -res.fun
            cvals = constraints(res.x)
            if np.min(cvals) >= -1e-8 and val > best_val:
                best_val = val
                best_vars = res.x.copy()
        except Exception:
            pass
            
        # Perturbation strategy to escape local minima
        if best_vars is not None and seed > 20 and seed % 6 == 0:
            x_pert = best_vars + np.random.normal(0, 0.003, size=best_vars.shape)
            # Scale radii down to break rigid contacts, then re-optimize
            x_pert[2::3] *= 0.96
            x_pert = project(x_pert)
            try:
                res2 = minimize(objective, x_pert, method='SLSQP', bounds=bounds,
                                constraints=cons, options={'maxiter': 15000, 'ftol': 1e-14, 'disp': False})
                val2 = -res2.fun
                cvals2 = constraints(res2.x)

                    best_vars = res2.x.copy()
            except Exception:
                pass
                
    # Fallback configuration
    if best_vars is None:
        best_vars = np.zeros(3*N)
        best_vars[2::3] = 0.09
        best_vars[0::3] = np.tile(np.linspace(0.12, 0.88, 6), 5)[:N]
        best_vars[1::3] = np.repeat(np.linspace(0.12, 0.88, 5), 6)[:N]
        best_vars = project(best_vars)

    radii = best_vars[2::3]

               centers[i,1] < radii[i] - 1e-10 or centers[i,1] > 1 - radii[i] + 1e-10:

                break

                for j in range(i+1, N):
                    if np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1]) < radii[i]+radii[j] - 1e-10:
                        valid = False
                        break
                if not valid:
                    break
        if valid:
            break
        radii *= 0.9995
        
    return centers, radii, float(np.sum(radii))
