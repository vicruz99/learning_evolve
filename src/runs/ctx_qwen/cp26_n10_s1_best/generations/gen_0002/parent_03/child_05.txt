from scipy.optimize import minimize

N = 26

def objective(vars):
    """Minimize negative sum of radii."""
    return -np.sum(vars[2::3])

def cons_func(vars):
    """Compute all inequality constraints as a single vector >= 0."""
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    # Boundary constraints
    b = np.empty(4*N)
    b[0:N] = x - r
    b[N:2*N] = 1.0 - x - r
    b[2*N:3*N] = y - r
    b[3*N:4*N] = 1.0 - y - r
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = x[:, None] - x
    dy = y[:, None] - y
    dr = r[:, None] + r
    
    tri = np.tril_indices(N, -1)
    o = dx[tri]**2 + dy[tri]**2 - dr[tri]**2
    
    return np.concatenate([b, o])

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-6, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)])
    return b

def hex_init(counts, r0, shift=0.0):
    """Generate a hexagonal lattice initial configuration."""
    centers = np.zeros((N, 2))
    radii = np.full(N, r0)
    idx = 0
    y = r0
    for i, cnt in enumerate(counts):
        x_start = r0 + (0.5*r0 if i%2==1 else 0.0) + shift
        for k in range(cnt):
            if idx < N:
                centers[idx] = [x_start + k*2*r0, y]
                idx += 1
        y += r0 * np.sqrt(3.0)
    return centers, radii

def rotate(centers, angle):
    """Rotate centers around the square center (0.5, 0.5)."""
    c, s = np.cos(angle), np.sin(angle)
    dx = centers[:, 0] - 0.5
    dy = centers[:, 1] - 0.5
    nc = np.zeros_like(centers)
    nc[:, 0] = dx*c - dy*s + 0.5
    nc[:, 1] = dx*s + dy*c + 0.5
    return nc

def force_init(n, seed):
    """Force-directed initialization to pack circles densely before optimization."""
    np.random.seed(seed)
    centers = np.random.uniform(0.2, 0.8, (n, 2))
    radii = np.full(n, 0.08)
    
    for _ in range(400):
        forces = np.zeros((n, 2))
        for i in range(n):
            for j in range(i+1, n):
                dx = centers[j,0] - centers[i,0]
                dy = centers[j,1] - centers[i,1]

                    forces[j] += [fx, fy]

                elif centers[i,d] > 1.0 - radii[i] - 0.05: forces[i,d] -= 0.005
                
        centers += forces * 0.01
        centers = np.clip(centers, 0.05, 0.95)
        
    # Estimate safe radii based on nearest neighbors/walls
    for i in range(n):
        min_d = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])

            if i != j:
                d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
                if d < min_d: min_d = d
        radii[i] = min_d / 2.0 * 0.85
    return centers, radii
