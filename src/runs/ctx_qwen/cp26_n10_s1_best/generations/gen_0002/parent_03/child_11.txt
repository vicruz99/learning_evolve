import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def objective(vars):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(vars[2::3])

def constraints(vars):
    """Inequality constraints: boundary and non-overlap."""
    n = N_CIRCLES
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    # Boundary constraints: circle inside [0,1]x[0,1]
    cons = [x - r, 1.0 - x - r, y - r, 1.0 - y - r]
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dr = r[:, None] + r[None, :]
    
    triu = np.triu_indices(n, 1)
    cons.append(dx[triu]**2 + dy[triu]**2 - dr[triu]**2)
    
    return np.concatenate(cons)

def inflate_packing(seed, n=26):
    """Force-directed inflation to generate a tight, feasible initial packing."""
    np.random.seed(seed)
    centers = np.random.uniform(0.2, 0.8, (n, 2))
    radii = np.ones(n) * 0.02
    
    # Gradually grow radii while resolving overlaps
    for _ in range(600):
        radii += 0.0003
        for _ in range(10):
            dx = centers[:, None, :] - centers[None, :, :]
            dist = np.sqrt(np.sum(dx**2, axis=2) + 1e-12)
            r_sum = radii[:, None] + radii[None, :]
            overlap = np.maximum(r_sum - dist, 0)
            
            inv_dist = 1.0 / (dist + 1e-12)
            centers[:, 0] += (dx[:, :, 0] * inv_dist * overlap).sum(axis=1) * 0.1
            centers[:, 1] += (dx[:, :, 1] * inv_dist * overlap).sum(axis=1) * 0.1
            
            # Strict boundary enforcement
            centers[:, 0] = np.clip(centers[:, 0], radii + 1e-7, 1.0 - radii - 1e-7)
            centers[:, 1] = np.clip(centers[:, 1], radii + 1e-7, 1.0 - radii - 1e-7)
            
    return centers, radii

def run_packing():
    n = N_CIRCLES
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * n
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_val = -np.inf
    best_vars = None
    
    # Phase 1: Multi-start optimization from inflated configurations
    for seed in range(20):
        centers, radii = inflate_packing(seed, n)
        x0 = np.zeros(3 * n)
        x0[0::3] = centers[:, 0]
        x0[1::3] = centers[:, 1]
        x0[2::3] = radii
        
        # Break exact symmetries to help SLSQP explore
        x0 += np.random.normal(0, 1e-5, 3*n)
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 10000, 'ftol': 1e-13, 'disp': False})
            if -res.fun > best_val:
                best_val = -res.fun
                best_vars = res.x.copy()
        except Exception:
            pass
            
    # Phase 2: Local refinement to escape shallow local minima
    if best_vars is not None:
        for _ in range(5):
            x0 = best_vars + np.random.normal(0, 1e-4, 3*n)
            # Project perturbed variables back to feasible bounds
            for i in range(n):
                r = max(1e-5, x0[3*i+2])
                x0[3*i+2] = r
                x0[3*i] = np.clip(x0[3*i], r, 1.0-r)
                x0[3*i+1] = np.clip(x0[3*i+1], r, 1.0-r)
                
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
                if -res.fun > best_val:
                    best_val = -res.fun
                    best_vars = res.x.copy()
            except Exception:
                pass
                
    # Extract final configuration
    centers = np.column_stack((best_vars[0::3], best_vars[1::3]))
    radii = best_vars[2::3]
    
    # Phase 3: Strict validity check and numerical repair
    for _ in range(100):
        valid = True

                valid = False
                break

                        valid = False
                        break

    return centers, radii, float(np.sum(radii))
