import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def objective(vars):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(vars[2::3])

def constraints(vars, n):
    """Inequality constraints: boundary and non-overlap (vectorized)."""
    xi = vars[0::3]
    yi = vars[1::3]
    ri = vars[2::3]
    
    # Boundary: x >= r, 1-x >= r, y >= r, 1-y >= r
    b = np.concatenate([xi - ri, 1.0 - xi - ri, yi - ri, 1.0 - yi - ri])
    
    # Overlap: dist^2 >= (r_i + r_j)^2
    dx = xi[:, None] - xi[None, :]
    dy = yi[:, None] - yi[None, :]
    dr = ri[:, None] + ri[None, :]
    
    triu = np.triu_indices(n, k=1)
    o = dx[triu]**2 + dy[triu]**2 - dr[triu]**2
    
    return np.concatenate([b, o])

def generate_hex_init(n, seed, rotation=0.0, scale=1.0):
    """Generate a hexagonal lattice initialization with optional rotation."""
    np.random.seed(seed)
    r_init = 0.095 * scale
    centers = []
    y = r_init
    row = 0
    while len(centers) < n + 5 and y + r_init <= 1.0:
        x = r_init + (row % 2) * r_init
        while len(centers) < n + 5 and x + r_init <= 1.0:
            centers.append([x, y])
            x += 2 * r_init
        y += r_init * np.sqrt(3.0)
        row += 1
        
    centers = np.array(centers[:n])
    
    # Apply rotation around center
    if rotation != 0.0:
        cx, cy = 0.5, 0.5
        cos_r, sin_r = np.cos(rotation), np.sin(rotation)
        shifted = centers - np.array([cx, cy])
        centers = np.array([
            shifted[:, 0] * cos_r - shifted[:, 1] * sin_r + cx,
            shifted[:, 0] * sin_r + shifted[:, 1] * cos_r + cy
        ]).T
        
    # Add noise and ensure feasibility bounds
    centers += np.random.normal(0, 0.005, centers.shape)
    centers = np.clip(centers, r_init, 1.0 - r_init)
    radii = np.full(n, r_init)
    return centers, radii

def generate_jammed_init(n, seed):
    """Generate a dense packing via force-directed relaxation with growing radii."""
    np.random.seed(seed)
    centers = np.random.uniform(0.25, 0.75, (n, 2))
    radii = np.full(n, 0.025)
    
    for _ in range(1500):
        forces = np.zeros_like(centers)

            # Wall repulsion
            for dim in range(2):
                if centers[i, dim] - radii[i] < 0.01:
                    forces[i, dim] += 0.15 * (0.01 - (centers[i, dim] - radii[i]))
                if centers[i, dim] + radii[i] > 0.99:
                    forces[i, dim] -= 0.15 * (centers[i, dim] + radii[i] - 0.99)
                    
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[j, 0] - centers[i, 0]
                dy = centers[j, 1] - centers[i, 1]
                dist = np.hypot(dx, dy)
                req = radii[i] + radii[j]
                if dist < req and dist > 1e-7:
                    f = (req - dist) * 0.4
                    nx, ny = dx / dist, dy / dist
                    forces[i] -= [nx * f, ny * f]
                    forces[j] += [nx * f, ny * f]
                    
        centers += forces * 0.04
        centers = np.clip(centers, 0.015, 0.985)
        radii *= 1.00025
        
    # Estimate final radii from jammed state
    new_radii = np.zeros(n)
    for i in range(n):
        d_min = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        for j in range(n):
            if i != j:
                d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                if d < d_min:
                    d_min = d
        new_radii[i] = d_min / 2.0 * 0.94
    return centers, new_radii

def generate_corner_bias_init(n, seed):

    centers = np.zeros((n, 2))
    radii = np.zeros(n)

        radii[i] = 0.085

        centers[i] = np.random.uniform(0.2, 0.8, 2)

                    nx, ny = dx / dist, dy / dist
                    centers[i] -= [nx * shift, ny * shift]
                    centers[j] += [nx * shift, ny * shift]
    return centers, radii
