import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def objective(vars):
    """Minimize negative sum of radii."""
    return -np.sum(vars[2::3])

def constraint_func(vars, n):
    """Vectorized inequality constraints: g(vars) >= 0."""
    xi = vars[0::3]
    yi = vars[1::3]
    ri = vars[2::3]
    
    num_overlap = n * (n - 1) // 2
    res = np.empty(n * 4 + num_overlap)
    
    res[:n] = xi - ri
    res[n:2*n] = 1.0 - xi - ri
    res[2*n:3*n] = yi - ri
    res[3*n:4*n] = 1.0 - yi - ri
    
    dx = xi[:, None] - xi[None, :]
    dy = yi[:, None] - yi[None, :]
    dr = ri[:, None] + ri[None, :]
    
    i_idx, j_idx = np.tril_indices(n, -1)
    res[4*n:] = dx[i_idx, j_idx]**2 + dy[i_idx, j_idx]**2 - dr[i_idx, j_idx]**2
    
    return res

def get_bounds(n):
    bounds = []
    for _ in range(n):
        bounds.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
    return bounds

def create_hex_init(n, r_init, seed):
    """Hexagonal lattice initialization with perturbation."""
    np.random.seed(seed)
    centers = []
    y = r_init
    row = 0
    while len(centers) < n and y + r_init <= 1.0:
        x = r_init + (row % 2) * r_init
        while len(centers) < n and x + r_init <= 1.0:
            centers.append([x, y])
            x += 2 * r_init
        y += r_init * np.sqrt(3)
        row += 1
    
    while len(centers) < n:
        centers.append([np.random.uniform(r_init, 1.0 - r_init),
                        np.random.uniform(r_init, 1.0 - r_init)])
    
    centers = np.array(centers[:n])
    centers += np.random.normal(0, 0.005, centers.shape)
    centers = np.clip(centers, r_init + 1e-6, 1.0 - r_init - 1e-6)
    
    # Add radius variation to break symmetry
    radii = r_init + np.random.normal(0, 0.008, n)
    radii = np.clip(radii, 0.03, 0.3)
    
    return centers, radii

def create_grid_init(n, seed):
    """Grid-based initialization."""
    np.random.seed(seed)
    centers = []
    r_init = 0.09
    
    idx = 0
    for row in range(6):
        for col in range(5):
            if idx >= n:
                break
            x = 0.09 + col * 0.18 + (row % 2) * 0.045
            y = 0.09 + row * 0.15
            centers.append([x, y])
            idx += 1

        centers.append([np.random.uniform(0.1, 0.9), np.random.uniform(0.1, 0.9)])
    
    centers = np.array(centers[:n])
    centers += np.random.normal(0, 0.008, centers.shape)
    centers = np.clip(centers, 0.03, 0.97)
    
    radii = 0.09 + np.random.normal(0, 0.006, n)
    radii = np.clip(radii, 0.03, 0.2)
    
    return centers, radii

def force_directed_init(n, seed):
    """Force-directed simulation to find good initial positions."""
    np.random.seed(seed)
    
    centers = np.random.uniform(0.2, 0.8, (n, 2))
    radii = np.full(n, 0.06)
    
    dt = 0.02
    repulsion_strength = 0.5
    
    for iteration in range(3000):
        forces = np.zeros((n, 2))
        cooling = max(0.1, 1.0 - iteration / 3000.0)
        
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[j, 0] - centers[i, 0]
                dy = centers[j, 1] - centers[i, 1]
                dist = np.hypot(dx, dy)
                
                if dist > 1e-8:
                    min_dist = radii[i] + radii[j]
                    if dist < min_dist * 2.5:
                        force = repulsion_strength * (min_dist * 2.5 - dist) / (dist * dist + 1e-6)
                        fx = force * dx / dist
                        fy = force * dy / dist
                        forces[i, 0] -= fx
                        forces[i, 1] -= fy
                        forces[j, 0] += fx
                        forces[j, 1] += fy
        
        # Push toward walls for better boundary utilization
        for i in range(n):
            r = radii[i]
            # Corner attraction
            corners = [(r + 0.05, r + 0.05), (1.0 - r - 0.05, r + 0.05),
                       (r + 0.05, 1.0 - r - 0.05), (1.0 - r - 0.05, 1.0 - r - 0.05)]
            nearest_corner = min(corners, key=lambda c: np.hypot(centers[i,0]-c[0], centers[i,1]-c[1]))
            forces[i, 0] += 0.05 * (nearest_corner[0] - centers[i, 0])
            forces[i, 1] += 0.05 * (nearest_corner[1] - centers[i, 1])
        
        centers += forces * dt * cooling
        centers = np.clip(centers, radii[:, None] + 1e-5, 1.0 - radii[:, None] - 1e-5)
    
    # Estimate radii based on nearest neighbors
    for i in range(n):
        min_d = min(centers[i, 0], 1.0 - centers[i, 0],
                    centers[i, 1], 1.0 - centers[i, 1])
        for j in range(n):
            if i != j:
                d = np.hypot(centers[i, 0] - centers[j, 0],
                             centers[i, 1] - centers[j, 1])
                min_d = min(min_d, d / 2.0)
        radii[i] = min_d * 0.95
    
    return centers, radii

def grow_and_pack_init(n, seed):
    """Grow circles from small size to find good packing."""
    np.random.seed(seed)
    
    centers = np.random.uniform(0.3, 0.7, (n, 2))
    radii = np.full(n, 0.015)
    
    grow_step = 0.0003
    
    for _ in range(8000):
        # Repulsion forces
        forces = np.zeros((n, 2))
        dt = 0.01
        
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[j, 0] - centers[i, 0]
                dy = centers[j, 1] - centers[i, 1]
                dist = np.hypot(dx, dy)
                
                if dist > 1e-8:
                    overlap = radii[i] + radii[j] - dist
                    if overlap > 0:
                        force = overlap * 2.0
                        nx, ny = dx / dist, dy / dist
                        forces[i] -= [nx * force, ny * force]
                        forces[j] += [nx * force, ny * force]
        
        # Wall forces
        for i in range(n):
            r = radii[i]
            if centers[i, 0] < r + 0.01:
                forces[i, 0] += 0.3
            if centers[i, 0] > 1.0 - r - 0.01:
                forces[i, 0] -= 0.3
            if centers[i, 1] < r + 0.01:
                forces[i, 1] += 0.3
            if centers[i, 1] > 1.0 - r - 0.01:
                forces[i, 1] -= 0.3
        
        centers += forces * dt
        centers = np.clip(centers, radii[:, None] + 1e-6, 1.0 - radii[:, None] - 1e-6)
        
        # Try to grow
        can_grow = True
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.hypot(centers[i, 0] - centers[j, 0],
                               centers[i, 1] - centers[j, 1])
                if dist < radii[i] + radii[j] + 0.003:
                    can_grow = False
                    break
            if not can_grow:
                break
            # Check walls
            r = radii[i]
            if centers[i, 0] < r + 0.003 or centers[i, 0] > 1.0 - r - 0.003 or \
               centers[i, 1] < r + 0.003 or centers[i, 1] > 1.0 - r - 0.003:
                can_grow = False
                break
        
        if can_grow:
            radii += grow_step
            grow_step *= 1.0002
        else:
            grow_step *= 0.999
    
    return centers, radii

def resolve_overlaps(centers, radii, n, iterations=500):
    """Resolve any overlaps by pushing circles apart."""
    for _ in range(iterations):
        max_violation = 0
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[j, 0] - centers[i, 0]
                dy = centers[j, 1] - centers[i, 1]
                dist = np.hypot(dx, dy)
                min_dist = radii[i] + radii[j]
                
                if dist < min_dist and dist > 1e-10:
                    overlap = min_dist - dist
                    max_violation = max(max_violation, overlap)
                    shift = overlap * 0.5
                    nx, ny = dx / dist, dy / dist
                    centers[i, 0] -= shift * nx
                    centers[i, 1] -= shift * ny
                    centers[j, 0] += shift * nx
                    centers[j, 1] += shift * ny
        
        # Clip to valid range
        for i in range(n):
            r = radii[i]
            centers[i, 0] = np.clip(centers[i, 0], r + 1e-8, 1.0 - r - 1e-8)
            centers[i, 1] = np.clip(centers[i, 1], r + 1e-8, 1.0 - r - 1e-8)
        
        if max_violation < 1e-12:
            break
    
    return centers, radii

def is_valid(centers, radii, n):
    """Check if configuration is valid."""
    for i in range(n):
        r = radii[i]
        if r < 0 or centers[i, 0] < r - 1e-10 or centers[i, 0] > 1.0 - r + 1e-10 or \
           centers[i, 1] < r - 1e-10 or centers[i, 1] > 1.0 - r + 1e-10:
            return False
    
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0],
                           centers[i, 1] - centers[j, 1])
            if dist < radii[i] + radii[j] - 1e-10:
                return False
    
    return True
