import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def objective(vars):
    """Minimize negative sum of radii."""
    return -np.sum(vars[2::3])

def constraints(vars):
    """Compute all inequality constraints g(vars) >= 0."""
    n = N_CIRCLES
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    b = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dr = r[:, None] + r[None, :]
    
    i_idx, j_idx = np.tril_indices(n, -1)
    o = dx[i_idx, j_idx]**2 + dy[i_idx, j_idx]**2 - dr[i_idx, j_idx]**2
    
    return np.concatenate([b, o])

def force_directed_init(n, seed, r_base=0.06):
    """Generate dense initial configuration using repulsive forces."""
    np.random.seed(seed)
    c = np.random.uniform(r_base, 1.0 - r_base, (n, 2))
    
    for _ in range(1500):
        f = np.zeros((n, 2))
        for i in range(n):
            for j in range(i + 1, n):
                dx = c[j, 0] - c[i, 0]
                dy = c[j, 1] - c[i, 1]
                dist = np.hypot(dx, dy)
                if dist < 0.3 and dist > 1e-6:
                    k = (0.3 - dist) / dist
                    fx, fy = k * dx, k * dy
                    f[i] -= [fx, fy]
                    f[j] += [fx, fy]
        c += f * 0.02
        c = np.clip(c, r_base, 1.0 - r_base)
        
    # Estimate safe initial radii based on local density
    radii = np.zeros(n)
    for i in range(n):
        min_d = min(c[i, 0], 1.0 - c[i, 0], c[i, 1], 1.0 - c[i, 1])
        for j in range(n):
            if i != j:
                d = np.hypot(c[i, 0] - c[j, 0], c[i, 1] - c[j, 1])
                if d < min_d:
                    min_d = d
        radii[i] = min_d * 0.45  # 0.45 ensures 2*r < min_dist, strictly feasible
        
    return c, radii

def hex_grid_init(n, seed):
    """Generate perturbed hexagonal lattice initialization."""
    np.random.seed(seed)
    r0 = 0.092
    c = np.zeros((n, 2))
    radii = np.full(n, r0)
    idx = 0
    y = r0
    row = 0
    
    while idx < n:
        x = r0 + (row % 2) * r0
        while idx < n and x + r0 <= 1.0:
            c[idx] = [x, y]
            idx += 1
            x += 2 * r0
        y += r0 * np.sqrt(3)
        row += 1
        
    c += np.random.normal(0, 0.002, c.shape)
    return c, radii

def run_packing():
    n = N_CIRCLES
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * n
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -np.inf
    best_vars = None
    
    # Phase 1: Diverse initializations
    inits = []
    for s in range(15):
        inits.append(('force', s))
    for s in range(8):
        inits.append(('hex', s))
        
    for method, seed in inits:
        if method == 'force':
            c, r = force_directed_init(n, seed)
        else:
            c, r = hex_grid_init(n, seed)
            
        x0 = np.zeros(3 * n)
        x0[0::3] = c[:, 0]
        x0[1::3] = c[:, 1]
        x0[2::3] = r
        
        # Project to strict bounds
        for i in range(n):
            ri = max(1e-6, x0[3*i+2])
            x0[3*i] = np.clip(x0[3*i], ri, 1.0 - ri)
            x0[3*i+1] = np.clip(x0[3*i+1], ri, 1.0 - ri)
            x0[3*i+2] = ri
            
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 20000, 'ftol': 1e-14, 'disp': False})
            if not np.isnan(res.fun) and -res.fun > best_sum:
                best_sum = -res.fun
                best_vars = res.x.copy()
        except Exception:
            continue
            
    # Phase 2: Local refinement to escape shallow minima
    if best_vars is not None:
        for _ in range(10):
            x_pert = best_vars + np.random.normal(0, 1e-4, 3 * n)
            for i in range(n):
                ri = max(1e-6, x_pert[3*i+2])
                x_pert[3*i+2] = ri
                x_pert[3*i] = np.clip(x_pert[3*i], ri, 1.0 - ri)
                x_pert[3*i+1] = np.clip(x_pert[3*i+1], ri, 1.0 - ri)
                
            try:
                res = minimize(objective, x_pert, method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 15000, 'ftol': 1e-14, 'disp': False})
                if not np.isnan(res.fun) and -res.fun > best_sum:
                    best_sum = -res.fun
                    best_vars = res.x.copy()
            except Exception:
                break
                
    # Fallback (should not trigger)
    if best_vars is None:
        best_vars = np.zeros(3 * n)
        best_vars[0::3] = np.tile(np.linspace(0.1, 0.9, 5), 6)[:n]
        best_vars[1::3] = np.repeat(np.linspace(0.1, 0.9, 6), 5)[:n]
        best_vars[2::3] = 0.08
        
    centers = np.column_stack((best_vars[0::3], best_vars[1::3]))
    radii = best_vars[2::3]
    
    # Phase 3: Strict validation & numerical repair
    for _ in range(100):
        valid = True
        for i in range(n):
            if radii[i] < 0: valid = False; break
            if centers[i,0] - radii[i] < -1e-9 or centers[i,0] + radii[i] > 1.0 + 1e-9: valid = False; break
            if centers[i,1] - radii[i] < -1e-9 or centers[i,1] + radii[i] > 1.0 + 1e-9: valid = False; break
        if valid:
            for i in range(n):
                for j in range(i + 1, n):
                    d = np.hypot(centers[i,0] - centers[j,0], centers[i,1] - centers[j,1])
                    if d < radii[i] + radii[j] - 1e-9:
                        valid = False; break
                if not valid: break
        if valid: break
        radii *= 0.995
        
    return centers, radii, float(np.sum(radii))
