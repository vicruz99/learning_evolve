import numpy as np
from scipy.optimize import minimize

def objective(vars, n):
    """Minimize negative sum of radii."""
    return -np.sum(vars[2*n:])

def compute_constraints(vars, n, i_idx, j_idx):
    """Compute all boundary and squared non-overlap constraints as a single vector >= 0."""
    x = vars[:n]
    y = vars[n:2*n]
    r = vars[2*n:]
    
    # Boundary constraints: g(vars) >= 0
    c = np.concatenate([
        x - r,
        1.0 - x - r,
        y - r,
        1.0 - y - r
    ])
    
    # Overlap constraints: dx^2 + dy^2 - (r_i + r_j)^2 >= 0
    xi, yi, ri = x[i_idx], y[i_idx], r[i_idx]
    xj, yj, rj = x[j_idx], y[j_idx], r[j_idx]
    
    dx = xi - xj
    dy = yi - yj
    dr = ri + rj
    
    overlap = dx**2 + dy**2 - dr**2
    return np.concatenate([c, overlap])

def make_init(n, angle, r_start, seed):
    """Generate a rotated hexagonal lattice initialization with perturbation."""
    np.random.seed(seed)
    cx, cy = 0.5, 0.5
    cos_a, sin_a = np.cos(angle), np.sin(angle)
    
    centers = []
    y_pos = r_start
    row = 0
    while len(centers) < n and y_pos + r_start <= 1.0:
        x_pos = r_start if row % 2 == 0 else 2.0 * r_start
        while len(centers) < n and x_pos + r_start <= 1.0:
            # Rotate around square center
            dx = x_pos - cx
            dy = y_pos - cy
            rx = dx * cos_a - dy * sin_a + cx
            ry = dx * sin_a + dy * cos_a + cy
            centers.append([rx, ry])
            x_pos += 2.0 * r_start
        y_pos += np.sqrt(3.0) * r_start
        row += 1
        
    # Fill remaining circles randomly if lattice runs out
    while len(centers) < n:
        centers.append([np.random.uniform(r_start, 1.0 - r_start),
                        np.random.uniform(r_start, 1.0 - r_start)])
                        
    centers = np.array(centers[:n])
    # Add controlled noise to break exact symmetry
    centers += np.random.normal(0, 0.012, centers.shape)
    # Project strictly inside bounds to give optimizer a feasible start
    centers = np.clip(centers, r_start + 1e-5, 1.0 - r_start - 1e-5)
    
    v = np.zeros(3 * n)
    v[:n] = centers[:, 0]
    v[n:2*n] = centers[:, 1]
    v[2*n:] = r_start
    return v

    best_sum = -1.0
    best_vars = None
    
    # Phase 1: Multi-start optimization with diverse lattice angles
    for seed in range(30):
        for angle in [0.0, np.pi/12, np.pi/6, np.pi/4]:
            v0 = make_init(n, angle, 0.085 + seed * 0.001, seed)
            try:
                res = minimize(objective, v0, args=(n,), method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
                s = -res.fun
                # Accept only if strictly feasible within tolerance and improves best
                c_vals = compute_constraints(res.x, n, i_idx, j_idx)
                if np.min(c_vals) >= -1e-7 and s > best_sum:
                    best_sum = s
                    best_vars = res.x.copy()
            except Exception:
                pass
                
    # Phase 2: Local refinement around best solution to escape shallow minima
    if best_vars is not None:
        for _ in range(10):
            v_pert = best_vars + np.random.normal(0, 1e-4, 3*n)

    if best_vars is None:
