import numpy as np
from scipy.optimize import minimize

N = 26

def objective(vars):
    """Objective: minimize negative sum of radii."""
    return -np.sum(vars[2::3])

def constraints(vars):
    """Compute all boundary and non-overlap constraints as a single vector >= 0."""
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    # Boundary constraints: circle inside [0,1]x[0,1]
    c = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    
    # Overlap constraints: (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dr = r[:, None] + r[None, :]
    
    iu, ju = np.triu_indices(N, k=1)
    c = np.concatenate([c, dx[iu, ju]**2 + dy[iu, ju]**2 - dr[iu, ju]**2])
    return c

def generate_hex_init(r_start, seed):
    """Generate a perturbed hexagonal lattice initialization."""
    np.random.seed(seed)
    centers = np.zeros((N, 2))
    radii = np.full(N, r_start)
    idx = 0
    y = r_start
    row = 0
    while idx < N and y + r_start <= 1.0:
        x_start = r_start if row % 2 == 0 else 2 * r_start
        x = x_start
        while idx < N and x + r_start <= 1.0:
            centers[idx] = [x, y]
            idx += 1
            x += 2 * r_start
        y += np.sqrt(3) * r_start
        row += 1
    while idx < N:
        centers[idx] = np.random.uniform(r_start, 1.0 - r_start, 2)
        idx += 1
    centers += np.random.normal(0, 0.005, centers.shape)
    centers = np.clip(centers, r_start, 1.0 - r_start)
    return np.concatenate([centers.flatten(), radii])

def generate_random_init(seed):
    """Generate a random valid initialization."""
    np.random.seed(seed)
    centers = np.random.uniform(0.15, 0.85, (N, 2))
    radii = np.full(N, 0.06)
    centers += np.random.normal(0, 0.02, centers.shape)
    centers = np.clip(centers, 0.05, 0.95)
    return np.concatenate([centers.flatten(), radii])

def is_valid(centers, radii):
    """Check strict feasibility of a packing configuration."""
    n = centers.shape[0]
    if np.any(centers[:, 0] - radii < -1e-12) or np.any(1.0 - centers[:, 0] - radii < -1e-12):
        return False
    if np.any(centers[:, 1] - radii < -1e-12) or np.any(1.0 - centers[:, 1] - radii < -1e-12):
        return False
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            if d < radii[i] + radii[j] - 1e-12:
                return False
    return True

def repair_and_optimize(x0):
    """Ensure feasibility and run SLSQP optimization."""
    centers = x0[:2*N].reshape(N, 2).copy()
    radii = x0[2*N:].copy()
    
    # Clip to bounds
    for i in range(N):
        r = max(0.01, radii[i])
        radii[i] = r
        centers[i, 0] = np.clip(centers[i, 0], r, 1.0 - r)
        centers[i, 1] = np.clip(centers[i, 1], r, 1.0 - r)
        
    # Iterative repulsion to resolve overlaps
    for _ in range(100):
        max_ov = 0
        for i in range(N):
            for j in range(i+1, N):
                dx = centers[j, 0] - centers[i, 0]
                dy = centers[j, 1] - centers[i, 1]
                d = np.hypot(dx, dy)
                req = radii[i] + radii[j]
                if d < req and d > 1e-8:
                    shift = (req - d) * 0.5
                    nx, ny = dx/d, dy/d
                    centers[i, 0] -= shift * nx
                    centers[i, 1] -= shift * ny
                    centers[j, 0] += shift * nx
                    centers[j, 1] += shift * ny
                    max_ov = max(max_ov, req - d)
        if max_ov < 1e-6:
            break
            
    # Re-clip after repulsion
    for i in range(N):
        r = radii[i]
        centers[i, 0] = np.clip(centers[i, 0], r, 1.0 - r)
        centers[i, 1] = np.clip(centers[i, 1], r, 1.0 - r)
        
    x0_fixed = np.concatenate([centers.flatten(), radii])
    
    cons = {'type': 'ineq', 'fun': constraints}
    bounds = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.25)] * N
    
    try:
        res = minimize(objective, x0_fixed, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': 15000, 'ftol': 1e-14, 'disp': False})
        if not np.isnan(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return x0_fixed, -np.sum(x0_fixed[2::3])

def run_packing():
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    best_x = None
    best_val = -np.inf
    
    # Phase 1: Hexagonal lattice starts with varying densities
    for r_start in [0.075, 0.08, 0.085, 0.09, 0.095]:

            x_opt, val = repair_and_optimize(x0)
            if val > best_val:
                best_val = val
                best_x = x_opt

    for seed in range(10):
        x0 = generate_random_init(seed)
        x_opt, val = repair_and_optimize(x0)

            best_val = val
            best_x = x_opt

            x_opt, val = repair_and_optimize(noisy)

                best_val = val
                best_x = x_opt

    radii = best_x[2*N:]
    
    # Phase 4: Strict validity check and numerical repair

        factor = 0.999
        for _ in range(100):
            radii *= factor
            if is_valid(centers, radii):
                break
                
    return centers, radii, float(np.sum(radii))
