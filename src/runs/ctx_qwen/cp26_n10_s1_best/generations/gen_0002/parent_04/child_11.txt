import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Compute all boundary and non-overlap constraints as a single vector >= 0."""
    n = N_CIRCLES
    cx, cy, cr = x[0::3], x[1::3], x[2::3]
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    b = np.concatenate([cx - cr, 1.0 - cx - cr, cy - cr, 1.0 - cy - cr])
    
    # Overlap constraints: dist(i,j) >= r_i + r_j
    # Vectorized for performance
    i, j = np.tril_indices(n, -1)
    dx = cx[i] - cx[j]
    dy = cy[i] - cy[j]
    dr = cr[i] + cr[j]
    dist = np.sqrt(dx**2 + dy**2)
    ovl = dist - dr
    
    return np.concatenate([b, ovl])

    def try_optimize(v0):
        nonlocal best_sum, best_vars

                           constraints=cons,

            s = -res.fun
            if s > best_sum:
                best_sum = s
                best_vars = res.x.copy()
        except Exception:

    # Phase 1: Structured Hexagonal Initializations

        [5, 5, 6, 6, 4],
        [4, 6, 6, 6, 4],
        [6, 4, 6, 6, 4],
        [5, 6, 6, 5, 4]
    ]

                    cy[idx] = y
                    cr[idx] = r_init
                    idx += 1
