

<strategy>
To maximize the sum of radii for 26 circles in a unit square, we employ a two-stage optimization strategy combined with multi-start local search:

1. **Stage 1 (Equal-Radius Packing)**: First, we optimize for a single shared radius $r$ across all circles. This simplifies the problem significantly (2N+1 variables instead of 3N) and reliably finds a densely packed configuration where all circles are roughly the same size. This serves as an excellent baseline, as optimal variable-radius packings for moderate N typically have nearly uniform radii.

2. **Stage 2 (Variable-Radius Expansion)**: We take the best configurations from Stage 1, assign each circle a slightly perturbed radius (±5%), and optimize the full 3N-variable problem. This allows the optimizer to exploit minor gaps by adjusting individual radii and positions, breaking symmetry and escaping the equal-radius local optimum.

3. **Stage 3 (Local Refinement)**: We perform iterative perturbations and re-optimizations around the best found solution to escape shallow local minima and polish the packing to machine precision.

**Key Technical Improvements**:
- **Smooth Constraints**: We use squared distance constraints `(dx^2 + dy^2) - (ri + rj)^2 >= 0` which are differentiable everywhere, improving SLSQP convergence compared to raw distance constraints.
- **Vectorized Evaluations**: Constraint and objective functions use NumPy broadcasting to compute all pairwise and boundary constraints simultaneously, drastically reducing overhead per iteration.
- **Robust Initialization**: Hexagonal lattice starts are perturbed and clipped to guarantee feasibility before optimization begins, preventing the solver from wasting iterations resolving severe overlaps.
- **Numerical Repair**: A final validation and controlled shrinkage loop ensures strict compliance with the provided `validate_packing` function, accounting for floating-point drift.

import numpy as np
from scipy.optimize import minimize

N = 26

def obj_equal(vars, n):
    """Objective for Stage 1: maximize shared radius."""
    return -vars[2*n]

def cons_equal(vars, n):
    """Constraints for Stage 1: boundary and non-overlap with shared radius."""
    centers = vars[:2*n].reshape(n, 2)
    r = vars[2*n]
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    b = np.concatenate([
        centers[:, 0] - r,
        1.0 - centers[:, 0] - r,
        centers[:, 1] - r,
        1.0 - centers[:, 1] - r
    ])
    
    # Overlap constraints: dist^2 >= (2r)^2
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist_sq = np.sum(diffs**2, axis=2)
    tri = np.tril_indices(n, -1)
    o = dist_sq[tri] - (2.0 * r)**2
    
    return np.concatenate([b, o])

def obj_var(vars, n):
    """Objective for Stage 2/3: maximize sum of individual radii."""
    return -np.sum(vars[2*n:])

def cons_var(vars, n):
    """Constraints for Stage 2/3: boundary and non-overlap with variable radii."""
    centers = vars[:2*n].reshape(n, 2)
    radii = vars[2*n:]
    
    # Boundary constraints
    b = np.concatenate([
        centers[:, 0] - radii,
        1.0 - centers[:, 0] - radii,
        centers[:, 1] - radii,
        1.0 - centers[:, 1] - radii
    ])
    
    # Overlap constraints: dist^2 >= (ri + rj)^2
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist_sq = np.sum(diffs**2, axis=2)
    r_sum_sq = (radii[:, np.newaxis] + radii[np.newaxis, :])**2
    tri = np.tril_indices(n, -1)
    o = dist_sq[tri] - r_sum_sq[tri]
    
    return np.concatenate([b, o])

def get_bounds_equal(n):
    """Bounds for Stage 1."""
    return [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)]

def get_bounds_var(n):
    """Bounds for Stage 2/3."""
    b = []
    for _ in range(n):
        b.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
    return b

def make_hex_init(n, seed):
    """Generate hexagonal lattice initialization for equal-radius stage."""
    np.random.seed(seed)
    r = 0.09
    centers = []
    y = r
    row = 0
    while len(centers) < n and y + r <= 1.0:
        x = r if row % 2 == 0 else 2.0 * r
        while len(centers) < n and x + r <= 1.0:
            centers.append([x, y])
            x += 2.0 * r
        y += np.sqrt(3.0) * r
        row += 1
        
    while len(centers) < n:
        centers.append([np.random.uniform(r, 1.0-r), np.random.uniform(r, 1.0-r)])
        
    centers = np.array(centers[:n])
    centers += np.random.normal(0, 0.005, centers.shape)
    centers = np.clip(centers, r, 1.0-r)
    
    vars = np.zeros(2*n + 1)
    vars[:2*n] = centers.flatten()
    vars[2*n] = r
    return vars

def run_packing():
    np.random.seed(42)  # Reproducibility
    n = N
    bounds_eq = get_bounds_equal(n)
    cons_eq = {'type': 'ineq', 'fun': cons_equal, 'args': (n,)}
    bounds_var = get_bounds_var(n)
    cons_var = {'type': 'ineq', 'fun': cons_var, 'args': (n,)}
    
    best_sum = -1.0
    best_vars = None
    
    # Stage 1: Equal radius packing to find dense configurations
    eq_results = []
    for s in range(30):
        v0 = make_hex_init(n, s)
        try:
            res = minimize(obj_equal, v0, args=(n,), method='SLSQP', bounds=bounds_eq,
                           constraints=cons_eq, options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
            if not np.isnan(res.fun):
                eq_results.append(res.x)
        except Exception:
            pass
            
    # Stage 2: Variable radius optimization from best equal-radius results
    for v_eq in eq_results:
        centers_eq = v_eq[:2*n].reshape(n, 2)
        r_eq = v_eq[2*n]
        
        # Perturb radii to break symmetry and allow exploration
        r_pert = r_eq * (1.0 + np.random.uniform(-0.05, 0.05, n))
        v0 = np.zeros(3*n)
        v0[:2*n] = centers_eq.flatten()
        v0[2*n:] = r_pert
        
        try:
            res = minimize(obj_var, v0, args=(n,), method='SLSQP', bounds=bounds_var,
                           constraints=cons_var, options={'maxiter': 10000, 'ftol': 1e-13, 'disp': False})
            s = -res.fun
            if s > best_sum:
                best_sum = s
                best_vars = res.x.copy()
        except Exception:
            pass
            
    # Stage 3: Local refinement to escape shallow local minima
    if best_vars is not None:
        for _ in range(15):
            v0 = best_vars + np.random.normal(0, 1e-4, 3*n)
            # Project perturbed variables to feasible bounds
            for k in range(n):
                r = np.clip(v0[2*n+k], 0.01, 0.5)
                v0[2*n+k] = r
                v0[k] = np.clip(v0[k], r, 1.0-r)
                v0[n+k] = np.clip(v0[n+k], r, 1.0-r)
                
            try:
                res = minimize(obj_var, v0, args=(n,), method='SLSQP', bounds=bounds_var,
                               constraints=cons_var, options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
                s = -res.fun
                if s > best_sum:
                    best_sum = s
                    best_vars = res.x.copy()
            except Exception:
                pass
                
    # Fallback if optimization completely fails
    if best_vars is None:
        v0 = make_hex_init(n, 0)
        best_vars = np.concatenate([v0[:2*n], np.full(n, v0[2*n])])
        
    centers = best_vars[:2*n].reshape(n, 2)
    radii = best_vars[2*n:]
    
    # Phase 4: Strict validity check and numerical repair
    valid = True
    for i in range(n):
        if radii[i] < 0 or centers[i,0] < radii[i] - 1e-12 or centers[i,0] + radii[i] > 1 + 1e-12 or \
           centers[i,1] < radii[i] - 1e-12 or centers[i,1] + radii[i] > 1 + 1e-12:
            valid = False
            break
    if valid:
        for i in range(n):
            for j in range(i+1, n):
                d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
                if d < radii[i] + radii[j] - 1e-12:
                    valid = False
                    break
            if not valid:
                break
                
    if not valid:
        # Aggressively shrink radii until strictly valid
        for _ in range(100):
            radii *= 0.995
            valid = True
            for i in range(n):
                if centers[i,0] < radii[i] - 1e-12 or centers[i,0] + radii[i] > 1 + 1e-12 or \
                   centers[i,1] < radii[i] - 1e-12 or centers[i,1] + radii[i] > 1 + 1e-12:
                    valid = False
                    break
            if valid:
                for i in range(n):
                    for j in range(i+1, n):
                        d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
                        if d < radii[i] + radii[j] - 1e-12:
                            valid = False
                            break
                    if not valid:
                        break
            if valid:
                break
                
    return centers, radii, float(np.sum(radii))
