import numpy as np
from scipy.optimize import minimize

N = 26
# Precompute upper triangle indices for pairwise constraints to speed up evaluation
TRI_U = np.triu_indices(N, 1)

def objective(vars):
    """Minimize negative sum of radii."""
    return -np.sum(vars[2::3])

def constraints(vars, n):
    """Compute all boundary and non-overlap constraints as a single vector >= 0."""
    xs = vars[0::3]
    ys = vars[1::3]
    rs = vars[2::3]
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    b = np.concatenate([xs - rs, 1.0 - xs - rs, ys - rs, 1.0 - ys - rs])
    
    # Overlap constraints using squared distances for numerical stability
    # dist^2 - (r_i + r_j)^2 >= 0
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    o = dx[TRI_U]**2 + dy[TRI_U]**2 - dr[TRI_U]**2
    
    return np.concatenate([b, o])

def make_init(n, seed):
    """Generate a perturbed hexagonal lattice initialization."""
    np.random.seed(seed)
    xs = np.zeros(n)
    ys = np.zeros(n)
    rs = np.full(n, 0.085)
    
    idx = 0
    y = rs[0]
    row = 0
    # Fill hexagonal pattern
    while idx < n and y + rs[0] <= 1.0:
        x = rs[0] if row % 2 == 0 else 2.0 * rs[0]
        while idx < n and x + rs[0] <= 1.0:
            xs[idx] = x + np.random.uniform(-0.004, 0.004)
            ys[idx] = y + np.random.uniform(-0.004, 0.004)
            idx += 1
            x += 2.0 * rs[0]
        y += np.sqrt(3.0) * rs[0]
        row += 1
        
    # Fallback for any remaining circles
    while idx < n:
        r = 0.06
        xs[idx] = np.random.uniform(r, 1.0 - r)
        ys[idx] = np.random.uniform(r, 1.0 - r)
        rs[idx] = r
        idx += 1
        
    v = np.empty(3 * n)
    v[0::3] = xs
    v[1::3] = ys
    v[2::3] = rs
    return v

def run_packing():
    n = N
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * n
    cons = {'type': 'ineq', 'fun': constraints, 'args': (n,)}
    
    best_sum = -1.0
    best_vars = None
    
    # Phase 1: Multi-start optimization from diverse hexagonal seeds
    for seed in range(30):
        v0 = make_init(n, seed)
        try:
            res = minimize(objective, v0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 6000, 'ftol': 1e-13, 'disp': False})
            s = -res.fun
            c_vals = constraints(res.x, n)
            # Accept if feasible within tolerance and improves best sum

                best_sum = s
                best_vars = res.x.copy()
        except Exception:
            pass
            
    # Phase 2: Local refinement around the best solution
    if best_vars is not None:

            v0 = best_vars + np.random.normal(0, 1e-4, 3 * n)

                res = minimize(objective, v0, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 4000, 'ftol': 1e-13, 'disp': False})
                s = -res.fun
                c_vals = constraints(res.x, n)
                if np.min(c_vals) >= -1e-8 and s > best_sum:
                    best_sum = s
                    best_vars = res.x.copy()
            except Exception:
                pass
                
    # Fallback in case optimization fails
    if best_vars is None:
        best_vars = make_init(n, 0)
        
    centers = np.column_stack((best_vars[0::3], best_vars[1::3]))

                        break
