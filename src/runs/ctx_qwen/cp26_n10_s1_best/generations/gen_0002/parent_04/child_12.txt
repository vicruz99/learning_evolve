from scipy.optimize import minimize

N_CIRCLES = 26

def objective(vars):
    """Minimize negative sum of radii."""
    return -np.sum(vars[2::3])

def constraint_func(vars):
    """Compute all boundary and non-overlap constraints as a single vector g(vars) >= 0."""
    n = N_CIRCLES
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    parts = []
    # Boundary constraints: circle inside [0, 1] x [0, 1]
    parts.append(x - r)              # x >= r
    parts.append(1.0 - x - r)        # 1-x >= r
    parts.append(y - r)              # y >= r
    parts.append(1.0 - y - r)        # 1-y >= r
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    # Using squared form avoids sqrt derivative singularities
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dr = r[:, None] + r[None, :]
    
    tri = np.tril_indices(n, -1)
    parts.append(dx[tri]**2 + dy[tri]**2 - dr[tri]**2)
    
    return np.concatenate(parts)

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-5, 0.5]."""
    b = []
    for _ in range(N_CIRCLES):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-5, 0.5)])
    return bounds

def force_directed_init(n, seed):
    """Generate initial configuration using repulsive force simulation."""
    np.random.seed(seed)
    centers = np.random.uniform(0.12, 0.88, (n, 2))
    radii = np.full(n, 0.08)
    
    for _ in range(600):
        forces = np.zeros_like(centers)
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[j, 0] - centers[i, 0]
                dy = centers[j, 1] - centers[i, 1]
                dist = np.hypot(dx, dy)
                if dist < 0.35 and dist > 1e-6:
                    f = 0.012 / (dist**2)
                    fx, fy = f * dx, f * dy
                    forces[i] -= [fx, fy]
                    forces[j] += [fx, fy]
        # Wall repulsion
        for i in range(n):
            if centers[i, 0] < 0.06: forces[i, 0] += 0.025
            elif centers[i, 0] > 0.94: forces[i, 0] -= 0.025
            if centers[i, 1] < 0.06: forces[i, 1] += 0.025
            elif centers[i, 1] > 0.94: forces[i, 1] -= 0.025
            
        centers += forces * 0.04
        centers = np.clip(centers, 0.03, 0.97)
        
    radii = np.full(n, 0.05)
    return centers, radii

def hex_init(n, seed):
    """Generate a dense hexagonal lattice initialization."""
    np.random.seed(seed)
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.09)
    idx = 0
    y = 0.09
    row = 0
    while idx < n and y + 0.09 <= 1.0:
        x = 0.09 if row % 2 == 0 else 0.18
        while idx < n and x + 0.09 <= 1.0:
            centers[idx] = [x, y]
            idx += 1
            x += 0.18
        y += 0.09 * np.sqrt(3.0)
        row += 1
        
    while idx < n:
        centers[idx] = np.random.uniform(0.15, 0.85, 2)
        idx += 1
        
    # Perturb to break exact symmetry
    centers += np.random.normal(0, 0.004, centers.shape)
    centers = np.clip(centers, 0.09, 0.91)
    return centers, radii

def grid_init(n, seed):
    """Generate a perturbed grid initialization."""
    np.random.seed(seed)
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.085)
    idx = 0

        idx += 1
