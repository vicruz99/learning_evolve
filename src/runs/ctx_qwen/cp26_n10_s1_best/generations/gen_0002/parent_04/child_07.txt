def obj_func(vars):

    return -np.sum(vars[2::3])

def compute_constraints(vars):
    """Compute all boundary and non-overlap constraints as a single vector >= 0."""
    n = len(vars) // 3
    xs = vars[0::3]
    ys = vars[1::3]
    rs = vars[2::3]
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    c = np.concatenate([
        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs
    ])
    
    # Pairwise non-overlap constraints (squared form for smooth gradients)
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]

    c_overlap = dx[i, j]**2 + dy[i, j]**2 - dr[i, j]**2
    
    return np.concatenate([c, c_overlap])
