import numpy as np
from scipy.optimize import minimize

N = 26

def obj_func(v):
    """Objective to minimize: negative sum of radii."""
    return -np.sum(v[2::3])

def constraints_func(v):
    """Compute all inequality constraints g(v) >= 0."""
    n = N
    xs = v[0::3]
    ys = v[1::3]
    rs = v[2::3]
    
    # Boundary constraints: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    b = np.concatenate([xs - rs, 1.0 - xs - rs, ys - rs, 1.0 - ys - rs])
    
    # Pairwise non-overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    idx = np.triu_indices(n, k=1)
    o = dx[idx]**2 + dy[idx]**2 - dr[idx]**2
    
    return np.concatenate([b, o])

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-6, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)])
    return b

def hex_init(seed, r_start=0.09):
    """Generate hexagonal lattice initialization."""
    np.random.seed(seed)
    centers = np.zeros((N, 2))
    radii = np.full(N, r_start)
    
    idx = 0
    y = r_start
    row = 0
    while idx < N and y + r_start <= 1.0:
        x = r_start if row % 2 == 0 else 2.0 * r_start
        while idx < N and x + r_start <= 1.0:
            centers[idx] = [x, y]
            idx += 1
            x += 2.0 * r_start
        y += np.sqrt(3.0) * r_start
        row += 1
        
    if idx < N:
        for i in range(idx, N):
            centers[i] = np.random.uniform(0.2, 0.8, 2)
            
    centers += np.random.normal(0, 0.015, centers.shape)
    radii += np.random.normal(0, 0.01, radii.shape)
    radii = np.clip(radii, 0.02, 0.4)
    
    centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
    centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)
    
    v = np.zeros(3 * N)
    v[0::3] = centers[:, 0]
    v[1::3] = centers[:, 1]
    v[2::3] = radii
    return v

def force_init(seed):
    """Generate initialization via repulsive force simulation."""
    np.random.seed(seed)
    centers = np.random.uniform(0.1, 0.9, (N, 2))
    radii = np.full(N, 0.05)
    
    for _ in range(600):
        forces = np.zeros((N, 2))
        for i in range(N):
            for j in range(i + 1, N):
                dx = centers[j, 0] - centers[i, 0]
                dy = centers[j, 1] - centers[i, 1]
                dist = np.hypot(dx, dy)
                if dist < 0.3 and dist > 1e-6:
                    f = 0.015 / (dist**2)
                    fx, fy = f * dx, f * dy
                    forces[i] -= [fx, fy]
                    forces[j] += [fx, fy]
                    
            for d in range(2):
                if centers[i, d] < radii[i] + 0.05:
                    forces[i, d] += 0.03
                elif centers[i, d] > 1.0 - radii[i] - 0.05:
                    forces[i, d] -= 0.03
                    
        centers += forces * 0.05
        centers = np.clip(centers, 0.01, 0.99)
        
    for i in range(N):
        min_d = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        for j in range(N):
            if i != j:
                d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                if d < min_d:
                    min_d = d
        radii[i] = min_d / 2.0 * 0.85
        
    v = np.zeros(3 * N)
    v[0::3] = centers[:, 0]
    v[1::3] = centers[:, 1]
    v[2::3] = radii
    return v

def run_packing():
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints_func}
    
    best_v = None
    best_val = -np.inf
    
    # Phase 1: Diverse multi-start optimization
    for seed in range(40):
        if seed % 4 == 0:
            v0 = hex_init(seed, r_start=0.09)
        elif seed % 4 == 1:
            v0 = hex_init(seed, r_start=0.10)
        else:
            v0 = force_init(seed)
            
        try:
            res = minimize(obj_func, v0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 5000, 'ftol': 1e-12, 'disp': False})
            if not np.isnan(res.fun) and -res.fun > best_val:
                best_val = -res.fun
                best_v = res.x.copy()
        except Exception:
            pass
            
    # Phase 2: Refinement and perturbation
    if best_v is not None:
        for _ in range(15):
            v_pert = best_v + np.random.normal(0, 1e-4, 3 * N)
            try:
                res = minimize(obj_func, v_pert, method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 5000, 'ftol': 1e-12, 'disp': False})
                if not np.isnan(res.fun) and -res.fun > best_val:
                    best_val = -res.fun
                    best_v = res.x.copy()
            except Exception:
                pass
                
    # Fallback
    if best_v is None:
        best_v = hex_init(0)
        
    centers = np.column_stack((best_v[0::3], best_v[1::3]))
    radii = best_v[2::3].copy()
    
    # Phase 3: Strict validity check and numerical repair
    for _ in range(100):
        valid = True
        for i in range(N):
            if radii[i] < 0 or centers[i, 0] < radii[i] - 1e-12 or centers[i, 0] > 1.0 - radii[i] + 1e-12 or \
               centers[i, 1] < radii[i] - 1e-12 or centers[i, 1] > 1.0 - radii[i] + 1e-12:
                valid = False
                break
        if valid:
            for i in range(N):
                for j in range(i + 1, N):
                    if np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1]) < radii[i] + radii[j] - 1e-12:
                        valid = False
                        break
                if not valid:
                    break
        if valid:
            break
        radii *= 0.995
        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)
        
    return centers, radii, float(np.sum(radii))
