import numpy as np
from scipy.optimize import minimize

def objective(vars, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(vars[2::3])

def compute_constraints(vars, n):
    """Compute all boundary and non-overlap constraints as a single vector >= 0."""
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    parts = []
    # Boundary constraints: g(vars) >= 0
    parts.append(x - r)
    parts.append(1.0 - x - r)
    parts.append(y - r)
    parts.append(1.0 - y - r)
    
    # Pairwise non-overlap: dist^2 - (r_i + r_j)^2 >= 0
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dr = r[:, None] + r[None, :]
    
    # Upper triangle indices to avoid duplicates and self-comparison
    tri = np.triu_indices(n, k=1)
    parts.append(dx[tri]**2 + dy[tri]**2 - dr[tri]**2)
    
    return np.concatenate(parts)

def force_directed_init(n, seed):
    """Generate initial configuration using repulsive force simulation."""
    np.random.seed(seed)
    centers = np.random.uniform(0.25, 0.75, (n, 2))
    radii = np.full(n, 0.05)
    
    # Repulsive forces to spread out circles naturally
    for _ in range(300):
        forces = np.zeros_like(centers)
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[j, 0] - centers[i, 0]
                dy = centers[j, 1] - centers[i, 1]
                d2 = dx*dx + dy*dy
                d = np.sqrt(d2) if d2 > 1e-12 else 1e-6
                f = 0.003 / (d2 + 0.01)
                fx, fy = f * dx, f * dy
                forces[i] -= [fx, fy]
                forces[j] += [fx, fy]
                
            # Boundary repulsion
            for dim in range(2):
                if centers[i, dim] < radii[i]:
                    forces[i, dim] += 0.01
                elif centers[i, dim] > 1.0 - radii[i]:
                    forces[i, dim] -= 0.01
                    
        centers += forces * 0.1
        centers = np.clip(centers, 0.01, 0.99)
        
    # Estimate safe initial radii based on nearest neighbors/walls
    for i in range(n):
        min_d = min(centers[i, 0], 1.0 - centers[i, 0], 
                    centers[i, 1], 1.0 - centers[i, 1])
        for j in range(n):
            if i != j:
                d = np.hypot(centers[i, 0] - centers[j, 0], 
                             centers[i, 1] - centers[j, 1])
                if d < min_d:
                    min_d = d
        radii[i] = min_d * 0.45
        
    return centers, radii

def hex_pattern_init(n, seed, row_counts):
    """Generate a structured hexagonal lattice initialization."""
    np.random.seed(seed)
    r_start = 0.09
    centers = []
    y = r_start
    for i, cnt in enumerate(row_counts):
        x_start = r_start + (r_start if i % 2 == 1 else 0.0)
        for _ in range(cnt):
            if len(centers) >= n:
                break
            centers.append([x_start, y])
            x_start += 2 * r_start
        y += np.sqrt(3) * r_start
        
    centers = np.array(centers[:n])
    # Add perturbation to break exact symmetry and help optimizer
    centers += np.random.uniform(-0.012, 0.012, centers.shape)
    centers = np.clip(centers, r_start, 1.0 - r_start)
    radii = np.full(n, r_start)
    return centers, radii
