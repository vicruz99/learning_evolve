

<strategy>
To maximize the sum of radii for 26 circles in a unit square, we need to explore the highly non-convex configuration space effectively. The optimal packing typically resembles a perturbed hexagonal lattice adapted to the square boundaries, with circles touching their neighbors and walls to form a rigid structure.

Key improvements in this approach:
1. **Diverse High-Quality Initializations**: Instead of pure random starts, we use hexagonal grids with systematic shifts (`shift_x`, `shift_y`) and varying base radii (`r0`). This covers the basin of attraction for the global optimum.
2. **Force-Directed Pre-Relaxation**: Before optimization, a fast force-based resolver pushes overlapping circles apart and against walls. This guarantees feasible or near-feasible starting points, significantly improving SLSQP convergence speed and reliability.
3. **Multi-Phase Optimization**: 
   - Phase 1: Exhaustive grid search over hex parameters.
   - Phase 2: Random starts resolved by forces.
   - Phase 3: Iterative perturbation and subset-jumping on the best found solution to escape shallow local minima.
4. **Vectorized Constraints**: The constraint function computes all boundary and pairwise non-overlap conditions simultaneously using NumPy broadcasting and `tril_indices`, reducing overhead per iteration.
5. **Strict Validation & Repair**: A post-processing loop checks for numerical drift. If any constraint is violated due to floating-point inaccuracies, radii are minimally scaled down until strict compliance is met, guaranteeing a valid output for the checker.

import numpy as np
from scipy.optimize import minimize

N = 26

def objective(vars):
    """Minimize negative sum of radii."""
    return -np.sum(vars[2::3])

def constraints(vars):
    """Compute all boundary and non-overlap constraints as a single vector >= 0."""
    c = vars.reshape(N, 3)
    x, y, r = c[:, 0], c[:, 1], c[:, 2]
    
    # Boundary constraints
    parts = [
        x - r,
        1.0 - x - r,
        y - r,
        1.0 - y - r
    ]
    
    # Pairwise non-overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dr = r[:, None] + r[None, :]
    
    i, j = np.tril_indices(N, -1)
    parts.append(dx[i, j]**2 + dy[i, j]**2 - dr[i, j]**2)
    
    return np.concatenate(parts)

def make_hex_init(r0, shift_x=0.0, shift_y=0.0, noise=0.0):
    """Generate a hexagonal lattice initialization with optional shifts and noise."""
    centers = []
    y = r0 + shift_y
    row = 0
    while len(centers) < N:
        x = r0 + shift_x + (r0 if row % 2 == 1 else 0.0)
        while len(centers) < N and x + r0 <= 1.0:
            centers.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
    centers = np.array(centers[:N])
    if noise > 0:
        centers += np.random.normal(0, noise, centers.shape)
    radii = np.full(N, r0)
    return centers, radii

def resolve_overlaps(centers, radii, iterations=300):
    """Fast force-directed relaxation to resolve initial overlaps and boundary violations."""
    n = len(centers)
    for _ in range(iterations):
        forces = np.zeros((n, 2))
        max_viol = 0.0
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[j, 0] - centers[i, 0]
                dy = centers[j, 1] - centers[i, 1]
                d = np.hypot(dx, dy)
                min_d = radii[i] + radii[j]
                if d < min_d and d > 1e-9:
                    shift = (min_d - d) * 0.5
                    fx, fy = shift * dx / d, shift * dy / d
                    forces[i] -= [fx, fy]
                    forces[j] += [fx, fy]
                    if min_d - d > max_viol:
                        max_viol = min_d - d
            for wall in range(2):
                if centers[i, wall] < radii[i]:
                    forces[i, wall] += (radii[i] - centers[i, wall]) * 2.0
                elif centers[i, wall] > 1.0 - radii[i]:
                    forces[i, wall] -= (centers[i, wall] - (1.0 - radii[i])) * 2.0
        centers += forces * 0.5
        centers = np.clip(centers, 1e-6, 1.0 - 1e-6)
        if max_viol < 1e-7:
            break
    return centers

def run_packing():
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -1.0
    best_vars = None
    
    np.random.seed(42)
    
    # Phase 1: Hexagonal grid variations with systematic shifts
    for r0 in [0.085, 0.09, 0.095, 0.10]:
        for sx in [0.0, 0.01, -0.01, 0.02]:
            for sy in [0.0, 0.005, -0.005]:
                c, r = make_hex_init(r0, sx, sy, noise=0.002)
                c = resolve_overlaps(c, r, 200)
                v0 = np.zeros(3 * N)
                v0[0::3] = c[:, 0]
                v0[1::3] = c[:, 1]
                v0[2::3] = r
                
                try:
                    res = minimize(objective, v0, method='SLSQP', bounds=bounds,
                                   constraints=cons, options={'maxiter': 4000, 'ftol': 1e-13, 'disp': False})
                    s = -res.fun
                    if s > best_sum:
                        best_sum = s
                        best_vars = res.x.copy()
                except:
                    pass
                    
    # Phase 2: Random starts resolved by force-directed layout
    for _ in range(15):
        c = np.random.uniform(0.1, 0.9, (N, 2))
        r = np.random.uniform(0.06, 0.12, N)
        c = resolve_overlaps(c, r, 400)
        v0 = np.zeros(3 * N)
        v0[0::3] = c[:, 0]
        v0[1::3] = c[:, 1]
        v0[2::3] = r
        try:
            res = minimize(objective, v0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 4000, 'ftol': 1e-13, 'disp': False})
            s = -res.fun
            if s > best_sum:
                best_sum = s
                best_vars = res.x.copy()
        except:
            pass

    # Phase 3: Refinement & Perturbation on best found solution
    if best_vars is not None:
        # Fine-tuning with small Gaussian noise
        for _ in range(20):
            v_pert = best_vars + np.random.normal(0, 0.0008, 3 * N)
            for i in range(N):
                r = max(0.01, v_pert[3*i+2])
                v_pert[3*i+2] = r
                v_pert[3*i] = np.clip(v_pert[3*i], r, 1.0 - r)
                v_pert[3*i+1] = np.clip(v_pert[3*i+1], r, 1.0 - r)
            try:
                res = minimize(objective, v_pert, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 5000, 'ftol': 1e-14, 'disp': False})
                s = -res.fun
                if s > best_sum:
                    best_sum = s
                    best_vars = res.x.copy()
            except:
                pass
                
        # Local search with larger jumps for a subset of circles to escape deeper local minima
        for _ in range(15):
            v_jump = best_vars.copy()
            idx = np.random.choice(N, size=np.random.randint(3, 7), replace=False)
            for i in idx:
                v_jump[3*i] = np.random.uniform(0.1, 0.9)
                v_jump[3*i+1] = np.random.uniform(0.1, 0.9)
                v_jump[3*i+2] = np.random.uniform(0.05, 0.15)
            try:
                res = minimize(objective, v_jump, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 3000, 'ftol': 1e-13, 'disp': False})
                s = -res.fun
                if s > best_sum:
                    best_sum = s
                    best_vars = res.x.copy()
            except:
                pass

    # Fallback in case all optimizations fail
    if best_vars is None:
        c, r = make_hex_init(0.09)
        best_vars = np.concatenate([c.flatten(), r])
        best_sum = np.sum(r)

    centers = best_vars[:2*N].reshape(N, 2)
    radii = best_vars[2*N:]
    
    # Phase 4: Strict validation check and numerical repair
    valid = True
    for i in range(N):
        if radii[i] < 0 or centers[i,0] < radii[i] - 1e-10 or centers[i,0] + radii[i] > 1 + 1e-10 or \
           centers[i,1] < radii[i] - 1e-10 or centers[i,1] + radii[i] > 1 + 1e-10:
            valid = False
            break
    if valid:
        for i in range(N):
            for j in range(i+1, N):
                d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
                if d < radii[i] + radii[j] - 1e-10:
                    valid = False
                    break
            if not valid:
                break
                
    if not valid:
        # Aggressively shrink radii until strictly valid
        for _ in range(100):
            radii *= 0.998
            ok = True
            for i in range(N):
                if centers[i,0] < radii[i] - 1e-12 or centers[i,0] + radii[i] > 1 + 1e-12 or \
                   centers[i,1] < radii[i] - 1e-12 or centers[i,1] + radii[i] > 1 + 1e-12:
                    ok = False; break
            if ok:
                for i in range(N):
                    for j in range(i+1, N):
                        if np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1]) < radii[i]+radii[j]-1e-12:
                            ok = False; break
                    if not ok: break
            if ok: break
        best_sum = np.sum(radii)
        
    return centers, radii, float(best_sum)
