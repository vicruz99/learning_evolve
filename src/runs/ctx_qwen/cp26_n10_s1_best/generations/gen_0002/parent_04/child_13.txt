import numpy as np
from scipy.optimize import minimize, differential_evolution

N_CIRCLES = 26

def unpack(vars_arr):
    """Convert flat array to centers and radii."""
    n = N_CIRCLES
    centers = vars_arr[:2 * n].reshape(n, 2)
    radii = vars_arr[2 * n:]
    return centers, radii

def pack(centers, radii):
    """Convert centers and radii to flat array."""
    return np.concatenate([centers.flatten(), radii])

def objective(vars_arr):
    """Minimize negative sum of radii."""
    radii = vars_arr[2 * N_CIRCLES:]
    return -np.sum(radii)

def constraints_squared(vars_arr):
    """Squared distance constraints (smoother optimization)."""
    centers, radii = unpack(vars_arr)
    n = N_CIRCLES
    
    parts = []
    
    # Boundary: x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
    parts.append(centers[:, 0] - radii)
    parts.append(1.0 - centers[:, 0] - radii)
    parts.append(centers[:, 1] - radii)
    parts.append(1.0 - centers[:, 1] - radii)
    
    # Squared overlap: dist^2 - (r_i + r_j)^2 >= 0
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dr = radii[i] + radii[j]
            parts.append(dx * dx + dy * dy - dr * dr)
    
    return np.concatenate(parts)

def constraints_linear(vars_arr):
    """Linear distance constraints (dist - r_i - r_j >= 0), better gradient near contact."""
    centers, radii = unpack(vars_arr)
    n = N_CIRCLES
    
    parts = []
    
    # Boundary
    parts.append(centers[:, 0] - radii)
    parts.append(1.0 - centers[:, 0] - radii)
    parts.append(centers[:, 1] - radii)
    parts.append(1.0 - centers[:, 1] - radii)
    
    # Linear overlap
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            parts.append(dist - radii[i] - radii[j])
    
    return np.concatenate(parts)

def force_directed_init(n, seed, num_iters=5000, init_radius=0.10):
    """Force-directed simulation for good initial configuration."""
    np.random.seed(seed)
    
    centers = np.random.uniform(0.15, 0.85, (n, 2))
    radii = np.full(n, init_radius)
    
    for iteration in range(num_iters):
        forces = np.zeros((n, 2))
        cooling = max(0.0001, 0.01 * (1.0 - iteration / num_iters))
        
        for i in range(n):
            # Wall repulsion
            for wall_x, sign in [(0.0, 1), (1.0, -1)]:
                dist_wall = (centers[i, 0] - wall_x) * sign
                if dist_wall < radii[i] + 0.02:
                    forces[i, 0] += sign * (radii[i] + 0.02 - dist_wall) * 10
            for wall_y, sign in [(0.0, 1), (1.0, -1)]:
                dist_wall = (centers[i, 1] - wall_y) * sign
                if dist_wall < radii[i] + 0.02:
                    forces[i, 1] += sign * (radii[i] + 0.02 - dist_wall) * 10
            
            for j in range(i + 1, n):
                dx = centers[j, 0] - centers[i, 0]
                dy = centers[j, 1] - centers[i, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                
                if dist < radii[i] + radii[j] + 0.05 and dist > 1e-10:
                    overlap = (radii[i] + radii[j] + 0.02) - dist
                    strength = overlap * 5.0 / (dist + 0.01)
                    fx = dx / dist * strength
                    fy = dy / dist * strength
                    forces[i] -= np.array([fx, fy])
                    forces[j] += np.array([fx, fy])
        
        centers += forces * cooling
        centers[:, 0] = np.clip(centers[:, 0], 0.01, 0.99)
        centers[:, 1] = np.clip(centers[:, 1], 0.01, 0.99)
    
    # Estimate safe radii
    for i in range(n):
        min_d = min(centers[i, 0], 1.0 - centers[i, 0], 
                    centers[i, 1], 1.0 - centers[i, 1])
        for j in range(n):
            if i != j:
                d = np.sqrt((centers[i, 0] - centers[j, 0])**2 + 
                           (centers[i, 1] - centers[j, 1])**2)
                if d < min_d:
                    min_d = d
        radii[i] = min_d / 2.0 * 0.85
    
    return centers, radii

def hex_lattice_init(n, r_start, row_counts, seed=0):
    """Generate hexagonal lattice with specified row counts."""
    np.random.seed(seed)
    centers = np.zeros((n, 2))
    radii = np.full(n, r_start)
    
    idx = 0
    y = r_start
    row = 0
    
    for count in row_counts:
        x_start = r_start if row % 2 == 0 else 2 * r_start
        for k in range(count):
            if idx >= n:
                break
            centers[idx, 0] = x_start + k * 2 * r_start
            centers[idx, 1] = y
            idx += 1
        y += np.sqrt(3) * r_start
        row += 1
    
    while idx < n:
        centers[idx, 0] = np.random.uniform(r_start, 1 - r_start)
        centers[idx, 1] = np.random.uniform(r_start, 1 - r_start)
        radii[idx] = r_start * 0.7
        idx += 1
    
    centers += np.random.normal(0, 0.005, centers.shape)
    centers[:, 0] = np.clip(centers[:, 0], radii + 0.001, 1 - radii - 0.001)
    centers[:, 1] = np.clip(centers[:, 1], radii + 0.001, 1 - radii - 0.001)
    
    return centers, radii

def get_bounds():
    """Variable bounds."""
    bounds = []
    for _ in range(N_CIRCLES):
        bounds.extend([(0.0, 1.0), (0.0, 1.0), (1e-8, 0.5)])
    return bounds

def project_to_feasible(centers, radii):
    """Project centers to respect boundary constraints."""
    for i in range(N_CIRCLES):
        r = max(0.001, radii[i])
        centers[i, 0] = np.clip(centers[i, 0], r, 1.0 - r)
        centers[i, 1] = np.clip(centers[i, 1], r, 1.0 - r)
    return centers

def is_valid_packing(centers, radii, tol=1e-10):
    """Check if packing is valid."""
    n = len(radii)
    for i in range(n):
        if radii[i] < 0:
            return False
        x, y = centers[i]
        r = radii[i]
        if x - r < -tol or x + r > 1 + tol or y - r < -tol or y + r > 1 + tol:
            return False
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt((centers[i, 0] - centers[j, 0])**2 + 
                          (centers[i, 1] - centers[j, 1])**2)
            if dist < radii[i] + radii[j] - tol:
                return False
    return True

def repair_packing(centers, radii):
    """Repair a slightly invalid packing by shrinking radii minimally."""
    for _ in range(200):
        if is_valid_packing(centers, radii, tol=1e-12):
            break
        radii *= 0.999
        centers = project_to_feasible(centers, radii)
    return centers, radii

def optimize_from_init(x0, max_iter=15000, method='SLSQP'):
    """Run optimization from an initial configuration."""
    bounds = get_bounds()
    
    if method == 'SLSQP':
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                          constraints={'type': 'ineq', 'fun': constraints_squared},
                          options={'maxiter': max_iter, 'ftol': 1e-14, 'disp': False})
            return res
        except:
            return None
    elif method == 'trust-constr':
        try:
            from scipy.optimize import NonlinearConstraint
            cons = NonlinearConstraint(constraints_squared, 0, np.inf)
            res = minimize(objective, x0, method='trust-constr', bounds=bounds,
                          constraints=[cons],
                          options={'maxiter': max_iter, 'gtol': 1e-10})
            return res
        except:
            return None
