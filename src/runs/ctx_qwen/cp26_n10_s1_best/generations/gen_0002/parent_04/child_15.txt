    return -np.sum(vars[2::3])

    xs = vars[0::3]
    ys = vars[1::3]
    rs = vars[2::3]
    
    parts = []
    # Boundary constraints
    parts.append(xs - rs)
    parts.append(1.0 - xs - rs)
    parts.append(ys - rs)
    parts.append(1.0 - ys - rs)
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    tri = np.tril_indices(n, -1)
    parts.append(dx[tri]**2 + dy[tri]**2 - dr[tri]**2)
    
    return np.concatenate(parts)
