from scipy.optimize import minimize

N = 26

def objective(vars):

    return -np.sum(vars[2::3])

def constraints(vars):
    """Compute all inequality constraints g(vars) >= 0."""
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    c = []
    
    # Boundary constraints: 4*N
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)
    
    # Overlap constraints: N*(N-1)/2 using squared distances for better conditioning
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dr = r[:, None] + r[None, :]

    c.append(dx[idx_i, idx_j]**2 + dy[idx_i, idx_j]**2 - dr[idx_i, idx_j]**2)

    return b
