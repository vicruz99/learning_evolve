import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def objective(vars):
    """Minimize negative sum of radii."""
    return -np.sum(vars[2::3])

def constraints(vars):
    """Returns array of constraint values g(vars) >= 0."""
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    # Boundary constraints (4*N)
    c = np.empty(4 * N_CIRCLES + N_CIRCLES * (N_CIRCLES - 1) // 2)
    idx = 0
    c[idx:idx+N_CIRCLES] = x - r; idx += N_CIRCLES
    c[idx:idx+N_CIRCLES] = 1.0 - x - r; idx += N_CIRCLES
    c[idx:idx+N_CIRCLES] = y - r; idx += N_CIRCLES
    c[idx:idx+N_CIRCLES] = 1.0 - y - r; idx += N_CIRCLES
    
    # Overlap constraints (N*(N-1)/2)
    # Using squared distance to maintain smooth gradients
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dr = r[:, None] + r[None, :]
    tri = np.triu_indices(N_CIRCLES, 1)
    c[idx:] = dx[tri]**2 + dy[tri]**2 - dr[tri]**2
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [0, 0.5]."""
    return [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N_CIRCLES

def generate_dense_init(seed):
    """Generates a dense initial configuration via growth simulation."""
    np.random.seed(seed)
    c = np.random.uniform(0.2, 0.8, (N_CIRCLES, 2))
    r = np.full(N_CIRCLES, 0.01)
    
    # Gradually grow circles while resolving overlaps
    for _ in range(1200):
        r *= 1.0005
        
        # Hard wall constraints
        for i in range(N_CIRCLES):
            for dim in range(2):
                if c[i, dim] - r[i] < 0: c[i, dim] = r[i]
                if c[i, dim] + r[i] > 1.0: c[i, dim] = 1.0 - r[i]
                
        # Pairwise repulsion to resolve overlaps
        for i in range(N_CIRCLES):
            for j in range(i + 1, N_CIRCLES):
                dx = c[j, 0] - c[i, 0]
                dy = c[j, 1] - c[i, 1]
                dist = np.hypot(dx, dy)
                min_d = r[i] + r[j]
                if dist < min_d and dist > 1e-9:
                    shift = (min_d - dist) * 0.5
                    nx, ny = dx / dist, dy / dist
                    c[i, 0] -= shift * nx
                    c[i, 1] -= shift * ny
                    c[j, 0] += shift * nx
                    c[j, 1] += shift * ny
                    
    # Final projection to strictly satisfy bounds
    r = np.clip(r, 0.01, 0.5)
    c[:, 0] = np.clip(c[:, 0], r, 1.0 - r)
    c[:, 1] = np.clip(c[:, 1], r, 1.0 - r)
    
    x0 = np.zeros(3 * N_CIRCLES)
    x0[0::3] = c[:, 0]
    x0[1::3] = c[:, 1]
    x0[2::3] = r
    return x0

def hex_init(r_start, seed):
    """Generates a perturbed hexagonal lattice initialization."""
    np.random.seed(seed)
    c = np.zeros((N_CIRCLES, 2))
    r = np.full(N_CIRCLES, r_start)
    idx = 0
    y = r_start
    row = 0
    
    while idx < N_CIRCLES:
        x = r_start
        if row % 2 == 1: x = 2 * r_start
        while idx < N_CIRCLES:
            if x + r_start > 1.0: break
            c[idx] = [x, y]
            idx += 1
            x += 2 * r_start
        y += np.sqrt(3) * r_start
        row += 1
        
    # Add noise to break exact symmetry
    c += np.random.normal(0, 0.004, c.shape)
    c[:, 0] = np.clip(c[:, 0], r_start + 1e-5, 1.0 - r_start - 1e-5)
    c[:, 1] = np.clip(c[:, 1], r_start + 1e-5, 1.0 - r_start - 1e-5)
    
    x0 = np.zeros(3 * N_CIRCLES)
    x0[0::3] = c[:, 0]
    x0[1::3] = c[:, 1]
    x0[2::3] = r
    return x0

def run_packing():

    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -1.0

            if -res.fun > best_sum:
                best_sum = -res.fun
                best_vars = res.x.copy()
        except Exception:
            continue

    # Shrinking radii creates slack, perturbation escapes local minima, re-optimization expands into new config

            x0 = best_vars.copy()
            x0[2::3] *= 0.97  # Shrink radii

            x0[1::3] += np.random.normal(0, 0.004, N_CIRCLES)
            
            # Project back to feasible bounds
            for i in range(N_CIRCLES):
                ri = max(0.01, x0[3*i+2])

                x0[3*i+1] = np.clip(x0[3*i+1], ri, 1.0 - ri)

                    best_sum = -res.fun

    # Strict validity repair for numerical drift

            break

                        ok = False
                        break
                if not ok:
                    break
            if ok:
                break
                
    return centers, radii, float(np.sum(radii))
