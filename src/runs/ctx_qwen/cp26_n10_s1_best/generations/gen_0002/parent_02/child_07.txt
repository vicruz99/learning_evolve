import numpy as np
from scipy.optimize import minimize

N = 26

def unpack(vars):
    """Converts flat optimization vector to centers and radii arrays."""
    c = vars.reshape(N, 3)
    return c[:, :2], c[:, 2]

def objective(vars):
    """Objective to minimize: negative sum of radii."""
    return -np.sum(vars[2::3])

def constraints(vars):
    """Returns array of constraint values g(x) >= 0."""
    c = vars.reshape(N, 3)
    x, y, r = c[:, 0], c[:, 1], c[:, 2]
    
    # Preallocate constraint array
    num_overlap = N * (N - 1) // 2
    res = np.empty(N * 4 + num_overlap)
    
    # Boundary constraints: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    res[:N] = x - r
    res[N:2*N] = 1.0 - x - r
    res[2*N:3*N] = y - r
    res[3*N:4*N] = 1.0 - y - r
    
    # Overlap constraints: squared distance - (r_i + r_j)^2 >= 0
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dr = r[:, None] + r[None, :]
    
    tri = np.tril_indices(N, -1)
    res[4*N:] = dx[tri]**2 + dy[tri]**2 - dr[tri]**2
    
    return res

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [0, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
    return b

    centers = np.zeros((N, 2))

    idx = 0
    y = r_start
    row = 0
    
    while idx < N and y + r_start <= 1.0:
        x = r_start + (row % 2) * r_start
        while idx < N and x + r_start <= 1.0:
            centers[idx] = [x, y]
            idx += 1
            x += 2 * r_start
        y += np.sqrt(3) * r_start
        row += 1
        
    # Fill remaining if grid didn't cover all 26
    while idx < N:
        centers[idx] = np.random.uniform(r_start, 1.0 - r_start, 2)
        idx += 1
        
    # Add slight noise to break exact symmetry
    centers += np.random.normal(0, 0.002, centers.shape)
    centers = np.clip(centers, r_start, 1.0 - r_start)
    return centers, radii
