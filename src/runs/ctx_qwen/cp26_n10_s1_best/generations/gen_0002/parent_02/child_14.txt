def objective(vars):

    return -np.sum(vars[2::3])

def compute_constraints(vars):
    """
    Computes all inequality constraints g(vars) >= 0.
    Uses vectorized operations for speed.
    """
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    c = np.concatenate([
        x - r,
        1.0 - x - r,
        y - r,
        1.0 - y - r
    ])
    
    # Squared overlap constraints: dist^2 >= (r_i + r_j)^2
    triu = np.triu_indices(N, k=1)
    dx = x[triu[0]] - x[triu[1]]
    dy = y[triu[0]] - y[triu[1]]
    dr = r[triu[0]] + r[triu[1]]
    c = np.concatenate([c, dx**2 + dy**2 - dr**2])
    
    return c
