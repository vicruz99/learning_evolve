import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [0, 0.5]."""
    b = []
    for _ in range(N_CIRCLES):
        b.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
    return b

def constraints_func(vars):
    """
    Computes all inequality constraints g(vars) >= 0.
    Boundary: x >= r, 1-x >= r, y >= r, 1-y >= r
    Overlap: dist(i,j) >= r_i + r_j
    """
    n = N_CIRCLES
    cs = vars.reshape(n, 3)
    x = cs[:, 0]
    y = cs[:, 1]
    r = cs[:, 2]
    
    # Boundary constraints
    b = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    
    # Overlap constraints
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist = np.sqrt(dx**2 + dy**2)
    dr = r[:, None] + r[None, :]
    
    tri = np.tril_indices(n, -1)
    o = dist[tri] - dr[tri]
    
    return np.concatenate([b, o])

def objective(vars):
    """Objective to minimize: negative sum of radii."""
    return -np.sum(vars[2::3])

def simulate_init(n, seed=0, steps=800):
    """
    Force-directed simulation to generate a densely packed initial configuration.
    Circles repel each other and walls, while gradually growing in radius.
    """
    np.random.seed(seed)
    centers = np.random.uniform(0.3, 0.7, (n, 2))
    radii = np.full(n, 0.02)
    
    for _ in range(steps):
        diff = centers[:, None, :] - centers[None, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2) + 1e-12)
        r_sum = radii[:, None] + radii[None, :]
        overlap = np.maximum(0.0, r_sum - dists)
        
        # Direction vectors from i to j
        dir_vec = diff / dists[..., None]
        # Repulsive force proportional to overlap
        f_mag = overlap * 0.8 / (dists + 1e-8)
        force_matrix = dir_vec * f_mag[..., None]
        
        # Net force on each circle
        forces = np.sum(force_matrix, axis=1)
        
        # Strong wall repulsion
        for i in range(n):
            if centers[i, 0] < radii[i]: forces[i, 0] += 2.0
            elif centers[i, 0] > 1.0 - radii[i]: forces[i, 0] -= 2.0
            if centers[i, 1] < radii[i]: forces[i, 1] += 2.0
            elif centers[i, 1] > 1.0 - radii[i]: forces[i, 1] -= 2.0
            
        centers += forces * 0.015
        centers = np.clip(centers, 0.005, 0.995)
        
        # Slowly inflate all circles to increase density
        radii += 0.000025
        
    return centers, radii

def ensure_feasibility(centers, radii, n):
    """Clips positions to boundaries and slightly shrinks radii if overlaps persist."""
    for i in range(n):
        radii[i] = max(radii[i], 0.01)
        centers[i, 0] = np.clip(centers[i, 0], radii[i], 1.0 - radii[i])
        centers[i, 1] = np.clip(centers[i, 1], radii[i], 1.0 - radii[i])
        
    # Quick overlap resolution
    for _ in range(30):
        dists = np.sqrt(np.sum((centers[:, None, :] - centers[None, :, :])**2, axis=2) + 1e-12)
        r_sums = radii[:, None] + radii[None, :]
        tri = np.tril_indices(n, -1)
        overlap = r_sums[tri] - dists[tri]
        if np.max(overlap) < 1e-6:
            break
        # Minimal shrinkage to guarantee strict feasibility for optimizer start
        radii *= 0.995
        for i in range(n):
            centers[i, 0] = np.clip(centers[i, 0], radii[i], 1.0 - radii[i])
            centers[i, 1] = np.clip(centers[i, 1], radii[i], 1.0 - radii[i])
            
    return centers, radii

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    n = N_CIRCLES
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints_func}

    best_radii = None
    best_sum = -1.0

        x0[0::3] = c_init[:, 0]
        x0[1::3] = c_init[:, 1]
        x0[2::3] = r_init
        
        try:

                           options={'maxiter': 15000, 'ftol': 1e-12, 'disp': False})

    return best_centers, best_radii, float(best_sum)
