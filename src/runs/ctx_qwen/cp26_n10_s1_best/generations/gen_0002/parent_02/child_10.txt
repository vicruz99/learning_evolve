import numpy as np
from scipy.optimize import minimize

N = 26
TRIL_INDICES = np.tril_indices(N, -1)

def unpack(vars):
    """Converts flat optimization vector to centers and radii arrays."""
    c = vars.reshape(N, 3)
    return c[:, :2], c[:, 2]

def objective(vars):
    """Objective to minimize: negative sum of radii."""
    _, r = unpack(vars)
    return -np.sum(r)

def constraints(vars):
    """Returns array of constraint values g(x) >= 0."""
    c = vars.reshape(N, 3)
    x, y, r = c[:, 0], c[:, 1], c[:, 2]
    cons = np.empty(N * 4 + N * (N - 1) // 2)
    k = 0
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    cons[k:k+N] = x - r; k += N
    cons[k:k+N] = 1.0 - x - r; k += N
    cons[k:k+N] = y - r; k += N
    cons[k:k+N] = 1.0 - y - r; k += N
    # Non-overlap constraints: dist^2 >= (r_i + r_j)^2
    ix, iy = TRIL_INDICES
    dx = x[ix] - x[iy]
    dy = y[ix] - y[iy]
    dr = r[ix] + r[iy]
    cons[k:] = dx*dx + dy*dy - dr*dr
    return cons

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [0, 0.5]."""
    return [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N

def hex_init(seed, r0):
    """Generates a perturbed hexagonal lattice initialization."""
    np.random.seed(seed)
    centers = []
    y = r0
    row = 0
    while len(centers) < N:
        x = r0 if row % 2 == 0 else 2 * r0
        while x + r0 <= 1.0 and len(centers) < N:
            centers.append([x, y])
            x += 2 * r0
        y += np.sqrt(3) * r0
        row += 1
    # Fill remaining spots if grid is sparse
    while len(centers) < N:
        centers.append([np.random.uniform(r0, 1-r0), np.random.uniform(r0, 1-r0)])
    centers = np.array(centers[:N])
    # Perturb to break exact symmetry and help optimizer explore
    centers += np.random.normal(0, 0.004, centers.shape)
    centers = np.clip(centers, r0 + 1e-6, 1.0 - r0 - 1e-6)
    radii = np.full(N, r0) + np.random.uniform(-0.001, 0.001, N)
    radii = np.clip(radii, 0.001, 0.4)
    return centers, radii

def run_packing():
    best_vars = None
    best_sum = -1.0
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    
    # Phase 1: Diverse hexagonal starts with varying densities
    inits = []
    for s in range(40):
        r0 = 0.085 + s * 0.0009
        inits.append(hex_init(s, r0))
    for s in range(20):
        inits.append(hex_init(s+100, 0.095))
        
    for c, r in inits:
        x0 = np.zeros(3 * N)
        x0[0::3] = c[:, 0]
        x0[1::3] = c[:, 1]
        x0[2::3] = r
        # Strictly project to feasible bounds to aid SLSQP convergence
        x0[0::3] = np.clip(x0[0::3], x0[2::3], 1.0 - x0[2::3])
        x0[1::3] = np.clip(x0[1::3], x0[2::3], 1.0 - x0[2::3])
        x0[2::3] = np.clip(x0[2::3], 0.001, 0.4)
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 12000, 'ftol': 1e-12, 'disp': False})
            if not np.isnan(res.fun):
                c_opt, r_opt = unpack(res.x)
                s = np.sum(r_opt)
                if s > best_sum:
                    valid = True
                    for i in range(N):
                        if r_opt[i] < 1e-6 or c_opt[i,0] < r_opt[i] - 1e-9 or c_opt[i,0] + r_opt[i] > 1 + 1e-9 or \
                           c_opt[i,1] < r_opt[i] - 1e-9 or c_opt[i,1] + r_opt[i] > 1 + 1e-9:
                            valid = False; break
                    if valid:
                        for i in range(N):
                            for j in range(i+1, N):
                                d = np.sqrt(np.sum((c_opt[i]-c_opt[j])**2))
                                if d < r_opt[i] + r_opt[j] - 1e-9:
                                    valid = False; break
                            if not valid: break
                    if valid:
                        best_sum = s
                        best_vars = res.x.copy()
        except Exception:
            pass
            
    # Phase 2: Local refinement around the best found solution
    if best_vars is not None:
        for _ in range(10):
            x_pert = best_vars + np.random.normal(0, 5e-5, 3*N)
            x_pert[0::3] = np.clip(x_pert[0::3], x_pert[2::3], 1.0 - x_pert[2::3])
            x_pert[1::3] = np.clip(x_pert[1::3], x_pert[2::3], 1.0 - x_pert[2::3])
            try:
                res = minimize(objective, x_pert, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 15000, 'ftol': 1e-13, 'disp': False})
                if not np.isnan(res.fun) and -res.fun > best_sum:
                    best_vars = res.x.copy()
                    best_sum = -res.fun
            except Exception:
                break
                
    # Fallback safety
    if best_vars is None:
        c, r = hex_init(0, 0.08)
        best_vars = np.zeros(3*N)
        best_vars[0::3] = c[:,0]; best_vars[1::3] = c[:,1]; best_vars[2::3] = r
        best_sum = np.sum(r)
        
    centers = best_vars.reshape(N, 3)[:, :2]
    radii = best_vars.reshape(N, 3)[:, 2]
    return centers, radii, float(best_sum)
