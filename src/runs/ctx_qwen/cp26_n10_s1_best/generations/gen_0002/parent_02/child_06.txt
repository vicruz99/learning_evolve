import numpy as np
from scipy.optimize import minimize

N = 26

def objective(vars):
    """Minimize negative sum of radii."""
    return -np.sum(vars[2::3])

def constraints(vars):
    """Compute all boundary and non-overlap constraints as a single vector >= 0."""
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    b = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    i, j = np.tril_indices(N, -1)
    dx = x[i] - x[j]
    dy = y[i] - y[j]
    dr = r[i] + r[j]
    o = dx*dx + dy*dy - dr*dr
    
    return np.concatenate([b, o])

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [0, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
    return b

def grow_and_relax(centers, radii, steps=1000, growth=1.0003):
    """Force-directed expansion to find dense initial packings."""
    for _ in range(steps):
        radii *= growth
        # Relax overlaps multiple times per growth step
        for _ in range(30):
            for i in range(N):
                fx, fy = 0.0, 0.0
                cx, cy = centers[i]
                ri = radii[i]
                
                # Wall repulsion
                for w, p, dim in [(0, cx, 0), (1, 1.0-cx, 0), (0, cy, 1), (1, 1.0-cy, 1)]:
                    if p < ri:
                        push = ri - p
                        if dim == 0:
                            fx += push if w == 1 else -push
                        else:
                            fy += push if w == 1 else -push
                            
                # Circle repulsion
                for j in range(i+1, N):
                    dx = centers[j,0] - cx
                    dy = centers[j,1] - cy
                    d = np.hypot(dx, dy)
                    req = ri + radii[j]
                    if d < req and d > 1e-9:
                        f = (req - d) / d * 0.5
                        fx += f * dx
                        fy += f * dy
                        
                centers[i,0] += fx
                centers[i,1] += fy
                
        # Project back to feasible box
        centers[:,0] = np.clip(centers[:,0], radii, 1.0-radii)
        centers[:,1] = np.clip(centers[:,1], radii, 1.0-radii)
        
    return centers, radii

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    best_sum = -1.0
    best_vars = None
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    
    configs = []

                x += 2.0*r0
            y += np.sqrt(3.0)*r0
            row += 1
        while len(centers) < N:
            centers.append([np.random.uniform(r0, 1.0-r0), np.random.uniform(r0, 1.0-r0)])
        centers = np.array(centers[:N])
        configs.append((centers, np.full(N, r0)))
        
    # 2. Random starts + growth relaxation to find high-density configurations
    np.random.seed(42)
    for _ in range(10):
        c = np.random.uniform(0.15, 0.85, (N, 2))
        r = np.full(N, 0.06)
        c, r = grow_and_relax(c, r, steps=800, growth=1.0004)
        configs.append((c, r))
        
    # 3. Perturbed hex grids to break symmetry
    for r0 in [0.09, 0.095]:
        c, r = configs[0]
        c_p = c + np.random.uniform(-0.02, 0.02, c.shape)
        c_p = np.clip(c_p, r0, 1.0-r0)
        configs.append((c_p, r.copy()))

    # Optimization loop over all configurations
    for c_init, r_init in configs:

        x0[1::3] = np.clip(x0[1::3], 1e-6, 1.0-1e-6)

                           options={'maxiter': 20000, 'ftol': 1e-13, 'disp': False})
            s = -res.fun
            vals = constraints(res.x)

            pass

                        break

    return centers, radii, float(np.sum(radii))
