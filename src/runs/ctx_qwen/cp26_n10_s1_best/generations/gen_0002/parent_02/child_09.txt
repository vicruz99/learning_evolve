import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def unpack(vars):
    """Converts flat optimization vector to centers and radii arrays."""
    centers = np.zeros((N_CIRCLES, 2))
    radii = np.zeros(N_CIRCLES)
    centers[:, 0] = vars[0::3]
    centers[:, 1] = vars[1::3]
    radii[:] = vars[2::3]
    return centers, radii

def objective(vars):
    """Objective to minimize: negative sum of radii."""
    return -np.sum(vars[2::3])

def constraints_func(vars):
    """Returns array of constraint values g(x) >= 0."""
    xs = vars[0::3]
    ys = vars[1::3]
    rs = vars[2::3]
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    b1 = xs - rs
    b2 = 1.0 - xs - rs
    b3 = ys - rs
    b4 = 1.0 - ys - rs
    
    # Overlap constraints (squared distance): dist^2 >= (r_i + r_j)^2
    dx = xs[:, None] - xs
    dy = ys[:, None] - ys
    dr = rs[:, None] + rs
    
    iu, ju = np.triu_indices(N_CIRCLES, k=1)
    o = dx[iu, ju]**2 + dy[iu, ju]**2 - dr[iu, ju]**2
    
    return np.concatenate([b1, b2, b3, b4, o])

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-4, 0.5]."""
    b = []
    for _ in range(N_CIRCLES):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-4, 0.5)])
    return b

def create_hex_init(pat, r_init, seed):
    """Generates a hexagonal lattice initialization."""
    np.random.seed(seed)
    centers = np.zeros((N_CIRCLES, 2))
    radii = np.full(N_CIRCLES, r_init)
    idx = 0
    y_step = np.sqrt(3) * r_init
    y = r_init
    for i, count in enumerate(pat):
        x_start = r_init + (r_init if i % 2 == 1 else 0.0)
        for k in range(count):
            if idx < N_CIRCLES:
                x = x_start + k * 2 * r_init
                centers[idx, 0] = x + np.random.normal(0, 0.005)
                centers[idx, 1] = y + np.random.normal(0, 0.005)
                idx += 1
        y += y_step
    while idx < N_CIRCLES:
        centers[idx, 0] = np.random.uniform(r_init, 1.0 - r_init)
        centers[idx, 1] = np.random.uniform(r_init, 1.0 - r_init)
        idx += 1
    return centers, radii

def create_force_init(seed):
    """Generates initial configuration using force-directed relaxation."""
    np.random.seed(seed)
    centers = np.random.uniform(0.15, 0.85, (N_CIRCLES, 2))
    radii = np.full(N_CIRCLES, 0.08)
    
    for _ in range(500):
        forces = np.zeros_like(centers)
        for i in range(N_CIRCLES):
            # Wall repulsion
            for d in range(2):
                if centers[i, d] < radii[i]:
                    forces[i, d] += 0.02
                elif centers[i, d] > 1.0 - radii[i]:
                    forces[i, d] -= 0.02
                    
            for j in range(N_CIRCLES):
                if i == j: continue
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.hypot(dx, dy)
                if dist < radii[i] + radii[j] and dist > 1e-6:
                    overlap = (radii[i] + radii[j] - dist) * 0.5
                    forces[i, 0] += overlap * dx / dist
                    forces[i, 1] += overlap * dy / dist
                    
        centers += forces * 0.15
        centers = np.clip(centers, 0.0, 1.0)
        
    # Recalculate safe radii based on final positions
    for i in range(N_CIRCLES):
        min_d = min(centers[i, 0], 1.0 - centers[i, 0], 
                    centers[i, 1], 1.0 - centers[i, 1])
        for j in range(N_CIRCLES):
            if i != j:
                d = np.hypot(centers[i, 0] - centers[j, 0], 
                             centers[i, 1] - centers[j, 1])
                if d < min_d:
                    min_d = d
        radii[i] = min_d / 2.0 * 0.95
        
    return centers, radii

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    best_vars = None
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints_func}
    
    # Prepare diverse initial configurations
    configs = []
    patterns = [

        [6, 5, 6, 5, 4],
        [5, 5, 5, 5, 6],

        [5, 6, 6, 5, 4]
    ]

            configs.append((c, r))

        configs.append((c, r))
        
    # Main optimization loop
    for c_init, r_init in configs:
        x0 = np.zeros(3 * N_CIRCLES)
        x0[0::3] = c_init[:, 0]
        x0[1::3] = c_init[:, 1]
        x0[2::3] = r_init

        x0 += np.random.normal(0, 1e-4, x0.shape)

            ri = np.clip(x0[3*i+2], 1e-4, 0.5)
            x0[3*i+2] = ri
            x0[3*i] = np.clip(x0[3*i], ri, 1.0 - ri)
            x0[3*i+1] = np.clip(x0[3*i+1], ri, 1.0 - ri)
            
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, 
                           constraints=cons, options={'maxiter': 20000, 'ftol': 1e-15, 'disp': False})
            if not np.isnan(res.fun):
                curr_sum = -res.fun
                if curr_sum > best_sum:
                    best_sum = curr_sum
                    best_vars = res.x.copy()
        except Exception:
            continue

           centers[i, 1] < r - 1e-9 or centers[i, 1] > 1.0 - r + 1e-9:
            valid = False
            break
    if valid:
        for i in range(N_CIRCLES):
            for j in range(i + 1, N_CIRCLES):
                dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                if dist < radii[i] + radii[j] - 1e-9:

                    break
