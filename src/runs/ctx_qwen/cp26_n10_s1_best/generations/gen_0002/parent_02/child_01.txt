import numpy as np
from scipy.optimize import minimize

N = 26

def unpack(vars):
    """Converts flat optimization vector to centers and radii arrays."""
    centers = np.zeros((N, 2))
    centers[:, 0] = vars[:N]
    centers[:, 1] = vars[N:2*N]
    radii = vars[2*N:]
    return centers, radii

def objective(vars):
    """Objective to minimize: negative sum of radii."""
    return -np.sum(vars[2*N:])

def compute_constraints(vars):
    """Returns array of constraint values g(x) >= 0."""
    x = vars[:N]
    y = vars[N:2*N]
    r = vars[2*N:]
    
    # Boundary constraints: x-r>=0, 1-x-r>=0, y-r>=0, 1-y-r>=0
    b = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    
    # Overlap constraints (squared): dist^2 - (ri+rj)^2 >= 0
    dx = x[:, None] - x
    dy = y[:, None] - y
    dr = r[:, None] + r
    
    i, j = np.triu_indices(N, k=1)
    o = dx[i, j]**2 + dy[i, j]**2 - dr[i, j]**2
    
    return np.concatenate([b, o])

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [0, 0.5]."""
    return [(0.0, 1.0)] * N + [(0.0, 1.0)] * N + [(0.0, 0.5)] * N

def create_init(seed, ptype='hex'):
    """Generates initial configuration vector."""
    np.random.seed(seed)
    if ptype == 'hex':
        centers = []
        r_init = 0.085 + np.random.uniform(0, 0.015)
        y = r_init
        row = 0
        while len(centers) < N:
            x = r_init if row % 2 == 0 else 2 * r_init
            while len(centers) < N and x + r_init <= 1.0:
                centers.append([x, y])
                x += 2 * r_init
            y += np.sqrt(3) * r_init
            row += 1
        centers = np.array(centers[:N])
    elif ptype == 'grid':
        centers = []
        step = 0.17 + np.random.uniform(0, 0.02)
        y = step
        while len(centers) < N:
            x = step
            while len(centers) < N and x + step <= 1.0:
                centers.append([x, y])
                x += step
            y += step
        centers = np.array(centers[:N])
    else:
        centers = np.random.uniform(0.15, 0.85, (N, 2))
        
    # Perturb to break symmetry
    centers += np.random.normal(0, 0.005, centers.shape)
    centers = np.clip(centers, 0.02, 0.98)
    radii = np.full(N, 0.07 + np.random.uniform(0, 0.02))
    
    # Flatten to [x1...xN, y1...yN, r1...rN]
    return np.concatenate([centers[:, 0], centers[:, 1], radii])

    bounds = get_bounds()

    for x0 in starts:

                best_sum = -res.fun
                best_vars = res.x.copy()
        except Exception:
            continue

    if best_vars is not None:

            # Project perturbed variables back to feasible bounds
            r = x0[2*N:]
            x0[:N] = np.clip(x0[:N], r, 1.0 - r)

    return centers, radii, float(np.sum(radii))
