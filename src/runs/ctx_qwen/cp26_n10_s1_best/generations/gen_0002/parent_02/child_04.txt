import numpy as np
from scipy.optimize import minimize

N = 26

def objective(vars):
    """Minimize negative sum of radii."""
    return -np.sum(vars[2*N:])

def constraints(vars):
    """Returns array of constraint values g(x) >= 0."""
    n = N
    x = vars[:n]
    y = vars[n:2*n]
    r = vars[2*n:]
    
    c = []
    # Boundary constraints
    c.append(x - r)
    c.append(1.0 - x - r)
    c.append(y - r)
    c.append(1.0 - y - r)
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = x[:, None] - x
    dy = y[:, None] - y
    dr = r[:, None] + r
    tri = np.tril_indices(n, -1)
    c.append(dx[tri]**2 + dy[tri]**2 - dr[tri]**2)
    
    return np.concatenate(c)

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [0.01, 0.2]."""
    return [(0.0, 1.0)] * (2 * N) + [(0.01, 0.2)] * N

def ensure_feasible(vars, n):
    """Clips variables to guarantee initial feasibility."""
    x = vars[:n].copy()
    y = vars[n:2*n].copy()
    r = vars[2*n:].copy()
    r = np.clip(r, 0.01, 0.2)
    x = np.clip(x, r, 1.0 - r)
    y = np.clip(y, r, 1.0 - r)
    vars[:n] = x
    vars[n:2*n] = y
    vars[2*n:] = r
    return vars

def hex_init(seed, r_val=0.085):
    """Generates a hexagonal lattice initialization."""
    np.random.seed(seed)
    vars = np.zeros(3*N)
    idx = 0
    y = r_val
    row = 0
    while idx < N and y + r_val <= 1.0:
        x = r_val + (r_val if row % 2 == 1 else 0)

            vars[idx] = x + np.random.uniform(-0.005, 0.005)
            vars[N+idx] = y + np.random.uniform(-0.005, 0.005)
            vars[2*N+idx] = r_val + np.random.uniform(-0.002, 0.002)
            idx += 1
            x += 2 * r_val

        vars[N+idx] = np.random.uniform(0.2, 0.8)

        idx += 1
    return vars

def force_init(seed):

        y = vars[N:2*N]

                dy = y[j] - y[i]

                    fx[j] += f * dx; fy[j] += f * dy

            fy[i] += 0.02 * (1.0 if y[i] < 0.05 else (-1.0 if y[i] > 0.95 else 0.0))

        y += fy * 0.05
        vars[:N] = np.clip(x, r, 1.0 - r)

    return vars
