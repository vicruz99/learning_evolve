from scipy.optimize import minimize

N = 26

def unpack(vars):
    """Converts flat optimization vector to centers and radii arrays."""
    centers = np.zeros((N, 2))
    radii = np.zeros(N)
    for i in range(N):
        centers[i, 0] = vars[3 * i]
        centers[i, 1] = vars[3 * i + 1]
        radii[i] = vars[3 * i + 2]
    return centers, radii

def objective(vars):
    """Objective to minimize: negative sum of radii."""
    _, radii = unpack(vars)
    return -np.sum(radii)

def constraint_func(vars):
    """Returns array of constraint values g(x) >= 0."""
    centers, radii = unpack(vars)
    x = centers[:, 0]
    y = centers[:, 1]
    r = radii
    
    cons = []
    
    # Boundary constraints: circle inside [0,1]x[0,1]
    cons.append(x - r)
    cons.append(1.0 - x - r)
    cons.append(y - r)
    cons.append(1.0 - y - r)
    
    # Non-overlap constraints: dist(i, j) >= r_i + r_j
    for i in range(N):
        for j in range(i + 1, N):
            dx = x[i] - x[j]
            dy = y[i] - y[j]
            dist = np.hypot(dx, dy)
            cons.append(dist - r[i] - r[j])
            
    return np.concatenate(cons)

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-6, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)])
    return b

def force_directed_init(n, seed, steps=1200):
    """Generate initial configuration using repulsive force simulation."""
    np.random.seed(seed)
    centers = np.random.uniform(0.15, 0.85, (n, 2))
    radii = np.full(n, 0.08)
    
    for _ in range(steps):
        forces = np.zeros((n, 2))
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[j, 0] - centers[i, 0]
                dy = centers[j, 1] - centers[i, 1]
                dist = np.hypot(dx, dy)
                if dist < 0.25 and dist > 1e-6:
                    f = 0.003 / (dist ** 2)
                    fx, fy = f * dx, f * dy
                    forces[i] -= [fx, fy]
                    forces[j] += [fx, fy]
                    
            # Wall repulsion
            if centers[i, 0] < 0.05: forces[i, 0] += 0.05
            elif centers[i, 0] > 0.95: forces[i, 0] -= 0.05
            if centers[i, 1] < 0.05: forces[i, 1] += 0.05
            elif centers[i, 1] > 0.95: forces[i, 1] -= 0.05
            
        centers += forces * 0.02
        centers = np.clip(centers, 0.02, 0.98)
        
    # Estimate safe initial radii based on nearest neighbors/walls
    for i in range(n):
        min_d = min(centers[i, 0], 1.0 - centers[i, 0], 
                    centers[i, 1], 1.0 - centers[i, 1])
        for j in range(n):
            if i != j:
                d = np.hypot(centers[i, 0] - centers[j, 0], 
                             centers[i, 1] - centers[j, 1])
                min_d = min(min_d, d)
        radii[i] = min_d / 2.0 * 0.85
    return centers, radii

def create_hex_init(rows_counts, r_init=0.09):
    """Generates a hexagonal lattice initialization."""
    centers = np.zeros((N, 2))
    radii = np.full(N, r_init)
    idx = 0
    y_step = np.sqrt(3) * r_init
    y = r_init
    for i, count in enumerate(rows_counts):
        x_start = r_init + (r_init if i % 2 == 1 else 0)
        for k in range(count):
            if idx < N:
                x = x_start + k * 2 * r_init
                centers[idx] = [x, y]
                idx += 1
        y += y_step
    # Fill remaining if needed
    while idx < N:
        centers[idx] = np.random.uniform(r_init + 0.01, 1.0 - r_init - 0.01, 2)
        idx += 1
    return centers, radii
