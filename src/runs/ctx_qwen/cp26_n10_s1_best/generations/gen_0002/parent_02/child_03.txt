import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def compute_objective(vars):
    """Objective to minimize: negative sum of radii."""
    return -np.sum(vars[2::3])

def compute_constraints(vars):
    """Vectorized inequality constraints g(vars) >= 0."""
    n = N_CIRCLES
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    # Boundary constraints: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    b = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dr = r[:, None] + r[None, :]
    
    triu = np.triu_indices(n, k=1)
    o = dx[triu]**2 + dy[triu]**2 - dr[triu]**2
    
    return np.concatenate([b, o])

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [0, 0.5]."""
    b = []
    for _ in range(N_CIRCLES):
        b.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
    return b

def create_hex_init(rows_counts, r_start=0.085, seed=0):
    """Generates a hexagonal lattice initialization."""
    np.random.seed(seed)
    centers = np.zeros((N_CIRCLES, 2))
    radii = np.full(N_CIRCLES, r_start)
    idx = 0
    y = r_start
    
    for i, count in enumerate(rows_counts):
        x_start = r_start + (r_start if i % 2 == 1 else 0.0)
        for k in range(count):
            if idx < N_CIRCLES:
                cx = x_start + k * 2.0 * r_start
                cy = y
                centers[idx] = [cx + np.random.uniform(-0.005, 0.005),
                                cy + np.random.uniform(-0.005, 0.005)]
                radii[idx] = r_start + np.random.uniform(-0.005, 0.005)
                idx += 1
        y += np.sqrt(3.0) * r_start
        
    while idx < N_CIRCLES:
        centers[idx] = np.random.uniform(0.15, 0.85, 2)
        radii[idx] = r_start * 0.8
        idx += 1
        
    return centers, radii

def ensure_feasibility(centers, radii):
    """Project and relax to guarantee initial feasibility."""
    n = len(radii)
    for _ in range(100):
        # Wall projection
        for i in range(n):
            r = radii[i]
            centers[i, 0] = np.clip(centers[i, 0], r, 1.0 - r)
            centers[i, 1] = np.clip(centers[i, 1], r, 1.0 - r)
            
        max_viol = 0.0
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                min_d = radii[i] + radii[j]
                if dist < min_d and dist > 1e-9:
                    overlap = min_d - dist
                    max_viol = max(max_viol, overlap)
                    shift = overlap * 0.5
                    nx, ny = dx / dist, dy / dist
                    centers[i, 0] -= shift * nx
                    centers[i, 1] -= shift * ny
                    centers[j, 0] += shift * nx
                    centers[j, 1] += shift * ny
                    
        if max_viol < 1e-6:
            break
    return centers, radii

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    n = N_CIRCLES
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': compute_constraints}
    
    best_sum = -1.0
    best_vars = None
    
    patterns = [
        [5, 6, 5, 6, 4],
        [6, 5, 6, 5, 4],
        [5, 6, 6, 5, 4],
        [4, 6, 6, 6, 4],
        [5, 5, 6, 5, 5],
        [6, 6, 5, 5, 4]
    ]
    
    # Phase 1: Structured hexagonal starts
    for pat in patterns:
        for seed in range(5):
            c, r = create_hex_init(pat, r_start=0.085, seed=seed)
            c, r = ensure_feasibility(c, r)
            
            x0 = np.zeros(3 * n)
            x0[0::3] = c[:, 0]
            x0[1::3] = c[:, 1]
            x0[2::3] = r
            
            try:
                res = minimize(compute_objective, x0, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 15000, 'ftol': 1e-13, 'disp': False})
                if not np.isnan(res.fun):
                    curr_sum = -res.fun
                    if curr_sum > best_sum:
                        best_sum = curr_sum
                        best_vars = res.x.copy()
            except Exception:
                pass
                
    # Phase 2: Random starts
    np.random.seed(42)
    for _ in range(15):
        c = np.random.uniform(0.15, 0.85, (n, 2))
        r = np.full(n, 0.06)
        c, r = ensure_feasibility(c, r)
        
        x0 = np.zeros(3 * n)
        x0[0::3] = c[:, 0]
        x0[1::3] = c[:, 1]
        x0[2::3] = r
        
        try:

                           constraints=cons, options={'maxiter': 15000, 'ftol': 1e-13, 'disp': False})
            if not np.isnan(res.fun):

                if curr_sum > best_sum:
                    best_sum = curr_sum
                    best_vars = res.x.copy()
        except Exception:
            pass
            
    # Phase 3: Refinement & Escape local minima
    if best_vars is not None:
        for _ in range(6):
            x0 = best_vars + np.random.normal(0, 1e-4, 3 * n)
            for i in range(n):
                r_val = max(0.0, x0[3*i+2])
                x0[3*i] = np.clip(x0[3*i], r_val, 1.0 - r_val)

                x0[3*i+2] = r_val
                
            try:
                res = minimize(compute_objective, x0, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 20000, 'ftol': 1e-14, 'disp': False})
                if not np.isnan(res.fun):
                    curr_sum = -res.fun
                    if curr_sum > best_sum:
                        best_sum = curr_sum
                        best_vars = res.x.copy()
            except Exception:
                break

    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        centers[i, 0] = best_vars[3*i]
        centers[i, 1] = best_vars[3*i+1]
        radii[i] = max(0.0, best_vars[3*i+2])

           centers[i, 1] < radii[i] - 1e-12 or centers[i, 1] > 1.0 - radii[i] + 1e-12:
            valid = False
            break
    if valid:
        for i in range(n):
            for j in range(i + 1, n):
                d = np.sqrt((centers[i, 0]-centers[j, 0])**2 + (centers[i, 1]-centers[j, 1])**2)
                if d < radii[i] + radii[j] - 1e-12:
                    valid = False
                    break
            if not valid:
                break
                
    if not valid:
        factor = 0.999
        for _ in range(100):
            radii *= factor
            valid = True
            for i in range(n):
                if centers[i, 0] < radii[i] - 1e-12 or centers[i, 0] > 1.0 - radii[i] + 1e-12 or \
                   centers[i, 1] < radii[i] - 1e-12 or centers[i, 1] > 1.0 - radii[i] + 1e-12:
                    valid = False
                    break
            if valid:
                for i in range(n):
                    for j in range(i + 1, n):
                        d = np.sqrt((centers[i, 0]-centers[j, 0])**2 + (centers[i, 1]-centers[j, 1])**2)
                        if d < radii[i] + radii[j] - 1e-12:
                            valid = False
                            break
                    if not valid:
                        break
            if valid:
                break
                
    return centers, radii, float(np.sum(radii))
