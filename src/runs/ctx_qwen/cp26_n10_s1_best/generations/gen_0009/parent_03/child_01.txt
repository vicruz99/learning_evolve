import numpy as np
from scipy.optimize import minimize, linprog
import warnings
warnings.filterwarnings('ignore')

N = 26

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Returns all inequality constraints g(x) >= 0 using hypot for stability."""
    xs, ys, rs = x[0::3], x[1::3], x[2::3]
    c = np.concatenate([xs - rs, 1.0 - xs - rs, ys - rs, 1.0 - ys - rs])
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    i, j = np.tril_indices(N, -1)
    c = np.concatenate([c, np.hypot(dx[i, j], dy[i, j]) - dr[i, j]])
    return c

def constraints_squared(x):
    """Squared distance constraints for some optimization runs."""
    xs, ys, rs = x[0::3], x[1::3], x[2::3]
    c = np.concatenate([xs - rs, 1.0 - xs - rs, ys - rs, 1.0 - ys - rs])
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    i, j = np.tril_indices(N, -1)
    c = np.concatenate([c, dx[i, j]**2 + dy[i, j]**2 - dr[i, j]**2])
    return c

def get_bounds():
    """Variable bounds."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-8, 0.5)])
    return b

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist)
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-9)
    except Exception:
        pass
    return np.full(n, 0.04)

def project_to_feasible(x):
    """Project variables to strictly satisfy bounds."""
    x = x.copy()
    for i in range(N):
        r = max(1e-7, x[3 * i + 2])
        x[3 * i] = np.clip(x[3 * i], r, 1.0 - r)
        x[3 * i + 1] = np.clip(x[3 * i + 1], r, 1.0 - r)
        x[3 * i + 2] = r
    return x

def make_hex_init(r0, angle=0.0, offset_x=0.0, offset_y=0.0):
    """Generates a hexagonal lattice initialization."""
    pts = []
    y = r0 + offset_y
    row = 0
    while len(pts) < N + 15:
        x_start = r0 + offset_x if row % 2 == 0 else 2.0 * r0 + offset_x
        x = x_start
        while x <= 1.0 - r0 and len(pts) < N + 15:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        if y > 1.0 + r0:
            break
    
    pts = np.array(pts[:N + 15])
    
    if angle != 0.0:
        cx, cy = 0.5, 0.5
        c, s = np.cos(angle), np.sin(angle)
        dx = pts[:, 0] - cx
        dy = pts[:, 1] - cy
        pts[:, 0] = dx * c - dy * s + cx
        pts[:, 1] = dx * s + dy * c + cy
    
    mask = (pts[:, 0] >= 0.01) & (pts[:, 0] <= 0.99) & (pts[:, 1] >= 0.01) & (pts[:, 1] <= 0.99)
    pts = pts[mask]
    
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.15, 0.85, (1, 2))])
    return pts[:N]

def make_staggered_init(r0, row_counts, angle=0.0):
    """Creates a staggered pattern with specified row counts."""
    pts = []
    n_rows = len(row_counts)
    y_spacing = 0.95 / n_rows
    y = y_spacing / 2
    
    for row_idx, count in enumerate(row_counts):
        if len(pts) >= N:
            break
        x_spacing = 0.95 / (count + 1)
        x = x_spacing
        offset = y_spacing * 0.5 if row_idx % 2 == 1 else 0
        for _ in range(count):
            if len(pts) >= N:
                break
            pts.append([x + offset, y])
            x += x_spacing
        y += y_spacing
    
    while len(pts) < N:
        pts.append(np.random.uniform(0.1, 0.9, 2).tolist())
    
    pts = np.array(pts[:N])
    
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
    
    return np.clip(pts, 0.02, 0.98)

def make_force_init(seed, steps=500):
    """Force-directed layout to spread points."""
    np.random.seed(seed)
    pts = np.random.uniform(0.12, 0.88, (N, 2))
    
    for step in range(steps):
        f = np.zeros_like(pts)
        lr = 0.03 * max(0.01, 1.0 - step / steps)
        
        for i in range(N):
            for j in range(i + 1, N):
                dx = pts[j] - pts[i]
                d = np.hypot(dx[0], dx[1])
                if d < 0.30 and d > 1e-5:
                    rep = 0.015 / (d**2 + 0.0005)
                    f[i] -= dx * rep / d
                    f[j] += dx * rep / d
            
            for dim in range(2):
                if pts[i, dim] < 0.10:
                    f[i, dim] += 0.06
                elif pts[i, dim] > 0.90:
                    f[i, dim] -= 0.06
        
        pts += f * lr
        pts = np.clip(pts, 0.02, 0.98)
    
    return pts

def make_corner_heavy_init():
    """Places circles strategically in corners and edges."""
    pts = np.zeros((N, 2))
    
    # Four corners
    r_c = 0.10
    pts[0] = [r_c, r_c]
    pts[1] = [1-r_c, r_c]
    pts[2] = [r_c, 1-r_c]
    pts[3] = [1-r_c, 1-r_c]
    
    # Mid edges
    pts[4] = [0.5, r_c * 0.7]
    pts[5] = [0.5, 1 - r_c * 0.7]
    pts[6] = [r_c * 0.7, 0.5]
    pts[7] = [1 - r_c * 0.7, 0.5]
    
    # Center
    pts[8] = [0.5, 0.5]
    
    # Fill remaining with hex-like pattern
    idx = 9
    for r in range(3):
        for c in range(4):
            if idx >= N:
                break
            x = 0.2 + c * 0.2 + (r % 2) * 0.1
            y = 0.2 + r * 0.2
            pts[idx] = [x, y]
            idx += 1
    
    while idx < N:
        pts[idx] = np.random.uniform(0.15, 0.85, 2)
        idx += 1
    
    pts += np.random.uniform(-0.005, 0.005, pts.shape)
    return np.clip(pts, 0.03, 0.97)

def run_slsqp_opt(x0, use_hypot=True):
    """Run SLSQP optimization with error handling."""
    bounds = get_bounds()
    cons_fn = constraints if use_hypot else constraints_squared
    cons = {'type': 'ineq', 'fun': cons_fn}
    
    try:
        res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 20000, 'ftol': 1e-14, 'disp': False})
        if res.success or not np.isnan(res.fun):
            return res.x, not np.isnan(res.fun)
    except Exception:
        pass
    return x0, False

def refine_with_lp(x0):
    """Refine radii using LP given centers."""
    centers = np.column_stack((x0[0::3], x0[1::3]))
    r_lp = solve_lp_radii(centers)
    x_new = x0.copy()
    x_new[2::3] = r_lp * 0.999
    return project_to_feasible(x_new), r_lp

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
