This transforms the problem into optimizing a single objective function $F(C) = \sum r_i^*(C)$ over center coordinates $C$.
2. **Multi-Start Hill Climbing**: I will generate diverse initial configurations (rotated/scaled hexagonal lattices and force-directed layouts). For each, I will run a greedy hill-climbing algorithm that perturbs centers, evaluates the new configuration via LP, and accepts improvements. 3. **SLSQP Polishing**: Once the best center configuration is found, I will run a high-precision SLSQP optimization on the full `[x, y, r]` variable vector to squeeze out remaining microscopic gains, using the LP-found configuration as a strictly feasible starting point.
4. **Strict Validation & Repair**: A final loop ensures all constraints strictly satisfy the validator's `1e-12` tolerance by minimally shrinking radii and re-clipping centers if any numerical drift occurs.

import numpy as np
from scipy.optimize import linprog, minimize
import warnings
warnings.filterwarnings('ignore')

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        x, y = centers[i]
        lim = min(x, 1.0 - x, y, 1.0 - y)
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist)
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-9)
    except Exception:
        pass
    return np.full(n, 0.01)

def make_hex_init(scale, angle, seed):
    """Generates a rotated and scaled hexagonal lattice initialization."""
    np.random.seed(seed)
    pts = []
    r0 = 0.09 * scale
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0 and len(pts) < N + 10:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
    pts = np.array(pts[:N])
    
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
        
    # Break exact symmetries
    pts += np.random.uniform(-0.002, 0.002, pts.shape)
    return np.clip(pts, 0.01, 0.99)

def make_force_init(seed):
    """Force-directed layout to spread points evenly and push to boundaries."""
    np.random.seed(seed + 5000)
    pts = np.random.uniform(0.1, 0.9, (N, 2))
    for _ in range(250):
        f = np.zeros_like(pts)
        for i in range(N):
            for j in range(i + 1, N):
                dx = pts[j] - pts[i]
                d = np.hypot(dx[0], dx[1])
                if d < 0.25 and d > 1e-5:
                    rep = 0.01 / (d**2 + 0.001)
                    f[i] -= dx * rep / d
                    f[j] += dx * rep / d
            for dim in range(2):
                if pts[i, dim] < 0.15: f[i, dim] += 0.05
                elif pts[i, dim] > 0.85: f[i, dim] -= 0.05
        pts += f * 0.05
        pts = np.clip(pts, 0.02, 0.98)
    return pts

def hill_climb(init_centers, steps=350):
    """Greedy hill climbing on centers with LP radius evaluation."""
    centers = init_centers.copy()
    radii = solve_lp_radii(centers)
    best_sum = np.sum(radii)
    best_c = centers.copy()
    best_r = radii.copy()
    
    step = 0.018
    for it in range(steps):
        pert = np.random.normal(0, step, centers.shape)
        new_c = np.clip(centers + pert, 0.005, 0.995)
        new_r = solve_lp_radii(new_c)
        new_sum = np.sum(new_r)
        
        if new_sum > best_sum:
            centers = new_c
            radii = new_r
            best_sum = new_sum
            best_c = centers.copy()
            best_r = radii.copy()
            step *= 1.005
            step = min(step, 0.04)
        else:
            step *= 0.997
            
        # Prevent step size from decaying too much too quickly
        if it % 100 == 99:
            step = max(step, 0.008)
            
    return best_c, best_r, best_sum

def polish_slsqlp(centers, radii):
    """High-precision SLSQP polish on the best configuration found."""
    n = N
    x0 = np.zeros(3 * n)
    x0[0::3] = centers[:, 0]
    x0[1::3] = centers[:, 1]
    x0[2::3] = radii * 0.9995  # Slight shrink for strict initial feasibility
    
    bounds = [(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)] * n
    
    def obj(x): 
        return -np.sum(x[2::3])
        
    def cons(x):
        cx, cy, cr = x[0::3], x[1::3], x[2::3]
        c = np.concatenate([cx - cr, 1.0 - cx - cr, cy - cr, 1.0 - cy - cr])
        dx = cx[:, None] - cx[None, :]
        dy = cy[:, None] - cy[None, :]
        dr = cr[:, None] + cr[None, :]
        c = np.concatenate([c, dx[TRIL_IDX]**2 + dy[TRIL_IDX]**2 - dr[TRIL_IDX]**2])
        return c
        
    res = minimize(obj, x0, method='SLSQP', bounds=bounds, constraints={'type': 'ineq', 'fun': cons},
                   options={'maxiter': 25000, 'ftol': 1e-14, 'disp': False})
                   
    if res.success and not np.isnan(res.fun):
        c_opt = np.column_stack((res.x[0::3], res.x[1::3]))
        r_opt = solve_lp_radii(c_opt)
        return c_opt, r_opt, np.sum(r_opt)
    return centers, radii, np.sum(radii)

def run_packing():
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    best_sum = 0.0
    best_c = None
    best_r = None
    
    # Phase 1: Generate diverse initial configurations
    inits = []
    for scale in np.linspace(0.9, 1.25, 5):
        for ang in np.linspace(-0.4, 0.4, 7):
            inits.append(make_hex_init(scale, ang, len(inits)))
    for s in range(12):
        inits.append(make_force_init(s))
        
    # Phase 2: Multi-start Hill Climbing
    for pts in inits:
        c, r, s = hill_climb(pts, steps=350)
        if s > best_sum:
            best_sum = s
            best_c = c.copy()
            best_r = r.copy()
            
    # Phase 3: SLSQP Polish
    best_c, best_r, best_sum = polish_slsqlp(best_c, best_r)
    
    # Phase 4: Final strict validation & numerical repair
    for _ in range(100):
        valid = True
        for i in range(N):
            x, y, r = best_c[i, 0], best_c[i, 1], best_r[i]
            if r < 0: valid = False; break
            if x - r < -1e-12 or x + r > 1.0 + 1e-12 or y - r < -1e-12 or y + r > 1.0 + 1e-12:
                valid = False; break
        if valid:
            for i in range(N):
                for j in range(i + 1, N):
                    d = np.hypot(best_c[i, 0] - best_c[j, 0], best_c[i, 1] - best_c[j, 1])
                    if d < best_r[i] + best_r[j] - 1e-12:
                        valid = False; break
                if not valid: break
        if valid: break
            
        # Minimal shrinkage to guarantee strict compliance
        best_r *= 0.9998
        best_c[:, 0] = np.clip(best_c[:, 0], best_r, 1.0 - best_r)
        best_c[:, 1] = np.clip(best_c[:, 1], best_r, 1.0 - best_r)
        
    return best_c, best_r, float(best_sum)
