import numpy as np
from scipy.optimize import minimize, linprog
import warnings
warnings.filterwarnings('ignore')

N = 26

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_ineq = n * 4 + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        x, y = centers[i]
        lim = min(x, 1.0 - x, y, 1.0 - y)
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    # Vectorized distance computation for speed
    dx = centers[:, 0, None] - centers[None, :, 0]
    dy = centers[:, 1, None] - centers[None, :, 1]
    dist_mat = np.hypot(dx, dy)
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist_mat[i, j])
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-9)
    except Exception:
        pass
    return np.full(n, 0.05)

def get_hex_init(scale, angle):
    """Generates a scaled and rotated hexagonal lattice."""
    pts = []
    r0 = 0.09 * scale
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0 and len(pts) < N + 10:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        if y > 1.0 + r0:
            break
            
    pts = np.array(pts[:N + 10])
    
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        center = np.array([0.5, 0.5])
        pts = (pts - center) @ np.array([[c, -s], [s, c]]) + center
        
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    if len(pts) < N:
        pad = N - len(pts)
        pts = np.vstack([pts, np.random.uniform(0.15, 0.85, (pad, 2))])
    return np.clip(pts[:N], 0.01, 0.99)

def get_force_init(seed):
    """Force-directed layout pushing to boundaries."""
    np.random.seed(seed)
    pts = np.random.uniform(0.1, 0.9, (N, 2))
    for step in range(600):
        f = np.zeros_like(pts)
        lr = 0.05 * (1.0 - step / 600.0)
        for i in range(N):
            for j in range(i + 1, N):
                dx = pts[j] - pts[i]
                d = np.hypot(dx[0], dx[1])
                if d < 0.25 and d > 1e-5:
                    rep = 0.015 / (d**2 + 0.001)
                    f[i] -= dx * rep / d
                    f[j] += dx * rep / d
            for dim in range(2):
                if pts[i, dim] < 0.12:
                    f[i, dim] += 0.08
                elif pts[i, dim] > 0.88:
                    f[i, dim] -= 0.08
        pts += f * lr
        pts = np.clip(pts, 0.02, 0.98)
    return pts

def optimize_centers_sa(C):
    """Simulated annealing + coordinate moves on centers."""
    best_C = C.copy()
    best_r = solve_lp_radii(C)
    best_sum = np.sum(best_r)
    
    temp = 0.015
    step_global = 0.02
    decay = 0.996
    
    for it in range(900):
        # 70% coordinate move, 30% global move
        if np.random.rand() < 0.7:
            i = np.random.randint(N)
            old_pos = best_C[i].copy()
            best_C[i] = np.random.uniform(0.02, 0.98, 2)
        else:
            pert = np.random.normal(0, step_global, best_C.shape)
            best_C = np.clip(best_C + pert, 0.01, 0.99)
            
        new_r = solve_lp_radii(best_C)
        new_sum = np.sum(new_r)
        
        delta = new_sum - best_sum
        if delta > 0 or np.random.rand() < np.exp(delta / temp):
            best_sum = new_sum
            best_r = new_r
        else:
            # Revert coordinate move if it was the cause
            if np.random.rand() < 0.7:
                best_C[i] = old_pos
            else:
                best_C = np.clip(best_C - pert, 0.01, 0.99)
                
        temp *= decay
        step_global *= decay
        
    return best_C, best_sum

def objective_joint(x):
    return -np.sum(x[2::3])

def constraints_joint(x):
    xs, ys, rs = x[0::3], x[1::3], x[2::3]
    c = np.concatenate([xs - rs, 1.0 - xs - rs, ys - rs, 1.0 - ys - rs])
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    triu = np.triu_indices(N, 1)
    c = np.concatenate([c, dx[triu]**2 + dy[triu]**2 - dr[triu]**2])
    return c

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    bounds_joint = [(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)] * N
    cons_joint = {'type': 'ineq', 'fun': constraints_joint}
    
    best_C = None
    best_r = None
    best_sum = -1.0
    
    # Phase 1: Diverse Initializations + SA Optimization
    inits = []
    for scale in [0.95, 1.0, 1.05]:
        for ang in np.linspace(-0.3, 0.3, 5):
            inits.append(get_hex_init(scale, ang))
    for s in range(8):
        inits.append(get_force_init(s))
        
    for C_init in inits:
        C_opt, s_opt = optimize_centers_sa(C_init)
        r_opt = solve_lp_radii(C_opt)
        if s_opt > best_sum:
            best_sum = s_opt
            best_C = C_opt.copy()
            best_r = r_opt.copy()
            
    # Phase 2: Joint SLSQP Polish
    if best_C is not None:
        x0 = np.zeros(3 * N)
        x0[0::3] = best_C[:, 0]
        x0[1::3] = best_C[:, 1]
        x0[2::3] = best_r * 0.995
        
        # Strict projection for solver stability
        for i in range(N):
            r = x0[3 * i + 2]
            x0[3 * i] = np.clip(x0[3 * i], r, 1.0 - r)
            x0[3 * i + 1] = np.clip(x0[3 * i + 1], r, 1.0 - r)
            
        try:
            res = minimize(objective_joint, x0, method='SLSQP', bounds=bounds_joint,
                           constraints=cons_joint, options={'maxiter': 20000, 'ftol': 1e-14, 'disp': False})
            if not np.isnan(res.fun):
                c_tmp = np.column_stack((res.x[0::3], res.x[1::3]))
                r_tmp = solve_lp_radii(c_tmp)
                if np.sum(r_tmp) > best_sum:
                    best_C = c_tmp
                    best_r = r_tmp
                    best_sum = np.sum(r_tmp)
        except Exception:
            pass
            
    # Fallback
    if best_C is None:
        best_C = get_hex_init(1.0, 0.0)
        best_r = solve_lp_radii(best_C)
        best_sum = np.sum(best_r)
        
    centers = best_C
    radii = best_r
    
    # Final strict validation repair against 1e-12 tolerance
    for _ in range(100):
        valid = True
        for i in range(N):
            if radii[i] < 0 or centers[i, 0] < radii[i] - 1e-9 or centers[i, 0] > 1.0 - radii[i] + 1e-9 or \
               centers[i, 1] < radii[i] - 1e-9 or centers[i, 1] > 1.0 - radii[i] + 1e-9:
                valid = False
                break
        if valid:
            for i in range(N):
                for j in range(i + 1, N):
                    d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                    if d < radii[i] + radii[j] - 1e-9:
                        valid = False
                        break
                if not valid:
                    break
        if valid:
            break
            
        # Minimal shrinkage to guarantee strict feasibility
        radii *= 0.9998
        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)
        
    return centers, radii, float(np.sum(radii))
