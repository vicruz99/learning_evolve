1. **Decouple Centers and Radii**: For any fixed set of centers, the optimal radii can be found exactly and rapidly using Linear Programming (`linprog`). This transforms the problem into optimizing center coordinates to maximize the sum of LP-computed radii.
2. **Derivative-Free Center Optimization**: The LP value function is non-differentiable at contact transitions. We use Powell's method (derivative-free) on the center coordinates, which handles non-smooth objectives gracefully and respects bounds.
3. **Multi-Phase Refinement**:
   - **Phase 1**: Diverse initial configurations (rotated hexagonal lattices, force-directed spreads) followed by LP snap -> Powell center optimization -> LP snap -> SLSQP joint refinement.
   - **Phase 2**: Deflation & Perturbation loop. Systematically shrink a random subset of radii, perturb centers, and re-run Powell optimization. This breaks rigid bottlenecks and allows the packing to reconfigure into denser topologies.
4. **Strict Validation & Repair**: A final numerical repair loop ensures strict compliance with the `1e-12` tolerance in the validator, gently shrinking radii only if necessary to fix microscopic overlaps.

import numpy as np
from scipy.optimize import minimize, linprog
import warnings
warnings.filterwarnings('ignore')
np.seterr(all='ignore')

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    x, y = centers[:, 0], centers[:, 1]
    
    # Boundary limits: r_i <= min(x, 1-x, y, 1-y)
    lims = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
    lims = np.maximum(lims, 0.0)
    
    # Pairwise distances
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dists = np.sqrt(dx**2 + dy**2)
    
    # Build LP: max sum(r) s.t. r_i <= lims[i], r_i + r_j <= dists[i,j]
    m = n + n * (n - 1) // 2
    A_ub = np.zeros((m, n))
    b_ub = np.zeros(m)
    
    # Boundary constraints
    A_ub[:n, :] = np.eye(n)
    b_ub[:n] = lims
    
    # Overlap constraints
    idx = n
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-9)
    except Exception:
        pass
    # Fallback: safe radii
    return np.maximum(lims * 0.5, 1e-9)

def obj_centers(c_flat):
    """Objective for center-only optimization: maximize sum of LP radii."""
    centers = c_flat.reshape(N, 2)
    centers = np.clip(centers, 1e-6, 1.0 - 1e-6)
    radii = solve_lp_radii(centers)
    return -np.sum(radii)

def generate_init(seed, strategy='hex', scale=1.0, angle=0.0):
    """Generate initial center configurations."""
    np.random.seed(seed)
    pts = np.zeros((N, 2))
    if strategy == 'hex':
        r0 = 0.09 * scale
        pts_list = []
        y = r0
        row = 0
        while len(pts_list) < N + 10:
            x = r0 if row % 2 == 0 else 2.0 * r0
            while x <= 1.0 - r0 and len(pts_list) < N + 10:
                pts_list.append([x, y])
                x += 2.0 * r0
            y += np.sqrt(3.0) * r0
            row += 1
        pts = np.array(pts_list[:N])
        if angle != 0.0:
            c, s = np.cos(angle), np.sin(angle)
            pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
    elif strategy == 'force':
        pts = np.random.uniform(0.1, 0.9, (N, 2))
        for _ in range(300):
            f = np.zeros_like(pts)
            for i in range(N):
                for j in range(i + 1, N):
                    dx = pts[j] - pts[i]
                    d = np.hypot(dx[0], dx[1])
                    if d < 0.3 and d > 1e-5:
                        rep = 0.01 / (d**2 + 0.001)
                        f[i] -= dx * rep / d
                        f[j] += dx * rep / d
                if pts[i, 0] < 0.15: f[i, 0] += 0.05
                elif pts[i, 0] > 0.85: f[i, 0] -= 0.05
                if pts[i, 1] < 0.15: f[i, 1] += 0.05
                elif pts[i, 1] > 0.85: f[i, 1] -= 0.05
            pts += f * 0.05
            pts = np.clip(pts, 0.02, 0.98)
    return pts

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    inits = []
    # Hexagonal variants with varied scales and rotations
    for scale in [0.9, 0.95, 1.0, 1.05, 1.1]:
        for ang in np.linspace(-0.2, 0.2, 7):
            inits.append(generate_init(0, 'hex', scale, ang))
    # Force-directed layouts
    for s in range(15):
        inits.append(generate_init(s, 'force'))
        
    center_bounds = [(0.001, 0.999)] * (2 * N)
    full_bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N
    
    def obj_full(x):
        return -np.sum(x[2::3])
        
    def con_full(x):
        xs, ys, rs = x[0::3], x[1::3], x[2::3]
        c = np.concatenate([xs - rs, 1.0 - xs - rs, ys - rs, 1.0 - ys - rs])
        dx = xs[:, None] - xs[None, :]
        dy = ys[:, None] - ys[None, :]
        dr = rs[:, None] + rs[None, :]
        c = np.concatenate([c, dx[TRIL_IDX]**2 + dy[TRIL_IDX]**2 - dr[TRIL_IDX]**2])
        return c
        
    cons = {'type': 'ineq', 'fun': con_full}
    
    for init_pts in inits:
        r_lp = solve_lp_radii(init_pts)
        c_curr = init_pts.copy()
        
        # Powell on centers to maximize LP radii sum
        c_flat = c_curr.flatten()
        try:
            res_powell = minimize(obj_centers, c_flat, method='Powell', bounds=center_bounds,
                                  options={'maxiter': 1000, 'ftol': 1e-12, 'disp': False})
            c_curr = res_powell.x.reshape(N, 2)
        except Exception:
            pass
            
        r_curr = solve_lp_radii(c_curr)
        
        # SLSQP joint refinement
        x0 = np.zeros(3 * N)
        x0[0::3] = c_curr[:, 0]
        x0[1::3] = c_curr[:, 1]
        x0[2::3] = r_curr * 0.999
        
        try:
            res_sl = minimize(obj_full, x0, method='SLSQP', bounds=full_bounds, constraints=cons,
                              options={'maxiter': 8000, 'ftol': 1e-14, 'disp': False})
            if not np.isnan(res_sl.fun):
                c_opt = np.column_stack((res_sl.x[0::3], res_sl.x[1::3]))
                r_opt = solve_lp_radii(c_opt)
                s = np.sum(r_opt)
                if s > best_sum:
                    best_sum = s
                    best_centers = c_opt.copy()
                    best_radii = r_opt.copy()
        except Exception:
            pass
            
    # Deflation & Refinement to escape local minima
    if best_centers is not None:
        for cycle in range(80):
            c_pert = best_centers.copy()
            r_pert = best_radii.copy()
            
            # Deflate random subset to break rigid contacts
            subset = np.random.choice(N, size=max(3, N // 3), replace=False)
            r_pert[subset] *= 0.85
            
            # Perturb centers
            noise = 0.004 * (0.96 ** (cycle // 5))
            c_pert += np.random.normal(0, noise, c_pert.shape)
            c_pert = np.clip(c_pert, 0.01, 0.99)
            
            r_pert = solve_lp_radii(c_pert)
            c_flat = c_pert.flatten()
            
            try:
                res_powell = minimize(obj_centers, c_flat, method='Powell', bounds=center_bounds,
                                      options={'maxiter': 400, 'ftol': 1e-12, 'disp': False})
                c_curr = res_powell.x.reshape(N, 2)
                r_curr = solve_lp_radii(c_curr)
                
                if np.sum(r_curr) > best_sum:
                    best_sum = np.sum(r_curr)
                    best_centers = c_curr.copy()
                    best_radii = r_curr.copy()
            except Exception:
                pass

    # Fallback
    if best_centers is None:
        best_centers = generate_init(0, 'hex', 1.0, 0.0)
        best_radii = solve_lp_radii(best_centers)
        best_sum = np.sum(best_radii)
        
    centers = best_centers
    radii = best_radii
    
    # Final strict validation & minimal numerical repair
    for _ in range(100):
        valid = True
        for i in range(N):
            if radii[i] < 0 or centers[i, 0] < radii[i] - 1e-9 or centers[i, 0] > 1.0 - radii[i] + 1e-9 or \
               centers[i, 1] < radii[i] - 1e-9 or centers[i, 1] > 1.0 - radii[i] + 1e-9:
                valid = False
                break
        if valid:
            for i in range(N):
                for j in range(i + 1, N):
                    d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                    if d < radii[i] + radii[j] - 1e-9:
                        valid = False
                        break
                if not valid:
                    break
        if valid:
            break
        radii *= 0.9998
        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)
        
    return centers, radii, float(np.sum(radii))
