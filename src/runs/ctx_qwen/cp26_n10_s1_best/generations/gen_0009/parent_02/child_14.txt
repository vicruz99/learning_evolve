import numpy as np
from scipy.optimize import minimize, linprog
import warnings

warnings.filterwarnings('ignore')

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        x_val, y_val = centers[i]
        lim = min(x_val, 1.0 - x_val, y_val, 1.0 - y_val)
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist)
            idx += 1
            
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=[(0.0, None)] * n, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-9)
    except Exception:
        pass
    return np.full(n, 1e-9)

def objective_slscp(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints_slscp(x):
    """Vectorized inequality constraints >= 0."""
    cx, cy, r = x[0::3], x[1::3], x[2::3]
    c = np.concatenate([
        cx - r,
        1.0 - cx - r,
        cy - r,
        1.0 - cy - r
    ])
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dr = r[:, None] + r[None, :]
    c = np.concatenate([c, dx[TRIL_IDX]**2 + dy[TRIL_IDX]**2 - dr[TRIL_IDX]**2])
    return c

def get_slscp_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-9, 0.5]."""
    return [(0.0, 1.0), (0.0, 1.0), (1e-9, 0.5)] * N

def optimize_slsqp(centers, radii):
    """Run SLSQP to optimize centers and radii jointly."""
    x0 = np.zeros(3 * N)
    x0[0::3] = centers[:, 0]
    x0[1::3] = centers[:, 1]
    x0[2::3] = radii
    
    cons = {'type': 'ineq', 'fun': constraints_slscp}
    try:
        res = minimize(objective_slscp, x0, method='SLSQP', bounds=get_slscp_bounds(),
                       constraints=cons, options={'maxiter': 4000, 'ftol': 1e-14, 'disp': False})
        if not np.isnan(res.fun):
            c_opt = np.column_stack((res.x[0::3], res.x[1::3]))
            r_opt = res.x[2::3]
            # Project to strict bounds
            for i in range(N):
                r = max(1e-9, r_opt[i])
                c_opt[i, 0] = np.clip(c_opt[i, 0], r, 1.0 - r)
                c_opt[i, 1] = np.clip(c_opt[i, 1], r, 1.0 - r)
            return c_opt, r_opt
    except Exception:
        pass
    return centers, radii

def nelder_refine(centers):
    """Refine centers using Nelder-Mead with exact LP radii evaluation."""
    def nelder_obj(c_flat):
        # Clip to valid interior to ensure LP feasibility
        c = np.clip(c_flat.reshape(-1, 2), 1e-6, 1.0 - 1e-6)
        r = solve_lp_radii(c)
        return -np.sum(r)
    
    try:
        res = minimize(nelder_obj, centers.reshape(-1), method='Nelder-Mead',
                       options={'maxiter': 400, 'xatol': 1e-6, 'fatol': 1e-7, 'disp': False})
        c_res = res.x.reshape(-1, 2)
        c_res = np.clip(c_res, 1e-5, 1.0 - 1e-5)
        return c_res
    except Exception:
        return centers

def generate_inits():

    inits = []
    np.random.seed(42)
    
    # 1. Rotated Hexagonal Lattices
    for r0 in [0.080, 0.085, 0.090, 0.095]:

                    x += 2.0 * r0
                y += np.sqrt(3.0) * r0
                row += 1
            pts = np.array(pts[:N + 8])
