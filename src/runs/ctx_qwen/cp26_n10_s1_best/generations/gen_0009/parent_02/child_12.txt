2. **Boundary-focused initialization**: Create configurations that explicitly push circles into corners and along edges where they can grow larger.

import numpy as np
from scipy.optimize import minimize, linprog
import warnings
warnings.filterwarnings('ignore')

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Returns all inequality constraints >= 0 (vectorized)."""
    xs, ys, rs = x[0::3], x[1::3], x[2::3]
    
    c = np.concatenate([
        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs
    ])
    
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    c = np.concatenate([c, dx[TRIL_IDX]**2 + dy[TRIL_IDX]**2 - dr[TRIL_IDX]**2])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-7, 0.5]."""
    return [(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)] * N

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist)
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-7)
    except Exception:
        pass
    return np.full(n, 0.05)

def circle_growth_sim(seed, n_grow=3000):
    """Simulate circles growing from small size with repulsive forces."""
    np.random.seed(seed)
    centers = np.random.uniform(0.2, 0.8, (N, 2))
    radii = np.full(N, 0.005)
    
    for step in range(n_grow):
        lr = 0.03 * (1.0 - step / n_grow) ** 1.5
        forces = np.zeros_like(centers)
        
        for i in range(N):
            for j in range(i + 1, N):
                dx = centers[j] - centers[i]
                d = np.hypot(dx[0], dx[1])
                target = radii[i] + radii[j] + 0.002
                if d < target and d > 1e-8:
                    rep = min(0.5, (target - d) / d)
                    forces[i] -= dx * rep / d
                    forces[j] += dx * rep / d
            
            for dim in range(2):
                if centers[i, dim] < radii[i] + 0.005:
                    forces[i, dim] += 0.15
                elif centers[i, dim] > 1.0 - radii[i] - 0.005:
                    forces[i, dim] -= 0.15
                # Push toward boundaries
                if centers[i, dim] < 0.15:
                    forces[i, dim] += 0.03
                elif centers[i, dim] > 0.85:
                    forces[i, dim] -= 0.03
        
        centers += forces * lr
        centers = np.clip(centers, 0.002, 0.998)
        
        # Grow circles gradually
        radii += 0.00003 * (1.0 - step / n_grow)
    
    return centers

def make_corner_edge_init(seed, angle=0.0):
    """Create init with circles in corners, along edges, and interior."""
    np.random.seed(seed)
    centers = np.zeros((N, 2))
    
    # 4 corner circles
    r_corner = 0.11
    centers[0] = [r_corner, r_corner]
    centers[1] = [1 - r_corner, r_corner]
    centers[2] = [r_corner, 1 - r_corner]
    centers[3] = [1 - r_corner, 1 - r_corner]
    
    # 8 edge circles (2 per edge)
    r_edge = 0.095
    centers[4] = [0.25, r_edge]
    centers[5] = [0.75, r_edge]
    centers[6] = [0.25, 1 - r_edge]
    centers[7] = [0.75, 1 - r_edge]
    centers[8] = [r_edge, 0.25]
    centers[9] = [r_edge, 0.75]
    centers[10] = [1 - r_edge, 0.25]
    centers[11] = [1 - r_edge, 0.75]
    
    # 14 interior circles in hexagonal pattern
    r_int = 0.09
    y = r_int + 0.12
    row = 0
    idx = 12
    while idx < N and y <= 0.88:
        x_start = r_int + 0.08 if row % 2 == 0 else r_int + 0.17
        x = x_start
        while x <= 0.92 and idx < N:
            centers[idx] = [x, y]
            idx += 1
            x += 2 * r_int + 0.02
        y += np.sqrt(3) * r_int + 0.01
        row += 1
    
    # Fill remaining if needed
    while idx < N:
        centers[idx] = [np.random.uniform(0.2, 0.8), np.random.uniform(0.2, 0.8)]
        idx += 1
    
    # Rotate if needed
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        centers = (centers - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
    
    # Add small noise
    centers += np.random.uniform(-0.005, 0.005, centers.shape)
    centers = np.clip(centers, 0.02, 0.98)
    
    return centers

def make_hex_init(r0, angle=0.0):
    """Generates a hexagonal lattice initialization."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        if y > 1.0 + r0:
            break
    
    pts = np.array(pts[:N + 10])
    
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
    
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.15, 0.85, (1, 2))])
    return pts[:N]

def project_to_feasible(x):
    """Project variables to strictly satisfy bounds."""
    x = x.copy()
    for i in range(N):
        r = max(1e-6, x[3 * i + 2])
        x[3 * i] = np.clip(x[3 * i], r, 1.0 - r)
        x[3 * i + 1] = np.clip(x[3 * i + 1], r, 1.0 - r)
        x[3 * i + 2] = r
    return x

def optimize_with_lp_squeeze(centers, radii, maxiter=15000):
    """Run SLSQP followed by LP radii squeeze."""
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    
    x0 = np.zeros(3 * N)
    x0[0::3] = centers[:, 0]
    x0[1::3] = centers[:, 1]
    x0[2::3] = np.maximum(radii * 0.99, 0.005)
    x0 = project_to_feasible(x0)
    
    try:
        res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14, 'disp': False})
        if not np.isnan(res.fun):
            c_opt = np.column_stack((res.x[0::3], res.x[1::3]))
            r_lp = solve_lp_radii(c_opt)
            return c_opt, r_lp, np.sum(r_lp)
    except Exception:
        pass
    return None, None, -1.0
