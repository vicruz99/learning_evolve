import numpy as np
from scipy.optimize import minimize, linprog
import warnings

warnings.filterwarnings('ignore')

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Vectorized inequality constraints: g(x) >= 0."""
    cx, cy, cr = x[0::3], x[1::3], x[2::3]
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    c = np.concatenate([
        cx - cr,
        1.0 - cx - cr,
        cy - cr,
        1.0 - cy - cr
    ])
    
    # Overlap constraints: dist >= r_i + r_j
    # Using Euclidean distance provides better gradients near contact
    dx = cx[TRIL_IDX[0]] - cx[TRIL_IDX[1]]
    dy = cy[TRIL_IDX[0]] - cy[TRIL_IDX[1]]
    dist = np.hypot(dx, dy)
    r_sum = cr[TRIL_IDX[0]] + cr[TRIL_IDX[1]]
    
    c = np.concatenate([c, dist - r_sum])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-7, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)])
    return b

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist)
            idx += 1
            
    bounds_lp = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_lp, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-7)
    except Exception:
        pass
    return np.full(n, 0.05)

def make_hex_init(r0, angle=0.0):
    """Generates a hexagonal lattice initialization with optional rotation."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 8:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        if y > 1.0 + r0:
            break
            
    pts = np.array(pts[:N + 8])
    
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
        
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.1, 0.9, (1, 2))])
    return pts[:N]

def force_directed_init(seed, steps=400):
    """Force-directed layout to spread points evenly and push to boundaries."""
    np.random.seed(seed)
    pts = np.random.uniform(0.15, 0.85, (N, 2))
    for step in range(steps):
        forces = np.zeros_like(pts)
        for i in range(N):
            for j in range(i + 1, N):
                dx = pts[j] - pts[i]
                d = np.linalg.norm(dx)
                if d < 0.3 and d > 1e-5:
                    f = 0.015 / (d**2 + 0.001)
                    forces[i] -= f * dx
                    forces[j] += f * dx
            for dim in range(2):
                if pts[i, dim] < 0.12: forces[i, dim] += 0.05
                elif pts[i, dim] > 0.88: forces[i, dim] -= 0.05
        lr = 0.04 * (1.0 - step / steps)
        pts += forces * lr
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def project_to_feasible(x):
    """Project variables to strictly satisfy bounds."""
    x = x.copy()
    for i in range(N):
        r = max(1e-6, x[3 * i + 2])
        x[3 * i] = np.clip(x[3 * i], r, 1.0 - r)
        x[3 * i + 1] = np.clip(x[3 * i + 1], r, 1.0 - r)
        x[3 * i + 2] = r
    return x

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    np.random.seed(42)
    bounds_obj = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -1.0
    best_x = None

    inits = []

    for scale in [0.90, 0.95, 1.00, 1.05, 1.10]:

        for ang in np.linspace(-0.35, 0.35, 9):
            inits.append(make_hex_init(r0, ang))

    for s in range(12):
        inits.append(force_directed_init(s))
        
    # Phase 2: Multi-start Optimization with LP-SLSQP Alternation
    for c_init in inits:
        x0 = np.zeros(3 * N)
        x0[0::3] = c_init[:, 0]
        x0[1::3] = c_init[:, 1]
        
        # Initialize radii via LP
        r_lp = solve_lp_radii(c_init)
        x0[2::3] = np.maximum(r_lp * 0.995, 0.005)
        x0 = project_to_feasible(x0)
        
        # First SLSQP run
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds_obj, constraints=cons,
                           options={'maxiter': 10000, 'ftol': 1e-13, 'disp': False})
            if not np.isnan(res.fun):
                x0 = res.x.copy()
                
                # LP snap to theoretical max radii for current centers
                centers_tmp = np.column_stack((x0[0::3], x0[1::3]))
                r_new = solve_lp_radii(centers_tmp)
                if r_new is not None:
                    x0[2::3] = r_new
                    x0 = project_to_feasible(x0)
                    
                    # Second SLSQP run with improved radii
                    res2 = minimize(objective, x0, method='SLSQP', bounds=bounds_obj, constraints=cons,
                                    options={'maxiter': 8000, 'ftol': 1e-13, 'disp': False})

                        # Final LP snap

                        r_final = solve_lp_radii(centers_tmp2)
                        if r_final is not None:
                            x0[2::3] = r_final
                            x0 = project_to_feasible(x0)
                            
                        curr_sum = np.sum(x0[2::3])
                        if curr_sum > best_sum:
                            vals = constraints(x0)
                            if np.min(vals) >= -1e-5:
                                best_sum = curr_sum
                                best_x = x0.copy()
        except Exception:
            continue
            
    # Phase 3: Deflation & Perturbation to Escape Local Minima
    if best_x is not None:

            shrink = 0.92 + 0.06 * (cyc / 40.0)
            x0[2::3] *= shrink

            x0[subset * 3 + 2] *= 0.5
            
            x0 = project_to_feasible(x0)
            
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds_obj, constraints=cons,
                               options={'maxiter': 8000, 'ftol': 1e-13, 'disp': False})
                if not np.isnan(res.fun):
                    c_tmp = np.column_stack((res.x[0::3], res.x[1::3]))
                    r_new = solve_lp_radii(c_tmp)

                        x_p[2::3] = r_new * 0.999

                            best_x = x_p.copy()

                pass

    radii = best_x[2::3].copy()

    for _ in range(100):
        valid = True
        for i in range(N):
            if radii[i] < 0: valid = False; break

            if centers[i, 1] - radii[i] < -1e-9 or centers[i, 1] + radii[i] > 1.0 + 1e-9: valid = False; break

        radii *= 0.9998
        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)
        
    return centers, radii, float(np.sum(radii))
