import numpy as np
from scipy.optimize import minimize, linprog

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist)
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return np.full(n, 0.01)

def obj_centers(centers_flat):
    """Objective for center optimization: maximize sum of LP-optimal radii."""
    centers = centers_flat.reshape(N, 2)
    centers = np.clip(centers, 0.005, 0.995)
    r = solve_lp_radii(centers)
    return -np.sum(r)

def make_hex_init(r0, angle=0.0, row_counts=None):
    """Generates a hexagonal lattice initialization with optional rotation and custom row counts."""
    pts = []
    if row_counts is None:
        # Default hex filling
        y = r0
        row = 0
        while len(pts) < N + 10:
            x = r0 if row % 2 == 0 else 2.0 * r0
            while x <= 1.0 - r0 and len(pts) < N + 10:
                pts.append([x, y])
                x += 2.0 * r0
            y += np.sqrt(3.0) * r0
            row += 1
    else:
        # Custom row counts (staggered)
        y = r0
        row = 0
        for count in row_counts:
            x = r0 if row % 2 == 0 else 2.0 * r0
            for _ in range(count):
                if len(pts) >= N + 10: break
                pts.append([x, y])
                x += 2.0 * r0
            y += np.sqrt(3.0) * r0
            row += 1
            
    pts = np.array(pts[:N + 10])
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
        
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    while len(pts) < N:
        pts = np.vstack([pts, [np.random.uniform(0.2, 0.8), np.random.uniform(0.2, 0.8)]])
    return pts[:N]

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    
    best_sum = -1.0
    best_centers = None
    
    # Phase 1: Generate diverse initial configurations
    inits = []
    
    # Standard hex grids with varied densities and rotations
    for r0 in [0.075, 0.085, 0.095, 0.105]:
        for ang in np.linspace(-0.35, 0.35, 9):
            inits.append(make_hex_init(r0, ang))
            
    # Staggered row counts matching 26 circles topology
    for counts in [[6,5,6,5,4], [5,6,5,6,4], [5,5,5,5,6], [6,6,5,5,4], [4,6,6,6,4]]:
        for ang in np.linspace(-0.2, 0.2, 5):
            inits.append(make_hex_init(0.09, ang, counts))
            
    # Randomized starts
    for s in range(15):
        np.random.seed(s + 100)
        pts = np.random.uniform(0.15, 0.85, (N, 2))
        # Quick repulsion spread
        for _ in range(100):
            f = np.zeros_like(pts)
            for i in range(N):
                for j in range(i+1, N):
                    dx = pts[j] - pts[i]
                    d = np.hypot(dx[0], dx[1])
                    if d < 0.25 and d > 1e-5:
                        rep = 0.015 / (d**2 + 0.001)
                        f[i] -= dx * rep / d
                        f[j] += dx * rep / d
                for dim in range(2):
                    if pts[i, dim] < 0.15: f[i, dim] += 0.04
                    elif pts[i, dim] > 0.85: f[i, dim] -= 0.04
            pts += f * 0.04
            pts = np.clip(pts, 0.05, 0.95)
        inits.append(pts)
        
    # Phase 2: Optimize centers using Nelder-Mead (robust for non-smooth LP objective)
    for pts in inits:
        x0 = pts.flatten()
        try:
            res = minimize(obj_centers, x0, method='Nelder-Mead',
                           options={'maxiter': 2500, 'fatol': 1e-9, 'xatol': 1e-9})
            if -res.fun > best_sum:
                best_sum = -res.fun
                best_centers = res.x.reshape(N, 2)
        except Exception:
            pass

                    best_centers = res.x.reshape(N, 2)

            xs, ys, rs = x[0::3], x[1::3], x[2::3]

            dy = ys[:, None] - ys[None, :]
            dr = rs[:, None] + rs[None, :]
            c = np.concatenate([c, dx[TRIL_IDX]**2 + dy[TRIL_IDX]**2 - dr[TRIL_IDX]**2])
            return c
            
        def obj_full(x):
            return -np.sum(x[2::3])
            
        bounds_full = [(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)] * N
        cons_full = {'type': 'ineq', 'fun': constraints_full}
        
        try:
            res = minimize(obj_full, x0, method='SLSQP', bounds=bounds_full, constraints=cons_full,
                           options={'maxiter': 25000, 'ftol': 1e-14, 'disp': False})
            if not np.isnan(res.fun):
                best_centers = np.column_stack((res.x[0::3], res.x[1::3]))
                best_sum = -res.fun
        except Exception:
            pass
            
    # Final LP snap to guarantee radii are maximally packed for the final centers
    radii = solve_lp_radii(best_centers)
    
    # Strict validation and numerical repair
    centers = best_centers.copy()
    for _ in range(100):
        valid = True
        for i in range(N):
            if radii[i] < 0 or centers[i, 0] < radii[i] - 1e-10 or centers[i, 0] > 1.0 - radii[i] + 1e-10 or \
               centers[i, 1] < radii[i] - 1e-10 or centers[i, 1] > 1.0 - radii[i] + 1e-10:
                valid = False
                break
        if valid:
            for i in range(N):
                for j in range(i + 1, N):
                    d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                    if d < radii[i] + radii[j] - 1e-10:
                        valid = False
                        break
                if not valid:
                    break
        if valid:
            break
            
        radii *= 0.9997
        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)
        
    return centers, radii, float(np.sum(radii))
