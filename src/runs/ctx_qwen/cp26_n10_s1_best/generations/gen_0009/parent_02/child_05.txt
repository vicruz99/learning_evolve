import numpy as np
from scipy.optimize import minimize, linprog
import warnings

warnings.filterwarnings('ignore')

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Returns all inequality constraints >= 0 (vectorized, linear distance for better conditioning)."""
    xs, ys, rs = x[0::3], x[1::3], x[2::3]
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    c = np.concatenate([
        xs - rs, 1.0 - xs - rs,
        ys - rs, 1.0 - ys - rs
    ])
    
    # Overlap constraints: dist >= r_i + r_j
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    c = np.concatenate([c, np.hypot(dx[TRIL_IDX], dy[TRIL_IDX]) - dr[TRIL_IDX]])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-6, 0.5]."""
    return [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist)
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-7)
    except Exception:
        pass
    return np.full(n, 0.01)

def project(x):
    """Project variables to strictly satisfy bounds."""
    x = x.copy()
    for i in range(N):
        r = max(1e-6, x[3 * i + 2])
        x[3 * i] = np.clip(x[3 * i], r, 1.0 - r)
        x[3 * i + 1] = np.clip(x[3 * i + 1], r, 1.0 - r)
        x[3 * i + 2] = r
    return x

def make_hex_init(angle=0.0, scale=1.0):
    """Generates a hexagonal lattice initialization with optional rotation."""
    pts = []
    r0 = 0.095 * scale
    y = r0
    row = 0
    while len(pts) < N + 5:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0 and len(pts) < N + 5:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        
    pts = np.array(pts[:N])
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
    return pts

def make_corner_init():
    """Generates an initialization focused on corners and edges."""
    pts = np.zeros((N, 2))
    r = 0.09
    pts[0] = [r, r]
    pts[1] = [1-r, r]
    pts[2] = [r, 1-r]
    pts[3] = [1-r, 1-r]
    idx = 4
    y = 2*r
    row = 0
    while idx < N and y + r <= 1.0:
        x = 2*r if row % 2 == 0 else 3*r
        while x + r <= 1.0 and idx < N:
            pts[idx] = [x, y]
            idx += 1
            x += 2*r
        y += np.sqrt(3)*r
        row += 1
    for i in range(idx, N):
        pts[i] = np.random.uniform(0.2, 0.8, 2)
    return pts

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -1.0
    best_x = None
    
    # Phase 1: Diverse Initial Configurations
    inits = []
    for ang in np.linspace(-0.4, 0.4, 17):
        for sc in [0.9, 1.0, 1.1]:
            inits.append(make_hex_init(ang, sc))
    for _ in range(5):
        inits.append(make_corner_init())
        
    for pts in inits:
        x0 = np.zeros(3 * N)
        x0[0::3] = pts[:, 0]
        x0[1::3] = pts[:, 1]
        x0[2::3] = solve_lp_radii(pts) * 0.99
        x0 = project(x0)
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 25000, 'ftol': 1e-14, 'disp': False})
            if not np.isnan(res.fun):
                c_opt = res.x.reshape(N, 3)[:, :2]
                r_lp = solve_lp_radii(c_opt)
                if r_lp is not None and np.sum(r_lp) > best_sum:
                    best_sum = np.sum(r_lp)
                    best_x = np.zeros(3*N)
                    best_x[0::3] = c_opt[:, 0]
                    best_x[1::3] = c_opt[:, 1]
                    best_x[2::3] = r_lp
        except Exception:
            pass
            
    # Phase 2: Aggressive Perturbation & Refinement to Escape Local Minima
    if best_x is not None:
        for cyc in range(180):
            x0 = best_x.copy()
            
            # Cooling noise schedule
            noise = 0.006 * (0.94 ** cyc)
            
            # Select random subset to perturb (breaks rigid contact networks)
            n_pert = np.random.randint(3, 9)
            idx_pert = np.random.choice(N, n_pert, replace=False)
            
            # Shrink radii in subset to create slack
            x0[idx_pert * 3 + 2] *= 0.85
            
            # Move centers in subset
            x0[idx_pert * 3] += np.random.normal(0, noise, n_pert)
            x0[idx_pert * 3 + 1] += np.random.normal(0, noise, n_pert)
            
            # Periodic global rotation & full deflation to reset topology
            if cyc % 12 == 0:
                ang = np.random.uniform(-0.15, 0.15)
                c_rot, s_rot = np.cos(ang), np.sin(ang)
                centers_curr = x0.reshape(N, 3)[:, :2]
                centers_curr = (centers_curr - 0.5) @ np.array([[c_rot, -s_rot], [s_rot, c_rot]]) + 0.5
                x0[0::3] = centers_curr[:, 0]
                x0[1::3] = centers_curr[:, 1]
                x0[2::3] *= 0.80
                
            x0 = project(x0)
            
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,

                if not np.isnan(res.fun):
                    c_opt = res.x.reshape(N, 3)[:, :2]
                    r_lp = solve_lp_radii(c_opt)
                    if r_lp is not None:
                        curr_sum = np.sum(r_lp)
                        if curr_sum > best_sum:
                            best_sum = curr_sum
                            best_x = np.zeros(3*N)
                            best_x[0::3] = c_opt[:, 0]
                            best_x[1::3] = c_opt[:, 1]
                            best_x[2::3] = r_lp
            except Exception:
                pass

    # Phase 3: Uniform Expansion Squeeze
    # Try to slightly expand all radii and push centers apart to find slack
    centers = best_x.reshape(N, 3)[:, :2]
    radii = best_x[2::3].copy()
    
    for _ in range(8):
        radii *= 1.0015
        # Force-directed relaxation to separate overlapping circles
        for step in range(150):
            f = np.zeros_like(centers)
            lr = 0.04 * (1.0 - step / 150.0)
            for i in range(N):
                for j in range(i + 1, N):
                    dx = centers[j] - centers[i]
                    d = np.hypot(dx[0], dx[1])
                    target = radii[i] + radii[j] + 0.0005
                    if d < target and d > 1e-5:
                        rep = 0.06 / (d**2 + 0.0005)
                        f[i] -= dx * rep / d
                        f[j] += dx * rep / d
                        
                for dim in range(2):
                    if centers[i, dim] < radii[i] + 0.001: f[i, dim] += 0.1
                    elif centers[i, dim] > 1.0 - radii[i] - 0.001: f[i, dim] -= 0.1
                    
            centers += f * lr
            centers = np.clip(centers, radii, 1.0 - radii)
            
        new_radii = solve_lp_radii(centers)
        if new_radii is not None and np.sum(new_radii) > best_sum:
            best_sum = np.sum(new_radii)
            radii = new_radii
            best_x = np.zeros(3*N)
            best_x[0::3] = centers[:, 0]
            best_x[1::3] = centers[:, 1]
            best_x[2::3] = radii

        best_x[1::3] = np.repeat(np.linspace(0.1, 0.9, 6), 5)[:N]
        best_x[2::3] = 0.06
        best_sum = np.sum(best_x[2::3])
        
    centers = best_x.reshape(N, 3)[:, :2]
    radii = best_x[2::3].copy()
    
    # Final strict validation repair against 1e-12 tolerance
    for _ in range(100):
        valid = True
        for i in range(N):

                valid = False
                break
        if valid:
            for i in range(N):
                for j in range(i + 1, N):
                    if np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1]) < radii[i] + radii[j] - 1e-9:
                        valid = False
                        break
                if not valid:
                    break
        if valid:

        radii *= 0.9995
        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)
        
    return centers, radii, float(np.sum(radii))
