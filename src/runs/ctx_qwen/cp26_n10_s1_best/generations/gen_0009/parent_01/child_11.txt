import numpy as np
from scipy.optimize import minimize, linprog
import warnings
warnings.filterwarnings('ignore')

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist)
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-7)
    except Exception:
        pass
    return np.full(n, 0.05)

def obj_centers(centers_flat):
    """Objective for center optimization: minimize negative sum of LP-computed radii."""
    centers = np.clip(centers_flat.reshape(N, 2), 0.005, 0.995)
    rs = solve_lp_radii(centers)
    return -np.sum(rs)

def make_hex_init(r0, angle):
    """Generates a rotated hexagonal lattice initialization."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0 and len(pts) < N + 10:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
    pts = np.array(pts[:N + 10])
    
    center = np.array([0.5, 0.5])
    pts -= center
    c, s = np.cos(angle), np.sin(angle)
    pts = pts @ np.array([[c, -s], [s, c]])
    pts += center
    
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    if len(pts) < N:
        pad = N - len(pts)
        pts = np.vstack([pts, np.random.uniform(0.2, 0.8, (pad, 2))])
    pts += np.random.uniform(-0.002, 0.002, pts.shape)
    return pts[:N]

def force_spread(centers, steps=200):
    """Force-directed relaxation to spread points evenly and push to boundaries."""
    pts = centers.copy()
    for step in range(steps):
        f = np.zeros_like(pts)
        lr = 0.02 * (1.0 - step / steps)
        for i in range(N):
            for j in range(i + 1, N):
                dx = pts[j] - pts[i]
                d = np.hypot(dx[0], dx[1])
                if d < 0.35 and d > 1e-5:
                    rep = 0.008 / (d**2 + 0.0002)
                    f[i] -= dx * rep / d
                    f[j] += dx * rep / d
            for dim in range(2):
                if pts[i, dim] < 0.08: f[i, dim] += 0.06
                elif pts[i, dim] > 0.92: f[i, dim] -= 0.06
        pts += f * lr
        pts = np.clip(pts, 0.02, 0.98)
    return pts

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Returns all inequality constraints >= 0 (vectorized)."""
    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]
    
    c = np.concatenate([xs - rs, 1.0 - xs - rs, ys - rs, 1.0 - ys - rs])
    
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    c = np.concatenate([c, dx[TRIL_IDX]**2 + dy[TRIL_IDX]**2 - dr[TRIL_IDX]**2])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-7, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)])
    return b

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Phase 1: Generate Diverse Initial Configurations
    inits = []
    for r0 in [0.08, 0.085, 0.09, 0.095, 0.10, 0.105]:
        for ang in np.linspace(-0.4, 0.4, 11):
            inits.append(make_hex_init(r0, ang))
    for s in range(10):
        np.random.seed(s + 200)
        pts = np.random.uniform(0.15, 0.85, (N, 2))
        inits.append(force_spread(pts, steps=200))
        
    # Phase 2: Multi-start Optimization using LP Oracle on Centers
    for pts in inits:
        x0 = pts.flatten()
        res = minimize(obj_centers, x0, method='Powell', options={'maxiter': 600, 'ftol': 1e-8})
        c_opt = res.x.reshape(N, 2)
        rs = solve_lp_radii(c_opt)
        curr_sum = np.sum(rs)
        if curr_sum > best_sum:
            best_sum = curr_sum
            best_centers = c_opt.copy()
            best_radii = rs.copy()
            
    # Phase 3: SLSQP Refinement and Perturbation Search
    if best_centers is not None:
        x0 = np.zeros(3 * N)
        x0[0::3] = best_centers[:, 0]
        x0[1::3] = best_centers[:, 1]
        x0[2::3] = best_radii * 0.995
        
        res_sl = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': 15000, 'ftol': 1e-12, 'disp': False})
        if not np.isnan(res_sl.fun) and -res_sl.fun > best_sum - 1e-6:
            best_sum = -res_sl.fun
            best_centers = np.column_stack((res_sl.x[0::3], res_sl.x[1::3]))
            best_radii = res_sl.x[2::3].copy()
            
        for _ in range(12):
            noise = 0.002 * np.random.randn(N, 2)
            c_pert = np.clip(best_centers + noise, 0.01, 0.99)
            x_p = np.zeros(3 * N)
            x_p[0::3] = c_pert[:, 0]
            x_p[1::3] = c_pert[:, 1]
            r_p = solve_lp_radii(c_pert)
            x_p[2::3] = r_p * 0.998
            
            res_p = minimize(objective, x_p, method='SLSQP', bounds=bounds, constraints=cons,
                             options={'maxiter': 8000, 'ftol': 1e-12, 'disp': False})
            if not np.isnan(res_p.fun) and -res_p.fun > best_sum:
                best_sum = -res_p.fun
                best_centers = np.column_stack((res_p.x[0::3], res_p.x[1::3]))
                best_radii = res_p.x[2::3].copy()
                
        final_r = solve_lp_radii(best_centers)
        if np.sum(final_r) > np.sum(best_radii) - 1e-7:
            best_radii = final_r.copy()
            best_sum = np.sum(best_radii)

    if best_centers is None:
        best_centers = make_hex_init(0.09, 0.0)
        best_radii = solve_lp_radii(best_centers)
        best_sum = np.sum(best_radii)

    # Final strict validation repair
    for _ in range(100):
        valid = True
        for i in range(N):
            if best_radii[i] < 0 or best_centers[i, 0] < best_radii[i] - 1e-9 or \
               best_centers[i, 0] > 1.0 - best_radii[i] + 1e-9 or \
               best_centers[i, 1] < best_radii[i] - 1e-9 or \
               best_centers[i, 1] > 1.0 - best_radii[i] + 1e-9:
                valid = False
                break
        if valid:
            for i in range(N):
                for j in range(i + 1, N):
                    d = np.hypot(best_centers[i, 0] - best_centers[j, 0], 
                                 best_centers[i, 1] - best_centers[j, 1])
                    if d < best_radii[i] + best_radii[j] - 1e-9:
                        valid = False
                        break
                if not valid:
                    break
        if valid:
            break
        best_radii *= 0.9995
        best_centers[:, 0] = np.clip(best_centers[:, 0], best_radii, 1.0 - best_radii)
        best_centers[:, 1] = np.clip(best_centers[:, 1], best_radii, 1.0 - best_radii)
        
    return best_centers, best_radii, float(np.sum(best_radii))
