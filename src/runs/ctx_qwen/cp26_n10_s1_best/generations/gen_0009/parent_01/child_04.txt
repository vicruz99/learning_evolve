Key observations:
1. The LP step for radius optimization is exact - the main challenge is finding good center configurations
2. Hexagonal lattices are good starting points but get stuck in local optima
3. The constraint formulation with `np.hypot` is numerically stable but might have gradient issues at contact points
4. Need better strategies to escape local minima and explore diverse topologies

<strategy>
1. Use a hybrid approach: Multiple optimization phases with different strategies
2. Better initialization: More diverse hex lattice angles, corner-focused configs, and layered patterns
3. Improved LP-solver calls with tighter tolerances
4. Better deflation strategy: Instead of random subset, deflate circles with smallest radii (most constrained) to allow topology changes
5. Add a "center relaxation" phase where we only optimize centers with fixed large radii to push them apart
6. Use trust-constr method as alternative to SLSQP (better constraint handling)
7. Add explicit corner/edge exploitation - some circles should be in corners where they can be larger
8. Better final polishing with multiple LP iterations on perturbed centers

import numpy as np
from scipy.optimize import minimize, linprog
import warnings

warnings.filterwarnings('ignore')

N = 26
TRIL_IDX = np.tril_indices(N, -1)


def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])


def constraints(x):
    """Returns all inequality constraints >= 0 (vectorized)."""
    xs, ys, rs = x[0::3], x[1::3], x[2::3]

    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    c = np.concatenate([
        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs
    ])

    # Overlap constraints using squared distances for numerical stability
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]

    c = np.concatenate([c, dx[TRIL_IDX]**2 + dy[TRIL_IDX]**2 - dr[TRIL_IDX]**2])
    return c


def constraints_hypot(x):
    """Alternative constraint formulation using hypot."""
    xs, ys, rs = x[0::3], x[1::3], x[2::3]

    c = np.concatenate([
        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs
    ])

    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]

    c = np.concatenate([c, np.hypot(dx[TRIL_IDX], dy[TRIL_IDX]) - dr[TRIL_IDX]])
    return c


def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-7, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)])
    return b


def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)

    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1

    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist)
            idx += 1

    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 10.0})
        if res.success:
            return np.maximum(res.x, 1e-7)
    except Exception:
        pass
    return np.full(n, 0.05)


def force_spread(centers, steps=400):
    """Force-directed relaxation to spread points evenly and push to boundaries."""
    pts = centers.copy()
    for step in range(steps):
        f = np.zeros_like(pts)
        lr = 0.025 * (1.0 - step / steps)
        for i in range(N):
            for j in range(i + 1, N):
                dx = pts[j] - pts[i]
                d = np.hypot(dx[0], dx[1])
                if d < 0.3 and d > 1e-5:
                    rep = 0.008 / (d**2 + 0.0005)
                    f[i] -= dx * rep / d
                    f[j] += dx * rep / d
            # Stronger boundary push
            for dim in range(2):
                if pts[i, dim] < 0.06:
                    f[i, dim] += 0.08
                elif pts[i, dim] > 0.94:
                    f[i, dim] -= 0.08
        pts += f * lr
        pts = np.clip(pts, 0.01, 0.99)
    return pts


def make_hex_init(r0, angle):
    """Generates a rotated hexagonal lattice initialization."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 15:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0 and len(pts) < N + 15:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
    pts = np.array(pts[:N + 15])

    # Rotate around center
    center = np.array([0.5, 0.5])
    pts -= center
    c, s = np.cos(angle), np.sin(angle)
    pts = pts @ np.array([[c, -s], [s, c]])
    pts += center

    # Filter and pad
    mask = (pts[:, 0] >= 0.01) & (pts[:, 0] <= 0.99) & (pts[:, 1] >= 0.01) & (pts[:, 1] <= 0.99)
    pts = pts[mask]
    if len(pts) < N:
        pad = N - len(pts)
        pts = np.vstack([pts, np.random.uniform(0.15, 0.85, (pad, 2))])

    # Add tiny noise to break exact symmetries
    pts += np.random.uniform(-0.0005, 0.0005, pts.shape)
    return pts[:N]


def make_corner_heavy_init():
    """Generates initialization with circles in corners for larger radii."""
    pts = np.zeros((N, 2))
    r = 0.085
    
    # Four corners
    pts[0] = [r, r]
    pts[1] = [1 - r, r]
    pts[2] = [r, 1 - r]
    pts[3] = [1 - r, 1 - r]
    
    # Mid-edges
    pts[4] = [0.5, r]
    pts[5] = [0.5, 1 - r]
    pts[6] = [r, 0.5]
    pts[7] = [1 - r, 0.5]
    
    # Inner layer (hex-like)
    pts[8] = [0.5, 0.5]
    
    idx = 9
    # Fill with staggered pattern
    ys = [r + 0.06, r + 0.13, r + 0.20, r + 0.27, 1 - r - 0.06, 1 - r - 0.13]
    xs = [r + 0.08, r + 0.18, r + 0.28, 1 - r - 0.08, 1 - r - 0.18, 1 - r - 0.28, 0.35, 0.65]
    
    for y in ys:
        for x in xs:
            if idx >= N:
                break
            pts[idx] = [x, y]
            idx += 1
    
    for i in range(idx, N):
        pts[i] = np.random.uniform(0.15, 0.85, 2)
    
    pts += np.random.uniform(-0.002, 0.002, pts.shape)
    pts = np.clip(pts, 0.02, 0.98)
    return pts


def make_layered_init():
    """Generates layered initialization with specific row counts."""
    pts = []
    r = 0.088
    
    # Row patterns for different numbers per row
    patterns = [5, 5, 5, 6, 5]
    y = r
    for row_idx, count in enumerate(patterns):
        if len(pts) >= N:
            break
        row_shift = r if row_idx % 2 == 1 else 0.0
        spacing = (1.0 - 2 * r - row_shift) / (count - 1) if count > 1 else 0
        for col in range(count):
            if len(pts) >= N:
                break
            x = r + row_shift + col * spacing
            pts.append([x, y])
        y += np.sqrt(3) * r * 0.95
    
    while len(pts) < N:
        pts.append([np.random.uniform(0.1, 0.9), np.random.uniform(0.1, 0.9)])
    
    pts = np.array(pts[:N])
    pts += np.random.uniform(-0.001, 0.001, pts.shape)
    pts = np.clip(pts, 0.02, 0.98)
    return pts


def project_to_feasible(x):
    """Project variables to strictly satisfy bounds."""
    x = x.copy()
    for i in range(N):
        r = max(1e-6, x[3 * i + 2])
        x[3 * i] = np.clip(x[3 * i], r, 1.0 - r)
        x[3 * i + 1] = np.clip(x[3 * i + 1], r, 1.0 - r)
        x[3 * i + 2] = r
    return x


def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    bounds = get_bounds()
    cons_sq = {'type': 'ineq', 'fun': constraints}
    cons_hp = {'type': 'ineq', 'fun': constraints_hypot}

    best_sum = -1.0
    best_x = None

    # Phase 1: Generate Diverse Initial Configurations
    inits = []
    
    # Wide range of hex lattice configurations
    for r0 in np.linspace(0.080, 0.108, 8):
        for ang in np.linspace(-0.50, 0.50, 17):
            inits.append(make_hex_init(r0, ang))

    # Corner-heavy configurations
    for _ in range(10):
        inits.append(make_corner_heavy_init())
    
    # Layered configurations
    for _ in range(8):
        inits.append(make_layered_init())

    # Force-directed layouts
    for s in range(15):
        np.random.seed(s + 200)
        pts = np.random.uniform(0.12, 0.88, (N, 2))
        inits.append(force_spread(pts, steps=350))

    # Phase 2: Multi-start Optimization with LP Initialization
    for pts in inits:
        x0 = np.zeros(3 * N)
        x0[0::3] = pts[:, 0]
        x0[1::3] = pts[:, 1]

        # Initialize radii via LP for maximum head-start
        r_lp = solve_lp_radii(pts)
        x0[2::3] = np.maximum(r_lp * 0.995, 0.008)
        x0 = project_to_feasible(x0)

        # Try both constraint formulations
        for cons in [cons_sq, cons_hp]:
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 20000, 'ftol': 1e-14, 'disp': False})
                if not np.isnan(res.fun):
                    c_tmp = np.column_stack((res.x[0::3], res.x[1::3]))
                    r_new = solve_lp_radii(c_tmp)
                    if r_new is not None:
                        s = np.sum(r_new)
                        if s > best_sum:
                            best_sum = s
                            best_x = res.x.copy()
                            best_x[2::3] = r_new
            except Exception:
                pass

    # Phase 3: Deflation & Perturbation to Escape Local Minima
    if best_x is not None:
        for cycle in range(80):
            noise_scale = 0.003 * (0.91 ** cycle)
            x0 = best_x + np.random.normal(0, noise_scale, 3 * N)

            # Smart deflation: deflate smallest radius circles (most constrained)
            r_current = x0[2::3]
            sorted_idx = np.argsort(r_current)
            n_deflate = max(3, N // 4)
            for idx in sorted_idx[:n_deflate]:
                x0[3 * idx + 2] *= 0.85

            # Also deflate one random circle aggressively
            ridx = np.random.randint(N)
            x0[3 * ridx + 2] *= 0.5

            x0 = project_to_feasible(x0)

            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons_sq,
                               options={'maxiter': 15000, 'ftol': 1e-14, 'disp': False})
                if not np.isnan(res.fun):
                    c_tmp = np.column_stack((res.x[0::3], res.x[1::3]))
                    r_new = solve_lp_radii(c_tmp)
                    if r_new is not None:
                        x_p = res.x.copy()
                        x_p[2::3] = np.maximum(r_new * 0.998, 0.005)
                        x_p = project_to_feasible(x_p)

                        s = np.sum(x_p[2::3])
                        vals = constraints(x_p)
                        if np.min(vals) >= -1e-4 and s > best_sum:
                            best_sum = s
                            best_x = x_p.copy()
            except Exception:
                pass

        # Phase 3b: Try center-only optimization (fix radii, move centers)
        best_centers = np.column_stack((best_x[0::3], best_x[1::3]))
        best_radii = best_x[2::3]
        
        for trial in range(20):
            np.random.seed(trial + 500)
            pert_centers = best_centers + np.random.normal(0, 0.005, best_centers.shape)
            pert_centers = np.clip(pert_centers, 0.02, 0.98)
            
            # Try to find better radii for perturbed centers
            r_try = solve_lp_radii(pert_centers)
            if r_try is not None:
                s_try = np.sum(r_try)
                if s_try > best_sum:
                    # Validate
                    valid = True
                    for i in range(N):
                        if r_try[i] < 1e-6:
                            valid = False
                            break
                        if pert_centers[i, 0] < r_try[i] - 1e-8 or pert_centers[i, 0] > 1 - r_try[i] + 1e-8:
                            valid = False
                            break
                        if pert_centers[i, 1] < r_try[i] - 1e-8 or pert_centers[i, 1] > 1 - r_try[i] + 1e-8:
                            valid = False
                            break
                    if valid:
                        for i in range(N):
                            for j in range(i + 1, N):
                                d = np.hypot(pert_centers[i, 0] - pert_centers[j, 0], 
                                           pert_centers[i, 1] - pert_centers[j, 1])
                                if d < r_try[i] + r_try[j] - 1e-8:
                                    valid = False
                                    break
                            if not valid:
                                break
                    if valid:
                        best_sum = s_try
                        best_centers = pert_centers.copy()
                        best_radii = r_try.copy()

    # Fallback (should not be reached)
    if best_x is None:
        best_x = np.zeros(3 * N)
        best_x[0::3] = np.tile(np.linspace(0.1, 0.9, 5), 6)[:N]
        best_x[1::3] = np.repeat(np.linspace(0.1, 0.9, 6), 5)[:N]
        best_x[2::3] = 0.06
        best_sum = np.sum(best_x[2::3])
    
    # Use best_centers/best_radii if better
    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3]
    
    # Final strict validation repair against 1e-12 tolerance
    for _ in range(100):
        valid = True
        for i in range(N):
            if radii[i] < 0 or centers[i, 0] < radii[i] - 1e-9 or centers[i, 0] > 1.0 - radii[i] + 1e-9 or \
               centers[i, 1] < radii[i] - 1e-9 or centers[i, 1] > 1.0 - radii[i] + 1e-9:
                valid = False
                break
        if valid:
            for i in range(N):
                for j in range(i + 1, N):
                    d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                    if d < radii[i] + radii[j] - 1e-9:
                        valid = False
                        break
                if not valid:
                    break
        if valid:
            break

        radii *= 0.9997
        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)

    return centers, radii, float(np.sum(radii))
