import numpy as np
from scipy.optimize import minimize, linprog
import warnings

warnings.filterwarnings('ignore')

N = 26
TRIL_IDX = np.tril_indices(N, -1)
N_OVERLAP = len(TRIL_IDX[0])

# Precompute constant part of the LP constraint matrix for overlap constraints
A_OVERLAP = np.zeros((N_OVERLAP, N))
A_OVERLAP[np.arange(N_OVERLAP), TRIL_IDX[0]] = 1.0
A_OVERLAP[np.arange(N_OVERLAP), TRIL_IDX[1]] = 1.0

def solve_radii_lp(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    limits = np.minimum(centers, 1.0 - centers).min(axis=1)
    
    dx = centers[:, 0, None] - centers[None, 0, :]
    dy = centers[:, 1, None] - centers[None, 1, :]
    dists = np.hypot(dx, dy)
    b_overlap = dists[TRIL_IDX]
    
    A_ub = np.vstack([np.eye(N), A_OVERLAP])
    b_ub = np.concatenate([limits, b_overlap])
    
    bounds = [(0.0, None)] * N
    try:
        res = linprog(-np.ones(N), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
    return np.zeros(N), 0.0

def sa_optimize(init_centers, steps=2500, T_init=0.025):
    """Simulated Annealing on centers, using LP for exact radius evaluation."""
    centers = init_centers.copy()
    current_radii, current_val = solve_radii_lp(centers)
    best_centers = centers.copy()
    best_val = current_val
    
    T = T_init
    cooling = (1e-6 / T_init) ** (1.0 / steps)
    
    for _ in range(steps):
        i = np.random.randint(N)
        delta = T * np.random.randn(2)
        new_c = centers.copy()
        new_c[i] = np.clip(new_c[i] + delta, 0.001, 0.999)
        
        new_radii, new_val = solve_radii_lp(new_c)
        d_val = new_val - current_val
        
        if d_val > 0 or np.random.rand() < np.exp(d_val / T):
            centers = new_c
            current_radii = new_radii
            current_val = new_val
            
        if current_val > best_val:
            best_val = current_val
            best_centers = centers.copy()
            
        T *= cooling
    return best_centers, best_val

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Returns all inequality constraints >= 0 (vectorized)."""
    xs, ys, rs = x[0::3], x[1::3], x[2::3]
    
    c = np.concatenate([
        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs
    ])
    
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    c = np.concatenate([c, np.hypot(dx[TRIL_IDX], dy[TRIL_IDX]) - dr[TRIL_IDX]])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-7, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)])
    return b

def make_hex_init(seed, scale=1.0):
    """Generates a rotated hexagonal lattice initialization."""
    np.random.seed(seed)
    pts = []
    r0 = 0.09 * scale
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0 and len(pts) < N + 10:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
    pts = np.array(pts[:N + 10])
    
    # Random rotation
    ang = np.random.uniform(-0.4, 0.4)
    c, s = np.cos(ang), np.sin(ang)
    pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
    
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    if len(pts) < N:
        pad = N - len(pts)
        pts = np.vstack([pts, np.random.uniform(0.2, 0.8, (pad, 2))])
        
    pts += np.random.uniform(-0.002, 0.002, pts.shape)
    return np.clip(pts[:N], 0.01, 0.99)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    best_sum = -1.0
    best_x = None
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    
    # Phase 1: Simulated Annealing exploration from diverse starts
    inits = []
    for s in range(4):
        inits.append(make_hex_init(s, scale=0.95 + 0.05*s))
    # Add a force-spread/random init
    for _ in range(2):
        pts = np.random.uniform(0.15, 0.85, (N, 2))
        for _ in range(100):
            f = np.zeros_like(pts)
            for i in range(N):
                for j in range(i+1, N):
                    dx = pts[j] - pts[i]
                    d = np.hypot(dx[0], dx[1])
                    if d < 0.25 and d > 1e-5:
                        rep = 0.005 / (d**2 + 0.001)
                        f[i] -= dx * rep / d
                        f[j] += dx * rep / d
            pts += f * 0.05
            pts = np.clip(pts, 0.05, 0.95)
        inits.append(pts)

    for init_c in inits:
        c_sa, val_sa = sa_optimize(init_c, steps=2500, T_init=0.03)
        
        # Phase 2: SLSQP joint refinement on SA results
        r_lp, _ = solve_radii_lp(c_sa)
        x0 = np.zeros(3 * N)
        x0[0::3] = c_sa[:, 0]
        x0[1::3] = c_sa[:, 1]
        x0[2::3] = r_lp * 0.9995
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 12000, 'ftol': 1e-14, 'disp': False})
            if not np.isnan(res.fun):
                curr_sum = -res.fun
                c_vals = constraints(res.x)
                if np.min(c_vals) >= -1e-6 and curr_sum > best_sum:
                    best_sum = curr_sum
                    best_x = res.x.copy()
        except Exception:
            pass
            
        # Keep raw SA best as fallback
        if val_sa > best_sum:
            best_sum = val_sa
            best_x = np.zeros(3 * N)
            best_x[0::3] = c_sa[:, 0]
            best_x[1::3] = c_sa[:, 1]
            best_x[2::3] = r_lp

    # Fallback initialization if all failed
    if best_x is None:
        best_x = np.zeros(3 * N)
        best_x[0::3] = np.tile(np.linspace(0.1, 0.9, 5), 6)[:N]
        best_x[1::3] = np.repeat(np.linspace(0.1, 0.9, 6), 5)[:N]
        best_x[2::3] = 0.08
        best_sum = np.sum(best_x[2::3])
        
    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3]
    
    # Final strict validation repair against 1e-12 tolerance
    for _ in range(100):
        valid = True
        for i in range(N):
            if radii[i] < 0 or centers[i, 0] < radii[i] - 1e-9 or centers[i, 0] > 1.0 - radii[i] + 1e-9 or \
               centers[i, 1] < radii[i] - 1e-9 or centers[i, 1] > 1.0 - radii[i] + 1e-9:
                valid = False
                break
        if valid:
            for i in range(N):
                for j in range(i + 1, N):
                    d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                    if d < radii[i] + radii[j] - 1e-9:
                        valid = False
                        break
                if not valid:
                    break
        if valid:
            break
            
        radii *= 0.9998
        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)
        
    return centers, radii, float(np.sum(radii))
