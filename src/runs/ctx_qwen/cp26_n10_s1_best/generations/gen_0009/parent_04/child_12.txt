import numpy as np
from scipy.optimize import minimize, linprog
import warnings
warnings.filterwarnings('ignore')

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraint_fun(x):
    """Returns all inequality constraints >= 0 (vectorized)."""
    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    c = np.concatenate([
        xs - rs, 1.0 - xs - rs,
        ys - rs, 1.0 - ys - rs
    ])
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    c = np.concatenate([c, dx[TRIL_IDX]**2 + dy[TRIL_IDX]**2 - dr[TRIL_IDX]**2])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-8, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-8, 0.5)])
    return b

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    for i in range(n):
        bound = min(centers[i, 0], 1.0 - centers[i, 0], 
                    centers[i, 1], 1.0 - centers[i, 1])
        A[idx, i] = 1.0
        b_ub[idx] = max(0.0, bound)
        idx += 1
        
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], 
                            centers[i, 1] - centers[j, 1])
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist)
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-7)
    except Exception:
        pass
    return np.full(n, 0.04)

def force_spread(centers, steps=400):
    """Force-directed relaxation to spread points evenly and push to boundaries."""
    pts = centers.copy()
    for step in range(steps):
        forces = np.zeros_like(pts)
        lr = 0.04 * (1.0 - step / steps)
        for i in range(N):
            for dim in range(2):
                if pts[i, dim] < 0.05: forces[i, dim] += 0.06
                elif pts[i, dim] > 0.95: forces[i, dim] -= 0.06
                
            for j in range(i + 1, N):
                diff = pts[j] - pts[i]
                d = np.linalg.norm(diff)
                if d < 0.35 and d > 1e-6:
                    f = 0.006 / (d**2 + 0.001)
                    fx, fy = f * diff / d
                    forces[i] -= [fx, fy]
                    forces[j] += [fx, fy]
                    
        pts += forces * lr
        pts = np.clip(pts, 0.01, 0.99)
    return pts

def project_x(x0):
    """Ensure optimization vector strictly respects bounds."""
    x0 = x0.copy()
    for i in range(N):
        r = max(1e-7, x0[3 * i + 2])
        x0[3 * i] = np.clip(x0[3 * i], r, 1.0 - r)
        x0[3 * i + 1] = np.clip(x0[3 * i + 1], r, 1.0 - r)
        x0[3 * i + 2] = r
    return x0

def make_hex_init(r0, angle):
    """Generates a hexagonal lattice initialization with optional rotation."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
    pts = np.array(pts[:N + 10])
    
    if angle != 0.0:
        c_a, s_a = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c_a, -s_a], [s_a, c_a]]) + 0.5
        
    mask = (pts[:, 0] > 0.02) & (pts[:, 0] < 0.98) & (pts[:, 1] > 0.02) & (pts[:, 1] < 0.98)
    pts = pts[mask]
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.15, 0.85, (1, 2))])
    return pts[:N]

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    best_sum = -1.0
    best_x = None
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraint_fun}
    
    # Phase 1: Generate diverse initial configurations
    inits = []
    
    # 1. Hexagonal lattices with varied density and rotation
    for r0 in np.linspace(0.075, 0.105, 8):
        for ang in np.linspace(-0.35, 0.35, 11):
            inits.append(make_hex_init(r0, ang))

    # 2. Force-directed spreads
    for s in range(8):
        np.random.seed(s)
        pts = np.random.uniform(0.15, 0.85, (N, 2))
        pts = force_spread(pts, steps=600)
        inits.append(pts)

    # 3. Corner/Edge biased layouts
    for s in range(6):
        np.random.seed(100 + s)
        pts = np.random.uniform(0.05, 0.95, (N, 2))
        pts[:10] = np.random.uniform(0.05, 0.3, (10, 2))
        pts[10:20] = np.random.uniform(0.7, 0.95, (10, 2))
        pts = force_spread(pts, steps=700)
        inits.append(pts)

    # Phase 2: Initial Optimization with LP-SLSQP alternation
    for pts in inits:
        r_lp = solve_lp_radii(pts)
        x0 = np.zeros(3 * N)
        x0[0::3] = pts[:, 0]
        x0[1::3] = pts[:, 1]
        x0[2::3] = np.maximum(r_lp * 0.995, 0.005)
        x0 = project_x(x0)
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 15000, 'ftol': 1e-13, 'disp': False})
            if not np.isnan(res.fun):
                s_val = -res.fun
                c_val = constraint_fun(res.x)
                if np.min(c_val) >= -1e-5 and s_val > best_sum:
                    best_sum = s_val
                    best_x = res.x.copy()
        except Exception:
            pass

    # Phase 3: Aggressive Topology Search & Asymmetric Deflation
    if best_x is not None:
        for cycle in range(90):
            x0 = best_x.copy()
            
            # Asymmetric radius deflation/perturbation to break symmetry
            shrink = 0.92 + 0.06 * (cycle / 90.0)
            x0[2::3] *= shrink
            subset = np.random.choice(N, size=max(5, N // 4), replace=False)
            x0[subset * 3 + 2] *= np.random.uniform(0.7, 1.3, len(subset))
            
            # Decaying noise perturbation
            noise_scale = 0.006 * (0.94 ** (cycle // 2))
            x0[0::3] += np.random.normal(0, noise_scale, N)
            x0[1::3] += np.random.normal(0, noise_scale, N)
            
            # Random swap to break lattice rigidity
            if cycle % 5 == 0:
                i_sw, j_sw = np.random.choice(N, 2, replace=False)
                x0[3*i_sw], x0[3*j_sw] = x0[3*j_sw], x0[3*i_sw]
                x0[3*i_sw+1], x0[3*j_sw+1] = x0[3*j_sw+1], x0[3*i_sw+1]
                
            x0 = project_x(x0)
            
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 10000, 'ftol': 1e-13, 'disp': False})
                if not np.isnan(res.fun):
                    s_val = -res.fun
                    if s_val > best_sum:
                        best_sum = s_val
                        best_x = res.x.copy()
                        
                    # LP refinement on new centers to maximize radii
                    c_tmp = np.column_stack((best_x[0::3], best_x[1::3]))
                    r_new = solve_lp_radii(c_tmp)
                    if r_new is not None:
                        best_x[2::3] = r_new * 0.998
                        best_sum = np.sum(best_x[2::3])
                        
                        # Secondary SLSQP pass to fine-tune centers
                        x0_fix = best_x.copy()
                        x0_fix[2::3] *= 0.988
                        x0_fix = project_x(x0_fix)
                        try:
                            res2 = minimize(objective, x0_fix, method='SLSQP', bounds=bounds, constraints=cons,
                                            options={'maxiter': 8000, 'ftol': 1e-13, 'disp': False})
                            if not np.isnan(res2.fun) and -res2.fun > best_sum:
                                best_sum = -res2.fun
                                best_x = res2.x.copy()
                        except Exception:
                            pass
            except Exception:
                pass

    # Phase 4: High-precision trust-constr polish (optional but powerful)
    if best_x is not None:
        try:
            from scipy.optimize import NonlinearConstraint
            # Define constraints for trust-constr
            cons_tc = NonlinearConstraint(constraint_fun, np.zeros(len(constraint_fun(best_x))), np.inf)
            res_tc = minimize(objective, best_x, method='trust-constr', bounds=bounds, 
                              constraints=cons_tc, options={'maxiter': 15000, 'verbose': 0})
            if not np.isnan(res_tc.fun) and -res_tc.fun > best_sum:
                c_val = constraint_fun(res_tc.x)
                if np.min(c_val) >= -1e-6:
                    best_sum = -res_tc.fun
                    best_x = res_tc.x.copy()
        except Exception:
            pass

    # Fallback (should not be reached)
    if best_x is None:
        best_x = np.zeros(3 * N)
        best_x[0::3] = np.tile(np.linspace(0.1, 0.9, 5), 6)[:N]
        best_x[1::3] = np.repeat(np.linspace(0.1, 0.9, 6), 5)[:N]
        best_x[2::3] = 0.08

    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3].copy()

    # Final strict validation repair to guarantee compliance with validator tolerances
    for _ in range(200):
        valid = True
        for i in range(N):
            if radii[i] < 0 or centers[i, 0] < radii[i] - 1e-9 or centers[i, 0] > 1.0 - radii[i] + 1e-9 or \
               centers[i, 1] < radii[i] - 1e-9 or centers[i, 1] > 1.0 - radii[i] + 1e-9:
                valid = False
                break
        if valid:
            for i in range(N):
                for j in range(i + 1, N):
                    d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                    if d < radii[i] + radii[j] - 1e-9:
                        valid = False
                        break
                if not valid:
                    break
        if valid:
            break
            
        # Minimal shrinkage to recover strict feasibility
        radii *= 0.9998
        for i in range(N):
            centers[i, 0] = np.clip(centers[i, 0], radii[i], 1.0 - radii[i])
            centers[i, 1] = np.clip(centers[i, 1], radii[i], 1.0 - radii[i])
            
    return centers, radii, float(np.sum(radii))
