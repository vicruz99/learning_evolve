import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Returns all inequality constraints >= 0 (vectorized)."""
    cx, cy, r = x[0::3], x[1::3], x[2::3]
    
    # Boundary constraints
    c = np.concatenate([
        cx - r, 1.0 - cx - r,
        cy - r, 1.0 - cy - r
    ])
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dr = r[:, None] + r[None, :]
    
    i, j = np.tril_indices(N, -1)
    c = np.concatenate([c, dx[i, j]**2 + dy[i, j]**2 - dr[i, j]**2])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-7, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)])
    return b

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], 
                  centers[i, 1], 1.0 - centers[i, 1])
        A[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], 
                            centers[i, 1] - centers[j, 1])
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist)
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return np.full(n, 0.05)

def force_spread(centers, steps=400):
    """Force-directed relaxation to spread points evenly and push to boundaries."""
    pts = centers.copy()
    for step in range(steps):
        forces = np.zeros_like(pts)
        for i in range(N):
            # Boundary repulsion
            for dim in range(2):
                if pts[i, dim] < 0.10: forces[i, dim] += 0.04
                elif pts[i, dim] > 0.90: forces[i, dim] -= 0.04
                
            # Pairwise repulsion
            for j in range(i + 1, N):
                diff = pts[j] - pts[i]
                d = np.linalg.norm(diff)
                if d < 0.35 and d > 1e-6:
                    f = 0.006 / (d**2 + 0.001)
                    forces[i] -= diff * f / d
                    forces[j] += diff * f / d
                    
        lr = 0.05 * (1.0 - step / steps)
        pts += forces * lr
        pts = np.clip(pts, 0.01, 0.99)
    return pts

def project_x(x0):
    """Ensure optimization vector strictly respects bounds."""
    x0 = x0.copy()
    for i in range(N):
        r = max(1e-7, x0[3 * i + 2])
        x0[3 * i] = np.clip(x0[3 * i], r, 1.0 - r)
        x0[3 * i + 1] = np.clip(x0[3 * i + 1], r, 1.0 - r)
        x0[3 * i + 2] = r
    return x0

def make_hex_init(r0, ang):
    """Generates a rotated hexagonal lattice initialization."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 8:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0 and len(pts) < N + 8:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        
    pts = np.array(pts[:N + 8])
    
    if ang != 0.0:
        c, s = np.cos(ang), np.sin(ang)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
        
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.15, 0.85, (1, 2))])
    return pts[:N]

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    best_sum = -1.0
    best_x = None
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    
    # Phase 1: Generate diverse initial configurations
    inits = []
    
    # 1. Hexagonal lattices with varied density and rotation
    for r0 in np.linspace(0.08, 0.105, 7):
        for ang in np.linspace(-0.3, 0.3, 11):
            inits.append(make_hex_init(r0, ang))

    # 2. Force-directed spreads
    for s in range(8):
        np.random.seed(s)
        pts = np.random.uniform(0.15, 0.85, (N, 2))
        pts = force_spread(pts, steps=500)
        inits.append(pts)

    # 3. Boundary-biased layouts
    for s in range(4):
        np.random.seed(100 + s)
        pts = np.random.uniform(0.05, 0.95, (N, 2))
        pts[:8] = np.random.uniform(0.05, 0.25, (8, 2))
        pts[8:16] = np.random.uniform(0.75, 0.95, (8, 2))
        pts = force_spread(pts, steps=600)
        inits.append(pts)

    # Phase 2: Initial Optimization with LP-SLSQP alternation
    for pts in inits:
        r_lp = solve_lp_radii(pts)
        x0 = np.zeros(3 * N)
        x0[0::3] = pts[:, 0]
        x0[1::3] = pts[:, 1]
        x0[2::3] = np.maximum(r_lp * 0.995, 0.005)
        x0 = project_x(x0)
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 15000, 'ftol': 1e-13, 'disp': False})
            if not np.isnan(res.fun):
                s_val = -res.fun
                if s_val > best_sum:
                    best_sum = s_val
                    best_x = res.x.copy()
        except Exception:
            pass

    # Phase 3: Aggressive Topology Search & Refinement
    if best_x is not None:
        for cycle in range(90):
            x0 = best_x.copy()
            
            # Deflate radii to create slack for center movement
            shrink = 0.82 + 0.14 * (cycle / 90.0)
            x0[2::3] *= shrink
            
            # Decaying noise perturbation
            noise_scale = 0.006 * (0.94 ** (cycle // 4))
            x0[0::3] += np.random.normal(0, noise_scale, N)
            x0[1::3] += np.random.normal(0, noise_scale, N)
            
            # Random swap of two circles to break lattice symmetries
            if cycle % 3 == 0:
                i_sw, j_sw = np.random.choice(N, 2, replace=False)
                x0[i_sw*3:i_sw*3+3], x0[j_sw*3:j_sw*3+3] = x0[j_sw*3:j_sw*3+3].copy(), x0[i_sw*3:i_sw*3+3].copy()
                
            x0 = project_x(x0)
            
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 10000, 'ftol': 1e-13, 'disp': False})
                if not np.isnan(res.fun):
                    # LP refinement on new centers to maximize radii
                    c_tmp = np.column_stack((res.x[0::3], res.x[1::3]))
                    r_new = solve_lp_radii(c_tmp)
                    if r_new is not None:
                        x0[2::3] = r_new * 0.998
                        x0 = project_x(x0)
                        
                        # Secondary SLSQP pass to fine-tune centers with expanded radii
                        try:
                            res2 = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                                            options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
                            if not np.isnan(res2.fun):
                                c_final = np.column_stack((res2.x[0::3], res2.x[1::3]))
                                r_final = solve_lp_radii(c_final)
                                if r_final is not None:
                                    s_val2 = np.sum(r_final)
                                    if s_val2 > best_sum:
                                        best_sum = s_val2
                                        best_x = np.zeros(3 * N)
                                        best_x[0::3] = c_final[:, 0]
                                        best_x[1::3] = c_final[:, 1]
                                        best_x[2::3] = r_final
                        except Exception:
                            pass
            except Exception:
                pass

    # Fallback (should not be reached)
    if best_x is None:
        best_x = np.zeros(3 * N)
        best_x[0::3] = np.tile(np.linspace(0.1, 0.9, 5), 6)[:N]
        best_x[1::3] = np.repeat(np.linspace(0.1, 0.9, 6), 5)[:N]
        best_x[2::3] = 0.08

    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3].copy()

    # Final strict validation repair to guarantee compliance with validator tolerances
    for _ in range(150):
        valid = True
        for i in range(N):
            if radii[i] < 0 or centers[i, 0] < radii[i] - 1e-9 or centers[i, 0] > 1.0 - radii[i] + 1e-9 or \
               centers[i, 1] < radii[i] - 1e-9 or centers[i, 1] > 1.0 - radii[i] + 1e-9:
                valid = False
                break
        if valid:
            for i in range(N):
                for j in range(i + 1, N):
                    d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                    if d < radii[i] + radii[j] - 1e-9:
                        valid = False
                        break
                if not valid:
                    break
        if valid:
            break
            
        # Minimal shrinkage to recover strict feasibility
        radii *= 0.9995
        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)
            
    return centers, radii, float(np.sum(radii))
