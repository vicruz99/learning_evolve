

<strategy>
1. **Hybrid Initialization**: Generate diverse starting configurations using rotated hexagonal lattices (varying pitch and angle), force-directed random layouts, and edge/corner-biased patterns. This ensures we explore multiple topological basins of attraction.
2. **LP-SLSQP Alternation**: Decouple the problem. Use Linear Programming (`linprog`) to compute the exact optimal radii for any fixed set of centers. Use SLSQP for joint optimization of centers and radii. This leverages LP's exactness for radii and SLSQP's ability to navigate the non-convex center space.
3. **Deflation & Perturbation Search**: To escape local minima, systematically deflate radii (creating slack) and inject decaying Gaussian noise into centers. Periodically swap random circle positions to break rigid lattice symmetries that trap standard optimizers.
4. **Iterative Refinement Loop**: Run a robust cycle: Perturb -> Deflate -> Project to bounds -> SLSQP -> LP polish -> Update best. Repeat with cooling parameters.
5. **Strict Feasibility Repair**: A final deterministic pass clips centers and minimally shrinks radii to guarantee strict compliance with the validator's 1e-12 tolerance, ensuring no numerical drift causes overlaps.
</strategy>
```python
import numpy as np
from scipy.optimize import minimize, linprog

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Returns all inequality constraints >= 0 (vectorized)."""
    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    c = np.concatenate([
        xs - rs, 1.0 - xs - rs,
        ys - rs, 1.0 - ys - rs
    ])
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    c = np.concatenate([c, dx[TRIL_IDX]**2 + dy[TRIL_IDX]**2 - dr[TRIL_IDX]**2])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-7, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)])
    return b

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        bound = min(centers[i, 0], 1.0 - centers[i, 0], 
                    centers[i, 1], 1.0 - centers[i, 1])
        A[idx, i] = 1.0
        b_ub[idx] = max(0.0, bound)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], 
                            centers[i, 1] - centers[j, 1])
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist)
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-8)
    except Exception:
        pass
    return np.full(n, 0.05)

def force_directed(centers, radii, steps=300):
    """Force-directed relaxation to spread points evenly respecting radii."""
    pts = centers.copy()
    rs = radii.copy()
    for step in range(steps):
        forces = np.zeros_like(pts)
        lr = 0.04 * (1.0 - step / steps)
        for i in range(N):
            # Boundary forces to push outward
            for dim in range(2):
                if pts[i, dim] < rs[i] + 0.015:
                    forces[i, dim] += 0.1
                elif pts[i, dim] > 1.0 - rs[i] - 0.015:
                    forces[i, dim] -= 0.1
                    
            # Pairwise repulsion based on target clearance
            for j in range(i + 1, N):
                diff = pts[j] - pts[i]
                d = np.linalg.norm(diff)
                target = rs[i] + rs[j] + 0.01
                if d < target and d > 1e-6:
                    f = 0.5 * (target - d) / (d**2 + 0.01)
                    fx, fy = f * diff / d
                    forces[i] -= [fx, fy]
                    forces[j] += [fx, fy]
                    
        pts += forces * lr
        pts = np.clip(pts, 0.001, 0.999)
    return pts

def project_x(x0):
    """Ensure optimization vector strictly respects bounds."""
    x0 = x0.copy()
    for i in range(N):
        r = max(1e-7, x0[3 * i + 2])
        x0[3 * i] = np.clip(x0[3 * i], r, 1.0 - r)
        x0[3 * i + 1] = np.clip(x0[3 * i + 1], r, 1.0 - r)
        x0[3 * i + 2] = r
    return x0

def run_packing() -> tuple:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    best_sum = -1.0
    best_x = None
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    
    inits = []
    
    # 1. Hexagonal lattices with varied density and rotation
    for r0 in np.linspace(0.082, 0.108, 6):
        for ang in np.linspace(-0.35, 0.35, 11):
            pts = []
            y = r0
            row = 0
            while len(pts) < N + 10:
                x = r0 if row % 2 == 0 else 2.0 * r0
                while x <= 1.0 - r0:
                    pts.append([x, y])
                    x += 2.0 * r0
                y += np.sqrt(3.0) * r0
                row += 1
            pts = np.array(pts[:N + 10])
            if ang != 0.0:
                c_a, s_a = np.cos(ang), np.sin(ang)
                pts = (pts - 0.5) @ np.array([[c_a, -s_a], [s_a, c_a]]) + 0.5
            mask = (pts[:, 0] > 0.02) & (pts[:, 0] < 0.98) & (pts[:, 1] > 0.02) & (pts[:, 1] < 0.98)
            pts = pts[mask]
            while len(pts) < N:
                pts = np.vstack([pts, np.random.uniform(0.15, 0.85, (1, 2))])
            inits.append(pts[:N])
            
    # 2. Force-directed spreads from random starts
    for s in range(10):
        np.random.seed(s)
        pts = np.random.uniform(0.15, 0.85, (N, 2))
        pts = force_directed(pts, np.full(N, 0.06), steps=400)
        inits.append(pts)
        
    # 3. Edge/Corner biased layouts
    for s in range(6):
        np.random.seed(100 + s)
        pts = np.random.uniform(0.05, 0.95, (N, 2))
        pts[:10] = np.random.uniform(0.05, 0.3, (10, 2))
        pts[10:20] = np.random.uniform(0.7, 0.95, (10, 2))
        pts = force_directed(pts, np.full(N, 0.06), steps=500)
        inits.append(pts)

    # Phase 1: Initial Optimization
    for pts in inits:
        r_lp = solve_lp_radii(pts)
        x0 = np.zeros(3 * N)
        x0[0::3] = pts[:, 0]
        x0[1::3] = pts[:, 1]
        x0[2::3] = np.maximum(r_lp * 0.995, 0.005)
        x0 = project_x(x0)
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 15000, 'ftol': 1e-13, 'disp': False})
            if not np.isnan(res.fun):
                s_val = -res.fun
                c_val = constraints(res.x)
                if np.min(c_val) >= -1e-5 and s_val > best_sum:
                    best_sum = s_val
                    best_x = res.x.copy()
        except Exception:
            pass

    # Phase 2: Aggressive Topology Search & Refinement
    if best_x is not None:
        for cyc in range(90):
            x0 = best_x.copy()
            
            # Deflate radii progressively to create slack
            shrink = 0.88 + 0.10 * (cyc / 90.0)
            x0[2::3] *= shrink
            
            # Decaying noise perturbation
            noise_scale = 0.006 * (0.94 ** cyc)
            x0[0::3] += np.random.normal(0, noise_scale, N)
            x0[1::3] += np.random.normal(0, noise_scale, N)
            
            # Random swap to break symmetries
            if cyc % 5 == 0:
                i_sw, j_sw = np.random.choice(N, 2, replace=False)
                for k in range(3):
                    x0[3*i_sw + k], x0[3*j_sw + k] = x0[3*j_sw + k], x0[3*i_sw + k]
                    
            x0 = project_x(x0)
            
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 12000, 'ftol': 1e-13, 'disp': False})
                if not np.isnan(res.fun):
                    s_val = -res.fun
                    if s_val > best_sum:
                        best_sum = s_val
                        best_x = res.x.copy()
                        
                    # LP refinement on new centers
                    c_tmp = np.column_stack((best_x[0::3], best_x[1::3]))
                    r_new = solve_lp_radii(c_tmp)
                    if r_new is not None:
                        x_lp = best_x.copy()
                        x_lp[2::3] = np.maximum(r_new * 0.998, 0.001)
                        x_lp = project_x(x_lp)
                        
                        try:
                            res2 = minimize(objective, x_lp, method='SLSQP', bounds=bounds, constraints=cons,
                                           options={'maxiter': 8000, 'ftol': 1e-13, 'disp': False})
                            if not np.isnan(res2.fun) and -res2.fun > best_sum:
                                best_sum = -res2.fun
                                best_x = res2.x.copy()
                        except Exception:
                            pass
            except Exception:
                pass

    # Fallback (should not be reached)
    if best_x is None:
        best_x = np.zeros(3 * N)
        best_x[0::3] = np.tile(np.linspace(0.1, 0.9, 5), 6)[:N]
        best_x[1::3] = np.repeat(np.linspace(0.1, 0.9, 6), 5)[:N]
        best_x[2::3] = 0.08

    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3].copy()

    # Final strict validation repair to guarantee compliance with validator tolerances
    for _ in range(200):
        valid = True
        for i in range(N):
            if radii[i] < 0 or centers[i, 0] < radii[i] - 1e-9 or centers[i, 0] > 1.0 - radii[i] + 1e-9 or \
               centers[i, 1] < radii[i] - 1e-9 or centers[i, 1] > 1.0 - radii[i] + 1e-9:
                valid = False
                break
        if valid:
            for i in range(N):
                for j in range(i + 1, N):
                    d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                    if d < radii[i] + radii[j] - 1e-9:
                        valid = False
                        break
                if not valid:
                    break
        if valid:
            break
            
        # Minimal shrinkage to recover strict feasibility
        radii *= 0.9995
        for i in range(N):
            centers[i, 0] = np.clip(centers[i, 0], radii[i], 1.0 - radii[i])
            centers[i, 1] = np.clip(centers[i, 1], radii[i], 1.0 - radii[i])
            
    return centers, radii, float(np.sum(radii))
```
