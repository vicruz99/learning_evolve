import numpy as np
from scipy.optimize import minimize, linprog
import warnings
warnings.filterwarnings('ignore')

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraint_fun(x):
    """Returns all inequality constraints >= 0 (vectorized)."""
    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]
    
    # Boundary constraints
    c = np.concatenate([
        xs - rs, 1.0 - xs - rs,
        ys - rs, 1.0 - ys - rs
    ])
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    c = np.concatenate([c, dx[TRIL_IDX]**2 + dy[TRIL_IDX]**2 - dr[TRIL_IDX]**2])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-8, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-8, 0.5)])
    return b

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        bound = min(centers[i, 0], 1.0 - centers[i, 0], 
                    centers[i, 1], 1.0 - centers[i, 1])
        A[idx, i] = 1.0
        b_ub[idx] = max(0.0, bound)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], 
                            centers[i, 1] - centers[j, 1])
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist)
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return np.full(n, 0.01)

def force_spread(centers, steps=300):
    """Force-directed relaxation to spread points evenly."""
    pts = centers.copy()
    for step in range(steps):
        forces = np.zeros_like(pts)
        for i in range(N):
            # Boundary repulsion
            for dim in range(2):
                if pts[i, dim] < 0.06: forces[i, dim] += 0.04
                elif pts[i, dim] > 0.94: forces[i, dim] -= 0.04
                
            # Pairwise repulsion
            for j in range(i + 1, N):
                diff = pts[j] - pts[i]
                d = np.linalg.norm(diff)
                if d < 0.35 and d > 1e-6:
                    f = 0.006 / (d**2 + 0.001)
                    fx, fy = f * diff / d
                    forces[i] -= [fx, fy]
                    forces[j] += [fx, fy]
                    
        lr = 0.06 * (1.0 - step / steps)
        pts += forces * lr
        pts = np.clip(pts, 0.01, 0.99)
    return pts

def project_x(x0):
    """Ensure optimization vector strictly respects bounds."""
    x0 = x0.copy()
    for i in range(N):
        r = max(1e-8, x0[3 * i + 2])
        x0[3 * i] = np.clip(x0[3 * i], r, 1.0 - r)
        x0[3 * i + 1] = np.clip(x0[3 * i + 1], r, 1.0 - r)
        x0[3 * i + 2] = r
    return x0

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    best_sum = -1.0
    best_x = None
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraint_fun}
    
    # Phase 1: Generate diverse initial configurations
    inits = []

            pts = []
            y = r0
            row = 0
            while len(pts) < N + 10:
                x = r0 if row % 2 == 0 else 2.0 * r0
                while x <= 1.0 - r0:
                    pts.append([x, y])
                    x += 2.0 * r0
                y += np.sqrt(3.0) * r0
                row += 1
            pts = np.array(pts[:N + 10])
            if ang != 0.0:
                c_a, s_a = np.cos(ang), np.sin(ang)
                pts = (pts - 0.5) @ np.array([[c_a, -s_a], [s_a, c_a]]) + 0.5
            mask = (pts[:, 0] > 0.02) & (pts[:, 0] < 0.98) & (pts[:, 1] > 0.02) & (pts[:, 1] < 0.98)
            pts = pts[mask]
            while len(pts) < N:
                pts = np.vstack([pts, np.random.uniform(0.15, 0.85, (1, 2))])
            inits.append(pts[:N])

    # 2. Force-directed spreads from random seeds
    for s in range(8):
        np.random.seed(s + 100)
        pts = np.random.uniform(0.15, 0.85, (N, 2))
        pts = force_spread(pts, steps=400)
        inits.append(pts)

    # 3. Corner/Edge biased layouts
    for s in range(6):
        np.random.seed(200 + s)
        pts = np.random.uniform(0.05, 0.95, (N, 2))
        pts[:8] = np.random.uniform(0.05, 0.25, (8, 2))
        pts[8:16] = np.random.uniform(0.75, 0.95, (8, 2))
        pts = force_spread(pts, steps=500)
        inits.append(pts)

    # Phase 2: Initial Optimization with LP-SLSQP alternation
    for pts in inits:

        x0[1::3] = pts[:, 1]

            if not np.isnan(res.fun):
                s_val = -res.fun
                c_val = constraint_fun(res.x)

                    best_sum = s_val
                    best_x = res.x.copy()
        except Exception:
            pass

    # Phase 3: Aggressive Topology Search & Refinement

            if cycle % 5 == 0:

                x0[3*i_sw+1], x0[3*j_sw+1] = x0[3*j_sw+1], x0[3*i_sw+1]
                
            x0 = project_x(x0)
            
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 10000, 'ftol': 1e-14, 'disp': False})

                    if s_val > best_sum:
                        best_sum = s_val
                        best_x = res.x.copy()

                    c_tmp = np.column_stack((best_x[0::3], best_x[1::3]))
                    r_new = solve_lp_radii(c_tmp)

                        best_x[2::3] = r_new * 0.999
                        best_sum = np.sum(best_x[2::3])
                        
                        # Secondary SLSQP pass to fine-tune centers with slightly deflated radii
                        x0_fix = best_x.copy()
                        x0_fix[2::3] *= 0.985

                            res2 = minimize(objective, x0_fix, method='SLSQP', bounds=bounds, constraints=cons,
                                            options={'maxiter': 6000, 'ftol': 1e-14, 'disp': False})
                            if not np.isnan(res2.fun) and -res2.fun > best_sum:

                                best_x = res2.x.copy()

                                if r_new2 is not None:

                                    best_sum = np.sum(r_new2)

                pass

    return centers, radii, float(np.sum(radii))
