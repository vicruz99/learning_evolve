import numpy as np
from scipy.optimize import minimize, linprog

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A = np.zeros((num_ineq, n))
    b = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A[idx, i] = 1.0
        b[idx] = max(0.0, lim)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = max(0.0, d)
            idx += 1
            
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0.0, None)] * n, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return np.full(n, 0.05)

def evaluate_sum_radii(centers):
    """Evaluate maximum sum of radii for given centers via LP."""
    r = solve_lp_radii(centers)
    return np.sum(r)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Returns all inequality constraints >= 0 (vectorized)."""
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    c = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dr = r[:, None] + r[None, :]
    
    c = np.concatenate([c, dx[TRIL_IDX] ** 2 + dy[TRIL_IDX] ** 2 - dr[TRIL_IDX] ** 2])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-8, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-8, 0.5)])
    return b

def project_x(x0):
    """Ensure optimization vector strictly respects bounds."""
    x0 = x0.copy()
    r = np.maximum(x0[2::3], 1e-8)
    x0[0::3] = np.clip(x0[0::3], r, 1.0 - r)
    x0[1::3] = np.clip(x0[1::3], r, 1.0 - r)
    x0[2::3] = r
    return x0

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    best_centers = None
    best_sum = -1.0
    bounds = get_bounds()
    
    # Generate diverse initial center configurations
    inits = []
    
    # 1. Hexagonal lattices with varied rotations
    for ang in np.linspace(-0.4, 0.4, 15):
        pts = []
        y = 0.09
        row = 0
        while len(pts) < N + 6:
            x = 0.09 if row % 2 == 0 else 0.18
            while x <= 0.91:
                pts.append([x, y])
                x += 0.18
            y += 0.09 * np.sqrt(3.0)
            row += 1
        pts = np.array(pts[:N])
        if ang != 0.0:
            c, s = np.cos(ang), np.sin(ang)
            pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
        pts = np.clip(pts, 0.01, 0.99)
        inits.append(pts)
        
    # 2. Force-directed spreads for non-lattice starts
    for s in range(8):
        np.random.seed(s)
        pts = np.random.uniform(0.15, 0.85, (N, 2))
        for _ in range(150):
            f = np.zeros_like(pts)
            for i in range(N):
                for j in range(i + 1, N):
                    dx = pts[j] - pts[i]
                    d = np.linalg.norm(dx)
                    if d < 0.25 and d > 1e-5:
                        fp = 0.008 / (d ** 2 + 0.001)
                        f[i] -= dx * fp / d
                        f[j] += dx * fp / d
                for k in range(2):
                    if pts[i, k] < 0.1: f[i, k] += 0.03
                    elif pts[i, k] > 0.9: f[i, k] -= 0.03
            pts += f * 0.05
            pts = np.clip(pts, 0.02, 0.98)
        inits.append(pts)

    # Simulated Annealing on centers with LP evaluation
    temperature = 0.06
    
    for init_idx, pts in enumerate(inits):
        curr_centers = pts.copy()
        curr_sum = evaluate_sum_radii(curr_centers)
        
        # Initial SLSQP polish to find local peak quickly
        x0 = np.zeros(3 * N)
        x0[0::3] = curr_centers[:, 0]
        x0[1::3] = curr_centers[:, 1]
        r_lp = solve_lp_radii(curr_centers)
        x0[2::3] = np.maximum(r_lp * 0.995, 0.01)
        x0 = project_x(x0)
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,

            if not np.isnan(res.fun):

            pass

            if step % 100 == 0:

                new_centers[[i_sw, j_sw]] = new_centers[[j_sw, i_sw]]

            delta = new_sum - best_sum_run

                best_in_run = new_centers
                best_sum_run = new_sum
                
            temp *= 0.997
            
            # Periodic SLSQP refinement to climb out of flat regions
            if step % 200 == 0 and step > 0:
                x0 = np.zeros(3 * N)
                x0[0::3] = best_in_run[:, 0]
                x0[1::3] = best_in_run[:, 1]
                r_lp = solve_lp_radii(best_in_run)
                x0[2::3] = np.maximum(r_lp * 0.99, 0.005)
                x0 = project_x(x0)
                try:
                    res = minimize(objective, x0, method='SLSQP', bounds=bounds, 
                                   constraints={'type': 'ineq', 'fun': constraints}, 
                                   options={'maxiter': 5000, 'ftol': 1e-12})
                    if not np.isnan(res.fun):
                        best_in_run = np.column_stack((res.x[0::3], res.x[1::3]))
                        r_lp = solve_lp_radii(best_in_run)
                        best_sum_run = np.sum(r_lp)
                except Exception:
                    pass
                    
        if best_sum_run > best_sum:
            best_sum = best_sum_run
            best_centers = best_in_run.copy()
            
    # Final global SLSQP polish
    x0 = np.zeros(3 * N)
    x0[0::3] = best_centers[:, 0]
    x0[1::3] = best_centers[:, 1]
    r_lp = solve_lp_radii(best_centers)
    x0[2::3] = r_lp * 0.998
    x0 = project_x(x0)
    
    try:
        res = minimize(objective, x0, method='SLSQP', bounds=bounds, 
                       constraints={'type': 'ineq', 'fun': constraints}, 
                       options={'maxiter': 20000, 'ftol': 1e-13})
        if not np.isnan(res.fun):
            best_centers = np.column_stack((res.x[0::3], res.x[1::3]))
    except Exception:
        pass
        
    best_radii = solve_lp_radii(best_centers)
    
    # Strict validation repair to guarantee compliance with validator tolerances
    centers = best_centers
    radii = best_radii
    for _ in range(100):
        valid = True
        for i in range(N):
            if radii[i] < 0 or centers[i, 0] < radii[i] - 1e-9 or centers[i, 0] > 1.0 - radii[i] + 1e-9 or \

                valid = False
                break
        if valid:

                for j in range(i + 1, N):
                    d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                    if d < radii[i] + radii[j] - 1e-9:

                        break
                if not valid:

        if valid:
            break
        radii *= 0.9995
        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)
        
    return centers, radii, float(np.sum(radii))
