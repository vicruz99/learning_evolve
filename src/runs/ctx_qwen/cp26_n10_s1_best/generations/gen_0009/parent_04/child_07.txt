import numpy as np
from scipy.optimize import minimize, linprog
import warnings
warnings.filterwarnings('ignore')

N = 26
TRIL_IDX = np.tril_indices(N, -1)


def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])


def constraints_sq(x):
    """Vectorized inequality constraints with squared distances >= 0."""
    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]
    
    c = np.concatenate([
        xs - rs, 1.0 - xs - rs,
        ys - rs, 1.0 - ys - rs
    ])
    
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    c = np.concatenate([c, dx[TRIL_IDX]**2 + dy[TRIL_IDX]**2 - dr[TRIL_IDX]**2])
    return c


def constraints_lin(x):
    """Vectorized inequality constraints with linear distances >= 0."""
    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]
    
    c = np.concatenate([
        xs - rs, 1.0 - xs - rs,
        ys - rs, 1.0 - ys - rs
    ])
    
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    dists = np.hypot(dx[TRIL_IDX], dy[TRIL_IDX])
    c = np.concatenate([c, dists - dr[TRIL_IDX]])
    return c


def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-8, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-8, 0.5)])
    return b


def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    for i in range(n):
        bound = min(centers[i, 0], 1.0 - centers[i, 0],
                    centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, bound)
        idx += 1
        
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0],
                            centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist)
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-8)
    except Exception:
        pass
    return np.full(n, 0.01)


def project_x(x0):
    """Project variables to feasible bounds."""
    x0 = x0.copy()
    for i in range(N):
        r = max(1e-8, x0[3 * i + 2])
        x0[3 * i] = np.clip(x0[3 * i], r, 1.0 - r)
        x0[3 * i + 1] = np.clip(x0[3 * i + 1], r, 1.0 - r)
        x0[3 * i + 2] = r
    return x0


def force_spread(centers, steps=500, target_sep=0.22):
    """Force-directed relaxation to spread points."""
    pts = centers.copy()
    for step in range(steps):
        forces = np.zeros_like(pts)
        lr = 0.04 * max(0.01, 1.0 - step / steps)
        
        for i in range(N):
            for dim in range(2):
                if pts[i, dim] < 0.06:
                    forces[i, dim] += 0.06
                elif pts[i, dim] > 0.94:
                    forces[i, dim] -= 0.06
            
            for j in range(i + 1, N):
                diff = pts[j] - pts[i]
                d = np.linalg.norm(diff)
                if d < target_sep and d > 1e-6:
                    f = 0.01 / (d * d + 0.0005)
                    fx, fy = f * diff[0] / d, f * diff[1] / d
                    forces[i] -= [fx, fy]
                    forces[j] += [fx, fy]
        
        pts += forces * lr
        pts = np.clip(pts, 0.01, 0.99)
    return pts


def make_hex_init(r0, angle):
    """Generate rotated hexagonal lattice initialization."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 12:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0 and len(pts) < N + 12:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
    
    pts = np.array(pts[:N + 12])
    
    if angle != 0.0:
        c_a, s_a = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c_a, -s_a], [s_a, c_a]]) + 0.5
    
    mask = (pts[:, 0] >= 0.01) & (pts[:, 0] <= 0.99) & \
           (pts[:, 1] >= 0.01) & (pts[:, 1] <= 0.99)
    pts = pts[mask]
    
    while len(pts) < N:
        pts = np.vstack([pts, [np.random.uniform(0.15, 0.85), np.random.uniform(0.15, 0.85)]])
    return pts[:N]


def make_corner_init(r_corner=0.105):

    pts = np.zeros((N, 2))
    pts[0] = [r_corner, r_corner]
    pts[1] = [1 - r_corner, r_corner]

    pts[6] = [r_edge, 0.5]
    pts[7] = [1 - r_edge, 0.5]
    
    idx = 8

            pts[idx] = [0.15 + j * 0.18, 0.15 + i * 0.17]
            idx += 1
    
    for i in range(idx, N):
        pts[i] = [np.random.uniform(0.15, 0.85), np.random.uniform(0.15, 0.85)]
    
    pts += np.random.uniform(-0.003, 0.003, pts.shape)
    return np.clip(pts, 0.02, 0.98)


def make_nested_hex_init():
    """Generate nested hexagonal pattern."""
    pts = []
    # Outer hexagonal ring
    r_outer = 0.12
    for angle in np.linspace(0, 2*np.pi, 8, endpoint=False):
        pts.append([0.5 + r_outer * np.cos(angle), 0.5 + r_outer * np.sin(angle)])
    
    # Inner hexagonal ring
    r_inner = 0.06
    for angle in np.linspace(0, 2*np.pi, 6, endpoint=False):
        pts.append([0.5 + r_inner * np.cos(angle), 0.5 + r_inner * np.sin(angle)])
    
    # Corner circles
    r_corner = 0.10
    pts.append([r_corner, r_corner])
    pts.append([1 - r_corner, r_corner])
    pts.append([r_corner, 1 - r_corner])
    pts.append([1 - r_corner, 1 - r_corner])
    
    # Fill remaining
    while len(pts) < N:
        x = np.random.uniform(0.15, 0.85)
        y = np.random.uniform(0.15, 0.85)
        pts.append([x, y])
    
    return np.array(pts[:N])
