from scipy.optimize import minimize

N_CIRCLES = 26

def objective(vars):
    """Objective to minimize: negative sum of radii."""
    return -np.sum(vars[2::3])

def constraint_fun(vars):
    """Returns vector of inequality constraints g(vars) >= 0."""
    n = N_CIRCLES
    xs = vars[0::3]
    ys = vars[1::3]
    rs = vars[2::3]
    
    # Boundary constraints: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    c = np.concatenate([
        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs
    ])
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    # Using squared distance avoids sqrt singularity and improves gradient behavior
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    i_idx, j_idx = np.triu_indices(n, 1)
    c = np.concatenate([c, dx[i_idx, j_idx]**2 + dy[i_idx, j_idx]**2 - dr[i_idx, j_idx]**2])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-6, 0.5]."""
    bounds = []
    for _ in range(N_CIRCLES):
        bounds.extend([(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)])
    return bounds

    np.random.seed(seed)
    pts = []
    y = r0
    row = 0
    # Generate extra points to allow rotation filtering

            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        if y > 1.0 + r0:
            break
            
    pts = np.array(pts[:N_CIRCLES + 10])
    
    # Rotate around center
    if angle != 0.0:

    x0[1::3] = centers[:, 1]
    x0[2::3] = radii

    x0 += np.random.normal(0, 0.002, 3 * N_CIRCLES)

    x0[0::3] = np.clip(x0[0::3], r_p, 1.0 - r_p)
    x0[1::3] = np.clip(x0[1::3], r_p, 1.0 - r_p)
    x0[2::3] = r_p
    
    return x0
