import numpy as np
from scipy.optimize import minimize, linprog

N = 26
TRIU_IDX = np.triu_indices(N, 1)

def unpack(vars):
    """Converts flat optimization vector to centers and radii arrays."""
    return vars.reshape(N, 3)

def objective(vars):
    """Objective to minimize: negative sum of radii."""
    return -np.sum(vars[2::3])

def constraints(vars):
    """Returns vector of inequality constraints g(vars) >= 0."""
    xv, yv, rv = vars[0::3], vars[1::3], vars[2::3]
    n_cons = 4 * N + N * (N - 1) // 2
    c = np.empty(n_cons)
    
    # Boundary constraints
    c[:N] = xv - rv
    c[N:2*N] = 1.0 - xv - rv
    c[2*N:3*N] = yv - rv
    c[3*N:4*N] = 1.0 - yv - rv
    
    # Overlap constraints: dist >= r_i + r_j
    dx = xv[:, None] - xv[None, :]
    dy = yv[:, None] - yv[None, :]
    dist = np.sqrt(dx**2 + dy**2)
    r_sum = rv[:, None] + rv[None, :]
    c[4*N:] = dist[TRIU_IDX] - r_sum[TRIU_IDX]
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [0, 0.5]."""
    return [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N

def hex_init(r0, angle_deg, seed=None):
    """Generates a hexagonal lattice initialization, optionally rotated."""
    if seed is not None:
        np.random.seed(seed)
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        if y > 1.0 + r0:
            break
            
    pts = np.array(pts[:N + 10])
    
    if angle_deg != 0:
        rad = np.deg2rad(angle_deg)
        cos_a, sin_a = np.cos(rad), np.sin(rad)
        center = [0.5, 0.5]
        dx_pts = pts[:, 0] - center[0]
        dy_pts = pts[:, 1] - center[1]
        pts[:, 0] = dx_pts * cos_a - dy_pts * sin_a + center[0]
        pts[:, 1] = dx_pts * sin_a + dy_pts * cos_a + center[1]
        
    # Filter points strictly inside the square
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    
    # Pad if rotation/filtering removed too many points
    while len(pts) < N:
        pts = np.vstack([pts, [np.random.uniform(0.1, 0.9), np.random.uniform(0.1, 0.9)]])
        
    pts = pts[:N] + np.random.normal(0, 0.003, (N, 2))
    return pts

def solve_lp_radii(centers):
    """Optimally scale radii for fixed centers using Linear Programming."""
    n = N
    c = -np.ones(n)  # Maximize sum of radii -> minimize negative sum
    
    num_ineq = 4 * n + n * (n - 1) // 2
    A = np.zeros((num_ineq, n))
    b = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        A[idx, i] = 1.0
        b[idx] = centers[i, 0]
        idx += 1
        
        A[idx, i] = 1.0
        b[idx] = 1.0 - centers[i, 0]
        idx += 1
        
        A[idx, i] = 1.0
        b[idx] = centers[i, 1]
        idx += 1
        
        A[idx, i] = 1.0
        b[idx] = 1.0 - centers[i, 1]
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = d
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return np.full(n, 0.05)

def is_strictly_valid(centers, radii, tol=1e-9):
    """Checks feasibility with a given tolerance."""
    if np.any(radii < -tol):
        return False
    if np.any(centers[:, 0] - radii < -tol) or np.any(centers[:, 0] + radii > 1.0 + tol):
        return False
    if np.any(centers[:, 1] - radii < -tol) or np.any(centers[:, 1] + radii > 1.0 + tol):
        return False
        
    dx = centers[:, 0][:, None] - centers[:, 0][None, :]
    dy = centers[:, 1][:, None] - centers[:, 1][None, :]
    dist = np.sqrt(dx**2 + dy**2)
    r_sum = radii[:, None] + radii[None, :]
    return np.all(dist[TRIU_IDX] >= r_sum[TRIU_IDX] - tol)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    best_sum = -1.0
    best_vars = None
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    
    # Phase 1: Diverse structural starts
    starts = []
    for r0 in [0.085, 0.09, 0.095, 0.10]:
        for ang in [0, 10, 15, 20, 30, 45]:
            starts.append(hex_init(r0, ang))
            
    np.random.seed(42)
    for _ in range(10):
        starts.append(np.random.uniform(0.15, 0.85, (N, 2)))
        
    for c0 in starts:
        # Start with small radii to guarantee initial feasibility for SLSQP
        r0 = np.full(N, 0.04)
        x0 = np.zeros(3 * N)
        x0[0::3] = c0[:, 0]
        x0[1::3] = c0[:, 1]
        x0[2::3] = r0
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 15000, 'ftol': 1e-14, 'disp': False})
            if not np.isnan(res.fun):
                # LP refinement guarantees optimal radii for the converged centers
                c_opt = res.x[:2 * N].reshape(N, 2)
                r_opt = solve_lp_radii(c_opt)
                curr_sum = np.sum(r_opt)
                
                if is_strictly_valid(c_opt, r_opt) and curr_sum > best_sum:
                    best_sum = curr_sum
                    best_vars = np.concatenate([c_opt.flatten(), r_opt])
        except Exception:
            continue
            
    # Phase 2: Iterative refinement to escape local minima
    if best_vars is not None:
        for step in range(20):
            noise_scale = 0.002 / (step + 1)
            x_pert = best_vars + np.random.normal(0, noise_scale, 3 * N)
            
            # Project perturbed variables back to feasible bounds
            r_p = np.maximum(x_pert[2::3], 0.01)
            x_pert[0::3] = np.clip(x_pert[0::3], r_p, 1.0 - r_p)
            x_pert[1::3] = np.clip(x_pert[1::3], r_p, 1.0 - r_p)
            x_pert[2::3] = r_p
            
            try:
                res = minimize(objective, x_pert, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 10000, 'ftol': 1e-14, 'disp': False})
                if not np.isnan(res.fun):
                    c_opt = res.x[:2 * N].reshape(N, 2)
                    r_opt = solve_lp_radii(c_opt)
                    s = np.sum(r_opt)
                    if is_strictly_valid(c_opt, r_opt) and s > best_sum:
                        best_sum = s
                        best_vars = np.concatenate([c_opt.flatten(), r_opt])
            except Exception:
                pass
                
    # Fallback (should rarely be reached)
    if best_vars is None:
        best_vars = np.zeros(3 * N)
        best_vars[2::3] = 0.06
        best_vars[0::3] = np.tile(np.linspace(0.1, 0.9, 5), 6)[:N]
        best_vars[1::3] = np.repeat(np.linspace(0.1, 0.9, 6), 5)[:N]
        
    centers = best_vars[:2 * N].reshape(N, 2)
    radii = best_vars[2 * N:]
    
    # Final strict validity adjustment against 1e-12 tolerance
    for _ in range(50):
        if is_strictly_valid(centers, radii, tol=1e-12):
            break
        radii *= 0.9995
        
    return centers, radii, float(np.sum(radii))
