import numpy as np
from scipy.optimize import minimize, linprog

N_CIRCLES = 26
UP_INDICES = np.triu_indices(N_CIRCLES, 1)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Vectorized inequality constraints: g(x) >= 0."""
    xs, ys, rs = x[0::3], x[1::3], x[2::3]
    
    # Boundary constraints
    c = np.concatenate([
        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs
    ])
    
    # Overlap constraints (squared for smoother gradients)
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    c = np.concatenate([c, dx[UP_INDICES]**2 + dy[UP_INDICES]**2 - dr[UP_INDICES]**2])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [0, 0.5]."""
    b = []
    for _ in range(N_CIRCLES):
        b.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
    return b

def get_safe_radii(centers):
    """Compute strictly feasible initial radii based on nearest neighbors/walls."""
    n = centers.shape[0]
    radii = np.zeros(n)
    for i in range(n):
        min_d = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
        for j in range(n):
            if i != j:
                d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
                if d < min_d:
                    min_d = d
        radii[i] = min_d * 0.45
    return radii

def solve_radii_lp(centers):
    """Optimally scale radii for fixed centers using Linear Programming."""
    n = N_CIRCLES
    c_obj = -np.ones(n)  # Maximize sum of radii
    
    num_pairs = n * (n - 1) // 2
    A_ub = np.zeros((n + num_pairs, n))
    b_ub = np.zeros(n + num_pairs)
    
    idx = 0
    # Boundary constraints: r_i <= min(dist to walls)
    for i in range(n):
        A_ub[idx, i] = 1.0
        b_ub[idx] = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dist
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -1.0
    best_x = None
    
    # Diverse hexagonal lattice patterns
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 6, 6, 5, 4],
        [4, 6, 6, 6, 4], [5, 5, 5, 5, 6], [6, 6, 6, 4, 4],
        [5, 7, 5, 6, 3], [6, 6, 5, 5, 4], [4, 5, 6, 5, 6],
        [6, 5, 5, 6, 4], [5, 5, 6, 5, 5], [5, 6, 4, 6, 5]
    ]
    
    inits = []
    for pat in patterns:
        centers = np.zeros((N_CIRCLES, 2))
        idx = 0
        y = 0.085
        y_step = 0.085 * np.sqrt(3)
        for row, count in enumerate(pat):
            x_start = 0.085 + (0.085 if row % 2 == 1 else 0.0)
            for k in range(count):
                if idx < N_CIRCLES:
                    centers[idx] = [x_start + k * 0.17, y]
                    idx += 1
            y += y_step
        while idx < N_CIRCLES:
            centers[idx] = [0.5, 0.5]
            idx += 1
            
        inits.append(centers.copy())
        # Add rotated variants to break symmetry
        for ang in [0.1, -0.1, 0.2, -0.2, 0.4, 0.7, 1.0]:
            c = centers.copy()
            c -= [0.5, 0.5]
            ca, sa = np.cos(ang), np.sin(ang)
            c = np.dot(c, [[ca, -sa], [sa, ca]])
            c += [0.5, 0.5]
            inits.append(c)

    # Add random starts
    np.random.seed(42)
    for _ in range(15):
        inits.append(np.random.uniform(0.2, 0.8, (N_CIRCLES, 2)))

    # Phase 1: Multi-start SLSQP optimization
    for c_init in inits:
        r_init = get_safe_radii(c_init)
        x0 = np.zeros(3 * N_CIRCLES)
        x0[0::3] = c_init[:, 0]
        x0[1::3] = c_init[:, 1]
        x0[2::3] = r_init
        
        # Perturb slightly to avoid exact symmetry
        x0 += np.random.normal(0, 1e-4, x0.shape)
        x0[2::3] = np.clip(x0[2::3], 0.01, 0.5)
        x0[0::3] = np.clip(x0[0::3], 0.01, 0.99)
        x0[1::3] = np.clip(x0[1::3], 0.01, 0.99)
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 15000, 'ftol': 1e-14})
            if not np.isnan(res.fun) and -res.fun > best_sum:
                best_sum = -res.fun
                best_x = res.x.copy()
        except Exception:
            continue

                best_x[2::3] = new_r
                curr_sum = np.sum(new_r)
                if curr_sum > best_sum:
                    best_sum = curr_sum
                    
                # Re-optimize positions with updated optimal radii
                x0 = best_x + np.random.normal(0, 1e-5, best_x.shape)
                try:
                    res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                                   constraints=cons, options={'maxiter': 8000, 'ftol': 1e-14})
                    if not np.isnan(res.fun) and -res.fun > best_sum:
                        best_sum = -res.fun
                        best_x = res.x.copy()
                except Exception:
                    pass

    # Phase 3: Perturbation-based local search
    if best_x is not None:
        for _ in range(10):
            noise = 0.002 * np.random.randn(3 * N_CIRCLES)
            x0 = best_x + noise
            r_pert = np.maximum(x0[2::3], 0.01)
            x0[0::3] = np.clip(x0[0::3], r_pert, 1.0 - r_pert)
            x0[1::3] = np.clip(x0[1::3], r_pert, 1.0 - r_pert)
            x0[2::3] = r_pert
            
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 5000, 'ftol': 1e-14})
                if not np.isnan(res.fun) and -res.fun > best_sum:
                    best_sum = -res.fun
                    best_x = res.x.copy()
            except Exception:
                pass

    # Fallback
    if best_x is None:
        best_x = np.zeros(3 * N_CIRCLES)
        best_x[2::3] = 0.06
        best_x[0::3] = np.tile(np.linspace(0.15, 0.85, 5), 6)[:N_CIRCLES]
        best_x[1::3] = np.repeat(np.linspace(0.15, 0.85, 6), 5)[:N_CIRCLES]
        
    centers = best_x.reshape(N_CIRCLES, 3)[:, :2]
    radii = best_x.reshape(N_CIRCLES, 3)[:, 2]
    
    # Final strict validity adjustment against 1e-12 tolerance
    for _ in range(100):
        valid = True
        for i in range(N_CIRCLES):
            if radii[i] < 0: valid=False; break
            if centers[i,0]-radii[i] < -1e-12 or centers[i,0]+radii[i] > 1.0+1e-12: valid=False; break
            if centers[i,1]-radii[i] < -1e-12 or centers[i,1]+radii[i] > 1.0+1e-12: valid=False; break
        if valid:
            for i in range(N_CIRCLES):
                for j in range(i+1, N_CIRCLES):
                    if np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1]) < radii[i]+radii[j]-1e-12:
                        valid=False; break
                if not valid: break
        if valid: break
        radii *= 0.9995
        
    return centers, radii, float(np.sum(radii))
