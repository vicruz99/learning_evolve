To improve the packing of 26 circles in a unit square, we will move beyond standard SLSQP multi-start optimization, which often gets trapped in local minima due to the highly non-convex constraint landscape. The strategy leverages three key components:

1. **Linear Programming for Fixed Centers**: For any given set of center coordinates, maximizing the sum of radii subject to non-overlap and boundary constraints is a Linear Program (LP). Solving this LP exactly gives the optimal radii for those centers, providing a precise, differentiable-free objective landscape for center optimization.
2. **Hybrid Initialization**: We generate diverse starting configurations using hexagonal lattices with varying row distributions, multiple rotation angles, and random perturbations. Each is immediately evaluated with the LP solver to guarantee feasible, tight initial radii.
3. **Alternating Optimization & Local Search**: 
   - **SLSQP Phase**: We first run smooth constrained optimization to jointly adjust centers and radii, using LP results as warm starts.
   - **LP-Driven Local Search**: After SLSQP converges, we perform a randomized hill-climbing on the *centers only*. For each perturbed center configuration, we solve the LP again. This effectively "shakes" the circles out of jammed positions while always maintaining mathematically optimal radii for the current layout.
   - **Deflation-Reinflation**: We slightly shrink the best radii, allow SLSQP to re-optimize positions (unlocking new spatial arrangements), and then re-expand radii via LP. This breaks symmetry and escapes shallow local minima.

This combination of exact LP radius computation, geometric diversity, and targeted deflation consistently pushes past standard gradient-based limits, targeting the 2.636+ sum of radii.

import numpy as np
from scipy.optimize import minimize, linprog

N_CIRCLES = 26
TRIL_IDX = np.tril_indices(N_CIRCLES, -1)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Returns vector of inequality constraints g(x) >= 0."""
    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]
    
    n_cons = 4 * N_CIRCLES + N_CIRCLES * (N_CIRCLES - 1) // 2
    c = np.empty(n_cons)
    
    # Boundary constraints
    c[:N_CIRCLES] = xs - rs
    c[N_CIRCLES:2*N_CIRCLES] = 1.0 - xs - rs
    c[2*N_CIRCLES:3*N_CIRCLES] = ys - rs
    c[3*N_CIRCLES:4*N_CIRCLES] = 1.0 - ys - rs
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    c[4*N_CIRCLES:] = dx[TRIL_IDX]**2 + dy[TRIL_IDX]**2 - dr[TRIL_IDX]**2
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-6, 0.5]."""
    b = []
    for _ in range(N_CIRCLES):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)])
    return b

def solve_lp_radii(centers):
    """Maximizes sum of radii for fixed centers using Linear Programming."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A = []
    b = []
    
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        bound = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
        row = np.zeros(n)
        row[i] = 1.0
        A.append(row)
        b.append(bound)
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i+1, n):
            d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A.append(row)
            b.append(d)
            
    bounds_lp = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=np.array(A), b_ub=np.array(b), bounds=bounds_lp, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return np.full(n, 0.01)

def generate_hex_init(row_counts, r0=0.09, angle=0.0):
    """Generates a hexagonal lattice initialization with optional rotation."""
    centers = np.zeros((N_CIRCLES, 2))
    idx = 0
    y = r0
    y_step = np.sqrt(3) * r0
    
    for i, cnt in enumerate(row_counts):
        x_start = r0 + (r0 if i % 2 == 1 else 0.0)
        for k in range(cnt):
            if idx < N_CIRCLES:
                centers[idx, 0] = x_start + k * 2 * r0
                centers[idx, 1] = y
                idx += 1
        y += y_step
        
    # Fill remaining if any
    while idx < N_CIRCLES:
        centers[idx] = np.random.uniform(r0, 1.0-r0, 2)
        idx += 1
        
    # Rotate around center
    if angle != 0.0:
        cx, cy = 0.5, 0.5
        ca, sa = np.cos(angle), np.sin(angle)
        dx = centers[:, 0] - cx
        dy = centers[:, 1] - cy
        centers[:, 0] = dx * ca - dy * sa + cx
        centers[:, 1] = dx * sa + dy * ca + cy
        
    centers = np.clip(centers, 0.01, 0.99)
    return centers

def local_search_on_centers(centers0, steps=400):
    """Gradient-free local search optimizing centers via LP radius evaluation."""
    centers = centers0.copy()
    radii = solve_lp_radii(centers)
    best_sum = np.sum(radii)
    step_size = 0.008
    
    for _ in range(steps):
        trial = centers + np.random.normal(0, step_size, centers.shape)
        trial = np.clip(trial, 0.01, 0.99)
        r_trial = solve_lp_radii(trial)
        s_trial = np.sum(r_trial)
        
        if s_trial > best_sum:
            centers = trial
            radii = r_trial
            best_sum = s_trial
            step_size = min(0.015, step_size * 1.05)
        else:
            step_size *= 0.995
            
    return centers, radii, best_sum

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Diverse hexagonal patterns summing to 26
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 6, 6, 5, 4],
        [4, 6, 6, 6, 4], [5, 5, 5, 5, 6], [6, 6, 5, 5, 4]
    ]
    
    inits = []
    for pat in patterns:
        for ang in [0.0, 0.2, -0.2, 0.4, -0.4, 0.6]:
            c = generate_hex_init(pat, r0=0.088, angle=ang)
            c += np.random.uniform(-0.002, 0.002, c.shape)
            inits.append(c)
            
    np.random.seed(42)
    for _ in range(15):
        c = np.random.uniform(0.15, 0.85, (N_CIRCLES, 2))
        inits.append(c)
        
    # Phase 1: SLSQP optimization from diverse starts
    for c_init in inits:
        r_init = solve_lp_radii(c_init)
        x0 = np.zeros(3 * N_CIRCLES)
        x0[0::3] = c_init[:, 0]
        x0[1::3] = c_init[:, 1]
        x0[2::3] = r_init
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 6000, 'ftol': 1e-12, 'disp': False})
            if not np.isnan(res.fun):
                c_opt = np.column_stack((res.x[0::3], res.x[1::3]))
                r_opt = res.x[2::3]
                
                # Clean up numerical drift
                for i in range(N_CIRCLES):
                    r_opt[i] = max(0.0, r_opt[i])
                    c_opt[i, 0] = np.clip(c_opt[i, 0], r_opt[i], 1.0-r_opt[i])
                    c_opt[i, 1] = np.clip(c_opt[i, 1], r_opt[i], 1.0-r_opt[i])
                    
                curr_sum = np.sum(r_opt)
                if curr_sum > best_sum:
                    best_sum = curr_sum
                    best_centers = c_opt.copy()
                    best_radii = r_opt.copy()
        except Exception:
            continue
            
    # Phase 2: LP-driven local search to escape SLSQP local minima
    if best_centers is not None:
        for _ in range(3):
            c_ls, r_ls, s_ls = local_search_on_centers(best_centers, steps=300)
            if s_ls > best_sum:
                best_sum = s_ls
                best_centers = c_ls
                best_radii = r_ls
                
        # Phase 3: Deflation & re-optimize to break symmetry/jams
        c_def = best_centers.copy()
        r_def = best_radii * 0.985
        x_def = np.zeros(3 * N_CIRCLES)
        x_def[0::3] = c_def[:, 0]
        x_def[1::3] = c_def[:, 1]
        x_def[2::3] = r_def
        
        try:
            res = minimize(objective, x_def, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 8000, 'ftol': 1e-12, 'disp': False})
            if not np.isnan(res.fun):
                c_opt = np.column_stack((res.x[0::3], res.x[1::3]))
                r_opt = res.x[2::3]
                curr_sum = np.sum(r_opt)
                if curr_sum > best_sum:
                    best_sum = curr_sum
                    best_centers = c_opt
                    best_radii = r_opt
        except Exception:
            pass
            
    # Final validation and safety shrinkage against numerical tolerances
    centers = best_centers.copy()
    radii = best_radii.copy()
    
    for _ in range(50):
        valid = True
        for i in range(N_CIRCLES):
            if radii[i] < 0: valid=False; break
            if centers[i,0]-radii[i] < -1e-12 or centers[i,0]+radii[i] > 1.0+1e-12: valid=False; break
            if centers[i,1]-radii[i] < -1e-12 or centers[i,1]+radii[i] > 1.0+1e-12: valid=False; break
        if valid:
            for i in range(N_CIRCLES):
                for j in range(i+1, N_CIRCLES):
                    d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
                    if d < radii[i]+radii[j]-1e-12:
                        valid=False; break
                if not valid: break
        if valid: break
        radii *= 0.999
        
    return centers, radii, float(np.sum(radii))
