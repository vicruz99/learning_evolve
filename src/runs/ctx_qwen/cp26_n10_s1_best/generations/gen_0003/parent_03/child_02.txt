import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def compute_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)  # Maximize sum -> Minimize negative sum
    num_cons = n + n * (n - 1) // 2
    A_ub = np.zeros((num_cons, n))
    b_ub = np.zeros(num_cons)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        bound = min(centers[i, 0], 1.0 - centers[i, 0], 
                    centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = bound
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], 
                            centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dist
            idx += 1
            
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=(0.0, 0.5), method='highs')
    if res.success:
        # Slight shrinkage ensures strict feasibility for the subsequent NLP step
        return np.clip(res.x * 0.995, 1e-6, 0.5)
    return None

def nlp_objective(x):
    """Objective: maximize sum of radii."""
    return -np.sum(x[2::3])

def nlp_constraints(x):
    """Vectorized inequality constraints g(x) >= 0."""
    n = N
    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]
    
    n_boundary = 4 * n
    n_overlap = n * (n - 1) // 2
    c = np.empty(n_boundary + n_overlap)
    
    # Boundary constraints
    c[:n] = xs - rs
    c[n:2*n] = 1.0 - xs - rs
    c[2*n:3*n] = ys - rs
    c[3*n:4*n] = 1.0 - ys - rs
    
    # Overlap constraints (squared form for better numerical stability)
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    i_idx, j_idx = np.tril_indices(n, -1)
    c[4*n:] = dx[i_idx, j_idx]**2 + dy[i_idx, j_idx]**2 - dr[i_idx, j_idx]**2
    
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [0, 0.5]."""
    bounds = []
    for _ in range(N):
        bounds.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
    return bounds

def make_hex_init(r0, seed):
    """Generate hexagonal lattice initialization."""
    np.random.seed(seed)
    c = np.zeros((N, 2))
    idx = 0
    y = r0
    row = 0
    while idx < N and y + r0 <= 1.0:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while idx < N and x + r0 <= 1.0:
            c[idx] = [x, y]
            idx += 1
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
    while idx < N:
        c[idx] = np.random.uniform(0.2, 0.8, 2)
        idx += 1
    c += np.random.normal(0, 0.002, c.shape)
    return np.clip(c, r0, 1.0 - r0)

def force_init(seed):
    """Force-directed layout to spread points naturally."""
    np.random.seed(seed)
    c = np.random.uniform(0.15, 0.85, (N, 2))
    for _ in range(500):
        f = np.zeros_like(c)
        for i in range(N):
            for j in range(i + 1, N):
                d = c[j] - c[i]
                dist = np.linalg.norm(d)
                if dist < 0.25 and dist > 1e-6:
                    force = 0.005 / (dist**2)
                    f[i] -= force * d
                    f[j] += force * d
            if c[i, 0] < 0.1: f[i, 0] += 0.02
            elif c[i, 0] > 0.9: f[i, 0] -= 0.02
            if c[i, 1] < 0.1: f[i, 1] += 0.02
            elif c[i, 1] > 0.9: f[i, 1] -= 0.02
        c += f * 0.05
        c = np.clip(c, 0.05, 0.95)
    return c

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    bounds = get_bounds()
    best_val = -1.0
    best_x = None
    
    inits = []
    # Hexagonal lattices with varied densities
    for s in range(5):
        for r0 in [0.08, 0.09, 0.10]:
            inits.append(make_hex_init(r0, s))
    # Force-directed spreads
    for s in range(5):
        inits.append(force_init(s))
    # Rotated hex lattices to break symmetry
    for ang in [0.1, -0.1, 0.2, -0.2]:
        base = make_hex_init(0.09, 0)
        cx, cy = 0.5, 0.5
        cos_a, sin_a = np.cos(ang), np.sin(ang)
        rot = (base - [cx, cy]) @ np.array([[cos_a, -sin_a], [sin_a, cos_a]]) + [cx, cy]
        inits.append(rot)

    # Primary optimization loop
    for c_init in inits:
        x0 = np.zeros(3 * N)
        x0[0::3] = c_init[:, 0]
        x0[1::3] = c_init[:, 1]
        x0[2::3] = 0.05  # Start with small radii
        
        # Alternating LP (radii) + NLP (centers+radii) cycles
        for cyc in range(4):
            centers = np.column_stack((x0[0::3], x0[1::3]))
            r_lp = compute_lp_radii(centers)
            if r_lp is not None:
                x0[2::3] = r_lp
                
            try:
                res = minimize(nlp_objective, x0, method='SLSQP', bounds=bounds,
                               constraints={'type': 'ineq', 'fun': nlp_constraints},
                               options={'maxiter': 3000, 'ftol': 1e-13, 'disp': False})
                x0 = res.x.copy()
            except Exception:
                pass
                
            # Perturb centers to explore nearby configurations in next cycle
            if cyc < 3:
                noise = 0.005 / (cyc + 1)
                x0[0::3] += np.random.normal(0, noise, N)
                x0[1::3] += np.random.normal(0, noise, N)
                x0[0::3] = np.clip(x0[0::3], 0.01, 0.99)
                x0[1::3] = np.clip(x0[1::3], 0.01, 0.99)
                
        curr_sum = -nlp_objective(x0)
        if curr_sum > best_val:
            best_val = curr_sum
            best_x = x0.copy()
            
    # Final intensive refinement on the best found configuration
    if best_x is not None:
        x0 = best_x.copy()
        for _ in range(3):
            try:
                res = minimize(nlp_objective, x0, method='SLSQP', bounds=bounds,
                               constraints={'type': 'ineq', 'fun': nlp_constraints},
                               options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
                x0 = res.x.copy()
            except Exception:
                pass
        best_x = x0

    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3]
    
    # Strict safety shrinkage to guarantee compliance with 1e-12 validator tolerance
    for _ in range(100):
        valid = True
        for i in range(N):
            if radii[i] < 0: valid = False; break
            if centers[i, 0] - radii[i] < -1e-12 or centers[i, 0] + radii[i] > 1.0 + 1e-12: valid = False; break
            if centers[i, 1] - radii[i] < -1e-12 or centers[i, 1] + radii[i] > 1.0 + 1e-12: valid = False; break
        if valid:
            for i in range(N):
                for j in range(i + 1, N):
                    d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                    if d < radii[i] + radii[j] - 1e-12: valid = False; break
                if not valid: break
        if valid: break
        radii *= 0.999
        
    return centers, radii, float(np.sum(radii))
