import numpy as np
from scipy.optimize import minimize, linprog

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Returns vector of inequality constraints g(x) >= 0."""
    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]
    c = np.empty(4*N + N*(N-1)//2)
    c[:N] = xs - rs
    c[N:2*N] = 1.0 - xs - rs
    c[2*N:3*N] = ys - rs
    c[3*N:4*N] = 1.0 - ys - rs
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    c[4*N:] = dx[TRIL_IDX]**2 + dy[TRIL_IDX]**2 - dr[TRIL_IDX]**2
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [0, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
    return b

def solve_radii_lp(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    A_ub = np.zeros((n + n*(n-1)//2, n))
    b_ub = np.zeros(n + n*(n-1)//2)
    
    idx = 0
    for i in range(n):
        x, y = centers[i]
        b_ub[idx] = min(x, 1.0-x, y, 1.0-y)
        A_ub[idx, i] = 1.0
        idx += 1
        
    for i in range(n):
        for j in range(i+1, n):
            dist = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dist
            idx += 1
            
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def force_relax(centers, radii, steps=150):
    """Force-directed layout to separate overlapping circles and push to boundaries."""
    n = centers.shape[0]
    for _ in range(steps):
        fx, fy = np.zeros(n), np.zeros(n)
        for i in range(n):
            for j in range(i+1, n):
                dx = centers[j,0] - centers[i,0]
                dy = centers[j,1] - centers[i,1]
                d = np.hypot(dx, dy)
                if d < radii[i] + radii[j] and d > 1e-7:
                    f = (radii[i] + radii[j] - d) * 0.5 / d
                    fx[i] -= f * dx; fx[j] += f * dx
                    fy[i] -= f * dy; fy[j] += f * dy
            r = radii[i]
            if centers[i,0] < r: fx[i] += 0.1
            elif centers[i,0] > 1.0-r: fx[i] -= 0.1
            if centers[i,1] < r: fy[i] += 0.1
            elif centers[i,1] > 1.0-r: fy[i] -= 0.1
        centers += np.column_stack([fx, fy]) * 0.1
        centers = np.clip(centers, 0.0, 1.0)
    return centers

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    n = N
    best_sum = 0.0
    best_c, best_r = None, None
    bounds_list = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    
    inits = []
    # Hexagonal patterns with varying densities
    for r0 in [0.075, 0.08, 0.085, 0.09, 0.095, 0.10]:
        pts = []
        y = r0
        row = 0
        while len(pts) < n + 2:
            x = r0 if row % 2 == 0 else 2.0 * r0
            while x <= 1.0 - r0 and len(pts) < n + 2:
                pts.append([x, y])
                x += 2.0 * r0
            y += np.sqrt(3.0) * r0
            row += 1
        pts = np.array(pts[:n])
        inits.append(pts)
        # Rotated variants to break symmetry and explore different boundary interactions
        for ang in [0.15, -0.15, 0.3, -0.3, 0.45, -0.45, 0.6]:
            c_pt = np.array([0.5, 0.5])
            d = pts - c_pt
            c_a, s_a = np.cos(ang), np.sin(ang)
            rot = d @ np.array([[c_a, -s_a], [s_a, c_a]]) + c_pt
            inits.append(rot)
            
    # Random starts
    np.random.seed(42)
    for _ in range(25):
        inits.append(np.random.uniform(0.1, 0.9, (n, 2)))
        
    for c0 in inits:
        # Force relax to separate overlapping circles
        r0 = np.full(n, 0.06)
        c_relaxed = force_relax(c0.copy(), r0, steps=100)
        
        # LP for optimal radii given relaxed centers
        r_lp = solve_radii_lp(c_relaxed)
        if r_lp is not None:
            r0 = r_lp
            
        x0 = np.zeros(3*n)
        x0[0::3] = c_relaxed[:, 0]
        x0[1::3] = c_relaxed[:, 1]
        x0[2::3] = r0
        
        # SLSQP optimization
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds_list,
                           constraints=cons, options={'maxiter': 10000, 'ftol': 1e-14, 'disp': False})
            if not np.isnan(res.fun):
                curr_sum = -res.fun
                if curr_sum > best_sum:
                    best_sum = curr_sum
                    best_c = res.x[0::3].reshape(n, 2).copy()
                    best_r = res.x[2::3].copy()
        except Exception:
            pass
            
        # Iterative refinement: shrink radii, perturb centers, re-optimize to escape local minima
        if best_c is not None and curr_sum > 2.63:
            for _ in range(4):
                c_pert = best_c + np.random.normal(0, 0.004, best_c.shape)
                c_pert = np.clip(c_pert, 0.01, 0.99)
                r_pert = best_r * 0.985
                
                x_pert = np.zeros(3*n)
                x_pert[0::3] = c_pert[:, 0]
                x_pert[1::3] = c_pert[:, 1]
                x_pert[2::3] = r_pert
                
                try:
                    res2 = minimize(objective, x_pert, method='SLSQP', bounds=bounds_list,
                                    constraints=cons, options={'maxiter': 8000, 'ftol': 1e-14, 'disp': False})
                    if not np.isnan(res2.fun):
                        s2 = -res2.fun
                        if s2 > best_sum:
                            best_sum = s2
                            best_c = res2.x[0::3].reshape(n, 2).copy()
                            best_r = res2.x[2::3].copy()
                except Exception:
                    pass

    # Final strict validity adjustment to guarantee compliance with 1e-12 tolerance
    if best_c is not None:
        for _ in range(100):
            valid = True
            for i in range(n):
                if best_r[i] < 0 or best_c[i,0] < best_r[i] - 1e-11 or best_c[i,0] > 1.0 - best_r[i] + 1e-11 or \
                   best_c[i,1] < best_r[i] - 1e-11 or best_c[i,1] > 1.0 - best_r[i] + 1e-11:
                    valid = False
                    break
            if valid:
                for i in range(n):
                    for j in range(i+1, n):
                        d = np.hypot(best_c[i,0]-best_c[j,0], best_c[i,1]-best_c[j,1])
                        if d < best_r[i] + best_r[j] - 1e-11:
                            valid = False
                            break
                    if not valid:
                        break
            if valid:
                break
            best_r *= 0.9998

    # Fallback (should rarely be reached)
    if best_c is None:
        best_c, best_r = force_relax(np.random.uniform(0.1, 0.9, (n, 2)), np.full(n, 0.06))
        best_sum = np.sum(best_r)
        
    return best_c, best_r, float(np.sum(best_r))
