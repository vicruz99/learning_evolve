import numpy as np
from scipy.optimize import minimize, linprog

N_CIRCLES = 26
TRI_INDICES = np.triu_indices(N_CIRCLES, 1)

def get_lp_radii(centers):
    """Solve LP to find maximum sum of radii for fixed centers."""
    n = N_CIRCLES
    c_obj = -np.ones(n)
    n_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((n_ineq, n))
    b_ub = np.zeros(n_ineq)
    
    idx = 0
    for i in range(n):
        x, y = centers[i]
        b_ub[idx] = min(x, 1.0 - x, y, 1.0 - y)
        A_ub[idx, i] = 1.0
        idx += 1
        
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = d
            idx += 1
            
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
    if res.success:
        return np.maximum(res.x, 0.005)
    return np.full(n, 0.05)

def force_spread(centers, steps=300):
    """Vectorized force-directed layout to spread points evenly."""
    n = centers.shape[0]
    for _ in range(steps):
        forces = np.zeros_like(centers)
        
        dx = centers[:, None, :] - centers[None, :, :]
        dist = np.sqrt(np.sum(dx**2, axis=2))
        np.fill_diagonal(dist, 1e6)
        
        mask = dist < 0.3
        inv_dist_sq = np.where(mask, 0.005 / (dist**2 + 0.001), 0.0)
        
        fx = np.sum(inv_dist_sq * dx[:, :, 0], axis=1)
        fy = np.sum(inv_dist_sq * dx[:, :, 1], axis=1)
        forces[:, 0] -= fx
        forces[:, 1] -= fy
        
        for i in range(n):
            if centers[i, 0] < 0.1: forces[i, 0] += 0.02
            elif centers[i, 0] > 0.9: forces[i, 0] -= 0.02
            if centers[i, 1] < 0.1: forces[i, 1] += 0.02
            elif centers[i, 1] > 0.9: forces[i, 1] -= 0.02
            
        centers += forces * 0.05
        centers = np.clip(centers, 0.05, 0.95)
    return centers

def objective(vars):
    """Objective to minimize: negative sum of radii."""
    return -np.sum(vars[2::3])

def constraints(vars):
    """Returns vector of inequality constraints g(vars) >= 0."""
    n = N_CIRCLES
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    c = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist = np.sqrt(dx**2 + dy**2)
    r_sum = r[:, None] + r[None, :]
    
    c = np.concatenate([c, dist[TRI_INDICES] - r_sum[TRI_INDICES]])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [0, 0.5]."""
    b = []
    for _ in range(N_CIRCLES):
        b.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
    return b

def hex_init(rows, r_start=0.09):
    """Generates a hexagonal lattice initialization."""
    centers = np.zeros((N_CIRCLES, 2))
    idx = 0
    y = r_start
    y_step = np.sqrt(3) * r_start
    for i, cnt in enumerate(rows):
        x_start = r_start + (r_start if i % 2 == 1 else 0.0)
        for k in range(cnt):
            if idx < N_CIRCLES:
                x = x_start + k * 2 * r_start
                centers[idx] = [x, y]
                idx += 1
        y += y_step
    while idx < N_CIRCLES:
        centers[idx] = np.random.uniform(r_start, 1.0 - r_start, 2)
        idx += 1
    return centers

def rotate_points(centers, angle):
    """Rotates a grid of centers around the square center."""
    cx, cy = 0.5, 0.5
    c, s = np.cos(angle), np.sin(angle)
    dx = centers[:, 0] - cx
    dy = centers[:, 1] - cy
    return np.column_stack([dx * c - dy * s + cx, dx * s + dy * c + cy])

def is_strictly_valid(centers, radii, tol=1e-12):
    """Checks feasibility with a given tolerance."""
    n = centers.shape[0]
    if np.any(radii < -tol):
        return False
    if np.any(centers[:, 0] - radii < -tol) or np.any(centers[:, 0] + radii > 1.0 + tol):
        return False
    if np.any(centers[:, 1] - radii < -tol) or np.any(centers[:, 1] + radii > 1.0 + tol):
        return False
    
    dx = centers[TRI_INDICES[0], 0] - centers[TRI_INDICES[1], 0]
    dy = centers[TRI_INDICES[0], 1] - centers[TRI_INDICES[1], 1]
    dist = np.hypot(dx, dy)
    r_sum = radii[TRI_INDICES[0]] + radii[TRI_INDICES[1]]
    return np.all(dist >= r_sum - tol)
