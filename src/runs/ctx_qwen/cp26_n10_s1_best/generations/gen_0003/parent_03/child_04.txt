import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26
TRIL_IDX = np.tril_indices(N_CIRCLES, -1)

def objective(vars):
    """Objective to minimize: negative sum of radii."""
    return -np.sum(vars[2::3])

def constraints(vars):
    """Returns vector of inequality constraints g(vars) >= 0."""
    n = N_CIRCLES
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    # Boundary constraints: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    c = np.concatenate([
        x - r,
        1.0 - x - r,
        y - r,
        1.0 - y - r
    ])
    
    # Overlap constraints: dist(i,j) - r_i - r_j >= 0
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist = np.hypot(dx, dy)
    r_sum = r[:, None] + r[None, :]
    
    c = np.concatenate([c, dist[TRIL_IDX] - r_sum[TRIL_IDX]])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-6, 0.5]."""
    b = []
    for _ in range(N_CIRCLES):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)])
    return b

def force_relax(centers, radii, steps=400):
    """Force-directed relaxation to resolve overlaps and boundary violations."""
    n = centers.shape[0]
    for _ in range(steps):
        fx, fy = np.zeros(n), np.zeros(n)
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[j, 0] - centers[i, 0]
                dy = centers[j, 1] - centers[i, 1]
                d = np.hypot(dx, dy)
                if d < radii[i] + radii[j] and d > 1e-6:
                    f = (radii[i] + radii[j] - d) * 0.5 / d
                    fx[i] -= f * dx
                    fy[i] -= f * dy
                    fx[j] += f * dx
                    fy[j] += f * dy
        for i in range(n):
            if centers[i, 0] < radii[i]: fx[i] += 0.1 * (radii[i] - centers[i, 0])
            if centers[i, 0] > 1.0 - radii[i]: fx[i] -= 0.1 * (centers[i, 0] - (1.0 - radii[i]))
            if centers[i, 1] < radii[i]: fy[i] += 0.1 * (radii[i] - centers[i, 1])
            if centers[i, 1] > 1.0 - radii[i]: fy[i] -= 0.1 * (centers[i, 1] - (1.0 - radii[i]))
            
        centers += np.column_stack([fx, fy]) * 0.12
        centers = np.clip(centers, 0.0, 1.0)
    return centers

def make_hex_init(r0, angle=0.0, seed=0):
    """Generates a hexagonal lattice initialization with optional rotation."""
    np.random.seed(seed)
    pts = []
    y = r0
    row = 0
    while len(pts) < N_CIRCLES + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        if y > 1.0 + r0:
            break
            
    pts = np.array(pts[:N_CIRCLES + 10])
    if angle != 0.0:
        rad = angle
        c, s = np.cos(rad), np.sin(rad)
        center = np.array([0.5, 0.5])
        pts = (pts - center) @ np.array([[c, -s], [s, c]]) + center
        
    mask = (pts[:, 0] >= 0.01) & (pts[:, 0] <= 0.99) & (pts[:, 1] >= 0.01) & (pts[:, 1] <= 0.99)
    pts = pts[mask]
    while len(pts) < N_CIRCLES:
        pts = np.vstack([pts, np.random.uniform(0.1, 0.9, 2)])
        
    centers = pts[:N_CIRCLES]
    centers += np.random.uniform(-0.002, 0.002, centers.shape)
    radii = np.full(N_CIRCLES, r0)
    return centers, radii

def is_valid_config(centers, radii, tol=1e-9):
    """Checks feasibility strictly."""
    n = centers.shape[0]
    if np.any(radii < 1e-7):
        return False
    if np.any(centers[:, 0] - radii < -tol) or np.any(centers[:, 0] + radii > 1.0 + tol):
        return False
    if np.any(centers[:, 1] - radii < -tol) or np.any(centers[:, 1] + radii > 1.0 + tol):
        return False
    
    d_mat = np.sqrt(np.sum((centers[:, None, :] - centers[None, :, :])**2, axis=2))
    r_s = radii[:, None] + radii[None, :]
    return np.all(d_mat[TRIL_IDX] >= r_s[TRIL_IDX] - tol)

        x0[1::3] = np.clip(x0[1::3], r_p, 1.0 - r_p)
        x0[2::3] = r_p
