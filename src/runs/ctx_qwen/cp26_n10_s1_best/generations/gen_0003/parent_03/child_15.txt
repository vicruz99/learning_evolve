import numpy as np
from scipy.optimize import minimize, linprog

N_CIRCLES = 26

def objective(x):
    """Objective to minimize: negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Returns vector of inequality constraints g(x) >= 0."""
    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]
    
    # Boundary constraints: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    c = np.concatenate([
        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs
    ])
    
    # Overlap constraints: dist^2 - (r_i + r_j)^2 >= 0
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    i_idx, j_idx = np.tril_indices(N_CIRCLES, -1)
    c = np.concatenate([c, dx[i_idx, j_idx]**2 + dy[i_idx, j_idx]**2 - dr[i_idx, j_idx]**2])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [0, 0.5]."""
    b = []
    for _ in range(N_CIRCLES):
        b.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
    return b

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N_CIRCLES
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    for i in range(n):
        x_val, y_val = centers[i]
        bound = min(x_val, 1.0 - x_val, y_val, 1.0 - y_val)
        A_ub[idx, i] = 1.0
        b_ub[idx] = bound
        idx += 1
        
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dist
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and not np.any(np.isnan(res.x)):
            return res.x
    except Exception:
        pass
    return np.full(n, 0.01)

def force_relax(centers, radii, steps=250):
    """Force-directed relaxation to resolve overlaps and push to boundaries."""
    n = N_CIRCLES
    cs = centers.copy()
    rs = radii.copy()
    
    for _ in range(steps):
        forces = np.zeros_like(cs)
        for i in range(n):
            for j in range(i + 1, n):
                dx = cs[j, 0] - cs[i, 0]
                dy = cs[j, 1] - cs[i, 1]
                dist = np.hypot(dx, dy)
                if dist < rs[i] + rs[j] and dist > 1e-9:
                    f = (rs[i] + rs[j] - dist) * 0.5 / dist
                    forces[i, 0] -= f * dx
                    forces[i, 1] -= f * dy
                    forces[j, 0] += f * dx
                    forces[j, 1] += f * dy
                    
            # Boundary repulsion
            for d in range(2):
                if cs[i, d] < rs[i]:
                    forces[i, d] += 0.15 * (rs[i] - cs[i, d])
                elif cs[i, d] > 1.0 - rs[i]:
                    forces[i, d] -= 0.15 * (cs[i, d] - (1.0 - rs[i]))
                    
        cs += forces * 0.12
        cs = np.clip(cs, 0.0, 1.0)
    return cs, rs

def generate_hex_init(seed, r0=0.095):
    """Generates a hexagonal lattice initialization with perturbation."""
    np.random.seed(seed)
    centers = []
    y = r0
    row = 0
    while len(centers) < N_CIRCLES + 5:
        x_start = r0 if row % 2 == 0 else 2.0 * r0
        x = x_start
        while x <= 1.0 - r0:
            centers.append([
                x + np.random.uniform(-0.004, 0.004),
                y + np.random.uniform(-0.004, 0.004)
            ])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        
    centers = np.array(centers[:N_CIRCLES])
    return centers, np.full(N_CIRCLES, r0 * 0.85)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    best_sum = -1.0
    best_x = None
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    
    # Phase 1: Diverse Initializations
    inits = []
    
    # Hexagonal lattices with varying densities and random seeds
    for s in range(25):
        r0 = 0.09 + 0.005 * (s % 5)
        c, r = generate_hex_init(s, r0)
        inits.append((c, r))
        
    # Random scattered starts
    for s in range(15):
        np.random.seed(s + 100)
        c = np.random.uniform(0.15, 0.85, (N_CIRCLES, 2))
        r = np.full(N_CIRCLES, 0.055)
        inits.append((c, r))
        
    # Structured grid start
    c_grid = np.zeros((N_CIRCLES, 2))
    idx = 0
    for i in range(6):
        for j in range(5):
            if idx < N_CIRCLES:
                c_grid[idx] = [0.1 + 0.17 * j, 0.1 + 0.16 * i]
                idx += 1
    inits.append((c_grid, np.full(N_CIRCLES, 0.065)))
    
    # Optimize each initialization
    for c_init, r_init in inits:
        # LP to get feasible, maximized radii for initial layout
        r_lp = solve_lp_radii(c_init)
        if np.sum(r_lp) < 1.5:  # Skip degenerate layouts
            continue
            
        # Force relaxation to spread circles
        c_rel, r_rel = force_relax(c_init, r_lp, steps=200)
        
        x0 = np.zeros(3 * N_CIRCLES)
        x0[0::3] = c_rel[:, 0]
        x0[1::3] = c_rel[:, 1]
        x0[2::3] = r_rel
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 8000, 'ftol': 1e-13, 'disp': False})
            if not np.isnan(res.fun) and -res.fun > best_sum:
                best_sum = -res.fun
                best_x = res.x.copy()
        except Exception:
            continue
            
    # Phase 2: Alternating Refinement (LP radii <-> SLSQP positions)
    if best_x is not None:
        for step in range(8):
            c_cur = best_x[0::3].copy()
            r_cur = best_x[2::3].copy()
            
            # Slight perturbation to escape local bottlenecks
            c_cur += np.random.normal(0, 0.0015, (N_CIRCLES, 2))
            c_cur = np.clip(c_cur, 0.01, 0.99)
            
            r_new = solve_lp_radii(c_cur)
            if np.sum(r_new) <= np.sum(r_cur):
                continue
                
            x0 = np.zeros(3 * N_CIRCLES)
            x0[0::3] = c_cur[:, 0]
            x0[1::3] = c_cur[:, 1]
            x0[2::3] = r_new
            
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 6000, 'ftol': 1e-13, 'disp': False})
                if not np.isnan(res.fun) and -res.fun > best_sum:
                    best_sum = -res.fun
                    best_x = res.x.copy()
            except Exception:
                pass
                
    # Phase 3: Deflation & Escape Strategy
    if best_x is not None:
        for _ in range(6):
            x_pert = best_x.copy()
            x_pert[2::3] *= 0.96  # Shrink radii to create breathing room
            x_pert[0::3] += np.random.normal(0, 0.004, N_CIRCLES)
            x_pert[1::3] += np.random.normal(0, 0.004, N_CIRCLES)
            
            # Project to feasible region
            for i in range(N_CIRCLES):
                r = max(0.005, x_pert[3*i+2])
                x_pert[3*i] = np.clip(x_pert[3*i], r, 1.0-r)
                x_pert[3*i+1] = np.clip(x_pert[3*i+1], r, 1.0-r)
                x_pert[3*i+2] = r
                
            try:
                res = minimize(objective, x_pert, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 7000, 'ftol': 1e-13, 'disp': False})
                if not np.isnan(res.fun) and -res.fun > best_sum:
                    best_sum = -res.fun
                    best_x = res.x.copy()
            except Exception:
                pass

    # Final Polish
    if best_x is not None:
        try:
            res = minimize(objective, best_x, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 15000, 'ftol': 1e-14, 'disp': False})
            if not np.isnan(res.fun) and -res.fun > best_sum:
                best_x = res.x.copy()
        except Exception:
            pass

    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3]
    
    # Strict validation and minimal repair
    for _ in range(100):
        valid = True
        for i in range(N_CIRCLES):
            if radii[i] < 0: valid = False; break
            if centers[i,0] < radii[i] - 1e-12 or centers[i,0] > 1.0 - radii[i] + 1e-12: valid = False; break
            if centers[i,1] < radii[i] - 1e-12 or centers[i,1] > 1.0 - radii[i] + 1e-12: valid = False; break
        if valid:
            for i in range(N_CIRCLES):
                for j in range(i + 1, N_CIRCLES):
                    d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
                    if d < radii[i] + radii[j] - 1e-12:
                        valid = False; break
                if not valid: break
        if valid: break
        radii *= 0.99995
        
    return centers, radii, float(np.sum(radii))
