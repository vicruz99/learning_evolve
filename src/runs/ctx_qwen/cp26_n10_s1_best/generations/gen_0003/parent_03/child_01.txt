import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26
TRIL_I, TRIL_J = np.tril_indices(N_CIRCLES, -1)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Returns vector of inequality constraints g(x) >= 0."""
    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]
    
    # Boundary constraints: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    b = np.concatenate([xs - rs, 1.0 - xs - rs, ys - rs, 1.0 - ys - rs])
    
    # Overlap constraints: dist^2 - (r_i + r_j)^2 >= 0
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    o = dx[TRIL_I, TRIL_J]**2 + dy[TRIL_I, TRIL_J]**2 - dr[TRIL_I, TRIL_J]**2
    return np.concatenate([b, o])

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-7, 0.5]."""
    bounds = []
    for _ in range(N_CIRCLES):
        bounds.extend([(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)])
    return bounds

def get_feasible_radii(centers):
    """Compute strictly feasible initial radii based on nearest neighbors/walls."""
    n = centers.shape[0]
    # Distance to boundaries
    dists_bound = np.minimum(centers, 1.0 - centers)
    min_d_bound = np.min(dists_bound, axis=1)
    
    # Distance to other centers
    diff = centers[:, None, :] - centers[None, :, :]
    dists_pair = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists_pair, np.inf)
    min_d_pair = np.min(dists_pair, axis=1)
    
    min_d = np.minimum(min_d_bound, min_d_pair)
    # Conservative scaling to guarantee initial feasibility
    return min_d * 0.45

    starts = []
    patterns = [[5,6,5,6,4], [6,5,6,5,4], [5,6,6,5,4], [4,6,6,6,4], [6,6,5,5,4]]

                    if idx < N_CIRCLES:
                        centers[idx] = [x_start + k * 2 * r0, y]
                        idx += 1
                y += y_step

                    c_rot = (c_rot - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
                    
                # Perturb and clip
                c_rot += np.random.uniform(-0.008, 0.008, c_rot.shape)
                c_rot = np.clip(c_rot, 0.05, 0.95)
                
                rs = get_feasible_radii(c_rot)
                x0 = np.zeros(3 * N_CIRCLES)
                x0[0::3] = c_rot[:, 0]
                x0[1::3] = c_rot[:, 1]
                x0[2::3] = rs
                starts.append(x0)

    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
