import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Returns vector of inequality constraints g(x) >= 0."""
    n = N
    cx = x[0::3]
    cy = x[1::3]
    cr = x[2::3]
    
    con = np.concatenate([
        cx - cr,
        1.0 - cx - cr,
        cy - cr,
        1.0 - cy - cr
    ])
    
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dist = np.sqrt(dx**2 + dy**2)
    r_sum = cr[:, None] + cr[None, :]
    
    i_idx, j_idx = np.triu_indices(n, 1)
    con = np.concatenate([con, dist[i_idx, j_idx] - r_sum[i_idx, j_idx]])
    return con

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-7, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)])
    return b

def solve_lp_radii(centers):
    """Solve LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]), 
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    
    dx = centers[:, 0:1] - centers[0:, None, 0]
    dy = centers[:, 1:2] - centers[0:, None, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    m = n + n * (n - 1) // 2
    A_ub = np.zeros((m, n))
    b_ub = np.zeros(m)
    
    k = 0
    for i in range(n):
        A_ub[k, i] = 1.0
        b_ub[k] = lims[i]
        k += 1
        
    tri_i, tri_j = np.triu_indices(n, 1)
    for idx in range(len(tri_i)):
        i, j = tri_i[idx], tri_j[idx]
        A_ub[k, i] = 1.0
        A_ub[k, j] = 1.0
        b_ub[k] = dists[i, j]
        k += 1
        
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-9)
    except Exception:
        pass
    return np.full(n, 0.05)

def generate_hex_init(counts, r_start=0.085, angle=0.0, seed=0):
    """Generate hexagonal lattice initialization with optional rotation."""
    np.random.seed(seed)
    centers = np.zeros((N, 2))
    idx = 0
    y = r_start
    y_step = np.sqrt(3) * r_start
    for i, cnt in enumerate(counts):
        x_start = r_start + (r_start if i % 2 == 1 else 0.0)
        for k in range(cnt):
            if idx < N:
                x = x_start + k * 2 * r_start
                centers[idx] = [x, y]
                idx += 1
        y += y_step
    while idx < N:
        centers[idx] = [np.random.uniform(r_start, 1.0 - r_start), 
                        np.random.uniform(r_start, 1.0 - r_start)]
        idx += 1
        
    if angle != 0.0:
        cx, cy = 0.5, 0.5
        c, s = np.cos(angle), np.sin(angle)
        dx = centers[:, 0] - cx
        dy = centers[:, 1] - cy
        centers[:, 0] = dx * c - dy * s + cx
        centers[:, 1] = dx * s + dy * c + cy
        
    centers += np.random.uniform(-0.002, 0.002, centers.shape)
    centers = np.clip(centers, 0.02, 0.98)
    return centers

def force_init(n, seed):
    """Force-directed spread initialization."""
    np.random.seed(seed)
    pts = np.random.uniform(0.15, 0.85, (n, 2))
    for _ in range(600):
        f = np.zeros_like(pts)
        for i in range(n):
            for j in range(i + 1, n):
                d = np.linalg.norm(pts[i] - pts[j])
                if d < 0.25 and d > 1e-5:
                    vec = pts[j] - pts[i]
                    f[i] -= vec / d
                    f[j] += vec / d
        for i in range(n):
            if pts[i, 0] < 0.1: f[i, 0] += 0.1
            elif pts[i, 0] > 0.9: f[i, 0] -= 0.1
            if pts[i, 1] < 0.1: f[i, 1] += 0.1
            elif pts[i, 1] > 0.9: f[i, 1] -= 0.1
        pts += f * 0.02
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -1.0
    best_x = None
    
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 6, 6, 5, 4],
        [4, 6, 6, 6, 4], [5, 5, 5, 5, 6], [6, 6, 6, 4, 4]
    ]
    
    inits = []
    init_id = 0
    for pat in patterns:
        for ang in [0.0, 0.15, -0.15, 0.3, -0.3, 0.45, -0.45]:
            inits.append(generate_hex_init(pat, r_start=0.085, angle=ang, seed=init_id))
            init_id += 1
            
    for s in range(15):
        inits.append(force_init(N, s))
        
    for c_init in inits:
        x0 = np.zeros(3 * N)
        x0[0::3] = c_init[:, 0]
        x0[1::3] = c_init[:, 1]
        x0[2::3] = 0.085
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 15000, 'ftol': 1e-14, 'disp': False})
            if not np.isnan(res.fun):
                s_val = -res.fun
                if s_val > best_sum:
                    c_opt = np.column_stack((res.x[0::3], res.x[1::3]))
                    r_opt = res.x[2::3]
                    valid = True
                    for i in range(N):
                        if r_opt[i] < 0 or c_opt[i, 0] < r_opt[i] - 1e-9 or c_opt[i, 0] > 1 - r_opt[i] + 1e-9 or \
                           c_opt[i, 1] < r_opt[i] - 1e-9 or c_opt[i, 1] > 1 - r_opt[i] + 1e-9:
                            valid = False
                            break
                    if valid:
                        dx = c_opt[:, 0:1] - c_opt[0:, None, 0]
                        dy = c_opt[:, 1:2] - c_opt[0:, None, 1]
                        dist = np.sqrt(dx**2 + dy**2)
                        r_sum = r_opt[:, None] + r_opt[None, :]
                        tri = np.triu_indices(N, 1)
                        if np.all(dist[tri] >= r_sum[tri] - 1e-9):
                            best_sum = s_val
                            best_x = res.x.copy()
        except Exception:
            continue
            
    if best_x is None:
        best_x = np.zeros(3 * N)
        best_x[2::3] = 0.08
        best_x[0::3] = np.tile(np.linspace(0.1, 0.9, 5), 6)[:N]
        best_x[1::3] = np.repeat(np.linspace(0.1, 0.9, 6), 5)[:N]

    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3].copy()
    
    for step in range(40):
        new_radii = solve_lp_radii(centers)
        
        pert_scale = 0.004 * (0.96 ** step)
        pert = np.random.normal(0, pert_scale, centers.shape)
        
        dx = centers[:, 0:1] - centers[0:, None, 0]
        dy = centers[:, 1:2] - centers[0:, None, 1]
        dists = np.sqrt(dx**2 + dy**2)
        r_sum = new_radii[:, None] + new_radii[None, :]
        
        push = np.zeros_like(centers)
        for i in range(N):
            for j in range(i + 1, N):
                if dists[i, j] < r_sum[i, j] + 0.002 and dists[i, j] > 1e-6:
                    f = (r_sum[i, j] - dists[i, j]) * 0.12 / dists[i, j]
                    push[i, 0] -= f * dx[i, j] / dists[i, j]
                    push[i, 1] -= f * dy[i, j] / dists[i, j]
                    push[j, 0] += f * dx[i, j] / dists[i, j]
                    push[j, 1] += f * dy[i, j] / dists[i, j]
                    
        centers = centers + pert + push
        centers = np.clip(centers, new_radii[:, None] + 1e-6, 1.0 - new_radii[:, None] - 1e-6)
        radii = new_radii
        
        x0 = np.zeros(3 * N)
        x0[0::3] = centers[:, 0]
        x0[1::3] = centers[:, 1]
        x0[2::3] = radii
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 6000, 'ftol': 1e-14, 'disp': False})
            if not np.isnan(res.fun):
                s_val = -res.fun
                if s_val > best_sum + 1e-8:
                    best_x = res.x.copy()
                    best_sum = s_val
        except Exception:
            pass
            
    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3]
    
    for _ in range(60):
        valid = True
        for i in range(N):
            if radii[i] < 0 or centers[i, 0] < radii[i] - 1e-12 or centers[i, 0] > 1 - radii[i] + 1e-12 or \
               centers[i, 1] < radii[i] - 1e-12 or centers[i, 1] > 1 - radii[i] + 1e-12:
                valid = False
                break
        if valid:
            dx = centers[:, 0:1] - centers[0:, None, 0]
            dy = centers[:, 1:2] - centers[0:, None, 1]
            dist = np.sqrt(dx**2 + dy**2)
            r_sum = radii[:, None] + radii[None, :]
            tri = np.triu_indices(N, 1)
            if np.any(dist[tri] < r_sum[tri] - 1e-12):
                valid = False
        if valid:
            break
        radii *= 0.9995
        centers[:, 0] = np.clip(centers[:, 0], radii, 1 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1 - radii)
        
    return centers, radii, float(np.sum(radii))
