from scipy.optimize import minimize

N_CIRCLES = 26
TRIU_IDX = np.triu_indices(N_CIRCLES, 1)

def objective(x):

    return -np.sum(x[2::3])

def constraints(x):

    ys = x[1::3]
    rs = x[2::3]

    c = np.empty(n_cons)

    c[N_CIRCLES:2*N_CIRCLES] = 1.0 - xs - rs
    c[2*N_CIRCLES:3*N_CIRCLES] = ys - rs
    c[3*N_CIRCLES:4*N_CIRCLES] = 1.0 - ys - rs

    dy = ys[TRIU_IDX[0]] - ys[TRIU_IDX[1]]

    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [0, 0.5]."""
    b = []
    for _ in range(N_CIRCLES):
        b.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])

def make_hex(counts, r0, angle=0.0):
    """Generates a hexagonal lattice initialization with optional rotation."""
    centers = []
    y = r0
    for row_idx, cnt in enumerate(counts):
        x_start = r0 + (r0 if row_idx % 2 == 1 else 0.0)
        for k in range(cnt):
            x = x_start + k * 2.0 * r0
            centers.append([x, y])
        y += np.sqrt(3.0) * r0
    centers = np.array(centers)
    
    if angle != 0:
        rad = np.deg2rad(angle)
        cos_a, sin_a = np.cos(rad), np.sin(rad)
        centers = (centers - 0.5) @ np.array([[cos_a, -sin_a], [sin_a, cos_a]]) + 0.5
    return centers

def make_feasible_radii(centers):
    """Computes strictly feasible radii based on nearest neighbors and walls."""
    n = centers.shape[0]
    rs = np.zeros(n)
    for i in range(n):
        d_wall = min(centers[i,0], 1.0 - centers[i,0], 
                     centers[i,1], 1.0 - centers[i,1])
        diffs = centers[i] - centers
        diffs[i] = np.inf  # Exclude self-distance

        # 0.85 safety factor guarantees strict initial feasibility
        rs[i] = min(d_wall, d_nn / 2.0) * 0.85
    return rs
