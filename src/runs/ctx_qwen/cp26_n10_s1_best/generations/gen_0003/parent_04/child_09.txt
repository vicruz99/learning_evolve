import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Compute all boundary and non-overlap constraints as a vector >= 0."""
    cx = x[0::3]
    cy = x[1::3]
    cr = x[2::3]
    
    # Boundary constraints: 4 * N
    b = np.concatenate([cx - cr, 1.0 - cx - cr, cy - cr, 1.0 - cy - cr])
    
    # Overlap constraints: N*(N-1)/2
    i, j = np.tril_indices(N_CIRCLES, -1)
    dx = cx[i] - cx[j]
    dy = cy[i] - cy[j]
    dr = cr[i] + cr[j]
    o = dx**2 + dy**2 - dr**2
    
    return np.concatenate([b, o])

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-6, 0.5]."""
    return [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N_CIRCLES

def resolve_and_grow(centers, steps=1500):
    """Force-directed relaxation that grows radii until they lock into a valid packing."""
    n = centers.shape[0]
    radii = np.full(n, 0.015)
    for step in range(steps):
        forces = np.zeros((n, 2))
        # Slowly grow radii to pack them tightly
        growth = 0.00006 * (1.0 - step / steps) + 0.000002
        radii += growth
        
        for i in range(n):
            # Boundary repulsion
            for d in [0, 1]:
                if centers[i, d] < radii[i]:
                    forces[i, d] += 8.0 * (radii[i] - centers[i, d])
                if centers[i, d] > 1.0 - radii[i]:
                    forces[i, d] -= 8.0 * (centers[i, d] - (1.0 - radii[i]))
                    
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[j, 0] - centers[i, 0]
                dy = centers[j, 1] - centers[i, 1]
                d = np.hypot(dx, dy)
                if d < radii[i] + radii[j] and d > 1e-8:
                    f = (radii[i] + radii[j] - d) / d
                    fx, fy = f * dx, f * dy
                    forces[i, 0] -= fx
                    forces[i, 1] -= fy
                    forces[j, 0] += fx
                    forces[j, 1] += fy
                    
        centers += forces * 0.06
        centers = np.clip(centers, radii, 1.0 - radii)
    return centers, radii

def create_hex_init(seed, r0, angle=0.0):
    """Generate a perturbed hexagonal lattice initialization."""
    np.random.seed(seed)
    pts = []
    y = r0
    row = 0
    while len(pts) < N_CIRCLES + 5:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        
    pts = np.array(pts[:N_CIRCLES + 5])
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
        
    # Filter points strictly inside the square
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    while len(pts) < N_CIRCLES:
        pts = np.vstack([pts, [np.random.uniform(0.2, 0.8), np.random.uniform(0.2, 0.8)]])
        
    centers = pts[:N_CIRCLES] + np.random.uniform(-0.004, 0.004, (N_CIRCLES, 2))
    return centers

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = N_CIRCLES
    bnds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -1.0
    best_x = None
    
    # Phase 1: Generate diverse initial configurations
    inits = []
    # Varied hexagonal densities
    for s in range(12):
        r0 = 0.082 + s * 0.005
        inits.append(create_hex_init(s, r0, angle=0.0))
    # Rotated hexagons to break symmetry
    for s in range(8):
        inits.append(create_hex_init(s + 50, 0.095, angle=0.12 + s * 0.04))
    # Pure random starts
    for s in range(10):
        centers = np.random.uniform(0.15, 0.85, (n, 2))
        inits.append(centers)
        
    # Phase 2: Relax and Optimize
    for centers in inits:
        c_res, r_res = resolve_and_grow(centers.copy())
        x0 = np.zeros(3 * n)
        x0[0::3] = c_res[:, 0]
        x0[1::3] = c_res[:, 1]
        x0[2::3] = r_res
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bnds, constraints=cons,
                           options={'maxiter': 30000, 'ftol': 1e-13, 'disp': False})
            curr_sum = -res.fun
            c_vals = constraints(res.x)
            # Accept if feasible within tolerance and improves best sum
            if np.min(c_vals) >= -1e-7 and curr_sum > best_sum:
                best_sum = curr_sum
                best_x = res.x.copy()
        except Exception:
            pass
            
    # Phase 3: Local refinement with decaying noise to escape locked minima
    if best_x is not None:
        for i in range(10):

            x0 = best_x + np.random.normal(0, noise, 3 * n)
            
            # Project perturbed variables back to strict bounds
            for k in range(n):
                r = np.clip(x0[3*k + 2], 1e-5, 0.5)
                x0[3*k] = np.clip(x0[3*k], r, 1.0 - r)

                x0[3*k + 2] = r

            except Exception:
                pass
                
    # Fallback if optimization completely fails (should not happen)
    if best_x is None:

        best_x = np.zeros(3 * n)
        best_x[0::3] = c_f[:, 0]
        best_x[1::3] = c_f[:, 1]
        best_x[2::3] = r_f
        best_sum = np.sum(r_f)
        
    # Extract results
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        centers[i, 0] = best_x[3*i]
        centers[i, 1] = best_x[3*i+1]
        radii[i] = best_x[3*i+2]
        
    # Phase 4: Strict validity check and numerical repair

                valid = False; break

                valid = False; break
            if centers[i, 1] < radii[i] - 1e-11 or centers[i, 1] > 1.0 - radii[i] + 1e-11:
                valid = False; break
        if valid:
            for i in range(n):
                for j in range(i + 1, n):
                    d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                    if d < radii[i] + radii[j] - 1e-11:
                        valid = False; break
                if not valid: break
        if valid: break
        
        # Minimal shrinkage to guarantee strict compliance
        radii *= 0.999
        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)
        
    return centers, radii, float(np.sum(radii))
