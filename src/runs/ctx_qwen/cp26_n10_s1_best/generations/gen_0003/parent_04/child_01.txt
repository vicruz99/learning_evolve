from scipy.optimize import minimize, linprog

N = 26

def get_lp_radii(centers):
    """Solve LP to find optimal radii for fixed centers."""
    c = -np.ones(N)
    n_con = 4 * N + N * (N - 1) // 2
    A_ub = np.zeros((n_con, N))
    b_ub = np.zeros(n_con)
    
    idx = 0
    for i in range(N):
        x, y = centers[i]
        A_ub[idx, i] = 1.0; b_ub[idx] = x; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - x; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = y; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - y; idx += 1
        
    for i in range(N):
        for j in range(i + 1, N):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = d
            idx += 1
            
    bounds = [(0.0, None)] * N
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except:
        pass
    return np.full(N, 0.05)

def objective(x):
    """Objective to minimize: negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Vectorized inequality constraints: g(x) >= 0."""
    cx = x[0::3]
    cy = x[1::3]
    cr = x[2::3]
    
    # Boundary constraints
    b = np.concatenate([cx - cr, 1.0 - cx - cr, cy - cr, 1.0 - cy - cr])
    
    # Overlap constraints (squared distance >= sum of radii^2)
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dr = cr[:, None] + cr[None, :]
    
    i_idx, j_idx = np.tril_indices(N, -1)
    o = dx[i_idx, j_idx]**2 + dy[i_idx, j_idx]**2 - dr[i_idx, j_idx]**2
    
    return np.concatenate([b, o])

def force_layout(seed):

    c = np.random.uniform(0.2, 0.8, (N, 2))
    r = np.full(N, 0.1)

        dist = np.sqrt(np.maximum(dist_sq, 1e-10))

        f[:, 1] += np.where(c[:, 1] < r, 1.0, 0.0) - np.where(c[:, 1] > 1.0 - r, 1.0, 0.0)
        
        alpha = 0.08 * (1.0 - step / 600.0)
        c += f * alpha
        c = np.clip(c, r, 1.0 - r)
    return c

def hex_init(seed, r0):
    """Generate a perturbed hexagonal lattice."""
    np.random.seed(seed)
    c = []
    y = r0
    row = 0
    while len(c) < N and y + r0 <= 1.0:
        x_start = r0 if row % 2 == 0 else 2.0 * r0
        x = x_start
        while x + r0 <= 1.0:
            c.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1

        c.append([np.random.uniform(0.2, 0.8), np.random.uniform(0.2, 0.8)])
    c = np.array(c[:N])
    c += np.random.normal(0, 0.005, c.shape)
    return c
