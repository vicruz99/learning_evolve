import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def obj(x):
    return -np.sum(x[2::3])

def constr(x):
    n = N
    cx = x[0::3]
    cy = x[1::3]
    cr = x[2::3]
    
    b = np.concatenate([cx - cr, 1.0 - cx - cr, cy - cr, 1.0 - cy - cr])
    
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dr = cr[:, None] + cr[None, :]
    i, j = np.tril_indices(n, -1)
    o = dx[i, j]**2 + dy[i, j]**2 - dr[i, j]**2
    
    return np.concatenate([b, o])

def bd():
    return [(0.0, 1.0), (0.0, 1.0), (1e-10, 0.5)] * N

def hex(rows, r, seed=0):
    np.random.seed(seed)
    c = np.zeros((N, 2))
    ra = np.full(N, r)
    idx = 0
    y = r
    for ri, cnt in enumerate(rows):
        xs = r if ri % 2 == 0 else 2 * r
        for k in range(cnt):
            if idx >= N:
                break
            c[idx] = [xs + k * 2 * r, y]
            idx += 1
        y += np.sqrt(3) * r
    while idx < N:
        c[idx] = np.random.uniform(0.2, 0.8, 2)
        ra[idx] = 0.05
        idx += 1
    return c + np.random.normal(0, 0.002, (N, 2)), ra

def rot(c, a):
    cx, cy = 0.5, 0.5
    co, si = np.cos(a), np.sin(a)
    nc = np.zeros_like(c)
    for i in range(len(c)):
        dx, dy = c[i, 0] - cx, c[i, 1] - cy
        nc[i] = [dx * co - dy * si + cx, dx * si + dy * co + cy]
    return np.clip(nc, 0.05, 0.95)

def lp_rad(centers):
    n = N
    c = -np.ones(n)
    A = np.zeros((n + n * (n - 1) // 2, n))
    b = np.zeros(n + n * (n - 1) // 2)
    idx = 0
    for i in range(n):
        b[idx] = min(centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])
        A[idx, i] = 1
        idx += 1
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A[idx, i] = 1
            A[idx, j] = 1
            b[idx] = d
            idx += 1
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')
    return res.x if res.success else None

def force_init(seed):
    np.random.seed(seed)
    c = np.random.uniform(0.15, 0.85, (N, 2))
    r = np.full(N, 0.09)
    
    for _ in range(800):
        fx = np.zeros(N)
        fy = np.zeros(N)
        for i in range(N):
            for j in range(i + 1, N):
                dx = c[j, 0] - c[i, 0]
                dy = c[j, 1] - c[i, 1]
                d = np.hypot(dx, dy)
                if d < 0.3 and d > 1e-6:
                    f = 0.01 / (d ** 2 + 0.001)
                    fx[i] -= f * dx
                    fy[i] -= f * dy
                    fx[j] += f * dx
                    fy[j] += f * dy
            if c[i, 0] < r[i]:
                fx[i] += 0.1
            if c[i, 0] > 1 - r[i]:
                fx[i] -= 0.1
            if c[i, 1] < r[i]:
                fy[i] += 0.1
            if c[i, 1] > 1 - r[i]:
                fy[i] -= 0.1
        c += np.column_stack([fx, fy]) * 0.02
        c = np.clip(c, r, 1 - r)
    
    return c, r

def run_packing():

    con = {'type': 'ineq', 'fun': constr}

                    x0 = np.zeros(3 * n)

                    x0[2::3] = r
                    
                    try:

                        if -res.fun > best_s:

                                best_x = res.x.copy()
                    except Exception:
                        pass
    
    for seed in range(8):
