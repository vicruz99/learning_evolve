import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26
TRIL_IDX = np.tril_indices(N_CIRCLES, -1)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Compute all boundary and non-overlap constraints as a vector >= 0."""
    cx, cy, cr = x[0::3], x[1::3], x[2::3]
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    c = np.concatenate([cx - cr, 1.0 - cx - cr, cy - cr, 1.0 - cy - cr])
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dr = cr[:, None] + cr[None, :]
    c = np.concatenate([c, dx[TRIL_IDX]**2 + dy[TRIL_IDX]**2 - dr[TRIL_IDX]**2])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [0, 0.5]."""
    return [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N_CIRCLES

def make_hex_init(seed, r_base):
    """Generate a perturbed hexagonal lattice initialization."""
    np.random.seed(seed)
    x = np.zeros(3 * N_CIRCLES)
    count = 0
    y = r_base
    row = 0
    while count < N_CIRCLES and y + r_base <= 1.0:
        x_start = r_base + (r_base if row % 2 == 1 else 0.0)
        col = 0
        while count < N_CIRCLES:
            cx = x_start + col * 2 * r_base
            if cx + r_base > 1.0:
                break
            x[3*count] = cx + np.random.uniform(-0.002, 0.002)
            x[3*count+1] = y + np.random.uniform(-0.002, 0.002)
            x[3*count+2] = r_base + np.random.uniform(-0.002, 0.002)
            count += 1
            col += 1
        y += np.sqrt(3.0) * r_base
        row += 1
    while count < N_CIRCLES:
        x[3*count] = np.random.uniform(0.15, 0.85)
        x[3*count+1] = np.random.uniform(0.15, 0.85)
        x[3*count+2] = 0.06
        count += 1
    return x

def make_force_init(seed):
    """Generate a dense initial configuration using repulsive force simulation."""
    np.random.seed(seed)
    cx = np.random.uniform(0.15, 0.85, N_CIRCLES)
    cy = np.random.uniform(0.15, 0.85, N_CIRCLES)
    cr = np.full(N_CIRCLES, 0.05) # Start small to avoid overlaps during sim
    
    for _ in range(2000):
        dx = cx[:, None] - cx[None, :]
        dy = cy[:, None] - cy[None, :]
        d2 = dx**2 + dy**2
        d = np.sqrt(d2)
        mask = (d < 0.25) & (d > 1e-6)
        f = np.zeros_like(d)
        f[mask] = 0.02 / d[mask]
        fx = np.sum(f * dx, axis=1)
        fy = np.sum(f * dy, axis=1)
        cx += fx * 0.02
        cy += fy * 0.02
        cx = np.clip(cx, cr, 1.0 - cr)
        cy = np.clip(cy, cr, 1.0 - cr)
        
    x = np.zeros(3 * N_CIRCLES)
    x[0::3] = cx
    x[1::3] = cy
    x[2::3] = 0.09 # Set initial optimization radius
    return x

def repair_and_check(centers, radii):
    """Ensure strict feasibility against the validator tolerance."""
    for _ in range(100):
        valid = True
        for i in range(N_CIRCLES):
            if radii[i] < 0 or centers[i,0] < radii[i] - 1e-10 or centers[i,0] > 1.0 - radii[i] + 1e-10 or \
               centers[i,1] < radii[i] - 1e-10 or centers[i,1] > 1.0 - radii[i] + 1e-10:
                valid = False
                break
        if valid:
            for i in range(N_CIRCLES):
                for j in range(i + 1, N_CIRCLES):
                    d = np.hypot(centers[i,0] - centers[j,0], centers[i,1] - centers[j,1])
                    if d < radii[i] + radii[j] - 1e-10:
                        valid = False
                        break
                if not valid:
                    break
        if valid:
            break
        radii *= 0.995
        centers[:,0] = np.clip(centers[:,0], radii, 1.0 - radii)
        centers[:,1] = np.clip(centers[:,1], radii, 1.0 - radii)
    return centers, radii

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    bnds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    best_sum = -1.0
    best_x = None

    # Phase 1: Extensive multi-start optimization
    inits = []
    for s in range(25):
        inits.append(make_hex_init(s, 0.092 + s * 0.0005))
        inits.append(make_hex_init(s+50, 0.10 - s * 0.0002))
    for s in range(15):
        inits.append(make_force_init(s))

    for x0 in inits:
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bnds, constraints=cons,
                           options={'maxiter': 25000, 'ftol': 1e-14, 'disp': False})
            if not np.isnan(res.fun):
                curr_sum = -res.fun
                c_vals = constraints(res.x)
                if np.min(c_vals) >= -1e-7 and curr_sum > best_sum:
                    best_sum = curr_sum
                    best_x = res.x.copy()
        except Exception:
            pass

    # Phase 2: Deflation & Refinement to escape local minima
    if best_x is not None:
        for _ in range(8):
            x0 = best_x * 0.98
            x0[2::3] *= 0.98
            for i in range(N_CIRCLES):
                r = x0[3*i+2]
                x0[3*i] = np.clip(x0[3*i], r, 1.0 - r)
                x0[3*i+1] = np.clip(x0[3*i+1], r, 1.0 - r)
            x0 += np.random.normal(0, 2e-4, 3 * N_CIRCLES)
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bnds, constraints=cons,
                               options={'maxiter': 20000, 'ftol': 1e-14, 'disp': False})
                if not np.isnan(res.fun):
                    curr_sum = -res.fun
                    c_vals = constraints(res.x)
                    if np.min(c_vals) >= -1e-7 and curr_sum > best_sum:
                        best_sum = curr_sum
                        best_x = res.x.copy()
            except Exception:
                pass

    # Fallback if optimization completely fails
    if best_x is None:
        best_x = make_hex_init(0, 0.095)

    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3]
    
    centers, radii = repair_and_check(centers, radii)
    
    return centers, radii, float(np.sum(radii))
