import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Compute all boundary and non-overlap constraints as a vector >= 0."""
    cx = x[0::3]
    cy = x[1::3]
    cr = x[2::3]
    
    # Boundary constraints: 4 * N
    c = np.concatenate([
        cx - cr,
        1.0 - cx - cr,
        cy - cr,
        1.0 - cy - cr
    ])
    
    # Overlap constraints: N*(N-1)/2
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dr = cr[:, None] + cr[None, :]
    
    i, j = np.tril_indices(N, -1)
    c = np.concatenate([c, dx[i, j]**2 + dy[i, j]**2 - dr[i, j]**2])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [0, 0.5]."""
    return [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N

def hex_init(seed, scale=1.0):
    """Generate a perturbed hexagonal lattice initialization."""
    np.random.seed(seed)
    centers = []
    r0 = 0.095 * scale
    y = r0
    row = 0
    while len(centers) < N + 5:
        x = r0 if row % 2 == 0 else 2 * r0
        while x + r0 <= 1.0 and len(centers) < N + 5:
            centers.append([x, y])
            x += 2 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        
    centers = np.array(centers[:N])
    angle = np.random.uniform(-0.2, 0.2)
    c, s = np.cos(angle), np.sin(angle)
    cx, cy = 0.5, 0.5
    centers = centers - [cx, cy]
    centers = centers @ np.array([[c, -s], [s, c]])
    centers += [cx, cy]
    centers += np.random.uniform(-0.005, 0.005, centers.shape)
    centers = np.clip(centers, 0.05, 0.95)
    return centers

def force_init(seed):
    """Generate a dense initial configuration using repulsive force simulation."""
    np.random.seed(seed)
    c = np.random.uniform(0.1, 0.9, (N, 2))
    for _ in range(600):
        f = np.zeros_like(c)
        for i in range(N):
            for j in range(i + 1, N):
                diff = c[j] - c[i]
                d = np.hypot(*diff)
                if d < 0.25 and d > 1e-5:
                    push = 0.012 / d
                    f[i] -= push * diff
                    f[j] += push * diff
        c += f * 0.05
        c = np.clip(c, 0.05, 0.95)
    return c

def solve_lp_radii(centers):
    """Optimally scale radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        x, y = centers[i]
        m = min(x, 1.0 - x, y, 1.0 - y)
        A_ub[idx, i] = 1.0
        b_ub[idx] = m
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = d
            idx += 1
            
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=[(0.0, None)] * n, method='highs')
        if res.success and np.all(res.x >= -1e-9):
            return res.x
    except Exception:
        pass
    return None

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    bnds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_x = None
    best_sum = -1.0
    
    # Generate diverse initial configurations
    inits = []
    for s in range(8):
        inits.append(hex_init(s))
    for s in range(4):
        inits.append(force_init(s))
        
    for centers in inits:
        radii = np.full(N, 0.05)
        x0 = np.zeros(3 * N)
        x0[0::3] = centers[:, 0]
        x0[1::3] = centers[:, 1]
        x0[2::3] = radii
        
        # Phase 1: Initial SLSQP optimization
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bnds, constraints=cons,
                           options={'maxiter': 12000, 'ftol': 1e-13, 'disp': False})
            x_curr = res.x
        except Exception:
            continue
            
        curr_sum = -res.fun
        if curr_sum > best_sum:
            best_sum = curr_sum
            best_x = x_curr.copy()
            
        # Phase 2: Alternating LP / SLSQP refinement
        # LP finds optimal radii for fixed centers, breaking symmetries and pushing limits
        for _ in range(6):
            c_curr = np.column_stack((x_curr[0::3], x_curr[1::3]))
            r_lp = solve_lp_radii(c_curr)
            if r_lp is None:
                break
                
            x_curr[2::3] = r_lp
            # Slight center perturbation to allow new local optima
            c_pert = c_curr + np.random.normal(0, 1e-4, c_curr.shape)
            c_pert = np.clip(c_pert, r_lp, 1.0 - r_lp)
            x_curr[0::3] = c_pert[:, 0]
            x_curr[1::3] = c_pert[:, 1]
            
            try:
                res2 = minimize(objective, x_curr, method='SLSQP', bounds=bnds, constraints=cons,
                                options={'maxiter': 8000, 'ftol': 1e-13, 'disp': False})
                x_curr = res2.x
            except Exception:
                break
                
            curr_sum = -res2.fun
            if curr_sum > best_sum:
                best_sum = curr_sum
                best_x = x_curr.copy()
                
    # Extract best result
    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3]
    
    # Phase 3: Strict validity check and numerical repair
    for _ in range(100):
        valid = True
        for i in range(N):
            if radii[i] < 0:
                valid = False
                break
            if centers[i, 0] < radii[i] - 1e-10 or centers[i, 0] > 1.0 - radii[i] + 1e-10:
                valid = False
                break
            if centers[i, 1] < radii[i] - 1e-10 or centers[i, 1] > 1.0 - radii[i] + 1e-10:
                valid = False
                break
                
        if valid:
            for i in range(N):
                for j in range(i + 1, N):
                    d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                    if d < radii[i] + radii[j] - 1e-10:
                        valid = False
                        break
                if not valid:
                    break
                    
        if valid:
            break
            
        # Minimal shrinkage to guarantee strict compliance
        radii *= 0.999
        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)
        
    return centers, radii, float(np.sum(radii))
