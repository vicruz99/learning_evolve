import numpy as np
from scipy.optimize import minimize, linprog

N_CIRCLES = 26

TRIL_IDX = np.tril_indices(N_CIRCLES, -1)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Compute all boundary and non-overlap constraints >= 0."""
    n = N_CIRCLES
    cx = x[0::3]
    cy = x[1::3]
    cr = x[2::3]
    
    # Boundary constraints: 4*N
    b = np.concatenate([cx - cr, 1.0 - cx - cr, cy - cr, 1.0 - cy - cr])
    
    # Overlap constraints: N*(N-1)/2 using squared distance
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dr = cr[:, None] + cr[None, :]
    
    o = dx[TRIL_IDX]**2 + dy[TRIL_IDX]**2 - dr[TRIL_IDX]**2
    
    return np.concatenate([b, o])

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [0, 0.5]."""
    return [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N_CIRCLES

def solve_lp_radii(centers):
    """Given fixed centers, solve LP to maximize sum of radii."""
    n = N_CIRCLES
    c_obj = -np.ones(n)
    num_ineq = 4 * n + n * (n - 1) // 2
    A = np.zeros((num_ineq, n))
    b = np.zeros(num_ineq)
    idx = 0
    for i in range(n):
        A[idx, i] = 1.0
        b[idx] = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        idx += 1
    for i in range(n):
        for j in range(i + 1, n):
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            idx += 1
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def make_hex_init(row_counts, r0, seed=0, angle=0.0):
    """Create hexagonal lattice initialization with specified row counts."""
    np.random.seed(seed)
    n = N_CIRCLES
    centers = np.zeros((n, 2))
    radii = np.full(n, r0)
    idx = 0
    y = r0
    y_step = np.sqrt(3.0) * r0
    for row_num, count in enumerate(row_counts):
        x_start = r0 + (r0 if row_num % 2 == 1 else 0.0)
        for k in range(count):
            if idx >= n:
                break
            x = x_start + k * 2 * r0
            if x + r0 > 1.0:
                break
            centers[idx, 0] = x
            centers[idx, 1] = y
            idx += 1
        y += y_step
        if y + r0 > 1.0:
            if idx >= n:
                break
    while idx < n:
        centers[idx, 0] = np.random.uniform(0.15, 0.85)
        centers[idx, 1] = np.random.uniform(0.15, 0.85)
        radii[idx] = 0.05
        idx += 1
    
    # Apply rotation around center
    if angle != 0.0:
        cos_a, sin_a = np.cos(angle), np.sin(angle)
        new_centers = np.zeros_like(centers)
        for i in range(n):
            dx = centers[i, 0] - 0.5
            dy = centers[i, 1] - 0.5
            new_centers[i, 0] = dx * cos_a - dy * sin_a + 0.5
            new_centers[i, 1] = dx * sin_a + dy * cos_a + 0.5
        centers = np.clip(new_centers, 0.02, 0.98)
    
    # Add small random perturbation to break symmetry
    centers += np.random.uniform(-0.002, 0.002, centers.shape)
    centers = np.clip(centers, 0.02, 0.98)
    
    x0 = np.zeros(3 * n)
    x0[0::3] = centers[:, 0]
    x0[1::3] = centers[:, 1]
    x0[2::3] = radii
    return x0

def make_force_init(seed):
    """Create dense initialization using repulsive force simulation."""
    np.random.seed(seed)
    n = N_CIRCLES
    centers = np.random.uniform(0.15, 0.85, (n, 2))
    radii = np.full(n, 0.085)
    
    for step in range(1000):
        forces = np.zeros_like(centers)
        for i in range(n):
            for j in range(i + 1, n):
                diff = centers[j] - centers[i]
                dist = np.hypot(diff[0], diff[1])
                if dist < 0.28 and dist > 1e-6:
                    f = 0.012 / (dist**2 + 0.0001)
                    forces[i] -= f * diff
                    forces[j] += f * diff
        # Wall repulsion to push circles toward boundaries
        for i in range(n):
            for coord in range(2):
                val = centers[i, coord]
                if val < radii[i] + 0.03:
                    forces[i, coord] += 0.02
                elif val > 1.0 - radii[i] - 0.03:
                    forces[i, coord] -= 0.02
        alpha = 0.025 * (1.0 - step / 1000.0)
        centers += forces * alpha
        centers = np.clip(centers, 0.05, 0.95)
    
    x0 = np.zeros(3 * n)
    x0[0::3] = centers[:, 0]
    x0[1::3] = centers[:, 1]
    x0[2::3] = radii
    return x0

def optimize_once(x0, bnds, cons, maxiter=20000):
    """Run one SLSQP optimization and return result if feasible."""
    try:
        res = minimize(objective, x0, method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-13, 'disp': False})
        if not np.isnan(res.fun):
            c_vals = constraints(res.x)
            if np.min(c_vals) >= -1e-5:
                return res.x.copy(), -res.fun
    except Exception:
        pass
    return None, None

def project_feasible(x0, n):
    """Project variables back to feasible bounds."""
    x0 = x0.copy()
    for k in range(n):
        r = max(0.01, min(0.49, x0[3 * k + 2]))
        x0[3 * k] = np.clip(x0[3 * k], r, 1.0 - r)
        x0[3 * k + 1] = np.clip(x0[3 * k + 1], r, 1.0 - r)
        x0[3 * k + 2] = r
    return x0

def check_validity(centers, radii, n):
    """Check if packing is valid."""
    for i in range(n):
        if radii[i] < 0:
            return False
        if (centers[i, 0] < radii[i] - 1e-9 or centers[i, 0] > 1.0 - radii[i] + 1e-9 or
            centers[i, 1] < radii[i] - 1e-9 or centers[i, 1] > 1.0 - radii[i] + 1e-9):
            return False
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            if d < radii[i] + radii[j] - 1e-9:
                return False
    return True

def run_packing():

    n = N_CIRCLES
    bnds = get_bounds()

        [4, 6, 5, 6, 5], [6, 5, 5, 6, 4], [5, 6, 4, 6, 5], [6, 4, 6, 6, 4],
        [4, 5, 6, 6, 5], [5, 4, 6, 6, 5], [6, 6, 4, 6, 4], [4, 6, 6, 5, 5],
    ]
    
    for pat in patterns:

                    configs.append(make_hex_init(pat, r0, seed, angle))
    
    # Force-directed configurations
    for seed in range(20):

    # Phase 2: Multi-start SLSQP optimization
    for x0 in configs:
        x_opt, s = optimize_once(x0, bnds, cons, maxiter=15000)
        if s is not None and s > best_sum:
            best_sum = s
            best_x = x_opt
    
    # Phase 3: Alternating LP/NLP optimization
    if best_x is not None:
        centers = np.column_stack((best_x[0::3], best_x[1::3]))
        
        for alt_round in range(6):
            # Step A: Fix centers, optimize radii via LP
            new_radii = solve_lp_radii(centers)
            if new_radii is not None:
