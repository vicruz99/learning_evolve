6. **Constraint formulation**: Use squared distance constraints `d² - (r₁+r₂)² >= 0` which have smooth gradients and are differentiable everywhere (including at the boundary).

- Boundary: simple ±1 entries
- Overlap (squared): ∂/∂x_i = 2·dx, ∂/∂y_i = 2·dy, ∂/∂r_i = -2·(r_i+r_j)

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
TRIL_I, TRIL_J = np.tril_indices(N, -1)
NUM_PAIRS = len(TRIL_I)


def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])


def objective_jac(x):
    """Gradient of objective."""
    g = np.zeros(3 * N)
    g[2::3] = -1.0
    return g


def constraints_val(x):
    """Constraint values (must be >= 0)."""
    cx = x[0::3]
    cy = x[1::3]
    cr = x[2::3]

    # Boundary: x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
    b = np.concatenate([cx - cr, 1.0 - cx - cr, cy - cr, 1.0 - cy - cr])

    # Overlap: d^2 - (r_i + r_j)^2 >= 0
    dx = cx[TRIL_I] - cx[TRIL_J]
    dy = cy[TRIL_I] - cy[TRIL_J]
    dr = cr[TRIL_I] + cr[TRIL_J]
    o = dx * dx + dy * dy - dr * dr

    return np.concatenate([b, o])


def constraints_jac(x):
    """Analytical Jacobian of constraints."""
    m = 4 * N + NUM_PAIRS
    nv = 3 * N
    J = np.zeros((m, nv))

    # Boundary constraints Jacobian
    for i in range(N):
        bi = 4 * i
        # x_i - r_i >= 0
        J[bi, 3 * i] = 1.0
        J[bi, 3 * i + 2] = -1.0
        # 1 - x_i - r_i >= 0
        J[bi + 1, 3 * i] = -1.0
        J[bi + 1, 3 * i + 2] = -1.0
        # y_i - r_i >= 0
        J[bi + 2, 3 * i + 1] = 1.0
        J[bi + 2, 3 * i + 2] = -1.0
        # 1 - y_i - r_i >= 0
        J[bi + 3, 3 * i + 1] = -1.0
        J[bi + 3, 3 * i + 2] = -1.0

    # Overlap constraints Jacobian: d^2 - (r_i + r_j)^2 >= 0
    off = 4 * N
    for k in range(NUM_PAIRS):
        i, j = TRIL_I[k], TRIL_J[k]
        r = off + k
        dx = x[3 * i] - x[3 * j]
        dy = x[3 * i + 1] - x[3 * j + 1]
        dr = x[3 * i + 2] + x[3 * j + 2]

        J[r, 3 * i] = 2.0 * dx
        J[r, 3 * i + 1] = 2.0 * dy
        J[r, 3 * i + 2] = -2.0 * dr
        J[r, 3 * j] = -2.0 * dx
        J[r, 3 * j + 1] = -2.0 * dy
        J[r, 3 * j + 2] = -2.0 * dr

    return J


def get_bounds():
    """Variable bounds."""
    return [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N


def solve_lp_radii(centers):
    """Given fixed centers, find optimal radii via linear programming.
    
    Maximizes sum of radii subject to:
    - r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i (boundary)
    - r_i + r_j <= dist(i,j) (non-overlap)
    - r_i >= 0
    """
    n = N
    c_obj = -np.ones(n)

    n_boundary = 4 * n
    n_overlap = n * (n - 1) // 2
    n_constr = n_boundary + n_overlap

    A = np.zeros((n_constr, n))
    b = np.zeros(n_constr)

    idx = 0
    for i in range(n):
        xv, yv = centers[i]
        # r_i <= x_i
        A[idx, i] = 1.0
        b[idx] = xv
        idx += 1
        # r_i <= 1 - x_i
        A[idx, i] = 1.0
        b[idx] = 1.0 - xv
        idx += 1
        # r_i <= y_i
        A[idx, i] = 1.0
        b[idx] = yv
        idx += 1
        # r_i <= 1 - y_i
        A[idx, i] = 1.0
        b[idx] = 1.0 - yv
        idx += 1

    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = d
            idx += 1

    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success and np.all(res.x >= -1e-10):
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def generate_hex(r0, seed, angle=0.0):
    """Generate hexagonal lattice initialization with optional rotation."""
    np.random.seed(seed)
    x = np.zeros(3 * N)
    count = 0
    y = r0
    row = 0

    while count < N and y + r0 <= 1.0:
        x_start = r0 + (r0 if row % 2 == 1 else 0.0)
        while count < N:
            cx = x_start
            if cx + r0 > 1.0:
                break
            x[3 * count] = cx + np.random.uniform(-0.003, 0.003)
            x[3 * count + 1] = y + np.random.uniform(-0.003, 0.003)
            x[3 * count + 2] = r0 + np.random.uniform(-0.002, 0.002)
            count += 1
            x_start += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1

    while count < N:
        x[3 * count] = np.random.uniform(0.15, 0.85)
        x[3 * count + 1] = np.random.uniform(0.15, 0.85)
        x[3 * count + 2] = 0.05
        count += 1

    # Apply rotation around center
    if abs(angle) > 1e-10:
        cx_arr = x[0::3].copy()
        cy_arr = x[1::3].copy()
        c, s = np.cos(angle), np.sin(angle)
        for i in range(N):
            dx = cx_arr[i] - 0.5
            dy = cy_arr[i] - 0.5
            x[3 * i] = dx * c - dy * s + 0.5
            x[3 * i + 1] = dx * s + dy * c + 0.5

    return x


def force_init(seed):
    """Generate dense initial configuration using repulsive force simulation."""
    np.random.seed(seed)
    cx = np.random.uniform(0.15, 0.85, N)
    cy = np.random.uniform(0.15, 0.85, N)
    cr = np.full(N, 0.08)

    for step in range(2000):
        fx = np.zeros(N)
        fy = np.zeros(N)
        alpha = 0.04 * max(0.0, 1.0 - step / 2000.0)

        for i in range(N):
            for j in range(i + 1, N):
                dx = cx[j] - cx[i]
                dy = cy[j] - cy[i]
                d = np.hypot(dx, dy)
                if d < 0.35 and d > 1e-8:
                    f = 0.025 / (d * d + 0.0002)
                    fxd = f * dx
                    fyd = f * dy
                    fx[i] -= fxd
                    fx[j] += fxd
                    fy[i] -= fyd
                    fy[j] += fyd

            # Wall repulsion
            if cx[i] < cr[i] + 0.02:
                fx[i] += 0.15
            if cx[i] > 1.0 - cr[i] - 0.02:
                fx[i] -= 0.15
            if cy[i] < cr[i] + 0.02:
                fy[i] += 0.15
            if cy[i] > 1.0 - cr[i] - 0.02:
                fy[i] -= 0.15

        cx += fx * alpha
        cy += fy * alpha
        cx = np.clip(cx, 0.02, 0.98)
        cy = np.clip(cy, 0.02, 0.98)

    x = np.zeros(3 * N)
    x[0::3] = cx
    x[1::3] = cy
    x[2::3] = cr
    return x


def perturb_and_clip(x0):
    """Clip perturbed variables to feasible bounds."""
    for i in range(N):
        r = max(0.005, x0[3 * i + 2])
        x0[3 * i + 2] = r
        x0[3 * i] = np.clip(x0[3 * i], r, 1.0 - r)
        x0[3 * i + 1] = np.clip(x0[3 * i + 1], r, 1.0 - r)
    return x0


def is_valid_config(centers, radii):
    """Check if configuration is valid."""
    n = centers.shape[0]
    for i in range(n):
        if radii[i] < 0:
            return False
        if centers[i, 0] < radii[i] - 1e-9 or centers[i, 0] > 1.0 - radii[i] + 1e-9:
            return False
        if centers[i, 1] < radii[i] - 1e-9 or centers[i, 1] > 1.0 - radii[i] + 1e-9:
            return False
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            if d < radii[i] + radii[j] - 1e-9:
                return False
    return True


def run_packing():
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    n = N
    bnds = get_bounds()
    cons_full = {
        'type': 'ineq',
        'fun': constraints_val,
        'jac': constraints_jac
    }

    best_sum = -1.0
    best_x = None

    # ============================
    # Phase 1: Extensive multi-start optimization
    # ============================
    inits = []

    # Varied hexagonal lattice starts
    for s in range(50):
        r0 = 0.082 + s * 0.0004
        inits.append(generate_hex(r0, seed=s))

    # Rotated hexagonal grids
    for s in range(30):
        angle = (s + 1) * np.pi / 30.0  # 6 to 180 degrees
        r0 = 0.087 + (s % 10) * 0.001
        inits.append(generate_hex(r0, seed=s + 100, angle=angle))

    # Force-directed starts
    for s in range(25):
        inits.append(force_init(seed=s + 200))

    # Optimize from each initialization
    for x0 in inits:
        try:
            res = minimize(
                objective, x0,
                method='SLSQP',
                bounds=bnds,
                constraints=cons_full,
                jac=objective_jac,
                options={'maxiter': 25000, 'ftol': 1e-15, 'disp': False}
            )
            curr_sum = -res.fun
            c_vals = constraints_val(res.x)
            if np.min(c_vals) >= -1e-6 and curr_sum > best_sum:
                best_sum = curr_sum
                best_x = res.x.copy()
        except Exception:
            pass

    # ============================
    # Phase 2: LP-SLSQP alternating refinement
    # ============================
    if best_x is not None:
        for cycle in range(20):
            # Step 1: LP for optimal radii given current centers
            centers = np.column_stack((best_x[0::3], best_x[1::3]))
            radii_lp = solve_lp_radii(centers)

            if radii_lp is not None:
                lp_sum = np.sum(radii_lp)
                if lp_sum > best_sum:
                    best_sum = lp_sum

                # Step 2: SLSQP with LP radii as starting point + perturbation
                noise = 2e-4 * (0.85 ** cycle)
                x0 = np.zeros(3 * N)
                x0[0::3] = centers[:, 0]
                x0[1::3] = centers[:, 1]
                x0[2::3] = radii_lp
                x0 += np.random.normal(0, noise, 3 * N)
                x0 = perturb_and_clip(x0)

                try:
                    res = minimize(
                        objective, x0,
                        method='SLSQP',
                        bounds=bnds,
                        constraints=cons_full,
                        jac=objective_jac,
                        options={'maxiter': 20000, 'ftol': 1e-15, 'disp': False}
                    )
                    curr_sum = -res.fun
                    c_vals = constraints_val(res.x)
                    if np.min(c_vals) >= -1e-6 and curr_sum > best_sum:
                        best_sum = curr_sum
                        best_x = res.x.copy()
                except Exception:
                    pass

    # ============================
    # Phase 3: Direct perturbation refinement
    # ============================
    if best_x is not None:
        for it in range(15):
            noise = 1e-3 * (0.75 ** it)
            x0 = best_x + np.random.normal(0, noise, 3 * N)
            x0 = perturb_and_clip(x0)

            try:
                res = minimize(
                    objective, x0,
                    method='SLSQP',
                    bounds=bnds,
                    constraints=cons_full,
                    jac=objective_jac,
                    options={'maxiter': 25000, 'ftol': 1e-15, 'disp': False}
                )
                curr_sum = -res.fun
                c_vals = constraints_val(res.x)
                if np.min(c_vals) >= -1e-6 and curr_sum > best_sum:
                    best_sum = curr_sum
                    best_x = res.x.copy()
            except Exception:
                pass

    # ============================
    # Phase 4: Final LP refinement
    # ============================
    if best_x is not None:
        centers = np.column_stack((best_x[0::3], best_x[1::3]))
        radii_lp = solve_lp_radii(centers)
        if radii_lp is not None:
            lp_sum = np.sum(radii_lp)
            if lp_sum > best_sum:
                best_sum = lp_sum
                best_x = np.zeros(3 * N)
                best_x[0::3] = centers[:, 0]
                best_x[1::3] = centers[:, 1]
                best_x[2::3] = radii_lp

    # ============================
    # Extract and validate results
    # ============================
    centers = np.zeros((N, 2))
    radii = np.zeros(N)
    for i in range(N):
        centers[i, 0] = best_x[3 * i]
        centers[i, 1] = best_x[3 * i + 1]
        radii[i] = best_x[3 * i + 2]

    # Safety validation and repair
    if not is_valid_config(centers, radii):
        for _ in range(200):
            radii *= 0.995
            for i in range(N):
                centers[i, 0] = np.clip(centers[i, 0], radii[i], 1.0 - radii[i])
                centers[i, 1] = np.clip(centers[i, 1], radii[i], 1.0 - radii[i])
            if is_valid_config(centers, radii):
                break

    return centers, radii, float(np.sum(radii))
