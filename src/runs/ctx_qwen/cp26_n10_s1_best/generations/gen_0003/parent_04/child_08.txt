1. **Use alternating optimization (LP for radii + optimization for centers)**: This is key - fixing centers and solving the LP gives optimal radii for those positions, which is a better subproblem than trying to optimize both simultaneously.

2. **Implement trust-constr method**: SLSQP can get stuck; trust-constr handles nonlinear constraints more robustly.

3. **Better initialization with force-directed layout**: Run longer, more careful force simulations that push circles to boundaries efficiently.

4. **Deflation-reinflation strategy**: Shrink radii, let circles repack into denser configurations, then expand again.

5. **Multiple diverse strategies**: Hexagonal lattices with different row counts, random perturbations, and grid-based starts.

The key insight is that the LP phase gives a provably optimal radius assignment for fixed centers, which provides a stronger signal for the subsequent center optimization.

import numpy as np
from scipy.optimize import minimize, linprog

N_CIRCLES = 26

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints_ineq(x):
    """Compute all boundary and non-overlap constraints as a vector >= 0."""
    cx = x[0::3]
    cy = x[1::3]
    cr = x[2::3]
    
    # Boundary constraints: 4 * N
    b = np.concatenate([cx - cr, 1.0 - cx - cr, cy - cr, 1.0 - cy - cr])
    
    # Overlap constraints: N*(N-1)/2
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dr = cr[:, None] + cr[None, :]
    
    idx = np.tril_indices(N_CIRCLES, -1)
    o = dx[idx]**2 + dy[idx]**2 - dr[idx]**2
    
    return np.concatenate([b, o])

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [0, 0.5]."""
    return [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N_CIRCLES

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming.
    Maximizes sum of radii subject to non-overlap and boundary constraints."""
    n = N_CIRCLES
    c_obj = -np.ones(n)
    
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x_i, 1-x_i, y_i, 1-y_i)
    for i in range(n):
        bound = min(centers[i, 0], 1.0 - centers[i, 0], 
                    centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = bound
        idx += 1
    
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], 
                          centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dist
            idx += 1
    
    bounds = [(0.0, 0.5)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def hex_init(r0, seed, row_pattern=None):
    """Generate a perturbed hexagonal lattice initialization."""
    np.random.seed(seed)
    x = np.zeros(3 * N_CIRCLES)
    count = 0
    y = r0
    row = 0
    
    if row_pattern is None:
        # Determine a good row pattern for 26 circles
        row_pattern = [6, 5, 6, 5, 4]
    
    for row_count in row_pattern:
        if count >= N_CIRCLES:
            break
        x_start = r0 + (r0 if row % 2 == 1 else 0.0)
        col = 0
        while count < N_CIRCLES and col < row_count:
            cx = x_start + col * 2 * r0
            if cx + r0 > 1.0:
                break
            x[3*count] = cx + np.random.uniform(-0.002, 0.002)
            x[3*count+1] = y + np.random.uniform(-0.002, 0.002)
            x[3*count+2] = r0 + np.random.uniform(-0.002, 0.002)
            count += 1
            col += 1
        y += np.sqrt(3.0) * r0
        row += 1
    
    while count < N_CIRCLES:
        x[3*count] = np.random.uniform(0.15, 0.85)
        x[3*count+1] = np.random.uniform(0.15, 0.85)
        x[3*count+2] = 0.05
        count += 1
    return x

def force_directed_layout(seed, steps=3000):
    """Generate initial layout using repulsive forces with boundary awareness."""
    np.random.seed(seed)
    n = N_CIRCLES
    
    # Start with random positions
    cx = np.random.uniform(0.2, 0.8, n)
    cy = np.random.uniform(0.2, 0.8, n)
    cr = np.full(n, 0.06)
    
    for step in range(steps):
        fx = np.zeros(n)
        fy = np.zeros(n)
        
        # Repulsive forces between circles
        for i in range(n):
            for j in range(i + 1, n):
                dx = cx[j] - cx[i]
                dy = cy[j] - cy[i]
                d = np.hypot(dx, dy)
                if d < 0.3 and d > 1e-8:
                    f = 0.008 / (d * d + 0.001)
                    fx[i] -= f * dx
                    fy[i] -= f * dy
                    fx[j] += f * dx
                    fy[j] += f * dy
        
        # Wall repulsion - push toward walls to utilize space
        for i in range(n):
            # Left wall
            if cx[i] < 0.3:
                fx[i] -= 0.02 * (0.3 - cx[i])
            elif cx[i] < 0.5:
                fx[i] -= 0.005
            elif cx[i] < 0.7:
                fx[i] += 0.005
            else:
                fx[i] += 0.02 * (cx[i] - 0.7)
                
            # Top/bottom wall
            if cy[i] < 0.3:
                fy[i] -= 0.02 * (0.3 - cy[i])
            elif cy[i] < 0.5:
                fy[i] -= 0.005
            elif cy[i] < 0.7:
                fy[i] += 0.005
            else:
                fy[i] += 0.02 * (cy[i] - 0.7)
        
        alpha = 0.05 * max(0.01, 1.0 - step / steps)
        cx += fx * alpha
        cy += fy * alpha
        cx = np.clip(cx, 0.02, 0.98)
        cy = np.clip(cy, 0.02, 0.98)
    
    x = np.zeros(3 * n)
    x[0::3] = cx
    x[1::3] = cy
    x[2::3] = cr
    return x

def grid_init(seed):
    """Generate a grid-based initialization."""
    np.random.seed(seed)
    x = np.zeros(3 * N_CIRCLES)
    count = 0
    for row in range(6):
        for col in range(5):
            if count >= N_CIRCLES:
                break
            cx = 0.1 + col * 0.18 + np.random.uniform(-0.008, 0.008)
            cy = 0.09 + row * 0.165 + np.random.uniform(-0.008, 0.008)
            x[3*count] = np.clip(cx, 0.05, 0.95)
            x[3*count+1] = np.clip(cy, 0.05, 0.95)
            x[3*count+2] = 0.085
            count += 1
    return x

def run_alternating_optimization(centers, radii, iterations=8):
    """Alternating optimization: LP for radii, NLP for centers."""
    n = N_CIRCLES
    
    for it in range(iterations):
        # Step 1: Optimize radii for fixed centers using LP
        new_radii = solve_lp_radii(centers)
        if new_radii is None:
            break
        radii = new_radii
        current_sum = np.sum(radii)
        
        # Step 2: Optimize centers for these radii using NLP
        x = np.zeros(3 * n)
        x[0::3] = centers[:, 0]
        x[1::3] = centers[:, 1]
        x[2::3] = radii
        
        # Add small perturbation to break symmetry
        if it > 0:
            x[0::3] += np.random.normal(0, 5e-5, n)
            x[1::3] += np.random.normal(0, 5e-5, n)
            # Clip to bounds
            for k in range(n):
                r = x[3*k + 2]
                x[3*k] = np.clip(x[3*k], r + 1e-8, 1.0 - r - 1e-8)
                x[3*k + 1] = np.clip(x[3*k + 1], r + 1e-8, 1.0 - r - 1e-8)
        
        cons = {'type': 'ineq', 'fun': constraints_ineq}
        bnds = get_bounds()
        
        try:
            res = minimize(objective, x, method='SLSQP', bounds=bnds,
                          constraints=cons,
                          options={'maxiter': 10000, 'ftol': 1e-14, 'disp': False})
            centers[:, 0] = res.x[0::3]
            centers[:, 1] = res.x[1::3]
            radii = res.x[2::3]
        except Exception:
            break
    
    return centers, radii, np.sum(radii)

def validate_and_repair(centers, radii):
    """Validate and repair the solution to ensure strict compliance."""
    n = N_CIRCLES
    valid = True
    
    for i in range(n):
        if (radii[i] < 0 or

                    valid = False
                    break
            if not valid:
