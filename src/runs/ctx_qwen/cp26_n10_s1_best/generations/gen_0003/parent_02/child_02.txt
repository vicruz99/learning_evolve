import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def objective(vars):
    """Minimize negative sum of radii."""
    return -np.sum(vars[2::3])

def constraint_func(vars):
    """Vectorized inequality constraints: g(vars) >= 0."""
    n = N_CIRCLES
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    # Boundary constraints
    b = np.concatenate([
        x - r,
        1.0 - x - r,
        y - r,
        1.0 - y - r
    ])
    
    # Overlap constraints: dist - (r_i + r_j) >= 0
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist = np.sqrt(dx**2 + dy**2)
    r_sum = r[:, None] + r[None, :]
    
    i, j = np.triu_indices(n, 1)
    o = dist[i, j] - r_sum[i, j]
    
    return np.concatenate([b, o])

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [0, 0.5]."""
    b = []
    for _ in range(N_CIRCLES):
        b.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
    return b

def generate_hex_points(n, r0, angle_deg):
    """Generate hexagonal lattice points, optionally rotated."""
    pts = []
    y = r0
    row = 0
    while len(pts) < n + 15:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0 + 1e-6:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        if y > 1.0 + r0:
            break
            
    pts = np.array(pts[:n + 15])
    
    # Rotate around center (0.5, 0.5)
    if angle_deg != 0:
        rad = np.deg2rad(angle_deg)
        c, s = np.cos(rad), np.sin(rad)
        center = np.array([0.5, 0.5])
        pts = (pts - center) @ np.array([[c, -s], [s, c]]) + center
        
    # Filter points strictly inside the square
    mask = (pts[:, 0] >= 0.01) & (pts[:, 0] <= 0.99) & (pts[:, 1] >= 0.01) & (pts[:, 1] <= 0.99)
    pts = pts[mask]
    
    # Pad if rotation/filtering removed too many points
    while len(pts) < n:
        pts = np.vstack([pts, [np.random.uniform(0.1, 0.9), np.random.uniform(0.1, 0.9)]])
        
    return pts[:n]

def force_directed_init(n, seed):
    """Generate initial configuration using repulsive force simulation."""
    np.random.seed(seed)
    centers = np.random.uniform(0.15, 0.85, (n, 2))
    
    for _ in range(1000):
        forces = np.zeros_like(centers)
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[j, 0] - centers[i, 0]
                dy = centers[j, 1] - centers[i, 1]
                dist = np.hypot(dx, dy)
                if dist < 0.25 and dist > 1e-6:
                    f = 0.008 / (dist**2 + 0.001)
                    fx, fy = f * dx, f * dy
                    forces[i] -= [fx, fy]
                    forces[j] += [fx, fy]
        for i in range(n):
            if centers[i, 0] < 0.12: forces[i, 0] += 0.03
            elif centers[i, 0] > 0.88: forces[i, 0] -= 0.03
            if centers[i, 1] < 0.12: forces[i, 1] += 0.03
            elif centers[i, 1] > 0.88: forces[i, 1] -= 0.03
            
        centers += forces * 0.05
        centers = np.clip(centers, 0.05, 0.95)
    return centers

def is_valid(centers, radii, tol=1e-10):
    """Check strict feasibility of a packing configuration."""
    n = centers.shape[0]
    if np.any(radii < -tol):
        return False
        
    # Boundary check
    if np.any(centers[:, 0] - radii < -tol) or np.any(centers[:, 0] + radii > 1.0 + tol):
        return False
    if np.any(centers[:, 1] - radii < -tol) or np.any(centers[:, 1] + radii > 1.0 + tol):
        return False
        
    # Overlap check
    dx = centers[:, 0, None] - centers[:, 0]
    dy = centers[:, 1, None] - centers[:, 1]
    dist = np.sqrt(dx**2 + dy**2)
    r_sum = radii[:, None] + radii[None, :]
    i, j = np.triu_indices(n, 1)
    if np.any(dist[i, j] < r_sum[i, j] - tol):
        return False
    return True
