import numpy as np
from scipy.optimize import minimize

N = 26
TRIL = np.tril_indices(N, -1)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Vectorized inequality constraints: g(x) >= 0."""
    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]
    
    # Boundary constraints
    c = np.concatenate([
        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs
    ])
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    c = np.concatenate([c, dx[TRIL]**2 + dy[TRIL]**2 - dr[TRIL]**2])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-7, 0.5]."""
    return [(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)] * N

def get_safe_radii(centers):
    """Compute strictly feasible initial radii based on nearest neighbors/walls."""
    n = centers.shape[0]
    radii = np.zeros(n)
    for i in range(n):
        d_wall = min(centers[i, 0], 1.0 - centers[i, 0], 
                     centers[i, 1], 1.0 - centers[i, 1])
        d_min = d_wall
        for j in range(n):
            if i != j:
                d = np.hypot(centers[i, 0] - centers[j, 0], 
                             centers[i, 1] - centers[j, 1])
                if d < d_min:
                    d_min = d
        radii[i] = max(1e-6, d_min * 0.35)
    return radii

def generate_hex(r0, angle_deg):
    """Generate hexagonal lattice points, optionally rotated."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 20:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        if y > 1.0 + r0:
            break
            
    pts = np.array(pts[:N + 20])
    
    if angle_deg != 0:
        rad = np.deg2rad(angle_deg)
        c, s = np.cos(rad), np.sin(rad)
        center = np.array([0.5, 0.5])
        pts = (pts - center) @ np.array([[c, -s], [s, c]]) + center
        
    mask = (pts[:, 0] >= 0.01) & (pts[:, 0] <= 0.99) & \
           (pts[:, 1] >= 0.01) & (pts[:, 1] <= 0.99)
    pts = pts[mask]
    
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.1, 0.9, 2)])
        
    return pts[:N]

def force_relax(centers, steps=80):
    """Force-directed layout to resolve overlaps and spread points."""
    n = centers.shape[0]
    radii = get_safe_radii(centers)
    for _ in range(steps):
        forces = np.zeros_like(centers)
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[j, 0] - centers[i, 0]
                dy = centers[j, 1] - centers[i, 1]
                d = np.hypot(dx, dy)
                if d < radii[i] + radii[j] and d > 1e-6:
                    shift = (radii[i] + radii[j] - d) * 0.5 / d
                    forces[i] -= [dx * shift, dy * shift]
                    forces[j] += [dx * shift, dy * shift]
        centers += forces * 0.1
        centers = np.clip(centers, radii, 1.0 - radii)
    return centers, radii

def is_valid(centers, radii):
    """Check strict feasibility against validator tolerance."""
    n = centers.shape[0]
    for i in range(n):
        if radii[i] < 1e-8:
            return False
        if centers[i, 0] - radii[i] < -1e-9 or centers[i, 0] + radii[i] > 1.0 + 1e-9:
            return False
        if centers[i, 1] - radii[i] < -1e-9 or centers[i, 1] + radii[i] > 1.0 + 1e-9:
            return False
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            if dist < radii[i] + radii[j] - 1e-9:
                return False
    return True

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = N
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    inits = []
    
    # 1. Rotated Hexagonal Lattices (varied densities & angles)
    for r0 in np.linspace(0.08, 0.11, 5):
        for ang in range(0, 91, 15):
            pts = generate_hex(r0, ang)
            pts += np.random.uniform(-0.005, 0.005, pts.shape)
            c, r = force_relax(pts, 60)
            inits.append((c, r))
            
    # 2. Force-Directed Random Layouts
    for _ in range(12):
        pts = np.random.uniform(0.15, 0.85, (n, 2))
        c, r = force_relax(pts, 100)
        inits.append((c, r))
        
    # Phase 1: Multi-start optimization
    for c_init, r_init in inits:
        x0 = np.zeros(3 * n)
        x0[0::3] = c_init[:, 0]
        x0[1::3] = c_init[:, 1]
        x0[2::3] = r_init
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 25000, 'ftol': 1e-14, 'disp': False})
            if not np.isnan(res.fun):
                curr_sum = -res.fun
                if curr_sum > best_sum:
                    c_opt = np.column_stack((res.x[0::3], res.x[1::3]))
                    r_opt = res.x[2::3]
                    if is_valid(c_opt, r_opt):
                        best_sum = curr_sum
                        best_centers = c_opt.copy()
                        best_radii = r_opt.copy()
        except Exception:
            continue

    # Phase 2: Basin-hopping refinement to escape local minima
    if best_centers is not None:
        for it in range(50):
            noise = 0.003 * (0.92 ** it)
            c_pert = best_centers + np.random.normal(0, noise, best_centers.shape)
            c_pert = np.clip(c_pert, 0.01, 0.99)
            r_pert = best_radii * (0.98 + 0.015 * np.random.rand(n))
            
            x0 = np.zeros(3 * n)
            x0[0::3] = c_pert[:, 0]
            x0[1::3] = c_pert[:, 1]
            x0[2::3] = r_pert
            
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 20000, 'ftol': 1e-14, 'disp': False})
                if not np.isnan(res.fun):
                    curr_sum = -res.fun
                    if curr_sum > best_sum:
                        c_opt = np.column_stack((res.x[0::3], res.x[1::3]))
                        r_opt = res.x[2::3]
                        if is_valid(c_opt, r_opt):
                            best_sum = curr_sum
                            best_centers = c_opt.copy()
                            best_radii = r_opt.copy()
            except Exception:
                pass
                
        # Phase 3: Uniform expansion push
        for _ in range(5):
            c_expand = best_centers.copy()
            r_expand = best_radii * 1.001
            x0 = np.zeros(3 * n)
            x0[0::3] = c_expand[:, 0]
            x0[1::3] = c_expand[:, 1]
            x0[2::3] = r_expand
            
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 15000, 'ftol': 1e-14, 'disp': False})
                if not np.isnan(res.fun):
                    curr_sum = -res.fun
                    if curr_sum > best_sum:
                        c_opt = np.column_stack((res.x[0::3], res.x[1::3]))
                        r_opt = res.x[2::3]
                        if is_valid(c_opt, r_opt):
                            best_sum = curr_sum
                            best_centers = c_opt.copy()
                            best_radii = r_opt.copy()
            except Exception:
                pass

    # Fallback (should rarely be reached)
    if best_centers is None:
        best_centers = generate_hex(0.09, 0)
        best_radii = get_safe_radii(best_centers)
        best_sum = np.sum(best_radii)
        
    return best_centers, best_radii, float(best_sum)
