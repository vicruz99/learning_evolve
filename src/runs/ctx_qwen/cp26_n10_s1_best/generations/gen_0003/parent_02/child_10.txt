The key improvements I'll make:

1. **Better constraint formulation**: Use non-squared distance constraints (`dist - r_i - r_j >= 0`) instead of squared distances. This provides non-zero gradients near contact points, helping the optimizer find better solutions.

2. **More diverse initializations**: Add more rotation angles to hexagonal patterns, more force-directed seeds, and additional grid variations.

3. **Improved alternating optimization**: More LP iterations followed by center refinement. LP optimally assigns radii to fixed centers, then NLP re-optimizes positions.

4. **Better perturbation strategy**: Exponentially decaying noise scales, combined with radius deflation to allow centers to move into tighter configurations.

5. **Longer optimization runs**: More iterations per optimization call to ensure convergence.

6. **Dual constraint approaches**: Try both squared and non-squared constraint formulations to leverage their different gradient behaviors.

import numpy as np
from scipy.optimize import minimize, linprog

N_CIRCLES = 26

def objective(vars):
    """Minimize negative sum of radii."""
    return -np.sum(vars[2::3])

def constraint_func(vars):
    """Vectorized inequality constraints using non-squared distances."""
    n = N_CIRCLES
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    # Boundary constraints
    b = np.concatenate([
        x - r,
        1.0 - x - r,
        y - r,
        1.0 - y - r
    ])
    
    # Overlap constraints using distance (better gradients at contact)
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist = np.sqrt(dx**2 + dy**2)
    r_sum = r[:, None] + r[None, :]
    
    i, j = np.tril_indices(n, -1)
    o = dist[i, j] - r_sum[i, j]
    
    return np.concatenate([b, o])

def constraint_func_squared(vars):
    """Vectorized inequality constraints using squared distances."""
    n = N_CIRCLES
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    b = np.concatenate([
        x - r,
        1.0 - x - r,
        y - r,
        1.0 - y - r
    ])
    
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dr = r[:, None] + r[None, :]
    
    i, j = np.tril_indices(n, -1)
    o = dx[i, j]**2 + dy[i, j]**2 - dr[i, j]**2
    
    return np.concatenate([b, o])

def get_bounds():
    b = []
    for _ in range(N_CIRCLES):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-8, 0.5)])
    return b

def solve_lp_radii(centers):
    """Given centers, find optimal radii using LP."""
    n = N_CIRCLES
    c_obj = -np.ones(n)
    
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    for i in range(n):
        bound = min(centers[i, 0], 1.0 - centers[i, 0],
                    centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = bound
        idx += 1
    
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0],
                           centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dist
            idx += 1
    
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def generate_hex_init(n, r0, angle=0.0):
    """Generate hexagonal lattice initialization."""
    pts = []
    y = r0
    row = 0
    while len(pts) < n + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x + r0 <= 1.0:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        if y + r0 > 1.0:
            break
    
    pts = np.array(pts[:n + 10])
    
    if angle != 0:
        c, s = np.cos(angle), np.sin(angle)
        center = np.array([0.5, 0.5])
        pts = (pts - center) @ np.array([[c, -s], [s, c]]) + center
    
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    while len(pts) < n:
        pts = np.vstack([pts, np.random.uniform(0.1, 0.9, 2)])
    
    return pts[:n]

def force_directed_init(n, seed):
    """Force-directed layout for initial positions."""
    np.random.seed(seed)
    pts = np.random.uniform(0.15, 0.85, (n, 2))
    
    for _ in range(1500):
        forces = np.zeros_like(pts)
        for i in range(n):
            for j in range(i + 1, n):
                dx = pts[j, 0] - pts[i, 0]
                dy = pts[j, 1] - pts[i, 1]
                d = np.hypot(dx, dy)
                if d < 0.3 and d > 1e-6:
                    f = 0.015 / (d**2 + 0.001)
                    fx, fy = f * dx, f * dy
                    forces[i] -= [fx, fy]
                    forces[j] += [fx, fy]
        for i in range(n):
            if pts[i, 0] < 0.1: forces[i, 0] += 0.03
            elif pts[i, 0] > 0.9: forces[i, 0] -= 0.03
            if pts[i, 1] < 0.1: forces[i, 1] += 0.03
            elif pts[i, 1] > 0.9: forces[i, 1] -= 0.03
        
        pts += forces * 0.06
        pts = np.clip(pts, 0.05, 0.95)
    
    return pts

def get_safe_radii(centers):
    """Compute feasible initial radii."""
    n = centers.shape[0]
    radii = np.zeros(n)
    for i in range(n):
        min_d = min(centers[i, 0], 1.0 - centers[i, 0],
                    centers[i, 1], 1.0 - centers[i, 1])
        for j in range(n):
            if i != j:
                d = np.hypot(centers[i, 0] - centers[j, 0],
                             centers[i, 1] - centers[j, 1])
                min_d = min(min_d, d / 2.0)
        radii[i] = min_d * 0.5
    return radii

def is_valid(centers, radii, tol=1e-9):
    """Check validity of packing."""
    n = centers.shape[0]
    for i in range(n):
        if radii[i] < 0:
            return False
        x, y, r = centers[i, 0], centers[i, 1], radii[i]
        if x - r < -tol or x + r > 1 + tol or y - r < -tol or y + r > 1 + tol:
            return False
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0],
                        centers[i, 1] - centers[j, 1])
            if d < radii[i] + radii[j] - tol:
                return False
    return True

def run_packing():

    best_radii = None

            pts = generate_hex_init(n, r0, angle)
            pts += np.random.uniform(-0.005, 0.005, pts.shape)
            inits.append(pts.copy())

    for seed in range(20):

                    break
                x = 0.08 + c * spacing_x

        pts = np.array(pts[:n])
        pts += np.random.uniform(-0.015, 0.015, pts.shape)
        inits.append(pts.copy())
    
    # 4. Staggered row patterns (like solution 7)

        [6, 6, 5, 5, 4],
    ]
    for pat in row_patterns:

            pts = np.zeros((n, 2))
            idx = 0
            y = r0
            for i, count in enumerate(pat):
                x_start = r0 + (r0 if i % 2 == 1 else 0.0)
                for k in range(count):
                    if idx < n:
                        x = x_start + k * 2.0 * r0
                        pts[idx] = [x, y]
                        idx += 1
                y += np.sqrt(3.0) * r0
