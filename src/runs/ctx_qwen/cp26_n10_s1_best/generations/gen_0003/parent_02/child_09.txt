   - Multiple hexagonal lattice patterns with different row counts

2. **LP-based Radius Optimization**: For fixed centers, solve LP to find optimal radii - this gives a convex subproblem that always finds the best radii.

3. **Alternating Optimization**: Alternate between optimizing centers (SLSQP) and radii (LP) to escape local minima.

4. **Better constraint formulation**: Use unsquared distance constraints for better gradient behavior near tight packings.

5. **More aggressive refinement**: Multiple rounds of perturbation with decreasing noise, combined with the LP radius step.

import numpy as np
from scipy.optimize import minimize, linprog

N_CIRCLES = 26

def objective(vars):
    """Minimize negative sum of radii."""
    return -np.sum(vars[2::3])

def constraint_func(vars):
    """Vectorized inequality constraints: g(vars) >= 0."""
    n = N_CIRCLES
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    # Boundary constraints: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    b = np.concatenate([
        x - r,
        1.0 - x - r,
        y - r,
        1.0 - y - r
    ])
    
    # Overlap constraints: dist^2 - (r1+r2)^2 >= 0
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dr = r[:, None] + r[None, :]
    
    i, j = np.tril_indices(n, -1)
    o = dx[i, j]**2 + dy[i, j]**2 - dr[i, j]**2
    
    return np.concatenate([b, o])

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-8, 0.5]."""
    b = []
    for _ in range(N_CIRCLES):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-8, 0.5)])
    return b

def solve_lp_radii(centers):
    """Optimally find radii for fixed centers using Linear Programming."""
    n = N_CIRCLES
    c_obj = -np.ones(n)  # Maximize sum of radii
    
    num_ineq = n * 4 + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        x_val, y_val = centers[i]
        bound = min(x_val, 1.0 - x_val, y_val, 1.0 - y_val)
        for wall in range(4):
            A_ub[idx, i] = 1.0
            b_ub[idx] = bound
            idx += 1
    
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dist
            idx += 1
    
    bounds = [(0.0, 0.5)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def force_directed_layout(centers, steps=1500):
    """Force-directed layout with strong boundary and repulsion forces."""
    n = centers.shape[0]
    for step in range(steps):
        forces = np.zeros_like(centers)
        alpha = 1.0 - step / steps
        
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[j, 0] - centers[i, 0]
                dy = centers[j, 1] - centers[i, 1]
                d = np.hypot(dx, dy)
                if d < 0.35 and d > 1e-8:
                    f = 0.008 * alpha / (d**2 + 0.001)
                    fx, fy = f * dx, f * dy
                    forces[i] -= [fx, fy]
                    forces[j] += [fx, fy]
        
        # Strong boundary attraction to push circles to edges
        for i in range(n):
            x, y = centers[i]
            # Repel from center, attract to boundaries
            dx_center = x - 0.5
            dy_center = y - 0.5
            forces[i, 0] += 0.003 * alpha * dx_center / (np.abs(dx_center) + 0.1)
            forces[i, 1] += 0.003 * alpha * dy_center / (np.abs(dy_center) + 0.1)
            
            # Boundary repulsion
            margin = 0.03
            if x < margin: forces[i, 0] += 0.02 * (margin - x) / margin
            elif x > 1.0 - margin: forces[i, 0] -= 0.02 * (x - (1.0 - margin)) / margin
            if y < margin: forces[i, 1] += 0.02 * (margin - y) / margin
            elif y > 1.0 - margin: forces[i, 1] -= 0.02 * (y - (1.0 - margin)) / margin
        
        centers += forces * 0.08 * alpha
        centers = np.clip(centers, 0.01, 0.99)
    return centers

def generate_hex_init(r0, row_counts=None, angle=0.0):
    """Generate hexagonal lattice with specific row counts."""
    if row_counts is None:
        row_counts = [5, 6, 5, 6, 4]
    
    centers = np.zeros((N_CIRCLES, 2))
    idx = 0
    y = r0
    y_step = np.sqrt(3.0) * r0
    
    for row_i, count in enumerate(row_counts):

            if idx >= N_CIRCLES:
                break
            x = x_start + k * 2.0 * r0
            centers[idx] = [x, y]
            idx += 1
        y += y_step
    
    # Fill remaining
    while idx < N_CIRCLES:
        centers[idx] = [np.random.uniform(0.15, 0.85), np.random.uniform(0.15, 0.85)]
        idx += 1

        s = np.sin(angle)
        cx, cy = 0.5, 0.5
        dx = centers[:, 0] - cx
        dy = centers[:, 1] - cy
        centers[:, 0] = dx * c - dy * s + cx
        centers[:, 1] = dx * s + dy * c + cy
        centers = np.clip(centers, 0.02, 0.98)
    
    return centers

def make_feasible(centers, radii):
    """Ensure configuration is strictly feasible."""
    n = N_CIRCLES
    # Clip centers to valid range
    for i in range(n):
        r = max(radii[i], 1e-8)
        centers[i, 0] = np.clip(centers[i, 0], r, 1.0 - r)
        centers[i, 1] = np.clip(centers[i, 1], r, 1.0 - r)
    
    # Resolve overlaps by shrinking radii
    for iteration in range(500):
        max_violation = 0
        for i in range(n):
            for j in range(i + 1, n):
                d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                if d < radii[i] + radii[j] - 1e-10:
                    violation = radii[i] + radii[j] - d
                    max_violation = max(max_violation, violation)
                    # Push apart

                        shift = violation * 0.5
                        nx, ny = (centers[j, 0] - centers[i, 0]) / d, (centers[j, 1] - centers[i, 1]) / d
                        centers[i, 0] -= shift * nx

        r = max(radii[i], 1e-8)
        centers[i, 0] = np.clip(centers[i, 0], r, 1.0 - r)
        centers[i, 1] = np.clip(centers[i, 1], r, 1.0 - r)
    
    return centers, radii

def check_validity(centers, radii, tol=1e-10):
    """Check if configuration is valid."""
    n = N_CIRCLES
    for i in range(n):
        if radii[i] < 0:
            return False
        x, y, r = centers[i, 0], centers[i, 1], radii[i]
        if x - r < -tol or x + r > 1.0 + tol or y - r < -tol or y + r > 1.0 + tol:
            return False
    
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            if d < radii[i] + radii[j] - tol:
                return False
    return True
