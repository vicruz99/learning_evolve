N_CIRCLES = 26

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraint_func(x):
    """Vectorized inequality constraints: g(x) >= 0."""
    n = N_CIRCLES
    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]
    
    # Boundary constraints
    c = np.concatenate([
        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs
    ])
    
    # Overlap constraints (squared distance - (r_i + r_j)^2 >= 0)
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    i_idx, j_idx = np.tril_indices(n, -1)
    c = np.concatenate([c, dx[i_idx, j_idx]**2 + dy[i_idx, j_idx]**2 - dr[i_idx, j_idx]**2])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [0, 0.5]."""
    bounds = []
    for _ in range(N_CIRCLES):
        bounds.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
    return bounds

def solve_radii_lp(centers):
    """Given fixed centers, solve LP to maximize sum of radii."""
    n = N_CIRCLES
    c_obj = -np.ones(n)

    A_ub = np.zeros((num_constraints, n))
    b_ub = np.zeros(num_constraints)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        x_val, y_val = centers[i, 0], centers[i, 1]
        bound = min(x_val, 1.0 - x_val, y_val, 1.0 - y_val)
        if bound < 0:
            return None
        A_ub[idx, i] = 1.0
        b_ub[idx] = bound
        idx += 1
    
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dist
            idx += 1
    
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success and res.fun > -np.inf:
        return res.x
    return None

                    # Push apart proportionally to overlap
                    shift = (min_dist - dist) * 0.5
                    forces[i, 0] -= shift * dx / dist
                    forces[i, 1] -= shift * dy / dist
                    forces[j, 0] += shift * dx / dist
                    forces[j, 1] += shift * dy / dist

            margin_x_max = 1.0 - centers[i, 0] - radii[i]

                forces[i, 1] += 0.5 * abs(margin_y_min)

def generate_hex_init(row_counts, r_start=0.09):

    radii = np.full(n, r_start)

        for k in range(count):
            if idx < n:
                x = x_start + k * 2 * r_start
                centers[idx] = [x, y]
                idx += 1
        y += y_step
    
    # Fill remaining if needed
    while idx < n:
        centers[idx] = np.random.uniform(r_start + 0.01, 1.0 - r_start - 0.01, 2)
        radii[idx] = r_start * 0.5
        idx += 1
    
    return centers, radii

def rotate_grid(centers, angle):
    """Rotates a grid of centers around the square center."""
    cx, cy = 0.5, 0.5
    cos_a, sin_a = np.cos(angle), np.sin(angle)
    dx = centers[:, 0] - cx
    dy = centers[:, 1] - cy
    new_x = dx * cos_a - dy * sin_a + cx
    new_y = dx * sin_a + dy * cos_a + cy
    return np.column_stack([new_x, new_y])

def is_valid_packing(centers, radii, tol=1e-10):
    """Check strict feasibility of a packing configuration."""
    n = centers.shape[0]
    for i in range(n):
        if radii[i] < -tol:
            return False
        if centers[i, 0] < radii[i] - tol or centers[i, 0] > 1.0 - radii[i] + tol:
            return False
        if centers[i, 1] < radii[i] - tol or centers[i, 1] > 1.0 - radii[i] + tol:
            return False
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            if dist < radii[i] + radii[j] - tol:
                return False
    return True

def run_slsqp(x0, bounds, cons, maxiter=15000):
    """Run SLSQP optimization with error handling."""
    try:
        res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14, 'disp': False})
        return res
    except Exception:
        return None

        x[3*i + 1] = np.clip(x[3*i + 1], r, 1.0 - r)

    return x
