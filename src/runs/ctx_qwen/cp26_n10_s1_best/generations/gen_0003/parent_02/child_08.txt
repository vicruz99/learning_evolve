import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def objective(vars):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(vars[2::3])

def constraint_func(vars):
    """Vectorized inequality constraints: g(vars) >= 0."""
    n = N_CIRCLES
    cx = vars[0::3]
    cy = vars[1::3]
    cr = vars[2::3]
    
    # Boundary constraints: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    b = np.concatenate([
        cx - cr,
        1.0 - cx - cr,
        cy - cr,
        1.0 - cy - cr
    ])
    
    # Overlap constraints: dist^2 - (r_i + r_j)^2 >= 0
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dr = cr[:, None] + cr[None, :]
    
    i_idx, j_idx = np.tril_indices(n, -1)
    o = dx[i_idx, j_idx]**2 + dy[i_idx, j_idx]**2 - dr[i_idx, j_idx]**2
    
    return np.concatenate([b, o])

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-7, 0.5]."""
    b = []
    for _ in range(N_CIRCLES):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)])
    return b

def generate_hex_points(n, r0, angle_deg):
    """Generate hexagonal lattice points, optionally rotated around center."""
    pts = []
    y = r0
    row = 0
    while len(pts) < n + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        if y > 1.0 + r0:
            break
            
    pts = np.array(pts[:n + 10])
    
    # Rotate around center (0.5, 0.5)
    if angle_deg != 0:
        rad = np.deg2rad(angle_deg)
        c, s = np.cos(rad), np.sin(rad)
        center = np.array([0.5, 0.5])
        pts = (pts - center) @ np.array([[c, -s], [s, c]]) + center
        
    # Filter points strictly inside the square
    mask = (pts[:, 0] >= 0.01) & (pts[:, 0] <= 0.99) & (pts[:, 1] >= 0.01) & (pts[:, 1] <= 0.99)
    pts = pts[mask]
    
    # Pad if rotation/filtering removed too many points
    while len(pts) < n:
        pts = np.vstack([pts, [np.random.uniform(0.1, 0.9), np.random.uniform(0.1, 0.9)]])
        
    return pts[:n]

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    n = N_CIRCLES
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraint_func}
    
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    inits = []
    
    # 1. Rotated Hexagonal Lattices (varied densities & angles)
    for r0 in [0.082, 0.088, 0.092, 0.098, 0.102]:
        for angle in [0, 5, 10, 15, 20, 25, 30, -10, -20, -30]:
            pts = generate_hex_points(n, r0, angle)
            pts += np.random.uniform(-0.004, 0.004, pts.shape)
            inits.append(pts.copy())
            
    # 2. Perturbed Grids
    for _ in range(8):
        pts = []
        for r in range(6):
            for c in range(5):
                if len(pts) >= n: break
                pts.append([0.08 + c * 0.18, 0.08 + r * 0.16])
        pts = np.array(pts[:n]) + np.random.uniform(-0.015, 0.015, (n, 2))
        inits.append(pts.copy())
        
    # 3. Force-Directed Layouts (captures non-lattice structures)
    for _ in range(6):
        pts = np.random.uniform(0.15, 0.85, (n, 2))
        for _ in range(600):
            forces = np.zeros_like(pts)
            for i in range(n):
                for j in range(i + 1, n):
                    dx = pts[j, 0] - pts[i, 0]
                    dy = pts[j, 1] - pts[i, 1]
                    d = np.hypot(dx, dy)
                    if d < 0.25 and d > 1e-6:
                        f = 0.02 / (d**2)
                        fx, fy = f * dx, f * dy
                        forces[i] -= [fx, fy]
                        forces[j] += [fx, fy]
            for i in range(n):
                if pts[i, 0] < 0.1: forces[i, 0] += 0.03
                elif pts[i, 0] > 0.9: forces[i, 0] -= 0.03
                if pts[i, 1] < 0.1: forces[i, 1] += 0.03
                elif pts[i, 1] > 0.9: forces[i, 1] -= 0.03
            pts += forces * 0.05

        inits.append(pts.copy())

    # Phase 1: Multi-start optimization
    for pts in inits:
        # Start with small feasible radii to help SLSQP converge
        radii = np.full(n, 0.04)
        x0 = np.zeros(3 * n)
        x0[0::3] = pts[:, 0]
        x0[1::3] = pts[:, 1]
        x0[2::3] = radii

            res = minimize(objective, x0, method='SLSQP', bounds=bounds,

                # Strict feasibility validation
                valid = True
                for i in range(n):

                            if d < r_opt[i] + r_opt[j] - 1e-9:
                                valid = False
                                break
                        if not valid: break
                
                if valid and curr_sum > best_sum:
                    best_sum = curr_sum
                    best_centers = c_opt.copy()
                    best_radii = r_opt.copy()
        except Exception:
            continue

    # Phase 2: Deflation-Reoptimization Refinement
    # Shrinking radii slightly allows centers to move into denser configurations

        for round in range(20):
            pert_c = best_centers + np.random.normal(0, 0.0012, best_centers.shape)
            pert_c = np.clip(pert_c, 0.01, 0.99)
            pert_r = best_radii * 0.988
            x0 = np.zeros(3 * n)
            x0[0::3] = pert_c[:, 0]
            x0[1::3] = pert_c[:, 1]
            x0[2::3] = pert_r
            
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 12000, 'ftol': 1e-13, 'disp': False})
                if not np.isnan(res.fun):
                    curr_sum = -res.fun
                    if curr_sum > best_sum:
                        c_opt = np.column_stack((res.x[0::3], res.x[1::3]))
                        r_opt = res.x[2::3]
                        
                        valid = True
                        for i in range(n):
                            if r_opt[i] < 1e-8 or c_opt[i,0] < r_opt[i] - 1e-9 or c_opt[i,0] > 1 - r_opt[i] + 1e-9 or \
                               c_opt[i,1] < r_opt[i] - 1e-9 or c_opt[i,1] > 1 - r_opt[i] + 1e-9:
                                valid = False
                                break
                        if valid:
                            for i in range(n):
                                for j in range(i + 1, n):
                                    d = np.hypot(c_opt[i,0] - c_opt[j,0], c_opt[i,1] - c_opt[j,1])
                                    if d < r_opt[i] + r_opt[j] - 1e-9:
                                        valid = False
                                        break
                                if not valid: break
                        if valid and curr_sum > best_sum:
                            best_sum = curr_sum
                            best_centers = c_opt.copy()
                            best_radii = r_opt.copy()
            except Exception:
                pass

    # Fallback (should rarely be reached)
    if best_centers is None:
        best_centers = generate_hex_points(n, 0.09, 0)
        best_radii = np.full(n, 0.04)
        best_sum = np.sum(best_radii)
        
    # Final safety shrink to guarantee strict compliance with validator
    for _ in range(100):
        valid = True
        for i in range(n):
            if best_radii[i] < 0: valid=False; break
            if best_centers[i,0] - best_radii[i] < -1e-12 or best_centers[i,0] + best_radii[i] > 1 + 1e-12: valid=False; break

            for i in range(n):
                for j in range(i + 1, n):
                    d = np.hypot(best_centers[i,0] - best_centers[j,0], best_centers[i,1] - best_centers[j,1])

                        valid=False; break
                if not valid: break
        if valid: break
        best_radii *= 0.9999
        best_centers[:, 0] = np.clip(best_centers[:, 0], best_radii, 1 - best_radii)
        best_centers[:, 1] = np.clip(best_centers[:, 1], best_radii, 1 - best_radii)
        
    return best_centers, best_radii, float(np.sum(best_radii))
