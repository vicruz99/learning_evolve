import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26
TRI_INDICES = np.tril_indices(N_CIRCLES, -1)

def get_constraints(x):
    """Vectorized inequality constraints: g(x) >= 0."""
    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]
    
    # Boundary constraints
    c = np.concatenate([
        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs
    ])
    
    # Overlap constraints (squared distance - sum_radii^2 >= 0)
    dx = xs[TRI_INDICES[0]] - xs[TRI_INDICES[1]]
    dy = ys[TRI_INDICES[0]] - ys[TRI_INDICES[1]]
    dr = rs[TRI_INDICES[0]] + rs[TRI_INDICES[1]]
    
    c = np.concatenate([c, dx**2 + dy**2 - dr**2])
    return c

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def get_bounds():
    """Variable bounds."""
    b = []
    for _ in range(N_CIRCLES):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)])
    return b

def generate_hex_init(n, r0, angle_deg, noise_scale=0.005):
    """Generate hexagonal lattice points with rotation and noise."""
    pts = []
    y = r0
    row = 0
    while len(pts) < n + 20:
        x = r0 + (row % 2) * r0
        while x <= 1.0 - r0:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
    pts = np.array(pts[:n + 20])
    
    # Rotate around center
    if angle_deg != 0:
        rad = np.deg2rad(angle_deg)
        c, s = np.cos(rad), np.sin(rad)
        center = np.array([0.5, 0.5])
        pts = (pts - center) @ np.array([[c, -s], [s, c]]) + center
        
    # Filter and pad
    mask = (pts[:, 0] >= 0.01) & (pts[:, 0] <= 0.99) & (pts[:, 1] >= 0.01) & (pts[:, 1] <= 0.99)
    pts = pts[mask]
    while len(pts) < n:
        pts = np.vstack([pts, [np.random.uniform(0.1, 0.9), np.random.uniform(0.1, 0.9)]])
        
    pts = np.random.permutation(pts[:n])
    pts += np.random.uniform(-noise_scale, noise_scale, pts.shape)
    return pts

def force_directed_init(n, seed):
    """Force-directed layout initialization."""
    np.random.seed(seed)
    pts = np.random.uniform(0.15, 0.85, (n, 2))
    for _ in range(600):
        forces = np.zeros_like(pts)
        for i in range(n):
            for j in range(i + 1, n):
                dx = pts[j, 0] - pts[i, 0]
                dy = pts[j, 1] - pts[i, 1]
                d = np.hypot(dx, dy)
                if d < 0.3 and d > 1e-6:
                    f = 0.012 / (d**2 + 0.0005)
                    fx, fy = f * dx, f * dy
                    forces[i] -= [fx, fy]
                    forces[j] += [fx, fy]
            # Boundary repulsion
            if pts[i, 0] < 0.12: forces[i, 0] += 0.02
            elif pts[i, 0] > 0.88: forces[i, 0] -= 0.02
            if pts[i, 1] < 0.12: forces[i, 1] += 0.02
            elif pts[i, 1] > 0.88: forces[i, 1] -= 0.02
        pts += forces * 0.06
        pts = np.clip(pts, 0.05, 0.95)
    return pts

    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': get_constraints}

    best_sum = -1.0

    inits = []
