import numpy as np
from scipy.optimize import minimize, linprog

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Vectorized inequality constraints g(x) >= 0."""
    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]
    
    # Boundary constraints
    c = np.concatenate([
        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs
    ])
    
    # Overlap constraints (squared distance - sum_radii^2 >= 0)
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    c = np.concatenate([c, dx[TRIL_IDX]**2 + dy[TRIL_IDX]**2 - dr[TRIL_IDX]**2])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-7, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)])
    return b

def solve_lp_radii(centers):
    """Solve LP to maximize sum of radii for fixed centers."""
    n = N
    c_obj = -np.ones(n)
    
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    # Boundary limits for each circle
    for i in range(n):
        x_val, y_val = centers[i]
        bound = min(x_val, 1.0 - x_val, y_val, 1.0 - y_val)
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(1e-9, bound)
        idx += 1
        
    # Pairwise distance limits
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(1e-9, dist)
            idx += 1
            
    bounds = [(0.0, 0.5)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def force_simulate(centers, steps=300):
    """Force-directed layout to spread points evenly and resolve overlaps."""
    n = centers.shape[0]
    for _ in range(steps):
        forces = np.zeros_like(centers)
        for i in range(n):
            for j in range(i + 1, n):
                diff = centers[j] - centers[i]
                dist = np.hypot(diff[0], diff[1])
                if dist < 0.25 and dist > 1e-6:
                    f = 0.005 / (dist**2 + 0.0001)
                    fx, fy = f * diff[0], f * diff[1]
                    forces[i] -= [fx, fy]
                    forces[j] += [fx, fy]
        # Push away from boundaries
        for i in range(n):
            if centers[i, 0] < 0.15: forces[i, 0] += 0.02
            elif centers[i, 0] > 0.85: forces[i, 0] -= 0.02
            if centers[i, 1] < 0.15: forces[i, 1] += 0.02
            elif centers[i, 1] > 0.85: forces[i, 1] -= 0.02
            
        centers += forces * 0.08
        centers = np.clip(centers, 0.05, 0.95)
    return centers

def generate_hex_init(r0, angle_deg):
    """Generate hexagonal lattice points, optionally rotated."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        if y > 1.0 + r0:
            break
            
    pts = np.array(pts[:N + 10])
    
    if angle_deg != 0:
        rad = np.deg2rad(angle_deg)
        c, s = np.cos(rad), np.sin(rad)
        center = np.array([0.5, 0.5])
        pts = (pts - center) @ np.array([[c, -s], [s, c]]) + center
        
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    
    while len(pts) < N:
        pts = np.vstack([pts, [np.random.uniform(0.1, 0.9), np.random.uniform(0.1, 0.9)]])
        
    return pts[:N]

def is_strictly_valid(centers, radii):
    """Check strict feasibility against validator tolerance."""
    n = centers.shape[0]
    for i in range(n):
        if radii[i] < 0: return False
        if centers[i,0] - radii[i] < -1e-12 or centers[i,0] + radii[i] > 1.0 + 1e-12: return False
        if centers[i,1] - radii[i] < -1e-12 or centers[i,1] + radii[i] > 1.0 + 1e-12: return False
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i,0] - centers[j,0], centers[i,1] - centers[j,1])
            if dist < radii[i] + radii[j] - 1e-12:
                return False
    return True

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Phase 1: Diverse Initializations
    inits = []
    for r0 in [0.08, 0.09, 0.10, 0.11]:
        for ang in [0, 10, 20, 30, 45]:
            pts = generate_hex_init(r0, ang)
            pts += np.random.uniform(-0.004, 0.004, pts.shape)
            inits.append(pts.copy())
            
    for _ in range(6):
        pts = np.random.uniform(0.15, 0.85, (N, 2))
        pts = force_simulate(pts, steps=400)
        inits.append(pts.copy())
        
    for c_init in inits:
        r_init = np.full(N, 0.085)
        # Safety clip for initial radii
        for i in range(N):
            r_init[i] = min(r_init[i], c_init[i,0], 1-c_init[i,0], c_init[i,1], 1-c_init[i,1])
            
        x0 = np.zeros(3 * N)
        x0[0::3] = c_init[:, 0]
        x0[1::3] = c_init[:, 1]
        x0[2::3] = r_init
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 15000, 'ftol': 1e-14, 'disp': False})
            if not np.isnan(res.fun):
                curr_sum = -res.fun
                if curr_sum > best_sum:
                    best_sum = curr_sum
                    best_centers = np.column_stack((res.x[0::3], res.x[1::3])).copy()
                    best_radii = res.x[2::3].copy()
        except Exception:
            continue

    # Phase 2: LP-SLSQP Alternating Refinement
    if best_centers is not None:
        for _ in range(8):
            # LP step: maximize radii for current centers
            lp_r = solve_lp_radii(best_centers)
            if lp_r is not None:
                best_radii = lp_r.copy()
                curr_sum = np.sum(best_radii)
                if curr_sum > best_sum:
                    best_sum = curr_sum
                    
            # SLSQP step: optimize both positions and radii starting from LP result
            x0 = np.zeros(3 * N)
            x0[0::3] = best_centers[:, 0]
            x0[1::3] = best_centers[:, 1]
            x0[2::3] = best_radii
            
            # Slight perturbation to escape flat gradients
            x0[0::3] += np.random.normal(0, 1e-5, N)
            x0[1::3] += np.random.normal(0, 1e-5, N)
            x0[0::3] = np.clip(x0[0::3], x0[2::3], 1.0 - x0[2::3])
            x0[1::3] = np.clip(x0[1::3], x0[2::3], 1.0 - x0[2::3])
            
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 20000, 'ftol': 1e-14, 'disp': False})
                if not np.isnan(res.fun):
                    curr_sum = -res.fun
                    if curr_sum > best_sum:
                        best_centers = np.column_stack((res.x[0::3], res.x[1::3])).copy()
                        best_radii = res.x[2::3].copy()
                        best_sum = curr_sum
            except Exception:
                pass

    # Phase 3: Deflation & Perturbation to escape local minima
    if best_centers is not None:
        for shrink in [0.985, 0.970, 0.955, 0.940]:
            pert_c = best_centers + np.random.normal(0, 0.002, best_centers.shape)
            pert_c = np.clip(pert_c, 0.01, 0.99)
            pert_r = best_radii * shrink
            
            x0 = np.zeros(3 * N)
            x0[0::3] = pert_c[:, 0]
            x0[1::3] = pert_c[:, 1]
            x0[2::3] = pert_r
            
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 15000, 'ftol': 1e-14, 'disp': False})
                if not np.isnan(res.fun):
                    curr_sum = -res.fun
                    if curr_sum > best_sum:
                        best_centers = np.column_stack((res.x[0::3], res.x[1::3])).copy()
                        best_radii = res.x[2::3].copy()
                        best_sum = curr_sum
            except Exception:
                continue
                
            # Run LP again after deflation
            lp_r = solve_lp_radii(best_centers)
            if lp_r is not None:
                best_radii = lp_r
                if np.sum(best_radii) > best_sum:
                    best_sum = np.sum(best_radii)

    # Fallback
    if best_centers is None:
        best_centers = generate_hex_init(0.09, 0)
        best_radii = np.full(N, 0.085)
        best_sum = np.sum(best_radii)
        
    # Final Validation & Minimal Repair
    for _ in range(100):
        if is_strictly_valid(best_centers, best_radii):
            break
        # Shrink radii minimally to satisfy validator tolerance
        best_radii *= 0.9999
        for i in range(N):
            best_centers[i, 0] = np.clip(best_centers[i, 0], best_radii[i], 1.0 - best_radii[i])
            best_centers[i, 1] = np.clip(best_centers[i, 1], best_radii[i], 1.0 - best_radii[i])
            
    best_sum = float(np.sum(best_radii))
    return best_centers, best_radii, best_sum
