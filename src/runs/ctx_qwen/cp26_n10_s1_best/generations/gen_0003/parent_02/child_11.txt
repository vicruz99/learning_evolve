

<strategy>
To maximize the sum of radii for 26 circles in a unit square, we need to escape the many local minima that plague gradient-based optimizers on this highly non-convex landscape. The strategy involves:

1. **Diverse High-Quality Initializations**: Generate starting points from multiple hexagonal lattices (varied densities and rotations), random distributions, and force-directed repulsion layouts. Each is equipped with strictly feasible initial radii (40% of the nearest neighbor/wall distance) to guarantee the optimizer starts inside the feasible region.
2. **Multi-Start SLSQP Optimization**: Run `scipy.optimize.minimize` with the SLSQP method from each initialization. SLSQP handles the inequality constraints (boundary and non-overlap) efficiently. We keep the best feasible solution found.
3. **Perturbation & Scaling Search**: Local minima often trap the optimizer near suboptimal radii. We systematically perturb the best configuration's positions and *scale up* the radii by a small factor, then re-optimize. This "push and relax" cycle encourages circles to expand into new, denser arrangements.
4. **Strict Validation & Repair**: After optimization, we rigorously check against the validator's tolerance (`1e-12`). If numerical drift causes micro-violations, we apply a minimal uniform shrinkage and re-clip centers to guarantee strict compliance without sacrificing meaningful score.
5. **Vectorized Constraint Evaluation**: All constraints are computed using NumPy broadcasting and `tril_indices` for maximum speed and numerical stability during the thousands of optimizer iterations.

import numpy as np
from scipy.optimize import minimize

N = 26

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraint_func(x):
    """Vectorized inequality constraints: g(x) >= 0."""
    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]
    
    # Boundary constraints
    c = np.concatenate([
        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs
    ])
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    i, j = np.tril_indices(N, -1)
    c = np.concatenate([c, dx[i, j]**2 + dy[i, j]**2 - dr[i, j]**2])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [0, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
    return b

def make_init_hex(r0, angle_deg=0):
    """Generate hexagonal lattice points, optionally rotated."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        if y > 1.0 + r0:
            break
    pts = np.array(pts[:N])
    if angle_deg != 0:
        rad = np.deg2rad(angle_deg)
        c_val, s_val = np.cos(rad), np.sin(rad)
        pts = (pts - 0.5) @ np.array([[c_val, -s_val], [s_val, c_val]]) + 0.5
    pts = np.clip(pts, 0.02, 0.98)
    return pts

def make_init_random(seed):
    """Generate random initial points."""
    np.random.seed(seed)
    return np.random.uniform(0.1, 0.9, (N, 2))

def make_init_force_directed(seed):
    """Generate force-directed repulsion layout."""
    np.random.seed(seed)
    pts = np.random.uniform(0.15, 0.85, (N, 2))
    for _ in range(600):
        forces = np.zeros_like(pts)
        # Repulsive forces between points
        dx = pts[:, None, 0] - pts[None, :, 0]
        dy = pts[:, None, 1] - pts[None, :, 1]
        d2 = dx**2 + dy**2 + 1e-8
        d = np.sqrt(d2)
        f = 0.015 / d2
        f[d < 0.01] = 0.0
        fx = f * dx
        fy = f * dy
        forces[:, 0] += np.sum(fx, axis=1)
        forces[:, 1] += np.sum(fy, axis=1)
        
        # Wall repulsion
        forces[:, 0] += np.where(pts[:, 0] < 0.15, 0.03, np.where(pts[:, 0] > 0.85, -0.03, 0.0))
        forces[:, 1] += np.where(pts[:, 1] < 0.15, 0.03, np.where(pts[:, 1] > 0.85, -0.03, 0.0))
        
        pts += forces * 0.06
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def get_safe_radii(centers):
    """Compute strictly feasible initial radii based on nearest neighbors/walls."""
    n = centers.shape[0]
    radii = np.zeros(n)
    for i in range(n):
        min_d = min(centers[i, 0], 1.0 - centers[i, 0],
                    centers[i, 1], 1.0 - centers[i, 1])
        for j in range(n):
            if i != j:
                d = np.hypot(centers[i, 0] - centers[j, 0],
                             centers[i, 1] - centers[j, 1])
                if d < min_d:
                    min_d = d
        # Leave 60% margin to guarantee initial feasibility
        radii[i] = min_d * 0.4
    return radii

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraint_func}
    
    best_sum = -1.0
    best_x = None
    
    # Phase 1: Diverse Multi-Start Optimization
    inits = []
    for r0 in [0.08, 0.085, 0.09, 0.095, 0.10, 0.105]:
        for ang in [0, 10, 20, 30, 45, 60]:
            inits.append(make_init_hex(r0, ang))
    for s in range(15):
        inits.append(make_init_random(s))
    for s in range(10):
        inits.append(make_init_force_directed(s))
        
    for pts in inits:
        rs = get_safe_radii(pts)
        x0 = np.zeros(3 * N)
        x0[0::3] = pts[:, 0]
        x0[1::3] = pts[:, 1]
        x0[2::3] = rs
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 15000, 'ftol': 1e-13, 'disp': False})
            if not np.isnan(res.fun) and -res.fun > best_sum:
                if np.min(constraint_func(res.x)) >= -1e-7:
                    best_sum = -res.fun
                    best_x = res.x.copy()
        except Exception:
            continue
            
    # Phase 2: Perturbation & Scaling Search to escape local minima
    if best_x is not None:
        for step in range(40):
            noise = 0.003 * (0.88 ** step)
            scale = 1.0 + 0.006 * (0.92 ** step)
            x0 = best_x + np.random.normal(0, noise, size=3 * N)
            x0[2::3] *= scale
            x0[0::3] = np.clip(x0[0::3], 0.0, 1.0)
            x0[1::3] = np.clip(x0[1::3], 0.0, 1.0)
            x0[2::3] = np.clip(x0[2::3], 0.0, 0.5)
            
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 8000, 'ftol': 1e-13, 'disp': False})
                if not np.isnan(res.fun) and -res.fun > best_sum:
                    if np.min(constraint_func(res.x)) >= -1e-7:
                        best_sum = -res.fun
                        best_x = res.x.copy()
            except Exception:
                pass
                
    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3]
    
    # Phase 3: Strict Validation & Minimal Repair
    valid = True
    for i in range(N):
        if radii[i] < 0: valid=False; break
        if centers[i,0] < radii[i] - 1e-10 or centers[i,0] > 1 - radii[i] + 1e-10: valid=False; break
        if centers[i,1] < radii[i] - 1e-10 or centers[i,1] > 1 - radii[i] + 1e-10: valid=False; break
    if valid:
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
                if d < radii[i]+radii[j] - 1e-10: valid=False; break
            if not valid: break
            
    if not valid:
        for _ in range(100):
            radii *= 0.999
            centers[:,0] = np.clip(centers[:,0], radii, 1-radii)
            centers[:,1] = np.clip(centers[:,1], radii, 1-radii)
            ok = True
            for i in range(N):
                for j in range(i+1, N):
                    if np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1]) < radii[i]+radii[j]-1e-12:
                        ok=False; break
                if not ok: break
            if ok: break
            
    return centers, radii, float(np.sum(radii))
