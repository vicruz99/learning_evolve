To improve the packing sum of radii beyond ~2.634, we leverage two key insights:
1. **Alternating Optimization**: The problem decomposes naturally. Given fixed centers, maximizing the sum of radii subject to non-overlap and boundary constraints is a Linear Program (LP). Given fixed radii, optimizing centers to satisfy constraints and allow larger radii is a nonlinear problem solvable via SLSQP. Alternating between these two steps often escapes local minima that trap simultaneous optimization.
2. **Deflation & Perturbation**: Temporarily shrinking radii slightly allows circles to rearrange into denser configurations (e.g., hexagonal vs grid). After rearrangement, radii are allowed to grow again, often reaching a higher sum.
3. **Diverse Initialization**: We generate initial configurations using hexagonal lattices (with rotations), force-directed repulsion simulations, and perturbed grids. Each is processed through the alternating LP-SLSQP pipeline.
4. **Robust Validation**: A final repair loop ensures strict compliance with the validator's tolerance (`1e-12`), shrinking radii minimally only if numerical drift causes micro-overlaps.

This approach systematically explores the configuration space, exploits the linear structure of the radius constraints, and fine-tunes positions for maximal density.

import numpy as np
from scipy.optimize import minimize, linprog

N_CIRCLES = 26

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Vectorized inequality constraints: g(x) >= 0."""
    n = N_CIRCLES
    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]
    
    # Boundary constraints
    c = np.concatenate([
        xs - rs, 1.0 - xs - rs,
        ys - rs, 1.0 - ys - rs
    ])
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    i_idx, j_idx = np.tril_indices(n, -1)
    c = np.concatenate([c, dx[i_idx, j_idx]**2 + dy[i_idx, j_idx]**2 - dr[i_idx, j_idx]**2])
    return c

def get_lp_radii(centers):
    """Solve LP to find maximal sum of radii for fixed centers."""
    n = N_CIRCLES
    c_obj = -np.ones(n)
    num_cons = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_cons, n))
    b_ub = np.zeros(num_cons)
    
    idx = 0
    for i in range(n):
        # x >= r
        A_ub[idx, i] = 1.0; b_ub[idx] = centers[i, 0]; idx += 1
        # 1-x >= r
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - centers[i, 0]; idx += 1
        # y >= r
        A_ub[idx, i] = 1.0; b_ub[idx] = centers[i, 1]; idx += 1
        # 1-y >= r
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - centers[i, 1]; idx += 1
        
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0; A_ub[idx, j] = 1.0; b_ub[idx] = max(d, 1e-10); idx += 1
            
    bounds = [(0.0, 0.5)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return np.full(n, 0.01)

def force_directed_init(n, seed):
    """Generate initial configuration using repulsive force simulation."""
    np.random.seed(seed)
    centers = np.random.uniform(0.15, 0.85, (n, 2))
    for _ in range(600):
        forces = np.zeros_like(centers)
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[j, 0] - centers[i, 0]
                dy = centers[j, 1] - centers[i, 1]
                d = np.hypot(dx, dy)
                if d < 0.28 and d > 1e-6:
                    f = 0.012 / (d**2)
                    fx, fy = f * dx, f * dy
                    forces[i] -= [fx, fy]
                    forces[j] += [fx, fy]
        for i in range(n):
            if centers[i, 0] < 0.12: forces[i, 0] += 0.025
            elif centers[i, 0] > 0.88: forces[i, 0] -= 0.025
            if centers[i, 1] < 0.12: forces[i, 1] += 0.025
            elif centers[i, 1] > 0.88: forces[i, 1] -= 0.025
        centers += forces * 0.06
        centers = np.clip(centers, 0.05, 0.95)
    return centers

def run_packing():
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    n = N_CIRCLES
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * n
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -1.0
    best_x = None
    
    inits = []
    
    # 1. Rotated Hexagonal Lattices
    for seed in range(10):
        np.random.seed(seed)
        r0 = 0.095
        pts = []
        y = r0
        row = 0
        while len(pts) < n + 5:
            x = r0 + (row % 2) * r0
            while x <= 1.0 - r0:
                pts.append([x + np.random.uniform(-0.004, 0.004),
                            y + np.random.uniform(-0.004, 0.004)])
                x += 2.0 * r0
            y += np.sqrt(3.0) * r0
            row += 1
            if y > 1.0 + r0: break
        pts = np.array(pts[:n])
        # Rotate to break symmetry with square boundaries
        ang = np.random.uniform(-0.2, 0.2)
        c, s = np.cos(ang), np.sin(ang)
        center = np.array([0.5, 0.5])
        pts = (pts - center) @ np.array([[c, -s], [s, c]]) + center
        pts = np.clip(pts, 0.05, 0.95)
        inits.append(pts)
        
    # 2. Force-Directed Layouts
    for seed in range(6):
        inits.append(force_directed_init(n, seed))
        
    # 3. Perturbed Grids
    for seed in range(4):
        np.random.seed(seed)
        pts = []
        for r in range(6):
            for c_idx in range(5):
                if len(pts) >= n: break
                pts.append([0.08 + c_idx * 0.18 + np.random.uniform(-0.01, 0.01),
                            0.08 + r * 0.16 + np.random.uniform(-0.01, 0.01)])
        inits.append(np.array(pts[:n]))

    # Optimization Pipeline
    for pts in inits:
        rs = get_lp_radii(pts)
        x0 = np.zeros(3 * n)
        x0[0::3] = pts[:, 0]
        x0[1::3] = pts[:, 1]
        x0[2::3] = rs
        
        try:
            # Phase 1: Initial SLSQP
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 25000, 'ftol': 1e-14, 'disp': False})
            curr_sum = -res.fun
            
            # Phase 2: Alternating LP + SLSQP Refinement
            for _ in range(8):
                c_tmp = np.column_stack((res.x[0::3], res.x[1::3]))
                # Small perturbation to escape flat spots
                c_tmp += np.random.normal(0, 1e-5, c_tmp.shape)
                c_tmp = np.clip(c_tmp, 0.01, 0.99)
                
                r_tmp = get_lp_radii(c_tmp)
                x_new = np.zeros(3 * n)
                x_new[0::3] = c_tmp[:, 0]
                x_new[1::3] = c_tmp[:, 1]
                x_new[2::3] = r_tmp
                
                # Mild deflation to allow rearrangement
                x_new[2::3] *= 0.997
                
                res = minimize(objective, x_new, method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 20000, 'ftol': 1e-14, 'disp': False})
                if -res.fun > curr_sum:
                    curr_sum = -res.fun
            
            if curr_sum > best_sum:
                best_sum = curr_sum
                best_x = res.x.copy()
        except Exception:
            continue

    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3]
    
    # Final validation and minimal repair to guarantee strict compliance
    for _ in range(100):
        valid = True
        for i in range(n):
            if radii[i] < 0: valid = False; break
            if centers[i,0] < radii[i] - 1e-12 or centers[i,0] > 1.0 - radii[i] + 1e-12: valid = False; break
            if centers[i,1] < radii[i] - 1e-12 or centers[i,1] > 1.0 - radii[i] + 1e-12: valid = False; break
        if valid:
            for i in range(n):
                for j in range(i + 1, n):
                    d = np.hypot(centers[i,0] - centers[j,0], centers[i,1] - centers[j,1])
                    if d < radii[i] + radii[j] - 1e-12:
                        valid = False; break
                if not valid: break
        if valid: break
        
        # Shrink minimally to recover validity
        radii *= 0.9995
        for i in range(n):
            centers[i,0] = np.clip(centers[i,0], radii[i], 1.0 - radii[i])
            centers[i,1] = np.clip(centers[i,1], radii[i], 1.0 - radii[i])
            
    return centers, radii, float(np.sum(radii))
