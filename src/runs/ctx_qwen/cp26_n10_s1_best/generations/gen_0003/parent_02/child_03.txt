import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraint_func(x):
    """Vectorized inequality constraints: g(x) >= 0."""
    n = N_CIRCLES
    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]
    
    # Boundary constraints
    c = np.empty(4 * n + n * (n - 1) // 2)
    c[:n] = xs - rs
    c[n:2*n] = 1.0 - xs - rs
    c[2*n:3*n] = ys - rs
    c[3*n:4*n] = 1.0 - ys - rs
    
    # Overlap constraints (squared distance formulation for stability)
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    i, j = np.tril_indices(n, -1)
    c[4*n:] = dx[i, j]**2 + dy[i, j]**2 - dr[i, j]**2
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-9, 0.5]."""
    b = []
    for _ in range(N_CIRCLES):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-9, 0.5)])
    return b

def get_safe_radii(centers):
    """Compute strictly feasible initial radii based on nearest neighbors/walls."""
    n = centers.shape[0]
    rs = np.zeros(n)
    for i in range(n):
        min_d = min(centers[i, 0], 1.0 - centers[i, 0],
                    centers[i, 1], 1.0 - centers[i, 1])
        for j in range(n):
            if i != j:
                d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                if d < min_d:
                    min_d = d
        # Leave margin to guarantee initial feasibility
        rs[i] = min_d * 0.45
    return rs

def generate_inits(n):
    """Generate diverse initial center configurations."""
    inits = []
    
    # 1. Rotated Hexagonal Lattices
    for r0 in [0.085, 0.09, 0.095]:
        for ang in np.linspace(0, np.pi/4, 6):
            pts = []
            y = r0
            row = 0
            while len(pts) < n + 5:
                x = r0 if row % 2 == 0 else 2.0 * r0
                while x <= 1.0 - r0:
                    pts.append([x, y])
                    x += 2.0 * r0
                y += np.sqrt(3.0) * r0
                row += 1
                if y > 1.0: break
            pts = np.array(pts[:n+5])
            
            if ang != 0.0:
                c, s = np.cos(ang), np.sin(ang)
                center = np.array([0.5, 0.5])
                pts = (pts - center) @ np.array([[c, -s], [s, c]]) + center
                
            mask = (pts[:, 0] > 0.02) & (pts[:, 0] < 0.98) & (pts[:, 1] > 0.02) & (pts[:, 1] < 0.98)
            pts = pts[mask]
            while len(pts) < n:
                pts = np.vstack([pts, [np.random.uniform(0.1, 0.9), np.random.uniform(0.1, 0.9)]])
            pts = pts[:n] + np.random.uniform(-0.005, 0.005, (n, 2))
            inits.append(pts)
            
    # 2. Corner-Biased Layouts
    for _ in range(10):
        pts = np.zeros((n, 2))
        pts[0] = [0.12, 0.12]
        pts[1] = [0.88, 0.12]
        pts[2] = [0.12, 0.88]
        pts[3] = [0.88, 0.88]
        y = 0.12
        row = 0
        idx = 4
        while idx < n:
            x = 0.12 if row % 2 == 0 else 0.24
            while x <= 0.88 and idx < n:
                pts[idx] = [x + np.random.uniform(-0.01, 0.01), y + np.random.uniform(-0.01, 0.01)]
                idx += 1
                x += 0.24
            y += 0.21
            row += 1
        inits.append(pts)
        
    # 3. Random Spread
    for _ in range(15):
        pts = np.random.uniform(0.1, 0.9, (n, 2))
        inits.append(pts)
        
    return inits

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    n = N_CIRCLES
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraint_func}
    
    best_sum = -1.0
    best_x = None
    
    inits = generate_inits(n)
    
    # Phase 1: Multi-start SLSQP optimization
    for pts in inits:
        x0 = np.zeros(3 * n)
        x0[0::3] = pts[:, 0]
        x0[1::3] = pts[:, 1]
        x0[2::3] = get_safe_radii(pts)
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 15000, 'ftol': 1e-14, 'disp': False})
            if not np.isnan(res.fun):
                s = -res.fun
                if s > best_sum:
                    best_sum = s
                    best_x = res.x.copy()
        except Exception:
            continue
            
    # Phase 2: Deflation-Reinflation to escape jammed local minima
    if best_x is not None:
        best_pts = np.column_stack((best_x[0::3], best_x[1::3]))
        best_rs = best_x[2::3]
        
        for factor in [0.98, 0.96, 0.94, 0.92]:
            for _ in range(6):
                noise_scale = 0.002 / (1.0 - factor + 0.01)
                pert_pts = best_pts + np.random.normal(0, noise_scale, best_pts.shape)
                pert_pts = np.clip(pert_pts, 0.02, 0.98)
                
                x0 = np.zeros(3 * n)
                x0[0::3] = pert_pts[:, 0]
                x0[1::3] = pert_pts[:, 1]
                # Deflated radii guarantee feasibility after perturbation
                x0[2::3] = get_safe_radii(pert_pts) * factor
                
                try:
                    res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                                   constraints=cons, options={'maxiter': 12000, 'ftol': 1e-14, 'disp': False})
                    if not np.isnan(res.fun):
                        s = -res.fun
                        if s > best_sum:
                            best_sum = s
                            best_x = res.x.copy()
                            best_pts = np.column_stack((best_x[0::3], best_x[1::3]))
                            best_rs = best_x[2::3]
                except Exception:
                    pass
                    
    # Phase 3: High-precision local refinement
    if best_x is not None:
        for _ in range(5):
            x0 = best_x + np.random.normal(0, 1e-6, 3*n)
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 20000, 'ftol': 1e-15, 'disp': False})
                if not np.isnan(res.fun) and -res.fun > best_sum:
                    best_sum = -res.fun
                    best_x = res.x.copy()
            except Exception:
                pass
                
    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3]
    
    # Final safety clipping to guarantee strict validator compliance
    for i in range(n):
        r = radii[i]
        centers[i, 0] = np.clip(centers[i, 0], r + 1e-10, 1.0 - r - 1e-10)
        centers[i, 1] = np.clip(centers[i, 1], r + 1e-10, 1.0 - r - 1e-10)
        
    return centers, radii, float(np.sum(radii))
