import numpy as np
from scipy.optimize import minimize, linprog

N_CIRCLES = 26

def objective(vars):
    """Minimize negative sum of radii."""
    return -np.sum(vars[2::3])

def constraint_func(vars):
    """Vectorized inequality constraints: g(vars) >= 0."""
    n = N_CIRCLES
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    # Boundary constraints: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    b = np.concatenate([
        x - r,
        1.0 - x - r,
        y - r,
        1.0 - y - r
    ])
    
    # Overlap constraints: dist^2 - (r1+r2)^2 >= 0
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dr = r[:, None] + r[None, :]
    
    i_idx, j_idx = np.tril_indices(n, -1)
    o = dx[i_idx, j_idx]**2 + dy[i_idx, j_idx]**2 - dr[i_idx, j_idx]**2
    
    return np.concatenate([b, o])

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [0, 0.5]."""
    b = []
    for _ in range(N_CIRCLES):
        b.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
    return b

def solve_lp_radii(centers):
    """Solve LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    n_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((n_ineq, n))
    b_ub = np.zeros(n_ineq)
    
    row = 0
    # Boundary constraints: r_i <= dist_to_wall
    for i in range(n):
        x, y = centers[i]
        d_wall = min(x, 1.0 - x, y, 1.0 - y)
        A_ub[row, i] = 1.0
        b_ub[row] = max(0.0, d_wall)
        row += 1
        
    # Overlap constraints: r_i + r_j <= dist_ij
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[row, i] = 1.0
            A_ub[row, j] = 1.0
            b_ub[row] = d
            row += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return np.full(n, 0.01)

def force_directed_layout(n, seed, steps=800):
    """Generate initial configuration using repulsive forces."""
    np.random.seed(seed)
    pts = np.random.uniform(0.15, 0.85, (n, 2))
    for _ in range(steps):
        forces = np.zeros_like(pts)
        for i in range(n):
            for j in range(i + 1, n):
                dx = pts[j, 0] - pts[i, 0]
                dy = pts[j, 1] - pts[i, 1]
                d = np.hypot(dx, dy)
                if d < 0.3 and d > 1e-6:
                    f = 0.015 / (d**2 + 0.001)
                    fx, fy = f * dx, f * dy
                    forces[i] -= [fx, fy]
                    forces[j] += [fx, fy]
            # Wall repulsion
            if pts[i, 0] < 0.12: forces[i, 0] += 0.03
            elif pts[i, 0] > 0.88: forces[i, 0] -= 0.03
            if pts[i, 1] < 0.12: forces[i, 1] += 0.03
            elif pts[i, 1] > 0.88: forces[i, 1] -= 0.03
        pts += forces * 0.05
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def hex_grid_layout(n, seed, r0=0.09):
    """Generate hexagonal grid initial configuration."""
    np.random.seed(seed)
    pts = []
    y = r0
    row = 0
    while len(pts) < n + 5:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
    pts = np.array(pts[:n])
    pts += np.random.uniform(-0.005, 0.005, pts.shape)
    return np.clip(pts, r0 + 0.01, 1.0 - r0 - 0.01)

def validate_and_shrink(centers, radii, tol=1e-11):
    """Ensure strict validity, shrink radii minimally if necessary."""
    n = centers.shape[0]
    for _ in range(100):
        valid = True
        for i in range(n):
            if radii[i] < 0:
                valid = False
                break
            if centers[i, 0] < radii[i] - tol or centers[i, 0] > 1.0 - radii[i] + tol:
                valid = False
                break
            if centers[i, 1] < radii[i] - tol or centers[i, 1] > 1.0 - radii[i] + tol:
                valid = False
                break
        if valid:
            for i in range(n):
                for j in range(i + 1, n):
                    d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                    if d < radii[i] + radii[j] - tol:
                        valid = False
                        break
                if not valid:
                    break
        if valid:
            return centers, radii
        radii *= 0.995
        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)
    return centers, radii

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    n = N_CIRCLES
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraint_func}
    
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    inits = []
    # 1. Force-directed layouts
    for s in range(15):
        inits.append(force_directed_layout(n, s, steps=800))
    # 2. Hexagonal grids with different densities
    for s in range(10):
        inits.append(hex_grid_layout(n, s, 0.09))
        inits.append(hex_grid_layout(n, s, 0.10))
    # 3. Perturbed grids
    for s in range(6):
        np.random.seed(s)
        pts = np.zeros((n, 2))
        idx = 0
        for r in range(6):
            for c in range(5):
                if idx < n:
                    pts[idx] = [0.08 + c * 0.18, 0.08 + r * 0.16]
                    idx += 1
        pts += np.random.uniform(-0.01, 0.01, pts.shape)
        inits.append(pts)
        
    # Phase 1: Multi-start SLSQP with LP-initialized radii
    for pts in inits:
        r_init = solve_lp_radii(pts)
        r_init = np.maximum(r_init, 0.005) * 0.95
        
        x0 = np.zeros(3 * n)
        x0[0::3] = pts[:, 0]
        x0[1::3] = pts[:, 1]
        x0[2::3] = r_init
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 15000, 'ftol': 1e-14, 'disp': False})
            if not np.isnan(res.fun):
                curr_sum = -res.fun
                c_opt = np.column_stack((res.x[0::3], res.x[1::3]))
                r_opt = res.x[2::3]
                
                if curr_sum > best_sum:
                    c_opt, r_opt = validate_and_shrink(c_opt, r_opt)
                    if np.sum(r_opt) > best_sum:
                        best_sum = np.sum(r_opt)
                        best_centers = c_opt
                        best_radii = r_opt
        except Exception:
            continue
            
    # Fallback initialization if all optimizations fail (rare)
    if best_centers is None:
        best_centers = force_directed_layout(n, 0, steps=1000)
        best_radii = solve_lp_radii(best_centers) * 0.95
        best_sum = np.sum(best_radii)
        
    # Phase 2: Alternating LP/SLSQP refinement
    if best_centers is not None:
        for _ in range(25):
            # LP step: expand radii to their theoretical maximum for current centers
            new_r = solve_lp_radii(best_centers)
            if np.sum(new_r) > best_sum * 0.999:
                best_radii = new_r * 0.999
                best_sum = np.sum(best_radii)
                
            # Perturb centers slightly to find new configurations
            pert_c = best_centers + np.random.normal(0, 0.002, best_centers.shape)
            pert_c = np.clip(pert_c, 0.01, 0.99)
            pert_r = best_radii * 0.995
            
            x0 = np.zeros(3 * n)
            x0[0::3] = pert_c[:, 0]
            x0[1::3] = pert_c[:, 1]
            x0[2::3] = pert_r
            
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 8000, 'ftol': 1e-14, 'disp': False})
                if not np.isnan(res.fun):
                    curr_sum = -res.fun
                    c_opt = np.column_stack((res.x[0::3], res.x[1::3]))
                    r_opt = res.x[2::3]
                    
                    c_opt, r_opt = validate_and_shrink(c_opt, r_opt)
                    curr_sum = np.sum(r_opt)
                    if curr_sum > best_sum:
                        best_sum = curr_sum
                        best_centers = c_opt
                        best_radii = r_opt
            except Exception:
                pass
                
    # Final validation and repair to guarantee strict compliance
    best_centers, best_radii = validate_and_shrink(best_centers, best_radii)
    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum
