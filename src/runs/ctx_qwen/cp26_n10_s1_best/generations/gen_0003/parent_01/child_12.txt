import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Returns all inequality constraints >= 0 (vectorized)."""
    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]
    
    # Boundary constraints: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    c = np.concatenate([xs - rs, 1.0 - xs - rs, ys - rs, 1.0 - ys - rs])
    
    # Overlap constraints: dist - (r_i + r_j) >= 0
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dist = np.sqrt(dx**2 + dy**2 + 1e-15)
    r_sum = rs[:, None] + rs[None, :]
    i_idx, j_idx = np.triu_indices(N, 1)
    c = np.concatenate([c, dist[i_idx, j_idx] - r_sum[i_idx, j_idx]])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-6, 0.5]."""
    return [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N

def safe_radii(centers):
    """Compute strictly feasible initial radii based on nearest neighbors/walls."""
    n = centers.shape[0]
    radii = np.zeros(n)
    for i in range(n):
        min_d = min(centers[i, 0], 1.0 - centers[i, 0],
                    centers[i, 1], 1.0 - centers[i, 1])
        for j in range(n):
            if i != j:
                d = np.hypot(centers[i, 0] - centers[j, 0],
                             centers[i, 1] - centers[j, 1])
                if d < min_d:
                    min_d = d
        radii[i] = max(1e-4, min_d * 0.42)
    return radii

def force_relax(centers, steps=200):
    """Force-directed relaxation to resolve overlaps and push towards boundaries."""
    n = centers.shape[0]
    rs = safe_radii(centers)
    for _ in range(steps):
        fx, fy = np.zeros(n), np.zeros(n)
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[j, 0] - centers[i, 0]
                dy = centers[j, 1] - centers[i, 1]
                d = np.hypot(dx, dy)
                if d < rs[i] + rs[j] and d > 1e-6:
                    f = (rs[i] + rs[j] - d) * 0.5 / d
                    fx[i] -= f * dx
                    fy[i] -= f * dy
                    fx[j] += f * dx
                    fy[j] += f * dy
            # Wall repulsion to utilize boundaries
            if centers[i, 0] < rs[i]: fx[i] += 0.1
            if centers[i, 0] > 1.0 - rs[i]: fx[i] -= 0.1
            if centers[i, 1] < rs[i]: fy[i] += 0.1
            if centers[i, 1] > 1.0 - rs[i]: fy[i] -= 0.1
        centers += np.column_stack([fx, fy]) * 0.05
        centers = np.clip(centers, 1e-4, 1.0 - 1e-4)
    return centers, safe_radii(centers)

def make_hex_init(seed, r0=0.09, angle=0.0):
    """Generate a hexagonal lattice initialization with optional rotation."""
    np.random.seed(seed)
    centers = []
    y = r0
    row = 0
    while len(centers) < N + 5:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x + r0 <= 1.0:
            centers.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
        row += 1
        if y + r0 > 1.0: break
    centers = np.array(centers[:N + 5])
    if angle != 0:
        c = np.array([0.5, 0.5])
        ca, sa = np.cos(angle), np.sin(angle)
        centers = ((centers - c) @ np.array([[ca, -sa], [sa, ca]]) + c)
    centers = centers[:N]
    centers += np.random.uniform(-0.01, 0.01, centers.shape)
    centers = np.clip(centers, 0.05, 0.95)
    return centers, np.full(N, r0)

def run_packing():
    np.random.seed(None)
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -1.0
    best_x = None
    
    inits = []
    # Phase 1: Diverse hexagonal starts with rotations and force relaxation
    for s in range(20):
        for r0 in [0.085, 0.09, 0.10, 0.11]:
            angle = np.random.uniform(-0.35, 0.35)
            c, r = make_hex_init(s, r0, angle)
            c, r = force_relax(c, steps=150)
            x0 = np.zeros(3 * N)
            x0[0::3] = c[:, 0]
            x0[1::3] = c[:, 1]
            x0[2::3] = r
            inits.append(x0)
            
    # Random + force starts to capture non-lattice structures
    for s in range(10):
        c = np.random.uniform(0.15, 0.85, (N, 2))
        c, r = force_relax(c, steps=200)
        x0 = np.zeros(3 * N)
        x0[0::3] = c[:, 0]
        x0[1::3] = c[:, 1]
        x0[2::3] = r
        inits.append(x0)
        
    # SLSQP Optimization
    for x0 in inits:
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 10000, 'ftol': 1e-13, 'disp': False})
            if not np.isnan(res.fun):
                val = constraints(res.x)
                if np.min(val) >= -1e-8:
                    s = -res.fun
                    if s > best_sum:
                        best_sum = s
                        best_x = res.x.copy()
        except Exception:
            pass
            
    # Phase 2: Alternating LP (radii) / SLSQP (centers) Refinement
    if best_x is not None:
        for iteration in range(12):
            centers = np.column_stack((best_x[0::3], best_x[1::3]))
            
            # LP: Maximize sum of radii for fixed centers
            n = N
            c_lp = -np.ones(n)
            num_ineq = n + n * (n - 1) // 2
            A_ub = np.zeros((num_ineq, n))
            b_ub = np.zeros(num_ineq)
            idx = 0
            for i in range(n):
                b = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
                A_ub[idx, i] = 1.0
                b_ub[idx] = b
                idx += 1
            for i in range(n):
                for j in range(i + 1, n):
                    d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                    A_ub[idx, i] = 1.0
                    A_ub[idx, j] = 1.0
                    b_ub[idx] = d
                    idx += 1
            
            try:
                res_lp = linprog(c_lp, A_ub=A_ub, b_ub=b_ub, bounds=[(0.0, None)] * n, method='highs')
                if res_lp.success:
                    best_x[2::3] = res_lp.x
                    new_sum = np.sum(res_lp.x)
                    if new_sum > best_sum:
                        best_sum = new_sum
                        # SLSQP on centers with updated radii
                        x0_ref = best_x + np.random.normal(0, 1e-5, 3 * N)
                        res_s = minimize(objective, x0_ref, method='SLSQP', bounds=bounds,
                                         constraints=cons, options={'maxiter': 6000, 'ftol': 1e-13})
                        if not np.isnan(res_s.fun):
                            val = constraints(res_s.x)
                            if np.min(val) >= -1e-8 and -res_s.fun > best_sum:
                                best_x = res_s.x.copy()
                                best_sum = -res_s.fun
            except Exception:
                pass
                
            # Perturbation to escape local minima periodically
            if iteration % 4 == 0:
                best_x += np.random.normal(0, 0.002, 3 * N)
                best_x[0::3] = np.clip(best_x[0::3], 0.01, 0.99)
                best_x[1::3] = np.clip(best_x[1::3], 0.01, 0.99)
                best_x[2::3] = np.clip(best_x[2::3], 0.005, 0.4)
                
    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3]
    
    # Final strict validation & minimal repair
    for _ in range(50):
        valid = True
        for i in range(N):
            if radii[i] < 0: valid = False; break
            if centers[i, 0] - radii[i] < -1e-11 or centers[i, 0] + radii[i] > 1 + 1e-11: valid = False; break
            if centers[i, 1] - radii[i] < -1e-11 or centers[i, 1] + radii[i] > 1 + 1e-11: valid = False; break
        if valid:
            for i in range(N):
                for j in range(i + 1, N):
                    if np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1]) < radii[i] + radii[j] - 1e-11:
                        valid = False; break
                if not valid: break
        if valid: break
        radii *= 0.998
        for i in range(N):
            centers[i, 0] = np.clip(centers[i, 0], radii[i], 1.0 - radii[i])
            centers[i, 1] = np.clip(centers[i, 1], radii[i], 1.0 - radii[i])
            
    return centers, radii, float(np.sum(radii))
