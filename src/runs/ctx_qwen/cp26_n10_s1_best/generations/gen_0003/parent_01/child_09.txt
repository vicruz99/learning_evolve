from scipy.optimize import minimize, linprog

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Returns all inequality constraints >= 0 (vectorized)."""
    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]
    
    c = np.empty(4*N + N*(N-1)//2)
    c[:N] = xs - rs
    c[N:2*N] = 1.0 - xs - rs
    c[2*N:3*N] = ys - rs
    c[3*N:4*N] = 1.0 - ys - rs
    
    i, j = TRIL_IDX
    dx = xs[i] - xs[j]
    dy = ys[i] - ys[j]
    dr = rs[i] + rs[j]
    c[4*N:] = dx**2 + dy**2 - dr**2
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [0, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
    return b

def force_relax(centers, steps=400):
    """Force-directed relaxation to spread points evenly."""
    n = centers.shape[0]
    for _ in range(steps):
        forces = np.zeros_like(centers)
        for i in range(n):
            for j in range(i + 1, n):
                diff = centers[j] - centers[i]
                dist = np.linalg.norm(diff)
                if dist < 0.25 and dist > 1e-6:
                    f = 0.008 / (dist**2 + 0.001)
                    forces[i] -= f * diff
                    forces[j] += f * diff
            for d in range(2):
                if centers[i, d] < 0.05: forces[i, d] += 0.03
                elif centers[i, d] > 0.95: forces[i, d] -= 0.03
        centers += forces * 0.12
        centers = np.clip(centers, 0.02, 0.98)
    return centers

def get_lp_radii(centers):
    """Optimally scale radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = 4*n + n*(n-1)//2
    A = np.zeros((num_ineq, n))
    b = np.zeros(num_ineq)
    idx = 0
    for i in range(n):
        x, y = centers[i]
        A[idx, i] = 1.0; b[idx] = x; idx += 1
        A[idx, i] = 1.0; b[idx] = 1.0 - x; idx += 1
        A[idx, i] = 1.0; b[idx] = y; idx += 1
        A[idx, i] = 1.0; b[idx] = 1.0 - y; idx += 1
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A[idx, i] = 1.0; A[idx, j] = 1.0; b[idx] = d; idx += 1
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return res.x
    except Exception:
        pass
    return None

def run_optimization(x0):
    """Run SLSQP optimizer from a given starting point."""
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    try:
        res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 10000, 'ftol': 1e-14, 'disp': False})
        if not np.isnan(res.fun):
            return res.x, -res.fun
    except Exception:
        pass
    return None, -np.inf

def run_packing():

    best_x = None

                x += 2 * r

            idx += 1

        x0[0::3] = centers[:, 0]
        x0[1::3] = centers[:, 1]
        x0[2::3] = 0.08

    for seed in range(10):

        if x_opt is not None and val > best_sum:
            best_sum = val
            best_x = x_opt.copy()

    if best_x is not None:

            break

    return centers, radii, float(np.sum(radii))
