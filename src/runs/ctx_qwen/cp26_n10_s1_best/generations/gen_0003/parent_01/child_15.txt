import numpy as np
from scipy.optimize import minimize, linprog
import math

N = 26

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [0, 0.5]."""
    bounds = []
    for _ in range(N):
        bounds.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
    return bounds

def constraints_func(x):
    """Returns all inequality constraints >= 0 (vectorized)."""
    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]
    
    # Boundary constraints
    c = np.concatenate([
        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs
    ])
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    i_idx, j_idx = np.tril_indices(N, -1)
    dx = xs[i_idx] - xs[j_idx]
    dy = ys[i_idx] - ys[j_idx]
    dr = rs[i_idx] + rs[j_idx]
    
    c = np.concatenate([c, dx**2 + dy**2 - dr**2])
    return c

def objective_func(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def solve_radii_lp(centers):
    """Optimally scale radii for fixed centers using Linear Programming."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    num_cons = n * 4 + n * (n - 1) // 2
    A = np.zeros((num_cons, n))
    b = np.zeros(num_cons)
    
    k = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        x_val, y_val = centers[i]
        A[k, i] = 1.0; b[k] = x_val; k += 1
        A[k, i] = 1.0; b[k] = 1.0 - x_val; k += 1
        A[k, i] = 1.0; b[k] = y_val; k += 1
        A[k, i] = 1.0; b[k] = 1.0 - y_val; k += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A[k, i] = 1.0
            A[k, j] = 1.0
            b[k] = d
            k += 1
            
    bounds_lp = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds_lp, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return np.full(n, 0.05)

def force_directed_centers(seed, steps=800):
    """Generate well-spread initial center positions using repulsive forces."""
    rng = np.random.default_rng(seed)
    centers = rng.uniform(0.15, 0.85, (N, 2))
    
    for _ in range(steps):
        forces = np.zeros((N, 2))
        # Pairwise repulsion (vectorized over pairs for efficiency)
        for i in range(N):
            for j in range(i + 1, N):
                dx = centers[j, 0] - centers[i, 0]
                dy = centers[j, 1] - centers[i, 1]
                d2 = dx*dx + dy*dy
                if d2 < 0.04 and d2 > 1e-8:
                    f = 0.005 / (d2 + 0.001)
                    fx, fy = f * dx, f * dy
                    forces[i, 0] -= fx
                    forces[i, 1] -= fy
                    forces[j, 0] += fx
                    forces[j, 1] += fy
                    
        # Boundary repulsion
        for i in range(N):
            if centers[i, 0] < 0.12: forces[i, 0] += 0.015
            elif centers[i, 0] > 0.88: forces[i, 0] -= 0.015
            if centers[i, 1] < 0.12: forces[i, 1] += 0.015
            elif centers[i, 1] > 0.88: forces[i, 1] -= 0.015
            
        centers += forces * 0.08
        centers = np.clip(centers, 0.05, 0.95)
        
    return centers

def run_packing():
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints_func}
    
    best_x = None
    best_sum = -1.0

    starts = []

            xs, ys, rs = [], [], []

                    xs.append(x)
                    ys.append(y)
                    rs.append(r0)
                    x += 2*r0

                row += 1
            while len(xs) < N:
                xs.append(0.5)
                ys.append(0.5)
                rs.append(r0)
                
            xs, ys, rs = np.array(xs), np.array(ys), np.array(rs)

                dx, dy = xs - cx, ys - cy

                ys = dx*s_ang + dy*c_ang + cy
                
            x0 = np.zeros(3*N)
            x0[0::3] = xs
            x0[1::3] = ys
            x0[2::3] = rs
            starts.append(x0)
            
    # Optimize each start with SLSQP
    for x0 in starts:

            if not np.isnan(res.fun):

                    best_sum = curr_sum

            pass

            x0[2::3] = new_r

            try:

            r = x0[3*i+2]

                best_sum = -res.fun
        except Exception:
            pass

    # Extract and validate
    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3]
    
    # Strict validation and minimal repair to guarantee compliance with 1e-12 tolerance
    for _ in range(100):

            if centers[i,1] - radii[i] < -1e-11 or centers[i,1] + radii[i] > 1 + 1e-11: valid=False; break

                if not valid: break
        if valid: break
        # Shrink radii minimally to resolve overlaps/boundary violations
        radii *= 0.9995
        
    return centers, radii, float(np.sum(radii))
