import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Returns all inequality constraints >= 0 (vectorized)."""
    n = N
    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]
    
    # Boundary constraints
    c = np.concatenate([
        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs
    ])
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    i_idx, j_idx = np.tril_indices(n, -1)
    c = np.concatenate([c, dx[i_idx, j_idx]**2 + dy[i_idx, j_idx]**2 - dr[i_idx, j_idx]**2])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-5, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-5, 0.5)])
    return b

def solve_lp_radii(centers):
    """Optimally scale radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        x_val, y_val = centers[i]
        bound = min(x_val, 1.0 - x_val, y_val, 1.0 - y_val)
        A_ub[idx, i] = 1.0
        b_ub[idx] = bound
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dist
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def generate_inits(n):
    """Generate diverse initial configurations."""
    inits = []
    patterns = [
        [5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [6, 6, 5, 5, 4], [5, 5, 5, 5, 6],
        [4, 6, 6, 6, 4], [5, 6, 6, 5, 4], [6, 4, 6, 5, 5], [5, 5, 6, 5, 5],
        [5, 6, 5, 5, 5], [6, 5, 5, 5, 5], [4, 5, 6, 5, 6]
    ]
    
    for pat in patterns:
        for r0 in [0.085, 0.090, 0.095, 0.100, 0.105]:
            x = np.zeros(3 * n)
            idx = 0
            y = r0
            for i, cnt in enumerate(pat):
                x_start = r0 + (r0 if i % 2 == 1 else 0)
                for k in range(cnt):
                    if idx >= n: break
                    cx = x_start + k * 2 * r0
                    x[3 * idx] = cx
                    x[3 * idx + 1] = y
                    x[3 * idx + 2] = r0
                    idx += 1
                y += np.sqrt(3) * r0
            while idx < n:
                x[3 * idx] = np.random.uniform(0.15, 0.85)
                x[3 * idx + 1] = np.random.uniform(0.15, 0.85)
                x[3 * idx + 2] = r0 * 0.7
                idx += 1
                
            # Ensure initial feasibility bounds
            for i in range(n):
                r = x[3*i+2]
                x[3*i] = np.clip(x[3*i], r + 1e-4, 1.0 - r - 1e-4)
                x[3*i+1] = np.clip(x[3*i+1], r + 1e-4, 1.0 - r - 1e-4)
                
            inits.append(x.copy())
            
            # Add rotated variants to break symmetry
            ang = np.random.uniform(-0.25, 0.25)
            pts = x.reshape(n, 3)[:, :2]
            cx_c, cy_c = 0.5, 0.5
            pts -= [cx_c, cy_c]
            rot = np.array([[np.cos(ang), -np.sin(ang)], [np.sin(ang), np.cos(ang)]])
            pts = (pts @ rot.T) + [cx_c, cy_c]
            
            x_rot = np.zeros(3 * n)
            x_rot[0::3] = pts[:, 0]
            x_rot[1::3] = pts[:, 1]
            x_rot[2::3] = x[2::3]
            
            for i in range(n):
                r = x_rot[3*i+2]
                x_rot[3*i] = np.clip(x_rot[3*i], r + 1e-4, 1.0 - r - 1e-4)
                x_rot[3*i+1] = np.clip(x_rot[3*i+1], r + 1e-4, 1.0 - r - 1e-4)
                
            inits.append(x_rot.copy())

    # Random perturbed grids
    for _ in range(15):
        x = np.zeros(3 * n)
        pts = []
        for r in range(6):
            for c in range(5):
                if len(pts) >= n: break

        pts = np.array(pts[:n])
        x[0::3] = pts[:, 0]
        x[1::3] = pts[:, 1]
        x[2::3] = 0.065

def run_packing():
    np.random.seed(42)
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -1.0
    best_x = None
    
    inits = generate_inits(N)
    
    # Phase 1: Multi-start optimization
    for x0 in inits:
        x0 += np.random.normal(0, 1e-4, 3 * N)
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, 
                           constraints=cons, options={'maxiter': 10000, 'ftol': 1e-13, 'disp': False})
            if not np.isnan(res.fun):
                curr_sum = -res.fun
                c_vals = constraints(res.x)
                if np.min(c_vals) >= -1e-6 and curr_sum > best_sum:
                    best_sum = curr_sum
                    best_x = res.x.copy()
        except Exception:
            pass
            
    # Phase 2: LP-Alternating Refinement
    if best_x is not None:
        for _ in range(6):
            centers = np.column_stack((best_x[0::3], best_x[1::3]))
            new_r = solve_lp_radii(centers)
            if new_r is not None:
                best_x[2::3] = new_r
                curr = np.sum(new_r)
                if curr > best_sum: best_sum = curr
                
                x0 = best_x + np.random.normal(0, 1e-5, 3 * N)
                try:
                    res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                                   constraints=cons, options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
                    if -res.fun > best_sum:
                        c_vals = constraints(res.x)
                        if np.min(c_vals) >= -1e-6:
                            best_x = res.x.copy()
                            best_sum = -res.fun
                except Exception: pass
                
    # Phase 3: Deflation & Re-optimization to escape local minima
    if best_x is not None:
        for scale in [0.995, 0.99, 0.985]:
            x0 = best_x.copy()
            x0[2::3] *= scale
            x0[0::3] += np.random.normal(0, 0.003, N)
            x0[1::3] += np.random.normal(0, 0.003, N)
            for i in range(N):
                r = x0[3*i+2]
                x0[3*i] = np.clip(x0[3*i], r, 1.0 - r)
                x0[3*i+1] = np.clip(x0[3*i+1], r, 1.0 - r)
                
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 8000, 'ftol': 1e-13, 'disp': False})
                if -res.fun > best_sum:
                    c_vals = constraints(res.x)
                    if np.min(c_vals) >= -1e-6:
                        best_x = res.x.copy()
                        best_sum = -res.fun
            except Exception: pass
            
    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3].copy()
    
    # Final safety shrink to guarantee strict compliance with validator
    for _ in range(50):
        valid = True
        for i in range(N):
            if radii[i] < 0: valid = False; break
            if centers[i,0] - radii[i] < -1e-10 or centers[i,0] + radii[i] > 1.0 + 1e-10: valid = False; break
            if centers[i,1] - radii[i] < -1e-10 or centers[i,1] + radii[i] > 1.0 + 1e-10: valid = False; break
        if valid:

                for j in range(i + 1, N):
                    if np.hypot(centers[i,0] - centers[j,0], centers[i,1] - centers[j,1]) < radii[i] + radii[j] - 1e-10:
                        valid = False; break
                if not valid: break
        if valid: break
        radii *= 0.995
        for i in range(N):
            centers[i,0] = np.clip(centers[i,0], radii[i], 1.0 - radii[i])
            centers[i,1] = np.clip(centers[i,1], radii[i], 1.0 - radii[i])
            
    return centers, radii, float(np.sum(radii))
