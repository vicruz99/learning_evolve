import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Returns all inequality constraints >= 0 (vectorized)."""
    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    c = np.concatenate([
        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs
    ])
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    i_idx, j_idx = np.tril_indices(N, -1)
    dist_sq = dx[i_idx, j_idx]**2 + dy[i_idx, j_idx]**2
    r_sum_sq = dr[i_idx, j_idx]**2
    
    c = np.concatenate([c, dist_sq - r_sum_sq])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-6, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)])
    return b

def greedy_init(n):
    """Generates a feasible initial configuration by greedily filling the largest gaps."""
    step = 0.02
    xs = np.arange(step, 1.0, step)
    ys = np.arange(step, 1.0, step)
    grid = np.array(np.meshgrid(xs, ys)).T.reshape(-1, 2)
    
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    used_mask = np.zeros(len(grid), dtype=bool)
    
    for i in range(n):
        # Distance to boundaries
        d_bound = np.minimum(grid[:, 0], 1.0 - grid[:, 0])
        d_bound = np.minimum(d_bound, np.minimum(grid[:, 1], 1.0 - grid[:, 1]))
        
        # Distance to existing circles
        d_circ = np.full(len(grid), 1.0)
        for j in range(i):
            dist = np.linalg.norm(grid - centers[j], axis=1)
            d_circ = np.minimum(d_circ, dist - radii[j])
            
        valid_d = np.minimum(d_bound, d_circ)
        valid_d[used_mask] = -1.0
        
        best_idx = np.argmax(valid_d)
        centers[i] = grid[best_idx]
        radii[i] = valid_d[best_idx] * 0.90  # Safety margin
        used_mask[best_idx] = True
        
    return centers, radii

def hex_init(n, seed, scale=1.0, angle=0.0):
    """Generates a hexagonal lattice initialization with perturbation."""
    np.random.seed(seed)
    r0 = 0.09 * scale
    centers = []
    y = r0
    row = 0
    while len(centers) < n and y + r0 <= 1.0:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while len(centers) < n and x + r0 <= 1.0:
            centers.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3) * r0
        row += 1
    while len(centers) < n:
        centers.append([np.random.uniform(0.1, 0.9), np.random.uniform(0.1, 0.9)])
        
    centers = np.array(centers[:n])
    radii = np.full(n, r0)
    
    if angle != 0.0:
        cx, cy = 0.5, 0.5
        ca, sa = np.cos(angle), np.sin(angle)
        dc = centers - [cx, cy]
        centers = dc @ np.array([[ca, -sa], [sa, ca]]) + [cx, cy]
        
    centers += np.random.normal(0, 0.004, centers.shape)
    centers = np.clip(centers, r0 + 1e-4, 1.0 - r0 - 1e-4)
    
    return centers, radii

def force_simulate(centers, radii, steps=300):
    """Simple force-directed relaxation to resolve overlaps."""
    n = centers.shape[0]
    for _ in range(steps):
        fx, fy = np.zeros(n), np.zeros(n)
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[j, 0] - centers[i, 0]
                dy = centers[j, 1] - centers[i, 1]
                dist = np.hypot(dx, dy)
                if dist < radii[i] + radii[j] and dist > 1e-6:
                    force = (radii[i] + radii[j] - dist) * 0.5 / dist
                    fx[i] -= force * dx
                    fy[i] -= force * dy
                    fx[j] += force * dx
                    fy[j] += force * dy
        for i in range(n):
            if centers[i, 0] < radii[i]: fx[i] += 0.1 * (radii[i] - centers[i, 0])
            if centers[i, 0] > 1.0 - radii[i]: fx[i] -= 0.1 * (centers[i, 0] - (1.0 - radii[i]))
            if centers[i, 1] < radii[i]: fy[i] += 0.1 * (radii[i] - centers[i, 1])
            if centers[i, 1] > 1.0 - radii[i]: fy[i] -= 0.1 * (centers[i, 1] - (1.0 - radii[i]))
            
        centers[:, 0] += fx * 0.1
        centers[:, 1] += fy * 0.1
        centers = np.clip(centers, 0.0, 1.0)
    return centers

def solve_lp_radii(centers):
    """Optimally scale radii for fixed centers using Linear Programming."""
    n = N
    c = -np.ones(n)
    
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    for i in range(n):
        x_val, y_val = centers[i]
        bound = min(x_val, 1.0 - x_val, y_val, 1.0 - y_val)
        A_ub[idx, i] = 1.0
        b_ub[idx] = bound
        idx += 1
        
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dist
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -1.0
    best_x = None

    # Greedy starts

        x0 = np.zeros(3*N)

                x0[0::3] = c[:,0]; x0[1::3] = c[:,1]; x0[2::3] = r
                inits.append(x0)
                
    # Phase 1: Optimization
    for x0 in inits:
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 15000, 'ftol': 1e-13, 'disp': False})
            if not np.isnan(res.fun):
                s_val = -res.fun
                c_vals = constraints(res.x)
                if np.min(c_vals) >= -1e-6 and s_val > best_sum:
                    best_sum = s_val
                    best_x = res.x.copy()
        except Exception:
            pass
            
    # Phase 2: Alternating Refinement (SLSQP positions <-> LP radii)
    if best_x is not None:
        for step in range(6):
            # Perturb to escape local minima
            x0 = best_x + np.random.normal(0, 0.0015 / (step + 1), 3*N)
            r_tmp = np.maximum(x0[2::3], 0.001)
            x0[0::3] = np.clip(x0[0::3], r_tmp, 1.0 - r_tmp)
            x0[1::3] = np.clip(x0[1::3], r_tmp, 1.0 - r_tmp)
            x0[2::3] = r_tmp
            
            # SLSQP optimization
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 10000, 'ftol': 1e-13})
                if not np.isnan(res.fun):
                    best_x = res.x.copy()
                    best_sum = -res.fun
                    
                    # LP radius refinement
                    centers = np.column_stack((best_x[0::3], best_x[1::3]))
                    new_radii = solve_lp_radii(centers)
                    if new_radii is not None:
                        best_x[2::3] = new_radii
                        curr_sum = np.sum(new_radii)
                        if curr_sum > best_sum:
                            best_sum = curr_sum
            except Exception:
                pass
                
    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3]

    for _ in range(50):
        valid = True

                    if d < radii[i]+radii[j]-1e-10:
                        valid=False; break
                if not valid: break
        if valid: break
        radii *= 0.999
        centers[:,0] = np.clip(centers[:,0], radii, 1-radii)
        centers[:,1] = np.clip(centers[:,1], radii, 1-radii)
        
    return centers, radii, float(np.sum(radii))
