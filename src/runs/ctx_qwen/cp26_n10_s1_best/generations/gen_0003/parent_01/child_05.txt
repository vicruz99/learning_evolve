import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def solve_lp_radii(centers):
    """Solve LP to maximize sum of radii for fixed centers."""
    n = N
    c = -np.ones(n)
    
    num_constraints = 4 * n + n * (n - 1) // 2
    A = np.zeros((num_constraints, n))
    b = np.zeros(num_constraints)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        x, y = centers[i]
        lim = np.min([x, 1.0 - x, y, 1.0 - y])
        A[idx, i] = 1.0
        b[idx] = lim
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = d
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def evaluate(centers):
    """Returns sum of radii and the radii array for fixed centers."""
    radii = solve_lp_radii(centers)
    if radii is not None:
        return np.sum(radii), radii
    return 0.0, None

def generate_hex_init(n, r0, angle_deg=0):
    """Generates a hexagonal lattice initialization, optionally rotated."""
    pts = []
    y = r0
    row = 0
    while len(pts) < n + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        if y > 1.0 + r0:
            break
            
    pts = np.array(pts[:n + 10])
    
    # Rotate around center
    if angle_deg != 0:
        rad = np.deg2rad(angle_deg)
        c, s = np.cos(rad), np.sin(rad)
        center = np.array([0.5, 0.5])
        pts = (pts - center) @ np.array([[c, -s], [s, c]]) + center
        
    # Filter and pad
    mask = (pts[:, 0] >= 0.05) & (pts[:, 0] <= 0.95) & \
           (pts[:, 1] >= 0.05) & (pts[:, 1] <= 0.95)
    pts = pts[mask]
    while len(pts) < n:
        pts = np.vstack([pts, [np.random.uniform(0.2, 0.8), np.random.uniform(0.2, 0.8)]])
        
    return pts[:n]

def run_simulated_annealing(centers, iterations=15000):
    """Runs SA to optimize circle positions using LP for radius evaluation."""
    np.random.seed(np.random.randint(0, 10000))
    best_centers = centers.copy()
    best_sum, best_radii = evaluate(best_centers)
    
    T = 0.06
    step_size = 0.035
    
    for _ in range(iterations):
        # Perturb centers
        noise = np.random.normal(0, step_size, centers.shape)
        new_centers = centers + noise
        new_centers = np.clip(new_centers, 0.015, 0.985)
        
        curr_sum, curr_radii = evaluate(new_centers)
        
        if curr_radii is not None:
            delta = curr_sum - best_sum
            if delta > 0 or np.random.rand() < np.exp(delta / T):
                best_centers = new_centers.copy()
                best_sum = curr_sum
                best_radii = curr_radii.copy()
                centers = new_centers.copy()
            
        T *= 0.9998
        step_size *= 0.9998
        
    return best_centers, best_radii, best_sum

def objective_joint(x):
    """Objective for joint optimization: minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints_joint(x):
    """Vectorized constraints for joint optimization."""
    n = N
    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]
    
    c = np.concatenate([
        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs
    ])
    
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    i_idx, j_idx = np.tril_indices(n, -1)
    c = np.concatenate([c, dx[i_idx, j_idx]**2 + dy[i_idx, j_idx]**2 - dr[i_idx, j_idx]**2])
    return c

def run_packing():
    np.random.seed(123)
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N
    
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Phase 1: Generate diverse initial configurations and run SA
    inits = []
    for r0 in [0.085, 0.095, 0.105]:
        for angle in [0, 12, 25, 40, 60]:
            inits.append(generate_hex_init(N, r0, angle))
            
    for _ in range(5):
        c = np.random.uniform(0.15, 0.85, (N, 2))
        inits.append(c)

    for c_init in inits:
        c_sa, r_sa, s_sa = run_simulated_annealing(c_init, iterations=12000)
        if s_sa > best_sum:
            best_sum = s_sa
            best_centers = c_sa.copy()
            best_radii = r_sa.copy()
            
    # Phase 2: Joint refinement with SLSQP
    x0 = np.zeros(3 * N)
    x0[0::3] = best_centers[:, 0]
    x0[1::3] = best_centers[:, 1]
    x0[2::3] = best_radii
    
    cons = {'type': 'ineq', 'fun': constraints_joint}
    
    # Multiple restarts around best SA solution
    for trial in range(15):
        x0_pert = x0 + np.random.normal(0, 1e-4 * (1.5 ** (trial % 5)), 3 * N)
        x0_pert[2::3] = np.clip(x0_pert[2::3], 0.005, 0.45)
        
        try:
            res = minimize(objective_joint, x0_pert, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 8000, 'ftol': 1e-14, 'disp': False})
            if not np.isnan(res.fun):
                curr_sum = -res.fun
                if curr_sum > best_sum:
                    best_sum = curr_sum
                    best_centers = np.column_stack((res.x[0::3], res.x[1::3]))
                    best_radii = res.x[2::3].copy()
                    x0 = res.x.copy()
        except Exception:
            continue
            
    # Phase 3: Final validation and minimal repair
    valid = True
    for i in range(N):
        if best_radii[i] < 0:
            valid = False; break
        if best_centers[i, 0] - best_radii[i] < -1e-10 or best_centers[i, 0] + best_radii[i] > 1.0 + 1e-10:
            valid = False; break

            best_centers[:, 0] = np.clip(best_centers[:, 0], best_radii, 1.0 - best_radii)
            best_centers[:, 1] = np.clip(best_centers[:, 1], best_radii, 1.0 - best_radii)
            
            is_ok = True
            for i in range(N):
                if best_radii[i] < 0: is_ok = False; break
                if best_centers[i, 0] - best_radii[i] < -1e-12 or best_centers[i, 0] + best_radii[i] > 1.0 + 1e-12: is_ok = False; break
                if best_centers[i, 1] - best_radii[i] < -1e-12 or best_centers[i, 1] + best_radii[i] > 1.0 + 1e-12: is_ok = False; break
            if is_ok:

                        if d < best_radii[i] + best_radii[j] - 1e-12:
                            is_ok = False; break
                    if not is_ok: break
            if is_ok:
                break
