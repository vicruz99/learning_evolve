import numpy as np
from scipy.optimize import minimize, linprog

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def unpack(x):
    """Converts flat optimization vector to centers and radii arrays."""
    return x[0::3], x[1::3], x[2::3]

def objective(x):
    """Objective to minimize: negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Returns vector of inequality constraints g(x) >= 0."""
    xs, ys, rs = unpack(x)
    
    # Boundary constraints: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    c = np.concatenate([
        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs
    ])
    
    # Overlap constraints: dist^2 - (r_i + r_j)^2 >= 0
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    c = np.concatenate([c, dx[TRIL_IDX[0], TRIL_IDX[1]]**2 + dy[TRIL_IDX[0], TRIL_IDX[1]]**2 - dr[TRIL_IDX[0], TRIL_IDX[1]]**2])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [0, 0.5]."""
    return [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N

def solve_lp_radii(centers):
    """Optimally scale radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    idx = 0
    
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        b_ub[idx] = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = d
            idx += 1
            
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=[(0.0, None)] * n, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def force_relax(centers, radii, steps=100):
    """Force-directed relaxation to resolve overlaps and boundary violations."""
    c = centers.copy()
    r = radii.copy()
    for _ in range(steps):
        f = np.zeros_like(c)
        for i in range(N):
            for j in range(i + 1, N):
                dx = c[j, 0] - c[i, 0]
                dy = c[j, 1] - c[i, 1]
                d = np.hypot(dx, dy)
                if d < r[i] + r[j] and d > 1e-8:
                    push = (r[i] + r[j] - d) * 0.5 / d
                    f[i, 0] -= push * dx
                    f[i, 1] -= push * dy
                    f[j, 0] += push * dx
                    f[j, 1] += push * dy
            # Repel from boundaries
            for axis, val in [(0, c[i, 0]), (0, 1.0 - c[i, 0]), (1, c[i, 1]), (1, 1.0 - c[i, 1])]:
                if val < r[i]:
                    f[i, axis] += (r[i] - val) * 0.5
        c += f * 0.1
        c = np.clip(c, 0.001, 0.999)
    return c, r

def run_packing():
    """Pack 26 circles in unit square to maximize sum of radii."""
    best_x = None
    best_val = -1.0
    bnds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    rng = np.random.RandomState(42)

    # Phase 1: Generate diverse initial configurations
    inits = []
    
    # Hexagonal lattices with various rotations to fit square boundaries better
    for ang in [0.0, 0.1, -0.1, 0.2, -0.2, 0.05, -0.05, 0.15]:
        c = np.zeros((N, 2))
        idx = 0
        y = 0.09
        row = 0
        while idx < N and y + 0.09 <= 1.0:
            x = 0.09 if row % 2 == 0 else 0.18
            while idx < N and x + 0.09 <= 1.0:
                c[idx] = [x, y]
                idx += 1
                x += 0.18
            y += 0.09 * np.sqrt(3)
            row += 1
        while idx < N:
            c[idx] = [0.5, 0.5]
            idx += 1
            
        if ang != 0.0:
            c -= 0.5
            rot = np.array([[np.cos(ang), -np.sin(ang)], [np.sin(ang), np.cos(ang)]])
            c = c @ rot
            c += 0.5
        inits.append(c.copy())

    # Random starts
    for _ in range(15):
        c = rng.uniform(0.1, 0.9, (N, 2))
        inits.append(c.copy())

    # Phase 2: SLSQP from multiple starts with force pre-relaxation
    for c_init in inits:
        r_init = np.full(N, 0.09)
        x0 = np.zeros(3 * N)
        x0[0::3] = c_init[:, 0]
        x0[1::3] = c_init[:, 1]
        x0[2::3] = r_init
        x0 += rng.normal(0, 0.001, 3 * N)
        x0[2::3] = np.clip(x0[2::3], 0.01, 0.4)

        # Resolve gross overlaps first
        c_f, r_f = force_relax(c_init, r_init, steps=80)
        x0[0::3] = c_f[:, 0]
        x0[1::3] = c_f[:, 1]
        x0[2::3] = r_f

        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bnds, constraints=cons,
                           options={'maxiter': 25000, 'ftol': 1e-13, 'disp': False})
            if not np.isnan(res.fun) and -res.fun > best_val:
                best_val = -res.fun
                best_x = res.x.copy()
        except Exception:
            continue

    # Phase 3: Alternating LP/SLSQP Refinement
    # This alternates between exact radius maximization (LP) and position optimization (SLSQP)
    # while slightly expanding radii to force denser configurations.
    if best_x is not None:
        for it in range(15):
            centers = np.column_stack((best_x[0::3], best_x[1::3]))
            new_radii = solve_lp_radii(centers)
            if new_radii is not None:
                best_x[2::3] = new_radii
                # Slight expansion pushes boundaries and forces centers to rearrange
                best_x[2::3] *= 1.002
                for i in range(N):
                    ri = best_x[3 * i + 2]
                    best_x[3 * i] = np.clip(best_x[3 * i], ri, 1.0 - ri)
                    best_x[3 * i + 1] = np.clip(best_x[3 * i + 1], ri, 1.0 - ri)

            # Adaptive perturbation to escape local minima
            pert_scale = 0.001 * (1.0 + 0.5 * (it % 5))
            x_pert = best_x + rng.normal(0, pert_scale, 3 * N)
            x_pert[0::3] = np.clip(x_pert[0::3], 0.001, 0.999)
            x_pert[1::3] = np.clip(x_pert[1::3], 0.001, 0.999)

            try:
                res = minimize(objective, x_pert, method='SLSQP', bounds=bnds, constraints=cons,
                               options={'maxiter': 20000, 'ftol': 1e-13, 'disp': False})
                if not np.isnan(res.fun) and -res.fun > best_val:
                    best_val = -res.fun
                    best_x = res.x.copy()
            except Exception:
                continue

    # Extract final results
    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3]

    # Final deterministic repair to guarantee strict compliance with validator tolerance
    for _ in range(100):
        valid = True
        for i in range(N):
            if radii[i] < 0:
                valid = False
                break
            if (centers[i, 0] - radii[i] < -1e-9 or centers[i, 0] + radii[i] > 1.0 + 1e-9 or
                centers[i, 1] - radii[i] < -1e-9 or centers[i, 1] + radii[i] > 1.0 + 1e-9):
                valid = False
                break
        if valid:
            for i in range(N):
                for j in range(i + 1, N):
                    d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                    if d < radii[i] + radii[j] - 1e-9:
                        valid = False
                        break
                if not valid:
                    break
        if valid:
            break
        # Minimal shrinkage to recover validity
        radii *= 0.998
        for i in range(N):
            centers[i, 0] = np.clip(centers[i, 0], radii[i], 1.0 - radii[i])
            centers[i, 1] = np.clip(centers[i, 1], radii[i], 1.0 - radii[i])

    return centers, radii, float(np.sum(radii))
