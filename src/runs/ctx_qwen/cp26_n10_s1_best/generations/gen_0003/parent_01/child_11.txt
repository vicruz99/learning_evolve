import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def unpack(x):
    """Unpack flat optimization vector into xs, ys, rs arrays."""
    return x[0::3], x[1::3], x[2::3]

def objective(x):
    """Objective to minimize: negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Vectorized inequality constraints g(x) >= 0."""
    xs, ys, rs = unpack(x)
    n = N
    num_overlap = n * (n - 1) // 2
    c = np.empty(4 * n + num_overlap)
    
    # Boundary constraints
    c[:n] = xs - rs
    c[n:2*n] = 1.0 - xs - rs
    c[2*n:3*n] = ys - rs
    c[3*n:4*n] = 1.0 - ys - rs
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    idx = 4 * n
    for i in range(n):
        for j in range(i + 1, n):
            c[idx] = (xs[i] - xs[j])**2 + (ys[i] - ys[j])**2 - (rs[i] + rs[j])**2
            idx += 1
    return c

def jacobian(x):
    """Exact Jacobian of the constraint function."""
    xs, ys, rs = unpack(x)
    n = N
    num_overlap = n * (n - 1) // 2
    num_cons = 4 * n + num_overlap
    J = np.zeros((num_cons, 3 * n))
    
    # Boundary Jacobian
    for i in range(n):
        row = 4 * i
        J[row, 3*i] = 1.0; J[row, 3*i+2] = -1.0
        J[row+1, 3*i] = -1.0; J[row+1, 3*i+2] = -1.0
        J[row+2, 3*i+1] = 1.0; J[row+2, 3*i+2] = -1.0
        J[row+3, 3*i+1] = -1.0; J[row+3, 3*i+2] = -1.0
        
    # Overlap Jacobian
    idx = 4 * n
    for i in range(n):
        for j in range(i + 1, n):
            dx = xs[i] - xs[j]
            dy = ys[i] - ys[j]
            dr = rs[i] + rs[j]
            
            J[idx, 3*i] = 2.0 * dx
            J[idx, 3*i+1] = 2.0 * dy
            J[idx, 3*i+2] = -2.0 * dr
            
            J[idx, 3*j] = -2.0 * dx
            J[idx, 3*j+1] = -2.0 * dy
            J[idx, 3*j+2] = -2.0 * dr
            idx += 1
    return J

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-7, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)])
    return b

def make_feasible(x0):
    """Ensure strict feasibility by computing safe radii for given centers."""
    xs, ys, rs = unpack(x0)
    n = N
    safe_rs = np.zeros(n)
    for i in range(n):
        max_r = min(xs[i], 1.0 - xs[i], ys[i], 1.0 - ys[i])
        for j in range(n):
            if i != j:
                d = np.hypot(xs[i] - xs[j], ys[i] - ys[j])
                max_r = min(max_r, d / 2.0)
        safe_rs[i] = max(max_r * 0.95, 1e-7)
        
    x0[2::3] = safe_rs
    for i in range(n):
        r = x0[3*i+2]
        x0[3*i] = np.clip(x0[3*i], r, 1.0 - r)
        x0[3*i+1] = np.clip(x0[3*i+1], r, 1.0 - r)
    return x0

def solve_lp_radii(centers):
    """Maximize sum of radii for fixed centers using Linear Programming."""
    n = N
    c = -np.ones(n)
    num_cons = n + n * (n - 1) // 2
    A_ub = np.zeros((num_cons, n))
    b_ub = np.zeros(num_cons)
    
    idx = 0
    for i in range(n):
        x, y = centers[i]
        A_ub[idx, i] = 1.0
        b_ub[idx] = min(x, 1.0 - x, y, 1.0 - y)
        idx += 1
        
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i,0] - centers[j,0], centers[i,1] - centers[j,1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = d
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def generate_hex_init(seed, r0=0.095, angle=0.0):
    """Generate hexagonal lattice initialization with optional rotation."""
    np.random.seed(seed)
    xs = np.zeros(N)
    ys = np.zeros(N)
    idx = 0
    y = r0
    row = 0
    while idx < N and y + r0 <= 1.0:
        x_start = r0 if row % 2 == 0 else 2 * r0
        x = x_start
        while idx < N and x + r0 <= 1.0:
            xs[idx] = x
            ys[idx] = y
            idx += 1
            x += 2 * r0
        y += np.sqrt(3) * r0
        row += 1
        
    while idx < N:
        xs[idx] = np.random.uniform(0.1, 0.9)
        ys[idx] = np.random.uniform(0.1, 0.9)
        idx += 1
        
    if angle != 0.0:
        c_x, c_y = 0.5, 0.5
        cos_a, sin_a = np.cos(angle), np.sin(angle)
        dx = xs - c_x
        dy = ys - c_y
        xs = dx * cos_a - dy * sin_a + c_x
        ys = dx * sin_a + dy * cos_a + c_y
        
    xs += np.random.normal(0, 0.002, N)
    ys += np.random.normal(0, 0.002, N)
    
    x0 = np.zeros(3 * N)
    x0[0::3] = xs
    x0[1::3] = ys
    x0[2::3] = r0
    return x0

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints, 'jac': jacobian}
    
    best_sum = -1.0
    best_x = None
    
    # Phase 1: Diverse starts
    inits = []
    for s in range(25):
        inits.append(generate_hex_init(s, r0=0.095, angle=0.0))
        inits.append(generate_hex_init(s, r0=0.10, angle=np.radians(15)))
        inits.append(generate_hex_init(s, r0=0.085, angle=np.radians(30)))
        
        # Random starts to break lattice symmetry
        if s % 5 == 0:
            x0_rand = np.zeros(3 * N)
            x0_rand[0::3] = np.random.uniform(0.1, 0.9, N)
            x0_rand[1::3] = np.random.uniform(0.1, 0.9, N)
            x0_rand[2::3] = 0.06
            inits.append(x0_rand)
            
    for x0 in inits:
        x0_safe = make_feasible(x0.copy())
        try:
            res = minimize(objective, x0_safe, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 12000, 'ftol': 1e-14, 'disp': False})
            if not np.isnan(res.fun):
                curr_sum = -res.fun
                if curr_sum > best_sum:
                    best_sum = curr_sum
                    best_x = res.x.copy()
        except Exception:
            pass
            
    # Phase 2: Alternating LP + SLSQP refinement
    if best_x is not None:
        for _ in range(12):
            centers = np.column_stack((best_x[0::3], best_x[1::3]))
            new_rs = solve_lp_radii(centers)
            
            if new_rs is not None:
                best_x[2::3] = new_rs
                curr_sum = np.sum(new_rs)
                if curr_sum > best_sum:
                    best_sum = curr_sum
                    
                # Perturb centers slightly to escape degeneracy
                x_pert = best_x.copy()
                x_pert[0::3] += np.random.normal(0, 1e-4, N)
                x_pert[1::3] += np.random.normal(0, 1e-4, N)
                x_pert = make_feasible(x_pert)
                
                try:
                    res = minimize(objective, x_pert, method='SLSQP', bounds=bounds,
                                   constraints=cons, options={'maxiter': 8000, 'ftol': 1e-14, 'disp': False})
                    if not np.isnan(res.fun):
                        curr_sum = -res.fun
                        if curr_sum > best_sum:
                            best_sum = curr_sum
                            best_x = res.x.copy()
                except Exception:
                    pass
                    
    # Phase 3: Cooling perturbations
    if best_x is not None:
        for temp in np.linspace(0.0015, 0.0001, 10):
            x_pert = best_x.copy()
            x_pert += np.random.normal(0, temp, 3 * N)
            x_pert = make_feasible(x_pert)
            
            try:
                res = minimize(objective, x_pert, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 5000, 'ftol': 1e-14, 'disp': False})
                if not np.isnan(res.fun):
                    curr_sum = -res.fun
                    if curr_sum > best_sum:
                        best_sum = curr_sum
                        best_x = res.x.copy()
            except Exception:
                pass

    centers = np.zeros((N, 2))
    radii = np.zeros(N)
    centers[:, 0] = best_x[0::3]
    centers[:, 1] = best_x[1::3]
    radii[:] = best_x[2::3]
    
    # Final validation and minimal repair to guarantee strict compliance
    for _ in range(100):
        valid = True
        for i in range(N):
            if radii[i] < 1e-9: valid = False; break
            if centers[i,0] - radii[i] < -1e-9 or centers[i,0] + radii[i] > 1 + 1e-9: valid = False; break
            if centers[i,1] - radii[i] < -1e-9 or centers[i,1] + radii[i] > 1 + 1e-9: valid = False; break
        if valid:
            for i in range(N):
                for j in range(i + 1, N):
                    d = np.hypot(centers[i,0] - centers[j,0], centers[i,1] - centers[j,1])
                    if d < radii[i] + radii[j] - 1e-9:
                        valid = False; break
                if not valid: break
        if valid: break
        radii *= 0.999
        for i in range(N):
            centers[i,0] = np.clip(centers[i,0], radii[i], 1-radii[i])
            centers[i,1] = np.clip(centers[i,1], radii[i], 1-radii[i])
            
    return centers, radii, float(np.sum(radii))
