import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def objective_ind(p):
    """Objective for independent radii: maximize sum of radii."""
    return -np.sum(p[2::3])

def constraints_ind(p):
    """Inequality constraints for independent radii packing."""
    xs = p[0::3]
    ys = p[1::3]
    rs = p[2::3]
    c = np.concatenate([xs - rs, 1.0 - xs - rs, ys - rs, 1.0 - ys - rs])
    i, j = np.tril_indices(N, -1)
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    c = np.concatenate([c, dx[i, j]**2 + dy[i, j]**2 - dr[i, j]**2])
    return c

def bounds_ind():
    """Variable bounds for independent optimization."""
    return [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N

def objective_eq(params):
    """Objective for equal radius packing: maximize single radius R."""
    return -params[-1]

def constraints_eq(params):
    """Inequality constraints for equal radius packing."""
    centers = params[:-1].reshape(N, 2)
    R = params[-1]
    c = np.concatenate([centers[:, 0] - R, 1.0 - centers[:, 0] - R,
                        centers[:, 1] - R, 1.0 - centers[:, 1] - R])
    i, j = np.tril_indices(N, -1)
    dx = centers[:, 0][i] - centers[:, 0][j]
    dy = centers[:, 1][i] - centers[:, 1][j]
    c = np.concatenate([c, dx**2 + dy**2 - 4.0 * R * R])
    return c

def bounds_eq():
    """Variable bounds for equal radius optimization."""
    return [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)]

def make_hex(counts, r_est=0.1):
    """Generate hexagonal lattice initialization."""
    centers = []
    y = r_est
    row = 0
    for cnt in counts:
        x = r_est + (r_est if row % 2 == 1 else 0)
        for _ in range(cnt):
            centers.append([x, y])
            x += 2 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
    c = np.array(centers[:N])
    return np.clip(c, 0.05, 0.95)

def force_layout(seed, steps=500):
    """Generate force-directed layout initialization."""
    np.random.seed(seed)
    pts = np.random.uniform(0.15, 0.85, (N, 2))
    for _ in range(steps):
        forces = np.zeros_like(pts)
        for i in range(N):
            for j in range(i + 1, N):
                d = pts[j] - pts[i]
                dist = np.hypot(d[0], d[1])
                if dist < 0.3 and dist > 1e-6:
                    f = 0.001 / (dist**2)
                    forces[i] -= f * d
                    forces[j] += f * d
        pts += forces * 0.1
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def solve_lp_radii(centers):
    """Solve LP to optimally assign radii for fixed centers."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    idx = 0
    for i in range(n):
        bound = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = bound
        idx += 1
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dist
            idx += 1
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=[(0.0, None)] * n, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    best_sum = 0.0
    best_centers = None
    best_radii = None

    # Phase 1: Target equal-circle dense packings (known to be near-optimal for sum of radii)
    patterns = [
        [6, 5, 6, 5, 4], [5, 6, 5, 6, 4], [6, 6, 5, 5, 4],
        [5, 5, 6, 6, 4], [7, 6, 7, 6], [6, 7, 6, 7], [5, 6, 7, 5, 3]
    ]
    
    inits_eq = []
    for pat in patterns:
        if sum(pat) >= N:
            inits_eq.append(make_hex(pat, 0.095))
    for s in range(5):
        inits_eq.append(force_layout(s))
        
    for layout in inits_eq:
        x0 = np.concatenate([layout.flatten(), [0.085]])
        try:
            res = minimize(objective_eq, x0, method='SLSQP', bounds=bounds_eq(),
                           constraints={'type': 'ineq', 'fun': constraints_eq},
                           options={'maxiter': 8000, 'ftol': 1e-14, 'disp': False})
            if not np.isnan(res.fun):
                R_opt = res.x[-1]
                c_opt = res.x[:-1].reshape(N, 2)
                
                # Validate equal config
                valid = True
                for i in range(N):
                    if c_opt[i,0] < R_opt - 1e-7 or c_opt[i,0] > 1-R_opt + 1e-7 or \
                       c_opt[i,1] < R_opt - 1e-7 or c_opt[i,1] > 1-R_opt + 1e-7:
                        valid = False; break
                if valid:
                    for i in range(N):
                        for j in range(i+1, N):
                            if np.hypot(c_opt[i,0]-c_opt[j,0], c_opt[i,1]-c_opt[j,1]) < 2*R_opt - 1e-7:
                                valid = False; break
                        if not valid: break
                        
                if valid:
                    # Phase 2: Independent radii refinement
                    radii_init = np.full(N, R_opt)
                    x0_ind = np.zeros(3*N)
                    x0_ind[0::3] = c_opt[:, 0]
                    x0_ind[1::3] = c_opt[:, 1]
                    x0_ind[2::3] = radii_init
                    
                    # LP pre-refinement for better starting radii
                    lp_r = solve_lp_radii(c_opt)
                    if lp_r is not None:
                        x0_ind[2::3] = lp_r
                        
                    # Joint SLSQP optimization
                    try:
                        res2 = minimize(objective_ind, x0_ind, method='SLSQP', bounds=bounds_ind(),
                                        constraints={'type': 'ineq', 'fun': constraints_ind},
                                        options={'maxiter': 10000, 'ftol': 1e-14, 'disp': False})
                        if not np.isnan(res2.fun):
                            s = -res2.fun
                            if s > best_sum:
                                best_sum = s
                                best_centers = res2.x[0::3].reshape(N, 2).copy()
                                best_radii = res2.x[2::3].copy()
                    except Exception:
                        pass
        except Exception:
            continue

    # Phase 3: Perturbation and local refinement to escape shallow minima
    if best_centers is not None:
        for _ in range(8):
            noise = np.random.normal(0, 0.002, best_centers.shape)
            p_c = np.clip(best_centers + noise, 0.02, 0.98)
            p_r = best_radii * 0.98
            x0 = np.zeros(3*N)
            x0[0::3] = p_c[:, 0]
            x0[1::3] = p_c[:, 1]
            x0[2::3] = p_r
            try:
                res3 = minimize(objective_ind, x0, method='SLSQP', bounds=bounds_ind(),
                                constraints={'type': 'ineq', 'fun': constraints_ind},
                                options={'maxiter': 5000, 'ftol': 1e-14, 'disp': False})
                if not np.isnan(res3.fun):
                    s = -res3.fun
                    if s > best_sum:
                        best_sum = s
                        best_centers = res3.x[0::3].reshape(N, 2).copy()
                        best_radii = res3.x[2::3].copy()
            except Exception:
                pass

    # Fallback
    if best_centers is None:
        best_centers = make_hex([6,5,6,5,4], 0.09)
        best_radii = np.full(N, 0.09)
        best_sum = np.sum(best_radii)

    # Strict validation and repair loop
    for _ in range(50):
        valid = True
        for i in range(N):
            if best_radii[i] < 0: valid = False; break
            if best_centers[i,0] < best_radii[i] - 1e-10 or best_centers[i,0] > 1 - best_radii[i] + 1e-10 or \
               best_centers[i,1] < best_radii[i] - 1e-10 or best_centers[i,1] > 1 - best_radii[i] + 1e-10:
                valid = False; break
        if valid:
            for i in range(N):
                for j in range(i+1, N):
                    if np.hypot(best_centers[i,0]-best_centers[j,0], best_centers[i,1]-best_centers[j,1]) < best_radii[i]+best_radii[j] - 1e-10:
                        valid = False; break
                if not valid: break
        if valid: break
        best_radii *= 0.995
        
    return best_centers, best_radii, float(np.sum(best_radii))
