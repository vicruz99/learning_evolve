import numpy as np
from scipy.optimize import minimize, linprog

N_CIRCLES = 26
TRIL_IDX = np.tril_indices(N_CIRCLES, -1)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def get_constraints(x):
    """Compute all inequality constraints g(x) >= 0 efficiently."""
    n = N_CIRCLES
    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]
    
    # Boundary constraints
    c = np.concatenate([
        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs
    ])
    
    # Overlap constraints: dist^2 - (r_i + r_j)^2 >= 0
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    c = np.concatenate([c, dx[TRIL_IDX]**2 + dy[TRIL_IDX]**2 - dr[TRIL_IDX]**2])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [0, 0.5]."""
    return [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N_CIRCLES

def generate_hex_init(r_start, angle=0.0, scale=1.0):

    n = N_CIRCLES
    centers = np.zeros((n, 2))
    idx = 0
    y = r_start
    row = 0
    while idx < n and y + r_start <= 1.0:
        x = r_start if row % 2 == 0 else 2.0 * r_start
        while idx < n and x + r_start <= 1.0:
            centers[idx] = [x, y]
            idx += 1
            x += 2.0 * r_start
        y += np.sqrt(3) * r_start
        row += 1

    while idx < n:
        centers[idx] = np.random.uniform(0.15, 0.85, 2)
        idx += 1

    if angle != 0.0 or scale != 1.0:
        cx, cy = 0.5, 0.5
        centers -= [cx, cy]
        centers *= scale

        centers = centers @ np.array([[cos_a, -sin_a], [sin_a, cos_a]])
        centers += [cx, cy]
        
    return centers

def force_spread_init(seed):

    np.random.seed(seed)
    n = N_CIRCLES
    centers = np.random.uniform(0.2, 0.8, (n, 2))

        forces = np.zeros_like(centers)

            for j in range(i + 1, n):
                diff = centers[j] - centers[i]
                dist = np.hypot(diff[0], diff[1])
                if dist < 0.35 and dist > 1e-6:
                    f = 0.002 / (dist**2 + 0.001)
                    forces[i] -= f * diff
                    forces[j] += f * diff
            # Push away from center slightly to utilize boundaries
            diff = centers[i] - 0.5
            dist = np.hypot(diff[0], diff[1])
            if dist > 0.01:
                forces[i] += 0.0008 * diff / dist
                
        centers += forces * 0.12
        centers = np.clip(centers, 0.05, 0.95)
    return centers

def solve_lp_radii(centers):
    """Solves LP to maximize sum of radii for fixed centers."""
    n = N_CIRCLES
    c_obj = -np.ones(n)
    
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        bound = min(centers[i, 0], 1.0 - centers[i, 0], 
                    centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = bound
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], 
                            centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dist
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None
