import numpy as np
from scipy.optimize import minimize, linprog

N_CIRCLES = 26
TRIL_IDX = np.tril_indices(N_CIRCLES, -1)

def objective(vars):
    """Minimize negative sum of radii."""
    return -np.sum(vars[2::3])

def constraint_func(vars):
    """Vectorized inequality constraints: g(vars) >= 0."""
    xs = vars[0::3]
    ys = vars[1::3]
    rs = vars[2::3]
    
    n_cons = 4 * N_CIRCLES + N_CIRCLES * (N_CIRCLES - 1) // 2
    cons = np.empty(n_cons)
    
    # Boundary constraints
    cons[:N_CIRCLES] = xs - rs
    cons[N_CIRCLES:2*N_CIRCLES] = 1.0 - xs - rs
    cons[2*N_CIRCLES:3*N_CIRCLES] = ys - rs
    cons[3*N_CIRCLES:4*N_CIRCLES] = 1.0 - ys - rs
    
    # Overlap constraints: dist >= r_i + r_j
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dist = np.sqrt(dx**2 + dy**2 + 1e-14)  # +epsilon prevents gradient singularity
    r_sum = rs[:, None] + rs[None, :]
    
    cons[4*N_CIRCLES:] = dist[TRIL_IDX] - r_sum[TRIL_IDX]
    return cons

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-7, 0.5]."""
    return [(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)] * N_CIRCLES

def solve_lp_radii(centers):
    """Solve LP to maximize sum of radii for fixed centers."""
    n = N_CIRCLES
    c_obj = -np.ones(n)
    num_con = n + n * (n - 1) // 2
    A = np.zeros((num_con, n))
    b = np.zeros(num_con)
    idx = 0
    
    # Boundary: r_i <= dist_to_wall
    for i in range(n):
        A[idx, i] = 1.0
        b[idx] = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        idx += 1
        
    # Overlap: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = d
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def generate_hex_init(seed, r_start=0.09, angle=0.0):
    """Generates a hexagonal lattice initialization with optional rotation."""
    np.random.seed(seed)
    centers = np.zeros((N_CIRCLES, 2))
    idx = 0
    y = r_start
    row = 0
    while idx < N_CIRCLES and y + r_start <= 1.0:
        x = r_start if row % 2 == 0 else 2.0 * r_start
        while idx < N_CIRCLES and x + r_start <= 1.0:
            centers[idx] = [x, y]
            idx += 1
            x += 2.0 * r_start
        y += np.sqrt(3.0) * r_start
        row += 1
    while idx < N_CIRCLES:
        centers[idx] = np.random.uniform(0.1, 0.9, 2)
        idx += 1
        
    if angle != 0.0:
        cx, cy = 0.5, 0.5
        centers -= [cx, cy]
        c, s = np.cos(angle), np.sin(angle)
        centers = centers @ np.array([[c, -s], [s, c]])
        centers *= np.random.uniform(0.95, 1.05)
        centers += [cx, cy]
        
    centers += np.random.uniform(-0.005, 0.005, centers.shape)
    centers = np.clip(centers, 0.02, 0.98)
    
    x0 = np.zeros(3 * N_CIRCLES)
    x0[0::3] = centers[:, 0]
    x0[1::3] = centers[:, 1]
    x0[2::3] = r_start
    return x0

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    n = N_CIRCLES
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraint_func}
    
    best_sum = -1.0
    best_x = None

        inits.append(generate_hex_init(s, r_start=0.08, angle=np.random.uniform(-0.4, 0.4)))
        
    for x0 in inits:
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,

            if not np.isnan(res.fun):
                curr_sum = -res.fun
                c_vals = constraint_func(res.x)
                if np.min(c_vals) >= -1e-6 and curr_sum > best_sum:
                    best_sum = curr_sum
                    best_x = res.x.copy()
        except Exception:
            continue

    if best_x is not None:
        for iteration in range(20):

            x_pert = best_x + np.random.normal(0, noise_scale, best_x.shape)
            
            for i in range(n):
                r = max(1e-6, x_pert[3*i+2])
                x_pert[3*i] = np.clip(x_pert[3*i], r, 1.0 - r)
                x_pert[3*i+1] = np.clip(x_pert[3*i+1], r, 1.0 - r)
                x_pert[3*i+2] = r

            new_radii = solve_lp_radii(centers)
            if new_radii is not None:

                    best_x[2::3] = new_radii
                    best_sum = lp_sum

                        best_x = res.x.copy()
            except Exception:
                pass
                
    # Fallback
    if best_x is None:
        best_x = generate_hex_init(0, 0.09)
        best_sum = -objective(best_x)
        
    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3]
    
    # Final strict validation and minimal repair
    valid = True
    for i in range(n):
        if radii[i] < 0: valid = False; break
        if centers[i,0] < radii[i] - 1e-9 or centers[i,0] > 1 - radii[i] + 1e-9: valid = False; break
        if centers[i,1] < radii[i] - 1e-9 or centers[i,1] > 1 - radii[i] + 1e-9: valid = False; break

        for i in range(n):
            for j in range(i+1, n):
                d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
                if d < radii[i] + radii[j] - 1e-9: valid = False; break
            if not valid: break

            radii *= factor
            for i in range(n):
                centers[i,0] = np.clip(centers[i,0], radii[i], 1-radii[i])

                if not is_ok: break
