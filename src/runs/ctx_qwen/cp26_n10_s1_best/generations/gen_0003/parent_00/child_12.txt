import numpy as np
from scipy.optimize import minimize, linprog

N_CIRCLES = 26
TRIL_IDX = np.tril_indices(N_CIRCLES, -1)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Compute all inequality constraints g(x) >= 0."""
    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]
    
    # Boundary constraints
    c = np.concatenate([
        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs
    ])
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    c = np.concatenate([c, dx[TRIL_IDX]**2 + dy[TRIL_IDX]**2 - dr[TRIL_IDX]**2])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [0, 0.5]."""
    return [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N_CIRCLES

def solve_lp_radii(centers):
    """Optimally scale radii for fixed centers using Linear Programming."""
    n = N_CIRCLES
    c_obj = -np.ones(n)
    
    num_constraints = n + n * (n - 1) // 2
    A_ub = np.zeros((num_constraints, n))
    b_ub = np.zeros(num_constraints)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        A_ub[idx, i] = 1.0
        b_ub[idx] = min(centers[i, 0], 1.0 - centers[i, 0], 
                        centers[i, 1], 1.0 - centers[i, 1])
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], 
                            centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dist
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.all(res.x >= 0):
            return res.x
    except Exception:
        pass
    return None

def generate_hex_configs(num_seeds=20):
    """Generates diverse hexagonal lattice initial configurations."""
    configs = []
    row_patterns = [
        [6, 5, 6, 5, 4],
        [5, 6, 5, 6, 4],
        [7, 6, 6, 7],
        [4, 5, 6, 5, 6],
        [5, 5, 5, 5, 6],
        [6, 6, 5, 5, 4]
    ]

        np.random.seed(seed)
        pattern = row_patterns[seed % len(row_patterns)]
        
        r0 = 0.085 + np.random.uniform(-0.005, 0.005)
        centers = np.zeros((N_CIRCLES, 2))
        idx = 0
        y = r0
        row = 0
        
        for count in pattern:
            x_start = r0 + (r0 if row % 2 == 1 else 0.0)
            for _ in range(count):
                if idx < N_CIRCLES:
                    x = x_start + np.random.uniform(-0.002, 0.002)
                    cy = y + np.random.uniform(-0.002, 0.002)
                    centers[idx] = [x, cy]
                    idx += 1
            y += np.sqrt(3) * r0
            row += 1
            
        # Rotate to break symmetry
        cx, cy = 0.5, 0.5
        centers -= [cx, cy]
        angle = np.random.uniform(-0.35, 0.35)
        c, s = np.cos(angle), np.sin(angle)
        centers = centers @ np.array([[c, -s], [s, c]])
        centers += [cx, cy]
        
        # Clip to valid range
        centers = np.clip(centers, 0.01, 0.99)
        configs.append(centers)
        
    return configs

def force_directed_feasibility(centers, steps=150):
    """Resolves overlaps and pushes to boundaries to ensure feasible start."""
    n = N_CIRCLES
    rs = np.full(n, 0.09)
    
    for _ in range(steps):
        forces = np.zeros_like(centers)
        for i in range(n):
            for j in range(i + 1, n):
                diff = centers[j] - centers[i]
                dist = np.hypot(diff[0], diff[1])
                min_d = rs[i] + rs[j]
                if dist < min_d and dist > 1e-6:
                    shift = (min_d - dist) * 0.5
                    forces[i] -= shift * diff / dist
                    forces[j] += shift * diff / dist
                    
        for i in range(n):
            if centers[i, 0] < rs[i]: forces[i, 0] += 0.01 * (rs[i] - centers[i, 0])
            elif centers[i, 0] > 1.0 - rs[i]: forces[i, 0] -= 0.01 * (centers[i, 0] - (1.0 - rs[i]))
            if centers[i, 1] < rs[i]: forces[i, 1] += 0.01 * (rs[i] - centers[i, 1])
            elif centers[i, 1] > 1.0 - rs[i]: forces[i, 1] -= 0.01 * (centers[i, 1] - (1.0 - rs[i]))
            
        centers += forces * 0.15
        centers = np.clip(centers, 0.005, 0.995)
        
    return centers

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
