import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26
TRIL_IDX = np.tril_indices(N_CIRCLES, -1)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Compute all inequality constraints g(x) >= 0."""
    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]
    
    n_cons = 4 * N_CIRCLES + N_CIRCLES * (N_CIRCLES - 1) // 2
    c = np.empty(n_cons)
    
    # Boundary constraints
    c[:N_CIRCLES] = xs - rs
    c[N_CIRCLES:2*N_CIRCLES] = 1.0 - xs - rs
    c[2*N_CIRCLES:3*N_CIRCLES] = ys - rs
    c[3*N_CIRCLES:4*N_CIRCLES] = 1.0 - ys - rs
    
    # Overlap constraints (squared form for smoothness)
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    c[4*N_CIRCLES:] = dx[TRIL_IDX]**2 + dy[TRIL_IDX]**2 - dr[TRIL_IDX]**2
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [0, 0.5]."""
    return [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N_CIRCLES

def make_hex_init(r0, angle=0.0, rng=None):
    """Generate a feasible initial configuration from a perturbed hexagonal lattice."""
    pts = []
    y = r0
    row = 0
    while y + r0 <= 1.0:
        x = r0 + (row % 2) * r0
        while x + r0 <= 1.0:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        
    pts = np.array(pts)
    if len(pts) < N_CIRCLES:
        missing = N_CIRCLES - len(pts)
        pts = np.vstack([pts, rng.uniform(0.1, 0.9, (missing, 2))])
        
    if rng is not None:
        rng.shuffle(pts)
    pts = pts[:N_CIRCLES]
    
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = pts - 0.5
        pts = pts @ np.array([[c, -s], [s, c]])
        pts = pts + 0.5
        
    pts = np.clip(pts, 0.02, 0.98)
    
    # Compute strictly feasible initial radii
    dists = np.sqrt(((pts[:, None, :] - pts[None, :, :])**2).sum(axis=2))
    np.fill_diagonal(dists, np.inf)
    min_d = dists.min(axis=1)
    bnd = np.minimum(np.minimum(pts[:,0], 1.0-pts[:,0]), np.minimum(pts[:,1], 1.0-pts[:,1]))
    rs = 0.45 * np.minimum(min_d, bnd)
    
    x0 = np.zeros(3 * N_CIRCLES)
    x0[0::3] = pts[:, 0]
    x0[1::3] = pts[:, 1]
    x0[2::3] = rs
    return x0

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -1.0
    best_x = None
    rng = np.random.default_rng(42)
    
    # Phase 1: Multiple diverse structured starts
    starts = []
    for r0 in [0.075, 0.08, 0.085, 0.09]:
        for ang in [0.0, 0.15, -0.15, 0.3, -0.3]:
            x0 = make_hex_init(r0, ang, rng)
            x0 += rng.normal(0, 0.003, x0.shape)
            starts.append(x0)
            
    for x0 in starts:
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 40000, 'ftol': 1e-14, 'disp': False})
            if not np.isnan(res.fun):
                val = -res.fun
                c_vals = constraints(res.x)
                if np.min(c_vals) >= -1e-5 and val > best_sum:
                    best_sum = val
                    best_x = res.x.copy()
        except Exception:
            continue
            
    # Phase 2: Iterative Refinement to escape local minima
    if best_x is not None:
        for step in range(15):
            x0 = best_x.copy()
            # Shrink radii to unlock tightly packed configurations
            x0[2::3] *= 0.97
            # Perturb positions to explore new arrangements
            x0[:2*N_CIRCLES] += rng.normal(0, 0.004, 2*N_CIRCLES)
            
            # Project variables back to feasible region
            rs = np.maximum(x0[2::3], 0.001)
            x0[2::3] = rs
            x0[0::3] = np.clip(x0[0::3], rs, 1.0 - rs)
            x0[1::3] = np.clip(x0[1::3], rs, 1.0 - rs)
            
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 40000, 'ftol': 1e-14, 'disp': False})
                if not np.isnan(res.fun):
                    val = -res.fun
                    c_vals = constraints(res.x)
                    if np.min(c_vals) >= -1e-5 and val > best_sum:
                        best_sum = val
                        best_x = res.x.copy()
            except Exception:
                pass
                
    # Extract results
    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3]
    
    # Final safety adjustment to guarantee strict compliance with validator tolerance
    for _ in range(100):
        valid = True
        for i in range(N_CIRCLES):
            if radii[i] < 0: valid = False; break
            if centers[i,0] < radii[i] - 1e-9 or centers[i,0] > 1 - radii[i] + 1e-9: valid = False; break
            if centers[i,1] < radii[i] - 1e-9 or centers[i,1] > 1 - radii[i] + 1e-9: valid = False; break
        if valid:
            for i in range(N_CIRCLES):
                for j in range(i + 1, N_CIRCLES):
                    d = np.hypot(centers[i,0] - centers[j,0], centers[i,1] - centers[j,1])
                    if d < radii[i] + radii[j] - 1e-9: valid = False; break
                if not valid: break
        if valid: break
        radii *= 0.9999
        
    return centers, radii, float(np.sum(radii))
