import numpy as np
from scipy.optimize import minimize, linprog

N_CIRCLES = 26
TRIL_IDX = np.tril_indices(N_CIRCLES, -1)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Compute all inequality constraints g(x) >= 0."""
    n = N_CIRCLES
    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]
    
    # Boundary constraints: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    c = np.concatenate([xs - rs, 1.0 - xs - rs, ys - rs, 1.0 - ys - rs])
    
    # Overlap constraints: dist^2 - (r_i + r_j)^2 >= 0
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    c = np.concatenate([c, dx[TRIL_IDX]**2 + dy[TRIL_IDX]**2 - dr[TRIL_IDX]**2])
    return c

def get_lp_radii(centers):
    """Solve LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    # Preallocate matrices
    n_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((n_ineq, n))
    b_ub = np.zeros(n_ineq)
    
    # Boundary: r_i <= min(x, 1-x, y, 1-y)
    bounds_arr = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                            np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    A_ub[:n] = np.eye(n)
    b_ub[:n] = bounds_arr
    
    # Overlap: r_i + r_j <= dist(i, j)
    idx = n
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = d
            idx += 1
            
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=[(0.0, None)] * n, method='highs')

            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

    y = r0 + shift[1]
    for i, cnt in enumerate(row_counts):
        x_start = r0 + (r0 if i % 2 == 1 else 0.0) + shift[0]
        for _ in range(cnt):
            centers.append([x_start, y])
            x_start += 2.0 * r0
        y += np.sqrt(3.0) * r0
        
    centers = np.array(centers[:N_CIRCLES])
    
    if angle_deg != 0.0:
        rad = np.deg2rad(angle_deg)
        cx, cy = 0.5, 0.5
        c, s = np.cos(rad), np.sin(rad)
        centers = (centers - [cx, cy]) @ np.array([[c, -s], [s, c]]) + [cx, cy]
        
    # Small random perturbation to break symmetry
    centers += np.random.uniform(-0.004, 0.004, centers.shape)
    centers = np.clip(centers, 0.01, 0.99)
    return centers

def force_init(seed):

    centers = np.random.uniform(0.1, 0.9, (N_CIRCLES, 2))
