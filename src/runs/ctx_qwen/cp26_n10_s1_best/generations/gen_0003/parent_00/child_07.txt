import numpy as np
from scipy.optimize import minimize, linprog

N_CIRCLES = 26
TRIL = np.tril_indices(N_CIRCLES, -1)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Compute all inequality constraints g(x) >= 0."""
    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]
    
    # Boundary constraints
    c = []
    c.append(xs - rs)
    c.append(1.0 - xs - rs)
    c.append(ys - rs)
    c.append(1.0 - ys - rs)
    
    # Overlap constraints (squared form for stability)
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    c.append(dx[TRIL]**2 + dy[TRIL]**2 - dr[TRIL]**2)
    
    return np.concatenate(c)

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [0, 0.5]."""
    return [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N_CIRCLES

def force_directed_init(seed):
    """Generate initial configuration using repulsive forces."""
    np.random.seed(seed)
    centers = np.random.uniform(0.15, 0.85, (N_CIRCLES, 2))
    for _ in range(1500):
        forces = np.zeros_like(centers)
        for i in range(N_CIRCLES):
            for j in range(i + 1, N_CIRCLES):
                diff = centers[j] - centers[i]
                d = np.hypot(diff[0], diff[1])
                if d < 0.35 and d > 1e-6:
                    f = 0.008 / (d**2 + 0.005)
                    forces[i] -= f * diff
                    forces[j] += f * diff
            for axis in range(2):
                if centers[i, axis] < 0.12: forces[i, axis] += 0.04
                elif centers[i, axis] > 0.88: forces[i, axis] -= 0.04
        centers += forces * 0.12
        centers = np.clip(centers, 0.02, 0.98)
    return centers

def hexagonal_init(seed):
    """Generate a perturbed hexagonal lattice."""
    np.random.seed(seed)
    centers = np.zeros((N_CIRCLES, 2))
    r = 0.095
    y = r
    row = 0
    idx = 0
    while idx < N_CIRCLES and y + r <= 1.0:
        x = r if row % 2 == 0 else 2.0 * r
        while idx < N_CIRCLES and x + r <= 1.0:
            centers[idx] = [x, y]
            idx += 1
            x += 2.0 * r
        y += np.sqrt(3.0) * r
        row += 1
    while idx < N_CIRCLES:
        centers[idx] = np.random.uniform(0.2, 0.8, 2)
        idx += 1
    centers += np.random.uniform(-0.008, 0.008, centers.shape)
    return np.clip(centers, 0.02, 0.98)

def solve_lp_radii(centers):
    """Optimally scale radii for fixed centers using Linear Programming."""
    n = N_CIRCLES
    c_obj = -np.ones(n)
    A_ub = np.zeros((n + n*(n-1)//2, n))
    b_ub = np.zeros(n + n*(n-1)//2)
    
    idx = 0
    for i in range(n):
        x, y = centers[i]
        lim = min(x, 1.0 - x, y, 1.0 - y)
        A_ub[idx, i] = 1.0
        b_ub[idx] = lim
        idx += 1
        
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i,0] - centers[j,0], centers[i,1] - centers[j,1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = d
            idx += 1
            
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, 
                      bounds=[(0.0, None)] * n, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def ensure_feasibility(x):
    """Project variables to strictly feasible bounds."""
    n = N_CIRCLES
    for i in range(n):
        r = max(0.001, x[3*i+2])
        x[3*i] = np.clip(x[3*i], r, 1.0 - r)
        x[3*i+1] = np.clip(x[3*i+1], r, 1.0 - r)
        x[3*i+2] = r
    return x

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    n = N_CIRCLES
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -1.0
    best_x = None
    
    # Generate diverse initial configurations
    inits = []
    for s in range(12):
        inits.append(force_directed_init(s))
    for s in range(12):
        inits.append(hexagonal_init(s))
        
    # Phase 1: SLSQP from diverse starts
    for c_init in inits:
        r_init = solve_lp_radii(c_init)
        if r_init is None:
            r_init = np.full(n, 0.06)
            
        x0 = np.zeros(3 * n)
        x0[0::3] = c_init[:, 0]
        x0[1::3] = c_init[:, 1]
        x0[2::3] = r_init
        x0 = ensure_feasibility(x0)
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 20000, 'ftol': 1e-13, 'disp': False})
            if not np.isnan(res.fun):
                cur = -res.fun
                c_val = constraints(res.x)
                if np.min(c_val) >= -1e-5 and cur > best_sum:
                    best_sum = cur
                    best_x = res.x.copy()
        except Exception:
            continue
            
    # Phase 2: Shrink & Perturb refinement to escape local minima
    if best_x is not None:
        for step in range(25):
            noise_scale = 0.006 * (0.78 ** step)
            x0 = best_x.copy()
            x0[2::3] *= 0.985  # Shrink radii slightly
            x0[0::3] += np.random.normal(0, noise_scale, n)
            x0[1::3] += np.random.normal(0, noise_scale, n)
            x0 = ensure_feasibility(x0)
            
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 15000, 'ftol': 1e-13, 'disp': False})
                if not np.isnan(res.fun):
                    cur = -res.fun
                    c_val = constraints(res.x)
                    if np.min(c_val) >= -1e-5 and cur > best_sum:
                        best_sum = cur
                        best_x = res.x.copy()
            except Exception:
                pass
                
    # Fallback
    if best_x is None:
        best_x = np.zeros(3 * n)
        best_x[0::3] = np.tile(np.linspace(0.1, 0.9, 5), 6)[:n]
        best_x[1::3] = np.repeat(np.linspace(0.1, 0.9, 6), 5)[:n]
        best_x[2::3] = 0.05
        best_sum = -objective(best_x)

    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3]

    for _ in range(80):
        valid = True
        for i in range(n):
            if radii[i] < 0: valid = False; break
            if centers[i,0] < radii[i] - 1e-9 or centers[i,0] > 1 - radii[i] + 1e-9: valid = False; break
            if centers[i,1] < radii[i] - 1e-9 or centers[i,1] > 1 - radii[i] + 1e-9: valid = False; break
        if valid:
            for i in range(n):
                for j in range(i + 1, n):
                    d = np.hypot(centers[i,0] - centers[j,0], centers[i,1] - centers[j,1])
                    if d < radii[i] + radii[j] - 1e-9:
                        valid = False
                        break
                if not valid: break
        if valid: break
        radii *= 0.9995
        
    return centers, radii, float(np.sum(radii))
