N_CIRCLES = 26
TRIL_IDX = np.tril_indices(N_CIRCLES, -1)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def get_constraints_sq(x):
    """Squared distance constraints (computationally efficient)."""
    n = N_CIRCLES
    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]

        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs
    ])

    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    c = np.concatenate([c, dx[TRIL_IDX]**2 + dy[TRIL_IDX]**2 - dr[TRIL_IDX]**2])
    return c

def get_constraints_dist(x):

    ys = x[1::3]
    rs = x[2::3]

        xs - rs,

        1.0 - ys - rs
    ])
    
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dist = np.sqrt(dx**2 + dy**2)
    dr = rs[:, None] + rs[None, :]
    
    c = np.concatenate([c, dist[TRIL_IDX] - dr[TRIL_IDX]])
    return c

def get_bounds():

def lp_optimize_radii(centers):

    n = N_CIRCLES
    A_ub = np.zeros((n + n * (n - 1) // 2, n))
    b_ub = np.zeros(n + n * (n - 1) // 2)
    idx = 0

        x_i, y_i = centers[i]
        A_ub[idx, i] = 1.0
        b_ub[idx] = min(x_i, 1.0 - x_i, y_i, 1.0 - y_i)
        idx += 1

        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])

            A_ub[idx, j] = 1.0
            b_ub[idx] = dist
            idx += 1

        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')

    except Exception:
        pass
    return None
