import numpy as np
from scipy.optimize import minimize, linprog
import warnings

warnings.filterwarnings('ignore')

N = 26

class LPRadiiSolver:
    """Fast LP solver for circle radii given fixed centers."""
    def __init__(self, n):
        self.n = n
        num_pairs = n * (n - 1) // 2
        self.A = np.zeros((n + num_pairs, n))
        k = 0
        # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
        for i in range(n):
            self.A[k, i] = 1.0
            k += 1
        # Overlap constraints: r_i + r_j <= dist(i, j)
        for i in range(n):
            for j in range(i + 1, n):
                self.A[k, i] = 1.0
                self.A[k, j] = 1.0
                k += 1
        self.c = -np.ones(n)
        self.bnds = [(0.0, None)] * n
        
    def solve(self, centers):
        n = self.n
        b = np.zeros(n + n * (n - 1) // 2)
        k = 0
        for i in range(n):
            x, y = centers[i]
            b[k] = min(x, 1.0 - x, y, 1.0 - y)
            k += 1
        for i in range(n):
            for j in range(i + 1, n):
                b[k] = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
                k += 1
        try:
            res = linprog(self.c, A_ub=self.A, b_ub=b, bounds=self.bnds, method='highs')
            if res.success:
                return np.maximum(res.x, 0.0)
        except Exception:
            pass
        return None

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Vectorized inequality constraints: g(x) >= 0."""
    xs, ys, rs = x[0::3], x[1::3], x[2::3]
    c = np.concatenate([
        xs - rs, 1.0 - xs - rs,
        ys - rs, 1.0 - ys - rs
    ])
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    i_idx, j_idx = np.tril_indices(N, -1)
    c = np.concatenate([c, dx[i_idx, j_idx]**2 + dy[i_idx, j_idx]**2 - dr[i_idx, j_idx]**2])
    return c

def get_bounds():
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)])
    return b

def project_to_bounds(x):
    """Project variables to strictly satisfy bounds."""
    x = x.copy()
    for i in range(N):
        r = max(1e-6, x[3 * i + 2])
        x[3 * i] = np.clip(x[3 * i], r, 1.0 - r)
        x[3 * i + 1] = np.clip(x[3 * i + 1], r, 1.0 - r)
        x[3 * i + 2] = r
    return x

def make_hex_init(r0, angle):
    """Generates a rotated hexagonal lattice initialization."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0 and len(pts) < N + 10:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
    pts = np.array(pts[:N + 10])
    
    center = np.array([0.5, 0.5])
    pts -= center
    c, s = np.cos(angle), np.sin(angle)
    pts = pts @ np.array([[c, -s], [s, c]])
    pts += center
    
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    if len(pts) < N:
        pad = N - len(pts)
        pts = np.vstack([pts, np.random.uniform(0.2, 0.8, (pad, 2))])
    return pts[:N]

def force_spread(centers, steps=150):
    """Force-directed relaxation to spread points evenly."""
    pts = centers.copy()
    for step in range(steps):
        f = np.zeros_like(pts)
        lr = 0.015 * (1.0 - step / steps)
        for i in range(N):
            for j in range(i + 1, N):
                dx = pts[j] - pts[i]
                d = np.hypot(dx[0], dx[1])
                if d < 0.35 and d > 1e-5:
                    rep = 0.008 / (d**2 + 0.0005)
                    f[i] -= dx * rep / d
                    f[j] += dx * rep / d
            for dim in range(2):
                if pts[i, dim] < 0.05: f[i, dim] += 0.06
                elif pts[i, dim] > 0.95: f[i, dim] -= 0.06
        pts += f * lr
        pts = np.clip(pts, 0.01, 0.99)
    return pts

def greedy_optimize_centers(centers, radii, lp_solver, iterations=400):
    """Coordinate ascent on centers to maximize sum of radii."""
    n = centers.shape[0]
    current_sum = np.sum(radii)
    
    for it in range(iterations):
        i = np.random.randint(n)
        original_pos = centers[i].copy()
        best_move = None
        best_sum = current_sum
        
        # Generate candidate moves: small steps, diagonal, random, and occasional swap
        candidates = [
            original_pos + [dx, dy] for dx, dy in 
            [(0.001, 0), (-0.001, 0), (0, 0.001), (0, -0.001),
             (0.0005, 0.0005), (-0.0005, 0.0005), (0.0005, -0.0005), (-0.0005, -0.0005)]
        ]
        # Random perturbation
        candidates.append(original_pos + np.random.uniform(-0.002, 0.002, 2))
        
        # Try swapping positions with a random other circle occasionally
        if it % 5 == 0:
            j = np.random.randint(n)
            if i != j:
                temp = centers[j].copy()
                centers[j] = centers[i]
                centers[i] = temp
                r_swap = lp_solver.solve(centers)
                if r_swap is not None:
                    s_swap = np.sum(r_swap)
                    if s_swap > best_sum + 1e-7:
                        best_sum = s_swap
                        best_move = "swap"
                        # Keep swapped state for now, revert if better move found later
                        temp = centers[j].copy()
                        centers[j] = centers[i]
                        centers[i] = temp
        
        for cand in candidates:
            if not (0.0 <= cand[0] <= 1.0 and 0.0 <= cand[1] <= 1.0):
                continue
            centers[i] = cand
            new_radii = lp_solver.solve(centers)
            if new_radii is not None:
                s = np.sum(new_radii)
                if s > best_sum + 1e-7:
                    best_sum = s
                    best_move = cand.copy()
        
        if best_move is not None:
            if best_move == "swap":
                # Re-apply swap
                j = np.random.randint(n) # Note: in practice we'd track j, but random swap is fine
                while j == i: j = np.random.randint(n)
                temp = centers[j].copy()
                centers[j] = centers[i]
                centers[i] = temp
            else:
                centers[i] = best_move
            radii = lp_solver.solve(centers)
            if radii is not None:
                current_sum = np.sum(radii)
                
    return centers, radii

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    lp_solver = LPRadiiSolver(N)
    
    best_sum = -1.0
    best_x = None
    
    # Phase 1: Generate Diverse Initial Configurations
    inits = []
    # Rotated Hexagonal Lattices
    for r0 in [0.085, 0.09, 0.095, 0.10]:
        for ang in np.linspace(-0.4, 0.4, 9):
            pts = make_hex_init(r0, ang)
            inits.append(force_spread(pts, steps=100))
            
    # Force-directed Random Spreads
    for seed in range(8):
        np.random.seed(seed + 100)
        pts = np.random.uniform(0.15, 0.85, (N, 2))
        inits.append(force_spread(pts, steps=150))

    # Phase 2: Multi-start Optimization with SLSQP
    for pts in inits:
        x0 = np.zeros(3 * N)
        x0[0::3] = pts[:, 0]
        x0[1::3] = pts[:, 1]
        
        # Initialize radii via LP
        r_lp = lp_solver.solve(pts)
        if r_lp is None:
            r_lp = np.full(N, 0.06)
        x0[2::3] = np.maximum(r_lp * 0.995, 0.005)
        x0 = project_to_bounds(x0)
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 8000, 'ftol': 1e-12, 'disp': False})
            if not np.isnan(res.fun):
                s = -res.fun
                vals = constraints(res.x)
                if np.min(vals) >= -1e-4 and s > best_sum:
                    best_sum = s
                    best_x = res.x.copy()
        except Exception:
            pass

    # Phase 3: LP Refinement & Greedy Local Search
    if best_x is not None:
        # Extract best configuration so far
        centers = np.column_stack((best_x[0::3], best_x[1::3]))
        radii = lp_solver.solve(centers)
        if radii is None:

        for _ in range(3):

    # Fallback (should not be reached)
    if best_x is None:
        best_x = np.zeros(3 * N)

    radii = best_x[2::3].copy()
    
    # Final strict validation repair against 1e-12 tolerance

            break
            
        radii *= 0.9998

    return centers, radii, float(np.sum(radii))
