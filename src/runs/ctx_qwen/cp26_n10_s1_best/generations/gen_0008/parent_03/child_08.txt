import numpy as np
from scipy.optimize import minimize, linprog

N = 26
TRIL_IDX = np.tril_indices(N, -1)
NUM_OVERLAP = N * (N - 1) // 2
NUM_CONSTRAINTS = N + NUM_OVERLAP

# Precompute constant parts of LP constraint matrix
A_ub = np.zeros((NUM_CONSTRAINTS, N))
for i in range(N):
    A_ub[i, i] = 1.0
idx = N
for i in range(N):
    for j in range(i + 1, N):
        A_ub[idx, i] = 1.0
        A_ub[idx, j] = 1.0
        idx += 1

def compute_b_ub(centers):
    """Compute the right-hand side of LP constraints for fixed centers."""
    b = np.zeros(NUM_CONSTRAINTS)
    xs = centers[:, 0]
    ys = centers[:, 1]
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    b[:N] = np.maximum(0.0, np.minimum(xs, np.minimum(1.0 - xs, np.minimum(ys, 1.0 - ys))))
    
    # Overlap constraints: r_i + r_j <= dist(i, j)
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dists = np.hypot(dx, dy)
    b[N:] = dists[TRIL_IDX]
    return b

def solve_lp_radii(centers):
    """Solve for optimal radii given fixed centers using Linear Programming."""
    b_ub = compute_b_ub(centers)
    res = linprog(-np.ones(N), A_ub=A_ub, b_ub=b_ub, bounds=[(0.0, None)] * N, method='highs')
    if res.success:
        return res.x
    return np.full(N, 0.05)

def evaluate_centers(centers):
    """Returns the maximum sum of radii for a given center configuration."""
    radii = solve_lp_radii(centers)
    return np.sum(radii), radii

def make_hex_init(r0, angle):
    """Generates a rotated hexagonal lattice initialization."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0 and len(pts) < N + 10:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
    pts = np.array(pts[:N + 10])
    center = np.array([0.5, 0.5])
    pts -= center
    c, s = np.cos(angle), np.sin(angle)
    pts = pts @ np.array([[c, -s], [s, c]])
    pts += center
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    if len(pts) < N:
        pad = N - len(pts)
        pts = np.vstack([pts, np.random.uniform(0.2, 0.8, (pad, 2))])
    return pts[:N]

def force_spread(centers, steps=150):
    """Force-directed relaxation to spread points evenly and push to boundaries."""
    pts = centers.copy()
    for step in range(steps):
        f = np.zeros_like(pts)
        lr = 0.02 * (1.0 - step / steps)
        for i in range(N):
            for j in range(i + 1, N):
                dx = pts[j] - pts[i]
                d = np.hypot(dx[0], dx[1])
                if d < 0.3 and d > 1e-5:
                    rep = 0.005 / (d**2 + 0.001)
                    f[i] -= dx * rep / d
                    f[j] += dx * rep / d
            for dim in range(2):
                if pts[i, dim] < 0.05: f[i, dim] += 0.05
                elif pts[i, dim] > 0.95: f[i, dim] -= 0.05
        pts += f * lr
        pts = np.clip(pts, 0.01, 0.99)
    return pts

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Returns all inequality constraints g(x) >= 0 (vectorized)."""
    xs, ys, rs = x[0::3], x[1::3], x[2::3]
    c = np.concatenate([xs - rs, 1.0 - xs - rs, ys - rs, 1.0 - ys - rs])
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    c = np.concatenate([c, dx[TRIL_IDX]**2 + dy[TRIL_IDX]**2 - dr[TRIL_IDX]**2])
    return c

def project_to_feasible(x):
    """Project variables to strictly satisfy bounds."""
    for i in range(N):
        r = max(1e-6, x[3 * i + 2])
        x[3 * i] = np.clip(x[3 * i], r, 1.0 - r)
        x[3 * i + 1] = np.clip(x[3 * i + 1], r, 1.0 - r)
        x[3 * i + 2] = r
    return x

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    
    T = 0.1
    noise_scale = 0.04
    best_centers = None
    best_score = -1.0
    
    # Prepare diverse starting configurations
    starts = [
        make_hex_init(0.09, 0.0),
        make_hex_init(0.09, 0.2),
        make_hex_init(0.09, -0.2),
        force_spread(np.random.uniform(0.15, 0.85, (N, 2)))
    ]
        
    for init_centers in starts:
        current_centers = init_centers.copy()
        current_score, _ = evaluate_centers(current_centers)
        
        if current_score > best_score:
            best_centers = current_centers.copy()
            best_score = current_score
            
        for step in range(3000):
            new_centers = current_centers + np.random.normal(0, noise_scale, (N, 2))
            new_centers = np.clip(new_centers, 0.005, 0.995)
            
            score, _ = evaluate_centers(new_centers)
            
            delta = score - current_score
            if delta > 0 or np.random.rand() < np.exp(delta / max(T, 1e-6)):
                current_centers = new_centers
                current_score = score
                if score > best_score:
                    best_centers = current_centers.copy()
                    best_score = score
                    
            noise_scale *= 0.999
            T *= 0.999
            
    # SLSQP Polish: Jointly optimize centers and radii for final precision
    r_lp = solve_lp_radii(best_centers)
    x0 = np.zeros(3 * N)
    x0[0::3] = best_centers[:, 0]
    x0[1::3] = best_centers[:, 1]
    x0[2::3] = r_lp * 0.999
    x0 = project_to_feasible(x0)
    
    bounds = [(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    try:
        res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 15000, 'ftol': 1e-14, 'disp': False})
        if not np.isnan(res.fun) and -res.fun > best_score:
            best_x = res.x.copy()
            best_score = -res.fun
        else:
            best_x = x0
    except Exception:
        best_x = x0
        
    # Second polish round with slight perturbation to escape remaining local minima
    for _ in range(5):
        x_pert = best_x + np.random.normal(0, 1e-4, 3 * N)
        x_pert[2::3] *= 0.995
        x_pert = project_to_feasible(x_pert)
        try:
            res = minimize(objective, x_pert, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 10000, 'ftol': 1e-14, 'disp': False})
            if not np.isnan(res.fun) and -res.fun > best_score:
                best_x = res.x.copy()
                best_score = -res.fun
        except Exception:
            pass
            
    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3]
    
    # Final strict validation repair against 1e-12 tolerance
    for _ in range(100):
        valid = True
        for i in range(N):
            if radii[i] < 0 or centers[i, 0] < radii[i] - 1e-9 or centers[i, 0] > 1.0 - radii[i] + 1e-9 or \
               centers[i, 1] < radii[i] - 1e-9 or centers[i, 1] > 1.0 - radii[i] + 1e-9:
                valid = False
                break
        if valid:
            for i in range(N):
                for j in range(i + 1, N):
                    d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                    if d < radii[i] + radii[j] - 1e-9:
                        valid = False
                        break
                if not valid:
                    break
        if valid:
            break
            
        radii *= 0.9998
        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)
        
    return centers, radii, float(np.sum(radii))
