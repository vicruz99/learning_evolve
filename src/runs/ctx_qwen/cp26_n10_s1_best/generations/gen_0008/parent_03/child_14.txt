import numpy as np
from scipy.optimize import minimize, linprog
import warnings

warnings.filterwarnings('ignore')

N = 26

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist)
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return np.full(n, 0.01)

def get_sum_radii(centers):
    """Compute sum of optimal radii for given centers."""
    return np.sum(solve_lp_radii(centers))

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    best_centers = None
    best_sum = -1.0
    
    # Phase 1: Generate diverse initial configurations
    inits = []
    
    # Rotated Hexagonal Lattices
    for r0 in [0.082, 0.090, 0.098]:
        for ang in np.linspace(0.0, np.pi, 9):
            pts = []
            y = r0
            row = 0
            while len(pts) < N + 8:
                x = r0 if row % 2 == 0 else 2.0 * r0
                while x <= 1.0 - r0 and len(pts) < N + 8:
                    pts.append([x, y])
                    x += 2.0 * r0
                y += np.sqrt(3.0) * r0
                row += 1
                
            pts = np.array(pts[:N + 8])
            center = np.array([0.5, 0.5])
            pts -= center
            c, s = np.cos(ang), np.sin(ang)
            pts = pts @ np.array([[c, -s], [s, c]])
            pts += center
            
            # Filter valid and pad if necessary
            mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
            pts = pts[mask]
            if len(pts) < N:
                pad = N - len(pts)
                pts = np.vstack([pts, np.random.uniform(0.15, 0.85, (pad, 2))])
            inits.append(pts[:N])
            
    # Random spreads
    for _ in range(8):
        pts = np.random.uniform(0.12, 0.88, (N, 2))
        inits.append(pts)
        
    # Phase 2: Gauss-Seidel Optimization on Centers
    for init_centers in inits:
        centers = init_centers.copy()
        current_sum = get_sum_radii(centers)
        
        epochs = 0
        max_epochs = 7
        improved = True
        
        while improved and epochs < max_epochs:
            improved = False
            epochs += 1
            order = np.random.permutation(N)
            
            # Occasional global perturbation to escape local minima
            if epochs == 4:
                num_perturb = N // 4
                idx_pert = np.random.choice(N, num_perturb, replace=False)
                centers[idx_pert] += np.random.normal(0, 0.03, (num_perturb, 2))
                centers = np.clip(centers, 0.05, 0.95)
                current_sum = get_sum_radii(centers)
                
            for i in order:
                # 2D optimization for circle i
                def obj_2d(pos):
                    new_c = centers.copy()
                    pos_clipped = np.clip(pos, 0.005, 0.995)
                    new_c[i] = pos_clipped
                    try:
                        val = get_sum_radii(new_c)
                        return -val
                    except Exception:
                        return 1e6
                        
                res = minimize(obj_2d, centers[i], method='Nelder-Mead',
                               options={'xatol': 1e-5, 'fatol': 1e-8, 'maxiter': 50})
                               
                new_sum = -res.fun
                if new_sum > current_sum + 1e-6:
                    centers[i] = res.x
                    current_sum = new_sum
                    improved = True
                    
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers.copy()
            
    # Phase 3: Final SLSQP Polish
    if best_centers is not None:
        radii_init = solve_lp_radii(best_centers)
        x0 = np.zeros(3 * N)
        x0[0::3] = best_centers[:, 0]
        x0[1::3] = best_centers[:, 1]
        x0[2::3] = radii_init
        
        def obj_sl(x):
            return -np.sum(x[2::3])
            
        def cons_sl(x):
            xs, ys, rs = x[0::3], x[1::3], x[2::3]
            c = np.concatenate([
                xs - rs,
                1.0 - xs - rs,
                ys - rs,
                1.0 - ys - rs
            ])
            dx = xs[:, None] - xs[None, :]
            dy = ys[:, None] - ys[None, :]
            dr = rs[:, None] + rs[None, :]
            i_idx, j_idx = np.tril_indices(N, -1)
            c = np.concatenate([c, dx[i_idx, j_idx]**2 + dy[i_idx, j_idx]**2 - dr[i_idx, j_idx]**2])
            return c
            
        bounds = [(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)] * N
        cons = {'type': 'ineq', 'fun': cons_sl}
        
        try:
            res = minimize(obj_sl, x0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 15000, 'ftol': 1e-12, 'disp': False})
            if not np.isnan(res.fun):
                final_sum = -res.fun
                if final_sum > best_sum:
                    best_sum = final_sum
                    best_centers = np.column_stack((res.x[0::3], res.x[1::3]))
                    radii = res.x[2::3]
                else:
                    radii = solve_lp_radii(best_centers)
            else:
                radii = solve_lp_radii(best_centers)
        except Exception:
            radii = solve_lp_radii(best_centers)
    else:
        best_centers = np.random.uniform(0.1, 0.9, (N, 2))
        radii = solve_lp_radii(best_centers)
        best_sum = np.sum(radii)
        
    # Phase 4: Strict Validation Repair
    for _ in range(100):
        valid = True
        for i in range(N):
            if radii[i] < 0 or best_centers[i, 0] < radii[i] - 1e-9 or best_centers[i, 0] > 1.0 - radii[i] + 1e-9 or \
               best_centers[i, 1] < radii[i] - 1e-9 or best_centers[i, 1] > 1.0 - radii[i] + 1e-9:
                valid = False
                break
        if valid:
            for i in range(N):
                for j in range(i + 1, N):
                    d = np.hypot(best_centers[i, 0] - best_centers[j, 0], best_centers[i, 1] - best_centers[j, 1])
                    if d < radii[i] + radii[j] - 1e-9:
                        valid = False
                        break
                if not valid:
                    break
        if valid:
            break
            
        radii *= 0.9995
        best_centers[:, 0] = np.clip(best_centers[:, 0], radii, 1.0 - radii)
        best_centers[:, 1] = np.clip(best_centers[:, 1], radii, 1.0 - radii)
        
    return best_centers, radii, float(np.sum(radii))
