import numpy as np
from scipy.optimize import minimize, linprog, NonlinearConstraint
import math

N = 26

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def objective_jac(x):
    """Gradient of objective."""
    grad = np.zeros(3 * N)
    grad[2::3] = -1.0
    return grad

def constraint_func(x):
    """Returns all inequality constraints g(x) >= 0."""
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    # Boundary constraints: 4*N
    c = np.concatenate([
        cx - r,
        1.0 - cx - r,
        cy - r,
        1.0 - cy - r
    ])
    
    # Overlap constraints: N*(N-1)/2
    i, j = np.triu_indices(N, 1)
    dx = cx[i] - cx[j]
    dy = cy[i] - cy[j]
    dr = r[i] + r[j]
    
    c = np.concatenate([c, dx**2 + dy**2 - dr**2])
    return c

def constraint_jac(x):
    """Jacobian of constraints."""
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    n_cons = 4 * N + N * (N - 1) // 2
    n_vars = 3 * N
    jac = np.zeros((n_cons, n_vars))
    
    # Boundary constraints
    for i in range(N):
        # cx[i] - r[i] >= 0
        idx = i
        jac[idx, 3*i] = 1.0
        jac[idx, 3*i+2] = -1.0
        
        # 1 - cx[i] - r[i] >= 0
        idx = N + i
        jac[idx, 3*i] = -1.0
        jac[idx, 3*i+2] = -1.0
        
        # cy[i] - r[i] >= 0
        idx = 2*N + i
        jac[idx, 3*i+1] = 1.0
        jac[idx, 3*i+2] = -1.0
        
        # 1 - cy[i] - r[i] >= 0
        idx = 3*N + i
        jac[idx, 3*i+1] = -1.0
        jac[idx, 3*i+2] = -1.0
    
    # Overlap constraints
    i_idx, j_idx = np.triu_indices(N, 1)
    base = 4 * N
    for k in range(len(i_idx)):
        i = i_idx[k]
        j = j_idx[k]
        row = base + k
        
        dx = cx[i] - cx[j]
        dy = cy[i] - cy[j]
        dr = r[i] + r[j]
        
        jac[row, 3*i] = 2.0 * dx
        jac[row, 3*j] = -2.0 * dx
        jac[row, 3*i+1] = 2.0 * dy
        jac[row, 3*j+1] = -2.0 * dy
        jac[row, 3*i+2] = -2.0 * dr
        jac[row, 3*j+2] = -2.0 * dr
    
    return jac

def get_bounds():
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)])
    return b

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A = np.zeros((num_ineq, n))
    b = np.zeros(num_ineq)
    
    idx = 0
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A[idx, i] = 1.0
        b[idx] = max(0.0, lim)
        idx += 1
        
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = max(0.0, d)
            idx += 1
            
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0.0, None)] * n, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-8)
    except Exception:
        pass
    return None

def project_to_bounds(x):
    x = x.copy()
    r = np.maximum(x[2::3], 1e-6)
    x[0::3] = np.clip(x[0::3], r, 1.0 - r)
    x[1::3] = np.clip(x[1::3], r, 1.0 - r)
    x[2::3] = r
    return x

def make_hex_init(r0, angle):
    """Generates a hexagonal lattice initialization with optional rotation."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        if y > 1.0 + r0:
            break
            
    pts = np.array(pts[:N + 10])
    
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
        
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.1, 0.9, (1, 2))])
    return pts[:N]

def make_staggered_init(r0, angle):
    """Creates a staggered grid that might pack better for non-divisible counts."""
    pts = []
    n_cols = 6
    spacing_x = 1.0 / (n_cols + 0.5)
    
    y = r0
    row = 0
    while len(pts) < N + 5:
        if row % 2 == 0:
            x_start = r0
            n_in_row = n_cols
        else:
            x_start = r0 + spacing_x / 2
            n_in_row = n_cols - 1
        
        for col in range(n_in_row):
            if len(pts) >= N + 5:
                break
            x = x_start + col * spacing_x
            if x + r0 <= 1.0:
                pts.append([x, y])
        
        y += r0 * 1.7
        row += 1
        if y > 1.0 + r0:
            break
            
    pts = np.array(pts[:N + 5])
    
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
    
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.15, 0.85, (1, 2))])
    return pts[:N]

def make_force_init(seed, steps=600):
    """Force-directed layout to spread points evenly."""
    np.random.seed(seed)
    pts = np.random.uniform(0.15, 0.85, (N, 2))
    radii = np.full(N, 0.05)
    
    for step in range(steps):
        f = np.zeros_like(pts)
        for i in range(N):
            for j in range(i + 1, N):
                dx = pts[j] - pts[i]
                d = np.hypot(dx[0], dx[1])
                if d < 0.28 and d > 1e-5:
                    rep = 0.015 / (d**2 + 0.001)
                    f[i] -= rep * dx
                    f[j] += rep * dx
            for dim in range(2):
                if pts[i, dim] < radii[i] + 0.08:
                    f[i, dim] += 0.05
                elif pts[i, dim] > 1.0 - radii[i] - 0.08:
                    f[i, dim] -= 0.05
        lr = 0.04 * (1.0 - step / steps) ** 0.5
        pts += f * lr
        pts = np.clip(pts, 0.03, 0.97)
    return pts

def try_optimize(x0, method='SLSQP', maxiter=15000):
    """Try to optimize from x0, return best result."""
    bounds = get_bounds()
    
    try:
        if method == 'SLSQP':
            cons = {'type': 'ineq', 'fun': constraint_func}
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-13, 'disp': False})
        elif method == 'trust-constr':
            cons = NonlinearConstraint(constraint_func, 0, np.inf)
            res = minimize(objective, x0, method='trust-constr', bounds=bounds, constraints=cons,
                          jac=objective_jac, options={'maxiter': maxiter, 'verbose': 0, 'gtol': 1e-10})
        else:
            return None
            
        if not np.isnan(res.fun) and not np.any(np.isnan(res.x)):
            return res
    except Exception:
        pass
    return None
