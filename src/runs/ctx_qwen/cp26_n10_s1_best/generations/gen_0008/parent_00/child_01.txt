import numpy as np
from scipy.optimize import minimize, linprog

N = 26
TRIU_IDX = np.triu_indices(N, 1)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Returns all inequality constraints >= 0."""
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    # Boundary constraints
    c = np.concatenate([
        cx - r,
        1.0 - cx - r,
        cy - r,
        1.0 - cy - r
    ])
    
    # Overlap constraints: dist >= ri + rj
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dr = r[:, None] + r[None, :]
    
    dists = np.sqrt(dx[TRIU_IDX[0], TRIU_IDX[1]]**2 + dy[TRIU_IDX[0], TRIU_IDX[1]]**2)
    c = np.concatenate([c, dists - dr[TRIU_IDX[0], TRIU_IDX[1]]])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-8, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-8, 0.5)])
    return b

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A = np.zeros((num_ineq, n))
    b = np.zeros(num_ineq)
    
    idx = 0
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A[idx, i] = 1.0
        b[idx] = max(1e-9, lim)
        idx += 1
        
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = max(1e-9, d)
            idx += 1
            
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0.0, None)] * n, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-8)
    except Exception:
        pass
    return None

def project(x):
    """Project optimization vector to respect bounds."""
    x = x.copy()
    r = np.maximum(x[2::3], 1e-8)
    x[0::3] = np.clip(x[0::3], r, 1.0 - r)
    x[1::3] = np.clip(x[1::3], r, 1.0 - r)
    x[2::3] = r
    return x

def make_hex_init(r0, angle, scale=1.0):
    """Generates a hexagonal lattice initialization."""
    pts = []
    y = r0 * scale
    row = 0
    while len(pts) < N + 10:
        x = r0 * scale if row % 2 == 0 else 2.0 * r0 * scale
        while x <= 1.0 - r0 * scale:
            pts.append([x, y])
            x += 2.0 * r0 * scale
        y += np.sqrt(3.0) * r0 * scale
        row += 1
        if y > 1.0 + r0 * scale:
            break
            
    pts = np.array(pts[:N + 10])
    
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
        
    mask = (pts[:, 0] >= 0.01) & (pts[:, 0] <= 0.99) & (pts[:, 1] >= 0.01) & (pts[:, 1] <= 0.99)
    pts = pts[mask]
    
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.1, 0.9, (1, 2))])
    return pts[:N]

def force_layout(centers, radii, steps=200):
    """Force-directed relaxation."""
    pts = centers.copy()
    for step in range(steps):
        forces = np.zeros_like(pts)
        for i in range(N):
            for j in range(i + 1, N):
                dx = pts[j, 0] - pts[i, 0]
                dy = pts[j, 1] - pts[i, 1]
                d = np.hypot(dx, dy)
                if d < 0.3 and d > 1e-6:
                    target = radii[i] + radii[j] + 0.01
                    if d < target:
                        f = 0.1 / (d + 0.01)
                        forces[i] -= f * np.array([dx, dy])
                        forces[j] += f * np.array([dx, dy])
            for dim in range(2):
                if pts[i, dim] < radii[i] + 0.05: forces[i, dim] += 0.05
                elif pts[i, dim] > 1.0 - radii[i] - 0.05: forces[i, dim] -= 0.05
        lr = 0.05 * (1.0 - step / steps)
        pts += forces * lr
        pts = np.clip(pts, 0.001, 0.999)
    return pts

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -1.0
    best_x = None
    
    inits = []
    # Diverse hex grids
    for r0 in np.linspace(0.07, 0.11, 6):
        for ang in np.linspace(-0.6, 0.6, 11):
            inits.append(make_hex_init(r0, ang))
            
    # Force-directed layouts
    for s in range(15):
        np.random.seed(s)
        pts = np.random.uniform(0.15, 0.85, (N, 2))
        pts = force_layout(pts, np.full(N, 0.08), steps=300)
        inits.append(pts)
        
    for c_init in inits:
        r_lp = solve_lp_radii(c_init)
        if r_lp is None:
            r_lp = np.full(N, 0.06)
            
        x0 = np.zeros(3 * N)
        x0[0::3] = c_init[:, 0]
        x0[1::3] = c_init[:, 1]
        x0[2::3] = r_lp
        x0 = project(x0)
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 25000, 'ftol': 1e-13, 'disp': False})
            if not np.isnan(res.fun):
                c_opt = np.column_stack((res.x[0::3], res.x[1::3]))
                r_lp = solve_lp_radii(c_opt)
                if r_lp is not None:
                    curr_sum = np.sum(r_lp)
                    if curr_sum > best_sum:
                        best_sum = curr_sum
                        best_x = np.zeros(3 * N)
                        best_x[0::3] = c_opt[:, 0]
                        best_x[1::3] = c_opt[:, 1]
                        best_x[2::3] = r_lp
        except Exception:
            continue
            
    # Refinement loop with subset deflation and cooling noise
    if best_x is not None:
        for step in range(60):
            x_p = best_x.copy()
            # Shrink radii progressively
            shrink = 0.85 + 0.14 * (step / 60.0)
            x_p[2::3] *= shrink
            
            # Perturb centers
            noise = 0.004 / (1.0 + step * 0.05)
            x_p[0::3] += np.random.normal(0, noise, N)
            x_p[1::3] += np.random.normal(0, noise, N)
            
            # Subset deflation: shrink random subset more to break symmetry
            if step % 3 == 0:
                subset = np.random.choice(N, size=N // 4, replace=False)
                x_p[2::3][subset] *= 0.7
                
            x_p = project(x_p)
            
            try:
                res = minimize(objective, x_p, method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 18000, 'ftol': 1e-13, 'disp': False})
                if not np.isnan(res.fun):
                    c_opt = np.column_stack((res.x[0::3], res.x[1::3]))
                    r_lp = solve_lp_radii(c_opt)
                    if r_lp is not None:
                        curr_sum = np.sum(r_lp)
                        if curr_sum > best_sum:
                            best_sum = curr_sum
                            best_x = np.zeros(3 * N)
                            best_x[0::3] = c_opt[:, 0]
                            best_x[1::3] = c_opt[:, 1]
                            best_x[2::3] = r_lp
            except Exception:
                pass
                
        # Final high-precision polish
        for _ in range(15):
            x_p = best_x.copy()
            x_p[2::3] *= 0.995
            x_p = project(x_p)
            try:
                res = minimize(objective, x_p, method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 30000, 'ftol': 1e-14, 'disp': False})
                if not np.isnan(res.fun):
                    c_opt = np.column_stack((res.x[0::3], res.x[1::3]))
                    r_lp = solve_lp_radii(c_opt)
                    if r_lp is not None:
                        curr_sum = np.sum(r_lp)
                        if curr_sum > best_sum:
                            best_sum = curr_sum
                            best_x = np.zeros(3 * N)
                            best_x[0::3] = c_opt[:, 0]
                            best_x[1::3] = c_opt[:, 1]
                            best_x[2::3] = r_lp
            except Exception:
                break

    if best_x is None:
        c_f = make_hex_init(0.09, 0.0)
        r_f = solve_lp_radii(c_f)
        if r_f is None: r_f = np.full(N, 0.06)
        best_x = np.zeros(3 * N)
        best_x[0::3] = c_f[:, 0]
        best_x[1::3] = c_f[:, 1]
        best_x[2::3] = r_f
        best_sum = np.sum(r_f)
        
    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3].copy()
    
    # Strict validation repair
    for _ in range(100):
        valid = True
        for i in range(N):
            if radii[i] < 0 or centers[i, 0] < radii[i] - 1e-10 or centers[i, 0] > 1.0 - radii[i] + 1e-10 or \
               centers[i, 1] < radii[i] - 1e-10 or centers[i, 1] > 1.0 - radii[i] + 1e-10:
                valid = False; break
        if valid:
            for i in range(N):
                for j in range(i + 1, N):
                    d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                    if d < radii[i] + radii[j] - 1e-10:
                        valid = False; break
                if not valid: break
        if valid: break
        radii *= 0.9998
        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)
        
    return centers, radii, float(np.sum(radii))
