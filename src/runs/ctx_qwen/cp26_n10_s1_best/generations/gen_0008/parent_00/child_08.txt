import numpy as np
from scipy.optimize import minimize, linprog

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Vectorized inequality constraints: g(x) >= 0."""
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    # Boundary constraints
    c = np.concatenate([
        cx - r,
        1.0 - cx - r,
        cy - r,
        1.0 - cy - r
    ])
    
    # Overlap constraints: dist^2 >= (ri + rj)^2
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dr = r[:, None] + r[None, :]
    
    c = np.concatenate([c, dx[TRIL_IDX]**2 + dy[TRIL_IDX]**2 - dr[TRIL_IDX]**2])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-7, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)])
    return b

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A = np.zeros((num_ineq, n))
    b = np.zeros(num_ineq)
    
    idx = 0
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A[idx, i] = 1.0
        b[idx] = max(0.0, lim)
        idx += 1
        
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = max(0.0, d)
            idx += 1
            
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0.0, None)] * n, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def project_to_bounds(x):
    """Ensure optimization vector strictly respects bounds."""
    x = x.copy()
    r = np.maximum(x[2::3], 1e-6)
    x[0::3] = np.clip(x[0::3], r, 1.0 - r)
    x[1::3] = np.clip(x[1::3], r, 1.0 - r)
    x[2::3] = r
    return x

def make_hex_init(r0, angle):
    """Generates a hexagonal lattice initialization with optional rotation."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        if y > 1.0 + r0:
            break
            
    pts = np.array(pts[:N + 10])
    
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
        
    mask = (pts[:, 0] >= 0.01) & (pts[:, 0] <= 0.99) & (pts[:, 1] >= 0.01) & (pts[:, 1] <= 0.99)
    pts = pts[mask]
    
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.1, 0.9, (1, 2))])
    return pts[:N]

def make_force_init(seed, steps=600):
    """Force-directed layout to spread points evenly and push to boundaries."""
    np.random.seed(seed)
    pts = np.random.uniform(0.15, 0.85, (N, 2))
    for step in range(steps):
        f = np.zeros_like(pts)
        for i in range(N):
            for j in range(i + 1, N):
                dx = pts[j] - pts[i]
                d = np.hypot(dx[0], dx[1])
                if d < 0.35 and d > 1e-5:
                    rep = 0.025 / (d**2 + 0.001)
                    f[i] -= rep * dx
                    f[j] += rep * dx
            for dim in range(2):
                if pts[i, dim] < 0.12:
                    f[i, dim] += 0.08
                elif pts[i, dim] > 0.88:
                    f[i, dim] -= 0.08
        lr = 0.06 * (1.0 - step / steps)
        pts += f * lr
        pts = np.clip(pts, 0.03, 0.97)
    return pts

def make_corner_init(seed):
    """Initialize with circles pinned near corners and edges."""
    np.random.seed(seed)
    pts = np.zeros((N, 2))
    
    # 4 corners
    corners = [[0.1, 0.1], [0.9, 0.1], [0.1, 0.9], [0.9, 0.9]]
    for i, c in enumerate(corners):
        pts[i] = [c[0] + np.random.uniform(-0.02, 0.02), c[1] + np.random.uniform(-0.02, 0.02)]
    
    # 8 edge positions
    edges = [[0.3, 0.08], [0.5, 0.08], [0.7, 0.08],
             [0.3, 0.92], [0.5, 0.92], [0.7, 0.92],
             [0.08, 0.3], [0.08, 0.5]]
    for i, e in enumerate(edges):
        pts[4 + i] = [e[0] + np.random.uniform(-0.02, 0.02), e[1] + np.random.uniform(-0.02, 0.02)]
    
    # 14 remaining interior
    for i in range(12, N):
        pts[i] = np.random.uniform(0.15, 0.85, 2)
    
    return pts

def make_mixed_size_init(seed):
    """Initialize with some large and some small circles."""
    np.random.seed(seed)
    pts = np.random.uniform(0.15, 0.85, (N, 2))
    # Spread them out
    for _ in range(200):
        forces = np.zeros_like(pts)
        for i in range(N):
            for j in range(i + 1, N):
                dx = pts[j] - pts[i]
                d = np.hypot(dx[0], dx[1])
                if d < 0.3 and d > 1e-5:
                    f = 0.02 / (d**2 + 0.001)
                    forces[i] -= f * dx
                    forces[j] += f * dx
        pts += forces * 0.03
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def optimize_from_init(x0, bounds, cons, method='SLSQP', maxiter=12000):
    """Run optimization from initial configuration."""
    try:
        res = minimize(objective, x0, method=method, bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-13, 'disp': False})
        if not np.isnan(res.fun):
            c_opt = np.column_stack((res.x[0::3], res.x[1::3]))
            r_lp = solve_lp_radii(c_opt)
            if r_lp is not None:
                return c_opt, r_lp, np.sum(r_lp)
            else:
                return None, None, -1.0
    except Exception:
        pass
    return None, None, -1.0
