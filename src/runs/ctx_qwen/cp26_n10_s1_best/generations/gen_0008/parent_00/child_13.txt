import numpy as np
from scipy.optimize import minimize, linprog

N = 26
TRIU = np.triu_indices(N, 1)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Vectorized inequality constraints: g(x) >= 0."""
    cx, cy, r = x[0::3], x[1::3], x[2::3]
    
    # Boundary constraints
    c = np.concatenate([
        cx - r, 1.0 - cx - r,
        cy - r, 1.0 - cy - r
    ])
    
    # Overlap constraints: dist^2 >= (ri + rj)^2
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dr = r[:, None] + r[None, :]
    
    c = np.concatenate([c, dx[TRIU]**2 + dy[TRIU]**2 - dr[TRIU]**2])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-7, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)])
    return b

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A = np.zeros((num_ineq, n))
    b = np.zeros(num_ineq)
    
    idx = 0
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A[idx, i] = 1.0
        b[idx] = max(0.0, lim)
        idx += 1
        
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = max(0.0, d)
            idx += 1
            
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0.0, None)] * n, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return np.full(n, 0.01)

def project_to_bounds(x):
    """Ensure optimization vector strictly respects bounds."""
    r = np.maximum(x[2::3], 1e-6)
    x[0::3] = np.clip(x[0::3], r, 1.0 - r)
    x[1::3] = np.clip(x[1::3], r, 1.0 - r)
    x[2::3] = r
    return x

def penalty_objective(x, penalty):
    """Smooth penalty objective for L-BFGS-B exploration."""
    cx, cy, r = x[0::3], x[1::3], x[2::3]
    obj = -np.sum(r)
    
    # Boundary penalties
    for i in range(N):
        for dim in range(2):
            coord = cx[i] if dim == 0 else cy[i]
            v1 = min(0.0, coord - r[i])
            v2 = min(0.0, 1.0 - coord - r[i])
            obj += penalty * (v1**2 + v2**2)
        obj += penalty * min(0.0, r[i])**2
            
    # Overlap penalties
    for i in range(N):
        for j in range(i + 1, N):
            d = np.hypot(cx[i] - cx[j], cy[i] - cy[j])
            v = min(0.0, d - r[i] - r[j])
            obj += penalty * v**2
    return obj

def make_hex_init(r0, angle):
    """Generates a hexagonal lattice initialization with optional rotation."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 8:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        if y > 1.0 + r0:
            break
            
    pts = np.array(pts[:N + 8])
    
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
        
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.1, 0.9, (1, 2))])
    return pts[:N]

def make_force_init(seed, steps=400):
    """Force-directed layout to spread points evenly and push to boundaries."""
    np.random.seed(seed)
    pts = np.random.uniform(0.15, 0.85, (N, 2))
    for step in range(steps):
        f = np.zeros_like(pts)
        for i in range(N):
            for j in range(i + 1, N):
                dx = pts[j] - pts[i]
                d = np.hypot(dx[0], dx[1])
                if d < 0.3 and d > 1e-5:
                    rep = 0.015 / (d**2 + 0.001)
                    f[i] -= rep * dx
                    f[j] += rep * dx
            for dim in range(2):
                if pts[i, dim] < 0.12: f[i, dim] += 0.04
                elif pts[i, dim] > 0.88: f[i, dim] -= 0.04
        lr = 0.05 * (1.0 - step / steps)
        pts += f * lr
        pts = np.clip(pts, 0.05, 0.95)
    return pts

    cons = {'type': 'ineq', 'fun': constraints}

    inits = []

            inits.append(make_hex_init(r0, ang))

    # Phase 2: Multi-start optimization with LP snapping
    for c_init in inits:
        r_init = solve_lp_radii(c_init)
        x0 = np.zeros(3 * N)
        x0[0::3] = c_init[:, 0]
        x0[1::3] = c_init[:, 1]
        x0[2::3] = r_init
        x0 = project_to_bounds(x0)
        
        # Break exact symmetries
        x0 += np.random.normal(0, 1e-5, x0.shape)
        
        # Try SLSQP directly
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 20000, 'ftol': 1e-13, 'disp': False})
            if not np.isnan(res.fun):
                c_opt = np.column_stack((res.x[0::3], res.x[1::3]))
                r_lp = solve_lp_radii(c_opt)
                if r_lp is not None:
                    curr_sum = np.sum(r_lp)
                    if curr_sum > best_sum:
                        best_sum = curr_sum
                        best_x = np.zeros(3 * N)
                        best_x[0::3] = c_opt[:, 0]
                        best_x[1::3] = c_opt[:, 1]
                        best_x[2::3] = r_lp
        except Exception:
            pass
            
    # Phase 3: Penalty-based exploration + LP snap for promising basins
    if best_x is not None:
        x_curr = best_x.copy()
        for _ in range(3):
            # Soft optimization to escape local minima
            for pen in [100, 500, 2000]:
                res = minimize(lambda x: penalty_objective(x, pen), x_curr, method='L-BFGS-B',
                               bounds=bounds, options={'maxiter': 5000, 'ftol': 1e-14})
                x_curr = res.x
                
            # Project and LP snap
            x_curr = project_to_bounds(x_curr)
            c_opt = np.column_stack((x_curr[0::3], x_curr[1::3]))
            r_lp = solve_lp_radii(c_opt)
            if r_lp is not None:
                x_curr[2::3] = r_lp

                ctrs = (ctrs - 0.5) @ rot.T + 0.5
