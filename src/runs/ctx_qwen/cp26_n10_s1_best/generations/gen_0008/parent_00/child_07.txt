def objective_centers(centers):
    """Compute negative sum of radii for given centers using LP."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A = np.zeros((num_ineq, n))
    b = np.zeros(num_ineq)
    
    idx = 0
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], 
                  centers[i, 1], 1.0 - centers[i, 1])
        A[idx, i] = 1.0
        b[idx] = max(0.0, lim)
        idx += 1
        
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], 
                        centers[i, 1] - centers[j, 1])
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = max(0.0, d)
            idx += 1
            
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0.0, None)] * n, method='highs')
        if res.success:
            return -np.sum(res.x)
    except Exception:
        pass
    return 0.0

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A = np.zeros((num_ineq, n))
    b = np.zeros(num_ineq)
    
    idx = 0
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], 
                  centers[i, 1], 1.0 - centers[i, 1])
        A[idx, i] = 1.0
        b[idx] = max(0.0, lim)
        idx += 1
        
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], 
                        centers[i, 1] - centers[j, 1])
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = max(0.0, d)
            idx += 1
            
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0.0, None)] * n, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def make_hex_init(r0, angle, n=N):
    """Generates a hexagonal lattice initialization with optional rotation."""
    pts = []
    y = r0
    row = 0
    while len(pts) < n + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        if y > 1.0 + r0:
            break
            
    pts = np.array(pts[:n + 10])
    
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
        
    mask = (pts[:, 0] >= 0.01) & (pts[:, 0] <= 0.99) & (pts[:, 1] >= 0.01) & (pts[:, 1] <= 0.99)
    pts = pts[mask]
    
    while len(pts) < n:
        pts = np.vstack([pts, np.random.uniform(0.1, 0.9, (1, 2))])
    return pts[:n]

def make_corner_init(seed):

    np.random.seed(seed)

    pts.append([corner_margin + np.random.uniform(-0.01, 0.01), 1 - corner_margin + np.random.uniform(-0.01, 0.01)])
    pts.append([1 - corner_margin + np.random.uniform(-0.01, 0.01), 1 - corner_margin + np.random.uniform(-0.01, 0.01)])

        t = 0.15 + i * 0.175
        edge_pts.append([t, 0.08 + np.random.uniform(-0.01, 0.01)])
        edge_pts.append([t, 0.92 + np.random.uniform(-0.01, 0.01)])
        if len(pts) + len(edge_pts) >= N:
            break
    pts.extend(edge_pts[:min(N - len(pts), len(edge_pts))])
    
    # Fill remaining with grid
    remaining = N - len(pts)

    return np.clip(pts, 0.02, 0.98)

def make_grid_init(n, spacing_x=0.18, spacing_y=0.17):
    """Create a grid initialization."""
    pts = []
    y = 0.1
    row = 0
    while len(pts) < n + 5:
        x = 0.1
        while x <= 0.9 and len(pts) < n + 5:
            pts.append([x, y])
            x += spacing_x
        y += spacing_y
        row += 1
    return np.array(pts[:n]) + np.random.uniform(-0.005, 0.005, (n, 2))

def make_staggered_init(n, seed):
    """Create staggered row initialization."""
    np.random.seed(seed)
    pts = []
    y = 0.09
    row = 0
    while len(pts) < n:
        x_start = 0.09 if row % 2 == 0 else 0.15
        x = x_start
        while x <= 0.91 and len(pts) < n:
            pts.append([x + np.random.uniform(-0.01, 0.01), y + np.random.uniform(-0.005, 0.005)])
            x += 0.18
        y += 0.15
        row += 1
    return np.array(pts[:n])

def force_spread(centers, steps=300, push_boundary=True):
    """Force-directed layout to spread points."""
    pts = centers.copy()
    n = pts.shape[0]
    for step in range(steps):
        f = np.zeros_like(pts)
        for i in range(n):
            for j in range(i + 1, n):
                dx = pts[j] - pts[i]
                d = np.hypot(dx[0], dx[1])
                if d < 0.35 and d > 1e-6:
                    rep = 0.02 / (d**2 + 0.002)
                    f[i] -= rep * dx
                    f[j] += rep * dx
            if push_boundary:
                for dim in range(2):
                    if pts[i, dim] < 0.12:
                        f[i, dim] += 0.08
                    elif pts[i, dim] > 0.88:
                        f[i, dim] -= 0.08
        lr = 0.04 * max(0.1, 1.0 - step / steps)
        pts += f * lr
        pts = np.clip(pts, 0.02, 0.98)
    return pts

def simulated_annealing(centers, initial_temp=0.05, final_temp=0.0001, cooling=0.97, n_steps=2000):
    """Simulated annealing on center positions with LP for radii."""
    n = centers.shape[0]
    best_centers = centers.copy()
    radii = solve_lp_radii(centers)
    if radii is None:
        radii = np.full(n, 0.05)
    best_sum = np.sum(radii)
    
    temp = initial_temp
    current_centers = centers.copy()
    current_radii = radii.copy()
    current_sum = best_sum
    
    for step in range(n_steps):
        # Perturb centers
        step_size = temp * 0.5
        new_centers = current_centers.copy()
        new_centers += np.random.normal(0, step_size, new_centers.shape)
        new_centers = np.clip(new_centers, 0.01, 0.99)
        
        # Get radii for new centers
        new_radii = solve_lp_radii(new_centers)
        if new_radii is None:
            temp *= cooling
            continue
            
        new_sum = np.sum(new_radii)
        delta = new_sum - current_sum
        
        # Accept or reject
        if delta > 0 or np.random.random() < np.exp(delta / temp):
            current_centers = new_centers
            current_radii = new_radii
            current_sum = new_sum
            
            if current_sum > best_sum:
                best_centers = current_centers.copy()
                best_sum = current_sum
        
        temp *= cooling
    
    return best_centers
