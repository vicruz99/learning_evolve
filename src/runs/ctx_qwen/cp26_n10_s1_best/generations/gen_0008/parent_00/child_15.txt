import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A = np.zeros((num_ineq, n))
    b = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A[idx, i] = 1.0
        b[idx] = max(0.0, lim)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = max(0.0, d)
            idx += 1
            
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0.0, None)] * n, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return np.full(n, 0.01)

def relax_centers(centers, radii, steps=120, step_size=0.008):
    """Vectorized force-directed relaxation to push overlapping circles apart."""
    c = centers.copy()
    r = radii.copy()
    for step in range(steps):
        forces = np.zeros_like(c)
        
        # Vectorized pair interactions
        dx = c[:, None, 0] - c[None, :, 0]
        dy = c[:, None, 1] - c[None, :, 1]
        d_sq = dx**2 + dy**2
        d_sq = np.maximum(d_sq, 1e-12)
        d = np.sqrt(d_sq)
        np.fill_diagonal(d, 0.0)
        
        overlap = r[:, None] + r[None, :] - d
        overlap = np.maximum(overlap, 0.0)
        np.fill_diagonal(overlap, 0.0)
        
        # Force magnitude: overlap / distance
        f_mag = overlap / d
        np.fill_diagonal(f_mag, 0.0)
        
        fx = f_mag * dx
        fy = f_mag * dy
        
        forces[:, 0] = np.sum(fx, axis=1)
        forces[:, 1] = np.sum(fy, axis=1)
        
        # Boundary repulsion forces
        for i in range(N):
            if c[i, 0] < r[i]:
                forces[i, 0] += (r[i] - c[i, 0]) * 2.0
            elif c[i, 0] > 1.0 - r[i]:
                forces[i, 0] -= (c[i, 0] - (1.0 - r[i])) * 2.0
                
            if c[i, 1] < r[i]:
                forces[i, 1] += (r[i] - c[i, 1]) * 2.0
            elif c[i, 1] > 1.0 - r[i]:
                forces[i, 1] -= (c[i, 1] - (1.0 - r[i])) * 2.0
                
        lr = step_size * (1.0 - step / steps)
        c += forces * lr
        c = np.clip(c, r, 1.0 - r)
        
    return c

def make_hex_init(seed):
    """Generates a hexagonal lattice initialization with optional perturbation."""
    np.random.seed(seed)
    pts = []
    y = 0.09
    row = 0
    while len(pts) < N + 5:
        x = 0.09 if row % 2 == 0 else 0.18
        while x <= 1.0 - 0.09:
            pts.append([x, y])
            x += 0.18
        y += np.sqrt(3.0) * 0.09
        row += 1
        if y > 1.0 + 0.09:
            break
    pts = np.array(pts[:N + 5])
    pts = (pts - 0.5) * 0.95 + 0.5
    pts = np.clip(pts, 0.05, 0.95)
    
    # Add small random perturbation
    pts += np.random.normal(0, 0.005, pts.shape)
    
    while len(pts) < N:
        pts = np.vstack([pts, [0.5, 0.5]])
    return pts[:N]

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Phase 1: Multi-start alternating LP + Relaxation
    for restart in range(10):
        c = make_hex_init(restart)
        r = solve_lp_radii(c)
        current_sum = np.sum(r)
        
        c_best = c.copy()
        r_best = r.copy()
        sum_best = current_sum
        
        for cycle in range(30):
            # Relax centers to resolve overlaps
            c = relax_centers(c, r, steps=120, step_size=0.008)
            # Expand radii optimally
            r = solve_lp_radii(c)
            current_sum = np.sum(r)
            
            if current_sum > sum_best:
                sum_best = current_sum
                c_best = c.copy()
                r_best = r.copy()
                
            # Swap perturbation to escape local topological minima
            if cycle % 5 == 4:
                idx1, idx2 = np.random.choice(N, 2, replace=False)
                c[idx1], c[idx2] = c[idx2].copy(), c[idx1].copy()
                r = r * 0.96
                for i in range(N):
                    c[i, 0] = np.clip(c[i, 0], r[i], 1.0 - r[i])
                    c[i, 1] = np.clip(c[i, 1], r[i], 1.0 - r[i])
                    
        if sum_best > best_sum:
            best_sum = sum_best
            best_centers = c_best.copy()
            best_radii = r_best.copy()
            
    # Phase 2: SLSQP High-Precision Polish
    x0 = np.zeros(3 * N)
    x0[0::3] = best_centers[:, 0]
    x0[1::3] = best_centers[:, 1]
    x0[2::3] = best_radii
    
    def obj(x):
        return -np.sum(x[2::3])

    def cons(x):
        cx, cy, rr = x[0::3], x[1::3], x[2::3]
        c = np.concatenate([cx - rr, 1.0 - cx - rr, cy - rr, 1.0 - cy - rr])
        dx = cx[:, None] - cx[None, :]
        dy = cy[:, None] - cy[None, :]
        dr = rr[:, None] + rr[None, :]
        i_idx, j_idx = np.tril_indices(N, -1)
        c = np.concatenate([c, dx[i_idx, j_idx]**2 + dy[i_idx, j_idx]**2 - dr[i_idx, j_idx]**2])
        return c
        
    bnds = [(0, 1), (0, 1), (0, 0.5)] * N
    try:
        res = minimize(obj, x0, method='SLSQP', bounds=bnds, 
                       constraints={'type': 'ineq', 'fun': cons},
                       options={'maxiter': 10000, 'ftol': 1e-13, 'disp': False})
        if not np.isnan(res.fun):
            c_polish = np.column_stack((res.x[0::3], res.x[1::3]))
            r_polish = solve_lp_radii(c_polish)
            if r_polish is not None and np.sum(r_polish) > best_sum:
                best_sum = np.sum(r_polish)
                best_centers = c_polish.copy()
                best_radii = r_polish.copy()
    except Exception:
        pass

    # Final strict validity adjustment against 1e-12 tolerance
    for _ in range(100):
        valid = True
        for i in range(N):
            if best_radii[i] < 0 or best_centers[i, 0] < best_radii[i] - 1e-10 or \
               best_centers[i, 0] > 1.0 - best_radii[i] + 1e-10 or \
               best_centers[i, 1] < best_radii[i] - 1e-10 or \
               best_centers[i, 1] > 1.0 - best_radii[i] + 1e-10:
                valid = False
                break
        if valid:
            for i in range(N):
                for j in range(i + 1, N):
                    d = np.hypot(best_centers[i, 0] - best_centers[j, 0], 
                                 best_centers[i, 1] - best_centers[j, 1])
                    if d < best_radii[i] + best_radii[j] - 1e-10:
                        valid = False
                        break
                if not valid:
                    break
        if valid:
            break
            
        # Minimal shrinkage to recover strict feasibility
        best_radii *= 0.9995
        best_centers[:, 0] = np.clip(best_centers[:, 0], best_radii, 1.0 - best_radii)
        best_centers[:, 1] = np.clip(best_centers[:, 1], best_radii, 1.0 - best_radii)
        
    return best_centers, best_radii, float(np.sum(best_radii))
