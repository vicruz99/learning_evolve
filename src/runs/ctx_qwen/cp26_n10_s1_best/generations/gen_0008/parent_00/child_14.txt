import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A = np.zeros((num_ineq, n))
    b = np.zeros(num_ineq)
    
    idx = 0
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A[idx, i] = 1.0
        b[idx] = max(0.0, lim)
        idx += 1
        
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = max(0.0, d)
            idx += 1
            
    bounds_lp = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds_lp, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return np.full(n, 0.01)


def make_hex_init(r0, angle, seed=0):
    """Generates a hexagonal lattice initialization with optional rotation."""
    np.random.seed(seed)
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        if y > 1.0 + r0:
            break
            
    pts = np.array(pts[:N + 10])
    
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
        
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.1, 0.9, (1, 2))])
    return pts[:N]


def make_corner_init(seed=0):
    """Generate configuration with circles near corners for larger radii."""
    np.random.seed(seed)
    pts = []
    
    # Place circles near corners with varying offsets
    for _ in range(4):
        offset = np.random.uniform(0.08, 0.20)
        cx = np.random.choice([offset, 1 - offset])
        cy = np.random.choice([offset, 1 - offset])
        pts.append([cx, cy])
    
    # Fill remaining with hexagonal pattern
    r = 0.08
    y = r
    row = 0
    while len(pts) < N:
        x = r if row % 2 == 0 else 2.0 * r
        while x <= 1.0 - r and len(pts) < N:
            pts.append([x, y])
            x += 2.0 * r
        y += np.sqrt(3.0) * r
        row += 1
        if y > 1.0 + r:
            break
    
    while len(pts) < N:
        pts.append(np.random.uniform(0.15, 0.85, 2))
    
    return np.array(pts[:N])


def make_staggered_grid_init(seed=0):
    """Generate a staggered grid configuration."""
    np.random.seed(seed)
    pts = []
    spacing_x = 0.18
    spacing_y = 0.16
    offset_x = np.random.uniform(0.0, 0.05)
    offset_y = np.random.uniform(0.0, 0.05)
    
    row = 0
    while len(pts) < N:
        col_offset = offset_x + (row % 2) * spacing_x * 0.5
        col = 0
        while len(pts) < N:
            x = col_offset + col * spacing_x + np.random.uniform(-0.01, 0.01)
            y = offset_y + row * spacing_y + np.random.uniform(-0.01, 0.01)
            if 0.05 <= x <= 0.95 and 0.05 <= y <= 0.95:
                pts.append([x, y])
            col += 1
        row += 1
        if row > 10:
            break
    
    while len(pts) < N:
        pts.append(np.random.uniform(0.1, 0.9, 2))
    
    return np.array(pts[:N])


def make_force_init(seed, steps=400):
    """Force-directed layout to spread points evenly."""
    np.random.seed(seed)
    pts = np.random.uniform(0.1, 0.9, (N, 2))
    
    for step in range(steps):
        f = np.zeros_like(pts)
        for i in range(N):
            for j in range(i + 1, N):
                dx = pts[j] - pts[i]
                d = np.hypot(dx[0], dx[1])
                if d < 0.3 and d > 1e-5:
                    rep = 0.02 / (d**2 + 0.001)
                    f[i] -= rep * dx / d
                    f[j] += rep * dx / d
            for dim in range(2):
                if pts[i, dim] < 0.12:
                    f[i, dim] += 0.06
                elif pts[i, dim] > 0.88:
                    f[i, dim] -= 0.06
        lr = 0.04 * max(0.0, 1.0 - step / steps)
        pts += f * lr
        pts = np.clip(pts, 0.05, 0.95)
    return pts


def simulated_annealing_centers(centers, max_iter=3000, T0=0.08, T_min=1e-8):
    """Simulated annealing on center positions with LP-based objective."""
    current_centers = centers.copy()
    current_radii = solve_lp_radii(current_centers)
    current_value = np.sum(current_radii)
    
    best_centers = current_centers.copy()
    best_radii = current_radii.copy()
    best_value = current_value
    
    for i in range(max_iter):
        T = T0 * (T_min / T0) ** (i / max_iter)
        
        # Adaptive noise scale
        noise_scale = T * 0.3
        new_centers = current_centers.copy()
        new_centers += np.random.normal(0, noise_scale, centers.shape)
        new_centers = np.clip(new_centers, 0.005, 0.995)
        
        new_radii = solve_lp_radii(new_centers)
        new_value = np.sum(new_radii)
        
        delta = new_value - current_value
        if delta > 0 or (np.random.random() < np.exp(min(delta / max(T, 1e-12), 10))):
            current_centers = new_centers.copy()
            current_radii = new_radii.copy()
            current_value = new_value
            
            if new_value > best_value:
                best_centers = new_centers.copy()
                best_radii = new_radii.copy()
                best_value = new_value
    
    return best_centers, best_radii, best_value


def cluster_perturbation(centers, radii, num_trials=500):
    """Perturb clusters of circles and recompute radii."""
    n = N
    best_centers = centers.copy()
    best_radii = radii.copy()
    best_sum = np.sum(radii)
    
    for trial in range(num_trials):
        k = np.random.randint(2, 7)
        idx = np.random.choice(n, k, replace=False)
        
        pert_centers = centers.copy()
        noise_scale = np.random.uniform(0.005, 0.03)
        pert_centers[idx] += np.random.normal(0, noise_scale, (k, 2))
        pert_centers = np.clip(pert_centers, 0.01, 0.99)
        
        pert_radii = solve_lp_radii(pert_centers)
        pert_sum = np.sum(pert_radii)
        
        if pert_sum > best_sum:
            best_sum = pert_sum
            best_centers = pert_centers.copy()
            best_radii = pert_radii.copy()
    
    return best_centers, best_radii, best_sum


def swap_perturbation(centers, radii, num_trials=200):
    """Swap positions of pairs of circles."""
    n = N
    best_centers = centers.copy()
    best_radii = radii.copy()
    best_sum = np.sum(radii)
    
    for _ in range(num_trials):
        i, j = np.random.choice(n, 2, replace=False)
        pert_centers = centers.copy()
        pert_centers[[i, j]] = pert_centers[[j, i]]
        
        pert_radii = solve_lp_radii(pert_centers)
        pert_sum = np.sum(pert_radii)
        
        if pert_sum > best_sum:
            best_sum = pert_sum
            best_centers = pert_centers.copy()
            best_radii = pert_radii.copy()
    
    return best_centers, best_radii, best_sum


def scale_rotate_perturbation(centers, radii):
    """Try scaling and rotating the configuration."""
    best_centers = centers.copy()
    best_radii = radii.copy()
    best_sum = np.sum(radii)
    
    for scale in np.linspace(0.92, 1.08, 30):
        for ang in np.linspace(-0.15, 0.15, 30):
            pert_centers = (centers - 0.5) * scale
            c, s = np.cos(ang), np.sin(ang)
            pert_centers = pert_centers @ np.array([[c, -s], [s, c]]) + 0.5
            pert_centers = np.clip(pert_centers, 0.01, 0.99)
            
            pert_radii = solve_lp_radii(pert_centers)

            if pert_sum > best_sum:

    return best_centers, best_radii, best_sum


def slesqp_refinement(centers, radii, max_iter=8000):
    """Use SLSQP for joint center-radius refinement."""
    n = N
    TRIL = np.tril_indices(n, -1)
    
    def obj(x):
        return -np.sum(x[2::3])
    
    def cons(x):
        cx = x[0::3]
        cy = x[1::3]
        r = x[2::3]
        c = np.concatenate([
            cx - r, 1.0 - cx - r,
            cy - r, 1.0 - cy - r
        ])
        dx = cx[:, None] - cx[None, :]
        dy = cy[:, None] - cy[None, :]
        dr = r[:, None] + r[None, :]
        c = np.concatenate([c, dx[TRIL]**2 + dy[TRIL]**2 - dr[TRIL]**2])
        return c
    
    bounds_opt = []
    for _ in range(n):
        bounds_opt.extend([(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)])
    
    x0 = np.zeros(3 * n)
    x0[0::3] = centers[:, 0]
    x0[1::3] = centers[:, 1]
    x0[2::3] = radii
    
    cons_dict = {'type': 'ineq', 'fun': cons}
    
    try:
        res = minimize(obj, x0, method='SLSQP', bounds=bounds_opt,
                       constraints=cons_dict,
                       options={'maxiter': max_iter, 'ftol': 1e-14, 'disp': False})
        if not np.isnan(res.fun):
            c_opt = np.column_stack((res.x[0::3], res.x[1::3]))
            r_opt = solve_lp_radii(c_opt)
            return c_opt, r_opt, np.sum(r_opt)
    except Exception:
        pass
    return centers, radii, np.sum(radii)
