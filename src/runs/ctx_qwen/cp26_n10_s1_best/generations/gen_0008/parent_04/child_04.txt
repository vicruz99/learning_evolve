1. **Diverse Initialization**: Generate a wide variety of starting configurations including rotated/scaled hexagonal lattices and force-directed spreads. This covers different topological arrangements.
2. **LP-SLSQP Alternation**: Linear Programming (LP) is exceptionally effective for finding optimal radii given fixed centers. We use LP to snap radii to their theoretical maximum, then use SLSQP to simultaneously adjust centers and radii. This alternation converges faster and more reliably than optimizing all variables at once.
3. **Continuation/Deflation Method**: To escape local minima, we progressively shrink the radii of the best configuration found so far. This creates "slack" space, allowing the optimizer to reposition centers into previously inaccessible, higher-quality arrangements. We gradually return the deflation factor to 1.0 to fine-tune.
4. **Topology Swapping**: Periodically swapping positions of two circles breaks symmetry and allows the optimizer to explore different contact graphs.
5. **Robust Validation & Repair**: A final strict check ensures compliance with the 1e-12 tolerance, with minimal shrinkage only if necessary.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
TRIL = np.tril_indices(N, -1)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Vectorized inequality constraints: g(x) >= 0."""
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    c = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dr = r[:, None] + r[None, :]
    
    c = np.concatenate([c, dx[TRIL]**2 + dy[TRIL]**2 - dr[TRIL]**2])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [0, 0.5]."""
    return [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N

def solve_lp(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    num_ineq = n + n * (n - 1) // 2
    A = np.zeros((num_ineq, n))
    b = np.zeros(num_ineq)
    
    idx = 0
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A[idx, i] = 1.0
        b[idx] = max(0.0, lim)
        idx += 1
        
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = max(0.0, d)
            idx += 1
            
    bounds_lp = [(0.0, None)] * n
    try:
        res = linprog(-np.ones(n), A_ub=A, b_ub=b, bounds=bounds_lp, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return np.full(n, 0.02)

def project_to_feasible(x):
    """Ensure optimization vector strictly respects variable bounds."""
    x = x.copy()
    for i in range(N):
        r = max(1e-7, x[3 * i + 2])
        x[3 * i] = np.clip(x[3 * i], r, 1.0 - r)
        x[3 * i + 1] = np.clip(x[3 * i + 1], r, 1.0 - r)
        x[3 * i + 2] = r
    return x

def generate_hex_init(r0, angle):
    """Generates a hexagonal lattice initialization with optional rotation."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        if y > 1.0 + r0:
            break
            
    pts = np.array(pts[:N + 10])
    
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
        
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.15, 0.85, (1, 2))])
    return pts[:N]

def generate_force_init(seed, steps=600):
    """Force-directed layout to spread points evenly and push to boundaries."""
    np.random.seed(seed)
    pts = np.random.uniform(0.15, 0.85, (N, 2))
    
    for step in range(steps):
        forces = np.zeros_like(pts)
        for i in range(N):
            for j in range(i + 1, N):
                diff = pts[j] - pts[i]
                d = np.linalg.norm(diff)
                if d < 0.3 and d > 1e-5:
                    f = 0.001 / (d**2 + 0.001)
                    forces[i] -= diff * f / d
                    forces[j] += diff * f / d
                    
            for dim in range(2):
                if pts[i, dim] < 0.12:
                    forces[i, dim] += 0.04
                elif pts[i, dim] > 0.88:
                    forces[i, dim] -= 0.04
                    
        lr = 0.05 * (1.0 - step / steps)
        pts += forces * lr
        pts = np.clip(pts, 0.05, 0.95)
        
    return pts

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    best_sum = -1.0
    best_x = None
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    
    # Phase 1: Generate diverse initial configurations
    inits = []
    for r0 in np.linspace(0.075, 0.105, 8):
        for ang in np.linspace(-0.4, 0.4, 9):
            inits.append(generate_hex_init(r0, ang))
            
    for s in range(15):
        inits.append(generate_force_init(s))
        
    # Phase 2: Multi-start Optimization with LP initialization
    for pts in inits:
        rad = solve_lp(pts)
        x0 = np.zeros(3 * N)
        x0[0::3] = pts[:, 0]
        x0[1::3] = pts[:, 1]
        x0[2::3] = rad
        x0 = project_to_feasible(x0)
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 12000, 'ftol': 1e-14, 'disp': False})
            if not np.isnan(res.fun) and np.min(constraints(res.x)) >= -1e-5:
                s = -res.fun
                if s > best_sum:
                    best_sum = s
                    best_x = res.x.copy()
        except Exception:
            continue

    # Phase 3: Continuation/Deflation Method & Topology Search
    if best_x is not None:
        for it in range(100):
            # Schedule deflation factor: starts lower to allow big moves, approaches 1.0
            shrink_factor = 0.92 + 0.06 * (it / 100.0)
            x0 = best_x.copy()
            x0[2::3] *= shrink_factor
            
            # Perturb centers with cooling noise
            noise_scale = 0.004 * (0.91 ** it)
            x0[0::3] += np.random.normal(0, noise_scale, N)
            x0[1::3] += np.random.normal(0, noise_scale, N)
            
            # Occasional swap to break symmetry and explore different topologies
            if it % 7 == 0:
                i_swap, j_swap = np.random.choice(N, 2, replace=False)
                for k in range(3):
                    x0[3*i_swap+k], x0[3*j_swap+k] = x0[3*j_swap+k], x0[3*i_swap+k]
                
            x0 = project_to_feasible(x0)
            
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 8000, 'ftol': 1e-14, 'disp': False})
                if not np.isnan(res.fun):
                    s = -res.fun
                    if s > best_sum:
                        best_sum = s
                        best_x = res.x.copy()
            except Exception:
                pass
                
            # LP refinement on current centers to maximize radii immediately
            c_tmp = np.column_stack((best_x[0::3], best_x[1::3]))
            r_lp = solve_lp(c_tmp)
            if r_lp is not None:
                best_x[2::3] = r_lp * 0.999
                curr_sum = np.sum(best_x[2::3])
                if curr_sum > best_sum:
                    best_sum = curr_sum
                    
                # Secondary polish with slightly deflated radii
                x0_fix = best_x.copy()
                x0_fix[2::3] *= 0.96
                x0_fix = project_to_feasible(x0_fix)
                try:
                    res = minimize(objective, x0_fix, method='SLSQP', bounds=bounds, constraints=cons,
                                   options={'maxiter': 6000, 'ftol': 1e-14, 'disp': False})
                    if not np.isnan(res.fun):
                        s = -res.fun
                        if s > best_sum:
                            best_sum = s
                            best_x = res.x.copy()
                except Exception:
                    pass

    # Fallback initialization (should rarely be reached)
    if best_x is None:
        best_x = np.zeros(3 * N)
        best_x[0::3] = np.tile(np.linspace(0.1, 0.9, 5), 6)[:N]
        best_x[1::3] = np.repeat(np.linspace(0.1, 0.9, 6), 5)[:N]
        best_x[2::3] = 0.08
        
    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3].copy()
    
    # Final strict validation repair to guarantee compliance with 1e-12 tolerance
    for _ in range(100):
        valid = True
        for i in range(N):
            if radii[i] < 0 or centers[i, 0] < radii[i] - 1e-9 or centers[i, 0] > 1.0 - radii[i] + 1e-9 or \
               centers[i, 1] < radii[i] - 1e-9 or centers[i, 1] > 1.0 - radii[i] + 1e-9:
                valid = False
                break
        if valid:
            for i in range(N):
                for j in range(i + 1, N):
                    d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                    if d < radii[i] + radii[j] - 1e-9:
                        valid = False
                        break
                if not valid:
                    break
        if valid:
            break
            
        # Minimal shrinkage to recover strict feasibility
        radii *= 0.9995
        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)
        
    return centers, radii, float(np.sum(radii))
