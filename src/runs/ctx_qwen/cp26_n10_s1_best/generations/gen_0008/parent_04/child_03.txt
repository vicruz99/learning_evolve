import numpy as np
from scipy.optimize import minimize, linprog

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraint_fun(x):
    """Returns all inequality constraints >= 0 (vectorized)."""
    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    c = np.concatenate([
        xs - rs, 1.0 - xs - rs,
        ys - rs, 1.0 - ys - rs
    ])
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    c = np.concatenate([c, dx[TRIL_IDX]**2 + dy[TRIL_IDX]**2 - dr[TRIL_IDX]**2])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [0, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
    return b

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        bound = min(centers[i, 0], 1.0 - centers[i, 0], 
                    centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, bound)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], 
                            centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist)
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return np.full(n, 0.01)

def project_to_bounds(x):
    """Ensure optimization vector strictly respects bounds."""
    x = x.copy()
    for i in range(N):
        r = max(1e-7, x[3 * i + 2])
        x[3 * i] = np.clip(x[3 * i], r, 1.0 - r)
        x[3 * i + 1] = np.clip(x[3 * i + 1], r, 1.0 - r)
        x[3 * i + 2] = r
    return x

def make_structured_init(r0, angle=0.0, shift_x=0.0, shift_y=0.0):
    """Generate structured hexagonal lattice initialization tailored for N=26."""
    pts = []
    y = r0
    row = 0
    # 5 rows pattern: 6, 5, 6, 5, 4
    counts = [6, 5, 6, 5, 4]
    
    for c in counts:
        x_start = r0 + shift_x if row % 2 == 0 else 2.0 * r0 + shift_x
        for _ in range(c):
            if x_start + r0 <= 1.0:
                pts.append([x_start, y + shift_y])
                x_start += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        
    pts = np.array(pts)
    if len(pts) < N:
        # Pad with random points if structure didn't fill exactly
        pad = N - len(pts)
        pts = np.vstack([pts, np.random.uniform(0.15, 0.85, (pad, 2))])
    pts = pts[:N]
    
    if angle != 0.0:
        c_a, s_a = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c_a, -s_a], [s_a, c_a]]) + 0.5
        
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.15, 0.85, (1, 2))])
    return pts[:N]

def force_relaxation(centers, radii, steps=150, k_rep=15.0, k_bound=25.0):
    """Physics-based force relaxation to clear overlaps and push to boundaries."""
    pts = centers.copy()
    r = radii.copy()
    lr = 0.012
    
    for step in range(steps):
        forces = np.zeros_like(pts)
        for i in range(N):
            # Boundary repulsion
            for d in range(2):
                if pts[i, d] < r[i]: forces[i, d] += k_bound * (r[i] - pts[i, d])
                if pts[i, d] > 1.0 - r[i]: forces[i, d] -= k_bound * (pts[i, d] - (1.0 - r[i]))
                
            # Pairwise repulsion
            for j in range(i + 1, N):
                diff = pts[j] - pts[i]
                dist = np.linalg.norm(diff)
                if dist < r[i] + r[j] and dist > 1e-8:
                    overlap = r[i] + r[j] - dist
                    f = k_rep * overlap
                    forces[i] += f * diff / dist
                    forces[j] -= f * diff / dist
                    
        pts += forces * lr
        pts = np.clip(pts, 1e-5, 1.0 - 1e-5)
        lr *= 0.98
        
    return pts

def optimize_config(x0):
    """Run SLSQP optimization on a configuration."""
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraint_fun}
    
    try:
        res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 15000, 'ftol': 1e-14, 'disp': False})
        if not np.isnan(res.fun):
            return res.x
    except Exception:
        pass
    return None

                    idx += 1

        res_x = optimize_config(x0)

            r_new = solve_lp_radii(c_tmp)
            if r_new is not None:

                if s > best_sum:
                    best_sum = s
                    best_x = np.zeros(3 * N)
                    best_x[0::3] = c_tmp[:, 0]
                    best_x[1::3] = c_tmp[:, 1]
