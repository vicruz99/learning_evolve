The previous solutions consistently converge around a sum of radii of ~2.634, which is significantly lower than the theoretical optimum (~2.76 for N=26). The main bottleneck is that standard joint optimization (SLSQP on centers and radii simultaneously) gets trapped in local minima where circles are tightly packed but suboptimally arranged.

To break through this barrier, I will implement an **Iterative Clearance Maximization** strategy:
1. **Linear Programming for Radii**: Given fixed centers, LP instantly computes the optimal radii that maximize the sum while respecting all pairwise and boundary distances.
2. **Clearance Maximization**: Given fixed radii, optimize centers to maximize the minimum clearance `t` such that `dist(i,j) >= r_i + r_j + t` and `wall_dist >= r_i + t`. This explicitly pushes circles apart to create breathing room.
3. **Growth Loop**: After finding clearance `t`, increase all radii by `t` (or a fraction), then re-run LP and clearance maximization. This iterative growth naturally escapes flat local minima and drives the packing towards tighter configurations.
4. **Multi-Start & Polish**: Apply this pipeline to diverse initial configurations (rotated/scaled hexagonal lattices and force-directed layouts). After growth, run a final joint SLSQP polish to fine-tune both centers and radii simultaneously.
5. **Strict Validation**: Ensure the final output passes the provided validator with numerical safety margins.

This approach decouples the non-convex center placement from radius scaling, allowing the optimizer to make larger, more directed progress in the configuration space.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [0, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
    return b

def constraints_func(x):
    """Vectorized inequality constraints: g(x) >= 0."""
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    c = np.concatenate([
        cx - r,
        1.0 - cx - r,
        cy - r,
        1.0 - cy - r
    ])
    
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dr = r[:, None] + r[None, :]
    
    i_idx, j_idx = np.tril_indices(N, -1)
    c = np.concatenate([c, dx[i_idx, j_idx]**2 + dy[i_idx, j_idx]**2 - dr[i_idx, j_idx]**2])
    return c

def objective_func(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], 
                  centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(1e-9, lim)
        idx += 1
        
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], 
                         centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(1e-9, d)
            idx += 1
            
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, 
                      bounds=[(0.0, None)] * n, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-7)
    except Exception:
        pass
    return np.full(n, 0.01)

def hex_init(scale, angle):
    """Generates a hexagonal lattice initialization."""
    pts = []
    r = 0.10 * scale
    y = r
    row = 0
    while len(pts) < N + 5:
        x = r if row % 2 == 0 else 2.0 * r
        while x <= 1.0 - r + 0.05:
            pts.append([x, y])
            x += 2.0 * r
        y += np.sqrt(3.0) * r
        row += 1
        if y > 1.0 + r:
            break
            
    pts = np.array(pts[:N + 5])
    
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        cx, cy = 0.5, 0.5
        dx = pts[:, 0] - cx
        dy = pts[:, 1] - cy
        pts[:, 0] = dx * c - dy * s + cx
        pts[:, 1] = dx * s + dy * c + cy
        
    mask = (pts[:, 0] >= 0.01) & (pts[:, 0] <= 0.99) & \
           (pts[:, 1] >= 0.01) & (pts[:, 1] <= 0.99)
    pts = pts[mask]
    
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.2, 0.8, (1, 2))])
    return pts[:N]

def force_init(seed, steps=400):
    """Force-directed layout to spread points evenly."""
    np.random.seed(seed)
    pts = np.random.uniform(0.15, 0.85, (N, 2))
    for step in range(steps):
        f = np.zeros_like(pts)
        for i in range(N):
            for j in range(i + 1, N):
                dx = pts[j] - pts[i]
                d = np.hypot(dx[0], dx[1])
                if d < 0.25 and d > 1e-6:
                    rep = 0.015 / (d**2 + 0.001)
                    f[i] -= rep * dx
                    f[j] += rep * dx
            for dim in range(2):
                if pts[i, dim] < 0.12: f[i, dim] += 0.04
                elif pts[i, dim] > 0.88: f[i, dim] -= 0.04
        lr = 0.05 * (1.0 - step / steps)
        pts += f * lr
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def maximize_clearance(centers, radii):
    """Optimize centers to maximize minimum clearance t."""
    n = N
    x0 = np.concatenate([centers[:, 0], centers[:, 1], [0.0]])
    bounds = [(0.0, 1.0)]*(2*n) + [(-0.05, 0.5)]
    
    i_idx, j_idx = np.tril_indices(n, -1)
    
    def obj(v):
        return -v[-1]
        
    def cons(v):
        cx = v[:n]
        cy = v[n:2*n]
        t = v[2*n]
        c = np.concatenate([
            cx - radii - t,
            1.0 - cx - radii - t,
            cy - radii - t,
            1.0 - cy - radii - t
        ])
        dx = cx[:, None] - cx[None, :]
        dy = cy[:, None] - cy[None, :]
        dr = radii[:, None] + radii[None, :]
        c = np.concatenate([c, dx[i_idx, j_idx]**2 + dy[i_idx, j_idx]**2 - (dr[i_idx, j_idx] + t)**2])
        return c
        
    try:
        res = minimize(obj, x0, method='SLSQP', bounds=bounds, 
                       constraints={'type': 'ineq', 'fun': cons},
                       options={'maxiter': 5000, 'ftol': 1e-10, 'disp': False})
        if res.success and res.x[2*n] > 1e-5:
            return res.x[:2*n].reshape(n, 2), res.x[2*n]
    except Exception:
        pass
    return centers, 0.0

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    inits = []
    for scale in np.linspace(0.95, 1.15, 5):
        for ang in np.linspace(-0.2, 0.2, 5):
            inits.append(hex_init(scale, ang))
    for s in range(8):
        inits.append(force_init(s))
        
    for c_init in inits:
        centers = c_init.copy()
        radii = solve_lp_radii(centers)
        
        # Iterative growth: maximize clearance, increase radii, repeat
        for _ in range(15):
            new_centers, t = maximize_clearance(centers, radii)
            if t <= 1e-5:
                break
            radii += t
            centers = new_centers
            
        # Perturb and polish
        for _ in range(3):
            centers += np.random.normal(0, 0.001, centers.shape)
            centers = np.clip(centers, 0.01, 0.99)
            radii = solve_lp_radii(centers)
            
        # Joint optimization
        x0 = np.zeros(3 * N)
        x0[0::3] = centers[:, 0]
        x0[1::3] = centers[:, 1]
        x0[2::3] = radii
        
        for i in range(N):
            r = max(0.0, x0[3*i+2])
            x0[3*i] = np.clip(x0[3*i], r, 1.0 - r)
            x0[3*i+1] = np.clip(x0[3*i+1], r, 1.0 - r)
            x0[3*i+2] = r
            
        try:
            res = minimize(objective_func, x0, method='SLSQP', bounds=get_bounds(),
                           constraints={'type': 'ineq', 'fun': constraints_func},
                           options={'maxiter': 15000, 'ftol': 1e-12, 'disp': False})
            if not np.isnan(res.fun):
                c_opt = np.column_stack((res.x[0::3], res.x[1::3]))
                r_opt = solve_lp_radii(c_opt)
                curr_sum = np.sum(r_opt)
                if curr_sum > best_sum:
                    best_sum = curr_sum
                    best_centers = c_opt.copy()
                    best_radii = r_opt.copy()
        except Exception:
            pass
            
    if best_centers is None:
        best_centers = hex_init(1.0, 0.0)
        best_radii = solve_lp_radii(best_centers)
        best_sum = np.sum(best_radii)
        
    # Final strict validity adjustment
    for _ in range(100):
        valid = True
        for i in range(N):
            if best_radii[i] < 0 or best_centers[i, 0] < best_radii[i] - 1e-10 or best_centers[i, 0] > 1.0 - best_radii[i] + 1e-10 or \
               best_centers[i, 1] < best_radii[i] - 1e-10 or best_centers[i, 1] > 1.0 - best_radii[i] + 1e-10:
                valid = False; break
        if valid:
            for i in range(N):
                for j in range(i + 1, N):
                    if np.hypot(best_centers[i, 0] - best_centers[j, 0], best_centers[i, 1] - best_centers[j, 1]) < best_radii[i] + best_radii[j] - 1e-10:
                        valid = False; break
                if not valid: break
        if valid: break
            
        best_radii *= 0.9995
        best_centers[:, 0] = np.clip(best_centers[:, 0], best_radii, 1.0 - best_radii)
        best_centers[:, 1] = np.clip(best_centers[:, 1], best_radii, 1.0 - best_radii)
        
    return best_centers, best_radii, float(np.sum(best_radii))
