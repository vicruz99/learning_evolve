import numpy as np
from scipy.optimize import minimize, linprog

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def obj_jac(x):
    """Analytical gradient of the objective function."""
    jac = np.zeros(3 * N)
    jac[2::3] = -1.0
    return jac

def constraints(x):
    """Vectorized inequality constraints: g(x) >= 0."""
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    # Boundary constraints
    c = np.concatenate([
        cx - r,
        1.0 - cx - r,
        cy - r,
        1.0 - cy - r
    ])
    
    # Overlap constraints: dist^2 >= (ri + rj)^2
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dr = r[:, None] + r[None, :]
    
    c = np.concatenate([c, dx[TRIL_IDX]**2 + dy[TRIL_IDX]**2 - dr[TRIL_IDX]**2])
    return c

def con_jac(x):
    """Analytical Jacobian matrix of the constraint function."""
    n = N
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    J = np.zeros((4*n + n*(n-1)//2, 3*n))
    
    # Boundary constraints jacobian
    J[0:4*n:4, 0::3] = 1.0
    J[0:4*n:4, 2::3] = -1.0
    J[1:4*n:4, 0::3] = -1.0
    J[1:4*n:4, 2::3] = -1.0
    J[2:4*n:4, 1::3] = 1.0
    J[2:4*n:4, 2::3] = -1.0
    J[3:4*n:4, 1::3] = -1.0
    J[3:4*n:4, 2::3] = -1.0
    
    # Overlap constraints jacobian
    i_idx, j_idx = TRIL_IDX
    ni = len(i_idx)
    off = 4*n
    
    dx = cx[i_idx] - cx[j_idx]
    dy = cy[i_idx] - cy[j_idx]
    sr = r[i_idx] + r[j_idx]
    
    J[off:off+ni, 3*i_idx] = 2.0 * dx
    J[off:off+ni, 3*j_idx] = -2.0 * dx
    J[off:off+ni, 3*i_idx+1] = 2.0 * dy
    J[off:off+ni, 3*j_idx+1] = -2.0 * dy
    J[off:off+ni, 3*i_idx+2] = -2.0 * sr
    J[off:off+ni, 3*j_idx+2] = -2.0 * sr
    
    return J

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-7, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)])
    return b

def project_to_bounds(x):
    """Ensure optimization vector strictly respects boundary constraints."""
    x = x.copy()
    r = np.maximum(x[2::3], 1e-7)
    x[0::3] = np.clip(x[0::3], r, 1.0 - r)
    x[1::3] = np.clip(x[1::3], r, 1.0 - r)
    x[2::3] = r
    return x

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], 
                  centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(1e-9, lim)
        idx += 1
        
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], 
                         centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(1e-9, d)
            idx += 1
            
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, 
                      bounds=[(0.0, None)] * n, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-7)
    except Exception:
        pass
    return None

def hex_init(scale, angle):
    """Generates a hexagonal lattice initialization with rotation."""
    pts = []
    r = 0.09 * scale
    y = r
    row = 0
    while len(pts) < N + 5:
        x = r if row % 2 == 0 else 2.0 * r
        while x <= 1.0 - r + 0.05:
            pts.append([x, y])
            x += 2.0 * r
        y += np.sqrt(3.0) * r
        row += 1
        if y > 1.0 + r:
            break
            
    pts = np.array(pts[:N + 5])
    
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        cx, cy = 0.5, 0.5
        dx = pts[:, 0] - cx
        dy = pts[:, 1] - cy
        pts[:, 0] = dx * c - dy * s + cx
        pts[:, 1] = dx * s + dy * c + cy
        
    mask = (pts[:, 0] >= 0.01) & (pts[:, 0] <= 0.99) & \
           (pts[:, 1] >= 0.01) & (pts[:, 1] <= 0.99)
    pts = pts[mask]
    
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.2, 0.8, (1, 2))])
    return pts[:N]

def force_init(seed, steps=400):
    """Force-directed layout to spread points evenly and push to boundaries."""
    np.random.seed(seed)
    pts = np.random.uniform(0.15, 0.85, (N, 2))
    for step in range(steps):
        f = np.zeros_like(pts)
        for i in range(N):
            for j in range(i + 1, N):
                dx = pts[j] - pts[i]
                d = np.hypot(dx[0], dx[1])
                if d < 0.25 and d > 1e-6:
                    rep = 0.015 / (d**2 + 0.001)
                    f[i] -= rep * dx
                    f[j] += rep * dx
            for dim in range(2):
                if pts[i, dim] < 0.12: f[i, dim] += 0.04
                elif pts[i, dim] > 0.88: f[i, dim] -= 0.04
        lr = 0.05 * (1.0 - step / steps)
        pts += f * lr
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints, 'jac': con_jac}
    
    best_sum = -1.0
    best_x = None
    
    # Phase 1: Diverse Initializations
    inits = []
    for scale in np.linspace(0.85, 1.15, 7):
        for ang in np.linspace(-0.3, 0.3, 13):
            inits.append(hex_init(scale, ang))
            
    for s in range(15):
        inits.append(force_init(s))
        
    # Phase 2: Multi-start Optimization with Analytical Gradients
    for c_init in inits:
        r_init = solve_lp_radii(c_init)
        if r_init is None:
            r_init = np.full(N, 0.08)
            
        x0 = np.zeros(3 * N)
        x0[0::3] = c_init[:, 0]
        x0[1::3] = c_init[:, 1]
        x0[2::3] = r_init
        x0 = project_to_bounds(x0)
        x0 += np.random.normal(0, 1e-5, 3 * N)
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, 
                           constraints=cons, jac=obj_jac,
                           options={'maxiter': 25000, 'ftol': 1e-14, 'disp': False})
            if not np.isnan(res.fun):
                c_val = constraints(res.x)
                if np.min(c_val) >= -1e-6:
                    curr = -res.fun
                    if curr > best_sum:
                        best_sum = curr
                        best_x = res.x.copy()
        except Exception:
            continue
            
    # Phase 3: Deflation & Perturbation Refinement to escape local minima
    if best_x is not None:
        for step in range(40):
            x_p = best_x.copy()
            x_p[2::3] *= 0.965
            noise = 0.003 / (step + 1)
            x_p[0::3] += np.random.normal(0, noise, N)
            x_p[1::3] += np.random.normal(0, noise, N)
            x_p = project_to_bounds(x_p)
            
            try:
                res = minimize(objective, x_p, method='SLSQP', bounds=bounds,
                               constraints=cons, jac=obj_jac,
                               options={'maxiter': 15000, 'ftol': 1e-14, 'disp': False})
                if not np.isnan(res.fun):
                    c_val = constraints(res.x)
                    if np.min(c_val) >= -1e-6 and -res.fun > best_sum:
                        best_sum = -res.fun
                        best_x = res.x.copy()
            except Exception:
                pass
                
    # Fallback
    if best_x is None:
        c_fb = hex_init(1.0, 0.0)
        r_fb = solve_lp_radii(c_fb)
        if r_fb is None:
            r_fb = np.full(N, 0.08)
        best_x = np.zeros(3 * N)
        best_x[0::3] = c_fb[:, 0]
        best_x[1::3] = c_fb[:, 1]
        best_x[2::3] = r_fb
        best_sum = np.sum(r_fb)
        
    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3].copy()
    
    # Phase 4: LP Polish to extract maximum radii for final centers
    lp_r = solve_lp_radii(centers)
    if lp_r is not None:
        radii = lp_r
        best_sum = np.sum(radii)
        
    # Final strict validity adjustment against 1e-12 tolerance
    for _ in range(100):
        valid = True
        for i in range(N):
            if radii[i] < 0 or centers[i, 0] < radii[i] - 1e-10 or centers[i, 0] > 1.0 - radii[i] + 1e-10 or \
               centers[i, 1] < radii[i] - 1e-10 or centers[i, 1] > 1.0 - radii[i] + 1e-10:
                valid = False
                break
        if valid:
            for i in range(N):
                for j in range(i + 1, N):
                    d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                    if d < radii[i] + radii[j] - 1e-10:
                        valid = False
                        break
                if not valid:
                    break
        if valid:
            break
            
        radii *= 0.9998
        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)
        
    return centers, radii, float(np.sum(radii))
