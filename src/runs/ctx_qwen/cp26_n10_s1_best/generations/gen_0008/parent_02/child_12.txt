N = 26
TRIL_I, TRIL_J = np.tril_indices(N, -1)

def objective(x):

    return -np.sum(x[2::3])

def constraints(x):

    xs, ys, rs = x[0::3], x[1::3], x[2::3]

    c = np.concatenate([xs - rs, 1.0 - xs - rs, ys - rs, 1.0 - ys - rs])

    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    c = np.concatenate([c, dx[TRIL_I, TRIL_J]**2 + dy[TRIL_I, TRIL_J]**2 - dr[TRIL_I, TRIL_J]**2])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-7, 0.5]."""
    b = []
    for _ in range(N):

def solve_lp_radii(centers):

    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)

        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1

            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist)
            idx += 1

        pass
    return np.full(n, 0.06)

def project_to_bounds(x):

    x = x.copy()

        x[3 * i + 1] = np.clip(x[3 * i + 1], r, 1.0 - r)
        x[3 * i + 2] = r
    return x
