2. **Alternating Optimization**: Instead of optimizing centers and radii jointly from the start, we alternate between:
   - **Linear Programming (LP) for Radii**: For fixed centers, optimally compute the largest feasible radii. This is a convex problem solved exactly.
   - **Force-Directed Center Adjustment**: For fixed radii, push overlapping circles apart and toward boundaries using a fast, gradient-free force method. This quickly finds locally optimal center positions.
3. **Aggressive Topology Search**: To escape local minima, we periodically:
   - Shrink a random subset of radii to create slack.
   - Perturb their centers with cooling noise.
   - Swap center coordinates of random circles to break rigid contact networks.
   - Rotate the entire best configuration slightly and re-optimize, as boundary interactions often favor specific angles.
4. **High-Precision Polish**: After finding a strong candidate, run a final joint SLSQP optimization with tight tolerances, followed by an exact LP radii squeeze to maximize the sum.
5. **Strict Feasibility Repair**: Apply a minimal, iterative shrinkage and clipping routine at the end to guarantee compliance with the `1e-12` tolerance in the validation function.

This hybrid approach leverages the exactness of LP for radii, the robustness of force methods for local center adjustment, and the precision of SLSQP for final polishing, significantly increasing the chances of finding the global or near-global optimum.

import numpy as np
from scipy.optimize import minimize, linprog
import warnings
warnings.filterwarnings('ignore')

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Returns all inequality constraints >= 0 (vectorized)."""
    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    c = np.concatenate([
        xs - rs, 1.0 - xs - rs,
        ys - rs, 1.0 - ys - rs
    ])
    
    # Overlap constraints: dist >= r_i + r_j
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    # np.hypot provides stable gradients at contact points
    c = np.concatenate([c, np.hypot(dx[TRIL_IDX], dy[TRIL_IDX]) - dr[TRIL_IDX]])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-7, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)])
    return b

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist)
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-7)
    except Exception:
        pass
    return np.full(n, 0.05)

def project_x(x0):
    """Project variables to strictly satisfy bounds."""
    x0 = x0.copy()
    for i in range(N):
        r = max(1e-7, x0[3 * i + 2])
        x0[3 * i] = np.clip(x0[3 * i], r, 1.0 - r)
        x0[3 * i + 1] = np.clip(x0[3 * i + 1], r, 1.0 - r)
        x0[3 * i + 2] = r
    return x0

def force_centers(centers, radii, steps=120):
    """Force-directed relaxation to spread points evenly and push to boundaries."""
    pts = centers.copy()
    n = pts.shape[0]
    for step in range(steps):
        f = np.zeros_like(pts)
        lr = 0.015 * (1.0 - step / steps)
        for i in range(n):
            for j in range(i + 1, n):
                dx = pts[j] - pts[i]
                d = np.linalg.norm(dx)
                min_d = radii[i] + radii[j]
                if d < min_d * 1.3 and d > 1e-6:
                    rep = (min_d - d) * 3.0 / d
                    f[i] -= dx * rep
                    f[j] += dx * rep
            for dim in range(2):
                if pts[i, dim] < radii[i] + 0.015:
                    f[i, dim] += 0.06
                elif pts[i, dim] > 1.0 - radii[i] - 0.015:
                    f[i, dim] -= 0.06
        pts += f * lr
        pts = np.clip(pts, 0.001, 0.999)
    return pts

def make_hex_init(angle=0.0, scale=1.0):
    """Generates a rotated hexagonal lattice initialization."""
    pts = []
    r0 = 0.09 * scale
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0 and len(pts) < N + 10:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
    pts = np.array(pts[:N + 10])
    
    if angle != 0.0:
        c_a, s_a = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c_a, -s_a], [s_a, c_a]]) + 0.5
        
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.1, 0.9, (1, 2))])
        
    return np.clip(pts[:N] + np.random.uniform(-0.002, 0.002, (N, 2)), 0.02, 0.98)

def make_corner_init():
    """Generates an initialization focused on corners and edges."""
    pts = np.zeros((N, 2))
    r = 0.1
    pts[0] = [r, r]
    pts[1] = [1-r, r]
    pts[2] = [r, 1-r]
    pts[3] = [1-r, 1-r]
    idx = 4
    y = r + 0.06
    row = 0
    while idx < N and y + r <= 1.0:
        x = r + 0.05 if row % 2 == 0 else r + 0.15
        while idx < N and x + r <= 1.0:
            pts[idx] = [x, y]
            idx += 1
            x += 0.20
        y += 0.14
        row += 1
    for i in range(idx, N):
        pts[i] = np.random.uniform(0.1, 0.9, 2)
    return np.clip(pts + np.random.uniform(-0.003, 0.003, pts.shape), 0.02, 0.98)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    best_sum = -1.0
    best_x = None
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    
    # Phase 1: Diverse Initial Configurations
    inits = []
    # Rotated hexagonal lattices
    for ang in np.linspace(-0.35, 0.35, 11):
        for sc in [0.95, 1.0, 1.05]:
            inits.append(make_hex_init(ang, sc))
    # Corner/Edge focused
    for _ in range(10):
        inits.append(make_corner_init())
    # Random + Force spread
    for s in range(8):
        np.random.seed(s + 100)
        pts = np.random.uniform(0.1, 0.9, (N, 2))
        r_init = np.full(N, 0.07)
        pts = force_centers(pts, r_init, steps=100)
        inits.append(pts)

    # Phase 2: Multi-start Optimization with LP-Force Alternation
    for pts in inits:
        c = pts.copy()
        r = solve_lp_radii(c)
        
        # Alternating refinement
        for _ in range(4):
            c = force_centers(c, r, steps=80)
            r = solve_lp_radii(c)

        x0 = np.zeros(3 * N)
        x0[0::3] = c[:, 0]
        x0[1::3] = c[:, 1]
        x0[2::3] = r * 0.995
        x0 = project_x(x0)

        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 15000, 'ftol': 1e-13, 'disp': False})
            if not np.isnan(res.fun):
                c_tmp = np.column_stack((res.x[0::3], res.x[1::3]))
                r_new = solve_lp_radii(c_tmp)
                s_val = np.sum(r_new)
                if s_val > best_sum:
                    best_sum = s_val
                    best_x = res.x.copy()
                    best_x[2::3] = r_new
        except Exception:
            pass

    # Phase 3: Aggressive Topology Search with Deflation, Swaps & Rotation
    if best_x is not None:
        for cyc in range(120):
            x0 = best_x.copy()
            subset = np.random.choice(N, size=N // 3, replace=False)
            noise = 0.003 / (cyc + 1)
            x0[subset * 3] += np.random.normal(0, noise, N // 3)
            x0[subset * 3 + 1] += np.random.normal(0, noise, N // 3)
            x0[subset * 3 + 2] *= 0.65

            # Random swap to break rigid contact networks
            if cyc % 5 == 0 and cyc > 0:
                i, j = np.random.choice(N, 2, replace=False)
                x0[[i*3, j*3]] = x0[[j*3, i*3]]
                x0[[i*3+1, j*3+1]] = x0[[j*3+1, i*3+1]]
                
            x0 = project_x(x0)

            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 8000, 'ftol': 1e-13, 'disp': False})
                if not np.isnan(res.fun):
                    c_tmp = np.column_stack((res.x[0::3], res.x[1::3]))
                    r_new = solve_lp_radii(c_tmp)
                    x_p = res.x.copy()
                    x_p[2::3] = r_new * 0.999
                    x_p = project_x(x_p)
                    s_val = np.sum(x_p[2::3])
                    if np.min(constraints(x_p)) >= -1e-5 and s_val > best_sum:
                        best_sum = s_val
                        best_x = x_p.copy()
            except Exception:
                pass

        # Global Rotation Search on best configuration
        best_c = np.column_stack((best_x[0::3], best_x[1::3]))
        for ang_deg in np.linspace(-12, 12, 13):
            ang = np.deg2rad(ang_deg)
            c_a, s_a = np.cos(ang), np.sin(ang)
            rot_c = (best_c - 0.5) @ np.array([[c_a, -s_a], [s_a, c_a]]) + 0.5
            rot_c = np.clip(rot_c, 0.01, 0.99)
            r_rot = solve_lp_radii(rot_c)
            
            x_rot = np.zeros(3 * N)
            x_rot[0::3] = rot_c[:, 0]
            x_rot[1::3] = rot_c[:, 1]
            x_rot[2::3] = r_rot * 0.99
            x_rot = project_x(x_rot)
            
            try:
                res = minimize(objective, x_rot, method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 6000, 'ftol': 1e-13, 'disp': False})
                if not np.isnan(res.fun):
                    c_tmp = np.column_stack((res.x[0::3], res.x[1::3]))
                    r_new = solve_lp_radii(c_tmp)
                    s_val = np.sum(r_new)
                    if s_val > best_sum:
                        best_sum = s_val
                        best_x = res.x.copy()
                        best_x[2::3] = r_new
            except Exception:
                pass

    # Fallback initialization
    if best_x is None:
        best_x = np.zeros(3 * N)
        best_x[0::3] = np.tile(np.linspace(0.1, 0.9, 5), 6)[:N]
        best_x[1::3] = np.repeat(np.linspace(0.1, 0.9, 6), 5)[:N]
        best_x[2::3] = 0.06
        best_sum = np.sum(best_x[2::3])

    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3].copy()
    
    # Final LP squeeze to maximize radii for the best centers found
    final_r = solve_lp_radii(centers)
    if np.sum(final_r) > np.sum(radii) - 1e-7:
        radii = final_r.copy()

    # Final strict validation repair against 1e-12 tolerance
    for _ in range(100):
        valid = True
        for i in range(N):
            if radii[i] < 0 or centers[i, 0] < radii[i] - 1e-9 or centers[i, 0] > 1.0 - radii[i] + 1e-9 or \
               centers[i, 1] < radii[i] - 1e-9 or centers[i, 1] > 1.0 - radii[i] + 1e-9:
                valid = False
                break
        if valid:
            for i in range(N):
                for j in range(i + 1, N):
                    d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                    if d < radii[i] + radii[j] - 1e-9:
                        valid = False
                        break
                if not valid:
                    break
        if valid:
            break
            
        radii *= 0.9995
        for i in range(N):
            centers[i, 0] = np.clip(centers[i, 0], radii[i], 1.0 - radii[i])
            centers[i, 1] = np.clip(centers[i, 1], radii[i], 1.0 - radii[i])

    return centers, radii, float(np.sum(radii))
