import numpy as np
from scipy.optimize import minimize, linprog
import warnings
warnings.filterwarnings('ignore')

N = 26

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist)
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-7)
    except Exception:
        pass
    return np.full(n, 0.05)

def make_lattice(angle, scale, tx, ty):
    """Generates a rotated and translated hexagonal lattice."""
    pts = []
    r0 = 0.09 * scale
    y = r0 + ty
    row = 0
    while len(pts) < N + 10:
        x = r0 + tx if row % 2 == 0 else 2.0 * r0 + tx
        while x <= 1.0 - r0 and len(pts) < N + 10:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
    pts = np.array(pts[:N + 10])
    
    if abs(angle) > 1e-6:
        c_a, s_a = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c_a, -s_a], [s_a, c_a]]) + 0.5
        
    mask = (pts[:, 0] >= 0.01) & (pts[:, 0] <= 0.99) & (pts[:, 1] >= 0.01) & (pts[:, 1] <= 0.99)
    valid_pts = pts[mask]
    if len(valid_pts) < N:
        pad = N - len(valid_pts)
        valid_pts = np.vstack([valid_pts, np.random.uniform(0.1, 0.9, (pad, 2))])
    return valid_pts[:N]

def objective_slsqp(x):
    return -np.sum(x[2::3])

def constraints_slsqp(x):
    xs, ys, rs = x[0::3], x[1::3], x[2::3]
    c = np.concatenate([xs - rs, 1.0 - xs - rs, ys - rs, 1.0 - ys - rs])
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    tril = np.tril_indices(N, -1)
    c = np.concatenate([c, np.hypot(dx[tril], dy[tril]) - dr[tril]])
    return c

def bounds_slsqp():
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)])
    return b

def project_x(x0):
    x0 = x0.copy()
    for i in range(N):
        r = max(1e-7, x0[3 * i + 2])
        x0[3 * i] = np.clip(x0[3 * i], r, 1.0 - r)
        x0[3 * i + 1] = np.clip(x0[3 * i + 1], r, 1.0 - r)
        x0[3 * i + 2] = r
    return x0

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    best_sum = -1.0
    best_x = None
    bounds_obj = bounds_slsqp()
    cons = {'type': 'ineq', 'fun': constraints_slsqp}

    # Phase 1: Random search over lattice parameters to find strong basins
    init_configs = []
    for _ in range(300):
        ang = np.random.uniform(-0.6, 0.6)

        r_lp = solve_lp_radii(pts)
        s = np.sum(r_lp)
        init_configs.append((s, pts))
        
    # Tuples sort by first element automatically. Sort descending.
    init_configs.sort(reverse=True)
    best_init_configs = [cfg[1] for cfg in init_configs[:15]]

                    d = np.linalg.norm(dx)
                    if d < 0.32 and d > 1e-5:

                        f[i] -= dx * rep / d

                    elif pts[i, dim] > 0.88: f[i, dim] -= 0.05
            pts += f * lr

                    best_x[2::3] = r_new
        except Exception:
            pass

    # Phase 3: Coordinate Descent / Local Search to escape rigid contacts
    if best_x is not None:

        curr_radii = best_x[2::3].copy()

            improved = False

                            new_centers[i, 0] = np.clip(new_centers[i, 0] + rx, 0.01, 0.99)
                            new_centers[i, 1] = np.clip(new_centers[i, 1] + ry, 0.01, 0.99)
                            
                            r_new = solve_lp_radii(new_centers)
                            s_new = np.sum(r_new)
                            if s_new > best_move_sum + 1e-7:
                                best_move_sum = s_new
                                best_move_centers = new_centers
                                best_move_radii = r_new
                                
                if best_move_sum > base_sum + 1e-7:
                    curr_centers = best_move_centers
                    curr_radii = best_move_radii
                    improved = True
                    
        # Update best if coordinate descent improved
        if np.sum(curr_radii) > best_sum:
            best_sum = np.sum(curr_radii)
            best_x = np.zeros(3 * N)
            best_x[0::3] = curr_centers[:, 0]
            best_x[1::3] = curr_centers[:, 1]
            best_x[2::3] = curr_radii

                x0[[i*3+1, j*3+1]] = x0[[j*3+1, i*3+1]]

                    x_p = res.x.copy()
                    x_p[2::3] = r_new * 0.999
                    x_p = project_x(x_p)
                    s_val = np.sum(x_p[2::3])
                    if np.min(constraints_slsqp(x_p)) >= -1e-6 and s_val > best_sum:
                        best_sum = s_val
                        best_x = x_p.copy()
            except Exception:
                pass

    # Phase 5: High-Precision Polish
    if best_x is not None:

            x0 = best_x.copy()

            centers[i, 0] = np.clip(centers[i, 0], radii[i], 1.0 - radii[i])
            centers[i, 1] = np.clip(centers[i, 1], radii[i], 1.0 - radii[i])

    return centers, radii, float(np.sum(radii))
