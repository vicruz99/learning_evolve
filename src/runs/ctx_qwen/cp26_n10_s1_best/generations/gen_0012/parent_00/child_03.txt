1. **High-Quality Initializations**: Generate configurations based on known dense hexagonal patterns (5-6-5-6-4, etc.) with varied rotations and scales, combined with force-spread layouts.
2. **Alternating Optimization**: Use a hybrid approach alternating between Linear Programming (to optimally compute radii for fixed centers) and L-BFGS-B with a smooth penalty function (to adjust centers for fixed radii). This decouples the non-smooth contact constraints and finds better local topologies.
3. **Squared-Distance SLSQP Polish**: Switch to squared-distance constraints for SLSQP to improve gradient conditioning near contact points, followed by LP radius updates.
4. **Aggressive Deflation & Regrowth**: Iteratively shrink radii, perturb centers, and re-optimize to break rigid contact networks and explore new packing topologies.
5. **Coordinate Descent Polish**: Fine-tune circle positions by testing small discrete displacements and re-solving the LP for radii. This effectively squeezes out remaining slack and is highly robust against shallow local minima.
6. **Strict Validation Repair**: Ensure final output strictly satisfies the 1e-12 tolerance by minimal uniform shrinkage if numerical drift occurs.

import numpy as np
from scipy.optimize import minimize, linprog
import warnings
warnings.filterwarnings('ignore')

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-6, 0.5]."""
    return [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N

def constraints_slsqp(x):
    """Inequality constraints for SLSQP: g(x) >= 0 using squared distances."""
    xs, ys, rs = x[0::3], x[1::3], x[2::3]
    c = np.concatenate([xs - rs, 1.0 - xs - rs, ys - rs, 1.0 - ys - rs])
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    c = np.concatenate([c, dx[TRIL_IDX]**2 + dy[TRIL_IDX]**2 - dr[TRIL_IDX]**2])
    return c

def objective_func(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist)
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-7)
    except Exception:
        pass
    return np.full(n, 0.05)

def penalty_centers(c_flat, radii):
    """Smooth penalty function for centers given fixed radii."""
    c = c_flat.reshape(-1, 2)
    p = 0.0
    x, y = c[:, 0], c[:, 1]
    r = radii
    p += np.sum(np.maximum(r - x, 0.0)**2)
    p += np.sum(np.maximum(x - (1.0 - r), 0.0)**2)
    p += np.sum(np.maximum(r - y, 0.0)**2)
    p += np.sum(np.maximum(y - (1.0 - r), 0.0)**2)
    
    for i in range(N):
        for j in range(i + 1, N):
            dx = c[j, 0] - c[i, 0]
            dy = c[j, 1] - c[i, 1]
            d = np.hypot(dx, dy)
            gap = d - r[i] - r[j]
            if gap < 0.0:
                p += gap**2
    return p

def alternating_opt(centers, radii, n_iter=6):
    """Alternating LP (radii) and L-BFGS-B (centers) optimization."""
    c = centers.copy()
    r = radii.copy()
    bounds_c = [(0.001, 0.999)] * (2 * N)
    
    for _ in range(n_iter):
        r = solve_lp_radii(c)
        res = minimize(penalty_centers, c.flatten(), args=(r,), method='L-BFGS-B', 
                       bounds=bounds_c, options={'maxiter': 1000, 'ftol': 1e-14})
        c = res.x.reshape(-1, 2)
        c = np.clip(c, 0.005, 0.995)
    return c, r

def make_hex_init(rows_pattern, scale=1.0, angle=0.0):
    """Generates a rotated hexagonal lattice initialization."""
    pts = []
    r = 0.10 * scale
    dy = np.sqrt(3.0) * r
    y = r
    for row_idx, ncols in enumerate(rows_pattern):
        x = r if row_idx % 2 == 0 else 2.0 * r
        for _ in range(ncols):
            pts.append([x, y])
            x += 2.0 * r
        y += dy
    pts = np.array(pts[:N])
    
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
        
    return np.clip(pts, 0.02, 0.98)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    bounds_obj = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints_slsqp}
    
    best_sum = -1.0
    best_x = None
    
    # Phase 1: Generate Diverse Initial Configurations
    inits = []
    patterns = [[5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [5, 5, 5, 5, 6], [6, 6, 5, 5, 4]]
    for pat in patterns:
        for ang in np.linspace(-0.25, 0.25, 5):
            inits.append(make_hex_init(pat, scale=1.0, angle=ang))
        for sc in [0.95, 1.02]:
            inits.append(make_hex_init(pat, scale=sc))
            
    for s in range(12):
        np.random.seed(s + 50)
        pts = np.random.uniform(0.15, 0.85, (N, 2))
        for _ in range(150):
            f = np.zeros_like(pts)
            for i in range(N):
                for j in range(i + 1, N):
                    dx = pts[j] - pts[i]
                    d = np.hypot(dx[0], dx[1])
                    if d < 0.3 and d > 1e-5:
                        rep = 0.012 / (d**2 + 0.001)
                        f[i] -= dx * rep / d
                        f[j] += dx * rep / d
                if pts[i, 0] < 0.1: f[i, 0] += 0.05
                elif pts[i, 0] > 0.9: f[i, 0] -= 0.05
                if pts[i, 1] < 0.1: f[i, 1] += 0.05
                elif pts[i, 1] > 0.9: f[i, 1] -= 0.05
            pts += f * 0.05
            pts = np.clip(pts, 0.02, 0.98)
        inits.append(pts)

    # Phase 2: Multi-start Optimization (Alternating + SLSQP Polish)
    for pts in inits:
        c_curr, r_curr = alternating_opt(pts, np.full(N, 0.07), n_iter=5)
        x0 = np.zeros(3 * N)
        x0[0::3] = c_curr[:, 0]
        x0[1::3] = c_curr[:, 1]
        r_lp = solve_lp_radii(c_curr)
        x0[2::3] = r_lp * 0.995
        
        for i in range(N):
            r_val = max(1e-6, x0[3*i+2])
            x0[3*i] = np.clip(x0[3*i], r_val, 1.0-r_val)
            x0[3*i+1] = np.clip(x0[3*i+1], r_val, 1.0-r_val)
            x0[3*i+2] = r_val

        try:
            res = minimize(objective_func, x0, method='SLSQP', bounds=bounds_obj, constraints=cons,
                           options={'maxiter': 20000, 'ftol': 1e-13, 'disp': False})
            if not np.isnan(res.fun):
                c_tmp = np.column_stack((res.x[0::3], res.x[1::3]))
                r_new = solve_lp_radii(c_tmp)
                s_val = np.sum(r_new)
                if s_val > best_sum:
                    best_sum = s_val
                    best_x = np.zeros(3*N)
                    best_x[0::3] = c_tmp[:, 0]
                    best_x[1::3] = c_tmp[:, 1]
                    best_x[2::3] = r_new
        except Exception:
            pass

    # Phase 3: Aggressive Topology Search (Deflation & Repositioning)
    if best_x is not None:
        for cyc in range(80):
            x0 = best_x.copy()
            noise = 0.0025 * (0.91 ** cyc)
            x0[0::3] += np.random.normal(0, noise, N)
            x0[1::3] += np.random.normal(0, noise, N)
            x0[2::3] *= 0.93
            
            subset = np.random.choice(N, size=max(3, N//4), replace=False)
            x0[subset * 3 + 2] *= 0.75
            
            for i in range(N):
                r_val = max(1e-6, x0[3*i+2])
                x0[3*i] = np.clip(x0[3*i], r_val, 1.0-r_val)
                x0[3*i+1] = np.clip(x0[3*i+1], r_val, 1.0-r_val)
                x0[3*i+2] = r_val

            try:
                res = minimize(objective_func, x0, method='SLSQP', bounds=bounds_obj, constraints=cons,
                               options={'maxiter': 12000, 'ftol': 1e-13, 'disp': False})
                if not np.isnan(res.fun):
                    c_tmp = np.column_stack((res.x[0::3], res.x[1::3]))
                    r_new = solve_lp_radii(c_tmp)
                    if r_new is not None:
                        x_p = res.x.copy()
                        x_p[2::3] = r_new
                        s_val = np.sum(r_new)
                        if s_val > best_sum and np.min(constraints_slsqp(x_p)) >= -1e-6:
                            best_sum = s_val
                            best_x = x_p.copy()
            except Exception:
                pass

        # Phase 4: Coordinate Descent Polish
        curr_c = np.column_stack((best_x[0::3], best_x[1::3]))
        curr_r = best_x[2::3].copy()
        improved = True
        while improved:
            improved = False
            for i in range(N):
                best_local_sum = np.sum(curr_r)
                for dx in [-0.004, -0.001, 0.001, 0.004]:
                    for dy in [-0.004, -0.001, 0.001, 0.004]:
                        if dx == 0 and dy == 0: continue
                        new_c = curr_c.copy()
                        new_c[i, 0] = np.clip(new_c[i, 0] + dx, 0.01, 0.99)
                        new_c[i, 1] = np.clip(new_c[i, 1] + dy, 0.01, 0.99)
                        r_new = solve_lp_radii(new_c)
                        if r_new is not None:
                            s_new = np.sum(r_new)
                            if s_new > best_local_sum + 1e-8:
                                curr_c = new_c
                                curr_r = r_new
                                best_local_sum = s_new
                                improved = True
                                break
                    if improved: break
        best_x[0::3] = curr_c[:, 0]
        best_x[1::3] = curr_c[:, 1]
        best_x[2::3] = curr_r
        best_sum = np.sum(curr_r)

    # Fallback (should not be reached)
    if best_x is None:
        best_x = np.zeros(3*N)
        best_x[2::3] = 0.06
        best_x[0::3] = np.tile(np.linspace(0.1, 0.9, 5), 6)[:N]
        best_x[1::3] = np.repeat(np.linspace(0.1, 0.9, 6), 5)[:N]
        best_sum = np.sum(best_x[2::3])

    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3].copy()

    # Final strict validity adjustment against 1e-12 tolerance
    for _ in range(100):
        valid = True
        for i in range(N):
            if radii[i] < 0 or centers[i, 0] < radii[i] - 1e-9 or centers[i, 0] > 1.0 - radii[i] + 1e-9 or \
               centers[i, 1] < radii[i] - 1e-9 or centers[i, 1] > 1.0 - radii[i] + 1e-9:
                valid = False
                break
        if valid:
            for i in range(N):
                for j in range(i + 1, N):
                    d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                    if d < radii[i] + radii[j] - 1e-9:
                        valid = False
                        break
                if not valid:
                    break
        if valid:
            break
            
        # Minimal shrinkage to recover strict feasibility
        radii *= 0.9999
        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)

    return centers, radii, float(np.sum(radii))
