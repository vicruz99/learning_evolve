

<strategy>
1. **Diverse Initialization**: Generate multiple starting configurations using rotated hexagonal lattices, corner/edge-seeded layouts, and force-directed random spreads. This ensures coverage of different topological basins.
2. **Robust Alternating Optimization**: Implement a tight loop alternating between:
   - **Linear Programming (LP)**: Maximizes the sum of radii for fixed centers, strictly respecting overlap and boundary constraints.
   - **Penalty-Based Center Optimization (L-BFGS-B)**: Minimizes a smooth penalty function for overlaps and boundary violations given fixed radii. This allows centers to slide efficiently into tight packings without getting stuck on hard constraint gradients.
   This cycle repeats 40-50 times per start to converge to a stable local topology.
3. **SLSQP Joint Polish**: Use the converged alternating result as a seed for `SLSQP` to jointly optimize centers and radii with hard constraints, squeezing out extra precision.
4. **Deflation & Basin Hopping**: To escape the local optimum around 2.634, implement an aggressive deflation routine: randomly shrink subsets of radii, perturb centers, and re-run the optimization pipeline. This breaks rigid contact networks and allows the solver to explore new packings.
5. **Final Validation & Repair**: Apply a strict validation check with minimal uniform shrinkage if necessary to guarantee compliance with the `1e-12` tolerance in the validation function.

import numpy as np
from scipy.optimize import minimize, linprog
import warnings

warnings.filterwarnings('ignore')

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def get_bounds():
    return [(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)] * N

def solve_lp_radii(centers):
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist)
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-8)
    except Exception:
        pass
    return np.full(n, 0.04)

def center_penalty(c_flat, radii):
    c = c_flat.reshape(-1, 2)
    p = 0.0
    n = c.shape[0]
    r = radii
    x, y = c[:, 0], c[:, 1]
    p += np.sum(np.maximum(r - x, 0.0)**2)
    p += np.sum(np.maximum(x - (1.0 - r), 0.0)**2)
    p += np.sum(np.maximum(r - y, 0.0)**2)
    p += np.sum(np.maximum(y - (1.0 - r), 0.0)**2)
    
    dx = c[:, 0:1] - c[None, :, 0]
    dy = c[:, 1:2] - c[None, :, 1]
    dist = np.hypot(dx, dy)
    sum_r = r[:, None] + r[None, :]
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    gaps = dist[mask] - sum_r[mask]
    p += np.sum(np.maximum(-gaps, 0.0)**2)
    return p

def alternating_opt(centers, radii, n_iter=40):
    c = centers.copy()
    r = radii.copy()
    bounds_c = [(0.001, 0.999)] * (2 * N)
    
    for it in range(n_iter):
        r = solve_lp_radii(c)
        c = np.clip(c, r, 1.0 - r)
        
        res = minimize(center_penalty, c.flatten(), args=(r,), 
                       method='L-BFGS-B', bounds=bounds_c, 
                       options={'maxiter': 500, 'ftol': 1e-15})
        c = res.x.reshape(-1, 2)
        c = np.clip(c, 0.005, 0.995)
        
    return c, r

def constraints_slsqp(x):
    xs, ys, rs = x[0::3], x[1::3], x[2::3]
    c = np.concatenate([xs - rs, 1.0 - xs - rs, ys - rs, 1.0 - ys - rs])
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    c = np.concatenate([c, np.hypot(dx[TRIL_IDX], dy[TRIL_IDX]) - dr[TRIL_IDX]])
    return c

def make_hex_init(r0, angle):
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0 and len(pts) < N + 10:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        
    pts = np.array(pts[:N + 10])
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
        
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.15, 0.85, (1, 2))])
    return pts[:N]

def make_corner_init():
    pts = np.zeros((N, 2))
    r = 0.09
    pts[0] = [r, r]
    pts[1] = [1-r, r]
    pts[2] = [r, 1-r]
    pts[3] = [1-r, 1-r]
    pts[4] = [0.5, r]
    pts[5] = [0.5, 1-r]
    pts[6] = [r, 0.5]
    pts[7] = [1-r, 0.5]
    pts[8] = [0.5, 0.5]
    idx = 9
    
    y = r + 0.04
    row = 0
    while idx < N and y + r <= 1.0:
        x = r + 0.04 if row % 2 == 0 else r + 0.16
        while idx < N and x + r <= 1.0:
            pts[idx] = [x, y]
            idx += 1
            x += 0.20
        y += 0.13
        row += 1
        
    for i in range(idx, N):
        pts[i] = np.random.uniform(0.1, 0.9, 2)
    pts += np.random.uniform(-0.002, 0.002, pts.shape)
    return pts

def make_force_init(seed):
    np.random.seed(seed)
    pts = np.random.uniform(0.15, 0.85, (N, 2))
    for step in range(300):
        f = np.zeros_like(pts)
        lr = 0.02 * (1.0 - step / 300.0)
        for i in range(N):
            for j in range(i + 1, N):
                dx = pts[j] - pts[i]
                d = np.hypot(dx[0], dx[1])
                if d < 0.3 and d > 1e-5:
                    rep = 0.015 / (d**2 + 0.001)
                    f[i] -= dx * rep / d
                    f[j] += dx * rep / d
            for dim in range(2):
                if pts[i, dim] < 0.08: f[i, dim] += 0.06
                elif pts[i, dim] > 0.92: f[i, dim] -= 0.06
        pts += f * lr
        pts = np.clip(pts, 0.03, 0.97)
    return pts

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    bounds_obj = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints_slsqp}
    
    best_sum = -1.0
    best_x = None
    
    inits = []
    for r0 in [0.085, 0.090, 0.095, 0.100]:
        for ang in np.linspace(-0.3, 0.3, 9):
            inits.append(make_hex_init(r0, ang))
    for s in range(10):
        inits.append(make_force_init(s))
    for _ in range(6):
        inits.append(make_corner_init())

    for pts in inits:
        c_curr, r_curr = pts.copy(), np.full(N, 0.07)
        c_curr, r_curr = alternating_opt(c_curr, r_curr, n_iter=40)
        
        x0 = np.zeros(3 * N)
        x0[0::3] = c_curr[:, 0]
        x0[1::3] = c_curr[:, 1]
        x0[2::3] = np.maximum(r_curr * 0.998, 1e-6)
        
        try:
            res = minimize(lambda x: -np.sum(x[2::3]), x0, method='SLSQP', 
                           bounds=bounds_obj, constraints=cons,
                           options={'maxiter': 20000, 'ftol': 1e-14, 'disp': False})
            if not np.isnan(res.fun):
                curr_sum = -res.fun
                if curr_sum > best_sum and np.min(constraints_slsqp(res.x)) >= -1e-6:
                    best_sum = curr_sum
                    best_x = res.x.copy()
        except Exception:
            continue

    if best_x is not None:
        for cyc in range(80):
            x0 = best_x.copy()
            noise = 0.0025 * (0.91 ** cyc)
            x0[0::3] += np.random.normal(0, noise, N)
            x0[1::3] += np.random.normal(0, noise, N)
            
            k = np.random.randint(3, 7)
            subset = np.random.choice(N, size=k, replace=False)
            x0[subset * 3 + 2] *= 0.80
            
            for i in range(N):
                r = max(1e-6, x0[3 * i + 2])
                x0[3 * i] = np.clip(x0[3 * i], r, 1.0 - r)
                x0[3 * i + 1] = np.clip(x0[3 * i + 1], r, 1.0 - r)
                x0[3 * i + 2] = r

            try:
                res = minimize(lambda x: -np.sum(x[2::3]), x0, method='SLSQP', 
                               bounds=bounds_obj, constraints=cons,
                               options={'maxiter': 12000, 'ftol': 1e-13, 'disp': False})
                if not np.isnan(res.fun):
                    centers_p = np.column_stack((res.x[0::3], res.x[1::3]))
                    r_p = solve_lp_radii(centers_p)
                    
                    x_p = res.x.copy()
                    x_p[2::3] = r_p
                    for i in range(N):
                        x_p[3 * i] = np.clip(x_p[3 * i], r_p[i], 1.0 - r_p[i])
                        x_p[3 * i + 1] = np.clip(x_p[3 * i + 1], r_p[i], 1.0 - r_p[i])
                        
                    curr_sum = np.sum(r_p)
                    if curr_sum > best_sum and np.min(constraints_slsqp(x_p)) >= -1e-6:
                        best_sum = curr_sum
                        best_x = x_p.copy()
            except Exception:
                pass

    if best_x is None:
        best_x = np.zeros(3 * N)
        best_x[0::3] = np.tile(np.linspace(0.1, 0.9, 5), 6)[:N]
        best_x[1::3] = np.repeat(np.linspace(0.1, 0.9, 6), 5)[:N]
        best_x[2::3] = 0.06
        best_sum = np.sum(best_x[2::3])

    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3].copy()

    for _ in range(100):
        valid = True
        for i in range(N):
            if radii[i] < 0 or centers[i, 0] < radii[i] - 1e-9 or centers[i, 0] > 1.0 - radii[i] + 1e-9 or \
               centers[i, 1] < radii[i] - 1e-9 or centers[i, 1] > 1.0 - radii[i] + 1e-9:
                valid = False
                break
        if valid:
            for i in range(N):
                for j in range(i + 1, N):
                    d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                    if d < radii[i] + radii[j] - 1e-9:
                        valid = False
                        break
                if not valid:
                    break
        if valid:
            break
            
        radii *= 0.9998
        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)

    return centers, radii, float(np.sum(radii))
