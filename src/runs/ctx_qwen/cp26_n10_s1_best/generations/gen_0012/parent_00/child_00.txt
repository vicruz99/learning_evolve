import numpy as np
from scipy.optimize import minimize, linprog
import warnings
warnings.filterwarnings('ignore')

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def get_bounds_obj():
    """Variable bounds: x,y in [0,1], r in [1e-7, 0.5]."""
    return [(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)] * N

def constraints_slsqp(x):
    """Inequality constraints for SLSQP: g(x) >= 0."""
    xs, ys, rs = x[0::3], x[1::3], x[2::3]
    c = np.concatenate([xs - rs, 1.0 - xs - rs, ys - rs, 1.0 - ys - rs])
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    c = np.concatenate([c, np.hypot(dx[TRIL_IDX], dy[TRIL_IDX]) - dr[TRIL_IDX]])
    return c

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    for i in range(n):
        x, y = centers[i]
        lim = min(x, 1.0 - x, y, 1.0 - y)
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    dx = centers[:, 0, None] - centers[:, 0]
    dy = centers[:, 1, None] - centers[:, 1]
    dists = np.hypot(dx[TRIL_IDX], dy[TRIL_IDX])
    
    rows, cols = TRIL_IDX
    A_ub[idx:, rows] = 1.0
    A_ub[idx:, cols] = 1.0
    b_ub[idx:] = dists
    
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-8)
    except Exception:
        pass
    return np.full(n, 0.02)

def hill_climb_centers(centers, steps=2500):
    """Coordinate-wise hill climbing on centers using LP for radii evaluation."""
    c = centers.copy()
    r = solve_lp_radii(c)
    best_sum = np.sum(r)
    step = 0.018
    
    for it in range(steps):
        # Adaptive step cooling
        step *= 0.9995
        
        # Random coordinate update
        idx = np.random.randint(N)
        c_new = c.copy()
        c_new[idx] += np.random.normal(0, step, 2)
        c_new[idx] = np.clip(c_new[idx], 0.005, 0.995)
        
        # Topology breaker: occasional swap
        if np.random.random() < 0.05:
            idx2 = np.random.randint(N)
            while idx2 == idx:
                idx2 = np.random.randint(N)
            c_new[[idx, idx2]] = c_new[[idx2, idx]]
            
        r_new = solve_lp_radii(c_new)
        s_new = np.sum(r_new)
        
        if s_new > best_sum + 1e-7:
            c = c_new
            best_sum = s_new
            step = min(0.025, step * 1.05)
            
    return c, best_sum

def make_hex_init(r0, angle):
    """Generates a rotated hexagonal lattice initialization."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0 and len(pts) < N + 10:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        
    pts = np.array(pts[:N + 10])
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
        
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.15, 0.85, (1, 2))])
    return pts[:N]

def make_force_init(seed):
    """Force-directed layout to spread points evenly and push to boundaries."""
    np.random.seed(seed)
    pts = np.random.uniform(0.15, 0.85, (N, 2))
    for _ in range(200):
        f = np.zeros_like(pts)
        for i in range(N):
            for j in range(i + 1, N):
                dx = pts[j] - pts[i]
                d = np.hypot(dx[0], dx[1])
                if d < 0.25 and d > 1e-5:
                    rep = 0.015 / (d**2 + 0.001)
                    f[i] -= dx * rep / d
                    f[j] += dx * rep / d
            for dim in range(2):
                if pts[i, dim] < 0.1: f[i, dim] += 0.06
                elif pts[i, dim] > 0.9: f[i, dim] -= 0.06
        pts += f * 0.05
        pts = np.clip(pts, 0.03, 0.97)
    return pts

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    np.random.seed(42)
    bounds_obj = get_bounds_obj()
    cons = {'type': 'ineq', 'fun': constraints_slsqp}
    
    best_sum = -1.0
    best_x = None

        inits.append(make_hex_init(0.092, ang))

        inits.append(make_force_init(s))
    for _ in range(3):
        inits.append(np.random.uniform(0.15, 0.85, (N, 2)))

    for pts in inits:
        c_curr = pts.copy()
        r_curr = solve_lp_radii(c_curr)
        
        # Run hill climbing on centers
        c_curr, s_val = hill_climb_centers(c_curr, steps=2000)
        r_curr = solve_lp_radii(c_curr)
        
        if np.sum(r_curr) > best_sum:
            best_sum = np.sum(r_curr)
            best_x = np.zeros(3 * N)
            best_x[0::3] = c_curr[:, 0]
            best_x[1::3] = c_curr[:, 1]
            best_x[2::3] = r_curr

    # Phase 2: Deflation & Perturbation to Escape Deep Local Minima
    if best_x is not None:
        for cyc in range(15):
            x0 = best_x.copy()
            
            # Shrink radii to create slack
            shrink = 0.80 + 0.20 * (cyc / 15.0)
            x0[2::3] *= shrink
            
            # Perturb centers
            noise = 0.004 * (0.85 ** cyc)
            x0[0::3] += np.random.normal(0, noise, N)
            x0[1::3] += np.random.normal(0, noise, N)
            
            # Project to feasible bounds
            for i in range(N):
                r = max(1e-6, x0[3 * i + 2])
                x0[3 * i] = np.clip(x0[3 * i], r, 1.0 - r)
                x0[3 * i + 1] = np.clip(x0[3 * i + 1], r, 1.0 - r)
                x0[3 * i + 2] = r

            # Short hill climb on deflated state
            c_def = np.column_stack((x0[0::3], x0[1::3]))
            c_def, _ = hill_climb_centers(c_def, steps=800)
            r_def = solve_lp_radii(c_def)
            
            x0[0::3] = c_def[:, 0]
            x0[1::3] = c_def[:, 1]
            x0[2::3] = r_def

                best_x = x0.copy()

    if best_x is not None:
        x0 = best_x.copy()
        # Slight shrink to ensure strict feasibility for SLSQP start
        x0[2::3] *= 0.995
        for i in range(N):
            x0[3 * i] = np.clip(x0[3 * i], x0[3 * i + 2], 1.0 - x0[3 * i + 2])
            x0[3 * i + 1] = np.clip(x0[3 * i + 1], x0[3 * i + 2], 1.0 - x0[3 * i + 2])

            res = minimize(lambda x: -np.sum(x[2::3]), x0, method='SLSQP', 
                           bounds=bounds_obj, constraints=cons,

            if not np.isnan(res.fun):

                r_pol = solve_lp_radii(c_pol)
                if np.sum(r_pol) > best_sum:
                    best_sum = np.sum(r_pol)
                    best_x = np.zeros(3 * N)

            pass

    for _ in range(100):

                break

                for j in range(i + 1, N):
                    d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])

                        break

            break

        radii *= 0.9998
        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)

    return centers, radii, float(np.sum(radii))
