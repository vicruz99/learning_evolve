import numpy as np
from scipy.optimize import linprog, minimize

N = 26

def get_bounds_centers():
    """Variable bounds for centers: x,y in [0,1]."""
    return [(0.0, 1.0)] * (2 * N)

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist)
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-8)
    except Exception:
        pass
    return np.full(n, 0.04)

def penalty_centers(cf, radii):
    """Smooth penalty function for centers given fixed radii."""
    n = radii.shape[0]
    c = cf.reshape(n, 2)
    p = 0.0
    x, y = c[:, 0], c[:, 1]
    r = radii
    
    # Wall penalties
    p += np.sum(np.maximum(r - x, 0.0)**2)
    p += np.sum(np.maximum(x + r - 1.0, 0.0)**2)
    p += np.sum(np.maximum(r - y, 0.0)**2)
    p += np.sum(np.maximum(y + r - 1.0, 0.0)**2)
    
    # Overlap penalties (vectorized)
    dx = c[:, 0, None] - c[:, None, 0]
    dy = c[:, 1, None] - c[:, None, 1]
    d = np.hypot(dx, dy)
    gap = r[:, None] + r[None, :] - d
    gap[np.diag_indices(n)] = 0.0
    p += np.sum(np.maximum(gap, 0.0)**2)
    
    return 5000.0 * p

def run_alternating(centers0, max_iter=25):
    """Alternating LP (radii) and L-BFGS-B (centers) optimization."""
    c = centers0.copy()
    r = solve_lp_radii(c)
    bounds_c = get_bounds_centers()
    
    for _ in range(max_iter):
        r = solve_lp_radii(c)
        try:
            res = minimize(penalty_centers, c.flatten(), args=(r,),
                           method='L-BFGS-B', bounds=bounds_c,
                           options={'maxiter': 1500, 'ftol': 1e-15, 'gtol': 1e-12})
            c = res.x.reshape(N, 2)
        except Exception:
            pass
        c = np.clip(c, 0.0001, 0.9999)
        
    r = solve_lp_radii(c)
    return c, r

def make_hex_init(r0, angle):
    """Generates a rotated hexagonal lattice initialization."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0 and len(pts) < N + 10:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        
    pts = np.array(pts[:N + 10])
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
        
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.15, 0.85, (1, 2))])
        
    # Break exact symmetries
    pts += np.random.uniform(-0.0005, 0.0005, pts.shape)
    return pts[:N]

def make_force_init(seed):
    """Force-directed layout to spread points evenly."""
    np.random.seed(seed)
    pts = np.random.uniform(0.2, 0.8, (N, 2))
    for _ in range(150):
        f = np.zeros_like(pts)
        for i in range(N):
            for j in range(i + 1, N):
                dx = pts[j] - pts[i]
                d = np.hypot(dx[0], dx[1])
                if d < 0.3 and d > 1e-5:
                    rep = 0.01 / (d**2 + 0.001)
                    f[i] -= dx * rep / d
                    f[j] += dx * rep / d
            for dim in range(2):
                if pts[i, dim] < 0.12: f[i, dim] += 0.05
                elif pts[i, dim] > 0.88: f[i, dim] -= 0.05
        pts += f * 0.05
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    best_sum = -1.0
    best_c = None
    best_r = None

    # Phase 1: Hex lattice rotations (primary topology search)
    for ang in np.linspace(-0.4, 0.4, 17):
        init_c = make_hex_init(0.10, ang)
        c, r = run_alternating(init_c, max_iter=30)
        s = np.sum(r)
        if s > best_sum:
            best_sum, best_c, best_r = s, c.copy(), r.copy()

    # Phase 2: Force-directed starts
    for s in range(12):
        init_c = make_force_init(s)
        c, r = run_alternating(init_c, max_iter=20)
        s_val = np.sum(r)
        if s_val > best_sum:
            best_sum, best_c, best_r = s_val, c.copy(), r.copy()

    # Phase 3: Random starts
    for _ in range(10):
        init_c = np.random.uniform(0.15, 0.85, (N, 2))
        c, r = run_alternating(init_c, max_iter=15)
        s_val = np.sum(r)
        if s_val > best_sum:
            best_sum, best_c, best_r = s_val, c.copy(), r.copy()

    # Phase 4: Deflation & Perturbation on best to escape local minima
    for cyc in range(80):
        c0 = best_c.copy()
        r0 = best_r.copy()
        
        # Shrink a random subset to break rigid contact networks
        k = np.random.randint(3, 8)
        subset = np.random.choice(N, size=k, replace=False)
        r0[subset] *= 0.75
        
        # Perturb centers with cooling noise
        noise = 0.003 * (0.91 ** cyc)
        c0 += np.random.normal(0, noise, c0.shape)
        c0 = np.clip(c0, 0.02, 0.98)
        
        c, r = run_alternating(c0, max_iter=20)
        s_val = np.sum(r)
        if s_val > best_sum:
            best_sum, best_c, best_r = s_val, c.copy(), r.copy()

    # Phase 5: Final LP squeeze
    final_r = solve_lp_radii(best_c)
    if np.sum(final_r) > best_sum - 1e-9:
        best_r = final_r
        best_sum = np.sum(best_r)

    # Final strict validation repair against 1e-12 tolerance
    for _ in range(100):
        valid = True
        for i in range(N):
            if best_r[i] < 0 or best_c[i, 0] < best_r[i] - 1e-9 or best_c[i, 0] > 1.0 - best_r[i] + 1e-9 or \
               best_c[i, 1] < best_r[i] - 1e-9 or best_c[i, 1] > 1.0 - best_r[i] + 1e-9:
                valid = False
                break
        if valid:
            for i in range(N):
                for j in range(i + 1, N):
                    d = np.hypot(best_c[i, 0] - best_c[j, 0], best_c[i, 1] - best_c[j, 1])
                    if d < best_r[i] + best_r[j] - 1e-9:
                        valid = False
                        break
                if not valid:
                    break
        if valid:
            break
            
        # Minimal shrinkage to recover strict feasibility
        best_r *= 0.9998
        best_c[:, 0] = np.clip(best_c[:, 0], best_r, 1.0 - best_r)
        best_c[:, 1] = np.clip(best_c[:, 1], best_r, 1.0 - best_r)

    return best_c, best_r, float(best_sum)
