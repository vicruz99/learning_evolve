import numpy as np
from scipy.optimize import linprog, minimize
import warnings

warnings.filterwarnings('ignore')

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = lim
        idx += 1
        
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dist
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-9)
    except Exception:
        pass
    return np.full(n, 0.05)

def center_penalty(c_flat, radii):
    """Smooth penalty function for centers given fixed radii."""
    c = c_flat.reshape(-1, 2)
    p = 0.0
    x, y = c[:, 0], c[:, 1]
    r = radii
    
    # Wall penalties
    p += np.sum(np.maximum(r - x, 0.0)**2)
    p += np.sum(np.maximum(x - (1.0 - r), 0.0)**2)
    p += np.sum(np.maximum(r - y, 0.0)**2)
    p += np.sum(np.maximum(y - (1.0 - r), 0.0)**2)
    
    # Overlap penalties
    for i in range(N):
        for j in range(i + 1, N):
            d = np.hypot(c[j, 0] - c[i, 0], c[j, 1] - c[i, 1])
            gap = d - r[i] - r[j]
            if gap < 0.0:
                p += gap**2
    return p

def optimize_centers_lbfgs(centers, radii, maxiter=800):
    """Optimize center positions to minimize overlap and boundary violations."""
    c0 = centers.flatten()
    bnds = [(0.001, 0.999)] * (2 * N)
    res = minimize(center_penalty, c0, args=(radii,), method='L-BFGS-B', bounds=bnds,
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    return res.x.reshape(-1, 2)

def alternating_opt(centers, radii, n_iter=8):
    """Alternating LP (radii) and L-BFGS-B (centers) optimization."""
    c = centers.copy()
    r = radii.copy()
    for _ in range(n_iter):
        r = solve_lp_radii(c)
        c = optimize_centers_lbfgs(c, r, maxiter=600)
    return c, r

def make_hex_init(r0, angle):
    """Generates a rotated hexagonal lattice initialization."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0 and len(pts) < N + 10:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        
    pts = np.array(pts[:N + 10])
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
        
    mask = (pts[:, 0] >= 0.01) & (pts[:, 0] <= 0.99) & (pts[:, 1] >= 0.01) & (pts[:, 1] <= 0.99)
    pts = pts[mask]
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.1, 0.9, (1, 2))])
    return pts[:N]

def force_init(seed):
    """Force-directed layout to spread points evenly and push to boundaries."""
    np.random.seed(seed)
    pts = np.random.uniform(0.15, 0.85, (N, 2))
    for step in range(300):
        f = np.zeros_like(pts)
        lr = 0.02 * (1.0 - step / 300.0)
        for i in range(N):
            for j in range(i + 1, N):
                dx = pts[j] - pts[i]
                d = np.hypot(dx[0], dx[1])
                if d < 0.25 and d > 1e-5:
                    rep = 0.01 / (d**2 + 0.0005)
                    f[i] -= dx * rep / d
                    f[j] += dx * rep / d
            for dim in range(2):
                if pts[i, dim] < 0.08:
                    f[i, dim] += 0.05
                elif pts[i, dim] > 0.92:
                    f[i, dim] -= 0.05
        pts += f * lr
        pts = np.clip(pts, 0.01, 0.99)
    return pts

def constraints_slsqp(x):
    """Inequality constraints for SLSQP: g(x) >= 0."""
    xs, ys, rs = x[0::3], x[1::3], x[2::3]
    c = np.concatenate([xs - rs, 1.0 - xs - rs, ys - rs, 1.0 - ys - rs])
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    c = np.concatenate([c, np.hypot(dx[TRIL_IDX], dy[TRIL_IDX]) - dr[TRIL_IDX]])
    return c

def objective_slsqp(x):
    """Objective to minimize: negative sum of radii."""
    return -np.sum(x[2::3])

def run_packing() -> tuple:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    bounds_obj = [(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints_slsqp}
    
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Phase 1: Generate Diverse Initial Configurations
    inits = []
    for ang in np.linspace(-0.35, 0.35, 11):
        inits.append(make_hex_init(0.092, ang))
    for s in range(12):
        inits.append(force_init(s))
    for _ in range(6):
        inits.append(np.random.uniform(0.15, 0.85, (N, 2)))

    # Phase 2: Alternating Optimization + SA Deflation Search
    for pts in inits:
        c_curr, r_curr = alternating_opt(pts, np.full(N, 0.07), n_iter=8)
        curr_sum = np.sum(r_curr)
        
        # SA-like perturbation loop on this basin
        temp = 0.005
        for cyc in range(60):
            noise = temp * (0.95 ** cyc)
            c_pert = c_curr + np.random.normal(0, noise, c_curr.shape)
            r_pert = r_curr.copy()
            
            # Deflate random subset to break rigid contacts
            k = np.random.randint(3, 8)
            subset = np.random.choice(N, size=k, replace=False)
            r_pert[subset] *= 0.85
            
            c_pert = np.clip(c_pert, 0.02, 0.98)
            c_pert, r_pert = alternating_opt(c_pert, r_pert, n_iter=4)
            
            new_sum = np.sum(r_pert)
            if new_sum > curr_sum:
                curr_sum = new_sum
                c_curr = c_pert
                r_curr = r_pert
                
        if curr_sum > best_sum:
            best_sum = curr_sum
            best_centers = c_curr.copy()
            best_radii = r_curr.copy()

    # Phase 3: Rotation Search on Best Configuration
    if best_centers is not None:
        for ang_deg in np.linspace(-12, 12, 25):
            ang = np.deg2rad(ang_deg)
            c, s = np.cos(ang), np.sin(ang)
            rot_centers = (best_centers - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
            rot_centers = np.clip(rot_centers, 0.01, 0.99)
            rot_radii = solve_lp_radii(rot_centers)
            rot_sum = np.sum(rot_radii)
            if rot_sum > best_sum:

                best_centers = rot_centers.copy()
                best_radii = rot_radii.copy()

    # Phase 4: SLSQP Final Polish
    if best_centers is not None:
        x0 = np.zeros(3 * N)
        x0[0::3] = best_centers[:, 0]
        x0[1::3] = best_centers[:, 1]
        x0[2::3] = best_radii * 0.998

            if not np.isnan(res.fun):

                r_tmp = solve_lp_radii(c_tmp)
                s_tmp = np.sum(r_tmp)
                if s_tmp > best_sum:
                    best_sum = s_tmp
                    best_centers = c_tmp.copy()
                    best_radii = r_tmp.copy()
        except Exception:
            pass

    # Fallback
    if best_centers is None:

    # Final strict validity adjustment against 1e-12 tolerance

                break

                    d = np.hypot(best_centers[i, 0] - best_centers[j, 0], best_centers[i, 1] - best_centers[j, 1])

        best_centers[:, 0] = np.clip(best_centers[:, 0], best_radii, 1.0 - best_radii)
        best_centers[:, 1] = np.clip(best_centers[:, 1], best_radii, 1.0 - best_radii)

    return best_centers, best_radii, float(np.sum(best_radii))
