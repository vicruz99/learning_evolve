import numpy as np
from scipy.optimize import minimize, linprog
import warnings

warnings.filterwarnings('ignore')

N = 26
TRIL = np.tril_indices(N, -1)

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-8, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-8, 0.5)])
    return b

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = centers.shape[0]
    num_ineq = n + n * (n - 1) // 2
    A = np.zeros((num_ineq, n))
    b = np.zeros(num_ineq)
    
    idx = 0
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A[idx, i] = 1.0
        b[idx] = max(0.0, lim)
        idx += 1
        
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = max(0.0, dist)
            idx += 1
            
    bounds_lp = [(0.0, None)] * n
    try:
        res = linprog(-np.ones(n), A_ub=A, b_ub=b, bounds=bounds_lp, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-8)
    except Exception:
        pass
    return np.full(n, 0.04)

def constraints_slsqp(x):
    """Inequality constraints for SLSQP: g(x) >= 0."""
    xs, ys, rs = x[0::3], x[1::3], x[2::3]
    c = np.concatenate([xs - rs, 1.0 - xs - rs, ys - rs, 1.0 - ys - rs])
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    c = np.concatenate([c, np.hypot(dx[TRIL], dy[TRIL]) - dr[TRIL]])
    return c

def objective_func(x):
    """Objective to minimize: negative sum of radii."""
    return -np.sum(x[2::3])

def make_hex_init(r0, angle):
    """Generates a rotated hexagonal lattice initialization."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 5:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0 and len(pts) < N + 5:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        
    pts = np.array(pts[:N + 5])
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
        
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.15, 0.85, (1, 2))])
    return pts[:N]

def hill_climb_centers(centers, steps=800, seed=0):
    """Hill-climb optimization over centers using LP for exact radii evaluation."""
    np.random.seed(seed)
    best_c = centers.copy()
    best_r = solve_lp_radii(best_c)
    best_sum = np.sum(best_r)
    
    for step in range(steps):
        # Adaptive cooling schedule
        scale = 0.012 * (1.0 - step / steps) + 0.0005
        
        # Decide perturbation type: shift or swap
        if np.random.rand() < 0.25:
            # Swap two random circles to explore different topologies
            idx = np.random.choice(N, size=2, replace=False)
            new_c = best_c.copy()
            new_c[[idx[0], idx[1]]] = new_c[[idx[1], idx[0]]]
        else:
            # Perturb 1-3 circles
            k = np.random.randint(1, 4)
            idx = np.random.choice(N, size=k, replace=False)
            new_c = best_c.copy()
            for i in idx:
                new_c[i] += np.random.uniform(-scale, scale, 2)
                new_c[i, 0] = np.clip(new_c[i, 0], 0.005, 0.995)
                new_c[i, 1] = np.clip(new_c[i, 1], 0.005, 0.995)
                
        new_r = solve_lp_radii(new_c)
        new_sum = np.sum(new_r)
        
        # Strict improvement with tiny tolerance
        if new_sum > best_sum + 1e-9:
            best_c = new_c
            best_r = new_r
            best_sum = new_sum
            
    return best_c, best_r, best_sum

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    np.random.seed(42)

            best_sum = s_val
            best_c = c.copy()
            best_r = r.copy()

    return best_c, best_r, float(np.sum(best_r))
