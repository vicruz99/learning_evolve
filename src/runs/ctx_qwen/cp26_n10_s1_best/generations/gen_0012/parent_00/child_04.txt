import numpy as np
from scipy.optimize import minimize, linprog
import warnings

warnings.filterwarnings('ignore')

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-7, 0.5]."""
    return [(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)] * N

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist)
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-8)
    except Exception:
        pass
    return np.full(n, 0.04)

def penalty_func(c_flat, r_arr, n_circles):
    """Smooth penalty function for centers given fixed radii."""
    cc = c_flat.reshape(-1, 2)
    p = 0.0
    x, y = cc[:, 0], cc[:, 1]
    rr = r_arr
    p += np.sum(np.maximum(rr - x, 0.0)**2)
    p += np.sum(np.maximum(x - (1.0 - rr), 0.0)**2)
    p += np.sum(np.maximum(rr - y, 0.0)**2)
    p += np.sum(np.maximum(y - (1.0 - rr), 0.0)**2)
    for i in range(n_circles):
        for j in range(i + 1, n_circles):
            dx = cc[j, 0] - cc[i, 0]
            dy = cc[j, 1] - cc[i, 1]
            d = np.hypot(dx, dy)
            gap = d - rr[i] - rr[j]
            if gap < 0.0:
                p += gap**2
    return p

def local_penalty_opt(centers, radii, n_iter=4):
    """Alternating center optimization using L-BFGS-B and LP."""
    c = centers.copy()
    r = radii.copy()
    bounds_c = [(0.001, 0.999)] * (2 * N)
    
    for _ in range(n_iter):
        r = solve_lp_radii(c)
        c_flat = c.flatten()
        try:
            res = minimize(penalty_func, c_flat, args=(r, N), 
                           method='L-BFGS-B', bounds=bounds_c, 
                           options={'maxiter': 800, 'ftol': 1e-14})
            c = res.x.reshape(-1, 2)
        except Exception:
            pass
    return c, r

def greedy_lp_search(centers, n_steps=600):
    """Hill-climbing search that uses LP to evaluate radii for candidate centers."""
    curr_c = centers.copy()
    curr_r = solve_lp_radii(curr_c)
    curr_sum = np.sum(curr_r)
    best_c = curr_c.copy()
    best_r = curr_r.copy()
    best_sum = curr_sum
    
    no_improve = 0
    for step in range(n_steps):
        idx = np.random.randint(N)
        new_c = curr_c.copy()
        new_r = curr_r.copy()
        
        # Shrink selected circle to allow movement
        shrink = 0.3 + 0.5 * (step / n_steps)
        new_r[idx] *= shrink
        
        # Move randomly with cooling step size
        step_size = 0.03 * (1.0 - step / n_steps)
        new_c[idx] += np.random.uniform(-step_size, step_size, 2)
        new_c[idx] = np.clip(new_c[idx], 0.02, 0.98)
        
        cand_r = solve_lp_radii(new_c)
        cand_sum = np.sum(cand_r)
        
        if cand_sum > curr_sum + 1e-8:
            curr_c = new_c
            curr_r = cand_r
            curr_sum = cand_sum
            no_improve = 0
            if curr_sum > best_sum:
                best_sum = curr_sum
                best_c = curr_c.copy()
                best_r = curr_r.copy()
        else:
            no_improve += 1
            if no_improve > 120:
                break
                
        # Periodically perturb multiple circles to break symmetries

                best_r = temp_r
                best_sum = np.sum(temp_r)
                curr_c, curr_r, curr_sum = best_c, best_r, best_sum
                no_improve = 0
                
    return best_c, best_r, best_sum

def constraints_slsqp(x):
    """Inequality constraints for SLSQP: g(x) >= 0."""
    xs, ys, rs = x[0::3], x[1::3], x[2::3]
    c = np.concatenate([xs - rs, 1.0 - xs - rs, ys - rs, 1.0 - ys - rs])
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    c = np.concatenate([c, np.hypot(dx[TRIL_IDX], dy[TRIL_IDX]) - dr[TRIL_IDX]])
    return c

def objective(x):
    return -np.sum(x[2::3])

def make_hex_init(r0, angle):
    """Generates a rotated hexagonal lattice initialization."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0 and len(pts) < N + 10:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        
    pts = np.array(pts[:N + 10])
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
        
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.15, 0.85, (1, 2))])
    return pts[:N]
