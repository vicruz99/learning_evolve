import numpy as np
from scipy.optimize import minimize, linprog
import warnings
warnings.filterwarnings('ignore')

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-7, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)])
    return b

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist)
            idx += 1
            
    bounds_lp = [(0.0, None)] * n
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds_lp, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-7)
    except Exception:
        pass
    return np.full(n, 0.04)

def penalty_function(c_flat, radii):
    """Smooth penalty function for centers given fixed radii."""
    c = c_flat.reshape(-1, 2)
    p = 0.0
    x, y = c[:, 0], c[:, 1]
    r = radii
    
    # Wall penalties
    p += np.sum(np.maximum(r - x, 0.0)**2)
    p += np.sum(np.maximum(x - (1.0 - r), 0.0)**2)
    p += np.sum(np.maximum(r - y, 0.0)**2)
    p += np.sum(np.maximum(y - (1.0 - r), 0.0)**2)
    
    # Overlap penalties
    for i in range(N):
        for j in range(i + 1, N):
            d = np.hypot(c[j, 0] - c[i, 0], c[j, 1] - c[i, 1])
            gap = d - r[i] - r[j]
            if gap < 0.0:
                p += gap**2
    return p

def optimize_centers_penalty(centers, radii):
    """Optimize centers to minimize overlaps/wall violations for fixed radii."""
    c0 = centers.flatten()
    bounds_c = [(0.001, 0.999)] * (2 * N)
    res = minimize(penalty_function, c0, args=(radii,), method='L-BFGS-B', 
                   bounds=bounds_c, options={'maxiter': 1000, 'ftol': 1e-15})
    return res.x.reshape(-1, 2)

def constraints_slsqp(x):
    """Inequality constraints for SLSQP: g(x) >= 0. Uses Euclidean distance for stability."""
    xs, ys, rs = x[0::3], x[1::3], x[2::3]
    
    # Boundary constraints
    c = np.concatenate([
        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs
    ])
    
    # Overlap constraints: dist >= r_i + r_j
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    c = np.concatenate([c, np.hypot(dx[TRIL_IDX], dy[TRIL_IDX]) - dr[TRIL_IDX]])
    return c

def objective_slsqp(x):
    """Objective to minimize: negative sum of radii."""
    return -np.sum(x[2::3])

def make_init(strategy, seed):
    """Generates diverse initial configurations."""
    np.random.seed(seed)
    if strategy == 'hex':
        pts = []
        r0 = 0.092
        y = r0
        row = 0
        while len(pts) < N + 10:
            x = r0 if row % 2 == 0 else 2.0 * r0
            while x <= 1.0 - r0 and len(pts) < N + 10:
                pts.append([x, y])
                x += 2.0 * r0
            y += np.sqrt(3.0) * r0
            row += 1
        pts = np.array(pts[:N + 10])
        
        ang = np.random.uniform(-0.35, 0.35)
        c = np.array([0.5, 0.5])
        pts = (pts - c) @ np.array([[np.cos(ang), -np.sin(ang)], [np.sin(ang), np.cos(ang)]]) + c
        
        mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
        pts = pts[mask]
        while len(pts) < N:
            pts = np.vstack([pts, np.random.uniform(0.1, 0.9, 2)])
        return pts[:N]
        
    elif strategy == 'force':
        pts = np.random.uniform(0.15, 0.85, (N, 2))
        for _ in range(200):
            f = np.zeros_like(pts)
            for i in range(N):
                for j in range(i + 1, N):
                    dx = pts[j] - pts[i]
                    d = np.hypot(dx[0], dx[1])
                    if d < 0.3 and d > 1e-5:
                        rep = 0.012 / (d**2 + 1e-4)
                        f[i] -= dx * rep / d
                        f[j] += dx * rep / d
                for dim in range(2):
                    if pts[i, dim] < 0.1: f[i, dim] += 0.04
                    elif pts[i, dim] > 0.9: f[i, dim] -= 0.04
            pts += f * 0.03
            pts = np.clip(pts, 0.02, 0.98)
        return pts
        
    elif strategy == 'edge':
        pts = []
        r = 0.09
        # Corners
        for cx in [r, 1.0 - r]:
            for cy in [r, 1.0 - r]:
                pts.append([cx, cy])
        # Edges
        for t in np.linspace(0, 1, 7)[1:-1]:
            pts.append([t, r])
            pts.append([t, 1.0 - r])
            pts.append([r, t])
            pts.append([1.0 - r, t])
        # Staggered fill
        y = r + 0.06
        row = 0
        while len(pts) < N:
            x = r + 0.06 if row % 2 == 0 else 2.0 * r + 0.06
            while x <= 1.0 - r and len(pts) < N:
                pts.append([x, y])
                x += 2.0 * r
            y += np.sqrt(3.0) * r
            row += 1
        return np.array(pts[:N])
        
    return np.random.uniform(0.1, 0.9, (N, 2))

def local_search_coordinate(centers, radii, max_passes=3):
    """Deterministic local search: displace circles to maximize LP radii sum."""
    c = centers.copy()
    r = radii.copy()
    
    for _ in range(max_passes):
        improved = True
        while improved:
            improved = False
            for i in range(N):
                best_sum_r = np.sum(r)
                for dx in [-0.004, -0.002, -0.001, 0.001, 0.002, 0.004]:
                    for dy in [-0.004, -0.002, -0.001, 0.001, 0.002, 0.004]:
                        if dx == 0 and dy == 0:
                            continue
                        nc = c.copy()
                        nc[i, 0] = np.clip(c[i, 0] + dx, 0.01, 0.99)
                        nc[i, 1] = np.clip(c[i, 1] + dy, 0.01, 0.99)
                        
                        nr = solve_lp_radii(nc)
                        s_new = np.sum(nr)
                        if s_new > best_sum_r + 1e-7:
                            c = nc
                            r = nr
                            best_sum_r = s_new
                            improved = True
                            break
                    if improved:
                        break
    return c, r

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints_slsqp}
    
    best_sum = -1.0
    best_x = None
    
    # Phase 1: Multi-start with Diverse Initializations
    inits = []
    for s in range(15):
        inits.append(make_init('hex', s))
    for s in range(12):
        inits.append(make_init('force', s))
    for s in range(8):
        inits.append(make_init('edge', s))
        
    for pts in inits:
        c_curr, r_curr = pts.copy(), np.full(N, 0.07)
        
        # Alternate LP and center optimization to find a strong local topology
        for _ in range(3):
            r_curr = solve_lp_radii(c_curr)
            c_curr = optimize_centers_penalty(c_curr, r_curr)
            
        x0 = np.zeros(3 * N)
        x0[0::3] = c_curr[:, 0]
        x0[1::3] = c_curr[:, 1]
        x0[2::3] = np.maximum(r_curr * 0.995, 1e-6)
        
        try:
            res = minimize(objective_slsqp, x0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 15000, 'ftol': 1e-14, 'disp': False})
            if not np.isnan(res.fun):
                c_opt = np.column_stack((res.x[0::3], res.x[1::3]))
                r_opt = solve_lp_radii(c_opt)
                if r_opt is not None:
                    s_val = np.sum(r_opt)
                    if s_val > best_sum:
                        best_sum = s_val
                        best_x = np.zeros(3 * N)
                        best_x[0::3] = c_opt[:, 0]
                        best_x[1::3] = c_opt[:, 1]
                        best_x[2::3] = r_opt
        except Exception:
            pass

    # Phase 2: Basin Hopping / Deflation to Escape Local Minima
    if best_x is not None:
        for step in range(80):
            x0 = best_x.copy()
            
            # Cooling perturbation
            noise = 0.0025 * (0.92 ** step)
            x0[0::3] += np.random.normal(0, noise, N)
            x0[1::3] += np.random.normal(0, noise, N)
            
            # Deflate a random subset to break rigid contact networks

            # Project to feasible bounds
            for i in range(N):

            try:

            except Exception:

    # Phase 3: Deterministic Local Search Polish

    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3].copy()

    # Final strict validation repair against 1e-12 tolerance
    for _ in range(100):

                valid = False
                break
        if valid:
            for i in range(N):
                for j in range(i + 1, N):
                    d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                    if d < radii[i] + radii[j] - 1e-9:
                        valid = False
                        break
                if not valid:

        if valid:
            break
            
        # Minimal shrinkage to recover strict feasibility
        radii *= 0.9995
        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)

    return centers, radii, float(np.sum(radii))
