import numpy as np
from scipy.optimize import linprog, minimize
import warnings
warnings.filterwarnings('ignore')

N = 26

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    dx = centers[:, 0, None] - centers[:, 0]
    dy = centers[:, 1, None] - centers[:, 1]
    dists = np.hypot(dx, dy)
    np.fill_diagonal(dists, 1.0)
    
    triu_idx = np.triu_indices(n, 1)
    A_ub[idx:, triu_idx[0]] = 1.0
    A_ub[idx:, triu_idx[1]] = 1.0
    b_ub[idx:] = dists[triu_idx]
    
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 1e-9)
    return np.full(n, 0.05)

def force_init():
    """Force-directed layout to spread points evenly and push to boundaries."""
    pts = np.random.uniform(0.15, 0.85, (N, 2))
    for _ in range(150):
        f = np.zeros_like(pts)
        for i in range(N):
            for j in range(i + 1, N):
                dx = pts[j] - pts[i]
                d = np.hypot(dx[0], dx[1])
                if d < 0.3 and d > 1e-5:
                    rep = 0.01 / (d**2 + 1e-4)
                    f[i] -= dx * rep / d
                    f[j] += dx * rep / d
            for dim in range(2):
                if pts[i, dim] < 0.1: f[i, dim] += 0.05
                elif pts[i, dim] > 0.9: f[i, dim] -= 0.05
        pts += f * 0.05
        pts = np.clip(pts, 0.02, 0.98)
    return pts

def hex_init(angle=0.0):
    """Generates a rotated hexagonal lattice initialization."""
    pts = []
    r0 = 0.09
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0 and len(pts) < N + 10:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
    pts = np.array(pts[:N + 10])
    if angle != 0.0:
        c = np.array([0.5, 0.5])
        pts = (pts - c) @ np.array([[np.cos(angle), -np.sin(angle)], [np.sin(angle), np.cos(angle)]]) + c
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.1, 0.9, (1, 2))])
    return pts[:N]

def hill_climb(centers, steps=2000):
    """Stochastic hill climbing on centers with LP radius evaluation."""
    curr = centers.copy()
    r = solve_lp_radii(curr)
    best_sum = np.sum(r)
    best_c = curr.copy()
    best_r = r.copy()
    sigma = 0.015
    
    for step in range(steps):
        sigma *= 0.998
        if step % 400 == 0:
            sigma = max(0.005, sigma * 1.8)
            
        i = np.random.randint(N)
        best_local_sum = best_sum
        best_local_pos = curr[i]
        best_local_r = None
        
        for _ in range(5):
            new_pos = curr[i] + np.random.normal(0, sigma, 2)
            new_pos = np.clip(new_pos, 0.005, 0.995)
            curr[i] = new_pos
            new_r = solve_lp_radii(curr)
            new_sum = np.sum(new_r)
            if new_sum > best_local_sum:
                best_local_sum = new_sum
                best_local_pos = new_pos.copy()
                best_local_r = new_r.copy()
                
        curr[i] = best_local_pos
        if best_local_sum > best_sum:
            best_sum = best_local_sum
            best_c = curr.copy()
            best_r = best_local_r
        else:
            curr[i] = best_c[i]
            
    return best_c, best_r, best_sum

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Phase 1: Diverse Initializations
    inits = []
    for ang in np.linspace(-0.4, 0.4, 9):
        inits.append(hex_init(ang))
    for _ in range(6):
        inits.append(force_init())
        
    # Phase 2: Hill Climbing Optimization
    for init_pts in inits:
        c, r, s = hill_climb(init_pts, steps=2000)
        if s > best_sum:
            best_sum = s
            best_centers = c
            best_radii = r
            
    # Phase 3: High-Precision SLSQP Polish
    if best_centers is not None:
        x0 = np.zeros(3 * N)
        x0[0::3] = best_centers[:, 0]
        x0[1::3] = best_centers[:, 1]
        x0[2::3] = best_radii * 0.999
        
        def obj(x):
            return -np.sum(x[2::3])
            
        def constr(x):
            xs, ys, rs = x[0::3], x[1::3], x[2::3]
            c = np.concatenate([xs - rs, 1.0 - xs - rs, ys - rs, 1.0 - ys - rs])
            dx = xs[:, None] - xs[None, :]
            dy = ys[:, None] - ys[None, :]
            dr = rs[:, None] + rs[None, :]
            triu = np.triu_indices(N, 1)
            c = np.concatenate([c, np.hypot(dx[triu], dy[triu]) - dr[triu]])
            return c
            
        bounds = [(0, 1), (0, 1), (1e-7, 0.5)] * N
        try:
            res = minimize(obj, x0, method='SLSQP', bounds=bounds, 
                           constraints={'type': 'ineq', 'fun': constr},
                           options={'maxiter': 10000, 'ftol': 1e-13})
            if not np.isnan(res.fun):
                c_pol = np.column_stack((res.x[0::3], res.x[1::3]))
                r_pol = solve_lp_radii(c_pol)
                s_pol = np.sum(r_pol)
                if s_pol > best_sum:
                    best_sum = s_pol
                    best_centers = c_pol
                    best_radii = r_pol
        except Exception:
            pass

    # Phase 4: Strict Validation Repair
    centers = best_centers
    radii = best_radii
    for _ in range(50):
        valid = True
        for i in range(N):
            if radii[i] < 1e-12 or centers[i, 0] < radii[i] - 1e-10 or centers[i, 0] > 1.0 - radii[i] + 1e-10 or \
               centers[i, 1] < radii[i] - 1e-10 or centers[i, 1] > 1.0 - radii[i] + 1e-10:
                valid = False; break
        if valid:
            for i in range(N):
                for j in range(i + 1, N):
                    d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                    if d < radii[i] + radii[j] - 1e-10:
                        valid = False; break
                if not valid: break
        if valid: break
        radii *= 0.999
        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)
        
    return centers, radii, float(np.sum(radii))
