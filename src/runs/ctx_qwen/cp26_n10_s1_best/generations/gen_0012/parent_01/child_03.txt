

<strategy>
To push the sum of radii beyond 2.636, I will:
1. Use `np.hypot` for overlap constraints to provide smoother gradients near contact points, avoiding the singularity of squared-distance formulations.
2. Create highly diverse initial configurations: rotated hexagonal lattices, boundary/corner-focused layouts, and force-directed spread configurations that explicitly push points toward walls.
3. Implement an alternating optimization scheme: fix centers -> solve LP for maximal radii -> fix radii -> optimize centers with L-BFGS-B using a penalty-based objective -> repeat. This often escapes local minima that SLSQP gets stuck in.
4. Run a multi-start strategy with basin-hopping from the best solution found, using controlled deflation to break rigid contact networks.
5. Polish the final result with high-precision SLSQP and a final LP snap.
6. Ensure strict feasibility with a careful validation repair step that minimally shrinks radii if needed.
</strategy>

import numpy as np
from scipy.optimize import minimize, linprog
import warnings
warnings.filterwarnings('ignore')

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-7, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)])
    return b

def constraints(x):
    """Inequality constraints g(x) >= 0."""
    xs, ys, rs = x[0::3], x[1::3], x[2::3]
    
    # Boundary constraints
    c = np.concatenate([
        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs
    ])
    
    # Overlap constraints: dist >= r_i + r_j
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    c = np.concatenate([c, np.hypot(dx[TRIL_IDX], dy[TRIL_IDX]) - dr[TRIL_IDX]])
    return c

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist)
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-7)
    except Exception:
        pass
    return np.full(n, 0.05)

def penalty_objective(c_flat, radii, penalty_weight=1000.0):
    """Penalty-based objective for optimizing centers with fixed radii."""
    c = c_flat.reshape(-1, 2)
    r = radii
    n = c.shape[0]
    
    # Base objective: we want to maximize sum of radii, but radii are fixed here.
    # Instead, we minimize overlap and boundary violations.
    penalty = 0.0
    
    # Boundary penalties
    x, y = c[:, 0], c[:, 1]
    penalty += penalty_weight * np.sum(np.maximum(r - x, 0.0)**2)
    penalty += penalty_weight * np.sum(np.maximum(x - (1.0 - r), 0.0)**2)
    penalty += penalty_weight * np.sum(np.maximum(r - y, 0.0)**2)
    penalty += penalty_weight * np.sum(np.maximum(y - (1.0 - r), 0.0)**2)
    
    # Overlap penalties
    for i in range(n):
        for j in range(i + 1, n):
            dx = c[j, 0] - c[i, 0]
            dy = c[j, 1] - c[i, 1]
            d = np.hypot(dx, dy)
            gap = d - r[i] - r[j]
            if gap < 0.0:
                penalty += penalty_weight * gap**2
                
    return penalty

def project_to_feasible(x):
    """Project variables to strictly satisfy bounds."""
    x = x.copy()
    for i in range(N):
        r = max(1e-6, x[3 * i + 2])
        x[3 * i] = np.clip(x[3 * i], r, 1.0 - r)
        x[3 * i + 1] = np.clip(x[3 * i + 1], r, 1.0 - r)
        x[3 * i + 2] = r
    return x

def make_hex_init(r0, angle, seed):
    """Generates a rotated hexagonal lattice initialization."""
    np.random.seed(seed)
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0 and len(pts) < N + 10:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        
    pts = np.array(pts[:N + 10])
    
    # Rotate around center
    c = np.array([0.5, 0.5])
    pts = (pts - c) @ np.array([[np.cos(angle), -np.sin(angle)], [np.sin(angle), np.cos(angle)]]) + c
    
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.1, 0.9, (1, 2))])
        
    # Add small noise
    pts += np.random.uniform(-0.002, 0.002, pts.shape)
    return pts[:N]

def make_boundary_init(seed):
    """Generates an initialization focused on corners and edges."""
    np.random.seed(seed)
    pts = []
    r = 0.085
    
    # Corners
    for cx in [r, 1.0 - r]:
        for cy in [r, 1.0 - r]:
            pts.append([cx + np.random.uniform(-0.01, 0.01), cy + np.random.uniform(-0.01, 0.01)])
            
    # Edges (excluding corners)
    for cx in np.linspace(r + 0.05, 1.0 - r - 0.05, 5):
        pts.append([cx, r + np.random.uniform(-0.01, 0.01)])
        pts.append([cx, 1.0 - r + np.random.uniform(-0.01, 0.01)])
    for cy in np.linspace(r + 0.05, 1.0 - r - 0.05, 5):
        pts.append([r + np.random.uniform(-0.01, 0.01), cy])
        pts.append([1.0 - r + np.random.uniform(-0.01, 0.01), cy])
        
    # Fill remaining with staggered pattern
    y = r + 0.06
    row = 0
    while len(pts) < N:
        x_start = r + 0.06 if row % 2 == 0 else 2.0 * r + 0.06
        x = x_start
        while x <= 1.0 - r and len(pts) < N:
            pts.append([x + np.random.uniform(-0.01, 0.01), y + np.random.uniform(-0.01, 0.01)])
            x += 2.0 * r + 0.02
        y += np.sqrt(3.0) * r + 0.01
        row += 1
        
    return np.array(pts[:N])

def make_force_init(seed):
    """Force-directed layout with wall attraction."""
    np.random.seed(seed)
    pts = np.random.uniform(0.15, 0.85, (N, 2))
    
    for step in range(200):
        f = np.zeros_like(pts)
        lr = 0.02 * (1.0 - step / 200.0)
        
        for i in range(N):
            # Pairwise repulsion
            for j in range(i + 1, N):
                dx = pts[j] - pts[i]
                d = np.hypot(dx[0], dx[1])
                if d < 0.35 and d > 1e-5:
                    rep = 0.008 / (d**2 + 0.001)
                    f[i] -= dx * rep / d
                    f[j] += dx * rep / d
                    
            # Wall attraction (push to boundaries)
            for dim in range(2):
                if pts[i, dim] < 0.25:
                    f[i, dim] -= 0.03
                elif pts[i, dim] > 0.75:
                    f[i, dim] += 0.03
                    
        pts += f * lr
        pts = np.clip(pts, 0.02, 0.98)
        
    return pts

def alternating_optimize(centers, radii, n_iters=15):
    """Alternating LP (radii) and L-BFGS-B (centers) optimization."""
    c = centers.copy()
    r = radii.copy()
    bounds_c = [(0.005, 0.995)] * (2 * N)
    
    for _ in range(n_iters):
        # 1. Optimize radii for fixed centers
        r = solve_lp_radii(c)
        
        # 2. Optimize centers for fixed radii
        res = minimize(penalty_objective, c.flatten(), args=(r,),
                       method='L-BFGS-B', bounds=bounds_c,
                       options={'maxiter': 1000, 'ftol': 1e-14})
        c = res.x.reshape(-1, 2)
        c = np.clip(c, 0.005, 0.995)
        
    return c, r

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -1.0
    best_x = None
    
    # Phase 1: Generate Diverse Initial Configurations
    inits = []
    
    # Rotated hexagonal lattices
    for r0 in [0.085, 0.090, 0.095, 0.100]:
        for ang in np.linspace(-0.35, 0.35, 9):
            inits.append(make_hex_init(r0, ang, seed=len(inits)))
            
    # Boundary focused
    for s in range(6):
        inits.append(make_boundary_init(s))
        
    # Force spread with wall attraction
    for s in range(8):
        inits.append(make_force_init(s))
        
    # Random starts
    for s in range(6):
        np.random.seed(s + 100)
        inits.append(np.random.uniform(0.1, 0.9, (N, 2)))

    # Phase 2: Multi-start Optimization
    for pts in inits:
        c_curr, r_curr = pts.copy(), np.full(N, 0.07)
        
        # Alternating optimization
        c_curr, r_curr = alternating_optimize(c_curr, r_curr, n_iters=12)
        
        x0 = np.zeros(3 * N)
        x0[0::3] = c_curr[:, 0]
        x0[1::3] = c_curr[:, 1]
        x0[2::3] = np.maximum(r_curr * 0.995, 1e-6)
        x0 = project_to_feasible(x0)
        
        # SLSQP Polish
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 12000, 'ftol': 1e-13, 'disp': False})
            if not np.isnan(res.fun):
                c_opt = np.column_stack((res.x[0::3], res.x[1::3]))
                r_opt = solve_lp_radii(c_opt)
                if r_opt is not None:
                    s_val = np.sum(r_opt)
                    if s_val > best_sum:
                        best_sum = s_val
                        best_x = np.zeros(3 * N)
                        best_x[0::3] = c_opt[:, 0]
                        best_x[1::3] = c_opt[:, 1]
                        best_x[2::3] = r_opt
        except Exception:
            pass

    # Phase 3: Basin Hopping / Deflation to Escape Local Minima
    if best_x is not None:
        for step in range(100):
            x0 = best_x.copy()
            
            # Cooling perturbation
            noise = 0.0025 * (0.92 ** step)
            x0[0::3] += np.random.normal(0, noise, N)
            x0[1::3] += np.random.normal(0, noise, N)
            
            # Deflate a random subset to break rigid contact networks
            k = np.random.randint(3, 8)
            subset = np.random.choice(N, size=k, replace=False)
            x0[subset * 3 + 2] *= 0.80
            
            x0 = project_to_feasible(x0)
            
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 8000, 'ftol': 1e-13, 'disp': False})
                if not np.isnan(res.fun):
                    c_tmp = np.column_stack((res.x[0::3], res.x[1::3]))
                    r_new = solve_lp_radii(c_tmp)
                    if r_new is not None:
                        x_p = res.x.copy()
                        x_p[2::3] = r_new * 0.999
                        x_p = project_to_feasible(x_p)
                        
                        s_val = np.sum(x_p[2::3])
                        if s_val > best_sum:
                            best_sum = s_val
                            best_x = x_p.copy()
            except Exception:
                pass

    # Phase 4: High-Precision Polish
    if best_x is not None:
        for _ in range(8):
            x_pol = best_x.copy()
            x_pol[2::3] *= 0.98
            x_pol = project_to_feasible(x_pol)
            try:
                res = minimize(objective, x_pol, method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 15000, 'ftol': 1e-14, 'disp': False})
                if not np.isnan(res.fun):
                    c_pol = np.column_stack((res.x[0::3], res.x[1::3]))
                    r_pol = solve_lp_radii(c_pol)
                    if r_pol is not None:
                        s_pol = np.sum(r_pol)
                        if s_pol > best_sum:
                            best_sum = s_pol
                            best_x = np.zeros(3 * N)
                            best_x[0::3] = c_pol[:, 0]
                            best_x[1::3] = c_pol[:, 1]
                            best_x[2::3] = r_pol
            except Exception:
                pass

    # Fallback
    if best_x is None:
        best_x = np.zeros(3 * N)
        best_x[0::3] = np.tile(np.linspace(0.1, 0.9, 5), 6)[:N]
        best_x[1::3] = np.repeat(np.linspace(0.1, 0.9, 6), 5)[:N]
        best_x[2::3] = 0.06
        best_sum = np.sum(best_x[2::3])

    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3].copy()

    # Final strict validation repair against 1e-12 tolerance
    for _ in range(100):
        valid = True
        for i in range(N):
            if radii[i] < 0 or centers[i, 0] < radii[i] - 1e-9 or centers[i, 0] > 1.0 - radii[i] + 1e-9 or \
               centers[i, 1] < radii[i] - 1e-9 or centers[i, 1] > 1.0 - radii[i] + 1e-9:
                valid = False
                break
        if valid:
            for i in range(N):
                for j in range(i + 1, N):
                    d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                    if d < radii[i] + radii[j] - 1e-9:
                        valid = False
                        break
                if not valid:
                    break
        if valid:
            break
            
        # Minimal shrinkage to recover strict feasibility
        radii *= 0.9998
        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)

    return centers, radii, float(np.sum(radii))
