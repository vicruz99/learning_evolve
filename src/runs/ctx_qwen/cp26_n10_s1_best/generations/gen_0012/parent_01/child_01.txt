

Looking at the problem, I need to improve circle packing for 26 circles in a unit square. The current best is ~2.63, and I need to reach 2.636. Let me analyze what's working and what can be improved.

<strategy>
1. Use `np.hypot` for overlap constraints instead of squared distances - provides better numerical conditioning near contact points
2. Implement coordinate descent: for each circle, try small perturbations in multiple directions, then re-optimize radii via LP. This can escape local minima that SLSQP gets stuck in
3. Add rotation search on the best solution - rotating a good packing can reveal better boundary interactions
4. More aggressive basin hopping with varied deflation strategies
5. Implement "gap-filling" strategy: identify large gaps and try to relocate circles there
6. Use alternating LP (exact radii) + L-BFGS-B (center optimization with penalty) more intensively
7. Generate more diverse initial configurations including specific 5-6-5-6-4 row patterns
</strategy>

import numpy as np
from scipy.optimize import minimize, linprog
import warnings
warnings.filterwarnings('ignore')

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Returns all inequality constraints >= 0 (using hypot for stability)."""
    xs, ys, rs = x[0::3], x[1::3], x[2::3]
    
    c = np.concatenate([
        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs
    ])
    
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    c = np.concatenate([c, np.hypot(dx[TRIL_IDX], dy[TRIL_IDX]) - dr[TRIL_IDX]])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-7, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)])
    return b

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist)
            idx += 1
            
    bounds_lp = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_lp, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-8)
    except Exception:
        pass
    return np.full(n, 0.05)

def project_to_feasible(x):
    """Project variables to strictly satisfy bounds."""
    x = x.copy()
    for i in range(N):
        r = max(1e-6, x[3 * i + 2])
        x[3 * i] = np.clip(x[3 * i], r, 1.0 - r)
        x[3 * i + 1] = np.clip(x[3 * i + 1], r, 1.0 - r)
        x[3 * i + 2] = r
    return x

def center_penalty(c_flat, radii):
    """Smooth penalty for center optimization given fixed radii."""
    c = c_flat.reshape(-1, 2)
    p = 0.0
    n = c.shape[0]
    
    x, y = c[:, 0], c[:, 1]
    r = radii
    p += np.sum(np.maximum(r - x, 0.0)**2 * 1000.0)
    p += np.sum(np.maximum(x - (1.0 - r), 0.0)**2 * 1000.0)
    p += np.sum(np.maximum(r - y, 0.0)**2 * 1000.0)
    p += np.sum(np.maximum(y - (1.0 - r), 0.0)**2 * 1000.0)
    
    for i in range(n):
        for j in range(i + 1, n):
            dx = c[j, 0] - c[i, 0]
            dy = c[j, 1] - c[i, 1]
            d = np.hypot(dx, dy)
            gap = d - r[i] - r[j]
            if gap < 0.0:
                p += gap**2 * 10000.0
    return p

def alternating_opt(centers, radii, n_iter=10):
    """Alternating LP (radii) and L-BFGS-B (centers) optimization."""
    c = centers.copy()
    r = radii.copy()
    bounds_c = [(0.001, 0.999)] * (2 * N)
    
    for it in range(n_iter):
        r = solve_lp_radii(c)
        
        res = minimize(center_penalty, c.flatten(), args=(r,), 
                       method='L-BFGS-B', bounds=bounds_c,
                       options={'maxiter': 1000, 'ftol': 1e-15})
        c = res.x.reshape(-1, 2)
        c = np.clip(c, 0.005, 0.995)
        
    return c, r

def make_hex_init(r0, angle):
    """Generates a rotated hexagonal lattice initialization."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0 and len(pts) < N + 10:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
    
    pts = np.array(pts[:N + 10])
    if angle != 0.0:
        center = np.array([0.5, 0.5])
        pts = (pts - center) @ np.array([[np.cos(angle), -np.sin(angle)], 
                                          [np.sin(angle), np.cos(angle)]]) + center
    
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    if len(pts) < N:
        pad = N - len(pts)
        pts = np.vstack([pts, np.random.uniform(0.2, 0.8, (pad, 2))])
    
    pts += np.random.uniform(-0.0005, 0.0005, pts.shape)
    return pts[:N]

def make_staggered_rows():
    """Generates a row-based staggered initialization tailored for 26 circles."""
    pts = []
    row_configs = [5, 6, 5, 6, 4]
    y_positions = [0.09, 0.25, 0.42, 0.60, 0.78]
    
    for row_idx, (count, y_base) in enumerate(zip(row_configs, y_positions)):
        if count == 1:
            pts.append([0.5, y_base])
        else:
            x_step = 0.9 / (count - 1)
            x_start = 0.05 + (row_idx % 2) * x_step * 0.5
            for k in range(count):
                pts.append([x_start + k * x_step, y_base])
    
    pts = np.array(pts[:N])
    pts += np.random.uniform(-0.005, 0.005, pts.shape)
    return np.clip(pts, 0.03, 0.97)

def make_force_init(seed):
    """Force-directed layout to spread points evenly."""
    np.random.seed(seed)
    pts = np.random.uniform(0.15, 0.85, (N, 2))
    
    for step in range(300):
        f = np.zeros_like(pts)
        lr = 0.02 * (1.0 - step / 300.0)
        
        for i in range(N):
            for j in range(i + 1, N):
                dx = pts[j] - pts[i]
                d = np.hypot(dx[0], dx[1])
                if d < 0.35 and d > 1e-5:
                    rep = 0.012 / (d**2 + 0.001)
                    f[i] -= dx * rep / d
                    f[j] += dx * rep / d
            
            for dim in range(2):
                if pts[i, dim] < 0.12:
                    f[i, dim] += 0.06
                elif pts[i, dim] > 0.88:
                    f[i, dim] -= 0.06
        
        pts += f * lr
        pts = np.clip(pts, 0.02, 0.98)
    
    return pts

def make_corner_edge_init(seed):
    """Places circles at corners and edges first, then fills interior."""
    np.random.seed(seed)
    pts = []
    r = 0.09
    
    # 4 corners
    for cx in [r, 1.0 - r]:
        for cy in [r, 1.0 - r]:
            pts.append([cx, cy])
    
    # Edge midpoints
    for cx in np.linspace(r + 0.05, 1.0 - r - 0.05, 4):
        pts.append([cx, r])
        pts.append([cx, 1.0 - r])
    for cy in np.linspace(r + 0.05, 1.0 - r - 0.05, 3):
        pts.append([r, cy])
        pts.append([1.0 - r, cy])
    
    # Fill remaining with hex pattern in center
    y = r + 0.12
    row = 0
    while len(pts) < N:
        x = r + 0.12 if row % 2 == 0 else 2.0 * r + 0.12
        while x <= 1.0 - r - 0.05 and len(pts) < N:
            pts.append([x, y])
            x += 2.0 * r
        y += np.sqrt(3.0) * r
        row += 1
    
    pts = np.array(pts[:N])
    pts += np.random.uniform(-0.003, 0.003, pts.shape)
    return np.clip(pts, 0.03, 0.97)

def coordinate_descent_step(centers, radii, perturbations):
    """Try moving each circle in various directions and re-optimize radii."""
    best_centers = centers.copy()
    best_radii = radii.copy()
    best_sum = np.sum(radii)
    
    for i in range(N):
        for dx in perturbations:
            for dy in perturbations:
                if dx == 0.0 and dy == 0.0:
                    continue
                
                new_centers = best_centers.copy()
                new_centers[i, 0] = np.clip(best_centers[i, 0] + dx, 0.01, 0.99)
                new_centers[i, 1] = np.clip(best_centers[i, 1] + dy, 0.01, 0.99)
                
                new_radii = solve_lp_radii(new_centers)
                new_sum = np.sum(new_radii)
                
                if new_sum > best_sum + 1e-8:
                    best_centers = new_centers
                    best_radii = new_radii
                    best_sum = new_sum
    
    return best_centers, best_radii, best_sum

def rotate_centers(centers, angle):
    """Rotate centers around the square center."""
    c, s = np.cos(angle), np.sin(angle)
    center = np.array([0.5, 0.5])
    rotated = (centers - center) @ np.array([[c, -s], [s, c]]) + center
    return np.clip(rotated, 0.01, 0.99)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""

    cons = {'type': 'ineq', 'fun': constraints}
