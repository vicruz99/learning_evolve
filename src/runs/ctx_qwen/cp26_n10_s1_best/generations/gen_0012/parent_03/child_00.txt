import numpy as np
from scipy.optimize import minimize, linprog

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Returns all inequality constraints >= 0 (vectorized)."""
    xs, ys, rs = x[0::3], x[1::3], x[2::3]
    c = np.concatenate([xs - rs, 1.0 - xs - rs, ys - rs, 1.0 - ys - rs])
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    c = np.concatenate([c, np.hypot(dx[TRIL_IDX], dy[TRIL_IDX]) - dr[TRIL_IDX]])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-7, 0.5]."""
    return [(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)] * N

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A = np.zeros((num_ineq, n))
    b = np.zeros(num_ineq)
    
    idx = 0
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A[idx, i] = 1.0
        b[idx] = max(0.0, lim)
        idx += 1
        
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = max(0.0, d)
            idx += 1
            
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0.0, None)] * n, method='highs')
    if res.success:
        return np.maximum(res.x, 1e-9)
    return np.full(n, 0.05)

def project(x):
    """Project variables to strictly satisfy bounds."""
    x = x.copy()
    for i in range(N):
        r = max(1e-6, x[3 * i + 2])
        x[3 * i] = np.clip(x[3 * i], r, 1.0 - r)
        x[3 * i + 1] = np.clip(x[3 * i + 1], r, 1.0 - r)
        x[3 * i + 2] = r
    return x

def make_hex_init(r0, angle):
    """Generates a hexagonal lattice initialization with optional rotation."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 5:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0 and len(pts) < N + 5:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
    pts = np.array(pts[:N + 5])
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
    mask = (pts[:, 0] >= 0.01) & (pts[:, 0] <= 0.99) & (pts[:, 1] >= 0.01) & (pts[:, 1] <= 0.99)
    pts = pts[mask]
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.1, 0.9, (1, 2))])
    return pts[:N]

def make_force_init(seed):
    """Force-directed layout to spread points evenly and push to boundaries."""
    np.random.seed(seed)
    pts = np.random.uniform(0.15, 0.85, (N, 2))
    for step in range(250):
        f = np.zeros_like(pts)
        lr = 0.015 * (1.0 - step / 250.0)
        for i in range(N):
            for j in range(i + 1, N):
                dx = pts[j] - pts[i]
                d = np.hypot(dx[0], dx[1])
                if d < 0.28 and d > 1e-5:
                    rep = 0.012 / (d**2 + 0.001)
                    f[i] -= dx * rep / d
                    f[j] += dx * rep / d
            for dim in range(2):
                if pts[i, dim] < 0.1: f[i, dim] += 0.05
                elif pts[i, dim] > 0.9: f[i, dim] -= 0.05
        pts += f * lr
        pts = np.clip(pts, 0.02, 0.98)
    return pts

def run_packing():
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -1.0
    best_x = None

    # Phase 1: Diverse Initial Configurations
    inits = []
    for r0 in [0.088, 0.092, 0.096, 0.100, 0.104]:
        for ang in np.linspace(-0.32, 0.32, 15):
            inits.append(make_hex_init(r0, ang))
    for s in range(15):
        inits.append(make_force_init(s))
        
    for pts in inits:
        x0 = np.zeros(3 * N)
        x0[0::3] = pts[:, 0]
        x0[1::3] = pts[:, 1]
        r_lp = solve_lp_radii(pts)
        x0[2::3] = r_lp * 0.995
        x0 = project(x0)
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 25000, 'ftol': 1e-13, 'disp': False})
            if not np.isnan(res.fun):
                s_val = -res.fun
                if s_val > best_sum and np.min(constraints(res.x)) >= -1e-5:
                    best_sum = s_val
                    best_x = res.x.copy()
        except Exception:
            pass
            
    # Phase 2: Iterative Deflation & Refinement to Escape Local Minima
    if best_x is not None:
        for cyc in range(60):
            x0 = best_x.copy()
            shrink = 0.80 + 0.20 * (cyc / 60.0)
            x0[2::3] *= shrink
            noise = 0.0025 / (cyc + 1)
            x0[0::3] += np.random.normal(0, noise, N)
            x0[1::3] += np.random.normal(0, noise, N)
            x0 = project(x0)
            
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 15000, 'ftol': 1e-13, 'disp': False})
                if not np.isnan(res.fun):
                    c_tmp = np.column_stack((res.x[0::3], res.x[1::3]))
                    r_new = solve_lp_radii(c_tmp)
                    x0[2::3] = r_new * 0.998
                    x0 = project(x0)
                    s_val = np.sum(x0[2::3])
                    if s_val > best_sum and np.min(constraints(x0)) >= -1e-5:
                        best_sum = s_val
                        best_x = x0.copy()
            except Exception:
                pass
                
        # Phase 3: Coordinate-wise Local Search on Centers
        c_best = np.column_stack((best_x[0::3], best_x[1::3]))
        r_best = best_x[2::3]
        best_lp_sum = np.sum(r_best)
        
        for _ in range(4):
            improved = True
            while improved:
                improved = False
                for i in range(N):
                    best_local = best_lp_sum
                    c_try = c_best[i].copy()
                    steps = [-0.006, -0.003, 0.0, 0.003, 0.006]
                    for dx in steps:
                        for dy in steps:
                            if dx == 0.0 and dy == 0.0:
                                continue
                            new_c = c_best.copy()
                            new_c[i, 0] = np.clip(c_best[i, 0] + dx, 0.005, 0.995)
                            new_c[i, 1] = np.clip(c_best[i, 1] + dy, 0.005, 0.995)
                            r_new = solve_lp_radii(new_c)
                            s_new = np.sum(r_new)
                            if s_new > best_local + 1e-8:
                                best_local = s_new
                                c_try = new_c[i].copy()
                    if best_local > best_lp_sum + 1e-8:
                        c_best[i] = c_try
                        best_lp_sum = best_local
                        r_best = solve_lp_radii(c_best)
                        best_x[0::3] = c_best[:, 0]
                        best_x[1::3] = c_best[:, 1]
                        best_x[2::3] = r_best
                        best_sum = best_lp_sum
                        improved = True
                        
        # Phase 4: Fine Rotation Polish
        best_c = np.column_stack((best_x[0::3], best_x[1::3]))
        best_s = np.sum(best_x[2::3])
        for ang in np.linspace(-0.06, 0.06, 25):
            ca, sa = np.cos(ang), np.sin(ang)
            rot_c = (best_c - 0.5) @ np.array([[ca, -sa], [sa, ca]]) + 0.5
            rot_c = np.clip(rot_c, 0.01, 0.99)
            r_rot = solve_lp_radii(rot_c)
            s_rot = np.sum(r_rot)
            if s_rot > best_s:
                best_s = s_rot
                best_c = rot_c
                best_x[0::3] = best_c[:, 0]
                best_x[1::3] = best_c[:, 1]
                best_x[2::3] = r_rot
                best_sum = best_s
    else:
        best_x = np.zeros(3 * N)
        best_x[0::3] = np.tile(np.linspace(0.1, 0.9, 5), 6)[:N]
        best_x[1::3] = np.repeat(np.linspace(0.1, 0.9, 6), 5)[:N]
        best_x[2::3] = 0.06
        
    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3].copy()
    
    # Final LP squeeze
    final_r = solve_lp_radii(centers)
    if np.sum(final_r) > np.sum(radii) - 1e-7:
        radii = final_r.copy()

    # Final strict validation repair against 1e-12 tolerance
    for _ in range(100):
        valid = True
        for i in range(N):
            if radii[i] < 0 or centers[i, 0] < radii[i] - 1e-9 or centers[i, 0] > 1.0 - radii[i] + 1e-9 or \
               centers[i, 1] < radii[i] - 1e-9 or centers[i, 1] > 1.0 - radii[i] + 1e-9:
                valid = False
                break
        if valid:
            for i in range(N):
                for j in range(i + 1, N):
                    d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                    if d < radii[i] + radii[j] - 1e-9:
                        valid = False
                        break
                if not valid:
                    break
        if valid:
            break
            
        radii *= 0.9995
        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)

    return centers, radii, float(np.sum(radii))
