import numpy as np
from scipy.optimize import minimize, linprog
import warnings
warnings.filterwarnings('ignore')

N = 26
TRIL_IDX = np.tril_indices(N, -1)


def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])


def constraints(x):
    """Returns all inequality constraints >= 0 (vectorized)."""
    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]

    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    c = np.concatenate([
        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs
    ])

    # Overlap constraints: dist >= r_i + r_j (using hypot for stable gradients)
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]

    c = np.concatenate([c, np.hypot(dx[TRIL_IDX], dy[TRIL_IDX]) - dr[TRIL_IDX]])
    return c


def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-7, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)])
    return b


def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    num_ineq = n + n * (n - 1) // 2
    A = np.zeros((num_ineq, n))
    b = np.zeros(num_ineq)

    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A[idx, i] = 1.0
        b[idx] = max(0.0, lim)
        idx += 1

    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = max(0.0, d)
            idx += 1

    try:
        res = linprog(-np.ones(n), A_ub=A, b_ub=b, bounds=[(0.0, None)] * n, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-8)
    except Exception:
        pass
    return np.full(n, 0.05)


def project(x):
    """Project variables to strictly satisfy bounds."""
    x = x.copy()
    for i in range(N):
        r = max(1e-6, x[3 * i + 2])
        x[3 * i] = np.clip(x[3 * i], r, 1.0 - r)
        x[3 * i + 1] = np.clip(x[3 * i + 1], r, 1.0 - r)
        x[3 * i + 2] = r
    return x


def make_hex_init(r0, ang, scale=1.0):
    """Generates a hexagonal lattice initialization with optional rotation and scaling."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 15:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0 and len(pts) < N + 15:
            pts.append([x * scale, y * scale])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1

    pts = np.array(pts[:N + 15])

    if ang != 0.0:
        c, s = np.cos(ang), np.sin(ang)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5

    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]

    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.15, 0.85, (1, 2))])
    return pts[:N]


def make_force_init(seed):
    """Force-directed layout to spread points evenly and push to boundaries."""
    np.random.seed(seed)
    pts = np.random.uniform(0.15, 0.85, (N, 2))
    for step in range(400):
        f = np.zeros_like(pts)
        lr = 0.025 * (1.0 - step / 400.0)
        for i in range(N):
            for j in range(i + 1, N):
                dx = pts[j] - pts[i]
                d = np.hypot(dx[0], dx[1])
                if d < 0.35 and d > 1e-5:
                    rep = 0.012 / (d**2 + 0.001)
                    f[i] -= dx * rep / d
                    f[j] += dx * rep / d
            for dim in range(2):
                if pts[i, dim] < 0.10:
                    f[i, dim] += 0.06
                elif pts[i, dim] > 0.90:
                    f[i, dim] -= 0.06
        pts += f * lr
        pts = np.clip(pts, 0.03, 0.97)
    return pts


def make_corner_init():
    """Generates an initialization focused on corners and edges."""
    pts = np.zeros((N, 2))
    r = 0.09
    pts[0] = [r, r]
    pts[1] = [1 - r, r]
    pts[2] = [r, 1 - r]
    pts[3] = [1 - r, 1 - r]
    pts[4] = [0.5, r]
    pts[5] = [0.5, 1 - r]
    pts[6] = [r, 0.5]
    pts[7] = [1 - r, 0.5]
    pts[8] = [0.5, 0.5]
    idx = 9

    y = r + 0.05
    row = 0
    while idx < N and y + r <= 1.0:
        x = r + 0.05 if row % 2 == 0 else r + 0.17
        while idx < N and x + r <= 1.0:
            pts[idx] = [x, y]
            idx += 1
            x += 0.22
        y += 0.14
        row += 1

    for i in range(idx, N):
        pts[i] = np.random.uniform(0.1, 0.9, 2)
    pts += np.random.uniform(-0.003, 0.003, pts.shape)
    return pts


def run_packing():
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}

    best_sum = -1.0
    best_x = None

    # Phase 1: Generate Diverse Initial Configurations
    inits = []
    # Hexagonal lattices with varied scales and rotations
    for scale in [0.90, 0.95, 1.0, 1.05, 1.1]:
        r0 = 0.092 * scale
        for ang in np.linspace(-0.40, 0.40, 15):
            inits.append(make_hex_init(r0, ang))

    # Corner/Edge focused
    for _ in range(8):
        inits.append(make_corner_init())

    # Random + Force spread
    for s in range(18):
        inits.append(make_force_init(s))

    # Phase 2: Multi-start Optimization
    for c_init in inits:
        x0 = np.zeros(3 * N)
        x0[0::3] = c_init[:, 0]
        x0[1::3] = c_init[:, 1]

        # Initial LP snap for radii
        r_lp = solve_lp_radii(c_init)
        x0[2::3] = r_lp * 0.995
        x0 = project(x0)

        # Primary SLSQP run
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 25000, 'ftol': 1e-14, 'disp': False})
            if not np.isnan(res.fun):
                c_tmp = np.column_stack((res.x[0::3], res.x[1::3]))
                r_new = solve_lp_radii(c_tmp)
                curr_sum = np.sum(r_new)
                if curr_sum > best_sum and np.min(constraints(res.x)) >= -1e-5:
                    best_sum = curr_sum
                    best_x = res.x.copy()
        except Exception:
            continue

    # Phase 3: Aggressive Topology Search (Deflation, Swapping, Rotation)
    if best_x is not None:
        # 3a: Adaptive Noise Deflation
        for step in range(120):
            noise = 0.003 * (0.91 ** step)
            x_p = best_x.copy()
            x_p += np.random.normal(0, noise, 3 * N)

            # Gradually recover from aggressive deflation
            shrink = 0.70 + 0.30 * (step / 120.0)
            x_p[2::3] *= shrink

            # Deflate a random subset to break rigid contact networks
            k = np.random.randint(4, 10)
            subset = np.random.choice(N, size=k, replace=False)
            x_p[subset * 3 + 2] *= 0.80

            x_p = project(x_p)

            try:
                res = minimize(objective, x_p, method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 12000, 'ftol': 1e-14, 'disp': False})
                if not np.isnan(res.fun):
                    centers_p = np.column_stack((res.x[0::3], res.x[1::3]))
                    r_p = solve_lp_radii(centers_p)
                    if r_p is not None:
                        x_p[2::3] = r_p
                        x_p = project(x_p)
                        curr_sum = np.sum(x_p[2::3])
                        if curr_sum > best_sum and np.min(constraints(x_p)) >= -1e-5:
                            best_sum = curr_sum
                            best_x = x_p.copy()
            except Exception:
                pass

        # 3b: Global Rotation Search from best
        best_centers = np.column_stack((best_x[0::3], best_x[1::3]))
        for ang_deg in np.linspace(-15, 15, 19):
            ang = np.deg2rad(ang_deg)
            c, s = np.cos(ang), np.sin(ang)
            rot_centers = (best_centers - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
            rot_centers = np.clip(rot_centers, 0.02, 0.98)

            r_rot = solve_lp_radii(rot_centers)
            if r_rot is not None:
                x_rot = np.zeros(3 * N)
                x_rot[0::3] = rot_centers[:, 0]
                x_rot[1::3] = rot_centers[:, 1]
                x_rot[2::3] = r_rot * 0.995
                x_rot = project(x_rot)

                try:
                    res = minimize(objective, x_rot, method='SLSQP', bounds=bounds, constraints=cons,
                                   options={'maxiter': 10000, 'ftol': 1e-14, 'disp': False})
                    if not np.isnan(res.fun):
                        centers_rot = np.column_stack((res.x[0::3], res.x[1::3]))
                        r_final = solve_lp_radii(centers_rot)
                        if r_final is not None:
                            x_rot[2::3] = r_final
                            x_rot = project(x_rot)
                            curr_sum = np.sum(x_rot[2::3])
                            if curr_sum > best_sum and np.min(constraints(x_rot)) >= -1e-5:
                                best_sum = curr_sum
                                best_x = x_rot.copy()
                except Exception:
                    pass

        # 3c: Coordinate Descent Local Search (Fine-grained position tuning)
        for _ in range(3):
            curr_centers = np.column_stack((best_x[0::3], best_x[1::3]))
            improved = True
            while improved:
                improved = False
                for i in range(N):
                    best_local_sum = np.sum(best_x[2::3])
                    # Try displacements in 8 directions with multiple step sizes
                    for ds in [0.003, 0.0015, 0.0007]:
                        for dx in [-ds, ds]:
                            for dy in [-ds, ds]:
                                if abs(dx) < 1e-10 and abs(dy) < 1e-10:
                                    continue
                                new_centers = curr_centers.copy()
                                new_centers[i, 0] = np.clip(curr_centers[i, 0] + dx, 0.01, 0.99)
                                new_centers[i, 1] = np.clip(curr_centers[i, 1] + dy, 0.01, 0.99)

                                r_new = solve_lp_radii(new_centers)
                                if r_new is not None:
                                    s_new = np.sum(r_new)
                                    if s_new > best_local_sum + 1e-8:
                                        curr_centers = new_centers
                                        best_local_sum = s_new
                                        best_x[0::3] = curr_centers[:, 0]
                                        best_x[1::3] = curr_centers[:, 1]
                                        best_x[2::3] = r_new
                                        best_sum = s_new
                                        improved = True
                                        break
                            if improved:
                                break
                        if improved:
                            break

    # Fallback (should not be reached)
    if best_x is None:
        best_x = np.zeros(3 * N)
        best_x[0::3] = np.tile(np.linspace(0.1, 0.9, 5), 6)[:N]
        best_x[1::3] = np.repeat(np.linspace(0.1, 0.9, 6), 5)[:N]
        best_x[2::3] = 0.06
        best_sum = np.sum(best_x[2::3])

    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3].copy()

    # Final LP squeeze to maximize radii for the best centers found
    final_r = solve_lp_radii(centers)
    if np.sum(final_r) > np.sum(radii) - 1e-7:
        radii = final_r.copy()

    # Final strict validation repair against 1e-12 tolerance
    for _ in range(150):
        valid = True
        for i in range(N):
            if radii[i] < 0 or centers[i, 0] < radii[i] - 1e-9 or centers[i, 0] > 1.0 - radii[i] + 1e-9 or \
               centers[i, 1] < radii[i] - 1e-9 or centers[i, 1] > 1.0 - radii[i] + 1e-9:
                valid = False
                break
        if valid:
            for i in range(N):
                for j in range(i + 1, N):
                    d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                    if d < radii[i] + radii[j] - 1e-9:
                        valid = False
                        break
                if not valid:
                    break
        if valid:
            break

        radii *= 0.9998
        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)

    return centers, radii, float(np.sum(radii))
