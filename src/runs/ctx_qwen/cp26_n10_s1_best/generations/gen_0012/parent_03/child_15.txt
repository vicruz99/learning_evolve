import numpy as np
from scipy.optimize import minimize, linprog
import warnings
warnings.filterwarnings('ignore')

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist)
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-8)
    except Exception:
        pass
    return np.full(n, 0.05)

def compute_center_penalty(c_flat, radii):
    """Smooth penalty function for centers given fixed radii."""
    c = c_flat.reshape(-1, 2)
    p = 0.0
    
    # Vectorized overlap penalty
    dx = c[:, 0, None] - c[:, 0]
    dy = c[:, 1, None] - c[:, 1]
    d = np.hypot(dx, dy)
    gap = d - (radii[:, None] + radii[None, :])
    mask = gap < 0.0
    p += np.sum(gap[mask] ** 2)
    
    # Vectorized boundary penalty
    x, y = c[:, 0], c[:, 1]
    r = radii
    p += np.sum(np.maximum(r - x, 0.0) ** 2)
    p += np.sum(np.maximum(x + r - 1.0, 0.0) ** 2)
    p += np.sum(np.maximum(r - y, 0.0) ** 2)
    p += np.sum(np.maximum(y + r - 1.0, 0.0) ** 2)
    
    return p

def optimize_centers(c0, radii, maxiter=2000):
    """Optimize center positions to minimize overlap and boundary penalties."""
    bounds = [(0.0, 1.0)] * (2 * N)
    res = minimize(compute_center_penalty, c0.flatten(), args=(radii,),
                   method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    c_opt = res.x.reshape(-1, 2)
    return np.clip(c_opt, 1e-6, 1.0 - 1e-6)

def constraints_slsqp(x):
    """Inequality constraints for SLSQP: g(x) >= 0."""
    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]
    
    c = np.concatenate([
        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs
    ])
    
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    c = np.concatenate([c, dx[TRIL_IDX]**2 + dy[TRIL_IDX]**2 - dr[TRIL_IDX]**2])
    return c

def objective_slsqp(x):
    """Objective to minimize: negative sum of radii."""
    return -np.sum(x[2::3])

def get_bounds_slsqp():
    """Variable bounds for SLSQP."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)])
    return b

def make_hex_init(r0, ang, scale=1.0):
    """Generates a rotated hexagonal lattice initialization."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0 and len(pts) < N + 10:
            pts.append([x * scale, y * scale])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
    pts = np.array(pts[:N + 10])
    
    if ang != 0.0:
        c, s = np.cos(ang), np.sin(ang)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
        
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.15, 0.85, (1, 2))])
    return pts[:N]

def make_force_init(seed):
    """Force-directed layout to spread points evenly and push to boundaries."""
    np.random.seed(seed)
    pts = np.random.uniform(0.15, 0.85, (N, 2))
    for step in range(400):
        f = np.zeros_like(pts)
        lr = 0.015 * (1.0 - step / 400.0)
        for i in range(N):
            for j in range(i + 1, N):
                dx = pts[j] - pts[i]
                d = np.hypot(dx[0], dx[1])
                if d < 0.35 and d > 1e-5:
                    rep = 0.01 / (d**2 + 0.001)
                    f[i] -= dx * rep / d
                    f[j] += dx * rep / d
            for dim in range(2):
                if pts[i, dim] < 0.1: f[i, dim] += 0.05
                elif pts[i, dim] > 0.9: f[i, dim] -= 0.05
        pts += f * lr
        pts = np.clip(pts, 0.02, 0.98)
    return pts

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    np.random.seed(42)
    bounds_obj = get_bounds_slsqp()
    
    inits = []

    # Force-directed spreads

    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Phase 1: Alternating LP + L-BFGS-B Optimization
    for c0 in inits:
        c_curr = c0.copy()
        r_curr = solve_lp_radii(c_curr)
        
        # Alternating refinement
        for _ in range(8):
            r_curr = solve_lp_radii(c_curr)
            c_curr = optimize_centers(c_curr, r_curr, maxiter=1500)
            
        # SLSQP Polish
        x0 = np.zeros(3 * N)
        x0[0::3] = c_curr[:, 0]
        x0[1::3] = c_curr[:, 1]
        x0[2::3] = r_curr * 0.999
        
        try:
            res = minimize(objective_slsqp, x0, method='SLSQP', bounds=bounds_obj,
                           constraints={'type': 'ineq', 'fun': constraints_slsqp},
                           options={'maxiter': 25000, 'ftol': 1e-13, 'disp': False})
            
            if not np.isnan(res.fun):
                curr_sum = -res.fun
                if curr_sum > best_sum and np.min(constraints_slsqp(res.x)) >= -1e-5:
                    best_sum = curr_sum
                    best_centers = np.column_stack((res.x[0::3], res.x[1::3]))
                    best_radii = res.x[2::3].copy()
        except Exception:
            continue
            
    # Phase 2: Aggressive Topology Search (Deflation & Repositioning)
    if best_centers is not None:
        c_best = best_centers.copy()
        r_best = best_radii.copy()
        
        for cyc in range(60):
            c_p = c_best.copy()
            r_p = r_best.copy()
            
            # Cooling perturbation
            noise = 0.003 * (0.91 ** cyc)
            c_p += np.random.normal(0, noise, c_p.shape)

            # Deflate a random subset to break rigid contact networks
            k = np.random.randint(4, 10)
            subset = np.random.choice(N, size=k, replace=False)
            r_p[subset] *= 0.82
            
            # Re-optimize radii and centers for perturbed state
            r_p = solve_lp_radii(c_p)
            c_p = optimize_centers(c_p, r_p, maxiter=1000)
            
            # SLSQP polish on perturbed state
            x0 = np.zeros(3 * N)
            x0[0::3] = c_p[:, 0]
            x0[1::3] = c_p[:, 1]
            x0[2::3] = r_p * 0.998
            
            try:
                res = minimize(objective_slsqp, x0, method='SLSQP', bounds=bounds_obj,
                               constraints={'type': 'ineq', 'fun': constraints_slsqp},

        best_radii = final_r.copy()
        
    centers = best_centers
    radii = best_radii
    
    # Final strict validation repair against 1e-12 tolerance

        valid = True
        for i in range(N):

                for j in range(i + 1, N):
                    d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])

                        break

            break
            
        radii *= 0.9998
        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)

    return centers, radii, float(np.sum(radii))
