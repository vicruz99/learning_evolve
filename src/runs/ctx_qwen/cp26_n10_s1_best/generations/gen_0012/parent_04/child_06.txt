import numpy as np
from scipy.optimize import minimize, linprog
import warnings
warnings.filterwarnings('ignore')

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def solve_lp(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    m = n + n * (n - 1) // 2
    A_ub = np.zeros((m, n))
    b_ub = np.zeros(m)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        x, y = centers[i]
        lim = min(x, 1.0 - x, y, 1.0 - y)
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist)
            idx += 1
            
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x, -res.fun
    return np.full(n, 1e-6), 0.0

def evaluate(centers):
    """Evaluate sum of radii for given centers."""
    centers = np.clip(centers, 1e-6, 1.0 - 1e-6)
    return solve_lp(centers)

def make_hex_init(r0, angle, scale=1.0):
    """Generates a rotated hexagonal lattice initialization."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0 and len(pts) < N + 10:
            pts.append([x * scale, y * scale])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        
    pts = np.array(pts[:N + 10])
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
        
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.1, 0.9, (1, 2))])
    return pts[:N]

def make_force_init(seed):
    """Force-directed layout to spread points evenly."""
    np.random.seed(seed)
    pts = np.random.uniform(0.15, 0.85, (N, 2))
    for _ in range(150):
        f = np.zeros_like(pts)
        for i in range(N):
            for j in range(i + 1, N):
                dx = pts[j] - pts[i]
                d = np.hypot(dx[0], dx[1])
                if d < 0.25 and d > 1e-5:
                    rep = 0.01 / (d**2 + 0.001)
                    f[i] -= dx * rep / d
                    f[j] += dx * rep / d
            for dim in range(2):
                if pts[i, dim] < 0.1: f[i, dim] += 0.05
                elif pts[i, dim] > 0.9: f[i, dim] -= 0.05
        pts += f * 0.03
        pts = np.clip(pts, 0.03, 0.97)
    return pts

def simulated_annealing(centers_init, steps=2500, init_temp=0.04):
    """Simulated Annealing on centers, evaluating radii via LP."""
    centers = centers_init.copy()
    radii, score = evaluate(centers)
    best_centers = centers.copy()
    best_radii = radii.copy()
    best_score = score
    temp = init_temp
    
    for step in range(steps):
        jump = temp * 0.7
        k = np.random.randint(1, 4)
        idx = np.random.choice(N, k, replace=False)
        
        new_centers = centers.copy()
        new_centers[idx] += np.random.uniform(-jump, jump, (k, 2))
        new_centers = np.clip(new_centers, 1e-5, 1.0 - 1e-5)
        
        new_radii, new_score = evaluate(new_centers)
        delta = new_score - score
        
        if delta > 0 or (temp > 1e-10 and np.random.rand() < np.exp(delta / temp)):
            centers = new_centers
            score = new_score
            radii = new_radii
            if score > best_score:
                best_score = score
                best_centers = centers.copy()
                best_radii = radii.copy()
        temp *= 0.9991
    return best_centers, best_radii, best_score

def slsqp_objective(x):
    return -np.sum(x[2::3])

def slsqp_constraints(x):
    xs, ys, rs = x[0::3], x[1::3], x[2::3]
    c = np.concatenate([xs - rs, 1.0 - xs - rs, ys - rs, 1.0 - ys - rs])
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    c = np.concatenate([c, np.hypot(dx[TRIL_IDX], dy[TRIL_IDX]) - dr[TRIL_IDX]])
    return c

def polish_slsqp(centers, radii, maxiter=8000):
    """Gradient-based refinement of the full variable space."""
    n = N
    x0 = np.zeros(3 * n)
    x0[0::3] = centers[:, 0]
    x0[1::3] = centers[:, 1]
    x0[2::3] = radii
    
    bounds = [(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)] * n
    cons = {'type': 'ineq', 'fun': slsqp_constraints}
    
    try:
        res = minimize(slsqp_objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'disp': False})
        if res.success and np.min(slsqp_constraints(res.x)) >= -1e-7:
            c = np.column_stack((res.x[0::3], res.x[1::3]))
            r = res.x[2::3]
            # Final LP snap to ensure radii are maximized for these exact centers
            _, s_val = evaluate(c)
            if s_val > np.sum(r) - 1e-7:
                _, r = evaluate(c)
            return c, r, np.sum(r)
    except Exception:
        pass
    return centers, radii, np.sum(radii)

def run_packing():

    best_centers = None
    best_radii = None

    for c_init in inits:

        if s_sa > best_score:
            best_score = s_sa
            best_centers = c_sa
            best_radii = r_sa
            
    # Phase 2: SLSQP Polish
    best_centers, best_radii, best_score = polish_slsqp(best_centers, best_radii)

    if best_centers is not None:
        c_def = best_centers.copy() + np.random.normal(0, 0.005, (N, 2))
        c_def = np.clip(c_def, 0.05, 0.95)

        if s_sa2 > best_score:
            best_centers = c_sa2
            best_radii = r_sa2

        c_pol, r_pol, s_pol = polish_slsqp(best_centers, best_radii)

            best_centers = c_pol
            best_radii = r_pol
            best_score = s_pol

    # Fallback initialization (should not be reached)
    if best_centers is None:
        best_centers = np.zeros((N, 2))
        for i in range(N):
            best_centers[i, 0] = 0.1 + (i % 5) * 0.18
            best_centers[i, 1] = 0.1 + (i // 5) * 0.16
        best_radii, best_score = evaluate(best_centers)

    # Final strict validation repair against 1e-12 tolerance
    centers = best_centers
    radii = best_radii
    for _ in range(100):
        valid = True
        for i in range(N):
            if radii[i] < 0 or centers[i, 0] < radii[i] - 1e-9 or centers[i, 0] > 1.0 - radii[i] + 1e-9 or \

                    d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])

        radii *= 0.9998
        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)

    return centers, radii, float(np.sum(radii))
