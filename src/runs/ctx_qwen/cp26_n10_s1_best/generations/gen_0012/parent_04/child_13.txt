import numpy as np
from scipy.optimize import minimize, linprog
import warnings
warnings.filterwarnings('ignore')

N = 26

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_ineq = n * 4 + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= x, 1-x, y, 1-y
    for i in range(n):
        x, y = centers[i]
        lims = [x, 1.0 - x, y, 1.0 - y]
        for lim in lims:
            A_ub[idx, i] = 1.0
            b_ub[idx] = max(0.0, lim)
            idx += 1
            
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist)
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return np.full(n, 0.01)

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-7, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)])
    return b

def constraints(x):
    """Returns all inequality constraints >= 0 (vectorized)."""
    xs, ys, rs = x[0::3], x[1::3], x[2::3]
    c = np.concatenate([
        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs
    ])
    
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    # Using hypot for stable gradients at contact
    tril = np.tril_indices(N, -1)
    c = np.concatenate([c, np.hypot(dx[tril], dy[tril]) - dr[tril]])
    return c

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def make_hex_init(r0, angle):
    """Generates a rotated hexagonal lattice initialization."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0 and len(pts) < N + 10:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        
    pts = np.array(pts[:N + 10])
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
        
    mask = (pts[:, 0] >= 0.05) & (pts[:, 0] <= 0.95) & (pts[:, 1] >= 0.05) & (pts[:, 1] <= 0.95)
    pts = pts[mask]
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.2, 0.8, (1, 2))])
    return pts[:N]

def force_spread(centers, steps=200):
    """Force-directed relaxation to spread points evenly."""
    pts = centers.copy()
    for step in range(steps):
        f = np.zeros_like(pts)
        lr = 0.015 * (1.0 - step / steps)
        for i in range(N):
            for j in range(i + 1, N):
                dx = pts[j] - pts[i]
                d = np.hypot(dx[0], dx[1])
                if d < 0.3 and d > 1e-4:
                    rep = 0.01 / (d**2 + 0.0005)
                    f[i] -= dx * rep / d
                    f[j] += dx * rep / d
            for dim in range(2):
                if pts[i, dim] < 0.1: f[i, dim] += 0.04
                elif pts[i, dim] > 0.9: f[i, dim] -= 0.04
        pts += f * lr
        pts = np.clip(pts, 0.02, 0.98)
    return pts

def hill_climbing(centers, steps=3000, init_step=0.025):
    """Optimizes centers by local search using LP radii evaluation."""
    pts = centers.copy()
    best_val = np.sum(solve_lp_radii(pts))
    
    for i in range(steps):
        step = init_step * (0.9985 ** i)
        idx = np.random.randint(N)
        new_pts = pts.copy()
        new_pts[idx] += np.random.normal(0, step, 2)
        new_pts = np.clip(new_pts, 0.002, 0.998)
        
        curr_val = np.sum(solve_lp_radii(new_pts))
        delta = curr_val - best_val
        
        # Accept if better, or with SA probability
        if delta > 0 or (step > 1e-5 and np.random.rand() < np.exp(delta / (step + 1e-9))):
            pts = new_pts
            if delta > 0:
                best_val = curr_val
    return pts, best_val

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    bounds_obj = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -1.0
    best_x = None
    
    # Phase 1: Diverse Initial Configurations
    inits = []
    for ang in np.linspace(-0.35, 0.35, 9):
        inits.append(make_hex_init(0.095, ang))
    for s in range(12):
        np.random.seed(s + 100)
        inits.append(force_spread(np.random.uniform(0.15, 0.85, (N, 2)), steps=250))

    # Phase 2: Hill-Climbing on Centers for each initialization
    for c_init in inits:
        c_opt, val = hill_climbing(c_init, steps=2500, init_step=0.02)
        if val > best_sum:
            best_sum = val
            best_centers = c_opt.copy()

    # Convert best centers to full optimization vector
    if best_centers is not None:
        x0 = np.zeros(3 * N)
        x0[0::3] = best_centers[:, 0]
        x0[1::3] = best_centers[:, 1]
        x0[2::3] = solve_lp_radii(best_centers)
        
        # Phase 3: SLSQP Joint Refinement
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds_obj, constraints=cons,
                           options={'maxiter': 15000, 'ftol': 1e-13, 'disp': False})
            if not np.isnan(res.fun):
                x0 = res.x.copy()
                centers_p = np.column_stack((x0[0::3], x0[1::3]))
                r_p = solve_lp_radii(centers_p)
                if np.sum(r_p) > best_sum:
                    best_sum = np.sum(r_p)
                    best_x = x0.copy()
        except Exception:
            pass

    # Phase 4: Deflation & Topology Search to escape rigid local minima
    if best_x is not None:
        for cyc in range(40):
            x_p = best_x.copy()
            # Shrink random subset of radii to create slack
            subset = np.random.choice(N, size=max(4, N // 3), replace=False)
            x_p[subset * 3 + 2] *= 0.80
            
            # Perturb corresponding centers inward
            for idx in subset:
                cx, cy = x_p[idx*3], x_p[idx*3+1]
                x_p[idx*3] = cx * 0.82 + 0.09
                x_p[idx*3+1] = cy * 0.82 + 0.09
                
            # Project to feasible bounds
            for i in range(N):
                r = max(1e-6, x_p[3 * i + 2])
                x_p[3 * i] = np.clip(x_p[3 * i], r, 1.0 - r)
                x_p[3 * i + 1] = np.clip(x_p[3 * i + 1], r, 1.0 - r)
                x_p[3 * i + 2] = r

            try:
                res = minimize(objective, x_p, method='SLSQP', bounds=bounds_obj, constraints=cons,
                               options={'maxiter': 10000, 'ftol': 1e-13, 'disp': False})
                if not np.isnan(res.fun):
                    centers_p = np.column_stack((res.x[0::3], res.x[1::3]))
                    r_p = solve_lp_radii(centers_p)
                    x_p[2::3] = r_p
                    curr_sum = np.sum(r_p)
                    if curr_sum > best_sum and np.min(constraints(x_p)) >= -1e-6:
                        best_sum = curr_sum
                        best_x = x_p.copy()
            except Exception:
                pass

    # Phase 5: Fine-grained Coordinate Descent
    if best_x is not None:
        curr_centers = np.column_stack((best_x[0::3], best_x[1::3]))
        for _ in range(3):
            improved = True
            while improved:
                improved = False
                for i in range(N):
                    best_local = best_sum
                    for dx in [-0.0015, -0.0005, 0.0005, 0.0015]:
                        for dy in [-0.0015, -0.0005, 0.0005, 0.0015]:
                            new_c = curr_centers.copy()
                            new_c[i, 0] = np.clip(curr_centers[i, 0] + dx, 0.001, 0.999)
                            new_c[i, 1] = np.clip(curr_centers[i, 1] + dy, 0.001, 0.999)
                            r_new = solve_lp_radii(new_c)
                            s_new = np.sum(r_new)
                            if s_new > best_local + 1e-8:
                                curr_centers = new_c
                                best_local = s_new
                                best_x[0::3] = curr_centers[:, 0]
                                best_x[1::3] = curr_centers[:, 1]
                                best_x[2::3] = r_new
                                best_sum = s_new
                                improved = True
                                break
                        if improved: break
                    if improved: break

    # Fallback
    if best_x is None:
        best_x = np.zeros(3 * N)
        best_x[0::3] = np.tile(np.linspace(0.1, 0.9, 5), 6)[:N]
        best_x[1::3] = np.repeat(np.linspace(0.1, 0.9, 6), 5)[:N]
        best_x[2::3] = 0.06
        best_sum = np.sum(best_x[2::3])

    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3].copy()

    # Final strict validity adjustment against 1e-12 tolerance
    for _ in range(100):
        valid = True
        for i in range(N):
            if radii[i] < 0: valid = False; break
            if centers[i, 0] < radii[i] - 1e-9 or centers[i, 0] > 1.0 - radii[i] + 1e-9: valid = False; break
            if centers[i, 1] < radii[i] - 1e-9 or centers[i, 1] > 1.0 - radii[i] + 1e-9: valid = False; break
        if valid:
            for i in range(N):
                for j in range(i + 1, N):
                    d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                    if d < radii[i] + radii[j] - 1e-9: valid = False; break
                if not valid: break
        if valid: break
            
        radii *= 0.9999
        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)

    return centers, radii, float(np.sum(radii))
