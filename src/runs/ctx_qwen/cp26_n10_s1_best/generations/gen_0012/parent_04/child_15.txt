import numpy as np
from scipy.optimize import minimize, linprog
import warnings
warnings.filterwarnings('ignore')

N = 26

def get_lp_pair_indices():
    idx_list = []
    for i in range(N):
        for j in range(i + 1, N):
            idx_list.append((i, j))
    return np.array(idx_list)

PAIR_IDX = get_lp_pair_indices()

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    m = n + n * (n - 1) // 2
    A_ub = np.zeros((m, n))
    b_ub = np.zeros(m)
    
    idx = 0
    for i in range(n):
        x, y = centers[i]
        lim = min(x, 1.0 - x, y, 1.0 - y)
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    # Vectorized pairwise constraints construction
    i_idx, j_idx = PAIR_IDX[:, 0], PAIR_IDX[:, 1]
    dx = centers[i_idx, 0] - centers[j_idx, 0]
    dy = centers[i_idx, 1] - centers[j_idx, 1]
    dists = np.hypot(dx, dy)
    
    A_ub[idx:, i_idx] = 1.0
    A_ub[idx:, j_idx] = 1.0
    b_ub[idx:] = np.maximum(dists, 0.0)
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=[(0.0, None)] * n, method='highs')
        if res.success and np.isfinite(res.fun):
            return -res.fun, np.maximum(res.x, 1e-8)
    except Exception:
        pass
    return 0.0, np.full(n, 0.04)

def objective_joint(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints_joint(x):
    """Vectorized inequality constraints for SLSQP: g(x) >= 0."""
    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]
    
    c = np.concatenate([
        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs
    ])
    
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    c = np.concatenate([c, np.hypot(dx[PAIR_IDX[:,0], PAIR_IDX[:,1]], dy[PAIR_IDX[:,0], PAIR_IDX[:,1]]) - dr[PAIR_IDX[:,0], PAIR_IDX[:,1]]])
    return c

def get_bounds_joint():
    """Variable bounds: x,y in [0,1], r in [1e-7, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)])
    return b

def make_hex_init(r0, angle):
    """Generates a rotated hexagonal lattice initialization."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0 and len(pts) < N + 10:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        
    pts = np.array(pts[:N + 10])
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
        
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.1, 0.9, (1, 2))])
    return pts[:N]

def simulated_annealing_centers(initial_centers):
    """Optimize centers to maximize LP-computed sum of radii."""
    rng = np.random.default_rng(int(np.sum(initial_centers) * 10000) % (2**31))
    centers = initial_centers.copy()
    curr_sum, _ = solve_lp_radii(centers)
    best_centers = centers.copy()
    best_sum = curr_sum
    
    temp = 0.045
    decay = 0.9988
    steps = 6000
    
    for step in range(steps):
        temp *= decay
        if temp < 1e-7: break
        
        k = rng.integers(2, 6)
        idx = rng.choice(N, k, replace=False)
        new_centers = centers.copy()
        new_centers[idx] += rng.normal(0, np.sqrt(temp), (k, 2))
        new_centers = np.clip(new_centers, 0.01, 0.99)
        
        new_sum, _ = solve_lp_radii(new_centers)
        delta = new_sum - curr_sum
        
        if delta > 0 or (rng.random() < np.exp(delta / temp)):
            centers = new_centers
            curr_sum = new_sum
            if curr_sum > best_sum:
                best_sum = curr_sum
                best_centers = centers.copy()
                
    return best_centers, best_sum

def local_refinement_centers(centers):
    """Coordinate descent to fine-tune center positions."""
    curr_sum, _ = solve_lp_radii(centers)
    improved = True
    step_sizes = [0.005, 0.002, 0.0005]
    
    while improved:
        improved = False
        for step_size in step_sizes:
            for i in range(N):
                cx, cy = centers[i]
                for dx in [-step_size, step_size]:
                    for dy in [-step_size, step_size]:
                        new_c = centers.copy()
                        new_c[i, 0] = np.clip(cx + dx, 0.01, 0.99)
                        new_c[i, 1] = np.clip(cy + dy, 0.01, 0.99)
                        s, _ = solve_lp_radii(new_c)
                        if s > curr_sum + 1e-7:
                            centers = new_c
                            curr_sum = s
                            improved = True
    return centers, curr_sum

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    bounds_obj = get_bounds_joint()
    cons = {'type': 'ineq', 'fun': constraints_joint}
    
    best_sum = -1.0
    best_x = None
    
    # Phase 1: Generate Diverse Initial Configurations
    inits = []
    for ang in np.linspace(-0.35, 0.35, 7):
        inits.append(make_hex_init(0.092, ang))
        
    for s in range(4):
        np.random.seed(s + 50)
        inits.append(np.random.uniform(0.15, 0.85, (N, 2)))
        
    for s in range(3):
        np.random.seed(s + 100)
        pts = make_hex_init(0.088, 0.0)
        pts += np.random.normal(0, 0.025, pts.shape)
        inits.append(np.clip(pts, 0.05, 0.95))

    # Phase 2: SA + Local Refinement on each init
    for c_init in inits:
        c_sa, s_sa = simulated_annealing_centers(c_init)
        c_lr, s_lr = local_refinement_centers(c_sa)
        
        if s_lr > best_sum:
            best_sum = s_lr
            _, radii_vec = solve_lp_radii(c_lr)
            x0 = np.zeros(3 * N)
            x0[0::3] = c_lr[:, 0]
            x0[1::3] = c_lr[:, 1]
            x0[2::3] = np.maximum(radii_vec * 0.999, 1e-7)
            best_x = x0.copy()

    # Phase 3: Joint SLSQP Polish to squeeze remaining slack
    if best_x is not None:
        try:
            res = minimize(objective_joint, best_x, method='SLSQP', bounds=bounds_obj, 
                           constraints=cons, options={'maxiter': 20000, 'ftol': 1e-14, 'disp': False})
            if not np.isnan(res.fun):
                c_final = np.column_stack((res.x[0::3], res.x[1::3]))
                s_final, r_final = solve_lp_radii(c_final)
                if s_final > best_sum - 1e-9:
                    best_sum = s_final
                    best_x[0::3] = c_final[:, 0]
                    best_x[1::3] = c_final[:, 1]
                    best_x[2::3] = r_final
        except Exception:
            pass

    # Fallback (should not be reached)
    if best_x is None:
        best_x = np.zeros(3 * N)
        best_x[0::3] = np.tile(np.linspace(0.1, 0.9, 5), 6)[:N]
        best_x[1::3] = np.repeat(np.linspace(0.1, 0.9, 6), 5)[:N]
        best_x[2::3] = 0.06
        best_sum = np.sum(best_x[2::3])

    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3].copy()

    # Final strict validity adjustment against 1e-12 tolerance
    for _ in range(100):
        valid = True
        for i in range(N):
            if radii[i] < 0 or centers[i, 0] < radii[i] - 1e-9 or centers[i, 0] > 1.0 - radii[i] + 1e-9 or \
               centers[i, 1] < radii[i] - 1e-9 or centers[i, 1] > 1.0 - radii[i] + 1e-9:
                valid = False
                break
        if valid:
            for i in range(N):
                for j in range(i + 1, N):
                    d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                    if d < radii[i] + radii[j] - 1e-9:
                        valid = False
                        break
                if not valid:
                    break
        if valid:
            break
            
        radii *= 0.9999
        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)

    return centers, radii, float(np.sum(radii))
