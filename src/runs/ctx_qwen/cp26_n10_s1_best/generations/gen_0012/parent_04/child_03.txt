Looking at the past solutions, the best achieved sum of radii is ~2.636. The key patterns that work well are:

1. **Diverse initialization**: Hexagonal lattices with various rotations/scales, force-spread random configurations, corner-focused configs
2. **LP for radii**: Given fixed centers, Linear Programming optimally computes radii
3. **SLSQP for joint optimization**: Optimizes centers and radii together
4. **Simulated annealing on centers**: Perturb centers and recompute radii via LP to escape local minima
5. **Coordinate descent**: Fine-tune individual circle positions
6. **Deflation strategy**: Periodically shrink radii to break rigid contact networks

My strategy:
- Generate many diverse initial configs (hexagonal with rotations, force-spread, corner-focused)
- Run SLSQP from each start with LP initialization
- Use simulated annealing on centers with LP-based evaluation
- Apply iterative deflation + SLSQP restarts to escape local minima
- Use coordinate descent for final fine-tuning
- Apply final LP squeeze to maximize radii

import numpy as np
from scipy.optimize import minimize, linprog
import warnings
warnings.filterwarnings('ignore')

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Returns all inequality constraints >= 0."""
    xs, ys, rs = x[0::3], x[1::3], x[2::3]
    c = np.concatenate([
        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs
    ])
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    c = np.concatenate([c, np.hypot(dx[TRIL_IDX], dy[TRIL_IDX]) - dr[TRIL_IDX]])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-7, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)])
    return b

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist)
            idx += 1
            
    bounds_lp = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_lp, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-7)
    except Exception:
        pass
    return np.full(n, 0.05)

def make_hex_init(r0, angle):
    """Generates a rotated hexagonal lattice initialization."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 15:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0 and len(pts) < N + 15:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        
    pts = np.array(pts[:N + 15])
    
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
    
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.15, 0.85, (1, 2))])
    return pts[:N]

def make_force_init(seed, steps=300):
    """Force-directed layout to spread points evenly and push to boundaries."""
    np.random.seed(seed)
    pts = np.random.uniform(0.15, 0.85, (N, 2))
    for step in range(steps):
        f = np.zeros_like(pts)
        lr = 0.02 * (1.0 - step / steps)
        for i in range(N):
            for j in range(i + 1, N):
                dx = pts[j] - pts[i]
                d = np.hypot(dx[0], dx[1])
                if d < 0.3 and d > 1e-5:
                    rep = 0.01 / (d**2 + 0.001)
                    f[i] -= dx * rep / d
                    f[j] += dx * rep / d
            for dim in range(2):
                if pts[i, dim] < 0.08:
                    f[i, dim] += 0.06
                elif pts[i, dim] > 0.92:
                    f[i, dim] -= 0.06
        pts += f * lr
        pts = np.clip(pts, 0.02, 0.98)
    return pts

def make_corner_init():
    """Generates an initialization focused on corners and edges."""
    pts = []
    r = 0.085
    pts.append([r, r])
    pts.append([1 - r, r])
    pts.append([r, 1 - r])
    pts.append([1 - r, 1 - r])
    pts.append([0.5, r])
    pts.append([0.5, 1 - r])
    pts.append([r, 0.5])
    pts.append([1 - r, 0.5])
    pts.append([0.5, 0.5])
    
    y = r + 0.04
    row = 0
    while len(pts) < N:
        x = r + 0.04 if row % 2 == 0 else r + 0.16
        while x + r <= 1.0 and len(pts) < N:
            pts.append([x, y])
            x += 0.20
        y += 0.13
        row += 1
    
    while len(pts) < N:
        pts.append(list(np.random.uniform(0.1, 0.9, 2)))
    
    pts = np.array(pts[:N])
    pts += np.random.uniform(-0.002, 0.002, pts.shape)
    return pts

def project_to_feasible(x):
    """Project variables to strictly satisfy bounds."""
    x = x.copy()
    for i in range(N):
        r = max(1e-6, x[3 * i + 2])
        x[3 * i] = np.clip(x[3 * i], r, 1.0 - r)
        x[3 * i + 1] = np.clip(x[3 * i + 1], r, 1.0 - r)
        x[3 * i + 2] = r
    return x

def run_slsqp(x0, bounds_obj, cons, maxiter=15000):
    """Run SLSQP optimization."""
    try:
        res = minimize(objective, x0, method='SLSQP', bounds=bounds_obj,
                       constraints=cons, options={'maxiter': maxiter,
                       'ftol': 1e-13, 'disp': False})
        return res
    except Exception:
        return None

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    bounds_obj = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -1.0
    best_x = None
    
    # Phase 1: Generate Diverse Initial Configurations
    inits = []
    
    # Hexagonal lattices with various scales and rotations
    for scale in [0.85, 0.9, 0.95, 1.0, 1.05, 1.1, 1.15]:
        r0 = 0.09 * scale
        for ang in np.linspace(-0.45, 0.45, 15):
            inits.append(make_hex_init(r0, ang))
    
    # Force-spread random starts
    for s in range(15):
        inits.append(make_force_init(s, steps=350))
    
    # Corner/edge focused
    for _ in range(8):
        inits.append(make_corner_init())
    
    # Phase 2: Multi-start SLSQP Optimization
    for c_init in inits:
        x0 = np.zeros(3 * N)
        x0[0::3] = c_init[:, 0]
        x0[1::3] = c_init[:, 1]
        
        # LP initialization for radii
        r_lp = solve_lp_radii(c_init)
        x0[2::3] = np.maximum(r_lp * 0.995, 1e-6)
        x0 = project_to_feasible(x0)
        
        # SLSQP refinement
        res = run_slsqp(x0, bounds_obj, cons, maxiter=15000)
        if res is not None and not np.isnan(res.fun):
            centers_p = np.column_stack((res.x[0::3], res.x[1::3]))
            r_p = solve_lp_radii(centers_p)
            if r_p is not None:
                x_p = res.x.copy()
                x_p[2::3] = r_p
                x_p = project_to_feasible(x_p)
                curr_sum = np.sum(r_p)
                vals = constraints(x_p)
                if curr_sum > best_sum and np.min(vals) >= -1e-6:
                    best_sum = curr_sum
                    best_x = x_p.copy()
    
    # Phase 3: Simulated Annealing on Centers with LP Radii
    if best_x is not None:
        curr_centers = np.column_stack((best_x[0::3], best_x[1::3]))
        best_val = np.sum(solve_lp_radii(curr_centers))
        best_centers_sa = curr_centers.copy()
        
        temp = 0.025
        for i in range(1000):
            temp *= 0.9985
            k = np.random.randint(2, 7)
            idx = np.random.choice(N, k, replace=False)
            new_centers = curr_centers.copy()
            new_centers[idx] += np.random.normal(0, temp, (k, 2))
            new_centers = np.clip(new_centers, 1e-3, 1.0 - 1e-3)
            
            new_r = solve_lp_radii(new_centers)
            new_val = np.sum(new_r)
            delta = new_val - best_val
            
            if delta > 0 or (temp > 1e-12 and np.random.rand() < np.exp(delta / (temp + 1e-12))):
                curr_centers = new_centers
                if new_val > best_val:
                    best_val = new_val
                    best_centers_sa = new_centers.copy()
        
        if best_val > best_sum:
            best_sum = best_val
            x_sa = np.zeros(3 * N)
            x_sa[0::3] = best_centers_sa[:, 0]
            x_sa[1::3] = best_centers_sa[:, 1]
            x_sa[2::3] = solve_lp_radii(best_centers_sa)
            best_x = x_sa.copy()
    
    # Phase 4: Iterative Deflation & SLSQP Restart
    if best_x is not None:
        for cyc in range(70):
            x0 = best_x.copy()
            
            # Shrink radii to create slack
            shrink = 0.70 + 0.30 * (cyc / 70.0)
            x0[2::3] *= shrink
            
            # Perturb centers
            noise = 0.003 * (0.91 ** cyc)
            x0[0::3] += np.random.normal(0, noise, N)
            x0[1::3] += np.random.normal(0, noise, N)
            
            # Deflate random subset
            k = np.random.randint(3, 8)
            subset = np.random.choice(N, size=k, replace=False)
            x0[subset * 3 + 2] *= 0.80
            
            x0 = project_to_feasible(x0)
            
            res = run_slsqp(x0, bounds_obj, cons, maxiter=10000)
            if res is not None and not np.isnan(res.fun):
                centers_p = np.column_stack((res.x[0::3], res.x[1::3]))
                r_p = solve_lp_radii(centers_p)
                if r_p is not None:
                    x_p = res.x.copy()
                    x_p[2::3] = np.maximum(r_p * 0.998, 1e-6)
                    x_p = project_to_feasible(x_p)
                    curr_sum = np.sum(x_p[2::3])
                    vals = constraints(x_p)
                    if curr_sum > best_sum and np.min(vals) >= -1e-6:
                        best_sum = curr_sum
                        best_x = x_p.copy()
    
    # Phase 5: Coordinate Descent Fine-Tuning
    if best_x is not None:
        curr_centers = np.column_stack((best_x[0::3], best_x[1::3]))
        improved = True
        outer_iter = 0
        while improved and outer_iter < 30:
            improved = False
            outer_iter += 1
            for i in range(N):
                best_local_sum = np.sum(best_x[2::3])
                found = False
                for dx in [-0.006, -0.003, -0.0015, -0.0005, 0.0005, 0.0015, 0.003, 0.006]:
                    if found:
                        break
                    for dy in [-0.006, -0.003, -0.0015, -0.0005, 0.0005, 0.0015, 0.003, 0.006]:
                        if dx == 0 and dy == 0:
                            continue
                        new_centers = curr_centers.copy()
                        new_centers[i, 0] = np.clip(curr_centers[i, 0] + dx, 0.005, 0.995)
                        new_centers[i, 1] = np.clip(curr_centers[i, 1] + dy, 0.005, 0.995)
                        r_new = solve_lp_radii(new_centers)
                        s_new = np.sum(r_new)
                        if s_new > best_local_sum + 1e-8:
                            curr_centers = new_centers
                            best_local_sum = s_new
                            best_x[0::3] = curr_centers[:, 0]
                            best_x[1::3] = curr_centers[:, 1]
                            best_x[2::3] = r_new
                            best_sum = s_new
                            improved = True
                            found = True
                            break
    
    # Phase 6: Final LP Squeeze - iteratively compute best radii for current centers
    if best_x is not None:
        centers = np.column_stack((best_x[0::3], best_x[1::3]))
        radii = solve_lp_radii(centers)
        best_sum = np.sum(radii)
    
    # Fallback
    if best_x is None:
        best_x = np.zeros(3 * N)
        best_x[0::3] = np.tile(np.linspace(0.1, 0.9, 5), 6)[:N]
        best_x[1::3] = np.repeat(np.linspace(0.1, 0.9, 6), 5)[:N]
        best_x[2::3] = 0.06
        best_sum = np.sum(best_x[2::3])
    
    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3].copy()
    
    # Final strict validity adjustment against 1e-12 tolerance
    for _ in range(100):
        valid = True
        for i in range(N):
            if radii[i] < 0 or centers[i, 0] < radii[i] - 1e-9 or centers[i, 0] > 1.0 - radii[i] + 1e-9 or \
               centers[i, 1] < radii[i] - 1e-9 or centers[i, 1] > 1.0 - radii[i] + 1e-9:
                valid = False
                break
        if valid:
            for i in range(N):
                for j in range(i + 1, N):
                    d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                    if d < radii[i] + radii[j] - 1e-9:
                        valid = False
                        break
                if not valid:
                    break
        if valid:
            break
        
        radii *= 0.9998
        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)
    
    return centers, radii, float(np.sum(radii))
