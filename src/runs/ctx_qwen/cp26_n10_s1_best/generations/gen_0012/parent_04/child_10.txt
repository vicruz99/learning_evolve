

<strategy>
To maximize the sum of radii for 26 circles in a unit square, we will employ a multi-stage optimization pipeline that directly optimizes the true objective function. The key insight is that for fixed centers, the optimal radii are given by a Linear Program (LP). This transforms the problem into optimizing only the $2N$ center coordinates, where the objective function is the sum of radii returned by the LP. This approach avoids the squared-distance approximations and constraint linearizations that can trap gradient-based methods.

Pipeline:
1. **Diverse Initialization**: Generate multiple starting configurations (rotated hexagonal lattices, staggered grids, corner-focused, and force-directed random layouts).
2. **Nelder-Mead Center Optimization**: For each start, use derivative-free Nelder-Mead to maximize the LP-derived sum of radii. This handles the non-smooth landscape of the objective well.
3. **Simulated Annealing (SA)**: Apply SA on the centers to escape local minima. The temperature schedule allows early exploration and late exploitation.
4. **Coordinate Descent**: A deterministic local search that moves one circle at a time on a fine grid to find local improvements in the LP sum.
5. **SLSQP Joint Polish**: Finally, run SLSQP on both centers and radii jointly starting from the best configuration found. This smoothly resolves any residual numerical drift and strictly enforces boundary/overlap constraints.
6. **Strict Validation Repair**: A final shrinkage loop guarantees the `validate_packing` function returns True by compensating for floating-point epsilon errors.

This hybrid approach leverages the exactness of LP, the robustness of derivative-free global search, and the precision of constrained gradient optimization.

import numpy as np
from scipy.optimize import linprog, minimize
import warnings
warnings.filterwarnings('ignore')

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = centers.shape[0]
    
    # Distance matrix
    dx = centers[:, 0, None] - centers[:, 0]
    dy = centers[:, 1, None] - centers[:, 1]
    dists = np.hypot(dx, dy)
    
    # Boundary limits: r_i <= min(x, 1-x, y, 1-y)
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]), 
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    lims = np.maximum(lims, 0.0)
    
    # Pairwise distances for i < j
    idx_i, idx_j = np.tril_indices(n, -1)
    d_ij = np.maximum(dists[idx_i, idx_j], 1e-9)
    
    # Build LP: max sum(r_i) s.t. r_i <= lims, r_i + r_j <= d_ij
    A_ub = np.zeros((n + len(idx_i), n))
    b_ub = np.zeros(n + len(idx_i))
    
    A_ub[:n, np.arange(n)] = 1.0
    b_ub[:n] = lims
    
    A_ub[n:, idx_i] = 1.0
    A_ub[n:, idx_j] = 1.0
    b_ub[n:] = d_ij
    
    bounds = [(0.0, None)] * n
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return np.full(n, 0.02)

def lp_objective(c_flat):
    """Objective for center optimization: negative sum of radii from LP."""
    c = c_flat.reshape(-1, 2)
    c = np.clip(c, 1e-4, 1.0 - 1e-4)
    r = solve_lp_radii(c)
    return -np.sum(r)

def make_hex_init(r0, angle):
    """Generates a rotated hexagonal lattice initialization."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0 and len(pts) < N + 10:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        
    pts = np.array(pts[:N + 10])
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
        
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.1, 0.9, (1, 2))])
    return pts[:N]

def make_staggered_init():
    """Generates a row-based initialization adapted to square boundaries."""
    pts = []
    rows = [6, 5, 6, 5, 4]
    y_step = 0.18
    y = 0.1
    for r_count in rows:
        x_step = 0.9 / (r_count - 1) if r_count > 1 else 0.5
        for k in range(r_count):
            x = 0.05 + k * x_step
            pts.append([x, y])
            if len(pts) >= N:
                break
        y += y_step
        if len(pts) >= N:
            break
    while len(pts) < N:
        pts.append(np.random.uniform(0.1, 0.9, 2))
    return np.array(pts[:N])

def make_corner_init():
    """Generates an initialization focused on corners and edges."""
    pts = np.zeros((N, 2))
    r = 0.09
    pts[0] = [r, r]
    pts[1] = [1-r, r]
    pts[2] = [r, 1-r]
    pts[3] = [1-r, 1-r]
    pts[4] = [0.5, r]
    pts[5] = [0.5, 1-r]
    pts[6] = [r, 0.5]
    pts[7] = [1-r, 0.5]
    pts[8] = [0.5, 0.5]
    idx = 9
    y = r + 0.04
    row = 0
    while idx < N and y + r <= 1.0:
        x = r + 0.04 if row % 2 == 0 else r + 0.16
        while idx < N and x + r <= 1.0:
            pts[idx] = [x, y]
            idx += 1
            x += 0.20
        y += 0.13
        row += 1
    for i in range(idx, N):
        pts[i] = np.random.uniform(0.1, 0.9, 2)
    return pts

def force_layout(pts, steps=100):
    """Force-directed relaxation to spread points evenly."""
    p = pts.copy()
    for _ in range(steps):
        f = np.zeros_like(p)
        for i in range(N):
            for j in range(i + 1, N):
                dx = p[j] - p[i]
                d = np.hypot(dx[0], dx[1])
                if d < 0.25 and d > 1e-5:
                    rep = 0.015 / (d**2 + 1e-6)
                    f[i] -= dx * rep / d
                    f[j] += dx * rep / d
            for dim in range(2):
                if p[i, dim] < 0.1: f[i, dim] += 0.05
                elif p[i, dim] > 0.9: f[i, dim] -= 0.05
        p += f * 0.02
        p = np.clip(p, 0.02, 0.98)
    return p

def coord_descent(centers, sweeps=3):
    """Deterministic local search moving one circle at a time."""
    c = centers.copy()
    for _ in range(sweeps):
        improved = True
        while improved:
            improved = False
            for i in range(N):
                best_s = np.sum(solve_lp_radii(c))
                for dx in [-0.005, -0.002, -0.001, 0.001, 0.002, 0.005]:
                    for dy in [-0.005, -0.002, -0.001, 0.001, 0.002, 0.005]:
                        nc = c.copy()
                        nc[i, 0] = np.clip(c[i, 0] + dx, 0.005, 0.995)
                        nc[i, 1] = np.clip(c[i, 1] + dy, 0.005, 0.995)
                        s = np.sum(solve_lp_radii(nc))
                        if s > best_s + 1e-7:
                            c = nc
                            best_s = s
                            improved = True
                            break
                    if improved: break
    return c

def sa_optimize(centers, steps=800):
    """Simulated Annealing on centers to escape local minima."""
    curr = centers.copy()
    best = curr.copy()
    best_val = np.sum(solve_lp_radii(best))
    curr_val = best_val
    temp_start, temp_end = 0.02, 1e-5
    
    for i in range(steps):
        temp = temp_start * (temp_end / temp_start) ** (i / steps)
        k = np.random.randint(1, 4)
        idx = np.random.choice(N, k, replace=False)
        new = curr.copy()
        new[idx] += np.random.normal(0, temp, (k, 2))
        new = np.clip(new, 0.005, 0.995)
        
        new_val = np.sum(solve_lp_radii(new))
        delta = new_val - curr_val
        
        if delta > 0 or (np.random.rand() < np.exp(delta / (temp + 1e-9))):
            curr = new
            curr_val = new_val
            if curr_val > best_val:
                best_val = curr_val
                best = curr.copy()
    return best

def constraints_slsqp(x):
    """Inequality constraints for SLSQP: g(x) >= 0."""
    xs, ys, rs = x[0::3], x[1::3], x[2::3]
    c = np.concatenate([xs - rs, 1.0 - xs - rs, ys - rs, 1.0 - ys - rs])
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    c = np.concatenate([c, np.hypot(dx[TRIL_IDX], dy[TRIL_IDX]) - dr[TRIL_IDX]])
    return c

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    best_centers = None
    best_val = -1.0
    
    # Phase 1: Generate diverse initial configurations
    inits = []
    for ang in np.linspace(-0.4, 0.4, 9):
        inits.append(make_hex_init(0.092, ang))
    for s in range(6):
        inits.append(force_layout(np.random.uniform(0.15, 0.85, (N, 2)), steps=80))
    for _ in range(4):
        inits.append(make_staggered_init())
    for _ in range(4):
        inits.append(make_corner_init())
        
    # Phase 2: Optimize each initialization
    for pts in inits:
        # Nelder-Mead polish
        try:
            res_nm = minimize(lp_objective, pts.flatten(), method='Nelder-Mead', 
                              options={'maxiter': 300, 'fatol': 1e-9, 'xatol': 1e-9})
            c_nm = np.clip(res_nm.x.reshape(-1, 2), 0.005, 0.995)
            
            # Simulated Annealing
            c_sa = sa_optimize(c_nm, steps=600)
            
            # Coordinate Descent
            c_cd = coord_descent(c_sa, sweeps=2)
            
            curr_val = np.sum(solve_lp_radii(c_cd))
            if curr_val > best_val:
                best_val = curr_val
                best_centers = c_cd.copy()
        except Exception:
            continue
            
    # Fallback
    if best_centers is None:
        best_centers = np.tile(np.linspace(0.1, 0.9, 6).reshape(-1, 1), (1, 2)).reshape(-1, 2)[:N]
        best_val = np.sum(solve_lp_radii(best_centers))

    # Phase 3: SLSQP Joint Polish (centers + radii) for precise constraint satisfaction
    x0 = np.zeros(3 * N)
    x0[0::3] = best_centers[:, 0]
    x0[1::3] = best_centers[:, 1]
    x0[2::3] = solve_lp_radii(best_centers)
    
    bounds_obj = [(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints_slsqp}
    
    try:
        res_sl = minimize(lambda x: -np.sum(x[2::3]), x0, method='SLSQP', 
                          bounds=bounds_obj, constraints=cons,
                          options={'maxiter': 15000, 'ftol': 1e-14, 'disp': False})
        if not np.isnan(res_sl.fun):
            final_c = np.column_stack((res_sl.x[0::3], res_sl.x[1::3]))
            final_r = solve_lp_radii(final_c)
            if np.sum(final_r) >= np.sum(x0[2::3]) - 1e-8:
                best_centers = final_c
                best_radii = final_r
                best_val = np.sum(best_radii)
            else:
                best_radii = res_sl.x[2::3]
        else:
            best_radii = solve_lp_radii(best_centers)
    except Exception:
        best_radii = solve_lp_radii(best_centers)

    # Phase 4: Final strict validity adjustment against 1e-12 tolerance
    for _ in range(100):
        valid = True
        for i in range(N):
            if best_radii[i] < 0:
                valid = False; break
            if best_centers[i, 0] < best_radii[i] - 1e-9 or best_centers[i, 0] > 1.0 - best_radii[i] + 1e-9:
                valid = False; break
            if best_centers[i, 1] < best_radii[i] - 1e-9 or best_centers[i, 1] > 1.0 - best_radii[i] + 1e-9:
                valid = False; break
        if valid:
            for i in range(N):
                for j in range(i + 1, N):
                    d = np.hypot(best_centers[i, 0] - best_centers[j, 0], 
                                 best_centers[i, 1] - best_centers[j, 1])
                    if d < best_radii[i] + best_radii[j] - 1e-9:
                        valid = False; break
                if not valid: break
        if valid: break
            
        # Minimal shrinkage to recover strict feasibility
        best_radii *= 0.9999
        best_centers[:, 0] = np.clip(best_centers[:, 0], best_radii, 1.0 - best_radii)
        best_centers[:, 1] = np.clip(best_centers[:, 1], best_radii, 1.0 - best_radii)

    return best_centers, best_radii, float(np.sum(best_radii))
