import numpy as np
from scipy.optimize import linprog

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], 
                  centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    dx = centers[:, 0, None] - centers[:, 0]
    dy = centers[:, 1, None] - centers[:, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    i_idx, j_idx = TRIL_IDX
    for k in range(len(i_idx)):
        A_ub[idx, i_idx[k]] = 1.0
        A_ub[idx, j_idx[k]] = 1.0
        b_ub[idx] = max(0.0, dists[i_idx[k], j_idx[k]])
        idx += 1
        
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-7)
    except Exception:
        pass
    return np.full(n, 0.05)

def force_relax(centers, radii, steps=300):
    """Position-based force simulation to resolve overlaps and spread circles."""
    pts = centers.copy()
    
    for step in range(steps):
        forces = np.zeros_like(pts)
        # Decaying learning rate for stability
        lr = 0.03 * (1.0 - step / steps) + 0.005
        
        for i in range(N):
            # Pairwise repulsion
            for j in range(i + 1, N):
                diff = pts[j] - pts[i]
                dist = np.linalg.norm(diff) + 1e-8
                target = radii[i] + radii[j]
                
                # Strong repulsion for overlaps, gentle repulsion for close neighbors
                if dist < target + 0.10:
                    gap = target + 0.02 - dist
                    f = 8.0 * gap / dist
                    forces[i] -= f * diff
                    forces[j] += f * diff
                    
            # Wall repulsion to keep circles inside and push to boundaries
            for d in range(2):
                if pts[i, d] < radii[i]:
                    forces[i, d] += 12.0 * (radii[i] - pts[i, d])
                elif pts[i, d] > 1.0 - radii[i]:
                    forces[i, d] -= 12.0 * (pts[i, d] - (1.0 - radii[i]))
                    
        pts += forces * lr
        pts = np.clip(pts, 1e-4, 1.0 - 1e-4)
    return pts

def make_hex_init(seed, angle=0.0, r0=0.09):
    """Generate a rotated hexagonal lattice initialization."""
    np.random.seed(seed)
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 5:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        if y > 1.0 + r0:
            break
            
    pts = np.array(pts[:N + 5])
    
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
        
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & \
           (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.2, 0.8, (1, 2))])
        
    return np.clip(pts[:N], 0.01, 0.99)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    np.random.seed(42)
    best_sum = -1.0
    best_centers = None

        inits.append(c)

    for s in range(15):
        c = np.random.uniform(0.1, 0.9, (N, 2))
        inits.append(c)
        
    # 3. Corner/Edge biased starts
    for s in range(10):
        c = np.random.uniform(0.1, 0.9, (N, 2))
        c[:4] = [[0.14, 0.14], [0.86, 0.14], [0.14, 0.86], [0.86, 0.86]]
        inits.append(c)

    # Phase 1: Gradual Expansion + Relaxation for each initialization

                radii = new_radii

                break

                        break

        radii *= 0.9995
        for i in range(N):
            centers[i, 0] = np.clip(centers[i, 0], radii[i], 1.0 - radii[i])
            centers[i, 1] = np.clip(centers[i, 1], radii[i], 1.0 - radii[i])
            
    return centers, radii, float(np.sum(radii))
