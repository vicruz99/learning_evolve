import numpy as np
from scipy.optimize import minimize, linprog

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Returns all inequality constraints g(x) >= 0."""
    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]
    
    # Boundary constraints: 4*N
    c = np.concatenate([
        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs
    ])
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    i_idx, j_idx = TRIL_IDX
    dist_sq = dx[i_idx, j_idx]**2 + dy[i_idx, j_idx]**2
    r_sum_sq = dr[i_idx, j_idx]**2
    
    c = np.concatenate([c, dist_sq - r_sum_sq])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-7, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)])
    return b

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], 
                  centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    dx = centers[:, 0, None] - centers[:, 0]
    dy = centers[:, 1, None] - centers[:, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    i_idx, j_idx = TRIL_IDX
    for k in range(len(i_idx)):
        A_ub[idx, i_idx[k]] = 1.0
        A_ub[idx, j_idx[k]] = 1.0
        b_ub[idx] = max(0.0, dists[i_idx[k], j_idx[k]])
        idx += 1
        
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-7)
    except Exception:
        pass
    return None

def safe_radii(centers):
    """Compute strictly feasible initial radii based on nearest neighbors/walls."""
    n = centers.shape[0]
    radii = np.zeros(n)
    for i in range(n):
        min_d = min(centers[i, 0], 1.0 - centers[i, 0],
                    centers[i, 1], 1.0 - centers[i, 1])
        for j in range(n):
            if i != j:
                d = np.hypot(centers[i, 0] - centers[j, 0],
                             centers[i, 1] - centers[j, 1])
                if d < min_d:
                    min_d = d
        radii[i] = min_d * 0.45
    return radii

def make_hex_init(seed, angle=0.0, r0=0.095):
    """Generate a rotated hexagonal lattice."""
    np.random.seed(seed)
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 5:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        if y > 1.0 + r0:
            break
            
    pts = np.array(pts[:N + 5])
    
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
        
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & \
           (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.2, 0.8, (1, 2))])
        
    return np.clip(pts[:N] + np.random.uniform(-0.005, 0.005, (N, 2)), 0.02, 0.98)

def force_directed_layout(centers, steps=400):
    """Spread points evenly using repulsive forces and boundary attraction."""
    pts = centers.copy()
    for step in range(steps):
        forces = np.zeros_like(pts)
        lr = 0.02 * (1.0 - step / steps)
        
        for i in range(N):
            for j in range(i + 1, N):
                diff = pts[j] - pts[i]
                dist = np.linalg.norm(diff)
                if dist < 0.25 and dist > 1e-5:
                    f = 0.015 / (dist**2 + 0.001)
                    fx, fy = f * diff[0], f * diff[1]
                    forces[i] -= [fx, fy]
                    forces[j] += [fx, fy]
                    
            for d in range(2):
                if pts[i, d] < 0.05:
                    forces[i, d] += 0.05
                elif pts[i, d] > 0.95:
                    forces[i, d] -= 0.05
                    
        pts += forces * lr
        pts = np.clip(pts, 0.02, 0.98)
    return pts

def project_to_bounds(x):
    """Ensure optimization vector strictly respects bounds."""
    x = x.copy()
    for i in range(N):
        r = max(1e-7, x[3 * i + 2])
        x[3 * i] = np.clip(x[3 * i], r, 1.0 - r)
        x[3 * i + 1] = np.clip(x[3 * i + 1], r, 1.0 - r)
        x[3 * i + 2] = r
    return x

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -1.0
    best_x = None
    
    # Phase 1: Diverse Initializations
    inits = []
    
    # 1. Rotated hexagonal lattices
    for s in range(20):
        ang = np.random.uniform(-0.4, 0.4)
        r0 = np.random.uniform(0.085, 0.115)
        c = make_hex_init(s, ang, r0)
        inits.append(c)
        
    # 2. Corner/Edge biased starts
    for s in range(10):
        c = np.zeros((N, 2))
        c[0] = [0.12, 0.12]; c[1] = [0.88, 0.12]
        c[2] = [0.12, 0.88]; c[3] = [0.88, 0.88]
        c[4:] = np.random.uniform(0.2, 0.8, (N-4, 2))
        c = force_directed_layout(c, steps=300)
        inits.append(c)
        
    # 3. Pure random force-directed
    for s in range(15):
        c = np.random.uniform(0.15, 0.85, (N, 2))
        c = force_directed_layout(c, steps=500)
        inits.append(c)

    # Phase 2: Multi-start Optimization with LP Snap
    for c_init in inits:
        r_lp = solve_lp_radii(c_init)
        if r_lp is None:
            r_lp = safe_radii(c_init)
            
        x0 = np.zeros(3 * N)
        x0[0::3] = c_init[:, 0]
        x0[1::3] = c_init[:, 1]
        x0[2::3] = r_lp
        
        # Break exact symmetries
        x0 += np.random.normal(0, 1e-5, 3 * N)
        x0 = project_to_bounds(x0)
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, 
                           constraints=cons, 
                           options={'maxiter': 15000, 'ftol': 1e-14, 'disp': False})
            
            if not np.isnan(res.fun):
                curr_sum = -res.fun
                c_vals = constraints(res.x)
                if np.min(c_vals) >= -1e-6 and curr_sum > best_sum:
                    best_sum = curr_sum
                    best_x = res.x.copy()
        except Exception:
            pass

    if best_x is None:
        centers = force_directed_layout(np.random.uniform(0.2, 0.8, (N, 2)))
        radii = solve_lp_radii(centers) or safe_radii(centers)
        return centers, radii, float(np.sum(radii))
        
    # Phase 3: Simulated Annealing Topology Search
    centers = best_x.reshape(N, 3)[:, :2]
    radii = best_x[2::3].copy()
    current_sum = np.sum(radii)
    
    T = 0.008
    for step in range(2500):
        T *= 0.995
        i = np.random.randint(N)
        old_pos = centers[i].copy()
        
        # Perturb center
        centers[i] += np.random.normal(0, T, 2)
        centers[i] = np.clip(centers[i], radii[i], 1.0 - radii[i])
        
        new_radii = solve_lp_radii(centers)
        if new_radii is not None:
            new_sum = np.sum(new_radii)
            delta = new_sum - current_sum
            if delta > 0 or np.random.random() < np.exp(delta / max(T, 1e-7)):
                radii = new_radii
                current_sum = new_sum
                # Update best if improved
                if current_sum > best_sum:
                    best_x = np.zeros(3 * N)
                    best_x[0::3] = centers[:, 0]
                    best_x[1::3] = centers[:, 1]
                    best_x[2::3] = radii
                    best_sum = current_sum
            else:
                centers[i] = old_pos
        else:
            centers[i] = old_pos

    # Phase 4: Iterative Deflation & SLSQP Refinement
    for rnd in range(30):
        noise_scale = 0.003 * (0.90 ** rnd)
        x0 = best_x + np.random.normal(0, noise_scale, 3 * N)
        
        # Deflate radii to allow centers to rearrange
        x0[2::3] *= 0.95
        
        # Randomly pick a subset to deflate more aggressively
        subset = np.random.choice(N, size=N // 3, replace=False)
        x0[subset * 3 + 2] *= 0.85
        
        x0 = project_to_bounds(x0)
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, 
                           constraints=cons, 
                           options={'maxiter': 12000, 'ftol': 1e-14, 'disp': False})
            
            if not np.isnan(res.fun):
                # LP refinement to snap radii to theoretical max for new centers
                c_cur = res.x.reshape(N, 3)[:, :2]
                r_lp = solve_lp_radii(c_cur)
                if r_lp is not None:
                    curr_sum = np.sum(r_lp)
                    if curr_sum > best_sum:
                        best_x = res.x.copy()
                        best_x[2::3] = r_lp
                        best_sum = curr_sum
        except Exception:
            pass

    # Extract results
    centers = best_x.reshape(N, 3)[:, :2]
    radii = best_x[2::3].copy()
    
    # Final strict validity check and minimal repair
    for _ in range(100):
        valid = True
        for i in range(N):
            if radii[i] < 0 or centers[i, 0] < radii[i] - 1e-9 or centers[i, 0] > 1.0 - radii[i] + 1e-9 or \
               centers[i, 1] < radii[i] - 1e-9 or centers[i, 1] > 1.0 - radii[i] + 1e-9:
                valid = False
                break
        if valid:
            for i in range(N):
                for j in range(i + 1, N):
                    d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                    if d < radii[i] + radii[j] - 1e-9:
                        valid = False
                        break
                if not valid:
                    break
        if valid:
            break
            
        # Gentle shrinkage to recover validity
        radii *= 0.9995
        for i in range(N):
            centers[i, 0] = np.clip(centers[i, 0], radii[i], 1.0 - radii[i])
            centers[i, 1] = np.clip(centers[i, 1], radii[i], 1.0 - radii[i])
            
    return centers, radii, float(np.sum(radii))
