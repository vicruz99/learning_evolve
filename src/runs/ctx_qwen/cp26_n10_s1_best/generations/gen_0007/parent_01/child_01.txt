import numpy as np
from scipy.optimize import minimize, linprog

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Returns all inequality constraints g(x) >= 0."""
    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]
    
    # Boundary constraints: 4*N
    c = np.concatenate([
        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs
    ])
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    i_idx, j_idx = TRIL_IDX
    dist_sq = dx[i_idx, j_idx]**2 + dy[i_idx, j_idx]**2
    r_sum_sq = dr[i_idx, j_idx]**2
    
    c = np.concatenate([c, dist_sq - r_sum_sq])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-7, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)])
    return b

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    num_ineq = n + N * (N - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros((num_ineq,))
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], 
                  centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    dx = centers[:, 0, None] - centers[:, 0]
    dy = centers[:, 1, None] - centers[:, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    i_idx, j_idx = TRIL_IDX
    for k in range(len(i_idx)):
        A_ub[idx, i_idx[k]] = 1.0
        A_ub[idx, j_idx[k]] = 1.0
        b_ub[idx] = max(0.0, dists[i_idx[k], j_idx[k]])
        idx += 1
        
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-7)
    except Exception:
        pass
    return None

def force_directed_layout(centers, steps=800):
    """Spread points evenly using repulsive forces and boundary attraction."""
    pts = centers.copy()
    for step in range(steps):
        forces = np.zeros_like(pts)
        lr = 0.015 * (1.0 - step / steps)
        
        for i in range(N):
            for j in range(i + 1, N):
                diff = pts[j] - pts[i]
                dist = np.linalg.norm(diff)
                if dist < 0.3 and dist > 1e-5:
                    f = 0.02 / (dist**2 + 0.002)
                    fx, fy = f * diff[0], f * diff[1]
                    forces[i] -= [fx, fy]
                    forces[j] += [fx, fy]
                    
            # Soft boundary repulsion
            for d in range(2):
                if pts[i, d] < 0.08:
                    forces[i, d] += 0.06
                elif pts[i, d] > 0.92:
                    forces[i, d] -= 0.06
                    
        pts += forces * lr
        pts = np.clip(pts, 0.02, 0.98)
    return pts

def make_hex_init(r0, angle=0.0):
    """Generate a rotated hexagonal lattice."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        if y > 1.0 + r0:
            break
            
    pts = np.array(pts[:N + 10])
    
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
        
    mask = (pts[:, 0] >= 0.01) & (pts[:, 0] <= 0.99) & \
           (pts[:, 1] >= 0.01) & (pts[:, 1] <= 0.99)
    pts = pts[mask]
    
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.15, 0.85, (1, 2))])
        
    return np.clip(pts[:N], 0.02, 0.98)

def make_staggered_init(offset_x=0.05, offset_y=0.05):
    """Generate staggered grid initialization."""
    pts = []
    # Row 1
    for x in np.linspace(0.12 + offset_x, 0.88 - offset_x, 6):
        pts.append([x, 0.12 + offset_y])
    # Row 2

    pts = np.array(pts[:N])
    pts += np.random.uniform(-0.01, 0.01, pts.shape)
    return np.clip(pts, 0.03, 0.97)

def project_to_bounds(x0):
    """Ensure optimization vector strictly respects bounds."""
    x0 = x0.copy()
    for i in range(N):
        r = max(1e-6, x0[3 * i + 2])
        x0[3 * i] = np.clip(x0[3 * i], r, 1.0 - r)
        x0[3 * i + 1] = np.clip(x0[3 * i + 1], r, 1.0 - r)
        x0[3 * i + 2] = r
    return x0

def run_optimization(x0, bounds, cons, maxiter=12000, do_lp=True):
    """Run SLSQP optimization with optional LP refinement."""
    try:
        res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'disp': False})
        
        if not np.isnan(res.fun):
            x_result = res.x.copy()
            if do_lp:
                centers_new = np.column_stack((x_result[0::3], x_result[1::3]))
                r_lp = solve_lp_radii(centers_new)
                if r_lp is not None:
                    x_result[2::3] = r_lp
                    x_result = project_to_bounds(x_result)
            return x_result
    except Exception:
        pass
    return None
