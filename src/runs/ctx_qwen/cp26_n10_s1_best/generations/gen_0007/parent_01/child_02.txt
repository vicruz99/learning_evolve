import numpy as np
from scipy.optimize import linprog, minimize

N = 26

def _build_lp_matrices():
    """Precompute the constant matrix A_ub for the Linear Programming formulation."""
    n_cons = 4 * N + N * (N - 1) // 2
    A_ub = np.zeros((n_cons, N))
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(N):
        A_ub[4*i, i] = 1.0
        A_ub[4*i+1, i] = 1.0
        A_ub[4*i+2, i] = 1.0
        A_ub[4*i+3, i] = 1.0
    # Pairwise constraints: r_i + r_j <= d_ij
    idx = 4 * N
    for i in range(N):
        for j in range(i + 1, N):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            idx += 1
    c_obj = -np.ones(N)  # Maximize sum(r) <=> Minimize -sum(r)
    bounds = [(0.0, None)] * N
    return A_ub, c_obj, bounds

A_LP, C_OBJ, BOUNDS_LP = _build_lp_matrices()

def solve_lp_and_compute_forces(centers):
    """
    Solves for optimal radii given fixed centers using LP.
    Returns radii, forces derived from dual variables (marginals), and sum of radii.
    """
    n_cons = A_LP.shape[0]
    b_ub = np.zeros(n_cons)
    
    xs = centers[:, 0]
    ys = centers[:, 1]
    # Set boundary constraint bounds
    for i in range(N):
        b_ub[4*i] = xs[i]
        b_ub[4*i+1] = 1.0 - xs[i]
        b_ub[4*i+2] = ys[i]
        b_ub[4*i+3] = 1.0 - ys[i]
        
    # Compute pairwise distances
    dx = centers[:, 0, None] - centers[:, 0]
    dy = centers[:, 1, None] - centers[:, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    # Set pairwise constraint bounds
    idx = 4 * N
    for i in range(N):
        for j in range(i + 1, N):
            b_ub[idx] = dists[i, j]
            idx += 1
            
    try:
        res = linprog(C_OBJ, A_ub=A_LP, b_ub=b_ub, bounds=BOUNDS_LP, method='highs',
                      options={'presolve': True, 'time_limit': 5.0})
        if not res.success:
            return None, None, None
            
        radii = res.x
        marginals = res.ineqlin.marginals
        # For minimization problems, active constraint marginals are typically >= 0
        marginals = np.maximum(marginals, 0.0)
        
        forces = np.zeros_like(centers)
        
        # Compute forces from boundary constraint marginals
        for i in range(N):
            # r <= x  -> relaxing pushes right (+)
            # r <= 1-x -> relaxing pushes left (-)
            forces[i, 0] += marginals[4*i] - marginals[4*i+1]
            # r <= y  -> relaxing pushes up (+)
            # r <= 1-y -> relaxing pushes down (-)
            forces[i, 1] += marginals[4*i+2] - marginals[4*i+3]
            
        # Compute repulsive forces from pairwise constraint marginals
        idx = 4 * N
        for i in range(N):
            for j in range(i + 1, N):
                mu = marginals[idx]
                if mu > 1e-12:
                    d = dists[i, j]
                    if d > 1e-9:
                        f = mu / d
                        dx_ij = centers[i, 0] - centers[j, 0]
                        dy_ij = centers[i, 1] - centers[j, 1]
                        forces[i, 0] += f * dx_ij
                        forces[i, 1] += f * dy_ij
                        forces[j, 0] -= f * dx_ij
                        forces[j, 1] -= f * dy_ij
                idx += 1
                
        return radii, forces, np.sum(radii)
    except Exception:
        return None, None, None

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    best_centers = None
    best_radii = None
    best_sum = -1.0
    
    # Phase 1: LP + Subgradient Ascent from multiple diverse starts
    for start in range(12):
        if start == 0:
            # Hexagonal lattice initialization (optimal packing structure)
            centers = np.zeros((N, 2))
            idx = 0
            r0 = 0.102
            y = r0
            row = 0
            while idx < N and y + r0 <= 1.0:
                x = r0 if row % 2 == 0 else 2.0 * r0
                while idx < N and x + r0 <= 1.0:
                    centers[idx] = [x + np.random.uniform(-0.005, 0.005), 
                                    y + np.random.uniform(-0.005, 0.005)]
                    idx += 1
                    x += 2.0 * r0
                y += np.sqrt(3.0) * r0
                row += 1
            while idx < N:
                centers[idx] = np.random.uniform(0.2, 0.8, 2)
                idx += 1
        else:
            # Random initialization with force-directed spread would be ideal, 
            # but simple uniform with clipping works well with the ascent
            centers = np.random.uniform(0.15, 0.85, (N, 2))
            
        centers = np.clip(centers, 0.02, 0.98)
        
        vel = np.zeros_like(centers)
        lr = 0.012
        
        for step in range(600):
            radii, forces, curr_sum = solve_lp_and_compute_forces(centers)
            if radii is None:
                break
                
            # Velocity Verlet-like integration with momentum
            vel = 0.5 * vel + forces * lr
            centers += vel
            centers = np.clip(centers, 0.005, 0.995)
            
            if curr_sum > best_sum:
                best_sum = curr_sum
                best_centers = centers.copy()
                best_radii = radii.copy()
                
            lr *= 0.997  # Cool down learning rate
            
        if curr_sum > best_sum:
            best_sum = curr_sum
            best_centers = centers.copy()
            best_radii = radii.copy()

    if best_centers is None:
        centers = np.random.uniform(0.2, 0.8, (N, 2))

    x0[2::3] = best_radii
    
    def obj_slsqp(x): return -np.sum(x[2::3])
    
    def cons_slsqp(x):
        cx, cy, cr = x[0::3], x[1::3], x[2::3]
        c = np.concatenate([cx-cr, 1.0-cx-cr, cy-cr, 1.0-cy-cr])
        dx = cx[:, None] - cx[None, :]
        dy = cy[:, None] - cy[None, :]
        dr = cr[:, None] + cr[None, :]
        i, j = np.tril_indices(N, -1)
        c = np.concatenate([c, dx[i,j]**2 + dy[i,j]**2 - dr[i,j]**2])
        return c
        
    try:
        res = minimize(obj_slsqp, x0, method='SLSQP',
                       constraints={'type': 'ineq', 'fun': cons_slsqp},
                       bounds=[(0,1),(0,1),(0,0.5)]*N,
                       options={'maxiter': 8000, 'ftol': 1e-13, 'disp': False})

            best_centers = np.column_stack((res.x[0::3], res.x[1::3]))
            best_radii = res.x[2::3]
            best_sum = -res.fun
    except Exception:
        pass
        
    # Final strict validity adjustment against 1e-12 tolerance
    for _ in range(100):
        valid = True
        for i in range(N):
            if best_radii[i] < 0 or best_centers[i,0] < best_radii[i]-1e-9 or \
               best_centers[i,0] > 1-best_radii[i]+1e-9 or \
               best_centers[i,1] < best_radii[i]-1e-9 or \
               best_centers[i,1] > 1-best_radii[i]+1e-9:
                valid = False; break
        if valid:
            for i in range(N):
                for j in range(i+1, N):
                    d = np.hypot(best_centers[i,0]-best_centers[j,0], best_centers[i,1]-best_centers[j,1])
                    if d < best_radii[i]+best_radii[j]-1e-9:
                        valid = False; break
                if not valid: break
        if valid: break
        # Minimal shrinkage to recover strict feasibility
        best_radii *= 0.9995
        best_centers[:,0] = np.clip(best_centers[:,0], best_radii, 1-best_radii)
        best_centers[:,1] = np.clip(best_centers[:,1], best_radii, 1-best_radii)
        
    return best_centers, best_radii, float(np.sum(best_radii))
