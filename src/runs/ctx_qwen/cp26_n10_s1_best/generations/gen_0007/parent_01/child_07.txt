import numpy as np
from scipy.optimize import minimize, linprog

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Returns all inequality constraints g(x) >= 0 (vectorized)."""
    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]
    
    # Boundary constraints: 4*N
    c = np.concatenate([
        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs
    ])
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    i_idx, j_idx = TRIL_IDX
    dist_sq = dx[i_idx, j_idx]**2 + dy[i_idx, j_idx]**2
    r_sum_sq = dr[i_idx, j_idx]**2
    
    return np.concatenate([c, dist_sq - r_sum_sq])

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-7, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)])
    return b

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + N * (N - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], 
                  centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    dx = centers[:, 0, None] - centers[:, 0]
    dy = centers[:, 1, None] - centers[:, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    i_idx, j_idx = TRIL_IDX
    for k in range(len(i_idx)):
        A_ub[idx, i_idx[k]] = 1.0
        A_ub[idx, j_idx[k]] = 1.0
        b_ub[idx] = max(0.0, dists[i_idx[k], j_idx[k]])
        idx += 1
        
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-7)
    except Exception:
        pass
    return None

def force_layout(centers, steps=500):
    """Spread points evenly using repulsive forces and boundary attraction."""
    pts = centers.copy()
    for step in range(steps):
        forces = np.zeros_like(pts)
        lr = 0.02 * (1.0 - step / steps)
        
        for i in range(N):
            for j in range(i + 1, N):
                diff = pts[j] - pts[i]
                dist = np.linalg.norm(diff)
                if dist < 0.3 and dist > 1e-5:
                    f = 0.012 / (dist**2 + 0.001)
                    fx, fy = f * diff[0], f * diff[1]
                    forces[i] -= [fx, fy]
                    forces[j] += [fx, fy]
                    
            for d in range(2):
                if pts[i, d] < 0.05:
                    forces[i, d] += 0.06
                elif pts[i, d] > 0.95:
                    forces[i, d] -= 0.06
                    
        pts += forces * lr
        pts = np.clip(pts, 0.02, 0.98)
    return pts

def make_hex_init(seed, angle=0.0, r0=0.095):
    """Generate a rotated hexagonal lattice."""
    np.random.seed(seed)
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 5:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        if y > 1.0 + r0:
            break
            
    pts = np.array(pts[:N + 5])
    
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
        
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & \
           (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.2, 0.8, (1, 2))])
        
    return np.clip(pts[:N] + np.random.uniform(-0.005, 0.005, (N, 2)), 0.02, 0.98)

def project_to_feasible(x0):
    """Ensure optimization vector strictly respects bounds."""
    x0 = x0.copy()
    for i in range(N):
        r = max(1e-6, x0[3 * i + 2])
        x0[3 * i] = np.clip(x0[3 * i], r, 1.0 - r)
        x0[3 * i + 1] = np.clip(x0[3 * i + 1], r, 1.0 - r)
        x0[3 * i + 2] = r
    return x0

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    np.random.seed(42)
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -1.0
    best_x = None
    
    # Phase 1: Diverse Initializations
    inits = []
    
    # 1. Rotated hexagonal lattices
    for s in range(20):
        ang = np.random.uniform(-0.4, 0.4)
        r0 = np.random.uniform(0.08, 0.12)
        c = make_hex_init(s, ang, r0)
        c = force_layout(c, steps=400)
        inits.append(c)
        
    # 2. Corner/Edge biased starts
    for s in range(10):
        c = np.zeros((N, 2))
        c[0] = [0.12, 0.12]; c[1] = [0.88, 0.12]
        c[2] = [0.12, 0.88]; c[3] = [0.88, 0.88]
        c[4:] = np.random.uniform(0.2, 0.8, (N-4, 2))
        c = force_layout(c, steps=500)
        inits.append(c)
        
    # 3. Pure random force-directed
    for s in range(15):
        c = np.random.uniform(0.15, 0.85, (N, 2))
        c = force_layout(c, steps=600)
        inits.append(c)

    # Phase 2: Multi-start Optimization with LP Snap
    for c_init in inits:
        r_lp = solve_lp_radii(c_init)
        if r_lp is None:
            r_lp = np.full(N, 0.08)
            
        x0 = np.zeros(3 * N)
        x0[0::3] = c_init[:, 0]
        x0[1::3] = c_init[:, 1]
        x0[2::3] = r_lp
        
        # Break exact symmetries & ensure feasibility
        x0 += np.random.normal(0, 1e-4, 3 * N)
        x0 = project_to_feasible(x0)
            
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, 
                           constraints=cons, 
                           options={'maxiter': 15000, 'ftol': 1e-13, 'disp': False})
            
            if not np.isnan(res.fun):
                curr_sum = -res.fun
                c_vals = constraints(res.x)
                if np.min(c_vals) >= -1e-6 and curr_sum > best_sum:
                    best_sum = curr_sum
                    best_x = res.x.copy()
        except Exception:
            pass

    if best_x is None:
        # Fallback
        centers = force_layout(np.random.uniform(0.2, 0.8, (N, 2)))
        radii = solve_lp_radii(centers) or np.full(N, 0.07)
        return centers, radii, float(np.sum(radii))
        
    # Phase 3: Uniform Scaling Push
    # Try to scale radii up uniformly to hit boundaries, then re-optimize
    centers_s = best_x.reshape(N, 3)[:, :2]
    radii_s = best_x[2::3].copy()
    # Find max scale factor
    min_ratio = 1.0
    for i in range(N):
        r = radii_s[i]
        if r < 1e-7: continue
        wall_lim = min(centers_s[i,0], 1-centers_s[i,0], centers_s[i,1], 1-centers_s[i,1])
        min_ratio = min(min_ratio, wall_lim / r)
        for j in range(N):
            if i == j: continue
            d = np.hypot(centers_s[i,0]-centers_s[j,0], centers_s[i,1]-centers_s[j,1])
            if d < 1e-7: continue
            min_ratio = min(min_ratio, d / (r + radii_s[j]))
            
    if min_ratio > 1.001:
        radii_s *= min_ratio * 0.99
        x0_s = np.zeros(3*N)
        x0_s[0::3] = centers_s[:, 0]
        x0_s[1::3] = centers_s[:, 1]
        x0_s[2::3] = radii_s
        x0_s = project_to_feasible(x0_s)
        try:
            res = minimize(objective, x0_s, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 10000, 'ftol': 1e-13, 'disp': False})

                if np.min(constraints(res.x)) >= -1e-6:
                    best_sum = -res.fun
                    best_x = res.x.copy()
        except: pass

    # Phase 4: Iterative Deflation & Topology Search
    for step in range(60):
        noise_scale = 0.0025 * (0.92 ** step)
        x0 = best_x + np.random.normal(0, noise_scale, 3 * N)
        
        # Deflate radii to allow centers to rearrange
        x0[2::3] *= 0.965
        
        # Randomly pick a subset to deflate more aggressively (breaks local grids)
        subset = np.random.choice(N, size=N // 3, replace=False)
        x0[subset * 3 + 2] *= 0.82
        
        x0 = project_to_feasible(x0)
            
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, 
                           constraints=cons, 
                           options={'maxiter': 10000, 'ftol': 1e-13, 'disp': False})
            
            if not np.isnan(res.fun):
                # LP refinement to snap radii to theoretical max for new centers
                c_cur = res.x.reshape(N, 3)[:, :2]
                r_lp = solve_lp_radii(c_cur)

                    cand_x[2::3] = r_lp
                    curr_sum = np.sum(r_lp)

        except Exception:
            pass
            
    # Extract results
    centers = best_x.reshape(N, 3)[:, :2]
    radii = best_x[2::3]
    
    # Final strict validity check and minimal repair
    for _ in range(150):
        valid = True
        for i in range(N):

                valid = False
                break
        if valid:
            for i in range(N):
                for j in range(i + 1, N):
                    d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                    if d < radii[i] + radii[j] - 1e-9:
                        valid = False
                        break
                if not valid:
                    break
        if valid:
            break
            
        # Gentle shrinkage to recover validity
        radii *= 0.9995
        for i in range(N):
            centers[i, 0] = np.clip(centers[i, 0], radii[i], 1.0 - radii[i])
            centers[i, 1] = np.clip(centers[i, 1], radii[i], 1.0 - radii[i])
            
    return centers, radii, float(np.sum(radii))
