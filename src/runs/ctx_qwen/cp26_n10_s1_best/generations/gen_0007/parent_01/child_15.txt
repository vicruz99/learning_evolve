2. **Global Optimization of Centers**: Use `differential_evolution` to globally optimize the center coordinates. 3. **Initialization**: Seed DE with multiple perturbed hexagonal lattice configurations. Hex grids are near-optimal for circle packing, so perturbing them gives DE a strong head start.
4. **Local Refinement**: After DE finds a high-quality center configuration, compute LP radii, then switch to `SLSQP` to jointly optimize centers and radii. This precisely pushes boundaries and resolves active constraints to machine precision.
5. **Perturbation & Deflation**: Run a scheduled perturbation loop on the best SLSQP result. Slightly deflating radii creates slack, allowing centers to jump over local barriers, followed by SLSQP to re-expand radii optimally.
6. **Strict Validation & Repair**: Apply a final iterative shrinkage routine to guarantee strict compliance with the validator's `1e-12` tolerance, ensuring no overlaps or boundary violations.

import numpy as np
from scipy.optimize import minimize, linprog, differential_evolution

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def objective_func(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Returns all inequality constraints g(x) >= 0."""
    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]
    
    # Boundary constraints: 4*N
    c = np.concatenate([
        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs
    ])
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    i_idx, j_idx = TRIL_IDX
    c = np.concatenate([c, dx[i_idx, j_idx]**2 + dy[i_idx, j_idx]**2 - dr[i_idx, j_idx]**2])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [0, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
    return b

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    bounds_val = np.minimum(np.minimum(centers[:,0], 1.0-centers[:,0]),
                            np.minimum(centers[:,1], 1.0-centers[:,1]))
    bounds_val = np.maximum(bounds_val, 0.0)
    
    # Pairwise distances
    dx = centers[:, 0, None] - centers[:, 0]
    dy = centers[:, 1, None] - centers[:, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    num_ineq = n + n*(n-1)//2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    for i in range(n):
        A_ub[idx, i] = 1.0
        b_ub[idx] = bounds_val[i]
        idx += 1
        
    i_idx, j_idx = TRIL_IDX
    for k in range(len(i_idx)):
        A_ub[idx, i_idx[k]] = 1.0
        A_ub[idx, j_idx[k]] = 1.0
        b_ub[idx] = dists[i_idx[k], j_idx[k]]
        idx += 1
        
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=[(0.0, None)]*n, method='highs')
        if res.success and np.min(res.x) >= -1e-6:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return np.full(n, 1e-4)

def de_objective(coords):
    """Objective for Differential Evolution: maximize sum of LP-optimized radii."""
    centers = coords.reshape(N, 2)
    radii = solve_lp_radii(centers)
    return -np.sum(radii)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    bounds_2d = [(0.0, 1.0)] * (2 * N)
    
    # Generate diverse hexagonal initializations for DE
    init_guesses = []
    for s in range(15):
        np.random.seed(s)
        r0 = 0.101 + np.random.uniform(-0.005, 0.005)
        pts = []
        y = r0
        row = 0
        while len(pts) < N + 3:
            x = r0 if row % 2 == 0 else 2.0 * r0
            while x <= 1.0 - r0 and len(pts) < N + 3:
                pts.append([x, y])
                x += 2.0 * r0
            y += np.sqrt(3.0) * r0
            row += 1
            if y > 1.0 + r0: break
        pts = np.array(pts[:N]) + np.random.normal(0, 0.015, (N, 2))
        pts = np.clip(pts, 0.01, 0.99)
        init_guesses.append(pts.flatten())
        
    # Phase 1: Global optimization of centers using Differential Evolution
    try:
        de_res = differential_evolution(de_objective, bounds_2d, seed=42,
                                        maxiter=250, popsize=15, mutation=(0.5, 1.0),
                                        recombination=0.9, tol=1e-7, polish=False,
                                        init=init_guesses)
        best_centers = de_res.x.reshape(N, 2)
    except Exception:
        best_centers = init_guesses[0].reshape(N, 2)
        
    best_radii = solve_lp_radii(best_centers)
    current_sum = np.sum(best_radii)
    
    # Phase 2: High-precision SLSQP refinement on (centers, radii)
    x0 = np.zeros(3 * N)
    x0[0::3] = best_centers[:, 0]
    x0[1::3] = best_centers[:, 1]
    x0[2::3] = best_radii
    
    # Project to strict feasible bounds
    for i in range(N):
        r = max(1e-6, x0[3*i+2])
        x0[3*i] = np.clip(x0[3*i], r, 1.0-r)
        x0[3*i+1] = np.clip(x0[3*i+1], r, 1.0-r)
        x0[3*i+2] = r
        
    cons = {'type': 'ineq', 'fun': constraints}
    bounds_3d = get_bounds()
    
    try:
        res = minimize(objective_func, x0, method='SLSQP', bounds=bounds_3d,
                       constraints=cons, options={'maxiter': 15000, 'ftol': 1e-13, 'disp': False})
        if not np.isnan(res.fun):
            c_vals = constraints(res.x)
            if np.min(c_vals) >= -1e-6:
                best_centers = res.x[0::3]
                best_radii = res.x[2::3]
                current_sum = -res.fun
    except Exception:
        pass
        
    # Phase 3: Perturbation & Deflation to escape local minima
    for step in range(35):
        x0 = np.column_stack((best_centers, best_radii)).flatten()
        noise = 0.0012 * (0.91 ** step)
        x0 += np.random.normal(0, noise, 3*N)
        x0[2::3] *= 0.975  # Deflate radii to create geometric slack
        
        for i in range(N):
            r = max(1e-5, x0[3*i+2])
            x0[3*i] = np.clip(x0[3*i], r, 1.0-r)
            x0[3*i+1] = np.clip(x0[3*i+1], r, 1.0-r)
            x0[3*i+2] = r
            
        try:
            res = minimize(objective_func, x0, method='SLSQP', bounds=bounds_3d,
                           constraints=cons, options={'maxiter': 8000, 'ftol': 1e-13, 'disp': False})
            if not np.isnan(res.fun):
                c_vals = constraints(res.x)
                if np.min(c_vals) >= -1e-6 and -res.fun > current_sum:
                    best_centers = res.x[0::3]
                    best_radii = res.x[2::3]
                    current_sum = -res.fun
        except Exception:
            pass
            
    centers = best_centers.reshape(N, 2)
    radii = best_radii
    
    # Final strict validation and minimal numerical repair
    for _ in range(100):
        valid = True
        for i in range(N):
            if radii[i] < 0 or centers[i,0] < radii[i]-1e-9 or centers[i,0] > 1.0-radii[i]+1e-9 or \
               centers[i,1] < radii[i]-1e-9 or centers[i,1] > 1.0-radii[i]+1e-9:
                valid = False
                break
        if valid:
            for i in range(N):
                for j in range(i+1, N):
                    d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
                    if d < radii[i]+radii[j]-1e-9:
                        valid = False
                        break
                if not valid:
                    break
        if valid:
            break
            
        # Gentle shrinkage to recover strict feasibility
        radii *= 0.9995
        centers[:,0] = np.clip(centers[:,0], radii, 1.0-radii)
        centers[:,1] = np.clip(centers[:,1], radii, 1.0-radii)
        
    return centers, radii, float(np.sum(radii))
