

Looking at the past solutions, they consistently converge to ~2.634292. To break through this barrier, I need a more aggressive strategy for escaping local optima and better utilization of boundary/corner placements.

<strategy>
1. Use direct distance constraints (not squared) to avoid zero-gradient issues at contact points
2. Implement a simulated-annealing-style perturbation that occasionally accepts worse solutions
3. Try "deflation" strategies where we shrink some circles to let others grow
4. Better corner/edge biased initialization - optimal packings always have circles touching boundaries
5. Use alternating LP (for radii) + nonlinear optimization (for centers) more aggressively
6. Add a "circle swapping" perturbation to change topology
7. More diverse initial configurations including known dense lattice patterns
</strategy>

```python
import numpy as np
from scipy.optimize import minimize, linprog

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints_direct(x):
    """Direct distance constraints: dist - (r_i + r_j) >= 0."""
    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]
    
    # Boundary constraints
    c = np.concatenate([
        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs
    ])
    
    # Direct overlap constraints
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    i_idx, j_idx = TRIL_IDX
    dists = np.sqrt(dx[i_idx, j_idx]**2 + dy[i_idx, j_idx]**2)
    c = np.concatenate([c, dists - dr[i_idx, j_idx]])
    return c

def constraints_squared(x):
    """Squared distance constraints: dist^2 - (r_i + r_j)^2 >= 0."""
    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]
    
    c = np.concatenate([
        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs
    ])
    
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    i_idx, j_idx = TRIL_IDX
    dist_sq = dx[i_idx, j_idx]**2 + dy[i_idx, j_idx]**2
    r_sum_sq = dr[i_idx, j_idx]**2
    
    c = np.concatenate([c, dist_sq - r_sum_sq])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-7, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)])
    return b

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    num_ineq = n + N * (N - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], 
                  centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    dx = centers[:, 0, None] - centers[:, 0]
    dy = centers[:, 1, None] - centers[:, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    i_idx, j_idx = TRIL_IDX
    for k in range(len(i_idx)):
        A_ub[idx, i_idx[k]] = 1.0
        A_ub[idx, j_idx[k]] = 1.0
        b_ub[idx] = max(0.0, dists[i_idx[k], j_idx[k]])
        idx += 1
        
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-7)
    except Exception:
        pass
    return None

def force_spread_to_boundaries(centers, steps=800):
    """Spread points and push toward boundaries for better packing."""
    pts = centers.copy()
    for step in range(steps):
        forces = np.zeros_like(pts)
        lr = 0.03 * (1.0 - step / steps) ** 0.5
        
        for i in range(N):
            for j in range(i + 1, N):
                diff = pts[j] - pts[i]
                dist = np.linalg.norm(diff)
                if dist < 0.3 and dist > 1e-6:
                    f = 0.02 / (dist**2 + 0.0005)
                    fx, fy = f * diff[0], f * diff[1]
                    forces[i] -= [fx, fy]
                    forces[j] += [fx, fy]
            
            # Stronger boundary attraction
            for d in range(2):
                if pts[i, d] < 0.2:
                    forces[i, d] += 0.03
                elif pts[i, d] > 0.8:
                    forces[i, d] -= 0.03
                    
        pts += forces * lr
        pts = np.clip(pts, 0.01, 0.99)
    return pts

def make_corner_init(seed):
    """Initialize with circles biased toward corners and edges."""
    np.random.seed(seed)
    pts = np.zeros((N, 2))
    
    # Place circles near corners
    corner_bias = 0.12
    pts[0] = [corner_bias, corner_bias]
    pts[1] = [1 - corner_bias, corner_bias]
    pts[2] = [corner_bias, 1 - corner_bias]
    pts[3] = [1 - corner_bias, 1 - corner_bias]
    
    # Edge midpoints
    pts[4] = [0.5, corner_bias]
    pts[5] = [0.5, 1 - corner_bias]
    pts[6] = [corner_bias, 0.5]
    pts[7] = [1 - corner_bias, 0.5]
    
    # Fill rest with perturbed hexagonal pattern
    remaining = N - 8
    r = 0.095
    idx = 8
    y = 0.25
    row = 0
    while idx < N and y <= 0.85:
        x_start = 0.25 if row % 2 == 0 else 0.35
        x = x_start
        while x <= 0.85 and idx < N:
            pts[idx] = [x, y]
            idx += 1
            x += 2 * r
        y += np.sqrt(3) * r
        row += 1
    
    while idx < N:
        pts[idx] = np.random.uniform(0.2, 0.8, 2)
        idx += 1
    
    # Small perturbation
    pts += np.random.uniform(-0.01, 0.01, pts.shape)
    pts = np.clip(pts, 0.03, 0.97)
    return pts

def make_hex_init(seed, angle=0.0, r0=0.095):
    """Generate a rotated hexagonal lattice."""
    np.random.seed(seed)
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 8:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        if y > 1.0 + r0:
            break
            
    pts = np.array(pts[:N + 8])
    
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
        
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & \
           (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.15, 0.85, (1, 2))])
        
    return np.clip(pts[:N] + np.random.uniform(-0.005, 0.005, (N, 2)), 0.02, 0.98)

def make_staggered_grid_init(seed):
    """Create a staggered grid that might pack better."""
    np.random.seed(seed)
    pts = np.zeros((N, 2))
    idx = 0
    
    # 5 rows with staggered columns
    for row in range(6):
        n_cols = 5 if row % 2 == 0 else 4
        x_start = 0.13 if row % 2 == 0 else 0.23
        y = 0.12 + row * 0.155
        
        for col in range(n_cols):
            if idx >= N:
                break
            x = x_start + col * 0.18
            pts[idx] = [x + np.random.uniform(-0.01, 0.01), 
                       y + np.random.uniform(-0.01, 0.01)]
            idx += 1
    
    while idx < N:
        pts[idx] = np.random.uniform(0.15, 0.85, 2)
        idx += 1
        
    return pts

def project_to_bounds(x):
    """Ensure optimization vector respects bounds."""
    x = x.copy()
    for i in range(N):
        r = max(1e-7, x[3 * i + 2])
        x[3 * i] = np.clip(x[3 * i], r, 1.0 - r)
        x[3 * i + 1] = np.clip(x[3 * i + 1], r, 1.0 - r)
        x[3 * i + 2] = r
    return x

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    bounds = get_bounds()
    cons_sq = {'type': 'ineq', 'fun': constraints_squared}
    cons_dir = {'type': 'ineq', 'fun': constraints_direct}
    
    best_sum = -1.0
    best_x = None
    
    # Phase 1: Generate diverse initial configurations
    inits = []
    
    # 1. Rotated hexagonal lattices
    for s in range(20):
        ang = np.random.uniform(-0.4, 0.4)
        r0 = np.random.uniform(0.085, 0.11)
        c = make_hex_init(s, ang, r0)
        c = force_spread_to_boundaries(c, steps=500)
        inits.append(c)
    
    # 2. Corner-biased initializations
    for s in range(15):
        c = make_corner_init(s)
        c = force_spread_to_boundaries(c, steps=600)
        inits.append(c)
    
    # 3. Staggered grids
    for s in range(10):
        c = make_staggered_grid_init(s)
        c = force_spread_to_boundaries(c, steps=400)
        inits.append(c)
    
    # 4. Pure random + force spread
    for s in range(10):
        c = np.random.uniform(0.1, 0.9, (N, 2))
        c = force_spread_to_boundaries(c, steps=700)
        inits.append(c)
    
    # Phase 2: Multi-start optimization
    for c_init in inits:
        r_lp = solve_lp_radii(c_init)
        if r_lp is None:
            r_lp = np.full(N, 0.08)
            
        x0 = np.zeros(3 * N)
        x0[0::3] = c_init[:, 0]
        x0[1::3] = c_init[:, 1]
        x0[2::3] = r_lp
        
        x0 += np.random.normal(0, 1e-4, 3 * N)
        x0 = project_to_bounds(x0)
        
        # Try both constraint formulations
        for cons, tag in [(cons_sq, 'sq'), (cons_dir, 'dir')]:
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds, 
                               constraints=cons,
                               options={'maxiter': 12000, 'ftol': 1e-13, 'disp': False})
                
                if not np.isnan(res.fun):
                    curr_sum = -res.fun
                    if tag == 'sq':
                        c_vals = constraints_squared(res.x)
                    else:
                        c_vals = constraints_direct(res.x)
                    
                    if np.min(c_vals) >= -1e-6 and curr_sum > best_sum:
                        best_sum = curr_sum
                        best_x = res.x.copy()
            except Exception:
                pass
    
    if best_x is None:
        centers = force_spread_to_boundaries(np.random.uniform(0.2, 0.8, (N, 2)))
        radii = solve_lp_radii(centers) or np.full(N, 0.07)
        return centers, radii, float(np.sum(radii))
    
    # Phase 3: Aggressive refinement with multiple strategies
    
    # 3a: Alternating LP + SLSQP
    for rnd in range(40):
        centers_cur = best_x.reshape(N, 3)[:, :2]
        r_lp = solve_lp_radii(centers_cur)
        if r_lp is not None:
            best_x[2::3] = r_lp
            curr_sum = np.sum(r_lp)
            if curr_sum > best_sum:
                best_sum = curr_sum
        
        # Perturb centers slightly
        noise = 0.0015 * (0.95 ** rnd)
        x0 = best_x.copy()
        x0[0::3] += np.random.normal(0, noise, N)
        x0[1::3] += np.random.normal(0, noise, N)
        x0 = project_to_bounds(x0)
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                           constraints=cons_sq,
                           options={'maxiter': 8000, 'ftol': 1e-13, 'disp': False})
            
            if not np.isnan(res.fun):
                c_vals = constraints_squared(res.x)
                if np.min(c_vals) >= -1e-6:
                    curr_sum = -res.fun
                    if curr_sum > best_sum:
                        best_sum = curr_sum
                        best_x = res.x.copy()
        except Exception:
            pass
    
    # 3b: Deflation + reoptimization to break local optima
    for step in range(50):
        noise_scale = 0.003 * (0.93 ** step)
        x0 = best_x + np.random.normal(0, noise_scale, 3 * N)
        
        # Deflate radii
        x0[2::3] *= 0.95
        
        # Random subset deflation (breaks rigid structures)
        subset = np.random.choice(N, size=max(4, N // 4), replace=False)
        x0[subset * 3 + 2] *= 0.75
        
        x0 = project_to_bounds(x0)
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                           constraints=cons_sq,
                           options={'maxiter': 10000, 'ftol': 1e-13, 'disp': False})
            
            if not np.isnan(res.fun):
                centers_res = np.column_stack((res.x[0::3], res.x[1::3]))
                r_lp = solve_lp_radii(centers_res)
                if r_lp is not None:
                    curr_sum = np.sum(r_lp)
                    if curr_sum > best_sum:
                        best_x[0::3] = centers_res[:, 0]
                        best_x[1::3] = centers_res[:, 1]
                        best_x[2::3] = r_lp
                        best_sum = curr_sum
        except Exception:
            pass
    
    # 3c: Try swapping random pairs of circle positions
    for _ in range(30):
        x0 = best_x.copy()
        i1, i2 = np.random.choice(N, 2, replace=False)
        
        # Swap positions
        x0[3*i1], x0[3*i2] = x0[3*i2], x0[3*i1]
        x0[3*i1+1], x0[3*i2+1] = x0[3*i2+1], x0[3*i1+1]
        
        # Shrink both slightly
        x0[3*i1+2] *= 0.85
        x0[3*i2+2] *= 0.85
        
        x0 = project_to_bounds(x0)
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                           constraints=cons_sq,
                           options={'maxiter': 6000, 'ftol': 1e-13, 'disp': False})
            
            if not np.isnan(res.fun):
                centers_res = np.column_stack((res.x[0::3], res.x[1::3]))
                r_lp = solve_lp_radii(centers_res)
                if r_lp is not None:
                    curr_sum = np.sum(r_lp)
                    if curr_sum > best_sum:
                        best_x[0::3] = centers_res[:, 0]
                        best_x[1::3] = centers_res[:, 1]
                        best_x[2::3] = r_lp
                        best_sum = curr_sum
        except Exception:
            pass
    
    # Phase 4: Final LP snap
    centers_final = best_x.reshape(N, 3)[:, :2]
    r_lp_final = solve_lp_radii(centers_final)
    if r_lp_final is not None:
        best_x[2::3] = r_lp_final
        best_sum = np.sum(r_lp_final)
    
    # Extract results
    centers = best_x.reshape(N, 3)[:, :2]
    radii = best_x[2::3]
    
    # Final strict validity check and minimal repair
    for _ in range(200):
        valid = True
        for i in range(N):
            if radii[i] < 0:
                valid = False
                break
            if centers[i, 0] - radii[i] < -1e-12 or centers[i, 0] + radii[i] > 1.0 + 1e-12:
                valid = False
                break
            if centers[i, 1] - radii[i] < -1e-12 or centers[i, 1] + radii[i] > 1.0 + 1e-12:
                valid = False
                break
        
        if valid:
            for i in range(N):
                for j in range(i + 1, N):
                    d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                    if d < radii[i] + radii[j] - 1e-12:
                        valid = False
                        break
                if not valid:
                    break
        if valid:
            break
        
        radii *= 0.9999
        for i in range(N):
            centers[i, 0] = np.clip(centers[i, 0], radii[i], 1.0 - radii[i])
            centers[i, 1] = np.clip(centers[i, 1], radii[i], 1.0 - radii[i])
    
    return centers, radii, float(np.sum(radii))
```
