1. The LP refinement in Phase 3 incorrectly uses old centers instead of new ones
2. The deflation strategy is too aggressive (0.97 factor)
3. Limited iteration depth in refinement phase

<strategy>
1. Fix the LP-SLSQP alternation loop to properly alternate: SLSQP moves centers, then LP computes optimal radii for new centers
2. Generate much more diverse initializations (hexagonal at various scales/rotations, force-directed, corner-biased)
3. Implement proper deflation strategy with decaying noise to escape local minima
4. Add more refinement iterations with careful perturbation
5. Use LP as a refinement step after each SLSQP run to snap radii to theoretical maximum
6. Increase optimization iterations and tighten tolerances

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def lp_optimal_radii(centers):
    """Compute maximum radii for fixed centers using Linear Programming."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], 
                  centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
    
    dx = centers[:, 0, None] - centers[:, 0]
    dy = centers[:, 1, None] - centers[:, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    i_idx, j_idx = TRIL_IDX
    for k in range(len(i_idx)):
        A_ub[idx, i_idx[k]] = 1.0
        A_ub[idx, j_idx[k]] = 1.0
        b_ub[idx] = dists[i_idx[k], j_idx[k]]
        idx += 1
    
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-7)
    except Exception:
        pass
    return None

def safe_initial_radii(centers):
    """Compute feasible radii for initial centers."""
    n = centers.shape[0]
    radii = np.zeros(n)
    for i in range(n):
        min_d = min(centers[i, 0], 1.0 - centers[i, 0], 
                    centers[i, 1], 1.0 - centers[i, 1])
        for j in range(n):
            if i != j:
                d = np.hypot(centers[i, 0] - centers[j, 0], 
                             centers[i, 1] - centers[j, 1])
                if d < min_d:
                    min_d = d
        radii[i] = min_d * 0.42
    return radii

def make_hex_init(r0, angle, seed):
    """Generate hexagonal lattice with optional rotation."""
    np.random.seed(seed)
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 15:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        if y > 1.0 + r0:
            break
    
    pts = np.array(pts[:N + 15])
    
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
    
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & \
           (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.2, 0.8, (1, 2))])
    
    return np.clip(pts[:N] + np.random.uniform(-0.004, 0.004, (N, 2)), 0.02, 0.98)

def force_directed_init(seed, steps=500):
    """Force-directed layout to spread points."""
    np.random.seed(seed)
    pts = np.random.uniform(0.15, 0.85, (N, 2))
    for step in range(steps):
        forces = np.zeros_like(pts)
        lr = 0.04 * (1.0 - step / steps)
        for i in range(N):
            for j in range(i + 1, N):
                diff = pts[j] - pts[i]
                d = np.linalg.norm(diff)
                if d < 0.28 and d > 1e-6:
                    f = 0.012 / (d**2 + 0.0008)
                    forces[i] -= f * diff
                    forces[j] += f * diff
            for dim in range(2):
                if pts[i, dim] < 0.1: forces[i, dim] += 0.04
                elif pts[i, dim] > 0.9: forces[i, dim] -= 0.04
        pts += forces * lr
        pts = np.clip(pts, 0.03, 0.97)
    return pts

def constraint_func(x):
    """All constraints >= 0."""
    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]
    
    c = np.concatenate([
        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs
    ])
    
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    i_idx, j_idx = TRIL_IDX
    c = np.concatenate([c, dx[i_idx, j_idx]**2 + dy[i_idx, j_idx]**2 - dr[i_idx, j_idx]**2])
    return c

def objective_func(x):
    return -np.sum(x[2::3])

def get_bounds():
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)])
    return b

def run_slsqp(x0, bounds, maxiter=15000):
    """Run SLSQP optimization."""
    cons = {'type': 'ineq', 'fun': constraint_func}
    try:
        res = minimize(objective_func, x0, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'disp': False})
        if not np.isnan(res.fun):
            return res
    except Exception:
        pass
    return None

def project_x(x0):
    """Project optimization variables to feasible bounds."""
    x0 = x0.copy()
    for i in range(N):
        r = max(1e-7, x0[3 * i + 2])
        x0[3 * i] = np.clip(x0[3 * i], r, 1.0 - r)
        x0[3 * i + 1] = np.clip(x0[3 * i + 1], r, 1.0 - r)
        x0[3 * i + 2] = r
    return x0

def optimize_config(c_init, bounds):
    """Full optimization from initial centers with LP-SLSQP alternation."""
    r_init = lp_optimal_radii(c_init)
    if r_init is None:
        r_init = safe_initial_radii(c_init)
    
    x0 = np.zeros(3 * N)
    x0[0::3] = c_init[:, 0]
    x0[1::3] = c_init[:, 1]
    x0[2::3] = r_init
    
    # Slight perturbation to break symmetry
    x0 += np.random.normal(0, 5e-5, 3 * N)
    x0 = project_x(x0)
    
    # Multiple LP-SLSQP alternation cycles
    best_x = x0.copy()
    best_sum = np.sum(x0[2::3])
    
    for cycle in range(6):
        res = run_slsqp(x0, bounds, maxiter=12000)
        if res is not None:
            x0 = res.x.copy()
            x0 = project_x(x0)
            
            # LP refinement: snap radii to optimal for new centers
            centers_new = np.column_stack((x0[0::3], x0[1::3]))
            r_lp = lp_optimal_radii(centers_new)
            if r_lp is not None:
                x0[2::3] = r_lp
                x0 = project_x(x0)
        
        curr_sum = np.sum(x0[2::3])
        c_vals = constraint_func(x0)
        if np.min(c_vals) >= -1e-7 and curr_sum > best_sum:
            best_sum = curr_sum
            best_x = x0.copy()
    
    return best_x, best_sum

def deflation_perturbation(best_x, bounds, step):
    """Deflate and perturb to escape local minima."""
    noise_scale = 0.003 * (0.91 ** step)
    x0 = best_x.copy()
    x0 += np.random.normal(0, noise_scale, 3 * N)
    
    # Deflate all radii slightly
    x0[2::3] *= 0.97
    
    # Aggressively deflate a random subset
    subset = np.random.choice(N, size=N // 3, replace=False)
    x0[subset * 3 + 2] *= 0.75
    
    x0 = project_x(x0)
    
    res = run_slsqp(x0, bounds, maxiter=10000)
    if res is not None:
        x0 = res.x.copy()
        x0 = project_x(x0)
        
        centers_new = np.column_stack((x0[0::3], x0[1::3]))
        r_lp = lp_optimal_radii(centers_new)
        if r_lp is not None:
            x0[2::3] = r_lp
            x0 = project_x(x0)
        
        curr_sum = np.sum(x0[2::3])
        c_vals = constraint_func(x0)
        if np.min(c_vals) >= -1e-7:
            return x0, curr_sum
    return None, 0.0

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    bounds = get_bounds()
    
    # Generate diverse initializations
    inits = []
    
    # Hexagonal lattices with various scales and rotations
    for r0 in [0.085, 0.09, 0.095, 0.10, 0.105, 0.11]:
        for ang in [-0.35, -0.2, -0.1, 0.0, 0.1, 0.2, 0.35]:
            for s in range(3):
                c = make_hex_init(r0, ang, s)
                inits.append(c)
    
    # Force-directed layouts
    for s in range(20):
        c = force_directed_init(s, steps=600)
        inits.append(c)
    
    # Corner/edge biased
    for s in range(10):
        np.random.seed(s + 100)
        c = np.zeros((N, 2))
        c[0] = [0.11, 0.11]
        c[1] = [0.89, 0.11]
        c[2] = [0.11, 0.89]
        c[3] = [0.89, 0.89]
        c[4] = [0.5, 0.11]
        c[5] = [0.5, 0.89]
        c[6] = [0.11, 0.5]
        c[7] = [0.89, 0.5]
        c[8:] = np.random.uniform(0.2, 0.8, (N - 8, 2))
        c = force_directed_init(s + 1000, steps=400)
        inits.append(c)
    
    best_x = None
    best_sum = 0.0
    
    # Phase 1: Multi-start optimization
    for idx, c_init in enumerate(inits):
        x_opt, curr_sum = optimize_config(c_init, bounds)
        if x_opt is not None and curr_sum > best_sum:
            best_sum = curr_sum
            best_x = x_opt.copy()
    
    if best_x is None:
        c_init = make_hex_init(0.095, 0.0, 0)
        best_x = np.zeros(3 * N)
        best_x[0::3] = c_init[:, 0]
        best_x[1::3] = c_init[:, 1]
        best_x[2::3] = lp_optimal_radii(c_init) or safe_initial_radii(c_init)
        best_sum = np.sum(best_x[2::3])
    
    # Phase 2: Iterative deflation and perturbation refinement
    for step in range(80):
        x_pert, curr_sum = deflation_perturbation(best_x, bounds, step)
        if x_pert is not None and curr_sum > best_sum:
            best_sum = curr_sum
            best_x = x_pert.copy()
    
    # Phase 3: Fine-grained refinement around best
    for rnd in range(30):
        noise = 0.0015 * (0.93 ** rnd)
        x0 = best_x.copy()
        x0 += np.random.normal(0, noise, 3 * N)
        x0[2::3] *= 0.99
        
        x0 = project_x(x0)
        
        res = run_slsqp(x0, bounds, maxiter=8000)
        if res is not None:
            x0 = res.x.copy()
            x0 = project_x(x0)
            
            centers_new = np.column_stack((x0[0::3], x0[1::3]))
            r_lp = lp_optimal_radii(centers_new)
            if r_lp is not None:
                x0[2::3] = r_lp
                x0 = project_x(x0)
            
            curr_sum = np.sum(x0[2::3])
            c_vals = constraint_func(x0)
            if np.min(c_vals) >= -1e-7 and curr_sum > best_sum:
                best_sum = curr_sum
                best_x = x0.copy()
    
    centers = best_x.reshape(N, 3)[:, :2]
    radii = best_x[2::3].copy()
    
    # Final validation and repair
    for _ in range(200):
        valid = True
        for i in range(N):
            if radii[i] < 0:
                valid = False
                break
            if centers[i, 0] < radii[i] - 1e-9 or centers[i, 0] > 1.0 - radii[i] + 1e-9:
                valid = False
                break
            if centers[i, 1] < radii[i] - 1e-9 or centers[i, 1] > 1.0 - radii[i] + 1e-9:
                valid = False
                break
        if valid:
            for i in range(N):
                for j in range(i + 1, N):
                    d = np.hypot(centers[i, 0] - centers[j, 0], 
                                 centers[i, 1] - centers[j, 1])
                    if d < radii[i] + radii[j] - 1e-9:
                        valid = False
                        break
                if not valid:
                    break
        if valid:
            break
        
        radii *= 0.9998
        for i in range(N):
            centers[i, 0] = np.clip(centers[i, 0], radii[i], 1.0 - radii[i])
            centers[i, 1] = np.clip(centers[i, 1], radii[i], 1.0 - radii[i])
    
    return centers, radii, float(np.sum(radii))
