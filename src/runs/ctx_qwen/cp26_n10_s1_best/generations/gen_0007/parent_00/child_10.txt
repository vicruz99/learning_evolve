2. **Linear Programming Refinement**: For any fixed center configuration, the optimal radii can be found exactly via LP (`max ∑r_i` s.t. `r_i + r_j ≤ d_ij` and boundary limits). import numpy as np
from scipy.optimize import minimize, linprog

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def objective_jac(x):
    """Analytical Jacobian of the objective."""
    jac = np.zeros(3 * N)
    jac[2::3] = -1.0
    return jac

def constraints(x):
    """Returns all inequality constraints g(x) >= 0."""
    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]
    
    # Boundary constraints
    c = np.concatenate([
        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs
    ])
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    c = np.concatenate([c, dx[TRIL_IDX[0], TRIL_IDX[1]]**2 + dy[TRIL_IDX[0], TRIL_IDX[1]]**2 - dr[TRIL_IDX[0], TRIL_IDX[1]]**2])
    return c

def constraints_jac(x):
    """Returns the exact Jacobian matrix of the constraints."""
    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]
    
    num_cons = 4 * N + N * (N - 1) // 2
    jac = np.zeros((num_cons, 3 * N))
    
    # Boundary constraints Jacobian
    for i in range(N):
        jac[i, 3*i] = 1.0; jac[i, 3*i+2] = -1.0
        jac[N+i, 3*i] = -1.0; jac[N+i, 3*i+2] = -1.0
        jac[2*N+i, 3*i+1] = 1.0; jac[2*N+i, 3*i+2] = -1.0
        jac[3*N+i, 3*i+1] = -1.0; jac[3*N+i, 3*i+2] = -1.0
        
    # Overlap constraints Jacobian
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    i_idx, j_idx = TRIL_IDX
    idx = 4 * N
    for k in range(len(i_idx)):
        i, j = i_idx[k], j_idx[k]
        row = idx + k
        dxi, dyi, dri = dx[i,j], dy[i,j], dr[i,j]
        
        jac[row, 3*i] = 2.0*dxi; jac[row, 3*i+1] = 2.0*dyi; jac[row, 3*i+2] = -2.0*dri
        jac[row, 3*j] = -2.0*dxi; jac[row, 3*j+1] = -2.0*dyi; jac[row, 3*j+2] = -2.0*dri
        
    return jac

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_ineq = n + n*(n-1)//2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    for i in range(n):
        lim = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    for i in range(n):
        for j in range(i+1, n):
            d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, d)
            idx += 1
            
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=[(0.0, None)]*n, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def gen_hex(scale, angle):
    """Generates a rotated and scaled hexagonal lattice initialization."""
    pts = []
    r0 = 0.09 * scale
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        if y > 1.0 + r0: break
            
    pts = np.array(pts[:N + 10])
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
        
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.15, 0.85, (1, 2))])
    return pts[:N]

def gen_force(seed):
    """Force-directed layout to spread points evenly."""
    np.random.seed(seed)
    pts = np.random.uniform(0.15, 0.85, (N, 2))
    for _ in range(400):
        f = np.zeros_like(pts)
        for i in range(N):
            for j in range(i+1, N):
                diff = pts[j] - pts[i]
                d = np.linalg.norm(diff)
                if d < 0.25 and d > 1e-5:
                    ff = 0.012 / (d**2 + 0.001)
                    f[i] -= ff * diff
                    f[j] += ff * diff
            for dim in range(2):
                if pts[i, dim] < 0.12: f[i, dim] += 0.025
                elif pts[i, dim] > 0.88: f[i, dim] -= 0.025
        pts += f * 0.05
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints, 'jac': constraints_jac}
    
    best_sum = 0.0
    best_x = None
    
    # Phase 1: Diverse Initial Configurations
    configs = []
    for scale in np.linspace(0.92, 1.08, 6):
        for ang in np.linspace(0, np.pi/3, 11):
            configs.append(gen_hex(scale, ang))
    for s in range(15):
        configs.append(gen_force(s))
        
    for c0 in configs:
        r0 = solve_lp_radii(c0)
        if r0 is None: r0 = np.full(N, 0.06)
        # Ensure strict initial feasibility
        r0 = np.maximum(r0 * 0.95, 0.01)
        c0 = np.clip(c0, r0, 1.0 - r0)
        
        x0 = np.zeros(3 * N)
        x0[0::3] = c0[:, 0]
        x0[1::3] = c0[:, 1]
        x0[2::3] = r0
        
        try:
            res = minimize(objective, x0, method='SLSQP', jac=objective_jac,
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 15000, 'ftol': 1e-14, 'disp': False})
            if not np.isnan(res.fun):
                curr = -res.fun
                if curr > best_sum:
                    best_x = res.x.copy()
                    best_sum = curr
                    
                # LP refinement to snap radii to theoretical max for these centers
                c_opt = res.x.reshape(N, 3)[:, :2]
                r_lp = solve_lp_radii(c_opt)
                if r_lp is not None and np.sum(r_lp) > best_sum:
                    best_x[2::3] = r_lp
                    best_sum = np.sum(r_lp)
        except Exception:
            pass
            
    # Phase 2: Iterative Refinement with Deflation & Perturbation
    if best_x is not None:
        for step in range(40):
            factor = 0.96 ** step
            x_p = best_x.copy()
            # Shrink radii to create slack for rearrangement
            x_p[2::3] *= (0.85 + 0.10 * factor)
            noise = 0.002 * factor
            x_p[0::3] += np.random.normal(0, noise, N)
            x_p[1::3] += np.random.normal(0, noise, N)
            
            # Project to strict bounds
            for i in range(N):
                r = max(0.005, x_p[3*i+2])
                x_p[3*i] = np.clip(x_p[3*i], r, 1.0-r)
                x_p[3*i+1] = np.clip(x_p[3*i+1], r, 1.0-r)
                x_p[3*i+2] = r
                
            try:
                res = minimize(objective, x_p, method='SLSQP', jac=objective_jac,
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 10000, 'ftol': 1e-14, 'disp': False})
                if not np.isnan(res.fun):
                    c_opt = res.x.reshape(N, 3)[:, :2]
                    r_lp = solve_lp_radii(c_opt)
                    if r_lp is not None:
                        s_lp = np.sum(r_lp)
                        if s_lp > best_sum:
                            best_x[2::3] = r_lp
                            best_x[0::3] = c_opt[:, 0]
                            best_x[1::3] = c_opt[:, 1]
                            best_sum = s_lp
                    elif -res.fun > best_sum:
                        best_x = res.x.copy()
                        best_sum = -res.fun
            except Exception:
                pass
                
    centers = best_x.reshape(N, 3)[:, :2]
    radii = best_x[2::3]
    
    # Final strict validity check and minimal numerical repair
    for _ in range(100):
        valid = True
        for i in range(N):
            if radii[i] < 0 or centers[i,0] < radii[i]-1e-9 or centers[i,0] > 1-radii[i]+1e-9 or \
               centers[i,1] < radii[i]-1e-9 or centers[i,1] > 1-radii[i]+1e-9:
                valid = False; break
        if valid:
            for i in range(N):
                for j in range(i+1, N):
                    if np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1]) < radii[i]+radii[j]-1e-9:
                        valid = False; break
                if not valid: break
        if valid: break
        radii *= 0.9995
        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)
        
    return centers, radii, float(np.sum(radii))
