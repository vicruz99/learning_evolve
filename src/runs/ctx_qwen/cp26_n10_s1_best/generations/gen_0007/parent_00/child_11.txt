import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Returns all inequality constraints g(x) >= 0."""
    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]
    
    # Boundary constraints: 4*N
    c = np.concatenate([
        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs
    ])
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    i_idx, j_idx = np.tril_indices(N, -1)
    dist_sq = dx[i_idx, j_idx]**2 + dy[i_idx, j_idx]**2
    r_sum_sq = dr[i_idx, j_idx]**2
    
    c = np.concatenate([c, dist_sq - r_sum_sq])
    return c

def constraints_jac(x):
    """Returns the exact Jacobian matrix of the constraints."""
    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]
    
    num_cons = 4 * N + N * (N - 1) // 2
    jac = np.zeros((num_cons, 3 * N))
    
    # Boundary constraints Jacobian
    for i in range(N):
        jac[i, 3 * i] = 1.0
        jac[i, 3 * i + 2] = -1.0
        jac[N + i, 3 * i] = -1.0
        jac[N + i, 3 * i + 2] = -1.0
        jac[2 * N + i, 3 * i + 1] = 1.0
        jac[2 * N + i, 3 * i + 2] = -1.0
        jac[3 * N + i, 3 * i + 1] = -1.0
        jac[3 * N + i, 3 * i + 2] = -1.0
        
    # Overlap constraints Jacobian
    idx = 4 * N
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    i_idx, j_idx = np.tril_indices(N, -1)
    for k in range(len(i_idx)):
        i = i_idx[k]
        j = j_idx[k]
        row = idx + k
        
        dxi = dx[i, j]
        dyi = dy[i, j]
        dri = dr[i, j]
        
        jac[row, 3 * i] = 2.0 * dxi
        jac[row, 3 * i + 1] = 2.0 * dyi
        jac[row, 3 * i + 2] = -2.0 * dri
        
        jac[row, 3 * j] = -2.0 * dxi
        jac[row, 3 * j + 1] = -2.0 * dyi
        jac[row, 3 * j + 2] = -2.0 * dri
        
    return jac

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist)
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def make_hex_init(r0, angle):
    """Generates a rotated hexagonal lattice initialization."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        if y > 1.0 + r0:
            break
            
    pts = np.array(pts[:N + 10])
    
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
        
    mask = (pts[:, 0] >= 0.01) & (pts[:, 0] <= 0.99) & (pts[:, 1] >= 0.01) & (pts[:, 1] <= 0.99)
    pts = pts[mask]
    
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.15, 0.85, (1, 2))])
    return pts[:N]

def grow_init(seed):
    """Physics-based disk growth simulation to find dense packings."""
    np.random.seed(seed)
    centers = np.random.uniform(0.15, 0.85, (N, 2))
    radii = np.full(N, 0.02)
    
    for step in range(1200):
        # Linear cooling schedule for growth rate
        growth = 0.00045 * max(0.0, 1.0 - step / 1200)
        radii += growth
        
        # Multiple relaxation passes per growth step
        for _ in range(3):
            for i in range(N):
                # Boundary repulsion
                centers[i, 0] = np.clip(centers[i, 0], radii[i], 1.0 - radii[i])
                centers[i, 1] = np.clip(centers[i, 1], radii[i], 1.0 - radii[i])
                
                # Overlap resolution
                for j in range(i + 1, N):
                    dx = centers[j, 0] - centers[i, 0]
                    dy = centers[j, 1] - centers[i, 1]
                    dist = np.hypot(dx, dy)
                    
                    if dist < radii[i] + radii[j] and dist > 1e-9:
                        shift = (radii[i] + radii[j] - dist) * 0.5 + 1e-7
                        nx, ny = dx / dist, dy / dist
                        centers[i] -= shift * np.array([nx, ny])
                        centers[j] += shift * np.array([nx, ny])
    return centers, radii

def project_to_bounds(x):
    """Ensure optimization vector strictly respects bounds."""
    x = x.copy()
    for i in range(N):
        r = max(1e-6, x[3 * i + 2])
        x[3 * i] = np.clip(x[3 * i], r, 1.0 - r)
        x[3 * i + 1] = np.clip(x[3 * i + 1], r, 1.0 - r)
        x[3 * i + 2] = r
    return x

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    np.random.seed(42)
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N

    best_sum = -1.0

    # Rotated Hexagonal Lattices
    for r0 in np.linspace(0.085, 0.110, 6):

            inits.append(pts)
            
    # Growth Simulations
    for s in range(12):
        c, r = grow_init(s)
        x0 = np.zeros(3 * N)
        x0[0::3] = c[:, 0]
        x0[1::3] = c[:, 1]
        x0[2::3] = r
        inits.append(project_to_bounds(x0))

    # Phase 2: Multi-start Optimization
    for init_data in inits:
        if init_data.ndim == 2:
            c_init = init_data
            x0 = np.zeros(3 * N)
            x0[0::3] = c_init[:, 0]
            x0[1::3] = c_init[:, 1]
            x0[2::3] = 0.06
        else:
            x0 = init_data.copy()
            
        # LP Snap to maximize initial radii
        centers_guess = np.column_stack((x0[0::3], x0[1::3]))
        r_lp = solve_lp_radii(centers_guess)
        if r_lp is not None:
            x0[2::3] = r_lp * 0.999
        x0 = project_to_bounds(x0)
        
        # Break exact symmetries
        x0 += np.random.normal(0, 5e-5, 3 * N)
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 25000, 'ftol': 1e-15, 'disp': False})
            if not np.isnan(res.fun):
                c_vals = constraints(res.x)
