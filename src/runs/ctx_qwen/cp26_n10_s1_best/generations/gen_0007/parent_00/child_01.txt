import numpy as np
from scipy.optimize import minimize, linprog

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def objective_jac(x):
    """Analytical Jacobian of the objective."""
    jac = np.zeros(3 * N)
    jac[2::3] = -1.0
    return jac

def constraints(x):
    """Returns all inequality constraints g(x) >= 0."""
    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]
    
    # Boundary constraints: 4*N
    c = np.concatenate([
        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs
    ])
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    c = np.concatenate([c, dx[TRIL_IDX]**2 + dy[TRIL_IDX]**2 - dr[TRIL_IDX]**2])
    return c

def constraints_jac(x):
    """Returns the exact Jacobian matrix of the constraints."""
    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]
    
    num_cons = 4 * N + N * (N - 1) // 2
    jac = np.zeros((num_cons, 3 * N))
    
    # Boundary constraints Jacobian
    for i in range(N):
        jac[i, 3 * i] = 1.0
        jac[i, 3 * i + 2] = -1.0
        jac[N + i, 3 * i] = -1.0
        jac[N + i, 3 * i + 2] = -1.0
        jac[2 * N + i, 3 * i + 1] = 1.0
        jac[2 * N + i, 3 * i + 2] = -1.0
        jac[3 * N + i, 3 * i + 1] = -1.0
        jac[3 * N + i, 3 * i + 2] = -1.0
        
    # Overlap constraints Jacobian
    idx = 4 * N
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    i_idx, j_idx = TRIL_IDX
    for k in range(len(i_idx)):
        i = i_idx[k]
        j = j_idx[k]
        row = idx + k
        
        dxi = dx[i, j]
        dyi = dy[i, j]
        dri = dr[i, j]
        
        jac[row, 3 * i] = 2.0 * dxi
        jac[row, 3 * i + 1] = 2.0 * dyi
        jac[row, 3 * i + 2] = -2.0 * dri
        
        jac[row, 3 * j] = -2.0 * dxi
        jac[row, 3 * j + 1] = -2.0 * dyi
        jac[row, 3 * j + 2] = -2.0 * dri
        
    return jac

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(1e-7, lim)
        idx += 1
        
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(1e-7, dist)
            idx += 1
            
    bounds = [(1e-7, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-7)
    except Exception:
        pass
    return None

def make_hex_init(seed, angle, scale):
    """Generates a rotated and scaled hexagonal lattice initialization."""
    np.random.seed(seed)
    pts = []
    r0 = 0.09 * scale
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        if y > 1.0 + r0:
            break
            
    pts = np.array(pts[:N + 10])
    
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
        
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.15, 0.85, (1, 2))])
    return pts[:N] + np.random.uniform(-0.002, 0.002, (N, 2))

def local_search_centers(centers, steps=300):
    """Hill-climbing on centers using LP evaluation for radii."""
    best_centers = centers.copy()
    best_radii = solve_lp_radii(best_centers)
    if best_radii is None:
        best_radii = np.full(N, 0.05)
    best_sum = np.sum(best_radii)
    step_size = 0.04
    
    for _ in range(steps):
        new_centers = best_centers + np.random.uniform(-step_size, step_size, (N, 2))
        new_centers = np.clip(new_centers, 0.01, 0.99)
        
        new_radii = solve_lp_radii(new_centers)
        if new_radii is not None:
            new_sum = np.sum(new_radii)
            if new_sum > best_sum:
                best_centers = new_centers
                best_radii = new_radii
                best_sum = new_sum
        step_size *= 0.998
        
    return best_centers, best_radii, best_sum

def get_score(item):
    return item[2]
