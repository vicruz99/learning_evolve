import numpy as np
from scipy.optimize import minimize, linprog
from functools import partial

N = 26
TRIU_IJ = np.triu_indices(N, 1)

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [0, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
    return b

def penalty_objective(x, P, Q):
    """Objective for penalty method: maximize sum radii - penalties for violations."""
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    # Base objective
    obj = -np.sum(r)
    
    # Boundary penalty: circles must stay inside [0,1]x[0,1]
    wall = np.minimum(np.minimum(cx, 1.0 - cx), np.minimum(cy, 1.0 - cy))
    obj += Q * np.sum(np.maximum(0.0, r - wall)**2)
    
    # Overlap penalty: circles must not overlap
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dist = np.sqrt(dx**2 + dy**2)
    r_sum = r[:, None] + r[None, :]
    obj += P * np.sum(np.maximum(0.0, r_sum[TRIU_IJ] - dist[TRIU_IJ])**2)
    
    return obj

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A = np.zeros((num_ineq, n))
    b = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A[idx, i] = 1.0
        b[idx] = max(1e-9, lim)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = max(1e-9, d)
            idx += 1
            
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0.0, None)] * n, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-6)
    except Exception:
        pass
    return None

def hex_init(r0, angle):
    """Generates a hexagonal lattice initialization with optional rotation."""
    centers = np.zeros((N, 2))
    idx = 0
    y = r0
    row = 0
    while idx < N and y + r0 <= 1.0:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while idx < N and x + r0 <= 1.0:
            centers[idx] = [x, y]
            idx += 1
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
    while idx < N:
        centers[idx] = np.random.uniform(0.1, 0.9, 2)
        idx += 1
        
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        cx, cy = 0.5, 0.5
        dx = centers[:, 0] - cx
        dy = centers[:, 1] - cy
        centers[:, 0] = dx * c - dy * s + cx
        centers[:, 1] = dx * s + dy * c + cy
        
    return centers

def force_init(seed, steps=500):
    """Force-directed layout to spread points evenly and push to boundaries."""
    np.random.seed(seed)
    pts = np.random.uniform(0.15, 0.85, (N, 2))
    for _ in range(steps):
        f = np.zeros_like(pts)
        for i in range(N):
            for j in range(i + 1, N):
                dx = pts[j] - pts[i]
                d = np.hypot(dx[0], dx[1])
                if d < 0.3 and d > 1e-5:
                    rep = 0.02 / (d**2 + 0.001)
                    f[i] -= rep * dx
                    f[j] += rep * dx
            for dim in range(2):
                if pts[i, dim] < 0.15:
                    f[i, dim] += 0.05
                elif pts[i, dim] > 0.85:
                    f[i, dim] -= 0.05
        pts += f * 0.05
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    bounds = get_bounds()
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Phase 1: Diverse Initializations
    inits = []
    for r0 in [0.08, 0.09, 0.10, 0.11, 0.12]:
        for ang in [0.0, 0.15, -0.15, 0.3, -0.3, 0.45, -0.45, 0.6, -0.6]:
            inits.append(hex_init(r0, ang))
    for s in range(15):
        inits.append(force_init(s))
        
    # Phase 2: Penalty method optimization (L-BFGS-B)
    for c_init in inits:
        x0 = np.zeros(3 * N)
        x0[0::3] = c_init[:, 0]
        x0[1::3] = c_init[:, 1]
        x0[2::3] = 0.09
        
        P, Q = 50.0, 50.0
        for _ in range(15):
            res = minimize(partial(penalty_objective, P=P, Q=Q), x0, 
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 2000, 'ftol': 1e-12})
            x0 = res.x
            P *= 1.8
            Q *= 1.8
            
        c_opt = np.column_stack((x0[0::3], x0[1::3]))
        r_lp = solve_lp_radii(c_opt)
        if r_lp is not None:
            curr_sum = np.sum(r_lp)
            if curr_sum > best_sum:
                best_sum = curr_sum
                best_centers = c_opt.copy()
                best_radii = r_lp.copy()
                
    # Phase 3: Deflation & Perturbation Refinement
    if best_centers is not None:
        for step in range(30):

            r_init = best_radii * 0.95
            x0 = np.zeros(3 * N)
            x0[0::3] = c_pert[:, 0]

            x0[2::3] = r_init
            
            P, Q = 200.0, 200.0

        if r_fallback is None:

    radii = best_radii

               centers[i, 1] < radii[i] - 1e-10 or centers[i, 1] > 1.0 - radii[i] + 1e-10:
                valid = False
                break
        if valid:

                    if np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1]) < radii[i] + radii[j] - 1e-10:
                        valid = False
                        break
                if not valid:
                    break
        if valid:
            break
            
        # Minimal shrinkage to recover strict feasibility
        radii *= 0.9995
        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)
        
    return centers, radii, float(np.sum(radii))
