

<strategy>
To improve upon the current packing sum of ~2.6279 and target >2.634, we need a more robust exploration-exploitation strategy. The main bottlenecks in circle packing are:
1. Getting stuck in rigid hexagonal-like local minima.
2. SLSQP struggling near the boundary where gradients vanish.
3. Inadequate initial configurations that don't explore non-lattice arrangements.

Strategy:
- **Enhanced Initialization**: Generate diverse starts including rotated/scaled hexagonal lattices, force-directed layouts, and specifically structured corner/edge biases.
- **Alternating LP & SLSQP**: Use Linear Programming to instantly snap radii to their theoretical maximum for fixed centers, then use SLSQP to adjust centers to create more space. This alternation is highly effective.
- **Deflation & Perturbation**: Systematically shrink radii to create slack, perturb centers (especially subsets of circles to break symmetry), and re-optimize. This escapes local minima more effectively than full random restarts.
- **Aggressive Refinement Loop**: Run multiple cycles of optimization with decaying noise and tight tolerances.
- **Strict Feasibility Repair**: A final repair loop ensures strict compliance with the 1e-12 tolerance in the validator by minimally shrinking radii and re-clipping centers if numerical drift occurs.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
TRIL = np.tril_indices(N, -1)

def get_constraints(x):
    """Vectorized inequality constraints: g(x) >= 0."""
    cx, cy, r = x[0::3], x[1::3], x[2::3]
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    c = np.concatenate([
        cx - r, 1.0 - cx - r,
        cy - r, 1.0 - cy - r
    ])
    
    # Overlap constraints: dist^2 >= (ri + rj)^2
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dr = r[:, None] + r[None, :]
    
    c = np.concatenate([c, dx[TRIL]**2 + dy[TRIL]**2 - dr[TRIL]**2])
    return c

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-7, 0.5]."""
    return [(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)] * N

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A = np.zeros((num_ineq, n))
    b = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], 
                  centers[i, 1], 1.0 - centers[i, 1])
        A[idx, i] = 1.0
        b[idx] = max(1e-9, lim)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], 
                         centers[i, 1] - centers[j, 1])
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = max(1e-9, d)
            idx += 1
            
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0.0, None)] * n, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-7)
    except Exception:
        pass
    return np.full(n, 1e-6)

def project_to_bounds(x):
    """Ensure optimization vector strictly respects bounds."""
    r = np.maximum(x[2::3], 1e-7)
    x = x.copy()
    x[0::3] = np.clip(x[0::3], r, 1.0 - r)
    x[1::3] = np.clip(x[1::3], r, 1.0 - r)
    x[2::3] = r
    return x

def hex_init(r0, angle):
    """Generates a hexagonal lattice initialization with optional rotation."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        if y > 1.0 + r0:
            break
            
    pts = np.array(pts[:N + 10])
    
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
        
    mask = (pts[:, 0] >= 0.01) & (pts[:, 0] <= 0.99) & \
           (pts[:, 1] >= 0.01) & (pts[:, 1] <= 0.99)
    pts = pts[mask]
    
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.2, 0.8, (1, 2))])
    return pts[:N]

def force_init(seed, steps=800):
    """Force-directed layout to spread points evenly and push to boundaries."""
    np.random.seed(seed)
    pts = np.random.uniform(0.12, 0.88, (N, 2))
    for step in range(steps):
        f = np.zeros_like(pts)
        for i in range(N):
            for j in range(i + 1, N):
                dx = pts[j] - pts[i]
                d = np.hypot(dx[0], dx[1])
                if d < 0.3 and d > 1e-6:
                    rep = 0.025 / (d**2 + 0.001)
                    f[i] -= rep * dx
                    f[j] += rep * dx
            for dim in range(2):
                if pts[i, dim] < 0.12:
                    f[i, dim] += 0.05
                elif pts[i, dim] > 0.88:
                    f[i, dim] -= 0.05
        lr = 0.06 * (1.0 - step / steps)
        pts += f * lr
        pts = np.clip(pts, 0.04, 0.96)
    return pts

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': get_constraints}
    
    best_sum = -1.0
    best_x = None
    
    # Phase 1: Diverse Initializations
    inits = []
    # Hexagonal lattices with varied scales and rotations
    for r0 in [0.075, 0.085, 0.095, 0.105, 0.115]:
        for ang in [0.0, 0.15, -0.15, 0.3, -0.3, 0.45, -0.45, 0.6]:
            inits.append(hex_init(r0, ang))
            
    # Force-directed layouts
    for s in range(25):
        inits.append(force_init(s))
        
    # Multi-start optimization with LP-SLSQP alternation
    for c_init in inits:
        r_init = solve_lp_radii(c_init)
        x0 = np.zeros(3 * N)
        x0[0::3] = c_init[:, 0]
        x0[1::3] = c_init[:, 1]
        x0[2::3] = r_init
        x0 = project_to_bounds(x0)
        
        # Break exact symmetries slightly
        x0 += np.random.normal(0, 1e-5, x0.shape)
        
        # Alternating refinement cycles
        for _ in range(4):
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 20000, 'ftol': 1e-13, 'disp': False})
                if not np.isnan(res.fun):
                    x0 = project_to_bounds(res.x)
                    c_opt = np.column_stack((x0[0::3], x0[1::3]))
                    r_lp = solve_lp_radii(c_opt)
                    x0[2::3] = r_lp
                    x0 = project_to_bounds(x0)
            except Exception:
                break
                
        curr_sum = np.sum(x0[2::3])
        c_vals = get_constraints(x0)
        if np.min(c_vals) >= -1e-7 and curr_sum > best_sum:
            best_sum = curr_sum
            best_x = x0.copy()
            
    # Phase 2: Deflation & Perturbation Refinement to escape local minima
    if best_x is not None:
        for step in range(40):
            x_pert = best_x.copy()
            # Shrink radii to create slack for repositioning
            shrink_factor = 0.92 + 0.06 * (step / 40.0)
            x_pert[2::3] *= shrink_factor
            
            noise = 0.003 / (step + 1)
            x_pert[0::3] += np.random.normal(0, noise, N)
            x_pert[1::3] += np.random.normal(0, noise, N)
            x_pert = project_to_bounds(x_pert)
            
            try:
                res = minimize(objective, x_pert, method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 15000, 'ftol': 1e-13, 'disp': False})
                if not np.isnan(res.fun):
                    x_pert = project_to_bounds(res.x)
                    c_opt = np.column_stack((x_pert[0::3], x_pert[1::3]))
                    r_lp = solve_lp_radii(c_opt)
                    x_pert[2::3] = r_lp
                    x_pert = project_to_bounds(x_pert)
                    
                    curr_sum = np.sum(x_pert[2::3])
                    if np.min(get_constraints(x_pert)) >= -1e-7 and curr_sum > best_sum:
                        best_sum = curr_sum
                        best_x = x_pert.copy()
            except Exception:
                pass
                
    # Phase 3: Subset Perturbation to break rigid lattice structures
    if best_x is not None:
        for _ in range(30):
            x_sub = best_x.copy()
            # Randomly select 1/3 of circles to perturb heavily
            subset = np.random.choice(N, size=N // 3, replace=False)
            x_sub[0::3][subset] += np.random.normal(0, 0.008, len(subset))
            x_sub[1::3][subset] += np.random.normal(0, 0.008, len(subset))
            x_sub[2::3][subset] *= 0.85 # Deflate selected radii
            x_sub = project_to_bounds(x_sub)
            
            try:
                res = minimize(objective, x_sub, method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 12000, 'ftol': 1e-13, 'disp': False})
                if not np.isnan(res.fun):
                    x_sub = project_to_bounds(res.x)
                    c_opt = np.column_stack((x_sub[0::3], x_sub[1::3]))
                    r_lp = solve_lp_radii(c_opt)
                    x_sub[2::3] = r_lp
                    x_sub = project_to_bounds(x_sub)
                    
                    curr_sum = np.sum(x_sub[2::3])
                    if np.min(get_constraints(x_sub)) >= -1e-7 and curr_sum > best_sum:
                        best_sum = curr_sum
                        best_x = x_sub.copy()
            except Exception:
                pass

    # Fallback (should not be reached with valid inits)
    if best_x is None:
        c_fallback = hex_init(0.09, 0.0)
        r_fallback = solve_lp_radii(c_fallback)
        best_x = np.zeros(3 * N)
        best_x[0::3] = c_fallback[:, 0]
        best_x[1::3] = c_fallback[:, 1]
        best_x[2::3] = r_fallback
        best_sum = np.sum(r_fallback)
        
    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3].copy()
    
    # Final strict validity adjustment against 1e-12 tolerance
    for _ in range(100):
        valid = True
        for i in range(N):
            if radii[i] < 0 or centers[i, 0] < radii[i] - 1e-10 or centers[i, 0] > 1.0 - radii[i] + 1e-10 or \
               centers[i, 1] < radii[i] - 1e-10 or centers[i, 1] > 1.0 - radii[i] + 1e-10:
                valid = False
                break
        if valid:
            for i in range(N):
                for j in range(i + 1, N):
                    d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                    if d < radii[i] + radii[j] - 1e-10:
                        valid = False
                        break
                if not valid:
                    break
        if valid:
            break
            
        # Minimal shrinkage to recover strict feasibility
        radii *= 0.9995
        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)
        
    return centers, radii, float(np.sum(radii))
