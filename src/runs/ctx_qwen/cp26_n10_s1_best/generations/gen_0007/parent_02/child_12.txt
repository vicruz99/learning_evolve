import numpy as np
from scipy.optimize import minimize, linprog, differential_evolution

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A = np.zeros((num_ineq, n))
    b = np.zeros(num_ineq)
    
    idx = 0
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        b[idx] = max(1e-9, lim)
        A[idx, i] = 1.0
        idx += 1
        
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            b[idx] = max(1e-9, d)
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            idx += 1
            
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0.0, None)] * n, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-6)
    except Exception:
        pass
    return None

def de_objective(centers_flat):
    """Objective for differential evolution: negative sum of LP-optimized radii."""
    centers = centers_flat.reshape(N, 2)
    r = solve_lp_radii(centers)
    if r is None:
        return 0.0
    for i in range(N):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        if r[i] > lim + 1e-6:
            return 0.0
    return -np.sum(r)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints_sq(x):
    """Squared distance constraint: dist^2 >= (r_i + r_j)^2"""
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    c = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dr = r[:, None] + r[None, :]
    c = np.concatenate([c, dx[TRIL_IDX]**2 + dy[TRIL_IDX]**2 - dr[TRIL_IDX]**2])
    return c

def constraints_lin(x):
    """Linear distance constraint: dist >= r_i + r_j"""
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    c = np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dr = r[:, None] + r[None, :]
    dists = np.sqrt(np.maximum(dx[TRIL_IDX]**2 + dy[TRIL_IDX]**2, 1e-20))
    c = np.concatenate([c, dists - dr[TRIL_IDX]])
    return c

def get_bounds():
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)])
    return b

def project_to_bounds(x):
    """Ensure optimization vector strictly respects bounds."""
    r = np.maximum(x[2::3], 1e-6)
    x[0::3] = np.clip(x[0::3], r, 1.0 - r)
    x[1::3] = np.clip(x[1::3], r, 1.0 - r)
    x[2::3] = r
    return x

def hex_init(r0, angle, seed):
    """Generates a hexagonal lattice initialization with optional rotation."""
    np.random.seed(seed)
    centers = np.zeros((N, 2))
    idx = 0
    y = r0
    row = 0
    while idx < N and y + r0 <= 1.0:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while idx < N and x + r0 <= 1.0:
            centers[idx] = [x, y]
            idx += 1
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
    while idx < N:
        centers[idx] = np.random.uniform(0.1, 0.9, 2)
        idx += 1
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        dx = centers[:, 0] - 0.5
        dy = centers[:, 1] - 0.5
        centers[:, 0] = dx * c - dy * s + 0.5
        centers[:, 1] = dx * s + dy * c + 0.5
    return centers

def force_init(seed, steps=500):
    """Force-directed layout to spread points evenly and push to boundaries."""
    np.random.seed(seed)
    pts = np.random.uniform(0.1, 0.9, (N, 2))
    for step in range(steps):
        f = np.zeros_like(pts)
        for i in range(N):
            for j in range(i + 1, N):
                dx = pts[j] - pts[i]
                d = np.hypot(dx[0], dx[1])
                if d < 0.3 and d > 1e-5:
                    rep = 0.02 / (d**2 + 0.001)
                    f[i] -= rep * dx
                    f[j] += rep * dx
            for dim in range(2):
                if pts[i, dim] < 0.15:
                    f[i, dim] += 0.06
                elif pts[i, dim] > 0.85:
                    f[i, dim] -= 0.06
        lr = 0.05 * (1.0 - step / steps)
        pts += f * lr
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def corner_init(seed):
    """Initialize with circles in corners and edges."""
    np.random.seed(seed)
    centers = np.zeros((N, 2))
    idx = 0
    # Four corners
    for cx in [0.06, 0.94]:
        for cy in [0.06, 0.94]:
            centers[idx] = [cx + np.random.uniform(-0.02, 0.02), cy + np.random.uniform(-0.02, 0.02)]
            idx += 1
    # Edge positions
    for t in np.linspace(0.15, 0.85, 6):
        centers[idx] = [t, 0.06 + np.random.uniform(-0.02, 0.02)]
        idx += 1
        if idx < N:
            centers[idx] = [t, 0.94 + np.random.uniform(-0.02, 0.02)]
            idx += 1
        if idx < N:
            centers[idx] = [0.06 + np.random.uniform(-0.02, 0.02), t]
            idx += 1
        if idx < N:
            centers[idx] = [0.94 + np.random.uniform(-0.02, 0.02), t]
            idx += 1
    # Fill remaining in center
    while idx < N:
        centers[idx] = [0.5 + np.random.uniform(-0.25, 0.25), 0.5 + np.random.uniform(-0.25, 0.25)]
        idx += 1
    return centers[:N]

def spiral_init(seed):
    """Initialize with spiral pattern."""
    np.random.seed(seed)
    centers = np.zeros((N, 2))
    for i in range(N):
        angle = i * 2.4 + np.random.uniform(-0.1, 0.1)
        radius = 0.05 + 0.35 * np.sqrt(i / N) + np.random.uniform(-0.02, 0.02)
        centers[i, 0] = 0.5 + radius * np.cos(angle)
        centers[i, 1] = 0.5 + radius * np.sin(angle)
    centers = np.clip(centers, 0.05, 0.95)
    return centers
