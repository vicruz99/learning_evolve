import numpy as np
from scipy.optimize import minimize, linprog

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraint_fun(x):
    """Returns all inequality constraints >= 0 (vectorized)."""
    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]
    
    c = np.concatenate([
        xs - rs, 1.0 - xs - rs,
        ys - rs, 1.0 - ys - rs
    ])
    
    i_idx, j_idx = np.tril_indices(N, -1)
    dx = xs[i_idx] - xs[j_idx]
    dy = ys[i_idx] - ys[j_idx]
    dr = rs[i_idx] + rs[j_idx]
    
    c = np.concatenate([c, dx**2 + dy**2 - dr**2])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-7, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)])
    return b

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    for i in range(n):
        bound = min(centers[i, 0], 1.0 - centers[i, 0], 
                    centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, bound)
        idx += 1
        
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], 
                            centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist)
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return np.full(n, 0.01)

def make_hex_init(r0, angle=0.0):
    """Generate hexagonal lattice initialization."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 15:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        if y > 1.0 + r0:
            break
    pts = np.array(pts[:N + 15])
    
    if angle != 0.0:
        c_a, s_a = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c_a, -s_a], [s_a, c_a]]) + 0.5
    
    mask = (pts[:, 0] >= 0.01) & (pts[:, 0] <= 0.99) & (pts[:, 1] >= 0.01) & (pts[:, 1] <= 0.99)
    pts = pts[mask]
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.15, 0.85, (1, 2))])
    return pts[:N]

def make_staggered_init(pattern, r0=0.09):
    """Create initialization with specific row pattern."""
    pts = []
    row_idx = 0
    y = r0
    
    for row_len in pattern:
        if row_idx % 2 == 0:
            x_start = r0
        else:
            x_start = 2.0 * r0
        
        if row_len >= 1:
            if row_len * 2.0 * r0 + 2.0 * r0 > 1.0:
                spacing = (1.0 - 2.0 * r0) / (row_len - 1) if row_len > 1 else 0.5
            else:
                spacing = 2.0 * r0
            
            x = x_start
            for _ in range(row_len):
                if x + r0 <= 1.0:
                    pts.append([x, y])
                    x += spacing
        
        y += np.sqrt(3.0) * r0
        row_idx += 1
    
    while len(pts) < N:
        pts.append(np.random.uniform(0.15, 0.85, 2))
    return np.array(pts[:N])

def force_spread_init(seed, steps=500):
    """Force-directed spread initialization."""
    np.random.seed(seed)
    pts = np.random.uniform(0.1, 0.9, (N, 2))
    for step in range(steps):
        f = np.zeros_like(pts)
        for i in range(N):
            for j in range(i + 1, N):
                diff = pts[j] - pts[i]
                d = np.linalg.norm(diff)
                if d < 0.3 and d > 1e-5:
                    force = 0.001 / (d**2 + 0.001)
                    f[i] -= diff * force / d
                    f[j] += diff * force / d
            for dim in range(2):
                if pts[i, dim] < 0.06:
                    f[i, dim] += 0.06
                elif pts[i, dim] > 0.94:
                    f[i, dim] -= 0.06
        lr = 0.06 * max(0.1, 1.0 - step / steps)
        pts += f * lr
        pts = np.clip(pts, 0.02, 0.98)
    return pts

def corner_pushed_init(seed):

    np.random.seed(seed)

    # Stronger attraction to corners

        f = np.zeros_like(pts)
        for i in range(N):
            # Repel from other circles
            for j in range(N):
                if i != j:
                    diff = pts[j] - pts[i]
                    d = np.linalg.norm(diff)
                    if d < 0.25 and d > 1e-5:
                        force = 0.002 / (d**2 + 0.001)
                        f[i] -= diff * force / d
            
            # Attract to nearest corner
            dists_to_corners = np.linalg.norm(corners - pts[i], axis=1)
            nearest_corner = corners[np.argmin(dists_to_corners)]
            diff = nearest_corner - pts[i]
            d = np.linalg.norm(diff)
            if d > 0.05:
                f[i] += diff * 0.02 / (d + 0.1)

            diff_center = pts[i] - np.array([0.5, 0.5])
            d_center = np.linalg.norm(diff_center)
            if d_center > 0.01:
                f[i] += diff_center * 0.01 / (d_center + 0.1)
        
        lr = 0.04 * max(0.05, 1.0 - step / steps)
        pts += f * lr
        pts = np.clip(pts, 0.02, 0.98)
    
    return pts

def project_to_bounds(x0):
    """Ensure optimization vector strictly respects bounds."""
    x0 = x0.copy()
    for i in range(N):
        r = max(1e-7, x0[3 * i + 2])
        x0[3 * i] = np.clip(x0[3 * i], r, 1.0 - r)
        x0[3 * i + 1] = np.clip(x0[3 * i + 1], r, 1.0 - r)
        x0[3 * i + 2] = r
    return x0

def run_single_opt(x0, bounds, cons, maxiter=8000):
    """Run a single SLSQP optimization."""
    try:
        res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-13, 'disp': False})
        return res
    except Exception:
        return None

def check_validity(x, tol=1e-5):
    """Check if optimization result is valid."""
    vals = constraint_fun(x)
    return np.min(vals) >= -tol

def do_lp_slqp_cycle(x0, bounds, cons, num_cycles=6):
    """Run alternating LP-SLSQP cycles."""
    for cycle in range(num_cycles):
        res = run_single_opt(x0, bounds, cons, maxiter=10000)
        if res is None or np.isnan(res.fun):
            break
        
        centers = np.column_stack((res.x[0::3], res.x[1::3]))
        r_new = solve_lp_radii(centers)
        if r_new is None:
            break
        
        x0 = np.zeros(3 * N)
        x0[0::3] = centers[:, 0]
        x0[1::3] = centers[:, 1]
        x0[2::3] = np.maximum(r_new * 0.995, 0.005)
        x0 = project_to_bounds(x0)
    
    return x0
