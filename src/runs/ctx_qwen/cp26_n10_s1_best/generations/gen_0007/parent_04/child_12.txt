import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist)
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return np.full(n, 0.01)

def sa_local_search(centers, steps=1500, temp=0.04):
    """Simulated annealing local search optimizing centers, evaluating with LP."""
    current_radii = solve_lp_radii(centers)
    best_sum = current_radii.sum()
    best_centers = centers.copy()
    curr_sum = best_sum
    
    for step in range(steps):
        # Annealing schedule
        t = temp * (1.0 - step / steps)
        i = np.random.randint(N)
        old = centers[i].copy()
        centers[i] += np.random.uniform(-0.04, 0.04, 2)
        centers[i] = np.clip(centers[i], 0.02, 0.98)
        
        radii = solve_lp_radii(centers)
        new_sum = radii.sum()
        
        # Metropolis criterion
        if new_sum > curr_sum or np.random.rand() < np.exp((new_sum - curr_sum) / max(t, 1e-6)):
            curr_sum = new_sum
            if new_sum > best_sum:
                best_sum = new_sum
                best_centers = centers.copy()
        else:
            centers[i] = old
            
    return best_centers, best_sum
