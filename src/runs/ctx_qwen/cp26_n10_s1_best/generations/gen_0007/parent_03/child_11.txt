import numpy as np
from scipy.optimize import minimize, linprog
import warnings
warnings.filterwarnings('ignore')

N = 26

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints_linear(x):
    """Returns all inequality constraints >= 0 using linear distance formulation."""
    xs, ys, rs = x[0::3], x[1::3], x[2::3]
    
    # Boundary constraints
    c = np.concatenate([xs - rs, 1.0 - xs - rs, ys - rs, 1.0 - ys - rs])
    
    # Overlap constraints: dist >= r_i + r_j (linear in radius, smoother gradient)
    dist_matrix = np.sqrt((xs[:, None] - xs[None, :])**2 + (ys[:, None] - ys[None, :])**2)
    rs_sum = rs[:, None] + rs[None, :]
    
    # Only upper triangle to avoid duplicates
    idx_i, idx_j = np.triu_indices(N, 1)
    c = np.concatenate([c, dist_matrix[idx_i, idx_j] - rs_sum[idx_i, idx_j]])
    
    return c

def constraints_squared(x):
    """Returns all inequality constraints >= 0 (vectorized squared version)."""
    xs, ys, rs = x[0::3], x[1::3], x[2::3]
    
    # Boundary constraints
    c = np.concatenate([xs - rs, 1.0 - xs - rs, ys - rs, 1.0 - ys - rs])
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    idx_i, idx_j = np.tril_indices(N, -1)
    c = np.concatenate([c, dx[idx_i, idx_j]**2 + dy[idx_i, idx_j]**2 - dr[idx_i, idx_j]**2])
    
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-7, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)])
    return b

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist)
            idx += 1
        
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-8)
    except Exception:
        pass
    return np.full(n, 0.06)

def project_to_feasible(x):
    """Project variables to strictly satisfy bounds."""
    for i in range(N):
        r = max(1e-6, x[3 * i + 2])
        x[3 * i] = np.clip(x[3 * i], r, 1.0 - r)
        x[3 * i + 1] = np.clip(x[3 * i + 1], r, 1.0 - r)
        x[3 * i + 2] = r
    return x

def force_directed_init(n, seed, steps=300, target_spacing=0.22):
    """Force-directed layout to spread points evenly and push to boundaries."""
    np.random.seed(seed)
    centers = np.random.uniform(0.15, 0.85, (n, 2))
    
    for step in range(steps):
        forces = np.zeros_like(centers)
        lr = 0.03 * max(0.1, 1.0 - step / steps)
        
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[j] - centers[i]
                d = np.hypot(dx[0], dx[1])
                if d < target_spacing * 1.5 and d > 1e-6:
                    # Repulsive force
                    rep = 0.01 / (d**2 + 0.0001)
                    fx, fy = rep * dx[0] / d, rep * dx[1] / d
                    forces[i] -= [fx, fy]
                    forces[j] += [fx, fy]
            
            # Boundary repulsion - push toward corners/edges strategically
            for dim in range(2):
                if centers[i, dim] < 0.08:
                    forces[i, dim] += 0.04
                elif centers[i, dim] > 0.92:
                    forces[i, dim] -= 0.04
        
        centers += forces * lr
        centers = np.clip(centers, 0.02, 0.98)
    
    return centers

def make_hex_init(r0, angle, scale=1.0):
    """Generate a rotated and scaled hexagonal lattice initialization."""
    pts = []
    y = r0 * scale
    row = 0
    while len(pts) < N + 15:
        x = r0 * scale if row % 2 == 0 else 2.0 * r0 * scale
        while x <= 1.0 - r0 * scale:
            pts.append([x, y])
            x += 2.0 * r0 * scale
        y += np.sqrt(3.0) * r0 * scale
        row += 1
        if y > 1.0 + r0 * scale:
            break
    
    pts = np.array(pts[:N + 15])
    
    # Rotate around center
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
    
    # Filter points inside bounds
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    
    # Pad if needed
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.15, 0.85, (1, 2))])
    
    return pts[:N].copy()

def optimize_centers_given_radii(centers, radii, maxiter=5000):
    """Optimize centers while keeping radii fixed."""
    n = N
    
    def obj_centers(c_flat):
        # Penalize overlaps heavily
        penalty = 0.0
        for i in range(n):
            x, y = c_flat[2*i], c_flat[2*i+1]
            r = radii[i]
            # Boundary penalty
            if x < r: penalty += 1000 * (r - x)**2
            if x > 1-r: penalty += 1000 * (x - (1-r))**2
            if y < r: penalty += 1000 * (r - y)**2
            if y > 1-r: penalty += 1000 * (y - (1-r))**2
        
        # Pairwise overlap penalty
        for i in range(n):
            for j in range(i+1, n):
                d = np.hypot(c_flat[2*i]-c_flat[2*j], c_flat[2*i+1]-c_flat[2*j+1])
                min_d = radii[i] + radii[j]
                if d < min_d:
                    penalty += 500 * (min_d - d)**2
        return penalty
    
    def bounds_centers():
        b = []
        for i in range(n):
            r = radii[i]
            b.extend([(r, 1-r), (r, 1-r)])
        return b
    
    x0 = centers.flatten().copy()
    try:
        res = minimize(obj_centers, x0, method='L-BFGS-B', bounds=bounds_centers(),
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return res.x.reshape(n, 2)
    except Exception:
        return centers

def identify_tight_pairs(centers, radii, tolerance=1e-4):
    """Find pairs of circles that are very close to touching."""
    tight = []
    for i in range(N):
        for j in range(i+1, N):
            d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
            if d < radii[i] + radii[j] + tolerance:
                tight.append((i, j))
    return tight
