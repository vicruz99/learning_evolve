import numpy as np
from scipy.optimize import minimize, linprog
import warnings
warnings.filterwarnings('ignore')

N = 26
TRIL = np.tril_indices(N, -1)


def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])


def constraints(x):
    """Returns all inequality constraints >= 0 (vectorized)."""
    xs, ys, rs = x[0::3], x[1::3], x[2::3]
    c = np.concatenate([xs - rs, 1.0 - xs - rs, ys - rs, 1.0 - ys - rs])
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    c = np.concatenate([c, dx[TRIL]**2 + dy[TRIL]**2 - dr[TRIL]**2])
    return c


def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-7, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)])
    return b


def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)

    idx = 0
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist)
            idx += 1

    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return np.full(n, 0.05)


def project_to_feasible(x):
    """Project variables to strictly satisfy bounds."""
    for i in range(N):
        r = max(1e-6, x[3 * i + 2])
        x[3 * i] = np.clip(x[3 * i], r, 1.0 - r)
        x[3 * i + 1] = np.clip(x[3 * i + 1], r, 1.0 - r)
        x[3 * i + 2] = r
    return x


def make_corner_init(seed, r_corner=0.12):
    """Place circles in corners and along edges, fill interior."""
    np.random.seed(seed)
    pts = np.zeros((N, 2))
    
    # 4 corners
    r = r_corner + np.random.uniform(-0.005, 0.005)
    pts[0] = [r, r]
    pts[1] = [1.0 - r, r]
    pts[2] = [r, 1.0 - r]
    pts[3] = [1.0 - r, 1.0 - r]
    
    idx = 4
    
    # Along edges
    edge_r = 0.07 + np.random.uniform(-0.005, 0.005)
    # Bottom edge
    x = r + edge_r * 2.5
    while x < 1.0 - r - 0.01 and idx < N:
        pts[idx] = [x + np.random.uniform(-0.01, 0.01), edge_r + np.random.uniform(-0.005, 0.005)]
        idx += 1
        x += edge_r * 2.2
    
    # Top edge
    x = r + edge_r * 2.5
    while x < 1.0 - r - 0.01 and idx < N:
        pts[idx] = [x + np.random.uniform(-0.01, 0.01), 1.0 - edge_r + np.random.uniform(-0.005, 0.005)]
        idx += 1
        x += edge_r * 2.2
    
    # Left edge
    y = r + edge_r * 2.5
    while y < 1.0 - r - 0.01 and idx < N:
        pts[idx] = [edge_r + np.random.uniform(-0.005, 0.005), y + np.random.uniform(-0.01, 0.01)]
        idx += 1
        y += edge_r * 2.2
    
    # Right edge
    y = r + edge_r * 2.5
    while y < 1.0 - r - 0.01 and idx < N:
        pts[idx] = [1.0 - edge_r + np.random.uniform(-0.005, 0.005), y + np.random.uniform(-0.01, 0.01)]
        idx += 1
        y += edge_r * 2.2
    
    # Fill interior with hex-like pattern
    int_r = 0.06 + np.random.uniform(-0.003, 0.003)
    y = 0.25
    row = 0
    while y < 0.75 and idx < N:
        start_x = 0.25 if row % 2 == 0 else 0.30
        x = start_x
        while x < 0.75 and idx < N:

        row += 1
    
    # Fill remaining randomly

        pts[idx] = np.random.uniform(0.15, 0.85, 2)
        idx += 1
    
    return pts


def make_hex_init(angle, scale=1.0, seed=0):
    """Generate hexagonal lattice with rotation and scaling."""
    np.random.seed(seed)
    pts = []
    r0 = 0.09 * scale
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        if y > 1.0 + r0:
            break

    pts = np.array(pts[:N + 10])
    
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5

    mask = (pts[:, 0] >= 0.01) & (pts[:, 0] <= 0.99) & (pts[:, 1] >= 0.01) & (pts[:, 1] <= 0.99)
    pts = pts[mask]
    
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.15, 0.85, (1, 2))])
    
    return pts[:N] + np.random.uniform(-0.002, 0.002, (N, 2))


def make_staggered_grid(seed, spacing=0.18):
    """Create a staggered grid with perturbations."""
    np.random.seed(seed)
    pts = []
    for row in range(7):
        y = 0.10 + row * spacing * 0.85
        offset = spacing * 0.5 if row % 2 == 1 else 0.0
        for col in range(5):
            if len(pts) >= N:
                break
            x = 0.10 + col * spacing + offset
            pts.append([x + np.random.uniform(-0.01, 0.01), y + np.random.uniform(-0.01, 0.01)])
    
    return np.array(pts[:N])


def make_force_init(seed, steps=400):
    """Force-directed initialization with boundary push."""
    np.random.seed(seed)
    pts = np.random.uniform(0.1, 0.9, (N, 2))
    
    for step in range(steps):
        forces = np.zeros_like(pts)
        lr = 0.04 * (1.0 - step / steps)
        
        for i in range(N):
            for j in range(i + 1, N):
                dx = pts[j] - pts[i]
                d = np.hypot(dx[0], dx[1])
                if d < 0.30 and d > 1e-6:
                    rep = 0.008 / (d * d + 0.002)
                    forces[i] -= dx * rep / d
                    forces[j] += dx * rep / d
            
            for dim in range(2):
                if pts[i, dim] < 0.12:
                    forces[i, dim] += 0.06
                elif pts[i, dim] > 0.88:
                    forces[i, dim] -= 0.06
        
        pts += forces * lr
        pts = np.clip(pts, 0.03, 0.97)
    
    return pts


def make_spiral_init(seed):
    """Spiral distribution of points."""
    np.random.seed(seed)
    pts = np.zeros((N, 2))
    golden_angle = np.pi * (3.0 - np.sqrt(5.0))
    
    for i in range(N):
        r = 0.42 * np.sqrt((i + 0.5) / N)
        theta = i * golden_angle + np.random.uniform(-0.1, 0.1)
        pts[i, 0] = 0.5 + r * np.cos(theta)
        pts[i, 1] = 0.5 + r * np.sin(theta)
        pts[i, 0] = np.clip(pts[i, 0], 0.05, 0.95)
        pts[i, 1] = np.clip(pts[i, 1], 0.05, 0.95)
    
    return pts


def optimize_one_start(pts, bounds, cons, max_cycles=5):
    """Optimize a single starting configuration."""
    x0 = np.zeros(3 * N)
    x0[0::3] = pts[:, 0]
    x0[1::3] = pts[:, 1]
    
    # LP initialization for radii

    best_x = x0.copy()
    best_sum = np.sum(x0[2::3])
    
    for cycle in range(max_cycles):
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 12000, 'ftol': 1e-13, 'disp': False})
            
            if not np.isnan(res.fun):
                c_tmp = np.column_stack((res.x[0::3], res.x[1::3]))
                r_new = solve_lp_radii(c_tmp)
                if r_new is not None:
                    x0[2::3] = np.maximum(r_new * 0.998, 0.005)
                    x0 = project_to_feasible(x0)
                    
                    curr_sum = np.sum(x0[2::3])
                    if curr_sum > best_sum:
                        best_sum = curr_sum
                        best_x = x0.copy()
                    
                    # Another SLSQP pass with LP radii

                        res2 = minimize(objective, x0, method='SLSQP', bounds=bounds,
                                        constraints=cons, options={'maxiter': 8000, 'ftol': 1e-13, 'disp': False})
                        if not np.isnan(res2.fun):

                                    best_sum = curr_sum2
                                    best_x = x0.copy()
                    except Exception:
                        pass
        except Exception:

    return best_x, best_sum
