import numpy as np
from scipy.optimize import minimize, linprog
import warnings
warnings.filterwarnings('ignore')

N = 26
TRIL = np.tril_indices(N, -1)

def objective_joint(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints_joint(x):
    """Returns all inequality constraints >= 0 (vectorized)."""
    xs, ys, rs = x[0::3], x[1::3], x[2::3]
    # Boundary constraints
    c = np.concatenate([xs - rs, 1.0 - xs - rs, ys - rs, 1.0 - ys - rs])
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    c = np.concatenate([c, dx[TRIL]**2 + dy[TRIL]**2 - dr[TRIL]**2])
    return c

def get_lp_radii(centers, A_ub, b_ub, triu_idx):
    """Optimally compute radii for fixed centers using Linear Programming."""
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    b_ub[:N] = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                           np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    # Overlap constraints: r_i + r_j <= dist(i, j)
    dx = centers[:, 0][:, None] - centers[:, 0][None, :]
    dy = centers[:, 1][:, None] - centers[:, 1][None, :]
    b_ub[N:] = np.hypot(dx, dy)[triu_idx]
    
    try:
        res = linprog(-np.ones(N), A_ub=A_ub, b_ub=b_ub, bounds=[(0.0, None)] * N, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return np.full(N, 0.01)

def sa_chain(init_centers, steps, A_ub, b_ub, triu_idx):
    """Simulated Annealing on centers using LP for objective evaluation."""
    curr_c = init_centers.copy()
    curr_sum = np.sum(get_lp_radii(curr_c, A_ub, b_ub, triu_idx))
    best_c = curr_c.copy()
    best_sum = curr_sum
    
    T = 0.05
    sigma = 0.03
    
    for step in range(steps):
        T *= 0.9995
        sigma = max(0.001, sigma * 0.9993)
        
        i = np.random.randint(N)
        old_pos = curr_c[i].copy()
        
        # Perturb one circle
        curr_c[i] = np.clip(curr_c[i] + np.random.normal(0, sigma, 2), 0.0, 1.0)
        
        new_r = get_lp_radii(curr_c, A_ub, b_ub, triu_idx)
        new_sum = np.sum(new_r)
        delta = new_sum - curr_sum
        
        # Metropolis criterion
        if delta > 0 or np.random.rand() < np.exp(delta / T):
            curr_sum = new_sum
            if new_sum > best_sum:
                best_sum = new_sum
                best_c = curr_c.copy()
        else:
            curr_c[i] = old_pos
            
    return best_c, best_sum

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

        for j in range(i + 1, N):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            idx += 1
            
    triu_idx = np.triu_indices(N, k=1)

    best_overall_sum = -1.0

        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
        starts.append(pts)

                        break

            break

        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)
        
    return centers, radii, float(np.sum(radii))
