import numpy as np
from scipy.optimize import minimize

def compute_objective(vars):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(vars[2::3])

def compute_constraints(vars):
    """
    Computes all inequality constraints g(vars) >= 0.
    Includes boundary constraints and pairwise non-overlap constraints.
    """
    n = len(vars) // 3
    constraints = []
    
    # Boundary constraints: circle must be inside [0,1]x[0,1]
    # x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
    for i in range(n):
        idx = 3 * i
        x, y, r = vars[idx], vars[idx+1], vars[idx+2]
        constraints.append(x - r)
        constraints.append(1.0 - x - r)
        constraints.append(y - r)
        constraints.append(1.0 - y - r)
        
    # Non-overlap constraints: dist(i,j) >= r_i + r_j
    # Equivalent to dist^2 - (r_i + r_j)^2 >= 0
    for i in range(n):
        for j in range(i + 1, n):
            idx_i = 3 * i
            idx_j = 3 * j
            dx = vars[idx_i] - vars[idx_j]
            dy = vars[idx_i + 1] - vars[idx_j + 1]
            dr = vars[idx_i + 2] + vars[idx_j + 2]
            constraints.append(dx*dx + dy*dy - dr*dr)
            
    return np.array(constraints)

def create_hex_init(n, angle_deg, r_guess):
    """Generates a rotated hexagonal lattice initialization."""
    points = []
    # Spacing parameters for a dense hex grid
    dx = 0.22
    dy = dx * np.sqrt(3.0) / 2.0
    
    for row in range(-15, 15):
        for col in range(-15, 15):
            x = col * dx + (0.0 if row % 2 == 0 else dx / 2.0)
            y = row * dy
            points.append([x, y])
            
    pts = np.array(points)
    min_pt = pts.min(axis=0)
    max_pt = pts.max(axis=0)
    center = (min_pt + max_pt) / 2.0
    size = np.max(max_pt - min_pt)

    pts_shifted = pts - center

    theta = np.radians(angle_deg)

    centers = np.dot(centers - 0.5, rot_mat.T) + 0.5

    vars[1::3] = centers[:, 1]
    vars[2::3] = r_guess
    return vars

    r = 0.09
    idx = 0

        vars[idx*3] = 0.5
        vars[idx*3+1] = 0.5
        vars[idx*3+2] = 0.04
    return vars
