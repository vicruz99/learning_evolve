Strategy:
1. **Initialization**: Generate a hexagonal grid of points, select the 26 points closest to the center (0.5, 0.5) to form a compact cluster. Set initial radii to 0.085 to ensure strict feasibility and give the optimizer room to expand.
2. **Optimization**: Use `scipy.optimize.minimize` with the `SLSQP` method. SLSQP handles non-linear inequality constraints well. We optimize centers and radii jointly to maximize $\sum r_i$.
3. **Constraints**: Enforce boundary constraints ($x_i \pm r_i \in [0,1]$, $y_i \pm r_i \in [0,1]$) and non-overlap constraints using squared distances: $(x_i-x_j)^2 + (y_i-y_j)^2 \ge (r_i+r_j)^2$. This avoids `sqrt` in constraints, improving gradient stability.
4. **Robustness**: Run the optimizer from 20 different random perturbations of the hexagonal initialization to escape local minima. Perform a high-precision polishing step on the best result.
5. **Validation & Safety**: Apply the provided validation function. If numerical precision causes tiny violations, slightly shrink radii until valid. All helper functions are top-level with no closures or lambdas.

import numpy as np
from scipy.optimize import minimize

N = 26

def objective(vars):
    """Objective to minimize: negative sum of radii."""
    return -np.sum(vars[2::3])

def constraint_fun(vars):
    """Computes all inequality constraints >= 0."""
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    cons = []
    
    # Boundary constraints
    cons.extend((x - r).tolist())
    cons.extend((1.0 - x - r).tolist())
    cons.extend((y - r).tolist())
    cons.extend((1.0 - y - r).tolist())
    
    # Non-overlap constraints: squared distance >= (r_i + r_j)^2
    for i in range(N):
        for j in range(i + 1, N):
            dx = x[i] - x[j]
            dy = y[i] - y[j]
            dr = r[i] + r[j]
            cons.append(dx*dx + dy*dy - dr*dr)
            
    return np.array(cons)

def get_initial_guess(seed):
    """Generates a hexagonal grid initialization centered in the square."""
    np.random.seed(seed)
    pts = []
    step = 0.2
    for row in range(15):
        for col in range(15):
            x = col * step + (row % 2) * (step / 2)
            y = row * step * np.sqrt(3) / 2
            if x <= 1.0 and y <= 1.0:
                pts.append([x, y])
                
    pts = np.array(pts)
    center = np.array([0.5, 0.5])
    dists = np.linalg.norm(pts - center, axis=1)
    idx = np.argsort(dists)[:N]
    centers = pts[idx].copy()
    radii = np.full(N, 0.085)
    
    # Add controlled perturbation to break symmetry
    centers += np.random.uniform(-0.003, 0.003, centers.shape)
    radii += np.random.uniform(-0.003, 0.003, N)
    radii = np.clip(radii, 0.06, 0.12)
    
    # Ensure strict initial feasibility
    margin = 0.005
    centers = np.clip(centers, radii + margin, 1.0 - radii - margin)
    
    vars = np.zeros(3 * N)
    vars[0::3] = centers[:, 0]
    vars[1::3] = centers[:, 1]
    vars[2::3] = radii
    return vars

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    n = centers.shape[0]
    if np.isnan(centers).any() or np.isnan(radii).any():
        return False
    for i in range(n):
        if radii[i] < 0:
            return False
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            return False
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt((centers[i,0]-centers[j,0])**2 + (centers[i,1]-centers[j,1])**2)
            if dist < radii[i] + radii[j] - 1e-12:
                return False
    return True

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    bounds = []
    for _ in range(N):
        bounds.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
        
    cons_dict = {'type': 'ineq', 'fun': constraint_fun}
    
    best_sum = -1.0
    best_vars = None
    
    # Multiple restarts from perturbed hexagonal configurations
    for seed in range(20):
        x0 = get_initial_guess(seed)
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons_dict, 
                           options={'maxiter': 5000, 'ftol': 1e-12, 'disp': False})
            if not np.isnan(res.fun):
                current_sum = -res.fun
                if current_sum > best_sum:
                    c_opt = np.zeros((N, 2))
                    r_opt = np.zeros(N)
                    c_opt[:, 0] = res.x[0::3]
                    c_opt[:, 1] = res.x[1::3]
                    r_opt = res.x[2::3]
                    
                    if validate_packing(c_opt, r_opt):
                        best_sum = current_sum
                        best_vars = res.x.copy()
        except Exception:
            continue

    # Fallback if optimization fails completely
    if best_vars is None:
        best_vars = get_initial_guess(0)
        
    centers = np.zeros((N, 2))
    radii = np.zeros(N)
    centers[:, 0] = best_vars[0::3]
    centers[:, 1] = best_vars[1::3]
    radii = best_vars[2::3]
    
    # High-precision polishing step on the best configuration found
    try:
        res2 = minimize(objective, best_vars, method='SLSQP', bounds=bounds, constraints=cons_dict, 
                        options={'maxiter': 10000, 'ftol': 1e-14, 'disp': False})
        if not np.isnan(res2.fun):
            c_polish = np.zeros((N, 2))
            r_polish = np.zeros(N)
            c_polish[:, 0] = res2.x[0::3]
            c_polish[:, 1] = res2.x[1::3]
            r_polish = res2.x[2::3]
            # Accept polish if valid and not significantly worse (tolerance for numerical noise)
            if validate_packing(c_polish, r_polish) and -res2.fun > best_sum - 1e-9:
                centers = c_polish
                radii = r_polish
    except Exception:
        pass

    # Final safety check: shrink radii slightly if numerical errors cause violations
    if not validate_packing(centers, radii):
        for scale in np.arange(0.999, 0.95, -0.001):
            if validate_packing(centers, radii * scale):
                radii = radii * scale
                break
                
    return centers, radii, float(np.sum(radii))
