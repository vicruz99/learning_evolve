import numpy as np
from scipy.optimize import minimize

# Precompute indices for upper triangular matrix (i < j) to speed up constraint evaluation
_N = 26
_TRIU_IDX = np.triu_indices(_N, k=1)

def compute_constraints(vars):
    """
    Computes all inequality constraints g(vars) >= 0.
    Constraints:
    - Boundary: x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
    - Non-overlap: dist^2 - (r_i + r_j)^2 >= 0 for all i < j
    """
    n = _N
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    # Boundary constraints
    cons = [x - r, 1.0 - x - r, y - r, 1.0 - y - r]
    
    # Overlap constraints (vectorized)
    dx = x[:, None] - x
    dy = y[:, None] - y
    dr = r[:, None] + r
    val = dx**2 + dy**2 - dr**2
    cons.append(val[_TRIU_IDX])
    
    return np.concatenate(cons)

def objective(vars):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(vars[2::3])

def generate_init(seed):
    """
    Generates an initial configuration based on a perturbed hexagonal grid.

    np.random.seed(seed)
    n = _N
    r_init = 0.095
    pts = []
    
    # Generate hexagonal lattice points
    for row in range(12):
        y = r_init + row * np.sqrt(3) * r_init

            break

            x = r_init + col * 2 * r_init
            if row % 2 == 1:
                x += r_init
            if x > 1.0 - r_init:
                break
            pts.append([x, y])
            
    # Fallback to random if grid didn't yield enough points
    if len(pts) < n:
        pts = [np.random.uniform(0.1, 0.9, 2) for _ in range(n)]
        
    # Select first n points and add Gaussian noise to break symmetry
    pts = np.array(pts[:n])
    pts += np.random.normal(0, 0.02, pts.shape)
    pts = np.clip(pts, 0.05, 0.95)
    
    # Flatten to optimization variables: [x1, y1, r1, x2, y2, r2, ...]
    vars = np.zeros(3 * n)
    vars[0::3] = pts[:, 0]

    vars[2::3] = r_init

            pass
