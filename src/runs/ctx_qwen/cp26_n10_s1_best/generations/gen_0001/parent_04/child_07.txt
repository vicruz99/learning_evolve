import numpy as np
from scipy.optimize import minimize

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square

    Args:
        centers: np.array of shape (n, 2) with (x, y) coordinates
        radii: np.array of shape (n) with radius of each circle

    Returns:
        True if valid, False otherwise
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True

def constraint_func(vars):
    """Computes all inequality constraints >= 0 for SLSQP."""
    cons = np.zeros(429)
    idx = 0
    n = 26
    
    # Boundary constraints: x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
    for i in range(n):
        bi = 3 * i
        x, y, r = vars[bi], vars[bi+1], vars[bi+2]
        cons[idx] = x - r; idx += 1
        cons[idx] = 1.0 - x - r; idx += 1
        cons[idx] = y - r; idx += 1
        cons[idx] = 1.0 - y - r; idx += 1
        
    # Non-overlap constraints: dist^2 - (r1 + r2)^2 >= 0
    for i in range(n):
        bi = 3 * i
        for j in range(i + 1, n):
            bj = 3 * j
            dx = vars[bi] - vars[bj]
            dy = vars[bi+1] - vars[bj+1]
            r_sum = vars[bi+2] + vars[bj+2]
            cons[idx] = dx*dx + dy*dy - r_sum*r_sum
            idx += 1
            
    return cons

def objective_func(vars):
    """Objective to minimize: negative sum of radii."""
    return -np.sum(vars[2::3])

def hex_init(n, r_start, seed):
    """Generate a hexagonal lattice initial configuration."""
    np.random.seed(seed)
    centers = np.zeros((n, 2))
    radii = np.full(n, r_start)
    idx = 0
    row = 0
    col = 0
    dy = np.sqrt(3) * r_start
    
    while idx < n:
        x = r_start + col * 2 * r_start
        y = r_start + row * dy
        if row % 2 == 1:
            x += r_start
            
        if x > 1 - r_start or y > 1 - r_start:
            row += 1
            col = 0
            continue
            
        centers[idx] = [x, y]
        radii[idx] = r_start
        idx += 1
        col += 1
        
    # Add small noise to break symmetry
    centers += np.random.uniform(-0.005, 0.005, centers.shape)
    centers = np.clip(centers, r_start, 1 - r_start)
    return centers, radii

def run_packing():
    n = 26
    bounds = []
    for _ in range(n):
        bounds.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])

    cons_dict = {'type': 'ineq', 'fun': constraint_func}
    
    best_centers = None
    best_radii = None
    best_sum = -1.0

    def try_opt(centers, radii):
        nonlocal best_centers, best_radii, best_sum
        vars0 = np.zeros(3 * n)
        vars0[0::3] = centers[:, 0]
        vars0[1::3] = centers[:, 1]
        vars0[2::3] = radii
        
        res = minimize(objective_func, vars0, method='SLSQP', bounds=bounds, 
                      constraints=cons_dict, options={'maxiter': 2500, 'ftol': 1e-12, 'disp': False})
        
        c = res.x.reshape(n, 3)
        new_centers = c[:, :2]
        new_radii = c[:, 2]
        s = np.sum(new_radii)
        
        if s > best_sum and validate_packing(new_centers, new_radii):
            best_sum = s
            best_centers = new_centers.copy()
            best_radii = new_radii.copy()
            
    # 1. Hexagonal lattice with varying densities
    for r in [0.085, 0.09, 0.095, 0.10, 0.105]:
        for seed in range(4):
            c, r_arr = hex_init(n, r, seed)
            try_opt(c, r_arr)
            
    # 2. Rectangular grid layout
    c_grid = np.zeros((n, 2))
    r_grid = np.full(n, 0.09)
    idx = 0
    for i in range(6):
        for j in range(5):
            if idx < n:
                c_grid[idx] = [0.1 + j*0.17, 0.1 + i*0.17]
                idx += 1
    try_opt(c_grid, r_grid)

    # 3. Random initialization with force-directed repulsion
    for seed in range(5):
        np.random.seed(seed)
        c_rand = np.random.uniform(0.1, 0.9, (n, 2))
        r_rand = np.full(n, 0.06)
        for _ in range(150):
            for i in range(n):
                for j in range(i+1, n):
                    d = np.linalg.norm(c_rand[i] - c_rand[j])
                    if d < 0.22 and d > 1e-6:
                        shift = (0.22 - d) * 0.015
                        diff = c_rand[i] - c_rand[j]
                        c_rand[i] += diff / d * shift
                        c_rand[j] -= diff / d * shift
        c_rand = np.clip(c_rand, 0.05, 0.95)
        try_opt(c_rand, r_rand)

    # 4. Perturbation refinement of the best found solution
    if best_centers is not None:
        for seed in range(20):
            np.random.seed(seed)
            c_pert = best_centers + np.random.uniform(-0.004, 0.004, best_centers.shape)
            r_pert = best_radii + np.random.uniform(-0.002, 0.002, n)
            r_pert = np.clip(r_pert, 0.01, 0.3)
            for i in range(n):
                c_pert[i, 0] = np.clip(c_pert[i, 0], r_pert[i], 1.0 - r_pert[i])
                c_pert[i, 1] = np.clip(c_pert[i, 1], r_pert[i], 1.0 - r_pert[i])
            try_opt(c_pert, r_pert)

    # Fallback if optimization fails completely
    if best_centers is None:
        best_centers = np.zeros((n, 2))
        best_radii = np.full(n, 0.05)
        best_sum = n * 0.05

    # Final validation and safety shrinkage for numerical precision
    if not validate_packing(best_centers, best_radii):
        for _ in range(100):
            best_radii *= 0.999
            if validate_packing(best_centers, best_radii):
                break
        best_sum = np.sum(best_radii)

    return best_centers, best_radii, float(best_sum)
