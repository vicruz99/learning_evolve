import numpy as np
from scipy.optimize import minimize

N = 26

def objective(vars):
    """Objective to minimize: negative sum of radii."""
    return -np.sum(vars[2::3])

def constraints(vars):
    """Returns constraint vector g(vars) >= 0 for boundary and non-overlap."""
    cs = vars.reshape(N, 3)
    x = cs[:, 0]
    y = cs[:, 1]
    r = cs[:, 2]
    
    num_cons = 4 * N + N * (N - 1) // 2
    cons = np.empty(num_cons)
    idx = 0
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    cons[idx:idx+N] = x - r; idx += N
    cons[idx:idx+N] = 1.0 - x - r; idx += N
    cons[idx:idx+N] = y - r; idx += N
    cons[idx:idx+N] = 1.0 - y - r; idx += N
    
    # Non-overlap constraints: dist_sq >= (r_i + r_j)^2
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dr = r[:, None] + r[None, :]
    dist_sq = dx * dx + dy * dy
    cons_sq = dist_sq - dr * dr
    
    mask = np.triu(np.ones((N, N), dtype=bool), k=1)
    cons[idx:] = cons_sq[mask]
    return cons

def generate_hex(rows_counts, r_init):
    """Generates a hexagonal lattice of circle centers."""
    pts = []
    y = r_init
    for i, cnt in enumerate(rows_counts):
        x_start = r_init if i % 2 == 0 else 2 * r_init
        for k in range(cnt):
            pts.append([x_start + k * 2 * r_init, y])
        y += np.sqrt(3) * r_init
    return np.array(pts)

def run_packing():
    n = N
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * n
    
    best_sum = -1.0
    best_vars = None
    
    np.random.seed(42)
    configs = []
    
    # Hexagonal configurations with varying row counts and rotations
    for rows in [[5, 6, 5, 6, 4], [6, 5, 6, 5, 4], [4, 6, 6, 5, 5]]:
        pts = generate_hex(rows, 0.1)
        pts = pts - pts.mean(axis=0)
        scale = 0.7 / np.max(np.abs(pts))
        pts = pts * scale + 0.5
        for angle in [0, np.pi/6, np.pi/4, np.pi/3, np.pi/5]:
            cos_a, sin_a = np.cos(angle), np.sin(angle)
            rx = (pts[:, 0] - 0.5) * cos_a - (pts[:, 1] - 0.5) * sin_a + 0.5
            ry = (pts[:, 0] - 0.5) * sin_a + (pts[:, 1] - 0.5) * cos_a + 0.5
            configs.append(np.column_stack([rx, ry]))
            
    # Rotated grid configurations
    for angle in [0, 0.2, 0.4, 0.6]:
        pts = np.array([[0.1 + 0.2*i, 0.1 + 0.2*j] for i in range(5) for j in range(5)])
        pts = np.vstack([pts, [[0.5, 0.5]]])
        cos_a, sin_a = np.cos(angle), np.sin(angle)
        rx = (pts[:, 0] - 0.5) * cos_a - (pts[:, 1] - 0.5) * sin_a + 0.5
        ry = (pts[:, 0] - 0.5) * sin_a + (pts[:, 1] - 0.5) * cos_a + 0.5
        configs.append(np.column_stack([rx, ry]))
        
    # Random configurations
    for _ in range(15):
        configs.append(np.random.uniform(0.15, 0.85, (n, 2)))
        
    # Optimize from each configuration
    for c_init in configs:
        c_safe = np.clip(c_init, 0.05, 0.95)
        r_init = np.full(n, 0.05)
        vars0 = np.column_stack([c_safe[:, 0], c_safe[:, 1], r_init]).ravel()
        
        try:
            res = minimize(objective, vars0, method='SLSQP', bounds=bounds,
                           constraints={'type': 'ineq', 'fun': constraints},
                           options={'maxiter': 5000, 'ftol': 1e-12})
            if not np.isnan(res.fun) and -res.fun > best_sum:
                cs = res.x.reshape(n, 3)
                r_opt = cs[:, 2]
                if np.all(r_opt >= 0):
                    best_sum = -res.fun
                    best_vars = res.x.copy()
        except Exception:
            continue
            
    # Fallback if optimization fails
    if best_vars is None:
        best_vars = np.column_stack([np.full((n, 2), 0.5), np.full(n, 0.01)]).ravel()
        
    cs = best_vars.reshape(n, 3)
    centers = cs[:, :2]
    radii = cs[:, 2]
    
    # Ensure strict validity to pass validation function
    radii = np.maximum(radii, 1e-9)
    centers[:, 0] = np.clip(centers[:, 0], radii + 1e-9, 1.0 - radii - 1e-9)
    centers[:, 1] = np.clip(centers[:, 1], radii + 1e-9, 1.0 - radii - 1e-9)
    
    return centers, radii, float(np.sum(radii))
