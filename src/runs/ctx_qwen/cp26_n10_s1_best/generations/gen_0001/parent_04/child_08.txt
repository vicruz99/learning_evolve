import numpy as np
from scipy.optimize import minimize

def objective(x, n):
    """Objective to minimize: negative sum of radii."""
    return -np.sum(x[2::3])

def all_constraints(x, n):
    """Computes all boundary and non-overlap constraints as a vector >= 0."""
    cons = []
    # Boundary constraints: x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
    for i in range(n):
        xi, yi, ri = x[3*i], x[3*i+1], x[3*i+2]
        cons.append(xi - ri)
        cons.append(1.0 - xi - ri)
        cons.append(yi - ri)
        cons.append(1.0 - yi - ri)
        
    # Overlap constraints: dist(i, j) >= r_i + r_j
    # Using sqrt directly provides better gradient behavior than squared form
    for i in range(n):
        xi, yi, ri = x[3*i], x[3*i+1], x[3*i+2]
        for j in range(i + 1, n):
            xj, yj, rj = x[3*j], x[3*j+1], x[3*j+2]
            dx = xi - xj
            dy = yi - yj
            dist = np.sqrt(dx*dx + dy*dy)
            cons.append(dist - ri - rj)
            
    return np.array(cons)

def get_init_grid(n, seed):
    """Generates a feasible initial configuration based on a perturbed hexagonal grid."""
    np.random.seed(seed)
    dx = 0.2
    dy = 0.17
    centers = []
    y = 0.1
    row = 0
    # Generate points in a staggered pattern
    while len(centers) < n:
        x = 0.1 if row % 2 == 0 else 0.2
        while len(centers) < n and x < 0.9:
            centers.append([x, y])
            x += dx
        y += dy
        row += 1
        
    centers = np.array(centers[:n])
    # Add noise to break symmetry and help escape grid-local-minima
    centers += np.random.uniform(-0.015, 0.015, centers.shape)
    centers = np.clip(centers, 0.05, 0.95)
    
    # Start with small radii to guarantee initial feasibility
    radii = np.full(n, 0.05)
    
    x0 = np.zeros(3 * n)
    x0[0::3] = centers[:, 0]
    x0[1::3] = centers[:, 1]
    x0[2::3] = radii
    return x0

def run_packing():
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    n = 26
    bounds = []
    for _ in range(n):
        # x in [0,1], y in [0,1], r in [1e-6, 0.5]
        bounds.extend([(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)])
        
    cons = {'type': 'ineq', 'fun': all_constraints, 'args': (n,)}
    
    best_x = None
    best_val = -np.inf
    
    # Multi-start optimization from different perturbed grids
    for seed in range(20):
        x0 = get_init_grid(n, seed)
        try:
            res = minimize(objective, x0, args=(n,), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 2000, 'ftol': 1e-12, 'disp': False})
            if -res.fun > best_val:
                best_val = -res.fun
                best_x = res.x.copy()
        except Exception:
            continue
            
    # Local refinement: perturb the best solution found and re-optimize 
    # to escape shallow local minima
    if best_x is not None:
        for _ in range(5):
            x_pert = best_x + np.random.normal(0, 0.002, size=best_x.shape)
            # Project perturbed variables back to feasible bounds
            for i in range(n):
                r = np.clip(x_pert[3*i+2], 1e-6, 0.5)
                x_pert[3*i+2] = r
                x_pert[3*i] = np.clip(x_pert[3*i], r, 1.0 - r)
                x_pert[3*i+1] = np.clip(x_pert[3*i+1], r, 1.0 - r)
                
            try:
                res = minimize(objective, x_pert, args=(n,), method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 2000, 'ftol': 1e-12, 'disp': False})
                if -res.fun > best_val:
                    best_val = -res.fun
                    best_x = res.x.copy()
            except Exception:
                break

    # Fallback if optimization completely fails
    if best_x is None:
        best_x = get_init_grid(n, 0)
        
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    centers[:, 0] = best_x[0::3]
    centers[:, 1] = best_x[1::3]
    radii[:] = best_x[2::3]
    
    return centers, radii, float(np.sum(radii))
