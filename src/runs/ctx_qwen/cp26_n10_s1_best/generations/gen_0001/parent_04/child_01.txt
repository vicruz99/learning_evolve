import numpy as np
from scipy.optimize import minimize

N = 26

def objective(v):
    """Objective to minimize: negative sum of radii."""
    return -np.sum(v[2::3])

def objective_grad(v):
    """Analytical gradient of the objective function."""
    g = np.zeros(3 * N)
    g[2::3] = -1.0
    return g

def constraints(v):
    """
    Computes all inequality constraints g(v) >= 0.
    Includes boundary constraints and non-overlap constraints.
    """
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    
    nc = 4 * N + N * (N - 1) // 2
    c = np.empty(nc)
    idx = 0
    
    # Boundary constraints
    c[idx:idx+N] = x - r; idx += N          # x >= r
    c[idx:idx+N] = 1.0 - x - r; idx += N    # x + r <= 1
    c[idx:idx+N] = y - r; idx += N          # y >= r
    c[idx:idx+N] = 1.0 - y - r; idx += N    # y + r <= 1
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    for i in range(N):
        xi, yi, ri = x[i], y[i], r[i]
        for j in range(i + 1, N):
            dx = xi - x[j]
            dy = yi - y[j]
            dr = ri + r[j]
            c[idx] = dx*dx + dy*dy - dr*dr
            idx += 1
            
    return c

def hex_init(r0, rot):
    """Generates an initial hexagonal lattice configuration."""
    pts = []
    for row in range(10):
        for col in range(10):
            x = col * r0 * 1.5
            y = row * r0 * np.sqrt(3) * 0.75
            if row % 2 == 1:
                x += r0 * 0.75
            pts.append([x, y])
    pts = np.array(pts) - 0.5
    
    c, s = np.cos(rot), np.sin(rot)
    pts = pts @ np.array([[c, -s], [s, c]]) + 0.5
    
    dists = np.sum((pts - 0.5)**2, axis=1)
    idx = np.argsort(dists)[:N]
    return pts[idx], np.full(N, r0)
