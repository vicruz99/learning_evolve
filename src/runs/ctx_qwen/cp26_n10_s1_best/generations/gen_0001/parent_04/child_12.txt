import numpy as np
from scipy.optimize import minimize

N = 26

def objective(v):
    """Minimize negative sum of radii."""
    return -np.sum(v[2::3])

def objective_jac(v):
    """Analytical gradient of the objective function."""
    jac = np.zeros(3 * N)
    jac[2::3] = -1.0
    return jac

def constraints_and_jac(v):
    """Compute constraint values and their Jacobian matrix."""
    pts = v.reshape(N, 3)
    x = pts[:, 0]
    y = pts[:, 1]
    r = pts[:, 2]
    
    n_cons = 4 * N + N * (N - 1) // 2
    c = np.zeros(n_cons)
    jac = np.zeros((n_cons, 3 * N))
    
    idx = 0
    # Boundary constraints: x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
    for i in range(N):
        base = 3 * i
        c[idx] = x[i] - r[i]
        jac[idx, base] = 1.0
        jac[idx, base + 2] = -1.0
        idx += 1
        
        c[idx] = 1.0 - x[i] - r[i]
        jac[idx, base] = -1.0
        jac[idx, base + 2] = -1.0
        idx += 1
        
        c[idx] = y[i] - r[i]
        jac[idx, base + 1] = 1.0
        jac[idx, base + 2] = -1.0
        idx += 1
        
        c[idx] = 1.0 - y[i] - r[i]
        jac[idx, base + 1] = -1.0
        jac[idx, base + 2] = -1.0
        idx += 1
        
    # Non-overlap constraints: d^2 - (r_i + r_j)^2 >= 0
    for i in range(N):
        bi = 3 * i
        xi, yi, ri = x[i], y[i], r[i]
        for j in range(i + 1, N):
            bj = 3 * j
            xj, yj, rj = x[j], y[j], r[j]
            
            dx = xi - xj
            dy = yi - yj
            dr = ri + rj
            
            val = dx*dx + dy*dy - dr*dr
            c[idx] = val
            
            jac[idx, bi] = 2.0 * dx
            jac[idx, bi + 1] = 2.0 * dy
            jac[idx, bi + 2] = -2.0 * dr
            
            jac[idx, bj] = -2.0 * dx
            jac[idx, bj + 1] = -2.0 * dy
            jac[idx, bj + 2] = -2.0 * dr
            idx += 1
            
    return c, jac

def cons_fun(v):
    """Wrapper for constraint values."""
    return constraints_and_jac(v)[0]

def cons_jac_fun(v):
    """Wrapper for constraint Jacobian."""
    return constraints_and_jac(v)[1]

def get_init_guess(method='hex', seed=0, angle=0.0):
    """Generate diverse initial configurations."""
    np.random.seed(seed)
    centers = np.zeros((N, 2))
    radii = np.full(N, 0.06)
    
    if method == 'hex':
        # 5 rows with counts 5, 6, 5, 6, 4 sums to 26
        counts = [5, 6, 5, 6, 4]
        r = 0.09
        h = np.sqrt(3) * r
        y = r
        idx = 0
        for row_idx, cnt in enumerate(counts):
            x_start = r if row_idx % 2 == 0 else 2 * r
            for k in range(cnt):
                x = x_start + k * 2 * r
                centers[idx, 0] = x
                centers[idx, 1] = y
                idx += 1
            y += h
    elif method == 'grid':
        step = 0.18
        idx = 0
        for i in range(6):
            for j in range(5):
                if idx < N:
                    centers[idx, 0] = 0.1 + i * step
                    centers[idx, 1] = 0.1 + j * step * 1.2
                    idx += 1
        if idx < N:
            centers[idx, :] = 0.5
    elif method == 'rand':
        centers = np.random.uniform(0.1, 0.9, (N, 2))
        
    # Apply rotation to break symmetry
    if angle != 0.0:
        cx, cy = 0.5, 0.5
        cos_a, sin_a = np.cos(angle), np.sin(angle)
        dx = centers[:, 0] - cx
        dy = centers[:, 1] - cy
        centers[:, 0] = dx * cos_a - dy * sin_a + cx
        centers[:, 1] = dx * sin_a + dy * cos_a + cy
        
    # Add minor noise and clamp to safe bounds
    centers += np.random.normal(0, 0.005, centers.shape)
    centers = np.clip(centers, radii, 1 - radii)
    
    v = np.zeros(3 * N)
    v[0::3] = centers[:, 0]
    v[1::3] = centers[:, 1]
    v[2::3] = radii
    return v

def run_packing():
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    bounds = [(0, 1)] * (2 * N) + [(0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': cons_fun, 'jac': cons_jac_fun}
    
    best_v = None
    best_val = -np.inf
    
    # Explore multiple geometric layouts with rotations
    configs = [
        ('hex', 0, 0.0), ('hex', 1, 0.1), ('hex', 2, -0.1),
        ('grid', 0, 0.0), ('grid', 1, 0.2),
        ('rand', 0, 0.0), ('rand', 1, 0.0), ('rand', 2, 0.0),
        ('rand', 3, 0.0), ('rand', 4, 0.0)
    ]
    
    for method, seed, angle in configs:
        v0 = get_init_guess(method, seed, angle)
        try:
            res = minimize(objective, v0, method='SLSQP', jac=objective_jac,
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 1500, 'ftol': 1e-12, 'disp': False})
            if not np.isnan(res.fun) and -res.fun > best_val:
                # Verify numerical feasibility
                c_val = cons_fun(res.x)
                if np.min(c_val) > -1e-5:
                    best_val = -res.fun
                    best_v = res.x.copy()
        except Exception:
            pass
            
    # Perturbation refinement to escape minor local optima
    if best_v is not None:
        for _ in range(5):
            v_pert = best_v + np.random.normal(0, 1e-4, 3 * N)
            v_pert[0::3] = np.clip(v_pert[0::3], 0, 1)
            v_pert[1::3] = np.clip(v_pert[1::3], 0, 1)
            v_pert[2::3] = np.clip(v_pert[2::3], 0, 0.5)
            
            try:
                res = minimize(objective, v_pert, method='SLSQP', jac=objective_jac,
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 800, 'ftol': 1e-12, 'disp': False})
                if not np.isnan(res.fun) and -res.fun > best_val:
                    c_val = cons_fun(res.x)
                    if np.min(c_val) > -1e-5:
                        best_val = -res.fun
                        best_v = res.x.copy()
            except Exception:
                pass
                
    # Fallback if optimization unexpectedly fails
    if best_v is None:
        best_v = get_init_guess('grid', 0, 0.0)
        
    centers = np.zeros((N, 2))
    radii = np.zeros(N)
    centers[:, 0] = best_v[0::3]
    centers[:, 1] = best_v[1::3]
    radii = best_v[2::3]
    
    return centers, radii, float(np.sum(radii))
