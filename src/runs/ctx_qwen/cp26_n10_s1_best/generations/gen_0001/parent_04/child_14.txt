import numpy as np
from scipy.optimize import minimize

N = 26
NUM_VARS = N * 3

def objective_func(vars):
    """Objective: minimize negative sum of radii."""
    return -np.sum(vars[2::3])

def objective_grad(vars):
    """Analytical gradient of the objective function."""
    grad = np.zeros(NUM_VARS)
    grad[2::3] = -1.0
    return grad

def constraint_func(vars):
    """
    Returns a 1D array of constraint values that must be >= 0.
    Includes boundary constraints and non-overlap constraints.
    """
    cons = []
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    for i in range(N):
        idx = 3 * i
        x, y, r = vars[idx], vars[idx+1], vars[idx+2]
        cons.append(x - r)
        cons.append(1.0 - x - r)
        cons.append(y - r)
        cons.append(1.0 - y - r)
        
    # Non-overlap constraints: dist^2 >= (r_i + r_j)^2
    for i in range(N):
        idx_i = 3 * i
        xi, yi, ri = vars[idx_i], vars[idx_i+1], vars[idx_i+2]
        for j in range(i + 1, N):
            idx_j = 3 * j
            xj, yj, rj = vars[idx_j], vars[idx_j+1], vars[idx_j+2]
            dx = xi - xj
            dy = yi - yj
            dr = ri + rj
            cons.append(dx*dx + dy*dy - dr*dr)
    return np.array(cons)

def generate_initial_guess(seed, rotation=0.0):
    """Generates a hexagonal lattice initial guess with optional rotation."""
    np.random.seed(seed)
    centers = np.zeros((N, 2))
    radii = np.full(N, 0.06)  # Start small to ensure feasibility
    
    count = 0
    dx = 2 * 0.08
    dy = 0.08 * np.sqrt(3)
    
    y = 0.08
    row = 0
    while count < N:
        x = 0.08 if row % 2 == 0 else 0.16
        while count < N:
            centers[count] = [x, y]
            count += 1
            x += dx
            if x + 0.08 > 1.0:
                break
        y += dy
        if y + 0.08 > 1.0:
            break
        row += 1
        
    # Center and rotate
    cx, cy = 0.5, 0.5
    cos_r, sin_r = np.cos(rotation), np.sin(rotation)
    for i in range(N):
        dx_val, dy_val = centers[i, 0] - cx, centers[i, 1] - cy
        centers[i, 0] = dx_val * cos_r - dy_val * sin_r + cx
        centers[i, 1] = dx_val * sin_r + dy_val * cos_r + cy
        
    # Add small noise
    centers += np.random.uniform(-0.005, 0.005, size=centers.shape)
    centers[:, 0] = np.clip(centers[:, 0], 0.05, 0.95)
    centers[:, 1] = np.clip(centers[:, 1], 0.05, 0.95)
    
    vars = np.zeros(NUM_VARS)
    vars[0::3] = centers[:, 0]
    vars[1::3] = centers[:, 1]
    vars[2::3] = radii
    return vars

def relax_configuration(centers, radii, iters=500):
    """
    Force-directed relaxation to resolve overlaps and satisfy boundary constraints.
    Prepares a feasible starting point for the gradient-based optimizer.
    """
    n = len(centers)
    dt = 0.15
    for _ in range(iters):
        forces = np.zeros((n, 2))
        for i in range(n):
            for j in range(i+1, n):
                dx = centers[j, 0] - centers[i, 0]
                dy = centers[j, 1] - centers[i, 1]
                d_sq = dx*dx + dy*dy
                d = np.sqrt(d_sq)
                if d < 1e-8:
                    d = 1e-8
                    dx, dy = 1e-8, 0.0
                overlap = radii[i] + radii[j] - d
                if overlap > 0:
                    f = overlap * dt / d
                    fx, fy = f * dx, f * dy
                    forces[i] -= [fx, fy]
                    forces[j] += [fx, fy]
                    
            # Boundary repulsion
            if centers[i, 0] < radii[i]:
                forces[i, 0] += (radii[i] - centers[i, 0]) * dt
            elif centers[i, 0] > 1 - radii[i]:
                forces[i, 0] -= (centers[i, 0] - (1 - radii[i])) * dt
            if centers[i, 1] < radii[i]:
                forces[i, 1] += (radii[i] - centers[i, 1]) * dt
            elif centers[i, 1] > 1 - radii[i]:
                forces[i, 1] -= (centers[i, 1] - (1 - radii[i])) * dt
                
        centers += forces
        # Keep centers strictly inside [0, 1] during relaxation
        centers = np.clip(centers, 0.0, 1.0)
    return centers

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    best_sum = -1.0
    best_vars = None
    
    bounds = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraint_func}
    
    # Strategy 1: Hexagonal lattices with various rotations
    for seed in range(3):
        for rot in [0.0, 0.15, 0.3, 0.45, 0.6]:
            x0 = generate_initial_guess(seed, rot)
            centers = x0[0::3].copy().reshape(N, 2)
            radii = x0[2::3].copy()
            centers = relax_configuration(centers, radii)
            x0[0::3] = centers[:, 0]
            x0[1::3] = centers[:, 1]
            
            try:
                res = minimize(objective_func, x0, jac=objective_grad, bounds=bounds, constraints=cons, 
                               method='SLSQP', options={'maxiter': 6000, 'ftol': 1e-12, 'disp': False})
                if -res.fun > best_sum:
                    best_sum = -res.fun
                    best_vars = res.x.copy()
            except Exception:
                pass

    # Strategy 2: Randomized starts with relaxation
    for seed in range(5):
        np.random.seed(seed)
        temp_vars = np.zeros(NUM_VARS)

        temp_vars[1::3] = np.random.uniform(0.15, 0.85, N)
        temp_vars[2::3] = np.full(N, 0.05)
        
        centers = temp_vars[0::3].copy().reshape(N, 2)
        radii = temp_vars[2::3].copy()
        centers = relax_configuration(centers, radii)
        temp_vars[0::3] = centers[:, 0]
        temp_vars[1::3] = centers[:, 1]
        
        try:
            res = minimize(objective_func, temp_vars, jac=objective_grad, bounds=bounds, constraints=cons,
                           method='SLSQP', options={'maxiter': 6000, 'ftol': 1e-12, 'disp': False})
            if -res.fun > best_sum:
                best_sum = -res.fun
                best_vars = res.x.copy()
        except Exception:
            pass
            
    # Fallback
    if best_vars is None:
        best_vars = generate_initial_guess(0, 0.0)
        
    centers = np.zeros((N, 2))
    radii = np.zeros(N)
    for i in range(N):
        centers[i, 0] = best_vars[3*i]
        centers[i, 1] = best_vars[3*i+1]
        radii[i] = best_vars[3*i+2]
        
    # Final validation and strict scaling to guarantee passing the checker
    scale = 1.0
    for _ in range(500):
        valid = True
        for i in range(N):
            x, y, r = centers[i, 0], centers[i, 1], radii[i] * scale
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                valid = False
                break
        if valid:
            for i in range(N):
                for j in range(i+1, N):
                    dist = np.sqrt((centers[i,0]-centers[j,0])**2 + (centers[i,1]-centers[j,1])**2)
                    if dist < radii[i]*scale + radii[j]*scale - 1e-12:
                        valid = False
                        break
                if not valid: break
        if valid:
            break
        scale *= 0.999
        
    radii *= scale
    return centers, radii, float(np.sum(radii))
