import numpy as np
from scipy.optimize import minimize
import math

def objective(vars):
    """Minimize negative sum of radii."""
    return -np.sum(vars[2::3])

def constraint_fun(vars, n, triu_indices):
    """
    Vectorized constraint function.
    Returns array where each element must be >= 0.
    Structure: [x-r, 1-x-r, y-r, 1-y-r for each circle] + [dist^2 - (ri+rj)^2 for each pair]
    """
    xs = vars[0::3]
    ys = vars[1::3]
    rs = vars[2::3]
    
    n_con = 4 * n + len(triu_indices)
    c = np.empty(n_con)
    
    # Wall constraints
    c[:n] = xs - rs
    c[n:2*n] = 1.0 - xs - rs
    c[2*n:3*n] = ys - rs
    c[3*n:4*n] = 1.0 - ys - rs
    
    # Overlap constraints: dist^2 >= (ri + rj)^2
    dx = xs[triu_indices[:, 0]] - xs[triu_indices[:, 1]]
    dy = ys[triu_indices[:, 0]] - ys[triu_indices[:, 1]]
    dr = rs[triu_indices[:, 0]] + rs[triu_indices[:, 1]]
    
    offset = 4 * n
    c[offset:] = dx*dx + dy*dy - dr*dr
    
    return c

class ConstraintHolder:
    """Stateless container to pass precomputed indices to constraint function without closures/lambdas."""
    def __init__(self, n, indices):
        self.n = n
        self.indices = indices
        
    def __call__(self, vars):
        return constraint_fun(vars, self.n, self.indices)

def run_packing():
    n = 26
    num_vars = 3 * n
    
    # Bounds: x, y in [0, 1], r in [0, 0.5]
    bounds = [(0.0, 1.0) for _ in range(2 * n)] + [(0.0, 0.5) for _ in range(n)]
    
    # Precompute upper triangle indices for i < j
    i_idx, j_idx = np.triu_indices(n, k=1)
    triu_indices = np.column_stack((i_idx, j_idx))
    
    cons = {'type': 'ineq', 'fun': ConstraintHolder(n, triu_indices)}
    
    best_sum = -1.0
    best_vars = None
    
    initializations = []
    
    # 1. Hexagonal lattice initializations at different scales
    for scale in [0.82, 0.87, 0.92, 0.97, 1.02]:
        v = np.zeros(num_vars)
        r0 = 0.088 * scale
        idx = 0
        y = r0
        while idx < n:
            x = r0
            while idx < n:
                if x + r0 > 1.0:
                    break
                v[3*idx] = x
                v[3*idx+1] = y
                v[3*idx+2] = r0
                idx += 1
                x += 2 * r0
            y += math.sqrt(3) * r0
            if y + r0 > 1.0:
                break
        initializations.append(v)
        
    # 2. Grid initialization
    v_grid = np.zeros(num_vars)
    idx = 0
    for i in range(5):
        for j in range(5):
            if idx < n:
                v_grid[3*idx] = 0.1 + i * 0.2
                v_grid[3*idx+1] = 0.1 + j * 0.2
                v_grid[3*idx+2] = 0.09
                idx += 1
    if idx < n:
        v_grid[3*idx] = 0.5
        v_grid[3*idx+1] = 0.5
        v_grid[3*idx+2] = 0.05
    initializations.append(v_grid)
    
    # 3. Random initializations
    for seed in range(20):
        np.random.seed(seed * 137)
        v = np.zeros(num_vars)
        for i in range(n):
            r = np.random.uniform(0.04, 0.13)
            v[3*i] = np.random.uniform(r, 1-r)
            v[3*i+1] = np.random.uniform(r, 1-r)
            v[3*i+2] = r
        initializations.append(v)
        
    # Main optimization loop
    for x0 in initializations:
        # Perturb to break exact symmetries
        x0 = x0 + np.random.normal(0, 1e-4, num_vars)

        for i in range(n):
            r = x0[3*i+2]
            if r < 0.01: r = 0.01
            x0[3*i] = np.clip(x0[3*i], r, 1-r)
            x0[3*i+1] = np.clip(x0[3*i+1], r, 1-r)
            x0[3*i+2] = r

    if best_vars is None:

    centers = np.zeros((n, 2))

                    if dist < radii[i] + radii[j] - 1e-11:

                        break
                if not valid:

    return centers, radii, float(np.sum(radii))
