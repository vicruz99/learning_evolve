def objective(vars):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(vars[2::3])

def compute_constraints(vars):
    """Compute all inequality constraints: boundary and non-overlap."""
    n = len(vars) // 3
    num_ineq = 4 * n + n * (n - 1) // 2
    cons = np.zeros(num_ineq)
    idx = 0
    
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    for i in range(n):
        r_i = vars[3 * i + 2]
        x_i = vars[3 * i]
        y_i = vars[3 * i + 1]
        cons[idx] = x_i - r_i
        cons[idx + 1] = 1.0 - x_i - r_i
        cons[idx + 2] = y_i - r_i
        cons[idx + 3] = 1.0 - y_i - r_i
        idx += 4
        
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    for i in range(n):
        r_i = vars[3 * i + 2]
        x_i = vars[3 * i]
        y_i = vars[3 * i + 1]
        for j in range(i + 1, n):

            x_j = vars[3 * j]
            y_j = vars[3 * j + 1]
            dx = x_i - x_j
            dy = y_i - y_j
            cons[idx] = dx * dx + dy * dy - (r_i + r_j) ** 2
            idx += 1
    return cons
