import numpy as np
from scipy.optimize import minimize
import math

N_CIRCLES = 26

def compute_constraints_and_jac(v):
    """Computes constraint values and their Jacobian matrix."""
    n = N_CIRCLES
    num_vars = 3 * n
    M = 4 * n + n * (n - 1) // 2
    C = np.zeros(M)
    J = np.zeros((M, num_vars))
    idx = 0
    
    # Wall constraints
    for i in range(n):
        xi, yi, ri = v[3*i], v[3*i+1], v[3*i+2]
        
        # x >= r
        C[idx] = xi - ri
        J[idx, 3*i] = 1.0
        J[idx, 3*i+2] = -1.0
        idx += 1
        
        # 1 - x >= r
        C[idx] = 1.0 - xi - ri
        J[idx, 3*i] = -1.0
        J[idx, 3*i+2] = -1.0
        idx += 1
        
        # y >= r
        C[idx] = yi - ri
        J[idx, 3*i+1] = 1.0
        J[idx, 3*i+2] = -1.0
        idx += 1
        
        # 1 - y >= r
        C[idx] = 1.0 - yi - ri
        J[idx, 3*i+1] = -1.0
        J[idx, 3*i+2] = -1.0
        idx += 1

    # Overlap constraints: dist^2 >= (r1+r2)^2
    for i in range(n):
        xi, yi, ri = v[3*i], v[3*i+1], v[3*i+2]
        for j in range(i + 1, n):
            xj, yj, rj = v[3*j], v[3*j+1], v[3*j+2]
            dx = xi - xj
            dy = yi - yj
            dr = ri + rj
            
            C[idx] = dx*dx + dy*dy - dr*dr
            J[idx, 3*i] = 2.0 * dx
            J[idx, 3*i+1] = 2.0 * dy
            J[idx, 3*i+2] = -2.0 * dr
            J[idx, 3*j] = -2.0 * dx
            J[idx, 3*j+1] = -2.0 * dy
            J[idx, 3*j+2] = -2.0 * dr
            idx += 1
            
    return C, J

def obj_fun(v):
    return -np.sum(v[2::3])

def obj_jac(v):
    jac = np.zeros(len(v))
    jac[2::3] = -1.0
    return jac

def constraint_fun(v):
    return compute_constraints_and_jac(v)[0]

def constraint_jac(v):
    return compute_constraints_and_jac(v)[1]

def generate_hex_init(n, seed):
    """Generates a perturbed hexagonal lattice initialization."""
    np.random.seed(seed)
    v = np.zeros(3 * n)
    r_init = 0.09
    count = 0
    y = r_init
    row = 0
    
    while count < n:
        x = r_init
        if row % 2 == 1:
            x += r_init
        while count < n and x + r_init <= 1.0:
            v[3*count] = x + np.random.uniform(-0.008, 0.008)
            v[3*count+1] = y + np.random.uniform(-0.008, 0.008)
            v[3*count+2] = r_init + np.random.uniform(-0.005, 0.005)
            count += 1
            x += 2 * r_init
        y += r_init * math.sqrt(3)
        row += 1
        
    # Fill remaining if grid was sparse
    while count < n:
        r = np.random.uniform(0.06, 0.12)
        v[3*count] = np.random.uniform(r, 1-r)
        v[3*count+1] = np.random.uniform(r, 1-r)
        v[3*count+2] = r
        count += 1
    return v

def run_packing():

    num_vars = 3 * n
    bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n
    
    cons = {
        'type': 'ineq',
        'fun': constraint_fun,
        'jac': constraint_jac
    }
    
    best_sum = -1.0
    best_v = None
    
    # Multi-start optimization
    for seed in range(40):

        # Strictly enforce bounds on initial guess
        for i in range(n):

            v0[3*i+1] = np.clip(v0[3*i+1], r, 1-r)
            v0[3*i+2] = r
            
        try:

                               bounds=bounds, constraints=cons,
