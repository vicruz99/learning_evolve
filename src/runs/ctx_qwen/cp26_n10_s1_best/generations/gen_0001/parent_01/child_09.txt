import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def objective(vars):
    """Maximize sum of radii by minimizing negative sum."""
    return -np.sum(vars[2::3])

def constraint_func(vars):
    """Compute all boundary and overlap constraints as a single vector >= 0."""
    n = N_CIRCLES
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    # Boundary constraints: x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
    bound = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    
    # Overlap constraints: dist^2 - (r_i + r_j)^2 >= 0
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist_sq = dx**2 + dy**2
    
    r_sum = r[:, None] + r[None, :]
    min_dist_sq = r_sum**2
    
    # Select upper triangle to avoid duplicate pairs and self-distance
    idx = np.triu_indices(n, k=1)
    overlap = dist_sq[idx] - min_dist_sq[idx]
    
    return np.concatenate([bound, overlap])

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    (Copied exactly from prompt specification)
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        return False

    if np.isnan(radii).any():
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            return False
        elif np.isnan(radii[i]):
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False

    return True

def resolve_overlaps(centers, radii, n):
    """Push circles apart until no overlaps exist, then clip to bounds."""
    for _ in range(200):
        max_viol = 0
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[j, 0] - centers[i, 0]
                dy = centers[j, 1] - centers[i, 1]
                d = np.sqrt(dx*dx + dy*dy)
                if d < radii[i] + radii[j] and d > 1e-10:
                    ov = radii[i] + radii[j] - d
                    shift = ov * 0.5
                    inv_d = 1.0 / d
                    centers[i, 0] -= shift * dx * inv_d
                    centers[i, 1] -= shift * dy * inv_d
                    centers[j, 0] += shift * dx * inv_d
                    centers[j, 1] += shift * dy * inv_d
                    if ov > max_viol:
                        max_viol = ov
        if max_viol < 1e-7:
            break
            
    # Clip to valid boundary range
    for i in range(n):
        r = radii[i]
        centers[i, 0] = np.clip(centers[i, 0], r, 1.0 - r)
        centers[i, 1] = np.clip(centers[i, 1], r, 1.0 - r)
    return centers, radii

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = N_CIRCLES
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * n
    cons = {'type': 'ineq', 'fun': constraint_func}
    
    best_sum = -1.0
    best_vars = None
    
    # Try multiple diverse initializations to escape local minima
    for seed in range(40):
        np.random.seed(seed)
        centers = np.zeros((n, 2))
        radii = np.full(n, 0.06)
        
        # Generate hexagonal lattice layout
        s = 0.18
        count = 0
        row = 0
        while count < n:
            y = s + row * s * np.sqrt(3) / 2
            if y > 1.0 - s:
                row += 1
                continue
            x = s
            if row % 2 == 1:
                x += s / 2
            while count < n:
                if x <= 1.0 - s:
                    centers[count] = [x, y]
                    count += 1
                    x += s
                else:
                    break
            row += 1
            
        # Add random perturbation to break symmetry
        centers += np.random.uniform(-0.005, 0.005, centers.shape)
        
        # Ensure initial configuration is strictly feasible
        centers, radii = resolve_overlaps(centers, radii, n)
        
        # Flatten to optimization vector
        x0 = np.zeros(3 * n)
        x0[0::3] = centers[:, 0]
        x0[1::3] = centers[:, 1]
        x0[2::3] = radii
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 8000, 'ftol': 1e-12, 'disp': False})
            
            if not np.isnan(res.fun):
                curr_sum = -res.fun
                if curr_sum > best_sum:
                    best_sum = curr_sum
                    best_vars = res.x.copy()
        except Exception:
            continue
            
    # Extract best result
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        centers[i] = [best_vars[3*i], best_vars[3*i+1]]
        radii[i] = best_vars[3*i+2]
        
    # Post-validation fix for numerical tolerance
    if not validate_packing(centers, radii):
        for _ in range(100):
            radii *= 0.999
            if validate_packing(centers, radii):
                break
                
    return centers, radii, float(np.sum(radii))
