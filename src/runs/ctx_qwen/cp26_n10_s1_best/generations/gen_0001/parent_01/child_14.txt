import numpy as np
from scipy.optimize import minimize

N = 26

def objective(vars):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(vars[2*N:])

def constraint_fun(vars):
    """
    Returns a vector of constraint values. All must be >= 0.
    Includes wall constraints and pairwise non-overlap constraints.
    Vectorized for performance.
    """
    centers = vars[:2*N].reshape(N, 2)
    radii = vars[2*N:]
    
    # Wall constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    c_wall = np.concatenate([
        centers[:, 0] - radii,
        1.0 - centers[:, 0] - radii,
        centers[:, 1] - radii,
        1.0 - centers[:, 1] - radii
    ])
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2  =>  dist^2 - (r_i + r_j)^2 >= 0
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist_sq = np.sum(diff**2, axis=2)
    r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    
    # Only need upper triangle (i < j) to avoid duplicates and self-comparison
    idx = np.triu_indices(N, k=1)
    c_overlap = dist_sq[idx] - r_sum[idx]**2
    
    return np.concatenate([c_wall, c_overlap])

def make_init(seed, base_vars):
    """Generates a feasible initial configuration."""
    np.random.seed(seed)
    vars_init = np.zeros(3*N)
    
    if base_vars is not None:
        # Perturb existing best solution
        vars_init = base_vars.copy()
        vars_init += np.random.normal(0, 0.0015, 3*N)
    else:
        # Staggered hexagonal grid: 6-5-6-5-4
        counts = [6, 5, 6, 5, 4]
        r_init = 0.09
        vars_init[2*N:] = r_init
        
        dy = np.sqrt(3) * r_init
        y_start = 0.5 - 2 * dy
        y_pos = np.array([y_start + i*dy for i in range(5)])
        
        idx = 0
        for row, cnt in enumerate(counts):
            x_start = 0.5 - (cnt - 1) * r_init
            if row % 2 == 1:
                x_start += r_init
            for col in range(cnt):
                vars_init[idx*3] = x_start + col * 2 * r_init
                vars_init[idx*3+1] = y_pos[row]
                idx += 1
                
        vars_init += np.random.normal(0, 0.004, 3*N)
        
    # Project to bounds to ensure strict feasibility initially
    for i in range(N):
        r = max(0.01, vars_init[2*N+i])
        vars_init[2*i] = np.clip(vars_init[2*i], r, 1.0 - r)
        vars_init[2*i+1] = np.clip(vars_init[2*i+1], r, 1.0 - r)
        vars_init[2*N+i] = r
        
    return vars_init

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    bounds = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraint_fun}
    
    best_sum = -1.0
    best_vars = None
    
    # Phase 1: Broad search from structured initializations
    for seed in range(15):
        x0 = make_init(seed, None)
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 6000, 'ftol': 1e-13, 'disp': False})
            if -res.fun > best_sum:
                best_sum = -res.fun
                best_vars = res.x.copy()
        except Exception:
            pass
            
    # Phase 2: Local refinement via perturbations of the best found
    if best_vars is not None:
        for seed in range(30):
            x0 = make_init(seed, best_vars)
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
                if -res.fun > best_sum:
                    best_sum = -res.fun
                    best_vars = res.x.copy()
            except Exception:
                pass
                
    # Fallback if optimization completely fails
    if best_vars is None:
        best_vars = make_init(0, None)
        
    centers = best_vars[:2*N].reshape(N, 2)
    radii = best_vars[2*N:]
    
    # Phase 3: Numerical safety adjustment
    # SLSQP may produce microscopic violations due to floating point precision.
    # We gently scale down radii until strictly valid.
    factor = 1.0
    for _ in range(100):
        # Check wall constraints
        wall_ok = (np.all(centers[:,0] >= radii*factor) and 
                   np.all(centers[:,0] + radii*factor <= 1.0) and 
                   np.all(centers[:,1] >= radii*factor) and 
                   np.all(centers[:,1] + radii*factor <= 1.0))
        if not wall_ok:
            factor *= 0.995
            continue
            
        # Check overlap constraints
        diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
        dist = np.sqrt(np.sum(diff**2, axis=2))
        idx = np.triu_indices(N, k=1)
        r_sum_f = radii[:, np.newaxis] * factor + radii[np.newaxis, :] * factor
        
        if np.all(dist[idx] >= r_sum_f[idx] - 1e-12):
            radii *= factor
            break
        factor *= 0.995
        
    return centers, radii, float(np.sum(radii))
