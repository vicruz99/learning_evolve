import numpy as np
from scipy.optimize import minimize

def compute_objective(x, n):
    """Objective function: minimize negative sum of radii."""
    # Radii are located at indices 2, 5, 8, ... (index 3*i + 2)
    return -np.sum(x[2::3])

def compute_constraints(x, n):
    """
    Computes all inequality constraints g(x) >= 0.
    Includes boundary and non-overlap constraints.
    """
    c = []
    
    # Boundary constraints: circle must be inside [0,1]x[0,1]
    for i in range(n):
        idx = 3 * i
        xi, yi, ri = x[idx], x[idx+1], x[idx+2]
        c.append(xi - ri)            # x >= r
        c.append(1.0 - xi - ri)      # x + r <= 1
        c.append(yi - ri)            # y >= r
        c.append(1.0 - yi - ri)      # y + r <= 1
        
    # Non-overlap constraints: dist(i,j) >= r_i + r_j
    # Formulated as squared distance to maintain smoothness
    for i in range(n):
        idx_i = 3 * i
        xi, yi, ri = x[idx_i], x[idx_i+1], x[idx_i+2]
        for j in range(i + 1, n):
            idx_j = 3 * j
            xj, yj, rj = x[idx_j], x[idx_j+1], x[idx_j+2]
            dx = xi - xj
            dy = yi - yj
            rs = ri + rj
            c.append(dx * dx + dy * dy - rs * rs)
            
    return np.array(c)

def constraint_wrapper(x):
    """Wrapper to pass constraints to scipy optimizer without closures."""
    return compute_constraints(x, 26)

def run_packing():
    n = 26
    
    # Variable bounds: x,y in [0,1], r in [0, 0.5]
    bounds = []

        bounds.append((0.0, 1.0))
        bounds.append((0.0, 1.0))
        bounds.append((0.0, 0.5))
        
    cons = {'type': 'ineq', 'fun': constraint_wrapper}
    
    best_sum = -1.0
    best_x = None
    
    # --- Generate Initial Configurations ---
    inits = []
    
    # 1. Hexagonal Lattice Init
    # Distribution 5,6,5,6,4 sums to 26. Matches known optimal topology.
    pts = []
    row_counts = [5, 6, 5, 6, 4]
    y = 0.09
    dy = 0.09 * np.sqrt(3)
    for rc in row_counts:
        x = 0.09
        for _ in range(rc):
            pts.append((x, y))
            x += 0.18
        y += dy
        
    # Scale and center the lattice to roughly [0.1, 0.9]
    xs = [p[0] for p in pts]
    ys = [p[1] for p in pts]
    mn_x, mx_x = min(xs), max(xs)
    mn_y, mx_y = min(ys), max(ys)
    scale = 0.8 / max(mx_x - mn_x, mx_y - mn_y, 1e-9)
    
    x0_hex = np.zeros(3 * n)
    for i, (px, py) in enumerate(pts[:n]):
        nx = (px - (mn_x + mx_x)/2) * scale + 0.5
        ny = (py - (mn_y + mx_y)/2) * scale + 0.5
        x0_hex[3*i] = nx
        x0_hex[3*i+1] = ny
        x0_hex[3*i+2] = 0.09
    inits.append(x0_hex)
    
    # 2. Grid Init (5x5 + 1)
    x0_grid = np.zeros(3 * n)
    idx = 0
    for r in range(5):
        for c in range(5):
            if idx < n:
                x0_grid[3*idx] = 0.1 + c * 0.2
                x0_grid[3*idx+1] = 0.1 + r * 0.2
                x0_grid[3*idx+2] = 0.1
                idx += 1
    if idx < n:
        x0_grid[3*idx] = 0.5
        x0_grid[3*idx+1] = 0.5
        x0_grid[3*idx+2] = 0.04
    inits.append(x0_grid)
    
    # 3. Random Valid Init
    np.random.seed(42)
    x0_rand = np.zeros(3 * n)
    for i in range(n):
        r = np.random.uniform(0.06, 0.12)
        x0_rand[3*i] = np.random.uniform(r, 1-r)
        x0_rand[3*i+1] = np.random.uniform(r, 1-r)
        x0_rand[3*i+2] = r
    inits.append(x0_rand)

                x_pert[3*i] = np.clip(x_pert[3*i], r, 1-r)

                x_pert[3*i+2] = r
            try:
                res = minimize(compute_objective, x_pert, args=(n,), method='SLSQP',
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 8000, 'ftol': 1e-12, 'disp': False})
                if -res.fun > best_sum:
                    best_sum = -res.fun
                    best_x = res.x.copy()
            except Exception:
                pass
                
    if best_x is None:
        best_x = x0_grid

    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        centers[i, 0] = best_x[3*i]
        centers[i, 1] = best_x[3*i+1]
        radii[i] = best_x[3*i+2]

            radii *= 0.995
            if validate_packing(centers, radii):
                break
