def objective(vars):

    return -np.sum(vars[2::3])

def constraints(vars):

    n = 26
    num_cons = 4 * n + n * (n - 1) // 2
    c = np.zeros(num_cons)
    idx = 0
    
    # Wall constraints: x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
    for i in range(n):
        ri = vars[3 * i + 2]
        xi = vars[3 * i]
        yi = vars[3 * i + 1]
        c[idx] = xi - ri
        c[idx + 1] = 1.0 - xi - ri
        c[idx + 2] = yi - ri
        c[idx + 3] = 1.0 - yi - ri
        idx += 4
        
    # Overlap constraints: dist^2 - (r_i + r_j)^2 >= 0

        xi, yi, ri = vars[3 * i], vars[3 * i + 1], vars[3 * i + 2]
        for j in range(i + 1, n):

            dx = xi - xj
            dy = yi - yj
            rs = ri + rj
            c[idx] = dx * dx + dy * dy - rs * rs
            idx += 1
            
    return c

def resolve_overlaps(vars, n, steps=150):

            if 1.0 - yi - ri < 0: grad[3 * i + 1] += 10.0
