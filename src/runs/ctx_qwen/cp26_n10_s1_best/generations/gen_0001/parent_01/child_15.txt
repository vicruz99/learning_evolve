import numpy as np
from scipy.optimize import minimize

def evaluate_objective(vars, n):
    """Minimize negative sum of radii."""
    return -np.sum(vars[2::3])

def evaluate_constraints(vars, n):
    """Vectorized constraint evaluation: all values must be >= 0."""
    xs = vars[0::3]
    ys = vars[1::3]
    rs = vars[2::3]
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    cons = np.concatenate([
        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs
    ])
    
    # Overlap constraints: dist^2 >= (r1+r2)^2
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dist_sq = dx*dx + dy*dy
    r_sum = rs[:, None] + rs[None, :]
    min_dist_sq = r_sum * r_sum
    
    # Extract lower triangle (i > j) to avoid duplicates and self-distance
    tri_idx = np.tril_indices(n, -1)
    overlap_cons = dist_sq[tri_idx] - min_dist_sq[tri_idx]
    
    return np.concatenate([cons, overlap_cons])

def create_bounds(n):
    """Create variable bounds: x,y in [0,1], r in [0, 0.5]."""
    bounds = []
    for _ in range(n):
        bounds.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
    return bounds

def generate_initial_vars(n, seed):
    """Generate a valid initial configuration using a hexagonal lattice."""
    np.random.seed(seed)
    vars_ = np.zeros(3*n)
    r_init = 0.095
    count = 0
    y = r_init
    row_idx = 0
    
    # Place circles in hexagonal pattern
    while count < n:
        x = r_init
        if row_idx % 2 == 1:
            x += r_init
        while count < n:
            if x + r_init > 1.0:
                break
            vars_[3*count] = x
            vars_[3*count+1] = y
            vars_[3*count+2] = r_init
            count += 1
            x += 2*r_init
        y += np.sqrt(3)*r_init
        row_idx += 1
        if y + r_init > 1.0:
            break
            
    # Fill remaining slots randomly if grid didn't produce enough
    while count < n:
        r = np.random.uniform(0.05, 0.1)
        x = np.random.uniform(r, 1-r)
        y = np.random.uniform(r, 1-r)
        vars_[3*count] = x
        vars_[3*count+1] = y
        vars_[3*count+2] = r
        count += 1
        
    # Add small noise to break symmetry and prevent grid-locking
    vars_ += np.random.normal(0, 1e-4, 3*n)
    
    # Ensure strict validity roughly before optimization starts
    for i in range(n):
        r = vars_[3*i+2]
        if r < 0.01: r = 0.01
        vars_[3*i+2] = r
        vars_[3*i] = np.clip(vars_[3*i], r, 1-r)
        vars_[3*i+1] = np.clip(vars_[3*i+1], r, 1-r)
    return vars_

def run_packing():
    n = 26
    bounds = create_bounds(n)
    cons = {'type': 'ineq', 'fun': evaluate_constraints, 'args': (n,)}
    
    best_sum = -1.0
    best_vars = None
    
    # Phase 1: Multiple random starts from structured initialization
    for seed in range(15):
        vars_ = generate_initial_vars(n, seed)
        try:
            res = minimize(evaluate_objective, vars_, args=(n,), method='SLSQP', 
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 2000, 'ftol': 1e-12, 'disp': False})
            if -res.fun > best_sum:
                best_sum = -res.fun
                best_vars = res.x.copy()
        except Exception:
            pass

    # Phase 2: Perturbation & Refinement to escape local minima
    if best_vars is not None:

            perturbed = best_vars + np.random.normal(0, 1e-3, 3*n)
            # Project back to feasible region roughly
            for i in range(n):
                r = perturbed[3*i+2]
                if r < 0.01: r = 0.01
                perturbed[3*i+2] = r
                perturbed[3*i] = np.clip(perturbed[3*i], r, 1-r)
                perturbed[3*i+1] = np.clip(perturbed[3*i+1], r, 1-r)
            try:
                res = minimize(evaluate_objective, perturbed, args=(n,), method='SLSQP', 
                               bounds=bounds, constraints=cons,

                if -res.fun > best_sum:
                    best_sum = -res.fun
                    best_vars = res.x.copy()
            except Exception:
                pass

    # Fallback initialization if all optimizations failed
    if best_vars is None:
        best_vars = generate_initial_vars(n, 42)

                    valid_check = False

                        dx = centers[i,0] - centers[j,0]
                        dy = centers[i,1] - centers[j,1]
                        dist = np.sqrt(dx*dx + dy*dy)
                        if dist < radii[i] + radii[j] - 1e-12:
                            valid_check = False
                            break
                    if not valid_check:
                        break
                if valid_check:
                    break
                    
    return centers, radii, float(np.sum(radii))
