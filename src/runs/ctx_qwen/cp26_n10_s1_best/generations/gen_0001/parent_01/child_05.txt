import numpy as np
from scipy.optimize import minimize
import math

N_CIRCLES = 26

def all_constraints(vars, n):
    """Compute all boundary and overlap constraints in vectorized form."""
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    # Wall constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    # Must be >= 0
    wall_cons = np.concatenate([
        x - r,
        1.0 - x - r,
        y - r,
        1.0 - y - r
    ])
    
    # Overlap constraints: dist(i,j)^2 >= (r_i + r_j)^2
    # dist^2 - (r1+r2)^2 >= 0
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dr = r[:, None] + r[None, :]
    
    dist_sq = dx**2 + dy**2
    rad_sum_sq = dr**2
    
    # Extract only upper triangle indices (i < j) to avoid duplicates and self-pairs
    triu_indices = np.triu_indices(n, k=1)
    overlap_cons = dist_sq[triu_indices] - rad_sum_sq[triu_indices]
    
    return np.concatenate([wall_cons, overlap_cons])

def objective(vars, n):
    """Objective to minimize: negative sum of radii."""
    return -np.sum(vars[2::3])

def make_hex_init(scale, offset_x, offset_y, n):
    """Generate a hexagonal lattice initialization scaled and shifted."""
    r_init = 0.09 * scale
    vars_init = np.zeros(3 * n)
    count = 0
    y = r_init + offset_y
    row = 0
    while count < n:
        x = r_init + offset_x
        if row % 2 == 1:
            x += r_init  # Shift odd rows horizontally
        while count < n:
            if x + r_init > 1.0:
                break
            vars_init[3 * count] = x
            vars_init[3 * count + 1] = y
            vars_init[3 * count + 2] = r_init
            count += 1
            x += 2 * r_init
        y += math.sqrt(3) * r_init
        row += 1
        if y + r_init > 1.0:
            break
            
    # Fill any remaining circles with random valid positions
    while count < n:
        r_rand = np.random.uniform(0.05, 0.12)
        vars_init[3 * count] = np.random.uniform(r_rand, 1 - r_rand)
        vars_init[3 * count + 1] = np.random.uniform(r_rand, 1 - r_rand)
        vars_init[3 * count + 2] = r_rand
        count += 1
    return vars_init

def run_packing():
    n = N_CIRCLES
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
    constraints = {'type': 'ineq', 'fun': all_constraints, 'args': (n,)}
    
    best_sum = -1.0
    best_vars = None
    
    init_points = []
    
    # 1. Systematic hexagonal grid variations
    for scale in [0.8, 0.9, 1.0, 1.1]:
        for ox in [0.0, 0.02, -0.02]:
            for oy in [0.0, 0.02, -0.02]:
                init_points.append(make_hex_init(scale, ox, oy, n))
                
    # 2. Perturbed hex grid to break symmetry
    np.random.seed(42)
    base_hex = make_hex_init(1.0, 0.0, 0.0, n)
    for _ in range(20):
        pert = base_hex.copy()
        pert += np.random.normal(0, 0.005, pert.shape)
        # Project back to feasible region roughly
        for i in range(n):
            r = max(0.02, pert[3*i+2])
            pert[3*i] = np.clip(pert[3*i], r, 1-r)
            pert[3*i+1] = np.clip(pert[3*i+1], r, 1-r)
            pert[3*i+2] = r
        init_points.append(pert)
        
    # 3. Fully random starts for diversity
    for _ in range(10):
        rand_init = np.zeros(3*n)
        for i in range(n):
            r = np.random.uniform(0.06, 0.12)
            rand_init[3*i] = np.random.uniform(r, 1-r)
            rand_init[3*i+1] = np.random.uniform(r, 1-r)
            rand_init[3*i+2] = r
        init_points.append(rand_init)
        
    # Optimize each initialization
    for x0 in init_points:
        try:
            res = minimize(objective, x0, args=(n,), method='SLSQP', bounds=bounds,

        except Exception:
            pass

    for i in range(n):

        centers[i, 1] = best_vars[3*i+1]
        radii[i] = best_vars[3*i+2]
        
    return centers, radii, float(np.sum(radii))
