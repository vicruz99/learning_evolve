from scipy.optimize import minimize

def compute_objective(v):
    """Minimize negative sum of radii."""
    return -np.sum(v[2::3])

def compute_constraints(v):
    """
    Compute all inequality constraints g(v) >= 0.
    Includes boundary and non-overlap (squared distance) constraints.
    """
    n = len(v) // 3
    xs = v[0::3]
    ys = v[1::3]
    rs = v[2::3]
    
    # Boundary constraints: 4 per circle
    # x >= r, 1-x >= r, y >= r, 1-y >= r
    c_bound = np.empty(4 * n)
    c_bound[0::4] = xs - rs
    c_bound[1::4] = 1.0 - xs - rs
    c_bound[2::4] = ys - rs
    c_bound[3::4] = 1.0 - ys - rs
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2  =>  dist^2 - (r_i + r_j)^2 >= 0
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    dist_sq = dx * dx + dy * dy
    min_dist_sq = dr * dr
    
    # Extract only i < j to avoid duplicates and self-comparisons
    idx = np.triu_indices(n, k=1)
    c_overlap = dist_sq[idx] - min_dist_sq[idx]
    
    return np.concatenate([c_bound, c_overlap])
