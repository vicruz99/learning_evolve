import numpy as np
from scipy.optimize import minimize, linprog

def compute_penalty(centers_flat, n, r_curr):
    """
    Computes a smooth, differentiable penalty for boundary violations and overlaps
    for a fixed equal radius r_curr.
    """
    centers = centers_flat.reshape(n, 2)
    p = 0.0
    
    # Boundary penalties: circles must stay within [r, 1-r]
    diff = np.maximum(0.0, r_curr - centers[:, 0])
    p += np.sum(diff**2)
    diff = np.maximum(0.0, centers[:, 0] + r_curr - 1.0)
    p += np.sum(diff**2)
    diff = np.maximum(0.0, r_curr - centers[:, 1])
    p += np.sum(diff**2)
    diff = np.maximum(0.0, centers[:, 1] + r_curr - 1.0)
    p += np.sum(diff**2)
    
    # Overlap penalties: distance between centers must be >= 2*r_curr
    c1 = centers[:, np.newaxis, :]
    c2 = centers[np.newaxis, :, :]
    dist = np.sqrt(np.sum((c1 - c2)**2, axis=2))
    tri = np.tril_indices(n, -1)
    overlaps = np.maximum(0.0, 2*r_curr - dist[tri])
    p += np.sum(overlaps**2)
    return p

def solve_lp_radii(centers, n):
    """
    Solves the LP to maximize sum of radii given fixed centers.
    Constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i,
                 r_i + r_j <= dist(i,j), r_i >= 0
    """
    c_obj = -np.ones(n)
    A_ub_rows = []
    b_ub_rows = []
    
    for i in range(n):
        x, y = centers[i]
        row = np.zeros(n)
        row[i] = 1.0
        A_ub_rows.append(row)
        b_ub_rows.append(x)
        
        A_ub_rows.append(row)
        b_ub_rows.append(1.0 - x)
        
        A_ub_rows.append(row)
        b_ub_rows.append(y)
        
        A_ub_rows.append(row)
        b_ub_rows.append(1.0 - y)
        
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            d = np.sqrt(dx*dx + dy*dy)
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub_rows.append(row)
            b_ub_rows.append(d)
            
    A_ub = np.array(A_ub_rows)
    b_ub = np.array(b_ub_rows)
    bounds_r = [(0.0, 0.5) for _ in range(n)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method='highs')
    if res.success:
        return res.x, np.sum(res.x)
    return None, 0.0

def run_packing():

    np.random.seed(42)
    best_sum = 0.0
    best_centers = None
    
    # Prepare multiple initial layouts to escape local minima
    inits = []
    pts1 = []
    for y in np.arange(0.1, 0.95, 0.17):
        for x in np.arange(0.1, 0.95, 0.18):
            pts1.append([x, y])
    pts1 = np.array(pts1[:n])
    inits.append(pts1 + np.random.uniform(-0.005, 0.005, pts1.shape))
    
    # Rotated hex grid
    angle = 0.15
    c_mat = np.cos(angle)
    s_mat = np.sin(angle)
    rot_pts = np.zeros_like(pts1)
    for i in range(n):
        dx, dy = pts1[i, 0] - 0.5, pts1[i, 1] - 0.5
        rot_pts[i, 0] = dx * c_mat - dy * s_mat + 0.5
        rot_pts[i, 1] = dx * s_mat + dy * c_mat + 0.5
    inits.append(rot_pts + np.random.uniform(-0.005, 0.005, rot_pts.shape))
    
    # Stage 1: Geometric relaxation to find dense center configuration
    for init_centers in inits:
        centers = init_centers.copy()
        r_curr = 0.09
        step_size = 0.0003
        
        for iteration in range(600):
            res = minimize(compute_penalty, centers.flatten(), args=(n, r_curr),
                           method='L-BFGS-B',
                           bounds=[(0.0, 1.0)]*(2*n),
                           options={'maxiter': 1000, 'ftol': 1e-12})
            centers = res.x.reshape(n, 2)
            
            # If configuration is valid (penalty ~0), grow the radius
            if res.fun < 1e-7:
                r_curr += step_size
                # Occasional perturbation to escape local minima

                    centers += np.random.uniform(-0.002, 0.002, centers.shape)
                    centers = np.clip(centers, 0.05, 0.95)
            else:

                r_curr -= step_size * 0.5
                
            current_sum = n * r_curr

                best_sum = current_sum
                best_centers = centers.copy()

    if best_centers is None:
        best_centers = inits[0]
        best_sum = 26 * 0.09
        
    return best_centers, np.full(n, best_sum/n), float(best_sum)
