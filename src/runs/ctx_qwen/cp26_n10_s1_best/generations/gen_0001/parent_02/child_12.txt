import numpy as np
from scipy.optimize import minimize

N = 26
VAR_LEN = 3 * N

def obj_func(x):
    """Objective: minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraint_func(x):
    """Returns all inequality constraints as a vector (must be >= 0)."""
    num_cons = 4 * N + N * (N - 1) // 2
    cons = np.empty(num_cons)
    idx = 0
    
    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]
    
    # Boundary constraints
    for i in range(N):
        cons[idx] = xs[i] - rs[i]; idx += 1
        cons[idx] = 1.0 - xs[i] - rs[i]; idx += 1
        cons[idx] = ys[i] - rs[i]; idx += 1
        cons[idx] = 1.0 - ys[i] - rs[i]; idx += 1
        
    # Overlap constraints
    for i in range(N):
        for j in range(i + 1, N):
            dx = xs[i] - xs[j]
            dy = ys[i] - ys[j]
            cons[idx] = np.sqrt(dx * dx + dy * dy) - rs[i] - rs[j]
            idx += 1
            
    return cons

def make_bounds():
    """Create variable bounds for [x, y, r] for each circle."""
    bounds = []
    for _ in range(N):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 0.5)) # r
    return bounds

def hex_init(seed):
    """Generate a feasible hexagonal initial guess with slight perturbation."""
    np.random.seed(seed)
    x = np.zeros(VAR_LEN)
    r = 0.09
    count = 0
    y = r
    row = 0
    while count < N:
        shift = r if row % 2 == 1 else 0.0
        cx = r + shift
        while count < N:
            if cx + r > 1.0:
                break
            x[3 * count] = cx
            x[3 * count + 1] = y
            x[3 * count + 2] = r
            count += 1
            cx += 2.0 * r
        y += np.sqrt(3) * r
        row += 1
        
    # Add small noise to break symmetry
    x += np.random.normal(0, 0.002, VAR_LEN)
    
    # Enforce bounds strictly to ensure feasibility at start
    for i in range(N):
        r_val = max(0.01, x[3 * i + 2])
        x[3 * i + 2] = r_val
        x[3 * i] = np.clip(x[3 * i], r_val, 1.0 - r_val)
        x[3 * i + 1] = np.clip(x[3 * i + 1], r_val, 1.0 - r_val)
    return x

def run_packing():
    """Pack 26 circles in a unit square to maximize sum of radii."""
    best_sum = -1.0
    best_x = None
    bounds = make_bounds()
    cons = {'type': 'ineq', 'fun': constraint_func}
    
    # Phase 1: Diverse structured starts
    for seed in range(15):
        x0 = hex_init(seed)
        try:
            res = minimize(obj_func, x0, method='SLSQP', bounds=bounds, constraints=cons,

            if not np.isnan(res.fun) and -res.fun > best_sum:
                best_sum = -res.fun
                best_x = res.x.copy()
        except Exception:
            continue

    for _ in range(30):

            break
        x_pert = best_x + np.random.normal(0, 0.004, VAR_LEN)
        
        # Project perturbed variables back to feasible bounds
        for i in range(N):
            r_val = max(0.01, x_pert[3 * i + 2])
            x_pert[3 * i + 2] = r_val
            x_pert[3 * i] = np.clip(x_pert[3 * i], r_val, 1.0 - r_val)
            x_pert[3 * i + 1] = np.clip(x_pert[3 * i + 1], r_val, 1.0 - r_val)

    centers = np.zeros((N, 2))
    radii = np.zeros(N)

        radii[i] = best_x[3 * i + 2]
