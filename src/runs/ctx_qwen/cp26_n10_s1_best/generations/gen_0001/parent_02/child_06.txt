import numpy as np
from scipy.optimize import minimize

N = 26

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def get_constraints(x):
    """Returns array of constraint values (all must be >= 0)."""
    cons = []
    # Boundary constraints
    for i in range(N):
        base = 3 * i
        cx, cy, r = x[base], x[base+1], x[base+2]
        cons.append(cx - r)
        cons.append(1.0 - cx - r)
        cons.append(cy - r)
        cons.append(1.0 - cy - r)
        
    # Overlap constraints (squared distance form for smoother gradients)
    for i in range(N):
        base_i = 3 * i
        cx_i, cy_i, r_i = x[base_i], x[base_i+1], x[base_i+2]
        for j in range(i + 1, N):
            base_j = 3 * j
            dx = cx_i - x[base_j]
            dy = cy_i - x[base_j+1]
            dr = r_i + x[base_j+2]
            cons.append(dx*dx + dy*dy - dr*dr)
            
    return np.array(cons)

def run_packing():
    num_vars = 3 * N
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N
    constraints = {'type': 'ineq', 'fun': get_constraints}
    
    best_sum = -1.0
    best_x = None
    
    def try_opt(x0):
        nonlocal best_sum, best_x
        # Perturb initial guess to break symmetry
        x0 = x0 + np.random.normal(0, 1e-4, num_vars)
        x0[0::3] = np.clip(x0[0::3], 0.0, 1.0)
        x0[1::3] = np.clip(x0[1::3], 0.0, 1.0)
        x0[2::3] = np.clip(x0[2::3], 0.0, 0.5)
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints,
                           options={'maxiter': 8000, 'ftol': 1e-13, 'disp': False})
            curr_sum = -res.fun
            if curr_sum > best_sum:
                best_sum = curr_sum
                best_x = res.x.copy()
        except Exception:
            pass

    # --- Generate Diverse Initial Configurations ---
    inits = []
    
    # 1. Hexagonal lattice
    pts = []
    r_hex = 0.096
    y = r_hex
    row_idx = 0
    while y <= 1.0 - r_hex:
        x = r_hex
        shift = r_hex if row_idx % 2 == 1 else 0
        while x <= 1.0 - r_hex:
            pts.append([x + shift, y, r_hex])
            x += 2.0 * r_hex
        y += r_hex * np.sqrt(3)
        row_idx += 1
    pts = np.array(pts[:N] + [[0.5, 0.5, 0.05]] * max(0, N - len(pts)))
    inits.append(pts.flatten())
    
    # 2. Random dense packings
    for s in range(20):
        np.random.seed(s)
        cx = np.random.uniform(0.15, 0.85, N)
        cy = np.random.uniform(0.15, 0.85, N)
        r = np.random.uniform(0.06, 0.11, N)
        x0 = np.zeros(num_vars)
        x0[0::3] = cx
        x0[1::3] = cy
        x0[2::3] = r
        inits.append(x0)
        
    # 3. Perturbed grids
    for s in range(10):
        np.random.seed(100 + s)
        x0 = np.zeros(num_vars)
        idx = 0
        for row in range(5):
            for col in range(6):
                if idx < N:
                    cx = 0.1 + col * 0.155 + np.random.uniform(-0.015, 0.015)
                    cy = 0.1 + row * 0.175 + np.random.uniform(-0.015, 0.015)
                    x0[3*idx] = cx
                    x0[3*idx+1] = cy
                    x0[3*idx+2] = 0.092
                    idx += 1
        inits.append(x0)

    # Run optimization from each start
    for x0 in inits:
        try_opt(x0)
        
    # Refine the best solution found with decreasing noise
    if best_x is not None:
        for noise in [1e-3, 5e-4, 2e-4]:
            x0 = best_x + np.random.normal(0, noise, num_vars)
            x0[0::3] = np.clip(x0[0::3], 0.0, 1.0)
            x0[1::3] = np.clip(x0[1::3], 0.0, 1.0)
            x0[2::3] = np.clip(x0[2::3], 0.0, 0.5)
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints,
                               options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
                if -res.fun > best_sum:
                    best_sum = -res.fun
                    best_x = res.x.copy()
            except Exception:
                pass

    # Fallback if optimization failed completely
    if best_x is None:
        best_x = inits[0]
        
    centers = best_x.reshape(N, 3)[:, :2]
    radii = best_x.reshape(N, 3)[:, 2]
    
    # Post-processing validation repair
    def is_valid(c, r):
        if np.any(r < -1e-9): return False
        if np.any(c[:,0] < r - 1e-9) or np.any(c[:,0] + r > 1.0 + 1e-9): return False
        if np.any(c[:,1] < r - 1e-9) or np.any(c[:,1] + r > 1.0 + 1e-9): return False
        for i in range(len(c)):
            for j in range(i+1, len(c)):
                d = np.hypot(c[i,0]-c[j,0], c[i,1]-c[j,1])
                if d < r[i] + r[j] - 1e-9: return False
        return True

    if not is_valid(centers, radii):
        for _ in range(100):
            radii *= 0.995
            if is_valid(centers, radii):
                break
                
    return centers, radii, float(np.sum(radii))
