import numpy as np
from scipy.optimize import minimize

def objective_func(x, n):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraint_func(x, n):
    """Returns a vector of constraint values. All must be >= 0."""
    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]
    
    n_bounds = 4 * n
    n_pairs = n * (n - 1) // 2
    cons = np.zeros(n_bounds + n_pairs)
    
    idx = 0
    # Boundary constraints: x-r>=0, 1-x-r>=0, y-r>=0, 1-y-r>=0
    for i in range(n):
        cons[idx] = xs[i] - rs[i]
        cons[idx+1] = 1.0 - xs[i] - rs[i]
        cons[idx+2] = ys[i] - rs[i]
        cons[idx+3] = 1.0 - ys[i] - rs[i]
        idx += 4
        
    # Non-overlap constraints: dist^2 >= (r1+r2)^2
    for i in range(n):
        for j in range(i + 1, n):
            dx = xs[i] - xs[j]
            dy = ys[i] - ys[j]
            dr = rs[i] + rs[j]
            cons[idx] = dx*dx + dy*dy - dr*dr
            idx += 1
            
    return cons

def get_bounds(n):
    """Variable bounds: x,y in [0,1], r in [0, 0.5]"""
    b = []
    for _ in range(n):
        b.append((0.0, 1.0))
        b.append((0.0, 1.0))
        b.append((0.0, 0.5))
    return b

def create_init_hex(n, angle=0.0):
    """Hexagonal layout with optional rotation."""
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.095)
    rows_counts = [6, 5, 6, 5, 4]  # Sums to 26
    idx = 0
    r = radii[0]
    for row, count in enumerate(rows_counts):
        for col in range(count):
            x = r + col * 2 * r
            if row % 2 == 1:
                x += r
            y = r + row * np.sqrt(3) * r
            if idx < n:
                centers[idx] = [x, y]
                idx += 1
                
    if angle != 0.0:
        cx, cy = 0.5, 0.5
        cos_a = np.cos(angle)
        sin_a = np.sin(angle)
        for i in range(n):
            dx = centers[i, 0] - cx
            dy = centers[i, 1] - cy
            centers[i, 0] = dx * cos_a - dy * sin_a + cx
            centers[i, 1] = dx * sin_a + dy * cos_a + cy
            
    return centers, radii

def create_init_random(n, seed):
    """Random feasible initialization."""
    np.random.seed(seed)
    centers = np.random.uniform(0.15, 0.85, (n, 2))
    radii = np.full(n, 0.04)  # Small enough to guarantee initial feasibility
    return centers, radii

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    n = 26
    bounds = get_bounds(n)
    cons = {'type': 'ineq', 'fun': constraint_func, 'args': (n,)}
    
    best_sum = -1.0
    best_x = None
    
    configs = []
    
    # 1. Hexagonal patterns with various rotations to break symmetry traps
    for angle in [0.0, 0.1, 0.2, -0.1, 0.3, 0.4]:
        c, r = create_init_hex(n, angle)
        x0 = np.zeros(3*n)
        for i in range(n):
            x0[3*i] = c[i, 0]
            x0[3*i+1] = c[i, 1]
            x0[3*i+2] = r[i]
        configs.append(x0)
        
    # 2. Random feasible starts for diversity
    for seed in range(15):
        c, r = create_init_random(n, seed)
        x0 = np.zeros(3*n)
        for i in range(n):
            x0[3*i] = c[i, 0]
            x0[3*i+1] = c[i, 1]
            x0[3*i+2] = r[i]
        configs.append(x0)
        
    # Optimize each configuration
    for x0 in configs:
        try:
            res = minimize(objective_func, x0, args=(n,), method='SLSQP', 
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 10000, 'ftol': 1e-13, 'disp': False})
            if not np.isnan(res.fun) and -res.fun > best_sum:
                best_sum = -res.fun
                best_x = res.x.copy()
        except Exception:
            continue
            
    # 3. Iterative refinement to escape shallow local optima
    if best_x is not None:
        current_x = best_x.copy()
        for _ in range(5):
            noise = np.random.normal(0, 1e-4, size=current_x.shape)
            x0 = current_x + noise
            for i in range(n):
                r = max(0.001, x0[3*i+2])
                x0[3*i+2] = r
                x0[3*i] = np.clip(x0[3*i], r, 1.0 - r)
                x0[3*i+1] = np.clip(x0[3*i+1], r, 1.0 - r)
                
            try:
                res = minimize(objective_func, x0, args=(n,), method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 10000, 'ftol': 1e-13, 'disp': False})
                if not np.isnan(res.fun) and -res.fun > best_sum:
                    best_sum = -res.fun
                    best_x = res.x.copy()
            except Exception:
                break

    # Extract and format results
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        centers[i, 0] = best_x[3*i]
        centers[i, 1] = best_x[3*i+1]
        radii[i] = best_x[3*i+2]
        
    # Final validation & numerical safety net
    for _ in range(20):
        valid = True
        for i in range(n):
            if radii[i] < 0:
                valid = False
                break
            if centers[i, 0] - radii[i] < -1e-10 or centers[i, 0] + radii[i] > 1.0 + 1e-10:
                valid = False; break
            if centers[i, 1] - radii[i] < -1e-10 or centers[i, 1] + radii[i] > 1.0 + 1e-10:
                valid = False; break
        if valid:
            for i in range(n):
                for j in range(i + 1, n):
                    d = np.sqrt((centers[i, 0] - centers[j, 0])**2 + (centers[i, 1] - centers[j, 1])**2)
                    if d < radii[i] + radii[j] - 1e-10:
                        valid = False
                        break
                if not valid:
                    break
        if valid:
            break
        radii *= 0.999
        
    return centers, radii, float(np.sum(radii))
