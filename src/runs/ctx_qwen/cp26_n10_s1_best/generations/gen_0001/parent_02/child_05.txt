import numpy as np
from scipy.optimize import minimize

def obj_func(vars, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(vars[2::3])

def con_func(vars, n):
    """Constraints: boundary and non-overlap (squared distance formulation)."""
    xs = vars[0::3]
    ys = vars[1::3]
    rs = vars[2::3]
    c = []
    
    # Boundary constraints
    for i in range(n):
        xi, yi, ri = xs[i], ys[i], rs[i]
        c.append(xi - ri)
        c.append(1.0 - xi - ri)
        c.append(yi - ri)
        c.append(1.0 - yi - ri)
        
    # Overlap constraints (squared)
    for i in range(n):
        xi, yi, ri = xs[i], ys[i], rs[i]
        for j in range(i + 1, n):
            dx = xi - xs[j]
            dy = yi - ys[j]
            dr = ri + rs[j]
            c.append(dx*dx + dy*dy - dr*dr)
            
    return np.array(c)

def get_init(n, seed, method):
    """Generate a feasible initial configuration."""
    np.random.seed(seed)
    vars = np.zeros(3 * n)
    
    if method == 'hex':
        r_init = 0.065
        vars[2::3] = r_init
        pts = []
        step_x = 2 * r_init
        step_y = np.sqrt(3) * r_init
        y = r_init
        row = 0
        while len(pts) < n:
            x = r_init if row % 2 == 0 else 2 * r_init
            while x <= 1.0 - r_init and len(pts) < n:
                pts.append((x, y))
                x += step_x
            y += step_y
            row += 1
            
        for i, (x, y) in enumerate(pts[:n]):
            vars[3*i] = x + np.random.uniform(-0.005, 0.005)
            vars[3*i+1] = y + np.random.uniform(-0.005, 0.005)
            
    elif method == 'grid':
        r_init = 0.055
        vars[2::3] = r_init
        pts = []
        step = 0.18
        for x in np.arange(0.1, 0.91, step):
            for y in np.arange(0.1, 0.91, step):
                pts.append((x, y))
        for i, (x, y) in enumerate(pts[:n]):
            vars[3*i] = x + np.random.uniform(-0.005, 0.005)
            vars[3*i+1] = y + np.random.uniform(-0.005, 0.005)
            
    else:
        vars[2::3] = np.random.uniform(0.04, 0.08, n)

            ri = vars[3*i+2]
            vars[3*i] = np.random.uniform(ri, 1.0 - ri)
            vars[3*i+1] = np.random.uniform(ri, 1.0 - ri)
            
    return vars

    for method, seed in configs:

        try:
            res = minimize(obj_func, v0, method='SLSQP', bounds=bounds, constraints=cons, args=(n,),

            curr_val = -res.fun
            if curr_val > best_val:
