import numpy as np
from scipy.optimize import minimize

def compute_objective_and_penalty(vars, n, mu, triu_idx):
    """
    Computes the negative sum of radii plus a penalty for boundary and overlap violations.
    vars: flattened array [x1, y1, r1, x2, y2, r2, ...]
    n: number of circles
    mu: penalty weight
    triu_idx: precomputed upper triangle indices for overlap pairs
    """
    c = vars.reshape(n, 3)
    x = c[:, 0]
    y = c[:, 1]
    r = c[:, 2]
    
    # Objective: maximize sum(r) => minimize -sum(r)
    obj = -np.sum(r)
    
    # Boundary penalties: circles must stay within [0, 1]
    # Penalizes r - x < 0, x + r - 1 > 0, r - y < 0, y + r - 1 > 0
    b_pen = np.sum(np.maximum(0.0, r - x)**2)
    b_pen += np.sum(np.maximum(0.0, x + r - 1.0)**2)
    b_pen += np.sum(np.maximum(0.0, r - y)**2)
    b_pen += np.sum(np.maximum(0.0, y + r - 1.0)**2)
    
    # Overlap penalties: distance between centers must be >= r_i + r_j
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    d = np.sqrt(dx**2 + dy**2 + 1e-12)
    r_sum = r[:, None] + r[None, :]
    overlap = np.maximum(0.0, r_sum - d)
    o_pen = np.sum(overlap[triu_idx]**2)
    
    return obj + mu * (b_pen + o_pen)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Pack 26 circles in a unit square to maximize the sum of radii.
    Uses a penalty-based continuation method with L-BFGS-B optimization.
    """
    n = 26
    triu_idx = np.triu_indices(n, k=1)
    
    best_sum = -np.inf
    best_centers = None
    best_radii = None
    
    # Bounds for x, y in [0, 1], r in [0, 0.5]
    bounds = [(0.0, 1.0) if k % 3 != 2 else (0.0, 0.5) for k in range(3 * n)]
    
    # Try multiple randomized hexagonal initial configurations
    for seed in range(5):
        np.random.seed(seed)
        r_init = 0.085
        
        # Generate a hexagonal lattice
        pts = []
        for i in range(-10, 10):
            for j in range(-10, 10):
                x = j * 2 * r_init + (i % 2) * r_init
                y = i * np.sqrt(3) * r_init
                pts.append([x, y])
        pts = np.array(pts)
        
        # Apply random rotation to break symmetry
        angle = np.random.uniform(-0.3, 0.3)
        c_x, c_y = 0.5, 0.5
        cos_a, sin_a = np.cos(angle), np.sin(angle)
        pts_rot = pts.copy()
        pts_rot -= [c_x, c_y]
        pts_rot = pts_rot @ np.array([[cos_a, -sin_a], [sin_a, cos_a]])
        pts_rot += [c_x, c_y]
        
        # Select the n points closest to the center of the square
        dists = np.linalg.norm(pts_rot - [0.5, 0.5], axis=1)
        idx = np.argsort(dists)[:n]
        centers_init = pts_rot[idx]
        
        # Ensure initial points are safely inside bounds
        centers_init = np.clip(centers_init, 0.05, 0.95)
        radii_init = np.full(n, r_init)
        
        # Flatten to optimization vector
        x0 = np.zeros(3 * n)
        x0[0::3] = centers_init[:, 0]
        x0[1::3] = centers_init[:, 1]
        x0[2::3] = radii_init
        
        current_vars = x0.copy()
        
        # Continuation: gradually increase penalty weight
        mus = [1.0, 10.0, 100.0, 1000.0, 10000.0, 100000.0]
        for mu in mus:
            try:
                res = minimize(
                    compute_objective_and_penalty, 
                    current_vars, 
                    args=(n, mu, triu_idx),
                    method='L-BFGS-B', 
                    bounds=bounds,
                    options={'maxiter': 3000, 'ftol': 1e-10, 'gtol': 1e-8}
                )
                current_vars = res.x.copy()
            except Exception:
                break
                
        # Extract results
        centers = current_vars.reshape(n, 3)[:, :2]
        radii = current_vars.reshape(n, 3)[:, 2]
        current_sum = np.sum(radii)
        
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers.copy()
            best_radii = radii.copy()
            
    return best_centers, best_radii, float(best_sum)
