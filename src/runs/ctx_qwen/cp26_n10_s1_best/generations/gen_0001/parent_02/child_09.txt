import numpy as np
from scipy.optimize import minimize

def objective_func(x, n):
    """Maximize sum of radii by minimizing negative sum."""
    r = x[2::3]
    return -np.sum(r)

def constraints_func(x, n, pair_i, pair_j):
    """Vectorized boundary and overlap constraints."""
    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]
    
    # Boundary constraints: 4 per circle
    b_cons = np.empty(4 * n)
    b_cons[0::4] = xs - rs          # x >= r
    b_cons[1::4] = 1.0 - xs - rs    # x + r <= 1
    b_cons[2::4] = ys - rs          # y >= r
    b_cons[3::4] = 1.0 - ys - rs    # y + r <= 1
    
    # Overlap constraints: dist >= r_i + r_j
    dx = xs[pair_i] - xs[pair_j]
    dy = ys[pair_i] - ys[pair_j]
    dist = np.sqrt(dx*dx + dy*dy)
    o_cons = dist - rs[pair_i] - rs[pair_j]
    
    return np.concatenate([b_cons, o_cons])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Pack 26 circles in a unit square to maximize the sum of radii.
    """
    n = 26
    pair_i, pair_j = np.triu_indices(n, k=1)
    
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * n
    
    cons = {
        'type': 'ineq',
        'fun': constraints_func,
        'args': (n, pair_i, pair_j)
    }
    
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Prepare initial configurations
    inits = []
    
    # 1. Hexagonal lattice initialization
    pts = []
    r_init = 0.095
    y = r_init
    row = 0
    while y <= 1.0 - r_init:
        x = r_init
        if row % 2 == 1:
            x += r_init
        while x <= 1.0 - r_init:
            pts.append((x, y))
            x += 2 * r_init
        y += np.sqrt(3) * r_init
        row += 1
        
    while len(pts) < n:
        pts.append((0.5 + 0.1 * np.random.rand(), 0.5 + 0.1 * np.random.rand()))
    inits.append(np.array(pts[:n]))
    
    # 2. Random initializations
    for s in range(5):
        np.random.seed(s * 1000 + 7)
        centers = np.random.uniform(0.15, 0.85, (n, 2))
        inits.append(centers)
        
    # 3. Grid initialization
    g = np.array(np.meshgrid(
        np.linspace(0.15, 0.85, 6), 
        np.linspace(0.15, 0.85, 5)
    )).T.reshape(-1, 2)
    inits.append(g[:n])

    # Optimize from each start
    for centers in inits:
        x0 = np.zeros(3 * n)
        x0[0::3] = centers[:, 0]
        x0[1::3] = centers[:, 1]
        x0[2::3] = 0.09  # Reasonable starting radius
        
        try:
            res = minimize(objective_func, x0, args=(n,), method='SLSQP', 
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 8000, 'ftol': 1e-13, 'disp': False})
            
            if not np.isnan(res.fun):
                sum_r = -res.fun
                if sum_r > best_sum:
                    best_sum = sum_r
                    best_centers = res.x[0::3].reshape(-1, 2)
                    best_radii = res.x[2::3]
        except Exception:
            pass

    # Refinement: perturb and re-optimize to escape shallow local minima

            x0 = np.zeros(3 * n)
            x0[0::3] = noisy[:, 0]
            x0[1::3] = noisy[:, 1]

                pass
