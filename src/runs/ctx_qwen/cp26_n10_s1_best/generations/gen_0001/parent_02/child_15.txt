import numpy as np
from scipy.optimize import minimize

def objective(v):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def objective_jac(v):
    """Gradient of the objective function."""
    jac = np.zeros_like(v)
    jac[2::3] = -1.0
    return jac

def constraints(v):
    """
    Inequality constraints: g(v) >= 0
    Includes boundary constraints and non-overlap constraints.
    """
    n = 26
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    
    # Total constraints: 4 boundary per circle + n*(n-1)/2 overlap pairs
    n_cons = 4 * n + n * (n - 1) // 2
    c = np.empty(n_cons)
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    c[:n] = x - r
    c[n:2*n] = 1.0 - x - r
    c[2*n:3*n] = y - r
    c[3*n:4*n] = 1.0 - y - r
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    idx = 4 * n
    for i in range(n):
        xi, yi, ri = x[i], y[i], r[i]
        for j in range(i + 1, n):
            dx = xi - x[j]
            dy = yi - y[j]
            dr = ri + r[j]
            c[idx] = dx * dx + dy * dy - dr * dr
            idx += 1
            
    return c

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * n
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -1.0
    best_v = None
    
    # Generate diverse initial configurations
    inits = []
    
    # 1. Hexagonal lattices with varying initial radii
    for r_init in [0.08, 0.09, 0.10, 0.11]:
        v = np.zeros(3 * n)
        count = 0
        row = 0
        while count < n:
            col = 0
            y = r_init + row * r_init * np.sqrt(3)
            if y > 1 - r_init:
                break
            while count < n:
                shift = r_init if row % 2 == 1 else 0
                x = r_init + shift + col * 2 * r_init
                if x > 1 - r_init:
                    break
                v[3 * count] = x
                v[3 * count + 1] = y
                v[3 * count + 2] = r_init
                count += 1
                col += 1
            row += 1
        # Fill remaining spots if grid didn't cover all 26
        while count < n:
            v[3 * count] = 0.5
            v[3 * count + 1] = 0.5
            v[3 * count + 2] = r_init * 0.5
            count += 1
        inits.append(v.copy())
        
    # 2. Perturbed hexagonal grids
    for seed in range(8):
        np.random.seed(seed)
        v = inits[0].copy()
        v += np.random.normal(0, 0.005, 3 * n)
        v[2::3] = np.clip(v[2::3], 0.01, 0.5)
        v[0::3] = np.clip(v[0::3], 0.01, 0.99)
        v[1::3] = np.clip(v[1::3], 0.01, 0.99)
        inits.append(v)
        
    # 3. Random initial configurations
    for seed in range(5):
        np.random.seed(seed + 1000)
        v = np.zeros(3 * n)
        for i in range(n):
            v[3 * i] = np.random.uniform(0.2, 0.8)
            v[3 * i + 1] = np.random.uniform(0.2, 0.8)
            v[3 * i + 2] = np.random.uniform(0.05, 0.12)
        inits.append(v)
        
    # Optimization loop
    for v0 in inits:
        try:
            res = minimize(objective, v0, method='SLSQP', jac=objective_jac,
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 6000, 'ftol': 1e-13})
            if not np.isnan(res.fun) and -res.fun > best_sum:
                best_sum = -res.fun
                best_v = res.x.copy()
        except Exception:
            pass
            
    # Refinement phase: perturb best solution and re-optimize
    if best_v is not None:
        for _ in range(5):
            v0 = best_v.copy()
            v0 += np.random.normal(0, 1e-4, 3 * n)
            v0[2::3] = np.clip(v0[2::3], 1e-8, 0.5)
            v0[0::3] = np.clip(v0[0::3], 0.0, 1.0)
            v0[1::3] = np.clip(v0[1::3], 0.0, 1.0)
            try:
                res = minimize(objective, v0, method='SLSQP', jac=objective_jac,
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 5000, 'ftol': 1e-14})
                if not np.isnan(res.fun) and -res.fun > best_sum:
                    best_sum = -res.fun
                    best_v = res.x.copy()
            except Exception:
                pass
                
    if best_v is None:
        best_v = inits[0]
        
    # Extract results
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        centers[i, 0] = best_v[3 * i]
        centers[i, 1] = best_v[3 * i + 1]
        radii[i] = best_v[3 * i + 2]
        
    # Post-processing: ensure strict feasibility within numerical tolerance
    radii = np.maximum(radii, 1e-9)
    centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
    centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)
    
    return centers, radii, float(np.sum(radii))
