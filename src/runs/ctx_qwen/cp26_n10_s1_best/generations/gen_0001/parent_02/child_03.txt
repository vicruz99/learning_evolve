import numpy as np
from scipy.optimize import minimize

def objective(x, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x, n):
    """Returns constraint values >= 0: boundaries and squared non-overlaps."""
    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]
    c = []
    
    # Boundary constraints
    c.extend(xs - rs)
    c.extend(1.0 - xs - rs)
    c.extend(ys - rs)
    c.extend(1.0 - ys - rs)
    
    # Overlap constraints (squared distance to avoid sqrt singularity)
    for i in range(n):
        for j in range(i + 1, n):
            dx = xs[i] - xs[j]
            dy = ys[i] - ys[j]
            dr = rs[i] + rs[j]
            c.append(dx*dx + dy*dy - dr*dr)
            
    return np.array(c)

def objective_wrapper(x):
    return objective(x, 26)

def constraints_wrapper(x):
    return constraints(x, 26)

def get_hex_init(n, seed, margin=0.1):
    """Generate a rotated, scaled hexagonal lattice initialization."""
    np.random.seed(seed)
    pts = []
    r_est = 0.1
    for row in range(12):
        for col in range(12):
            x = col * 2 * r_est
            y = row * np.sqrt(3) * r_est
            if row % 2 == 1:
                x += r_est
            pts.append([x, y])
    pts = np.array(pts)
    mask = (pts[:, 0] <= 1.0) & (pts[:, 1] <= 1.0)
    pts = pts[mask]
    
    if len(pts) < n:
        pts = np.random.uniform(margin, 1.0 - margin, (n, 2))
    else:
        dists = np.hypot(pts[:, 0] - 0.5, pts[:, 1] - 0.5)
        idx = np.argsort(dists)[:n]
        pts = pts[idx]
        
    # Random rotation to break symmetry
    angle = np.random.uniform(-15, 15) * np.pi / 180
    c = np.array([0.5, 0.5])
    rot_pts = np.zeros_like(pts)
    for i in range(len(pts)):
        dx, dy = pts[i, 0] - c[0], pts[i, 1] - c[1]
        rot_pts[i, 0] = dx * np.cos(angle) - dy * np.sin(angle) + c[0]
        rot_pts[i, 1] = dx * np.sin(angle) + dy * np.cos(angle) + c[1]
        
    xs, ys = rot_pts[:, 0], rot_pts[:, 1]
    min_x, max_x = np.min(xs), np.max(xs)
    min_y, max_y = np.min(ys), np.max(ys)
    w, h = max_x - min_x, max_y - min_y
    s = min((1.0 - 2*margin)/w, (1.0 - 2*margin)/h)
    rot_pts[:, 0] = (xs - min_x) * s + margin
    rot_pts[:, 1] = (ys - min_y) * s + margin
    
    centers = rot_pts
    radii = np.full(n, 0.08)
    return centers, radii

def resolve_overlaps(centers, radii, n):
    """Force-directed relaxation to resolve initial overlaps."""
    for _ in range(200):
        max_ov = 0
        for i in range(n):
            for j in range(i+1, n):
                dx = centers[j, 0] - centers[i, 0]
                dy = centers[j, 1] - centers[i, 1]
                d = np.hypot(dx, dy)
                r_sum = radii[i] + radii[j]
                if d < r_sum and d > 1e-8:
                    push = (r_sum - d) * 0.5
                    nx, ny = dx/d, dy/d
                    centers[i, 0] -= push * nx
                    centers[i, 1] -= push * ny
                    centers[j, 0] += push * nx
                    centers[j, 1] += push * ny
                    max_ov = max(max_ov, r_sum - d)
        if max_ov < 1e-6:
            break
    for i in range(n):
        r = radii[i]
        centers[i, 0] = np.clip(centers[i, 0], r, 1.0 - r)
        centers[i, 1] = np.clip(centers[i, 1], r, 1.0 - r)
    return centers, radii
