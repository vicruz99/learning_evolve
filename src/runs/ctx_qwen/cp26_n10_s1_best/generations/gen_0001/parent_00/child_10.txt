from scipy.optimize import minimize

def objective(vars):
    """Minimize negative sum of radii."""
    N = len(vars) // 3
    return -np.sum(vars[2::3])

def constraint(vars):
    """Vectorized constraints: boundary and non-overlap (all >= 0)."""
    N = len(vars) // 3
    xs = vars[0::3]
    ys = vars[1::3]
    rs = vars[2::3]
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    c = np.concatenate([
        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs
    ])
    
    # Pairwise non-overlap: dist(i,j) >= r_i + r_j
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dist = np.sqrt(dx**2 + dy**2 + 1e-16)
    min_dist = rs[:, None] + rs[None, :]

    c = np.concatenate([c, dist[i, j] - min_dist[i, j]])
    return c

def create_hex_init(N, seed):

def create_random_init(N, seed):

                    centers[i, 0] -= shift * nx
                    centers[i, 1] -= shift * ny
                    centers[j, 0] += shift * nx
                    centers[j, 1] += shift * ny
        if max_viol < 1e-12:
            break
