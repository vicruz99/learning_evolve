import numpy as np
from scipy.optimize import minimize

def objective(params):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(params[2::3])

def constraint_func(params):
    """
    Returns array of constraint values. All must be >= 0.
    Constraints:
    1. x - r >= 0
    2. 1 - x - r >= 0
    3. y - r >= 0
    4. 1 - y - r >= 0
    5. r >= 0
    6. (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0 (squared distance constraint)
    """
    n = len(params) // 3
    xs = params[0::3]
    ys = params[1::3]
    rs = params[2::3]
    
    c = list(xs - rs)
    c.extend(1.0 - xs - rs)
    c.extend(ys - rs)
    c.extend(1.0 - ys - rs)
    c.extend(rs)
    
    for i in range(n):
        for j in range(i + 1, n):
            dx = xs[i] - xs[j]
            dy = ys[i] - ys[j]
            dist_sq = dx*dx + dy*dy
            r_sum = rs[i] + rs[j]
            c.append(dist_sq - r_sum*r_sum)
            
    return np.array(c)

def bounds_func(n):
    """Returns bounds for variables: x, y in [0, 1], r in [0, 0.5]."""
    return [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * n

def force_expand(n, steps=1500, seed=0):
    """
    Force-directed expansion to find a dense initial packing.
    Circles grow uniformly and repel each other to jam into a dense configuration.
    """
    np.random.seed(seed)
    centers = np.random.uniform(0.2, 0.8, (n, 2))
    radii = np.full(n, 0.06)
    
    for step in range(steps):
        # Grow radii slowly
        radii *= 1.0002
        
        forces = np.zeros((n, 2))
        
        # Pairwise repulsion for overlapping/close circles
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[j, 0] - centers[i, 0]
                dy = centers[j, 1] - centers[i, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii[i] + radii[j] and dist > 1e-10:
                    overlap = radii[i] + radii[j] - dist
                    # Spring-like repulsive force
                    f = overlap * 20.0 / dist
                    fx, fy = f * dx, f * dy
                    forces[i, 0] -= fx
                    forces[i, 1] -= fy
                    forces[j, 0] += fx
                    forces[j, 1] += fy
                    
        # Boundary repulsion
        for i in range(n):
            for d in range(2):
                if centers[i, d] < radii[i]:
                    forces[i, d] += 20.0 * (radii[i] - centers[i, d])
                elif centers[i, d] > 1.0 - radii[i]:
                    forces[i, d] -= 20.0 * (centers[i, d] - (1.0 - radii[i]))
                    
        # Apply forces with adaptive step size
        step_size = 0.005 / (1.0 + step * 0.001)
        centers += forces * step_size
        np.clip(centers, 0.0, 1.0, out=centers)
        
    return centers, radii

def run_packing():

        params[1::3] = c[:, 1]
        params[2::3] = r
