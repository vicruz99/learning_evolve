import numpy as np
from scipy.optimize import minimize

N = 26

def objective(params):
    """Objective: minimize negative sum of radii."""
    r = params[2::3]
    return -np.sum(r)

def compute_constraints(params):
    """Vectorized constraint evaluation: returns array where all values must be >= 0."""
    x = params[0::3]
    y = params[1::3]
    r = params[2::3]
    
    # Boundary constraints: x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
    c = np.empty(4 * N)
    c[:N] = x - r
    c[N:2*N] = 1.0 - x - r
    c[2*N:3*N] = y - r
    c[3*N:4*N] = 1.0 - y - r
    
    # Overlap constraints: dist(i,j) - (r_i + r_j) >= 0
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist = np.sqrt(dx**2 + dy**2)
    r_sum = r[:, None] + r[None, :]
    
    idx = np.tril_indices(N, -1)
    overlaps = dist[idx] - r_sum[idx]
    
    return np.concatenate([c, overlaps])

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [0,0.5]."""
    bounds = []
    for _ in range(N):
        bounds.append((0.0, 1.0))
        bounds.append((0.0, 1.0))
        bounds.append((0.0, 0.5))
    return bounds

def generate_hex_init(angle_deg):
    """Generate a rotated hexagonal lattice initialization."""
    angle = np.deg2rad(angle_deg)
    pts = []
    for row in range(8):
        for col in range(8):
            x = 0.1 + col * 0.2
            y = 0.1 + row * 0.1732  # approx sqrt(3)/2 * 0.2
            if row % 2 == 1:
                x += 0.1
            pts.append([x, y])
            
    pts = np.array(pts[:N])
    pts -= 0.5
    c, s = np.cos(angle), np.sin(angle)
    rotated = pts @ np.array([[c, -s], [s, c]])
    
    max_ext = np.max(np.abs(rotated))
    if max_ext > 0:
        rotated *= 0.4 / max_ext
    rotated += 0.5
    
    centers = np.clip(rotated, 0.15, 0.85)
    radii = np.full(N, 0.09)
    
    params = np.zeros(3 * N)
    params[0::3] = centers[:, 0]
    params[1::3] = centers[:, 1]
    params[2::3] = radii
    return params

def resolve_overlaps(params):
    """Force-directed relaxation to resolve overlaps and push to boundaries."""
    step_size = 0.01
    x = params[0::3].copy()
    y = params[1::3].copy()
    r = params[2::3]
    
    for _ in range(300):
        max_ov = 0.0
        fx, fy = np.zeros(N), np.zeros(N)
        
        for i in range(N):
            for j in range(i + 1, N):
                dx = x[j] - x[i]
                dy = y[j] - y[i]
                d = np.sqrt(dx*dx + dy*dy)
                if d < r[i] + r[j] and d > 1e-8:
                    ov = r[i] + r[j] - d
                    if ov > max_ov:
                        max_ov = ov
                    f = ov / d
                    fx[i] -= f * dx
                    fy[i] -= f * dy
                    fx[j] += f * dx
                    fy[j] += f * dy
                    
        if max_ov < 1e-6:
            break
            
        x += fx * step_size
        y += fy * step_size
        x = np.clip(x, r, 1 - r)
        y = np.clip(y, r, 1 - r)
        
    params[0::3] = x
    params[1::3] = y
    return params

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in unit square to maximize sum of radii."""
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': compute_constraints}
    
    best_params = None
    best_val = -np.inf
    
    # 1. Try multiple hexagonal rotations
    angles = [0, 5, 10, 15, 20, 30, 45]
    for ang in angles:
        p = generate_hex_init(ang)
        p = resolve_overlaps(p)
        
        res = minimize(objective, p, method='SLSQP', bounds=bounds, 
                       constraints=cons, options={'maxiter': 5000, 'ftol': 1e-12})
        
        val = -res.fun
        if val > best_val:
            best_val = val
            best_params = res.x.copy()
            
    # 2. Try random starts
    np.random.seed(42)
    for seed in range(5):
        np.random.seed(seed + 100)
        c = np.random.uniform(0.2, 0.8, (N, 2))
        r = np.full(N, 0.07)
        p = np.zeros(3 * N)
        p[0::3] = c[:, 0]
        p[1::3] = c[:, 1]
        p[2::3] = r
        p = resolve_overlaps(p)
        
        res = minimize(objective, p, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': 5000, 'ftol': 1e-12})
        val = -res.fun
        if val > best_val:
            best_val = val
            best_params = res.x.copy()
            
    # 3. Final refinement with perturbation to escape shallow local minima
    if best_params is not None:
        np.random.seed(0)
        for _ in range(3):
            perturbed = best_params + np.random.normal(0, 1e-4, 3 * N)
            res = minimize(objective, perturbed, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 10000, 'ftol': 1e-14})
            val = -res.fun
            if val > best_val:
                best_val = val
                best_params = res.x.copy()
                
    centers = np.zeros((N, 2))
    radii = best_params[2::3]
    centers[:, 0] = best_params[0::3]
    centers[:, 1] = best_params[1::3]
    
    # Safety shrink to strictly satisfy validation tolerances if on boundary
    radii = np.maximum(radii, 0.0)
    
    return centers, radii, float(np.sum(radii))
