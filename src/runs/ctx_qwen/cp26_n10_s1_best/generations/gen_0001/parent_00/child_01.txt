import numpy as np
from scipy.optimize import minimize

def get_init_centers(n, method='hex', seed=0):
    np.random.seed(seed)
    centers = np.zeros((n, 2))
    if method == 'hex':
        pts = []
        dx, dy = 0.15, 0.15 * np.sqrt(3) / 2
        for i in range(10):
            for j in range(10):
                x = 0.05 + j*dx + (i%2)*(dx/2)
                y = 0.05 + i*dy
                if 0.1 <= x <= 0.9 and 0.1 <= y <= 0.9:
                    pts.append([x, y])
        pts = np.array(pts)
        if len(pts) < n:
            pts = np.random.uniform(0.2, 0.8, (n, 2))
        centers = pts[:n] + np.random.normal(0, 0.005, (n, 2))
    elif method == 'grid':
        xs = np.linspace(0.1, 0.9, 5)
        ys = np.linspace(0.1, 0.9, 5)
        grid = np.meshgrid(xs, ys)
        pts = np.vstack([grid[0].ravel(), grid[1].ravel()]).T
        pts = pts[:n] + np.random.normal(0, 0.01, (n, 2))
        centers = pts
    else:
        centers = np.random.uniform(0.15, 0.85, (n, 2))
        
    centers = np.clip(centers, 0.1, 0.9)
    return centers

def compute_safe_radii(centers, n):
    r = np.full(n, 0.5)
    for i in range(n):
        cx, cy = centers[i]
        r[i] = min(cx, 1.0-cx, cy, 1.0-cy)
        for j in range(n):
            if i == j:
                continue
            d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
            if d/2.0 < r[i]:
                r[i] = d/2.0
    return r

def objective(params, n):
    return -np.sum(params[2::3])

def constraint_func(params, n):
    n_cons = 4*n + n*(n-1)//2
    c = np.empty(n_cons)
    idx = 0
    for i in range(n):
        x, y, r = params[3*i], params[3*i+1], params[3*i+2]
        c[idx] = x - r; idx += 1
        c[idx] = 1.0 - x - r; idx += 1
        c[idx] = y - r; idx += 1
        c[idx] = 1.0 - y - r; idx += 1
        
    for i in range(n):
        for j in range(i+1, n):
            dx = params[3*i] - params[3*j]
            dy = params[3*i+1] - params[3*j+1]
            dr = params[3*i+2] + params[3*j+2]
            c[idx] = dx*dx + dy*dy - dr*dr
            idx += 1
    return c

def run_packing():
    n = 26
    bounds = []
    for _ in range(n):
        bounds.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])

    for seed in range(20):

            try:

                               constraints={'type': 'ineq', 'fun': constraint_func, 'args': (n,)},
                               options={'maxiter': 5000, 'ftol': 1e-12, 'disp': False})
                
                if res.success or -res.fun > best_sum:
                    centers_opt = res.x.reshape(n, 3)[:, :2]
                    radii_opt = res.x.reshape(n, 3)[:, 2].copy()

                    radii_opt *= 0.9999
