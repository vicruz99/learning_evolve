import numpy as np
from scipy.optimize import minimize

def objective(params, n):
    """Maximize sum of radii (return negative for minimization)."""
    return -np.sum(params[2::3])

def constraints(params, n):
    """All constraints >= 0: boundary, non-negative radius, non-overlap."""
    c_list = []
    
    # Boundary constraints
    for i in range(n):
        idx = 3 * i
        x, y, r = params[idx], params[idx+1], params[idx+2]
        c_list.append(x - r)            # x >= r
        c_list.append(1.0 - x - r)      # x + r <= 1
        c_list.append(y - r)            # y >= r
        c_list.append(1.0 - y - r)      # y + r <= 1
        c_list.append(r)                # r >= 0

    # Non-overlap constraints using squared distance for smoothness
    # (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
    for i in range(n):
        idx_i = 3 * i
        xi, yi, ri = params[idx_i], params[idx_i+1], params[idx_i+2]
        for j in range(i + 1, n):
            idx_j = 3 * j
            xj, yj, rj = params[idx_j], params[idx_j+1], params[idx_j+2]
            dx = xi - xj
            dy = yi - yj
            dr = ri + rj
            c_list.append(dx*dx + dy*dy - dr*dr)

    return np.array(c_list)

def resolve_overlaps(centers, radii, n):
    """Simple iterative repulsion to resolve initial overlaps."""
    for _ in range(300):
        max_viol = 0.0
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[j, 0] - centers[i, 0]
                dy = centers[j, 1] - centers[i, 1]
                dist = np.hypot(dx, dy)
                req = radii[i] + radii[j]
                if dist < req and dist > 1e-9:
                    shift = (req - dist) * 0.5
                    inv_dist = 1.0 / dist
                    centers[i, 0] -= shift * dx * inv_dist
                    centers[i, 1] -= shift * dy * inv_dist
                    centers[j, 0] += shift * dx * inv_dist
                    centers[j, 1] += shift * dy * inv_dist
                    max_viol = max(max_viol, req - dist)
        if max_viol < 1e-8:
            break

    # Clip to valid range
    for i in range(n):
        r = radii[i]
        centers[i, 0] = np.clip(centers[i, 0], r, 1.0 - r)
        centers[i, 1] = np.clip(centers[i, 1], r, 1.0 - r)
    return centers

def create_hex_init(n, seed):
    """Hexagonal lattice initialization."""
    np.random.seed(seed)
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.09)
    idx = 0
    row_h = 0.09 * np.sqrt(3)
    for r in range(8):
        for c in range(6):
            if idx >= n: break
            x = 0.12 + c * 0.18 + (r % 2) * 0.09
            y = 0.12 + r * row_h
            centers[idx] = [x, y]
            radii[idx] = 0.09 + np.random.uniform(-0.01, 0.01)
            idx += 1
    return centers, radii

def create_corner_init(n, seed):
    """Corner-anchored initialization with varying radii."""
    np.random.seed(seed)
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.08)
    
    # Place 4 larger circles near corners
    corner_r = 0.18
    corners = [[corner_r, corner_r], [1-corner_r, corner_r], 
               [corner_r, 1-corner_r], [1-corner_r, 1-corner_r]]
    for i, pos in enumerate(corners):
        centers[i] = pos
        radii[i] = corner_r
        
    # Fill rest in a staggered grid in the middle
    idx = 4
    step = 0.12
    for row in range(6):
        for col in range(5):
            if idx >= n: break
            x = 0.3 + col * step + (row % 2) * (step/2)
            y = 0.3 + row * step
            centers[idx] = [x, y]
            radii[idx] = 0.07 + np.random.uniform(-0.01, 0.01)
            idx += 1
    return centers, radii

def create_random_init(n, seed):
    """Random valid initialization."""
    np.random.seed(seed)
    centers = np.random.uniform(0.2, 0.8, (n, 2))
    radii = np.full(n, 0.06)
    return centers, radii

def run_packing():
    n = 26
    best_sum = -np.inf
    best_params = None
    
    # Generate diverse initial configurations
    init_configs = []
    for s in range(5):
        init_configs.append(create_hex_init(n, s))
    for s in range(5):
        init_configs.append(create_corner_init(n, s))
    for s in range(5):
        init_configs.append(create_random_init(n, s))
        
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * n

    for centers, radii in init_configs:
        centers, radii = resolve_overlaps(centers, radii, n)
        params = np.zeros(3 * n)
        for i in range(n):
            params[3*i] = centers[i, 0]
            params[3*i+1] = centers[i, 1]
            params[3*i+2] = radii[i]
            
        try:
            res = minimize(objective, params, args=(n,), method='SLSQP',
                          bounds=bounds,
                          constraints={'type': 'ineq', 'fun': constraints, 'args': (n,)},
                          options={'maxiter': 5000, 'ftol': 1e-13})
            
            if not np.isnan(res.fun) and -res.fun > best_sum:
                # Quick validity check
                c_opt = res.x.reshape(n, 3)
                valid = True
                for i in range(n):
                    if c_opt[i,2] < 0 or c_opt[i,0] < c_opt[i,2] - 1e-10 or c_opt[i,0]+c_opt[i,2] > 1+1e-10 or \
                       c_opt[i,1] < c_opt[i,2] - 1e-10 or c_opt[i,1]+c_opt[i,2] > 1+1e-10:
                        valid = False
                        break

                    best_sum = -res.fun
                    best_params = res.x.copy()
        except Exception:
            continue

                res = minimize(objective, noisy_params, args=(n,), method='SLSQP',
                              bounds=bounds,
                              constraints={'type': 'ineq', 'fun': constraints, 'args': (n,)},
                              options={'maxiter': 5000, 'ftol': 1e-13})
                
                if not np.isnan(res.fun) and -res.fun > current_sum:
                    current_sum = -res.fun
                    current_params = res.x.copy()
            except Exception:
                pass
        best_sum = current_sum
        best_params = current_params

    # Extract and finalize
    if best_params is None:

        radii = np.full(n, 0.05)
        idx = 0

                    radii[idx] = 0.1
                    idx += 1

            radii[idx] = 0.04
        return centers, radii, np.sum(radii)

    radii = np.zeros(n)
    for i in range(n):
        centers[i, 0] = best_params[3*i]
        centers[i, 1] = best_params[3*i+1]
        radii[i] = best_params[3*i+2]

    # Safety shrinkage to guarantee strict validity against checker
    if not validate_packing(centers, radii):
        factor = 0.999
        for _ in range(20):
            radii *= factor
            if validate_packing(centers, radii):
                break
    final_sum = np.sum(radii)
    return centers, radii, float(final_sum)

def validate_packing(centers, radii):
    """Validate packing constraints."""
    n = centers.shape[0]
    if np.isnan(centers).any() or np.isnan(radii).any():
        return False
    for i in range(n):
        if radii[i] < 0: return False
        x, y, r = centers[i, 0], centers[i, 1], radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            return False
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.hypot(dx, dy)
            if dist < radii[i] + radii[j] - 1e-12:
                return False
    return True
