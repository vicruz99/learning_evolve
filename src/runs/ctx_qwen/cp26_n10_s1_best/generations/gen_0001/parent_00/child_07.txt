import numpy as np
from scipy.optimize import minimize

N = 26

def objective(params):
    """Objective: minimize negative sum of radii."""
    return -np.sum(params[2::3])

def constraint_func(params):
    """Returns array of constraint values, all must be >= 0."""
    c = []
    # Boundary constraints for each circle
    for i in range(N):
        x = params[3*i]
        y = params[3*i+1]
        r = params[3*i+2]
        c.append(x - r)
        c.append(1.0 - x - r)
        c.append(y - r)
        c.append(1.0 - y - r)
        
    # Non-overlap constraints using squared distances for smoothness
    for i in range(N):
        for j in range(i + 1, N):
            xi, yi, ri = params[3*i], params[3*i+1], params[3*i+2]
            xj, yj, rj = params[3*j], params[3*j+1], params[3*j+2]
            dist_sq = (xi - xj)**2 + (yi - yj)**2
            min_dist_sq = (ri + rj)**2
            c.append(dist_sq - min_dist_sq)
            
    return np.array(c)

def resolve_overlaps(centers, radii, iters=150):
    """Pushes overlapping circles apart and clips to valid bounds."""
    for _ in range(iters):
        any_overlap = False
        for i in range(N):
            for j in range(i + 1, N):
                dx = centers[j, 0] - centers[i, 0]
                dy = centers[j, 1] - centers[i, 1]
                d = np.sqrt(dx*dx + dy*dy)
                if d < radii[i] + radii[j] and d > 1e-10:
                    overlap = radii[i] + radii[j] - d
                    shift = overlap * 0.5 / d
                    centers[i, 0] -= shift * dx
                    centers[i, 1] -= shift * dy
                    centers[j, 0] += shift * dx
                    centers[j, 1] += shift * dy
                    any_overlap = True
        if not any_overlap:
            break
            
    # Clip to ensure strict boundary satisfaction
    for i in range(N):
        r = radii[i]
        centers[i, 0] = max(r + 1e-9, min(1.0 - r - 1e-9, centers[i, 0]))
        centers[i, 1] = max(r + 1e-9, min(1.0 - r - 1e-9, centers[i, 1]))
    return centers, radii

def make_params(centers, radii):
    """Flattens centers and radii into optimization vector."""
    p = np.zeros(3*N)
    for i in range(N):
        p[3*i] = centers[i, 0]
        p[3*i+1] = centers[i, 1]
        p[3*i+2] = radii[i]
    return p

def init_hex(seed):
    """Hexagonal lattice initialization."""
    np.random.seed(seed)
    centers = np.zeros((N, 2))
    radii = np.full(N, 0.09)
    idx = 0
    row_counts = [6, 5, 6, 5, 4]
    for r_idx, count in enumerate(row_counts):
        y = 0.12 + r_idx * 0.16
        x_start = 0.12 + (0.0 if r_idx % 2 == 0 else 0.08)
        for c_idx in range(count):
            if idx >= N: break
            x = x_start + c_idx * 0.16
            centers[idx] = [x + np.random.uniform(-0.01, 0.01),
                            y + np.random.uniform(-0.01, 0.01)]
            idx += 1
    return centers, radii

def init_random(seed):
    """Random scatter initialization."""
    np.random.seed(seed)
    centers = np.random.uniform(0.15, 0.85, (N, 2))
    radii = np.full(N, 0.075)
    return centers, radii

def run_packing():
    """Pack 26 circles in unit square to maximize sum of radii."""
    bounds = [(0.0, 1.0)] * (2*N) + [(0.0001, 0.5)] * N
    
    best_sum = -1.0
    best_centers = None
    best_radii = None

    # Multi-start optimization
    for seed in range(40):
        if seed % 2 == 0:
            c, r = init_hex(seed)
        else:
            c, r = init_random(seed)
            
        c, r = resolve_overlaps(c, r)
        p0 = make_params(c, r)
        
        try:
            res = minimize(objective, p0, method='SLSQP', bounds=bounds,
                           constraints={'type': 'ineq', 'fun': constraint_func},
                           options={'maxiter': 8000, 'ftol': 1e-13})
            sum_r = -res.fun
            if sum_r > best_sum:
                best_sum = sum_r
                best_centers = np.zeros((N, 2))
                best_radii = np.zeros(N)

                    best_centers[i, 0] = res.x[3*i]
                    best_centers[i, 1] = res.x[3*i+1]
                    best_radii[i] = res.x[3*i+2]
        except Exception:
            continue

    if best_centers is None:

        idx = 0

            best_centers[idx] = [0.5, 0.5]
            best_radii[idx] = 0.02
        best_sum = np.sum(best_radii)

    return best_centers, best_radii, float(best_sum)
