import numpy as np
from scipy.optimize import minimize

N = 26

def objective(params):
    """Objective: maximize sum of radii -> minimize negative sum."""
    return -np.sum(params[2::3])

def objective_grad(params):
    """Gradient of the objective function."""
    grad = np.zeros_like(params)
    grad[2::3] = -1.0
    return grad

def constraint_func(params):
    """Constraint function: returns array of constraint values (must be >= 0)."""
    n = N
    num_constraints = 4 * n + n * (n - 1) // 2
    c = np.empty(num_constraints)
    idx = 0
    
    # Boundary constraints: x-r>=0, 1-x-r>=0, y-r>=0, 1-y-r>=0
    for i in range(n):
        base = 3 * i
        x, y, r = params[base], params[base + 1], params[base + 2]
        c[idx] = x - r; idx += 1
        c[idx] = 1.0 - x - r; idx += 1
        c[idx] = y - r; idx += 1
        c[idx] = 1.0 - y - r; idx += 1
        
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    for i in range(n):
        bi = 3 * i
        xi, yi, ri = params[bi], params[bi + 1], params[bi + 2]
        for j in range(i + 1, n):
            bj = 3 * j
            xj, yj, rj = params[bj], params[bj + 1], params[bj + 2]
            dx = xi - xj
            dy = yi - yj
            dr = ri + rj
            c[idx] = dx * dx + dy * dy - dr * dr
            idx += 1
    return c

def constraint_jac(params):
    """Jacobian of the constraint function."""
    n = N
    num_constraints = 4 * n + n * (n - 1) // 2
    jac = np.zeros((num_constraints, 3 * n))
    row = 0
    
    # Boundary constraint gradients
    for i in range(n):
        base = 3 * i
        # x - r >= 0
        jac[row, base] = 1.0; jac[row, base + 2] = -1.0; row += 1
        # 1 - x - r >= 0
        jac[row, base] = -1.0; jac[row, base + 2] = -1.0; row += 1
        # y - r >= 0
        jac[row, base + 1] = 1.0; jac[row, base + 2] = -1.0; row += 1
        # 1 - y - r >= 0
        jac[row, base + 1] = -1.0; jac[row, base + 2] = -1.0; row += 1
        
    # Overlap constraint gradients
    for i in range(n):
        bi = 3 * i
        xi, yi, ri = params[bi], params[bi + 1], params[bi + 2]
        for j in range(i + 1, n):
            bj = 3 * j
            xj, yj, rj = params[bj], params[bj + 1], params[bj + 2]
            dx = xi - xj
            dy = yi - yj
            dr = ri + rj
            
            # Gradients for circle i
            jac[row, bi] = 2 * dx
            jac[row, bi + 1] = 2 * dy
            jac[row, bi + 2] = -2 * dr
            
            # Gradients for circle j
            jac[row, bj] = -2 * dx
            jac[row, bj + 1] = -2 * dy
            jac[row, bj + 2] = -2 * dr
            row += 1
    return jac

def create_init_params(n, seed):
    """Generate initial parameters: hexagonal layout with perturbation."""
    np.random.seed(seed)
    params = np.zeros(3 * n)
    
    r_init = 0.06
    dx = 2 * r_init
    dy = np.sqrt(3) * r_init
    
    centers = []
    for row in range(8):
        for col in range(6):
            x = 0.2 + col * dx + (row % 2) * (dx / 2)
            y = 0.2 + row * dy
            if x <= 0.9 and y <= 0.9:
                centers.append((x, y))
                
    while len(centers) < n:
        centers.append((np.random.uniform(0.2, 0.8), np.random.uniform(0.2, 0.8)))
        
    centers = np.array(centers[:n])
    centers += np.random.uniform(-0.02, 0.02, centers.shape)
    centers = np.clip(centers, 0.15, 0.85)
    
    for i in range(n):
        params[3 * i] = centers[i, 0]
        params[3 * i + 1] = centers[i, 1]
        params[3 * i + 2] = r_init
        
    return params

def run_packing():
    """Pack 26 circles in unit square to maximize sum of radii."""
    n = N
    best_sum = -1.0
    best_params = None
    
    bounds_list = []
    for _ in range(n):
        bounds_list.append((0.0, 1.0)) # x
        bounds_list.append((0.0, 1.0)) # y
        bounds_list.append((1e-6, 0.5)) # r

    constraints = {'type': 'ineq', 'fun': constraint_func, 'jac': constraint_jac}
    
    # Multiple starts to escape local minima
    for seed in range(15):
        x0 = create_init_params(n, seed)
        try:
            res = minimize(objective, x0, method='SLSQP', jac=objective_grad,
                          bounds=bounds_list, constraints=constraints,
                          options={'maxiter': 8000, 'ftol': 1e-12, 'disp': False})
            if -res.fun > best_sum:
                best_sum = -res.fun
                best_params = res.x.copy()
        except Exception:
            pass
            
    if best_params is None:
        best_params = create_init_params(n, 0)
        best_sum = -objective(best_params)
        
    # Extract results
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        centers[i, 0] = best_params[3 * i]
        centers[i, 1] = best_params[3 * i + 1]
        radii[i] = best_params[3 * i + 2]
        
    # Post-processing to strictly satisfy constraints and avoid numerical edge cases
    for i in range(n):
        r = radii[i]
        centers[i, 0] = max(r + 1e-12, min(1.0 - r - 1e-12, centers[i, 0]))
        centers[i, 1] = max(r + 1e-12, min(1.0 - r - 1e-12, centers[i, 1]))
        
    return centers, radii, float(np.sum(radii))
