import numpy as np
from scipy.optimize import minimize

def objective(params, n):
    """Objective function: maximize sum of radii (minimize negative sum)."""
    return -sum(params[3 * i + 2] for i in range(n))

def constraint_fun(params, n):
    """
    Returns array of constraint values. All must be >= 0.
    Uses squared distances to avoid sqrt derivative issues at zero.
    """
    cons = []
    # Boundary constraints: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    for i in range(n):
        x, y, r = params[3 * i], params[3 * i + 1], params[3 * i + 2]
        cons.append(x - r)
        cons.append(1.0 - x - r)
        cons.append(y - r)
        cons.append(1.0 - y - r)
        
    # Non-overlap constraints: dist^2 >= (r_i + r_j)^2
    for i in range(n):
        for j in range(i + 1, n):
            dx = params[3 * i] - params[3 * j]
            dy = params[3 * i + 1] - params[3 * j + 1]
            dr = params[3 * i + 2] + params[3 * j + 2]
            cons.append(dx * dx + dy * dy - dr * dr)
            
    return np.array(cons)

def resolve_overlaps(centers, radii, n):
    """Iteratively push overlapping circles apart and fix boundary violations."""
    for _ in range(1000):
        max_violation = 0.0
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[j, 0] - centers[i, 0]
                dy = centers[j, 1] - centers[i, 1]
                dist_sq = dx * dx + dy * dy
                min_dist = radii[i] + radii[j]
                if dist_sq < min_dist * min_dist and dist_sq > 1e-12:
                    dist = np.sqrt(dist_sq)
                    overlap = min_dist - dist
                    max_violation = max(max_violation, overlap)
                    shift = overlap * 0.5
                    inv_dist = 0.5 / dist
                    centers[i, 0] -= shift * inv_dist * dx
                    centers[i, 1] -= shift * inv_dist * dy
                    centers[j, 0] += shift * inv_dist * dx
                    centers[j, 1] += shift * inv_dist * dy
                    
        # Boundary corrections
        for i in range(n):
            r = radii[i]
            centers[i, 0] = max(r + 1e-9, min(1.0 - r - 1e-9, centers[i, 0]))
            centers[i, 1] = max(r + 1e-9, min(1.0 - r - 1e-9, centers[i, 1]))
            
        if max_violation < 1e-10:
            break
    return centers, radii

def create_hex_init(n, seed):
    """Hexagonal lattice initialization."""
    np.random.seed(seed)
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.09)
    idx = 0
    r_est = 0.10
    y_step = r_est * np.sqrt(3)
    x_step = 2.0 * r_est
    
    row = 0
    while idx < n:
        col = 0
        y = 0.1 + row * y_step
        if row % 2 == 1:
            x_start = 0.1 + r_est
        else:
            x_start = 0.1
        while idx < n:
            x = x_start + col * x_step
            if x + r_est > 1.0:
                break
            centers[idx] = [x + np.random.uniform(-0.01, 0.01), 
                            y + np.random.uniform(-0.01, 0.01)]
            idx += 1
            col += 1
        row += 1
    return centers, radii

def create_grid_init(n, seed):
    """Perturbed grid initialization."""
    np.random.seed(seed)
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.085)
    idx = 0
    for r in range(6):
        for c in range(5):
            if idx >= n:
                break
            x = 0.15 + c * 0.175 + np.random.uniform(-0.015, 0.015)
            y = 0.15 + r * 0.175 + np.random.uniform(-0.015, 0.015)
            centers[idx] = [x, y]
            idx += 1
    return centers, radii

def create_random_init(n, seed):
    """Dense random initialization."""
    np.random.seed(seed)
    centers = np.random.uniform(0.15, 0.85, (n, 2))
    radii = np.full(n, 0.05)
    return centers, radii

def extract_result(params, n):
    """Extract centers and radii from flat parameter vector."""
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        centers[i, 0] = params[3 * i]
        centers[i, 1] = params[3 * i + 1]
        radii[i] = max(0.0, params[3 * i + 2])
    return centers, radii

def run_optimization(centers, radii, n):
    """Run SLSQP on a given initial configuration."""
    params = np.zeros(3 * n)
    for i in range(n):
        params[3 * i] = centers[i, 0]
        params[3 * i + 1] = centers[i, 1]
        params[3 * i + 2] = radii[i]
        
    bounds = []
    for _ in range(n):
        bounds.append((0.0, 1.0))
        bounds.append((0.0, 1.0))
        bounds.append((0.0, 0.5))
        
    try:
        res = minimize(objective, params, args=(n,), method='SLSQP',
                       constraints={'type': 'ineq', 'fun': constraint_fun, 'args': (n,)},
                       bounds=bounds,
                       options={'maxiter': 20000, 'ftol': 1e-14, 'disp': False})
        return res.x, -res.fun
    except Exception:
        return params, -sum(params[3*i+2] for i in range(n))

def run_packing():
