import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def objective(vars):
    """Maximize sum of radii (return negative for minimization)."""
    return -np.sum(vars[2::3])

def constraint_func(vars, n):
    """All constraints >= 0: boundary, non-negative radius, non-overlap."""
    cons = []
    
    # Boundary and radius constraints
    for i in range(n):
        x = vars[3 * i]
        y = vars[3 * i + 1]
        r = vars[3 * i + 2]
        
        # x >= r, 1 - x >= r, y >= r, 1 - y >= r
        cons.append(x - r)
        cons.append(1.0 - x - r)
        cons.append(y - r)
        cons.append(1.0 - y - r)
        # r >= 0
        cons.append(r)
        
    # Non-overlap constraints using squared distance for smoothness
    # dist^2 >= (r_i + r_j)^2  =>  dist^2 - (r_i + r_j)^2 >= 0
    for i in range(n):
        xi = vars[3 * i]
        yi = vars[3 * i + 1]
        ri = vars[3 * i + 2]
        for j in range(i + 1, n):
            xj = vars[3 * j]
            yj = vars[3 * j + 1]
            rj = vars[3 * j + 2]
            dx = xi - xj
            dy = yi - yj
            cons.append(dx * dx + dy * dy - (ri + rj) ** 2)
            
    return np.array(cons)

def generate_hex_init(n, seed):
    """Create hexagonal initial configuration with slight perturbation."""
    np.random.seed(seed)
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.09)
    idx = 0
    
    r0 = 0.09
    y_step = r0 * np.sqrt(3)
    x_step = 2 * r0
    
    row = 0
    col = 0
    while idx < n:
        x = r0 + col * x_step
        y = r0 + row * y_step
        if row % 2 == 1:
            x += r0
            
        x += np.random.uniform(-0.02, 0.02)
        y += np.random.uniform(-0.02, 0.02)
        
        if 0 <= x <= 1 and 0 <= y <= 1:
            centers[idx] = [x, y]
            radii[idx] = max(0.02, min(0.2, r0 + np.random.uniform(-0.01, 0.01)))
            idx += 1
            col += 1
        else:
            row += 1
            col = 0
            
    return centers, radii

def generate_grid_init(n, seed):
    """Create grid-like initial configuration."""
    np.random.seed(seed)
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.09)
    idx = 0
    
    for r in range(6):
        for c in range(5):
            if idx >= n: break
            x = 0.15 + c * 0.16 + np.random.uniform(-0.01, 0.01)
            y = 0.1 + r * 0.16 + np.random.uniform(-0.01, 0.01)
            centers[idx] = [x, y]
            radii[idx] = 0.09 + np.random.uniform(-0.005, 0.005)
            idx += 1
            
    return centers, radii

def generate_random_init(n, seed):
    """Create random initial configuration."""
    np.random.seed(seed)
    centers = np.random.uniform(0.2, 0.8, (n, 2))
    radii = np.random.uniform(0.06, 0.12, n)
    return centers, radii

def create_vars(centers, radii, n):
    """Convert centers and radii to flat optimization vector."""
    vars = np.zeros(3 * n)
    for i in range(n):
        vars[3 * i] = centers[i, 0]
        vars[3 * i + 1] = centers[i, 1]
        vars[3 * i + 2] = radii[i]
    return vars

def extract_vars(vars, n):
    """Convert flat optimization vector back to centers and radii."""
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        centers[i, 0] = vars[3 * i]
        centers[i, 1] = vars[3 * i + 1]
        radii[i] = vars[3 * i + 2]
    return centers, radii

def resolve_overlaps_fast(centers, radii, n):
    """Resolve overlaps by pushing circles apart iteratively."""
    for _ in range(300):
        changed = False
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[j, 0] - centers[i, 0]
                dy = centers[j, 1] - centers[i, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                req = radii[i] + radii[j]
                if dist < req and dist > 1e-9:
                    shift = (req - dist) * 0.5 + 1e-6
                    fx = dx / dist * shift
                    fy = dy / dist * shift
                    centers[i, 0] -= fx
                    centers[i, 1] -= fy
                    centers[j, 0] += fx
                    centers[j, 1] += fy
                    changed = True
        if not changed:
            break
            
    # Clip to valid range
    for i in range(n):
        r = radii[i]
        centers[i, 0] = np.clip(centers[i, 0], r + 1e-7, 1.0 - r - 1e-7)
        centers[i, 1] = np.clip(centers[i, 1], r + 1e-7, 1.0 - r - 1e-7)
    return centers, radii

def run_packing():
    """Pack 26 circles in unit square to maximize sum of radii."""
    n = N_CIRCLES
    best_sum = -1.0
    best_vars = None
    
    bounds = []
    for _ in range(n):
        bounds.append((0.0, 1.0))
        bounds.append((0.0, 1.0))
        bounds.append((0.0, 0.5))
        
    cons = {'type': 'ineq', 'fun': constraint_func, 'args': (n,)}
    
    # Generate diverse initial configurations
    inits = []
    for seed in range(12):
        c, r = generate_hex_init(n, seed)
        inits.append((c, r))
    for seed in range(8):
        c, r = generate_grid_init(n, seed)
        inits.append((c, r))
    for seed in range(15):
        c, r = generate_random_init(n, seed)
        inits.append((c, r))
        
    # Optimize each configuration
    for c, r in inits:
        # Ensure feasibility before SLSQP
        c, r = resolve_overlaps_fast(c.copy(), r.copy(), n)
        v0 = create_vars(c, r, n)
        
        try:
            res = minimize(objective, v0, method='SLSQP', bounds=bounds, 
                          constraints=cons, options={'maxiter': 8000, 'ftol': 1e-12, 'disp': False})
            
            current_sum = -res.fun
            if current_sum > best_sum:
                best_sum = current_sum
                best_vars = res.x.copy()
        except Exception:
            continue
            
    # Further refinement: use best result as starting point with noise
    if best_vars is not None:
        for _ in range(5):
            noisy = best_vars + np.random.normal(0, 1e-4, best_vars.shape)
            # Project back to bounds roughly to help SLSQP
            for i in range(n):
                r = noisy[3*i+2]
                noisy[3*i] = np.clip(noisy[3*i], r, 1-r)
                noisy[3*i+1] = np.clip(noisy[3*i+1], r, 1-r)
                noisy[3*i+2] = max(0.0, r)
                
            try:
                res = minimize(objective, noisy, method='SLSQP', bounds=bounds,
                              constraints=cons, options={'maxiter': 8000, 'ftol': 1e-12, 'disp': False})
                current_sum = -res.fun
                if current_sum > best_sum:
                    best_sum = current_sum
                    best_vars = res.x.copy()
            except Exception:
                break
                
    # Extract final result
    centers, radii = extract_vars(best_vars, n)
    
    # Final validation fix: shrink radii slightly if numerical errors cause violations
    factor = 1.0
    for _ in range(100):
        valid = True
        for i in range(n):
            x, y = centers[i, 0], centers[i, 1]
            r = radii[i] * factor
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                valid = False
                break
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii[i] * factor + radii[j] * factor - 1e-12:
                    valid = False
                    break
            if not valid: break
            
        if valid:
            radii *= factor
            break
        factor *= 0.999
        
    return centers, radii, float(np.sum(radii))
