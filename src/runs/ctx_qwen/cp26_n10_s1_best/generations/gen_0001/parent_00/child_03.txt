The previous solutions rely heavily on SLSQP with explicit constraints, which often gets trapped in local optima for high-dimensional non-convex packing problems. To break out of these traps and push closer to the theoretical limit (~2.636), I will switch to a **penalty-based optimization approach** combined with a **continuation method** and **multi-start strategy**.

Key improvements:
1. **Penalty Objective**: Instead of hard constraints, I'll use a smooth penalty function for boundary violations and circle overlaps. This allows the use of `L-BFGS-B`, which is generally faster and more robust than SLSQP for this type of problem. The penalty is squared to ensure smoothness.
2. **Continuation Method**: Start with a low penalty coefficient (`mu`) to quickly increase the sum of radii and find a feasible region, then gradually increase `mu` to strictly enforce non-overlap and boundary constraints. This prevents the optimizer from getting stuck early.
3. **Multi-Start & Perturbation**: Run optimization from diverse initial configurations (hexagonal, grid, random). After each `mu` stage, take the best solution, apply random perturbations, and continue optimization. This actively escapes local minima.
4. **Vectorized Computation**: Compute pairwise distances and penalties using NumPy broadcasting for efficiency.
5. **Strict Validation & Fallback**: After optimization, verify strict feasibility. If numerical errors cause tiny violations, uniformly shrink radii until valid. This guarantees the output passes the validation function.

This strategy is well-suited for circle packing problems and typically yields sums of radii very close to the global optimum.

import numpy as np
from scipy.optimize import minimize

def compute_penalty(params, n, mu):
    """
    Computes the objective value: -sum(radii) + mu * (boundary_penalty + overlap_penalty)
    Lower value is better. High mu enforces constraints strictly.
    """
    # Unpack parameters: [x0, y0, r0, x1, y1, r1, ...]
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        centers[i, 0] = params[3 * i]
        centers[i, 1] = params[3 * i + 1]
        radii[i] = params[3 * i + 2]
        
    # 1. Boundary Penalty: Circles must stay within [0, 1] x [0, 1]
    bp = 0.0
    bp += np.sum(np.maximum(0.0, radii - centers[:, 0])**2)          # x - r >= 0
    bp += np.sum(np.maximum(0.0, centers[:, 0] + radii - 1.0)**2)    # x + r <= 1
    bp += np.sum(np.maximum(0.0, radii - centers[:, 1])**2)          # y - r >= 0
    bp += np.sum(np.maximum(0.0, centers[:, 1] + radii - 1.0)**2)    # y + r <= 1
    
    # 2. Overlap Penalty: dist(i, j) >= r_i + r_j
    # Vectorized pairwise distance computation
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dist = np.sqrt(dx**2 + dy**2)
    
    r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    overlap = np.maximum(0.0, r_sum - dist)
    np.fill_diagonal(overlap, 0.0)  # Ignore self-distance
    op = np.sum(overlap**2)
    
    # Objective: maximize sum of radii -> minimize negative sum, plus penalties
    return -np.sum(radii) + mu * (bp + op)

def run_optimizer(init_params, n, mu, max_iter=10000):
    """Runs L-BFGS-B optimization with given parameters."""
    bounds = []
    for _ in range(n):
        bounds.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
        
    res = minimize(
        compute_penalty,
        init_params,
        args=(n, mu),
        method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-9}
    )
    return res.x

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Pack 26 circles in a unit square to maximize the sum of radii.
    Uses penalty-based L-BFGS-B with continuation and multi-start.
    """
    n = 26
    best_params = None
    best_sum = -1.0
    
    # Continuation schedule: gradually tighten constraints
    mus = [50.0, 300.0, 1500.0]
    
    for mu in mus:
        seeds = list(range(12))
        for seed in seeds:
            np.random.seed(seed)
            
            # Generate diverse initial configurations
            centers = np.zeros((n, 2))
            radii = np.ones(n) * 0.08
            
            if seed < 4:
                # Hexagonal lattice initialization
                r_hex = 0.09
                idx = 0
                for row in range(6):
                    for col in range(6):
                        if idx >= n:
                            break
                        x = 0.12 + col * 0.18 + (row % 2) * 0.09
                        y = 0.12 + row * 0.15
                        x += np.random.uniform(-0.02, 0.02)
                        y += np.random.uniform(-0.02, 0.02)
                        centers[idx] = [x, y]
                        radii[idx] = r_hex
                        idx += 1
            elif seed < 8:
                # Staggered grid initialization
                idx = 0
                for row in range(5):
                    for col in range(6):
                        if idx >= n:
                            break
                        x = 0.15 + col * 0.15
                        y = 0.15 + row * 0.17
                        x += np.random.uniform(-0.015, 0.015)
                        y += np.random.uniform(-0.015, 0.015)
                        centers[idx] = [x, y]
                        radii[idx] = 0.085
                        idx += 1
            else:
                # Random initialization
                centers = np.random.uniform(0.15, 0.85, (n, 2))
                radii = np.ones(n) * 0.07
                
            # Flatten to 1D parameter vector
            params = np.zeros(3 * n)
            for i in range(n):
                params[3*i] = centers[i, 0]
                params[3*i+1] = centers[i, 1]
                params[3*i+2] = radii[i]
                
            # Run optimization
            opt_params = run_optimizer(params, n, mu)
            current_sum = np.sum(opt_params[2::3])
            
            if current_sum > best_sum:
                best_sum = current_sum
                best_params = opt_params.copy()
                
        # Perturb best solution to escape local minima before increasing mu
        if best_params is not None:
            perturbed = best_params + np.random.normal(0, 0.004, 3*n)
            # Soft projection to keep within reasonable bounds
            for i in range(n):
                perturbed[3*i] = np.clip(perturbed[3*i], 0.02, 0.98)
                perturbed[3*i+1] = np.clip(perturbed[3*i+1], 0.02, 0.98)
                perturbed[3*i+2] = np.maximum(0.02, perturbed[3*i+2])
                
            opt_params = run_optimizer(perturbed, n, mu)
            current_sum = np.sum(opt_params[2::3])
            if current_sum > best_sum:
                best_sum = current_sum
                best_params = opt_params.copy()

    # Extract final centers and radii
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        centers[i, 0] = best_params[3*i]
        centers[i, 1] = best_params[3*i+1]
        radii[i] = max(0.0, best_params[3*i+2])
        
    # Strict validation check
    valid = True
    for i in range(n):
        if radii[i] < -1e-9:
            valid = False
        if centers[i, 0] - radii[i] < -1e-9 or centers[i, 0] + radii[i] > 1.0 + 1e-9:
            valid = False
        if centers[i, 1] - radii[i] < -1e-9 or centers[i, 1] + radii[i] > 1.0 + 1e-9:
            valid = False
            
    if valid:
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                d = np.sqrt(dx*dx + dy*dy)
                if d < radii[i] + radii[j] - 1e-9:
                    valid = False
                    break
            if not valid:
                break
                
    # Fallback: uniformly shrink radii if numerical errors caused violations
    if not valid:
        factor = 0.99
        for _ in range(100):
            test_r = radii * factor
            ok = True
            for i in range(n):
                if test_r[i] < 0: ok = False; break
                if centers[i,0] - test_r[i] < -1e-12 or centers[i,0] + test_r[i] > 1.0 + 1e-12:
                    ok = False; break
                if centers[i,1] - test_r[i] < -1e-12 or centers[i,1] + test_r[i] > 1.0 + 1e-12:
                    ok = False; break
            if ok:
                for i in range(n):
                    for j in range(i+1, n):
                        dx = centers[i,0] - centers[j,0]
                        dy = centers[i,1] - centers[j,1]
                        d = np.sqrt(dx*dx + dy*dy)
                        if d < test_r[i] + test_r[j] - 1e-12:
                            ok = False; break
                    if not ok: break
            if ok:
                radii = test_r
                break
            factor *= 0.999

    return centers, radii, float(np.sum(radii))
