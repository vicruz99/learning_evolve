import numpy as np
from scipy.optimize import minimize


def run_packing():
    """
    Pack 26 circles in unit square to maximize sum of radii.
    Uses squared distance constraints for smoother optimization,
    multiple hexagonal initializations, and perturbation refinement.
    """
    n = 26

    def objective(params):
        """Maximize sum of radii (return negative for minimization)."""
        return -np.sum(params[2::3])

    def constraints(params):
        """All constraints >= 0: boundary, non-negative radius, non-overlap (squared distance)."""
        c = []

        # Boundary constraints: x-r>=0, 1-x-r>=0, y-r>=0, 1-y-r>=0, r>=0
        for i in range(n):
            x = params[3 * i]
            y = params[3 * i + 1]
            r = params[3 * i + 2]
            c.append(x - r)
            c.append(1 - x - r)
            c.append(y - r)
            c.append(1 - y - r)
            c.append(r)

        # Non-overlap constraints using squared distance: dist^2 >= (r_i + r_j)^2
        for i in range(n):
            for j in range(i + 1, n):
                dx = params[3 * i] - params[3 * j]
                dy = params[3 * i + 1] - params[3 * j + 1]
                dr = params[3 * i + 2] + params[3 * j + 2]
                c.append(dx * dx + dy * dy - dr * dr)

        return np.array(c)

    # Variable bounds: x,y in [0,1], r in [0, 0.5]
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * n

    best_params = None
    best_val = -np.inf

    # Phase 1: Try many hexagonal initial configurations with different densities
    for seed in range(40):
        for base_r in np.linspace(0.08, 0.11, 7):
            params = np.zeros(3 * n)
            np.random.seed(seed * 300 + int(base_r * 1000))

            idx = 0
            y = base_r
            while idx < n:
                x = base_r
                # Calculate row index for hexagonal offset
                row_idx = int((y - base_r) / (base_r * np.sqrt(3) + 1e-12))
                if row_idx % 2 == 1:
                    x += base_r  # Offset odd rows
                while idx < n and x + base_r <= 1.0:
                    params[3 * idx] = np.clip(x + np.random.uniform(-0.015, 0.015), 0.01, 0.99)
                    params[3 * idx + 1] = np.clip(y + np.random.uniform(-0.015, 0.015), 0.01, 0.99)
                    params[3 * idx + 2] = base_r
                    idx += 1
                    x += 2 * base_r
                y += base_r * np.sqrt(3)

            if idx >= n:
                try:
                    res = minimize(objective, params, method='SLSQP', bounds=bounds,
                                  constraints={'type': 'ineq', 'fun': constraints},
                                  options={'maxiter': 20000, 'ftol': 1e-14})
                    val = -res.fun
                    if not np.isnan(val) and val > best_val:
                        best_val = val
                        best_params = res.x.copy()
                except Exception:
                    pass

    # Phase 2: Perturbation refinement from best solution
    if best_params is not None:
        for trial in range(200):
            np.random.seed(trial + 80000)
            params = best_params.copy()
            # Decreasing noise amplitude (simulated annealing style)
            noise_amp = 0.012 * max(0.05, 1.0 - trial / 400.0)
            params += np.random.normal(0, noise_amp, params.shape)

            # Ensure feasibility after perturbation
            for i in range(n):
                r = max(0.001, params[3 * i + 2])
                params[3 * i] = np.clip(params[3 * i], r, 1 - r)
                params[3 * i + 1] = np.clip(params[3 * i + 1], r, 1 - r)
                params[3 * i + 2] = r

            try:
                res = minimize(objective, params, method='SLSQP', bounds=bounds,
                              constraints={'type': 'ineq', 'fun': constraints},
                              options={'maxiter': 15000, 'ftol': 1e-14})
                val = -res.fun
                if not np.isnan(val) and val > best_val:
                    best_val = val
                    best_params = res.x.copy()
            except Exception:
                pass

    # Extract final results
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        centers[i] = [best_params[3 * i], best_params[3 * i + 1]]
        radii[i] = max(0.0, best_params[3 * i + 2])

    return centers, radii, float(np.sum(radii))
