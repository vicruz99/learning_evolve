import numpy as np
from scipy.optimize import minimize


def objective(params):
    """Objective function: minimize negative sum of radii."""
    # Radii are at indices 2, 5, 8, ...
    return -np.sum(params[2::3])


def constraints(params):
    """
    Returns array of constraint values >= 0.
    Constraints:
    1. x_i - r_i >= 0
    2. 1 - x_i - r_i >= 0
    3. y_i - r_i >= 0
    4. 1 - y_i - r_i >= 0
    5. dist(i,j) - r_i - r_j >= 0 for all i < j
    """
    n = 26
    xs = params[0::3]
    ys = params[1::3]
    rs = params[2::3]
    
    # Boundary constraints
    c = []
    c.extend(xs - rs)
    c.extend(1.0 - xs - rs)
    c.extend(ys - rs)
    c.extend(1.0 - ys - rs)
    
    # Overlap constraints (vectorized)
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dist = np.hypot(dx, dy)
    r_sum = rs[:, None] + rs[None, :]
    
    # Extract lower triangle to avoid double counting and self-distance
    i_idx, j_idx = np.tril_indices(n, -1)
    c.extend(dist[i_idx, j_idx] - r_sum[i_idx, j_idx])
    
    return np.array(c)


def create_guess(seed, angle_deg):
    """Generates a rotated hexagonal lattice initial configuration."""
    np.random.seed(seed)
    pts = []
    r_est = 0.1
    dx = 2 * r_est
    dy = np.sqrt(3) * r_est
    
    # Generate enough points to cover the square
    for row in range(10):
        for col in range(10):
            x = col * dx + (row % 2) * (dx / 2)
            y = row * dy
            pts.append([x, y])
    pts = np.array(pts)
    
    # Center and scale to fit inside unit square with margin
    cx, cy = np.mean(pts, axis=0)
    pts = pts - np.array([cx, cy])
    max_d = np.max(np.sqrt(pts[:, 0]**2 + pts[:, 1]**2))
    scale = 0.35 / max_d
    pts = pts * scale
    
    # Apply rotation
    if angle_deg != 0.0:
        ang = np.deg2rad(angle_deg)
        c_arr, s_arr = np.cos(ang), np.sin(ang)
        rot = np.array([[c_arr, -s_arr], [s_arr, c_arr]])
        pts = pts @ rot.T
        
    # Shift to [0,1]
    pts = pts + 0.5
    
    # Select 26 points closest to center for dense packing
    dists = np.sqrt((pts[:, 0] - 0.5)**2 + (pts[:, 1] - 0.5)**2)
    order = np.argsort(dists)[:26]
    centers = pts[order]
    
    # Add noise to break symmetry
    centers += np.random.uniform(-0.015, 0.015, size=centers.shape)
    centers = np.clip(centers, 0.05, 0.95)
    
    radii = np.full(26, 0.06)
    return centers, radii


def resolve_overlaps(centers, radii):
    """Iteratively pushes overlapping circles apart to ensure feasible start."""
    n = 26
    for _ in range(200):
        max_v = 0.0
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[j, 0] - centers[i, 0]
                dy = centers[j, 1] - centers[i, 1]
                d = np.hypot(dx, dy)
                if d < radii[i] + radii[j] and d > 1e-10:
                    overlap = radii[i] + radii[j] - d
                    max_v = max(max_v, overlap)
                    shift = overlap * 0.5
                    centers[i, 0] -= shift * dx / d
                    centers[i, 1] -= shift * dy / d
                    centers[j, 0] += shift * dx / d
                    centers[j, 1] += shift * dy / d
        if max_v < 1e-8:
            break
            
    # Clip to valid range
    for i in range(n):
        centers[i, 0] = max(radii[i], min(1.0 - radii[i], centers[i, 0]))
        centers[i, 1] = max(radii[i], min(1.0 - radii[i], centers[i, 1]))
        
    return centers, radii


def run_packing():
    """Pack 26 circles in unit square to maximize sum of radii."""
    n = 26
    best_sum = -1.0
    best_centers = None
    best_radii = None

    cons = {'type': 'ineq', 'fun': constraints}

    seeds = list(range(15))
    angles = [0.0, 15.0, 30.0, 45.0, 60.0]

            centers, radii = resolve_overlaps(centers, radii)
            
            # Flatten to optimization vector
            params = np.zeros(3 * n)
            params[0::3] = centers[:, 0]
            params[1::3] = centers[:, 1]
            params[2::3] = radii
            
            try:
                res = minimize(objective, params, method='SLSQP',
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 15000, 'ftol': 1e-12, 'disp': False})
                
                cur_sum = -res.fun
                if cur_sum > best_sum:
                    best_sum = cur_sum
                    best_centers = np.column_stack((res.x[0::3], res.x[1::3]))
                    best_radii = res.x[2::3].copy()
            except Exception:
                pass
                
    # Ensure final result is strictly valid (handle numerical precision)
    if best_centers is not None:
        for i in range(n):
            best_radii[i] = max(0.0, best_radii[i])
            best_centers[i, 0] = max(best_radii[i], min(1.0 - best_radii[i], best_centers[i, 0]))
            best_centers[i, 1] = max(best_radii[i], min(1.0 - best_radii[i], best_centers[i, 1]))

    if best_centers is None:
        best_centers = np.random.uniform(0.15, 0.85, (n, 2))
        best_radii = np.full(n, 0.01)
        best_sum = np.sum(best_radii)
        
    return best_centers, best_radii, float(best_sum)
