from scipy.optimize import minimize

def objective(vars):

def constraint_func(vars, n):
    """Returns a single array of all constraint values that must be >= 0."""
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    # Total constraints: 4*n boundary + n*(n-1)/2 overlap
    n_constraints = 4 * n + n * (n - 1) // 2
    cons = np.empty(n_constraints)
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    cons[:n] = x - r
    cons[n:2*n] = 1.0 - x - r
    cons[2*n:3*n] = y - r
    cons[3*n:4*n] = 1.0 - y - r
    
    # Non-overlap constraints: dist(i,j) >= r_i + r_j
    idx = 4 * n
    for i in range(n):

            dy = y[i] - y[j]
            cons[idx] = np.hypot(dx, dy) - r[i] - r[j]
            idx += 1
            
    return cons
