from scipy.optimize import minimize

N_CIRCLES = 26

def compute_objective(vars):
    """Objective function: minimize negative sum of radii."""
    # Radii are stored at indices 2, 5, 8, ... (step of 3 starting at 2)
    radii = vars[2::3]
    return -np.sum(radii)

def compute_constraints(vars):
    """Computes all inequality constraints g(vars) >= 0."""
    n = N_CIRCLES
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    # 1. Boundary Constraints: circles must stay inside [0, 1] x [0, 1]
    # x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
    bound_cons = np.concatenate([
        x - r,
        1.0 - x - r,
        y - r,
        1.0 - y - r
    ])
    
    # 2. Non-Overlap Constraints: dist(i, j) >= r_i + r_j
    # Using squared distance for smoothness: dist^2 - (r_i + r_j)^2 >= 0
    X = x[:, np.newaxis] - x[np.newaxis, :]
    Y = y[:, np.newaxis] - y[np.newaxis, :]
    R_Sum = r[:, np.newaxis] + r[np.newaxis, :]
    
    # Extract lower triangular indices to avoid duplicates and self-pairs
    i_idx, j_idx = np.tril_indices(n, -1)
    overlap_cons = X[i_idx, j_idx]**2 + Y[i_idx, j_idx]**2 - R_Sum[i_idx, j_idx]**2
    
    return np.concatenate([bound_cons, overlap_cons])
