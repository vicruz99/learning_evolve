import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def objective(params):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(params[2::3])

def constraints_func(params):
    """
    Computes all inequality constraints: g(x) >= 0.
    Uses vectorized operations for speed.
    """
    n = N_CIRCLES
    x = params[0::3]
    y = params[1::3]
    r = params[2::3]
    
    total_cons = 4 * n + n * (n - 1) // 2
    cons = np.empty(total_cons)
    idx = 0
    
    # Boundary constraints
    cons[idx:idx+n] = x - r; idx += n
    cons[idx:idx+n] = 1.0 - x - r; idx += n
    cons[idx:idx+n] = y - r; idx += n
    cons[idx:idx+n] = 1.0 - y - r; idx += n
    
    # Overlap constraints using squared distances
    # i < j indices
    i, j = np.triu_indices(n, 1)
    dx = x[i] - x[j]
    dy = y[i] - y[j]
    dr = r[i] + r[j]
    cons[idx:] = dx*dx + dy*dy - dr*dr
    
    return cons

def resolve_overlaps(centers, radii, n):
    """Iteratively pushes overlapping circles apart until feasible."""
    for _ in range(300):
        max_v = 0.0
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[j, 0] - centers[i, 0]
                dy = centers[j, 1] - centers[i, 1]
                d = np.hypot(dx, dy)
                if d < radii[i] + radii[j] and d > 1e-12:
                    ov = radii[i] + radii[j] - d
                    max_v = max(max_v, ov)
                    s = ov * 0.5
                    nx, ny = dx / d, dy / d
                    centers[i, 0] -= s * nx
                    centers[i, 1] -= s * ny
                    centers[j, 0] += s * nx
                    centers[j, 1] += s * ny
        if max_v < 1e-8:
            break
            
    # Clip to valid bounds
    for i in range(n):
        centers[i, 0] = np.clip(centers[i, 0], radii[i], 1.0 - radii[i])
        centers[i, 1] = np.clip(centers[i, 1], radii[i], 1.0 - radii[i])
    return centers, radii

                col = 0
                continue
            c[idx] = [x + np.random.uniform(-0.015, 0.015), 
                      y + np.random.uniform(-0.015, 0.015)]

            idx += 1
            col += 1

            idx += 1
        c, r = resolve_overlaps(c, r, n)
        p = np.empty(3 * n)
        p[0::3] = c[:, 0]
        p[1::3] = c[:, 1]
        p[2::3] = r
        init_configs.append(p)
