import numpy as np
from scipy.optimize import minimize

def objective(vars):
    """Objective function: maximize sum of radii -> minimize negative sum."""
    return -np.sum(vars[2::3])

def constraints(vars):
    """
    Computes all inequality constraints g(vars) >= 0.
    Layout of vars: [x1, y1, r1, x2, y2, r2, ..., xN, yN, rN]
    """
    n = len(vars) // 3
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    num_ineq = 4 * n + n * (n - 1) // 2
    cons = np.empty(num_ineq)
    idx = 0
    
    # Boundary constraints
    cons[idx:idx+n] = x - r              # x >= r
    idx += n
    cons[idx:idx+n] = 1.0 - x - r        # x + r <= 1
    idx += n
    cons[idx:idx+n] = y - r              # y >= r
    idx += n
    cons[idx:idx+n] = 1.0 - y - r        # y + r <= 1
    idx += n
    
    # Non-overlap constraints (vectorized)
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    d2 = dx**2 + dy**2
    r_sum = r[:, None] + r[None, :]
    
    i, j = np.triu_indices(n, k=1)
    cons[idx:] = d2[i, j] - r_sum[i, j]**2
    
    return cons

def hex_init(n, seed):
    """Generate initial configuration from a hexagonal lattice."""
    np.random.seed(seed)
    pts = []
    sp = 0.18
    for row in range(12):
        for col in range(12):
            x = col * sp * 2 + (row % 2) * sp
            y = row * sp * np.sqrt(3)
            if 0.02 <= x <= 0.98 and 0.02 <= y <= 0.98:
                pts.append([x, y])
    pts = np.array(pts)
    
    if len(pts) < n:
        while len(pts) < n:
            pts = np.vstack([pts, [np.random.uniform(0.1, 0.9, 2)]])
            
    dists = np.hypot(pts[:, 0] - 0.5, pts[:, 1] - 0.5)
    idx = np.argsort(dists)[:n]
    centers = pts[idx]
    
    x0 = np.zeros(3 * n)
    x0[0::3] = centers[:, 0]
    x0[1::3] = centers[:, 1]
    x0[2::3] = 0.05
    return x0

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * n
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -1.0
    best_vars = None
    
    def try_opt(x0):
        nonlocal best_sum, best_vars
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 10000, 'ftol': 1e-13, 'disp': False})
            if -res.fun > best_sum:
                r = res.x[2::3]
                if np.all(r > 1e-7):
                    x = res.x[0::3]
                    y = res.x[1::3]
                    # Fast feasibility check
                    if (np.all(x >= r - 1e-8) and np.all(x + r <= 1 + 1e-8) and 
                        np.all(y >= r - 1e-8) and np.all(y + r <= 1 + 1e-8)):
                        dx = x[:, None] - x[None, :]
                        dy = y[:, None] - y[None, :]
                        d2 = dx**2 + dy**2
                        r_s = r[:, None] + r[None, :]
                        i, j = np.triu_indices(n, k=1)
                        if np.all(d2[i, j] >= r_s[i, j]**2 - 1e-8):
                            best_sum = -res.fun
                            best_vars = res.x.copy()
        except Exception:
            pass

    # 1. Multiple starts from hexagonal lattice
    for s in range(15):
        try_opt(hex_init(n, s))
        
    # 2. Stochastic perturbation to escape local minima
    if best_vars is not None:
        for _ in range(60):
            x0 = best_vars.copy()
            x0[0::3] += np.random.normal(0, 0.004, n)
            x0[1::3] += np.random.normal(0, 0.004, n)
            x0[2::3] += np.random.normal(0, 0.001, n)
            
            # Project back to feasible boundary region
            for i in range(n):
                ri = x0[2 + 3*i]
                if ri < 1e-4: ri = 1e-4
                x0[3*i] = np.clip(x0[3*i], ri, 1.0 - ri)
                x0[3*i+1] = np.clip(x0[3*i+1], ri, 1.0 - ri)
                x0[3*i+2] = ri
                
            try_opt(x0)
            
    # Fallback
    if best_vars is None:
        best_vars = hex_init(n, 0)
        
    # Extract results
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        centers[i, 0] = best_vars[3*i]
        centers[i, 1] = best_vars[3*i+1]
        radii[i] = best_vars[3*i+2]
        
    # Strict validation and numerical repair
    for _ in range(10):
        valid = True
        for i in range(n):
            for j in range(i+1, n):
                d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
                if d < radii[i] + radii[j] - 1e-9:
                    valid = False
                    break
            if not valid: break
        if valid:
            break
        radii *= 0.995
        
    return centers, radii, float(np.sum(radii))
