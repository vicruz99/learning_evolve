import numpy as np
from scipy.optimize import minimize

def obj_func(v, n):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def constr_func(v, n):
    """Constraint function: returns array of values that must be >= 0."""
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    c = []
    
    # Boundary constraints: circles must stay within [0, 1] x [0, 1]
    c.extend(x - r)
    c.extend(1.0 - x - r)
    c.extend(y - r)
    c.extend(1.0 - y - r)
    
    # Overlap constraints: squared distance >= (r_i + r_j)^2
    for i in range(n):
        for j in range(i + 1, n):
            dx = x[i] - x[j]
            dy = y[i] - y[j]
            dr = r[i] + r[j]
            c.append(dx*dx + dy*dy - dr*dr)
            
    return np.array(c)

def generate_init(n, r_start, seed):
    """Generate initial configuration based on a hexagonal lattice."""
    np.random.seed(seed)
    pts = []
    step_x = 2.0 * r_start
    step_y = np.sqrt(3.0) * r_start
    
    # Generate hexagonal points covering the unit square
    for iy in range(int(1.0 / step_y) + 10):
        for ix in range(int(1.0 / step_x) + 10):
            x = r_start + ix * step_x
            y = r_start + iy * step_y
            if iy % 2 == 1:
                x += step_x / 2.0
            if 0.0 <= x <= 1.0 and 0.0 <= y <= 1.0:
                pts.append([x, y])
                
    pts = np.array(pts)
    if len(pts) < n:
        # Fallback to random placement if grid is too sparse
        pts = np.random.uniform(0.15, 0.85, (n, 2))
        
    # Select n points closest to the center to maximize initial density
    dists = np.sum((pts - 0.5)**2, axis=1)
    idx = np.argsort(dists)[:n]
    c = pts[idx]
    r = np.full(n, r_start)
    
    v = np.zeros(3 * n)
    v[0::3] = c[:, 0]
    v[1::3] = c[:, 1]
    v[2::3] = r
    
    # Small perturbation to break exact symmetry and allow optimization movement
    v += np.random.normal(0, 0.001, 3 * n)
    
    # Ensure initial feasibility for boundaries
    v[2::3] = np.clip(v[2::3], 0.01, 0.25)
    v[0::3] = np.clip(v[0::3], v[2::3], 1.0 - v[2::3])
    v[1::3] = np.clip(v[1::3], v[2::3], 1.0 - v[2::3])
    
    return v

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    n = 26
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * n
    cons = {'type': 'ineq', 'fun': constr_func, 'args': (n,)}
    
    best_val = -1.0
    best_v = None
    
    # Multiple restarts with different initial radii and seeds
    starts = [(0.09, 0), (0.09, 1), (0.10, 0), (0.10, 1), (0.11, 0), (0.11, 1)]
    
    for r_start, seed in starts:
        v0 = generate_init(n, r_start, seed)
        
        # Primary optimization
        res = minimize(obj_func, v0, args=(n,), method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 3000, 'ftol': 1e-14, 'disp': False})
        val = -res.fun
        if val > best_val:
            best_val = val
            best_v = res.x.copy()
            
        # Refinement step: perturb best solution and optimize again to escape local minima
        if best_v is not None:
            v0_ref = best_v + np.random.normal(0, 0.0005, 3 * n)
            v0_ref[2::3] = np.clip(v0_ref[2::3], 0.01, 0.5)
            v0_ref[0::3] = np.clip(v0_ref[0::3], 0.0, 1.0)
            v0_ref[1::3] = np.clip(v0_ref[1::3], 0.0, 1.0)
            
            res = minimize(obj_func, v0_ref, args=(n,), method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 2000, 'ftol': 1e-14, 'disp': False})
            val = -res.fun
            if val > best_val:
                best_val = val
                best_v = res.x.copy()

    # Extract results
    c = np.zeros((n, 2))
    r = np.zeros(n)
    c[:, 0] = best_v[0::3]
    c[:, 1] = best_v[1::3]
    r[:] = best_v[2::3]
    
    # Post-processing: ensure strict numerical validity
    # SLSQP may leave tiny violations due to floating point precision.
    # We shrink radii minimally only if necessary.
    for _ in range(10):
        violation = False
        for i in range(n):
            if r[i] < 0:
                r[i] = 0.0
            if c[i, 0] < r[i] - 1e-12 or c[i, 0] > 1.0 - r[i] + 1e-12 or \
               c[i, 1] < r[i] - 1e-12 or c[i, 1] > 1.0 - r[i] + 1e-12:
                violation = True
                break
        if not violation:
            for i in range(n):
                for j in range(i + 1, n):
                    dist = np.hypot(c[i, 0] - c[j, 0], c[i, 1] - c[j, 1])
                    if dist < r[i] + r[j] - 1e-12:
                        violation = True
                        break
                if violation:
                    break
                    
        if violation:
            r *= 0.9999
        else:
            break
            
    return c, r, float(np.sum(r))
