import numpy as np
from scipy.optimize import minimize

def compute_penalty(x_flat, n, pairs, radii):
    """Smooth penalty function for boundary and overlap violations."""
    centers = x_flat.reshape(n, 2)
    p = 0.0
    x = centers[:, 0]
    y = centers[:, 1]
    
    # Boundary penalties
    p += np.sum(np.maximum(0.0, radii - x)**2)
    p += np.sum(np.maximum(0.0, x + radii - 1.0)**2)
    p += np.sum(np.maximum(0.0, radii - y)**2)
    p += np.sum(np.maximum(0.0, y + radii - 1.0)**2)
    
    # Overlap penalties
    for i, j in pairs:
        dx = centers[i, 0] - centers[j, 0]
        dy = centers[i, 1] - centers[j, 1]
        d = np.hypot(dx, dy)
        min_d = radii[i] + radii[j]
        if d < min_d:
            p += (min_d - d)**2
    return p

def objective_slsqp(v):
    """Objective to minimize: negative sum of radii."""
    return -np.sum(v[2::3])

def constraints_slsqp(v, n, pairs):
    """Inequality constraints g(v) >= 0 for SLSQP."""
    xs = v[0::3]
    ys = v[1::3]
    rs = v[2::3]
    c = []
    
    # Boundary constraints
    c.extend(xs - rs)
    c.extend(1.0 - xs - rs)
    c.extend(ys - rs)
    c.extend(1.0 - ys - rs)
    
    # Non-overlap constraints
    for i, j in pairs:
        dx = xs[i] - xs[j]
        dy = ys[i] - ys[j]
        c.append(np.hypot(dx, dy) - rs[i] - rs[j])
        
    return np.array(c)
