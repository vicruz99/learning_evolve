from scipy.optimize import minimize

N_CIRCLES = 26

def compute_objective(v):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def compute_constraints(v):
    """Returns array of constraint values g(v) >= 0 for SLSQP."""
    cs = v.reshape(N_CIRCLES, 3)
    x, y, r = cs[:, 0], cs[:, 1], cs[:, 2]
    
    c_list = []
    # Boundary constraints: circles inside [0,1]x[0,1]
    c_list.append(x - r)
    c_list.append(1.0 - x - r)
    c_list.append(y - r)
    c_list.append(1.0 - y - r)

    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist = np.sqrt(dx*dx + dy*dy)
    r_sum = r[:, None] + r[None, :]

    c_list.append(dist[mask] - r_sum[mask])
    
    return np.concatenate(c_list)

    return True
