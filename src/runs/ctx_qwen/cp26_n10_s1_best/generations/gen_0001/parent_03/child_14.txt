import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def objective(vars):
    """Objective to minimize: negative sum of radii."""
    return -np.sum(vars[2::3])

def constraint_fn(vars):
    """
    Returns an array of constraint values >= 0.
    Includes boundary and pairwise non-overlap constraints.
    """
    n = N_CIRCLES
    res = []
    
    # Boundary constraints: circle must be inside [0, 1] x [0, 1]
    for i in range(n):
        xi = vars[3*i]
        yi = vars[3*i+1]
        ri = vars[3*i+2]
        res.append(xi - ri)          # x >= r
        res.append(1.0 - xi - ri)    # 1 - x >= r
        res.append(yi - ri)          # y >= r
        res.append(1.0 - yi - ri)    # 1 - y >= r
        
    # Overlap constraints: squared distance >= (r_i + r_j)^2
    for i in range(n):
        xi = vars[3*i]
        yi = vars[3*i+1]
        ri = vars[3*i+2]
        for j in range(i + 1, n):
            xj = vars[3*j]
            yj = vars[3*j+1]
            rj = vars[3*j+2]
            dx = xi - xj
            dy = yi - yj
            dr = ri + rj
            res.append(dx*dx + dy*dy - dr*dr)
            
    return np.array(res)

def generate_init(method, n, seed):
    """Generates an initial feasible configuration."""
    np.random.seed(seed)
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    if method == 'grid':
        idx = 0
        for r in range(5):
            for c in range(5):
                if idx < n:
                    centers[idx] = [0.1 + c*0.2 + np.random.uniform(-0.005, 0.005),
                                    0.1 + r*0.2 + np.random.uniform(-0.005, 0.005)]
                    radii[idx] = 0.08
                    idx += 1
        while idx < n:
            centers[idx] = [0.5 + np.random.uniform(-0.05, 0.05),
                            0.5 + np.random.uniform(-0.05, 0.05)]
            radii[idx] = 0.05
            idx += 1
            
    elif method == 'hex':
        idx = 0
        y = 0.08
        while idx < n:
            x = 0.08
            while idx < n:
                centers[idx] = [x + np.random.uniform(-0.005, 0.005),
                                y + np.random.uniform(-0.005, 0.005)]
                radii[idx] = 0.08
                idx += 1
                x += 0.17
                if x + 0.08 > 1.0:
                    break
            y += 0.148
            if y + 0.08 > 1.0:
                break
        while idx < n:
            centers[idx] = [np.random.uniform(0.1, 0.9), np.random.uniform(0.1, 0.9)]
            radii[idx] = 0.05
            idx += 1
            
    elif method == 'random':
        for i in range(n):
            centers[i] = np.random.uniform(0.15, 0.85, 2)
            radii[i] = 0.06
            
    elif method == 'border':
        idx = 0
        for i in range(n):
            side = i % 4
            t = (i // 4) / max(1, n//4 - 1)
            t = np.clip(t, 0.05, 0.95)
            if side == 0: centers[idx] = [t, 0.08]
            elif side == 1: centers[idx] = [0.92, t]
            elif side == 2: centers[idx] = [1-t, 0.08]
            else: centers[idx] = [0.08, t]
            radii[idx] = 0.08
            idx += 1
            
    # Flatten to optimization vector [x1, y1, r1, x2, y2, r2, ...]
    vars = np.zeros(3*n)
    vars[0::3] = centers[:, 0]
    vars[1::3] = centers[:, 1]
    vars[2::3] = radii
    return vars

def run_packing():
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    n = N_CIRCLES
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * n
    cons = {'type': 'ineq', 'fun': constraint_fn}
    
    best_sum = -1.0
    best_vars = None
    
    # Phase 1: Multiple restarts with different layouts and seeds
    methods = ['grid', 'hex', 'random', 'border', 'random', 'grid', 'hex', 'random']
    for idx, method in enumerate(methods):
        for seed in range(3):
            seed_val = idx * 100 + seed
            x0 = generate_init(method, n, seed_val)
            
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 5000, 'ftol': 1e-15, 'disp': False})
                if -res.fun > best_sum:
                    best_sum = -res.fun

                pass

                        valid = False
                        break
                if not valid:
                    break
        if valid:
            break
        radii *= factor
        
    return centers, radii, float(np.sum(radii))
