import numpy as np
from scipy.optimize import minimize

N = 26

def objective(vars, n):
    """Objective: maximize sum of radii -> minimize negative sum."""
    return -np.sum(vars[2::3])

def constraint_func(vars, n):
    """
    Constraint function for SLSQP.
    Returns a vector of constraint values >= 0.
    Uses squared distances to avoid sqrt singularities.
    """
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    c = []
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    c.extend(x - r)
    c.extend(1.0 - x - r)
    c.extend(y - r)
    c.extend(1.0 - y - r)
    
    # Non-overlap constraints: dist^2 >= (r_i + r_j)^2
    for i in range(n):
        for j in range(i + 1, n):
            dx = x[i] - x[j]
            dy = y[i] - y[j]
            dr = r[i] + r[j]
            c.append(dx*dx + dy*dy - dr*dr)
            
    return np.array(c)

def make_init(seed, n):
    """Generate a perturbed hexagonal lattice initialization."""
    np.random.seed(seed)
    r_init = 0.05
    pts = []
    
    # Generate dense hex grid points
    for row in range(15):
        for col in range(15):
            x = col * 2 * r_init + (row % 2) * r_init
            y = row * r_init * np.sqrt(3)
            # Add slight perturbation to break symmetry
            x += np.random.uniform(-0.01, 0.01)
            y += np.random.uniform(-0.01, 0.01)
            if 0.05 <= x <= 0.95 and 0.05 <= y <= 0.95:
                pts.append([x, y, r_init])
                
    # Fallback grid if hex grid didn't yield enough points
    if len(pts) < n:
        pts = []
        for i in range(10):
            for j in range(10):
                pts.append([0.1 + i*0.09 + np.random.uniform(-0.005, 0.005), 
                            0.1 + j*0.09 + np.random.uniform(-0.005, 0.005), 0.05])
                            
    arr = np.array(pts)
    # Pick n points closest to the center (0.5, 0.5) for compact density
    dists = np.sum((arr[:, :2] - 0.5)**2, axis=1)
    idx = np.argsort(dists)[:n]
    sel = arr[idx]
    
    # Flatten to [x0, y0, r0, x1, y1, r1, ...]
    return np.concatenate(sel.T).flatten()

def run_packing():
    n = N
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * n
    cons = {'type': 'ineq', 'fun': constraint_func, 'args': (n,)}
    
    best_sum = -1.0
    best_vars = None

    def update_best(res, current_sum):

        x = res.x[0::3]
        y = res.x[1::3]
        r = res.x[2::3]

        if np.any(r < -1e-7):
            return

           np.any(y < r - 1e-7) or np.any(y + r > 1 + 1e-7):
            return
            
        for ii in range(n):
            for jj in range(ii + 1, n):
                d2 = (x[ii] - x[jj])**2 + (y[ii] - y[jj])**2
                if d2 < (r[ii] + r[jj])**2 - 1e-7:
                    return
                    
        if current_sum > best_sum:
            best_sum = current_sum
            best_vars = res.x.copy()

    # Phase 1: Multiple diverse starts
    for i in range(25):
        x0 = make_init(i, n)
        try:
            res = minimize(objective, x0, args=(n,), method='SLSQP', bounds=bounds, 
                           constraints=cons, options={'maxiter': 5000, 'ftol': 1e-12, 'disp': False})
            update_best(res, -res.fun)
        except Exception:
            pass
            
    # Phase 2: Local refinement on best found configuration

            except Exception:

                        break

        radii *= 0.9999
        
    return centers, radii, float(np.sum(radii))
