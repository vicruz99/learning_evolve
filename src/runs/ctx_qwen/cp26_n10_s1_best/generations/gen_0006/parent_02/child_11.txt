import numpy as np
from scipy.optimize import minimize, linprog

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Returns all inequality constraints g(x) >= 0."""
    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]
    
    # Boundary constraints: 4*N
    c = np.concatenate([
        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs
    ])
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    c = np.concatenate([c, dx[TRIL_IDX]**2 + dy[TRIL_IDX]**2 - dr[TRIL_IDX]**2])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-7, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)])
    return b

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist)
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return np.full(n, 0.05)

def force_optimize_centers(centers, radii, steps=1500):
    """Physics-based simulation to push centers apart and increase clearance."""
    n = centers.shape[0]
    dt = 0.005
    for _ in range(steps):
        forces = np.zeros_like(centers)
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[j] - centers[i]
                d = np.hypot(dx[0], dx[1])
                # Repel if boundaries are too close
                if d < radii[i] + radii[j] + 0.015 and d > 1e-8:
                    rep = (radii[i] + radii[j] + 0.015 - d) * 30.0 / (d + 1e-8)
                    forces[i] -= rep * dx / d
                    forces[j] += rep * dx / d
                    
            # Boundary repulsion
            for dim in range(2):
                if centers[i, dim] < radii[i] + 0.015:
                    forces[i, dim] += 60.0 * (radii[i] + 0.015 - centers[i, dim])
                elif centers[i, dim] > 1.0 - radii[i] - 0.015:
                    forces[i, dim] -= 60.0 * (centers[i, dim] - (1.0 - radii[i] - 0.015))
                    
        # Clamp forces to prevent instability
        forces = np.clip(forces, -5.0, 5.0)
        centers += forces * dt
        centers = np.clip(centers, 0.001, 0.999)
        dt *= 0.9995 # Cooling schedule
    return centers

def generate_hex_init(angle, scale):
    """Generates a rotated and scaled hexagonal lattice initialization."""
    pts = []
    r = 0.095 * scale
    y = r
    row = 0
    while len(pts) < N + 10:
        x = r if row % 2 == 0 else 2.0 * r
        while x <= 1.0 - r + 0.02:
            pts.append([x, y])
            x += 2.0 * r
        y += np.sqrt(3.0) * r
        row += 1
        if y > 1.0 + r: break
        
    pts = np.array(pts[:N + 10])
    
    # Rotate around center
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
        
    # Filter valid points and pad if necessary
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.2, 0.8, (1, 2))])
    return pts[:N]

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Phase 1: Diverse starts with alternating LP & Force optimization
    for angle in np.linspace(-0.25, 0.25, 13):
        for scale in [0.92, 0.98, 1.04]:
            centers = generate_hex_init(angle, scale)
            radii = solve_lp_radii(centers)
            
            # Alternating refinement
            for _ in range(20):
                centers = force_optimize_centers(centers, radii, steps=800)
                new_radii = solve_lp_radii(centers)
                if np.sum(new_radii) > np.sum(radii) * 1.00005:
                    radii = new_radii
                else:
                    break
                    
            curr_sum = np.sum(radii)
            if curr_sum > best_sum:
                best_sum = curr_sum
                best_centers = centers.copy()
                best_radii = radii.copy()
                
    # Phase 2: SLSQP fine-tuning on the best found configuration
    if best_centers is not None:
        x0 = np.zeros(3 * N)
        x0[0::3] = best_centers[:, 0]
        x0[1::3] = best_centers[:, 1]
        x0[2::3] = best_radii
        
        # Project to strict bounds for optimizer stability
        for i in range(N):
            r = max(1e-7, x0[3*i+2])
            x0[3*i] = np.clip(x0[3*i], r, 1.0-r)
            x0[3*i+1] = np.clip(x0[3*i+1], r, 1.0-r)
            x0[3*i+2] = r
            
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=get_bounds(),
                           constraints={'type': 'ineq', 'fun': constraints},
                           options={'maxiter': 25000, 'ftol': 1e-14, 'disp': False})
            if not np.isnan(res.fun):
                curr = -res.fun
                if curr > best_sum:
                    best_sum = curr
                    best_centers = np.column_stack((res.x[0::3], res.x[1::3]))
                    best_radii = res.x[2::3]
        except Exception:
            pass
            
    # Phase 3: Final LP snap to guarantee maximal radii for the converged centers
    lp_radii = solve_lp_radii(best_centers)
    if lp_radii is not None and np.sum(lp_radii) > best_sum:
        best_radii = lp_radii
        best_sum = np.sum(lp_radii)
        
    # Phase 4: Strict validity adjustment against 1e-12 tolerance
    radii = best_radii.copy()
    centers = best_centers.copy()
    for _ in range(100):
        valid = True
        for i in range(N):
            if radii[i] < 0 or centers[i,0] < radii[i]-1e-9 or centers[i,0] > 1-radii[i]+1e-9 or \
               centers[i,1] < radii[i]-1e-9 or centers[i,1] > 1-radii[i]+1e-9:
                valid = False
                break
        if valid:
            for i in range(N):
                for j in range(i+1, N):
                    d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
                    if d < radii[i]+radii[j]-1e-9:
                        valid = False
                        break
                if not valid:
                    break
        if valid:
            break
            
        # Minimal shrinkage to recover strict feasibility
        radii *= 0.9999
        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0-radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0-radii)
        
    return centers, radii, float(np.sum(radii))
