To maximize the sum of radii for 26 circles in a unit square, we face a highly non-convex optimization problem with many local minima. Pure gradient-based methods (like SLSQP) often get trapped in suboptimal symmetric arrangements. 

Strategy:
1. **Decouple Centers and Radii**: Fix circle centers and solve a Linear Programming (LP) problem to find the exact maximum radii that satisfy all boundary and non-overlap constraints. LP is fast, exact, and handles the linear `r_i + r_j <= dist(i,j)` constraints efficiently.
2. **Simulated Annealing (SA) on Centers**: Use SA to explore the center space. At each step, perturb centers, solve LP for radii, and accept/reject based on the sum of radii and a decaying temperature. This escapes local minima effectively.
3. **SLSQP Polishing**: After SA converges to a high-quality configuration, run SLSQP to jointly optimize centers and radii. This handles the non-linear squared-distance constraints smoothly and fine-tunes boundary contacts.
4. **Diverse Initializations**: Start SA from multiple configurations: rotated/scaled hexagonal lattices (theoretical packing optimums) and force-directed layouts (push points to boundaries and spread them out).
5. **Strict Validation Repair**: Apply a minimal shrinkage loop to guarantee compliance with the `1e-12` tolerance in the validation function, correcting any floating-point drift from optimizers.

This hybrid approach leverages the global search capability of SA, the exactness of LP, and the precision of SLSQP, consistently outperforming single-method optimizers for dense circle packing.

import numpy as np
from scipy.optimize import linprog, minimize

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = lim
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dist
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Returns all inequality constraints g(x) >= 0."""
    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]
    
    # Boundary constraints: 4*N
    c = np.concatenate([
        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs
    ])
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    dist_sq = dx[TRIL_IDX]**2 + dy[TRIL_IDX]**2
    r_sum_sq = dr[TRIL_IDX]**2
    
    return np.concatenate([c, dist_sq - r_sum_sq])

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-6, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)])
    return b

def sa_and_polish(centers_init, sa_steps=350, temp_init=0.03):
    """Simulated Annealing on centers + LP for radii + SLSQP polish."""
    centers = centers_init.copy()
    best_c = centers.copy()
    best_r = solve_lp_radii(centers)
    if best_r is None:
        best_r = np.zeros(N)
    best_sum = np.sum(best_r)
    curr_sum = best_sum
    temp = temp_init
    
    for step in range(sa_steps):
        step_sz = 0.015 * (0.995**step) + 0.0005
        new_c = centers + np.random.normal(0, step_sz, centers.shape)
        new_c = np.clip(new_c, 0.005, 0.995)
        
        r_new = solve_lp_radii(new_c)
        if r_new is None:
            continue
            
        s_new = np.sum(r_new)
        
        # Acceptance criterion
        if s_new > curr_sum:
            accept = True
        else:
            prob = np.exp((s_new - curr_sum) / max(temp, 1e-6))
            accept = np.random.rand() < prob
            
        if accept:
            centers = new_c
            curr_sum = s_new
            if s_new > best_sum:
                best_sum = s_new
                best_c = centers.copy()
                best_r = r_new.copy()
        temp *= 0.996
        
    # SLSQP polish
    x0 = np.zeros(3 * N)
    x0[0::3] = best_c[:, 0]
    x0[1::3] = best_c[:, 1]
    x0[2::3] = best_r * 0.9995
    
    cons = {'type': 'ineq', 'fun': constraints}
    bounds = get_bounds()
    
    try:
        res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 10000, 'ftol': 1e-12, 'disp': False})
        if not np.isnan(res.fun):
            c_vals = constraints(res.x)
            if np.min(c_vals) >= -1e-7:
                best_c = np.column_stack((res.x[0::3], res.x[1::3]))
                best_r = res.x[2::3]
                best_sum = -res.fun
                
                # Final LP snap for current centers to guarantee optimality
                r_lp = solve_lp_radii(best_c)
                if r_lp is not None:
                    best_r = r_lp
                    best_sum = np.sum(best_r)
    except Exception:
        pass
        
    return best_c, best_r, best_sum

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    best_c, best_r, best_sum = None, None, -1.0
    
    # Generate diverse starts
    starts = []
    
    # Hexagonal lattices with rotation and scaling
    for s in range(10):
        r0 = 0.09 + 0.015 * np.random.uniform(-0.5, 0.5)
        c = np.zeros((N, 2))
        idx = 0
        y = r0
        row = 0
        while idx < N and y + r0 <= 1.0:
            sx = r0 if row % 2 == 0 else 2.0 * r0
            cx = sx
            while idx < N and cx + r0 <= 1.0:
                c[idx] = [cx, y]
                idx += 1
                cx += 2.0 * r0
            y += np.sqrt(3.0) * r0
            row += 1
        while idx < N:
            c[idx] = np.random.uniform(0.1, 0.9, 2)
            idx += 1
            
        ang = np.random.uniform(-0.4, 0.4)
        ca, sa_val = np.cos(ang), np.sin(ang)
        c = (c - 0.5) @ np.array([[ca, -sa_val], [sa_val, ca]]) + 0.5
        c += np.random.uniform(-0.005, 0.005, c.shape)
        c = np.clip(c, 0.02, 0.98)
        starts.append(c)
        
    # Force-directed random starts
    for s in range(5):
        c = np.random.uniform(0.2, 0.8, (N, 2))
        for _ in range(300):
            f = np.zeros_like(c)
            for i in range(N):
                for j in range(i+1, N):
                    diff = c[j] - c[i]
                    d = np.linalg.norm(diff)
                    if d < 0.25 and d > 1e-4:
                        ff = 0.01 / (d**2 + 0.001)
                        f[i] -= ff * diff
                        f[j] += ff * diff
                for dim in range(2):
                    if c[i, dim] < 0.1: f[i, dim] += 0.03
                    elif c[i, dim] > 0.9: f[i, dim] -= 0.03
            c += f * 0.05
            c = np.clip(c, 0.02, 0.98)
        starts.append(c)
        
    for c_init in starts:
        c_opt, r_opt, s_opt = sa_and_polish(c_init, sa_steps=350, temp_init=0.03)
        if s_opt > best_sum:
            best_c, best_r, best_sum = c_opt, r_opt, s_opt
            
    # Final validation and repair
    centers = best_c
    radii = best_r
    for _ in range(50):
        valid = True
        for i in range(N):
            if radii[i] < 0 or centers[i,0] < radii[i]-1e-9 or centers[i,0] > 1-radii[i]+1e-9 or \
               centers[i,1] < radii[i]-1e-9 or centers[i,1] > 1-radii[i]+1e-9:
                valid = False; break
        if valid:
            for i in range(N):
                for j in range(i+1, N):
                    if np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1]) < radii[i]+radii[j]-1e-9:
                        valid = False; break
                if not valid: break
        if valid: break
        radii *= 0.999
        centers[:,0] = np.clip(centers[:,0], radii, 1-radii)
        centers[:,1] = np.clip(centers[:,1], radii, 1-radii)
        
    return centers, radii, float(np.sum(radii))
