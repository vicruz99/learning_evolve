

<strategy>
To maximize the sum of radii for 26 circles in a unit square, we must escape local optima that trap standard gradient-based optimizers. The key insight is that for fixed center positions, the optimal radii can be computed exactly and rapidly via Linear Programming (LP). This decouples the problem: we can optimize centers using derivative-free methods that evaluate the LP objective, while using SLSQP for precise local refinement of both centers and radii jointly.

Strategy:
1. **Diverse Initialization**: Generate hexagonal lattices with various rotations, force-directed layouts, and corner-biased configurations. Each is immediately paired with its optimal LP radii.
2. **Joint SLSQP Refinement**: Run SLSQP to locally optimize centers and radii simultaneously, leveraging analytic constraint structures. This quickly converges to a local optimum.
3. **Simulated Annealing on Centers**: The LP-sum objective landscape is piecewise-linear and non-smooth. Standard gradients fail. We implement Simulated Annealing (SA) that perturbs center coordinates (including occasional pairwise swaps to change packing topology) and accepts moves based on the LP-computed sum of radii. This effectively jumps between combinatorial constraint faces.
4. **Alternating Cycle**: Run SA followed by SLSQP refinement multiple times. SA explores new topologies, SLSQP precisely tightens constraints and maximizes radii within the new topology.
5. **Strict Validation & Repair**: Final geometric checks with minimal radius shrinkage guarantee compliance with the `1e-12` tolerance validator.

This hybrid LP + SA + SLSQP pipeline is highly effective for circle packing problems where constraint activity changes discretely.

```python
import numpy as np
import math
from scipy.optimize import minimize, linprog

N = 26

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Returns all inequality constraints g(x) >= 0."""
    cx, cy, cr = x[0::3], x[1::3], x[2::3]
    
    # Boundary constraints
    c = np.concatenate([cx - cr, 1.0 - cx - cr, cy - cr, 1.0 - cy - cr])
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dr = cr[:, None] + cr[None, :]
    
    i_idx, j_idx = np.tril_indices(N, -1)
    c = np.concatenate([c, dx[i_idx, j_idx]**2 + dy[i_idx, j_idx]**2 - dr[i_idx, j_idx]**2])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-6, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)])
    return b

def solve_lp(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A = np.zeros((num_ineq, n))
    b = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A[idx, i] = 1.0
        b[idx] = lim
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = d
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return np.full(n, 0.01)

def generate_inits():
    """Generate diverse initial configurations."""
    inits = []
    
    # 1. Hexagonal lattices with varied rotations
    for ang in np.linspace(-0.4, 0.4, 9):
        r0 = 0.10
        pts = []
        y = r0
        row = 0
        while len(pts) < N + 5:
            x = r0 if row % 2 == 0 else 2.0 * r0
            while x <= 1.0 - r0:
                pts.append([x, y])
                x += 2.0 * r0
            y += np.sqrt(3.0) * r0
            row += 1
        pts = np.array(pts[:N])
        
        c, s = np.cos(ang), np.sin(ang)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
        pts += np.random.uniform(-0.005, 0.005, pts.shape)
        pts = np.clip(pts, 0.02, 0.98)
        inits.append(pts)
        
    # 2. Force-directed layouts
    for s in range(6):
        np.random.seed(s * 137)
        pts = np.random.uniform(0.15, 0.85, (N, 2))
        for _ in range(300):
            forces = np.zeros_like(pts)
            for i in range(N):
                for j in range(i + 1, N):
                    diff = pts[j] - pts[i]
                    dist = np.linalg.norm(diff)
                    if dist < 0.25 and dist > 1e-5:
                        f = 0.012 / (dist**2 + 0.001)
                        forces[i] -= f * diff
                        forces[j] += f * diff
            pts += forces * 0.05
            pts = np.clip(pts, 0.05, 0.95)
        inits.append(pts)
        
    return inits

def simulated_annealing(centers, radii, steps=1500, init_temp=0.004):
    """SA on center coordinates optimizing LP sum of radii."""
    T = init_temp
    for step in range(steps):
        T = init_temp * (0.996 ** step)
        
        centers_new = centers.copy()
        
        # Perturb 1 to 3 circles
        n_pert = np.random.choice([1, 2, 3], p=[0.5, 0.35, 0.15])
        idxs = np.random.choice(N, n_pert, replace=False)
        moves = np.random.normal(0, T, (n_pert, 2))
        centers_new[idxs] += moves
        
        # Occasional swap to change topology
        if np.random.rand() < 0.12:
            i, j = np.random.choice(N, 2, replace=False)
            centers_new[i], centers_new[j] = centers_new[j].copy(), centers_new[i].copy()
            
        centers_new = np.clip(centers_new, 0.01, 0.99)
        
        r_new = solve_lp(centers_new)
        s_new = np.sum(r_new)
        s_old = np.sum(radii)
        
        # Metropolis criterion
        if s_new > s_old or np.random.rand() < math.exp((s_new - s_old) / max(T, 1e-7)):
            centers = centers_new
            radii = r_new
            
    return centers, radii

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -1.0
    best_x = None
    
    # Phase 1: Multi-start SLSQP
    inits = generate_inits()
    
    for c_init in inits:
        r_init = solve_lp(c_init)
        x0 = np.zeros(3 * N)
        x0[0::3] = c_init[:, 0]
        x0[1::3] = c_init[:, 1]
        x0[2::3] = r_init
        
        # Project to strict bounds
        for i in range(N):
            r = max(1e-5, x0[3 * i + 2])
            x0[3 * i] = np.clip(x0[3 * i], r, 1.0 - r)
            x0[3 * i + 1] = np.clip(x0[3 * i + 1], r, 1.0 - r)
            x0[3 * i + 2] = r
            
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 10000, 'ftol': 1e-13, 'disp': False})
            
            if not np.isnan(res.fun):
                curr_sum = -res.fun
                c_vals = constraints(res.x)
                if np.min(c_vals) >= -1e-5 and curr_sum > best_sum:
                    best_sum = curr_sum
                    best_x = res.x.copy()
        except Exception:
            pass
            
    if best_x is None:
        centers = np.random.uniform(0.2, 0.8, (N, 2))
        radii = solve_lp(centers)
        return centers, radii, float(np.sum(radii))
        
    centers = best_x.reshape(N, 3)[:, :2]
    radii = solve_lp(centers)
    
    # Phase 2: SA + SLSQP Alternation to escape local minima
    for rnd in range(3):
        centers, radii = simulated_annealing(centers, radii, steps=1500, init_temp=0.0035)
        
        # Refine with SLSQP
        x0 = np.zeros(3 * N)
        x0[0::3] = centers[:, 0]
        x0[1::3] = centers[:, 1]
        x0[2::3] = radii
        
        for i in range(N):
            r = max(1e-5, x0[3 * i + 2])
            x0[3 * i] = np.clip(x0[3 * i], r, 1.0 - r)
            x0[3 * i + 1] = np.clip(x0[3 * i + 1], r, 1.0 - r)
            x0[3 * i + 2] = r
            
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 8000, 'ftol': 1e-13, 'disp': False})
            if not np.isnan(res.fun):
                curr_sum = -res.fun
                c_vals = constraints(res.x)
                if np.min(c_vals) >= -1e-5 and curr_sum > best_sum:
                    best_sum = curr_sum
                    best_x = res.x.copy()
        except Exception:
            pass
            
    # Final extraction
    centers = best_x.reshape(N, 3)[:, :2]
    radii = best_x[2::3]
    
    # Final LP snap to guarantee maximal radii for final centers
    radii = solve_lp(centers)
    
    # Strict validation and minimal repair
    for _ in range(100):
        valid = True
        for i in range(N):
            if radii[i] < 0 or centers[i, 0] < radii[i] - 1e-9 or centers[i, 0] > 1.0 - radii[i] + 1e-9 or \
               centers[i, 1] < radii[i] - 1e-9 or centers[i, 1] > 1.0 - radii[i] + 1e-9:
                valid = False
                break
        if valid:
            for i in range(N):
                for j in range(i + 1, N):
                    d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                    if d < radii[i] + radii[j] - 1e-9:
                        valid = False
                        break
                if not valid:
                    break
        if valid:
            break
            
        # Minimal shrinkage to recover strict feasibility
        radii *= 0.9995
        for i in range(N):
            centers[i, 0] = np.clip(centers[i, 0], radii[i], 1.0 - radii[i])
            centers[i, 1] = np.clip(centers[i, 1], radii[i], 1.0 - radii[i])
            
    return centers, radii, float(np.sum(radii))
```
