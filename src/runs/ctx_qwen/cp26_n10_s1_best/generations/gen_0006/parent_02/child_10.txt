N = 26

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Returns all inequality constraints g(x) >= 0."""
    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]
    
    # Boundary constraints
    c = np.concatenate([
        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs
    ])
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    i_idx, j_idx = np.tril_indices(N, -1)
    dist_sq = dx[i_idx, j_idx]**2 + dy[i_idx, j_idx]**2
    r_sum_sq = dr[i_idx, j_idx]**2
    
    c = np.concatenate([c, dist_sq - r_sum_sq])
    return c

def get_bounds():
    """Variable bounds."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
    return b

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    for i in range(n):
        x_val, y_val = centers[i]
        bound = min(x_val, 1.0 - x_val, y_val, 1.0 - y_val)
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, bound)
        idx += 1
        
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist)
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def project_to_feasible(centers, radii):
    """Project centers and radii to satisfy bounds."""
    centers = centers.copy()
    radii = radii.copy()
    for i in range(N):
        r = max(1e-7, radii[i])
        centers[i, 0] = np.clip(centers[i, 0], r, 1.0 - r)
        centers[i, 1] = np.clip(centers[i, 1], r, 1.0 - r)
        radii[i] = min(r, min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1]))
    return centers, radii

def force_directed_optimal(n, seed, steps=1500):
    """Advanced force-directed layout optimizing for packing density."""
    np.random.seed(seed)
    centers = np.random.uniform(0.1, 0.9, (n, 2))
    
    for step in range(steps):
        forces = np.zeros_like(centers)
        alpha = max(0.01, 1.0 - step / steps)
        
        for i in range(n):
            for j in range(i + 1, n):
                diff = centers[j] - centers[i]
                dist = np.linalg.norm(diff)
                if dist < 0.4 and dist > 1e-6:
                    # Repulsion: stronger at close range
                    f = 0.02 / (dist**2 + 0.001)
                    forces[i] -= f * diff / dist
                    forces[j] += f * diff / dist
            
            # Boundary attraction - push towards edges
            for dim in range(2):
                val = centers[i, dim]
                if val < 0.3:
                    forces[i, dim] += 0.002 * alpha
                elif val > 0.7:
                    forces[i, dim] -= 0.002 * alpha
                    
        step_size = 0.05 * (1.0 + 0.5 * np.exp(-step / 300.0))
        centers += forces * step_size
        centers = np.clip(centers, 0.01, 0.99)
        
    return centers

def make_hex_init(r0, angle=0.0, perturb=0.0):
    """Hexagonal lattice with optional rotation and perturbation."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        if y > 1.0 + r0:
            break
            
    pts = np.array(pts[:N + 10])
    
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
    
    mask = (pts[:, 0] >= 0.01) & (pts[:, 0] <= 0.99) & (pts[:, 1] >= 0.01) & (pts[:, 1] <= 0.99)
    pts = pts[mask]
    
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.15, 0.85, (1, 2))])
    pts = pts[:N]
    
    if perturb > 0:
        pts += np.random.uniform(-perturb, perturb, pts.shape)
    pts = np.clip(pts, 0.02, 0.98)
    return pts

def simulated_annealing_step(centers, radii, temperature, current_sum):
    """Try a move and accept based on simulated annealing criterion."""
    move_type = np.random.randint(0, 4)
    new_centers = centers.copy()
    new_radii = radii.copy()
    
    if move_type == 0:
        # Single circle perturbation
        i = np.random.randint(N)
        new_centers[i] += np.random.normal(0, temperature, 2)
    elif move_type == 1:
        # Pair swap with perturbation
        i, j = np.random.choice(N, 2, replace=False)
        mid = (new_centers[i] + new_centers[j]) / 2
        new_centers[i] = mid + np.random.normal(0, temperature * 0.5, 2)
        new_centers[j] = mid + np.random.normal(0, temperature * 0.5, 2)
    elif move_type == 2:
        # Cluster move - move a group of nearby circles
        i = np.random.randint(N)
        neighbors = [i]
        for j in range(N):
            if j != i and np.linalg.norm(centers[j] - centers[i]) < 0.3:
                neighbors.append(j)
                if len(neighbors) >= 4:
                    break
        shift = np.random.normal(0, temperature * 0.5, 2)
        for k in neighbors:
            new_centers[k] += shift
    else:
        # Random subset repositioning
        subset_size = np.random.randint(3, 8)
        subset = np.random.choice(N, subset_size, replace=False)
        new_centers[subset] += np.random.normal(0, temperature * 0.3, (subset_size, 2))
    
    new_centers = np.clip(new_centers, 0.01, 0.99)
    
    # Compute optimal radii for new centers
    r_new = solve_lp_radii(new_centers)
    if r_new is not None:
        new_sum = np.sum(r_new)
        # Accept with SA probability
        delta = new_sum - current_sum
        if delta > 0 or np.random.random() < np.exp(delta / max(temperature, 0.001)):
            return new_centers, r_new, new_sum
    
    return centers, radii, current_sum

def crossover_configs(c1, r1, c2, r2):
    """Crossover between two configurations."""
    new_centers = c1.copy()
    # Take random subset from second config
    subset_size = np.random.randint(N // 3, 2 * N // 3)
    subset = np.random.choice(N, subset_size, replace=False)
    new_centers[subset] = c2[subset]
    
    r_new = solve_lp_radii(new_centers)
    if r_new is not None:
        return new_centers, r_new
    return None, None

                for j in range(i + 1, N):
                    d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                    if d < radii[i] + radii[j] - 1e-11:
                        valid = False
                        break
                if not valid:
                    break
        if valid:
            break
        
        radii *= 0.9998
        for i in range(N):
            centers[i, 0] = np.clip(centers[i, 0], radii[i], 1.0 - radii[i])
            centers[i, 1] = np.clip(centers[i, 1], radii[i], 1.0 - radii[i])
    
    return centers, radii, float(np.sum(radii))
