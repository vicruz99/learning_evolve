1. **Diverse Topology Initialization**: Generate multiple starting configurations using rotated/scaled hexagonal lattices, force-directed point spreading (to maximize inter-circle distances), and corner-biased layouts. This ensures the optimizer explores different packing symmetries.
2. **LP-SLSQP Alternation**: For fixed centers, the optimal radii problem is a Linear Program. We use `scipy.optimize.linprog` to snap radii to their theoretical maximum for any given center configuration. This provides a highly feasible and tight starting point for the non-linear SLSQP optimizer.
3. **Deflation & Perturbation Loop**: After finding a good local optimum, we intentionally shrink all radii by ~2%, add Gaussian noise to centers, and re-optimize. 4. **Boundary Utilization Heuristics**: Force-directed layouts include boundary repulsion/attraction terms to push circles toward edges and corners, which is critical for high-density packings as boundary circles can often achieve larger effective radii or create more interior space.
5. **Strict Numerical Repair**: A final validation loop applies minimal shrinkage and clipping to guarantee strict compliance with the `1e-12` tolerance in the validator, preventing false failures due to floating-point drift.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Returns all inequality constraints g(x) >= 0."""
    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]
    
    # Boundary constraints: 4*N
    c = np.concatenate([
        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs
    ])
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    dist_sq = dx[TRIL_IDX]**2 + dy[TRIL_IDX]**2
    r_sum_sq = dr[TRIL_IDX]**2
    
    c = np.concatenate([c, dist_sq - r_sum_sq])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-7, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)])
    return b

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist)
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-8)
    except Exception:
        pass
    return np.full(n, 0.05)

def force_spread(centers, steps=600):
    """Force-directed relaxation to spread points evenly and push to boundaries."""
    pts = centers.copy()
    for step in range(steps):
        forces = np.zeros_like(pts)
        alpha = 1.0 - step / steps
        
        for i in range(N):
            # Pairwise repulsion
            for j in range(i + 1, N):
                diff = pts[j] - pts[i]
                dist = np.linalg.norm(diff)
                if dist < 0.25 and dist > 1e-5:
                    f = 0.015 / (dist**2 + 0.001)
                    forces[i] -= f * diff
                    forces[j] += f * diff
                    
            # Boundary attraction/repulsion to utilize edges
            for dim in range(2):
                val = pts[i, dim]
                if val < 0.05:
                    forces[i, dim] += 0.06
                elif val > 0.95:
                    forces[i, dim] -= 0.06
                else:
                    # Gentle push outward to edges
                    forces[i, dim] += 0.02 * (-1.0 if val < 0.5 else 1.0)
                    
        pts += forces * 0.04 * alpha
        pts = np.clip(pts, 0.02, 0.98)
    return pts

def make_init_hex(r0, angle=0.0):
    """Generate a rotated hexagonal lattice initialization."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 5:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        
    pts = np.array(pts[:N + 5])
    
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
        
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.15, 0.85, (1, 2))])
    return pts[:N]

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -1.0
    best_x = None
    
    # Phase 1: Diverse Initial Configurations
    configs = []
    
    # 1. Rotated hexagonal lattices with varying densities
    for r0 in [0.085, 0.09, 0.095, 0.10]:
        for ang in np.linspace(-0.4, 0.4, 11):
            configs.append(make_init_hex(r0, ang))
            
    # 2. Force-directed layouts from random starts
    for s in range(12):
        c = np.random.uniform(0.15, 0.85, (N, 2))
        configs.append(force_spread(c, steps=500))
        
    # 3. Explicit boundary/corner biased starts
    for s in range(8):
        c = np.zeros((N, 2))
        c[0] = [0.15, 0.15]
        c[1] = [0.85, 0.15]
        c[2] = [0.15, 0.85]
        c[3] = [0.85, 0.85]
        c[4:] = np.random.uniform(0.2, 0.8, (N - 4, 2))
        configs.append(force_spread(c, steps=400))

    # Phase 2: Multi-start Optimization with LP-SLSQP Alternation
    for c_init in configs:
        # LP to snap radii tightly to feasible maximum for these centers
        r_init = solve_lp_radii(c_init) * 0.96
        
        x0 = np.zeros(3 * N)
        x0[0::3] = c_init[:, 0]
        x0[1::3] = c_init[:, 1]
        x0[2::3] = r_init
        
        # Project to strict bounds for SLSQP stability
        for i in range(N):
            r = max(1e-6, x0[3 * i + 2])
            x0[3 * i] = np.clip(x0[3 * i], r, 1.0 - r)
            x0[3 * i + 1] = np.clip(x0[3 * i + 1], r, 1.0 - r)
            x0[3 * i + 2] = r
            
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 15000, 'ftol': 1e-13, 'disp': False})
            
            if not np.isnan(res.fun):
                curr_sum = -res.fun
                c_vals = constraints(res.x)
                if np.min(c_vals) >= -1e-6 and curr_sum > best_sum:
                    best_sum = curr_sum
                    best_x = res.x.copy()
        except Exception:
            continue

    if best_x is None:
        # Fallback
        best_x = np.zeros(3 * N)
        best_x[0::3] = np.random.uniform(0.1, 0.9, N)
        best_x[1::3] = np.random.uniform(0.1, 0.9, N)
        best_x[2::3] = 0.06
        best_sum = np.sum(best_x[2::3])

    # Phase 3: Deflation & Perturbation Refinement to Escape Local Minima
    for step in range(70):
        noise_scale = 0.0025 * (0.96 ** step)
        x0 = best_x + np.random.normal(0, noise_scale, 3 * N)
        
        # Deflate radii to create slack for topological rearrangement
        x0[2::3] *= 0.975
        
        # Re-project bounds
        for i in range(N):
            r = max(1e-6, x0[3 * i + 2])
            x0[3 * i] = np.clip(x0[3 * i], r, 1.0 - r)
            x0[3 * i + 1] = np.clip(x0[3 * i + 1], r, 1.0 - r)
            x0[3 * i + 2] = r
            
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 12000, 'ftol': 1e-13, 'disp': False})
            
            if not np.isnan(res.fun):
                curr_sum = -res.fun
                c_vals = constraints(res.x)
                if np.min(c_vals) >= -1e-6 and curr_sum > best_sum:
                    best_sum = curr_sum
                    best_x = res.x.copy()
                    
                # LP refinement: snap radii to optimal for current centers
                c_cur = best_x.reshape(N, 3)[:, :2]
                r_lp = solve_lp_radii(c_cur)
                if r_lp is not None:
                    best_x[2::3] = r_lp
                    lp_sum = np.sum(r_lp)
                    if lp_sum > best_sum:
                        best_sum = lp_sum
        except Exception:
            pass

    # Extract final results
    centers = best_x.reshape(N, 3)[:, :2]
    radii = best_x[2::3]
    
    # Final strict validity adjustment against 1e-12 tolerance
    for _ in range(100):
        valid = True
        for i in range(N):
            if radii[i] < 0:
                valid = False; break
            if centers[i, 0] - radii[i] < -1e-10 or centers[i, 0] + radii[i] > 1.0 + 1e-10:
                valid = False; break
            if centers[i, 1] - radii[i] < -1e-10 or centers[i, 1] + radii[i] > 1.0 + 1e-10:
                valid = False; break
        if valid:
            for i in range(N):
                for j in range(i + 1, N):
                    d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                    if d < radii[i] + radii[j] - 1e-10:
                        valid = False; break
                if not valid:
                    break
        if valid:
            break
            
        # Minimal shrinkage to recover strict feasibility
        radii *= 0.9995
        for i in range(N):
            centers[i, 0] = np.clip(centers[i, 0], radii[i], 1.0 - radii[i])
            centers[i, 1] = np.clip(centers[i, 1], radii[i], 1.0 - radii[i])
            
    return centers, radii, float(np.sum(radii))
