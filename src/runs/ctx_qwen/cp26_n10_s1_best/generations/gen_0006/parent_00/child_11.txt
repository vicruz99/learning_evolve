import numpy as np
from scipy.optimize import minimize, linprog

N_CIRCLES = 26
TRIL_IDX = np.tril_indices(N_CIRCLES, -1)
PAIR_COUNT = len(TRIL_IDX[0])
NUM_CONSTRAINTS = 4 * N_CIRCLES + PAIR_COUNT

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Returns all inequality constraints g(x) >= 0."""
    cx, cy, r = x[0::3], x[1::3], x[2::3]
    c = np.empty(NUM_CONSTRAINTS)
    c[:4*N_CIRCLES:4] = cx - r
    c[1:4*N_CIRCLES:4] = 1.0 - cx - r
    c[2:4*N_CIRCLES:4] = cy - r
    c[3:4*N_CIRCLES:4] = 1.0 - cy - r
    
    dx = cx[TRIL_IDX[0]] - cx[TRIL_IDX[1]]
    dy = cy[TRIL_IDX[0]] - cy[TRIL_IDX[1]]
    dr = r[TRIL_IDX[0]] + r[TRIL_IDX[1]]
    c[4*N_CIRCLES:] = dx**2 + dy**2 - dr**2
    return c

def jacobian(x):
    """Exact analytic Jacobian for constraints."""
    cx, cy, r = x[0::3], x[1::3], x[2::3]
    jac = np.zeros((NUM_CONSTRAINTS, 3*N_CIRCLES))
    
    # Boundary constraints Jacobian
    jac[:4*N_CIRCLES:4, 0::3] = 1.0
    jac[:4*N_CIRCLES:4, 2::3] = -1.0
    jac[1:4*N_CIRCLES:4, 0::3] = -1.0
    jac[1:4*N_CIRCLES:4, 2::3] = -1.0
    jac[2:4*N_CIRCLES:4, 1::3] = 1.0
    jac[2:4*N_CIRCLES:4, 2::3] = -1.0
    jac[3:4*N_CIRCLES:4, 1::3] = -1.0
    jac[3:4*N_CIRCLES:4, 2::3] = -1.0
    
    # Overlap constraints Jacobian
    dx = cx[TRIL_IDX[0]] - cx[TRIL_IDX[1]]
    dy = cy[TRIL_IDX[0]] - cy[TRIL_IDX[1]]
    dr = r[TRIL_IDX[0]] + r[TRIL_IDX[1]]
    base = 4*N_CIRCLES
    for k in range(PAIR_COUNT):
        i = TRIL_IDX[0][k]
        j = TRIL_IDX[1][k]
        idx = base + k
        jac[idx, 3*i] = 2.0*dx[k]
        jac[idx, 3*i+1] = 2.0*dy[k]
        jac[idx, 3*i+2] = -2.0*dr[k]
        jac[idx, 3*j] = -2.0*dx[k]
        jac[idx, 3*j+1] = -2.0*dy[k]
        jac[idx, 3*j+2] = -2.0*dr[k]
    return jac

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-7, 0.5]."""
    b = []
    for _ in range(N_CIRCLES):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)])
    return b

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N_CIRCLES
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], 
                  centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(1e-7, lim)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], 
                            centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(1e-7, dist)
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and np.all(res.x >= 0):
            return res.x
    except Exception:
        pass
    return None

def make_hex_init(r0, angle=0.0):
    """Generates a hexagonal lattice initialization with optional rotation."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N_CIRCLES + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        if y > 1.0 + r0: break
            
    pts = np.array(pts[:N_CIRCLES + 10])
    
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
        
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & \
           (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    
    while len(pts) < N_CIRCLES:
        pts = np.vstack([pts, np.random.uniform(0.15, 0.85, (1, 2))])
    return pts[:N_CIRCLES]

def make_force_init(seed, steps=300):
    """Force-directed layout to spread points evenly."""
    np.random.seed(seed)
    pts = np.random.uniform(0.15, 0.85, (N_CIRCLES, 2))
    for step in range(steps):
        forces = np.zeros_like(pts)
        for i in range(N_CIRCLES):
            for j in range(i + 1, N_CIRCLES):
                dx = pts[j] - pts[i]
                d = np.linalg.norm(dx)
                if d < 0.3 and d > 1e-5:
                    f = 0.01 / (d**2 + 0.001)
                    forces[i] -= f * dx
                    forces[j] += f * dx
            for dim in range(2):
                if pts[i, dim] < 0.15: forces[i, dim] += 0.02
                elif pts[i, dim] > 0.85: forces[i, dim] -= 0.02
        pts += forces * 0.05 * (1.0 - step/steps)
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def project_to_bounds(x0):
    """Ensure optimization vector strictly respects bounds."""
    x0 = x0.copy()
    for i in range(N_CIRCLES):
        r = max(1e-7, x0[3 * i + 2])
        x0[3 * i] = np.clip(x0[3 * i], r, 1.0 - r)
        x0[3 * i + 1] = np.clip(x0[3 * i + 1], r, 1.0 - r)
        x0[3 * i + 2] = r
    return x0
