import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A = np.zeros((num_ineq, n))
    b = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A[idx, i] = 1.0
        b[idx] = max(0.0, lim)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = max(0.0, d)
            idx += 1
            
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0.0, None)] * n, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-9)
    except Exception:
        pass
    return np.full(n, 0.05)

def relax_centers(centers, radii, steps=300):
    """Force-directed relaxation to spread centers according to current radii."""
    n = N
    for t in range(steps):
        forces = np.zeros_like(centers)
        lr = 0.06 * (1.0 - t / steps)
        
        # Boundary repulsion
        for i in range(n):
            for d in range(2):
                if centers[i, d] < radii[i]:
                    forces[i, d] += 15.0 * (radii[i] - centers[i, d])
                elif centers[i, d] > 1.0 - radii[i]:
                    forces[i, d] -= 15.0 * (centers[i, d] - (1.0 - radii[i]))
                    
        # Pairwise repulsion
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[j, 0] - centers[i, 0]
                dy = centers[j, 1] - centers[i, 1]
                dist = np.hypot(dx, dy)
                target = radii[i] + radii[j]
                if dist < target and dist > 1e-8:
                    overlap = target - dist
                    f_mag = overlap * 8.0 / (dist + 1e-6)
                    fx = f_mag * dx / dist
                    fy = f_mag * dy / dist
                    forces[i] -= [fx, fy]
                    forces[j] += [fx, fy]
                    
        centers += forces * lr
        centers = np.clip(centers, 1e-7, 1.0 - 1e-7)
    return centers

def get_constraints(x):
    """Vectorized inequality constraints g(x) >= 0 for SLSQP."""
    cx, cy, r = x[0::3], x[1::3], x[2::3]
    c = np.concatenate([
        cx - r,
        1.0 - cx - r,
        cy - r,
        1.0 - cy - r
    ])
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dr = r[:, None] + r[None, :]
    i_idx, j_idx = np.tril_indices(N, -1)
    c = np.concatenate([c, dx[i_idx, j_idx]**2 + dy[i_idx, j_idx]**2 - dr[i_idx, j_idx]**2])
    return c

def get_bounds():
    """Variable bounds for SLSQP."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)])
    return b

def make_hex_init(r0, angle, shift_x, shift_y):
    """Generates a hexagonal lattice initialization with rotation and shift."""
    pts = []
    y = r0 + shift_y
    row = 0
    while len(pts) < N + 10:
        x = r0 + shift_x + (row % 2) * r0
        while x <= 1.0 - r0:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        if y > 1.0 + r0:
            break
            
    pts = np.array(pts[:N + 10])
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
        
    mask = (pts[:, 0] >= 0.04) & (pts[:, 0] <= 0.96) & (pts[:, 1] >= 0.04) & (pts[:, 1] <= 0.96)
    pts = pts[mask]
    
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.2, 0.8, (1, 2))])
    return pts[:N]
