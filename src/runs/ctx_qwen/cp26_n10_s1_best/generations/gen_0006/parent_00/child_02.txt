import numpy as np
from scipy.optimize import minimize, linprog

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Returns all inequality constraints g(x) >= 0."""
    cx, cy, r = x[0::3], x[1::3], x[2::3]
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    c = np.concatenate([
        cx - r,
        1.0 - cx - r,
        cy - r,
        1.0 - cy - r
    ])
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dr = r[:, None] + r[None, :]
    
    c = np.concatenate([c, dx[TRIL_IDX]**2 + dy[TRIL_IDX]**2 - dr[TRIL_IDX]**2])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [0, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
    return b

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A = np.zeros((num_ineq, n))
    b = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A[idx, i] = 1.0
        b[idx] = max(0.0, lim)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = max(0.0, d)
            idx += 1
            
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0.0, None)] * n, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-9)
    except Exception:
        pass
    return np.full(n, 0.05)

def make_hex_init(r0, angle=0.0, scale=1.0):
    """Generates a hexagonal lattice initialization with optional rotation and scaling."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        if y > 1.0 + r0: break
            
    pts = np.array(pts[:N + 10])
    
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
        
    pts = pts * scale + (1.0 - scale) * 0.5
    
    mask = (pts[:, 0] >= 0.01) & (pts[:, 0] <= 0.99) & (pts[:, 1] >= 0.01) & (pts[:, 1] <= 0.99)
    pts = pts[mask]
    
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.15, 0.85, (1, 2))])
    return pts[:N]

def force_push(centers, radii, steps=150):
    """Resolve overlaps and push circles to boundaries via simple force simulation."""
    pts = centers.copy()
    r = radii.copy()
    for _ in range(steps):
        f = np.zeros_like(pts)
        for i in range(N):
            for j in range(i + 1, N):
                dx = pts[j] - pts[i]
                d = np.linalg.norm(dx)
                if d < r[i] + r[j] and d > 1e-6:
                    force = (r[i] + r[j] - d) * 0.5 / d
                    f[i] -= force * dx
                    f[j] += force * dx
            for dim in range(2):
                if pts[i, dim] < r[i]: f[i, dim] += 0.1 * (r[i] - pts[i, dim])
                elif pts[i, dim] > 1.0 - r[i]: f[i, dim] -= 0.1 * (pts[i, dim] - (1.0 - r[i]))
        pts += f * 0.1
        pts = np.clip(pts, 0.001, 0.999)
    return pts

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    best_sum = -1.0
    best_x = None
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}

    # Phase 1: Diverse Initial Configurations
    inits = []
    for r0 in [0.09, 0.095, 0.10, 0.105]:
        for ang in np.linspace(-0.25, 0.25, 11):
            for sc in [0.95, 1.0, 1.05]:
                inits.append(make_hex_init(r0, ang, sc))
    for s in range(6):
        inits.append(np.random.uniform(0.15, 0.85, (N, 2)))

    for c_init in inits:
        r_init = solve_lp_radii(c_init)
        c_init = force_push(c_init, r_init, steps=120)
        x0 = np.zeros(3 * N)
        x0[0::3] = c_init[:, 0]
        x0[1::3] = c_init[:, 1]
        x0[2::3] = r_init

        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 15000, 'ftol': 1e-13, 'disp': False})
            if not np.isnan(res.fun):
                c_vals = constraints(res.x)
                if np.min(c_vals) >= -1e-7 and -res.fun > best_sum:
                    best_sum = -res.fun
                    best_x = res.x.copy()
        except Exception:
            pass

    # Phase 2: Adaptive Deflation & Refinement to escape local minima
    if best_x is not None:
        for step in range(40):
            noise = 0.0025 * (0.92 ** step)
            x_p = best_x + np.random.normal(0, noise, 3 * N)
            x_p[2::3] *= 0.985  # Deflate radii to create movement slack
            
            # Project to bounds
            for i in range(N):
                r = max(1e-7, x_p[3*i+2])
                x_p[3*i] = np.clip(x_p[3*i], r, 1.0-r)
                x_p[3*i+1] = np.clip(x_p[3*i+1], r, 1.0-r)
                x_p[3*i+2] = r

            try:
                res = minimize(objective, x_p, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 10000, 'ftol': 1e-13, 'disp': False})
                if not np.isnan(res.fun):
                    centers_p = np.column_stack((res.x[0::3], res.x[1::3]))
                    r_lp = solve_lp_radii(centers_p)
                    if r_lp is not None:
                        res.x[2::3] = r_lp
                        c_vals = constraints(res.x)
                        if np.min(c_vals) >= -1e-7 and -res.fun > best_sum:
                            best_sum = -res.fun
                            best_x = res.x.copy()
            except Exception:
                pass

    # Fallback
    if best_x is None:
        best_x = np.zeros(3 * N)
        best_x[0::3] = np.tile(np.linspace(0.12, 0.88, 5), 6)[:N]
        best_x[1::3] = np.repeat(np.linspace(0.12, 0.88, 6), 5)[:N]
        best_x[2::3] = 0.06

    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3].copy()

    # Final strict validity adjustment against 1e-12 tolerance
    for _ in range(100):
        valid = True
        for i in range(N):
            if radii[i] < 0: valid = False; break
            if centers[i, 0] - radii[i] < -1e-9 or centers[i, 0] + radii[i] > 1.0 + 1e-9: valid = False; break
            if centers[i, 1] - radii[i] < -1e-9 or centers[i, 1] + radii[i] > 1.0 + 1e-9: valid = False; break
        if valid:
            for i in range(N):
                for j in range(i + 1, N):
                    d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                    if d < radii[i] + radii[j] - 1e-9: valid = False; break
                if not valid: break
        if valid: break
        
        # Minimal shrinkage to recover strict feasibility
        radii *= 0.9995
        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)

    return centers, radii, float(np.sum(radii))
