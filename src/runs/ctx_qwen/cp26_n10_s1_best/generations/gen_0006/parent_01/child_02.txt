import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Returns all inequality constraints g(x) >= 0."""
    cx, cy, r = x[0::3], x[1::3], x[2::3]
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    c = np.concatenate([
        cx - r, 1.0 - cx - r,
        cy - r, 1.0 - cy - r
    ])
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dr = r[:, None] + r[None, :]
    i, j = np.tril_indices(N, -1)
    c = np.concatenate([c, dx[i, j]**2 + dy[i, j]**2 - dr[i, j]**2])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [0, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
    return b

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, d)
            idx += 1
            
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=[(0.0, None)] * n, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return np.full(n, 0.05)

def greedy_init(n, seed, num_samples=8000):
    """Generates a high-clearance initial configuration using sequential greedy placement."""
    np.random.seed(seed)
    centers = np.zeros((n, 2))
    for i in range(n):
        pts = np.random.uniform(0.0, 1.0, (num_samples, 2))
        d_bound = np.min(np.column_stack([pts, 1.0 - pts]), axis=1)
        d_pts = d_bound.copy()
        if i > 0:
            for k in range(i):
                d = np.hypot(pts[:, 0] - centers[k, 0], pts[:, 1] - centers[k, 1])
                d_pts = np.minimum(d_pts, d)
        best_idx = np.argmax(d_pts)
        centers[i] = pts[best_idx]
    return centers

def force_spread(centers, radii, steps=100):
    """Simple force-directed relaxation to resolve overlaps and push to boundaries."""
    pts = centers.copy()
    for _ in range(steps):
        f = np.zeros_like(pts)
        for i in range(N):
            for j in range(i + 1, N):
                dx = pts[j, 0] - pts[i, 0]
                dy = pts[j, 1] - pts[i, 1]
                d = np.hypot(dx, dy)
                if d < 0.3 and d > 1e-6:
                    rep = 0.012 / (d**2 + 0.001)
                    fx, fy = rep * dx / d, rep * dy / d
                    f[i] -= [fx, fy]
                    f[j] += [fx, fy]
            for dim in range(2):
                if pts[i, dim] < radii[i] + 0.02: f[i, dim] += 0.06
                elif pts[i, dim] > 1.0 - radii[i] - 0.02: f[i, dim] -= 0.06
        pts += f * 0.05
        pts = np.clip(pts, 0.01, 0.99)
    return pts

def project_to_bounds(x0):
    """Ensure optimization vector strictly respects variable bounds."""
    res = x0.copy()
    for i in range(N):
        r = max(1e-6, res[3 * i + 2])
        res[3 * i] = np.clip(res[3 * i], r, 1.0 - r)
        res[3 * i + 1] = np.clip(res[3 * i + 1], r, 1.0 - r)
        res[3 * i + 2] = r
    return res

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    best_sum = -1.0
    best_x = None
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}

    inits = []
    # Phase 1: Diverse Initial Configurations
    # 1. Greedy high-clearance layouts
    for s in range(12):
        inits.append(greedy_init(N, seed=s))
        
    # 2. Rotated & scaled hexagonal lattices
    for r0 in [0.085, 0.09, 0.095, 0.10]:
        for ang in np.linspace(-0.25, 0.25, 5):
            pts = np.zeros((N, 2))
            idx = 0
            y = r0
            row = 0
            while idx < N and y + r0 <= 1.0:
                x = r0 if row % 2 == 0 else 2.0 * r0
                while idx < N and x + r0 <= 1.0:
                    pts[idx] = [x, y]
                    idx += 1
                    x += 2.0 * r0
                y += np.sqrt(3.0) * r0
                row += 1
            while idx < N:
                pts[idx] = np.random.uniform(0.1, 0.9, 2)
                idx += 1
            c, s = np.cos(ang), np.sin(ang)
            pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
            inits.append(pts)

    # Optimize each initialization
    for c_init in inits:
        r_init = np.full(N, 0.06)
        c_spread = force_spread(c_init, r_init, steps=80)
        r_lp = solve_lp_radii(c_spread)
        
        x0 = np.zeros(3 * N)
        x0[0::3] = c_spread[:, 0]
        x0[1::3] = c_spread[:, 1]
        x0[2::3] = r_lp
        x0 = project_to_bounds(x0)

        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 15000, 'ftol': 1e-14, 'disp': False})
            if not np.isnan(res.fun):
                c_vals = constraints(res.x)
                if np.min(c_vals) >= -1e-6:
                    curr_sum = -res.fun
                    if curr_sum > best_sum:
                        best_sum = curr_sum
                        best_x = res.x.copy()
        except Exception:
            pass

    # Phase 2: Iterative Refinement & Escape
    if best_x is not None:
        # Alternate LP radii snap and SLSQP position optimization
        for _ in range(6):
            centers = np.column_stack((best_x[0::3], best_x[1::3]))
            r_lp = solve_lp_radii(centers)
            best_x[2::3] = r_lp
            x0 = project_to_bounds(best_x.copy())
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 10000, 'ftol': 1e-14, 'disp': False})
                if not np.isnan(res.fun) and -res.fun > best_sum:
                    best_x = res.x.copy()
                    best_sum = -res.fun
            except Exception:
                pass

        # Deflation + Perturbation to escape local minima
        for step in range(30):
            noise = 0.003 * (0.92 ** step)
            x_pert = best_x + np.random.normal(0, noise, 3 * N)
            x_pert[2::3] *= 0.98  # Deflate to create slack
            x_pert = project_to_bounds(x_pert)
            
            try:
                res = minimize(objective, x_pert, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 8000, 'ftol': 1e-14, 'disp': False})
                if not np.isnan(res.fun):
                    centers_p = np.column_stack((res.x[0::3], res.x[1::3]))
                    r_p_lp = solve_lp_radii(centers_p)
                    if r_p_lp is not None:
                        x_pert[2::3] = r_p_lp
                        x_pert = project_to_bounds(x_pert)
                        curr_sum = np.sum(x_pert[2::3])
                        if np.min(constraints(x_pert)) >= -1e-6 and curr_sum > best_sum:
                            best_sum = curr_sum
                            best_x = x_pert.copy()
            except Exception:
                pass

    # Fallback (should not be reached)
    if best_x is None:
        best_x = np.zeros(3 * N)
        best_x[0::3] = np.tile(np.linspace(0.1, 0.9, 5), 6)[:N]
        best_x[1::3] = np.repeat(np.linspace(0.1, 0.9, 6), 5)[:N]
        best_x[2::3] = 0.06

    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3].copy()

    # Final strict validity adjustment against 1e-12 tolerance
    for _ in range(100):
        valid = True
        for i in range(N):
            if radii[i] < 0 or centers[i, 0] < radii[i] - 1e-10 or centers[i, 0] > 1.0 - radii[i] + 1e-10 or \
               centers[i, 1] < radii[i] - 1e-10 or centers[i, 1] > 1.0 - radii[i] + 1e-10:
                valid = False; break
        if valid:
            for i in range(N):
                for j in range(i + 1, N):
                    if np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1]) < radii[i] + radii[j] - 1e-10:
                        valid = False; break
                if not valid: break
        if valid: break
        
        # Minimal shrinkage to recover strict feasibility
        radii *= 0.9995
        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)

    return centers, radii, float(np.sum(radii))
