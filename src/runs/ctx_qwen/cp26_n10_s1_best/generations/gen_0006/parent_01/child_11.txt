import numpy as np
from scipy.optimize import minimize, linprog

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Returns all inequality constraints g(x) >= 0."""
    cx, cy, r = x[0::3], x[1::3], x[2::3]
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    c = np.concatenate([
        cx - r,
        1.0 - cx - r,
        cy - r,
        1.0 - cy - r
    ])
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dr = r[:, None] + r[None, :]
    
    c = np.concatenate([c, dx[TRIL_IDX]**2 + dy[TRIL_IDX]**2 - dr[TRIL_IDX]**2])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [0, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
    return b

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, d)
            idx += 1
            
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=[(0.0, None)] * n, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-8)
    except Exception:
        pass
    return np.full(n, 0.05)

def push_apart(centers, radii, steps=150):
    """Iteratively resolve overlaps by pushing centers apart."""
    n = centers.shape[0]
    for _ in range(steps):
        max_overlap = 0.0
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[j, 0] - centers[i, 0]
                dy = centers[j, 1] - centers[i, 1]
                d = np.hypot(dx, dy)
                if d < radii[i] + radii[j] and d > 1e-7:
                    overlap = radii[i] + radii[j] - d
                    max_overlap = max(max_overlap, overlap)
                    shift = overlap * 0.5 / d
                    centers[i, 0] -= shift * dx
                    centers[i, 1] -= shift * dy
                    centers[j, 0] += shift * dx
                    centers[j, 1] += shift * dy
        if max_overlap < 1e-6:
            break
    return centers

def project_to_bounds(centers, radii):
    """Clip centers to strictly respect boundary constraints."""
    centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
    centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)
    return centers

def generate_init_configs():
    """Create diverse initial center configurations."""
    configs = []
    
    # 1. Hexagonal lattices with varying densities and rotations
    for r0 in np.linspace(0.082, 0.108, 6):
        for ang in np.linspace(-0.25, 0.25, 7):
            pts = []
            y = r0
            row = 0
            while len(pts) < N + 10:
                x = r0 if row % 2 == 0 else 2.0 * r0
                while x <= 1.0 - r0 and len(pts) < N + 10:
                    pts.append([x, y])
                    x += 2.0 * r0
                y += np.sqrt(3.0) * r0
                row += 1
            pts = np.array(pts[:N + 10])
            
            if ang != 0.0:
                c, s = np.cos(ang), np.sin(ang)
                pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
                
            mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
            pts = pts[mask]
            while len(pts) < N:
                pts = np.vstack([pts, np.random.uniform(0.2, 0.8, (1, 2))])
            configs.append(pts[:N].copy())
            
    # 2. Corner-heavy layouts (often optimal for square packing)
    for _ in range(8):
        pts = np.array([[0.15, 0.15], [0.85, 0.15], [0.15, 0.85], [0.85, 0.85]])
        y = 0.35
        row = 0
        r0 = 0.095
        while len(pts) < N:
            x = 0.35 if row % 2 == 0 else 0.45
            while x <= 0.65 and len(pts) < N:
                pts = np.vstack([pts, [x + np.random.uniform(-0.01, 0.01), 
                                       y + np.random.uniform(-0.01, 0.01)]])
                x += 2.0 * r0
            y += np.sqrt(3.0) * r0
            row += 1
        np.random.shuffle(pts)
        configs.append(pts[:N].copy())
        
    # 3. Random force-directed starts
    for seed in range(10):
        np.random.seed(seed)
        pts = np.random.uniform(0.2, 0.8, (N, 2))
        for _ in range(200):
            forces = np.zeros_like(pts)
            for i in range(N):
                for j in range(i + 1, N):
                    dx = pts[j] - pts[i]
                    d = np.linalg.norm(dx)
                    if d < 0.3 and d > 1e-5:
                        f = 0.01 / (d**2 + 0.001)
                        forces[i] -= f * dx
                        forces[j] += f * dx
                if pts[i, 0] < 0.15: forces[i, 0] += 0.02
                if pts[i, 0] > 0.85: forces[i, 0] -= 0.02
                if pts[i, 1] < 0.15: forces[i, 1] += 0.02
                if pts[i, 1] > 0.85: forces[i, 1] -= 0.02
            pts += forces * 0.05
            pts = np.clip(pts, 0.05, 0.95)
        configs.append(pts.copy())
        
    return configs

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    best_sum = -1.0
    best_centers = None
    best_radii = None
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}
    
    configs = generate_init_configs()
    
    # Phase 1: Multi-start alternating LP + SLSQP
    for c_init in configs:
        x0 = np.zeros(3 * N)
        x0[0::3] = c_init[:, 0]
        x0[1::3] = c_init[:, 1]
        
        # Initialize with LP-optimal radii for these centers
        r_lp = solve_lp_radii(c_init)
        x0[2::3] = r_lp
        
        # Project to strict feasibility
        for i in range(N):
            r = x0[3*i+2]
            x0[3*i] = np.clip(x0[3*i], r, 1.0 - r)
            x0[3*i+1] = np.clip(x0[3*i+1], r, 1.0 - r)
            x0[3*i+2] = max(r, 1e-6)
            
        # Alternating refinement
        for _ in range(4):
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 10000, 'ftol': 1e-13, 'disp': False})
                if not np.isnan(res.fun):
                    x0 = res.x.copy()
                    c_new = np.column_stack((x0[0::3], x0[1::3]))
                    r_new = solve_lp_radii(c_new)
                    x0[2::3] = r_new
                    # Re-project
                    for i in range(N):
                        r = x0[3*i+2]
                        x0[3*i] = np.clip(x0[3*i], r, 1.0 - r)
                        x0[3*i+1] = np.clip(x0[3*i+1], r, 1.0 - r)
            except Exception:
                break
                
        curr_sum = np.sum(x0[2::3])
        c_vals = constraints(x0)
        if np.min(c_vals) >= -1e-6 and curr_sum > best_sum:
            best_sum = curr_sum
            best_centers = np.column_stack((x0[0::3], x0[1::3])).copy()
            best_radii = x0[2::3].copy()
            
    if best_centers is not None:
        # Phase 2: Topology search via Radial Expansion & Swapping

            c_pert = best_centers.copy()

            x0 = np.zeros(3 * N)
            x0[0::3] = c_pert[:, 0]
            x0[1::3] = c_pert[:, 1]
            x0[2::3] = r_pert
            
            try:

                    c_opt = np.column_stack((x0[0::3], x0[1::3]))
                    r_opt = solve_lp_radii(c_opt)

            except Exception:
                pass
                
    # Fallback

        best_centers = np.column_stack((np.tile(np.linspace(0.1, 0.9, 5), 6)[:N],
                                        np.repeat(np.linspace(0.1, 0.9, 6), 5)[:N]))

    radii = best_radii
    for _ in range(100):

        for i in range(N):

            if centers[i,0] - radii[i] < -1e-10 or centers[i,0] + radii[i] > 1.0 + 1e-10: valid = False; break

        if valid: break

        centers = project_to_bounds(centers, radii)
        
    return centers, radii, float(np.sum(radii))
