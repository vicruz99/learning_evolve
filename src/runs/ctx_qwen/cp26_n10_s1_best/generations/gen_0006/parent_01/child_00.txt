1. **Alternating Optimization**: For fixed centers, the optimal radii are found exactly via Linear Programming (LP). For fixed radii (or coupled), center positions are optimized via SLSQP. Alternating between these two subproblems effectively navigates the non-convex landscape.
2. **Diverse Topologies**: Hexagonal lattices, force-directed layouts, and structured grids are excellent starting points. We will generate dozens of variants with different scales, rotations, and perturbations.
3. **Symmetry Breaking & Deflation**: Local minima are common. We introduce controlled noise and radius deflation (shrinking radii slightly to create slack) during refinement to allow centers to rearrange into better topologies.
4. **Strict Feasibility Enforcement**: A final repair loop guarantees compliance with the validator's `1e-12` tolerance by minimally shrinking radii and clipping centers if numerical drift occurs.
5. **Vectorized Constraints**: Computing all boundary and pairwise overlap constraints in vectorized NumPy operations accelerates the SLSQP steps significantly.

The strategy combines multi-start initialization, alternating LP/SLSQP refinement, aggressive perturbation to escape local optima, and a strict validity repair step.

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x):
    """Returns all inequality constraints g(x) >= 0 (vectorized)."""
    cx, cy, r = x[0::3], x[1::3], x[2::3]
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    c = np.concatenate([
        cx - r, 1.0 - cx - r,
        cy - r, 1.0 - cy - r
    ])
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dr = r[:, None] + r[None, :]
    
    c = np.concatenate([c, dx[TRIL_IDX]**2 + dy[TRIL_IDX]**2 - dr[TRIL_IDX]**2])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [0, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
    return b

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A = np.zeros((num_ineq, n))
    b = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A[idx, i] = 1.0
        b[idx] = max(0.0, lim)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = max(0.0, d)
            idx += 1
            
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0.0, None)] * n, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return np.full(n, 0.06)

def make_hex_init(r0, angle, seed):
    """Generates a hexagonal lattice initialization with optional rotation."""
    np.random.seed(seed)
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 5:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        if y > 1.0 + r0:
            break
            
    pts = np.array(pts[:N + 5])
    
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
        
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.1, 0.9, (1, 2))])
        
    pts += np.random.uniform(-0.003, 0.003, pts.shape)
    return np.clip(pts[:N], 0.02, 0.98)

def make_force_init(seed):
    """Force-directed layout to spread points evenly and push to boundaries."""
    np.random.seed(seed)
    pts = np.random.uniform(0.15, 0.85, (N, 2))
    for _ in range(500):
        forces = np.zeros_like(pts)
        for i in range(N):
            for j in range(i + 1, N):
                dx = pts[j] - pts[i]
                d = np.linalg.norm(dx)
                if d < 0.25 and d > 1e-5:
                    f = 0.012 / (d**2 + 0.0005)
                    forces[i] -= f * dx
                    forces[j] += f * dx
            for dim in range(2):
                if pts[i, dim] < 0.12: forces[i, dim] += 0.025
                elif pts[i, dim] > 0.88: forces[i, dim] -= 0.025
        pts += forces * 0.05
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def project_to_bounds(x0):
    """Ensure optimization vector strictly respects bounds."""
    x0 = x0.copy()
    for i in range(N):
        r = max(1e-6, x0[3 * i + 2])
        x0[3 * i] = np.clip(x0[3 * i], r, 1.0 - r)
        x0[3 * i + 1] = np.clip(x0[3 * i + 1], r, 1.0 - r)
        x0[3 * i + 2] = r
    return x0

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    best_sum = -1.0
    best_x = None
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}

    # Generate diverse initial configurations
    configs = []
    for r0 in np.linspace(0.08, 0.11, 5):
        configs.append(make_hex_init(r0, 0.0, 0))
        for ang in np.linspace(-0.25, 0.25, 9):
            configs.append(make_hex_init(r0, ang, 0))
            
    for s in range(15):
        configs.append(make_force_init(s))
        
    # Structured Grid
    pts_grid = []
    for i in range(6):
        for j in range(5):
            if len(pts_grid) >= N: break
            pts_grid.append([0.1 + j * 0.18, 0.1 + i * 0.16])
    configs.append(np.array(pts_grid[:N]) + np.random.uniform(-0.005, 0.005, (N, 2)))

    # Phase 1: Multi-start with Alternating LP + SLSQP
    for c_init in configs:
        x0 = np.zeros(3 * N)
        x0[0::3] = c_init[:, 0]
        x0[1::3] = c_init[:, 1]
        
        # Initial LP to snap radii to feasible maximum
        x0[2::3] = solve_lp_radii(c_init)
        x0 = project_to_bounds(x0)

        # Alternating refinement cycles
        for _ in range(4):
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 10000, 'ftol': 1e-14, 'disp': False})
                if not np.isnan(res.fun):
                    x0 = res.x.copy()
                    centers_new = x0.reshape(N, 3)[:, :2]
                    r_new = solve_lp_radii(centers_new)
                    x0[2::3] = r_new
                    x0 = project_to_bounds(x0)
            except Exception:
                break

        curr_sum = np.sum(x0[2::3])
        c_vals = constraints(x0)
        if np.min(c_vals) >= -1e-6 and curr_sum > best_sum:
            best_sum = curr_sum
            best_x = x0.copy()

    # Phase 2: Deflation & Perturbation to escape local minima
    if best_x is not None:
        for step in range(40):
            noise = 0.0015 * (0.96 ** step)
            x_pert = best_x + np.random.normal(0, noise, 3 * N)
            
            # Shrink radii slightly to create slack for repositioning
            r_pert = np.maximum(x_pert[2::3] * 0.97, 0.005)
            x_pert[2::3] = r_pert
            x_pert = project_to_bounds(x_pert)

            try:
                res = minimize(objective, x_pert, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 8000, 'ftol': 1e-14, 'disp': False})
                if not np.isnan(res.fun):
                    centers_pert = res.x.reshape(N, 3)[:, :2]
                    r_pert_lp = solve_lp_radii(centers_pert)
                    x_pert[2::3] = r_pert_lp
                    x_pert = project_to_bounds(x_pert)
                    
                    curr_sum = np.sum(x_pert[2::3])
                    c_vals = constraints(x_pert)
                    if np.min(c_vals) >= -1e-6 and curr_sum > best_sum:
                        best_sum = curr_sum
                        best_x = x_pert.copy()
            except Exception:
                pass

    # Fallback (should not be reached)
    if best_x is None:
        best_x = np.zeros(3 * N)
        best_x[0::3] = np.tile(np.linspace(0.1, 0.9, 5), 6)[:N]
        best_x[1::3] = np.repeat(np.linspace(0.1, 0.9, 6), 5)[:N]
        best_x[2::3] = 0.08

    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3].copy()

    # Final strict validity adjustment against 1e-12 tolerance
    for _ in range(100):
        valid = True
        for i in range(N):
            if radii[i] < 0: valid = False; break
            if centers[i, 0] - radii[i] < -1e-10 or centers[i, 0] + radii[i] > 1.0 + 1e-10: valid = False; break
            if centers[i, 1] - radii[i] < -1e-10 or centers[i, 1] + radii[i] > 1.0 + 1e-10: valid = False; break
        if valid:
            for i in range(N):
                for j in range(i + 1, N):
                    d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                    if d < radii[i] + radii[j] - 1e-10: valid = False; break
                if not valid: break
        if valid: break
        
        # Minimal shrinkage to recover strict feasibility
        radii *= 0.9999
        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)

    return centers, radii, float(np.sum(radii))
