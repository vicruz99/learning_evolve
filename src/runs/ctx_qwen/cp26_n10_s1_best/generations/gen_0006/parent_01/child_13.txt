import numpy as np
from scipy.optimize import minimize, linprog
import math

N = 26

def objective(x):
    return -np.sum(x[2::3])

def constraints(x):
    cx, cy, r = x[0::3], x[1::3], x[2::3]
    c = np.concatenate([
        cx - r,
        1.0 - cx - r,
        cy - r,
        1.0 - cy - r
    ])
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dr = r[:, None] + r[None, :]
    i_idx, j_idx = np.tril_indices(N, -1)
    c = np.concatenate([c, dx[i_idx, j_idx]**2 + dy[i_idx, j_idx]**2 - dr[i_idx, j_idx]**2])
    return c

def get_bounds():
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
    return b

def solve_lp_radii(centers):
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    idx = 0
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist)
            idx += 1
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-8)
    except Exception:
        pass
    return np.full(n, 0.05)

def project_to_feasible(x0):
    x0 = x0.copy()
    for i in range(N):
        r = max(1e-6, x0[3 * i + 2])
        x0[3 * i] = np.clip(x0[3 * i], r, 1.0 - r)
        x0[3 * i + 1] = np.clip(x0[3 * i + 1], r, 1.0 - r)
        x0[3 * i + 2] = r
    return x0

def hex_lattice(r0, angle=0.0, seed=0):
    np.random.seed(seed)
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0:
            pts.append([x + np.random.uniform(-0.002, 0.002), y + np.random.uniform(-0.002, 0.002)])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        if y > 1.0 + r0:
            break
    pts = np.array(pts[:N + 10])
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
    mask = (pts[:, 0] >= 0.01) & (pts[:, 0] <= 0.99) & (pts[:, 1] >= 0.01) & (pts[:, 1] <= 0.99)
    pts = pts[mask]
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.1, 0.9, (1, 2))])
    return pts[:N]

def corner_anchored_init(r0, angle=0.0, seed=0):
    np.random.seed(seed)
    pts = []
    corner_r = r0 * 1.15
    pts.append([corner_r, corner_r])
    pts.append([1.0 - corner_r, corner_r])
    pts.append([corner_r, 1.0 - corner_r])
    pts.append([1.0 - corner_r, 1.0 - corner_r])
    edge_r = r0 * 1.05
    pts.append([0.5, edge_r])
    pts.append([0.5, 1.0 - edge_r])
    pts.append([edge_r, 0.5])
    pts.append([1.0 - edge_r, 0.5])
    y = corner_r + r0
    row = 0
    while len(pts) < N + 5:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0 and len(pts) < N + 5:
            pts.append([x + np.random.uniform(-0.002, 0.002), y + np.random.uniform(-0.002, 0.002)])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        if y > 1.0 + r0:
            break
    pts = np.array(pts[:N + 5])
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
    mask = (pts[:, 0] >= 0.01) & (pts[:, 0] <= 0.99) & (pts[:, 1] >= 0.01) & (pts[:, 1] <= 0.99)
    pts = pts[mask]
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.1, 0.9, (1, 2))])
    return pts[:N]

def ring_init(seed=0):
    np.random.seed(seed)
    pts = []
    center = [0.5, 0.5]
    r1 = 0.12
    n1 = 6
    for i in range(n1):
        angle = 2 * math.pi * i / n1
        pts.append([center[0] + r1 * math.cos(angle), center[1] + r1 * math.sin(angle)])
    r2 = 0.24
    n2 = 12
    for i in range(n2):
        angle = 2 * math.pi * i / n2 + math.pi / n2
        pts.append([center[0] + r2 * math.cos(angle), center[1] + r2 * math.sin(angle)])
    while len(pts) < N:
        pts.append(center.copy())
    while len(pts) < N:
        pts.append(center.copy())
    pts = np.array(pts[:N]) + np.random.uniform(-0.005, 0.005, (N, 2))
    return np.clip(pts, 0.05, 0.95)

def force_simulate(centers, steps=800):
    pts = centers.copy()
    n = N
    for step in range(steps):
        forces = np.zeros_like(pts)
        lr = 0.03 * (1.0 - step / steps) ** 0.7
        for i in range(n):
            for j in range(i + 1, n):
                dx = pts[j] - pts[i]
                d = np.linalg.norm(dx)
                if d < 0.2 and d > 1e-8:
                    f = 0.008 / (d * d + 0.0005)
                    forces[i] -= f * dx / d
                    forces[j] += f * dx / d
            for dim in range(2):
                if pts[i, dim] < 0.08:
                    forces[i, dim] += 0.03
                elif pts[i, dim] > 0.92:
                    forces[i, dim] -= 0.03
        pts += forces * lr
        pts = np.clip(pts, 0.02, 0.98)
    return pts

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    best_sum = -1.0
    best_x = None
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraints}

    configs = []
    for r0 in np.linspace(0.085, 0.105, 7):
        for ang in np.linspace(-0.4, 0.4, 11):
            configs.append(hex_lattice(r0, ang, seed=int(r0 * 1000)))
    for r0 in np.linspace(0.085, 0.105, 5):
        for ang in np.linspace(-0.3, 0.3, 7):
            configs.append(corner_anchored_init(r0, ang, seed=int(r0 * 1000)))
    for s in range(10):
        configs.append(ring_init(seed=s))
    for s in range(15):
        pts = np.random.uniform(0.12, 0.88, (N, 2))
        configs.append(force_simulate(pts, steps=600))

    for c_init in configs:
        r_lp = solve_lp_radii(c_init)
        x0 = np.zeros(3 * N)
        x0[0::3] = c_init[:, 0]
        x0[1::3] = c_init[:, 1]
        x0[2::3] = r_lp
        x0 = project_to_feasible(x0)

        for cycle in range(6):
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 15000, 'ftol': 1e-14, 'disp': False})
                if not np.isnan(res.fun):
                    x0 = res.x.copy()
                    centers_new = np.column_stack((x0[0::3], x0[1::3]))
                    r_new = solve_lp_radii(centers_new)
                    x0[2::3] = r_new
                    x0 = project_to_feasible(x0)
            except Exception:
                break

        curr_sum = np.sum(x0[2::3])
        c_vals = constraints(x0)
        if np.min(c_vals) >= -1e-6 and curr_sum > best_sum:
            best_sum = curr_sum
            best_x = x0.copy()

    if best_x is not None:
        for rnd in range(40):
            noise = 0.003 * (0.90 ** rnd)
            x_pert = best_x + np.random.normal(0, noise, 3 * N)
            r_pert = np.maximum(x_pert[2::3] * 0.97, 0.008)
            x_pert[2::3] = r_pert
            x_pert = project_to_feasible(x_pert)
            try:
                res = minimize(objective, x_pert, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 10000, 'ftol': 1e-14, 'disp': False})
                if not np.isnan(res.fun):
                    centers_p = np.column_stack((res.x[0::3], res.x[1::3]))
                    r_lp = solve_lp_radii(centers_p)
                    if r_lp is not None:
                        x_pert[2::3] = r_lp
                        x_pert = project_to_feasible(x_pert)
                        curr = np.sum(x_pert[2::3])
                        if curr > best_sum and np.min(constraints(x_pert)) >= -1e-6:
                            best_sum = curr
                            best_x = x_pert.copy()
            except Exception:
                pass

    if best_x is not None:
        for rnd in range(30):
            x_sub = best_x.copy()
            subset = np.random.choice(N, size=N // 3, replace=False)
            x_sub[3 * subset + 2] *= 0.5
            x_sub = project_to_feasible(x_sub)
            try:
                res = minimize(objective, x_sub, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 10000, 'ftol': 1e-14, 'disp': False})
                if not np.isnan(res.fun):
                    centers_s = np.column_stack((res.x[0::3], res.x[1::3]))
                    r_lp = solve_lp_radii(centers_s)
                    if r_lp is not None:
                        x_sub[2::3] = r_lp
                        x_sub = project_to_feasible(x_sub)
                        curr = np.sum(x_sub[2::3])
                        if curr > best_sum and np.min(constraints(x_sub)) >= -1e-6:
                            best_sum = curr
                            best_x = x_sub.copy()
            except Exception:
                pass

    if best_x is not None:
        for rnd in range(20):
            ang = np.random.uniform(-0.15, 0.15)
            best_centers = np.column_stack((best_x[0::3], best_x[1::3]))
            c, s = np.cos(ang), np.sin(ang)
            rot_centers = (best_centers - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
            rot_centers = np.clip(rot_centers, 0.02, 0.98)
            r_lp = solve_lp_radii(rot_centers)
            x_rot = np.zeros(3 * N)
            x_rot[0::3] = rot_centers[:, 0]
            x_rot[1::3] = rot_centers[:, 1]
            x_rot[2::3] = r_lp
            x_rot = project_to_feasible(x_rot)
            try:
                res = minimize(objective, x_rot, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 8000, 'ftol': 1e-14, 'disp': False})
                if not np.isnan(res.fun):
                    centers_r = np.column_stack((res.x[0::3], res.x[1::3]))
                    r_final = solve_lp_radii(centers_r)
                    if r_final is not None:
                        x_rot[2::3] = r_final
                        x_rot = project_to_feasible(x_rot)
                        curr = np.sum(x_rot[2::3])
                        if curr > best_sum and np.min(constraints(x_rot)) >= -1e-6:
                            best_sum = curr
                            best_x = x_rot.copy()
            except Exception:
                pass

    if best_x is not None:
        for _ in range(5):
            try:
                res = minimize(objective, best_x, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 20000, 'ftol': 1e-15, 'disp': False})
                if not np.isnan(res.fun):
                    centers_final = np.column_stack((res.x[0::3], res.x[1::3]))
                    r_final = solve_lp_radii(centers_final)
                    if r_final is not None:
                        best_x[2::3] = r_final
                        best_x = project_to_feasible(best_x)
                        curr = np.sum(best_x[2::3])
                        if curr > best_sum and np.min(constraints(best_x)) >= -1e-6:
                            best_sum = curr
            except Exception:
                pass

    if best_x is None:
        best_x = np.zeros(3 * N)
        best_x[0::3] = np.tile(np.linspace(0.1, 0.9, 5), 6)[:N]
        best_x[1::3] = np.repeat(np.linspace(0.1, 0.9, 6), 5)[:N]
        best_x[2::3] = 0.06

    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3].copy()

    for _ in range(100):
        valid = True
        for i in range(N):
            if radii[i] < 0:
                valid = False
                break
            if centers[i, 0] - radii[i] < -1e-10 or centers[i, 0] + radii[i] > 1.0 + 1e-10:
                valid = False
                break

                valid = False
                break

            for i in range(N):
                for j in range(i + 1, N):
                    d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                    if d < radii[i] + radii[j] - 1e-10:
                        valid = False
                        break
                if not valid:
                    break
        if valid:
            break
        radii *= 0.9999
        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)

    return centers, radii, float(np.sum(radii))
