import numpy as np
from scipy.optimize import minimize, linprog

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [0, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
    return b

def constraint_func(x):
    """Returns all inequality constraints g(x) >= 0 (vectorized)."""
    cx, cy, r = x[0::3], x[1::3], x[2::3]
    
    # Boundary constraints
    c = np.concatenate([
        cx - r,
        1.0 - cx - r,
        cy - r,
        1.0 - cy - r
    ])
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dr = r[:, None] + r[None, :]
    
    c = np.concatenate([c, dx[TRIL_IDX]**2 + dy[TRIL_IDX]**2 - dr[TRIL_IDX]**2])
    return c

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, d)
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and not np.any(np.isnan(res.x)):
            return res.x
    except Exception:
        pass
    return np.full(n, 0.05)

def generate_hex_init(r0, angle, seed):
    """Generates a rotated and perturbed hexagonal lattice."""
    rng = np.random.default_rng(seed)
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 5:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        if y > 1.0 + r0:
            break
            
    pts = np.array(pts[:N + 5])
    
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
        
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    
    while len(pts) < N:
        pts = np.vstack([pts, rng.uniform(0.15, 0.85, (1, 2))])
        
    pts = pts[:N] + rng.uniform(-0.005, 0.005, (N, 2))
    return np.clip(pts, 0.01, 0.99)

def generate_boundary_grid_init(seed):
    """Places circles near boundaries and corners, then fills interior."""
    rng = np.random.default_rng(seed)
    pts = []
    # Corners
    pts.extend([[0.12, 0.12], [0.88, 0.12], [0.12, 0.88], [0.88, 0.88]])
    # Edges
    for i in range(4):
        pts.append([0.25 + i*0.06, 0.1])
        pts.append([0.25 + i*0.06, 0.9])
        pts.append([0.1, 0.25 + i*0.06])
        pts.append([0.9, 0.25 + i*0.06])
    # Interior fill
    while len(pts) < N:
        pts.append(rng.uniform(0.2, 0.8, 2))
        
    pts = np.array(pts[:N]) + rng.uniform(-0.01, 0.01, (N, 2))
    return np.clip(pts, 0.02, 0.98)

def project_to_bounds(x0):
    """Ensure optimization vector strictly respects geometric bounds."""
    for i in range(N):
        r = max(1e-6, x0[3 * i + 2])
        x0[3 * i] = np.clip(x0[3 * i], r, 1.0 - r)
        x0[3 * i + 1] = np.clip(x0[3 * i + 1], r, 1.0 - r)
        x0[3 * i + 2] = r
    return x0

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    np.random.seed(42)
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraint_func}

    best_x = None

    configs = []
    
    # Hexagonal lattices with varied scales and rotations
    for r0 in np.linspace(0.085, 0.105, 5):
        configs.append(generate_hex_init(r0, 0.0, 0))
        for ang in np.linspace(-0.25, 0.25, 7):
            configs.append(generate_hex_init(r0, ang, hash(ang) % 1000))
            
    # Boundary-aligned grids
    for s in range(6):
        configs.append(generate_boundary_grid_init(s))
        
    # Random perturbed grids
    for s in range(6):
        rng = np.random.default_rng(s)
        c = np.zeros((N, 2))
        idx = 0
        for i in range(6):
            for j in range(5):
                if idx < N:
                    c[idx] = [0.15 + j * 0.16, 0.15 + i * 0.15]
                    idx += 1
        c += rng.uniform(-0.02, 0.02, (N, 2))
        configs.append(np.clip(c, 0.05, 0.95))

    # Phase 2: Multi-start Optimization with LP-SLSQP Alternation
    for c_init in configs:

        x0 = project_to_bounds(x0)

            try:
                res = minimize(lambda v: -np.sum(v[2::3]), x0, method='SLSQP',
