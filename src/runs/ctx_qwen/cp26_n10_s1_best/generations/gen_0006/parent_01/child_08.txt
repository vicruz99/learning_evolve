

<strategy>
The core challenge in circle packing is navigating the highly non-convex constraint landscape. Gradient-based optimizers like SLSQP often get trapped in local minima corresponding to suboptimal lattice distortions. To overcome this, I will use a hybrid strategy:

1. **LP-based Radius Optimization**: For any fixed set of center coordinates, computing the maximum sum of radii subject to non-overlap and boundary constraints is a Linear Programming problem. This is exact, fast, and guarantees feasibility.
2. **Simulated Annealing (SA) on Centers**: Instead of optimizing centers and radii simultaneously, I will optimize only the centers. The objective function evaluated at each step is the LP-optimal sum of radii. SA's stochastic nature and temperature schedule allow it to accept temporary "worse" configurations, helping it escape shallow local minima and discover better topological arrangements (e.g., boundary adaptations, lattice tilts).
3. **Vectorized Evaluation**: Pairwise distances and boundary limits are computed using NumPy vectorization, and the HiGHS LP solver handles the radius optimization in milliseconds.
4. **Gradient Refinement**: After SA identifies the best basin, I switch to SLSQP to perform precise local refinement, leveraging its ability to converge to exact constraint boundaries.
5. **Diverse Initializations**: Multiple rotated/scaled hexagonal grids and repulsive-force layouts ensure the search starts from promising regions of the configuration space.

This decoupling dramatically improves robustness and typically yields higher packing sums than monolithic non-linear solvers.
</strategy>

import numpy as np
from scipy.optimize import minimize, linprog

N = 26
PAIRS = [(i, j) for i in range(N) for j in range(i + 1, N)]
N_PAIRS = len(PAIRS)

# Precompute LP constraint matrix structure
A_LP = np.zeros((N + N_PAIRS, N))
idx = 0
for i in range(N):
    A_LP[idx, i] = 1.0
    idx += 1
for i, j in PAIRS:
    A_LP[idx, i] = 1.0
    A_LP[idx, j] = 1.0
    idx += 1

bounds_lp = [(0.0, None)] * N
tri_i, tri_j = np.triu_indices(N, 1)

def compute_lp(centers):
    """Solve LP to find optimal radii for fixed centers."""
    b = np.empty(N + N_PAIRS)
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    b[:N] = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                        np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    
    # Overlap constraints: r_i + r_j <= dist(i, j)
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    b[N:] = dists[tri_i, tri_j]
    
    res = linprog(-np.ones(N), A_ub=A_LP, b_ub=b, bounds=bounds_lp, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0), res.fun * -1.0
    return None, -1.0

def objective_slsqp(x):
    return -np.sum(x[2::3])

def constraints_slsqp(x):
    cx, cy, r = x[0::3], x[1::3], x[2::3]
    c = np.concatenate([
        cx - r,
        1.0 - cx - r,
        cy - r,
        1.0 - cy - r
    ])
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dr = r[:, None] + r[None, :]
    c = np.concatenate([c, dx[tri_i, tri_j]**2 + dy[tri_i, tri_j]**2 - dr[tri_i, tri_j]**2])
    return c

def get_bounds_slsqp():
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
    return b

def generate_hex_init(seed, scale=1.0, angle=0.0):
    np.random.seed(seed)
    r0 = 0.095 * scale
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 5:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        if y > 1.0 + r0: break
        
    pts = np.array(pts[:N + 5])
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
        
    mask = (pts[:, 0] >= 0.05) & (pts[:, 0] <= 0.95) & (pts[:, 1] >= 0.05) & (pts[:, 1] <= 0.95)
    pts = pts[mask]
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.15, 0.85, (1, 2))])
    return pts[:N]

def generate_force_init(seed):
    np.random.seed(seed)
    pts = np.random.uniform(0.1, 0.9, (N, 2))
    for _ in range(150):
        f = np.zeros_like(pts)
        for i in range(N):
            for j in range(i + 1, N):
                d = np.hypot(pts[i, 0] - pts[j, 0], pts[i, 1] - pts[j, 1])
                if d < 0.25 and d > 1e-5:
                    v = pts[j] - pts[i]
                    v /= d
                    f[i] -= v / (d**2 + 0.001)
                    f[j] += v / (d**2 + 0.001)
        pts += f * 0.008
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    best_centers = None
    best_radii = None
    best_sum = -1.0
    
    # Phase 1: Diverse Starts + Simulated Annealing
    starts = []
    for s in range(12):
        starts.append(generate_hex_init(s, scale=np.random.uniform(0.85, 1.15), 
                                        angle=np.random.uniform(-0.4, 0.4)))
    for s in range(8):
        starts.append(generate_force_init(s))
        
    for c_init in starts:
        centers = c_init.copy()
        radii, curr_sum = compute_lp(centers)
        if radii is None: continue
        
        best_c_run = centers.copy()
        best_r_run = radii.copy()
        best_s_run = curr_sum
        
        temp = 0.004
        step = 0.012
        
        for t in range(700):
            noise = np.random.normal(0, step, centers.shape)
            new_c = np.clip(centers + noise, 0.03, 0.97)
            new_r, new_s = compute_lp(new_c)
            if new_r is None: continue
            
            if new_s > best_s_run:
                best_s_run = new_s
                best_c_run = new_c.copy()
                best_r_run = new_r.copy()
                
            if new_s > curr_sum:
                centers = new_c.copy()
                curr_sum = new_s
                radii = new_r.copy()
            else:
                prob = np.exp((new_s - curr_sum) / max(temp, 1e-6))
                if np.random.rand() < prob:
                    centers = new_c.copy()
                    curr_sum = new_s
                    radii = new_r.copy()
                    
            temp *= 0.996
            step *= 0.998
            
        if best_s_run > best_sum:
            best_sum = best_s_run
            best_centers = best_c_run.copy()
            best_radii = best_r_run.copy()

    # Phase 2: SLSQP Gradient Refinement
    bounds = get_bounds_slsqp()
    cons = {'type': 'ineq', 'fun': constraints_slsqp}
    
    x0 = np.zeros(3 * N)
    x0[0::3] = best_centers[:, 0]
    x0[1::3] = best_centers[:, 1]
    x0[2::3] = best_radii
    
    for i in range(N):
        r = max(1e-6, x0[3*i+2])
        x0[3*i] = np.clip(x0[3*i], r, 1.0-r)
        x0[3*i+1] = np.clip(x0[3*i+1], r, 1.0-r)
        x0[3*i+2] = r
        
    for _ in range(3):
        try:
            res = minimize(objective_slsqp, x0, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 20000, 'ftol': 1e-14, 'disp': False})
            if -res.fun > best_sum:
                best_centers = np.column_stack((res.x[0::3], res.x[1::3]))
                best_radii = res.x[2::3]
                best_sum = -res.fun
                x0 = res.x.copy()
        except Exception:
            pass
            
        # Slight perturbation to escape flat regions
        x0[0::3] += np.random.normal(0, 1e-5, N)
        x0[1::3] += np.random.normal(0, 1e-5, N)
        for i in range(N):
            r = max(1e-6, x0[3*i+2])
            x0[3*i] = np.clip(x0[3*i], r, 1.0-r)
            x0[3*i+1] = np.clip(x0[3*i+1], r, 1.0-r)

    # Final strict validation & minimal repair
    for _ in range(100):
        valid = True
        for i in range(N):
            if best_radii[i] < 0 or best_centers[i,0] < best_radii[i] - 1e-10 or best_centers[i,0] > 1 - best_radii[i] + 1e-10 or \
               best_centers[i,1] < best_radii[i] - 1e-10 or best_centers[i,1] > 1 - best_radii[i] + 1e-10:
                valid = False; break
        if valid:
            for i in range(N):
                for j in range(i+1, N):
                    d = np.hypot(best_centers[i,0]-best_centers[j,0], best_centers[i,1]-best_centers[j,1])
                    if d < best_radii[i] + best_radii[j] - 1e-10:
                        valid = False; break
                if not valid: break
        if valid: break
        best_radii *= 0.9999
        best_centers[:, 0] = np.clip(best_centers[:, 0], best_radii, 1.0 - best_radii)
        best_centers[:, 1] = np.clip(best_centers[:, 1], best_radii, 1.0 - best_radii)
        
    return best_centers, best_radii, float(np.sum(best_radii))
