import numpy as np
from scipy.optimize import minimize, linprog

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def get_constraints(x):
    """Vectorized inequality constraints: g(x) >= 0."""
    cx, cy, r = x[0::3], x[1::3], x[2::3]
    
    # Boundary constraints
    c = np.concatenate([
        cx - r,
        1.0 - cx - r,
        cy - r,
        1.0 - cy - r
    ])
    
    # Overlap constraints: dist^2 >= (ri + rj)^2
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dr = r[:, None] + r[None, :]
    
    c = np.concatenate([c, dx[TRIL_IDX]**2 + dy[TRIL_IDX]**2 - dr[TRIL_IDX]**2])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-7, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)])
    return b

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A = np.zeros((num_ineq, n))
    b = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
        b[idx] = max(0.0, lim)
        A[idx, i] = 1.0
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i+1, n):
            d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
            b[idx] = max(0.0, d)
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            idx += 1
            
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0.0, None)]*n, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def force_init(seed, steps=600):
    """Force-directed layout to spread points evenly and push to boundaries."""
    np.random.seed(seed)
    pts = np.random.uniform(0.1, 0.9, (N, 2))
    for step in range(steps):
        f = np.zeros_like(pts)
        for i in range(N):
            for j in range(i+1, N):
                dx = pts[j] - pts[i]
                d = np.hypot(dx[0], dx[1])
                if d < 0.25 and d > 1e-5:
                    rep = 0.015 / (d**2 + 0.001)
                    f[i] -= rep * dx
                    f[j] += rep * dx
            for dim in range(2):
                if pts[i, dim] < 0.15: f[i, dim] += 0.06
                elif pts[i, dim] > 0.85: f[i, dim] -= 0.06
        pts += f * 0.04 * (1.0 - step/steps)
        pts = np.clip(pts, 0.02, 0.98)
    return pts

def hex_init(r0, angle, scale=1.0):
    """Generates a hexagonal lattice initialization with rotation and scaling."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 + r0:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        if y > 1.0 + r0:
            break
            
    pts = np.array(pts[:N + 10]) * scale
    
    # Rotate around center
    center = np.array([0.5, 0.5])
    pts = (pts - center) @ np.array([[np.cos(angle), -np.sin(angle)], [np.sin(angle), np.cos(angle)]]) + center
    
    # Filter valid points and pad if necessary
    mask = (pts[:, 0] >= 0.01) & (pts[:, 0] <= 0.99) & (pts[:, 1] >= 0.01) & (pts[:, 1] <= 0.99)
    pts = pts[mask]
    
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.1, 0.9, (1, 2))])
    return pts[:N]

def project_to_valid(x0):
    """Project optimization vector to strictly satisfy bounds."""
    x0 = x0.copy()
    for i in range(N):
        r = max(1e-7, x0[3 * i + 2])
        x0[3 * i] = np.clip(x0[3 * i], r, 1.0 - r)
        x0[3 * i + 1] = np.clip(x0[3 * i + 1], r, 1.0 - r)
        x0[3 * i + 2] = r
    return x0

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    cons = {'type': 'ineq', 'fun': get_constraints}

                configs.append(hex_init(r0, ang, sc))

    for s in range(10):
        configs.append(force_init(s))
        
    # Phase 2: Multi-start Optimization with LP-SLSQP Alternation
    for c_init in configs:
        x0 = np.zeros(3 * N)
        x0[0::3] = c_init[:, 0]
        x0[1::3] = c_init[:, 1]
        
        # Initial LP to snap radii to feasible maximum
        r_lp = solve_lp_radii(c_init)
        if r_lp is None:

        x0[2::3] = r_lp
        x0 = project_to_valid(x0)

        # Alternating refinement cycles

            try:

                               constraints=cons, options={'maxiter': 15000, 'ftol': 1e-14, 'disp': False})
                if not np.isnan(res.fun):
                    x0 = res.x.copy()
                    centers_new = np.column_stack((x0[0::3], x0[1::3]))
                    r_new = solve_lp_radii(centers_new)
                    if r_new is not None:
                        x0[2::3] = r_new
                        x0 = project_to_valid(x0)
            except Exception:
                break

        c_vals = get_constraints(x0)

            best_x = x0.copy()

                pass

    # Fallback (should not be reached)

        best_x[0::3] = np.tile(np.linspace(0.1, 0.9, 5), 6)[:N]
        best_x[1::3] = np.repeat(np.linspace(0.1, 0.9, 6), 5)[:N]
        best_x[2::3] = 0.06

    centers = np.column_stack((best_x[0::3], best_x[1::3]))

    # Final strict validity adjustment against 1e-12 tolerance
    for _ in range(100):

    return centers, radii, float(np.sum(radii))
