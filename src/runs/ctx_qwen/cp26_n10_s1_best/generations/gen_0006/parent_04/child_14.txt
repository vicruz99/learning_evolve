N = 26
TRIL_IDX = np.tril_indices(N, -1)

def objective(x):

    return -np.sum(x[2::3])

def get_constraints(x):
    """Vectorized inequality constraints: g(x) >= 0."""
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]

    c = np.concatenate([
        cx - r,
        1.0 - cx - r,
        cy - r,
        1.0 - cy - r
    ])

    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dr = r[:, None] + r[None, :]
    
    c = np.concatenate([c, dx[TRIL_IDX]**2 + dy[TRIL_IDX]**2 - dr[TRIL_IDX]**2])

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-6, 0.5]."""

        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)])
    return b

def solve_lp_radii(centers):

    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A = np.zeros((num_ineq, n))
    b = np.zeros(num_ineq)
    
    idx = 0

        lim = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])

        A[idx, i] = 1.0
        idx += 1

            A[idx, i] = 1.0
            A[idx, j] = 1.0
            idx += 1

        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0.0, None)]*n, method='highs')

            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def project_to_bounds(x0):

    x0 = x0.copy()

        r = max(1e-6, x0[3*i+2])

        x0[3*i+2] = r
    return x0

        pts += f * 0.05
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def hex_init(pat, r0, angle=0.0):
