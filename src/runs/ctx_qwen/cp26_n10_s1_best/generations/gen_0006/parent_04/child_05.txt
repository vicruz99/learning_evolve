import numpy as np
from scipy.optimize import minimize, linprog

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def get_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    M = n + n * (n - 1) // 2
    A_ub = np.zeros((M, n))
    b_ub = np.zeros(M)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], 
                  centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    # Pairwise overlap constraints: r_i + r_j <= dist(i, j)
    dx = centers[:, 0:1] - centers[None, :, 0]
    dy = centers[:, 1:2] - centers[None, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    d_vals = dists[TRIL_IDX]
    
    i_idx, j_idx = TRIL_IDX
    A_ub[idx:, i_idx] = 1.0
    A_ub[idx:, j_idx] = 1.0
    b_ub[idx:] = np.maximum(0.0, d_vals)
    
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return np.full(n, 0.06)

def constraints_joint(x):
    """Vectorized inequality constraints: g(x) >= 0 for SLSQP."""
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    # Boundary constraints
    c = np.concatenate([
        cx - r,
        1.0 - cx - r,
        cy - r,
        1.0 - cy - r
    ])
    
    # Overlap constraints (squared form)
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dr = r[:, None] + r[None, :]
    
    c = np.concatenate([c, dx[TRIL_IDX]**2 + dy[TRIL_IDX]**2 - dr[TRIL_IDX]**2])
    return c

def objective_joint(x):
    """Objective to minimize: negative sum of radii."""
    return -np.sum(x[2::3])

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-7, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)])
    return b

def hex_init(r0, angle=0.0, seed=0):
    """Generates a hexagonal lattice initialization."""
    np.random.seed(seed)
    centers = np.zeros((N, 2))
    idx = 0
    y = r0
    row = 0
    while idx < N and y + r0 <= 1.0:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while idx < N and x + r0 <= 1.0:
            centers[idx] = [x, y]
            idx += 1
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        
    while idx < N:
        centers[idx] = np.random.uniform(0.1, 0.9, 2)
        idx += 1
        
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        cx, cy = 0.5, 0.5
        dx = centers[:, 0] - cx
        dy = centers[:, 1] - cy
        centers[:, 0] = dx * c - dy * s + cx
        centers[:, 1] = dx * s + dy * c + cy
    return centers
