import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A = np.zeros((num_ineq, n))
    b = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i,0], 1.0 - centers[i,0], centers[i,1], 1.0 - centers[i,1])
        b[idx] = max(0.0, lim)
        A[idx, i] = 1.0
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers[i,0] - centers[j,0], centers[i,1] - centers[j,1])
            b[idx] = max(0.0, d)
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            idx += 1
            
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0.0, None)] * n, method='highs')
        if res.success:
            return np.sum(res.x), res.x
    except Exception:
        pass
    return 0.0, np.zeros(n)

def lp_objective(centers_flat):
    """Objective for center optimization: negative sum of optimal radii."""
    centers = centers_flat.reshape(N, 2)
    centers = np.clip(centers, 1e-7, 1.0 - 1e-7)
    s, _ = solve_lp_radii(centers)
    return -s

def generate_hex_centers(pattern, angle, r0):
    """Generates a hexagonal lattice initialization."""
    centers = np.zeros((N, 2))
    idx = 0
    y = r0
    y_step = np.sqrt(3.0) * r0
    for i, cnt in enumerate(pattern):
        x_start = r0 + (r0 if i % 2 == 1 else 0.0)
        for k in range(cnt):
            if idx < N:
                centers[idx, 0] = x_start + k * 2.0 * r0
                centers[idx, 1] = y
                idx += 1
        y += y_step
    while idx < N:
        centers[idx, 0] = np.random.uniform(0.1, 0.9)
        centers[idx, 1] = np.random.uniform(0.1, 0.9)
        idx += 1
        
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        cx, cy = 0.5, 0.5
        dx = centers[:,0] - cx
        dy = centers[:,1] - cy
        centers[:,0] = dx * c - dy * s + cx
        centers[:,1] = dx * s + dy * c + cy
        
    return centers

def objective_slsqp(x):
    """Minimize negative sum of radii for SLSQP."""
    return -np.sum(x[2::3])

def constraints_slsqp(x):
    """Vectorized inequality constraints for SLSQP: g(x) >= 0."""
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    c = np.concatenate([
        cx - r,
        1.0 - cx - r,
        cy - r,
        1.0 - cy - r
    ])
    
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dr = r[:, None] + r[None, :]
    
    i, j = np.tril_indices(N, -1)
    c = np.concatenate([c, dx[i, j]**2 + dy[i, j]**2 - dr[i, j]**2])
    return c

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    patterns = [[6,5,6,5,4], [5,6,5,6,4], [5,5,6,6,4], [4,6,5,6,5], [6,6,5,5,4], [5,6,6,5,4]]
    inits = []
    
    for p in patterns:
        for ang in np.linspace(-0.25, 0.25, 7):
            c = generate_hex_centers(p, ang, 0.085 + np.random.uniform(-0.005, 0.005))
            c += np.random.normal(0, 0.003, c.shape)
            inits.append(c.reshape(-1))
            
    for _ in range(12):
        c = np.random.uniform(0.15, 0.85, (N, 2))
        inits.append(c.reshape(-1))
        
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Phase 1: Derivative-free center optimization via LP objective
    for x0 in inits:
        try:
            res = minimize(lp_objective, x0, method='Nelder-Mead', 
                           options={'maxiter': 3500, 'xatol': 1e-5, 'fatol': 1e-7})
            curr_sum = -res.fun
            if curr_sum > best_sum:
                best_sum = curr_sum
                best_centers = res.x.reshape(N, 2)
                _, best_radii = solve_lp_radii(best_centers)
        except Exception:
            continue
            
    # Phase 2: Gradient-based joint refinement
    x0_full = np.zeros(3 * N)
    x0_full[0::3] = best_centers[:, 0]
    x0_full[1::3] = best_centers[:, 1]
    x0_full[2::3] = best_radii
    
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints_slsqp}
    
    try:
        res_slsqp = minimize(objective_slsqp, x0_full, method='SLSQP', bounds=bounds,
                             constraints=cons, options={'maxiter': 20000, 'ftol': 1e-14})
        if not np.isnan(res_slsqp.fun):
            c_opt = np.column_stack((res_slsqp.x[0::3], res_slsqp.x[1::3]))
            _, r_opt = solve_lp_radii(c_opt)
            if np.sum(r_opt) > best_sum:
                best_sum = np.sum(r_opt)
                best_centers = c_opt
                best_radii = r_opt
    except Exception:
        pass

        pert_c = np.clip(pert_c, 0.01, 0.99)
        
        try:
            res_pert = minimize(lp_objective, pert_c.reshape(-1), method='Nelder-Mead', 
                                options={'maxiter': 1200, 'xatol': 1e-5, 'fatol': 1e-7})
            curr_sum = -res_pert.fun
            if curr_sum > best_sum:
                best_sum = curr_sum

                _, best_radii = solve_lp_radii(best_centers)
        except Exception:
            continue
            
    # Phase 4: Final SLSQP polish
    x0_full = np.zeros(3 * N)
    x0_full[0::3] = best_centers[:, 0]
    x0_full[1::3] = best_centers[:, 1]
    x0_full[2::3] = best_radii
    
    try:
        res_final = minimize(objective_slsqp, x0_full, method='SLSQP', bounds=bounds,
                             constraints=cons, options={'maxiter': 15000, 'ftol': 1e-14})
        if not np.isnan(res_final.fun):
            c_opt = np.column_stack((res_final.x[0::3], res_final.x[1::3]))
            _, r_opt = solve_lp_radii(c_opt)
            if np.sum(r_opt) > best_sum:
                best_sum = np.sum(r_opt)
                best_centers = c_opt
                best_radii = r_opt
    except Exception:
        pass

    # Final strict validity adjustment
    centers = best_centers.copy()
    radii = best_radii.copy()
    
    for _ in range(100):
        valid = True
        for i in range(N):
            if radii[i] < 0 or centers[i,0] < radii[i] - 1e-9 or centers[i,0] > 1.0 - radii[i] + 1e-9 or \
               centers[i,1] < radii[i] - 1e-9 or centers[i,1] > 1.0 - radii[i] + 1e-9:
                valid = False
                break
        if valid:
            for i in range(N):
                for j in range(i + 1, N):
                    d = np.hypot(centers[i,0] - centers[j,0], centers[i,1] - centers[j,1])
                    if d < radii[i] + radii[j] - 1e-9:
                        valid = False
                        break
                if not valid:
                    break
        if valid:
            break
            
        radii *= 0.9995
        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)
        
    return centers, radii, float(best_sum)
