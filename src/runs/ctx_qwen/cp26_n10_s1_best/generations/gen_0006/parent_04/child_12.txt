def constraints(x):

    c = np.concatenate([

        1.0 - cy - r
    ])

    dy = cy[:, None] - cy[None, :]

    dr = r[:, None] + r[None, :]
    
    c = np.concatenate([c, dist[TRIL_IDX] - dr[TRIL_IDX]])
    return c
