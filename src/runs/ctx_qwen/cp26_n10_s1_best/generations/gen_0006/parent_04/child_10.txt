import numpy as np
from scipy.optimize import minimize, linprog
import math

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def get_radii_lp(centers):
    """Solve LP to maximize sum of radii for fixed centers."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], 
                  centers[i, 1], 1.0 - centers[i, 1])
        b_ub[idx] = max(0.0, lim)
        A_ub[idx, i] = 1.0
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    dx = centers[:, 0][:, None] - centers[:, 0][None, :]
    dy = centers[:, 1][:, None] - centers[:, 1][None, :]
    dists = np.hypot(dx, dy)
    
    A_ub[idx:, TRIL_IDX[0]] = 1.0
    A_ub[idx:, TRIL_IDX[1]] = 1.0
    b_ub[idx:] = np.maximum(dists[TRIL_IDX], 0.0)
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, 
                      bounds=[(0.0, None)] * n, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return np.full(n, 0.0)

def objective_joint(x):
    """Objective for SLSQP polish: minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints_joint(x):
    """Joint constraints for SLSQP: g(x) >= 0."""
    cx, cy, r = x[0::3], x[1::3], x[2::3]
    c = np.concatenate([
        cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r
    ])
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dr = r[:, None] + r[None, :]
    c = np.concatenate([c, dx[TRIL_IDX]**2 + dy[TRIL_IDX]**2 - dr[TRIL_IDX]**2])
    return c

def get_bounds_joint():
    return [(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)] * N

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Phase 1: Generate diverse initial configurations
    configs = []
    
    # Hexagonal lattices with varied scales and rotations
    for r0 in np.linspace(0.08, 0.11, 5):
        for ang in np.linspace(-0.4, 0.4, 9):
            pts = []
            y = r0
            row = 0
            while len(pts) < N + 5:
                x = r0 if row % 2 == 0 else 2.0 * r0
                while x <= 1.0 - r0 and len(pts) < N + 5:
                    pts.append([x, y])
                    x += 2.0 * r0
                y += math.sqrt(3.0) * r0
                row += 1
                if y > 1.0 + r0: break
                
            pts = np.array(pts[:N + 5])
            if ang != 0.0:
                c, s = np.cos(ang), np.sin(ang)
                pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
                
            mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & \
                   (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
            pts = pts[mask]
            while len(pts) < N:
                pts = np.vstack([pts, np.random.uniform(0.15, 0.85, (1, 2))])
            configs.append(pts[:N])
            
    # Random starts
    for _ in range(15):
        configs.append(np.random.uniform(0.12, 0.88, (N, 2)))
        
    # Phase 2: Simulated Annealing + Force-Directed Refinement
    for c_init in configs:
        cur_c = c_init.copy()
        cur_r = get_radii_lp(cur_c)
        cur_sum = np.sum(cur_r)
        
        best_local_c = cur_c.copy()
        best_local_r = cur_r.copy()
        best_local_sum = cur_sum
        
        T = 0.015
        step = 0.008
        
        for it in range(2000):
            noise = np.random.normal(0, step, (N, 2))
            new_c = cur_c + noise
            new_c = np.clip(new_c, 0.005, 0.995)
            
            new_r = get_radii_lp(new_c)
            new_sum = np.sum(new_r)
            
            delta = new_sum - cur_sum
            accept = delta > 0 or (np.random.rand() < math.exp(delta / max(T, 1e-5)))
            
            if accept:
                cur_c = new_c
                cur_r = new_r
                cur_sum = new_sum
                if cur_sum > best_local_sum:
                    best_local_c = cur_c.copy()
                    best_local_r = cur_r.copy()
                    best_local_sum = cur_sum
                    
            T *= 0.997
            step *= 0.9995
            
        # Force-directed polish
        fc = best_local_c.copy()
        for _ in range(400):
            fr = get_radii_lp(fc)
            forces = np.zeros_like(fc)
            
            dx = fc[:, 0][:, None] - fc[:, 0][None, :]
            dy = fc[:, 1][:, None] - fc[:, 1][None, :]
            dists = np.hypot(dx, dy)
            
            tight = np.maximum(0.0, fr[TRIL_IDX[0]] + fr[TRIL_IDX[1]] - dists[TRIL_IDX])
            
            for k in range(len(TRIL_IDX[0])):
                i, j = TRIL_IDX[0][k], TRIL_IDX[1][k]
                if tight[k] > 1e-7:
                    f = tight[k] * 0.6
                    d = dists[i, j]
                    if d > 1e-6:
                        fx, fy = dx[i, j], dy[i, j]
                        forces[i] -= f * np.array([fx, fy]) / d
                        forces[j] += f * np.array([fx, fy]) / d
                        
            for i in range(N):
                for dim in range(2):
                    if fc[i, dim] < fr[i] + 0.002: forces[i, dim] += 0.15
                    elif fc[i, dim] > 1.0 - fr[i] - 0.002: forces[i, dim] -= 0.15
                    
            fc += forces * 0.015
            fc = np.clip(fc, 0.005, 0.995)
            
        fc_r = get_radii_lp(fc)
        fc_sum = np.sum(fc_r)
        if fc_sum > best_local_sum:
            best_local_c = fc.copy()
            best_local_r = fc_r.copy()
            best_local_sum = fc_sum
            
        if best_local_sum > best_sum:
            best_sum = best_local_sum
            best_centers = best_local_c.copy()
            best_radii = best_local_r.copy()
            
    # Phase 3: SLSQP Polish
    x0 = np.zeros(3 * N)
    x0[0::3] = best_centers[:, 0]
    x0[1::3] = best_centers[:, 1]
    x0[2::3] = best_radii
    
    cons = {'type': 'ineq', 'fun': constraints_joint}
    bounds = get_bounds_joint()
    
    try:
        res = minimize(objective_joint, x0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': 10000, 'ftol': 1e-14, 'disp': False})
        if not np.isnan(res.fun):
            c_opt = np.column_stack((res.x[0::3], res.x[1::3]))
            r_opt = res.x[2::3]
            # Re-solve LP for strict feasibility after SLSQP
            r_lp = get_radii_lp(c_opt)
            if r_lp is not None and np.sum(r_lp) > best_sum:
                best_centers = c_opt
                best_radii = r_lp
                best_sum = np.sum(r_lp)
    except Exception:
        pass

    # Phase 4: Final validation & minimal numerical repair
    centers = best_centers.copy()
    radii = best_radii.copy()
    
    for _ in range(200):
        valid = True
        for i in range(N):
            if radii[i] < 0 or centers[i, 0] < radii[i] - 1e-10 or \
               centers[i, 0] > 1.0 - radii[i] + 1e-10 or \
               centers[i, 1] < radii[i] - 1e-10 or \
               centers[i, 1] > 1.0 - radii[i] + 1e-10:
                valid = False
                break
        if valid:
            for i in range(N):
                for j in range(i + 1, N):
                    d = np.hypot(centers[i, 0] - centers[j, 0], 
                                 centers[i, 1] - centers[j, 1])
                    if d < radii[i] + radii[j] - 1e-10:
                        valid = False
                        break
                if not valid: break
        if valid: break
            
        radii *= 0.9998
        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)
        
    return centers, radii, float(np.sum(radii))
