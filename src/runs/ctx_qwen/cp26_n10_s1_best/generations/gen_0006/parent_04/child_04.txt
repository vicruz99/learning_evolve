1. **LP for Radii, SLSQP for Centers**: For fixed centers, the optimal radii are given by a Linear Program (maximizing sum of radii subject to boundary and pairwise distance constraints). This is exact and fast. SLSQP is then used to jointly optimize centers and radii, leveraging gradient information.
2. **Diverse Topologies**: Initial configurations should cover hexagonal lattices (rotated and scaled), force-directed layouts, and corner-biased placements to capture different packing topologies.
3. **Iterative Deflation & Perturbation**: To escape local minima, we repeatedly shrink radii slightly, perturb centers, and re-optimize. This allows circles to "slide" past each other into denser arrangements.
4. **Subset Perturbation**: Perturbing only a random subset of circles per iteration reduces the chance of destructive global moves and increases the probability of finding incremental improvements.
5. **Strict Feasibility Enforcement**: A final repair loop ensures the solution strictly satisfies the `1e-12` tolerance of the validator by minimally shrinking radii if numerical drift occurs.

import numpy as np
from scipy.optimize import minimize, linprog

N_CIRCLES = 26
TRIL_IDX = np.tril_indices(N_CIRCLES, -1)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def get_constraints(x):
    """Vectorized inequality constraints: g(x) >= 0."""
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    # Boundary constraints
    c = np.concatenate([
        cx - r,
        1.0 - cx - r,
        cy - r,
        1.0 - cy - r
    ])
    
    # Overlap constraints: dist^2 >= (ri + rj)^2
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dr = r[:, None] + r[None, :]
    
    c = np.concatenate([c, dx[TRIL_IDX]**2 + dy[TRIL_IDX]**2 - dr[TRIL_IDX]**2])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-6, 0.5]."""
    b = []
    for _ in range(N_CIRCLES):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)])
    return b

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N_CIRCLES
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A = np.zeros((num_ineq, n))
    b = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i,0], 1.0-centers[i,0], centers[i,1], 1.0-centers[i,1])
        b[idx] = max(0.0, lim)
        A[idx, i] = 1.0
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i+1, n):
            d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
            b[idx] = max(0.0, d)
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            idx += 1
            
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0.0, None)]*n, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def project_to_feasible(x0):
    """Ensure optimization vector strictly respects bounds and basic geometry."""
    x0 = x0.copy()
    for i in range(N_CIRCLES):
        r = max(1e-6, x0[3*i+2])
        x0[3*i] = np.clip(x0[3*i], r, 1.0-r)
        x0[3*i+1] = np.clip(x0[3*i+1], r, 1.0-r)
        x0[3*i+2] = r
    return x0

def force_init(seed):
    """Force-directed layout to spread points evenly."""
    np.random.seed(seed)
    pts = np.random.uniform(0.1, 0.9, (N_CIRCLES, 2))
    for _ in range(1200):
        f = np.zeros_like(pts)
        for i in range(N_CIRCLES):
            for j in range(i+1, N_CIRCLES):
                dx = pts[j] - pts[i]
                d = np.hypot(dx[0], dx[1])
                if d < 0.28 and d > 1e-5:
                    rep = 0.025 / (d**2 + 0.001)
                    f[i] -= rep * dx
                    f[j] += rep * dx
            for dim in range(2):
                if pts[i, dim] < 0.12: f[i, dim] += 0.06
                elif pts[i, dim] > 0.88: f[i, dim] -= 0.06
        pts += f * 0.05
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def hex_init(r0, angle=0.0, seed=0):
    """Generates a hexagonal lattice initialization."""
    np.random.seed(seed)
    centers = np.zeros((N_CIRCLES, 2))
    idx = 0
    y = r0
    row = 0
    while idx < N_CIRCLES and y + r0 <= 1.0:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while idx < N_CIRCLES and x + r0 <= 1.0:
            centers[idx, 0] = x
            centers[idx, 1] = y
            idx += 1
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
    while idx < N_CIRCLES:
        centers[idx] = np.random.uniform(0.1, 0.9, 2)
        idx += 1
        
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        cx, cy = 0.5, 0.5
        dx = centers[:,0] - cx
        dy = centers[:,1] - cy
        centers[:,0] = dx*c - dy*s + cx
        centers[:,1] = dx*s + dy*c + cy
        
    return centers

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    np.random.seed(42)
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': get_constraints}
    
    best_sum = -1.0
    best_x = None
    
    # Phase 1: Diverse Initializations
    configs = []
    # Hexagonal lattices with varied scales and rotations
    for r0 in np.linspace(0.078, 0.098, 7):
        for ang in [0.0, 0.1, -0.1, 0.2, -0.2, 0.4, -0.4, 0.6]:
            configs.append(hex_init(r0, ang))
            
    # Force-directed layouts
    for s in range(15):
        configs.append(force_init(s))
        
    # Optimize each initialization
    for c_init in configs:
        r_init = solve_lp_radii(c_init)
        if r_init is None:
            r_init = np.full(N_CIRCLES, 0.06)
            
        x0 = np.zeros(3 * N_CIRCLES)
        x0[0::3] = c_init[:,0]
        x0[1::3] = c_init[:,1]
        x0[2::3] = r_init
        
        # Break symmetry slightly
        x0 += np.random.normal(0, 1e-5, x0.shape)
        x0 = project_to_feasible(x0)
        
        # SLSQP -> LP refinement cycle
        for _ in range(3):
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 25000, 'ftol': 1e-14, 'disp': False})
                if not np.isnan(res.fun):
                    c_opt = np.column_stack((res.x[0::3], res.x[1::3]))
                    r_lp = solve_lp_radii(c_opt)
                    if r_lp is not None:
                        curr_sum = np.sum(r_lp)
                        if curr_sum > best_sum:
                            best_sum = curr_sum
                            best_x = np.zeros(3 * N_CIRCLES)
                            best_x[0::3] = c_opt[:,0]
                            best_x[1::3] = c_opt[:,1]
                            best_x[2::3] = r_lp
                        # Update x0 for next cycle with LP radii
                        x0[2::3] = r_lp
                        x0 = project_to_feasible(x0)
            except Exception:
                break

    # Phase 2: Aggressive Basin Hopping / Subset Deflation
    if best_x is not None:
        for step in range(40):
            x_pert = best_x.copy()
            
            # Shrink radii to allow movement
            shrink = 0.94 + 0.005 * step
            x_pert[2::3] *= shrink
            
            # Perturb random subset of centers to escape local topology
            subset_size = max(3, N_CIRCLES // 4)
            subset_idx = np.random.choice(N_CIRCLES, size=subset_size, replace=False)
            noise = 0.004 / (step + 1)
            x_pert[3*subset_idx] += np.random.normal(0, noise, subset_size)
            x_pert[3*subset_idx + 1] += np.random.normal(0, noise, subset_size)
            
            x_pert = project_to_feasible(x_pert)
            
            try:
                res = minimize(objective, x_pert, method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 20000, 'ftol': 1e-14, 'disp': False})
                if not np.isnan(res.fun):
                    c_opt = np.column_stack((res.x[0::3], res.x[1::3]))
                    r_lp = solve_lp_radii(c_opt)
                    if r_lp is not None:
                        curr_sum = np.sum(r_lp)
                        if curr_sum > best_sum:
                            best_sum = curr_sum
                            best_x = np.zeros(3 * N_CIRCLES)
                            best_x[0::3] = c_opt[:,0]
                            best_x[1::3] = c_opt[:,1]
                            best_x[2::3] = r_lp
            except Exception:
                pass

    # Fallback (should not be reached)
    if best_x is None:
        c_fallback = hex_init(0.085, 0.0)
        r_fallback = solve_lp_radii(c_fallback)
        if r_fallback is None: r_fallback = np.full(N_CIRCLES, 0.06)
        best_x = np.zeros(3 * N_CIRCLES)
        best_x[0::3] = c_fallback[:,0]
        best_x[1::3] = c_fallback[:,1]
        best_x[2::3] = r_fallback
        best_sum = np.sum(r_fallback)
        
    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3]
    
    # Final strict validity adjustment against 1e-12 tolerance
    for _ in range(100):
        valid = True
        for i in range(N_CIRCLES):
            if radii[i] < 0: valid = False; break
            if centers[i,0] < radii[i]-1e-10 or centers[i,0] > 1.0-radii[i]+1e-10: valid = False; break
            if centers[i,1] < radii[i]-1e-10 or centers[i,1] > 1.0-radii[i]+1e-10: valid = False; break
        if valid:
            for i in range(N_CIRCLES):
                for j in range(i+1, N_CIRCLES):
                    d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
                    if d < radii[i]+radii[j]-1e-10:
                        valid = False; break
                if not valid: break
        if valid: break
        
        # Minimal shrinkage to recover strict feasibility
        radii *= 0.9998
        centers[:,0] = np.clip(centers[:,0], radii, 1.0-radii)
        centers[:,1] = np.clip(centers[:,1], radii, 1.0-radii)
        
    return centers, radii, float(np.sum(radii))
