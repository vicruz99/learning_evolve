def constraints(x):

    ys = x[1::3]
    rs = x[2::3]

    c = np.concatenate([
        xs - rs, 1.0 - xs - rs,
        ys - rs, 1.0 - ys - rs
    ])
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]

    dist_sq = dx[i_idx, j_idx]**2 + dy[i_idx, j_idx]**2
    r_sum_sq = dr[i_idx, j_idx]**2

    return c

    b = []

        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)])
    return b

    n = N

    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)

        b_ub[idx] = max(0.0, lim)
        idx += 1

            A_ub[idx, j] = 1.0

            idx += 1

        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')

    except Exception:
        pass
    return None
