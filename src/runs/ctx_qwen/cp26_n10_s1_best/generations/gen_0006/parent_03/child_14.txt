import numpy as np
from scipy.optimize import minimize, linprog

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraint_fun(x):
    """Returns all inequality constraints >= 0 (vectorized)."""
    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]
    
    # Boundary constraints: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    c = np.concatenate([
        xs - rs, 1.0 - xs - rs,
        ys - rs, 1.0 - ys - rs
    ])
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    c = np.concatenate([c, dx[TRIL_IDX]**2 + dy[TRIL_IDX]**2 - dr[TRIL_IDX]**2])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-7, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)])
    return b

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        lim = min(centers[i, 0], 1.0 - centers[i, 0], 
                  centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, lim)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], 
                            centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dist
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def force_init(seed, steps=400):
    """Generate initial configuration using repulsive force simulation."""
    np.random.seed(seed)
    pts = np.random.uniform(0.2, 0.8, (N, 2))
    for _ in range(steps):
        f = np.zeros_like(pts)
        for i in range(N):
            for j in range(i + 1, N):
                diff = pts[j] - pts[i]
                d = np.hypot(diff[0], diff[1])
                if d < 0.25 and d > 1e-5:
                    ff = 0.01 / (d**2 + 1e-4)
                    f[i] -= diff * ff / d
                    f[j] += diff * ff / d
            # Push to boundaries to utilize edge space
            for dim in range(2):
                if pts[i, dim] < 0.15: f[i, dim] += 0.06
                elif pts[i, dim] > 0.85: f[i, dim] -= 0.06
        pts += f * 0.05
        pts = np.clip(pts, 0.02, 0.98)
    return pts

def hex_init(r0, ang):
    """Generates a hexagonal lattice initialization with optional rotation."""
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0 and len(pts) < N + 10:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
    pts = np.array(pts[:N + 10])
    
    if ang != 0.0:
        c_a, s_a = np.cos(ang), np.sin(ang)
        pts = (pts - 0.5) @ np.array([[c_a, -s_a], [s_a, c_a]]) + 0.5
        
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.1, 0.9, (1, 2))])
    return pts[:N]

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    best_sum = -1.0
    best_x = None
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraint_fun}
    
    inits = []
    # Diverse initial configurations
    for s in range(15):
        inits.append(force_init(s))
    for r0 in np.linspace(0.085, 0.115, 6):
        for ang in np.linspace(-0.35, 0.35, 8):
            inits.append(hex_init(r0, ang))
            
    # Phase 1: Multi-start with LP-initialized radii
    for pts in inits:
        r_lp = solve_lp_radii(pts)
        if r_lp is None:
            r_lp = np.full(N, 0.06)
            
        x0 = np.zeros(3 * N)
        x0[0::3] = pts[:, 0]
        x0[1::3] = pts[:, 1]
        x0[2::3] = r_lp * 0.995  # Slight margin for strict feasibility
        
        # Project strictly feasible
        for i in range(N):
            r = x0[3 * i + 2]
            x0[3 * i] = np.clip(x0[3 * i], r, 1.0 - r)
            x0[3 * i + 1] = np.clip(x0[3 * i + 1], r, 1.0 - r)
            
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 30000, 'ftol': 1e-13})
            if not np.isnan(res.fun):
                s = -res.fun
                vals = constraint_fun(res.x)
                if np.min(vals) >= -1e-6 and s > best_sum:
                    best_sum = s
                    best_x = res.x.copy()
        except Exception:
            pass

    # Phase 2: Alternating LP <-> SLSQP Refinement on best
    if best_x is not None:
        cur_x = best_x.copy()
        for _ in range(8):
            # LP snap
            c_tmp = np.column_stack((cur_x[0::3], cur_x[1::3]))
            r_new = solve_lp_radii(c_tmp)
            if r_new is not None:
                cur_x[2::3] = r_new * 0.998
                # Perturb centers slightly to break symmetry/degeneracy
                cur_x[0::3] += np.random.normal(0, 1e-5, N)
                cur_x[1::3] += np.random.normal(0, 1e-5, N)
                for i in range(N):
                    r = cur_x[3 * i + 2]
                    cur_x[3 * i] = np.clip(cur_x[3 * i], r, 1.0 - r)
                    cur_x[3 * i + 1] = np.clip(cur_x[3 * i + 1], r, 1.0 - r)
                try:
                    res = minimize(objective, cur_x, method='SLSQP', bounds=bounds, constraints=cons,
                                   options={'maxiter': 15000, 'ftol': 1e-13})
                    if not np.isnan(res.fun) and -res.fun > best_sum:
                        best_x = res.x.copy()
                        best_sum = -res.fun
                        cur_x = best_x.copy()
                except Exception:
                    break
            else:
                break
                
        # Phase 3: Aggressive perturbation to escape local minima
        cur_x = best_x.copy()
        for step in range(40):
            noise = 0.003 * (0.92 ** step)
            x_p = cur_x.copy()
            x_p[2::3] *= 0.975  # Deflate radii to allow center movement
            x_p[0::3] += np.random.normal(0, noise, N)
            x_p[1::3] += np.random.normal(0, noise, N)
            
            for i in range(N):
                r = max(0.01, x_p[3 * i + 2])
                x_p[3 * i] = np.clip(x_p[3 * i], r, 1.0 - r)
                x_p[3 * i + 1] = np.clip(x_p[3 * i + 1], r, 1.0 - r)
                
            try:
                res = minimize(objective, x_p, method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 10000, 'ftol': 1e-13})
                if not np.isnan(res.fun):
                    s = -res.fun
                    vals = constraint_fun(res.x)
                    if np.min(vals) >= -1e-6 and s > best_sum:
                        best_x = res.x.copy()
                        best_sum = s
                        cur_x = best_x.copy()
            except Exception:
                pass

    # Fallback (should not be reached)
    if best_x is None:
        best_x = np.zeros(3 * N)
        best_x[0::3] = np.tile(np.linspace(0.12, 0.88, 5), 6)[:N]
        best_x[1::3] = np.repeat(np.linspace(0.12, 0.88, 6), 5)[:N]
        best_x[2::3] = 0.08

    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3]
    
    # Final strict validation repair to guarantee compliance with 1e-12 tolerance
    for _ in range(100):
        valid = True
        for i in range(N):
            if radii[i] < 0 or centers[i, 0] < radii[i] - 1e-9 or centers[i, 0] > 1.0 - radii[i] + 1e-9 or \
               centers[i, 1] < radii[i] - 1e-9 or centers[i, 1] > 1.0 - radii[i] + 1e-9:
                valid = False
                break
        if valid:
            for i in range(N):
                for j in range(i + 1, N):
                    d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                    if d < radii[i] + radii[j] - 1e-9:
                        valid = False
                        break
                if not valid:
                    break
        if valid:
            break
        radii *= 0.9995
        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)
        
    return centers, radii, float(np.sum(radii))
