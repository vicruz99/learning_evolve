import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraint_fun(x):
    """Returns all inequality constraints >= 0 (vectorized)."""
    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]
    
    # Boundary constraints: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    c = np.concatenate([xs - rs, 1.0 - xs - rs, ys - rs, 1.0 - ys - rs])
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    i, j = np.tril_indices(N, -1)
    c = np.concatenate([c, dx[i, j]**2 + dy[i, j]**2 - dr[i, j]**2])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [0, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
    return b

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        bound = min(centers[i, 0], 1.0 - centers[i, 0], 
                    centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, bound)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], 
                            centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist)
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def generate_hex_init(r0, angle, seed):
    """Generate a rotated and perturbed hexagonal lattice."""
    np.random.seed(seed)
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 10:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0 and len(pts) < N + 10:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        
    pts = np.array(pts[:N + 10])
    if angle != 0.0:
        c_a, s_a = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c_a, -s_a], [s_a, c_a]]) + 0.5
        
    pts += np.random.normal(0, 0.005, pts.shape)
    mask = (pts[:, 0] >= 0.02) & (pts[:, 0] <= 0.98) & (pts[:, 1] >= 0.02) & (pts[:, 1] <= 0.98)
    pts = pts[mask]
    while len(pts) < N:
        pts = np.vstack([pts, np.random.uniform(0.1, 0.9, (1, 2))])
    return pts[:N]

def force_directed_init(seed):
    """Generate initial configuration using repulsive force simulation."""
    np.random.seed(seed)
    pts = np.random.uniform(0.15, 0.85, (N, 2))
    for _ in range(500):
        f = np.zeros_like(pts)
        for i in range(N):
            for j in range(i + 1, N):
                diff = pts[j] - pts[i]
                d = np.hypot(diff[0], diff[1])
                if d < 0.25 and d > 1e-5:
                    force = 0.008 / (d**2 + 0.001)
                    f[i] -= diff * force / d
                    f[j] += diff * force / d
            # Boundary repulsion
            if pts[i, 0] < 0.12: f[i, 0] += 0.02
            elif pts[i, 0] > 0.88: f[i, 0] -= 0.02
            if pts[i, 1] < 0.12: f[i, 1] += 0.02
            elif pts[i, 1] > 0.88: f[i, 1] -= 0.02
        pts += f * 0.05
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def project_to_feasible(x0):
    """Project variables to strictly satisfy bounds and non-negative radii."""
    x0 = x0.copy()
    for i in range(N):
        r = max(0.005, x0[3 * i + 2])
        x0[3 * i + 2] = r
        x0[3 * i] = np.clip(x0[3 * i], r, 1.0 - r)
        x0[3 * i + 1] = np.clip(x0[3 * i + 1], r, 1.0 - r)
    return x0

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    np.random.seed(42)
    best_sum = -1.0
    best_x = None
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraint_fun}
    
    # Phase 1: Generate diverse initial configurations
    inits = []
    # Hexagonal lattices with varied scales, rotations, and perturbations
    for r0 in np.linspace(0.08, 0.11, 6):
        for ang in np.linspace(-0.3, 0.3, 9):
            for s in range(4):
                inits.append(generate_hex_init(r0, ang, s))
    # Force-directed spreads
    for s in range(8):
        inits.append(force_directed_init(s))
        
    # Phase 2: Multi-start Optimization
    for pts in inits:
        x0 = np.zeros(3 * N)
        x0[0::3] = pts[:, 0]
        x0[1::3] = pts[:, 1]
        
        # Initialize radii via LP for maximum head-start
        r_lp = solve_lp_radii(pts)
        if r_lp is not None:
            x0[2::3] = r_lp * 0.995
        else:
            x0[2::3] = 0.075
            
        x0 = project_to_feasible(x0)
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 25000, 'ftol': 1e-14, 'disp': False})
            if not np.isnan(res.fun):
                s_val = -res.fun
                c_vals = constraint_fun(res.x)
                if np.min(c_vals) >= -1e-6 and s_val > best_sum:
                    best_sum = s_val
                    best_x = res.x.copy()
        except Exception:
            pass
            
    # Phase 3: Alternating Refinement with Deflation
    if best_x is not None:
        for round_idx in range(30):
            x0 = best_x.copy()
            noise = 0.0015 / (round_idx + 1)
            x0 += np.random.normal(0, noise, 3 * N)
            
            # Deflation: shrink a subset of radii to break local lattice structures
            subset = np.random.choice(N, size=N // 5, replace=False)
            rad_indices = np.arange(2, 3 * N, 3)[subset]
            x0[rad_indices] *= 0.85
            x0 = project_to_feasible(x0)
            
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 15000, 'ftol': 1e-14, 'disp': False})
                if not np.isnan(res.fun):
                    s_val = -res.fun
                    if s_val > best_sum:
                        best_sum = s_val
                        best_x = res.x.copy()
            except Exception:
                pass
                
            # LP update: optimally scale radii for current centers
            c_tmp = np.column_stack((best_x[0::3], best_x[1::3]))
            r_new = solve_lp_radii(c_tmp)
            if r_new is not None:
                best_x[2::3] = r_new
                best_sum = np.sum(r_new)
                
                # Re-optimize positions with fixed optimal radii (allows centers to slide into better positions)

                x0_fix[0::3] += np.random.normal(0, 1e-5, N)
                x0_fix[1::3] += np.random.normal(0, 1e-5, N)
                x0_fix = project_to_feasible(x0_fix)
                try:
                    res = minimize(objective, x0_fix, method='SLSQP', bounds=bounds, constraints=cons,
                                   options={'maxiter': 10000, 'ftol': 1e-14, 'disp': False})
                    if not np.isnan(res.fun) and -res.fun > best_sum:
                        best_x = res.x.copy()
                        best_sum = -res.fun
                except Exception:
                    pass

    # Fallback (should rarely be reached)
    if best_x is None:
        best_x = np.zeros(3 * N)
        best_x[0::3] = np.tile(np.linspace(0.1, 0.9, 5), 6)[:N]

        best_x[2::3] = 0.08
        
    centers = np.column_stack((best_x[0::3], best_x[1::3]))

    # Final strict validation repair against 1e-12 tolerance
    for _ in range(100):
        valid = True
        for i in range(N):

                for j in range(i + 1, N):
                    d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                    if d < radii[i] + radii[j] - 1e-9:

                        break

            break

        radii *= 0.9995
        for i in range(N):
            centers[i, 0] = np.clip(centers[i, 0], radii[i], 1.0 - radii[i])
            centers[i, 1] = np.clip(centers[i, 1], radii[i], 1.0 - radii[i])
            
    return centers, radii, float(np.sum(radii))
