import numpy as np
from scipy.optimize import minimize, linprog

N = 26
TRIL_IDX = np.tril_indices(N, -1)

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraint_fun(x):
    """Returns all inequality constraints g(x) >= 0 (vectorized)."""
    xs = x[0::3]
    ys = x[1::3]
    rs = x[2::3]
    
    # Boundary constraints
    c = np.concatenate([
        xs - rs, 1.0 - xs - rs,
        ys - rs, 1.0 - ys - rs
    ])
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    c = np.concatenate([c, dx[TRIL_IDX]**2 + dy[TRIL_IDX]**2 - dr[TRIL_IDX]**2])
    return c

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-8, 0.5]."""
    b = []
    for _ in range(N):
        b.extend([(0.0, 1.0), (0.0, 1.0), (1e-8, 0.5)])
    return b

def solve_lp_radii(centers):
    """Optimally compute radii for fixed centers using Linear Programming."""
    n = N
    c_obj = -np.ones(n)
    num_ineq = n + n * (n - 1) // 2
    A_ub = np.zeros((num_ineq, n))
    b_ub = np.zeros(num_ineq)
    
    idx = 0
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        bound = min(centers[i, 0], 1.0 - centers[i, 0], 
                    centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 1.0
        b_ub[idx] = max(0.0, bound)
        idx += 1
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], 
                            centers[i, 1] - centers[j, 1])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = max(0.0, dist)
            idx += 1
            
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 1e-8)
    except Exception:
        pass
    return None

def generate_hex_init(r0, angle, seed):
    """Hexagonal lattice with rotation and perturbation."""
    np.random.seed(seed)
    pts = []
    y = r0
    row = 0
    while len(pts) < N + 5:
        x = r0 if row % 2 == 0 else 2.0 * r0
        while x <= 1.0 - r0 and len(pts) < N + 5:
            pts.append([x, y])
            x += 2.0 * r0
        y += np.sqrt(3.0) * r0
        row += 1
        
    pts = np.array(pts[:N])
    if angle != 0.0:
        c, s = np.cos(angle), np.sin(angle)
        pts = (pts - 0.5) @ np.array([[c, -s], [s, c]]) + 0.5
    pts += np.random.normal(0, 0.001, pts.shape)
    return np.clip(pts, 0.01, 0.99)

def generate_grid_init(seed):
    """Structured grid initialization."""
    np.random.seed(seed)
    pts = []
    for r in range(6):
        for c in range(5):
            if len(pts) >= N: break
            x = 0.06 + c * 0.18
            y = 0.06 + r * 0.16
            pts.append([x, y])
    pts = np.array(pts[:N]) + np.random.normal(0, 0.005, (N, 2))
    return np.clip(pts, 0.02, 0.98)

def generate_force_init(seed):
    """Force-directed spread from random points."""
    np.random.seed(seed)
    pts = np.random.uniform(0.2, 0.8, (N, 2))
    for _ in range(250):
        f = np.zeros_like(pts)
        for i in range(N):
            for j in range(i + 1, N):
                diff = pts[j] - pts[i]
                d = np.linalg.norm(diff)
                if d < 0.25 and d > 1e-6:
                    force = 0.0008 / (d**2 + 1e-5)
                    f[i] -= diff * force / d
                    f[j] += diff * force / d
        pts += f * 0.05
        pts = np.clip(pts, 0.05, 0.95)
    return pts

def project_to_feasible(x0):
    """Ensure strict feasibility for optimizer stability."""
    x = x0.copy()
    for i in range(N):
        r = max(1e-8, x[3 * i + 2])
        x[3 * i] = np.clip(x[3 * i], r, 1.0 - r)
        x[3 * i + 1] = np.clip(x[3 * i + 1], r, 1.0 - r)
        x[3 * i + 2] = r
    return x

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    best_sum = -1.0
    best_x = None
    bounds = get_bounds()
    cons = {'type': 'ineq', 'fun': constraint_fun}

        inits.append(generate_grid_init(s))
        
    # 3. Force-Directed Spreads
    for s in range(12):

    # Phase 1: Multi-start Optimization
    for pts in inits:

        if r_lp is None:
            r_lp = np.full(N, 0.07)
            
        x0 = np.zeros(3 * N)
        x0[0::3] = pts[:, 0]
        x0[1::3] = pts[:, 1]
        x0[2::3] = r_lp * 0.992  # Slight deflation for strict feasibility
        x0 = project_to_feasible(x0)
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 15000, 'ftol': 1e-13, 'disp': False})
            if not np.isnan(res.fun):
                s_val = -res.fun
                if np.min(constraint_fun(res.x)) >= -1e-5 and s_val > best_sum:
                    best_sum = s_val
                    best_x = res.x.copy()
        except Exception:
            pass

    # Phase 2: Aggressive Refinement & Topology Search
    if best_x is not None:
        for cycle in range(35):
            # Adaptive noise and deflation
            noise_scale = 0.002 * (0.93 ** cycle)
            deflation = 0.985 - 0.005 * (cycle / 35.0)
            
            x0 = best_x + np.random.normal(0, noise_scale, 3 * N)
            x0[2::3] *= max(0.90, deflation)
            x0 = project_to_feasible(x0)
            
            try:
                res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,

                if not np.isnan(res.fun):
                    s_val = -res.fun
                    if np.min(constraint_fun(res.x)) >= -1e-5 and s_val > best_sum:

                    # LP snap to theoretical max radii for current centers

                            if not np.isnan(res.fun) and -res.fun > best_sum:
                                best_x = res.x.copy()
                                best_sum = -res.fun
                        except Exception:
                            pass
            except Exception:
                pass

    # Fallback (should not be reached)
    if best_x is None:
        best_x = np.zeros(3 * N)
        best_x[0::3] = np.tile(np.linspace(0.12, 0.88, 5), 6)[:N]
        best_x[1::3] = np.repeat(np.linspace(0.12, 0.88, 6), 5)[:N]
        best_x[2::3] = 0.08
        
    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3].copy()
    
    # Final strict validation repair against 1e-12 tolerance
    for _ in range(100):
        valid = True
        for i in range(N):

            centers[i, 1] = np.clip(centers[i, 1], radii[i], 1.0 - radii[i])
            
    return centers, radii, float(np.sum(radii))
