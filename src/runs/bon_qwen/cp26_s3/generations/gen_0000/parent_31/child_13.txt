import numpy as np
from scipy.optimize import minimize

def compute_penalty(x, r, N):
    """
    Computes the overlap penalty for given centers x and radius r.
    x is flattened centers array of shape (2*N,).
    """
    c = x.reshape(N, 2)
    p = 0.0
    for i in range(N):
        for j in range(i + 1, N):
            dx = c[i, 0] - c[j, 0]
            dy = c[i, 1] - c[j, 1]
            d = np.hypot(dx, dy)
            if d < 2.0 * r:
                diff = 2.0 * r - d
                p += diff * diff
    return p

def compute_grad(x, r, N):
    """
    Computes the gradient of the overlap penalty.
    """
    c = x.reshape(N, 2)
    g = np.zeros_like(c)
    for i in range(N):
        for j in range(i + 1, N):
            dx = c[i, 0] - c[j, 0]
            dy = c[i, 1] - c[j, 1]
            d = np.hypot(dx, dy)
            if d < 2.0 * r:
                diff = 2.0 * r - d
                if d > 1e-7:
                    factor = -2.0 * diff / d
                    g[i, 0] += factor * dx
                    g[i, 1] += factor * dy
                    g[j, 0] -= factor * dx
                    g[j, 1] -= factor * dy
                else:
                    # Fallback for singularity
                    g[i, 0] += 1e-5
                    g[j, 0] -= 1e-5
    return g.ravel()

def run_packing():
    np.random.seed(42)
    N = 26
    best_r = 0.0
    best_centers = None
    
    # Run multiple attempts with different seeds to escape local minima
    for attempt in range(4):
        np.random.seed(attempt * 123)
        centers = np.random.uniform(0.1, 0.9, (N, 2))
        r = 0.05
        
        for step in range(400):
            # Box bounds enforce boundary constraints strictly
            bounds = [(r, 1 - r)] * (2 * N)
            res = minimize(
                compute_penalty,
                centers.ravel(),
                args=(r, N),
                jac=compute_grad,
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 600, 'ftol': 1e-12, 'disp': False}
            )
            centers = res.x.reshape(N, 2)
            
            # Adaptive radius growth
            if res.fun < 1e-6:
                r *= 1.0015
            else:
                if res.fun > 1e-3:
                    r *= 0.985
                else:
                    r *= 1.0005
                    
            if r > best_r:
                best_r = r
                best_centers = centers.copy()
                
        # Early exit if we surpassed the target threshold
        if best_r > 0.1015:
            break
            
    radii = np.full(N, best_r)
    # Final strict clipping to guarantee boundary compliance
    best_centers = np.clip(best_centers, best_r, 1 - best_r)
    return best_centers, radii, float(np.sum(radii))
