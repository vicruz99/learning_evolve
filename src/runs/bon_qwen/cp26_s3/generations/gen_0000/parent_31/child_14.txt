import numpy as np
from scipy.optimize import linprog
import math

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # Initialize centers in a grid pattern
    # 6 columns, 5 rows -> 30 points. We need 26.
    # Using linspace to create a uniform grid inside [0,1]
    # We offset slightly from edges to allow room for growth
    xs = np.linspace(0.1, 0.9, 6)
    ys = np.linspace(0.1, 0.9, 5)
    
    grid_points = []
    for y in ys:
        for x in xs:
            grid_points.append([x, y])
            
    # Take the first 26 points
    centers = np.array(grid_points[:n])
    
    # Precompute LP matrix structure (A_ub)
    # Variables: r_0, ..., r_25
    # Constraints:
    # 1. Boundary: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i (4 per circle)
    # 2. Pairs: r_i + r_j <= dist(i, j) (1 per pair)
    
    rows = []
    
    # Boundary constraints coefficients (Identity matrix essentially)
    for i in range(n):
        row = np.zeros(n)
        row[i] = 1.0
        rows.append(row) # r_i <= x_i
        rows.append(row.copy()) # r_i <= 1-x_i
        rows.append(row.copy()) # r_i <= y_i
        rows.append(row.copy()) # r_i <= 1-y_i
        
    # Pair constraints coefficients
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            rows.append(row)
            
    A_ub = np.array(rows)
    c_obj = np.ones(n) * -1.0 # Maximize sum(r) => Minimize -sum(r)
    bounds = [(0, None)] * n
    
    current_centers = centers.copy()
    best_sum_r = 0.0
    best_centers = current_centers.copy()
    best_radii = np.zeros(n)
    
    num_iters = 2000
    step_size = 0.015 # Initial step size for center movement
    
    # Optimization Loop
    for it in range(num_iters):
        # 1. Construct RHS (b_ub) based on current centers
        b_ub = np.zeros(4 * n + (n * (n - 1) // 2))
        
        idx = 0
        for i in range(n):
            x, y = current_centers[i]
            b_ub[idx] = x; idx += 1
            b_ub[idx] = 1.0 - x; idx += 1
            b_ub[idx] = y; idx += 1
            b_ub[idx] = 1.0 - y; idx += 1
            
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt((current_centers[i][0] - current_centers[j][0])**2 + 
                               (current_centers[i][1] - current_centers[j][1])**2)
                b_ub[idx] = dist
                idx += 1
                
        # 2. Solve LP
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
            if res.success:
                radii = res.x
                sum_r = np.sum(radii)
                
                if sum_r > best_sum_r:
                    best_sum_r = sum_r
                    best_centers = current_centers.copy()
                    best_radii = radii.copy()
                
                # 3. Calculate Forces based on active constraints
                forces = np.zeros((n, 2))
                threshold = 1e-4 # Constraint considered active if slack < threshold
                
                # Boundary Forces
                for i in range(n):
                    x, y = current_centers[i]
                    r = radii[i]
                    
                    if x - r < threshold: forces[i, 0] += 1.0
                    if (1.0 - x) - r < threshold: forces[i, 0] -= 1.0
                    if y - r < threshold: forces[i, 1] += 1.0
                    if (1.0 - y) - r < threshold: forces[i, 1] -= 1.0
                
                # Pair Forces
                for i in range(n):
                    for j in range(i + 1, n):
                        dist = np.sqrt((current_centers[i][0] - current_centers[j][0])**2 + 
                                       (current_centers[i][1] - current_centers[j][1])**2)
                        if dist < 1e-9: 
                            # If centers coincide, push apart randomly or arbitrarily
                            vec = np.random.rand(2) * 2 - 1
                            vec /= np.linalg.norm(vec)
                            forces[i] += vec * 5.0
                            forces[j] -= vec * 5.0
                        else:
                            r_sum = radii[i] + radii[j]
                            if dist - r_sum < threshold:
                                # Repulsive force along the line connecting centers
                                vec = (current_centers[i] - current_centers[j]) / dist
                                # Force magnitude could be adaptive, but constant works for heuristic
                                forces[i] += vec
                                forces[j] -= vec
                
                # 4. Update Centers
                # Decay step size slightly to stabilize
                current_step = step_size * (1.0 / (1.0 + it * 0.001))
                
                current_centers += current_step * forces
                
                # Clip centers to stay strictly inside [0, 1] to maintain validity of LP bounds logic
                # Although LP handles boundary constraints, keeping centers in (0,1) is safer
                current_centers = np.clip(current_centers, 1e-6, 1.0 - 1e-6)
                
        except Exception:
            pass

    # Final Validation and Adjustment
    # Due to numerical issues or aggressive moves, the best solution might be slightly invalid.
    # We validate and if invalid, scale down radii slightly.
    
    final_centers = best_centers
    final_radii = best_radii
    
    if validate_packing(final_centers, final_radii):
        return final_centers, final_radii, np.sum(final_radii)
    else:
        # Try to fix by scaling radii
        # Find max overlap or boundary violation to determine scale factor
        # Simple heuristic: reduce radii by 1% until valid
        scale = 1.0
        for _ in range(100):
            scale -= 0.01
            if scale <= 0: break
            test_radii = final_radii * scale
            if validate_packing(final_centers, test_radii):
                return final_centers, test_radii, np.sum(test_radii)
        
        # Fallback: very small circles
        fallback_centers = np.random.rand(26, 2) * 0.8 + 0.1
        fallback_radii = np.ones(26) * 0.01
        return fallback_centers, fallback_radii, 0.26

# Run the packing
centers, radii, sum_radii = run_packing()
