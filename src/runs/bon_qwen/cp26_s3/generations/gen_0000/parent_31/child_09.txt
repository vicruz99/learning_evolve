# Helper function to compute the objective and penalty
def compute_loss(v, n_circles, penalty_weight):
    """
    Computes the loss function: -sum(radii) + penalty * (violations)
    v: flattened array [x1, y1, r1, x2, y2, r2, ...]
    """
    # Unpack
    centers = v[:2 * n_circles].reshape((n_circles, 2))
    radii = v[2 * n_circles:]

    # Objective: maximize sum of radii -> minimize -sum(radii)
    sum_radii = np.sum(radii)

    # 1. Overlap Penalty
    # Calculate pairwise distances
    # Centers shape: (n, 2)
    # Diff shape: (n, n, 2)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))

    # Sum of radii matrix
    r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]

    # Violation: r_i + r_j - dist_ij > 0
    # We only consider i < j to avoid double counting
    # Using upper triangle indices
    triu_idx = np.triu_indices(n_circles, k=1)
    violations_overlap = r_sum[triu_idx] - dists[triu_idx]
    overlap_penalty = np.sum(np.maximum(0, violations_overlap)**2)

    # 2. Boundary Penalty
    # Constraints:
    # x - r >= 0  => violation: r - x
    # x + r <= 1  => violation: r + x - 1
    # y - r >= 0  => violation: r - y
    # y + r <= 1  => violation: r + y - 1

    # X constraints
    v_x_in = radii - centers[:, 0]
    v_x_out = radii + centers[:, 0] - 1.0

    # Y constraints
    v_y_in = radii - centers[:, 1]
    v_y_out = radii + centers[:, 1] - 1.0

    boundary_penalty = (
        np.sum(np.maximum(0, v_x_in)**2) +
        np.sum(np.maximum(0, v_x_out)**2) +
        np.sum(np.maximum(0, v_y_in)**2) +
        np.sum(np.maximum(0, v_y_out)**2)
    )

    # Total Loss
    total_penalty = overlap_penalty + boundary_penalty
    loss = -sum_radii + penalty_weight * total_penalty

    return loss
