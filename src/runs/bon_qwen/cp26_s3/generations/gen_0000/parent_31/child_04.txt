def objective(vars, n):
    # vars is flattened [x0, y0, r0, x1, y1, r1, ...]

    r = vars[2::3]
    return -np.sum(r)

def constraint_boundary(vars, n, idx):
    # Returns value that must be >= 0
    # Constraint i: 
    # x_i >= r_i  => x_i - r_i >= 0
    # x_i <= 1 - r_i => 1 - x_i - r_i >= 0
    # y_i >= r_i  => y_i - r_i >= 0
    # y_i <= 1 - r_i => 1 - y_i - r_i >= 0
    
    x = vars[3*idx]
    y = vars[3*idx + 1]
    r = vars[3*idx + 2]
    
    return [
        x - r,
        1.0 - x - r,
        y - r,
        1.0 - y - r
    ]

def constraint_pairwise(vars, n, i, j):
    # (x_i - x_j)^2 + (y_i - y_j)^2 >= (r_i + r_j)^2
    # => (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
    
    xi, yi, ri = vars[3*i], vars[3*i+1], vars[3*i+2]
    xj, yj, rj = vars[3*j], vars[3*j+1], vars[3*j+2]
    
    dist_sq = (xi - xj)**2 + (yi - yj)**2
    sum_r_sq = (ri + rj)**2
    
    return dist_sq - sum_r_sq
