        R = vars[2*n:]
        return -np.sum(R)

        R = vars[2*n:]              # Radii (n,)
        
        # 1. Boundary Constraints
        # x >= r  => x - r >= 0
        # 1-x >= r => 1 - x - r >= 0
        # y >= r => y - r >= 0
        # 1-y >= r => 1 - y - r >= 0

        bx2 = 1.0 - C[:, 0] - R
        by1 = C[:, 1] - R
        by2 = 1.0 - C[:, 1] - R
        
        boundaries = np.concatenate([bx1, bx2, by1, by2])

        # ||Ci - Cj|| >= ri + rj  => ||Ci - Cj|| - (ri + rj) >= 0

        # diff shape (n, n, 2)
        diff = C[:, np.newaxis, :] - C[np.newaxis, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2)) # (n, n)
