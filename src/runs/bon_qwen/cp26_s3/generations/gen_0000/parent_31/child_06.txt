import numpy as np
import scipy.optimize as opt

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    Packs 26 circles in a unit square to maximize the sum of radii.
    
    Returns:
        centers: np.array of shape (26, 2) with (x, y) coordinates
        radii: np.array of shape (26) with radius of each circle
        sum_radii: float sum of all radii
    """
    n = 26
    best_sum = -np.inf
    best_centers = None
    best_radii = None

    def objective(vars):
        # vars contains x, y, r for each circle: [x1, y1, r1, x2, y2, r2, ...]
        return -np.sum(vars[2::3]) # Maximize sum of radii

    def constraints(vars):
        cons = []
        # Boundary constraints: x >= r, x <= 1-r, y >= r, y <= 1-r
        for i in range(n):
            idx = i * 3
            x, y, r = vars[idx:idx+3]
            cons.append(x - r)       # x >= r
            cons.append(x - (1 - r)) # x <= 1-r
            cons.append(y - r)       # y >= r
            cons.append(y - (1 - r)) # y <= 1-r
            cons.append(r)           # r >= 0
        
        # Overlap constraints: dist(i, j) >= r_i + r_j
        for i in range(n):
            for j in range(i + 1, n):
                idx_i = i * 3
                idx_j = j * 3
                x_i, y_i, r_i = vars[idx_i:idx_i+3]
                x_j, y_j, r_j = vars[idx_j:idx_j+3]
                dist = np.sqrt((x_i - x_j)**2 + (y_i - y_j)**2)
                cons.append(dist - (r_i + r_j))
        return cons

        """Generates a hexagonal grid of 26 points."""
        centers = np.zeros((n, 2))
        # 4 rows: 7, 6, 7, 6
        row_counts = [7, 6, 7, 6]
        idx = 0
        y_spacing = 0.15
        x_spacing = 0.15
        y_offset = 0.1
        x_offset = 0.1
        
        for r_idx, count in enumerate(row_counts):
            y = y_offset + r_idx * y_spacing
            for c_idx in range(count):
                x = x_offset + c_idx * x_spacing
                centers[idx] = [x, y]
                idx += 1
        return centers

        """Generates a random set of points."""
        return np.random.rand(n, 2) * 0.8 + 0.1

                 [(0.0, 0.5) for _ in range(n)]

        centers = np.clip(centers, radii, 1 - radii)

    centers = generate_grid()

    return centers, radii, float(np.sum(radii))

# Run the packing function
