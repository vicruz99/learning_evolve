    penalties = [500, 2000, 8000, 20000]
    
    for phase in range(num_phases):

            wrapper,
            x_opt,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 800, 'ftol': 1e-15, 'gtol': 1e-10}
        )
        x_opt = result.x
    
    return x_opt


def phase2_optimize(x0, n, constraints, bounds, objective):
    result = minimize(
        objective,
        x0,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 3000, 'ftol': 1e-14, 'disp': False}
    )
    return result.x
