def compute_centers_radii(p, n):
    centers = p[:2 * n].reshape(n, 2)
    radii = p[2 * n:]
    return centers, radii


def boundary_constraint_x_low(p, i, n):

    return centers[i, 0] - radii[i]


def boundary_constraint_x_high(p, i, n):
    centers, radii = compute_centers_radii(p, n)
    return 1 - centers[i, 0] - radii[i]


def boundary_constraint_y_low(p, i, n):
    centers, radii = compute_centers_radii(p, n)
    return centers[i, 1] - radii[i]


def boundary_constraint_y_high(p, i, n):
    centers, radii = compute_centers_radii(p, n)
    return 1 - centers[i, 1] - radii[i]


def non_overlap_constraint(p, i, j, n):
    centers, radii = compute_centers_radii(p, n)
    dx = centers[i, 0] - centers[j, 0]
    dy = centers[i, 1] - centers[j, 1]
    dist = np.sqrt(dx * dx + dy * dy)
    return dist - radii[i] - radii[j]


def objective_func(p, n):
    centers, radii = compute_centers_radii(p, n)
    return -np.sum(radii)

        r = radii[i]
        if x - r < 0:
            total_violation += (r - x) ** 2
        if x + r > 1:
            total_violation += (x + r - 1) ** 2
        if y - r < 0:
            total_violation += (r - y) ** 2
        if y + r > 1:
            total_violation += (y + r - 1) ** 2
