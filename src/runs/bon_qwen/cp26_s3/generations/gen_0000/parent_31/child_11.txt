import numpy as np
from scipy.optimize import minimize

def objective_func(vars, n, pw):
    """
    Computes the penalized objective: -sum(radii) + penalty * (boundary_violations + overlap_violations)
    vars: 1D array of shape (3*n,) containing [x0, y0, r0, x1, y1, r1, ...]
    n: number of circles
    pw: penalty weight
    """
    centers = vars[:2*n].reshape(n, 2)
    radii = vars[2*n:]

    obj = -np.sum(radii)

    # Boundary penalties: max(0, violation)^2
    p1 = np.maximum(0, radii - centers[:, 0])**2
    p2 = np.maximum(0, radii - centers[:, 1])**2
    p3 = np.maximum(0, centers[:, 0] + radii - 1)**2
    p4 = np.maximum(0, centers[:, 1] + radii - 1)**2
    b_pen = np.sum(p1 + p2 + p3 + p4)

    # Overlap penalties
    dx = centers[:, 0, np.newaxis] - centers[:, 0]
    dy = centers[:, 1, np.newaxis] - centers[:, 1]
    dist = np.sqrt(dx**2 + dy**2)
    r_sum = radii[:, np.newaxis] + radii
    
    i_idx, j_idx = np.triu_indices(n, k=1)
    dist_u = dist[i_idx, j_idx]
    r_sum_u = r_sum[i_idx, j_idx]
    o_pen = np.sum(np.maximum(0, r_sum_u - dist_u)**2)

    return obj + pw * (b_pen + o_pen)

def run_packing():
    n = 26
    best_val = np.inf
    best_vars = None
    np.random.seed(42)

    # Generate diverse initial guesses
    x0_list = []
    for _ in range(25):
        # Random positions in central region, small radii
        pos = np.random.uniform(0.2, 0.8, (n, 2))
        rad = np.full(n, 0.06)
        x0_list.append(np.concatenate([pos.flatten(), rad]))

    for _ in range(15):
        # Grid-like initialization with jitter
        grid_x = np.random.uniform(0.1, 0.9, (5, 5)).flatten()[:n]
        grid_y = np.random.uniform(0.1, 0.9, (5, 5)).flatten()[:n]
        pos = np.column_stack((grid_x, grid_y))
        rad = np.full(n, 0.08)
        x0_list.append(np.concatenate([pos.flatten(), rad]))

    # Continuation strategy: gradually increase penalty weight
    penalty_weights = [50.0, 500.0, 5000.0, 20000.0]

    for x0 in x0_list:
        curr = x0.copy()
        for pw in penalty_weights:
            bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n
            res = minimize(objective_func, curr, args=(n, pw), method='L-BFGS-B', 
                           bounds=bounds, options={'maxiter': 500, 'ftol': 1e-12})
            curr = res.x.copy()

        # Evaluate with a high penalty to gauge feasibility of the result
        val = objective_func(curr, n, 1e6)
        if val < best_val:
            best_val = val
            best_vars = curr.copy()

    centers = best_vars[:2*n].reshape(n, 2)
    init_radii = best_vars[2*n:]

    # Post-processing to guarantee strict feasibility per validation tolerance
    scale = 1.0
    for _ in range(100):
        curr_radii = init_radii * scale
        valid = True
        
        # Boundary check
        if np.any(curr_radii - centers[:, 0] > 1e-10) or np.any(curr_radii - centers[:, 1] > 1e-10):
            valid = False
        if np.any(centers[:, 0] + curr_radii - 1 > 1e-10) or np.any(centers[:, 1] + curr_radii - 1 > 1e-10):
            valid = False
            
        if valid:
            # Overlap check
            dx = centers[:, 0, np.newaxis] - centers[:, 0]
            dy = centers[:, 1, np.newaxis] - centers[:, 1]
            dist = np.sqrt(dx**2 + dy**2)
            r_sum = curr_radii[:, np.newaxis] + curr_radii
            i_idx, j_idx = np.triu_indices(n, k=1)
            if np.any(r_sum[i_idx, j_idx] - dist[i_idx, j_idx] > 1e-10):
                valid = False
                
        if valid:
            break
        scale *= 0.99
        
    final_radii = init_radii * scale
    return centers, final_radii, np.sum(final_radii)
