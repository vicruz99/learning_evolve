def compute_objective_and_penalty(vars, penalty_weight):
    """
    Computes the objective (negative sum of radii) plus penalty for constraint violations.
    vars: [x1, y1, r1, x2, y2, r2, ...]
    """
    centers = vars.reshape((N, 3))[:, :2] # x, y
    radii = vars.reshape((N, 3))[:, 2]   # r
    
    sum_r = np.sum(radii)
    objective = -sum_r
    
    penalty = 0.0
    
    # Boundary penalties
    # r <= x => x - r >= 0. Violation if r > x. Penalty (r-x)^2
    for i in range(N):
        x, y, r = centers[i, 0], centers[i, 1], radii[i]
        
        # Left
        if r > x: penalty += (r - x)**2
        # Right
        if r > 1 - x: penalty += (r - (1 - x))**2
        # Bottom
        if r > y: penalty += (r - y)**2
        # Top
        if r > 1 - y: penalty += (r - (1 - y))**2
        
    # Overlap penalties
    # dist >= r1 + r2. Violation if dist < r1 + r2.
    # Penalty (r1 + r2 - dist)^2
    # Vectorized computation might be faster but N=26 is small.
    for i in range(N):
        for j in range(i + 1, N):
            dist = np.sqrt((centers[i,0]-centers[j,0])**2 + (centers[i,1]-centers[j,1])**2)
            sum_r_ij = radii[i] + radii[j]
            if sum_r_ij > dist:
                penalty += (sum_r_ij - dist)**2
                
    return objective + penalty_weight * penalty
