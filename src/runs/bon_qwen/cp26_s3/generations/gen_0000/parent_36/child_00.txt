import numpy as np
from scipy.optimize import minimize

def get_constraints_and_jac(v, n):
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    
    n_pair = n * (n - 1) // 2
    n_cons = 4 * n + n_pair
    vals = np.empty(n_cons)
    
    # Boundary constraints: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    vals[:n] = x - r
    vals[n:2*n] = 1.0 - x - r
    vals[2*n:3*n] = y - r
    vals[3*n:4*n] = 1.0 - y - r
    
    # Pairwise constraints: dist^2 - (r_i + r_j)^2 >= 0
    idx = np.triu_indices(n, k=1)
    i_idxs, j_idxs = idx
    dx = x[i_idxs] - x[j_idxs]
    dy = y[i_idxs] - y[j_idxs]
    rs = r[i_idxs] + r[j_idxs]
    vals[4*n:] = dx**2 + dy**2 - rs**2
    
    # Jacobian matrix
    jac = np.zeros((n_cons, 3*n))
    
    # Boundary jacobians
    jac[:n, 0::3] = 1.0
    jac[:n, 2::3] = -1.0
    jac[n:2*n, 0::3] = -1.0
    jac[n:2*n, 2::3] = -1.0
    jac[2*n:3*n, 1::3] = 1.0
    jac[2*n:3*n, 2::3] = -1.0
    jac[3*n:4*n, 1::3] = -1.0
    jac[3*n:4*n, 2::3] = -1.0
    
    # Pairwise jacobians
    c_idx = 4*n
    for k in range(n_pair):
        i = i_idxs[k]
        j = j_idxs[k]
        rs_k = rs[k]
        
        jac[c_idx, 3*i] = 2.0 * dx[k]
        jac[c_idx, 3*j] = -2.0 * dx[k]
        jac[c_idx, 3*i+1] = 2.0 * dy[k]
        jac[c_idx, 3*j+1] = -2.0 * dy[k]
        jac[c_idx, 3*i+2] = -2.0 * rs_k
        jac[c_idx, 3*j+2] = -2.0 * rs_k
        c_idx += 1
        
    return vals, jac

def cons_fun(v):
    return get_constraints_and_jac(v, 26)[0]

def cons_jac(v):
    return get_constraints_and_jac(v, 26)[1]

def obj_func(v):
    return -np.sum(v[2::3])

def run_packing():
    np.random.seed(42)
    n = 26
    
    # Initialize on a 6x5 grid
    xs = np.array([(i % 6 + 0.5) / 6.0 for i in range(n)])
    ys = np.array([(i // 6 + 0.5) / 6.0 for i in range(n)])
    rs = np.full(n, 0.06)
    
    v0 = np.zeros(3*n)
    v0[0::3] = xs
    v0[1::3] = ys
    v0[2::3] = rs
    
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)] * n + [(0.0, 0.5)] * n
    
    cons = {'type': 'ineq', 'fun': cons_fun, 'jac': cons_jac}
    
    best_res = None
    best_val = -np.inf
    
    # Multiple restarts to escape local optima
    for _ in range(5):
        v_start = v0 + np.random.randn(3*n) * 0.01
        for i in range(len(bounds)):
            v_start[i] = np.clip(v_start[i], bounds[i][0], bounds[i][1])
            
        res = minimize(obj_func, v_start, method='SLSQP', bounds=bounds, 
                       constraints=cons, options={'maxiter': 2000, 'ftol': 1e-12})
        if -res.fun > best_val:
            best_val = -res.fun
            best_res = res
            
    centers = np.column_stack((best_res.x[0::3], best_res.x[1::3]))
    radii = best_res.x[2::3]
    
    return centers, radii, np.sum(radii)
