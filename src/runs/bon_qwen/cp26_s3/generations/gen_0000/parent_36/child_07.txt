1.  **Inner Problem (Linear Programming)**: For any fixed set of circle centers, the problem of finding the largest possible radii (maximizing the sum) subject to non-overlap and boundary constraints becomes a Linear Programming (LP) problem. The constraints are linear inequalities of the form $r_i + r_j \leq \|c_i - c_j\|$ and $r_i \leq \min(x_i, 1-x_i, y_i, 1-y_i)$. This can be solved efficiently using `scipy.optimize.linprog`.

2.  **Outer Problem (Global Optimization)**: We need to find the optimal center coordinates $c_1, \dots, c_{26}$ that maximize the objective function returned by the inner LP.     *   **Local Optimization**: Use `scipy.optimize.minimize` with the `Nelder-Mead` method (derivative-free) to refine the centers. Since the LP solver handles the constraints strictly, the optimizer can explore the space of centers, and the LP solver will adapt the radii to maximize the sum for each configuration.

import numpy as np
from scipy.optimize import minimize, linprog

def solve_radii_lp(centers):
    """
    Given a set of centers, solve the Linear Programming problem to find
    radii that maximize sum(r_i) subject to non-overlap and boundary constraints.
    
    Args:
        centers: np.array of shape (n, 2)
        
    Returns:
        radii: np.array of shape (n) with optimal radii
        sum_radii: float, the sum of radii
        success: bool, whether LP was solved successfully
    """
    n = centers.shape[0]
    
    # Objective: Maximize sum(r_i) -> Minimize -sum(r_i)
    c = -np.ones(n)
    
    # Constraints: A_ub * r <= b_ub
    # 1. Pairwise non-overlap: r_i + r_j <= dist(i, j)
    # 2. Boundary constraints: r_i <= x_i, r_i <= 1-x_i, etc.
    
    constraints_pairs = []
    constraints_bounds = []
    
    # Pairwise constraints
    # For each pair (i, j), r_i + r_j <= dist
    # We construct the matrix rows.
    # To avoid creating a huge list of lists, we can build arrays.
    # Number of pairs is n*(n-1)/2.
    
    num_pairs = n * (n - 1) // 2
    A_pairs = np.zeros((num_pairs, n))
    b_pairs = np.zeros(num_pairs)
    
    pair_idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            A_pairs[pair_idx, i] = 1.0
            A_pairs[pair_idx, j] = 1.0
            b_pairs[pair_idx] = dist
            pair_idx += 1
            
    # Boundary constraints
    # 4 per circle
    A_bounds = np.zeros((4 * n, n))
    b_bounds = np.zeros(4 * n)
    
    for i in range(n):
        x, y = centers[i]
        # r_i <= x
        A_bounds[4*i, i] = 1.0
        b_bounds[4*i] = x
        # r_i <= 1 - x
        A_bounds[4*i+1, i] = 1.0
        b_bounds[4*i+1] = 1.0 - x
        # r_i <= y
        A_bounds[4*i+2, i] = 1.0
        b_bounds[4*i+2] = y
        # r_i <= 1 - y
        A_bounds[4*i+3, i] = 1.0
        b_bounds[4*i+3] = 1.0 - y
        
    # Combine constraints
    A_ub = np.vstack([A_pairs, A_bounds])
    b_ub = np.concatenate([b_pairs, b_bounds])
    
    # Bounds for r_i: r_i >= 0
    # Upper bound can be 1 (or 0.5), but LP handles it via constraints.
    # linprog requires bounds as (min, max).
    bounds = [(0, 1) for _ in range(n)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            radii = res.x
            return radii, -res.fun, True
        else:
            # Fallback if LP fails (rare), return small radii
            radii = np.full(n, 1e-6)
            return radii, 26e-6, False
    except Exception:
        return np.full(n, 1e-6), 26e-6, False

def objective_function(centers_flat):
    """
    Objective function for the optimizer.
    Maximizes sum of radii by solving LP for given centers.
    """
    n = 26
    centers = centers_flat.reshape((n, 2))
    
    # Clip centers to [0, 1] to keep them valid for boundary constraints logic
    # though LP handles it, keeping them in bounds is safer for stability
    centers = np.clip(centers, 0, 1)
    
    radii, sum_r, success = solve_radii_lp(centers)
    
    # We want to minimize negative sum
    return -sum_r

def generate_hex_grid(n_circles):
    """Generate an initial configuration using a hexagonal grid layout."""
    centers = []
    # Approximate radius 0.1, spacing 0.2
    # Hex spacing vertical: 0.2 * sqrt(3)/2 ~ 0.1732
    # We try to fit 26 circles. 
    # Pattern: 5, 4, 5, 4, 5, 3 (sum 26)
    
    rows_config = [5, 4, 5, 4, 5, 3]
    
    y = 0.1
    spacing_y = 0.173205 # 0.1 * sqrt(3)
    spacing_x = 0.2
    
    for count in rows_config:
        # Center the row
        # Width of row with 'count' circles: (count-1)*spacing_x + 2*0.1? 
        # Actually centers at 0.1 + k*0.2.
        # Let's just place them starting from 0.1
        start_x = 0.1
        # If row is offset (odd index in list? No, let's alternate)
        # We decided on 5, 4, 5, 4...
        # Row 0 (5): 0.1, 0.3, 0.5, 0.7, 0.9
        # Row 1 (4): 0.2, 0.4, 0.6, 0.8 (offset by 0.1)
        # Row 2 (5): 0.1...
        
        row_idx = len(centers) // (np.mean(rows_config)) # Rough check
        # Actually simpler:
        # If count is odd (5), start at 0.1
        # If count is even (4, 3), start at 0.2?
        # 3 circles: 0.2, 0.4, 0.6? No, width is 0.4. 0.1 to 0.9 is 0.8.
        # 3 circles at 0.3, 0.5, 0.7 fits well.
        
        if count == 5:
            xs = [0.1, 0.3, 0.5, 0.7, 0.9]
        elif count == 4:
            xs = [0.2, 0.4, 0.6, 0.8]
        elif count == 3:
            xs = [0.3, 0.5, 0.7]
        else:
            xs = [0.1 + i*0.2 for i in range(count)]
            
        for x in xs:
            centers.append([x, y])
        y += spacing_y
        
    return np.array(centers[:n_circles])

def generate_random_grid(n_circles):
    """Generate random points with some spacing."""
    centers = np.random.rand(n_circles, 2)
    return centers

def run_packing():
    n_circles = 26
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Strategy: Run multiple local optimizations from different starts
    
    starts = []
    
    # 1. Hexagonal grid
    starts.append(generate_hex_grid(n_circles))
    
    # 2. Square grid (6x5 grid, pick first 26)
    grid_centers = []
    y = 0.1
    while len(grid_centers) < n_circles:
        x = 0.1
        while x <= 0.9 and len(grid_centers) < n_circles:
            grid_centers.append([x, y])
            x += 0.2
        y += 0.2
    starts.append(np.array(grid_centers[:n_circles]))
    
    # 3. Random starts (few of them)
    for _ in range(3):
        starts.append(generate_random_grid(n_circles))

    for initial_centers in starts:
        # Flatten for optimizer
        x0 = initial_centers.flatten()
        
        # Bounds for centers [0, 1]
        bounds = [(0, 1)] * (n_circles * 2)
        
        # Run Nelder-Mead (derivative free, good for non-smooth objectives)
        # Maxiter needs to be sufficient. 
        try:
            res = minimize(
                objective_function, 
                x0, 
                method='Nelder-Mead', 
                bounds=bounds,
                options={'maxiter': 5000, 'xatol': 1e-6, 'fatol': 1e-9}
            )
            
            if res.success or res.fun < -best_sum:
                current_sum = -res.fun
                if current_sum > best_sum:
                    best_sum = current_sum
                    best_centers = res.x.reshape((n_circles, 2))
                    # Re-solve LP to get exact radii for best centers
                    radii, _, _ = solve_radii_lp(best_centers)
                    best_radii = radii
        except Exception as e:
            print(f"Optimization error: {e}")
            continue

    # Final validation check and cleanup
    if best_centers is None:
        # Fallback
        best_centers = generate_hex_grid(n_circles)
        best_radii, best_sum, _ = solve_radii_lp(best_centers)

    # Clip centers to [0, 1] just in case
    best_centers = np.clip(best_centers, 0, 1)
    
    # Ensure radii are consistent with final centers
    best_radii, final_sum, _ = solve_radii_lp(best_centers)
    
    return best_centers, best_radii, final_sum
