

Looking at this problem, I need to pack 26 circles in a unit square to maximize the sum of radii. The target is 2.636.

<strategy>

2. **Key insight**: Maximizing sum(r_i) (not area) benefits from having larger circles where space allows, with smaller circles filling gaps. Corner and edge placement is crucial.

3. **Strategy**: Use a force-based simulation that:
   - Starts with a good hexagonal grid arrangement
   - Iteratively updates radii based on constraints (nearest boundary and neighbor)
   - Applies repulsion forces to separate circles and create space
   - Uses cooling schedule for convergence
   - Multiple restarts to avoid local optima

4. **Implementation**: 
   - Initialize with hexagonal pattern (5 rows: 5,6,5,6,4 = 26 circles)
   - Run force-based simulation with boundary repulsion
   - After convergence, compute final radii from constraints
   - Run multiple restarts and keep best solution
   - Final fine-tuning pass with smaller step sizes

5. **Optimization tricks**:
   - Use different initial perturbations for multiple restarts
   - Decay learning rate gradually
   - Ensure circles are pushed toward boundaries where they can grow larger

import numpy as np


def compute_radii_from_centers(centers, radii, n):
    """Compute maximum valid radii given centers and other radii."""
    new_radii = np.zeros(n)
    for i in range(n):
        r = min(centers[i][0], 1.0 - centers[i][0],
                centers[i][1], 1.0 - centers[i][1])
        for j in range(n):
            if i != j:
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                r = min(r, dist - radii[j])
        new_radii[i] = max(0.0, r)
    return new_radii


def simulate(centers, radii, n, iterations, dt_initial):
    """Run force-based simulation to optimize packing."""
    dt = dt_initial
    
    for it in range(iterations):
        # Compute current radii
        radii = compute_radii_from_centers(centers, radii, n)
        
        # Compute forces
        forces = np.zeros((n, 2))
        
        for i in range(n):
            # Boundary repulsion
            for d in range(2):
                margin = radii[i] + 0.002
                if centers[i][d] < margin:
                    forces[i][d] += 200.0 * (margin - centers[i][d])
                if centers[i][d] > 1.0 - margin:
                    forces[i][d] -= 200.0 * (centers[i][d] - (1.0 - margin))
            
            # Inter-circle repulsion
            for j in range(i + 1, n):
                diff = centers[i] - centers[j]
                dist = np.sqrt(np.sum(diff ** 2))
                min_dist = radii[i] + radii[j]
                
                if dist < min_dist + 0.015:
                    force_mag = 80.0 * max(0.0, min_dist + 0.015 - dist)
                    if dist > 1e-10:
                        forces[i] += diff * force_mag / dist
                        forces[j] -= diff * force_mag / dist
        
        # Update positions
        centers += dt * forces
        centers = np.clip(centers, 0.0001, 0.9999)
        
        # Decay dt
        dt *= 0.9993
    
    return centers, radii


def make_hexagonal_init(n, seed):
    """Create hexagonal packing initialization with some perturbation."""
    rng = np.random.RandomState(seed)
    centers = np.zeros((n, 2))
    idx = 0
    
    # 5 rows with varying columns to get 26
    row_counts = [5, 6, 5, 6, 4]
    spacing = 0.18
    
    for row in range(5):
        y = 0.08 + row * spacing * 0.88
        if row % 2 == 0:
            x_start = 0.08
        else:
            x_start = 0.08 + spacing * 0.5
        
        for col in range(row_counts[row]):
            if idx < n:
                x = x_start + col * spacing
                centers[idx] = [x + rng.uniform(-0.01, 0.01),
                                y + rng.uniform(-0.01, 0.01)]
                idx += 1
    
    # Fill any remaining
    while idx < n:
        centers[idx] = [0.5 + rng.uniform(-0.3, 0.3),
                        0.5 + rng.uniform(-0.3, 0.3)]
        idx += 1
    
    centers = np.clip(centers, 0.05, 0.95)
    radii = np.ones(n) * 0.07
    return centers, radii


def run_packing():
    n = 26
    best_centers = None
    best_radii = None
    best_sum = -1.0
    
    # Try multiple restarts
    for seed in range(15):
        centers, radii = make_hexagonal_init(n, seed)
        
        # Phase 1: coarse optimization
        centers, radii = simulate(centers, radii, n, 3000, 0.008)
        
        # Phase 2: fine optimization
        centers, radii = simulate(centers, radii, n, 4000, 0.003)
        
        # Phase 3: very fine optimization
        centers, radii = simulate(centers, radii, n, 5000, 0.001)
        
        # Final radius computation
        radii = compute_radii_from_centers(centers, radii, n)
        
        # Check validity and compute sum
        current_sum = np.sum(radii)
        
        # Verify no overlaps
        valid = True
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist < radii[i] + radii[j] - 1e-10:
                    valid = False
                    break
            if not valid:
                break
        
        if valid and current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers.copy()
            best_radii = radii.copy()
    
    # Final shrink to ensure validity
    best_radii *= 0.99995
    
    # Recompute to be safe
    for i in range(n):
        best_radii[i] = min(best_centers[i][0], 1.0 - best_centers[i][0],
                           best_centers[i][1], 1.0 - best_centers[i][1])
        for j in range(n):
            if i != j:
                dist = np.sqrt(np.sum((best_centers[i] - best_centers[j]) ** 2))
                best_radii[i] = min(best_radii[i], dist - best_radii[j])
        best_radii[i] = max(0.0, best_radii[i])
    
    best_radii *= 0.9999
    
    final_sum = np.sum(best_radii)
    
    return best_centers, best_radii, final_sum
