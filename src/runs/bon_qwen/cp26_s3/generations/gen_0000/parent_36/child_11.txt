

To maximize the sum of radii for 26 circles in a unit square, a numerical optimization approach is required. The strategy involves two stages:
1.  **Initialization**: Placing circles in a hexagonal (triangular) lattice, which provides the densest possible arrangement in the plane and a robust starting point for optimization.
2.  **Numerical Optimization**: Using the `scipy.optimize.minimize` function with the `SLSQP` method to maximize $\sum r_i$ while satisfying non-overlap and boundary constraints.

```python
import numpy as np
from scipy.optimize import minimize
import math

def run_packing():
    """
    Pack 26 circles in a unit square to maximize the sum of radii.
    Uses a hexagonal initialization followed by SLSQP optimization.
    """
    n_circles = 26
    
    # --- 1. Hexagonal Initialization ---
    centers, radii = _hexagonal_init(n_circles)
    
    # Flatten variables: [x0, y0, r0, x1, y1, r1, ...]
    x0 = np.zeros(3 * n_circles)
    for i in range(n_circles):
        x0[3 * i] = centers[i, 0]
        x0[3 * i + 1] = centers[i, 1]
        x0[3 * i + 2] = radii[i]

    # --- 2. Optimization Setup ---
    # Bounds: x, y in [0, 1], r in [0, 0.5]
    bounds = [(0, 1), (0, 1), (0, 0.5)] * n_circles

    # Objective: Minimize negative sum of radii (maximize sum of radii)
    def objective(v):
        return -np.sum(v[2::3])

    # Constraints
    constraints = []
    
    # Boundary Constraints
    for i in range(n_circles):
        idx = 3 * i
        # x >= r, 1-x >= r, y >= r, 1-y >= r
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: v[3*i] - v[3*i+2]})
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: 1 - v[3*i] - v[3*i+2]})
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: v[3*i+1] - v[3*i+2]})
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: 1 - v[3*i+1] - v[3*i+2]})

    # Overlap Constraints (Distance^2 >= (r1 + r2)^2)
    # Defined as a single vectorized function for efficiency
    def overlap_constraints(v):
        dist_sq = []
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                dx = v[3*i] - v[3*j]
                dy = v[3*i+1] - v[3*j+1]
                dist_sq.append(dx**2 + dy**2 - (v[3*i+2] + v[3*j+2])**2)
        return np.array(dist_sq)

    constraints.append({'type': 'ineq', 'fun': overlap_constraints})

    # Run SLSQP
    try:
        result = minimize(
            objective, 
            x0, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=constraints,
            options={'ftol': 1e-10, 'maxiter': 5000, 'disp': False}
        )
        
        if result.success:
            final_centers = result.x[0::3].reshape(-1, 2)
            final_radii = result.x[2::3]
            sum_radii = np.sum(final_radii)
            return final_centers, final_radii, sum_radii
    except Exception as e:
        print(f"Optimization failed: {e}")

    # Fallback to initialization if optimization fails
    return centers, radii, np.sum(radii)

def _hexagonal_init(n):
    """Initializes centers and radii using a hexagonal packing pattern."""
    r = 0.10
    spacing_x = 2 * r
    spacing_y = r * math.sqrt(3)
    shift_x = r
    
    centers = []
    row = 0
    while len(centers) < n:
        y = r + row * spacing_y
        if y + r > 1.0:
            break
            
        start_x = shift_x if row % 2 == 1 else r
        x = start_x
        while x + r <= 1.0 and len(centers) < n:
            centers.append([x, y])
            x += spacing_x
        row += 1
    
    return np.array(centers), np.ones(n) * r
```
