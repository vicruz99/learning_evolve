import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Pack 26 circles in a unit square to maximize the sum of radii.
    """
    n = 26
    
    # 1. Initial Guess: Hexagonal Lattice
    # We generate points in a hexagonal pattern and select the first n valid ones.
    points = []
    r_init = 0.05 # Small initial radius to ensure valid packing
    
    # Hexagonal spacing parameters
    row_spacing = r_init * np.sqrt(3)
    col_spacing = 2 * r_init
    
    y = r_init
    row_idx = 0
    
    while len(points) < n:
        x = r_init + (row_idx % 2) * col_spacing * 0.5
        while x + r_init <= 1.0 and len(points) < n:
            points.append([x, y, r_init])
            x += col_spacing
        y += row_spacing
        row_idx += 1
        
    x0 = np.array(points, dtype=float).flatten()
    
    # 2. Objective Function
    # We maximize sum of radii, so we minimize negative sum
    def objective(vars_flat):
        radii = vars_flat[2::3]
        return -np.sum(radii)
    
    # 3. Constraints
    constraints = []
    
    # Boundary constraints: x - r >= 0, 1 - x - r >= 0, etc.
    for i in range(n):
        # x_i - r_i >= 0
        cons_x1 = {'type': 'ineq', 'fun': lambda v, i=i: v[3*i] - v[3*i+2]}
        constraints.append(cons_x1)
        
        # 1 - x_i - r_i >= 0
        cons_x2 = {'type': 'ineq', 'fun': lambda v, i=i: 1.0 - v[3*i] - v[3*i+2]}
        constraints.append(cons_x2)
        
        # y_i - r_i >= 0
        cons_y1 = {'type': 'ineq', 'fun': lambda v, i=i: v[3*i+1] - v[3*i+2]}
        constraints.append(cons_y1)
        
        # 1 - y_i - r_i >= 0
        cons_y2 = {'type': 'ineq', 'fun': lambda v, i=i: 1.0 - v[3*i+1] - v[3*i+2]}
        constraints.append(cons_y2)

    # Non-overlap constraints: dist(i, j) >= r_i + r_j
    # (x_i - x_j)^2 + (y_i - y_j)^2 >= (r_i + r_j)^2
    # dist_sq - (r_i + r_j)^2 >= 0
    for i in range(n):
        for j in range(i + 1, n):
            def non_overlap(v, i=i, j=j):
                xi, yi, ri = v[3*i], v[3*i+1], v[3*i+2]
                xj, yj, rj = v[3*j], v[3*j+1], v[3*j+2]
                dist_sq = (xi - xj)**2 + (yi - yj)**2
                return dist_sq - (ri + rj)**2
            
            constraints.append({'type': 'ineq', 'fun': non_overlap})

    # 4. Bounds
    # x, y in [0, 1], r in [0, 0.5]
    bounds = [(0, 1), (0, 1), (0, 0.5)] * n
    
    # 5. Optimization
    # Using SLSQP for constrained non-linear optimization
    result = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints, 
                      options={'maxiter': 1000, 'ftol': 1e-12})
    
    # 6. Extract and Format Results
    optimal_vars = result.x
    centers = np.array([optimal_vars[3*i:3*i+2] for i in range(n)])
    radii = np.array([optimal_vars[3*i+2] for i in range(n)])
    sum_radii = float(np.sum(radii))
    
    return centers, radii, sum_radii

if __name__ == "__main__":
