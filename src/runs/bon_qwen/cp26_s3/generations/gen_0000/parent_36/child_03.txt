import numpy as np
from scipy.optimize import minimize
import math

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # We need to return centers (n, 2), radii (n,), sum_radii
    
    # Optimization variables vector v of length 3*n
    # v[0:n]     -> x coordinates
    # v[n:2*n]   -> y coordinates
    # v[2*n:3*n] -> radii
    
    # Objective function: maximize sum of radii -> minimize negative sum
    def objective(v):
        return -np.sum(v[2*n : 3*n])
    
    # We will collect constraints in a list
    constraints = []
    
    # 1. Boundary Constraints
    # For each circle i:
    # x_i - r_i >= 0
    # 1 - x_i - r_i >= 0
    # y_i - r_i >= 0
    # 1 - y_i - r_i >= 0
    
    for i in range(n):
        idx_x = i
        idx_y = n + i
        idx_r = 2 * n + i
        
        # x - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: v[idx_x] - v[idx_r]
        })
        # 1 - x - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: 1.0 - v[idx_x] - v[idx_r]
        })
        # y - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: v[idx_y] - v[idx_r]
        })
        # 1 - y - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: 1.0 - v[idx_y] - v[idx_r]
        })

    # 2. Overlap Constraints
    # For each pair i < j:
    # dist(i, j) - (r_i + r_j) >= 0

    pairs = []
    for i in range(n):
        for j in range(i + 1, n):
            pairs.append((i, j))
            
    for i, j in pairs:
        idx_xi = i
        idx_yi = n + i
        idx_ri = 2 * n + i
        
        idx_xj = j
        idx_yj = n + j
        idx_rj = 2 * n + j
        
        constraints.append({
            'type': 'ineq',

        })

            # Approx area per circle 1/n. r ~ sqrt(1/(pi*n)) ~ 0.11

                        break

                    break
