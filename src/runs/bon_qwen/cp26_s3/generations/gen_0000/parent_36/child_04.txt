import numpy as np
from scipy.optimize import minimize
import math

def get_hexagonal_initialization(n, seed=42):
    """
    Generates initial centers for n circles based on a hexagonal lattice
    packed inside the unit square, with some random perturbation.
    """
    rng = np.random.RandomState(seed)
    
    # Estimate optimal radius for equal circles to set scale
    # Density of hex packing ~ 0.9069
    # Area = n * pi * r^2 ~ 0.9
    # r ~ sqrt(0.9 / (n * pi))
    r_est = math.sqrt(0.9 / (n * math.pi))
    
    # Spacing for hexagonal packing
    # Horizontal distance between centers in same row: 2*r
    # Vertical distance between rows: sqrt(3)*r
    dx = 2 * r_est
    dy = math.sqrt(3) * r_est
    
    centers = []
    
    # Try to fill the square with a hexagonal grid
    # We iterate rows and columns
    row = 0
    while True:
        # Offset for hexagonal rows (every other row shifted by dx/2)
        x_offset = (dx / 2) if (row % 2 == 1) else 0
        
        # Determine starting y for this row
        y_start = r_est + row * dy
        
        # If this row is too high, break
        if y_start > 1 - r_est:
            break
            
        # Fill columns for this row
        col = 0
        while True:
            x = r_est + x_offset + col * dx
            if x > 1 - r_est:
                break
            centers.append([x, y_start])
            col += 1
        
        row += 1
        
        # Safety break if we generated too many or got stuck
        if len(centers) > n + 100:
            break

    # If we have more points than needed, trim them (prefer keeping those closer to center or just truncate)
    # If we have fewer, we might need to add random points, but hex grid usually covers well.
    if len(centers) > n:
        centers = centers[:n]
    elif len(centers) < n:
        # Add random points in empty space
        # Simple strategy: add random points until we hit n
        while len(centers) < n:
            pt = rng.uniform(r_est, 1 - r_est, 2)
            # Check if too close to existing
            dists = np.linalg.norm(np.array(centers) - pt, axis=1)
            if np.min(dists) > r_est:
                centers.append(pt)
            else:
                # If crowded, just add random
                centers.append(pt)

    # Add small random perturbation to avoid symmetry locks
    centers = np.array(centers)
    centers += rng.normal(0, 0.005, centers.shape)
    
    return centers

def objective_function(vars_flat, n):
    """
    Objective function to minimize: -Sum(Radii) + Penalties
    
    vars_flat structure:
    [x0, x1, ..., xn-1, y0, y1, ..., yn-1, r0, r1, ..., rn-1]
    """
    centers_x = vars_flat[:n]
    centers_y = vars_flat[n:2*n]
    radii = vars_flat[2*n:]
    
    current_sum_radii = np.sum(radii)
    penalty = 0.0
    
    # Penalty parameters
    wall_penalty_weight = 1000.0
    overlap_penalty_weight = 1000.0
    
    # 1. Wall constraints penalty
    # x - r >= 0  => r - x <= 0
    # 1 - x - r >= 0 => r + x - 1 <= 0
    # Same for y
    
    # We want to penalize if constraints are violated.
    # Violation amount: max(0, r - x), etc.
    
    # Left wall: x - r >= 0. Violation: r - x > 0.
    # We use quadratic penalty for smoothness
    viol_x_left = np.maximum(0, radii - centers_x)
    penalty += wall_penalty_weight * np.sum(viol_x_left ** 2)
    
    # Right wall: 1 - x - r >= 0. Violation: x + r - 1 > 0.
    viol_x_right = np.maximum(0, centers_x + radii - 1.0)
    penalty += wall_penalty_weight * np.sum(viol_x_right ** 2)
    
    # Bottom wall: y - r >= 0.
    viol_y_bot = np.maximum(0, radii - centers_y)
    penalty += wall_penalty_weight * np.sum(viol_y_bot ** 2)
    
    # Top wall: 1 - y - r >= 0.
    viol_y_top = np.maximum(0, centers_y + radii - 1.0)
    penalty += wall_penalty_weight * np.sum(viol_y_top ** 2)
    
    # 2. Overlap constraints penalty
    # dist(i, j) >= r_i + r_j
    # Violation: r_i + r_j - dist(i, j) > 0
    
    # Vectorized distance calculation might be slow for N=26 if done fully inside loop 
    # but N is small enough.
    
    # To save time, we can check only if r_i + r_j is large? 
    # But for safety let's do full check. 26*25/2 = 325 pairs. Cheap.
    
    # Compute pairwise distances
    # Centers array (n, 2)
    C = np.column_stack((centers_x, centers_y))
    
    # Distance matrix
    # Using broadcasting: (n, 1, 2) - (1, n, 2) -> (n, n, 2)
    diff = C[:, np.newaxis, :] - C[np.newaxis, :, :]
    dists = np.linalg.norm(diff, axis=2)
    
    # Mask upper triangle
    # We only care about i < j
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    
    # Sum of radii matrix
    r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    
    # Violations
    violations = np.maximum(0, r_sum - dists)
    
    # Sum squared violations for active pairs
    penalty += overlap_penalty_weight * np.sum(violations[mask] ** 2)
    
    return -current_sum_radii + penalty

def run_packing():
    n = 26
    
    best_sum_radii = -np.inf
    best_centers = None
    best_radii = None
    
    # We will run optimization multiple times with different seeds
    seeds = [42, 123, 456, 789, 0, 10, 20]
    
    for seed in seeds:
        # 1. Initialization
        init_centers = get_hexagonal_initialization(n, seed=seed)
        
        # Initial radii guess. 
        # A safe guess is slightly smaller than the grid spacing suggests.
        # For hex packing with r_est, spacing is 2*r_est.
        # We can start with r_est * 0.5 to be safe from overlaps initially?
        # Or just let optimizer figure it out.
        # Let's start with small radii to avoid huge initial penalty.
        init_radii = np.full(n, 0.05)
        
        # Flatten variables
        x0 = np.concatenate([init_centers[:, 0], init_centers[:, 1], init_radii])
        
        # Bounds
        # x, y in [0, 1]
        # r in [0, 0.5] (cannot be larger than 0.5 in unit square)
        bounds = []
        for _ in range(n):
            bounds.append((0.0, 1.0)) # x
        for _ in range(n):
            bounds.append((0.0, 1.0)) # y
        for _ in range(n):
            bounds.append((0.0, 0.5)) # r
            
        # 2. Optimization
        # Use L-BFGS-B or SLSQP. L-BFGS-B is faster for large vars but handles box bounds.
        # SLSQP handles non-linear constraints, but we used penalty function.
        # So L-BFGS-B is good choice.
        
        try:
            res = minimize(objective_function, x0, args=(n,), method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 1000, 'ftol': 1e-9})
            
            if res.success or res.fun < 0: # fun is negative sum + penalty
                # Extract result
                centers_opt = np.column_stack((res.x[:n], res.x[n:2*n]))
                radii_opt = res.x[2*n:]
                
                # Calculate actual sum of radii
                sum_r = np.sum(radii_opt)
                
                # Validate roughly (check penalty part of objective)
                # The objective is -sum + penalty. 
                # If penalty is small, it's valid.
                # We can reconstruct penalty to check validity strictly?
                # Or just check constraints manually.
                
                # Check validity
                valid = True
                
                # Check walls
                for i in range(n):
                    x, y = centers_opt[i]
                    r = radii_opt[i]
                    if x < -1e-6 or y < -1e-6 or x > 1 + 1e-6 or y > 1 + 1e-6:
                        valid = False; break
                    if r < -1e-6: valid = False; break
                    if x - r < -1e-6 or x + r > 1 + 1e-6: valid = False; break
                    if y - r < -1e-6 or y + r > 1 + 1e-6: valid = False; break
                
                if valid:
                    # Check overlaps
                    for i in range(n):
                        for j in range(i + 1, n):
                            d = math.sqrt((centers_opt[i, 0] - centers_opt[j, 0])**2 + 
                                          (centers_opt[i, 1] - centers_opt[j, 1])**2)
                            if d < radii_opt[i] + radii_opt[j] - 1e-7:
                                valid = False
                                break
                        if not valid: break
                
                if valid and sum_r > best_sum_radii:
                    best_sum_radii = sum_r
                    best_centers = centers_opt.copy()
                    best_radii = radii_opt.copy()
                    
        except Exception as e:
            continue

    # Final fallback if nothing valid found (unlikely)
    if best_centers is None:
        # Return a valid trivial solution
        best_centers = np.random.rand(26, 2) * 0.5 + 0.25
        best_radii = np.full(26, 0.01)
        best_sum_radii = 0.26

    return best_centers, best_radii, float(best_sum_radii)
