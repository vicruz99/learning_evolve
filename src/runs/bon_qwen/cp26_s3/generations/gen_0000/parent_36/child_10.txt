def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    points = []
    row = 0

        # Hex grid logic: row spacing sqrt(3), horizontal spacing 2, offset 1 for odd rows
        cols = 8 # generous number of columns
        for i in range(cols):
            if len(points) >= n:
                break
            x = 2.0 * i + (row % 2)

            points.append([x, y])
        row += 1

    r_init = 0.05

    height = max_vals[1] - min_vals[1]

        centers = np.random.uniform(r_init, 1 - r_init, (n, 2))

        scale = min((1.0 - 2.0 * r_init) / width, (1.0 - 2.0 * r_init) / height)
        centers = np.zeros((n, 2))

        centers[:, 1] = (points[:, 1] - min_vals[1]) * scale + r_init
    
    radii = np.full(n, r_init)

    growth_step = 0.0003
    max_iterations = 2000
    relax_steps = 30
    force_k = 100.0
    position_step = 0.008

        radii += growth_step

            # Compute pairwise repulsive forces

            diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
            dists = np.sqrt(np.sum(diff**2, axis=2))
            
            # Avoid division by zero

            # Direction vectors and force magnitudes
            inv_dists = 1.0 / dists[:, :, np.newaxis]
            unit_diff = diff * inv_dists
            mag = overlap * force_k
            force_vec = unit_diff * mag[:, :, np.newaxis]
            
            # Accumulate forces
            forces += np.sum(force_vec, axis=1)
            
            # Boundary forces
            for i in range(n):
                r = radii[i]
                x, y = centers[i]
                
                # X-bounds
                if x < r:
                    forces[i, 0] += force_k * (r - x)
                elif x > 1 - r:
                    forces[i, 0] -= force_k * (x - (1 - r))
                
                # Y-bounds
                if y < r:
                    forces[i, 1] += force_k * (r - y)
                elif y > 1 - r:
                    forces[i, 1] -= force_k * (y - (1 - r))
            
            # Clamp forces to limit step size (stability)
            force_mags = np.sqrt(np.sum(forces**2, axis=1))
            max_force = 0.1
            limit = np.maximum(force_mags, max_force)
            forces = forces * (max_force / limit[:, np.newaxis])
            
            centers += position_step * forces
            
            # Hard clamp to ensure validity

            centers[:, 1] = np.clip(centers[:, 1], radii, 1 - radii)

    return centers, radii, sum_radii
