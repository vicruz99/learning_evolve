import numpy as np
from scipy.optimize import minimize

def objective(x):
    # Minimize negative sum of radii
    return -np.sum(x[2::3])

def constraint_fun(x):
    n = 26
    n_con = 4 * n + n * (n - 1) // 2
    res = np.empty(n_con)
    k = 0
    
    # Boundary constraints
    for i in range(n):
        idx = 3 * i
        xi, yi, ri = x[idx], x[idx + 1], x[idx + 2]
        res[k] = xi - ri
        res[k + 1] = 1.0 - xi - ri
        res[k + 2] = yi - ri
        res[k + 3] = 1.0 - yi - ri
        k += 4
        
    # Non-overlap constraints (squared distance formulation)
    for i in range(n):
        for j in range(i + 1, n):
            idx_i, idx_j = 3 * i, 3 * j
            dx = x[idx_i] - x[idx_j]
            dy = x[idx_i + 1] - x[idx_j + 1]
            dr = x[idx_i + 2] + x[idx_j + 2]
            res[k] = dx * dx + dy * dy - dr * dr
            k += 1
            
    return res

def run_packing():
    n = 26
    np.random.seed(42)
    
    # Hexagonal lattice initialization
    points = []
    s = 0.2
    for i in range(10):
        for j in range(10):
            x = j * s + (i % 2) * s / 2
            y = i * s * np.sqrt(3) / 2
            if 0.0 <= x <= 1.0 and 0.0 <= y <= 1.0:
                points.append((x, y))
                
    np.random.shuffle(points)
    points = np.array(points[:n])
    
    # Perturb to break symmetry and ensure initial feasibility
    points += np.random.uniform(-0.015, 0.015, size=(n, 2))
    points = np.clip(points, 0.12, 0.88)
    
    # Flatten to optimization vector: [x1, y1, r1, x2, y2, r2, ...]
    x0 = np.zeros(3 * n)
    for i in range(n):
        x0[3 * i] = points[i, 0]
        x0[3 * i + 1] = points[i, 1]
        x0[3 * i + 2] = 0.025  # Feasible initial radius
        
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
    cons = {'type': 'ineq', 'fun': constraint_fun}
    
    # Run SLSQP optimizer
    res = minimize(
        objective, x0, method='SLSQP', bounds=bounds, 
        constraints=cons, options={'ftol': 1e-14, 'maxiter': 4000, 'disp': False}
    )
    
    # Extract and sanitize results
    centers = np.clip(res.x[:2 * n].reshape(n, 2), 0.0, 1.0)
    radii = np.maximum(res.x[2 * n:], 0.0)
    
    return centers, radii, np.sum(radii)
