from scipy.optimize import minimize

def run_packing():

    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    
    Returns:
        tuple: (centers, radii, sum_radii)
    """

    # 1. Initialization
    # Place 25 circles in a 5x5 grid and 1 in a gap
    grid_coords = np.array([0.1, 0.3, 0.5, 0.7, 0.9])
    centers = []
    
    # 5x5 Grid
    for y in grid_coords:
        for x in grid_coords:
            centers.append([x, y])
    
    # 26th circle in a gap (e.g., between (0.1, 0.1) and (0.3, 0.3))
    # Center of gap is (0.2, 0.2)
    centers.append([0.2, 0.2])
    
    centers = np.array(centers)
    radii = np.full(n_circles, 0.05) # Initial small radius to ensure feasibility
    
    # Flatten variables for optimizer: [x1, y1, r1, x2, y2, r2, ...]
    x0 = np.zeros(n_circles * 3)
    for i in range(n_circles):
        x0[3*i] = centers[i, 0]
        x0[3*i + 1] = centers[i, 1]
        x0[3*i + 2] = radii[i]
        
    # Bounds: x, y in [0, 1], r in [0, 0.5]
    bounds = []
    for _ in range(n_circles):
        bounds.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])

    # 2. Objective and Penalty Function
    penalty_weight = 1000.0
    
    def cost_function(vars_flat):
        # Reshape
        centers_opt = vars_flat.reshape(-1, 3)[:, :2]
        radii_opt = vars_flat.reshape(-1, 3)[:, 2]
        
        # Objective: Minimize negative sum of radii
        obj = -np.sum(radii_opt)
        
        penalty = 0.0
        
        # Boundary penalties
        # Circle must be inside [0, 1]x[0, 1]
        # x - r >= 0  => r - x <= 0
        # 1 - x - r >= 0 => x + r - 1 <= 0
        # Same for y
        for i in range(n_circles):
            x, y, r = centers_opt[i, 0], centers_opt[i, 1], radii_opt[i]
            
            # Left
            if r - x > 0: penalty += penalty_weight * (r - x)**2
            # Right
            if x + r - 1 > 0: penalty += penalty_weight * (x + r - 1)**2
            # Bottom
            if r - y > 0: penalty += penalty_weight * (r - y)**2
            # Top
            if y + r - 1 > 0: penalty += penalty_weight * (y + r - 1)**2
            
        # Overlap penalties
        # dist >= r1 + r2 => r1 + r2 - dist <= 0
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                dx = centers_opt[i, 0] - centers_opt[j, 0]
                dy = centers_opt[i, 1] - centers_opt[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                r_sum = radii_opt[i] + radii_opt[j]
                
                if r_sum - dist > 0:
                    penalty += penalty_weight * (r_sum - dist)**2
                    
        return obj + penalty

    # 3. Optimization
    # Using L-BFGS-B for bound constraints
    result = minimize(cost_function, x0, method='L-BFGS-B', bounds=bounds, 
                      options={'maxiter': 1000, 'ftol': 1e-12})
    
    # 4. Extract results
    best_vars = result.x
    best_centers = best_vars.reshape(-1, 3)[:, :2]
    best_radii = best_vars.reshape(-1, 3)[:, 2]
    
    # Ensure non-negative radii (clip just in case, though bounds should handle it)
    best_radii = np.maximum(best_radii, 0.0)
    
    sum_radii = np.sum(best_radii)
    
    return (best_centers, best_radii, float(sum_radii))
