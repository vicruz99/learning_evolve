1.  **Outer Optimization (Positions):** Use `scipy.optimize.differential_evolution` to search for optimal center coordinates $(x_i, y_i)$. This global optimizer is chosen for its ability to escape local minima in the high-dimensional search space (52 dimensions).

import numpy as np
from scipy.optimize import linprog, differential_evolution

def run_packing():
    """
    Finds an optimal packing of 26 circles in a unit square to maximize sum of radii.
    Uses Differential Evolution for positions and Linear Programming for radii.
    """
    n_circles = 26
    bounds = [(0.05, 0.95)] * (n_circles * 2) # Keep centers away from extreme edges initially

    def neg_sum_radii(flat_coords):
        """
        Objective function: Returns negative sum of radii.
        The LP maximizes sum(r) subject to geometric constraints.
        """
        centers = flat_coords.reshape(-1, 2)
        
        # 1. Define LP variables: r_0, ..., r_{n-1}
        # Maximize sum(r) => Minimize -sum(r)
        c = -np.ones(n_circles)
        
        # 2. Define Constraints: A_ub @ r <= b_ub
        A_ub = []
        b_ub = []
        
        # Pairwise distance constraints: r_i + r_j <= dist(i, j)
        # Optimized calculation
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                d = np.linalg.norm(centers[i] - centers[j])
                row = np.zeros(n_circles)
                row[i] = 1
                row[j] = 1
                A_ub.append(row)
                b_ub.append(d)
        
        # Boundary constraints: r_i <= x_i, 1-x_i, y_i, 1-y_i
        for i in range(n_circles):
            x, y = centers[i]
            row_i = np.zeros(n_circles)
            row_i[i] = 1
            
            # x bound
            A_ub.append(row_i)
            b_ub.append(x)
            # 1-x bound
            A_ub.append(row_i)
            b_ub.append(1 - x)
            # y bound
            A_ub.append(row_i)
            b_ub.append(y)
            # 1-y bound
            A_ub.append(row_i)
            b_ub.append(1 - y)

        # 3. Solve LP
        # Use high bound for radii (e.g., 0.5) to avoid numerical issues, 
        # though boundaries constrain them tighter.
        bounds_lp = [(0, 0.5)] * n_circles
        
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds_lp, method='highs')
            if res.success:
                return -res.fun
            else:
                # Return a low value if LP fails
                return -0.01
        except Exception:
            return -0.01

    # 4. Run Global Optimization
    # We use multiple strategies (initial seeds) conceptually by letting DE explore,
    # but we can bias the search range.
    
    # Using Differential Evolution
    result = differential_evolution(
        neg_sum_radii,
        bounds,
        seed=42,
        maxiter=150,
        popsize=30,
        tol=1e-6,
        mutation=(0.5, 1.5),
        recombination=0.7,
        disp=False
    )
    
    # 5. Recover final configuration
    final_centers = result.x.reshape(-1, 2)
    
    # Solve LP one last time to get optimal radii for the best centers
    # Re-calculate LP for the best found centers
    centers = final_centers
    c = -np.ones(n_circles)
    A_ub = []
    b_ub = []
    
    for i in range(n_circles):
        for j in range(i + 1, n_circles):
            d = np.linalg.norm(centers[i] - centers[j])
            row = np.zeros(n_circles)
            row[i] = 1
            row[j] = 1
            A_ub.append(row)
            b_ub.append(d)
            
        x, y = centers[i]
        row_i = np.zeros(n_circles)
        row_i[i] = 1
        A_ub.extend([row_i, row_i, row_i, row_i])
        b_ub.extend([x, 1-x, y, 1-y])
        
    bounds_lp = [(0, 0.5)] * n_circles
    final_res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds_lp, method='highs')
    
    final_radii = final_res.x
    sum_radii = np.sum(final_radii)
    
    return final_centers, final_radii, sum_radii

if __name__ == "__main__":
    centers, radii, s = run_packing()
