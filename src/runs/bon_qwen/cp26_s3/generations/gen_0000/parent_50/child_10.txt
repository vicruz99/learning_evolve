import numpy as np
from scipy.optimize import minimize

def compute_constraints(params):
    """
    Computes inequality constraints for the packing problem.
    Constraints are defined as g(params) >= 0.
    """
    n = 26
    centers = params[:2*n].reshape((n, 2))
    radii = params[2*n:]
    cons = []
    
    # Boundary and non-negative radius constraints
    for i in range(n):
        r = radii[i]
        x, y = centers[i]
        cons.append(x - r)           # x >= r
        cons.append(1.0 - x - r)     # x + r <= 1
        cons.append(y - r)           # y >= r
        cons.append(1.0 - y - r)     # y + r <= 1
        cons.append(r)               # r >= 0
        
    # Pairwise non-overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx*dx + dy*dy)
            cons.append(dist - (radii[i] + radii[j])) # dist >= r_i + r_j
            
    return np.array(cons)

def run_packing():
    n = 26
    
    # Initialize centers in a structured grid to ensure feasibility
    centers = []
    for r in range(6):
        for c in range(5):
            if len(centers) >= n: break
            centers.append([0.1 + c * 0.15, 0.1 + r * 0.15])
    centers = np.array(centers[:n])
    
    # Break symmetry with fixed deterministic perturbation
    centers[:, 0] += 0.002 * np.arange(n)
    centers[:, 1] += 0.002 * np.arange(n)
    centers = np.clip(centers, 0.05, 0.95)
    
    # Initial radii
    radii = np.ones(n) * 0.07
    
    # Flatten to optimizer format
    x0 = np.zeros(n * 3)
    for i in range(n):
        x0[3*i] = centers[i, 0]
        x0[3*i+1] = centers[i, 1]
        x0[3*i+2] = radii[i]
        
    # Bounds for variables
    bounds = [(0, 1), (0, 1), (0, 1)] * n
    
    def objective(params):
        return -np.sum(params[2*n:])
        
    # Define constraint dictionary
    cons = {'type': 'ineq', 'fun': compute_constraints}
    
    # Run optimization
    res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, 
                   options={'maxiter': 3000, 'ftol': 1e-12, 'disp': False})
                   
    centers_opt = res.x[:2*n].reshape((n, 2))
    radii_opt = res.x[2*n:]
    
    # Ensure strict validity against numerical tolerance
    radii_opt = np.clip(radii_opt, 0, None)
    centers_opt = np.clip(centers_opt, 0, 1)
    
    return centers_opt, radii_opt, np.sum(radii_opt)
