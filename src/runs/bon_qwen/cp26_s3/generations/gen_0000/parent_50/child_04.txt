import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    
    Returns:
        centers: np.array of shape (26, 2) with (x, y) coordinates
        radii: np.array of shape (26,) with radius of each circle
        sum_radii: float sum of radii
    """
    n_circles = 26
    
    # --- 1. Initialization Strategy ---
    # We start with a valid packing using a hexagonal lattice pattern.
    # This provides a dense initial configuration which is a good starting point for optimization.
    
    def generate_initial_packing():
        # We aim for a radius slightly smaller than the expected optimum to ensure feasibility.
        # For 26 circles, a radius of ~0.08 is safe.
        r_init = 0.08
        
        # Hexagonal lattice parameters
        # Horizontal spacing between centers in a row
        dx = 2 * r_init
        # Vertical spacing between rows
        dy = r_init * np.sqrt(3)
        
        centers = []
        y = r_init
        
        # Generate points row by row
        while len(centers) < n_circles:
            x = r_init
            row_parity = 0
            # Check if we should shift the row (standard hex packing)
            # If previous row had points, shift this one by dx/2
            if len(centers) > 0:
                row_parity = 1
            
            while x <= 1.0 - r_init and len(centers) < n_circles:
                # Check if point fits
                centers.append([x + row_parity * (dx / 2), y])
                x += dx
            
            y += dy
            
        # Ensure we have exactly n_circles
        centers = centers[:n_circles]
        
        # If generation stopped early (unlikely with these params), fill with random valid points
        while len(centers) < n_circles:
            cx = np.random.uniform(r_init, 1 - r_init)
            cy = np.random.uniform(r_init, 1 - r_init)
            # Simple check to avoid immediate huge overlap, though not strictly necessary
            # as the optimizer will fix it.
            centers.append([cx, cy])
            
        return np.array(centers), np.full(n_circles, r_init)

    # --- 2. Objective and Constraints ---

    def objective(vars_vec):
        # vars_vec contains [x1, y1, r1, x2, y2, r2, ...]
        # We want to maximize sum(r_i), so we minimize -sum(r_i)
        radii = vars_vec[2::3]
        return -np.sum(radii)

    def get_constraints():
        constraints = []
        
        # Wall constraints: r <= x <= 1-r  =>  x - r >= 0  AND  1 - x - r >= 0
        # Same for y.
        for i in range(n_circles):
            idx_x = 3 * i
            idx_y = 3 * i + 1
            idx_r = 3 * i + 2
            
            # x - r >= 0
            constraints.append({
                'type': 'ineq',
                'fun': lambda v, i=i: v[idx_x] - v[idx_r]
            })
            # 1 - x - r >= 0
            constraints.append({
                'type': 'ineq',
                'fun': lambda v, i=i: 1.0 - v[idx_x] - v[idx_r]
            })
            # y - r >= 0
            constraints.append({
                'type': 'ineq',
                'fun': lambda v, i=i: v[idx_y] - v[idx_r]
            })
            # 1 - y - r >= 0
            constraints.append({
                'type': 'ineq',
                'fun': lambda v, i=i: 1.0 - v[idx_y] - v[idx_r]
            })
            
            # Radius non-negativity r >= 0
            constraints.append({
                'type': 'ineq',
                'fun': lambda v, i=i: v[idx_r]
            })

        # Overlap constraints: dist^2 >= (r_i + r_j)^2
        # dist^2 - (r_i + r_j)^2 >= 0
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                idx_xi, idx_yi, idx_ri = 3 * i, 3 * i + 1, 3 * i + 2
                idx_xj, idx_yj, idx_rj = 3 * j, 3 * j + 1, 3 * j + 2
                
                constraints.append({
                    'type': 'ineq',
                    'fun': lambda v, i=i, j=j: \
                        (v[idx_xi] - v[idx_xj])**2 + (v[idx_yi] - v[idx_yj])**2 - \
                        (v[idx_ri] + v[idx_rj])**2
                })
        
        return constraints

    # --- 3. Optimization Loop ---
    
    best_sum_radii = -1.0
    best_centers = None
    best_radii = None
    
    constraints = get_constraints()
    
    # Run optimization multiple times with slight perturbations to avoid local minima
    num_restarts = 5
    
    for restart in range(num_restarts):
        centers_init, radii_init = generate_initial_packing()
        
        # Add small random noise to centers to break symmetry
        noise = np.random.uniform(-0.001, 0.001, centers_init.shape)
        centers_init = centers_init + noise
        # Ensure centers stay within bounds after noise
        centers_init = np.clip(centers_init, radii_init, 1.0 - radii_init)

        # Flatten variables: [x1, y1, r1, x2, y2, r2, ...]
        x0 = np.zeros(3 * n_circles)
        x0[0::3] = centers_init[:, 0]
        x0[1::3] = centers_init[:, 1]
        x0[2::3] = radii_init
        
        # Bounds for optimization
        # x in [0, 1], y in [0, 1], r in [0, 0.5]
        bounds = []
        for i in range(n_circles):
            bounds.append((0.0, 1.0)) # x
            bounds.append((0.0, 1.0)) # y
            bounds.append((0.0, 0.5)) # r
            
        # Run SLSQP
        try:
            res = minimize(
                objective, 
                x0, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=constraints,
                options={'maxiter': 1000, 'ftol': 1e-9, 'disp': False}
            )
            
            if res.success or res.nit > 10:
                sum_r = -res.fun # Objective was negative sum
                if sum_r > best_sum_radii:
                    best_sum_radii = sum_r
                    # Extract solution
                    final_centers = np.zeros((n_circles, 2))
                    final_radii = np.zeros(n_circles)
                    final_centers[:, 0] = res.x[0::3]
                    final_centers[:, 1] = res.x[1::3]
                    final_radii = res.x[2::3]
                    best_centers = final_centers
                    best_radii = final_radii
                    
        except Exception as e:
            # If optimization fails, try next restart
            continue

    # If optimization didn't find anything (unlikely), fallback to init
    if best_centers is None:
        best_centers, best_radii = generate_initial_packing()
        best_sum_radii = np.sum(best_radii)

    # Final Validation check (internal sanity check)
    # In a real scenario, we trust the optimizer constraints, but let's just return.
    # Note: Numerical issues might cause tiny constraint violations.

    best_radii = np.maximum(best_radii, 0.0)
    
    return best_centers, best_radii, float(best_sum_radii)
