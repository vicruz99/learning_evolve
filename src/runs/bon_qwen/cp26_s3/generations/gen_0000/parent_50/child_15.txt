    *   **Objective**: Maximize $\sum r_i$ (equivalent to minimizing $-\sum r_i$).
    *   **Constraints**:

        *   Non-overlap: $(x_i - x_j)^2 + (y_i - y_j)^2 \ge (r_i + r_j)^2$ for all $i < j$.

import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # --- Initialization ---
    # Generate a hexagonal lattice of potential centers
    # We use a tentative radius to space the grid points
    r_tentative = 0.09
    points = []
    
    # Hexagonal grid generation
    # Vertical spacing: r * sqrt(3)
    # Horizontal spacing: 2 * r
    # Odd rows are shifted by r
    
    y = r_tentative
    row_idx = 0
    while y <= 1.0 - r_tentative:
        # Determine x start for this row
        # Even rows (0, 2, ...) start at r_tentative
        # Odd rows (1, 3, ...) start at r_tentative + r_tentative (shifted by r)
        if row_idx % 2 == 0:
            x_start = r_tentative
        else:
            x_start = r_tentative + r_tentative # shift by r
            
        x = x_start
        while x <= 1.0 - r_tentative:
            points.append([x, y])
            x += 2 * r_tentative
        y += r_tentative * np.sqrt(3)
        row_idx += 1
    
    # We need exactly 26 points
    # If we generated fewer, we might need a denser grid, but with r=0.09 we should have plenty.
    # If more, just take the first 26.
    centers_init = np.array(points[:n])
    
    # Set initial radii small to ensure feasibility
    # The distance between points in our grid is at least 2*r_tentative = 0.18
    # So radius 0.08 would touch, 0.05 is safe.
    r_init_val = 0.05
    radii_init = np.full(n, r_init_val)
    
    # Flatten variables: [x0, y0, x1, y1, ..., r0, r1, ...]
    # Actually, let's keep them grouped by circle for easier indexing in constraints?
    # Or just flat array. 
    # Let's use order: x_0, y_0, x_1, y_1, ..., x_25, y_25, r_0, r_1, ..., r_25
    x0 = np.concatenate([centers_init.flatten(), radii_init])
    
    # --- Optimization Setup ---
    
    def objective(vars_arr):
        # vars_arr: [x0, y0, ..., x25, y25, r0, ..., r25]
        # Radii are the last n elements
        r = vars_arr[2*n:]
        return -np.sum(r) # Minimize negative sum

    def constraints_function(vars_arr):
        # Extract centers and radii
        centers = vars_arr[:2*n].reshape(n, 2)
        radii = vars_arr[2*n:]
        
        # 1. Boundary Constraints
        # x >= r  => x - r >= 0
        # x <= 1 - r => 1 - x - r >= 0
        # y >= r  => y - r >= 0
        # y <= 1 - r => 1 - y - r >= 0
        
        # x - r
        cons_x_min = centers[:, 0] - radii
        # 1 - x - r
        cons_x_max = 1.0 - centers[:, 0] - radii
        # y - r
        cons_y_min = centers[:, 1] - radii
        # 1 - y - r
        cons_y_max = 1.0 - centers[:, 1] - radii
        
        # 2. Non-overlap Constraints
        # dist_sq >= (r_i + r_j)^2
        # dist_sq - (r_i + r_j)^2 >= 0
        
        # Compute squared Euclidean distance matrix
        # centers: (n, 2)
        # diff: (n, n, 2)
        diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
        dist_sq = np.sum(diff**2, axis=2)
        
        # Compute (r_i + r_j)^2
        r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
        r_sum_sq = r_sum**2
        
        # Constraint violation (should be >= 0)
        non_overlap = dist_sq - r_sum_sq
        
        # We only need upper triangle (i < j) to avoid duplicates and self-comparison
        mask = np.triu(np.ones((n, n), dtype=bool), k=1)
        non_overlap_unique = non_overlap[mask]
        
        # Concatenate all constraints
        all_cons = np.concatenate([
            cons_x_min, cons_x_max, 
            cons_y_min, cons_y_max, 
            non_overlap_unique
        ])
        
        return all_cons

    cons = {'type': 'ineq', 'fun': constraints_function}
    
    # Bounds
    # x, y in [0, 1]
    # r in [0, 0.5]
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
    
    # Run Optimization
    # SLSQP is suitable for this type of constrained optimization
    res = minimize(
        objective, 
        x0, 
        method='SLSQP', 
        bounds=bounds, 
        constraints=cons, 
        options={
            'maxiter': 2000, 
            'ftol': 1e-12, 
            'disp': False
        }
    )
    
    # Extract results
    final_centers = res.x[:2*n].reshape(n, 2)
    final_radii = res.x[2*n:]
    total_sum = -res.fun
    
    # Ensure radii are non-negative (just in case numerical error occurs)
    final_radii = np.maximum(final_radii, 0.0)
    
    # If optimization failed or result is bad, we might want to fallback, 
    # but SLSQP usually returns a point. 
    # Let's just return the optimized result.
    
    return final_centers, final_radii, total_sum

if __name__ == "__main__":
