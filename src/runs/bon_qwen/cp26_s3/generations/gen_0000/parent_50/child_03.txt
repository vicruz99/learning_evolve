

Looking at this problem, I need to pack 26 circles in a unit square to maximize the sum of radii. Let me think about the optimal strategy.

**Key Insights:**
1. For equal circles, 26 circles in a square gives radius ≈ 0.0808, sum ≈ 2.10
2. Unequal circles can potentially achieve higher sums by placing larger circles in spacious regions
3. Hexagonal packing is the densest arrangement for equal circles
4. The target sum of 2.636 suggests we need quite aggressive optimization

**Strategy:**
- Use hexagonal grid initialization with multiple restarts
- Apply penalty-based L-BFGS-B optimization with adaptive penalty weights
- Try various initial configurations (hexagonal, random, boundary-biased)
- Use multiple optimization stages with increasing penalty strength
- Focus on getting circles to boundaries and corners where they can grow larger

```python
import numpy as np
from scipy.optimize import minimize

def validate_packing(centers, radii):
    n = centers.shape[0]
    if np.isnan(centers).any() or np.isnan(radii).any():
        return False
    for i in range(n):
        if radii[i] < 0:
            return False
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            return False
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False
    return True

def compute_loss(centers, radii, n, penalty_weight):
    loss = -np.sum(radii)
    
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        loss += penalty_weight * max(0, r - x) ** 2
        loss += penalty_weight * max(0, r - (1 - x)) ** 2
        loss += penalty_weight * max(0, r - y) ** 2
        loss += penalty_weight * max(0, r - (1 - y)) ** 2
    
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            overlap = radii[i] + radii[j] - dist
            if overlap > 0:
                loss += penalty_weight * overlap ** 2
    
    return loss

def init_hex_grid(n, seed=0):
    np.random.seed(seed)
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.035
    
    idx = 0
    row = 0
    while idx < n:
        n_cols = min(6, n - idx)
        if n_cols <= 0:
            break
        offset = 0.085 if row % 2 == 1 else 0
        spacing = 0.165
        
        for col in range(n_cols):
            if idx >= n:
                break
            x = 0.08 + col * spacing + offset + np.random.randn() * 0.005
            y = 0.08 + row * 0.145 + np.random.randn() * 0.005
            centers[idx] = [x, y]
            idx += 1
        row += 1
    
    return centers, radii

def init_random(n, seed=0):
    np.random.seed(seed + 1000)
    centers = np.random.rand(n, 2) * 0.8 + 0.1
    radii = np.ones(n) * 0.03
    return centers, radii

def init_boundary(n, seed=0):
    np.random.seed(seed + 2000)
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.03
    
    idx = 0
    for corner in [(0.05, 0.05), (0.95, 0.05), (0.05, 0.95), (0.95, 0.95)]:
        for i in range(min(2, n - idx)):
            centers[idx] = [corner[0] + np.random.rand() * 0.05, corner[1] + np.random.rand() * 0.05]
            idx += 1
    
    while idx < n:
        centers[idx] = np.random.rand(2) * 0.6 + 0.2
        idx += 1
    
    return centers, radii

def optimize_packing(n, centers, radii):
    x0 = np.concatenate([centers.ravel(), radii])
    
    penalty_weights = [20, 80, 300, 1000, 3000, 8000]
    
    for pw in penalty_weights:
        def loss_fn(x, pw=pw):
            c = x[:2 * n].reshape(n, 2)
            r = np.maximum(x[2 * n:], 1e-12)
            return compute_loss(c, r, n, pw)
        
        bounds = [(0.0005, 0.9995)] * (2 * n) + [(1e-12, 0.5)] * n
        
        try:
            result = minimize(loss_fn, x0, method='L-BFGS-B', bounds=bounds,
                             options={'maxiter': 3000, 'ftol': 1e-15, 'gtol': 1e-12})
            x0 = result.x.copy()
        except:
            pass
    
    centers_opt = x0[:2 * n].reshape(n, 2)
    radii_opt = np.maximum(x0[2 * n:], 0)
    
    return centers_opt, radii_opt

def run_packing():
    n = 26
    best_centers = None
    best_radii = None
    best_sum = -1
    
    generators = [init_hex_grid, init_random, init_boundary]
    
    for gen_func in generators:
        for trial in range(15):
            centers, radii = gen_func(n, seed=trial)
            
            centers_opt, radii_opt = optimize_packing(n, centers, radii)
            
            if validate_packing(centers_opt, radii_opt):
                current_sum = np.sum(radii_opt)
                if current_sum > best_sum:
                    best_sum = current_sum
                    best_centers = centers_opt.copy()
                    best_radii = radii_opt.copy()
    
    if best_centers is None:
        best_centers, best_radii = init_hex_grid(n, seed=0)
        best_sum = np.sum(best_radii)
    
    return best_centers, best_radii, best_sum
```
