import numpy as np
from scipy.optimize import minimize

def calculate_loss(params, n, penalty_weight):
    """
    Calculates the loss function for the packing problem.
    Loss = -sum(radii) + penalty(overlaps) + penalty(boundaries)
    """
    # params is shape (3*n,) -> [x0, y0, r0, x1, y1, r1, ...]
    # Reshape to (n, 3) for easier handling
    pts = params.reshape(n, 3)
    centers = pts[:, :2] # (n, 2)
    radii = pts[:, 2]    # (n,)
    
    # Objective: maximize sum of radii => minimize -sum(radii)
    loss = -np.sum(radii)
    
    # Boundary penalties
    # x >= r  => r - x <= 0
    # x <= 1-r => x + r - 1 <= 0
    # y >= r  => r - y <= 0
    # y <= 1-r => y + r - 1 <= 0
    
    # Left wall: r - x
    viol = radii - centers[:, 0]
    loss += penalty_weight * np.sum(np.maximum(0, viol)**2)
    
    # Right wall: r + x - 1
    viol = radii + centers[:, 0] - 1.0
    loss += penalty_weight * np.sum(np.maximum(0, viol)**2)
    
    # Bottom wall: r - y
    viol = radii - centers[:, 1]
    loss += penalty_weight * np.sum(np.maximum(0, viol)**2)
    
    # Top wall: r + y - 1
    viol = radii + centers[:, 1] - 1.0
    loss += penalty_weight * np.sum(np.maximum(0, viol)**2)
    
    # Overlap penalties
    # Distance matrix
    # centers shape (n, 2)
    # diff shape (n, n, 2)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    # Radii sum matrix
    # radii shape (n,)
    rad_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    
    # Overlaps
    overlaps = rad_sum - dists
    
    # Set diagonal to -infinity to ignore self-overlap
    np.fill_diagonal(overlaps, -1e9)
    
    # Penalty for positive overlaps
    pos_overlaps = np.maximum(0, overlaps)
    loss += penalty_weight * np.sum(pos_overlaps**2)
    
    return loss

def get_initial_guess(n):
    """
    Generates an initial valid packing of n circles using a hexagonal grid.
    Returns flattened array [x0, y0, r0, ...]
    """
    # Try to fit n circles with a reasonable radius
    # Estimate radius based on area density of hexagonal packing ~ 0.9069
    # n * pi * r^2 = 0.9 * Area => r = sqrt(0.9 / (n * pi))
    # For n=26, r approx 0.105
    # Start slightly smaller to ensure validity
    r_start = 0.09
    
    centers_list = []
    
    # Hexagonal packing generation
    # Rows
    # y = r + k * sqrt(3) * r
    # Even rows (k=0, 2, ...): x = r, 3r, 5r ...
    # Odd rows (k=1, 3, ...): x = 2r, 4r, 6r ... (shifted by r)
    
    row_idx = 0
    while True:
        y = r_start + row_idx * np.sqrt(3) * r_start
        if y + r_start > 1.0:
            break
        
        if row_idx % 2 == 0:
            x_start = r_start
        else:
            x_start = 2 * r_start
        
        x = x_start
        while x + r_start <= 1.0:
            centers_list.append((x, y, r_start))
            x += 2 * r_start
        
        row_idx += 1
        if len(centers_list) >= n:
            break
            
    if len(centers_list) < n:
        # If not enough circles, decrease r_start and retry (simplified logic: just add random valid spots or smaller circles)
        # For robustness in this specific task, we assume 0.09 fits 26 easily. 
        # If not, we can just place remaining in available space or reduce r.
        # But 0.09 should fit ~30 circles in hex grid.
        pass
        
    # Take first n circles
    final_centers = centers_list[:n]
    
    # Flatten to [x0, y0, r0, x1, y1, r1, ...]
    params = []
    for c in final_centers:
        params.extend(c)
        
    return np.array(params)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    penalty_weight = 10000.0
    
    # Get initial guess
    x0 = get_initial_guess(n)
    
    # Bounds: x, y in [0, 1], r in [0, 0.5]
    bounds = [(0, 1), (0, 1), (0, 0.5)] * n
    
    best_params = x0

    for _ in range(5):

        current_x0 = x0.copy()

        # Keep radii fixed during perturbation to maintain validity roughly
        noise[2::3] = 0

        res = minimize(
            calculate_loss,
            current_x0,
            args=(n, penalty_weight),
            method='L-BFGS-B',
            bounds=bounds,
            options={'ftol': 1e-9, 'gtol': 1e-6, 'maxiter': 2000}
        )
        
        if res.fun < best_loss:
            best_loss = res.fun
            best_params = res.x
            
    # Extract centers and radii
    pts = best_params.reshape(n, 3)
    centers = pts[:, :2]
    radii = pts[:, 2]

    # This is a safety net for numerical precision

        overlaps = rad_sum - dists
        np.fill_diagonal(overlaps, 0)
        max_overlap = np.max(overlaps)
        
        # Check boundaries
        # x >= r
        viol = radii - centers[:, 0]
        max_viol = np.max(viol)
        if max_viol > max_overlap:
            max_overlap = max_viol
            
        # x <= 1-r => r <= 1-x

        if max_viol > max_overlap:
            max_overlap = max_viol
            
        # y >= r
        viol = radii - centers[:, 1]

        if max_viol > max_overlap:

        # y <= 1-r

            max_overlap = max_viol
            
        if max_overlap > 1e-12:
            # Reduce all radii slightly proportional to the overlap
            # A simple way is to reduce all radii by a small amount relative to overlap
            # Or just scale down. Scaling down might be safer.

        else:

    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
