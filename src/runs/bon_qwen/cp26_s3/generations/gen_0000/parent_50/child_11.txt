import numpy as np
from scipy.optimize import differential_evolution, linprog
from itertools import combinations

def get_optimal_radii_sum_and_radii(centers):
    """
    Given a set of centers, solve the LP to maximize sum of radii.
    Constraints:
    1. r_i >= 0
    2. r_i + r_j <= dist(i, j) for all pairs
    3. r_i <= min(x_i, 1-x_i, y_i, 1-y_i) for all i
    
    Returns (max_sum, radii_array)
    """
    n = centers.shape[0]
    
    # Variables: r_0, r_1, ..., r_{n-1}
    # Objective: Maximize sum(r_i) => Minimize -sum(r_i)
    c_obj = np.ones(n)
    
    # Constraints: A_ub * r <= b_ub
    
    # Boundary constraints: r_i <= bound_i
    # r_i <= x_i
    # r_i <= 1 - x_i
    # r_i <= y_i
    # r_i <= 1 - y_i
    # So r_i <= min(x_i, 1-x_i, y_i, 1-y_i)
    bounds_i = np.minimum(np.minimum(centers[:, 0], 1 - centers[:, 0]), 
                          np.minimum(centers[:, 1], 1 - centers[:, 1]))
    # Clamp bounds to be non-negative (though centers should be in [0,1])
    bounds_i = np.maximum(bounds_i, 0.0)
    
    # Matrix for boundary constraints: Identity matrix
    A_ub = np.eye(n)
    b_ub = bounds_i
    
    # Pairwise distance constraints: r_i + r_j <= dist_ij
    # This adds n*(n-1)/2 constraints
    n_pairs = n * (n - 1) // 2
    A_pair = np.zeros((n_pairs, n))
    b_pair = np.zeros(n_pairs)
    
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            A_pair[idx, i] = 1.0
            A_pair[idx, j] = 1.0
            b_pair[idx] = dist
            idx += 1
            
    # Combine constraints
    A_ub_full = np.vstack([A_ub, A_pair])
    b_ub_full = np.concatenate([b_ub, b_pair])
    
    # Bounds for variables (r_i >= 0)
    bounds_vars = [(0, None) for _ in range(n)]
    
    try:
        # Solve LP
        # linprog minimizes c.T x
        res = linprog(-c_obj, A_ub=A_ub_full, b_ub=b_ub_full, bounds=bounds_vars, method='highs')
        
        if res.success:
            return res.fun * -1, res.x
        else:
            # Fallback if LP fails (e.g. infeasible, though shouldn't happen with 0 radii)
            return 0.0, np.zeros(n)
    except Exception:
        return 0.0, np.zeros(n)

def fitness(centers_flat):
    """
    Wrapper for differential evolution.
    Converts flat array to centers, calls LP solver, returns negative sum (for minimization).
    """
    n = 26
    centers = centers_flat.reshape((n, 2))
    
    # Ensure centers are within bounds for the LP to make sense (though DE handles bounds)
    # If centers are outside, the LP might return 0 or weird results.
    # The DE bounds will keep them in [0, 1].
    
    total_sum, _ = get_optimal_radii_sum_and_radii(centers)
    
    # We want to maximize sum, so return negative
    return -total_sum

def generate_hex_grid_centers(n_points=26):
    """
    Generates initial centers on a hexagonal grid pattern.
    """
    centers = []
    # Estimate spacing to fit ~26 points
    # Area per point approx 1/26. Hex grid density factor sqrt(3)/2.
    # But simple heuristic: try spacing 0.15
    
    spacing = 0.15
    y = 0.1
    row = 0
    
    while len(centers) < n_points:
        x = 0.1
        # Offset alternate rows
        offset = (spacing / 2) if row % 2 == 1 else 0
        x += offset
        
        while x + 0.1 <= 1.0 and len(centers) < n_points:
            centers.append([x, y])
            x += spacing
        y += spacing * np.sqrt(3) / 2
        row += 1
        
    # If we didn't get enough, fill with random points or adjust spacing
    # Or just return the first 26
    centers = centers[:n_points]
    
    # If we have fewer than 26 (unlikely with this loop), fill randomly
    while len(centers) < n_points:
        centers.append([np.random.rand(), np.random.rand()])
        
    return np.array(centers)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Number of circles
    n = 26
    
    # 1. Initialization
    # Try a few random starts or fixed hex grid
    # Hex grid is a very good start
    init_centers = generate_hex_grid_centers(n)
    
    # Perturb slightly to avoid symmetry issues if any
    init_centers += np.random.normal(0, 0.005, init_centers.shape)
    # Clip to valid range
    init_centers = np.clip(init_centers, 0.01, 0.99)
    
    # 2. Optimization using Differential Evolution
    # Variables: 26 * 2 = 52 floats
    # Bounds: [0, 1] for all
    
    bounds = [(0.0, 1.0)] * (n * 2)
    
    # Differential Evolution parameters
    # popsize: number of candidates (default 15*D)
    # maxiter: number of generations
    # tol: tolerance
    # seed: for reproducibility
    
    try:
        result = differential_evolution(
            fitness,
            bounds=bounds,
            popsize=30, # Larger population for better global search
            maxiter=200, # Generations
            tol=1e-7,
            mutation=(0.5, 1.0),
            recombination=0.9,
            seed=42,
            polish=True, # Run a local optimizer at the end
            init='latinhypercube',
            workers=1 # Sequential to avoid issues
        )
        
        best_centers_flat = result.x
        best_centers = best_centers_flat.reshape((n, 2))
        
    except Exception as e:
        print(f"Optimization failed: {e}")
        # Fallback to init
        best_centers = init_centers

    # 3. Compute final radii for the best centers
    # Use the LP solver to get the optimal radii for this configuration
    total_sum, final_radii = get_optimal_radii_sum_and_radii(best_centers)
    
    # 4. Validation and Correction
    # Ensure radii are not negative
    final_radii = np.maximum(final_radii, 0.0)
    
    # Check for any numerical overlaps or boundary violations and fix if needed
    # Although LP guarantees validity, numerical precision might be an issue.
    # The validation function checks with tolerance 1e-12.
    
    # Just to be safe, if any radius is 0 (shouldn't be), force it small
    # But LP will give 0 if completely blocked, which is valid.
    
    return best_centers, final_radii, float(total_sum)
