import numpy as np
from scipy.optimize import minimize

def _get_initial_guess(n, seed_val):
    rng = np.random.default_rng(seed_val)
    # Hexagonal arrangement: 5 rows with alternating counts
    row_counts = [5, 6, 5, 6, 4]
    centers = np.zeros((n, 2))
    idx = 0
    y_spacing = 1.0 / 5.2
    x_spacing = 1.0 / 5.2
    
    for r_idx, count in enumerate(row_counts):
        if idx >= n: break
        y = (r_idx + 0.5) * y_spacing
        offset = (r_idx % 2) * x_spacing / 2
        for c in range(count):
            if idx >= n: break
            x = offset + (c + 0.5) * x_spacing
            centers[idx] = [x, y]
            idx += 1
            
    # Add controlled perturbation to avoid symmetric local minima
    centers += rng.uniform(-0.02, 0.02, size=centers.shape)
    centers = np.clip(centers, 0.05, 0.95)
    radii = np.ones(n) * 0.06
    return centers, radii

def _optimize_packing(centers, radii):
    n = len(radii)
    v0 = np.zeros(3 * n)
    for i in range(n):
        v0[3*i] = centers[i, 0]
        v0[3*i+1] = centers[i, 1]
        v0[3*i+2] = radii[i]
        
    bounds = [(0, 1)] * (2*n) + [(0, 0.5)] * n
    
    def objective(v):
        return -np.sum(v[2::3])
        
    def con_ineq(v):
        xs = v[0::3]
        ys = v[1::3]
        rs = v[2::3]
        
        # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
        b_cons = np.concatenate([xs - rs, 1 - xs - rs, ys - rs, 1 - ys - rs])
        
        # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
        diff_x = xs[:, None] - xs[None, :]
        diff_y = ys[:, None] - ys[None, :]
        dist_sq = diff_x**2 + diff_y**2
        rad_sum_sq = (rs[:, None] + rs[None, :])**2
        
        mask = np.triu(np.ones((n, n), dtype=bool), k=1)
        p_cons = (dist_sq - rad_sum_sq)[mask]
        return np.concatenate([b_cons, p_cons])

    cons = {'type': 'ineq', 'fun': con_ineq}
    
    res = minimize(objective, v0, method='SLSQP', bounds=bounds, constraints=cons, 
                   options={'maxiter': 2500, 'ftol': 1e-12, 'disp': False})
    return res.x, res.fun

def run_packing():
    n = 26
    best_sum = -1.0
    best_v = None
    
    # Multiple restarts to escape local optima
    starts = [0, 1, 2, 123, 456, 789]
    for seed in starts:

        v, obj = _optimize_packing(centers, radii)
        current_sum = -obj
        if current_sum > best_sum:
            best_sum = current_sum
            best_v = v
            
    centers_out = best_v[:2*n].reshape(n, 2)
    radii_out = best_v[2*n:]

    centers_out[:, 0] = np.clip(centers_out[:, 0], radii_out, 1.0 - radii_out)
    centers_out[:, 1] = np.clip(centers_out[:, 1], radii_out, 1.0 - radii_out)
    
    return centers_out, radii_out, float(np.sum(radii_out))
