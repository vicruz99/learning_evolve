import numpy as np
from scipy.optimize import minimize

def compute_penalty(centers, radii):
    """Compute penalty for constraint violations."""
    n = len(radii)
    # Boundary penalties: circle must be inside [0,1]x[0,1]
    pen = np.sum(np.maximum(0, radii - centers[:, 0])**2)
    pen += np.sum(np.maximum(0, radii - (1 - centers[:, 0]))**2)
    pen += np.sum(np.maximum(0, radii - centers[:, 1])**2)
    pen += np.sum(np.maximum(0, radii - (1 - centers[:, 1]))**2)
    
    # Overlap penalties: distance between centers >= sum of radii
    c_diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist = np.sqrt(np.sum(c_diff**2, axis=2))
    np.fill_diagonal(dist, np.inf)
    r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    overlaps = np.maximum(0, r_sum - dist)
    pen += np.sum(overlaps**2)
    
    return pen

def objective(params, n):
    """Objective function to maximize sum of radii with penalty for constraints."""
    centers = params[:2*n].reshape(n, 2)
    radii = params[2*n:]
    # Negative sum because minimize() finds minimum
    return -np.sum(radii) + 500.0 * compute_penalty(centers, radii)

def run_packing():
    n = 26

    res = minimize(objective, x0, args=(n,), method='L-BFGS-B', bounds=bounds,

    r_opt *= min_scale
    
    return c_opt, r_opt, float(np.sum(r_opt))
