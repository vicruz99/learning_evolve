import numpy as np
from scipy.optimize import minimize
import math

def objective(vars_flat):
    """Objective function: minimize negative sum of radii."""
    n = 26
    radii = vars_flat[2*n:]
    return -np.sum(radii)

def get_constraints(n):
    """
    Define constraints for the optimization problem.
    1. Circles inside [0, 1] x [0, 1]
    2. No overlaps between circles
    """
    constraints = []
    
    # Boundary constraints:
    # x_i - r_i >= 0  => r_i - x_i <= 0 (using inequality form g(x) <= 0)
    # Actually scipy SLSQP uses 'ineq' meaning g(x) >= 0.
    # So x_i - r_i >= 0 is correct form for 'ineq'.
    # x_i + r_i <= 1 => 1 - x_i - r_i >= 0
    # Same for y.
    
    def bound_x_minus_r(vars_flat):
        # x - r >= 0
        centers_x = vars_flat[0:n]
        radii = vars_flat[2*n:]
        return centers_x - radii

    def bound_x_plus_r(vars_flat):
        # 1 - x - r >= 0
        centers_x = vars_flat[0:n]
        radii = vars_flat[2*n:]
        return 1.0 - centers_x - radii

    def bound_y_minus_r(vars_flat):
        # y - r >= 0
        centers_y = vars_flat[n:2*n]
        radii = vars_flat[2*n:]
        return centers_y - radii

    def bound_y_plus_r(vars_flat):
        # 1 - y - r >= 0
        centers_y = vars_flat[n:2*n]
        radii = vars_flat[2*n:]
        return 1.0 - centers_y - radii

    constraints.append({'type': 'ineq', 'fun': bound_x_minus_r})
    constraints.append({'type': 'ineq', 'fun': bound_x_plus_r})
    constraints.append({'type': 'ineq', 'fun': bound_y_minus_r})
    constraints.append({'type': 'ineq', 'fun': bound_y_plus_r})

    # Non-overlap constraints:
    # dist^2 >= (r_i + r_j)^2
    # dist^2 - (r_i + r_j)^2 >= 0
    # (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
    
    # To avoid computing all pairs every time if not needed, but n=26 is small.
    # 325 constraints. Might be slow but acceptable.
    # Let's define a function that returns a vector of all pairwise constraints.
    
    def non_overlap(vars_flat):
        centers_x = vars_flat[0:n]
        centers_y = vars_flat[n:2*n]
        radii = vars_flat[2*n:]
        
        vals = []
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers_x[i] - centers_x[j]
                dy = centers_y[i] - centers_y[j]
                dist_sq = dx*dx + dy*dy
                r_sum = radii[i] + radii[j]
                vals.append(dist_sq - r_sum*r_sum)
        return np.array(vals)

    constraints.append({'type': 'ineq', 'fun': non_overlap})
    
    return constraints

def get_bounds(n):
    """
    Bounds for variables.
    x in [0, 1], y in [0, 1], r in [0, 1] (loose upper bound)
    Actually r cannot exceed 0.5, but 1 is safe.
    """
    bounds = []
    for _ in range(n):
        bounds.append((0.0, 1.0)) # x
    for _ in range(n):
        bounds.append((0.0, 1.0)) # y
    for _ in range(n):
        bounds.append((0.0, 0.5)) # r (radius cannot be > 0.5 in unit square)
    return bounds

def generate_hexagonal_init(n):

    Generate an initial configuration of n circles using a hexagonal lattice.
