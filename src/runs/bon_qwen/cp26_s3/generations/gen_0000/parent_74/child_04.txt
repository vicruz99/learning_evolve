import numpy as np
from scipy.optimize import minimize

def objective(vars_flat):
    n = 26
    # Maximize sum of radii -> minimize negative sum
    return -np.sum(vars_flat[2 * n:])

def constraints(vars_flat):
    n = 26
    centers = vars_flat[:2 * n].reshape(n, 2)
    radii = vars_flat[2 * n:]
    res = []
    # Circle-circle non-overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            res.append(dx * dx + dy * dy - (radii[i] + radii[j]) ** 2)
    # Boundary constraints
    for i in range(n):
        res.append(centers[i, 0] - radii[i])
        res.append(1.0 - centers[i, 0] - radii[i])
        res.append(centers[i, 1] - radii[i])
        res.append(1.0 - centers[i, 1] - radii[i])
    return np.array(res)

def get_hex_initialization(n):
    centers = []
    row = 0
    while len(centers) < n:
        for col in range(7):
            x = col * 0.15 + (row % 2) * 0.075 + 0.05
            y = row * 0.13 + 0.05
            if len(centers) < n:
                centers.append([x, y])
        row += 1
    return np.array(centers)

def run_packing():
    n = 26
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = 0.0
    best_result = None
    
    # Prepare initializations: 1 hex grid + 3 random
    initializations = [get_hex_initialization(n)]
    for _ in range(3):
        np.random.seed()
        centers = np.random.rand(n, 2) * 0.6 + 0.2
        initializations.append(centers)
        
    for init_centers in initializations:
        radii = np.full(n, 0.03)
        x0 = np.concatenate([init_centers.ravel(), radii])
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, 
                           constraints=cons, options={'maxiter': 3000, 'ftol': 1e-9})
            if np.sum(res.x[2 * n:]) > best_sum:
                best_sum = np.sum(res.x[2 * n:])
                best_result = res.x
        except Exception:
            continue
            
    if best_result is not None:
        centers = best_result[:2 * n].reshape(n, 2)
        radii = best_result[2 * n:]
    else:
        # Fallback safe configuration
        centers = get_hex_initialization(n)
        radii = np.full(n, 0.05)
        
    # Slight shrink to guarantee strict validity against numerical tolerances
    radii *= 0.9999
    centers = np.clip(centers, radii, 1.0 - radii)
    
    return centers, radii, np.sum(radii)
