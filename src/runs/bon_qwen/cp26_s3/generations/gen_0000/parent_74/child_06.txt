import numpy as np
from scipy.optimize import minimize

def compute_radii(centers):
    """Calculate maximum valid radii for given centers based on boundaries and mutual distances."""
    n = centers.shape[0]
    x = centers[:, 0]
    y = centers[:, 1]
    
    # Distance to square boundaries
    r_bound = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
    
    # Pairwise distances
    dists = np.linalg.norm(centers[:, np.newaxis, :] - centers[np.newaxis, :, :], axis=2)
    np.fill_diagonal(dists, np.inf)  # Ignore self-distance
    r_circle = np.min(dists / 2.0, axis=1)
    
    # Effective radius is the tighter constraint
    return np.minimum(r_bound, r_circle)

def objective(p):
    """Objective function: negative sum of radii. Optimizer minimizes this."""
    # Transform unconstrained parameters to (0,1) interval
    centers = 1.0 / (1.0 + np.exp(-p)).reshape(-1, 2)
    radii = compute_radii(centers)
    return -np.sum(radii)

    for seed in range(5):

                    x += x_step / 2.0
                centers[idx] = [x, y]
                idx += 1

    return best_centers, best_radii, float(best_sum)
