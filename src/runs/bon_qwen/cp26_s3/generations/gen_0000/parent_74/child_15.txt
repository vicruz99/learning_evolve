To maximize the sum of radii for 26 circles in a unit square, we treat the problem as a continuous optimization task. The constraints are non-convex (distance constraints involving radii), but we can decouple the problem:
1.  **LP Sub-problem**: For any fixed set of centers, finding the optimal radii is a Linear Programming (LP) problem. We maximize $\sum r_i$ subject to $r_i + r_j \le d_{ij}$ (distance constraints) and $r_i \le b_i$ (boundary constraints).
2.  **Center Optimization**: We optimize the centers to maximize the objective value of the LP. We use a force-directed heuristic: if circles are "touching" ($r_i + r_j \approx d_{ij}$) or touching the boundary, we apply repulsive forces to push them apart/inward, creating space for larger radii.

The algorithm iteratively solves the LP to get current radii, calculates repulsion forces based on active constraints, updates centers, and repeats. We start with a hexagonal grid initialization to ensure a good geometric baseline.

import numpy as np
from scipy.optimize import linprog

def get_optimal_radii(centers):
    """
    Solves the LP to find optimal radii for fixed centers.
    Maximizes sum(r_i) subject to non-overlap and boundary constraints.
    """
    n = len(centers)
    c_obj = -np.ones(n)  # Maximize sum(r) => Minimize -sum(r)
    
    constraints = []
    bounds_rhs = []
    
    # Pairwise distance constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.sqrt(np.sum((centers[i] - centers[j])**2))
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            constraints.append(row)
            bounds_rhs.append(d)
            
    # Boundary constraints: r_i <= x_i, 1-x_i, y_i, 1-y_i
    for i in range(n):
        x, y = centers[i]
        row = np.zeros(n)
        row[i] = 1.0
        
        constraints.append(row); bounds_rhs.append(x)
        constraints.append(row); bounds_rhs.append(1.0 - x)
        constraints.append(row); bounds_rhs.append(y)
        constraints.append(row); bounds_rhs.append(1.0 - y)
        
    A_ub = np.array(constraints)
    b_ub = np.array(bounds_rhs)
    bnds = [(0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bnds, method='highs')
        if res.success:
            return res.x, -res.fun
    except Exception:
        pass
        
    return np.zeros(n), 0.0

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # 1. Initialize centers with a hexagonal grid
    centers = []
    r_est = 0.1
    # 6 rows of hexagonal packing
    for row_idx in range(6):
        y = r_est + row_idx * r_est * np.sqrt(3)
        if y > 1 - r_est: break
        x_start = r_est if row_idx % 2 == 0 else r_est * 2
        while x_start < 1 - r_est:
            centers.append([x_start, y])
            x_start += 2 * r_est
            if len(centers) >= n: break
        if len(centers) >= n: break
        
    # If we don't have 26, fill remaining randomly (should not happen with this logic)
    while len(centers) < n:
        centers.append([np.random.rand(), np.random.rand()])
    centers = np.array(centers[:n])
    
    best_centers = centers.copy()
    best_radii = np.zeros(n)
    best_sum = 0.0
    
    # 2. Optimization Loop
    num_iters = 2000
    step_size = 0.005
    tol = 1e-4
    
    for it in range(num_iters):
        # Solve LP for radii
        radii, current_sum = get_optimal_radii(centers)
        
        if current_sum > best_sum:
            best_sum = current_sum
            best_radii = radii.copy()
            best_centers = centers.copy()
            
        # Calculate forces
        forces = np.zeros_like(centers)
        
        # Pairwise repulsion
        for i in range(n):
            for j in range(i + 1, n):
                d = np.sqrt(np.sum((centers[i] - centers[j])**2))
                if d > 1e-6:
                    # If touching or overlapping
                    if radii[i] + radii[j] >= d - tol:
                        vec = (centers[i] - centers[j]) / d
                        forces[i] += vec
                        forces[j] -= vec
                        
        # Boundary repulsion
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if r >= x - tol: forces[i][0] += 1.0
            if r >= 1 - x - tol: forces[i][0] -= 1.0
            if r >= y - tol: forces[i][1] += 1.0
            if r >= 1 - y - tol: forces[i][1] -= 1.0
            
        # Update centers
        # Normalize forces to prevent explosion
        magnitudes = np.linalg.norm(forces, axis=1)
        magnitudes[magnitudes == 0] = 1.0
        forces = forces / magnitudes[:, np.newaxis]
        
        # Adaptive step size (annealing)
        current_step = step_size * (1.0 - it / num_iters)
        centers += forces * current_step
        
        # Clamp to valid range [0, 1]
        centers = np.clip(centers, 1e-5, 1 - 1e-5)
        
    # Final validation and adjustment
    final_radii, final_sum = get_optimal_radii(best_centers)
    
    return best_centers, final_radii, final_sum
