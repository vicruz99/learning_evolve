import numpy as np
from scipy.optimize import minimize

def compute_loss(params, n):
    centers = params[:2*n].reshape(n, 2)
    radii = params[2*n:]
    
    # Objective: maximize sum of radii -> minimize negative sum
    obj = -np.sum(radii)
    
    # Boundary penalty: circles must be inside [0,1]x[0,1]
    b1 = np.maximum(0, radii - centers[:, 0])
    b2 = np.maximum(0, radii + centers[:, 0] - 1)
    b3 = np.maximum(0, radii - centers[:, 1])
    b4 = np.maximum(0, radii + centers[:, 1] - 1)
    boundary_pen = np.sum(b1**2 + b2**2 + b3**2 + b4**2)
    
    # Overlap penalty: distance between centers >= sum of radii
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    r_sums = radii[:, np.newaxis] + radii[np.newaxis, :]
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    overlaps = np.maximum(0, r_sums - dists)
    overlap_pen = np.sum(overlaps[mask]**2)
    
    # High penalty weight ensures constraints are respected while maximizing radii
    return obj + 5000.0 * (boundary_pen + overlap_pen)

def run_packing():
    np.random.seed(42)
    n = 26
    
    # Initialize with a 5x5 grid plus one center circle
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.1)
    
    idx = 0
    for i in range(5):
        for j in range(5):
            centers[idx] = [0.1 + i * 0.2, 0.1 + j * 0.2]
            idx += 1
    centers[idx] = [0.5, 0.5]
    radii[idx] = 0.05
    
    # Add random jitter to break symmetry and aid optimization
    centers += np.random.uniform(-0.02, 0.02, size=centers.shape)
    centers = np.clip(centers, 0.01, 0.99)
    
    params = np.concatenate([centers.flatten(), radii])
    bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n
    
    best_params = params
    best_loss = compute_loss(params, n)
    
    # Multiple restarts to escape local optima
    for _ in range(8):
        res = minimize(compute_loss, params, args=(n,), method='L-BFGS-B', 
                       bounds=bounds, options={'maxiter': 3000, 'ftol': 1e-12})
        if res.fun < best_loss:
            best_loss = res.fun
            best_params = res.x
        # Perturb for next iteration
        params = best_params + np.random.normal(0, 0.005, size=best_params.shape)
        
    centers = best_params[:2*n].reshape(n, 2)
    radii = best_params[2*n:]
    
    # Post-processing: strictly enforce non-overlap and boundaries
    for _ in range(10):
        diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        r_sums = radii[:, np.newaxis] + radii[np.newaxis, :]
        mask = np.triu(np.ones((n, n), dtype=bool), k=1)
        max_viol = np.max(r_sums[mask] - dists[mask])
        if max_viol > 1e-9:
            radii -= max_viol / 2.0
        else:
            break
            
    # Final boundary clamping
    radii = np.minimum(radii, np.minimum(centers[:,0], 1.0 - centers[:,0]))
    radii = np.minimum(radii, np.minimum(centers[:,1], 1.0 - centers[:,1]))
    radii = np.maximum(radii, 0.0)
    
    return centers, radii, float(np.sum(radii))
