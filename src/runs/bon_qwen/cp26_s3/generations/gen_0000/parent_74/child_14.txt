I'll use a multi-phase optimization strategy:

1. **Initial Configuration**: Use a force-based simulation to pack circles efficiently. Start with a hexagonal-inspired grid and let repulsive forces settle circles into good positions while gradually expanding radii.

2. **Key Insight**: To maximize sum of radii (not just minimum radius), we should use unequal circles. Some circles can be larger if they're in corners/edges where they don't compete with as many neighbors.

3. **Phase 1 - Force-based simulation**: 
   - Initialize circles in a grid/hexagonal pattern
   - Apply repulsive forces between overlapping circles
   - Apply boundary forces to keep circles inside
   - Gradually expand radii while maintaining non-overlap
   - This gives a good feasible starting point

4. **Phase 2 - Local optimization with scipy**:
   - Use a penalty-based objective with scipy's optimization
   - Multiple restarts from different initial configurations
   - The penalty method handles the non-convex constraints

5. **Phase 3 - Post-processing**:
   - Ensure all constraints are satisfied
   - Try to "grow" each circle individually while pushing others apart
   - This greedy expansion can squeeze out extra radius

6. **Important considerations**:
   - Use high penalty weights to enforce constraints strictly
   - Run multiple trials with different initializations
   - Use both Nelder-Mead and BFGS/L-BFGS-B optimizers
   - Add random perturbations to escape local optima

import numpy as np
from scipy.optimize import minimize
from scipy.optimize import dual_annealing

N_CIRCLES = 26


def compute_penalty_and_objective(params):
    """Combined objective + penalty function for optimization."""
    n = N_CIRCLES
    centers = params[:2 * n].reshape(n, 2)
    radii = params[2 * n:]

    # Objective: negative sum of radii (we minimize)
    obj = -np.sum(radii)

    # Penalty for constraint violations
    penalty = 0.0
    weight = 2000.0

    # Boundary penalties
    for i in range(n):
        r = radii[i]
        if r < 0:
            penalty += weight * 10.0 * r * r
        x, y = centers[i]
        if x - r < 0:
            penalty += weight * (x - r) * (x - r)
        if x + r > 1:
            penalty += weight * (x + r - 1) * (x + r - 1)
        if y - r < 0:
            penalty += weight * (y - r) * (y - r)
        if y + r > 1:
            penalty += weight * (y + r - 1) * (y + r - 1)

    # Overlap penalties
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist_sq = dx * dx + dy * dy
            dist = np.sqrt(dist_sq) if dist_sq > 1e-20 else 1e-10
            overlap = radii[i] + radii[j] - dist
            if overlap > 0:
                penalty += weight * overlap * overlap

    return obj + penalty


def force_based_simulation(initial_centers, initial_radii, n, num_iters=30000):
    """Force-based simulation to pack circles."""
    centers = initial_centers.copy()
    radii = initial_radii.copy()

    velocities = np.zeros((n, 2))
    dt = 0.0005
    K_repulse = 500.0
    K_boundary = 500.0
    damping = 0.85

    for iteration in range(num_iters):
        forces = np.zeros((n, 2))

        # Circle-circle repulsion
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist_sq = dx * dx + dy * dy
                dist = np.sqrt(dist_sq) if dist_sq > 1e-16 else 1e-8

                if dist < 1e-8:
                    # Push apart randomly
                    dx = np.random.randn() * 0.01
                    dy = np.random.randn() * 0.01
                    dist = np.sqrt(dx * dx + dy * dy)

                min_sep = radii[i] + radii[j]
                if dist < min_sep * 1.05:
                    overlap = min_sep - dist
                    if overlap > 0:
                        force_mag = K_repulse * overlap / (dist + 1e-10)
                        fx = force_mag * dx
                        fy = force_mag * dy
                        forces[i, 0] += fx
                        forces[i, 1] += fy
                        forces[j, 0] -= fx
                        forces[j, 1] -= fy

        # Boundary repulsion
        for i in range(n):
            r = radii[i]
            if centers[i, 0] < r + 0.001:
                forces[i, 0] += K_boundary * (r + 0.001 - centers[i, 0])
            if centers[i, 0] > 1 - r - 0.001:
                forces[i, 0] -= K_boundary * (centers[i, 0] - (1 - r - 0.001))
            if centers[i, 1] < r + 0.001:
                forces[i, 1] += K_boundary * (r + 0.001 - centers[i, 1])
            if centers[i, 1] > 1 - r - 0.001:
                forces[i, 1] -= K_boundary * (centers[i, 1] - (1 - r - 0.001))

        # Update
        velocities += dt * forces
        velocities *= damping
        centers += dt * velocities

        # Clip centers
        centers[:, 0] = np.clip(centers[:, 0], 0.001, 0.999)
        centers[:, 1] = np.clip(centers[:, 1], 0.001, 0.999)

        # Expand radii periodically
        if iteration % 5 == 0:
            for i in range(n):
                max_r = min(
                    centers[i, 0] - 0.0005,
                    1 - centers[i, 0] - 0.0005,
                    centers[i, 1] - 0.0005,
                    1 - centers[i, 1] - 0.0005
                )
                for j in range(n):
                    if i != j:
                        dx = centers[i, 0] - centers[j, 0]
                        dy = centers[i, 1] - centers[j, 1]
                        dist = np.sqrt(dx * dx + dy * dy)
                        max_r = min(max_r, dist - radii[j] - 0.0005)

                if max_r > 0:
                    target = max_r
                    radii[i] = radii[i] + 0.002 * (target - radii[i])

    return centers, radii


def make_initialization_grid():
    """Create grid-based initialization."""
    n = N_CIRCLES
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.015

    idx = 0
    for i in range(5):
        for j in range(5):
            centers[idx, 0] = (i + 1) / 6.0
            centers[idx, 1] = (j + 1) / 6.0
            idx += 1
    centers[n - 1, 0] = 0.5 / 6.0
    centers[n - 1, 1] = 0.5 / 6.0

    return centers, radii


def make_initialization_hexagonal():
    """Create hexagonal-inspired initialization."""
    n = N_CIRCLES
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.015

    spacing_x = 1.0 / 6.0
    spacing_y = spacing_x * np.sqrt(3) / 2

    idx = 0
    row = 0
    while idx < n:
        y = (row + 1) * spacing_y
        x_offset = spacing_x / 2 if row % 2 == 1 else 0
        max_cols = 6
        for col in range(max_cols):
            if idx >= n:
                break
            x = (col + 1) * spacing_x + x_offset
            if x < 0.95 and y < 0.95:
                centers[idx] = [x, y]
                idx += 1
        row += 1

    return centers, radii


def make_initialization_corners():
    """Create initialization with emphasis on corners and edges."""
    n = N_CIRCLES
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.015

    # 4 corners
    corner_positions = [[0.1, 0.1], [0.9, 0.1], [0.1, 0.9], [0.9, 0.9]]
    for i, pos in enumerate(corner_positions):
        centers[i] = pos
        radii[i] = 0.03

    # 8 edge centers
    edge_positions = [
        [0.5, 0.08], [0.08, 0.5], [0.92, 0.5], [0.5, 0.92],
        [0.25, 0.08], [0.75, 0.08], [0.25, 0.92], [0.75, 0.92]
    ]
    for i, pos in enumerate(edge_positions):
        centers[i + 4] = pos
        radii[i + 4] = 0.02

    # Remaining 14 in interior
    remaining = n - 12
    interior_positions = []
    for i in range(3):
        for j in range(3):
            interior_positions.append([0.3 + i * 0.2, 0.3 + j * 0.2])
    for i, pos in enumerate(interior_positions[:remaining]):
        centers[12 + i] = pos

    return centers, radii


def make_initialization_random():
    """Create random initialization."""
    n = N_CIRCLES
    centers = np.random.uniform(0.1, 0.9, (n, 2))
    radii = np.ones(n) * 0.015
    return centers, radii


def optimize_from_initialization(centers_init, radii_init, n, optimizer_type='nelder'):
    """Run optimization from a given initialization."""
    x0 = np.concatenate([centers_init.flatten(), radii_init])

    # Bounds: x, y in [0, 1], r in [0, 0.5]
    bounds = []
    for i in range(n):
        bounds.append((0, 1))
        bounds.append((0, 1))
        bounds.append((0, 0.45))

    if optimizer_type == 'nelder':
        result = minimize(
            compute_penalty_and_objective,
            x0,
            method='Nelder-Mead',
            options={'maxiter': 150000, 'xatol': 1e-13, 'fatol': 1e-13, 'adaptive': True}
        )
    elif optimizer_type == 'lbfgs':
        result = minimize(
            compute_penalty_and_objective,
            x0,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 100000, 'ftol': 1e-15}
        )
    else:
        result = minimize(
            compute_penalty_and_objective,
            x0,
            method='Powell',
            options={'maxiter': 100000, 'xtol': 1e-13, 'ftol': 1e-13}
        )

    centers_opt = result.x[:2 * n].reshape(n, 2)
    radii_opt = result.x[2 * n:]

    return centers_opt, radii_opt, result.fun


def fix_and_validate(centers, radii, n):
    """Fix constraint violations and ensure validity."""
    centers = centers.copy()
    radii = radii.copy()

    # Ensure non-negative radii
    radii = np.maximum(radii, 0)

    # Clip to boundary constraints
    for i in range(n):
        r = radii[i]
        centers[i, 0] = np.clip(centers[i, 0], r, 1 - r)
        centers[i, 1] = np.clip(centers[i, 1], r, 1 - r)
        radii[i] = min(r, centers[i, 0], 1 - centers[i, 0],
                       centers[i, 1], 1 - centers[i, 1])

    # Iteratively resolve overlaps
    for _ in range(500):
        max_overlap = 0
        max_pair = (-1, -1)
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                overlap = radii[i] + radii[j] - dist
                if overlap > max_overlap:
                    max_overlap = overlap
                    max_pair = (i, j)

        if max_overlap < 1e-14:
            break

        i, j = max_pair
        # Reduce radii proportionally
        total_r = radii[i] + radii[j]
        if total_r > 0:
            frac_i = radii[i] / total_r
            frac_j = radii[j] / total_r
            radii[i] -= max_overlap * frac_i * 0.99
            radii[j] -= max_overlap * frac_j * 0.99

    radii = np.maximum(radii, 0)

    # Final boundary check
    for i in range(n):
        r = radii[i]
        centers[i, 0] = np.clip(centers[i, 0], r, 1 - r)
        centers[i, 1] = np.clip(centers[i, 1], r, 1 - r)
        radii[i] = min(r, centers[i, 0], 1 - centers[i, 0],
                       centers[i, 1], 1 - centers[i, 1])

    return centers, radii


def greedy_expand(centers, radii, n, num_rounds=50):
    """Try to expand each circle greedily."""
    centers = centers.copy()
    radii = radii.copy()

    for round_idx in range(num_rounds):
        for i in range(n):
            # Find max possible radius for circle i
            max_r = min(
                centers[i, 0],
                1 - centers[i, 0],
                centers[i, 1],
                1 - centers[i, 1]
            )
            for j in range(n):
                if i != j:
                    dx = centers[i, 0] - centers[j, 0]
                    dy = centers[i, 1] - centers[j, 1]
                    dist = np.sqrt(dx * dx + dy * dy)
                    max_r = min(max_r, dist - radii[j])

            max_r = max(max_r, 0)
            if max_r > radii[i] + 1e-10:
                # Try to expand
                new_r = (radii[i] + max_r) / 2
                radii[i] = new_r

                # Push neighbors away if needed
                for j in range(n):
                    if i != j:
                        dx = centers[j, 0] - centers[i, 0]
                        dy = centers[j, 1] - centers[i, 1]
                        dist = np.sqrt(dx * dx + dy * dy)
                        min_dist = radii[i] + radii[j]
                        if dist < min_dist and dist > 1e-10:
                            # Push j away
                            push = (min_dist - dist) * 0.5
                            dx_norm = dx / dist
                            dy_norm = dy / dist
                            centers[j, 0] += push * dx_norm
                            centers[j, 1] += push * dy_norm

                            # Clip
                            r_j = radii[j]
                            centers[j, 0] = np.clip(centers[j, 0], r_j, 1 - r_j)
                            centers[j, 1] = np.clip(centers[j, 1], r_j, 1 - r_j)

        # Fix any boundary violations
        for i in range(n):
            r = radii[i]
            centers[i, 0] = np.clip(centers[i, 0], r, 1 - r)
            centers[i, 1] = np.clip(centers[i, 1], r, 1 - r)
            radii[i] = min(r, centers[i, 0], 1 - centers[i, 0],
                          centers[i, 1], 1 - centers[i, 1])

    return centers, radii


def run_packing():
    """Main function to pack 26 circles in a unit square."""
    n = N_CIRCLES
    best_centers = None
    best_radii = None
    best_sum = -np.inf

    # Phase 1: Force-based simulation from multiple initializations
    init_funcs = [
        make_initialization_grid,
        make_initialization_hexagonal,
        make_initialization_corners,
        make_initialization_random,
    ]

    force_results = []
    for init_func in init_funcs:
        centers_init, radii_init = init_func()
        centers_sim, radii_sim = force_based_simulation(
            centers_init, radii_init, n, num_iters=20000
        )
        force_results.append((centers_sim, radii_sim))

    # Phase 2: Optimize from each force simulation result
    for centers_sim, radii_sim in force_results:
        for opt_type in ['nelder', 'lbfgs', 'powell']:
            centers_opt, radii_opt, _ = optimize_from_initialization(
                centers_sim, radii_sim, n, opt_type
            )
            centers_opt, radii_opt = fix_and_validate(centers_opt, radii_opt, n)
            current_sum = np.sum(radii_opt)
            if current_sum > best_sum:
                best_sum = current_sum
                best_centers = centers_opt.copy()
                best_radii = radii_opt.copy()

    # Phase 3: Optimize from raw initializations without force simulation
    for init_func in init_funcs:
        centers_init, radii_init = init_func()
        for opt_type in ['nelder', 'lbfgs']:
            centers_opt, radii_opt, _ = optimize_from_initialization(
                centers_init, radii_init, n, opt_type
            )
            centers_opt, radii_opt = fix_and_validate(centers_opt, radii_opt, n)
            current_sum = np.sum(radii_opt)
            if current_sum > best_sum:
                best_sum = current_sum
                best_centers = centers_opt.copy()
                best_radii = radii_opt.copy()

    # Phase 4: Greedy expansion on best result
    if best_centers is not None:
        best_centers, best_radii = greedy_expand(best_centers, best_radii, n, 100)
        best_sum = np.sum(best_radii)

    # Phase 5: Final local optimization on the best result
    x0 = np.concatenate([best_centers.flatten(), best_radii])
    result = minimize(
        compute_penalty_and_objective,
        x0,
        method='Nelder-Mead',
        options={'maxiter': 200000, 'xatol': 1e-14, 'fatol': 1e-14, 'adaptive': True}
    )
    best_centers = result.x[:2 * n].reshape(n, 2)
    best_radii = result.x[2 * n:]
    best_centers, best_radii = fix_and_validate(best_centers, best_radii, n)

    # Phase 6: One more round of greedy expansion
    best_centers, best_radii = greedy_expand(best_centers, best_radii, n, 80)
    best_sum = np.sum(best_radii)

    # Final validation and fix
    best_centers, best_radii = fix_and_validate(best_centers, best_radii, n)
    best_sum = np.sum(best_radii)

    return best_centers, best_radii, best_sum
