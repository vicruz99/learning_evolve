import numpy as np
from scipy.optimize import minimize

N = 26

def objective(v):
    """Objective function: minimize negative sum of radii"""
    return -np.sum(v[2::3])

def con_wall_x_min(v):
    """Constraint: x_i >= r_i"""
    return v[0::3] - v[2::3]

def con_wall_x_max(v):
    """Constraint: x_i <= 1 - r_i"""
    return 1.0 - v[0::3] - v[2::3]

def con_wall_y_min(v):
    """Constraint: y_i >= r_i"""
    return v[1::3] - v[2::3]

def con_wall_y_max(v):
    """Constraint: y_i <= 1 - r_i"""
    return 1.0 - v[1::3] - v[2::3]

def con_overlap(v):
    """Constraint: dist^2 >= (r_i + r_j)^2 for all pairs"""
    res = np.empty(N * (N - 1) // 2)
    idx = 0
    for i in range(N):
        xi, yi, ri = v[3 * i], v[3 * i + 1], v[3 * i + 2]
        for j in range(i + 1, N):
            dx = xi - v[3 * j]
            dy = yi - v[3 * j + 1]
            dr = ri + v[3 * j + 2]
            res[idx] = dx * dx + dy * dy - dr * dr
            idx += 1
    return res

def get_constraints():
    """Return list of constraint dictionaries for SLSQP"""
    return [
        {'type': 'ineq', 'fun': con_wall_x_min},
        {'type': 'ineq', 'fun': con_wall_x_max},
        {'type': 'ineq', 'fun': con_wall_y_min},
        {'type': 'ineq', 'fun': con_wall_y_max},
        {'type': 'ineq', 'fun': con_overlap}
    ]

def get_initial_guess():
    """Generate a feasible initial configuration based on hexagonal lattice"""
    pts = []
    rows = 5
    cols = 6
    s = 0.18
    for r in range(rows):
        y = 0.2 + r * s * 0.866
        shift = (r % 2) * s / 2
        for c in range(cols):
            if len(pts) == N:
                break
            x = 0.2 + c * s + shift
            pts.append((x, y))
    
    xs = np.array([p[0] for p in pts])
    ys = np.array([p[1] for p in pts])
    
    def norm(arr):
        mn, mx = np.min(arr), np.max(arr)
        return 0.25 + (arr - mn) / (mx - mn) * 0.5 if mx > mn else np.full_like(arr, 0.5)
    
    xs = norm(xs)
    ys = norm(ys)
    
    v = np.zeros(3 * N)
    v[0::3] = xs
    v[1::3] = ys
    v[2::3] = 0.04  # Initial small radius
    return v

def run_packing():
    """Optimize circle packing to maximize sum of radii"""
    np.random.seed(42)
    best_v = None
    best_obj = np.inf
    constraints = get_constraints()
    bounds = [(0.0, 1.0) if i % 3 != 2 else (0.001, 0.5) for i in range(3 * N)]

    for trial in range(5):
        x0 = get_initial_guess()
        if trial > 0:
            # Perturb initial guess to escape local optima
            noise = np.random.normal(0, 0.015, size=x0.shape)
            x0 = x0 + noise
            x0[0::3] = np.clip(x0[0::3], 0.05, 0.95)
            x0[1::3] = np.clip(x0[1::3], 0.05, 0.95)
            x0[2::3] = np.clip(x0[2::3], 0.02, 0.4)
        
        res = minimize(
            objective, 
            x0, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=constraints, 
            options={'maxiter': 4000, 'ftol': 1e-11, 'disp': False}
        )
        
        if res.success and res.fun < best_obj - 1e-8:
            best_obj = res.fun
            best_v = res.x.copy()
            
    if best_v is None:
        # Fallback to base guess if all optimizations fail
        best_v = get_initial_guess()
        
    centers = np.column_stack((best_v[0::3], best_v[1::3]))
    radii = best_v[2::3]
    sum_r = np.sum(radii)
    
    # Ensure strict feasibility against numerical drift
    centers = np.clip(centers, 0.0, 1.0)
    radii = np.maximum(radii, 1e-6)
    
    return centers, radii, float(sum_r)
