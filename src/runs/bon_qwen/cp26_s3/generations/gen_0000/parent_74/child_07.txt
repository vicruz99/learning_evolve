The problem thus reduces to finding the largest radius $r$ such that 26 equal circles fit in $[0,1]^2$.

Known optimal packings for $n=26$ involve hexagonal lattice arrangements perturbed to fit the square boundaries. A 5-row pattern with circle counts `[5, 6, 5, 6, 4]` (summing to 26) provides an excellent initial configuration.

2. **Optimization**: Maximize $r$ subject to:
   - Boundary constraints: $r \le x_i, y_i \le 1-r$
   - Non-overlap constraints: $\|c_i - c_j\| \ge 2r$
   The SLSQP solver handles these inequality constraints efficiently.
3. **Fallback**: If SLSQP fails to converge, a penalty-based method with BFGS is used as a robust alternative.
4. **Validation & Safety**: Apply a tiny safety factor to the final radius to ensure strict compliance with the validation function's tolerance ($10^{-12}$), then return the configuration.

import numpy as np
import scipy.optimize as opt

def _objective(vars, n):
    """Objective: maximize radius R (minimize -R)"""
    return -vars[0]

def _constraints(vars, n):
    """Constraints: boundary and non-overlap"""
    R = vars[0]
    c = vars[1:].reshape(n, 2)
    res = []
    
    # Boundary constraints: R <= x_i <= 1-R, R <= y_i <= 1-R
    for i in range(n):
        res.append(c[i, 0] - R)
        res.append(1.0 - R - c[i, 0])
        res.append(c[i, 1] - R)
        res.append(1.0 - R - c[i, 1])
        
    # Overlap constraints: dist^2 >= (2R)^2
    for i in range(n):
        for j in range(i + 1, n):
            d2 = (c[i, 0] - c[j, 0])**2 + (c[i, 1] - c[j, 1])**2
            res.append(d2 - 4 * R * R)
            
    return np.array(res)

def _penalty(vars, n, lam):
    """Fallback penalty objective for packing"""
    R = vars[0]
    c = vars[1:].reshape(n, 2)
    p = 0.0
    for i in range(n):
        p += max(0, R - c[i,0])**2 + max(0, c[i,0] - (1-R))**2
        p += max(0, R - c[i,1])**2 + max(0, c[i,1] - (1-R))**2
    for i in range(n):
        for j in range(i+1, n):
            d = np.hypot(c[i,0]-c[j,0], c[i,1]-c[j,1])
            if d < 2*R:
                p += (2*R - d)**2
    return -R + lam*p

def run_packing():
    n = 26
    np.random.seed(42)
    
    # Strategy: Start from a perturbed hexagonal arrangement which is dense.
    # 5 rows with counts 5, 6, 5, 6, 4 sum to 26.
    r_init = 0.07
    centers = np.zeros((n, 2))
    idx = 0
    y_step = np.sqrt(3) * r_init
    row_counts = [5, 6, 5, 6, 4]
    
    for r_idx, count in enumerate(row_counts):
        y = r_init + r_idx * y_step
        x_start = (1.0 - (count - 1) * 2 * r_init) / 2.0
        for c_idx in range(count):
            x = x_start + c_idx * 2 * r_init
            centers[idx] = [x, y]
            idx += 1
            
    # Add small random perturbation to break symmetry and avoid local minima
    centers += np.random.normal(0, 0.002, size=centers.shape)
    # Ensure initial strict feasibility
    centers = np.clip(centers, r_init + 1e-4, 1.0 - r_init - 1e-4)
    
    x0 = np.concatenate([[r_init], centers.flatten()])
    
    # Setup constraints for SLSQP
    constr = {
        'type': 'ineq',
        'fun': _constraints,
        'args': (n,)
    }
    
    bounds = [(0, 0.5)] + [(0, 1)] * (2 * n)
    
    # Primary optimization
    res = opt.minimize(_objective, x0, args=(n,), method='SLSQP', 
                       bounds=bounds, constraints=constr,
                       options={'maxiter': 3000, 'ftol': 1e-12})
                       
    if res.success:
        R_opt = res.x[0]
        c_opt = res.x[1:].reshape(n, 2)
    else:
        # Fallback: Penalty method if SLSQP fails
        res2 = opt.minimize(_penalty, x0, args=(n, 50000), method='BFGS', 
                            options={'maxiter': 3000})
        R_opt = res2.x[0]
        c_opt = res2.x[1:].reshape(n, 2)

    # Safety margin to satisfy validation tolerance
    R_final = R_opt * 0.9995
    centers_final = c_opt
    
    # Verify and adjust if necessary
    for i in range(n):
        for j in range(i+1, n):
            d = np.linalg.norm(centers_final[i] - centers_final[j])
            if d < 2*R_final - 1e-9:
                R_final = (d + 1e-9) / 2.0
        d_wall = min(centers_final[i,0], 1-centers_final[i,0], 
                     centers_final[i,1], 1-centers_final[i,1])
        if d_wall < R_final - 1e-9:
            R_final = d_wall + 1e-9

    radii_final = np.full(n, R_final)
    return centers_final, radii_final, np.sum(radii_final)
