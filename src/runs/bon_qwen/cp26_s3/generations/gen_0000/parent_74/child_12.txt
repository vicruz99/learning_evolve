import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # Define objective function
    def objective(vars):
        # vars shape (78,)
        # radii are at indices 2, 5, 8, ... (3*i + 2)
        r = vars[2::3]
        return -np.sum(r)
    
    # Define constraints
    # We will use a list of constraint dictionaries.
    # Constraints:
    # 1. Boundary: x >= r, x <= 1-r, y >= r, y <= 1-r
    #    => x - r >= 0
    #    => 1 - x - r >= 0
    #    => y - r >= 0
    #    => 1 - y - r >= 0
    # 2. Non-overlap: (xi-xj)^2 + (yi-yj)^2 - (ri+rj)^2 >= 0
    
    constraints = []
    
    # Precompute indices to avoid repetition
    idx_x = np.arange(0, 3*n, 3)
    idx_y = np.arange(1, 3*n, 3)
    idx_r = np.arange(2, 3*n, 3)
    
    # Boundary constraints
    for i in range(n):
        ix, iy, ir = idx_x[i], idx_y[i], idx_r[i]
        
        # x - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, ix=ix, ir=ir: v[ix] - v[ir]
        })
        # 1 - x - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, ix=ix, ir=ir: 1.0 - v[ix] - v[ir]
        })
        # y - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, iy=iy, ir=ir: v[iy] - v[ir]
        })
        # 1 - y - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, iy=iy, ir=ir: 1.0 - v[iy] - v[ir]
        })

    # Non-overlap constraints
    # To save time, we can be smarter, but 325 constraints is manageable.
    # We define them lazily or just list them.
    # Let's create a function to generate the constraint fun to avoid huge list literal
    def make_overlap_constraint(i, j):
        ix_i, iy_i, ir_i = idx_x[i], idx_y[i], idx_r[i]
        ix_j, iy_j, ir_j = idx_x[j], idx_y[j], idx_r[j]
        
        def fun(v):
            xi, yi, ri = v[ix_i], v[iy_i], v[ir_i]
            xj, yj, rj = v[ix_j], v[iy_j], v[ir_j]
            return (xi - xj)**2 + (yi - yj)**2 - (ri + rj)**2
        return fun

    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({
                'type': 'ineq',
                'fun': make_overlap_constraint(i, j)
            })

    # Bounds
    # x, y in [0, 1], r in [0, 0.5]
    bounds = [(0.0, 1.0) if k % 3 != 2 else (0.0, 0.5) for k in range(3 * n)]

    best_result = None
    best_sum_r = -1.0

    # Helper to generate initial guess using hexagonal lattice
    def generate_hexagonal_guess():
        # We want to place 26 circles.
        # Let's try a pattern.
        # Approximate radius for 26 circles in unit square is around 0.102.
        # Spacing 2r approx 0.2.
        
        # Let's try to fit a grid.
        # 5 rows, varying columns.
        # Or just a simple grid and let optimizer fix it.
        
        # Hexagonal packing coordinates
        # Row 0: 5 circles
        # Row 1: 5 circles
        # Row 2: 5 circles
        # Row 3: 5 circles
        # Row 4: 6 circles
        # Total 26.
        
        # Let's construct coordinates
        centers = []
        r_approx = 0.1
        dx = 2 * r_approx
        dy = r_approx * np.sqrt(3)
        
        # Row 0
        for k in range(5):
            x = r_approx + k * dx
            y = r_approx
            centers.append((x, y))
        # Row 1 (shifted)
        for k in range(5):
            x = r_approx + 0.5 * dx + k * dx # Shift by r
            y = r_approx + dy
            centers.append((x, y))
        # Row 2
        for k in range(5):
            x = r_approx + k * dx
            y = r_approx + 2 * dy
            centers.append((x, y))
        # Row 3 (shifted)
        for k in range(5):
            x = r_approx + 0.5 * dx + k * dx
            y = r_approx + 3 * dy
            centers.append((x, y))
        # Row 4 (extra circle to reach 26)
        # Row 4 has 5 circles normally. We need 1 more.
        # Let's add a 6th circle in Row 4.
        # Check space.
        # x range for 5 circles: r to r + 4*2r = 9r = 0.9.
        # 6th circle would be at r + 5*2r = 1.1 (out of bounds).
        # Maybe compress grid or place in gap.
        # Let's just place it at end of row 4, but x might be > 1-r.
        # Optimizer will fix it.
        for k in range(6):
            x = r_approx + k * dx
            y = r_approx + 4 * dy
            centers.append((x, y))
            
        # We have 5+5+5+5+6 = 26 centers.
        # Some might be out of bounds or overlapping initially.
        # Clip to valid range [0.1, 0.9] for centers (since r approx 0.1)
        clipped_centers = []
        for x, y in centers:
            x = np.clip(x, 0.1, 0.9)
            y = np.clip(y, 0.1, 0.9)
            clipped_centers.append((x, y))
            
        # Construct vars array
        vars = np.zeros(3 * n)
        for i, (x, y) in enumerate(clipped_centers):
            vars[3*i] = x
            vars[3*i+1] = y
            vars[3*i+2] = 0.1 # Initial radius
            
        return vars

    # Helper to generate random valid guess
    def generate_random_guess():
        r_init = 0.05
        centers = []
        attempts = 0
        while len(centers) < n and attempts < 5000:
            attempts += 1
            x = np.random.uniform(r_init, 1 - r_init)
            y = np.random.uniform(r_init, 1 - r_init)
            valid = True
            for cx, cy in centers:
                if (x - cx)**2 + (y - cy)**2 < (2 * r_init)**2:
                    valid = False
                    break
            if valid:
                centers.append((x, y))
        
        if len(centers) < n:
            # Fallback to grid
            vars = generate_hexagonal_guess()
            # Perturb slightly
            vars[0::3] += np.random.normal(0, 0.01, n)
            vars[1::3] += np.random.normal(0, 0.01, n)
            return vars

        vars = np.zeros(3 * n)
        for i, (x, y) in enumerate(centers):
            vars[3*i] = x
            vars[3*i+1] = y
            vars[3*i+2] = r_init
        return vars

    # Run optimization multiple times
    # Use different seeds/strategies
    strategies = [generate_hexagonal_guess] + [generate_random_guess] * 4
    
    for idx, strat_func in enumerate(strategies):
        np.random.seed(idx)
        try:
            x0 = strat_func()
            # Small perturbation for random ones
            if idx > 0:
                x0[0::3] += np.random.normal(0, 0.005, n)
                x0[1::3] += np.random.normal(0, 0.005, n)
                
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, 
                           constraints=constraints, options={'maxiter': 500, 'ftol': 1e-10})
            
            if res.success:
                sum_r = -res.fun
                if sum_r > best_sum_r:
                    best_sum_r = sum_r
                    best_result = res
        except Exception:
            pass

    if best_result is None:
        # Fallback to hexagonal guess with small radii
        vars_opt = generate_hexagonal_guess()
        vars_opt[2::3] = 0.001
        best_sum_r = np.sum(vars_opt[2::3])
    else:
        vars_opt = best_result.x

    # Extract results
    centers = np.array([[vars_opt[3*i], vars_opt[3*i+1]] for i in range(n)])
    radii = np.array([vars_opt[3*i+2] for i in range(n)])
    
    # Post-processing: Ensure strict validity within tolerance
    # The solver might be close but slightly violating.
    # We can check and shrink radii if needed.
    # But usually SLSQP is good.
    # Let's verify overlap dist >= r1 + r2 - 1e-12
    
    # Check for any severe violations and fix
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        # Boundary check
        if x - r < -1e-12:
            r = max(0, x + 1e-12) # This would move center? No, just shrink radius
            # Actually better to adjust center?
            # Just clamp radius
            radii[i] = max(0, min(r, x, 1-x, y, 1-y))
        # Re-eval radius clamp
        r = radii[i]
        max_r_bound = min(x, 1-x, y, 1-y)
        if r > max_r_bound:
            radii[i] = max_r_bound
            
    # Check pairwise
    for i in range(n):
        for j in range(i+1, n):
            dist = np.sqrt((centers[i,0]-centers[j,0])**2 + (centers[i,1]-centers[j,1])**2)
            sum_r_ij = radii[i] + radii[j]
            if dist < sum_r_ij - 1e-12:
                # Shrink radii equally to fix
                shrink = (sum_r_ij - dist) / 2 + 1e-12
                radii[i] -= shrink
                radii[j] -= shrink
                # Ensure non-negative
                radii[i] = max(0, radii[i])
                radii[j] = max(0, radii[j])
                
    # Recalculate sum
    final_sum_r = np.sum(radii)
    
    # If we shrunk radii significantly, maybe we lost optimality, but it's valid.
    # Actually, the optimizer should have found a valid point.
    # The post-processing is just safety.
    
    return centers, radii, final_sum_r
