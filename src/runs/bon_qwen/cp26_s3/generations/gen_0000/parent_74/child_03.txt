    return -np.sum(radii)

def constraint_non_overlap(vars, i, j, n):
    """
    Constraint: dist(i, j) >= r_i + r_j
    """
    xi, yi, ri = vars[3*i], vars[3*i+1], vars[3*i+2]
    xj, yj, rj = vars[3*j], vars[3*j+1], vars[3*j+2]
    dist_sq = (xi - xj)**2 + (yi - yj)**2
    # dist >= ri + rj  <=>  dist^2 >= (ri + rj)^2
    # But dist^2 constraint is not convex. 
    # Linearized approximation or just dist >= ri + rj
    # Using dist - (ri + rj) >= 0
    dist = np.sqrt(dist_sq)
    return dist - (ri + rj)

    return vars[3*i] - vars[3*i+2]

def constraint_boundary_x_max(vars, i, n):
    """ x_i + r_i <= 1 """

def constraint_boundary_y_min(vars, i, n):

def constraint_boundary_y_max(vars, i, n):

    return 1.0 - (vars[3*i+1] + vars[3*i+2])

def constraint_radius_nonneg(vars, i, n):
