import numpy as np
from scipy.optimize import minimize
import math

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    Packs 26 circles in a unit square [0,1]x[0,1] to maximize the sum of radii.
    Uses a force-directed optimization approach starting from a hexagonal lattice.
    """
    n_circles = 26

    # We estimate an initial radius. 26 circles, area approx 0.85. 
    # r_approx = sqrt(0.85 / (26 * pi)) approx 0.102

    r_init = 0.09
    
    centers = []
    radii = []
    
    # Generate hexagonal grid points
    # Rows are staggered
    row_height = math.sqrt(3) * r_init * 1.1 # 10% buffer
    x_spacing = 2 * r_init * 1.1
    
    y = r_init
    row_idx = 0
    while len(centers) < n_circles:
        # Determine x offset for staggered rows
        if row_idx % 2 == 0:
            x_start = r_init
        else:
            x_start = r_init + x_spacing / 2
            
        x = x_start
        while x <= 1 - r_init:
            if len(centers) < n_circles:
                centers.append([x, y])
                radii.append(r_init)
            x += x_spacing
        y += row_height
        row_idx += 1
        
    centers = np.array(centers[:n_circles])
    radii = np.array(radii[:n_circles])
    
    # 2. Force-Directed Optimization
    # We will run a simulation to push circles apart and grow them.
    # We optimize centers to minimize overlap penalty.
    
    num_iterations = 2000
    learning_rate = 0.01

    # Strategy: Keep radii somewhat uniform but allow slight variations if needed.
    # Actually, for max sum of radii, equal radii is usually optimal or near optimal.
    # We will treat radii as a global variable r that we try to increase,
    # but we optimize centers for that r. 
    # However, to be safe and creative, we allow individual radii to adjust slightly 
    # based on local space, but we primarily focus on center positions.
    
    # Let's use a repulsive force model.
    # Force between i and j: F = max(0, r_i + r_j - dist)^2 * direction
    # Force with boundary: F = max(0, r - dist_to_wall)^2 * direction
    
    # We will run multiple passes.
    # Pass 1: Optimize centers for current radii.
    # Pass 2: Increase radii slightly.
    # Repeat.

    # with a penalty function for constraints.
    
    # Let's define an energy function
    def energy_fn(vars):
        # vars: [x1, y1, r1, x2, y2, r2, ...]
        n = len(vars) // 3
        penalty = 0.0
        sum_r = 0.0
        
        # Reshape
        c = vars[:2*n].reshape(n, 2)
        r = vars[2*n:]
        
        # Boundary constraints
        for i in range(n):
            x, y = c[i]
            ri = r[i]
            # Soft constraint for being inside
            # We want x-r >= 0 => x-r + margin >= 0
            # Penalty if x-r < 0
            val = x - ri
            if val < 0: penalty += 1000 * val**2
            val = 1 - (x + ri)
            if val < 0: penalty += 1000 * val**2
            val = y - ri
            if val < 0: penalty += 1000 * val**2
            val = 1 - (y + ri)
            if val < 0: penalty += 1000 * val**2
            
            sum_r += ri
            
        # Overlap constraints
        for i in range(n):
            for j in range(i+1, n):
                dx = c[i, 0] - c[j, 0]
                dy = c[i, 1] - c[j, 1]
                dist_sq = dx*dx + dy*dy
                dist = math.sqrt(dist_sq + 1e-9) # avoid nan
                req_dist = r[i] + r[j]
                overlap = req_dist - dist
                if overlap > 0:
                    penalty += 100 * overlap**2
                    
        # We want to maximize sum_r - penalty

        return penalty - sum_r

    # Final validation and potential local fix
    # If any overlaps remain due to optimization stopping early, try to fix
    # But with high penalty, it should be clean.

    sum_radii = np.sum(best_radii)
    
    return best_centers, best_radii, sum_radii
