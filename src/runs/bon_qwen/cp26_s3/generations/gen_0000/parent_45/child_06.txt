import numpy as np

def compute_forces(positions, r):
    """Compute repulsion forces between circles and boundaries."""
    n = positions.shape[0]
    forces = np.zeros_like(positions)
    
    # Pairwise repulsion
    for i in range(n):
        for j in range(i + 1, n):
            diff = positions[i] - positions[j]
            d = np.linalg.norm(diff)
            if d < 2 * r and d > 1e-9:
                f_mag = (2 * r - d) / d
                forces[i] += diff * f_mag
                forces[j] -= diff * f_mag
            elif d < 1e-9:
                # Push apart randomly if exactly coincident
                rand_dir = np.random.randn(2)
                forces[i] += rand_dir
                forces[j] -= rand_dir
                
    # Boundary repulsion
    for i in range(n):
        x, y = positions[i]
        if x < r: forces[i, 0] += (r - x) * 2
        if x > 1 - r: forces[i, 0] -= (x - (1 - r)) * 2
        if y < r: forces[i, 1] += (r - y) * 2
        if y > 1 - r: forces[i, 1] -= (y - (1 - r)) * 2
    return forces

def run_packing():
    np.random.seed(2024)
    n = 26
    best_r = 0.0
    best_centers = None

    configs = []

    hex_cfg = []

            hex_cfg.append([x, y])
    while len(hex_cfg) < n:
        hex_cfg.append([np.random.rand(), np.random.rand()])

    # 2. Random layout
    configs.append(np.random.rand(n, 2))
    
    # 3. Uniform grid layout
    grid_cfg = np.zeros((n, 2))
    idx = 0
    for i in range(6):
        for j in range(5):
            if idx < n:
                grid_cfg[idx] = [j * 0.2 + 0.1, i * 0.2 + 0.1]
                idx += 1
    configs.append(grid_cfg)

    # Optimize for each configuration
    for cfg in configs:
        pos = cfg.copy()
        r = 0.02
        lr = 0.06
        
        for step in range(4000):
            forces = compute_forces(pos, r)
            pos += lr * forces
            pos = np.clip(pos, 0, 1)
            
            # Check for overlaps
            has_overlap = False
            for i in range(n):
                for j in range(i + 1, n):
                    if np.linalg.norm(pos[i] - pos[j]) < 2 * r - 1e-6:
                        has_overlap = True
                        break
                if has_overlap: break
            
            if not has_overlap:
                r += 0.00005
                if step % 150 == 0:
                    lr = 0.06  # Reset learning rate to maintain dynamics
            else:
                lr = max(0.01, lr * 0.95)  # Slow down if stuck
                
            # Periodic perturbation to escape local minima
            if step % 400 == 0 and step > 0:
                pos += np.random.randn(n, 2) * 0.0005
                pos = np.clip(pos, 0, 1)
                
        # Compute the exact maximum radius for this configuration
        min_d = 1.0
        for i in range(n):
            for j in range(i + 1, n):
                d = np.linalg.norm(pos[i] - pos[j])
                if d < min_d: min_d = d
            min_d = min(min_d, pos[i,0], 1-pos[i,0], pos[i,1], 1-pos[i,1])
            
        curr_r = min_d / 2
        if curr_r > best_r:
            best_r = curr_r
            best_centers = pos.copy()
            
    radii = np.full(n, best_r)
    return best_centers, radii, n * best_r
