import numpy as np
from scipy.optimize import minimize

def run_packing():
    """
    Pack 26 circles in a unit square to maximize the sum of radii.
    Returns (centers, radii, sum_radii).
    """
    n_circles = 26
    best_centers = None
    best_radii = None
    best_sum = -1.0

    # We will perform multiple restarts to find the global optimum
    n_restarts = 20

    for restart in range(n_restarts):
        # 1. Generate Initial Configuration
        # Strategy: Hexagonal packing grid
        # We try to fit circles with a radius that allows roughly 26 circles.
        # A hexagonal grid allows rows to be offset.
        # Let's estimate a radius. For 25 circles in 5x5 grid, r=0.1.
        # For 26, maybe slightly less, or different arrangement.
        # Let's try r_init = 0.095 to start safe, or 0.1 if we can fit.
        
        # Generate hex grid coordinates
        # Rows roughly 5 or 6.
        # Let's try to generate points and select 26 best fitting.
        
        r_init = 0.09 + 0.01 * np.random.rand() # Randomize initial radius slightly
        # Vertical spacing for hex: sqrt(3)*r
        # Horizontal spacing: 2*r
        
        # We want to fill the square.
        # Let's generate a grid and filter points inside.
        
        points = []
        # Number of rows needed?
        # Height 1. Row spacing ~ 0.15. ~ 6-7 rows.
        # Width 1. Col spacing ~ 0.2. ~ 5 cols.
        
        rows = 8
        cols = 6
        
        y_step = np.sqrt(3) * r_init
        x_step = 2 * r_init
        
        for row in range(rows):
            for col in range(cols):
                # Offset every other row
                x = r_init + col * x_step
                if row % 2 == 1:
                    x += r_init # Shift by r to nest in gaps
                y = r_init + row * y_step
                
                if 0 <= x <= 1 and 0 <= y <= 1:
                    points.append([x, y])
        
        # Shuffle points to pick random 26
        np.random.shuffle(points)
        if len(points) < n_circles:
            # Fallback to random valid positions if grid is too sparse
            # This happens if r_init is too large
            centers_init = np.random.rand(n_circles, 2) * 0.8 + 0.1 # Keep away from edges initially
            radii_init = np.full(n_circles, 0.01)
        else:
            # Pick first 26
            selected_points = np.array(points[:n_circles])
            # Perturb slightly to break symmetry
            selected_points += np.random.normal(0, 0.005, size=selected_points.shape)
            # Clip to [0.1, 0.9] to ensure initial validity with small radius
            selected_points = np.clip(selected_points, 0.1, 0.9)
            centers_init = selected_points
            radii_init = np.full(n_circles, r_init)

        # 2. Define Optimization Problem
        # Variables: [x1, y1, r1, x2, y2, r2, ...]
        # Shape: (n_circles * 3,)
        
        def flatten(vars_2d):
            # vars_2d is (n, 3) -> x, y, r
            return vars_2d.flatten()

        def unflatten(vars_1d):
            return vars_1d.reshape(n_circles, 3)

        initial_vars_2d = np.hstack([centers_init, radii_init.reshape(-1, 1)])
        x0 = flatten(initial_vars_2d)

        # Objective: Minimize -sum(radii)
        def objective(vars_1d):
            vars_2d = unflatten(vars_1d)
            radii = vars_2d[:, 2]
            return -np.sum(radii)

        # Constraints
        constraints = []
        
        # 1. Boundary constraints: r <= x <= 1-r  =>  r - x <= 0, x + r - 1 <= 0
        #    Actually constraints should be g(x) >= 0 for minimize? 
        #    scipy 'SLSQP' expects fun(x) >= 0 for inequality constraints? 
        #    Docs: "inequality constraints (fun(x) >= 0)"
        
        # Constraint: x - r >= 0  =>  x - r >= 0
        # Constraint: 1 - (x + r) >= 0 => 1 - x - r >= 0
        # Same for y
        
        # We can define a function that returns an array of constraint values
        def boundary_constraints(vars_1d):
            vars_2d = unflatten(vars_1d)
            xs = vars_2d[:, 0]
            ys = vars_2d[:, 1]
            rs = vars_2d[:, 2]
            
            # x >= r
            c1 = xs - rs
            # x + r <= 1
            c2 = 1 - (xs + rs)
            # y >= r
            c3 = ys - rs
            # y + r <= 1
            c4 = 1 - (ys + rs)
            # r >= 0
            c5 = rs
            
            return np.concatenate([c1, c2, c3, c4, c5])

        constraints.append({'type': 'ineq', 'fun': boundary_constraints})

        # 2. Non-overlap constraints: dist(i, j) >= r_i + r_j
        #    dist^2 >= (r_i + r_j)^2
        #    (xi-xj)^2 + (yi-yj)^2 - (ri+rj)^2 >= 0
        
        def non_overlap_constraints(vars_1d):
            vars_2d = unflatten(vars_1d)
            xs = vars_2d[:, 0]
            ys = vars_2d[:, 1]
            rs = vars_2d[:, 2]
            
            cons = []
            for i in range(n_circles):
                for j in range(i + 1, n_circles):
                    dx = xs[i] - xs[j]
                    dy = ys[i] - ys[j]
                    dist_sq = dx*dx + dy*dy
                    sum_r = rs[i] + rs[j]
                    cons.append(dist_sq - sum_r*sum_r)
            return np.array(cons)

        constraints.append({'type': 'ineq', 'fun': non_overlap_constraints})

        # Bounds
        # x, y in [0, 1]
        # r in [0, 0.5]
        bounds = []
        for _ in range(n_circles):
            bounds.append((0, 1)) # x
            bounds.append((0, 1)) # y
            bounds.append((0, 0.5)) # r

        # 3. Run Optimizer
        try:
            # Use SLSQP
            res = minimize(
                objective,
                x0,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 500, 'ftol': 1e-12, 'disp': False}
            )
            
            if res.success or res.fun < -best_sum: # Note: objective is negative sum
                # Check if valid manually just in case
                curr_centers = unflatten(res.x)[:, :2]
                curr_radii = unflatten(res.x)[:, 2]
                
                # Simple validation check
                valid = True
                # Check bounds
                if np.any(curr_radii < -1e-9) or np.any(np.isnan(curr_radii)):
                    valid = False
                if np.any(np.isnan(curr_centers)):
                    valid = False
                
                if valid:
                    # Check overlaps with tolerance
                    overlap = False
                    for i in range(n_circles):
                        for j in range(i + 1, n_circles):
                            dist = np.sqrt(np.sum((curr_centers[i] - curr_centers[j])**2))
                            if dist < curr_radii[i] + curr_radii[j] - 1e-7:
                                overlap = True
                                break
                        if overlap: break
                    
                    if not overlap:
                        # Check square bounds
                        out = False
                        for i in range(n_circles):
                            x, y = curr_centers[i]
                            r = curr_radii[i]
                            if x - r < -1e-7 or x + r > 1 + 1e-7 or y - r < -1e-7 or y + r > 1 + 1e-7:
                                out = True
                                break
                        
                        if not out:
                            curr_sum = np.sum(curr_radii)
                            if curr_sum > best_sum:
                                best_sum = curr_sum
                                best_centers = curr_centers
                                best_radii = curr_radii

        except Exception as e:
            # Continue on error
            pass

    # Final fallback if optimization failed or result invalid
    if best_centers is None:
        # Return a simple valid grid
        # 5x5 grid + 1 small circle in center of a gap?
        # Actually just return a valid small packing
        centers = np.random.rand(n_circles, 2) * 0.8 + 0.1
        radii = np.full(n_circles, 0.01)
        best_centers = centers
        best_radii = radii
        best_sum = np.sum(radii)

    # Ensure output format
    final_centers = np.array(best_centers)
    final_radii = np.array(best_radii)
    final_sum = float(best_sum)

    return final_centers, final_radii, final_sum
