import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    
    Returns:
        centers: np.array of shape (26, 2)
        radii: np.array of shape (26,)
        sum_radii: float
    """
    n = 26
    
    # 1. Initialization: Hexagonal Grid
    # We arrange circles in rows with horizontal offsets to approximate hexagonal packing.
    # Row configuration: 5, 6, 5, 6, 4 circles (Total 26)
    rows_config = [5, 6, 5, 6, 4]
    
    centers_init = []
    # Initial spacing parameters (will be refined by optimizer)
    # We place them somewhat centered to leave room for expansion
    dx = 0.18
    dy = dx * np.sqrt(3) / 2
    
    current_y = 0.15
    for r_idx, count in enumerate(rows_config):
        # Calculate x positions for the current row
        # Center the row horizontally
        row_width = (count - 1) * dx
        start_x = (1.0 - row_width) / 2
        
        # Offset for hexagonal pattern (even rows vs odd rows)
        # Actually, standard hex grid shifts every row. 
        # Let's apply a shift of dx/2 for odd-indexed rows relative to a base grid
        if r_idx % 2 == 1:
            start_x += dx / 2
            
        for c_idx in range(count):
            x = start_x + c_idx * dx
            centers_init.append([x, current_y])
        
        current_y += dy
        
    centers_init = np.array(centers_init)
    
    # Initial radii: small enough to not overlap, but non-zero
    radii_init = np.ones(n) * 0.05
    
    # Flatten for optimizer
    # Variables: [x0, y0, r0, x1, y1, r1, ..., x25, y25, r25]
    x0 = np.hstack([centers_init.T, radii_init])
    
    # Bounds: x, y in [0, 1], r >= 0
    bounds = []
    for i in range(n):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((1e-6, 0.5)) # r (small lower bound to avoid degenerate circles if possible, though 0 is allowed)
        
    # Constraints
    cons = []
    
    # Boundary constraints:
    # x - r >= 0
    # 1 - x - r >= 0
    # y - r >= 0
    # 1 - y - r >= 0
    # Implemented as linear constraints for SLSQP efficiency where possible, 
    # but SLSQP handles general nonlinear.
    # We will add them as general constraints for clarity.
    
    def boundary_constraint_x_min(vars, i):
        return vars[3*i] - vars[3*i + 2]
    
    def boundary_constraint_x_max(vars, i):
        return 1.0 - vars[3*i] - vars[3*i + 2]

    def boundary_constraint_y_min(vars, i):
        return vars[3*i + 1] - vars[3*i + 2]

    def boundary_constraint_y_max(vars, i):
        return 1.0 - vars[3*i + 1] - vars[3*i + 2]

    for i in range(n):
        cons.append({'type': 'ineq', 'fun': lambda v, idx=i: boundary_constraint_x_min(v, idx)})
        cons.append({'type': 'ineq', 'fun': lambda v, idx=i: boundary_constraint_x_max(v, idx)})
        cons.append({'type': 'ineq', 'fun': lambda v, idx=i: boundary_constraint_y_min(v, idx)})
        cons.append({'type': 'ineq', 'fun': lambda v, idx=i: boundary_constraint_y_max(v, idx)})

    # Pairwise non-overlap constraints:
    # dist^2 >= (r_i + r_j)^2
    # (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
    
    def overlap_constraint(vars, i, j):
        xi, yi, ri = vars[3*i], vars[3*i+1], vars[3*i+2]
        xj, yj, rj = vars[3*j], vars[3*j+1], vars[3*j+2]
        
        dist_sq = (xi - xj)**2 + (yi - yj)**2
        sum_r = ri + rj
        return dist_sq - sum_r**2

    for i in range(n):
        for j in range(i + 1, n):
            cons.append({'type': 'ineq', 'fun': lambda v, ii=i, jj=j: overlap_constraint(v, ii, jj)})

    # Objective function: Maximize sum of radii -> Minimize -sum(r)
    def objective(vars):
        r = vars[2::3]
        return -np.sum(r)

    # Optimization
    # Using SLSQP. It might take some time but is robust for this size.
    # We set a reasonable maxiter and tolerance.
    res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, 
                   options={'maxiter': 1000, 'ftol': 1e-9})
    
    # Extract results
    final_vars = res.x
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    for i in range(n):
        centers[i, 0] = final_vars[3*i]
        centers[i, 1] = final_vars[3*i+1]
        radii[i] = final_vars[3*i+2]
        
    # Ensure non-negative radii (numerical safety)
    radii = np.maximum(radii, 0.0)
    
    sum_radii = np.sum(radii)
    
    # Final check and clipping if necessary (though optimizer should handle it)
    # Sometimes numerical errors push slightly out, but validate_packing handles 1e-12 tolerance.
    
    return centers, radii, sum_radii
