

<strategy>
This is a circle packing problem where we need to pack 26 circles in a unit square to maximize the sum of radii (target: 2.636).

Key insights:
1. With variable radii, we don't need all circles to be equal size - we just need to maximize the sum.
2. For 26 equal circles in a 5×5 grid, r=0.1 gives sum=2.5. We need ~2.636, so we need better packing.
3. Hexagonal packing is more space-efficient than square packing.
4. The problem decomposes into: (a) choose centers, (b) given centers, solve LP for optimal radii.

Strategy:
1. Generate initial centers using hexagonal lattice packing (5 rows with alternating offsets)
2. Use LP (scipy linprog) to find optimal radii given fixed centers
3. Perform local search: perturb each center, re-solve LP, keep if improvement
4. Use decreasing step sizes for refinement
5. Try multiple initial configurations and keep the best

The LP formulation:
- Variables: r_0, ..., r_25
- Maximize: sum(r_i)
- Constraints: r_i <= boundary(i), r_i + r_j <= distance(i,j), r_i >= 0

This LP has ~430 constraints and 26 variables, solvable quickly with HiGHS.

import numpy as np
from scipy.optimize import linprog


def solve_radii_lp(centers):
    """Given fixed centers, solve LP to maximize sum of radii."""
    n = centers.shape[0]
    
    # Build constraint matrix A_ub @ r <= b_ub
    # Constraints:
    # 1. r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i (4 per circle)
    # 2. r_i + r_j <= dist(i,j) for all i < j (overlap)
    
    n_boundary = 4 * n
    n_overlap = n * (n - 1) // 2
    n_constraints = n_boundary + n_overlap
    
    A = np.zeros((n_constraints, n))
    b = np.zeros(n_constraints)
    
    idx = 0
    for i in range(n):
        x, y = centers[i]
        for val in [x, 1.0 - x, y, 1.0 - y]:
            A[idx, i] = 1.0
            b[idx] = max(val, 0.0)
            idx += 1
    
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = dist
            idx += 1
    
    c = -np.ones(n)
    bounds = [(0.0, None)] * n
    
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                  options={'disp': False, 'presolve': True})
    
    if res.success:
        return res.x, -res.fun
    return np.zeros(n), 0.0


def generate_hexagonal_centers(n, rows, cols_per_row, s):
    """Generate centers in a hexagonal lattice pattern."""
    centers = []
    rh = s * np.sqrt(3) / 2.0
    
    for row in range(rows):
        ncols = cols_per_row[row] if row < len(cols_per_row) else cols_per_row[0]
        x_offset = s / 2.0 if row % 2 == 1 else 0.0
        
        for col in range(ncols):
            if len(centers) >= n:
                break
            x = 0.5 + (col - (ncols - 1) / 2.0) * s + x_offset
            y = 0.5 + (row - (rows - 1) / 2.0) * rh
            centers.append([x, y])
    
    return np.array(centers[:n])


def try_configuration(centers, n=26, max_iter=120, init_step=0.025):
    """Local search optimization starting from given centers."""
    best_centers = centers.copy()
    radii, best_sum = solve_radii_lp(best_centers)
    
    if best_sum < 1e-10:
        return best_centers, best_radii if 'best_radii' in dir() else np.zeros(n), best_sum
    
    step = init_step
    step_decay = 0.88
    min_step = 1e-8
    
    # Directions for perturbation
    directions = [
        (step, 0.0), (-step, 0.0), (0.0, step), (0.0, -step),
        (step/np.sqrt(2), step/np.sqrt(2)), (-step/np.sqrt(2), -step/np.sqrt(2)),
        (step/np.sqrt(2), -step/np.sqrt(2)), (-step/np.sqrt(2), step/np.sqrt(2)),
    ]
    
    improved_steps = 0
    for iteration in range(max_iter):
        improved = False
        for i in range(n):
            for ddx, ddy in directions:
                trial = best_centers.copy()
                trial[i, 0] = np.clip(best_centers[i, 0] + ddx, 0.002, 0.998)
                trial[i, 1] = np.clip(best_centers[i, 1] + ddy, 0.002, 0.998)
                
                r, s = solve_radii_lp(trial)
                if s > best_sum + 1e-12:
                    best_centers = trial
                    radii = r
                    best_sum = s
                    improved = True
                    improved_steps = 0
                    # Update direction magnitudes
                    directions = [
                        (step, 0.0), (-step, 0.0), (0.0, step), (0.0, -step),
                        (step/np.sqrt(2), step/np.sqrt(2)), (-step/np.sqrt(2), -step/np.sqrt(2)),
                        (step/np.sqrt(2), -step/np.sqrt(2)), (-step/np.sqrt(2), step/np.sqrt(2)),
                    ]
        
        improved_steps += 1
        if not improved or improved_steps > 5:
            step *= step_decay
            improved_steps = 0
            directions = [
                (step, 0.0), (-step, 0.0), (0.0, step), (0.0, -step),
                (step/np.sqrt(2), step/np.sqrt(2)), (-step/np.sqrt(2), -step/np.sqrt(2)),
                (step/np.sqrt(2), -step/np.sqrt(2)), (-step/np.sqrt(2), step/np.sqrt(2)),
            ]
        
        if step < min_step:
            break
    
    return best_centers, radii, best_sum


def run_packing():
    n = 26
    
    best_centers = None
    best_radii = None
    best_sum = 0.0
    
    # Try multiple initial configurations
    configs = []
    
    # Config 1: 5 rows, 5+5+5+5+6 arrangement
    for s in [0.13, 0.14, 0.15, 0.16]:
        centers = generate_hexagonal_centers(n, 5, [5, 5, 5, 5, 6], s)
        configs.append(centers)
    
    # Config 2: 6 rows, 4+5+4+5+4+4 arrangement
    for s in [0.14, 0.15, 0.16]:
        centers = generate_hexagonal_centers(n, 6, [4, 5, 4, 5, 4, 4], s)
        configs.append(centers)
    
    # Config 3: 5 rows, 6+5+5+5+5 arrangement
    for s in [0.13, 0.14, 0.15]:
        centers = generate_hexagonal_centers(n, 5, [6, 5, 5, 5, 5], s)
        configs.append(centers)
    
    # Config 4: 6 rows, 5+4+5+4+5+3 arrangement
    for s in [0.14, 0.15]:
        centers = generate_hexagonal_centers(n, 6, [5, 4, 5, 4, 5, 3], s)
        configs.append(centers)
    
    # Config 5: 7 rows alternating
    for s in [0.15, 0.16]:
        centers = generate_hexagonal_centers(n, 7, [4, 3, 4, 3, 4, 3, 5], s)
        configs.append(centers)
    
    # Config 6: Nearly square grid with slight hexagonal shift
    for s in [0.17, 0.18]:
        centers = generate_hexagonal_centers(n, 5, [5, 5, 5, 5, 6], s)
        configs.append(centers)
    
    # Config 7: Random-ish perturbation of hexagonal
    for s in [0.14, 0.15]:
        centers = generate_hexagonal_centers(n, 5, [5, 5, 5, 5, 6], s)
        # Add small random-like perturbation
        np.random.seed(42)
        centers += np.random.uniform(-0.01, 0.01, size=centers.shape)
        configs.append(centers)
    
    # Config 8: 5x5 grid + 1 extra
    for _ in range(3):
        centers = np.zeros((n, 2))
        for i in range(5):
            for j in range(5):
                centers[i * 5 + j] = [0.1 + j * 0.2, 0.1 + i * 0.2]
        # Place 26th circle
        centers[25] = [0.5, 0.55]
        configs.append(centers)
    
    # Optimize each configuration
    for config_idx, centers in enumerate(configs):
        try:
            centers_opt, radii_opt, sum_opt = try_configuration(
                centers.copy(), n=n, max_iter=150, init_step=0.03
            )
            if sum_opt > best_sum:
                best_sum = sum_opt
                best_centers = centers_opt
                best_radii = radii_opt
        except Exception:
            continue
    
    # Final refinement on best configuration
    if best_centers is not None:
        best_centers, best_radii, best_sum = try_configuration(
            best_centers.copy(), n=n, max_iter=200, init_step=0.02
        )
    
    # Final LP solve to ensure consistency
    final_radii, final_sum = solve_radii_lp(best_centers)
    
    return best_centers, final_radii, final_sum
