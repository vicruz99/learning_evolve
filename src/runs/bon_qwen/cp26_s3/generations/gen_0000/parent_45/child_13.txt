The objective is to maximize $\sum r_i$, which is equivalent to minimizing $-\sum r_i$. The constraints ensure circles remain within $[0,1]^2$ and do not overlap: $r_i \le x_i \le 1-r_i$, $r_i \le y_i \le 1-r_i$, and $\|c_i - c_j\|_2 \ge r_i + r_j$.

import numpy as np
from scipy.optimize import minimize

def compute_constraints(x):
    """
    Computes all inequality constraints for the circle packing problem.
    Returns a 1D array where all values must be >= 0 for feasibility.
    """
    n = 26
    centers = x[:2*n].reshape(n, 2)
    radii = x[2*n:]
    
    # Boundary constraints: 4 per circle -> x >= r, 1-x >= r, y >= r, 1-y >= r
    b_con = np.zeros(4*n)
    b_con[0::4] = centers[:, 0] - radii
    b_con[1::4] = 1.0 - centers[:, 0] - radii
    b_con[2::4] = centers[:, 1] - radii
    b_con[3::4] = 1.0 - centers[:, 1] - radii

    # Pairwise non-overlap constraints: dist(i,j) >= r_i + r_j
    n_pairs = n * (n - 1) // 2
    p_con = np.zeros(n_pairs)
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            p_con[idx] = dist - radii[i] - radii[j]
            idx += 1
            
    return np.concatenate([b_con, p_con])

def objective(x):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(x[52:])

def run_packing():
    n = 26
    
    # Initialize centers in a hexagonal/triangular lattice pattern
    centers = np.zeros((n, 2))
    idx = 0
    row_counts = [5, 4, 5, 4, 5, 3]  # Sums to 26
    y_pos = np.linspace(0.15, 0.85, len(row_counts))
    
    for r, y in enumerate(y_pos):
        nc = row_counts[r]
        # Shift odd-indexed rows to form hexagonal packing
        shift = 0.0 if r % 2 == 0 else 0.075
        x_coords = np.linspace(0.15 + shift, 0.85 - shift, nc)
        for x in x_coords:
            centers[idx] = [x, y]
            idx += 1
            
    # Initial feasible radii
    radii_init = np.full(n, 0.05)
    x0 = np.concatenate([centers.flatten(), radii_init])
    
    # Variable bounds
    bnds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
    cons = {'type': 'ineq', 'fun': compute_constraints}
    
    # Run SLSQP optimization
    res = minimize(objective, x0, method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': 3000, 'ftol': 1e-12, 'disp': False})
                   
    centers_opt = res.x[:52].reshape(n, 2)
    radii_opt = res.x[52:]
    
    # Post-processing to strictly satisfy numerical tolerances
    radii_opt = np.maximum(radii_opt, 0.0)
    centers_opt = np.clip(centers_opt, 0.0, 1.0)
    
    return centers_opt, radii_opt, float(np.sum(radii_opt))
