import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def objective_func(x):
    """Objective function to maximize sum of radii (minimized as negative sum)."""
    return -np.sum(x[2*N_CIRCLES:])

def constraint_func(x, n):
    """Returns inequality constraints >= 0."""
    c = []
    ctr = x[:2*n].reshape(n, 2)
    rad = x[2*n:]
    
    # Wall constraints: 0 <= x-r and x+r <= 1 => x-r >= 0, 1-x-r >= 0
    c.append(ctr[:, 0] - rad)
    c.append(1.0 - ctr[:, 0] - rad)
    c.append(ctr[:, 1] - rad)
    c.append(1.0 - ctr[:, 1] - rad)
    
    # Pairwise non-overlap constraints: dist^2 >= (ri + rj)^2
    diff = ctr[:, None, :] - ctr[None, :, :]
    dist_sq = np.sum(diff**2, axis=2)
    r_sum = rad[:, None] + rad[None, :]
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    c.append(dist_sq[mask] - r_sum[mask]**2)
    
    return np.concatenate(c)

def run_packing():
    n = N_CIRCLES
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Multiple restarts to escape local optima
    seeds = [42, 13, 7, 100, 200, 300, 500]
    
    for seed in seeds:
        rng = np.random.RandomState(seed)
        
        # Hexagonal lattice initialization
        cols = 6
        rows = 5
        dx = 1.0 / (cols + 1)
        dy = np.sqrt(3) / 2.0 * dx
        
        centers = []
        radii = []
        idx = 0
        for r in range(rows):
            for c in range(cols):
                if idx < n:
                    x = (c + 0.5 + (0.5 if r % 2 == 1 else 0.0)) * dx
                    y = (r + 0.5) * dy
                    # Small random perturbation
                    x += rng.randn() * 0.01
                    y += rng.randn() * 0.01
                    # Ensure strict feasibility initially
                    x = np.clip(x, 0.1, 0.9)
                    y = np.clip(y, 0.1, 0.9)
                    centers.append([x, y])
                    radii.append(0.03)
                    idx += 1
                    
        centers = np.array(centers)
        radii = np.array(radii)
        
        x0 = np.concatenate([centers.flatten(), radii])
        bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n
        
        cons = {'type': 'ineq', 'fun': constraint_func, 'args': (n,)}
        
        try:
            res = minimize(objective_func, x0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 5000, 'ftol': 1e-10, 'disp': False})
            if res.fun is not None and -res.fun > best_sum:
                curr_centers = res.x[:2*n].reshape(n, 2)
                curr_radii = res.x[2*n:]
                if np.all(curr_radii >= 0):
                    best_sum = -res.fun
                    best_centers = curr_centers.copy()
                    best_radii = curr_radii.copy()
        except Exception:
            continue
            
    if best_centers is None:
        # Fallback configuration
        best_centers = np.random.rand(n, 2) * 0.8 + 0.1
        best_radii = np.full(n, 0.05)
        best_sum = np.sum(best_radii)
        
    best_radii = np.maximum(best_radii, 1e-9)
    return best_centers, best_radii, best_sum
