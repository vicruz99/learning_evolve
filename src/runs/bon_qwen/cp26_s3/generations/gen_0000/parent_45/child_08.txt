        return centers, radii

        return -np.sum(radii) # Minimize negative sum

        # Constraint: dist^2 >= (r_i + r_j)^2

        ci = vars[i]
        cj = vars[j]

        r_j = cj[2]

        return vars[i, 0] - vars[i, 2] # x - r >= 0

        return 1 - vars[i, 0] - vars[i, 2] # 1 - x - r >= 0
