import numpy as np
from scipy.optimize import minimize

def calculate_repulsive_energy(centers, radii):
    """Calculates a penalty for overlapping circles."""
    penalty = 0.0
    n = len(radii)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            min_dist = radii[i] + radii[j]
            if dist < min_dist:
                # Quadratic penalty for overlap
                penalty += (min_dist - dist) ** 2
    return penalty

            radii[i] = x[3*i+2]
        
        # Penalty for circles going outside [0,1]
        boundary_penalty = 0.0
        for i in range(n_circles):
            x, y = centers[i]
            r = radii[i]
            if x < r: boundary_penalty += 1e6 * (r - x)**2
            if x > 1 - r: boundary_penalty += 1e6 * (x - (1 - r))**2
            if y < r: boundary_penalty += 1e6 * (r - y)**2
            if y > 1 - r: boundary_penalty += 1e6 * (y - (1 - r))**2

        # Penalty for overlap
        overlap_penalty = calculate_repulsive_energy(centers, radii)
        
        # Objective: Maximize sum of radii - penalties
        return -np.sum(radii) + boundary_penalty + overlap_penalty

        """Ensure centers are within [0,1] and radii are non-negative."""
