To maximize the sum of radii for 26 circles in a unit square, we formulate the problem as a constrained non-linear optimization task. We want to maximize $\sum r_i$ subject to:
1. Boundary constraints: $r_i \le x_i \le 1-r_i$ and $r_i \le y_i \le 1-r_i$
2. Non-overlap constraints: $\|c_i - c_j\| \ge r_i + r_j$ for all $i < j$

We use `scipy.optimize.minimize` with the SLSQP algorithm, which handles inequality constraints efficiently. 
- **Initialization**: We place circles on a perturbed hexagonal lattice, which is known to be dense for circle packing. This provides a high-quality starting point.
- **Vectorized Constraints**: All boundary and pairwise distance constraints are computed simultaneously using NumPy broadcasting to accelerate the constraint evaluation function.
- **Optimization**: SLSQP minimizes the negative sum of radii (equivalent to maximization) while keeping all constraint values $\ge 0$.
- **Post-processing**: To guarantee strict compliance with the validation function, we clamp radii to boundary limits and iteratively shrink overlapping circles proportionally until all distances satisfy the non-overlap condition with a safe margin.
- **Symmetry Breaking**: Small random perturbations are added to the initial configuration to help the optimizer escape symmetric flat directions and find a tighter local optimum.

import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def compute_objective(vars):
    """Objective: maximize sum of radii (minimize negative sum)"""
    return -np.sum(vars[2*N_CIRCLES:])

def compute_constraints(vars):
    """
    Returns array of constraint values. All must be >= 0 for feasibility.
    Includes 4*N boundary constraints and N*(N-1)/2 non-overlap constraints.
    """
    n = N_CIRCLES
    cx = vars[:n]
    cy = vars[n:2*n]
    cr = vars[2*n:]
    
    vals = []
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    vals.extend(cx - cr)
    vals.extend(1.0 - cx - cr)
    vals.extend(cy - cr)
    vals.extend(1.0 - cy - cr)
    
    # Non-overlap constraints: dist >= r_i + r_j
    # Vectorized computation for speed
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dist = np.hypot(dx, dy)
    np.fill_diagonal(dist, np.inf)  # Ignore self-distances
    overlap = dist - (cr[:, None] + cr[None, :])
    
    # Extract only upper triangular part to avoid duplicates
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    vals.extend(overlap[mask])
    
    return np.array(vals)

def run_packing():
    np.random.seed(42)
    n = N_CIRCLES
    
    # 1. Initialize on a hexagonal lattice
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.085)
    
    pts = []
    for i in range(7):
        for j in range(6):
            if len(pts) >= n:
                break
            x = j * 0.2 + (0.5 if i % 2 else 0) * 0.1
            y = i * 0.17320508
            pts.append((x, y))
            
    xs = [p[0] for p in pts]
    ys = [p[1] for p in pts]
    sx = (0.85 - 0.15) / (max(xs) - min(xs))
    sy = (0.85 - 0.15) / (max(ys) - min(ys))
    
    for idx, p in enumerate(pts):
        centers[idx, 0] = 0.15 + sx * (p[0] - min(xs))
        centers[idx, 1] = 0.15 + sy * (p[1] - min(ys))
        
    # 2. Flatten variables and add small random perturbation to break symmetry
    x0 = np.zeros(3 * n)
    for i in range(n):
        x0[3*i] = centers[i, 0] + np.random.uniform(-0.002, 0.002)
        x0[3*i+1] = centers[i, 1] + np.random.uniform(-0.002, 0.002)
        x0[3*i+2] = radii[i]
        
    # Ensure initial guess strictly respects bounds
    x0[:2*n] = np.clip(x0[:2*n], 0.01, 0.99)
    x0[2*n:] = np.clip(x0[2*n:], 0.02, 0.25)
        
    bnds = [(0.0, 1.0)] * (2*n) + [(0.01, 0.3)] * n
    
    cons = {'type': 'ineq', 'fun': compute_constraints}
    
    # 3. Run optimization
    res = minimize(compute_objective, x0, method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': 2000, 'ftol': 1e-12, 'disp': False})
                   
    best_vars = res.x
    best_centers = np.zeros((n, 2))
    best_radii = np.zeros(n)
    for i in range(n):
        best_centers[i, 0] = best_vars[3*i]
        best_centers[i, 1] = best_vars[3*i+1]
        best_radii[i] = best_vars[3*i+2]
        
    # 4. Post-processing to guarantee strict validity
    # Clamp radii so circles are inside the square
    for i in range(n):
        max_r = min(best_centers[i,0], 1-best_centers[i,0], best_centers[i,1], 1-best_centers[i,1])
        if best_radii[i] > max_r:
            best_radii[i] = max_r
            
    # Iteratively resolve overlaps by shrinking radii
    for _ in range(100):
        changed = False
        for i in range(n):
            for j in range(i+1, n):
                dist = np.hypot(best_centers[i,0]-best_centers[j,0], best_centers[i,1]-best_centers[j,1])
                if dist < best_radii[i] + best_radii[j] - 1e-9:
                    excess = best_radii[i] + best_radii[j] - dist
                    best_radii[i] -= excess / 2
                    best_radii[j] -= excess / 2
                    changed = True
        if not changed:
            break
            
    best_radii = np.maximum(best_radii, 0.0)
    total = np.sum(best_radii)
    
    return best_centers, best_radii, float(total)
