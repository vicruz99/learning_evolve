import numpy as np
import math
from scipy.optimize import minimize


def make_hexagonal_centers(n, arrangement, r):
    """Generate initial hexagonal packing centers given a row arrangement."""
    centers = []
    for i, ncols in enumerate(arrangement):
        y = r + i * r * math.sqrt(3)
        row_width = ncols * 2 * r
        x_start = (1.0 - row_width) / 2.0
        for j in range(ncols):
            x = x_start + j * 2 * r
            centers.append([x, y])
    return np.array(centers)


def optimize_with_slsqp(params0, n=26):
    """Run SLSQP optimization with explicit constraints."""
    
    def objective(params):
        return -np.sum(params[2 * n:])
    
    constraints = []
    
    # Boundary constraints for each circle
    for i in range(n):
        xi = 2 * i
        yi = 2 * i + 1
        ri = 2 * n + i
        
        # x_i >= r_i  =>  x_i - r_i >= 0
        constraints.append({'type': 'ineq', 'fun': lambda p, i=i: p[2 * i] - p[2 * n + i]})
        # x_i <= 1 - r_i  =>  1 - x_i - r_i >= 0
        constraints.append({'type': 'ineq', 'fun': lambda p, i=i: 1.0 - p[2 * i] - p[2 * n + i]})
        # y_i >= r_i  =>  y_i - r_i >= 0
        constraints.append({'type': 'ineq', 'fun': lambda p, i=i: p[2 * i + 1] - p[2 * n + i]})
        # y_i <= 1 - r_i  =>  1 - y_i - r_i >= 0
        constraints.append({'type': 'ineq', 'fun': lambda p, i=i: 1.0 - p[2 * i + 1] - p[2 * n + i]})
    
    # Overlap constraints: ||c_i - c_j|| >= r_i + r_j
    for i in range(n):
        for j in range(i + 1, n):
            ci = np.array([2 * i, 2 * i + 1])
            cj = np.array([2 * j, 2 * j + 1])
            ri = 2 * n + i
            rj = 2 * n + j
            constraints.append({
                'type': 'ineq',
                'fun': lambda p, i=i, j=j, ci=ci, cj=cj, ri=ri, rj=rj: 
                    np.sqrt(np.sum((p[ci] - p[cj]) ** 2)) - p[ri] - p[rj]
            })
    
    # Bounds
    bounds = []
    for i in range(n):
        bounds.append((0.0, 1.0))  # x_i
        bounds.append((0.0, 1.0))  # y_i
    for i in range(n):
        bounds.append((0.0, 0.5))  # r_i
    
    result = minimize(
        objective,
        params0,
        method='SLSQP',
        constraints=constraints,
        bounds=bounds,
        options={'maxiter': 2000, 'ftol': 1e-14, 'disp': False}
    )
    
    return result.x
