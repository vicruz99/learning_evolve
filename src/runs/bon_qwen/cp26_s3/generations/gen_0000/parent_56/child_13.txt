2.  **Non-overlap Constraints**: For every pair of circles $i, j$, the distance between centers must be at least the sum of radii: $\sqrt{(x_i-x_j)^2 + (y_i-y_j)^2} \ge r_i + r_j$. Squaring both sides avoids square roots: $(x_i-x_j)^2 + (y_i-y_j)^2 \ge (r_i+r_j)^2$.

import numpy as np
from scipy.optimize import minimize

def run_packing():
    n = 26
    
    # Helper function to generate initial configuration
    def get_initial_guess(seed=0):
        rng = np.random.default_rng(seed)
        
        # Base configuration: 5x5 grid for 25 circles
        # Coordinates spaced by 0.2, centered at 0.1, 0.3, 0.5, 0.7, 0.9
        grid_coords = []
        for i in range(5):
            for j in range(5):
                grid_coords.append([0.1 + i * 0.2, 0.1 + j * 0.2])
        
        # Add 26th circle. 
        # Placing it at (0.5, 0.05) near the bottom edge center seems promising 
        # as it's equidistant from the two bottom circles (0.3, 0.1) and (0.7, 0.1)
        # but might need to push them apart.
        # Or placing it in a gap like (0.5, 0.5) is occupied. 
        # Let's try (0.5, 0.05) or random position.
        extra = [0.5, 0.05]
        
        centers = np.array(grid_coords + [extra])
        
        # Add small random noise to break symmetry
        noise = rng.uniform(-0.01, 0.01, centers.shape)
        centers = centers + noise
        centers = np.clip(centers, 0.05, 0.95) # Keep strictly inside to avoid boundary issues initially
        
        # Initial radii
        # Start with a small radius to ensure validity, let optimizer grow them
        radii = np.full(n, 0.04)
        
        # Flatten to [x1, y1, r1, x2, y2, r2, ...]
        x0 = np.zeros(3 * n)
        for i in range(n):
            x0[3*i] = centers[i, 0]
            x0[3*i+1] = centers[i, 1]
            x0[3*i+2] = radii[i]
            
        return x0

    # Objective function: Minimize negative sum of radii
    def objective(x):
        # Radii are at indices 2, 5, 8, ... (2::3)
        return -np.sum(x[2::3])

    # Constraints
    # 1. Boundary constraints: r <= x <= 1-r and r <= y <= 1-r
    # Inequality g(x) >= 0
    def boundary_constraints(x):
        # x shape (3n,)
        # Reshape to (n, 3) where rows are [x_i, y_i, r_i]
        # But our x is interleaved: x0, y0, r0, x1, y1, r1...
        # So we can reshape directly if we permute, or just slice
        
        # Extract vectors
        xs = x[0::3]
        ys = x[1::3]
        rs = x[2::3]
        
        constraints = []
        # x - r >= 0
        constraints.extend(xs - rs)
        # 1 - x - r >= 0
        constraints.extend(1.0 - xs - rs)
        # y - r >= 0
        constraints.extend(ys - rs)
        # 1 - y - r >= 0
        constraints.extend(1.0 - ys - rs)
        
        # r >= 0 is handled by bounds, but can add here if needed.
        # constraints.extend(rs)
        
        return np.array(constraints)

    # 2. Non-overlap constraints: dist_ij^2 >= (r_i + r_j)^2
    def non_overlap_constraints(x):
        # Extract centers and radii
        xs = x[0::3]
        ys = x[1::3]
        rs = x[2::3]
        
        # Vectorized distance calculation
        # xs shape (n,)
        # diff_x shape (n, n)
        diff_x = xs[:, None] - xs[None, :]
        diff_y = ys[:, None] - ys[None, :]
        
        dist_sq = diff_x**2 + diff_y**2
        
        # Radii sum squared
        r_sum = rs[:, None] + rs[None, :]
        r_sum_sq = r_sum**2
        
        # Constraint: dist_sq - r_sum_sq >= 0
        # We only need upper triangle (i < j)
        triu_idx = np.triu_indices(n, k=1)
        
        return dist_sq[triu_idx] - r_sum_sq[triu_idx]

    # Bounds for variables
    # x, y in [0, 1], r in [0, 0.5]
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * n
    
    # Define constraints for SLSQP
    # SLSQP expects constraints as list of dicts or Constraint objects
    # Using dicts for compatibility
    cons = [
        {'type': 'ineq', 'fun': boundary_constraints},
        {'type': 'ineq', 'fun': non_overlap_constraints}
    ]
    
    best_x = None
    best_val = -np.inf
    best_sum_r = 0.0
    
    # Run optimization multiple times with different seeds
    seeds = [42, 123, 456, 789, 101, 202, 303, 404]
    
    for seed in seeds:
        x0 = get_initial_guess(seed)
        
        try:
            res = minimize(
                objective, 
                x0, 
                method='SLSQP', 
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 2000, 'ftol': 1e-10, 'disp': False}
            )
            
            if res.success or res.fun < -2.0: # Threshold heuristic
                sum_r = -res.fun
                # Validate constraints roughly to ensure we aren't returning invalid result
                # Check overlap
                xs = res.x[0::3]
                ys = res.x[1::3]
                rs = res.x[2::3]
                
                # Check boundaries
                valid = True
                for i in range(n):
                    if xs[i] < rs[i] - 1e-6 or xs[i] > 1 - rs[i] + 1e-6 or \
                       ys[i] < rs[i] - 1e-6 or ys[i] > 1 - rs[i] + 1e-6:
                        valid = False
                        break
                
                if valid:
                    if sum_r > best_sum_r:
                        best_sum_r = sum_r
                        best_x = res.x
        except Exception as e:
            continue

    if best_x is None:
        # Fallback if optimization failed
        # Return a simple grid packing
        centers = np.array([[0.1 + 0.2*i, 0.1 + 0.2*j] for i in range(5) for j in range(5)] + [[0.5, 0.05]])
        radii = np.full(26, 0.04)
        return centers, radii, np.sum(radii)

    # Reconstruct centers and radii from best_x
    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3]
    
    return centers, radii, float(np.sum(radii))
