    def compute_constraints(vars, n):

            idx = 3 * i

            yi = vars[idx + 1]
            ri = vars[idx + 2]

            # x - r >= 0  =>  xi - ri >= 0
            # 1 - x - r >= 0 => 1 - xi - ri >= 0
            # y - r >= 0  =>  yi - ri >= 0
            # 1 - y - r >= 0 => 1 - yi - ri >= 0
            constraints.append(xi - ri)
            constraints.append(1.0 - xi - ri)
            constraints.append(yi - ri)
            constraints.append(1.0 - yi - ri)
            
            # Overlap constraints with subsequent circles
            for j in range(i + 1, n):
                jdx = 3 * j
                xj = vars[jdx]
                yj = vars[jdx + 1]
                rj = vars[jdx + 2]
                
                # (xi - xj)^2 + (yi - yj)^2 >= (ri + rj)^2
                # Constraint: dist_sq - (sum_r)^2 >= 0
                dist_sq = (xi - xj)**2 + (yi - yj)**2
                sum_r = ri + rj
                constraints.append(dist_sq - sum_r**2)
        
        return np.array(constraints)

            return compute_constraints(vars, n)
