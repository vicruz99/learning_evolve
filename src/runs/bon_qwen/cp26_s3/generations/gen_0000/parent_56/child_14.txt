import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    n = 26
    best_score = -1.0
    best_state = None

    def obj_func(x):
        # Unpack variables: r (26), cx (26), cy (26)
        r = x[:n]
        cx = x[n:2*n]
        cy = x[2*n:]
        
        # Objective: maximize sum of radii (we minimize the negative)
        score = -np.sum(r)
        
        # Calculate distances for overlap penalty
        # Vectorized distance calculation
        cx_diff = cx[:, np.newaxis] - cx[np.newaxis, :]
        cy_diff = cy[:, np.newaxis] - cy[np.newaxis, :]
        dist_mat = np.sqrt(cx_diff**2 + cy_diff**2)
        r_mat = r[:, np.newaxis] + r[np.newaxis, :]
        
        # Overlap penalty: if r_i + r_j > dist_ij
        violation = r_mat - dist_mat
        overlap_mask = violation > 0
        violation[~overlap_mask] = 0.0
        overlap_penalty = 5.0 * np.sum(violation**2)
        
        # Boundary penalty
        # r_i <= cx_i <= 1 - r_i  =>  r_i - cx_i <= 0 and cx_i + r_i - 1 <= 0
        # r_i <= cy_i <= 1 - r_i  =>  r_i - cy_i <= 0 and cy_i + r_i - 1 <= 0
        b1 = r - cx
        b2 = cx + r - 1.0
        b3 = r - cy
        b4 = cy + r - 1.0
        
        b1[b1 < 0] = 0; b2[b2 < 0] = 0; b3[b3 < 0] = 0; b4[b4 < 0] = 0
        boundary_penalty = 5.0 * (np.sum(b1**2) + np.sum(b2**2) + np.sum(b3**2) + np.sum(b4**2))
        
        return score + overlap_penalty + boundary_penalty

    def get_seed(seed_type):
        r_init = np.full(n, 0.08)
        cx, cy = np.zeros(n), np.zeros(n)
        
        if seed_type == 'hex':
            # Hexagonal grid seed
            k = 0
            rows = 7
            for i in range(rows):
                count = 5 if i % 2 == 0 else 4
                if k + count > n: count = n - k
                for j in range(count):
                    cx[k] = 0.06 + j * 0.16 + (0.08 if i % 2 != 0 else 0)
                    cy[k] = 0.06 + i * 0.13
                    r_init[k] = 0.095
                    k += 1
        elif seed_type == 'rand':
            r_init = np.random.uniform(0.05, 0.12, n)
            cx = np.random.uniform(0.1, 0.9, n)
            cy = np.random.uniform(0.1, 0.9, n)
        
        x0 = np.concatenate([r_init, cx, cy])
        # Bounds: r >= 0.01, coords in [0,1]
        bnds = [(0.01, 0.5)] * n + [(0.0, 1.0)] * (2 * n)
        return x0, bnds

    seed_types = ['hex', 'rand', 'hex'] # Run a few variations
    for st in seed_types:
        x0, bnds = get_seed(st)
        res = minimize(obj_func, x0, method='L-BFGS-B', bounds=bnds, options={'maxiter': 2000, 'ftol': 1e-12})
        
        r_final = res.x[:n]
        cx_final = res.x[n:2*n]
        cy_final = res.x[2*n:]
        centers_final = np.column_stack([cx_final, cy_final])
        
        if np.sum(r_final) > best_score:
            # Final cleanup to ensure strict non-negativity and boundary clamping
            r_final = np.clip(r_final, 0, None)
            
            # Clamp centers to be valid for the radii
            cx_final = np.clip(cx_final, r_final, 1.0 - r_final)
            cy_final = np.clip(cy_final, r_final, 1.0 - r_final)
            
            best_score = np.sum(r_final)
            best_state = (centers_final, r_final)

    centers, radii = best_state
    return centers, radii, np.sum(radii)
