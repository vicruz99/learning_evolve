from scipy.optimize import minimize

def compute_distance(c1, c2):
    """Compute Euclidean distance between two centers."""
    return np.sqrt(np.sum((c1 - c2) ** 2))

def get_initial_config(n=26):

    est_r = np.sqrt(1 / (n * np.pi * 0.85))

    # Pitch: 2*r, Vertical spacing: sqrt(3)*r
    pitch = 2 * est_r

    y_start = 0.5

def objective(params, n):
    """Objective function: negative sum of radii."""
    radii = params[2*n:]
    return -np.sum(radii)

def constraints_factory(n):
    """Factory to create constraints for optimization."""
    constraints = []
    
    # Boundary constraints: x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
    # x_i - r_i >= 0  =>  params[2*i] - params[2*n + i] >= 0
    # 1 - x_i - r_i >= 0 => 1 - params[2*i] - params[2*n + i] >= 0
    # y_i - r_i >= 0 => params[2*i + 1] - params[2*n + i] >= 0
    # 1 - y_i - r_i >= 0 => 1 - params[2*i + 1] - params[2*n + i] >= 0
    
    # We will define a single constraint function that returns an array of values >= 0
    # But scipy allows list of constraint dicts or a single dict returning array.
    # Let's use a single function returning array for efficiency.
    
    def boundary_and_overlap(params):
        centers = params[:2*n].reshape(n, 2)
        radii = params[2*n:]
        
        vals = []
        
        # Boundary
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            vals.append(x - r)
            vals.append(1.0 - x - r)
            vals.append(y - r)
            vals.append(1.0 - y - r)
            
        # Overlap: dist >= r_i + r_j
        # dist^2 >= (r_i + r_j)^2
        # dist^2 - (r_i + r_j)^2 >= 0
        # This avoids sqrt but is non-linear. SLSQP handles it.
        # However, dist - (r_i + r_j) >= 0 is also fine and convex-ish locally.
        # Let's use dist - (r_i + r_j) >= 0
        
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                vals.append(dist - (radii[i] + radii[j]))
                
        return np.array(vals)
    
    return {'type': 'ineq', 'fun': boundary_and_overlap}

def run_packing():
    """
    Pack 26 circles in a unit square to maximize sum of radii.
    
    Returns:
        tuple: (centers, radii, sum_radii)
    """
    n = 26
    
    # 1. Initial Guess
    centers_init, radii_init = get_initial_config(n)
    
    # Flatten for optimizer: [x1, y1, x2, y2, ..., r1, r2, ...]
    x0 = np.concatenate([centers_init.flatten(), radii_init])
    
    # Bounds
    # Centers in [0, 1]
    # Radii >= 0
    bounds = [(0, 1)] * (2 * n) + [(0, 0.5)] * n # Max radius 0.5
    
    # Constraints
    cons = constraints_factory(n)
    
    # 2. Optimization
    # Use SLSQP for constrained optimization
    # We might need to run multiple times or use a robust method.
    # SLSQP is good for this.
    
    # To help with non-convexity, we can try to start with a larger valid radius if possible
    # But the init function provides a valid small radius.
    
    res = minimize(
        objective, 
        x0, 
        args=(n,),
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'ftol': 1e-12, 'maxiter': 1000, 'disp': False}
    )
    
    if not res.success:
        # Fallback or warning, but usually it works or gives best found
        pass
        
    best_params = res.x
    best_centers = best_params[:2*n].reshape(n, 2)
    best_radii = best_params[2*n:]
    
    # Clip radii to be non-negative just in case
    best_radii = np.maximum(best_radii, 0.0)
    
    sum_radii = np.sum(best_radii)
    
    return best_centers, best_radii, sum_radii
