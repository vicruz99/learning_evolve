def get_constraints(centers, radii):
    """
    Returns a list of constraint dictionaries for scipy.optimize.
    Constraints: 
    1. Circles inside [0, 1]x[0, 1]
    2. No overlaps between circles
    """
    n = len(centers)
    cons = []

    # Boundary constraints: x >= r, x <= 1-r, y >= r, y <= 1-r
    # This is equivalent to r <= x <= 1-r and r <= y <= 1-r
    # We handle this via bounds or constraints. Let's use bounds in minimize.
    
    # Overlap constraints: dist(i, j) >= r_i + r_j
    # dist^2 >= (r_i + r_j)^2
    for i in range(n):
        for j in range(i + 1, n):
            # Constraint function: (x_i-x_j)^2 + (y_i-y_j)^2 - (r_i+r_j)^2 >= 0
            def overlap_constraint(x, i=i, j=j):
                c_i = np.array([x[3*i], x[3*i+1]])
                c_j = np.array([x[3*j], x[3*j+1]])
                r_i = x[3*i+2]
                r_j = x[3*j+2]
                return np.sum((c_i - c_j)**2) - (r_i + r_j)**2
            
            cons.append({'type': 'ineq', 'fun': overlap_constraint})
            
    return cons
