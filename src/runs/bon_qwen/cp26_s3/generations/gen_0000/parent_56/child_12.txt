import numpy as np
from scipy.optimize import minimize


def make_constraints_and_jacobian(n=26):
    """Build constraint functions for SLSQP optimization."""
    def constraint_func(params):
        centers = params[:2 * n].reshape(n, 2)
        radii = params[2 * n:]
        c = []
        # Boundary constraints
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            c.append(x - r)
            c.append(1.0 - x - r)
            c.append(y - r)
            c.append(1.0 - y - r)
            c.append(r)
        # Non-overlap constraints
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                c.append(dist - radii[i] - radii[j])
        return np.array(c)

    return constraint_func


def objective_func(params, n=26):
    """Negative sum of radii."""
    return -np.sum(params[2 * n:])


def hexagonal_init(n, rows=None, cols=None):
    """Initialize circles in hexagonal pattern."""
    if rows is None:
        rows = min(6, int(np.ceil(n / 5)))
    if cols is None:
        cols = min(5, int(np.ceil(n / rows)))
    
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    idx = 0
    r_init = 0.08
    
    for row in range(rows):
        n_in_row = cols if row % 2 == 0 else (cols - 1 if cols > 1 else cols)
        shift = 0.5 * (2 * r_init) if row % 2 == 1 else 0
        for col in range(n_in_row):
            if idx >= n:
                break
            x = r_init + col * (2 * r_init) + shift
            y = r_init + row * np.sqrt(3) * r_init
            centers[idx] = [x, y]
            radii[idx] = r_init
            idx += 1
    
    return centers, radii


def grid_init(n):
    """Initialize circles in a grid pattern."""
    k = int(np.ceil(np.sqrt(n)))
    m = int(np.ceil(n / k))
    
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    dx = 1.0 / (k + 1)
    dy = 1.0 / (m + 1)
    r = min(dx, dy) * 0.5
    
    idx = 0
    for i in range(1, k + 1):
        for j in range(1, m + 1):
            if idx >= n:
                break
            centers[idx] = [i * dx, j * dy]
            radii[idx] = r
            idx += 1
    
    return centers, radii


def random_init_with_relaxation(n, seed=0):
    """Random initialization with force-based relaxation."""
    rng = np.random.RandomState(seed)
    
    # Start with random positions and small radii
    centers = rng.rand(n, 2)
    radii = np.ones(n) * 0.03
    
    # Force-directed relaxation
    lr = 0.02
    for iteration in range(5000):
        forces = np.zeros_like(centers)
        
        for i in range(n):
            for j in range(i + 1, n):
                diff = centers[j] - centers[i]
                dist = np.linalg.norm(diff)
                if dist < 1e-10:
                    dist = 1e-10
                    diff = rng.rand(2) * 0.01
                desired = radii[i] + radii[j]
                if dist < desired:
                    f = (desired - dist) / dist * 0.5
                    forces[i] -= f * diff
                    forces[j] += f * diff
            
            for d in range(2):
                if centers[i, d] - radii[i] < 0:
                    forces[i, d] += (radii[i] - centers[i, d]) * 5.0
                if centers[i, d] + radii[i] > 1.0:
                    forces[i, d] -= (centers[i, d] + radii[i] - 1.0) * 5.0
        
        centers += forces * lr
        centers = np.clip(centers, 0.001, 0.999)
        
        # Gradually increase radii
        for i in range(n):
            max_r = min(centers[i, 0], 1 - centers[i, 0],
                       centers[i, 1], 1 - centers[i, 1])
            for j in range(n):
                if i != j:
                    d = np.linalg.norm(centers[i] - centers[j])
                    max_r = min(max_r, d - radii[j])
            radii[i] = min(max_r * 0.99, radii[i] * 1.005)
            radii[i] = max(radii[i], 0.01)
        
        lr *= 0.9995
    
    return centers, radii


def concentric_ring_init(n):
    """Initialize circles in concentric ring pattern."""
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    idx = 0
    # Central circle
    centers[idx] = [0.5, 0.5]
    radii[idx] = 0.15
    idx += 1
    
    # Rings
    ring = 1
    while idx < n:
        if ring == 1:
            n_in_ring = min(6, n - idx)
        else:
            n_in_ring = min(6 * ring, n - idx)
        
        angle_step = 2 * np.pi / n_in_ring
        radius_ring = 0.15 + ring * 0.12
        r_circle = 0.08 / (1 + ring * 0.3)
        
        for k in range(n_in_ring):
            if idx >= n:
                break
            angle = k * angle_step + ring * 0.3
            x = 0.5 + radius_ring * np.cos(angle)
            y = 0.5 + radius_ring * np.sin(angle)
            centers[idx] = [np.clip(x, r_circle + 0.001, 1 - r_circle - 0.001),
                          np.clip(y, r_circle + 0.001, 1 - r_circle - 0.001)]
            radii[idx] = r_circle
            idx += 1
        ring += 1
    
    return centers, radii

    return centers, radii


def optimize_packing(centers_init, radii_init, n=26, verbose=False):
    """Optimize packing using scipy SLSQP."""
    centers = centers_init.copy()
    radii = radii_init.copy()
    
    # Combine into parameter vector
    params = np.concatenate([centers.flatten(), radii])
    
    # Bounds: x,y in [0,1], r in [0, 0.5]
    bounds = []
    for i in range(2 * n):
        bounds.append((0.0, 1.0))
    for i in range(n):
        bounds.append((0.0, 0.5))
    
    constraint_func = make_constraints_and_jacobian(n)
    
    # Add small perturbation to avoid degenerate starts
    rng = np.random.RandomState(42)
    params += rng.randn(len(params)) * 0.001
    
    try:
        result = minimize(
            objective_func,
            params,
            args=(n,),
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'ineq', 'fun': constraint_func},
            options={'maxiter': 2000, 'ftol': 1e-15, 'disp': False}
        )
        
        if result.success or result.fun < -2.0:
            final_centers = result.x[:2 * n].reshape(n, 2)
            final_radii = result.x[2 * n:]
            
            # Clip to valid range
            final_radii = np.maximum(final_radii, 1e-10)
            
            # Ensure boundary constraints are satisfied
            for i in range(n):
                r = final_radii[i]
                final_centers[i, 0] = np.clip(final_centers[i, 0], r, 1 - r)
                final_centers[i, 1] = np.clip(final_centers[i, 1], r, 1 - r)
            
            sum_r = np.sum(final_radii)
            
            if verbose:
                print(f"Optimization result: sum_radii = {sum_r:.6f}, success = {result.success}")
            
            return final_centers, final_radii, sum_r
    except Exception as e:
        if verbose:
            print(f"Optimization failed: {e}")
    
    return centers, radii, np.sum(radii)


def force_optimize(centers_init, radii_init, n=26, iterations=8000):
    """Force-based optimization to improve packing."""
    centers = centers_init.copy()
    radii = radii_init.copy()
    
    lr = 0.01
    rng = np.random.RandomState(123)
    
    for it in range(iterations):
        forces = np.zeros_like(centers)
        total_force = 0.0
        
        # Repulsive forces between circles
        for i in range(n):
            for j in range(i + 1, n):
                diff = centers[j] - centers[i]
                dist_sq = diff[0]**2 + diff[1]**2
                dist = np.sqrt(dist_sq)
                if dist < 1e-10:
                    dist = 1e-10
                    diff = rng.rand(2) * 0.001
                
                desired = radii[i] + radii[j]
                if dist < desired + 0.01:
                    overlap = desired - dist
                    if overlap > 0:
                        f = overlap / dist * 10.0
                    else:
                        f = 0.01 / (dist + 0.01)
                    forces[i] -= f * diff
                    forces[j] += f * diff
                    total_force += f
        
        # Wall forces
        for i in range(n):
            for d in range(2):
                if centers[i, d] - radii[i] < 0.001:
                    forces[i, d] += max(0, (radii[i] - centers[i, d]) * 50.0)
                if centers[i, d] + radii[i] > 0.999:
                    forces[i, d] -= max(0, (centers[i, d] + radii[i] - 1.0) * 50.0)
        
        # Apply forces
        step = lr / (1 + it * 0.0003)
        centers += forces * step
        centers = np.clip(centers, 0.0001, 0.9999)
        
        # Inflate radii slightly
        for i in range(n):
            max_r = min(centers[i, 0], 1 - centers[i, 0],
                       centers[i, 1], 1 - centers[i, 1])
            for j in range(n):
                if i != j:
                    d = np.linalg.norm(centers[i] - centers[j])
                    max_r = min(max_r, d - radii[j])
            target = max_r * 0.98
            radii[i] = radii[i] + (target - radii[i]) * 0.01
            radii[i] = max(radii[i], 0.005)
    
    sum_r = np.sum(radii)
    return centers, radii, sum_r


def validate_and_fix(centers, radii, n=26):
    """Validate and fix any constraint violations."""
    # Fix boundary violations by reducing radius
    for i in range(n):
        r = radii[i]
        x, y = centers[i]
        max_r = min(x, 1 - x, y, 1 - y)
        if r > max_r:
            radii[i] = max_r * 0.999
            r = radii[i]
        centers[i, 0] = np.clip(x, r, 1 - r)
        centers[i, 1] = np.clip(y, r, 1 - r)
    
    # Fix overlaps by reducing radii
    for iteration in range(100):
        max_violation = 0
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.linalg.norm(centers[i] - centers[j])
                needed = radii[i] + radii[j]
                if dist < needed - 1e-12:
                    violation = needed - dist
                    max_violation = max(max_violation, violation)
                    # Reduce both radii proportionally
                    factor = dist / (radii[i] + radii[j])
                    radii[i] *= factor * 0.999
                    radii[j] *= factor * 0.999
        
        if max_violation < 1e-12:
            break
    
    return centers, radii
