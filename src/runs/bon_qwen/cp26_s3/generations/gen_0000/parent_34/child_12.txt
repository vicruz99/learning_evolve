import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Finds a packing of 26 circles in a unit square maximizing sum of radii.
    """
    N = 26
    
    def hexagonal_points(n, r_spacing):
        """Generate points in a hexagonal grid."""
        points = []
        rows = int(np.ceil(np.sqrt(n * 2 / np.sqrt(3))))
        while len(points) < n:
            for r in range(rows):
                y = r * r_spacing * np.sqrt(3) / 2
                shift = r_spacing / 2 if r % 2 == 1 else 0
                cols = int(np.ceil((1.0 + r_spacing) / r_spacing))
                for c in range(cols):
                    x = c * r_spacing + shift
                    if len(points) < n:
                        points.append([x, y])
        return np.array(points[:n])

    # Pre-generate a hexagonal grid structure for initialization
    base_centers = hexagonal_points(N, 0.15)
    
    # Scale to fit in square [0, 1] x [0, 1]
    min_coords = base_centers.min(axis=0)
    max_coords = base_centers.max(axis=0)
    dims = max_coords - min_coords
    if dims[0] > 0 and dims[1] > 0:
        base_centers = (base_centers - min_coords) / dims * 0.8 + 0.1

    best_sum = 0
    best_result = None

    # Run multiple restarts to find the best local optimum
    for trial in range(20):
        # Perturb initial positions
        current_centers = base_centers.copy()
        if trial > 0:
            current_centers += np.random.normal(0, 0.02, size=current_centers.shape)
        
        # Clip to valid range
        current_centers = np.clip(current_centers, 0.05, 0.95)

        # Variables: [x0, y0, r0, x1, y1, r1, ...]
        x0 = current_centers.flatten()
        r0 = np.full(N, 0.04) # Start with small valid radius
        params0 = np.concatenate([x0, r0])

        def objective(p):
            # Maximize sum of radii => Minimize negative sum
            return -np.sum(p[2::3])

        def constraints_func(p):
            cons = []
            x = p[0::3]
            y = p[1::3]
            r = p[2::3]

            # Boundary constraints
            # x - r >= 0, x + r <= 1 => r <= x, r <= 1-x
            # y - r >= 0, y + r <= 1 => r <= y, r <= 1-y
            cons.append({'type': 'ineq', 'fun': lambda p: p[2::3] - (1 - p[0::3]), 'args': ()}) # r <= 1-x
            cons.append({'type': 'ineq', 'fun': lambda p: p[0::3] - p[2::3], 'args': ()})    # x >= r
            cons.append({'type': 'ineq', 'fun': lambda p: p[2::3] - (1 - p[1::3]), 'args': ()}) # r <= 1-y
            cons.append({'type': 'ineq', 'fun': lambda p: p[1::3] - p[2::3], 'args': ()})    # y >= r

            # Overlap constraints
            # dist(i, j) >= r_i + r_j
            for i in range(N):
                for j in range(i + 1, N):
                    idx_i = i * 3
                    idx_j = j * 3
                    def make_overlap(i, j):
                        def overlap(p):
                            dx = p[idx_i] - p[idx_j + 0]
                            dy = p[idx_i + 1] - p[idx_j + 1]
                            dist = np.sqrt(dx**2 + dy**2)
                            return dist - (p[idx_i + 2] + p[idx_j + 2])
                        return overlap
                    cons.append({'type': 'ineq', 'fun': make_overlap(i, j)})
            
            # Minimum radius to prevent collapse (small but positive)
            cons.append({'type': 'ineq', 'fun': lambda p: p[2::3] - 0.01})
            return cons

        # Bounds: x,y in [0,1], r in [0, 0.5]
        bounds = []
        for i in range(N):
            bounds.append((0.0, 1.0)) # x
            bounds.append((0.0, 1.0)) # y
            bounds.append((0.0, 0.5)) # r

        # SLSQP solver
        try:
            res = minimize(objective, params0, method='SLSQP', bounds=bounds, 
                          constraints=constraints_func(params0), 
                          options={'ftol': 1e-8, 'maxiter': 1000})
            
            if res.success:
                sum_r = -res.fun
                if sum_r > best_sum:
                    best_sum = sum_r
                    best_result = res
        except Exception:
            continue

    if best_result is None:
        # Fallback to a simple grid if optimization fails completely
        centers = hexagonal_points(N, 0.15)
        centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0)) * 0.8 + 0.1
        radii = np.full(N, 0.04)

    # Extract best solution
    best_params = best_result.x
    centers = np.column_stack((best_params[0::3], best_params[1::3]))
    radii = best_params[2::3]
    
    # Clip radii just in case of numerical drift
    radii = np.clip(radii, 1e-6, 0.5)

    return centers, radii, float(np.sum(radii))
