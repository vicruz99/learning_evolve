import numpy as np
from scipy.optimize import minimize

def run_packing():
    n = 26
    
    # 1. Initialize with a hexagonal grid
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Configuration: 5 rows. Counts: 6, 5, 6, 5, 4 (Total 26)
    row_counts = [6, 5, 6, 5, 4]
    
    # Estimate initial radius to fit in unit square
    # Approximation for hexagonal packing in unit square
    r_est = 0.08 
    dy = r_est * np.sqrt(3)
    dx = 2 * r_est
    
    current_idx = 0
    for row_idx, count in enumerate(row_counts):
        y = r_est + row_idx * dy
        
        # Calculate starting x for the row to center it horizontally
        total_width = (count - 1) * dx
        x_start = (1.0 - total_width) / 2.0
        
        for col_idx in range(count):
            x = x_start + col_idx * dx
            centers[current_idx] = [x, y]
            radii[current_idx] = r_est
            current_idx += 1

    # 2. Define the objective function (maximize sum of radii -> minimize negative sum)
    def objective(x_vars):
        return -np.sum(x_vars[n*2:n*2+n]) # radii are the last n variables

    # 3. Define constraints
    def constraint_boundary(x_vars):
        # Returns array of (x - r), (1 - x - r), (y - r), (1 - y - r) for all circles
        cons = []
        for i in range(n):
            x = x_vars[i*2]
            y = x_vars[i*2+1]
            r = x_vars[n*2+i]
            cons.extend([
                x - r,
                1 - x - r,
                y - r,
                1 - y - r
            ])
        return np.array(cons)

    def constraint_non_overlap(x_vars):
        # Returns array of (dist^2 - (r_i + r_j)^2) for all pairs
        cons = []
        for i in range(n):
            for j in range(i + 1, n):
                x1, y1 = x_vars[i*2], x_vars[i*2+1]
                r1 = x_vars[n*2+i]
                x2, y2 = x_vars[j*2], x_vars[j*2+1]
                r2 = x_vars[n*2+j]
                
                dist_sq = (x1 - x2)**2 + (y1 - y2)**2
                min_dist = (r1 + r2)**2
                cons.append(dist_sq - min_dist)
        return np.array(cons)

    # Flatten variables: [x1, y1, x2, y2, ..., x26, y26, r1, r2, ..., r26]
    x0 = np.concatenate((centers.flatten(), radii))
    
    # Bounds for variables
    bounds = []
    for i in range(n):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 0.5)) # r
    
    # Solve using SLSQP
    result = minimize(
        objective, 
        x0, 
        method='SLSQP', 
        bounds=bounds, 
        constraints=[
            {'type': 'ineq', 'fun': constraint_boundary},
            {'type': 'ineq', 'fun': constraint_non_overlap}
        ],
        options={'maxiter': 1000, 'ftol': 1e-9}
    )

    # Extract results
    final_centers = result.x[:n*2].reshape((n, 2))
    final_radii = result.x[n*2:]
    sum_radii = np.sum(final_radii)

    return final_centers, final_radii, sum_radii

if __name__ == "__main__":
    centers, radii, total = run_packing()
    print(f"Optimization completed. Sum of radii: {total:.6f}")
    print(f"Minimum radius: {np.min(radii):.6f}, Maximum radius: {np.max(radii):.6f}")
