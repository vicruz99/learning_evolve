import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Packs 26 circles in a unit square [0,1]x[0,1] to maximize the sum of radii.
    
    Returns:
        tuple: (centers, radii, sum_radii)
    """
    n_circles = 26
    
    # 1. Initialization: Hexagonal Grid
    # We generate a hexagonal grid pattern to serve as a strong starting point.
    # We assume a tentative radius to space the circles, then let the optimizer adjust.
    # A radius of ~0.09 fits easily, giving the optimizer room to expand.
    
    initial_centers = []
    # Parameters for hexagonal packing
    # Rows are horizontal. Vertical spacing is sqrt(3)*r. Horizontal spacing is 2r.
    # Shift odd rows by r horizontally.
    
    # Let's try to fit rows. 
    # If we have roughly 5 rows, we can fit ~5 circles per row.
    # 5*5 = 25. We need 26. So maybe 6, 5, 5, 5, 5 or 5, 6, 5, 5, 5.
    
    # Let's generate points in a dense hex grid and pick the best 26?
    # Or just manually construct a valid set.
    
    # Strategy: Fill rows.
    # Row 0: y = r. x = r, 3r, 5r...
    # Row 1: y = r + sqrt(3)r. x = 2r, 4r, 6r...
    
    # Let's pick a safe initial radius r_init = 0.08
    r_init = 0.08
    h = np.sqrt(3) * r_init
    
    rows_y = []
    y_curr = r_init
    while y_curr + r_init <= 1.0:
        rows_y.append(y_curr)
        y_curr += h
    
    # Determine how many circles we can fit in each row
    # Row i: if i is even, x starts at r_init, step 2*r_init.
    # if i is odd, x starts at 3*r_init (shifted by r_init), step 2*r_init.
    # Actually shifted by r_init means x starts at r_init + r_init = 2*r_init.
    
    candidates = []
    for i, y in enumerate(rows_y):
        if i % 2 == 0:
            x_start = r_init
        else:
            x_start = 3 * r_init # Shift by one diameter? No, shift by radius in hex packing is usually r?
            # In hex packing, horizontal distance between adjacent circles in same row is 2r.
            # Vertical distance is sqrt(3)r.
            # Horizontal shift between rows is r.
            # So if row 0 starts at r, row 1 starts at 2r.
            x_start = 2 * r_init 
        
        x = x_start
        while x + r_init <= 1.0:
            candidates.append([x, y])
            x += 2 * r_init
            
    # If we have enough candidates, pick first 26. If not, add some.
    # With r=0.08, 2r=0.16.
    # Width 1.0. Max x count approx 1/0.16 ~ 6.
    # Height 1.0. h ~ 0.138. Max rows ~ 1/0.138 ~ 7.
    # We should have plenty of candidates.
    
    if len(candidates) < n_circles:
        # Fallback to random grid if hex generation fails (unlikely with r=0.08)
        candidates = []
        grid_size = 6
        for i in range(grid_size):
            for j in range(grid_size):
                if len(candidates) >= n_circles:
                    break
                candidates.append([ (i + 0.5) / grid_size, (j + 0.5) / grid_size ])
    
    selected_centers = np.array(candidates[:n_circles])
    initial_radii = np.full(n_circles, r_init)
    
    # 2. Optimization Setup
    # Variables: [x1, y1, r1, x2, y2, r2, ..., x26, y26, r26]
    # Total 3 * 26 = 78 variables.
    
    x0 = np.concatenate([selected_centers.flatten(), initial_radii])
    
    # Bounds: x, y in [0, 1], r in [0, 0.5]
    bounds = []
    for _ in range(n_circles):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 0.5)) # r
    
    # Objective: Maximize sum of radii => Minimize -sum(r)
    def objective(x):
        radii = x[2::3]
        return -np.sum(radii)
    
    # Constraints
    constraints = []
    
    # Boundary constraints:
    # x_i >= r_i  => x_i - r_i >= 0
    # x_i + r_i <= 1 => 1 - x_i - r_i >= 0
    # y_i >= r_i  => y_i - r_i >= 0
    # y_i + r_i <= 1 => 1 - y_i - r_i >= 0
    
    for i in range(n_circles):
        idx_x = 3 * i
        idx_y = 3 * i + 1
        idx_r = 3 * i + 2
        
        # x - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda x, i=i: x[idx_x] - x[idx_r]
        })
        # 1 - x - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda x, i=i: 1.0 - x[idx_x] - x[idx_r]
        })
        # y - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda x, i=i: x[idx_y] - x[idx_r]
        })
        # 1 - y - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda x, i=i: 1.0 - x[idx_y] - x[idx_r]
        })
        
    # Pairwise non-overlap constraints:
    # dist(i, j) >= r_i + r_j
    # dist^2 >= (r_i + r_j)^2
    # (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
    
    for i in range(n_circles):
        for j in range(i + 1, n_circles):
            idx_xi, idx_yi, idx_ri = 3 * i, 3 * i + 1, 3 * i + 2
            idx_xj, idx_yj, idx_rj = 3 * j, 3 * j + 1, 3 * j + 2
            
            constraints.append({
                'type': 'ineq',
                'fun': lambda x, i=i, j=j: \
                    (x[3*i] - x[3*j])**2 + (x[3*i+1] - x[3*j+1])**2 - (x[3*i+2] + x[3*j+2])**2
            })
            
    # Solve
    # Using SLSQP method
    try:
        result = minimize(
            objective,
            x0,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 1000, 'ftol': 1e-9, 'disp': False}
        )
        
        if result.success:
            final_centers = result.x[:2*n_circles].reshape(n_circles, 2)
            final_radii = result.x[2*n_circles:]
        else:
            # Fallback to initial if optimization fails
            final_centers = selected_centers
            final_radii = initial_radii
    except Exception:
        final_centers = selected_centers
        final_radii = initial_radii

    # 3. Validation and Cleanup
    # Ensure no negative radii and clamp if necessary (though optimizer should handle)
    final_radii = np.maximum(final_radii, 0.0)
    
    # Recalculate sum
    sum_radii = np.sum(final_radii)
    
    return final_centers, final_radii, sum_radii
