import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26
PENALTY_WEIGHT = 1e6

def compute_objective(v):
    """Computes the negative sum of radii plus penalty for constraint violations."""
    n = N_CIRCLES
    c = v[:2*n].reshape(n, 2)
    r = v[2*n:]
    
    # Pairwise overlap penalty
    diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
    dist = np.sqrt(np.sum(diff**2, axis=2) + 1e-12)
    overlap = r[:, np.newaxis] + r[np.newaxis, :] - dist
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    pen_pair = np.sum(np.maximum(overlap[mask], 0.0)**2)
    
    # Boundary penalty
    pen_bound = np.sum(np.maximum(0.0, r - c[:, 0])**2)
    pen_bound += np.sum(np.maximum(0.0, r - (1.0 - c[:, 0]))**2)
    pen_bound += np.sum(np.maximum(0.0, r - c[:, 1])**2)
    pen_bound += np.sum(np.maximum(0.0, r - (1.0 - c[:, 1]))**2)
    
    # Radii non-negativity penalty
    pen_r = np.sum(np.maximum(0.0, -r)**2)
    
    return -np.sum(r) + PENALTY_WEIGHT * (pen_pair + pen_bound + pen_r)

def get_grid_init():
    """Returns initial configuration based on a perturbed 5x5 grid plus one circle."""
    r0 = 0.09
    c = np.array([[i*0.2 + 0.1, j*0.2 + 0.1] for i in range(5) for j in range(5)])
    c = np.vstack([c, [0.5, 0.15]])
    c += np.random.randn(*c.shape) * 0.002
    v0 = np.concatenate([c.flatten(), np.ones(N_CIRCLES)*r0])
    return v0

def get_hex_init():
    """Returns initial configuration based on a hexagonal lattice."""
    r0 = 0.095
    c = []
    for row in range(6):
        for col in range(5):
            if len(c) >= N_CIRCLES: break
            x = col * 2 * r0 + r0
            y = row * np.sqrt(3) * r0 + r0
            if row % 2 == 1: 
                x += r0
            if x + r0 <= 1 and y + r0 <= 1:
                c.append([x, y])
        if len(c) >= N_CIRCLES: break
    c = np.array(c)
    c += np.random.randn(*c.shape) * 0.002
    v0 = np.concatenate([c.flatten(), np.ones(N_CIRCLES)*r0])
    return v0

def get_random_init():
    """Returns a random initial configuration."""
    c = np.random.rand(N_CIRCLES, 2) * 0.8 + 0.1
    v0 = np.concatenate([c.flatten(), np.full(N_CIRCLES, 0.08)])
    return v0

def validate_and_fix(centers, radii):
    """Ensures the packing strictly satisfies constraints within tolerance."""
    n = centers.shape[0]
    c = centers.copy()
    r = radii.copy()
    
    # Boundary clamping
    for i in range(n):
        max_r = min(c[i,0], 1.0 - c[i,0], c[i,1], 1.0 - c[i,1])
        if r[i] > max_r:
            r[i] = max_r
            
    # Overlap resolution by slight shrinking
    for _ in range(50):
        valid = True
        for i in range(n):
            for j in range(i + 1, n):
                d = np.linalg.norm(c[i] - c[j])
                if d < r[i] + r[j] - 1e-12:
                    valid = False
                    break
            if not valid: break
        if valid: break
        r *= 0.9995
        
    return c, r

def run_packing():
    """Finds the optimal packing of 26 circles in a unit square."""
    best_sum = 0.0
    best_v = None
    
    # Try multiple initializations
    inits = [get_grid_init(), get_hex_init(), get_random_init(), get_random_init(), get_random_init()]
    
    bounds = [(0.0, 1.0) for _ in range(2*N_CIRCLES)] + [(0.0, 0.5) for _ in range(N_CIRCLES)]
    
    for v0 in inits:
        res = minimize(compute_objective, v0, method='L-BFGS-B', bounds=bounds, 
                       options={'maxiter': 8000, 'ftol': 1e-12, 'gtol': 1e-8})
        
        c_opt = res.x[:2*N_CIRCLES].reshape(N_CIRCLES, 2)
        r_opt = res.x[2*N_CIRCLES:]
        current_sum = np.sum(r_opt)
        
        if current_sum > best_sum:
            best_sum = current_sum
            best_v = res.x
            
    centers = best_v[:2*N_CIRCLES].reshape(N_CIRCLES, 2)
    radii = best_v[2*N_CIRCLES:]
    
    # Post-processing to guarantee validation passes
    centers, radii = validate_and_fix(centers, radii)
    
    return centers, radii, np.sum(radii)
