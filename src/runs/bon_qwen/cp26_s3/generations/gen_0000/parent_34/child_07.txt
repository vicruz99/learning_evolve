import numpy as np
from scipy.optimize import minimize
import itertools

def run_packing():

    Packs 26 circles in a unit square [0,1]x[0,1] to maximize the sum of radii.
    Uses a hexagonal initialization followed by SLSQP optimization.
    """
    n_circles = 26
    
    # Function to generate a hexagonal initial packing
    def generate_hex_packing(n, r_init=0.08):
        centers = []
        # Row configuration: 5, 4, 5, 4, 5, 3 sums to 26
        row_counts = [5, 4, 5, 4, 5, 3]
        
        y_curr = r_init
        row_idx = 0
        
        while len(centers) < n and row_idx < len(row_counts):
            count = row_counts[row_idx]
            # Width of row with 'count' circles of radius r
            # Total width needed = 2*r*count
            # We want to center the row in [0, 1]
            # Start x = (1 - 2*r*count) / 2 + r
            # But simpler: centers at r, 3r, 5r...
            # Let's just place them spaced by 2*r
            
            # For odd rows (0, 2, 4...), align with left? Or center?
            # Let's center the row.
            row_width = 2 * r_init * count
            start_x = (1.0 - row_width) / 2.0 + r_init
            
            for i in range(count):
                x = start_x + i * (2 * r_init)
                centers.append([x, y_curr])
                if len(centers) >= n:
                    break
            
            y_curr += r_init * np.sqrt(3)
            row_idx += 1
            
        # Ensure we have exactly n centers
        centers = centers[:n]
        
        # Convert to array and initialize radii
        centers_arr = np.array(centers)
        radii_arr = np.full(n, r_init)
        
        return centers_arr, radii_arr

    # Optimization function
    # Variables: [x0, y0, r0, x1, y1, r1, ...]
    # Size: 3 * n
    
    best_sum_radii = -1.0
    best_centers = None
    best_radii = None

    # Try multiple random restarts to find a better local optimum
    num_restarts = 10
    
    for _ in range(num_restarts):
        # 1. Initialize
        centers_init, radii_init = generate_hex_packing(n_circles, r_init=0.08)
        
        # Add small random noise to break symmetry and help optimization
        noise_scale = 0.01
        centers_init += np.random.uniform(-noise_scale, noise_scale, centers_init.shape)

        centers_init = np.clip(centers_init, radii_init, 1 - radii_init)
        
        # Flatten variables
        x0 = np.hstack([centers_init.flatten(), radii_init])

        constraints = []

        # 1 - x_i - r_i >= 0
        # y_i - r_i >= 0
        # 1 - y_i - r_i >= 0
        for i in range(n_circles):
            idx_x = 3 * i
            idx_y = 3 * i + 1
            idx_r = 3 * i + 2
            
            # x - r >= 0
            constraints.append({
                'type': 'ineq',
                'fun': lambda v, i=i: v[idx_x] - v[idx_r]
            })
            # 1 - x - r >= 0
            constraints.append({
                'type': 'ineq',
                'fun': lambda v, i=i: 1.0 - v[idx_x] - v[idx_r]
            })
            # y - r >= 0
            constraints.append({
                'type': 'ineq',
                'fun': lambda v, i=i: v[idx_y] - v[idx_r]
            })
            # 1 - y - r >= 0
            constraints.append({
                'type': 'ineq',
                'fun': lambda v, i=i: 1.0 - v[idx_y] - v[idx_r]
            })
            # r >= 0 (handled by bounds usually, but let's add constraint or bounds)
        
        # Non-overlap constraints: dist(i, j) >= r_i + r_j
        # dist^2 >= (r_i + r_j)^2
        # (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0

            for j in range(i + 1, n_circles):
                idx_xi = 3 * i
                idx_yi = 3 * i + 1
                idx_ri = 3 * i + 2
                idx_xj = 3 * j
                idx_yj = 3 * j + 1
                idx_rj = 3 * j + 2
                
                # Using squared distance to avoid sqrt in constraint function if possible?
                # Actually SLSQP might handle sqrt fine, but squared is safer for smoothness?
                # sqrt is smooth for dist > 0.
                # Let's use dist - (r_i + r_j) >= 0
                
                def constraint_overlap(v, i=i, j=j):
                    xi, yi, ri = v[3*i], v[3*i+1], v[3*i+2]
                    xj, yj, rj = v[3*j], v[3*j+1], v[3*j+2]
                    dist = np.sqrt((xi - xj)**2 + (yi - yj)**2)
                    return dist - (ri + rj)
                
                constraints.append({
                    'type': 'ineq',

                })

            r = v[2::3] # Every 3rd element starting from index 2
            return -np.sum(r)

                objective, 
                x0, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=constraints,
                options={'maxiter': 1000, 'ftol': 1e-9, 'disp': False}
            )

    # Fallback if optimization fails completely

        best_centers = centers_fallback
