We arrange them in rows of counts `[6, 5, 6, 5, 4]` which sums to 26, fitting naturally within the square's aspect ratio.

import numpy as np
from scipy.optimize import minimize

def packing_objective(vars, n):
    """
    Objective function for circle packing optimization.
    Minimizes negative sum of radii + penalty for overlaps and boundary violations.
    """
    centers = vars[:2*n].reshape(n, 2)
    radii = vars[2*n:]
    
    # Calculate pairwise distances efficiently using broadcasting
    diffs = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2) + 1e-12)
    
    # Overlap penalty: squared violation of dist >= r_i + r_j
    r_sum = radii[:, None] + radii[None, :]
    overlap = np.maximum(0, r_sum - dists)
    np.fill_diagonal(overlap, 0)
    overlap_pen = np.sum(overlap**2)
    
    # Boundary penalty: squared violation of circle staying inside [0,1]^2
    x = centers[:, 0]
    y = centers[:, 1]
    bound_pen = np.sum(np.maximum(0, radii - x)**2 +
                       np.maximum(0, radii - (1 - x))**2 +
                       np.maximum(0, radii - y)**2 +
                       np.maximum(0, radii - (1 - y))**2)
                       
    # Objective: maximize sum(radii) => minimize -sum(radii) + weighted penalties
    return -np.sum(radii) + 1000 * (overlap_pen + bound_pen)

def run_packing():
    n = 26
    
    # 1. Initialization: Hexagonal grid layout
    # Rows configured to sum to 26: 6+5+6+5+4
    centers = np.zeros((n, 2))
    idx = 0
    y = 0.15
    row_h = np.sqrt(3)/2 * 0.14
    row_counts = [6, 5, 6, 5, 4]
    
    for rc in row_counts:
        x_start = (6 - rc) * 0.07
        for c in range(rc):
            if idx < n:
                centers[idx] = [x_start + c * 0.14, y]
                idx += 1
        y += row_h
    
    radii = np.full(n, 0.05)
    x0 = np.concatenate([centers.flatten(), radii])
    
    # Bounds: centers in [0,1], radii >= 1e-5
    bounds = [(0, 1)] * (2*n) + [(1e-5, None)] * n
    
    # 2. Optimization using L-BFGS-B with penalty method
    res = minimize(packing_objective, x0, args=(n,), method='L-BFGS-B', 
                   bounds=bounds, options={'maxiter': 3000, 'ftol': 1e-9})
    
    centers_opt = res.x[:2*n].reshape(n, 2)
    radii_opt = res.x[2*n:]
    
    # 3. Relaxation phase: strictly enforce non-overlap and boundary constraints
    for _ in range(2000):
        moved = False
        # Boundary clamping
        for i in range(n):
            for dim in range(2):
                if centers_opt[i, dim] < radii_opt[i]:
                    centers_opt[i, dim] = radii_opt[i]
                    moved = True
                if centers_opt[i, dim] > 1 - radii_opt[i]:
                    centers_opt[i, dim] = 1 - radii_opt[i]
                    moved = True
        # Overlap resolution
        for i in range(n):
            for j in range(i+1, n):
                d = np.sqrt(np.sum((centers_opt[i] - centers_opt[j])**2))
                min_d = radii_opt[i] + radii_opt[j]
                if d < min_d:
                    if d < 1e-12: 
                        d = 1e-12
                    overlap = min_d - d
                    dir = (centers_opt[j] - centers_opt[i]) / d
                    centers_opt[i] -= dir * overlap * 0.5
                    centers_opt[j] += dir * overlap * 0.5
                    moved = True
        if not moved:
            break
            
    # 4. Expansion phase: try to uniformly grow radii to maximize sum
    for _ in range(1000):
        can_grow = True
        test_r = radii_opt * 1.001
        # Check boundaries
        for i in range(n):
            if test_r[i] > centers_opt[i, 0] or test_r[i] > 1 - centers_opt[i, 0] or \
               test_r[i] > centers_opt[i, 1] or test_r[i] > 1 - centers_opt[i, 1]:
                can_grow = False
                break
        if not can_grow:
            break
        # Check overlaps
        for i in range(n):
            for j in range(i+1, n):
                d = np.sqrt(np.sum((centers_opt[i] - centers_opt[j])**2))
                if d < test_r[i] + test_r[j]:
                    can_grow = False
                    break
            if not can_grow:
                break
        if can_grow:
            radii_opt = test_r
        else:
            break
            
    total_sum = np.sum(radii_opt)
    return centers_opt, radii_opt, total_sum
