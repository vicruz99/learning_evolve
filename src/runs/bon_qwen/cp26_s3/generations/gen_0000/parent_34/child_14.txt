from scipy.optimize import minimize
import math

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    best_centers = None
    best_radii = None
    best_sum = -1.0
    
    # Optimization function
    def objective(x):
        # x contains [x1, y1, r1, x2, y2, r2, ...]
        # We want to maximize sum of radii, so we minimize negative sum
        radii = x[2::3]
        return -np.sum(radii)

    def constraints_factory(n):
        cons = []
        # Boundary constraints: r <= x <= 1-r  =>  r - x <= 0 AND r - (1-x) <= 0
        # Indices: 3i for x, 3i+1 for y, 3i+2 for r
        
        for i in range(n):
            idx_x = 3 * i
            idx_y = 3 * i + 1
            idx_r = 3 * i + 2
            
            # Left wall: x - r >= 0 => -(x - r) <= 0 => r - x <= 0
            cons.append({'type': 'ineq', 'fun': lambda v, i=i: v[idx_x] - v[idx_r]})
            # Right wall: 1 - x - r >= 0 => x + r <= 1 => -(x + r - 1) <= 0 => 1 - x - r >= 0
            # Standard scipy 'ineq' means fun(x) >= 0.
            # So we want 1 - x - r >= 0
            cons.append({'type': 'ineq', 'fun': lambda v, i=i: 1.0 - v[idx_x] - v[idx_r]})
            # Bottom wall: y - r >= 0
            cons.append({'type': 'ineq', 'fun': lambda v, i=i: v[idx_y] - v[idx_r]})
            # Top wall: 1 - y - r >= 0
            cons.append({'type': 'ineq', 'fun': lambda v, i=i: 1.0 - v[idx_y] - v[idx_r]})

        # dist^2 >= (r_i + r_j)^2

                idx_xj, idx_yj, idx_rj = 3*j, 3*j+1, 3*j+2
                
                def non_overlap(v, i=i, j=j):
                    xi, yi, ri = v[idx_xi], v[idx_yi], v[idx_ri]
                    xj, yj, rj = v[idx_xj], v[idx_yj], v[idx_rj]
                    dist_sq = (xi - xj)**2 + (yi - yj)**2
                    sum_r_sq = (ri + rj)**2
                    return dist_sq - sum_r_sq
                
                cons.append({'type': 'ineq', 'fun': non_overlap})
        
        return cons

    constraints = constraints_factory(n)
