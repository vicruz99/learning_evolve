import numpy as np
from scipy.optimize import minimize


def init_hexagonal(n, trial):
    """Initialize with hexagonal-like patterns."""
    centers = np.zeros((n, 2))
    
    row_configs = [
        [6, 5, 6, 5, 4],
        [5, 6, 5, 6, 4],
        [5, 5, 5, 5, 6],
        [6, 6, 5, 5, 4],
        [5, 6, 6, 5, 4],
        [6, 5, 5, 6, 4],
    ]
    row_counts = row_configs[trial % len(row_configs)]
    
    idx = 0
    dx = 0.17
    dy = 0.155
    
    for row, count in enumerate(row_counts):
        y = 0.07 + row * dy
        if row % 2 == 1:
            x_start = 0.07 + dx / 2
        else:
            x_start = 0.07
        
        for col in range(count):
            if idx < n:
                x = x_start + col * dx
                centers[idx] = [np.clip(x, 0.01, 0.99), np.clip(y, 0.01, 0.99)]
                idx += 1
    
    radii = np.full(n, 0.08)
    return centers, radii

def init_perturbed_hex(n, trial):
    """Initialize with perturbed hexagonal pattern."""
    rng = np.random.RandomState(42 + trial)
    centers = np.zeros((n, 2))
    
    row_counts = [6, 5, 6, 5, 4]
    idx = 0
    dx = 0.17
    dy = 0.155
    
    for row, count in enumerate(row_counts):
        y = 0.07 + row * dy
        if row % 2 == 1:
            x_start = 0.07 + dx / 2
        else:
            x_start = 0.07
        
        for col in range(count):
            if idx < n:
                x = x_start + col * dx + rng.uniform(-0.02, 0.02)
                y_pert = y + rng.uniform(-0.02, 0.02)
                centers[idx] = [np.clip(x, 0.01, 0.99), np.clip(y_pert, 0.01, 0.99)]
                idx += 1
    
    radii = np.full(n, 0.075)
    return centers, radii


def init_random_grid(n, trial):
    """Initialize with random positions."""
    rng = np.random.RandomState(100 + trial)
    centers = np.zeros((n, 2))
    
    # Create a structured but random-ish layout
    for i in range(n):
        x = 0.05 + rng.uniform(0, 0.9)
        y = 0.05 + rng.uniform(0, 0.9)
        centers[i] = [x, y]
    
    radii = np.full(n, 0.05)
    return centers, radii


def init_packing(n, trial):
    """Select initialization strategy based on trial number."""
    if trial < 6:
        return init_hexagonal(n, trial)
    elif trial < 26:
        return init_grid_plus_one(n, trial - 6)
    elif trial < 36:
        return init_perturbed_hex(n, trial - 26)
    else:
        return init_random_grid(n, trial - 36)


def compute_dist_sq(c1, c2):
    """Compute squared distance between two centers."""
    dx = c1[0] - c2[0]
    dy = c1[1] - c2[1]
    return dx * dx + dy * dy


def objective_and_penalty(params, n, pw):
    """Compute objective (negative sum of radii) plus constraint penalties."""
    centers = params[:2 * n].reshape((n, 2))
    radii = params[2 * n:]
    
    # Objective: minimize negative sum of radii
    obj = -np.sum(radii)
    
    # Boundary penalties
    penalty = 0.0
    for i in range(n):
        x = centers[i, 0]
        y = centers[i, 1]
        r = radii[i]
        
        # Left
        v = r - x
        if v > 0:
            penalty += pw * v * v
        # Right
        v = r - (1.0 - x)
        if v > 0:
            penalty += pw * v * v
        # Bottom
        v = r - y
        if v > 0:
            penalty += pw * v * v
        # Top
        v = r - (1.0 - y)
        if v > 0:
            penalty += pw * v * v
    
    # Overlap penalties
    for i in range(n):
        for j in range(i + 1, n):
            d2 = compute_dist_sq(centers[i], centers[j])
            d = np.sqrt(max(d2, 0))
            overlap = radii[i] + radii[j] - d
            if overlap > 0:
                penalty += pw * overlap * overlap
    
    return obj + penalty


def optimize_packing(centers, radii, n, penalty_weights=None, max_iters=15000):
    """Optimize circle positions and radii using L-BFGS-B."""
    if penalty_weights is None:
        penalty_weights = [2000, 5000, 10000, 20000]
    
    params = np.concatenate([centers.ravel(), radii])
    
    bounds = []
    for i in range(n):
        bounds.append((0.0, 1.0))
        bounds.append((0.0, 1.0))
        bounds.append((0.0, 0.5))
    
    for pw in penalty_weights:
        result = minimize(
            objective_and_penalty,
            params,
            args=(n, pw),
            bounds=bounds,
            method='L-BFGS-B',
            options={'maxiter': max_iters, 'ftol': 1e-15, 'gtol': 1e-12}
        )
        params = result.x
    
    centers_opt = params[:2 * n].reshape((n, 2))
    radii_opt = params[2 * n:]
    
    centers_opt = np.clip(centers_opt, 0.0, 1.0)
    radii_opt = np.maximum(radii_opt, 0.0)
    
    return centers_opt, radii_opt


def inflate_radii(centers, radii, n, max_passes=200):
    """Increase each radius to its maximum feasible value."""
    radii = radii.copy()
    
    for _ in range(max_passes):
        improved = False
        
        for i in range(n):
            max_r = min(
                centers[i, 0],
                1.0 - centers[i, 0],
                centers[i, 1],
                1.0 - centers[i, 1]
            )
            
            for j in range(n):
                if i != j:
                    d2 = compute_dist_sq(centers[i], centers[j])
                    d = np.sqrt(max(d2, 0))
                    max_r = min(max_r, d - radii[j])
            
            max_r = max(max_r, 0.0)
            
            if max_r > radii[i] + 1e-12:
                radii[i] = max_r
                improved = True
        
        if not improved:
            break
    
    return radii


def safe_adjust_radii(centers, radii, n):
    """Ensure all constraints are satisfied by reducing radii if needed."""
    radii = radii.copy()
    
    for i in range(n):
        x = centers[i, 0]
        y = centers[i, 1]
        r = radii[i]
        
        r = min(r, x, 1.0 - x, y, 1.0 - y)
        
        for j in range(n):
            if i != j:
                d2 = compute_dist_sq(centers[i], centers[j])
                d = np.sqrt(max(d2, 0))
                r = min(r, d - radii[j])
        
        radii[i] = max(r, 0.0)
    
    return radii


def check_validity(centers, radii, n):
    """Check if the packing satisfies all constraints."""
    for i in range(n):
        x = centers[i, 0]
        y = centers[i, 1]
        r = radii[i]
        
        if r < -1e-12:
            return False
        if x - r < -1e-10 or x + r > 1.0 + 1e-10:
            return False
        if y - r < -1e-10 or y + r > 1.0 + 1e-10:
            return False
    
    for i in range(n):
        for j in range(i + 1, n):
            d2 = compute_dist_sq(centers[i], centers[j])
            d = np.sqrt(max(d2, 0))
            if d < radii[i] + radii[j] - 1e-10:
                return False
    
    return True


def run_packing():
    """Main function to pack 26 circles in unit square maximizing sum of radii."""
    n = 26
    
    best_centers = np.zeros((n, 2))
    best_radii = np.zeros(n)
    best_sum = 0.0
    
    num_trials = 40
    
    for trial in range(num_trials):
        centers, radii = init_packing(n, trial)
        
        centers, radii = optimize_packing(centers, radii, n)
        
        # Inflate radii to maximum feasible values
        radii = inflate_radii(centers, radii, n)
        
        # Safety adjustment
        radii = safe_adjust_radii(centers, radii, n)
        
        # Validate
        if check_validity(centers, radii, n):
            current_sum = np.sum(radii)
            if current_sum > best_sum:
                best_sum = current_sum
                best_centers = centers.copy()
                best_radii = radii.copy()
    
    # Final validation pass
    if not check_validity(best_centers, best_radii, n):
        best_radii = safe_adjust_radii(best_centers, best_radii, n)
        best_sum = np.sum(best_radii)
    
    return best_centers, best_radii, best_sum
