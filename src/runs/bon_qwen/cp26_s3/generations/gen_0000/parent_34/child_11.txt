import numpy as np
from scipy.optimize import minimize, dual_annealing

def check_validity(centers, radii):
    """
    Helper to quickly check validity without printing
    """
    n = centers.shape[0]
    # Check boundaries
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-7 or x + r > 1 + 1e-7 or y - r < -1e-7 or y + r > 1 + 1e-7:
            return False
    
    # Check overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-7:
                return False
    return True

def generate_hex_grid(n, r):
    """
    Generate a hexagonal grid of centers for n circles with estimated radius r.
    """
    centers = []
    # Hexagonal packing parameters
    # Vertical spacing
    dy = np.sqrt(3) * r
    # Horizontal spacing
    dx = 2 * r
    
    y = r
    row = 0
    count = 0
    while count < n:
        x_start = r + (row % 2) * r
        x = x_start
        while x <= 1 - r:
            centers.append([x, y])
            count += 1
            if count >= n:
                break
            x += dx
        row += 1
        y += dy

def solve_packing():
    n = 26
    # Initial radius estimate
    r_init = 0.10
    
    # Generate initial centers
    centers_init = generate_hex_grid(n, r_init)
    
    # If centers are out of bounds, clamp them
    centers_init = np.clip(centers_init, r_init, 1 - r_init)
    
    # Initial parameters: [x1, y1, ..., x26, y26, r]
    x0 = np.concatenate([centers_init.flatten(), [r_init]])
    
    # Bounds: x, y in [0, 1], r in [0, 0.5]
    # Actually r should be at least 0.
    bounds = []
    for i in range(n):
        bounds.extend([(0, 1), (0, 1)])
    bounds.append((0, 0.5))
    
    # Constraints
    # 1. Boundary constraints: x >= r, x <= 1-r => x - r >= 0, 1 - r - x >= 0
    # 2. Non-overlap: dist >= 2r => dist^2 - 4r^2 >= 0 (non-linear inequality)
    
    def obj(params):
        # We want to maximize sum of radii. Assuming equal radii r = params[-1].
        # So maximize n * r => minimize -n * r
        r = params[-1]
        return -n * r

    def constraint_boundary(params):
        # Returns array of constraint values, must be >= 0
        # x_i - r >= 0
        # 1 - r - x_i >= 0
        # y_i - r >= 0
        # 1 - r - y_i >= 0
        
        centers = params[:-1].reshape(-1, 2)
        r = params[-1]
        
        c_vals = []
        for i in range(n):
            x, y = centers[i]
            c_vals.append(x - r)
            c_vals.append(1 - r - x)
            c_vals.append(y - r)
            c_vals.append(1 - r - y)
        return np.array(c_vals)

    def constraint_overlap(params):
        # dist_ij >= 2r
        # (x_i - x_j)^2 + (y_i - y_j)^2 - 4r^2 >= 0
        centers = params[:-1].reshape(-1, 2)
        r = params[-1]
        
        c_vals = []
        for i in range(n):
            for j in range(i + 1, n):
                dist_sq = np.sum((centers[i] - centers[j])**2)
                c_vals.append(dist_sq - 4 * r**2)
        return np.array(c_vals)

    # SLSQP might struggle with many constraints. 
    # Let's try to solve it.
    
    cons = [
        {'type': 'ineq', 'fun': constraint_boundary},
        {'type': 'ineq', 'fun': constraint_overlap}
    ]
    
    best_params = None

            x0_trial[2*n] = x0[-1]

            x0_trial = x0.copy()

                    best_params = res.x
        except Exception as e:
            print(f"Optimization failed in trial {trial}: {e}")
            continue

    if best_params is not None:
        centers = best_params[:-1].reshape(-1, 2)
        r = best_params[-1]
        radii = np.full(n, r)

            centers = np.clip(centers, r, 1-r)
        
        sum_radii = np.sum(radii)
        return centers, radii, sum_radii
    else:
        # Fallback
        return np.zeros((n, 2)), np.zeros(n), 0.0

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    centers, radii, sum_radii = solve_packing()
