import numpy as np
from scipy.optimize import linprog

def calculate_max_radii(centers):
    """
    Solves an LP to maximize sum of radii for a fixed set of centers.
    """
    n = len(centers)
    # Maximize sum(r_i) <=> Minimize -sum(r_i)
    c_obj = np.ones(n) * -1

    # A_ub * x <= b_ub
    A_ub = []
    b_ub = []

    # Pairwise non-overlap constraints: r_i + r_j <= distance(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            row = [0.0] * n
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dist)

    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        x, y = centers[i]
        max_r = min(x, 1 - x, y, 1 - y)
        if max_r < 0:
            return None, 0.0 # Center outside square
        row = [0.0] * n
        row[i] = 1.0
        A_ub.append(row)
        b_ub.append(max_r)

    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    # Bounds for r_i >= 0
    bounds = [(0, None) for _ in range(n)]

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return -res.x, -res.fun
        return None, 0.0
    except Exception:
        return None, 0.0

def run_packing():
    n = 26
    
    # 1. Initialize with a dense 5x5 grid structure
    centers = []
    # 25 circles in 5x5 grid
    for i in range(5):
        for j in range(5):
            centers.append([0.1 + i * 0.2, 0.1 + j * 0.2])
    # 26th circle in the middle of a gap
    centers.append([0.4, 0.4])
    centers = np.array(centers)
    
    # Initial radius solve
    _, current_sum = calculate_max_radii(centers)
    
    best_centers = centers.copy()
    
    # 2. Iterative perturbation to improve centers
    # Using a coarse-to-fine approach
    for step in range(200):
        # Perturbation magnitude decreases over time
        magnitude = 0.005 * (0.95 ** (step // 20))
        
        best_move = None
        best_move_sum = current_sum
        
        # Try small shifts for a few random circles to speed up
        for _ in range(50):
            idx = np.random.randint(n)
            dx = np.random.uniform(-magnitude, magnitude)
            dy = np.random.uniform(-magnitude, magnitude)
            
            trial_centers = centers.copy()
            trial_centers[idx] += [dx, dy]
            
            # Keep centers within a reasonable boundary range for the LP
            trial_centers[idx] = np.clip(trial_centers[idx], 0.01, 0.99)
            
            _, trial_sum = calculate_max_radii(trial_centers)
            
            if trial_sum > best_move_sum:
                best_move = (idx, dx, dy)
                best_move_sum = trial_sum
        
        if best_move:
            idx, dx, dy = best_move
            centers[idx] += [dx, dy]
            centers[idx] = np.clip(centers[idx], 0.01, 0.99)
            _, current_sum = calculate_max_radii(centers)
            best_centers = centers.copy()
        else:
            # If no improvement, reduce step size or break if very small
            if magnitude < 1e-5:
                break
                
    # 3. Final solve for the best centers found
    final_radii, final_sum = calculate_max_radii(best_centers)
    if final_radii is None:
        # Fallback to a guaranteed valid grid if LP fails
        final_radii = np.ones(26) * 0.1
        final_radii[-1] = 0.04
        final_sum = np.sum(final_radii)
        
    return best_centers, final_radii, final_sum
