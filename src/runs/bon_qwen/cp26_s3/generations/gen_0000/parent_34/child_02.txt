import numpy as np
from scipy.optimize import minimize

N = 26

def objective(params):
    return -np.sum(params[2 * N :])


def objective_grad(params):
    g = np.zeros(len(params))
    g[2 * N :] = -1.0
    return g


def constraint_xmin(params, i):
    return params[2 * i] - params[2 * N + i]


def constraint_xmax(params, i):
    return 1.0 - params[2 * i] - params[2 * N + i]


def constraint_ymin(params, i):
    return params[2 * i + 1] - params[2 * N + i]


def constraint_ymax(params, i):
    return 1.0 - params[2 * i + 1] - params[2 * N + i]


def constraint_overlap(params, i, j):
    dx = params[2 * i] - params[2 * j]
    dy = params[2 * i + 1] - params[2 * j + 1]
    return np.sqrt(dx * dx + dy * dy) - params[2 * N + i] - params[2 * N + j]


def build_constraints():
    constraints = []
    for i in range(N):
        constraints.append({"type": "ineq", "fun": constraint_xmin, "args": (i,)})
        constraints.append({"type": "ineq", "fun": constraint_xmax, "args": (i,)})
        constraints.append({"type": "ineq", "fun": constraint_ymin, "args": (i,)})
        constraints.append({"type": "ineq", "fun": constraint_ymax, "args": (i,)})
    for i in range(N):
        for j in range(i + 1, N):
            constraints.append(
                {"type": "ineq", "fun": constraint_overlap, "args": (i, j)}
            )
    return constraints


def make_bounds():
    b = []
    for _ in range(2 * N):
        b.append((0.0, 1.0))
    for _ in range(N):
        b.append((0.0, 0.5))
    return b


def generate_hex_init(pattern):
    centers = np.zeros((N, 2))
    idx = 0
    nr = len(pattern)
    for r in range(nr):
        nc = pattern[r]
        y = (r + 1) / (nr + 1)
        for cc in range(nc):
            if r % 2 == 0:
                x = (cc + 1) / (nc + 1)
            else:
                x = (cc + 0.5) / nc
            centers[idx] = [x, y]
            idx += 1
    return centers


def run_optimization(x0, constraints, bounds, max_iter=2000):
    res = minimize(
        objective,
        x0,
        method="SLSQP",
        jac=objective_grad,
        bounds=bounds,
        constraints=constraints,
        options={"maxiter": max_iter, "ftol": 1e-15, "disp": False},
    )
    return res


def fix_solution(centers, radii):
    n = len(radii)
    radii = radii.copy()

    # Ensure boundary constraints
    for i in range(n):
        max_r = min(
            centers[i, 0],
            1.0 - centers[i, 0],
            centers[i, 1],
            1.0 - centers[i, 1],
        )
        radii[i] = min(radii[i], max(max_r, 0.0))

    # Fix overlaps by shrinking
    for _ in range(200):
        max_violation = 0.0
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                required = radii[i] + radii[j]
                if dist < required - 1e-14:
                    violation = required - dist
                    max_violation = max(max_violation, violation)
                    avg = (radii[i] + radii[j]) / 2.0
                    new_sum = dist
                    radii[i] = new_sum * radii[i] / (2.0 * avg) if avg > 1e-15 else 0
                    radii[j] = new_sum * radii[j] / (2.0 * avg) if avg > 1e-15 else 0
                    radii[i] = max(radii[i], 0.0)
                    radii[j] = max(radii[j], 0.0)
        if max_violation < 1e-12:
            break

    return centers, radii


def run_packing():
    constraints = build_constraints()
    bounds = make_bounds()

    patterns = [
        [5, 6, 5, 6, 4],
        [6, 5, 6, 5, 4],
        [5, 5, 5, 5, 6],
        [4, 6, 6, 6, 4],
        [6, 5, 5, 5, 5],
        [5, 5, 6, 5, 5],
        [4, 5, 6, 6, 5],
        [5, 4, 6, 6, 5],
        [5, 6, 4, 6, 5],
        [6, 4, 6, 5, 5],
    ]

    best_centers = None
    best_radii = None
    best_sum = 0.0

    for pat in patterns:
        if sum(pat) != N:
            continue

        centers = generate_hex_init(pat)
        radii = np.ones(N) * 0.095
        x0_base = np.concatenate([centers.flatten(), radii])

        for trial in range(8):
            pert = np.random.randn(len(x0_base)) * 0.008
            x0_pert = x0_base + pert
            x0_pert[: 2 * N] = np.clip(x0_pert[: 2 * N], 0.05, 0.95)
            x0_pert[2 * N :] = np.maximum(x0_pert[2 * N :], 0.05)

            res = run_optimization(x0_pert, constraints, bounds, max_iter=3000)

            if res.success or (not np.isnan(res.fun)):
                c = res.x[: 2 * N].reshape(N, 2)
                r = res.x[2 * N :]
                r = np.maximum(r, 0.0)

                s = np.sum(r)
                if s > best_sum:
                    best_sum = s
                    best_centers = c.copy()
                    best_radii = r.copy()

    if best_centers is None:
        best_centers = generate_hex_init([5, 6, 5, 6, 4])
        best_radii = np.ones(N) * 0.09

    best_centers, best_radii = fix_solution(best_centers, best_radii)
    final_sum = np.sum(best_radii)

    return best_centers, best_radii, final_sum
