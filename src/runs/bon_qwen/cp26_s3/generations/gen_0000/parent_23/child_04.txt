import numpy as np
from scipy.optimize import minimize

def compute_loss_and_grad(vars, n, penalty_weight, i_idx, j_idx):
    # Objective: maximize sum of radii -> minimize -sum(r)
    loss = -np.sum(vars[2::3])
    grad = np.zeros_like(vars)
    grad[2::3] -= 1.0
    
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    # Pairwise overlap penalties
    dx = x[i_idx] - x[j_idx]
    dy = y[i_idx] - y[j_idx]
    dr = r[i_idx] + r[j_idx]
    
    dist = np.sqrt(dx**2 + dy**2 + 1e-16)
    overlap = dr - dist
    penalty_pair = np.maximum(0.0, overlap)**2
    loss += penalty_weight * np.sum(penalty_pair)
    
    active = overlap > 0.0
    if np.any(active):
        active_overlap = overlap[active]
        grad_term = 2.0 * penalty_weight * active_overlap
        inv_dist = 1.0 / dist[active]
        
        # Gradients for x
        grad[3*i_idx[active]] += grad_term * dx[active] * inv_dist
        grad[3*j_idx[active]] -= grad_term * dx[active] * inv_dist
        
        # Gradients for y
        grad[3*i_idx[active] + 1] += grad_term * dy[active] * inv_dist
        grad[3*j_idx[active] + 1] -= grad_term * dy[active] * inv_dist
        
        # Gradients for r
        grad[3*i_idx[active] + 2] += grad_term
        grad[3*j_idx[active] + 2] += grad_term
        
    # Boundary penalties
    # Left: x - r >= 0
    b1 = np.maximum(0.0, r - x)
    loss += penalty_weight * np.sum(b1**2)
    active_b1 = b1 > 0.0
    if np.any(active_b1):
        gt = 2.0 * penalty_weight * b1[active_b1]
        grad[3*active_b1 + 2] += gt
        grad[3*active_b1] -= gt
        
    # Right: 1 - x - r >= 0
    b2 = np.maximum(0.0, x + r - 1.0)
    loss += penalty_weight * np.sum(b2**2)
    active_b2 = b2 > 0.0
    if np.any(active_b2):
        gt = 2.0 * penalty_weight * b2[active_b2]
        grad[3*active_b2] += gt
        grad[3*active_b2 + 2] += gt
        
    # Bottom: y - r >= 0
    b3 = np.maximum(0.0, r - y)
    loss += penalty_weight * np.sum(b3**2)
    active_b3 = b3 > 0.0
    if np.any(active_b3):
        gt = 2.0 * penalty_weight * b3[active_b3]
        grad[3*active_b3 + 2] += gt
        grad[3*active_b3 + 1] -= gt
        
    # Top: 1 - y - r >= 0
    b4 = np.maximum(0.0, y + r - 1.0)
    loss += penalty_weight * np.sum(b4**2)
    active_b4 = b4 > 0.0

        gt = 2.0 * penalty_weight * b4[active_b4]
        grad[3*active_b4 + 1] += gt
        grad[3*active_b4 + 2] += gt
        
    return loss, grad

def run_packing():
    n = 26
    np.random.seed(42)
    
    # 1. Initialize on a hexagonal lattice
    s = 0.2
    rows = [5, 5, 5, 5, 6]
    xs, ys = [], []
    for row_idx, count in enumerate(rows):
        y = row_idx * s * np.sqrt(3) / 2
        for col in range(count):
            x = col * s + (0.5 * s if row_idx % 2 == 1 else 0.0)
            xs.append(x)
            ys.append(y)
            
    xs = np.array(xs)
    ys = np.array(ys)
    
    # Center and scale to fit in [0, 1]
    margin = 0.05
    xs = (xs - xs.min()) / (xs.max() - xs.min() + 1e-8) * (1.0 - 2*margin) + margin
    ys = (ys - ys.min()) / (ys.max() - ys.min() + 1e-8) * (1.0 - 2*margin) + margin
    
    # Initial radii with slight randomness to break symmetry
    r0 = 0.05 + np.random.uniform(0, 0.005, n)
    vars0 = np.concatenate([xs, ys, r0])
    
    # Precompute pair indices for efficiency
    i_idx, j_idx = np.triu_indices(n, k=1)
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * n
    
    # 2. Optimize with increasing penalty weights
    current_vars = vars0.copy()
    for w in [1000.0, 5000.0, 20000.0]:
        res = minimize(
            compute_loss_and_grad,
            current_vars,
            args=(n, w, i_idx, j_idx),
            method='L-BFGS-B',
            jac=True,
            bounds=bounds,
            options={'maxiter': 2000, 'ftol': 1e-12}
        )
        current_vars = res.x
        
    centers = np.column_stack((current_vars[0::3], current_vars[1::3]))
    radii = current_vars[2::3]
    
    # 3. Ensure strict validity by shrinking radii if necessary
    max_iter = 100
    for _ in range(max_iter):
        valid = True
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-9 or x + r > 1.0 + 1e-9 or y - r < -1e-9 or y + r > 1.0 + 1e-9:
                valid = False
                break
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii[i] + radii[j] - 1e-9:
                    valid = False
                    break
            if not valid:
                break
        if valid:
            break
        radii *= 0.995
        
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
