import numpy as np
from scipy.optimize import minimize

def loss(vars, n, C):
    """
    Objective: -sum(radii) + penalty for boundary/overlap violations.
    vars: flattened array of [x0, y0, r0, x1, y1, r1, ...]
    """
    pts = vars.reshape(-1, 3)
    x = pts[:, 0]
    y = pts[:, 1]
    r = pts[:, 2]

    # Primary objective: maximize sum of radii
    obj = -np.sum(r)

    # Boundary penalty: r <= x <= 1-r  and  r <= y <= 1-r
    pen_b = np.sum(
        np.maximum(0, r - x)**2 +
        np.maximum(0, r + x - 1)**2 +
        np.maximum(0, r - y)**2 +
        np.maximum(0, r + y - 1)**2
    )

    # Overlap penalty: dist(i,j) >= r_i + r_j
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dr = r[:, None] + r[None, :]
    dist = np.sqrt(dx**2 + dy**2)
    overlap = dr - dist
    np.fill_diagonal(overlap, 0)  # Ignore self-interaction
    pen_o = np.sum(np.maximum(0, overlap)**2)

    # Total loss
    return obj + C * pen_b + 0.5 * C * pen_o

def get_hexagonal_init(n):
    """Generate a structured hexagonal-like initial guess."""
    pts = []
    # Pattern designed to fill the square densely: 5, 6, 5, 6, 4
    row_counts = [5, 6, 5, 6, 4]
    y_pos = np.linspace(0.12, 0.88, len(row_counts))
    
    for i, count in enumerate(row_counts):
        span = 0.8
        step = span / (count + 1)
        xs = [0.1 + step * k for k in range(1, count + 1)]
        
        # Stagger odd rows for hexagonal packing effect
        if i % 2 == 1:
            xs = [x + step / 2 for x in xs]
            
        for xp in xs:
            pts.append([xp, y_pos[i], 0.09]) # Initial radius ~0.09
    return np.array(pts).flatten()

def run_packing():
    n = 26
    best_vars = None
    best_loss = np.inf
    
    np.random.seed(42)
    inits = [get_hexagonal_init(n)]
    
    # Add random initializations for robustness
    for _ in range(3):
        pts = np.random.uniform(0.2, 0.8, (n, 2))
        r = np.ones(n) * 0.08
        inits.append(np.hstack([pts, r[:, None]]).flatten())

    # Bounds: x,y in [0,1], r in [0, 0.5]
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * n
    C = 5000.0

    # Run optimization from each initial guess
    for x0 in inits:
        res = minimize(loss, x0, args=(n, C), method='L-BFGS-B', 
                       bounds=bounds, options={'maxiter': 2000, 'ftol': 1e-12})
        if res.fun < best_loss:
            best_loss = res.fun
            best_vars = res.x

    # Extract results
    pts = best_vars.reshape(-1, 3)
    centers = pts[:, :2].copy()
    radii = pts[:, 2].copy()

    # Post-processing: strictly enforce validity
    # 1. Fix boundary violations by clipping radii
    for i in range(n):
        r = radii[i]
        x, y = centers[i]
        if x < r: r = x
        if x > 1 - r: r = 1 - x
        if y < r: r = y
        if y > 1 - r: r = 1 - y
        radii[i] = r

    # 2. Fix overlap violations by uniform radius reduction
    max_ov = 0.0
    for i in range(n):
        for j in range(i + 1, n):
            d = np.sqrt((centers[i, 0] - centers[j, 0])**2 + (centers[i, 1] - centers[j, 1])**2)
            ov = radii[i] + radii[j] - d
            if ov > max_ov:
                max_ov = ov

    if max_ov > 1e-12:
        reduction = max_ov / 2.0
        radii -= reduction
        radii = np.maximum(radii, 0.0)

    return centers, radii, np.sum(radii)
