import numpy as np
from scipy.optimize import minimize

def run_packing():
    """
    Optimizes the packing of 26 circles in a unit square to maximize sum of radii.
    """
    N = 26
    
    # --- Helper Functions ---
    def get_objective_and_constraints(p, N):
        """
        Computes objective (-sum of radii) and constraints (violations) for SLSQP.
        SLSQP minimizes objective. Constraints must be >= 0.
        """
        x = p[0::3]
        y = p[1::3]
        r = p[2::3]
        
        # Objective: maximize sum(r) => minimize -sum(r)
        obj = -np.sum(r)
        
        # 1. Wall Constraints: center must be >= r away from walls
        # x >= r => x - r >= 0
        # 1-x >= r => 1 - x - r >= 0
        # y >= r => y - r >= 0
        # 1-y >= r => 1 - y - r >= 0
        wall_cons = np.concatenate([
            x - r,
            1 - x - r,
            y - r,
            1 - y - r
        ])
        
        # 2. Pairwise Non-overlap Constraints
        # dist(i,j) >= r_i + r_j => dist(i,j) - (r_i + r_j) >= 0
        # We compute for all pairs i < j to reduce size
        # Coordinates as (N, 1) matrices for broadcasting
        cx = x[:, np.newaxis]
        cy = y[:, np.newaxis]
        rx = r[:, np.newaxis]
        
        dx = cx - cx.T
        dy = cy - cy.T
        dists = np.sqrt(dx**2 + dy**2)
        
        r_sum = rx + rx.T
        
        # Get upper triangle indices (i < j)
        i_upper, j_upper = np.triu_indices(N, k=1)
        pair_cons = dists[i_upper, j_upper] - r_sum[i_upper, j_upper]
        
        # Combine all constraints
        cons = np.concatenate([wall_cons, pair_cons])
        
        return obj, cons

    def get_grad(p, N):
        """
        Computes gradient of objective and constraints for faster convergence.
        Returns (grad_obj, jacobian_cons).
        """
        # Numerical gradient might be safer and robust enough for 78 vars
        # But analytical is faster. Let's stick to numerical if SLSQP approximates,
        # but providing it is better. However, Jacobian is large.
        # SLSQP can approximate it. Let's return None for simplicity and robustness.
        return None

    # --- Initialization Strategy ---
    best_result = None
    best_sum_r = -np.inf

    # Try multiple restarts with perturbations
    num_trials = 5
    
    for trial in range(num_trials):
        # 1. Generate Hexagonal Lattice
        centers = np.zeros((N, 2))
        idx = 0
        
        # We'll pack rows. For 26 circles, maybe 5 rows?
        # Row lengths: 6, 5, 6, 5, 4 (sum 26)
        row_counts = [6, 5, 6, 5, 4]
        
        # Estimate spacing for unit square
        # Vertical spacing for hex lattice is sqrt(3)/2 * spacing
        # Horizontal spacing is spacing
        # Let's assume diameter ~ 0.2 => spacing ~ 0.2
        # But we will optimize, so initial scale doesn't strictly matter as long as valid.
        # Let's target r ~ 0.1.
        spacing = 0.25 
        
        y_pos = 0.15 # Start a bit inside
        row_height = spacing * np.sqrt(3) / 2
        
        for r_idx, count in enumerate(row_counts):
            # Offset every other row by half spacing
            x_offset = 0.15
            if r_idx % 2 == 1:
                x_offset += spacing / 2
            
            for c_idx in range(count):
                x = x_offset + c_idx * spacing
                y = y_pos + r_idx * row_height
                centers[idx] = [x, y]
                idx += 1
        
        # Initial radii: small enough to be valid
        radii = np.full(N, 0.05)
        
        # 2. Add random perturbation to break symmetry and explore space
        # Perturbation scale small relative to spacing
        noise_scale = 0.05
        centers += np.random.normal(0, noise_scale, centers.shape)
        
        # Clip to valid range [0.1, 0.9] to keep away from walls initially
        centers = np.clip(centers, 0.1, 0.9)
        
        # 3. Flatten to optimization vector
        # Order: x1, y1, r1, x2, y2, r2, ...
        p0 = np.zeros(3 * N)
        p0[0::3] = centers[:, 0]
        p0[1::3] = centers[:, 1]
        p0[2::3] = radii
        
        # 4. Bounds
        # x, y in [0, 1]
        # r in [0, 0.5] (radius cannot exceed 0.5)
        bounds = [
            (0.0, 1.0), (0.0, 1.0), (0.0, 0.5) for _ in range(N)
        ]
        
        # 5. Optimization
        try:
            res = minimize(
                lambda p: get_objective_and_constraints(p, N)[0],
                p0,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'ineq', 'fun': lambda p: get_objective_and_constraints(p, N)[1]},
                options={'maxiter': 2000, 'ftol': 1e-9, 'disp': False}
            )
            
            if res.success:
                sum_r = -res.fun
                if sum_r > best_sum_r:
                    best_sum_r = sum_r
                    best_result = res.x
        except Exception as e:
            print(f"Optimization failed in trial {trial}: {e}")
            continue

    # --- Finalize and Validate ---
    if best_result is None:
        # Fallback to initial hexagonal if optimization failed everywhere
        # (Should not happen)
        p_final = np.zeros(3 * N)
        p_final[2::3] = 0.05
    else:
        p_final = best_result

    # Extract centers and radii
    centers = np.column_stack((p_final[0::3], p_final[1::3]))
    radii = p_final[2::3]
    
    # Ensure radii are non-negative (numerical stability)
    radii = np.maximum(radii, 0.0)
    
    # Ensure centers are within bounds (numerical stability)
    centers = np.clip(centers, 0.0, 1.0)
    
    # Compute sum of radii
    total_sum = np.sum(radii)
    
    return centers, radii, total_sum
