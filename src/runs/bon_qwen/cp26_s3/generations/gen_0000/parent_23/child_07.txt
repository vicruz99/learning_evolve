import numpy as np
from scipy.optimize import minimize

def compute_objective(vars, n):
    """Objective: minimize negative sum of radii (equivalent to maximizing sum)."""
    return -np.sum(vars[2::3])

def compute_constraints(vars, n):
    """
    Returns an array of constraint values. All values must be >= 0 for a valid packing.
    Boundary constraints: x - r >= 0, 1 - (x + r) >= 0, same for y.
    Pairwise constraints: dist^2 - (r_i + r_j)^2 >= 0.
    """
    X = vars[0::3]
    Y = vars[1::3]
    R = vars[2::3]
    
    # Boundary constraints
    con = np.concatenate([
        X - R,
        1.0 - (X + R),
        Y - R,
        1.0 - (Y + R)
    ])
    
    # Pairwise distance constraints using broadcasting
    dx = X[:, None] - X[None, :]
    dy = Y[:, None] - Y[None, :]
    dr = R[:, None] + R[None, :]
    dist_sq = dx**2 + dy**2
    
    # Extract only the upper triangular part (i < j) to avoid duplicates and self-comparison
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    pair_cons = dist_sq[mask] - dr[mask]**2
    
    return np.concatenate([con, pair_cons])

def run_packing():
    n = 26
    
    # 1. Initial Guess: Dense grid + center circle
    centers = np.zeros((n, 2))
    idx = 0
    for row in range(5):
        y = 0.125 + row * 0.25
        for col in range(5):
            if idx >= 25:
                break
            x = 0.125 + col * 0.25
            centers[idx] = [x, y]
            idx += 1
    centers[25] = [0.5, 0.5]  # 26th circle in the center
    
    radii = np.full(n, 0.12)
    
    # Flatten variables: [x1, y1, r1, x2, y2, r2, ...]
    x0 = np.zeros(3 * n)
    for i in range(n):
        x0[3*i] = centers[i, 0]
        x0[3*i+1] = centers[i, 1]
        x0[3*i+2] = radii[i]
        
    # Bounds: centers in [0,1], radii in [epsilon, 0.5]
    bounds = []
    for i in range(n):
        bounds.append((0.0, 1.0))
        bounds.append((0.0, 1.0))
        bounds.append((1e-6, 0.5))
        
    cons = {'type': 'ineq', 'fun': compute_constraints, 'args': (n,)}
    
    # 2. Optimization
    result = minimize(
        compute_objective, 
        x0, 
        args=(n,), 
        method='SLSQP', 
        bounds=bounds, 
        constraints=cons, 
        options={'maxiter': 10000, 'ftol': 1e-13}
    )
                      
    x_opt = result.x
    centers_opt = np.zeros((n, 2))
    radii_opt = np.zeros(n)
    for i in range(n):
        centers_opt[i, 0] = x_opt[3*i]
        centers_opt[i, 1] = x_opt[3*i+1]
        radii_opt[i] = x_opt[3*i+2]
        
    # 3. Post-processing: Ensure strict validity within numerical tolerance
    violations = -np.min(compute_constraints(x_opt, n))
    if violations > 1e-9:
        # Scale radii down slightly to resolve any microscopic overlaps
        scale = 1.0 - violations * 2.0
        if scale < 0.90:
            scale = 0.90
        radii_opt *= scale
        
    sum_radii = float(np.sum(radii_opt))
    return centers_opt, radii_opt, sum_radii
