import numpy as np
from scipy.optimize import linprog, minimize

def solve_radii_lp(centers):
    """Solve LP to find optimal radii given centers."""
    n = centers.shape[0]
    
    # Objective: maximize sum(r_i) => minimize -sum(r_i)
    c_obj = -np.ones(n)
    
    # Build constraint matrix A_ub @ r <= b_ub
    n_boundary = 4 * n
    n_pair = n * (n - 1) // 2
    n_constr = n_boundary + n_pair
    
    A_ub = np.zeros((n_constr, n))
    b_ub = np.zeros(n_constr)
    
    k = 0
    for i in range(n):
        x, y = centers[i]
        # r_i <= x
        A_ub[k, i] = 1
        b_ub[k] = x
        k += 1
        # r_i <= 1 - x
        A_ub[k, i] = 1
        b_ub[k] = 1.0 - x
        k += 1
        # r_i <= y
        A_ub[k, i] = 1
        b_ub[k] = y
        k += 1
        # r_i <= 1 - y
        A_ub[k, i] = 1
        b_ub[k] = 1.0 - y
        k += 1
    
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            A_ub[k, i] = 1
            A_ub[k, j] = 1
            b_ub[k] = dist
            k += 1
    
    bounds = [(0, None)] * n
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        return result.x
    return np.zeros(n)


def get_sum_radii(centers):
    """Get sum of optimal radii for given centers."""
    radii = solve_radii_lp(centers)
    return np.sum(radii)


def optimal_position_for_circle(centers, i, radii_others):
    """Find position for circle i that maximizes its radius given fixed others."""
    n = centers.shape[0]
    
    def neg_radius(pos):
        x, y = pos
        # Boundary constraints
        r = min(x, 1.0 - x, y, 1.0 - y)
        # Circle constraints
        for j in range(n):
            if i != j:
                dx = x - centers[j, 0]
                dy = y - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                r = min(r, dist - radii_others[j])
        return -max(0.0, r)
    
    # Try multiple starting points
    best_val = float('inf')
    best_pos = centers[i].copy()
    
    # Start from current position
    result = minimize(neg_radius, centers[i], bounds=[(0.001, 0.999), (0.001, 0.999)], 
                      method='Nelder-Mead', options={'maxiter': 1000, 'xatol': 1e-8, 'fatol': 1e-10})
    if result.fun < best_val:
        best_val = result.fun
        best_pos = result.x.copy()
    
    # Try some random starts
    for _ in range(5):
        start = np.array([np.random.uniform(0.05, 0.95), np.random.uniform(0.05, 0.95)])
        result = minimize(neg_radius, start, bounds=[(0.001, 0.999), (0.001, 0.999)],
                          method='Nelder-Mead', options={'maxiter': 1000, 'xatol': 1e-8, 'fatol': 1e-10})
        if result.fun < best_val:
            best_val = result.fun
            best_pos = result.x.copy()
    
    return best_pos, max(0.0, -best_val)


def hexagonal_init(n, rows=None):
    """Initialize centers in hexagonal pattern."""
    centers = np.zeros((n, 2))
    
    if rows is None:
        rows = [6, 5, 6, 5, 4]  # 26 circles
    
    idx = 0
    y_spacing = 0.18
    x_spacing = 0.14
    
    for row_i, count in enumerate(rows):
        y = 0.12 + row_i * y_spacing
        for col_j in range(count):
            x = 0.06 + col_j * x_spacing + (row_i % 2) * (x_spacing / 2.0)
            x = max(0.01, min(0.99, x))
            y = max(0.01, min(0.99, y))
            centers[idx] = [x, y]
            idx += 1
    
    return centers
