import numpy as np
from scipy.optimize import minimize

def objective(params, n):
    xs = params[0::3]
    ys = params[1::3]
    rs = params[2::3]
    
    sum_r = np.sum(rs)
    
    # Boundary penalties: circles must stay inside [0,1]x[0,1]
    p = np.sum(np.maximum(0.0, rs - xs)**2)
    p += np.sum(np.maximum(0.0, rs - (1.0 - xs))**2)
    p += np.sum(np.maximum(0.0, rs - ys)**2)
    p += np.sum(np.maximum(0.0, rs - (1.0 - ys))**2)
    
    # Overlap penalties: distance between centers must be >= sum of radii
    diff_x = xs[:, None] - xs[None, :]
    diff_y = ys[:, None] - ys[None, :]
    dist = np.sqrt(diff_x**2 + diff_y**2)
    sum_r_mat = rs[:, None] + rs[None, :]
    
    # Avoid self-overlap penalty by setting diagonal to a large distance
    dist = dist.copy()
    np.fill_diagonal(dist, 10.0)
    
    overlap = np.maximum(0.0, sum_r_mat - dist)
    p += np.sum(overlap**2)
    
    # Maximize sum of radii by minimizing negative sum, plus penalties
    return -sum_r + 1000.0 * p

def run_packing():
    n = 26
    np.random.seed(42)
    
    best_val = np.inf
    best_params = None
    
    # Bounds: x,y in [0,1], r in [0, 0.5]
    bnds = list(zip([0.0]*n, [1.0]*n)) + \
           list(zip([0.0]*n, [1.0]*n)) + \
           list(zip([0.0]*n, [0.5]*n))
           
    for trial in range(20):
        if trial == 0:
            # Structured start: 5x5 grid + 1 center circle
            gx = np.linspace(0.1, 0.9, 5)
            gy = np.linspace(0.1, 0.9, 5)
            ix, iy = np.meshgrid(gx, gy)
            ix = ix.flatten()
            iy = iy.flatten()
            ix = np.append(ix, 0.5)
            iy = np.append(iy, 0.5)
            ir = np.ones(n) * 0.06
        else:
            # Random starts with small radii to encourage exploration
            ix = np.random.rand(n) * 0.8 + 0.1
            iy = np.random.rand(n) * 0.8 + 0.1
            ir = np.random.rand(n) * 0.05 + 0.01
            
        params0 = np.zeros(3 * n)
        params0[0::3] = ix
        params0[1::3] = iy
        params0[2::3] = ir
        
        # L-BFGS-B efficiently handles bounds and is robust for medium-sized problems
        res = minimize(objective, params0, method='L-BFGS-B', 
                       bounds=bnds, args=(n,),
                       options={'maxiter': 10000, 'ftol': 1e-12, 'gtol': 1e-10})
        
        if res.fun < best_val:
            best_val = res.fun
            best_params = res.x.copy()
            
    xs = best_params[0::3]
    ys = best_params[1::3]
    rs = best_params[2::3]
    
    centers = np.column_stack((xs, ys))
    radii = rs
    return centers, radii, float(np.sum(radii))
