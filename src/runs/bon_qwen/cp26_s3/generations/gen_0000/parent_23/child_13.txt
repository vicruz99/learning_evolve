import numpy as np
from scipy.optimize import minimize

def run_packing():
    n = 26
    best_sum = -1.0
    best_centers = np.zeros((n, 2))
    best_radii = np.zeros(n)
    
    def objective(x):
        # Minimize negative sum of radii to maximize sum of radii
        return -np.sum(x[2::3])
        
    def constr_func(x):
        xs = x[0::3]
        ys = x[1::3]
        rs = x[2::3]
        
        # Total constraints: 5 boundary/nn per circle + pairwise non-overlap
        m = 5 * n + n * (n - 1) // 2
        vals = np.zeros(m)
        
        # 1. r_i >= 0
        vals[:n] = rs
        # 2. c_x - r >= 0
        vals[n:2*n] = xs - rs
        # 3. 1 - c_x - r >= 0
        vals[2*n:3*n] = 1.0 - xs - rs
        # 4. c_y - r >= 0
        vals[3*n:4*n] = ys - rs
        # 5. 1 - c_y - r >= 0
        vals[4*n:5*n] = 1.0 - ys - rs
        
        # 6. Non-overlap: dist(c_i, c_j) - (r_i + r_j) >= 0
        idx = 5 * n
        for i in range(n):
            for j in range(i + 1, n):
                dx = xs[i] - xs[j]
                dy = ys[i] - ys[j]
                vals[idx] = np.sqrt(dx*dx + dy*dy) - rs[i] - rs[j]
                idx += 1
        return vals

    cons = {'type': 'ineq', 'fun': constr_func}
    # Bounds help guide the optimizer: centers in [0,1], radii in [0, 0.5]
    bounds = [(0.0, 1.0) if k % 3 != 2 else (0.0, 0.5) for k in range(3 * n)]

    # Hexagonal packing row structure for 26 circles
    row_counts = [6, 5, 6, 5, 4]
    
    # Multi-start optimization to escape local minima
    for seed in range(5):
        rng = np.random.default_rng(seed)
        centers = np.zeros((n, 2))
        radii = np.zeros(n)
        
        # Base radius slightly below theoretical max for hex packing
        r_init = 0.095 + rng.uniform(-0.005, 0.005)
        idx = 0
        for row_i, count in enumerate(row_counts):
            y = r_init + row_i * 1.732 * r_init
            for col_j in range(count):
                # Offset odd rows for hexagonal arrangement
                x = r_init + col_j * 2 * r_init + (1.0 * r_init if row_i % 2 == 1 else 0)
                centers[idx] = [x + rng.uniform(-0.02, 0.02), y + rng.uniform(-0.02, 0.02)]
                radii[idx] = r_init + rng.uniform(-0.005, 0.005)
                idx += 1
                
        # Scale down if initial guess exceeds boundaries
        max_val = np.max(centers) + np.max(radii)
        if max_val > 0.99:
            scale = 0.95 / max_val
            centers *= scale
            radii *= scale
            
        x0 = np.zeros(3 * n)
        x0[0::3] = centers[:, 0]
        x0[1::3] = centers[:, 1]
        x0[2::3] = radii
        
        res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
        
        current_sum = -res.fun
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = np.column_stack((res.x[0::3], res.x[1::3]))
            best_radii = res.x[2::3].copy()
            
    # Strict post-processing to guarantee validity against numerical errors
    for i in range(n):
        r = best_radii[i]
        best_centers[i, 0] = np.clip(best_centers[i, 0], r, 1.0 - r)
        best_centers[i, 1] = np.clip(best_centers[i, 1], r, 1.0 - r)
        best_radii[i] = max(best_radii[i], 0.0)
        
    return best_centers, best_radii, float(np.sum(best_radii))
