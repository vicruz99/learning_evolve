import numpy as np
from scipy.optimize import minimize

def run_packing():

    Pack 26 circles in a unit square to maximize the sum of radii.
    """
    n = 26
    best_sum = -1.0
    best_state = None

    # We will try multiple random initializations to find a good local optimum
    # and pick the best one.
    num_restarts = 20
    
    for restart in range(num_restarts):
        # 1. Generate Initial Configuration
        # A hexagonal packing initialization is generally better than random.
        # We place points in a hexagonal grid and pick the first 'n' that fit.
        
        centers = []
        radii_init = []

        # We try a few different densities for initialization
        density_factor = 0.15 + restart * 0.01 # vary slightly

            x = density_factor
            shift = 0.5 * density_factor if int((y - density_factor) / (density_factor * np.sqrt(3))) % 2 == 1 else 0
            
            # Adjust shift logic for hexagonal packing
            # Row k: shift by r if k is odd
            row_idx = int((y - density_factor) / (density_factor * np.sqrt(3)))
            current_shift = density_factor / 2 if row_idx % 2 == 1 else 0
            
            x = density_factor + current_shift
            while x < 1 - density_factor:
                centers.append([x, y])
                radii_init.append(density_factor / 2) # Initial radius guess
                
                if len(centers) >= n:
                    break
                x += density_factor # Horizontal spacing approx 2r
            
            if len(centers) >= n:
                break
            
            y += density_factor * np.sqrt(3) / 2 # Vertical spacing approx sqrt(3)r

        # If grid didn't fill enough (unlikely), pad with random
        while len(centers) < n:
            centers.append([np.random.rand(), np.random.rand()])
            radii_init.append(0.05)

        centers = np.array(centers[:n])
        radii_init = np.array(radii_init[:n])

        # Flatten variables: [x1, y1, r1, x2, y2, r2, ...]
        x0 = np.hstack([centers.flatten(), radii_init])

        # 2. Define Objective and Constraints for scipy
        
        def objective(vars_flat):
            # We want to maximize sum of radii, so minimize negative sum
            rs = vars_flat[2::3]
            return -np.sum(rs)

        def constraint_boundary(vars_flat):
            # Constraints: r_i >= 0, x_i - r_i >= 0, 1 - (x_i + r_i) >= 0, etc.
            constraints = []
            for i in range(n):
                idx = i * 3
                x, y, r = vars_flat[idx], vars_flat[idx+1], vars_flat[idx+2]
                
                # r >= 0
                constraints.append(r)
                # x - r >= 0
                constraints.append(x - r)
                # 1 - x - r >= 0
                constraints.append(1 - x - r)
                # y - r >= 0
                constraints.append(y - r)
                # 1 - y - r >= 0
                constraints.append(1 - y - r)
            return np.array(constraints)

        def constraint_overlap(vars_flat):
            # Constraints: dist_ij >= r_i + r_j  => dist_ij^2 >= (r_i + r_j)^2
            constraints = []
            for i in range(n):
                for j in range(i + 1, n):
                    idx_i = i * 3
                    idx_j = j * 3
                    
                    xi, yi, ri = vars_flat[idx_i], vars_flat[idx_i+1], vars_flat[idx_i+2]
                    xj, yj, rj = vars_flat[idx_j], vars_flat[idx_j+1], vars_flat[idx_j+2]
                    
                    dist_sq = (xi - xj)**2 + (yi - yj)**2
                    sum_r = ri + rj
                    
                    # We want dist_sq - sum_r^2 >= 0
                    # Note: Using dist - sum_r is smoother but dist_sq - sum_r^2 is differentiable everywhere
                    # However, dist - sum_r is more standard for distance.
                    # Let's use dist - sum_r.
                    dist = np.sqrt(dist_sq)
                    constraints.append(dist - sum_r)
            return np.array(constraints)

        # Define constraint objects for SLSQP
        # SLSQP expects a list of dictionaries
        cons = []

        # Let's use bounds for x, y in [0, 1] and r >= 0?
        # No, x must be >= r. So bounds are not sufficient.

        # Individual constraints are safer for SLSQP.
        
        # Boundary constraints
        for i in range(n):
            idx = i * 3
            # r >= 0
            cons.append({
                'type': 'ineq',
                'fun': lambda v, i=i: v[idx+2]
            })
            # x - r >= 0
            cons.append({
                'type': 'ineq',
                'fun': lambda v, i=i: v[idx] - v[idx+2]
            })
            # 1 - x - r >= 0
            cons.append({
                'type': 'ineq',
                'fun': lambda v, i=i: 1 - v[idx] - v[idx+2]
            })
            # y - r >= 0
            cons.append({
                'type': 'ineq',
                'fun': lambda v, i=i: v[idx+1] - v[idx+2]
            })
            # 1 - y - r >= 0
            cons.append({
                'type': 'ineq',
                'fun': lambda v, i=i: 1 - v[idx+1] - v[idx+2]
            })

                cons.append({

                })

        # SLSQP is good for constrained non-linear optimization
        try:

                objective, 
                x0, 
                method='SLSQP', 
                constraints=cons,
                options={'maxiter': 1000, 'ftol': 1e-9}
            )

                    best_state = res.x.copy()
        except Exception as e:

            continue

    # 4. Extract Result
    if best_state is None:
        # Fallback to a simple grid if optimization failed completely
        centers = []

        # 5x5 grid

        radii = np.zeros(n)

            centers[i] = [best_state[i*3], best_state[i*3+1]]
            radii[i] = best_state[i*3+2]

    # 5. Post-processing to ensure validity
    # Shrink radii slightly to account for numerical precision errors

    epsilon = 1e-10
    radii = radii * (1 - epsilon) 
    
    # Ensure centers are valid (clip to bounds)
    centers[:, 0] = np.clip(centers[:, 0], 0, 1)
    centers[:, 1] = np.clip(centers[:, 1], 0, 1)
    
    # Ensure r >= 0
    radii = np.maximum(radii, 0)

    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
