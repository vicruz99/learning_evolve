def compute_distances(centers):
    """Compute pairwise distances between centers."""
    n = centers.shape[0]
    dists = np.zeros((n, n))
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            d = math.hypot(dx, dy)
            dists[i, j] = d
            dists[j, i] = d
    return dists
