import numpy as np
from scipy.optimize import minimize
import math

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    Returns (centers, radii, sum_radii).
    """
    n_circles = 26
    np.random.seed(42)  # Fixed seed for reproducibility, but we will vary it internally

    def objective(x):
        # x contains [x_1, y_1, r_1, x_2, y_2, r_2, ..., x_n, y_n, r_n]
        # We want to maximize sum of radii, so minimize negative sum
        radii = x[2::3]
        return -np.sum(radii)

    def boundary_constraints(x):
        # For each circle i:
        # r_i <= x_i <= 1 - r_i  =>  r_i - x_i <= 0, x_i + r_i - 1 <= 0
        # r_i <= y_i <= 1 - r_i  =>  r_i - y_i <= 0, y_i + r_i - 1 <= 0
        constraints = []
        for i in range(n_circles):
            idx = 3 * i
            xi, yi, ri = x[idx], x[idx+1], x[idx+2]
            constraints.append(ri - xi)      # x_i - r_i >= 0
            constraints.append(xi + ri - 1)  # x_i + r_i <= 1
            constraints.append(ri - yi)      # y_i - r_i >= 0
            constraints.append(yi + ri - 1)  # y_i + r_i <= 1
        return constraints

    def overlap_constraints(x):
        # For each pair (i, j) with i < j:
        # dist(i, j) >= r_i + r_j
        # (xi - xj)^2 + (yi - yj)^2 >= (ri + rj)^2
        # This is non-linear. For SLSQP, we can use the inequality constraint:
        # (ri + rj)^2 - (dist)^2 <= 0
        constraints = []
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                idx_i = 3 * i
                idx_j = 3 * j
                xi, yi, ri = x[idx_i], x[idx_i+1], x[idx_i+2]
                xj, yj, rj = x[idx_j], x[idx_j+1], x[idx_j+2]
                
                dist_sq = (xi - xj)**2 + (yi - yj)**2
                sum_radii_sq = (ri + rj)**2
                
                # Constraint: sum_radii_sq - dist_sq <= 0
                constraints.append(sum_radii_sq - dist_sq)
        return constraints

    # Helper function for the main optimization loop
    def solve_optimization(initial_centers):
        n = n_circles
        # Initial radii
        initial_radii = np.full(n, 0.1)
        
        # Construct initial vector x
        x0 = np.zeros(3 * n)
        for i in range(n):
            x0[3*i] = initial_centers[i, 0]
            x0[3*i+1] = initial_centers[i, 1]
            x0[3*i+2] = initial_radii[i]

        # Bounds for variables
        # x, y in [0, 1], r in [0, 0.5] (since diameter <= 1)
        bounds = []
        for _ in range(n):
            bounds.append((0, 1)) # x
            bounds.append((0, 1)) # y
            bounds.append((1e-5, 0.5)) # r (small positive lower bound)

        # Define constraints for SLSQP
        # SLSQP accepts a list of constraint dictionaries
        cons = []
        
        # Boundary constraints
        # We define a function that returns the vector of constraint values
        def fun_bounds(x):
            return boundary_constraints(x)
        cons.append({'type': 'ineq', 'fun': lambda x: [-c for c in fun_bounds(x)]}) 
        # Note: SLSQP ineq is fun(x) >= 0. Our constraints are <= 0, so we negate.

        # Overlap constraints
        def fun_overlap(x):
            return overlap_constraints(x)
        cons.append({'type': 'ineq', 'fun': lambda x: [-c for c in fun_overlap(x)]})

        # Run optimization
        res = minimize(
            objective, 
            x0, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons, 
            options={'maxiter': 2000, 'ftol': 1e-9}
        )
        
        if res.success:
            centers = np.column_stack((res.x[0::3], res.x[1::3]))
            radii = res.x[2::3]
            return centers, radii, -res.fun
        else:
            return None, None, 0

    def generate_initial_config():
        # Randomly place circles, then push them apart slightly to avoid immediate overlap
        centers = np.random.rand(n_circles, 2)
        # Simple relaxation to spread points
        for _ in range(50):
            forces = np.zeros_like(centers)
            for i in range(n_circles):
                for j in range(i+1, n_circles):
                    diff = centers[i] - centers[j]
                    dist = np.linalg.norm(diff)
                    if dist < 0.15: # Repulsion radius
                        repulsion = 0.01 * diff / (dist + 1e-6)
                        forces[i] += repulsion
                        forces[j] -= repulsion
            centers += forces
            # Project back to [0,1]
            centers = np.clip(centers, 0, 1)
        return centers

    best_centers = None
    best_radii = None
    best_sum = -np.inf

    # Run multiple times with different seeds to find a good solution
    for trial in range(5):
        np.random.seed(100 + trial)
        init_centers = generate_initial_config()
        c, r, s = solve_optimization(init_centers)
        if s > best_sum:
            best_sum = s
            best_centers = c
            best_radii = r

    # Final validation and adjustment if needed
    # Sometimes optimization might violate constraints slightly due to numerical tolerance
    # We scale down radii slightly if validation fails, but with good solver it shouldn't.
    
    # Let's verify and ensure strict compliance
    centers = best_centers
    radii = best_radii
    
    # Check constraints manually and shrink if necessary
    # This is a safety net
    max_violation = 0
    n = len(radii)
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        # Boundary
        if x - r < 0:
            max_violation = max(max_violation, -(x - r))
        if x + r > 1:
            max_violation = max(max_violation, x + r - 1)
        if y - r < 0:
            max_violation = max(max_violation, -(y - r))
        if y + r > 1:
            max_violation = max(max_violation, y + r - 1)
            
        for j in range(i + 1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            if dist < radii[i] + radii[j]:
                max_violation = max(max_violation, radii[i] + radii[j] - dist)

    if max_violation > 1e-6:
        # Scale radii down slightly to fix violations
        # A simple global scaling factor
        scale_factor = 1.0 - max_violation * 2 # Heuristic
        radii *= max(0.95, scale_factor)
        # Recalculate sum
        best_sum = np.sum(radii)

    return centers, radii, best_sum

# Helper for the prompt requirement (not strictly needed for run_packing but good practice)
def validate_packing(centers, radii):
    n = centers.shape[0]
    if np.isnan(centers).any() or np.isnan(radii).any():
        return False
    for i in range(n):
        if radii[i] < 0:
            return False
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            return False
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False
    return True
