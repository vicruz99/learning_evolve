import numpy as np

def calculate_energy(centers, radii):
    """Calculate total overlap energy. Lower is better (0 is valid)."""
    energy = 0.0
    n = centers.shape[0]
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        # Boundary penalty
        if x - r < 0: energy += (r - x)**2
        if x + r > 1: energy += (x + r - 1)**2
        if y - r < 0: energy += (r - y)**2
        if y + r > 1: energy += (y + r - 1)**2
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            overlap = radii[i] + radii[j] - dist
            if overlap > 0:
                energy += overlap**2
    return energy
