import numpy as np
from scipy.optimize import minimize

def compute_cost(vars, n):
    centers = vars[:2 * n].reshape(n, 2)
    radii = vars[2 * n:]
    
    # Objective: maximize sum of radii -> minimize negative sum
    cost = -np.sum(radii)
    penalty = 1.0e4
    
    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x < r:
            cost += penalty * (r - x) ** 2
        if x > 1.0 - r:
            cost += penalty * (x - (1.0 - r)) ** 2
        if y < r:
            cost += penalty * (r - y) ** 2
        if y > 1.0 - r:
            cost += penalty * ((1.0 - r) - y) ** 2
            
    # Overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            r_sum = radii[i] + radii[j]
            if dist < r_sum:
                cost += penalty * (r_sum - dist) ** 2
                
    return cost

def run_packing():
    n = 26
    best_cost = np.inf
    best_vars = None
    
    # Multiple restarts to avoid local minima
    for seed in range(8):
        rng = np.random.default_rng(seed)
        
        # Hexagonal initial layout perturbed by noise
        centers = np.zeros((n, 2))
        idx = 0
        for r in range(5):
            for c in range(6):
                if idx >= n:
                    break
                x = 0.08 + c * 0.13 + (r % 2) * 0.065
                y = 0.08 + r * 0.16
                centers[idx] = [x, y]
                idx += 1
                
        # Add controlled noise
        centers += rng.normal(0, 0.02, centers.shape)
        centers = np.clip(centers, 0.1, 0.9)
        
        radii = np.full(n, 0.025)
        x0 = np.concatenate([centers.flatten(), radii])
        
        bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
        
        res = minimize(compute_cost, x0, args=(n,), method='L-BFGS-B', 
                       bounds=bounds, options={'maxiter': 5000, 'ftol': 1e-10})
        
        if res.fun < best_cost:
            best_cost = res.fun
            best_vars = res.x
            
    centers = best_vars[:2 * n].reshape(n, 2)
    radii = best_vars[2 * n:]
    
    # Post-optimization projection to guarantee strict feasibility
    # 1. Enforce boundary constraints strictly
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        # Clamp radius to fit within boundaries
        max_r_boundary = min(x, 1.0 - x, y, 1.0 - y)
        if radii[i] > max_r_boundary + 1e-12:
            radii[i] = max_r_boundary
            
    # 2. Resolve overlaps by scaling down radii of conflicting pairs
    # Iterate a few times to propagate adjustments
    for _ in range(5):
        overlap_found = False
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                r_sum = radii[i] + radii[j]
                
                if dist < r_sum - 1e-12:
                    overlap_found = True
                    # Scale both radii down to exactly touch
                    if r_sum > 1e-9:
                        factor = dist / r_sum
                        radii[i] *= np.sqrt(factor)
                        radii[j] *= np.sqrt(factor)
                    else:
                        radii[i] = dist / 2.0
                        radii[j] = dist / 2.0
                        
        if not overlap_found:
            break
            
    # Re-enforce boundary constraints after radius adjustments
    for i in range(n):
        x, y = centers[i]
        max_r_boundary = min(x, 1.0 - x, y, 1.0 - y)
        if radii[i] > max_r_boundary:
            radii[i] = max_r_boundary
            
    return centers, radii, np.sum(radii)
