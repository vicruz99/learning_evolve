import numpy as np
import scipy.optimize as opt
import math

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    
    Returns:
        centers: np.array of shape (26, 2) with (x, y) coordinates
        radii: np.array of shape (26) with radius of each circle
        sum_radii: float representing the sum of radii
    """
    n = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None

    # Optimization parameters
    penalty_weight = 10000.0
    boundary_weight = 10000.0
    max_iter = 1000

    def objective(params):
        """Objective function: maximize sum of radii with penalties for overlaps and boundaries."""
        centers = params[:2*n].reshape((n, 2))
        radii = params[2*n:]
        
        # Penalty for negative radii
        penalty = np.sum(np.maximum(0, -radii))
        
        # Penalty for boundary violations
        # Circle i must satisfy: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            # Distance to boundaries
            d_left = x
            d_right = 1 - x
            d_bottom = y
            d_top = 1 - y
            dist_to_boundary = min(d_left, d_right, d_bottom, d_top)
            if r > dist_to_boundary:
                penalty += boundary_weight * (r - dist_to_boundary)**2
        
        # Penalty for overlaps
        # Distance between centers must be >= r_i + r_j
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                min_dist = radii[i] + radii[j]
                if dist < min_dist:
                    penalty += penalty_weight * (min_dist - dist)**2
        
        # We want to maximize sum(radii), so we minimize -sum(radii) + penalty
        return -np.sum(radii) + penalty

    def get_initial_guess(strategy, seed):
        """Generates initial guesses based on different strategies."""
        np.random.seed(seed)
        centers = np.zeros((n, 2))
        radii = np.ones(n) * 0.05 # Start with small radii

            centers = np.random.uniform(0.1, 0.9, (n, 2))

                    coords.append([0.1 + i*0.2, 0.1 + j*0.2])

                    x = 0.1 + col * 0.2 + shift
                    centers[idx] = [x, y]
                    idx += 1
                row_count += 1
            radii = np.ones(n) * 0.095
            
        return np.hstack([centers.flatten(), radii])

    strategies = ['random', 'grid', 'hexagonal']
    seeds = list(range(20)) # 20 restarts

    for strategy in strategies:

            params0 = get_initial_guess(strategy, seed)
            
            # Bounds: centers in [0, 1], radii in [0, 0.5]
            bounds = [(0, 1)] * (2 * n) + [(0, 0.5)] * n

            try:
                # Use L-BFGS-B for local optimization with bounds
                res = opt.minimize(
                    objective,
                    params0,
                    method='L-BFGS-B',
                    bounds=bounds,
                    options={'maxiter': max_iter, 'ftol': 1e-12}

                                is_valid = False
                                break
                        if not is_valid: break

                        if current_sum > best_sum:
                            best_sum = current_sum
                            best_centers = centers_opt.copy()
                            best_radii = radii_opt.copy()

    # If we found a valid packing, try to scale radii up slightly

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square

    Args:
        centers: np.array of shape (n, 2) with (x, y) coordinates
        radii: np.array of shape (n) with radius of each circle

    Returns:
        True if valid, False otherwise
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True
