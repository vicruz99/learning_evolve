import numpy as np
from scipy.optimize import minimize
import math

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    N = 26
    
    # 1. Initial Guess: Hexagonal packing
    r_init = 0.09
    centers_init = []
    r_list_init = []
    
    # Hexagonal lattice generation
    row_idx = 0
    y = r_init
    # Calculate vertical step
    y_step = math.sqrt(3) * r_init
    
    while len(centers_init) < N and y <= 1 - r_init + 1e-6:
        # Alternating row start for hexagonal packing
        if row_idx % 2 == 0:
            x_start = r_init
        else:
            x_start = 2 * r_init
        
        x = x_start
        while x <= 1 - r_init + 1e-6 and len(centers_init) < N:
            centers_init.append([x, y])
            r_list_init.append(r_init)
            x += 2 * r_init
        
        y += y_step
        row_idx += 1
        
    # Fallback if hexagonal generation didn't yield 26 points
    if len(centers_init) < N:
        # Grid fallback
        idx = 0
        for i in range(6):
            for j in range(5):
                if idx < N:
                    centers_init.append([0.1 + j * 0.2, 0.1 + i * 0.2])
                    r_list_init.append(0.08)
                    idx += 1

    # Truncate to N
    centers = np.array(centers_init[:N])
    radii = np.array(r_list_init[:N])
    
    # 2. Perturbation to avoid symmetry and local minima
    np.random.seed(42)
    centers += np.random.uniform(-0.005, 0.005, centers.shape)
    radii += np.random.uniform(-0.002, 0.002, radii.shape)
    
    # Clip to ensure valid starting bounds
    centers = np.clip(centers, 0.01, 0.99)
    radii = np.clip(radii, 0.001, 0.49)

    # 3. Optimization Setup
    x0 = np.concatenate([centers.flatten(), radii])
    
    # Bounds: x, y in [0, 1], r in [0, 0.5]
    bounds = []
    for i in range(N):
        bounds.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])

    def constraint_func_vec(vars):
        cs = vars[:2*N].reshape(N, 2)
        rs = vars[2*N:]
        
        # Boundary constraints: x - r >= 0, 1 - x - r >= 0, etc.
        c1 = cs[:, 0] - rs
        c2 = 1.0 - cs[:, 0] - rs
        c3 = cs[:, 1] - rs
        c4 = 1.0 - cs[:, 1] - rs
        
        # Overlap constraints: dist - (r1 + r2) >= 0
        # Vectorized pairwise distance
        diff = cs[:, np.newaxis, :] - cs[np.newaxis, :, :] # (N, N, 2)
        dists = np.sqrt(np.sum(diff**2, axis=2)) # (N, N)
        sum_radii = rs[:, np.newaxis] + rs[np.newaxis, :] # (N, N)
        
        # Upper triangle indices
        mask = np.triu(np.ones((N, N), dtype=bool), k=1)
        overlaps = (dists - sum_radii)[mask]
        
        return np.concatenate([c1, c2, c3, c4, overlaps])

    cons = {'type': 'ineq', 'fun': constraint_func_vec}
    
    def objective(vars):
        rs = vars[2*N:]
        return -np.sum(rs)

    # Run optimizer
    try:
        res = minimize(
            objective, 
            x0, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons, 
            options={'maxiter': 2000, 'ftol': 1e-10}
        )
        centers_opt = res.x[:2*N].reshape(N, 2)
        radii_opt = res.x[2*N:]
    except Exception as e:
        # Fallback if optimization fails
        print(f"Optimization failed: {e}")
        centers_opt = centers
        radii_opt = radii

    return centers_opt, radii_opt, np.sum(radii_opt)
