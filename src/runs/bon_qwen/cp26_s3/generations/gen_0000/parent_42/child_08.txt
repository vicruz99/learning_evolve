For fixed centers, the problem of finding maximum radii is a **Linear Programming (LP)** problem. The constraints are linear in $r_i$:
-   Boundary: $r_i \le \min(x_i, 1-x_i, y_i, 1-y_i)$
-   Non-overlap: $r_i + r_j \le \text{dist}(c_i, c_j)$ for all pairs $i, j$.

import numpy as np
from scipy.optimize import linprog, minimize

def solve_radii(centers):
    """
    Given centers, solve the LP to find maximum radii.
    Maximize sum(r_i)
    Subject to:
      r_i + r_j <= dist(c_i, c_j)
      r_i <= min(x_i, 1-x_i, y_i, 1-y_i)
      r_i >= 0
    """
    n = centers.shape[0]
    
    # Objective: Maximize sum(r) => Minimize -sum(r)
    c_obj = -np.ones(n)
    
    # Constraints: A_ub @ r <= b_ub
    # 1. Pairwise constraints: r_i + r_j <= d_ij
    # 2. Boundary constraints: r_i <= bound_i
    
    # We will build A_ub and b_ub
    # Number of variables: n
    # Number of constraints: n(n-1)/2 + 4*n
    
    # Using sparse matrix construction might be faster but n=26 is small.
    # Let's just build lists.
    
    A_ub_list = []
    b_ub_list = []
    
    # Pairwise constraints
    # There are n*(n-1)/2 pairs.
    # We can iterate i < j
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub_list.append(row)
            b_ub_list.append(dist)
            
    # Boundary constraints
    # r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        x, y = centers[i]
        bound = min(x, 1.0 - x, y, 1.0 - y)
        # r_i <= bound  => 1*r_i <= bound
        row = np.zeros(n)
        row[i] = 1.0
        A_ub_list.append(row)
        b_ub_list.append(bound)
        
    A_ub = np.array(A_ub_list)
    b_ub = np.array(b_ub_list)
    
    # Bounds for r_i: r_i >= 0
    # linprog bounds default to (0, inf) for variables? 
    # Actually default is (-inf, inf). We need (0, inf).
    bounds = [(0, None)] * n
    
    # Solve LP
    # method='highs' is default in newer scipy, robust.
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return -res.fun, res.x
    else:
        # Fallback or error handling
        # If LP fails (e.g. infeasible), return 0 and zeros
        return 0.0, np.zeros(n)

def objective_function(centers_flat):
    """
    Objective function for optimizing centers.
    Returns negative sum of radii.
    """
    centers = centers_flat.reshape(-1, 2)
    # Ensure centers are within [0,1] roughly, though optimizer bounds handle this.
    # However, to avoid numerical issues with distance, maybe clamp?
    # Not strictly necessary if bounds are set.
    
    _, radii = solve_radii(centers)
    sum_radii = np.sum(radii)
    return -sum_radii
