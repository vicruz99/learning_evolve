import numpy as np
from scipy.spatial.distance import pdist
from scipy.optimize import linprog

def generate_initial_configs(n, num_configs):
    configs = []
    # Random configurations spread within safe margins
    for _ in range(num_configs // 2):
        configs.append(np.random.uniform(0.15, 0.85, (n, 2)))
        
    # Hexagonal-like grid configurations to promote dense packing
    for shift_x in np.linspace(0, 0.2, 4):
        for shift_y in np.linspace(0, 0.2, 4):
            pts = []
            rows, cols = 6, 5
            spacing = 0.22
            for i in range(rows):
                for j in range(cols):
                    if len(pts) >= n:
                        break
                    x = 0.1 + j * spacing + (i % 2) * spacing / 2 + shift_x
                    y = 0.1 + i * spacing * 0.866 + shift_y
                    if 0 <= x <= 1 and 0 <= y <= 1:
                        pts.append([x, y])
                if len(pts) >= n:
                    break
            if len(pts) == n:
                configs.append(np.array(pts))
                
    return configs[:num_configs]

def simulate_packing(centers_init, n):
    centers = centers_init.copy()
    radii = np.full(n, 0.01)
    
    dt = 0.015
    expand_rate = 5e-5
    max_iter = 2500
    cooling = 0.9997
    
    for _ in range(max_iter):
        radii += expand_rate
        
        # Compute pairwise distances
        dists = pdist(centers)
        r_sum_full = radii[:, None] + radii[None, :]
        dists_full = np.zeros((n, n))
        idx = np.triu_indices(n, k=1)
        dists_full[idx] = dists
        dists_full += dists_full.T
        
        overlaps = np.maximum(0, r_sum_full - dists_full)
        
        if np.sum(overlaps) < 1e-12:
            expand_rate *= 0.95
            continue
            
        forces = np.zeros_like(centers)
        
        # Pairwise repulsion
        diff = centers[:, None, :] - centers[None, :, :]
        dists_safe = np.maximum(dists_full, 1e-9)
        directions = diff / dists_safe[:, :, None]
        
        # Force proportional to overlap depth
        force_mag = overlaps
        forces += np.sum(directions * force_mag[:, :, None], axis=1)
        
        # Boundary repulsion
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < 0:
                forces[i, 0] += (r - x) * 8.0
            if x + r > 1:
                forces[i, 0] -= (x + r - 1) * 8.0
            if y - r < 0:
                forces[i, 1] += (r - y) * 8.0
            if y + r > 1:
                forces[i, 1] -= (y + r - 1) * 8.0
                
        centers += dt * forces
        centers = np.clip(centers, 0.0, 1.0)
        
        expand_rate *= cooling
        
    return centers

def solve_radii_lp(centers, n):
    dists = pdist(centers)
    A_ub = []
    b_ub = []
    
    # Non-overlap constraints: r_i + r_j <= d_ij
    for idx, (i, j) in enumerate(np.triu_indices(n, k=1)):
        row = np.zeros(n)
        row[i] = 1.0
        row[j] = 1.0
        A_ub.append(row)
        b_ub.append(dists[idx])
        
    # Boundary constraints: r_i <= x_i, 1-x_i, y_i, 1-y_i
    for i in range(n):
        bounds = [centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1]]
        for b in bounds:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(max(0.0, b))
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)] * n
    
    # Maximize sum(r) -> Minimize -sum(r)
    res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return res.x, np.sum(res.x)
    return None, 0.0

def run_packing():
    np.random.seed(42)
    n = 26
    configs = generate_initial_configs(n, num_configs=50)
    
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    for cfg in configs:
        centers = simulate_packing(cfg, n)
        radii, s = solve_radii_lp(centers, n)
        if radii is not None and s > best_sum:
            best_sum = s
            best_centers = centers.copy()
            best_radii = radii.copy()
            
    return best_centers, best_radii, best_sum
