def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True

        radii[i] = params[3 * i + 2]

    # Clip radii to be non-negative (though optimization should handle this)
    radii = np.maximum(radii, 0)
    
    # Penalty for boundary constraints
    penalty = 0.0
    
    # Boundary check
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        # Circle must be inside [0, 1]x[0, 1]
        # x - r >= 0 => r - x <= 0
        # x + r <= 1 => x + r - 1 <= 0
        # y - r >= 0 => r - y <= 0

        val2 = x + r - 1
        val3 = r - y
        val4 = y + r - 1
        
        if val1 > 0: penalty += 1000 * (val1 ** 2)
        if val2 > 0: penalty += 1000 * (val2 ** 2)
        if val3 > 0: penalty += 1000 * (val3 ** 2)
        if val4 > 0: penalty += 1000 * (val4 ** 2)

    # Overlap check
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = math.sqrt(dx*dx + dy*dy)
            min_dist = radii[i] + radii[j]
            if dist < min_dist:
                overlap = min_dist - dist
                penalty += 1000 * (overlap ** 2)

    # We want to maximize sum of radii, so minimize negative sum
    return -np.sum(radii) + penalty
