import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # 1. Initialization: Hexagonal Grid
    # We place circles in a hexagonal lattice pattern to start with a dense, valid packing.
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Parameters for initial hexagonal packing
    r_init = 0.08
    d = 2 * r_init
    
    idx = 0
    y = r_init
    row = 0
    
    # Generate hexagonal points
    while idx < n and y + r_init <= 1.0:
        # Shift odd rows for hexagonal packing
        shift = (d / 2.0) if (row % 2 == 1) else 0.0
        x = r_init + shift
        
        while x + r_init <= 1.0 and idx < n:
            centers[idx] = [x, y]
            radii[idx] = r_init
            x += d
            idx += 1
        
        y += d * np.sqrt(3) / 2.0
        row += 1
        
    # Fallback to random placement if grid generation failed (unlikely with small r)
    if idx < n:
        centers = np.random.rand(n, 2) * 0.8 + 0.1
        radii = np.ones(n) * 0.01

    # 2. Optimization Setup
    # Flatten variables into a single vector: [x1...xn, y1...yn, r1...rn]
    x0 = np.concatenate([centers[:, 0], centers[:, 1], radii])
    
    # Bounds: x, y in [0, 1], r >= 0
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)] * n + [(0.0, 0.5)] * n
    
    # Objective: Maximize sum of radii (minimize negative sum)
    def objective(vars):
        r_vals = vars[2 * n:]
        return -np.sum(r_vals)
    
    # Constraints Function
    def constraints(vars):
        x_vals = vars[:n]
        y_vals = vars[n:2 * n]
        r_vals = vars[2 * n:]
        
        all_constraints = []
        
        # Boundary constraints
        # x - r >= 0
        all_constraints.extend(x_vals - r_vals)
        # 1 - x - r >= 0
        all_constraints.extend(1.0 - x_vals - r_vals)
        # y - r >= 0
        all_constraints.extend(y_vals - r_vals)
        # 1 - y - r >= 0
        all_constraints.extend(1.0 - y_vals - r_vals)
        
        # Pairwise non-overlap constraints
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt((x_vals[i] - x_vals[j])**2 + (y_vals[i] - y_vals[j])**2)
                # dist - (r_i + r_j) >= 0
                all_constraints.append(dist - (r_vals[i] + r_vals[j]))
                
        return np.array(all_constraints)

    cons = {'type': 'ineq', 'fun': constraints}

    # Run Optimization
    try:
        res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, 
                       options={'maxiter': 2000, 'ftol': 1e-10, 'disp': False})
        
        # Extract results
        final_x = res.x[:n]
        final_y = res.x[n:2 * n]
        final_r = res.x[2 * n:]
        
        # Safety: ensure radii are non-negative
        final_r = np.maximum(final_r, 0.0)
        
        # Construct output arrays
        final_centers = np.column_stack((final_x, final_y))
        final_radii = final_r
        
    except Exception:
        # If optimization fails, return the initial valid configuration
        final_centers = centers
        final_radii = radii

    sum_radii = np.sum(final_radii)
    
    # Final validation and safety adjustments
    # (Though SLSQP should satisfy constraints, we ensure strict compliance)
    final_radii = np.maximum(final_radii, 0.0)
    
    return final_centers, final_radii, float(sum_radii)

if __name__ == "__main__":
    centers, radii, s = run_packing()
    print(f"Sum of radii: {s}")
    print(f"Centers shape: {centers.shape}")
