import numpy as np
from scipy.optimize import minimize


def compute_constraints(params, n):
    """Compute all inequality constraints (must be >= 0)."""
    c = params[:2 * n].reshape(n, 2)
    r = params[2 * n:]
    
    cons = []
    # Boundary constraints
    for i in range(n):
        cons.append(c[i, 0] - r[i])         # x >= r
        cons.append(1.0 - c[i, 0] - r[i])   # x + r <= 1
        cons.append(c[i, 1] - r[i])         # y >= r
        cons.append(1.0 - c[i, 1] - r[i])   # y + r <= 1
    
    # Overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            dx = c[i, 0] - c[j, 0]
            dy = c[i, 1] - c[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            cons.append(dist - r[i] - r[j])
    
    return np.array(cons)


def force_based_relaxation(centers, radii, n, max_steps=2000):
    """Grow circles and resolve overlaps via force-based relaxation."""
    centers = centers.copy().astype(np.float64)
    radii = radii.copy().astype(np.float64)
    
    growth_step = 0.0005
    
    for step in range(max_steps):
        # Try to grow all circles
        radii += growth_step
        
        # Resolve overlaps iteratively
        converged = False
        for _ in range(200):
            moved = False
            for i in range(n):
                for j in range(i + 1, n):
                    dx = centers[i, 0] - centers[j, 0]
                    dy = centers[i, 1] - centers[j, 1]
                    dist = np.sqrt(dx * dx + dy * dy)

                        dist = 1e-15
                        dx = 0.01
                        dy = 0.01

                    if overlap > 1e-14:
                        nx = dx / dist
                        ny = dy / dist
                        push = overlap / 2.0 + 1e-7
                        
                        centers[i, 0] += nx * push
                        centers[i, 1] += ny * push
                        centers[j, 0] -= nx * push
                        centers[j, 1] -= ny * push
                        moved = True
            
            # Handle boundary violations
            for i in range(n):
                if centers[i, 0] < radii[i] + 1e-14:
                    centers[i, 0] = radii[i] + 1e-14
                    moved = True
                if centers[i, 0] > 1.0 - radii[i] - 1e-14:
                    centers[i, 0] = 1.0 - radii[i] - 1e-14
                    moved = True
                if centers[i, 1] < radii[i] + 1e-14:
                    centers[i, 1] = radii[i] + 1e-14
                    moved = True
                if centers[i, 1] > 1.0 - radii[i] - 1e-14:
                    centers[i, 1] = 1.0 - radii[i] - 1e-14
                    moved = True
            
            if not moved:
                converged = True
                break
        
        if not converged:
            # Growth too fast, shrink back
            radii -= growth_step * 2
            growth_step *= 0.5
            if growth_step < 1e-8:
                break
    
    return centers, radii


def optimize_given_centers(centers, n):
    """Optimize radii given fixed centers using iterative update."""
    radii = np.ones(n) * 0.01
    
    for _ in range(200):
        new_radii = np.zeros(n)
        for i in range(n):
            # Distance to boundaries
            r_max = min(
                centers[i, 0],
                1.0 - centers[i, 0],
                centers[i, 1],
                1.0 - centers[i, 1]
            )
            # Distance to other circles (accounting for their radii)
            for j in range(n):
                if i != j:
                    dx = centers[i, 0] - centers[j, 0]
                    dy = centers[i, 1] - centers[j, 1]
                    dist = np.sqrt(dx * dx + dy * dy)
                    r_max = min(r_max, dist - radii[j])
            new_radii[i] = max(0, r_max)
        
        if np.allclose(radii, new_radii, atol=1e-12):
            break
        radii = new_radii
    
    return radii


def optimize_centers_given_radii(centers, radii, n, max_iter=500):

def slsqp_optimization(centers, radii, n):
    """Use SLSQP to optimize both centers and radii."""
    params0 = np.concatenate([centers.flatten(), radii])
    
    def objective(params):
        return -np.sum(params[2 * n:])
    
    def constraint_fun(params):
        return compute_constraints(params, n)
    
    bounds = []
    for i in range(n):
        bounds.extend([(0.0, 1.0), (0.0, 1.0)])
        bounds.append((0.0, 0.5))
    
    constraint_dict = {
        'type': 'ineq',
        'fun': constraint_fun
    }
    
    result = minimize(
        objective,
        params0,
        method='SLSQP',
        bounds=bounds,
        constraints=constraint_dict,
        options={'maxiter': 3000, 'ftol': 1e-14, 'disp': False}
    )
    
    opt_centers = result.x[:2 * n].reshape(n, 2)
    opt_radii = result.x[2 * n:]
    
    # Clip to ensure validity
    for i in range(n):
        opt_radii[i] = max(0, opt_radii[i])
        opt_centers[i, 0] = np.clip(opt_centers[i, 0], opt_radii[i] + 1e-13, 1.0 - opt_radii[i] - 1e-13)
        opt_centers[i, 1] = np.clip(opt_centers[i, 1], opt_radii[i] + 1e-13, 1.0 - opt_radii[i] - 1e-13)
    
    return opt_centers, opt_radii
