def run_packing():

    Packs 26 circles in a unit square to maximize the sum of radii.
    Uses a penalty-based optimization approach with L-BFGS-B.
    """
    n = 26
    best_sum_radii = -1.0
    best_centers = None
    best_radii = None
    
    # Number of random restarts to find a good global optimum
    n_restarts = 15

        row_counts = [5, 5, 6, 5, 5] # Sum = 26
        y_spacing = 0.9 / 4.0 # Spread over 0.1 to 0.9
        start_y = 0.1
        
        idx = 0
        for r, count in enumerate(row_counts):
            y = start_y + r * y_spacing
            # Spread x from 0.1 to 0.9
            if count > 0:
                xs = np.linspace(0.1, 0.9, count)
                for x in xs:
                    centers_init[idx, 0] = x
                    centers_init[idx, 1] = y
                    idx += 1
        
        # Add some random jitter to break symmetry and avoid exact grid locking
        jitter_scale = 0.02
        rng = np.random.RandomState(seed)
        centers_init += rng.uniform(-jitter_scale, jitter_scale, size=centers_init.shape)
        
        # Clip to valid range
        centers_init = np.clip(centers_init, 0.0, 1.0)

            s_r = np.sum(radii)

            cost += lam * (boundary_penalty + overlap_penalty)
            
            return cost

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square

    Args:
        centers: np.array of shape (n, 2) with (x, y) coordinates
        radii: np.array of shape (n) with radius of each circle

    Returns:
        True if valid, False otherwise
    """
    import numpy as np
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True
