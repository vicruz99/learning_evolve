    *   **Evolution:** Run the PSO algorithm for 200 iterations. Particles update their positions (centers and radii) based on their own best known position and the global best position found by the swarm.

    *   Take the best configuration found by the PSO and use `scipy.optimize.minimize` with the SLSQP method for a high-precision local search.

import numpy as np
from scipy.optimize import minimize
from scipy.optimize import differential_evolution

def get_bounds():
    """Define bounds for the optimization variables: [x1, y1, r1, ..., x26, y26, r26]"""
    # x, y in [0, 1], r in [0, 0.5]
    bnds = []
    for _ in range(26):
        bnds.append((0.0, 1.0)) # x
        bnds.append((0.0, 1.0)) # y
        bnds.append((0.0, 0.5)) # r
    return bnds

def get_constraints(centers, radii):
    """Calculate overlap and boundary penalty for a given configuration."""
    n = 26
    penalty = 0.0
    
    # Check boundaries
    for i in range(n):
        x, y, r = centers[i, 0], centers[i, 1], radii[i]
        # Boundary constraints: 0 <= x-r and x+r <= 1
        if x - r < 0: penalty += (x - r) ** 2
        if x + r > 1: penalty += (x + r - 1) ** 2
        if y - r < 0: penalty += (y - r) ** 2
        if y + r > 1: penalty += (y + r - 1) ** 2

    # Check overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt((centers[i, 0] - centers[j, 0])**2 + (centers[i, 1] - centers[j, 1])**2)
            min_dist = radii[i] + radii[j]
            if dist < min_dist:
                penalty += (min_dist - dist) ** 2
                
    return penalty

def objective_function(params):
    """
    Objective function for optimization.
    We want to maximize sum of radii, so we minimize -sum(radii) + penalties.
    """
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Unpack params
    idx = 0
    for i in range(n):
        centers[i, 0] = params[idx]     # x
        centers[i, 1] = params[idx+1]   # y
        radii[i] = params[idx+2]        # r
        idx += 3
        
    penalty = get_constraints(centers, radii)
    
    # Objective: -sum(radii) + penalty
    # We use a weight for penalty to ensure constraints are met.
    # In the global phase, we rely on the penalty being large enough.
    return -np.sum(radii) + penalty * 1000.0

def run_packing():
    n = 26
    num_vars = n * 3
    
    # Phase 1: Global Optimization using Differential Evolution
    # DE is robust for non-convex problems and avoids local minima better than PSO
    # when implemented simply.
    
    bounds = get_bounds()
    
    # We minimize the objective function.
    # Strategy: Use a wrapper to handle the constraint checking inside the objective
    # for the global search, then refine locally.
    
    best_res = None
    
    # Differential Evolution parameters
    # popsize: Number of individuals in the population
    # maxiter: Maximum number of iterations
    # tol: Relative error for convergence
    # mutation: Mutation constant
    # recombination: Recombination constant
    # seed: Random seed for reproducibility
    
    # Run DE
    result_de = differential_evolution(
        objective_function, 
        bounds, 
        popsize=20, 
        maxiter=200, 
        tol=1e-7, 
        mutation=(0.5, 1), 
        recombination=0.7, 
        seed=42,
        polish=True # Try to polish the result at the end
    )
    
    # Extract best parameters from DE
    best_params = result_de.x
    
    # Unpack best_params into centers and radii
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    idx = 0
    for i in range(n):
        centers[i, 0] = best_params[idx]
        centers[i, 1] = best_params[idx+1]
        radii[i] = best_params[idx+2]
        idx += 3
        
    # Phase 2: Local Refinement using SLSQP with explicit constraints
    # This ensures strict adherence to geometric constraints
    
    # Define constraints for SLSQP
    cons = []
    
    # Boundary constraints: x >= r, x <= 1-r, y >= r, y <= 1-r
    # Inequality constraints are defined as: cons >= 0
    # x - r >= 0  => x - r >= 0
    # 1 - r - x >= 0 => 1 - x - r >= 0
    # y - r >= 0
    # 1 - r - y >= 0
    
    for i in range(n):
        # x_i >= r_i  => x_i - r_i >= 0
        def make_con_x_lower(i):
            return lambda params, idx=i: params[idx] - params[idx+2]
        cons.append({'type': 'ineq', 'fun': make_con_x_lower(i)})
        
        # 1 - x_i - r_i >= 0
        def make_con_x_upper(i):
            return lambda params, idx=i: 1.0 - params[idx] - params[idx+2]
        cons.append({'type': 'ineq', 'fun': make_con_x_upper(i)})
        
        # y_i >= r_i  => y_i - r_i >= 0
        def make_con_y_lower(i):
            return lambda params, idx=i: params[idx+1] - params[idx+2]
        cons.append({'type': 'ineq', 'fun': make_con_y_lower(i)})
        
        # 1 - y_i - r_i >= 0
        def make_con_y_upper(i):
            return lambda params, idx=i: 1.0 - params[idx+1] - params[idx+2]
        cons.append({'type': 'ineq', 'fun': make_con_y_upper(i)})

    # Overlap constraints: dist >= r_i + r_j
    # sqrt((xi-xj)^2 + (yi-yj)^2) >= ri + rj
    # (xi-xj)^2 + (yi-yj)^2 >= (ri + rj)^2
    # (xi-xj)^2 + (yi-yj)^2 - (ri + rj)^2 >= 0
    
    for i in range(n):
        for j in range(i + 1, n):
            idx_i = i * 3
            idx_j = j * 3
            
            def make_con_overlap(i, j):
                idx_i = i * 3
                idx_j = j * 3
                def fun(params):
                    xi, yi, ri = params[idx_i], params[idx_i+1], params[idx_i+2]
                    xj, yj, rj = params[idx_j], params[idx_j+1], params[idx_j+2]
                    dist_sq = (xi - xj)**2 + (yi - yj)**2
                    rad_sum_sq = (ri + rj)**2
                    return dist_sq - rad_sum_sq
                return fun
            cons.append({'type': 'ineq', 'fun': make_con_overlap(i, j)})
            
    # Run local optimization
    # Objective for SLSQP: minimize -sum(radii)
    def obj_slsqp(params):
        sum_r = 0
        for i in range(n):
            sum_r += params[i*3 + 2]
        return -sum_r

    try:
        result_slsqp = minimize(
            obj_slsqp, 
            best_params, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 1000, 'ftol': 1e-12}
        )
        if result_slsqp.success:
            best_params = result_slsqp.x
    except Exception:
        pass # Fallback to DE result if SLSQP fails

    # Final unpacking
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    idx = 0
    for i in range(n):
        centers[i, 0] = best_params[idx]
        centers[i, 1] = best_params[idx+1]
        radii[i] = best_params[idx+2]
        idx += 3
        
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
