from scipy.optimize import minimize

def run_packing():

    best_centers = None
    best_radii = None

    # Helper function to calculate penalties for constraints
    def calculate_loss(vars_flat):
        centers = vars_flat[:2*n].reshape((n, 2))
        radii = vars_flat[2*n:]
        
        loss = 0.0
        
        # Penalty for radii being too small or large (soft bounds)
        # We want radii to be large, but constrained by geometry.
        # The objective is handled in optimization, but we need to penalize invalid states heavily.
        
        # Boundary penalties
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < 0:
                loss += 1000.0 * (x - r) ** 2
            if x + r > 1:
                loss += 1000.0 * (x + r - 1) ** 2
            if y - r < 0:
                loss += 1000.0 * (y - r) ** 2
            if y + r > 1:
                loss += 1000.0 * (y + r - 1) ** 2
            
            # Penalty for negative radii
            if r < 0:
                loss += 1000.0 * r ** 2
        
        # Overlap penalties
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                min_dist = radii[i] + radii[j]
                if dist < min_dist:
                    loss += 1000.0 * (min_dist - dist) ** 2
                    
        return loss

    # Objective function: Maximize sum of radii => Minimize negative sum
    def objective(vars_flat):
        radii = vars_flat[2*n:]
        return -np.sum(radii) + calculate_loss(vars_flat)
