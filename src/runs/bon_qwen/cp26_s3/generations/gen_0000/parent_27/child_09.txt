Therefore, we employ a strategy of unequal radii: we keep most circles large (slightly less than 0.1) to maintain a high sum, but shrink specific circles to accommodate the 26th circle in the resulting gaps.

2.  **Gradient-Based Optimization**: We use `scipy.optimize.minimize` (SLSQP method) to maximize the sum of radii. The solver will iteratively expand radii while strictly respecting boundary and non-overlap constraints.

import numpy as np
import math
from scipy.optimize import minimize

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    n = centers.shape[0]
    if np.isnan(centers).any() or np.isnan(radii).any():
        return False
    for i in range(n):
        if radii[i] < 0:
            return False
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            return False
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False
    return True

def run_packing():
    n = 26
    best_centers = None
    best_radii = None
    best_sum = 0.0

    # Base configuration: 5x5 grid
    grid_coords = np.array([[0.1 + 0.2*i, 0.1 + 0.2*j] for i in range(5) for j in range(5)])
    
    # Create initial 26-circle configuration (25 grid + 1 center gap)
    # Shrink slightly to ensure initial validity
    centers_0 = np.vstack([grid_coords, [0.5, 0.5]])
    radii_0 = np.full(n, 0.09) 

    # Helper for optimization
    def optimization_problem(variables):
        centers = variables[:2*n].reshape((n, 2))
        radii = variables[2*n:]
        
        # Objective: maximize sum of radii (minimize negative sum)
        objective_val = -np.sum(radii)
        
        constraints = []
        
        # Boundary constraints
        for i in range(n):
            constraints.append(centers[i, 0] - radii[i])       # x >= r
            constraints.append(1.0 - (centers[i, 0] + radii[i])) # x <= 1-r
            constraints.append(centers[i, 1] - radii[i])       # y >= r
            constraints.append(1.0 - (centers[i, 1] + radii[i])) # y <= 1-r

        # Non-overlap constraints
        for i in range(n):
            for j in range(i + 1, n):
                dist = math.sqrt((centers[i, 0] - centers[j, 0])**2 + 
                                 (centers[i, 1] - centers[j, 1])**2)
                constraints.append(dist - (radii[i] + radii[j]))

        # Aggregate constraints using minimum to keep array size small
        # If any constraint is violated (negative), min will be negative
        if constraints:
            min_constraint = np.min(constraints)
            # Scale violation to impact objective if violated
            # We use a soft penalty here, but SLSQP handles hard constraints better.
            # However, for performance, we return min constraint value.
            return objective_val + 1000.0 * min(0, min_constraint)
        
        return objective_val

    def objective_hard(variables):
        radii = variables[2*n:]
        return -np.sum(radii)

    def constraint_hard(variables):
        centers = variables[:2*n].reshape((n, 2))
        radii = variables[2*n:]
        con_list = []
        for i in range(n):
            con_list.append(centers[i, 0] - radii[i])
            con_list.append(1.0 - (centers[i, 0] + radii[i]))
            con_list.append(centers[i, 1] - radii[i])
            con_list.append(1.0 - (centers[i, 1] + radii[i]))
        for i in range(n):
            for j in range(i + 1, n):
                dist = math.sqrt((centers[i, 0] - centers[j, 0])**2 + 
                                 (centers[i, 1] - centers[j, 1])**2)
                con_list.append(dist - (radii[i] + radii[j]))
        return np.array(con_list)

    # Run multiple optimization trials from perturbed starts
    np.random.seed(42)
    for trial in range(5):
        # Perturb initial configuration
        if trial == 0:
            init_centers = centers_0.copy()
        else:
            noise = np.random.uniform(-0.05, 0.05, size=(n, 2))
            init_centers = np.clip(centers_0 + noise, 0.01, 0.99)
        
        init_radii = radii_0.copy()
        init_radii[-1] = 0.05 # Keep center circle small initially
        
        x0 = np.concatenate([init_centers.flatten(), init_radii])
        
        # Bounds: x, y in [0, 1], r in [0, 1]
        bounds = []
        for _ in range(n):
            bounds.extend([(0, 1), (0, 1)]) # x, y
            bounds.append((0, 1))            # r

        cons = ({'type': 'ineq', 'fun': constraint_hard})

        try:
            res = minimize(objective_hard, x0, method='SLSQP', bounds=bounds, 
                          constraints=cons, options={'maxiter': 1000, 'ftol': 1e-9})
            
            res_centers = res.x[:2*n].reshape((n, 2))
            res_radii = res.x[2*n:]
            
            if validate_packing(res_centers, res_radii):
                current_sum = np.sum(res_radii)
                if current_sum > best_sum:
                    best_sum = current_sum
                    best_centers = res_centers
                    best_radii = res_radii
        except Exception:
            continue

    # Final fallback if optimization failed
    if best_centers is None:
        return centers_0, radii_0, np.sum(radii_0)

    # Sort circles by radius descending for consistency
    sort_idx = np.argsort(-best_radii)
    best_centers = best_centers[sort_idx]
    best_radii = best_radii[sort_idx]
    best_sum = np.sum(best_radii)

    return best_centers, best_radii, best_sum

# Run the function to get the result
