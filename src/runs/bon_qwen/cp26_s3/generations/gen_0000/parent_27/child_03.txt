import numpy as np
from scipy.optimize import minimize

def compute_min_sep(centers):
    """Compute the maximum feasible equal radius given center positions."""
    n = centers.shape[0]
    x = centers[:, 0]
    y = centers[:, 1]
    
    # Distance to walls
    d_wall = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
    min_wall = np.min(d_wall)
    
    # Pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    min_pair = np.min(dists)
    
    # Radius is limited by half the smallest pairwise distance or wall distance
    return min(min_wall, min_pair / 2.0)

def objective(params):
    """Objective to minimize: negative of max feasible radius, plus boundary penalties."""
    centers = params.reshape(26, 2)
    penalty = 0.0
    for i in range(26):
        x, y = centers[i]
        if x < 0: penalty += 1000.0 * x**2
        elif x > 1: penalty += 1000.0 * (x - 1)**2
        if y < 0: penalty += 1000.0 * y**2
        elif y > 1: penalty += 1000.0 * (y - 1)**2
        
    if penalty > 1e-6:
        return 1e6
        
    return -compute_min_sep(centers)

def run_packing():
    best_R = -1.0
    best_centers = None
    
    # 1. Structured start: 5x5 grid + center

            grid[i*5 + j] = [0.1 + 0.2*i, 0.1 + 0.2*j]
    grid[25] = [0.5, 0.5]
    inits = [grid]

    np.random.seed(42)
    for _ in range(8):
        inits.append(np.random.rand(26, 2) * 0.8 + 0.1)
        
    for init_pts in inits:
        try:
            # Clamp to safe region to avoid initial penalties
            init_clamped = np.clip(init_pts, 0.01, 0.99)
            res = minimize(objective, init_clamped.flatten(), method='Nelder-Mead', 
                          options={'maxiter': 5000, 'xatol': 1e-7, 'fatol': 1e-7})
            
            # Only accept if optimization didn't get stuck in penalty region
            if res.fun < 1e5:
                R = -res.fun

                    best_R = R
                    best_centers = res.x.reshape(26, 2)
        except Exception:

    if best_centers is None:
        best_centers = grid
        best_R = 0.09
        
    # Strictly enforce bounds
    best_centers = np.clip(best_centers, 0.0, 1.0)

    R_safe = max(0.0, best_R * 0.9995)
    radii = np.full(26, R_safe)
    
    return best_centers, radii, np.sum(radii)
