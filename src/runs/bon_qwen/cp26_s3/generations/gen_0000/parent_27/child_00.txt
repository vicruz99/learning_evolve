def simulate_packing(N: int) -> tuple[np.ndarray, np.ndarray]:

    Simulates the growth of N circles using force-directed layout.
    """

    centers = np.random.rand(N, 2) * 0.6 + 0.2  # Start in center to avoid immediate boundary issues

    repulsion_strength = 10.0
    growth_rate = 0.0005
    damping = 0.95

            # Top wall
