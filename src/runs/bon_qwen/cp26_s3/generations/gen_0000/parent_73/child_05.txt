import numpy as np
from scipy.optimize import linprog


def solve_radii_lp(centers):
    """Given fixed centers, solve LP to maximize sum of radii."""
    n = centers.shape[0]
    c_obj = -np.ones(n)  # Maximize sum(r_i)
    
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        x, y = centers[i]
        for val in [x, 1.0 - x, y, 1.0 - y]:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(val)
    
    # Pairwise constraints: r_i + r_j <= ||c_i - c_j||
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dist)
    
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)] * n
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x
    else:
        # Fallback: small equal radii
        return np.ones(n) * 0.01


def optimize_centers_force(centers, radii, n_iter=300, lr_init=0.005):
    """Optimize centers using force-directed repulsion."""
    n = centers.shape[0]
    centers = centers.copy()
    lr = lr_init
    
    for it in range(n_iter):
        forces = np.zeros_like(centers)
        
        # Boundary forces: push circles away from boundaries
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            margin = 1e-6
            # Left boundary
            if x < r + margin:
                forces[i, 0] += 10.0 * (r + margin - x)
            # Right boundary
            if x > 1.0 - r - margin:
                forces[i, 0] -= 10.0 * (x - (1.0 - r - margin))
            # Bottom boundary
            if y < r + margin:
                forces[i, 1] += 10.0 * (r + margin - y)
            # Top boundary
            if y > 1.0 - r - margin:
                forces[i, 1] -= 10.0 * (y - (1.0 - r - margin))
        
        # Pairwise repulsive forces
        for i in range(n):
            for j in range(i + 1, n):
                diff = centers[j] - centers[i]
                dist = np.linalg.norm(diff)
                desired = radii[i] + radii[j]
                
                if dist > 1e-10:
                    if dist < desired + 0.15:
                        # Repulsion strength increases as circles get closer
                        overlap = desired - dist
                        if overlap > 0:
                            strength = 50.0 * overlap
                            direction = diff / dist
                            forces[i] -= strength * direction
                            forces[j] += strength * direction
                        else:
                            # Small repulsion to maintain separation
                            strength = 2.0 * (0.15 - overlap)
                            direction = diff / dist
                            forces[i] -= strength * direction
                            forces[j] += strength * direction
        
        centers += lr * forces
        # Clamp centers to valid region
        for i in range(n):
            centers[i, 0] = max(radii[i], min(1.0 - radii[i], centers[i, 0]))
            centers[i, 1] = max(radii[i], min(1.0 - radii[i], centers[i, 1]))
        
        # Decay learning rate
        if it % 30 == 0 and it > 0:
            lr *= 0.92
    
    return centers


def initialize_hexagonal(n=26):
    """Initialize centers in a hexagonal pattern."""
    centers = []
    # Pattern: 6 + 5 + 6 + 5 + 4 = 26 circles
    row_counts = [6, 5, 6, 5, 4]
    r_est = 0.085
    
    for row_idx, count in enumerate(row_counts):
        y = r_est + row_idx * np.sqrt(3) * r_est
        offset = (row_idx % 2) * r_est
        
        if count > 1:
            spacing = (1.0 - 2.0 * r_est - offset) / (count - 1)
        else:
            spacing = 0
        
        for col in range(count):
            if len(centers) >= n:
                break
            x = r_est + offset + col * spacing
            centers.append([x, y])
    
    return np.array(centers[:n])


def run_single_trial(init_centers, n_iterations=40):
    """Run alternating optimization from given initial centers."""
    n = init_centers.shape[0]
    centers = init_centers.copy()
    radii = np.ones(n) * 0.05
    
    for iteration in range(n_iterations):
        # Step 1: Solve LP for optimal radii
        radii = solve_radii_lp(centers)
        
        # Ensure radii are valid
        radii = np.maximum(radii, 1e-8)
        
        # Step 2: Optimize centers with force-directed method
        n_force_iter = max(50, 200 - iteration * 3)
        lr = 0.008 / (1 + iteration * 0.05)
        centers = optimize_centers_force(centers, radii, n_iter=n_force_iter, lr_init=lr)
    
    return centers, radii


def run_packing():
    """Main function to find optimal packing of 26 circles in unit square."""
    np.random.seed(42)
    n = 26
    
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Try multiple trials with different initial configurations
    for trial in range(25):
        # Initialize with hexagonal pattern
        centers = initialize_hexagonal(n)
        
        # Add random perturbation for diversity
        if trial > 0:
            sigma = 0.015 + trial * 0.002
            centers += np.random.normal(0, sigma, centers.shape)
            centers = np.clip(centers, 0.05, 0.95)
        
        # Run optimization
        opt_centers, opt_radii = run_single_trial(centers, n_iterations=40)
        
        current_sum = np.sum(opt_radii)
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = opt_centers.copy()
            best_radii = opt_radii.copy()
    
    # Final cleanup: ensure all circles are properly within bounds
    for i in range(n):
        x, y = best_centers[i]
        r = best_radii[i]
        best_centers[i, 0] = max(r, min(1.0 - r, x))
        best_centers[i, 1] = max(r, min(1.0 - r, y))
    
    # Re-solve LP one final time to ensure optimal radii
    best_radii = solve_radii_lp(best_centers)
    
    return best_centers, best_radii, np.sum(best_radii)
