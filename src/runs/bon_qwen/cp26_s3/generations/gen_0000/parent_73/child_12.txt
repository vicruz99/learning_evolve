import numpy as np
from scipy.optimize import minimize

def _objective(v, n):
    # We want to maximize sum of radii, so minimize negative sum
    return -np.sum(v[2*n:])

def _constraints(v, n):
    # Extract centers and radii from the flattened variable array
    c = v[:2*n].reshape(n, 2)
    r = v[2*n:]
    
    # Boundary constraints: circle inside [0,1]x[0,1]
    # x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
    b1 = c[:, 0] - r
    b2 = 1.0 - c[:, 0] - r
    b3 = c[:, 1] - r
    b4 = 1.0 - c[:, 1] - r
    bnds = np.concatenate([b1, b2, b3, b4])
    
    # Pairwise non-overlap constraints: dist >= r_i + r_j
    # Compute all pairwise distances efficiently using broadcasting
    diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    # We only need constraints for i < j to avoid duplicates and self-checks
    triu_idx = np.triu_indices(n, k=1)
    pairs = dists[triu_idx] - r[triu_idx[0]] - r[triu_idx[1]]
    
    # Return all inequality constraints as a single array (f(v) >= 0)
    return np.concatenate([bnds, pairs])

def run_packing():
    n = 26
    
    # Initialize centers on a perturbed grid to break symmetry and ensure distinct positions
    # This provides a good starting point for the optimizer and avoids degenerate starts
    xs = np.linspace(0.15, 0.85, 6)
    ys = np.linspace(0.15, 0.85, 5)
    centers = np.array([[x, y] for x in xs for y in ys])[:n]
    
    # Add small random perturbation
    rng = np.random.RandomState(42)
    centers += rng.uniform(-0.02, 0.02, centers.shape)
    centers = np.clip(centers, 0.05, 0.95)
    
    # Initial radii guess
    radii = np.full(n, 0.08)
    
    # Flatten variables: [x_0, y_0, ..., x_25, y_25, r_0, ..., r_25]
    x0 = np.concatenate([centers.flatten(), radii])
    
    # Define bounds for variables
    bnds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n
    
    # Define constraints
    cons = {'type': 'ineq', 'fun': _constraints, 'args': (n,)}
    
    # Run optimization using SLSQP
    # SLSQP is robust for medium-sized NLP problems with inequality constraints
    res = minimize(_objective, x0, method='SLSQP', bounds=bnds, constraints=cons, 
                   args=(n,), options={'maxiter': 5000, 'ftol': 1e-12, 'disp': False})
                   
    centers_opt = res.x[:2*n].reshape((n, 2))
    radii_opt = res.x[2*n:]
    
    # Post-processing: slightly shrink radii to guarantee strict validity against numerical tolerance
    # This handles any tiny floating-point violations from the optimizer while preserving the solution quality
    for i in range(n):
        r_max = min(centers_opt[i, 0], 1.0 - centers_opt[i, 0], 
                    centers_opt[i, 1], 1.0 - centers_opt[i, 1])
        for j in range(n):
            if i != j:
                d = np.sqrt(np.sum((centers_opt[i] - centers_opt[j])**2))
                r_max = min(r_max, d - radii_opt[j])
        radii_opt[i] = min(radii_opt[i], r_max * 0.9999999)
        
    return centers_opt, radii_opt, np.sum(radii_opt)
