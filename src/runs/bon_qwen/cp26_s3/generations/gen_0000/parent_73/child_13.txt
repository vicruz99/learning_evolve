import numpy as np
from scipy.optimize import minimize

def run_packing():
    """
    Solves the circle packing problem for 26 circles in a unit square.
    Returns (centers, radii, sum_radii).
    """
    n = 26
    num_vars = n * 3  # x, y, r for each circle

    # --- 1. Initialization: Hexagonal Grid ---
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Hexagonal parameters
    r_init = 0.09  # Starting radius, slightly loose to allow optimizer to grow them
    
    idx = 0
    y = r_init
    while idx < n:
        x = r_init
        while x < 1 - r_init and idx < n:
            centers[idx] = [x, y]
            radii[idx] = r_init
            idx += 1
            x += 2 * r_init
        
        y += np.sqrt(3) * r_init
        r_offset = r_init  # Horizontal offset for hexagonal packing
        x = r_init + r_offset
        while x < 1 - r_init and idx < n:
            centers[idx] = [x, y]
            radii[idx] = r_init
            idx += 1
            x += 2 * r_init
            
        if y > 1 - r_init:
            break

    # Flatten initial state: [x0, y0, r0, x1, y1, r1, ...]
    x0 = np.zeros(num_vars)
    for i in range(n):
        x0[3*i] = centers[i, 0]
        x0[3*i + 1] = centers[i, 1]
        x0[3*i + 2] = radii[i]

    # --- 2. Optimization Setup ---
    
    def objective(params):
        # Maximize sum of radii -> Minimize negative sum
        return -np.sum(params[2::3])

    def jacobian_objective(params):
        jac = np.zeros(num_vars)
        jac[2::3] = -1.0  # Derivative wrt r_i is -1
        return jac

    # Helper to get distance and derivatives
    def pair_constraint_dist(params, i, j):
        xi, yi, ri = params[3*i], params[3*i+1], params[3*i+2]
        xj, yj, rj = params[3*j], params[3*j+1], params[3*j+2]
        dx = xi - xj
        dy = yi - yj
        dist = np.sqrt(dx**2 + dy**2)
        return dist - (ri + rj)

    # We will group constraints for efficiency if possible, but SLSQP handles lists well.
    # To keep it clean and robust, we define a function that returns the vector of all constraints
    # and its Jacobian.
    
    # Number of pair constraints
    num_pairs = n * (n - 1) // 2
    # Number of boundary constraints (4 per circle)
    num_bounds = 4 * n
    total_constraints = num_pairs + num_bounds
    
    def constraints_func(params):
        c = np.zeros(total_constraints)
        k = 0
        
        # 1. Circle-Circle Non-overlap: dist(i,j) >= ri + rj  => dist - (ri+rj) >= 0
        for i in range(n):
            for j in range(i + 1, n):
                xi, yi = params[3*i], params[3*i+1]
                xj, yj = params[3*j], params[3*j+1]
                ri, rj = params[3*i+2], params[3*j+2]
                
                dx = xi - xj
                dy = yi - yj
                dist = np.sqrt(dx**2 + dy**2)
                c[k] = dist - (ri + rj)
                k += 1
                
        # 2. Boundary Constraints: 
        # x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
        for i in range(n):
            x, y, r = params[3*i], params[3*i+1], params[3*i+2]
            c[k] = x - r       # Left wall
            k += 1
            c[k] = 1.0 - x - r # Right wall
            k += 1
            c[k] = y - r       # Bottom wall
            k += 1
            c[k] = 1.0 - y - r # Top wall
            k += 1
            
        return c

    def jacobian_constraints(params):
        jac = np.zeros((total_constraints, num_vars))
        k = 0
        
        # 1. Circle-Circle Jacobian
        for i in range(n):
            for j in range(i + 1, n):
                xi, yi = params[3*i], params[3*i+1]
                xj, yj = params[3*j], params[3*j+1]
                ri, rj = params[3*i+2], params[3*j+2]
                
                dx = xi - xj
                dy = yi - yj
                dist = np.sqrt(dx**2 + dy**2)
                
                if dist > 1e-9:
                    d_dist_dx = dx / dist
                    d_dist_dy = dy / dist
                else:
                    d_dist_dx = 0.0
                    d_dist_dy = 0.0

                # Derivatives wrt xi, yi
                jac[k, 3*i] = d_dist_dx
                jac[k, 3*i + 1] = d_dist_dy
                # Derivatives wrt xj, yj
                jac[k, 3*j] = -d_dist_dx
                jac[k, 3*j + 1] = -d_dist_dy
                # Derivatives wrt ri, rj (from -(ri+rj))
                jac[k, 3*i + 2] = -1.0
                jac[k, 3*j + 2] = -1.0
                
                k += 1

        # 2. Boundary Jacobian
        for i in range(n):
            # c = x - r
            jac[k, 3*i] = 1.0
            jac[k, 3*i + 2] = -1.0
            k += 1
            
            # c = 1 - x - r
            jac[k, 3*i] = -1.0
            jac[k, 3*i + 2] = -1.0
            k += 1
            
            # c = y - r
            jac[k, 3*i + 1] = 1.0
            jac[k, 3*i + 2] = -1.0
            k += 1
            
            # c = 1 - y - r
            jac[k, 3*i + 1] = -1.0
            jac[k, 3*i + 2] = -1.0
            k += 1
            
        return jac

    # Define constraint dictionary
    cons = {
        'type': 'ineq',
        'fun': constraints_func,
        'jac': jacobian_constraints
    }

    # Bounds for variables
    # x, y in [0, 1]
    # r in [0, 0.5] (max possible radius in unit square)
    bounds = []
    for _ in range(n):
        bounds.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])

    # Run optimization
    # Using SLSQP which handles constraints well
    res = minimize(
        objective, 
        x0, 
        method='SLSQP', 
        bounds=bounds, 
        constraints=cons,
        jac=jacobian_objective,
        options={'maxiter': 1000, 'ftol': 1e-9, 'disp': False}
    )

    if not res.success:
        # If optimization failed, we might still have a valid solution in x0 or res.x
        # But let's try to return the best found
        pass

    best_params = res.x
    
    # Extract results
    final_centers = np.zeros((n, 2))
    final_radii = np.zeros(n)
    
    for i in range(n):
        final_centers[i, 0] = best_params[3*i]
        final_centers[i, 1] = best_params[3*i + 1]
        final_radii[i] = best_params[3*i + 2]
        
    # Clamp radii to be non-negative (just in case of numerical noise)
    final_radii = np.maximum(final_radii, 0.0)
    
    sum_radii = np.sum(final_radii)
    
    return final_centers, final_radii, sum_radii

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square

    Args:
        centers: np.array of shape (n, 2) with (x, y) coordinates
        radii: np.array of shape (n) with radius of each circle

    Returns:
        True if valid, False otherwise
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True

# Run and validate
if __name__ == "__main__":
    centers, radii, s = run_packing()
    is_valid = validate_packing(centers, radii)
    print(f"Valid: {is_valid}")
    print(f"Sum of radii: {s}")
