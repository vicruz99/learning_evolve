import numpy as np
from scipy.optimize import minimize

def packing_objective(params, n, beta):
    """
    Computes the objective value for circle packing.
    Minimizes negative sum of radii + penalty terms for overlaps and boundary violations.
    """
    c = params[:2*n].reshape(n, 2)
    r = params[2*n:]
    
    # Primary objective: maximize sum of radii
    obj = -np.sum(r)
    
    # Pairwise distances using broadcasting
    diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
    dist = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dist, np.inf)  # Ignore self-distance
    
    # Overlap penalty: circles should not intersect
    min_req = r[:, np.newaxis] + r[np.newaxis, :]
    overlap = np.maximum(0, min_req - dist)
    obj += beta * np.sum(overlap**2)
    
    # Boundary penalties: circles must stay within [0,1]^2
    # Left: x >= r  => r - x <= 0
    obj += beta * np.sum(np.maximum(0, r - c[:, 0])**2)
    # Right: x + r <= 1 => r - (1 - x) <= 0
    obj += beta * np.sum(np.maximum(0, r - (1 - c[:, 0]))**2)
    # Bottom: y >= r
    obj += beta * np.sum(np.maximum(0, r - c[:, 1])**2)
    # Top: y + r <= 1
    obj += beta * np.sum(np.maximum(0, r - (1 - c[:, 1]))**2)
    
    return obj

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # 1. Initialization: Hexagonal lattice pattern
    # Hexagonal packing is denser than square grid
    rows, cols = 5, 6
    xg, yg = np.meshgrid(np.linspace(0.0, 1.0, cols), np.linspace(0.0, 1.0, rows))
    xs = xg.copy()
    # Shift odd rows to create hexagonal offset
    for i in range(1, rows, 2):
        xs[i, :] += 0.5 / (cols - 1)
        
    pts = np.column_stack([xs.ravel(), yg.ravel()])[:n]
    # Center and scale to leave a safety margin
    pts = (pts - pts.min(axis=0)) / (pts.max(axis=0) - pts.min(axis=0))
    pts = pts * 0.82 + 0.09
    
    centers = pts
    radii = np.full(n, 0.045)  # Small initial radius to start feasible
    
    p0 = np.concatenate([centers.ravel(), radii])
    
    # Bounds: centers in [0,1], radii in [0, 0.5]
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
    
    # 2. Optimization: Iterative penalty increase
    # Gradually tighten constraints while expanding radii
    p = p0.copy()
    beta = 50.0
    for _ in range(9):
        res = minimize(packing_objective, p, args=(n, beta), method='L-BFGS-B',
                       bounds=bounds, options={'ftol': 1e-11, 'gtol': 1e-9, 'maxiter': 3000, 'disp': False})
        p = res.x
        beta *= 2.5  # Exponentially increase penalty weight
        if res.success:
            break
            
    centers = p[:2*n].reshape(n, 2)
    radii = p[2*n:]
    
    # Ensure strict non-negativity to satisfy validator
    radii = np.maximum(radii, 1e-6)
    
    return centers, radii, float(np.sum(radii))
