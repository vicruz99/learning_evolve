- Variables: $(x_i, y_i, r_i)$ for $i=1,\dots,26$.
- Objective: Maximize $\sum r_i$ (equivalently, minimize $-\sum r_i$).
- Constraints: 
  1. Boundary: $0 \le x_i - r_i$ and $x_i + r_i \le 1$ (similarly for $y$).

  3. Non-negativity: $r_i \ge 0$.

import numpy as np
from scipy.optimize import minimize

# Precompute pairwise indices for efficient constraint evaluation
_TRI_U_INDICES = np.triu_indices(26, k=1)

def _objective(z):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(z[52:])

def _constraints(z):
    """
    Compute constraint values that must be >= 0.
    Includes boundary constraints and non-overlap constraints.
    """
    n = 26
    xs = z[:n]
    ys = z[n:2*n]
    rs = z[2*n:]
    
    # Total constraints: 4*n boundary + n*(n-1)/2 overlaps
    c = np.empty(n*4 + n*(n-1)//2)
    
    # Boundary constraints
    c[:n] = xs - rs
    c[n:2*n] = 1.0 - xs - rs
    c[2*n:3*n] = ys - rs
    c[3*n:4*n] = 1.0 - ys - rs
    
    # Non-overlap constraints: dist(i,j) - r_i - r_j >= 0
    ii, jj = _TRI_U_INDICES
    dx = xs[ii] - xs[jj]
    dy = ys[ii] - ys[jj]
    dists = np.sqrt(dx*dx + dy*dy)
    c[4*n:] = dists - rs[ii] - rs[jj]
    
    return c

def run_packing():
    n = 26
    bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n
    cons = [{'type': 'ineq', 'fun': _constraints}]
    
    best_val = -np.inf
    best_sol = None
    
    # Multiple restarts to escape local minima
    for trial in range(15):
        np.random.seed(trial + 123)
        
        # Initialize with a hexagonal-like pattern for good density
        centers = []
        r0 = 0.09
        y = r0
        row = 0
        while len(centers) < n:
            count = 6 if row % 2 == 0 else 4
            needed = n - len(centers)
            if needed < count:
                count = needed
            x_start = r0 if row % 2 == 0 else 2*r0
            for i in range(count):
                centers.append([x_start + 2*i*r0, y])
            y += r0 * np.sqrt(3)
            row += 1
            
        centers = np.array(centers)
        radii = np.full(n, r0)
        
        # Add controlled random perturbation
        centers += np.random.uniform(-0.02, 0.02, centers.shape)
        radii += np.random.uniform(-0.01, 0.01, radii.shape)
        centers = np.clip(centers, 0.02, 0.98)
        radii = np.clip(radii, 0.02, 0.4)
        
        z0 = np.concatenate([centers[:, 0], centers[:, 1], radii])
        
        try:
            res = minimize(_objective, z0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 500, 'ftol': 1e-9, 'disp': False})
            val = -res.fun
            if val > best_val:
                best_val = val
                best_sol = res.x
        except Exception:
            continue

    if best_sol is None:
        return np.zeros((n, 2)), np.zeros(n), 0.0
        
    centers = best_sol[:2*n].reshape(n, 2)
    radii = best_sol[2*n:]
    return centers, radii, best_val
