import numpy as np
from scipy.optimize import minimize

def run_packing():
    n = 26
    
    # 1. Initialize with a hexagonal lattice pattern
    # This arrangement is near-optimal for dense circle packing.
    centers = []
    r_init = 0.085
    for i in range(6):
        # Number of circles per row in a staggered hexagonal pattern
        if i == 0: cnt = 5
        elif i == 1: cnt = 4
        elif i == 2: cnt = 5
        elif i == 3: cnt = 4
        elif i == 4: cnt = 5
        else: cnt = 3
        
        y = r_init + i * np.sqrt(3) * r_init
        if i % 2 == 0:
            x_start = r_init
        else:
            x_start = 2 * r_init
            
        for j in range(cnt):
            x = x_start + j * 2 * r_init
            centers.append([x, y])
            
    centers = np.array(centers)
    # Variables: [x0, y0, x1, y1, ..., x25, y25, r]
    x0 = np.concatenate([centers.flatten(), [r_init]])
    
    # Bounds for coordinates and radius
    bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)]
    
    # Precompute indices for pairwise constraints to speed up evaluation
    i_indices, j_indices = np.triu_indices(n, k=1)
    
    def objective(vars_):
        # Maximize radius r (minimize -r)
        return -vars_[-1]
        
    def constraints(vars_):
        coords = vars_[:2*n].reshape(-1, 2)
        r = vars_[-1]
        c = []
        
        # Boundary constraints: distance to each side must be >= r
        c.append(coords[:, 0] - r)          # x - r >= 0
        c.append(1.0 - coords[:, 0] - r)    # 1 - x - r >= 0
        c.append(coords[:, 1] - r)          # y - r >= 0
        c.append(1.0 - coords[:, 1] - r)    # 1 - y - r >= 0
        
        # Pairwise constraints: distance between centers must be >= 2r
        xi, yi = coords[i_indices, 0], coords[i_indices, 1]
        xj, yj = coords[j_indices, 0], coords[j_indices, 1]
        d2 = (xi - xj)**2 + **(yi - yj)2
        c.append(d2 - 4*r**2)
        
        return np.concatenate(c)
        
    cons = {'type': 'ineq', 'fun': constraints}
    
    # Run optimization
    try:
        res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, 
                       options={'maxiter': 3000, 'ftol': 1e-12, 'disp': False})
        best_x = res.x
    except Exception:
        best_x = x0
        
    centers_opt = best_x[:-1].reshape(-1, 2)
    r_opt = best_x[-1]
    radii_opt = np.full(n, r_opt)
    
    # Ensure strictly valid output within numerical tolerance
    # The optimizer should already satisfy constraints, but we clip just in case
    centers_opt = np.clip(centers_opt, 1e-9, 1.0 - 1e-9)
    
    return centers_opt, radii_opt, float(np.sum(radii_opt))
