import numpy as np
from scipy.optimize import minimize


def compute_penalty(params, n):
    centers = params[:2 * n].reshape(n, 2)
    radii = params[2 * n:]

    penalty = 0.0

    for i in range(n):
        x = centers[i, 0]
        y = centers[i, 1]
        r = radii[i]
        if r > x:
            penalty += (r - x) ** 2
        if r > 1 - x:
            penalty += (r - (1 - x)) ** 2
        if r > y:
            penalty += (r - y) ** 2
        if r > 1 - y:
            penalty += (r - (1 - y)) ** 2

    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            overlap = radii[i] + radii[j] - dist
            if overlap > 0:
                penalty += overlap ** 2

    return penalty


def objective(params, n, penalty_weight):
    radii = params[2 * n:]
    penalty = compute_penalty(params, n)
    return -np.sum(radii) + penalty_weight * penalty


def generate_hex_arrangement(rows, n):
    centers = np.zeros((n, 2))
    num_rows = len(rows)
    idx = 0

    for row_idx, count in enumerate(rows):
        y = (row_idx + 1) * (1.0 / (num_rows + 1))

        if row_idx % 2 == 1:
            x_start = 0.5 / (count + 1)
        else:
            x_start = 1.0 / (count + 1)

        x_spacing = 1.0 / (count + 1)
        for col in range(count):
            x = x_start + col * x_spacing
            centers[idx, 0] = x
            centers[idx, 1] = y
            idx += 1

    return centers


def optimize_from_initial(centers, n, initial_r=0.04):
    radii = np.ones(n) * initial_r
    x0 = np.concatenate([centers.flatten(), radii])

    bounds = [(0, 1)] * (2 * n) + [(0, 0.5)] * n

    penalty_weight = 50
    current_x = x0.copy()

    for iteration in range(100):
        result = minimize(objective, current_x, args=(n, penalty_weight),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': 5000, 'ftol': 1e-16, 'gtol': 1e-10})
        current_x = result.x
        penalty_weight *= 1.25

    final_centers = current_x[:2 * n].reshape(n, 2)
    final_radii = current_x[2 * n:]
    return final_centers, final_radii


def relax_positions(centers, radii, n, num_iters=2000, step=0.005):
    centers = centers.copy()
    radii = radii.copy()

    for _ in range(num_iters):
        max_moved = 0
        for i in range(n):
            grad = np.array([0.0, 0.0])

            for j in range(n):
                if i == j:
                    continue
                diff = centers[i] - centers[j]
                dist_sq = np.sum(diff ** 2)
                dist = np.sqrt(dist_sq)
                if dist < radii[i] + radii[j] and dist > 1e-10:
                    overlap = radii[i] + radii[j] - dist
                    grad += overlap * diff / dist

            r = radii[i]
            if centers[i, 0] < r:
                grad[0] += (r - centers[i, 0]) * 200
            elif centers[i, 0] > 1 - r:
                grad[0] -= (centers[i, 0] - (1 - r)) * 200
            if centers[i, 1] < r:
                grad[1] += (r - centers[i, 1]) * 200
            elif centers[i, 1] > 1 - r:
                grad[1] -= (centers[i, 1] - (1 - r)) * 200

            centers[i] -= step * grad
            centers[i, 0] = max(r + 1e-10, min(1 - r - 1e-10, centers[i, 0]))
            centers[i, 1] = max(r + 1e-10, min(1 - r - 1e-10, centers[i, 1]))

            moved = np.linalg.norm(step * grad)
            if moved > max_moved:
                max_moved = moved

        if max_moved < 1e-10:
            break

    return centers


def try_scale_up(centers, radii, n):
    """Try to uniformly scale up radii while maintaining validity"""
    centers = centers.copy()
    radii = radii.copy()

    for _ in range(500):
        radii *= 1.001
        centers = relax_positions(centers, radii, n, num_iters=500, step=0.003)
        penalty = compute_penalty(np.concatenate([centers.flatten(), radii]), n)
        if penalty > 1e-6:
            radii /= 1.001
            break

    return centers, radii
