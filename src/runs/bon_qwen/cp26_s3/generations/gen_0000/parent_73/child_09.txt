import numpy as np
import scipy.optimize as opt

def run_packing():
    n = 26
    
    def evaluate(centers):
        """
        Given centers, calculates optimal radii and sum of radii.
        Also returns a penalty term if constraints are violated.
        """
        radii = np.ones(n)
        # Boundary constraints
        dists_to_bound = np.column_stack([
            centers[:, 0], 
            1 - centers[:, 0], 
            centers[:, 1], 
            1 - centers[:, 1]
        ])
        radii = np.minimum(radii, dists_to_bound.min(axis=1))
        
        # Neighbor constraints
        # For each circle i, r_i <= (dist(i,j) - r_j) for all j
        # We solve this iteratively or via a simple relaxation as it's a graph problem
        # Simple iterative relaxation for r_i
        for _ in range(10): # Iterate to propagate constraints
            r_prev = radii.copy()
            for i in range(n):
                # Calculate min radius allowed by neighbors
                min_r = radii[i] # start with current or boundary limit
                for j in range(n):
                    if i == j:
                        continue
                    dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                    r_ij = (dist - radii[j]) / 2.0
                    if r_ij < min_r:
                        min_r = r_ij
                
                # Apply boundary constraint as well
                b_min = min(centers[i, 0], 1-centers[i, 0], centers[i, 1], 1-centers[i, 1])
                min_r = min(min_r, b_min)
                radii[i] = max(0, min_r)
        
        return radii

    def objective(centers_flat):
        centers = centers_flat.reshape((n, 2))
        radii = evaluate(centers)
        
        # Objective: Negative sum of radii (we minimize)
        obj = -np.sum(radii)
        
        # Penalty for negative radii
        if np.any(radii < 0):
            obj += 1e6 * np.sum(np.maximum(0, -radii))
            
        return obj

    best_obj = np.inf
    best_centers = None
    
    # Initial guesses
    guesses = []
    
    # 1. Grid initialization (5x5 + 1)
    x = np.linspace(0, 1, 6)[:5]
    y = np.linspace(0, 1, 6)[:5]

        guesses.append(r_centers.flatten())

    # 3. Hexagonal packing initialization
    # 6 rows with 5 or 4 circles
    h_centers = []
    rows = [5, 6, 5, 6, 4] # Sum 26
    # Scale factor to fit roughly

    while len(h_centers) < n:

        res = opt.minimize(
            objective, 
            guess, 
            method='L-BFGS-B',

        )
        
        if res.fun < best_obj:
            best_obj = res.fun
            best_centers = res.x.reshape((n, 2))

    # Final validation and radius calculation
    final_radii = evaluate(best_centers)
    sum_radii = np.sum(final_radii)
    
    return best_centers, final_radii, sum_radii
