import numpy as np
from scipy.optimize import minimize

def _objective(X):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(X[2::3])

def _constr_boundary(X):
    """Boundary constraints: circles must stay inside [0,1]^2."""
    x = X[0::3]
    y = X[1::3]
    r = X[2::3]
    # x >= r, x <= 1-r, y >= r, y <= 1-r
    return np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])

def _constr_overlap(X, i_idx, j_idx):
    """Non-overlap constraints: distance >= sum of radii."""
    x = X[0::3]
    y = X[1::3]
    r = X[2::3]
    dx = x[i_idx] - x[j_idx]
    dy = y[i_idx] - y[j_idx]
    dr = r[i_idx] + r[j_idx]
    dist = np.sqrt(dx**2 + dy**2)
    return dist - dr

def _generate_start(n, seed):
    """Generate initial configuration using a perturbed hexagonal layout."""
    np.random.seed(seed)
    pts = []
    # Hexagonal-ish layout encourages dense initial packing
    for row in range(6):
        y = 0.12 + row * 0.15
        cols = 5 if row % 2 == 0 else 6
        for col in range(cols):
            x = 0.12 + col * 0.15 + (0.075 if row % 2 != 0 else 0.0)
            pts.append([x, y])
        if len(pts) >= n:
            break
            
    pts = np.array(pts[:n])
    if len(pts) < n:
        extra = np.random.rand(n - len(pts), 2) * 0.7 + 0.15
        pts = np.vstack([pts, extra])
        
    X = np.zeros(3 * n)
    X[0::3] = pts[:, 0]
    X[1::3] = pts[:, 1]
    X[2::3] = 0.05 + 0.01 * np.random.rand(n)
    return X

def _validate_and_fix(X, n, i_idx, j_idx):
    """Post-process solution to guarantee strict feasibility."""
    x = X[0::3].copy()
    y = X[1::3].copy()
    r = X[2::3].copy()
    
    # Enforce boundary constraints strictly
    r = np.minimum(r, x)
    r = np.minimum(r, 1.0 - x)
    r = np.minimum(r, y)
    r = np.minimum(r, 1.0 - y)
    
    # Iteratively resolve overlaps by shrinking radii
    for _ in range(50):
        dx = x[i_idx] - x[j_idx]
        dy = y[i_idx] - y[j_idx]
        dist = np.sqrt(dx**2 + dy**2)
        dr = r[i_idx] + r[j_idx]
        overlaps = dr - dist
        
        if np.all(overlaps <= 1e-9):
            break
            
        max_idx = np.argmax(overlaps)
        overlap_val = overlaps[max_idx]
        if overlap_val <= 0:
            break
            
        i, j = i_idx[max_idx], j_idx[max_idx]
        shrink = overlap_val / 2.0 + 1e-9
        r[i] = max(0.0, r[i] - shrink)
        r[j] = max(0.0, r[j] - shrink)
        
    # Re-check boundaries after shrinking
    r = np.minimum(r, x)
    r = np.minimum(r, 1.0 - x)
    r = np.minimum(r, y)
    r = np.minimum(r, 1.0 - y)
    
    X[2::3] = r
    return X

def run_packing():
    n = 26
    i_idx, j_idx = np.tril_indices(n, -1)
    
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)] * n + [(0.0, 0.5)] * n
    
    best_X = None
    best_sum = -np.inf
    
    constraints = [
        {'type': 'ineq', 'fun': _constr_boundary},
        {'type': 'ineq', 'fun': _constr_overlap, 'args': (i_idx, j_idx)}
    ]
    
    # Run optimization from multiple starts to find global optimum
    for seed in range(5):
        X0 = _generate_start(n, seed)
        try:
            res = minimize(_objective, X0, method='SLSQP', bounds=bounds,
                           constraints=constraints,
                           options={'maxiter': 2000, 'ftol': 1e-9, 'disp': False})
            
            # Ensure strict feasibility
            X_fixed = _validate_and_fix(res.x, n, i_idx, j_idx)
            current_sum = np.sum(X_fixed[2::3])
            
            if current_sum > best_sum:
                best_sum = current_sum
                best_X = X_fixed.copy()
        except Exception:
            continue
            
    if best_X is None:
        best_X = _validate_and_fix(_generate_start(n, 0), n, i_idx, j_idx)
        
    centers = np.column_stack((best_X[0::3], best_X[1::3]))
    radii = best_X[2::3]
    return centers, radii, float(np.sum(radii))
