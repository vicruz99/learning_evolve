import numpy as np
from scipy.optimize import differential_evolution
import math

def compute_min_dist(centers):
    """
    Computes the minimum distance between any pair of circles or between a circle and the boundary.
    """
    n = centers.shape[0]
    
    # Distance to boundaries
    x_coords = centers[:, 0]
    y_coords = centers[:, 1]
    dist_to_boundary = np.minimum(
        np.minimum(x_coords, 1.0 - x_coords),
        np.minimum(y_coords, 1.0 - y_coords)
    )
    min_b = np.min(dist_to_boundary)
    
    # Distance between circles
    # Compute pairwise distances efficiently
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    min_c = np.min(dists)
    
    return min(min_b, min_c)

def objective(params):
    """
    Objective function for optimization.
    We want to maximize the minimum separation, so we minimize its negative.
    """
    centers = params.reshape(26, 2)
    return -compute_min_dist(centers)

def run_packing():
    """
    Optimizes the positions of 26 circles to maximize the sum of their radii.
    Assumes equal radii for simplicity and effectiveness, which is near-optimal for this problem.
    """
    N = 26
    bounds = [(0.0, 1.0)] * (2 * N)
