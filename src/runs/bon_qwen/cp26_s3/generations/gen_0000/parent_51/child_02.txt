import numpy as np
from scipy.optimize import minimize


def boundary_x_min(x_vec, i, n):
    """Circle i must satisfy: x_i - r_i >= 0"""
    return x_vec[2 * i] - x_vec[2 * n + i]


def boundary_x_max(x_vec, i, n):
    """Circle i must satisfy: x_i + r_i <= 1"""
    return 1.0 - x_vec[2 * i] - x_vec[2 * n + i]


def boundary_y_min(x_vec, i, n):
    """Circle i must satisfy: y_i - r_i >= 0"""
    return x_vec[2 * i + 1] - x_vec[2 * n + i]


def boundary_y_max(x_vec, i, n):
    """Circle i must satisfy: y_i + r_i <= 1"""
    return 1.0 - x_vec[2 * i + 1] - x_vec[2 * n + i]


def non_overlap(x_vec, i, j, n):
    """Circles i and j must not overlap: dist >= r_i + r_j"""
    xi = x_vec[2 * i]
    yi = x_vec[2 * i + 1]
    xj = x_vec[2 * j]
    yj = x_vec[2 * j + 1]
    ri = x_vec[2 * n + i]
    rj = x_vec[2 * n + j]
    dx = xi - xj
    dy = yi - yj
    dist = np.sqrt(dx * dx + dy * dy)
    return dist - ri - rj


def objective_func(x_vec, n):
    """Negative sum of radii (we minimize this)"""
    return -np.sum(x_vec[2 * n:])


def build_constraints(n):
    """Build all constraints for n circles"""
    constraints = []
    for i in range(n):
        constraints.append({'type': 'ineq', 'fun': boundary_x_min, 'args': (i, n)})
        constraints.append({'type': 'ineq', 'fun': boundary_x_max, 'args': (i, n)})
        constraints.append({'type': 'ineq', 'fun': boundary_y_min, 'args': (i, n)})
        constraints.append({'type': 'ineq', 'fun': boundary_y_max, 'args': (i, n)})
    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({'type': 'ineq', 'fun': non_overlap, 'args': (i, j, n)})
    return constraints


def make_bounds(n):
    """Build bounds for all variables"""
    bounds = []
    for i in range(2 * n):
        bounds.append((0.0, 1.0))
    for i in range(n):
        bounds.append((0.0, 0.5))
    return bounds


def init_grid_plus_extra(n):
    """Initialize with 5x5 grid + extra circles, adjusted for n"""
    centers = []
    radii_list = []
    r = 0.09
    
    # Fill rows with varying counts to get exactly n circles
    row_sizes = [6, 5, 6, 5, 4]  # Total: 26
    idx = 0
    for row in range(5):
        if idx >= n:
            break
        n_in_row = min(row_sizes[row], n - idx)
        y = r + row * r * 1.6  # Slightly compressed vertical spacing
        for col in range(n_in_row):
            x = r + col * 2 * r + (r if row % 2 == 1 else 0)
            centers.append([x, y])
            radii_list.append(r)
            idx += 1
    
    centers = np.array(centers[:n])
    radii_list = radii_list[:n]
    return centers, np.array(radii_list)


def init_staggered(n):
    """Initialize with staggered/hexagonal packing"""
    centers = []
    radii_list = []
    r = 0.085
    idx = 0
    
    for row in range(6):
        if idx >= n:
            break
        if row % 2 == 0:
            n_in_row = 5
            x_offset = r
        else:
            n_in_row = 4
            x_offset = 2 * r
        y = r + row * r * np.sqrt(3)
        actual_n = min(n_in_row, n - idx)
        for col in range(actual_n):
            x = x_offset + col * 2 * r
            if x + r <= 1.0:
                centers.append([x, y])
                radii_list.append(r)
                idx += 1
        if idx >= n:
            break
    
    # If we still need more circles, add them
    while len(centers) < n:
        x = 0.5 + (len(centers) - len(centers)) * 0.01
        y = 0.5 + (len(centers) - len(centers)) * 0.01
        centers.append([0.5, 0.5])
        radii_list.append(r * 0.5)
    
    centers = np.array(centers[:n])
    radii_list = radii_list[:n]
    return centers, np.array(radii_list)


def init_dense_grid(n):
    """Initialize with a dense grid layout"""
    centers = []
    radii_list = []
    
    # Try to fit n circles in a grid-like pattern
    cols = int(np.ceil(np.sqrt(n)))
    rows = int(np.ceil(n / cols))
    
    margin = 0.05
    x_spacing = (1 - 2 * margin) / (cols - 1) if cols > 1 else 0
    y_spacing = (1 - 2 * margin) / (rows - 1) if rows > 1 else 0
    
    r_est = min(x_spacing, y_spacing) / 2 * 0.9
    
    idx = 0
    for row in range(rows):
        for col in range(cols):
            if idx >= n:
                break
            x = margin + col * x_spacing
            y = margin + row * y_spacing
            centers.append([x, y])
            radii_list.append(r_est)
            idx += 1
        if idx >= n:
            break
    
    centers = np.array(centers[:n])
    radii_list = radii_list[:n]
    return centers, np.array(radii_list)


def optimize_from_init(centers, radii, n, max_iter=5000):
    """Run optimization from a given initial configuration"""
    x0 = np.concatenate([centers.flatten(), radii])
    constraints = build_constraints(n)
    bounds = make_bounds(n)
    
    result = minimize(
        objective_func,
        x0,
        args=(n,),
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={
            'maxiter': max_iter,
            'ftol': 1e-14,
            'disp': False
        }
    )
    
    opt_centers = result.x[:2 * n].reshape(n, 2)
    opt_radii = result.x[2 * n:]
    
    # Ensure non-negative radii
    opt_radii = np.maximum(opt_radii, 0.0)
    
    # Clamp centers to valid range
    for i in range(n):
        opt_centers[i, 0] = max(opt_radii[i], min(1.0 - opt_radii[i], opt_centers[i, 0]))
        opt_centers[i, 1] = max(opt_radii[i], min(1.0 - opt_radii[i], opt_centers[i, 1]))
    
    return opt_centers, opt_radii, np.sum(opt_radii)


def perturb_centers(centers, radii, n, scale=0.02):
    """Add small random perturbation to centers"""
    new_centers = centers.copy()
    for i in range(n):
        dx = np.random.uniform(-scale, scale)
        dy = np.random.uniform(-scale, scale)
        new_centers[i, 0] = max(radii[i], min(1.0 - radii[i], new_centers[i, 0] + dx))
        new_centers[i, 1] = max(radii[i], min(1.0 - radii[i], new_centers[i, 1] + dy))
    return new_centers


def run_packing():

    best_centers = None

            x0[2 * i + 1] = max(radii[i], min(1.0 - radii[i], x0[2 * i + 1]))
            x0[2 * n + i] = max(0.0, min(0.5, x0[2 * n + i]))
        
        centers = x0[:2 * n].reshape(n, 2)
        radii = x0[2 * n:]
        
        opt_centers, opt_radii, opt_sum = optimize_from_init(centers, radii, n, max_iter=8000)
        
        if opt_sum > best_sum:
            best_sum = opt_sum
            best_centers = opt_centers.copy()
