import numpy as np
from scipy.optimize import minimize

def compute_constraints(params, n_circles):
    """
    Computes the inequality constraints for the circle packing problem.
    Returns a numpy array of constraint values. All values must be >= 0.
    
    Constraints:
    1. Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r (4 per circle)
    2. Non-overlap constraints: dist^2 - (r_i + r_j)^2 >= 0 (N*(N-1)/2 constraints)
    """
    # Reshape parameters
    # params is shape (3 * n_circles,) or (n_circles, 3) depending on usage
    # Here we assume flattened [x0, y0, r0, x1, y1, r1, ...]
    # Actually, let's stick to a consistent layout. 
    # Let's use shape (n_circles, 3) -> columns [x, y, r]
    
    data = params.reshape(n_circles, 3)
    centers = data[:, :2] # (n, 2)
    radii = data[:, 2]    # (n,)
    
    constraints = []
    
    # 1. Boundary constraints
    # x >= r  => x - r >= 0
    # 1 - x >= r => 1 - x - r >= 0
    # y >= r  => y - r >= 0
    # 1 - y >= r => 1 - y - r >= 0
    
    constraints.append(centers[:, 0] - radii)
    constraints.append(1.0 - centers[:, 0] - radii)
    constraints.append(centers[:, 1] - radii)
    constraints.append(1.0 - centers[:, 1] - radii)
    
    # 2. Non-overlap constraints
    # (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
    # We only need to compute for i < j
    
    # Compute pairwise squared distances
    # Centers shape (N, 2)
    # Diff shape (N, N, 2)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist_sq = np.sum(diff**2, axis=2) # Shape (N, N)
    
    # Compute pairwise sum of radii squared
    # radii shape (N,)
    r_sum = radii[:, np.newaxis] + radii[np.newaxis, :] # Shape (N, N)
    r_sum_sq = r_sum**2
    
    # Constraint matrix
    overlap_constraints = dist_sq - r_sum_sq
    
    # Extract upper triangle (i < j) to avoid duplicates and self-constraints
    # np.triu_indices returns indices for upper triangle excluding diagonal
    row_idx, col_idx = np.triu_indices(n_circles, k=1)
    constraints.append(overlap_constraints[row_idx, col_idx])
    
    return np.concatenate(constraints)

def objective_function(params, n_circles):
    """
    Objective function to minimize: -sum(radii)
    """
    data = params.reshape(n_circles, 3)
    radii = data[:, 2]
    return -np.sum(radii)

def run_packing():
    """
    Optimizes the packing of 26 circles in a unit square to maximize sum of radii.
    """
    n_circles = 26
    
    # --- Initialization ---
    # We start with a grid configuration to ensure a dense, valid starting point.
    # A 5x5 grid gives 25 points. We need 26.
    # We will place 25 points in a 5x5 grid and 1 point slightly perturbed or in a gap.
    # Grid spacing: If we place centers at 0.1, 0.3, 0.5, 0.7, 0.9, spacing is 0.2.
    # This allows radius 0.1.
    
    x_vals = np.array([0.1, 0.3, 0.5, 0.7, 0.9])
    y_vals = np.array([0.1, 0.3, 0.5, 0.7, 0.9])
    
    # Generate grid points
    grid_x, grid_y = np.meshgrid(x_vals, y_vals)
    centers_init = np.column_stack((grid_x.flatten(), grid_y.flatten()))
    
    # We have 25 points. We need 1 more.
    # Let's add a point in the center (0.5, 0.5) but it's occupied.
    # Let's add it at (0.2, 0.2) - close to (0.1, 0.1) and (0.3, 0.1) etc.
    # To make it valid initially, we start with a very small radius.
    extra_center = np.array([0.2, 0.2])
    centers_init = np.vstack((centers_init, extra_center))
    
    # Initial radii: small enough to avoid overlaps
    # With grid spacing 0.2, max radius is 0.1.
    # The extra point at (0.2, 0.2) is dist 0.1414 from (0.1, 0.1).
    # r + r <= 0.1414 => r <= 0.07.
    # So init radius 0.05 is safe.
    radii_init = np.full(n_circles, 0.05)
    
    # Pack into params vector: [x0, y0, r0, x1, y1, r1, ...]
    params_init = np.zeros(n_circles * 3)
    params_init[0::3] = centers_init[:, 0]
    params_init[1::3] = centers_init[:, 1]
    params_init[2::3] = radii_init
    
    # --- Optimization ---
    # We use SLSQP to handle constraints.
    # Bounds: x, y in [0, 1], r in [0, 0.5]
    bounds = []
    for _ in range(n_circles):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 0.5)) # r
    
    constraints_dict = {
        'type': 'ineq',
        'fun': lambda p: compute_constraints(p, n_circles)
    }
    
    # Run optimization
    # options={'maxiter': 1000} might be needed if it stalls
    res = minimize(
        objective_function,
        params_init,
        args=(n_circles,),
        method='SLSQP',
        bounds=bounds,
        constraints=constraints_dict,
        options={'maxiter': 2000, 'ftol': 1e-12}
    )
    
    # --- Post-processing ---
    # Extract result
    params_opt = res.x
    data_opt = params_opt.reshape(n_circles, 3)
    centers_opt = data_opt[:, :2]
    radii_opt = data_opt[:, 2]
    
    # Check validity and fix minor numerical issues if necessary

    # However, if the optimizer pushed boundaries slightly, we might need to clamp.
    # But SLSQP respects bounds.
    
    # Just to be safe, ensure radii are non-negative
    radii_opt = np.maximum(radii_opt, 0.0)
    
    sum_radii = np.sum(radii_opt)
    
    return centers_opt, radii_opt, sum_radii
