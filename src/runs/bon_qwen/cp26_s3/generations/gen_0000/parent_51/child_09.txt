import numpy as np
from scipy.optimize import minimize


def run_packing():
    n = 26

    def objective(x):
        centers = x[:2 * n].reshape(n, 2)
        radii = x[2 * n:]
        radii = np.maximum(radii, 0)

        obj = -np.sum(radii)

        mu = 8000.0
        penalty = 0.0

        for i in range(n):
            r = radii[i]
            penalty += mu * max(0, r - centers[i, 0]) ** 2
            penalty += mu * max(0, r - (1 - centers[i, 0])) ** 2
            penalty += mu * max(0, r - centers[i, 1]) ** 2
            penalty += mu * max(0, r - (1 - centers[i, 1])) ** 2

        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                gap = dist - radii[i] - radii[j]
                if gap < 0:
                    penalty += mu * gap ** 2

        return obj + penalty

    def force_directed_relax(centers, radii, steps=2000):
        centers = centers.copy()
        radii = radii.copy()

        for step in range(steps):
            dt = 0.05 * (1 - step / steps)
            grow_rate = 0.001 * (1 - step / steps)

            for i in range(n):
                fx, fy = 0.0, 0.0

                for j in range(n):
                    if i != j:
                        dx = centers[i, 0] - centers[j, 0]
                        dy = centers[i, 1] - centers[j, 1]
                        dist = np.sqrt(dx * dx + dy * dy)

                        if dist > 1e-10:
                            overlap = radii[i] + radii[j] - dist
                            if overlap > 0:
                                force = overlap / dist
                                fx += force * dx
                                fy += force * dy

                centers[i, 0] += dt * fx
                centers[i, 1] += dt * fy

                centers[i, 0] = np.clip(centers[i, 0], 1e-6, 1 - 1e-6)
                centers[i, 1] = np.clip(centers[i, 1], 1e-6, 1 - 1e-6)

            for i in range(n):
                max_r = min(centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])
                for j in range(n):
                    if i != j:
                        d = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                        max_r = min(max_r, d - radii[j])
                max_r = max(max_r, 0)
                radii[i] += grow_rate * max(0, max_r - radii[i])

        return centers, radii

    def make_hex_init(offset_x=0, offset_y=0):
        centers = np.zeros((n, 2))
        radii = np.ones(n) * 0.085

        dx = 0.21
        dy = dx * np.sqrt(3) / 2

        idx = 0
        for row in range(9):
            for col in range(6):
                if idx >= n:
                    break
                x = 0.08 + col * dx + (dx / 2 if row % 2 == 1 else 0) + offset_x
                y = 0.08 + row * dy + offset_y
                if 0.05 < x < 0.95 and 0.05 < y < 0.95:
                    centers[idx] = [x, y]
                    idx += 1

        while idx < n:
            centers[idx] = [0.5 + 0.05 * np.cos(idx * 2.5), 0.5 + 0.05 * np.sin(idx * 2.5)]
            idx += 1

        return centers, radii

    def make_grid_init():
        centers = np.zeros((n, 2))
        radii = np.ones(n) * 0.085

        idx = 0
        for row in range(6):
            for col in range(5):
                if idx >= n:
                    break
                x = 0.1 + col * 0.2 + (0.1 if row % 2 == 1 else 0)
                y = 0.1 + row * 0.18
                centers[idx] = [x, y]
                idx += 1

        while idx < n:
            centers[idx] = [0.5, 0.5]
            idx += 1

        return centers, radii

    def check_validity(centers, radii):
        for i in range(n):
            r = radii[i]
            if r < 0:
                return False
            if centers[i, 0] - r < -1e-9 or centers[i, 0] + r > 1 + 1e-9:
                return False
            if centers[i, 1] - r < -1e-9 or centers[i, 1] + r > 1 + 1e-9:
                return False

        for i in range(n):
            for j in range(i + 1, n):
                d = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if d < radii[i] + radii[j] - 1e-9:
                    return False

        return True

    def fix_violations(centers, radii):
        centers = centers.copy()
        radii = radii.copy()

        for _ in range(100):
            max_violation = 0
            for i in range(n):
                r = radii[i]
                if centers[i, 0] - r < 0:
                    r = centers[i, 0]
                if centers[i, 0] + r > 1:
                    r = 1 - centers[i, 0]
                if centers[i, 1] - r < 0:
                    r = centers[i, 1]
                if centers[i, 1] + r > 1:
                    r = 1 - centers[i, 1]
                radii[i] = max(r, 0)

            changed = True
            while changed:
                changed = False
                for i in range(n):
                    for j in range(i + 1, n):
                        d = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                        if d < radii[i] + radii[j] - 1e-10:
                            overlap = radii[i] + radii[j] - d
                            if radii[i] <= radii[j]:
                                radii[i] -= overlap * 0.5
                            else:
                                radii[j] -= overlap * 0.5
                            changed = True

            max_violation = 0
            for i in range(n):
                for j in range(i + 1, n):
                    d = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                    gap = radii[i] + radii[j] - d
                    max_violation = max(max_violation, gap)

            if max_violation < 1e-12:
                break

        return centers, radii

    best_sum = 0
    best_centers = None
    best_radii = None

    np.random.seed(42)

    configs = []
    configs.append(make_hex_init())
    configs.append(make_grid_init())

    for ox in [-0.02, 0, 0.02]:
        for oy in [-0.02, 0, 0.02]:
            configs.append(make_hex_init(ox, oy))

    for trial_idx, (centers, radii) in enumerate(configs):
        centers_relax, radii_relax = force_directed_relax(centers, radii, steps=1500)

        x0 = np.concatenate([centers_relax.flatten(), radii_relax])
        bounds = [(0, 1)] * (2 * n) + [(0, 0.5)] * n

        result = minimize(
            objective,
            x0,
            method="L-BFGS-B",
            bounds=bounds,
            options={"maxiter": 30000, "ftol": 1e-15, "gtol": 1e-12},
        )

        c = result.x[: 2 * n].reshape(n, 2)
        r = result.x[2 * n:]
        r = np.maximum(r, 0)

        c, r = fix_violations(c, r)

        if check_validity(c, r):
            s = np.sum(r)
            if s > best_sum:
                best_sum = s
                best_centers = c.copy()
                best_radii = r.copy()
        else:
            s = np.sum(r)
            if s > best_sum * 0.95:
                c_fixed, r_fixed = fix_violations(c, r)
                if check_validity(c_fixed, r_fixed):
                    s = np.sum(r_fixed)
                    if s > best_sum:
                        best_sum = s
                        best_centers = c_fixed.copy()
                        best_radii = r_fixed.copy()

    if best_centers is None:
        centers, radii = make_hex_init()
        best_centers, best_radii = fix_violations(centers, radii)
        best_sum = np.sum(best_radii)

    return best_centers, best_radii, best_sum
