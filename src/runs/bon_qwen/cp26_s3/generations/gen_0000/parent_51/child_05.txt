import numpy as np
from scipy.optimize import differential_evolution, minimize

def compute_sum_radii(centers):
    """
    Calculates the sum of radii for a given set of centers.
    Radii are determined by the minimum distance to boundaries and neighbors.
    """
    n = centers.shape[0]
    
    # 1. Boundary constraints
    # x >= r, 1-x >= r => r <= min(x, 1-x)
    # y >= r, 1-y >= r => r <= min(y, 1-y)
    r_boundary = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]), 
                            np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    
    # 2. Pairwise constraints
    # Distance between centers >= r_i + r_j
    # To maximize sum, we approximate by assuming equal radii locally: r_i <= dist/2
    # This is a valid upper bound for equal packing.
    
    # Compute pairwise distances efficiently
    # Using broadcasting: (n, 1, 2) - (1, n, 2) -> (n, n, 2)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    # Mask diagonal to infinity so min doesn't pick it
    np.fill_diagonal(dists, np.inf)
    
    # Minimum distance to any other circle
    min_dists = np.min(dists, axis=1)
    
    # Neighbor-limited radii
    r_neighbor = min_dists / 2.0
    
    # Final radii
    radii = np.minimum(r_boundary, r_neighbor)
    
    return np.sum(radii), radii

def objective(centers_flat):

    centers = centers_flat.reshape(-1, 2)
    s, _ = compute_sum_radii(centers)

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))

                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True
