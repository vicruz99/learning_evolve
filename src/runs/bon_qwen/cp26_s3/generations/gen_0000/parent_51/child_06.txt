import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    # 1. Initialization: Hexagonal lattice pattern
    # This provides a dense starting configuration close to optimal.
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.05)
    
    # Grid parameters
    rows = 6
    cols = 5
    cell_h = np.sqrt(3) / 10.0 # Approximate vertical spacing
    cell_w = 2.0 / 10.0        # Approximate horizontal spacing
    
    count = 0
    for r in range(rows):
        # Stagger even and odd rows for hexagonal packing
        x_start = (0.5 * cell_w) if r % 2 == 1 else 0.0
        for c in range(cols):
            if count < n:
                centers[count, 0] = 0.1 + c * cell_w + x_start
                centers[count, 1] = 0.1 + r * cell_h
                count += 1
    
    # 2. Define the optimization problem
    # Variables: [x1, y1, r1, x2, y2, r2, ..., x26, y26, r26]
    x0 = np.zeros(3 * n)
    for i in range(n):
        x0[3 * i] = centers[i, 0]
        x0[3 * i + 1] = centers[i, 1]
        x0[3 * i + 2] = radii[i]
    
    # Objective: Minimize negative sum of radii (equivalent to maximizing sum)
    def objective(x):
        return -np.sum(x[2::3])
    
    # Constraints
    constraints = []
    
    # Boundary and non-negativity constraints
    for i in range(n):
        idx = 3 * i
        # x - r >= 0  =>  r - x <= 0
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: x[idx] - x[idx+2]})
        # 1 - (x + r) >= 0  =>  x + r - 1 <= 0
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: 1.0 - x[idx] - x[idx+2]})
        # y - r >= 0
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: x[idx+1] - x[idx+2]})
        # 1 - (y + r) >= 0
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: 1.0 - x[idx+1] - x[idx+2]})
        # r >= 0 (slack allowed, but enforced by boundaries mostly)
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: x[idx+2]})

    # Non-overlap constraints: dist(i, j) >= r_i + r_j
    # (x_i - x_j)^2 + (y_i - y_j)^2 >= (r_i + r_j)^2
    for i in range(n):
        for j in range(i + 1, n):
            idx_i = 3 * i
            idx_j = 3 * j
            
            def separation(x, i=i, j=j):
                xi, yi, ri = x[3*i], x[3*i+1], x[3*i+2]
                xj, yj, rj = x[3*j], x[3*j+1], x[3*j+2]
                dx = xi - xj
                dy = yi - yj
                dist_sq = dx**2 + dy**2
                rad_sum = ri + rj
                return dist_sq - rad_sum**2
            
            constraints.append({'type': 'ineq', 'fun': separation})

    # 3. Run Optimization
    # SLSQP is a sequential least squares programming method, suitable for non-linear constraints
    result = minimize(objective, x0, method='SLSQP', constraints=constraints, 
                      options={'maxiter': 2000, 'ftol': 1e-10, 'disp': False})

    # 4. Extract Results
    final_centers = np.zeros((n, 2))
    final_radii = np.zeros(n)
    
    for i in range(n):
        final_centers[i, 0] = result.x[3 * i]
        final_centers[i, 1] = result.x[3 * i + 1]
        final_radii[i] = max(0.0, result.x[3 * i + 2]) # Ensure non-negative

    total_sum = np.sum(final_radii)
    
    return final_centers, final_radii, total_sum
