import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # 1. Initialization: Hexagonal packing pattern
    # We aim for a layout that fits 26 circles.
    # A pattern of rows with counts [5, 4, 5, 4, 5, 3] sums to 26.
    # This is a hexagonal arrangement.
    
    r_init = 0.085  # Initial safe radius
    row_counts = [5, 4, 5, 4, 5, 3]
    
    centers = []
    current_y = r_init
    
    for idx, count in enumerate(row_counts):
        # Shift for hexagonal staggering
        shift = r_init if idx % 2 == 1 else 0.0
        
        # Calculate starting x to center the row
        # Width of row with 'count' circles of radius r is 2*r*count (approx)
        # More precisely, distance from center of first to last is (count-1)*2r
        # Leftmost point of first circle is x_first - r.
        # Rightmost point of last circle is x_last + r.
        # x_last = x_first + (count-1)*2r
        # Width = x_last + r - (x_first - r) = x_first - r + (count-1)*2r + 2r = x_first + (count+1)*2r?
        # Let's just center the centers.
        # Span of centers = (count-1)*2r.
        # Center of row at 0.5.
        # Start x = 0.5 - (count-1)*r + shift.
        
        start_x = 0.5 - (count - 1) * r_init + shift
        
        for k in range(count):
            cx = start_x + k * 2 * r_init
            cy = current_y
            centers.append([cx, cy])
        
        # Vertical spacing for hex packing is sqrt(3)*r
        current_y += np.sqrt(3) * r_init
        
    centers = np.array(centers)
    radii = np.full(n, r_init)
    
    # Verify and adjust if needed (though 0.085 should be safe for this config)
    # Flatten variables: [x1, y1, ..., xn, yn, r1, ..., rn]
    x0 = np.concatenate([centers.flatten(), radii])
    
    # Define bounds
    # x, y in [0, 1], r in [0, 0.5]
    bounds = [(0, 1)] * (2 * n) + [(0, 0.5)] * n
    
    # Objective: Maximize sum of radii -> Minimize -sum(radii)
    def objective(vars):
        # Radii are the last n elements
        r = vars[2 * n:]
        return -np.sum(r)
    
    # Constraints
    # 1. Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    # 2. Boundary: x >= r, x <= 1-r, y >= r, y <= 1-r
    
    def constraint_func(vars):
        n = len(vars) // 3
        centers = vars[:2 * n].reshape(n, 2)
        radii = vars[2 * n:]
        
        con_list = []
        
        # Pairwise constraints
        # Vectorized computation
        # centers shape (n, 2)
        # diff shape (n, n, 2)
        diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
        dist_sq = np.sum(diff ** 2, axis=2)
        
        # radii sum squared shape (n, n)
        r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
        r_sum_sq = r_sum ** 2
        
        # We only need i < j constraints (upper triangle)
        # But to keep it simple, we can return all, or use mask.
        # Using mask to reduce constraint count
        mask = np.triu(np.ones((n, n), dtype=bool), k=1)
        pairwise_cons = dist_sq[mask] - r_sum_sq[mask]
        con_list.append(pairwise_cons)
        
        # Boundary constraints
        # x - r >= 0
        # 1 - x - r >= 0
        # y - r >= 0
        # 1 - y - r >= 0
        
        x = centers[:, 0]
        y = centers[:, 1]
        r = radii
        
        con_list.append(x - r)
        con_list.append(1.0 - x - r)
        con_list.append(y - r)
        con_list.append(1.0 - y - r)
        
        return np.concatenate(con_list)
    
    # SLSQP requires a list of constraint dictionaries
    cons = ({'type': 'ineq', 'fun': constraint_func})
    
    # Optimization
    # Use SLSQP
    res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, 
                   options={'maxiter': 1000, 'ftol': 1e-9, 'disp': False})
    
    # Extract results

    # Ensure non-negative radii (just in case)
    final_radii = np.maximum(final_radii, 0.0)
    
    sum_radii = np.sum(final_radii)
    
    return final_centers, final_radii, sum_radii
