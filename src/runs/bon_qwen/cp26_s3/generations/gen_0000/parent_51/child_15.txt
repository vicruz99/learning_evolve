import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def objective_func(x):
    """Minimize negative sum of radii (i.e., maximize sum of radii)"""
    radii = x[2 * N_CIRCLES:]
    return -np.sum(radii)


def boundary_constraint_lower_x(x, circle_idx):
    """x_i >= r_i"""
    return x[2 * circle_idx] - x[2 * N_CIRCLES + circle_idx]


def boundary_constraint_upper_x(x, circle_idx):
    """x_i + r_i <= 1"""
    return 1.0 - x[2 * circle_idx] - x[2 * N_CIRCLES + circle_idx]


def boundary_constraint_lower_y(x, circle_idx):
    """y_i >= r_i"""
    return x[2 * circle_idx + 1] - x[2 * N_CIRCLES + circle_idx]


def boundary_constraint_upper_y(x, circle_idx):
    """y_i + r_i <= 1"""
    return 1.0 - x[2 * circle_idx + 1] - x[2 * N_CIRCLES + circle_idx]


def overlap_constraint(x, i, j):
    """Distance between circles i and j >= r_i + r_j"""
    dx = x[2 * i] - x[2 * j]
    dy = x[2 * i + 1] - x[2 * j + 1]
    dist = np.sqrt(dx * dx + dy * dy)
    r_sum = x[2 * N_CIRCLES + i] + x[2 * N_CIRCLES + j]
    return dist - r_sum


def create_hexagonal_initial():
    """Create initial hexagonal-like configuration with small radii"""
    n = N_CIRCLES
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.035

    idx = 0
    y = 0.04
    row = 0

    while idx < n:
        if row % 2 == 0:
            start_x = 0.04
        else:
            start_x = 0.14

        x = start_x
        while x <= 0.96 and idx < n:
            centers[idx] = [x, y]
            idx += 1
            x += 0.20

        y += 0.16
        row += 1

    return centers, radii


def create_corner_biased_initial():
    """Create initial config biased toward corners (larger potential radii)"""
    n = N_CIRCLES
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.035

    # Place 4 circles near corners first
    centers[0] = [0.12, 0.12]
    centers[1] = [0.88, 0.12]
    centers[2] = [0.12, 0.88]
    centers[3] = [0.88, 0.88]
    radii[:4] = 0.05

    # Fill remaining with hexagonal pattern
    idx = 4
    y = 0.15
    row = 0
    while idx < n:
        if row % 2 == 0:
            start_x = 0.18
        else:
            start_x = 0.30

        x = start_x
        while x <= 0.82 and idx < n:
            centers[idx] = [x, y]
            idx += 1
            x += 0.22

        y += 0.18
        row += 1

    return centers, radii


def create_grid_initial():
    """Create a uniform grid initial configuration"""
    n = N_CIRCLES
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.035

    # 5x5 grid = 25, plus 1 extra
    positions = []
    for i in range(5):
        for j in range(5):
            x = 0.1 + i * 0.2
            y = 0.1 + j * 0.2
            positions.append([x, y])
    # Add one more in the center
    positions.append([0.5, 0.5])

    for idx in range(n):
        centers[idx] = positions[idx]

    return centers, radii


def build_constraints():
    """Build list of constraint dictionaries"""
    n = N_CIRCLES
    constraints = []

    for i in range(n):
        constraints.append({'type': 'ineq', 'fun': boundary_constraint_lower_x, 'args': (i,)})
        constraints.append({'type': 'ineq', 'fun': boundary_constraint_upper_x, 'args': (i,)})
        constraints.append({'type': 'ineq', 'fun': boundary_constraint_lower_y, 'args': (i,)})
        constraints.append({'type': 'ineq', 'fun': boundary_constraint_upper_y, 'args': (i,)})

    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({'type': 'ineq', 'fun': overlap_constraint, 'args': (i, j)})

    return constraints


def run_single_optimization(x0, constraints, bounds, seed_val):
    """Run a single optimization from x0"""
    np.random.seed(seed_val)

    # Add tiny perturbation to avoid exact symmetries
    x0_perturbed = x0.copy()
    x0_perturbed += np.random.uniform(-1e-6, 1e-6, size=x0.shape)

    result = minimize(objective_func, x0_perturbed, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': 5000, 'ftol': 1e-15, 'disp': False})

    return result


def refine_packing(centers, radii, constraints, bounds):
    """Refine packing by trying to grow each circle individually"""
    n = N_CIRCLES
    opt_centers = centers.copy()
    opt_radii = radii.copy()

    # Try growing each circle one at a time
    for trial in range(3):
        improved = True
        while improved:
            improved = False
            for i in range(n):
                # Try to grow circle i
                test_radii = opt_radii.copy()
                test_radii[i] *= 1.05  # Try 5% growth

                x0 = np.concatenate([opt_centers.flatten(), test_radii])

                # Fix other variables, only optimize circle i
                # Actually, let's do a full re-optimization with circle i's radius as starting point
                x0_start = np.concatenate([opt_centers.flatten(), opt_radii])

                result = minimize(objective_func, x0_start, method='SLSQP',
                                  bounds=bounds, constraints=constraints,
                                  options={'maxiter': 500, 'ftol': 1e-14, 'disp': False})

                new_sum = np.sum(result.x[2 * N_CIRCLES:])
                old_sum = np.sum(opt_radii)

                if new_sum > old_sum + 1e-10:
                    opt_centers = result.x[:2 * n].reshape(n, 2)
                    opt_radii = result.x[2 * n:]
                    opt_radii = np.maximum(opt_radii, 0)
                    improved = True

    return opt_centers, opt_radii
