import numpy as np

def run_packing():
    """
    Pack 26 circles in a unit square [0,1]x[0,1] to maximize the sum of radii.
    
    Uses Simulated Annealing with vectorized repulsive forces to find a dense packing.
    """
    n = 26
    
    # Initialize centers on a dense hexagonal grid pattern
    # This provides a good starting point closer to optimal than random
    centers = np.zeros((n, 2))
    idx = 0
    for row in range(5):
        # Calculate y position for the row
        y = 0.125 + row * 0.22 
        # Calculate number of columns for this row to fit 26 circles
        # Alternating 6 and 5 columns works well for hexagonal packing
        n_cols = 6 if row % 2 == 0 else 5
        for col in range(n_cols):
            if idx >= n:
                break
            # x position, offset for odd rows
            x_offset = 0.08 if row % 2 != 0 else 0.0
            x = 0.08 + x_offset + col * 0.20
            centers[idx] = [x, y]
            idx += 1

    # Simulation parameters
    r = 0.05          # Starting radius
    max_r = 0.12      # Target upper bound for radius
    temperature = 0.05 # Starting "thermal" noise
    cooling = 0.9995  # Temperature reduction per step
    dt = 0.2          # Integration step size
    k_rep = 50.0      # Repulsive force constant
    max_iter = 3000   # Number of simulation steps

    # Main optimization loop (Simulated Annealing)
    for step in range(max_iter):
        # Slowly increase radius
        if r < max_r:
            r += 0.00002
        
        # Decrease temperature
        temperature *= cooling
        
        # Vectorized distance calculation
        # diffs shape: (26, 26, 2)
        diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
        
        # dists shape: (26, 26)
        dists = np.sqrt(np.sum(diffs**2, axis=2))
        
        # Handle diagonal (self-distance) by setting to infinity
        np.fill_diagonal(dists, np.inf)
        
        # Calculate repulsive forces for overlapping pairs
        # Overlap amount: max(0, 2r - dist)
        overlap = np.maximum(0, 2 * r - dists)
        
        # Force magnitude proportional to overlap
        force_mag = k_rep * overlap
        
        # Unit vectors for direction (diffs / dists)
        # Avoid division by zero by masking inf distances
        inv_dists = 1.0 / dists
        inv_dists[np.isinf(inv_dists)] = 0.0
        
        # Unit vectors shape: (26, 26, 2)
        unit_vecs = diffs * inv_dists[:, :, np.newaxis]
        
        # Pairwise forces shape: (26, 26, 2)
        forces_pair = unit_vecs * force_mag[:, :, np.newaxis]
        
        # Net force on each circle by summing over neighbors
        net_force = np.sum(forces_pair, axis=1)
        
        # Boundary repulsion forces (walls at 0 and 1)
        # If circle is closer to wall than r, push it back
        # Left wall (x=0)
        mask_l = centers[:, 0] < r
        net_force[mask_l, 0] += k_rep * (r - centers[mask_l, 0])
        
        # Right wall (x=1)
        mask_r = centers[:, 0] > 1 - r
        net_force[mask_r, 0] -= k_rep * (centers[mask_r, 0] - (1 - r))
        
        # Bottom wall (y=0)
        mask_b = centers[:, 1] < r
        net_force[mask_b, 1] += k_rep * (r - centers[mask_b, 1])
        
        # Top wall (y=1)
        mask_t = centers[:, 1] > 1 - r
        net_force[mask_t, 1] -= k_rep * (centers[mask_t, 1] - (1 - r))
        
        # Update centers: deterministic movement + random noise
        centers += dt * net_force + np.random.randn(n, 2) * temperature
        
        # Strict boundary constraints to keep circles inside
        centers[:, 0] = np.clip(centers[:, 0], r, 1 - r)
        centers[:, 1] = np.clip(centers[:, 1], r, 1 - r)

    # Final relaxation pass to clear any minor overlaps and stabilize
    # Keep r fixed at the highest value reached
    final_r = r
    for _ in range(1000):
        diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
        dists = np.sqrt(np.sum(diffs**2, axis=2))
        np.fill_diagonal(dists, np.inf)
        
        overlap = np.maximum(0, 2 * final_r - dists)
        inv_dists = 1.0 / dists
        inv_dists[np.isinf(inv_dists)] = 0.0
        
        # Use a strong repulsive force to clear overlaps quickly
        force_mag = 100.0 * overlap
        unit_vecs = diffs * inv_dists[:, :, np.newaxis]
        forces_pair = unit_vecs * force_mag[:, :, np.newaxis]
        net_force = np.sum(forces_pair, axis=1)
        
        # Boundary correction
        centers[:, 0] = np.clip(centers[:, 0], final_r, 1 - final_r)
        centers[:, 1] = np.clip(centers[:, 1], final_r, 1 - final_r)
        
        # Apply force
        centers += 0.1 * net_force
        
        # Re-apply boundaries immediately
        centers[:, 0] = np.clip(centers[:, 0], final_r, 1 - final_r)
        centers[:, 1] = np.clip(centers[:, 1], final_r, 1 - final_r)

    # Verify and correct radius if overlaps persist
    # Recalculate min distance between any pair
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    min_dist = np.min(dists)
    
    # If min_dist < 2*r, we have overlaps. 
    # The required radius is min_dist / 2.
    # We also check boundaries.
    min_x = np.min(centers[:, 0])
    max_x = np.max(centers[:, 0])
    min_y = np.min(centers[:, 1])
    max_y = np.max(centers[:, 1])
    
    r_boundary_x = min(min_x, 1 - max_x)
    r_boundary_y = min(min_y, 1 - max_y)
    
    # The valid radius is limited by the tightest constraint:
    # 1. Separation between circles: min_dist / 2
    # 2. Distance to walls: r_boundary_x, r_boundary_y
    # We subtract a tiny epsilon for numerical safety
    final_r_valid = (min_dist / 2.0)
    final_r_valid = min(final_r_valid, r_boundary_x, r_boundary_y)
    final_r_valid -= 1e-5 # Safety margin

    # If the final radius is surprisingly small (optimization failed), 
    # fallback to a safe grid packing
    if final_r_valid < 0.05:
        centers = np.array([[(i+1)*0.18, (j+1)*0.18] for i in range(5) for j in range(5) if (i*5+j)<26])
        final_r_valid = 0.09

    radii = np.full(n, final_r_valid)
    sum_radii = np.sum(radii)
    
    return centers, radii, float(sum_radii)
