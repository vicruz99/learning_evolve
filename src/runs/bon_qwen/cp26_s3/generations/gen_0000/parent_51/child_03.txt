import numpy as np
from scipy.optimize import minimize

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    n = centers.shape[0]

    if np.isnan(centers).any():
        return False
    if np.isnan(radii).any():
        return False

    for i in range(n):
        if radii[i] < 0:
            return False
        elif np.isnan(radii[i]):
            return False

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-9 or x + r > 1 + 1e-9 or y - r < -1e-9 or y + r > 1 + 1e-9:
            return False

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-9:
                return False

    return True

def hexagonal_init(n):
    # Estimate radius for hex packing in 1x1 square
    # Area approx: N * pi * r^2 <= 1 * pi / sqrt(12) => r approx 1/sqrt(N*sqrt(12))
    # A safer estimate for boundaries: r ~ 0.105
    r_guess = 0.105
    
    centers = np.zeros((n, 2))
    radii = np.full(n, r_guess)
    
    idx = 0
    row = 0
    while idx < n:
        # Calculate y for current row
        y = r_guess + row * (np.sqrt(3) * r_guess)
        if y + r_guess > 1:
            # Row too high, reduce r_guess slightly and restart? 
            # Or just break and adjust. 
            # For robustness, we just fill as much as possible
            pass
            
        # Determine start x based on row parity
        start_x = r_guess + (0.5 * r_guess if row % 2 == 1 else 0)
        
        col = 0
        x = start_x + col * (2 * r_guess)
        while x + r_guess <= 1 and idx < n:
            centers[idx] = [x, y]
            idx += 1
            col += 1
            x = start_x + col * (2 * r_guess)
        
        row += 1
        if row > 10: break # Safety
        
    # If we didn't fill n circles, the guess was too large or logic simple
    # Fill remaining with small random circles if needed
    while idx < n:
        centers[idx] = [np.random.rand() * 0.8 + 0.1, np.random.rand() * 0.8 + 0.1]
        radii[idx] = 0.01
        idx += 1
        
    return centers, radii

def run_packing():
    n = 26
    best_sum = -np.inf
    best_centers = None
    best_radii = None
    
    np.random.seed(123) # For reproducibility in seeding
    
    # We will try a few optimization runs
    for seed in range(10):
        # Generate initial hexagonal configuration
        centers_init, radii_init = hexagonal_init(n)
        
        # Add some noise to escape local minima of the init function
        noise = np.random.normal(0, 0.005, (n, 2))
        centers_init += noise
        
        # Ensure bounds for initial state
        centers_init = np.clip(centers_init, 0, 1)
        radii_init = np.clip(radii_init, 0.01, 0.5)
        
        # Flatten for scipy
        x0 = np.concatenate([centers_init.flatten(), radii_init])
        
        # Objective function with penalty
        def objective(vars):
            c = vars[:2*n].reshape(n, 2)
            r = vars[2*n:]
            
            # Maximize sum(r) => minimize -sum(r)
            score = -np.sum(r)
            
            # Penalty for overlaps and boundaries
            penalty = 0.0
            
            # Boundary penalties
            # x - r >= 0, x + r <= 1 => 1 - x - r >= 0
            # y - r >= 0, y + r <= 1 => 1 - y - r >= 0
            # Using squared max(0, violation)
            
            # Vectorized boundary check
            # Left
            violations = np.maximum(0, r - c[:, 0])
            penalty += np.sum(violations**2)
            # Right
            violations = np.maximum(0, r + c[:, 0] - 1)
            penalty += np.sum(violations**2)
            # Bottom
            violations = np.maximum(0, r - c[:, 1])
            penalty += np.sum(violations**2)
            # Top
            violations = np.maximum(0, r + c[:, 1] - 1)
            penalty += np.sum(violations**2)
            
            # Overlap penalties
            # O(N^2) loop is fine for N=26
            for i in range(n):
                for j in range(i + 1, n):
                    dist = np.sqrt((c[i,0]-c[j,0])**2 + (c[i,1]-c[j,1])**2)
                    overlap = r[i] + r[j] - dist
                    if overlap > 0:
                        penalty += overlap**2
            
            # Weight for penalty. Higher is stricter.
            return score + 1000 * penalty

        # Bounds
        bounds = []
        for _ in range(n):
            bounds.append((0.0, 1.0)) # x
            bounds.append((0.0, 1.0)) # y
            bounds.append((0.0, 0.5)) # r
        
        # Optimize
        try:
            res = minimize(objective, x0, method='L-BFGS-B', bounds=bounds, 
                           options={'maxiter': 2000, 'ftol': 1e-12})
            
            curr_centers = res.x[:2*n].reshape(n, 2)
            curr_radii = res.x[2*n:]
            
            # Check if valid
            if validate_packing(curr_centers, curr_radii):
                s = np.sum(curr_radii)
                if s > best_sum:
                    best_sum = s
                    best_centers = curr_centers.copy()
                    best_radii = curr_radii.copy()
        except Exception:
            continue
            
    # If no valid packing found in loop (unlikely), return a safe fallback
    if best_centers is None:

                    idx += 1

             centers[idx] = [np.random.rand(), np.random.rand()]
             radii[idx] = 0.01
             idx += 1
        best_centers = centers
        best_radii = radii
        best_sum = np.sum(radii)

    return best_centers, best_radii, best_sum
