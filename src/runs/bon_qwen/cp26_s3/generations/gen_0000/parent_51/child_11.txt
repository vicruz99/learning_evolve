import numpy as np
from scipy.optimize import minimize
from functools import partial


def obj_func(x, n):
    s = 0.0
    for i in range(n):
        s += x[3 * i + 2]
    return -s


def bc_xl(x, i):
    return x[3 * i] - x[3 * i + 2]


def bc_xh(x, i):
    return 1.0 - x[3 * i] - x[3 * i + 2]


def bc_yl(x, i):
    return x[3 * i + 1] - x[3 * i + 2]


def bc_yh(x, i):
    return 1.0 - x[3 * i + 1] - x[3 * i + 2]


def nonoverlap_con(x, i, j):
    dx = x[3 * i] - x[3 * j]
    dy = x[3 * i + 1] - x[3 * j + 1]
    return np.sqrt(dx * dx + dy * dy) - x[3 * i + 2] - x[3 * j + 2]


def init_hexagonal(n, variant=0):
    cx = np.zeros(n)
    cy = np.zeros(n)
    r = np.zeros(n)

    # Estimate a good starting radius
    # For hexagonal packing of ~26 circles, r ~ 0.1
    r0 = 0.095

    # Define row patterns - different variants for diversity
    if variant == 0:
        pattern = [6, 5, 6, 5, 4]
    elif variant == 1:
        pattern = [5, 6, 5, 6, 4]
    elif variant == 2:
        pattern = [6, 6, 5, 5, 4]
    elif variant == 3:
        pattern = [5, 5, 6, 5, 5]
    else:
        pattern = [6, 5, 6, 5, 4]

    idx = 0
    for row_idx, ncols in enumerate(pattern):
        # Horizontal spacing = 2*r0 for hexagonal
        # Vertical spacing = r0*sqrt(3)

            x_offset = r0
        else:
            x_offset = r0 + r0 * 0.51  # Slightly less than full offset to fit more
        
        y_pos = r0 + row_idx * r0 * np.sqrt(3)
        
        for col in range(ncols):
            if idx >= n:
                break

            cx[idx] = x_pos
            cy[idx] = y_pos

            idx += 1
        
        if idx >= n:
            break
    
    return cx, cy, r


def init_grid(n, variant=0):
    cx = np.zeros(n)
    cy = np.zeros(n)
    r = np.zeros(n)

    if variant == 0:
        cols = 6
        rows = 5
    elif variant == 1:
        cols = 5
        rows = 6
    else:
        cols = 7
        rows = 4
    
    x_step = 1.0 / (cols + 1)
    y_step = 1.0 / (rows + 1)
    
    for i in range(n):
        row = i // cols
        col = i % cols
        cx[i] = (col + 1) * x_step
        cy[i] = (row + 1) * y_step
        margin = min(cx[i], 1.0 - cx[i], cy[i], 1.0 - cy[i])
        r[i] = margin * 0.55

    return cx, cy, r


def build_constraints(n):
    cons = []
    for i in range(n):
        cons.append({'type': 'ineq', 'fun': partial(bc_xl, i=i)})
        cons.append({'type': 'ineq', 'fun': partial(bc_xh, i=i)})
        cons.append({'type': 'ineq', 'fun': partial(bc_yl, i=i)})
        cons.append({'type': 'ineq', 'fun': partial(bc_yh, i=i)})
    
    for i in range(n):
        for j in range(i + 1, n):
            cons.append({'type': 'ineq', 'fun': partial(nonoverlap_con, i=i, j=j)})
    
    return cons


def build_bounds(n):
    bounds = []
    for k in range(3 * n):
        if k % 3 < 2:
            bounds.append((0.0, 1.0))
        else:
            bounds.append((1e-12, 0.5))
    return bounds


def flatten(centers, radii, n):
    x = np.zeros(3 * n)
    for i in range(n):
        x[3 * i] = centers[i, 0]
        x[3 * i + 1] = centers[i, 1]
        x[3 * i + 2] = radii[i]
    return x


def unflatten(x, n):
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        centers[i, 0] = x[3 * i]
        centers[i, 1] = x[3 * i + 1]
        radii[i] = max(0.0, x[3 * i + 2])
    return centers, radii
