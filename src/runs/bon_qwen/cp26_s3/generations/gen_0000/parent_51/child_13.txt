    *   The constraints are non-convex: circles must not overlap (distance $\ge$ sum of radii) and must stay within the unit square boundaries.
    *   The problem is sensitive to local optima, so multiple initializations are required.

    *   **Initial Placement**: We use a hexagonal lattice pattern. This arrangement is highly efficient and serves as a strong starting point for optimization. We place 26 circles with a conservative initial radius (0.08) to guarantee validity before optimization.
    *   **Optimization**: We use `scipy.optimize.minimize` with the L-BFGS-B method, which supports bound constraints.
    *   **Penalty Method**: Since standard L-BFGS-B does not handle complex inequality constraints (overlaps), we incorporate a penalty function into the objective.

import numpy as np
from scipy.optimize import minimize
import math

def run_packing():
    n = 26
    
    # We maximize sum of radii, so we minimize -sum(radii) + penalty
    def objective(vars):
        val = -np.sum(vars[2::3]) # -sum of radii
        
        penalty = 0.0
        weight = 1000.0
        
        # Boundary penalties
        for i in range(n):
            x = vars[3*i]
            y = vars[3*i+1]
            r = vars[3*i+2]
            
            # Check left
            d = x - r
            if d < 0: penalty += d*d
            # Check right
            d = 1 - x - r
            if d < 0: penalty += d*d
            # Check bottom
            d = y - r
            if d < 0: penalty += d*d
            # Check top
            d = 1 - y - r
            if d < 0: penalty += d*d
            
        # Overlap penalties
        # Using squared distance to keep it smooth and differentiable
        # Constraint: dist^2 >= (r1 + r2)^2
        for i in range(n):
            xi = vars[3*i]
            yi = vars[3*i+1]
            ri = vars[3*i+2]
            for j in range(i+1, n):
                xj = vars[3*j]
                yj = vars[3*j+1]
                rj = vars[3*j+2]
                
                dist_sq = (xi - xj)**2 + (yi - yj)**2
                sum_r_sq = (ri + rj)**2
                
                constraint_val = dist_sq - sum_r_sq
                if constraint_val < 0:
                    penalty += constraint_val**2
                    
        return val + weight * penalty

    def get_valid_solution(x, y, r):
        """
        Check if the configuration is valid. 
        If not, shrink radii to make it valid.
        Returns (centers, radii, sum_radii, is_valid_original)
        """
        centers = np.column_stack((x, y))
        radii = r
        valid = True
        
        # Check boundaries
        for i in range(n):
            if radii[i] < 0:
                radii[i] = 0
            if x[i] < radii[i]: valid = False
            if x[i] > 1 - radii[i]: valid = False
            if y[i] < radii[i]: valid = False
            if y[i] > 1 - radii[i]: valid = False

        # Check overlaps and determine max valid scale factor
        scale_factor = 1.0
        
        for i in range(n):
            # Boundary scaling
            # Need r <= x <= 1-r => r <= x AND r <= 1-x => r <= min(x, 1-x)
            limit = min(x[i], 1 - x[i])
            if radii[i] > limit:
                if radii[i] > 1e-9:
                    scale_factor = min(scale_factor, limit / radii[i])
                valid = False
            
            limit = min(y[i], 1 - y[i])
            if radii[i] > limit:
                if radii[i] > 1e-9:
                    scale_factor = min(scale_factor, limit / radii[i])
                valid = False

            # Overlap scaling
            for j in range(i + 1, n):
                dist = math.hypot(x[i] - x[j], y[i] - y[j])
                sum_r = radii[i] + radii[j]
                if dist < sum_r:
                    if sum_r > 1e-9:
                        scale_factor = min(scale_factor, dist / sum_r)
                    valid = False
        
        # Apply scaling if needed to ensure validity
        # We apply a slightly stricter scale to be safe against numerical errors
        safe_scale = scale_factor * 0.99999 
        if scale_factor < 1.0:
            radii *= safe_scale
        
        # Final strict check (optional but good for debug)
        # We assume scaling made it valid.
        
        sum_radii = np.sum(radii)
        return centers, radii, sum_radii, valid

    # Bounds
    bounds = []
    for _ in range(n):
        bounds.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])

    best_sum = -1.0
    best_centers = None
    best_radii = None

    # 1. Grid / Hexagonal Initialization
    # Try to place 26 circles in a hexagonal grid with small radius
    r_init = 0.08
    count = 0
    x, y, r = [], [], []
    
    row = 0
    while count < n:
        y_row = r_init + row * r_init * math.sqrt(3)
        if y_row + r_init > 1:
            # Try to fill remaining with random or just stop if out of bounds
            # Fallback to random placement for remaining
            while count < n:
                x.append(np.random.uniform(0.1, 0.9))
                y.append(np.random.uniform(0.1, 0.9))
                r.append(0.01)
                count += 1
            break
            
        offset = 0
        if row % 2 == 1:
            offset = r_init
            
        x_pos = r_init + offset
        while x_pos + r_init <= 1 and count < n:
            x.append(x_pos)
            y.append(y_row)
            r.append(r_init)
            count += 1
            x_pos += 2 * r_init
        row += 1
        
    vars_init = np.array([val for row_vals in [x, y, r] for val in row_vals])
    
    # Optimize from grid start
    try:
        res = minimize(objective, vars_init, method='L-BFGS-B', bounds=bounds, options={'maxiter': 2000})
        cx, cy, cr = res.x[0::3], res.x[1::3], res.x[2::3]
        c, rd, s, v = get_valid_solution(cx, cy, cr)
        if s > best_sum:
            best_sum = s
            best_centers = c
            best_radii = rd
    except:
        pass

    # 2. Random Restarts
    for i in range(50):
        # Random valid initialization
        # Place points randomly, start with small radius
        rand_x = np.random.uniform(0.15, 0.85, n)
        rand_y = np.random.uniform(0.15, 0.85, n)
        rand_r = np.full(n, 0.02)
        
        vars_rand = np.array([val for row_vals in [rand_x, rand_y, rand_r] for val in row_vals])
        
        try:
            res = minimize(objective, vars_rand, method='L-BFGS-B', bounds=bounds, options={'maxiter': 1500})
            cx, cy, cr = res.x[0::3], res.x[1::3], res.x[2::3]
            c, rd, s, v = get_valid_solution(cx, cy, cr)
            if s > best_sum:
                best_sum = s
                best_centers = c
                best_radii = rd
        except:
            pass

    # 3. Equal Radii Optimization (Heuristic)
    # Sometimes forcing equal radii helps finding a dense local optimum
    # We take the best centers found so far and optimize only radii (assuming equal) or full
    # Let's just try one run with equal radii initialized at the average of best_radii
    if best_radii is not None:
        avg_r = np.mean(best_radii)
        vars_eq = np.array([val for row_vals in [best_centers[:,0], best_centers[:,1], np.full(n, avg_r)] for val in row_vals])
        # We can relax bounds for radii to allow optimization
        # But we keep x,y fixed? No, let's optimize everything but initialize with equal r
        try:
            res = minimize(objective, vars_eq, method='L-BFGS-B', bounds=bounds, options={'maxiter': 1000})
            cx, cy, cr = res.x[0::3], res.x[1::3], res.x[2::3]
            c, rd, s, v = get_valid_solution(cx, cy, cr)
            if s > best_sum:
                best_sum = s
                best_centers = c
                best_radii = rd
        except:
            pass

    return best_centers, best_radii, best_sum
