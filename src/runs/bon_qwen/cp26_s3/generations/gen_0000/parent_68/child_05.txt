

Looking at this circle packing problem, I need to pack 26 circles in a unit square to maximize the sum of radii. Let me think systematically about the approach.

<strategy>
1. **Problem Analysis**: Maximizing sum of radii is a nonconvex optimization problem with 78 variables (26 circles × 3: x, y, r). Constraints include boundary containment and non-overlap.

2. **Initialization Strategy**: Use a hexagonal grid pattern which is the densest packing arrangement. For 26 circles, a 5×5-6 arrangement works well.

3. **Optimization Approach**: 
   - Use L-BFGS-B with smooth penalty method for fast convergence
   - Multiple restarts to escape local optima
   - Iterative penalty increase for better constraint satisfaction
   - Final refinement with SLSQP using explicit constraints

4. **Key Insights**:
   - Hexagonal packing gives ~15% better density than square grid
   - Unequal circles can exploit boundary geometry - circles near edges/corners can adjust sizes differently
   - Need to balance: interior circles have more neighbors but no boundary constraints; boundary circles have fewer neighbors but are constrained by walls

5. **Implementation**: Multi-phase optimization:
   - Phase 1: Penalty-based L-BFGS-B from multiple starts
   - Phase 2: SLSQP refinement on best result
   - Phase 3: Small random perturbations + re-optimization to escape local optima

import numpy as np
from scipy.optimize import minimize


def initialize_hexagonal(n, trial=0):
    """Initialize circles in a hexagonal grid pattern with some noise."""
    rng = np.random.RandomState(42 + trial)
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.05

    cols = 6
    rows = 5
    x_spacing = 0.165
    y_spacing = 0.165 * np.sqrt(3) / 2

    idx = 0
    for row in range(rows):
        for col in range(cols):
            if idx >= n:
                break
            x = 0.08 + col * x_spacing + (row % 2) * x_spacing / 2
            y = 0.08 + row * y_spacing
            x += rng.uniform(-0.015, 0.015)
            y += rng.uniform(-0.015, 0.015)
            centers[idx] = [x, y]
            idx += 1

    return centers, radii


def initialize_random(n, trial=0):
    """Initialize circles randomly with small radii, then expand."""
    rng = np.random.RandomState(100 + trial)
    centers = rng.uniform(0.1, 0.9, (n, 2))
    radii = np.ones(n) * 0.03
    return centers, radii


def compute_pairwise_distances(centers):
    """Compute pairwise Euclidean distance matrix."""
    n = centers.shape[0]
    dist = np.zeros((n, n))
    for i in range(n):
        for j in range(i + 1, n):
            d = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            dist[i, j] = d
            dist[j, i] = d
    return dist


def penalty_objective(x, n, lam=1000.0):
    """Smooth penalty objective: maximize sum of radii subject to constraints."""
    centers = x[: 2 * n].reshape(n, 2)
    radii = x[2 * n :]

    # Primary objective: maximize sum of radii
    obj = -np.sum(radii)

    # Overlap penalty (smooth)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            overlap = radii[i] + radii[j] - dist
            if overlap > 1e-12:
                obj += lam * overlap * overlap

    # Boundary penalties
    for i in range(n):
        xi, yi = centers[i]
        ri = radii[i]
        # Left boundary
        v = xi - ri
        if v < 0:
            obj += lam * v * v
        # Right boundary
        v = 1 - xi - ri
        if v < 0:
            obj += lam * v * v
        # Bottom boundary
        v = yi - ri
        if v < 0:
            obj += lam * v * v
        # Top boundary
        v = 1 - yi - ri
        if v < 0:
            obj += lam * v * v

    # Negative radius penalty
    for i in range(n):
        if radii[i] < 0:
            obj += lam * radii[i] * radii[i]

    return obj


def constraint_functions(x, n):
    """Return array of constraint values (all should be >= 0)."""
    centers = x[: 2 * n].reshape(n, 2)
    radii = x[2 * n :]

    con = []
    # Boundary constraints
    for i in range(n):
        con.append(centers[i, 0] - radii[i])
        con.append(1 - centers[i, 0] - radii[i])
        con.append(centers[i, 1] - radii[i])
        con.append(1 - centers[i, 1] - radii[i])

    # Overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            con.append(dist - radii[i] - radii[j])

    return np.array(con)


def constraint_jacobian(x, n):
    """Compute Jacobian of constraint functions."""
    centers = x[: 2 * n].reshape(n, 2)
    radii = x[2 * n :]

    num_constraints = 4 * n + n * (n - 1) // 2
    jac = np.zeros((num_constraints, 3 * n))

    # Boundary constraint Jacobians
    for i in range(n):
        base = 4 * i
        # x_i >= r_i:  constraint = x_i - r_i
        jac[base, 3 * i] = 1.0
        jac[base, 3 * i + 2] = -1.0
        # x_i + r_i <= 1: constraint = 1 - x_i - r_i
        jac[base + 1, 3 * i] = -1.0
        jac[base + 1, 3 * i + 2] = -1.0
        # y_i >= r_i
        jac[base + 2, 3 * i + 1] = 1.0
        jac[base + 2, 3 * i + 2] = -1.0
        # y_i + r_i <= 1
        jac[base + 3, 3 * i + 1] = -1.0
        jac[base + 3, 3 * i + 2] = -1.0

    # Overlap constraint Jacobians
    idx = 4 * n
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            if dist > 1e-12:
                # constraint = dist - r_i - r_j
                jac[idx, 3 * i] = dx / dist
                jac[idx, 3 * i + 1] = dy / dist
                jac[idx, 3 * i + 2] = -1.0
                jac[idx, 3 * j] = -dx / dist
                jac[idx, 3 * j + 1] = -dy / dist
                jac[idx, 3 * j + 2] = -1.0
            idx += 1

    return jac


def objective_simple(x, n):
    """Simple objective: minimize negative sum of radii."""
    radii = x[2 * n :]
    return -np.sum(radii)


def check_validity(centers, radii, n):
    """Check if packing is valid (returns True if valid)."""
    for i in range(n):
        xi, yi = centers[i]
        ri = radii[i]
        if ri < -1e-10:
            return False
        if xi - ri < -1e-10 or xi + ri > 1 + 1e-10:
            return False
        if yi - ri < -1e-10 or yi + ri > 1 + 1e-10:
            return False
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-10:
                return False
    return True


def repair_packing(centers, radii, n):
    """Repair invalid packing by reducing radii."""
    centers = centers.copy()
    radii = radii.copy()

    # Fix boundary violations
    for i in range(n):
        xi, yi = centers[i]
        ri = radii[i]
        max_r = min(xi, 1 - xi, yi, 1 - yi, 0.5)
        if ri > max_r:
            radii[i] = max(0, max_r - 1e-8)

    # Fix overlaps by reducing radii
    max_iter = 1000
    for _ in range(max_iter):
        violated = False
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                needed = radii[i] + radii[j]
                if needed > dist + 1e-10:
                    excess = (needed - dist) / 2 + 1e-9
                    radii[i] = max(0, radii[i] - excess)
                    radii[j] = max(0, radii[j] - excess)
                    violated = True
        if not violated:
            break

    # Fix boundaries again
    for i in range(n):
        xi, yi = centers[i]
        ri = radii[i]
        max_r = min(xi, 1 - xi, yi, 1 - yi, 0.5)
        if ri > max_r:
            radii[i] = max(0, max_r - 1e-8)

    return centers, radii


def run_packing():
    """Pack 26 circles in unit square to maximize sum of radii."""
    n = 26
    best_sum = 0.0
    best_centers = np.zeros((n, 2))
    best_radii = np.zeros(n)

    # Phase 1: Multiple restarts with penalty method + L-BFGS-B
    for trial in range(15):
        if trial < 10:
            centers, radii = initialize_hexagonal(n, trial)
        else:
            centers, radii = initialize_random(n, trial)

        x0 = np.concatenate([centers.flatten(), radii])
        bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n

        # Try with increasing penalty weights
        for lam in [500.0, 2000.0, 5000.0]:
            try:
                result = minimize(
                    penalty_objective,
                    x0,
                    args=(n, lam),
                    method="L-BFGS-B",
                    bounds=bounds,
                    options={
                        "maxiter": 8000,
                        "ftol": 1e-15,
                        "gtol": 1e-10,
                    },
                )
                x0 = result.x.copy()
            except Exception:
                pass

        c_opt = x0[: 2 * n].reshape(n, 2)
        r_opt = x0[2 * n :]

        # Repair if needed
        c_opt, r_opt = repair_packing(c_opt, r_opt, n)

        current_sum = np.sum(r_opt)
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = c_opt.copy()
            best_radii = r_opt.copy()

    # Phase 2: SLSQP refinement on best result
    x0 = np.concatenate([best_centers.flatten(), best_radii])
    cons = {
        "type": "ineq",
        "fun": lambda x: constraint_functions(x, n),
        "jac": lambda x: constraint_jacobian(x, n),
    }
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n

    try:
        result = minimize(
            lambda x: objective_simple(x, n),
            x0,
            method="SLSQP",
            constraints=cons,
            bounds=bounds,
            options={"maxiter": 3000, "ftol": 1e-14},
        )
        if result.success or result.fun < objective_simple(x0, n):
            x_opt = result.x
            c_opt = x_opt[: 2 * n].reshape(n, 2)
            r_opt = x_opt[2 * n :]
            c_opt, r_opt = repair_packing(c_opt, r_opt, n)
            if np.sum(r_opt) > best_sum:
                best_sum = np.sum(r_opt)
                best_centers = c_opt.copy()
                best_radii = r_opt.copy()
    except Exception:
        pass

    # Phase 3: Perturbation-based local search
    for _ in range(20):
        rng = np.random.RandomState()
        pert_centers = best_centers.copy()
        pert_radii = best_radii.copy()

        # Add small perturbation to positions
        for i in range(n):
            pert_centers[i, 0] += rng.uniform(-0.005, 0.005)
            pert_centers[i, 1] += rng.uniform(-0.005, 0.005)
            pert_centers[i, 0] = np.clip(pert_centers[i, 0], 0.01, 0.99)
            pert_centers[i, 1] = np.clip(pert_centers[i, 1], 0.01, 0.99)

        # Slightly reduce radii to ensure feasibility after perturbation
        pert_radii *= 0.98

        x0 = np.concatenate([pert_centers.flatten(), pert_radii])
        lam = 5000.0
        try:
            result = minimize(
                penalty_objective,
                x0,
                args=(n, lam),
                method="L-BFGS-B",
                bounds=bounds,
                options={"maxiter": 5000, "ftol": 1e-15},
            )
            x_opt = result.x
            c_opt = x_opt[: 2 * n].reshape(n, 2)
            r_opt = x_opt[2 * n :]
            c_opt, r_opt = repair_packing(c_opt, r_opt, n)

            if np.sum(r_opt) > best_sum:
                best_sum = np.sum(r_opt)
                best_centers = c_opt.copy()
                best_radii = r_opt.copy()
        except Exception:
            pass

    # Phase 4: Final SLSQP refinement
    x0 = np.concatenate([best_centers.flatten(), best_radii])
    try:
        result = minimize(
            lambda x: objective_simple(x, n),
            x0,
            method="SLSQP",
            constraints=cons,
            bounds=bounds,
            options={"maxiter": 5000, "ftol": 1e-15},
        )
        x_opt = result.x
        c_opt = x_opt[: 2 * n].reshape(n, 2)
        r_opt = x_opt[2 * n :]
        c_opt, r_opt = repair_packing(c_opt, r_opt, n)
        if np.sum(r_opt) > best_sum:
            best_sum = np.sum(r_opt)
            best_centers = c_opt.copy()
            best_radii = r_opt.copy()
    except Exception:
        pass

    # Ensure non-negative radii
    best_radii = np.maximum(best_radii, 0.0)

    # Final verification and repair
    best_centers, best_radii = repair_packing(best_centers, best_radii, n)
    best_sum = np.sum(best_radii)

    return best_centers, best_radii, best_sum
