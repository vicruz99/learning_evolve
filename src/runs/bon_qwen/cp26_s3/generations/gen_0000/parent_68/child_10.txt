import numpy as np
import scipy.optimize

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square

    Args:
        centers: np.array of shape (n, 2) with (x, y) coordinates
        radii: np.array of shape (n) with radius of each circle

    Returns:
        True if valid, False otherwise
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True

def initialize_hex_grid(n_circles, square_size=1.0):
    """
    Initialize circle centers in a hexagonal grid pattern.
    """
    centers = []
    r_est = 0.09  # Initial estimated radius slightly less than 0.1 to allow 26 circles
    
    # Approximate spacing for hex grid
    h_spacing = 2 * r_est
    v_spacing = np.sqrt(3) * r_est
    
    row = 0
    while len(centers) < n_circles:
        y = r_est + row * v_spacing
        if y + r_est > square_size:
            break
            
        # Determine x-offset for hexagonal staggering
        x_offset = r_est
        if row % 2 == 1:
            x_offset += r_est  # Shift by one radius to nestle between previous row
            
        col = 0
        while True:
            x = x_offset + col * h_spacing
            if x + r_est > square_size:
                break
            
            if len(centers) < n_circles:
                centers.append([x, y])
            col += 1
        row += 1
        
    # If we didn't get enough circles, fill remaining randomly or on grid
    if len(centers) < n_circles:
        # Fallback: simple grid fill
        pass
        
    centers = np.array(centers)
    radii = np.full(n_circles, r_est)
    
    # If we generated more than needed (unlikely with logic above, but safe), trim
    if centers.shape[0] > n_circles:
        centers = centers[:n_circles]
        radii = radii[:n_circles]
        
    # If we generated fewer (shouldn't happen with r=0.09), duplicate or pad
    while centers.shape[0] < n_circles:
        centers = np.vstack([centers, centers])
        radii = np.concatenate([radii, radii])
    
    centers = centers[:n_circles]
    radii = radii[:n_circles]
    
    return centers, radii

def get_objective(vars, n_circles):
    """
    Objective function: minimize negative sum of radii.
    """
    radii = vars[2::3]
    return -np.sum(radii)

def get_constraints(vars, n_circles):
    """
    Return list of constraints for optimization.
    """
    centers = np.zeros((n_circles, 2))
    centers[:, 0] = vars[0::3]
    centers[:, 1] = vars[1::3]
    radii = vars[2::3]
    
    constraints = []
    
    # Boundary constraints: x >= r, x <= 1-r, y >= r, y <= 1-r
    # x - r >= 0
    c1 = centers[:, 0] - radii
    constraints.append({'type': 'ineq', 'fun': lambda v: v[0::3] - v[2::3]})
    
    # 1 - x - r >= 0
    constraints.append({'type': 'ineq', 'fun': lambda v: 1.0 - v[0::3] - v[2::3]})
    
    # y - r >= 0
    constraints.append({'type': 'ineq', 'fun': lambda v: v[1::3] - v[2::3]})
    
    # 1 - y - r >= 0
    constraints.append({'type': 'ineq', 'fun': lambda v: 1.0 - v[1::3] - v[2::3]})
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    # (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
    
    # To avoid defining 325 lambda functions, we define one function that computes all
    def overlap_constraints(v):
        c = np.zeros((n_circles, 2))
        c[:, 0] = v[0::3]
        c[:, 1] = v[1::3]
        r = v[2::3]
        
        # Compute pairwise distances and radii sums efficiently
        # Using broadcasting
        # dx = c[:, 0, np.newaxis] - c[np.newaxis, :, 0] # (n, n)
        # dy = c[:, 1, np.newaxis] - c[np.newaxis, :, 1]
        # dr = r[:, np.newaxis] + r[np.newaxis, :]
        
        # Upper triangle indices
        i_indices, j_indices = np.triu_indices(n_circles, k=1)
        
        dx = c[i_indices, 0] - c[j_indices, 0]
        dy = c[i_indices, 1] - c[j_indices, 1]
        dr = r[i_indices] + r[j_indices]
        
        dist_sq = dx**2 + dy**2
        rad_sum_sq = dr**2
        
        return dist_sq - rad_sum_sq

    constraints.append({'type': 'ineq', 'fun': overlap_constraints})
    
    return constraints

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n_circles = 26
    
    # 1. Initialize positions and radii
    centers_init, radii_init = initialize_hex_grid(n_circles)
    
    # Flatten into optimization variable vector [x1, y1, r1, x2, y2, r2, ...]
    x0 = np.zeros(3 * n_circles)
    x0[0::3] = centers_init[:, 0]
    x0[1::3] = centers_init[:, 1]
    x0[2::3] = radii_init
    
    # 2. Define bounds
    # x, y in [0, 1], r in [0, 0.5]
    bounds = []
    for _ in range(n_circles):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 0.5)) # r
    
    # 3. Define constraints
    cons = get_constraints(x0, n_circles)
    
    # 4. Optimize
    # Using SLSQP as it handles constraints well
    res = scipy.optimize.minimize(
        fun=get_objective,
        x0=x0,
        args=(n_circles,),
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 1000, 'ftol': 1e-9, 'disp': False}
    )
    
    if not res.success:
        # Fallback or warning if optimization failed, but we proceed with result
        pass
    
    # 5. Extract results
    final_x = res.x[0::3]
    final_y = res.x[1::3]
    final_r = res.x[2::3]
    
    final_centers = np.column_stack((final_x, final_y))
    
    # Ensure radii are not negative due to numerical noise
    final_r = np.maximum(final_r, 0.0)
    
    # Final validation and clamping to strictly satisfy constraints if needed
    # Sometimes optimizer leaves tiny violations. We can try to fix them.
    
    # Adjust centers/radii slightly to ensure validity
    # This is a safety measure
    # If any constraint is violated, we shrink radii slightly
    
    valid = validate_packing(final_centers, final_r)
    
    if not valid:
        # If invalid, we try to reduce radii until valid
        # This is a brute-force fallback
        reduction_factor = 1.0
        step = 0.05
        while not valid and reduction_factor > 0.1:
            final_r *= (1 - step)
            # Re-check boundary
            # Clamping centers if needed
            final_centers[:, 0] = np.clip(final_centers[:, 0], final_r, 1 - final_r)
            final_centers[:, 1] = np.clip(final_centers[:, 1], final_r, 1 - final_r)
            valid = validate_packing(final_centers, final_r)
            reduction_factor -= step

    sum_radii = np.sum(final_r)
    
    return final_centers, final_r, sum_radii

# To run the solution
if __name__ == "__main__":
    centers, radii, total_sum = run_packing()
    print(f"Sum of radii: {total_sum}")
    print(f"Validation: {validate_packing(centers, radii)}")
