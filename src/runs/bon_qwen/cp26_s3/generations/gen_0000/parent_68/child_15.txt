import numpy as np
from scipy.optimize import minimize

def compute_objective(x, n):
    """Returns the negative sum of radii for minimization."""
    return -np.sum(x[2*n:])

def compute_constraints(x, n):
    """Returns array of constraint values that must be >= 0."""
    # Boundary constraints: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    c_bound = np.zeros(4*n)
    for i in range(n):
        c_bound[4*i] = x[3*i] - x[3*i+2]
        c_bound[4*i+1] = 1.0 - x[3*i] - x[3*i+2]
        c_bound[4*i+2] = x[3*i+1] - x[3*i+2]
        c_bound[4*i+3] = 1.0 - x[3*i+1] - x[3*i+2]
        
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    X = x[:2*n].reshape(n, 2)
    R = x[2*n:]
    
    # Vectorized pairwise computation
    diff = X[:, np.newaxis, :] - X[np.newaxis, :, :]
    dist2 = np.sum(diff**2, axis=2)
    r_sum = R[:, np.newaxis] + R[np.newaxis, :]
    
    # Extract upper triangle to avoid duplicates and self-comparison
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    c_overlap = (dist2 - r_sum**2)[mask]
    
    return np.concatenate([c_bound, c_overlap])

def _obj_wrapper(x):
    return compute_objective(x, 26)

def _con_wrapper(x):
    return compute_constraints(x, 26)

def run_packing():
    n = 26
    # Initial guess: Staggered grid to promote hexagonal packing density
    pts = []
    for i in range(6):
        y = 0.1 + i * 0.17
        for j in range(5):
            x = 0.1 + j * 0.2 + (0.1 if i % 2 == 1 else 0.0)
            if x < 0.95 and y < 0.95:
                pts.append((x, y))
                
    if len(pts) < n:
        pts += [(0.5, 0.5), (0.2, 0.8), (0.8, 0.2)]
    pts = pts[:n]
    
    x0 = np.zeros(3*n)
    r_init = 0.04
    for k, (px, py) in enumerate(pts):
        x0[3*k] = px
        x0[3*k+1] = py
        x0[3*k+2] = r_init
        
    bounds = [(0.0, 1.0)]*(2*n) + [(0.0, 0.5)]*n
    cons = {'type': 'ineq', 'fun': _con_wrapper}
    
    # Run optimization
    res = minimize(_obj_wrapper, x0, method='SLSQP', 
                   bounds=bounds, constraints=cons, 
                   options={'maxiter': 2000, 'ftol': 1e-12})
                   
    centers = res.x[:2*n].reshape(n, 2)
    radii = res.x[2*n:]
    
    # Ensure numerical validity
    radii = np.maximum(radii, 0.0)
    centers = np.clip(centers, 0.0, 1.0)
    
    return centers, radii, np.sum(radii)
