import numpy as np
from scipy.optimize import minimize


def hex_init(n):
    """Generate hexagonal grid initial configuration."""
    centers = np.zeros((n, 2))
    idx = 0
    for row in range(7):
        for col in range(5):
            if idx >= n:
                break
            x = (col + 0.5 + (row % 2) * 0.5) * 0.17 + 0.04
            y = row * 0.155 + 0.08
            centers[idx] = [x, y]
            idx += 1
        if idx >= n:
            break
    radii = np.full(n, 0.09)
    return np.concatenate([centers.flatten(), radii])


def grid_init(n):

def random_init(n, seed):
    """Generate random initial configuration."""
    np.random.seed(seed)
    centers = np.random.uniform(0.15, 0.85, (n, 2))
    radii = np.full(n, 0.07)
    return np.concatenate([centers.flatten(), radii])


def smooth_penalty_obj(x, n):
    """Objective with smooth penalty for constraints."""
    centers = x[:2 * n].reshape(n, 2)
    radii = x[2 * n:]

    obj = -np.sum(radii)
    penalty = 0.0
    alpha = 50000.0

    # Boundary penalties (smooth)
    for i in range(n):
        xi = centers[i, 0]
        yi = centers[i, 1]
        ri = radii[i]

        # x >= r
        gap = xi - ri
        if gap < 0:
            penalty += alpha * gap * gap
        # x <= 1-r
        gap = 1.0 - xi - ri
        if gap < 0:
            penalty += alpha * gap * gap
        # y >= r
        gap = yi - ri
        if gap < 0:
            penalty += alpha * gap * gap
        # y <= 1-r
        gap = 1.0 - yi - ri
        if gap < 0:
            penalty += alpha * gap * gap

    # Overlap penalties (smooth)
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            d = np.sqrt(dx * dx + dy * dy)
            gap = d - radii[i] - radii[j]
            if gap < 0:
                penalty += alpha * gap * gap

    return obj + penalty


def barrier_obj(x, n):
    """Objective with log barrier for constraints."""
    centers = x[:2 * n].reshape(n, 2)
    radii = x[2 * n:]

    obj = -np.sum(radii)
    barrier = 0.0
    mu = 0.005
    eps = 1e-8

    # Boundary barriers
    for i in range(n):
        xi = centers[i, 0]
        yi = centers[i, 1]
        ri = radii[i]

        barrier += -mu * np.log(xi - ri + eps)
        barrier += -mu * np.log(1.0 - xi - ri + eps)
        barrier += -mu * np.log(yi - ri + eps)
        barrier += -mu * np.log(1.0 - yi - ri + eps)

    # Overlap barriers
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            d = np.sqrt(dx * dx + dy * dy + eps)
            barrier += -mu * np.log(d - radii[i] - radii[j] + eps)

    return obj + barrier


def get_bounds(n):
    """Get bounds for optimization variables."""
    bounds = []
    for i in range(2 * n):
        bounds.append((0.0, 1.0))
    for i in range(n):
        bounds.append((0.0, 0.25))
    return bounds


def make_perturbed_hex(n, seed):
    """Create perturbed hexagonal initial configuration."""
    np.random.seed(seed)
    centers = np.zeros((n, 2))
    idx = 0
    for row in range(7):
        for col in range(5):
            if idx >= n:
                break
            x = (col + 0.5 + (row % 2) * 0.5) * 0.17 + 0.04
            y = row * 0.155 + 0.08
            centers[idx] = [x, y]
            idx += 1
        if idx >= n:
            break

    # Add perturbation
    perturbation = np.random.uniform(-0.04, 0.04, (n, 2))
    centers = centers + perturbation
    centers = np.clip(centers, 0.05, 0.95)

    # Vary radii slightly
    radii = np.full(n, 0.09)
    radii = radii + np.random.uniform(-0.015, 0.015, n)
    radii = np.clip(radii, 0.04, 0.15)

    return np.concatenate([centers.flatten(), radii])


def make_two_scale_init(n, seed):
    """Initialize with two different radii scales."""
    np.random.seed(seed)
    centers = np.zeros((n, 2))
    idx = 0

    # Place larger circles first in interior
    large_count = 12
    for row in range(5):
        for col in range(5):
            if idx >= n:
                break
            x = (col + 0.5) * 0.22 + 0.08
            y = (row + 0.5) * 0.22 + 0.08
            centers[idx] = [x, y]
            idx += 1
        if idx >= n:
            break

    # Fill remaining with smaller circles in gaps
    while idx < n:
        centers[idx] = [np.random.uniform(0.05, 0.95), np.random.uniform(0.05, 0.95)]
        idx += 1

    radii = np.zeros(n)
    radii[:large_count] = 0.11
    radii[large_count:] = 0.07

    return np.concatenate([centers.flatten(), radii])


def clip_to_feasible(centers, radii, n):
    """Post-process to ensure feasibility by clipping radii."""
    radii_new = radii.copy()

    # Clip to boundary constraints
    for i in range(n):
        ri = radii_new[i]
        max_r = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        radii_new[i] = min(ri, max(0, max_r))

    # Iteratively clip for overlap
    for iteration in range(10):
        changed = False
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                d = np.sqrt(dx * dx + dy * dy)
                if d < radii_new[i] + radii_new[j]:
                    excess = radii_new[i] + radii_new[j] - d
                    # Reduce the larger radius
                    if radii_new[i] >= radii_new[j]:
                        radii_new[i] = max(0, radii_new[i] - excess / 2 - 1e-10)
                    else:
                        radii_new[j] = max(0, radii_new[j] - excess / 2 - 1e-10)
                    changed = True
        if not changed:
            break

    return centers, radii_new


def refine_local(centers, radii, n, max_iter=5000):
    """Local refinement using penalty method."""
    x0 = np.concatenate([centers.flatten(), radii])
    bounds = get_bounds(n)

    def local_obj(x):
        return smooth_penalty_obj(x, n)

    result = minimize(local_obj, x0, method='L-BFGS-B', bounds=bounds,
                     options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-12})

    centers_opt = result.x[:2 * n].reshape(n, 2)
    radii_opt = result.x[2 * n:]
    return centers_opt, radii_opt
