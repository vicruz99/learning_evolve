from scipy.optimize import minimize


def make_constraints(n):
    def constraints(x):
        c = x[:2 * n].reshape((n, 2))
        r = x[2 * n:]

        cons = []

        for i in range(n):
            cons.append(c[i, 0] - r[i] - 1e-10)
            cons.append(1.0 - c[i, 0] - r[i] - 1e-10)
            cons.append(c[i, 1] - r[i] - 1e-10)
            cons.append(1.0 - c[i, 1] - r[i] - 1e-10)

        for i in range(n):
            for j in range(i + 1, n):
                dx = c[i, 0] - c[j, 0]
                dy = c[i, 1] - c[j, 1]
                dist = np.sqrt(dx ** 2 + dy ** 2 + 1e-20)
                cons.append(dist - r[i] - r[j] - 1e-10)

        return np.array(cons)

    return constraints


def hexagonal_init(n, seed=0):
    rng = np.random.RandomState(seed)

    centers = []
    radii = []

    rows_config = [5, 4, 5, 4, 5, 3]

    row_height = 0.155
    col_width = 0.185

    idx = 0
    row = 0
    for cols in rows_config:
        for col in range(cols):
            if idx >= n:
                break
            base_x = col * col_width + 0.11
            base_y = row * row_height + 0.10
            if row % 2 == 1:
                base_x += col_width / 2.0

            x = base_x + rng.uniform(-0.02, 0.02)
            y = base_y + rng.uniform(-0.02, 0.02)

            y = np.clip(y, 0.05, 0.95)

            centers.append([x, y])
            radii.append(0.085 + rng.uniform(-0.01, 0.01))
            idx += 1
        if idx >= n:
            break
        row += 1

    return np.array(centers[:n]), np.array(radii[:n])


def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any():
        return False

    if np.isnan(radii).any():
        return False

    for i in range(n):
        if radii[i] < 0:
            return False

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            return False

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False

    return True
