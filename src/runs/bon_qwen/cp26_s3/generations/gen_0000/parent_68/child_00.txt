To maximize the sum of radii for 26 circles in a unit square, we recognize that the optimal configuration typically consists of equal or nearly equal circles. This transforms the problem into a maximin problem: find positions $C = \{c_1, \dots, c_{26}\}$ that maximize the minimum distance between any two circles and between any circle and the boundary. The optimal radius will then be half this minimum distance.

Strategy:
1. **Objective Function**: Define a function that computes the minimum pairwise distance and minimum boundary distance for a given set of centers. We want to maximize this value.

3. **Optimization**: Apply `scipy.optimize.minimize` with the `Nelder-Mead` method. This derivative-free optimizer is robust for non-smooth maximin objectives and handles the 52-dimensional search space (2 coordinates per circle) efficiently.
4. **Validation Safety**: After optimization, slightly shrink the computed optimal radius and clamp centers to ensure strict compliance with the validation function's $10^{-12}$ tolerance, accounting for floating-point precision limits.

import numpy as np
from scipy.optimize import minimize

def compute_min_dist(centers):
    """
    Compute the minimum distance between all pairs of circles 
    and the minimum distance to the square boundaries.
    """
    n = centers.shape[0]
    
    # Vectorized pairwise squared distances
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists_sq = np.sum(diffs**2, axis=2)
    np.fill_diagonal(dists_sq, np.inf)
    min_pair = np.sqrt(np.min(dists_sq))
    
    # Minimum distance to boundaries [0, 1]
    min_x = np.min(np.minimum(centers[:, 0], 1.0 - centers[:, 0]))
    min_y = np.min(np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    min_bound = np.min(min_x, min_y)
    
    return min(min_pair, min_bound)

def objective(centers_flat):
    """
    Negative minimum distance to be minimized.
    """
    centers = centers_flat.reshape(-1, 2)
    return -compute_min_dist(centers)

def run_packing():
    n = 26
    best_centers = None
    best_min_dist = -1.0
    
    # Prepare multiple initialization strategies
    inits = []
    rng = np.random.RandomState(42)
    
    # 1. Random uniform within safe margins
    inits.append(rng.rand(n, 2) * 0.6 + 0.2)
    
    # 2. Hexagonal lattice packing
    hex_c = []
    r_h = 0.105
    y = r_h
    row = 0
    while len(hex_c) < n:
        x = r_h if row % 2 == 0 else r_h * 1.5
        while x < 1.0 - r_h:
            if len(hex_c) < n:
                hex_c.append([x, y])
            x += r_h * 2
        y += r_h * np.sqrt(3)
        row += 1
    inits.append(np.array(hex_c[:n]))
    
    # 3. Perturbed 5x5 grid (25 points) + 1 center point
    grid_c = []
    for i in range(5):
        for j in range(5):
            if len(grid_c) < n:
                grid_c.append([(j + 0.5) / 5.0, (i + 0.5) / 5.0])
    if len(grid_c) < n:
        grid_c.append([0.5, 0.5])
    inits.append(np.array(grid_c[:n]) + rng.randn(n, 2) * 0.02)

    # Optimize from each initialization
    for init in inits:
        init = np.clip(init, 0.15, 0.85)
        res = minimize(objective, init.flatten(), method='Nelder-Mead',
                       options={'maxiter': 50000, 'xatol': 1e-9, 'fatol': 1e-11})
        
        if -res.fun > best_min_dist:
            best_min_dist = -res.fun
            best_centers = res.x.reshape(-1, 2)
            
    # Slight shrinkage to guarantee validation passes with 1e-12 tolerance
    r = (best_min_dist - 1e-10) / 2.0
    best_centers = np.clip(best_centers, r + 1e-9, 1.0 - r - 1e-9)
    
    radii = np.full(n, r)
    sum_r = float(np.sum(radii))
    
    return best_centers, radii, sum_r
