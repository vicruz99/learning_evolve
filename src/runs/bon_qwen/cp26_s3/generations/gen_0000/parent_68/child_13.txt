import numpy as np
from scipy.optimize import minimize
import math


def compute_penalty(centers, radii, n, alpha):
    """Compute constraint violation penalty"""
    penalty = 0.0
    
    # Boundary violations
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < 0:
            penalty += alpha * (x - r) ** 2
        if x + r > 1:
            penalty += alpha * (x + r - 1) ** 2
        if y - r < 0:
            penalty += alpha * (y - r) ** 2
        if y + r > 1:
            penalty += alpha * (y + r - 1) ** 2
    
    # Overlap violations
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = math.sqrt(dx * dx + dy * dy)
            overlap = radii[i] + radii[j] - dist
            if overlap > 0:
                penalty += alpha * overlap ** 2
    
    return penalty


def objective_function(vars, n, alpha):
    """Objective: maximize sum of radii (minimize negative) with penalty"""
    centers = vars[:2 * n].reshape((n, 2))
    radii = vars[2 * n:]
    
    # Main objective: negative sum of radii
    obj = -np.sum(radii)
    
    # Add penalty for constraint violations
    pen = compute_penalty(centers, radii, n, alpha)
    
    return obj + pen


def check_validity(n, centers, radii):
    """Check if packing is valid"""
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if r < -1e-12 or np.isnan(r):
            return False
        if np.isnan(x) or np.isnan(y):
            return False
        if x - r < -1e-10 or x + r > 1 + 1e-10:
            return False
        if y - r < -1e-10 or y + r > 1 + 1e-10:
            return False
    
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = math.sqrt(dx * dx + dy * dy)

def create_initial_hexagonal(n):

    ]
    
    idx = 0

    y_spacing = 0.17

            if idx < n:
                x = x_start + col * x_spacing
                centers[idx] = [x, y_pos]
                radii[idx] = 0.065
                idx += 1
        y_pos += y_spacing
    
    # Fill any remaining spots
    while idx < n:
        centers[idx] = [0.5, 0.5]
        radii[idx] = 0.01
        idx += 1
    
    return centers, radii


def create_random_initial(n, seed):
    """Create random initial configuration"""
    np.random.seed(seed)
    centers = np.random.uniform(0.15, 0.85, (n, 2))
    radii = np.full(n, 0.05)
    return centers, radii


def optimize_with_penalty(centers, radii, n, alpha, max_iter=2000):
    """Run optimization with given penalty weight"""
    vars0 = np.concatenate([centers.flatten(), radii])
    
    # Bounds: centers in [0.001, 0.999], radii in [0.001, 0.5]
    bounds = []
    for i in range(n):
        bounds.extend([(0.001, 0.999), (0.001, 0.999)])
        bounds.append((0.001, 0.5))
    
    result = minimize(
        objective_function,
        vars0,
        args=(n, alpha),
        method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-10}
    )
    
    opt_centers = result.x[:2 * n].reshape((n, 2))
    opt_radii = result.x[2 * n:]
    
    return opt_centers, opt_radii, result.success


def refine_radii(centers, radii, n, margin=1e-4):
    """Reduce radii slightly to ensure validity, then try to increase"""
    # First, ensure validity by shrinking
    for i in range(n):
        x, y = centers[i]
        max_r = min(x, 1 - x, y, 1 - y)
        for j in range(n):
            if i != j:
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = math.sqrt(dx * dx + dy * dy)
                max_r = min(max_r, dist - radii[j])
        radii[i] = max(0.001, max_r - margin)
    
    return radii


def try_increase_radii(centers, radii, n, steps=10):
    """Try to increase radii greedily"""
    for step in range(steps):

            x, y = centers[i]
            current_r = radii[i]
            
            # Compute maximum possible radius for circle i

                    dy = centers[i, 1] - centers[j, 1]
                    dist = math.sqrt(dx * dx + dy * dy)
                    max_r = min(max_r, dist - radii[j])
            
            max_r = max(0.001, max_r)
            
            if max_r > current_r + 1e-6:
                new_r = (current_r + max_r) / 2
                if new_r > current_r + 1e-8:
                    radii[i] = new_r
                    improved = True
        
        if not improved:
            break
    
    return radii
