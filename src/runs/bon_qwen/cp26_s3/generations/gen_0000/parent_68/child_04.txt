2.  **Center Optimization (SLSQP)**: We use `scipy.optimize.minimize` with the 'SLSQP' method to maximize a common radius $r$ for all circles. 3.  **Radius Optimization (Linear Programming)**: Once an optimal center configuration is found, we relax the constraint that all radii must be equal. We formulate a Linear Program (LP) to maximize the sum of individual radii $\sum r_i$, subject to the distance constraints defined by the fixed centers and boundary limits. This allows boundary circles to shrink slightly if it permits inner circles to expand, or allows for better gap utilization.

import numpy as np
from scipy.optimize import minimize, linprog

def get_hex_centers(n, width=1.0, height=1.0, scale=0.9):
    """Generate initial centers in a hexagonal pattern."""
    # Estimate grid dimensions
    rows = int(np.ceil(np.sqrt(n * 1.5))) 
    cols = int(np.ceil(n / rows))
    
    centers = []
    count = 0
    r_est = 0.05 # Initial small radius estimate
    
    # Hexagonal spacing
    row_h = r_est * np.sqrt(3)
    col_w = 2 * r_est
    
    for i in range(rows):
        for j in range(cols):
            if count >= n:
                break
            x = (j + 0.5 + (i % 2) * 0.5) * col_w
            y = i * row_h
            centers.append([x, y])
            count += 1
        if count >= n:
            break
            
    centers = np.array(centers[:n])
    
    # Normalize to unit square with margin
    if len(centers) > 0:
        min_c = centers.min(axis=0)
        max_c = centers.max(axis=0)
        span = max_c - min_c
        # Scale to fit in [margin, 1-margin]
        margin = 0.1
        if span[0] > 0:
            centers[:, 0] = margin + (centers[:, 0] - min_c[0]) / span[0] * (1 - 2 * margin)
        if span[1] > 0:
            centers[:, 1] = margin + (centers[:, 1] - min_c[1]) / span[1] * (1 - 2 * margin)
            
    return centers

def run_packing() -> tuple:
    n = 26
    
    # 1. Initialization
    centers_init = get_hex_centers(n)
    
    # 2. Optimize centers for equal radii (Maximize r)
    # Variables: [x1, y1, ..., x26, y26, r]
    # Objective: minimize -r
    def obj_equal(v):
        return -v[-1] # r is the last variable

    def constraints_equal(v):
        r = v[-1]
        c = v[:-1].reshape(-1, 2)
        cons = []
        # Boundary constraints
        for i in range(n):
            cons.append(c[i, 0] - r) # x >= r
            cons.append(1 - c[i, 0] - r) # x <= 1-r
            cons.append(c[i, 1] - r) # y >= r
            cons.append(1 - c[i, 1] - r) # y <= 1-r
        # Separation constraints
        for i in range(n):
            for j in range(i + 1, n):
                dist_sq = np.sum((c[i] - c[j])**2)
                cons.append(dist_sq - (2 * r)**2) # dist >= 2r
        return np.array(cons)

    # Initial guess
    v0 = np.concatenate([centers_init.flatten(), [0.05]])
    
    # Bounds for x, y, r
    bounds = [(0, 1)] * (2 * n) + [(0, 0.5)] # r <= 0.5
    
    # SLSQP solver
    res = minimize(obj_equal, v0, method='SLSQP', bounds=bounds, 
                   constraints={'type': 'ineq', 'fun': constraints_equal},
                   options={'maxiter': 1000, 'ftol': 1e-9})
    
    best_v = res.x
    centers_opt = best_v[:-1].reshape(-1, 2)
    r_common = best_v[-1]
    
    # 3. Optimize individual radii using Linear Programming
    # Maximize sum(r_i) s.t. r_i + r_j <= dist(i,j) and boundary constraints
    # This is an LP.
    
    # Distance matrix
    dists = np.zeros((n, n))
    for i in range(n):
        for j in range(n):
            dists[i, j] = np.linalg.norm(centers_opt[i] - centers_opt[j])
            
    # LP Setup: minimize -sum(r_i)
    c_obj = -np.ones(n)
    
    # Constraints: A_ub r <= b_ub
    # Boundary constraints:
    # r_i <= x_i
    # r_i <= 1 - x_i
    # r_i <= y_i
    # r_i <= 1 - y_i
    # Pairwise: r_i + r_j <= dist(i, j)
    
    A_ub = []
    b_ub = []
    
    for i in range(n):
        # r_i <= x_i
        row = np.zeros(n)
        row[i] = 1
        A_ub.append(row)
        b_ub.append(centers_opt[i, 0])
        
        # r_i <= 1 - x_i
        row = np.zeros(n)
        row[i] = 1
        A_ub.append(row)
        b_ub.append(1 - centers_opt[i, 0])
        
        # r_i <= y_i
        row = np.zeros(n)
        row[i] = 1
        A_ub.append(row)
        b_ub.append(centers_opt[i, 1])
        
        # r_i <= 1 - y_i
        row = np.zeros(n)
        row[i] = 1
        A_ub.append(row)
        b_ub.append(1 - centers_opt[i, 1])
        
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1
            row[j] = 1
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    bounds_r = [(0, None) for _ in range(n)]
    
    # Solve LP
    lp_res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method='highs')
    
    if lp_res.success:
        radii_opt = lp_res.x
    else:
        # Fallback to common radius if LP fails
        radii_opt = np.full(n, r_common)

    # Calculate sum of radii
    sum_radii = np.sum(radii_opt)
    
    return centers_opt, radii_opt, sum_radii
