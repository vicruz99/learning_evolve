from scipy.optimize import minimize
import math

def get_initial_centers(n=26):
    """
    Generates an initial configuration of n circles using a hexagonal packing pattern.
    """
    centers = []
    # Estimate radius to fit in unit square roughly
    # For n=26, radius approx 0.1
    r_est = 0.08 
    
    # Hexagonal packing logic
    # Rows
    row_y = r_est
    row_idx = 0
    
    # We want to fill the square. 
    # Let's just place them in a hexagonal grid and then let the optimizer fix positions.
    # Grid spacing
    dx = 2 * r_est
    dy = math.sqrt(3) * r_est
    
    y = r_est
    while y <= 1 - r_est:
        x = r_est
        # Offset every other row
        if row_idx % 2 == 1:
            x = r_est + r_est # Shift by r? No, shift by r means distance is sqrt(r^2 + (r*sqrt3)^2) = 2r. Correct.
            # Actually, standard hex packing shift is r horizontally.
            # But let's just shift by r_est to ensure distance.
        
        while x <= 1 - r_est:
            if len(centers) < n:
                centers.append([x, y])
            x += dx
        y += dy
        row_idx += 1
    
    # If we have fewer than n, add some random valid positions or just pad
    while len(centers) < n:
        # Random position
        cx = np.random.uniform(r_est, 1 - r_est)
        cy = np.random.uniform(r_est, 1 - r_est)
        centers.append([cx, cy])
        
    # If we have more (unlikely with loop), truncate
    centers = centers[:n]
    
    return np.array(centers)

        centers[i, 0] = np.clip(centers[i, 0], r, 1 - r)
        centers[i, 1] = np.clip(centers[i, 1], r, 1 - r)
