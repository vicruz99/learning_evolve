import numpy as np
from scipy.optimize import linprog, differential_evolution, minimize

def get_max_radii_sum(centers):
    """
    Given fixed centers, solve the LP to find radii that maximize the sum of radii.
    Constraints:
    r_i + r_j <= dist(i, j)
    r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    r_i >= 0
    """
    n = centers.shape[0]
    
    # Objective: maximize sum(r) => minimize -sum(r)
    c = -np.ones(n)
    
    # Inequality constraints A_ub @ r <= b_ub
    # We will build these dynamically
    
    A_rows = []
    b_rows = []
    
    # Pairwise distance constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_rows.append(row)
            b_rows.append(dist)
            
    # Boundary constraints
    # r_i <= x_i  => r_i <= centers[i, 0]
    # r_i <= 1-x_i => r_i <= 1 - centers[i, 0]
    # r_i <= y_i  => r_i <= centers[i, 1]
    # r_i <= 1-y_i => r_i <= 1 - centers[i, 1]
    
    for i in range(n):
        x, y = centers[i]
        # r_i <= x
        row = np.zeros(n)
        row[i] = 1.0
        A_rows.append(row)
        b_rows.append(x)
        
        # r_i <= 1-x
        row = np.zeros(n)
        row[i] = 1.0
        A_rows.append(row)
        b_rows.append(1 - x)
        
        # r_i <= y
        row = np.zeros(n)
        row[i] = 1.0
        A_rows.append(row)
        b_rows.append(y)
        
        # r_i <= 1-y
        row = np.zeros(n)
        row[i] = 1.0
        A_rows.append(row)
        b_rows.append(1 - y)

    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)
    
    # Bounds for r_i >= 0
    bounds = [(0, None) for _ in range(n)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return -res.fun, res.x
        else:
            # If LP fails, return 0 sum and small radii to avoid crash
            return 0.0, np.zeros(n)
    except Exception:
        return 0.0, np.zeros(n)
