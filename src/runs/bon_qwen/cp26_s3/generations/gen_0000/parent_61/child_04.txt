2. **Optimization Formulation**: We directly maximize $\sum r_i$ subject to:

3. **Solver Choice**: `scipy.optimize.minimize` with the `SLSQP` algorithm handles the non-linear inequality constraints efficiently.

import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def _objective(vars_flat):
    """Objective function: maximize sum of radii (minimize negative sum)"""
    radii = vars_flat.reshape(N_CIRCLES, 3)[:, 2]
    return -np.sum(radii)

def _constraint_func(vars_flat):
    """
    Computes all constraints as an array of values >= 0.
    Includes pairwise non-overlap and boundary containment.
    """
    c = vars_flat.reshape(N_CIRCLES, 3)
    xs = c[:, 0]
    ys = c[:, 1]
    rs = c[:, 2]

    cons = []
    
    # Pairwise distance constraints: dist(i,j) >= r_i + r_j
    iu, ju = np.triu_indices(N_CIRCLES, k=1)
    dx = xs[:, None] - xs
    dy = ys[:, None] - ys
    dists = np.sqrt(dx**2 + dy**2)
    r_sum = rs[:, None] + rs
    cons.extend(dists[iu, ju] - r_sum[iu, ju])

    # Boundary constraints: r <= x <= 1-r  and  r <= y <= 1-r
    cons.extend(xs - rs)
    cons.extend(1.0 - (xs + rs))
    cons.extend(ys - rs)
    cons.extend(1.0 - (ys + rs))
    
    return np.array(cons)

def _get_initial_guess():
    """Generate a hexagonal lattice configuration as a strong starting point"""
    centers = []
    # Row configuration for 26 circles: 6, 5, 6, 5, 4
    row_counts = [6, 5, 6, 5, 4]
    y_pos = 0.10
    
    for i, count in enumerate(row_counts):
        for j in range(count):
            if i % 2 == 0:
                x_pos = 0.12 + j * 0.145
            else:
                x_pos = 0.24 + j * 0.145
            # Start with smaller radii to ensure feasibility
            centers.append([x_pos, y_pos, 0.06])
        y_pos += 0.125
        
    return np.array(centers).flatten()

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes circle packing to maximize sum of radii.
    Returns (centers, radii, sum_radii)
    """
    np.random.seed(42)
    x0 = _get_initial_guess()
    
    # Variable bounds: x,y in [0,1], r in [0, 0.5]
    bounds = [(0.0, 1.0)] * (2 * N_CIRCLES) + [(0.0, 0.5)] * N_CIRCLES
    cons = {'type': 'ineq', 'fun': _constraint_func}
    
    best_res = None
    best_val = -np.inf

    # Multiple restarts to escape local minima
    for _ in range(3):
        # Small random perturbation to initial guess
        x0_trial = x0 + np.random.normal(0, 0.005, size=x0.shape)
        # Clip to keep within reasonable bounds initially
        x0_trial = np.clip(x0_trial.reshape(N_CIRCLES, 3), 0.0, 1.0).flatten()
        
        res = minimize(
            _objective, 
            x0_trial, 
            method='SLSQP', 
            bounds=bounds,
            constraints=cons, 
            options={'maxiter': 500, 'ftol': 1e-9, 'disp': False}
        )
        
        if res.success:
            curr_val = -res.fun
            if curr_val > best_val:
                best_val = curr_val
                best_res = res

    # Fallback if optimization fails (unlikely with this setup)
    if best_res is None:
        centers = np.random.rand(N_CIRCLES, 2) * 0.5 + 0.25
        radii = np.full(N_CIRCLES, 0.05)
        return centers, radii, float(np.sum(radii))

    final_state = best_res.x.reshape(N_CIRCLES, 3)
    centers = final_state[:, :2]
    radii = final_state[:, 2]
    return centers, radii, float(np.sum(radii))
