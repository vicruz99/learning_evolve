import numpy as np
from scipy.optimize import linprog, minimize

def solve_radii(centers):
    """
    Solves the Linear Programming problem to find optimal radii for fixed centers.
    Maximizes sum of radii subject to non-overlap and boundary constraints.
    """
    n = len(centers)
    if n == 0:
        return 0.0, np.array([])

    # Objective: Maximize sum(r) => Minimize -sum(r)
    c = -np.ones(n)

    # Constraints: r_i + r_j <= distance(i, j)
    A_ub = []
    b_ub = []

    # Pairwise distance constraints
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dist)

    # If constraints list is empty (1 circle), handle gracefully
    if not A_ub:
        A_ub = np.empty((0, n))
        b_ub = np.empty(0)
    else:
        A_ub = np.array(A_ub)
        b_ub = np.array(b_ub)

    # Bounds for each radius: 0 <= r_i <= min_dist_to_boundary
    bounds = []
    for i in range(n):
        x, y = centers[i]
        # Distance to boundaries: x, 1-x, y, 1-y
        max_r = min(x, 1 - x, y, 1 - y)
        # Ensure non-negative bound
        if max_r < 0:
            max_r = 0
        bounds.append((0, max_r))

    # Solve LP
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')

    if res.success:
        return -res.fun, res.x
    else:
        # Fallback if LP fails (e.g., infeasible due to numerical issues)
        return 0.0, np.zeros(n)

def objective_with_bounds(centers_flat):
    """
    Objective function for scipy optimizer.
    Reshapes centers, penalizes out-of-bounds, and returns negative max sum of radii.
    """
    centers = centers_flat.reshape(-1, 2)
    
    # Penalty for boundaries
    penalty = 0.0
    for pt in centers:
        x, y = pt
        if x < 0 or x > 1 or y < 0 or y > 1:
            penalty += 1000.0 # Strong penalty

    sum_radii, _ = solve_radii(centers)
    
    # We want to maximize sum_radii, so minimize -sum_radii + penalty
    return -sum_radii + penalty

def run_packing():

    Main function to pack 26 circles.
    Returns (centers, radii, sum_radii).

    n = 26
    
    # --- Step 1: Initial Guess (Hexagonal Lattice) ---
    centers = []
    r_est = 0.10 # Initial estimate for radius
    
    # Generate points in a hexagonal pattern
    # 5 rows seems reasonable for r ~ 0.1
    rows_to_fill = 6
    for row in range(rows_to_fill):
        y = r_est + row * r_est * np.sqrt(3)
        
        # Alternate row offsets for hexagonal packing
        if row % 2 == 0:
            # Even row: standard spacing
            # x coordinates: r, 3r, 5r...
            max_x = 1 - r_est
            x = r_est
            while x <= max_x and len(centers) < n:
                centers.append([x, y])
                x += 2 * r_est
        else:
            # Odd row: shifted by r
            # x coordinates: 2r, 4r, 6r...
            max_x = 1 - r_est
            x = 2 * r_est
            while x <= max_x and len(centers) < n:
                centers.append([x, y])
                x += 2 * r_est

    # If we need more circles (unlikely with 6 rows, but safe guard), pad with random
    while len(centers) < n:
        centers.append([np.random.uniform(0, 1), np.random.uniform(0, 1)])
    
    # Trim to exactly 26
    centers = np.array(centers[:n])

    # Flatten centers for optimizer
