import numpy as np
from scipy.optimize import linprog
import math

def get_pair_index(i, j, n):
    """
    Maps a pair (i, j) with i < j to a unique index 0..n(n-1)/2 - 1.
    """
    if i > j:
        i, j = j, i
    return i * n - (i * (i + 1)) // 2 + j - (i + 1)

def solve_lp(centers, n):
    """
    Solves the Linear Programming problem to maximize sum of radii for fixed centers.
    
    Args:
        centers: np.array (n, 2)
        n: int
    
    Returns:
        radii: np.array (n,)
        sum_radii: float
        duals: np.array (num_constraints,) - shadow prices
        success: bool
    """
    num_vars = n
    
    # Objective: Maximize sum(r_i) => Minimize -sum(r_i)
    c_obj = -np.ones(num_vars)
    
    # Constraints:
    # 1. Pairwise: r_i + r_j <= dist(i, j)
    # 2. Boundary: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    # 3. Non-negativity: r_i >= 0 (handled by bounds)
    
    # Calculate distances and boundary limits
    num_pairs = n * (n - 1) // 2
    num_boundaries = 4 * n
    num_constraints = num_pairs + num_boundaries
    
    A_ub = np.zeros((num_constraints, num_vars))
    b_ub = np.zeros(num_constraints)
    
    # Fill Pairwise Constraints
    pair_idx = 0
    dists = np.zeros((n, n)) # Store distances for gradient
    
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            d = math.hypot(dx, dy)
            dists[i, j] = d
            dists[j, i] = d
            
            # Avoid division by zero later by ensuring min distance
            if d < 1e-12:
                d = 1e-12 
                # Force centers apart slightly to maintain numerical stability? 
                # Actually, just setting d small is fine, r will be small.
                # But gradient calculation needs valid direction.
            
            A_ub[pair_idx, i] = 1.0
            A_ub[pair_idx, j] = 1.0
            b_ub[pair_idx] = d
            pair_idx += 1
            
    # Fill Boundary Constraints
    # r_i <= x_i  =>  r_i - x_i <= 0 (Wait, standard is Ax <= b)
    # Constraint: 1*r_i <= x_i
    # Constraint: 1*r_i <= 1 - x_i
    # Constraint: 1*r_i <= y_i
    # Constraint: 1*r_i <= 1 - y_i
    
    for i in range(n):
        x, y = centers[i]
        
        # Left: r_i <= x
        idx = num_pairs + 4 * i + 0
        A_ub[idx, i] = 1.0
        b_ub[idx] = x
        
        # Right: r_i <= 1 - x
        idx = num_pairs + 4 * i + 1
        A_ub[idx, i] = 1.0
        b_ub[idx] = 1.0 - x
        
        # Bottom: r_i <= y
        idx = num_pairs + 4 * i + 2
        A_ub[idx, i] = 1.0
        b_ub[idx] = y
        
        # Top: r_i <= 1 - y
        idx = num_pairs + 4 * i + 3
        A_ub[idx, i] = 1.0
        b_ub[idx] = 1.0 - y
        
    bounds = [(0, None) for _ in range(num_vars)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            radii = res.x
            sum_radii = -res.fun
            # Marginals correspond to inequality constraints (A_ub x <= b_ub)
            # In scipy >= 1.9, marginals are in result.ineqlin.marginals
            # Note: linprog minimizes, so marginal is -d(lambda)/d(b). 
            # But for maximization of sum (min of -sum), sensitivity is positive if relaxing constraint helps.
            # Actually, standard interpretation: dual variable mu >= 0 for min problem.
            # d(opt)/d(b) = -mu.
            # Here opt = -Sum. So d(-Sum)/db = -mu => d(Sum)/db = mu.
            # So marginals are directly the sensitivities we want (positive contribution to Sum).
            duals = res.ineqlin.marginals if hasattr(res, 'ineqlin') and hasattr(res.ineqlin, 'marginals') else np.zeros(num_constraints)
            return radii, sum_radii, duals, True
        else:
            return np.zeros(n), 0.0, np.zeros(num_constraints), False
    except Exception:
        return np.zeros(n), 0.0, np.zeros(num_constraints), False

def compute_gradient(centers, duals, n):
    """
    Computes gradient of Sum(radii) w.r.t centers using LP duals.
    """
    num_pairs = n * (n - 1) // 2
    grad_x = np.zeros(n)
    grad_y = np.zeros(n)
    
    # 1. Pairwise constraints contribution
    # Duals[0..num_pairs-1] correspond to r_i + r_j <= dist(i, j)
    # Gradient contribution: mu_ij * (c_i - c_j) / dist(i, j)
    
    # We iterate pairs again to reconstruct which dual belongs to which pair
    pair_idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            mu = duals[pair_idx]
            pair_idx += 1
            
            if mu > 1e-12: # Only active constraints matter significantly
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = math.hypot(dx, dy)
                
                if dist > 1e-12:
                    unit_x = dx / dist
                    unit_y = dy / dist
                    
                    grad_x[i] += mu * unit_x
                    grad_y[i] += mu * unit_y
                    
                    grad_x[j] -= mu * unit_x
                    grad_y[j] -= mu * unit_y
                else:
                    # If centers coincide, push apart randomly or ignore
                    # Gradient undefined, but mu implies we want to increase distance.
                    pass

    # 2. Boundary constraints contribution
    # Duals[num_pairs .. end]
    # Mapping: for circle i, indices 4i+0 (L), 4i+1 (R), 4i+2 (B), 4i+3 (T)
    
    for i in range(n):
        base = num_pairs + 4 * i
        
        # Left: r_i <= x_i. RHS is x_i. Dual v_L.
        # d(Sum)/d(x_i) contribution: +v_L (since increasing x_i relaxes constraint)
        v_L = duals[base + 0]
        
        # Right: r_i <= 1 - x_i. RHS is 1 - x_i. Dual v_R.
        # d(Sum)/d(x_i) contribution: -v_R (since increasing x_i tightens constraint RHS 1-x_i decreases)
        v_R = duals[base + 1]
        
        # Bottom: r_i <= y_i. RHS is y_i. Dual v_B.
        # d(Sum)/d(y_i) contribution: +v_B
        v_B = duals[base + 2]
        
        # Top: r_i <= 1 - y_i. RHS is 1 - y_i. Dual v_T.
        # d(Sum)/d(y_i) contribution: -v_T
        v_T = duals[base + 3]
        
        grad_x[i] += v_L - v_R
        grad_y[i] += v_B - v_T
        
    return np.column_stack((grad_x, grad_y))

def run_packing():
    np.random.seed(42) # For reproducibility
    n = 26
    
    # Initialization: Hexagonal packing
    # Estimate radius r ~ 0.1. 
    # Grid spacing: x-step 2r, y-step sqrt(3)r
    # But we don't know r yet. Let's use a normalized grid.
    
    centers = np.zeros((n, 2))
    
    # Simple grid initialization scaled to fit
    # 5 rows x 6 cols approx
    rows = 6
    cols = 5
    # Adjust to fit 26
    # Row 0: 5, Row 1: 5, ...
    # Hex pattern: shift odd rows
    
    idx = 0
    y_step = 1.0 / (rows + 1) # roughly
    x_step = 1.0 / (cols + 1)
    
    # Better hex init
    r_est = 0.1
    x_spacing = 2 * r_est * 1.1
    y_spacing = math.sqrt(3) * r_est * 1.1
    
    current_idx = 0
    y = r_est + 0.05 # Margin
    row = 0
    
    while current_idx < n:
        x = r_est + 0.05
        if row % 2 == 1:
            x += x_spacing / 2
        
        while x < 1.0 - r_est - 0.05 and current_idx < n:
            centers[current_idx, 0] = x
            centers[current_idx, 1] = y
            current_idx += 1
            x += x_spacing
        
        y += y_spacing
        row += 1
        
    # If not enough, fill randomly or extend
    while current_idx < n:
        centers[current_idx, 0] = np.random.uniform(0.2, 0.8)
        centers[current_idx, 1] = np.random.uniform(0.2, 0.8)
        current_idx += 1
        
    best_sum = 0.0
    best_centers = centers.copy()
    best_radii = np.zeros(n)
    
    # Optimization Loop
    # We try to improve the configuration
    for iteration in range(150):
        # Solve LP
        radii, sum_r, duals, success = solve_lp(centers, n)
        
        if not success:
            # If LP fails, maybe centers are bad. Perturb.
            centers += np.random.normal(0, 0.01, centers.shape)
            centers = np.clip(centers, 0.01, 0.99)
            continue
            
        if sum_r > best_sum + 1e-9:
            best_sum = sum_r
            best_centers = centers.copy()
            best_radii = radii.copy()
            
        # Compute Gradient
        grad = compute_gradient(centers, duals, n)
        
        # Step size decay
        alpha = 0.05 / (1 + iteration * 0.05)
        
        # Gradient Ascent
        centers = centers + alpha * grad
        
        # Project back to valid box [0, 1]
        # Actually centers must allow r >= 0. 
        # r <= x implies x >= 0. r <= 1-x implies x <= 1.
        # So [0, 1] is correct.
        centers = np.clip(centers, 0.0, 1.0)
        
        # Add small noise to escape local minima
        if iteration % 20 == 0 and iteration > 0:
            centers += np.random.normal(0, 0.005, centers.shape)
            centers = np.clip(centers, 0.0, 1.0)

    # Final polish: Solve LP one last time on best_centers
    radii, sum_r, _, _ = solve_lp(best_centers, n)
    
    # Validation check (internal)
    # Ensure valid
    # If best_sum is very low, fallback?
    # But with good init, it should be fine.
    
    # To ensure strict validity with epsilon tolerance
    # The LP ensures r_i + r_j <= dist.
    # But numerical errors might exist.
    # The validation function allows 1e-12.
    
    return best_centers, radii, sum_r

# Helper for the requested output format
def run_packing_wrapper():
    centers, radii, sum_r = run_packing()
    return centers, radii, float(sum_r)

# Ensure function signature matches
