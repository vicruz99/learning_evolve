from scipy.optimize import minimize

def _constraint_func(v, n):

    Returns an array of constraint values. All must be >= 0.
    v is a flattened array of [x1, y1, r1, x2, y2, r2, ...]
    """

        cons.append(xi - ri)
        cons.append(1.0 - xi - ri)
        cons.append(yi - ri)
        cons.append(1.0 - yi - ri)

            d2 = (xi - xj)**2 + (yi - yj)**2
            dr2 = (ri + rj)**2
            cons.append(d2 - dr2)
    return np.array(cons)

def _objective_func(v):

    return -np.sum(v[2::3])
