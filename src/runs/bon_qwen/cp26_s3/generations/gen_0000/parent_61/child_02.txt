import numpy as np
from scipy.optimize import minimize
import math


def objective(params, n):
    """Negative sum of radii (we minimize negative to maximize)."""
    radii = params[2*n:]
    return -np.sum(radii)


def constraint_func(params, n):
    """All constraints: must return values >= 0 for feasible points."""
    centers = params[:2*n].reshape(n, 2)
    radii = params[2*n:]
    
    constraints = []
    
    # Boundary constraints: r <= x, r <= 1-x, r <= y, r <= 1-y
    for i in range(n):
        constraints.append(centers[i, 0] - radii[i])
        constraints.append(1.0 - centers[i, 0] - radii[i])
        constraints.append(centers[i, 1] - radii[i])
        constraints.append(1.0 - centers[i, 1] - radii[i])
    
    # Non-overlap constraints: dist >= r_i + r_j
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = math.sqrt(dx * dx + dy * dy)
            constraints.append(dist - radii[i] - radii[j])
    
    return np.array(constraints)


def make_hex_initial(n, row_sizes, r):
    """Create hexagonal lattice initial placement."""
    centers = []
    y = r
    row_idx = 0
    
    for size in row_sizes:
        if size == 0:
            y += r * math.sqrt(3)
            row_idx += 1
            continue

            xs = [0.5]

            xs = np.linspace(r, 1.0 - r, size).tolist()

            xs = [max(r, min(1.0 - r, x)) for x in xs]
        
        for x in xs:
            centers.append([x, y])
        
        y += r * math.sqrt(3)
        row_idx += 1
    
    return np.array(centers), np.full(n, r)


def try_optimization(n, row_sizes, r_init):

    params = np.concatenate([centers.flatten(), radii])

    try:
        result = minimize(
            fun=objective,
            x0=params,
            args=(n,),
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'ineq', 'fun': constraint_func, 'args': (n,)},
            options={'maxiter': 5000, 'ftol': 1e-12}

        current_sum = np.sum(result.x[2*n:])
        return result, current_sum
    except Exception:
        return None, 0
