import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    """
    n_circles = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None

    # Define constraints
    def boundary_constraints(x_vars):
        # x_vars structure: [x1, y1, r1, x2, y2, r2, ...]
        # Shape: (3 * n_circles,)
        cons = []
        for i in range(n_circles):
            idx = i * 3
            xi, yi, ri = x_vars[idx], x_vars[idx+1], x_vars[idx+2]
            # r <= x, r <= 1-x, r <= y, r <= 1-y
            # Formatted as g(x) >= 0
            cons.append({'type': 'ineq', 'fun': lambda v, ind=i: v[ind*3+2] - v[ind*3]})
            cons.append({'type': 'ineq', 'fun': lambda v, ind=i: v[ind*3+2] - (1 - v[ind*3])})
            cons.append({'type': 'ineq', 'fun': lambda v, ind=i: v[ind*3+2] - v[ind*3+1]})
            cons.append({'type': 'ineq', 'fun': lambda v, ind=i: v[ind*3+2] - (1 - v[ind*3+1])})
        return cons

    def overlap_constraints(x_vars):
        cons = []
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                idx_i = i * 3
                idx_j = j * 3
                # (xi - xj)^2 + (yi - yj)^2 >= (ri + rj)^2
                # (xi - xj)^2 + (yi - yj)^2 - (ri + rj)^2 >= 0
                def cons_func(v, i=i, j=j):
                    xi, yi, ri = v[i*3], v[i*3+1], v[i*3+2]
                    xj, yj, rj = v[j*3], v[j*3+1], v[j*3+2]
                    dx = xi - xj
                    dy = yi - yj
                    dr = ri + rj
                    return (dx*dx + dy*dy) - (dr*dr)
                cons.append({'type': 'ineq', 'fun': cons_func})
        return cons

    constraints = boundary_constraints(None) + overlap_constraints(None)

    # Objective: Maximize sum of radii -> Minimize -sum(radii)
    def objective(x_vars):
        r_sum = 0
        for i in range(n_circles):
            r_sum += x_vars[i*3 + 2]
        return -r_sum

    # Bounds for variables
    # x, y in [0, 1], r in [0, 1]
    bounds = [(0, 1) for _ in range(n_circles)] + \
             [(0, 1) for _ in range(n_circles)] + \
             [(0, 1) for _ in range(n_circles)]

    def get_initial_guess(method="hexagonal"):
        centers = np.zeros((n_circles, 2))
        radii = np.zeros(n_circles)
        
        if method == "hexagonal":
            # Generate hexagonal grid points
            # Estimate spacing to fit ~26 circles
            # Area ~ 1, density ~ 0.9. 26 * pi * r^2 ~ 0.9 => r ~ 0.105. s ~ 0.21
            # But boundary constraints shrink effective area.
            # Try s = 0.22
            s = 0.22 
            points = []
            # Iterate rows and cols
            for row in range(10):
                for col in range(10):
                    x = col * s + (row % 2) * (s / 2) + 0.1
                    y = row * (s * np.sqrt(3)/2) + 0.1
                    if 0 <= x <= 1 and 0 <= y <= 1:
                        points.append([x, y])
            
            # Select top 26 points closest to center to reduce boundary issues initially?
            # Or just take first 26
            if len(points) >= n_circles:
                # Sort by distance from center (0.5, 0.5) to pick central ones first?
                # Actually, packing usually fills from center or grid.
                # Let's just take the first n_circles generated
                selected = points[:n_circles]
            else:
                # Fallback random if grid is too sparse
                selected = np.random.rand(n_circles, 2)
                
            centers = np.array(selected)
            # Initial radius small to ensure feasibility
            radii = np.full(n_circles, 0.05)
            
        elif method == "random":
            centers = np.random.rand(n_circles, 2)
            radii = np.full(n_circles, 0.05)
            
        x0 = np.zeros(3 * n_circles)
        for i in range(n_circles):
            x0[i*3] = centers[i, 0]
            x0[i*3+1] = centers[i, 1]
            x0[i*3+2] = radii[i]
        return x0

    # Run optimization from multiple starts
    strategies = ["hexagonal", "random", "random"]
    
    for strategy in strategies:
        x0 = get_initial_guess(strategy)
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints, 
                           options={'maxiter': 1000, 'ftol': 1e-9})
            if res.success or res.fun < -best_sum: # We are minimizing negative sum
                current_sum = -res.fun
                if current_sum > best_sum:
                    best_sum = current_sum
                    best_centers = np.zeros((n_circles, 2))

                        best_centers[i, 0] = res.x[i*3]

                        best_radii[i] = res.x[i*3+2]
        except Exception as e:
            continue

    # Sometimes optimizer returns slightly invalid due to numerical error

    if best_radii is not None:

        final_sum = np.sum(best_radii)
        # If validation fails, we might need to shrink radii slightly
        # But the validator allows 1e-12 tolerance.

    if best_centers is None:

        best_centers = best_centers[:26]
        best_radii = np.full(26, 0.05)
        final_sum = np.sum(best_radii)
    else:
        final_sum = best_sum

    return best_centers, best_radii, final_sum
