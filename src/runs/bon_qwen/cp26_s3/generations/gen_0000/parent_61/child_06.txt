import numpy as np
import cvxpy as cp
import scipy.optimize as opt

def get_boundary_distances(centers):
    """Calculate the minimum distance of each center to the square boundaries."""
    x, y = centers[:, 0], centers[:, 1]
    r_max = np.minimum(np.minimum(x, 1 - x), np.minimum(y, 1 - y))
    return np.maximum(r_max, 0)

def solve_optimal_radii(centers, n_circles):
    """
    Solves the LP: max sum(r) s.t. r_i + r_j <= dist(i,j), r_i <= boundary_dist(i)
    """
    r = cp.Variable(n_circles, nonneg=True)
    constraints = []
    
    # Pairwise distance constraints
    for i in range(n_circles):
        for j in range(i + 1, n_circles):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            constraints.append(r[i] + r[j] <= dist)
            
    # Boundary constraints
    boundaries = get_boundary_distances(centers)
    constraints.append(r <= boundaries)
    
    objective = cp.Maximize(cp.sum(r))
    problem = cp.Problem(objective, constraints)
    
    try:
        status = problem.solve()
        if status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return r.value, problem.value
        else:
            return None, 0
    except Exception:
        return None, 0
