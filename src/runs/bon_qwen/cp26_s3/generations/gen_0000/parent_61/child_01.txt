import numpy as np
import scipy.optimize

def run_packing():

    Optimizes the packing of 26 circles in a unit square to maximize sum of radii.
    
    Returns:
        centers: np.array of shape (26, 2) with (x, y) coordinates
        radii: np.array of shape (26) with radius of each circle
        sum_radii: float sum of all radii
    """
    np.random.seed(42)
    N = 26
    
    # 1. Initialization: Hexagonal Grid Pattern
    # Approximate radius for 26 circles is ~0.1. 
    # We start with a slightly smaller radius to ensure a valid starting configuration.
    init_r = 0.06
    
    centers_init = []
    r_init = [init_r] * N
    
    # Hexagonal packing logic
    # Row height spacing
    dy = init_r * np.sqrt(3)
    # Col width spacing
    dx = 2 * init_r
    
    current_idx = 0
    row = 0
    
    # Generate positions until we have N circles
    while current_idx < N:
        y = init_r + row * dy
        
        # Check if row fits vertically
        if y + init_r > 1.0:
            # If we can't fit a row, try to squeeze by reducing row height slightly or stopping
            # For initialization, if we run out of vertical space, we stop adding rows
            # and fill remaining circles in last row or break.
            # However, for 26 circles, 5-6 rows should fit.
            break
            
        # Determine number of circles in this row to fit horizontally
        # Alternating shifts for hexagonal packing
        shift = (init_r if row % 2 == 1 else 0)
        x = shift + init_r
        
        while x + init_r <= 1.0 and current_idx < N:
            centers_init.append([x, y])
            current_idx += 1
            x += dx
            
        row += 1
        
    # Fill any remaining spots if loop terminated early (unlikely with this logic)
    while current_idx < N:
        # Fallback to random placement in safe zone if grid fails
        cx = np.random.uniform(init_r, 1 - init_r)
        cy = np.random.uniform(init_r, 1 - init_r)
        centers_init.append([cx, cy])
        current_idx += 1

    centers_arr = np.array(centers_init[:N])
    
    # 2. Optimization Setup
    # Variables: [x1, y1, r1, x2, y2, r2, ...]
    # Total variables = N * 3
    init_vars = []
    for i in range(N):
        init_vars.extend([centers_arr[i, 0], centers_arr[i, 1], r_init[i]])
    
    init_vars = np.array(init_vars)
    
    # Bounds for variables
    # x, y in [0, 1], r in [0, 0.5]
    bounds = []
    for i in range(N):
        bounds.extend([(0, 1), (0, 1), (0, 0.5)])

    # 3. Objective Function with Penalties
    def objective(vars):
        # Reshape vars back to centers and radii
        cs = vars.reshape(N, 3)
        cx = cs[:, 0]
        cy = cs[:, 1]
        r = cs[:, 2]
        
        # Objective: Maximize sum of radii -> Minimize negative sum
        obj = -np.sum(r)
        
        # Penalty for overlaps
        overlap_penalty = 0.0
        # Vectorized distance calculation
        # cx_diff shape (N, N), same for cy
        cx_diff = cx[:, np.newaxis] - cx[np.newaxis, :]
        cy_diff = cy[:, np.newaxis] - cy[np.newaxis, :]
        dist_sq = cx_diff**2 + cy_diff**2
        dist = np.sqrt(dist_sq)
        
        # Sum of radii matrix
        r_sum = r[:, np.newaxis] + r[np.newaxis, :]
        
        # Overlap condition: dist < r_sum
        gap = r_sum - dist
        # Only consider positive gaps (overlaps) and upper triangle (i < j) to avoid double counting
        # Mask for upper triangle
        mask = np.triu(np.ones((N, N), dtype=bool), k=1)
        overlaps = np.maximum(0, gap) * mask
        
        # Quadratic penalty
        overlap_penalty = np.sum(overlaps**2)
        
        # Penalty for boundaries
        boundary_penalty = 0.0
        # Constraints: r <= x <= 1-r  => x-r >= 0, 1-x-r >= 0
        # Same for y
        # gap_x1 = r - x, gap_x2 = x + r - 1
        # We penalize if gap > 0
        
        # x boundaries
        gap_x_left = r - cx
        gap_x_right = cx + r - 1.0
        # y boundaries
        gap_y_bottom = r - cy
        gap_y_top = cy + r - 1.0
        
        boundary_penalty += np.sum(np.maximum(0, gap_x_left)**2)
        boundary_penalty += np.sum(np.maximum(0, gap_x_right)**2)
        boundary_penalty += np.sum(np.maximum(0, gap_y_bottom)**2)
        boundary_penalty += np.sum(np.maximum(0, gap_y_top)**2)
        
        # Total cost
        # High weights to enforce constraints strictly
        penalty_weight = 5000.0
        
        return obj + penalty_weight * (overlap_penalty + boundary_penalty)

    # 4. Run Optimization
    # Using L-BFGS-B which handles bounds well
    # We might run multiple restarts or just one strong optimization
    result = scipy.optimize.minimize(
        objective, 
        init_vars, 
        method='L-BFGS-B', 
        bounds=bounds,
        options={'maxiter': 1000, 'ftol': 1e-9, 'gtol': 1e-6}
    )
    
    # 5. Post-processing and Validation
    best_vars = result.x
    best_cs = best_vars.reshape(N, 3)
    centers_final = best_cs[:, :2]
    radii_final = best_cs[:, 2]

            # Neighbor check

                if i == j: continue
                xj, yj = centers_final[j]
                rj = radii_final[j]
                d = np.sqrt((x - xj)**2 + (y - yj)**2)
                # r + rj <= d  => r <= d - rj
                limit = d - rj

                    max_r_neighbor = limit
            
            allowed_r = min(max_r_boundary, max_r_neighbor)
            
            if radii_final[i] > allowed_r + 1e-9:
                radii_final[i] = max(0, allowed_r)
                changed = True
        
        if not changed:
            break

    # Final validation check (internal)

    # Check boundaries

        r = radii_final[i]

            valid = False
            break
    # Check overlaps

                    break
            if not valid: break

    return centers_final, radii_final, sum_radii
