import numpy as np
from scipy.optimize import minimize

def smooth_relu(x, eps=1e-4):
    """Smooth approximation of ReLU(x) = max(0, x)"""
    return 0.5 * (x + np.sqrt(x**2 + eps**2))

def get_cost_grad(params, n, mu, eps=1e-4):
    """
    Computes the cost function and its gradient for the circle packing problem.
    Cost = -sum(radii) + mu * penalty
    """
    centers = params[:2*n].reshape(n, 2)
    radii = params[2*n:]

    cost = -np.sum(radii)
    grad = np.zeros_like(params)
    grad[2*n:] = -1.0  # Gradient of -sum(r) w.r.t r is -1

    # Overlap constraints: dist(i,j) >= r_i + r_j  =>  r_i + r_j - dist <= 0
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx**2 + dy**2)
            sum_r = radii[i] + radii[j]
            overlap = sum_r - dist

            if overlap > -0.01:  # Only process if close to violation
                s_overlap = smooth_relu(overlap, eps)
                cost += mu * s_overlap**2

                # Derivative of smooth_relu
                d_s = 0.5 * (1.0 + overlap / np.sqrt(overlap**2 + eps**2))
                # Chain rule: d(cost)/d(overlap) = 2 * mu * s_overlap * d_s
                dP_d_overlap = 2.0 * mu * s_overlap * d_s

                if dist > 1e-8:
                    inv_dist = 1.0 / dist
                    # d(overlap)/d(x_i) = -dx/dist, d(overlap)/d(y_i) = -dy/dist
                    # d(overlap)/d(r_i) = 1
                    grad[3*i]     += dP_d_overlap * (-dx * inv_dist)
                    grad[3*i + 1] += dP_d_overlap * (-dy * inv_dist)
                    grad[3*i + 2] += dP_d_overlap

                    grad[3*j]     += dP_d_overlap * (dx * inv_dist)
                    grad[3*j + 1] += dP_d_overlap * (dy * inv_dist)
                    grad[3*j + 2] += dP_d_overlap

    # Boundary constraints: 0 <= x,y <= 1 and r <= coord <= 1-r
    # Left/Bottom: r - coord <= 0
    # Right/Top:   r + coord - 1 <= 0
    for i in range(n):
        for dim in range(2):
            c = centers[i, dim]
            r = radii[i]

            # Left/Bottom violation
            val_lb = r - c
            if val_lb > -0.01:
                s_val = smooth_relu(val_lb, eps)
                cost += mu * s_val**2
                d_s = 0.5 * (1.0 + val_lb / np.sqrt(val_lb**2 + eps**2))
                dP = 2.0 * mu * s_val * d_s
                
                # d(val)/d(c) = -1, d(val)/d(r) = 1
                grad[2*i + dim] -= dP
                grad[2*n + i]   += dP

            # Right/Top violation
            val_rt = r + c - 1.0
            if val_rt > -0.01:
                s_val = smooth_relu(val_rt, eps)
                cost += mu * s_val**2
                d_s = 0.5 * (1.0 + val_rt / np.sqrt(val_rt**2 + eps**2))
                dP = 2.0 * mu * s_val * d_s
                
                # d(val)/d(c) = 1, d(val)/d(r) = 1
                grad[2*i + dim] += dP
                grad[2*n + i]   += dP

    return cost, grad

def fun(p, n, mu):
    return get_cost_grad(p, n, mu)[0]

def jac(p, n, mu):
    return get_cost_grad(p, n, mu)[1]

def generate_initial_config(n):
    """Generates a grid-based initial configuration with small radii."""
    cols = int(np.ceil(np.sqrt(n)))
    rows = int(np.ceil(n / cols))
    pts = []
    for r in range(rows):
        for c in range(cols):
            if len(pts) >= n:
                break
            x = (c + 0.5) / cols
            y = (r + 0.5) / rows
            pts.append([x, y])
    
    centers = np.array(pts[:n])
    radii = np.full(n, 0.05)
    return centers, radii

def run_packing():
    n = 26
    best_sum_r = -np.inf
    best_centers = None
    best_radii = None

    # Bounds for L-BFGS-B
    bounds = [(0.0, 1.0) for _ in range(2 * n)] + [(0.0, 0.5) for _ in range(n)]

    np.random.seed(42)  # For reproducibility

    for restart in range(15):
        centers, radii = generate_initial_config(n)
        # Add random perturbation
        centers += np.random.randn(*centers.shape) * 0.03
        centers = np.clip(centers, 0.05, 0.95)

        x0 = np.concatenate([centers.flatten(), radii])

        # Annealing schedule for penalty weight
        mus = [10, 100, 1000, 5000, 10000]
        current_x = x0.copy()

        for mu in mus:
            res = minimize(
                fun, current_x, args=(n, mu),
                jac=jac,
                bounds=bounds,
                method='L-BFGS-B',
                options={'ftol': 1e-12, 'gtol': 1e-8, 'maxiter': 5000, 'disp': False}
            )
            current_x = res.x

        sum_r = np.sum(current_x[2*n:])
        if sum_r > best_sum_r:
            best_sum_r = sum_r
            best_centers = current_x[:2*n].reshape(n, 2)
            best_radii = current_x[2*n:]

    return best_centers, best_radii, best_sum_r
