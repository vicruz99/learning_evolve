import numpy as np
from scipy.optimize import minimize

def run_packing():
    n = 26
    # 1. Initialize centers on a dense grid
    # We use a 6x5 grid (30 points) and select the first 26
    x = np.linspace(0.1, 0.9, 6)
    y = np.linspace(0.1, 0.9, 5)
    grid_x, grid_y = np.meshgrid(x, y)
    centers = np.column_stack([grid_x.ravel(), grid_y.ravel()])[:n]
    
    # 2. Force-directed relaxation to spread points
    # Repulsion forces to maximize minimum distance between points
    for _ in range(500):
        # Initialize forces
        forces = np.zeros_like(centers)
        
        for i in range(n):
            # Boundary repulsion (soft)
            for axis in range(2):
                if centers[i, axis] < 0.05:
                    forces[i, axis] += 0.001
                elif centers[i, axis] > 0.95:
                    forces[i, axis] -= 0.001
            
            for j in range(i + 1, n):
                diff = centers[i] - centers[j]
                dist = np.linalg.norm(diff)
                if dist < 0.15: # Repel if too close
                    force_mag = 0.01 / (dist + 1e-6)
                    forces[i] += diff * force_mag
                    forces[j] -= diff * force_mag
        
        centers += forces
        centers = np.clip(centers, 0.0, 1.0)

    # Initial radii (small but non-zero)
    radii = np.full(n, 0.01)

    # 3. Objective function: maximize sum of radii (minimize negative sum)
    def objective(params):
        c = params[:2 * n].reshape((n, 2))
        r = params[2 * n:]
        return -np.sum(r)

    # 4. Constraints
    constraints = []

    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    # Reformulated as: x - r >= 0, 1 - x - r >= 0, etc.
    for i in range(n):
        constraints.append({'type': 'ineq', 'fun': lambda p, idx=i: p[2*idx] - p[2*n + idx]}) # x - r >= 0
        constraints.append({'type': 'ineq', 'fun': lambda p, idx=i: 1.0 - p[2*idx] - p[2*n + idx]}) # 1-x-r >= 0
        constraints.append({'type': 'ineq', 'fun': lambda p, idx=i: p[2*idx+1] - p[2*n + idx]}) # y - r >= 0
        constraints.append({'type': 'ineq', 'fun': lambda p, idx=i: 1.0 - p[2*idx+1] - p[2*n + idx]}) # 1-y-r >= 0

    # Non-overlap constraints: dist(c_i, c_j) >= r_i + r_j
    # Squared form: ||c_i - c_j||^2 - (r_i + r_j)^2 >= 0
    for i in range(n):
        for j in range(i + 1, n):
            def make_constraint(i, j):
                def constraint(p):
                    c_i = p[2*i : 2*i+2]
                    c_j = p[2*j : 2*j+2]
                    r_i = p[2*n + i]
                    r_j = p[2*n + j]
                    dist_sq = np.sum((c_i - c_j)**2)
                    return dist_sq - (r_i + r_j)**2
                return constraint
            
            constraints.append({'type': 'ineq', 'fun': make_constraint(i, j)})

    # 5. Setup initial parameters
    x0 = np.concatenate([centers.ravel(), radii])

    # 6. Optimization
    bounds = []
    # Bounds for centers (0 to 1)
    for _ in range(2 * n):
        bounds.append((0.0, 1.0))
    # Bounds for radii (0 to 0.5)
    for _ in range(n):
        bounds.append((0.0, 0.5))

    result = minimize(
        objective, 
        x0, 
        method='SLSQP', 
        bounds=bounds, 
        constraints=constraints,
        options={'maxiter': 1000, 'ftol': 1e-9}
    )

    # 7. Extract and return results
    optimal_centers = result.x[:2 * n].reshape((n, 2))
    optimal_radii = result.x[2 * n:]
    sum_radii = np.sum(optimal_radii)

    return optimal_centers, optimal_radii, sum_radii
