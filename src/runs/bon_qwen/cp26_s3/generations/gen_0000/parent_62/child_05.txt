        # Right
        if x + r > 1: penalty += (x + r - 1)**2
        # Bottom
        if y - r < 0: penalty += (y - r)**2
        # Top
        if y + r > 1: penalty += (y + r - 1)**2

            req_dist = radii[i] + radii[j]
            if d < req_dist:
                penalty += (req_dist - d)**2
                
    valid = penalty < 1e-8
    return sum_r, valid, penalty
