```python
import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def objective(vars):
    # Maximize sum of radii -> minimize negative sum
    # vars layout: [x0, y0, r0, x1, y1, r1, ...]
    return -np.sum(vars[2*N_CIRCLES:])

def con_boundary(vars):
    cx = vars[0::3]
    cy = vars[1::3]
    cr = vars[2::3]
    # Constraints: cx >= cr, cx + cr <= 1, cy >= cr, cy + cr <= 1
    # Formulated as >= 0 for SLSQP 'ineq' type
    return np.concatenate([
        cx - cr,
        1 - cx - cr,
        cy - cr,
        1 - cy - cr
    ])

def con_overlap(vars):
    cx = vars[0::3]
    cy = vars[1::3]
    cr = vars[2::3]
    cons = []
    for i in range(N_CIRCLES):
        for j in range(i + 1, N_CIRCLES):
            dx = cx[i] - cx[j]
            dy = cy[i] - cy[j]
            dist_sq = dx**2 + dy**2
            r_sum = cr[i] + cr[j]
            # dist >= r_sum  =>  dist^2 - r_sum^2 >= 0
            cons.append(dist_sq - r_sum**2)
    return np.array(cons)

def get_initial_guess():
    r_init = 0.08
    sqrt3 = np.sqrt(3)
    centers = []
    row_idx = 0
    count = 0
    while count < N_CIRCLES:
        y = r_init + row_idx * r_init * sqrt3
        if y + r_init > 1.0:
            break
        is_odd = (row_idx % 2 == 1)
        x_start = r_init * (2 if is_odd else 1)
        x = x_start
        while x + r_init <= 1.0 and count < N_CIRCLES:
            centers.append([x, y])
            x += 2 * r_init
            count += 1
        row_idx += 1
        
    centers = np.array(centers[:N_CIRCLES])
    radii = np.full(N_CIRCLES, r_init)
    
    init_vars = np.zeros(3 * N_CIRCLES)
    init_vars[0::3] = centers[:, 0]
    init_vars[1::3] = centers[:, 1]
    init_vars[2::3] = radii
    return init_vars

def run_packing():
    x0 = get_initial_guess()
    bounds = [(0.0, 1.0)] * (2 * N_CIRCLES) + [(0.0, 0.5)] * N_CIRCLES
    
    constraints = [
        {'type': 'ineq', 'fun': con_boundary},
        {'type': 'ineq', 'fun': con_overlap}
    ]
    
    res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints,
                   options={'maxiter': 3000, 'ftol': 1e-12, 'disp': False})
    
    cx_opt = res.x[0::3]
    cy_opt = res.x[1::3]
    cr_opt = res.x[2::3]
    
    centers_opt = np.column_stack([cx_opt, cy_opt])
    sum_radii = float(np.sum(cr_opt))
    
    return centers_opt, cr_opt, sum_radii
```
