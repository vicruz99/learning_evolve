import numpy as np

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes the packing of 26 circles in a unit square to maximize sum of radii.
    Uses a force-directed layout with radius expansion.
    """
    n = 26
    
    # 1. Initialization: Hexagonal Lattice
    # Arrange in rows to fill the square efficiently
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.05  # Start with small radius
    
    row, col = 0, 0
    for i in range(n):
        # Hexagonal spacing
        x = col * 1.0 + (row % 2) * 0.5
        y = row * 0.866  # sin(60) approx 0.866
        
        # Normalize to fit roughly in 1x1 initially
        centers[i] = [x, y]
        
        # Advance grid
        col += 1
        if col > 5: # Approximate 5-6 circles per row
            col = 0
            row += 1
    
    # Normalize positions to be inside [0,1]x[0,1] initially
    min_x, min_y = centers.min(axis=0)
    max_x, max_y = centers.max(axis=0)
    width = max_x - min_x
    height = max_y - min_y
    
    if width > 0:
        centers[:, 0] = (centers[:, 0] - min_x) / width
    if height > 0:
        centers[:, 1] = (centers[:, 1] - min_y) / height
        
    # Add small padding to ensure strict interior start
    centers *= 0.95
    centers += 0.025

    # 2. Optimization Parameters
    iterations = 5000
    expansion_rate = 1.0003
    damping = 0.95
    repulsion_strength = 10.0
    wall_strength = 5.0

    for _ in range(iterations):
        # Expand radii
        radii *= expansion_rate

        velocities = np.zeros((n, 2))

        # Vectorized computation for efficiency
        # diff[i, j] = centers[i] - centers[j]
        diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
        dist_sq = np.sum(diff**2, axis=2)
        
        # Avoid self-interaction and division by zero
        dist_sq = np.maximum(dist_sq, 1e-12)
        dist = np.sqrt(dist_sq)
        min_dist = radii[:, np.newaxis] + radii[np.newaxis, :]

        overlap = min_dist - dist

        inv_dist = 1.0 / dist
        inv_dist[dist < 1e-6] = 0
        
        force_vec = diff * inv_dist[:, :, np.newaxis] * force_mag[:, :, np.newaxis]
        
        # Sum forces on each circle
        forces += np.sum(force_vec, axis=1)
        
        # Wall forces
        # Push back if touching walls
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            
            # Left wall
            if x - r < 0:
                forces[i, 0] += wall_strength * (r - x)
            # Right wall
            if x + r > 1:
                forces[i, 0] -= wall_strength * (x + r - 1)
            # Bottom wall
            if y - r < 0:
                forces[i, 1] += wall_strength * (r - y)
            # Top wall
            if y + r > 1:
                forces[i, 1] -= wall_strength * (y + r - 1)
                
        # Update velocities and positions
        velocities = velocities * damping + forces * 0.01
        centers += velocities

        centers[:, 1] = np.clip(centers[:, 1], radii, 1 - radii)
