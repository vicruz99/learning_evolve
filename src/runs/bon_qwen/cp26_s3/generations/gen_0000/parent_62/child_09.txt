import numpy as np
from scipy.optimize import minimize, differential_evolution, basinhopping


def compute_max_radii(centers, n, max_iter=2000):
    """Compute maximum feasible radii for given centers."""
    radii = np.ones(n) * 0.5
    
    for _ in range(max_iter):
        new_radii = np.ones(n) * 0.5
        
        for i in range(n):
            r = min(centers[i, 0], 1.0 - centers[i, 0],
                    centers[i, 1], 1.0 - centers[i, 1])
            
            for j in range(n):
                if i != j:
                    dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                    r = min(r, dist - radii[j])
            
            new_radii[i] = max(0.0, r)
        
        if np.max(np.abs(new_radii - radii)) < 1e-13:
            break
        radii = new_radii
    
    return radii


def hexagonal_init(n, seed=0):
    """Initialize centers in a hexagonal lattice pattern."""
    rng = np.random.RandomState(seed)
    centers = np.zeros((n, 2))
    
    idx = 0
    row = 0
    spacing = 0.19
    
    while idx < n:
        if row % 2 == 0:
            cols = 5
        else:
            cols = 4
        
        for col in range(cols):
            if idx < n:
                x = 0.08 + col * spacing + (spacing / 2.0 if row % 2 == 1 else 0.0)
                y = 0.08 + row * spacing * np.sqrt(3.0) / 2.0
                centers[idx] = [x, y]
                idx += 1
        row += 1
    
    # Add small random perturbation
    centers += rng.uniform(-0.005, 0.005, (n, 2))
    centers[:, 0] = np.clip(centers[:, 0], 0.03, 0.97)
    centers[:, 1] = np.clip(centers[:, 1], 0.03, 0.97)
    
    return centers


def grid_init(n, seed=0):
    """Initialize centers in a grid pattern."""
    rng = np.random.RandomState(seed)
    centers = np.zeros((n, 2))
    
    # 5x5 grid plus 1
    rows = 5
    cols = 5
    idx = 0
    
    for i in range(rows):
        for j in range(cols):
            if idx < n:
                centers[idx, 0] = 0.1 + j * 0.2

                idx += 1

    centers[:, 0] = np.clip(centers[:, 0], 0.03, 0.97)
    centers[:, 1] = np.clip(centers[:, 1], 0.03, 0.97)
    
    return centers


def random_init(n, seed=0):
    """Initialize centers randomly."""
    rng = np.random.RandomState(seed)
    centers = rng.uniform(0.05, 0.95, (n, 2))
    return centers


def force_optimize(centers, n, iterations=40000, lr_init=0.008):
    """Force-based optimization with expanding circles."""
    radii = np.ones(n) * 0.015
    lr = lr_init
    centers = centers.copy()
    
    for it in range(iterations):
        # Gradually expand target radii
        expansion = 1.0 + 3e-5 * max(0, 1.0 - it / 30000.0)
        radii *= expansion
        
        # Compute forces
        forces = np.zeros_like(centers)
        
        for i in range(n):
            # Boundary repulsion forces
            for d in range(2):
                # Left/bottom boundary
                gap = centers[i, d] - radii[i]
                if gap < 0:
                    forces[i, d] += gap * 80.0
                elif gap < 0.02:
                    forces[i, d] += gap * 5.0
                
                # Right/top boundary
                gap = (1.0 - centers[i, d]) - radii[i]
                if gap < 0:
                    forces[i, d] -= gap * 80.0
                elif gap < 0.02:
                    forces[i, d] -= gap * 5.0
            
            # Neighbor repulsion
            for j in range(i + 1, n):
                diff = centers[i] - centers[j]
                dist_sq = np.sum(diff ** 2)
                dist = np.sqrt(dist_sq) if dist_sq > 1e-20 else 1e-10
                
                overlap = radii[i] + radii[j] - dist
                if overlap > 0:
                    # Strong repulsion when overlapping
                    force_magnitude = overlap * 150.0
                    force = diff / dist * force_magnitude
                    forces[i] += force
                    forces[j] -= force
        
        # Update positions
        centers = centers + lr * forces
        centers[:, 0] = np.clip(centers[:, 0], 0.002, 0.998)
        centers[:, 1] = np.clip(centers[:, 1], 0.002, 0.998)
        
        # Decay learning rate
        if it % 4000 == 0 and lr > 1e-9:
            lr *= 0.85
        
        # Slow down expansion near end
        if it > 25000:
            pass  # Let positions settle
    
    # Compute final exact radii
    radii = compute_max_radii(centers, n)
    return centers, radii


def optimize_with_scipy(centers_init, n, maxiter=100):
    """Use scipy optimizer to refine positions."""
    radii = compute_max_radii(centers_init, n)
    
    def objective(x):
        c = x.reshape((n, 2))
        r = compute_max_radii(c, n, max_iter=500)
        return -np.sum(r)
    
    result = minimize(objective, centers_init.flatten(), 
                     method='Nelder-Mead',
                     options={'maxiter': maxiter, 'xatol': 1e-8, 'fatol': 1e-10})
    
    centers_opt = result.x.reshape((n, 2))
    radii_opt = compute_max_radii(centers_opt, n)
    return centers_opt, radii_opt


def try_configuration(init_func, seed, n, use_scipy=False):
    """Try a single configuration."""
    centers = init_func(n, seed)
    centers, radii = force_optimize(centers, n)
    
    if use_scipy:
        centers, radii = optimize_with_scipy(centers, n, maxiter=50)
    
    return centers, radii
