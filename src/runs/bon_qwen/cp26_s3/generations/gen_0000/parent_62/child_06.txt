The objective function will be the negative sum of radii plus a large quadratic penalty for boundary violations and circle overlaps. This transforms the constrained problem into an unconstrained (bounded) one that gradient-based methods can handle efficiently.

To ensure robustness and escape local minima:
1. We run multiple optimization trials initialized from structured hexagonal/grid patterns with deterministic jitter.
2. We use a high penalty weight to strongly enforce non-overlap and boundary constraints.
3. After optimization, we apply a strict post-processing step that computes the maximum valid radius for each circle given its current position and shrinks it slightly to guarantee validation passes within numerical tolerances.
4. A fallback simple grid packing is included to guarantee a valid solution is always returned, though the optimizer will typically find a superior configuration.

This approach balances computational efficiency with geometric optimality, effectively pushing circles against each other and the square boundaries to maximize the total radius sum.

import numpy as np
from scipy.optimize import minimize

def compute_loss(params, n):
    """
    Objective function to minimize: negative sum of radii + penalty for constraint violations.
    """
    centers = params[:2*n].reshape(n, 2)
    radii = params[2*n:]
    
    loss = -np.sum(radii)
    penalty_weight = 1.0e6
    
    # Boundary penalties: ensure r <= x <= 1-r and r <= y <= 1-r
    loss += penalty_weight * np.sum(np.maximum(0, radii - centers[:,0])**2)
    loss += penalty_weight * np.sum(np.maximum(0, radii - (1.0 - centers[:,0]))**2)
    loss += penalty_weight * np.sum(np.maximum(0, radii - centers[:,1])**2)
    loss += penalty_weight * np.sum(np.maximum(0, radii - (1.0 - centers[:,1]))**2)
    
    # Overlap penalties: penalize when dist < r_i + r_j
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx*dx + dy*dy)
            overlap = radii[i] + radii[j] - dist
            if overlap > 0.0:
                loss += penalty_weight * overlap**2
                
    return loss

def run_packing():
    np.random.seed(42)
    n = 26
    
    best_centers = None
    best_radii = None
    best_sum = 0.0
    
    # Run multiple trials with different structured initializations
    for trial in range(8):
        coords = []
        s = 0.18
        y = 0.1 + trial * 0.005
        while len(coords) < n:
            x = 0.1 + (trial % 3) * 0.03
            row_shift = (len(coords) // 5) % 2
            if row_shift: 
                x += s/2
            while x < 0.9 and len(coords) < n:
                # Add small deterministic jitter to break symmetry
                jitter_x = np.sin(trial * 10.0 + len(coords)) * 0.02
                jitter_y = np.cos(trial * 10.0 + len(coords)) * 0.02
                coords.append([x + jitter_x, y + jitter_y])
                x += s
            y += s * np.sqrt(3) / 2
            
        init_centers = np.array(coords[:n])
        init_radii = np.full(n, 0.07)
        
        params = np.concatenate([init_centers.flatten(), init_radii])
        bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n
        
        res = minimize(compute_loss, params, args=(n,), method='L-BFGS-B', 
                       bounds=bounds, options={'maxiter': 4000, 'ftol': 1e-15})
        
        centers = res.x[:2*n].reshape(n, 2)
        radii = res.x[2*n:]
        
        # Validate and track best
        valid = True
        for i in range(n):
            if radii[i] < 0 or centers[i,0] < radii[i] - 1e-7 or centers[i,0] > 1 - radii[i] + 1e-7 or \
               centers[i,1] < radii[i] - 1e-7 or centers[i,1] > 1 - radii[i] + 1e-7:
                valid = False
                break
        if valid:
            for i in range(n):
                for j in range(i+1, n):
                    d = np.linalg.norm(centers[i] - centers[j])
                    if d < radii[i] + radii[j] - 1e-7:
                        valid = False
                        break
                if not valid: break
                
        if valid:
            current_sum = np.sum(radii)
            if current_sum > best_sum:
                best_sum = current_sum
                best_centers = centers.copy()
                best_radii = radii.copy()
                
    # Fallback to a valid simple packing if optimization failed to find a valid config
    if best_centers is None:
        best_centers = np.zeros((n, 2))
        best_radii = np.full(n, 0.12)
        idx = 0
        for r in range(5):
            for c in range(5):
                if idx < n:
                    best_centers[idx] = [0.12 + c*0.2, 0.12 + r*0.2]
                    idx += 1
        if idx < n:
            best_centers[idx] = [0.5, 0.5]
            best_radii[idx] = 0.01
            
    # Final strict adjustment to guarantee validation passes within tolerances
    for i in range(n):
        max_r = min(best_centers[i,0], 1-best_centers[i,0], best_centers[i,1], 1-best_centers[i,1])
        for j in range(n):
            if i != j:
                d = np.linalg.norm(best_centers[i] - best_centers[j])
                max_r = min(max_r, d - best_radii[j])
        best_radii[i] = min(best_radii[i], max(0.0, max_r - 1e-9))
        
    return best_centers, best_radii, np.sum(best_radii)
