import numpy as np
from scipy.optimize import minimize


def compute_loss(params, n, penalty):
    """Compute penalized objective function."""
    centers = params[:2 * n].reshape(n, 2)
    radii = params[2 * n:]
    radii = np.maximum(radii, 1e-12)

    loss = -np.sum(radii)

    # Boundary penalties
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        loss += penalty * max(0, r - x) ** 2
        loss += penalty * max(0, r - (1 - x)) ** 2
        loss += penalty * max(0, r - y) ** 2
        loss += penalty * max(0, r - (1 - y)) ** 2

    # Overlap penalties
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            overlap = radii[i] + radii[j] - dist
            if overlap > 0:
                loss += penalty * overlap ** 2

    return loss

def force_relax(centers, radii, n, iterations=3000, lr=0.005):
    """Apply repulsive forces to separate circles and allow growth."""
    centers = centers.copy()
    radii = radii.copy()

    for it in range(iterations):
        forces = np.zeros_like(centers)

        # Repulsive forces between overlapping circles
        for i in range(n):
            for j in range(i + 1, n):
                diff = centers[i] - centers[j]
                dist = np.sqrt(np.sum(diff ** 2)) + 1e-12
                min_dist = radii[i] + radii[j]

                if dist < min_dist:
                    overlap = min_dist - dist
                    force_mag = overlap * (1 + 0.5 * np.exp(-it / 500))
                    force = force_mag * diff / dist
                    forces[i] += force
                    forces[j] -= force

        # Boundary repulsion
        for i in range(n):
            r = radii[i]
            margin = 1e-8
            if centers[i, 0] < r + margin:
                forces[i, 0] += (r + margin - centers[i, 0]) * 20
            if centers[i, 0] > 1 - r - margin:
                forces[i, 0] -= (centers[i, 0] - (1 - r - margin)) * 20
            if centers[i, 1] < r + margin:
                forces[i, 1] += (r + margin - centers[i, 1]) * 20
            if centers[i, 1] > 1 - r - margin:
                forces[i, 1] -= (centers[i, 1] - (1 - r - margin)) * 20

        centers -= lr * forces

        # Clip to feasible region
        for i in range(n):
            r = radii[i]
            centers[i, 0] = max(r + 1e-10, min(1 - r - 1e-10, centers[i, 0]))
            centers[i, 1] = max(r + 1e-10, min(1 - r - 1e-10, centers[i, 1]))

        lr *= 0.999

    return centers, radii


def grow_radii_iteratively(centers, radii, n, steps=2000):
    """Grow radii by computing max feasible radius for each circle."""
    centers = centers.copy()
    radii = radii.copy()

    for step in range(steps):
        max_r = np.full(n, 0.5)

        for i in range(n):
            # Boundary constraints
            max_r[i] = min(
                centers[i, 0], 1 - centers[i, 0],
                centers[i, 1], 1 - centers[i, 1]
            )

            # Neighbor constraints
            for j in range(n):
                if i != j:
                    dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                    max_r[i] = min(max_r[i], dist - radii[j])

        max_r = np.maximum(max_r, 1e-10)

        # Grow radii toward max
        alpha = 0.05 / (1 + step * 0.002)
        radii = radii + alpha * (max_r - radii)
        radii = np.minimum(radii, max_r - 1e-12)
        radii = np.maximum(radii, 1e-10)

    return centers, radii


def ensure_feasibility(centers, radii, n):

    centers = centers.copy()
    radii = radii.copy()

    # Clip to boundaries
    for i in range(n):
        r = radii[i]
        radii[i] = min(r, centers[i, 0], 1 - centers[i, 0],
                        centers[i, 1], 1 - centers[i, 1])
        radii[i] = max(radii[i], 1e-10)

    # Fix overlaps by reducing radii
