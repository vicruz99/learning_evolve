Empirical and theoretical results for circle packing indicate that for a fixed number of circles, equal radii typically yield the maximum sum of radii due to symmetry and the concavity of the constraint region relative to the objective.

We define variables for the center coordinates $(x_i, y_i)$ of all 26 circles and a single radius parameter $r$. The objective is to minimize $-26r$ (equivalent to maximizing $\sum r_i$). Constraints enforce:
1. Boundary conditions: $r \le x_i \le 1-r$ and $r \le y_i \le 1-r$
2. Non-overlap: $(x_i-x_j)^2 + (y_i-y_j)^2 \ge (2r)^2$ for all $i<j$

We use a two-stage optimization approach:
- **Initialization**: Place circles in a perturbed hexagonal lattice to provide a geometrically sound starting configuration.
- **Stage 1 (Global/Robust)**: Use COBYLA (Constrained Optimization BY Linear Approximations), a derivative-free method highly effective for complex geometric constraints and robust against local minima.
- **Stage 2 (Local/Precise)**: Refine the result using SLSQP, which utilizes gradient information for high-precision convergence.

import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def objective(vars):
    return -N_CIRCLES * vars[-1]

def constraint_boundary(vars):
    r = vars[-1]
    cx = vars[:-1].reshape(N_CIRCLES, 2)[:, 0]
    cy = vars[:-1].reshape(N_CIRCLES, 2)[:, 1]
    return np.concatenate([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])

def constraint_overlap(vars):
    r = vars[-1]
    cx = vars[:-1].reshape(N_CIRCLES, 2)
    diff = cx[:, np.newaxis, :] - cx[np.newaxis, :, :]
    dist_sq = np.sum(diff**2, axis=2)
    ij = np.triu_indices(N_CIRCLES, k=1)
    return dist_sq[ij] - 4.0 * r**2

def run_packing():
    rng = np.random.default_rng(42)
    
    # Hexagonal-inspired grid initialization
    cols = 6
    rows = 5
    step_x = 1.0 / (cols + 1)
    step_y = 1.0 / (rows + 1)
    
    centers = []
    for i in range(rows):
        for j in range(cols):
            if len(centers) >= N_CIRCLES:
                break
            x = step_x * (j + 1)
            y = step_y * (i + 1)
            if i % 2 == 1:
                x += step_x * 0.5
            centers.append([x, y])
            
    centers = np.array(centers[:N_CIRCLES])
    centers += rng.uniform(-0.01, 0.01, size=centers.shape)
    centers = np.clip(centers, 0.05, 0.95)
    
    r_init = 0.05
    x0 = np.concatenate([centers.flatten(), [r_init]])
    
    constraints = [
        {'type': 'ineq', 'fun': constraint_boundary},
        {'type': 'ineq', 'fun': constraint_overlap}
    ]
    
    bounds = [(0.0, 1.0)] * (2 * N_CIRCLES) + [(0.0, 0.5)]
    
    # Stage 1: Robust derivative-free optimization
    res1 = minimize(objective, x0, method='COBYLA', constraints=constraints, 
                    bounds=bounds, options={'rhobeg': 0.02, 'rhofinal': 1e-5, 'maxiter': 4000})
                    
    # Stage 2: Gradient-based local refinement for precision
    res2 = minimize(objective, res1.x, method='SLSQP', constraints=constraints, 
                    bounds=bounds, options={'maxiter': 1500, 'ftol': 1e-8})
                    
    final_res = res2 if res2.success else res1
    
    cx = final_res.x[:2 * N_CIRCLES].reshape(N_CIRCLES, 2)
    r_val = final_res.x[2 * N_CIRCLES]
    
    radii = np.full(N_CIRCLES, max(0.0, r_val))
    sum_radii = float(N_CIRCLES * r_val)
    
    return cx, radii, sum_radii
