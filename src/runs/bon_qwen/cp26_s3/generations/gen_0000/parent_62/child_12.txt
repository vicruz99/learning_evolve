2. **Force-Directed Simulation**: Before mathematical optimization, we run a physical simulation where circles repel each other and boundaries. This "grows" the circles iteratively, allowing them to settle into a valid, tightly packed configuration that satisfies all constraints. This phase effectively handles the non-convex geometry and ensures we start SLSQP from a feasible point.
3. **SLSQP Optimization**: We use `scipy.optimize.minimize` with the SLSQP algorithm to fine-tune the positions and maximize the sum of radii. SLSQP is robust for inequality-constrained problems and will push the radii to their theoretical limits given the fixed topology.

import numpy as np
from scipy.optimize import minimize

N = 26

def objective(vars):
    """Objective function: maximize sum of radii (minimize negative sum)"""
    radii = vars[2::3]
    return -np.sum(radii)

def compute_constraints(vars):
    """
    Computes all inequality constraints c(x) >= 0.
    Returns a 1D numpy array of constraint values.
    """
    centers = vars.reshape((N, 3))[:, :2]
    radii = vars.reshape((N, 3))[:, 2]
    
    constraints = []
    
    # 1. Pairwise non-overlap: ||c_i - c_j|| >= r_i + r_j
    # We only need i < j
    indices = np.triu_indices(N, k=1)
    c_i = centers[indices[0]]
    c_j = centers[indices[1]]
    r_i = radii[indices[0]]
    r_j = radii[indices[1]]
    
    dists = np.linalg.norm(c_i - c_j, axis=1)
    pairwise_cons = dists - r_i - r_j
    constraints.append(pairwise_cons)
    
    # 2. Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    constraints.append(centers[:, 0] - radii)       # Left
    constraints.append(1.0 - centers[:, 0] - radii) # Right
    constraints.append(centers[:, 1] - radii)       # Bottom
    constraints.append(1.0 - centers[:, 1] - radii) # Top
    
    # 3. Non-negative radii
    constraints.append(radii)
    
    return np.concatenate(constraints)

def hex_init():
    """Generates an initial hexagonal grid configuration for 26 circles"""
    centers = []
    # Approximate spacing for 26 circles
    dy = 1.0 / 7.0
    dx = dy * np.sqrt(3)
    
    y = dy
    while y < 1.0 - dy:
        x = dx / 2.0
        # Shift odd rows
        if len(centers) % 2 == 1:
            x = dx
        while x < 1.0 - dx/2.0:
            centers.append([x, y])
            x += dx
            if len(centers) >= N:
                break
        y += dy
        if len(centers) >= N:
            break
            
    # Pad or trim to exactly N
    centers = np.array(centers[:N])
    if len(centers) < N:
        # Fallback to random if grid is too small (unlikely for 26)
        pad = N - len(centers)
        rand_c = np.random.rand(pad, 2) * 0.8 + 0.1
        centers = np.vstack([centers, rand_c])
        
    return centers

def simulate_packing(centers, initial_r=0.02, steps=800):
    """
    Force-directed simulation to pack circles tightly and grow radii.
    Acts as a robust pre-optimizer for SLSQP.
    """
    radii = np.full(N, initial_r)
    c = centers.copy()
    
    for step in range(steps):
        # Compute pairwise distances
        diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
        dists = np.linalg.norm(diff, axis=2)
        
        # Repulsion forces for overlaps
        sums_r = radii[:, np.newaxis] + radii[np.newaxis, :]
        overlaps = sums_r - dists
        overlaps = np.maximum(overlaps, 0.0)
        
        # Direction vectors (avoid division by zero)
        safe_dists = np.where(dists > 1e-6, dists, 1e-6)
        directions = diff / safe_dists[:, :, np.newaxis]
        
        # Force magnitude: proportional to overlap, inversely to distance
        forces_mag = overlaps / safe_dists * 0.05
        force_vectors = forces_mag[:, :, np.newaxis] * directions
        
        # Accumulate forces
        total_force = np.sum(force_vectors, axis=1)
        
        # Update positions
        c += total_force
        
        # Boundary constraints: project back to valid region [r, 1-r]
        # This is more stable than soft boundary forces for packing
        for i in range(N):
            c[i, 0] = np.clip(c[i, 0], radii[i], 1.0 - radii[i])
            c[i, 1] = np.clip(c[i, 1], radii[i], 1.0 - radii[i])
            
        # Slowly grow radii
        radii *= 1.0003
        
    return c, radii

def run_packing():
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Try multiple seeds/configurations
    for seed in range(5):
        np.random.seed(seed)
        
        # 1. Initialize
        init_centers = hex_init()
        # Add small random perturbation to break symmetry
        init_centers += np.random.randn(N, 2) * 0.005
        init_centers = np.clip(init_centers, 0.05, 0.95)
        
        # 2. Simulate to find a tight valid packing
        centers, radii = simulate_packing(init_centers, initial_r=0.03, steps=600)
        
        # 3. Flatten for optimizer
        x0 = np.zeros(N * 3)
        for i in range(N):
            x0[3*i] = centers[i, 0]
            x0[3*i+1] = centers[i, 1]
            x0[3*i+2] = radii[i]
            
        # Bounds: x,y in [0,1], r in [0, 0.5]
        bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N
        
        # Constraints setup for SLSQP
        cons = {'type': 'ineq', 'fun': compute_constraints}
        
        # 4. Optimize
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, 
                          constraints=cons, options={'maxiter': 150, 'ftol': 1e-9})
            
            if res.success or res.fun < -best_sum:
                # Extract and validate
                opt_centers = res.x.reshape((N, 3))[:, :2]
                opt_radii = res.x.reshape((N, 3))[:, 2]
                current_sum = np.sum(opt_radii)
                
                # Quick validation check (simulating the validator logic)
                valid = True
                for i in range(N):
                    x, y, r = opt_centers[i,0], opt_centers[i,1], opt_radii[i]
                    if x - r < -1e-9 or x + r > 1 + 1e-9 or y - r < -1e-9 or y + r > 1 + 1e-9:
                        valid = False; break
                if valid:
                    for i in range(N):
                        for j in range(i+1, N):
                            d = np.sqrt(np.sum((opt_centers[i] - opt_centers[j])**2))
                            if d < opt_radii[i] + opt_radii[j] - 1e-9:
                                valid = False; break
                        if not valid: break
                            
                if valid and current_sum > best_sum:
                    best_sum = current_sum
                    best_centers = opt_centers
                    best_radii = opt_radii
                    
        except Exception:
            continue
            
    # Fallback if optimization failed (shouldn't happen)
    if best_centers is None:
        init_centers = hex_init()
        best_centers = init_centers
        best_radii = np.full(N, 0.04)
        best_sum = np.sum(best_radii)
        
    return best_centers, best_radii, best_sum
