import numpy as np
from scipy.optimize import minimize
import math

class BoundaryConstraint:
    def __init__(self, circle_idx, boundary_type, n):
        self.circle_idx = circle_idx
        self.boundary_type = boundary_type
        self.n = n
    
    def __call__(self, x):
        c = self.circle_idx
        n = self.n
        r = x[2 * n + c]
        cx = x[2 * c]
        cy = x[2 * c + 1]
        if self.boundary_type == 0:  # left
            return cx - r
        elif self.boundary_type == 1:  # right
            return 1.0 - cx - r
        elif self.boundary_type == 2:  # bottom
            return cy - r
        else:  # top
            return 1.0 - cy - r


class NonOverlapConstraint:
    def __init__(self, i, j, n):
        self.i = i
        self.j = j
        self.n = n
    
    def __call__(self, x):
        n = self.n
        ci = 2 * self.i
        cj = 2 * self.j
        ri = 2 * n + self.i
        rj = 2 * n + self.j
        dx = x[ci] - x[cj]
        dy = x[ci + 1] - x[cj + 1]
        rsum = x[ri] + x[rj]
        return dx * dx + dy * dy - rsum * rsum


class RadiusNonNegConstraint:
    def __init__(self, circle_idx, n):
        self.circle_idx = circle_idx
        self.n = n
    
    def __call__(self, x):
        return x[2 * self.n + self.circle_idx]


def objective_func(x, n):
    radii = x[2 * n:]
    return -np.sum(radii)


def create_bounds(n):
    bounds = []
    for i in range(2 * n):
        bounds.append((0.0, 1.0))
    for i in range(n):
        bounds.append((1e-12, 0.5))
    return bounds


def create_constraints(n):
    constraints = []
    for i in range(n):
        for bt in range(4):
            constraints.append({'type': 'ineq', 'fun': BoundaryConstraint(i, bt, n)})
    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({'type': 'ineq', 'fun': NonOverlapConstraint(i, j, n)})
    for i in range(n):
        constraints.append({'type': 'ineq', 'fun': RadiusNonNegConstraint(i, n)})
    return constraints


def init_hexagonal_v1(n):

    radii = np.full(n, 0.08)
    
    rows = [5, 6, 5, 5, 5]

    idx = 0
    for row_i, num_cols in enumerate(rows):
        y = r + row_i * dy
        if row_i % 2 == 0:
            x_start = r
        else:
            x_start = 2 * r
        for j in range(num_cols):
            x = x_start + j * 2 * r
            centers[idx] = [x, y]
            idx += 1
    
    return centers, radii

def optimize_packing(x0, n, max_iter=20000):
    bounds = create_bounds(n)
    constraints = create_constraints(n)
    
    result = minimize(
        objective_func,
        x0,
        args=(n,),
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={
            'maxiter': max_iter,
            'ftol': 1e-14,
            'disp': False,
            'iprint': -1
        }
    )
    return result

def project_to_feasible(centers, radii, n, tol=1e-8):
    new_centers = centers.copy()
    new_radii = radii.copy()
    
    for _ in range(100):
        violated = False
        
        for i in range(n):
            new_radii[i] = min(new_radii[i], 
                              new_centers[i, 0],
                              1.0 - new_centers[i, 0],
                              new_centers[i, 1],
                              1.0 - new_centers[i, 1])
            new_radii[i] = max(new_radii[i], 1e-10)
        
        for i in range(n):
            for j in range(i + 1, n):
                dx = new_centers[i, 0] - new_centers[j, 0]
                dy = new_centers[i, 1] - new_centers[j, 1]
                dist = math.sqrt(dx * dx + dy * dy)
                min_dist = new_radii[i] + new_radii[j] + tol
                if dist < min_dist and dist > 1e-14:
                    scale = (min_dist - dist) / (2.0 * min_dist + 1e-14)
                    new_radii[i] = max(new_radii[i] * (1 - scale), 1e-10)
                    new_radii[j] = max(new_radii[j] * (1 - scale), 1e-10)
                    violated = True
        
        if not violated:
            break
    
    return new_centers, new_radii
