import numpy as np
import cvxpy as cp
from scipy.optimize import minimize, differential_evolution


def solve_radii_lp(centers):
    """Given fixed centers, solve LP to find optimal radii maximizing sum of radii."""
    n = centers.shape[0]
    r = cp.Variable(n)
    
    constraints = [r >= 0]
    
    for i in range(n):
        x, y = centers[i]
        constraints.append(r[i] <= x)
        constraints.append(r[i] <= 1.0 - x)
        constraints.append(r[i] <= y)
        constraints.append(r[i] <= 1.0 - y)
    
    for i in range(n):
        for j in range(i + 1, n):
            d = np.linalg.norm(centers[i] - centers[j])
            constraints.append(r[i] + r[j] <= d)
    
    prob = cp.Problem(cp.Maximize(cp.sum(r)), constraints)
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
    except Exception:
        prob.solve(solver=cp.SCS, verbose=False)
    
    if prob.status in ['optimal', 'optimal_inaccurate']:
        return np.maximum(r.value, 0.0)
    return np.zeros(n)


def neg_sum_radii(positions_flat):
    """Objective function for position optimization."""
    n = 26
    centers = positions_flat.reshape(n, 2)
    # Ensure centers are valid
    centers = np.clip(centers, 0, 1)
    radii = solve_radii_lp(centers)
    return -np.sum(radii)


def generate_hexagonal_initial(n, seed=0):
    """Generate hexagonal lattice initial positions."""
    np.random.seed(seed)
    positions = []
    
    r_est = 0.1 + np.random.uniform(-0.02, 0.02)
    h = 2.0 * r_est
    v = h * np.sqrt(3) / 2.0
    
    margin = r_est * 0.8
    
    row = 0
    while len(positions) < n:
        col = 0
        offset = (row % 2) * h / 2.0
        while len(positions) < n:
            x = col * h + offset + margin
            y = row * v + margin
            if x - r_est >= 0 and x + r_est <= 1.0 and y - r_est >= 0 and y + r_est <= 1.0:
                positions.append([x, y])
                col += 1
            else:
                break
        row += 1
    
    result = np.array(positions[:n])
    
    if len(result) < n:
        # Fill remaining with random positions
        while len(result) < n:
            x = np.random.uniform(0.15, 0.85)
            y = np.random.uniform(0.15, 0.85)
            too_close = False
            for p in result:
                if np.linalg.norm([x, y] - p) < 2 * r_est * 0.8:
                    too_close = True
                    break
            if not too_close:
                result = np.vstack([result, [x, y]])
    
    return result[:n]


def generate_grid_initial(n, seed=0):
    """Generate grid-based initial positions."""
    np.random.seed(seed + 100)
    
    # Try to arrange in roughly sqrt(n) x sqrt(n)
    cols = int(np.ceil(np.sqrt(n * 1.2)))
    rows = int(np.ceil(n / cols))
    
    positions = []
    for row in range(rows):
        for col in range(cols):
            if len(positions) >= n:
                break
            x = (col + 0.5) / cols
            y = (row + 0.5) / rows
            # Add small random perturbation
            x += np.random.uniform(-0.02, 0.02)
            y += np.random.uniform(-0.02, 0.02)
            x = np.clip(x, 0.05, 0.95)
            y = np.clip(y, 0.05, 0.95)
            positions.append([x, y])
        if len(positions) >= n:
            break
    
    return np.array(positions[:n])


def run_packing():
    n = 26
    best_centers = None
    best_sum_radii = -1.0
    
    # Try multiple initial configurations with local optimization
    initial_configs = []
    
    # Hexagonal patterns with different seeds
    for seed in range(8):
        initial_configs.append(generate_hexagonal_initial(n, seed))
    
    # Grid patterns
    for seed in range(4):
        initial_configs.append(generate_grid_initial(n, seed))
    
    for centers_init in initial_configs:
        x0 = centers_init.flatten()
        
        # Nelder-Mead optimization
        result = minimize(
            neg_sum_radii,
            x0,
            method='Nelder-Mead',
            options={
                'maxiter': 80000,
                'xatol': 1e-10,
                'fatol': 1e-13,
                'adaptive': True
            }
        )
        
        centers_opt = result.x.reshape(n, 2)
        radii_opt = solve_radii_lp(centers_opt)
        current_sum = np.sum(radii_opt)
        
        if current_sum > best_sum_radii:
            best_sum_radii = current_sum
            best_centers = centers_opt.copy()
            best_radii = radii_opt.copy()
    
    # Try a few random restarts from best found
    for _ in range(5):
        perturbed = best_centers + np.random.normal(0, 0.01, best_centers.shape)
        perturbed = np.clip(perturbed, 0, 1)
        
        result = minimize(
            neg_sum_radii,
            perturbed.flatten(),
            method='Nelder-Mead',
            options={
                'maxiter': 50000,
                'xatol': 1e-10,
                'fatol': 1e-13
            }
        )
        
        centers_opt = result.x.reshape(n, 2)
        radii_opt = solve_radii_lp(centers_opt)
        current_sum = np.sum(radii_opt)
        
        if current_sum > best_sum_radii:
            best_sum_radii = current_sum
            best_centers = centers_opt.copy()
            best_radii = radii_opt.copy()
    
    # Final cleanup: ensure all radii are non-negative
    best_radii = np.maximum(best_radii, 0.0)
    
    return best_centers, best_radii, float(best_sum_radii)
