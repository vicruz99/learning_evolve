import numpy as np
from scipy.optimize import minimize
import math

def compute_loss(z, n, mu, bound_mu):
    """Computes the negative sum of radii plus penalty terms for constraint violations."""
    centers = z[:2*n].reshape(n, 2)
    radii = z[2*n:]
    loss = -np.sum(radii)
    
    # Pairwise overlap penalties
    for i in range(n):
        for j in range(i+1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = math.hypot(dx, dy)
            gap = radii[i] + radii[j] - dist
            if gap > 0:
                loss += mu * gap**2
                
    # Boundary violation penalties
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        # Gaps: r - x, r - (1-x), r - y, r - (1-y)
        gaps = [r - x, r + x - 1, r - y, r + y - 1]
        for g in gaps:
            if g > 0:
                loss += bound_mu * g**2
                
    return loss

def get_initial_configs(n, rng):
    """Generates multiple diverse initial configurations to improve global search."""
    configs = []
    
    # 1. 5x5 Grid with 1 extra
    z1 = np.zeros(3*n)
    idx = 0
    for r_idx in range(5):
        for c_idx in range(5):
            z1[2*idx] = 0.1 + c_idx * 0.2
            z1[2*idx+1] = 0.1 + r_idx * 0.2
            z1[2*n + idx] = 0.09
            idx += 1
    z1[2*25] = 0.5
    z1[2*25+1] = 0.5
    z1[2*n + 25] = 0.09
    configs.append(z1 + 0.005 * rng.normal(size=3*n))

    # 2. Hexagonal-like staggered rows
    z2 = np.zeros(3*n)
    idx = 0
    row = 0
    while idx < 26:
        start_x = 0.12 + (0.1 if row % 2 else 0)
        col = 0
        while idx < 26:
            x = start_x + col * 0.21
            y = 0.12 + row * 0.18
            if x < 0.9 and y < 0.9:
                z2[2*idx] = x
                z2[2*idx+1] = y
                z2[2*n + idx] = 0.10
                idx += 1
            col += 1
        row += 1
    configs.append(z2 + 0.005 * rng.normal(size=3*n))
    
    # 3. Random dense
    z3 = np.zeros(3*n)
    z3[:2*n] = rng.uniform(0.2, 0.8, size=(2*n,))
    z3[2*n:] = 0.06 + 0.02 * rng.random(n)
    configs.append(z3)
    
    # 4. Random spread
    z4 = np.zeros(3*n)
    z4[:2*n] = rng.uniform(0.15, 0.85, size=(2*n,))
    z4[2*n:] = 0.07 + 0.03 * rng.random(n)
    configs.append(z4)
    
    return configs

def run_packing():
    n = 26
    rng = np.random.default_rng(42)
    
    # Penalty weights. High values enforce strict constraints.
    mu = 5000.0
    bound_mu = 10000.0
    
    configs = get_initial_configs(n, rng)
    # Bounds for coordinates [0,1] and radii [1e-5, 0.5]
    bounds = [(0.0, 1.0) for _ in range(2*n)] + [(1e-5, 0.5) for _ in range(n)]
    
    best_z = None
    best_sum_r = -1.0
    
    # Optimize from each starting configuration
    for z0 in configs:
        res = minimize(compute_loss, z0, args=(n, mu, bound_mu), method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': 5000, 'ftol': 1e-12, 'gtol': 1e-10, 'disp': False})
        
        current_sum_r = np.sum(res.x[2*n:])
        if current_sum_r > best_sum_r:
            best_z = res.x
            best_sum_r = current_sum_r
            
    centers = best_z[:2*n].reshape(n, 2)
    radii = best_z[2*n:]
    
    # Post-optimization validation and strict correction
    max_viol = 0.0
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        v = max(0.0, r - x, r - (1.0 - x), r - y, r - (1.0 - y))
        if v > max_viol:
            max_viol = v
            
    for i in range(n):
        for j in range(i+1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = math.hypot(dx, dy)
            v = max(0.0, radii[i] + radii[j] - dist)
            if v > max_viol:
                max_viol = v
                
    # If any violation exists (even numerical), scale down radii uniformly to guarantee validity
    if max_viol > 1e-11:
        scale = 1.0 - max_viol - 1e-10
        if scale > 0:
            radii *= scale
            best_sum_r = np.sum(radii)
            
    return centers, radii, best_sum_r
