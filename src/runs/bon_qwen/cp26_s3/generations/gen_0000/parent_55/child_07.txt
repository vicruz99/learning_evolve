To maximize the sum of radii for 26 circles in a unit square, we cannot rely on a standard equal-circle packing because $r(25) = 0.1$ (for a $5 \times 5$ grid) and $r(n)$ is non-increasing. Thus, the maximum sum for equal circles is 2.6, which is below the target of 2.636.

To exceed this, we must use unequal circles, specifically concentrating radius in larger circles that are more efficient per unit area. The optimal strategy involves creating a "core" of large circles in the center and filling the boundaries with a ring of smaller circles. This arrangement allows us to fit 26 circles with a total sum of radii greater than 2.6.

I will implement a hybrid approach:
1.  **Hardcoded Base Configuration**: Start with a geometrically sound initial layout consisting of a $3 \times 3$ central block of larger circles and a surrounding layer of smaller circles. This provides a much better starting point than random or grid placement.
2.  **Force-Directed Optimization**: Run a simulation where circles repel each other and the boundaries, while a central pressure force attempts to expand all radii simultaneously. This local optimization will refine the positions and radii to find a valid, high-scoring configuration.

import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    
    # 1. Create a geometrically sound initial configuration
    # A 3x3 core with radius ~0.15 and a ring of smaller circles
    centers = np.zeros((26, 2))
    radii = np.zeros(26)
    
    # Central 3x3 block (9 circles)
    core_size = 0.13
    for i in range(3):
        for j in range(3):
            idx = i * 3 + j
            centers[idx] = [0.25 + j * 0.25, 0.25 + i * 0.25]
            radii[idx] = core_size

    # Middle ring (8 circles)
    mid_r = 0.09
    for i in range(8):
        idx = 9 + i
        # Place in the gaps of the 3x3 grid
        if i < 4:
            centers[idx] = [0.125 + i * 0.25, 0.125]
        elif i < 6:
            centers[idx] = [0.125, 0.375 + (i - 4) * 0.25]
        elif i < 7:
            centers[idx] = [0.875, 0.375 + (i - 6) * 0.25]
        else:
            centers[idx] = [0.125 + (i - 7) * 0.25, 0.875]
        radii[idx] = mid_r

    # Outer ring (9 circles) - corners and edges
    outer_r = 0.07
    for i in range(9):
        idx = 17 + i
        if i < 4:
            centers[idx] = [0.0625 + i * 0.25, 0.0625]
        elif i < 6:
            centers[idx] = [0.0625, 0.3125 + (i - 4) * 0.25]
        elif i < 7:
            centers[idx] = [0.9375, 0.3125 + (i - 6) * 0.25]
        else:
            centers[idx] = [0.0625 + (i - 7) * 0.25, 0.9375]
        radii[idx] = outer_r

    # 2. Force-Directed Optimization
    # We define a repulsion energy and a pressure term to expand radii
    # Variables: [x0, y0, x1, y1, ..., x25, y25, r0, r1, ..., r25]
    # However, to maintain stability, we optimize positions given fixed radii,
    # then iteratively adjust radii.
    
    positions = centers.copy()
    current_radii = radii.copy()
    
    # Optimization parameters
    lr = 0.005
    pressure = 0.002
    max_iters = 2000
    temperature = 0.05
    
    for _ in range(max_iters):
        forces = np.zeros_like(positions)
        
        # Calculate repulsion forces between circles
        for i in range(26):
            for j in range(i + 1, 26):
                diff = positions[i] - positions[j]
                dist_sq = np.sum(diff**2)
                dist = np.sqrt(dist_sq) if dist_sq > 1e-12 else 1e-12
                overlap = current_radii[i] + current_radii[j] - dist
                if overlap > 0:
                    force_magnitude = overlap / dist * 1000
                    forces[i] += diff * force_magnitude
                    forces[j] -= diff * force_magnitude
        
        # Boundary forces
        for i in range(26):
            for axis in range(2):
                if positions[i, axis] - current_radii[i] < 0:
                    forces[i, axis] += (current_radii[i] - positions[i, axis]) * 1000
                if positions[i, axis] + current_radii[i] > 1:
                    forces[i, axis] -= (positions[i, axis] + current_radii[i] - 1) * 1000
        
        # Apply forces
        positions += forces * lr
        
        # Ensure boundaries
        positions = np.clip(positions, current_radii[:, None], 1 - current_radii[:, None])
        
        # Expand radii (Pressure)
        current_radii += pressure * np.ones(26)
        
        # Reduce pressure over time
        pressure *= 0.9995
        
    # 3. Final cleanup to ensure validity
    for i in range(26):
        current_radii[i] = max(1e-4, current_radii[i])
        positions[i, 0] = np.clip(positions[i, 0], current_radii[i], 1 - current_radii[i])
        positions[i, 1] = np.clip(positions[i, 1], current_radii[i], 1 - current_radii[i])

    # Resolve final overlaps by slightly shrinking overlapping circles
    for _ in range(50):
        for i in range(26):
            for j in range(i + 1, 26):
                dist = np.sqrt(np.sum((positions[i] - positions[j])**2))
                overlap = (current_radii[i] + current_radii[j]) - dist
                if overlap > 1e-9:
                    # Split the overlap reduction proportionally
                    shrink = overlap / 2.0
                    current_radii[i] -= shrink
                    current_radii[j] -= shrink
                    current_radii[i] = max(1e-4, current_radii[i])
                    current_radii[j] = max(1e-4, current_radii[j])

    sum_radii = np.sum(current_radii)
    return positions, current_radii, sum_radii

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True
