import numpy as np
from scipy.optimize import minimize
import math

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        return False
    if np.isnan(radii).any():
        return False

    # Check if radii are nonnegative
    for i in range(n):
        if radii[i] < 0:
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False

    return True

def objective_function(params, n, alpha):
    """
    Objective function with penalties for constraints.
    params: flattened array [x1...xn, y1...yn, r1...rn]
    """
    x = params[0:n]
    y = params[n:2*n]
    r = params[2*n:3*n]
    
    # Primary objective: minimize negative sum of radii
    val = -np.sum(r)
    
    # Penalty for boundary violations
    # Constraints: r <= x <= 1-r  =>  x-r >= 0, 1-x-r >= 0
    #               r <= y <= 1-r  =>  y-r >= 0, 1-y-r >= 0
    
    # Using a large alpha ensures constraints are respected
    penalty = 0.0
    
    # Vectorized boundary checks
    lower_x = x - r
    upper_x = 1 - x - r
    lower_y = y - r
    upper_y = 1 - y - r
    
    # max(0, -violation)^2
    v1 = np.maximum(0, -lower_x)
    v2 = np.maximum(0, -upper_x)
    v3 = np.maximum(0, -lower_y)
    v4 = np.maximum(0, -upper_y)
    
    penalty += alpha * (np.sum(v1**2) + np.sum(v2**2) + np.sum(v3**2) + np.sum(v4**2))
    
    # Penalty for overlaps
    # Constraint: dist_ij >= r_i + r_j
    # Violation: r_i + r_j - dist_ij > 0
    
    # Iterate over pairs (N=26 is small enough for O(N^2))
    for i in range(n):
        for j in range(i + 1, n):
            dx = x[i] - x[j]
            dy = y[i] - y[j]
            dist = math.sqrt(dx*dx + dy*dy)
            sum_r = r[i] + r[j]
            violation = sum_r - dist
            if violation > 0:
                penalty += alpha * violation**2
                
    return val + penalty

def generate_initial_config(n, seed=None):
    """
    Generates an initial configuration of centers and radii.
    Uses a hexagonal lattice pattern for good density.
    """
    if seed is not None:
        np.random.seed(seed)
        
    # Estimate a reasonable radius based on density
    # Max density ~ 0.906. Area = 1.
    # n * pi * r^2 approx 0.906
    r_est = math.sqrt(0.906 / (n * math.pi))
    
    # Generate points on a hexagonal grid
    points = []
    # Vertical spacing for hex grid is r * sqrt(3)
    # Horizontal spacing is 2r. Odd rows shifted by r.
    
    # We iterate to fill n points
    # Start with a radius slightly smaller than estimate to ensure fit
    r_start = r_est * 0.95
    
    y = r_start
    row_idx = 0
    while len(points) < n:
        # Determine x offset for this row
        # Even rows (0, 2, ...): start at r
        # Odd rows (1, 3, ...): start at 2r (shifted by r relative to grid points)
        # Actually standard hex:
        # Row 0: (r, r), (3r, r), (5r, r)...
        # Row 1: (2r, r+sqrt(3)r), (4r, r+sqrt(3)r)...
        
        x_start = r_start if row_idx % 2 == 0 else 2 * r_start
        
        x = x_start
        while x <= 1 - r_start: # Ensure within bounds roughly
            if len(points) < n:
                points.append([x, y])
            x += 2 * r_start
        
        y += r_start * math.sqrt(3)
        row_idx += 1
        
    # If we didn't get enough points (unlikely with logic), fill randomly
    while len(points) < n:
        points.append([0.5, 0.5])
        
    centers = np.array(points[:n])
    radii = np.full(n, r_start)
    
    # Add small random noise to break symmetry and help optimization
    noise_scale = 0.01
    centers += np.random.normal(0, noise_scale, centers.shape)
    
    # Clip centers to [0, 1]
    centers = np.clip(centers, 0, 1)
    
    return centers, radii

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    # Penalty weight. Needs to be high enough to enforce constraints.
    alpha = 10000.0
    
    best_obj = -np.inf
    best_centers = None
    best_radii = None
    
    # Try multiple restarts to find a good local optimum
    # We'll try a hexagonal start and a few random perturbations
    seeds = [None, 42, 123, 456, 789, 1000]
    
    # Also try a random initialization strategy
    for i in range(10):
        if i < 6:
            centers, radii = generate_initial_config(n, seed=seeds[i])
        else:
            # Random initialization
            centers = np.random.uniform(0.1, 0.9, (n, 2))
            radii = np.full(n, 0.05)
            
        # Flatten parameters: [x1...xn, y1...yn, r1...rn]
        x = centers[:, 0]
        y = centers[:, 1]
        r = radii
        x0 = np.concatenate([x, y, r])
        
        # Bounds: x,y in [0,1], r in [0, 0.5]
        bounds = [(0, 1)] * (2 * n) + [(0, 0.5)] * n
        
        # Optimization options
        options = {'maxiter': 2000, 'ftol': 1e-9, 'gtol': 1e-6}
        
        try:
            res = minimize(
                objective_function, 
                x0, 
                args=(n, alpha), 
                method='L-BFGS-B', 
                bounds=bounds, 
                options=options
            )
            
            if res.success or res.fun < 0: # res.fun is negative sum + penalty
                # Extract results
                final_x = res.x[0:n]
                final_y = res.x[n:2*n]
                final_r = res.x[2*n:3*n]
                
                # Calculate actual sum of radii (without penalty)
                # Note: res.fun includes penalty. We want sum(r).
                # But if penalty is high, the solution is invalid.
                # We check validity separately.
                
                c = np.column_stack((final_x, final_y))
                r_arr = final_r
                
                # Quick validity check
                if validate_packing(c, r_arr):
                    current_sum = np.sum(r_arr)
                    if current_sum > best_obj:
                        best_obj = current_sum
                        best_centers = c
                        best_radii = r_arr
                else:
                    # If invalid, maybe scale down radii slightly to make valid?
                    # But better to rely on penalty method finding valid ones.
                    # If penalty was high, the objective would be poor anyway.
                    pass
                    
        except Exception as e:
            pass

    # Fallback if optimization failed to produce valid packing (unlikely)
    if best_centers is None:
        # Default safe packing
        best_centers, best_radii = generate_initial_config(n)
        best_obj = np.sum(best_radii)

    return best_centers, best_radii, best_obj

    centers, radii, total_sum = run_packing()
