import numpy as np
from scipy.optimize import minimize, LinearConstraint, NonlinearConstraint

def objective_func(v):
    """Minimize negative sum of radii to maximize total radius."""
    return -np.sum(v[2::3])

def circle_constraint_func(v, n):
    """
    Compute overlap constraints.
    Returns array of values (r_i + r_j)^2 - dist_ij^2.
    We require this array <= 0 for non-overlapping circles.
    """
    m = n * (n - 1) // 2
    vals = np.empty(m)
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            ix, iy, ir = 3 * i, 3 * i + 1, 3 * i + 2
            jx, jy, jr = 3 * j, 3 * j + 1, 3 * j + 2
            dx = v[ix] - v[jx]
            dy = v[iy] - v[jy]
            r_sum = v[ir] + v[jr]
            vals[idx] = r_sum * r_sum - (dx * dx + dy * dy)
            idx += 1
    return vals

def generate_initial_centers(n):
    """Generate a hexagonal lattice initial guess inside [0,1]^2."""
    pts = []
    s = 0.18  # Lattice scaling factor
    for row in range(12):
        y = row * s * np.sqrt(3) / 2
        for col in range(12):
            x = col * s + (row % 2) * s / 2
            if 0.05 <= x <= 0.95 and 0.05 <= y <= 0.95:
                pts.append([x, y])
                
    pts = np.array(pts)
    if len(pts) > n:
        # Keep points closest to the center of the square
        dists = np.sqrt((pts[:, 0] - 0.5)**2 + (pts[:, 1] - 0.5)**2)
        order = np.argsort(dists)
        pts = pts[order[:n]]
    elif len(pts) < n:
        # Fallback to random if lattice doesn't yield enough points
        extra = np.random.uniform(0.1, 0.9, size=(n - len(pts), 2))
        pts = np.vstack([pts, extra])
        
    return pts[:n]

def run_packing():
    n = 26
    
    # Linear constraints for boundaries: A @ v <= b
    # Variables layout: [x1, y1, r1, x2, y2, r2, ...]
    A = np.zeros((4 * n, 3 * n))
    b = np.zeros(4 * n)
    for i in range(n):
        ix, iy, ir = 3 * i, 3 * i + 1, 3 * i + 2
        # x - r >= 0  =>  -x + r <= 0
        A[4 * i, ix] = -1.0
        A[4 * i, ir] = 1.0
        # x + r <= 1
        A[4 * i + 1, ix] = 1.0
        A[4 * i + 1, ir] = 1.0
        b[4 * i + 1] = 1.0
        # y - r >= 0  =>  -y + r <= 0
        A[4 * i + 2, iy] = -1.0
        A[4 * i + 2, ir] = 1.0
        # y + r <= 1
        A[4 * i + 3, iy] = 1.0
        A[4 * i + 3, ir] = 1.0
        b[4 * i + 3] = 1.0

    lin_cons = LinearConstraint(A, -np.inf, b)
    
    # Nonlinear constraints for circle overlaps
    nlc = NonlinearConstraint(
        circle_constraint_func, 
        -np.inf, 
        np.zeros(n * (n - 1) // 2), 
        args=(n,)
    )
    
    # Variable bounds: x,y in [0,1], r in [0, 0.5]
    bounds = [(0.0, 1.0) if i % 3 < 2 else (0.0, 0.5) for i in range(3 * n)]

    best_obj = -np.inf
    best_x = None

    # Multi-start optimization to avoid local minima
    for trial in range(8):
        np.random.seed(trial * 137 + 42)
        
        centers = generate_initial_centers(n)
        # Add small perturbation
        centers += np.random.normal(0, 0.005, size=centers.shape)
        centers = np.clip(centers, 0.05, 0.95)

        x0 = np.concatenate([centers.flatten(), radii])

            x0, 
            method='SLSQP',

    # to guarantee validator tolerance pass
    radii = np.maximum(radii, 1e-9)

    return centers, radii, np.sum(radii)
