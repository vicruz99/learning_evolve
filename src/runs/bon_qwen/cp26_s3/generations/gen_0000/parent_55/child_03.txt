import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def objective_func(v):
    """Minimize negative sum of radii."""
    return -np.sum(v[2 * N_CIRCLES:])

def constraint_func(v):
    """Non-overlap constraints: dist^2 >= (r_i + r_j)^2"""
    n = N_CIRCLES
    centers = v[:2 * n].reshape((n, 2))
    radii = v[2 * n:]
    cons = []
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            cons.append(dx * dx + dy * dy - (radii[i] + radii[j]) ** 2)
    return np.array(cons)

def create_hex_init():
    """Create a hexagonal lattice initialization."""
    v0 = np.zeros(3 * N_CIRCLES)
    idx = 0
    for r in range(6):
        y = (r + 0.5) / 6.0
        shift = 0.1 if r % 2 == 1 else 0.0
        for c in range(5):
            if idx < N_CIRCLES:
                x = (c + 0.5) / 5.0 + shift
                v0[idx * 3] = x
                v0[idx * 3 + 1] = y
                v0[idx * 3 + 2] = 0.08  # Initial radius
                idx += 1
    return v0

def run_packing():
    np.random.seed(42)
    bounds = [(0, 1)] * (2 * N_CIRCLES) + [(0, 0.5)] * N_CIRCLES
    cons = {'type': 'ineq', 'fun': constraint_func}
    
    options = {'ftol': 1e-10, 'maxiter': 1000, 'disp': False}
    
    best_sum = -np.inf
    best_v = None
    
    # Generate initial configurations
    inits = [create_hex_init()]

        x0 = np.random.uniform(0.12, 0.88, N_CIRCLES)
        y0 = np.random.uniform(0.12, 0.88, N_CIRCLES)

        inits.append(np.concatenate([x0, y0, r0]))

            current_sum = np.sum(res.x[2 * N_CIRCLES:])

                best_sum = current_sum
                best_v = res.x.copy()
        except Exception:
            continue

        best_v = create_hex_init()

                       constraints=cons, options=options)
        best_v = res.x
    except Exception:
        pass
        
    # Refinement phase 2: Perturb and optimize to escape local optima
    for _ in range(3):
        perturbed = best_v + np.random.normal(0, 0.005, size=best_v.shape)
        # Clip to bounds to avoid invalid starts
        perturbed[:2*N_CIRCLES] = np.clip(perturbed[:2*N_CIRCLES], 0, 1)
        perturbed[2*N_CIRCLES:] = np.clip(perturbed[2*N_CIRCLES:], 0, 0.5)

                best_sum = np.sum(best_v[2*N_CIRCLES:])
