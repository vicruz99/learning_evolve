import numpy as np

from scipy.spatial.distance import pdist, squareform

def get_optimal_radii(centers):
    """
    Solves the LP to find radii that maximize sum(r) for fixed centers.
    Constraints:
    1. r_i + r_j <= distance(c_i, c_j)
    2. r_i <= x_i, 1-x_i, y_i, 1-y_i (inside square)
    3. r_i >= 0
    """
    n = centers.shape[0]
    
    # Precompute pairwise distances
    dists = squareform(pdist(centers))
    
    # Objective: Maximize sum(r) -> Minimize -sum(r)
    c_obj = -np.ones(n)
    
    # Constraints matrix A_ub and vector b_ub for A_ub @ r <= b_ub
    A_ub = []
    b_ub = []
    
    # Wall constraints: r_i <= x_i, r_i <= 1-x_i, etc.
    for i in range(n):
        x, y = centers[i]
        # Constraint row for r_i
        row = np.zeros(n)
        row[i] = 1.0
        
        # r_i <= x
        A_ub.append(row.copy())
        b_ub.append(x)
        
        # r_i <= 1 - x
        A_ub.append(row.copy())
        b_ub.append(1.0 - x)
        
        # r_i <= y
        A_ub.append(row.copy())
        b_ub.append(y)
        
        # r_i <= 1 - y
        A_ub.append(row.copy())
        b_ub.append(1.0 - y)
        
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            d = dists[i, j]
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(d)
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    # Bounds for radii: r_i >= 0
    bounds = [(0, None) for _ in range(n)]
    
    try:
        # Use HiGHS solver for speed and reliability
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, np.sum(res.x)
        else:
            # Fallback if infeasible (should not happen if centers valid)
            return np.zeros(n), 0.0
    except Exception:
        return np.zeros(n), 0.0
