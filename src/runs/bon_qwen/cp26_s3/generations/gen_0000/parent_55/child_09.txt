import math
from scipy.optimize import minimize

def run_packing():

    Pack 26 circles in a unit square to maximize the sum of radii.

        row_idx = int(round((y - r_init) / dy))
        if row_idx % 2 == 0:
            x_start = r_init
        else:
            x_start = r_init + r_init # shift by r (half of 2r spacing)
            
        x = x_start
        while x < 1 - r_init and idx < n:
            centers[idx] = [x, y]
            radii[idx] = r_init
            idx += 1
            x += 2 * r_init # Horizontal spacing 2r
        
        y += dy

    for step in range(500):
        forces = np.zeros_like(centers)
        
        # Compute forces between circles
        for i in range(n):
            for j in range(i + 1, n):
                diff = centers[i] - centers[j]
                dist = np.linalg.norm(diff)
                min_dist = radii[i] + radii[j]
                
                if dist < min_dist and dist > 1e-6:
                    # Repulsive force proportional to overlap
                    overlap = min_dist - dist
                    # Force direction is along diff
                    force_mag = repulsion_k * overlap
                    force_vec = force_mag * (diff / dist)
                    forces[i] += force_vec
                    forces[j] -= force_vec
                elif dist < 1e-6:
                    # Very close, push apart randomly or based on index
                    forces[i] += np.random.rand(2) * 0.01
                    forces[j] -= np.random.rand(2) * 0.01

        # Compute wall forces
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            
            # Left wall
            if x - r < 0:
                forces[i, 0] += wall_k * (r - x) # Push right
            # Right wall
            if x + r > 1:
                forces[i, 0] -= wall_k * (x + r - 1) # Push left
            # Bottom wall
            if y - r < 0:
                forces[i, 1] += wall_k * (r - y) # Push up
            # Top wall
            if y + r > 1:
                forces[i, 1] -= wall_k * (y + r - 1) # Push down

        # Update centers
        centers += dt * forces

            )

                        max_r = limit

    y0 = centers[:, 1].copy()

        bounds.append((1e-6, 0.5)) # r

        r_vars = vars[2::3]
        return -np.sum(r_vars)

    def constraint_wall(vars):
        c = vars[0::3]
        s = vars[1::3]
        r = vars[2::3]
        # x - r >= 0
        cons1 = c - r
        # 1 - x - r >= 0
        cons2 = 1 - c - r
        # y - r >= 0
        cons3 = s - r
        # 1 - y - r >= 0
        cons4 = 1 - s - r
        return np.concatenate([cons1, cons2, cons3, cons4])

        s = vars[1::3]
        r = vars[2::3]

                dist_sq = dx*dx + dy*dy

                val = np.sqrt(dist_sq) - (r[i] + r[j])
                dist_cons.append(val)
        return np.array(dist_cons)

    # Define constraints for optimizer
    cons = []
    # Wall constraints
    cons.append({'type': 'ineq', 'fun': constraint_wall})
    # Distance constraints
    cons.append({'type': 'ineq', 'fun': constraint_distance})

    # Run optimizer
    # Method SLSQP is suitable for constrained optimization

        final_x = res.x[0::3]
        final_y = res.x[1::3]
        final_r = res.x[2::3]
    else:
        # Fallback to force simulation result if optimizer fails
        final_x = centers[:, 0]
        final_y = centers[:, 1]
        final_r = radii

    final_centers = np.column_stack((final_x, final_y))
    
    # Final validation check and cleanup
    # Ensure strictly valid
    for i in range(n):
        r = final_r[i]
        x, y = final_centers[i]
        # Project to valid region if slightly out
        if x < r: x = r
        if x > 1-r: x = 1-r
        if y < r: y = r
        if y > 1-r: y = 1-r
        final_centers[i] = [x, y]

    return final_centers, final_r, np.sum(final_r)
