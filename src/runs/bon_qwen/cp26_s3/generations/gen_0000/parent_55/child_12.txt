import numpy as np
from scipy.optimize import linprog, minimize


def solve_radii_lp(centers):
    """Given fixed centers, solve LP to maximize sum of radii."""
    n = centers.shape[0]
    
    # Objective: maximize sum(r_i) => minimize -sum(r_i)
    c_obj = -np.ones(n)
    
    A_ub = []
    b_ub = []
    
    # Pairwise non-overlap: r_i + r_j <= dist(i,j)
    for i in range(n):
        for j in range(i + 1, n):
            diff = centers[i] - centers[j]
            dist = np.sqrt(np.dot(diff, diff))
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dist)
    
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        bounds_i = min(centers[i, 0], 1.0 - centers[i, 0],
                       centers[i, 1], 1.0 - centers[i, 1])
        row = np.zeros(n)
        row[i] = 1.0
        A_ub.append(row)
        b_ub.append(bounds_i)
    
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None) for _ in range(n)]
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        return result.x
    else:
        # Fallback: compute radii from distances
        radii = np.ones(n) * 0.01
        for i in range(n):
            max_r = min(centers[i, 0], 1.0 - centers[i, 0],
                        centers[i, 1], 1.0 - centers[i, 1])
            for j in range(n):
                if i != j:
                    dist = np.sqrt(np.dot(centers[i] - centers[j], centers[i] - centers[j]))
                    max_r = min(max_r, dist / 2.0)
            radii[i] = max_r
        return radii


def neg_sum_radii(centers_flat):
    """Negative sum of radii for minimization."""
    n = 26
    centers = centers_flat.reshape(n, 2)
    centers = np.clip(centers, 0.005, 0.995)
    radii = solve_radii_lp(centers)
    return -np.sum(radii)


def hexagonal_init(n, row_config, perturb=0.0):
    """Initialize centers in hexagonal pattern."""
    centers = np.zeros((n, 2))
    idx = 0
    num_rows = len(row_config)
    
    for row_idx, count in enumerate(row_config):
        y = 0.08 + row_idx * (0.84 / (num_rows - 1)) if num_rows > 1 else 0.5
        for col_idx in range(count):
            x_spacing = 0.82 / (max(count, 1) + 0.5)
            x = 0.08 + col_idx * x_spacing + (x_spacing * 0.5 if row_idx % 2 == 1 else 0)
            centers[idx] = [x, y]
            idx += 1
    
    if perturb > 0:
        centers += np.random.normal(0, perturb, centers.shape)
    
    centers = np.clip(centers, 0.05, 0.95)
    return centers


def run_packing():
    n = 26
    
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Try various hexagonal configurations
    row_configs = [
        [6, 5, 6, 5, 4],
        [5, 6, 5, 6, 4],
        [7, 6, 7, 6],
        [6, 6, 6, 8],
        [4, 6, 6, 6, 4],
        [5, 5, 5, 5, 6],
        [6, 6, 5, 5, 4],
        [5, 7, 5, 7, 2],
        [6, 6, 6, 6, 2],
        [8, 6, 6, 6],
    ]
    
    all_configs = []
    for rc in row_configs:
        if sum(rc) == n:
            all_configs.append(rc)
    
    # Phase 1: Multiple restarts with local optimization
    for config_idx, rc in enumerate(all_configs):
        for restart in range(8):
            perturb = 0.04 * (0.95 ** restart)
            centers = hexagonal_init(n, rc, perturb=perturb)
            
            # Add more randomness for later restarts
            if restart > 3:
                centers += np.random.normal(0, 0.03, centers.shape)
                centers = np.clip(centers, 0.05, 0.95)
            
            try:
                result = minimize(neg_sum_radii, centers.flatten(),
                                method='Nelder-Mead',
                                options={'maxiter': 8000, 'xatol': 1e-10, 'fatol': 1e-11})
                
                current_centers = result.x.reshape(n, 2)
                current_centers = np.clip(current_centers, 0.005, 0.995)
                current_radii = solve_radii_lp(current_centers)
                current_sum = np.sum(current_radii)
                
                if current_sum > best_sum:
                    best_sum = current_sum
                    best_centers = current_centers.copy()
                    best_radii = current_radii.copy()
            except Exception:
                pass
    
    # Phase 2: Refinement from best solution
    if best_centers is not None:
        for refine in range(15):
            centers = best_centers.copy()

            centers = np.clip(centers, 0.02, 0.98)
            
            try:
                result = minimize(neg_sum_radii, centers.flatten(),
                                method='Nelder-Mead',
                                options={'maxiter': 12000, 'xatol': 1e-11, 'fatol': 1e-12})
                
                current_centers = result.x.reshape(n, 2)

                    best_radii = current_radii.copy()
            except Exception:
                pass

    for random_start in range(10):
        centers = np.random.uniform(0.1, 0.9, (n, 2))

            current_centers = result.x.reshape(n, 2)

                best_sum = current_sum
                best_centers = current_centers.copy()
                best_radii = current_radii.copy()
        except Exception:
            pass
    
    return best_centers, best_radii, float(best_sum)
