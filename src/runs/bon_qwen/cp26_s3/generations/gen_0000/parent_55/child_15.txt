import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Runs a constrained optimization to pack 26 circles in a unit square.
    """
    n = 26
    
    # 1. Heuristic Initialization (Hexagonal Lattice)
    # We generate a hexagonal grid and select the first 26 points that fit.
    # Hexagonal packing has a vertical row spacing of sqrt(3)/2 * diameter.
    # Assuming a starting r, we can place centers at:
    # (row * spacing, col * 2r) or (row * spacing, col * 2r + r)
    
    # Let's create a grid of potential centers
    # We'll use a rough guess of r = 0.1013
    r_guess = 0.1013
    row_spacing = r_guess * np.sqrt(3)
    
    centers_list = []
    # Generate rows within a slightly larger box to ensure we have enough points
    for r_idx in range(10):
        y = r_guess + r_idx * row_spacing
        if y > 1.0:
            continue
        # Shift odd rows by r_guess
        shift = r_guess if r_idx % 2 == 1 else 0
        for c_idx in range(10):
            x = r_guess + shift + c_idx * (2 * r_guess)
            if 0 <= x <= 1.0 and 0 <= y <= 1.0:
                centers_list.append([x, y])
                
    # If we have enough points, take the first 26
    # If not, fallback to random
    if len(centers_list) >= n:
        # Try to find a subset that is well-distributed? 
        # Simply taking the first 26 from a dense grid usually works well.
        # To ensure they are packed tightly, we might want to pick the "most inner" or just first valid.
        # For the optimizer, just a valid start is enough.
        init_centers = np.array(centers_list[:n])
    else:
        # Fallback: Random valid centers
        init_centers = np.random.rand(n, 2) * 0.8 + 0.1
        # Sort to keep determinism
        init_centers = np.sort(init_centers, axis=0)

    # 2. Optimization Setup
    # Variables: x1, y1, ..., x26, y26, r (Total 53)
    # We flatten the centers array.
    x0 = np.hstack([init_centers.flatten(), [r_guess]])
    
    # Bounds for variables
    # x, y in [0, 1]
    # r in [0, 0.5] (theoretical max for single circle)
    bounds = []
    for _ in range(n * 2):
        bounds.append((0, 1))
    bounds.append((0, 0.5))

    # Objective: Maximize r => Minimize -r
    def objective(vars):
        return -vars[-1]

    # Constraints:
    # 1. Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    #    => x - r >= 0, 1 - r - x >= 0, y - r >= 0, 1 - r - y >= 0
    # 2. Non-overlap: dist^2 >= (2r)^2 => (xi-xj)^2 + (yi-yj)^2 - 4r^2 >= 0
    
    def constraints(vars):
        centers = vars[:52].reshape(n, 2)
        r = vars[52]
        cons = []
        
        # Boundary constraints
        # x >= r
        cons.append(centers[:, 0] - r)
        # x <= 1-r
        cons.append(1 - r - centers[:, 0])
        # y >= r
        cons.append(centers[:, 1] - r)
        # y <= 1-r
        cons.append(1 - r - centers[:, 1])
        
        # Pairwise non-overlap constraints
        # We compute all pair distances squared
        # Vectorized distance calculation
        # centers shape (n, 2)
        # diff shape (n, n, 2)
        # dist_sq shape (n, n)
        diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
        dist_sq = np.sum(diff**2, axis=2)
        
        # We only need upper triangle (i < j)
        # Flatten the upper triangle
        upper_tri_indices = np.triu_indices(n, k=1)
        pair_dists_sq = dist_sq[upper_tri_indices]
        
        # Constraint: dist_sq >= 4r^2
        cons.append(pair_dists_sq - 4 * r * r)
        
        return cons

    # Define constraints in scipy format
    # Each constraint is a dict with 'type' and 'fun'
    # 'fun' returns an array of values. 
    # For SLSQP, we can pass a list of constraint dicts or a single dict with a function returning array?
    # Actually, standard SLSQP expects 'fun' to return a scalar for one constraint, 
    # or we can pass a list of constraints. 
    # However, passing a single constraint with 'type': 'ineq' and 'fun' returning a numpy array 
    # is supported in recent SciPy versions (treats each element as a constraint).
    
    # Let's define a wrapper to return a flat array of all constraints
    def combined_constraints(vars):
        centers = vars[:52].reshape(n, 2)
        r = vars[52]
        
        # Boundary
        b1 = centers[:, 0] - r
        b2 = 1 - r - centers[:, 0]
        b3 = centers[:, 1] - r
        b4 = 1 - r - centers[:, 1]
        
        # Pairwise
        diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
        dist_sq = np.sum(diff**2, axis=2)
        upper_tri_indices = np.triu_indices(n, k=1)
        pair_dists_sq = dist_sq[upper_tri_indices]
        
        # Combine all
        return np.concatenate([b1, b2, b3, b4, pair_dists_sq - 4 * r * r])

    constraint_dict = {
        'type': 'ineq',
        'fun': combined_constraints
    }

    # Run Optimization
    # Use SLSQP for constrained optimization
    # Options: maxiter, ftol, ...
    res = minimize(objective, x0, method='SLSQP', bounds=bounds, 
                   constraints=constraint_dict, 
                   options={'maxiter': 1000, 'ftol': 1e-9})

    # Extract results
    best_centers = res.x[:52].reshape(n, 2)
    best_r = res.x[52]
    
    # If optimization failed to improve, fallback to initialization
    if not res.success:
        # Check if current solution is valid
        # We can try to compute valid r for init_centers
        # But usually minimize finds something even if not fully converged
        pass

    # Construct radii array (all equal)
    best_radii = np.full(n, best_r)
    sum_radii = np.sum(best_radii)

    return best_centers, best_radii, sum_radii
