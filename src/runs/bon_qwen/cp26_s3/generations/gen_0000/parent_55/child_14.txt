import numpy as np
from scipy.optimize import minimize

N = 26

def objective(vars):
    return -np.sum(vars[2*N:])

def constraint(vars):
    c = vars[:2*N].reshape(N, 2)
    r = vars[2*N:]
    cons = []
    # Boundary constraints
    cons.append(c[:, 0] - r)
    cons.append(1 - c[:, 0] - r)
    cons.append(c[:, 1] - r)
    cons.append(1 - c[:, 1] - r)
    # Pairwise non-overlap constraints
    for i in range(N):
        for j in range(i + 1, N):
            d2 = np.sum((c[i] - c[j])**2)
            cons.append(d2 - (r[i] + r[j])**2)
    return np.concatenate(cons)

def run_packing():
    centers = []
    y_start = 0.15
    dx = 0.18
    dy = dx * np.sqrt(3) / 2
    offset_x = dx / 2
    
    # Hexagonal initialization: 5 rows with counts [5, 5, 5, 5, 6]
    row_counts = [5, 5, 5, 5, 6]
    for i, count in enumerate(row_counts):
        if len(centers) >= N:
            break
        y = y_start + i * dy
        x_start = 0.1 + (offset_x if i % 2 == 1 else 0)
        for j in range(count):
            x = x_start + j * dx
            centers.append([x, y])
            
    centers = np.array(centers[:N])
    radii = np.full(N, 0.08)
    
    x0 = np.concatenate([centers.flatten(), radii])
    bounds = [(0, 1)] * (2 * N) + [(0, 0.5)] * N
    
    cons = {'type': 'ineq', 'fun': constraint}
    
    # Optimize
    res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 2000, 'ftol': 1e-12})
    
    final_vars = res.x
    final_centers = final_vars[:2*N].reshape(N, 2)
    final_radii = final_vars[2*N:]
    
    # Safety corrections for numerical precision and strict validity
    final_radii = np.maximum(final_radii, 0)
    final_radii *= 0.9999  # Tiny margin to avoid boundary/overlap tolerance issues
    
    for i in range(N):
        final_centers[i, 0] = np.clip(final_centers[i, 0], final_radii[i], 1 - final_radii[i])
        final_centers[i, 1] = np.clip(final_centers[i, 1], final_radii[i], 1 - final_radii[i])
        
    return final_centers, final_radii, np.sum(final_radii)
