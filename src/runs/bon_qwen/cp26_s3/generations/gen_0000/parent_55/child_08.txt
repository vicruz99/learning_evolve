import numpy as np
from scipy.optimize import minimize

NUM_CIRCLES = 26
PENALTY_WEIGHT = 3000.0

def compute_objective(vars, n):
    """
    Computes the negative sum of radii plus penalties for constraint violations.
    vars: 1D array of shape [2*n + n] containing [x1, y1, r1, ..., xn, yn, rn]
    """
    centers = vars[:2*n].reshape(n, 2)
    radii = vars[2*n:]
    
    loss = 0.0
    
    # Boundary constraint penalties
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        # Circle must stay within [0, 1]
        if x < r: loss += (r - x)**2
        if x > 1.0 - r: loss += (x + r - 1.0)**2
        if y < r: loss += (r - y)**2
        if y > 1.0 - r: loss += (y + r - 1.0)**2
        
    # Overlap constraint penalties
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx*dx + dy*dy)
            req_dist = radii[i] + radii[j]
            if dist < req_dist:
                loss += (req_dist - dist)**2
                
    # We want to maximize sum of radii => minimize negative sum
    return -np.sum(radii) + PENALTY_WEIGHT * loss

def run_packing():
    n = NUM_CIRCLES
    best_loss = float('inf')
    best_centers = None
    best_radii = None
    
    # Try multiple initial configurations to avoid poor local minima
    for seed in range(4):
        np.random.seed(seed)
        
        # Initialize with a grid pattern scaled to leave room for boundaries
        pts_x = np.linspace(0.1, 0.9, 6)
        pts_y = np.linspace(0.1, 0.9, 5)
        grid = np.zeros((30, 2))
        idx = 0
        for y in pts_y:
            for x in pts_x:
                grid[idx, 0] = x
                grid[idx, 1] = y
                idx += 1
                
        init_centers = grid[:n].copy()
        # Add small random noise to break symmetry
        init_centers += np.random.uniform(-0.015, 0.015, size=init_centers.shape)
        init_centers = np.clip(init_centers, 0.0, 1.0)
        
        init_radii = np.ones(n) * 0.05
        init_vars = np.concatenate([init_centers.ravel(), init_radii])
        
        bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n
        
        res = minimize(compute_objective, init_vars, args=(n,),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': 3000, 'ftol': 1e-10})
        
        if res.fun < best_loss:
            best_loss = res.fun
            best_centers = res.x[:2*n].reshape(n, 2).copy()
            best_radii = res.x[2*n:].copy()
            
    # Post-processing: Strictly enforce constraints by shrinking radii if necessary
    # This guarantees the output passes validation with 1e-12 tolerance
    for _ in range(100):
        any_change = False
        for i in range(n):
            max_allowed = min(best_centers[i, 0], 1.0 - best_centers[i, 0],
                              best_centers[i, 1], 1.0 - best_centers[i, 1])
            
            for j in range(n):
                if i != j:
                    dist = np.sqrt(np.sum((best_centers[i] - best_centers[j])**2))
                    allowed_by_j = dist - best_radii[j]

                        max_allowed = allowed_by_j
                        
            if max_allowed < best_radii[i] - 1e-12:
                best_radii[i] = max(max_allowed, 0.0)
                any_change = True
        if not any_change:
            break
            
    return best_centers, best_radii, np.sum(best_radii)
