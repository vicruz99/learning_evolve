1. Equal circles would give ~1.47 sum (r≈0.0566), far from target
2. Unequal circles allow much better packing: large circles in corners, smaller ones filling gaps
3. At optimality, circles should be "jammed" - touching neighbors/boundaries

**Strategy:**
1. **Smart Initialization**: Start with hexagonal-ish grid patterns, with some circles slightly larger in corners
2. **Force-Directed Growth (Lubachevsky-Stillinger style)**: Grow all radii slowly while resolving overlaps via repulsive forces. This naturally finds jammed configurations.
3. **Gradient-Based Refinement**: Use L-BFGS-B with penalty method to fine-tune positions and radii simultaneously
4. **Multiple Random Restarts**: Try different seeds and initial patterns to escape local optima
5. **Progressive Growth**: Run multiple phases of growth with decreasing step sizes for fine convergence

**Implementation Details:**
- 78 variables (2 center coords + 1 radius per circle × 26 circles)
- Constraints handled via penalty functions: boundary penalties and overlap penalties
- Boundary forces push circles away from edges by a margin
- Repulsion proportional to overlap amount

import numpy as np
from scipy.optimize import minimize


def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26

    def is_valid(centers, radii):
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                return False
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist < radii[i] + radii[j] - 1e-12:
                    return False
        return True

    def force_directed_optimize(centers, radii, n, steps, growth_rate):
        for step in range(steps):
            forces = np.zeros_like(centers)

            # Repulsive forces between overlapping/nearby circles
            for i in range(n):
                for j in range(i + 1, n):
                    diff = centers[i] - centers[j]
                    dist = np.linalg.norm(diff)
                    min_dist = radii[i] + radii[j]

                    if dist < min_dist and dist > 1e-10:
                        overlap = min_dist - dist
                        force_mag = overlap * 0.5
                        forces[i] += force_mag * diff / dist
                        forces[j] -= force_mag * diff / dist
                    elif dist < min_dist * 1.3 and dist > 1e-10:
                        # Weak attraction to keep packing tight
                        attr = (min_dist * 1.3 - dist) * 0.005
                        forces[i] -= attr * diff / dist
                        forces[j] += attr * diff / dist

            # Boundary repulsion
            for i in range(n):
                x, y = centers[i]
                r = radii[i]
                margin = 0.005
                if x - r < margin:
                    forces[i, 0] += (margin - (x - r)) * 50
                if x + r > 1 - margin:
                    forces[i, 0] -= (x + r - (1 - margin)) * 50
                if y - r < margin:
                    forces[i, 1] += (margin - (y - r)) * 50
                if y + r > 1 - margin:
                    forces[i, 1] -= (y + r - (1 - margin)) * 50

            # Apply forces
            lr = 0.003 / (1 + step * 0.0005)
            centers += lr * forces

            # Clamp centers to valid range
            for i in range(n):
                centers[i, 0] = np.clip(centers[i, 0], radii[i] + 1e-7, 1 - radii[i] - 1e-7)
                centers[i, 1] = np.clip(centers[i, 1], radii[i] + 1e-7, 1 - radii[i] - 1e-7)

            # Grow radii
            radii += growth_rate
            radii = np.clip(radii, 0.005, 0.49)

        return centers, radii

    def gradient_refine(centers, radii, n):
        x0 = np.concatenate([centers.flatten(), radii])

        def penalized_obj(x):
            c = x[:2 * n].reshape(n, 2)
            r = x[2 * n:]
            val = -np.sum(r)

            # Boundary penalties
            for i in range(n):
                xi, yi = c[i]
                ri = r[i]
                if xi - ri < 0:
                    val += 2000 * (ri - xi) ** 2
                if xi + ri > 1:
                    val += 2000 * (xi + ri - 1) ** 2
                if yi - ri < 0:
                    val += 2000 * (ri - yi) ** 2
                if yi + ri > 1:
                    val += 2000 * (yi + ri - 1) ** 2

            # Overlap penalties
            for i in range(n):
                for j in range(i + 1, n):
                    d = np.linalg.norm(c[i] - c[j])
                    md = r[i] + r[j]
                    if d < md:
                        val += 2000 * (md - d) ** 2

            return val

        result = minimize(
            penalized_obj, x0, method="L-BFGS-B",
            bounds=[(0.0001, 0.9999)] * 2 * n + [(0.0001, 0.499)] * n,
            options={"maxiter": 3000, "ftol": 1e-15, "gtol": 1e-12}
        )

        fc = result.x[:2 * n].reshape(n, 2)
        fr = result.x[2 * n:]

        # Final clamping
        for i in range(n):
            fc[i, 0] = np.clip(fc[i, 0], fr[i], 1 - fr[i])
            fc[i, 1] = np.clip(fc[i, 1], fr[i], 1 - fr[i])

        return fc, fr

    best_centers = np.zeros((n, 2))
    best_radii = np.zeros(n)
    best_sum = 0.0

    # Try multiple seeds with different initialization strategies
    for seed in range(30):
        np.random.seed(seed * 7919 + 17)

        centers = np.zeros((n, 2))
        radii = np.ones(n) * 0.04

        # Different initialization patterns based on seed
        init_type = seed % 4
        idx = 0

        if init_type == 0:
            # Hexagonal-ish grid
            for row in range(8):
                ncols = 5 if row % 2 == 0 else 4
                offset = 0.09 if row % 2 == 1 else 0.0
                for col in range(ncols):
                    if idx >= n:
                        break
                    x = 0.07 + col * 0.18 + offset
                    y = 0.07 + row * 0.13
                    if x < 1 and y < 1:
                        centers[idx] = [x, y]
                        idx += 1

        elif init_type == 1:
            # 5x5 grid with one extra
            for row in range(6):
                for col in range(5):
                    if idx >= n:
                        break
                    x = 0.1 + col * 0.18
                    y = 0.1 + row * 0.15
                    if x < 1 and y < 1:
                        centers[idx] = [x, y]
                        idx += 1

        elif init_type == 2:
            # Random but clustered
            for i in range(n):
                x = np.random.uniform(0.12, 0.88)
                y = np.random.uniform(0.12, 0.88)
                centers[i] = [x, y]
                radii[i] = 0.03 + 0.02 * np.random.random()

        elif init_type == 3:
            # Corner-biased: larger circles near corners
            corners = [(0.15, 0.15), (0.85, 0.15), (0.15, 0.85), (0.85, 0.85)]
            for i, (cx, cy) in enumerate(corners):
                centers[i] = [cx, cy]
                radii[i] = 0.18
            idx = 4
            # Fill rest in grid
            for row in range(7):
                for col in range(5):
                    if idx >= n:
                        break
                    x = 0.2 + col * 0.14
                    y = 0.2 + row * 0.12
                    if x < 1 and y < 1:
                        # Check not too close to corners
                        too_close = False
                        for k in range(4):
                            if np.linalg.norm(centers[k] - [x, y]) < 0.35:
                                too_close = True
                                break
                        if not too_close:
                            centers[idx] = [x, y]
                            radii[idx] = 0.06
                            idx += 1
            while idx < n:
                centers[idx] = [np.random.uniform(0.15, 0.85), np.random.uniform(0.15, 0.85)]
                radii[idx] = 0.04
                idx += 1

        # Fill any remaining spots
        while idx < n:
            x = np.random.uniform(0.1, 0.9)
            y = np.random.uniform(0.1, 0.9)
            centers[idx] = [x, y]
            radii[idx] = 0.03
            idx += 1

        # Phase 1: Quick growth to get initial jamming
        centers, radii = force_directed_optimize(centers, radii, n, 2000, 0.00004)

        # Phase 2: Slower, more careful growth
        centers, radii = force_directed_optimize(centers, radii, n, 3000, 0.00002)

        # Phase 3: Very slow growth for fine packing
        centers, radii = force_directed_optimize(centers, radii, n, 2000, 0.00001)

        # Gradient refinement
        centers, radii = gradient_refine(centers, radii, n)

        # Second round of force-directed with current radii
        centers, radii = force_directed_optimize(centers, radii, n, 1500, 0.000015)

        # Final gradient refinement
        centers, radii = gradient_refine(centers, radii, n)

        # Validate and track best
        if is_valid(centers, radii):
            current_sum = np.sum(radii)
            if current_sum > best_sum:
                best_sum = current_sum
                best_centers = centers.copy()
                best_radii = radii.copy()

    # Final cleanup: ensure all values are valid
    for i in range(n):
        best_centers[i, 0] = np.clip(best_centers[i, 0], best_radii[i], 1 - best_radii[i])
        best_centers[i, 1] = np.clip(best_centers[i, 1], best_radii[i], 1 - best_radii[i])

    return best_centers, best_radii, best_sum
