def solve_radii(centers):
    """
    Solves for the optimal radii given fixed centers using Linear Programming.
    Maximizes sum(radii) subject to boundary and non-overlap constraints.
    """
    n = len(centers)
    if n == 0:
        return np.array([])

    # Objective: minimize -sum(radii)
    c = -np.ones(n)

    # Constraints: A_ub @ r <= b_ub
    A_ub = []
    b_ub = []

    # Precompute distances to avoid redundant calculations in loops if needed,
    # but for N=26, simple loop is fast enough.
    
    # Boundary constraints: r_i <= dist_to_boundary
    # dist_to_boundary = min(x, 1-x, y, 1-y)
    # This translates to 4 inequalities per circle:
    # r_i <= x_i
    # r_i <= 1 - x_i
    # r_i <= y_i
    # r_i <= 1 - y_i
    
    # We can aggregate these into single constraints r_i <= min_dist, 
    # but linprog handles multiple rows. Let's add them.
    # Actually, just one constraint r_i <= min(x_i, 1-x_i, y_i, 1-y_i) is sufficient.
    # Let's compute min_dist for each circle.
    min_dists = []
    for i in range(n):
        x, y = centers[i]
        d = min(x, 1 - x, y, 1 - y)
        min_dists.append(d)
    
    for i in range(n):
        row = np.zeros(n)
        row[i] = 1
        A_ub.append(row)
        b_ub.append(min_dists[i])

    # Non-overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = math.sqrt((centers[i][0] - centers[j][0])**2 +

            row = np.zeros(n)
            row[i] = 1
            row[j] = 1
            A_ub.append(row)
            b_ub.append(dist)

    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)

    # Bounds: r_i >= 0
    bounds = [(0, None) for _ in range(n)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
