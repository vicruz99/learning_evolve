import numpy as np
from scipy.optimize import minimize
import math

def compute_constraints(vars, n):
    constraints = []
    # Boundary constraints: x >= r, x <= 1-r, y >= r, y <= 1-r
    for i in range(n):
        idx = i * 3
        xi, yi, ri = vars[idx], vars[idx+1], vars[idx+2]
        constraints.extend([
            xi - ri,
            1.0 - xi - ri,
            yi - ri,
            1.0 - yi - ri
        ])
    # Non-overlap constraints: dist^2 >= (r_i + r_j)^2
    for i in range(n):
        idx_i = i * 3
        xi, yi, ri = vars[idx_i], vars[idx_i+1], vars[idx_i+2]
        for j in range(i + 1, n):
            idx_j = j * 3
            xj, yj, rj = vars[idx_j], vars[idx_j+1], vars[idx_j+2]
            dist_sq = (xi - xj)**2 + (yi - yj)**2
            sum_r_sq = (ri + rj)**2
            constraints.append(dist_sq - sum_r_sq)
    return np.array(constraints)

def objective(vars, n):
    # Maximize sum of radii -> minimize negative sum
    return -np.sum(vars[2::3])

def constraint_fun_26(vars):
    return compute_constraints(vars, 26)

def make_initial_guess(n):
    centers = []
    radii = []
    r = 0.08
    row_idx = 0
    # Generate a hexagonal lattice pattern
    while len(centers) < n:
        y = r + row_idx * r * math.sqrt(3)
        if y + r > 0.95:
            break
        shift = r if row_idx % 2 == 1 else 0.0
        k = 0
        while True:
            x = r + shift + k * (2 * r)
            if x + r > 0.95:
                break
            if len(centers) < n:
                centers.append([x, y])
                radii.append(r)
            k += 1
        row_idx += 1
    # Fallback fill if pattern yields fewer than n
    while len(centers) < n:
        centers.append([0.5, 0.5])
        radii.append(r)
        
    x0 = np.zeros(3 * n)
    for i in range(n):
        x0[3*i] = centers[i][0]
        x0[3*i+1] = centers[i][1]
        x0[3*i+2] = radii[i]
    return x0

def run_packing():
    n = 26
    x0 = make_initial_guess(n)
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
    
    cons = {'type': 'ineq', 'fun': constraint_fun_26}
    
    res = minimize(objective, x0, args=(n,), method='SLSQP', 
                   bounds=bounds, constraints=cons, 
                   options={'maxiter': 400, 'ftol': 1e-10})
    
    vars_opt = res.x
    
    centers = np.zeros((n, 2))
    centers[:, 0] = vars_opt[0::3]
    centers[:, 1] = vars_opt[1::3]
    radii = vars_opt[2::3]
    
    # Ensure radii are non-negative
    radii = np.maximum(radii, 0.0)
    sum_r = np.sum(radii)
    
    return centers, radii, sum_r
