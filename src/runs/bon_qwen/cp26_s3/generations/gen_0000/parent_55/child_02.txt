import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def objective(z):
    """Objective function: maximize sum of radii (minimize negative sum)"""
    return -np.sum(z[2::3])

def constraints(z):
    """Inequality constraints: g(z) >= 0"""
    x = z[0::3]
    y = z[1::3]
    r = z[2::3]
    cons = []
    
    # Boundary constraints: circles inside [0,1]x[0,1]
    cons.extend(x - r)
    cons.extend(1.0 - x - r)
    cons.extend(y - r)
    cons.extend(1.0 - y - r)
    
    # Non-overlap constraints: distance^2 >= (r_i + r_j)^2
    for i in range(N_CIRCLES):
        dx = x[i+1:] - x[i]
        dy = y[i+1:] - y[i]
        dr = r[i+1:] + r[i]
        d2 = dx**2 + dy**2
        cons.extend(d2 - dr**2)
        
    return np.array(cons)

def run_packing():
    trials = []
    
    # 1. Random initializations
    np.random.seed(42)
    for _ in range(4):
        x = np.random.uniform(0.2, 0.8, N_CIRCLES)
        y = np.random.uniform(0.2, 0.8, N_CIRCLES)
        r = np.full(N_CIRCLES, 0.06)
        z = np.empty(3 * N_CIRCLES)
        z[0::3] = x
        z[1::3] = y
        z[2::3] = r
        trials.append(z)
        
    # 2. Hexagonal lattice initialization (efficient packing structure)
    row_cols = [6, 5, 6, 5, 4]  # Sums to 26
    cx, cy = [], []
    dy = 0.8 / 4
    dx = 0.8 / 5
    for i, nc in enumerate(row_cols):
        y = 0.1 + i * dy
        offset = (0.5 if i % 2 == 1 else 0) * dx / 2
        start = 0.1 + offset
        end = 0.9 - offset
        xs = np.linspace(start, end, nc)
        cx.extend(xs)
        cy.extend([y] * nc)
        
    z_hex = np.empty(3 * N_CIRCLES)
    z_hex[0::3] = cx
    z_hex[1::3] = cy
    z_hex[2::3] = 0.08
    trials.append(z_hex)
    
    # Variable bounds: x,y in [0,1], r in [0, 0.4]
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.4)] * N_CIRCLES
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -1.0
    best_z = None
    
    # Optimize from each trial
    for z0 in trials:
        res = minimize(objective, z0, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 3000, 'ftol': 1e-10, 'disp': False})
        cur_sum = -res.fun
        if cur_sum > best_sum:
            best_sum = cur_sum
            best_z = res.x
            
    # Extract results
    centers = np.empty((N_CIRCLES, 2))
    centers[:, 0] = best_z[0::3]
    centers[:, 1] = best_z[1::3]
    radii = best_z[2::3]
    
    return centers, radii, best_sum
