import numpy as np
from scipy.optimize import minimize
import warnings

warnings.filterwarnings('ignore')

N_CIRCLES = 26

def get_initial_vars():
    """Generate initial configuration on a hexagonal lattice."""
    centers = []
    s = 0.2
    dy = s * np.sqrt(3) / 2
    row = 0
    while len(centers) < N_CIRCLES:
        y = row * dy
        if y > 0.95:
            break
        col = 0
        while True:
            x = col * s + (row % 2) * (s / 2)
            if x > 0.95:
                break
            centers.append([x, y])
            col += 1
        row += 1

    centers = np.array(centers[:N_CIRCLES])
    centers += np.random.uniform(-0.008, 0.008, size=(N_CIRCLES, 2))
    centers = np.clip(centers, 0.12, 0.88)
    radii = np.ones(N_CIRCLES) * 0.05

    vars = np.zeros(N_CIRCLES * 3)
    for i in range(N_CIRCLES):
        vars[3*i] = centers[i, 0]
        vars[3*i+1] = centers[i, 1]
        vars[3*i+2] = radii[i]
    return vars

def objective_func(v):
    """Return negative sum of radii for minimization."""
    return -np.sum(v[2::3])

def constraint_func(v):
    """Compute all inequality constraints: boundary and non-overlap."""
    c = []
    for i in range(N_CIRCLES):
        x = v[3*i]
        y = v[3*i+1]
        r = v[3*i+2]
        c.extend([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
        
    for i in range(N_CIRCLES):
        xi, yi, ri = v[3*i], v[3*i+1], v[3*i+2]
        for j in range(i + 1, N_CIRCLES):
            xj, yj, rj = v[3*j], v[3*j+1], v[3*j+2]
            dx = xi - xj
            dy = yi - yj
            dist = np.sqrt(dx*dx + dy*dy)
            c.append(dist - ri - rj)
    return np.array(c)

def run_single_optimization(v0):
    """Run SLSQP optimization from a given starting point."""
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.25)] * N_CIRCLES
    cons = {'type': 'ineq', 'fun': constraint_func}
    res = minimize(objective_func, v0, method='SLSQP', bounds=bounds, 
                   constraints=cons, options={'ftol': 1e-9, 'maxiter': 800, 'disp': False})
    return res.x

def extract_config(v):
    """Extract centers and radii from flattened variable array."""
    centers = np.zeros((N_CIRCLES, 2))
    radii = np.zeros(N_CIRCLES)
    for i in range(N_CIRCLES):
        centers[i, 0] = v[3*i]
        centers[i, 1] = v[3*i+1]
        radii[i] = v[3*i+2]
    return centers, radii

def is_valid(centers, radii):
    """Check validity against constraints with tight tolerance."""
    for i in range(N_CIRCLES):
        x, y, r = centers[i, 0], centers[i, 1], radii[i]
        if x - r < -1e-10 or x + r > 1.0 + 1e-10 or y - r < -1e-10 or y + r > 1.0 + 1e-10:
            return False
    for i in range(N_CIRCLES):
        for j in range(i + 1, N_CIRCLES):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-10:
                return False
    return True

def repair_and_return(v):
    """Ensure strict validity by slightly scaling down radii if needed."""
    centers, radii = extract_config(v)
    if not is_valid(centers, radii):
        scale = 0.99
        for _ in range(10):
            radii *= scale
            centers, _ = extract_config(v)
            centers[:, 0] = v[0::3]
            centers[:, 1] = v[1::3]
            if is_valid(centers, radii):
                break
            scale *= 0.98
        v[2::3] = radii
    centers, radii = extract_config(v)
    return centers, radii, np.sum(radii)

def run_packing():
    """Main function to find and return the optimal packing."""
    np.random.seed(42)
    best_v = None
    best_sum = -1.0

    # Run multiple restarts to escape local minima
    for _ in range(4):
        v0 = get_initial_vars()
        v_opt = run_single_optimization(v0)
        centers, radii = extract_config(v_opt)
        if is_valid(centers, radii):
            s = np.sum(radii)
            if s > best_sum:
                best_sum = s
                best_v = v_opt.copy()

    if best_v is None:
        v0 = get_initial_vars()
        v0[2::3] = np.ones(N_CIRCLES) * 0.04
        best_v = v0

    return repair_and_return(best_v)
