import numpy as np
from scipy.optimize import minimize

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square

    Args:
        centers: np.array of shape (n, 2) with (x, y) coordinates
        radii: np.array of shape (n) with radius of each circle

    Returns:
        True if valid, False otherwise
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True

def generate_hexagonal_init(n):
    """Generate an initial hexagonal packing configuration for n circles."""
    centers = []
    # Try to fit circles in rows with alternating shifts
    # Target pattern to reach 26: 6, 5, 6, 5, 4 (Sum=26)
    # We will adjust radii to fit, but here we just place centers on a lattice
    # Assume a tentative radius r = 0.1 for spacing
    r_tentative = 0.1
    row_counts = [6, 5, 6, 5, 4]
    
    y_curr = r_tentative
    row_idx = 0
    count = 0

        # We center this in [0, 1].

        # Actually, first center at r + (1 - 2*n*r)/2 ?
        # Let's just place them centered.
        # Center of row at x=0.5.
        # Extent of row: 0.5 +/- (n_in_row-1)*r.

        for k in range(n_in_row):
            if count >= n: break
            x = start_x + k * 2 * r_tentative
            centers.append([x, y_curr])
            count += 1
        
        y_curr += r_tentative * np.sqrt(3)

    return np.array(centers[:n])

def objective(x_vars, n):
    """Objective function: maximize sum of radii (minimize negative sum)."""
    radii = x_vars[2*n:]
    return -np.sum(radii)

def constraints_builder(n):
    """Build constraints for the optimizer."""
    cons = []
    
    # Boundary constraints: x >= r, x <= 1-r, y >= r, y <= 1-r
    # x - r >= 0
    def bound_x_lower(x_vars):
        centers = x_vars[:2*n].reshape(n, 2)
        radii = x_vars[2*n:]
        return centers[:, 0] - radii
    
    def bound_x_upper(x_vars):
        centers = x_vars[:2*n].reshape(n, 2)
        radii = x_vars[2*n:]
        return (1 - centers[:, 0]) - radii
        
    def bound_y_lower(x_vars):
        centers = x_vars[:2*n].reshape(n, 2)
        radii = x_vars[2*n:]
        return centers[:, 1] - radii
        
    def bound_y_upper(x_vars):
        centers = x_vars[:2*n].reshape(n, 2)
        radii = x_vars[2*n:]
        return (1 - centers[:, 1]) - radii

    cons.append({'type': 'ineq', 'fun': bound_x_lower})
    cons.append({'type': 'ineq', 'fun': bound_x_upper})
    cons.append({'type': 'ineq', 'fun': bound_y_lower})
    cons.append({'type': 'ineq', 'fun': bound_y_upper})
    
    # Non-overlap constraints: dist >= r_i + r_j
    # sqrt((xi-xj)^2 + (yi-yj)^2) - ri - rj >= 0
    # To avoid sqrt in derivative (optional, but smooth), we can use squared distance?
    # No, constraint is dist >= sum_r. Squaring is valid if dist, sum_r >= 0.
    # (dist)^2 >= (sum_r)^2.
    # (xi-xj)^2 + (yi-yj)^2 - (ri+rj)^2 >= 0.
    
    def overlap_constraint(x_vars, i, j):
        ci = x_vars[2*i]
        cj = x_vars[2*j]
        di = x_vars[2*i+1]
        dj = x_vars[2*j+1]
        ri = x_vars[2*n + i]
        rj = x_vars[2*n + j]
        
        dist_sq = (ci - cj)**2 + (di - dj)**2
        sum_r = ri + rj
        return dist_sq - sum_r**2

    for i in range(n):
        for j in range(i + 1, n):
            # Create a closure or lambda? 
            # Instructions say "no closures from function nesting" and "no lambda".
            # So we define a specific function for each pair or pass i,j.
            # But scipy expects a callable. 
            # We can add a list of constraint dicts.
            cons.append({'type': 'ineq', 'fun': lambda x_vars, i=i, j=j: overlap_constraint(x_vars, i, j)})

    # Non-negative radii
    # r >= 0. 
    # Handled by bounds in optimize, but adding constraint for safety?
    # SLSQP handles bounds well.
    
    return cons

def run_packing():
    n = 26
    
    # 1. Initialization
    # Start with a hexagonal-ish pattern
    centers_init = generate_hexagonal_init(n)
    
    # Estimate initial radius. 
    # For 26 circles, r approx 0.1.
    r_init = 0.09 * np.ones(n)
    
    # Combine into variables vector: [x1, y1, ..., x26, y26, r1, ..., r26]
    x0 = np.concatenate([centers_init.flatten(), r_init])
    
    # Bounds: x, y in [0, 1], r in [0, 0.5]
    bounds = []
    for _ in range(2 * n):
        bounds.append((0.0, 1.0)) # x and y
    for _ in range(n):
        bounds.append((1e-6, 0.5)) # r must be positive
    
    # Constraints
    constraints = constraints_builder(n)
    
    # 2. Optimization
    # Run optimization
    res = minimize(
        objective, 
        x0, 
        args=(n,), 
        method='SLSQP', 
        bounds=bounds, 
        constraints=constraints,
        options={'maxiter': 1000, 'ftol': 1e-12, 'disp': False}
    )
    
    # Extract results
    centers_opt = res.x[:2*n].reshape(n, 2)
    radii_opt = res.x[2*n:]
    
    # Check if valid, if not, try to repair or use previous
    # But SLSQP should satisfy constraints.
    
    sum_radii = np.sum(radii_opt)
    
    # Just in case, ensure radii are positive
    radii_opt = np.maximum(radii_opt, 1e-9)
    
    # Recalculate sum
    sum_radii = np.sum(radii_opt)
    
    return centers_opt, radii_opt, sum_radii

# Run the packing
centers, radii, sum_radii = run_packing()

# Validation
is_valid = validate_packing(centers, radii)
