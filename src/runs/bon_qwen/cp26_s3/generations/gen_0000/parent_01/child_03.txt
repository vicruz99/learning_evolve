import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimize the placement and radii of 26 circles in a unit square.
    
    Strategy:
    1. Initialize circles in a hexagonal-like grid (rows of 6, 5, 6, 5, 4).
    2. Use scipy.optimize to maximize the sum of radii.
    3. Use a penalty-based objective function to handle overlap and boundary constraints.
    """
    n = 26
    np.random.seed(42) # For reproducibility of initial jitter

    # 1. Initial Configuration (Hybrid Hexagonal Grid)
    # Rows: 6, 5, 6, 5, 4
    rows = [6, 5, 6, 5, 4]
    init_centers = []
    init_radii = []
    
    y = 0.1
    for r_count in rows:
        # Distribute centers in x
        x_coords = np.linspace(0.1, 0.9, r_count)
        # Shift odd-indexed rows for hexagonal effect (rows 1, 3 are shifted)
        # rows indices: 0(6), 1(5), 2(6), 3(5), 4(4)
        # Shift rows with 5 circles (indices 1, 3)
        if r_count == 5:
            shift = 0.1
            x_coords = x_coords + shift
            # Clamp to ensure we stay roughly inside, optimization will fix it
            x_coords = np.clip(x_coords, 0.1, 0.9)
        
        for x in x_coords:
            init_centers.append([x, y])
            init_radii.append(0.09) # Initial radius guess
        
        # Increment y based on hexagonal packing density
        y += 0.22 

    centers = np.array(init_centers)
    radii = np.array(init_radii)

    # 2. Define Objective Function (Maximize Sum of Radii with Penalties)
    # We minimize the negative sum of radii + penalty terms.
    def objective(params):
        # Reshape parameters
        # Params: [x1, y1, r1, x2, y2, r2, ...]
        # Total params: 3 * n
        c = params[0::3].reshape(-1, 2)
        r = params[1::3]
        
        # Hard floor for radii to prevent collapse
        r = np.maximum(r, 1e-4)
        
        sum_r = np.sum(r)
        penalty = 0.0
        
        # Boundary Penalty: Center must be in [r, 1-r]
        # If x < r, penalty. If x > 1-r, penalty.
        # Using a quadratic penalty for smoothness
        for i in range(n):
            x, y_val = c[i]
            ri = r[i]
            
            # Left/Right
            if x < ri:
                penalty += 1000 * (x - ri) ** 2
            elif x > 1.0 - ri:
                penalty += 1000 * (x - (1.0 - ri)) ** 2
            
            # Top/Bottom
            if y_val < ri:
                penalty += 1000 * (y_val - ri) ** 2
            elif y_val > 1.0 - ri:
                penalty += 1000 * (y_val - (1.0 - ri)) ** 2

        # Overlap Penalty
        # For all pairs, dist >= r_i + r_j
        # dist^2 >= (r_i + r_j)^2
        # Penalty if dist < r_i + r_j
        # We check pairs. 26*25/2 = 325 pairs.
        # To be efficient, we can vectorize or just loop for small N.
        
        # Vectorized distance calculation
        # c is (n, 2)
        # diff = c[:, np.newaxis, :] - c[np.newaxis, :, :] # (n, n, 2)
        # dist_sq = np.sum(diff**2, axis=2) # (n, n)
        
        # We only need upper triangle
        dist_mat = np.sqrt(np.sum((c[:, np.newaxis, :] - c[np.newaxis, :, :])**2, axis=2))
        sum_r_mat = r[:, np.newaxis] + r[np.newaxis, :]
        
        overlap = sum_r_mat - dist_mat
        # Only care about positive overlaps
        overlap = np.maximum(0, overlap - 1e-8) # small tolerance
        
        # Sum of squared overlaps, heavily penalized
        penalty += 5000 * np.sum(overlap**2)
        
        # Return negative sum_r so minimization works
        return -sum_r + penalty

    # 3. Setup Optimization
    # Flatten parameters: [x1, r1, y1, x2, r2, y2, ...]
    # This order helps gradient descent sometimes, but any order works.
    init_params = np.zeros(3 * n)
    for i in range(n):
        init_params[3*i] = centers[i, 0]
        init_params[3*i+1] = radii[i]
        init_params[3*i+2] = centers[i, 1]

    # Bounds: x, y in [0, 1], r in [0, 0.5]
    bounds = [(0.0, 1.0) for _ in range(3*n)]
    # Tighten radius bounds slightly for stability
    for i in range(0, 3*n, 3):
        bounds[i+1] = (1e-4, 0.5)

    # 4. Run Optimization
    # Use L-BFGS-B for bound-constrained optimization
    res = minimize(objective, init_params, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': 2000, 'ftol': 1e-9})

    # 5. Extract and Validate Solution
    final_params = res.x
    final_centers = np.zeros((n, 2))
    final_radii = np.zeros(n)

    for i in range(n):
        final_centers[i, 0] = final_params[3*i]
        final_radii[i] = final_params[3*i+1]
        final_centers[i, 1] = final_params[3*i+2]

    # Post-processing: Clamp radii to positive and ensure boundaries
    final_radii = np.maximum(final_radii, 1e-5)
    
    # Final safety check and adjustment (simple projection)
    for i in range(n):
        r = final_radii[i]
        x, y = final_centers[i]
        # Clamp center
        x = max(r, min(1.0 - r, x))
        y = max(r, min(1.0 - r, y))
        final_centers[i] = [x, y]

    sum_radii = np.sum(final_radii)
    
    return final_centers, final_radii, sum_radii
