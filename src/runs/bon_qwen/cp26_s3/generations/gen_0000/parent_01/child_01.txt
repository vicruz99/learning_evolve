import numpy as np
from scipy.optimize import minimize

def objective_func(theta, lam, N):
    """Computes the penalized objective function for circle packing."""
    centers = theta[:2*N].reshape(N, 2)
    radii = theta[2*N:]
    
    # Objective: maximize sum of radii (minimize negative sum)
    obj = -np.sum(radii)
    pen = 0.0
    
    # Inter-circle overlap penalty
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2) + 1e-12)
    sum_r = radii[:, None] + radii[None, :]
    
    idx = np.triu_indices(N, k=1)
    overlaps = np.maximum(0, sum_r[idx] - dists[idx])
    pen += np.sum(overlaps**2)
    
    # Boundary constraint penalty
    pen += np.sum(np.maximum(0, radii - centers[:, 0])**2)
    pen += np.sum(np.maximum(0, radii - (1.0 - centers[:, 0]))**2)
    pen += np.sum(np.maximum(0, radii - centers[:, 1])**2)
    pen += np.sum(np.maximum(0, radii - (1.0 - centers[:, 1]))**2)
    
    return obj + lam * pen

def solve_instance(N, theta0, bounds):
    """Runs the annealing optimization for a single initialization."""
    current_theta = theta0.copy()
    # Gradually increase penalty to enforce constraints
    lams = np.logspace(-2, 4, 25)
    
    for lam in lams:
        res = minimize(objective_func, current_theta, args=(lam, N), 
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': 500, 'ftol': 1e-10})
        current_theta = res.x
    return current_theta

def run_packing():
    """Packs 26 circles in a unit square to maximize sum of radii."""
    N = 26
    bounds = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N
    
    best_sum_r = 0.0
    best_centers = np.zeros((N, 2))
    best_radii = np.zeros(N)
    
    # Multiple restarts to avoid poor local minima
    for restart in range(4):
        np.random.seed(restart * 137 + 7)
        
        # Initialize on a perturbed 5x5 grid + 1 center circle
        base = np.linspace(0.15, 0.85, 5)
        cx, cy = np.meshgrid(base, base)
        centers_init = np.column_stack([cx.ravel(), cy.ravel()])
        
        cx2 = 0.5 + np.random.normal(0, 0.08)
        cy2 = 0.5 + np.random.normal(0, 0.08)
        centers_init = np.vstack([centers_init, [cx2, cy2]])
        
        radii_init = np.full(N, 0.06)
        theta0 = np.hstack([centers_init.flatten(), radii_init])
        
        # Optimize
        theta_opt = solve_instance(N, theta0, bounds)
        
        curr_centers = theta_opt[:2*N].reshape(N, 2)
        curr_radii = theta_opt[2*N:]
        
        # Strict projection to guarantee boundary feasibility
        r_lim = np.minimum(curr_centers[:, 0], 1.0 - curr_centers[:, 0])
        r_lim = np.minimum(r_lim, np.minimum(curr_centers[:, 1], 1.0 - curr_centers[:, 1]))
        curr_radii = np.minimum(curr_radii, r_lim)
        
        # Iteratively shrink radii to guarantee non-overlap
        for _ in range(100):
            changed = False
            for i in range(N):
                for j in range(i + 1, N):
                    d = np.sqrt(np.sum((curr_centers[i] - curr_centers[j])**2))
                    if d < curr_radii[i] + curr_radii[j]:
                        shrink = (curr_radii[i] + curr_radii[j] - d) / 2.0 + 1e-6
                        curr_radii[i] -= shrink
                        curr_radii[j] -= shrink
                        curr_radii[i] = max(0.0, curr_radii[i])
                        curr_radii[j] = max(0.0, curr_radii[j])
                        changed = True
            if not changed:
                break
                
        curr_sum = np.sum(curr_radii)
        if curr_sum > best_sum_r:
            best_sum_r = curr_sum
            best_centers = curr_centers.copy()
            best_radii = curr_radii.copy()
            
    return best_centers, best_radii, best_sum_r
