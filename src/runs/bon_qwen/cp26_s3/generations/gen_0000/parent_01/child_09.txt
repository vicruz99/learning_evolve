import numpy as np
import scipy.optimize as opt

def run_packing():
    """
    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    Uses a hexagonal lattice initialization and SLSQP optimization.
    """
    n_circles = 26
    
    # 1. Initialization: Hexagonal Lattice
    # We generate points in a hexagonal pattern and select the first n_circles.
    # This provides a dense initial configuration.
    initial_spacing = 0.2 # Diameter approx 0.2, radius 0.1
    # Vertical step for hex packing
    dy = initial_spacing * np.sqrt(3) / 2
    
    points = []
    y = initial_spacing / 2 # Start with offset to fit better in boundaries
    row_idx = 0
    
    # Generate enough points
    while len(points) < n_circles:
        x_offset = 0
        if row_idx % 2 == 1:
            x_offset = initial_spacing / 2
        
        x = initial_spacing / 2 + x_offset
        while x < 1.0:
            points.append([x, y])
            x += initial_spacing
            if len(points) >= n_circles:
                break
        y += dy
        row_idx += 1
    
    # Take exactly n_circles
    initial_centers = np.array(points[:n_circles])
    initial_radii = np.full(n_circles, initial_spacing / 2 * 0.9) # Slightly smaller to be safe
    
    # 2. Define Objective and Constraints for Optimization
    # We want to maximize sum(radii), so we minimize -sum(radii)
    
    def objective(vars):
        # vars layout: x0, y0, r0, x1, y1, r1, ...
        radii = vars[2::3]
        return -np.sum(radii)

    def boundary_constraints(vars):
        constraints = []
        for i in range(n_circles):
            x = vars[3*i]
            y = vars[3*i+1]
            r = vars[3*i+2]
            # x - r >= 0  => x - r >= 0
            constraints.append(x - r)
            # 1 - x - r >= 0
            constraints.append(1.0 - x - r)
            # y - r >= 0
            constraints.append(y - r)
            # 1 - y - r >= 0
            constraints.append(1.0 - y - r)
        return constraints

    def non_overlap_constraints(vars):
        constraints = []
        centers = np.array([
            [vars[3*i], vars[3*i+1]] for i in range(n_circles)
        ])
        radii = vars[2::3]
        
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                # dist >= r_i + r_j
                # dist = sqrt((xi-xj)^2 + (yi-yj)^2)
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                constraints.append(dist - radii[i] - radii[j])
        return constraints

    # 3. Setup Optimization
    # Initial guess vector
    x0 = np.zeros(3 * n_circles)
    for i in range(n_circles):
        x0[3*i] = initial_centers[i, 0]
        x0[3*i+1] = initial_centers[i, 1]
        x0[3*i+2] = initial_radii[i]
    
    # Bounds: x, y in [0, 1], r in [0, 1] (loose upper bound)
    bounds = []
    for _ in range(n_circles):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 0.5)) # r (cannot exceed 0.5)
    
    # Define constraints dict for scipy
    cons = []
    
    # Boundary constraints
    # We can define a function that returns an array
    def bound_ineq(vars):
        res = []
        for i in range(n_circles):
            x = vars[3*i]
            y = vars[3*i+1]
            r = vars[3*i+2]
            res.extend([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
        return np.array(res)

    cons.append({'type': 'ineq', 'fun': bound_ineq})
    
    # Overlap constraints
    def overlap_ineq(vars):
        res = []
        centers = np.zeros((n_circles, 2))
        radii = np.zeros(n_circles)
        for i in range(n_circles):
            centers[i, 0] = vars[3*i]
            centers[i, 1] = vars[3*i+1]
            radii[i] = vars[3*i+2]
        
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                dist = np.linalg.norm(centers[i] - centers[j])
                res.append(dist - radii[i] - radii[j])
        return np.array(res)

    cons.append({'type': 'ineq', 'fun': overlap_ineq})

    # 4. Run Optimization
    # SLSQP is a good choice for constrained nonlinear optimization
    try:
        res = opt.minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, 
                           options={'maxiter': 1000, 'ftol': 1e-9})
        optimized_vars = res.x
    except Exception as e:
        print(f"Optimization failed: {e}")
        # Fallback to initial configuration
        optimized_vars = x0

    # 5. Extract Results
    centers = np.array([
        [optimized_vars[3*i], optimized_vars[3*i+1]] for i in range(n_circles)
    ])
    radii = optimized_vars[2::3]
    
    # Ensure radii are non-negative (clamp)
    radii = np.maximum(radii, 0.0)
    
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
