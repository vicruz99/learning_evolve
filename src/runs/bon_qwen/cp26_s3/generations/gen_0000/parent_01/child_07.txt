import numpy as np
from scipy.optimize import minimize

def run_packing():
    n = 26
    
    # 1. Generate a feasible initial hexagonal layout
    pos = []
    r_init = 0.06  # Start with a safe radius
    for i in range(7):
        for j in range(5):
            if len(pos) >= n:
                break
            # Hexagonal spacing
            x = r_init + j * 2 * r_init + (i % 2) * r_init
            y = r_init + i * r_init * np.sqrt(3)
            pos.append([x, y])
            
    centers_init = np.array(pos[:n])
    
    # Normalize initial positions to sit comfortably inside [0.1, 0.9]
    min_c = centers_init.min(axis=0)
    max_c = centers_init.max(axis=0)
    centers_init = (centers_init - min_c) / (max_c - min_c) * 0.8 + 0.1
    
    radii_init = np.full(n, r_init)
    
    # Flatten variables: [x1, y1, r1, x2, y2, r2, ...]
    x0 = np.zeros(3 * n)
    for i in range(n):
        x0[3*i] = centers_init[i, 0]
        x0[3*i+1] = centers_init[i, 1]
        x0[3*i+2] = radii_init[i]
        
    # Bounds for variables
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
    
    # Objective: maximize sum of radii -> minimize negative sum
    def objective(v):
        return -np.sum(v[2::3])
        
    # Constraints: boundary + non-overlap
    def constraints(v):
        x = v[0::3]
        y = v[1::3]
        r = v[2::3]
        
        # Boundary constraints: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
        c_bound = np.concatenate([
            x - r,
            1.0 - x - r,
            y - r,
            1.0 - y - r
        ])
        
        # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
        dx = x[:, None] - x[None, :]
        dy = y[:, None] - y[None, :]
        dist_sq = dx**2 + dy**2
        r_sum = r[:, None] + r[None, :]
        
        # Only need upper triangle to avoid duplicate constraints
        idx = np.triu_indices(n, k=1)
        c_pair = dist_sq[idx] - r_sum[idx]**2
        
        return np.concatenate([c_bound, c_pair])
        
    cons = {'type': 'ineq', 'fun': constraints}
    
    # Optimization
    res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'tol': 1e-9, 'maxiter': 5000, 'disp': False})
                   
    # Extract results
    final_centers = res.x.reshape(n, 3)[:, :2]
    final_radii = res.x.reshape(n, 3)[:, 2]
    
    # Ensure radii are strictly positive (optimizer might push to 0 if stuck)
    final_radii = np.maximum(final_radii, 1e-9)
    
    total_radius = np.sum(final_radii)
    return final_centers, final_radii, total_radius
