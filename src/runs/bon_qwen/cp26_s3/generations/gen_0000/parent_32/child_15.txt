def validate_packing(centers, radii):

    Validate that circles don't overlap and are inside the unit square

    Objective function for scipy optimizer.
    Params: flattened array of [x1, y1, r1, x2, y2, r2, ...]
    Returns: -sum(radii) + penalty for violations

    radii = params[2*n:]
    
    penalty = 0.0
    
    # Boundary constraints
    for i in range(n):

        r = radii[i]
        # Soft constraint: if r > x, penalty
        if r > x: penalty += 1000 * (r - x)**2
        if r > 1 - x: penalty += 1000 * (r - (1 - x))**2
        if r > y: penalty += 1000 * (r - y)**2
        if r > 1 - y: penalty += 1000 * (r - (1 - y))**2
        
    # Overlap constraints

        for j in range(i + 1, n):

            if dist < min_dist:
                # Quadratic penalty for overlap
                overlap = min_dist - dist
                penalty += 1000 * overlap**2
                
    return -np.sum(radii) + penalty
