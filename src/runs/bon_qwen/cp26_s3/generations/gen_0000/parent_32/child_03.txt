def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    maximizing the sum of radii using a force-directed optimization.
    
    Returns:
        centers: np.array of shape (26, 2) with (x, y) coordinates
        radii: np.array of shape (26,) with radius of each circle
        sum_radii: float, the sum of the radii
    """

    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.09 # Start slightly smaller than 0.1 to allow movement

    for i in range(5):
        for j in range(5):
            idx = i * 5 + j
            centers[idx] = [0.1 + 0.2 * j, 0.1 + 0.2 * i]
            
    # Place the 26th circle in the center
    centers[25] = [0.5, 0.5]
    
    # 2. Optimization Loop
    num_iterations = 10000
    learning_rate = 0.005
    damping = 0.9
    
    velocities = np.zeros((n, 2))
    
    for step in range(num_iterations):
        # Increase radii slowly to expand the packing
        radii_increase = 0.00005 * (1.0 - step/num_iterations)
        radii += radii_increase
        
        forces = np.zeros((n, 2))
        
        # Calculate forces
        for i in range(n):
            for j in range(i + 1, n):
                vec = centers[j] - centers[i]
                dist = np.linalg.norm(vec)
                
                # Non-overlap constraint: distance >= r_i + r_j
                required_dist = radii[i] + radii[j]
                
                if dist < required_dist and dist > 1e-6:
                    # Repulsive force proportional to overlap
                    overlap = required_dist - dist
                    force_mag = overlap * 10.0 # Spring constant
                    
                    # Normalize and apply force
                    force_dir = vec / dist
                    forces[i] -= force_dir * force_mag
                    forces[j] += force_dir * force_mag
            else:
                # Boundary constraints
                for k in range(2): # x and y
                    # Wall at 0

                        forces[i, k] -= (centers[i, k] + radii[i] - 1) * 20.0

        velocities = velocities * damping + forces * learning_rate

        np.clip(centers, 0, 1, out=centers)

    total_radius = np.sum(radii)
    
    return centers, radii, float(total_radius)
