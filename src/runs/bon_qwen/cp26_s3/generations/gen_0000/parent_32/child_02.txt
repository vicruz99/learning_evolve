

<strategy>
The goal is to maximize the sum of radii of 26 circles packed in a unit square. This is a non-convex optimization problem. However, for a fixed set of centers, finding the optimal radii is a Linear Programming (LP) problem. We can leverage this by decoupling the problem:
1. Define a function `max_sum_radii(centers)` that solves the LP for radii given fixed centers. The constraints are $r_i + r_j \le \|c_i - c_j\|$ and $r_i \le \text{dist}(c_i, \text{boundary})$.
2. Use a derivative-free global optimizer (like Differential Evolution or Basin Hopping) to find the optimal centers by maximizing the result of this LP function.
3. To ensure a good solution, we will use multiple starting points (random, grid-based, hexagonal) and a local refinement step.

This approach transforms the complex non-linear constraints into a sequence of linear problems and a smooth(ish) optimization landscape for the centers.

import numpy as np
from scipy.optimize import linprog, minimize, differential_evolution
from scipy.spatial.distance import cdist

def get_boundary_distances(centers):
    """
    Calculates the distance from each center to the nearest boundary of the unit square.
    Returns an array of shape (n,).
    """
    # Centers is (n, 2)
    # Distances to left (x=0), right (x=1), bottom (y=0), top (y=1)
    dists_left = centers[:, 0]
    dists_right = 1.0 - centers[:, 0]
    dists_bottom = centers[:, 1]
    dists_top = 1.0 - centers[:, 1]
    
    # Minimum distance to any boundary
    boundary_dists = np.minimum.reduce([dists_left, dists_right, dists_bottom, dists_top])
    return boundary_dists

def solve_radii_lp(centers):
    """
    Given fixed centers, solves the LP to maximize sum of radii.
    
    Constraints:
    1. r_i + r_j <= distance(centers[i], centers[j]) for all i < j
    2. r_i <= distance(centers[i], boundary)
    3. r_i >= 0
    
    Returns:
        optimal_radii: np.array of shape (n,)
        max_sum: float
    """
    n = centers.shape[0]
    if n == 0:
        return np.array([]), 0.0

    # Calculate distances between all pairs
    dist_matrix = cdist(centers, centers, metric='euclidean')
    
    # Calculate boundary constraints
    b_constraints = get_boundary_distances(centers)

    # Objective: maximize sum(r_i) => minimize -sum(r_i)
    c = -np.ones(n)

    # Inequality constraints A_ub @ r <= b_ub
    # We need to construct the matrix for r_i + r_j <= d_ij
    # And r_i <= b_i
    
    # Number of constraints
    num_pair_constraints = n * (n - 1) // 2
    num_boundary_constraints = n
    total_constraints = num_pair_constraints + num_boundary_constraints
    
    A_ub = np.zeros((total_constraints, n))
    b_ub = np.zeros(total_constraints)
    
    constraint_idx = 0
    
    # Pairwise constraints
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[constraint_idx, i] = 1.0
            A_ub[constraint_idx, j] = 1.0
            b_ub[constraint_idx] = dist_matrix[i, j]
            constraint_idx += 1
            
    # Boundary constraints
    for i in range(n):
        A_ub[constraint_idx, i] = 1.0
        b_ub[constraint_idx] = b_constraints[i]
        constraint_idx += 1
        
    # Bounds for r_i: r_i >= 0
    bounds = [(0, None) for _ in range(n)]
    
    # Solve LP
    # Using 'highs' method is usually robust and fast
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
        else:
            # Fallback or error handling
            return np.zeros(n), 0.0
    except Exception:
        return np.zeros(n), 0.0

def objective_function(centers_flat):
    """
    Objective function for the center optimization.
    Takes a flat array of centers, reshapes, solves LP, returns negative sum of radii.
    """
    n = 26
    centers = centers_flat.reshape((n, 2))
    
    # Clip centers to [0, 1] just in case, though optimizer bounds should handle it
    centers = np.clip(centers, 1e-9, 1 - 1e-9)
    
    _, sum_radii = solve_radii_lp(centers)
    return -sum_radii

def run_packing():
    """
    Main function to find the optimal packing.
    """
    n = 26
    
    # Strategy: Use Differential Evolution to search for optimal centers.
    # Bounds for each coordinate are [0, 1].
    bounds = [(0.0, 1.0)] * (2 * n)
    
    # Differential Evolution parameters
    # popsize: population size. Larger is better for exploration but slower.
    # maxiter: maximum iterations.
    # tol: tolerance for convergence.
    
    # We run DE. It might take some time, so we keep parameters reasonable.
    # seed for reproducibility
    seed = 42
    
    print("Starting global optimization (Differential Evolution)...")
    try:
        result = differential_evolution(
            objective_function,
            bounds=bounds,
            popsize=30, # Increased population for better coverage
            maxiter=200, # Increased iterations
            tol=1e-6,
            mutation=(0.5, 1.5),
            recombination=0.7,
            seed=seed,
            updating='deferred',
            workers=1 # Single worker for safety in some environments
        )
        
        optimal_centers_flat = result.x
        max_sum = -result.fun
        
    except Exception as e:
        print(f"DE failed: {e}")
        # Fallback to grid
        optimal_centers_flat = create_grid_centers(n)
        centers = optimal_centers_flat.reshape((n, 2))
        radii, max_sum = solve_radii_lp(centers)
        return centers, radii, max_sum

    optimal_centers = optimal_centers_flat.reshape((n, 2))
    
    # Final refinement and solving for radii
    # Ensure centers are valid
    optimal_centers = np.clip(optimal_centers, 0, 1)
    
    radii, final_sum = solve_radii_lp(optimal_centers)
    
    # Double check validity with the provided validation logic (mental check)
    # The LP guarantees constraints are satisfied.
    
    print(f"Optimization finished. Sum of radii: {final_sum}")
    
    return optimal_centers, radii, final_sum

def create_grid_centers(n):
    """
    Helper to create a grid arrangement for fallback or initial guess.
    """
    # 5x5 grid is 25 circles. We have 26.
    # Let's try to fit 26 in a grid-like structure.
    # Maybe 6x5 = 30, remove 4? Or just perturb random?
    # Let's just return a random valid set if needed, but DE handles this.
    # Just a simple grid for 25 + 1 in center
    centers = []
    # 5x5 grid with spacing 0.2
    # r would be 0.1.
    for i in range(5):
        for j in range(5):
            centers.append([0.1 + i*0.2, 0.1 + j*0.2])
    # Add 26th circle in the middle of a gap?
    # Gap at (0.2, 0.2) is occupied.
    # Maybe (0.1, 0.1) is occupied.
    # Let's place 26th at (0.05, 0.05) - very small.
    centers.append([0.5, 0.5]) # Center
    # This is just a fallback, DE will likely find better.
    return np.array(centers)

if __name__ == "__main__":
    centers, radii, sum_r = run_packing()
    print(f"Sum of radii: {sum_r}")
