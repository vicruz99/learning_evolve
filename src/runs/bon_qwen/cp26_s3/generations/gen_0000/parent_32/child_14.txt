The objective is to maximize the sum of radii of 26 circles packed in a unit square. This is a non-linear optimization problem. 

**Strategy:**
1.  **Dual Approach:** We decouple the problem. For a fixed set of circle centers, finding the optimal radii is a Linear Programming (LP) problem. This allows us to treat the sum of radii as a function of the centers only.
2.  **Initialization:** We start with a structured hexagonal grid arrangement. This is geometrically denser than a square grid and provides a strong initial configuration for the optimizer.
3.  **Iterative Relaxation (Repulsive Forces):** We iteratively improve the centers. For a given configuration, we compute the maximal radii. We then identify "active" constraints (where circles touch each other or boundaries). We apply repulsive forces to move centers apart from neighbors and away from constraining walls. This simulates a "gas expansion" to find the optimal packing density.
4.  **Validation:** Finally, we ensure the resulting packing satisfies all geometric constraints and passes the validation function.

import numpy as np
from scipy.optimize import linprog

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Packs 26 circles in a unit square to maximize sum of radii.
    
    Returns:
        centers: np.array (26, 2)
        radii: np.array (26)
        sum_radii: float
    """
    N = 26
    
    # --- 1. Initialization ---
    # Use a hexagonal grid pattern for better density than a square grid.
    centers = np.zeros((N, 2))
    
    # Arrange 26 points in roughly 6 rows
    # Row distribution: 4, 5, 4, 5, 4, 4 = 26 points
    row_counts = [4, 5, 4, 5, 4, 4]
    row_spacing = 1.0 / 6.0
    idx = 0
    
    for r_idx, count in enumerate(row_counts):
        y_coord = 0.5 + r_idx * (row_spacing * 2 - row_spacing) * 0.5 # Simple distribution
        # Better y-distribution for hex packing:
        # Total height 1. 6 rows.
        y_coord = 0.125 + r_idx * (0.75 / 5.0)
        
        # Stagger odd rows (if we index 0-based, usually row 1, 3, 5 are shifted)
        # But here row_counts varies. Let's just center them.
        x_start = (1.0 - (count - 1) * 0.15) / 2.0
        x_spacing = 0.15 # Initial spacing
        
        # Shift alternating rows slightly for hex effect
        shift = 0.075 if r_idx % 2 == 1 else 0.0
        
        for c_idx in range(count):
            centers[idx, 0] = x_start + c_idx * x_spacing + shift
            centers[idx, 1] = y_coord
            idx += 1

    # Add small random noise to break symmetry
    centers += np.random.uniform(-0.01, 0.01, centers.shape)
    centers = np.clip(centers, 0, 1)

    # --- 2. Optimization Loop ---
    # We will iteratively solve for radii and move centers to relieve pressure
    
    for iteration in range(500):
        # --- 2a. Solve LP for radii given centers ---
        # Maximize sum(r_i) s.t. r_i + r_j <= dist(i,j), r_i <= boundaries
        # Linprog minimizes, so we minimize -sum(r_i)
        
        c_obj = -np.ones(N)
        
        # Constraints matrix A_ub * r <= b_ub
        # 1. Pairwise constraints: r_i + r_j <= dist
        # 2. Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
        
        A_ub = []
        b_ub = []
        
        # Pairwise
        for i in range(N):
            for j in range(i + 1, N):
                dist = np.linalg.norm(centers[i] - centers[j])
                row = np.zeros(N)
                row[i] = 1.0
                row[j] = 1.0
                A_ub.append(row)
                b_ub.append(dist)
        
        # Boundaries
        for i in range(N):
            row = np.zeros(N)
            row[i] = 1.0
            
            # x constraints
            A_ub.append(row); b_ub.append(centers[i, 0])
            A_ub.append(row); b_ub.append(1.0 - centers[i, 0])
            # y constraints
            A_ub.append(row); b_ub.append(centers[i, 1])
            A_ub.append(row); b_ub.append(1.0 - centers[i, 1])
            
        A_ub = np.array(A_ub)
        b_ub = np.array(b_ub)
        
        # Bounds for r_i >= 0
        bounds = [(0, None) for _ in range(N)]
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
            radii = res.x
        except:
            # Fallback if linprog fails
            radii = np.zeros(N)
            
        sum_radii = np.sum(radii)
        
        # --- 2b. Compute Forces and Move Centers ---
        forces = np.zeros((N, 2))
        tol = 1e-4
        
        # Pairwise repulsion (if touching)
        for i in range(N):
            for j in range(i + 1, N):
                diff = centers[i] - centers[j]
                dist = np.linalg.norm(diff)
                if dist < 1e-9: continue
                
                dir_ij = diff / dist
                # If constraint is active (tight), push apart
                if radii[i] + radii[j] > dist - tol:
                    # Force magnitude proportional to tightness or just constant
                    force_mag = 0.1 
                    forces[i] += force_mag * dir_ij
                    forces[j] -= force_mag * dir_ij

        # Boundary repulsion (if touching wall)
        for i in range(N):
            x, y = centers[i]
            r = radii[i]
            
            # Left wall (x=0): constraint r <= x. If r approx x, push right
            if r > x - tol:
                forces[i, 0] += 0.1
            # Right wall (x=1): constraint r <= 1-x. If r approx 1-x, push left
            if r > (1.0 - x) - tol:
                forces[i, 0] -= 0.1
            # Bottom wall (y=0): constraint r <= y. If r approx y, push up
            if r > y - tol:
                forces[i, 1] += 0.1
            # Top wall (y=1): constraint r <= 1-y. If r approx 1-y, push down
            if r > (1.0 - y) - tol:
                forces[i, 1] -= 0.1

        # Apply displacement
        # Adaptive step size: smaller as we iterate to settle down
        step_size = 0.02 * (1.0 / (1.0 + iteration * 0.01))
        centers += step_size * forces
        
        # Clip to valid square
        centers = np.clip(centers, 1e-6, 1.0 - 1e-6)
        
        # Break if convergence (optional, but loop is fixed)
        if iteration % 100 == 0:
            pass # print(f"Iter {iteration}: sum_r = {sum_radii:.4f}")

    # --- 3. Final Validation and Adjustment ---
    # Re-solve LP one last time for the final centers
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        radii = res.x
    except:
        radii = np.zeros(N)
        
    # Ensure strict validity for the validator (handle float precision)
    # If any circle is slightly outside due to clip, adjust radius
    for i in range(N):
        x, y = centers[i]
        r = radii[i]
        max_r = min(x, 1-x, y, 1-y)
        if r > max_r:
            radii[i] = max_r
            
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
