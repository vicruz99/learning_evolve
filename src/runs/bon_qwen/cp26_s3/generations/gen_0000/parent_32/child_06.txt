import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    
    Returns:
        centers: np.array of shape (26, 2) with (x, y) coordinates.
        radii: np.array of shape (26,) with radius of each circle.
        sum_radii: float sum of all radii.
    """
    n = 26

    # 1. Initialize with a dense 5x5 grid + 1 circle in a gap
    xs = np.linspace(0.1, 0.9, 5)
    ys = np.linspace(0.1, 0.9, 5)
    
    centers = np.empty((n, 2))
    radii = np.empty(n)
    
    idx = 0
    for x in xs:
        for y in ys:
            centers[idx] = [x, y]
            radii[idx] = 0.1
            idx += 1
            
    # 26th circle placed in the center of the top-left quadrant gap
    centers[25] = [0.2, 0.2]
    radii[25] = 0.04
    
    # Flatten to vector for optimization: [x0, y0, r0, x1, y1, r1, ...]
    x0 = np.zeros(3 * n)
    for i in range(n):
        x0[3*i] = centers[i, 0]
        x0[3*i+1] = centers[i, 1]
        x0[3*i+2] = radii[i]
    
    # Bounds: x, y in [0, 1], r in [0, 0.5]
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * n

    def objective(vars):
        # Unpack vector
        xs = vars[0::3]
        ys = vars[1::3]
        rs = vars[2::3]
        
        # Boundary constraints: r <= dist_to_wall
        dist_wall = np.minimum(np.minimum(xs, 1.0 - xs), np.minimum(ys, 1.0 - ys))
        overlap_wall = np.maximum(0.0, rs - dist_wall)
        penalty_wall = np.sum(overlap_wall**2)
        
        # Pairwise constraints: r_i + r_j <= dist(c_i, c_j)
        centers = np.column_stack((xs, ys))
        
        # Compute pairwise distances
        dists = pdist(centers)
        dist_matrix = squareform(dists)
        
        # Compute sum of radii matrix
        sum_rs = rs[:, None] + rs[None, :]
        
        # Calculate overlap (positive means violation)
        # max(0, sum_rs - dist_matrix)
        overlap = np.maximum(0.0, sum_rs - dist_matrix)
        
        # Sum squared overlaps for the upper triangle (avoid double counting)
        penalty_pair = np.sum(np.triu(overlap)**2)
        
        # Total penalty with a high weight
        penalty = penalty_pair + penalty_wall
        
        # Objective: maximize sum(r) => minimize -sum(r) + lambda * penalty
        lam = 5000.0 
        return -np.sum(rs) + lam * penalty

    # 2. Optimize centers and radii
    res = minimize(objective, x0, bounds=bounds, method='L-BFGS-B', 
                   options={'maxiter': 2000, 'ftol': 1e-12, 'gtol': 1e-8})
    
    # 3. Extract and clean results
    final_x = res.x
    final_centers = np.column_stack((final_x[0::3], final_x[1::3]))
    final_radii = final_x[2::3]
    
    # Ensure non-negative radii (clipping tiny negative values due to numerical noise)
    final_radii = np.maximum(0.0, final_radii)
    
    sum_radii = float(np.sum(final_radii))
    
    return final_centers, final_radii, sum_radii
