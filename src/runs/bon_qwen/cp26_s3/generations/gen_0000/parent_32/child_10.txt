import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Pack 26 circles in a unit square [0,1]x[0,1] to maximize sum of radii.
    
    Returns:
        centers: np.array of shape (26, 2)
        radii: np.array of shape (26,)
        sum_radii: float
    """
    n = 26
    
    # 1. Initialization: Hexagonal Lattice
    # We create a hexagonal grid and select n points that fit best or just generate enough
    # A hexagonal grid has rows shifted by half a spacing.
    
    # Estimate initial radius based on area or simple grid
    # For 26 circles, a rough guess is r ~ 0.1
    r_init = 0.1
    spacing = 2 * r_init
    h_spacing = np.sqrt(3) * r_init
    
    # Generate points
    points = []
    row = 0
    while len(points) < n:
        # Determine number of circles in this row
        # Shift for odd rows
        x_offset = spacing / 2.0 if row % 2 == 1 else 0.0
        
        # Generate x coordinates for this row
        # We can fit roughly 1 / spacing circles
        # Let's just generate a dense set and filter or take first n
        col = 0
        while True:
            x = x_offset + col * spacing
            y = row * h_spacing
            points.append((x, y))
            col += 1
            if x > 1.5: # Limit search space
                break
        row += 1
    
    # Take the first n points
    points = points[:n]
    points = np.array(points)
    
    # Center and scale to fit in [0,1]x[0,1] with some margin
    # Find bounding box
    min_x, min_y = np.min(points, axis=0)
    max_x, max_y = np.max(points, axis=0)
    
    # Center in [0,1]
    points[:, 0] = (points[:, 0] - min_x) / (max_x - min_x + 1e-9)
    points[:, 1] = (points[:, 1] - min_y) / (max_y - min_y + 1e-9)
    
    # Add a small margin to ensure initial validity
    points = points * 0.9 + 0.05
    
    # 2. Optimization
    # Variables: [x1, y1, ..., x26, y26, r]
    # Initial guess
    x0 = np.concatenate([points.flatten(), [r_init]])
    
    # Bounds: x, y in [0, 1], r in [0, 0.5]
    bounds = []
    for _ in range(2 * n):
        bounds.append((0, 1))
    bounds.append((0, 0.5))
    
    # Constraints
    cons = []
    
    # Boundary constraints: r <= x_i <= 1-r, r <= y_i <= 1-r
    # This is equivalent to: x_i - r >= 0, 1 - x_i - r >= 0, etc.
    # In scipy minimize, constraints are usually defined as fun(x) >= 0 or fun(x) <= 0 or fun(x) == 0
    # 'ineq' means fun(x) >= 0
    
    for i in range(n):
        idx_x = 2 * i
        idx_y = 2 * i + 1
        idx_r = 2 * n
        
        # x_i - r >= 0
        cons.append({
            'type': 'ineq',
            'fun': lambda vars, i=i, idx_x=idx_x, idx_r=idx_r: vars[idx_x] - vars[idx_r]
        })
        # 1 - x_i - r >= 0 => -x_i - r >= -1 => x_i + r <= 1
        cons.append({
            'type': 'ineq',
            'fun': lambda vars, i=i, idx_x=idx_x, idx_r=idx_r: 1.0 - vars[idx_x] - vars[idx_r]
        })
        # y_i - r >= 0
        cons.append({
            'type': 'ineq',
            'fun': lambda vars, i=i, idx_y=idx_y, idx_r=idx_r: vars[idx_y] - vars[idx_r]
        })
        # 1 - y_i - r >= 0
        cons.append({
            'type': 'ineq',
            'fun': lambda vars, i=i, idx_y=idx_y, idx_r=idx_r: 1.0 - vars[idx_y] - vars[idx_r]
        })

    # Non-overlap constraints: dist(c_i, c_j) >= 2r
    # sqrt((xi-xj)^2 + (yi-yj)^2) - 2r >= 0
    # To avoid sqrt issues and derivatives, we can use squared distance if we are careful,
    # but sqrt is smooth enough except at 0. Since circles are distinct, dist > 0.
    # Actually, standard distance is fine.
    # However, (xi-xj)^2 + (yi-yj)^2 >= 4r^2 is non-convex but polynomial.
    # Let's use distance constraint directly: dist - 2r >= 0
    
    for i in range(n):
        for j in range(i + 1, n):
            idx_xi, idx_yi = 2 * i, 2 * i + 1
            idx_xj, idx_yj = 2 * j, 2 * j + 1
            idx_r = 2 * n
            
            cons.append({
                'type': 'ineq',
                'fun': lambda vars, i=i, j=j, idx_xi=idx_xi, idx_yi=idx_yi, idx_xj=idx_xj, idx_yj=idx_yj, idx_r=idx_r: 
                       np.sqrt((vars[idx_xi] - vars[idx_xj])**2 + (vars[idx_yi] - vars[idx_yj])**2) - 2 * vars[idx_r]
            })

    # Objective: Maximize sum of radii = 26 * r
    # Minimize -26 * r
    def objective(vars):
        r = vars[-1]
        return -26 * r

    # Run optimization
    # Using SLSQP which handles non-linear constraints
    result = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, 
                      options={'maxiter': 1000, 'ftol': 1e-12, 'disp': False})
    
    if result.success or result.nit > 0:
        # Extract results
        best_vars = result.x
        best_centers = best_vars[:2*n].reshape(n, 2)
        best_radius = best_vars[-1]
        best_radii = np.full(n, best_radius)
        best_sum = 26 * best_radius
    else:
        # Fallback to initial configuration if optimization fails
        best_centers = points
        best_radius = r_init
        best_radii = np.full(n, best_radius)
        best_sum = 26 * best_radius

    return best_centers, best_radii, best_sum
