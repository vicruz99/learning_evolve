import numpy as np
import math

def run_packing():

    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    Uses an iterative force-directed relaxation approach starting from multiple heuristics.
    """

    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Parameters for the simulation
    num_restarts = 10
    growth_steps = 4000  # Number of iterations to grow the radius
    initial_radius = 0.02
    radius_growth_rate = 0.00002  # Small step to increment radius
    damping = 0.9  # Velocity damping factor
    repulsion_strength = 5.0
    boundary_strength = 20.0
    
    np.random.seed(42)

    for restart_idx in range(num_restarts):
        # 1. Initialization
        # We try different initial patterns. 
        # Pattern 0: Hexagonal-ish
        # Pattern 1: Random valid
        # Pattern 2: Perturbed Grid
        
        pattern = restart_idx % 3
        centers = np.zeros((n, 2))
        
        if pattern == 0: # Hexagonal Grid
            # Try to fit a hexagonal pattern
            # 26 circles: maybe 5, 6, 5, 6, 4 rows? or similar
            # Let's just fill a grid and perturb
            idx = 0
            y = 0.1
            while idx < n:
                x = 0.1
                row_count = 0
                # Alternate row lengths for hex packing
                is_odd_row = (int(y / 0.17) % 2 == 0) 
                # Heuristic for spacing
                while x <= 1.0 - 0.1 and idx < n:
                    centers[idx, 0] = x
                    centers[idx, 1] = y
                    idx += 1
                    x += 0.17
                y += 0.15
                if idx >= n: break
        elif pattern == 1: # Random
            centers = np.random.rand(n, 2) * 0.8 + 0.1
        else: # Perturbed Grid (5x5 + 1)
            # 5x5 grid points

                    centers[count, 0] = grid_x[i]
                    centers[count, 1] = grid_y[j]
                    count += 1

            centers[25, 0] = 0.5
            centers[25, 1] = 0.5
            
            # Add small random noise
            centers += np.random.randn(n, 2) * 0.01
            # Clip to valid range for initial radius
            centers = np.clip(centers, initial_radius, 1.0 - initial_radius)

        radii = np.full(n, initial_radius)
        velocities = np.zeros_like(centers)
        current_r = initial_radius

        for step in range(growth_steps):
            # Grow radius
            # Slow down growth as we get larger to be more precise
            current_r += radius_growth_rate * (1.0 - current_r) 
            radii[:] = current_r

            forces = np.zeros_like(centers)
            
            # Pairwise Repulsion
            # Vectorized pairwise distance calculation could be faster but O(N^2) loop is fine for N=26
            for i in range(n):
                for j in range(i + 1, n):
                    diff = centers[i] - centers[j]
                    dist_sq = np.sum(diff**2)
                    dist = math.sqrt(dist_sq)
                    min_dist = 2 * current_r
                    
                    if dist < min_dist and dist > 1e-9:
                        overlap = min_dist - dist
                        # Force proportional to overlap
                        # Direction is normalized diff
                        force_vec = (diff / dist) * overlap * repulsion_strength
                        forces[i] += force_vec
                        forces[j] -= force_vec
            
            # Boundary Forces
            for i in range(n):
                for d in range(2):
                    if centers[i, d] < current_r:
                        forces[i, d] += boundary_strength * (current_r - centers[i, d])
                    elif centers[i, d] > 1.0 - current_r:
                        forces[i, d] -= boundary_strength * (centers[i, d] - (1.0 - current_r))

            velocities = velocities * damping + forces * 0.01 # 0.01 is time step for forces

                        forces[j] -= force_vec

                    max_overlap = max(max_overlap, centers[i, d] - (1.0 - current_r))

        # A safe reduction is max_overlap / 2 (since 2 circles overlap, each gives half)
        # Actually, if dist = 2r - delta, we need new_r such that dist >= 2*new_r => new_r <= dist/2.
        # So new_r <= r - delta/2.

            current_r -= max_overlap / 2.0
            # Also ensure boundary validity
            for i in range(n):
                r_i = current_r
                centers[i, 0] = np.clip(centers[i, 0], r_i, 1.0 - r_i)

        radii[:] = current_r

                    break

                    if dist < radii[i] + radii[j] - 1e-9:

                        break

        if valid and current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers.copy()
            best_radii = radii.copy()

    return best_centers, best_radii, best_sum

def validate_packing(centers, radii):
    n = centers.shape[0]
    if np.isnan(centers).any(): return False
    if np.isnan(radii).any(): return False
    for i in range(n):
        if radii[i] < 0: return False
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            return False
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False
    return True

if __name__ == "__main__":
