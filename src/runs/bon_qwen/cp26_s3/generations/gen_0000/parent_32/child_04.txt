import numpy as np
from scipy.optimize import minimize

def compute_obj(v, n, mu):
    """Objective function: negative sum of radii + penalty for constraint violations."""
    loss = -np.sum(v[2::3])
    xs = v[0::3]
    ys = v[1::3]
    rs = v[2::3]
    
    # Boundary penalties
    for i in range(n):
        xi, yi, ri = xs[i], ys[i], rs[i]
        loss += mu * (max(0, ri - xi)**2 + max(0, ri + xi - 1)**2 + 
                      max(0, ri - yi)**2 + max(0, ri + yi - 1)**2)
                      
    # Overlap penalties
    for i in range(n):
        for j in range(i + 1, n):
            xi, yi, ri = xs[i], ys[i], rs[i]
            xj, yj, rj = xs[j], ys[j], rs[j]
            dist = np.sqrt((xi - xj)**2 + (yi - yj)**2)
            ov = ri + rj - dist
            if ov > 0:
                loss += mu * ov**2
    return loss

def compute_grad(v, n, mu):
    """Analytical gradient of the penalized objective function."""
    grad = np.zeros_like(v)
    xs = v[0::3]
    ys = v[1::3]
    rs = v[2::3]
    
    for i in range(n):
        xi, yi, ri = xs[i], ys[i], rs[i]
        
        # Boundary gradients
        if ri > xi:
            err = ri - xi
            grad[i*3]   -= 2 * mu * err
            grad[i*3+2] += 2 * mu * err
        if ri + xi > 1:
            err = ri + xi - 1
            grad[i*3]   += 2 * mu * err
            grad[i*3+2] += 2 * mu * err
        if ri > yi:
            err = ri - yi
            grad[i*3+1] -= 2 * mu * err
            grad[i*3+2] += 2 * mu * err
        if ri + yi > 1:
            err = ri + yi - 1
            grad[i*3+1] += 2 * mu * err
            grad[i*3+2] += 2 * mu * err
            
    # Overlap gradients
    for i in range(n):
        for j in range(i + 1, n):
            xi, yi, ri = xs[i], ys[i], rs[i]
            xj, yj, rj = xs[j], ys[j], rs[j]
            
            dx = xi - xj
            dy = yi - yj
            dist = np.sqrt(dx**2 + dy**2)
            ov = ri + rj - dist
            
            if ov > 0:
                if dist < 1e-9:
                    dist = 1e-9
                    
                term = 2 * mu * ov / dist
                grad[i*3]   -= term * dx
                grad[i*3+1] -= term * dy
                grad[i*3+2] += 2 * mu * ov
                
                grad[j*3]   += term * dx
                grad[j*3+1] += term * dy
                grad[j*3+2] += 2 * mu * ov
                
    # Objective gradient (-1 for each radius)
    grad[2::3] -= 1.0
    return grad

def run_packing():
    n = 26
    bounds = [(0, 1), (0, 1), (0, 1)] * n
    
    best_v = None
    best_sum_r = -np.inf
    
    # Multiple restarts to avoid local minima
    for restart in range(10):
        v0 = np.zeros(n * 3)
        centers = np.zeros((n, 2))
        
        # Hexagonal grid initialization
        idx = 0
        dx = 0.2
        dy = 0.2 * np.sqrt(3)
        for r in range(7):
            for c in range(7):
                if idx >= n: break
                cx = 0.1 + c * dx + (r % 2) * dx / 2
                cy = 0.1 + r * dy
                if 0.05 <= cx <= 0.95 and 0.05 <= cy <= 0.95:
                    centers[idx] = [cx, cy]
                    idx += 1
        while idx < n:
            centers[idx] = [np.random.uniform(0.1, 0.9), np.random.uniform(0.1, 0.9)]
            idx += 1
            
        v0[:n*2] = centers.ravel()
        v0[n*2:] = 0.05 + 0.01 * np.random.rand(n)
        
        # Add jitter
        v0 += np.random.uniform(-0.02, 0.02, v0.shape)

        v = v0.copy()

            best_v = v.copy()
            
    centers = best_v[:n*2].reshape((n, 2))
