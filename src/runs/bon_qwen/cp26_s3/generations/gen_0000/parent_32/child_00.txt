```python
import numpy as np
from scipy.optimize import minimize
from functools import partial

# --- Top-level helper functions ---

def boundary_constraint_x_lower(v, i, n_circles):
    """Constraint: x_i - r_i >= 0"""
    return v[2*i] - v[2*n_circles + i]

def boundary_constraint_x_upper(v, i, n_circles):
    """Constraint: 1 - x_i - r_i >= 0"""
    return 1.0 - v[2*i] - v[2*n_circles + i]

def boundary_constraint_y_lower(v, i, n_circles):
    """Constraint: y_i - r_i >= 0"""
    return v[2*i+1] - v[2*n_circles + i]

def boundary_constraint_y_upper(v, i, n_circles):
    """Constraint: 1 - y_i - r_i >= 0"""
    return 1.0 - v[2*i+1] - v[2*n_circles + i]

def circle_separation_constraint(v, i, j, n_circles):
    """Constraint: dist^2 - (r_i + r_j)^2 >= 0"""
    xi, yi = v[2*i], v[2*i+1]
    xj, yj = v[2*j], v[2*j+1]
    ri, rj = v[2*n_circles + i], v[2*n_circles + j]
    return (xi - xj)**2 + (yi - yj)**2 - (ri + rj)**2

def objective_neg_sum_radii(v, n_circles):
    """Objective: minimize negative sum of radii"""
    return -np.sum(v[2*n_circles:])

# --- Main packing function ---

def run_packing():
    n_circles = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Run optimization with multiple restarts to escape local optima
    n_restarts = 30
    for seed in range(n_restarts):
        rng = np.random.RandomState(seed)
        
        # 1. Initialization: Hexagonal grid pattern
        # This provides a dense, feasible starting configuration
        centers_init = []
        # Base spacing, adjusted to fit within [0,1] with margin
        spacing = 0.18
        for row in range(6):
            for col in range(6):
                if len(centers_init) >= n_circles:
                    break
                y = 0.08 + row * spacing
                x = 0.08 + col * spacing + (row % 2) * (spacing * 0.5)
                if 0 <= x <= 1 and 0 <= y <= 1:
                    centers_init.append([x, y])
                    
        # Fill remaining slots if grid didn't provide enough
        while len(centers_init) < n_circles:
            centers_init.append([rng.uniform(0.15, 0.85), rng.uniform(0.15, 0.85)])
            
        centers_init = np.array(centers_init[:n_circles])
        
        # Initial radii set small to ensure strict feasibility
        r_init = np.full(n_circles, 0.04)
        
        # Flatten variables for scipy: [x0, y0, x1, y1, ..., r0, r1, ...]
        x0 = np.concatenate([centers_init.ravel(), r_init])
        
        # Bounds for variables
        bounds = [(0.0, 1.0)] * (2 * n_circles) + [(0.0, 1.0)] * n_circles
        
        # Define constraints
        constraints = []
        for i in range(n_circles):
            constraints.append({'type': 'ineq', 'fun': partial(boundary_constraint_x_lower, i=i, n_circles=n_circles)})
            constraints.append({'type': 'ineq', 'fun': partial(boundary_constraint_x_upper, i=i, n_circles=n_circles)})
            constraints.append({'type': 'ineq', 'fun': partial(boundary_constraint_y_lower, i=i, n_circles=n_circles)})
            constraints.append({'type': 'ineq', 'fun': partial(boundary_constraint_y_upper, i=i, n_circles=n_circles)})
            
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                constraints.append({'type': 'ineq', 'fun': partial(circle_separation_constraint, i=i, j=j, n_circles=n_circles)})
                
        # Run optimization
        try:
            res = minimize(
                partial(objective_neg_sum_radii, n_circles=n_circles),
                x0,
                bounds=bounds,
                constraints=constraints,
                method='SLSQP',
                options={'maxiter': 2000, 'ftol': 1e-12, 'disp': False}
            )
            
            if res.success and -res.fun > best_sum:
                best_sum = -res.fun
                best_centers = res.x[:2*n_circles].reshape(n_circles, 2)
                best_radii = res.x[2*n_circles:]
        except Exception:
            continue
            
    return best_centers, best_radii, best_sum
```
