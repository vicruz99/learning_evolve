def compute_violations(centers, radii):
    """
    Computes the sum of squared constraint violations.
    Violations include:
    1. Circles outside the unit square [0,1]x[0,1].
    2. Circles overlapping each other.
    """
    n = centers.shape[0]
    violation = 0.0
    
    # Boundary violations
    # x - r >= 0 => r - x <= 0. Violation if x - r < 0.
    # x + r <= 1 => x + r - 1 <= 0. Violation if x + r > 1.
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        
        # Left boundary
        if x - r < 0:
            violation += (x - r)**2
        # Right boundary
        if x + r > 1:
            violation += (x + r - 1)**2
        # Bottom boundary
        if y - r < 0:
            violation += (y - r)**2
        # Top boundary
        if y + r > 1:
            violation += (y + r - 1)**2
            
        # Negative radius check (though bounds should handle this)
        if r < 0:
            violation += r**2

    # Overlap violations
    # dist >= r1 + r2 => dist - (r1 + r2) >= 0.
    # Violation if dist < r1 + r2.
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = math.sqrt(dx*dx + dy*dy)
            min_dist = radii[i] + radii[j]
            
            if dist < min_dist:
                diff = dist - min_dist
                violation += diff**2
                
    return violation
