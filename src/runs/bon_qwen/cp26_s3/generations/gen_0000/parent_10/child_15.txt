import numpy as np
from scipy.optimize import minimize
import typing

def _compute_objective(vars, n, penalty, triu):
    """
    Computes the objective: negative sum of radii + penalty terms for constraint violations.
    """
    centers = vars[:2*n].reshape(n, 2)
    radii = vars[2*n:]
    radii = np.maximum(radii, 0.0)
    
    # Boundary violations: circles must stay inside [0,1]x[0,1]
    b1 = np.maximum(0, radii - centers[:, 0])      # left wall
    b2 = np.maximum(0, radii + centers[:, 0] - 1)  # right wall
    b3 = np.maximum(0, radii - centers[:, 1])      # bottom wall
    b4 = np.maximum(0, radii + centers[:, 1] - 1)  # top wall
    boundary_pen = penalty * np.sum(b1**2 + b2**2 + b3**2 + b4**2)
    
    # Overlap violations: distance between centers must be >= sum of radii
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.linalg.norm(diffs, axis=2)
    sum_r = radii[:, np.newaxis] + radii[np.newaxis, :]
    overlaps = np.maximum(0, sum_r - dists)
    
    # Apply penalty only to upper triangle to avoid double counting
    overlap_pen = penalty * np.sum((overlaps * triu)**2)
    
    return -np.sum(radii) + boundary_pen + overlap_pen

def _get_init(n):
    """
    Generates an initial hexagonal/staggered grid placement for n circles.
    """
    centers = np.zeros((n, 2))
    idx = 0
    row_height = 0.14
    col_width = 0.15
    for r in range(6):
        for c in range(6):
            if idx >= n: break
            x = 0.12 + c * col_width
            y = 0.12 + r * row_height + (0.07 if c % 2 else 0)
            centers[idx] = [x, y]
            idx += 1
        if idx >= n: break
    return centers

def run_packing() -> typing.tuple[np.ndarray, np.ndarray, float]:
    n = 26
    triu = np.triu(np.ones((n, n), dtype=bool), k=1)
    bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n
    
    best_loss = np.inf
    best_vars = None
    
    # Multi-restart optimization
    for seed in range(40):
        np.random.seed(seed)
        centers_init = _get_init(n)
        # Add controlled jitter to break symmetry and explore configuration space
        centers_init += np.random.uniform(-0.015, 0.015, centers_init.shape)
        centers_init = np.clip(centers_init, 0.05, 0.95)
        
        vars0 = np.concatenate([centers_init.flatten(), np.full(n, 0.08)])
        current_vars = vars0.copy()
        
        # Sequential penalty annealing: gradually tighten constraints
        for pen in [50, 300, 1500, 8000, 30000]:
            res = minimize(_compute_objective, current_vars, args=(n, pen, triu), 
                           method='L-BFGS-B', bounds=bounds, 
                           options={'maxiter': 5000, 'ftol': 1e-12, 'gtol': 1e-10})
            current_vars = res.x
            
        loss = res.fun
        if loss < best_loss:
            best_loss = loss
            best_vars = current_vars.copy()
            
    centers = best_vars[:2*n].reshape(n, 2)
    radii = best_vars[2*n:]
    radii = np.maximum(radii, 0.0)
    
    return centers, radii, float(np.sum(radii))
