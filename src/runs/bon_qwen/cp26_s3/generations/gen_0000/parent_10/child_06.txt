            x = 0.05 + col * 0.15 + (0.075 if row % 2 == 1 else 0.0)

            # Ensure points are within the unit square (loosely)
            if 0 <= x <= 1 and 0 <= y <= 1:
                centers.append([x, y])

                    break
        row += 1

    # Start with a small radius to ensure valid initial state
    radii = np.full(n, 0.05)

    # Initial state vector: [x1, y1, r1, x2, y2, r2, ...]
    x0 = np.zeros(3 * n)
    for i in range(n):
        x0[3 * i] = centers[i, 0]
        x0[3 * i + 1] = centers[i, 1]
        x0[3 * i + 2] = radii[i]

    # Objective: Maximize sum of radii (Minimize negative sum)
    def objective(z):
        # Radii are at indices 2, 5, 8, ... (i.e., 3*i + 2)
        radii_vals = z[2::3]
        return -np.sum(radii_vals)

    constraints = []

    # Linear constraints for boundaries:
    # r <= x <= 1-r => x-r >= 0 and x+r <= 1
    # r <= y <= 1-r => y-r >= 0 and y+r <= 1
    for i in range(n):

        # x - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda z, i=i: z[3 * i] - z[3 * i + 2]
        })
        # 1 - x - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda z, i=i: 1.0 - z[3 * i] - z[3 * i + 2]
        })
        # y - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda z, i=i: z[3 * i + 1] - z[3 * i + 2]

        })

    # Quadratic non-linear constraints for non-overlap

    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({

            })

    # Bounds for all variables [0, 1]
    bounds = [(0, 1)] * (3 * n)

    # Run Optimization
    # SLSQP is suitable for non-linear constraints
    res = minimize(

        x0, 
        method='SLSQP', 
        bounds=bounds,
