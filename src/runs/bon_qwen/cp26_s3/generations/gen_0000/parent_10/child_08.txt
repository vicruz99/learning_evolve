import numpy as np

import math

def compute_cost(params, n=26):
    """
    Computes the cost function: negative sum of radii + penalty for overlaps and boundary violations.
    params: array of shape (n, 3) -> [x, y, r] for each circle
    """
    centers = params[:, :2]
    radii = params[:, 2]
    
    sum_radii = np.sum(radii)
    
    penalty = 0.0
    
    # Boundary penalties
    # Circle i must be inside [0, 1] x [0, 1]
    # Constraints: x >= r, x <= 1-r, y >= r, y <= 1-r
    # Equivalently: r <= x, r <= 1-x, r <= y, r <= 1-y
    # Violation: r - x > 0 => penalty (r - x)^2
    
    for i in range(n):
        x, y, r = params[i]
        # Boundary checks
        if x < r: penalty += (r - x)**2
        if x > 1 - r: penalty += (r - (1 - x))**2
        if y < r: penalty += (r - y)**2
        if y > 1 - r: penalty += (r - (1 - y))**2
        
        # Overlap checks (only need to check pairs)
        # To optimize speed, we can vectorize, but for n=26 loop is fine
        for j in range(i + 1, n):
            x2, y2, r2 = params[j]
            dist = np.sqrt((x - x2)**2 + (y - y2)**2)
            min_dist = r + r2
            if dist < min_dist:
                penalty += (min_dist - dist)**2
                
    return -sum_radii + 1000.0 * penalty
