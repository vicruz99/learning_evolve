from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # --- Initialization ---
    # Create a staggered grid (approx hexagonal) to get close to optimal density
    rows = 6
    cols = 5
    spacing_x = 1.0 / (cols + 1)
    spacing_y = 1.0 / (rows + 1)
    
    centers = []
    radii = []
    count = 0
    
    for r in range(rows):
        for c in range(cols):
            if count >= n:
                break
            x = (c + 1) * spacing_x
            y = (r + 1) * spacing_y
            # Offset every other row to create hexagonal packing
            if r % 2 == 1:
                x += spacing_x / 2
            centers.append([x, y])
            radii.append(0.05) # Small initial radius
            count += 1
        if count >= n:
            break
            
    centers = np.array(centers[:n])
    radii = np.array(radii[:n])

    # --- Optimization ---
    # Objective: Maximize sum(radii) -> Minimize -sum(radii)
    def objective(x):
        # x is a 1D array [x0, y0, r0, x1, y1, r1, ...]
        return -np.sum(x[2::3]) # Sum of radii is at indices 2, 5, 8...

    # Constraints
    def boundary_constraints(x):
        cons = []
        for i in range(n):
            idx = i * 3
            x_pos, y_pos, r = x[idx], x[idx+1], x[idx+2]
            # r >= 0
            cons.append(r)
            # x - r >= 0
            cons.append(x_pos - r)
            # 1 - (x + r) >= 0
            cons.append(1.0 - (x_pos + r))
            # y - r >= 0
            cons.append(y_pos - r)
            # 1 - (y + r) >= 0
            cons.append(1.0 - (y_pos + r))
        return cons

    def separation_constraints(x):
        cons = []
        for i in range(n):
            idx_i = i * 3
            for j in range(i + 1, n):
                idx_j = j * 3
                x_i, y_i, r_i = x[idx_i], x[idx_i+1], x[idx_i+2]
                x_j, y_j, r_j = x[idx_j], x[idx_j+1], x[idx_j+2]
                
                dist_sq = (x_i - x_j)**2 + (y_i - y_j)**2
                sum_r_sq = (r_i + r_j)**2
                
                # dist >= r_i + r_j  => dist^2 >= (r_i + r_j)^2
                # We add a small tolerance to help numerical stability
                cons.append(dist_sq - sum_r_sq + 1e-5) 
        return cons

    # Flatten initial state
    x0 = np.zeros(n * 3)
    for i in range(n):
        x0[i*3] = centers[i, 0]
        x0[i*3+1] = centers[i, 1]
        x0[i*3+2] = radii[i]

    constraints = []

            })

    # Run optimization
    result = minimize(objective, x0, method='SLSQP', constraints=constraints, options={'maxiter': 1000})

    # Extract results

    final_radii = result.x.reshape(-1, 3)[:, 2]

    final_radii = np.maximum(final_radii, 0.0)
    
    sum_radii = np.sum(final_radii)

    return final_centers, final_radii, sum_radii
