import numpy as np
from scipy.optimize import minimize

def run_packing():
    """
    Pack 26 circles in a unit square [0,1]x[0,1] to maximize the sum of radii.
    """
    n = 26
    
    # 1. Initialization: Hexagonal Grid Layout
    # We aim for a pattern of rows with counts 6, 5, 6, 5, 4 to get 26 circles.
    # This is a dense hexagonal packing structure.
    
    # Start with a radius small enough to fit this pattern easily, e.g., 0.08
    # With r=0.08, width of row with 6 circles is 12r = 0.96 < 1.
    # Height is roughly 9r = 0.72 < 1.
    r_init = 0.08
    
    centers = []
    # Row parameters
    # Row 0: 6 circles, y = r, x = r, 3r, 5r, 7r, 9r, 11r
    # Row 1: 5 circles, y = r + sqrt(3)r, x = 2r, 4r, 6r, 8r, 10r
    # Row 2: 6 circles, y = r + 2sqrt(3)r, x = r, ...
    # Row 3: 5 circles, y = r + 3sqrt(3)r, x = 2r, ...
    # Row 4: 4 circles, y = r + 4sqrt(3)r, x = r, 3r, 5r, 7r
    
    sqrt3 = np.sqrt(3)
    rows_config = [
        (r_init, 0, 6),       # y_offset=0, x_offset=0, count=6
        (r_init + sqrt3*r_init, 1, 5), # y_offset, x_offset=1 (shift by r), count=5
        (r_init + 2*sqrt3*r_init, 0, 6),
        (r_init + 3*sqrt3*r_init, 1, 5),
        (r_init + 4*sqrt3*r_init, 0, 4)
    ]
    
    for y_base, x_offset_idx, count in rows_config:
        # If offset index is 0, start x at r_init. If 1, start at r_init + r_init (which is 2r? No, shift is r)
        # In hex packing, shift is r horizontally (since dx=2r, shift is dx/2 = r).
        # So if offset is 0, x = r, 3r... (indices 0, 2, 4...) relative to r step?
        # Actually, x coordinates are r, 3r, 5r... for even rows.
        # And 2r, 4r... for odd rows.
        # So start x is r_init + x_offset_idx * r_init.
        start_x = r_init + x_offset_idx * r_init
        
        for k in range(count):
            x = start_x + k * (2 * r_init)
            centers.append([x, y_base])
    
    # Fill any missing circles if count < 26 (should be exactly 26 with config above)
    # 6+5+6+5+4 = 26.
    
    # Initialize variables vector: [x1, y1, r1, x2, y2, r2, ...]
    # Shape (78,)
    vars0 = np.zeros(3 * n)
    for i in range(n):
        vars0[3*i] = centers[i][0]
        vars0[3*i+1] = centers[i][1]
        vars0[3*i+2] = r_init

    # 2. Define Objective and Constraints
    
    # Objective: Maximize sum of radii => Minimize -sum(radii)
    def objective(v):
        # Radii are at indices 2, 5, 8, ...
        return -np.sum(v[2::3])

    # Boundary Constraints:
    # For each circle i: r <= x <= 1-r  => x - r >= 0, 1 - x - r >= 0
    #                    r <= y <= 1-r  => y - r >= 0, 1 - y - r >= 0
    def boundary_constraints(v):
        res = np.zeros(4 * n)
        for i in range(n):
            x = v[3*i]
            y = v[3*i+1]
            r = v[3*i+2]
            res[4*i]   = x - r
            res[4*i+1] = 1.0 - x - r
            res[4*i+2] = y - r
            res[4*i+3] = 1.0 - y - r
        return res

    # Overlap Constraints:
    # For all i < j: dist_ij^2 >= (r_i + r_j)^2
    # dist^2 - (r_i + r_j)^2 >= 0
    def overlap_constraints(v):
        # Extract coordinates and radii
        xs = v[0::3]
        ys = v[1::3]
        rs = v[2::3]
        
        # Compute pairwise squared distances using broadcasting
        # shape (n, n)
        diff_x = xs[:, None] - xs[None, :]
        diff_y = ys[:, None] - ys[None, :]
        dist_sq = diff_x**2 + diff_y**2
        
        # Compute pairwise sum of radii squared
        r_sum = rs[:, None] + rs[None, :]
        r_sum_sq = r_sum**2
        
        # We need constraints for i < j.
        # Upper triangular part (excluding diagonal)
        # Mask for upper triangle
        mask = np.triu(np.ones((n, n), dtype=bool), k=1)
        
        vals = dist_sq[mask] - r_sum_sq[mask]
        return vals

    # Bounds for variables
    # x, y in [0, 1], r >= 0
    bounds = []
    for _ in range(n):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 1.0)) # r (upper bound 1 is safe)

    constraints = [
        {'type': 'ineq', 'fun': boundary_constraints},
        {'type': 'ineq', 'fun': overlap_constraints}
    ]

    # 3. Optimization
    # Using SLSQP which handles non-linear constraints well.
    # We set a reasonable maxiter and tolerance.
    result = minimize(
        objective, 
        vars0, 
        method='SLSQP', 
        bounds=bounds, 
        constraints=constraints,
        options={
            'maxiter': 2000,
            'ftol': 1e-12,
            'disp': False
        }
    )
    
    # 4. Extract and Format Result
    if result.success or result.fun < -2.5: # Heuristic check
        final_centers = np.zeros((n, 2))
        final_radii = np.zeros(n)
        for i in range(n):
            final_centers[i, 0] = result.x[3*i]
            final_centers[i, 1] = result.x[3*i+1]
            final_radii[i] = result.x[3*i+2]
    else:
        # Fallback to initial config if optimization failed badly
        final_centers = np.array(centers)
        final_radii = np.full(n, r_init)

    # Ensure radii are non-negative (just in case)
    final_radii = np.maximum(final_radii, 0)
    
    sum_radii = np.sum(final_radii)
    
    return final_centers, final_radii, sum_radii
