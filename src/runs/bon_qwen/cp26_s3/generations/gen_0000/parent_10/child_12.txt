import numpy as np

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    np.random.seed(42)
    
    # Hexagonal initialization for better starting configuration
    centers = []
    spacing = 0.20
    offset = 0.10
    for r in range(6):
        for c in range(6):
            if len(centers) >= n: break
            x = c * spacing + (r % 2) * spacing / 2 + offset
            y = r * spacing * np.sqrt(3)/2 + offset
            centers.append([x, y])
        if len(centers) >= n: break
    centers = np.array(centers[:n])
    centers = np.clip(centers, 0.0, 1.0)
    
    # Optimization parameters
    lr = 0.008
    r_target = 0.05
    r_inc = 0.0002
    max_iter = 8000
    
    for it in range(max_iter):
        forces = np.zeros_like(centers)
        min_dist = 1.0
        
        # Pairwise repulsion forces
        for i in range(n):
            for j in range(i+1, n):
                diff = centers[i] - centers[j]
                d = np.linalg.norm(diff)
                if d < 1e-9: d = 1e-9
                if d < min_dist: min_dist = d
                
                if d < 2 * r_target:
                    # Repulsive force proportional to overlap
                    f = (2 * r_target - d) / d
                    forces[i] += f * diff
                    forces[j] -= f * diff
                    
        # Boundary repulsion forces
        for i in range(n):
            x, y = centers[i]
            # Update min_dist with boundary distances
            for val in [x, 1-x, y, 1-y]:
                if val < min_dist: min_dist = val
            
            # Push away from boundaries if too close
            if x < r_target: forces[i, 0] += (r_target - x) * 20
            if x > 1 - r_target: forces[i, 0] -= (x - (1 - r_target)) * 20
            if y < r_target: forces[i, 1] += (r_target - y) * 20
            if y > 1 - r_target: forces[i, 1] -= (y - (1 - r_target)) * 20
            
        # Update positions
        centers += lr * forces
        centers = np.clip(centers, 0.0, 1.0)
        
        # Adaptive step and target radius
        if min_dist >= 2 * r_target:
            r_target += r_inc
        else:
            lr *= 0.997
            
        if lr < 1e-7: break
        
    # Compute final valid radius based on tightest constraint
    min_d = 1.0
    for i in range(n):
        for j in range(i+1, n):
            d = np.linalg.norm(centers[i] - centers[j])
            if d < min_d: min_d = d
        for val in [centers[i,0], 1-centers[i,0], centers[i,1], 1-centers[i,1]]:
            if val < min_d: min_d = val
            
    r = 0.5 * min_d
    radii = np.full(n, r)
    
    return centers, radii, float(np.sum(radii))
