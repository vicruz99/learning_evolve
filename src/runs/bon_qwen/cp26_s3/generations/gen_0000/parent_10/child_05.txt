I'll use an alternating optimization approach:

1. **LP-based radius computation**: For fixed centers, use linear programming (LP) to compute the maximum possible radii. This is because given fixed centers, the constraints on radii are all linear: r_i ≤ x_i, r_i ≤ 1-x_i, r_i ≤ y_i, r_i ≤ 1-y_i, and r_i + r_j ≤ ||c_i - c_j||.

2. **Gradient-based center optimization**: For fixed radii, use gradient descent to move centers. The idea is to push circles away from tight constraints (overlapping circles and boundaries).

3. **Multiple restarts**: Try different initial configurations (hexagonal grids with different spacings, random perturbations) to escape local optima.

4. **Iterative refinement**: Alternate between LP for radii and gradient descent for centers, with decaying learning rates.

Key insight: The sum of radii is maximized when circles are as large as possible while maintaining non-overlap and boundary constraints. Hexagonal packing is optimal for equal circles, and the LP formulation handles unequal circles naturally.

The LP has n variables and ~n + n(n-1)/2 constraints, which is manageable for n=26.

import numpy as np
from scipy.optimize import linprog


def compute_radii_lp(centers):
    """Given fixed centers, compute maximum radii using linear programming."""
    n = centers.shape[0]
    
    # Objective: maximize sum(r_i) -> minimize -sum(r_i)
    c_obj = np.ones(n) * (-1.0)
    
    # Build constraint matrix A_ub @ r <= b_ub
    n_boundary = 4 * n
    n_overlap = n * (n - 1) // 2
    A_ub = np.zeros((n_boundary + n_overlap, n))
    b_ub = np.zeros(n_boundary + n_overlap)
    
    idx = 0
    for i in range(n):
        x, y = centers[i]
        # r_i <= x_i
        A_ub[idx, i] = 1.0
        b_ub[idx] = x
        idx += 1
        # r_i <= 1 - x_i
        A_ub[idx, i] = 1.0
        b_ub[idx] = 1.0 - x
        idx += 1
        # r_i <= y_i
        A_ub[idx, i] = 1.0
        b_ub[idx] = y
        idx += 1
        # r_i <= 1 - y_i
        A_ub[idx, i] = 1.0
        b_ub[idx] = 1.0 - y
        idx += 1
    
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dist
            idx += 1
    
    bounds = [(0.0, None)] * n
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        return result.x, -result.fun
    else:
        return np.zeros(n), 0.0


def init_hex_grid(n, spacing, random_seed=0):
    """Initialize centers in a hexagonal grid pattern."""
    np.random.seed(random_seed)
    centers = np.zeros((n, 2))
    idx = 0
    
    row = 0
    max_rows = 20
    while idx < n and row < max_rows:
        col = 0
        x_off = (row % 2) * spacing * 0.5
        while idx < n:
            x = spacing * 0.5 + x_off + col * spacing
            y = spacing * 0.5 + row * spacing * np.sqrt(3) / 2.0
            
            if x <= 1.0 - spacing * 0.5 and y <= 1.0 - spacing * 0.5:
                centers[idx] = [x, y]
                idx += 1
            col += 1
        row += 1
    
    # Add small random perturbation
    centers += np.random.randn(n, 2) * 0.001
    
    # Ensure within bounds
    centers = np.clip(centers, 0.01, 0.99)
    
    return centers


def optimize_centers_gradient(centers, radii, n, lr=0.005, n_iters=500, cool_rate=0.998):
    """
    Optimize centers for fixed radii using gradient-based approach.
    Push circles apart when they're too close, push away from boundaries.
    """
    centers = centers.copy()
    current_lr = lr
    
    for iteration in range(n_iters):
        grad = np.zeros((n, 2))
        
        for i in range(n):
            # Boundary repulsion
            x, y = centers[i]
            r = radii[i]
            
            # Distance to left boundary
            if x - r < 0.01:
                grad[i, 0] += 1.0 / max(0.001, x - r)
            # Distance to right boundary
            if 1.0 - x - r < 0.01:
                grad[i, 0] -= 1.0 / max(0.001, 1.0 - x - r)
            # Distance to bottom boundary
            if y - r < 0.01:
                grad[i, 1] += 1.0 / max(0.001, y - r)
            # Distance to top boundary
            if 1.0 - y - r < 0.01:
                grad[i, 1] -= 1.0 / max(0.001, 1.0 - y - r)
            
            # Inter-circle repulsion
            for j in range(n):
                if i != j:
                    dx = centers[i, 0] - centers[j, 0]
                    dy = centers[i, 1] - centers[j, 1]
                    dist = np.sqrt(dx * dx + dy * dy)
                    min_dist = radii[i] + radii[j]
                    
                    if dist < min_dist + 0.05 and dist > 1e-8:
                        repulsion_strength = (min_dist + 0.05 - dist) / (dist + 1e-8)
                        grad[i, 0] += dx * repulsion_strength
                        grad[i, 1] += dy * repulsion_strength
        
        # Apply gradient
        centers += current_lr * grad
        
        # Project back to feasible region
        for i in range(n):
            centers[i, 0] = np.clip(centers[i, 0], radii[i] + 1e-8, 1.0 - radii[i] - 1e-8)
            centers[i, 1] = np.clip(centers[i, 1], radii[i] + 1e-8, 1.0 - radii[i] - 1e-8)
        
        current_lr *= cool_rate
    
    return centers


def alternating_optimization(centers_init, n, max_outer=20, max_inner=300):
    """
    Alternate between optimizing radii (LP) and centers (gradient).
    """
    centers = centers_init.copy()
    
    best_sum = 0.0
    best_centers = centers.copy()
    best_radii = None
    
    for outer in range(max_outer):
        # Step 1: Compute optimal radii via LP
        radii, sum_r = compute_radii_lp(centers)
        
        # Check if any radius is too small (degenerate)
        if np.any(radii < 1e-8):
            radii = np.maximum(radii, 1e-6)
        
        if sum_r > best_sum:
            best_sum = sum_r
            best_centers = centers.copy()
            best_radii = radii.copy()
        
        # Step 2: Optimize centers for these radii
        lr = 0.01 / (1 + outer * 0.2)
        centers = optimize_centers_gradient(centers, radii, n, lr=lr, n_iters=max_inner)
    
    # Final LP to get optimal radii for best centers
    if best_radii is not None:
        radii_final, sum_final = compute_radii_lp(best_centers)
        if np.any(radii_final < 1e-8):
            radii_final = np.maximum(radii_final, 1e-6)
        return best_centers, radii_final, np.sum(radii_final)
    
    radii_final, sum_final = compute_radii_lp(centers)
    if np.any(radii_final < 1e-8):
        radii_final = np.maximum(radii_final, 1e-6)
    return centers, radii_final, np.sum(radii_final)


def run_packing():
    n = 26
    
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Try multiple initial configurations
    spacings = [0.16, 0.17, 0.18, 0.19, 0.20]
    seeds = [0, 1, 2, 3, 4, 5, 10, 20, 30, 42]
    
    for spacing in spacings:
        for seed in seeds:
            centers_init = init_hex_grid(n, spacing, seed)
            centers_opt, radii_opt, sum_r = alternating_optimization(centers_init, n)
            
            # Validate
            if validate_internal(centers_opt, radii_opt, n):
                if sum_r > best_sum:
                    best_sum = sum_r
                    best_centers = centers_opt.copy()
                    best_radii = radii_opt.copy()
    
    if best_centers is None:
        # Fallback
        centers_init = init_hex_grid(n, 0.18, 0)
        best_centers, best_radii, best_sum = alternating_optimization(centers_init, n)
    
    return best_centers, best_radii, float(best_sum)


def validate_internal(centers, radii, n):
    """Internal validation (same logic as the provided validate function)."""
    for i in range(n):
        if np.isnan(centers[i, 0]) or np.isnan(centers[i, 1]):
            return False
        if np.isnan(radii[i]) or radii[i] < 0:
            return False
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-9 or x + r > 1.0 + 1e-9 or y - r < -1e-9 or y + r > 1.0 + 1e-9:
            return False
    
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            if dist < radii[i] + radii[j] - 1e-9:
                return False
    
    return True
