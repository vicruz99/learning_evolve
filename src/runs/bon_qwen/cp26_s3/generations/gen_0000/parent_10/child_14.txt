import numpy as np
from scipy.optimize import minimize

def compute_loss(vars, n):
    """
    Objective function: minimize negative sum of radii + overlap penalty.
    vars layout: [u1_x, u1_y, r1, u2_x, u2_y, r2, ..., u26_x, u26_y, r26]
    """
    radii = vars[2::3]
    u_x = vars[0::3]
    u_y = vars[1::3]
    
    centers = np.zeros((n, 2))
    for i in range(n):
        r = radii[i]
        w = 1.0 - 2.0 * r
        centers[i, 0] = r + w * u_x[i]
        centers[i, 1] = r + w * u_y[i]
        
    loss = -np.sum(radii)
    penalty = 0.0
    for i in range(n):
        ri = radii[i]
        ci = centers[i]
        for j in range(i + 1, n):
            rj = radii[j]
            cj = centers[j]
            dx = ci[0] - cj[0]
            dy = ci[1] - cj[1]
            dist = np.sqrt(dx*dx + dy*dy)
            min_dist = ri + rj
            if dist < min_dist:
                diff = min_dist - dist
                penalty += diff * diff
    return loss + 2000.0 * penalty

def run_packing():
    np.random.seed(42)
    n = 26
    
    # Initial configuration: hexagonal-ish grid
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.05)
    idx = 0
    row_counts = [6, 5, 6, 5, 4]
    for r_idx, count in enumerate(row_counts):
        y_init = 0.15 + r_idx * 0.16
        for c_idx in range(count):
            x_init = 0.1 + c_idx * 0.14 + (0.07 if r_idx % 2 == 1 else 0)
            centers[idx] = [x_init + np.random.uniform(-0.02, 0.02), 
                            y_init + np.random.uniform(-0.02, 0.02)]
            idx += 1
            
    # Map to parameter space u in [0,1], r in [0.01, 0.5]
    # c = r + (1-2r)*u  =>  u = (c - r) / (1-2r)
    x0 = np.zeros(3*n)
    for i in range(n):
        denom = 1.0 - 2.0 * radii[i]
        u = (centers[i] - radii[i]) / denom
        x0[3*i] = np.clip(u[0], 0, 1)
        x0[3*i+1] = np.clip(u[1], 0, 1)
        x0[3*i+2] = radii[i]
        
    bounds = [(0, 1) for _ in range(2*n)] + [(0.01, 0.5) for _ in range(n)]
    
    res = minimize(compute_loss, x0, args=(n,), method='L-BFGS-B', 
                   bounds=bounds, options={'maxiter': 5000, 'ftol': 1e-10, 'gtol': 1e-8})
                   
    radii_opt = res.x[2::3]
    u_x = res.x[0::3]
    u_y = res.x[1::3]
    
    centers_opt = np.zeros((n, 2))
    for i in range(n):
        r = radii_opt[i]
        w = 1.0 - 2.0 * r
        centers_opt[i, 0] = r + w * u_x[i]
        centers_opt[i, 1] = r + w * u_y[i]
        
    return centers_opt, radii_opt, np.sum(radii_opt)
