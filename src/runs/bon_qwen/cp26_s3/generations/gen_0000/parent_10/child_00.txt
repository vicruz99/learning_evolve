def compute_overlap_penalties(centers, radii):
    """
    Compute a penalty value based on overlaps and boundary violations.
    Returns 0 if valid, positive if invalid.
    """
    penalty = 0.0
    n = len(centers)
    
    # Boundary checks
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        # Distance to walls
        d_x = min(x - r, 1 - (x + r))
        d_y = min(y - r, 1 - (y + r))
        if d_x < 0:
            penalty += (-d_x) * 1000
        if d_y < 0:
            penalty += (-d_y) * 1000
            
    # Overlap checks
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            r_sum = radii[i] + radii[j]
            if dist < r_sum:
                penalty += (r_sum - dist) * 1000
                
    return penalty
