import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def evaluate_constraints(x):
    """Evaluates non-overlap constraints for all pairs of circles."""
    n = N_CIRCLES
    num_constraints = n * (n - 1) // 2
    constraints = np.zeros(num_constraints)
    idx = 0
    for i in range(n):
        xi, yi, ri = x[3*i], x[3*i+1], x[3*i+2]
        for j in range(i + 1, n):
            xj, yj, rj = x[3*j], x[3*j+1], x[3*j+2]
            dx = xi - xj
            dy = yi - yj
            dist_sq = dx*dx + dy*dy
            sum_r = ri + rj
            constraints[idx] = dist_sq - sum_r*sum_r
            idx += 1
    return constraints

def objective(x):
    """Objective function: negative sum of radii (to be minimized)."""
    return -np.sum(x[2::3])

def run_packing():
    n = N_CIRCLES
    # Initialize positions on a structured grid to ensure feasibility
    init_pos = np.zeros((n, 2))
    cols = 6
    rows = (n + cols - 1) // cols
    
    # Create a hex-like or grid pattern
    for i in range(n):
        col = i % cols
        row = i // cols
        # Offset even rows for hexagonal packing
        offset = 0.08 if row % 2 == 1 else 0.0
        init_pos[i, 0] = 0.1 + col * 0.15 + offset
        init_pos[i, 1] = 0.1 + row * 0.15

    # Flatten initial state: [x0, y0, r0, x1, y1, r1, ...]
    x0 = np.zeros(3 * n)
    init_radius = 0.06
    for i in range(n):
        x0[3*i] = init_pos[i, 0]
        x0[3*i+1] = init_pos[i, 1]
        x0[3*i+2] = init_radius

    # Bounds: x, y in [0, 1], r in [0, 1]
    bounds = [(0, 1)] * (2 * n) + [(0, 1)] * n

    # Constraints definition
    cons = {'type': 'ineq', 'fun': evaluate_constraints}

    # Run optimization
    result = minimize(
        objective,
        x0,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 5000, 'ftol': 1e-10, 'disp': False}
    )

    if result.success:
        x_opt = result.x
    else:
        # Fallback to initial if optimization fails (unlikely)
        x_opt = x0

    centers = x_opt[0::3].reshape(-1, 2)
    radii = x_opt[2::3]

    # Iteratively resolve any minor constraint violations due to numerical precision
    for _ in range(20):
        violation = False
        # Check boundaries
        for i in range(n):
            cx, cy, cr = centers[i, 0], centers[i, 1], radii[i]
            max_r_bound = min(cx, 1 - cx, cy, 1 - cy)
            if max_r_bound < cr - 1e-9:
                radii[i] = max(0, max_r_bound)
                violation = True

        # Check overlaps

            for j in range(i + 1, n):
                dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                sum_r = radii[i] + radii[j]
                if dist < sum_r - 1e-9:
                    # Shrink proportionally to resolve overlap
                    factor = dist / sum_r
                    radii[i] *= factor
                    radii[j] *= factor
                    violation = True
        if not violation:
            break

    total_sum = np.sum(radii)
    return centers, radii, total_sum
