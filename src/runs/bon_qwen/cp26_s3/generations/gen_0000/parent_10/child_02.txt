import numpy as np
from scipy.optimize import minimize
import random

def run_packing():
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    Returns (centers, radii, sum_radii).
    """
    n = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None

    # Helper to evaluate and validate a solution
    def evaluate_solution(vars_array):
        radii = vars_array[2::3]
        sum_r = np.sum(radii)
        # Basic check for negative radii
        if np.any(radii < 0):
            return -1e9
        return sum_r

    # Objective function: Minimize negative sum of radii
    def objective(vars_array):
        return -np.sum(vars_array[2::3])

    # Constraint function for SLSQP
    # Returns an array of values that must be >= 0
    def constraints(vars_array):
        x = vars_array[0::3]
        y = vars_array[1::3]
        r = vars_array[2::3]
        
        cons = []
        
        # Boundary constraints: x - r >= 0, y - r >= 0, 1 - x - r >= 0, 1 - y - r >= 0
        cons.append(x - r)
        cons.append(y - r)
        cons.append(1 - x - r)
        cons.append(1 - y - r)
        
        # Overlap constraints: dist^2 >= (r_i + r_j)^2
        # dist^2 - (r_i + r_j)^2 >= 0
        for i in range(n):
            for j in range(i + 1, n):
                dx = x[i] - x[j]
                dy = y[i] - y[j]
                dist_sq = dx*dx + dy*dy
                r_sum = r[i] + r[j]
                cons.append(dist_sq - r_sum * r_sum)
        
        return np.concatenate(cons)

    # Variable bounds
    # x, y in [0, 1], r in [0, 0.5]
    bounds = [(0, 1)] * n + [(0, 1)] * n + [(0, 0.5)] * n

    # Initial guess generation function
    def get_initial_guess(method='hexagonal'):
        x_init = np.zeros(n)
        y_init = np.zeros(n)
        r_init = np.full(n, 0.08) # Small initial radius
        
        if method == 'hexagonal':
            # Generate hexagonal lattice points
            # Spacing such that we can fit roughly 26 points
            # sqrt(3)/2 * spacing approx row height
            # 2 * spacing approx col width
            # Try spacing ~ 0.2
            spacing = 0.22
            points = []
            # Rows
            for row in range(8):
                y = 0.1 + row * spacing * np.sqrt(3) / 2
                if y + r_init[0] > 1: break
                offset = (row % 2) * spacing / 2
                for col in range(8):
                    x = 0.1 + col * spacing + offset
                    if x + r_init[0] > 1: break
                    points.append((x, y))
            
            # If we don't have enough points, fill with random
            if len(points) < n:
                while len(points) < n:
                    points.append((random.uniform(0.1, 0.9), random.uniform(0.1, 0.9)))
            
            # Select first n points (or shuffle if we have more)
            # Actually, just take first n to keep structure
            selected = points[:n]
            
        elif method == 'random':
            selected = [(random.uniform(0.1, 0.9), random.uniform(0.1, 0.9)) for _ in range(n)]
            
        for i in range(n):
            x_init[i] = selected[i][0]
            y_init[i] = selected[i][1]
            
        return np.concatenate([x_init, y_init, r_init])

    # Optimization settings
    # SLSQP is good for constrained non-linear problems
    options = {'ftol': 1e-8, 'maxiter': 500, 'disp': False}

    # Try multiple starts
    methods_to_try = ['hexagonal', 'hexagonal', 'random', 'random', 'hexagonal']
    
    # Add some jitter to hexagonal starts to explore nearby local optima
    starts = []
    for _ in range(3):
        starts.append(get_initial_guess('hexagonal'))
    
    for _ in range(2):
        starts.append(get_initial_guess('random'))

    for i, x0 in enumerate(starts):
        try:
            # Add small noise to avoid getting stuck in exact symmetric points if they are saddle points
            if i == 0:
                noise = np.zeros_like(x0)
            else:
                noise = np.random.normal(0, 0.005, size=x0.shape)
            
            x_start = x0 + noise
            # Clip radii to be positive
            x_start[2::3] = np.clip(x_start[2::3], 0.01, 0.5)

            result = minimize(
                objective,
                x_start,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'ineq', 'fun': constraints},
                options=options
            )

            if result.success:
                current_sum = -result.fun
                if current_sum > best_sum:
                    # Validate constraints manually with tolerance
                    r = result.x[2::3]
                    x = result.x[0::3]
                    y = result.x[1::3]
                    
                    valid = True
                    if np.any(r < -1e-9):
                        valid = False
                    else:
                        # Check boundaries
                        for k in range(n):
                            if x[k] - r[k] < -1e-9 or x[k] + r[k] > 1 + 1e-9:
                                valid = False; break
                            if y[k] - r[k] < -1e-9 or y[k] + r[k] > 1 + 1e-9:
                                valid = False; break
                        # Check overlaps
                        if valid:
                            for k1 in range(n):
                                for k2 in range(k1 + 1, n):
                                    dist = np.sqrt((x[k1]-x[k2])**2 + (y[k1]-y[k2])**2)
                                    if dist < r[k1] + r[k2] - 1e-9:
                                        valid = False; break
                                if not valid: break
                    
                    if valid:
                        best_sum = current_sum
                        best_centers = np.column_stack((x, y))
                        best_radii = r
                        # print(f"New best found: sum={best_sum:.5f}")

        except Exception as e:
            # print(f"Optimization failed: {e}")
            pass

    # Fallback if nothing found (should not happen)
    if best_centers is None:
        # Return a safe default, e.g., small circles in grid
        centers = np.zeros((n, 2))
        radii = np.zeros(n)
        idx = 0
        for r_row in range(5):
            for c_col in range(5):
                if idx < n:
                    centers[idx] = [0.1 + c_col * 0.2, 0.1 + r_row * 0.2]
                    radii[idx] = 0.09 # slightly smaller to be safe
                    idx += 1
        if idx < n:
             centers[idx] = [0.5, 0.5]
             radii[idx] = 0.01
             idx += 1
        best_centers = centers
        best_radii = radii
        best_sum = np.sum(radii)

    return best_centers, best_radii, float(best_sum)
