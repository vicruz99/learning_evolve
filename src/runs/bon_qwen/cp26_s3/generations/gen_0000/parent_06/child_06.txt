import numpy as np
from scipy.optimize import minimize

N = 26

def compute_constraints(v):
    """Compute all inequality constraint values (must be >= 0)"""
    m = 4 * N + N * (N - 1) // 2
    c = np.empty(m)
    idx = 0
    for i in range(N):
        xi = v[3 * i]
        yi = v[3 * i + 1]
        ri = v[3 * i + 2]
        c[idx] = xi - ri
        c[idx + 1] = 1.0 - xi - ri
        c[idx + 2] = yi - ri
        c[idx + 3] = 1.0 - yi - ri
        idx += 4
    for i in range(N):
        xi = v[3 * i]
        yi = v[3 * i + 1]
        ri = v[3 * i + 2]
        for j in range(i + 1, N):
            xj = v[3 * j]
            yj = v[3 * j + 1]
            rj = v[3 * j + 2]
            dist_sq = (xi - xj) ** 2 + (yi - yj) ** 2
            c[idx] = dist_sq - (ri + rj) ** 2
            idx += 1
    return c

def constraint_jac(v):
    """Compute the Jacobian matrix of the constraints"""
    m = 4 * N + N * (N - 1) // 2
    jac = np.zeros((m, 3 * N))
    idx = 0
    for i in range(N):
        # x - r >= 0
        jac[idx, 3 * i] = 1.0
        jac[idx, 3 * i + 2] = -1.0
        idx += 1
        # 1 - x - r >= 0
        jac[idx, 3 * i] = -1.0
        jac[idx, 3 * i + 2] = -1.0
        idx += 1
        # y - r >= 0
        jac[idx, 3 * i + 1] = 1.0
        jac[idx, 3 * i + 2] = -1.0
        idx += 1
        # 1 - y - r >= 0
        jac[idx, 3 * i + 1] = -1.0
        jac[idx, 3 * i + 2] = -1.0
        idx += 1
        
    for i in range(N):
        xi = v[3 * i]
        yi = v[3 * i + 1]
        ri = v[3 * i + 2]
        for j in range(i + 1, N):
            xj = v[3 * j]
            yj = v[3 * j + 1]
            rj = v[3 * j + 2]
            dx = xi - xj
            dy = yi - yj
            sr = ri + rj
            jac[idx, 3 * i] = 2.0 * dx
            jac[idx, 3 * i + 1] = 2.0 * dy
            jac[idx, 3 * i + 2] = -2.0 * sr
            jac[idx, 3 * j] = -2.0 * dx
            jac[idx, 3 * j + 1] = -2.0 * dy
            jac[idx, 3 * j + 2] = -2.0 * sr
            idx += 1
    return jac

def constraint_fun(v):
    return compute_constraints(v)

def objective(v):
    # We maximize sum of radii by minimizing negative sum
    return -sum(v[3 * i + 2] for i in range(N))

def generate_initial_guess(n, seed=42):
    np.random.seed(seed)
    pts = []
    s = 0.18
    rows = 5
    cols_per_row = [6, 5, 6, 5, 4]
    for r in range(rows):
        nc = cols_per_row[r]
        for c in range(nc):
            x = c * s + (r % 2) * s / 2.0
            y = r * s * np.sqrt(3) / 2.0
            pts.append([x, y])
        if len(pts) >= n:
            break
            
    pts = np.array(pts[:n])
    min_c = np.min(pts, axis=0)
    max_c = np.max(pts, axis=0)
    span = max_c - min_c
    scale = 0.85 / np.max(span)
    centers = (pts - min_c) * scale + (1.0 - scale) / 2.0
    
    # Add slight random perturbation to break symmetry
    centers += np.random.uniform(-0.02, 0.02, size=centers.shape)
    centers = np.clip(centers, 0.1, 0.9)
    
    radii = np.full(n, 0.04)
    x0 = np.zeros(3 * n)
    for i in range(n):
        x0[3 * i] = centers[i, 0]
        x0[3 * i + 1] = centers[i, 1]
        x0[3 * i + 2] = radii[i]
    return x0

def run_packing():
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Try multiple seeds to find the best local optimum
    for seed in [42, 123, 456, 789, 1000]:
        x0 = generate_initial_guess(N, seed)
        bounds = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
        cons = {'type': 'ineq', 'fun': constraint_fun, 'jac': constraint_jac}
        
        res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'ftol': 1e-10, 'maxiter': 5000, 'disp': False})
                       
        if res.success:
            centers = np.array([[res.x[3 * i], res.x[3 * i + 1]] for i in range(N)])
            radii = np.array([res.x[3 * i + 2] for i in range(N)])
            radii = np.clip(radii, 0.0, None)
            s = np.sum(radii)
            if s > best_sum:
                best_sum = s
                best_centers = centers
                best_radii = radii
                
    return best_centers, best_radii, best_sum
