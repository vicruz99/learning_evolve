import numpy as np
import scipy.optimize as opt

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        return False
    if np.isnan(radii).any():
        return False

    # Check if radii are nonnegative
    for i in range(n):
        if radii[i] < 0:
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False

    return True

def hexagonal_init(n):
    """Generates a hexagonal packing initialization for n circles."""
    # Determine grid dimensions that are roughly square to fit in [0,1]
    # We look for k, m such that k*m >= n
    # Hexagonal rows usually have alternating counts.
    
    # Try to fit in a square bounding box of side ~4 in "unit distance" space
    # But let's just try to generate a set of points.
    
    # Simple approach: Try different row counts
    best_points = None
    best_score = float('inf')

    max_k = int(np.ceil(np.sqrt(2 * n))) + 2
    
    for k in range(1, max_k):

        # Row 1: k, Row 2: k-1 (or k), Row 3: k...

        # Rows with k, k-1, k, k-1...

def objective_and_gradient(params, n):
    """
    Computes the negative sum of radii (to minimize) and the penalty for violations.
    params shape: (3*n,) -> [x1, y1, r1, x2, y2, r2, ...]
    """
    centers = params[0::3].reshape(n, 2)
    radii = params[2::3] # r is at index 2, 5, 8... wait.
    # Let's map indices carefully.
    # params indices: 0:x, 1:y, 2:r, 3:x, 4:y, 5:r ...
    # centers: params[0::3] -> [x1, x2, ...], params[1::3] -> [y1, y2, ...]
    
    xs = params[0::3]
    ys = params[1::3]
    rs = params[2::3]
    
    centers = np.vstack((xs, ys)).T
    
    # Objective: Maximize sum(r) -> Minimize -sum(r)
    obj = -np.sum(rs)
    
    # Penalty terms
    penalty = 0.0
    
    # 1. Boundary constraints: x-r >= 0, x+r <= 1, y-r >= 0, y+r <= 1
    # Violations:
    # x - r < 0  => penalty += (x - r)^2
    # x + r > 1  => penalty += (x + r - 1)^2
    # ...
    
    # Vectorized boundary penalties
    # x - r
    p1 = np.maximum(0, rs - xs)
    # 1 - (x + r) < 0 => x + r - 1 > 0
    p2 = np.maximum(0, xs + rs - 1)
    # y - r
    p3 = np.maximum(0, rs - ys)
    # 1 - (y + r)
    p4 = np.maximum(0, ys + rs - 1)
    
    boundary_penalty = np.sum(p1**2) + np.sum(p2**2) + np.sum(p3**2) + np.sum(p4**2)
    
    # 2. Overlap constraints: dist(i,j) >= r_i + r_j
    # dist^2 >= (r_i + r_j)^2
    # If violated, penalty += (r_i + r_j - dist)^2
    
    overlap_penalty = 0.0
    # Compute all pairwise distances
    # Efficient computation
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.linalg.norm(diff, axis=2)
    
    # Lower triangle indices
    i, j = np.tril_indices(n, -1)
    
    radii_sum = rs[i] + rs[j]
    dists_subset = dists[i, j]
    
    violations = np.maximum(0, radii_sum - dists_subset)
    overlap_penalty = np.sum(violations**2)
    
    # Total cost
    # We use a multiplier for penalty to prioritize feasibility, 
    # but since we want to maximize radii, we need a balance.
    # A simple approach is to minimize -sum(r) + lambda * (boundary_penalty + overlap_penalty)
    # If lambda is too high, it restricts r growth.
    # However, standard approach in these problems is to make penalty heavy enough 
    # that the solver tries to satisfy it.
    
    cost = obj + 100.0 * (boundary_penalty + overlap_penalty)
    
    return cost
