import numpy as np
from scipy.optimize import minimize

def hex_grid_init(n, r):
    """Generate initial positions using a hexagonal lattice pattern."""
    points = []
    row = 0
    while len(points) < n:
        y = r + row * r * np.sqrt(3)
        if y > 1 - r:
            break
        col = 0
        while True:
            x = r + col * 2 * r + (row % 2) * r
            if x > 1 - r:
                break
            points.append([x, y])
            col += 1
            if len(points) >= n:
                break
        row += 1
    # Fill remaining points randomly if grid doesn't yield enough (rare for small r)
    while len(points) < n:
        points.append([np.random.uniform(r, 1 - r), np.random.uniform(r, 1 - r)])
    return np.array(points[:n])

def objective(params, n, r):
    """Penalty function for constraint violations."""
    c = params.reshape((n, 2))
    val = 0.0
    # Boundary violations
    for i in range(n):
        x, y = c[i]
        if x < r: val += (r - x)**2
        if x > 1 - r: val += (x - (1 - r))**2
        if y < r: val += (r - y)**2
        if y > 1 - r: val += (y - (1 - r))**2
    # Overlap violations
    for i in range(n):
        for j in range(i + 1, n):
            dx = c[i, 0] - c[j, 0]
            dy = c[i, 1] - c[j, 1]
            d = np.sqrt(dx*dx + dy*dy)
            if d < 2 * r:
                val += (2 * r - d)**2
    return val

def find_max_r(n):
    """Binary search for the maximum feasible equal radius."""
    np.random.seed(42)
    lo, hi = 0.05, 0.15
    best_c = None
    while hi - lo > 1e-6:
        mid = (lo + hi) / 2
        init_c = hex_grid_init(n, mid)
        init_c = np.clip(init_c, mid, 1 - mid)
        bounds = [(mid, 1 - mid) for _ in range(2 * n)]
        res = minimize(objective, init_c.flatten(), args=(n, mid), method='L-BFGS-B', bounds=bounds, options={'maxiter': 2000, 'ftol': 1e-12})
        if res.fun < 1e-4:
            lo = mid
            best_c = res.x.reshape((n, 2))
        else:
            hi = mid
    return lo, best_c

def run_packing():
    n = 26
    r_opt, centers = find_max_r(n)
    radii = np.full(n, r_opt)
    # Slight reduction to guarantee strict feasibility against numerical tolerance
    radii *= 0.9999
    return centers, radii, np.sum(radii)
