import numpy as np

import math

def calculate_loss_and_grad(params, n, penalty_weight):
    """
    Calculates the loss and gradient for the packing optimization.
    Loss = -Sum(radii) + penalty_weight * (overlap_penalties + boundary_penalties)
    """
    centers = params[:2*n].reshape(n, 2)
    radii = params[2*n:]
    
    # Objective: Maximize sum of radii (minimize negative sum)
    obj = -np.sum(radii)
    
    # Penalty for boundaries
    # Constraints: r <= x <= 1-r  =>  x - r >= 0, 1 - x - r >= 0
    # Similarly for y
    # Violation: max(0, r - x), max(0, r + x - 1)
    
    x = centers[:, 0]
    y = centers[:, 1]
    r = radii
    
    boundary_violations = np.maximum(0, r - x) ** 2 + \
                          np.maximum(0, r + x - 1) ** 2 + \
                          np.maximum(0, r - y) ** 2 + \
                          np.maximum(0, r + y - 1) ** 2
    
    boundary_penalty = np.sum(boundary_violations)
    
    # Penalty for overlaps
    # Constraint: dist >= r_i + r_j
    # Violation: max(0, r_i + r_j - dist)^2
    
    overlap_penalty = 0.0
    # Vectorized distance calculation can be heavy for gradient, 
    # but for n=26 it's manageable. We compute scalar penalty for objective.
    # Gradient computation will be done numerically or approximated if needed, 
    # but scipy can do finite differences. For speed, we return just the value 
    # and let scipy compute gradients, OR we implement analytic gradients.
    # Given the complexity of analytic gradients for the max function, 
    # let's rely on scipy's numerical gradient or finite difference, 
    # but providing the value is enough for minimize if method supports it.
    # However, 'L-BFGS-B' requires gradient or uses finite diff.
    # Let's compute overlap penalty value.
    
    # To speed up, we can check only pairs with potential overlap, 
    # but O(N^2) is small for N=26.
    
    for i in range(n):
        xi, yi, ri = centers[i, 0], centers[i, 1], r[i]
        for j in range(i + 1, n):
            xj, yj, rj = centers[j, 0], centers[j, 1], r[j]
            dist = math.sqrt((xi - xj)**2 + (yi - yj)**2)
            min_dist = ri + rj
            if dist < min_dist:
                overlap_penalty += (min_dist - dist)**2
                
    loss = obj + penalty_weight * (boundary_penalty + overlap_penalty)
    return loss
