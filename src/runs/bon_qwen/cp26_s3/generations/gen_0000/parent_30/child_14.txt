from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    Returns:
        centers: np.array of shape (26, 2)
        radii: np.array of shape (26,)
        sum_radii: float

    initial_radius = 0.09

        # Even rows: x starts at r, step 2r
        # Odd rows: x starts at 2r, step 2r

            x_start = initial_radius
        else:
            x_start = 2 * initial_radius

            if len(centers_init) >= n_circles:
                break
            centers_init.append([x, y])
            x += 2 * initial_radius
        
        y += initial_radius * np.sqrt(3)
        row_idx += 1
    
    # Trim to exactly 26 circles
    centers_init = np.array(centers_init[:n_circles])
    radii_init = np.full(n_circles, initial_radius)
    
    # Flatten variables for optimization: [x1, y1, r1, x2, y2, r2, ...]

        r_sum = 0

        return -r_sum

                'type': 'ineq',
                'fun': lambda v, i=i: 1 - v[idx_x] - v[idx_r]
            })

                'type': 'ineq',
                'fun': lambda v, i=i: v[idx_y] - v[idx_r]
            })
            # y <= 1 - r => 1 - y - r >= 0

                'type': 'ineq',
                'fun': lambda v, i=i: 1 - v[idx_y] - v[idx_r]

    # No, constraint must be >= 0.
