import numpy as np
from scipy.optimize import minimize

def compute_penalty(x, n):
    """Compute penalty for constraint violations."""
    centers = x[:2 * n].reshape(n, 2)
    radii = x[2 * n:]
    
    penalty = 0.0
    
    # Circle-circle overlap penalty
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.hypot(dx, dy)
            min_dist = radii[i] + radii[j]
            overlap = min_dist - dist
            if overlap > 0:
                penalty += overlap ** 2
    
    # Boundary violation penalty
    for i in range(n):
        xi = centers[i, 0]
        yi = centers[i, 1]
        ri = radii[i]
        
        if xi - ri < 0:
            penalty += (xi - ri) ** 2
        if xi + ri > 1:
            penalty += (xi + ri - 1) ** 2
        if yi - ri < 0:
            penalty += (yi - ri) ** 2
        if yi + ri > 1:
            penalty += (yi + ri - 1) ** 2
    
    return penalty


def compute_max_violation(x, n):
    """Compute the maximum constraint violation."""
    centers = x[:2 * n].reshape(n, 2)
    radii = x[2 * n:]
    
    max_viol = 0.0
    
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.hypot(dx, dy)
            min_dist = radii[i] + radii[j]
            overlap = min_dist - dist
            if overlap > max_viol:
                max_viol = overlap
    
    for i in range(n):
        xi = centers[i, 0]
        yi = centers[i, 1]
        ri = radii[i]
        
        violations = [ri - xi, xi + ri - 1, ri - yi, yi + ri - 1]
        for v in violations:
            if v > max_viol:
                max_viol = v
    
    return max_viol


class PackingObjective:
    """Objective function for circle packing optimization."""
    
    def __init__(self, n, penalty_weight):
        self.n = n
        self.penalty_weight = penalty_weight
    
    def __call__(self, x):
        radii = x[2 * self.n:]
        return -np.sum(radii) + self.penalty_weight * compute_penalty(x, self.n)


def make_hexagonal_init(n, margin=0.04):
    """Create hexagonal grid initialization for circle centers."""
    centers = np.zeros((n, 2))
    idx = 0
    
    cols = 6
    rows = 5
    
    spacing_x = (1 - 2 * margin) / (cols - 0.5)
    spacing_y = spacing_x * np.sqrt(3) / 2
    
    for row in range(rows):
        for col in range(cols):
            if idx >= n:
                break
            x = margin + col * spacing_x + (row % 2) * spacing_x / 2
            y = margin + row * spacing_y
            centers[idx] = [x, y]
            idx += 1
    
    return centers


def make_random_init(n, margin=0.05):
    """Create random initialization for circle centers."""
    centers = np.random.uniform(margin, 1 - margin, size=(n, 2))
    return centers


def make_grid_init(n):
    """Create uniform grid initialization."""
    centers = np.zeros((n, 2))
    idx = 0
    rows = 6
    cols = 5
    for row in range(rows):
        for col in range(cols):
            if idx >= n:
                break
            centers[idx] = [0.1 + col * 0.18, 0.1 + row * 0.15]
            idx += 1
    return centers


def force_relaxation(centers, radii, n, steps=3000, lr=0.02):
    """Apply force-directed relaxation to reduce overlaps."""
    for step in range(steps):
        forces = np.zeros((n, 2))
        current_lr = lr * (1 - step / steps) ** 2
        
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.hypot(dx, dy)
                min_dist = radii[i] + radii[j]
                
                if dist < min_dist and dist > 1e-10:
                    force_mag = (min_dist - dist) / dist
                    fx = dx * force_mag
                    fy = dy * force_mag
                    forces[i, 0] += fx
                    forces[i, 1] += fy
                    forces[j, 0] -= fx
                    forces[j, 1] -= fy
            
            for d in range(2):
                if centers[i, d] - radii[i] < 0:
                    forces[i, d] += (radii[i] - centers[i, d]) * 5
                if centers[i, d] + radii[i] > 1:
                    forces[i, d] -= (centers[i, d] + radii[i] - 1) * 5
        
        centers = centers + forces * current_lr
    
    return centers


def optimize_packing(x0, n, bounds, penalty_weights):
    """Sequential optimization with increasing penalty weights."""
    x = x0.copy()
    
    for pw in penalty_weights:
        obj = PackingObjective(n, pw)
        result = minimize(
            obj,
            x,
            method='L-BFGS-B',
            bounds=bounds,
            options={
                'maxiter': 30000,
                'ftol': 1e-16,
                'gtol': 1e-12
            }
        )
        x = result.x
    
    return x
