Strategy:
1. Formulate the problem as maximizing $\sum r_i$ subject to $(x_i-x_j)^2 + (y_i-y_j)^2 \ge (r_i+r_j)^2$ and $r_i \le x_i, y_i \le 1-r_i$.
2. Use `scipy.optimize.minimize` with the SLSQP algorithm, which handles nonlinear constraints efficiently.
3. Initialize with a structured 5x5 grid plus one center circle, all with radius 0.08. This is a valid, dense packing close to the optimum.
4. Run the optimizer 5 times with slightly perturbed initial conditions to escape local minima and increase the chance of finding the global optimum (target ~2.636).

import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def objective(vars):
    """Objective function: maximize sum of radii (minimize negative sum)."""
    return -np.sum(vars[2::3])

def constraints(vars):
    """Constraint function: boundary and non-overlap conditions."""
    n = N_CIRCLES
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    c_list = []
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    c_list.extend(x - r)
    c_list.extend(1 - x - r)
    c_list.extend(y - r)
    c_list.extend(1 - y - r)
    # Non-overlap constraints: dist^2 >= (r_i + r_j)^2
    for i in range(n):
        for j in range(i + 1, n):
            dx = x[i] - x[j]
            dy = y[i] - y[j]
            c_list.append(dx*dx + dy*dy - (r[i] + r[j])**2)
    return np.array(c_list)

def get_initial_guess():
    """Generate a valid initial configuration (5x5 grid + center)."""
    n = N_CIRCLES
    centers = []
    for i in range(5):
        for j in range(5):
            centers.append([0.1 + i * 0.2, 0.1 + j * 0.2])
    centers.append([0.5, 0.5])
    centers = np.array(centers)
    radii = np.full(n, 0.08)
    return np.concatenate([centers, radii]).flatten()

def run_packing():
    n = N_CIRCLES
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * n
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_vars = None
    best_sum = -1.0
    
    # Run multiple trials to improve robustness against local minima
    for trial in range(5):
        vars0 = get_initial_guess()
        if trial > 0:
            noise = np.random.randn(len(vars0)) * 0.015
            vars0 = np.clip(vars0 + noise, 0.02, 0.98)
            vars0[2::3] = np.clip(vars0[2::3], 0.02, 0.45)
            
        try:
            res = minimize(objective, vars0, method='SLSQP', bounds=bounds, 
                           constraints=cons, options={'maxiter': 1500, 'ftol': 1e-12})
            if -res.fun > best_sum:
                best_sum = -res.fun
                best_vars = res.x
        except Exception:
            pass

    if best_vars is None:
        best_vars = get_initial_guess()
        
    centers = best_vars[:-n].reshape(n, 2)
    radii = best_vars[-n:]
    
    # Post-processing: ensure strict validity within tolerance
    for _ in range(100):
        valid = True
        for i in range(n):
            if (centers[i][0] - radii[i] < -1e-9 or centers[i][0] + radii[i] > 1 + 1e-9 or 
                centers[i][1] - radii[i] < -1e-9 or centers[i][1] + radii[i] > 1 + 1e-9):
                valid = False
                break
        if valid:
            for i in range(n):
                for j in range(i + 1, n):
                    dist = np.sqrt((centers[i][0] - centers[j][0])**2 + (centers[i][1] - centers[j][1])**2)
                    if dist < radii[i] + radii[j] - 1e-9:
                        valid = False
                        break
                if not valid:
                    break
                    
        if not valid:
            radii *= 0.998
            centers = np.clip(centers, radii.reshape(-1,1), 1 - radii.reshape(-1,1))
        else:
            break
            
    return centers, radii, np.sum(radii)
