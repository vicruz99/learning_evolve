import numpy as np
from scipy.optimize import minimize


def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    best_radii = None
    best_sum = 0.0

def initialize_hexagonal(n, seed):
    """Initialize circles in a hexagonal pattern with slight random perturbation."""
    rng = np.random.RandomState(seed)
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.085)

    idx = 0
    row = 0
    base_spacing_x = 0.175
    base_spacing_y = 0.152

    while idx < n:
        if row % 2 == 0:
            ncols = 5
            offset_x = 0.0
        else:
            ncols = 5
            offset_x = base_spacing_x * 0.5

        for col in range(ncols):
            if idx >= n:
                break
            x = 0.09 + col * base_spacing_x + offset_x
            y = 0.09 + row * base_spacing_y
            x += rng.uniform(-0.005, 0.005)
            y += rng.uniform(-0.005, 0.005)
            centers[idx, 0] = np.clip(x, 0.05, 0.95)
            centers[idx, 1] = np.clip(y, 0.05, 0.95)
            idx += 1
        row += 1

    return centers, radii


def objective_with_penalty(x, n, penalty_weight):
    """Objective function: maximize sum of radii with penalties for constraint violations."""
    centers = x[:2 * n].reshape(n, 2)
    radii = x[2 * n:]

    obj = -np.sum(radii)
    penalty = 0.0

    # Boundary penalties (smooth barrier)
    for i in range(n):
        for c in range(2):
            violation_low = centers[i, c] - radii[i]
            if violation_low < 0:
                penalty += penalty_weight * violation_low ** 2
            violation_high = 1.0 - centers[i, c] - radii[i]
            if violation_high < 0:
                penalty += penalty_weight * violation_high ** 2

    # Overlap penalties
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            min_dist = radii[i] + radii[j]
            if dist < min_dist:
                overlap = min_dist - dist
                penalty += penalty_weight * overlap ** 2

    return obj + penalty


def optimize_packing(centers, radii, n):
    """Optimize packing using L-BFGS-B with penalty method."""
    x0 = np.concatenate([centers.flatten(), radii])

    # Bounds for variables
    bounds = []
    for i in range(2 * n):
        bounds.append((0.001, 0.999))
    for i in range(n):
        bounds.append((0.001, 0.5))

    # Multi-stage optimization with increasing penalty
    x = x0.copy()
    penalty_weights = [500, 2000, 10000, 50000]

    for pw in penalty_weights:
        def obj(x_inner):
            return objective_with_penalty(x_inner, n, pw)

        result = minimize(
            obj, x, method='L-BFGS-B', bounds=bounds,
            options={
                'maxiter': 3000,
                'ftol': 1e-14,
                'gtol': 1e-10
            }
        )
        x = result.x

    centers_opt = x[:2 * n].reshape(n, 2)
    radii_opt = x[2 * n:]

    return centers_opt, radii_opt


def project_to_feasible(centers, radii, n):
    """Project solution to feasible region."""
    centers = centers.copy()
    radii = radii.copy()

    # Enforce boundary constraints
    for i in range(n):
        for c in range(2):
            if centers[i, c] - radii[i] < 0:
                centers[i, c] = radii[i]
            if centers[i, c] + radii[i] > 1.0:
                centers[i, c] = 1.0 - radii[i]

    # Iteratively resolve overlaps
    for iteration in range(500):
        max_overlap = 0.0
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                if dist < radii[i] + radii[j]:
                    overlap = radii[i] + radii[j] - dist
                    max_overlap = max(max_overlap, overlap)
                    # Reduce both radii equally
                    reduction = overlap / 2 + 1e-10
                    radii[i] = max(radii[i] - reduction, 0.001)
                    radii[j] = max(radii[j] - reduction, 0.001)

        if max_overlap < 1e-12:
            break

    # Final boundary check
    for i in range(n):
        for c in range(2):
            if centers[i, c] - radii[i] < 0:
                centers[i, c] = radii[i]
            if centers[i, c] + radii[i] > 1.0:
                centers[i, c] = 1.0 - radii[i]

    return centers, radii


def validate_internal(centers, radii, n):
    """Internal validation check."""
    for i in range(n):
        if radii[i] < 0:
            return False
        for c in range(2):
            if centers[i, c] - radii[i] < -1e-10:
                return False
            if centers[i, c] + radii[i] > 1.0 + 1e-10:
                return False

    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            if dist < radii[i] + radii[j] - 1e-10:
                return False

    return True
