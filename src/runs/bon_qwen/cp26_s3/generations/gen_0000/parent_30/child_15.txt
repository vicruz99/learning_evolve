import numpy as np
from scipy.optimize import minimize

def _loss(params, n, mu):
    """
    Computes the negative sum of radii plus penalty for constraint violations.
    """
    c = params[:2*n].reshape((n, 2))
    r = params[2*n:]
    
    # Objective: minimize negative sum of radii
    obj = -np.sum(r)
    p = 0.0
    
    # Boundary penalties: circle must be inside [0,1]x[0,1]
    for i in range(n):
        x, y = c[i]
        ri = r[i]
        gaps = [x - ri, 1 - x - ri, y - ri, 1 - y - ri]
        for gap in gaps:
            if gap < 0:
                p += gap**2
                
    # Overlap penalties: distance between centers must be >= sum of radii
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(c[i, 0] - c[j, 0], c[i, 1] - c[j, 1])
            gap = dist - r[i] - r[j]
            if gap < 0:
                p += gap**2
                
    return obj + mu * p

def run_packing():
    n = 26
    best_loss = np.inf
    best_params = None
    
    # Try multiple random initializations to escape local minima
    for seed in range(15):
        np.random.seed(seed)
        
        # Initialize centers in a staggered grid pattern
        centers = np.zeros((n, 2))
        radii = np.ones(n) * 0.095
        
        row_counts = [5, 6, 5, 6, 4]  # Sums to 26
        idx = 0
        for r_idx, count in enumerate(row_counts):
            y = (r_idx + 0.5) / 5.0
            for c_idx in range(count):
                x = (c_idx + 0.5) / (count + 0.5) + (r_idx % 2) * 0.08
                centers[idx] = [x, y]
                idx += 1
                
        # Add controlled noise
        centers += np.random.randn(*centers.shape) * 0.02
        centers = np.clip(centers, 0.05, 0.95)
        
        x0 = np.hstack([centers.flatten(), radii])
        bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
        
        # Progressive penalty optimization
        current_x = x0
        for mu in [500, 2000, 10000]:
            res = minimize(_loss, current_x, method='L-BFGS-B', 
                           bounds=bounds, args=(n, mu),
                           options={'maxiter': 1000, 'ftol': 1e-12})
            current_x = res.x
            
        # Evaluate with a high penalty to check feasibility quality
        final_loss = _loss(current_x, n, 100000)
        if final_loss < best_loss:
            best_loss = final_loss
            best_params = current_x.copy()
            
    if best_params is None:
        # Fallback to a valid but suboptimal configuration
        best_params = np.hstack([np.zeros((n, 2)), np.ones(n) * 0.05])
        
    final_centers = best_params[:2*n].reshape((n, 2))
    final_radii = best_params[2*n:]
    
    # Post-processing to guarantee strict validity under numerical tolerances
    # 1. Enforce boundary constraints
    for i in range(n):
        x, y = final_centers[i]
        max_r_boundary = min(x, 1 - x, y, 1 - y)
        if final_radii[i] > max_r_boundary:
            final_radii[i] = max_r_boundary
            
    # 2. Enforce non-overlap constraints by scaling down overlapping pairs
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(final_centers[i, 0] - final_centers[j, 0], 
                            final_centers[i, 1] - final_centers[j, 1])
            r_sum = final_radii[i] + final_radii[j]
            if r_sum > dist:
                scale = dist / r_sum
                final_radii[i] *= scale
                final_radii[j] *= scale
                
    return final_centers, final_radii, float(np.sum(final_radii))
