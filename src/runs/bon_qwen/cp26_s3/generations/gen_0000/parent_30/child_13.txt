from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    best_centers = np.zeros((n_circles, 2))
    best_radii = np.zeros(n_circles)
    best_sum = -1.0

    def objective(params):
        centers = params[:2 * n_circles].reshape(n_circles, 2)
        radii = params[2 * n_circles:]
        return -np.sum(radii)

    def constraints(params):
        centers = params[:2 * n_circles].reshape(n_circles, 2)
        radii = params[2 * n_circles:]
        cons = []

        # Boundary constraints
        for i in range(n_circles):
            cons.append(centers[i, 0] - radii[i])
            cons.append(1.0 - (centers[i, 0] + radii[i]))
            cons.append(centers[i, 1] - radii[i])
            cons.append(1.0 - (centers[i, 1] + radii[i]))
            cons.append(radii[i]) # Non-negative radius

        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                cons.append(dist - (radii[i] + radii[j]))

        return cons

    # 2. Random initialization
