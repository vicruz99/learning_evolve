import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    
    Returns:
        centers: np.array of shape (26, 2) with (x, y) coordinates
        radii: np.array of shape (26) with radius of each circle
        sum_radii: float sum of all radii
    """
    N = 26
    PENALTY_WEIGHT = 5000.0
    
    def compute_penalty(centers, radii):
        """Calculates the penalty for constraint violations."""
        penalty = 0.0
        
        # Boundary violations
        for i in range(N):
            x, y = centers[i]
            r = radii[i]
            if x - r < 0: penalty += (x - r)**2
            if x + r > 1: penalty += (x + r - 1)**2
            if y - r < 0: penalty += (y - r)**2
            if y + r > 1: penalty += (y + r - 1)**2
            
        # Overlap violations
        for i in range(N):
            for j in range(i + 1, N):
                dist_sq = np.sum((centers[i] - centers[j])**2)
                dist = np.sqrt(dist_sq)
                req_dist = radii[i] + radii[j]
                if dist < req_dist:
                    penalty += (req_dist - dist)**2
                    
        return penalty

    def objective(vars_flat):
        """Objective function to minimize: negative sum of radii + penalty."""
        centers = vars_flat[:2*N].reshape(N, 2)
        radii = vars_flat[2*N:]
        
        # Penalize negative radii heavily (though bounds handle this)
        if np.any(radii < -1e-6):
            return 1e9
            
        pen = compute_penalty(centers, radii)
        return -np.sum(radii) + PENALTY_WEIGHT * pen

    def generate_hex_grid_centers(n_points, perturbation_scale=0.0):
        """Generates initial centers using a hexagonal lattice pattern."""
        centers = []
        # Estimate radius for grid spacing
        r_est = 0.1 
        s = 2 * r_est

        y = r_est

            start_x = r_est + (s / 2 if row % 2 == 1 else 0)
            x = start_x

                centers.append([x, y])
                x += s
            y += v_dist
            row += 1

        while len(centers) < n_points:
            centers.append([np.random.rand(), np.random.rand()])
            
        centers = np.array(centers[:n_points])
        
        # Add perturbation
        if perturbation_scale > 0:
            centers += np.random.normal(0, perturbation_scale, centers.shape)
            centers = np.clip(centers, 0, 1)
            
        return centers

    restarts = 15

        init_centers = generate_hex_grid_centers(N, perturbation_scale=pert_scale)
        init_radii = np.ones(N) * 0.05
        x0 = np.concatenate([init_centers.flatten(), init_radii])
        
        res = minimize(objective, x0, method='L-BFGS-B', bounds=bounds, 
                       options={'ftol': 1e-10, 'maxiter': 2000, 'disp': False})

            # Actually, obj = -sum_r + penalty. So sum_r approx = -obj + penalty.
            # But if penalty is non-zero, the sum is not exactly recoverable from obj.
            # Let's compute sum directly.
            current_centers = res.x[:2*N].reshape(N, 2)
            current_radii = res.x[2*N:]
            current_sum = np.sum(current_radii)
            
            if current_sum > max_sum_radii:
                max_sum_radii = current_sum
                best_result = (current_centers.copy(), current_radii.copy())

    if best_result is None:
        # Fallback to simple grid
        init_centers = generate_hex_grid_centers(N, 0.0)
        init_radii = np.ones(N) * 0.01
        x0 = np.concatenate([init_centers.flatten(), init_radii])
        res = minimize(objective, x0, method='L-BFGS-B', bounds=bounds, options={'maxiter': 2000})
        best_result = (res.x[:2*N].reshape(N, 2), res.x[2*N:])

    centers, radii = best_result
    
    # Post-processing: Ensure strict validity by shrinking radii slightly if needed
    # Check overlaps and boundaries
    valid = True
    tolerance = 1e-7
    
    # Check boundaries
    for i in range(N):
        x, y = centers[i]
        r = radii[i]
        if x - r < -tolerance or x + r > 1 + tolerance or y - r < -tolerance or y + r > 1 + tolerance:

                    valid = False
                    break

                break

                    # We need r_i + r_j <= dist => r_i <= dist - r_j
                    # This is coupled. Simple iterative reduction.
                    pass

            for i in range(N):
                r_i = radii[i]
