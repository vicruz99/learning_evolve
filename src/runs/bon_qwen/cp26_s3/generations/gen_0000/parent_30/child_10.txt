import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    
    Returns:
        centers: (26, 2) array of (x, y) coordinates.
        radii: (26,) array of radii.
        sum_radii: float, the sum of all radii.
    """
    N = 26
    
    # 1. Initialization: Hexagonal Grid
    # We arrange circles in a staggered pattern (6, 5, 6, 5, 4) to fit 26 circles.
    centers = []
    radii = []
    
    # Define grid parameters
    # We aim to cover the space [0, 1] x [0, 1]. 
    # We leave a margin initially to allow the optimizer to expand.
    y_vals = np.linspace(0.15, 0.85, 5)
    x_vals_full = np.linspace(0.15, 0.85, 6)
    shift = (x_vals_full[1] - x_vals_full[0]) / 2.0
    
    idx = 0
    for r_idx in range(5):
        y = y_vals[r_idx]
        
        # Determine x values for this row
        if r_idx % 2 == 0:
            # Full row of 6 points
            xs = x_vals_full
        else:
            # Staggered row of 5 points (shifted)
            xs = x_vals_full[1:-1]
            
        for x in xs:
            if idx < N:
                centers.append([x, y])
                radii.append(0.05) # Start with small radius
                idx += 1
            else:
                break
        if idx >= N:
            break
            
    centers = np.array(centers)
    radii = np.array(radii)
    
    # If grid logic didn't fill exactly 26 (unlikely with 6+5+6+5+4=26), pad
    if len(centers) < N:
        # Fallback to random placement for missing circles
        missing = N - len(centers)
        extra_centers = np.random.uniform(0.1, 0.9, (missing, 2))
        extra_radii = np.full(missing, 0.05)
        centers = np.vstack([centers, extra_centers])
        radii = np.concatenate([radii, extra_radii])

    # 2. Variable Setup
    # Variables vector: [x0, y0, r0, x1, y1, r1, ..., x25, y25, r25]
    x0 = np.zeros(3 * N)
    for i in range(N):
        x0[3*i] = centers[i, 0]
        x0[3*i+1] = centers[i, 1]
        x0[3*i+2] = radii[i]
    
    # Bounds: x, y in [0, 1], r in [0, 0.5]
    bounds = []
    for _ in range(N):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 0.5)) # r
    
    # 3. Objective Function
    # We want to maximize sum(r), so we minimize -sum(r)
    def objective(v):
        # Radii are at indices 2, 5, 8, ...
        return -np.sum(v[2::3])

    # 4. Vectorized Constraint Function
    # Returns an array of constraint values (all must be >= 0)
    def constraints_func(v):
        # Extract variables
        cx = v[0::3]
        cy = v[1::3]
        cr = v[2::3]
        
        # Boundary Constraints (4 per circle)
        # 1. x - r >= 0
        c1 = cx - cr
        # 2. 1 - x - r >= 0
        c2 = 1.0 - cx - cr
        # 3. y - r >= 0
        c3 = cy - cr
        # 4. 1 - y - r >= 0
        c4 = 1.0 - cy - cr
        
        # Overlap Constraints
        # (xi - xj)^2 + (yi - yj)^2 >= (ri + rj)^2
        # We compute this for all pairs i < j
        
        # Broadcasting differences
        dx = cx[:, None] - cx[None, :]
        dy = cy[:, None] - cy[None, :]
        dr = cr[:, None] + cr[None, :]
        
        dist_sq = dx**2 + dy**2
        rad_sum_sq = dr**2
        
        # We only need upper triangle (i < j) to avoid duplicates and self-comparison
        # Create a mask for i < j
        mask = np.triu(np.ones((N, N), dtype=bool), k=1)
        
        overlap_constraints = dist_sq[mask] - rad_sum_sq[mask]
        
        # Concatenate all constraints
        return np.concatenate([c1, c2, c3, c4, overlap_constraints])

    # 5. Optimization
    # Using SLSQP for handling constraints
    cons = {'type': 'ineq', 'fun': constraints_func}
    
    # Options for convergence
    options = {
        'maxiter': 2000,
        'ftol': 1e-10,
        'disp': False
    }
    
    res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, options=options)
    
    # 6. Extract and Clean Results
    final_centers = res.x[:2*N].reshape(N, 2)
    final_radii = res.x[2*N:]
    
    # Ensure radii are non-negative (numerical safety)
    final_radii = np.maximum(final_radii, 0.0)
    
    # Clamp centers to [0, 1] just in case, though bounds should handle it
    final_centers = np.clip(final_centers, 0.0, 1.0)
    
    sum_radii = np.sum(final_radii)
    
    return final_centers, final_radii, float(sum_radii)
