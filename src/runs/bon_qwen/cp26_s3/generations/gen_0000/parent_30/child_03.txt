    centers = np.random.rand(n, 2) * 0.8 + 0.1 # Keep away from edges initially

    # We will iteratively update positions and radii.
    # Objective: Maximize sum(radii)
    # Constraints: No overlap, inside [0,1]x[0,1]
    
    # Parameters
    steps = 2000
    dt = 0.01
    repulsion_strength = 1.0
    expansion_rate = 0.0005 # How fast radii grow
    boundary_force = 10.0   # Force to keep inside

    pos = centers
    rad = radii
    
    for step in range(steps):
        forces = np.zeros_like(pos)

        # Optimization: use broadcasting for pairwise distances
        # pos shape (n, 2)
        # diff shape (n, n, 2)
        diff = pos[:, np.newaxis, :] - pos[np.newaxis, :, :]
        dist_sq = np.sum(diff**2, axis=2)
        dist_sq = np.maximum(dist_sq, 1e-9) # Avoid div by zero
        dist = np.sqrt(dist_sq)

        # To avoid O(n^2) loop in python, we do vector ops.

        # If overlap > 0, push apart. 
        # If overlap < 0, maybe a tiny attraction to keep structure? No, we want them apart.

        # Force on i from j: pushes i in direction -diff (away from j)

        # Usually defined as pushing apart.

        # force_magnitude shape (n, n)
        # dir_norm shape (n, n, 2)

        # Note: force_magnitude[i,i] is 0 (handled by max/overlap logic usually, dist=0)

        # Using np.einsum or simple multiplication
        # F[i] = sum_j ( mag[i,j] * dir_norm[i,j] )
        
        # Mask out self-interaction just in case
        np.fill_diagonal(force_magnitude, 0)
        
        # Calculate forces
        # force_vec shape (n, n, 2)
        force_vec = force_magnitude[:, :, np.newaxis] * dir_norm
        
        # Sum over axis 1 (j)
        net_forces = np.sum(force_vec, axis=1)
        
        # Apply forces
        pos += dt * net_forces

        for i in range(n):
            x, y = pos[i]
            r = rad[i]
            
            # X boundaries
            if x < r:
                pos[i, 0] = r
                net_forces[i, 0] += boundary_force * (r - x) # Additional push

                pos[i, 0] = 1 - r

                net_forces[i, 1] += boundary_force * (r - y)

        rad += expansion_rate
