def compute_violations_and_gradient(centers, radii):
    """
    Compute total squared violation of constraints.
    Constraints:
    1. Circles inside [0,1]x[0,1]: r <= x <= 1-r, r <= y <= 1-r
    2. No overlaps: dist(i,j) >= r_i + r_j
    """
    n = len(radii)
    penalty = 0.0
    
    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        
        # Left
        if x - r < 0: penalty += (x - r)**2
        # Right
        if x + r > 1: penalty += (x + r - 1)**2
        # Bottom
        if y - r < 0: penalty += (y - r)**2
        # Top
        if y + r > 1: penalty += (y + r - 1)**2
        
    # Pairwise constraints
    # Vectorized calculation for performance
    # centers shape (n, 2), radii shape (n)
    # dist matrix
    # We can compute upper triangle differences
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            r_sum = radii[i] + radii[j]
            if dist < r_sum:
                penalty += (r_sum - dist)**2
                
    return penalty

def objective_function(params, n):
    """
    Objective to maximize sum of radii.
    We minimize negative sum of radii + penalty.
    """
    centers = params[:2*n].reshape(n, 2)
    radii = params[2*n:]
    
    # Ensure radii are non-negative (clamped in bounds usually, but just in case)
    radii = np.maximum(radii, 0)
    
    sum_radii = np.sum(radii)
    penalty = compute_violations_and_gradient(centers, radii)

    # If violation > 0, we want to push it to 0.
    # A large multiplier ensures we find a feasible solution first.
    penalty_weight = 1e4
    
    return -sum_radii + penalty_weight * penalty
