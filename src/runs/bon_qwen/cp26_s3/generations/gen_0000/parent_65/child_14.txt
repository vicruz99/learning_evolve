import numpy as np
from scipy.optimize import minimize

def _compute_loss(params, N):
    centers = params[:2*N].reshape(N, 2)
    radii = params[2*N:]
    
    # Boundary penalties: penalize if circle crosses x=0, x=1, y=0, y=1
    pen = np.sum(np.maximum(0.0, radii - centers[:, 0])**2)
    pen += np.sum(np.maximum(0.0, radii - (1.0 - centers[:, 0]))**2)
    pen += np.sum(np.maximum(0.0, radii - centers[:, 1])**2)
    pen += np.sum(np.maximum(0.0, radii - (1.0 - centers[:, 1]))**2)
    
    # Overlap penalties: penalize if distance < r_i + r_j
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2) + 1e-12)
    mask = np.triu(np.ones((N, N), dtype=bool), k=1)
    overlaps = radii[:, None] + radii[None, :] - dists[mask]
    pen += np.sum(np.maximum(0.0, overlaps)**2)
    
    # Objective: maximize sum of radii (minimize negative sum) subject to penalties
    return -np.sum(radii) + 5000.0 * pen

def run_packing():
    N = 26
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Suppress optimization warnings for cleaner output
    np.seterr(all='ignore')
    
    for seed in range(15):
        np.random.seed(seed)
        # Hexagonal-ish initialization for efficient packing
        init_c = np.zeros((N, 2))
        idx = 0
        for r in range(6):
            y = 0.12 + r * 0.135
            for c in range(5):
                if idx >= N: break
                x = 0.12 + c * 0.175 + (0.0875 if r % 2 == 1 else 0.0)
                init_c[idx] = [x, y]
                idx += 1
        init_r = np.full(N, 0.08)
        init_params = np.concatenate([init_c.flatten(), init_r])
        
        # L-BFGS-B handles bounds efficiently
        res = minimize(_compute_loss, init_params, args=(N,), 
                       method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * 2*N + [(0.0, 0.5)] * N,
                       options={'ftol': 1e-12, 'maxiter': 30000, 'gtol': 1e-8})
                       
        # Post-processing: compute exact maximum feasible radii for optimized centers
        c_opt = res.x[:2*N].reshape(N, 2)
        r_opt = np.zeros(N)
        for i in range(N):
            # Distance to boundaries
            max_r = min(c_opt[i, 0], 1.0 - c_opt[i, 0], c_opt[i, 1], 1.0 - c_opt[i, 1])
            # Distance to other circles
            for j in range(N):
                if i == j: continue
                d = np.sqrt(np.sum((c_opt[i] - c_opt[j])**2))
                if d / 2.0 < max_r:
                    max_r = d / 2.0
            # Shrink slightly to guarantee strict validity within 1e-12 tolerance
            r_opt[i] = max_r - 1e-10
            
        s = np.sum(r_opt)
        if s > best_sum:
            best_sum = s
            best_centers = c_opt
            best_radii = r_opt
            
    return best_centers, best_radii, best_sum
