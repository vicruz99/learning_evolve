import numpy as np
import scipy.optimize as opt
import random

def run_packing():

    def objective(vars_flat):
        n = len(vars_flat) // 3
        radii = vars_flat[2::3]
        return -np.sum(radii)

    def constraints(vars_flat):
        n = len(vars_flat) // 3
        centers = np.array([vars_flat[i::3] for i in range(n)]).T
        radii = vars_flat[2::3]
        cons = []

        cons.append(centers[:, 0] - radii)
        cons.append(1 - centers[:, 0] - radii)
        cons.append(centers[:, 1] - radii)
        cons.append(1 - centers[:, 1] - radii)

        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                cons.append(dist - (radii[i] + radii[j]))

        return np.concatenate(cons)

    def get_initial_guess():

        r_est = 0.11
        centers = []
        y_step = r_est * np.sqrt(3)
        x_step = 2 * r_est
        y = r_est

            x = r_est

            y += y_step

            vars_flat,

    return centers, radii, np.sum(radii)
