import numpy as np
from scipy.optimize import minimize

def run_packing():
    n = 26
    
    # 1. Initialize centers in a hexagonal lattice pattern
    r_est = 0.10
    centers = []
    y = r_est
    row = 0
    while len(centers) < n:
        # Stagger odd rows by one radius
        x = r_est + (row % 2) * r_est
        while x + r_est <= 1.0 and len(centers) < n:
            centers.append([x, y])
            x += 2.0 * r_est
        y += r_est * np.sqrt(3)
        row += 1
    centers = np.array(centers[:n])
    
    # 2. Add deterministic symmetry-breaking perturbation
    for i in range(n):
        centers[i, 0] += 0.0008 * (i % 9)
        centers[i, 1] += 0.0008 * (i // 9)
    centers = np.clip(centers, 0.05, 0.95)
    
    # 3. Setup optimization variables: [x1, y1, ..., x26, y26, r]
    x0 = np.concatenate([centers.flatten(), [0.05]])
    bounds = [(0.0, 1.0)] * (n * 2) + [(0.0, 0.5)]
    
    def objective(vars):
        # We want to maximize r, so minimize -r
        return -vars[-1]
        
    def constraints(vars):
        r = vars[-1]
        c = vars[:n*2].reshape(n, 2)
        con = []
        # Boundary constraints: distance to edges >= r
        for i in range(n):
            con.append(c[i, 0] - r)
            con.append(1.0 - c[i, 0] - r)
            con.append(c[i, 1] - r)
            con.append(1.0 - c[i, 1] - r)
        # Pairwise non-overlap constraints: distance >= 2r
        for i in range(n):
            for j in range(i + 1, n):
                d = np.linalg.norm(c[i] - c[j])
                con.append(d - 2.0 * r)
        return np.array(con)
        
    cons = {'type': 'ineq', 'fun': constraints}
    
    # 4. Run SLSQP optimizer
    res = minimize(
        objective, 
        x0, 
        method='SLSQP', 
        bounds=bounds, 
        constraints=cons,
        options={'maxiter': 3000, 'ftol': 1e-12, 'disp': False}
    )
    
    # 5. Extract and format results
    if res.success or res.x[-1] > 0.09:
        final_centers = res.x[:n*2].reshape(n, 2)
        final_radii = np.full(n, res.x[-1])
        return final_centers, final_radii, np.sum(final_radii)
    else:
        # Fallback to initial configuration if optimization fails
        return centers, np.full(n, 0.05), n * 0.05
