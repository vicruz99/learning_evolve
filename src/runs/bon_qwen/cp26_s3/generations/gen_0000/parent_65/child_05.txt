def compute_overlap_penalty(centers, radii):
    """Calculates penalty for overlapping circles and boundaries."""
    penalty = 0.0
    n = len(radii)
    
    # Circle-circle overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j]:
                overlap = (radii[i] + radii[j] - dist)
                penalty += overlap ** 2
                
    # Boundary violations
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x < r: penalty += (x - r) ** 2
        if x > 1 - r: penalty += (x - (1 - r)) ** 2
        if y < r: penalty += (y - r) ** 2
        if y > 1 - r: penalty += (y - (1 - r)) ** 2
        
    return penalty
