import numpy as np
from scipy.optimize import minimize


def compute_max_radius(centers, radii, idx, n):
    """Compute maximum valid radius for circle idx given current positions and other radii"""
    x, y = centers[idx]
    r = min(x, 1 - x, y, 1 - y)
    for j in range(n):
        if j == idx:
            continue
        dist = np.sqrt(np.sum((centers[idx] - centers[j]) ** 2))
        r = min(r, dist - radii[j])
    return max(0.0, r)


def compute_all_max_radii(centers, radii, n):
    """Compute maximum valid radii for all circles given positions"""
    new_radii = np.zeros(n)
    for i in range(n):
        new_radii[i] = compute_max_radius(centers, radii, i, n)
    return new_radii


def objective_func(params, n):
    """Objective: negative sum of radii"""
    centers = params[:2 * n].reshape(n, 2)
    radii = params[2 * n:3 * n]
    return -np.sum(radii)


def boundary_constraint_x_min(params, n, i):
    return params[2 * i] - params[2 * n + i]


def boundary_constraint_x_max(params, n, i):
    return 1.0 - params[2 * i] - params[2 * n + i]


def boundary_constraint_y_min(params, n, i):
    return params[2 * i + 1] - params[2 * n + i]


def boundary_constraint_y_max(params, n, i):
    return 1.0 - params[2 * i + 1] - params[2 * n + i]


def overlap_constraint(params, n, i, j):
    x_diff = params[2 * i] - params[2 * j]
    y_diff = params[2 * i + 1] - params[2 * j + 1]
    dist = np.sqrt(x_diff * x_diff + y_diff * y_diff)
    return dist - params[2 * n + i] - params[2 * n + j]


def make_boundary_constraint_x_min_closure(n, i):
    def fn(params):
        return params[2 * i] - params[2 * n + i]
    return fn


def make_boundary_constraint_x_max_closure(n, i):
    def fn(params):
        return 1.0 - params[2 * i] - params[2 * n + i]
    return fn


def make_boundary_constraint_y_min_closure(n, i):
    def fn(params):
        return params[2 * i + 1] - params[2 * n + i]
    return fn


def make_boundary_constraint_y_max_closure(n, i):
    def fn(params):
        return 1.0 - params[2 * i + 1] - params[2 * n + i]
    return fn


def make_overlap_closure(n, i, j):
    def fn(params):
        x_diff = params[2 * i] - params[2 * j]
        y_diff = params[2 * i + 1] - params[2 * j + 1]
        dist = np.sqrt(x_diff * x_diff + y_diff * y_diff)
        return dist - params[2 * n + i] - params[2 * n + j]
    return fn


def make_objective_closure(n):
    def fn(params):
        radii = params[2 * n:3 * n]
        return -np.sum(radii)
    return fn


def coordinate_descent_step(centers, radii, n, search_range=0.05, num_steps=30):
    """Perform one step of coordinate descent to optimize circle positions"""
    centers_new = centers.copy()
    radii_new = radii.copy()

    for i in range(n):
        best_pos = centers_new[i].copy()
        best_r = compute_max_radius(centers_new, radii_new, i, n)

        # Grid search around current position
        deltas = np.linspace(-search_range, search_range, num_steps)
        for dx in deltas:
            for dy in deltas:
                new_pos = centers_new[i] + np.array([dx, dy])
                if not (0.0 <= new_pos[0] <= 1.0 and 0.0 <= new_pos[1] <= 1.0):
                    continue
                centers_new[i] = new_pos
                new_r = compute_max_radius(centers_new, radii_new, i, n)
                if new_r > best_r + 1e-12:
                    best_r = new_r
                    best_pos = new_pos.copy()
                centers_new[i] = best_pos  # restore if not better

        centers_new[i] = best_pos
        radii_new[i] = best_r

    return centers_new, radii_new


def coordinate_descent_fine(centers, radii, n, search_range=0.01, num_steps=15):
    """Fine-grained coordinate descent"""
    centers_new = centers.copy()
    radii_new = radii.copy()

    for i in range(n):
        best_pos = centers_new[i].copy()
        best_r = compute_max_radius(centers_new, radii_new, i, n)

        deltas = np.linspace(-search_range, search_range, num_steps)
        for dx in deltas:
            for dy in deltas:
                new_pos = centers_new[i] + np.array([dx, dy])
                if not (0.0 <= new_pos[0] <= 1.0 and 0.0 <= new_pos[1] <= 1.0):
                    continue
                centers_new[i] = new_pos
                new_r = compute_max_radius(centers_new, radii_new, i, n)
                if new_r > best_r + 1e-12:
                    best_r = new_r
                    best_pos = new_pos.copy()
                centers_new[i] = best_pos

        centers_new[i] = best_pos
        radii_new[i] = best_r

    return centers_new, radii_new
