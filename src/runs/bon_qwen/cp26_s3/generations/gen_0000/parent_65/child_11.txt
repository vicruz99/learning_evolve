import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes the packing of 26 circles in a unit square to maximize sum of radii.
    """
    n = 26
    
    # 1. Initialization
    # Start with a 5x5 grid plus one extra circle to break symmetry and provide a good starting point.
    np.random.seed(42)
    xs = np.linspace(0.1, 0.9, 5)
    ys = np.linspace(0.1, 0.9, 5)
    
    centers = []
    radii = []
    
    # Fill 5x5 grid
    for y in ys:
        for x in xs:
            centers.append([x, y])
            radii.append(0.05)
            
    # Add 26th circle near a gap
    centers.append([0.5, 0.05]) 
    radii.append(0.05)
    
    centers = np.array(centers)
    radii = np.array(radii)
    
    # 2. Variable setup for optimization
    # Format: [x0, y0, r0, x1, y1, r1, ...]
    x0 = np.zeros(3 * n)
    for i in range(n):
        x0[3*i] = centers[i, 0]
        x0[3*i+1] = centers[i, 1]
        x0[3*i+2] = radii[i]
        
    bounds = []
    for i in range(n):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 0.5)) # r
        
    # 3. Define constraints
    # Boundary constraints: x >= r, x <= 1-r, y >= r, y <= 1-r
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    
    def get_constraints(vars):
        c = []
        for i in range(n):
            xi, yi, ri = vars[3*i], vars[3*i+1], vars[3*i+2]
            c.append(xi - ri)           # x >= r
            c.append(1.0 - xi - ri)     # x <= 1-r
            c.append(yi - ri)           # y >= r
            c.append(1.0 - yi - ri)     # y <= 1-r
            
            for j in range(i + 1, n):
                xj, yj, rj = vars[3*j], vars[3*j+1], vars[3*j+2]
                dx = xi - xj
                dy = yi - yj
                dist_sq = dx*dx + dy*dy
                rad_sum = ri + rj
                c.append(dist_sq - rad_sum*rad_sum)
        return np.array(c)
        
    cons = {'type': 'ineq', 'fun': get_constraints}
    
    # 4. Optimization function
    def objective(vars):
        return -np.sum(vars[2::3]) # Maximize sum of radii
        
    def run_optimization(init_vars):
        res = minimize(objective, init_vars, method='SLSQP', bounds=bounds, 
                       constraints=cons, options={'maxiter': 500, 'ftol': 1e-12})
        return res.x
    
    # 5. Execute optimization and validate
    optimized_vars = run_optimization(x0)
    
    centers_opt = np.array([[optimized_vars[3*i], optimized_vars[3*i+1]] for i in range(n)])
    radii_opt = np.array([optimized_vars[3*i+2] for i in range(n)])
    
    # Validate function (copy of the provided logic)
    def validate_packing(c, r):
        if np.isnan(c).any() or np.isnan(r).any():
            return False
        for i in range(len(c)):
            if r[i] < 0: return False
            x, y = c[i]
            if x - r[i] < -1e-12 or x + r[i] > 1 + 1e-12 or \
               y - r[i] < -1e-12 or y + r[i] > 1 + 1e-12:
                return False
        for i in range(len(c)):
            for j in range(i + 1, len(c)):
                dist = np.sqrt(np.sum((c[i] - c[j]) ** 2))
                if dist < r[i] + r[j] - 1e-12:
                    return False
        return True

    if not validate_packing(centers_opt, radii_opt):
        # Secondary optimization attempt if first fails
        x0_random = np.zeros(3 * n)
        for i in range(n):
            x0_random[3*i] = np.random.rand() * 0.8 + 0.1
            x0_random[3*i+1] = np.random.rand() * 0.8 + 0.1
            x0_random[3*i+2] = 0.02
        optimized_vars = run_optimization(x0_random)
        centers_opt = np.array([[optimized_vars[3*i], optimized_vars[3*i+1]] for i in range(n)])
        radii_opt = np.array([optimized_vars[3*i+2] for i in range(n)])

    sum_radii = np.sum(radii_opt)
    return centers_opt, radii_opt, sum_radii
