import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # --- 1. Initialization: Hexagonal Lattice ---
    # Hexagonal packing is denser than square grid.
    # We arrange rows with offset x-coordinates and sqrt(3) vertical spacing.
    y = 0.15
    row_count = 0
    idx = 0
    while idx < n:
        # Determine number of circles in current row
        # Alternating 5 and 4 circles allows for hexagonal offsets
        num_in_row = 5 if row_count % 2 == 0 else 4
        
        # Calculate x coordinates for this row
        # For a row with k circles, we distribute them evenly in [r, 1-r]
        # Here we fix spacing initially to 0.2 (diameter)
        x_coords = np.linspace(0.1, 0.9, num_in_row)
        
        # If it's an odd row (0-indexed), shift x by half the diameter
        if row_count % 2 == 1:
            x_coords += 0.1
            
        for x in x_coords:
            if idx < n:
                centers[idx] = [x, y]
                radii[idx] = 0.1
                idx += 1
        
        # Vertical spacing for hexagonal packing: r * sqrt(3)
        y += 0.1 * np.sqrt(3)
        row_count += 1

    # --- 2. Optimization Setup ---
    # Variables: [x1, y1, r1, x2, y2, r2, ...]
    # Flattening the arrays to a single vector for scipy
    x0 = np.concatenate([centers.flatten(), radii])
    
    # Objective: Maximize sum of radii (Minimize negative sum)
    def objective(vars_flat):
        radii_opt = vars_flat[2::3]
        return -np.sum(radii_opt)

    # Constraints
    cons = []
    
    # 1. Boundary constraints: r <= x <= 1-r and r <= y <= 1-r
    # Implemented as x - r >= 0, 1 - (x + r) >= 0, etc.
    for i in range(n):
        base = 3 * i
        # x - r >= 0
        cons.append({'type': 'ineq', 'fun': lambda v, i=i: v[base] - v[base+2]})
        # 1 - (x + r) >= 0
        cons.append({'type': 'ineq', 'fun': lambda v, i=i: 1.0 - v[base] - v[base+2]})
        # y - r >= 0
        cons.append({'type': 'ineq', 'fun': lambda v, i=i: v[base+1] - v[base+2]})
        # 1 - (y + r) >= 0
        cons.append({'type': 'ineq', 'fun': lambda v, i=i: 1.0 - v[base+1] - v[base+2]})

    # 2. Overlap constraints: dist(i, j) >= r_i + r_j
    # dist^2 >= (r_i + r_j)^2
    for i in range(n):
        for j in range(i + 1, n):
            base_i = 3 * i
            base_j = 3 * j
            cons.append({
                'type': 'ineq', 
                'fun': lambda v, i=base_i, j=base_j: 
                    (v[i] - v[j])**2 + (v[i+1] - v[j+1])**2 - (v[i+2] + v[j+2])**2
            })

    # 3. Non-negative radii
    bounds = []
    for _ in range(n):
        bounds.extend([
            (0.0, 1.0), # x bounds
            (0.0, 1.0), # y bounds
            (1e-6, 1.0) # r bounds (small epsilon to prevent zero)
        ])

    # --- 3. Run Optimizer ---
    # SLSQP is suitable for constrained non-linear optimization
    result = minimize(
        objective, 
        x0, 
        method='SLSQP', 
        bounds=bounds, 
        constraints=cons,
        options={'maxiter': 1000, 'ftol': 1e-8}
    )

    # --- 4. Extract and Validate ---
    best_vars = result.x
    final_centers = best_vars[0::3].reshape(n, 2)
    final_radii = best_vars[2::3]
    
    sum_r = np.sum(final_radii)
    
    return final_centers, final_radii, sum_r
