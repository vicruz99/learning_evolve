import numpy as np
from scipy.optimize import minimize

def objective(vars, n):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(vars[2*n:])

def get_constraints(vars, n):
    """Compute all boundary and pairwise non-overlap constraints."""
    centers = vars[:2*n].reshape(n, 2)
    radii = vars[2*n:]
    
    # Preallocate constraint array: 4 boundary per circle + n(n-1)/2 pairwise
    num_constraints = 4*n + n*(n-1)//2
    cons = np.zeros(num_constraints)
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    cons[:n] = centers[:, 0] - radii
    cons[n:2*n] = 1.0 - centers[:, 0] - radii
    cons[2*n:3*n] = centers[:, 1] - radii
    cons[3*n:4*n] = 1.0 - centers[:, 1] - radii
    
    # Pairwise overlap constraints: dist^2 >= (r_i + r_j)^2
    # Vectorized difference matrix
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist_sq = np.sum(diff**2, axis=2)
    r_sum_sq = (radii[:, np.newaxis] + radii[np.newaxis, :])**2
    overlap_vals = dist_sq - r_sum_sq
    
    # Extract strictly upper triangular elements (i < j)
    idx_i, idx_j = np.triu_indices(n, k=1)
    cons[4*n:] = overlap_vals[idx_i, idx_j]
    
    return cons

def run_packing():
    n = 26
    
    # 1. Initialize with a hexagonal grid pattern
    r0 = 0.06
    centers = []
    y = 0.5
    row_idx = 0
    while len(centers) < n:
        cols = 5 if row_idx % 2 == 0 else 4
        x_start = 0.5 - (cols - 1) * 0.2
        for c in range(cols):
            if len(centers) >= n:
                break
            centers.append([x_start + c * 0.2, y])
        y -= np.sqrt(3) / 2 * 0.2
        row_idx += 1
        
    centers = np.array(centers[:n])
    
    # Center and scale to fit comfortably inside [0,1]^2
    cx, cy = centers.mean(axis=0)
    centers = centers - np.array([cx, cy])
    max_extent = np.max(np.abs(centers)) + r0
    scale = 0.42 / max_extent
    centers = centers * scale + 0.5
    
    radii_init = np.full(n, r0)
    x0 = np.concatenate([centers.flatten(), radii_init])
    
    # 2. Set up bounds and constraints
    bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n
    cons = {'type': 'ineq', 'fun': get_constraints, 'args': (n,)}
    
    # 3. Optimize using SLSQP
    res = minimize(
        objective, x0, args=(n,), method='SLSQP', bounds=bounds, constraints=cons,
        options={'maxiter': 10000, 'ftol': 1e-12, 'disp': False}
    )
    
    centers_opt = res.x[:2*n].reshape(n, 2)
    radii_opt = res.x[2*n:]
    
    # 4. Post-process to ensure strict feasibility against numerical tolerance
    radii_opt = np.maximum(radii_opt, 0.0)
    centers_opt[:, 0] = np.clip(centers_opt[:, 0], radii_opt, 1.0 - radii_opt)
    centers_opt[:, 1] = np.clip(centers_opt[:, 1], radii_opt, 1.0 - radii_opt)
    
    return centers_opt, radii_opt, float(np.sum(radii_opt))
