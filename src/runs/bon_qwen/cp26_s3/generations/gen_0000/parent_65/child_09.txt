import numpy as np
import scipy.optimize as opt

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True

def compute_loss(params, n_circles):
    """
    Computes the loss function: -sum(radii) + penalties for violations.
    params: array of shape (3 * n_circles) -> [x1, y1, r1, x2, y2, r2, ...]
    """
    centers = params[0::3].reshape(-1, 2)
    radii = params[2::3].reshape(-1)
    
    # Ensure radii are non-negative for distance calculations (though optimizer might explore negative)
    # We use a soft penalty for negative radii or clip it for distance checks?

    loss = -current_sum_radii
    
    # Penalty for boundary violations

    # x + r <= 1 => x + r - 1 <= 0

        v4 = max(0, y + r - 1)
        boundary_violations.append(v1 + v2 + v3 + v4)
    
    loss += 1000 * np.sum(np.array(boundary_violations)**2)
    
    # Penalty for overlaps

            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            overlap = radii[i] + radii[j] - dist
            if overlap > 0:
                loss += 1000 * overlap**2
                
    return loss
