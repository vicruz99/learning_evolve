import numpy as np
from scipy.optimize import minimize

def compute_sum_radii(centers):
    """
    Computes the maximum sum of radii for a given configuration of centers.
    Each radius is constrained by boundaries and half the distance to the nearest neighbor.
    """
    x = centers[:, 0]
    y = centers[:, 1]
    
    # Distance to the four boundaries
    d_bound = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
    
    # Pairwise distances using broadcasting
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    # Ignore self-distances by setting them to a large value
    dists[np.eye(26, dtype=bool)] = 2.0
    
    # Minimum distance to any other circle
    min_pair_dists = np.min(dists, axis=1)
    
    # Effective radius for each circle
    radii = np.minimum(d_bound, 0.5 * min_pair_dists)
    return np.sum(radii)

def objective(centers_flat):
    """
    Objective function for optimization: minimize negative sum of radii.
    Centers are clipped to [0, 1] to stay within bounds.
    """
    centers = centers_flat.reshape(26, 2)
    centers = np.clip(centers, 0.0, 1.0)
    return -compute_sum_radii(centers)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    best_sum = 0.0
    best_centers = None
    
    # Prepare initial configurations
    starts = []
    
    # 1. Hexagonal-like grid (efficient packing structure)
    c_hex = []
    cols_per_row = [6, 5, 6, 5, 4]  # Totals 26
    for r in range(5):
        row_y = 0.15 + r * 0.19
        for c in range(cols_per_row[r]):
            row_x = 0.15 + c * 0.22 + (r % 2) * 0.11
            c_hex.append([row_x, row_y])
    starts.append(np.array(c_hex).flatten())
    
    # 2. Random starts to escape local minima
    np.random.seed(42)
    for _ in range(10):
        starts.append(np.random.rand(26, 2).flatten())
        
    # Optimization loop
    for c0 in starts:
        res = minimize(
            objective, 
            c0, 
            method='Nelder-Mead', 
            options={
                'maxiter': 50000, 
                'xatol': 1e-7, 
                'fatol': 1e-9
            }
        )
        current_sum = -res.fun
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = res.x.reshape(26, 2)
            
    # Final validation and radius computation
    final_centers = np.clip(best_centers, 0.0, 1.0)
    final_radii = np.zeros(26)
    
    for i in range(26):
        x, y = final_centers[i]
        # Start with boundary limits
        r = min(x, 1.0 - x, y, 1.0 - y)
        
        # Limit by distances to all other circles
        for j in range(26):
            if i != j:
                dx = final_centers[i, 0] - final_centers[j, 0]
                dy = final_centers[i, 1] - final_centers[j, 1]
                d = np.sqrt(dx*dx + dy*dy)
                r = min(r, d / 2.0)
        final_radii[i] = r
        
    return final_centers, final_radii, float(np.sum(final_radii))
