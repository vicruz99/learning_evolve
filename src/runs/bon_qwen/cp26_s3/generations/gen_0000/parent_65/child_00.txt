import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def obj_func(x):
    # Maximize sum of radii -> minimize negative sum
    return -np.sum(x[:N_CIRCLES])

def con_func(x):
    r = x[:N_CIRCLES]
    c = x[N_CIRCLES:].reshape(N_CIRCLES, 2)
    cons = []
    
    # Boundary constraints: circle i must be inside [0,1]x[0,1]
    for i in range(N_CIRCLES):
        cons.append(c[i, 0] - r[i])          # x - r >= 0
        cons.append(1.0 - c[i, 0] - r[i])    # 1 - (x + r) >= 0
        cons.append(c[i, 1] - r[i])          # y - r >= 0
        cons.append(1.0 - c[i, 1] - r[i])    # 1 - (y + r) >= 0
        
    # Non-overlap constraints: dist^2 >= (r_i + r_j)^2
    for i in range(N_CIRCLES):
        for j in range(i + 1, N_CIRCLES):
            dx = c[i, 0] - c[j, 0]
            dy = c[i, 1] - c[j, 1]
            cons.append(dx*dx + dy*dy - (r[i] + r[j])**2)
            
    return np.array(cons)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Hexagonal initialization for dense packing
    r_init = 0.09
    centers = []
    y = r_init
    dy = r_init * np.sqrt(3)
    dx = 2 * r_init
    rows = [6, 5, 6, 5, 4]  # 26 circles total
    
    for row_len in rows:
        row_width = (row_len - 1) * dx
        x_start = (1.0 - row_width) / 2.0
        for k in range(row_len):
            centers.append([x_start + k * dx, y])
        y += dy
        
    centers = np.array(centers)
    radii = np.full(N_CIRCLES, r_init)
    
    # Flatten parameters: [r0..r25, x0..x25, y0..y25]
    x0 = np.concatenate([radii, centers.flatten()])
    bounds = [(0.0, 0.5)] * N_CIRCLES + [(0.0, 1.0)] * (2 * N_CIRCLES)
    
    cons = {'type': 'ineq', 'fun': con_func}
    
    # Run constrained optimization
    res = minimize(obj_func, x0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': 3000, 'ftol': 1e-10})
                   
    r_opt = np.maximum(res.x[:N_CIRCLES], 0.0)
    c_opt = res.x[N_CIRCLES:].reshape(N_CIRCLES, 2)
    
    # Post-processing to guarantee strict validity against numerical tolerance
    tol = 1e-8
    for i in range(N_CIRCLES):
        for j in range(i+1, N_CIRCLES):
            d = np.hypot(c_opt[i,0]-c_opt[j,0], c_opt[i,1]-c_opt[j,1])
            if d < r_opt[i] + r_opt[j] - tol:
                red = (r_opt[i] + r_opt[j] - d) / 2.0 + tol
                r_opt[i] -= red
                r_opt[j] -= red
                
    for i in range(N_CIRCLES):
        max_r = min(c_opt[i,0], 1-c_opt[i,0], c_opt[i,1], 1-c_opt[i,1])
        if r_opt[i] > max_r:
            r_opt[i] = max_r
            
    r_opt = np.maximum(r_opt, 0.0)
    
    return c_opt, r_opt, float(np.sum(r_opt))
