import numpy as np
from scipy.optimize import minimize

def run_packing():
    """
    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    
    Returns:
        centers: np.array of shape (26, 2)
        radii: np.array of shape (26,)
        sum_radii: float
    """
    n = 26
    
    # 1. Initialization: Hexagonal Lattice
    # Create a grid and perturb it to break symmetry and avoid local minima
    rng = np.random.default_rng(42)
    
    # Estimate number of rows for hexagonal packing
    # Area density approx pi/sqrt(12) ~ 0.9069
    # r approx sqrt(1 / (26 * pi)) ~ 0.11. Diameter ~ 0.22.
    # 1/0.22 ~ 4.5 rows.
    cols = 5
    rows = 6
    
    centers_init = []
    for r in range(rows):
        for c in range(cols):
            if len(centers_init) >= n:
                break
            # Hexagonal offset
            x = (c * 0.2) + (0.5 * (0.2 * np.sqrt(3)/2) if r % 2 == 1 else 0)
            y = r * (0.2 * np.sqrt(3)/2)
            # Add small random noise to escape symmetry
            x += (rng.random() - 0.5) * 0.02
            y += (rng.random() - 0.5) * 0.02
            centers_init.append([x, y])
        if len(centers_init) >= n:
            break
    
    centers_init = np.array(centers_init[:n])
    # Normalize coordinates to [0, 1] relative to a safe bounding box inside [0,1]
    # We start with a safe radius of 0.05
    r_init = 0.05
    centers_init = centers_init * 0.8 + 0.1 # Center in the square
    
    # 2. Define Objective and Constraints
    def objective(vars_flat):
        # vars_flat contains: x0, y0, x1, y1, ..., x25, y25, r0, r1, ..., r25
        centers = vars_flat[:2 * n].reshape((n, 2))
        radii = vars_flat[2 * n:]
        # Maximize sum of radii -> Minimize negative sum
        return -np.sum(radii)

    def bounds():
        bnds = []
        # Centers in [0, 1]
        for _ in range(2 * n):
            bnds.append((0.0, 1.0))
        # Radii >= 0 (and practically <= 0.5)
        for _ in range(n):
            bnds.append((0.0, 0.5))
        return bnds

    def constraints():
        cons = []
        for i in range(n):
            x = 2 * i
            y = 2 * i + 1
            r_idx = 2 * n + i
            
            # Boundary constraints: center - radius >= 0  =>  center >= radius
            # We define inequality constraints g(vars) >= 0
            # x - r >= 0
            cons.append({'type': 'ineq', 'fun': lambda v, i=i: v[x] - v[r_idx]})
            # 1 - x - r >= 0 => 1 - x >= r
            cons.append({'type': 'ineq', 'fun': lambda v, i=i: 1.0 - v[x] - v[r_idx]})
            # y - r >= 0
            cons.append({'type': 'ineq', 'fun': lambda v, i=i: v[y] - v[r_idx]})
            # 1 - y - r >= 0
            cons.append({'type': 'ineq', 'fun': lambda v, i=i: 1.0 - v[y] - v[r_idx]})

        # Overlap constraints: dist(i, j) >= r_i + r_j
        # dist^2 >= (r_i + r_j)^2
        # (xi-xj)^2 + (yi-yj)^2 - (ri+rj)^2 >= 0
        for i in range(n):
            for j in range(i + 1, n):
                idx_x_i = 2 * i
                idx_y_i = 2 * i + 1
                idx_r_i = 2 * n + i
                
                idx_x_j = 2 * j
                idx_y_j = 2 * j + 1
                idx_r_j = 2 * n + j
                
                cons.append({
                    'type': 'ineq',
                    'fun': lambda v, ii=i, jj=j: (
                        (v[idx_x_i] - v[idx_x_j])**2 + (v[idx_y_i] - v[idx_y_j])**2 
                        - (v[idx_r_i] + v[idx_r_j])**2
                    )
                })
        return cons

    # 3. Optimization
    initial_vars = np.concatenate([centers_init.flatten(), np.full(n, 0.05)])
    cons = constraints()
    bnds = bounds()
    
    # Use SLSQP for constrained optimization
    # Max iterations increased for convergence
    result = minimize(
        objective, 
        initial_vars, 
        method='SLSQP', 
        bounds=bnds, 
        constraints=cons,
        options={'maxiter': 1000, 'ftol': 1e-12, 'disp': False}
    )
    
    # 4. Extract Results
    optimal_vars = result.x
    centers = optimal_vars[:2 * n].reshape((n, 2))
    radii = optimal_vars[2 * n:]
    
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
