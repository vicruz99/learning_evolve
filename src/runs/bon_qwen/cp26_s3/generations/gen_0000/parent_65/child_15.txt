import numpy as np
from scipy.optimize import minimize, dual_annealing


def hexagonal_initial(n, r_init, square_size=1.0):
    """Create initial hexagonal packing of n circles with given radius."""
    centers_list = []
    radii_list = []
    
    row = 0
    while len(centers_list) < n:
        y = r_init + row * r_init * np.sqrt(3)
        if y + r_init > square_size + 1e-10:
            break
        offset = r_init * (row % 2)
        col = 0
        while len(centers_list) < n:
            x = r_init + offset + col * 2 * r_init
            if x + r_init > square_size + 1e-10:
                break
            centers_list.append([x, y])
            radii_list.append(r_init)
            col += 1
        row += 1
    
    centers = np.array(centers_list[:n])
    radii = np.array(radii_list[:n])
    
    return centers, radii


def grid_initial(n, r_init, square_size=1.0):

    centers_list = []
    radii_list = []
    
    # Determine grid dimensions
    n_rows = int(np.ceil(np.sqrt(n * 4 / 3)))  # Approximate for hex-like density
    n_cols = int(np.ceil(n / n_rows))
    
    if n_cols * 2 * r_init + 2 * r_init > square_size:
        n_cols = int(np.floor((square_size - 2 * r_init) / (2 * r_init))) + 1
    if n_rows * 2 * r_init + 2 * r_init > square_size:
        n_rows = int(np.floor((square_size - 2 * r_init) / (2 * r_init))) + 1

def compute_smooth_violation(params, n):
    """Compute smooth penalty for constraint violations."""
    centers = params[:2*n].reshape(n, 2)
    radii = params[2*n:]
    
    violation = 0.0
    
    # Boundary violations (smooth quadratic)
    for i in range(n):
        x = centers[i, 0]
        y = centers[i, 1]
        r = radii[i]
        
        # x - r >= 0
        v = r - x
        if v > 0:
            violation += v * v
        
        # x + r <= 1
        v = r + x - 1
        if v > 0:
            violation += v * v
        
        # y - r >= 0
        v = r - y
        if v > 0:
            violation += v * v
        
        # y + r <= 1
        v = r + y - 1
        if v > 0:
            violation += v * v
    
    # Non-overlap violations (smooth)
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            
            v = radii[i] + radii[j] - dist
            if v > 0:
                violation += v * v
    
    return violation


def objective_penalty(params, n, penalty_weight):
    """Objective function with penalty for constraints."""
    radii = params[2*n:]
    obj = -np.sum(radii)
    obj += penalty_weight * compute_smooth_violation(params, n)
    return obj


def optimize_from_initial(params, n, bounds, penalty_start=100.0, max_outer=60):
    """Optimize packing starting from given initial parameters."""

    current_params = params.copy()

def validate_and_fix(centers, radii, n):
    """Validate and fix any small constraint violations."""
    # Clip centers to valid range
    for i in range(n):
        r = radii[i]
        centers[i, 0] = np.clip(centers[i, 0], r, 1 - r)
        centers[i, 1] = np.clip(centers[i, 1], r, 1 - r)

                dy = centers[i, 1] - centers[j, 1]

                if overlap > max_overlap:
                    max_overlap = overlap
                    max_pair = (i, j)

            break

            radii[j] -= max_overlap * 0.51
        radii[i] = max(0, radii[i])
        radii[j] = max(0, radii[j])
    
    # Ensure boundary constraints
    for i in range(n):
        r = min(radii[i], centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])
        radii[i] = max(0, r)
    
    return centers, radii
