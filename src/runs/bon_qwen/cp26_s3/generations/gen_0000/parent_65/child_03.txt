def get_max_radii_sum(centers, n):
    """
    Given fixed centers, solve the LP to find the maximum radii 
    such that circles don't overlap and are inside the square.
    Maximizes sum(r) subject to r_i + r_j <= dist(i, j) and boundary constraints.
    """
    # Cost function: minimize -sum(r) => maximize sum(r)
    c = -np.ones(n)
    
    # Inequality constraints: A_ub @ r <= b_ub
    # We need r_i + r_j <= dist_ij
    # And r_i <= boundary_i
    
    n_constraints = n * (n - 1) // 2 + 4 * n
    A_ub = np.zeros((n_constraints, n))
    b_ub = np.zeros(n_constraints)
    
    idx = 0
    
    # Pairwise distance constraints
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = dist
            idx += 1
            
    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        # r <= x
        A_ub[idx, i] = 1
        b_ub[idx] = x
        idx += 1
        # r <= 1 - x
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - x
        idx += 1
        # r <= y
        A_ub[idx, i] = 1
        b_ub[idx] = y
        idx += 1
        # r <= 1 - y
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - y
        idx += 1
        
    bounds = [(0, None) for _ in range(n)]
    
    try:
        res = opt.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.fun * -1, res.x
        else:
            return 0, None
    except Exception:
        return 0, None
