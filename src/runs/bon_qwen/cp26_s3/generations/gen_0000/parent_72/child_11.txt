import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Pack 26 circles in a unit square to maximize sum of radii.
    Returns centers, radii, sum_radii.
    """
    N = 26
    
    # Best solution tracker
    best_sum = 0.0
    best_centers = None
    best_radii = None

    def get_overlap_penalty(centers, radii):
        """Calculate penalty for overlaps and boundary violations."""
        penalty = 0.0
        # Boundary violations
        for i in range(N):
            x, y = centers[i]
            r = radii[i]
            if x < r: penalty += (r - x)**2
            if x > 1 - r: penalty += (x - (1 - r))**2
            if y < r: penalty += (r - y)**2
            if y > 1 - r: penalty += (y - (1 - r))**2
            
        # Overlap violations
        for i in range(N):
            for j in range(i + 1, N):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                min_dist = radii[i] + radii[j]
                if dist < min_dist:
                    penalty += (min_dist - dist)**2
        return penalty

    def optimize_from_init(centers_init, radii_init):
        """
        Optimization wrapper to maximize sum of radii.
        Uses a penalty method combined with constrained optimization logic.
        """
        # Flatten variables: [x1, y1, r1, ..., x26, y26, r26]
        # Actually let's keep x, y, r separate arrays for easier handling, 
        # but scipy needs 1D. Let's use [x..., y..., r...]
        
        initial_x = centers_init[:, 0]
        initial_y = centers_init[:, 1]
        initial_r = radii_init
        
        x0 = np.concatenate([initial_x, initial_y, initial_r])
        
        # Bounds
        # x, y in [0, 1]
        # r >= 0
        bnds_x = [(0, 1)] * N
        bnds_y = [(0, 1)] * N
        bnds_r = [(0, None)] * N
        bounds = bnds_x + bnds_y + bnds_r
        
        # Objective: Maximize sum of radii -> Minimize -sum(r)
        # We add a penalty term for constraint violations to guide the optimizer.
        # This is a trust region / penalty approach.
        
        def objective(vars_1d):
            x = vars_1d[:N]
            y = vars_1d[N:2*N]
            r = vars_1d[2*N:]
            
            # Objective: -sum(r)
            # We want to maximize sum(r), so we minimize -sum(r).
            # However, pure -sum(r) ignores constraints.
            # We rely on constraints in scipy or penalty.
            # Using scipy constraints is more robust.
            return -np.sum(r)

        # Constraints for scipy
        # 1. Boundary: x >= r, x <= 1-r, etc.
        #    => x - r >= 0
        #    => 1 - x - r >= 0
        #    => y - r >= 0
        #    => 1 - y - r >= 0
        
        # 2. Overlap: dist >= r_i + r_j
        #    => dist^2 >= (r_i + r_j)^2
        
        constraints = []
        
        # Add boundary constraints
        for i in range(N):
            # x_i >= r_i
            cons_x_low = {
                'type': 'ineq',
                'fun': lambda v, i=i: v[i] - v[2*N + i],
                'jac': None # Optional
            }
            constraints.append(cons_x_low)
            
            # 1 - x_i >= r_i => 1 - x_i - r_i >= 0
            cons_x_high = {
                'type': 'ineq',
                'fun': lambda v, i=i: 1 - v[i] - v[2*N + i],
            }
            constraints.append(cons_x_high)
            
            # y_i >= r_i
            cons_y_low = {
                'type': 'ineq',
                'fun': lambda v, i=i: v[N+i] - v[2*N + i],
            }
            constraints.append(cons_y_low)
            
            # 1 - y_i >= r_i
            cons_y_high = {
                'type': 'ineq',
                'fun': lambda v, i=i: 1 - v[N+i] - v[2*N + i],
            }
            constraints.append(cons_y_high)

        # Add overlap constraints
        # To reduce number of constraints, we can batch them or just add them all.
        # N=26 -> 325 constraints. SLSQP can handle this.
        for i in range(N):
            for j in range(i + 1, N):
                # (x_i - x_j)^2 + (y_i - y_j)^2 >= (r_i + r_j)^2
                # dist_sq - (r_i + r_j)^2 >= 0
                
                def make_overlap_con(i, j):
                    def overlap_constraint(v):
                        xi, yi = v[i], v[N+i]
                        xj, yj = v[j], v[N+j]
                        ri, rj = v[2*N+i], v[2*N+j]
                        
                        dist_sq = (xi - xj)**2 + (yi - yj)**2
                        sum_r_sq = (ri + rj)**2
                        return dist_sq - sum_r_sq
                    return overlap_constraint

                cons = {
                    'type': 'ineq',
                    'fun': make_overlap_con(i, j)
                }
                constraints.append(cons)

        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints, 
                           options={'maxiter': 1000, 'ftol': 1e-9})
            
            if res.success:
                x_opt = res.x[:N]
                y_opt = res.x[N:2*N]
                r_opt = res.x[2*N:]
                
                # Verify validity manually with tolerance
                centers_opt = np.column_stack([x_opt, y_opt])
                penalty = get_overlap_penalty(centers_opt, r_opt)
                
                if penalty < 1e-8: # Valid packing
                    current_sum = np.sum(r_opt)
                    return centers_opt, r_opt, current_sum
        except Exception:
            pass
        return None, None, 0.0

    # Helper to generate hexagonal grid initialization
    def hex_grid_init(n_circles, scale_factor=0.1):
        centers = []
        y = scale_factor
        row_count = 0
        while len(centers) < n_circles:
            x = scale_factor
            offset = 0
            if row_count % 2 == 1:
                offset = 2 * scale_factor * np.cos(np.pi/3) # 1.0 * r? No.
                # In hex packing, row shift is r (if spacing is 2r).
                # Here scale_factor is roughly r.
                offset = scale_factor 
            
            while x + scale_factor <= 1.0 and len(centers) < n_circles:
                centers.append([x, y])
                x += 2 * scale_factor # Spacing 2r
            
            y += np.sqrt(3) * scale_factor
            row_count += 1
        
        # Fill remaining if needed with random
        while len(centers) < n_circles:
            centers.append([np.random.rand(), np.random.rand()])
            
        return np.array(centers[:n_circles]), np.array([scale_factor] * n_circles)

    # Try multiple strategies
    strategies = []
    
    # 1. Hexagonal grid estimates
    for r_init in [0.08, 0.09, 0.10, 0.11]:
        c, r = hex_grid_init(N, r_init)
        strategies.append((c, r))

    # 2. Random initializations
    for _ in range(5):
        c = np.random.rand(N, 2)
        r = np.array([0.05] * N) # Small initial radius
        strategies.append((c, r))
        
    # 3. Grid initialization (perturbed)
    # 5x5 grid is 25. Add one.
    grid_pts = []
    for i in range(5):
        for j in range(5):
            grid_pts.append([0.125 + i*0.25, 0.125 + j*0.25])
    # Add one more at center of a gap or random
    grid_pts.append([0.5, 0.5]) # Overlaps, but optimizer will fix
    c_grid = np.array(grid_pts[:N])
    r_grid = np.array([0.08] * N)
    strategies.append((c_grid, r_grid))

    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    for idx, (c_init, r_init) in enumerate(strategies):
        centers_res, radii_res, sum_res = optimize_from_init(c_init, r_init)
        if centers_res is not None and sum_res > best_sum:
            best_sum = sum_res
            best_centers = centers_res
            best_radii = radii_res
            
    # Final check and clipping
    if best_centers is None:
        # Fallback
        return np.random.rand(N, 2), np.ones(N)*0.01, 0.26

    # Ensure radii are non-negative (though optimizer bounds should handle)
    best_radii = np.maximum(best_radii, 0)
    
    return best_centers, best_radii, float(best_sum)
