import numpy as np
from scipy.optimize import minimize

def compute_max_equal_radius(centers):
    """Computes the maximum feasible equal radius for a given set of centers."""
    n = centers.shape[0]
    # Distance to boundaries
    d_b_x = np.minimum(centers[:, 0], 1.0 - centers[:, 0])
    d_b_y = np.minimum(centers[:, 1], 1.0 - centers[:, 1])
    d_b = np.minimum(d_b_x, d_b_y)
    
    # Distance to other circles
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    d_c = np.min(dists, axis=1) / 2.0
    
    # The bottleneck radius is the minimum over all constraints
    return np.min(np.minimum(d_b, d_c))

def run_packing():

    np.random.seed(42)
    
    # 1. Hexagonal-like initialization
    centers = np.zeros((n, 2))
    idx = 0
    y = 0.1
    row = 0
    while idx < n:
        x = 0.1
        while x <= 0.9 and idx < n:
            offset = 0.05 if row % 2 == 1 else 0.0
            centers[idx] = [x + offset, y]
            idx += 1
            x += 0.15
        y += 0.147
        row += 1
        
    # 2. Force-directed expansion simulation
    r = 0.01
    k = 5.0
    alpha = 0.005
    
    for step in range(4000):
        # Vectorized pairwise distances and repulsion
        diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        np.fill_diagonal(dists, np.inf)
        
        target = 2.0 * r
        overlap = np.maximum(0, target - dists)
        safe_dists = np.maximum(dists, 1e-12)
        dir_vec = diff / safe_dists[:, :, np.newaxis]
        f_mag = k * overlap / safe_dists
        forces_mat = dir_vec * f_mag[:, :, np.newaxis]
        forces = np.sum(forces_mat, axis=1)
        
        # Boundary repulsion
        bound_f = np.zeros_like(centers)
        for dim in [0, 1]:
            inside_low = centers[:, dim] < r
            inside_high = centers[:, dim] > 1 - r
            bound_f[inside_low, dim] += k * (r - centers[inside_low, dim])
            bound_f[inside_high, dim] -= k * (centers[inside_high, dim] - (1 - r))
        forces += bound_f
        
        # Normalize and apply step
        norms = np.linalg.norm(forces, axis=1)
        max_norm = np.max(norms)
        
        if max_norm > 1e-12:
            step_vec = (forces / np.maximum(norms[:, np.newaxis], 1e-12)) * alpha
            centers += step_vec
            centers = np.clip(centers, 0.0, 1.0)
            
        # Slowly grow the target radius
        r += 0.00004
        
        # Periodic shake to escape local minima if forces vanish
        if max_norm < 1e-5 and step % 150 == 0:
            centers += np.random.uniform(-0.002, 0.002, centers.shape)
            
    # 3. Local optimization to fine-tune positions
    x0 = centers.flatten()
    
    def obj(x):
        c = np.clip(x.reshape(-1, 2), 0, 1)
        diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        np.fill_diagonal(dists, np.inf)
        min_c = np.min(dists) / 2.0
        min_b = np.min(np.minimum(np.minimum(c[:, 0], 1-c[:, 0]), np.minimum(c[:, 1], 1-c[:, 1])))
        return -np.min(min_c, min_b)
        
    best_res = None
    # Multiple restarts with slight perturbations for robustness
    for trial in range(4):
        x_trial = x0 + np.random.normal(0, 0.0015, x0.shape)
        res = minimize(obj, x_trial, method='Powell', options={'maxiter': 6000, 'xtol': 1e-7, 'ftol': 1e-9})
        if best_res is None or res.fun < best_res.fun:
            best_res = res

    r_eq = compute_max_equal_radius(final_centers)
    # Tiny safety margin for numerical tolerance in validator
    final_radii = np.full(n, r_eq - 1e-10)
    sum_radii = r_eq * n
    
    return final_centers, final_radii, sum_radii
