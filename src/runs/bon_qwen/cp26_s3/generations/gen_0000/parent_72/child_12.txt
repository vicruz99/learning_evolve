def objective_penalty(params, n, penalty_weight=500.0):
    """Objective function: negative sum of radii + penalty for violations"""
    centers = params[:2*n].reshape(n, 2)
    radii = params[2*n:]
    
    obj = -np.sum(radii)
    
    # Boundary penalties (smooth)
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        
        # Left
        v = r - x
        if v > 0:
            obj += penalty_weight * v * v
        # Right
        v = x + r - 1
        if v > 0:
            obj += penalty_weight * v * v
        # Bottom
        v = r - y
        if v > 0:
            obj += penalty_weight * v * v
        # Top
        v = y + r - 1
        if v > 0:
            obj += penalty_weight * v * v
    
    # Overlap penalties
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx*dx + dy*dy)
            v = radii[i] + radii[j] - dist
            if v > 0:
                obj += penalty_weight * v * v
    
    return obj


def objective_penalty_grad(params, n, penalty_weight=500.0):

    return grad


def optimize_from_initial(centers, radii, n, penalty_weight=500.0):
    """Optimize from given initial configuration"""
    params = np.concatenate([centers.flatten(), radii])

    result = minimize(
        objective_penalty,
        params,
        args=(n, penalty_weight),
        method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': 10000, 'ftol': 1e-15, 'gtol': 1e-12}
    )
    
    final_centers = result.x[:2*n].reshape(n, 2)
    final_radii = np.maximum(result.x[2*n:], 0.0)
    
    return final_centers, final_radii, result.success


def check_validity(centers, radii, n):
    """Quick validity check"""
    eps = 1e-10
    
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if r < -eps:
            return False
        if x - r < -eps or x + r > 1 + eps:
            return False
        if y - r < -eps or y + r > 1 + eps:
            return False
    
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx*dx + dy*dy)
            if dist < radii[i] + radii[j] - eps:
                return False
    
    return True

            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx*dx + dy*dy)
            overlap = radii[i] + radii[j] - dist
            if overlap > max_overlap:
                max_overlap = overlap
    
    if max_overlap > 1e-8:
        # Reduce all radii slightly

        r = radii[i]
        max_r = min(x, 1-x, y, 1-y)
        if r > max_r + 1e-10:
            radii[i] = max(max_r, 0.0)
    
    return centers, radii
