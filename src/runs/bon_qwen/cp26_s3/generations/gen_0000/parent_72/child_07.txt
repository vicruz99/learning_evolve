import numpy as np
from scipy.optimize import minimize

N = 26

def compute_objective(vars):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(vars[2::3])

def compute_constraints(vars):
    """
    Compute all inequality constraints c(vars) >= 0.
    Returns an array of constraint values.
    """
    res = []
    
    # Boundary constraints for each circle
    for i in range(N):
        x = vars[3*i]
        y = vars[3*i+1]
        r = vars[3*i+2]
        
        # x >= r  and  1 - x >= r
        res.append(x - r)
        res.append(1.0 - x - r)
        # y >= r  and  1 - y >= r
        res.append(y - r)
        res.append(1.0 - y - r)
        
    # Non-overlap constraints for each pair
    for i in range(N):
        for j in range(i + 1, N):
            dx = vars[3*i] - vars[3*j]
            dy = vars[3*i+1] - vars[3*j+1]
            dr = vars[3*i+2] + vars[3*j+2]
            
            # dist^2 >= (r_i + r_j)^2  =>  dist^2 - (r_i + r_j)^2 >= 0
            res.append(dx*dx + dy*dy - dr*dr)
            
    return np.array(res)

def run_packing():
    # 1. Initialize with a hexagonal lattice pattern
    centers = np.zeros((N, 2))
    radii = np.zeros(N)
    
    pts = []
    r0 = 0.08  # Initial safe radius
    y = r0
    x_init = r0
    
    while len(pts) < N:
        x = x_init
        while x < 1.0 - r0 and len(pts) < N:
            pts.append([x, y])
            x += 2 * r0
        y += r0 * np.sqrt(3) / 2
        # Alternate starting x for hexagonal rows
        x_init = 2 * r0 if x_init == r0 else r0
        
    centers = np.array(pts[:N])
    radii = np.full(N, r0)
    
    # 2. Prepare optimization variables and bounds
    x0 = np.zeros(3 * N)
    for i in range(N):
        x0[3*i] = centers[i, 0]
        x0[3*i+1] = centers[i, 1]
        x0[3*i+2] = radii[i]
        
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N

    cons = {'type': 'ineq', 'fun': compute_constraints}
    
    res = minimize(
        compute_objective, 
        x0, 
        method='SLSQP', 
        bounds=bounds, 
        constraints=cons, 
        options={'maxiter': 1000, 'ftol': 1e-10, 'disp': False}
    )
    
    # 4. Extract results
    if res.success:
        opt_centers = res.x.reshape(N, 3)[:, :2]
        opt_radii = res.x.reshape(N, 3)[:, 2]
    else:
        # Fallback to initial configuration if optimization fails
        opt_centers = centers
        opt_radii = radii
        
    total_sum = np.sum(opt_radii)
    return opt_centers, opt_radii, total_sum
