import numpy as np
from scipy.optimize import minimize


def compute_violation(centers, radii, n):
    v = 0.0
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        v += max(0.0, r - x) ** 2
        v += max(0.0, x + r - 1) ** 2
        v += max(0.0, r - y) ** 2
        v += max(0.0, y + r - 1) ** 2
        v += max(0.0, 1e-10 - r) ** 2
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i][0] - centers[j][0]
            dy = centers[i][1] - centers[j][1]
            d = np.sqrt(dx * dx + dy * dy)
            o = radii[i] + radii[j] - d
            v += max(0.0, o) ** 2
    return v


def objective_fn(x, n, penalty_weight):
    centers = x[: 2 * n].reshape(n, 2)
    radii = x[2 * n :]
    return -np.sum(radii) + penalty_weight * compute_violation(centers, radii, n)


def init_hexagonal(n):
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.075
    idx = 0
    rows = [6, 5, 6, 5, 4]
    s = 0.155
    for ri, cnt in enumerate(rows):
        y = 0.5 + (ri - 2) * s * np.sqrt(3) / 2
        off = s * 0.5 if ri % 2 == 1 else 0
        for ci in range(cnt):
            x = 0.5 + (ci - (cnt - 1) / 2) * s + off
            if idx < n:
                centers[idx] = [x, y]
                idx += 1
    return centers, radii


def init_grid(n):
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.075
    idx = 0
    for row in range(5):
        for col in range(5):
            x = 0.12 + col * 0.19
            y = 0.12 + row * 0.19
            if idx < n:
                centers[idx] = [x, y]
                idx += 1
    return centers, radii


def init_concentric(n):
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.07
    centers[0] = [0.5, 0.5]
    radii[0] = 0.14
    for i in range(1, 7):
        a = 2 * np.pi * (i - 1) / 6
        centers[i] = [0.5 + 0.22 * np.cos(a), 0.5 + 0.22 * np.sin(a)]
        radii[i] = 0.08
    for i in range(7, 19):
        a = 2 * np.pi * (i - 7) / 12
        centers[i] = [0.5 + 0.37 * np.cos(a), 0.5 + 0.37 * np.sin(a)]
        radii[i] = 0.055
    for i in range(19, n):
        t = (i - 19) / (n - 20) if n > 20 else 0
        centers[i] = [0.08 + t * 0.84, 0.08 + (1 - t) * 0.0]
        radii[i] = 0.05
    return centers, radii


def run_optimization(x0, n, bounds):
    x = x0.copy()
    penalty_schedule = [50, 150, 300, 600, 1000, 2000, 4000, 8000, 15000, 30000]
    for pw in penalty_schedule:
        result = minimize(
            objective_fn,
            x,
            args=(n, pw),
            method="L-BFGS-B",
            bounds=bounds,
            options={"maxiter": 4000, "ftol": 1e-15, "gtol": 1e-10},
        )
        x = result.x
    return x


def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    bounds = [(0.0, 1.0) for _ in range(2 * n)] + [(0.0001, 0.5) for _ in range(n)]

    init_functions = [init_hexagonal, init_grid, init_concentric]
    best_centers = None
    best_radii = None
    best_sum = 0.0
    best_violation = float("inf")

    np.random.seed(42)

    for init_idx, init_fn in enumerate(init_functions):
        for restart in range(5):
            seed = init_idx * 100 + restart + 1
            np.random.seed(seed)

            centers, radii = init_fn(n)

            if restart > 0:
                centers += np.random.randn(n, 2) * 0.025
                centers = np.clip(centers, 0.05, 0.95)
                radii += np.random.randn(n) * 0.015
                radii = np.clip(radii, 0.02, 0.35)

            x0 = np.concatenate([centers.flatten(), radii])
            x_opt = run_optimization(x0, n, bounds)

            c_opt = x_opt[: 2 * n].reshape(n, 2)
            r_opt = x_opt[2 * n :]
            v = compute_violation(c_opt, r_opt, n)
            s = np.sum(r_opt)

            if v < best_violation or (abs(v - best_violation) < 1e-10 and s > best_sum):
                best_violation = v
                best_centers = c_opt.copy()
                best_radii = r_opt.copy()
                best_sum = s

    if best_centers is None:
        best_centers = x_opt[: 2 * n].reshape(n, 2)
        best_radii = x_opt[2 * n :]
        best_sum = np.sum(best_radii)

    # Post-processing: fix boundary violations
    for i in range(n):
        r = best_radii[i]
        best_centers[i][0] = np.clip(best_centers[i][0], r, 1.0 - r)
        best_centers[i][1] = np.clip(best_centers[i][1], r, 1.0 - r)

    # Post-processing: fix any remaining overlaps by slightly reducing radii
    for iteration in range(200):
        max_overlap = 0.0
        max_i, max_j = -1, -1
        for i in range(n):
            for j in range(i + 1, n):
                dx = best_centers[i][0] - best_centers[j][0]
                dy = best_centers[i][1] - best_centers[j][1]
                d = np.sqrt(dx * dx + dy * dy)
                o = best_radii[i] + best_radii[j] - d
                if o > max_overlap:
                    max_overlap = o
                    max_i, max_j = i, j
        if max_overlap < 1e-12:
            break
        reduction = max_overlap / 2.0
        best_radii[max_i] -= reduction
        best_radii[max_j] -= reduction
        best_radii = np.maximum(best_radii, 0.0)

    final_sum = np.sum(best_radii)
    return best_centers, best_radii, float(final_sum)
