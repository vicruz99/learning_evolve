import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    Uses a perturbed hexagonal grid initialization and L-BFGS-B optimization.
    """
    np.random.seed(42)
    N = 26
    
    # --- 1. Initialization ---
    # Generate a perturbed hexagonal grid
    centers_init = []
    rows = 6
    # Distribute N items into rows as evenly as possible
    counts = [N // rows + (1 if i < N % rows else 0) for i in range(rows)]
    
    current_y = 0.1
    for r_idx, count in enumerate(counts):
        row_x_start = 0.1
        row_width = 0.8
        if count > 1:
            spacing = row_width / (count - 1)
        else:
            spacing = 0 # Will be handled by center logic if needed, but count >= 1
            
        # Hexagonal offset for odd rows
        offset = spacing / 2.0 if r_idx % 2 == 1 else 0.0
        
        for i in range(count):
            x = row_x_start + i * spacing + offset
            y = current_y
            centers_init.append([x, y])
        
        current_y += 0.16 # Vertical spacing for hex grid

    centers_init = np.array(centers_init[:N])
    initial_radii = np.full(N, 0.1) # Start with a reasonable radius estimate

    # Flatten variables for optimizer: [x0, y0, r0, x1, y1, r1, ...]
    initial_params = np.zeros(3 * N)
    for i in range(N):
        initial_params[3*i] = centers_init[i, 0]
        initial_params[3*i+1] = centers_init[i, 1]
        initial_params[3*i+2] = initial_radii[i]

    # --- 2. Objective and Constraints ---
    def objective(params):
        # Unpack
        cx = params[0::3]
        cy = params[1::3]
        radii = params[2::3]
        
        # Objective: Maximize sum of radii
        return -np.sum(radii)

    def constraints_func(params):
        cx = params[0::3]
        cy = params[1::3]
        radii = params[2::3]
        
        violation = 0.0
        
        # Boundary constraints: cx - r >= 0, cx + r <= 1, cy - r >= 0, cy + r <= 1
        # We use a soft penalty approach or hard constraints? 
        # L-BFGS-B supports bounds. Let's use bounds for [0, 1] on x, y.
        # But r is coupled. We will handle boundary overlap in penalty.
        
        # Penalties for boundaries
        for i in range(N):
            r = radii[i]
            x, y = cx[i], cy[i]
            
            if x - r < 0:
                violation += (x - r)**2 * 1000
            if x + r > 1:
                violation += (x + r - 1)**2 * 1000
            if y - r < 0:
                violation += (y - r)**2 * 1000
            if y + r > 1:
                violation += (y + r - 1)**2 * 1000
                
        # Overlap constraints: dist >= r_i + r_j

                dx = cx[i] - cx[j]
                dy = cy[i] - cy[j]
                dist_sq = dx*dx + dy*dy
                min_dist = radii[i] + radii[j]
                
                if dist_sq < min_dist**2:
                    overlap = min_dist - np.sqrt(dist_sq)
                    violation += overlap**2 * 1000
                    
        return -violation # We are maximizing, so return negative penalty to be added to obj? 
                          # Actually, let's just use a combined objective.

    def combined_objective(params):
        cx = params[0::3]
        cy = params[1::3]
        radii = params[2::3]
        
        penalty = 0.0
        
        # Boundary penalties
        for i in range(N):
            r = radii[i]
            x, y = cx[i], cy[i]
            
            # Soft constraints using max(0, violation)
            penalty += max(0, -(x - r))**2 * 1000 # x < r
            penalty += max(0, x + r - 1)**2 * 1000 # x > 1-r
            penalty += max(0, -(y - r))**2 * 1000 # y < r
            penalty += max(0, y + r - 1)**2 * 1000 # y > 1-r

        for i in range(N):
            for j in range(i + 1, N):
                dx = cx[i] - cx[j]
                dy = cy[i] - cy[j]
                dist = np.sqrt(dx*dx + dy*dy)
                min_dist = radii[i] + radii[j]
                
                if dist < min_dist:
                    overlap = min_dist - dist
                    penalty += overlap**2 * 1000
                    
        # Primary objective: Maximize sum of radii

        return -np.sum(radii) + penalty

    # --- 3. Optimization ---
    # Bounds: x in [0, 1], y in [0, 1], r in [0, 1]
    bounds = []
    for i in range(N):
        bounds.append((0, 1)) # x
        bounds.append((0, 1)) # y
        bounds.append((0, 1)) # r (radius cannot be negative, 1 is loose upper bound)

    # To handle the non-smooth penalty, we might need multiple restarts or a different method,

    # Final Validation and Cleanup
    # Ensure no negative radii due to numerical noise
    radii = np.maximum(radii, 1e-9)
    
    # Verify validity manually to be safe before returning
    # (Though the penalty should have minimized violations)
    
    return centers, radii, np.sum(radii)
