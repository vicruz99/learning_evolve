    n = 26
    centers = vars[:2*n].reshape(n, 2)
    radii = vars[2*n:]
    
    obj = -np.sum(radii)
    
    c_list = []
    
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    # r - x <= 0
    c_list.extend(radii - centers[:, 0])
    # x + r - 1 <= 0
    c_list.extend(centers[:, 0] + radii - 1.0)
    # r - y <= 0
    c_list.extend(radii - centers[:, 1])
    # y + r - 1 <= 0
    c_list.extend(centers[:, 1] + radii - 1.0)

    # (ri+rj)^2 - dist^2 <= 0

import numpy as np
from scipy.optimize import minimize

def compute_objective_and_constraints(vars):
    """
    Computes the objective function (negative sum of radii) and constraint violations.
    Constraints are formulated as g(x) <= 0.
    
    Args:
        vars: Array of shape (78,) containing [x1, y1, r1, x2, y2, r2, ..., x26, y26, r26]
    
    Returns:
        obj: The objective value (to be minimized)
        cons: Array of constraint values (should be <= 0)
    """
    n = 26
    centers = vars[:2*n].reshape(n, 2)
    radii = vars[2*n:]
    
    # Objective: maximize sum(radii) => minimize -sum(radii)
    obj = -np.sum(radii)
    
    c_list = []
    
    # Boundary constraints: 
    # 1. r_i <= x_i  => r_i - x_i <= 0
    # 2. x_i <= 1 - r_i => x_i + r_i - 1 <= 0
    # 3. r_i <= y_i  => r_i - y_i <= 0
    # 4. y_i <= 1 - r_i => y_i + r_i - 1 <= 0
    
    c_list.extend(radii - centers[:, 0])
    c_list.extend(centers[:, 0] + radii - 1.0)
    c_list.extend(radii - centers[:, 1])
    c_list.extend(centers[:, 1] + radii - 1.0)
    
    # Overlap constraints:
    # Distance between centers >= sum of radii
    # dist^2 >= (r_i + r_j)^2
    # (r_i + r_j)^2 - dist^2 <= 0
    
    # Compute squared distances between all pairs efficiently
    # diff shape (n, n, 2)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist_sq = np.sum(diff**2, axis=2)
    
    # Compute squared sum of radii
    rad_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    rad_sum_sq = rad_sum**2
    
    # We only need constraints for pairs i < j (upper triangle excluding diagonal)
    i_idx, j_idx = np.triu_indices(n, k=1)
    
    overlaps = rad_sum_sq[i_idx, j_idx] - dist_sq[i_idx, j_idx]
    c_list.extend(overlaps)
    
    return obj, np.array(c_list)

def objective_func(vars):
    """Wrapper for objective function."""
    obj, _ = compute_objective_and_constraints(vars)
    return obj

def constraint_func(vars):
    """
    Wrapper for constraints.
    SLSQP requires inequality constraints of the form g(x) >= 0.
    Our constraints are c(x) <= 0, so we return -c(x).
    """
    _, cons = compute_objective_and_constraints(vars)
    return -cons

def get_initial_guess(seed=0):
    """
    Generates an initial guess for the variables.
    Starts with a 5x5 grid of circles (25 circles) and adds one in a gap,
    then adds random perturbation.
    """
    n = 26
    np.random.seed(seed)
    coords = []
    
    # 5x5 grid

        for j in range(5):
            coords.append([0.1 + 0.2*i, 0.1 + 0.2*j])
    
    # Add 26th point in a gap (0.2, 0.2)
    coords.append([0.2, 0.2])
    
    coords = np.array(coords)
    
    # Add small random perturbation to break symmetry and avoid singularities
    pert = np.random.normal(0, 0.02, size=(n, 2))
    coords += pert
    
    # Ensure centers are within reasonable bounds [0.05, 0.95]
    coords = np.clip(coords, 0.05, 0.95)
    
    # Initial radii small enough to ensure no overlap
    r0 = np.full(n, 0.05)

def run_packing():

    Solves the circle packing problem to maximize the sum of radii for 26 circles.
    """

    # Bounds for variables:
