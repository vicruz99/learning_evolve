import numpy as np
from scipy.optimize import minimize
import itertools

def compute_loss(vars, n=26, mu=500.0):
    """
    Computes the penalized loss function for circle packing.
    vars: array of shape (3*n,) -> [x_0..x_{n-1}, y_0..y_{n-1}, r_0..r_{n-1}]
    """
    x = vars[:n]
    y = vars[n:2*n]
    r = vars[2*n:]
    
    # Objective: maximize sum of radii -> minimize -sum(r)
    loss = -np.sum(r)
    
    # Boundary penalties: r <= x <= 1-r, r <= y <= 1-r
    # Violations: r - x, r - (1-x), r - y, r - (1-y)
    loss += mu * np.sum(np.maximum(r - x, 0)**2)
    loss += mu * np.sum(np.maximum(r - (1.0 - x), 0)**2)
    loss += mu * np.sum(np.maximum(r - y, 0)**2)
    loss += mu * np.sum(np.maximum(r - (1.0 - y), 0)**2)
    
    # Distance penalties: ||c_i - c_j|| >= r_i + r_j
    # Using broadcasting for efficiency
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    d_sq = dx**2 + dy**2
    
    r_sum = r[:, None] + r[None, :]
    
    # Upper triangle indices to avoid double counting and self-comparison
    idx = np.triu_indices(n, k=1)
    d_sq_tri = d_sq[idx]
    r_sum_tri = r_sum[idx]
    
    # Penalty for overlaps: max(0, r_i + r_j - dist_ij)^2
    dist = np.sqrt(np.maximum(d_sq_tri, 0.0))
    viol = np.maximum(r_sum_tri - dist, 0.0)
    loss += mu * np.sum(viol**2)
    
    return loss

def generate_initial_guess(n=26, seed=0):
    """Generates an initial configuration using a hexagonal lattice pattern."""
    rng = np.random.default_rng(seed)
    
    # Base spacing
    spacing = 0.12
    pts = []
    y = 0.08
    row = 0
    while len(pts) < n:
        x = 0.08
        if row % 2 == 1:
            x = 0.14
        while x <= 0.92 and len(pts) < n:
            pts.append((x, y))
            x += spacing
        y += spacing * np.sqrt(3) / 2
        row += 1
        
    # Convert to array and add small random jitter to break symmetry
    centers = np.array(pts[:n])
    centers += rng.uniform(-0.005, 0.005, size=centers.shape)
    
    # Initial small radii
    radii = np.full(n, 0.04)
    
    # Shuffle to randomize order slightly
    perm = rng.permutation(n)
    centers = centers[perm]
    radii = radii[perm]
    
    vars = np.concatenate([centers[:, 0], centers[:, 1], radii])
    return vars

    n = 26
    best_sum_r = -1.0
    best_centers = None
    best_radii = None
    
    # Bounds: x,y in [0,1], r in [0, 0.5]
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)] * n + [(0.0, 0.5)] * n
    
    # Optimization settings
    opt_options = {'maxiter': 1500, 'ftol': 1e-15, 'gtol': 1e-12}
    
    # Try multiple seeds/configurations
    for seed in range(10):
        x0 = generate_initial_guess(n, seed=seed)
        
        # Phase 1: Moderate penalty to find good positions and grow radii
        res1 = minimize(
            fun=lambda v: compute_loss(v, n=n, mu=200.0),
            x0=x0,
            method='L-BFGS-B',
            bounds=bounds,
            options=opt_options
        )
        
        # Phase 2: High penalty to strictly enforce constraints and polish
        res2 = minimize(
            fun=lambda v: compute_loss(v, n=n, mu=5000.0),
            x0=res1.x,
            method='L-BFGS-B',
            bounds=bounds,
            options=opt_options
        )
        
        # Extract result
        x_final = res2.x[:n]
        y_final = res2.x[n:2*n]

                valid = False
                break

                valid = False
                break
            if y_final[i] - r_final[i] < -1e-9 or y_final[i] + r_final[i] > 1 + 1e-9:
                valid = False
                break
        
        if valid:

                for j in range(i+1, n):

                        valid = False
                        break

        rng = np.random.default_rng(42)
        centers = rng.uniform(0.1, 0.9, size=(n, 2))
        radii = np.full(n, 0.03)
        return centers, radii, float(np.sum(radii))
