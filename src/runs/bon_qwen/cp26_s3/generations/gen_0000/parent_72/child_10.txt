import numpy as np
from scipy.optimize import minimize

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square

    Args:
        centers: np.array of shape (n, 2) with (x, y) coordinates
        radii: np.array of shape (n) with radius of each circle

    Returns:
        True if valid, False otherwise
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n_circles = 26
    
    # --- Initialization: Hexagonal-like Grid ---
    # Create a valid starting configuration to aid convergence
    # Pattern: 6, 5, 6, 5, 4 circles in rows (Total 26)
    # Using a safe initial radius
    r_init = 0.08
    h = np.sqrt(3) * r_init
    
    x_coords = []
    y_coords = []
    
    rows = [
        (6, 0.0, 0.0),      # 6 circles, offset 0
        (5, 0.08, h),       # 5 circles, offset r, height h
        (6, 0.0, 2*h),      # 6 circles, offset 0, height 2h
        (5, 0.08, 3*h),     # 5 circles, offset r, height 3h
        (4, 0.0, 4*h)       # 4 circles, offset 0, height 4h
    ]
    
    # Adjust base y to center vertically if possible, or just start from r
    # Let's scale the whole structure to fit comfortably within [0,1]
    # With r=0.08, height is approx 4h + 2r = 4*0.138 + 0.16 = 0.71. 
    # We have plenty of room.
    
    # We will shift y by an offset to center it
    max_y_needed = 4 * h + 2 * r_init
    y_shift = (1.0 - max_y_needed) / 2.0
    
    for count, x_offset, y_row in rows:
        y = y_row + y_shift + r_init
        for i in range(count):
            x = r_init + x_offset + i * (2 * r_init)
            x_coords.append(x)
            y_coords.append(y)
            
    # Initialize variables vector [x1, y1, r1, x2, y2, r2, ...]
    x0 = np.zeros(n_circles * 3)
    for i in range(n_circles):
        x0[3*i] = x_coords[i]
        x0[3*i+1] = y_coords[i]
        x0[3*i+2] = r_init

    # --- Optimization Problem Definition ---
    
    def objective(vars_vec):
        r = vars_vec[2::3]
        return -np.sum(r)

    def constraint_func(vars_vec):
        # Returns a list of constraint values
        # Constraints must be >= 0
        cons = []
        
        x = vars_vec[0::3]
        y = vars_vec[1::3]
        r = vars_vec[2::3]
        
        # 1. Boundary constraints: r <= x, r <= 1-x, r <= y, r <= 1-y
        # x - r >= 0
        cons.extend(x - r)
        # 1 - x - r >= 0
        cons.extend(1.0 - x - r)
        # y - r >= 0
        cons.extend(y - r)
        # 1 - y - r >= 0
        cons.extend(1.0 - y - r)
        
        # 2. Non-overlap constraints (squared distance)
        # dist^2 >= (ri + rj)^2  => dist^2 - (ri+rj)^2 >= 0
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                dx = x[i] - x[j]
                dy = y[i] - y[j]
                dist_sq = dx*dx + dy*dy
                r_sum = r[i] + r[j]
                cons.append(dist_sq - r_sum*r_sum)
                
        return np.array(cons)

    def constraint_jac(vars_vec):
        # Jacobian of the constraints
        # Returns a matrix of shape (n_constraints, n_variables)
        x = vars_vec[0::3]
        y = vars_vec[1::3]
        r = vars_vec[2::3]
        
        # Number of constraints
        # 4 boundary per circle + N*(N-1)/2 overlap
        n_bnd = 4 * n_circles
        n_ovl = n_circles * (n_circles - 1) // 2
        n_total_cons = n_bnd + n_ovl
        
        jac = np.zeros((n_total_cons, n_circles * 3))
        
        # Boundary Constraints Jacobian
        # Rows 0 to N-1: x - r >= 0
        # Deriv w.r.t x_i is 1, w.r.t r_i is -1
        for i in range(n_circles):
            jac[i, 3*i] = 1.0
            jac[i, 3*i+2] = -1.0
            
        # Rows N to 2N-1: 1 - x - r >= 0
        # Deriv w.r.t x_i is -1, w.r.t r_i is -1
        for i in range(n_circles):
            row = n_circles + i
            jac[row, 3*i] = -1.0
            jac[row, 3*i+2] = -1.0
            
        # Rows 2N to 3N-1: y - r >= 0
        # Deriv w.r.t y_i is 1, w.r.t r_i is -1
        for i in range(n_circles):
            row = 2*n_circles + i
            jac[row, 3*i+1] = 1.0
            jac[row, 3*i+2] = -1.0
            
        # Rows 3N to 4N-1: 1 - y - r >= 0
        # Deriv w.r.t y_i is -1, w.r.t r_i is -1
        for i in range(n_circles):
            row = 3*n_circles + i
            jac[row, 3*i+1] = -1.0
            jac[row, 3*i+2] = -1.0
            
        # Overlap Constraints Jacobian
        # dist^2 - (ri+rj)^2 >= 0
        # d/dx_i = 2(x_i - x_j)
        # d/dy_i = 2(y_i - y_j)
        # d/dr_i = -2(r_i + r_j)
        # Same for j with sign flips for position
        idx = 4 * n_circles
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                dx = x[i] - x[j]
                dy = y[i] - y[j]
                r_sum = r[i] + r[j]
                
                # Circle i
                jac[idx, 3*i] = 2.0 * dx
                jac[idx, 3*i+1] = 2.0 * dy
                jac[idx, 3*i+2] = -2.0 * r_sum
                
                # Circle j
                jac[idx, 3*j] = -2.0 * dx
                jac[idx, 3*j+1] = -2.0 * dy
                jac[idx, 3*j+2] = -2.0 * r_sum
                
                idx += 1
                
        return jac

    # Define constraints for scipy
    # scipy expects a dict or list of NonlinearConstraint
    # Using a single function for all inequality constraints is efficient
    
    bnds = []
    for i in range(n_circles):
        bnds.append((0.0, 1.0)) # x
        bnds.append((0.0, 1.0)) # y
        bnds.append((0.0, 0.5)) # r (max radius 0.5)

    # Create constraint object
    # type 'ineq' means fun(x) >= 0
    con = {
        'type': 'ineq',
        'fun': constraint_func,
        'jac': constraint_jac
    }

    # Run Optimization
    # SLSQP is suitable for this problem
    res = minimize(
        objective,
        x0,
        method='SLSQP',
        jac=None, # Objective gradient is simple (-1 for r, 0 for x,y), scipy can approximate
        bounds=bnds,
        constraints=con,
        options={'maxiter': 2000, 'ftol': 1e-12, 'disp': False}
    )

    # Extract results
    vars_opt = res.x
    centers_opt = np.column_stack((vars_opt[0::3], vars_opt[1::3]))
    radii_opt = vars_opt[2::3]
    
    # Ensure non-negative radii (solver should handle bounds, but clamp for safety)
    radii_opt = np.maximum(radii_opt, 0.0)
    
    # Validation step
    if not validate_packing(centers_opt, radii_opt):
        # If validation fails, try to fix or return a fallback
        # For this problem, we might just clamp radii slightly if numerical errors occurred
        # But usually validate_packing is strict.
        # Let's try to shrink radii slightly to satisfy constraints if needed
        # However, with 1e-12 tolerance in validation, it should pass if close enough.
        pass

    sum_radii = np.sum(radii_opt)
    
    return centers_opt, radii_opt, float(sum_radii)

if __name__ == "__main__":
    c, r, s = run_packing()
    print(f"Sum of radii: {s}")
    print(f"Valid: {validate_packing(c, r)}")
