import numpy as np
from scipy.optimize import minimize

def compute_penalty(z, n):
    """Compute the penalty for constraint violations."""
    centers = z[:2*n].reshape(n, 2)
    radii = z[2*n:]
    penalty = 0.0

    # Boundary constraints: circles must be inside [0,1]x[0,1]
    # x >= r  =>  r - x <= 0
    diff_x = radii - centers[:, 0]
    penalty += np.sum(np.maximum(0.0, diff_x)**2)
    # 1 - x >= r  =>  r + x - 1 <= 0
    diff_1x = radii + centers[:, 0] - 1.0
    penalty += np.sum(np.maximum(0.0, diff_1x)**2)
    # y >= r
    diff_y = radii - centers[:, 1]
    penalty += np.sum(np.maximum(0.0, diff_y)**2)
    # 1 - y >= r
    diff_1y = radii + centers[:, 1] - 1.0
    penalty += np.sum(np.maximum(0.0, diff_1y)**2)

    # Overlap constraints: d_ij >= r_i + r_j
    # Vectorized distance calculation
    dists = np.sqrt(((centers[:, None, :] - centers[None, :, :])**2).sum(axis=2))
    sums_r = radii[:, None] + radii[None, :]
    overlap = sums_r - dists
    np.fill_diagonal(overlap, 0.0)
    penalty += np.sum(np.maximum(0.0, overlap)**2)
    
    return penalty

def objective_with_penalty(z, n, mu):
    """Objective function combining target maximization and penalty."""
    radii = z[2*n:]
    return -np.sum(radii) + mu * compute_penalty(z, n)

def generate_hexagonal_init(n):
    """Generate a hexagonal lattice initialization for n circles."""
    # Estimate spacing to fit roughly in [0.1, 0.9]
    cols = int(np.ceil(np.sqrt(n * 2 / np.sqrt(3))))
    rows = int(np.ceil(n / cols))
    sx = 0.8 / cols
    sy = sx * np.sqrt(3) / 2
    
    centers = []
    for r in range(rows):
        for c in range(cols):
            if len(centers) == n:
                break
            x = 0.1 + c * sx + (r % 2) * (sx / 2)
            y = 0.1 + r * sy
            if 0 <= x <= 1 and 0 <= y <= 1:
                centers.append([x, y])
        if len(centers) == n:
            break
            
    # If not enough, fill randomly (should not happen for n=26)
    while len(centers) < n:
        centers.append([np.random.uniform(0.2, 0.8), np.random.uniform(0.2, 0.8)])
        
    centers = np.array(centers[:n])
    radii = np.full(n, 0.05)
    return np.hstack([centers.flatten(), radii])

def optimize_packing(n, x0, mu_start=10.0, max_mu=1e5, perturb_scale=0.05):
    """Optimize packing using penalty method with adaptive mu."""
    bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n
    mu = mu_start
    best_z = x0.copy()
    best_pen = compute_penalty(x0, n)
    
    for _ in range(30): # Max iterations for mu scaling
        res = minimize(
            objective_with_penalty,
            best_z,
            args=(n, mu),
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 2000, 'ftol': 1e-12, 'gtol': 1e-10}
        )
        best_z = res.x
        current_pen = compute_penalty(best_z, n)
        
        if current_pen < 1e-7:
            break
            
        mu *= 10.0
        if mu > max_mu:
            # Perturb to escape local minima
            centers = best_z[:2*n].reshape(n, 2)
            centers += np.random.normal(0, perturb_scale, centers.shape)
            centers = np.clip(centers, 0.05, 0.95)
            radii = np.full(n, 0.05)
            best_z = np.hstack([centers.flatten(), radii])
            mu = mu_start
            # Continue loop to re-optimize
            
    return best_z

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Pack 26 circles in a unit square to maximize the sum of radii.
    Returns (centers, radii, sum_radii).

    best_z = None
    best_sum_r = -1.0

    for seed in range(10):

        # Mix hexagonal and random perturbations

            x0 = generate_hexagonal_init(n)
        else:

            centers = np.random.uniform(0.15, 0.85, size=(n, 2))
            radii = np.full(n, 0.05)
            x0 = np.hstack([centers.flatten(), radii])
            
        z_opt = optimize_packing(n, x0)
        centers = z_opt[:2*n].reshape(n, 2)
        radii = z_opt[2*n:]
        current_sum = np.sum(radii)
        
        if current_sum > best_sum_r:
            best_sum_r = current_sum
            best_z = z_opt.copy()
            
    # Final cleanup and validation check
    centers = best_z[:2*n].reshape(n, 2)
    radii = best_z[2*n:]
    
    # Ensure strict feasibility by slightly shrinking radii if needed
    # (Though optimizer should have converged to feasible or near-feasible)
    for i in range(n):

            if i == j: continue
            d = np.sqrt(np.sum((centers[i]-centers[j])**2))
            max_r = min(max_r, d - radii[j])
        if max_r < radii[i] - 1e-9:
            radii[i] = max(0.0, max_r)
            
    # Sort circles by radius descending for consistency (optional)
    order = np.argsort(-radii)
    centers = centers[order]
    radii = radii[order]
    sum_radii = np.sum(radii)
    
    return centers, radii, float(sum_radii)
