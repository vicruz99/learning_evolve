def _calculate_forces(centers: np.ndarray, radii: np.ndarray) -> np.ndarray:

def _clamp(centers: np.ndarray, radii: np.ndarray) -> np.ndarray:
    # Ensure centers stay within valid bounds [r, 1-r]
    for i in range(len(centers)):
        r = radii[i]
        centers[i, 0] = np.clip(centers[i, 0], r, 1 - r)
        centers[i, 1] = np.clip(centers[i, 1], r, 1 - r)
    return centers

def _validate(centers, radii):

    return True
