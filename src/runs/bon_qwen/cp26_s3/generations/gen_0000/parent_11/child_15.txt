import numpy as np
from scipy.optimize import minimize

def compute_penalty(centers, radii):
    """Compute squared penalty for boundary violations and overlaps."""
    n = centers.shape[0]
    loss = 0.0
    
    # Boundary penalties
    x = centers[:, 0]
    y = centers[:, 1]
    r = radii
    loss += np.sum(np.maximum(0.0, r - x)**2 + np.maximum(0.0, r - (1.0 - x))**2)
    loss += np.sum(np.maximum(0.0, r - y)**2 + np.maximum(0.0, r - (1.0 - y))**2)
    
    # Overlap penalties
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2) + 1e-12)
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    overlaps = np.maximum(0.0, r_sum - dists)
    loss += np.sum(overlaps[mask]**2)
    
    return loss

def objective_func_flat(c_flat, n, radii):
    """Wrapper for scipy optimizer."""
    centers = c_flat.reshape(n, 2)
    return compute_penalty(centers, radii)

def optimize_positions(centers, radii):
    """Optimize centers given fixed radii."""
    n = centers.shape[0]
    res = minimize(objective_func_flat, centers.flatten(), args=(n, radii),
                   method='L-BFGS-B', bounds=[(0.0, 1.0)]*(2*n),
                   options={'maxiter': 500, 'ftol': 1e-14})
    return res.x.reshape(n, 2)

            if idx < n:
                grid_c.append([(col + 0.5) / ncols, y])
                idx += 1
    while len(grid_c) < n:

            hex_c.append([x, y])
            x += 0.2
        y += 0.15 * np.sqrt(3)
    inits.append(np.array(hex_c[:n]))

    # Optimization loop over initializations
    for init_c in inits:
        centers = init_c.copy()
        radii = np.ones(n) * 0.05

            centers = optimize_positions(centers, radii)

            alpha = 1.0

                r = radii[i]
