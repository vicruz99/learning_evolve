import numpy as np

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square

    Args:
        centers: np.array of shape (n, 2) with (x, y) coordinates
        radii: np.array of shape (n) with radius of each circle

    Returns:
        True if valid, False otherwise
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True

def generate_initial_configurations(n_circles, num_configs):

    base_points = []
    count = 0
    for r in range(rows):
        for c in range(cols):
            if count >= n_circles:
                break
            # Hex offset
            x = (c + 0.5 + (0.5 if r % 2 == 1 else 0.0)) * base_x_step
            y = (r + 0.5) * base_y_step
            base_points.append([x, y])
            count += 1
        if count >= n_circles:
            break
            
    base_points = np.array(base_points[:n_circles])
    
    for k in range(num_configs):
        # Add small random noise to break symmetry and explore local optima
        noise_x = np.random.uniform(-0.02, 0.02, n_circles)
        noise_y = np.random.uniform(-0.02, 0.02, n_circles)
        
        centers = base_points + np.column_stack([noise_x, noise_y])
        
        # Ensure centers are strictly inside (0.1, 0.9) to allow radius growth
        centers = np.clip(centers, 0.05, 0.95)
        
        # Initial radii: small enough to not overlap

        radii = np.full(n_circles, 0.04)
        
        configs.append((centers, radii))
        
    return configs

def get_constraints(n_circles):
    """
    Returns the list of constraint dictionaries for scipy.optimize.minimize
    Constraints:
    1. x_i >= r_i  => x_i - r_i >= 0
    2. x_i <= 1 - r_i => 1 - x_i - r_i >= 0
    3. y_i >= r_i  => y_i - r_i >= 0
    4. y_i <= 1 - r_i => 1 - y_i - r_i >= 0
    5. dist_ij >= r_i + r_j => (x_i-x_j)^2 + (y_i-y_j)^2 - (r_i+r_j)^2 >= 0
    """
    constraints = []
    
    # Precompute indices for pairwise constraints
    pairs = []
    for i in range(n_circles):
        for j in range(i + 1, n_circles):
            pairs.append((i, j))
    
    def boundary_constraints(vars):
        vals = []
        for i in range(n_circles):
            idx = 3 * i
            x, y, r = vars[idx], vars[idx+1], vars[idx+2]
            vals.append(x - r)          # x >= r
            vals.append(1 - x - r)      # x <= 1 - r
            vals.append(y - r)          # y >= r
            vals.append(1 - y - r)      # y <= 1 - r
        return vals

    def pairwise_constraints(vars):
        vals = []
        for i, j in pairs:
            idx_i = 3 * i
            idx_j = 3 * j
            x_i, y_i, r_i = vars[idx_i], vars[idx_i+1], vars[idx_i+2]
            x_j, y_j, r_j = vars[idx_j], vars[idx_j+1], vars[idx_j+2]
            
            dx = x_i - x_j
            dy = y_i - y_j
            dist_sq = dx*dx + dy*dy
            r_sum = r_i + r_j
            
            # Constraint: dist_sq >= r_sum^2
            # Equivalent to dist_sq - r_sum^2 >= 0
            vals.append(dist_sq - r_sum**2)
        return vals

    # We can combine constraints into a single function or list of dicts.
    # Scipy allows a list of constraint dicts.
    
    # Boundary constraints (nonlinear but simple)
    constraints.append({
        'type': 'ineq',
        'fun': boundary_constraints,
        'jac': None # Numerical differentiation
    })
    
    # Pairwise constraints
    constraints.append({
        'type': 'ineq',
        'fun': pairwise_constraints,
        'jac': None
    })
    
    return constraints
