import numpy as np
from scipy.optimize import minimize

def generate_hex_centers(N=26):
    """Generate initial circle centers based on a hexagonal lattice."""
    centers = []
    spacing = 0.22
    y = spacing
    row = 0
    while len(centers) < N:
        x = spacing if row % 2 == 0 else 2 * spacing
        while x <= 1 - spacing and len(centers) < N:
            centers.append([x, y])
            x += 2 * spacing
        y += spacing * np.sqrt(3) / 2
        row += 1
    centers = np.array(centers[:N])
    # Normalize to fit comfortably inside [0,1]^2 with padding
    cs = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))
    return cs * 0.9 + 0.05

def obj_func(v):
    """Objective function: maximize sum of radii -> minimize negative sum."""
    return -np.sum(v[2::3])

def constraint_func(v):
    """Compute all boundary and pairwise non-overlap constraints."""
    N = 26
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    
    cons = []
    # Boundary constraints: x >= r, x + r <= 1, y >= r, y + r <= 1
    cons.append(x - r)
    cons.append(1.0 - x - r)
    cons.append(y - r)
    cons.append(1.0 - y - r)
    
    # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    i_idx, j_idx = np.triu_indices(N, k=1)
    dx = x[i_idx] - x[j_idx]
    dy = y[i_idx] - y[j_idx]
    dr = r[i_idx] + r[j_idx]
    cons.append(dx**2 + dy**2 - dr**2)
    
    return np.concatenate(cons)

def run_packing():
    N = 26
    best_v = None
    best_sum_r = -1.0
    
    np.random.seed(42)
    
    # Run multiple optimization trials with perturbed initial conditions
    for trial in range(3):
        centers_init = generate_hex_centers(N)
        centers_init += np.random.randn(N, 2) * 0.03
        centers_init = np.clip(centers_init, 0.05, 0.95)
        
        r_init = np.full(N, 0.08 + 0.02 * np.random.rand())
        
        v0 = np.zeros(3 * N)
        v0[0::3] = centers_init[:, 0]
        v0[1::3] = centers_init[:, 1]
        v0[2::3] = r_init
        
        bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N
        constraints = {'type': 'ineq', 'fun': constraint_func}
        
        res = minimize(obj_func, v0, method='SLSQP', bounds=bounds, constraints=constraints,
                       options={'maxiter': 2500, 'ftol': 1e-12})
        
        v_opt = res.x
        current_sum_r = np.sum(v_opt[2::3])
        
        # Check feasibility
        cons_vals = constraint_func(v_opt)
        if np.min(cons_vals) >= -1e-7:
            if current_sum_r > best_sum_r:
                best_sum_r = current_sum_r
                best_v = v_opt.copy()
                
    # Fallback if optimization fails to find feasible solution
    if best_v is None:
        centers_init = generate_hex_centers(N)
        best_v = np.zeros(3 * N)
        best_v[0::3] = centers_init[:, 0]
        best_v[1::3] = centers_init[:, 1]
        best_v[2::3] = 0.01
        best_sum_r = np.sum(best_v[2::3])

    centers = np.column_stack((best_v[0::3], best_v[1::3]))
    radii = best_v[2::3]
    radii = np.maximum(radii, 0.0)
    
    # Slight shrinkage to guarantee strict validity against 1e-12 tolerance
    min_con = np.min(constraint_func(best_v))
    if min_con < -1e-6:
        radii *= 1.0 - 1e-5
        best_sum_r = np.sum(radii)
        
    return centers, radii, float(best_sum_r)
