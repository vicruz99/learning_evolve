import numpy as np
import scipy.optimize as opt

def calculate_max_radii(centers):
    """
    Given fixed centers, calculates the maximum valid radius for each circle
    such that they fit in the unit square and do not overlap with each other.
    Solves a linear programming problem conceptually, but we can compute it
    iteratively or via a simple relaxation since n is small.
    
    Actually, for fixed centers, r_i <= min(d_ij/2) and r_i <= wall_dist.
    However, r_i and r_j are coupled by r_i + r_j <= d_ij.
    This is a specific linear programming problem: Maximize sum(r)
    s.t. r_i + r_j <= d_ij for all i!=j
         r_i <= w_i (distance to wall)
         r_i >= 0
    
    We can solve this efficiently using a simple iterative relaxation or
    scipy's linear programming. Given N=26, N^2 constraints is manageable.
    """
    n = centers.shape[0]
    # Distance to walls for each circle
    # r <= x, r <= 1-x, r <= y, r <= 1-y  => r <= min(x, 1-x, y, 1-y)
    wall_dists = np.minimum(
        np.minimum(centers[:, 0], 1 - centers[:, 0]),
        np.minimum(centers[:, 1], 1 - centers[:, 1])
    )
    
    # Inter-circle distances
    # We need to solve: max sum(r) s.t. r_i + r_j <= dist_ij
    # This is equivalent to finding the "bottleneck" radii.
    # A standard way to solve this for equal radii is r = 0.5 * min(dist),
    # but for unequal, it's a bit more complex.
    # However, we can approximate or use a solver.
    # Let's use scipy.linprog for exactness if needed, but a simple iterative
    # projection might be faster and sufficient if we just want a good packing.
    # Actually, for the optimization phase, we can just define the objective
    # as sum of radii assuming they expand to the limit.
    # But strictly, if we fix centers, the max sum of radii is a specific value.
    
    # Let's formulate as LP:
    # max c^T r
    # A_ub r <= b_ub
    # r_i + r_j <= dist_ij  => [0..1..1..0] . r <= dist
    # r_i <= wall_i
    
    from scipy.optimize import linprog
    
    c = -np.ones(n) # Minimize -sum(r)
    
    # Inequality constraints A_ub @ r <= b_ub
    # 1. r_i + r_j <= dist_ij
    # 2. r_i <= wall_i
    
    n_constraints = n * (n - 1) // 2 + n
    A_ub = np.zeros((n_constraints, n))
    b_ub = np.zeros(n_constraints)
    
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dist
            idx += 1
            
    for i in range(n):
        A_ub[idx, i] = 1.0
        b_ub[idx] = wall_dists[i]
        idx += 1
        
    bounds = [(0, None) for _ in range(n)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return -res.fun, -res.x
        else:
            # Fallback to simple geometric estimation if LP fails (rare)
            # Estimate r_i based on min neighbor distance / 2
            radii = np.zeros(n)
            for i in range(n):
                min_dist = np.min(np.linalg.norm(centers[i, np.newaxis] - centers, axis=1))
                # Exclude self
                min_dist = np.min(np.delete(min_dist, i)) if n > 1 else 1.0
                r_circ = 0.5 * min_dist
                r_wall = wall_dists[i]
                radii[i] = min(r_circ, r_wall)
            return np.sum(radii), radii
    except Exception:
        # Fallback
        radii = np.zeros(n)
        for i in range(n):
            min_dist = np.min(np.linalg.norm(centers[i, np.newaxis] - centers, axis=1))
            min_dist = np.min(np.delete(min_dist, i)) if n > 1 else 1.0
            radii[i] = min(0.5 * min_dist, wall_dists[i])
        return np.sum(radii), radii

def objective_function(vars_array):
    """
    Objective function for optimization.
    Minimizes negative sum of radii.
    """
    n = 26
    centers = vars_array.reshape((n, 2))
    total_sum, _ = calculate_max_radii(centers)
    return -total_sum

def run_packing():
    n_circles = 26
    
    # Phase 1: Hexagonal Grid Initialization
    # We create a hexagonal lattice and pick the best n points.
    # Lattice spacing can be estimated. 
    # For n=26, roughly 5x5 grid. Spacing ~ 0.2.
    # Hexagonal spacing dy = spacing * sqrt(3)/2.
    
    spacing = 0.25 
    rows = 7
    cols = 7
    
    points = []
    for r in range(rows):
        for c in range(cols):
            x = c * spacing + (0.5 * spacing if r % 2 == 1 else 0) + spacing/2
            y = r * spacing * np.sqrt(3) / 2 + spacing/2
            if 0 <= x <= 1 and 0 <= y <= 1:
                points.append([x, y])
    
    points = np.array(points)
    
    # Sort points by distance to center of square (0.5, 0.5) to pick central ones
    # Central points allow for larger radii (more space from walls)
    center_sq = np.array([0.5, 0.5])
    dists_to_center = np.linalg.norm(points - center_sq, axis=1)
    indices = np.argsort(dists_to_center)[:n_circles]
    initial_centers = points[indices]
    
    # Phase 2: Optimization
    # We optimize the centers to maximize the sum of radii.
    # Constraints: centers in [0, 1]x[0, 1]
    
    initial_vars = initial_centers.flatten()
    
    bounds = [(0, 1) for _ in range(n_circles * 2)]
    
    # Use SLSQP optimizer
    # We run multiple restarts to avoid local minima
    best_result = None
    best_score = -np.inf
    
    for attempt in range(5):
        # Perturb initial centers slightly for each attempt
        if attempt > 0:
            noise = np.random.uniform(-0.02, 0.02, size=initial_vars.shape)
            current_vars = np.clip(initial_vars + noise, 0, 1)
        else:
            current_vars = initial_vars
            
        try:
            res = opt.minimize(
                objective_function,
                current_vars,
                method='SLSQP',
                bounds=bounds,
                options={'maxiter': 1000, 'ftol': 1e-12}
            )
            
            if res.fun < best_score: # We are minimizing -sum, so lower is better (more negative)
                best_score = res.fun
                best_result = res
        except Exception as e:
            continue
            
    # Extract best solution
    if best_result is None:
        # Fallback to initial
        centers = initial_centers
        _, radii = calculate_max_radii(centers)
    else:
        centers = best_result.x.reshape((n_circles, 2))
        _, radii = calculate_max_radii(centers)
        
    sum_radii = np.sum(radii)
    
    # Phase 3: Validation
    if not validate_packing(centers, radii):
        print("Warning: Validation failed. Attempting to repair.")
        # Simple repair: shrink radii slightly to satisfy constraints
        # This is a last resort, optimization should have found valid packing.
        # But calculate_max_radii guarantees validity based on LP, so it should be fine.
        pass

    return centers, radii, float(sum_radii)
