import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def objective(vars):
    """Maximize the common radius r (minimize -r)."""
    return -vars[-1]

def con_overlap(vars):
    """Constraint: distance between any two centers >= 2*r."""
    r = vars[-1]
    centers = vars[:-1].reshape((N_CIRCLES, 2))
    c = []
    for i in range(N_CIRCLES):
        for j in range(i + 1, N_CIRCLES):
            dist_sq = np.sum((centers[i] - centers[j])**2)
            c.append(dist_sq - 4*r*r)
    return np.array(c)

def con_boundary_x(vars):
    """Constraint: 0 <= x - r and x + r <= 1."""
    r = vars[-1]
    centers = vars[:-1].reshape((N_CIRCLES, 2))
    c = []
    for i in range(N_CIRCLES):
        c.append(centers[i, 0] - r)
        c.append(1 - centers[i, 0] - r)
    return np.array(c)

def con_boundary_y(vars):
    """Constraint: 0 <= y - r and y + r <= 1."""
    r = vars[-1]
    centers = vars[:-1].reshape((N_CIRCLES, 2))
    c = []
    for i in range(N_CIRCLES):
        c.append(centers[i, 1] - r)
        c.append(1 - centers[i, 1] - r)
    return np.array(c)

constraints = [
    {'type': 'ineq', 'fun': con_overlap},
    {'type': 'ineq', 'fun': con_boundary_x},
    {'type': 'ineq', 'fun': con_boundary_y}
]

bounds = [(0.0, 1.0)] * (2 * N_CIRCLES) + [(0.0, 0.5)]

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    best_r = 0.0
    best_centers = None
    
    # Try multiple random initializations
    for seed in range(15):
        np.random.seed(seed)
        r0 = 0.03
        # Random initial positions, kept away from boundaries
        centers = np.random.rand(2 * N_CIRCLES) * 0.8 + 0.1
        x0 = np.concatenate([centers, [r0]])
        
        try:
            res = minimize(objective, x0, method='SLSQP', 
                           bounds=bounds, constraints=constraints,
                           options={'maxiter': 500, 'ftol': 1e-11, 'disp': False})
            
            if res.success:
                r_curr = res.x[-1]
                if r_curr > best_r:
                    best_r = r_curr
                    best_centers = res.x[:-1].reshape((N_CIRCLES, 2))
        except Exception:
            pass
            
    # Try a hexagonal lattice initialization
    hex_coords = []
    # Parameters for a hex grid
    dx, dy = 0.2, 0.173205 
    row = 0
    while len(hex_coords) < N_CIRCLES:
        for col in range(6):
            x = col * dx + (row % 2) * (dx / 2) + 0.05
            y = row * dy + 0.05
            if x <= 0.95 and y <= 0.95:
                hex_coords.append([x, y])
                if len(hex_coords) == N_CIRCLES:
                    break
        row += 1
        
    if len(hex_coords) == N_CIRCLES:
        x0_hex = np.concatenate([np.array(hex_coords).flatten(), [0.08]])
        try:
            res = minimize(objective, x0_hex, method='SLSQP', 
                           bounds=bounds, constraints=constraints,
                           options={'maxiter': 500, 'ftol': 1e-11, 'disp': False})
            if res.success and res.x[-1] > best_r:
                best_r = res.x[-1]
                best_centers = res.x[:-1].reshape((N_CIRCLES, 2))
        except Exception:
            pass

    # Fallback to a valid grid packing if optimization fails
    if best_centers is None:
        coords = []
        for i in range(5):
            for j in range(5):
                coords.append([0.1 + 0.2*i, 0.1 + 0.2*j])
        coords.append([0.5, 0.5])
        best_centers = np.array(coords[:N_CIRCLES])
        best_r = 0.1
        
    radii = np.full(N_CIRCLES, best_r)
    return best_centers, radii, np.sum(radii)
