import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26
TRI_U_IDX = np.triu_indices(N_CIRCLES, k=1)

def _objective(v):
    """Minimize negative sum of radii."""
    return -np.sum(v[2*N_CIRCLES:])

def _constraint(v):
    """
    Returns array of constraint values. All must be >= 0.
    Constraints: boundary compliance and non-overlap.
    """
    centers = v[:2*N_CIRCLES].reshape(N_CIRCLES, 2)
    radii = v[2*N_CIRCLES:]
    
    # Boundary constraints
    b1 = centers[:, 0] - radii
    b2 = 1 - centers[:, 0] - radii
    b3 = centers[:, 1] - radii
    b4 = 1 - centers[:, 1] - radii
    
    # Non-overlap constraints
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    d2 = np.sum(diff**2, axis=2)
    r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    olap = d2 - r_sum**2
    olap = olap[TRI_U_IDX]
    
    return np.concatenate([b1, b2, b3, b4, olap])

def run_packing():
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    bounds = [(0.0, 1.0) for _ in range(2*N_CIRCLES)] + [(1e-6, 0.5) for _ in range(N_CIRCLES)]
    
    # Multiple restarts to avoid local optima
    for seed in range(10):
        np.random.seed(seed)
        
        # Initialize with hex-like grid
        rows, cols = 5, 6
        c0 = np.zeros((N_CIRCLES, 2))
        idx = 0
        for r in range(rows):
            for c in range(cols):
                if idx >= N_CIRCLES: break
                x = (c + 0.5 * (r % 2)) * 0.12 + 0.05
                y = r * 0.12 * np.sqrt(3) + 0.05
                c0[idx] = [x, y]
                idx += 1
                
        # Perturb positions and radii
        c0 += np.random.uniform(-0.03, 0.03, c0.shape)
        c0 = np.clip(c0, 0.02, 0.98)
        r0 = np.ones(N_CIRCLES) * 0.035 + np.random.uniform(0, 0.02, N_CIRCLES)
        
        v0 = np.concatenate([c0.ravel(), r0])
        
        try:
            res = minimize(_objective, v0, bounds=bounds, 
                          constraints={'type': 'ineq', 'fun': _constraint}, 
                          method='SLSQP', options={'maxiter': 5000, 'ftol': 1e-10})
            
            cur_sum = -res.fun
            if cur_sum > best_sum:
                best_sum = cur_sum
                best_centers = res.x[:2*N_CIRCLES].reshape(N_CIRCLES, 2).copy()
                best_radii = res.x[2*N_CIRCLES:].copy()
        except Exception:
            continue
            
    # Final high-precision refinement
    if best_centers is not None:
        v_best = np.concatenate([best_centers.ravel(), best_radii])
        res = minimize(_objective, v_best, bounds=bounds,
                      constraints={'type': 'ineq', 'fun': _constraint},
                      method='SLSQP', options={'maxiter': 8000, 'ftol': 1e-12})
        if -res.fun > best_sum:
            best_sum = -res.fun
            best_centers = res.x[:2*N_CIRCLES].reshape(N_CIRCLES, 2).copy()
            best_radii = res.x[2*N_CIRCLES:].copy()
            
    # Post-optimization safety: resolve tiny overlaps and clamp boundaries
    for _ in range(100):
        c = best_centers[:, np.newaxis, :] - best_centers[np.newaxis, :, :]
        d = np.sqrt(np.sum(c**2, axis=2) + 1e-12)
        r_sum = best_radii[:, np.newaxis] + best_radii[np.newaxis, :]
        overlap = r_sum - d
        mask = overlap > 1e-9
        if not np.any(mask):
            break
        i_idx, j_idx = np.unravel_index(np.argmax(overlap), overlap.shape)
        dist = d[i_idx, j_idx]
        dir_vec = (best_centers[i_idx] - best_centers[j_idx]) / dist
        push = overlap[i_idx, j_idx] / 2.0 + 1e-8
        best_centers[i_idx] += dir_vec * push
        best_centers[j_idx] -= dir_vec * push
        
    best_radii = np.maximum(best_radii, 0.0)
    best_centers = np.clip(best_centers, best_radii[:, np.newaxis], 1 - best_radii[:, np.newaxis])
    
    return best_centers, best_radii, np.sum(best_radii)
