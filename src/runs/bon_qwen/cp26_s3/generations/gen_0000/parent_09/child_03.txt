import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def objective(vars_flat):
    # Maximize sum of radii -> Minimize negative sum
    return -np.sum(vars_flat[2::3])

def constraint_boundary(vars_flat):
    x = vars_flat[0::3]
    y = vars_flat[1::3]
    r = vars_flat[2::3]
    # Circles must be inside [0,1]x[0,1]
    return np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])

def constraint_overlap(vars_flat):
    v = vars_flat.reshape(N_CIRCLES, 3)
    # Vectorized distance and radius sum calculations
    dx = v[:, 0, None] - v[None, :, 0]
    dy = v[:, 1, None] - v[None, :, 1]
    dr = v[:, 2, None] + v[None, :, 2]
    # Upper triangular indices to avoid duplicate constraints and self-overlap
    idx = np.triu_indices(N_CIRCLES, 1)
    # Squared distance minus squared sum of radii must be >= 0
    return dx[idx]**2 + dy[idx]**2 - dr[idx]**2

def run_packing():
    n = N_CIRCLES
    # Bounds: x,y in [0,1], r in [1e-5, 0.5]
    bounds = [(0.0, 1.0)] * (2 * n) + [(1e-5, 0.5)] * n
    
    constraints = [
        {'type': 'ineq', 'fun': constraint_boundary},
        {'type': 'ineq', 'fun': constraint_overlap}
    ]
    
    best_sum = -1.0
    best_vars = None
    
    inits = []
    # 1. 5x5 Grid + 1 center
    pts1 = np.zeros((n, 2))
    idx = 0
    for i in range(5):
        for j in range(5):
            pts1[idx] = [0.1 + i * 0.2, 0.1 + j * 0.2]
            idx += 1
    pts1[idx] = [0.5, 0.5]
    inits.append(pts1)
    
    # 2. Hexagonal layout
    pts2 = np.zeros((n, 2))
    idx = 0
    for row in range(6):
        y = 0.05 + row * 0.16
        shift = 0.085 if row % 2 == 1 else 0.0
        for i in range(5):
            if idx >= n: break
            x = 0.05 + i * 0.17 + shift
            pts2[idx] = [x, y]
            idx += 1
    inits.append(pts2)
    
    # 3. Random distribution
    np.random.seed(42)
    inits.append(np.random.rand(n, 2) * 0.6 + 0.2)
    
    for init_pts in inits:
        x0 = np.zeros(n * 3)
        x0[0::3] = init_pts[:, 0]
        x0[1::3] = init_pts[:, 1]
        # Slightly randomized initial radii to break symmetry
        x0[2::3] = 0.04 + np.random.rand(n) * 0.01
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                           constraints=constraints, options={'maxiter': 1000, 'ftol': 1e-12})
            if res.fun is not None and not np.isinf(res.fun):
                if -res.fun > best_sum:
                    best_sum = -res.fun
                    best_vars = res.x.copy()
        except Exception:
            pass
            
    if best_vars is None:
        # Fallback configuration
        best_vars = np.zeros(n * 3)
        best_vars[0::3] = 0.5
        best_vars[1::3] = 0.5
        best_vars[2::3] = 0.01
        best_sum = 0.26
        
    centers = best_vars.reshape(n, 3)[:, :2]
    radii = best_vars.reshape(n, 3)[:, 2]
    
    # Ensure strict validity within tolerance
    radii = np.maximum(radii, 1e-9)
    centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
    centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)
    
    return centers, radii, np.sum(radii)
