import numpy as np
from scipy.optimize import linprog, minimize

def compute_max_radii(centers):
    """
    Given fixed centers, solves the Linear Program to maximize the sum of radii.
    Returns the optimal radii vector.
    """
    n = centers.shape[0]
    c = -np.ones(n)  # Minimize -sum(r_i) to maximize sum(r_i)
    
    A_ub = []
    b_ub = []
    
    # 1. Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        x, y = centers[i]
        row = np.zeros(n)
        row[i] = 1
        
        A_ub.append(row); b_ub.append(x)
        A_ub.append(row); b_ub.append(1 - x)
        A_ub.append(row); b_ub.append(y)
        A_ub.append(row); b_ub.append(1 - y)
        
    # 2. Pairwise non-overlap constraints: r_i + r_j <= distance
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            row = np.zeros(n)
            row[i] = 1
            row[j] = 
            A_ub.append(row); b_ub.append(dist)
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0, None) for _ in range(n)]
    
    # Use HiGHS solver for efficiency
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
        
    return np.zeros(n)

def objective(centers_flat):
    centers = centers_flat.reshape(-1, 2)
    radii = compute_max_radii(centers)
    return -np.sum(radii)

def run_packing():
    n = 26
    best_sum = 0.0
    best_centers = np.zeros((n, 2))
    best_radii = np.zeros(n)
    
    # Strategy: Multiple restarts with different initial configurations
    initial_configs = []
    
    # 1. Random initialization
    np.random.seed(42)
    initial_configs.append(np.random.rand(n, 2))
    
    # 2. Perturbed Grid (5x5 with one extra)
    coords = np.linspace(0.1, 0.9, 5)
    grid_centers = np.array(np.meshgrid(coords, coords)).T.reshape(-1, 2)
    perturbation = np.random.normal(0, 0.02, grid_centers.shape)
    initial_configs.append(grid_centers + perturbation)
    
    # 3. Perturbed Hexagonal Lattice
    hex_centers = []
    r_init = 0.1
    # Row 1: 4 circles
    for x in np.linspace(0.5 * r_init, 1 - 0.5 * r_init, 4):
        hex_centers.append([x, r_init])
    # Row 2: 4 circles (offset)
    for x in np.linspace(0.5 * r_init + r_init, 1 - 0.5 * r_init - r_init, 4):
        hex_centers.append([x, r_init * np.sqrt(3) + r_init])
    # Row 3: 4 circles
    for x in np.linspace(0.5 * r_init, 1 - 0.5 * r_init, 4):
        hex_centers.append([x, 2 * r_init * np.sqrt(3) + r_init])
    # Row 4: 4 circles
    for x in np.linspace(0.5 * r_init + r_init, 1 - 0.5 * r_init - r_init, 4):
        hex_centers.append([x, 3 * r_init * np.sqrt(3) + r_init])
    # Row 5: 4 circles
    for x in np.linspace(0.5 * r_init, 1 - 0.5 * r_init, 4):
        hex_centers.append([x, 4 * r_init * np.sqrt(3) + r_init])
    # Row 6: 2 circles
    for x in np.linspace(0.5 * r_init + r_init, 1 - 0.5 * r_init - r_init, 2):
        hex_centers.append([x, 5 * r_init * np.sqrt(3) + r_init])
        
    if len(hex_centers) >= n:
        initial_configs.append(np.array(hex_centers[:n]))

    # Optimization loop
    for centers in initial_configs:
        try:
            # Nelder-Mead is robust for non-smooth functions
            res = minimize(objective, centers.flatten(), method='Nelder-Mead', 
                           options={'maxiter': 3000, 'xatol': 1e-8, 'fatol': 1e-8})
            
            current_sum = -res.fun
            if current_sum > best_sum:
                best_sum = current_sum
                best_centers = res.x.reshape(-1, 2)
                best_radii = compute_max_radii(best_centers)
        except Exception:
            continue
            
    # Final validation and clamping to [0,1] just in case
    best_centers = np.clip(best_centers, 0.0, 1.0)
    best_radii = np.clip(best_radii, 0.0, None)
    
    return best_centers, best_radii, float(best_sum)
