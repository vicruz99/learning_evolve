    """
    Objective function for circle packing.
    Minimizes -sum(radii) + weight * penalties.
    """
    # Unpack variables
    # vars layout: x0, y0, r0, x1, y1, r1, ...
    xs = vars[0::3]
    ys = vars[1::3]
    rs = vars[2::3]

    sum_radii = np.sum(rs)
    
    penalty = 0.0

    # r <= x  => x - r >= 0. Violation if x - r < 0.
    # penalty = (min(0, x-r))^2
    for i in range(n):
        xi, yi, ri = xs[i], ys[i], rs[i]
        
        # x >= r
        val = xi - ri
        if val < 0:
            penalty += val * val
            
        # 1-x >= r => 1 - x - r >= 0
        val = 1.0 - xi - ri
        if val < 0:
            penalty += val * val
            
        # y >= r
        val = yi - ri
        if val < 0:
            penalty += val * val
            
        # 1-y >= r
        val = 1.0 - yi - ri
        if val < 0:
            penalty += val * val

    # dist^2 >= (r_i + r_j)^2
    # Violation if dist < r_i + r_j
    # Using squared distance to avoid sqrt in penalty?

        if val < 0:
            penalty += val * val

        if val < 0:
