import numpy as np
import scipy.optimize as opt

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    Packs 26 circles in a unit square to maximize the sum of radii.
    Uses SLSQP optimization starting from a hexagonal lattice initialization.
    """
    n = 26
    
    # --- 1. Initialization ---
    # Create a hexagonal lattice arrangement
    # We arrange circles in rows. 
    # Hexagonal packing vertical spacing is sqrt(3) * r.
    # Horizontal spacing is 2 * r.
    # Staggered rows have horizontal offset of r.
    
    # Let's estimate an initial radius that fits roughly.
    # For 26 circles, a hexagonal packing might allow r ~ 0.09-0.10.
    # Let's start with a safe small radius to ensure validity.
    r_init = 0.04 
    
    centers = np.zeros((n, 2))
    radii = np.full(n, r_init)
    
    # Fill rows
    row = 0
    col = 0
    idx = 0
    
    # Approximate number of circles per row for hexagonal packing
    # Width 1. 2*r*col + 2*r <= 1 => col approx 1/(2r). 
    # With r=0.04, 1/0.08 = 12. So ~12 per row.
    # But we want to fill 26. Maybe 5-6 rows.
    # Let's just fill row by row with some staggering.
    
    y = r_init
    while idx < n:
        if row % 2 == 0:
            x = r_init
        else:
            x = 2 * r_init # Shift by one diameter? Or just r? 
            # In hexagonal, shift is r (radius) to sit in gap.
            # But if we shift by r, the first circle is at 2r.
            # Let's shift by r.
            x = 2 * r_init 
            # Wait, if row 0 starts at r, row 1 starts at 2r?
            # Distance between (r, y) and (2r, y') must be >= 2r.
            # sqrt(r^2 + (sqrt(3)r)^2) = sqrt(r^2 + 3r^2) = 2r. Correct.
        
        x_current = x
        while x_current + r_init <= 1.0 + 1e-9: # Allow slight tolerance for init
            centers[idx] = [x_current, y]
            idx += 1
            if idx >= n:
                break
            x_current += 2 * r_init
            
        y += np.sqrt(3) * r_init
        row += 1
        
    # Ensure we placed n circles
    # (The loop logic above might be slightly off if grid doesn't align perfectly, 
    # but with small r it should be fine. If not, we clamp or adjust.)
    # Let's fix the array size just in case.
    centers = centers[:n]
    radii = radii[:n]
    
    # Reshape variables for optimizer: [x0, y0, r0, x1, y1, r1, ...]
    x0 = centers[:, 0]
    y0 = centers[:, 1]
    r0 = radii[:]
    
    # Interleave
    vars_init = np.empty(3 * n)
    vars_init[0::3] = x0
    vars_init[1::3] = y0
    vars_init[2::3] = r0
    
    # --- 2. Optimization Setup ---
    
    # Bounds: x, y in [0, 1], r in [0, 0.5]
    bounds = []
    for _ in range(n):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 0.5)) # r
    
    # Constraints
    constraints = []
    
    # Boundary constraints: r <= x <= 1-r  =>  x - r >= 0  and  1 - x - r >= 0
    # Same for y
    def bound_constraint_x(v):
        # v is flattened
        res = np.empty(n)
        for i in range(n):
            x = v[3*i]
            r = v[3*i+2]
            res[i] = x - r # >= 0
        return res
    
    def bound_constraint_x_upper(v):
        res = np.empty(n)
        for i in range(n):
            x = v[3*i]
            r = v[3*i+2]
            res[i] = 1.0 - x - r # >= 0
        return res

    def bound_constraint_y(v):
        res = np.empty(n)
        for i in range(n):
            y = v[3*i+1]
            r = v[3*i+2]
            res[i] = y - r # >= 0
        return res

    def bound_constraint_y_upper(v):
        res = np.empty(n)
        for i in range(n):
            y = v[3*i+1]
            r = v[3*i+2]
            res[i] = 1.0 - y - r # >= 0
        return res

    constraints.append({'type': 'ineq', 'fun': bound_constraint_x})
    constraints.append({'type': 'ineq', 'fun': bound_constraint_x_upper})
    constraints.append({'type': 'ineq', 'fun': bound_constraint_y})
    constraints.append({'type': 'ineq', 'fun': bound_constraint_y_upper})

    # (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
    # To avoid creating n*(n-1)/2 separate constraint functions (which is slow),
    # we can return a vector of all violations. SLSQP supports vector constraints.
    
    def non_overlap_constraint(v):
        # Extract arrays
        xs = v[0::3]
        ys = v[1::3]
        rs = v[2::3]
        
        # Compute all pairwise squared distances
        # Using broadcasting might be memory intensive for large N, but N=26 is small (26^2 ~ 676)
        # dist_sq[i, j] = (x_i-x_j)^2 + (y_i-y_j)^2
        diff_x = xs[:, np.newaxis] - xs[np.newaxis, :]
        diff_y = ys[:, np.newaxis] - ys[np.newaxis, :]
        dist_sq = diff_x**2 + diff_y**2
        
        # Sum of radii squared
        # sum_r[i, j] = (r_i + r_j)^2
        sum_r = (rs[:, np.newaxis] + rs[np.newaxis, :])**2

        # We only need upper triangle or lower triangle to avoid duplicates and self-check
        # Self check: dist_sq[i,i] = 0, sum_r[i,i] = (2r)^2. 0 - 4r^2 < 0. 
        # But i != j is what matters.
        
        mask = np.triu(np.ones((n, n), dtype=bool), k=1)
        violations = dist_sq[mask] - sum_r[mask]
        
        return violations

    constraints.append({'type': 'ineq', 'fun': non_overlap_constraint})
    
    # Objective: Maximize sum(r) -> Minimize -sum(r)
    def objective(v):
        rs = v[2::3]
        return -np.sum(rs)

            objective, 
            vars_init, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=constraints,
            options={'maxiter': 1000, 'ftol': 1e-9, 'disp': False}
        )
        vars_opt = result.x
    except Exception as e:
        # Fallback to initial if optimization fails
        vars_opt = vars_init
        print(f"Optimization failed: {e}")

    # --- 4. Extract Results ---
    x_opt = vars_opt[0::3]
    y_opt = vars_opt[1::3]
    r_opt = vars_opt[2::3]
    
    centers_opt = np.column_stack((x_opt, y_opt))
    radii_opt = r_opt

    # Given the constraints are exact, the result should be valid within tolerance.
    # However, numerical errors might occur.

    # but SLSQP with tight tolerances usually works.
    
    # Check for any tiny negative radii due to float errors
    radii_opt = np.maximum(radii_opt, 0.0)
    
    sum_radii = np.sum(radii_opt)
    
    return centers_opt, radii_opt, sum_radii
