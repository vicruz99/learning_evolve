import numpy as np
from scipy.optimize import linprog, differential_evolution, minimize

def solve_radii(centers):
    """Solve LP to find maximum radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    n_pairs = n * (n - 1) // 2
    n_bounds = 4 * n
    n_constraints = n_pairs + n_bounds
    
    A_ub = np.zeros((n_constraints, n))
    b_ub = np.zeros(n_constraints)
    
    idx = 0
    # Pairwise non-overlap constraints: r_i + r_j <= distance(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dist
            idx += 1
            
    # Boundary constraints: r_i <= x_i, 1-x_i, y_i, 1-y_i
    for i in range(n):
        x, y = centers[i]
        A_ub[idx, i] = 1.0
        b_ub[idx] = x
        idx += 1
        A_ub[idx, i] = 1.0
        b_ub[idx] = 1.0 - x
        idx += 1
        A_ub[idx, i] = 1.0
        b_ub[idx] = y
        idx += 1
        A_ub[idx, i] = 1.0
        b_ub[idx] = 1.0 - y
        idx += 1
        
    bounds_r = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method='highs')
    
    if res.success:
        return -res.fun, res.x
    return 0.0, np.zeros(n)

def objective(centers_flat):
    """Objective function for center optimization: negative sum of radii."""
    centers = centers_flat.reshape((26, 2))
    sum_r, _ = solve_radii(centers)
    return -sum_r

def run_packing():
    np.random.seed(42)
    
    # Initialize with a perturbed hexagonal lattice pattern
    centers_init = np.zeros((26, 2))
    idx = 0
    rows = 5
    cols_per_row = [5, 5, 6, 5, 5]
    dx, dy = 0.2, 0.17
    for r in range(rows):
        n_cols = cols_per_row[r]
        shift = dx / 2.0 if r % 2 == 1 else 0.0
        base_x = 0.5 - (n_cols * dx) / 2.0 + shift
        base_y = 0.5 - (rows * dy) / 2.0 + r * dy
        for c in range(n_cols):
            if idx < 26:
                centers_init[idx] = [base_x + c * dx, base_y]
                idx += 1
                
    centers_init = np.clip(centers_init, 0.05, 0.95)
    x0 = centers_init.flatten()
    bounds = [(0.0, 1.0)] * 52
    
    # Global optimization of center positions
    de_res = differential_evolution(
        objective, 
        bounds, 
        popsize=25, 
        maxiter=200, 
        tol=1e-7, 
        seed=42, 
        workers=1, 
        mutation=(0.5, 1.0), 
        recombination=0.8, 
        polish=False
    )
    
    best_centers = de_res.x.reshape((26, 2))
    
    # Local refinement to tighten packing
    lm_res = minimize(
        objective, 
        best_centers.flatten(), 
        method='Nelder-Mead', 
        options={'xatol': 1e-6, 'fatol': 1e-7, 'maxiter': 1500, 'adaptive': True}
    )
    
    optimal_centers = lm_res.x.reshape((26, 2))
    total_sum, radii = solve_radii(optimal_centers)
    
    return optimal_centers, radii, total_sum
