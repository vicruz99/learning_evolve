

To maximize the sum of radii for 26 circles in a unit square, we use a numerical optimization approach. We start with a 5x5 grid configuration (25 circles) and add a 26th circle, then employ the `SLSQP` algorithm to expand the circles until they are constrained by the square boundaries and mutual non-overlap.

### Strategy
1.  **Initialization**: Generate a base 5x5 grid of circles with a radius slightly smaller than 0.1 (e.g., 0.09) to allow for movement. Place the 26th circle at the center of the square.
2.  **Optimization**: Use `scipy.optimize.minimize` to maximize the sum of radii. 
    -   **Objective**: $-\sum r_i$ (minimizing negative sum is equivalent to maximizing sum).
    -   **Constraints**: 
        -   Boundary constraints: $r_i \le x_i \le 1-r_i$ and $r_i \le y_i \le 1-r_i$.
        -   Non-overlap constraints: Distance between centers $\ge$ sum of radii, implemented as $(x_i - x_j)^2 + (y_i - y_j)^2 \ge (r_i + r_j)^2$.
3.  **Validation**: Ensure all returned values are non-negative and satisfy the geometric constraints before outputting.

```python
import numpy as np
from scipy.optimize import minimize

def get_initial_guess(n):
    """Create a 5x5 grid plus one extra circle for 26 circles."""
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    r_init = 0.09
    centers[0:25] = np.array([(x * 0.2 + 0.1, y * 0.2 + 0.1) for y in range(5) for x in range(5)])
    centers[25] = [0.5, 0.5]
    radii[:] = r_init
    return centers, radii

def constraints_factory(centers, radii, n):
    """Define constraints for the SLSQP optimizer."""
    constraints = []
    # Boundary constraints: r <= x <= 1-r and r <= y <= 1-r
    for i in range(n):
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: v[2*i] - v[2*i + n]})       # x >= r
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: 1 - v[2*i] - v[2*i + n]}) # x <= 1-r
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: v[2*i + 1] - v[2*i + n]}) # y >= r
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: 1 - v[2*i + 1] - v[2*i + n]}) # y <= 1-r
    
    # Non-overlap constraints: (x_i - x_j)^2 + (y_i - y_j)^2 >= (r_i + r_j)^2
    for i in range(n):
        for j in range(i + 1, n):
            def non_overlap(v, i=i, j=j):
                xi, yi = v[2*i], v[2*i + 1]
                xj, yj = v[2*j], v[2*j + 1]
                ri, rj = v[2*i + n], v[2*j + n]
                return (xi - xj)**2 + (yi - yj)**2 - (ri + rj)**2
            constraints.append({'type': 'ineq', 'fun': non_overlap})
            
    return constraints

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    centers, radii = get_initial_guess(n)
    
    # Initial vector: [x1, y1, ..., xn, yn, r1, r2, ..., rn]
    x0 = np.concatenate([centers.flatten(), radii])
    
    # Bounds: x, y in [0, 1], r in [0, 1]
    bounds = [(0, 1)] * (2 * n) + [(0, 1)] * n
    
    cons = constraints_factory(centers, radii, n)
    
    result = minimize(lambda v: -np.sum(v[2*n:]), x0, method='SLSQP', 
                     bounds=bounds, constraints=cons, 
                     options={'maxiter': 500, 'ftol': 1e-9})
    
    if result.success:
        final_x = result.x[:2*n].reshape(n, 2)
        final_r = result.x[2*n:]
        # Clean up very small radii
        final_r[final_r < 1e-9] = 0
        return final_x, final_r, float(np.sum(final_r))
    else:
        return centers, radii, 0.0
```
