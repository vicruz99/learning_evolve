                overlaps = dists[mask] < (min_dists[mask] - 1e-9)
                if np.any(overlaps):
                    valid = False
            
            if valid:
                current_sum = np.sum(r)
                if current_sum > best_sum:
                    best_sum = current_sum
                    best_centers = c.copy()
                    best_radii = r.copy()
        except Exception:
            continue
            
    if best_centers is not None:
        return best_centers, best_radii, float(best_sum)
    
    # Fallback solution: 5x5 grid + 1 small circle in a gap
    # This guarantees a valid packing with sum ~2.54
    centers = []
    radii = []
    # 5x5 grid of radius 0.1
    for i in range(5):
        for j in range(5):
            centers.append([0.1 + i * 0.2, 0.1 + j * 0.2])
            radii.append(0.1)
    # Add one small circle in the gap at (0.4, 0.4)
    # Distance to nearest grid points (0.3,0.3) etc is ~0.1414
    # Radius 0.1 leaves ~0.0414 space.
    centers.append([0.4, 0.4])
    radii.append(0.04)
    
    return np.array(centers), np.array(radii), float(np.sum(radii))
