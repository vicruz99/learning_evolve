from scipy.optimize import minimize

N_CIRCLES = 26

    """Objective: maximize radius r => minimize -r"""
    return -vars[-1]

def bound_func(vars):
    """Boundary constraints: r <= coord <= 1-r => coord - r >= 0 and 1-r - coord >= 0"""

    r = vars[-1]

        centers[:, 0] - r,

        1 - r - centers[:, 1]

            cons[idx] = dist - 2 * r
            idx += 1
