import numpy as np
from scipy.optimize import minimize


def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26

    centers, radii = init_hexagonal(n)

    best_centers, best_radii, best_sum = optimize_iterative(centers, radii, n)

    return best_centers, best_radii, best_sum


def init_hexagonal(n):
    row_counts = [6, 5, 6, 5, 4]
    centers = []
    r = 0.085

    y = r
    for row_idx, count in enumerate(row_counts):
        offset = r * 0.5 if row_idx % 2 == 1 else 0
        for j in range(count):
            x = r + offset + j * 2 * r
            centers.append([x, y])
        y += r * np.sqrt(3)

    centers = np.array(centers[:n])
    radii = np.ones(n) * r

    max_x = np.max(centers[:, 0]) + r
    max_y = np.max(centers[:, 1]) + r
    scale = min((0.99) / max_x, (0.99) / max_y)
    centers *= scale
    radii *= scale

    return centers, radii


def optimize_iterative(centers, radii, n):
    best_centers = centers.copy()
    best_radii = radii.copy()
    best_sum = np.sum(radii)

    np.random.seed(42)

    for iteration in range(8):
        current_centers = best_centers.copy()
        current_radii = best_radii.copy()

        perturbation_scale = 0.003 * (iteration + 1)
        current_centers += np.random.randn(n, 2) * perturbation_scale
        current_radii *= (1 + np.random.randn(n) * 0.01 * (iteration + 1))
        current_radii = np.maximum(current_radii, 1e-6)

        current_centers = np.clip(current_centers, 0.001, 0.999)

        result = optimize_slsqp(current_centers, current_radii, n)

        if result[2] > best_sum:
            best_centers = result[0]
            best_radii = result[1]
            best_sum = result[2]

    return best_centers, best_radii, best_sum


def optimize_slsqp(centers, radii, n):
    x0 = np.concatenate([centers.flatten(), radii])

    bounds = [(0, 1)] * (2 * n) + [(1e-8, 0.5)] * n

    constraints = []
    for i in range(n):
        constraints.append({'type': 'ineq', 'fun': con_x_ge_r, 'args': (n, i)})
        constraints.append({'type': 'ineq', 'fun': con_1_minus_x_ge_r, 'args': (n, i)})
        constraints.append({'type': 'ineq', 'fun': con_y_ge_r, 'args': (n, i)})
        constraints.append({'type': 'ineq', 'fun': con_1_minus_y_ge_r, 'args': (n, i)})

    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({'type': 'ineq', 'fun': con_no_overlap, 'args': (n, i, j)})

    result = minimize(
        obj_neg_sum, x0, args=(n,),
        bounds=bounds, constraints=constraints,
        method='SLSQP',
        options={'maxiter': 2000, 'ftol': 1e-12, 'disp': False}
    )

    if result.success:
        final_centers = result.x[:2 * n].reshape(n, 2)
        final_radii = result.x[2 * n:]
    else:
        final_centers = x0[:2 * n].reshape(n, 2)
        final_radii = x0[2 * n:]

    final_centers = np.clip(final_centers, 0, 1)
    final_radii = np.maximum(final_radii, 1e-10)

    return final_centers, final_radii, np.sum(final_radii)


def obj_neg_sum(x, n):
    return -np.sum(x[2 * n:])


def con_x_ge_r(x, args):
    n, i = args
    return x[2 * i] - x[2 * n + i]


def con_1_minus_x_ge_r(x, args):
    n, i = args
    return 1 - x[2 * i] - x[2 * n + i]


def con_y_ge_r(x, args):
    n, i = args
    return x[2 * i + 1] - x[2 * n + i]


def con_1_minus_y_ge_r(x, args):
    n, i = args
    return 1 - x[2 * i + 1] - x[2 * n + i]


def con_no_overlap(x, args):
    n, i, j = args
    xi, yi = x[2 * i], x[2 * i + 1]
    xj, yj = x[2 * j], x[2 * j + 1]
    ri, rj = x[2 * n + i], x[2 * n + j]
    dx = xi - xj
    dy = yi - yj
    dist = np.sqrt(dx * dx + dy * dy)
    return dist - ri - rj
