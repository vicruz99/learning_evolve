from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    def evaluate_sum_radii(params):
        centers = params[:n*2].reshape(n, 2)
        radii = params[n*2:]
        
        # Boundary constraints (inside [0,1])
        for i in range(n):
            r = radii[i]
            x, y = centers[i]
            if x < r or x > 1 - r or y < r or y > 1 - r:
                return -100.0 # Penalty

        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                if dist < radii[i] + radii[j]:
                    return -100.0 # Penalty
                    
        return -np.sum(radii) # Negative for minimization

        centers = params[:n*2].reshape(n, 2)

            if x < r: penalty += 1e5 * (r - x)**2
            if x > 1 - r: penalty += 1e5 * (x - (1 - r))**2

                if dist < sum_r:
                    penalty += 1e5 * (sum_r - dist)**2
