import numpy as np
from scipy.optimize import minimize


def hexagonal_initial(n):
    """Generate a hexagonal lattice initial placement."""
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.06
    
    idx = 0
    r_spacing = 0.12
    y_spacing = np.sqrt(3) / 2 * r_spacing
    
    y = r_spacing
    while idx < n:
        is_offset = (int((y - r_spacing) / y_spacing) if y > r_spacing else 0) % 2
        x = r_spacing if not is_offset else r_spacing + r_spacing / 2
        while x < 1.0 - r_spacing and idx < n:
            centers[idx, 0] = x
            centers[idx, 1] = y
            idx += 1
            x += r_spacing
        y += y_spacing
    
    # If we didn't fill all positions, just use what we have
    return centers, radii


def grid_initial(n):
    """Generate a grid-based initial placement."""
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.05
    
    # Try to arrange in roughly sqrt(n) x sqrt(n) grid
    rows = int(np.ceil(np.sqrt(n)))
    cols = int(np.ceil(n / rows))
    
    # Adjust rows/cols to be balanced
    if cols > rows + 1:
        rows += 1
        cols = int(np.ceil(n / rows))
    
    spacing_x = 1.0 / (cols + 1)
    spacing_y = 1.0 / (rows + 1)
    
    idx = 0
    for j in range(rows):
        for i in range(cols):
            if idx >= n:
                break
            centers[idx, 0] = (i + 1) * spacing_x
            centers[idx, 1] = (j + 1) * spacing_y
            radii[idx] = 0.05
            idx += 1
    
    return centers, radii


def perturbed_initial(n, seed=42):
    """Generate a perturbed grid initial placement."""
    rng = np.random.RandomState(seed)
    centers, radii = grid_initial(n)
    perturbation = rng.uniform(-0.03, 0.03, size=centers.shape)
    centers = np.clip(centers + perturbation, 0.05, 0.95)
    return centers, radii


def known_packing_initial(n):

            x = 0.11 + i * 0.19
            y = 0.11 + j * 0.19

        if idx < 25:
            centers[idx] = pos
        else:
            # Place the extra circle in a strategic gap
            centers[idx] = [0.255, 0.255]  # Between grid points
    
    radii = np.ones(n) * 0.06
    
    return centers, radii


def all_inequality_constraints(x, n):
    """Compute all inequality constraints as a vector."""
    centers = x[:2 * n].reshape(n, 2)
    radii = x[2 * n:]
    
    constraint_list = []
    
    # Non-overlap constraints: dist^2 >= (r1 + r2)^2
    for i in range(n):
        for j in range(i + 1, n):
            dist_sq = np.sum((centers[i] - centers[j]) ** 2)
            min_dist_sq = (radii[i] + radii[j]) ** 2
            constraint_list.append(dist_sq - min_dist_sq)
    
    # Boundary constraints
    for i in range(n):
        # Left: x - r >= 0
        constraint_list.append(centers[i, 0] - radii[i])
        # Right: 1 - x - r >= 0
        constraint_list.append(1.0 - centers[i, 0] - radii[i])
        # Bottom: y - r >= 0
        constraint_list.append(centers[i, 1] - radii[i])
        # Top: 1 - y - r >= 0
        constraint_list.append(1.0 - centers[i, 1] - radii[i])
    
    # Non-negative radii
    for i in range(n):
        constraint_list.append(radii[i])
    
    return np.array(constraint_list)


def objective_function(x, n):
    """Objective: maximize sum of radii (minimize negative sum)."""
    radii = x[2 * n:]
    return -np.sum(radii)


def optimize_packing(centers, radii, n=26, max_iter=3000, verbose=False):
    """Optimize circle packing using SLSQP."""
    x0 = np.concatenate([centers.flatten(), radii])
    
    # Create constraint dictionary
    cons = {'type': 'ineq', 'fun': lambda x: all_inequality_constraints(x, n)}
    
    # Bounds: centers in [0, 1], radii in [0, 0.5]
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
    
    # Run optimization
    result = minimize(
        objective_function,
        x0,
        args=(n,),
        method='SLSQP',
        constraints=cons,
        bounds=bounds,
        options={
            'maxiter': max_iter,
            'ftol': 1e-14,
            'disp': verbose,
            'iprint': -1,
        }
    )
    
    centers_opt = result.x[:2 * n].reshape(n, 2)
    radii_opt = np.maximum(result.x[2 * n:], 0.0)
    sum_radii = np.sum(radii_opt)
    
    if verbose:
        print(f"Optimization result: success={result.success}, sum_radii={sum_radii:.6f}")
        print(f"Message: {result.message}")
    
    return centers_opt, radii_opt, sum_radii


def refine_packing(centers, radii, n=26, iterations=3):
    """Refine packing by running optimization multiple times from perturbed positions."""
    best_centers = centers.copy()
    best_radii = radii.copy()

    for iteration in range(iterations):
        # Small perturbation

        perturbed_radii = radii.copy()

        noise = rng.uniform(-0.001, 0.001, size=perturbed_centers.shape)

        # Slightly reduce radii to ensure feasibility
        perturbed_radii *= 0.98
        
        centers_opt, radii_opt, sum_opt = optimize_packing(

        )
        
        if sum_opt > best_sum:
            best_centers = centers_opt
            best_radii = radii_opt
            best_sum = sum_opt
    
    return best_centers, best_radii, best_sum
