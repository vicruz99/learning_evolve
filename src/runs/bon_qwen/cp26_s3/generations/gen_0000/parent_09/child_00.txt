import numpy as np
from scipy.optimize import minimize

def constraint_func(v, n):
    """
    Computes all packing constraints.
    Returns an array where each element must be >= 0 for a valid packing.
    """
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    c1 = x - r
    c2 = 1.0 - x - r
    c3 = y - r
    c4 = 1.0 - y - r
    
    # Pairwise non-overlap constraints: dist^2 >= (r_i + r_j)^2
    xx = x[:, None] - x[None, :]
    yy = y[:, None] - y[None, :]
    rr = r[:, None] + r[None, :]
    
    diff = xx**2 + yy**2 - rr**2
    
    # Extract upper triangle (j > i) to avoid duplicates and self-checks
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    c_pairwise = diff[mask]
    
    return np.concatenate([c1, c2, c3, c4, c_pairwise])

def objective(v, n):
    """
    Objective function to minimize: negative sum of radii.
    Maximizing sum(r_i) is equivalent to minimizing -sum(r_i).
    """
    return -np.sum(v[2::3])

def run_packing():
    n = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Setup optimization constraints and bounds
    cons = {'type': 'ineq', 'fun': constraint_func, 'args': (n,)}
    bounds = [(0, 1)] * n + [(0, 1)] * n + [(0, 0.5)] * n
    
    # Run optimization from multiple diverse starting points
    for seed in range(10):
        np.random.seed(seed)
        
        # Generate hexagonal-ish initial layout
        centers = []
        row_counts = [5, 5, 5, 5, 6]  # Sums to 26
        y_vals = np.linspace(0.15, 0.85, len(row_counts))
        
        for i, cnt in enumerate(row_counts):
            y = y_vals[i]
            # Shift odd rows to create hexagonal packing structure
            shift = 0.05 if i % 2 == 1 else 0.0
            x_vals = np.linspace(0.05 + shift, 0.95 + shift, cnt)
            centers.extend(zip(x_vals, [y]*cnt))
            
        centers = np.array(centers)
        
        # Add controlled random perturbation to break symmetry
        noise = np.random.randn(n, 2) * 0.01
        centers = np.clip(centers + noise, 0.05, 0.95)
        
        # Construct initial variable vector [x1, y1, r1, x2, y2, r2, ...]
        x0 = np.zeros(3 * n)
        x0[0::3] = centers[:, 0]
        x0[1::3] = centers[:, 1]
        x0[2::3] = 0.05  # Start with small feasible radii
        
        try:
            res = minimize(
                objective, 
                x0, 
                args=(n,), 
                method='SLSQP',
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 1000, 'ftol': 1e-12}
            )
            
            if res.success:
                current_sum = -res.fun
                if current_sum > best_sum:
                    best_sum = current_sum
                    best_centers = np.column_stack((res.x[0::3], res.x[1::3]))
                    best_radii = res.x[2::3]
        except Exception:
            continue
            
    # Fallback if optimization fails to produce a valid result
    if best_centers is None:
        best_centers = centers
        best_radii = np.full(n, 0.04)
        best_sum = np.sum(best_radii)
        
    return best_centers, best_radii, best_sum
