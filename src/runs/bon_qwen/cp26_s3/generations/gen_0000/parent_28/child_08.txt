import numpy as np
from scipy.optimize import minimize

# Constants
N_CIRCLES = 26
TOLERANCE = 1e-12

def get_distance(c1, c2):
    return np.sqrt((c1[0] - c2[0])**2 + (c1[1] - c2[1])**2)

def check_validity(centers, radii):
    """
    Returns True if packing is valid, False otherwise.
    Used for internal checks.
    """
    n = centers.shape[0]
    
    # Check NaN
    if np.isnan(centers).any() or np.isnan(radii).any():
        return False
    
    # Check radii non-negative
    if np.any(radii < 0):
        return False
    
    # Check boundaries
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -TOLERANCE or x + r > 1 + TOLERANCE or \
           y - r < -TOLERANCE or y + r > 1 + TOLERANCE:
            return False
            
    # Check overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - TOLERANCE:
                return False
                
    return True

def force_directed_packing(centers, radii, iterations=2000, temp=0.1):
    """
    A simple force-directed layout to expand circles and resolve overlaps.
    Increases radii and pushes centers apart.
    """
    n = centers.shape[0]
    # Convert to list for easier mutation if needed, but numpy arrays are mutable
    # We work on copies
    cur_centers = centers.copy()
    cur_radii = radii.copy()
    
    for _ in range(iterations):
        # Try to expand all radii slightly
        expansion = 1e-5 * (1 + 0.5 * np.random.rand())
        cur_radii += expansion
        
        # Resolve overlaps by pushing centers apart
        for _ in range(10): # Multiple passes per expansion step
            moved = False
            for i in range(n):
                for j in range(i + 1, n):
                    dx = cur_centers[j, 0] - cur_centers[i, 0]
                    dy = cur_centers[j, 1] - cur_centers[i, 1]
                    dist = np.hypot(dx, dy)
                    
                    min_dist = cur_radii[i] + cur_radii[j]
                    
                    if dist < min_dist and dist > 1e-9:
                        # Overlap detected
                        overlap = min_dist - dist
                        # Normalize direction
                        nx = dx / dist
                        ny = dy / dist
                        
                        # Push apart equally
                        shift = overlap / 2.0 + 1e-6 # Small jitter
                        
                        cur_centers[i, 0] -= nx * shift
                        cur_centers[i, 1] -= ny * shift
                        cur_centers[j, 0] += nx * shift
                        cur_centers[j, 1] += ny * shift
                        moved = True
                        
                        # Clamp to boundaries immediately to avoid getting stuck
                        cur_centers[i, 0] = np.clip(cur_centers[i, 0], cur_radii[i], 1 - cur_radii[i])
                        cur_centers[i, 1] = np.clip(cur_centers[i, 1], cur_radii[i], 1 - cur_radii[i])
                        cur_centers[j, 0] = np.clip(cur_centers[j, 0], cur_radii[j], 1 - cur_radii[j])
                        cur_centers[j, 1] = np.clip(cur_centers[j, 1], cur_radii[j], 1 - cur_radii[j])
            
            if not moved:
                break
        
        # Clamp boundaries after each expansion step
        for i in range(n):
            cur_centers[i, 0] = np.clip(cur_centers[i, 0], cur_radii[i], 1 - cur_radii[i])
            cur_centers[i, 1] = np.clip(cur_centers[i, 1], cur_radii[i], 1 - cur_radii[i])
            
    return cur_centers, cur_radii

def run_packing():
    np.random.seed(42)
    n = N_CIRCLES
    
    # 1. Initialization: Hexagonal Grid Pattern
    # We try to place centers in a hexagonal pattern.
    # Spacing for radius ~0.08 initially.
    # Let's generate points and pick the best 26 that fit?
    # Or just generate a dense cloud.
    
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.05
    
    # Generate a grid of points
    # 10x10 grid is 100 points. We pick 26.
    # But picking random 26 might be sparse.
    # Let's pick a dense subset.
    # Actually, let's just place them in a hex lattice structure.
    
    # Hex lattice parameters
    # Row spacing h = r * sqrt(3) * 2 ? No, h = d * sqrt(3)/2 = r * sqrt(3)
    # Let's use r=0.08. d=0.16.
    r_start = 0.08
    d_start = 2 * r_start
    h_start = d_start * np.sqrt(3) / 2
    
    points = []
    y = r_start
    row_idx = 0
    while y <= 1 - r_start:
        x = r_start
        if row_idx % 2 == 1:
            x += d_start / 2 # Shift odd rows
        while x <= 1 - r_start:
            points.append([x, y])
            x += d_start
        y += h_start
        row_idx += 1
    
    # If we don't have enough points, expand grid or use fallback
    if len(points) < n:
        # Fallback to random dense packing
        points = []
        attempts = 0
        while len(points) < n and attempts < 10000:
            cx = np.random.uniform(r_start, 1 - r_start)
            cy = np.random.uniform(r_start, 1 - r_start)
            valid = True
            for p in points:
                if np.hypot(cx - p[0], cy - p[1]) < 1.5 * r_start: # loose check
                    valid = False
                    break
            if valid:
                points.append([cx, cy])
            attempts += 1
        # If still not enough, just fill grid
        if len(points) < n:
            pts_grid = []
            for i in range(6):
                for j in range(5):
                    pts_grid.append([0.1 + i*0.15, 0.1 + j*0.15])
            points = pts_grid[:n]

    centers, radii = force_directed_packing(centers, radii, iterations=3000)

    # Variables: x1, y1, r1, x2, y2, r2, ...
    # Total 3*n variables
    x0 = np.hstack([centers.flatten(), radii])
    
    def objective(vars):
        # Maximize sum of radii -> Minimize -sum(r)
        r = vars[2*n:]
        return -np.sum(r)

    def constraint_penalty(vars):
        c = vars[:2*n].reshape(n, 2)
        r = vars[2*n:]
        penalty = 0.0
        weight = 1000.0
        
        # Boundary constraints: x >= r, x <= 1-r => r <= x <= 1-r
        # Violation: max(0, r - x) + max(0, x + r - 1)
        # Same for y
        
        # Vectorized boundary penalty
        # x - r >= 0
        viol = np.maximum(0, r - c[:, 0])
        penalty += np.sum(viol**2)
        # x + r <= 1
        viol = np.maximum(0, c[:, 0] + r - 1)
        penalty += np.sum(viol**2)
        # y - r >= 0
        viol = np.maximum(0, r - c[:, 1])
        penalty += np.sum(viol**2)
        # y + r <= 1
        viol = np.maximum(0, c[:, 1] + r - 1)
        penalty += np.sum(viol**2)
        
        # Overlap constraints
        # dist(i, j) >= r_i + r_j
        # violation = max(0, r_i + r_j - dist)
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.hypot(c[i, 0] - c[j, 0], c[i, 1] - c[j, 1])
                viol = r[i] + r[j] - dist
                if viol > 0:
                    penalty += viol**2
                    
        return weight * penalty

    def obj_with_penalty(vars):
        return objective(vars) + constraint_penalty(vars)

    # Bounds
    # Centers in [0, 1], Radii in [0, 0.5]
    bounds = []
    for _ in range(2 * n):
        bounds.append((0.0, 1.0))
    for _ in range(n):
        bounds.append((0.0, 0.5))

    # Run optimization
    # Use L-BFGS-B
    try:
        res = minimize(obj_with_penalty, x0, method='L-BFGS-B', bounds=bounds, 
                       options={'maxiter': 2000, 'ftol': 1e-12, 'gtol': 1e-10})
        final_vars = res.x
    except Exception:
        # Fallback to current if optimization fails
        final_vars = x0

    final_centers = final_vars[:2*n].reshape(n, 2)
    final_radii = final_vars[2*n:]
    
    # 4. Final Validation and Correction
    # If there are small violations, we might need to shrink radii slightly or fix positions.
    # The penalty method should have handled it, but let's be safe.
    
    # Check validity
    if not check_validity(final_centers, final_radii):
        # If invalid, try to shrink radii slightly until valid
        # This is a safeguard
        scale = 1.0
        step = 0.001
        while not check_validity(final_centers, final_radii * scale) and scale > 0.5:
            scale -= step
        final_radii = final_radii * scale
        # After scaling, check if we can improve by re-running force directed with smaller radii?
        # For simplicity, just return the scaled valid packing.
        
    # Ensure strict bounds [0,1] for centers and non-negative radii
    final_centers = np.clip(final_centers, 0, 1)
    final_radii = np.maximum(final_radii, 0)
    
    sum_radii = np.sum(final_radii)
    
    return final_centers, final_radii, sum_radii
