import numpy as np
import math

def compute_max_radii(centers, radii):
    """Compute maximum possible radius for each circle."""
    n = len(radii)
    new_radii = np.zeros(n)
    for i in range(n):
        max_r = min(centers[i, 0], 1 - centers[i, 0],
                   centers[i, 1], 1 - centers[i, 1])
        for j in range(n):
            if i != j:
                diff = centers[i] - centers[j]
                dist = math.sqrt(diff[0]**2 + diff[1]**2)
                max_r = min(max_r, dist - radii[j])
        new_radii[i] = max(0, max_r)
    return new_radii

def optimize_packing(centers_init, num_iterations=10000):
    """Optimize a packing starting from given centers."""

    centers = centers_init.copy()

                        force += diff / (dist + 1e-10)
