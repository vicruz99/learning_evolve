import numpy as np
from scipy.optimize import linprog

def compute_optimal_radii(centers):
    """
    Given fixed centers, solve the LP to find radii that maximize sum(r_i)
    subject to non-overlap and boundary constraints.
    """
    n = centers.shape[0]
    n_pairs = n * (n - 1) // 2
    n_bnd = 4 * n
    
    # Constraint matrices
    A_ub = np.zeros((n_pairs + n_bnd, n))
    b_ub = np.zeros(n_pairs + n_bnd)
    
    idx = 0
    # Pairwise non-overlap constraints: r_i + r_j <= distance(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.linalg.norm(centers[i] - centers[j])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = d
            idx += 1
            
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        x, y = centers[i]
        vals = [x, 1.0 - x, y, 1.0 - y]
        for v in vals:
            A_ub[idx, i] = 1.0
            b_ub[idx] = v
            idx += 1
            
    bounds = [(0.0, None)] * n
    
    # Linprog minimizes c^T x, so we use c = -1 to maximize sum
    res = linprog(c=-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        radii = np.maximum(res.x, 0.0)  # Handle potential tiny negative floats
        return radii, np.sum(radii)
    else:
        # Fallback should rarely trigger
        r_fallback = 0.05
        return np.full(n, r_fallback), n * r_fallback

def run_packing():
    np.random.seed(42)
    n = 26
    
    # 1. Initialize with a staggered hexagonal-like grid
    centers = np.zeros((n, 2))
    rows = [5, 4, 5, 4, 5, 3]  # Total 26 circles
    idx = 0
    r_init = 0.13
    for r_idx, count in enumerate(rows):
        y = 0.08 + r_idx * r_init * np.sqrt(3)
        x_start = 0.08 + (0.5 if r_idx % 2 == 1 else 0.0) * 2 * r_init
        for c in range(count):
            if idx < n:
                centers[idx, 0] = x_start + c * 2 * r_init
                centers[idx, 1] = y
                idx += 1
                
    # Add small random jitter to break symmetry
    centers += np.random.randn(*centers.shape) * 0.01
    centers = np.clip(centers, 0.0, 1.0)
    
    # 2. Simulated Annealing to optimize centers
    best_centers = centers.copy()
    _, best_obj = compute_optimal_radii(centers)
    
    temp = 0.03
    step = 0.06
    iterations = 2000
    
    for it in range(iterations):
        i = np.random.randint(n)
        move = np.random.randn(2) * step
        new_centers = best_centers.copy()
        new_centers[i] = np.clip(new_centers[i] + move, 0.0, 1.0)
        
        if np.linalg.norm(move) < 1e-6:
            continue
            
        _, obj = compute_optimal_radii(new_centers)
        
        accept = False
        if obj > best_obj:
            accept = True
        else:
            diff = obj - best_obj
            if temp > 1e-8 and np.random.rand() < np.exp(diff / temp):
                accept = True
                
        if accept:
            best_centers = new_centers
            best_obj = obj
            
        temp *= 0.9995
        step *= 0.9995
        
    # 3. Final evaluation
    radii, sum_r = compute_optimal_radii(best_centers)
    return best_centers, radii, sum_r
