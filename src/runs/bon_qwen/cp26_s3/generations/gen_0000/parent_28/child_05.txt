import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26
PENALTY_WEIGHT = 500000.0


def compute_objective(state):
    radii = state[:N_CIRCLES]
    cx = state[N_CIRCLES:2 * N_CIRCLES]
    cy = state[2 * N_CIRCLES:3 * N_CIRCLES]

    obj = -np.sum(radii)

    for i in range(N_CIRCLES):
        r_i = radii[i]
        x_i = cx[i]
        y_i = cy[i]

        obj += PENALTY_WEIGHT * max(0.0, r_i - x_i) ** 2
        obj += PENALTY_WEIGHT * max(0.0, r_i - (1.0 - x_i)) ** 2
        obj += PENALTY_WEIGHT * max(0.0, r_i - y_i) ** 2
        obj += PENALTY_WEIGHT * max(0.0, r_i - (1.0 - y_i)) ** 2
        obj += PENALTY_WEIGHT * max(0.0, -r_i) ** 2

    for i in range(N_CIRCLES):
        for j in range(i + 1, N_CIRCLES):
            dx = cx[i] - cx[j]
            dy = cy[i] - cy[j]
            dist = np.sqrt(dx * dx + dy * dy)
            overlap = radii[i] + radii[j] - dist
            obj += PENALTY_WEIGHT * max(0.0, overlap) ** 2

    return obj


def create_hexagonal_state():
    row_counts = [5, 4, 5, 4, 5, 3]
    n_rows = len(row_counts)

    centers = []
    for row_i, count in enumerate(row_counts):
        y = (row_i + 0.5) / n_rows
        for col_i in range(count):
            if row_i % 2 == 0:
                x = (col_i + 0.5) / count
            else:
                x = (col_i + 1.0) / count
            centers.append([x, y])

    radii = np.full(N_CIRCLES, 0.07)
    centers_arr = np.array(centers)

    state = np.concatenate([radii, centers_arr[:, 0], centers_arr[:, 1]])
    return state


def create_alternative_state_1():

def perturb_state(base_state, seed):
    rng = np.random.RandomState(seed)
    state = base_state.copy()
    state[:N_CIRCLES] = np.maximum(state[:N_CIRCLES] + rng.randn(N_CIRCLES) * 0.003, 0.005)
    state[N_CIRCLES:2 * N_CIRCLES] = np.clip(
        state[N_CIRCLES:2 * N_CIRCLES] + rng.randn(N_CIRCLES) * 0.03, 0.02, 0.98
    )
    state[2 * N_CIRCLES:3 * N_CIRCLES] = np.clip(
        state[2 * N_CIRCLES:3 * N_CIRCLES] + rng.randn(N_CIRCLES) * 0.03, 0.02, 0.98
    )
    return state


def optimize_from_state(state, bounds):
    result = minimize(
        compute_objective,
        state,
        method="L-BFGS-B",
        bounds=bounds,
        options={"maxiter": 100000, "ftol": 1e-15, "gtol": 1e-12, "maxfun": 200000},
    )
    return result


def refine_packing(centers, radii, bounds):
    state = np.concatenate([radii, centers[:, 0], centers[:, 1]])
    result = optimize_from_state(state, bounds)
    new_radii = np.maximum(result.x[:N_CIRCLES], 0.0)
    new_centers = np.column_stack(
        [result.x[N_CIRCLES:2 * N_CIRCLES], result.x[2 * N_CIRCLES:3 * N_CIRCLES]]
    )
    return new_centers, new_radii


def run_packing():
    np.random.seed(42)

    bounds = [(0.001, 0.5)] * N_CIRCLES + [(0.0, 1.0)] * N_CIRCLES + [(0.0, 1.0)] * N_CIRCLES

    initial_states = [
        create_hexagonal_state(),
        create_alternative_state_1(),
        create_alternative_state_2(),
        create_alternative_state_3(),
    ]

    best_obj = float("inf")
    best_state = None

    all_starts = []
    for base in initial_states:
        for s in range(15):
            all_starts.append(perturb_state(base, s))

    for idx, start_state in enumerate(all_starts):
        result = optimize_from_state(start_state, bounds)

        if result.fun < best_obj:
            best_obj = result.fun
            best_state = result.x.copy()

    radii = np.maximum(best_state[:N_CIRCLES], 0.0)
    cx = best_state[N_CIRCLES:2 * N_CIRCLES]
    cy = best_state[2 * N_CIRCLES:3 * N_CIRCLES]
    centers = np.column_stack([cx, cy])

    for _ in range(3):
        centers, radii = refine_packing(centers, radii, bounds)

    radii = np.maximum(radii, 0.0)

    return centers, radii, float(np.sum(radii))
