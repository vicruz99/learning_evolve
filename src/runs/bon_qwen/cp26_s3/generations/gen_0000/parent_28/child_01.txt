import numpy as np
import math
import random

def run_packing() -> tuple:
    # Constants
    N = 26
    num_iterations = 15000
    init_temp = 1.0
    min_temp = 1e-6
    cooling_rate = 0.9995
    
    # Helper to calculate overlap cost
    # Returns a high cost if overlapping, otherwise 0
    def get_overlap_cost(centers, radii):
        cost = 0.0
        for i in range(N):
            x, y = centers[i]
            r = radii[i]
            # Boundary cost
            if x < r: cost += (r - x) ** 2
            if x > 1 - r: cost += (x - (1 - r)) ** 2
            if y < r: cost += (r - y) ** 2
            if y > 1 - r: cost += (y - (1 - r)) ** 2
            
            # Pairwise overlap cost
            for j in range(i + 1, N):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = math.sqrt(dx*dx + dy*dy)
                min_dist = radii[i] + radii[j]
                if dist < min_dist:
                    cost += (min_dist - dist) ** 2
        return cost

    # Initialize with a perturbed grid to ensure valid start
    centers = np.zeros((N, 2))
    radii = np.ones(N) * 0.09 # Slightly smaller than 0.1 to fit 26 roughly

    idx = 0
    for row in range(5):
        for col in range(5):
            if idx < 25:
                centers[idx, 0] = 0.1 + col * 0.2 + np.random.uniform(-0.02, 0.02)
                centers[idx, 1] = 0.1 + row * 0.2 + np.random.uniform(-0.02, 0.02)
                idx += 1
    
    # Place 26th circle in center-ish gap
    centers[25, 0] = 0.5 + np.random.uniform(-0.05, 0.05)
    centers[25, 1] = 0.5 + np.random.uniform(-0.05, 0.05)

    # We will let the optimizer grow them
    
    current_centers = centers.copy()
    current_radii = radii.copy()
    current_cost = get_overlap_cost(current_centers, current_radii)
    current_sum_radii = np.sum(current_radii)
    
    best_centers = current_centers.copy()
    best_radii = current_radii.copy()
    best_sum_radii = current_sum_radii
    
    temp = init_temp
    
    for step in range(num_iterations):
        # Pick random circle
        i = random.randint(0, N - 1)
        
        # Decide move type: move center or change radius
        move_type = random.random()
        
        trial_centers = current_centers.copy()
        trial_radii = current_radii.copy()

            # Perturb center
            dx = np.random.uniform(-0.02, 0.02) * (temp / init_temp + 0.1)
            dy = np.random.uniform(-0.02, 0.02) * (temp / init_temp + 0.1)
            trial_centers[i, 0] += dx
            trial_centers[i, 1] += dy
            
            # Clamp to [0, 1]
            trial_centers[i, 0] = max(0.0, min(1.0, trial_centers[i, 0]))
            trial_centers[i, 1] = max(0.0, min(1.0, trial_centers[i, 1]))
            
        else: # 30% chance to change radius
            # Try to increase radius slightly, or decrease if stuck
            # Bias towards increasing to maximize sum
            change = np.random.uniform(-0.005, 0.015) * (temp / init_temp + 0.05)
            trial_radii[i] = max(0.01, trial_radii[i] + change)
            
        # Calculate new cost and sum
        new_cost = get_overlap_cost(trial_centers, trial_radii)
        new_sum_radii = np.sum(trial_radii)
        
        # Score: We want to minimize cost and maximize sum_radii.
        # Let's define a score: - (cost + weight * (max_possible_sum - new_sum_radii))
        # Actually, easier: Accept if valid and better sum.

        # Combined objective: Maximize Sum, Penalize Cost heavily.
        # Objective = new_sum_radii - penalty * new_cost
        penalty = 100.0
        current_obj = current_sum_radii - penalty * current_cost
        new_obj = new_sum_radii - penalty * new_cost
        
        delta = new_obj - current_obj
        
        # Acceptance
        if delta > 0:
            accept = True
        else:
            # Boltzmann factor
            # We need a scale for delta. Sum of radii is ~2.5, Cost can be large.
            # If cost is 0, delta is just change in sum.
            # If cost > 0, we usually reject unless temp is high.
            exp_val = delta / (temp * 0.1) # Scaling temp

                prob = 0.0
            else:
                prob = math.exp(exp_val)

            current_sum_radii = new_sum_radii

                best_centers = current_centers.copy()
                best_radii = current_radii.copy()
                best_sum_radii = new_sum_radii
        
        # Cooling
        temp *= cooling_rate

                best_radii[i] = r # Revert

    # If any overlap, shrink the offending circles slightly

        overlap_found = False
        for i in range(N):
            for j in range(i + 1, N):
                dx = best_centers[i, 0] - best_centers[j, 0]
                dy = best_centers[i, 1] - best_centers[j, 1]
                dist = math.sqrt(dx*dx + dy*dy)
                r_sum = best_radii[i] + best_radii[j]
                if dist < r_sum:
                    # Shrink both equally
                    shrink = (r_sum - dist) / 2 + 1e-4
                    best_radii[i] -= shrink / 2
                    best_radii[j] -= shrink / 2
                    overlap_found = True
            if overlap_found: break
        if not overlap_found:
            # Check boundaries
            boundary_fix = False
            for i in range(N):
                r = best_radii[i]
                x, y = best_centers[i]
                # Project to valid region
                x = max(r, min(1-r, x))
                y = max(r, min(1-r, y))
                if x != best_centers[i, 0] or y != best_centers[i, 1]:
                     best_centers[i] = [x, y]
                     boundary_fix = True
            if not boundary_fix:
                break

    sum_radii = np.sum(best_radii)
    
    return best_centers, best_radii, sum_radii
