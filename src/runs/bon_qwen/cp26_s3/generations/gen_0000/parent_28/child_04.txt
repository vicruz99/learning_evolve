2.  **Penalty Function**: Define a loss function to minimize. The objective is to maximize $\sum r_i$, which is equivalent to minimizing $-\sum r_i$. We add penalty terms for:

    *   Circles crossing the boundaries of the unit square.
    The penalty function uses squared violations to ensure differentiability (mostly) and smooth gradients.
3.  **Gradient Computation**: Implement the analytical gradient of the loss function with respect to all variables ($x, y, r$ for each circle). This speeds up convergence compared to finite differences.
4.  **Continuation Method**: Run the optimization iteratively with an increasing penalty weight ($\lambda$). This allows the solver to first find a feasible configuration and then tighten the constraints while maximizing radii.
5.  **Refinement**: After optimization, perform a final check and adjustment to ensure all constraints are strictly satisfied within the tolerance.
6.  **Random Restarts**: To avoid local minima, the solver will attempt multiple random initializations (perturbing the grid) and select the best result.

import numpy as np
from scipy.optimize import minimize
import time

def get_loss_and_grad(params, n, lambda_penalty):
    """
    Computes the loss and gradient for the circle packing problem.
    
    Loss = - sum(radii) + lambda * (penalty for overlaps + penalty for boundaries)
    
    Args:
        params: numpy array of shape (3*n,)
                Structure: [x1, y1, r1, x2, y2, r2, ...]
        n: number of circles
        lambda_penalty: weight for the penalty terms
    
    Returns:
        loss: scalar
        grad: numpy array of shape (3*n,)
    """
    # Reshape params
    # x, y, r vectors
    # We can access them by slicing or reshaping. 
    # Let's keep it as a flat array but extract columns.
    
    x = params[0::3]
    y = params[1::3]
    r = params[2::3]
    
    # Initialize gradient array
    grad = np.zeros_like(params)
    
    # Objective: maximize sum(r), so minimize -sum(r)
    # Gradient contribution: -1 for each r
    grad[2::3] = -1.0
    
    current_loss = -np.sum(r)
    
    # 1. Boundary Penalties
    # Constraints: r <= x, r <= y, x + r <= 1, y + r <= 1
    # Violations:
    # v1 = r - x  (should be <= 0)
    # v2 = r - y
    # v3 = x + r - 1
    # v4 = y + r - 1
    
    # We use squared penalty for positive violations: max(0, v)^2
    # To make it differentiable at 0, we can rely on the optimizer or use smooth approx, 
    # but exact max(0,v)^2 has 0 derivative at 0 from both sides, so it is C1.
    
    # Vectorized computation
    # We need to accumulate gradients to the correct indices
    
    # Helper to apply penalty and gradient for a violation vector v
    # If v > 0, penalty += v^2, grad += 2v * dv/dparam
    
    # v1 = r - x
    v1 = r - x
    mask1 = v1 > 0
    if np.any(mask1):
        current_loss += lambda_penalty * np.sum(v1[mask1]**2)
        # d(v1)/dr = 1, d(v1)/dx = -1
        # grad contribution: 2 * v1 * dv/dparam
        # For r (index 2::3): +2*v1
        # For x (index 0::3): -2*v1
        grad_r_indices = 2 + 3*np.where(mask1)[0]
        grad_x_indices = 0 + 3*np.where(mask1)[0]
        grad[grad_r_indices] += 2 * v1[mask1]
        grad[grad_x_indices] -= 2 * v1[mask1]

    # v2 = r - y
    v2 = r - y
    mask2 = v2 > 0
    if np.any(mask2):
        current_loss += lambda_penalty * np.sum(v2[mask2]**2)
        # d(v2)/dr = 1, d(v2)/dy = -1
        grad_r_indices = 2 + 3*np.where(mask2)[0]
        grad_y_indices = 1 + 3*np.where(mask2)[0]
        grad[grad_r_indices] += 2 * v2[mask2]
        grad[grad_y_indices] -= 2 * v2[mask2]

    # v3 = x + r - 1
    v3 = x + r - 1.0
    mask3 = v3 > 0
    if np.any(mask3):
        current_loss += lambda_penalty * np.sum(v3[mask3]**2)
        # d(v3)/dx = 1, d(v3)/dr = 1
        grad_x_indices = 0 + 3*np.where(mask3)[0]
        grad_r_indices = 2 + 3*np.where(mask3)[0]
        grad[grad_x_indices] += 2 * v3[mask3]
        grad[grad_r_indices] += 2 * v3[mask3]

    # v4 = y + r - 1
    v4 = y + r - 1.0
    mask4 = v4 > 0
    if np.any(mask4):
        current_loss += lambda_penalty * np.sum(v4[mask4]**2)
        # d(v4)/dy = 1, d(v4)/dr = 1
        grad_y_indices = 1 + 3*np.where(mask4)[0]
        grad_r_indices = 2 + 3*np.where(mask4)[0]
        grad[grad_y_indices] += 2 * v4[mask4]
        grad[grad_r_indices] += 2 * v4[mask4]

    # 2. Overlap Penalties
    # Constraint: (ri + rj)^2 <= (xi - xj)^2 + (yi - yj)^2
    # Violation u = (ri + rj)^2 - dist_sq
    # Penalty: max(0, u)^2
    
    # Compute pairwise squared distances and radius sums
    # n is small (26), so O(n^2) is fine.
    
    # Using broadcasting
    # x[:, None] - x[None, :] gives difference matrix
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist_sq = dx**2 + dy**2
    
    r_sum = r[:, None] + r[None, :]
    r_sum_sq = r_sum**2
    
    violation = r_sum_sq - dist_sq
    
    # We only care about upper triangle i < j
    # Create a mask for i < j
    mask_ij = np.triu(np.ones((n, n), dtype=bool), k=1)
    
    active_violations = violation * mask_ij # 0 where i>=j or violation<=0 (handled by max)
    
    # Find indices where violation > 0
    # We need to iterate or vectorize the gradient update.
    # Vectorizing over pairs is tricky for gradient accumulation into flat array.
    # But n=26 is small enough to loop over active pairs or use sparse accumulation.
    # Let's find indices of active violations.
    
    row_idx, col_idx = np.where((violation > 0) & mask_ij)
    
    if len(row_idx) > 0:
        v_vals = violation[row_idx, col_idx]
        # Penalty addition
        current_loss += lambda_penalty * np.sum(v_vals**2)
        
        # Gradient update
        # d(u)/dr_i = 2(r_i + r_j)
        # d(u)/dx_i = -2(x_i - x_j)
        # d(u)/dy_i = -2(y_i - y_j)
        # Contribution to Loss gradient (for minimizing):
        # d(L)/dr_i += lambda * 2 * u * du/dr_i = 4 * lambda * u * (r_i + r_j)
        # d(L)/dx_i += lambda * 2 * u * du/dx_i = -4 * lambda * u * (x_i - x_j)
        
        # Calculate terms
        # r_i + r_j
        r_i_j = r[row_idx] + r[col_idx]
        # x_i - x_j
        x_i_j = dx[row_idx, col_idx]
        # y_i - y_j
        y_i_j = dy[row_idx, col_idx]
        
        # Common factor
        factor = 4 * lambda_penalty * v_vals 
        
        # Update r gradients
        # Indices in flat array: 2 + 3*i
        r_indices_i = 2 + 3 * row_idx
        r_indices_j = 2 + 3 * col_idx
        grad[r_indices_i] += factor * r_i_j
        grad[r_indices_j] += factor * r_i_j # Symmetric for r_j
        
        # Update x gradients
        # d(L)/dx_i proportional to -(x_i - x_j) = (x_j - x_i)
        # Note: factor is positive (since v > 0).
        # x_i component: -4 * lambda * u * (x_i - x_j)
        x_indices_i = 0 + 3 * row_idx
        x_indices_j = 0 + 3 * col_idx
        grad[x_indices_i] += factor * (-x_i_j)
        grad[x_indices_j] += factor * (x_i_j) # For j, dx/dx_j = 2(x_j - x_i) = -2(x_i - x_j)
        
        # Update y gradients
        y_indices_i = 1 + 3 * row_idx
        y_indices_j = 1 + 3 * col_idx
        grad[y_indices_i] += factor * (-y_i_j)
        grad[y_indices_j] += factor * (y_i_j)

    return current_loss, grad

def generate_initial_grid(n):
    """
    Generates an initial configuration of n circles using a hexagonal grid pattern.
    Returns a flattened array of params.
    """
    # We want to fit n circles.
    # Try to arrange in rows.
    # Approximation: area per circle in hex packing is 2*sqrt(3)*r^2.
    # 1.0 / (26 * 2 * sqrt(3) * r^2) = 1 => r approx 0.1.
    # Let's place centers on a grid.
    
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.05) # Small initial radius
    
    idx = 0
    # Heuristic grid generation
    # 5 rows
    row_counts = [6, 5, 6, 5, 4] # Sum = 26
    # y spacing
    y_step = 0.25
    x_spacing = 0.2
    
    current_y = 0.25
    for i, count in enumerate(row_counts):
        # Center the row
        start_x = (1.0 - (count - 1) * x_spacing) / 2.0
        # Shift odd rows
        if i % 2 == 1:
            start_x += x_spacing / 2.0
            
        for j in range(count):
            if idx < n:
                centers[idx, 0] = start_x + j * x_spacing
                centers[idx, 1] = current_y
                idx += 1
        current_y += y_step
    
    # Flatten to params
    params = np.zeros(3 * n)
    params[0::3] = centers[:, 0]
    params[1::3] = centers[:, 1]
    params[2::3] = radii
    
    return params

def run_packing():
    n = 26
    best_sum_radii = 0.0
    best_params = None
    
    # We will try a few runs with different initializations
    # But to save time, we rely on the grid init and maybe a random perturbation
    
    lambdas = [100, 1000, 10000, 50000]
    
    # Generate base initialization
    params = generate_initial_grid(n)
    
    # Add small random noise to break symmetry
    np.random.seed(42)
    params += np.random.normal(0, 0.01, size=params.shape)
    
    # Bounds
    bounds = []
    for i in range(n):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 0.5)) # r
    
    current_params = params
    
    for lam in lambdas:
        # Optimization
        # Use L-BFGS-B
        res = minimize(
            fun=lambda p: get_loss_and_grad(p, n, lam)[0],
            x0=current_params,
            jac=lambda p: get_loss_and_grad(p, n, lam)[1],
            method='L-BFGS-B',
            bounds=bounds,
            options={'ftol': 1e-9, 'gtol': 1e-6, 'maxiter': 200}
        )
        current_params = res.x
        
        # Check loss to see if we are stuck or improving
        # Not strictly necessary but good for debugging
        # print(f"Lambda {lam}, Loss: {res.fun}")

    # Final extraction
    x_final = current_params[0::3]
    y_final = current_params[1::3]
    r_final = current_params[2::3]
    
    # Construct outputs
    centers = np.column_stack((x_final, y_final))
    radii = r_final
    
    # Validate and repair if necessary (projection)
    # Although the optimizer should have done it, let's ensure strict validity.
    # Clamp centers and radii.
    
    # 1. Ensure inside square
    for i in range(n):
        r = radii[i]
        x, y = centers[i]
        
        # Clip center to be at least r distance from walls
        # But reducing r is better than moving center?
        # Actually, if x < r, we must increase x or decrease r.
        # Decreasing r reduces objective. Moving x is free (mostly).
        # So we move x to max(r, min(1-r, x))
        
        # However, moving x might cause overlap.
        # But for final output, we just need a valid config.
        # The optimizer should have found a valid one.

        radii[i] = max(0.0, radii[i])
        
        # Ensure bounds
        # If circle is outside, shrink radius?
        # x - r < 0 => r > x. New r = x.

        r = radii[i]
        x = centers[i, 0]
        y = centers[i, 1]
        
        # Left
        if x - r < 0:
            radii[i] = x # Reduce radius to touch wall? Or move center?

            centers[i, 0] = max(x, radii[i]) 
        # Right
        if x + radii[i] > 1:
            centers[i, 0] = min(x, 1.0 - radii[i])
        # Bottom
        if y - radii[i] < 0:
            centers[i, 1] = max(y, radii[i])
        # Top
        if y + radii[i] > 1:
            centers[i, 1] = min(y, 1.0 - radii[i])

    # Re-calculate sum
    sum_radii = np.sum(radii)
    
    # If this is better than previous (initially 0), store it.
    # Since we run sequentially, this is the best we found.
    
    # Double check with a local search for overlaps?
    # If overlaps persist, we might need to reduce radii.
    # But with high lambda, overlaps should be negligible.
    
    # Let's run a quick check and fix overlaps by reducing radii
    # This is a fallback.
    max_iter = 100
    for _ in range(max_iter):
        overlap_found = False
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                r_sum = radii[i] + radii[j]
                if dist < r_sum - 1e-12:
                    # Overlap detected
                    # Reduce radii proportionally to fix overlap
                    # We need dist = r_sum_new

                    reduction = overlap / 2.0
                    radii[i] = max(0, radii[i] - reduction)
                    radii[j] = max(0, radii[j] - reduction)
                    overlap_found = True
        if not overlap_found:
            break
            
    # Final sum
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
