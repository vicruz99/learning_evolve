import numpy as np
import scipy.optimize as opt
import scipy.spatial

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    np.random.seed(42)

    # ---------------------------------------------------------
    # Step 1: Optimize centers for equal radii (Maximize r)
    # ---------------------------------------------------------
    # We want to find centers c_1...c_n and max radius r such that
    # all circles fit. This is equivalent to minimizing overlap for a fixed r,
    # or finding max r.
    
    # Heuristic initialization: Hexagonal grid pattern
    # Approximate spacing for 26 circles. sqrt(26) ~ 5.1
    # Let's try to place them in a grid and perturb.
    
    # Create a 5x5 grid (25 points) and add one in the center or gap
    # But we need 26.
    # Let's just place them randomly in a valid way to start, or grid.
    # A perturbed grid is safer.
    
    # 5x5 grid points
    grid_x = np.linspace(0.1, 0.9, 5)
    grid_y = np.linspace(0.1, 0.9, 5)
    xs, ys = np.meshgrid(grid_x, grid_y)
    centers_init = np.column_stack([xs.ravel(), ys.ravel()]) # 25 points
    
    # Add 26th point. Maybe in the center of a 2x2 block?
    # Center of square is (0.5, 0.5), which is occupied.
    # Let's add a point at (0.5, 0.5) perturbed? 
    # Actually, 26th point can be placed in a gap.
    # Let's just add a point at (0.2, 0.2) offset?
    # Better: Generate a hexagonal lattice.
    
    # Hexagonal lattice generation
    # Spacing d.
    # Rows of 5.
    # y coordinates: 0.1, 0.1 + h, ...
    # Let's just use a random valid initialization to be safe and general.
    
    centers_init = np.random.rand(n, 2) * 0.8 + 0.1 # Random in [0.1, 0.9]
    
    # Function to calculate penalty for a given configuration and radius r
    # Penalty = sum of squared overlaps
    def get_penalty(centers, r):
        # Boundary violations
        # x - r >= 0 => x >= r. Violation: max(0, r - x)^2
        # x + r <= 1 => x <= 1-r. Violation: max(0, x + r - 1)^2
        
        dist_left = centers[:, 0] - r
        dist_right = (1 - centers[:, 0]) - r
        dist_down = centers[:, 1] - r
        dist_up = (1 - centers[:, 1]) - r
        
        b_viol = np.maximum(0, -dist_left)**2 + np.maximum(0, -dist_right)**2 + \
                 np.maximum(0, -dist_down)**2 + np.maximum(0, -dist_up)**2
        
        # Circle-circle violations
        # dist(c_i, c_j) >= 2r. Violation: max(0, 2r - dist)^2
        # Using cdist for efficiency
        dists = scipy.spatial.distance.cdist(centers, centers, 'euclidean')
        np.fill_diagonal(dists, np.inf)
        
        # We only need upper triangle, but squaring handles it symmetrically roughly
        # Actually sum over all pairs
        overlap = 2 * r - dists
        c_viol = np.sum(np.maximum(0, overlap)**2)
        
        return np.sum(b_viol) + c_viol

    # We will use a binary search on r combined with minimization
    # Or just minimize -r with penalty.
    # Let's try to minimize -r + lambda * penalty.
    # But lambda needs to be tuned.
    # Instead, let's fix r and check feasibility.
    
    def is_feasible(r):
        # Minimize penalty for fixed r
        # Variables: 2*n centers
        x0 = centers_init.ravel()
        
        # Bounds for centers [0, 1]
        bounds = [(0, 1)] * (2 * n)
        
        # Objective: penalty
        def objective(x):
            c = x.reshape(n, 2)
            return get_penalty(c, r)
        
        # Use L-BFGS-B for speed
        res = opt.minimize(objective, x0, method='L-BFGS-B', bounds=bounds, 
                           options={'maxiter': 500, 'ftol': 1e-12})
        
        if res.fun < 1e-8: # Tolerance for validity
            return True, res.x
        return False, res.x

    # Binary search for max equal radius
    r_low = 0.05
    r_high = 0.12
    best_r = r_low
    best_centers = np.zeros((n, 2))
    
    # 20 iterations of binary search
    for _ in range(25):
        r_mid = (r_low + r_high) / 2
        feasible, x_opt = is_feasible(r_mid)
        if feasible:
            best_r = r_mid
            best_centers = x_opt.reshape(n, 2)
            r_low = r_mid
        else:
            r_high = r_mid
            # Use the best centers found from the failed attempt to guide next search?
            # The optimization might have moved centers to reduce penalty even if not 0.
            # But for binary search, we just need a flag.
            # However, keeping track of best valid centers is good.
            # If not feasible, the centers from 'is_feasible' are the best attempt for that r.
            # But they are invalid. We stick to best_centers from valid r.
            pass

    print(f"Max equal radius found: {best_r:.6f}, Sum: {26*best_r:.6f}")

    # ---------------------------------------------------------
    # Step 2: Optimize radii using Linear Programming (LP)
    # ---------------------------------------------------------
    # Given best_centers, maximize sum(r_i)
    # Constraints:
    # r_i + r_j <= dist(c_i, c_j)
    # r_i <= dist(c_i, wall)
    # r_i >= 0
    
    # This is an LP.
    # Variables: r_1, ..., r_26
    # Maximize sum(r_i) -> Minimize -sum(r_i)
    
    # Objective coefficients
    c_obj = -np.ones(n)
    
    # Constraints matrix A_ub @ x <= b_ub
    # r_i + r_j <= d_ij
    # Wall constraints: r_i <= w_i
    
    A_ub = []
    b_ub = []
    
    # Distance to walls for each circle
    walls = np.minimum(np.minimum(best_centers[:, 0], 1 - best_centers[:, 0]),
                       np.minimum(best_centers[:, 1], 1 - best_centers[:, 1]))
    
    for i in range(n):
        # Wall constraint: 1 * r_i <= walls[i]
        row = np.zeros(n)
        row[i] = 1.0
        A_ub.append(row)
        b_ub.append(walls[i])
        
    # Pairwise constraints
    # To save memory, we can construct sparse matrix, but 26x26 is small.
    # A_ub is a list of lists or 2D array.
    
    # Calculate pairwise distances
    dists = scipy.spatial.distance.pdist(best_centers, 'euclidean')
    # pdist returns condensed distance matrix.
    # Convert to indices
    indices = scipy.spatial.distance.squareform(np.arange(n*(n-1)//2), checks=False)
    # indices is a set of (i, j) pairs
    
    for k in range(len(dists)):
        i, j = indices[k, 0], indices[k, 1]
        row = np.zeros(n)
        row[i] = 1.0
        row[j] = 1.0
        A_ub.append(row)
        b_ub.append(dists[k])
        
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    # Bounds for r_i >= 0
    bounds_r = [(0, None)] * n
    
    # Solve LP
    try:
        res_lp = opt.linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method='highs')
        if res_lp.success:
            radii = res_lp.x
            centers = best_centers
            sum_radii = np.sum(radii)
            print(f"LP Success. Sum of radii: {sum_radii:.6f}")
        else:
            # Fallback to equal radii if LP fails
            radii = np.full(n, best_r)
            centers = best_centers
            sum_radii = np.sum(radii)
            print(f"LP Failed. Using equal radii. Sum: {sum_radii:.6f}")
            
    except Exception as e:
        print(f"LP Error: {e}")
        radii = np.full(n, best_r)
        centers = best_centers
        sum_radii = np.sum(radii)

    # ---------------------------------------------------------
    # Step 3: Local refinement of centers and radii
    # ---------------------------------------------------------
    # The LP gives optimal radii for FIXED centers.
    # We might improve by moving centers to allow larger radii.
    # However, the objective is non-smooth.
    # Let's try a simple gradient-based or random perturbation local search.
    
    # We can define a combined objective: Maximize sum(r)
    # But r depends on centers.
    # Let's just perturb centers slightly and re-solve LP?
    # That might be too slow (LP is fast, but 100 LPs might take time).
    # Let's try 10 iterations of perturbation.
    
    current_centers = centers.copy()
    current_radii = radii.copy()
    current_sum = sum_radii
    
    # Try random perturbations
    for _ in range(50):
        # Perturb centers
        perturbation = np.random.normal(0, 0.005, size=current_centers.shape)
        new_centers = current_centers + perturbation
        
        # Clip to ensure somewhat inside (though LP handles bounds)
        # Actually LP handles wall constraints, so centers can be anywhere,
        # but walls constraint r_i <= dist(c, wall) will force r_i small if c is near wall.
        # But we want c inside [0,1].
        # The LP constraint r_i <= dist(c, wall) implicitly assumes c is valid?
        # If c is outside, dist is negative?
        # dist to wall definition: min(x, 1-x, y, 1-y). If x < 0, dist < 0.
        # Then r_i <= negative => r_i must be negative (impossible) or 0.
        # So LP will force r_i=0 if center is outside.
        # So it's safe. But we prefer centers inside.
        
        # Re-solve LP for new centers
        # Update walls and distances
        walls_new = np.minimum(np.minimum(new_centers[:, 0], 1 - new_centers[:, 0]),
                               np.minimum(new_centers[:, 1], 1 - new_centers[:, 1]))
        
        # Update A_ub and b_ub
        # Walls
        for i in range(n):
            A_ub[i, i] = 1.0 # Should be 1.0
            b_ub[i] = walls_new[i]
            
        # Pairs
        dists_new = scipy.spatial.distance.pdist(new_centers, 'euclidean')
        for k in range(len(dists_new)):
            i, j = indices[k, 0], indices[k, 1]
            row_idx = n + k
            A_ub[row_idx, i] = 1.0
            A_ub[row_idx, j] = 1.0
            b_ub[row_idx] = dists_new[k]
            
        try:
            res_lp_new = opt.linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method='highs')
            if res_lp_new.success:
                new_sum = -res_lp_new.fun
                if new_sum > current_sum:
                    current_sum = new_sum
                    current_centers = new_centers
                    current_radii = res_lp_new.x
        except:
            pass

    # Validate the final packing
    # We should ensure the returned packing is valid
    # The LP guarantees r_i + r_j <= dist, and r_i <= dist_wall
    # However, numerical errors might occur.
    # Let's do a quick validation and shrink if necessary.
    
    # Check for overlaps
    dists_final = scipy.spatial.distance.cdist(current_centers, current_centers, 'euclidean')
    np.fill_diagonal(dists_final, np.inf)
    
    # Check circle-circle
    min_gap = np.min(dists_final - current_radii[:, None] - current_radii[None, :])
    
    # Check circle-wall
    min_wall_gap = np.min(np.minimum(current_centers, 1 - current_centers)) - current_radii
    
    # If invalid, shrink radii slightly
    if min_gap < -1e-9 or min_wall_gap < -1e-9:
        # Scale down radii to fix overlaps
        # Calculate required scale
        scale = 1.0
        if min_gap < 0:
            # dist < r_i + r_j => need to reduce sum of radii
            # This is complex to fix by scaling all.
            # Just reduce all radii by a small factor
            scale = 0.99
        if min_wall_gap < 0:
            scale = min(scale, 0.99)
        
        current_radii *= scale
        # Recalculate sum
        current_sum = np.sum(current_radii)
        
    return current_centers, current_radii, current_sum
