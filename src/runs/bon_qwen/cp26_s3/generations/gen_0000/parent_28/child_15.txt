import numpy as np
from scipy.optimize import minimize

def compute_energy(centers, r):
    """
    Computes the penalty energy for a configuration of circles with radius r.
    Minimizing this energy drives circles to non-overlapping positions inside the square.
    """
    N = centers.shape[0]
    # Vectorized pairwise distance calculation
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dist, np.inf)
    
    # Overlap penalty: penalize when distance < 2r
    ov = np.maximum(0, 2 * r - dist)
    e_pair = np.sum(ov**2) / 2.0
    
    # Boundary penalty: penalize when center is too close to walls
    low_pen = np.sum(np.maximum(0, r - centers)**2)
    high_pen = np.sum(np.maximum(0, centers - (1 - r))**2)
    
    return e_pair + low_pen + high_pen
