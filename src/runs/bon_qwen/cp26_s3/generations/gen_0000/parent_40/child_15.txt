import numpy as np

def compute_radii(centers):
    """Compute maximum possible radii for given centers constrained by boundaries and each other."""
    b_dist = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]), 
                        np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.linalg.norm(diff, axis=2)
    np.fill_diagonal(dists, np.inf)
    neighbor_dist = np.min(dists, axis=1)
    return np.minimum(b_dist, neighbor_dist / 2.0)

    pts = []

        for x in xs:
            pts.append([x, y])
        xs = xs + 0.04  # Shift for hexagonal staggering
    inits.append(np.array(pts[:n]))

    inits.append(np.random.uniform(0.1, 0.9, (n, 2)))
    
    # 3. Structured grid with extra point
    grid = np.array([(i * 0.15 + 0.1, j * 0.15 + 0.1) for j in range(5) for i in range(5)])
    grid = np.vstack([grid, [0.5, 0.5]])
    inits.append(grid)

    for init in inits:
        centers = init.copy()
        lr = 0.005
        
        # Iterative repulsion-based optimization
        for step in range(4000):
            radii = compute_radii(centers)
            
            forces = np.zeros_like(centers)
            
            # Boundary repulsion forces
            forces[:, 0] += np.where(centers[:, 0] < radii, (radii - centers[:, 0]), 0.0) * 10.0
            forces[:, 0] -= np.where(centers[:, 0] > 1.0 - radii, (centers[:, 0] - (1.0 - radii)), 0.0) * 10.0
            forces[:, 1] += np.where(centers[:, 1] < radii, (radii - centers[:, 1]), 0.0) * 10.0
            forces[:, 1] -= np.where(centers[:, 1] > 1.0 - radii, (centers[:, 1] - (1.0 - radii)), 0.0) * 10.0
            
            # Pairwise repulsion forces
            diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
            dists = np.linalg.norm(diff, axis=2)
            
            r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
            mask = dists < r_sum + 1e-6
            np.fill_diagonal(mask, False)
            
            safe_dists = np.where(mask, dists, 1.0) + 1e-9
            dirs = diff / safe_dists[:, :, np.newaxis]
            overlap = np.where(mask, r_sum - dists, 0.0)
            mag = overlap * 5.0
            
            forces += np.sum(dirs * mag[:, :, np.newaxis], axis=1)

            centers += forces * lr

        # Final exact radii computation
        radii = compute_radii(centers)
        s = np.sum(radii)
        if s > best_sum:
            best_sum = s
            best_centers = centers.copy()
            best_radii = radii.copy()
            
    return best_centers, best_radii, best_sum
