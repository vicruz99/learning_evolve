import numpy as np
from scipy.optimize import minimize

def objective(params):
    # params contains [x1, y1, x2, y2, ..., x26, y26, r1, r2, ..., r26]
    # We want to maximize sum(r), so we minimize -sum(r)
    return -np.sum(params[52:])

def constraint_func(params):
    n = 26
    # 4 boundary constraints per circle + pairwise overlap constraints
    n_constr = 4 * n + n * (n - 1) // 2
    vals = np.zeros(n_constr)
    idx = 0
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    for i in range(n):
        ix, iy, ir = 2 * i, 2 * i + 1, 52 + i
        vals[idx] = params[ix] - params[ir]       # x >= r
        idx += 1
        vals[idx] = 1.0 - params[ix] - params[ir] # 1-x >= r
        idx += 1
        vals[idx] = params[iy] - params[ir]       # y >= r
        idx += 1
        vals[idx] = 1.0 - params[iy] - params[ir] # 1-y >= r
        idx += 1

    # Overlap constraints: dist_ij >= ri + rj
    for i in range(n):
        for j in range(i + 1, n):
            ix_i, iy_i, ir_i = 2 * i, 2 * i + 1, 52 + i
            ix_j, iy_j, ir_j = 2 * j, 2 * j + 1, 52 + j
            
            dx = params[ix_i] - params[ix_j]
            dy = params[iy_i] - params[iy_j]
            dist = np.sqrt(dx * dx + dy * dy)
            vals[idx] = dist - params[ir_i] - params[ir_j]
            idx += 1
            
    return vals

def run_packing():
    n = 26
    best_sum = -1.0
    best_centers = None
    best_radii = None

    # Run optimization from multiple seeded initializations
    for seed in range(5):
        rng = np.random.RandomState(seed + 7)
        cols = int(np.ceil(np.sqrt(n * 1.2)))
        rows = int(np.ceil(n / cols))
        
        centers = np.zeros((n, 2))
        for i in range(n):
            col = i % cols
            row = i // cols
            # Hexagonal grid coordinates
            x = (col + 0.5 + (row % 2) * 0.5) / cols
            y = (row + 0.5) / rows
            # Add controlled random perturbation
            px = x + rng.uniform(-0.02, 0.02)
            py = y + rng.uniform(-0.02, 0.02)
            # Keep strictly inside initially to help solver start feasible
            centers[i] = np.clip([px, py], 0.15, 0.85)

        radii = np.full(n, 0.04)
        params0 = np.concatenate([centers.flatten(), radii])

        # Bounds: x,y in [0,1], r in [1e-5, 0.5]
        bounds = [(0.0, 1.0)] * 52 + [(1e-5, 0.5)] * 26
        cons = {'type': 'ineq', 'fun': constraint_func}

        res = minimize(objective, params0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': 5000, 'ftol': 1e-12, 'disp': False})

        # Track best solution
        if not np.isnan(res.fun):
            cur_sum = -res.fun
            if cur_sum > best_sum:
                best_sum = cur_sum
                best_centers = res.x[:52].reshape((n, 2))
                best_radii = res.x[52:]

    # Fallback if optimizer failed entirely
    if best_centers is None:
        best_centers = np.zeros((n, 2))
        best_radii = np.zeros(n)
        best_sum = 0.0

    # Ensure radii are non-negative
    best_radii = np.maximum(best_radii, 0.0)
    
    return best_centers, best_radii, float(best_sum)
