**Strategy:**
1. **Penalty Method**: We formulate the problem as minimizing a penalty function. The objective is $-\sum r_i$ (since we want to maximize radii). We add quadratic penalty terms for:
   - Overlaps: $(\max(0, r_i + r_j - d_{ij}))^2$
   - Boundary violations: $(\max(0, r_i - x_i))^2$, etc.
   A large penalty weight ensures constraints are satisfied while the optimizer pushes radii to grow.
2. **Initialization**: We use multiple random initial configurations within the square. Starting with small radii helps the optimizer navigate the constraint landscape before expanding circles.
3. **Optimizer**: `scipy.optimize.minimize` with the `L-BFGS-B` method is chosen for its efficiency with bound constraints and medium-dimensional problems (78 variables here). We run ~80 restarts to escape poor local minima.
4. **Post-processing**: We ensure radii are non-negative and format the output correctly. The penalty formulation naturally converges to configurations where circles touch each other or boundaries, which is characteristic of optimal packings.

import numpy as np
from scipy.optimize import minimize

def compute_objective(vars, N):
    """
    Computes the objective function: negative sum of radii + constraint penalties.
    vars layout: [x1, x2, ..., xN, y1, y2, ..., yN, r1, r2, ..., rN]
    """
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]

    penalty_weight = 2000.0
    total_penalty = 0.0

    # Overlap penalties between all pairs
    for i in range(N):
        for j in range(i + 1, N):
            dist = np.hypot(x[i] - x[j], y[i] - y[j])
            overlap = r[i] + r[j] - dist
            if overlap > 0.0:
                total_penalty += overlap ** 2

    # Boundary penalties for each circle
    for i in range(N):
        total_penalty += max(0.0, r[i] - x[i]) ** 2
        total_penalty += max(0.0, r[i] - (1.0 - x[i])) ** 2
        total_penalty += max(0.0, r[i] - y[i]) ** 2
        total_penalty += max(0.0, r[i] - (1.0 - y[i])) ** 2

    return -np.sum(r) + penalty_weight * total_penalty

def run_packing():
    """
    Optimizes packing of 26 circles in a unit square to maximize sum of radii.
    Returns (centers, radii, sum_radii)
    """
    N = 26
    best_vars = None
    best_score = np.inf
    best_sum_r = -1.0

    # Deterministic seed for reproducibility, but multiple starts provide exploration
    rng = np.random.default_rng(2023)

    # Multiple restarts to find a good local/global optimum
    num_restarts = 100
    for _ in range(num_restarts):
        # Initialize centers randomly in the interior to avoid immediate boundary penalties
        x0 = rng.uniform(0.2, 0.8, N)
        y0 = rng.uniform(0.2, 0.8, N)
        # Start with small radii so the optimizer can place them first, then grow them
        r0 = np.full(N, 0.04)
        vars0 = np.concatenate([x0, y0, r0])

        # Bounds: centers in [0, 1], radii in [0, 0.5]
        bounds = [(0.0, 1.0)] * N + [(0.0, 1.0)] * N + [(0.0, 0.5)] * N

        try:
            res = minimize(
                compute_objective,
                vars0,
                args=(N,),
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 3000, 'ftol': 1e-12, 'gtol': 1e-9}
            )
            
            # Check if this run improved upon the best found so far
            if res.fun < best_score:
                best_score = res.fun
                best_vars = res.x.copy()
                cur_r = best_vars[2::3]
                best_sum_r = np.sum(cur_r)
        except Exception:
            continue

    if best_vars is not None:
        centers = np.column_stack((best_vars[0::3], best_vars[1::3]))
        radii = np.maximum(best_vars[2::3], 0.0) # Ensure non-negative
        return centers, radii, np.sum(radii)

    # Fallback (should not be reached with reasonable runtime)
    centers = np.tile([0.5, 0.5], (N, 1))
    radii = np.zeros(N)
    return centers, radii, 0.0
