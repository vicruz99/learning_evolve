from scipy.optimize import minimize

    Runs a packing optimization for 26 circles in a unit square.
    Returns centers, radii, and sum of radii.

    best_centers = None
    best_radii = None

        r = x[2 * n:]
        return -np.sum(r)

    def boundary_constraints(x):
        centers = x[:2 * n].reshape(n, 2)
        r = x[2 * n:]

        # x_i - r_i >= 0
        # 1 - x_i - r_i >= 0
        # y_i - r_i >= 0
        # 1 - y_i - r_i >= 0
        cons = []
        for i in range(n):
            cons.append(centers[i, 0] - r[i])
            cons.append(1.0 - centers[i, 0] - r[i])
            cons.append(centers[i, 1] - r[i])
            cons.append(1.0 - centers[i, 1] - r[i])
        return cons

        centers = x[:2 * n].reshape(n, 2)
