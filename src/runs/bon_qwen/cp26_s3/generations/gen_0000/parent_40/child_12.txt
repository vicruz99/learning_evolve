import numpy as np

def get_lp_radii(centers):
    """
    Given fixed centers, solve LP to find max radii.
    Maximize sum(r_i)
    Subject to:
    r_i + r_j <= dist(i, j) for all i < j
    r_i <= dist(i, boundary) for all i
    r_i >= 0
    """
    n = centers.shape[0]
    
    # Objective: maximize sum(r_i) -> minimize -sum(r_i)
    c_obj = np.zeros(n)
    c_obj[:] = -1.0

    # Inequality constraints A_ub * r <= b_ub
    # We need to construct these constraints.
    # Constraints are of the form r_i + r_j <= d_ij
    # and r_i <= b_i
    
    # Number of constraints
    # Pairwise: n*(n-1)/2
    # Boundary: 4*n
    
    n_constraints = n * (n - 1) // 2 + 4 * n
    A_ub = np.zeros((n_constraints, n))
    b_ub = np.zeros(n_constraints)
    
    idx = 0
    
    # Pairwise distance constraints
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            # r_i + r_j <= dist
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dist
            idx += 1
            
    # Boundary constraints
    # x - r >= 0 => r <= x
    # 1 - x - r >= 0 => r <= 1 - x
    # y - r >= 0 => r <= y
    # 1 - y - r >= 0 => r <= 1 - y
    
    for i in range(n):
        x, y = centers[i]
        
        # r_i <= x
        A_ub[idx, i] = 1.0
        b_ub[idx] = x
        idx += 1
        
        # r_i <= 1 - x
        A_ub[idx, i] = 1.0
        b_ub[idx] = 1.0 - x
        idx += 1
        
        # r_i <= y
        A_ub[idx, i] = 1.0
        b_ub[idx] = y
        idx += 1
        
        # r_i <= 1 - y
        A_ub[idx, i] = 1.0
        b_ub[idx] = 1.0 - y
        idx += 1

    # Bounds for r_i: r_i >= 0
    bounds = [(0, None) for _ in range(n)]

    # Solve LP
    # Using 'highs' method if available, else 'simplex' or 'interior-point'
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        radii = res.x
        # Clean up small negative values due to numerical error
        radii = np.maximum(radii, 0.0)
        return radii, -res.fun
    else:
        # Fallback if LP fails (should not happen for valid centers)
        # Return small radii
        return np.full(n, 1e-9), 0.0
