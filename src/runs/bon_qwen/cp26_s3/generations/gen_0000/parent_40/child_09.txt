import numpy as np
from scipy.optimize import minimize

def run_packing():
    """
    Pack 26 circles in a unit square to maximize the sum of radii.
    
    Returns:
        tuple: (centers, radii, sum_radii)
    """
    N = 26
    best_sum = -1.0
    best_centers = None
    best_radii = None

    # Function to perform optimization from a specific seed
    def optimize_seed(seed):
        np.random.seed(seed)
        
        # 1. Initialization
        # We initialize with a grid-like distribution to ensure good coverage
        # and then perturb slightly to break symmetry and avoid grid artifacts.
        # We use a 6x5 grid (30 points) and select 26, or just random with separation.
        # Here we use a random initialization with a repulsive force pre-processing step
        # to ensure the starting point is valid (non-overlapping).
        
        centers = np.random.rand(N, 2)
        radii = np.full(N, 0.04) # Start with small radius
        
        # Relaxation step to separate overlapping circles if any
        # This ensures we start from a feasible point for SLSQP
        for _ in range(200):
            overlap_found = False
            for i in range(N):
                for j in range(i + 1, N):
                    dx = centers[i, 0] - centers[j, 0]
                    dy = centers[i, 1] - centers[j, 1]
                    d = np.hypot(dx, dy)
                    
                    min_dist = radii[i] + radii[j]
                    
                    if d < min_dist and d > 1e-6:
                        # Push apart
                        # Normalize vector
                        nx = dx / d
                        ny = dy / d
                        
                        # Shift amount: half the overlap plus a tiny buffer
                        shift = (min_dist - d + 1e-5) / 2
                        
                        centers[i, 0] += nx * shift
                        centers[i, 1] += ny * shift
                        centers[j, 0] -= nx * shift
                        centers[j, 1] -= ny * shift
                        overlap_found = True
            
            # Check and fix boundary violations
            for i in range(N):
                x, y = centers[i]
                r = radii[i]
                changed = False
                
                if x - r < 0:
                    x = r
                    changed = True
                elif x + r > 1:
                    x = 1 - r
                    changed = True
                    
                if y - r < 0:
                    y = r
                    changed = True
                elif y + r > 1:
                    y = 1 - r
                    changed = True
                
                if changed:
                    centers[i] = [x, y]
                    overlap_found = True # Trigger another pass if moved

            if not overlap_found:
                break

        # 2. Define Optimization Problem
        # Variables: [x1, y1, r1, x2, y2, r2, ...]
        x0 = np.hstack([centers.ravel(), radii])
        
        # Objective: Maximize sum(r) -> Minimize -sum(r)
        def objective(v):
            return -np.sum(v[2::3])
        
        # Constraints function
        # Returns a vector of constraint values, must be >= 0
        def constraints(v):
            x = v[0::3]
            y = v[1::3]
            r = v[2::3]
            
            cons = []
            
            # Boundary constraints:
            # x - r >= 0
            cons.append(x - r)
            # 1 - x - r >= 0
            cons.append(1.0 - x - r)
            # y - r >= 0
            cons.append(y - r)
            # 1 - y - r >= 0
            cons.append(1.0 - y - r)
            
            # Overlap constraints:
            # dist(i, j) >= r_i + r_j
            # dist - (r_i + r_j) >= 0
            
            # Vectorized distance calculation
            # dx shape (N, N)
            dx = x[:, None] - x[None, :]
            dy = y[:, None] - y[None, :]
            dist = np.sqrt(dx**2 + dy**2)
            
            r_sum = r[:, None] + r[None, :]
            
            # Constraint value: dist - r_sum
            # We only need upper triangle (i < j) to avoid duplicates and diagonal
            mask = np.triu(np.ones((N, N), dtype=bool), k=1)
            overlap_vals = (dist - r_sum)[mask]
            cons.append(overlap_vals)
            
            return np.concatenate(cons)
        
        # Define constraints for SLSQP
        cons_dict = {'type': 'ineq', 'fun': constraints}
        
        # Bounds for variables
        # x, y in [0, 1]
        # r >= 0 (and implicitly <= 0.5 due to bounds)
        bounds = []
        for _ in range(N):
            bounds.append((0.0, 1.0)) # x
            bounds.append((0.0, 1.0)) # y
            bounds.append((0.0, 0.5)) # r
            
        # Run optimization
        # SLSQP is a good choice for this scale
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, 
                           constraints=cons_dict, 
                           options={'maxiter': 2000, 'ftol': 1e-12, 'disp': False})
            
            if res.success or res.fun < -best_sum: # res.fun is negative sum
                curr_sum = -res.fun
                if curr_sum > best_sum:
                    best_sum = curr_sum
                    best_centers = np.column_stack((res.x[0::3], res.x[1::3]))
                    best_radii = res.x[2::3]
        except Exception:
            pass # Ignore failed optimizations

    # 3. Run multiple times with different seeds to escape local minima
    seeds = range(10) # 10 restarts
    
    for s in seeds:
        optimize_seed(s)

    # 4. Final Validation and Formatting
    if best_centers is None:
        # Fallback if everything failed (unlikely)
        best_centers = np.random.rand(N, 2)
        best_radii = np.full(N, 0.01)
        best_sum = np.sum(best_radii)

    # Ensure types
    best_centers = np.array(best_centers)
    best_radii = np.array(best_radii)
    
    # Return
    return best_centers, best_radii, float(np.sum(best_radii))
