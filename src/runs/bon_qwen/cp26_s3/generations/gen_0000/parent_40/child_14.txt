import numpy as np
from scipy.optimize import linprog

def run_packing():
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    """
    n = 26
    
    # 1. Initialization
    # Create a 5x5 grid and add one extra point, then perturb
    centers = np.zeros((n, 2))
    idx = 0
    for i in range(5):
        for j in range(5):
            centers[idx] = [0.1 + 0.2 * i, 0.1 + 0.2 * j]
            idx += 1
    centers[25] = [0.5, 0.2] # Add 26th point in a gap
    
    # Add random jitter to break symmetry
    centers += np.random.uniform(-0.02, 0.02, centers.shape)
    
    # Clamp to valid range
    centers = np.clip(centers, 0.01, 0.99)
    
    # Initial repulsive layout to spread points evenly
    for _ in range(50):
        forces = np.zeros_like(centers)
        for i in range(n):
            for j in range(i + 1, n):
                vec = centers[i] - centers[j]
                dist = np.linalg.norm(vec)
                if dist < 0.1: # Repel if too close
                    force_mag = (0.1 - dist) * 0.5
                    if dist > 0:
                        forces[i] += (vec / dist) * force_mag
                        forces[j] -= (vec / dist) * force_mag
        centers += forces
        centers = np.clip(centers, 0.01, 0.99)

    # 2. Iterative Optimization
    # Alternating LP for radii and Force relaxation for centers
    best_sum = 0
    best_centers = centers.copy()
    best_radii = np.zeros(n)

    # Helper to compute pairwise distances
    # Vectorized distance calculation
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.linalg.norm(diff, axis=2)
    np.fill_diagonal(dists, np.inf) # Distance to self is infinite

    for iteration in range(100):
        # --- LP Step: Maximize sum of radii for fixed centers ---
        c_obj = -np.ones(n) # Maximize sum r_i <=> Minimize -sum r_i
        
        # Constraints: r_i + r_j <= d_ij
        # We only need constraints for pairs that are close or all pairs?
        # All pairs to be safe, but sparse is better. 
        # Given n=26, dense matrix is fine.
        
        n_pairs = n * (n - 1) // 2
        A_ub = np.zeros((n_pairs, n))
        b_ub = np.zeros(n_pairs)
        
        pair_idx = 0
        for i in range(n):
            for j in range(i + 1, n):
                A_ub[pair_idx, i] = 1
                A_ub[pair_idx, j] = 1
                b_ub[pair_idx] = dists[i, j]
                pair_idx += 1

        # Bounds: 0 <= r_i <= min_dist_to_boundary
        # min_dist_to_boundary for center (x,y) is min(x, 1-x, y, 1-y)
        bounds = []
        for i in range(n):
            x, y = centers[i]
            max_r = min(x, 1 - x, y, 1 - y)
            bounds.append((0, max_r))
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
            if res.success:
                radii = res.x
            else:
                # Fallback if LP fails (rare)
                radii = np.zeros(n)
        except Exception:
            radii = np.zeros(n)

        current_sum = np.sum(radii)
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers.copy()
            best_radii = radii.copy()

        # --- Force Step: Move centers to increase potential radii ---
        # We want to separate circles that are touching (tight constraints)
        # Force is repulsive proportional to "tightness" or just constant for touching
        # Tightness = r_i + r_j - d_ij. If positive, overlap (impossible in valid LP sol).
        # If close to 0, it's active. We push apart to increase d_ij.
        
        # Recompute distances from best_centers to be safe, or use current
        diff = best_centers[:, np.newaxis, :] - best_centers[np.newaxis, :, :]
        dists = np.linalg.norm(diff, axis=2)
        np.fill_diagonal(dists, np.inf)
        
        forces = np.zeros_like(best_centers)
        step_size = 0.01
        
        for i in range(n):
            xi, yi = best_centers[i]
            ri = best_radii[i]
            
            # Wall forces
            if xi - ri < 1e-5: forces[i, 0] += 1.0
            if xi + ri > 1 - 1e-5: forces[i, 0] -= 1.0
            if yi - ri < 1e-5: forces[i, 1] += 1.0
            if yi + ri > 1 - 1e-5: forces[i, 1] -= 1.0
            
            # Pairwise forces
            for j in range(i + 1, n):
                d = dists[i, j]
                r_sum = best_radii[i] + best_radii[j]
                
                # If touching or very close, repel
                if r_sum > d - 1e-4:
                    # Repulsion force
                    # Vector from j to i
                    vec = best_centers[i] - best_centers[j]
                    if d > 1e-6:
                        # Force magnitude: proportional to how tight? 
                        # Just a constant push to separate touching circles
                        f_mag = 0.5 
                        forces[i] += (vec / d) * f_mag
                        forces[j] -= (vec / d) * f_mag

        # Update centers
        best_centers += step_size * forces
        
        # Clip and project back to valid range (inside square)
        # If a center is too close to wall for its radius, push it in
        for i in range(n):
            ri = best_radii[i]
            best_centers[i, 0] = np.clip(best_centers[i, 0], ri, 1 - ri)
            best_centers[i, 1] = np.clip(best_centers[i, 1], ri, 1 - ri)

    diff = best_centers[:, np.newaxis, :] - best_centers[np.newaxis, :, :]
    dists = np.linalg.norm(diff, axis=2)
    np.fill_diagonal(dists, np.inf)
    
    A_ub = np.zeros((n * (n - 1) // 2, n))
    b_ub = np.zeros(n * (n - 1) // 2)
    pair_idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[pair_idx, i] = 1
            A_ub[pair_idx, j] = 1
            b_ub[pair_idx] = dists[i, j]
            pair_idx += 1
            
    bounds = []
    for i in range(n):
        x, y = best_centers[i]
        max_r = min(x, 1 - x, y, 1 - y)
        # Ensure non-negative bound
        bounds.append((0, max(0, max_r)))
        
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            final_radii = res.x
        else:
            final_radii = best_radii
    except Exception:
        final_radii = best_radii

        xc, yc = final_centers[i]
        # Ensure inside
        if xc < r: xc = r
        if xc > 1 - r: xc = 1 - r
        if yc < r: yc = r
        if yc > 1 - r: yc = 1 - r
        final_centers[i] = [xc, yc]
        
    # Recalculate radii one last time based on clamped centers to be 100% safe
    # Though LP should have handled it, clamping might slightly alter constraints
    # Actually, if we clamp centers, distances might decrease, violating r_i + r_j <= d.
    # So we must re-solve LP.
    
    diff = final_centers[:, np.newaxis, :] - final_centers[np.newaxis, :, :]
    dists = np.linalg.norm(diff, axis=2)
    np.fill_diagonal(dists, np.inf)
    
    A_ub = np.zeros((n * (n - 1) // 2, n))
    b_ub = np.zeros(n * (n - 1) // 2)
    pair_idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[pair_idx, i] = 1
            A_ub[pair_idx, j] = 1
            b_ub[pair_idx] = dists[i, j]
            pair_idx += 1
            
    bounds = []
    for i in range(n):
        x, y = final_centers[i]
        max_r = min(x, 1 - x, y, 1 - y)
        bounds.append((0, max(0, max_r)))
        
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            final_radii = res.x
    except:
        pass # Keep previous
