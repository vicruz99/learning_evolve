import numpy as np
from scipy.optimize import minimize
import math


def compute_penalty(params, n):
    """Compute constraint violation penalty."""
    c = params[:2 * n].reshape(n, 2)
    r = params[2 * n:]
    
    penalty = 0.0
    
    # Boundary constraints: x - r >= 0, x + r <= 1, y - r >= 0, y + r <= 1
    for i in range(n):
        penalty += max(0.0, r[i] - c[i, 0]) ** 2
        penalty += max(0.0, r[i] - (1.0 - c[i, 0])) ** 2
        penalty += max(0.0, r[i] - c[i, 1]) ** 2
        penalty += max(0.0, r[i] - (1.0 - c[i, 1])) ** 2
    
    # Overlap constraints: distance >= r_i + r_j
    for i in range(n):
        for j in range(i + 1, n):
            dx = c[i, 0] - c[j, 0]
            dy = c[i, 1] - c[j, 1]
            d = math.sqrt(dx * dx + dy * dy)
            min_d = r[i] + r[j]
            if d < min_d:
                penalty += (min_d - d) ** 2
    
    return penalty


def objective(params, n):
    """Objective: maximize sum of radii with penalty for constraint violations."""
    r = params[2 * n:]
    penalty = compute_penalty(params, n)
    # Use large penalty coefficient to enforce constraints
    return -np.sum(r) + 5000.0 * penalty


def make_initial_params(row_counts, seed=0):

def optimize_packing(params_init, n):
    """Run optimization from given initial parameters."""
    bounds = [(0.02, 0.98)] * (2 * n) + [(0.02, 0.3)] * n
    
    result = minimize(
        objective,
        params_init,
        args=(n,),
        bounds=bounds,
        method='L-BFGS-B',
        options={
            'maxiter': 100000,
            'ftol': 1e-18,
            'gtol': 1e-14,
            'disp': False
        }
    )
    return result
