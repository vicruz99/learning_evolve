from scipy.optimize import minimize

def calculate_overlap_penalty(centers, radii, lam):

    penalty = 0.0
    
    # Boundary penalties
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        margin = r - 1e-5 # Small buffer to stay strictly inside

                penalty += lam * (r - val)**2

            min_dist = radii[i] + radii[j]
            if dist_sq < min_dist**2:
                penetration = min_dist - np.sqrt(dist_sq)
                penalty += lam * penetration**2
                
    return penalty
