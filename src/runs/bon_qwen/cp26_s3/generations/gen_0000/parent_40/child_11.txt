import numpy as np
from scipy.optimize import minimize
import math


def objective_func(x, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(x[2*n:])


def boundary_x_ge_r(x, n, i):
    """x >= r constraint."""
    return x[2*i] - x[2*n+i]


def boundary_x_le_1_minus_r(x, n, i):
    """1-x >= r constraint."""
    return 1 - x[2*i] - x[2*n+i]


def boundary_y_ge_r(x, n, i):
    """y >= r constraint."""
    return x[2*i+1] - x[2*n+i]


def boundary_y_le_1_minus_r(x, n, i):
    """1-y >= r constraint."""
    return 1 - x[2*i+1] - x[2*n+i]


def non_overlap_constraint(x, n, i, j):
    """Non-overlap: dist^2 >= (r_i + r_j)^2."""
    dx = x[2*i] - x[2*j]
    dy = x[2*i+1] - x[2*j+1]
    dist_sq = dx*dx + dy*dy
    r_sum = x[2*n+i] + x[2*n+j]
    return dist_sq - r_sum*r_sum


def hex_init(n):
    """Initialize circles in hexagonal pattern: 6-5-6-5-4 = 26."""
    rows_pattern = [6, 5, 6, 5, 4]
    d = 1.0 / 6.0
    r = d / 2.0  # r = 1/12
    
    centers = np.zeros((n, 2))
    radii = np.ones(n) * r
    
    idx = 0
    for row_idx, ncols in enumerate(rows_pattern):
        y = r + row_idx * d * math.sqrt(3) / 2.0
        
        if row_idx % 2 == 0:
            for col in range(ncols):
                x = r + col * d
                centers[idx] = [x, y]
                idx += 1
        else:
            for col in range(ncols):
                x = r + d/2.0 + col * d
                centers[idx] = [x, y]
                idx += 1
    
    return centers, radii


def grid_init(n):
    """Initialize circles in a grid pattern."""
    cols = 6
    rows = int(math.ceil(n / cols))
    
    centers = np.zeros((n, 2))
    dx = 1.0 / (cols + 1)
    dy = 1.0 / (rows + 1)
    
    idx = 0
    for i in range(rows):
        for j in range(cols):
            if idx < n:
                centers[idx, 0] = (j + 1) * dx
                centers[idx, 1] = (i + 1) * dy
                idx += 1
    
    radii = np.ones(n) * 0.04
    return centers, radii


def random_init(n, seed):
    """Random initialization with some structure."""
    rng = np.random.RandomState(seed)
    
    # Place in hex-like random positions
    centers = rng.rand(n, 2) * 0.8 + 0.1
    radii = np.ones(n) * 0.03
    
    return centers, radii


def build_constraints(n):
    """Build all constraints for the optimization."""
    cons = []
    
    # Boundary constraints
    for i in range(n):
        cons.append({'type': 'ineq', 'args': (n, i), 'fun': boundary_x_ge_r})
        cons.append({'type': 'ineq', 'args': (n, i), 'fun': boundary_x_le_1_minus_r})
        cons.append({'type': 'ineq', 'args': (n, i), 'fun': boundary_y_ge_r})
        cons.append({'type': 'ineq', 'args': (n, i), 'fun': boundary_y_le_1_minus_r})
    
    # Non-overlap constraints
    for i in range(n):
        for j in range(i+1, n):
            cons.append({'type': 'ineq', 'args': (n, i, j), 'fun': non_overlap_constraint})
    
    return cons


def optimize_from_initial(centers, radii, n, maxiter=3000):
    """Run optimization from a given initial configuration."""
    x0 = np.concatenate([centers.ravel(), radii])
    
    bounds = [(0, 1)] * (2*n) + [(0, 0.5)] * n
    cons = build_constraints(n)
    
    result = minimize(
        objective_func,
        x0,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        args=(n,),
        options={'maxiter': maxiter, 'ftol': 1e-15, 'disp': False}
    )
    
    centers_opt = result.x[:2*n].reshape((n, 2))
    radii_opt = np.maximum(result.x[2*n:], 0)
    
    return centers_opt, radii_opt, np.sum(radii_opt)
