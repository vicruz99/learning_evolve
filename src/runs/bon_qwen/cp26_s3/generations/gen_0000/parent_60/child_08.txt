import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    
    Returns:
        centers: np.array of shape (26, 2)
        radii: np.array of shape (26,)
        sum_radii: float
    """
    n = 26
    
    # Initial Configuration: 5x5 grid + 1 circle
    # We place 25 circles in a grid and 1 in the center, 
    # then slightly randomize to allow the optimizer to move them freely.
    
    # Grid parameters
    rows = 5
    cols = 5
    
    # Initial centers
    initial_centers = []
    
    # Add 25 circles in a grid
    spacing = 0.25 
    start_pos = 0.125
    
    for r in range(rows):
        for c in range(cols):
            x = start_pos + c * spacing
            y = start_pos + r * spacing
            initial_centers.append([x, y])
            
    # Add 26th circle (initially at center, will be moved)
    initial_centers.append([0.5, 0.5])
    
    initial_centers = np.array(initial_centers)
    initial_radii = np.full(n, 0.1) # Start with reasonable radius
    
    # Flatten for optimizer: [x1, y1, r1, x2, y2, r2, ...]
    x0 = np.zeros(n * 3)
    for i in range(n):
        x0[3*i] = initial_centers[i, 0]
        x0[3*i + 1] = initial_centers[i, 1]
        x0[3*i + 2] = initial_radii[i]
        
    # Bounds for variables
    # x, y in [0, 1], r >= 0
    # We can set tighter bounds for x,y based on r, but simple bounds work for SLSQP
    bounds = []
    for i in range(n):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 1.0)) # r (max radius 0.5 theoretically, 1.0 safe)

    def objective(vars_arr):
        # Sum of radii is the negative of the objective to minimize
        r_sum = 0.0
        for i in range(n):
            r_sum += vars_arr[3*i + 2]
        return -r_sum

    def constraint_overlap(vars_arr):
        # Returns distance - (r1 + r2) >= 0
        # We return an array of constraints
        constraints = []
        for i in range(n):
            for j in range(i + 1, n):
                xi, yi = vars_arr[3*i], vars_arr[3*i + 1]
                ri = vars_arr[3*i + 2]
                
                xj, yj = vars_arr[3*j], vars_arr[3*j + 1]
                rj = vars_arr[3*j + 2]
                
                dist = np.sqrt((xi - xj)**2 + (yi - yj)**2)
                constraints.append(dist - (ri + rj))
        return np.array(constraints)

    def constraint_boundary(vars_arr):
        # Returns margins >= 0
        # x - r >= 0, 1 - x - r >= 0, etc.
        constraints = []
        for i in range(n):
            xi, yi = vars_arr[3*i], vars_arr[3*i + 1]
            ri = vars_arr[3*i + 2]
            
            constraints.append(xi - ri)       # Left
            constraints.append(1.0 - xi - ri)  # Right
            constraints.append(yi - ri)       # Bottom
            constraints.append(1.0 - yi - ri) # Top
        return np.array(constraints)

    # Non-linear constraint definitions for SLSQP
    nonlin_con = {
        'type': 'ineq',
        'fun': constraint_overlap,
        'jac': '2-point' # Approximate Jacobian
    }
    
    boundary_con = {
        'type': 'ineq',
        'fun': constraint_boundary,
        'jac': '2-point'
    }
    
    constraints = [nonlin_con, boundary_con]

    best_sol = None
    best_val = -np.inf
    
    # Try multiple restarts with slight perturbations to escape local minima
    # Since 26 circles is a moderate size, a few restarts help.
    num_restarts = 5
    
    for restart in range(num_restarts):
        # Perturb initial guess slightly
        current_x0 = x0.copy()
        if restart > 0:
            # Random shuffle of circles or random perturbation
            # Shuffling indices helps break symmetry if present
            perm = np.random.permutation(n)
            for i in range(n):
                idx_old = perm[i]
                idx_new = i
                # Swap variables
                for k in range(3):
                    current_x0[3*i + k] = x0[3*idx_old + k]
        
        # Small random noise
        noise = np.random.normal(0, 0.01, size=current_x0.shape)
        # Clamp to valid ranges approximately to keep solver happy
        current_x0[::3] = np.clip(current_x0[::3] + noise[::3], 0, 1)
        current_x0[1::3] = np.clip(current_x0[1::3] + noise[1::3], 0, 1)
        current_x0[2::3] = np.clip(current_x0[2::3] + noise[2::3], 0.01, 0.5) # r > 0

        try:
            res = minimize(objective, current_x0, method='SLSQP', bounds=bounds, 
                           constraints=constraints, 
                           options={'maxiter': 1000, 'ftol': 1e-8, 'disp': False})
            
            if res.success and -res.fun > best_val:
                best_val = -res.fun
                best_sol = res.x
                
        except Exception as e:
            print(f"Optimization error in restart {restart}: {e}")
            continue

    if best_sol is None:
        # Fallback if optimization failed (use original grid with valid radii)
        # This should theoretically not happen with valid start
        return initial_centers, initial_radii, np.sum(initial_radii)

    # Extract solution
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    for i in range(n):
        centers[i, 0] = best_sol[3*i]
        centers[i, 1] = best_sol[3*i + 1]
        radii[i] = best_sol[3*i + 2]
        
    final_sum = np.sum(radii)
    
    return centers, radii, final_sum
