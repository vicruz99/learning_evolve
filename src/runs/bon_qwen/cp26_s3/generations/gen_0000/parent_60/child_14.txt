import numpy as np
from scipy.optimize import linprog, minimize

N = 26

def solve_radii(centers):
    """
    Solves for optimal radii given fixed centers using Linear Programming.
    
    Args:
        centers: np.array of shape (N, 2)
        
    Returns:
        sum_radii: float
        radii: np.array of shape (N,)
    """
    # We want to maximize sum(r_i), so we minimize -sum(r_i)
    c_obj = np.ones(N) * -1.0
    
    A_ub = []
    b_ub = []
    
    # Pairwise distance constraints: r_i + r_j <= dist(i, j)
    # We only need to add constraints for pairs that are likely to be active or all pairs.
    # Adding all pairs is safe and N=26 is small enough (325 constraints).
    for i in range(N):
        for j in range(i + 1, N):
            dist = np.sqrt((centers[i][0] - centers[j][0])**2 + (centers[i][1] - centers[j][1])**2)
            row = np.zeros(N)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dist)
            
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(N):
        x, y = centers[i]
        # r_i <= x
        row = np.zeros(N)
        row[i] = 1.0
        A_ub.append(row)
        b_ub.append(x)
        
        # r_i <= 1 - x
        row = np.zeros(N)
        row[i] = 1.0
        A_ub.append(row)
        b_ub.append(1.0 - x)
        
        # r_i <= y
        row = np.zeros(N)
        row[i] = 1.0
        A_ub.append(row)
        b_ub.append(y)
        
        # r_i <= 1 - y
        row = np.zeros(N)
        row[i] = 1.0
        A_ub.append(row)
        b_ub.append(1.0 - y)
        
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    # Bounds for radii: r_i >= 0
    bounds = [(0, None) for _ in range(N)]
    
    # Solve LP
    # method='highs' is robust and fast
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return -res.fun, -res.x
    else:
        # Fallback if LP fails (should not happen for valid geometry)
        return 0.0, np.zeros(N)

def objective(centers_flat):
    """
    Objective function for optimization.
    Minimizes the negative sum of radii.
    """
    centers = centers_flat.reshape(-1, 2)
    # Clip centers to valid range to avoid numerical issues in LP bounds
    # Although bounds in LP handle x<0 (r<=x implies r<=negative, so r=0), 
    # it's cleaner to keep centers in [0, 1].
    centers = np.clip(centers, 1e-9, 1.0 - 1e-9)
    
    sum_r, radii = solve_radii(centers)
    return -sum_r

def generate_hexagonal_init(n_circles, perturbation=0.01):
    """
    Generates an initial configuration of centers in a hexagonal pattern.
    """
    centers = []
    # Approximate number of columns based on aspect ratio 1:1
    # For square, roughly sqrt(n) columns
    cols = int(np.ceil(np.sqrt(n_circles * 2 / np.sqrt(3))))
    
    count = 0
    row = 0
    while count < n_circles:
        # Determine number of points in this row
        # Alternate between cols and cols-1 or similar to fit
        # Simple hexagonal packing logic:
        # Row 0: cols points
        # Row 1: cols-1 points (offset)
        # ...
        # But we need to fill square.
        
        # Let's just try to place points in a grid and shift odd rows
        # Determine y coordinate
        # Vertical spacing for hex packing is sqrt(3)/2 * diameter
        # Let's assume diameter ~ 0.2 for now
        y = 0.1 + row * 0.1732 # approx
        
        if y > 0.9: 
            break # Adjust if needed, but loop condition handles count
            
        # X coordinates
        if row % 2 == 0:
            # Even rows: 0.1, 0.3, 0.5, ...
            num_in_row = cols
            x_start = 0.1
        else:
            # Odd rows: offset by half step
            num_in_row = cols - 1
            x_start = 0.2 # 0.1 + 0.1
            
        step = 0.2
        for k in range(num_in_row):
            if count >= n_circles:
                break
            x = x_start + k * step
            if x <= 0.9:
                centers.append([x, y])
                count += 1
            else:
                # If x goes out, stop row
                break
        row += 1
        
    # If we didn't get enough, fill remaining randomly or extend
    while len(centers) < n_circles:
        # Simple fallback: add to last row extended or new row
        centers.append([0.5, 0.5]) # Placeholder
        # Actually let's just use a grid fallback if this fails
        pass

    # Convert to array and add perturbation
    if len(centers) < n_circles:
        # Fallback to grid
        centers = []
        # Grid 6x5 = 30 points, take 26
        for r in range(5):
            for c in range(6):
                centers.append([0.1 + c * 0.166, 0.1 + r * 0.2])
        centers = centers[:n_circles]

    centers = np.array(centers)
    centers += np.random.normal(0, perturbation, centers.shape)
    centers = np.clip(centers, 0.05, 0.95)
    
    return centers.flatten()

def generate_grid_init(n_circles, perturbation=0.01):
    """
    Generates an initial configuration of centers in a square grid.
    """
    # Try to arrange in roughly square shape
    cols = int(np.ceil(np.sqrt(n_circles)))
    rows = int(np.ceil(n_circles / cols))
    
    centers = []
    count = 0
    
    # Determine spacing to fit in [0.1, 0.9]
    # Actually just linspace
    x_coords = np.linspace(0.1, 0.9, cols)
    y_coords = np.linspace(0.1, 0.9, rows)
    
    for y in y_coords:
        for x in x_coords:
            if count < n_circles:
                centers.append([x, y])
                count += 1
            else:
                break
    
    centers = np.array(centers)
    centers += np.random.normal(0, perturbation, centers.shape)
    centers = np.clip(centers, 0.05, 0.95)
    return centers.flatten()

def run_packing():

        # Nelder-Mead is derivative-free, good for non-smooth objectives
        try:

    # The solve_radii function ensures validity, but let's be sure

    return best_centers, best_radii, best_sum
