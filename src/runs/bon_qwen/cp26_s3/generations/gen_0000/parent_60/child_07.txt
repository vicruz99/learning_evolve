import numpy as np

def run_packing():
    """
    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    Uses a force-directed relaxation approach with iterative radius growth.
    """
    np.random.seed(42)
    n = 26
    
    # Initialization: Random positions, small radii
    centers = np.random.rand(n, 2)
    radii = np.ones(n) * 0.001
    
    # Parameters
    max_steps = 15000
    dt_radius = 2e-5  # Radius growth step
    learning_rate = 0.15 # Position update scaling
    damping = 0.85     # Velocity damping
    velocity = np.zeros((n, 2))
    
    # Preallocate arrays for efficiency
    diff = np.zeros((n, n, 2))
    dist = np.zeros((n, n))
    req_dist = np.zeros((n, n))
    overlap = np.zeros((n, n))
    inv_dist = np.zeros((n, n))
    unit_diff = np.zeros((n, n, 2))
    force_vectors = np.zeros((n, n, 2))
    forces = np.zeros((n, 2))

    # Main Optimization Loop
    for step in range(max_steps):
        # 1. Grow radii
        radii += dt_radius
        
        # 2. Calculate Pairwise Forces
        diff[:] = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
        dist[:] = np.sqrt(np.sum(diff**2, axis=2))
        np.maximum(dist, 1e-10, out=dist) # Avoid division by zero
        
        req_dist[:] = radii[:, np.newaxis] + radii[np.newaxis, :]
        np.fill_diagonal(req_dist, 0) # No self-interaction
        
        overlap[:] = np.maximum(req_dist - dist, 0)
        
        inv_dist[:] = 1.0 / dist
        unit_diff[:] = diff * inv_dist[:, :, np.newaxis]
        force_vectors[:] = unit_diff * overlap[:, :, np.newaxis]
        forces[:] = np.sum(force_vectors, axis=1)
        
        # 3. Calculate Boundary Forces
        # Left wall: x < r -> push right
        mask_left = centers[:, 0] < radii
        forces[mask_left, 0] += (radii[mask_left] - centers[mask_left, 0])
        # Right wall: x > 1-r -> push left
        mask_right = centers[:, 0] > 1 - radii
        forces[mask_right, 0] -= (centers[mask_right, 0] - (1 - radii[mask_right]))

        mask_bottom = centers[:, 1] < radii
        forces[mask_bottom, 1] += (radii[mask_bottom] - centers[mask_bottom, 1])
        # Top wall: y > 1-r -> push down
        mask_top = centers[:, 1] > 1 - radii
        forces[mask_top, 1] -= (centers[mask_top, 1] - (1 - radii[mask_top]))
        
        # 4. Update Velocity and Position
        velocity = damping * velocity + learning_rate * forces
        centers += velocity
        
        # Clamp positions to stay within [0, 1] to prevent divergence
        np.clip(centers, 1e-9, 1 - 1e-9, out=centers)

    for _ in range(1000):

        unit_diff[:] = diff * inv_dist[:, :, np.newaxis]
        force_vectors[:] = unit_diff * overlap[:, :, np.newaxis]
        forces[:] = np.sum(force_vectors, axis=1)
        
        # Boundary forces
        mask_left = centers[:, 0] < radii
        forces[mask_left, 0] += (radii[mask_left] - centers[mask_left, 0])

        forces[mask_right, 0] -= (centers[mask_right, 0] - (1 - radii[mask_right]))

        forces[mask_bottom, 1] += (radii[mask_bottom] - centers[mask_bottom, 1])
