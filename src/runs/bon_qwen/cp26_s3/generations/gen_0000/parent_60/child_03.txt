import numpy as np
from scipy.optimize import minimize


def run_packing():

    best_radii = None
    best_sum = 0.0

        x0 = np.concatenate([centers0.flatten(), radii0])

            bounds.extend([(0.0, 1.0), (0.0, 1.0)])
            bounds.append((1e-8, 0.5))

        try:

        except Exception:

    if best_centers is None:

    return best_centers, best_radii, best_sum


def init_config(n, trial):
    """Generate various initial configurations."""
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.04
    idx = 0
    
    pattern = trial % 5
    
    if pattern == 0:
        # Hexagonal staggered grid
        row_spacing = 0.175
        for row in range(7):
            y = 0.1 + row * row_spacing
            if y > 0.9:
                break
            if row % 2 == 0:
                xs = np.arange(0.1, 0.92, 0.2)
            else:
                xs = np.arange(0.15, 0.92, 0.2)
            for x in xs:
                if idx < n and x < 0.95:
                    centers[idx] = [x, y]
                    idx += 1
    
    elif pattern == 1:
        # Regular grid 5x6
        for i in range(6):
            for j in range(5):
                if idx < n:
                    centers[idx] = [0.1 + j * 0.2, 0.08 + i * 0.16]
                    idx += 1
    
    elif pattern == 2:
        # Concentric rings
        cx, cy = 0.5, 0.5
        ring_radii = [0.35, 0.25, 0.15, 0.05]
        counts = [8, 8, 6, 4]
        pos = 0
        for rr, nc in zip(ring_radii, counts):
            for k in range(nc):
                if idx < n:
                    angle = 2 * np.pi * k / nc + pos * 0.1
                    centers[idx] = [cx + rr * np.cos(angle), cy + rr * np.sin(angle)]
                    idx += 1
            pos += 1
    
    elif pattern == 3:
        # 4 clusters in corners
        corner_centers = [[0.25, 0.25], [0.75, 0.25], [0.25, 0.75], [0.75, 0.75]]
        per_cluster = n // 4
        extra = n % 4
        for ci, cc in enumerate(corner_centers):
            count = per_cluster + (1 if ci < extra else 0)
            for k in range(count):
                if idx < n:
                    angle = 2 * np.pi * k / count
                    r = 0.12 * (1 + 0.3 * np.sin(k * 2.5))
                    centers[idx] = [cc[0] + r * np.cos(angle), cc[1] + r * np.sin(angle)]
                    idx += 1
    
    else:
        # Random with some structure
        np.random.seed(trial * 13 + 5)
        for i in range(n):
            centers[i] = [np.random.uniform(0.1, 0.9), np.random.uniform(0.1, 0.9)]
    
    # Fill any remaining spots
    while idx < n:
        centers[idx] = [0.5, 0.5]
        idx += 1
    
    return centers, radii


def objective_fun(x):
    """Negate sum of radii (we minimize, so negation maximizes)."""
    n = len(x) // 3
    radii = x[2::3]
    return -np.sum(radii)


def constraint_fun(x):
    """
    Returns array of constraint values. All must be >= 0.
    Uses squared distance for smoothness.
    """
    n = len(x) // 3
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        centers[i, 0] = x[3*i]
        centers[i, 1] = x[3*i+1]
        radii[i] = x[3*i+2]
    
    result = []
    
    # Boundary constraints: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    for i in range(n):
        result.append(centers[i, 0] - radii[i])
        result.append(1.0 - centers[i, 0] - radii[i])
        result.append(centers[i, 1] - radii[i])
        result.append(1.0 - centers[i, 1] - radii[i])
    
    # Pairwise distance constraints: ||c_i - c_j||^2 >= (r_i + r_j)^2
    for i in range(n):
        for j in range(i+1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            ri_plus_rj = radii[i] + radii[j]
            result.append(dx*dx + dy*dy - ri_plus_rj * ri_plus_rj)
    
    return np.array(result)


def unpack_fun(x, n):
    """Unpack optimization variable to centers and radii."""
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        centers[i, 0] = x[3*i]
        centers[i, 1] = x[3*i+1]
        radii[i] = x[3*i+2]
    return centers, radii


def valid_fun(centers, radii, n):
    """Check if packing is valid."""
    for i in range(n):
        xc, yc = centers[i]
        r = radii[i]
        if r < -1e-10:
            return False
        if xc - r < -1e-10 or xc + r > 1 + 1e-10:
            return False
        if yc - r < -1e-10 or yc + r > 1 + 1e-10:
            return False
    for i in range(n):
        for j in range(i+1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx*dx + dy*dy)
            if dist < radii[i] + radii[j] - 1e-10:
                return False
    return True


def force_refine(centers, radii, n):
    """
    Force-based refinement: push overlapping circles apart and
    toward optimal positions. Can escape some local minima.
    """
    step_size = 0.005
    decay = 0.998
    
    for iteration in range(5000):
        forces = np.zeros((n, 2))
        
        # Compute repulsive forces from overlaps
        for i in range(n):
            for j in range(i+1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist_sq = dx*dx + dy*dy
                dist = np.sqrt(dist_sq) if dist_sq > 1e-20 else 1e-10
                min_dist = radii[i] + radii[j]
                
                if dist < min_dist:
                    overlap = min_dist - dist
                    # Strong repulsive force
                    force_mag = overlap * 50.0 / dist
                    fx = dx * force_mag
                    fy = dy * force_mag
                    forces[i, 0] += fx
                    forces[i, 1] += fy
                    forces[j, 0] -= fx
                    forces[j, 1] -= fy
                elif dist < min_dist * 1.5:
                    # Gentle pushing force
                    force_mag = 0.1 / (dist + 0.01)
                    fx = dx * force_mag
                    fy = dy * force_mag
                    forces[i, 0] += fx
                    forces[i, 1] += fy
                    forces[j, 0] -= fx
                    forces[j, 1] -= fy
        
        # Boundary repulsion
        for i in range(n):
            xc, yc = centers[i]
            r = radii[i]
            
            if xc - r < 0.01:
                forces[i, 0] += (0.01 - (xc - r)) * 20.0
            if xc + r > 0.99:
                forces[i, 0] -= (xc + r - 0.99) * 20.0
            if yc - r < 0.01:
                forces[i, 1] += (0.01 - (yc - r)) * 20.0
            if yc + r > 0.99:
                forces[i, 1] -= (yc + r - 0.99) * 20.0
        
        # Apply forces
        centers += forces * step_size * decay**iteration
        centers = np.clip(centers, 0.001, 0.999)
    
    return centers, radii
