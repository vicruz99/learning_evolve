def solve_lp_for_centers(centers, n, n_pairs, pair_indices, c_obj, bounds):
    """
    Solves the Linear Programming problem to find max sum of radii for fixed centers.
    
    Args:
        centers: np.array (n, 2)
        n: int
        n_pairs: int
        pair_indices: np.array (n_pairs, 2) with indices of circle pairs
        c_obj: np.array (n,) objective coefficients
        bounds: list of bounds
    
    Returns:
        sum_radii: float
        radii: np.array (n,)
    """
    # Number of constraints: 4*n (boundary) + n_pairs (pairwise)
    total_constraints = 4 * n + n_pairs
    
    # Construct A_ub and b_ub
    # A_ub shape: (total_constraints, n)
    # We can build this efficiently.
    
    # Initialize A_ub as zeros
    # For performance with N=26, dense matrix is fine.
    A_ub = np.zeros((total_constraints, n))
    b_ub = np.zeros(total_constraints)
    
    # 1. Boundary constraints
    # r_i <= x_i
    # r_i <= 1 - x_i
    # r_i <= y_i
    # r_i <= 1 - y_i
    
    for i in range(n):
        x, y = centers[i]
        
        # Constraint 1: r_i <= x_i  => 1*r_i <= x_i
        row_idx = 4 * i
        A_ub[row_idx, i] = 1.0
        b_ub[row_idx] = x
        
        # Constraint 2: r_i <= 1 - x_i
        row_idx = 4 * i + 1
        A_ub[row_idx, i] = 1.0
        b_ub[row_idx] = 1.0 - x
        
        # Constraint 3: r_i <= y_i
        row_idx = 4 * i + 2
        A_ub[row_idx, i] = 1.0
        b_ub[row_idx] = y
        
        # Constraint 4: r_i <= 1 - y_i
        row_idx = 4 * i + 3
        A_ub[row_idx, i] = 1.0
        b_ub[row_idx] = 1.0 - y

    # 2. Pairwise constraints
    # r_i + r_j <= dist_ij
    
    # Precompute distances
    # dist_sq = (x_i - x_j)^2 + (y_i - y_j)^2
    # We can vectorize this slightly but loop is fine for N=26
    
    # Extract x and y arrays
    xs = centers[:, 0]
    ys = centers[:, 1]
    
    # Calculate distances for all pairs
    # pair_indices has shape (n_pairs, 2)
    # indices i and j
    i_indices = pair_indices[:, 0]
    j_indices = pair_indices[:, 1]
    
    diffs_x = xs[i_indices] - xs[j_indices]
    diffs_y = ys[i_indices] - ys[j_indices]
    dists = np.sqrt(diffs_x**2 + diffs_y**2)
    
    start_row = 4 * n
    
    # Fill A_ub for pairwise constraints
    # For each pair (i, j), row k has 1 at col i and 1 at col j
    # We can use advanced indexing
    
    rows = np.arange(n_pairs) + start_row
    
    # Set coefficients to 1
    A_ub[rows, i_indices] = 1.0
    A_ub[rows, j_indices] = 1.0
    
    # Set RHS
    b_ub[start_row:] = dists
    
    # Solve LP
    # Maximize sum(r) <=> Minimize -sum(r)
    # linprog minimizes c^T x
    # c = -1s
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        
        if res.success:
            radii = res.x
            # Numerical noise might make radii slightly negative or constraints violated by epsilon
            # Clip to 0 just in case
            radii = np.maximum(radii, 0.0)
            return np.sum(radii), radii
        else:
            # If LP fails, return 0
            return 0.0, np.zeros(n)
    except Exception:
        return 0.0, np.zeros(n)
