import numpy as np
from scipy.optimize import minimize


def boundary_violation(centers, radii, n):
    """Compute total squared boundary violation."""
    v = 0.0
    for i in range(n):
        xi = centers[i, 0]
        yi = centers[i, 1]
        ri = radii[i]
        v += max(0.0, ri - xi) ** 2
        v += max(0.0, ri + xi - 1.0) ** 2
        v += max(0.0, ri - yi) ** 2
        v += max(0.0, ri + yi - 1.0) ** 2
    return v


def overlap_violation(centers, radii, n):
    """Compute total squared overlap violation."""
    v = 0.0
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            min_dist = radii[i] + radii[j]
            if min_dist > dist:
                v += (min_dist - dist) ** 2
    return v


def penalty_objective(x, n, penalty):
    """Objective: maximize sum of radii with penalty for constraint violations."""
    centers = x[:2 * n].reshape((n, 2))
    radii = x[2 * n:]
    obj = -np.sum(radii)
    bv = boundary_violation(centers, radii, n)
    ov = overlap_violation(centers, radii, n)
    return obj + penalty * (bv + ov)


def make_hex_initial(n, seed=0):
    """Create initial hexagonal packing with small random perturbations."""
    rng = np.random.RandomState(seed)
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.035

    r0 = 0.035
    idx = 0
    row = 0
    while idx < n:
        y = r0 + row * np.sqrt(3) * r0
        if y + r0 > 1.0:
            for k in range(idx, n):
                centers[k] = [0.5 + rng.uniform(-0.05, 0.05), 0.5 + rng.uniform(-0.05, 0.05)]
                idx = k + 1
            break
        offset = np.sqrt(3) * r0 / 2.0 if row % 2 == 1 else 0.0
        col = 0
        while idx < n:
            x = r0 + offset + col * 2 * r0
            if x + r0 > 1.0:
                break
            centers[idx] = [x + rng.uniform(-0.005, 0.005), y + rng.uniform(-0.005, 0.005)]
            idx += 1
            col += 1
        row += 1

    return centers, radii


def make_dense_initial(n, seed=0):
    """Create a denser initial packing estimate."""
    rng = np.random.RandomState(seed)
    centers = np.zeros((n, 2))
    
    # Estimate radius for n circles in hexagonal packing
    # Area per circle ≈ (2/√3) * r^2, so n * π * r^2 ≈ 0.9069 * 1
    # r ≈ sqrt(0.9069 / (n * π)) ≈ 0.105 for n=26
    r_est = 0.09
    radii = np.ones(n) * r_est

    idx = 0
    row = 0
    while idx < n:
        y = r_est + row * np.sqrt(3) * r_est
        if y + r_est > 1.0:
            for k in range(idx, n):
                centers[k] = [0.5 + rng.uniform(-0.02, 0.02), 0.5 + rng.uniform(-0.02, 0.02)]
                idx = k + 1
            break
        offset = np.sqrt(3) * r_est / 2.0 if row % 2 == 1 else 0.0
        col = 0
        while idx < n:
            x = r_est + offset + col * 2 * r_est
            if x + r_est > 1.0:
                break
            centers[idx] = [x + rng.uniform(-0.003, 0.003), y + rng.uniform(-0.003, 0.003)]
            idx += 1
            col += 1
        row += 1

    return centers, radii


def optimize_packing(x0, n, max_iter=60, init_penalty=50.0):
    """Optimize packing from initial point using increasing penalty method."""
    bounds = []
    for i in range(n):
        bounds.extend([(0.0, 1.0), (0.0, 1.0), (1e-9, 0.5)])

    penalty = init_penalty
    x = x0.copy()

    for iteration in range(max_iter):
        result = minimize(penalty_objective, x, args=(n, penalty),
                         method='L-BFGS-B', bounds=bounds,
                         options={'maxiter': 5000, 'ftol': 1e-17})
        x = result.x
        penalty *= 2.0

        centers_cur = x[:2 * n].reshape((n, 2))
        radii_cur = x[2 * n:]
        bv = boundary_violation(centers_cur, radii_cur, n)
        ov = overlap_violation(centers_cur, radii_cur, n)

        if bv < 1e-15 and ov < 1e-15:
            break

    return x, np.sum(x[2 * n:])


def refine_packing(centers, radii, n, num_refine=5):
    """Further refine an already feasible packing by small perturbations."""
    x = np.concatenate([centers.flatten(), radii])
    bounds = []
    for i in range(n):
        bounds.extend([(0.0, 1.0), (0.0, 1.0), (1e-9, 0.5)])

    # Use a very high penalty to maintain feasibility while optimizing
    penalty = 1e8
    for ref_iter in range(num_refine):
        result = minimize(penalty_objective, x, args=(n, penalty),
                         method='L-BFGS-B', bounds=bounds,
                         options={'maxiter': 3000, 'ftol': 1e-17})
        x = result.x
        penalty *= 2

    return x[:2 * n].reshape((n, 2)), x[2 * n:]


def run_packing():
    n = 26
    best_sum_r = 0.0
    best_x = None

    # Try dense initial packings with various seeds
    for seed in range(8):
        centers, radii = make_dense_initial(n, seed=seed)
        x0 = np.concatenate([centers.flatten(), radii])
        x_opt, sum_r = optimize_packing(x0, n, max_iter=50, init_penalty=30.0)

        if sum_r > best_sum_r:
            best_sum_r = sum_r
            best_x = x_opt.copy()

    # Also try sparser initial packings that might lead to different local optima
    for seed in range(5):
        centers, radii = make_hex_initial(n, seed=seed)
        x0 = np.concatenate([centers.flatten(), radii])
        x_opt, sum_r = optimize_packing(x0, n, max_iter=50, init_penalty=30.0)

        if sum_r > best_sum_r:
            best_sum_r = sum_r
            best_x = x_opt.copy()

    # Extract and finalize best solution
    centers_opt = best_x[:2 * n].reshape((n, 2))
    radii_opt = np.maximum(best_x[2 * n:], 0.0)

    # Final projection to ensure strict boundary feasibility
    for i in range(n):
        r = radii_opt[i]
        centers_opt[i, 0] = np.clip(centers_opt[i, 0], r + 1e-13, 1.0 - r - 1e-13)
        centers_opt[i, 1] = np.clip(centers_opt[i, 1], r + 1e-13, 1.0 - r - 1e-13)

    # Try to refine the best solution further
    centers_opt, radii_opt = refine_packing(centers_opt, radii_opt, n, num_refine=8)

    # Final boundary projection after refinement
    for i in range(n):
        r = radii_opt[i]
        centers_opt[i, 0] = np.clip(centers_opt[i, 0], r + 1e-13, 1.0 - r - 1e-13)
        centers_opt[i, 1] = np.clip(centers_opt[i, 1], r + 1e-13, 1.0 - r - 1e-13)

    final_sum_r = np.sum(radii_opt)
    return centers_opt, radii_opt, final_sum_r
