def solve_radii_lp(centers):
    """
    Solves the Linear Programming problem to find optimal radii for fixed centers.
    Maximizes sum(r_i) subject to non-overlap and boundary constraints.
    """
    n = centers.shape[0]
    
    # Variables: r_0, r_1, ..., r_{n-1}
    # Objective: Maximize sum(r_i) -> Minimize -sum(r_i)
    c_obj = np.ones(n)
    
    # Inequality constraints A_ub @ x <= b_ub
    constraints_A = []
    constraints_b = []
    
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    # This can be formulated as r_i <= min(x_i, 1-x_i, y_i, 1-y_i)
    # We can just add upper bounds to variables instead of explicit constraints 
    # to keep A_ub small, but linprog supports bounds directly.
    
    bounds = []
    for i in range(n):
        x, y = centers[i]
        max_r = min(x, 1 - x, y, 1 - y)
        bounds.append((0, max_r))
        
    # Non-overlap constraints: r_i + r_j <= distance(i, j)
    # r_i + r_j - dist <= 0
    # We need to construct A_ub matrix rows.
    # However, constructing a dense 325x26 matrix is fine.
    
    # Let's use list of lists for constraints to build sparse or just dense array
    # Number of pairwise constraints = n*(n-1)/2
    num_pairs = n * (n - 1) // 2
    A_ub = np.zeros((num_pairs, n))
    b_ub = np.zeros(num_pairs)
    
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            # r_i + r_j <= dist
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dist
            idx += 1
            
    try:
        # Use high performance solver if available, otherwise default
        res = opt.linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            radii = res.x
            sum_radii = np.sum(radii)
            return radii, sum_radii
        else:
            # Fallback or error handling
