   - Violations of the unit square boundary.
4. **Convergence**: Increase the radius step when the penalty is low (valid packing found). This "expanding circles" approach naturally pushes circles into a jammed, optimal configuration.

import numpy as np
from scipy.optimize import minimize

def compute_loss(vars, r, n):
    """
    Compute penalty for overlaps and boundary violations.
    vars: flattened array of centers
    r: current target radius
    n: number of circles
    """
    c = vars.reshape((n, 2))
    loss = 0.0
    
    # Boundary constraints
    for k in range(n):
        x, y = c[k]
        if x < r: 
            loss += (x - r)**2
        elif x > 1 - r: 
            loss += (x - (1 - r))**2
            
        if y < r: 
            loss += (y - r)**2
        elif y > 1 - r: 
            loss += (y - (1 - r))**2
            
    # Overlap constraints
    for k in range(n):
        for l in range(k + 1, n):
            dx = c[k, 0] - c[l, 0]
            dy = c[k, 1] - c[l, 1]
            dist_sq = dx*dx + dy*dy
            min_dist = 2 * r
            
            if dist_sq < min_dist**2:
                # Use smooth approximation for gradient stability
                dist = np.sqrt(dist_sq)
                loss += (min_dist - dist)**2
                
    return loss

def run_packing():
    n = 26
    # Hexagonal lattice initialization
    # Row distribution: 4, 5, 4, 5, 4, 4 sums to 26
    row_counts = [4, 5, 4, 5, 4, 4]
    
    r = 0.04  # Initial small radius
    centers = np.zeros((n, 2))
    idx = 0
    
    for i, count in enumerate(row_counts):
        y = r + i * np.sqrt(3) * r
        # Shift odd rows for hexagonal packing
        if i % 2 == 0:
            x_start = r
        else:
            x_start = 2 * r
            
        for j in range(count):
            x = x_start + j * 2 * r
            centers[idx] = [x, y]
            idx += 1
            
    radii = np.full(n, r)
    target_sum = 2.636
    target_r = target_sum / n
    
    # Iteratively expand circles
    for _ in range(300):
        # Optimize positions for current radius
        res = minimize(compute_loss, centers.flatten(), args=(r, n), method='L-BFGS-B',
                       bounds=[(0.0, 1.0) for _ in range(2 * n)], 
                       options={'maxiter': 200, 'ftol': 1e-10})
        
        centers = res.x.reshape((n, 2))
        
        # If valid packing found, increase radius significantly
        if res.fun < 1e-4:
            r += 0.0003
            radii[:] = r
        else:
            # If stuck, increase slowly to encourage movement
            r += 0.00005
            radii[:] = r
            
        if r >= target_r:
            break
            
    # Final refinement
    res_final = minimize(compute_loss, centers.flatten(), args=(r, n), method='L-BFGS-B',
                         bounds=[(0.0, 1.0) for _ in range(2 * n)], 
                         options={'maxiter': 500, 'ftol': 1e-12})
    
    centers = res_final.x.reshape((n, 2))
    radii = np.full(n, r)
    
    return centers, radii, np.sum(radii)
