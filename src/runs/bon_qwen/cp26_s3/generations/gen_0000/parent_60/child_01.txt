import numpy as np

def compute_interactions(centers, radii, k_force=50.0):
    """
    Computes repulsive forces and minimum clearances for each circle.
    Returns forces array and clearances array.
    """
    N = len(radii)
    forces = np.zeros_like(centers)
    clearances = np.full(N, 1.0)
    
    for i in range(N):
        r = radii[i]
        x, y = centers[i]
        
        # Boundary interactions
        # X-axis
        if x < r:
            forces[i, 0] += k_force * (r - x)
            clearances[i] = min(clearances[i], x - r)
        elif x > 1.0 - r:
            forces[i, 0] -= k_force * (x - (1.0 - r))
            clearances[i] = min(clearances[i], 1.0 - x - r)
        else:
            clearances[i] = min(clearances[i], x - r, 1.0 - x - r)
            
        # Y-axis
        if y < r:
            forces[i, 1] += k_force * (r - y)
            clearances[i] = min(clearances[i], y - r)
        elif y > 1.0 - r:
            forces[i, 1] -= k_force * (y - (1.0 - r))
            clearances[i] = min(clearances[i], 1.0 - y - r)
        else:
            clearances[i] = min(clearances[i], y - r, 1.0 - y - r)
            
        # Pairwise interactions
        for j in range(i + 1, N):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.hypot(dx, dy)
            
            if dist < 1e-8:
                dist = 1e-8
                dx = np.random.rand() * 1e-4
                dy = np.random.rand() * 1e-4
                
            r_sum = radii[i] + radii[j]
            penetration = r_sum - dist
            
            if penetration > 0:
                f_mag = k_force * penetration
                fx = f_mag * dx / dist
                fy = f_mag * dy / dist
                forces[i, 0] += fx
                forces[i, 1] += fy
                forces[j, 0] -= fx
                forces[j, 1] -= fy
                
            c = dist - r_sum
            if c < clearances[i]:
                clearances[i] = c
            if c < clearances[j]:
                clearances[j] = c
                
    return forces, clearances

def run_packing():
    np.random.seed(42)
    N = 26
    best_centers = None
    best_radii = None
