def compute_max_radii_sum(centers):
    """
    Solves the LP to find the optimal radii for fixed centers 
    that maximize the sum of radii.
    
    Args:
        centers: np.array of shape (n, 2)
        
    Returns:
        sum_radii: float, sum of optimal radii
        radii: np.array, optimal radii
    """
    n = centers.shape[0]
    
    # Objective: Maximize sum(r_i) -> Minimize sum(-r_i)
    c_obj = -np.ones(n)
    
    # Constraints: r_i + r_j <= dist(c_i, c_j)

    pair_constraints = []
    pair_b = []
    
    for i in range(n):
        for j in range(i + 1, n):
            # r_i + r_j <= dist
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            row = np.zeros(n)
            row[i] = 1.0

            pair_constraints.append(row)
            pair_b.append(dist)
