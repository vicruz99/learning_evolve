import numpy as np
from scipy.optimize import minimize

def compute_violation(centers_flat, r):
    """
    Computes the sum of squared violations for boundary and overlap constraints.
    centers_flat: 1D array of length 2*n with [x0, y0, x1, y1, ...]
    r: target radius
    """
    n = len(centers_flat) // 2
    centers = centers_flat.reshape((n, 2))
    v = 0.0
    
    # Boundary violations: circle must be inside [0,1]x[0,1]
    for i in range(n):
        x, y = centers[i]
        for val in (x, 1.0 - x, y, 1.0 - y):
            diff = r - val
            if diff > 0.0:
                v += diff**2
                
    # Overlap violations: distance between centers must be >= 2*r
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.hypot(dx, dy)
            diff = 2.0 * r - dist
            if diff > 0.0:
                v += diff**2
                
    return v

            centers[idx, 1] = y
            idx += 1
        y += np.sqrt(3) / 2 * 0.2
        
    # Add tiny random perturbation to break symmetry and avoid local minima
    centers += np.random.randn(n, 2) * 1e-5
    
    # Binary search for the maximum feasible radius
    r_lo, r_hi = 0.1, 0.105
    best_r = 0.1
    best_centers = centers.copy()

    current_centers = centers.copy()
    
    for _ in range(15):
        r_mid = (r_lo + r_hi) / 2.0
        
        # Optimize centers to minimize constraint violations for r_mid
        res = minimize(
            compute_violation,
            current_centers.flatten(),
            args=(r_mid,),
            method='L-BFGS-B',
            bounds=[(1e-7, 1.0 - 1e-7)] * (2 * n),

            best_centers = current_centers.copy()
            r_lo = r_mid
        else:
            r_hi = r_mid
            
    radii = np.full(n, best_r)
    total_sum = float(np.sum(radii))
    
    return best_centers, radii, total_sum
