**Analysis:**
1.  **Objective:** Maximize $\sum r_i$.
2.  **Constraints:** Non-overlap ($r_i + r_j \le d_{ij}$) and boundary containment ($r_i \le \text{dist}(center_i, \text{boundary})$).
3.  **Strategy:** This is a non-convex optimization problem. A hybrid approach is best:
    *   **Center Optimization:** Use a derivative-free global optimizer (Simulated Annealing) to find optimal centers.
    *   **Radius Calculation:** For any fixed set of centers, the problem of finding radii to maximize sum is a Linear Programming (LP) problem. We solve this efficiently using `scipy.optimize.linprog`.
4.  **Heuristics:** Start with a hexagonal grid (densest packing) and perturb. Handle boundary conditions strictly.

**Implementation Plan:**
1.  **Initialization:** Place circles in a hexagonal pattern.
2.  **LP Solver:** A function `solve_radii(centers)` returns the optimal radii given fixed centers.

import numpy as np
import scipy.optimize as opt

# Set random seed for reproducibility
np.random.seed(42)

def solve_radii(centers):
    """
    Solves the Linear Programming problem to maximize sum of radii
    given fixed centers.
    
    Args:
        centers: np.array of shape (n, 2)
        
    Returns:
        radii: np.array of shape (n)
    """
    n = centers.shape[0]
    
    # Objective: Maximize sum(r_i) => Minimize -sum(r_i)
    c_obj = -np.ones(n)
    
    # Variables: r_0, r_1, ..., r_n-1
    # Lower bound: r_i >= 0
    # Upper bound: r_i <= min(x, 1-x, y, 1-y)
    bounds = []
    for i in range(n):
        x, y = centers[i]
        # Distance to boundaries
        dist_to_bound = min(x, 1 - x, y, 1 - y)
        # Radius cannot be negative, and limited by boundary distance
        # If center is outside, dist is negative, radius must be 0 (handled by bounds)
        low = 0.0
        high = max(0.0, dist_to_bound)
        bounds.append((low, high))
    
    # Constraints: r_i + r_j <= dist(i, j)
    # Matrix form: A_ub @ r <= b_ub
    num_constraints = n * (n - 1) // 2
    A_ub = np.zeros((num_constraints, n))
    b_ub = np.zeros(num_constraints)
    
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dist
            idx += 1
            
    # Solve LP
    # Method 'highs' is fast and robust
    try:
        res = opt.linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
        else:
            # Fallback if LP fails (should not happen with valid centers)
            return np.zeros(n)
    except Exception:
        return np.zeros(n)

def evaluate_sum_radii(centers):
    """
    Computes the sum of radii for a given configuration of centers.
    """
    # Clip centers to [0, 1] to ensure validity, though optimizer should keep them there
    centers_clipped = np.clip(centers, 1e-5, 1 - 1e-5)
    radii = solve_radii(centers_clipped)
    return np.sum(radii), radii, centers_clipped

def run_packing():
    n_circles = 26
    
    # 1. Initialization: Hexagonal packing approximation
    # We arrange circles in rows with staggered positions
    # Aim for roughly 5 rows to fit 26 circles
    centers = np.zeros((n_circles, 2))
    
    # Grid parameters
    # 5 rows, varying number of circles
    # Row 0: 5, Row 1: 6, Row 2: 5, Row 3: 6, Row 4: 4 (Sum 26) -> Too sparse
    # Let's try to pack them tightly.
    # Standard hex grid coordinates
    x = 0.1
    y = 0.1
    dx = 0.18  # Horizontal spacing
    dy = 0.15  # Vertical spacing (sqrt(3)/2 * dx approx)
    
    idx = 0
    row = 0
    while idx < n_circles and row < 10:
        # Determine number of circles in this row
        # Staggered rows have slightly fewer or same count depending on width
        # Simple heuristic: fill rows
        count_in_row = 6
        if row % 2 == 1:
            count_in_row = 5 # Staggered
        
        # Adjust count to fit in square
        # Max x extent for 'k' circles starting at x with spacing dx:
        # x + (k-1)*dx + r <= 1. Approx r~0.1.
        
        # Just place them, optimizer will fix positions
        for i in range(count_in_row):
            if idx >= n_circles:
                break
            # Shift x for staggered rows
            offset = dx / 2 if row % 2 == 1 else 0
            
            cx = 0.15 + i * dx + offset
            cy = 0.15 + row * dy
            
            # Keep within bounds roughly
            if cx > 0.9: cx = 0.9
            if cy > 0.9: cy = 0.9
            
            centers[idx] = [cx, cy]
            idx += 1
        row += 1

    # Fill remaining if any (should be none with loop logic above)
    # But ensure we have 26
    # The logic above might not be perfect, let's just initialize randomly if needed, 
    # but the hex pattern is a good start.
    # Let's verify count
    if centers.shape[0] < n_circles:
        # Fill rest randomly
        for i in range(centers.shape[0], n_circles):
            centers[i] = [np.random.uniform(0.2, 0.8), np.random.uniform(0.2, 0.8)]
    
    # Resize to exactly n_circles
    centers = centers[:n_circles]

    # 2. Optimization using Simulated Annealing
    # We want to maximize sum_radii. SA minimizes energy, so Energy = -sum_radii
    
    best_sum = 0
    best_centers = centers
    best_radii = np.zeros(n_circles)
    
    # Current state
    current_centers = best_centers.copy()
    current_sum, current_radii, _ = evaluate_sum_radii(current_centers)
    if current_sum > best_sum:
        best_sum = current_sum
        best_centers = current_centers
        best_radii = current_radii

    # SA Parameters
    temperature = 0.1
    cooling_rate = 0.995
    step_size = 0.05
    iterations = 5000 # Number of iterations
    
    for i in range(iterations):
        # Perturb centers
        # Only perturb a subset or all? Perturbing all is fine for small n.
        # Random move for each circle
        noise = np.random.normal(0, step_size, size=current_centers.shape)
        proposed_centers = current_centers + noise
        
        # Clip to [0, 1]
        proposed_centers = np.clip(proposed_centers, 0, 1)
        
        # Evaluate
        proposed_sum, proposed_radii, proposed_centers_valid = evaluate_sum_radii(proposed_centers)
        
        # Acceptance criterion
        delta = proposed_sum - current_sum
        
        if delta > 0:
            current_centers = proposed_centers_valid
            current_radii = proposed_radii
            current_sum = proposed_sum
            
            # Update best
            if current_sum > best_sum:
                best_sum = current_sum
                best_centers = current_centers
                best_radii = current_radii
        else:
            # Accept with probability exp(delta / T)
            # delta is negative
            prob = np.exp(delta / temperature)
            if np.random.random() < prob:
                current_centers = proposed_centers_valid
                current_radii = proposed_radii
                current_sum = proposed_sum
        
        # Cool down
        temperature *= cooling_rate
        step_size *= 0.999 # Reduce step size slowly
        
    # Final check and return
    # Re-run LP on best centers to ensure consistency
    final_sum, final_radii, final_centers = evaluate_sum_radii(best_centers)
    
    return final_centers, final_radii, final_sum

if __name__ == "__main__":
    centers, radii, sum_r = run_packing()
    print(f"Sum of radii: {sum_r}")
    print(f"Max radius: {np.max(radii)}, Min radius: {np.min(radii)}")
