import numpy as np
from scipy.optimize import minimize


def make_boundary_constraint(i, n, ctype):
    """Create boundary constraint for circle i."""
    def constraint(p):
        if ctype == 0:
            return p[i] - p[2 * n + i]
        elif ctype == 1:
            return 1.0 - p[i] - p[2 * n + i]
        elif ctype == 2:
            return p[i + n] - p[2 * n + i]
        else:
            return 1.0 - p[i + n] - p[2 * n + i]
    return constraint


def make_overlap_constraint(i, j, n):
    """Create non-overlap constraint for circles i and j."""
    def constraint(p):
        dx = p[i] - p[j]
        dy = p[i + n] - p[j + n]
        dist = np.sqrt(dx * dx + dy * dy)
        return dist - p[2 * n + i] - p[2 * n + j]
    return constraint


def make_radius_constraint(i, n):
    """Create positive radius constraint for circle i."""
    def constraint(p):
        return p[2 * n + i]
    return constraint


def generate_hexagonal_positions(n, arrangement):
    """Generate hexagonal packing positions for given row arrangement."""
    positions = []
    r_guess = 0.1
    h = r_guess * np.sqrt(3.0)

    for row, num_in_row in enumerate(arrangement):
        y = r_guess + row * h
        for col in range(num_in_row):
            x = r_guess + col * 2.0 * r_guess
            if row % 2 == 1:
                x += r_guess
            positions.append([x, y])

    positions = np.array(positions[:n])
    return positions


def normalize_positions(positions, margin=0.08):
    """Scale and center positions within [margin, 1-margin]."""
    positions = positions.copy()
    x_min, y_min = positions.min(axis=0)
    x_max, y_max = positions.max(axis=0)

    range_x = x_max - x_min
    range_y = y_max - y_min

    if range_x < 1e-10:
        range_x = 1.0
    if range_y < 1e-10:
        range_y = 1.0

    scale_x = (1.0 - 2.0 * margin) / range_x
    scale_y = (1.0 - 2.0 * margin) / range_y
    scale = min(scale_x, scale_y)

    positions[:, 0] = margin + (positions[:, 0] - x_min) * scale
    positions[:, 1] = margin + (positions[:, 1] - y_min) * scale

    return positions


def compute_min_separation(positions, n):
    """Compute minimum separation between any two points."""
    min_dist = float('inf')
    for i in range(n):
        for j in range(i + 1, n):
            d = np.linalg.norm(positions[i] - positions[j])
            if d < min_dist:
                min_dist = d
    return min_dist


def run_optimization(positions, radii, n, max_iter=4000):
    """Run SLSQP optimization to maximize sum of radii."""
    params = np.concatenate([positions.flatten(), radii])

    constraints = []

    for i in range(n):
        for ctype in range(4):
            constraints.append({
                'type': 'ineq',
                'fun': make_boundary_constraint(i, n, ctype)
            })

    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({
                'type': 'ineq',
                'fun': make_overlap_constraint(i, j, n)
            })

    for i in range(n):
        constraints.append({
            'type': 'ineq',
            'fun': make_radius_constraint(i, n)
        })

    bounds = [(0.0, 1.0)] * (2 * n) + [(1e-8, 0.5)] * n

    def objective(p):
        return -np.sum(p[2 * n:])

    result = minimize(
        objective, params, method='SLSQP',
        constraints=constraints, bounds=bounds,
        options={'maxiter': max_iter, 'ftol': 1e-14, 'disp': False}
    )

    centers = result.x[:2 * n].reshape(n, 2)
    radii = np.maximum(result.x[2 * n:], 1e-8)

    # Clip centers to be valid
    for i in range(n):
        centers[i, 0] = np.clip(centers[i, 0], radii[i], 1.0 - radii[i])
        centers[i, 1] = np.clip(centers[i, 1], radii[i], 1.0 - radii[i])

    return centers, radii


def refine_packing(centers, radii, n, iterations=200):
    """Refine packing by iteratively adjusting positions and expanding radii."""
    centers = centers.copy()
    radii = radii.copy()

    for iteration in range(iterations):
        learning_rate = 0.005 / (1.0 + iteration * 0.02)
        moved = False

        for i in range(n):
            fx, fy = 0.0, 0.0

            # Forces from other circles
            for j in range(n):
                if i == j:
                    continue
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                if dist < 1e-12:
                    dist = 1e-12

                desired = radii[i] + radii[j] + 0.002
                if dist < desired:
                    force = (desired - dist) / dist * 0.5
                    fx += dx * force
                    fy += dy * force

            # Forces from boundaries
            if centers[i, 0] - radii[i] < 0.005:
                fx += 0.3
            if 1.0 - centers[i, 0] - radii[i] < 0.005:
                fx -= 0.3
            if centers[i, 1] - radii[i] < 0.005:
                fy += 0.3
            if 1.0 - centers[i, 1] - radii[i] < 0.005:
                fy -= 0.3

            centers[i, 0] += fx * learning_rate
            centers[i, 1] += fy * learning_rate
            moved = True

        # Recalculate radii based on constraints
        for i in range(n):
            max_r = min(
                centers[i, 0],
                1.0 - centers[i, 0],
                centers[i, 1],
                1.0 - centers[i, 1]
            )
            for j in range(n):
                if i == j:
                    continue
                d = np.linalg.norm(centers[i] - centers[j])
                max_r = min(max_r, d - radii[j])
            radii[i] = max(1e-8, max_r)

    return centers, radii

        current_sum = np.sum(radii)

        best_centers[i, 1] = np.clip(best_centers[i, 1], best_radii[i], 1.0 - best_radii[i])
