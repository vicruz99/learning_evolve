import numpy as np
from scipy.optimize import minimize, linprog

def objective_function(vars, n):
    c = vars[:2*n].reshape(n, 2)
    r = vars[2*n:]
    
    # Objective: maximize sum(r) => minimize -sum(r)
    obj = -np.sum(r)
    
    penalty = 0.0
    lambda_pen = 1000.0
    
    # Boundary penalties
    # r <= x  => r - x <= 0
    # r <= 1-x => r + x - 1 <= 0
    # r <= y
    # r <= 1-y
    
    # Vectorized
    diff_x = r - c[:, 0]
    diff_1x = r + c[:, 0] - 1.0
    diff_y = r - c[:, 1]
    diff_1y = r + c[:, 1] - 1.0
    
    p_x = np.maximum(0, diff_x)**2
    p_1x = np.maximum(0, diff_1x)**2
    p_y = np.maximum(0, diff_y)**2
    p_1y = np.maximum(0, diff_1y)**2
    
    penalty += np.sum(p_x + p_1x + p_y + p_1y)
    
    # Overlap penalties
    # r_i + r_j <= dist
    # dist = ||c_i - c_j||
    
    # Using broadcasting for pairwise distances
    # c shape (n, 2)
    # c[:, np.newaxis, :] -> (n, 1, 2)
    # c[np.newaxis, :, :] -> (1, n, 2)
    diff_matrix = c[:, np.newaxis, :] - c[np.newaxis, :, :]
    dist_matrix = np.sqrt(np.sum(diff_matrix**2, axis=2))
    
    r_sum_matrix = r[:, np.newaxis] + r[np.newaxis, :]
    
    violation = r_sum_matrix - dist_matrix
    
    # Upper triangle mask
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    violation_tri = violation[mask]
    
    penalty += np.sum(np.maximum(0, violation_tri)**2)
    
    return obj + lambda_pen * penalty
