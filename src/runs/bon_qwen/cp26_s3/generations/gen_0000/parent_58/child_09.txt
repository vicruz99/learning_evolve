def check_validity(centers, radii):
    n = centers.shape[0]
    # Boundary
    for i in range(n):
        if radii[i] < 0: return False
        if centers[i][0] < radii[i] or centers[i][0] > 1 - radii[i]: return False
        if centers[i][1] < radii[i] or centers[i][1] > 1 - radii[i]: return False
    # Overlap
    for i in range(n):
        for j in range(i+1, n):
            dist = np.sqrt(np.sum((centers[i]-centers[j])**2))
            if dist < radii[i] + radii[j] - 1e-9:
                return False
    return True
