import numpy as np
from scipy.optimize import minimize


def compute_dist_matrix(centers):
    """Compute pairwise distances between all circle centers."""
    n = centers.shape[0]
    dists = np.zeros((n, n))
    for i in range(n):
        for j in range(i + 1, n):
            d = np.sqrt((centers[i, 0] - centers[j, 0]) ** 2 + (centers[i, 1] - centers[j, 1]) ** 2)
            dists[i, j] = d
            dists[j, i] = d
    return dists


def iterative_expand(centers, radii, n, iterations=500):
    """Iteratively expand circles to their maximum possible radii."""
    for _ in range(iterations):
        perm = np.random.permutation(n)
        for i in perm:
            max_r = min(centers[i, 0], 1.0 - centers[i, 0],
                        centers[i, 1], 1.0 - centers[i, 1])
            
            for j in range(n):
                if i != j:
                    d = np.sqrt((centers[i, 0] - centers[j, 0]) ** 2 + 
                               (centers[i, 1] - centers[j, 1]) ** 2)
                    max_r = min(max_r, d - radii[j])
            
            radii[i] = max(1e-8, max_r)
    return centers, radii


def hexagonal_initial(n, seed=0):

    np.random.seed(seed)
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.085
    
    idx = 0
    for row in range(7):
        n_in_row = 5 if row % 2 == 0 else 4
        for col in range(n_in_row):
            if idx < n:
                x_spacing = 1.0 / (n_in_row + 1)
                offset = 0.5 * x_spacing if row % 2 == 1 else 0
                centers[idx, 0] = (col + 1) * x_spacing + offset
                centers[idx, 1] = (row + 1) * 1.0 / 8
                idx += 1
    
    while idx < n:
        centers[idx, 0] = np.random.uniform(0.15, 0.85)
        centers[idx, 1] = np.random.uniform(0.15, 0.85)
        idx += 1
    
    return centers, radii

def objective(params, n):
    """Objective: maximize sum of radii (negated for minimization)."""
    return -np.sum(params[2 * n:])


def boundary_constraints(params, n):
    """Boundary constraints: circles must be inside [0,1]^2."""
    centers = params[:2 * n].reshape(n, 2)
    radii = params[2 * n:]
    con = []
    for i in range(n):
        con.append(centers[i, 0] - radii[i])
        con.append(1.0 - centers[i, 0] - radii[i])
        con.append(centers[i, 1] - radii[i])
        con.append(1.0 - centers[i, 1] - radii[i])
        con.append(radii[i])
    return con


def non_overlap_constraints(params, n):
    """Non-overlap constraints: distance >= sum of radii."""
    centers = params[:2 * n].reshape(n, 2)
    radii = params[2 * n:]
    con = []
    for i in range(n):
        for j in range(i + 1, n):
            d = np.sqrt((centers[i, 0] - centers[j, 0]) ** 2 + 
                       (centers[i, 1] - centers[j, 1]) ** 2)
            con.append(d - radii[i] - radii[j])
    return con


def all_constraints(params, n):
    """Combine all constraints."""
    con = boundary_constraints(params, n)
    con.extend(non_overlap_constraints(params, n))
    return con


def local_optimize(centers, radii, n, max_iter=20000):
    """Run local optimization using SLSQP."""
    x0 = np.concatenate([centers.flatten(), radii])
    
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
    
    cons = [{'type': 'ineq', 'fun': lambda p: all_constraints(p, n)}]
    
    result = minimize(
        objective,
        x0,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        args=(n,),
        options={'maxiter': max_iter, 'ftol': 1e-15, 'disp': False}
    )
    
    if result.success or result.fun < -0:
        opt_centers = result.x[:2 * n].reshape(n, 2)
        opt_radii = result.x[2 * n:]
        return opt_centers, opt_radii
    return centers, radii
