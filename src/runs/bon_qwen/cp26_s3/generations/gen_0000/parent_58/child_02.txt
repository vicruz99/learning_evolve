    Main function to run the circle packing optimization.
    """
    n = 26
    
    # Bounds: x, y in [0, 1], r in [0, 0.5]
    bounds = []
    for _ in range(n):
        bounds.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])

    best_centers = None
    best_radii = None
    max_sum_radii = -1.0
    
    np.random.seed(42)
    num_restarts = 10
    
    for _ in range(num_restarts):
        # Initialization: Hexagonal grid perturbed by randomness
        centers_init = []
        radii_init = []
        
        # Start with a small radius to ensure valid initial placement
        r_start = 0.05 
        spacing = 2 * r_start
        row_height = spacing * np.sqrt(3) / 2
        
        y = r_start
        row_idx = 0
        while y < 1 - r_start:
            # Offset for hexagonal packing
            offset = 0 if row_idx % 2 == 0 else spacing / 2
            x = r_start + offset
            while x < 1 - r_start:
                if len(centers_init) < n:
                    centers_init.append([x, y])
                    radii_init.append(r_start)
                x += spacing
            y += row_height
            row_idx += 1
            
        # If grid didn't fill enough, add random points
        if len(centers_init) < n:
            remaining = n - len(centers_init)
            random_centers = np.random.uniform(r_start, 1 - r_start, size=(remaining, 2))
            centers_init.extend(random_centers.tolist())
            radii_init.extend([r_start] * remaining)
            
        # Ensure we have exactly n circles
        centers_init = centers_init[:n]
        radii_init = radii_init[:n]
        
        # Flatten to optimization variables vector
        vars_vec = np.zeros(3 * n)
        for i in range(n):
            vars_vec[3*i] = centers_init[i][0]
            vars_vec[3*i+1] = centers_init[i][1]
            vars_vec[3*i+2] = radii_init[i]
            
        # Run optimization
        res = optimize_packing(n, vars_vec, bounds)
        
        # Extract results
        c_x = res.x[0::3]
        c_y = res.x[1::3]
        r = res.x[2::3]
        
        sum_r = np.sum(r)
        
        # Verify feasibility with high penalty
        # If penalty is low, the solution is valid
        penalty_val = calc_loss(res.x, n, 10000) - (-sum_r)
        
        # Update best solution if sum of radii is higher and solution is feasible
        if sum_r > max_sum_radii and penalty_val < 1e-5:
            max_sum_radii = sum_r
            best_centers = np.column_stack((c_x, c_y))
            best_radii = r
            
        # Fallback if no strictly feasible solution found yet (rare)
        if best_centers is None:
            best_centers = np.column_stack((c_x, c_y))
            best_radii = r
            max_sum_radii = sum_r

    return best_centers, best_radii, max_sum_radii
