import numpy as np
from scipy.optimize import minimize

def compute_min_dist(centers):
    """Compute the maximum feasible radius for the given centers."""
    n = centers.shape[0]
    
    # Distance to boundaries
    dx = np.minimum(centers[:, 0], 1.0 - centers[:, 0])
    dy = np.minimum(centers[:, 1], 1.0 - centers[:, 1])
    min_bound = np.min(np.minimum(dx, dy))
    
    # Pairwise distances
    diff = centers[:, None, :] - centers[None, :, :]
    dists_sq = np.sum(diff**2, axis=2)
    
    # Use lower triangle indices (excluding diagonal)
    idx = np.tril_indices(n, -1)
    min_pair = np.sqrt(np.min(dists_sq[idx])) / 2.0
    
    return min(min_bound, min_pair)

def objective(pos, n):
    """Objective to minimize: negative of the maximum feasible radius."""
    c = pos.reshape(-1, 2)
    c = np.clip(c, 0.0, 1.0)  # Ensure centers stay within the square
    return -compute_min_dist(c)

def run_packing():
    n = 26
    best_val = -1.0
    best_centers = None
    
    # Fallback configuration: 5x5 grid + 1
    fallback = np.zeros((n, 2))
    for i in range(n):
        fallback[i, 0] = 0.1 + (i % 5) * 0.2
        fallback[i, 1] = 0.1 + (i // 5) * 0.2
    best_centers = fallback.copy()
    best_val = 0.1
    
    # Hexagonal multi-start optimization
    for seed in range(5):
        np.random.seed(42 + seed)
        centers = []
        # Row counts for 26 circles in a hexagonal-ish pattern
        row_counts = [5, 4, 5, 4, 5, 3]
        y = 0.12
        for i, cnt in enumerate(row_counts):
            x = 0.12 + (0.5 if i % 2 == 1 else 0.0) * 0.14
            for j in range(cnt):
                centers.append([x + j * 0.14, y])
            y += 0.14
            
        centers = np.array(centers)
        centers += np.random.uniform(-0.015, 0.015, size=centers.shape)
        
        try:
            res = minimize(
                objective, 
                centers.flatten(), 
                args=(n,), 
                method='Nelder-Mead',
                options={'maxiter': 8000, 'xatol': 1e-7, 'fatol': 1e-8}
            )
            if -res.fun > best_val:
                best_val = -res.fun
                best_centers = res.x.reshape(-1, 2)
        except Exception:
            continue
            
    radii = np.full(n, best_val)
    return best_centers, radii, sum(radii)
