def calculate_distances(centers):
    """Calculate pairwise distances between centers."""
    n = centers.shape[0]
    # Compute squared distances to avoid sqrt where possible, but need actual dist for constraints
    # Using broadcasting for efficiency
    # centers shape (n, 2)
    # diff shape (n, n, 2)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    return dists
