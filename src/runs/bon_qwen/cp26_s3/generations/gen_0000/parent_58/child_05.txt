import numpy as np
import scipy.optimize as opt
import math

def get_lp_matrix_and_rhs(centers, radii_bounds=None):
    """
    Constructs the matrix A and vector b for the LP:
    Maximize sum(r)
    Subject to A @ r <= b
    """
    n = centers.shape[0]
    num_pair_constraints = n * (n - 1) // 2
    num_boundary_constraints = 4 * n
    total_constraints = num_pair_constraints + num_boundary_constraints
    
    # A_ub has shape (total_constraints, n)
    # b_ub has shape (total_constraints,)
    
    # Initialize lists
    rows = []
    rhs = []
    
    # 1. Pairwise constraints: r_i + r_j <= dist(i, j)
    pair_idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            rows.append(row)
            rhs.append(dist)
            pair_idx += 1
            
    # 2. Boundary constraints: r_i <= x_i, r_i <= 1-x_i, etc.
    for i in range(n):
        x, y = centers[i]
        
        # r_i <= x
        row = np.zeros(n)
        row[i] = 1.0
        rows.append(row)
        rhs.append(x)
        
        # r_i <= 1 - x
        row = np.zeros(n)
        row[i] = 1.0
        rows.append(row)
        rhs.append(1.0 - x)
        
        # r_i <= y
        row = np.zeros(n)
        row[i] = 1.0
        rows.append(row)
        rhs.append(y)
        
        # r_i <= 1 - y
        row = np.zeros(n)
        row[i] = 1.0
        rows.append(row)
        rhs.append(1.0 - y)

    A_ub = np.array(rows)
    b_ub = np.array(rhs)
    
    return A_ub, b_ub

def solve_radii(centers):
    """
    Solves the LP to find maximum radii for given centers.
    Returns radii array and sum of radii.
    """
    n = centers.shape[0]
    A_ub, b_ub = get_lp_matrix_and_rhs(centers)
    
    # Objective: Maximize sum(r) => Minimize -sum(r)
    c = -np.ones(n)
    
    # Bounds: r_i >= 0
    bounds = [(0, None) for _ in range(n)]
    
    # Solve LP
    # Using 'highs' method if available (default in newer scipy), else 'simplex' or 'interior-point'
    try:
        res = opt.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    except:
        # Fallback for older scipy
        res = opt.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='interior-point')
        
    if res.success:
        radii = res.x
        # Clip tiny negative values due to numerical error
        radii = np.maximum(radii, 0.0)
        return radii, np.sum(radii)
    else:
        # Fallback if LP fails (should not happen usually)
        # Return small radii
        return np.full(n, 1e-6), 26 * 1e-6

def get_score(centers):
    """
    Wrapper to get the score (sum of radii) for a set of centers.
    """
    _, score = solve_radii(centers)
    return score
