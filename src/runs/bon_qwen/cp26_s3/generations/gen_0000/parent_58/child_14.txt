import numpy as np
import scipy.optimize as opt

def generate_initial_packing(n):
    """
    Generates a hexagonal lattice packing for n circles with a safe initial radius.
    """
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Start with a conservative radius that fits easily
    r_init = 0.08
    d = 2 * r_init
    h = np.sqrt(3)/2 * d
    
    count = 0
    row = 0
    
    while count < n:
        y = r_init + row * h
        if y + r_init > 1.0:
            break

            x_start = r_init
        else:
            x_start = r_init + d / 2

            centers[count] = [x, y]
            radii[count] = r_init
            count += 1
            x += d
        
        row += 1
    
    # If we didn't fill n circles (unlikely with r=0.08), fill randomly
    while count < n:
        centers[count] = [np.random.uniform(0.1, 0.9), np.random.uniform(0.1, 0.9)]
        radii[count] = 0.05
        count += 1
        
    return centers, radii

def objective(x_flat, n):
    """
    Objective function to minimize: -sum(radii)
    x_flat contains [x1, y1, r1, x2, y2, r2, ..., xn, yn, rn]
    """
    radii = x_flat[2::3]
    return -np.sum(radii)

def boundary_constraints(x_flat, n):
    """
    Returns list of constraints: x-r >= 0, x+r <= 1, y-r >= 0, y+r <= 1
    Formatted for scipy: c(x) >= 0
    """
    cons = []
    for i in range(n):
        x = x_flat[3*i]
        y = x_flat[3*i+1]
        r = x_flat[3*i+2]
        
        # x - r >= 0
        cons.append(x - r)
        # 1 - (x + r) >= 0  =>  x + r <= 1
        cons.append(1.0 - (x + r))
        # y - r >= 0
        cons.append(y - r)
        # 1 - (y + r) >= 0 => y + r <= 1
        cons.append(1.0 - (y + r))
        # r >= 0 (usually handled by bounds, but good to have)
        cons.append(r)
        
    return np.array(cons)

def non_overlap_constraints(x_flat, n):
    """
    Returns list of constraints: dist(i, j) - (r_i + r_j) >= 0
    """
    cons = []
    for i in range(n):
        xi, yi, ri = x_flat[3*i], x_flat[3*i+1], x_flat[3*i+2]
        for j in range(i + 1, n):
            xj, yj, rj = x_flat[3*j], x_flat[3*j+1], x_flat[3*j+2]
            dist = np.sqrt((xi - xj)**2 + (yi - yj)**2)
            cons.append(dist - (ri + rj))
    return np.array(cons)
