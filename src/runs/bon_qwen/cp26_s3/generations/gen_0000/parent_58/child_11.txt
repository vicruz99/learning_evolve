def compute_overlap_penalty(centers, radii, penalty_weight=1000.0):
    """Compute penalty for overlaps and boundary violations."""
    n = len(radii)
    penalty = 0.0

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            min_dist = radii[i] + radii[j]
            if dist < min_dist:
                overlap = min_dist - dist
                penalty += penalty_weight * overlap ** 2
    
    # Boundary penalties

        x, y = centers[i]
        r = radii[i]
        if x - r < 0:
            penalty += penalty_weight * (x - r) ** 2

def slsqp_optimize(centers, radii, n):

            cons.append(ctrs[i, 1] - rds[i] - 1e-11)
            cons.append(1.0 - ctrs[i, 1] - rds[i] - 1e-11)
        return np.array(cons)

        return np.array(cons)

    ]

    return None
