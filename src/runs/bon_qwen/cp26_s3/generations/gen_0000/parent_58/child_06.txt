import numpy as np
from scipy.optimize import minimize
import math

N_CIRCLES = 26

def objective(vars_arr):
    """Objective function: minimize negative sum of radii."""
    radii = vars_arr[2 * N_CIRCLES:]
    return -np.sum(radii)

def constraint_bounds(vars_arr):
    """Boundary constraints: circles must be inside [0,1]x[0,1]."""
    centers = vars_arr[:2 * N_CIRCLES].reshape(N_CIRCLES, 2)
    radii = vars_arr[2 * N_CIRCLES:]
    c = np.zeros(4 * N_CIRCLES)
    for i in range(N_CIRCLES):
        x, y = centers[i]
        r = radii[i]
        c[4 * i]     = x - r
        c[4 * i + 1] = 1.0 - x - r
        c[4 * i + 2] = y - r
        c[4 * i + 3] = 1.0 - y - r
    return c

def constraint_overlaps(vars_arr):
    """Overlap constraints: distance between centers >= sum of radii."""
    centers = vars_arr[:2 * N_CIRCLES].reshape(N_CIRCLES, 2)
    radii = vars_arr[2 * N_CIRCLES:]
    n_pairs = N_CIRCLES * (N_CIRCLES - 1) // 2
    c = np.zeros(n_pairs)
    idx = 0
    for i in range(N_CIRCLES):
        for j in range(i + 1, N_CIRCLES):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist_sq = dx * dx + dy * dy
            sum_r = radii[i] + radii[j]
            # Constraint: dist^2 - (r_i + r_j)^2 >= 0
            c[idx] = dist_sq - sum_r**2
            idx += 1
    return c

def generate_hex_grid_centers(n):
    """Generate initial centers using a hexagonal lattice pattern."""
    centers = []
    # Approximate grid dimensions
    cols = math.ceil(math.sqrt(n))
    rows = math.ceil(n / cols)
    spacing = 0.12
    for r in range(rows):
        for c in range(cols):
            if len(centers) >= n:
                break
            x = 0.15 + c * spacing * 1.0
            y = 0.15 + r * spacing * math.sqrt(3) / 2
            if c % 2 == 1:
                y += spacing * math.sqrt(3) / 4
            centers.append([x, y])
        if len(centers) >= n:
            break
    return np.array(centers[:n])

def solve_packing(seed_val):
    """Run the optimization for a given random seed."""
    rng = np.random.default_rng(seed_val)
    
    # Initialize centers
    centers = generate_hex_grid_centers(N_CIRCLES)
    # Add small random perturbation to break symmetry
    centers += rng.uniform(-0.01, 0.01, size=centers.shape)
    # Clip to valid range
    centers = np.clip(centers, 0.05, 0.95)
    
    # Initial radii
    init_radii = np.full(N_CIRCLES, 0.05)
    
    # Flatten variables
    x0 = np.concatenate([centers.ravel(), init_radii])
    
    # Bounds: centers in [0, 1], radii in [0, 0.5]
    bounds = [(0.0, 1.0)] * (2 * N_CIRCLES) + [(0.0, 0.5)] * N_CIRCLES
    
    # Constraints setup
    cons = [
        {'type': 'ineq', 'fun': constraint_bounds},
        {'type': 'ineq', 'fun': constraint_overlaps}
    ]
    
    # Run SLSQP optimizer
    try:
        res = minimize(
            objective,
            x0,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 1000, 'ftol': 1e-9, 'disp': False}
        )
        return res
    except Exception:
        return None

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Main function to pack circles and return the best result."""
    best_result = None
    best_sum_r = -1.0
    
    # Run optimization multiple times with different seeds
    for seed in [42, 123, 456]:
        res = solve_packing(seed)
        if res is not None and res.success:
            centers = res.x[:2 * N_CIRCLES].reshape(N_CIRCLES, 2)
            radii = res.x[2 * N_CIRCLES:]
            
            # Post-process: ensure non-negative radii and valid bounds
            radii = np.maximum(radii, 1e-9)
            centers = np.clip(centers, radii[:, np.newaxis], 1.0 - radii[:, np.newaxis])
            
            sum_r = np.sum(radii)
            if sum_r > best_sum_r:
                best_sum_r = sum_r
                best_result = (centers, radii, sum_r)
                
    # Fallback if optimization fails (should not happen with this setup)
    if best_result is None:
        # Simple 5x5 grid + 1 fallback
        centers = np.zeros((26, 2))
        radii = np.full(26, 0.045)
        idx = 0
        for i in range(5):

                centers[idx, 0] = 0.1 + i * 0.2

                idx += 1
        centers[25, 0] = 0.5

        best_result = (centers, radii, np.sum(radii))
        
    return best_result
