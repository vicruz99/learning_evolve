import numpy as np

def compute_max_radius(centers):
    """Compute the maximum valid radius for a given set of centers."""
    n = centers.shape[0]
    max_r = 1.0
    # Check boundaries
    for i in range(n):
        max_r = min(max_r, centers[i, 0], 1.0 - centers[i, 0], 
                    centers[i, 1], 1.0 - centers[i, 1])
    # Check pairwise distances
    for i in range(n):
        for j in range(i + 1, n):
            diff = centers[i] - centers[j]
            dist = np.sqrt(np.dot(diff, diff))
            max_r = min(max_r, dist / 2.0)
    return max_r

def optimize_once(n, seed):

    np.random.seed(seed)

    centers = 0.1 + 0.8 * np.random.rand(n, 2)
    
    r = 0.06
    step_size = 0.015
    best_r = r
    
    for step in range(3000):
        forces = np.zeros_like(centers)
        is_valid = True
        
        for i in range(n):
            # Boundary repulsion
            for d in range(2):
                if centers[i, d] < r:
                    forces[i, d] += (r - centers[i, d]) * 20.0

                    forces[i, d] -= (centers[i, d] - (1.0 - r)) * 20.0

                diff = centers[i] - centers[j]

                if dist < 2.0 * r:

                    force_mag = overlap / dist

        centers += forces * step_size

        if is_valid:
            r = min(r + 8e-5, 0.15)
        else:
            r = max(r - 1e-5, 0.01)

    return centers, best_r

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Main function to pack 26 circles and return the best configuration."""
    n = 26
    best_sum = -1.0
    best_centers = None

            best_sum = current_sum
            best_centers = centers.copy()
            best_radii = np.full(n, actual_r)
            
    return best_centers, best_radii, best_sum
