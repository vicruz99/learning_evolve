import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def objective_sum_radii(p):
    """Negative sum of radii for minimization."""
    return -np.sum(p[2::3])

def constraints_function(p):
    """All constraints as a single array."""
    n = N_CIRCLES
    c = []
    
    for i in range(n):
        c.append(p[3 * i] - p[3 * i + 2])
        c.append(1.0 - p[3 * i] - p[3 * i + 2])
        c.append(p[3 * i + 1] - p[3 * i + 2])
        c.append(1.0 - p[3 * i + 1] - p[3 * i + 2])
    
    for i in range(n):
        for j in range(i + 1, n):
            dx = p[3 * i] - p[3 * j]
            dy = p[3 * i + 1] - p[3 * j + 1]
            dist = np.sqrt(dx * dx + dy * dy)
            c.append(dist - p[3 * i + 2] - p[3 * j + 2])
    
    return np.array(c)

def make_bounds():
    """Create bounds for all variables."""
    n = N_CIRCLES
    bounds = []
    for k in range(3 * n):
        if k % 3 != 2:
            bounds.append((0.0, 1.0))
        else:
            bounds.append((1e-8, 0.5))
    return bounds

def hexagonal_initialization(radius, n_target):
    """Create hexagonal grid of circles."""
    centers = np.zeros((n_target, 2))
    radii = np.full(n_target, radius)
    
    count = 0
    row = 0
    row_spacing = radius * np.sqrt(3) * 1.0
    col_spacing = 2 * radius * 1.0
    
    while count < n_target:
        y = radius + row * row_spacing
        if y + radius > 1.0:
            break
        
        offset = (row % 2) * radius
        x_start = radius + offset
        col = 0
        while count < n_target:
            x = x_start + col * col_spacing
            if x + radius > 1.0:
                break
            centers[count] = [x, y]
            radii[count] = radius
            count += 1
            col += 1
        row += 1
    
    return centers, radii

def force_relax(centers, radii, iterations=5000):
    """Force-based relaxation to resolve overlaps."""
    n = N_CIRCLES
    centers = centers.copy()
    radii = radii.copy()
    
    dt = 0.001
    stiffness = 500.0
    
    for _ in range(iterations):
        forces = np.zeros((n, 2))
        
        for i in range(n):
            for j in range(i + 1, n):
                diff = centers[i] - centers[j]
                dist = np.sqrt(np.sum(diff ** 2))
                min_dist = radii[i] + radii[j]
                
                if dist < min_dist * 1.2 and dist > 1e-12:
                    overlap = max(0, min_dist - dist)
                    force_mag = overlap * stiffness
                    force = force_mag * diff / dist
                    forces[i] += force
                    forces[j] -= force
        
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            
            if x - r < 0:
                forces[i, 0] += (r - x) * stiffness
            if x + r > 1:
                forces[i, 0] -= (x + r - 1) * stiffness
            if y - r < 0:
                forces[i, 1] += (r - y) * stiffness
            if y + r > 1:
                forces[i, 1] -= (y + r - 1) * stiffness
        
        centers += dt * forces
    
    return centers, radii

def optimize_equal_radii(centers_init):
    """Optimize positions for equal circles to maximize minimum distance."""
    n = N_CIRCLES
    
    def neg_min_dist(params):
        c = params.reshape(n, 2)
        min_d = float('inf')
        for i in range(n):
            for j in range(i + 1, n):
                d = np.sqrt(np.sum((c[i] - c[j]) ** 2))
                if d < min_d:
                    min_d = d
        return -min_d
    
    def boundary_constraints(params):
        c = params.reshape(n, 2)
        cons = []
        for i in range(n):
            cons.append(c[i, 0])
            cons.append(1.0 - c[i, 0])
            cons.append(c[i, 1])
            cons.append(1.0 - c[i, 1])
        return np.array(cons)
    
    bounds = [(0, 1) for _ in range(2 * n)]
    cons = {'type': 'ineq', 'fun': boundary_constraints}
    
    result = minimize(neg_min_dist, centers_init.flatten(), 
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': 3000, 'ftol': 1e-14})
    
    return result.x.reshape(n, 2), -result.fun

def run_packing():
    n = N_CIRCLES
    best_sum = 0
    best_centers = None
    best_radii = None
    
    # Try multiple initial configurations
    initial_radii = [0.09, 0.095, 0.10, 0.085]
    
    for init_r in initial_radii:
        centers, radii = hexagonal_initialization(init_r, n)
        
        if len(centers) < n:
            centers = np.zeros((n, 2))
            radii = np.full(n, init_r)
            idx = 0

                if y + init_r > 1.0:
                    break
                offset = (row % 2) * init_r
                for col in range(8):
                    if idx >= n:
                        break
                    x = init_r + offset + col * 2 * init_r
                    if x + init_r > 1.0:
                        break
                    centers[idx] = [x, y]
                    idx += 1
            for i in range(idx, n):
                centers[i] = [0.5, 0.5]
        
        # Force relaxation
        centers, radii = force_relax(centers, radii, iterations=3000)
        
        # Optimize equal radii positions
        centers_eq, min_dist = optimize_equal_radii(centers)
        equal_r = min_dist / 2.0
        
        # Build parameters for full optimization
        params = np.zeros(3 * n)
        for i in range(n):
            params[3 * i] = centers_eq[i, 0]
            params[3 * i + 1] = centers_eq[i, 1]
            params[3 * i + 2] = equal_r * 0.95
        
        # Full SLSQP optimization with variable radii
        bnds = make_bounds()
        cons = {'type': 'ineq', 'fun': constraints_function}
        
        result = minimize(objective_sum_radii, params, method='SLSQP',
                         bounds=bnds, constraints=cons,
                         options={'maxiter': 8000, 'ftol': 1e-15})
        
        centers_opt = result.x[:2 * n].reshape(n, 2)
        radii_opt = result.x[2 * n:]
        current_sum = np.sum(radii_opt)
        
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers_opt.copy()
            best_radii = radii_opt.copy()
    
    # Second pass: refine with small perturbations
    for trial in range(5):
        perturbed = best_centers.copy()
        np.random.seed(42 + trial)
        perturbation = np.random.randn(n, 2) * 0.01
        perturbed = perturbed + perturbation
        perturbed = np.clip(perturbed, best_radii.reshape(-1, 1), 1.0 - best_radii.reshape(-1, 1))
        
        centers_refined, radii_refined = force_relax(perturbed, best_radii, iterations=2000)
        
        params = np.zeros(3 * n)
        for i in range(n):
            params[3 * i] = centers_refined[i, 0]
            params[3 * i + 1] = centers_refined[i, 1]
            params[3 * i + 2] = radii_refined[i]
        
        result = minimize(objective_sum_radii, params, method='SLSQP',
                         bounds=make_bounds(), constraints=cons,
                         options={'maxiter': 5000, 'ftol': 1e-15})
        
        centers_opt = result.x[:2 * n].reshape(n, 2)
        radii_opt = result.x[2 * n:]
        current_sum = np.sum(radii_opt)
        
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers_opt.copy()
            best_radii = radii_opt.copy()
    
    return best_centers, best_radii, best_sum
