import numpy as np
from scipy.optimize import minimize

N = 26
IDX_X = np.arange(0, 3*N, 3)
IDX_Y = np.arange(1, 3*N, 3)
IDX_R = np.arange(2, 3*N, 3)

def objective(v):
    return -np.sum(v[IDX_R])

def constraints(v):
    x = v[IDX_X]
    y = v[IDX_Y]
    r = v[IDX_R]
    
    # Boundary constraints: x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
    c = np.concatenate([x - r, 1 - x - r, y - r, 1 - y - r])
    
    # Overlap constraints: dist(i,j) - r_i - r_j >= 0
    ov = np.empty(N*(N-1)//2)
    k = 0
    for i in range(N):
        dx = x[i] - x[i+1:]
        dy = y[i] - y[i+1:]
        ov[k:] = np.sqrt(dx**2 + dy**2 + 1e-14) - r[i] - r[i+1:]
        k += len(dx)
        
    return np.concatenate([c, ov])

def check_packing_validity(centers, radii):
    tol = 1e-9
    n = len(radii)
    for i in range(n):
        if radii[i] < -tol:
            return False
        x, y = centers[i]
        r = radii[i]
        if x < r - tol or x > 1 - r + tol:
            return False
        if y < r - tol or y > 1 - r + tol:
            return False
    for i in range(n):
        for j in range(i+1, n):
            d = np.sqrt(np.sum((centers[i] - centers[j])**2))
            if d < radii[i] + radii[j] - tol:
                return False
    return True

def run_packing():
    np.random.seed(42)
    bounds = [(0, 1)]*(2*N) + [(0, 0.5)]*N
    cons_dict = {'type': 'ineq', 'fun': constraints}
    
    best_val = -np.inf
    best_v = None
    
    v0_list = []
    
    # 1. Structured hexagonal lattice guess
    v0 = np.zeros(3*N)
    s = 0.25
    for i in range(N):
        row, col = divmod(i, 6)
        v0[IDX_X[i]] = 0.2 + col*s + (row%2)*s/2
        v0[IDX_Y[i]] = 0.2 + row*s*np.sqrt(3)/2
        v0[IDX_R[i]] = 0.08
    v0_list.append(v0)
    
    # 2. Random initial guesses
    for _ in range(6):
        v_rand = np.random.rand(3*N)
        v_rand[IDX_R] *= 0.12
        v0_list.append(v_rand)
        
    for v_start in v0_list:
        # Perturb to avoid exact local minima
        v_start = v_start + np.random.randn(3*N) * 0.03
        v_start = np.clip(v_start, 0.02, 0.98)
        v_start[IDX_R] = np.clip(v_start[IDX_R], 0.02, 0.4)
        
        res = minimize(objective, v_start, method='SLSQP', bounds=bounds, constraints=cons_dict,
                       options={'maxiter': 500, 'ftol': 1e-10, 'disp': False})
        if -res.fun > best_val:
            best_val = -res.fun
            best_v = res.x
            
    centers = np.column_stack((best_v[IDX_X], best_v[IDX_Y]))
    radii = best_v[IDX_R].copy()
    
    # Ensure strict validity after optimization (handle numerical drift)
    if not check_packing_validity(centers, radii):
        for _ in range(200):
            radii *= 0.995
            if check_packing_validity(centers, radii):
                break
                
    # Aggressively scale up radii to fill remaining slack
    for _ in range(300):
        r_trial = radii * 1.002
        if check_packing_validity(centers, r_trial):
            radii = r_trial
        else:
            break
            
    return centers, radii, np.sum(radii)
