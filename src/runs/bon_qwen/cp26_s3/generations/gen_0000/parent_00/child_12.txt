import numpy as np

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square

    Args:
        centers: np.array of shape (n, 2) with (x, y) coordinates
        radii: np.array of shape (n) with radius of each circle

    Returns:
        True if valid, False otherwise
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True

def get_initial_hex_centers(n):

def objective_function(vars, n):
    """
    Objective function to minimize: -sum(radii) + penalty for violations.
    vars = [x1, y1, x2, y2, ..., r1, r2, ...]
    """
    centers = vars[:2*n].reshape(n, 2)
    radii = vars[2*n:]
    
    # Soft boundary constraints
    # radii must be <= x, <= 1-x, <= y, <= 1-y
    # So x >= r, 1-x >= r => r <= min(x, 1-x)
    # We want to maximize sum(r), so we penalize if r > boundary
    # Or better, just enforce r = min(distances) implicitly? 
    # Here we treat r as variable but penalize if it violates geometry.
    
    penalty = 0.0
    lambda_pen = 1000.0
    
    # Boundary penalties
    # For each circle, it must be inside [0,1]x[0,1]
    # r <= x, r <= 1-x, r <= y, r <= 1-y
    # Violation: max(0, r - x, r - (1-x), r - y, r - (1-y))
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        v1 = r - x
        v2 = r - (1 - x)
        v3 = r - y
        v4 = r - (1 - y)
        violations = [v for v in [v1, v2, v3, v4] if v > 0]
        if violations:
            penalty += lambda_pen * sum([v**2 for v in violations])
            
        # Also ensure r >= 0
        if r < 0:
            penalty += lambda_pen * r**2

    # Overlap penalties
    # dist(i, j) >= r_i + r_j
    # Violation: r_i + r_j - dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            v = radii[i] + radii[j] - dist
            if v > 0:
                penalty += lambda_pen * v**2
                
    return -np.sum(radii) + penalty
