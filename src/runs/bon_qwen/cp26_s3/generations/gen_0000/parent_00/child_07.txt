import numpy as np
from scipy.optimize import minimize

def init_positions(n, grid_size=80):
    """Generates n positions in [0,1]^2 maximizing minimum distance between points."""
    grid = np.mgrid[:grid_size, :grid_size].T.reshape(-1, 2) / (grid_size - 1)
    pts = [np.array([0.5, 0.5])]
    available = grid.copy()
    for _ in range(n - 1):
        dists = np.min(np.linalg.norm(available[:, np.newaxis, :] - pts[np.newaxis, :, :], axis=2), axis=1)
        best_idx = np.argmax(dists)
        pts.append(available[best_idx])
        available = np.delete(available, best_idx, axis=0)
    return np.array(pts)

def compute_constraints(v, n, iu, ju):
    """Computes constraint values for circle packing problem."""
    r = v[-1]
    p = v[:-1].reshape(-1, 2)
    
    # Boundary constraints: p - r >= 0 and 1 - p - r >= 0
    c = np.concatenate([
        p[:, 0] - r,
        1.0 - p[:, 0] - r,
        p[:, 1] - r,
        1.0 - p[:, 1] - r
    ])
    
    # Pairwise non-overlap constraints: dist >= 2r
    diff = p[:, np.newaxis, :] - p[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    c = np.concatenate([c, dists[iu, ju] - 2.0 * r])
    return c

def run_packing():
    n = 26
    iu, ju = np.triu_indices(n, k=1)
    
    # We define the constraint function that minimize will call
    def constr_fun(v):
        return compute_constraints(v, n, iu, ju)

    def obj(v):
        return -v[-1]  # Maximize r
        
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)]
    cons = {'type': 'ineq', 'fun': constr_fun}
    
    best_res = None
    best_val = -np.inf
    
    # Run optimization with multiple initial seeds to avoid local minima
    for seed in range(5):
        np.random.seed(seed)
        pos0 = init_positions(n, grid_size=60)
        # Scale positions to be safely inside the unit square
        pos0 = pos0 * 0.7 + 0.15
        r0 = 0.005  # Small initial radius to ensure feasibility
        x0 = np.concatenate([pos0.flatten(), [r0]])
        
        try:
            res = minimize(obj, x0, bounds=bounds, constraints=cons, method='SLSQP', 
                           options={'maxiter': 5000, 'ftol': 1e-12})
            if res.success and -res.fun > best_val:
                best_val = -res.fun
                best_res = res
        except Exception:
            continue
            
    if best_res is None:
        # Fallback configuration
        pos0 = init_positions(n, grid_size=60)
        pos0 = pos0 * 0.7 + 0.15
        r0 = 0.005
        return pos0, np.full(n, r0), n * r0

    final_r = best_res.x[-1]
    final_pos = best_res.x[:-1].reshape(-1, 2)
    
    # Ensure strict feasibility for the validator
    final_pos = np.clip(final_pos, final_r + 1e-9, 1.0 - final_r - 1e-9)
    
    radii = np.full(n, final_r)
    return final_pos, radii, np.sum(radii)
