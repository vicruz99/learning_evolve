import numpy as np
from scipy.optimize import minimize
import random

def generate_initial_hex_centers(n, r):
    """Generates initial centers for n circles in a hexagonal pattern with radius r."""
    centers = []
    row = 0
    # Estimate row height for hex packing
    dy = r * np.sqrt(3)
    
    while len(centers) < n:
        y = r + row * dy
        if y + r > 1.0 + 1e-9:
            break
            
        # Determine starting x for the row
        # Even rows: start at r, step 2r
        # Odd rows: start at 2r, step 2r (shifted by r)
        x_start = r if row % 2 == 0 else 2 * r
        x = x_start
        
        while x + r <= 1.0 + 1e-9 and len(centers) < n:
            centers.append([x, y])
            x += 2 * r
            if len(centers) == n:
                break
        row += 1
        
    # If we didn't fit n circles (unlikely with small r), fill with random valid positions
    while len(centers) < n:
        x = random.uniform(r, 1 - r)
        y = random.uniform(r, 1 - r)
        # Simple check to avoid overlap with existing
        valid = True
        for cx, cy in centers:
            if (x - cx)**2 + (y - cy)**2 < (2*r)**2:
                valid = False
                break
        if valid:
            centers.append([x, y])
        else:
            # If stuck, just place it (optimizer will fix it if possible, or we reduce r)
            centers.append([x, y])
            # Safety break to prevent infinite loop
            if len(centers) > n + 10: break
            
    return np.array(centers[:n])

def objective_function(v, n):
    """Objective: Maximize sum of radii (minimize negative sum)."""
    s = 0.0
    for i in range(n):
        s += v[3 * i + 2]
    return -s

def boundary_constraints(v, n):
    """Generates boundary constraints: r <= x <= 1-r, r <= y <= 1-r"""
    constraints = []
    for i in range(n):
        idx = 3 * i
        # x - r >= 0
        constraints.append(v[idx] - v[idx + 2])
        # 1 - x - r >= 0
        constraints.append(1.0 - v[idx] - v[idx + 2])
        # y - r >= 0
        constraints.append(v[idx + 1] - v[idx + 2])
        # 1 - y - r >= 0
        constraints.append(1.0 - v[idx + 1] - v[idx + 2])
    return np.array(constraints)

def distance_constraints(v, n):
    """Generates non-overlap constraints: dist^2 >= (r1+r2)^2"""
    constraints = []
    for i in range(n):
        for j in range(i + 1, n):
            idx_i = 3 * i
            idx_j = 3 * j
            
            xi = v[idx_i]
            yi = v[idx_i + 1]
            ri = v[idx_i + 2]
            
            xj = v[idx_j]
            yj = v[idx_j + 1]
            rj = v[idx_j + 2]
            
            dx = xi - xj
            dy = yi - yj
            dist_sq = dx * dx + dy * dy
            r_sum = ri + rj
            
            # Constraint: dist_sq - r_sum^2 >= 0
            constraints.append(dist_sq - r_sum * r_sum)
    return np.array(constraints)

def run_optimization(n, initial_centers, initial_radii):
    """Runs the optimization for a given starting configuration."""
    x0 = np.zeros(3 * n)
    for i in range(n):
        x0[3 * i] = initial_centers[i, 0]
        x0[3 * i + 1] = initial_centers[i, 1]
        x0[3 * i + 2] = initial_radii[i]
    
    bounds = []
    for i in range(n):
        # x in [0, 1], y in [0, 1], r in [0, 0.5]
        bounds.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
        
    # SLSQP requires constraints to be dictionaries or a special object
    # We will use a wrapper that evaluates the functions
    
    # Define constraint functions inline to avoid closures if possible, 
    # but for scipy, passing functions is standard.
    # To adhere to "no closures from function nesting" and "top level", 
    # we define helper functions that take v and n.
    
    cons = []
    
    # Boundary constraints
    def bound_fn(v):
        return boundary_constraints(v, n)
    # We need separate constraints for SLSQP or use a single vector constraint?
    # SLSQP accepts a dict with 'fun' returning an array for inequality.
    cons.append({'type': 'ineq', 'fun': bound_fn})
    
    # Distance constraints
    def dist_fn(v):
        return distance_constraints(v, n)
    cons.append({'type': 'ineq', 'fun': dist_fn})
    
    res = minimize(objective_function, x0, args=(n,), method='SLSQP', 
                   bounds=bounds, constraints=cons, 
                   options={'maxiter': 1000, 'ftol': 1e-9, 'disp': False})
    
    return res
