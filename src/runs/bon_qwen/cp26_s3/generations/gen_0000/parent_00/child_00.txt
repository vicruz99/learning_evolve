def run_packing():

    Finds a packing of 26 circles in a unit square maximizing the sum of radii.
    """
    n = 26
    best_sum_radii = 0.0
    best_centers = None

        """Calculate max radii for given centers."""
        radii = np.zeros(n)
        for i in range(n):
            # Boundary constraints
            r_bound = min(centers[i, 0], 1 - centers[i, 0], 
                          centers[i, 1], 1 - centers[i, 1])
            
            # Neighbor constraints
            dists = np.sqrt(np.sum((centers - centers[i]) ** 2, axis=1))
            dists[i] = np.inf # Ignore self-distance
            r_neighbor = 0.5 * np.min(dists)
            
            radii[i] = min(r_bound, r_neighbor)
        return radii

    def objective(x):
        """Objective function to minimize: -sum(radii)."""
        centers = x.reshape(n, 2)
        radii = calculate_radii(centers)
        return -np.sum(radii)
