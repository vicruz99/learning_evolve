import numpy as np

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Stage 1: Hexagonal Initialization
    # Arrange in rows to maximize density. 
    # Pattern: 6, 5, 6, 5, 4 circles per row (total 26)
    row_counts = [6, 5, 6, 5, 4]
    r_init = 0.095  # Starting radius slightly smaller than theoretical max
    
    # Vertical spacing for hexagonal packing: sqrt(3)/2 * diameter = sqrt(3) * r
    v_step = np.sqrt(3) * r_init
    total_height = 2 * r_init + (len(row_counts) - 1) * v_step
    
    # Calculate Y positions centered vertically in the unit square
    y_positions = np.linspace(0.5 - total_height/2, 0.5 + total_height/2, len(row_counts))
    
    idx = 0
    for row_i, count in enumerate(row_counts):
        y = y_positions[row_i]
        # Horizontal span of 'count' circles: (count-1)*2r + 2r = count*2r? 
        # Actually, extent is 2*r (left) to 2*r*count (right) relative to first center?
        # Bounding box width needed: (count - 1) * 2 * r + 2 * r = count * 2 * r
        # But for hexagonal, rows are offset. 
        # Width occupied is (count-1)*2r + 2r.
        width = count * 2 * r_init
        
        # Center the row horizontally
        x_start = 0.5 - width / 2 + r_init

        if row_i % 2 == 1:
            x_start += r_init

            x = x_start + c * 2 * r_init
            centers[idx] = [x, y]
            radii[idx] = r_init
            idx += 1

    # We will push circles apart to maximize their size
    
    current_time = 0.0
    max_time = 100.0 # Iteration count for "time"

    temperature = 0.05
    cooling_rate = 0.995
    
    for t in range(2000):
        forces = np.zeros((n, 2))
        
        for i in range(n):
            # 1. Boundary Forces (Push inward if touching walls)
            xi, yi = centers[i]
            ri = radii[i]
            
            # Left wall
            if xi - ri < 0:
                forces[i, 0] += 0.1
            # Right wall
            elif xi + ri > 1:
                forces[i, 0] -= 0.1
            # Bottom wall
            if yi - ri < 0:
                forces[i, 1] += 0.1
            # Top wall
            elif yi + ri > 1:
                forces[i, 1] -= 0.1

            # 2. Inter-circle Repulsion (Push apart if overlapping)
            for j in range(i + 1, n):
                dx = centers[j, 0] - centers[i, 0]
                dy = centers[j, 1] - centers[i, 1]
                dist = np.sqrt(dx*dx + dy*dy)

                    dist = 1e-9
                    dx, dy = 1.0, 0.0 # Default direction if coincident

                req_dist = radii[i] + radii[j]
                
                # Penalty/Force proportional to overlap squared (stiff spring)
                overlap = req_dist - dist
                
                if overlap > 0:
                    force_mag = 0.5 * overlap # Repulsive force
                    fx = (dx / dist) * force_mag
                    fy = (dy / dist) * force_mag
                    
                    forces[i, 0] -= fx
                    forces[i, 1] -= fy
                    forces[j, 0] += fx
                    forces[j, 1] += fy

        # 3. Update Centers with Temperature (Perturbation)
        # Move centers in direction of forces
        for i in range(n):
            centers[i, 0] += forces[i, 0]
            centers[i, 1] += forces[i, 1]

            centers[i, 0] += np.random.normal(0, temperature)
            centers[i, 1] += np.random.normal(0, temperature)
            
            # Hard clamp to keep centers within [0, 1]
            centers[i, 0] = np.clip(centers[i, 0], 0, 1)
            centers[i, 1] = np.clip(centers[i, 1], 0, 1)

        # 4. Expand Radii
        # Slowly grow all radii to fill the available space
        # Growth rate decreases over time
        growth = 0.0005 * (1.0 + 0.1 * np.sin(current_time)) # Oscillating growth helps
        radii += growth
        current_time += 0.1
        
        # 5. Cool down temperature
        temperature *= cooling_rate
        
        # 6. Check for "stuck" state: if radius is too small relative to bounds,

    # Final cleanup: Ensure strict validity by clipping radii to boundaries
    # If a circle is too close to a wall, reduce its radius slightly to be safe

        # Radius cannot exceed distance to any wall

        if radii[i] > max_r + 1e-6:
