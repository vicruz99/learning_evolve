def generate_hex_grid(n, scale=0.95, shift=(0.025, 0.025)):
    """
    Generates centers for a hexagonal grid pattern to initialize the packing.
    """
    # Approximate number of rows/cols needed
    # Area per circle in hex packing ~ 2*sqrt(3)*r^2
    # We want to fit n circles.
    # Let's just generate a dense grid and pick n points.
    
    # Create a grid of points
    points = []
    # Hexagonal spacing logic
    # Let's assume a target radius r ~ 0.1 for spacing
    spacing_x = 0.15 
    spacing_y = spacing_x * np.sqrt(3) / 2
    
    y = 0.0
    row = 0
    while y <= 1.0:
        x = 0.0 if row % 2 == 0 else spacing_x / 2
        while x <= 1.0:
            points.append([x, y])
            x += spacing_x
        y += spacing_y
        row += 1
        
    points = np.array(points)
    
    # Filter points strictly inside [0, 1]x[0, 1]
    mask = (points[:, 0] >= 0) & (points[:, 0] <= 1) & \
           (points[:, 1] >= 0) & (points[:, 1] <= 1)
    points = points[mask]
    
    # If we have more than n, select a subset (e.g., random or first n)
    # To ensure good distribution, maybe shuffle
    if len(points) > n:
        np.random.shuffle(points)
        points = points[:n]

        # Fill remaining with random points
        missing = n - len(points)
        extra = np.random.rand(missing, 2)
        points = np.vstack([points, extra])
        
    return points
