import numpy as np
from scipy.optimize import differential_evolution, minimize

def compute_penalty_and_objective(x, n_circles, penalty_weight):
    centers = x[:2 * n_circles].reshape(n_circles, 2)
    radii = x[2 * n_circles:]
    
    obj = -np.sum(radii)
    
    pen = 0.0
    
    for i in range(n_circles):
        x_i = centers[i, 0]
        y_i = centers[i, 1]
        r_i = radii[i]
        
        pen += max(0, r_i - x_i) ** 2
        pen += max(0, r_i - (1 - x_i)) ** 2
        pen += max(0, r_i - y_i) ** 2
        pen += max(0, r_i - (1 - y_i)) ** 2
    
    for i in range(n_circles):
        for j in range(i + 1, n_circles):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            overlap = radii[i] + radii[j] - dist
            if overlap > 0:
                pen += overlap ** 2
    
    return obj + penalty_weight * pen

def create_hexagonal_initial(n_circles):
    rows = 6
    cols = 5
    points = []
    
    for row in range(rows):
        for col in range(cols):
            if len(points) >= n_circles:
                break
            x = 0.12 + col * 0.165 + (row % 2) * 0.0825
            y = 0.1 + row * 0.14
            points.append([x, y])
        if len(points) >= n_circles:
            break
    
    centers = np.array(points[:n_circles])
    radii = np.full(n_circles, 0.035)
    return centers, radii

def adjust_for_feasibility(centers, radii, n_circles):
    for i in range(n_circles):
        x_i = centers[i, 0]
        y_i = centers[i, 1]
        r_i = radii[i]
        
        max_r_boundary = min(x_i, 1 - x_i, y_i, 1 - y_i)
        radii[i] = min(r_i, max_r_boundary - 1e-11)
    
    for iteration in range(200):
        max_violation = 0.0
        viol_i = -1
        viol_j = -1
        
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                violation = radii[i] + radii[j] - dist
                if violation > max_violation:
                    max_violation = violation
                    viol_i = i
                    viol_j = j
        
        if max_violation < 1e-12:
            break
        
        if viol_i >= 0 and viol_j >= 0:
            reduction = max_violation / 2 + 1e-12
            radii[viol_i] -= reduction
            radii[viol_j] -= reduction
    
    radii = np.maximum(radii, 0.0)
    return centers, radii

        updating='deferred'
    )
    
    centers = result.x[:2 * n_circles].reshape(n_circles, 2)
