import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def objective(vars):
    """Objective: minimize negative sum of radii"""
    return -np.sum(vars[2*N_CIRCLES:])

def constr_radii(vars):
    """Radii must be non-negative"""
    return vars[2*N_CIRCLES:]

def constr_x_left(vars):
    """Circles must not cross left boundary"""
    return vars[:N_CIRCLES] - vars[2*N_CIRCLES:]

def constr_x_right(vars):
    """Circles must not cross right boundary"""
    return 1.0 - vars[:N_CIRCLES] - vars[2*N_CIRCLES:]

def constr_y_bottom(vars):
    """Circles must not cross bottom boundary"""
    return vars[N_CIRCLES:2*N_CIRCLES] - vars[2*N_CIRCLES:]

def constr_y_top(vars):
    """Circles must not cross top boundary"""
    return 1.0 - vars[N_CIRCLES:2*N_CIRCLES] - vars[2*N_CIRCLES:]

def constr_overlap(vars):
    """Pairwise non-overlap constraints: dist^2 >= (r_i + r_j)^2"""
    xc = vars[:N_CIRCLES]
    yc = vars[N_CIRCLES:2*N_CIRCLES]
    rc = vars[2*N_CIRCLES:]
    vals = []
    for i in range(N_CIRCLES):
        for j in range(i + 1, N_CIRCLES):
            d2 = (xc[i] - xc[j])**2 + (yc[i] - yc[j])**2
            vals.append(d2 - (rc[i] + rc[j])**2)
    return np.array(vals)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initial feasible configuration: structured grid
    # 6 columns, 5 rows layout, taking first 26 positions
    x_init = np.repeat(np.linspace(0.1, 0.9, 6), 5)[:N_CIRCLES]

    r_init = 0.04 * np.ones(N_CIRCLES)
    
    # Concatenate variables: [x_1..x_n, y_1..y_n, r_1..r_n]
    x0 = np.concatenate([x_init, y_init, r_init])
    
    # Define constraints dictionary
    cons = [
        {'type': 'ineq', 'fun': constr_radii},
        {'type': 'ineq', 'fun': constr_x_left},
        {'type': 'ineq', 'fun': constr_x_right},
        {'type': 'ineq', 'fun': constr_y_bottom},
        {'type': 'ineq', 'fun': constr_y_top},
        {'type': 'ineq', 'fun': constr_overlap}

    bounds = [(0.0, 1.0)] * (2 * N_CIRCLES) + [(0.0, 0.5)] * N_CIRCLES

    res = minimize(
        objective, 
        x0, 
        method='SLSQP', 
        bounds=bounds, 
        constraints=cons,
        options={'maxiter': 5000, 'ftol': 1e-10}
    )
    
    # Extract results
    if res.success:
        xc = res.x[:N_CIRCLES]
        yc = res.x[N_CIRCLES:2*N_CIRCLES]
        rc = res.x[2*N_CIRCLES:]
    else:
        # Fallback to initial if optimization fails
        xc = x0[:N_CIRCLES]
        yc = x0[N_CIRCLES:2*N_CIRCLES]
        rc = x0[2*N_CIRCLES:]

    rc -= 1e-9
    rc = np.maximum(rc, 0.0)

        # Enforce boundary constraints strictly

    return centers, rc, float(np.sum(rc))
