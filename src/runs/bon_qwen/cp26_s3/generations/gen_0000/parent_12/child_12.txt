def get_optimal_radii(centers):
    """
    Given fixed centers, solve the LP to find radii that maximize sum(r).
    Constraints:
    1. r_i + r_j <= dist(centers[i], centers[j]) for all i < j
    2. r_i <= min(x_i, 1-x_i, y_i, 1-y_i) for all i (boundary constraints)
    3. r_i >= 0
    """
    n = len(centers)
    
    # Objective: maximize sum(r) => minimize -sum(r)
    c_obj = np.ones(n) * -1.0
    
    # Inequality constraints: A_ub @ r <= b_ub
    A_ub = []
    b_ub = []
    
    # Pairwise distance constraints: r_i + r_j <= dist_ij
    # Precompute distances
    dists = np.zeros((n, n))
    for i in range(n):
        for j in range(i + 1, n):
            d = np.linalg.norm(centers[i] - centers[j])
            dists[i, j] = d
            dists[j, i] = d
            
            # Row for constraint r_i + r_j <= d
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(d)
            
    # Boundary constraints: r_i <= wall_dist_i
    for i in range(n):
        x, y = centers[i]
        # Distance to nearest wall
        w = min(x, 1 - x, y, 1 - y)
        
        # Ensure wall distance is non-negative (clamp to 0 if center is outside, though shouldn't happen)
        w = max(0.0, w)
        
        row = np.zeros(n)
        row[i] = 1.0
        A_ub.append(row)
        b_ub.append(w)
        
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    # Bounds: r_i >= 0
    bounds = [(0, None) for _ in range(n)]
    
    # Solve LP
    # 'highs' method is generally fast and robust
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            radii = -res.x
            sum_radii = np.sum(radii)
            return sum_radii, radii
        else:
            # If LP fails (e.g., infeasible due to numerical issues), return zeros
            return 0.0, np.zeros(n)
    except Exception:
