import numpy as np
from scipy.optimize import minimize

def compute_objective(vars, n):
    """
    Computes the objective: -sum(radii) + penalty for overlaps and boundary violations.
    """
    centers = vars[:2*n].reshape(n, 2)
    radii = vars[2*n:]
    
    penalty = 0.0
    
    # Boundary penalties: push circles inside [0, 1]x[0, 1]
    penalty += np.sum(np.maximum(0, radii - centers[:, 0])**2)
    penalty += np.sum(np.maximum(0, radii - centers[:, 1])**2)
    penalty += np.sum(np.maximum(0, centers[:, 0] + radii - 1.0)**2)
    penalty += np.sum(np.maximum(0, centers[:, 1] + radii - 1.0)**2)
    
    # Pairwise overlap penalties
    diffs = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    r_sum = radii[:, None] + radii[None, :]
    overlaps = np.maximum(0, r_sum - dists)
    np.fill_diagonal(overlaps, 0)
    penalty += np.sum(overlaps**2)
    
    return -np.sum(radii) + 1000 * penalty

def run_packing():
    n = 26
    best_sum = -np.inf
    best_centers = None
    best_radii = None
    
    # Initialization 1: Hexagonal lattice
    init_centers = []
    y = 0.1
    row_counts = [6, 5, 6, 5, 4]
    for count in row_counts:
        x_start = (1.0 - (count - 1) * 0.15) / 2.0
        for i in range(count):
            init_centers.append([x_start + i * 0.15, y])
        y += 0.15 * np.sqrt(3) / 2
    init_centers = np.array(init_centers)
    
    # Initialization 2: Perturbed hex lattice to break symmetry
    np.random.seed(123)
    perturbed = init_centers.copy()
    for _ in range(30):
        idx = np.random.randint(n)
        perturbed[idx] += np.random.uniform(-0.02, 0.02, 2)
        
    candidates = [init_centers, perturbed]
    
    for init_c in candidates:
        radii = np.ones(n) * 0.08
        x0 = np.concatenate([init_c.flatten(), radii])
        bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
        
        try:
            res = minimize(
                compute_objective, 
                x0, 
                args=(n,), 
                bounds=bounds, 
                method='L-BFGS-B', 
                options={'maxiter': 20000, 'ftol': 1e-12, 'gtol': 1e-8}
            )
            
            c = res.x[:2*n].reshape(n, 2)
            r = res.x[2*n:]
            curr_sum = np.sum(r)
            
            # Check max overlap to ensure validity
            diffs = c[:, None, :] - c[None, :, :]
            dists = np.sqrt(np.sum(diffs**2, axis=2))
            r_sum = r[:, None] + r[None, :]
            max_overlap = np.max(np.maximum(0, r_sum - dists))
            
            # Accept if overlaps are negligible and sum is improved
            if max_overlap < 1e-5 and curr_sum > best_sum:
                best_sum = curr_sum
                best_centers = c.copy()
                best_radii = r.copy()
        except Exception:
            continue
            
    # Fallback to a valid but small packing if optimization fails
    if best_centers is None:
        best_centers = np.tile([0.5, 0.5], (n, 1))
        best_radii = np.zeros(n)
        best_sum = 0.0
        
    return best_centers, best_radii, best_sum
