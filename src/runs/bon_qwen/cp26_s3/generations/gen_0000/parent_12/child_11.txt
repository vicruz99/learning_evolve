To maximize the sum of radii for 26 circles in a unit square, we employ a penalty-based optimization approach using `scipy.optimize.minimize` with the L-BFGS-B algorithm. 

**Strategy:**
1. **Formulation:** We maximize $\sum r_i$ subject to non-overlap and boundary constraints. This is transformed into a minimization problem by negating the objective and adding a quadratic penalty for constraint violations.
2. **Penalty Function:** We use a high penalty factor $\beta = 10000$. The loss function is:
   $L = -\sum r_i + \beta \sum_{i<j} \max(0, r_i + r_j - \|c_i - c_j\|)^2 + \beta \sum_i \sum_{walls} \max(0, r_i - \text{dist\_to\_wall})^2$
   This smoothly penalizes overlaps and boundary breaches, guiding the optimizer toward a valid, dense packing.
3. **Initialization:** Circle packing is highly non-convex. We generate multiple starting configurations:
   - A perturbed hexagonal lattice (optimal for equal circles).
   - Multiple random restarts to escape local minima.
4. **Optimization:** L-BFGS-B is used with bounds $c \in [0,1]$ and $r \in [\epsilon, 0.5]$. It efficiently handles the bounded variables and converges quickly for this dimensionality.
5. **Post-processing:** After optimization, we apply a strict clamping and slight shrinkage step to guarantee that all circles satisfy the validation constraints within numerical tolerance ($10^{-12}$).

import numpy as np
from scipy.optimize import minimize
import math
import random

def packing_loss(vars, beta):
    """
    Computes the objective function value.
    Minimizes negative sum of radii + penalty for overlaps and boundary violations.
    """
    N = 26
    centers = vars[:2*N].reshape(N, 2)
    radii = vars[2*N:]
    loss = -np.sum(radii)
    
    # Pairwise repulsion penalty
    for i in range(N):
        for j in range(i+1, N):
            diff = centers[i] - centers[j]
            dist = math.sqrt(np.sum(diff**2))
            if dist < 1e-12: 
                dist = 1e-12
            v = radii[i] + radii[j] - dist
            if v > 0:
                loss += beta * v**2
                
    # Boundary constraints penalty
    for i in range(N):
        v = radii[i] - centers[i, 0]
        if v > 0:
            loss += beta * v**2
        v = radii[i] - (1.0 - centers[i, 0])
        if v > 0:
            loss += beta * v**2
        v = radii[i] - centers[i, 1]
        if v > 0:
            loss += beta * v**2
        v = radii[i] - (1.0 - centers[i, 1])
        if v > 0:
            loss += beta * v**2
            
    return loss

def run_packing():
    N = 26
    best_sum_r = 0.0
    best_c = None
    best_r = None
    
    # Generate initial configurations
    inits = []
    
    # 1. Hexagonal grid initialization
    r0 = 0.08
    pts = []
    for row in range(8):
        for col in range(8):
            y = r0 + row * math.sqrt(3) * r0
            x = r0 + col * 2 * r0
            if row % 2 == 1: x += r0
            if 0 <= x <= 1 and 0 <= y <= 1:
                pts.append([x, y])
    random.seed(42)
    if len(pts) > 26:
        random.shuffle(pts)
        pts = pts[:26]
    else:
        while len(pts) < 26:
            pts.append([np.random.rand(), np.random.rand()])
    inits.append(np.concatenate([np.array(pts).flatten(), np.full(26, 0.05)]))
    
    # 2. Random restarts
    for _ in range(10):
        c = np.random.rand(26, 2) * 0.8 + 0.1
        r = np.full(26, 0.08)
        inits.append(np.concatenate([c.flatten(), r]))
        
    bounds = [(0,1)]*2*N + [(1e-6, 0.5)]*N
    
    # Optimize from each starting point
    for init in inits:
        # Wrap loss to pass beta
        def obj(vars):
            return packing_loss(vars, 10000.0)
            
        res = minimize(obj, init, method='L-BFGS-B', 
                       bounds=bounds, options={'ftol':1e-12, 'gtol':1e-10, 'maxiter':5000})
        
        c = res.x[:2*N].reshape(N, 2)
        r = res.x[2*N:]
        
        # Post-process to ensure strict validity
        # Iteratively shrink if any overlap/boundary violation remains
        for _ in range(5):
            overlap = False
            for i in range(N):
                for j in range(i+1, N):
                    if np.linalg.norm(c[i]-c[j]) < r[i]+r[j] - 1e-12:
                        overlap = True
                        break
                if overlap: break
            if not overlap:
                for i in range(N):
                    if c[i,0]-r[i]<-1e-12 or c[i,0]+r[i]>1+1e-12 or c[i,1]-r[i]<-1e-12 or c[i,1]+r[i]>1+1e-12:
                        overlap = True
                        break
            if overlap:
                r *= 0.99
            else:
                break
                
        if np.sum(r) > best_sum_r:
            best_sum_r = np.sum(r)
            best_c = c.copy()
            best_r = r.copy()
            
    # Final strict clamping to guarantee validation passes
    if best_c is not None:
        # Clamp to boundaries
        for i in range(N):
            max_r = min(best_c[i,0], 1-best_c[i,0], best_c[i,1], 1-best_c[i,1]) - 1e-13
            if best_r[i] > max_r:
                best_r[i] = max_r
                
        # Clamp pairwise
        for _ in range(3):
            changed = False
            for i in range(N):
                for j in range(i+1, N):
                    req = np.linalg.norm(best_c[i]-best_c[j]) - 1e-13
                    if best_r[i] + best_r[j] > req:
                        scale = max(0.0, req / (best_r[i] + best_r[j] + 1e-15))
                        best_r[i] *= scale
                        best_r[j] *= scale
                        changed = True
            if not changed: break

    return best_c, best_r, np.sum(best_r)
