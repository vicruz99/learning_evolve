import numpy as np
from scipy.optimize import minimize

def compute_objective(vars, n):
    """
    Objective function to maximize sum of radii with penalty for constraint violations.
    vars: flattened array [r_1..r_n, x_1..x_n, y_1..y_n]
    """
    r = vars[:n]
    x = vars[n:2*n]
    y = vars[2*n:3*n]
    
    penalty = 0.0
    PW = 1e5  # Penalty weight
    
    # Boundary constraints: r <= x <= 1-r and r <= y <= 1-r
    for i in range(n):
        d_left = x[i] - r[i]
        if d_left < 0: penalty += d_left**2
        d_right = 1 - r[i] - x[i]
        if d_right < 0: penalty += d_right**2
        d_down = y[i] - r[i]
        if d_down < 0: penalty += d_down**2
        d_up = 1 - r[i] - y[i]
        if d_up < 0: penalty += d_up**2
        
    # Non-overlap constraints: dist >= r_i + r_j
    for i in range(n):
        for j in range(i+1, n):
            dist_sq = (x[i]-x[j])**2 + (y[i]-y[j])**2
            dist = np.sqrt(dist_sq)
            overlap = r[i] + r[j] - dist
            if overlap > 0: penalty += overlap**2
            
    return -np.sum(r) + PW * penalty

def run_packing():
    n = 26
    best_sum = -1.0
    best_centers = None
    best_radii = None

    bounds = [(0.0001, 0.5)] * n + [(0.0, 1.0)] * n + [(0.0, 1.0)] * n

    base_y = []

            base_y.append(0.1 + i * 0.2)
    base_x.append(0.5)
    base_y.append(0.5)
    base_r = np.array([0.1]*25 + [0.01])
    
    np.random.seed(42)
    
    # Multiple restarts to escape local minima
    for restart in range(20):
        if restart < 10:
            # Perturb grid start
            scale = 0.02 * (restart + 1)
            x0 = base_x + np.random.uniform(-scale, scale, n)
            y0 = base_y + np.random.uniform(-scale, scale, n)

            r0 = np.random.uniform(0.05, 0.15, n)

        y0 = np.clip(y0, 0.0, 1.0)

        for i in range(n):
            r0[i] = min(r0[i], x0[i], 1-x0[i], y0[i], 1-y0[i])
            
        vars0 = np.concatenate([r0, x0, y0])
        
        # Local optimization
        res = minimize(compute_objective, vars0, args=(n,), method='L-BFGS-B', 
                       bounds=bounds, options={'maxiter': 3000, 'ftol': 1e-12})
        
        cur_r = res.x[:n]
        cur_x = res.x[n:2*n]
        cur_y = res.x[2*n:3*n]
        cur_sum = np.sum(cur_r)

    return best_centers, best_radii, best_sum
