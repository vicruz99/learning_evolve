import numpy as np
from scipy.optimize import minimize, linprog

def sllsqp_objective(params, n):
    """Objective function: maximize sum of radii (minimize negative sum)."""
    return -np.sum(params[2 * n:])

def sllsqp_constraints(params, n):
    """
    Inequality constraints for SLSQP.
    Returns array of values that must be >= 0.
    Includes boundary constraints and non-overlap constraints.
    """
    centers = params[:2 * n].reshape(n, 2)
    radii = params[2 * n:]
    vals = []
    
    # Boundary constraints: r <= x, r <= 1-x, r <= y, r <= 1-y
    for i in range(n):
        vals.append(centers[i, 0] - radii[i])
        vals.append(1.0 - centers[i, 0] - radii[i])
        vals.append(centers[i, 1] - radii[i])
        vals.append(1.0 - centers[i, 1] - radii[i])
        
    # Non-overlap constraints: dist >= r_i + r_j
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2) + 1e-12)
            vals.append(dist - radii[i] - radii[j])
            
    return np.array(vals)

def get_lp_matrices(centers):
    """
    Constructs A_ub and b_ub for the LP:
    Maximize sum(r_i)  <=>  Minimize -sum(r_i)
    Subject to: r_i + r_j <= dist_ij, and boundary constraints.
    """
    n = centers.shape[0]
    n_constr = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((n_constr, n))
    b_ub = np.zeros(n_constr)
    idx = 0
    
    for i in range(n):
        x, y = centers[i]
        # r_i <= x
        A_ub[idx, i] = 1.0; b_ub[idx] = x; idx += 1
        # r_i <= 1-x
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - x; idx += 1
        # r_i <= y
        A_ub[idx, i] = 1.0; b_ub[idx] = y; idx += 1
        # r_i <= 1-y
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - y; idx += 1
        
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2) + 1e-12)
            # r_i + r_j <= dist
            A_ub[idx, i] = 1.0; A_ub[idx, j] = 1.0; b_ub[idx] = dist; idx += 1
            
    return A_ub, b_ub

def run_packing():
    np.random.seed(42)
    n = 26
    best_centers = None
    best_radii = None
    best_sum = 0.0
    
    # Multi-start optimization
    for trial in range(15):
        # Initialize centers randomly within the square
        centers_init = np.random.rand(n, 2) * 0.8 + 0.1
        radii_init = np.full(n, 0.01)
        x0 = np.concatenate([centers_init.flatten(), radii_init])
        
        bounds = [(0, 1)] * (2 * n) + [(0, 0.5)] * n
        cons = {'type': 'ineq', 'fun': sllsqp_constraints, 'args': (n,)}
        
        res = minimize(sllsqp_objective, x0, args=(n,), method='SLSQP', 
                       bounds=bounds, constraints=cons, 
                       options={'maxiter': 1000, 'ftol': 1e-9})
        
        if res.success:
            centers_opt = res.x[:2 * n].reshape(n, 2)
            
            # LP refinement to maximize radii for fixed centers
            A, b = get_lp_matrices(centers_opt)
            res_lp = linprog(-np.ones(n), A_ub=A, b_ub=b, 
                             bounds=[(0, None)] * n, method='highs')
            
            if res_lp.success:
                s = np.sum(res_lp.x)
                if s > best_sum:
                    best_sum = s
                    best_centers = centers_opt.copy()
                    best_radii = res_lp.x.copy()
                    
    # Fallback if optimization fails (unlikely)
    if best_centers is None:
        best_centers = np.random.rand(n, 2) * 0.8 + 0.1
        best_radii = np.full(n, 0.01)
        
    # Final LP pass to ensure radii are strictly optimal for best centers
    A, b = get_lp_matrices(best_centers)
    res_lp = linprog(-np.ones(n), A_ub=A, b_ub=b, 
                     bounds=[(0, None)] * n, method='highs')
    if res_lp.success:
        best_radii = res_lp.x
        best_sum = np.sum(best_radii)
        
    return best_centers, best_radii, float(best_sum)
