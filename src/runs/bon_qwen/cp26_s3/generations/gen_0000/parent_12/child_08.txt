import numpy as np
from scipy.optimize import minimize


def init_packing(n):
    """Initialize circles in a hexagonal pattern."""
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.05

    idx = 0
    spacing = 0.17
    y_spacing = spacing * np.sqrt(3) / 2

    for row in range(8):
        num_cols = 5
        x_offset = spacing / 2 if row % 2 == 1 else 0

        for col in range(num_cols):
            if idx >= n:
                break
            x = 0.08 + col * spacing + x_offset
            y = 0.08 + row * y_spacing
            centers[idx] = [np.clip(x, 0.01, 0.99), np.clip(y, 0.01, 0.99)]
            idx += 1

    return centers, radii


def compute_forces(centers, radii, n):
    """Compute repulsion forces between circles and from boundaries."""
    forces = np.zeros((n, 2))

    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[j][0] - centers[i][0]
            dy = centers[j][1] - centers[i][1]
            dist_sq = dx * dx + dy * dy

            if dist_sq < 1e-14:
                dx = 0.0001
                dy = 0.0001
                dist_sq = dx * dx + dy * dy

            dist = np.sqrt(dist_sq)
            overlap = radii[i] + radii[j] - dist

            if overlap > 0:
                f = overlap * 80.0 / dist
                forces[i][0] -= dx * f
                forces[i][1] -= dy * f
                forces[j][0] += dx * f
                forces[j][1] += dy * f

        for d in range(2):
            if centers[i][d] < radii[i]:
                forces[i][d] += (radii[i] - centers[i][d]) * 200.0
            if centers[i][d] > 1 - radii[i]:
                forces[i][d] -= (centers[i][d] - (1 - radii[i])) * 200.0

    return forces


def check_can_grow(centers, radii, growth, n):
    """Check if all radii can be increased by growth amount."""
    new_radii = radii + growth

    for i in range(n):
        if new_radii[i] < 0:
            return False
        for d in range(2):
            if centers[i][d] - new_radii[i] < -1e-10:
                return False
            if centers[i][d] + new_radii[i] > 1 + 1e-10:
                return False
        for j in range(i + 1, n):
            dx = centers[i][0] - centers[j][0]
            dy = centers[i][1] - centers[j][1]
            dist = np.sqrt(dx * dx + dy * dy)
            if dist < new_radii[i] + new_radii[j] - 1e-10:
                return False

    return True


def relax_and_grow(centers, radii, n, max_iter=80000):
    """Force-directed relaxation with iterative radius growth."""
    dt = 0.0008

    for it in range(max_iter):
        forces = compute_forces(centers, radii, n)
        centers = centers + forces * dt

        if it % 200 == 0:
            if check_can_grow(centers, radii, 5e-5, n):
                radii = radii + 5e-5

    return centers, radii


def packing_objective(params, n):
    """Objective function: maximize sum of radii with penalty for violations."""
    c = params[: 2 * n].reshape(n, 2)
    r = params[2 * n :]
    loss = -np.sum(r)

    for i in range(n):
        for j in range(i + 1, n):
            dx = c[i][0] - c[j][0]
            dy = c[i][1] - c[j][1]
            d = np.sqrt(dx * dx + dy * dy)
            ov = r[i] + r[j] - d
            if ov > 0:
                loss -= ov * ov * 50000

        for d_dim in range(2):
            violation_left = r[i] - c[i][d_dim]
            if violation_left > 0:
                loss -= violation_left * violation_left * 50000
            violation_right = c[i][d_dim] + r[i] - 1
            if violation_right > 0:
                loss -= violation_right * violation_right * 50000

    return loss


def optimize_positions(centers, radii, n):
    """Fine-tune with scipy L-BFGS-B optimization."""
    params = np.concatenate([centers.flatten(), radii])
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n

    result = minimize(
        packing_objective,
        params,
        args=(n,),
        method="L-BFGS-B",
        bounds=bounds,
        options={"maxiter": 15000, "ftol": 1e-15, "gtol": 1e-10},
    )

    return result.x[: 2 * n].reshape(n, 2), result.x[2 * n :]


def repair_packing(centers, radii, n):
    """Repair packing to ensure validity."""
    for _ in range(500):
        max_violation = 0

        for i in range(n):
            for d in range(2):
                if centers[i][d] < radii[i]:
                    max_violation = max(max_violation, radii[i] - centers[i][d])
                if centers[i][d] > 1 - radii[i]:
                    max_violation = max(max_violation, centers[i][d] - (1 - radii[i]))

        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                d = np.sqrt(dx * dx + dy * dy)
                v = radii[i] + radii[j] - d
                if v > max_violation:
                    max_violation = v

        if max_violation < 1e-12:
            break

        for i in range(n):
            radii[i] -= max_violation / 2

        for i in range(n):
            radii[i] = max(0, radii[i])
            centers[i][0] = np.clip(centers[i][0], radii[i], 1 - radii[i])
            centers[i][1] = np.clip(centers[i][1], radii[i], 1 - radii[i])

    return centers, radii


def run_single_attempt(centers, radii, n):
    """Run one complete optimization attempt."""
    centers, radii = relax_and_grow(centers, radii, n, max_iter=80000)
    centers, radii = optimize_positions(centers, radii, n)
    centers, radii = repair_packing(centers, radii, n)
    return centers, radii, np.sum(radii)


def run_packing():
    """Main packing function. Returns centers, radii, sum_radii."""
    n = 26
    best_sum = -1.0
    best_centers = None
    best_radii = None

    # Multiple restarts with slight perturbations
    for restart in range(8):
        centers, radii = init_packing(n)

        # Add small random perturbation for diversity
        if restart > 0:
            noise = np.random.randn(2 * n) * 0.02
            centers = centers + noise.reshape(n, 2)
            centers = np.clip(centers, 0.05, 0.95)
            radii = np.ones(n) * (0.05 + np.random.randn() * 0.01)

        c, r, s = run_single_attempt(centers, radii, n)

        if s > best_sum:
            best_sum = s
            best_centers = c.copy()
            best_radii = r.copy()

    return best_centers, best_radii, best_sum
