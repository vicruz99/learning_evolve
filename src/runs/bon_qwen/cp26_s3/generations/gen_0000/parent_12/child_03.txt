import numpy as np
from scipy.optimize import minimize, NonlinearConstraint

def objective(vars):
    """Objective: minimize negative sum of radii (equivalent to maximizing sum)"""
    return -np.sum(vars[:26])

def packing_constraints(vars):
    """
    Computes all packing constraints.
    Returns an array where each element must be <= 0.
    """
    r = vars[:26]
    c = vars[26:].reshape(26, 2)
    
    # Pairwise distances
    diffs = c[:, None, :] - c[None, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2) + 1e-20)
    
    # Lower triangle mask for unique pairs (i < j)
    mask = np.tril(np.ones((26, 26), dtype=bool), k=-1)
    
    # Non-overlap constraint: r_i + r_j <= dist_ij  =>  r_i + r_j - dist_ij <= 0
    pair_c = (r[:, None][mask] + r[None, :][mask]) - dists[mask]
    
    # Wall constraints: circles must stay inside [0,1]^2
    wall_c = np.concatenate([
        r - c[:, 0],          # r <= x
        r + c[:, 0] - 1.0,    # r + x <= 1
        r - c[:, 1],          # r <= y
        r + c[:, 1] - 1.0     # r + y <= 1
    ])
    
    return np.concatenate([pair_c, wall_c])

def get_initial_guess():
    """Generates a feasible initial configuration using a hexagonal grid."""
    centers = np.zeros((26, 2))
    radii = np.full(26, 0.05)
    idx = 0
    step_x = 0.15
    step_y = step_x * np.sqrt(3) / 2
    y = 0.1
    offset = 0.0
    
    while idx < 26:
        x = 0.1 + offset
        while x < 0.9 and idx < 26:
            centers[idx] = [x, y]
            idx += 1
            x += step_x
        y += step_y
        offset = (1 - offset) * step_x
        
    # Break symmetry to avoid degenerate optimizer paths
    np.random.seed(42)
    centers += np.random.randn(26, 2) * 0.005
    return radii, centers

def run_packing():
    radii0, centers0 = get_initial_guess()
    vars0 = np.concatenate([radii0, centers0.flatten()])
    
    # Bounds: radii in [0, 0.5], centers in [0, 1]
    bounds = [(0, 0.5)] * 26 + [(0, 1)] * 52
    cons = NonlinearConstraint(packing_constraints, -np.inf, 0)
    
    # trust-constr handles vector constraints efficiently
    res = minimize(objective, vars0, method='trust-constr', bounds=bounds, 
                   constraints=cons, options={'maxiter': 5000, 'verbose': 0})
                   
    r_opt = np.maximum(res.x[:26], 0.0)
    c_opt = np.clip(res.x[26:].reshape(26, 2), 0.0, 1.0)
    
    return c_opt, r_opt, np.sum(r_opt)
