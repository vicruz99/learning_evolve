def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    def compute_loss_and_grad(vars, penalty_weight):

        # vars structure: [x0, y0, r0, x1, y1, r1, ..., x25, y25, r25]
        # Total 78 variables
        centers = np.zeros((n, 2))
        radii = np.zeros(n)
        for i in range(n):
            centers[i, 0] = vars[3 * i]
            centers[i, 1] = vars[3 * i + 1]
            radii[i] = vars[3 * i + 2]

        obj = -np.sum(radii)
        grad_obj = np.zeros_like(vars)
        for i in range(n):
            grad_obj[3 * i + 2] = -1.0

        # Penalties
        penalty = 0.0
        grad_penalty = np.zeros_like(vars)

        # 1. Boundary constraints
        # x - r >= 0  => max(0, r - x)^2
        # x + r <= 1  => max(0, x + r - 1)^2
        # Same for y

            x, y, r = centers[i, 0], centers[i, 1], radii[i]

            viol_x_minus = r - x

                penalty += viol_x_minus**2

                grad_penalty[3 * i] += d_viol * (-1) # deriv wrt x
                grad_penalty[3 * i + 2] += d_viol * (1) # deriv wrt r

                grad_penalty[3 * i] += d_viol * (1)
                grad_penalty[3 * i + 2] += d_viol * (1)

            viol_y_minus = r - y

                penalty += viol_y_plus**2

                grad_penalty[3 * i + 1] += d_viol * (1)
                grad_penalty[3 * i + 2] += d_viol * (1)

                dy = centers[i, 1] - centers[j, 1]

                dist = np.sqrt(dist_sq)

                    dist = 1e-9
                
                sum_r = radii[i] + radii[j]
                viol = sum_r - dist
                
                if viol > 0:
                    penalty += viol**2
                    d_viol = 2.0 * viol
                    
                    # Gradient wrt radii

                    grad_penalty[3 * j + 2] += d_viol * (1.0)

                    grad_penalty[3 * i] += factor * dx
                    grad_penalty[3 * i + 1] += factor * dy
                    grad_penalty[3 * j] += -factor * dx
                    grad_penalty[3 * j + 1] += -factor * dy

        return total_loss, total_grad
