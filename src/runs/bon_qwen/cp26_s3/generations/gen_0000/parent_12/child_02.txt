The problem is a non-convex geometric optimization: maximize $\sum r_i$ subject to boundary and non-overlap constraints. 

**Strategy:**
1. **Initial Configuration:** Start with a hexagonal lattice packing, which is known to be highly dense for equal circles. This provides a geometrically sound starting point close to the optimum. We place ~26 circles with a conservative initial radius (0.085) ensuring feasibility.
2. **Optimization Formulation:** Formulate as a nonlinear programming problem. Variables are $(x_i, y_i, r_i)$ for each circle. 
   - Objective: Maximize $\sum r_i$ (minimize $-\sum r_i$).
   - Constraints: 
     - Boundary: $r_i \le x_i \le 1-r_i$ and $r_i \le y_i \le 1-r_i$.
     - Non-overlap: $(x_i-x_j)^2 + (y_i-y_j)^2 \ge (r_i+r_j)^2$.
3. **Solver:** Use `scipy.optimize.minimize` with the `SLSQP` algorithm, which handles nonlinear equality/inequality constraints efficiently. 
4. **Robustness:** Run the optimizer multiple times (3 trials) with slight random perturbations to escape local minima and ensure we find a high-quality packing. 
5. **Post-processing:** Clamp radii to be non-negative and return the best configuration found.

This approach leverages the known efficiency of hexagonal packing while allowing the solver to adjust radii unequally and reposition centers to squeeze out maximum total radius, targeting the 2.636 benchmark.

import numpy as np
from scipy.optimize import minimize

def boundary_x_lower(v, i):
    return v[3*i] - v[3*i+2]

def boundary_x_upper(v, i):
    return 1.0 - v[3*i] - v[3*i+2]

def boundary_y_lower(v, i):
    return v[3*i+1] - v[3*i+2]

def boundary_y_upper(v, i):
    return 1.0 - v[3*i+1] - v[3*i+2]

def overlap_constraint(v, i, j):
    dx = v[3*i] - v[3*j]
    dy = v[3*i+1] - v[3*j+1]
    dr = v[3*i+2] + v[3*j+2]
    return dx*dx + dy*dy - dr*dr

def run_packing():
    n = 26
    np.random.seed(42)
    
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Bounds: x,y in [0,1], r in [0, 0.5]
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
    
    constraints = []
    for i in range(n):
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: boundary_x_lower(v, i)})
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: boundary_x_upper(v, i)})
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: boundary_y_lower(v, i)})
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: boundary_y_upper(v, i)})
        
    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({'type': 'ineq', 'fun': lambda v, i=i, j=j: overlap_constraint(v, i, j)})

    def objective(v):
        return -sum(v[3*i+2] for i in range(n))

    def get_init_guess(perturb_scale):
        # Hexagonal lattice initialization
        r_est = 0.085
        centers = []
        row_y = r_est
        row_idx = 0
        count = 0
        while count < n and row_y + r_est <= 1.0:
            shift = r_est if row_idx % 2 == 1 else 0.0
            x = r_est + shift
            while x + r_est <= 1.0 and count < n:
                centers.append([x, row_y])
                count += 1
                x += 2.0 * r_est
            row_y += r_est * np.sqrt(3.0)
            row_idx += 1
            
        # Fill remaining if any
        while len(centers) < n:
            centers.append([0.5 + (np.random.rand() - 0.5) * 0.3, 
                            0.5 + (np.random.rand() - 0.5) * 0.3])
                            
        centers = np.array(centers[:n])
        if perturb_scale > 0:
            centers += np.random.normal(0, perturb_scale, centers.shape)
            centers = np.clip(centers, 0.01, 0.99)
            
        x0 = np.zeros(3 * n)
        for i in range(n):
            x0[3*i] = centers[i, 0]
            x0[3*i+1] = centers[i, 1]
            x0[3*i+2] = r_est
        return x0

    # Multiple restarts to avoid local minima
    for trial in range(3):
        x0 = get_init_guess(perturb_scale=0.0 if trial == 0 else 0.01)
        res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints, 
                       options={'maxiter': 2500, 'ftol': 1e-11})
        
        current_sum = -res.fun
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = np.zeros((n, 2))
            best_radii = np.zeros(n)
            for i in range(n):
                best_centers[i] = [res.x[3*i], res.x[3*i+1]]
                best_radii[i] = res.x[3*i+2]

    best_radii = np.maximum(best_radii, 0.0)
    return best_centers, best_radii, best_sum
