    # Define constraints and objective
    def objective(vars):
        # vars shape: (n*3,) -> [x1, y1, r1, x2, y2, r2, ...]
        # We want to maximize sum(r)
        # Minimize -sum(r)
        radii = vars[2::3]
        return -np.sum(radii)

    def grad_objective(vars):
        # Gradient of -sum(r) wrt vars
        # dr_i is at index 3*i + 2
        grad = np.zeros_like(vars)
        for i in range(n):
            grad[3*i + 2] = -1.0
        return grad

    # Constraints
    # 1. r >= 0 (bounds handled by bounds parameter in minimize)
    # 2. Boundary: r <= x <= 1-r, r <= y <= 1-r
    #    x - r >= 0  => x - r >= 0
    #    1 - x - r >= 0 => x + r <= 1
    #    y - r >= 0
    #    1 - y - r >= 0
    def boundary_constraints(vars):
        con = []
        for i in range(n):
            x = vars[3*i]
            y = vars[3*i+1]
            r = vars[3*i+2]
            # x - r >= 0
            con.append(x - r)
            # 1 - (x + r) >= 0
            con.append(1 - x - r)
            # y - r >= 0
            con.append(y - r)
            # 1 - (y + r) >= 0
            con.append(1 - y - r)
        return np.array(con)

    def jac_boundary_constraints(vars):
        # Jacobian of boundary constraints
        # 4 constraints per circle.
        # Constraint 0 (x-r): dx=1, dy=0, dr=-1
        # Constraint 1 (1-x-r): dx=-1, dy=0, dr=-1
        # Constraint 2 (y-r): dx=0, dy=1, dr=-1
        # Constraint 3 (1-y-r): dx=0, dy=-1, dr=-1
        J = np.zeros((n*4, n*3))
        for i in range(n):
            idx = 4*i
            # x - r >= 0
            J[idx, 3*i] = 1.0
            J[idx, 3*i+2] = -1.0
            # 1 - x - r >= 0
            J[idx+1, 3*i] = -1.0
            J[idx+1, 3*i+2] = -1.0
            # y - r >= 0
            J[idx+2, 3*i+1] = 1.0
            J[idx+2, 3*i+2] = -1.0
            # 1 - y - r >= 0
            J[idx+3, 3*i+1] = -1.0
            J[idx+3, 3*i+2] = -1.0
        return J

    # 3. Non-overlap: dist >= r_i + r_j

    # We can formulate as: dist - (r_i + r_j) >= 0 ? 
    # But dist is sqrt, not smooth at 0.
    # Better: (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0.

    def non_overlap_constraints(vars):
        cons = []
        for i in range(n):
            for j in range(i + 1, n):
                x_i, y_i, r_i = vars[3*i], vars[3*i+1], vars[3*i+2]
                x_j, y_j, r_j = vars[3*j], vars[3*j+1], vars[3*j+2]
                
                dx = x_i - x_j
                dy = y_i - y_j
                dr = r_i + r_j
                
                # Constraint: dx^2 + dy^2 - dr^2 >= 0
                val = dx*dx + dy*dy - dr*dr
                cons.append(val)
        return np.array(cons)

    def jac_non_overlap_constraints(vars):
        # Number of constraints
        num_cons = n * (n - 1) // 2
        num_vars = n * 3
        J = np.zeros((num_cons, num_vars))
        
        idx = 0
        for i in range(n):
            for j in range(i + 1, n):
                x_i, y_i, r_i = vars[3*i], vars[3*i+1], vars[3*i+2]
                x_j, y_j, r_j = vars[3*j], vars[3*j+1], vars[3*j+2]
                
                dx = x_i - x_j
                dy = y_i - y_j
                dr = r_i + r_j
                
                # Partial derivatives of (dx^2 + dy^2 - dr^2)
                # w.r.t x_i: 2*dx
                # w.r.t y_i: 2*dy
                # w.r.t r_i: -2*dr
                # w.r.t x_j: -2*dx
                # w.r.t y_j: -2*dy
                # w.r.t r_j: -2*dr
                
                J[idx, 3*i] = 2 * dx
                J[idx, 3*i+1] = 2 * dy
                J[idx, 3*i+2] = -2 * dr
                
                J[idx, 3*j] = -2 * dx
                J[idx, 3*j+1] = -2 * dy
                J[idx, 3*j+2] = -2 * dr
                
                idx += 1
        return J

    # Combine constraints
    def constraints_func(vars):
        b_cons = boundary_constraints(vars)
        o_cons = non_overlap_constraints(vars)
        return np.concatenate([b_cons, o_cons])

    def jac_constraints_func(vars):
        J_b = jac_boundary_constraints(vars)
        J_o = jac_non_overlap_constraints(vars)
        return np.concatenate([J_b, J_o], axis=0)

    # Initial Guess
    centers_init, radii_init = get_initial_guess(n)
    
    # Flatten
    x0 = np.zeros(n * 3)
    for i in range(n):
        x0[3*i] = centers_init[i, 0]
        x0[3*i+1] = centers_init[i, 1]
        x0[3*i+2] = radii_init[i]

    # Bounds: x in [0,1], y in [0,1], r in [0, 0.5] (max possible radius)
    bounds = []
    for i in range(n):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 0.5)) # r

    # Optimization options
    options = {
        'maxiter': 1000,
        'ftol': 1e-12,
        'disp': False
    }

    try:
        res = scipy.optimize.minimize(
            objective,
            x0,
            method='SLSQP',
            jac=grad_objective,
            bounds=bounds,
            constraints={'type': 'ineq', 'fun': constraints_func, 'jac': jac_constraints_func},
            options=options
        )
        
        vars_opt = res.x
    except Exception as e:
        # Fallback or error handling if optimization fails
        # Just return initial guess
        vars_opt = x0

    # Unpack
    centers = vars_opt[0::3].reshape(-1, 2)
    radii = vars_opt[2::3]
    
    sum_radii = np.sum(radii)
    
    # Final validation check (optional, but good for debugging if we could print)
    # The validate_packing function will check it.
    
    return centers, radii, sum_radii
