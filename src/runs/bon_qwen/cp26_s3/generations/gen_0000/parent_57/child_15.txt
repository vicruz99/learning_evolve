2. **Numerical Optimization**: We use `scipy.optimize.minimize` with the SLSQP algorithm, which handles nonlinear inequality constraints effectively. The objective is to minimize the negative sum of radii.
3. **Constraints**:

4. **Multiple Restarts**: Circle packing landscapes are highly non-convex with many local optima. We run 10 optimization passes from slightly perturbed initial configurations to increase the chance of finding a global or near-global optimum.

import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def objective(vars):
    """Objective function: negative sum of radii."""
    return -np.sum(vars[2*N_CIRCLES:])

def constraint_fun(vars):
    """Inequality constraints: boundaries and non-overlap."""
    n = N_CIRCLES
    centers = vars[:2*n].reshape((n, 2))
    radii = vars[2*n:]
    
    c = []
    # Boundary constraints
    c.append(centers[:, 0] - radii)
    c.append(1 - centers[:, 0] - radii)
    c.append(centers[:, 1] - radii)
    c.append(1 - centers[:, 1] - radii)
    
    # Distance constraints (non-overlap)
    diff = centers[:, None, :] - centers[None, :, :]
    dist_sq = np.sum(diff**2, axis=2)
    r_sum = radii[:, None] + radii[None, :]
    dist_constraints = dist_sq - r_sum**2
    
    iu, ju = np.triu_indices(n, k=1)
    c.append(dist_constraints[iu, ju])
    
    return np.concatenate(c)

def run_packing():
    n = N_CIRCLES
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Initialize with a 5x5 grid + 1 extra circle
    k = 0
    for i in range(5):
        for j in range(5):
            centers[k, 0] = 0.1 + j * 0.2
            centers[k, 1] = 0.1 + i * 0.2
            radii[k] = 0.095
            k += 1
            
    centers[25, 0] = 0.5
    centers[25, 1] = 0.9
    radii[25] = 0.05
    
    best_result = None
    best_sum = -1.0
    
    np.random.seed(42)
    
    # Multiple restarts to escape local optima
    for _ in range(10):
        x0 = np.zeros(3 * n)
        x0[:2*n] = centers.flatten()
        x0[2*n:] = radii.flatten()
        
        # Perturb initial configuration
        x0[:2*n] += np.random.uniform(-0.005, 0.005, 2*n)
        x0[2*n:] += np.random.uniform(-0.002, 0.002, n)
        x0[:2*n] = np.clip(x0[:2*n], 0, 1)
        x0[2*n:] = np.clip(x0[2*n:], 0, 0.2)
        
        bounds = [(0, 1)] * 2*n + [(0, 0.2)] * n
        cons = {'type': 'ineq', 'fun': constraint_fun}
        
        res = minimize(
            objective, 
            x0, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons, 
            options={'maxiter': 2000, 'ftol': 1e-12, 'disp': False}
        )
        
        current_sum = -res.fun
        if current_sum > best_sum:
            best_sum = current_sum
            best_result = res.x
            
    centers = best_result[:2*n].reshape((n, 2))
    radii = best_result[2*n:]
    
    # Safety clipping to ensure strict validity
    radii = np.maximum(radii, 0)
    centers = np.clip(centers, 0, 1)
    
    return centers, radii, np.sum(radii)
