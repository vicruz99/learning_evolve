import numpy as np
from scipy.optimize import minimize
import math

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    """
    n_circles = 26
    
    # --- Step 1: Initialization using Farthest Point Sampling ---
    # Generate a dense grid to sample from
    grid_size = 50
    x_grid = np.linspace(0, 1, grid_size)
    y_grid = np.linspace(0, 1, grid_size)
    grid_points = np.array(np.meshgrid(x_grid, y_grid)).T.reshape(-1, 2)
    
    # Select points using Farthest Point Sampling
    centers = np.empty((n_circles, 2))
    # Start with a point near the center
    centers[0] = [0.5, 0.5]
    
    for i in range(1, n_circles):
        # Calculate min distance from each grid point to the current set of centers
        dists = np.min(np.linalg.norm(grid_points[:, np.newaxis, :] - centers[np.newaxis, :, :], axis=2), axis=1)
        # Pick the grid point that is farthest from existing centers
        idx = np.argmax(dists)
        centers[i] = grid_points[idx]
        
    # Initialize radii to a small safe value
    radii = np.full(n_circles, 0.01)
    
    # --- Step 2: Optimization Setup ---
    # Variables: [x1, y1, r1, x2, y2, r2, ...]
    x0 = np.hstack([centers.flatten(), radii])
    
    # Bounds: 0 <= x, y <= 1, 0 <= r
    bounds = []
    for _ in range(n_circles):
        bounds.extend([(0, 1), (0, 1), (0, 0.5)]) # r max 0.5 for safety
    
    # Constraints: Non-overlap and boundary containment
    # We define a function that returns the constraint values (>= 0)
    def constraints_callback(vars_flat):
        c = []
        # Reshape back to circles
        circles = vars_flat.reshape(-1, 3) # (x, y, r)
        
        # 1. Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
        # This is equivalent to: x - r >= 0, etc.
        for i in range(n_circles):
            x, y, r = circles[i]
            c.append(x - r)
            c.append(1.0 - x - r)
            c.append(y - r)
            c.append(1.0 - y - r)
            
        # 2. Non-overlap constraints: dist^2 >= (r1 + r2)^2
        # dist^2 - (r1+r2)^2 >= 0
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                x1, y1, r1 = circles[i]
                x2, y2, r2 = circles[j]
                
                dx = x1 - x2
                dy = y1 - y2
                dist_sq = dx*dx + dy*dy
                
                min_dist_sq = (r1 + r2)**2
                
                c.append(dist_sq - min_dist_sq)
                
        return np.array(c)

    # Define constraint dict for scipy
    cons = {'type': 'ineq', 'fun': constraints_callback}
    
    # --- Step 3: Run Optimization ---
    # Objective: Maximize sum(r) => Minimize -sum(r)
    def objective(vars_flat):
        circles = vars_flat.reshape(-1, 3)
        return -np.sum(circles[:, 2])
    
    # Use SLSQP solver
    try:
        res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, 
                       options={'maxiter': 1000, 'ftol': 1e-9})
        
        if res.success:
            final_vars = res.x
        else:
            # Fallback to best found or initial if failed
            final_vars = x0 
    except Exception as e:
        # In case of solver failure, return initialized state
        final_vars = x0

    # --- Step 4: Extract and Return Results ---
    final_circles = final_vars.reshape(-1, 3)
    final_centers = final_circles[:, :2]
    final_radii = final_circles[:, 2]
    
    # Enforce strict non-negativity due to numerical noise
    final_radii = np.maximum(final_radii, 0.0)
    
    # Ensure centers are within bounds (clamp if necessary)
    final_centers[:, 0] = np.clip(final_centers[:, 0], final_radii, 1 - final_radii)
    final_centers[:, 1] = np.clip(final_centers[:, 1], final_radii, 1 - final_radii)
    
    sum_radii = np.sum(final_radii)
    
    return (final_centers, final_radii, sum_radii)
