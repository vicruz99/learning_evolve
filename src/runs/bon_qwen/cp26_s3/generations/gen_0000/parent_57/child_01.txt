import numpy as np
import scipy.optimize
import math

def run_packing():
    n = 26
    np.random.seed(42)
    
    # Initialization: Grid pattern
    # Create a 6x5 grid of points (30 points)
    grid_x = np.linspace(0.1, 0.9, 6)
    grid_y = np.linspace(0.1, 0.9, 5)
    centers = []
    for y in grid_y:
        for x in grid_x:
            centers.append([x, y])
    # We have 30 points. Take first 26.
    centers = np.array(centers[:n])
    
    # Add small noise to break symmetry
    centers += np.random.randn(n, 2) * 0.01
    centers = np.clip(centers, 0.02, 0.98)

    # Repulsion simulation to spread points nicely
    for _ in range(100):
        forces = np.zeros_like(centers)
        for i in range(n):
            for j in range(i + 1, n):
                diff = centers[i] - centers[j]
                d2 = np.sum(diff**2)
                d = math.sqrt(d2)
                if d < 0.01: 
                    # Very close, strong push
                    forces[i] += diff * 100 
                    forces[j] -= diff * 100
                elif d < 0.2: 
                    # Moderate repulsion
                    f_mag = 1.0 / d2
                    forces[i] += f_mag * diff 
                    forces[j] -= f_mag * diff
            
            # Wall repulsion
            for k in range(2):
                if centers[i][k] < 0.05: 
                    forces[i][k] += 10 * (0.05 - centers[i][k])
                elif centers[i][k] > 0.95: 
                    forces[i][k] -= 10 * (centers[i][k] - 0.95)
        
        centers += 0.02 * forces
        centers = np.clip(centers, 0.02, 0.98)

    best_sum = -1.0
    best_centers = centers.copy()
    best_radii = np.zeros(n)
    
    # Precompute pair indices
    pair_indices = [(i, j) for i in range(n) for j in range(i + 1, n)]
    num_pairs = len(pair_indices)
    
    # Optimization Loop
    max_iter = 800

        # 1. Solve LP for radii
        c_obj = -np.ones(n) # Maximize sum(r) => Minimize -sum(r)

        A_ub = np.zeros((num_constraints, n))
        b_ub = np.zeros(num_constraints)
        
        idx = 0

        for i, j in pair_indices:
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dist
            idx += 1

        for i in range(n):

            A_ub[idx, i] = 1.0; b_ub[idx] = x; idx += 1

            A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - x; idx += 1

            A_ub[idx, i] = 1.0; b_ub[idx] = y; idx += 1
            # r_i <= 1-y
            A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - y; idx += 1
            
        bounds = [(0, None) for _ in range(n)]

            best_radii = radii.copy()

            if radii[i] + radii[j] >= dist - tol:
                diff = centers[i] - centers[j]
                u = diff / dist
                forces[i] += u
                forces[j] -= u

            # If touching top wall (y=1), push down
            if r >= 1.0 - y - tol: forces[i][1] -= 1.0

        step = 0.05 * (1.0 / (1.0 + it * 0.005))
        centers += step * forces
