import numpy as np
from scipy.optimize import linprog, minimize
import warnings

# Suppress linprog warnings for cleaner output
warnings.filterwarnings('ignore')

def solve_radii(centers):
    """
    Solves the LP to find optimal radii for fixed centers.
    Maximizes sum(r_i) s.t. r_i + r_j <= dist(i,j) and r_i <= dist(i, wall).
    """
    n = centers.shape[0]
    if n == 0:
        return 0.0, np.array([])
        
    # Objective: Maximize sum(r_i) <=> Minimize -sum(r_i)
    c_obj = -np.ones(n)
    
    # Constraints matrix A_ub and vector b_ub
    # Each row represents a constraint: A_ub @ r <= b_ub
    
    constraints_A = []
    constraints_b = []
    
    # 1. Pairwise non-overlap constraints: r_i + r_j <= dist(i, j)
    # Iterate over all unique pairs
    for i in range(n):
        # Optimization: vectorized distance calculation for row i
        dists_i = np.linalg.norm(centers[i] - centers[i+1:], axis=1)
        for k, j in enumerate(range(i + 1, n)):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            constraints_A.append(row)
            constraints_b.append(dists_i[k])
            
    # 2. Wall constraints: r_i <= dist(i, wall)
    for i in range(n):
        x, y = centers[i]
        # r_i <= x
        row = np.zeros(n)
        row[i] = 1.0
        constraints_A.append(row)
        constraints_b.append(x)
        
        # r_i <= 1 - x
        row = np.zeros(n)
        row[i] = 1.0
        constraints_A.append(row)
        constraints_b.append(1.0 - x)
        
        # r_i <= y
        row = np.zeros(n)
        row[i] = 1.0
        constraints_A.append(row)
        constraints_b.append(y)
        
        # r_i <= 1 - y
        row = np.zeros(n)
        row[i] = 1.0
        constraints_A.append(row)
        constraints_b.append(1.0 - y)
        
    A_ub = np.array(constraints_A)
    b_ub = np.array(constraints_b)
    
    # Bounds for radii: r_i >= 0
    bounds = [(0, None) for _ in range(n)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return -res.fun, res.x
        else:
            # Fallback or error handling
            return 0.0, np.zeros(n)
    except Exception:
        return 0.0, np.zeros(n)

def objective_func(centers_flat):
    """
    Objective function for the optimizer.
    Input: flattened array of centers (n, 2)
    Output: negative sum of radii (to be minimized)
    """
    n = 26
    centers = centers_flat.reshape(n, 2)
    
    # Handle invalid centers
    if np.any(np.isnan(centers)) or np.any(np.isinf(centers)):
        return 1e6
        
    sum_r, _ = solve_radii(centers)
    return -sum_r
