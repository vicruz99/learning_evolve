import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    Packs 26 circles in a unit square to maximize the sum of radii.
    Uses a hexagonal initial layout and SLSQP optimization.
    """
    n = 26
    
    # 1. Generate Initial Configuration (Hexagonal Packing)
    # We aim to fit 26 circles. A hexagonal pattern with rows of 5 and 4/5 circles is efficient.
    # Let's try a pattern that sums to 26.
    # Rows: 5, 5, 5, 5, 5, 1? Or 5, 4, 5, 4, 5, 3?
    # Let's try to fit a 6x5 grid area with hexagonal spacing.
    
    # Initial guess parameters
    # We want radius around 0.101. Start with 0.09 to ensure validity.
    r_init = 0.09
    
    centers = []
    
    # Generate rows
    # Pattern: 5, 5, 5, 5, 5, 1 is hard to pack.
    # Pattern: 5, 4, 5, 4, 5, 3 sums to 26.
    row_lengths = [5, 5, 5, 5, 5, 1] # Sum = 26. Wait, 5*5+1 = 26. 
    # But fitting a single circle at the bottom is easy.
    # Let's try to pack 5 rows of 5, and one extra?
    # Actually, 5x5 grid has 25. Adding 1 is hard in grid.
    # Let's use a more balanced hexagonal approach.
    # 6 rows: 5, 4, 5, 4, 5, 3 -> Sum = 26.
    
    row_counts = [5, 4, 5, 4, 5, 3]
    
    # Vertical spacing for hex packing: sqrt(3)/2 * (2r) = sqrt(3)r approx 1.732r
    # But we use spacing = 2*r_init for x, and sqrt(3)*r_init for y shift?
    # In a triangular lattice, distance between rows is sqrt(3) * diameter/2? No.
    # Distance between centers is 2r. Vertical distance is sqrt(3)r.
    
    y_step = np.sqrt(3) * r_init
    
    current_y = r_init
    
    # We need to fit this in [0,1]. 
    # If 6 rows, height approx 2r + 5*sqrt(3)r = r(2 + 8.66) = 10.66r.
    # With r=0.09, height ~ 0.96. Fits.
    
    for i, count in enumerate(row_counts):
        # Shift every other row by r to create hexagonal nesting
        if i % 2 == 1:
            x_start = 2 * r_init # Shifted by one diameter unit? 
            # Actually, in hex packing, shifted row starts at x = 2r if unshifted starts at r.
            # Unshifted: r, 3r, 5r...
            # Shifted: 2r, 4r, 6r...
        else:
            x_start = r_init
            
        x_spacing = 2 * r_init
        
        for j in range(count):
            x = x_start + j * x_spacing
            # Check bounds roughly, if out, cap or adjust? 
            # With r=0.09, 5 circles width: 1 + 4*0.18 = 1.72? No.
            # Width of 5 circles: (5-1)*2r + 2r = 10r = 0.9. Fits.
            # Shifted row of 4 circles: start 2r=0.18. End 2r + 3*2r = 8r = 0.72. Fits.
            centers.append([x, current_y])
            
        current_y += y_step

    # Convert to numpy array
    centers = np.array(centers)
    
    # If we have fewer than 26 due to bad config, pad or adjust?
    # row_counts sums to 26.
    
    radii = np.full(n, r_init)
    
    # Initial state vector: [x0, y0, r0, x1, y1, r1, ...]
    x0 = np.zeros(3 * n)
    for i in range(n):
        x0[3*i] = centers[i, 0]
        x0[3*i + 1] = centers[i, 1]
        x0[3*i + 2] = radii[i]

    # 2. Define Objective and Constraints for Optimizer
    
    # Objective: Minimize -sum(radii)
    def objective(vars):
        r_sum = np.sum(vars[2::3])
        return -r_sum

    # Constraints
    # We need to pass variables to a function. 
    # SLSQP supports constraints defined as functions returning >= 0.
    
    # 1. Boundary constraints
    # x >= r  => x - r >= 0
    # 1 - x >= r => 1 - x - r >= 0
    # Same for y
    
    def constraint_boundary_x(vars):
        x = vars[0::3]
        r = vars[2::3]
        return x - r # x >= r
    
    def constraint_boundary_1_x(vars):
        x = vars[0::3]
        r = vars[2::3]
        return 1.0 - x - r # 1-x >= r

    def constraint_boundary_y(vars):
        y = vars[1::3]
        r = vars[2::3]
        return y - r

    def constraint_boundary_1_y(vars):
        y = vars[1::3]
        r = vars[2::3]
        return 1.0 - y - r

    def constraint_radii_positive(vars):
        r = vars[2::3]
        return r

    # 2. Non-overlap constraints
    # dist(i, j) >= r_i + r_j
    # sqrt((xi-xj)^2 + (yi-yj)^2) - (ri + rj) >= 0
    # This is non-smooth at 0, but we are far from overlap initially.
    # Alternatively, square distance constraint:
    # (xi-xj)^2 + (yi-yj)^2 - (ri+rj)^2 >= 0
    # This is differentiable.
    
    def constraint_no_overlap(vars):
        violations = []
        for i in range(n):
            for j in range(i + 1, n):
                xi = vars[3*i]
                yi = vars[3*i + 1]
                ri = vars[3*i + 2]
                
                xj = vars[3*j]
                yj = vars[3*j + 1]
                rj = vars[3*j + 2]
                
                dist_sq = (xi - xj)**2 + (yi - yj)**2
                r_sum = ri + rj
                
                # Constraint: dist_sq - r_sum^2 >= 0
                violations.append(dist_sq - r_sum**2)
        return np.array(violations)

    # Define constraints list for SLSQP
    cons = [
        {'type': 'ineq', 'fun': constraint_boundary_x},
        {'type': 'ineq', 'fun': constraint_boundary_1_x},
        {'type': 'ineq', 'fun': constraint_boundary_y},
        {'type': 'ineq', 'fun': constraint_boundary_1_y},
        {'type': 'ineq', 'fun': constraint_radii_positive},
        {'type': 'ineq', 'fun': constraint_no_overlap}
    ]

    # Bounds for variables
    # x, y in [0, 1]
    # r >= 0 (handled by constraint, but bounds help)
    bounds = []
    for i in range(n):
        bounds.append((0, 1)) # x
        bounds.append((0, 1)) # y
        bounds.append((0, 0.5)) # r (radius cannot be > 0.5)

    # 3. Run Optimizer
    # SLSQP is good for constrained nonlinear optimization.
    result = minimize(
        objective,
        x0,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 1000, 'ftol': 1e-9, 'disp': False}
    )
    
    # Extract results
    final_vars = result.x
    final_centers = np.zeros((n, 2))
    final_radii = np.zeros(n)
    
    for i in range(n):
        final_centers[i, 0] = final_vars[3*i]
        final_centers[i, 1] = final_vars[3*i + 1]
        final_radii[i] = final_vars[3*i + 2]
        
    sum_radii = np.sum(final_radii)
    
    return final_centers, final_radii, sum_radii
