import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # --- 1. Initialization ---
    # We start with a 5x5 grid (25 circles) and add one more.
    # Grid points at 0.1, 0.3, 0.5, 0.7, 0.9
    grid_points = np.linspace(0.1, 0.9, 5)
    centers_list = []
    
    # Create 25 circles in a grid
    for y in grid_points:
        for x in grid_points:
            centers_list.append([x, y])
            
    # Add 26th circle in a gap (center of a grid cell)
    # (0.2, 0.2) is a gap center between (0.1,0.1), (0.3,0.1), (0.1,0.3), (0.3,0.3)
    centers_list.append([0.2, 0.2])
    
    centers = np.array(centers_list)
    radii = np.ones(n) * 0.05  # Small initial radius to ensure no overlap
    
    # Flatten variables for optimizer: [x1, y1, r1, x2, y2, r2, ...]
    x0 = np.concatenate([centers.flatten(), radii])
    
    # --- 2. Optimization Function and Constraints ---
    def objective(vars):
        # We want to maximize sum of radii, so minimize negative sum
        r = vars[2::3]  # Extract radii (every 3rd element starting from index 2)
        return -np.sum(r)

    def constraint_boundary(vars):
        constraints = []
        for i in range(n):
            x = vars[3*i]
            y = vars[3*i + 1]
            r = vars[3*i + 2]
            # x - r >= 0  => r - x <= 0
            # 1 - x - r >= 0 => x + r - 1 <= 0
            # y - r >= 0
            # 1 - y - r >= 0
            constraints.append(r - x)      # Should be <= 0
            constraints.append(x + r - 1)  # Should be <= 0
            constraints.append(r - y)      # Should be <= 0
            constraints.append(y + r - 1)  # Should be <= 0
        return constraints

    def constraint_overlap(vars):
        constraints = []
        for i in range(n):
            for j in range(i + 1, n):
                x1, y1 = vars[3*i], vars[3*i + 1]
                x2, y2 = vars[3*j], vars[3*j + 1]
                r1, r2 = vars[3*i + 2], vars[3*j + 2]
                
                dist_sq = (x1 - x2)**2 + (y1 - y2)**2
                sum_r_sq = (r1 + r2)**2
                
                # dist_sq >= sum_r_sq  => sum_r_sq - dist_sq <= 0
                constraints.append(sum_r_sq - dist_sq)
        return constraints

    # --- 3. Optimization Execution ---
    
    # Helper to build scipy constraint objects
    def make_ineq_constraint(fun, n_constraints):
        return {
            'type': 'ineq',
            'fun': lambda vars: [c for c in fun(vars)], # Returns list of values, all must be >= 0? 
            # Wait, scipy 'ineq' means fun(vars) >= 0.
            # My functions return values that should be <= 0.
            # So I need to return -fun(vars).
            'fun': lambda vars: [-c for c in fun(vars)],
            'jac': '2-point' # Approximate jacobian
        }

    # Actually, let's define constraints properly for SLSQP
    # Inequality constraints must be >= 0.
    # My boundary: x - r >= 0 -> fun = x - r
    # My overlap: dist^2 - (r1+r2)^2 >= 0 -> fun = dist^2 - (r1+r2)^2
    
    bnds = [(0.0, 1.0) for _ in range(2*n)] + [(0.0, 1.0) for _ in range(n)]
    
    # Boundary constraints: x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
    cons_boundary = {
        'type': 'ineq',
        'fun': lambda vars: np.array([
            vars[3*i] - vars[3*i+2] for i in range(n) # x - r
        ] + [
            1.0 - vars[3*i] - vars[3*i+2] for i in range(n) # 1 - x - r
        ] + [
            vars[3*i+1] - vars[3*i+2] for i in range(n) # y - r
        ] + [
            1.0 - vars[3*i+1] - vars[3*i+2] for i in range(n) # 1 - y - r
        ]),
        'jac': '2-point'
    }
    
    # Overlap constraints: dist^2 - (r1+r2)^2 >= 0
    cons_overlap = {
        'type': 'ineq',
        'fun': lambda vars: np.array([
            (vars[3*i] - vars[3*j])**2 + (vars[3*i+1] - vars[3*j+1])**2 - (vars[3*i+2] + vars[3*j+2])**2
            for i in range(n) for j in range(i+1, n)
        ]),
        'jac': '2-point'
    }
    
    constraints = [cons_boundary, cons_overlap]
    
    # Run optimization
    # We use a two-step approach for robustness
    # Step 1: Coarse optimization
    res1 = minimize(objective, x0, method='SLSQP', bounds=bnds, constraints=constraints, 
                    options={'ftol': 1e-6, 'maxiter': 500, 'disp': False})
    
    # Step 2: Fine optimization from result of Step 1
    res2 = minimize(objective, res1.x, method='SLSQP', bounds=bnds, constraints=constraints, 
                    options={'ftol': 1e-9, 'maxiter': 1000, 'disp': False})
    
    # Extract results
    final_vars = res2.x
    final_centers = np.array([[final_vars[3*i], final_vars[3*i+1]] for i in range(n)])
    final_radii = np.array([final_vars[3*i+2] for i in range(n)])
    total_radius = np.sum(final_radii)
    
    return (final_centers, final_radii, total_radius)
