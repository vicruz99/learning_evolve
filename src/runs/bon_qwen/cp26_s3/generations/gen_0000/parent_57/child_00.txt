import numpy as np
from scipy.optimize import minimize
import math


def validate_packing_check(centers, radii, n):
    if np.isnan(centers).any() or np.isnan(radii).any():
        return False
    for i in range(n):
        if radii[i] < -1e-12:
            return False
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-10 or x + r > 1 + 1e-10 or y - r < -1e-10 or y + r > 1 + 1e-10:
            return False
    for i in range(n):
        for j in range(i + 1, n):
            dist = math.sqrt(sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-10:
                return False
    return True


def generate_initial_hex(n, seed):
    np.random.seed(seed)
    centers = np.zeros((n, 2))
    idx = 0
    
    base_spacing = 0.19
    row_spacing = base_spacing * math.sqrt(3) / 2
    
    for row in range(8):
        for col in range(6):
            if idx >= n:
                break
            x = 0.06 + col * base_spacing + (row % 2) * base_spacing / 2
            y = 0.06 + row * row_spacing
            if x < 0.95 and y < 0.95:
                centers[idx] = [x, y]
                idx += 1
        if idx >= n:
            break
    
    # Add small random perturbation
    centers += np.random.randn(n, 2) * 0.01
    centers = np.clip(centers, 0.02, 0.98)
    
    return centers


def run_packing():
    n = 26
    
    def objective(params):
        radii = params[2 * n:]
        return -np.sum(radii)
    
    def make_boundary_constraints(n_circles, centers_shape_idx):
        constraints = []
        for i in range(n_circles):
            # x - r >= 0
            c_idx_x = 2 * i
            r_idx = 2 * n_circles + i
            constraints.append({
                'type': 'ineq',
                'fun': lambda p, ci=c_idx_x, ri=r_idx: p[ci] - p[ri]
            })
            # 1 - x - r >= 0
            constraints.append({
                'type': 'ineq',
                'fun': lambda p, ci=c_idx_x, ri=r_idx: 1 - p[ci] - p[ri]
            })
            # y - r >= 0
            c_idx_y = 2 * i + 1
            constraints.append({
                'type': 'ineq',
                'fun': lambda p, ci=c_idx_y, ri=r_idx: p[ci] - p[ri]
            })
            # 1 - y - r >= 0
            constraints.append({
                'type': 'ineq',
                'fun': lambda p, ci=c_idx_y, ri=r_idx: 1 - p[ci] - p[ri]
            })
        return constraints
    
    def make_overlap_constraints(n_circles):
        constraints = []
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                c_i_x = 2 * i
                c_i_y = 2 * i + 1
                c_j_x = 2 * j
                c_j_y = 2 * j + 1
                r_i = 2 * n_circles + i
                r_j = 2 * n_circles + j
                
                def overlap_fun(p, ci_x=c_i_x, ci_y=c_i_y, cj_x=c_j_x, cj_y=c_j_y, ri=r_i, rj=r_j):
                    dx = p[ci_x] - p[cj_x]
                    dy = p[ci_y] - p[cj_y]
                    dist = math.sqrt(dx * dx + dy * dy)
                    return dist - p[ri] - p[rj]
                
                constraints.append({
                    'type': 'ineq',
                    'fun': overlap_fun
                })
        return constraints
    
    all_constraints = make_boundary_constraints(n, 2 * n) + make_overlap_constraints(n)
    
    bounds = [(0.001, 0.999)] * (2 * n) + [(0.001, 0.5)] * n
    
    best_centers = None
    best_radii = None
    best_sum = 0
    
    # Try many different initial configurations
    for seed in range(80):
        centers_init = generate_initial_hex(n, seed)
        radii_init = np.full(n, 0.06 + seed * 0.001)
        params_init = np.concatenate([centers_init.flatten(), radii_init])
        
        try:
            result = minimize(
                objective,
                params_init,
                bounds=bounds,
                constraints=all_constraints,
                method='SLSQP',
                options={'maxiter': 2000, 'ftol': 1e-14, 'disp': False}
            )
            
            centers_opt = result.x[:2 * n].reshape(n, 2)
            radii_opt = result.x[2 * n:]
            radii_opt = np.maximum(radii_opt, 0.0001)
            
            # Post-processing: try to grow radii where possible
            for _ in range(50):
                for i in range(n):
                    x, y = centers_opt[i]
                    max_r = min(x, 1 - x, y, 1 - y)
                    for j in range(n):
                        if i != j:
                            dist = math.sqrt(
                                (centers_opt[i, 0] - centers_opt[j, 0]) ** 2 +
                                (centers_opt[i, 1] - centers_opt[j, 1]) ** 2
                            )
                            max_r = min(max_r, dist - radii_opt[j])
                    max_r = max(max_r, radii_opt[i])
                    if max_r > radii_opt[i] + 1e-12:
                        radii_opt[i] = max_r
            
            # Final validation
            valid = True
            for i in range(n):
                x, y = centers_opt[i]
                r = radii_opt[i]
                if r < 0:
                    valid = False
                    break
                if x - r < -1e-10 or x + r > 1 + 1e-10 or y - r < -1e-10 or y + r > 1 + 1e-10:
                    valid = False
                    break
            if valid:
                for i in range(n):
                    for j in range(i + 1, n):
                        dist = math.sqrt(
                            (centers_opt[i, 0] - centers_opt[j, 0]) ** 2 +
                            (centers_opt[i, 1] - centers_opt[j, 1]) ** 2
                        )
                        if dist < radii_opt[i] + radii_opt[j] - 1e-10:
                            valid = False
                            break
                    if not valid:
                        break
            
            if valid:
                s = np.sum(radii_opt)
                if s > best_sum:
                    best_sum = s
                    best_centers = centers_opt.copy()
                    best_radii = radii_opt.copy()
                    if best_sum > 2.6:
                        print(f"Seed {seed}: sum = {best_sum:.6f}")
        except Exception as e:
            pass
    
    if best_centers is None:
        best_centers = generate_initial_hex(n, 0)
        best_radii = np.full(n, 0.05)
        best_sum = np.sum(best_radii)
    
    # Ensure non-negative radii
    best_radii = np.maximum(best_radii, 0)
    
    return best_centers, best_radii, float(best_sum)
