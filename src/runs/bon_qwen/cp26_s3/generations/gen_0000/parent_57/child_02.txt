def compute_max_radius(centers, radii, i, n):
    """Compute the maximum possible radius for circle i given current positions."""
    min_dist = float('inf')
    
    # Distance to boundaries
    min_dist = min(min_dist, 
                   centers[i, 0], 
                   1 - centers[i, 0], 
                   centers[i, 1], 
                   1 - centers[i, 1])
    
    # Distance to other circles (minus their radius)
    for j in range(n):
        if i != j:
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            min_dist = min(min_dist, dist - radii[j])
    
    return max(0.0, min_dist)
