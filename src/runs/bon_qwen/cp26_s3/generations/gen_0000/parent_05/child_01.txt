import numpy as np


def compute_repulsion_forces(centers, radii, n, force_scale):
    """Compute repulsion forces between circles and from boundaries."""
    forces = np.zeros((n, 2))
    eps = 0.0005

    for i in range(n):
        # Boundary repulsion forces
        for d in range(2):
            if centers[i, d] < radii[i]:
                gap = radii[i] - centers[i, d]
                forces[i, d] += gap * force_scale / (gap + eps)
            if centers[i, d] > 1 - radii[i]:
                gap = centers[i, d] - (1 - radii[i])
                forces[i, d] -= gap * force_scale / (gap + eps)

        # Pairwise repulsion forces
        for j in range(i + 1, n):
            diff = centers[i] - centers[j]
            dist_sq = np.sum(diff ** 2)
            dist = np.sqrt(dist_sq)
            if dist < 1e-12:
                dist = 1e-12
                diff = np.array([np.random.randn() * 0.0001, np.random.randn() * 0.0001])
            min_dist = radii[i] + radii[j]
            if dist < min_dist:
                overlap = min_dist - dist
                force_mag = overlap * force_scale / dist
                forces[i] += force_mag * diff
                forces[j] -= force_mag * diff

    return forces


def project_to_valid(centers, radii, n):
    """Project centers to valid region [r_i, 1-r_i] in each dimension."""
    for i in range(n):
        centers[i, 0] = max(radii[i], min(1 - radii[i], centers[i, 0]))
        centers[i, 1] = max(radii[i], min(1 - radii[i], centers[i, 1]))
