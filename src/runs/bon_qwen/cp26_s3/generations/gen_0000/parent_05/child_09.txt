import numpy as np
from scipy.optimize import minimize
import random

def run_packing():
    # Target number of circles
    n = 26
    best_sum = -1.0
    best_centers = None
    best_radii = None

    # Helper for distance calculation
    def dist(p1, p2):
        return np.sqrt(np.sum((p1 - p2)**2))

    # Generate a hexagonal lattice initialization
    def generate_hex_lattice(n):
        coords = []
        r_est = 0.09  # Estimated radius for hex packing
        
        # Generate a grid of points
        rows = 8
        cols = 8
        dx = r_est * 2.0
        dy = r_est * np.sqrt(3.0)
        
        points = []
        for i in range(rows):
            for j in range(cols):
                x = j * dx + (0.5 if i % 2 else 0.0) * dx + 0.05
                y = i * dy + 0.05
                if x < 1.0 and y < 1.0:
                    points.append([x, y])
        
        # Shuffle and take first n
        random.shuffle(points)
        selected = points[:n]
        
        # Add random perturbation
        centers = np.array(selected) + np.random.uniform(-0.01, 0.01, (n, 2))
        # Clip to valid range
        centers = np.clip(centers, 0.02, 0.98)
        
        return centers, np.full(n, 0.05) # Start with small radii

    # Optimization function
    def solve_from_init(seed):
        random.seed(seed)
        np.random.seed(seed)
        
        centers_init, radii_init = generate_hex_lattice(n)
        # Flatten for optimizer: [x1, y1, r1, x2, y2, r2, ...]
        x0 = np.concatenate([centers_init.ravel(), radii_init])
        
        # Objective: Minimize negative sum of radii
        def objective(x):
            radii = x[2::3]
            return -np.sum(radii)

        # Constraints
        constraints = []
        bounds = []
        
        # Bounds for x, y, r
        for i in range(n):
            # x, y in [0, 1]
            bounds.append((0, 1)) # x
            bounds.append((0, 1)) # y
            bounds.append((0, 0.5)) # r <= 0.5
        
        # Boundary constraints: r <= x <= 1-r  =>  x - r >= 0, x + r <= 1
        for i in range(n):
            idx_x = i * 3
            idx_y = i * 3 + 1
            idx_r = i * 3 + 2
            
            # Left boundary
            constraints.append({
                'type': 'ineq',
                'fun': lambda x, i=i: x[idx_x] - x[idx_r]
            })
            # Right boundary
            constraints.append({
                'type': 'ineq',
                'fun': lambda x, i=i: 1 - x[idx_x] - x[idx_r]
            })
            # Bottom boundary
            constraints.append({
                'type': 'ineq',
                'fun': lambda x, i=i: x[idx_y] - x[idx_r]
            })
            # Top boundary
            constraints.append({
                'type': 'ineq',
                'fun': lambda x, i=i: 1 - x[idx_y] - x[idx_r]
            })

        # Non-overlap constraints: dist(i,j) >= r_i + r_j
        for i in range(n):
            for j in range(i + 1, n):
                idx_xi, idx_yi, idx_ri = i * 3, i * 3 + 1, i * 3 + 2
                idx_xj, idx_yj, idx_rj = j * 3, j * 3 + 1, j * 3 + 2
                
                def constraint_func(x, i=i, j=j):
                    xi, yi, ri = x[idx_xi], x[idx_yi], x[idx_ri]
                    xj, yj, rj = x[idx_xj], x[idx_yj], x[idx_rj]
                    d = np.sqrt((xi - xj)**2 + (yi - yj)**2)
                    return d - (ri + rj)
                
                constraints.append({'type': 'ineq', 'fun': constraint_func})

        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints,
                           options={'maxiter': 1000, 'ftol': 1e-6})
            
            if res.success:
                centers = res.x[:2*n].reshape(n, 2)
                radii = res.x[2*n:]
                sum_r = np.sum(radii)
                return centers, radii, sum_r
        except Exception:
            pass
            
        return None, None, -1.0

    # Try multiple seeds to find a good global optimum
    for seed in range(50):
        c, r, s = solve_from_init(seed)
        if s > best_sum:
            best_sum = s
            best_centers = c
            best_radii = r

    # If optimizer failed to find a valid one better than baseline, 
    # we might return a trivial valid one, but the optimizer usually finds > 0.
    if best_sum < 0:
        # Fallback to a simple grid if optimization completely fails
        best_centers = np.zeros((26, 2))
        best_radii = np.zeros(26)
        for i in range(26):
            row = i // 5
            col = i % 5
            best_centers[i] = [0.1 + col * 0.2, 0.1 + row * 0.2]
            best_radii[i] = 0.1
        best_sum = np.sum(best_radii)

    return best_centers, best_radii, float(best_sum)
