

Here's my strategy for solving this circle packing problem:

<strategy>
1. **Start with a hexagonal grid**: For 26 circles, use a 5×5+1 arrangement with hexagonal spacing, which is near-optimal for dense packings.

2. **Force-based expansion simulation**: Gradually increase radii while applying repulsive forces between overlapping circles and boundary-pushing forces. This mimics the "circle packing by expansion" algorithm.

3. **Linear Programming refinement**: After force simulation converges, solve an LP to compute optimal radii given fixed centers. The LP maximizes Σrᵢ subject to rᵢ ≤ boundary distances and rᵢ + rⱼ ≤ ||cᵢ - cⱼ||.

4. **Position refinement**: After LP gives radii, adjust positions based on tight constraints (pairs where rᵢ + rⱼ ≈ distance), then re-solve LP. Repeat for several iterations.

5. **Multiple restarts**: Try different initial grid spacings and perturbations to avoid local minima.

6. **Final validation**: Clip small violations and return the best valid solution.

Key insight: maximizing sum of radii favors unequal circles - larger ones near corners/boundaries where they have more space, smaller ones filling gaps. The LP step is crucial because it optimally allocates radii for fixed centers, unlike naive equal-radius approaches.
</strategy>

import numpy as np
import math
from scipy.optimize import linprog


def run_packing():
    n = 26
    best_sum = 0
    best_centers = None
    best_radii = None

    for trial in range(15):
        np.random.seed(42 + trial)

        # Hexagonal initial grid
        centers = np.zeros((n, 2))
        idx = 0
        base_spacing = 0.14 + 0.02 * np.random.rand()
        offset_x = 0.08 + 0.05 * np.random.rand()
        offset_y = 0.08 + 0.05 * np.random.rand()

        for row in range(6):
            for col in range(5):
                if idx < n:
                    x = col * base_spacing * math.sqrt(3) + offset_x
                    y = row * base_spacing * 1.5 + offset_y
                    if row % 2 == 1:
                        x += base_spacing * math.sqrt(3) / 2
                    centers[idx] = [x, y]
                    idx += 1

        # Add random perturbation
        centers += np.random.randn(n, 2) * 0.02
        centers = np.clip(centers, 0.05, 0.95)

        radii = np.full(n, 0.04)

        # Force-based expansion
        lr = 0.008
        growth = 0.00015

        for iteration in range(30000):
            forces = np.zeros((n, 2))

            for i in range(n):
                for j in range(i + 1, n):
                    diff = centers[i] - centers[j]
                    dist = np.linalg.norm(diff)
                    if dist > 1e-10:
                        desired = radii[i] + radii[j]
                        if dist < desired:
                            force_mag = (desired - dist) / dist * 2.0
                            forces[i] += diff * force_mag
                            forces[j] -= diff * force_mag

            for i in range(n):
                x, y = centers[i]
                r = radii[i]
                if x - r < 0:
                    forces[i, 0] += max(0, r - x) * 15
                if x + r > 1:
                    forces[i, 0] -= max(0, x + r - 1) * 15
                if y - r < 0:
                    forces[i, 1] += max(0, r - y) * 15
                if y + r > 1:
                    forces[i, 1] -= max(0, y + r - 1) * 15

            centers += forces * lr
            centers = np.clip(centers, 0.002, 0.998)
            radii += growth

            if iteration % 5000 == 0:
                lr *= 0.92
                growth *= 0.96

        # LP to compute optimal radii for fixed centers
        c_obj = -np.ones(n)
        A_ub_list = []
        b_ub_list = []

        for i in range(n):
            row = np.zeros(n)
            row[i] = 1
            A_ub_list.append(row.copy())
            b_ub_list.append(centers[i, 0])
            A_ub_list.append(row.copy())
            b_ub_list.append(centers[i, 1])
            A_ub_list.append(row.copy())
            b_ub_list.append(1 - centers[i, 0])
            A_ub_list.append(row.copy())
            b_ub_list.append(1 - centers[i, 1])

        for i in range(n):
            for j in range(i + 1, n):
                dist = np.linalg.norm(centers[i] - centers[j])
                row = np.zeros(n)
                row[i] = 1
                row[j] = 1
                A_ub_list.append(row)
                b_ub_list.append(dist)

        A_ub = np.array(A_ub_list)
        b_ub = np.array(b_ub_list)
        bounds = [(0, None) for _ in range(n)]

        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')

        if result.success:
            radii_lp = result.x
            current_sum = np.sum(radii_lp)

            # Refine positions based on tight constraints
            for refine_iter in range(50):
                move = np.zeros((n, 2))
                for i in range(n):
                    for j in range(i + 1, n):
                        dist = np.linalg.norm(centers[i] - centers[j])
                        target = radii_lp[i] + radii_lp[j]
                        if abs(dist - target) < 0.05 and dist < target + 0.01:
                            diff = centers[i] - centers[j]
                            if dist > 1e-10:
                                push = (target - dist) * 0.02 / dist
                                move[i] += diff * push
                                move[j] -= diff * push

                centers += move
                centers = np.clip(centers, 0.002, 0.998)

                # Re-solve LP
                for idx_j in range(len(b_ub_list)):
                    if idx_j < 4 * n:
                        b_ub[idx_j] = b_ub_list[idx_j]
                    else:
                        pair_idx = (idx_j - 4 * n)
                        i = pair_idx // n
                        j = pair_idx % n
                        if i < j:
                            b_ub[idx_j] = np.linalg.norm(centers[i] - centers[j])

                result2 = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
                if result2.success:
                    radii_lp = result2.x
                    current_sum = np.sum(radii_lp)

            # Final cleanup: ensure validity
            valid, final_radii = validate_and_fix(centers, radii_lp, n)
            if valid and current_sum > best_sum:
                best_sum = current_sum
                best_centers = centers.copy()
                best_radii = final_radii.copy()

    if best_centers is None:
        # Fallback
        best_centers = np.random.rand(n, 2) * 0.8 + 0.1
        best_radii = np.full(n, 0.02)
        best_sum = np.sum(best_radii)

    return best_centers, best_radii, best_sum


def validate_and_fix(centers, radii, n):
    """Validate and fix small violations."""
    radii_fixed = radii.copy()
    valid = True

    # Fix boundary violations
    for i in range(n):
        x, y = centers[i]
        r = radii_fixed[i]
        max_r = min(x, y, 1 - x, 1 - y)
        if r > max_r:
            radii_fixed[i] = max(0, max_r - 1e-12)

    # Fix overlap violations
    for iteration in range(20):
        changed = False
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.linalg.norm(centers[i] - centers[j])
                r_sum = radii_fixed[i] + radii_fixed[j]
                if r_sum > dist - 1e-12:
                    # Reduce radii proportionally
                    excess = r_sum - dist + 1e-12
                    if radii_fixed[i] > 0 and radii_fixed[j] > 0:
                        factor = dist / r_sum if r_sum > 0 else 0
                        radii_fixed[i] *= factor
                        radii_fixed[j] *= factor
                    elif radii_fixed[i] > 0:
                        radii_fixed[i] = dist
                    elif radii_fixed[j] > 0:
                        radii_fixed[j] = dist
                    changed = True

        if not changed:
            break

    # Final validation
    for i in range(n):
        x, y = centers[i]
        r = radii_fixed[i]
        if x - r < -1e-10 or x + r > 1 + 1e-10 or y - r < -1e-10 or y + r > 1 + 1e-10:
            valid = False
            break
        if r < -1e-10:
            radii_fixed[i] = 0

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            if dist < radii_fixed[i] + radii_fixed[j] - 1e-10:
                valid = False
                break

    return valid, radii_fixed
