import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Pack 26 circles in a unit square to maximize the sum of radii.
    Uses a constraint-based optimization approach with multiple restarts.
    """
    n = 26
    best_sum_radii = -1.0
    best_centers = None
    best_radii = None

    # Number of optimization attempts
    num_restarts = 15

    for _ in range(num_restarts):
        # Generate initial guess
        # Mix of grid and random perturbations to explore space
        if np.random.rand() > 0.5:
            # Hexagonal-like grid initialization
            centers_init = np.zeros((n, 2))
            idx = 0
            # Try to fit in a pattern, e.g., 5 rows
            rows = 6
            cols_per_row = [5, 5, 5, 5, 4, 2] # Sum = 26
            # Adjust to fit in square roughly
            # Simple random scatter inside [0.2, 0.8]
            centers_init = np.random.uniform(0.15, 0.85, size=(n, 2))
        else:
            # Random initialization
            centers_init = np.random.uniform(0.1, 0.9, size=(n, 2))
        
        # Initial radii: small valid value
        # Max possible radius for 26 circles is < 0.12 roughly. 
        # Start with 0.05 to be safe.
        radii_init = np.full(n, 0.05)

        # Flatten variables: [x0, y0, r0, x1, y1, r1, ...]
        x0 = np.zeros(3 * n)
        for i in range(n):
            x0[3 * i] = centers_init[i, 0]
            x0[3 * i + 1] = centers_init[i, 1]
            x0[3 * i + 2] = radii_init[i]

        # Bounds: x, y in [0, 1], r in [0, 0.5]
        bounds = []
        for _ in range(n):
            bounds.append((0, 1)) # x
            bounds.append((0, 1)) # y
            bounds.append((0, 0.5)) # r

        # Define objective function: minimize -sum(r)
        def objective(vars_vec):
            r = vars_vec[2::3]
            return -np.sum(r)

        # Define constraints
        # We use a function that returns an array of constraint values >= 0
        def constraint_func(vars_vec):
            constraints = []
            
            # Extract x, y, r
            x = vars_vec[0::3]
            y = vars_vec[1::3]
            r = vars_vec[2::3]

            # Boundary constraints: x >= r, 1-x >= r => x-r >= 0, 1-x-r >= 0
            # Same for y
            constraints.extend(x - r)
            constraints.extend(1 - x - r)
            constraints.extend(y - r)
            constraints.extend(1 - y - r)

            # Non-overlap constraints: dist_ij >= r_i + r_j
            # dist^2 >= (r_i + r_j)^2
            # x_i - x_j, y_i - y_j
            # We compute pairwise differences efficiently
            
            # Using broadcasting for all pairs might be memory heavy for large n, 
            # but for n=26, 26x26 is small.
            # However, we only need upper triangle.
            
            # Create matrices
            # X shape (1, n), X.T shape (n, 1)
            # Diff matrix shape (n, n)
            dx = x[:, None] - x[None, :]
            dy = y[:, None] - y[None, :]
            dr_sum = r[:, None] + r[None, :]
            
            dist_sq = dx**2 + dy**2
            min_dist_sq = dr_sum**2
            
            # We need dist_sq >= min_dist_sq for i != j
            # Set diagonal to 0 or large positive to ignore self-constraint?
            # Actually constraint is dist >= r+r, for i=j dist=0, r+r=2r > 0.
            # So self constraint 0 >= 4r^2 would fail.
            # We must exclude i=j.
            
            # Flatten upper triangle (excluding diagonal)
            # indices where i < j
            i_idx, j_idx = np.triu_indices(n, k=1)
            constraints.extend(dist_sq[i_idx, j_idx] - min_dist_sq[i_idx, j_idx])

            return np.array(constraints)

        # SLSQP constraint format
        cons = {'type': 'ineq', 'fun': constraint_func}

        # Run optimization
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, 
                           options={'maxiter': 500, 'ftol': 1e-9, 'disp': False})
            
            if res.success or res.fun < -best_sum_radii: # res.fun is negative sum
                current_sum = -res.fun
                if current_sum > best_sum_radii:
                    # Extract solution
                    res_x = res.x[0::3]
                    res_y = res.x[1::3]
                    res_r = res.x[2::3]
                    
                    # Verify validity explicitly before accepting
                    # (Solver might slightly violate constraints due to tolerance)
                    valid = True
                    
                    # Check bounds
                    for i in range(n):
                        if res_x[i] < -1e-9 or res_x[i] > 1 + 1e-9 or \
                           res_y[i] < -1e-9 or res_y[i] > 1 + 1e-9 or \
                           res_r[i] < -1e-9:
                            valid = False
                            break
                    
                    if valid:
                        # Check overlaps and boundaries strictly
                        for i in range(n):
                            if res_x[i] - res_r[i] < -1e-7 or res_x[i] + res_r[i] > 1 + 1e-7 or \
                               res_y[i] - res_r[i] < -1e-7 or res_y[i] + res_r[i] > 1 + 1e-7:
                                valid = False
                                break
                            
                        if valid:
                            for j in range(i + 1, n):
                                dist = np.sqrt((res_x[i]-res_x[j])**2 + (res_y[i]-res_y[j])**2)
                                if dist < res_r[i] + res_r[j] - 1e-7:
                                    valid = False
                                    break
                            
                    if valid:
                        best_sum_radii = current_sum
                        best_centers = np.column_stack((res_x, res_y))
                        best_radii = res_r

        except Exception as e:
            continue

    # If optimization failed to find valid or we want to ensure a valid fallback
    # We can return the best found. If best_centers is None, fallback to grid.
    if best_centers is None:
        # Fallback: 5x5 grid + 1
        # This is not optimal but valid
        centers_fallback = np.zeros((26, 2))
        radii_fallback = np.zeros(26)
        # 5x5 grid

        idx = 0
        for i in range(5):

                centers_fallback[idx] = [coords[i], coords[j]]
                radii_fallback[idx] = 0.1
                idx += 1
        # 26th circle in a gap? 
        # Gap at (0.2, 0.2) relative to grid? 
        # Center (0.2, 0.2) is distance sqrt(0.1^2+0.1^2) ~ 0.141 from (0.1,0.1).
        # Radius can be 0.141 - 0.1 = 0.041
        centers_fallback[25] = [0.2, 0.2]
        radii_fallback[25] = 0.04
        best_centers = centers_fallback
        best_radii = radii_fallback
        best_sum_radii = np.sum(best_radii)

    return best_centers, best_radii, float(best_sum_radii)
