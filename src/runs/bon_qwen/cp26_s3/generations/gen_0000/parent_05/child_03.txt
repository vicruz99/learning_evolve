import numpy as np
from scipy.optimize import minimize

def compute_loss_and_grad(params, n, P):
    c = params[:2*n].reshape(n, 2)
    r = params[2*n:]
    loss = -np.sum(r)
    grad_c = np.zeros_like(c)
    grad_r = -np.ones_like(r)
    
    # Boundary constraints penalty
    for i in range(n):
        dists = np.array([c[i,0], 1-c[i,0], c[i,1], 1-c[i,1]])
        overlaps = np.maximum(0, r[i] - dists)
        loss += P * np.sum(overlaps**2)
        grad_r[i] += 2 * P * np.sum(overlaps)
        
    # Inter-circle constraints penalty
    for i in range(n):
        for j in range(i+1, n):
            diff = c[i] - c[j]
            d = np.sqrt(np.sum(diff**2))
            if d < 1e-9: d = 1e-9
            overlap = np.maximum(0, r[i] + r[j] - d)
            loss += P * overlap**2
            grad_r[i] += 2 * P * overlap
            grad_r[j] += 2 * P * overlap
            force = 2 * P * overlap / d
            grad_c[i] -= force * diff
            grad_c[j] += force * diff
            
    grad = np.concatenate([grad_c.flatten(), grad_r])
    return loss, grad

def run_packing():
    n = 26
    P = 50000.0
    best_loss = float('inf')
    best_params = None
    
    for seed in range(50):
        np.random.seed(seed)
        
        # Initialization: mix of hexagonal grids and random placements
        if seed % 5 == 0:
            centers = []
            for i in range(8):
                for j in range(8):
                    if len(centers) >= n:
                        break
                    x = (2*j + 0.5 + (i % 2)*0.5) * 0.12
                    y = i * 0.12 * np.sqrt(3)
                    centers.append([x, y])
                if len(centers) >= n:
                    break
            centers = np.array(centers[:n])
            cx, cy = centers.mean(axis=0)
            centers -= [cx, cy]
            scale = 0.8 / np.max(np.abs(centers))
            centers *= scale
            centers += 0.5
            # Add jitter to break symmetry
            centers += np.random.uniform(-0.01, 0.01, centers.shape)
        else:
            centers = np.random.uniform(0.15, 0.85, (n, 2))
            
        radii = np.full(n, 0.08)
        params0 = np.concatenate([centers.flatten(), radii])
        bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n
        
        res = minimize(compute_loss_and_grad, params0, args=(n, P), 
                       method='L-BFGS-B', jac=True, bounds=bounds, 
                       options={'maxiter': 3000, 'ftol': 1e-12, 'gtol': 1e-8})
                       
        if res.fun < best_loss:
            best_loss = res.fun
            best_params = res.x.copy()
            
    c_best = best_params[:2*n].reshape(n, 2)
    r_best = best_params[2*n:]
    
    # Clip to bounds strictly
    c_best = np.clip(c_best, 0.0, 1.0)
    r_best = np.clip(r_best, 0.0, 0.5)
    
    # Post-process to guarantee strict feasibility
    for _ in range(100):
        new_r = r_best.copy()
        changed = False
        for i in range(n):
            limit = min(c_best[i,0], 1-c_best[i,0], c_best[i,1], 1-c_best[i,1])
            for j in range(n):
                if i != j:
                    d = np.sqrt(np.sum((c_best[i] - c_best[j])**2))
                    limit = min(limit, d - r_best[j])
            if limit < r_best[i] - 1e-12:
                new_r[i] = max(0.0, limit)
                changed = True
        if not changed:
            break
        r_best = new_r
        
    return c_best, r_best, np.sum(r_best)
