import numpy as np
import math
from scipy.optimize import minimize

def penalty_function(centers_flat, radii):
    centers = centers_flat.reshape(-1, 2)
    n = len(radii)
    p = 0.0
    
    # Boundary violations
    x, y = centers[:, 0], centers[:, 1]
    r = radii
    
    p += np.sum(np.maximum(0, r - x)**2) * 10.0
    p += np.sum(np.maximum(0, x - (1 - r))**2) * 10.0
    p += np.sum(np.maximum(0, r - y)**2) * 10.0
    p += np.sum(np.maximum(0, y - (1 - r))**2) * 10.0
    
    # Overlap violations
    c_diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(c_diff**2, axis=2))
    req_matrix = radii[:, np.newaxis] + radii[np.newaxis, :]
    viol = np.maximum(0, req_matrix - dists)
    
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    p += np.sum(viol[mask]**2) * 10.0
    
    return p

def optimize_positions(centers, radii, max_iter=100):
    centers_flat = centers.flatten()
    bounds = [(0.0, 1.0)] * centers_flat.shape[0]
    
    best_pen = float('inf')
    best_flat = centers_flat
    
    for _ in range(2):
        res = minimize(penalty_function, centers_flat, args=(radii,), 
                       method='L-BFGS-B', bounds=bounds, 
                       options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-10})
        if res.fun < best_pen:
            best_pen = res.fun
            best_flat = res.x
            centers_flat = best_flat
            
    return best_flat.reshape(-1, 2), best_pen
