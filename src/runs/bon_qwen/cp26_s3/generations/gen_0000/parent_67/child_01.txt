import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def neg_sum_radii(vars):
    return -np.sum(vars[2::3])

def penalty_objective(vars, n, lam):
    sum_r = np.sum(vars[2::3])
    pen = 0.0
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    # Boundary penalties
    pen += np.sum(np.maximum(0, r - x)**2)
    pen += np.sum(np.maximum(0, r - (1 - x))**2)
    pen += np.sum(np.maximum(0, r - y)**2)
    pen += np.sum(np.maximum(0, r - (1 - y))**2)
    
    # Overlap penalties
    centers = np.column_stack((x, y))
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    overlap = np.maximum(0, r_sum - dists)
    idx = np.triu_indices(n, k=1)
    pen += np.sum(overlap[idx]**2)
    
    return -sum_r + lam * pen

def exact_constraints(vars):
    n = N_CIRCLES
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    vals = []
    # Boundary constraints
    vals.extend(x - r)
    vals.extend(1 - x - r)
    vals.extend(y - r)
    vals.extend(1 - y - r)
    
    # Pairwise non-overlap constraints
    centers = np.column_stack((x, y))
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    idx = np.triu_indices(n, k=1)
    vals.extend(dists[idx] - r_sum[idx])
    return np.array(vals)

def run_packing():

    best_sum = 0.0
    best_centers = np.zeros((n, 2))
    best_radii = np.zeros(n)

    bounds = [(0, 1)]*n + [(0, 1)]*n + [(0, 0.5)]*n

        init_x = np.zeros(n)
        init_y = np.zeros(n)

                if k >= n: break
                init_x[k] = 0.1 + j * 0.17 + (0.085 if i % 2 == 1 else 0)
                init_y[k] = 0.1 + i * 0.15
                k += 1
            if k >= n: break

            k += 1

        # Stage 2: Precise refinement with exact constraints
        res2 = minimize(neg_sum_radii, res.x,

                        options={'maxiter': 1000, 'ftol': 1e-12})
