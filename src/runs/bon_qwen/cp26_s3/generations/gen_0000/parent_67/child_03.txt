import numpy as np
from scipy.optimize import linprog
from scipy.spatial.distance import cdist
import math

def solve_radii_lp(centers):
    """
    Solve the LP: Maximize sum(r) s.t. r_i + r_j <= dist(i,j) and boundary constraints.
    """
    n = centers.shape[0]
    
    # c for linprog minimizes -c^T x, so we set c = -1 to maximize sum(r)
    c = -np.ones(n)
    
    # Constraints: A_ub @ x <= b_ub
    # We will build these lists and convert to arrays
    a_rows = []
    b_rows = []
    
    # 1. Boundary constraints: r_i <= dist to walls
    for i in range(n):
        x, y = centers[i]
        # Max radius based on boundaries: x, 1-x, y, 1-y
        r_max_bound = min(x, 1 - x, y, 1 - y)
        # r_i <= r_max_bound
        row = np.zeros(n)
        row[i] = 1.0
        a_rows.append(row)
        b_rows.append(r_max_bound)
        
    # 2. Pairwise constraints: r_i + r_j <= dist(i, j)
    # We only need to check constraints for circles that are relatively close to avoid a huge matrix,
    # but for N=26, N^2 is small (325 constraints), so we can add all.
    dists = cdist(centers, centers)
    
    for i in range(n):
        for j in range(i + 1, n):
            d = dists[i, j]
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            a_rows.append(row)
            b_rows.append(d)
            
    A_ub = np.array(a_rows)
    b_ub = np.array(b_rows)
    
    # Bounds for r_i: 0 <= r_i
    bounds = [(0, None) for _ in range(n)]
    
    # Solve LP
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return res.fun * -1, res.x
    else:
        return 0.0, np.zeros(n)

                idx += 1
