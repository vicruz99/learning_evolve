import numpy as np
from scipy.optimize import minimize

N = 26

def objective(v):
    """Objective function: maximize sum of radii (minimize negative sum)"""
    return -np.sum(v[2*N:])

def constraints(v):
    """Nonlinear inequality constraints: boundary and pairwise non-overlap"""
    c = v[:2*N].reshape((N, 2))
    r = v[2*N:]
    
    # Boundary constraints: 4 per circle
    b = np.vstack([
        c[:, 0] - r,
        1.0 - c[:, 0] - r,
        c[:, 1] - r,
        1.0 - c[:, 1] - r
    ]).flatten()
    
    # Pairwise non-overlap constraints: N*(N-1)/2
    dx = c[:, 0, None] - c[None, :, 0]
    dy = c[:, 1, None] - c[None, :, 1]
    dr = r[:, None] + r[None, :]
    
    dist_sq = dx**2 + dy**2
    min_dist_sq = dr**2
    
    mask = np.triu(np.ones((N, N), dtype=bool), k=1)
    p = dist_sq[mask] - min_dist_sq[mask]
    
    return np.concatenate([b, p])

def run_packing():
    np.random.seed(42)
    
    # 1. Generate initial hexagonal packing
    r_init = 0.075
    centers = []
    idx = 0
    y = r_init
    shift = 0.0
    while idx < N:
        x = r_init + shift * (np.sqrt(3) * r_init / 2)
        dx = np.sqrt(3) * r_init
        max_x = 1.0 - r_init
        count = int((max_x - x) / dx) + 1
        count = min(count, N - idx)
        for _ in range(count):
            centers.append([x, y])
            x += dx
            idx += 1
        y += 1.5 * r_init
        shift = 1.0 - shift
        
    centers = np.array(centers)
    # Add small perturbation to break symmetry
    centers += np.random.uniform(-0.002, 0.002, centers.shape)
    centers = np.clip(centers, 0, 1)
    
    # Flatten variables: [x1, y1, ..., xN, yN, r1, ..., rN]
    x0 = np.zeros(3 * N)
    x0[:2*N] = centers.flatten()
    x0[2*N:] = r_init
    
    bounds = [(0.0, 1.0)] * (2*N) + [(0.0, 1.0)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_sum = 0.0
    best_v = x0
    
    # 2. Multi-start optimization
    for trial in range(4):
        x_init = x0 + np.random.uniform(-0.005, 0.005, x0.shape)
        try:
            res = minimize(objective, x_init, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 1000, 'ftol': 1e-10, 'disp': False})
            if -res.fun > best_sum:
                best_sum = -res.fun
                best_v = res.x
        except Exception:
            continue
            
    c_opt = best_v[:2*N].reshape((N, 2))
    r_opt = best_v[2*N:]
    
    # 3. Post-processing to guarantee strict validity
    c_out = c_opt.copy()
    r_out = r_opt.copy()
    
    # Project centers to satisfy boundary constraints strictly
    for i in range(N):
        c_out[i, 0] = np.clip(c_out[i, 0], r_out[i], 1 - r_out[i])
        c_out[i, 1] = np.clip(c_out[i, 1], r_out[i], 1 - r_out[i])
        
    # Iteratively shrink overlapping circles
    for _ in range(20):
        changed = False
        for i in range(N):
            for j in range(i+1, N):
                dist = np.linalg.norm(c_out[i] - c_out[j])
                r_sum = r_out[i] + r_out[j]
                if dist < r_sum - 1e-11:
                    shrink = (r_sum - dist) / 2
                    r_out[i] -= shrink
                    r_out[j] -= shrink
                    changed = True
        if not changed:
            break
            
    # Final boundary clamp
    for i in range(N):
        r_out[i] = min(r_out[i], c_out[i,0], 1-c_out[i,0], c_out[i,1], 1-c_out[i,1])
        r_out[i] = max(r_out[i], 0.0)
        
    return c_out, r_out, float(np.sum(r_out))
