import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes the positions of 26 circles in a unit square to maximize the sum of radii.
    
    Returns:
        centers: np.array of shape (26, 2)
        radii: np.array of shape (26,)
        sum_radii: float
    """
    n = 26
    
    def compute_radii(centers: np.ndarray) -> np.ndarray:
        """
        Computes the maximum possible radius for each circle given their centers.
        Radii are constrained by the square boundaries and non-overlap with other circles.
        """
        radii = np.zeros(n)
        
        # 1. Distance to boundaries for each circle
        # Distance to left wall (x=0) and right wall (x=1)
        dist_x = np.minimum(centers[:, 0], 1.0 - centers[:, 0])
        # Distance to bottom wall (y=0) and top wall (y=1)
        dist_y = np.minimum(centers[:, 1], 1.0 - centers[:, 1])
        
        # The limiting distance to any wall
        wall_dist = np.minimum(dist_x, dist_y)
        
        # 2. Distance to nearest neighbor for each circle
        # Initialize with a large value
        min_pair_dist = np.full(n, np.inf)
        
        # Compute pairwise distances
        # Vectorized approach for efficiency
        # Expand dims to use broadcasting: (n, 1, 2) - (1, n, 2) -> (n, n, 2)
        diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        
        # Set diagonal (self-distance) to infinity so it doesn't affect min
        np.fill_diagonal(dists, np.inf)
        
        # Find minimum distance to any other circle for each circle
        min_pair_dist = np.min(dists, axis=1)
        
        # 3. Radius is the minimum of wall distance and half the nearest neighbor distance
        # If a center is outside the square, wall_dist will be negative.
        # We treat this as a hard constraint: radius cannot be negative.
        # However, for optimization purposes, we might want to penalize negative radii heavily.
        # Here we simply compute the geometric limit.
        raw_radii = np.minimum(wall_dist, min_pair_dist / 2.0)
        
        # Radii must be non-negative
        return np.maximum(raw_radii, 0.0)

    def objective(centers_flat: np.ndarray) -> float:
        """
        Objective function to minimize: -sum of radii.
        Includes a penalty term if centers are outside the unit square to guide the optimizer.
        """
        centers = centers_flat.reshape((n, 2))

        # This helps the optimizer keep circles inside the box
        penalty = 0.0
        for i in range(n):
            x, y = centers[i]
            if x < 0 or x > 1:
                penalty += 100 * min(x**2, (x-1)**2)
            if y < 0 or y > 1:
                penalty += 100 * min(y**2, (y-1)**2)
        
        radii = compute_radii(centers)
        return -np.sum(radii) + penalty

            # Row 1: 5 circles (shifted)

                        centers[idx] = [x, y]
                        idx += 1

                idx += 1

            centers += np.random.normal(0, 0.01, centers.shape)

            centers = np.clip(centers, 0.01, 0.99)
            
        elif method == 'random':
            centers = np.random.rand(n, 2) * 0.8 + 0.1 # Keep inside [0.1, 0.9] initially
            
        return centers.flatten()

    best_obj = np.inf
    best_centers_flat = None

        try:
            result = minimize(objective, x0, method='Nelder-Mead', options={'maxiter': 5000, 'xatol': 1e-6, 'fatol': 1e-8})

    sum_radii = np.sum(final_radii)
    
    return best_centers, final_radii, sum_radii
