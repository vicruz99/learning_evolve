import numpy as np
from scipy.optimize import linprog
from scipy.optimize import minimize

def run_packing():
    """
    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    
    Returns:
        centers: np.array of shape (26, 2)
        radii: np.array of shape (26,)
        sum_radii: float
    """
    n = 26
    np.random.seed(42)
    
    # 1. Initialize centers in a hexagonal grid pattern
    centers = np.zeros((n, 2))
    row, col = 0, 0
    idx = 0
    
    # Estimate rows and columns for hexagonal packing
    # Area per circle ~ pi * r^2. Total area 1. 
    # For equal circles, r ~ 0.11.
    # Hex grid spacing dx = 2r, dy = sqrt(3)r.
    
    current_y = 0.1
    row_idx = 0
    
    while idx < n:
        current_x = 0.1 if row_idx % 2 == 0 else 0.2 # Offset for hex
        while current_x < 0.95 and idx < n:
            centers[idx] = [current_x, current_y]
            idx += 1
            current_x += 0.22 # Rough spacing
        current_y += 0.19
        row_idx += 1
        
    # Ensure all centers are strictly inside the square initially
    centers = np.clip(centers, 0.05, 0.95)

    # 2. Iterative Optimization
    for _ in range(2000):
        # A. Solve LP for radii given current centers
        # Variables: r_0 ... r_25
        c_obj = -np.ones(n)  # Minimize -sum(r)
        
        A_ub = []
        b_ub = []
        
        # Constraints: r_i + r_j <= dist(i, j)  =>  r_i + r_j - dist <= 0
        # Also r_i + r_j <= dist is linear in r, but dist is constant here
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.linalg.norm(centers[i] - centers[j])
                if dist < 1e-9: dist = 1e-9 # Prevent division by zero issues later if needed, though LP handles it
                
                row = np.zeros(n)
                row[i] = 1
                row[j] = 1
                A_ub.append(row)
                b_ub.append(dist)
                
                # Symmetric constraint isn't needed as we iterate i<j
        
        # Constraints: r_i <= dist_to_wall
        for i in range(n):
            x, y = centers[i]
            d_wall = min(x, 1-x, y, 1-y)
            row = np.zeros(n)
            row[i] = 1
            A_ub.append(row)
            b_ub.append(d_wall)
            
        # Constraints: r_i >= 0
        bounds = [(0, None) for _ in range(n)]
        
        A_ub = np.array(A_ub)
        b_ub = np.array(b_ub)
        
        # Solve LP
        # Using 'highs' method for efficiency
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        
        if not res.success:
            # Fallback if LP fails, keep previous or small radii
            radii = np.ones(n) * 0.01
        else:
            radii = res.x

        forces = np.zeros_like(centers)
        
        # Force parameters

        k_wall = 10.0
        step_size = 0.005
        tolerance = 1e-5

        # If r_i + r_j is close to dist, they are touching. Push apart.

            for j in range(i + 1, n):
                dist = np.linalg.norm(centers[i] - centers[j])
                r_sum = radii[i] + radii[j]
                
                # Slack: positive means gap, negative means overlap (shouldn't happen if LP solved correctly)
                # We want to push apart if slack is small
                if dist > 1e-9:
                    slack = dist - r_sum
                    # If slack is very small, apply force
                    if slack < 0.1: # Threshold for "tightness"
                        # Force magnitude inversely proportional to slack? 
                        # Or just constant repulsion if touching?
                        # Let's use a soft repulsion that increases as they get closer
                        if slack > 0:
                            force_mag = k_repulse * (0.1 - slack) / (slack + 1e-5)
                        else:
                            force_mag = k_repulse * 100 # Strong repulsion if overlapping (numerical error)
                        
                        vec = centers[i] - centers[j]
                        forces[i] += vec * force_mag
                        forces[j] -= vec * force_mag
        
        # 2. Wall forces
        # If r_i is close to wall distance, push center away from wall
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            
            # Left wall
            if x - r < tolerance:
                forces[i, 0] += k_wall * (tolerance - (x - r))
            # Right wall
            if (1 - x) - r < tolerance:
                forces[i, 0] -= k_wall * (tolerance - ((1 - x) - r))
            # Bottom wall
            if y - r < tolerance:
                forces[i, 1] += k_wall * (tolerance - (y - r))
            # Top wall
            if (1 - y) - r < tolerance:
                forces[i, 1] -= k_wall * (tolerance - ((1 - y) - r))

        centers += step_size * forces

        # Although forces should keep them in, numerical drift can happen.
        # But we must ensure r_i <= dist_to_wall. 
        # If center goes outside, LP will constrain r to 0 or negative.
        # Better to clip centers to [0, 1].
        centers = np.clip(centers, 0.0, 1.0)

    # Final LP solve to get precise radii
    c_obj = -np.ones(n)
    A_ub = []
    b_ub = []
    
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            if dist < 1e-9: dist = 1e-9 
            row = np.zeros(n)
            row[i] = 1
            row[j] = 1
            A_ub.append(row)
            b_ub.append(dist)
            
    for i in range(n):
        x, y = centers[i]
        d_wall = min(x, 1-x, y, 1-y)
        row = np.zeros(n)
        row[i] = 1
        A_ub.append(row)
        b_ub.append(d_wall)
        
    bounds = [(0, None) for _ in range(n)]
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    radii = res.x
    
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
