import numpy as np
from scipy.optimize import minimize

def generate_initial_config(n):
    """Generates a hexagonal lattice arrangement for n circles."""
    # Hexagonal row configuration for 26 circles
    row_counts = [5, 5, 6, 5, 5]
    centers = []
    for i, count in enumerate(row_counts):
        y = i * np.sqrt(3)/2
        for j in range(count):
            x = j + (i % 2) * 0.5
            centers.append([x, y])
    centers = np.array(centers)
    
    # Normalize and center in [0,1]^2 with margin
    min_c = centers.min(axis=0)
    max_c = centers.max(axis=0)
    extent = max_c - min_c
    scale = 0.8 / max(extent)
    centers = (centers - min_c) * scale
    centers += (1 - scale * extent) / 2
    
    radii = np.full(n, 0.05)
    return centers, radii

def boundary_constraints(vars, n):
    """Returns array of constraint values >= 0 for circle boundaries."""
    c = vars[:2*n].reshape(n, 2)
    r = vars[2*n:]
    con = np.zeros(4*n)
    con[0::4] = c[:, 0] - r          # x >= r
    con[1::4] = 1 - r - c[:, 0]      # x <= 1 - r
    con[2::4] = c[:, 1] - r          # y >= r
    con[3::4] = 1 - r - c[:, 1]      # y <= 1 - r
    return con

def non_overlap_constraints(vars, n):
    """Returns array of constraint values >= 0 for non-overlapping circles."""
    c = vars[:2*n].reshape(n, 2)
    r = vars[2*n:]
    con = []
    for i in range(n):
        for j in range(i+1, n):
            dist_sq = np.sum((c[i] - c[j])**2)
            con.append(dist_sq - (r[i] + r[j])**2)
    return np.array(con)

def run_packing():
    n = 26
    best_sum = -1.0
    best_centers = None
    best_radii = None

    for trial in range(5):

        centers += np.random.uniform(-0.005, 0.005, size=centers.shape)
        centers = np.clip(centers, 0.01, 0.99)

            return -np.sum(vars[2*n:])
            
        constraints = [
            {'type': 'ineq', 'fun': boundary_constraints, 'args': (n,)},
            {'type': 'ineq', 'fun': non_overlap_constraints, 'args': (n,)}
        ]
        bounds = [(0.0, 1.0)]*(2*n) + [(0.0, 0.5)]*n
        
        res = minimize(objective, x0, method='SLSQP', bounds=bounds,

                    break

                for i in range(n):

                        dist = np.sqrt(np.sum((c_test[i] - c_test[j])**2))

                            valid = False
                            break
