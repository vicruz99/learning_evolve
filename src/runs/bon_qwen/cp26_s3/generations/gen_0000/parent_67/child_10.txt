import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def make_constraint(idx, jdx, type_):
    """Factory function to create constraint evaluators."""
    def constraint(theta):
        xi = theta[3*idx]
        yi = theta[3*idx+1]
        ri = theta[3*idx+2]
        if type_ == 'x_min':
            return xi - ri
        if type_ == 'x_max':
            return 1.0 - xi - ri
        if type_ == 'y_min':
            return yi - ri
        if type_ == 'y_max':
            return 1.0 - yi - ri
        if type_ == 'dist':
            xj = theta[3*jdx]
            yj = theta[3*jdx+1]
            rj = theta[3*jdx+2]
            dx = xi - xj
            dy = yi - yj
            return dx*dx + dy*dy - (ri + rj)**2
        return 0.0
    return constraint

def objective(theta):
    """Negative sum of radii for minimization."""
    return -sum(theta[3*i+2] for i in range(N_CIRCLES))

def get_initial_guess(n):
    """Generate hexagonal lattice initial positions."""
    r = 0.085
    pts = []
    row = 0
    y = r
    while y + r < 1.0:
        x = r if row % 2 == 0 else 2 * r
        while x + r < 1.0:
            pts.append((x, y))
            x += 2 * r
        if len(pts) >= n:
            break
        y += np.sqrt(3) * r
        row += 1
        
    # Fill remaining circles if needed
    while len(pts) < n:
        pts.append((np.random.uniform(r, 1-r), np.random.uniform(r, 1-r)))
        
    theta = []
    for (x, y) in pts[:n]:
        theta.extend([x, y, r])
    return np.array(theta)

def run_packing():
    np.random.seed(42)
    best_sum = -1.0
    best_theta = None
    
    # Bounds: x,y in [0,1], r in [1e-6, 0.5]
    bounds = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * N_CIRCLES
    
    # Build constraint list
    constraints = []
    for i in range(N_CIRCLES):
        constraints.append({'type': 'ineq', 'fun': make_constraint(i, -1, 'x_min')})
        constraints.append({'type': 'ineq', 'fun': make_constraint(i, -1, 'x_max')})
        constraints.append({'type': 'ineq', 'fun': make_constraint(i, -1, 'y_min')})
        constraints.append({'type': 'ineq', 'fun': make_constraint(i, -1, 'y_max')})
    for i in range(N_CIRCLES):
        for j in range(i+1, N_CIRCLES):
            constraints.append({'type': 'ineq', 'fun': make_constraint(i, j, 'dist')})
            
    # Multiple restarts to avoid local minima
    for _ in range(5):
        x0 = get_initial_guess(N_CIRCLES)
        noise = np.random.normal(0, 0.01, size=x0.shape)
        x0 = x0 + noise
        
        # Clip to valid bounds
        x0[0::3] = np.clip(x0[0::3], 0.0, 1.0)
        x0[1::3] = np.clip(x0[1::3], 0.0, 1.0)
        x0[2::3] = np.clip(x0[2::3], 0.01, 0.5)
        
        res = minimize(objective, x0, method='SLSQP', bounds=bounds, 
                       constraints=constraints, options={'maxiter': 300, 'ftol': 1e-9, 'disp': False})
        
        current_sum = -res.fun
        if current_sum > best_sum:
            best_sum = current_sum
            best_theta = res.x
            
    centers = best_theta.reshape(-1, 3)[:, :2].copy()
    radii = best_theta.reshape(-1, 3)[:, 2].copy()
    
    # Strict projection to guarantee validity against numerical tolerance
    radii = np.maximum(radii, 1e-9)
    centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
    centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)
    
    return centers, radii, float(np.sum(radii))
