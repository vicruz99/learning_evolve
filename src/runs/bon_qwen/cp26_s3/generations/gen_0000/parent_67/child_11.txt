import numpy as np
from scipy.optimize import minimize

def objective_func(vars, n):
    """Objective function to maximize sum of radii (minimize negative sum)."""
    return -np.sum(vars[2*n:])

def constraints_func(vars, n, triu_idx):
    """Inequality constraints: boundary containment and non-overlap."""
    c = vars[:2*n].reshape(n, 2)
    r = vars[2*n:]
    
    # Boundary constraints: r <= x, y <= 1-r
    cons = np.concatenate([
        c[:, 0] - r,
        1.0 - c[:, 0] - r,
        c[:, 1] - r,
        1.0 - c[:, 1] - r
    ])
    
    # Pairwise non-overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = c[:, 0] - c[:, 0][:, None]
    dy = c[:, 1] - c[:, 1][:, None]
    d2 = dx**2 + dy**2
    r_sum = r[:, None] + r[None, :]
    
    cons_pair = d2[triu_idx] - r_sum[triu_idx]**2
    return np.concatenate([cons, cons_pair])

def run_packing():
    n = 26
    # Precompute indices for upper triangle to avoid repeated allocation
    triu_idx = np.triu_indices(n, k=1)
    
    # Generate initial hexagonal-like grid configuration
    centers = []
    spacing = 0.25
    y = 0.25
    row_idx = 0
    while len(centers) < n:
        x = 0.25
        if row_idx % 2 == 1:
            x = 0.375
        while x <= 0.75 and len(centers) < n:
            centers.append([x, y])
            x += spacing
        y += spacing * np.sqrt(3) / 2
        row_idx += 1
        
    centers_init = np.array(centers[:n])
    radii_init = np.full(n, 0.04)
    
    # Flatten variables: [x1, y1, ..., x_n, y_n, r1, ..., r_n]
    x0 = np.concatenate([centers_init.flatten(), radii_init])
    
    # Bounds for variables
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
    
    # Run optimization
    res = minimize(
        objective_func, 
        x0, 
        args=(n,), 
        method='SLSQP', 
        bounds=bounds, 
        constraints={'type': 'ineq', 'fun': constraints_func, 'args': (n, triu_idx)},
        options={'maxiter': 3000, 'ftol': 1e-10, 'disp': False}
    )
    
    # Extract and format results
    c_opt = res.x[:2*n].reshape(n, 2)
    r_opt = res.x[2*n:]

    c_opt = np.clip(c_opt, 1e-9, 1.0 - 1e-9)
    r_opt = np.maximum(r_opt, 1e-9)
    
    return c_opt, r_opt, np.sum(r_opt)
