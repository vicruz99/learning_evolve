import numpy as np
from scipy.optimize import minimize

def generate_initial_config(n):
    """Generates a valid hexagonal lattice configuration."""
    r = 0.08
    centers = []
    y = r
    row_idx = 0
    while len(centers) < n:
        x_start = r if row_idx % 2 == 0 else 2 * r
        x = x_start
        while x + r <= 1.0 + 1e-9 and len(centers) < n:
            centers.append([x, y])
            x += 2 * r
        y += np.sqrt(3) * r
        row_idx += 1
    centers = np.array(centers[:n])
    radii = np.full(n, r)
    return centers, radii

def compute_objective_and_constraints(vars, n):
    """Calculates objective value and constraint array."""
    radii = vars[2::3]
    x_coords = vars[0::3]
    y_coords = vars[1::3]
    
    # Objective: maximize sum of radii
    obj = -np.sum(radii)
    
    constraints = []
    
    # 1. Boundary constraints: r <= x <= 1-r and r <= y <= 1-r
    constraints.extend(x_coords - radii)
    constraints.extend(1.0 - x_coords - radii)
    constraints.extend(y_coords - radii)
    constraints.extend(1.0 - y_coords - radii)
    
    # 2. Non-overlap constraints: dist^2 >= (r_i + r_j)^2
    for i in range(n):
        for j in range(i + 1, n):
            dx = x_coords[i] - x_coords[j]
            dy = y_coords[i] - y_coords[j]
            dist_sq = dx*dx + dy*dy
            min_dist = radii[i] + radii[j]
            constraints.append(dist_sq - min_dist * min_dist)
            
    return obj, np.array(constraints)

def run_packing():
    n = 26
    
    # Initial Setup
    centers, radii = generate_initial_config(n)
    x0 = np.zeros(3 * n)
    for i in range(n):
        x0[3*i] = centers[i, 0]
        x0[3*i+1] = centers[i, 1]
        x0[3*i+2] = radii[i]
        
    bounds = []
    for _ in range(n):
        bounds.extend([(0, 1), (0, 1), (0, 0.5)])
        
    best_vars = x0.copy()
    best_score = np.sum(radii)

    for _ in range(5):
        result = minimize(
            fun=lambda x: compute_objective_and_constraints(x, n)[0],
            x0=best_vars,
            bounds=bounds,
            constraints={'type': 'ineq', 'fun': lambda x: compute_objective_and_constraints(x, n)[1]},
            method='SLSQP',
            options={'ftol': 1e-12, 'maxiter': 1000, 'disp': False}
        )
        
        if result.success and (-result.fun) > best_score:
            best_vars = result.x
            best_score = -result.fun
            
    # Final radius maximization push
    final_centers = np.array([[best_vars[i], best_vars[i+1]] for i in range(0, len(best_vars), 3)])
    final_radii = np.array([best_vars[i+2] for i in range(0, len(best_vars), 3)])
    
    # Jiggling to maximize radii at contact points
    # Try to increase all radii by a tiny epsilon if constraints allow
    for _ in range(2):
        for k in range(n):
            curr_r = final_radii[k]
            max_r = 0.5
            # Boundary limits
            max_r = min(max_r, final_centers[k, 0])
            max_r = min(max_r, 1.0 - final_centers[k, 0])
            max_r = min(max_r, final_centers[k, 1])
            max_r = min(max_r, 1.0 - final_centers[k, 1])
            
            # Neighbor limits
            for m in range(n):
                if k != m:
                    dx = final_centers[k, 0] - final_centers[m, 0]
                    dy = final_centers[k, 1] - final_centers[m, 1]
                    dist = np.sqrt(dx*dx + dy*dy)
                    max_r = min(max_r, dist - final_radii[m])
            
            if max_r > curr_r:
                final_radii[k] = max(max_r - 1e-10, curr_r)
                
    return final_centers, final_radii, np.sum(final_radii)
