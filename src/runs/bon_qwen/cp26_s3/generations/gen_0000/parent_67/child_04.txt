def compute_energy(centers_flat, r):
    """
    Computes the penalty energy for a given configuration of centers and radius r.
    Energy is 0 if the packing is valid (no overlaps, inside bounds with margin r).
    """
    n_circles = 26
    centers = centers_flat.reshape(n_circles, 2)
    n = n_circles
    energy = 0.0
    
    # Boundary penalties
    # Centers must be within [r, 1-r] for both x and y
    x = centers[:, 0]
    y = centers[:, 1]
    
    # Penalty for x < r
    term_x_low = np.maximum(0, r - x)
    # Penalty for x > 1-r
    term_x_high = np.maximum(0, x - (1 - r))
    # Penalty for y < r
    term_y_low = np.maximum(0, r - y)
    # Penalty for y > 1-r
    term_y_high = np.maximum(0, y - (1 - r))
    
    energy += np.sum(term_x_low**2) + np.sum(term_x_high**2)
    energy += np.sum(term_y_low**2) + np.sum(term_y_high**2)
    
    # Overlap penalties
    # Distance between any two centers must be >= 2r
    # Penalty is (2r - dist)^2 if dist < 2r
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            d = np.sqrt(dx*dx + dy*dy)
            overlap = 2*r - d
            if overlap > 0:
                energy += overlap**2
    return energy
