import numpy as np
from scipy.optimize import minimize

def obj_func(x, n):
    r = x[2*n:]
    return -np.sum(r)

def constraint_func(x, n):
    centers = x[:2*n].reshape(n, 2)
    r = x[2*n:]
    
    constraints = []
    
    # Boundary constraints: x - r >= 0, 1 - x - r >= 0, etc.
    constraints.append(centers[:, 0] - r)
    constraints.append(1 - centers[:, 0] - r)
    constraints.append(centers[:, 1] - r)
    constraints.append(1 - centers[:, 1] - r)
    
    # Overlap constraints: (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
    # Vectorized calculation for O(N^2) constraints
    dx = centers[:, 0][:, np.newaxis] - centers[:, 0][np.newaxis, :]
    dy = centers[:, 1][:, np.newaxis] - centers[:, 1][np.newaxis, :]
    dist_sq = dx**2 + dy**2
    
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    vals = dist_sq[mask] - r_sum[mask]**2
    constraints.append(vals)
    
    return np.concatenate(constraints)
