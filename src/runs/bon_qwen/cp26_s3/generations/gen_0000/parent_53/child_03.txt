import numpy as np
from scipy.optimize import minimize

def compute_penalty(vars, n):
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    pen = 0.0
    # Boundary penalties
    pen += np.sum(np.maximum(0.0, r - x)**2)
    pen += np.sum(np.maximum(0.0, r - (1.0 - x))**2)
    pen += np.sum(np.maximum(0.0, r - y)**2)
    pen += np.sum(np.maximum(0.0, r - (1.0 - y))**2)
    
    # Overlap penalties
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt((x[i] - x[j])**2 + (y[i] - y[j])**2)
            overlap = r[i] + r[j] - dist
            if overlap > 0:
                pen += overlap**2
    return pen

def objective(vars, n):
    # Maximize sum of radii => minimize negative sum
    # Add large penalty for constraint violations
    return -np.sum(vars[2::3]) + 2000.0 * compute_penalty(vars, n)

def get_initialization(n, seed):
    np.random.seed(seed)
    init_x = []
    init_y = []
    # Hexagonal spacing approximations
    dy = np.sqrt(3)/2 * 0.18
    dx = 0.18
    base_y = 0.12 + np.random.uniform(-0.02, 0.02)
    
    for i in range(5):
        y = base_y + i * dy
        offset = dx/2 if i % 2 == 1 else 0
        j = 0
        while len(init_x) < n:
            x = 0.12 + offset + j * dx + np.random.uniform(-0.01, 0.01)
            if x < 0.95:
                init_x.append(x)
                init_y.append(y)
                j += 1
            else:
                break
                
    init_x = np.array(init_x[:n])
    init_y = np.array(init_y[:n])
    r0 = np.full(n, 0.05)
    return np.concatenate([init_x, init_y, r0])

def run_packing():
    n = 26
    best_vars = None
    best_val = np.inf
    
    # Variable bounds: x in [0,1], y in [0,1], r in [0, 0.5]
    bounds = [(0.0, 1.0)] * (3 * n)
    for i in range(2, 3 * n, 3):
        bounds[i] = (0.0, 0.5)
        
    # Run optimization from multiple random initializations
    for seed in range(5):
        vars0 = get_initialization(n, seed)
        try:
            res = minimize(objective, vars0, args=(n,), bounds=bounds, method='L-BFGS-B',
                           options={'ftol': 1e-12, 'maxiter': 50000, 'disp': False})
            if res.fun < best_val:
                best_val = res.fun
                best_vars = res.x
        except Exception:
            continue
            
    if best_vars is None:
        best_vars = get_initialization(n, 0)
        
    centers = np.column_stack((best_vars[0::3], best_vars[1::3]))
    radii = best_vars[2::3]
    
    # Strict enforcement of constraints to guarantee validity against numerical errors
    for i in range(n):
        radii[i] = min(radii[i], centers[i,0], 1.0 - centers[i,0], centers[i,1], 1.0 - centers[i,1])
        
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            if dist < radii[i] + radii[j]:
                excess = radii[i] + radii[j] - dist
                radii[i] -= excess / 2
                radii[j] -= excess / 2
                
    radii = np.maximum(radii, 0.0)
    return centers, radii, np.sum(radii)
