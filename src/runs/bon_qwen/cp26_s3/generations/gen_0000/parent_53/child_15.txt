                    if x - r < -1e-6 or x + r > 1 + 1e-6 or y - r < -1e-6 or y + r > 1 + 1e-6:
                        valid = False; break
                if valid:
                    for i in range(n):
                        for j in range(i+1, n):
                            dist = np.linalg.norm(final_centers[i] - final_centers[j])
                            if dist < 2*final_r - 1e-6:
                                valid = False; break
                        if not valid: break
                
                if valid:
                    current_sum = n * final_r
                    if best_result is None or current_sum > best_result[2]:
                        best_result = (final_centers, np.full(n, final_r), current_sum)
        except Exception:
            pass

    # Strategy 2: Random initialization (sometimes finds better local minima)
    for i in range(20):
        np.random.seed(1000 + i)
        centers = np.random.rand(n, 2) * 0.6 + 0.2 # Centered
        r_init = 0.08
        initial_params = np.concatenate([centers.flatten(), [r_init]])
        
        bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)]
        
        try:
            res = minimize(calculate_loss, initial_params, args=(n,), method='L-BFGS-B', bounds=bounds, 
                           options={'maxiter': 3000, 'ftol': 1e-13})
            
            final_r = res.x[2*n]
            final_centers = res.x[:2*n].reshape(n, 2)
            loss_val = calculate_loss(res.x, n)
            penalty_part = loss_val + final_r
            
            if penalty_part < 1e-4:
                valid = True
                for i in range(n):
                    x, y = final_centers[i]
                    r = final_r
                    if x - r < -1e-6 or x + r > 1 + 1e-6 or y - r < -1e-6 or y + r > 1 + 1e-6:
                        valid = False; break
                if valid:
                    for i in range(n):
                        for j in range(i+1, n):
                            dist = np.linalg.norm(final_centers[i] - final_centers[j])
                            if dist < 2*final_r - 1e-6:
                                valid = False; break
                        if not valid: break
                
                if valid:
                    current_sum = n * final_r
                    if best_result is None or current_sum > best_result[2]:
                        best_result = (final_centers, np.full(n, final_r), current_sum)
        except Exception:
            pass

    if best_result is None:
        # Fallback to a simple grid if optimization fails
        centers = np.zeros((n, 2))
        idx = 0
        for r in range(5):
            for c in range(5):
                if idx < n:
                    centers[idx] = [0.1 + c*0.2, 0.1 + r*0.2]
                    idx += 1
        # Fill last one
        if idx < n:
            centers[idx] = [0.5, 0.5]
        radii = np.full(n, 0.095) # Slightly smaller to be safe
        # Check if 0.095 fits grid? 0.1 spacing is 0.2, radius 0.095 -> dist 0.2 >= 0.19. Yes.
        # But sum is low.
        # Let's try to optimize this grid locally
        initial_params = np.concatenate([centers.flatten(), [0.095]])
        bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)]
        try:
             res = minimize(calculate_loss, initial_params, args=(n,), method='L-BFGS-B', bounds=bounds, options={'maxiter': 5000})
             final_r = res.x[2*n]
             final_centers = res.x[:2*n].reshape(n, 2)
             # Check validity
             valid = True
             for i in range(n):
                 x, y = final_centers[i]
                 if x - final_r < -1e-6 or x + final_r > 1 + 1e-6 or y - final_r < -1e-6 or y + final_r > 1 + 1e-6:
                     valid = False
             if valid:
                 for i in range(n):
                     for j in range(i+1, n):
                         if np.linalg.norm(final_centers[i] - final_centers[j]) < 2*final_r - 1e-6:
                             valid = False
             if valid:
                 best_result = (final_centers, np.full(n, final_r), n * final_r)
        except:
            pass

    if best_result is None:
        # Absolute fallback
        centers = np.zeros((n, 2))
        idx = 0
        for r in range(5):
            for c in range(5):
                if idx < n:
                    centers[idx] = [0.1 + c*0.2, 0.1 + r*0.2]
                    idx += 1
        if idx < n: centers[idx] = [0.5, 0.5]
        radii = np.full(n, 0.09)
        best_result = (centers, radii, n * 0.09)

    return best_result[0], best_result[1], best_result[2]
