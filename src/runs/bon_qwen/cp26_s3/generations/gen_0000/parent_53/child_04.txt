import numpy as np
from scipy.optimize import linprog
from scipy.spatial.distance import pdist, squareform
import random

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Packs 26 circles in a unit square to maximize sum of radii.
    Returns centers (26, 2), radii (26,), and sum of radii.
    """
    n = 26
    best_sum_radii = 0.0
    best_centers = None
    best_radii = None
    
    # --- Helper Function to Solve for Radii given Centers ---
    def solve_radii_lp(centers):
        """
        Solves the Linear Programming problem to find max radii for fixed centers.
        Maximize sum(r)
        s.t.
          r_i + r_j <= dist(i, j)
          r_i <= min(x_i, 1-x_i, y_i, 1-y_i)
          r_i >= 0
        """
        # Objective: maximize sum(r) => minimize -sum(r)
        c_obj = np.ones(n) * -1.0
        
        # Constraints A_ub @ r <= b_ub
        # We construct the matrix A_ub and vector b_ub
        # Rows 0-3n-1: Boundary constraints (r_i <= dist_to_wall)
        # Rows 3n-end: Pairwise distance constraints (r_i + r_j <= dist)
        
        # Pre-calculate distances
        dists = pdist(centers, metric='euclidean')
        # dists is condensed (upper triangle). Map indices back to i, j
        # Number of pairwise constraints
        n_pairs = n * (n - 1) // 2
        
        A_ub = np.zeros((4 * n + n_pairs, n))
        b_ub = np.zeros(4 * n + n_pairs)
        
        # 1. Boundary Constraints
        # r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
        idx = 0
        for i in range(n):
            xi, yi = centers[i]
            
            # r_i <= xi
            A_ub[idx, i] = 1.0
            b_ub[idx] = xi
            idx += 1
            
            # r_i <= 1 - xi
            A_ub[idx, i] = 1.0
            b_ub[idx] = 1.0 - xi
            idx += 1
            
            # r_i <= yi
            A_ub[idx, i] = 1.0
            b_ub[idx] = yi
            idx += 1
            
            # r_i <= 1 - yi
            A_ub[idx, i] = 1.0
            b_ub[idx] = 1.0 - yi
            idx += 1
            
        # 2. Pairwise Constraints
        # r_i + r_j <= dist(i, j)
        dist_idx = 0
        for i in range(n):
            for j in range(i + 1, n):
                A_ub[idx, i] = 1.0
                A_ub[idx, j] = 1.0
                b_ub[idx] = dists[dist_idx]
                idx += 1
                dist_idx += 1
                
        # Bounds for r_i >= 0
        bounds = [(0, None) for _ in range(n)]
        
        # Solve LP
        # method='highs' is fast and robust in modern scipy
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
            if res.success:
                return res.x, -res.fun
            else:
                # Fallback if LP fails (should not happen for valid centers)
                return np.zeros(n), 0.0
        except Exception:
            return np.zeros(n), 0.0

    # --- Initial Configuration: Hexagonal Lattice ---
    centers = []
    # Approximate spacing based on target r ~ 0.1
    # dx = 2*r, dy = sqrt(3)*r
    dx = 0.2
    dy = 0.1732
    
    y = 0.1
    row = 0
    while y <= 0.95:
        x = 0.1
        # Shift every other row for hexagonal packing
        if row % 2 == 1:
            x = 0.1 + dx / 2
        
        while x <= 0.95:
            centers.append([x, y])
            x += dx
        y += dy
        row += 1
        
    centers = np.array(centers)
    
    # Adjust size to exactly 26
    if len(centers) > n:
        centers = centers[:n]
    elif len(centers) < n:
        # If we don't have enough, fill with random points
        # (Though hex grid usually produces enough for small area)
        while len(centers) < n:
            p = np.random.rand(2)
            # Ensure it's somewhat spread
            if np.min(np.linalg.norm(centers - p, axis=1)) > 0.05:
                centers = np.vstack([centers, p])
    
    centers = centers[:n]
    
    # --- Iterative Optimization (Simulated Annealing / Hill Climbing) ---
    # We optimize centers. Radii are determined by LP.
    
    current_centers = centers.copy()
    current_radii, current_sum = solve_radii_lp(current_centers)
    
    # If initial sum is good, store it
    if current_sum > best_sum_radii:
        best_sum_radii = current_sum
        best_centers = current_centers.copy()
        best_radii = current_radii.copy()
    
    # Optimization Parameters
    step_size = 0.05  # Initial step size for moving centers
    iterations = 2000
    
    for step in range(iterations):
        # Decay step size slowly to fine-tune
        if step % 500 == 0 and step > 0:
            step_size *= 0.8
            
        # Pick a random circle to move
        idx = random.randint(0, n - 1)
        
        # Propose a move
        # Random direction
        angle = random.uniform(0, 2 * np.pi)
        move_x = random.uniform(-step_size, step_size)
        move_y = random.uniform(-step_size, step_size)
        
        new_centers = current_centers.copy()
        new_centers[idx, 0] += move_x
        new_centers[idx, 1] += move_y
        
        # Clip to boundaries (centers must be in [0,1])
        # Actually, LP handles boundaries, but keeping centers in [0,1] is safer for distance logic
        # If center is at 0, radius must be 0.
        new_centers = np.clip(new_centers, 1e-9, 1.0 - 1e-9)
        
        # Solve for radii with new centers
        new_radii, new_sum = solve_radii_lp(new_centers)
        
        # Accept if better
        if new_sum > current_sum:
            current_centers = new_centers
            current_radii = new_radii
            current_sum = new_sum
            
            # Track best
            if current_sum > best_sum_radii:
                best_sum_radii = current_sum
                best_centers = current_centers.copy()
                best_radii = current_radii.copy()
        else:
            # Small probability to accept worse moves (Simulated Annealing element)
            # to escape local minima
            T = 0.01 * (1 - step / iterations) # Temperature decaying
            if T > 0 and random.random() < np.exp((new_sum - current_sum) / T):
                current_centers = new_centers
                current_radii = new_radii
                current_sum = new_sum
            else:
                # Keep step size constant or increase if stuck?
                pass

    # Final verification and cleanup
    # Ensure radii are strictly positive if possible, though LP might return 0
    # The problem asks for non-negative.
    
    # Re-run LP on best centers to be absolutely sure of radii consistency
    final_radii, final_sum = solve_radii_lp(best_centers)
    
    # Round tiny negative radii to 0 if any numerical noise
    final_radii = np.maximum(final_radii, 0.0)
    
    return best_centers, final_radii, np.sum(final_radii)
