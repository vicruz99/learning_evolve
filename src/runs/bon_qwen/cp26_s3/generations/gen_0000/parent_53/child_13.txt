import numpy as np
from scipy.optimize import minimize

def compute_objective_and_penalty(vars_flat, n, pen_weight):
    """Computes the negative sum of radii plus quadratic penalty for constraint violations."""
    centers = vars_flat[:2*n].reshape(n, 2)
    radii = vars_flat[2*n:]
    
    # Objective: minimize -sum(radii)
    obj = -np.sum(radii)
    
    # Boundary violations: circle must be inside [0,1]x[0,1]
    v = np.maximum(0, radii - centers[:, 0])
    v = np.maximum(v, centers[:, 0] + radii - 1)
    v = np.maximum(v, radii - centers[:, 1])
    v = np.maximum(v, centers[:, 1] + radii - 1)
    boundary_pen = np.sum(v**2)
    
    # Overlap violations: distance >= sum of radii
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.linalg.norm(diff, axis=2)
    r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    v_overlap = np.maximum(0, r_sum - dists)
    np.fill_diagonal(v_overlap, 0)
    overlap_pen = np.sum(v_overlap**2)
    
    return obj + pen_weight * (boundary_pen + overlap_pen), boundary_pen + overlap_pen

def make_objective(n, pen_weight):
    """Factory for the objective function to satisfy top-level helper requirement."""
    def objective(vars_flat):
        return compute_objective_and_penalty(vars_flat, n, pen_weight)[0]
    return objective

def run_packing():
    np.random.seed(42)
    n = 26
    bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n
    
    best_centers = None
    best_radii = None
    best_sum = -1.0
    
    # Generate diverse initial configurations
    init_configs = []
    
    # 1. Regular Grid
    gx = np.linspace(0.15, 0.85, 6)
    gy = np.linspace(0.15, 0.85, 5)
    c1, c2 = [], []
    for y in gy:
        for x in gx:
            if len(c1) < 26:
                c1.append(x)
                c2.append(y)
    init_configs.append(np.column_stack([c1, c2]))
    
    # 2. Hexagonal Lattice
    h_c1, h_c2 = [], []
    for i in range(26):
        col = i % 6
        row = i // 6
        y = 0.15 + row * 0.14
        x = 0.15 + col * 0.14 + (0.07 if row % 2 == 1 else 0.0)
        if len(h_c1) < 26:
            h_c1.append(x)
            h_c2.append(y)
    init_configs.append(np.column_stack([h_c1, h_c2]))
    
    # 3. Randomly Perturbed Grid
    init_configs.append(init_configs[0] + np.random.normal(0, 0.02, (26, 2)))
    init_configs[-1] = np.clip(init_configs[-1], 0.05, 0.95)
    
    # Optimize each initialization
    for init_c in init_configs:
        vars_flat = np.concatenate([init_c.flatten(), np.ones(n) * 0.06])
        
        # Progressive penalty annealing for robust convergence
        for pen in [50, 200, 800]:
            res = minimize(make_objective(n, pen), vars_flat, method='L-BFGS-B', 
                           bounds=bounds, options={'maxiter': 3000, 'ftol': 1e-9, 'gtol': 1e-6})
            vars_flat = res.x
            
        centers = vars_flat[:2*n].reshape(n, 2)
        radii = vars_flat[2*n:]
        curr_sum = np.sum(radii)
        
        if curr_sum > best_sum:
            best_sum = curr_sum
            best_centers = centers.copy()
            best_radii = radii.copy()
            
    # Final high-precision polishing step
    if best_centers is not None:
        vars_flat = np.concatenate([best_centers.flatten(), best_radii])
        res = minimize(make_objective(n, 2000), vars_flat, method='L-BFGS-B', 
                       bounds=bounds, options={'maxiter': 5000, 'ftol': 1e-12, 'gtol': 1e-8})
        best_centers = res.x[:2*n].reshape(n, 2)
        best_radii = res.x[2*n:]
        best_sum = np.sum(best_radii)
        
    # Deterministic projection to guarantee strict validity within tolerance
    tolerance = 1e-10
    for i in range(n):
        for j in range(i+1, n):
            d = np.linalg.norm(best_centers[i] - best_centers[j])
            req = best_radii[i] + best_radii[j]
            if d < req - tolerance:
                overlap = req - d
                best_radii[i] -= overlap / 2
                best_radii[j] -= overlap / 2
                
    for i in range(n):
        x, y = best_centers[i]
        r = best_radii[i]
        if x - r < -tolerance: best_radii[i] = x
        if x + r > 1 + tolerance: best_radii[i] = 1 - x
        if y - r < -tolerance: best_radii[i] = y
        if y + r > 1 + tolerance: best_radii[i] = 1 - y
        
    best_radii = np.maximum(best_radii, 1e-6)
    best_sum = np.sum(best_radii)
    
    return best_centers, best_radii, best_sum
