import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def objective_func(v):
    """Minimize negative sum of radii to maximize total radius."""
    return -np.sum(v[52:])

def constraint_func(v):
    """Evaluate all inequality constraints >= 0."""
    n = N_CIRCLES
    centers = v[:52].reshape(n, 2)
    radii = v[52:]
    
    num_constraints = 4 * n + n * (n - 1) // 2
    c = np.empty(num_constraints)
    idx = 0
    
    # Boundary constraints: r <= x, r <= 1-x, r <= y, r <= 1-y
    for i in range(n):
        c[idx] = centers[i, 0] - radii[i]; idx += 1
        c[idx] = 1.0 - centers[i, 0] - radii[i]; idx += 1
        c[idx] = centers[i, 1] - radii[i]; idx += 1
        c[idx] = 1.0 - centers[i, 1] - radii[i]; idx += 1
        
    # Pairwise non-overlap constraints: dist >= r_i + r_j
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            c[idx] = np.sqrt(dx * dx + dy * dy) - radii[i] - radii[j]
            idx += 1
    return c

def run_packing():
    n = N_CIRCLES
    # Bounds: centers in [0,1], radii >= 0
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, None)] * n
    cons = {'type': 'ineq', 'fun': constraint_func}
    
    best_centers = None
    best_radii = None
    best_sum = -np.inf
    
    # Run optimization from multiple random starts
    for seed in range(10):
        np.random.seed(seed)
        centers_init = np.random.rand(n, 2) * 0.8 + 0.1
        radii_init = np.ones(n) * 0.03
        x0 = np.concatenate([centers_init.flatten(), radii_init])
        
        res = minimize(objective_func, x0, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 5000, 'ftol': 1e-11})
        
        if not np.isnan(res.fun):
            current_sum = -res.fun
            if current_sum > best_sum:
                best_sum = current_sum
                best_centers = res.x[:52].reshape(n, 2).copy()
                best_radii = res.x[52:].copy()

    if best_centers is None:
        best_centers = np.random.rand(n, 2) * 0.8 + 0.1
        best_radii = np.ones(n) * 0.01
        
    centers = best_centers
    radii = best_radii

    centers = np.clip(centers, 0.0, 1.0)
    radii = np.maximum(radii, 1e-12)
    
    # Fix boundary penetrations
    for i in range(n):
        cx, cy = centers[i]
        r = radii[i]
        if cx < r: cx = r
        if cx > 1 - r: cx = 1 - r
        if cy < r: cy = r
        if cy > 1 - r: cy = 1 - r
        centers[i] = [cx, cy]
        
    # Fix minor overlaps
    for _ in range(5):
        changed = False
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                min_dist = radii[i] + radii[j]

                    push = (min_dist - dist) * 0.5 + 1e-10

                        ux, uy = dx / dist, dy / dist
                        centers[i, 0] += ux * push
                        centers[i, 1] += uy * push

                        centers[j, 1] -= uy * push
                        changed = True

                        changed = True

    centers = np.clip(centers, 0.0, 1.0)
    return centers, radii, np.sum(radii)
