import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Pack 26 circles in a unit square to maximize the sum of radii.
    """
    N = 26
    
    # --- 1. Initialization ---
    # Generate a hexagonal grid configuration for initial centers
    centers = np.zeros((N, 2))
    radii = np.zeros(N)
    
    # We use an estimated radius to place points, then optimizer will adjust sizes.
    # A radius of 0.1 fits 5x5 grid. Hexagonal packing might allow slightly better density.
    # We start with a valid configuration (small radii) to ensure solver starts feasible.
    
    init_radius = 0.05 # Small enough to avoid overlap initially
    
    # Hexagonal grid generation
    # Spacing based on a hypothetical radius r_hyp
    r_hyp = 0.1
    dx = 2 * r_hyp
    dy = np.sqrt(3) * r_hyp
    
    points = []
    y = r_hyp
    row_idx = 0
    
    while len(points) < N:
        # Alternate row offsets
        if row_idx % 2 == 0:
            start_x = r_hyp
        else:
            start_x = r_hyp + dx / 2 # Offset by r_hyp
            
        x = start_x
        while x <= 1.0 - r_hyp + 1e-9:
            if len(points) < N:
                points.append([x, y])
            else:
                break
            x += dx
        
        y += dy
        row_idx += 1
        if y + r_hyp > 1.0 + 1e-9:
            break
            
    # If we didn't get enough points (unlikely with r_hyp=0.1), fill remaining with random or grid
    # With r_hyp=0.1, row 0 has 5, row 1 has 4 (0.2 to 0.8), row 2 has 5...
    # 5+4+5+4+5+4 = 27. So we get 26 easily.
    
    centers[:len(points)] = np.array(points[:N])
    # If somehow fewer, fill rest with random valid positions (edge case handling)
    if len(points) < N:
        for i in range(len(points), N):
            centers[i] = [np.random.uniform(0.1, 0.9), np.random.uniform(0.1, 0.9)]
            
    # Set initial radii to a safe small value
    radii[:] = init_radius
    
    # --- 2. Optimization Setup ---
    # Variables: [x1, y1, r1, x2, y2, r2, ...]
    # Total variables: 3 * N
    
    x0 = np.zeros(3 * N)
    for i in range(N):
        x0[3*i] = centers[i, 0]
        x0[3*i + 1] = centers[i, 1]
        x0[3*i + 2] = radii[i]
        
    # Bounds
    bounds = []
    for i in range(N):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 1.0)) # r (upper bound 1.0 is loose)
        
    def objective(vars):
        # Maximize sum of radii => Minimize negative sum
        r = vars[2::3]
        return -np.sum(r)
        
    def constraint_distance(vars):
        # Non-overlap constraints: dist >= r_i + r_j
        # We return the slack: dist - (r_i + r_j) >= 0
        # Since scipy expects function values >= 0 for inequality constraints (or <= 0 depending on spec)
        # minimize with method='SLSQP' expects cons type 'ineq' => func(x) >= 0
        
        # Vectorized computation
        x = vars[0::3]
        y = vars[1::3]
        r = vars[2::3]
        
        # We need to check all pairs. For N=26, 325 pairs.
        # Doing this inside objective/constraint evaluation might be slow if done naively in Python loop.
        # But 325 is small.
        
        slacks = []
        for i in range(N):
            xi, yi, ri = x[i], y[i], r[i]
            for j in range(i + 1, N):
                xj, yj, rj = x[j], y[j], r[j]
                dist_sq = (xi - xj)**2 + (yi - yj)**2
                sum_r = ri + rj
                # We need dist >= sum_r <=> dist^2 >= sum_r^2
                # But sqrt is expensive. 
                # However, constraint is dist - sum_r >= 0.
                # Using squared distance constraint: dist^2 - sum_r^2 >= 0 is equivalent ONLY if dist, sum_r >= 0.
                # Which they are.
                slack = dist_sq - sum_r**2
                slacks.append(slack)
        return np.array(slacks)

    # Boundary constraints:
    # x - r >= 0  => x - r >= 0
    # x + r <= 1  => 1 - x - r >= 0
    # y - r >= 0  => y - r >= 0
    # y + r <= 1  => 1 - y - r >= 0
    
    def constraint_boundaries(vars):
        x = vars[0::3]
        y = vars[1::3]
        r = vars[2::3]
        
        slacks = []
        for i in range(N):
            slacks.append(x[i] - r[i])
            slacks.append(1.0 - x[i] - r[i])
            slacks.append(y[i] - r[i])
            slacks.append(1.0 - y[i] - r[i])
        return np.array(slacks)

    # Define constraints for SLSQP
    cons = []
    cons.append({'type': 'ineq', 'fun': constraint_distance})
    cons.append({'type': 'ineq', 'fun': constraint_boundaries})
    
    # --- 3. Run Optimization ---
    # Using SLSQP
    try:
        result = minimize(
            objective,
            x0,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 1000, 'ftol': 1e-12, 'disp': False}
        )
        
        if result.success:
            x_opt = result.x
        else:
            # If failed, take best result anyway or keep initial
            x_opt = x0 # Fallback, though result.x usually holds best found
            
    except Exception as e:
        print(f"Optimization error: {e}")
        x_opt = x0 # Fallback
        
    # --- 4. Extract Results ---
    final_centers = np.zeros((N, 2))
    final_radii = np.zeros(N)
    
    for i in range(N):
        final_centers[i, 0] = x_opt[3*i]
        final_centers[i, 1] = x_opt[3*i + 1]
        final_radii[i] = x_opt[3*i + 2]
        
    sum_radii = np.sum(final_radii)
    
    # Post-processing / Validation check (internal)
    # Just to be safe, ensure radii are non-negative (optimizer might drift slightly)
    final_radii = np.maximum(final_radii, 0.0)
    
    # Re-center if any coordinate is out of bounds due to numerical noise
    final_centers[:, 0] = np.clip(final_centers[:, 0], 0.0, 1.0)
    final_centers[:, 1] = np.clip(final_centers[:, 1], 0.0, 1.0)
    
    # Adjust radii to strictly satisfy boundary constraints if clipped
    for i in range(N):
        x, y = final_centers[i]
        r = final_radii[i]
        max_r = min(x, 1-x, y, 1-y)
        if r > max_r:
            final_radii[i] = max(max_r, 0.0)
            
    return final_centers, final_radii, np.sum(final_radii)
