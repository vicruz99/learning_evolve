import numpy as np
import math
from scipy.optimize import differential_evolution

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    n = centers.shape[0]
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False
    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False
    
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False
            
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False
    return True

def get_objective(values):
    """
    Objective function to maximize: Sum of radii.
    values: flattened array of (x, y) coordinates for 26 circles.
    """
    n = 26
    centers = values.reshape(n, 2)
    
    # Clip centers to [0, 1] to ensure validity during optimization
    centers = np.clip(centers, 0, 1)
    
    radii = np.zeros(n)
    
    for i in range(n):
        x, y = centers[i]
        
        # Max radius based on boundaries
        r_boundary = min(x, 1 - x, y, 1 - y)
        
        # Max radius based on other circles
        r_neighbors = np.inf
        for j in range(n):
            if i == j:
                continue
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = math.sqrt(dx*dx + dy*dy)
            r_neighbor = 0.5 * dist
            if r_neighbor < r_neighbors:
                r_neighbors = r_neighbor
        
        # The valid radius is the minimum of constraints
        # If r_neighbors is 0 (overlap), radius becomes 0, penalizing the objective
        radii[i] = min(r_boundary, r_neighbors)
        
    return -np.sum(radii) # Negative because we maximize

def generate_initial_guess(seed):
    """
    Generate a hexagonal lattice initial guess.
    """
    n = 26
    rng = np.random.default_rng(seed)
    
    # Try to fit circles in a hexagonal pattern
    # Rows and columns estimation
    # Approx radius 0.1 -> 5x5 grid.
    # Hexagonal packing allows tighter fit.
    
    # Let's try a few random shifts/scales for the lattice
    # Base spacing
    r_est = 0.095 
    dx = 2 * r_est
    dy = r_est * math.sqrt(3)
    
    # Grid dimensions to cover roughly 26 circles
    # Maybe 6x5 or 5x6 area
    cols = 6
    rows = 5
    
    points = []
    y = r_est
    for r in range(rows):
        x = r_est
        # Offset every other row
        if r % 2 == 1:
            x += r_est # Shift by radius (half diameter)
        
        for c in range(cols):
            if len(points) >= 26:
                break
            points.append((x, y))
            x += dx
        y += dy
        
    if len(points) < 26:
        # Fallback to random if grid is too small
        points = [(rng.uniform(0, 1), rng.uniform(0, 1)) for _ in range(26)]
        
    # Convert to numpy array
    pts = np.array(points[:26])
    
    # Add some jitter
    pts += rng.uniform(-0.01, 0.01, pts.shape)
    pts = np.clip(pts, 0, 1)
    
    return pts.flatten()

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Run the packing optimization.
    """
    n = 26
    bounds = [(0, 1) for _ in range(2 * n)]
    
    best_score = np.inf
    best_solution = None
    
    # Run differential evolution multiple times with different seeds/starts
    # to find a good global optimum
    seeds = [42, 123, 456, 789, 999, 101, 202, 303]
    
    for seed in seeds:
        initial_guess = generate_initial_guess(seed)
        
        # Differential Evolution
        # popsize: number of trial vectors maintained at each step
        # maxiter: number of iterations
        # tol: termination tolerance on improvement
        
        try:
            result = differential_evolution(
                get_objective, 
                bounds, 
                strategy='best1bin', 
                maxiter=200, 
                popsize=30, 
                mutation=(0.5, 1), 
                recombination=0.9, 
                seed=seed,
                initial_pop=initial_guess.reshape(1, -1) # Provide one init guess, rest random? 
                # Note: DE in scipy doesn't take a single array as full pop, 
                # but we can pass seed to control randomness.
                # Actually, providing initial_pop as shape (M, D) is allowed.
            )
            
            if result.success or result.fun < best_score:
                best_score = result.fun
                best_solution = result.x
        except Exception as e:
            print(f"Optimization failed for seed {seed}: {e}")
            continue

    if best_solution is None:
        # Fallback to simple random
        best_solution = np.random.uniform(0, 1, 2 * n)

    # Extract centers
    centers = best_solution.reshape(n, 2)
    
    # Calculate final radii based on the optimized centers
    radii = np.zeros(n)
    for i in range(n):
        x, y = centers[i]
        r_boundary = min(x, 1 - x, y, 1 - y)
        r_neighbors = np.inf
        for j in range(n):
            if i == j: continue
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            r_neighbors = min(r_neighbors, 0.5 * dist)
        radii[i] = min(r_boundary, r_neighbors)
        
        # Safety clamp to 0
        if radii[i] < 0:
            radii[i] = 0
            
    sum_radii = np.sum(radii)
    
    # Final Validation
    is_valid = validate_packing(centers, radii)
    if not is_valid:
        # If invalid (should not happen with this construction), 
        # we might need to reduce radii slightly.
        # But min(r_boundary, 0.5*dist) guarantees validity.
        # The only issue is numerical precision.
        pass
        
    return centers, radii, float(sum_radii)
