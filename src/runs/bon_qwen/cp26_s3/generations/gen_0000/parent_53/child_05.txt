import numpy as np
from scipy.optimize import minimize

def loss_function(params, n):
    """
    Objective function to minimize.
    Minimizes: -sum(radii) + penalty_for_overlaps + penalty_for_boundaries
    
    params: flattened array [x1, y1, r1, x2, y2, r2, ..., x26, y26, r26]
    """
    centers = params[0:2*n].reshape(n, 2)
    radii = params[2*n:]
    
    # Objective: minimize negative sum of radii (maximize sum of radii)
    obj = -np.sum(radii)
    
    penalty_weight = 1000.0
    penalty = 0.0
    
    # Boundary penalties
    # Constraints: r <= x <= 1-r  =>  x-r >= 0, 1-r-x >= 0
    # Violation: max(0, r - x) and max(0, r - (1-x))
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        
        # Left/Right
        viol_left = max(0, r - x)
        viol_right = max(0, r - (1 - x))
        # Top/Bottom
        viol_bottom = max(0, r - y)
        viol_top = max(0, r - (1 - y))
        
        penalty += (viol_left**2 + viol_right**2 + viol_bottom**2 + viol_top**2)
    
    # Overlap penalties
    # Constraints: dist(c_i, c_j) >= r_i + r_j
    # Violation: max(0, r_i + r_j - dist)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            min_dist = radii[i] + radii[j]
            viol = max(0, min_dist - dist)
            penalty += viol**2
            
    return obj + penalty_weight * penalty

def run_packing():

    r_init = 0.075
    centers = []

    y = r_init
    h = r_init * np.sqrt(3) # Vertical spacing in hex packing
    
    for row_idx, count in enumerate(row_counts):
        x_start = r_init
        # Shift every other row by r_init
        if row_idx % 2 == 1:
            x_start += r_init
            
        for _ in range(count):
            centers.append([x_start, y])
            x_start += 2 * r_init
        y += h
        
    centers = np.array(centers)
    
    # 2. Construct initial parameter vector
    # [x1, y1, ..., x26, y26, r1, ..., r26]
    # Initialize radii to r_init

    bounds = [(0, 1), (0, 1)] * n + [(0, 0.5)] * n
    
    # 3. Optimize
    # Using L-BFGS-B for bounded variables
    res = minimize(
        loss_function,
        initial_params,
        args=(n,),
        method='L-BFGS-B',
        bounds=bounds,
        options={'ftol': 1e-12, 'gtol': 1e-8, 'maxiter': 1000}
    )
    
    # 4. Extract results
    best_params = res.x

    # Clamp radii to be strictly inside boundaries and non-overlapping
    # This handles any tiny numerical drift
    for i in range(n):
        x, y = best_centers[i]
        r = best_radii[i]
        
        # Boundary constraint
        dist_to_bound = min(x, 1-x, y, 1-y)
        if dist_to_bound < r:
            r = dist_to_bound
        
        best_radii[i] = r
