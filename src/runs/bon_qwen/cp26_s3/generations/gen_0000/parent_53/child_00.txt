def generate_hexagonal_initialization(n, square_size=1.0):

    row_height = square_size / 5.0 
    
    y = row_height
    row_idx = 0

        x = x_start
        while x + row_height/2 <= square_size:

                points.append([x, y])
            x += row_height
        y += row_height
