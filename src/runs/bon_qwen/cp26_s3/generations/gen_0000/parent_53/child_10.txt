def compute_cost(vars, n=26):
    # vars order: x0, y0, r0, x1, y1, r1, ...
    x = vars[0::3]
    y = vars[1::3]
    radii = vars[2::3]
    
    centers = np.column_stack((x, y))
    
    obj = -np.sum(radii)
    
    # Boundary penalties
    # r <= x => r - x <= 0. Violation max(0, r - x)
    # r <= 1-x => r + x - 1 <= 0. Violation max(0, r + x - 1)
    # r <= y => r - y <= 0. Violation max(0, r - y)
    # r <= 1-y => r + y - 1 <= 0. Violation max(0, r + y - 1)
    
    v1 = np.maximum(0, radii - x)
    v2 = np.maximum(0, radii + x - 1)
    v3 = np.maximum(0, radii - y)
    v4 = np.maximum(0, radii + y - 1)
    
    boundary_penalty = np.sum(v1**2 + v2**2 + v3**2 + v4**2)
    
    # Overlap penalties
    # Vectorized distance
    # centers shape (n, 2)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist_matrix = np.sqrt(np.sum(diff**2, axis=2))
    
    # Avoid 0 distance on diagonal causing issues with max(0, 2r - 0)
    # Set diagonal to a large value
    np.fill_diagonal(dist_matrix, np.inf)
    
    rad_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    violations = np.maximum(0, rad_sum - dist_matrix)
    
    # Sum of squared violations. 
    # Matrix is symmetric, so sum is 2 * sum_{i<j}.
    # We want to penalize each pair once? Or total energy?
    # Summing all entries is fine, just scales the weight.
    # Let's sum all and divide by 2 to be consistent with pair count.
    overlap_penalty = np.sum(violations**2) / 2.0
    
    cost = obj + 10000.0 * boundary_penalty + 10000.0 * overlap_penalty
    return cost
