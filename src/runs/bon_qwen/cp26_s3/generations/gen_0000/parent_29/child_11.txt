import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    """
    n = 26
    
    # Initial guess: 
    # 25 circles in a 5x5 grid with radius 0.09
    # 1 circle in a gap
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    r_grid = 0.09
    gap_r = 0.05
    
    idx = 0
    # Fill 5x5 grid
    for i in range(5):
        for j in range(5):
            centers[idx] = [0.1 + j * 0.2, 0.1 + i * 0.2]
            radii[idx] = r_grid
            idx += 1
            
    # Place 26th circle in a gap, e.g., between (0.1, 0.1), (0.3, 0.1), (0.1, 0.3), (0.3, 0.3)
    # Center of this gap is (0.2, 0.2)
    centers[idx] = [0.2, 0.2]
    radii[idx] = gap_r
    
    # Reshape to optimization vector: [x0, y0, r0, x1, y1, r1, ...]
    x0 = np.concatenate([centers.flatten(), radii])
    
    # Bounds: x, y in [0, 1], r in [0, 0.5]
    bounds = []
    for i in range(n):
        bounds.extend([
            (0.0, 1.0), # x
            (0.0, 1.0), # y
            (0.0, 0.5)  # r
        ])
    
    # Define constraints
    constraints = []
    
    # 1. Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    # x - r >= 0
    # 1 - r - x >= 0
    # y - r >= 0
    # 1 - r - y >= 0
    
    def constraint_boundary_x_lower(params):
        # x_i - r_i >= 0
        vals = []
        for i in range(n):
            x = params[3*i]
            r = params[3*i + 2]
            vals.append(x - r)
        return np.array(vals)

    def constraint_boundary_x_upper(params):
        # 1 - r_i - x_i >= 0
        vals = []
        for i in range(n):
            x = params[3*i]
            r = params[3*i + 2]
            vals.append(1.0 - r - x)
        return np.array(vals)

    def constraint_boundary_y_lower(params):
        # y_i - r_i >= 0
        vals = []
        for i in range(n):
            y = params[3*i + 1]
            r = params[3*i + 2]
            vals.append(y - r)
        return np.array(vals)

    def constraint_boundary_y_upper(params):
        # 1 - r_i - y_i >= 0
        vals = []
        for i in range(n):
            y = params[3*i + 1]
            r = params[3*i + 2]
            vals.append(1.0 - r - y)
        return np.array(vals)

    constraints.append({'type': 'ineq', 'fun': constraint_boundary_x_lower})
    constraints.append({'type': 'ineq', 'fun': constraint_boundary_x_upper})
    constraints.append({'type': 'ineq', 'fun': constraint_boundary_y_lower})
    constraints.append({'type': 'ineq', 'fun': constraint_boundary_y_upper})

    # 2. Overlap constraints: dist(i, j) - r_i - r_j >= 0
    # To improve performance, we can combine these into a single function returning an array
    # But scipy expects a function returning a float or an array? 
    # If returning array, it's treated as multiple constraints.
    
    def constraint_overlaps(params):
        vals = []
        for i in range(n):
            xi = params[3*i]
            yi = params[3*i+1]
            ri = params[3*i+2]
            for j in range(i + 1, n):
                xj = params[3*j]
                yj = params[3*j+1]
                rj = params[3*j+2]
                
                dist = np.sqrt((xi - xj)**2 + (yi - yj)**2)
                vals.append(dist - ri - rj)
        return np.array(vals)

    constraints.append({'type': 'ineq', 'fun': constraint_overlaps})

    # Objective: Maximize sum(radii) -> Minimize -sum(radii)
    def objective(params):
        # Extract radii
        r_vals = params[2::3] # Every 3rd element starting from index 2
        return -np.sum(r_vals)

    # Run optimization
    # SLSQP is suitable for constrained non-linear optimization
    res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints, 
                   options={'maxiter': 1000, 'ftol': 1e-9})
    
    if res.success:
        final_params = res.x
    else:
        # If optimization failed, return the initial guess (which is valid)
        final_params = x0
        
    # Extract results
    final_centers = np.zeros((n, 2))
    final_radii = np.zeros(n)
    
    for i in range(n):
        final_centers[i] = [final_params[3*i], final_params[3*i+1]]
        final_radii[i] = final_params[3*i+2]
        
    sum_radii = np.sum(final_radii)
    
    return final_centers, final_radii, sum_radii
