        # x >= r
        constraints.append({
            'type': 'ineq',

        })
        # 1 - x >= r  => 1 - x - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: 1.0 - v[3*i] - v[3*i+2]

                xi, yi, ri = v[3*i], v[3*i+1], v[3*i+2]

                dist_sq = dx*dx + dy*dy
