import numpy as np
import scipy.optimize
import math

def calculate_penalty(centers, radii):
    """
    Calculates the penalty for constraint violations.
    Returns 0 if valid, positive value otherwise.
    """
    n = centers.shape[0]
    penalty = 0.0
    
    # Boundary constraints: x - r >= 0, x + r <= 1, y - r >= 0, y + r <= 1
    # Violation amount is max(0, violation)
    # We square them for smoothness and stronger penalty
    
    # x boundaries
    penalty += np.sum(np.maximum(0, radii - centers[:, 0])**2) # x < r
    penalty += np.sum(np.maximum(0, centers[:, 0] + radii - 1)**2) # x > 1-r
    
    # y boundaries
    penalty += np.sum(np.maximum(0, radii - centers[:, 1])**2) # y < r
    penalty += np.sum(np.maximum(0, centers[:, 1] + radii - 1)**2) # y > 1-r
    
    # Overlap constraints: dist >= r_i + r_j
    # dist^2 >= (r_i + r_j)^2
    # Violation: (r_i + r_j)^2 - dist^2 > 0
    
    # Vectorized overlap calculation
    # Centers shape (n, 2)
    # dist_sq shape (n, n)
    dist_sq = np.sum((centers[:, np.newaxis, :] - centers[np.newaxis, :, :])**2, axis=2)
    # radii sum shape (n, n)
    radii_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    
    # Violation amount squared
    # We only care if dist < r_i + r_j
    violation = radii_sum**2 - dist_sq
    # Clip negative values (valid cases) to 0
    valid_violations = np.maximum(0, violation)
    
    # Sum up violations (excluding diagonal where i=j, violation is 0 anyway)
    penalty += np.sum(valid_violations)
    
    return penalty

def objective(params, n):
    """
    Objective function to minimize.
    params: flattened array [x1, y1, r1, x2, y2, r2, ...]
    """
    centers = params[:2*n].reshape(n, 2)
    radii = params[2*n:]
    
    # We want to maximize sum(radii), so minimize -sum(radii)
    # Add a large penalty for invalid configurations
    penalty = calculate_penalty(centers, radii)
    
    # Dynamic penalty weight? 
    # A large constant weight ensures we prioritize validity.
    # However, if penalty is 0, we just maximize radius.
    # If penalty > 0, we must reduce it.
    
    # Using a high multiplier to ensure constraints are met
    penalty_weight = 1e6 
    
    return -np.sum(radii) + penalty_weight * penalty
