import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # 1. Initialization: Hexagonal Grid
    # We estimate a base radius. For 25 circles in a 5x5 grid, r=0.1.
    # For 26, slightly smaller or denser packing. Start with 0.09.
    r_init = 0.09
    dx = 2 * r_init
    dy = np.sqrt(3) * r_init
    
    centers = []
    idx = 0
    # Generate hexagonal points until we have 26
    j = 0
    while idx < n:
        i = 0
        while idx < n:
            x = i * dx + (j % 2) * (dx / 2) + r_init
            y = j * dy + r_init
            # Perturb slightly
            x += np.random.uniform(-0.005, 0.005)
            y += np.random.uniform(-0.005, 0.005)
            
            if 0 <= x <= 1 and 0 <= y <= 1:
                centers.append([x, y])
                idx += 1
            i += 1
        j += 1
        
    centers = np.array(centers[:n])
    radii = np.full(n, r_init)
    
    # 2. Optimization Setup
    # Variables: [x1, y1, r1, x2, y2, r2, ...]
    x0 = np.concatenate([centers.flatten(), radii])
    
    # Bounds: r in [0, 0.5], x in [0, 1], y in [0, 1]
    bounds = []
    for _ in range(n):
        bounds.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])

    def objective(vars):
        # Maximize sum of radii (minimize negative sum)
        r = vars[2::3]
        return -np.sum(r)

    def constraints(vars):
        cons = []
        for i in range(n):
            xi, yi, ri = vars[3*i], vars[3*i+1], vars[3*i+2]
            # Boundary constraints
            cons.append(xi - ri) # x >= r
            cons.append(1.0 - (xi + ri)) # x + r <= 1
            cons.append(yi - ri) # y >= r
            cons.append(1.0 - (yi + ri)) # y + r <= 1
            
            # Overlap constraints
            for j in range(i + 1, n):
                xj, yj, rj = vars[3*j], vars[3*j+1], vars[3*j+2]
                # Use squared distance to avoid sqrt issues, though distance is better for smoothness
                # dist^2 >= (ri + rj)^2
                dx = xi - xj
                dy = yi - yj
                dist_sq = dx*dx + dy*dy
                cons.append(dist_sq - (ri + rj)**2)
        return cons

    # Run optimization
    result = minimize(objective, x0, method='SLSQP', bounds=bounds, 
                      constraints={'type': 'ineq', 'fun': constraints},
                      options={'maxiter': 500, 'ftol': 1e-9})

    # 3. Extract and Finalize
    final_centers = result.x.reshape(-1, 3)[:, :2]
    final_radii = result.x.reshape(-1, 3)[:, 2]

    # Validation and clipping to ensure numerical compliance
    # If any radius is negative, clip to 0
    final_radii = np.maximum(final_radii, 0.0)
    
    # Clip centers and adjust radii if they violate boundary
    for i in range(n):
        x, y = final_centers[i]
        r = final_radii[i]
        
        # Adjust radius to fit in square
        r = min(r, x, 1-x, y, 1-y)
        final_radii[i] = max(0.0, r) # Ensure non-negative

    # Final check for overlaps and slight correction (relaxation)
    # This loop ensures no circle is strictly overlapping others after optimization
    # It's a "push out" pass for safety.
    for _ in range(50):
        moved = False
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.linalg.norm(final_centers[i] - final_centers[j])
                min_dist = final_radii[i] + final_radii[j]
                if dist < min_dist and dist > 1e-9:
                    # Move apart
                    vec = (final_centers[j] - final_centers[i]) / dist
                    shift = (min_dist - dist) / 2 + 1e-6
                    final_centers[i] -= vec * shift
                    final_centers[j] += vec * shift
                    moved = True
        
        # Re-clamp to square
        for i in range(n):
            x, y = final_centers[i]
            r = final_radii[i]
            # Clamp center
            final_centers[i] = np.clip(final_centers[i], [r, r], [1-r, 1-r])
            
    sum_radii = np.sum(final_radii)
    
    return final_centers, final_radii, sum_radii
