import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def objective(params, N):
    """
    Objective function: maximize sum of radii, penalize constraint violations.
    Variables: [x0, y0, r0, x1, y1, r1, ..., xN, yN, rN]
    """
    c = params[:2*N].reshape(N, 2)
    r = params[2*N:]
    
    # Primary objective: maximize sum of radii (minimize negative sum)
    obj = -np.sum(r)
    
    # Penalty for constraint violations
    pen = 0.0
    
    # Boundary constraints: circle must be inside [0,1]x[0,1]
    for i in range(N):
        x, y = c[i]
        ri = r[i]
        # Left boundary: x >= r
        if x < ri:
            pen += (x - ri)**2
        # Right boundary: x <= 1 - r
        if x > 1.0 - ri:
            pen += (x - (1.0 - ri))**2
        # Bottom boundary: y >= r
        if y < ri:
            pen += (y - ri)**2
        # Top boundary: y <= 1 - r
        if y > 1.0 - ri:
            pen += (y - (1.0 - ri))**2
            
    # Non-overlap constraints: dist(c_i, c_j) >= r_i + r_j
    # Using squared distance for efficiency and smoothness
    for i in range(N):
        for j in range(i + 1, N):
            dist_sq = np.sum((c[i] - c[j])**2)
            min_dist_sq = (r[i] + r[j])**2
            if dist_sq < min_dist_sq:
                pen += (dist_sq - min_dist_sq)**2
                
    # Weight for penalty term
    PENALTY_WEIGHT = 5000.0
    return obj + PENALTY_WEIGHT * pen

def get_bounds(N):
    """
    Returns bounds for the optimization variables.
    Variables order: x0, y0, r0, x1, y1, r1, ...
    """
    bounds = []
    for _ in range(N):
        # x and y coordinates in [0, 1]
        bounds.append((0.0, 1.0))
        bounds.append((0.0, 1.0))
        # radius in [0, 0.5]
        bounds.append((0.0, 0.5))
    return bounds

def run_packing():
    """
    Pack 26 circles in a unit square to maximize sum of radii.
    """
    N = N_CIRCLES
    
    # Initial configuration: Hexagonal grid pattern
    centers = np.zeros((N, 2))
    radii = np.full(N, 0.05)
    
    # Row structure: 5, 6, 5, 6, 4 circles sum to 26
    # This staggered arrangement approximates optimal packing density
    row_counts = [5, 6, 5, 6, 4]
    y_pos = 0.12
    idx = 0
    
    for i, count in enumerate(row_counts):
        x_start = 0.12 if i % 2 == 1 else 0.12
        # Calculate spacing to fit within a safe margin of the square
        dx = 0.76 / (count - 1) if count > 1 else 0.0
        for j in range(count):
            centers[idx, 0] = x_start + j * dx
            centers[idx, 1] = y_pos
            idx += 1
        y_pos += 0.17
        
    # Flatten parameters: [x0, y0, r0, x1, y1, r1, ...]
    x0 = np.concatenate([centers.flatten(), radii])
    
    # Run optimization
    # L-BFGS-B is efficient for bound-constrained problems
    res = minimize(objective, x0, args=(N,), method='L-BFGS-B', 
                   bounds=get_bounds(N), options={'maxiter': 10000, 'ftol': 1e-15})
                   
    c_opt = res.x[:2*N].reshape(N, 2)
    r_opt = res.x[2*N:]
    
    # Ensure radii are non-negative
    r_opt = np.maximum(r_opt, 0.0)
    
    # Sum of radii
    sum_radii = np.sum(r_opt)
    
    return c_opt, r_opt, sum_radii
