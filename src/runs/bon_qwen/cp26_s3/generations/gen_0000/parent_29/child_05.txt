import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # 1. Initialization: Hexagonal Grid
    # We place circles in a hexagonal pattern.
    # Spacing s determines initial density. 
    # Max radius in unit square is 0.5. 
    # We start with a conservative radius.
    
    r_init = 0.06
    # Hexagonal spacing: distance between centers in a row is 2*r
    # Vertical distance is sqrt(3)*r
    # We use a slightly larger spacing to ensure no initial overlap
    s = 2.2 * r_init 
    
    centers = []
    radii = []
    
    row = 0
    y = r_init
    # Generate grid points
    while len(centers) < n:
        # Offset for odd rows
        x_offset = (row % 2) * (s / 2)
        x = r_init + x_offset
        
        while x <= 1 - r_init and len(centers) < n:
            centers.append([x, y])
            x += s
        y += s * np.sqrt(3) / 2
        row += 1
        
    # Fill remaining spots if grid didn't produce enough (unlikely with these params)
    while len(centers) < n:
        # Add small circles in random valid positions if needed
        # Just picking a valid spot near existing ones or random
        cx = 0.5
        cy = 0.5
        # Simple fallback
        centers.append([cx, cy])
        radii.append(0.01) # Placeholder
        
    centers = np.array(centers[:n])
    radii = np.full(n, r_init)
    
    # 2. Setup Optimization
    # Variables: [x0, y0, r0, x1, y1, r1, ..., x25, y25, r25]
    # Total variables: 3 * n = 78
    
    p0 = np.zeros(3 * n)
    for i in range(n):
        p0[3*i] = centers[i, 0]
        p0[3*i+1] = centers[i, 1]
        p0[3*i+2] = radii[i]
        
    # Bounds: x, y in [0, 1], r in [0, 0.5]
    bounds = []
    for i in range(n):
        bounds.extend([(0, 1), (0, 1), (0, 0.5)])
        
    # Objective function with penalty
    def objective(p, K):
        # Unpack
        # Reshaping is faster than loop for large n, but loop is fine for 26
        # Using view/slicing for speed
        cs = p.reshape(n, 3)
        xc = cs[:, 0]
        yc = cs[:, 1]
        rc = cs[:, 2]
        
        # Objective: Maximize sum of radii -> Minimize -sum
        val = -np.sum(rc)
        
        # Penalties
        penalty = 0.0
        
        # 1. Boundary Penalties
        # Constraints: r <= x <= 1-r  =>  x-r >= 0 AND x+r <= 1
        # r <= y <= 1-y  =>  y-r >= 0 AND y+r <= 1
        
        # Violation amounts (positive if violated)
        # Left: r - x > 0
        v_left = np.maximum(0, rc - xc)
        # Right: r - (1-x) > 0 => r + x - 1 > 0
        v_right = np.maximum(0, rc + xc - 1)
        # Bottom: r - y > 0
        v_bottom = np.maximum(0, rc - yc)
        # Top: r - (1-y) > 0 => r + y - 1 > 0
        v_top = np.maximum(0, rc + yc - 1)
        
        penalty += np.sum(v_left**2)
        penalty += np.sum(v_right**2)
        penalty += np.sum(v_bottom**2)
        penalty += np.sum(v_top**2)
        
        # 2. Overlap Penalties
        # Constraint: dist(i, j) >= ri + rj
        # Violation: ri + rj - dist > 0
        
        # Vectorized distance calculation
        # diff_ij = [xi - xj, yi - yj]
        # dist = sqrt(diff_x^2 + diff_y^2)
        
        # To avoid O(N^2) loop overhead in python, we can use broadcasting
        # But for N=26, loop is acceptable and memory efficient.
        
        # Using broadcasting for efficiency
        # Shape (n, 1, 2) - (1, n, 2) -> (n, n, 2)
        # This creates a matrix of differences
        diff = cs[:, np.newaxis, :2] - cs[np.newaxis, :, :2] # (n, n, 2)
        dist_sq = np.sum(diff**2, axis=2) # (n, n)
        # Distance matrix (symmetric, 0 on diag)
        dist = np.sqrt(dist_sq + 1e-12) # epsilon to avoid div by 0 if needed, though not used here
        
        # Radius sum matrix
        r_sum = rc[:, np.newaxis] + rc[np.newaxis, :] # (n, n)
        
        # Overlap amount matrix
        overlap = r_sum - dist
        # We only care where overlap > 0
        # Also only upper triangle to avoid double counting, though squaring makes it symmetric anyway
        # Using np.maximum on the whole matrix is fine, but let's just sum the positive parts
        # Note: overlap[i,i] = 2*ri - 0 = 2ri > 0. This is a problem!
        # Distance is 0 on diagonal. r_sum is 2*ri.
        # We must ignore diagonal.
        
        np.fill_diagonal(overlap, 0)
        
        # Positive overlaps
        pos_overlap = np.maximum(0, overlap)
        penalty += np.sum(pos_overlap**2)
        
        return val + K * penalty

    # 3. Optimization Loop
    current_p = p0.copy()
    K = 100.0
    
    # Run multiple phases with increasing penalty
    for _ in range(15):
        # Minimize
        # L-BFGS-B is good for bound constrained problems
        res = minimize(objective, current_p, args=(K,), method='L-BFGS-B', 
                       bounds=bounds, options={'maxiter': 1000, 'ftol': 1e-12, 'gtol': 1e-8})
        current_p = res.x
        
        # Increase penalty weight to enforce constraints strictly
        K *= 2.5
        
    # 4. Extract Results
    cs_final = current_p.reshape(n, 3)
    centers_res = cs_final[:, :2]
    radii_res = cs_final[:, 2]
    
    # 5. Validation / Cleaning
    # Ensure non-negative radii (bounds should handle, but just in case)
    radii_res = np.maximum(radii_res, 0)
    
    # Check validity and maybe scale down if needed
    # Although K should be high enough, numerical errors might persist.
    # We can compute a scaling factor alpha <= 1.
    
    # Check boundary
    alpha = 1.0
    for i in range(n):
        x, y = centers_res[i]
        r = radii_res[i]
        if r > 1e-9:
            # x - r >= 0 => r <= x => alpha * r <= x => alpha <= x/r
            alpha = min(alpha, x / r)
            # x + r <= 1 => r <= 1-x => alpha * r <= 1-x => alpha <= (1-x)/r
            alpha = min(alpha, (1 - x) / r)
            # y - r >= 0
            alpha = min(alpha, y / r)
            # y + r <= 1
            alpha = min(alpha, (1 - y) / r)
            
    # Check overlaps
    # dist >= r_i + r_j => alpha(r_i + r_j) <= dist => alpha <= dist / (r_i + r_j)
    for i in range(n):
        for j in range(i + 1, n):
            ri = radii_res[i]
            rj = radii_res[j]
            if ri + rj > 1e-9:
                dist = np.sqrt((centers_res[i,0] - centers_res[j,0])**2 + 
                               (centers_res[i,1] - centers_res[j,1])**2)
                alpha = min(alpha, dist / (ri + rj))
                
    # Apply scaling if necessary (alpha should be very close to 1)
    if alpha < 1.0:
        radii_res = radii_res * alpha
        # We could also slightly move centers, but scaling radii is safer for validity.
        # However, scaling radii reduces the objective.
        # Given the high K, alpha should be >= 1.0 (or very close, e.g. 0.99999999)
        # If alpha is significantly less than 1, the optimization failed.
        # But with K=100*2.5^15 ~ 1.5e9, it should be very tight.
        # Let's trust the optimizer unless alpha is very small.
        # Actually, if alpha < 1, we MUST scale to be valid.
        
    sum_radii = np.sum(radii_res)
    
    return centers_res, radii_res, sum_radii
