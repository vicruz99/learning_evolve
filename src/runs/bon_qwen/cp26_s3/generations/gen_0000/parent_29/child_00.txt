

Looking at this problem, I need to pack 26 circles in a unit square to maximize the sum of radii. The target is 2.636.

**Key Insight:** The target sum of 2.636 for 26 circles suggests average radius ~0.1014, which is better than a simple 5×5 grid (r=0.1, sum=2.5 for 25 circles). This means we need a denser packing than a simple grid — likely a hexagonal arrangement or hybrid pattern with unequal radii.

**Strategy:**
1. Try multiple initial configurations (hexagonal lattices with different spacings, grid patterns)
2. Use scipy's SLSQP optimizer to jointly optimize centers and radii
3. Validate each solution against constraints
4. Use multi-start optimization to avoid local optima
5. Apply an iterative radius computation to refine solutions

The hexagonal lattice is denser than square grid (packing density ~0.907 vs ~0.785), allowing larger radii. I'll also try unequal radii where circles near corners can be larger and gap-filling circles smaller.

```python
import numpy as np
from scipy.optimize import minimize


def hexagonal_init(n, d, y_start=None):
    """Generate n circle centers in hexagonal pattern with spacing d."""
    r_init = d / 2
    centers = []
    y = r_init if y_start is None else y_start
    row = 0
    while len(centers) < n:
        x = r_init
        if row % 2 == 1:
            x += d / 2
        while x <= 1 - r_init and len(centers) < n:
            centers.append([x, y])
            x += d
        y += d * np.sqrt(3) / 2
        row += 1
    return np.array(centers[:n])


def grid_init(n, spacing):
    """Generate n circle centers in a square grid."""
    centers = []
    x = spacing / 2
    while x <= 1 - spacing / 2 and len(centers) < n:
        y = spacing / 2
        while y <= 1 - spacing / 2 and len(centers) < n:
            centers.append([x, y])
            y += spacing
        x += spacing
    return np.array(centers[:n])


def my_objective(params, n):
    """Negative sum of radii (we minimize this)."""
    radii = params[2 * n:]
    return -np.sum(radii)


def my_constraint(params, n):
    """All constraints: non-overlap and boundary."""
    centers = params[:2 * n].reshape(n, 2)
    radii = params[2 * n:]
    constraints = []
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            constraints.append(dist - radii[i] - radii[j])
    for i in range(n):
        x, y = centers[i, 0], centers[i, 1]
        r = radii[i]
        constraints.append(x - r)
        constraints.append(1 - x - r)
        constraints.append(y - r)
        constraints.append(1 - y - r)
    return np.array(constraints)


def validate_solution(centers, radii, n):
    """Check if solution satisfies all constraints."""
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-10 or x + r > 1 + 1e-10:
            return False
        if y - r < -1e-10 or y + r > 1 + 1e-10:
            return False
        if r < -1e-10:
            return False
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            if dist < radii[i] + radii[j] - 1e-10:
                return False
    return True


def refine_radii(centers, n, iterations=200):
    """Iteratively compute feasible radii for fixed centers."""
    radii = np.ones(n) * 0.05
    for _ in range(iterations):
        new_radii = np.zeros(n)
        for i in range(n):
            x, y = centers[i]
            max_r = min(x, 1 - x, y, 1 - y)
            for j in range(n):
                if i != j:
                    dx = centers[i, 0] - centers[j, 0]
                    dy = centers[i, 1] - centers[j, 1]
                    dist = np.sqrt(dx * dx + dy * dy)
                    max_r = min(max_r, dist - radii[j])
            new_radii[i] = max(0, max_r)
        if np.max(np.abs(new_radii - radii)) < 1e-12:
            break
        radii = new_radii
    return radii


def run_packing():
    n = 26

    best_sum = 0
    best_centers = None
    best_radii = None

    initializations = []

    for d in [0.19, 0.20, 0.205, 0.21, 0.215, 0.22, 0.23]:
        centers = hexagonal_init(n, d)
        initializations.append(centers)

    for d in [0.195, 0.205, 0.215, 0.225]:
        centers = hexagonal_init(n, d, y_start=0.08)
        initializations.append(centers)

    for spacing in [0.19, 0.20, 0.21, 0.22, 0.23]:
        centers = grid_init(n, spacing)
        initializations.append(centers)

    for seed in range(8):
        np.random.seed(seed)
        centers = np.random.uniform(0.12, 0.88, (n, 2))
        initializations.append(centers)

    for centers_init in initializations:
        radii_init = np.ones(n) * 0.04
        params0 = np.concatenate([centers_init.flatten(), radii_init])

        bounds = [(0, 1)] * (2 * n) + [(0, 0.5)] * n
        cons = {'type': 'ineq', 'fun': my_constraint, 'args': (n,)}

        try:
            result = minimize(
                my_objective, params0, args=(n,),
                method='SLSQP', bounds=bounds, constraints=cons,
                options={'maxiter': 3000, 'ftol': 1e-14}
            )

            centers_opt = result.x[:2 * n].reshape(n, 2)
            radii_opt = result.x[2 * n:]
            radii_opt = np.maximum(radii_opt, 1e-10)

            if validate_solution(centers_opt, radii_opt, n):
                current_sum = np.sum(radii_opt)
                if current_sum > best_sum:
                    best_sum = current_sum
                    best_centers = centers_opt.copy()
                    best_radii = radii_opt.copy()
        except Exception:
            pass

    # Additional refinement pass on best solution
    if best_centers is not None:
        for centers_init in [best_centers]:
            radii_refined = refine_radii(centers_init, n, iterations=300)
            params0 = np.concatenate([centers_init.flatten(), radii_refined])

            bounds = [(0, 1)] * (2 * n) + [(0, 0.5)] * n
            cons = {'type': 'ineq', 'fun': my_constraint, 'args': (n,)}

            try:
                result = minimize(
                    my_objective, params0, args=(n,),
                    method='SLSQP', bounds=bounds, constraints=cons,
                    options={'maxiter': 3000, 'ftol': 1e-14}
                )

                centers_opt = result.x[:2 * n].reshape(n, 2)
                radii_opt = result.x[2 * n:]
                radii_opt = np.maximum(radii_opt, 1e-10)

                if validate_solution(centers_opt, radii_opt, n):
                    current_sum = np.sum(radii_opt)
                    if current_sum > best_sum:
                        best_sum = current_sum
                        best_centers = centers_opt.copy()
                        best_radii = radii_opt.copy()
            except Exception:
                pass

    # Second refinement with refined radii
    if best_centers is not None:
        radii_refined = refine_radii(best_centers, n, iterations=500)
        params0 = np.concatenate([best_centers.flatten(), radii_refined])

        bounds = [(0, 1)] * (2 * n) + [(0, 0.5)] * n
        cons = {'type': 'ineq', 'fun': my_constraint, 'args': (n,)}

        try:
            result = minimize(
                my_objective, params0, args=(n,),
                method='SLSQP', bounds=bounds, constraints=cons,
                options={'maxiter': 5000, 'ftol': 1e-14}
            )

            centers_opt = result.x[:2 * n].reshape(n, 2)
            radii_opt = result.x[2 * n:]
            radii_opt = np.maximum(radii_opt, 1e-10)

            if validate_solution(centers_opt, radii_opt, n):
                current_sum = np.sum(radii_opt)
                if current_sum > best_sum:
                    best_sum = current_sum
                    best_centers = centers_opt.copy()
                    best_radii = radii_opt.copy()
        except Exception:
            pass

    if best_centers is not None:
        # Clamp to ensure validity
        for i in range(n):
            x, y = best_centers[i]
            r = best_radii[i]
            best_radii[i] = min(r, x, 1 - x, y, 1 - y, 0.5)
            best_radii[i] = max(best_radii[i], 1e-10)

        return best_centers, best_radii, np.sum(best_radii)

    # Fallback: simple grid
    centers = np.zeros((n, 2))
    idx = 0
    for i in range(5):
        for j in range(5):
            if idx < n:
                centers[idx] = [0.1 + j * 0.2, 0.1 + i * 0.2]
                idx += 1
    centers[25] = [0.5, 0.5]
    radii = np.ones(n) * 0.06
    return centers, radii, np.sum(radii)
```
