import numpy as np
from scipy.optimize import minimize

def compute_radii(positions, n):
    """Compute optimal radii given fixed positions."""
    radii = np.ones(n) * 1.0
    for iteration in range(50):
        new_radii = np.zeros(n)
        for i in range(n):
            max_r = min(positions[i][0], 1 - positions[i][0],
                       positions[i][1], 1 - positions[i][1])
            for j in range(n):
                if i != j:
                    dist = np.sqrt(np.sum((positions[i] - positions[j])**2))
                    max_r = min(max_r, dist - radii[j])
            new_radii[i] = max(0.0001, max_r)
        radii = new_radii
    return radii

def boundary_penalty(x_flat, positions, radii, n):
    """Compute penalty for boundary violations."""
    pos = x_flat.reshape((n, 2))
    penalty = 0.0
    for i in range(n):
        r = radii[i]
        if pos[i][0] < r:
            penalty += (r - pos[i][0]) ** 2
        if pos[i][0] > 1 - r:
            penalty += (pos[i][0] - (1 - r)) ** 2
        if pos[i][1] < r:
            penalty += (r - pos[i][1]) ** 2
        if pos[i][1] > 1 - r:
            penalty += (pos[i][1] - (1 - r)) ** 2
    return penalty

def overlap_penalty(x_flat, positions, radii, n):
    """Compute penalty for overlapping circles."""
    pos = x_flat.reshape((n, 2))
    penalty = 0.0
    for i in range(n):
        for j in range(i + 1, n):
            dx = pos[i][0] - pos[j][0]
            dy = pos[i][1] - pos[j][1]
            dist = np.sqrt(dx * dx + dy * dy)
            if dist < radii[i] + radii[j]:
                penalty += (radii[i] + radii[j] - dist) ** 2
    return penalty

def total_penalty(x_flat, radii, n):
    """Total penalty function for position optimization."""
    return boundary_penalty(x_flat, None, radii, n) + overlap_penalty(x_flat, None, radii, n)

def force_based_optimize(positions, radii, n, iterations=500):
    """Use force-based method to optimize positions."""
    lr = 0.05
    for it in range(iterations):
        decayed_lr = lr * (0.99 ** it)
        
        forces = np.zeros((n, 2))
        
        # Repulsive forces between circles
        for i in range(n):
            for j in range(i + 1, n):
                dx = positions[i][0] - positions[j][0]
                dy = positions[i][1] - positions[j][1]
                dist = np.sqrt(dx * dx + dy * dy)
                min_dist = radii[i] + radii[j]
                
                if dist < min_dist + 0.03 and dist > 0.0001:
                    repulsion = (min_dist + 0.03 - dist) * 5.0
                    fx = (dx / dist) * repulsion
                    fy = (dy / dist) * repulsion
                    forces[i][0] += fx
                    forces[i][1] += fy
                    forces[j][0] -= fx
                    forces[j][1] -= fy
        
        # Boundary forces
        for i in range(n):
            r = radii[i]
            if positions[i][0] < r + 0.01:
                forces[i][0] += (r + 0.01 - positions[i][0]) * 10
            if positions[i][0] > 1 - r - 0.01:
                forces[i][0] -= (positions[i][0] - (1 - r - 0.01)) * 10
            if positions[i][1] < r + 0.01:
                forces[i][1] += (r + 0.01 - positions[i][1]) * 10
            if positions[i][1] > 1 - r - 0.01:
                forces[i][1] -= (positions[i][1] - (1 - r - 0.01)) * 10
        
        positions += forces * decayed_lr
        positions = np.clip(positions, 0.001, 0.999)
    
    return positions

def hexagonal_init(n, spacing=0.18, offset=0.09):
    """Initialize positions in hexagonal pattern."""
    positions = np.zeros((n, 2))
    idx = 0
    
    for row in range(8):
        y = offset + row * spacing * np.sqrt(3) / 2
        x_offset = spacing / 2 if row % 2 == 1 else 0
        
        for col in range(6):
            if idx >= n:
                break
            x = offset + col * spacing + x_offset
            if 0 <= x <= 1 and 0 <= y <= 1:
                positions[idx] = [x, y]
                idx += 1
    
    # Fill remaining
    while idx < n:
        row = idx // 6
        col = idx % 6
        positions[idx] = [0.1 + col * 0.18, 0.1 + row * 0.18]
        idx += 1
    
    return positions

def grid_init(n, spacing=0.2, offset=0.1):
    """Initialize positions in grid pattern."""
    positions = np.zeros((n, 2))
    idx = 0
    
    for i in range(6):
        for j in range(6):
            if idx >= n:
                break
            x = offset + i * spacing
            y = offset + j * spacing
            if x <= 1 and y <= 1:
                positions[idx] = [x, y]
                idx += 1
    
    while idx < n:
        positions[idx] = [0.5, 0.5]
        idx += 1
    
    return positions

def run_packing():
    np.random.seed(42)
    n = 26
    
    best_sum_radii = 0
    best_positions = None
    best_radii = None
    
    # Try multiple initialization strategies
    init_configs = []
    
    # Config 1: Hexagonal with different spacings
    for spacing in [0.17, 0.18, 0.19]:
        pos = hexagonal_init(n, spacing=spacing)
        init_configs.append(pos.copy())
    
    # Config 2: Grid
    for spacing in [0.18, 0.19, 0.2]:
        pos = grid_init(n, spacing=spacing)
        init_configs.append(pos.copy())
    
    # Config 3: Random perturbations of hexagonal
    for seed in [0, 1, 2]:
        np.random.seed(seed)
        pos = hexagonal_init(n, spacing=0.18)
        pos += np.random.uniform(-0.02, 0.02, size=pos.shape)
        pos = np.clip(pos, 0.05, 0.95)
        init_configs.append(pos.copy())
    
    for config_idx, init_pos in enumerate(init_configs):
        positions = init_pos.copy()
        radii = compute_radii(positions, n)
        
        # Iterative optimization
        for outer_iter in range(60):
            # Update radii
            radii = compute_radii(positions, n)
            
            # Force-based position optimization
            positions = force_based_optimize(positions, radii, n, iterations=200)
            
            # Local optimization using scipy
            x0 = positions.flatten()
            
            def obj_func(x_flat):
                return total_penalty(x_flat, radii, n)
            
            bounds = [(0.001, 0.999) for _ in range(2 * n)]
            
            result = minimize(obj_func, x0, method='L-BFGS-B', bounds=bounds,
                            options={'maxiter': 800, 'ftol': 1e-14})
            positions = result.x.reshape((n, 2))
        
        # Final radius computation
        radii = compute_radii(positions, n)
        current_sum = np.sum(radii)
        
        if current_sum > best_sum_radii:
            best_sum_radii = current_sum
            best_positions = positions.copy()
            best_radii = radii.copy()
    
    # Final cleanup and validation
    best_positions = np.clip(best_positions, 0.001, 0.999)
    best_radii = compute_radii(best_positions, n)
    
    return best_positions, best_radii, np.sum(best_radii)
