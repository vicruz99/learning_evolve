from scipy.optimize import minimize


def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    best_sum = 0.0
    best_centers = None
    best_radii = None

        x0 = np.concatenate([centers.flatten(), radii])

        # Multi-phase optimization with increasing penalty weight
        x_current = x0.copy()
        for pw in [100.0, 500.0, 2000.0, 10000.0, 50000.0]:
            result = minimize(
                partial_objective,
                x_current,
                args=(n, pw),
                method="L-BFGS-B",
                bounds=bounds,
                options={"maxiter": 10000, "ftol": 1e-15, "gtol": 1e-10},
            )
            x_current = result.x

        # Extract and project to feasible region
        centers_opt = x_current[: 2 * n].reshape(n, 2)
        radii_opt = x_current[2 * n :]

        # Ensure non-negative radii
        radii_opt = np.maximum(radii_opt, 0.0)

        # Project centers to ensure circles stay in bounds
        for i in range(n):
            centers_opt[i, 0] = max(radii_opt[i], min(1.0 - radii_opt[i], centers_opt[i, 0]))
            centers_opt[i, 1] = max(radii_opt[i], min(1.0 - radii_opt[i], centers_opt[i, 1]))

        current_sum = np.sum(radii_opt)

        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers_opt.copy()
            best_radii = radii_opt.copy()

    return best_centers, best_radii, best_sum


def partial_objective(x_opt, n, penalty_weight):
    """Objective with specified penalty weight - no closures."""
    return penalized_objective(x_opt, n, penalty_weight)


def penalized_objective(x_opt, n, penalty_weight):
    """
    Negative sum of radii plus penalties for constraint violations.
    
    Parameters
    ----------
    x_opt : array of shape (2*n + n,) - [x1, y1, x2, y2, ..., xn, yn, r1, r2, ..., rn]
    n : int - number of circles
    penalty_weight : float - weight for constraint violations
    """
    centers = x_opt[: 2 * n].reshape(n, 2)
    radii = x_opt[2 * n :]

    # Primary objective: maximize sum of radii (negated for minimization)
    obj = -np.sum(radii)

    pw = penalty_weight

    # Boundary constraints (vectorized for efficiency)
    x = centers[:, 0]
    y = centers[:, 1]
    r = radii

    # x - r >= 0 means r - x <= 0, violation when r - x > 0
    v1 = np.maximum(0.0, r - x)
    obj += pw * np.sum(v1 ** 2)

    # x + r <= 1, violation when x + r - 1 > 0
    v2 = np.maximum(0.0, x + r - 1.0)
    obj += pw * np.sum(v2 ** 2)

    # y - r >= 0
    v3 = np.maximum(0.0, r - y)
    obj += pw * np.sum(v3 ** 2)

    # y + r <= 1
    v4 = np.maximum(0.0, y + r - 1.0)
    obj += pw * np.sum(v4 ** 2)

    # Overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.hypot(dx, dy)
            overlap = radii[i] + radii[j] - dist
            if overlap > 0.0:
                obj += pw * overlap ** 2

    return obj


def create_initial_hexagonal(n, trial):
    """
    Create initial hexagonal grid configuration with trial-specific perturbation.
    
    Uses 5 rows with counts [6, 5, 6, 5, 4] = 26 circles total.
    """
    row_counts = [6, 5, 6, 5, 4]

    centers = []
    base_spacing = 1.0 / 6.0

    # Deterministic perturbation based on trial number
    perturbation = trial * 0.0003
    phase = trial * 0.5

    for row_idx, count in enumerate(row_counts):
        if len(centers) >= n:
            break

        # Hexagonal vertical spacing
        y = (row_idx + 0.5) * base_spacing * np.sqrt(3.0) / 2.0 + perturbation * np.sin(phase + row_idx)
        # Offset alternating rows for hexagonal pattern
        offset = base_spacing / 2.0 if row_idx % 2 == 1 else 0.0

        for col in range(count):
            if len(centers) >= n:
                break
            x = offset + (col + 0.5) * base_spacing + perturbation * np.cos(phase + col)
            centers.append([x, y])

    centers = np.array(centers[:n])
    # Start with reasonable radii (about 40% of half-spacing to allow growth)
    radii = np.ones(n) * 0.035

    return centers, radii
