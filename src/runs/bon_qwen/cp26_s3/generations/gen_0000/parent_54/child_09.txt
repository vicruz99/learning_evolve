1. **Radius Optimization (LP):** For a fixed set of centers, finding the optimal radii is a Linear Programming (LP) problem. We maximize $\sum r_i$ subject to $r_i + r_j \leq \text{dist}(i, j)$ and boundary constraints.
2. **Center Optimization (Nelder-Mead):** Since the optimal sum of radii as a function of centers is non-smooth, we use the Nelder-Mead simplex method to iteratively perturb center positions.

import numpy as np
from scipy.optimize import linprog, minimize

N = 26
SQ_SIZE = 1.0

def solve_lp(centers):
    """
    Solves the linear program to maximize sum of radii for a fixed set of centers.
    Returns (sum_radii, radii_array).
    """
    n_vars = N
    c = -np.ones(n_vars) # Minimize negative sum to maximize sum
    
    A_ub = []
    b_ub = []
    
    # Precompute distances to walls and between circles
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    # Wall constraints: r_i <= dist_to_wall(i)
    for i in range(N):
        x, y = centers[i]
        dist_wall = min(x, SQ_SIZE - x, y, SQ_SIZE - y)
        row = np.zeros(n_vars)
        row[i] = 1.0
        A_ub.append(row)
        b_ub.append(dist_wall)
        
    # Overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(N):
        for j in range(i + 1, N):
            d = dists[i, j]
            row = np.zeros(n_vars)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(d)
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
        if res.success:
            return -res.fun, res.x
        else:
            return 0.0, np.zeros(N)
    except Exception:
        return 0.0, np.zeros(N)

def objective(centers_flat):
    """
    Objective function for the center optimizer.
    Returns the negative of the optimal sum of radii.
    """
    centers = centers_flat.reshape((N, 2))
    
    # Penalty for centers outside the unit square
    if np.any(centers < 0) or np.any(centers > SQ_SIZE):
        return 1e6
        
    sum_r, _ = solve_lp(centers)
    return -sum_r

def generate_grid(n_rows, n_cols):
    """Generates a regular grid of points."""
    xs = np.linspace(0.1, 0.9, n_cols) if n_cols > 1 else np.array([0.5])
    ys = np.linspace(0.1, 0.9, n_rows) if n_rows > 1 else np.array([0.5])
    points = []
    for y in ys:
        for x in xs:
            points.append([x, y])
    return np.array(points[:N])

def generate_hex(n_rows, cols_per_row):
    """Generates a hexagonal lattice of points."""
    points = []
    row_height = np.sqrt(3) / 2
    for r in range(n_rows):
        y = (r + 0.5) * row_height
        offset = (row_height * 0.5) if r % 2 == 1 else 0
        for c in range(cols_per_row[r]):
            x = (c + 0.5) + offset
            if 0 <= x <= 1:
                points.append([x, y])
    return np.array(points[:N])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    best_sum = -1
    best_centers = None
    best_radii = None
    
    # Initializations: Grid and Hexagonal patterns
    inits = [
        generate_grid(5, 5),
        generate_grid(4, 6),
        generate_hex(7, [4, 4, 5, 5, 5, 5, 3])
    ]
    
    # Add random initialization
    np.random.seed(42)
    inits.append(np.random.rand(N, 2) * 0.8 + 0.1)

    for x0 in inits:
        x0_flat = x0.flatten()
        try:
            res = minimize(objective, x0_flat, method='Nelder-Mead', 
                          options={'maxiter': 2000, 'xatol': 1e-5, 'fatol': 1e-5})
            
            centers = res.x.reshape((N, 2))
            sum_r, radii = solve_lp(centers)
            
            if sum_r > best_sum:
                best_sum = sum_r
                best_centers = centers
                best_radii = radii
        except Exception:
            continue
            
    return best_centers, best_radii, best_sum
