import numpy as np
from scipy.optimize import linprog
import math

def solve_radii_with_duals(centers):
    """Solve LP for optimal radii given centers, returning radii and dual variables."""
    n = centers.shape[0]
    
    n_boundary = 4 * n
    n_pair = n * (n - 1) // 2
    n_constraints = n_boundary + n_pair
    
    A_ub = np.zeros((n_constraints, n))
    b_ub = np.zeros(n_constraints)
    
    idx = 0
    for i in range(n):
        A_ub[idx, i] = 1.0
        b_ub[idx] = centers[i, 0]
        idx += 1
        
        A_ub[idx, i] = 1.0
        b_ub[idx] = 1.0 - centers[i, 0]
        idx += 1
        
        A_ub[idx, i] = 1.0
        b_ub[idx] = centers[i, 1]
        idx += 1
        
        A_ub[idx, i] = 1.0
        b_ub[idx] = 1.0 - centers[i, 1]
        idx += 1
    
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = math.sqrt(dx * dx + dy * dy)
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dist
            idx += 1
    
    c_obj = -np.ones(n)
    bounds = [(0.0, None) for _ in range(n)]
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        radii = result.x
        duals = result.ineqlin.marginals
        return -result.fun, radii, duals
    else:
        return 0.0, np.zeros(n), np.zeros(n_constraints)


def compute_dual_gradient(centers, radii, duals):
    """Compute gradient of sum of radii w.r.t. centers using dual variables."""
    n = centers.shape[0]
    grad = np.zeros((n, 2))
    
    idx = 0
    for i in range(n):
        # r_i <= x_i: moving +x increases room
        grad[i, 0] += duals[idx]
        idx += 1
        # r_i <= 1 - x_i: moving -x increases room
        grad[i, 0] -= duals[idx]
        idx += 1
        # r_i <= y_i: moving +y increases room
        grad[i, 1] += duals[idx]
        idx += 1
        # r_i <= 1 - y_i: moving -y increases room
        grad[i, 1] -= duals[idx]
        idx += 1
    
    for i in range(n):
        for j in range(i + 1, n):
            dual = duals[idx]
            if dual > 1e-10:
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = math.sqrt(dx * dx + dy * dy)
                if dist > 1e-10:
                    grad[i, 0] += dual * dx / dist
                    grad[i, 1] += dual * dy / dist
                    grad[j, 0] -= dual * dx / dist
                    grad[j, 1] -= dual * dy / dist
            idx += 1
    
    return grad


def hexagonal_init(n):
    """Initialize centers in hexagonal pattern."""
    centers = []
    spacing = 0.18
    y = spacing * math.sqrt(3) / 2.0
    row = 0
    while len(centers) < n:
        x_off = 0.0 if row % 2 == 0 else spacing / 2.0
        x = 0.05 + x_off
        while x < 0.95 and len(centers) < n:
            centers.append([x, y])
            x += spacing
        y += spacing * math.sqrt(3) / 2.0
        row += 1
    
    result = np.array(centers[:n])
    result = np.clip(result, 0.01, 0.99)
    return result


def grid_init(n):
    """Initialize centers in grid pattern."""
    k = int(math.ceil(math.sqrt(n)))
    centers = []
    for i in range(k):
        for j in range(k):
            if len(centers) < n:
                x = (i + 0.5) / k
                y = (j + 0.5) / k
                centers.append([x, y])
    return np.array(centers[:n])


def random_init(n):
    """Initialize centers randomly with some spacing."""
    centers = np.random.rand(n, 2) * 0.8 + 0.1
    return centers


def optimize_from(centers, max_iter=3000, lr_init=0.02):
    """Run gradient ascent optimization from given centers."""
    n = centers.shape[0]
    best_sum = 0.0
    best_centers = centers.copy()
    best_radii = None
    
    for iteration in range(max_iter):
        s, radii, duals = solve_radii_with_duals(centers)
        
        if s > best_sum:
            best_sum = s
            best_centers = centers.copy()
            best_radii = radii.copy()
        
        grad = compute_dual_gradient(centers, radii, duals)
        
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad = grad / grad_norm
        
        lr = lr_init / (1.0 + iteration * 0.0003)
        
        centers = centers + lr * grad
        centers = np.clip(centers, 1e-8, 1.0 - 1e-8)
    
    s, radii, _ = solve_radii_with_duals(best_centers)
    return best_centers, radii, s

    for init_idx, init_fn in enumerate(init_methods):

            bc, br, bs = optimize_from(centers, max_iter=4000, lr_init=0.015)
            
            if bs > global_best_sum:
                global_best_sum = bs

                global_best_radii = br.copy()

def verify_packing(centers, radii):
    """Verify the packing is valid."""
    n = centers.shape[0]
    
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if r < 0:
            print(f"Circle {i} has negative radius")
            return False
        if x - r < -1e-10 or x + r > 1 + 1e-10:
            print(f"Circle {i} outside x bounds")
            return False
        if y - r < -1e-10 or y + r > 1 + 1e-10:
            print(f"Circle {i} outside y bounds")
            return False
    
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = math.sqrt(dx * dx + dy * dy)
            if dist < radii[i] + radii[j] - 1e-10:
                print(f"Circles {i} and {j} overlap")
                return False
    
    return True
