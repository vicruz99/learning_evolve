import numpy as np
import math
from scipy.optimize import minimize

def compute_penalty(vars):

    centers = vars[:52].reshape(26, 2)
    radii = vars[52:]

        viol = np.maximum(0, radii - wall)
        p += np.sum(viol**2)

    c2 = centers[np.newaxis, :, :]

    r2 = radii[np.newaxis, :]

    viol = np.maximum(0, reqs - dists)
    p += np.sum((viol[mask])**2)
    
    return p

def objective_function(vars):
    """Objective: Maximize sum of radii subject to constraints."""
    # Penalty weight balances constraint satisfaction vs objective
    return 1000.0 * compute_penalty(vars) - np.sum(vars[52:])
