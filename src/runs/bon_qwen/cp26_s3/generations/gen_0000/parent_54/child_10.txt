import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Packs 26 circles in a unit square [0,1]x[0,1] to maximize the sum of radii.
    Returns (centers, radii, sum_radii).
    """
    n = 26
    
    # 1. Initialization: Hexagonal packing layout
    # We create a dense initial configuration.
    # A hexagonal grid is a good starting point.
    # Let's try to fit roughly 26 circles. 
    # 5 rows of 5, 6, 5, 6, 4? Or just a dense grid.
    # We'll generate a grid and select 26 points.
    
    # Generate a hexagonal grid points
    points = []
    # Estimate radius for initialization to avoid overlaps
    # If we want to fit 26, maybe r ~ 0.08 initially is safe
    r_init = 0.08
    
    # Row spacing sqrt(3) * r_init
    row_h = np.sqrt(3) * r_init
    # Col spacing 2 * r_init
    col_w = 2 * r_init
    
    # We need to fit within [0, 1]
    # Let's just place them in a grid pattern that fits
    # 6 columns, 5 rows -> 30 points. We take 26.
    
    # To fit 6 columns of radius 0.08 (diameter 0.16), width needed ~ 0.16*6 = 0.96. Fits.
    # Height needed for 5 rows: 2*r + 4*sqrt(3)*r approx 2*0.08 + 4*0.138 = 0.16 + 0.55 = 0.71. Fits.
    
    y = r_init
    count = 0
    rows = 0
    while count < n and y + r_init <= 1:
        x = r_init
        if rows % 2 == 1:
            x += r_init # Shift odd rows
        while x + r_init <= 1 and count < n:
            points.append([x, y])
            count += 1
            x += col_w
        y += row_h
        rows += 1
        
    # If we didn't get enough points, fill with random or adjust
    # But the logic above should yield ~25-30 points.
    # If points < 26, we might need to tighten the grid, but for init, small r is fine.
    # Actually, with r=0.08, we fit many.
    
    # Ensure we have exactly 26 points
    centers_init = np.array(points[:n])
    radii_init = np.ones(n) * r_init
    
    # If we have fewer than 26 points due to bounds, pad with random valid points
    if len(centers_init) < n:
        # This shouldn't happen with r=0.08, but just in case
        pass

    # 2. Define Optimization Variables and Bounds
    # Variables: [x1, y1, r1, x2, y2, r2, ...]
    x0 = []
    for i in range(n):
        x0.extend([centers_init[i, 0], centers_init[i, 1], radii_init[i]])
    x0 = np.array(x0)
    
    bounds = []
    for i in range(n):
        bounds.extend([
            (0.0, 1.0), # x
            (0.0, 1.0), # y
            (0.0, 0.5)  # r (max radius 0.5)
        ])
    
    # 3. Define Objective Function
    def objective(x_vars, n):
        # Maximize sum of radii -> Minimize negative sum
        r_sum = 0.0
        for i in range(n):
            r_sum += x_vars[3 * i + 2]
        return -r_sum

    # 4. Define Constraint Functions
    # Inequality constraints must return >= 0
    
    def boundary_constraints(x_vars, n):
        vals = np.zeros(4 * n)
        idx = 0
        for i in range(n):
            xi = x_vars[3 * i]
            yi = x_vars[3 * i + 1]
            ri = x_vars[3 * i + 2]
            
            # x - r >= 0
            vals[idx] = xi - ri
            idx += 1
            # 1 - x - r >= 0
            vals[idx] = 1.0 - xi - ri
            idx += 1
            # y - r >= 0
            vals[idx] = yi - ri
            idx += 1
            # 1 - y - r >= 0
            vals[idx] = 1.0 - yi - ri
            idx += 1
        return vals

    def pairwise_constraints(x_vars, n):
        # dist^2 >= (r1 + r2)^2
        num_pairs = n * (n - 1) // 2
        vals = np.zeros(num_pairs)
        k = 0
        for i in range(n):
            xi = x_vars[3 * i]
            yi = x_vars[3 * i + 1]
            ri = x_vars[3 * i + 2]
            for j in range(i + 1, n):
                xj = x_vars[3 * j]
                yj = x_vars[3 * j + 1]
                rj = x_vars[3 * j + 2]
                
                dist_sq = (xi - xj)**2 + (yi - yj)**2
                sum_r = ri + rj
                vals[k] = dist_sq - sum_r**2
                k += 1
        return vals

    # 5. Setup Constraints for SLSQP
    cons = [
        {'type': 'ineq', 'fun': lambda x: boundary_constraints(x, n)},
        {'type': 'ineq', 'fun': lambda x: pairwise_constraints(x, n)}
    ]

    # 6. Run Optimization
    # SLSQP is a robust method for non-linear constraints
    try:
        res = minimize(
            objective,
            x0,
            args=(n,),
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 1000, 'ftol': 1e-9}
        )
        optimal_params = res.x
    except Exception as e:
        # Fallback to initial if optimization fails
        optimal_params = x0
        print(f"Optimization failed: {e}")

    # 7. Extract results
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        centers[i, 0] = optimal_params[3 * i]
        centers[i, 1] = optimal_params[3 * i + 1]
        radii[i] = optimal_params[3 * i + 2]

    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
