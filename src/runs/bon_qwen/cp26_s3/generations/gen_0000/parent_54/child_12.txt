import numpy as np
from scipy.optimize import minimize

def objective(vars, n):
    r = vars[2::3]
    return -np.sum(r)

def constraints(vars, n):
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    cons = []
    # Boundary constraints: r <= x <= 1-r  and  r <= y <= 1-r
    cons.extend(x - r)
    cons.extend(1.0 - x - r)
    cons.extend(y - r)
    cons.extend(1.0 - y - r)
    
    # Non-overlap constraints: dist^2 >= (r_i + r_j)^2
    for i in range(n):
        for j in range(i + 1, n):
            dx = x[i] - x[j]
            dy = y[i] - y[j]
            dr = r[i] + r[j]
            cons.append(dx*dx + dy*dy - dr*dr)
            
    return np.array(cons)

def constraint_fun(vars, n):
    return constraints(vars, n)

def run_packing():
    n = 26
    np.random.seed(42)
    
    # 1. Generate initial positions using a hexagonal lattice
    pts = []
    d = 0.2
    for row in range(8):
        for col in range(8):
            x = col * d + (row % 2) * d / 2
            y = row * d * np.sqrt(3) / 2
            if 0 <= x <= 1 and 0 <= y <= 1:
                pts.append([x, y])
                
    # Fallback to grid if not enough points (unlikely with d=0.2)
    if len(pts) < n:
        step = 1.0 / (int(np.sqrt(n)) + 1)
        for i in range(6):
            for j in range(6):
                if len(pts) < n:
                    pts.append([step * (i + 1), step * (j + 1)])
                    
    pts = pts[:n]
    centers_init = np.array(pts)
    
    # 2. Perturb and ensure feasibility
    centers_init += np.random.uniform(-0.01, 0.01, centers_init.shape)
    centers_init = np.clip(centers_init, 0.02, 0.98)
    
    # Compute feasible initial radii
    radii_init = np.array([min(c[0], 1.0-c[0], c[1], 1.0-c[1]) for c in centers_init])
    for i in range(n):
        for j in range(i+1, n):
            dist = np.sqrt(np.sum((centers_init[i] - centers_init[j])**2))
            r_lim = dist / 2.0
            radii_init[i] = min(radii_init[i], r_lim)
            radii_init[j] = min(radii_init[j], r_lim)
            
    # Flatten to 1D array: [x1, y1, r1, x2, y2, r2, ...]
    vars_init = np.zeros(3 * n)
    vars_init[0::3] = centers_init[:, 0]
    vars_init[1::3] = centers_init[:, 1]
    vars_init[2::3] = radii_init
    
    # Bounds: 0 <= x,y <= 1, 0 <= r <= 0.5
    bounds = [(0, 1), (0, 1), (0, 0.5)] * n
    cons = {'type': 'ineq', 'fun': constraint_fun, 'args': (n,)}
    
    # 3. Optimization
    res = minimize(objective, vars_init, args=(n,), method='SLSQP', 
                   bounds=bounds, constraints=cons, 
                   options={'maxiter': 3000, 'ftol': 1e-14, 'disp': False})
    
    final_vars = res.x
    centers = np.column_stack((final_vars[0::3], final_vars[1::3]))
    radii = final_vars[2::3]
    
    # Ensure non-negative radii and clip tiny numerical violations
    radii = np.maximum(radii, 0.0)
    
    sum_radii = float(np.sum(radii))
    return centers, radii, sum_radii
