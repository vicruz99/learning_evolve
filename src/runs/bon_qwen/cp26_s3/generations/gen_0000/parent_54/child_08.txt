import numpy as np
from scipy.optimize import minimize

N = 26


def encode(cen, rad):
    """Pack centers and radii into a flat vector for optimization."""
    x = np.empty(3 * N)
    x[::3] = cen[:, 0]
    x[1::3] = cen[:, 1]
    x[2::3] = rad
    return x


def decode(x):
    """Unpack optimization vector into centers and radii arrays."""
    cx = x[::3].copy()
    cy = x[1::3].copy()
    cr = x[2::3].copy()
    centers = np.column_stack([cx, cy])
    return centers, cr


def objective(x):
    """Negative sum of radii (minimizing this maximizes sum)."""
    return -np.sum(x[2::3])


def boundary_constraints(x):
    """Ensure all circles stay within [0,1]x[0,1].
    
    For each circle i: r_i <= x_i <= 1-r_i and r_i <= y_i <= 1-r_i
    Returns vector of constraint values (must be >= 0).
    """
    cx = x[::3]
    cy = x[1::3]
    cr = x[2::3]
    return np.concatenate([
        cx - cr,        # x >= r  (left boundary)
        1.0 - cx - cr,  # x + r <= 1  (right boundary)
        cy - cr,        # y >= r  (bottom boundary)
        1.0 - cy - cr   # y + r <= 1  (top boundary)
    ])


def overlap_constraints(x):
    """Ensure no two circles overlap.
    
    For each pair (i,j): distance(i,j)^2 >= (r_i + r_j)^2
    Using squared formulation avoids square root and is equivalent.
    Returns vector of constraint values (must be >= 0).
    """
    cx = x[::3]
    cy = x[1::3]
    cr = x[2::3]
    constraints = []
    for i in range(N):
        for j in range(i + 1, N):
            dx = cx[i] - cx[j]
            dy = cy[i] - cy[j]
            rsum = cr[i] + cr[j]
            constraints.append(dx * dx + dy * dy - rsum * rsum)
    return np.array(constraints)


def init_hexagonal(seed):
    """Generate initial hexagonal lattice configuration.
    
    Uses staggered rows with spacing based on hexagonal packing geometry.
    Row spacing = circle_diameter * sqrt(3)/2 for optimal hexagonal packing.
    """
    np.random.seed(seed)
    centers = np.zeros((N, 2))
    radii = np.full(N, 0.085)
    
    idx = 0
    spacing = 0.145
    
    for row in range(8):
        y = 0.08 + row * spacing * 0.8660254037844387  # sqrt(3)/2
        
        if y > 0.92:
            break
            
        for col in range(8):
            if row % 2 == 0:
                x = 0.08 + col * spacing
            else:
                x = 0.08 + spacing * 0.5 + col * spacing
            
            if x > 0.92 or idx >= N:
                break
            
            centers[idx] = [x, y]
            radii[idx] = 0.085
            idx += 1
            if idx >= N:
                break
    
    return centers, radii


def init_random(seed):
    """Generate random initial configuration with small radii."""
    np.random.seed(seed)
    centers = np.random.uniform(0.15, 0.85, (N, 2))
    radii = np.full(N, 0.05)
    return centers, radii


def run_packing():
    """Pack 26 circles in unit square to maximize sum of radii.
    
    Uses SLSQP optimization with multiple restarts from diverse initial
    configurations (hexagonal lattice and random).
    
    Returns:
        centers: np.array of shape (26, 2) with (x, y) coordinates
        radii: np.array of shape (26,) with radius of each circle
        sum_radii: float sum of all radii
    """
    best_sum = -1e10
    best_centers = None
    best_radii = None
    
    constraints = [
        {'type': 'ineq', 'fun': boundary_constraints},
        {'type': 'ineq', 'fun': overlap_constraints}
    ]
    
    num_trials = 25
    
    for trial in range(num_trials):
        if trial < 15:
            cen0, rad0 = init_hexagonal(seed=trial)
        else:
            cen0, rad0 = init_random(seed=trial * 37 + 13)
        
        x0 = encode(cen0, rad0)

            result = minimize(
                objective,
                x0,
                method='SLSQP',
                constraints=constraints,
                options={
                    'maxiter': 12000,
                    'ftol': 1e-15,
                    'disp': False
                }

            if current_sum > best_sum:
                best_sum = current_sum
                best_centers = centers_opt.copy()
                best_radii = radii_opt.copy()
                
        except Exception:
            continue
    
    # Fallback if optimization failed
    if best_centers is None:
        best_centers, best_radii = init_hexagonal(seed=0)
        best_sum = np.sum(best_radii)
    
    return best_centers, best_radii, best_sum
