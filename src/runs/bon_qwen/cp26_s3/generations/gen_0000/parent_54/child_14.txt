def get_constraints_func(n):
    """Returns a function that computes constraint values given a flattened variable array."""
    def constraint_func(vars):
        # vars is (78,) -> reshape to (26, 3)
        # Format: x, y, r
        data = vars.reshape((n, 3))
        x = data[:, 0]
        y = data[:, 1]
        r = data[:, 2]
        
        constraints = []
        
        # Boundary constraints
        # x >= r  => x - r >= 0
        constraints.extend(x - r)
        # x <= 1 - r => 1 - x - r >= 0
        constraints.extend(1 - x - r)
        # y >= r => y - r >= 0
        constraints.extend(y - r)
        # y <= 1 - r => 1 - y - r >= 0
        constraints.extend(1 - y - r)
        
        # Non-overlap constraints
        # dist^2 >= (r_i + r_j)^2
        # dist^2 - (r_i + r_j)^2 >= 0
        for i in range(n):
            for j in range(i + 1, n):
                dist_sq = (x[i] - x[j])**2 + (y[i] - y[j])**2
                rad_sum_sq = (r[i] + r[j])**2
                constraints.append(dist_sq - rad_sum_sq)
                
        return np.array(constraints)
    
    return constraint_func

def objective_func(vars):
    """Objective function to minimize (negative sum of radii)."""
    # vars is (78,) -> reshape to (26, 3)
    # We only need r values. They are at indices 2, 5, 8, ...
    # Or simply sum the 3rd column
    data = vars.reshape((26, 3))
    return -np.sum(data[:, 2])

def run_packing():
    n = 26
    best_sum_r = -1.0
    best_centers = None
    best_radii = None
    
    # Generate initial configurations
    configs = generate_initial_configs()
    
    # Define constraints for SLSQP
    # SLSQP expects a list of dicts: {'type': 'ineq', 'fun': ...}
    # But for efficiency with many constraints, we can pass a single function
    # that returns an array of constraint values.
    
    cons_func = get_constraints_func(n)
    
    # We need to wrap the constraint function for scipy
    # SLSQP 'fun' for inequality constraints should return >= 0
    def scipy_constraint(vars):
        return cons_func(vars)
    
    # Bounds for variables
    # x, y in [0, 1], r in [0, 0.5]
    # To be safe, r can go up to 0.5
    bounds = []
    for _ in range(n):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 0.5)) # r
        
    constraints = {'type': 'ineq', 'fun': scipy_constraint}
    
    for centers_init, radii_init in configs:
        # Flatten initial guess
        # Format: x, y, r for each circle
        init_vars = np.zeros(n * 3)

            init_vars[3*i+1] = centers_init[i, 1]
            init_vars[3*i+2] = radii_init[i]
        
        try:
            # Run optimization
            res = minimize(objective_func, init_vars, method='SLSQP', bounds=bounds, 
                           constraints=constraints, options={'ftol': 1e-9, 'maxiter': 1000, 'disp': False})
            
            if res.success:
                sum_r = -res.fun

                    best_sum_r = sum_r
                    best_res = res

            continue
