import numpy as np
from scipy.optimize import minimize

def obj(v):
    return -np.sum(v[2::3])

def con_bnd_x(v, i): return v[3*i] - v[3*i+2]
def con_bnd_1mx(v, i): return 1.0 - v[3*i] - v[3*i+2]
def con_bnd_y(v, i): return v[3*i+1] - v[3*i+2]
def con_bnd_1my(v, i): return 1.0 - v[3*i+1] - v[3*i+2]

def con_pair(v, i, j):
    dx = v[3*i] - v[3*j]
    dy = v[3*i+1] - v[3*j+1]
    dr = v[3*i+2] + v[3*j+2]
    return dx*dx + dy*dy - dr*dr

def jac_bnd_x(v, i):
    j = np.zeros(len(v))
    j[3*i] = 1.0; j[3*i+2] = -1.0
    return j

def jac_bnd_1mx(v, i):
    j = np.zeros(len(v))
    j[3*i] = -1.0; j[3*i+2] = -1.0
    return j

def jac_bnd_y(v, i):
    j = np.zeros(len(v))
    j[3*i+1] = 1.0; j[3*i+2] = -1.0
    return j

def jac_bnd_1my(v, i):
    j = np.zeros(len(v))
    j[3*i+1] = -1.0; j[3*i+2] = -1.0
    return j

def jac_pair(v, i, j):
    j = np.zeros(len(v))
    dx = v[3*i] - v[3*j]
    dy = v[3*i+1] - v[3*j+1]
    dr = v[3*i+2] + v[3*j+2]
    j[3*i] = 2*dx; j[3*i+1] = 2*dy; j[3*i+2] = -2*dr
    j[3*j] = -2*dx; j[3*j+1] = -2*dy; j[3*j+2] = -2*dr
    return j

def run_packing():
    n = 26
    best_sum = -1.0
    best_x = None
    
    inits = []
    # 1. Grid initialization
    g1 = []
    for i in range(5):
        for j in range(5):
            g1.append([0.1 + i*0.2, 0.1 + j*0.2])
    g1.append([0.2, 0.2])
    inits.append(g1)
    
    # 2. Hexagonal initialization
    g2 = []
    r0 = 0.09
    for row in range(6):
        y = r0 + row * np.sqrt(3)*r0
        x0 = r0 if row%2==0 else r0 + r0
        for col in range(5):
            x = x0 + col * 2*r0
            if x < 1-r0 and y < 1-r0:
                g2.append([x, y])
        if len(g2) >= n: break
    while len(g2) < n: g2.append([0.5, 0.5])
    inits.append(g2[:n])
    
    # 3. Random initializations
    for seed in [7, 99, 2024]:
        rng = np.random.default_rng(seed)
        inits.append(rng.uniform(0.05, 0.95, size=(n, 2)).tolist())

    bounds = [(0.0, 1.0), (0.0, 1.0), (1e-6, 1.0)] * n
    cons = []
    for i in range(n):
        cons.append({'type': 'ineq', 'fun': con_bnd_x, 'jac': jac_bnd_x, 'args': (i,)})
        cons.append({'type': 'ineq', 'fun': con_bnd_1mx, 'jac': jac_bnd_1mx, 'args': (i,)})
        cons.append({'type': 'ineq', 'fun': con_bnd_y, 'jac': jac_bnd_y, 'args': (i,)})
        cons.append({'type': 'ineq', 'fun': con_bnd_1my, 'jac': jac_bnd_1my, 'args': (i,)})
    for i in range(n):
        for j in range(i+1, n):
            cons.append({'type': 'ineq', 'fun': con_pair, 'jac': jac_pair, 'args': (i,j)})

    for guess in inits:
        x0 = np.zeros(3*n)
        for i in range(n):
            x0[3*i] = guess[i][0]
            x0[3*i+1] = guess[i][1]
            x0[3*i+2] = 0.09
            
        try:
            res = minimize(obj, x0, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 5000, 'ftol': 1e-12})
            curr = -res.fun
            if curr > best_sum:
                best_sum = curr
                best_x = res.x
        except:
            pass

    if best_x is None:
        best_x = np.zeros(3*n)
        for i in range(n):
            best_x[3*i] = 0.1 + (i%5)*0.2
            best_x[3*i+1] = 0.1 + (i//5)*0.2
            best_x[3*i+2] = 0.1
        best_sum = 2.5

    centers = best_x[:2*n].reshape(n, 2)
    radii = best_x[2::3]
    return centers, radii, best_sum
