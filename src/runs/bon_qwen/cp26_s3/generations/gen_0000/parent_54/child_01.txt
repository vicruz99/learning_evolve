import numpy as np
from scipy.optimize import linprog
import math

def solve_radii(centers):
    """
    Solves the LP to find optimal radii for a given set of centers.
    Maximize sum(r_i) subject to non-overlap and boundary constraints.
    """
    n = centers.shape[0]
    
    # Objective: minimize -sum(r) => c = -1
    c_obj = -np.ones(n)
    
    # Constraints: A_ub * r <= b_ub
    # Number of constraints: pairs + 4 boundaries per circle
    n_pairs = n * (n - 1) // 2
    n_bound = 4 * n
    n_constraints = n_pairs + n_bound
    
    # Initialize matrix and vector
    # Using dense matrix is fine for n=26 (approx 430 constraints)
    A_ub = np.zeros((n_constraints, n))
    b_ub = np.zeros(n_constraints)
    
    row = 0
    
    # 1. Pairwise constraints: r_i + r_j <= dist(i, j)
    # Compute distance matrix
    # centers is (n, 2)
    # Using broadcasting for speed
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dists = np.sqrt(dx*dx + dy*dy)
    
    # Fill upper triangle constraints
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[row, i] = 1.0
            A_ub[row, j] = 1.0
            b_ub[row] = dists[i, j]
            row += 1
            
    # 2. Boundary constraints
    # r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        x, y = centers[i]
        
        # r_i <= x
        A_ub[row, i] = 1.0
        b_ub[row] = x
        row += 1
        
        # r_i <= 1 - x
        A_ub[row, i] = 1.0
        b_ub[row] = 1.0 - x
        row += 1
        
        # r_i <= y
        A_ub[row, i] = 1.0
        b_ub[row] = y
        row += 1
        
        # r_i <= 1 - y
        A_ub[row, i] = 1.0
        b_ub[row] = 1.0 - y
        row += 1
        
    # Bounds: r_i >= 0
    bounds = [(0, None) for _ in range(n)]
    
    # Solve LP
    # 'highs' method is efficient
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return -res.fun, res.x
    else:
        # Fallback if LP fails (should not happen with valid inputs)
        return 0.0, np.zeros(n)
