import numpy as np
from scipy.optimize import linprog

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    np.random.seed(42)

    # 1. Geometric Layout Optimization
    # Initialize centers in the middle of the square with some randomness
    centers = np.random.uniform(0.2, 0.8, (n, 2))
    
    # Iterative growth and relaxation
    current_r = 0.01
    max_r = 0.15
    steps = 500
    
    for _ in range(steps):
        current_r += (max_r - current_r) * 0.01
        
        # Relaxation iterations per growth step
        for _ in range(50):
            forces = np.zeros_like(centers)
            
            # Inter-circle repulsion
            for i in range(n):
                for j in range(i + 1, n):
                    diff = centers[i] - centers[j]
                    dist = np.linalg.norm(diff)
                    min_dist = 2 * current_r
                    
                    if dist < min_dist and dist > 1e-9:
                        push = (min_dist - dist) / dist
                        forces[i] += diff * push * 0.5
                        forces[j] -= diff * push * 0.5
            # Boundary repulsion
            for i in range(n):
                if centers[i, 0] < current_r:
                    forces[i, 0] += (current_r - centers[i, 0]) * 10
                elif centers[i, 0] > 1 - current_r:
                    forces[i, 0] -= (centers[i, 0] - (1 - current_r)) * 10
                    
                if centers[i, 1] < current_r:
                    forces[i, 1] += (current_r - centers[i, 1]) * 10
                elif centers[i, 1] > 1 - current_r:
                    forces[i, 1] -= (centers[i, 1] - (1 - current_r)) * 10

            centers += forces * 0.5
            # Clamp
            centers = np.clip(centers, current_r, 1 - current_r)

    # 2. Size Optimization (Linear Programming)
    # Maximize sum(r_i)
    # Variables: r_0, ..., r_{25}
    c = np.ones(n)
    
    A_ub = []
    b_ub = []
    
    # Pairwise non-overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist_ij = np.linalg.norm(centers[i] - centers[j])
            row = np.zeros(n)
            row[i] = 1
            row[j] = 1
            A_ub.append(row)
            b_ub.append(dist_ij)
            
    # Boundary constraints: r_i <= min(x_i, 1-x_i, y_i, 1-y_i)
    for i in range(n):
        x, y = centers[i]
        max_r_i = min(x, 1 - x, y, 1 - y)
        row = np.zeros(n)
        row[i] = 1
        A_ub.append(row)
        b_ub.append(max_r_i)

    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    # Bounds for r_i: [0, inf]
    bounds = [(0, None) for _ in range(n)]

    # Solve LP
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    radii = res.x
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
