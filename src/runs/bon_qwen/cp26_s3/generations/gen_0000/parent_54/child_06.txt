import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def objective(v):
    """Objective function: maximize sum of radii -> minimize negative sum."""
    return -np.sum(v[2*N_CIRCLES:])

def constraint_fun(v):
    """Computes non-overlap constraints: dist^2 - (r_i + r_j)^2 >= 0"""
    n = N_CIRCLES
    centers = v[:2*n].reshape((n, 2))
    radii = v[2*n:]
    c_i = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    diff_sq = np.sum(c_i**2, axis=2)
    r_sum_sq = (radii[:, np.newaxis] + radii[np.newaxis, :])**2
    idx = np.tril_indices(n, -1)
    return diff_sq[idx] - r_sum_sq[idx]

def constraint_jac(v):
    """Exact Jacobian of the non-overlap constraints."""
    n = N_CIRCLES
    m = n*(n-1)//2
    jac = np.zeros((m, 3*n))
    centers = v[:2*n].reshape((n, 2))
    radii = v[2*n:]
    c_i = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    idx = np.tril_indices(n, -1)
    
    for k, (i, j) in enumerate(zip(idx[0], idx[1])):
        diff = c_i[i, j]
        sr = radii[i] + radii[j]
        jac[k, 2*i:2*i+2] = 2 * diff
        jac[k, 2*j:2*j+2] = -2 * diff
        jac[k, 2*n+i] = -2 * sr
        jac[k, 2*n+j] = -2 * sr
    return jac

def is_feasible(v):
    """Checks if the configuration satisfies all constraints within tolerance."""
    n = N_CIRCLES
    centers = v[:2*n].reshape((n, 2))
    radii = v[2*n:]
    if np.any(radii < -1e-9) or np.any(centers < -1e-9) or np.any(centers > 1.0 + 1e-9):
        return False
    c_diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist_sq = np.sum(c_diff**2, axis=2)
    r_sum_sq = (radii[:, np.newaxis] + radii[np.newaxis, :])**2
    idx = np.tril_indices(n, -1)
    return np.all(dist_sq[idx] >= r_sum_sq[idx] - 1e-9)

def run_packing():
    n = N_CIRCLES
    bounds_c = [(0.0, 1.0)] * (2 * n)
    bounds_r = [(1e-4, 0.5)] * n
    bounds = bounds_c + bounds_r
    
    cons = {'type': 'ineq', 'fun': constraint_fun, 'jac': constraint_jac}
    
    best_sum = 0.0
    best_v = None
    
    # Multi-start optimization to escape local minima
    for seed in range(15):
        rng = np.random.RandomState(seed)
        v0 = np.zeros(3 * n)
        
        # Initialize with a perturbed grid layout
        grid_x = np.linspace(0.12, 0.88, 6)
        grid_y = np.linspace(0.12, 0.88, 5)
        pts = []
        for y in grid_y:
            for x in grid_x:
                if len(pts) < n:
                    pts.append([x, y])
        
        pts = np.array(pts) + rng.uniform(-0.03, 0.03, size=(n, 2))
        pts = np.clip(pts, 0.05, 0.95)
        v0[:2*n] = pts.flatten()
        v0[2*n:] = 0.04 * np.ones(n)
        
        # Ensure initial feasibility by scaling radii if necessary
        scale = 1.0
        while scale > 0.1:
            v0[2*n:] *= scale
            if is_feasible(v0):
                break
            scale *= 0.8
            
        try:
            res = minimize(objective, v0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 400, 'ftol': 1e-10})
            cur_sum = -res.fun
            if cur_sum > best_sum:
                if is_feasible(res.x):
                    best_sum = cur_sum
                    best_v = res.x.copy()
        except Exception:
            pass
            
    # Fallback safe configuration if optimization fails entirely
    if best_v is None:
        centers = np.array([[0.2, 0.2], [0.8, 0.8]] + [[0.5, 0.5]]*24)
        radii = np.ones(26) * 0.03
        return centers, radii, float(26*0.03)
        
    centers = best_v[:2*n].reshape((n, 2))
    radii = best_v[2*n:]
    return centers, radii, float(best_sum)
