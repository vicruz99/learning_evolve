import numpy as np
import scipy.optimize as opt

def objective_vectorized(p, n, lam):
    """
    Objective function to minimize: -sum(radii) + penalty terms.
    p: flattened array of shape (2n + n,) = (x1, y1, r1, x2, y2, r2, ...)
    n: number of circles
    lam: penalty weight
    """
    centers = p[:2*n].reshape(n, 2)
    radii = p[2*n:]
    
    # We want to maximize sum of radii, so we minimize its negative
    val = -np.sum(radii)
    
    # Boundary penalties: ensure circles stay inside [0,1]x[0,1]
    # Penalize if x - r < 0  =>  r - x > 0
    val += lam * np.sum(np.maximum(0, radii - centers[:, 0])**2)
    # Penalize if x + r > 1  =>  x - (1 - r) > 0
    val += lam * np.sum(np.maximum(0, radii - (1 - centers[:, 0]))**2)
    val += lam * np.sum(np.maximum(0, radii - centers[:, 1])**2)
    val += lam * np.sum(np.maximum(0, radii - (1 - centers[:, 1]))**2)
    
    # Pairwise non-overlap penalties
    # Compute distance matrix efficiently using broadcasting
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    # Avoid self-comparison issues
    np.fill_diagonal(dists, 1.0)
    
    # Extract lower triangle to count each pair exactly once
    tri_dists = np.tril(dists, -1)
    rad_sum = radii[:, None] + radii[None, :]
    tri_rad_sum = np.tril(rad_sum, -1)
    
    # Penalize overlaps: distance < r_i + r_j
    overlaps = np.maximum(0, tri_rad_sum - tri_dists)
    val += lam * np.sum(overlaps**2)
    
    return val
