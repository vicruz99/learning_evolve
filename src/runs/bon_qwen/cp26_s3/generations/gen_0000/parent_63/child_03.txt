import numpy as np
from scipy.optimize import minimize

def objective(vars):
    radii = vars[2::3]
    return -np.sum(radii)

def constraint_func(vars):
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    # Boundary constraints: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    b = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    
    # Distance constraints: dist(i,j) - r_i - r_j >= 0
    n = len(r)
    x_col = x[:, np.newaxis]
    y_col = y[:, np.newaxis]
    dx = x_col - x_col.T
    dy = y_col - y_col.T
    dist_mat = np.sqrt(dx*dx + dy*dy)
    
    i, j = np.triu_indices(n, k=1)
    dists = dist_mat[i, j] - r[i] - r[j]
    
    return np.concatenate([b, dists])

def run_optimization(initial_centers, initial_radii):
    n = len(initial_radii)
    x0 = np.zeros(3*n)
    x0[0::3] = initial_centers[:, 0]
    x0[1::3] = initial_centers[:, 1]
    x0[2::3] = initial_radii
    
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)] * n + [(1e-7, 0.5)] * n
    cons = {'type': 'ineq', 'fun': constraint_func}
    
    res = minimize(objective, x0, method='SLSQP', bounds=bounds, 
                   constraints=cons, options={'maxiter': 5000, 'ftol': 1e-12})
    return res.x, -res.fun

def generate_hex_initial(n=26):
    centers = []
    r_init = 0.09
    s = 2.0 * r_init
    dy = s * np.sqrt(3.0) / 2.0
    row = 0
    while len(centers) < n:
        y = r_init + row * dy
        if y + r_init > 1.0:
            break
        if row % 2 == 0:
            xs = np.arange(r_init, 1.0 - r_init + 1e-9, s)
        else:
            xs = np.arange(r_init + s/2.0, 1.0 - r_init + 1e-9, s)
        for x in xs:
            if len(centers) < n:
                centers.append([x, y])
        row += 1
        
    centers = np.array(centers)
    if len(centers) < n:
        missing = n - len(centers)
        c_rand = np.random.rand(missing, 2) * 0.6 + 0.2
        centers = np.vstack([centers, c_rand])
        
    return centers, np.full(min(len(centers), n), r_init)

def run_packing():
    np.random.seed(42)
    best_vars = None
    best_sum = 0.0
    
    starts = []
    
    # 1. Hexagonal lattice start
    c1, r1 = generate_hex_initial(26)
    starts.append((c1, r1))
    
    # 2. Perturbed hexagonal
    c2 = c1.copy() + np.random.randn(26, 2) * 0.02
    r2 = r1.copy()
    starts.append((c2, r2))
    
    # 3. Grid start (5x5 + 1 center) with slight perturbation to break symmetry
    c_grid = np.array([[0.1 + 0.2*i, 0.1 + 0.2*j] for i in range(5) for j in range(5)])
    c_grid = np.vstack([c_grid, [0.5, 0.5]])
    c_grid += np.random.randn(26, 2) * 0.005
    r_grid = np.full(26, 0.095)
    starts.append((c_grid, r_grid))
    
    # Run optimization from each starting configuration
    for c, r in starts:
        try:
            res_vars, res_sum = run_optimization(c, r)
            if res_sum > best_sum:
                best_sum = res_sum
                best_vars = res_vars.copy()
        except Exception:
            continue
            
    # Final refinement pass on the best found configuration
    if best_vars is not None:
        c_final = np.column_stack((best_vars[0::3], best_vars[1::3]))
        r_final = best_vars[2::3]
        res_vars, res_sum = run_optimization(c_final, r_final)
        best_vars = res_vars
        best_sum = res_sum
        
    centers = np.column_stack((best_vars[0::3], best_vars[1::3]))
    radii = best_vars[2::3]
    
    # Numerical safety: if any constraint is slightly violated due to float precision,
    # uniformly shrink radii to guarantee validity.
    con_vals = constraint_func(best_vars)
    min_con = np.min(con_vals)
    if min_con < -1e-8:
        radii = radii + min_con  # min_con is negative, so this shrinks radii
        centers = centers
        best_sum = np.sum(radii)
        
    return centers, radii, best_sum
