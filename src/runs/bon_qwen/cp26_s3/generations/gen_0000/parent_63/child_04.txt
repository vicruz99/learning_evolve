import numpy as np
from scipy.optimize import minimize, linprog


def run_packing() -> tuple:
    n = 26

    best_centers = None
    best_radii = None
    best_sum = 0.0

    # Try many different initializations
    for trial in range(50):
        if trial < 15:
            centers = _init_hexagonal(n, trial)
        elif trial < 35:
            centers = _init_corner_heavy(n, trial)
        else:
            centers = _init_random_spread(n, trial)

        current_sum, centers, radii = _full_optimize(centers, n)

        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers.copy()
            best_radii = radii.copy()

    # Final refinement on best solution
    best_sum, best_centers, best_radii = _final_refine(best_centers, n)

    return best_centers, best_radii, best_sum


def _init_hexagonal(n, variant):
    centers = np.zeros((n, 2))
    idx = 0

    r_base = 0.075 + variant * 0.003
    offset_x = variant * 0.02
    offset_y = variant * 0.02

    for row in range(8):
        for col in range(8):
            if idx >= n:
                break
            x = r_base + offset_x + col * 2 * r_base + (r_base if row % 2 == 1 else 0)
            y = r_base + offset_y + row * np.sqrt(3) * r_base

            if x <= 1 - r_base and y <= 1 - r_base:
                centers[idx] = [x, y]
                idx += 1
        if idx >= n:
            break

    while idx < n:
        centers[idx] = [0.5 + 0.05 * np.sin(idx * 0.7), 0.5 + 0.05 * np.cos(idx * 0.7)]
        idx += 1

    return centers


def _init_corner_heavy(n, variant):
    centers = np.zeros((n, 2))
    np.random.seed(100 + variant)

    # Place larger circles in corners
    corner_r = 0.18 + variant * 0.01
    centers[0] = [corner_r, corner_r]
    centers[1] = [1 - corner_r, corner_r]
    centers[2] = [corner_r, 1 - corner_r]
    centers[3] = [1 - corner_r, 1 - corner_r]

    idx = 4
    # Fill remaining with hexagonal pattern in center
    r_mid = 0.06
    for row in range(6):
        for col in range(6):
            if idx >= n:
                break
            x = 0.3 + col * 2 * r_mid + (r_mid if row % 2 == 1 else 0)
            y = 0.3 + row * np.sqrt(3) * r_mid

            if x <= 0.7 and y <= 0.7:
                centers[idx] = [x, y]
                idx += 1
        if idx >= n:
            break

    while idx < n:
        centers[idx] = np.random.rand(2) * 0.6 + 0.2
        idx += 1

    return centers


def _init_random_spread(n, seed):
    np.random.seed(seed)
    centers = np.zeros((n, 2))

    # Use jittered grid
    grid_n = int(np.ceil(np.sqrt(n)))
    for i in range(n):
        row = i // grid_n
        col = i % grid_n
        cell_size = 1.0 / grid_n
        margin = cell_size * 0.2
        x = margin + col * cell_size + np.random.rand() * (cell_size - 2 * margin)
        y = margin + row * cell_size + np.random.rand() * (cell_size - 2 * margin)
        centers[i] = [x, y]

    return centers


def _solve_radii_lp(centers, n):
    """Solve LP: maximize sum(r_i) s.t. r_i + r_j <= dist(i,j) and boundary constraints"""
    c = -np.ones(n)

    A_ub_list = []
    b_ub_list = []

    # Pairwise distance constraints
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub_list.append(row)
            b_ub_list.append(dist)

    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        max_r = min(x, 1.0 - x, y, 1.0 - y)
        if max_r < 0:
            max_r = 0.0
        row = np.zeros(n)
        row[i] = 1.0
        A_ub_list.append(row)
        b_ub_list.append(max_r)

    A_ub = np.array(A_ub_list)
    b_ub = np.array(b_ub_list)
    bounds = [(0.0, None) for _ in range(n)]

    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            return np.maximum(result.x, 0.0)
    except Exception:
        pass

    # Fallback: boundary-limited radii
    radii = np.zeros(n)
    for i in range(n):
        x, y = centers[i]
        radii[i] = max(0.0, min(x, 1.0 - x, y, 1.0 - y))

    return radii


def _compute_slack(centers, radii, n):
    """Compute minimum slack (clearance)"""
    slack = 1e10

    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            required = radii[i] + radii[j]
            slack = min(slack, dist - required)

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        slack = min(slack, x - r, 1.0 - x - r, y - r, 1.0 - y - r)

    return slack


def _optimize_positions(centers, radii, n, max_iter=50000):
    """Optimize positions to maximize minimum slack"""
    best_centers = centers.copy()
    best_slack = _compute_slack(centers, radii, n)

    def objective(vars):
        c = vars.reshape(n, 2)
        slack = _compute_slack(c, radii, n)
        return -slack

    try:
        result = minimize(
            objective,
            best_centers.flatten(),
            method='Nelder-Mead',
            options={
                'maxiter': max_iter,
                'xatol': 1e-12,
                'fatol': 1e-12,
                'adaptive': True
            }
        )
        new_centers = result.x.reshape(n, 2)
        new_slack = _compute_slack(new_centers, radii, n)
        if new_slack > best_slack:
            best_centers = new_centers
            best_slack = new_slack
    except Exception:
        pass

    return best_centers


def _perturb_centers(centers, n, strength=0.02):
    """Add random perturbation to escape local optima"""
    new_centers = centers.copy()
    perturbation = np.random.randn(n, 2) * strength
    new_centers += perturbation
    new_centers = np.clip(new_centers, 0.01, 0.99)
    return new_centers


def _full_optimize(centers, n):
    """Full optimization loop"""
    np.random.seed(None)
    best_centers = centers.copy()
    best_radii = _solve_radii_lp(centers, n)
    best_sum = np.sum(best_radii)

    for iteration in range(40):
        current_sum = np.sum(best_radii)

        # Perturbation with annealing
        strength = 0.03 * (1.0 - iteration / 40.0)
        new_centers = _perturb_centers(best_centers, n, strength)

        # Solve for radii
        new_radii = _solve_radii_lp(new_centers, n)
        new_sum = np.sum(new_radii)

        if new_sum > current_sum:
            best_centers = new_centers
            best_radii = new_radii
            best_sum = new_sum

        # Position optimization with fixed radii
        if np.min(best_radii) > 1e-8:
            opt_centers = _optimize_positions(best_centers, best_radii, n, max_iter=20000)
            opt_radii = _solve_radii_lp(opt_centers, n)
            opt_sum = np.sum(opt_radii)

            if opt_sum > best_sum:
                best_centers = opt_centers
                best_radii = opt_radii
                best_sum = opt_sum

    return best_sum, best_centers, best_radii


def _final_refine(centers, n):
    """Final careful refinement"""
    radii = _solve_radii_lp(centers, n)
    current_sum = np.sum(radii)

    for _ in range(20):
        # Very fine position optimization
        centers = _optimize_positions(centers, radii, n, max_iter=100000)
        radii = _solve_radii_lp(centers, n)
        new_sum = np.sum(radii)

        if new_sum <= current_sum + 1e-10:
            break
        current_sum = new_sum

    return current_sum, centers, radii
