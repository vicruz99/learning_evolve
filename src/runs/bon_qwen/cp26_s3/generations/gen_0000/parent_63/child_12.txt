import numpy as np
from scipy.optimize import minimize

N = 26


def compute_violation(p):
    centers = p[:2 * N].reshape(N, 2)
    radii = p[2 * N:]

    violation = 0.0

    for i in range(N):
        x, y = centers[i]
        r = radii[i]
        violation += max(0, r - x) ** 2
        violation += max(0, r - (1 - x)) ** 2
        violation += max(0, r - y) ** 2
        violation += max(0, r - (1 - y)) ** 2

    for i in range(N):
        for j in range(i + 1, N):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            overlap = radii[i] + radii[j] - dist
            if overlap > 0:
                violation += overlap ** 2

    return violation


def objective_penalty(p):
    radii = p[2 * N:]
    v = compute_violation(p)
    return -np.sum(radii) + 10000 * v


def optimize_radii_given_positions(centers):
    n = centers.shape[0]
    radii = np.zeros(n)

    for i in range(n):
        x, y = centers[i]
        radii[i] = min(x, 1 - x, y, 1 - y)

    for iteration in range(200):
        old_radii = radii.copy()

        for i in range(n):
            max_r = min(centers[i, 0], 1 - centers[i, 0],
                        centers[i, 1], 1 - centers[i, 1])

            for j in range(n):
                if i != j:
                    dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                    max_r = min(max_r, dist - old_radii[j])

            radii[i] = max(1e-10, max_r)

        if np.max(np.abs(radii - old_radii)) < 1e-12:
            break

    return radii


def optimize_positions_given_radii(centers, radii, lr=0.001, n_iter=300):
    n = centers.shape[0]

    for iteration in range(n_iter):
        forces = np.zeros_like(centers)

        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist_sq = dx * dx + dy * dy
                dist = np.sqrt(dist_sq)

                min_dist = radii[i] + radii[j]
                if dist < min_dist and dist > 1e-10:
                    overlap = min_dist - dist
                    force = overlap / dist
                    fx = dx * force
                    fy = dy * force
                    forces[i, 0] += fx
                    forces[i, 1] += fy
                    forces[j, 0] -= fx
                    forces[j, 1] -= fy

        for i in range(n):
            x, y = centers[i]
            r = radii[i]

            if x - r < 0:
                forces[i, 0] += abs(x - r) * 500
            if x + r > 1:
                forces[i, 0] -= abs(x + r - 1) * 500
            if y - r < 0:
                forces[i, 1] += abs(y - r) * 500
            if y + r > 1:
                forces[i, 1] -= abs(y + r - 1) * 500

        centers = centers + lr * forces

        for i in range(n):
            centers[i, 0] = np.clip(centers[i, 0], radii[i] + 1e-10, 1 - radii[i] - 1e-10)
            centers[i, 1] = np.clip(centers[i, 1], radii[i] + 1e-10, 1 - radii[i] - 1e-10)

    return centers


def create_initial_hexagonal(row_counts, spacing_x, spacing_y, offset_factor=0.5):
    centers = np.zeros((N, 2))
    radii = np.ones(N) * 0.05

    idx = 0
    for row_idx, count in enumerate(row_counts):
        y = spacing_y * (row_idx + 1) / (len(row_counts) + 0.5)
        for col_idx in range(count):
            if idx >= N:
                break
            x_spacing = spacing_x / (count + 1)
            x = x_spacing * (col_idx + 1)
            if row_idx % 2 == 1:
                x += x_spacing * offset_factor
            centers[idx] = [x, y]
            idx += 1

    return centers, radii


def run_packing():
    best_centers = None
    best_radii = None
    best_sum = -np.inf

    # Multiple configuration templates
    configs = [
        ([7, 6, 7, 6], 0.95, 0.85),
        ([6, 7, 6, 7], 0.95, 0.85),
        ([5, 6, 6, 6, 3], 0.90, 0.90),
        ([4, 6, 6, 6, 4], 0.85, 0.92),
        ([6, 6, 6, 6, 2], 0.88, 0.88),
    ]

    for config_idx, (row_counts, sx, sy) in enumerate(configs):
        for seed in range(3):
            np.random.seed(config_idx * 100 + seed)

            centers, radii = create_initial_hexagonal(row_counts, sx, sy)

            # Add perturbation
            centers = centers + np.random.randn(*centers.shape) * 0.008
            radii = radii + np.random.randn(N) * 0.003
            radii = np.maximum(radii, 0.01)

            # Alternating optimization - coarse
            for phase in range(3):
                lr = 0.003 * (0.5 ** phase)
                n_outer = 80 // (10 ** phase)
                n_inner = 400 // (10 ** phase)

                for outer_iter in range(n_outer):
                    radii = optimize_radii_given_positions(centers)
                    centers = optimize_positions_given_radii(centers, radii, lr=lr, n_iter=n_inner)

                    # Recalculate radii after position adjustment
                    radii = optimize_radii_given_positions(centers)

            current_sum = np.sum(radii)
            if current_sum > best_sum:
                best_sum = current_sum
                best_centers = centers.copy()
                best_radii = radii.copy()

    # Also try penalty-based optimization from best alternating result
    if best_centers is not None:
        x0 = np.concatenate([best_centers.flatten(), best_radii])

        result = minimize(objective_penalty, x0, method='Nelder-Mead',
                          options={'maxiter': 300000, 'xatol': 1e-12, 'fatol': 1e-12, 'adaptive': True})

        centers_pen = result.x[:2 * N].reshape(N, 2)
        radii_pen = result.x[2 * N:]
        radii_pen = np.maximum(radii_pen, 1e-8)

        # Clip to valid range
        for i in range(N):
            centers_pen[i, 0] = np.clip(centers_pen[i, 0], radii_pen[i] + 1e-10, 1 - radii_pen[i] - 1e-10)
            centers_pen[i, 1] = np.clip(centers_pen[i, 1], radii_pen[i] + 1e-10, 1 - radii_pen[i] - 1e-10)

        pen_violation = compute_violation(result.x)
        if pen_violation < 1e-6:
            pen_sum = np.sum(radii_pen)
            if pen_sum > best_sum:
                best_sum = pen_sum
                best_centers = centers_pen.copy()
                best_radii = radii_pen.copy()

    # Final fine-tuning with alternating method
    for phase in range(3):
        lr = 0.0005 * (0.3 ** phase)
        for outer_iter in range(50):
            best_radii = optimize_radii_given_positions(best_centers)
            best_centers = optimize_positions_given_radii(best_centers, best_radii, lr=lr, n_iter=200)
            best_radii = optimize_radii_given_positions(best_centers)

    # Final cleanup
    best_radii = np.maximum(best_radii, 1e-8)
    for i in range(N):
        best_centers[i, 0] = np.clip(best_centers[i, 0], best_radii[i] + 1e-10, 1 - best_radii[i] - 1e-10)
        best_centers[i, 1] = np.clip(best_centers[i, 1], best_radii[i] + 1e-10, 1 - best_radii[i] - 1e-10)

    final_sum = np.sum(best_radii)

    return best_centers, best_radii, final_sum
