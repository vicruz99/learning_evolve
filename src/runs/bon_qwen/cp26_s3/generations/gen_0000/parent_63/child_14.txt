def run_packing():

                    centers[idx] = [x, y]
                    idx += 1
        else:

            centers = np.random.rand(n_circles, 2)

                if x - r < 0:
                    forces[i, 0] += k_wall * (x - r)

                    forces[i, 0] -= k_wall * (x + r - 1)

                    forces[i, 1] += k_wall * (y - r)

                    forces[i, 1] -= k_wall * (y + r - 1)
