from scipy.optimize import minimize
import math

def generate_initial(n, seed):

def hexagonal_grid(n):
    """Create a hexagonal grid arrangement."""
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.04
    
    row_counts = [5, 6, 5, 6, 4]
    idx = 0
    r0 = 0.04
    
    for row_i, count in enumerate(row_counts):
        y = r0 + row_i * math.sqrt(3) * r0
        shift = r0 if row_i % 2 == 1 else 0
        spacing = 2 * r0
        total_width = (count - 1) * spacing + 2 * r0
        x_start = (1 - total_width) / 2 + shift
        
        for col_i in range(count):
            x = x_start + col_i * spacing
            centers[idx] = [x, y]
            idx += 1
    
    centers += np.random.randn(n, 2) * 0.005
    return centers, radii


def perturbed_grid(n):

def random_valid(n):
    """Generate random valid positions."""
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.025
    
    for i in range(n):
        centers[i, 0] = np.random.uniform(radii[i], 1 - radii[i])
        centers[i, 1] = np.random.uniform(radii[i], 1 - radii[i])
    
    return centers, radii

def dense_hexagonal(n):

def get_bounds(n):
    """Get bounds for optimization variables."""
    bounds = []
    # Center coordinates: [0, 1]
    for _ in range(2 * n):
        bounds.append((0.0, 1.0))
    # Radii: [0, 0.5]
    for _ in range(n):
        bounds.append((0.0, 0.5))
    return bounds


def objective_function(params, n, lam):
    """
    Objective: maximize sum of radii with soft constraints.
    
    Minimizes: -sum(radii) + penalty(violations)
    """
    centers = params[:2 * n].reshape(n, 2)
    radii = params[2 * n:]
    
    # Primary objective: maximize sum of radii
    loss = -np.sum(radii)
    
    # Boundary constraint penalties
    for i in range(n):
        x, y = centers[i]
        r = max(0.0, radii[i])
        margins = [x - r, 1.0 - x - r, y - r, 1.0 - y - r]
        for m in margins:
            if m < 0:
                loss += lam * m * m
    
    # Non-overlap constraint penalties
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = math.hypot(dx, dy)
            overlap = radii[i] + radii[j] - dist
            if overlap > 0:
                loss += lam * overlap * overlap
    
    return loss


def cleanup_solution(centers, radii, n):
    """
    Clean up solution after optimization.
    Ensures all constraints are satisfied.
    """
    # Ensure non-negative radii
    radii = np.maximum(radii, 1e-10)
    
    # Clip centers to valid boundary range
    for i in range(n):
        r = radii[i]
        centers[i, 0] = np.clip(centers[i, 0], r, 1.0 - r)
        centers[i, 1] = np.clip(centers[i, 1], r, 1.0 - r)
    
    # Iteratively resolve overlaps
    for _ in range(500):
        overlap_found = False
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = math.hypot(dx, dy)
                required = radii[i] + radii[j]
                
                if dist < required - 1e-12:
                    overlap_found = True
                    excess = required - dist + 1e-10
                    
                    # Shrink circles proportionally to their radii
                    total_r = radii[i] + radii[j]
                    if total_r > 0:
                        shrink_i = excess * radii[i] / total_r
                        shrink_j = excess * radii[j] / total_r
                    else:
                        shrink_i = shrink_j = excess / 2.0
                    
                    radii[i] = max(1e-10, radii[i] - shrink_i)
                    radii[j] = max(1e-10, radii[j] - shrink_j)
        
        if not overlap_found:
            break
    
    # Final boundary check - shrink if needed
    for i in range(n):
        x, y = centers[i]
        max_r = min(x, 1.0 - x, y, 1.0 - y)
        radii[i] = min(radii[i], max(1e-10, max_r))

    for _ in range(100):
        overlap_found = False

                    radii[j] = max(1e-10, radii[j] - shrink_j)
        if not overlap_found:
            break
    
    return centers, radii
