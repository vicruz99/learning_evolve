import numpy as np
from scipy.optimize import minimize

def evaluate_objective(params, n, weight):
    """
    Objective function for scipy optimization.
    Minimizes negative sum of radii plus penalties for constraint violations.
    """
    centers = params[:n*2].reshape((n, 2))
    radii = params[n*2:]
    
    # Primary objective: maximize sum of radii (minimize negative)
    obj = -np.sum(radii)
    
    # Quadratic penalty for boundary violations
    # Left and Right walls
    pen = np.sum(np.maximum(0, radii - centers[:, 0])**2)
    pen += np.sum(np.maximum(0, centers[:, 0] + radii - 1)**2)
    # Bottom and Top walls
    pen += np.sum(np.maximum(0, radii - centers[:, 1])**2)
    pen += np.sum(np.maximum(0, centers[:, 1] + radii - 1)**2)
    
    # Quadratic penalty for circle-circle overlaps
    i_idx, j_idx = np.triu_indices(n, k=1)
    diff = centers[i_idx] - centers[j_idx]
    dists = np.sqrt(np.sum(diff**2, axis=1))
    rad_sum = radii[i_idx] + radii[j_idx]
    overlaps = rad_sum - dists
    pen += np.sum(np.maximum(0, overlaps)**2)
    
    return obj + weight * pen

    n = 26

            idx += 1

    centers = np.clip(centers, 0.05, 0.95)
    
    radii = np.ones(n) * 0.06
    
    # Simulation parameters
    dt = 0.005
    damping = 0.85
    growth = 1.0008
    max_steps = 50000
    
    velocities = np.zeros_like(centers)
    best_sum = 0.0
    best_centers = centers.copy()
    best_radii = radii.copy()
    
    i_idx, j_idx = np.triu_indices(n, k=1)

        diff = centers[i_idx] - centers[j_idx]
        dists = np.sqrt(np.sum(diff**2, axis=1))
        dists[dists < 1e-8] = 1e-8
        rad_sum = radii[i_idx] + radii[j_idx]
        overlaps = rad_sum - dists
        
        forces = np.zeros_like(centers)
        # Apply repulsive forces for overlapping circles
        for k in range(len(i_idx)):
            if overlaps[k] > 0:
                f_mag = overlaps[k]
                f_vec = diff[k] * (f_mag / dists[k])
                forces[i_idx[k]] += f_vec
                forces[j_idx[k]] -= f_vec
                
        # Apply boundary repulsion forces
        forces[:, 0] += np.maximum(0, radii - centers[:, 0])
        forces[:, 0] -= np.maximum(0, centers[:, 0] + radii - 1)
        forces[:, 1] += np.maximum(0, radii - centers[:, 1])
        forces[:, 1] -= np.maximum(0, centers[:, 1] + radii - 1)
        
        # Update physics
        velocities = velocities * damping + forces * dt
        centers += velocities * dt
        centers = np.clip(centers, 0, 1)

        if step % 50 == 0:
            radii *= growth
            
        # Record best valid configuration found during simulation
        if step % 1000 == 0:
            valid = True

                if rad_sum[k] - dists[k] > 1e-5:
                    valid = False
                    break
            if valid:
                for i in range(n):
