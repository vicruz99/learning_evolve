import numpy as np
from scipy.optimize import minimize

    return best_centers, best_radii, best_sum


def _fallback_packing(n):
    """Fallback: place circles in a grid pattern."""
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    r = 0.09
    idx = 0
    for i in range(6):
        for j in range(5):
            if idx >= n:
                break
            centers[idx] = [r + i * 2 * r, r + j * 2 * r]
            radii[idx] = r
            idx += 1
            if idx >= n:
                break
        if idx >= n:
            break
    return centers, radii, np.sum(radii)


def _penalized_objective(x, n, penalty_weight):
    """Compute penalized objective: minimize negative sum of radii plus penalty for violations."""
    centers = x[:2 * n].reshape(n, 2)
    radii = x[2 * n:]

    obj = -np.sum(radii)
    penalty = 0.0

    # Boundary penalties
    for i in range(n):
        cx, cy = centers[i]
        r = radii[i]

        if cx - r < 0:
            penalty += penalty_weight * (cx - r) ** 2
        if cx + r > 1:
            penalty += penalty_weight * (cx + r - 1) ** 2
        if cy - r < 0:
            penalty += penalty_weight * (cy - r) ** 2
        if cy + r > 1:
            penalty += penalty_weight * (cy + r - 1) ** 2

    # Overlap penalties (use squared distance to avoid sqrt)
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist_sq = dx * dx + dy * dy
            min_dist = radii[i] + radii[j]
            min_dist_sq = min_dist * min_dist
            if dist_sq < min_dist_sq:
                diff = min_dist_sq - dist_sq
                penalty += penalty_weight * diff * diff / (min_dist_sq + 1e-10)

    return obj + penalty


def _check_validity(centers, radii, n):
    """Check if packing is valid."""
    for i in range(n):
        cx, cy = centers[i]
        r = radii[i]
        if r < 0:
            return False
        if cx - r < -1e-10 or cx + r > 1 + 1e-10:
            return False
        if cy - r < -1e-10 or cy + r > 1 + 1e-10:
            return False

    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            if dist < radii[i] + radii[j] - 1e-10:
                return False

    return True


def _init_hexagonal(n, row_configs, r_init, perturbation=0.0):
    """Initialize circles in hexagonal pattern."""
    centers = []
    y = r_init
    for row_idx, nc in enumerate(row_configs):
        x_offset = r_init if row_idx % 2 == 1 else 0
        total_width = 2 * r_init + (nc - 1) * 2 * r_init
        x_start = (1 - total_width) / 2 + r_init + x_offset

        for i in range(nc):
            x = x_start + i * 2 * r_init
            if perturbation > 0:
                x += np.random.uniform(-perturbation, perturbation)
                y_curr = y + np.random.uniform(-perturbation, perturbation)
            else:
                y_curr = y
            centers.append([x, y_curr])
        y += r_init * np.sqrt(3)

    return np.array(centers[:n]), np.ones(n) * r_init


def _try_packing(n, seed):
    """Try to pack circles with given seed."""
    np.random.seed(seed)

    row_configs_list = [
        [5, 6, 5, 6, 4],
        [6, 5, 6, 5, 4],
        [5, 5, 6, 5, 5],
        [5, 6, 5, 5, 5],
        [6, 5, 5, 5, 5],
        [5, 5, 5, 6, 5],
        [4, 6, 6, 6, 4],
        [5, 5, 5, 5, 5, 1],
    ]

    best_result = None
    best_sum = 0

    for row_configs in row_configs_list:
        total = sum(row_configs)
        if total < n:
            continue

        for r_init in [0.09, 0.095, 0.098, 0.10, 0.101]:
            for pert in [0.0, 0.005, 0.01, 0.02]:
                try:
                    centers, radii = _init_hexagonal(n, row_configs, r_init, pert)
                    x0 = np.concatenate([centers.flatten(), radii])

                    bounds = []
                    for _ in range(2 * n):
                        bounds.append((0, 1))
                    for _ in range(n):
                        bounds.append((0, 0.5))

                    # Phase 1: optimize with moderate penalty
                    result1 = minimize(
                        _penalized_objective,
                        x0,
                        args=(n, 1000),
                        method='L-BFGS-B',
                        bounds=bounds,
                        options={'maxiter': 50000, 'ftol': 1e-15}
                    )

                    # Phase 2: optimize with higher penalty
                    result2 = minimize(
                        _penalized_objective,
                        result1.x,
                        args=(n, 10000),
                        method='L-BFGS-B',
                        bounds=bounds,
                        options={'maxiter': 50000, 'ftol': 1e-15}
                    )

                    centers = result2.x[:2 * n].reshape(n, 2)

                    # Ensure non-negative radii
                    radii = np.maximum(radii, 0)

                        s = np.sum(radii)
                        if s > best_sum:
                            best_sum = s
                            best_result = (centers.copy(), radii.copy(), s)
                except Exception:
                    continue

        # Also try with equal radii optimization (position only)
        for r_target in [0.099, 0.101, 0.103, 0.105]:
            try:
                centers, _ = _init_hexagonal(n, row_configs, r_target, 0.01)
                x0 = centers.flatten()

                bounds_pos = [(0, 1)] * (2 * n)

                def pos_objective(x):
                    c = x.reshape(n, 2)
                    p = 0.0
                    for i in range(n):
                        cx, cy = c[i]
                        if cx - r_target < 0:

                            p += 10000 * (cy - r_target) ** 2

                            p += 10000 * (cy + r_target - 1) ** 2

                            dy = c[i, 1] - c[j, 1]
                            d2 = dx * dx + dy * dy
                            md2 = (2 * r_target) ** 2

                                p += 10000 * (md2 - d2) ** 2 / (md2 + 1e-10)

                result = minimize(pos_objective, x0, method='L-BFGS-B', bounds=bounds_pos,
                                 options={'maxiter': 100000, 'ftol': 1e-15})
                centers = result.x.reshape(n, 2)

                if _check_validity(centers, radii, n):
                    s = np.sum(radii)
                    if s > best_sum:
                        best_sum = s
                        best_result = (centers.copy(), radii.copy(), s)
            except Exception:
                continue

    return best_result


def _run_coordinate_descent(centers, radii, n, iterations=200):
    """Refine packing using coordinate descent."""
    best_centers = centers.copy()
    best_radii = radii.copy()

    for it in range(iterations):
        improved = False
        for i in range(n):
            best_c = best_centers[i].copy()
            best_r = best_radii[i]

            for _ in range(50):
                dx = np.random.uniform(-0.01, 0.01)
                dy = np.random.uniform(-0.01, 0.01)
                dr = np.random.uniform(-0.003, 0.003)

                nc = best_centers[i] + np.array([dx, dy])
                nr = max(0.001, best_radii[i] + dr)

                valid = True
                cx, cy = nc
                if cx - nr < 0 or cx + nr > 1 or cy - nr < 0 or cy + nr > 1:
                    valid = False

                if valid:
                    for j in range(n):
                        if i == j:
                            continue
                        dx2 = nc[0] - best_centers[j, 0]
                        dy2 = nc[1] - best_centers[j, 1]
                        d = np.sqrt(dx2 * dx2 + dy2 * dy2)
                        if d < nr + best_radii[j] - 1e-12:
                            valid = False
                            break

                if valid and nr > best_r:
                    best_c = nc
                    best_r = nr
                    improved = True

            best_centers[i] = best_c
            best_radii[i] = best_r

        if not improved:
            break

    return best_centers, best_radii

5. **Coordinate descent refinement** to fine-tune individual circle positions
