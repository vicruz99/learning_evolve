import numpy as np
from scipy.optimize import minimize
import math

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Pack 26 circles in a unit square [0,1]x[0,1] to maximize the sum of radii.
    """
    N = 26
    
    # --- Initialization: Hexagonal Lattice ---
    # We use a hexagonal pattern to start with a dense, non-overlapping configuration.
    # Initial radius estimate to allow some growth during optimization
    r_init = 0.08
    centers = []
    y = r_init
    while len(centers) < N:
        x = r_init
        while x <= 1.0 - r_init:
            centers.append([x, y])
            x += math.sqrt(3) * r_init
        # Shift x for next row to create hexagonal offset
        y += 1.5 * r_init
        if len(centers) < N:
            centers.append([1.0 - r_init, y]) # Ensure we fill if needed
            
    # Take only the first N points if we generated more
    centers = np.array(centers[:N])
    
    # Mirror points that fall outside the unit square into the square
    for i in range(N):
        x, y = centers[i]
        if x < 0: centers[i, 0] = -x
        elif x > 1: centers[i, 0] = 2 - x
        if y < 0: centers[i, 1] = -y
        elif y > 1: centers[i, 1] = 2 - y
        
    radii = np.full(N, r_init)
    
    # --- Optimization Objective and Constraints ---
    # We minimize a penalty function: -Sum(r) + Penalty(Boundary) + Penalty(Overlap)
    
    def objective(vars):
        """
        vars contains [x0, y0, r0, x1, y1, r1, ..., x25, y25, r25]
        """
        n = len(vars) // 3
        penalty_weight = 1000.0
        
        loss = 0.0
        radii_sum = 0.0
        
        # Extract and apply bounds/clamping for numerical stability
        # (Though L-BFGS-B handles bounds, this helps inside the function logic)
        current_centers = vars[:2*n].reshape(n, 2)
        current_radii = vars[2*n:]
        
        # Boundary Check
        for i in range(n):
            x, y = current_centers[i]
            r = current_radii[i]
            radii_sum += r
            
            # Calculate violation distance
            dist_left = x - r
            dist_right = r - (x - 1) # x + r - 1
            dist_down = y - r
            dist_up = r - (y - 1)   # y + r - 1
            
            # Penalty if violation > 0
            if dist_left < 0: loss += penalty_weight * dist_left**2
            if dist_right < 0: loss += penalty_weight * dist_right**2
            if dist_down < 0: loss += penalty_weight * dist_down**2
            if dist_up < 0: loss += penalty_weight * dist_up**2

        # Overlap Check
        for i in range(n):
            for j in range(i + 1, n):
                dist_sq = np.sum((current_centers[i] - current_centers[j])**2)
                r_sum = current_radii[i] + current_radii[j]
                
                # If dist < r_sum, they overlap
                if dist_sq < r_sum**2:
                    overlap_dist = r_sum - math.sqrt(dist_sq)
                    loss += penalty_weight * overlap_dist**2
                    
        # We want to maximize sum of radii, so we minimize negative sum
        return -radii_sum + loss

    # --- Optimization Setup ---
    # Variables: x0, y0, r0, x1, y1, r1, ...
    x0 = np.concatenate([centers.flatten(), radii])
    
    # Bounds for L-BFGS-B: x in [0, 1], y in [0, 1], r in [0, 0.5]
    bounds = []
    for _ in range(N):
        bounds.append((0, 1)) # x
        bounds.append((0, 1)) # y
        bounds.append((0, 0.5)) # r

    # Run optimization with multiple restarts to improve results
    best_vars = x0
    best_obj = objective(x0)
    
    for iteration in range(3): # 3 restarts
        # Use L-BFGS-B
        res = minimize(
            objective,
            best_vars,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 1000, 'ftol': 1e-12, 'gtol': 1e-10}
        )
        
        if res.fun < best_obj:
            best_vars = res.x
            best_obj = res.fun
            
        # Prepare for next restart: Perturb centers and increase radii slightly
        # Extract current best
        c_best = best_vars[:2*N].reshape(N, 2)
        r_best = best_vars[2*N:]
        
        # Add small noise to centers to escape local minima
        noise = np.random.normal(0, 0.005, c_best.shape)
        c_perturbed = c_best + noise
        
        # Clamp perturbed centers to valid range
        c_perturbed = np.clip(c_perturbed, 1e-5, 1 - 1e-5)
        
        # Increase radii slightly to encourage growth
        r_perturbed = r_best * 1.05
        r_perturbed = np.clip(r_perturbed, 1e-6, 0.5)
        
        # Reconstruct variables for next iteration
        best_vars = np.concatenate([c_perturbed.flatten(), r_perturbed])

    # --- Final Extraction and Formatting ---
    final_centers = best_vars[:2*N].reshape(N, 2)
    final_radii = best_vars[2*N:]
    
    # Ensure non-negative radii (clipping just in case)
    final_radii = np.maximum(final_radii, 0.0)
    
    # Calculate final sum
    sum_radii = np.sum(final_radii)
    
    return final_centers, final_radii, sum_radii
