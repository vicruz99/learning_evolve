import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    # 1. Initial configuration: Hexagonal lattice pattern
    centers = np.zeros((n, 2))
    idx = 0
    y = 0.1
    row = 0
    
    # Fill rows with hexagonal spacing (approximate)
    while idx < n and y <= 0.9:
        x = 0.1
        offset = 0.05 if row % 2 == 1 else 0.0
        while x <= 0.9 and idx < n:
            centers[idx] = [x + offset, y]
            idx += 1
            x += 0.2
        y += 0.17320508  # sqrt(3) * 0.1
        row += 1

    while idx < n:
        centers[idx] = [0.5, 0.5]
        idx += 1

    centers += np.random.uniform(-0.005, 0.005, size=centers.shape)
    radii = np.full(n, 0.06)

    v0 = np.concatenate([centers.ravel(), radii])
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
    
    def objective(v, mu):
        c = v[:2 * n].reshape(n, 2)
        r = v[2 * n:]
        val = -np.sum(r)  # Maximize sum of radii
        
        # Quadratic penalty for overlaps
        for i in range(n):
            for j in range(i + 1, n):
                d = np.hypot(c[i, 0] - c[j, 0], c[i, 1] - c[j, 1])
                overlap = r[i] + r[j] - d
                if overlap > 0.0:
                    val += mu * overlap ** 2
                    
        # Quadratic penalty for boundary violations
        for i in range(n):
            if c[i, 0] < r[i]:
                val += mu * (r[i] - c[i, 0]) ** 2
            if c[i, 0] > 1.0 - r[i]:
                val += mu * (r[i] - (1.0 - c[i, 0])) ** 2
            if c[i, 1] < r[i]:
                val += mu * (r[i] - c[i, 1]) ** 2
            if c[i, 1] > 1.0 - r[i]:
                val += mu * (r[i] - (1.0 - c[i, 1])) ** 2
                
        return val

    # 2. Continuation method: Optimize with increasing penalty weight
    v = v0
    mu = 500.0
    for _ in range(7):
        res = minimize(
            objective, v, args=(mu,), method='L-BFGS-B', bounds=bounds,
            options={'maxiter': 800, 'ftol': 1e-15, 'gtol': 1e-10}
        )
        v = res.x
        mu *= 10.0
        
    centers_opt = v[:2 * n].reshape(n, 2)
    radii_opt = v[2 * n:].copy()
    
    # 3. Post-processing: Ensure strict feasibility within validator tolerance
    # Iteratively shrink radii if constraints are violated due to numerical precision
    for _ in range(3):  # A few passes are enough to resolve conflicts
        for i in range(n):
            r = radii_opt[i]
            x, y = centers_opt[i]
            
            # Enforce boundary constraints
            r = min(r, x, 1.0 - x, y, 1.0 - y)
            
            # Enforce pairwise non-overlap constraints
            for j in range(n):
                if i == j:
                    continue

                if d < radii_opt[i] + radii_opt[j]:
                    # Reduce radius i just enough to satisfy d >= r_i + r_j
                    required_r = d - radii_opt[j]
                    if required_r < radii_opt[i]:
                        radii_opt[i] = required_r
                        
            radii_opt[i] = max(radii_opt[i], 0.0)
            
    sum_r = float(np.sum(radii_opt))
    return centers_opt, radii_opt, sum_r
