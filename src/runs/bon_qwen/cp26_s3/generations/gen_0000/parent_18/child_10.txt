def compute_optimal_radii(centers):
    """
    Solves the LP to maximize the sum of radii for fixed centers.
    """
    n = centers.shape[0]
    # Variables: r_1, ..., r_n
    
    # Objective: maximize sum(r) => minimize -sum(r)
    c = np.ones(n)
    
    # Inequality constraints: A_ub @ r <= b_ub
    # 1. Boundary constraints: r_i <= x_i, r_i <= 1-x_i, etc.
    # 2. Pairwise constraints: r_i + r_j <= distance(i, j)
    
    A_ub = []
    b_ub = []
    
    # Helper to append constraints
    for i in range(n):
        x, y = centers[i]
        # r_i <= x_i
        row = np.zeros(n)
        row[i] = 1.0
        A_ub.append(row)
        b_ub.append(x)
        
        # r_i <= 1 - x_i
        A_ub.append(row)
        b_ub.append(1.0 - x)
        
        # r_i <= y_i
        A_ub.append(row)
        b_ub.append(y)
        
        # r_i <= 1 - y_i
        A_ub.append(row)
        b_ub.append(1.0 - y)

    # Pairwise constraints
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dist)

    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    # Bounds for radii: 0 <= r_i
    bounds = [(0, None) for _ in range(n)]
    
    try:
        # Method 'highs' is robust for LP
        res = scipy.optimize.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
        else:
            # Fallback to small radii if LP fails
            return np.full(n, 1e-6)
    except Exception:
        return np.full(n, 1e-6)
