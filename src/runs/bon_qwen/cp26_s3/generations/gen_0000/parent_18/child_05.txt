So `dx^2 + dy^2 >= dr^2` => `dist >= sum_radii`.

import numpy as np
from scipy.optimize import minimize

def objective(vars, n):
    """
    Objective function to minimize.
    We want to maximize sum of radii, so we minimize negative sum.
    vars layout: [x1..xn, y1..yn, r1..rn]
    """
    radii = vars[2*n:]
    return -np.sum(radii)

def constraint_func(vars, n):
    """
    Constraint function for SLSQP.
    Returns an array of constraint values that must be >= 0.
    vars layout: [x1..xn, y1..yn, r1..rn]
    """
    x = vars[:n]
    y = vars[n:2*n]
    r = vars[2*n:]
    
    # Total constraints: 4 boundary constraints per circle + 1 overlap constraint per pair
    num_cons = 4 * n + n * (n - 1) // 2
    cons = np.empty(num_cons)
    idx = 0
    
    # Boundary constraints for each circle
    # 1. x - r >= 0  (left wall)
    cons[idx:idx+n] = x - r
    idx += n
    # 2. 1 - x - r >= 0 (right wall)
    cons[idx:idx+n] = 1.0 - x - r
    idx += n
    # 3. y - r >= 0 (bottom wall)
    cons[idx:idx+n] = y - r
    idx += n
    # 4. 1 - y - r >= 0 (top wall)
    cons[idx:idx+n] = 1.0 - y - r
    idx += n
    
    # Overlap constraints: dist(i, j) >= r_i + r_j
    # Equivalent to: (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
    for i in range(n):
        xi, yi, ri = x[i], y[i], r[i]
        for j in range(i + 1, n):
            dx = xi - x[j]
            dy = yi - y[j]
            dr = ri + r[j]
            cons[idx] = dx*dx + dy*dy - dr*dr
            idx += 1
            
    return cons

def run_packing():
    n = 26
    # Bounds: x, y in [0, 1], r in [0, 0.5]
    # Layout matches vars: first 2n are coords, last n are radii
    bounds = [(0.0, 1.0) for _ in range(2 * n)] + [(0.0, 0.5) for _ in range(n)]
    
    # Initialize variables
    # Layout: [x_0...x_25, y_0...y_25, r_0...r_25]
    init_vars = np.zeros(3 * n)
    
    # Generate initial positions based on a dense grid pattern
    # 5x5 grid covers 25 circles. We add a 26th circle in a gap.
    grid_vals = np.array([0.1, 0.3, 0.5, 0.7, 0.9])
    r_init = 0.09 # Slightly less than 0.1 to allow room for optimization
    
    # Collect coordinates for the grid
    coords = []
    for x in grid_vals:
        for y in grid_vals:
            coords.append((x, y))
    
    # Add 26th circle at a gap position (0.2, 0.2) which is empty in the grid
    coords.append((0.2, 0.2))
    
    # Fill x and y coordinates
    init_vars[:n] = np.array([c[0] for c in coords])
    init_vars[n:2*n] = np.array([c[1] for c in coords])
    
    # Fill radii
    # Set initial radii to r_init
    init_vars[2*n:] = r_init
    # Set the 26th circle (last one) to a smaller radius to ensure initial feasibility
    init_vars[3*n - 1] = 0.04
    
    # Optimization 1: From structured grid start
    res1 = minimize(objective, init_vars, args=(n,), method='SLSQP', bounds=bounds, 
                    constraints={'type': 'ineq', 'fun': constraint_func, 'args': (n,)},
                    options={'maxiter': 2000, 'ftol': 1e-12})
    
    # Optimization 2: From perturbed start to escape local minima
    init_vars2 = init_vars.copy()
    np.random.seed(42)
    # Perturb x and y coordinates
    init_vars2[:2*n] += np.random.normal(0, 0.02, (2*n))
    # Ensure coordinates stay within bounds
    init_vars2[:2*n] = np.clip(init_vars2[:2*n], 0.0, 1.0)
    # Scale down radii slightly to ensure constraints are satisfied initially
    init_vars2[2*n:] *= 0.9
    
    res2 = minimize(objective, init_vars2, args=(n,), method='SLSQP', bounds=bounds, 
                    constraints={'type': 'ineq', 'fun': constraint_func, 'args': (n,)},
                    options={'maxiter': 2000, 'ftol': 1e-12})
    
    # Select the best result (objective is negative sum, so lower is better)
    if res2.fun < res1.fun:
        best_vars = res2.x
    else:
        best_vars = res1.x
        
    # Extract results
    # x is first n, y is next n
    centers = np.column_stack((best_vars[:n], best_vars[n:2*n]))
    radii = best_vars[2*n:]
    
    # Ensure non-negative radii (though bounds should handle this)
    radii = np.maximum(radii, 0)
    
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
