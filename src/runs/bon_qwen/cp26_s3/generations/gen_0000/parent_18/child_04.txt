import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def constraints_vec(vars):
    """
    Returns an array of constraint values. All must be >= 0 for feasibility.
    Constraints:
    - x_i - r_i >= 0
    - y_i - r_i >= 0
    - 1 - x_i - r_i >= 0
    - 1 - y_i - r_i >= 0
    - (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
    """
    c = vars.reshape(N_CIRCLES, 3)
    centers = c[:, :2]
    radii = c[:, 2]
    
    cons = []
    # Boundary constraints
    cons.append(centers[:, 0] - radii)
    cons.append(centers[:, 1] - radii)
    cons.append(1.0 - centers[:, 0] - radii)
    cons.append(1.0 - centers[:, 1] - radii)
    
    # Non-overlap constraints (squared distance >= squared sum of radii)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist_sq = np.sum(diff**2, axis=2)
    rad_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    
    # Only need upper triangle for i < j
    triu = np.triu_indices(N_CIRCLES, k=1)
    cons.append(dist_sq[triu] - rad_sum[triu]**2)
    
    return np.concatenate(cons)

def objective_func(vars):
    """Objective: maximize sum of radii <=> minimize negative sum."""
    return -np.sum(vars.reshape(N_CIRCLES, 3)[:, 2])

def run_packing():
    np.random.seed(42)
    best_sum_r = -1.0
    best_centers = None
    best_radii = None
    
    bounds = [(0.0, 1.0)] * (2 * N_CIRCLES) + [(0.0, 0.5)] * N_CIRCLES
    cons = {'type': 'ineq', 'fun': constraints_vec}
    
    # Run optimization multiple times with different seeds to escape local minima
    for seed in range(15):
        np.random.seed(seed)
        
        # Generate initial positions: perturbed hexagonal grid
        centers_init = np.zeros((N_CIRCLES, 2))
        idx = 0
        # Hexagonal spacing factor tuned to fit ~26 circles in [0,1]^2
        sp = 0.155
        for row in range(6):
            for col in range(5):
                if idx >= N_CIRCLES: break
                x = (col + 0.5 + 0.5 * (row % 2)) * sp
                y = (row + 0.5) * sp * np.sqrt(3) / 2
                # Add controlled jitter to break symmetry
                x += np.random.randn() * 0.02
                y += np.random.randn() * 0.02
                centers_init[idx] = [
                    np.clip(x, 0.05, 0.95),
                    np.clip(y, 0.05, 0.95)
                ]
                idx += 1
                
        radii_init = np.full(N_CIRCLES, 0.01)
        x0 = np.concatenate([centers_init.flatten(), radii_init])
        
        try:
            res = minimize(
                objective_func,
                x0,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 2000, 'ftol': 1e-10, 'disp': False}
            )
            
            if res.success:
                c_opt = res.x.reshape(N_CIRCLES, 3)
                centers_opt = c_opt[:, :2]
                radii_opt = c_opt[:, 2]
                
                # Verify strict feasibility within numerical tolerance
                feasible = True
                for i in range(N_CIRCLES):
                    cx, cy, cr = centers_opt[i, 0], centers_opt[i, 1], radii_opt[i]
                    if (cx - cr < -1e-9 or cx + cr > 1 + 1e-9 or 
                        cy - cr < -1e-9 or cy + cr > 1 + 1e-9):
                        feasible = False
                        break
                        
                if feasible:
                    s_r = np.sum(radii_opt)
                    if s_r > best_sum_r:
                        best_sum_r = s_r
                        best_centers = centers_opt.copy()
                        best_radii = radii_opt.copy()
        except Exception:
            continue
            
    return best_centers, best_radii, best_sum_r
