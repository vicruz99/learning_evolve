import numpy as np
from scipy.optimize import minimize

N = 26
TRIU_IDX = np.triu_indices(N, k=1)

def obj(vars):
    """Objective: maximize sum of radii -> minimize negative sum."""
    return -np.sum(vars[2*N:])

def con(vars):
    """
    Inequality constraints: g(vars) >= 0
    1. Boundary constraints for all circles
    2. Non-overlap constraints for all unique pairs
    """
    c = vars[:2*N].reshape((N, 2))
    r = vars[2*N:]
    
    # Boundary constraints: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    cons = np.concatenate([
        c[:, 0] - r,
        1.0 - c[:, 0] - r,
        c[:, 1] - r,
        1.0 - c[:, 1] - r
    ])
    
    # Pairwise distance constraints: dist_ij - (r_i + r_j) >= 0
    diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    
    overlaps = dists[TRIU_IDX] - r_sum[TRIU_IDX]
    return np.concatenate([cons, overlaps])

def run_packing():
    best_sum = 0.0
    best_x = None
    bounds = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N
    constraints = {'type': 'ineq', 'fun': con}
    
    # Multiple restarts to escape local minima
    for seed in range(20):
        np.random.seed(seed)
        # Initialize centers in [0.2, 0.8] and small radii to ensure feasibility
        c_init = np.random.rand(N, 2) * 0.6 + 0.2
        r_init = 0.04 + np.random.rand(N) * 0.02
        x0 = np.concatenate([c_init.flatten(), r_init])
        
        res = minimize(obj, x0, method='SLSQP', bounds=bounds, constraints=constraints,
                       options={'maxiter': 400, 'ftol': 1e-12, 'disp': False})
        
        c = res.x[:2*N].reshape((N, 2))
        r = res.x[2*N:]
        current_sum = np.sum(r)
        
        if current_sum > best_sum:
            # Strict validation step
            valid = True
            if np.any(c < -1e-6) or np.any(c > 1.0+1e-6) or np.any(r < -1e-6):
                valid = False
            
            if valid:
                diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
                dists = np.sqrt(np.sum(diff**2, axis=2))
                r_sum_mat = r[:, np.newaxis] + r[np.newaxis, :]
                if np.any(dists[TRIU_IDX] < r_sum_mat[TRIU_IDX] - 1e-6):
                    valid = False
                    
            if valid:
                best_sum = current_sum
                best_x = res.x.copy()
                
    return best_x[:2*N].reshape((N, 2)), best_x[2*N:], best_sum
