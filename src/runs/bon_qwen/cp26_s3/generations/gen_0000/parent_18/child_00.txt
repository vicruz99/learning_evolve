import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    Packs 26 circles in a unit square [0,1]x[0,1] to maximize the sum of radii.
    Uses SLSQP optimization to handle boundary and non-overlap constraints.
    """
    n = 26
    np.random.seed(42)
    
    # 1. Initialization
    # Place centers randomly in the interior and assign small initial radii
    centers = np.random.uniform(0.1, 0.9, size=(n, 2))

    # Flatten variables: [x1, y1, r1, x2, y2, r2, ...]
    x0 = np.concatenate([centers.flatten(), radii])
    
    # 2. Objective Function: Maximize sum of radii -> Minimize negative sum
    def objective(x):
        r = x[2::3]
        return -np.sum(r)
    
    # 3. Constraints
    # SLSQP uses 'type': 'ineq' for f(x) >= 0
    
    # Boundary Constraints: r <= x <= 1-r  =>  x - r >= 0  and  1 - x - r >= 0
    def boundary_constraint(x):
        x_coords = x[0::3]
        y_coords = x[1::3]
        r = x[2::3]
        cons = []
        for i in range(n):
            # x - r >= 0
            cons.append(x_coords[i] - r[i])
            # 1 - x - r >= 0
            cons.append(1 - x_coords[i] - r[i])
            # y - r >= 0
            cons.append(y_coords[i] - r[i])
            # 1 - y - r >= 0
            cons.append(1 - y_coords[i] - r[i])
        return np.array(cons)
    
    # Non-overlap Constraints: dist(c_i, c_j) >= r_i + r_j

        r = x[2::3]
        cons = []

                dy = y_coords[i] - y_coords[j]

        return np.array(cons)

        (0, 0.5) for _ in range(n) # r bounds

    constraints = [
        {'type': 'ineq', 'fun': boundary_constraint},
        {'type': 'ineq', 'fun': non_overlap_constraints}
    ]
    
    # Run optimization
    res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints,

    return centers_opt, radii_opt, sum_radii
