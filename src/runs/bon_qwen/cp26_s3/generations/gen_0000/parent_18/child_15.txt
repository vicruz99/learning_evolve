import numpy as np
from scipy.optimize import minimize, linprog
import math


def max_radii_lp(centers, n=26):
    """Solve LP to find optimal radii for given centers.
    
    Maximize sum(r_i) subject to:
    - r_i + r_j <= ||c_i - c_j|| for all i < j
    - 0 <= r_i <= min(x_i, 1-x_i, y_i, 1-y_i) for all i
    """
    m = n * (n - 1) // 2
    A = np.zeros((m, n))
    b = np.zeros(m)

    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = math.sqrt(dx * dx + dy * dy)
            A[idx, i] = 1
            A[idx, j] = 1
            b[idx] = dist
            idx += 1

    bounds = []
    for i in range(n):
        x, y = centers[i]
        boundary_limit = min(x, 1 - x, y, 1 - y)
        if boundary_limit < 0:
            boundary_limit = 0
        bounds.append((0, boundary_limit))

    c = -np.ones(n)

    try:
        result = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if result.success and result.fun is not None:
            return result.x, -result.fun
    except Exception:
        pass

    # Fallback
    return np.ones(n) * 0.01, 0.26


def position_objective(centers_flat, n=26):
    """Objective for position optimization: negative of max achievable sum of radii."""
    centers = centers_flat.reshape(n, 2)
    # Clip centers to valid range to avoid issues
    centers = np.clip(centers, 0, 1)
    radii, sum_r = max_radii_lp(centers, n)
    return -sum_r


def joint_objective(params, n=26):
    """Joint optimization of positions and radii with penalty method."""
    centers = params[:2 * n].reshape(n, 2)
    radii = params[2 * n:]
    radii = np.maximum(radii, 1e-10)

    # Primary objective
    obj = -np.sum(radii)

    # Penalty for constraint violations
    penalty = 0.0

    # Boundary penalties
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < 0:
            penalty += 5000 * (r - x) ** 2
        if x + r > 1:
            penalty += 5000 * (x + r - 1) ** 2
        if y - r < 0:
            penalty += 5000 * (r - y) ** 2
        if y + r > 1:
            penalty += 5000 * (y + r - 1) ** 2

    # Overlap penalties
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = math.sqrt(dx * dx + dy * dy)
            r_sum = radii[i] + radii[j]
            if dist < r_sum:
                penalty += 5000 * (r_sum - dist) ** 2

    return obj + penalty


def validate_solution(centers, radii, n=26):
    """Check if solution is valid."""
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if r < 0:
            return False
        if x - r < -1e-9 or x + r > 1 + 1e-9:
            return False
        if y - r < -1e-9 or y + r > 1 + 1e-9:
            return False

    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = math.sqrt(dx * dx + dy * dy)
            if dist < radii[i] + radii[j] - 1e-9:
                return False

    return True


def run_packing():
    n = 26

    best_sum = 0
    best_centers = None
    best_radii = None

    # =====================================================
    # PHASE 1: LP-based position optimization
    # =====================================================
    # Generate diverse initial configurations
    initial_centers_list = []

    # Config 1: 5x5 grid + 1 small circle
    centers = np.zeros((n, 2))
    for i in range(5):
        for j in range(5):
            centers[i * 5 + j] = [0.1 + j * 0.2, 0.1 + i * 0.2]
    centers[25] = [0.5, 0.5]
    initial_centers_list.append(centers.copy())

    # Config 2: 5x5 grid with tighter spacing
    centers = np.zeros((n, 2))
    for i in range(5):
        for j in range(5):
            centers[i * 5 + j] = [0.09 + j * 0.205, 0.09 + i * 0.205]
    centers[25] = [0.5, 0.5]
    initial_centers_list.append(centers.copy())

    # Config 3: Hexagonal packing
    centers = np.zeros((n, 2))
    r = 0.085
    idx = 0
    for row in range(8):
        for col in range(8):
            if idx >= n:
                break
            x = col * r * 1.5 + r
            y = row * r * math.sqrt(3) + r
            if col % 2 == 1:
                y += r * math.sqrt(3) / 2
            if r <= x <= 1 - r and r <= y <= 1 - r:
                centers[idx] = [x, y]
                idx += 1
        if idx >= n:
            break
    while idx < n:
        centers[idx] = [0.5, 0.5]
        idx += 1
    initial_centers_list.append(centers.copy())

    # Config 4: Corner-heavy
    centers = np.zeros((n, 2))
    centers[0] = [0.15, 0.15]
    centers[1] = [0.85, 0.15]
    centers[2] = [0.15, 0.85]
    centers[3] = [0.85, 0.85]
    for i in range(4, n):
        row = (i - 4) // 5
        col = (i - 4) % 5
        centers[i] = [0.3 + col * 0.1, 0.3 + row * 0.1]

    # Config 5-8: Random initializations

    for centers_init in initial_centers_list:

            position_objective,

            args=(n,),

                'xatol': 1e-13,
                'fatol': 1e-13,
                'adaptive': True
            }

        opt_centers = result.x.reshape(n, 2)

    # =====================================================
    # PHASE 2: Joint optimization with penalty method
    # =====================================================
    # Use best solution from Phase 1 as starting point, plus random restarts
    joint_configs = []

        joint_configs.append(params.copy())

    for seed in range(30):
        np.random.seed(seed + 200)
        centers = np.random.rand(n, 2) * 0.6 + 0.2
        radii = np.random.rand(n) * 0.06 + 0.04

        joint_configs.append(params.copy())

    # Grid-based joint configs
    for spacing in [0.18, 0.20, 0.22]:
        centers = np.zeros((n, 2))
        radii = np.ones(n) * spacing / 2
        idx = 0
        for row in range(6):
            for col in range(5):
                if idx >= n:
                    break
                x = spacing / 2 + col * spacing
                y = spacing / 2 + row * spacing
                if x <= 1 - spacing / 2 and y <= 1 - spacing / 2:
                    centers[idx] = [x, y]
                    idx += 1
            if idx >= n:
                break
        while idx < n:
            centers[idx] = [0.5, 0.5]
            radii[idx] = 0.01
            idx += 1
        params = np.concatenate([centers.flatten(), radii])
        joint_configs.append(params.copy())

    for params_init in joint_configs:
        try:
            result = minimize(
                joint_objective,
                params_init,
                args=(n,),
                method='Nelder-Mead',
                options={
                    'maxiter': 300000,
                    'xatol': 1e-14,
                    'fatol': 1e-14
                }
            )

            opt_params = result.x
            opt_centers = opt_params[:2 * n].reshape(n, 2)
            opt_radii = np.maximum(opt_params[2 * n:], 1e-10)

            # Fine-tune radii using LP
            opt_radii_refined, refined_sum = max_radii_lp(opt_centers, n)

            if validate_solution(opt_centers, opt_radii_refined, n):
                if refined_sum > best_sum:
                    best_sum = refined_sum
                    best_centers = opt_centers.copy()
                    best_radii = opt_radii_refined.copy()
        except Exception:
            continue

    # =====================================================
    # PHASE 3: Refinement - iterative position + radius updates
    # =====================================================
    if best_centers is not None:
        for iteration in range(50):
            improved = False
            # Try small random perturbations
            np.random.seed(iteration * 31 + 7)
            for _ in range(100):
                perturbed = best_centers.copy()
                perturbed += np.random.randn(n, 2) * 0.005

                if validate_solution(perturbed, opt_radii_new, n):

                        best_sum = new_sum
                        best_centers = perturbed.copy()

                break

    # Fallback if nothing worked
    if best_centers is None or best_sum < 1:
        best_centers = np.zeros((n, 2))
        best_radii = np.ones(n) * 0.05
        for i in range(n):
            row = i // 6
            col = i % 6
            best_centers[i] = [0.1 + col * 0.15, 0.1 + row * 0.17]
        best_sum = np.sum(best_radii)

    return best_centers, best_radii, best_sum
