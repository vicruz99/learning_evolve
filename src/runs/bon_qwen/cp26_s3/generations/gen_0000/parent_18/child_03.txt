def run_packing():
    n = 26

    best_sum = 0
    best_centers = None
    best_radii = None

    for trial in range(40):
        centers = np.zeros((n, 2))
        idx = 0

        for row in range(8):
            for col in range(6):
                if idx >= n:
                    break
                x = col * 0.185 + (0.0925 if row % 2 == 1 else 0)
                y = row * 0.16
                cx = x + 0.09 + np.random.rand() * 0.12
                cy = y + 0.09 + np.random.rand() * 0.12
                cx = np.clip(cx, 0.05, 0.95)
                cy = np.clip(cy, 0.05, 0.95)
                centers[idx] = [cx, cy]
                idx += 1

        radii = np.ones(n) * 0.012

        dt = 0.12
        repulsion_k = 1.2
        boundary_k = 8.0
        margin = 0.0005
        desired_extra = 0.0004
        expansion_base = 0.00008

        for iteration in range(25000):
            forces = np.zeros((n, 2))

            for i in range(n):
                for j in range(i + 1, n):
                    diff = centers[i] - centers[j]
                    dist_sq = diff[0] * diff[0] + diff[1] * diff[1]
                    dist = np.sqrt(dist_sq + 1e-20)

                    desired_dist = radii[i] + radii[j] + desired_extra
                    if dist < desired_dist:
                        overlap = desired_dist - dist
                        force_mag = overlap / dist * repulsion_k
                        dir_x = diff[0] / dist
                        dir_y = diff[1] / dist
                        forces[i, 0] += force_mag * dir_x
                        forces[i, 1] += force_mag * dir_y
                        forces[j, 0] -= force_mag * dir_x
                        forces[j, 1] -= force_mag * dir_y

            for i in range(n):
                x = centers[i, 0]
                y = centers[i, 1]
                r = radii[i]

                if x - r < margin:
                    forces[i, 0] += boundary_k * (margin - (x - r))
                if x + r > 1.0 - margin:
                    forces[i, 0] -= boundary_k * (x + r - 1.0 + margin)
                if y - r < margin:
                    forces[i, 1] += boundary_k * (margin - (y - r))
                if y + r > 1.0 - margin:
                    forces[i, 1] -= boundary_k * (y + r - 1.0 + margin)

            adaptive_dt = dt / (1.0 + iteration * 0.0003)
            centers[:, 0] += adaptive_dt * forces[:, 0]
            centers[:, 1] += adaptive_dt * forces[:, 1]

            low = margin + 1e-8
            high = 1.0 - margin - 1e-8
            centers[:, 0] = np.clip(centers[:, 0], low, high)
            centers[:, 1] = np.clip(centers[:, 1], low, high)

            decay = 1.0 / (1.0 + iteration * 0.0005)
            radii += expansion_base * decay

            for i in range(n):
                max_r = min(
                    centers[i, 0],
                    1.0 - centers[i, 0],
                    centers[i, 1],
                    1.0 - centers[i, 1]
                )
                for j in range(n):
                    if i != j:
                        dx = centers[i, 0] - centers[j, 0]
                        dy = centers[i, 1] - centers[j, 1]
                        d = np.sqrt(dx * dx + dy * dy)
                        allowed = (d - radii[j]) / 2.0
                        if allowed < max_r:
                            max_r = allowed
                radii[i] = max(1e-10, max_r)

        current_sum = np.sum(radii)
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers.copy()
            best_radii = radii.copy()

    for i in range(n):
        max_r = min(
            best_centers[i, 0],
            1.0 - best_centers[i, 0],
            best_centers[i, 1],
            1.0 - best_centers[i, 1]
        )
        for j in range(n):
            if i != j:
                dx = best_centers[i, 0] - best_centers[j, 0]
                dy = best_centers[i, 1] - best_centers[j, 1]
                d = np.sqrt(dx * dx + dy * dy)
                allowed = (d - best_radii[j]) / 2.0
                if allowed < max_r:
                    max_r = allowed
        best_radii[i] = max(0.0, max_r)

    return best_centers, best_radii, float(np.sum(best_radii))
