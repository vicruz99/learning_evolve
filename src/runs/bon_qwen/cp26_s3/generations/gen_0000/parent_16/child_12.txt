We formulate the problem as maximizing $\sum r_i$ subject to non-overlap and boundary constraints. This is solved using a penalty-based continuous optimization approach:

1. **Objective & Penalty**: We minimize $L = -\sum r_i + \lambda \cdot (\sum_{i<j}\max(0, r_i+r_j - d_{ij})^2 + \sum_i \sum_{walls}\max(0, r_i - \text{dist}_w)^2)$. The quadratic penalty is smooth enough for gradient descent and strictly enforces constraints as $\lambda$ increases.
2. **Initialization**: Multiple starting configurations are generated using a perturbed uniform grid. This provides good geometric spread while avoiding poor local minima.
3. **Gradient Descent**: Analytical gradients are computed for both centers and radii. Centers repel each other and boundaries when overlaps occur. Radii grow when space is available and shrink when constrained. Learning rates decay over time, and the penalty weight $\lambda$ is annealed upward to tighten constraints near convergence.
4. **Validation & Correction**: A final pass computes the maximum constraint violation. If any violation exists (due to numerical precision), all radii are uniformly scaled down to guarantee strict validity. Centers are then clipped to the feasible region $[r_i, 1-r_i]$.

This hybrid approach balances exploration (multiple seeds), exploitation (analytical gradients), and robustness (constraint projection), reliably finding high-density packings.

import numpy as np

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    N = 26
    best_centers = None
    best_radii = None
    best_sum = -1.0
    
    # Try multiple seeds to escape poor local minima
    seeds = list(range(40))
    
    for seed in seeds:
        np.random.seed(seed)
        # Initialize on a perturbed grid layout
        cx, cy = np.meshgrid(np.linspace(0.2, 0.8, 5), np.linspace(0.2, 0.8, 5))
        grid = np.column_stack((cx.ravel(), cy.ravel()))
        grid = np.vstack((grid, [0.5, 0.5]))  # 26 points total
        idx = np.random.permutation(N)[:N]
        centers = grid[idx].copy() + 0.03 * np.random.randn(N, 2)
        centers = np.clip(centers, 0.1, 0.9)
        radii = np.full(N, 0.06)
        
        centers, radii = _optimize(centers, radii)
        s = np.sum(radii)
        if s > best_sum:
            best_sum = s
            best_centers = centers.copy()
            best_radii = radii.copy()
            
    # Enforce strict validity constraints
    best_centers, best_radii = _enforce_validity(best_centers, best_radii)
    best_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, best_sum

def _optimize(centers: np.ndarray, radii: np.ndarray) -> tuple[np.ndarray, np.ndarray]:
    N = centers.shape[0]
    lr_c = 0.01
    lr_r = 0.005
    alpha = 10.0
    
    for step in range(4000):
        # Pairwise distances
        diff = centers[:, None, :] - centers[None, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        np.fill_diagonal(dists, np.inf)
        
        # Circle overlaps
        radii_sum = radii[:, None] + radii[None, :]
        overlap = radii_sum - dists
        overlap[overlap < 0] = 0.0
        
        # Boundary violations
        dist_wall = np.min(np.array([centers[:, 0], 1-centers[:, 0], centers[:, 1], 1-centers[:, 1]]), axis=0)
        wall_overlap = radii - dist_wall
        wall_overlap[wall_overlap < 0] = 0.0
        
        # Gradient for radii: d(-sum(r) + penalty)/dr
        grad_r = 2.0 * (np.sum(overlap, axis=1) + wall_overlap) - 1.0
        
        # Gradient for centers from overlaps
        safe_dists = np.where(dists < 1e-10, 1e-10, dists)
        mask = (overlap > 1e-10)
        F = np.zeros((N, N))
        F[mask] = 2.0 * overlap[mask] / safe_dists[mask]
        
        row_sum = np.sum(F, axis=1, keepdims=True)
        grad_c = F @ centers - row_sum * centers
        
        # Gradient for centers from walls
        dx = np.zeros(N)
        dy = np.zeros(N)
        left = centers[:, 0] < radii
        right = centers[:, 0] > 1 - radii
        bot = centers[:, 1] < radii
        top = centers[:, 1] > 1 - radii
        
        dx[left] = -2.0 * wall_overlap[left]
        dx[right] = 2.0 * wall_overlap[right]
        dy[bot] = -2.0 * wall_overlap[bot]
        dy[top] = 2.0 * wall_overlap[top]
        
        grad_c[:, 0] += dx
        grad_c[:, 1] += dy
        
        # Update parameters
        radii -= lr_r * alpha * grad_r
        centers -= lr_c * alpha * grad_c
        
        # Keep radii positive
        radii = np.maximum(radii, 1e-4)
        
        # Adaptive schedule
        if step > 2000:
            alpha *= 1.002  # Increase penalty weight to enforce constraints
        if step % 500 == 0:
            lr_c *= 0.88
            lr_r *= 0.88
            
    return centers, radii

def _enforce_validity(centers: np.ndarray, radii: np.ndarray) -> tuple[np.ndarray, np.ndarray]:
    # Compute maximum constraint violation
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    radii_sum = radii[:, None] + radii[None, :]
    max_viol = float(np.max(radii_sum - dists))
    
    wall_dist = np.min(np.array([centers[:, 0], 1-centers[:, 0], centers[:, 1], 1-centers[:, 1]]), axis=0)
    max_viol = max(max_viol, float(np.max(radii - wall_dist)))
    
    # If violations exist, uniformly shrink radii to satisfy constraints
    if max_viol > 1e-9:
        factor = 1.0 - (max_viol + 1e-8) / 0.5
        radii *= factor
        
    # Strictly clamp centers to the valid region [r, 1-r]
    centers[:, 0] = np.clip(centers[:, 0], radii, 1 - radii)
    centers[:, 1] = np.clip(centers[:, 1], radii, 1 - radii)
    
    return centers, radii
