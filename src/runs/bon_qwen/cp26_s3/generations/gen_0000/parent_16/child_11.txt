

<strategy>
To maximize the sum of radii for 26 circles in a unit square, we formulate a non-linear constrained optimization problem:
1. **Variables**: $(x_i, y_i, r_i)$ for each circle $i=1,\dots,26$.
2. **Objective**: Maximize $\sum r_i$ (equivalently, minimize $-\sum r_i$).
3. **Constraints**: 
   - Boundary: $r_i \le x_i \le 1-r_i$, $r_i \le y_i \le 1-r_i$
   - Non-overlap: $(x_i-x_j)^2 + (y_i-y_j)^2 \ge (r_i+r_j)^2$ for all $i<j$
   - Non-negativity: $r_i \ge 0$

Since the problem is non-convex, a single optimization run may get stuck in local optima. We use a **multi-start local optimization** strategy:
- Generate multiple initial configurations based on a perturbed hexagonal grid pattern, which is geometrically efficient for packing.
- Run `scipy.optimize.minimize` with the `SLSQP` method on each starting point.
- Track the best feasible solution found across all runs.
- Helper functions are kept top-level with no closures or lambda functions as required.
- The optimizer naturally adjusts radii to be unequal where beneficial, filling gaps efficiently to push the sum of radii toward the target.
</strategy>

```python
import numpy as np
from scipy.optimize import minimize

def objective_fn(params, n):
    """Objective: maximize sum of radii => minimize negative sum."""
    return -np.sum(params[2*n:])

def compute_constraints(params, n):
    """Compute all inequality constraints as an array."""
    centers = params[:2*n].reshape(n, 2)
    radii = params[2*n:]
    
    # 4 constraints per circle for boundaries + pairwise non-overlap constraints
    m = 4*n + n*(n-1)//2
    vals = np.empty(m)
    idx = 0
    
    # x >= r
    vals[idx:idx+n] = centers[:, 0] - radii
    idx += n
    # 1 - x >= r
    vals[idx:idx+n] = 1.0 - centers[:, 0] - radii
    idx += n
    # y >= r
    vals[idx:idx+n] = centers[:, 1] - radii
    idx += n
    # 1 - y >= r
    vals[idx:idx+n] = 1.0 - centers[:, 1] - radii
    idx += n
    
    # dist^2 >= (r_i + r_j)^2
    for i in range(n):
        for j in range(i+1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            vals[idx] = dx*dx + dy*dy - (radii[i] + radii[j])**2
            idx += 1
    return vals

def constraint_fun(params):
    """Wrapper to match scipy's expected signature without using lambdas."""
    return compute_constraints(params, 26)

def run_packing():
    n = 26
    bounds = [(0.0, 1.0)]*(2*n) + [(0.0, 0.5)]*n
    cons = {'type': 'ineq', 'fun': constraint_fun}
    
    best_sum = -1.0
    best_params = None
    
    # Multi-start optimization to escape local optima
    for seed in range(8):
        np.random.seed(seed)
        centers = np.zeros((n, 2))
        radii = np.full(n, 0.08)
        
        idx = 0
        # Initialize with a perturbed hexagonal grid pattern
        for row in range(5):
            y = 0.2 * row + 0.1
            if row == 4: cols = 4
            elif row % 2 == 0: cols = 6
            else: cols = 5
            
            start_x = (1.0 - (cols - 1) * 0.2) / 2.0
            for col in range(cols):
                if idx < n:
                    centers[idx, 0] = start_x + col * 0.2 + (np.random.random()-0.5)*0.02
                    centers[idx, 1] = y + (np.random.random()-0.5)*0.02
                    idx += 1
                    
        params0 = np.concatenate([centers.flatten(), radii])
        try:
            res = minimize(objective_fn, params0, args=(n,), bounds=bounds, 
                           constraints=cons, method='SLSQP', 
                           options={'maxiter': 1500, 'ftol': 1e-12, 'disp': False})
            if res.success:
                s = -res.fun
                if s > best_sum:
                    best_sum = s
                    best_params = res.x
        except Exception:
            pass
            
    # Fallback configuration in case optimization fails
    if best_params is None:
        centers = np.zeros((n, 2))
        radii = np.full(n, 0.01)
        for i in range(n):
            centers[i, 0] = (i % 5) * 0.2 + 0.1
            centers[i, 1] = (i // 5) * 0.2 + 0.1
            if i >= 25:
                centers[i, 0] = 0.5
                centers[i, 1] = 0.5
        return centers, radii, np.sum(radii)
        
    centers_out = best_params[:2*n].reshape(n, 2)
    radii_out = best_params[2*n:]
    return centers_out, radii_out, np.sum(radii_out)
```
