import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def objective_func(v):
    r = v[2*N_CIRCLES:]
    return -np.sum(r)

def compute_constraints(v):
    N = N_CIRCLES
    C = v.reshape((N, 3))
    x, y, r = C[:, 0], C[:, 1], C[:, 2]
    
    # Boundary constraints: circles must stay inside [0,1]x[0,1]
    c_bound = np.concatenate([
        x - r,
        1.0 - x - r,
        y - r,
        1.0 - y - r
    ])
    
    # Pairwise non-overlap constraints
    dx = C[:, 0][:, None] - C[:, 0][None, :]
    dy = C[:, 1][:, None] - C[:, 1][None, :]
    dr = C[:, 2][:, None] + C[:, 2][None, :]
    
    dists = np.sqrt(dx**2 + dy**2)
    mask = np.triu(np.ones((N, N), dtype=bool), k=1)
    c_pairwise = dists[mask] - dr[mask]
    
    return np.concatenate([c_bound, c_pairwise])

def generate_initial_guess(strategy, seed):
    np.random.seed(seed)
    if strategy == 'grid':
        cx = np.repeat(np.linspace(1/12, 11/12, 6), 5)
        cy = np.tile(np.linspace(1/10, 9/10, 5), 6)
        centers = np.column_stack((cx, cy))
        radii = np.full(N_CIRCLES, 0.04)
    elif strategy == 'perturbed_grid':
        cx = np.repeat(np.linspace(1/12, 11/12, 6), 5)
        cy = np.tile(np.linspace(1/10, 9/10, 5), 6)
        centers = np.column_stack((cx, cy))
        centers += np.random.uniform(-0.02, 0.02, centers.shape)
        centers = np.clip(centers, 0.05, 0.95)
        radii = np.full(N_CIRCLES, 0.035)
    elif strategy == 'random':
        centers = np.random.uniform(0.1, 0.9, (N_CIRCLES, 2))
        radii = np.full(N_CIRCLES, 0.025)
    else: # hexagonal
        centers = []
        for row in range(8):
            y = (row + 0.5) * 0.22
            if y > 0.95: break
            x_start = 0.11 if row % 2 == 0 else 0.11 * np.sqrt(3)
            x = x_start
            while x < 0.9:
                centers.append([x, y])
                x += 0.22 * np.sqrt(3)
        centers = np.array(centers[:N_CIRCLES])
        while len(centers) < N_CIRCLES:
            centers = np.vstack([centers, np.random.uniform(0.2, 0.8, 2)])
        radii = np.full(N_CIRCLES, 0.035)
        
    return np.hstack([centers.T, radii])

def run_packing():
    N = N_CIRCLES
    bounds = [(0.0, 1.0)]*N + [(0.0, 1.0)]*N + [(0.0, 0.5)]*N
    
    best_sum = 0.0
    best_v = None
    
    strategies = ['grid', 'perturbed_grid', 'hexagonal', 'random']
    seeds = [42, 123, 456, 789, 101112, 202324]
    
    for strat in strategies:
        for seed in seeds:
            v0 = generate_initial_guess(strat, seed)
            
            # Ensure initial configuration is feasible
            cons_vals = compute_constraints(v0)
            if np.min(cons_vals) < -0.05:
                v0[2*N:] *= 0.7
                
            try:
                res = minimize(objective_func, v0, method='SLSQP', bounds=bounds,
                               constraints={'type': 'ineq', 'fun': compute_constraints},
                               options={'maxiter': 5000, 'ftol': 1e-12, 'disp': False})
                
                final_cons = compute_constraints(res.x)
                # Accept if constraints are satisfied within numerical tolerance
                if np.min(final_cons) >= -1e-10:
                    current_sum = -res.fun
                    if current_sum > best_sum:
                        best_sum = current_sum
                        best_v = res.x.copy()
            except Exception:
                continue
                
    # Fallback if optimization fails entirely
    if best_v is None:
        v0 = generate_initial_guess('grid', 42)
        v0[2*N:] = 0.03
        best_v = v0
        best_sum = -np.sum(v0[2*N:])
        
    centers = best_v[:2*N].reshape((N, 2))
    radii = best_v[2*N:]
    return centers, radii, best_sum
