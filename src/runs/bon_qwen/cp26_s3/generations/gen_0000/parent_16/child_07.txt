import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes the placement of 26 circles in a unit square to maximize the sum of radii.
    """
    N = 26
    best_sum = -1.0
    best_centers = None
    best_radii = None

    def objective(x_flat):
        # x_flat contains [x1, y1, r1, x2, y2, r2, ...]
        # We want to maximize sum of radii, so minimize negative sum
        radii = x_flat[2::3]
        return -np.sum(radii)

    def constraint_boundary(x_flat):
        # Constraints: r <= x <= 1-r  =>  x-r >= 0, 1-r-x >= 0
        # Same for y
        cons = []
        for i in range(N):
            x = x_flat[3 * i]
            y = x_flat[3 * i + 1]
            r = x_flat[3 * i + 2]
            cons.append(x - r)
            cons.append(1.0 - r - x)
            cons.append(y - r)
            cons.append(1.0 - r - y)
        return np.array(cons)

    def constraint_overlap(x_flat):
        # Constraints: dist >= r_i + r_j  =>  dist^2 >= (r_i + r_j)^2
        # To avoid sqrt, we use squared distance.
        # (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
        cons = []
        centers = np.array([x_flat[i*3:(i+1)*3] for i in range(N)]).reshape(-1, 3)
        for i in range(N):
            for j in range(i + 1, N):
                xi, yi, ri = centers[i]
                xj, yj, rj = centers[j]
                dist_sq = (xi - xj)**2 + (yi - yj)**2
                sum_r_sq = (ri + rj)**2
                cons.append(dist_sq - sum_r_sq)
        return np.array(cons)

    def run_optimization(initial_x):
        bounds = [(0, 1) for _ in range(2*N)] # x, y
        bounds += [(0, 0.6) for _ in range(N)] # r (max possible radius is 0.5)
        
        constraints = [
            {'type': 'ineq', 'fun': constraint_boundary},
            {'type': 'ineq', 'fun': constraint_overlap}
        ]

        try:
            res = minimize(
                objective,
                initial_x,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 500, 'ftol': 1e-9}
            )
            return res
        except Exception:
            return None

    # Helper to generate hexagonal packing initialization
    def hex_init():
        # Try to fit 26 circles in a hexagonal pattern
        # Estimate radius
        r_est = 0.1
        rows = []
        y = r_est
        row_idx = 0
        while len(rows) < 6: # Max 6 rows
            # Determine number of circles in this row
            # Shift odd rows by r
            shift = r_est if row_idx % 2 == 1 else 0
            x = r_est + shift
            row_circles = []
            while x <= 1 - r_est:
                row_circles.append([x, y])
                x += 2 * r_est
            
            rows.append(row_circles)
            y += r_est * np.sqrt(3)
            row_idx += 1
            
        centers = np.array(rows).flatten()[:26*2].reshape(-1, 2)
        # If we have fewer than 26, pad with random
        if centers.shape[0] < N:
            pad = np.random.rand(N - centers.shape[0], 2) * 0.2 + 0.4
            centers = np.vstack([centers, pad])
        # If more, trim
        centers = centers[:N]
        
        return centers, r_est

    # Run multiple attempts
    # 1. Hexagonal init
    centers_hex, r_hex = hex_init()
    x0 = np.zeros(3 * N)
    x0[0::3] = centers_hex[:, 0]
    x0[1::3] = centers_hex[:, 1]
    x0[2::3] = r_hex
    res = run_optimization(x0)
    if res and res.fun < -best_sum:
        best_sum = -res.fun
        best_centers = res.x[0::3].reshape(-1, 2)
        best_radii = res.x[2::3]

    # 2. Random starts
    for _ in range(20):
        centers_rand = np.random.rand(N, 2) * 0.8 + 0.1 # Keep away from boundaries initially
        radii_rand = np.ones(N) * 0.05
        x0 = np.zeros(3 * N)
        x0[0::3] = centers_rand[:, 0]
        x0[1::3] = centers_rand[:, 1]
        x0[2::3] = radii_rand
        
        res = run_optimization(x0)
        if res and res.fun < -best_sum:
            best_sum = -res.fun
            best_centers = res.x[0::3].reshape(-1, 2)
            best_radii = res.x[2::3]

    # 3. Grid start (5x5 + 1)
    grid_x = np.linspace(0.1, 0.9, 5)
    grid_points = [(x, y) for x in grid_x for y in grid_x]
    # Add center point if not enough, or just take first 26
    grid_points.append((0.5, 0.5))
    grid_points = grid_points[:N]
    centers_grid = np.array(grid_points)
    x0 = np.zeros(3 * N)
    x0[0::3] = centers_grid[:, 0]
    x0[1::3] = centers_grid[:, 1]
    x0[2::3] = 0.08 # Small radius to start
    res = run_optimization(x0)
    if res and res.fun < -best_sum:
        best_sum = -res.fun
        best_centers = res.x[0::3].reshape(-1, 2)
        best_radii = res.x[2::3]

    # Final cleanup: ensure strict non-negativity and bounds
    if best_centers is not None:
        best_radii = np.maximum(best_radii, 0.0)
        # Clip centers just in case
        best_centers[:, 0] = np.clip(best_centers[:, 0], 0, 1)
        best_centers[:, 1] = np.clip(best_centers[:, 1], 0, 1)

    return best_centers, best_radii, best_sum
