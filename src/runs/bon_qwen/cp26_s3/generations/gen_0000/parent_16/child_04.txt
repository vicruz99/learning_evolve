import numpy as np
from scipy.optimize import linprog

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    Uses a hexagonal lattice initialization, iterative expansion, and a final LP refinement.
    """
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    # 1. Initialization: 6x5 Hexagonal Lattice
    row_diam = 6
    row_pitch = np.sqrt(3) / 2
    y = 0.5 / row_diam
    idx = 0
    for r in range(5):
        # Offset odd rows for hexagonal packing
        offset = 0.5 / row_diam if r % 2 == 1 else 0
        for c in range(row_diam):
            if idx < n:
                centers[idx, 0] = 0.5 / row_diam + c / row_diam + offset
                centers[idx, 1] = y + r * (1 / row_diam)
                idx += 1
    radii[:] = 0.08

    # 2. Iterative "Jiggle and Expand" Optimization
    for step in range(5000):
        # Expand radii by 0.02%
        radii *= 1.0002
        
        # Apply noise (decaying) to escape local minima
        noise_mag = 0.002 * (1 - step / 5000)
        centers += np.random.normal(0, noise_mag, centers.shape)

        # Resolve Collisions (5 passes per step for stability)
        for _ in range(5):
            for i in range(n):
                for j in range(i + 1, n):
                    c1, c2 = centers[i], centers[j]
                    r1, r2 = radii[i], radii[j]
                    dist_vec = c2 - c1
                    dist = np.linalg.norm(dist_vec)
                    
                    if dist < r1 + r2:
                        overlap = (r1 + r2 - dist) / 2
                        if dist > 1e-9:
                            shift = overlap * dist_vec / dist
                            centers[i] -= shift
                            centers[j] += shift
                        else:
                            # Handle overlap at exact center
                            centers[i] -= np.random.rand(2) * 0.01
                            centers[j] += np.random.rand(2) * 0.01

        # Resolve Boundaries (Project into valid area)
        for i in range(n):
            r = radii[i]
            centers[i, 0] = np.clip(centers[i, 0], r, 1 - r)
            centers[i, 1] = np.clip(centers[i, 1], r, 1 - r)

    # 3. Final LP Optimization for sum of radii
    # Maximize sum(r_i) s.t. r_i + r_j <= dist_ij and r_i <= bounds
    m = 26
    dists = np.linalg.norm(centers[:, np.newaxis, :] - centers[np.newaxis, :, :], axis=2)
    ub = np.maximum(dists, 0)
    
    # Build A_ub for constraints: r_i + r_j <= dist_ij
    rows, cols = [], []
    for i in range(m):
        for j in range(i + 1, m):
            rows.extend([i, j])
            cols.extend([j, i])
    A_ub_pair = np.zeros((len(rows), m))
    A_ub_pair[np.arange(len(rows)), cols] = 1
    b_ub_pair = dists[np.arange(m), np.arange(m)]
    # Only need upper triangle constraints
    valid_pairs = np.triu_indices(m, 1)
    A_ub_pair = A_ub_pair[np.ix_(valid_pairs, range(m))]
    b_ub_pair = ub[valid_pairs]

    # Build constraints for boundaries
    A_ub_bound = np.eye(m) * 0
    for i in range(m):
        A_ub_bound[i, i] = 1
        # Bounds: r_i <= x_i, r_i <= 1-x_i, etc.
        pass 
    # We'll combine these into a single A_ub for linprog
    # Total rows: pair_constraints + 4*m
    n_pairs = m * (m - 1) // 2
    A_ub = np.zeros((n_pairs + 4 * m, m))
    b_ub = np.zeros(n_pairs + 4 * m)
    
    # Add pair constraints
    k = 0
    for i in range(m):
        for j in range(i + 1, m):
            A_ub[k, i] = 1
            A_ub[k, j] = 1
            b_ub[k] = dists[i, j]
            k += 1
            
    # Add boundary constraints
    for i in range(m):
        A_ub[k, i] = 1; b_ub[k] = centers[i, 0]; k += 1
        A_ub[k, i] = 1; b_ub[k] = 1 - centers[i, 0]; k += 1
        A_ub[k, i] = 1; b_ub[k] = centers[i, 1]; k += 1
        A_ub[k, i] = 1; b_ub[k] = 1 - centers[i, 1]; k += 1

    # Solve LP: max sum(r) -> min -sum(r)
    c_obj = -np.ones(m)
    bounds = [(0, None)] * m
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    radii_final = res.x
    sum_radii = -res.fun
    
    # Update centers to be consistent with final radii (clipping)
    for i in range(n):
        centers[i, 0] = np.clip(centers[i, 0], radii_final[i], 1 - radii_final[i])
        centers[i, 1] = np.clip(centers[i, 1], radii_final[i], 1 - radii_final[i])

    return centers, radii_final, sum_radii
