import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # --- 1. Hexagonal Initialization ---
    # Create a dense set of points in a hexagonal lattice pattern
    candidates = []
    r_init = 0.12
    y = r_init
    row_idx = 0
    
    # Generate enough points to pick the top 26
    while len(candidates) < n + 10:
        x = r_init
        if row_idx % 2 == 1:
            x += r_init # Shift for hexagonal packing
        
        while x + r_init <= 1.0:
            candidates.append([x, y])
            x += 2 * r_init
        y += r_init * np.sqrt(3)
        row_idx += 1

    # Convert to numpy array and select first 26
    centers_init = np.array(candidates[:n])
    
    # Initial radii (slightly smaller than lattice spacing to ensure initial validity)
    radii_init = np.full(n, 0.08)
    
    # --- 2. Optimization Setup ---
    # Concatenate centers and radii into a single optimization vector
    x0 = np.concatenate([centers_init, radii_init])
    
    def objective(x_vars):
        # We want to maximize sum of radii, so minimize negative sum
        radii_vars = x_vars[2*n:]
        return -np.sum(radii_vars)
    
    def constraint_inter(x_vars):
        centers_vars = x_vars[:2*n].reshape(n, 2)
        radii_vars = x_vars[2*n:]
        
        # Compute squared distance matrix
        # dist_sq[i, j] = ||c_i - c_j||^2
        dist_sq = np.sum((centers_vars[:, np.newaxis, :] - centers_vars[np.newaxis, :, :]) ** 2, axis=2)
        
        # Compute required squared distance (sum of radii)^2
        r_sum = radii_vars[:, np.newaxis] + radii_vars[np.newaxis, :]
        min_dist_sq = r_sum**2
        
        # Constraint: dist_sq >= min_dist_sq
        # Return slack: dist_sq - min_dist_sq >= 0
        # We only need upper triangle (i < j)
        mask = np.triu(np.ones((n, n), dtype=bool), k=1)
        return dist_sq[mask] - min_dist_sq[mask]
        
    def constraint_boundary(x_vars):
        centers_vars = x_vars[:2*n].reshape(n, 2)
        radii_vars = x_vars[2*n:]
        
        # x - r >= 0  => x - r >= 0
        # x + r <= 1  => 1 - (x + r) >= 0
        # y - r >= 0
        # y + r <= 1
        
        constraints = []
        for i in range(n):
            cx, cy = centers_vars[i]
            r = radii_vars[i]
            constraints.append(cx - r)
            constraints.append(1 - (cx + r))
            constraints.append(cy - r)
            constraints.append(1 - (cy + r))
            
        return np.array(constraints)
    
    def constraint_radii_pos(x_vars):
        # r >= 0
        return x_vars[2*n:]

    # --- 3. Solver Execution ---
    constraints = [
        {'type': 'ineq', 'fun': constraint_inter},
        {'type': 'ineq', 'fun': constraint_boundary},
        {'type': 'ineq', 'fun': constraint_radii_pos}
    ]

    # SLSQP is robust for non-linear constraints
    result = minimize(
        objective, 
        x0, 
        method='SLSQP', 
        constraints=constraints, 
        options={'ftol': 1e-10, 'maxiter': 1000, 'disp': False}
    )
    
    if result.success:
        optimal_centers = result.x[:2*n].reshape(n, 2)
        optimal_radii = result.x[2*n:]
        sum_radii = np.sum(optimal_radii)
        return optimal_centers, optimal_radii, sum_radii
    else:
        # Fallback if optimization fails
        return centers_init, radii_init, np.sum(radii_init)

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square

    Args:
        centers: np.array of shape (n, 2) with (x, y) coordinates
        radii: np.array of shape (n) with radius of each circle

    Returns:
        True if valid, False otherwise
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True
