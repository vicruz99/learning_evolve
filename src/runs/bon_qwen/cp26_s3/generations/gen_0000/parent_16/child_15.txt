import numpy as np
from scipy.optimize import minimize

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True

def objective(params):
    """
    Objective function to minimize: -sum(radii)
    params layout: [x1, y1, r1, x2, y2, r2, ...]
    """
    n = len(params) // 3
    radii = params[2::3]
    return -np.sum(radii)

def constraints_func(params):
    """
    Returns list of constraint values >= 0
    """
    n = len(params) // 3
    centers = np.zeros((n, 2))
    centers[:, 0] = params[0::3] # x
    centers[:, 1] = params[1::3] # y
    radii = params[2::3]
    
    constraints_list = []
    
    # 1. Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    # x - r >= 0
    constraints_list.append(params[0::3] - radii)
    # (1 - r) - x >= 0 => 1 - x - r >= 0
    constraints_list.append(1 - params[0::3] - radii)
    # y - r >= 0
    constraints_list.append(params[1::3] - radii)
    # 1 - y - r >= 0
    constraints_list.append(1 - params[1::3] - radii)
    
    # 2. Non-overlap: dist^2 >= (r_i + r_j)^2
    # (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
    # To avoid squaring issues and keep it simpler for gradient, 
    # we can use dist - (r_i + r_j) >= 0 but that involves sqrt.
    # Squared form is differentiable everywhere except 0, but dist > 0 usually.
    # Let's use squared form.
    
    # We only need to check pairs.
    for i in range(n):
        for j in range(i + 1, n):
            xi, yi = centers[i]
            xj, yj = centers[j]
            ri, rj = radii[i], radii[j]
            
            dist_sq = (xi - xj)**2 + (yi - yj)**2
            sum_r_sq = (ri + rj)**2
            
            # Constraint: dist_sq - sum_r_sq >= 0
            constraints_list.append(dist_sq - sum_r_sq)
            
    return np.concatenate(constraints_list)
