import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # Helper to convert vars to centers and radii
    def get_centers_radii(vars):
        # vars shape (n*3,)
        params = np.reshape(vars, (n, 3))
        r = params[:, 0]
        u = params[:, 1]
        v = params[:, 2]
        
        # Centers calculation ensuring boundary constraints
        # x = r + u * (1 - 2r) => if u=0, x=r; if u=1, x=1-r
        width = 1.0 - 2.0 * r
        x = r + u * width
        y = r + v * width
        return x, y, r

    # Objective: Minimize negative sum of radii
    def objective(vars):
        _, _, r = get_centers_radii(vars)
        return -np.sum(r)

    # Constraints: Non-overlap between all pairs
    # Returns a 1D array of constraint values, all must be >= 0
    def constraints(vars):
        x, y, r = get_centers_radii(vars)
        
        # Compute squared distances matrix (n x n)
        # Broadcasting: (n, 1) - (1, n)
        diff_x = x[:, np.newaxis] - x[np.newaxis, :]
        diff_y = y[:, np.newaxis] - y[np.newaxis, :]
        dist_sq = diff_x**2 + diff_y**2
        
        # Compute squared sum of radii matrix
        r_sum = r[:, np.newaxis] + r[np.newaxis, :]
        r_sum_sq = r_sum**2
        
        # Constraint: dist_sq >= r_sum_sq  => dist_sq - r_sum_sq >= 0
        mat = dist_sq - r_sum_sq
        
        # We only need constraints for i < j (upper triangle)
        mask = np.triu(np.ones((n, n), dtype=bool), k=1)
        return mat[mask]

    # Generate initial guess: Hexagonal grid
    def get_initial_guess(perturbation=0.0):
        # Estimate a starting radius. 
        # For 26 circles, r approx 0.1 is feasible. 
        # Start smaller to ensure feasibility.
        r_start = 0.04 
        
        centers = []
        # Hexagonal packing logic
        # Row spacing: sqrt(3)/2 * diameter = sqrt(3) * r_start
        # Col spacing: diameter = 2 * r_start
        # Shift every other row by r_start
        
        y = r_start
        row = 0
        while len(centers) < n:
            x_start = r_start + (row % 2) * r_start
            x = x_start
            while x <= 1.0 - r_start and len(centers) < n:
                centers.append([x, y])
                x += 2 * r_start
            y += np.sqrt(3) * r_start
            row += 1
        
        centers = np.array(centers[:n])
        
        # Convert centers to (r, u, v)
        # x = r + u(1-2r) => u = (x - r) / (1 - 2r)
        # Handle case where 1-2r is 0 (r=0.5), though unlikely here
        r_vals = np.full(n, r_start)
        denom = 1.0 - 2.0 * r_start
        
        u_vals = (centers[:, 0] - r_start) / denom
        v_vals = (centers[:, 1] - r_start) / denom
        
        # Clip u, v to [0, 1] just in case of slight numerical overshoot in generation
        u_vals = np.clip(u_vals, 0, 1)
        v_vals = np.clip(v_vals, 0, 1)
        
        # Stack into flat array
        params = np.column_stack((r_vals, u_vals, v_vals)).flatten()
        
        # Add perturbation
        if perturbation > 0:
            # Perturb u, v slightly, keep r
            noise = np.random.normal(0, perturbation, size=(n, 2))
            u_vals_noise = np.clip(u_vals + noise[:, 0], 0, 1)
            v_vals_noise = np.clip(v_vals + noise[:, 1], 0, 1)
            params[1::3] = u_vals_noise
            params[2::3] = v_vals_noise
            
        return params

    # Bounds for variables
    # r in [0, 0.5], u in [0, 1], v in [0, 1]
    bounds = []
    for i in range(n):
        bounds.append((0.0, 0.5)) # r
        bounds.append((0.0, 1.0)) # u
        bounds.append((0.0, 1.0)) # v

    # Optimization setup
    constraints_def = {'type': 'ineq', 'fun': constraints}
    
    best_result = None
    best_sum_radii = -np.inf
    
    # Run optimization multiple times with different perturbations to find global optimum
    # Use a fixed seed for reproducibility if needed, but random helps escape local minima
    np.random.seed(42)
    
    # Try 5 different initial conditions
    for seed in range(5):
        # Perturb initial hex grid
        perturbation = 0.02 * (seed + 1)
        x0 = get_initial_guess(perturbation)
        
        try:
            res = minimize(
                objective, 
                x0, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=constraints_def,
                options={'maxiter': 2000, 'ftol': 1e-9}
            )
            
            if res.success or res.nit > 10: # Check if optimization ran
                current_sum = -res.fun
                if current_sum > best_sum_radii:
                    best_sum_radii = current_sum
                    best_result = res
                    
        except Exception:
            continue

    if best_result is not None:
        x, y, r = get_centers_radii(best_result.x)
        centers = np.column_stack((x, y))
        radii = r
        sum_radii = np.sum(radii)
        
        # Final validation check (optional but good for debug)
        # Note: The solver ensures constraints, but float precision might be tight.
        # The provided validate function handles small errors (1e-12).
        
        return centers, radii, sum_radii
    else:
        # Fallback: return initial guess if optimization failed
        x0 = get_initial_guess(0.0)
        x, y, r = get_centers_radii(x0)
        return np.column_stack((x, y)), r, np.sum(r)
