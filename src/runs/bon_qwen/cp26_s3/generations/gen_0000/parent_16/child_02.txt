def distance_matrix(centers):
    """Compute pairwise distance matrix for centers."""
    # centers shape (N, 2)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    return np.sqrt(np.sum(diff**2, axis=2))

def penalty_function(params, n):
    """
    Computes the objective: -sum(radii) + penalty for constraints.
    params: flat array [r_1...r_n, x_1...x_n, y_1...y_n]
    """
    radii = params[:n]
    centers = params[n:].reshape(n, 2)
    
    # Ensure non-negative radii for stability (though bounds handle this, 
    # numerical issues can occur in penalty calc)
    r_safe = np.maximum(radii, 0)
    
    # 1. Sum of radii (to maximize, so negative for minimization)
    obj = -np.sum(r_safe)
    
    # 2. Boundary violations
    # Circle i is inside if r_i <= x_i <= 1-r_i and r_i <= y_i <= 1-r_i
    # Violations: x_i < r_i  => x_i - r_i < 0
    #              x_i > 1-r_i => x_i + r_i - 1 > 0
    #              y_i < r_i
    #              y_i > 1-r_i
    
    left_viol = np.maximum(0, r_safe - centers[:, 0])
    right_viol = np.maximum(0, centers[:, 0] + r_safe - 1)
    bottom_viol = np.maximum(0, r_safe - centers[:, 1])
    top_viol = np.maximum(0, centers[:, 1] + r_safe - 1)
    
    boundary_penalty = np.sum(left_viol**2 + right_viol**2 + bottom_viol**2 + top_viol**2)
    
    # 3. Overlap violations
    # dist(i, j) < r_i + r_j
    dists = distance_matrix(centers)
    # r_sum[i, j] = r_i + r_j
    r_sum = r_safe[:, np.newaxis] + r_safe[np.newaxis, :]
    
    # Overlap amount
    overlap = np.maximum(0, r_sum - dists)
    # Only consider i < j to avoid double counting and diagonal
    triu_overlap = np.triu(overlap, k=1)
    overlap_penalty = np.sum(triu_overlap**2)
    
    # Total objective
    # Penalty weight should be large enough to enforce constraints
    # but not so large that it dominates the gradient of radii sum.
    # A value around 1000-10000 is typical for this scale.
    penalty_weight = 5000.0
    
    obj = obj + penalty_weight * (boundary_penalty + overlap_penalty)
    
    return obj
