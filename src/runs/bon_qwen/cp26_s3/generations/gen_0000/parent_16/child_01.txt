

Looking at this problem, I need to pack 26 circles in a unit square to maximize the sum of radii. This is fundamentally different from equal-radius packing - we want varied sizes that fill space efficiently.

**Key insight**: Given fixed centers, the optimal radii can be found via linear programming (maximize Σr_i subject to r_i + r_j ≤ dist(i,j) and boundary constraints). Then I optimize centers to maximize this optimal sum.

```python
import numpy as np
from scipy.optimize import linprog, minimize, differential_evolution


def compute_optimal_radii(centers, n):
    """Given fixed centers, find optimal radii via LP to maximize sum of radii."""
    c = -np.ones(n)  # Minimize -sum(r) = maximize sum(r)
    A_ub = []
    b_ub = []
    
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            b_ub.append(dist)
    
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        row = np.zeros(n)
        row[i] = 1.0
        A_ub.append(row)
        bl = min(centers[i, 0], 1.0 - centers[i, 0],
                 centers[i, 1], 1.0 - centers[i, 1])
        b_ub.append(max(0.0, bl))
    
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)] * n
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success and result.x is not None:
        return result.x
    return np.zeros(n)


def objective_func(x, n):
    """Objective: negative sum of optimal radii for given centers."""
    centers = x.reshape(n, 2)
    centers = np.clip(centers, 0.001, 0.999)
    radii = compute_optimal_radii(centers, n)
    return -np.sum(radii)


def hexagonal_init(n, variant=0):
    """Initialize centers in hexagonal packing patterns."""
    centers = np.zeros((n, 2))
    
    # Try multiple row arrangements
    arrangements = [
        [5, 4, 5, 4, 5, 3],   # variant 0
        [4, 5, 4, 5, 4, 4],   # variant 1
        [5, 5, 5, 5, 4, 2],   # variant 2
        [6, 5, 5, 5, 4, 1],   # variant 3
        [5, 5, 5, 6, 5],      # variant 4 (5 rows)
    ]
    
    row_counts = arrangements[variant % len(arrangements)]
    n_rows = len(row_counts)
    
    # Adjust spacing based on arrangement
    max_cols = max(row_counts)
    dx = 1.0 / max_cols
    dy = 1.0 / n_rows
    
    idx = 0
    for row, count in enumerate(row_counts):
        offset = dx * 0.5 if row % 2 == 1 else 0.0
        for col in range(count):
            if idx >= n:
                break
            centers[idx, 0] = (col + 0.5) * dx + offset
            centers[idx, 1] = (row + 0.5) * dy
            idx += 1
    
    return centers


def force_based_optimize(centers, n, n_iter=3000):
    """Use force-based simulation to improve center positions."""
    radii = compute_optimal_radii(centers, n)
    
    for it in range(n_iter):
        forces = np.zeros((n, 2))
        lr = 0.008 / (1.0 + it * 0.0005)
        
        # Inter-circle repulsive forces
        for i in range(n):
            for j in range(i + 1, n):
                diff = centers[i] - centers[j]
                dist = np.sqrt(np.sum(diff ** 2))
                if dist < 1e-12:
                    dist = 1e-12
                    diff = np.array([1e-12, 0.0])
                
                desired = radii[i] + radii[j] + 1e-5
                if dist < desired:
                    f = (desired - dist) / (dist + 1e-12) * 0.5
                    forces[i] += diff * f
                    forces[j] -= diff * f
        
        # Boundary repulsion forces
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            margin = 1e-5
            
            if x < r + margin:
                forces[i, 0] += 15.0 * (r + margin - x)
            if x > 1.0 - r - margin:
                forces[i, 0] -= 15.0 * (x - (1.0 - r - margin))
            if y < r + margin:
                forces[i, 1] += 15.0 * (r + margin - y)
            if y > 1.0 - r - margin:
                forces[i, 1] -= 15.0 * (y - (1.0 - r - margin))
        
        centers += forces * lr
        centers[:, 0] = np.clip(centers[:, 0], 0.001, 0.999)
        centers[:, 1] = np.clip(centers[:, 1], 0.001, 0.999)
        
        # Periodically recompute radii
        if it % 100 == 0:
            radii = compute_optimal_radii(centers, n)
    
    return centers


def random_init(n, seed=42):
    """Initialize centers randomly with some structure."""
    rng = np.random.RandomState(seed)
    centers = rng.uniform(0.1, 0.9, size=(n, 2))
    return centers


def corner_init(n):
    """Initialize with circles near corners and edges."""
    centers = np.zeros((n, 2))
    idx = 0
    
    # 4 corners
    corners = [[0.15, 0.15], [0.85, 0.15], [0.15, 0.85], [0.85, 0.85]]
    for c in corners:
        if idx < n:
            centers[idx] = c
            idx += 1
    
    # Edge centers
    edges = [
        [0.5, 0.1], [0.5, 0.9], [0.1, 0.5], [0.9, 0.5],
        [0.3, 0.1], [0.7, 0.1], [0.3, 0.9], [0.7, 0.9],
        [0.1, 0.3], [0.1, 0.7], [0.9, 0.3], [0.9, 0.7]
    ]
    for e in edges:
        if idx < n:
            centers[idx] = e
            idx += 1
    
    # Fill remaining in center area
    for i in range(idx, n):
        centers[i, 0] = 0.3 + (i - idx) * 0.1
        centers[i, 1] = 0.3 + ((i - idx) // 3) * 0.2
    
    return centers


def try_optimization(centers_init, n, local_method='Nelder-Mead'):
    """Run optimization from given initial centers."""
    centers = force_based_optimize(centers_init, n, n_iter=3000)
    
    x0 = centers.flatten()
    bounds = [(0.001, 0.999)] * (2 * n)
    
    if local_method == 'Nelder-Mead':
        result = minimize(objective_func, x0, args=(n,), method='Nelder-Mead',
                          options={'maxiter': 20000, 'xatol': 1e-10, 'fatol': 1e-12,
                                   'adaptive': True})
    elif local_method == 'L-BFGS-B':
        result = minimize(objective_func, x0, args=(n,), method='L-BFGS-B',
                          bounds=bounds, options={'maxiter': 10000, 'ftol': 1e-15})
    else:
        result = minimize(objective_func, x0, args=(n,), method='Powell',
                          options={'maxiter': 20000, 'xtol': 1e-10})
    
    centers_opt = result.x.reshape(n, 2)
    centers_opt = np.clip(centers_opt, 0.001, 0.999)
    radii_opt = compute_optimal_radii(centers_opt, n)
    
    return centers_opt, radii_opt, np.sum(radii_opt)


def try_force_then_lp(centers_init, n, force_iters=5000):
    """Alternative: extended force optimization then LP."""
    centers = centers_init.copy()
    radii = compute_optimal_radii(centers, n)
    
    for it in range(force_iters):
        forces = np.zeros((n, 2))
        lr = 0.006 / (1.0 + it * 0.0003)
        
        for i in range(n):
            for j in range(i + 1, n):
                diff = centers[i] - centers[j]
                dist = np.sqrt(np.sum(diff ** 2))
                if dist < 1e-12:
                    continue
                
                desired = radii[i] + radii[j] + 1e-6
                if dist < desired:
                    f = (desired - dist) / (dist + 1e-12) * 0.3
                    forces[i] += diff * f
                    forces[j] -= diff * f
        
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            
            if x < r + 1e-6:
                forces[i, 0] += 20.0 * (r + 1e-6 - x)
            if x > 1.0 - r - 1e-6:
                forces[i, 0] -= 20.0 * (x - (1.0 - r - 1e-6))
            if y < r + 1e-6:
                forces[i, 1] += 20.0 * (r + 1e-6 - y)
            if y > 1.0 - r - 1e-6:
                forces[i, 1] -= 20.0 * (y - (1.0 - r - 1e-6))
        
        centers += forces * lr
        centers[:, 0] = np.clip(centers[:, 0], 0.001, 0.999)
        centers[:, 1] = np.clip(centers[:, 1], 0.001, 0.999)
        
        if it % 50 == 0:
            new_radii = compute_optimal_radii(centers, n)
            radii = new_radii
    
    radii = compute_optimal_radii(centers, n)
    return centers, radii, np.sum(radii)


def run_packing():
    n = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Strategy 1: Multiple hexagonal variants with force + local opt
    for variant in range(5):
        centers_init = hexagonal_init(n, variant)
        centers, radii, s = try_optimization(centers_init, n, local_method='Nelder-Mead')
        if s > best_sum:
            best_sum = s
            best_centers = centers.copy()
            best_radii = radii.copy()
    
    # Strategy 2: Corner-based initialization
    centers_init = corner_init(n)
    centers, radii, s = try_optimization(centers_init, n, local_method='Nelder-Mead')
    if s > best_sum:
        best_sum = s
        best_centers = centers.copy()
        best_radii = radii.copy()
    
    # Strategy 3: Random seeds with force-based optimization
    for seed in range(5, 15):
        centers_init = random_init(n, seed)
        centers, radii, s = try_force_then_lp(centers_init, n, force_iters=5000)
        if s > best_sum:
            best_sum = s
            best_centers = centers.copy()
            best_radii = radii.copy()
    
    # Strategy 4: Refine best solution with L-BFGS-B
    if best_centers is not None:
        x0 = best_centers.flatten()
        bounds = [(0.001, 0.999)] * (2 * n)
        result = minimize(objective_func, x0, args=(n,), method='L-BFGS-B',
                          bounds=bounds, options={'maxiter': 5000, 'ftol': 1e-15})
        centers = result.x.reshape(n, 2)
        centers = np.clip(centers, 0.001, 0.999)
        radii = compute_optimal_radii(centers, n)
        s = np.sum(radii)
        if s > best_sum:
            best_sum = s
            best_centers = centers.copy()
            best_radii = radii.copy()
    
    # Strategy 5: Extended force optimization on best solution
    if best_centers is not None:
        centers, radii, s = try_force_then_lp(best_centers, n, force_iters=10000)
        if s > best_sum:
            best_sum = s
            best_centers = centers.copy()
            best_radii = radii.copy()
    
    # Final refinement with Nelder-Mead on best solution
    if best_centers is not None:
        x0 = best_centers.flatten()
        result = minimize(objective_func, x0, args=(n,), method='Nelder-Mead',
                          options={'maxiter': 30000, 'xatol': 1e-12, 'fatol': 1e-13,
                                   'adaptive': True})
        centers = result.x.reshape(n, 2)
        centers = np.clip(centers, 0.001, 0.999)
        radii = compute_optimal_radii(centers, n)
        s = np.sum(radii)
        if s > best_sum:
            best_sum = s
            best_centers = centers.copy()
            best_radii = radii.copy()
    
    # Ensure radii are valid (non-negative, within boundaries)
    if best_radii is not None:
        for i in range(n):
            x, y = best_centers[i]
            r = best_radii[i]
            r = min(r, x, 1.0 - x, y, 1.0 - y)
            best_radii[i] = max(0.0, r)
    
    # Final validation: recompute optimal radii one more time
    if best_centers is not None:
        final_radii = compute_optimal_radii(best_centers, n)
        final_sum = np.sum(final_radii)
        if final_sum >= best_sum:
            best_radii = final_radii
            best_sum = final_sum
    
    return best_centers, best_radii, best_sum
```
