import numpy as np

def get_overlap_penalty(centers, radii):
    """
    Calculate the squared overlap penalty between all pairs of circles.
    Returns 0 if no overlaps, positive value otherwise.
    """
    n = centers.shape[0]
    # Vectorized distance calculation
    # centers: (n, 2)
    # diff: (n, n, 2)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist_sq = np.sum(diff**2, axis=2)
    # Add small epsilon to avoid sqrt(0) issues, though 0 dist implies infinite overlap
    dist = np.sqrt(np.maximum(dist_sq, 1e-12))
    
    # Sum of radii matrix
    r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    
    # Overlap amount: positive means overlap
    overlap = r_sum - dist
    
    # We only care about positive overlaps (violations)
    overlap = np.maximum(overlap, 0)
    
    # Sum of squared overlaps (upper triangle to avoid double counting, though sum over all is fine too)
    # Using np.triu_indices to sum unique pairs
    u, v = np.triu_indices(n, k=1)
    return np.sum(overlap[u, v]**2)

def get_boundary_penalty(centers, radii):
    """
    Calculate penalty for circles going out of [0,1]x[0,1].

    x = centers[:, 0]
    y = centers[:, 1]
    r = radii
    
    penalty = 0.0
    # x - r >= 0  => r - x <= 0. Violation is max(0, r - x)
    v1 = np.maximum(r - x, 0)
    penalty += np.sum(v1**2)
    
    # x + r <= 1  => x + r - 1 <= 0. Violation is max(0, x + r - 1)
    v2 = np.maximum(x + r - 1, 0)
    penalty += np.sum(v2**2)
    
    # y - r >= 0
    v3 = np.maximum(r - y, 0)
    penalty += np.sum(v3**2)
    
    # y + r <= 1
    v4 = np.maximum(y + r - 1, 0)
    penalty += np.sum(v4**2)
    
    return penalty
