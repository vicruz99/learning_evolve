import numpy as np

def generate_hex_init(n, scale=0.8, offset=0.1):
    """Generate a hexagonal lattice initialization scaled to fit in the unit square."""
    pts = []
    for i in range(12):
        for j in range(12):
            x = j + (i % 2) * 0.5
            y = i * np.sqrt(3) / 2
            pts.append([x, y])
    pts = np.array(pts)
    pts -= pts.min(axis=0)
    pts /= (pts.max(axis=0) - pts.min(axis=0))
    idx = np.random.choice(len(pts), n, replace=False)
    return pts[idx] * scale + offset

def generate_random_init(n):
    """Generate a random initialization inside the unit square."""
    return np.random.uniform(0.15, 0.85, (n, 2))

def optimize_iterative(centers, n_iters=5000):

    Iteratively expand circles and resolve overlaps using repulsion forces.
    Returns the best (centers, radii) configuration found during the process.
    """
    n = centers.shape[0]
    radii = np.full(n, 0.005)
    centers = centers.copy().astype(float)
    
    dr = 0.0002
    best_sum = 0
    best_centers = centers.copy()
    best_radii = radii.copy()
    
    for it in range(n_iters):
        radii += dr
        diff = centers[:, None, :] - centers[None, :, :]
        dist = np.sqrt(np.sum(diff**2, axis=2))
        np.fill_diagonal(dist, np.inf)
        
        r_sum = radii[:, None] + radii[None, :]
        overlap = r_sum - dist
        mask = overlap > 0
        
        if mask.any():
            safe_dist = np.where(dist < 1e-8, 1e-8, dist)
            norm = safe_dist / (overlap + 1e-9)
            
            dx = diff[:, :, 0] / norm * (overlap / 2)
            dy = diff[:, :, 1] / norm * (overlap / 2)
            
            centers[:, 0] += np.sum(dx * mask, axis=1)
            centers[:, 1] += np.sum(dy * mask, axis=1)
            
            total_overlap = np.sum(overlap[mask])
            if total_overlap > 0.05:
                dr *= 0.8
            elif total_overlap > 0.01:
                dr *= 0.95
            else:
                dr = min(dr * 1.02, 0.001)
        else:
            dr = min(dr * 1.05, 0.002)
            
        # Enforce boundary constraints
        centers[:, 0] = np.clip(centers[:, 0], radii, 1 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1 - radii)

        if it % 50 == 0:
            temp = 0.004 * (1 - it / n_iters)
            centers += np.random.uniform(-temp, temp, centers.shape)
            centers = np.clip(centers, 0, 1)
            
        curr_sum = np.sum(radii)
        if curr_sum > best_sum:
            best_sum = curr_sum
            best_centers = centers.copy()
            best_radii = radii.copy()
            
    return best_centers, best_radii

def run_packing():

    best_centers = None
    best_radii = None

        s = np.sum(r)
        if s > best_sum:
            best_sum = s
            best_centers = c
            best_radii = r
            
    # Ensure radii are strictly positive
    best_radii = np.maximum(best_radii, 1e-6)
    
    return best_centers, best_radii, best_sum
