from scipy.optimize import minimize, linprog, differential_evolution
import math


def compute_optimal_radii_lp(centers, n=26):
    """
    Solve LP to find maximum radii for given positions.
    Maximizes sum(r_i) subject to:
    - r_i <= distance to boundary for circle i
    - r_i + r_j <= distance between circles i and j
    - r_i >= 0
    """
    c = -np.ones(n)
    
    # Boundary constraints: max radius for each circle based on position
    b = np.zeros(n)
    for i in range(n):
        x, y = centers[i]
        b[i] = min(x, 1.0 - x, y, 1.0 - y)
    b = np.maximum(b, 0.0)
    
    # Build constraint matrix A_ub @ r <= b_ub
    constraints = []
    bounds_rhs = []
    
    # Boundary constraints: r_i <= b[i]
    for i in range(n):
        row = np.zeros(n)
        row[i] = 1.0
        constraints.append(row)
        bounds_rhs.append(b[i])
    
    # Non-overlap constraints: r_i + r_j <= distance(i,j)
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = math.sqrt(dx * dx + dy * dy) + 1e-14
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            constraints.append(row)
            bounds_rhs.append(dist)
    
    A_ub = np.array(constraints)
    b_ub = np.array(bounds_rhs)
    bounds = [(0.0, b[i]) for i in range(n)]
    
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success and result.fun is not None:
            radii = np.maximum(result.x, 0.0)
            return radii, -result.fun
    except Exception:
        pass
    
    # Fallback: return zeros
    return np.zeros(n), 0.0


def objective_for_positions(centers_flat, n=26):
    """Objective function for position optimization: negative sum of optimal radii."""
    centers = centers_flat.reshape(n, 2)
    centers = np.clip(centers, 1e-10, 1.0 - 1e-10)
    _, score = compute_optimal_radii_lp(centers, n)
    return -score


def hexagonal_init(pattern, r_init=0.08):
    """Create initial centers in hexagonal pattern with given row counts."""
    centers = []
    for row_idx, count in enumerate(pattern):
        y = r_init + row_idx * r_init * math.sqrt(3.0)
        offset = r_init if row_idx % 2 == 1 else 0.0
        x_start = r_init + offset
        for col_idx in range(count):
            x = x_start + col_idx * 2.0 * r_init
            centers.append([x, y])
    return np.array(centers)


def random_init(n=26, seed=None):
    """Create random initial centers."""
    rng = np.random.RandomState(seed)
    centers = rng.uniform(0.05, 0.95, size=(n, 2))
    return centers


def grid_init(n=26, cols=6):
    """Create grid-like initial centers."""
    rows = int(math.ceil(n / cols))
    centers = []
    r_init = 0.08
    for i in range(rows):
        for j in range(cols):
            if len(centers) >= n:
                break
            x = r_init + j * 2.0 * r_init
            y = r_init + i * 2.0 * r_init
            centers.append([x, y])
        if len(centers) >= n:
            break
    return np.array(centers[:n])
