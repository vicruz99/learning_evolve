import numpy as np
from scipy.optimize import minimize

def compute_cost(vars, n, mu):
    """
    Computes the objective function: -sum(radii) + penalty * constraint_violations
    """
    c_x = vars[:n]
    c_y = vars[n:2*n]
    r = vars[2*n:]
    
    # Objective: maximize sum of radii -> minimize negative sum
    cost = -np.sum(r)
    
    # Vectorized overlap penalty
    diff_x = c_x[:, np.newaxis] - c_x[np.newaxis, :]
    diff_y = c_y[:, np.newaxis] - c_y[np.newaxis, :]
    dist = np.sqrt(diff_x**2 + diff_y**2)
    rad_sum = r[:, np.newaxis] + r[np.newaxis, :]
    
    # Consider only upper triangle for pairs i < j to avoid double counting
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    violation = np.where(mask, rad_sum - dist, 0.0)
    overlap_pen = np.sum(np.maximum(0.0, violation)**2)
    
    # Boundary penalty
    pen_b = np.sum(np.maximum(0.0, r - c_x)**2)
    pen_b += np.sum(np.maximum(0.0, c_x + r - 1.0)**2)
    pen_b += np.sum(np.maximum(0.0, r - c_y)**2)
    pen_b += np.sum(np.maximum(0.0, c_y + r - 1.0)**2)
    
    return cost + mu * (overlap_pen + pen_b)

def run_packing():
    n = 26
    # Initial hexagonal packing configuration
    r_init = 0.08
    dx = 2 * r_init
    dy = np.sqrt(3) * r_init
    
    centers = []
    radii = []
    
    row_y = r_init
    row_idx = 0
    while row_y + r_init <= 1.0 and len(centers) < n:
        row_x = r_init if row_idx % 2 == 0 else r_init + dx / 2
        while row_x + r_init <= 1.0 and len(centers) < n:
            centers.append([row_x, row_y])
            radii.append(r_init)
            row_x += dx
        row_y += dy
        row_idx += 1
        
    # Flatten to optimization variable vector: [x1, y1, r1, x2, y2, r2, ...]
    x0 = np.zeros(3 * n)
    for i in range(n):
        x0[3*i] = centers[i][0]
        x0[3*i+1] = centers[i][1]
        x0[3*i+2] = radii[i]
        
    bounds = [(0.0, 1.0)] * (2*n) + [(1e-5, 0.5)] * n
    
    current_vars = x0.copy()
    # Continuation method: increase penalty weight gradually to enforce constraints
    for mu in [1.0, 5.0, 10.0, 20.0, 50.0, 100.0, 200.0, 500.0]:
        res = minimize(compute_cost, current_vars, args=(n, mu), 
                       method='L-BFGS-B', bounds=bounds, 
                       options={'maxiter': 1500, 'ftol': 1e-12})
        current_vars = res.x
        
    c_x = current_vars[:n]
    c_y = current_vars[n:2*n]
    r = current_vars[2*n:]
    
    # Post-processing to strictly satisfy boundary constraints
    for i in range(n):
        r[i] = max(r[i], 0.0)
        c_x[i] = np.clip(c_x[i], r[i], 1.0 - r[i])
        c_y[i] = np.clip(c_y[i], r[i], 1.0 - r[i])
        
    centers_opt = np.column_stack((c_x, c_y))
    radii_opt = r
    
    return centers_opt, radii_opt, np.sum(radii_opt)
