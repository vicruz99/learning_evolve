import numpy as np
from scipy.optimize import minimize

def get_max_equal_radius(centers):
    """
    Calculates the maximum possible equal radius r for a given set of centers
    such that all circles fit inside the unit square and do not overlap.
    
    r is limited by:
    1. Distance to boundaries: r <= min(x, 1-x, y, 1-y) for all circles.
    2. Distance to other circles: r <= 0.5 * dist(ci, cj) for all pairs.
    """
    n = centers.shape[0]
    
    # 1. Distance to boundaries
    # For each center (x, y), distance to boundaries is min(x, 1-x, y, 1-y)
    # Since centers must be within [0,1], we assume valid inputs, but clamp just in case
    xs = centers[:, 0]
    ys = centers[:, 1]
    
    dists_boundary = np.minimum(
        np.minimum(xs, 1.0 - xs),
        np.minimum(ys, 1.0 - ys)
    )
    min_boundary_dist = np.min(dists_boundary)
    
    # 2. Pairwise distances
    # Compute all pairwise distances efficiently
    # Centers shape: (n, 2)
    # Diff shape: (n, n, 2)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    # Euclidean distance
    dists_matrix = np.sqrt(np.sum(diff**2, axis=2))
    
    # We only need upper triangular part (i < j)
    # Create a mask for upper triangle excluding diagonal
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    pairwise_dists = dists_matrix[mask]
    
    # Constraint: 2*r <= dist => r <= dist / 2
    min_pairwise_dist = np.min(pairwise_dists)
    max_r_from_pairs = 0.5 * min_pairwise_dist
    
    # The feasible radius is the minimum of these two constraints
    return min(min_boundary_dist, max_r_from_pairs)
