import numpy as np
from scipy.optimize import minimize

N = 26

def _objective(vars_):
    return -np.sum(vars_[2::3])

def _constraint_bounds(vars_):
    res = np.empty(4 * N)
    for i in range(N):
        idx = 3 * i
        r = vars_[idx + 2]
        res[4 * i]     = vars_[idx] - r
        res[4 * i + 1] = 1.0 - vars_[idx] - r
        res[4 * i + 2] = vars_[idx + 1] - r
        res[4 * i + 3] = 1.0 - vars_[idx + 1] - r
    return res

def _constraint_overlaps(vars_):
    m = N * (N - 1) // 2
    res = np.empty(m)
    k = 0
    for i in range(N):
        xi, yi, ri = vars_[3 * i], vars_[3 * i + 1], vars_[3 * i + 2]
        for j in range(i + 1, N):
            dx = xi - vars_[3 * j]
            dy = yi - vars_[3 * j + 1]
            r_sum = ri + vars_[3 * j + 2]
            res[k] = dx * dx + dy * dy - r_sum * r_sum
            k += 1
    return res

def run_packing():
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    for trial in range(5):
        rng = np.random.RandomState(2024 + trial)
        
        centers = np.zeros((N, 2))
        idx = 0
        for row in range(5):
            ncols = 5 if row % 2 == 0 else 6
            for col in range(ncols):
                cx = 0.05 + col * 0.16 + (row % 2) * 0.08
                cy = 0.05 + row * 0.19
                centers[idx] = [cx, cy]
                idx += 1
                
        centers += rng.uniform(-0.02, 0.02, size=centers.shape)
        centers = np.clip(centers, 0.1, 0.9)
        
        radii = np.full(N, 0.05)
        x0 = np.concatenate([centers.ravel(), radii])
        
        bounds = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
        constraints = [
            {'type': 'ineq', 'fun': _constraint_bounds},
            {'type': 'ineq', 'fun': _constraint_overlaps}
        ]
        
        try:
            res = minimize(_objective, x0, method='SLSQP', bounds=bounds, 
                           constraints=constraints, options={'maxiter': 500, 'ftol': 1e-9, 'disp': False})
            if res.success:
                current_sum = -res.fun
                if current_sum > best_sum:
                    best_sum = current_sum
                    best_centers = res.x[:2 * N].reshape(N, 2)
                    best_radii = res.x[2 * N:]
        except Exception:
            pass
            
    if best_centers is None:
        best_centers = np.tile([0.5, 0.5], (N, 1))
        best_radii = np.full(N, 0.01)
        best_sum = 0.26
        
    return best_centers, best_radii, best_sum
