from scipy.optimize import minimize

def run_packing():

    Returns a packing of 26 circles in a unit square that maximizes the sum of radii.
    """
    n_circles = 26
    # 1. Create a good hexagonal-like initial guess
    # 5 rows of 5 circles
    r_init = 0.1
    centers = np.zeros((n_circles, 2))
    
    # Row heights for hexagonal packing: dy = r * sqrt(3)
    dy = r_init * np.sqrt(3)
    
    current_idx = 0
    for row in range(6):
        y = r_init + row * dy
        
        # Determine number of circles and x-offset
        # Rows 0, 2, 4 (even) start at x = r_init (aligned with left wall)
        # Rows 1, 3, 5 (odd) start at x = 2 * r_init (shifted right)
        
        if row % 2 == 0:
            count = 5
            x_start = r_init
        else:
            count = 5
            x_start = 2 * r_init
            
        for i in range(count):
            x = x_start + i * (2 * r_init)
            centers[current_idx, 0] = x
            centers[current_idx, 1] = y
            current_idx += 1

    radii = np.ones(n_circles) * r_init
    
    # 2. Optimization variables: [x1, y1, r1, x2, y2, r2, ...]
    # Total variables: 3 * 26 = 78
    initial_params = np.concatenate([centers.flatten(), radii])
    
    # 3. Define Objective: Maximize sum of radii => Minimize negative sum
    def objective(params):
        return -np.sum(params[52:]) # last 26 params are radii

    # 4. Define Constraints
    def get_constraints():
        constraints = []
        # Boundary constraints: x >= r, x <= 1-r, y >= r, y <= 1-r
        for i in range(n_circles):
            idx_x = i * 3
            idx_y = i * 3 + 1
            idx_r = 52 + i
            
            # x - r >= 0
            constraints.append({
                'type': 'ineq',
                'fun': lambda p, ix=idx_x, ir=idx_r: p[ix] - p[ir]
            })
            # 1 - r - x >= 0
            constraints.append({
                'type': 'ineq',
                'fun': lambda p, ix=idx_x, ir=idx_r: 1 - p[ir] - p[ix]
            })
            # y - r >= 0
            constraints.append({
                'type': 'ineq',
                'fun': lambda p, iy=idx_y, ir=idx_r: p[iy] - p[ir]
            })
            # 1 - r - y >= 0
            constraints.append({
                'type': 'ineq',
                'fun': lambda p, iy=idx_y, ir=idx_r: 1 - p[ir] - p[iy]
            })
            # r >= 0
            constraints.append({
                'type': 'ineq',
                'fun': lambda p, ir=idx_r: p[ir]
            })

        # Pairwise non-overlap constraints: dist >= r_i + r_j
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                idx_xi = i * 3
                idx_yi = i * 3 + 1
                idx_ri = 52 + i
                
                idx_xj = j * 3
                idx_yj = j * 3 + 1
                idx_rj = 52 + j
                
                constraints.append({
                    'type': 'ineq',
                    'fun': lambda p, ix1=idx_xi, iy1=idx_yi, ir1=idx_ri, ix2=idx_xj, iy2=idx_yj, ir2=idx_rj: \
                        np.sqrt((p[ix1] - p[ix2])**2 + (p[iy1] - p[iy2])**2) - (p[ir1] + p[ir2])
                })
        return constraints

    # 5. Run Optimization
    # Using SLSQP which handles constraints well
    constraints_list = get_constraints()
    
    # Bounds for variables to keep things sane during optimization
    # x, y in [0, 1], r in [0, 0.5]
    bounds = []
    for _ in range(26):

    if res.success:

        sort_idx = np.argsort(final_radii)[::-1]
        final_centers = final_centers[sort_idx]
        final_radii = final_radii[sort_idx]
        
        return final_centers, final_radii, np.sum(final_radii)
    else:
        # Fallback to initial guess if optimization fails (should not happen)
        return centers, radii, np.sum(radii)
