import numpy as np
from scipy.optimize import minimize

def objective(vars, n):
    """Maximize sum of radii (minimize negative sum)"""
    return -np.sum(vars[2::3])

def constraints(vars, n):
    """Define packing constraints"""
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    cons = []
    
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    cons.extend(x - r)
    cons.extend(1.0 - x - r)
    cons.extend(y - r)
    cons.extend(1.0 - y - r)
    
    # Non-overlap constraints: dist^2 >= (r_i + r_j)^2
    for i in range(n):
        for j in range(i + 1, n):
            d2 = (x[i] - x[j])**2 + (y[i] - y[j])**2
            cons.append(d2 - (r[i] + r[j])**2)
            
    return np.array(cons)

def run_packing():
    n = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # We try multiple random starts to escape local optima
    for seed in range(10):
        rng = np.random.RandomState(seed)
        
        # Initialize centers on a perturbed hexagonal lattice
        centers = []
        row = 0
        while len(centers) < n:
            col = 0
            y = row * 0.18 * np.sqrt(3) / 2
            while len(centers) < n:
                x = col * 0.18 + (0.09 if row % 2 == 1 else 0)
                if x <= 1.0 and y <= 1.0:
                    centers.append([x, y])
                col += 1
            row += 1
            
        centers = np.array(centers[:n])
        centers += rng.randn(n, 2) * 0.01
        centers = np.clip(centers, 0.05, 0.95)
        
        radii = np.ones(n) * 0.06
        
        # Flatten to optimization vector [x1, y1, r1, x2, y2, r2, ...]
        x0 = np.zeros(3 * n)
        x0[0::3] = centers[:, 0]
        x0[1::3] = centers[:, 1]
        x0[2::3] = radii
        
        # Bounds for variables
        bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
        
        try:
            res = minimize(objective, x0, args=(n,), method='SLSQP',
                           bounds=bounds,
                           constraints={'type': 'ineq', 'fun': constraints, 'args': (n,)},
                           options={'maxiter': 1500, 'ftol': 1e-9, 'disp': False})
            
            if res.success:
                current_sum = -res.fun
                if current_sum > best_sum:
                    best_sum = current_sum
                    best_centers = np.column_stack((res.x[0::3], res.x[1::3]))
                    best_radii = res.x[2::3]
        except Exception:
            continue
            
    # Post-processing to ensure strict validity against 1e-12 tolerance
    if best_centers is not None:
        # Simple iterative push to resolve any remaining tiny overlaps
        for _ in range(50):
            moved = False
            for i in range(n):
                for j in range(i + 1, n):
                    dx = best_centers[j, 0] - best_centers[i, 0]
                    dy = best_centers[j, 1] - best_centers[i, 1]
                    dist = np.sqrt(dx*dx + dy*dy)
                    min_dist = best_radii[i] + best_radii[j]
                    if dist < min_dist - 1e-10:
                        overlap = min_dist - dist
                        if dist > 1e-12:
                            nx, ny = dx/dist, dy/dist
                            shift = overlap / 2.0
                            best_centers[i, 0] -= shift * nx
                            best_centers[i, 1] -= shift * ny
                            best_centers[j, 0] += shift * nx
                            best_centers[j, 1] += shift * ny
                            moved = True
            if not moved:
                break
                
        # Final boundary check and radius shrink if needed
        for i in range(n):
            x, y = best_centers[i]
            r = best_radii[i]
            max_r = min(x, y, 1-x, 1-y)
            if r > max_r:
                best_radii[i] = max_r
                
        best_sum = np.sum(best_radii)
        
    return best_centers, best_radii, best_sum
