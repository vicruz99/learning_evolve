import numpy as np
from scipy.optimize import minimize, linprog
from functools import partial

def boundary_x_lower(x, n, i):
    return x[2*i] - x[2*n + i]

def boundary_x_upper(x, n, i):
    return 1 - x[2*i] - x[2*n + i]

def boundary_y_lower(x, n, i):
    return x[2*i + 1] - x[2*n + i]

def boundary_y_upper(x, n, i):
    return 1 - x[2*i + 1] - x[2*n + i]

def overlap_constraint(x, n, i, j):
    ci = x[2*i:2*i+2]
    cj = x[2*j:2*j+2]
    ri = x[2*n + i]
    rj = x[2*n + j]
    dist_sq = np.sum((ci - cj)**2)
    return dist_sq - (ri + rj)**2

def objective_func(x, n):
    return -np.sum(x[2*n:])

def hexagonal_centers(n, rows, r_init=0.06):
    centers = np.zeros((n, 2))
    idx = 0
    for row_idx, num_circles in enumerate(rows):
        y = r_init + row_idx * np.sqrt(3) * r_init
        x_offset = r_init + (row_idx % 2) * r_init
        spacing = 2 * r_init
        for col in range(num_circles):
            centers[idx, 0] = x_offset + col * spacing
            centers[idx, 1] = y
            idx += 1
    return centers

def solve_radii_lp(centers, n):
    radii = np.zeros(n)
    for trial in range(5):
        A_ub = []
        b_ub = []
        for i in range(n):
            A_ub.append([-1.0] * n)
            b_ub.append(-1e-6)
        for i in range(n):
            bound = min(centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])
            row = [0.0] * n
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(bound)
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                row = [0.0] * n
                row[i] = 1.0
                row[j] = 1.0
                A_ub.append(row)
                b_ub.append(dist)
        A_ub = np.array(A_ub)
        b_ub = np.array(b_ub)
        c_obj = -np.ones(n)
        bounds = [(0, None)] * n
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            radii = result.x.copy()
            break
    return radii

def run_packing():

            centers_init = np.clip(centers_init, 0.02, 0.98)
            radii_init = solve_radii_lp(centers_init, n)
