import numpy as np
from scipy.optimize import minimize

NUM_CIRCLES = 26

def objective(v):
    # Maximize sum of radii -> minimize negative sum
    return -np.sum(v[2::3])

def objective_grad(v):
    g = np.zeros_like(v)
    g[2::3] = -1.0
    return g

def constraints(v):
    n = NUM_CIRCLES
    m = 4 * n + n * (n - 1) // 2
    vals = np.empty(m)
    k = 0
    
    # Boundary constraints: r <= x, r <= 1-x, r <= y, r <= 1-y
    for i in range(n):
        x, y, r = v[3*i], v[3*i+1], v[3*i+2]
        vals[k] = x - r; k += 1
        vals[k] = 1.0 - x - r; k += 1
        vals[k] = y - r; k += 1
        vals[k] = 1.0 - y - r; k += 1
        
    # Pairwise non-overlap constraints
    for i in range(n):
        xi, yi, ri = v[3*i], v[3*i+1], v[3*i+2]
        for j in range(i + 1, n):
            xj, yj, rj = v[3*j], v[3*j+1], v[3*j+2]
            dx = xi - xj
            dy = yi - yj
            vals[k] = dx * dx + dy * dy - (ri + rj) ** 2
            k += 1
    return vals

def generate_initial_guesses():
    guesses = []
    n = NUM_CIRCLES
    
    # Guess 1: 5x5 Grid + 1 center
    c1 = np.zeros((n, 2))
    r1 = np.full(n, 0.085)
    idx = 0
    for i in range(5):
        for j in range(5):
            if idx < n:
                c1[idx] = [i * 0.2 + 0.1, j * 0.2 + 0.1]
                idx += 1
    c1[25] = [0.5, 0.5]
    r1[25] = 0.05
    guesses.append((c1, r1))
    
    # Guess 2: Hexagonal-ish lattice
    c2 = np.zeros((n, 2))
    r2 = np.full(n, 0.09)
    idx = 0
    y = 0.05
    for row in range(6):
        x = 0.05 + (row % 2) * 0.1
        count = 5 if row % 2 == 0 else 4
        for _ in range(count):
            if idx < n:
                c2[idx] = [x, y]
                idx += 1
                x += 0.2
        y += 0.15
    guesses.append((c2, r2))
    
    # Guess 3: Randomized perturbation of grid
    c3 = np.zeros((n, 2))
    r3 = np.full(n, 0.08)
    idx = 0
    for i in range(5):
        for j in range(5):
            if idx < n:
                c3[idx] = [i * 0.2 + 0.1, j * 0.2 + 0.1]
                idx += 1
    c3[25] = [0.5, 0.5]
    np.random.seed(42)
    c3 += np.random.uniform(-0.03, 0.03, size=c3.shape)
    c3 = np.clip(c3, 0.05, 0.95)
    guesses.append((c3, r3))
    
    return guesses

def run_packing():
    n = NUM_CIRCLES
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
    cons = {'type': 'ineq', 'fun': constraints}
    
    guesses = generate_initial_guesses()
    
    for c_init, r_init in guesses:
        x0 = np.zeros(3 * n)
        for i in range(n):
            x0[3*i] = c_init[i, 0]
            x0[3*i+1] = c_init[i, 1]
            x0[3*i+2] = r_init[i]
            
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, 
                           constraints=cons, jac=objective_grad,
                           options={'maxiter': 2000, 'ftol': 1e-10, 'disp': False})
            
            if res.success:
                current_sum = -res.fun
                if current_sum > best_sum:
                    best_sum = current_sum
                    best_centers = res.x[:2*n].reshape(n, 2)
                    best_radii = res.x[2*n:]
        except Exception:
            continue

    # Fallback to a known valid configuration if optimization fails
    if best_centers is None:
        best_centers = np.zeros((n, 2))
        best_radii = np.full(n, 0.09)
        for i in range(n):
            best_centers[i] = [(i % 5) * 0.2 + 0.1, (i // 5) * 0.2 + 0.1]
        best_radii[-1] = 0.05
        best_sum = np.sum(best_radii)
    
    # Post-processing: ensure strict validity within numerical tolerance
    best_radii = np.maximum(best_radii, 1e-7)
    
    # Iterative shrink to satisfy constraints robustly
    for _ in range(20):
        valid = True
        # Check boundaries
        for i in range(n):
            x, y, r = best_centers[i, 0], best_centers[i, 1], best_radii[i]
            if x - r < -1e-7 or x + r > 1.0 + 1e-7 or y - r < -1e-7 or y + r > 1.0 + 1e-7:
                valid = False
                break
        if not valid:
            best_radii *= 0.998
            continue
            
        # Check overlaps
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((best_centers[i] - best_centers[j]) ** 2))
                if dist < best_radii[i] + best_radii[j] - 1e-7:
                    valid = False
                    break
            if not valid:
                break
                
        if valid:
            break
        best_radii *= 0.998
        
    return best_centers, best_radii, float(np.sum(best_radii))
