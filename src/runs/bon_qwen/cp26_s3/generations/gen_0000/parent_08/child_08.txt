### Strategy
1.  **Central Anchor:** Place a single large circle at the center $(0.5, 0.5)$ with radius $0.5$.
2.  **Corner Anchors:** Place four circles of radius $r_c \approx 0.0857$ in the corners to be tangent to the walls and the central circle.
3.  **Edge Fillers:** Place 8 circles along the edges between the corner anchors to utilize the remaining boundary space.
4.  **Internal Fillers:** Place 13 circles in the internal gaps using a hexagonal-inspired offset pattern to maximize space utilization.
5.  **Optimization:** Use a gradient-free optimization approach (simulated annealing-style jitter) to refine the positions and radii, ensuring all constraints (boundary and non-overlap) are met.

import numpy as np
from scipy.optimize import minimize

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    import math
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False

    for i in range(n):
        if radii[i] < 0:
            return False
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            return False

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False
    return True

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    # 1. Initial placement based on strategy
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    # Large central circle

    radii[0] = 0.5

    # Four corner circles (tangent to center and walls)

    # Convert to flat array for scipy
    params = np.concatenate([centers.flatten(), radii])

        return -np.sum(x[52:]) # Maximize sum of radii

        c = x[:52].reshape(26, 2)
        r = x[52:]
        # Boundary constraints
        cons = []
        for i in range(n):
            cons.append(c[i, 0] - r[i])
            cons.append(1 - c[i, 0] - r[i])
            cons.append(c[i, 1] - r[i])
            cons.append(1 - c[i, 1] - r[i])

                cons.append(dist - (r[i] + r[j]))

    centers_opt = res.x[:52].reshape(26, 2)

    max_r_shrink = 0.0

        if any(radii_opt < 0):
            radii_opt[radii_opt < 0] = 0

    return centers_opt, radii_opt, np.sum(radii_opt)
