import numpy as np
from scipy.optimize import minimize


def create_hexagonal_grid(n, seed=0):

    idx = 0
    
    row = 0
    while idx < n:

            n_cols = min(4, n - idx)

            idx += 1
        row += 1

    radii = np.ones(n) * 0.07
    return centers, radii


def compute_penalty(x, n, weight=10000):
    """Compute penalty for constraint violations."""
    centers = x[:2*n].reshape(n, 2)
    radii = x[2*n:]
    
    penalty = 0.0
    
    # Boundary penalties
    for i in range(n):
        r = radii[i]
        cx, cy = centers[i]
        
        # Left: x >= r
        violation = max(0, r - cx)
        penalty += weight * violation**2
        
        # Right: x <= 1 - r
        violation = max(0, r - (1 - cx))
        penalty += weight * violation**2
        
        # Bottom: y >= r
        violation = max(0, r - cy)
        penalty += weight * violation**2
        
        # Top: y <= 1 - r
        violation = max(0, r - (1 - cy))
        penalty += weight * violation**2
    
    # Overlap penalties

        for j in range(i + 1, n):

            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx*dx + dy*dy)
            min_dist = radii[i] + radii[j]
            violation = max(0, min_dist - dist)
            penalty += weight * violation**2
    
    return penalty


def objective(x, n, weight=10000):
    """Objective: maximize sum of radii (minimize negative sum) plus penalty."""
    radii = x[2*n:]
    return -np.sum(radii) + compute_penalty(x, n, weight)


def force_relaxation(centers, radii, n, iterations=200, strength=0.01):
    """Use force-based relaxation to improve packing."""
    centers = centers.copy()
    radii = radii.copy()
    
    for _ in range(iterations):
        forces = np.zeros((n, 2))
        
        # Repulsion forces for overlapping circles
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[j, 0] - centers[i, 0]
                dy = centers[j, 1] - centers[i, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii[i] + radii[j] and dist > 1e-10:
                    overlap = radii[i] + radii[j] - dist
                    fx = (dx / dist) * overlap * strength
                    fy = (dy / dist) * overlap * strength
                    forces[i] -= [fx, fy]
                    forces[j] += [fx, fy]
        
        # Boundary repulsion
        for i in range(n):
            r = radii[i]
            if centers[i, 0] < r:
                forces[i, 0] += strength * (r - centers[i, 0])
            if centers[i, 0] > 1 - r:
                forces[i, 0] -= strength * (centers[i, 0] - (1 - r))
            if centers[i, 1] < r:
                forces[i, 1] += strength * (r - centers[i, 1])
            if centers[i, 1] > 1 - r:
                forces[i, 1] -= strength * (centers[i, 1] - (1 - r))
        
        centers += forces
        centers = np.clip(centers, 0, 1)
    
    return centers


def expand_radii(centers, radii, n, iterations=50, factor=0.0005):
