import numpy as np
from scipy.optimize import linprog, minimize, differential_evolution
from scipy.spatial.distance import cdist
import math

def hexagonal_initialization(n=26):
    """Create a hexagonal initial packing for n circles."""
    patterns = [
        [5, 5, 5, 5, 6],
        [6, 5, 6, 5, 4],
        [5, 6, 5, 6, 4],
        [6, 6, 5, 5, 4],
        [4, 5, 5, 5, 4, 3],
        [5, 5, 5, 5, 5, 1],
        [5, 5, 5, 5, 4, 2],
        [6, 6, 6, 4, 4],
    ]
    
    best_centers = None
    best_score = -1
    
    for pattern in patterns:
        if sum(pattern) != n:
            continue
        
        r = 0.095
        centers = []
        y = r
        
        for row_idx, count in enumerate(pattern):
            if row_idx % 2 == 0:
                x_start = r
            else:
                x_start = 2 * r
            
            for i in range(count):
                x = x_start + i * 2 * r
                if 0 <= x - r and x + r <= 1:
                    centers.append([x, y])
            y += math.sqrt(3) * r
        
        if len(centers) >= n:
            centers = np.array(centers[:n])
            score = 0
            for i in range(n):
                for j in range(i + 1, n):
                    d = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                    if d < 2 * r:
                        score -= (2 * r - d) ** 2
            if score > best_score:
                best_score = score
                best_centers = centers.copy()
    
    if best_centers is None:
        centers = np.random.uniform(0.1, 0.9, (n, 2))
        return centers, np.full(n, 0.05)
    
    radii = np.full(n, r)
    return best_centers, radii


def solve_radii_given_centers(centers, n=26):
    """Solve LP to find optimal radii given fixed centers."""
    A_ub = []
    b_ub = []
    
    for i in range(n):
        x, y = centers[i]
        row = np.zeros(n)
        row[i] = 1.0
        A_ub.append(row.copy())
        b_ub.append(x)
        A_ub.append(row.copy())
        b_ub.append(1.0 - x)
        A_ub.append(row.copy())
        b_ub.append(y)
        A_ub.append(row.copy())
        b_ub.append(1.0 - y)
    
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i][0] - centers[j][0]
            dy = centers[i][1] - centers[j][1]
            dist = math.sqrt(dx * dx + dy * dy)
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dist)
    
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    c = -np.ones(n)
    bounds = [(0.0, None)] * n
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success and np.all(result.x >= -1e-12):
        return np.maximum(result.x, 0.0)
    
    min_r = 0.1
    for i in range(n):
        min_r = min(min_r, centers[i][0], 1 - centers[i][0], centers[i][1], 1 - centers[i][1])
        for j in range(i + 1, n):
            d = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            min_r = min(min_r, d / 2)
    return np.full(n, max(1e-10, min_r))


def optimize_centers_given_radii(centers, radii, n=26, iterations=2000, init_lr=0.005):
    """Optimize centers using force-directed repulsion."""
    centers = centers.copy().astype(float)
    
    for iteration in range(iterations):
        lr = init_lr / (1.0 + iteration * 0.002)
        forces = np.zeros_like(centers)
        has_violation = False
        
        for i in range(n):
            for j in range(i + 1, n):
                diff = centers[i] - centers[j]
                dx, dy = diff[0], diff[1]
                dist = math.sqrt(dx * dx + dy * dy)
                min_dist = radii[i] + radii[j]
                
                if dist < min_dist and dist > 1e-12:
                    has_violation = True
                    overlap = min_dist - dist
                    inv_dist = 1.0 / dist
                    forces[i][0] += overlap * dx * inv_dist
                    forces[i][1] += overlap * dy * inv_dist
                    forces[j][0] -= overlap * dx * inv_dist
                    forces[j][1] -= overlap * dy * inv_dist
        
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < 0:
                forces[i][0] += (r - x) * 10
                has_violation = True
            if x + r > 1:
                forces[i][0] -= (x + r - 1) * 10
                has_violation = True
            if y - r < 0:
                forces[i][1] += (r - y) * 10
                has_violation = True
            if y + r > 1:
                forces[i][1] -= (y + r - 1) * 10
                has_violation = True
        
        if has_violation:
            centers += forces * lr
            for i in range(n):
                centers[i][0] = max(radii[i], min(1 - radii[i], centers[i][0]))
                centers[i][1] = max(radii[i], min(1 - radii[i], centers[i][1]))
        else:
            break
    
    return centers


def alternating_optimization(n=26, num_phases=5):
    """Run alternating optimization with multiple phases."""
    centers, radii = hexagonal_initialization(n)
    best_sum = np.sum(radii)
    best_centers = centers.copy()
    best_radii = radii.copy()
    
    lr_decay = [0.02, 0.01, 0.005, 0.002, 0.001]
    iter_decay = [500, 1000, 2000, 3000, 5000]
    
    for phase in range(num_phases):
        for cycle in range(3):
            radii = solve_radii_given_centers(centers, n)
            lr = lr_decay[min(phase, len(lr_decay) - 1)]
            niter = iter_decay[min(phase, len(iter_decay) - 1)]
            centers = optimize_centers_given_radii(centers, radii, n, iterations=niter, init_lr=lr)
            radii = solve_radii_given_centers(centers, n)
        
        current_sum = np.sum(radii)
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers.copy()
            best_radii = radii.copy()
    
    return best_centers, best_radii


def penalty_objective(params, n=26):
    """Objective for scipy optimization: -sum_radii + penalty."""
    centers = params[:2 * n].reshape(n, 2)
    radii = np.maximum(params[2 * n:], 1e-12)
    
    penalty = 0.0
    pw = 1e8
    
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        v1 = min(0, x - r)
        v2 = min(0, r - x + 1)
        v3 = min(0, y - r)
        v4 = min(0, r - y + 1)
        penalty += pw * (v1 * v1 + v2 * v2 + v3 * v3 + v4 * v4)
    
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i][0] - centers[j][0]
            dy = centers[i][1] - centers[j][1]
            dist = math.sqrt(dx * dx + dy * dy)
            overlap = radii[i] + radii[j] - dist
            if overlap > 0:
                penalty += pw * overlap * overlap
    
    return -np.sum(radii) + penalty


def local_refinement(centers, radii, n=26):
    """Use scipy local optimization to refine."""
    params = np.concatenate([centers.flatten(), radii])
    
    bounds = []
    for i in range(n):
        bounds.append((0, 1))
        bounds.append((0, 1))
    for i in range(n):
        bounds.append((1e-10, 0.5))
    
    try:
        result = minimize(penalty_objective, params, args=(n,), method='L-BFGS-B',
                         bounds=bounds, options={'maxiter': 5000, 'ftol': 1e-15})
        if result.success or result.fun < penalty_objective(params, n):
            new_centers = result.x[:2 * n].reshape(n, 2)
            new_radii = np.maximum(result.x[2 * n:], 1e-12)
            
            sum_orig = np.sum(radii)
            sum_new = np.sum(new_radii)
            
            violation = False
            for i in range(n):
                x, y = new_centers[i]
                r = new_radii[i]
                if x - r < -1e-8 or x + r > 1 + 1e-8 or y - r < -1e-8 or y + r > 1 + 1e-8:
                    violation = True
                    break
                for j in range(i + 1, n):
                    d = np.sqrt(np.sum((new_centers[i] - new_centers[j]) ** 2))
                    if d < new_radii[i] + new_radii[j] - 1e-8:
                        violation = True
                        break
                if violation:
                    break
            
            if not violation and sum_new > sum_orig:
                return new_centers, new_radii
    except Exception:
        pass
    
    return centers, radii


def random_perturbation(centers, radii, n=26, strength=0.02):
    """Apply random perturbation to centers."""
    new_centers = centers.copy()
    for i in range(n):
        r = radii[i]
        dx = np.random.uniform(-strength * r, strength * r)
        dy = np.random.uniform(-strength * r, strength * r)
        new_centers[i][0] = max(r, min(1 - r, new_centers[i][0] + dx))
        new_centers[i][1] = max(r, min(1 - r, new_centers[i][1] + dy))
    return new_centers
