import numpy as np
from scipy.optimize import minimize
import random

def compute_constraints(v):
    """
    Computes all inequality constraints for the packing problem.
    Returns an array where each element must be >= 0.
    """
    n = len(v) // 3
    out = []
    
    # Boundary constraints: 0 <= x, y <= 1 and circles inside
    for i in range(n):
        ix, iy, ir = 2*i, 2*i+1, 2*n+i
        out.append(v[ix] - v[ir])          # x >= r
        out.append(1.0 - v[ix] - v[ir])    # 1 - x >= r
        out.append(v[iy] - v[ir])          # y >= r
        out.append(1.0 - v[iy] - v[ir])    # 1 - y >= r
        
    # Pairwise non-overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            ix1, iy1, ir1 = 2*i, 2*i+1, 2*n+i
            ix2, iy2, ir2 = 2*j, 2*j+1, 2*n+j
            dx = v[ix1] - v[ix2]
            dy = v[iy1] - v[iy2]
            r_sum = v[ir1] + v[ir2]
            out.append(dx*dx + dy*dy - r_sum*r_sum)
            
    return np.array(out)

def objective(v):
    """Objective function: minimize negative sum of radii"""
    n = len(v) // 3
    return -np.sum(v[2*n:])

def run_packing():
    np.random.seed(42)
    random.seed(42)
    n = 26
    
    # 1. Initial Configuration: Hexagonal Grid
    points = []
    dx, dy = 0.22, 0.19
    # Create a hexagonal layout with 26 points
    for r in range(5):
        cols = 5 + (1 if r == 4 else 0)
        for c in range(cols):
            x = 0.1 + c * dx + (r % 2) * (dx / 2)
            y = 0.1 + r * dy
            points.append([x, y])
            
    # Fill remaining if needed
    while len(points) < n:
        points.append([random.uniform(0, 1), random.uniform(0, 1)])
    points = np.array(points[:n])
    
    # Scale to fit comfortably within the square
    xs, ys = points[:, 0], points[:, 1]
    span_x = np.max(xs) - np.min(xs)
    span_y = np.max(ys) - np.min(ys)
    scale = min(0.8 / span_x, 0.8 / span_y) if max(span_x, span_y) > 1e-6 else 1.0
    
    centers_init = np.copy(points)
    centers_init[:, 0] = (centers_init[:, 0] - np.min(xs)) * scale + 0.1
    centers_init[:, 1] = (centers_init[:, 1] - np.min(ys)) * scale + 0.1
    
    # Add small random perturbation to break symmetry and aid convergence
    centers_init += np.random.uniform(-0.005, 0.005, centers_init.shape)
    centers_init = np.clip(centers_init, 0.05, 0.95)
    
    # Start with small radii to ensure initial feasibility
    radii_init = np.full(n, 0.03)
    x0 = np.concatenate([centers_init.flatten(), radii_init])
    
    # Bounds for variables
    bounds = []
    for i in range(n):
        bounds.extend([(0.0, 1.0), (0.0, 1.0)]) # x, y
        bounds.append((1e-4, 0.5))              # r
        
    cons = {'type': 'ineq', 'fun': compute_constraints}
    
    # 2. Optimization
    res = minimize(
        fun=objective,
        x0=x0,
        bounds=bounds,
        constraints=cons,
        method='SLSQP',
        options={'maxiter': 5000, 'ftol': 1e-12, 'disp': False}
    )
    
    centers = res.x[:2*n].reshape((n, 2))
    radii = res.x[2*n:]
    
    # 3. Post-processing for strict validity
    # Clip to boundaries
    for i in range(n):
        r = radii[i]
        centers[i, 0] = np.clip(centers[i, 0], r, 1.0 - r)
        centers[i, 1] = np.clip(centers[i, 1], r, 1.0 - r)
        
    # Relax any numerical overlaps
    for _ in range(200):
        overlap_found = False
        for i in range(n):
            for j in range(i+1, n):
                dx = centers[i,0] - centers[j,0]
                dy = centers[i,1] - centers[j,1]
                dist = np.hypot(dx, dy)
                r_sum = radii[i] + radii[j]
                if dist < r_sum - 1e-8:
                    overlap_found = True
                    if dist < 1e-10: dist = 1e-10
                    push = (r_sum - dist) * 0.5 + 1e-8
                    nx, ny = dx/dist, dy/dist
                    centers[i, 0] += nx * push
                    centers[i, 1] += ny * push
                    centers[j, 0] -= nx * push
                    centers[j, 1] -= ny * push
        if not overlap_found:
            break
            
    # Final boundary clipping
    for i in range(n):
        r = radii[i]
        centers[i, 0] = np.clip(centers[i, 0], r, 1.0 - r)
        centers[i, 1] = np.clip(centers[i, 1], r, 1.0 - r)
        
    return centers, radii, float(np.sum(radii))
