import numpy as np

def resolve_packing(init_centers):
    """
    Iteratively expand circles starting from init_centers while resolving overlaps
    and respecting boundary constraints. Returns a valid configuration.
    """
    n = init_centers.shape[0]
    centers = init_centers.copy()
    r = 0.05
    lr = 0.005
    
    # Expansion phase: slowly grow circles and push them apart
    for step in range(4000):
        r *= 1.0003
        if r > 0.105: 
            r = 0.105  # Safety cap near theoretical optimum
            
        for _ in range(40):
            forces = np.zeros((n, 2))
            for i in range(n):
                # Boundary repulsion
                for d in range(2):
                    if centers[i, d] < r:
                        forces[i, d] += (r - centers[i, d]) * 500.0
                    elif centers[i, d] > 1.0 - r:
                        forces[i, d] -= (centers[i, d] - (1.0 - r)) * 500.0
                        
                # Circle-circle repulsion
                for j in range(i + 1, n):
                    dx = centers[i, 0] - centers[j, 0]
                    dy = centers[i, 1] - centers[j, 1]
                    dist_sq = dx*dx + dy*dy
                    dist = np.sqrt(dist_sq)
                    if dist < 2.0 * r and dist > 1e-9:
                        repulsion = (2.0 * r - dist) * 200.0
                        fx = (dx / dist) * repulsion
                        fy = (dy / dist) * repulsion
                        forces[i, 0] += fx
                        forces[i, 1] += fy
                        forces[j, 0] -= fx
                        forces[j, 1] -= fy
                        
            centers += forces * lr
            # Hard project to feasible region [r, 1-r]
            centers[:, 0] = np.clip(centers[:, 0], r, 1.0 - r)
            centers[:, 1] = np.clip(centers[:, 1], r, 1.0 - r)
            
            lr *= 0.9998
            if lr < 1e-7: lr = 1e-7

    # Correction phase: guarantee zero overlap by slightly reducing r if needed
    for _ in range(200):
        max_ov = 0.0
        for i in range(n):
            for j in range(i + 1, n):
                d = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                if d < 2.0 * r:
                    ov = 2.0 * r - d
                    if ov > max_ov:
                        max_ov = ov
        if max_ov < 1e-8:
            break
        r -= max_ov * 0.6
        centers[:, 0] = np.clip(centers[:, 0], r, 1.0 - r)
        centers[:, 1] = np.clip(centers[:, 1], r, 1.0 - r)
        
    radii = np.full(n, r)
    return centers, radii

    for init in inits:
