import numpy as np
from scipy.optimize import minimize

def compute_penalty(centers, radii, n):
    """
    Compute the smooth penalty for boundary violations and overlaps.
    Centers is shape (n, 2), radii is shape (n).
    """
    penalty = 0.0
    
    # Boundary penalties (using soft constraints for smoothness in objective)
    # x >= r, x + r <= 1 => x - r >= 0, 1 - (x + r) >= 0
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        
        # x boundaries
        if x < r:
            penalty += (x - r)**2
        elif x > 1 - r:
            penalty += (x - (1 - r))**2
            
        # y boundaries
        if y < r:
            penalty += (y - r)**2
        elif y > 1 - r:
            penalty += (y - (1 - r))**2

    # Overlap penalties
    # Vectorized approach for efficiency
    # Calculate squared distances between all pairs
    # centers shape (n, 2)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist_sq = np.sum(diff**2, axis=2)
    
    # Create lower triangle mask to avoid double counting and self-distance
    # i < j
    i_indices, j_indices = np.tril_indices(n, k=-1)
    
    # Radii sums squared
    r_sum_sq = (radii[i_indices] + radii[j_indices])**2
    
    # Overlap exists if dist_sq < r_sum_sq
    overlap_sq = r_sum_sq - dist_sq[i_indices, j_indices]
    
    # We only care about positive overlaps (when dist < sum of radii)
    # Using max(0, overlap)^2
    penalty += np.sum(np.maximum(0, overlap_sq)**2)
    
    return penalty
