To maximize the sum of radii for 26 circles in a unit square, we treat the circle radii as dependent variables determined by the positions of their centers. Specifically, for any configuration of centers, the maximum valid radius for each circle is half the minimum distance to the nearest other center or to the square's boundary. This transforms the problem into a 52-dimensional continuous optimization problem (optimizing x and y coordinates for 26 circles) where the objective is to maximize the sum of these induced radii.

The problem is highly non-convex with many local optima. We will use Differential Evolution (DE), a robust global optimization algorithm, to search the space of center coordinates. DE is particularly effective for this type of geometric packing problem because it handles non-smooth objective functions well and explores the search space efficiently. We will run DE with multiple random seeds to increase the chance of finding a global optimum. The best result from DE will then be polished using the Nelder-Mead local optimizer to refine positions to high precision. import numpy as np
from scipy.optimize import differential_evolution, minimize

def compute_objective(centers_flat):
    """
    Computes the negative sum of radii for a given set of centers.
    The radius of each circle is determined by half the minimum distance 
    to any other center or to the boundary of the unit square.
    """
    centers = centers_flat.reshape(26, 2)
    cx, cy = centers[:, 0], centers[:, 1]
    
    # Distance to boundaries
    dist_bound = np.minimum(np.minimum(cx, 1.0 - cx), np.minimum(cy, 1.0 - cy))
    
    # Pairwise distances between centers
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist_matrix = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dist_matrix, np.inf)
    dist_min_neighbors = np.min(dist_matrix, axis=1)
    
    # Max valid radius is half the minimum of boundary and neighbor distances
    radii = 0.5 * np.minimum(dist_bound, dist_min_neighbors)
    
    return -np.sum(radii)

def run_packing():
    # Search bounds for each coordinate (x, y) in [0, 1]
    bounds = [(0.0, 1.0)] * 52
    
    best_val = np.inf
    best_x = None
    
    # Run Differential Evolution with multiple seeds to find a good global configuration
    for seed in [0, 1, 2, 3, 4]:
        res = differential_evolution(
            compute_objective, 
            bounds, 
            maxiter=200, 
            popsize=8,
            mutation=(0.5, 1.5), 
            recombination=0.7,
            seed=seed, 
            tol=1e-8, 
            polish=False
        )
        if res.fun < best_val:
            best_val = res.fun
            best_x = res.x
            
    # Local optimization to refine the best global result
    if best_x is not None:
        try:
            res_loc = minimize(
                compute_objective, 
                best_x, 
                method='Nelder-Mead',
                options={'xatol': 1e-8, 'fatol': 1e-8, 'maxiter': 3000}
            )
            if res_loc.fun < best_val:
                best_x = res_loc.x
        except Exception:
            pass
            
    # Compute final radii from optimized centers
    centers = best_x.reshape(26, 2)
    cx, cy = centers[:, 0], centers[:, 1]
    dist_bound = np.minimum(np.minimum(cx, 1.0 - cx), np.minimum(cy, 1.0 - cy))
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist_matrix = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dist_matrix, np.inf)
    dist_min_neighbors = np.min(dist_matrix, axis=1)
    
    radii = 0.5 * np.minimum(dist_bound, dist_min_neighbors)
    
    return centers, radii, np.sum(radii)
