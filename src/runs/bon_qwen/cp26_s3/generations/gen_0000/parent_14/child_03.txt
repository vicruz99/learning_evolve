import numpy as np
import scipy.optimize

def generate_hexagonal_grid(n_points, padding=0.05):
    """
    Generates an initial configuration of n_points in a hexagonal grid pattern 
    scaled to fit within the unit square with a specified padding.
    """
    # Estimate grid dimensions to fit n_points
    # Area of hex cell ~ 0.866 * s^2. n_points * 0.866 * s^2 ~ (1-2*padding)^2
    area = (1 - 2 * padding) ** 2
    spacing = np.sqrt(area / (n_points * 0.866))
    
    points = []
    # Vertical spacing
    row_height = spacing * np.sqrt(3) / 2
    # Horizontal spacing
    col_width = spacing
    
    y = padding
    row_idx = 0
    
    while len(points) < n_points:
        row_shift = (spacing / 2) * (row_idx % 2)
        x = padding + row_shift
        
        while x < 1 - padding and len(points) < n_points:
            points.append([x, y])
            x += col_width
        y += row_height
        row_idx += 1
        
    return np.array(points[:n_points])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes the packing of 26 circles in a unit square to maximize sum of radii.
    Returns (centers, radii, sum_radii).
    """
    n = 26
    
    # 1. Initialization: Generate hexagonal grid for centers
    initial_centers = generate_hexagonal_grid(n, padding=0.02)
    
    # We optimize 4 variables per circle: [x_min, x_max, y_min, y_max]
    # Center = (min + max) / 2, Radius = (max - min) / 2
    # Initial small radius to ensure non-overlap at start
    r_init = 0.02
    initial_bounds = []
    for c in initial_centers:
        initial_bounds.append([c[0] - r_init, c[0] + r_init, c[1] - r_init, c[1] + r_init])
    
    x0 = np.array(initial_bounds).flatten()
    
    # Bounds for the optimization variables: 0 <= min <= max <= 1
    bounds = []
    for _ in range(n):
        bounds.extend([(0, 1), (0, 1), (0, 1), (0, 1)])
        
    # 2. Define Objective and Constraints
    def objective(vars_flat):
        # Maximize sum of radii -> Minimize negative sum of radii
        radii = (vars_flat[1::4] - vars_flat[0::4] + 
                 vars_flat[3::4] - vars_flat[2::4]) / 2
        return -np.sum(radii)
        
    constraints = []
    
    # Constraint: min <= max for all dimensions
    # x_min <= x_max
    for i in range(n):
        idx_min, idx_max = i*4, i*4 + 1
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: v[idx_max] - v[idx_min]})
        
        # y_min <= y_max
        idx_min, idx_max = i*4 + 2, i*4 + 3
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: v[idx_max] - v[idx_min]})

    # Constraints: No overlaps between any pair of circles
    # Distance between centers >= sum of radii
    # (c1 - c2)^2 >= (r1 + r2)^2
    for i in range(n):
        for j in range(i + 1, n):
            # Indices for circle i
            i_min_x, i_max_x = i*4, i*4 + 1
            i_min_y, i_max_y = i*4 + 2, i*4 + 3
            
            # Indices for circle j
            j_min_x, j_max_x = j*4, j*4 + 1
            j_min_y, j_max_y = j*4 + 2, j*4 + 3
            
            def overlap_constraint(v, i=i, j=j):
                # Calculate centers
                cx_i = (v[i_min_x] + v[i_max_x]) / 2
                cy_i = (v[i_min_y] + v[i_max_y]) / 2
                cx_j = (v[j_min_x] + v[j_max_x]) / 2
                cy_j = (v[j_min_y] + v[j_max_y]) / 2
                
                # Calculate radii
                r_i = (v[i_max_x] - v[i_min_x]) / 2
                r_j = (v[j_max_x] - v[j_min_x]) / 2
                
                # Distance squared
                dist_sq = (cx_i - cx_j)**2 + (cy_i - cy_j)**2
                
                # Sum of radii squared
                r_sum_sq = (r_i + r_j)**2
                
                # Constraint: dist_sq - r_sum_sq >= 0
                # Add small tolerance for numerical stability
                return dist_sq - r_sum_sq + 1e-6

            constraints.append({'type': 'ineq', 'fun': overlap_constraint})

            objective, 
            x0, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=constraints, 
            options={'maxiter': 2000, 'ftol': 1e-9, 'disp': False}

        opt_vars = result.x

        # Fallback to initial grid if optimization fails
        opt_vars = x0

    sum_radii = np.sum(radii)
    
    return centers, radii, float(sum_radii)
