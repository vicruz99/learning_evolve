import numpy as np
import math
from scipy.optimize import minimize

def check_validity(centers, radii, tol=1e-9):
    """Helper to check validity manually during optimization if needed, though SLSQP handles constraints."""
    n = centers.shape[0]
    # Boundary
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x < r - tol or x > 1 - r + tol or y < r - tol or y > 1 - r + tol:
            return False
    # Overlap
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - tol:
                return False
    return True

def run_packing():
    n_circles = 26
    
    # Initial guess for hexagonal packing
    # We want to fit 26 circles. 
    # A hexagonal packing with rows of alternating lengths is efficient.
    # Let's try to distribute 26 circles into rows.
    # Possible row configurations summing to 26:
    # 5, 4, 5, 4, 5, 3 (6 rows) -> Total 26.
    # 5, 4, 5, 4, 4, 4 (6 rows) -> Total 26.
    # Let's try 5, 4, 5, 4, 5, 3.
    
    # Estimate radius for initialization. 
    # Target sum 2.636 -> avg r ~ 0.1014.
    # Let's start with r = 0.09 to be safe and let optimizer expand.
    r_init = 0.09
    
    rows_counts = [5, 4, 5, 4, 5, 3]
    centers_init = []
    radii_init = [r_init] * n_circles
    
    # Height calculation for hexagonal packing
    # Vertical spacing between rows is r * sqrt(3)
    # We need to fit these rows in [0, 1]
    # Let's space them evenly first
    
    num_rows = len(rows_counts)
    y_spacing = 1.0 / (num_rows + 1) # Simple spacing initially
    
    current_circle_idx = 0
    for row_idx, count in enumerate(rows_counts):
        y = (row_idx + 1) * y_spacing
        # Horizontal spacing
        # For a row with 'count' circles, width occupied is roughly (count-1)*2r + 2r = 2*count*r
        # But in hex packing, adjacent rows are shifted by r.
        # Let's just space them evenly in x for initialization
        x_spacing = 1.0 / (count + 1)
        for col_idx in range(count):
            x = (col_idx + 1) * x_spacing
            centers_init.append([x, y])
            current_circle_idx += 1
            
    centers_init = np.array(centers_init)
    radii_init = np.array(radii_init)
    
    # Combine into optimization vector
    # Variables: [x0, y0, r0, x1, y1, r1, ..., x25, y25, r25]
    x0 = np.concatenate([centers_init.flatten(), radii_init])
    
    # Objective: Maximize sum of radii -> Minimize -sum(radii)
    def objective(vars):
        radii = vars[2::3] # r0, r1, ...
        return -np.sum(radii)

    # Constraints
    constraints = []
    
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    # x - r >= 0  =>  x - r >= 0
    # 1 - x - r >= 0 => 1 - x - r >= 0
    # y - r >= 0
    # 1 - y - r >= 0
    for i in range(n_circles):
        idx_x = 3 * i
        idx_y = 3 * i + 1
        idx_r = 3 * i + 2
        
        # x - r >= 0
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: v[idx_x] - v[idx_r]})
        # 1 - x - r >= 0
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: 1.0 - v[idx_x] - v[idx_r]})
        # y - r >= 0
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: v[idx_y] - v[idx_r]})
        # 1 - y - r >= 0
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: 1.0 - v[idx_y] - v[idx_r]})
        
        # r >= 0 (though usually handled by bounds or logic, added for safety)
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: v[idx_r]})

    # Non-overlap constraints: dist(i, j) >= r_i + r_j
    # dist^2 >= (r_i + r_j)^2
    # (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
    for i in range(n_circles):
        for j in range(i + 1, n_circles):
            idx_xi = 3 * i
            idx_yi = 3 * i + 1
            idx_ri = 3 * i + 2
            idx_xj = 3 * j
            idx_yj = 3 * j + 1
            idx_rj = 3 * j + 2
            
            def overlap_constraint(v, i=i, j=j):
                xi, yi, ri = v[idx_xi], v[idx_yi], v[idx_ri]
                xj, yj, rj = v[idx_xj], v[idx_yj], v[idx_rj]
                dist_sq = (xi - xj)**2 + (yi - yj)**2
                return dist_sq - (ri + rj)**2
            
            constraints.append({'type': 'ineq', 'fun': overlap_constraint})

    # Bounds for variables to keep search space reasonable
    # x, y in [0, 1]
    # r in [0, 0.5]
    bounds = []
    for i in range(n_circles):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 0.5)) # r

    # Optimization
    # Using SLSQP as it handles constraints well
    try:
        res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints, 
                       options={'maxiter': 1000, 'ftol': 1e-9, 'disp': False})
        
        if res.success:
            final_vars = res.x
            final_centers = np.column_stack((final_vars[0::3], final_vars[1::3]))
            final_radii = final_vars[2::3]
        else:
            # Fallback if optimization fails, return initial guess
            final_centers = centers_init
            final_radii = radii_init
            
    except Exception as e:
        final_centers = centers_init
        final_radii = radii_init

    sum_radii = np.sum(final_radii)
    return final_centers, final_radii, sum_radii
