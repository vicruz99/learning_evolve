def solve_lp_for_radii(centers):
    """
    Given centers, solve the LP to find radii that maximize sum(r_i).
    
    Maximize sum(r_i)
    Subject to:
      r_i + r_j <= dist(centers[i], centers[j])
      r_i <= x_i
      r_i <= 1 - x_i
      r_i <= y_i
      r_i <= 1 - y_i
      r_i >= 0
    """
    n = len(centers)
    c = -np.ones(n) # Maximize sum(r) -> Minimize -sum(r)
    
    # Inequality constraints A_ub @ r <= b_ub
    A_ub = []
    b_ub = []
    
    # Pairwise constraints
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dist)
            
    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        # r_i <= x
        row = np.zeros(n); row[i] = 1.0; A_ub.append(row); b_ub.append(x)
        # r_i <= 1 - x
        row = np.zeros(n); row[i] = 1.0; A_ub.append(row); b_ub.append(1 - x)
        # r_i <= y
        row = np.zeros(n); row[i] = 1.0; A_ub.append(row); b_ub.append(y)
        # r_i <= 1 - y
        row = np.zeros(n); row[i] = 1.0; A_ub.append(row); b_ub.append(1 - y)

    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    bounds = [(0, None) for _ in range(n)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
        else:
            return None, -np.inf
    except Exception:
        return None, -np.inf

def objective_and_constraints(params, n_circles):
    """
    Objective function and constraints for the nonlinear optimizer.
    We optimize centers and radii simultaneously to avoid the non-differentiable LP step.
    
    Variables: [x1, y1, r1, x2, y2, r2, ...]
    """
    # Reshape params
    centers = params[0::3].reshape(-1, 2)
    radii = params[2::3] # Actually params are x, y, r. 
    # Let's re-structure: params indices: 0:x, 1:y, 2:r, 3:x...
    # Wait, easier:
    # params shape (3*n,)
    # centers[:, 0] = params[0::3]
    # centers[:, 1] = params[1::3]
    # radii = params[2::3]
    
    centers = np.zeros((n_circles, 2))
    centers[:, 0] = params[0::3]
    centers[:, 1] = params[1::3]
    radii = params[2::3]
    
    # Objective: Minimize negative sum of radii
    obj_val = -np.sum(radii)
    
    # Constraints
    constraints_list = []
    
    # Boundary constraints for centers and radii
    # 0 <= x - r, x + r <= 1 => r <= x, r <= 1-x
    # Similarly for y
    # We handle these as bounds in the optimizer, but let's double check or enforce via penalty if needed.
    # Better to use bounds in minimize.
    
    # Overlap constraints: dist >= r_i + r_j
    # dist^2 >= (r_i + r_j)^2
    # (r_i + r_j)^2 - dist^2 <= 0
    
    for i in range(n_circles):
        for j in range(i + 1, n_circles):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist_sq = dx*dx + dy*dy
            sum_r = radii[i] + radii[j]
            # Constraint: (r_i + r_j)^2 - dist^2 <= 0
            # But this is non-convex and might be sensitive.
            # A safer constraint for optimizer: r_i + r_j <= dist
            # But dist is sqrt.
            # Let's use r_i + r_j - dist <= 0
            val = radii[i] + radii[j] - np.sqrt(max(1e-12, dist_sq))
            constraints_list.append({
                'type': 'ineq', 
                'fun': lambda p, v=val: -v # Wait, 'ineq' means fun >= 0. So we want -val >= 0 => val <= 0
            })
            
    return obj_val, constraints_list

def run_optimization(initial_centers):
    n = 26
    # Initialize radii small
    radii_init = np.ones(n) * 0.01
    
    # Combine into params: x, y, r
    x_init = initial_centers[:, 0]
    y_init = initial_centers[:, 1]
    r_init = radii_init
    
    params_init = np.empty(3 * n)
    params_init[0::3] = x_init
    params_init[1::3] = y_init
    params_init[2::3] = r_init
    
    # Bounds: x, y in [0, 1], r >= 0
    bounds = []
    for _ in range(n):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, None)) # r
    
    # Constraints: r_i + r_j <= distance
    def constr_dist(p):
        c = p[0::3].reshape(-1, 2)
        r = p[2::3]
        vals = []
        for i in range(n):
            for j in range(i+1, n):
                d = np.linalg.norm(c[i] - c[j])
                vals.append(d - (r[i] + r[j]))
        return vals
    
    nonlin_con = {'type': 'ineq', 'fun': constr_dist}
    
    # Also boundary constraints for radii relative to centers
    def constr_bound(p):
        c = p[0::3].reshape(-1, 2)
        r = p[2::3]
        vals = []
        for i in range(n):
            vals.append(c[i, 0] - r[i])       # x - r >= 0
            vals.append(1 - c[i, 0] - r[i])   # 1 - x - r >= 0
            vals.append(c[i, 1] - r[i])       # y - r >= 0
            vals.append(1 - c[i, 1] - r[i])   # 1 - y - r >= 0
        return vals

    nonlin_con_bound = {'type': 'ineq', 'fun': constr_bound}

    constraints = [nonlin_con, nonlin_con_bound]

    try:
        res = minimize(fun=lambda p: -np.sum(p[2::3]), 
                       x0=params_init, 
                       method='SLSQP', 
                       bounds=bounds,
                       constraints=constraints,
                       options={'ftol': 1e-9, 'maxiter': 500})
        
        if res.success:
            best_params = res.x
        else:
            # Fallback to initial
            best_params = params_init
    except Exception:
        best_params = params_init

    # Extract results
    best_centers = best_params[0::3].reshape(-1, 2)
    best_radii = best_params[2::3]
    
    # Clean up: ensure radii are valid for final returned centers using LP
    # The optimization might have slight drift. Let's re-solve LP to be sure.
    final_radii, final_sum = solve_lp_for_radii(best_centers)
    if final_radii is not None:
        best_radii = final_radii
        final_sum_val = final_sum
    else:
        final_sum_val = np.sum(best_radii)
        
    return best_centers, best_radii, final_sum_val

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    configs = generate_initial_configurations()
    
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    for cfg in configs:
        c, r, s = run_optimization(cfg)
        if s > best_sum:
            best_sum = s
            best_centers = c
            best_radii = r
            
    return best_centers, best_radii, best_sum
