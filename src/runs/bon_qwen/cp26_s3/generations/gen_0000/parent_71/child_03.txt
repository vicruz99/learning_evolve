To maximize the sum of radii for 26 circles in a unit square, I need to exploit several geometric advantages:

1. **Corner circles** can be larger since they only need clearance on 2 sides (can approach radius 0.5 in isolation, though constrained by neighbors)
2. **Edge circles** have one free boundary to expand toward
3. **Interior circles** are fully constrained by neighbors

My strategy:
- Start with an initial configuration that places larger circles at corners and edges
- Use scipy's optimization with a penalty method to grow radii while maintaining valid packing
- Run multiple restarts from different initial configurations to escape local minima
- Implement a secondary "grow phase" after optimization to squeeze out remaining space

The penalty-based objective function minimizes negative sum of radii (i.e., maximizes sum) with quadratic penalties for:
- Overlapping circles (penalty proportional to overlap²)
- Circles extending beyond the unit square boundary

I'll use multiple random restarts with L-BFGS-B optimizer and then perform a final greedy growth pass to maximize each radius individually given fixed positions of others.

import numpy as np
from scipy.optimize import minimize


def run_packing():
    n = 26

    best_centers = None
    best_radii = None
    best_sum = 0.0

    def objective(params):
        c = params[:2 * n].reshape(n, 2)
        r = params[2 * n:]
        obj = -np.sum(r)
        penalty = 8000.0

        for i in range(n):
            for j in range(i + 1, n):
                dx = c[i, 0] - c[j, 0]
                dy = c[i, 1] - c[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                overlap = r[i] + r[j] - dist
                if overlap > 0:
                    obj += penalty * overlap * overlap

        for i in range(n):
            for d in range(2):
                if c[i, d] < r[i]:
                    obj += penalty * (r[i] - c[i, d]) * (r[i] - c[i, d])
                if c[i, d] > 1 - r[i]:
                    obj += penalty * (c[i, d] - 1 + r[i]) * (c[i, d] - 1 + r[i])

        return obj

    def generate_hexagonal(n_circles):
        ctrs = np.zeros((n_circles, 2))
        idx = 0
        for row in range(7):
            for col in range(5):
                if idx >= n_circles:
                    break
                x = 0.06 + col * 0.18 + (row % 2) * 0.09
                y = 0.06 + row * 0.155
                x = np.clip(x, 0.05, 0.95)
                y = np.clip(y, 0.05, 0.95)
                ctrs[idx] = [x, y]
                idx += 1
        return ctrs

    def generate_corner_emphasis(n_circles):
        ctrs = np.zeros((n_circles, 2))
        idx = 0

        ctrs[idx] = [0.15, 0.15]
        idx += 1
        ctrs[idx] = [0.85, 0.15]
        idx += 1
        ctrs[idx] = [0.15, 0.85]
        idx += 1
        ctrs[idx] = [0.85, 0.85]
        idx += 1

        edge_positions = [
            [0.35, 0.08], [0.65, 0.08],
            [0.35, 0.92], [0.65, 0.92],
            [0.08, 0.35], [0.08, 0.65],
            [0.92, 0.35], [0.92, 0.65],
        ]
        for ep in edge_positions:
            if idx >= n_circles:
                break
            ctrs[idx] = ep
            idx += 1

        for row in range(2, 6):
            for col in range(2, 6):
                if idx >= n_circles:
                    break
                ctrs[idx] = [0.08 + col * 0.175, 0.08 + row * 0.175]
                idx += 1

        return ctrs

    initial_generators = [generate_hexagonal, generate_corner_emphasis]

    for gen_func in initial_generators:
        for restart in range(8):
            centers0 = gen_func(n)
            centers0 = centers0 + np.random.randn(n, 2) * 0.015
            centers0 = np.clip(centers0, 0.05, 0.95)

            if gen_func == generate_corner_emphasis:
                radii0 = np.zeros(n)
                radii0[:4] = 0.12
                radii0[4:12] = 0.07
                radii0[12:] = 0.05
            else:
                radii0 = np.full(n, 0.05) + np.random.randn(n) * 0.008
                radii0 = np.maximum(radii0, 0.02)

            params0 = np.concatenate([centers0.flatten(), radii0])

            result = minimize(
                objective,
                params0,
                method="L-BFGS-B",
                bounds=[(0.0, 1.0)] * (2 * n) + [(0.0, 0.48)] * n,
                options={"maxiter": 30000, "ftol": 1e-15, "gtol": 1e-10},
            )

            centers_opt = result.x[:2 * n].reshape(n, 2)
            radii_opt = result.x[2 * n:]
            radii_opt = np.maximum(radii_opt, 0.0)

            total = np.sum(radii_opt)
            if total > best_sum:
                best_sum = total
                best_centers = centers_opt.copy()
                best_radii = radii_opt.copy()

    centers, radii = best_centers, best_radii

    greedy_grow(centers, radii, n)

    cleanup(centers, radii, n)

    final_sum = np.sum(radii)
    return centers, radii, final_sum


def greedy_grow(centers, radii, n):
    for iteration in range(500):
        improved = False
        for i in range(n):
            max_r = min(
                centers[i, 0],
                1 - centers[i, 0],
                centers[i, 1],
                1 - centers[i, 1],
            )
            for j in range(n):
                if i == j:
                    continue
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                max_r = min(max_r, dist - radii[j])
            max_r = max(0.0, max_r)
            if max_r > radii[i] + 1e-10:
                radii[i] = max_r
                improved = True
        if not improved:
            break

    for iteration in range(300):
        improved = False
        for i in range(n):
            for trial in range(4):
                delta = 0.001 * (0.5 ** trial)
                new_r = radii[i] + delta
                valid = True
                max_r_bound = min(
                    centers[i, 0],
                    1 - centers[i, 0],
                    centers[i, 1],
                    1 - centers[i, 1],
                )
                if new_r > max_r_bound + 1e-12:
                    valid = False
                if valid:
                    for j in range(n):
                        if i == j:
                            continue
                        dx = centers[i, 0] - centers[j, 0]
                        dy = centers[i, 1] - centers[j, 1]
                        dist = np.sqrt(dx * dx + dy * dy)
                        if dist < new_r + radii[j] - 1e-12:
                            valid = False
                            break
                if valid:
                    radii[i] = new_r
                    improved = True
        if not improved:
            break


def cleanup(centers, radii, n):
    for i in range(n):
        radii[i] = min(
            radii[i],
            centers[i, 0] + 1e-12,
            1 - centers[i, 0] + 1e-12,
            centers[i, 1] + 1e-12,
            1 - centers[i, 1] + 1e-12,
        )
        radii[i] = max(0.0, radii[i])

    for _ in range(100):
        changed = False
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                needed = radii[i] + radii[j] - 1e-12
                if dist < needed:
                    overlap = needed - dist
                    if radii[i] <= radii[j]:
                        radii[i] -= overlap
                    else:
                        radii[j] -= overlap
                    changed = True
        if not changed:
            break

    for i in range(n):
        radii[i] = max(0.0, radii[i])
