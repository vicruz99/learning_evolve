

Looking at this problem, I need to pack 26 circles in a unit square to maximize the sum of radii. Let me reason about the strategy and then implement an optimization approach.

## Strategy

**Key Observations:**
1. For 26 equal circles, a naive 5×5 grid gives r=0.1, sum=2.5. The target 2.636 suggests average r≈0.1014, meaning circles should be slightly larger than 0.1.
2. Hexagonal packing is denser than square packing, allowing slightly larger circles.
3. The objective (maximize sum of radii) benefits from dense packing where circles share space efficiently.

**Approach:**
1. Initialize with a hexagonal pattern (~5-6 rows, ~5 circles each)
2. Use scipy SLSQP with explicit non-convex constraints for boundary and non-overlap
3. Run multiple random restarts to escape local optima
4. Add a post-processing "squeeze" phase: iteratively expand all radii uniformly until first contact, then optimize again

**Hexagonal layout for 26 circles:**
- Row 1: 5 circles, Row 2: 5 circles (shifted), Row 3: 5 circles, Row 4: 5 circles (shifted), Row 5: 5 circles, Row 6: 1 circle
- This uses the √3 vertical spacing of hexagonal packing to fit more circles

```python
import numpy as np
from scipy.optimize import minimize


def hexagonal_init(n, r_init, seed=0):
    """Initialize circles in a hexagonal pattern with optional random perturbation."""
    rng = np.random.RandomState(seed)
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    idx = 0
    for row in range(10):
        y = r_init + row * r_init * np.sqrt(3)
        if y + r_init > 1.0:
            break
        x_offset = 0.0 if row % 2 == 0 else r_init
        for col in range(8):
            if idx >= n:
                break
            x = r_init + col * 2 * r_init + x_offset
            if x + r_init > 1.0:
                continue
            centers[idx, 0] = x + rng.uniform(-0.005, 0.005)
            centers[idx, 1] = y + rng.uniform(-0.005, 0.005)
            radii[idx] = r_init + rng.uniform(-0.003, 0.003)
            idx += 1
        if idx >= n:
            break

    # Clip to valid range
    for i in range(n):
        radii[i] = max(0.005, min(0.4, radii[i]))
        centers[i, 0] = max(radii[i], min(1.0 - radii[i], centers[i, 0]))
        centers[i, 1] = max(radii[i], min(1.0 - radii[i], centers[i, 1]))

    return centers, radii


def check_validity(centers, radii, n, tol=1e-10):
    """Check if a packing is valid."""
    for i in range(n):
        xi, yi, ri = centers[i, 0], centers[i, 1], radii[i]
        if ri < -tol:
            return False
        if xi < ri - tol or xi > 1.0 - ri + tol:
            return False
        if yi < ri - tol or yi > 1.0 - ri + tol:
            return False
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            if dist < radii[i] + radii[j] - tol:
                return False
    return True


def try_optimize(n, centers_init, radii_init):
    """Run SLSQP optimization from given initial point."""
    x0 = np.concatenate([centers_init.flatten(), radii_init])

    def objective(x):
        return -np.sum(x[2 * n:])

    bounds = [(0.0, 1.0)] * (2 * n) + [(1e-10, 0.45)] * n

    cons = []

    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    for i in range(n):
        cons.append({'type': 'ineq', 'fun': lambda x, i=i: x[2 * i] - x[2 * n + i]})
        cons.append({'type': 'ineq', 'fun': lambda x, i=i: 1.0 - x[2 * i] - x[2 * n + i]})
        cons.append({'type': 'ineq', 'fun': lambda x, i=i: x[2 * i + 1] - x[2 * n + i]})
        cons.append({'type': 'ineq', 'fun': lambda x, i=i: 1.0 - x[2 * i + 1] - x[2 * n + i]})

    # Separation constraints
    for i in range(n):
        for j in range(i + 1, n):
            cons.append({'type': 'ineq', 'fun': lambda x, i=i, j=j:
                         np.sqrt((x[2 * i] - x[2 * j]) ** 2 + (x[2 * i + 1] - x[2 * j + 1]) ** 2)
                         - x[2 * n + i] - x[2 * n + j] - 1e-12})

    result = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': 3000, 'ftol': 1e-16})

    c = result.x[:2 * n].reshape(n, 2)
    r = result.x[2 * n:]
    return c, r


def squeeze_phase(n, centers, radii):
    """Uniformly expand all radii until first contact is reached."""
    # Find minimum gap
    min_gap = 1.0
    for i in range(n):
        xi, yi, ri = centers[i, 0], centers[i, 1], radii[i]
        min_gap = min(min_gap, xi - ri, 1.0 - xi - ri, yi - ri, 1.0 - yi - ri)
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            min_gap = min(min_gap, dist - radii[i] - radii[j])

    if min_gap > 1e-8:
        # Expand all radii uniformly
        radii_new = radii + min_gap / 2.0
        # Re-optimize positions
        c, r = try_optimize(n, centers, radii_new)
        return c, r
    return centers, radii


def run_packing():
    n = 26

    best_sum = -1.0
    best_centers = None
    best_radii = None

    # Try multiple initializations
    for seed in range(40):
        for r_init in [0.085, 0.09, 0.093, 0.095, 0.097, 0.098, 0.10]:
            centers_init, radii_init = hexagonal_init(n, r_init, seed=seed)

            # Optimize
            centers_opt, radii_opt = try_optimize(n, centers_init, radii_init)

            # Squeeze phase
            centers_opt, radii_opt = squeeze_phase(n, centers_opt, radii_opt)

            # Second squeeze + optimize
            centers_opt, radii_opt = squeeze_phase(n, centers_opt, radii_opt)
            centers_opt, radii_opt = try_optimize(n, centers_opt, radii_opt)

            current_sum = np.sum(radii_opt)

            if current_sum > best_sum:
                # Validate
                if check_validity(centers_opt, radii_opt, n, tol=1e-9):
                    best_sum = current_sum
                    best_centers = centers_opt.copy()
                    best_radii = radii_opt.copy()

    if best_centers is None:
        # Fallback: equal circles
        centers_fb = np.zeros((n, 2))
        radii_fb = np.zeros(n)
        r_fb = 0.09
        idx = 0
        for row in range(5):
            for col in range(5):
                if idx < n:
                    centers_fb[idx] = [0.1 + col * 0.18, 0.1 + row * 0.18]
                    radii_fb[idx] = r_fb
                    idx += 1
        return centers_fb, radii_fb, np.sum(radii_fb)

    return best_centers, best_radii, best_sum
```
