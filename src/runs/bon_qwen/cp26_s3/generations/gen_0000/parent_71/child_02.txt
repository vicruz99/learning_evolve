import numpy as np
from scipy.optimize import minimize


def obj_func(x, n):
    return -np.sum(x[2 * n:])


def con_xmin(x, idx, n):
    return x[2 * idx] - x[2 * n + idx]


def con_xmax(x, idx, n):
    return 1.0 - x[2 * idx] - x[2 * n + idx]


def con_ymin(x, idx, n):
    return x[2 * idx + 1] - x[2 * n + idx]


def con_ymax(x, idx, n):
    return 1.0 - x[2 * idx + 1] - x[2 * n + idx]


def con_nonoverlap(x, i, j, n):
    dx = x[2 * i] - x[2 * j]
    dy = x[2 * i + 1] - x[2 * j + 1]
    ri = x[2 * n + i]
    rj = x[2 * n + j]
    return dx * dx + dy * dy - (ri + rj) * (ri + rj)


def build_constraints(n):
    cons = []
    for i in range(n):
        cons.append({'type': 'ineq', 'fun': con_xmin, 'args': (i, n)})
        cons.append({'type': 'ineq', 'fun': con_xmax, 'args': (i, n)})
        cons.append({'type': 'ineq', 'fun': con_ymin, 'args': (i, n)})
        cons.append({'type': 'ineq', 'fun': con_ymax, 'args': (i, n)})
    for i in range(n):
        for j in range(i + 1, n):
            cons.append({'type': 'ineq', 'fun': con_nonoverlap, 'args': (i, j, n)})
    return cons


def build_bounds(n):
    return [(0.0, 1.0)] * (2 * n) + [(0.0, 1.0)] * n


def make_hex_init(n, row_sizes, r):
    centers = np.zeros((n, 2))
    radii = np.full(n, r)
    idx = 0
    h = np.sqrt(3) * r
    for ri, sz in enumerate(row_sizes):
        y = r + ri * h
        if ri % 2 == 0:
            x0 = r
        else:
            x0 = 2 * r
        for ci in range(sz):
            centers[idx, 0] = x0 + ci * 2 * r
            centers[idx, 1] = y
            idx += 1
    return centers, radii


def make_grid_init(n, nc, nr, r):
    centers = np.zeros((n, 2))
    radii = np.full(n, r)
    idx = 0
    dx = 1.0 / (nc + 1)
    dy = 1.0 / (nr + 1)
    for i in range(nr):
        for j in range(nc):
            if idx < n:
                centers[idx, 0] = dx * (j + 1)
                centers[idx, 1] = dy * (i + 1)
                idx += 1
    while idx < n:
        centers[idx, 0] = 0.5
        centers[idx, 1] = 0.5
        idx += 1
    return centers, radii


def repair_packing(centers, radii, n):

    centers = centers.copy()
    for iteration in range(50):
        changed = False
        for i in range(n):
            r_bound = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
            radii[i] = min(radii[i], r_bound)

            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                required = radii[i] + radii[j]
                if dist < required - 1e-12:
                    scale = dist / (required + 1e-12)

                    radii[j] *= scale * 0.999
                    changed = True
        if not changed:
            break
    radii = np.maximum(radii, 1e-15)
    centers = np.clip(centers, 0, 1)
    return centers, radii
