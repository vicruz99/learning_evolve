import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    # Initial configuration: Hexagonal lattice points
    centers = np.zeros((n, 2))
    s = 0.2  # Initial spacing for lattice generation
    points = []
    j = 0
    # Generate points in a hexagonal pattern
    while len(points) < n:
        x_start = (s / 2) if j % 2 == 1 else 0
        k = 0
        while True:
            x = x_start + k * s
            y = j * s * np.sqrt(3) / 2
            # Keep points well within the square to allow margin for expansion
            if 0.05 <= x <= 0.95 and 0.05 <= y <= 0.95:
                points.append((x, y))
            if len(points) >= n:
                break
            k += 1
        j += 1
    
    # Select exactly n points
    centers = np.array(points[:n])
    
    # Initial small radius to ensure valid start
    radii = np.full(n, 0.01)
    
    # Flatten variables for optimizer: [x1, y1, r1, x2, y2, r2, ...]
    x0 = np.zeros(3 * n)
    for i in range(n):
        x0[3 * i] = centers[i, 0]
        x0[3 * i + 1] = centers[i, 1]
        x0[3 * i + 2] = radii[i]
    
    # Bounds for variables
    # x, y in [0, 1], r in [0, 0.5]
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
    
    # Vectorized constraint function
    def constraints(x_vars):
        vars_arr = x_vars.reshape((n, 3))
        X = vars_arr[:, 0]
        Y = vars_arr[:, 1]
        R = vars_arr[:, 2]
        
        c_list = []
        
        # Boundary constraints: x >= r, x <= 1-r, y >= r, y <= 1-r
        c_list.append(X - R)              # x - r >= 0
        c_list.append(1.0 - X - R)        # 1 - x - r >= 0
        c_list.append(Y - R)              # y - r >= 0
        c_list.append(1.0 - Y - R)        # 1 - y - r >= 0
        
        # Pairwise non-overlap constraints
        # We only need the upper triangle of the pairwise distance matrix
        dx = X[:, None] - X[None, :]
        dy = Y[:, None] - Y[None, :]
        dr = R[:, None] + R[None, :]
        
        dist_sq = dx**2 + dy**2
        rad_sum_sq = dr**2
        
        # Mask for upper triangle
        mask = np.triu(np.ones((n, n), dtype=bool), k=1)
        diffs = dist_sq[mask] - rad_sum_sq[mask]
        c_list.append(diffs)
        
        return np.concatenate(c_list)
    
    # Objective: minimize negative sum of radii
    def objective(x_vars):
        return -np.sum(x_vars[2::3])
    
    # Run optimization
    # maxiter set high to allow convergence, ftol tight
    res = minimize(objective, x0, method='SLSQP', bounds=bounds, 
                   constraints={'type': 'ineq', 'fun': constraints},
                   options={'maxiter': 1000, 'ftol': 1e-12, 'disp': False})
    
    # Extract results
    final_vars = res.x.reshape((n, 3))
    final_centers = final_vars[:, :2]
    final_radii = final_vars[:, 2]
    sum_radii = np.sum(final_radii)
    
    return final_centers, final_radii, sum_radii
