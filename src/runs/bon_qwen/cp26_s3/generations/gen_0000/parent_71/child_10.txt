2. **Force-Based Simulation**: Use repulsive forces between overlapping circles and boundary repulsion to spread circles apart while gradually increasing target radii. This helps escape poor local configurations.

3. **SLSQP Optimization**: Use scipy's Sequential Least Squares Programming to simultaneously optimize positions and radii. The constraint function returns all boundary and non-overlap constraints as a single array.

4. **Post-processing**: Clamp values to ensure validity and handle numerical edge cases.

Key design decisions:
- All helper functions are top-level with no closures
- No lambda functions used

import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26


def hexagonal_init(n, r):
    """Initialize circles in a hexagonal lattice pattern."""
    centers = np.zeros((n, 2))
    pattern = [6, 5, 6, 5, 4]
    k = 0
    for row, ncols in enumerate(pattern):
        y = r + row * r * np.sqrt(3)
        offset = (row % 2) * r
        for col in range(ncols):
            x = r + offset + col * 2 * r
            if x <= 1 - r:
                centers[k] = [x, y]
                k += 1
    return centers[:n]


def force_simulate(centers, radii, n, target_r, steps=3000):
    """Force-based simulation to spread circles and grow radii."""
    for step in range(steps):
        forces = np.zeros((n, 2))
        # Repulsive forces between circles
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                d2 = dx * dx + dy * dy
                d = np.sqrt(d2) if d2 > 1e-20 else 1e-10
                req = radii[i] + radii[j]
                if d < req:
                    f = (req - d) / d
                    forces[i, 0] += f * dx
                    forces[i, 1] += f * dy
                    forces[j, 0] -= f * dx
                    forces[j, 1] -= f * dy
        # Boundary repulsion
        for i in range(n):
            r = radii[i]
            if centers[i, 0] < r:
                forces[i, 0] += 10 * (r - centers[i, 0])
            elif centers[i, 0] > 1 - r:
                forces[i, 0] -= 10 * (centers[i, 0] - 1 + r)
            if centers[i, 1] < r:
                forces[i, 1] += 10 * (r - centers[i, 1])
            elif centers[i, 1] > 1 - r:
                forces[i, 1] -= 10 * (centers[i, 1] - 1 + r)
        # Apply forces
        step_size = 0.1 / (1 + step * 0.002)
        centers += step_size * forces
        # Clip to valid region
        for i in range(n):
            r = radii[i]
            centers[i, 0] = max(r, min(1 - r, centers[i, 0]))
            centers[i, 1] = max(r, min(1 - r, centers[i, 1]))
        # Gradually increase radii
        current_max = np.max(radii)
        target = min(target_r, current_max + (target_r - current_max) / steps * 0.02)
        radii[:] = target
    return centers, radii


def objective(x):
    """Negative sum of radii (to maximize)."""
    return -np.sum(x[2::3])


def constraints(x):
    """All constraints as an array (must be >= 0)."""
    n = N_CIRCLES
    c = []
    # Boundary constraints for each circle
    for i in range(n):
        # x - r >= 0
        c.append(x[3 * i] - x[3 * i + 2])
        # 1 - x - r >= 0
        c.append(1 - x[3 * i] - x[3 * i + 2])
        # y - r >= 0
        c.append(x[3 * i + 1] - x[3 * i + 2])
        # 1 - y - r >= 0
        c.append(1 - x[3 * i + 1] - x[3 * i + 2])
        # r >= 0
        c.append(x[3 * i + 2])
    # Non-overlap constraints for each pair
    for i in range(n):
        for j in range(i + 1, n):
            dx = x[3 * i] - x[3 * j]
            dy = x[3 * i + 1] - x[3 * j + 1]
            dist = np.sqrt(dx * dx + dy * dy)
            c.append(dist - x[3 * i + 2] - x[3 * j + 2])
    return np.array(c)


def run_packing():
    """Pack 26 circles in unit square to maximize sum of radii."""
    n = N_CIRCLES
    
    # Phase 1: Hexagonal initialization
    r_init = 0.1
    centers = hexagonal_init(n, r_init)
    radii = np.ones(n) * r_init
    
    # Phase 2: Force-based simulation
    centers, radii = force_simulate(centers, radii, n, 0.115, steps=3000)
    
    # Phase 3: SLSQP optimization
    x0 = np.zeros(3 * n)
    for i in range(n):
        x0[3 * i] = centers[i, 0]
        x0[3 * i + 1] = centers[i, 1]
        x0[3 * i + 2] = radii[i]
    
    bounds = [(0, 1)] * (2 * n) + [(1e-10, 0.5)] * n
    constraint_dict = {'type': 'ineq', 'fun': constraints}
    
    result = minimize(
        objective,
        x0,
        method='SLSQP',
        bounds=bounds,
        constraints=constraint_dict,
        options={'maxiter': 5000, 'ftol': 1e-14, 'disp': False}
    )
    
    centers = result.x[:2 * n].reshape(n, 2)
    radii = result.x[2 * n:]
    
    # Post-processing: ensure validity
    for i in range(n):
        r = radii[i]
        centers[i, 0] = max(r + 1e-12, min(1 - r - 1e-12, centers[i, 0]))
        centers[i, 1] = max(r + 1e-12, min(1 - r - 1e-12, centers[i, 1]))
        radii[i] = max(1e-10, radii[i])
    
    return centers, radii, np.sum(radii)
