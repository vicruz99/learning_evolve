import numpy as np
from scipy.optimize import minimize

def calculate_penalty(vars, n, mu):
    """Compute the penalty for constraint violations."""
    centers = vars[:2*n].reshape(n, 2)
    radii = vars[2*n:]
    penalty = 0.0
    
    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if r > x: 
            penalty += mu * (r - x)**2
        if x + r > 1: 
            penalty += mu * (x + r - 1)**2
        if r > y: 
            penalty += mu * (r - y)**2
        if y + r > 1: 
            penalty += mu * (y + r - 1)**2
            
    # Pairwise non-overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx*dx + dy*dy)
            req = radii[i] + radii[j]
            if req > dist:
                penalty += mu * (req - dist)**2
    return penalty

def objective_function(vars, n, mu):
    """Objective: minimize negative sum of radii plus penalties."""
    radii = vars[2*n:]
    return -np.sum(radii) + calculate_penalty(vars, n, mu)

def run_packing():
    """
    Pack 26 circles in a unit square to maximize sum of radii.
    Returns (centers, radii, sum_radii).
    """
    n = 26
    mu = 10000.0
    
    # 1. Initialization: Hexagonal-like grid placement
    centers = np.zeros((n, 2))
    idx = 0
    # Row configuration summing to 26
    rows = [6, 5, 6, 5, 4] 
    y_pos = 0.12
    y_step = 0.18
    
    for r_idx, count in enumerate(rows):
        x_pos = 0.12
        # Calculate step to fit `count` circles in [0.12, 0.88]
        span = 0.76
        x_step = span / max(count - 1, 1)
        
        # Stagger odd rows for hexagonal packing
        if r_idx % 2 == 1:
            x_pos += x_step / 2.0
            
        for c in range(count):
            if idx < n:
                centers[idx] = [x_pos, y_pos]
                idx += 1
            x_pos += x_step
        y_pos += y_step
        
    radii_init = np.full(n, 0.02)
    vars0 = np.concatenate([centers.flatten(), radii_init])
    
    # 2. Setup bounds for optimization
    bounds = []
    for _ in range(n):
        bounds.append((1e-4, 1 - 1e-4))  # x
        bounds.append((1e-4, 1 - 1e-4))  # y
        bounds.append((1e-6, 0.5))        # r
        
    # 3. Optimization
    result = minimize(
        objective_function, 
        vars0, 
        args=(n, mu), 
        method='L-BFGS-B', 
        bounds=bounds, 
        options={'maxiter': 3000, 'ftol': 1e-12, 'gtol': 1e-8}
    )
    
    opt_vars = result.x
    centers_opt = opt_vars[:2*n].reshape(n, 2)
    
    # 4. Post-processing: Project to strictly valid radii
    # For each circle, radius is limited by boundaries and nearest neighbor
    final_radii = np.zeros(n)
    for i in range(n):
        # Boundary limits
        r = min(
            centers_opt[i, 0], 
            1.0 - centers_opt[i, 0], 
            centers_opt[i, 1], 
            1.0 - centers_opt[i, 1]
        )
        # Neighbor limits
        for j in range(n):
            if i == j: 
                continue
            dist = np.sqrt(np.sum((centers_opt[i] - centers_opt[j])**2))
            # To satisfy dist >= r_i + r_j, we conservatively set r_i <= dist/2
            # This guarantees validity for all pairs simultaneously
            r = min(r, dist / 2.0)
        final_radii[i] = r
        
    sum_radii = np.sum(final_radii)
    return centers_opt, final_radii, sum_radii
