import numpy as np
from scipy.optimize import minimize

def run_packing():
    n = 26
    
    def objective(x):
        # Minimize negative sum of radii = maximize sum of radii
        return -np.sum(x[2*n:])
    
    def constraint_vector(x):
        c = x[:2*n].reshape(n, 2)
        r = x[2*n:]
        
        res = []
        # Boundary constraints: each circle must fit in [0,1]x[0,1]
        for i in range(n):
            res.append(c[i, 0] - r[i])       # x >= r (left boundary)
            res.append(1 - c[i, 0] - r[i])   # 1-x >= r (right boundary)
            res.append(c[i, 1] - r[i])       # y >= r (bottom boundary)
            res.append(1 - c[i, 1] - r[i])   # 1-y >= r (top boundary)
        
        # Non-overlap constraints: distance between centers >= sum of radii
        for i in range(n):
            for j in range(i + 1, n):
                dx = c[i, 0] - c[j, 0]
                dy = c[i, 1] - c[j, 1]
                d = np.sqrt(dx * dx + dy * dy)
                res.append(d - r[i] - r[j])
        
        return np.array(res)
    
    def constraint_jacobian(x):
        # Return sparse Jacobian for efficiency
        n_constraints = 4 * n + n * (n - 1) // 2
        jacobian = np.zeros((n_constraints, 3 * n))
        
        c = x[:2*n].reshape(n, 2)
        r = x[2*n:]
        
        idx = 0
        # Boundary constraint Jacobians
        for i in range(n):
            # x_i - r_i >= 0
            jacobian[idx, 2*i] = 1
            jacobian[idx, 2*n + i] = -1
            idx += 1
            # 1 - x_i - r_i >= 0
            jacobian[idx, 2*i] = -1
            jacobian[idx, 2*n + i] = -1
            idx += 1
            # y_i - r_i >= 0
            jacobian[idx, 2*i + 1] = 1
            jacobian[idx, 2*n + i] = -1
            idx += 1
            # 1 - y_i - r_i >= 0
            jacobian[idx, 2*i + 1] = -1
            jacobian[idx, 2*n + i] = -1
            idx += 1
        
        # Non-overlap constraint Jacobians
        for i in range(n):
            for j in range(i + 1, n):
                dx = c[i, 0] - c[j, 0]
                dy = c[i, 1] - c[j, 1]
                d = np.sqrt(dx * dx + dy * dy)
                if d > 1e-10:
                    jacobian[idx, 2*i] = dx / d
                    jacobian[idx, 2*i + 1] = dy / d
                    jacobian[idx, 2*j] = -dx / d
                    jacobian[idx, 2*j + 1] = -dy / d
                    jacobian[idx, 2*n + i] = -1
                    jacobian[idx, 2*n + j] = -1
                idx += 1
        
        return jacobian
    
    best_x = None
    best_sum = 0.0
    
    # Try multiple initial configurations with hexagonal lattice pattern
    for seed in range(60):
        np.random.seed(seed)
        
        # Hexagonal lattice initialization - known to be dense for circle packing
        centers = np.zeros((n, 2))
        radii = np.ones(n) * 0.09
        
        # Arrange in rows with alternating offset (hexagonal packing)
        y_pos = 0.14
        idx = 0
        for row in range(6):
            x_start = 0.12
            if row % 2 == 1:
                x_start = 0.23  # Offset for hexagonal pattern
            for col in range(5):
                if idx < n:
                    # Add small perturbation to break symmetry
                    centers[idx, 0] = x_start + col * 0.22 + np.random.normal(0, 0.015)
                    centers[idx, 1] = y_pos + np.random.normal(0, 0.015)
                    idx += 1
            y_pos += 0.18 * np.sqrt(3) / 2
        
        # Ensure initial positions are valid
        centers = np.clip(centers, 0.05, 0.95)
        
        x0 = np.concatenate([centers.flatten(), radii])
        
        # Bounds for variables
        bounds = [(0.02, 0.98)] * (2 * n) + [(0.02, 0.45)] * n
        
        try:
            result = minimize(
                objective,
                x0,
                method='SLSQP',
                jac=None,
                constraints={
                    'type': 'ineq',
                    'fun': constraint_vector,
                    'jac': constraint_jacobian
                },
                bounds=bounds,
                options={
                    'maxiter': 4000,
                    'ftol': 1e-14,
                    'disp': False
                }
            )
            
            current_sum = -result.fun
            if current_sum > best_sum:
                best_sum = current_sum
                best_x = result.x.copy()
        except Exception:
            continue
    
    # If optimization failed, use fallback
    if best_x is None:
        centers = np.random.uniform(0.15, 0.85, (n, 2))
        radii = np.ones(n) * 0.06
        return centers, radii, np.sum(radii)
    
    final_centers = best_x[:2*n].reshape(n, 2)
    final_radii = best_x[2*n:]
    
    # Post-processing: fix any numerical violations
    for _ in range(2000):
        valid = True
        
        # Check overlaps
        for i in range(n):
            for j in range(i + 1, n):
                dx = final_centers[i, 0] - final_centers[j, 0]
                dy = final_centers[i, 1] - final_centers[j, 1]
                d = np.sqrt(dx * dx + dy * dy)
                required = final_radii[i] + final_radii[j]
                if d < required - 1e-12:
                    valid = False
                    # Reduce radii proportionally to fix overlap
                    if required > 1e-15:
                        scale = (d + 1e-12) / required
                        final_radii[i] *= np.sqrt(scale)
                        final_radii[j] *= np.sqrt(scale)
        
        # Check boundary violations
        for i in range(n):
            x, y = final_centers[i]
            r = final_radii[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                valid = False
                max_r = min(x, 1 - x, y, 1 - y)
                if max_r > 0:
                    final_radii[i] = max_r * 0.9999
        
        if valid:
            break
    
    # Final safety scaling if still invalid
    scale = 0.99999
    for _ in range(100):
        valid = True
        for i in range(n):
            x, y = final_centers[i]
            r = final_radii[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                valid = False
                break
        
        if valid:
            for i in range(n):
                for j in range(i + 1, n):
                    dx = final_centers[i, 0] - final_centers[j, 0]
                    dy = final_centers[i, 1] - final_centers[j, 1]
                    d = np.sqrt(dx * dx + dy * dy)
                    if d < final_radii[i] + final_radii[j] - 1e-12:
                        valid = False
                        break
                if not valid:
                    break
        
        if valid:
            break
        final_radii *= scale
        scale *= 0.999
    
    return final_centers, final_radii, np.sum(final_radii)
