To maximize the sum of radii for 26 circles in a unit square, I will employ a hybrid optimization strategy:

2.  **Linear Programming (LP) Solver**: For any fixed configuration of centers, the problem of finding the maximum radii becomes a Linear Program. This is solved efficiently for each evaluation.
3.  **Global Optimization**: Use `scipy.optimize.differential_evolution` to search the 52-dimensional space of circle centers, using the LP-derived sum of radii as the objective function.

import numpy as np
from scipy.optimize import linprog, differential_evolution, minimize

def get_max_radii_sum(centers):
    """
    Solves the LP to maximize sum of radii given fixed centers.
    Maximize sum(r_i)
    Subject to:
        r_i + r_j <= dist(i, j) for all i < j
        r_i <= x_i, 1 - x_i, y_i, 1 - y_i (boundary constraints)
        r_i >= 0
    """
    n = centers.shape[0]
    
    # Coefficients for objective function (maximize sum of radii -> minimize negative sum)
    c_obj = -np.ones(n)
    
    # Inequality constraints matrix A_ub * r <= b_ub
    # We use lists to build the matrix and RHS vector for memory efficiency
    A_rows = []
    b_rows = []
    
    # Pairwise distance constraints: r_i + r_j <= dist
    # There are n*(n-1)/2 such constraints.
    # We can iterate over all pairs.
    for i in range(n):
        xi, yi = centers[i]
        for j in range(i + 1, n):
            xj, yj = centers[j]
            dist = np.sqrt((xi - xj)**2 + (yi - yj)**2)
            
            # Create a row vector for this constraint
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_rows.append(row)
            b_rows.append(dist)
            
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        xi, yi = centers[i]
        bounds_for_r = [xi, 1 - xi, yi, 1 - yi]
        for val in bounds_for_r:
            row = np.zeros(n)
            row[i] = 1.0
            A_rows.append(row)
            b_rows.append(val)
            
    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)
    
    # Bounds for variables r_i: [0, None]
    bounds = [(0, None) for _ in range(n)]
    
    # Solve LP
    # method='highs' is recommended if available, falls back to default
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    except ValueError:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='interior-point')
        
    if res.success:
        return -res.fun, -res.x
    else:
        # If LP fails, return 0 and zeros
        return 0.0, np.zeros(n)

def objective(centers_flat):
    """Objective function for the optimizer."""
    n = 26
    centers = centers_flat.reshape(n, 2)
    sum_r, radii = get_max_radii_sum(centers)
    # We want to maximize sum_r, so we return negative for minimization
    return -sum_r

def generate_hexagonal_initial_guess(n=26):
    """Generates an initial hexagonal packing configuration."""
    centers = []
    # Attempt to fit points in a hex grid
    # Approximate spacing based on area
    # Area per point ~ 1/n. Hex area factor ~ sqrt(3)/2.
    # Side of hexagon cell ~ sqrt(1/n).
    
    # Try a few grid sizes to find one that fits ~26 points
    for row_count in range(3, 10):
        for start_cols in range(3, 10):
            temp_centers = []
            spacing_x = 1.0 / (start_cols + 0.5) 
            spacing_y = spacing_x * np.sqrt(3) / 2
            
            current_y = 0.5
            for r in range(row_count):
                # Offset every other row
                offset = spacing_x / 2 if r % 2 == 1 else 0
                cols = start_cols - (1 if r % 2 == 1 else 0)
                current_x = offset
                
                for c in range(cols):
                    if len(temp_centers) >= n:
                        break
                    temp_centers.append([current_x, current_y])
                    current_x += spacing_x
                current_y += spacing_y
                if len(temp_centers) >= n:
                    break
        
        if len(temp_centers) >= n:
            # Take first n and normalize/center if needed, though they are likely within bounds
            # Simple scaling to fit in [0,1]
            if len(temp_centers) == n:
                return np.array(temp_centers)
    
    # Fallback to random if grid generation logic is too strict
    return np.random.rand(n, 2)

def run_packing():
    n = 26
    bounds = [(0, 1)] * (n * 2)
    
    # 1. Generate initial guess
    initial_guess = generate_hexagonal_initial_guess(n)
    initial_guess_flat = initial_guess.flatten()
    
    # 2. Global Optimization using Differential Evolution
    # We use a moderate population size to balance speed and coverage
    # vectorized=False is safer for custom objectives, but slower.
    # Here we use simple iteration.
    
    print("Starting Differential Evolution...")
    result_de = differential_evolution(
        objective, 
        bounds, 
        seed=42, 
        popsize=20, 
        maxiter=500, 
        tol=1e-7, 
        mutation=(0.5, 1.5), 
        recombination=0.7,
        polishing=False # Do not polish here, we will do local search manually
    )
    
    best_centers = result_de.x.reshape(n, 2)
    best_sum, best_radii = get_max_radii_sum(best_centers)
    print(f"DE Best Sum: {best_sum}")

    # 3. Local Refinement using Nelder-Mead
    # Nelder-Mead is good for non-smooth objectives (like LP value)
    print("Starting Local Refinement...")
    result_local = minimize(
        objective, 
        best_centers.flatten(), 
        method='Nelder-Mead', 
        options={'maxiter': 2000, 'xatol': 1e-6, 'fatol': 1e-6}
    )
    
    final_centers = result_local.x.reshape(n, 2)
    final_sum, final_radii = get_max_radii_sum(final_centers)
    
    print(f"Final Sum: {final_sum}")
    
    return final_centers, final_radii, final_sum

# Example usage check
if __name__ == "__main__":
    centers, radii, total = run_packing()
    print(f"Total Sum of Radii: {total}")
    print(f"Number of circles: {len(radii)}")
