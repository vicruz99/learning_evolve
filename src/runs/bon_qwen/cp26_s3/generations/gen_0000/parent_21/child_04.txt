To maximize the sum of radii for 26 circles in a unit square, we formulate the problem as a constrained non-linear optimization problem. The decision variables are the centers $(x_i, y_i)$ and radii $r_i$ for each circle.

**Formulation:**
- **Objective:** Maximize $\sum r_i$ (equivalently, minimize $-\sum r_i$)
- **Constraints:**
  1. Boundary: $r_i \le x_i \le 1-r_i$ and $r_i \le y_i \le 1-r_i$
  2. Non-overlap: $(x_i-x_j)^2 + (y_i-y_j)^2 \ge (r_i+r_j)^2$ for all $i < j$
  3. Non-negativity: $r_i \ge 0$

**Algorithm:**
1. Use `scipy.optimize.minimize` with the SLSQP method, which handles non-linear constraints efficiently.
2. Initialize with multiple starting configurations:
   - A hexagonal lattice (optimal for dense packing)
   - Several random valid configurations to explore different basins of attraction
3. For each start, run SLSQP. Track the best feasible solution.
4. Scale pairwise constraints to match boundary constraint magnitudes for better numerical stability.
5. Return the centers, radii, and sum of radii from the best run.

This approach leverages robust local optimization with multi-start initialization to reliably find high-density packings.

import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def objective(x):
    """Minimize negative sum of radii."""
    return -np.sum(x[2 * N_CIRCLES:])

def constraints(x):
    """Compute constraint violations: boundaries and pairwise distances."""
    n = N_CIRCLES
    centers = x[:2 * n].reshape(n, 2)
    radii = x[2 * n:]
    
    n_con = 4 * n + n * (n - 1) // 2
    con_vals = np.zeros(n_con)
    idx = 0
    
    # Boundary constraints
    for i in range(n):
        con_vals[idx] = centers[i, 0] - radii[i]
        con_vals[idx + 1] = 1.0 - centers[i, 0] - radii[i]
        con_vals[idx + 2] = centers[i, 1] - radii[i]
        con_vals[idx + 3] = 1.0 - centers[i, 1] - radii[i]
        idx += 4
        
    # Pairwise non-overlap constraints
    # Scaled by 50.0 to balance magnitudes with boundary constraints
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist_sq = dx * dx + dy * dy
            con_vals[idx] = 50.0 * (dist_sq - (radii[i] + radii[j]) ** 2)
            idx += 1
            
    return con_vals

def get_hex_init():
    """Generate initial centers on a hexagonal grid."""
    pts = []
    spacing = 0.22
    y = spacing
    row = 0
    while len(pts) < N_CIRCLES:
        x = spacing
        if row % 2 == 1:
            x = spacing * 1.5
        while x < 1.0 and len(pts) < N_CIRCLES:
            pts.append([x, y])
            x += spacing * 2
        y += spacing * np.sqrt(3) / 2
        row += 1
    return np.array(pts).flatten()

def get_random_init():
    """Generate random initial centers with minimum separation."""
    pts = []
    while len(pts) < N_CIRCLES:
        p = np.random.rand(2) * 0.6 + 0.2
        valid = True
        for q in pts:
            if np.linalg.norm(np.array(p) - q) < 0.15:
                valid = False
                break
        if valid:
            pts.append(p)
    return np.array(pts).flatten()

def run_packing():
    n = N_CIRCLES
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
    
    constraint = {'type': 'ineq', 'fun': constraints}
    
    best_sum = -np.inf
    best_x = None
    
    # Multiple starting points to escape local minima
    starts = [get_hex_init()]
    for _ in range(8):
        starts.append(get_random_init())
        
    for x0_pos in starts:
        r_init = []
        centers0 = x0_pos.reshape(n, 2)
        for i in range(n):
            r = min(centers0[i, 0], 1 - centers0[i, 0], centers0[i, 1], 1 - centers0[i, 1])
            r_init.append(max(r * 0.3, 0.01))
            
        x0 = np.concatenate([x0_pos, np.array(r_init)])
        
        # Strictly enforce bounds on initial guess
        for i in range(2 * n):
            x0[i] = max(bounds[i][0], min(bounds[i][1], x0[i]))
        for i in range(n):
            x0[2 * n + i] = max(bounds[2 * n + i][0], min(bounds[2 * n + i][1], x0[2 * n + i]))
            
        res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                       constraints=constraint, options={'maxiter': 2000, 'ftol': 1e-10, 'disp': False})
        
        if res.success or res.nit > 150:
            con_vals = constraints(res.x)
            # Accept if nearly feasible (allowing tiny numerical slack)
            if np.min(con_vals) >= -1e-5:
                s = -res.fun
                if s > best_sum:
                    best_sum = s
                    best_x = res.x.copy()
                    
    # Fallback if optimization fails completely
    if best_x is None:
        best_x = np.zeros(3 * n)
        best_x[:2 * n] = 0.5
        best_x[2 * n:] = 0.05
        
    centers = best_x[:2 * n].reshape(n, 2)
    radii = best_x[2 * n:]
    radii = np.maximum(radii, 0.0)
    
    return centers, radii, np.sum(radii)
