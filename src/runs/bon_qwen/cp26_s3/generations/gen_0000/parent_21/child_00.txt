import numpy as np
from scipy.optimize import minimize
import cvxpy as cp


def position_penalty(x, radii, n):
    """Compute constraint violation for given positions and radii."""
    centers = x.reshape((n, 2))
    violation = 0.0
    
    for i in range(n):
        xi, yi = centers[i]
        ri = radii[i]
        
        if xi < ri + 1e-8:
            violation += (xi - ri) ** 2
        if xi > 1 - ri - 1e-8:
            violation += (xi - (1 - ri)) ** 2
        if yi < ri + 1e-8:
            violation += (yi - ri) ** 2
        if yi > 1 - ri - 1e-8:
            violation += (yi - (1 - ri)) ** 2
    
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            min_dist = radii[i] + radii[j]
            if dist < min_dist + 1e-8:
                violation += (min_dist - dist) ** 2
    
    return violation


def optimize_positions_given_radii(centers, radii, n):
    """Optimize positions for given radii using penalty method."""
    x0 = centers.flatten()
    bounds = [(0.001, 0.999)] * (2 * n)
    
    result = minimize(position_penalty, x0, args=(radii, n),
                     method='L-BFGS-B', bounds=bounds,
                     options={'maxiter': 1000, 'ftol': 1e-15})
    
    return result.x.reshape((n, 2))


def optimize_radii_given_positions(centers, n):
    """Optimize radii for given positions using LP."""
    r = cp.Variable(n)
    
    objective = cp.Maximize(cp.sum(r))
    
    constraints = []
    
    for i in range(n):
        x, y = centers[i]
        constraints.append(r[i] <= x - 1e-10)
        constraints.append(r[i] <= 1 - x - 1e-10)
        constraints.append(r[i] <= y - 1e-10)
        constraints.append(r[i] <= 1 - y - 1e-10)
        constraints.append(r[i] >= 1e-10)
    
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            constraints.append(r[i] + r[j] <= dist - 1e-10)
    
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.ECOS)
    except Exception:
        try:
            prob.solve(solver=cp.SCS)
        except Exception:
            prob.solve()
    
    if r.value is None:
        return np.ones(n) * 0.01
    
    return np.maximum(r.value, 1e-10)


def initialize_hexagonal(n, seed, spacing=0.21):
    """Initialize with hexagonal grid pattern."""
    np.random.seed(seed)
    
    centers = np.zeros((n, 2))
    
    row_spacing = spacing * np.sqrt(3) / 2
    
    idx = 0
    for row in range(10):
        for col in range(10):
            if idx >= n:
                break
            x = 0.05 + col * spacing + (row % 2) * spacing / 2
            y = 0.05 + row * row_spacing
            if x <= 0.95 and y <= 0.95:
                centers[idx] = [x, y]
                idx += 1
    
    centers += np.random.uniform(-0.008, 0.008, centers.shape)
    
    radii = np.ones(n) * spacing / 2.2
    return centers, radii


def initialize_random(n, seed):
    """Initialize with random placement."""
    np.random.seed(seed)
    
    centers = np.random.uniform(0.1, 0.9, (n, 2))
    radii = np.ones(n) * 0.08
    
    return centers, radii


def initialize_corners_and_fill(n, seed):

    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    corner_positions = [[0.15, 0.15], [0.85, 0.15], [0.15, 0.85], [0.85, 0.85]]

        centers[i] = pos
        radii[i] = 0.15

    idx = 4

    for row in range(8):
        for col in range(8):
            if idx >= n:
                break

            y = 0.1 + row * row_spacing
            if x <= 0.9 and y <= 0.9:
                centers[idx] = [x, y]
                radii[idx] = 0.08
                idx += 1
    
    centers += np.random.uniform(-0.005, 0.005, centers.shape)
    return centers, radii


def combined_objective(x, n):
    """Combined objective for positions and radii optimization."""
    centers = x[:2 * n].reshape((n, 2))
    radii = x[2 * n:]
    
    obj = -np.sum(radii)
    
    penalty = 0.0
    mu = 500.0
    
    for i in range(n):
        xi, yi = centers[i]
        ri = radii[i]
        
        if xi < ri:
            penalty += mu * (xi - ri) ** 2
        if xi > 1 - ri:
            penalty += mu * (xi - (1 - ri)) ** 2
        if yi < ri:
            penalty += mu * (yi - ri) ** 2
        if yi > 1 - ri:
            penalty += mu * (yi - (1 - ri)) ** 2
    
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            min_dist = radii[i] + radii[j]
            if dist < min_dist:
                penalty += mu * (min_dist - dist) ** 2
    
    return obj + penalty


def is_valid_packing(centers, radii, n):
    """Check if the packing satisfies all constraints."""
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-10 or x + r > 1 + 1e-10:
            return False
        if y - r < -1e-10 or y + r > 1 + 1e-10:
            return False
    
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-10:
                return False
    
    return True
