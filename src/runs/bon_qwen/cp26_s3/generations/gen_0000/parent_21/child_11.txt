import numpy as np
from scipy.optimize import minimize

def packing_loss(params, N, weight):
    """
    Computes the objective function: negative sum of radii plus penalty terms.
    params: flattened array [x1, y1, r1, x2, y2, r2, ..., xN, yN, rN]
    """
    centers = params[:2*N].reshape(N, 2)
    radii = params[2*N:]
    
    loss = -np.sum(radii)
    
    # Boundary penalties: ensure r <= x <= 1-r and r <= y <= 1-r
    # x - r >= 0  =>  r - x <= 0
    loss += weight * np.sum(np.maximum(0.0, radii - centers[:, 0])**2)
    # x + r <= 1  =>  x + r - 1 <= 0
    loss += weight * np.sum(np.maximum(0.0, centers[:, 0] + radii - 1.0)**2)
    # y - r >= 0
    loss += weight * np.sum(np.maximum(0.0, radii - centers[:, 1])**2)
    # y + r <= 1
    loss += weight * np.sum(np.maximum(0.0, centers[:, 1] + radii - 1.0)**2)
    
    # Overlap penalties: ensure distance >= r_i + r_j
    diffs = centers[:, None, :] - centers[None, :, :]  # Shape (N, N, 2)
    dists = np.sqrt(np.sum(diffs**2, axis=2) + 1e-16)   # Shape (N, N), add eps for stability
    radii_sum = radii[:, None] + radii[None, :]          # Shape (N, N)
    
    # Zero out diagonal to avoid self-penalization
    np.fill_diagonal(dists, 1e9)
    overlap = np.maximum(0.0, radii_sum - dists)
    loss += weight * np.sum(overlap**2)
    
    return loss

def run_packing() -> tuple:
    N = 26
    best_centers = None
    best_radii = None
    best_sum = -1.0
    
    # Optimization settings
    penalty_schedule = [10, 50, 200, 1000, 5000]
    bounds = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N
    opt_options = {'maxiter': 5000, 'ftol': 1e-15, 'gtol': 1e-12}
    
    # Run multiple trials to escape local minima
    for trial in range(4):
        np.random.seed(42 + trial)
        # Random initialization within safe margins
        c_init = np.random.uniform(0.15, 0.85, (N, 2))
        r_init = np.full(N, 0.045)
        x0 = np.concatenate([c_init.flatten(), r_init])
        current_x = x0.copy()
        
        # Continuation method: gradually increase penalty weight
        for w in penalty_schedule:
            res = minimize(packing_loss, current_x, args=(N, w), 
                           bounds=bounds, method='L-BFGS-B', options=opt_options)
            current_x = res.x
            
        centers = current_x[:2*N].reshape(N, 2)
        radii = current_x[2*N:]
        
        # Safeguard: if minor violations exist, shrink radii slightly to guarantee validity
        # This ensures the validator passes even with numerical noise
        max_violation = 0.0
        for i in range(N):
            x, y, r = centers[i, 0], centers[i, 1], radii[i]
            v1 = max(0, r - x)
            v2 = max(0, r - (1-x))
            v3 = max(0, r - y)
            v4 = max(0, r - (1-y))
            max_violation = max(max_violation, v1, v2, v3, v4)
            
        for i in range(N):
            for j in range(i+1, N):
                d = np.sqrt((centers[i,0]-centers[j,0])**2 + (centers[i,1]-centers[j,1])**2)
                v = radii[i] + radii[j] - d
                if v > max_violation:
                    max_violation = v
                    
        if max_violation > 1e-10:
            scale = 1.0 - (max_violation / np.min(radii) * 1.1) if np.min(radii) > 0 else 0.99
            radii *= max(0.95, scale)
            
        current_sum = np.sum(radii)
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers.copy()
            best_radii = radii.copy()
            
    return best_centers, best_radii, best_sum
