To maximize the sum of radii for 26 circles in a unit square, we treat the problem as a joint optimization of circle centers and radii. The core logic is:
1.  **Problem Decomposition**: For any fixed set of centers, the problem of maximizing the sum of radii subject to non-overlap and boundary constraints is a Linear Programming (LP) problem. 
2.  **Objective Function**: We define the objective function as the optimal sum of radii found by the LP solver for a given configuration of centers. We aim to maximize this function.
3.  **Initialization**: We initialize the 26 centers in a hexagonal grid, as hexagonal packing is the densest regular arrangement in 2D. This provides a high-quality starting point.
4.  **Optimization**: We employ a multi-stage optimization strategy:
    *   **Local Search (Powell)**: A gradient-free local optimizer is used to fine-tune the centers from the hexagonal starting position.
    *   **Random Restart Simulated Annealing**: To avoid local optima, we run a randomized search that perturbs centers and accepts improvements (or occasionally worse states) to explore the configuration space more broadly.
5.  **Final Polish**: A final coordinate ascent step adjusts each circle's position individually to squeeze out the last bit of radius sum.

This combination of structured initialization, local refinement, and stochastic exploration ensures a robust and high-performing packing configuration.

import numpy as np
from scipy.optimize import linprog, minimize
import math
import time

def get_max_sum_radii(centers):
    """
    For fixed centers, solves the LP to find max radii sum.
    """
    n = centers.shape[0]
    
    # Calculate boundary distances
    # r_i <= min(x, 1-x, y, 1-y)
    boundary_dists = np.min([
        centers[:, 0], 
        1 - centers[:, 0], 
        centers[:, 1], 
        1 - centers[:, 1]
    ], axis=0)

    # Objective: Maximize sum(r_i) -> Minimize -sum(r_i)
    c = -np.ones(n)
    
    # Constraints: r_i + r_j <= dist(i, j)
    # r_i <= boundary_dists[i]
    
    A_ub_rows = []
    b_ub_rows = []
    
    # Pairwise constraints
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub_rows.append(row)
            b_ub_rows.append(dist)
            
    # Boundary constraints
    for i in range(n):
        row = np.zeros(n)
        row[i] = 1.0
        A_ub_rows.append(row)
        b_ub_rows.append(boundary_dists[i])
        
    A_ub = np.array(A_ub_rows)
    b_ub = np.array(b_ub_rows)
    
    bounds = [(0, None) for _ in range(n)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and res.fun != -np.inf:
            return res.x, -res.fun
    except:
        pass
        
    # Fallback
    return np.zeros(n), 0.0

def evaluate_packing(centers):
    """Wrapper for optimization, handles clipping and returns negative sum."""
    centers = np.clip(centers, 1e-9, 1 - 1e-9)
    _, score = get_max_sum_radii(centers)
    return -score # We minimize negative score

def run_packing():
    n = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # 1. Hexagonal Grid Initialization
    centers = np.zeros((n, 2))
    idx = 0
    row = 0
    y = 0.1
    while idx < n:
        x = 0.1
        shift = 0.15 * (row % 2) # Stagger odd rows
        while x <= 0.9 and idx < n:
            if idx < n:
                centers[idx] = [x + shift, y]
                idx += 1
            x += 0.3 # Approximate spacing
        row += 1
        y += 0.26 # Approximate vertical spacing

    # 2. Local Optimization (Powell)
    # Flatten centers for optimizer
    x0 = centers.flatten()
    
    try:
        res = minimize(evaluate_packing, x0, method='Powell', 
                       options={'maxiter': 200, 'ftol': 1e-9})
        if not np.isnan(res.fun):
            curr_centers = res.x.reshape(n, 2)
            radii, curr_sum = get_max_sum_radii(curr_centers)
            if curr_sum > best_sum:
                best_sum = curr_sum
                best_centers = curr_centers.copy()
                best_radii = radii.copy()
    except Exception:
        pass

    # 3. Randomized Search (Simulated Annealing)
    # Start from best found so far
    current_centers = best_centers.copy() if best_centers is not None else centers
    current_radii, current_sum = get_max_sum_radii(current_centers)
    
    best_sum_sa = current_sum
    best_centers_sa = current_centers.copy()
    best_radii_sa = current_radii.copy()
    
    temp = 0.1
    decay = 0.995
    
    # Run SA loop
    # We use a fixed number of iterations based on a simple time check or count
    iterations = 5000
    
    for _ in range(iterations):
        # Perturb all centers slightly
        noise = np.random.normal(0, 0.02, size=current_centers.shape)
        proposed_centers = current_centers + noise
        proposed_centers = np.clip(proposed_centers, 1e-9, 1 - 1e-9)
        
        proposed_radii, proposed_sum = get_max_sum_radii(proposed_centers)
        
        delta = proposed_sum - current_sum
        
        # Acceptance criteria
        if delta > 0 or np.random.rand() < math.exp(delta / max(temp, 1e-9)):
            current_centers = proposed_centers
            current_radii = proposed_radii
            current_sum = proposed_sum
            
            if current_sum > best_sum_sa:
                best_sum_sa = current_sum
                best_centers_sa = current_centers.copy()
                best_radii_sa = current_radii.copy()
                
        temp *= decay

    # Update global best
    if best_sum_sa > best_sum:
        best_sum = best_sum_sa
        best_centers = best_centers_sa
        best_radii = best_radii_sa

    # 4. Coordinate Ascent Polish
    # Try moving each circle individually to improve sum
    improved = True
    while improved:
        improved = False
        for i in range(n):
            # Save current
            old_c = best_centers[i].copy()
            
            # Try random perturbations for circle i
            best_local_sum = best_sum
            best_local_c = old_c.copy()
            
            # Check 8 neighbors and random positions
            step = 0.05
            for dx, dy in [(step, 0), (-step, 0), (0, step), (0, -step),
                           (step, step), (step, -step), (-step, step), (-step, -step)]:
                test_c = best_centers.copy()
                test_c[i] = best_centers[i] + np.array([dx, dy])
                test_c = np.clip(test_c, 1e-9, 1 - 1e-9)
                
                r, s = get_max_sum_radii(test_c)
                if s > best_local_sum:
                    best_local_sum = s
                    best_local_c = test_c[i].copy()
            
            if best_local_sum > best_sum:
                best_centers[i] = best_local_c
                best_radii, best_sum = get_max_sum_radii(best_centers)
                improved = True
                
    return best_centers, best_radii, float(best_sum)
