import numpy as np
from scipy.optimize import minimize
import warnings
warnings.filterwarnings('ignore')

def compute_all_constraints(vars):
    """
    Computes all boundary and non-overlap constraints for 26 circles.
    Returns a 1D array where each element must be >= 0 for feasibility.
    """
    n = 26
    centers = vars[:2*n].reshape(n, 2)
    radii = vars[2*n:]
    
    # Preallocate constraint array
    cons = np.empty(4*n + n*(n-1)//2)
    
    # Boundary constraints: x-r >= 0, x+r <= 1, y-r >= 0, y+r <= 1
    for i in range(n):
        cons[4*i] = centers[i, 0] - radii[i]
        cons[4*i+1] = 1.0 - (centers[i, 0] + radii[i])
        cons[4*i+2] = centers[i, 1] - radii[i]
        cons[4*i+3] = 1.0 - (centers[i, 1] + radii[i])
        
    # Pairwise non-overlap constraints: dist^2 >= (r1+r2)^2
    idx = 4*n
    for i in range(n):
        for j in range(i+1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist_sq = dx*dx + dy*dy
            sum_r = radii[i] + radii[j]
            cons[idx] = dist_sq - sum_r*sum_r
            idx += 1
            
    return cons
