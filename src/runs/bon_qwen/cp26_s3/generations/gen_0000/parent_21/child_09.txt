import numpy as np
from scipy.optimize import minimize

def obj_func(v):
    """Objective function: negative sum of radii"""
    return -np.sum(v[2::3])

def constraints_26(v):
    """
    Computes all boundary and non-overlap constraints for 26 circles.
    Returns a 1D numpy array where each element must be >= 0.
    """
    n = 26
    # Boundary constraints: 4 per circle -> 104 total
    bnd = np.zeros(4 * n)
    for i in range(n):
        idx = i * 3
        x, y, r = v[idx], v[idx + 1], v[idx + 2]
        bnd[4 * i]     = x - r          # left
        bnd[4 * i + 1] = 1.0 - x - r    # right
        bnd[4 * i + 2] = y - r          # bottom
        bnd[4 * i + 3] = 1.0 - y - r    # top
        
    # Overlap constraints: n*(n-1)/2 -> 325 total
    m = n * (n - 1) // 2
    olp = np.zeros(m)
    k = 0
    for i in range(n):
        xi, yi, ri = v[i * 3], v[i * 3 + 1], v[i * 3 + 2]
        for j in range(i + 1, n):
            xj, yj, rj = v[j * 3], v[j * 3 + 1], v[j * 3 + 2]
            dist = np.hypot(xi - xj, yi - yj)
            olp[k] = dist - ri - rj
            k += 1
            
    return np.concatenate([bnd, olp])

def run_packing():
    n = 26
    best_v = None
    best_score = -np.inf
    
    # Run multiple trials to escape local optima
    for trial in range(5):
        # Random initialization with small radii to ensure initial feasibility
        x0 = np.random.rand(3 * n)
        x0[2::3] = 0.02  # radii
        x0[0::3] = np.random.rand(n)  # x
        x0[1::3] = np.random.rand(n)  # y
        
        # Variable bounds
        bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * n
        cons = {'type': 'ineq', 'fun': constraints_26}
        
        try:
            res = minimize(
                obj_func, 
                x0, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons, 
                options={'ftol': 1e-12, 'maxiter': 5000}
            )
            if res.success:
                score = -res.fun
                if score > best_score:
                    best_score = score
                    best_v = res.x.copy()
        except Exception:
            continue
            
    # Fallback in case all trials fail (should not happen with proper scipy)
    if best_v is None:
        best_v = np.random.rand(3 * n)
        best_v[2::3] = 0.01
        
    # Extract results
    centers = best_v.reshape(n, 3)[:, :2]
    radii = best_v.reshape(n, 3)[:, 2]
    
    # Clamp to satisfy strict bounds due to numerical precision
    centers = np.clip(centers, 0.0, 1.0)
    radii = np.maximum(radii, 0.0)
    
    return centers, radii, float(np.sum(radii))
