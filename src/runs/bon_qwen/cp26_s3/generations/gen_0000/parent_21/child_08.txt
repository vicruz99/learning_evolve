import numpy as np
from scipy.optimize import minimize

def check_distance_constraint(params, i, j):
    """
    Constraint: distance between circle i and j >= r_i + r_j
    Equivalent to: dist^2 - (r_i + r_j)^2 >= 0
    """
    # Extract parameters
    # params layout: [x0, y0, r0, x1, y1, r1, ...]
    # Index for circle k is 3*k
    idx_i = 3 * i
    idx_j = 3 * j
    
    xi, yi, ri = params[idx_i], params[idx_i+1], params[idx_i+2]
    xj, yj, rj = params[idx_j], params[idx_j+1], params[idx_j+2]
    
    dx = xi - xj
    dy = yi - yj
    dist_sq = dx*dx + dy*dy
    sum_r = ri + rj
    
    # We use dist^2 - (r_i + r_j)^2 >= 0
    # This avoids square root and is differentiable
    return dist_sq - sum_r * sum_r

def check_boundary_x_left(params, i):
    """Constraint: x_i - r_i >= 0"""
    idx = 3 * i
    return params[idx] - params[idx+2]

def check_boundary_x_right(params, i):
    """Constraint: (1 - x_i) - r_i >= 0"""
    idx = 3 * i
    return (1.0 - params[idx]) - params[idx+2]

def check_boundary_y_bottom(params, i):
    """Constraint: y_i - r_i >= 0"""
    idx = 3 * i
    return params[idx+1] - params[idx+2]

def check_boundary_y_top(params, i):
    """Constraint: (1 - y_i) - r_i >= 0"""
    idx = 3 * i
    return (1.0 - params[idx+1]) - params[idx+2]

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # 1. Initialization
    # Create a dense grid of potential centers to ensure a valid starting configuration
    # Grid spacing 0.18, starting at 0.1
    # 6 columns x 5 rows = 30 points. We need 26.
    points = []
    for r in range(5):
        for c in range(6):
            x = 0.1 + c * 0.18
            y = 0.1 + r * 0.18
            points.append([x, y])
    
    # Select first 26 points
    init_centers = np.array(points[:n])
    
    # Initial radii: 0.05 (safe, since grid spacing is 0.18)
    init_radii = np.full(n, 0.05)
    
    # Flatten to [x0, y0, r0, x1, y1, r1, ...]
    x0 = np.zeros(3 * n)
    for i in range(n):
        x0[3*i] = init_centers[i, 0]
        x0[3*i+1] = init_centers[i, 1]
        x0[3*i+2] = init_radii[i]
        
    # 2. Bounds
    # x, y in [0, 1], r in [0, 0.5]
    bounds = []
    for _ in range(n):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 0.5)) # r

    # 3. Constraints
    cons = []
    
    # Boundary constraints
    for i in range(n):
        # x - r >= 0
        cons.append({
            'type': 'ineq',
            'fun': check_boundary_x_left,
            'args': (i,)
        })
        # (1-x) - r >= 0
        cons.append({
            'type': 'ineq',
            'fun': check_boundary_x_right,
            'args': (i,)
        })
        # y - r >= 0
        cons.append({
            'type': 'ineq',
            'fun': check_boundary_y_bottom,
            'args': (i,)
        })
        # (1-y) - r >= 0
        cons.append({
            'type': 'ineq',
            'fun': check_boundary_y_top,
            'args': (i,)
        })
        
    # Pairwise distance constraints
    for i in range(n):
        for j in range(i + 1, n):
            cons.append({
                'type': 'ineq',
                'fun': check_distance_constraint,
                'args': (i, j)
            })

    # 4. Objective Function
    # Maximize sum(r_i) -> Minimize -sum(r_i)
    def objective(params):
        s = 0.0
        for i in range(n):
            s += params[3*i + 2] # r_i
        return -s

    # 5. Optimization
    # Using SLSQP which handles bounds and non-linear constraints
    res = minimize(
        objective,
        x0,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 1000, 'ftol': 1e-12, 'disp': False}
    )
    
    # 6. Extract Results
    best_params = res.x
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    for i in range(n):
        centers[i, 0] = best_params[3*i]
        centers[i, 1] = best_params[3*i+1]
        radii[i] = best_params[3*i+2]
        
    # Ensure radii are non-negative (just in case of numerical noise)
    radii = np.maximum(radii, 0.0)
    
    total_sum = np.sum(radii)
    
    return centers, radii, total_sum

if __name__ == "__main__":
    c, r, s = run_packing()
    print(f"Sum of radii: {s}")
    print(f"Centers:\n{c}")
    print(f"Radii:\n{r}")
