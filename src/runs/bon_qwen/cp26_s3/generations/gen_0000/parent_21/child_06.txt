import numpy as np
from scipy.optimize import minimize

def compute_packing_loss(params, n, penalty_factor):
    """
    Computes the objective function: negative sum of radii + penalty for violations.
    params: flattened array [x1, y1, ..., xn, yn, r1, ..., rn]
    """
    centers = params[:2*n].reshape(n, 2)
    radii = params[2*n:]
    
    # Objective: maximize sum of radii -> minimize negative sum
    loss = -np.sum(radii)
    
    # Boundary penalties: circle must be inside [0,1]x[0,1]
    # x >= r
    val = radii - centers[:, 0]
    loss += penalty_factor * np.sum(np.maximum(0, val)**2)
    # x + r <= 1
    val = centers[:, 0] + radii - 1.0
    loss += penalty_factor * np.sum(np.maximum(0, val)**2)
    # y >= r
    val = radii - centers[:, 1]
    loss += penalty_factor * np.sum(np.maximum(0, val)**2)
    # y + r <= 1
    val = centers[:, 1] + radii - 1.0
    loss += penalty_factor * np.sum(np.maximum(0, val)**2)
    
    # Overlap penalties: dist(i,j) >= r_i + r_j
    # Vectorized pairwise computation
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    r_sum = radii[:, None] + radii[None, :]
    
    # Upper triangle mask to avoid double counting and self-comparison
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    pair_dists = dists[mask]
    pair_r_sum = r_sum[mask]
    
    val = pair_r_sum - pair_dists
    loss += penalty_factor * np.sum(np.maximum(0, val)**2)
    
    return loss

    inits = []

                pts_hex.append([x, y])

        x0 = np.concatenate([x0_centers, np.full(n, 0.09)])
