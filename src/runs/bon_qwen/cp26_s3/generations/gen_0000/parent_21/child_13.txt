def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square

    Args:
        centers: np.array of shape (n, 2) with (x, y) coordinates
        radii: np.array of shape (n) with radius of each circle

    Returns:
        True if valid, False otherwise
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True

def get_overlap_energy(centers, radii):
    """Calculates the energy due to overlaps and boundary violations."""
    n = centers.shape[0]
    energy = 0.0
    
    # Overlap energy
    # Vectorized calculation for overlaps
    # centers shape (n, 2)
    # diff shape (n, n, 2)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist_sq = np.sum(diff**2, axis=2)
    dist = np.sqrt(dist_sq)
    
    # Radii sum matrix
    r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    
    # Overlap depth (negative means no overlap)
    overlap = r_sum - dist
    
    # Set diagonal and lower triangle to 0 to avoid double counting and self-overlap
    np.fill_diagonal(overlap, 0)
    mask = np.triu(np.ones_like(overlap), k=1)
    overlap = overlap * mask
    
    # Sum of squared positive overlaps
    energy += np.sum(np.maximum(0, overlap)**2)
    
    # Boundary energy
    # x - r < 0 -> r - x > 0
    # x + r > 1 -> x + r - 1 > 0
    # Same for y
    
    # Left wall violation
    viol_l = radii - centers[:, 0]
    energy += np.sum(np.maximum(0, viol_l)**2)
    
    # Right wall violation
    viol_r = centers[:, 0] + radii - 1
    energy += np.sum(np.maximum(0, viol_r)**2)
    
    # Bottom wall violation
    viol_b = radii - centers[:, 1]
    energy += np.sum(np.maximum(0, viol_b)**2)
    
    # Top wall violation
    viol_t = centers[:, 1] + radii - 1
    energy += np.sum(np.maximum(0, viol_t)**2)
    
    return energy
