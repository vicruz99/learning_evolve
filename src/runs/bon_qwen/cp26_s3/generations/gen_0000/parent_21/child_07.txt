def is_valid(centers, radii, n, tol=1e-10):
    """Check if packing is valid without printing."""
    # Check radii non-negative
    if np.any(radii < -tol):
        return False
    if np.any(np.isnan(radii)):
        return False
    if np.any(np.isnan(centers)):
        return False

    # Check boundaries
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -tol or x + r > 1 + tol or y - r < -tol or y + r > 1 + tol:
            return False

    # Check overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - tol:
                return False

    return True

def hexagonal_from_config(config):
    """Create hexagonal packing from row configuration."""
    n_rows = len(config)
    n = sum(config)

    # Estimate radius based on configuration
    # The horizontal constraint: max circles in a row determines max radius
    max_cols = max(config)

    # For a row with k circles in hexagonal packing:
    # Width needed = 2*r*(k-1) + 2*r = 2*r*k for non-offset row
    # But for offset rows, width = 2*r*(k-1) + 2*r as well
    # Actually, for k circles with spacing 2r, width = 2r*(k-1) + 2r = 2rk
    # So r <= 1/(2*k) for the widest row

    # Vertical constraint: n_rows rows with spacing r*sqrt(3)
    # Height = r + (n_rows-1)*r*sqrt(3) + r = 2r + (n_rows-1)*r*sqrt(3)
    # So r <= 1 / (2 + (n_rows-1)*sqrt(3))

    r_horiz = 1.0 / (2 * max_cols)
    r_vert = 1.0 / (2 + (n_rows - 1) * np.sqrt(3))
    r = min(r_horiz, r_vert) * 0.95  # Slightly smaller to allow optimization room

    centers = np.zeros((n, 2))
    radii = np.ones(n) * r

    idx = 0
    for row_idx in range(n_rows):
        n_in_row = config[row_idx]
        y = r + row_idx * r * np.sqrt(3)

        # Offset for odd rows in hexagonal packing
        if row_idx % 2 == 1:
            x_start = 2 * r
        else:
            x_start = r

        # Center the row horizontally
        if n_in_row > 0:
            row_width = (n_in_row - 1) * 2 * r
            available_width = 1 - 2 * r
            x_offset = (available_width - row_width) / 2
        else:
            x_offset = 0

        for j in range(n_in_row):
            if idx >= n:
                break
            x = x_start + x_offset + j * 2 * r
            centers[idx] = [x, y]
            idx += 1

    if idx < n:
        return None, None

    return centers[:n], radii[:n]

def optimize_packing(centers, radii, n, max_iter=2000):
    """Optimize packing using scipy.optimize."""
    # Flatten variables: [x1, y1, x2, y2, ..., x26, y26, r1, r2, ..., r26]
    x0 = np.concatenate([centers.flatten(), radii])

    # Objective: maximize sum of radii (minimize negative)
    def objective(vars):
        return -np.sum(vars[2 * n:])

    # Bounds
    bounds = [(0, 1)] * (2 * n) + [(0, 0.5)] * n

    # Constraints
    constraints = []

    # Boundary constraints: x_i - r_i >= 0, etc.
    for i in range(n):
        # x_i >= r_i
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: v[2 * i] - v[2 * n + i]
        })
        # x_i <= 1 - r_i
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: 1 - v[2 * i] - v[2 * n + i]
        })
        # y_i >= r_i
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: v[2 * i + 1] - v[2 * n + i]
        })
        # y_i <= 1 - r_i
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: 1 - v[2 * i + 1] - v[2 * n + i]
        })

    # Non-overlap constraints: dist(i,j) >= r_i + r_j
    # Using squared distance for differentiability
    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({
                'type': 'ineq',
                'fun': lambda v, i=i, j=j:
                    (v[2 * i] - v[2 * j]) ** 2 + (v[2 * i + 1] - v[2 * j + 1]) ** 2 -
                    (v[2 * n + i] + v[2 * n + j]) ** 2
            })

    # Try SLSQP
    try:
        result = minimize(
            objective,
            x0,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={
                'maxiter': max_iter,
                'ftol': 1e-15,
                'disp': False
            }
        )

        if result.success or result.fun < 0:
            centers_opt = result.x[:2 * n].reshape((n, 2))
            radii_opt = result.x[2 * n:]

            # Ensure radii are non-negative
            radii_opt = np.maximum(radii_opt, 0)

            # Clamp centers to valid range
            for i in range(n):
                centers_opt[i, 0] = np.clip(centers_opt[i, 0], radii_opt[i], 1 - radii_opt[i])
                centers_opt[i, 1] = np.clip(centers_opt[i, 1], radii_opt[i], 1 - radii_opt[i])

            return centers_opt, radii_opt
    except Exception:
        pass

    # Fallback: return original
    return centers, radii


def optimize_with_penalty(centers, radii, n, lr=0.001, max_iter=5000):
    """Optimize using penalty method for speed."""
    centers = centers.copy()
    radii = radii.copy()

    for iteration in range(max_iter):
        # Compute gradient of objective (w.r.t. radii): +1 for each radius
        grad_r = np.ones(n)

        # Compute penalty gradients
        alpha = 1000  # Penalty weight

        # Boundary penalties
        for i in range(n):
            # x_i - r_i < 0 penalty
            val = centers[i, 0] - radii[i]
            if val < 0:
                grad_r[i] -= alpha * (-val)
            # 1 - x_i - r_i < 0 penalty
            val = 1 - centers[i, 0] - radii[i]
            if val < 0:
                grad_r[i] -= alpha * (-val)
            # y_i - r_i < 0 penalty
            val = centers[i, 1] - radii[i]
            if val < 0:
                grad_r[i] -= alpha * (-val)
            # 1 - y_i - r_i < 0 penalty
            val = 1 - centers[i, 1] - radii[i]
            if val < 0:
                grad_r[i] -= alpha * (-val)

        # Overlap penalties
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                if dist < radii[i] + radii[j]:
                    overlap = radii[i] + radii[j] - dist
                    if dist > 1e-10:
                        # Gradient w.r.t. centers
                        factor = alpha * overlap / dist
                        centers[i, 0] += factor * dx * lr
                        centers[j, 0] -= factor * dx * lr
                        centers[i, 1] += factor * dy * lr
                        centers[j, 1] -= factor * dy * lr
                    grad_r[i] -= alpha * overlap
                    grad_r[j] -= alpha * overlap

        # Update radii
        radii += grad_r * lr
        radii = np.maximum(radii, 0.001)

        # Clamp centers
        for i in range(n):
            centers[i, 0] = np.clip(centers[i, 0], radii[i], 1 - radii[i])
            centers[i, 1] = np.clip(centers[i, 1], radii[i], 1 - radii[i])

    return centers, radii


def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square

    Args:
        centers: np.array of shape (n, 2) with (x, y) coordinates
        radii: np.array of shape (n) with radius of each circle

    Returns:
        True if valid, False otherwise
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True
