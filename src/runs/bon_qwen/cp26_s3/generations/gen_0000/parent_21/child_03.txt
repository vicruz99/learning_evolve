import numpy as np
from scipy.optimize import minimize

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square

    Args:
        centers: np.array of shape (n, 2) with (x, y) coordinates
        radii: np.array of shape (n) with radius of each circle

    Returns:
        True if valid, False otherwise
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    
    Returns:
        tuple: (centers, radii, sum_radii)
    """
    n = 26
    np.random.seed(42)

    # 1. Initialization Strategy: Hexagonal Packing (Staggered Rows)
    # Pattern: 5, 4, 5, 4, 5, 3 circles in 6 rows
    # This allows for high density.
    row_counts = [5, 4, 5, 4, 5, 3]
    centers_init = np.zeros((n, 2))
    r_guess = 0.09  # Initial guess smaller than 0.1 to allow space
    
    idx = 0
    # Approximate vertical spacing for hexagonal packing: r * sqrt(3)
    # We distribute rows evenly in [0, 1]
    y_step = 1.0 / (len(row_counts) - 1) if len(row_counts) > 1 else 1.0
    
    for i, count in enumerate(row_counts):
        y = i * y_step
        # Center the row horizontally
        # Width of row with 'count' circles is (count * 2r) if touching boundaries?
        # Actually, we place centers at even spacing
        x_positions = np.linspace(0, 1, count + 2)[1:-1] # Evenly spaced in [0,1] initially
        
        # Shift alternating rows to create staggered effect
        if i % 2 == 1:
            # Shift by half a step
            step = 1.0 / (count + 1)
            x_positions = x_positions + step / 2
        
        for x in x_positions:
            centers_init[idx] = [x, y]
            idx += 1
            
    # 2. Optimization Function
    def objective(vars):
        # vars = [x0, y0, r0, x1, y1, r1, ...]
        # Structure: 3 variables per circle (x, y, r)
        rs = vars[2::3]
        return -np.sum(rs) # Minimize negative sum = Maximize sum

    def constraint_boundary(vars):
        # x - r >= 0, x + r <= 1, y - r >= 0, y + r <= 1
        # x >= r, y >= r, x <= 1-r, y <= 1-r
        # x - r >= 0
        cs = vars.reshape(-1, 3)
        res = []
        for c in cs:
            x, y, r = c
            res.append(x - r)       # >= 0
            res.append(y - r)       # >= 0
            res.append(x + r - 1)   # <= 0 -> 1 - (x+r) >= 0
            res.append(y + r - 1)   # <= 0 -> 1 - (y+r) >= 0
            res.append(r)           # >= 0
        return np.array(res)

    def constraint_overlap(vars):
        # dist_ij >= r_i + r_j  =>  (xi-xj)^2 + (yi-yj)^2 - (ri+rj)^2 >= 0
        cs = vars.reshape(-1, 3)
        res = []
        for i in range(n):
            for j in range(i + 1, n):
                xi, yi, ri = cs[i]
                xj, yj, rj = cs[j]
                dist_sq = (xi - xj)**2 + (yi - yj)**2
                rad_sum = ri + rj
                res.append(dist_sq - rad_sum**2)
        return np.array(res)

    # 3. Run Optimization
    # Initial vars: [x, y, r] for each circle
    r_init = np.full(n, r_guess)
    vars_init = np.zeros(n * 3)
    vars_init[0::3] = centers_init[:, 0]
    vars_init[1::3] = centers_init[:, 1]
    vars_init[2::3] = r_init

    # Bounds for x, y in [0, 1], r in [0, 0.5]
    bounds = [(0, 1)] * (n * 3)
    # Tighten r bounds? No, let optimizer decide.
    
    # Constraints
    cons = [
        {'type': 'ineq', 'fun': constraint_boundary},
        {'type': 'ineq', 'fun': constraint_overlap}
    ]

    # Try SLSQP
    try:
        res = minimize(objective, vars_init, method='SLSQP', bounds=bounds, constraints=cons, 
                       options={'maxiter': 1000, 'ftol': 1e-12})
        vars_opt = res.x
    except Exception as e:
        print(f"Optimization failed: {e}")
        # Fallback to initialization if optimization fails completely
        vars_opt = vars_init

    # 4. Extract results
    centers = vars_opt[:2*n].reshape(n, 2)
    radii = vars_opt[2*n:]
    
    # Ensure non-negative radii due to numerical noise
    radii = np.maximum(radii, 0.0)

    # 5. Validation and Cleanup
    # Check if valid, if not, try to fix (e.g. reduce radii slightly)
    if not validate_packing(centers, radii):
        print("Warning: Initial optimization result invalid. Attempting correction...")
        # Simple correction: reduce radii until valid
        scale = 1.0
        while scale > 0.1:
            radii_adj = radii * scale
            if validate_packing(centers, radii_adj):
                radii = radii_adj
                break
            scale -= 0.01
            
        # Re-validate
        if not validate_packing(centers, radii):
             # If still failing, return a safe small packing
             centers = np.random.rand(n, 2)
             radii = np.full(n, 0.01)
             
    sum_radii = np.sum(radii)
    
    # Ensure outputs are clean numpy arrays
    centers = np.array(centers)
    radii = np.array(radii)

    return centers, radii, float(sum_radii)
