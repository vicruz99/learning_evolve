import numpy as np
from scipy.optimize import minimize

def generate_hexagonal_packing(n_circles):
    """Generates a hexagonal packing configuration for n_circles."""
    best_config = None
    best_diff = float('inf')
    
    # Search for optimal number of rows (m) to match square aspect ratio
    for m in range(3, 15):
        base = n_circles // m
        rem = n_circles % m
        row_sizes = [base + 1] * rem + [base] * (m - rem)
        
        # Estimate aspect ratio for hexagonal packing
        W = 2 * max(row_sizes)
        H = (m - 1) * np.sqrt(3) + 2
        diff = abs(W - H)
        if diff < best_diff:
            best_diff = diff
            best_config = row_sizes
            
    # Construct coordinates assuming unit radius
    centers = []
    r_temp = 1.0
    current_y = 0.0
    for i, size in enumerate(best_config):
        shift = r_temp if i % 2 == 1 else 0.0
        xs = [k * 2 * r_temp + shift for k in range(size)]
        for x in xs:
            centers.append([x, current_y])
        current_y += np.sqrt(3) * r_temp
        
    centers = np.array(centers)
    min_x, max_x = centers[:, 0].min(), centers[:, 0].max()
    min_y, max_y = centers[:, 1].min(), centers[:, 1].max()
    
    # Calculate bounding box including radii
    W_shape = max_x - min_x + 2 * r_temp
    H_shape = max_y - min_y + 2 * r_temp
    
    # Scale to fit unit square and center
    scale = 1.0 / max(W_shape, H_shape)
    centers_scaled = (centers - np.array([min_x, min_y])) * scale
    r_init = r_temp * scale
    
    # Center the configuration
    min_x_s, max_x_s = centers_scaled[:, 0].min(), centers_scaled[:, 0].max()
    min_y_s, max_y_s = centers_scaled[:, 1].min(), centers_scaled[:, 1].max()
    shift_x = (1 - (max_x_s + 2 * r_init)) / 2 + r_init - min_x_s
    shift_y = (1 - (max_y_s + 2 * r_init)) / 2 + r_init - min_y_s
    centers_scaled[:, 0] += shift_x
    centers_scaled[:, 1] += shift_y
    
    return centers_scaled, np.full(n_circles, r_init)

def evaluate_and_validate(centers, radii):
    """Evaluates the sum of radii and validates the packing."""
    n = len(radii)
    sum_radii = np.sum(radii)
    
    # Check boundaries
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            return False, 0
            
    # Check overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, 0
                
    return True, sum_radii

def objective_function(variables):
    """Objective: Maximize sum of radii (minimize negative sum)."""
    radii = variables[52:78]
    return -np.sum(radii)

def get_constraints():
    """Define constraints for the optimization problem."""
    constraints = []
    
    # Boundary constraints: r <= x <= 1-r and r <= y <= 1-r
    for i in range(26):
        x_idx, y_idx, r_idx = 2*i, 2*i + 1, 52 + i
        # x - r >= 0
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: v[x_idx] - v[r_idx]})
        # 1 - x - r >= 0
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: 1 - v[x_idx] - v[r_idx]})
        # y - r >= 0
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: v[y_idx] - v[r_idx]})
        # 1 - y - r >= 0
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: 1 - v[y_idx] - v[r_idx]})
        
    # Non-overlap constraints: dist^2 >= (r_i + r_j)^2
    for i in range(26):
        for j in range(i + 1, 26):
            xi, yi = 2*i, 2*i + 1
            xj, yj = 2*j, 2*j + 1
            ri, rj = 52 + i, 52 + j

                dx = v[xi] - v[xj]
                dy = v[yi] - v[yj]
                return (dx**2 + dy**2) - (v[ri] + v[rj])**2
            constraints.append({'type': 'ineq', 'fun': dist_constraint})
            
    return constraints

def run_packing():
