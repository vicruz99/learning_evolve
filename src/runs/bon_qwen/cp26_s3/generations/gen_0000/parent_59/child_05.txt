import numpy as np
from scipy.optimize import minimize

def run_packing():
    """
    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    Uses a hexagonal initialization and numerical optimization to maximize the minimum clearance.
    """
    n_circles = 26
    
    # --- 1. Initialization ---
    # We use a hexagonal packing pattern as a starting point.
    # A 6-row arrangement with counts [5, 5, 5, 5, 4, 2] sums to 26.
    # Or [5, 5, 5, 4, 4, 3] = 26.
    # Let's try to pack them as tightly as possible.
    
    # Approximate radius for 26 equal circles is around 0.101.
    # Diameter ~ 0.202.
    # Vertical spacing for hex packing: r * sqrt(3) ~ 0.175.
    # 6 rows height: 2r + 5 * r*sqrt(3) ~ 0.2 + 0.875 = 1.075 (too tall for r=0.101?)
    # Actually 5 gaps: 5 * 0.175 = 0.875. Plus 2r = 0.202. Total 1.077.
    # So 6 rows of r=0.101 might be tight.
    # Let's try 5 rows? 5*5=25. Need 1 more.
    # Maybe 5 rows of 5, and 1 small?
    # Or a denser packing.
    
    # Let's initialize with a grid and perturb.
    # 5x5 grid centers
    centers = []
    # 5 rows, 5 cols
    for r in range(5):
        for c in range(5):
            x = 0.1 + c * 0.2
            y = 0.1 + r * 0.2
            centers.append([x, y])
    # Add 26th circle in the center gap?
    # Center of square is 0.5, 0.5.
    # Gap in 5x5 grid (radius 0.1) is small.
    # Let's place it at 0.5, 0.5 with small radius initially.
    centers.append([0.5, 0.5])
    
    centers = np.array(centers)
    
    # Perturb slightly to break symmetry and help optimization
    np.random.seed(42)
    centers += np.random.normal(0, 0.01, centers.shape)
    
    # Clip to valid range
    centers = np.clip(centers, 0.05, 0.95)

    # --- 2. Optimization ---
    # We want to maximize the minimum distance between any two centers (divided by 2) 
    # and the distance to the walls.
    # Let x be the flattened array of centers (shape 52).
    
    def objective(x):
        """
        Negative of the minimum clearance.
        Clearance is min( dist(i,j)/2 for all i,j, dist(i, wall) for all i ).
        """
        centers_opt = x.reshape(-1, 2)
        min_clearance = float('inf')
        
        n = centers_opt.shape[0]
        
        # Check wall distances
        # Distance to left wall is x, to right is 1-x, etc.
        # Clearance allowed is min(x, 1-x, y, 1-y).
        dists_wall = np.min([centers_opt[:, 0], 1 - centers_opt[:, 0], 
                             centers_opt[:, 1], 1 - centers_opt[:, 1]], axis=0)
        min_clearance = np.min(dists_wall)
        
        # Check inter-circle distances
        # We only need to check pairwise distances.
        # Optimization can be slow if we compute all pairs every step, but n=26 is small.
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers_opt[i] - centers_opt[j]) ** 2))
                # The radius sum would be dist. Max radius for each is dist/2.
                # So clearance contributed by this pair is dist/2.
                if dist / 2 < min_clearance:
                    min_clearance = dist / 2
                    # Early exit if very small
                    if min_clearance < 1e-5:
                        return -min_clearance
                        
        return -min_clearance

    # Bounds: centers must be in [0, 1]
    bounds = [(0, 1)] * (n_circles * 2)
    
    # Initial guess
    x0 = centers.flatten()
    
    # We try a few optimizations with different initial perturbations or just run one robustly.
    # SLSQP is generally good for constrained problems, but here we encoded constraints in objective.
    # However, boundary violations are handled by the wall distance term in objective.
    # But we must ensure centers stay in [0,1]. The bounds handle that.
    
    # To encourage distinct positions and avoid local minima where circles overlap (dist=0),
    # the objective naturally pushes them apart.
    
    # We might want to use a method that handles non-smooth min function well.
    # The min function is non-differentiable.
    # However, for packing, usually the active constraints are a few pairs.
    
    # Let's try 'Nelder-Mead' or 'Powell' as they don't require gradients and handle non-smoothness okay.
    # Or 'L-BFGS-B' with bounds.
    
    best_res = None
    best_fun = -1.0
    
    # Run multiple times with slight noise
    for seed in range(10):
        np.random.seed(seed)
        x_current = x0.copy()
        # Add small noise
        x_current += np.random.normal(0, 0.005, x_current.shape)
        x_current = np.clip(x_current, 0.01, 0.99)
        
        # Optimization
        res = minimize(objective, x_current, method='L-BFGS-B', bounds=bounds, options={'maxiter': 1000, 'ftol': 1e-10})
        
        if -res.fun > best_fun:
            best_fun = -res.fun
            best_res = res
            
        # Also try Nelder-Mead for robustness
        res2 = minimize(objective, x_current, method='Nelder-Mead', options={'maxiter': 2000, 'xatol': 1e-8, 'fatol': 1e-8})
        if -res2.fun > best_fun:
            best_fun = -res2.fun
            best_res = res2

    # Extract best centers
    optimal_centers = best_res.x.reshape(-1, 2)
    optimal_r = best_fun
    
    # --- 3. Final Validation and Output ---
    # The problem asks for radii array.
    # With equal radii packing, all radii are optimal_r.
    # However, we might be able to increase some radii if the packing is not uniform.
    # But for the purpose of a robust solution, equal radii = optimal_r is valid.
    
    radii = np.full(n_circles, optimal_r)
    
    # Check validity
    # If optimal_r is very small, something went wrong.
    # With 26 circles, r should be around 0.1.
    
    # Calculate sum
    total_sum = np.sum(radii)
    
    return optimal_centers, radii, total_sum
