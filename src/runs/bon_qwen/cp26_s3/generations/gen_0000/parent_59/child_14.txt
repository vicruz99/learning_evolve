### Strategy

2.  **Numerical Optimization:** Use `scipy.optimize.minimize` with the `SLSQP` method to maximize $\sum r_i$. The constraints include circle-circle non-overlap ($d_{ij} \ge r_i + r_j$) and boundary adherence.
3.  **Constraints Vectorization:** All 429 constraints (boundary and overlap) are vectorized into a single function to improve computational efficiency.
4.  **Randomization:** Since SLSQP is a local optimizer, we run multiple iterations from different random starting points to escape local minima and find a near-global maximum.

import numpy as np
from scipy.optimize import minimize
import math

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square

    Args:
        centers: np.array of shape (n, 2) with (x, y) coordinates
        radii: np.array of shape (n) with radius of each circle

    Returns:
        True if valid, False otherwise
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True

def _generate_greedy_layout(n):
    """
    Places circles in the largest available gaps using a simple grid sampling approach.
    """
    centers = []
    radii = []
    resolution = 200
    xs = np.linspace(0.01, 0.99, resolution)
    ys = np.linspace(0.01, 0.99, resolution)
    
    # Start with a random seed circle to ensure validity and variety
    seed_x = np.random.uniform(0.1, 0.9)
    seed_y = np.random.uniform(0.1, 0.9)
    centers.append([seed_x, seed_y])
    radii.append(0.02)

    for _ in range(n - 1):
        dist_map = np.full((resolution, resolution), -1)
        
        # Calculate distance from each grid point to all existing circles
        # Using broadcasting for efficiency
        existing_centers = np.array(centers)
        existing_radii = np.array(radii)
        
        # Compute distances to centers
        dist_to_centers = np.sqrt(np.sum((xs[:, None] - existing_centers[:, 0])**2 + 
                                          (ys[:, None] - existing_centers[:, 1])**2, axis=2))
        
        # Available space is distance to center minus radius
        available_space = dist_to_centers - existing_radii
        
        # The bottleneck radius for a point is the minimum distance to any existing circle
        min_dist_to_existing = np.min(available_space, axis=2)
        
        # Also consider distance to square boundaries
        dist_to_border = np.minimum(np.minimum(xs, 1 - xs), np.minimum(ys, 1 - ys))
        
        # The radius we can fit at (x, y) is limited by the closest existing circle and the border
        # However, we are looking for the "gap" size.
        # A simpler heuristic: find the point in the grid that is furthest from all existing circles
        # and the boundaries.
        
        # We want to place a circle with radius r.
        # We need center (cx, cy) such that dist(cx, cy, ci) >= ri + r
        # => dist(cx, cy, ci) - ri >= r
        # So the potential radius at a grid point is min_i(dist_to_center_i - ri_i)
        # But it must also be <= dist_to_border
        
        potential_radii = np.minimum(min_dist_to_existing, dist_to_border)
        
        # Mask out areas already occupied by existing circles (where potential radius < 0)
        valid_mask = potential_radii >= 0
        
        if not np.any(valid_mask):
            # Fallback: random placement if grid is full or blocked
            while True:
                cx, cy = np.random.uniform(0.1, 0.9, 2)
                r = 0.01
                # Quick check
                is_valid = True
                for k, (ec, er) in enumerate(zip(existing_centers, existing_radii)):
                    if np.sqrt(np.sum((cx - ec)**2)) < er + r - 1e-5:
                        is_valid = False
                        break
                if is_valid:
                    centers.append([cx, cy])
                    radii.append(r)
                    break
            continue

        valid_potentials = potential_radii[valid_mask]
        if len(valid_potentials) == 0:
            continue
            
        max_r = np.max(valid_potentials)
        # Find the index of the max radius
        idx = np.argmax(potential_radii)
        ix, iy = np.unravel_index(idx, potential_radii.shape)
        
        new_center = [xs[ix], ys[iy]]
        # Clamp radius slightly to avoid boundary issues during optimization
        new_r = min(max_r, 0.15) 
        
        # Add the new circle
        centers.append(new_center)
        radii.append(new_r)

    return np.array(centers), np.array(radii)

def _solve_optimization(centers, radii):
    """
    Optimizes the packing of circles given an initial guess.
    """
    n = len(radii)
    
    # Variable vector: [x1, y1, ..., xn, yn, r1, ..., rn]
    # x indices: 0, 2, ..., 2n-2
    # y indices: 1, 3, ..., 2n-1
    # r indices: 2n, 2n+1, ..., 3n-1
    
    x0 = np.concatenate([centers.flatten(), radii])
    
    # Bounds
    # x, y in [0, 1]
    # r in [0, 0.5]
    bounds = [(0, 1)] * (2 * n) + [(0, 0.5)] * n
    
    # Objective: Maximize sum of radii => Minimize -sum of radii
    def objective(v):
        r_vals = v[2*n:]
        return -np.sum(r_vals)
    
    # Constraints:
    # 1. Boundary: x >= r, 1-x >= r, y >= r, 1-y >= r
    # 2. Overlap: (xi-xj)^2 + (yi-yj)^2 >= (ri+rj)^2
    
    def constraints_func(v):
        n_c = len(v) // 3
        if 3 * n_c != len(v):
            return np.array([])
            
        centers_opt = v[:2*n_c].reshape((n_c, 2))
        radii_opt = v[2*n_c:]
        
        cons = []
        
        # Boundary constraints
        # x >= r  => x - r >= 0
        cons.append(centers_opt[:, 0] - radii_opt)
        # 1 - x >= r => 1 - x - r >= 0
        cons.append(1.0 - centers_opt[:, 0] - radii_opt)
        # y >= r => y - r >= 0
        cons.append(centers_opt[:, 1] - radii_opt)
        # 1 - y >= r => 1 - y - r >= 0
        cons.append(1.0 - centers_opt[:, 1] - radii_opt)
        
        # Overlap constraints
        # (xi-xj)^2 + (yi-yj)^2 - (ri+rj)^2 >= 0
        # We only need i < j
        
        # Calculate pairwise distance squared
        # diff shape: (n, n, 2)
        diff = centers_opt[:, np.newaxis, :] - centers_opt[np.newaxis, :, :]
        dist_sq = np.sum(diff**2, axis=2)
        
        # Calculate (ri + rj)^2
        r_sum = radii_opt[:, np.newaxis] + radii_opt[np.newaxis, :]
        r_sum_sq = r_sum**2
        
        # Difference
        diff_sq = dist_sq - r_sum_sq
        
        # Upper triangle (excluding diagonal)
        mask = np.triu(np.ones((n_c, n_c), dtype=bool), k=1)
        overlap_cons = diff_sq[mask]
        
        cons.append(overlap_cons)
        
        return np.concatenate(cons)
    
    cons = {'type': 'ineq', 'fun': constraints_func}
    
    # Optimization
    res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, 
                   options={'maxiter': 2000, 'ftol': 1e-10})
    
    if res.success:
        final_centers = res.x[:2*n].reshape((n, 2))
        final_radii = res.x[2*n:]
        return final_centers, final_radii, np.sum(final_radii)
    else:
        # Return initial if optimization failed
        return centers, radii, np.sum(radii)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    best_sum = 0.0
    best_centers = np.zeros((n, 2))
    best_radii = np.zeros(n)
    
    # Try multiple random starts
    num_trials = 15
    
    for _ in range(num_trials):
        # 1. Generate initial layout
        centers_init, radii_init = _generate_greedy_layout(n)
        
        # 2. Optimize
        centers_opt, radii_opt, current_sum = _solve_optimization(centers_init, radii_init)
        
        # 3. Validate
        if validate_packing(centers_opt, radii_opt):
            if current_sum > best_sum:
                best_sum = current_sum
                best_centers = centers_opt
                best_radii = radii_opt
        else:
            # If invalid, try to fix small overlaps by slightly reducing radii
            # This is a fallback, but ideally optimization handles it.
            # We can try to shrink radii slightly until valid
            shrink_factor = 1.0
            for _ in range(10):
                shrink_factor -= 0.01
                test_radii = best_radii * shrink_factor
                if validate_packing(centers_opt, test_radii):
                    best_radii = test_radii
                    best_sum = np.sum(test_radii)
                    break

    return best_centers, best_radii, best_sum
