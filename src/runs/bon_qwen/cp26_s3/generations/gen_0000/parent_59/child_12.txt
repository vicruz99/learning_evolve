

<strategy>
The problem is to pack 26 circles in a unit square to maximize the sum of radii, with a target of 2.636.

Key observations:
1. Equal-radius packing for n=26 gives ~2.504, but unequal radii can achieve ~2.636
2. Corner circles can be larger (constrained by 2 boundaries), edge circles medium, interior smaller
3. Need to use nonlinear optimization (SLSQP) with proper constraint formulation
4. Multiple random restarts are essential to avoid local minima

Strategy:
- Use scipy.optimize.minimize with SLSQP method
- Formulate constraints explicitly: boundary (4 per circle) + non-negativity + non-overlap (325 pairwise)
- Use multiple initial configurations (grid, hexagonal, random, corner-biased)
- Run many restarts and keep the best valid solution
- Vectorize constraint computation for efficiency
</strategy>

```python
import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26


def compute_objective(x):
    """Objective: minimize negative sum of radii (i.e., maximize sum of radii)"""
    n = N_CIRCLES
    return -np.sum(x[2 * n:])


def compute_constraints(x):
    """
    Returns vector of constraint values (all should be >= 0 for inequality constraints).
    Constraints:
    - For each circle: x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0, r >= 0
    - For each pair: dist(i,j) - r_i - r_j >= 0
    """
    n = N_CIRCLES
    centers = x[:2 * n].reshape(n, 2)
    radii = x[2 * n:]

    c_boundary = np.zeros(5 * n)
    for i in range(n):
        c_boundary[5 * i] = centers[i, 0] - radii[i]
        c_boundary[5 * i + 1] = 1.0 - centers[i, 0] - radii[i]
        c_boundary[5 * i + 2] = centers[i, 1] - radii[i]
        c_boundary[5 * i + 3] = 1.0 - centers[i, 1] - radii[i]
        c_boundary[5 * i + 4] = radii[i]

    n_overlap = n * (n - 1) // 2
    c_overlap = np.zeros(n_overlap)
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy + 1e-20)
            c_overlap[idx] = dist - radii[i] - radii[j]
            idx += 1

    return np.concatenate([c_boundary, c_overlap])


def make_grid_config(seed):
    """Grid-based initial configuration"""
    np.random.seed(seed)
    n = N_CIRCLES
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.095

    idx = 0
    for row in range(6):
        for col in range(5):
            if idx >= n:
                break
            x = (col + 0.5) / 5.0
            y = (row + 0.5) / 6.0
            centers[idx] = [x, y]
            idx += 1

    centers += np.random.randn(n, 2) * 0.02
    centers = np.clip(centers, 0.05, 0.95)
    return centers, radii


def make_hexagonal_config(seed):
    """Hexagonal/staggered initial configuration"""
    np.random.seed(seed)
    n = N_CIRCLES
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.095

    idx = 0
    for row in range(6):
        num_in_row = 5 if row % 2 == 0 else 4
        if row % 2 == 0:
            x_start = 1.0 / 10.0
            x_step = 2.0 / 5.0
        else:
            x_start = 3.0 / 10.0
            x_step = 2.0 / 5.0

        y_pos = (row + 0.5) / 6.0

        for col in range(num_in_row):
            if idx >= n:
                break
            x = x_start + col * x_step
            centers[idx] = [x, y_pos]
            idx += 1

    centers += np.random.randn(n, 2) * 0.015
    centers = np.clip(centers, 0.05, 0.95)
    return centers, radii


def make_corner_bias_config(seed):
    """Configuration with larger circles near corners"""
    np.random.seed(seed)
    n = N_CIRCLES
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    # Place 4 corner circles larger
    idx = 0
    corners = [[0.15, 0.15], [0.85, 0.15], [0.15, 0.85], [0.85, 0.85]]
    for c in corners:
        centers[idx] = c
        radii[idx] = 0.14
        idx += 1

    # Place edge circles
    edge_positions = []
    for i in range(1, 5):
        edge_positions.append([i * 0.2, 0.1])
        edge_positions.append([i * 0.2, 0.9])
    for i in range(1, 5):
        edge_positions.append([0.1, i * 0.2])
        edge_positions.append([0.9, i * 0.2])

    for ep in edge_positions:
        if idx >= n:
            break
        centers[idx] = ep
        radii[idx] = 0.09
        idx += 1

    # Fill remaining with interior circles
    while idx < n:
        x = np.random.uniform(0.2, 0.8)
        y = np.random.uniform(0.2, 0.8)
        centers[idx] = [x, y]
        radii[idx] = 0.08
        idx += 1

    centers += np.random.randn(n, 2) * 0.01
    centers = np.clip(centers, radii + 0.01, 1 - radii - 0.01)
    return centers, radii


def make_random_config(seed):
    """Random initial configuration"""
    np.random.seed(seed)
    n = N_CIRCLES
    centers = np.random.rand(n, 2) * 0.8 + 0.1
    radii = np.ones(n) * 0.09
    return centers, radii


def is_valid(centers, radii):
    """Check if packing is valid"""
    n = centers.shape[0]

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if r < -1e-15:
            return False
        if x - r < -1e-12 or x + r > 1 + 1e-12:
            return False
        if y - r < -1e-12 or y + r > 1 + 1e-12:
            return False

    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            if dist < radii[i] + radii[j] - 1e-12:
                return False

    return True


def fix_invalid_packing(centers, radii):
    """Try to fix minor constraint violations"""
    n = centers.shape[0]
    centers = centers.copy()
    radii = radii.copy()

    # Clip radii to be non-negative
    radii = np.maximum(radii, 0)

    # Clip centers to respect boundaries
    for i in range(n):
        centers[i, 0] = np.clip(centers[i, 0], radii[i], 1 - radii[i])
        centers[i, 1] = np.clip(centers[i, 1], radii[i], 1 - radii[i])

    return centers, radii


def run_packing():
    """Main function to find optimal circle packing"""
    constraints = {
        'type': 'ineq',
        'fun': compute_constraints
    }

    bounds = [(0.0, 1.0)] * (2 * N_CIRCLES) + [(0.0, 1.0)] * N_CIRCLES

    best_centers = None
    best_radii = None
    best_sum = -1.0

    # Define initial configuration generators
    config_generators = [
        make_grid_config,
        make_hexagonal_config,
        make_corner_bias_config,
        make_random_config,
    ]

    # Run multiple restarts
    for cfg_idx, gen in enumerate(config_generators):
        for seed in range(15):
            centers, radii = gen(seed)
            x0 = np.concatenate([centers.flatten(), radii])

            # First pass with Nelder-Mead (more robust, no gradients needed)
            result_nm = minimize(
                compute_objective,
                x0,
                method='Nelder-Mead',
                options={
                    'maxiter': 20000,
                    'xatol': 1e-10,
                    'fatol': 1e-12,
                    'adaptive': True
                }
            )

            # Second pass with SLSQP for refinement
            result_sl = minimize(
                compute_objective,
                result_nm.x,
                method='SLSQP',
                constraints=constraints,
                bounds=bounds,
                options={
                    'maxiter': 5000,
                    'ftol': 1e-14,
                    'disp': False
                }
            )

            # Use the better of the two results
            x_opt = result_sl.x if result_sl.fun < result_nm.fun else result_nm.x

            centers_opt = x_opt[:2 * N_CIRCLES].reshape(N_CIRCLES, 2)
            radii_opt = x_opt[2 * N_CIRCLES:]

            # Fix minor violations
            centers_opt, radii_opt = fix_invalid_packing(centers_opt, radii_opt)

            # Validate
            if is_valid(centers_opt, radii_opt):
                s = np.sum(radii_opt)
                if s > best_sum:
                    best_centers = centers_opt.copy()
                    best_radii = radii_opt.copy()
                    best_sum = s

    # If we still don't have a valid solution, use a simple fallback
    if best_centers is None:
        centers, radii = make_grid_config(0)
        best_centers = centers
        best_radii = radii
        best_sum = np.sum(radii)

    return best_centers, best_radii, best_sum
```
