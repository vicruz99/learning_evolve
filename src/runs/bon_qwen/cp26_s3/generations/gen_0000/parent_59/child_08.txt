import numpy as np
from scipy.optimize import minimize

def run_packing():
    np.random.seed(42)
    n = 26
    best_sum = -1.0
    best_centers = None
    best_radii = None

    # Objective function with penalty method
    def objective(vars):
        c = vars[:2*n].reshape((n, 2))
        r = vars[2*n:]
        
        # Primary objective: maximize sum of radii
        loss = -np.sum(r)
        
        # Pairwise non-overlap penalty
        dists = np.linalg.norm(c[:, None, :] - c[None, :, :], axis=2)
        triu_mask = np.triu(np.ones((n, n)), k=1).astype(bool)
        overlaps = r[:, None] + r[None, :] - dists
        overlaps[~triu_mask] = 0.0
        pen_pair = np.sum(np.maximum(0.0, overlaps)**2)
        
        # Boundary containment penalty
        pen_bound = (np.sum(np.maximum(0.0, r - c[:, 0])**2) + 
                     np.sum(np.maximum(0.0, r - (1.0 - c[:, 0]))**2) + 
                     np.sum(np.maximum(0.0, r - c[:, 1])**2) + 
                     np.sum(np.maximum(0.0, r - (1.0 - c[:, 1]))**2))
                     
        # High weight ensures constraints are prioritized
        return loss + 10000.0 * (pen_pair + pen_bound)

    # Bounds for coordinates [0, 1] and radii [1e-4, 0.5]
    bounds = [(0.0, 1.0)] * (2*n) + [(1e-4, 0.5)] * n
    
    # Run multiple trials with different initializations
    for trial in range(10):
        centers = np.zeros((n, 2))
        
        # Mix of hexagonal and random initializations for robustness
        if trial < 6:
            # Hexagonal pattern
            idx = 0
            row = 0
            while idx < n:
                cols = 6 if row % 2 == 0 else 5
                for col in range(cols):
                    if idx >= n: break
                    cx = col * 0.16 + (0.08 if row % 2 != 0 else 0) + np.random.uniform(-0.02, 0.02)
                    cy = row * 0.14 + np.random.uniform(-0.02, 0.02)
                    centers[idx] = [cx, cy]
                    idx += 1
                row += 1
            centers = np.clip(centers, 0.05, 0.95)
        else:
            # Random placement
            centers = np.random.uniform(0.1, 0.9, size=(n, 2))
            
        radii = np.full(n, 0.08)
        x0 = np.concatenate([centers.flatten(), radii])
        
        # Optimize
        res = minimize(objective, x0, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': 2000, 'ftol': 1e-12, 'gtol': 1e-8})
        
        if res.fun is not None:
            c_opt = res.x[:2*n].reshape((n, 2))
            r_opt = res.x[2*n:]
            s = np.sum(r_opt)
            
            # Validate result
            valid = True
            if s > 2.0:
                for i in range(n):
                    x, y = c_opt[i]
                    ri = r_opt[i]
                    if x - ri < -1e-4 or x + ri > 1 + 1e-4 or y - ri < -1e-4 or y + ri > 1 + 1e-4:
                        valid = False
                        break
                if valid:
                    for i in range(n):
                        for j in range(i+1, n):
                            d = np.sqrt(np.sum((c_opt[i]-c_opt[j])**2))
                            if d < r_opt[i] + r_opt[j] - 1e-4:
                                valid = False
                                break
                        if not valid: break
                        
            if valid and s > best_sum:
                best_sum = s
                best_centers = c_opt.copy()
                best_radii = r_opt.copy()

    return best_centers, best_radii, best_sum
