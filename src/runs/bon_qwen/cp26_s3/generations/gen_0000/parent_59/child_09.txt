import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # 1. Initialization: Grid Layout
    # Use a 5x6 grid and select the first 26 points
    x_grid = np.linspace(0.15, 0.85, 5)
    y_grid = np.linspace(0.15, 0.85, 6)
    xx, yy = np.meshgrid(x_grid, y_grid)
    centers_init = np.vstack([xx.flatten(), yy.flatten()]).T[:26]
    radii_init = np.full(n, 0.01)
    
    # 2. Physics-based Repulsion (Relaxation)
    # This phase moves centers apart to resolve initial overlaps and find a natural spacing.
    current_centers = centers_init.copy()
    # Run 500 iterations of repulsion forces
    for _ in range(500):
        forces_x = np.zeros(n)
        forces_y = np.zeros(n)
        
        # Pairwise repulsion
        for i in range(n):
            for j in range(i + 1, n):
                dx = current_centers[j, 0] - current_centers[i, 0]
                dy = current_centers[j, 1] - current_centers[i, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist > 0:
                    # Force magnitude inversely proportional to distance squared
                    f_mag = 0.005 / (dist**2)
                    fx = f_mag * dx
                    fy = f_mag * dy
                    forces_x[i] -= fx
                    forces_y[i] -= fy
                    forces_x[j] += fx
                    forces_y[j] += fy
        
        # Boundary repulsion (keep centers inside)
        for i in range(n):
            x, y = current_centers[i]
            if x < 0.05: forces_x[i] += 0.01
            if x > 0.95: forces_x[i] -= 0.01
            if y < 0.05: forces_y[i] += 0.01
            if y > 0.95: forces_y[i] -= 0.01
            
        current_centers[:, 0] += forces_x * 0.5
        current_centers[:, 1] += forces_y * 0.5
        
        # Clamp to valid range [0, 1]
        current_centers[:, 0] = np.clip(current_centers[:, 0], 0.0, 1.0)
        current_centers[:, 1] = np.clip(current_centers[:, 1], 0.0, 1.0)

    # 3. Numerical Optimization
    # Flatten centers and radii into a single vector
    x0 = np.concatenate([current_centers.flatten(), radii_init])
    
    # Define bounds: x, y in [0, 1], r in [0, 0.5]
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n

    # Objective: Minimize negative sum of radii
    def objective(v):
        radii = v[2 * n:]
        return -np.sum(radii)

    # Define constraints for the optimizer
    constraints = []

    # Boundary constraints: r <= x, r <= 1-x, r <= y, r <= 1-y
    for i in range(n):
        idx = i * 2
        # x - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i, idx=idx: v[idx] - v[2 * n + i]
        })
        # 1 - x - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i, idx=idx: 1.0 - v[idx] - v[2 * n + i]
        })
        idx_y = i * 2 + 1
        # y - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i, idx=idx_y: v[idx_y] - v[2 * n + i]
        })
        # 1 - y - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i, idx=idx_y: 1.0 - v[idx_y] - v[2 * n + i]
        })

    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    for i in range(n):
        for j in range(i + 1, n):
            # Variables indices
            xi, yi = 2 * i, 2 * i + 1
            xj, yj = 2 * j, 2 * j + 1
            ri, rj = 2 * n + i, 2 * n + j
            
            constraints.append({
                'type': 'ineq',
                'fun': lambda v, xi=xi, yi=yi, xj=xj, yj=yj, ri=ri, rj=rj: \
                       (v[xi] - v[xj])**2 + (v[yi] - v[yj])**2 - (v[ri] + v[rj])**2
            })

    # Run optimizer
    # SLSQP is chosen for handling non-linear constraints and bounds
    res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints, 
                   options={'maxiter': 1000, 'disp': False})

    # Extract final results
    final_centers = res.x[:2 * n].reshape((n, 2))
    final_radii = res.x[2 * n:]
    final_sum = np.sum(final_radii)

    return final_centers, final_radii, float(final_sum)
