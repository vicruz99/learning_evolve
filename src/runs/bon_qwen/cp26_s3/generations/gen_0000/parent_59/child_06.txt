def compute_penalty(params, n):
    """Compute constraint violation penalty."""
    centers = params[:2 * n].reshape(n, 2)
    radii = params[2 * n:]
    
    penalty = 0.0
    
    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x < r:
            penalty += (r - x) ** 2
        if x + r > 1:
            penalty += (x + r - 1) ** 2
        if y < r:
            penalty += (r - y) ** 2
        if y + r > 1:
            penalty += (y + r - 1) ** 2
    
    # Non-overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            if dist < radii[i] + radii[j]:
                penalty += (radii[i] + radii[j] - dist) ** 2
    
    return penalty


def objective(params, n, penalty_weight):
    """Objective: maximize sum of radii with penalty for violations."""
    radii = params[2 * n:]

    return -np.sum(radii) + penalty_weight * penalty


def is_valid(centers, radii):
