import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    
    Returns:
        centers: np.array of shape (26, 2)
        radii: np.array of shape (26,)
        sum_radii: float
    """
    n = 26
    
    # Helper function to generate a hexagonal lattice initialization
    def get_initial_guess(n_circles):
        # Estimate a radius for initialization (slightly less than optimal)
        # For 26 circles, r is roughly 0.1. We start with 0.08 to be safe.
        r_init = 0.08
        
        # Generate hex grid points
        points = []
        y = r_init
        row_idx = 0
        
        while len(points) < n_circles:
            # Calculate x offset for hexagonal shift
            x_start = r_init if row_idx % 2 == 0 else r_init + r_init * np.sqrt(3)
            
            x = x_start
            while x <= 1 - r_init:
                points.append((x, y))
                x += 2 * r_init
            
            y += r_init * np.sqrt(3)
            row_idx += 1
            
        # If we have more points than needed, trim them
        points = points[:n_circles]
        
        # If we have fewer (unlikely with this spacing), pad with random
        while len(points) < n_circles:
            # Random point avoiding existing
            while True:
                p = np.random.rand(2)
                valid = True
                for ep in points:
                    if np.linalg.norm(np.array(p) - np.array(ep)) < 0.15:
                        valid = False
                        break
                if valid:
                    points.append(p)
                    break
                    
        return np.array(points), np.full(n_circles, r_init)

    # Generate initial centers and radii
    centers_init, radii_init = get_initial_guess(n)
    
    # Flatten variables for optimizer: [x0, y0, r0, x1, y1, r1, ...]
    x0 = np.zeros(3 * n)
    for i in range(n):
        x0[3*i] = centers_init[i, 0]
        x0[3*i+1] = centers_init[i, 1]
        x0[3*i+2] = radii_init[i]
        
    # Bounds for variables: x, y in [0, 1], r in [0, 0.5]
    bounds = []
    for i in range(n):
        bounds.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
        
    # Constraints
    cons = []
    
    # 1. Boundary constraints
    # x - r >= 0  => x - r >= 0
    # 1 - x - r >= 0
    # y - r >= 0
    # 1 - y - r >= 0
    for i in range(n):
        cons.append({'type': 'ineq', 'fun': lambda v, i=i: v[3*i] - v[3*i+2]})
        cons.append({'type': 'ineq', 'fun': lambda v, i=i: 1.0 - v[3*i] - v[3*i+2]})
        cons.append({'type': 'ineq', 'fun': lambda v, i=i: v[3*i+1] - v[3*i+2]})
        cons.append({'type': 'ineq', 'fun': lambda v, i=i: 1.0 - v[3*i+1] - v[3*i+2]})
        
    # 2. Non-overlap constraints
    # (xi - xj)^2 + (yi - yj)^2 >= (ri + rj)^2
    for i in range(n):
        for j in range(i + 1, n):
            cons.append({'type': 'ineq', 'fun': lambda v, i=i, j=j: 
                (v[3*i] - v[3*j])**2 + (v[3*i+1] - v[3*j+1])**2 - (v[3*i+2] + v[3*j+2])**2
            })
            
    # Objective function: maximize sum(r) <=> minimize -sum(r)
    def objective(v):
        return -sum(v[3*i+2] for i in range(n))
        
    # Run optimization
    # SLSQP is suitable for this type of problem
    res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, 
                   options={'ftol': 1e-9, 'maxiter': 2000, 'disp': False})
    
    if res.success or res.nit > 0:
        final_centers = np.zeros((n, 2))
        final_radii = np.zeros(n)
        for i in range(n):
            final_centers[i] = (res.x[3*i], res.x[3*i+1])
            final_radii[i] = res.x[3*i+2]
        return final_centers, final_radii, np.sum(final_radii)
    
    # Fallback to initial guess if optimization fails
    return centers_init, radii_init, np.sum(radii_init)
