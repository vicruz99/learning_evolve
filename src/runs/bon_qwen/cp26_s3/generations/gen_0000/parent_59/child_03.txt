   - Non-overlap constraints (distance between centers ≥ sum of radii)
   - Non-negative radii

3. **Multi-start strategy**: Run optimization from multiple initial configurations (different grid arrangements) and keep the best result to avoid local minima.

4. **Refinement**: After the main optimization, do additional refinement passes to squeeze out more radius sum.

- Hexagonal packing is theoretically optimal for equal circles
- Allowing different radii can improve the total sum (corners allow slightly larger circles)

import numpy as np
from scipy.optimize import minimize


def init_hexagonal_grid(n):
    """Initialize circles on a hexagonal grid pattern."""
    points = []
    spacing = 0.35

    for row in range(10):
        for col in range(10):
            x = col * spacing
            y = row * spacing * np.sqrt(3) / 2
            if row % 2 == 1:
                x += spacing / 2
            points.append([x, y])
            if len(points) >= n:
                break
        if len(points) >= n:
            break

    points = np.array(points[:n])

    x_min, x_max = points[:, 0].min(), points[:, 0].max()
    y_min, y_max = points[:, 1].min(), points[:, 1].max()

    margin = 0.15
    x_range = x_max - x_min
    y_range = y_max - y_min
    scale_x = (1 - 2 * margin) / x_range if x_range > 0 else 1
    scale_y = (1 - 2 * margin) / y_range if y_range > 0 else 1
    scale = min(scale_x, scale_y)

    points[:, 0] = (points[:, 0] - x_min) * scale + margin
    points[:, 1] = (points[:, 1] - y_min) * scale + margin

    radii = np.ones(n) * 0.07

    return points, radii


def init_alternating_grid(n):
    """Initialize with alternating row sizes for better packing."""
    points = []
    spacing = 0.22

    n_cols_even = 6
    n_cols_odd = 5

    row = 0
    while len(points) < n:
        n_cols = n_cols_even if row % 2 == 0 else n_cols_odd
        for col in range(n_cols):
            x = col * spacing + (spacing / 2 if row % 2 == 1 else 0)
            y = row * spacing * np.sqrt(3) / 2
            points.append([x, y])
            if len(points) >= n:
                break
        row += 1

    points = np.array(points[:n])

    x_min, x_max = points[:, 0].min(), points[:, 0].max()
    y_min, y_max = points[:, 1].min(), points[:, 1].max()

    margin = 0.1
    x_range = x_max - x_min
    y_range = y_max - y_min
    scale_x = (1 - 2 * margin) / x_range if x_range > 0 else 1
    scale_y = (1 - 2 * margin) / y_range if y_range > 0 else 1
    scale = min(scale_x, scale_y)

    points[:, 0] = (points[:, 0] - x_min) * scale + margin
    points[:, 1] = (points[:, 1] - y_min) * scale + margin

    radii = np.ones(n) * 0.09

    return points, radii


def init_random_grid(n):
    """Initialize with perturbed grid for diversity."""
    centers, radii = init_hexagonal_grid(n)
    centers += np.random.randn(n, 2) * 0.02
    centers[:, 0] = np.clip(centers[:, 0], 0.05, 0.95)
    centers[:, 1] = np.clip(centers[:, 1], 0.05, 0.95)
    return centers, radii


def get_all_constraints(x, n):
    """Compute all constraint values for the optimization."""
    centers = x[:2 * n].reshape(n, 2)
    radii = x[2 * n:]

    constraints = []

    # Boundary constraints: circle i must be inside unit square
    for i in range(n):
        constraints.append(x[2 * i] - x[2 * n + i])
        constraints.append(1.0 - x[2 * i] - x[2 * n + i])
        constraints.append(x[2 * i + 1] - x[2 * n + i])
        constraints.append(1.0 - x[2 * i + 1] - x[2 * n + i])

    # Non-overlap constraints: distance between centers >= sum of radii
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            constraints.append(dist - radii[i] - radii[j])

    return np.array(constraints)


def packing_objective(x, n):
    """Objective: maximize sum of radii (minimize negative sum)."""
    radii = x[2 * n:]
    return -np.sum(radii)


def constraint_wrapper(x, n):
    """Wrapper for constraint function to pass to optimizer."""
    return get_all_constraints(x, n)


def run_single_optimization(centers, radii, n, maxiter=3000):
    """Run a single optimization from given initial configuration."""
    x0 = np.concatenate([centers.flatten(), radii])

    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n

    constraints = {'type': 'ineq', 'fun': constraint_wrapper, 'args': (n,)}

    result = minimize(
        packing_objective,
        x0,
        args=(n,),
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': maxiter, 'ftol': 1e-15, 'disp': False}
    )

    centers_opt = result.x[:2 * n].reshape(n, 2)
    radii_opt = result.x[2 * n:]
    radii_opt = np.maximum(radii_opt, 0)

    return centers_opt, radii_opt, -result.fun


def fix_constraint_violations(centers, radii, n, iterations=50):
    """Iteratively fix constraint violations using penalty method."""
    centers = centers.copy()
    radii = radii.copy()

    for it in range(iterations):
        penalty = 1000 + 100 * it
        valid = True

        # Fix boundary violations
        for i in range(n):
            x, y = centers[i]
            r = radii[i]

            if x - r < 0:
                centers[i, 0] = r
            if x + r > 1:
                centers[i, 0] = 1 - r
            if y - r < 0:
                centers[i, 1] = r
            if y + r > 1:
                centers[i, 1] = 1 - r

        # Push apart overlapping circles
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[j, 0] - centers[i, 0]
                dy = centers[j, 1] - centers[i, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                r_sum = radii[i] + radii[j]

                if dist < r_sum and dist > 1e-12:
                    overlap = r_sum - dist
                    push = overlap / 2 + 0.0001
                    nx = dx / dist
                    ny = dy / dist
                    centers[i, 0] -= nx * push * 0.5
                    centers[i, 1] -= ny * push * 0.5
                    centers[j, 0] += nx * push * 0.5
                    centers[j, 1] += ny * push * 0.5
                    valid = False

        # Clip to valid region
        for i in range(n):
            centers[i, 0] = np.clip(centers[i, 0], radii[i], 1 - radii[i])
            centers[i, 1] = np.clip(centers[i, 1], radii[i], 1 - radii[i])

        if valid:
            break

    return centers, radii


def run_packing():
    n = 26
    best_sum = -1
    best_centers = None
    best_radii = None

    # Try multiple initial configurations
    initial_configs = [
        init_hexagonal_grid(n),
        init_alternating_grid(n),
        init_random_grid(n),
    ]

    for centers_init, radii_init in initial_configs:
        centers_opt, radii_opt, current_sum = run_single_optimization(
            centers_init, radii_init, n, maxiter=5000
        )

        # Fix any remaining constraint violations
        centers_opt, radii_opt = fix_constraint_violations(centers_opt, radii_opt, n)

        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers_opt.copy()
            best_radii = radii_opt.copy()

    # Try to squeeze more by scaling up radii slightly and re-optimizing
    for attempt in range(3):
        radii_scaled = best_radii * 1.02

            best_centers, radii_scaled, n, maxiter=5000
        )
        centers_temp, radii_temp = fix_constraint_violations(centers_temp, radii_temp, n)
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers_temp.copy()
            best_radii = radii_temp.copy()

    # Final cleanup: ensure all constraints are satisfied
    best_centers, best_radii = fix_constraint_violations(best_centers, best_radii, n, iterations=200)

    # Recompute sum after fixes
    best_sum = np.sum(best_radii)

    return best_centers, best_radii, best_sum
