import numpy as np
from scipy.optimize import minimize

# Global constant for the number of circles
N_CIRCLES = 26

def objective(x):
    """
    Objective function to minimize: -sum(radii).
    x is a flat array of shape (3*N,): [x1, y1, r1, x2, y2, r2, ...]
    """
    # Radii are at indices 2, 5, 8, ...
    radii = x[2::3]
    return -np.sum(radii)

def constraints_fun(x):
    """
    Computes all inequality constraints.
    Returns a 1D numpy array of constraint values (all must be >= 0).
    """
    # Reshape x to (N, 3) where columns are x, y, r
    points = x.reshape(N_CIRCLES, 3)
    centers = points[:, :2]
    radii = points[:, 2]
    
    all_constraints = []
    
    # 1. Boundary constraints
    # x_i >= r_i  => x_i - r_i >= 0
    all_constraints.append(centers[:, 0] - radii)
    # 1 - x_i >= r_i => 1 - x_i - r_i >= 0
    all_constraints.append(1.0 - centers[:, 0] - radii)
    # y_i >= r_i
    all_constraints.append(centers[:, 1] - radii)
    # 1 - y_i >= r_i
    all_constraints.append(1.0 - centers[:, 1] - radii)
    
    # 2. Pairwise non-overlap constraints
    # (x_i - x_j)^2 + (y_i - y_j)^2 >= (r_i + r_j)^2
    # dist_sq - r_sum_sq >= 0
    
    # Get indices for upper triangle of pairwise comparisons
    i_indices, j_indices = np.triu_indices(N_CIRCLES, k=1)

    diff = centers[i_indices] - centers[j_indices]
    dist_sq = np.sum(diff**2, axis=1)
    
    # Sum of radii squared
    r_sum = radii[i_indices] + radii[j_indices]
    r_sum_sq = r_sum**2
    
    pairwise_constraints = dist_sq - r_sum_sq
    all_constraints.append(pairwise_constraints)
    
    # Concatenate all constraints into a single array
    return np.concatenate(all_constraints)

def resolve_conflicts(x, n, max_iter=200):
    """
    Iteratively adjusts centers to resolve overlaps and boundary violations.
    Ensures the initial point is feasible for the optimizer.
    """
    points = x.reshape(n, 3)
    centers = points[:, :2].copy()
    radii = points[:, 2].copy()
    
    for _ in range(max_iter):
        changed = False
        
        # Fix boundary violations
        for i in range(n):
            r = radii[i]
            c = centers[i]
            
            # x boundary
            if c[0] < r:
                centers[i, 0] = r
                changed = True
            elif c[0] > 1 - r:
                centers[i, 0] = 1 - r
                changed = True
            
            # y boundary
            if c[1] < r:
                centers[i, 1] = r
                changed = True
            elif c[1] > 1 - r:
                centers[i, 1] = 1 - r
                changed = True
        
        # Fix overlaps
        for i in range(n):
            for j in range(i + 1, n):
                d_vec = centers[i] - centers[j]
                dist = np.sqrt(np.sum(d_vec**2))
                r_sum = radii[i] + radii[j]
                
                if dist < r_sum and dist > 1e-9:
                    # Push apart along the line connecting centers
                    overlap = r_sum - dist
                    # Normalize direction vector
                    direction = d_vec / dist
                    # Move each circle half the overlap distance
                    move = direction * (overlap / 2.0)
                    centers[i] -= move
                    centers[j] += move
                    changed = True
        
        if not changed:
            break
    
    # Reconstruct flat array
    new_x = np.zeros(3 * n)
    for i in range(n):
        new_x[3 * i] = centers[i, 0]
        new_x[3 * i + 1] = centers[i, 1]
        new_x[3 * i + 2] = radii[i]
        
    return new_x

def run_packing():
    n = N_CIRCLES
    best_x = None
    best_obj = np.inf  # We are minimizing negative sum, so lower is better

    # Optimization constraints
    constraints = {'type': 'ineq', 'fun': constraints_fun}
    
    # Multi-start optimization
    num_restarts = 10
    
    for restart in range(num_restarts):

        np.random.seed(restart) # Use seed for reproducibility if needed, or remove for true randomness

            x0[3 * i + 2] = radii_init[i]
        
        # Resolve conflicts to ensure feasible start
        x0_feasible = resolve_conflicts(x0, n, max_iter=100)
        
        # Run optimization

            res = minimize(
                objective,
                x0_feasible,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 1000, 'ftol': 1e-12, 'disp': False}
            )

                best_obj = res.fun

            continue
