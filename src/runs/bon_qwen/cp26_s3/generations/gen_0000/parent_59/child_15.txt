import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # --- 1. Generate a robust initial configuration (Hexagonal Grid) ---
    # We want to fit 26 circles. A 5x5 grid fits 25. 
    # We can arrange them in rows. 5 rows of roughly 5 circles.
    # Hexagonal packing is denser.
    
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.095) # Initial guess slightly smaller than 0.1 to allow optimization
    
    # Arrange in 5 rows
    # Row heights in hexagonal packing are r * sqrt(3)
    # Let's try to fit 5 rows
    # To fit 26, maybe 6, 5, 5, 5, 5? Or 5, 5, 5, 5, 6?
    # Let's distribute counts to maximize space.
    # Rows: 5, 6, 5, 6, 4? (Total 26)
    # Or 5, 5, 6, 5, 5?
    
    row_counts = [5, 6, 5, 6, 4] # Sum = 26
    # Actually, alternating row counts in hex packing allows offsetting.
    # Let's just place them in a grid first, optimization will fix it.
    
    idx = 0
    num_rows = 5
    # Calculate initial r based on 5 rows hexagonal
    # Height constraint: 2r + (num_rows-1)*r*sqrt(3) <= 1
    # r(2 + 4*1.732) <= 1 => r(8.928) <= 1 => r <= 0.112
    # Width constraint for row with 6 circles: 2r + 5*2r <= 1 => 12r <= 1 => r <= 0.0833
    # So 6 circles in a row is restrictive. 
    # Let's use a configuration that avoids long rows if possible, or just let optimizer fix.
    # A 6x5 grid is 30 circles. We remove 4.
    
    # Let's create a dense 6x5 grid (30 circles) and remove 4 from corners/centers to make 26
    # Grid step
    # If we have 6 columns, width 1. r ~ 1/12 = 0.083.
    # If we have 5 columns, width 1. r ~ 0.1.
    
    # Better initial guess: 5x5 grid (25 circles) + 1 small circle in a gap?
    # But we need to optimize all radii.
    
    # Let's try a perturbed 5x5 grid plus one extra circle.
    # 5x5 grid centers: 0.1, 0.3, 0.5, 0.7, 0.9
    # This leaves gaps.
    
    grid_vals = np.linspace(0.1, 0.9, 5)
    
    count = 0
    for i in range(5):
        for j in range(5):
            if count < n:
                centers[count] = [grid_vals[i], grid_vals[j]]
                radii[count] = 0.1
                count += 1
    
    # Place the 26th circle in the center gap (0.2, 0.2) with small radius
    if count < n:
        centers[count] = [0.2, 0.2] # Gap center
        radii[count] = 0.05
        count += 1
        
    # If we still need more (though 25+1=26), fill remaining
    while count < n:
        centers[count] = [0.5, 0.5]
        radii[count] = 0.01
        count += 1

    # --- 2. Define Objective and Constraints ---
    
    def objective(x):
        # x contains [x1, y1, r1, x2, y2, r2, ...]
        # We want to maximize sum(r), so minimize -sum(r)
        radii_part = x[2::3]
        return -np.sum(radii_part)

    def constraints_factory(n):
        constraints = []
        
        # Boundary constraints: x >= r, x <= 1-r, y >= r, y <= 1-r
        # x[i*3] >= x[i*3+2] => x[i*3] - x[i*3+2] >= 0
        # x[i*3] <= 1 - x[i*3+2] => x[i*3] + x[i*3+2] - 1 <= 0 => 1 - x[i*3] - x[i*3+2] >= 0
        # Same for y (index +1)
        
        for i in range(n):
            idx_x = i * 3
            idx_y = i * 3 + 1
            idx_r = i * 3 + 2
            
            # x - r >= 0
            constraints.append({'type': 'ineq', 'fun': lambda x, i=i: x[idx_x] - x[idx_r]})
            # x + r <= 1  => 1 - x - r >= 0
            constraints.append({'type': 'ineq', 'fun': lambda x, i=i: 1.0 - x[idx_x] - x[idx_r]})
            # y - r >= 0
            constraints.append({'type': 'ineq', 'fun': lambda x, i=i: x[idx_y] - x[idx_r]})
            # y + r <= 1  => 1 - y - r >= 0
            constraints.append({'type': 'ineq', 'fun': lambda x, i=i: 1.0 - x[idx_y] - x[idx_r]})
            
            # r >= 0 (handled by bounds)
            
        # Non-overlap constraints: dist >= r_i + r_j
        # dist^2 >= (r_i + r_j)^2
        # (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
        
        for i in range(n):
            for j in range(i + 1, n):
                idx_xi, idx_yi, idx_ri = i * 3, i * 3 + 1, i * 3 + 2
                idx_xj, idx_yj, idx_rj = j * 3, j * 3 + 1, j * 3 + 2
                
                def non_overlap(x, i=i, j=j):
                    dx = x[idx_xi] - x[idx_xj]
                    dy = x[idx_yi] - x[idx_yj]
                    dr = x[idx_ri] + x[idx_rj]
                    return (dx**2 + dy**2) - dr**2
                
                constraints.append({'type': 'ineq', 'fun': non_overlap})
                
        return constraints

    constraints = constraints_factory(n)

    # Bounds for variables: x in [0,1], y in [0,1], r in [0, 0.5]
    bounds = [(0, 1) for _ in range(n)] + [(0, 1) for _ in range(n)] + [(0, 0.5) for _ in range(n)]

    # Flatten initial guess
    x0 = np.zeros(3 * n)
    for i in range(n):
        x0[3*i] = centers[i, 0]
        x0[3*i+1] = centers[i, 1]
        x0[3*i+2] = radii[i]

    # --- 3. Optimization ---
    # SLSQP is a good choice for constrained non-linear optimization
    
    # Run multiple times to avoid local minima
    best_result = None
    min_obj = np.inf
    
    # We can try a few random perturbations of the initial grid
    for attempt in range(10):
        # Perturb x0
        x0_perturbed = x0.copy()
        # Add small noise to positions
        x0_perturbed[:2*n] += np.random.uniform(-0.01, 0.01, 2*n)
        # Clip positions
        x0_perturbed[:2*n] = np.clip(x0_perturbed[:2*n], 0.001, 0.999)
        
        try:
            res = minimize(objective, x0_perturbed, method='SLSQP', bounds=bounds, constraints=constraints,
                           options={'maxiter': 1000, 'ftol': 1e-12})
            if res.success and res.fun < min_obj:
                min_obj = res.fun
                best_result = res
        except Exception:
            continue
            
    if best_result is None:
        # Fallback to unoptimized or previous best
        return centers, radii, np.sum(radii)

    # Extract solution
    final_x = best_result.x
    final_centers = np.zeros((n, 2))
    final_radii = np.zeros(n)
    
    for i in range(n):
        final_centers[i, 0] = final_x[3*i]
        final_centers[i, 1] = final_x[3*i+1]
        final_radii[i] = final_x[3*i+2]
        
    # Ensure validity (numerical checks)
    # The optimizer might hit boundaries exactly.
    
    sum_radii = np.sum(final_radii)
    return final_centers, final_radii, sum_radii
