from scipy.optimize import minimize

def run_packing():

    Pack 26 circles in a unit square to maximize the sum of radii.
    Uses a hexagonal initialization followed by SLSQP optimization.
    """
    n = 26

        return (p1[0] - p2[0])**2 + (p1[1] - p2[1])**2

    def create_initial_params():
        """
        Creates initial centers and radii for 26 circles.
        Uses a hexagonal packing layout.
        """
        centers = np.zeros((n, 2))
        radii = np.full(n, 0.09) # Start with a safe radius
        
        # Arrange in hexagonal rows
        # Row counts: 5, 5, 5, 5, 5, 1
        row_counts = [5, 5, 5, 5, 5, 1]
        
        y_pos = 0.1
        row_idx = 0
        
        for r_count in row_counts:
            # Horizontal spacing in hex packing is 2*r
            # But since r varies, we use a fixed spacing for init based on initial r
            # Initial r = 0.09 -> spacing = 0.18
            spacing = 0.19 
            
            # Calculate starting x to center the row
            total_width = (r_count - 1) * spacing
            start_x = (1.0 - total_width) / 2
            
            for i in range(r_count):
                idx = row_idx + i
                if idx >= n:
                    break
                x_pos = start_x + i * spacing
                centers[idx] = [x_pos, y_pos]
            
            row_idx += r_count
            # Vertical spacing in hex packing is sqrt(3)/2 * 2r = sqrt(3)*r
            y_pos += np.sqrt(3) * 0.09
            
        # Flatten parameters: [x0, y0, r0, x1, y1, r1, ...]
        params = np.zeros(3 * n)
        for i in range(n):
            params[3*i] = centers[i, 0]
            params[3*i + 1] = centers[i, 1]
            params[3*i + 2] = radii[i]
            
        return params

    def get_bounds():
        """
        Bounds for each variable:
        x in [0, 1]
        y in [0, 1]
        r in [0.001, 0.5] (lower bound > 0 to avoid singularities)
        """
        bounds = []
        for _ in range(n):
            bounds.append((0.0, 1.0)) # x
            bounds.append((0.0, 1.0)) # y
            bounds.append((1e-4, 0.5)) # r
        return bounds

    def objective(params):
        """
        Objective function: Minimize negative sum of radii.
        """
        radii = params[2::3]
        return -np.sum(radii)

    def constraint_bounds(params):
        """
        Returns array of constraint values. 
        All constraints must be >= 0.
        Includes:
        1. Distance from left wall: x - r >= 0
        2. Distance from right wall: (1 - x) - r >= 0
        3. Distance from bottom wall: y - r >= 0
        4. Distance from top wall: (1 - y) - r >= 0
        """
        cons = []
        for i in range(n):
            x = params[3*i]
            y = params[3*i + 1]
            r = params[3*i + 2]
            
            # Left wall
            cons.append(x - r)
            # Right wall
            cons.append((1.0 - x) - r)
            # Bottom wall
            cons.append(y - r)
            # Top wall
            cons.append((1.0 - y) - r)
        return np.array(cons)

    def constraint_non_overlap(params):
        """
        Returns array of constraint values for non-overlapping circles.
        dist(c_i, c_j) - (r_i + r_j) >= 0
        """
        cons = []
        for i in range(n):
            for j in range(i + 1, n):
                x1, y1, r1 = params[3*i], params[3*i+1], params[3*i+2]
                x2, y2, r2 = params[3*j], params[3*j+1], params[3*j+2]
                
                dist = np.sqrt((x1 - x2)**2 + (y1 - y2)**2)
                cons.append(dist - (r1 + r2))
        return np.array(cons)

    # Initial guess
    x0 = create_initial_params()

    bounds = get_bounds()
    
    cons = []
    
    # Wall constraints
    cons.append({
        'type': 'ineq',
        'fun': constraint_bounds
    })
    
    # Non-overlap constraints
    cons.append({
        'type': 'ineq',
        'fun': constraint_non_overlap
    })

    # Run optimizer
    # SLSQP is suitable for non-linear problems with constraints
    result = minimize(
        objective, 
        x0, 
        method='SLSQP', 
        bounds=bounds, 
        constraints=cons,
        options={'maxiter': 1000, 'ftol': 1e-9}
    )
    
    # Extract results

    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    for i in range(n):
        centers[i, 0] = final_params[3*i]
        centers[i, 1] = final_params[3*i + 1]
        radii[i] = final_params[3*i + 2]
        
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
