import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Pack 26 circles in a unit square [0,1]x[0,1] to maximize the sum of radii.
    
    Returns:
        centers: np.array of shape (26, 2)
        radii: np.array of shape (26,)
        sum_radii: float
    """
    n = 26
    
    # 1. Initialization
    # We initialize centers on a grid to ensure they are separated, 
    # with small radii to satisfy initial constraints.
    # A 5x5 grid gives 25 points. We add one more point slightly perturbed 
    # or place them in a dense pattern.
    
    centers_init = np.zeros((n, 2))
    radii_init = np.ones(n) * 0.05 # Start with small radii
    
    # Create a grid of points
    # We can fit roughly 5 circles per side with radius ~0.1.
    # Let's generate a 6x5 grid (30 points) and pick 26.
    # Or just a dense random initialization. Grid is more deterministic.
    
    # Let's use a hexagonal-like grid or just a dense rectangular grid
    # Grid spacing approx 0.2
    x_coords = np.linspace(0.1, 0.9, 6) # 6 points
    y_coords = np.linspace(0.1, 0.9, 5) # 5 points
    
    idx = 0
    for y in y_coords:
        for x in x_coords:
            if idx < n:
                centers_init[idx] = [x, y]
                idx += 1
            else:
                break
        if idx >= n:
            break
            
    # Flatten variables: [x0, y0, r0, x1, y1, r1, ...]
    # Actually, let's keep structure clear.
    # vars layout: x (n), y (n), r (n)
    # Total 3n variables
    x0 = np.concatenate([centers_init[:, 0], centers_init[:, 1], radii_init])
    
    # Bounds for variables
    # x, y in [0, 1]
    # r in [0, 0.5]
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)] * n + [(0.0, 0.5)] * n

    def objective(vars):
        # Maximize sum of radii -> Minimize negative sum
        radii = vars[2*n:]
        return -np.sum(radii)

    def constraints(vars):
        # Extract variables
        xs = vars[:n]
        ys = vars[n:2*n]
        rs = vars[2*n:]
        
        c = []
        
        # Boundary constraints:
        # x >= r  => x - r >= 0
        # 1-x >= r => 1 - x - r >= 0
        # Same for y
        for i in range(n):
            c.append(xs[i] - rs[i])
            c.append(1.0 - xs[i] - rs[i])
            c.append(ys[i] - rs[i])
            c.append(1.0 - ys[i] - rs[i])
            
        # Non-overlap constraints:
        # dist_sq >= (r_i + r_j)^2
        # dist_sq - (r_i + r_j)^2 >= 0
        # We compute this for all pairs i < j
        for i in range(n):
            for j in range(i + 1, n):
                dx = xs[i] - xs[j]
                dy = ys[i] - ys[j]
                dist_sq = dx*dx + dy*dy
                r_sum = rs[i] + rs[j]
                c.append(dist_sq - r_sum*r_sum)
                
        return np.array(c)

    # Define constraints for scipy
    # We pass the function, but scipy needs to know it returns inequality constraints (>= 0)
    # By default, inequality constraints are g(x) >= 0.
    cons = {'type': 'ineq', 'fun': constraints}
    
    # Run optimization
    # maxiter might need to be high for convergence
    res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': 2000, 'ftol': 1e-10, 'disp': False})
    
    # Check if optimization was successful or at least converged
    # We take the result regardless, assuming it improved from initial guess
    
    # Reshape results
    final_xs = res.x[:n]
    final_ys = res.x[n:2*n]
    final_rs = res.x[2*n:]
    
    final_centers = np.column_stack((final_xs, final_ys))
    final_radii = final_rs
    
    # Calculate sum
    sum_r = np.sum(final_radii)
    
    # Return results
    return final_centers, final_radii, sum_r
