import numpy as np
from scipy.optimize import minimize

def compute_constraints(x, n):
    """
    Computes the vector of constraint values for the circle packing problem.
    Returns a vector where each element must be >= 0.
    x is a flattened array of [x1, y1, r1, x2, y2, r2, ...]
    """
    # Extract centers and radii
    centers = x[:2*n].reshape((n, 2))
    radii = x[2*n:]
    
    constraints = []
    
    # 1. Boundary constraints
    # Ensure circles are inside the unit square [0, 1] x [0, 1]
    # x - r >= 0
    constraints.append(centers[:, 0] - radii)
    # 1 - x - r >= 0
    constraints.append(1.0 - centers[:, 0] - radii)
    # y - r >= 0
    constraints.append(centers[:, 1] - radii)
    # 1 - y - r >= 0
    constraints.append(1.0 - centers[:, 1] - radii)
    
    # 2. Overlap constraints
    # Ensure distance between centers is at least sum of radii
    # dist^2 >= (r_i + r_j)^2
    
    # Compute squared Euclidean distance matrix (n x n)
    # diff has shape (n, n, 2)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist_sq = np.sum(diff**2, axis=2)
    
    # Compute (r_i + r_j)^2 matrix
    r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    r_sum_sq = r_sum**2
    
    # Constraint: dist_sq - r_sum_sq >= 0
    overlap_constraints = dist_sq - r_sum_sq
    
    # We only need to enforce this for i < j (upper triangle)
    # Create a mask for the upper triangle
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    constraints.append(overlap_constraints[mask])
    
    # Concatenate all constraints into a single array
    return np.concatenate(constraints)

def objective_func(x, n):
    """
    Objective function to minimize: -sum(radii)
    """
    radii = x[2*n:]
    return -np.sum(radii)

def run_packing():
    n = 26
    best_x = None
    best_sum = -1.0
    
    # Bounds for variables
    # x_i, y_i in [0, 1]
    # r_i in [0, 0.5] (radius cannot exceed half the square side)
    bounds = [(0.0, 1.0) for _ in range(2 * n)] + [(1e-6, 0.5) for _ in range(n)]
    
    rng = np.random.RandomState(42)
    
    start_points = []
    
    # Strategy 1: Random Starts
    # Place centers randomly in the middle of the square with small radii
    # to ensure initial feasibility (mostly)
    for _ in range(50):
        centers = rng.uniform(0.1, 0.9, size=(n, 2))
        radii = np.full(n, 0.02) # Small initial radius
        x0 = np.concatenate([centers.flatten(), radii])
        start_points.append(x0)
        
    # Strategy 2: Grid Starts
    # A perturbed grid is a good local starting point
    grid_centers = []
    # Try to fit in a 5x5 or 6x5 pattern
    # 5 rows, 6 cols = 30 points, take first 26
    for r in range(5):
        for c in range(6):
            if len(grid_centers) >= n:
                break
            # Place on a coarse grid
            cx = 0.1 + c * 0.15
            cy = 0.1 + r * 0.2
            grid_centers.append([cx, cy])
    
    if len(grid_centers) < n:
        # Fill remaining if any (shouldn't happen with 30 capacity)
        pass
        
    centers_grid = np.array(grid_centers[:n])
    # Add noise to break symmetry
    centers_grid += rng.normal(0, 0.01, size=centers_grid.shape)
    centers_grid = np.clip(centers_grid, 0.05, 0.95)
    radii_grid = np.full(n, 0.02)
    x0_grid = np.concatenate([centers_grid.flatten(), radii_grid])
    start_points.append(x0_grid)
    
    # Strategy 3: Hexagonal Lattice Starts
    # Hexagonal packing is denser than square packing
    hex_centers = []
    y_pos = 0.1
    # Define row structure to sum to 26
    # e.g., 5, 5, 6, 5, 5 = 26
    rows = [5, 5, 6, 5, 5]
    
    for i, count in enumerate(rows):
        # Stagger every other row
        x_start = 0.1 + (i % 2) * 0.08 
        if count == 1:
            x_pos = 0.5
        else:
            # Distribute points evenly in width [0.1, 0.9] approx
            width = 0.8
            step = width / (count - 1)
            for k in range(count):
                x_pos = x_start + k * step
                hex_centers.append([x_pos, y_pos])
        # Vertical spacing for hex packing is sqrt(3)/2 * diameter approx
        # Assuming nominal diameter ~0.2, height ~0.173
        y_pos += 0.18 
        if len(hex_centers) >= n:
            break
            
    centers_hex = np.array(hex_centers[:n])
    centers_hex += rng.normal(0, 0.01, size=centers_hex.shape)
    centers_hex = np.clip(centers_hex, 0.05, 0.95)
    radii_hex = np.full(n, 0.02)
    x0_hex = np.concatenate([centers_hex.flatten(), radii_hex])
    start_points.append(x0_hex)

    # Optimization Loop
    for x0 in start_points:
        try:
            res = minimize(
                objective_func,
                x0,
                args=(n,),
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'ineq', 'fun': lambda x, n=n: compute_constraints(x, n)},
                options={'maxiter': 2000, 'ftol': 1e-12, 'disp': False}
            )
            
            # Check if this result is better than the current best
            # res.fun is the value of the objective function (-sum_radii)
            current_sum = -res.fun
            if current_sum > best_sum:
                best_sum = current_sum
                best_x = res.x.copy()
        except Exception:
            continue
            
    if best_x is not None:
        centers = best_x[:2*n].reshape((n, 2))
        radii = best_x[2*n:]
        # Ensure radii are non-negative (though bounds should handle this)
        radii = np.maximum(radii, 0.0)
        return centers, radii, np.sum(radii)
    else:
        # Fallback to a valid random packing if optimization fails
        centers = rng.uniform(0.2, 0.8, (n, 2))
        radii = np.full(n, 0.05)
        return centers, radii, np.sum(radii)
