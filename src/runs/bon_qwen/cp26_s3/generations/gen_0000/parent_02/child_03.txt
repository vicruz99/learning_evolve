import numpy as np
from scipy.optimize import minimize

def objective_func(vars, n):
    """Objective: maximize sum of radii (minimize negative sum)"""
    r = vars[2::3]
    return -np.sum(r)

def constraint_func(vars, n):
    """Constraints: non-overlap and boundary containment"""
    c = vars.reshape(-1, 3)
    cons = []
    # Non-overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            dx = c[i, 0] - c[j, 0]
            dy = c[i, 1] - c[j, 1]
            dr = c[i, 2] + c[j, 2]
            cons.append(dx**2 + dy**2 - dr**2)
    # Boundary constraints
    for i in range(n):
        cons.append(c[i, 0] - c[i, 2])          # x >= r
        cons.append(1 - c[i, 0] - c[i, 2])      # 1-x >= r
        cons.append(c[i, 1] - c[i, 2])          # y >= r
        cons.append(1 - c[i, 1] - c[i, 2])      # 1-y >= r
    return np.array(cons)

def validate_and_adjust(centers, radii, n):

    feasible = True
