import numpy as np
from scipy.optimize import minimize

def get_initial_guess_hex(n, side=1.0):
    """Generate an initial guess using a hexagonal lattice pattern."""
    # Estimate radius for n circles in unit square using area density approximation
    # Area = 1, Density ~ 0.9069 (hex), pi * r^2 * n ~ 0.9
    # r_est = sqrt(0.9 / (pi * n))
    r_est = np.sqrt(0.8 / (np.pi * n)) # Conservative estimate
    spacing = 2 * r_est
    h = spacing * np.sqrt(3) / 2
    
    centers = []
    r = 0
    c = 0
    
    # Try to fill the square
    y = r_est
    while y + r_est <= side and len(centers) < n:
        x = r_est
        offset = (len(centers) % 2) * (spacing / 2)
        current_x = r_est + offset
        while current_x + r_est <= side and len(centers) < n:
            centers.append([current_x, y])
            current_x += spacing
            c += 1
        y += h
    
    # If we didn't get enough circles, just pad or adjust
    while len(centers) < n:
        centers.append([0.5, 0.5]) # Fallback
        
    centers = np.array(centers[:n])
    radii = np.full(n, r_est)
    
    # Flatten to vector: x1, y1, r1, x2, y2, r2, ...
    vars_vec = np.zeros(3 * n)
    for i in range(n):
        vars_vec[3*i] = centers[i, 0]
        vars_vec[3*i+1] = centers[i, 1]
        vars_vec[3*i+2] = radii[i]
        
    return vars_vec

def get_initial_guess_grid(n, side=1.0):
    """Generate an initial guess using a grid pattern."""
    # Try to fit in a sqrt(n) x sqrt(n) grid
    cols = int(np.ceil(np.sqrt(n)))
    rows = int(np.ceil(n / cols))
    
    # Determine spacing
    # We want to fit 'cols' circles in width 'side'
    # (cols) * 2r <= side ? No, distance between centers.
    # With cols circles, we have (cols-1) gaps.
    # Total width = 2r + (cols-1)*2r = 2r*cols.
    # So 2r*cols <= side => r <= side / (2*cols)
    
    r_est = side / (2 * max(cols, rows)) * 0.9 # Slightly smaller to be safe
    
    centers = []
    idx = 0
    for r_idx in range(rows):
        for c_idx in range(cols):
            if idx >= n: break
            x = r_est + c_idx * 2 * r_est
            y = r_est + r_idx * 2 * r_est
            centers.append([x, y])
            idx += 1
        if idx >= n: break
        
    while len(centers) < n:
        centers.append([0.5, 0.5])
        
    centers = np.array(centers[:n])
    radii = np.full(n, r_est)
    
    vars_vec = np.zeros(3 * n)
    for i in range(n):
        vars_vec[3*i] = centers[i, 0]
        vars_vec[3*i+1] = centers[i, 1]
        vars_vec[3*i+2] = radii[i]
        
    return vars_vec

def objective(vars_vec, n):
    """Minimize negative sum of radii."""
    radii = vars_vec[2::3]
    return -np.sum(radii)

def get_constraints(n):
    """Generate constraint functions and their jacobians."""
    constraints = []
    
    # 1. Boundary constraints: x - r >= 0, 1 - x - r >= 0, same for y
    # x_i >= r_i  =>  x_i - r_i >= 0
    # x_i <= 1 - r_i => 1 - x_i - r_i >= 0
    # y_i >= r_i  =>  y_i - r_i >= 0
    # y_i <= 1 - r_i => 1 - y_i - r_i >= 0
    
    for i in range(n):
        idx_x = 3*i
        idx_y = 3*i + 1
        idx_r = 3*i + 2
        
        # x - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: v[idx_x] - v[idx_r],
            'jac': lambda v, i=i: np.zeros(3*n) + 0.0 # Will fill manually? No, lambda captures i
            # Actually passing jac is tricky with lambda in list. 
            # Let's define a helper or use finite diff if needed, but SLSQP supports fun only.
            # Providing 'jac' is better.
        })
        
        # 1 - x - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: 1.0 - v[idx_x] - v[idx_r]
        })
        
        # y - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: v[idx_y] - v[idx_r]
        })
        
        # 1 - y - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: 1.0 - v[idx_y] - v[idx_r]
        })

    # 2. Non-overlap constraints: dist^2 >= (r_i + r_j)^2
    # (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
    for i in range(n):
        for j in range(i + 1, n):
            idx_xi, idx_yi, idx_ri = 3*i, 3*i+1, 3*i+2
            idx_xj, idx_yj, idx_rj = 3*j, 3*j+1, 3*j+2
            
            def fun_overlap(v, i=i, j=j):
                xi, yi, ri = v[idx_xi], v[idx_yi], v[idx_ri]
                xj, yj, rj = v[idx_xj], v[idx_yj], v[idx_rj]
                return (xi - xj)**2 + (yi - yj)**2 - (ri + rj)**2

            # Jacobian for overlap constraint
            def jac_overlap(v, i=i, j=j):
                jac = np.zeros(3*n)
                xi, yi, ri = v[idx_xi], v[idx_yi], v[idx_ri]
                xj, yj, rj = v[idx_xj], v[idx_yj], v[idx_rj]
                
                d_dx = 2 * (xi - xj)
                d_dy = 2 * (yi - yj)
                d_dr = -2 * (ri + rj)
                
                jac[idx_xi] = d_dx
                jac[idx_yi] = d_dy
                jac[idx_ri] = d_dr
                
                jac[idx_xj] = -d_dx
                jac[idx_yj] = -d_dy
                jac[idx_rj] = d_dr # Note: -( -2(r_i+r_j) ) ? 
                # Wait, derivative of -(r_i+r_j)^2 w.r.t r_j is -2(r_i+r_j).
                # Constraint is dist^2 - (r_i+r_j)^2.
                # d/dr_j [ - (r_i+r_j)^2 ] = -2(r_i+r_j).
                # My code above: d_dr = -2*(ri+rj).
                # So jac[idx_rj] should be d_dr.
                jac[idx_rj] = d_dr
                
                return jac

            constraints.append({
                'type': 'ineq',
                'fun': fun_overlap,
                'jac': jac_overlap
            })
            
    # Provide jacs for boundary constraints too?
    # SLSQP might approximate if not provided, but providing is safer.
    # However, updating the constraints list in place is annoying.
    # Let's rely on numerical differentiation for boundaries or add jacs.
    # Given the number of constraints, providing jacs is good practice.
    # But for brevity and robustness, let's stick to 'fun' for boundaries 
    # as they are simple linear constraints, SLSQP handles them well.
    # Actually, let's just use 'fun' for all to keep code clean, 
    # but for overlap, providing jac is very helpful.
    
    return constraints

def run_packing():
    n = 26
    
    # Bounds for variables
    # x, y in [0, 1], r in [0, 0.5]
    bounds = []
    for _ in range(n):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 0.5)) # r
    
    constraints = get_constraints(n)
    
    best_sum_radii = 0.0
    best_vars = None
    
    # Try multiple initializations
    # 1. Hexagonal
    vars_hex = get_initial_guess_hex(n)
    # 2. Grid
    vars_grid = get_initial_guess_grid(n)
    # 3. Random valid (small radii)
    np.random.seed(42)
    vars_rand = np.zeros(3*n)
    for i in range(n):
        vars_rand[3*i] = np.random.uniform(0.1, 0.9)
        vars_rand[3*i+1] = np.random.uniform(0.1, 0.9)
        vars_rand[3*i+2] = 0.05 # Small radius to start valid
    
    initial_guesses = [vars_hex, vars_grid, vars_rand]
    
    for k, x0 in enumerate(initial_guesses):
        try:
            res = minimize(objective, x0, args=(n,), method='SLSQP', 
                           bounds=bounds, constraints=constraints,
                           options={'maxiter': 1000, 'ftol': 1e-9, 'disp': False})
            
            if res.success or res.nit > 0:
                # Extract radii and check validity manually just in case
                radii = res.x[2::3]
                current_sum = np.sum(radii)
                
                # We want to maximize sum, so minimize -sum.
                # res.fun is -sum.
                if -res.fun > best_sum_radii:
                    # Validate roughly
                    centers = np.column_stack((res.x[0::3], res.x[1::3]))
                    # Simple check: radii non-negative, inside square
                    valid_r = np.all(radii >= -1e-9)
                    if valid_r:
                        # Check overlaps roughly
                        valid_pack = True
                        for i in range(n):
                            for j in range(i+1, n):
                                dist = np.sqrt(np.sum((centers[i]-centers[j])**2))
                                if dist < radii[i] + radii[j] - 1e-5:
                                    valid_pack = False
                                    break
                            if not valid_pack: break
                        
                        # Check boundaries
                        for i in range(n):
                            x, y, r = centers[i,0], centers[i,1], radii[i]
                            if x < -r - 1e-5 or x > 1 + r + 1e-5 or \
                               y < -r - 1e-5 or y > 1 + r + 1e-5:
                                valid_pack = False
                                break
                                
                        if valid_pack:
                            best_sum_radii = current_sum
                            best_vars = res.x.copy()
        except Exception as e:
            print(f"Optimization run {k} failed: {e}")
            continue

    if best_vars is None:
        # Fallback to hex guess
        best_vars = vars_hex
        
    centers = np.column_stack((best_vars[0::3], best_vars[1::3]))
    radii = best_vars[2::3]
    
    # Ensure radii are non-negative
    radii = np.maximum(radii, 0.0)
    
    return centers, radii, float(np.sum(radii))
