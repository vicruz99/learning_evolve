import numpy as np
from scipy.optimize import minimize
import warnings

# Suppress warnings for cleaner output
warnings.filterwarnings('ignore')

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    n = centers.shape[0]

    if np.isnan(centers).any():
        return False
    if np.isnan(radii).any():
        return False

    for i in range(n):
        if radii[i] < 0:
            return False
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            return False

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False
    return True

def generate_hex_initial(n, r_est):
    """
    Generates a hexagonal packing of n circles with estimated radius r_est.
    """
    points = []
    # Hexagonal spacing
    dx = 2 * r_est
    dy = np.sqrt(3) * r_est
    
    # We want centers in [r_est, 1-r_est]
    y_curr = r_est
    row = 0
    
    while len(points) < n:
        # Determine x offset for zigzag pattern
        offset = 0.0
        if row % 2 == 1:
            offset = dx / 2.0
        
        x_curr = r_est + offset
        
        while x_curr <= 1 - r_est + 1e-9:
            if y_curr >= r_est - 1e-9 and y_curr <= 1 - r_est + 1e-9:
                points.append([x_curr, y_curr])
            
            if len(points) >= n:
                break
            x_curr += dx
        
        if len(points) >= n:
            break
        y_curr += dy
        row += 1
        
        # Safety break
        if row > 20:
            break
            
    if len(points) >= n:
        return np.array(points[:n])
    else:
        # Fallback to grid if hex fails (unlikely for small n)
        grid_x = np.linspace(r_est, 1-r_est, 6)
        grid_y = np.linspace(r_est, 1-r_est, 5)
        pts = []
        for x in grid_x:
            for y in grid_y:
                pts.append([x, y])
                if len(pts) >= n: break
            if len(pts) >= n: break
        return np.array(pts[:n])

def run_packing():
    n_circles = 26
    best_sum_r = -1.0
    best_centers = None
    best_radii = None
    
    # Estimated initial radius
    # Area of 26 circles <= 1. Density ~ 0.9. 
    # 26 * pi * r^2 <= 0.9 => r <= sqrt(0.9 / 26pi) ~ 0.105
    r_init = 0.09 
    
    # Try multiple seeds/initializations to find global optimum
    num_restarts = 10
    
    for restart in range(num_restarts):
        # Generate initial positions
        if restart == 0:
            # Pure hexagonal
            centers = generate_hex_initial(n_circles, r_init)
        else:
            # Perturbed hexagonal
            centers = generate_hex_initial(n_circles, r_init)
            # Add small noise
            noise = np.random.uniform(-0.02, 0.02, centers.shape)
            centers = centers + noise
            # Clamp to valid range [r_init, 1-r_init]
            centers = np.clip(centers, r_init, 1 - r_init)
            
        initial_radii = np.full(n_circles, r_init)
        
        # Flatten variables: x1, y1, r1, x2, y2, r2, ...
        x0 = []
        for i in range(n_circles):
            x0.extend([centers[i, 0], centers[i, 1], initial_radii[i]])
        x0 = np.array(x0)
        
        # Objective: Maximize sum of radii -> Minimize negative sum
        def objective(x):
            radii = x[2::3]
            return -np.sum(radii)
        
        # Constraints
        constraints = []
        
        # 1. Non-overlap constraints
        # (xi - xj)^2 + (yi - yj)^2 >= (ri + rj)^2
        # Rearranged: (xi - xj)^2 + (yi - yj)^2 - (ri + rj)^2 >= 0
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                # Indices in x vector
                idx_i_x = 3 * i
                idx_i_y = 3 * i + 1
                idx_i_r = 3 * i + 2
                
                idx_j_x = 3 * j
                idx_j_y = 3 * j + 1
                idx_j_r = 3 * j + 2
                
                def constraint_pair(idx_i_x=idx_i_x, idx_i_y=idx_i_y, idx_i_r=idx_i_r,
                                    idx_j_x=idx_j_x, idx_j_y=idx_j_y, idx_j_r=idx_j_r):
                    def func(x):
                        xi = x[idx_i_x]
                        yi = x[idx_i_y]
                        ri = x[idx_i_r]
                        xj = x[idx_j_x]
                        yj = x[idx_j_y]
                        rj = x[idx_j_r]
                        return (xi - xj)**2 + (yi - yj)**2 - (ri + rj)**2
                    return func
                
                constraints.append({'type': 'ineq', 'fun': constraint_pair()})
        
        # 2. Boundary constraints
        # x >= r, x <= 1-r => x - r >= 0, 1 - x - r >= 0
        # Same for y
        for i in range(n_circles):
            idx_x = 3 * i
            idx_y = 3 * i + 1
            idx_r = 3 * i + 2
            
            # Left boundary: x - r >= 0
            constraints.append({'type': 'ineq', 'fun': lambda x, ix=idx_x, ir=idx_r: x[ix] - x[ir]})
            # Right boundary: 1 - x - r >= 0
            constraints.append({'type': 'ineq', 'fun': lambda x, ix=idx_x, ir=idx_r: 1.0 - x[ix] - x[ir]})
            # Bottom boundary: y - r >= 0
            constraints.append({'type': 'ineq', 'fun': lambda x, iy=idx_y, ir=idx_r: x[iy] - x[ir]})
            # Top boundary: 1 - y - r >= 0
            constraints.append({'type': 'ineq', 'fun': lambda x, iy=idx_y, ir=idx_r: 1.0 - x[iy] - x[ir]})
            
            # Radius non-negative: r >= 0
            constraints.append({'type': 'ineq', 'fun': lambda x, ir=idx_r: x[ir]})
            
        # Bounds for variables to help optimizer
        # x, y in [0, 1], r in [0, 0.5]
        bounds = []
        for i in range(n_circles):
            bounds.append((0.0, 1.0)) # x
            bounds.append((0.0, 1.0)) # y
            bounds.append((0.0, 0.5)) # r
            
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints, 
                           options={'maxiter': 500, 'ftol': 1e-9})
            
            if res.success:
                # Extract results
                opt_centers = np.array([[res.x[3*i], res.x[3*i+1]] for i in range(n_circles)])
                opt_radii = np.array([res.x[3*i+2] for i in range(n_circles)])
                
                # Validate
                if validate_packing(opt_centers, opt_radii):
                    current_sum = np.sum(opt_radii)
                    if current_sum > best_sum_r:
                        best_sum_r = current_sum
                        best_centers = opt_centers
                        best_radii = opt_radii
        except Exception as e:
            continue

    # Fallback if optimization fails (should not happen with good init)
    if best_centers is None:
        best_centers = generate_hex_initial(n_circles, 0.05)
        best_radii = np.full(n_circles, 0.05)
        best_sum_r = np.sum(best_radii)

    return best_centers, best_radii, best_sum_r
