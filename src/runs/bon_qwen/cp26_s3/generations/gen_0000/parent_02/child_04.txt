import numpy as np
from scipy.optimize import minimize


def hexagonal_init(n):
    """Initialize circles in a hexagonal pattern"""
    row_counts = [6, 5, 6, 5, 4]
    num_rows = len(row_counts)
    y_spacing = 1.0 / (num_rows + 1)
    
    points = []
    for row_idx, num_cols in enumerate(row_counts):
        y = (row_idx + 1) * y_spacing
        x_spacing = 1.0 / (num_cols + 1)
        offset = x_spacing * 0.5 if row_idx % 2 == 1 else 0.0
        
        for col in range(num_cols):
            x = (col + 1) * x_spacing + offset
            points.append([x, y])
    
    centers = np.array(points[:n])
    radii = np.full(n, 0.06)
    return centers, radii


def compute_penalty(params, n, weight):
    """Compute penalty for constraint violations"""
    centers = params[:2 * n].reshape(n, 2)
    radii = params[2 * n:]
    penalty = 0.0
    
    # Boundary penalties
    for i in range(n):
        x = centers[i, 0]
        y = centers[i, 1]
        r = radii[i]
        
        v1 = min(0.0, x - r)
        v2 = min(0.0, 1.0 - x - r)
        v3 = min(0.0, y - r)
        v4 = min(0.0, 1.0 - y - r)
        
        penalty += v1 * v1
        penalty += v2 * v2
        penalty += v3 * v3
        penalty += v4 * v4
    
    # Overlap penalties
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            min_dist = radii[i] + radii[j]
            v = min(0.0, dist - min_dist)
            penalty += v * v
    
    return weight * penalty


def objective(params, n, weight):
    """Objective: maximize sum of radii with penalty for violations"""
    radii = params[2 * n:]
    return -np.sum(radii) + compute_penalty(params, n, weight)

    for _ in range(n):
        bounds.append((0.0, 1.0))
        bounds.append((0.0, 1.0))

        y = centers[i, 1]

            break
    
    return centers, radii


def optimize_single_run(centers, radii, n):
