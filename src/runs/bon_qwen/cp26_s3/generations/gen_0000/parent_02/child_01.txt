def get_max_sum_radii(centers):
    """
    Given fixed centers, find the maximum sum of radii using Linear Programming.
    Constraints:
    1. r_i + r_j <= distance(i, j) for all i < j
    2. r_i <= x_i, r_i <= 1 - x_i, r_i <= y_i, r_i <= 1 - y_i
    3. r_i >= 0
    Objective: Maximize sum(r_i)
    """
    n = centers.shape[0]
    
    # Variables: r_0, r_1, ..., r_{n-1}
    # Objective: maximize sum(r) => minimize -sum(r)
    c = -np.ones(n)
    
    # Inequality constraints: A_ub * r <= b_ub
    A_ub = []
    b_ub = []
    
    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        # r_i <= x
        row = np.zeros(n)
        row[i] = 1
        A_ub.append(row)
        b_ub.append(x)
        
        # r_i <= 1 - x
        A_ub.append(row)
        b_ub.append(1 - x)
        
        # r_i <= y
        A_ub.append(row)
        b_ub.append(y)
        
        # r_i <= 1 - y
        A_ub.append(row)
        b_ub.append(1 - y)
        
    # Pairwise constraints: r_i + r_j <= dist
    # Precompute distances to speed up
    dists = np.linalg.norm(centers[np.newaxis, :] - centers[:, np.newaxis], axis=2)
    
    for i in range(n):
        for j in range(i + 1, n):
            dist = dists[i, j]
            row = np.zeros(n)
            row[i] = 1
            row[j] = 1
            A_ub.append(row)
            b_ub.append(dist)
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    # Bounds for r_i: [0, None]
    bounds = [(0, None) for _ in range(n)]
    
    # Solve LP
    # We use 'highs' method for efficiency if available, else 'simplex' or 'interior-point'
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    except ValueError:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='simplex')
        
    if res.success:
        return -res.fun, res.x
    else:
        # Fallback: small radii
        return 0.0, np.zeros(n)
