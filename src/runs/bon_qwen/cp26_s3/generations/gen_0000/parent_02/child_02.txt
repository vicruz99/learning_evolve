

Looking at this circle packing problem, I need to pack 26 circles in a unit square to maximize the sum of radii (target: 2.636).

<strategy>

**Strategy Analysis:**

1. **Initial Configuration**: Use a hexagonal grid pattern as starting point, which is the densest packing for equal circles. For 26 circles, I'll arrange them in staggered rows.

2. **Optimization Approach**: Use SLSQP (Sequential Least Squares Programming) with explicit inequality constraints:
   - Boundary constraints: x ≥ r, x ≤ 1-r, y ≥ r, y ≤ 1-r for each circle
   - Non-overlap constraints: dist(i,j) ≥ r_i + r_j for all pairs
   - Objective: maximize sum(r_i)

3. **Multiple Restarts**: Run optimization from 15 different starting configurations to avoid local optima. Vary the grid spacing, offsets, and random perturbations.

4. **Force-Based Pre-optimization**: Before SLSQP, use a simple force-based simulation to resolve overlaps and get closer to a feasible region.

5. **Refinement**: After getting a feasible solution, run additional SLSQP refinement passes.

Key insight: The optimal solution likely has circles of varying sizes - larger in the center, smaller near boundaries/corners. SLSQP will naturally discover this through the constraint structure.

</strategy>

import numpy as np
from scipy.optimize import minimize


def objective(params):
    """Minimize negative sum of radii (maximize sum of radii)."""
    return -np.sum(params[2::3])


def bound_x_ge_r(params, i):
    """Constraint: x_i >= r_i"""
    return params[3 * i] - params[3 * i + 2]


def bound_x_le_1mr(params, i):
    """Constraint: x_i + r_i <= 1"""
    return 1.0 - params[3 * i] - params[3 * i + 2]


def bound_y_ge_r(params, i):
    """Constraint: y_i >= r_i"""
    return params[3 * i + 1] - params[3 * i + 2]


def bound_y_le_1mr(params, i):
    """Constraint: y_i + r_i <= 1"""
    return 1.0 - params[3 * i + 1] - params[3 * i + 2]


def nonoverlap_constraint(params, pair):
    """Constraint: dist(i,j) >= r_i + r_j"""
    i, j = pair
    xi, yi, ri = params[3 * i], params[3 * i + 1], params[3 * i + 2]
    xj, yj, rj = params[3 * j], params[3 * j + 1], params[3 * j + 2]
    dx = xi - xj
    dy = yi - yj
    dist = np.sqrt(dx * dx + dy * dy)
    return dist - ri - rj


def make_bounds(n):
    """Create bound constraints for x, y, r."""
    bounds = []
    for i in range(n):
        bounds.append((0.0, 1.0))    # x in [0, 1]
        bounds.append((0.0, 1.0))    # y in [0, 1]
        bounds.append((0.0, 0.5))    # r in [0, 0.5]
    return bounds


def make_constraints(n):
    """Create all constraint dictionaries."""
    constraints = []
    for i in range(n):
        constraints.append({'type': 'ineq', 'fun': bound_x_ge_r, 'args': (i,)})
        constraints.append({'type': 'ineq', 'fun': bound_x_le_1mr, 'args': (i,)})
        constraints.append({'type': 'ineq', 'fun': bound_y_ge_r, 'args': (i,)})
        constraints.append({'type': 'ineq', 'fun': bound_y_le_1mr, 'args': (i,)})
    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({'type': 'ineq', 'fun': nonoverlap_constraint, 'args': ((i, j),)})
    return constraints


def create_hexagonal_init(n, spacing, offset_x, offset_y):
    """Create hexagonal grid initial configuration."""
    centers = np.zeros((n, 2))
    radii = np.full(n, spacing * 0.4)
    
    idx = 0
    row = 0
    while idx < n:
        for col in range(10):
            if idx >= n:
                break
            x = offset_x + col * spacing * np.sqrt(3) * 0.5
            y = offset_y + row * spacing * 1.5
            if col % 2 == 1:
                y += spacing * 0.75
            if 0.05 <= x <= 0.95 and 0.05 <= y <= 0.95:
                centers[idx] = [x, y]
                idx += 1
        row += 1
        if row > 15:
            break
    
    return centers, radii


def force_relax(centers, radii, n, iterations=5000):
    """Simple force-based relaxation to resolve overlaps."""
    dt = 0.005
    for it in range(iterations):
        forces = np.zeros((n, 2))
        
        # Repulsion from overlaps
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[j, 0] - centers[i, 0]
                dy = centers[j, 1] - centers[i, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                if dist < 1e-10:
                    dist = 1e-10
                    dx = 1e-10
                    dy = 1e-10
                min_dist = radii[i] + radii[j]
                if dist < min_dist:
                    overlap = (min_dist - dist) / dist
                    fx = dx * overlap * 5.0
                    fy = dy * overlap * 5.0
                    forces[i, 0] -= fx
                    forces[i, 1] -= fy
                    forces[j, 0] += fx
                    forces[j, 1] += fy
        
        # Boundary repulsion
        for i in range(n):
            x, y, r = centers[i, 0], centers[i, 1], radii[i]
            if x - r < 0:
                forces[i, 0] += (r - x) * 10.0
            if x + r > 1:
                forces[i, 0] -= (x + r - 1) * 10.0
            if y - r < 0:
                forces[i, 1] += (r - y) * 10.0
            if y + r > 1:
                forces[i, 1] -= (y + r - 1) * 10.0
        
        centers += forces * dt
        centers = np.clip(centers, 0.01, 0.99)
    
    return centers, radii


def run_packing():
    n = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None
    all_constraints = make_constraints(n)
    all_bounds = make_bounds(n)
    
    # Try multiple starting configurations
    for trial in range(20):
        rng = np.random.RandomState(trial * 7919 + 13)

        offset_y = 0.08 + rng.uniform(0, 0.12)
        
        centers, radii = create_hexagonal_init(n, spacing, offset_x, offset_y)
        
        # Add small random perturbation
        centers += rng.uniform(-0.02, 0.02, size=centers.shape)
        centers = np.clip(centers, 0.05, 0.95)
        radii += rng.uniform(-0.005, 0.005, size=n)
        radii = np.clip(radii, 0.02, 0.25)
        
        # Force-based pre-relaxation
        centers, radii = force_relax(centers, radii, n, iterations=2000)
        
        # Flatten to params
        params = np.zeros(3 * n)
        for i in range(n):
            params[3 * i] = centers[i, 0]
            params[3 * i + 1] = centers[i, 1]
            params[3 * i + 2] = radii[i]
        
        # Try SLSQP optimization
        result = minimize(
            objective, params, method='SLSQP', bounds=all_bounds,
            constraints=all_constraints,
            options={'maxiter': 8000, 'ftol': 1e-15, 'disp': False}

        # Extract and validate
        current_centers = np.zeros((n, 2))
        current_radii = np.zeros(n)

            current_radii[i] = result.x[3 * i + 2]
