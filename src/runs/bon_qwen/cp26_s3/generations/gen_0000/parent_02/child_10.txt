def get_initial_configs(n):

    dy = np.sqrt(3) * r_est

            y = y + 0.05

def pack_circles(n=26):
    """
    Optimizes circle packing for n circles in a unit square.
    Returns (centers, radii, sum_radii).
    """
    best_sum = -1.0
    best_centers = None
    best_radii = None

    configs = get_initial_configs(n)

    for i, (c0, r0) in enumerate(configs):
        # Flatten variables: [x1, y1, r1, x2, y2, r2, ...]
        x0 = np.concatenate([c0.flatten(), r0])
        
        # Bounds
        # x, y in [0, 1], r in [0, 0.5]
        bounds = [(0, 1) for _ in range(2 * n)] + [(0, 0.5) for _ in range(n)]

        # Constraints
        constraints = []
        
        # Non-overlap constraints: dist(i, j) >= r_i + r_j
        # (x_i - x_j)^2 + (y_i - y_j)^2 >= (r_i + r_j)^2
        # We define function g(x) >= 0
        for k in range(n):
            idx_xi = 3 * k
            idx_yi = 3 * k + 1
            idx_ri = 3 * k + 2
            
            # Boundary constraints
            # x - r >= 0
            constraints.append({
                'type': 'ineq',
                'fun': lambda v, idx_x=idx_xi, idx_r=idx_ri: v[idx_x] - v[idx_r]
            })
            # 1 - x - r >= 0
            constraints.append({
                'type': 'ineq',
                'fun': lambda v, idx_x=idx_xi, idx_r=idx_ri: 1.0 - v[idx_x] - v[idx_r]
            })
            # y - r >= 0
            constraints.append({
                'type': 'ineq',
                'fun': lambda v, idx_y=idx_yi, idx_r=idx_ri: v[idx_y] - v[idx_r]
            })
            # 1 - y - r >= 0
            constraints.append({
                'type': 'ineq',
                'fun': lambda v, idx_y=idx_yi, idx_r=idx_ri: 1.0 - v[idx_y] - v[idx_r]
            })

            # Overlap constraints with subsequent circles
            for l in range(k + 1, n):
                idx_xj = 3 * l
                idx_yj = 3 * l + 1
                idx_rj = 3 * l + 2
                
                # dist^2 - (r_i + r_j)^2 >= 0
                def dist_sq_constraint(v, i=k, j=l):
                    xi, yi, ri = v[3*i], v[3*i+1], v[3*i+2]
                    xj, yj, rj = v[3*j], v[3*j+1], v[3*j+2]
                    d2 = (xi - xj)**2 + (yi - yj)**2
                    r_sum_sq = (ri + rj)**2
                    return d2 - r_sum_sq
                
                constraints.append({
                    'type': 'ineq',
                    'fun': dist_sq_constraint
                })

        # Objective: maximize sum(r_i) -> minimize -sum(r_i)
        def objective(v):
            radii = v[2::3]
            return -np.sum(radii)

        # Run optimization
        # SLSQP is generally good for constrained non-linear problems
        try:
            res = opt.minimize(
                objective,
                x0,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 500, 'ftol': 1e-9}
            )
            
            if res.success:
                sum_r = -res.fun

        except Exception as e:
            continue

                    idx += 1

            centers[idx] = [0.5, 0.5] # Overlap, but validation will catch? No, we need valid.

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True
