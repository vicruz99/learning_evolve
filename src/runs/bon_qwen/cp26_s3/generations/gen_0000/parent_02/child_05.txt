import numpy as np
from scipy.optimize import minimize

def obj_func(v, n):
    return -np.sum(v[2*n:])

def bound_constr_x_lower(v, n, i):
    return v[i] - v[2*n + i]

def bound_constr_x_upper(v, n, i):
    return 1.0 - v[i] - v[2*n + i]

def bound_constr_y_lower(v, n, i):
    return v[n + i] - v[2*n + i]

def bound_constr_y_upper(v, n, i):
    return 1.0 - v[n + i] - v[2*n + i]

def overlap_constr(v, n, i, j):
    xi, yi = v[i], v[n+i]
    xj, yj = v[j], v[n+j]
    ri, rj = v[2*n+i], v[2*n+j]
    return (xi - xj)**2 + (yi - yj)**2 - (ri + rj)**2

def run_packing():
    n = 26
    np.random.seed(42)
    centers = np.random.rand(n, 2) * 0.8 + 0.1
    radii = np.full(n, 0.05)
    x0 = np.concatenate([centers.flatten(), radii])
    
    bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n
    
    constraints = []
    for i in range(n):
        constraints.append({'type': 'ineq', 'fun': bound_constr_x_lower, 'args': (n, i)})
        constraints.append({'type': 'ineq', 'fun': bound_constr_x_upper, 'args': (n, i)})
        constraints.append({'type': 'ineq', 'fun': bound_constr_y_lower, 'args': (n, i)})
        constraints.append({'type': 'ineq', 'fun': bound_constr_y_upper, 'args': (n, i)})

            constraints.append({'type': 'ineq', 'fun': overlap_constr, 'args': (n, i, j)})
            
    res = minimize(obj_func, x0, args=(n,), bounds=bounds, constraints=constraints, method='SLSQP', options={'maxiter': 5000, 'ftol': 1e-10})
    
    centers_opt = res.x[:2*n].reshape(n, 2)
    radii_opt = res.x[2*n:]

    min_gap = 1.0
