import numpy as np
from scipy.optimize import minimize
import math

def create_hexagonal_init(n):
    """Create hexagonal grid initialization for n circles."""
    centers = np.zeros((n, 2))
    idx = 0
    row = 0
    while idx < n:
        cols = 6 if row % 2 == 0 else 5
        y = (row + 0.5) / 6.0
        x_spacing = 1.0 / (cols + 1)
        for col in range(cols):
            if idx >= n:
                break
            x = (col + 1) * x_spacing
            if row % 2 == 1:
                x += x_spacing / 2.0
            centers[idx] = [np.clip(x, 0.02, 0.98), np.clip(y, 0.02, 0.98)]
            idx += 1
        row += 1
    return centers


def smooth_penalty(violation, k=50.0):
    """Smooth approximation of max(0, violation)^2 using softplus."""
    return (np.log(1.0 + np.exp(k * violation)) / k) ** 2


def penalty_objective(vars, n, penalty_weight=10000.0, k=50.0):
    """Objective function with smooth penalty for constraint violations."""
    centers = vars[:2*n].reshape(n, 2)
    radii = vars[2*n:]
    
    # Objective: maximize sum of radii
    obj = -np.sum(radii)
    
    # Boundary penalties
    for i in range(n):
        x, y = centers[i]
        ri = radii[i]
        # x >= r (violation: r - x > 0)
        obj += penalty_weight * smooth_penalty(ri - x, k)
        # x <= 1 - r (violation: x + r - 1 > 0)
        obj += penalty_weight * smooth_penalty(x + ri - 1.0, k)
        # y >= r
        obj += penalty_weight * smooth_penalty(ri - y, k)
        # y <= 1 - r
        obj += penalty_weight * smooth_penalty(y + ri - 1.0, k)
    
    # Overlap penalties
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = math.sqrt(dx*dx + dy*dy + 1e-15)
            min_dist = radii[i] + radii[j]
            violation = min_dist - dist
            obj += penalty_weight * smooth_penalty(violation, k)
    
    return obj


def optimize_with_penalties(n, centers_init, radii_init):
    """Run optimization with progressively increasing penalties."""
    x0 = np.concatenate([centers_init.flatten(), radii_init])
    
    bounds = []
    for _ in range(2 * n):
        bounds.append((0.001, 0.999))
    for _ in range(n):
        bounds.append((0.001, 0.5))
    
    best_centers = None
    best_radii = None
    best_sum = 0.0
    
    # Try multiple penalty values, increasing
    for penalty in [500, 2000, 10000, 50000]:
        # Smoothness parameter
        k_val = 30.0 if penalty < 10000 else 50.0
        
        result = minimize(
            penalty_objective,
            x0,
            args=(n, penalty, k_val),
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 5000, 'ftol': 1e-15, 'gtol': 1e-10}
        )
        
        c_opt = result.x[:2*n].reshape(n, 2)
        r_opt = np.maximum(result.x[2*n:], 0.001)
        
        # Post-process
        c_opt, r_opt = ensure_validity(n, c_opt, r_opt)
        
        current_sum = np.sum(r_opt)
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = c_opt.copy()
            best_radii = r_opt.copy()
        
        x0 = result.x.copy()
    
    if best_centers is None:
        best_centers = centers_init
        best_radii = radii_init
    
    return best_centers, best_radii, best_sum


def ensure_validity(n, centers, radii):
    """Ensure all constraints are strictly satisfied."""
    radii = radii.copy()
    centers = centers.copy()
    
    # Enforce boundary constraints
    for i in range(n):
        x, y = centers[i]
        radii[i] = min(radii[i], x - 1e-9, 1.0 - x - 1e-9, y - 1e-9, 1.0 - y - 1e-9)
        radii[i] = max(radii[i], 0.0)
    
    # Enforce non-overlap constraints
    for _ in range(100):
        changed = False
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = math.sqrt(dx*dx + dy*dy + 1e-15)
                combined_r = radii[i] + radii[j]
                if combined_r > dist - 1e-9:
                    ratio = max(0.0, (dist - 1e-9)) / max(combined_r, 1e-12)
                    radii[i] *= ratio
                    radii[j] *= ratio
                    changed = True
        if not changed:
            break
    
    return centers, radii
