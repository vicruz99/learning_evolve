import numpy as np
import scipy.optimize
import scipy.spatial

def get_hex_grid_points(n_points):
    """
    Generate n_points in a hexagonal grid pattern within [0, 1]x[0, 1].
    Returns an array of shape (n_points, 2).
    """
    points = []
    # Estimate spacing based on area
    # Area per point approx 1/n. Sides of hex approx sqrt(2/ (sqrt(3)/2 * n)) ? 
    # Just use a heuristic spacing.
    # For N=26, density is high. Spacing approx 0.2-0.3.
    spacing = 0.25 
    dy = spacing * np.sqrt(3) / 2
    
    y = 0.0
    while True:
        x = 0.0
        # Alternate x offset for hexagonal packing
        if len(points) % 2 == 0:
            x_start = 0.0
        else:
            x_start = spacing / 2.0
        
        while x < 1.0 + spacing/2:
            if 0 <= x <= 1.0 and 0 <= y <= 1.0:
                points.append([x, y])
                if len(points) >= n_points:
                    return np.array(points[:n_points])
            x += spacing
        y += dy
        if y > 1.0 + dy:
            break
            
    # Fallback to random or grid if not enough points generated
    # (Should not happen with appropriate spacing)
    grid_x = np.linspace(0.1, 0.9, 6)
    grid_y = np.linspace(0.1, 0.9, 5)
    pts = []
    for y in grid_y:
        for x in grid_x:
            pts.append([x, y])
            if len(pts) >= n_points:
                return np.array(pts[:n_points])
    return np.array(pts[:n_points])

def compute_sum_radii(centers):
    """
    Given centers, solve LP to maximize sum of radii.
    Returns (sum_radii, radii_array).
    """
    n = centers.shape[0]
    if n == 0:
        return 0.0, np.array([])

    # Variables: r_0, ..., r_{n-1}
    # Objective: Maximize sum(r_i) => Minimize -sum(r_i)
    c_obj = -np.ones(n)

    # Constraints: A_ub * r <= b_ub
    
    # 1. Non-overlap: r_i + r_j <= dist(i, j)
    # 2. Boundary: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    # 3. Non-negativity: r_i >= 0 (handled by bounds)

    # Calculate pairwise distances
    # Using scipy.spatial.distance.cdist for efficiency
    dists = scipy.spatial.distance.cdist(centers, centers)
    
    # We only need upper triangular part for constraints i < j
    # But we can just build the full matrix or loop. 
    # Loop is fine for n=26.
    
    num_constraints = 0
    # Estimate size: n*(n-1)/2 pairs + 4*n boundary constraints
    # 26*25/2 = 325 + 104 = 429 constraints.
    
    A_ub = []
    b_ub = []
    
    # Pairwise constraints
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        bounds_list = [x, 1.0 - x, y, 1.0 - y]
        for val in bounds_list:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(val)
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    # Bounds for r_i: r_i >= 0
    bounds = [(0, None)] * n
    
    try:
        # Use high-level method, 'highs' is default and fast
        res = scipy.optimize.linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        
        if res.success:
            radii = res.x
            return -res.fun, radii
        else:
            # Fallback if LP fails (should not happen)
            return 0.0, np.zeros(n)
    except Exception:
        return 0.0, np.zeros(n)

def objective(centers_flat):
    """
    Objective function for optimization.
    Minimizes negative sum of radii.
    """
    n = 26
    centers = centers_flat.reshape(n, 2)
    # Ensure centers are within bounds (optimizer should handle this, but safe to clip or check)
    # Actually, bounds are passed to minimize.
    
    sum_r, radii = compute_sum_radii(centers)
    return -sum_r # Minimize negative sum

def run_packing():
    n = 26
    
    # 1. Initialization
    # Try to generate a hexagonal grid
    centers_init = get_hex_grid_points(n)
    
    # If we didn't get exactly n points (unlikely with logic above), pad
    if centers_init.shape[0] < n:
        # Fill remaining with random valid points
        missing = n - centers_init.shape[0]
        random_pts = np.random.rand(missing, 2)
        centers_init = np.vstack([centers_init, random_pts])
    elif centers_init.shape[0] > n:
        centers_init = centers_init[:n]
        
    # Scale/shift to ensure strictly inside if needed, but [0,1] is fine.
    # The hex grid generator uses [0, 1] range.
    
    # 2. Optimization
    # We want to maximize sum of radii.
    # Use Powell method as it doesn't require gradient and handles non-smoothness well.
    # Bounds for centers: [0, 1] for both x and y.
    bounds = [(0, 1)] * (2 * n)
    
    # Flatten initial centers
    x0 = centers_init.flatten()
    
    # Run optimization
    # Maximize sum_radii => Minimize -sum_radii
    try:
        res = scipy.optimize.minimize(
            objective, 
            x0, 
            method='Powell', 
            bounds=bounds,
            options={'maxiter': 500, 'ftol': 1e-8, 'xtol': 1e-8}
        )
        optimal_centers = res.x.reshape(n, 2)
        sum_r, optimal_radii = compute_sum_radii(optimal_centers)
    except Exception as e:
        # Fallback to initial if optimization fails
        print(f"Optimization failed: {e}")
        optimal_centers = centers_init
        sum_r, optimal_radii = compute_sum_radii(optimal_centers)
        
    # 3. Post-processing / Validation
    # Ensure strict validity
    # The LP ensures r_i + r_j <= dist. 
    # However, numerical errors might occur.
    # Also, LP might return radii that slightly violate constraints due to tolerance.
    # We can shrink radii slightly if needed, but 'highs' is precise.
    
    # Let's do a quick check and slight adjustment if necessary
    # But usually the LP solution is valid.
    
    # Just to be safe with the validation function's epsilon (1e-12)
    # We can clamp radii to be strictly feasible if needed, but LP is robust.
    
    return optimal_centers, optimal_radii, sum_r

# Self-test helper
def validate_packing(centers, radii):
    """
    Validation function provided in prompt.
    """
    n = centers.shape[0]
    if np.isnan(centers).any(): return False
    if np.isnan(radii).any(): return False
    for i in range(n):
        if radii[i] < 0: return False
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            return False
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False
    return True

if __name__ == "__main__":
