import numpy as np
from scipy.optimize import minimize

def objective_func(v, n):
    """Objective: maximize sum of radii => minimize negative sum"""
    return -np.sum(v[2::3])

def constraint_func(v, n):
    """Constraints: boundaries and non-overlap"""
    C = v.reshape((n, 3))
    X = C[:, 0]
    Y = C[:, 1]
    R = C[:, 2]
    
    vals = []
    # Boundary constraints: circle must be inside [0,1]x[0,1]
    vals.extend(X - R)
    vals.extend(1.0 - X - R)
    vals.extend(Y - R)
    vals.extend(1.0 - Y - R)
    
    # Pairwise non-overlap constraints: dist >= r_i + r_j
    dx = X[:, None] - X[None, :]
    dy = Y[:, None] - Y[None, :]
    # Add small epsilon to prevent sqrt(0) issues, though dist>0 typically
    D = np.sqrt(dx**2 + dy**2 + 1e-12)
    R_sum = R[:, None] + R[None, :]
    diff = D - R_sum
    # Only need upper triangle for i < j
    vals.extend(np.triu(diff, k=1).flatten())
    
    return np.array(vals)

def run_packing():
    n = 26
    best_sum = 0.0
    best_centers = np.zeros((n, 2))
    best_radii = np.zeros(n)

    # Try multiple seeds to find a better local/global optimum
    for seed in range(5):
        np.random.seed(seed)
        
        # Initial grid layout
        centers = np.zeros((n, 2))
        cols = 6
        rows = 5
        for i in range(n):
            centers[i, 0] = (i % cols + 0.5) / cols
            centers[i, 1] = (i // cols + 0.5) / rows
            
        # Add small random perturbation to break symmetry
        centers += np.random.uniform(-0.02, 0.02, size=centers.shape)
        centers = np.clip(centers, 0.05, 0.95)
        
        # Start with reasonable radii
        radii = np.ones(n) * 0.08
        x0 = np.concatenate([centers.flatten(), radii])

        # Bounds: coordinates in [0,1], radii in [0, 0.5]
        bounds = [(0.0, 1.0) for _ in range(2*n)] + [(0.0, 0.5) for _ in range(n)]

        # Optimization
        res = minimize(objective_func, x0, args=(n,), method='SLSQP', 
                       bounds=bounds, 
                       constraints={'type': 'ineq', 'fun': constraint_func, 'args': (n,)}, 
                       options={'maxiter': 2000, 'ftol': 1e-9, 'disp': False})
        
        if not np.isnan(res.x).any():
            current_sum = np.sum(res.x[2::3])
            if current_sum > best_sum:
                best_sum = current_sum
                best_centers = res.x[:2*n].reshape((n, 2))
                best_radii = res.x[2*n:]

    return best_centers, best_radii, best_sum
