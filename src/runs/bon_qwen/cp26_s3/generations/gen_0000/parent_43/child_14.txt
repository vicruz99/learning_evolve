import numpy as np
from scipy.optimize import minimize

def compute_penalty(centers, radii):
    """
    Computes a penalty score for the given centers and radii.
    Penalty is high if circles overlap or are outside the unit square.
    """
    n = centers.shape[0]
    penalty = 0.0
    
    # Boundary constraints:
    # Each circle i must satisfy: r_i <= x_i <= 1 - r_i  and  r_i <= y_i <= 1 - r_i
    # We penalize violations quadratically.
    
    # Left wall: x_i < r_i  => violation r_i - x_i
    left_viol = np.maximum(0, radii - centers[:, 0])
    # Right wall: x_i > 1 - r_i  => x_i + r_i > 1 => violation x_i + r_i - 1
    right_viol = np.maximum(0, centers[:, 0] + radii - 1.0)
    # Bottom wall: y_i < r_i
    bottom_viol = np.maximum(0, radii - centers[:, 1])
    # Top wall: y_i > 1 - r_i
    top_viol = np.maximum(0, centers[:, 1] + radii - 1.0)
    
    boundary_penalty = np.sum(left_viol**2) + np.sum(right_viol**2) + \
                       np.sum(bottom_viol**2) + np.sum(top_viol**2)
    
    # Overlap constraints:
    # Distance between centers i and j must be >= r_i + r_j
    # Squared distance >= (r_i + r_j)^2
    # Violation: (r_i + r_j)^2 - dist^2 > 0
    
    # Efficient pairwise distance calculation
    # centers shape (n, 2)
    # Compute squared distances using broadcasting
    # diff shape (n, n, 2)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist_sq = np.sum(diff**2, axis=2)
    
    # Radii sum squared
    r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    r_sum_sq = r_sum**2
    
    # Violation matrix (upper triangle only is needed, but we compute all and divide by 2 later or just sum)
    # Note: diagonal is dist_sq=0, r_sum_sq=(2r)^2=4r^2, so huge violation on diagonal.
    # We must ignore diagonal.
    
    violation = r_sum_sq - dist_sq
    np.fill_diagonal(violation, 0)
    
    # Only positive violations contribute
    overlap_penalty = np.sum(np.maximum(0, violation)**2)
    
    # Weight boundary and overlap penalties appropriately
    # Overlaps are generally harder constraints to satisfy initially
    return boundary_penalty + overlap_penalty

def optimize_positions(centers_init, radii, n_runs=5):
    """
    Optimizes positions for fixed radii to minimize penalty.
    Tries multiple restarts to avoid local minima.
    """
    best_centers = centers_init.copy()
    best_penalty = compute_penalty(centers_init, radii)
    
    n = len(radii)
    x0 = centers_init.flatten()
    
    # Bounds for centers: [0, 1] roughly, but effectively [min(r), 1-min(r)]
    # We use [0, 1] for bounds and rely on penalty for strictness, 
    # or set tighter bounds. Let's use [0,1] to allow escape.
    bounds = [(0.0, 1.0)] * (2 * n)
    
    for _ in range(n_runs):
        # Random perturbation for restart
        current_centers = centers_init.copy()
        current_centers += np.random.uniform(-0.01, 0.01, current_centers.shape)
        
        # Clip to valid range roughly
        current_centers = np.clip(current_centers, 0.0, 1.0)
        
        res = minimize(
            fun=lambda x: compute_penalty(x.reshape(n, 2), radii),
            x0=current_centers.flatten(),
            method='L-BFGS-B',
            bounds=bounds,
            options={'ftol': 1e-12, 'gtol': 1e-8, 'maxiter': 500}
        )
        
        final_centers = res.x.reshape(n, 2)
        final_penalty = compute_penalty(final_centers, radii)
        
        if final_penalty < best_penalty:
            best_penalty = final_penalty
            best_centers = final_centers
            
    return best_centers, best_penalty

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square

    Args:
        centers: np.array of shape (n, 2) with (x, y) coordinates
        radii: np.array of shape (n) with radius of each circle

    Returns:
        True if valid, False otherwise
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True
