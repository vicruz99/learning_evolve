def calculate_distance(p1, p2):
    return np.sqrt(np.sum((p1 - p2) ** 2))

def initial_hex_packing(n):

    if n == 0:
        return np.array([])

    row_count = 0
    col_count = 0

    spacing_y = approx_r * math.sqrt(3)

        shift = (spacing_x / 2) if r % 2 == 1 else 0

        # We want to center the row roughly
        # But for optimization, simple grid is fine
        
        # Estimate how many fit in width

def objective(p, n):

    Objective function to maximize the minimum distance.
    We minimize the negative of the minimum distance.
    """

    min_dist = 1.0

    # Actually, if we find max min_dist = D, then r = D/2.
    # Constraint: centers must be in [r, 1-r].

    # This must be >= r.
    # So 2 * dist_to_boundary >= 2r = D.
    # So we require 2 * dist_to_boundary >= D.
    # Or dist_to_boundary >= D/2.
    # In the optimization, we are looking for the configuration.

    # We want to maximize the "clearance".
    # Clearance = min( min(d_ij/2), min(d_ijk_wall) )
    # We want to maximize this clearance.
    
    # Let's compute raw distances
    np.fill_diagonal(dists, np.inf) # Ignore self
    min_pairwise = np.min(dists)
    
    # Boundary distances
    x = centers[:, 0]
    y = centers[:, 1]
    d_bound_x = np.minimum(x, 1 - x)
    d_bound_y = np.minimum(y, 1 - y)
    d_bound = np.minimum(d_bound_x, d_bound_y)
    min_bound = np.min(d_bound)
    
    # The effective radius r is limited by min_pairwise / 2 and min_bound.
    # So r_eff = min(min_pairwise / 2, min_bound).
    # We want to maximize r_eff.
    
    r_eff = min(min_pairwise / 2.0, min_bound)
    
    return -r_eff

def run_packing():
