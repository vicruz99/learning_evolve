from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    Uses optimization starting from a hexagonal lattice.

    def generate_hex_packing(n, shift_x=0.0, shift_y=0.0):

                points.append([x + shift_x, y + shift_y])
                x += 2 * r_est
            y += np.sqrt(3) * r_est
            row += 1

        radii = np.full(n, r_est)
        return centers, radii

        radii = params[2*n:]

        penalty = 0.0
        eps = 1e-6 # Tolerance for constraints

            if r > x + eps: penalty += 1e4 * (r - x - eps)**2
            if r > 1 - x + eps: penalty += 1e4 * (r + x - 1 - eps)**2
