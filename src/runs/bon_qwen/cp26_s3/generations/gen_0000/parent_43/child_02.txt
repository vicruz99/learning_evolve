import numpy as np
from scipy.optimize import minimize
import math

def compute_objective_and_violations(vars_flat, n=26):
    """
    Computes the negative sum of radii and the total penalty for constraint violations.
    
    Args:
        vars_flat: 1D numpy array of shape (n*3,) containing [x1, y1, r1, x2, y2, r2, ...]
        n: Number of circles
        
    Returns:
        obj_value: The value to minimize (negative sum of radii + penalty)
        violations: The scalar sum of squared violations (for tracking)
    """
    # Reshape variables
    x = vars_flat[0::3]
    y = vars_flat[1::3]
    r = vars_flat[2::3]
    
    # 1. Objective: Maximize sum of radii -> Minimize negative sum
    sum_radii = np.sum(r)
    
    # 2. Penalties
    penalty = 0.0
    penalty_weight = 1000.0 # Will be passed or handled externally? 
    # Actually, to make this function clean, we can compute raw violations and scale outside.
    # But for speed, let's compute the weighted value directly if we assume a fixed lambda?
    # No, lambda changes. Let's return raw violations and objective part.
    
    # Boundary Violations
    # r <= x <= 1-r  => x - r >= 0, 1 - x - r >= 0
    # r <= y <= 1-y  => y - r >= 0, 1 - y - r >= 0
    
    # Vectorized boundary checks
    # Left boundary
    v_left = np.minimum(0, x - r)
    # Right boundary
    v_right = np.minimum(0, 1 - x - r)
    # Bottom boundary
    v_bottom = np.minimum(0, y - r)
    # Top boundary
    v_top = np.minimum(0, 1 - y - r)
    
    boundary_violation_sum = np.sum(v_left**2 + v_right**2 + v_bottom**2 + v_top**2)
    
    # Overlap Violations
    # dist_ij >= r_i + r_j
    # dist_sq_ij >= (r_i + r_j)^2
    # Violation is max(0, r_i + r_j - dist_ij)
    # To avoid sqrt in every iteration if possible? 
    # But sqrt is needed for distance. 
    # Optimization is faster with squared distance if we adjust constraint?
    # Constraint: (x_i - x_j)^2 + (y_i - y_j)^2 >= (r_i + r_j)^2
    # Violation: max(0, (r_i + r_j)^2 - dist_sq_ij)
    # This is differentiable almost everywhere.
    
    overlap_violation_sum = 0.0
    
    # Compute pairwise squared distances and radius sums
    # Using broadcasting for efficiency
    # X matrix (n, 1)
    X = x.reshape(-1, 1)
    Y = y.reshape(-1, 1)
    R = r.reshape(-1, 1)
    
    # Diff matrices
    dx = X - X.T
    dy = Y - Y.T
    dr_sum = R + R.T
    
    # Squared distances
    dist_sq = dx**2 + dy**2
    
    # Radius sum squared
    r_sum_sq = dr_sum**2
    
    # We only need upper triangle to avoid double counting and self
    # But matrix ops are fast enough for n=26 (26x26 = 676 ops)
    
    # Violation: if r_sum_sq > dist_sq, penalty
    # violation = max(0, r_sum_sq - dist_sq)
    # We need to sum these.
    
    # Mask for upper triangle (excluding diagonal)
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    
    diff_sq = r_sum_sq - dist_sq
    violations_ij = np.maximum(0, diff_sq)
    
    overlap_violation_sum = np.sum(violations_ij[mask])
    
    total_violation = boundary_violation_sum + overlap_violation_sum
    
    # The function to minimize needs to return a scalar.
    # We will combine them in the caller with lambda.
    return -sum_radii, total_violation

def solve_packing(n_circles=26, n_restarts=5):
    """
    Solves the circle packing problem using penalty method optimization.
    """
    best_sum_radii = -1.0
    best_centers = None
    best_radii = None
    
    # Helper for the objective function with fixed lambda
    def objective(vars_flat, lam):
        obj_val, violations = compute_objective_and_violations(vars_flat, n_circles)
        return obj_val + lam * violations

    # Define bounds: x, y in [0, 1], r in [0, 0.5]
    # Actually r can't exceed 0.5.
    bounds = []
    for _ in range(n_circles):
        bounds.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])

    # Strategy: Progressive penalty increase
    # Start with small lambda, allow violations, then tighten.
    lambdas = [10, 100, 1000, 5000]
    
    for restart in range(n_restarts):
        # Initialization
        if restart == 0:
            # Structured grid init
            init_x = np.zeros(n_circles)
            init_y = np.zeros(n_circles)
            init_r = np.full(n_circles, 0.05) # Small valid radius
            
            # 5x5 grid points
            idx = 0
            for row in range(5):
                for col in range(5):
                    if idx < n_circles:
                        init_x[idx] = 0.1 + col * 0.2
                        init_y[idx] = 0.1 + row * 0.2
                        idx += 1
            
            # Place remaining circles (if any) in gaps
            # For 26, we have 1 extra. Place at (0.2, 0.2) roughly or just random
            if idx < n_circles:
                # Random valid positions for remaining
                while idx < n_circles:
                    rx = np.random.uniform(0.1, 0.9)
                    ry = np.random.uniform(0.1, 0.9)
                    # Check basic distance to others to avoid immediate huge penalty
                    # Simple check
                    min_dist = 1.0
                    for k in range(idx):
                        d = math.sqrt((init_x[k]-rx)**2 + (init_y[k]-ry)**2)
                        if d < min_dist:
                            min_dist = d
                    
                    if min_dist > 0.1: # 2 * r
                        init_x[idx] = rx
                        init_y[idx] = ry
                        idx += 1
                    else:
                        # If stuck, just random
                        init_x[idx] = np.random.uniform(0.1, 0.9)
                        init_y[idx] = np.random.uniform(0.1, 0.9)
                        idx += 1
        else:
            # Random restart
            init_x = np.random.uniform(0.1, 0.9, n_circles)
            init_y = np.random.uniform(0.1, 0.9, n_circles)
            init_r = np.full(n_circles, 0.05)
            
            # Shuffle to ensure variety
            perm = np.random.permutation(n_circles)
            init_x = init_x[perm]
            init_y = init_y[perm]

        vars_flat = np.concatenate([init_x, init_y, init_r])

        # Optimization loop with increasing penalty
        current_vars = vars_flat.copy()
        
        for lam in lambdas:
            # Minimize
            res = minimize(objective, current_vars, args=(lam,), method='L-BFGS-B', bounds=bounds, 
                           options={'ftol': 1e-9, 'gtol': 1e-6, 'maxiter': 500})
            current_vars = res.x
            
            # Check if we improved the objective (sum radii)
            # We need to extract sum radii from the result
            # Note: res.fun includes penalty, so we recompute actual sum
            _, violations = compute_objective_and_violations(current_vars, n_circles)
            # If violations are small, the obj is close to -sum_r
            # But res.fun = -sum_r + lam * violations.
            # We can just compute -sum_r from vars.
            current_r = current_vars[2::3]
            current_sum_r = np.sum(current_r)
            
            if current_sum_r > best_sum_radii:
                # Validate validity roughly before accepting? 
                # Or just keep track.
                # Let's check violations. If too high, it's invalid.
                if violations < 1e-4: # Tolerance for valid packing
                    best_sum_radii = current_sum_r
                    best_centers = np.column_stack((current_vars[0::3], current_vars[1::3]))
                    best_radii = current_vars[2::3].copy()

    # If best_centers is None (all failed validation), fallback to last result if violations low
    if best_centers is None:
        # Take the last result from the last restart
        _, violations = compute_objective_and_violations(current_vars, n_circles)
        if violations < 1e-3:
            best_centers = np.column_stack((current_vars[0::3], current_vars[1::3]))
            best_radii = current_vars[2::3].copy()
            best_sum_radii = np.sum(best_radii)

    # Final fallback if still None
    if best_centers is None:
        best_centers = np.column_stack((current_vars[0::3], current_vars[1::3]))
        best_radii = current_vars[2::3]
        best_sum_radii = np.sum(best_radii)
        # Clean up slightly if needed? No, just return.

    return best_centers, best_radii, best_sum_radii

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Returns (centers, radii, sum_radii)
    """
    centers, radii, sum_r = solve_packing(n_circles=26, n_restarts=5)
    return centers, radii, sum_r
