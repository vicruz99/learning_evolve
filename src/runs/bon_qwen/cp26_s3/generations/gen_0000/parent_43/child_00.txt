import numpy as np
from scipy.optimize import minimize

def compute_objective(vars, n, penalty_weight):
    """
    Computes the negative sum of radii plus penalty terms for overlaps and boundary violations.
    """
    centers = vars[:2 * n].reshape(n, 2)
    radii = vars[2 * n:]
    
    # Primary objective: maximize sum of radii
    obj = -np.sum(radii)
    
    # Pairwise distance matrix
    diff = centers[:, None, :] - centers[None, :, :]
    dist = np.sqrt(np.sum(diff ** 2, axis=2))
    
    # Lower triangular indices for unique pairs
    idx = np.tril_indices(n, -1)
    r_sum = radii[idx[0]] + radii[idx[1]]
    overlap = r_sum - dist[idx[0], idx[1]]
    penalty_overlap = np.sum(np.maximum(0, overlap) ** 2)
    
    # Boundary penalties
    penalty_bound = 0.0
    penalty_bound += np.sum(np.maximum(0, -centers[:, 0] + radii) ** 2)
    penalty_bound += np.sum(np.maximum(0, centers[:, 0] - 1.0 + radii) ** 2)
    penalty_bound += np.sum(np.maximum(0, -centers[:, 1] + radii) ** 2)
    penalty_bound += np.sum(np.maximum(0, centers[:, 1] - 1.0 + radii) ** 2)
    
    return obj + penalty_weight * (penalty_overlap + penalty_bound)

def obj_stage1(vars):
    return compute_objective(vars, 26, 100.0)

def obj_stage2(vars):
    return compute_objective(vars, 26, 2000.0)

def run_packing():
    n = 26
    r_init = 0.09  # Start slightly loose to ensure feasibility
    centers = []
    y = r_init
    
    # Hexagonal row structure: 6, 5, 6, 5, 4 circles
    row_counts = [6, 5, 6, 5, 4]
    for row, count in enumerate(row_counts):
        width = (count - 1) * 2 * r_init
        x_start = (1.0 - width) / 2
        shift = (row % 2) * r_init
        for col in range(count):
            x = x_start + col * 2 * r_init + shift
            centers.append([x, y])
        y += r_init * np.sqrt(3)
        
    centers = np.array(centers)
    
    # Center vertically in the square
    y_max = centers[:, 1].max()
    centers[:, 1] += (1.0 - y_max) / 2 - r_init
    radii = np.full(n, r_init)
    
    # Small random perturbation to break symmetry
    np.random.seed(42)
    centers += np.random.randn(n, 2) * 0.002
    radii += np.random.randn(n) * 0.001
    radii = np.maximum(radii, 0.01)
    
    x0 = np.concatenate([centers.ravel(), radii])
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
    
    # Stage 1: Exploration & Expansion
    res1 = minimize(obj_stage1, x0, method='L-BFGS-B', bounds=bounds, 
                    options={'maxiter': 3000, 'ftol': 1e-12, 'gtol': 1e-10})
    
    # Stage 2: Strict Feasibility Refinement
    res2 = minimize(obj_stage2, res1.x, method='L-BFGS-B', bounds=bounds, 
                    options={'maxiter': 3000, 'ftol': 1e-12, 'gtol': 1e-10})
    
    final_centers = res2.x[:2 * n].reshape(n, 2)
    final_radii = res2.x[2 * n:]
    
    # Ensure strict non-negativity
    final_radii = np.maximum(final_radii, 0.0)
    
    # Clip centers to respect boundaries explicitly
    final_centers[:, 0] = np.clip(final_centers[:, 0], final_radii, 1.0 - final_radii)
    final_centers[:, 1] = np.clip(final_centers[:, 1], final_radii, 1.0 - final_radii)
    
    sum_r = float(np.sum(final_radii))
    return final_centers, final_radii, sum_r
