import numpy as np
from scipy.optimize import minimize

def objective(vars, n):
    """Objective function: maximize sum of radii (minimize negative sum)"""
    return -np.sum(vars[2*n:])

def gradient(vars, n):
    """Analytical gradient of the objective function"""
    grad = np.zeros_like(vars)
    grad[2*n:] = -1.0
    return grad

def constraint_func(vars, n):
    """
    Returns array of constraint slacks. All must be >= 0.
    Includes pairwise non-overlap and boundary containment constraints.
    """
    c = vars[:2*n].reshape(n, 2)
    r = vars[2*n:]
    
    # Pairwise constraints: dist_ij - r_i - r_j >= 0
    diff = c[:, None, :] - c[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    r_sum = r[:, None] + r[None, :]
    slacks_pair = dists - r_sum
    
    # Extract lower triangle to avoid duplicate constraints
    idx = np.tril_indices(n, -1)
    slacks = slacks_pair[idx]
    
    # Boundary constraints: r_i <= dist_to_boundary
    # x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
    slacks_bound = np.concatenate([
        c[:, 0] - r,
        1.0 - c[:, 0] - r,
        c[:, 1] - r,
        1.0 - c[:, 1] - r
    ])
    
    return np.concatenate([slacks, slacks_bound])

def run_packing():
    n = 26
    r_init = 0.075
    centers = []
    radii = []
    
    # Hexagonal packing layout: 5, 6, 5, 6, 4 circles per row
    rows = [5, 6, 5, 6, 4]
    row_idx = 0
    for count in rows:
        y = r_init * (1 + row_idx * np.sqrt(3))
        x_start = r_init if row_idx % 2 == 0 else 2 * r_init
        for k in range(count):
            x = x_start + k * 2 * r_init
            centers.append([x, y])
            radii.append(r_init)
        row_idx += 1
        
    centers = np.array(centers)
    radii = np.array(radii)
    
    # Center the configuration in the unit square to allow symmetric expansion
    x_min, x_max = centers[:, 0].min(), centers[:, 0].max()
    y_min, y_max = centers[:, 1].min(), centers[:, 1].max()
    
    cx, cy = 0.5, 0.5
    centers[:, 0] += cx - (x_min + x_max) / 2
    centers[:, 1] += cy - (y_min + y_max) / 2
    
    # Flatten to optimizer variable vector: [x1, y1, r1, x2, y2, r2, ...]
    x0 = np.hstack([centers.flatten(), radii])
    
    # Define constraints and bounds
    cons = {'type': 'ineq', 'fun': constraint_func, 'args': (n,)}
    bounds = [(0.0, 1.0) if i % 3 < 2 else (0.0, 0.5) for i in range(3*n)]
    
    # Run constrained optimization
    res = minimize(objective, x0, jac=gradient, method='SLSQP', 
                   bounds=bounds, constraints=cons, args=(n,),
                   options={'maxiter': 5000, 'ftol': 1e-10, 'disp': False})
    
    # Extract and clean results
    final_centers = res.x[:2*n].reshape(n, 2)
    final_radii = np.maximum(res.x[2*n:], 0.0)
    
    return final_centers, final_radii, np.sum(final_radii)
