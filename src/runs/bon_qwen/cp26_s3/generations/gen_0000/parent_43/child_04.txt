import numpy as np
from scipy.optimize import minimize
import math

NUM_CIRCLES = 26
UNIT_SQUARE_SIZE = 1.0

def objective_function(x):
    """
    Objective function: Minimize negative sum of radii.
    x is a flattened array [x0, y0, r0, x1, y1, r1, ...]
    """
    radii = x[2::3]
    return -np.sum(radii)

def constraints_function(x):
    """
    Returns a vector of constraint values. All must be >= 0.
    Constraints:
    1. Boundary constraints for each circle (4 per circle)
       x_i - r_i >= 0
       1 - x_i - r_i >= 0
       y_i - r_i >= 0
       1 - y_i - r_i >= 0
    2. Distance constraints for each pair (N*(N-1)/2)
       (xi-xj)^2 + (yi-yj)^2 - (ri+rj)^2 >= 0
    """
    n = NUM_CIRCLES
    num_dist_constraints = n * (n - 1) // 2
    num_constraints = 4 * n + num_dist_constraints
    constraints_vals = np.zeros(num_constraints)
    
    # Boundary constraints
    idx = 0
    for i in range(n):
        base = 3 * i
        xi = x[base]
        yi = x[base+1]
        ri = x[base+2]
        
        constraints_vals[idx] = xi - ri
        constraints_vals[idx+1] = UNIT_SQUARE_SIZE - xi - ri
        constraints_vals[idx+2] = yi - ri
        constraints_vals[idx+3] = UNIT_SQUARE_SIZE - yi - ri
        idx += 4
        
    # Distance constraints
    for i in range(n):
        base_i = 3 * i
        xi = x[base_i]
        yi = x[base_i+1]
        ri = x[base_i+2]
        
        for j in range(i + 1, n):
            base_j = 3 * j
            xj = x[base_j]
            yj = x[base_j+1]
            rj = x[base_j+2]
            
            dx = xi - xj
            dy = yi - yj
            dr = ri + rj
            
            constraints_vals[idx] = dx*dx + dy*dy - dr*dr
            idx += 1
            
    return constraints_vals

def get_bounds():
    """
    Returns bounds for the optimization variables.
    x, y in [0, 1]
    r in [0, 0.5]
    """
    bnds = []
    for _ in range(NUM_CIRCLES):
        bnds.append((0.0, UNIT_SQUARE_SIZE)) # x
        bnds.append((0.0, UNIT_SQUARE_SIZE)) # y
        bnds.append((0.0, 0.5))              # r
    return bnds

def generate_hexagonal_centers(n):
    """
    Generates initial center positions based on a hexagonal lattice.
    """
    centers = []
    # Estimate spacing to fit n circles in unit square roughly
    # Area per circle ~ 1/n. Hexagonal density factor.
    # Spacing s approx sqrt(2 / (n * sqrt(3)))
    s = math.sqrt(2.0 / (n * math.sqrt(3.0))) * 1.1 # 1.1 for padding
    
    y = 0.1
    row = 0
    while len(centers) < n:
        offset = (row % 2) * (s / 2.0)
        x = 0.1 + offset
        while x < 0.9 and len(centers) < n:
            centers.append([x, y])
            x += s
        y += s * math.sqrt(3.0) / 2.0
        row += 1
        if y > 1.0:
            # If we exceed vertical bounds, try to shift or break
            # For a robust start, just reset y or break
            break
            
    return np.array(centers)

def generate_random_centers(n):
    """
    Generates random initial center positions.
    """
    return np.random.uniform(0.1, 0.9, size=(n, 2))

def run_packing():
    """
    Main function to run the optimization and return the best packing.
    """
    best_sum_radii = -1.0
    best_centers = None
    best_radii = None
    
    # Define constraints dictionary for SLSQP
    # We use a single function returning a vector
    cons = {'type': 'ineq', 'fun': constraints_function}
    bounds = get_bounds()
    
    # Try multiple initialization strategies
    # 1. Hexagonal grid
    # 2. Random grids (multiple seeds)
    
    strategies = []
    strategies.append(generate_hexagonal_centers(NUM_CIRCLES))
    
    # Add 5 random strategies
    for seed in range(5):
        np.random.seed(seed * 12345)
        strategies.append(generate_random_centers(NUM_CIRCLES))

    for centers in strategies:
        # Construct initial x vector
        x0 = np.zeros(3 * NUM_CIRCLES)
        for i in range(NUM_CIRCLES):
            x0[3*i] = centers[i, 0]
            x0[3*i+1] = centers[i, 1]
            x0[3*i+2] = 0.01 # Start with small radius to be feasible
            
        try:
            # Run optimization
            res = minimize(objective_function, x0, method='SLSQP', bounds=bounds, constraints=cons, 
                           options={'ftol': 1e-9, 'maxiter': 1000})
            
            if res.success:
                current_sum = -res.fun # fun returns negative sum
                if current_sum > best_sum_radii:
                    best_sum_radii = current_sum
                    # Extract best solution

                    best_radii = np.zeros(NUM_CIRCLES)

                        best_centers[i, 0] = res.x[3*i]
                        best_centers[i, 1] = res.x[3*i+1]
                        best_radii[i] = res.x[3*i+2]
        except Exception as e:
            continue

    # If no success, fallback to a safe simple packing (though optimizer should find better)
    if best_centers is None:
        # Fallback: Grid
        best_centers = np.zeros((NUM_CIRCLES, 2))
        best_radii = np.zeros(NUM_CIRCLES)
        # Simple grid placement
        count = 0
        for r in range(6):
            for c in range(6):
                if count < NUM_CIRCLES:
                    best_centers[count] = [0.1 + c * 0.15, 0.1 + r * 0.15]
                    best_radii[count] = 0.05
                    count += 1
        best_sum_radii = np.sum(best_radii)

    return best_centers, best_radii, best_sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]
    if np.isnan(centers).any():
        return False
    if np.isnan(radii).any():
        return False
    for i in range(n):
        if radii[i] < 0:
            return False
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            return False
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False
    return True

if __name__ == "__main__":
    centers, radii, sum_r = run_packing()
    print(f"Sum of radii: {sum_r}")
    print(f"Valid: {validate_packing(centers, radii)}")
