import numpy as np
from scipy.optimize import minimize

def objective(v):
    """Objective: minimize negative sum of radii"""
    return -sum(v[i*3+2] for i in range(26))

def all_constraints(v):
    """
    Returns inequality constraints >= 0:
    - Boundary constraints for all circles
    - Non-overlap constraints for all pairs
    """
    n = 26
    c = []
    # Boundary constraints
    for i in range(n):
        c.append(v[i*3] - v[i*3+2])
        c.append(1 - (v[i*3] + v[i*3+2]))
        c.append(v[i*3+1] - v[i*3+2])
        c.append(1 - (v[i*3+1] + v[i*3+2]))
    # Overlap constraints
    for i in range(n):
        for j in range(i+1, n):
            dx = v[i*3] - v[j*3]
            dy = v[i*3+1] - v[j*3+1]
            dr = v[i*3+2] + v[j*3+2]
            c.append(dx*dx + dy*dy - dr*dr)
    return np.array(c)

def run_packing():
    n = 26
    # Initial hexagonal arrangement: rows of 6, 5, 6, 5, 4
    row_counts = [6, 5, 6, 5, 4]
    r_init = 0.09
    
    centers = []
    radii = []
    y = r_init
    for idx, count in enumerate(row_counts):
        # Stagger odd rows by one radius horizontally
        x_start = r_init if idx % 2 == 0 else 2 * r_init
        for j in range(count):
            centers.append([x_start + j * 2 * r_init, y])
            radii.append(r_init)
        y += r_init * np.sqrt(3)
        
    centers = np.array(centers)
    radii = np.array(radii)
    
    # Flatten variables: [x0, y0, r0, x1, y1, r1, ..., x25, y25, r25]
    x0 = np.concatenate([centers.flatten(), radii])
    bounds = [(0, 1) for _ in range(n * 3)]
    cons = {'type': 'ineq', 'fun': all_constraints}
    
    # Optimize
    res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, 
                   options={'ftol': 1e-12, 'maxiter': 2000, 'disp': False})
                   
    final_x = res.x
    
    # Extract results
    final_centers = np.array([[final_x[i*3], final_x[i*3+1]] for i in range(n)])
    final_radii = np.array([final_x[i*3+2] for i in range(n)])
    
    # Ensure non-negative radii
    final_radii = np.maximum(final_radii, 1e-9)
    
    # Clip centers to respect boundaries strictly
    final_centers[:, 0] = np.clip(final_centers[:, 0], final_radii, 1 - final_radii)
    final_centers[:, 1] = np.clip(final_centers[:, 1], final_radii, 1 - final_radii)
    
    # Post-processing: resolve any residual overlaps from numerical tolerance
    for _ in range(10):
        overlap = False
        for i in range(n):
            for j in range(i+1, n):
                dist = np.sqrt(np.sum((final_centers[i] - final_centers[j])**2))
                if dist < final_radii[i] + final_radii[j] - 1e-12:
                    final_radii[i] *= 0.999
                    final_radii[j] *= 0.999
                    overlap = True
        if not overlap:
            break
            
    return final_centers, final_radii, np.sum(final_radii)
