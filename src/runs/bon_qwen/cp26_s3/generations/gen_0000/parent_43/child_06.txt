def solve_lp(centers):
    """
    Solves the Linear Programming problem to find optimal radii for fixed centers.
    Maximizes sum(r_i) subject to:
    1. r_i >= 0
    2. r_i <= distance to boundary
    3. r_i + r_j <= distance between centers
    """
    n = centers.shape[0]
    if n == 0:
        return 0.0, np.array([])

    # Objective: maximize sum(r) => minimize -sum(r)
    c_obj = -np.ones(n)

    # Constraints: A_ub * r <= b_ub
    # We will construct the constraints list
    constraints_A = []
    constraints_b = []

    # Precompute distances and boundaries
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    boundaries = np.array([
        min(c[0], 1 - c[0], c[1], 1 - c[1]) for c in centers
    ])

    # Pairwise distances
    # dists[i, j] is distance between i and j
    # We only need upper triangle
    dists = np.linalg.norm(centers[:, None] - centers[None, :], axis=2)

    # 1. Boundary constraints: 1 * r_i <= boundary_i
    for i in range(n):
        row = np.zeros(n)
        row[i] = 1.0
        constraints_A.append(row)
        constraints_b.append(boundaries[i])

    # 2. Pairwise constraints: 1 * r_i + 1 * r_j <= dists[i, j]
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            constraints_A.append(row)
            constraints_b.append(dists[i, j])

    A_ub = np.array(constraints_A)
    b_ub = np.array(constraints_b)

    # Bounds for r_i: [0, infinity)
    bounds = [(0, None) for _ in range(n)]

    # Solve LP
    # Using 'highs' method which is efficient
    try:
        res = opt.linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return -res.fun, res.x
        else:
            # Fallback if LP fails (should not happen for valid centers)
            return 0.0, np.zeros(n)
    except Exception:
        return 0.0, np.zeros(n)
