import numpy as np
from scipy.optimize import minimize


def penalty_objective(x, n, penalty_weight):
    centers = x[:2 * n].reshape(n, 2)
    radii = x[2 * n:]
    obj = -np.sum(radii)
    penalty = 0.0
    for i in range(n):
        v1 = max(0.0, radii[i] - centers[i, 0])
        penalty += penalty_weight * v1 * v1
        v2 = max(0.0, centers[i, 0] + radii[i] - 1.0)
        penalty += penalty_weight * v2 * v2
        v3 = max(0.0, radii[i] - centers[i, 1])
        penalty += penalty_weight * v3 * v3
        v4 = max(0.0, centers[i, 1] + radii[i] - 1.0)
        penalty += penalty_weight * v4 * v4
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            min_dist = radii[i] + radii[j]
            if dist < min_dist:
                violation = min_dist - dist
                penalty += penalty_weight * violation * violation
    return obj + penalty


def hex_init(n):
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.08)
    idx = 0
    row_spacing = 1.0 / 5.5
    col_spacing = row_spacing * np.sqrt(3) * 0.85
    for row in range(6):
        for col in range(5):
            if idx >= n:
                break
            x = col * col_spacing + 0.05 + (row % 2) * (col_spacing * 0.5)
            y = row * row_spacing + 0.05
            centers[idx] = [np.clip(x, 0.02, 0.98), np.clip(y, 0.02, 0.98)]
            idx += 1
    while idx < n:
        centers[idx] = [0.5, 0.5]
        idx += 1
    return centers, radii


def grid_init(n):
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.07)
    cols = int(np.ceil(np.sqrt(n)))
    rows = int(np.ceil(n / cols))
    idx = 0
    for row in range(rows):
        for col in range(cols):
            if idx >= n:
                break
            x = (col + 0.5) / cols
            y = (row + 0.5) / rows
            centers[idx] = [x, y]
            idx += 1
    while idx < n:
        centers[idx] = [0.5, 0.5]
        idx += 1
    return centers, radii


def random_init(n, seed):
    np.random.seed(seed)
    centers = np.random.rand(n, 2) * 0.8 + 0.1
    radii = np.full(n, 0.04)
    return centers, radii


def force_refine(centers, radii, n, iterations=2000):
    centers = centers.copy()
    radii = radii.copy()
    for it in range(iterations):
        cooling = max(0.01, 1.0 - it / iterations)
        forces = np.zeros((n, 2))
        max_growth = np.zeros(n)
        for i in range(n):
            max_r = min(centers[i, 0], 1.0 - centers[i, 0],
                        centers[i, 1], 1.0 - centers[i, 1])
            for j in range(n):
                if i != j:
                    dx = centers[i] - centers[j]
                    dist = np.sqrt(dx[0] * dx[0] + dx[1] * dx[1])
                    max_r = min(max_r, dist - radii[j])
            max_growth[i] = max(max_r - radii[i], 0.0)
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i] - centers[j]
                dist = np.sqrt(dx[0] * dx[0] + dx[1] * dx[1])
                min_dist = radii[i] + radii[j]
                if dist < min_dist and dist > 1e-12:
                    push = (min_dist - dist) / dist
                    forces[i] += push * dx
                    forces[j] -= push * dx
        for i in range(n):
            if centers[i, 0] - radii[i] < 0:
                forces[i, 0] += 10 * (radii[i] - centers[i, 0])
            if centers[i, 0] + radii[i] > 1:
                forces[i, 0] -= 10 * (centers[i, 0] + radii[i] - 1)
            if centers[i, 1] - radii[i] < 0:
                forces[i, 1] += 10 * (radii[i] - centers[i, 1])
            if centers[i, 1] + radii[i] > 1:
                forces[i, 1] -= 10 * (centers[i, 1] + radii[i] - 1)
        lr = 0.5 * cooling
        centers += lr * forces
        for i in range(n):
            if max_growth[i] > 0:
                radii[i] += max_growth[i] * 0.05 * cooling
        for i in range(n):
            centers[i, 0] = np.clip(centers[i, 0], radii[i], 1.0 - radii[i])
            centers[i, 1] = np.clip(centers[i, 1], radii[i], 1.0 - radii[i])
    return centers, radii


def run_packing():
    n = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None
    bounds = [(0.0, 1.0)] * 2 * n + [(1e-10, 0.5)] * n
    all_configs = []
    np.random.seed(0)
    np.random.seed(0)
    for trial in range(15):
        if trial < 5:
            centers, radii = hex_init(n)
            np.random.seed(trial * 7 + 1)
            centers += np.random.randn(n, 2) * 0.005
        elif trial < 10:
            centers, radii = grid_init(n)
            np.random.seed(trial * 11 + 2)
            centers += np.random.randn(n, 2) * 0.01
        else:
            centers, radii = random_init(n, trial * 13 + 3)
        x0 = np.concatenate([centers.ravel(), radii])
        for pw in [500, 2000, 8000, 20000]:
            def wrapped_obj(x, _n=n, _pw=pw):
                return penalty_objective(x, _n, _pw)
            result = minimize(wrapped_obj, x0, method='L-BFGS-B', bounds=bounds,
                              options={'maxiter': 5000, 'ftol': 1e-15, 'gtol': 1e-10})
            x0 = result.x.copy()
        centers_opt = result.x[:2 * n].reshape(n, 2)
        radii_opt = np.maximum(result.x[2 * n:], 1e-10)
        for i in range(n):
            centers_opt[i, 0] = np.clip(centers_opt[i, 0], radii_opt[i], 1.0 - radii_opt[i])
            centers_opt[i, 1] = np.clip(centers_opt[i, 1], radii_opt[i], 1.0 - radii_opt[i])
        centers_opt, radii_opt = force_refine(centers_opt, radii_opt, n, iterations=500)
        current_sum = np.sum(radii_opt)
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers_opt.copy()
            best_radii = radii_opt.copy()
    return best_centers, best_radii, best_sum
