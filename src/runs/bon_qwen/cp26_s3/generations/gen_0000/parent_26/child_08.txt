import numpy as np
from scipy.optimize import minimize

def run_packing():
    """
    Optimizes the positions and radii of 26 circles in a unit square
    to maximize the sum of radii.
    """
    n = 26
    np.random.seed(42) # For reproducibility

    # Function to generate a hexagonal grid initial guess
    def generate_initial_guess(n):
        # Estimate a starting radius. For 26 circles, roughly 5x5 grid implies r ~ 0.1.
        # We start slightly smaller to be safe.
        r_init = 0.095
        
        # Hexagonal packing parameters
        # Horizontal spacing = 2*r
        # Vertical spacing = sqrt(3)*r
        # Try to fit roughly sqrt(n) rows/cols
        cols = int(np.ceil(np.sqrt(n)))
        rows = int(np.ceil(n / cols))
        
        # Adjust density to fit in 1x1
        # Width constraint: cols * 2*r <= 1  => r <= 1/(2*cols)
        # Height constraint: (rows-1)*sqrt(3)*r + 2*r <= 1 => r <= 1 / (2 + (rows-1)*sqrt(3))
        
        r_x = 1.0 / (2 * cols)
        r_y = 1.0 / (2 + (rows - 1) * np.sqrt(3))
        
        r_start = min(r_x, r_y) * 0.9 # 90% of theoretical max for safety
        
        centers = []
        radii = []
        count = 0
        
        # Generate hexagonal grid
        for row in range(rows):
            for col in range(cols):
                if count >= n:
                    break
                x = (col + 0.5) * 2 * r_start
                y = (row + 0.5) * np.sqrt(3) * r_start
                
                # Stagger odd rows
                if row % 2 == 1:
                    x += r_start
                
                # Clamp to bounds [r_start, 1-r_start]
                x = np.clip(x, r_start, 1 - r_start)
                y = np.clip(y, r_start, 1 - r_start)
                
                centers.append([x, y])
                radii.append(r_start)
                count += 1
            if count >= n:
                break
                
        return np.array(centers), np.array(radii)

    # Constraint definitions
    def boundary_constraints(x_vars, n):
        # x_vars is flattened [x0, y0, r0, x1, y1, r1, ...]
        # Returns list of constraint dictionaries
        cons = []
        for i in range(n):
            idx_x = 3 * i
            idx_y = 3 * i + 1
            idx_r = 3 * i + 2
            
            # x_i - r_i >= 0
            cons.append({
                'type': 'ineq',
                'fun': lambda x_vars, i=i: x_vars[idx_x] - x_vars[idx_r]
            })
            # 1 - x_i - r_i >= 0
            cons.append({
                'type': 'ineq',
                'fun': lambda x_vars, i=i: 1.0 - x_vars[idx_x] - x_vars[idx_r]
            })
            # y_i - r_i >= 0
            cons.append({
                'type': 'ineq',
                'fun': lambda x_vars, i=i: x_vars[idx_y] - x_vars[idx_r]
            })
            # 1 - y_i - r_i >= 0
            cons.append({
                'type': 'ineq',
                'fun': lambda x_vars, i=i: 1.0 - x_vars[idx_y] - x_vars[idx_r]
            })
        return cons

    def overlap_constraints(x_vars, n):
        cons = []
        for i in range(n):
            for j in range(i + 1, n):
                idx_xi, idx_yi, idx_ri = 3*i, 3*i+1, 3*i+2
                idx_xj, idx_yj, idx_rj = 3*j, 3*j+1, 3*j+2
                
                # dist^2 - (ri + rj)^2 >= 0
                def constraint_func(x_vars, i=i, j=j):
                    xi, yi, ri = x_vars[idx_xi], x_vars[idx_yi], x_vars[idx_ri]
                    xj, yj, rj = x_vars[idx_xj], x_vars[idx_yj], x_vars[idx_rj]
                    dist_sq = (xi - xj)**2 + (yi - yj)**2
                    sum_r = ri + rj
                    return dist_sq - sum_r**2
                
                cons.append({
                    'type': 'ineq',
                    'fun': constraint_func
                })
        return cons

    # Objective function: Minimize -sum(radii)
    def objective(x_vars):
        radii = x_vars[2::3]
        return -np.sum(radii)

    # Bounds for variables: x, y in [0, 1], r in [0, 0.5]
    bounds = [(0, 1), (0, 1), (0, 0.5)] * n

    # Run optimization multiple times to find best global optimum
    best_result = None
    best_val = -np.inf
    best_centers = None
    best_radii = None

    # We will run a few iterations with random perturbations on initial guess
    for attempt in range(3):
        centers_init, radii_init = generate_initial_guess(n)
        
        # Add small random noise to break symmetry
        noise = np.random.normal(0, 0.01, centers_init.shape)
        centers_init = np.clip(centers_init + noise, 0, 1)
        
        # Flatten variables
        x0 = np.zeros(3 * n)
        x0[0::3] = centers_init[:, 0]
        x0[1::3] = centers_init[:, 1]
        x0[2::3] = radii_init
        
        # Define constraints
        # Note: Creating constraints inside loop is expensive, but for 26 circles it's manageable
        # To speed up, we could define them once, but they depend on n which is constant.
        # However, lambda closures capture i, j correctly if defined carefully.
        # Let's define them once outside if possible, but n is fixed.
        
        cons = []
        cons.extend(boundary_constraints(None, n)) # dummy arg
        # Boundary constraints don't depend on variable values, just indices.
        # But the lambda needs to access x_vars.
        # Re-defining boundary constraints logic cleanly:
        boundary_cons_list = []
        for i in range(n):
            boundary_cons_list.append({'type': 'ineq', 'fun': lambda x, i=i: x[3*i] - x[3*i+2]})
            boundary_cons_list.append({'type': 'ineq', 'fun': lambda x, i=i: 1.0 - x[3*i] - x[3*i+2]})
            boundary_cons_list.append({'type': 'ineq', 'fun': lambda x, i=i: x[3*i+1] - x[3*i+2]})
            boundary_cons_list.append({'type': 'ineq', 'fun': lambda x, i=i: 1.0 - x[3*i+1] - x[3*i+2]})
        
        overlap_cons_list = []
        for i in range(n):
            for j in range(i+1, n):
                # Indices
                idx_i = 3*i
                idx_j = 3*j
                overlap_cons_list.append({
                    'type': 'ineq',
                    'fun': lambda x, i=idx_i, j=idx_j: (x[i]-x[j])**2 + (x[i+1]-x[j+1])**2 - (x[i+2]+x[j+2])**2
                })
        
        all_constraints = boundary_cons_list + overlap_cons_list

        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=all_constraints, options={'maxiter': 200, 'ftol': 1e-9})
            
            if res.success:
                current_sum = -res.fun
                if current_sum > best_val:
                    best_val = current_sum
                    best_result = res
        except Exception as e:
            print(f"Optimization failed in attempt {attempt}: {e}")
            continue

    if best_result is None:
        # Fallback to initial guess if optimization fails completely
        centers, radii = generate_initial_guess(n)
        return centers, radii, np.sum(radii)

    # Extract best solution
    x_opt = best_result.x
    centers_opt = np.zeros((n, 2))
    radii_opt = np.zeros(n)
    
    for i in range(n):
        centers_opt[i, 0] = x_opt[3*i]
        centers_opt[i, 1] = x_opt[3*i+1]
        radii_opt[i] = x_opt[3*i+2]
    
    sum_radii = np.sum(radii_opt)
    
    return centers_opt, radii_opt, sum_radii
