import numpy as np

def compute_constraints(centers, radii):
    """
    Computes constraint violations for a given packing configuration.
    Returns an array of constraint values. We want all values <= 0 for feasibility.
    Specifically:
    - Boundary constraints: r - x <= 0, x - (1-r) <= 0, r - y <= 0, y - (1-r) <= 0
    - Overlap constraints: (r_i + r_j)^2 - dist^2 <= 0
    """
    n = centers.shape[0]
    constraints_list = []

    # Boundary constraints
    # x - r >= 0  => r - x <= 0
    # x + r <= 1  => x + r - 1 <= 0
    # y - r >= 0  => r - y <= 0
    # y + r <= 1  => y + r - 1 <= 0
    
    # Using a slightly relaxed bound for numerical stability in optimization, 
    # but validation is strict.
    
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        
        # Boundary: r <= x <= 1-r  =>  r - x <= 0  AND  x + r - 1 <= 0
        constraints_list.append(r - x)
        constraints_list.append(x + r - 1)
        
        # Boundary: r <= y <= 1-r  =>  r - y <= 0  AND  y + r - 1 <= 0
        constraints_list.append(r - y)
        constraints_list.append(y + r - 1)

    # Overlap constraints
    # (ri + rj)^2 <= (xi - xj)^2 + (yi - yj)^2
    # => (ri + rj)^2 - dist^2 <= 0
    for i in range(n):
        for j in range(i + 1, n):
            dist_sq = np.sum((centers[i] - centers[j]) ** 2)
            r_sum = radii[i] + radii[j]
            constraints_list.append(r_sum**2 - dist_sq)

    return np.array(constraints_list)

def objective_function(params, n):

    centers = params[0:2*n].reshape(n, 2)
    radii = params[2*n:]
    return -np.sum(radii)
