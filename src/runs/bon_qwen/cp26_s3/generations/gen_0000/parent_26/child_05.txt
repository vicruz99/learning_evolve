import numpy as np
from scipy.optimize import minimize

# Global constant for number of circles
N_CIRCLES = 26

def _pair_constraint(v, i, j):
    """
    Constraint: distance between centers >= sum of radii
    Returns dist - (r_i + r_j) >= 0
    """
    xi, yi = v[2*i], v[2*i+1]
    xj, yj = v[2*j], v[2*j+1]
    ri = v[2*N_CIRCLES + i]
    rj = v[2*N_CIRCLES + j]
    
    dist = np.sqrt((xi - xj)**2 + (yi - yj)**2)
    return dist - (ri + rj)

def _boundary_x(v, i):
    """Constraint: r_i <= x_i"""
    return v[2*i] - v[2*N_CIRCLES + i]

def _boundary_1x(v, i):
    """Constraint: r_i <= 1 - x_i"""
    return (1 - v[2*i]) - v[2*N_CIRCLES + i]

def _boundary_y(v, i):
    """Constraint: r_i <= y_i"""
    return v[2*i+1] - v[2*N_CIRCLES + i]

def _boundary_1y(v, i):
    """Constraint: r_i <= 1 - y_i"""
    return (1 - v[2*i+1]) - v[2*N_CIRCLES + i]

    n = N_CIRCLES
    
    # --- Stage 1: Initialization ---
    # Generate points on a hexagonal lattice to start with a dense packing
    s = 0.18 # Approximate spacing
    points = []
    # Iterate over a grid to find points inside [0,1]x[0,1]
    for i in range(-10, 20):
        for j in range(-10, 20):
            x = i * s + j * 0.5 * s + 0.5
            y = j * (np.sqrt(3)/2) * s + 0.5
            if 0 <= x <= 1 and 0 <= y <= 1:
                points.append([x, y])
    
    # Shuffle to avoid bias from generation order and pick n points
    np.random.seed(42)
    np.random.shuffle(points)
    
    if len(points) >= n:
        init_centers = np.array(points[:n])
    else:

        init_centers = np.random.uniform(0.1, 0.9, (n, 2))
        
    # Initial radii guess
    radii_init = np.full(n, 0.05)
    
    # Flatten variables: [x0, y0, ..., xn, yn, r0, ..., rn]
    x0 = np.concatenate([init_centers.flatten(), radii_init])
    
    # --- Stage 2: Optimization ---
    
    def objective(vars):
        # Maximize sum of radii => Minimize negative sum
        return -np.sum(vars[2*n:])

    constraints_list = []
    
    # Pairwise non-overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            constraints_list.append({
                'type': 'ineq', 
                'fun': lambda v, i=i, j=j: _pair_constraint(v, i, j)

        constraints_list.append({'type': 'ineq', 'fun': lambda v, i=i: _boundary_y(v, i)})
