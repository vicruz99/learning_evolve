import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    N = 26
    
    # Generate diverse initial configurations
    def generate_inits():
        inits = []
        
        # 1. 5x5 Grid + 1 center
        cx, cy = np.meshgrid(np.linspace(0.1, 0.9, 5), np.linspace(0.1, 0.9, 5))
        centers = np.concatenate([cx.ravel().reshape(-1, 1), cy.ravel().reshape(-1, 1)], axis=1)
        inits.append(np.column_stack([centers, np.full(N, 0.01)]))
        
        # 2. Random Initialization
        for _ in range(5):
            r_centers = np.random.rand(N, 2) * 0.8 + 0.1
            inits.append(np.column_stack([r_centers, np.full(N, 0.01)]))
            
        return inits

    def objective(var):
        # Minimize negative sum of radii to maximize sum
        r = var[2::3]
        return -np.sum(r)

    def constraints_factory(var, N):
        con = []
        x, y, r = var[0::3], var[1::3], var[2::3]
        
        # Boundary constraints
        for i in range(N):
            con.append({'type': 'ineq', 'fun': lambda v, idx=i: v[3*idx] - v[3*idx+2]})
            con.append({'type': 'ineq', 'fun': lambda v, idx=i: 1 - v[3*idx] - v[3*idx+2]})
            con.append({'type': 'ineq', 'fun': lambda v, idx=i: v[3*idx+1] - v[3*idx+2]})
            con.append({'type': 'ineq', 'fun': lambda v, idx=i: 1 - v[3*idx+1] - v[3*idx+2]})
        
        # Pairwise non-overlap constraints (squared)
        for i in range(N):
            for j in range(i + 1, N):
                def dist_con(v, i=i, j=j):
                    return (v[3*i] - v[3*j])**2 + (v[3*i+1] - v[3*j+1])**2 - (v[3*i+2] + v[3*j+2])**2
                con.append({'type': 'ineq', 'fun': dist_con})
        return con

    bounds = [(0, 1) if i % 3 != 2 else (0, 0.5) for i in range(3 * N)]
    constraints = constraints_factory(np.zeros(3*N), N)
    
    best_val = -np.inf
    best_var = None
    
    for init in generate_inits():
        res = minimize(objective, init.flatten(), method='SLSQP', bounds=bounds, 
                       constraints=constraints, options={'ftol': 1e-8, 'maxiter': 1000})
        if res.fun < best_val:
            best_val = res.fun
            best_var = res.x

    centers = np.column_stack([best_var[0::3], best_var[1::3]])
    radii = best_var[2::3]
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
