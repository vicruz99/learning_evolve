import numpy as np
from scipy.optimize import linprog

def get_max_radii(centers):
    """
    Given centers, solve the LP to find the maximum possible radii.
    Maximizes sum(r_i) subject to r_i + r_j <= dist(i,j) and boundary constraints.
    """
    n = centers.shape[0]
    # Objective: maximize sum(r_i) => minimize -sum(r_i)
    c_obj = -np.ones(n)
    
    # Constraints matrix A_ub @ r <= b_ub
    # 1. Pairwise constraints: r_i + r_j <= dist(i, j)
    # 2. Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    
    num_pairwise = n * (n - 1) // 2
    num_boundary = 4 * n
    num_constraints = num_pairwise + num_boundary
    
    A = np.zeros((num_constraints, n))
    b = np.zeros(num_constraints)
    
    idx = 0
    
    # Pairwise constraints
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = dist
            idx += 1
            
    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        # r_i <= x
        A[idx, i] = 1.0
        b[idx] = x
        idx += 1
        # r_i <= 1 - x
        A[idx, i] = 1.0
        b[idx] = 1.0 - x
        idx += 1
        # r_i <= y
        A[idx, i] = 1.0
        b[idx] = y
        idx += 1
        # r_i <= 1 - y
        A[idx, i] = 1.0
        b[idx] = 1.0 - y
        idx += 1
        
    # Bounds for r_i >= 0
    bounds = [(0, None) for _ in range(n)]
    
    # Solve LP
    # Using 'highs' method is generally fast and robust
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success:
            return -res.fun, res.x
        else:
            # Fallback or error handling if LP fails (should not happen with valid centers)
            return 0.0, np.zeros(n)
    except Exception:
        return 0.0, np.zeros(n)
