def solve_radii_lp(centers):
    """
    Given centers, solve LP to maximize sum of radii.
    Variables: r_0, ..., r_{n-1}
    Maximize sum(r_i)
    Subject to:
      r_i >= 0
      r_i <= x_i
      r_i <= 1 - x_i
      r_i <= y_i
      r_i <= 1 - y_i
      r_i + r_j <= dist(i, j)
    """
    n = centers.shape[0]
    
    # Objective: maximize sum(r) -> minimize -sum(r)
    c_obj = -np.ones(n)
    
    # Inequalities: A_ub * r <= b_ub
    # 1. Boundary constraints
    # r_i <= x_i  =>  r_i - x_i <= 0 ? No, r_i is variable.
    # Constraint: r_i <= x_i  =>  1*r_i <= x_i
    #              r_i <= 1-x_i => 1*r_i <= 1-x_i
    #              r_i <= y_i  =>  1*r_i <= y_i
    #              r_i <= 1-y_i => 1*r_i <= 1-y_i
    
    # Matrix for boundaries: diagonal matrix with 1s.
    # But we have 4 constraints per variable.
    # Let's construct A_ub and b_ub.
    
    # Rows for boundaries
    A_ub = []
    b_ub = []
    
    for i in range(n):
        xi, yi = centers[i]
        # r_i <= xi
        row = np.zeros(n)
        row[i] = 1.0
        A_ub.append(row)
        b_ub.append(xi)
        
        # r_i <= 1 - xi
        row = np.zeros(n)
        row[i] = 1.0
        A_ub.append(row)
        b_ub.append(1.0 - xi)
        
        # r_i <= yi
        row = np.zeros(n)
        row[i] = 1.0
        A_ub.append(row)
        b_ub.append(yi)
        
        # r_i <= 1 - yi
        row = np.zeros(n)
        row[i] = 1.0
        A_ub.append(row)
        b_ub.append(1.0 - yi)
        
    # 2. Overlap constraints
    # r_i + r_j <= dist_ij
    # 1*r_i + 1*r_j <= dist
    for i in range(n):
        for j in range(i + 1, n):
            dist = math.hypot(centers[i][0] - centers[j][0], centers[i][1] - centers[j][1])
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dist)
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    # Bounds for r_i >= 0
    bounds = [(0, None) for _ in range(n)]
    
    # Solve
    # Method 'highs' is efficient
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        radii = res.x
        return radii, -res.fun
    else:
        # Fallback if LP fails (should not happen if centers valid)
        return np.full(n, 0.0), 0.0
