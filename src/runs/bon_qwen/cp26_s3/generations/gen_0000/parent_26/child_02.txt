from scipy.optimize import minimize
from itertools import combinations

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None

    def objective(params):
        centers = params[:n * 2].reshape(n, 2)
        radii = params[n * 2:]
        
        # 1. Boundary penalty: ensure circles are inside [0, 1]x[0, 1]
        # Penalty is applied if x-r < 0 or x+r > 1, etc.
        penalty = 0.0
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            penalty += max(0, r - x) ** 2
            penalty += max(0, r - (1 - x)) ** 2
            penalty += max(0, r - y) ** 2
            penalty += max(0, r - (1 - y)) ** 2

            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                overlap = radii[i] + radii[j] - dist
                if overlap > 0:
                    penalty += overlap ** 2
                    
        return -np.sum(radii) + 1000.0 * penalty

            np.random.seed(seed)
