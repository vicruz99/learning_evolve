import numpy as np
from scipy.optimize import minimize

def compute_objective(params, n, penalty):
    """
    Computes the objective value: -sum(radii) + penalty * (boundary_penalties + overlap_penalties)
    """
    c = params.reshape(n, 3)
    radii = c[:, 2]
    
    # Objective: maximize sum of radii
    loss = -np.sum(radii)
    
    # Boundary penalties
    x, y = c[:, 0], c[:, 1]
    pen = 0.0
    pen += np.sum(np.maximum(0, radii - x)**2)
    pen += np.sum(np.maximum(0, radii - (1 - x))**2)
    pen += np.sum(np.maximum(0, radii - y)**2)
    pen += np.sum(np.maximum(0, radii - (1 - y))**2)
    
    # Overlap penalties
    dx_mat = c[:, 0:1] - c[:, 0:1].T
    dy_mat = c[:, 1:2] - c[:, 1:2].T
    dist = np.sqrt(dx_mat**2 + dy_mat**2)
    r_sum = radii[:, None] + radii[None, :]
    ov = r_sum - dist
    np.fill_diagonal(ov, 0)
    pen += np.sum(np.maximum(0, ov)**2)
    
    return loss + penalty * pen

def run_packing():
    n = 26
    
    # 1. Initialize with a hexagonal/triangular lattice
    pts = []
    dx = 0.15
    dy = dx * np.sqrt(3) / 2
    for i in range(10):
        for j in range(10):
            x = j * dx + (i % 2) * dx / 2 + 0.05
            y = i * dy + 0.05
            if 0 <= x <= 1 and 0 <= y <= 1:
                pts.append([x, y, 0.08])
                
    if len(pts) > n:
        pts = pts[:n]
    elif len(pts) < n:
        np.random.seed(42)
        while len(pts) < n:
            pts.append([np.random.rand(), np.random.rand(), 0.05])
            
    X0 = np.array(pts).flatten()
    bounds = [(0, 1)] * (2 * n) + [(0, 0.5)] * n
    
    # 2. Multi-stage optimization with increasing penalties
    current_x = X0.copy()
    penalties = [50, 200, 1000, 5000, 20000]
    
    for pen in penalties:
        res = minimize(compute_objective, current_x, args=(n, pen),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': 5000, 'ftol': 1e-15})
        current_x = res.x.copy()
        
    # 3. Extract and clean results
    centers = current_x.reshape(n, 3)[:, :2]
    radii = current_x.reshape(n, 3)[:, 2]
    
    # Ensure strict feasibility within numerical tolerance
    radii = np.maximum(radii, 1e-7)
    centers[:, 0] = np.clip(centers[:, 0], radii, 1 - radii)
    centers[:, 1] = np.clip(centers[:, 1], radii, 1 - radii)
    
    # Conservative overlap fix to guarantee validation passes
    for _ in range(3):
        for i in range(n):
            for j in range(i + 1, n):
                d = np.sqrt((centers[i, 0] - centers[j, 0])**2 + 
                            (centers[i, 1] - centers[j, 1])**2)
                if d < radii[i] + radii[j] - 1e-12:
                    overlap = radii[i] + radii[j] - d
                    radii[i] -= overlap / 2
                    radii[j] -= overlap / 2
                    centers[i, 0] = np.clip(centers[i, 0], radii[i], 1 - radii[i])
                    centers[i, 1] = np.clip(centers[i, 1], radii[i], 1 - radii[i])
                    centers[j, 0] = np.clip(centers[j, 0], radii[j], 1 - radii[j])
                    centers[j, 1] = np.clip(centers[j, 1], radii[j], 1 - radii[j])
                    
    return centers, radii, np.sum(radii)
