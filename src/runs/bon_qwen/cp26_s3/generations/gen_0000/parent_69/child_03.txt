import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def get_initial_guess():
    # Generate a hexagonal lattice pattern for dense packing
    pts = []
    s = 0.18
    for j in range(8):
        for i in range(8):
            x = i * s + (j % 2) * s / 2
            y = j * s * np.sqrt(3) / 2
            pts.append([x, y])
        if len(pts) >= N_CIRCLES:
            break
            
    pts = np.array(pts[:N_CIRCLES])
    # Normalize to fit comfortably inside the square
    pts -= pts.min(axis=0)
    spread = pts.max(axis=0) - pts.min(axis=0)
    pts = pts / spread * 0.7 + 0.15
    
    r_init = np.full(N_CIRCLES, 0.095)
    return np.concatenate([pts.flatten(), r_init])

def objective(vars):
    return -np.sum(vars[2*N_CIRCLES:])

def constraint_boundary(vars):
    X = vars[:2*N_CIRCLES].reshape(N_CIRCLES, 2)
    R = vars[2*N_CIRCLES:]
    return np.concatenate([
        X[:, 0] - R,
        1 - X[:, 0] - R,
        X[:, 1] - R,
        1 - X[:, 1] - R,
        R
    ])

def constraint_pairwise(vars):
    X = vars[:2*N_CIRCLES].reshape(N_CIRCLES, 2)
    R = vars[2*N_CIRCLES:]
    idx = np.triu_indices(N_CIRCLES, k=1)
    dx = X[idx[0], 0] - X[idx[1], 0]
    dy = X[idx[0], 1] - X[idx[1], 1]
    dist_sq = dx**2 + dy**2
    rad_sum = R[idx[0]] + R[idx[1]]
    return dist_sq - rad_sum**2

def run_packing():
    x0 = get_initial_guess()
    bounds = [(0.0, 1.0)] * (2*N_CIRCLES) + [(0.0, 0.5)] * N_CIRCLES
    
    cons = [
        {'type': 'ineq', 'fun': constraint_boundary},
        {'type': 'ineq', 'fun': constraint_pairwise}
    ]
    
    best_x = x0
    best_val = objective(x0)
    
    # Run multiple restarts to escape local minima
    for seed in range(5):
        np.random.seed(seed)
        x_try = x0 + np.random.randn(len(x0)) * 0.005
        for i in range(len(bounds)):
            x_try[i] = np.clip(x_try[i], bounds[i][0], bounds[i][1])
            
        res = minimize(objective, x_try, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': 5000, 'ftol': 1e-12, 'disp': False})
        
        if res.fun < best_val:
            best_val = res.fun
            best_x = res.x
            
    centers = np.zeros((N_CIRCLES, 2))
    radii = np.zeros(N_CIRCLES)
    for i in range(N_CIRCLES):
        centers[i, 0] = best_x[i*2]
        centers[i, 1] = best_x[i*2+1]
        radii[i] = best_x[2*N_CIRCLES + i]
        
    return centers, radii, np.sum(radii)
