import numpy as np
from scipy.optimize import linprog, differential_evolution

def get_max_radii_sum(centers):
    """
    Solves the LP to find the maximum sum of radii for a fixed set of centers.
    
    Args:
        centers: np.array of shape (n, 2)
        
    Returns:
        max_sum: float, the maximum sum of radii
        radii: np.array of shape (n), the optimal radii
    """
    n = centers.shape[0]
    
    # Objective: maximize sum(r_i) => minimize -sum(r_i)
    c = -np.ones(n)
    
    # Constraints matrix A_ub * x <= b_ub
    # We have constraints:
    # 1. r_i <= distance to boundary for each circle i
    # 2. r_i + r_j <= distance between centers i and j for all pairs (i, j) with i < j
    
    A_ub = []
    b_ub = []
    
    # 1. Boundary constraints
    for i in range(n):
        x, y = centers[i]
        dist_to_boundary = min(x, 1 - x, y, 1 - y)
        
        # Row for r_i <= dist
        row = np.zeros(n)
        row[i] = 1.0
        A_ub.append(row)
        b_ub.append(dist_to_boundary)
        
    # 2. Overlap constraints (r_i + r_j <= dist)
    # Compute pairwise distances efficiently
    # centers shape (n, 2)
    # diff shape (n, n, 2)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.linalg.norm(diff, axis=2)
    
    for i in range(n):
        for j in range(i + 1, n):
            dist = dists[i, j]
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dist)
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    # Bounds for r_i: 0 <= r_i < inf
    bounds = [(0, None) for _ in range(n)]
    
    # Solve LP
    # Using high-precision method if possible, but default is usually fine
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        max_sum = -res.fun
        radii = res.x
        return max_sum, radii
    else:
        # If LP fails, return 0 (should not happen with valid inputs and 0 lower bound)
        return 0.0, np.zeros(n)
