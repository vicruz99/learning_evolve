import numpy as np
from scipy.optimize import minimize

def compute_constraints(v):
    """
    Computes all inequality constraints for the packing problem.
    Constraints must be >= 0.
    Returns a 1D numpy array of constraint values.
    """
    n = len(v) // 3
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    
    constraints = []
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    constraints.append(x - r)
    constraints.append(1.0 - x - r)
    constraints.append(y - r)
    constraints.append(1.0 - y - r)
    
    # Pairwise distance constraints: d^2 >= (r_i + r_j)^2
    for i in range(n):
        for j in range(i + 1, n):
            d2 = (x[i] - x[j])**2 + (y[i] - y[j])**2
            rsum2 = (r[i] + r[j])**2
            constraints.append(d2 - rsum2)
            
    return np.concatenate(constraints)

def objective_func(v):
    """Objective: maximize sum of radii -> minimize negative sum."""
    n = len(v) // 3
    return -np.sum(v[2::3])

def run_packing():
    n = 26
    
    # Initialize with a hexagonal-like pattern for dense packing
    centers0 = []
    centers0.extend([(x, 0.15) for x in np.linspace(0.1, 0.9, 6)])
    centers0.extend([(x, 0.32) for x in np.linspace(0.2, 0.8, 5)])
    centers0.extend([(x, 0.49) for x in np.linspace(0.1, 0.9, 6)])
    centers0.extend([(x, 0.66) for x in np.linspace(0.2, 0.8, 5)])
    centers0.extend([(x, 0.83) for x in np.linspace(0.25, 0.75, 4)])

    r0 = np.ones(n) * 0.04
    v0 = np.zeros(3 * n)
    for i in range(n):
        v0[3*i] = centers0[i][0]

        v0[3*i+2] = r0[i]

    # Bounds for x, y, r
    bounds = [(0.0, 1.0) for _ in range(2*n)] + [(0.0, 0.5) for _ in range(n)]
    constraints = [{'type': 'ineq', 'fun': compute_constraints}]

    # Run optimization
    res = minimize(

        v0, 
        method='SLSQP',

    return centers, radii, float(np.sum(radii))
