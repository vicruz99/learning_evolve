import numpy as np
from scipy.optimize import minimize

N = 26

def objective(vars):
    """Objective: maximize sum of radii (minimize negative sum)"""
    return -np.sum(vars[2*N:])
    
def jac_obj(vars):
    """Jacobian of the objective function"""
    jac = np.zeros(3*N)
    jac[2*N:] = -1.0
    return jac
    
def constr_boundary(vars):
    """Boundary constraints: circles must be inside [0,1]x[0,1]"""
    c = vars[:2*N].reshape(N, 2)
    r = vars[2*N:]
    return np.concatenate([c[:,0] - r, 1 - c[:,0] - r, c[:,1] - r, 1 - c[:,1] - r])
    
def jac_boundary(vars):
    """Jacobian of boundary constraints"""
    jac = np.zeros((4*N, 3*N))
    for i in range(N):
        r_idx = 2*N + i
        # x_i - r_i >= 0
        jac[i, 2*i] = 1.0
        jac[i, r_idx] = -1.0
        # 1 - x_i - r_i >= 0
        jac[N+i, 2*i] = -1.0
        jac[N+i, r_idx] = -1.0
        # y_i - r_i >= 0
        jac[2*N+i, 2*i+1] = 1.0
        jac[2*N+i, r_idx] = -1.0
        # 1 - y_i - r_i >= 0
        jac[3*N+i, 2*i+1] = -1.0
        jac[3*N+i, r_idx] = -1.0
    return jac
    
def constr_overlap(vars):
    """Non-overlap constraints: squared distance >= (r_i + r_j)^2"""
    c = vars[:2*N].reshape(N, 2)
    r = vars[2*N:]
    cons = np.empty(N*(N-1)//2)
    idx = 0
    for i in range(N):
        for j in range(i+1, N):
            dx = c[i,0] - c[j,0]
            dy = c[i,1] - c[j,1]
            cons[idx] = dx*dx + dy*dy - (r[i] + r[j])**2
            idx += 1
    return cons
    
def jac_overlap(vars):
    """Jacobian of overlap constraints"""
    jac = np.zeros((N*(N-1)//2, 3*N))
    idx = 0
    for i in range(N):
        for j in range(i+1, N):
            ci0 = 2*i; ci1 = 2*i+1; cri = 2*N+i
            cj0 = 2*j; cj1 = 2*j+1; crj = 2*N+j
            dx = vars[ci0] - vars[cj0]
            dy = vars[ci1] - vars[cj1]
            dr = vars[cri] + vars[crj]
            jac[idx, ci0] = 2*dx
            jac[idx, ci1] = 2*dy
            jac[idx, cj0] = -2*dx
            jac[idx, cj1] = -2*dy
            jac[idx, cri] = -2*dr
            jac[idx, crj] = -2*dr
            idx += 1
    return jac

def run_packing():
    np.random.seed(42)
    
    # Initialization: Dense grid pattern + jitter
    xs = np.linspace(0.1, 0.9, 5)
    ys = np.linspace(0.1, 0.9, 5)
    centers = np.array([(x, y) for x in xs for y in ys])[:25]
    centers = np.vstack([centers, [0.5, 0.5]]) # 26th circle
    centers += np.random.uniform(-0.02, 0.02, centers.shape)
    radii = np.full(N, 0.04)
    
    # Feasibility correction: resolve initial overlaps and clamp to boundaries
    for _ in range(15):
        for i in range(N):
            for j in range(i+1, N):
                dist = np.linalg.norm(centers[i] - centers[j])
                req = radii[i] + radii[j]
                if dist < req:
                    if dist > 1e-9:
                        vec = (centers[i] - centers[j]) / dist
                        push = (req - dist) / 2.0
                        centers[i] += vec * push
                        centers[j] -= vec * push
        for i in range(N):
            centers[i] = np.clip(centers[i], radii[i], 1-radii[i])
            
    x0 = np.concatenate([centers.flatten(), radii])
    
    bounds = [(0, 1)] * (2*N) + [(1e-6, 0.5)] * N
    cons = [
        {'type': 'ineq', 'fun': constr_boundary, 'jac': jac_boundary},
        {'type': 'ineq', 'fun': constr_overlap, 'jac': jac_overlap}
    ]
    
    res = minimize(objective, x0, method='SLSQP', jac=jac_obj, bounds=bounds, 
                   constraints=cons, options={'maxiter': 1500, 'ftol': 1e-12})
    
    centers_opt = res.x[:2*N].reshape(N, 2)
    radii_opt = res.x[2*N:]
    radii_opt = np.maximum(radii_opt, 0.0)
    
    return centers_opt, radii_opt, np.sum(radii_opt)
