import numpy as np
from scipy.optimize import linprog

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    Uses a hybrid approach: Hexagonal initialization -> LP for radii -> Gradient for centers.
    """
    n = 26
    iterations = 800
    
    # 1. Initialization: Hexagonal Grid
    # Create a dense hex grid and select n points.
    # Spacing is chosen to be slightly less than theoretical optimal to allow room.
    s = 0.21 
    h = s * np.sqrt(3) / 2
    candidates = []
    rows = 7
    cols = 6
    for r in range(rows):
        for c in range(cols):
            x = c * s + (r % 2) * (s / 2) + 0.1
            y = r * h + 0.1
            if 0 <= x <= 1 and 0 <= y <= 1:
                candidates.append([x, y])
    
    # Sort candidates to pick a good subset (e.g., by row/col order)
    # We want to pick n points that are well distributed.
    # Just taking the first n is usually decent for a dense grid.
    centers = np.array(candidates[:n])
    
    # 2. Optimization Loop
    for step in range(iterations):
        # Step A: Solve LP for optimal radii given fixed centers
        # Variables: r_i for i in 0..25
        # Objective: Maximize sum(r_i) -> Minimize -sum(r_i)
        c_obj = -np.ones(n)
        
        # Constraints: r_i + r_j <= dist(i, j)
        #               r_i <= boundaries
        
        # Precompute distances
        # Vectorized distance calculation
        diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        
        # Build A_ub and b_ub
        # Constraints:
        # 1. Pairwise: r_i + r_j <= dists[i,j]
        # 2. Boundary: r_i <= x_i, 1-x_i, y_i, 1-y_i
        
        num_pairs = n * (n - 1) // 2
        num_bound = 4 * n
        total_constraints = num_pairs + num_bound
        
        A_ub = np.zeros((total_constraints, n))
        b_ub = np.zeros(total_constraints)
        
        # Pairwise constraints
        idx = 0
        for i in range(n):
            for j in range(i + 1, n):
                A_ub[idx, i] = 1.0
                A_ub[idx, j] = 1.0
                b_ub[idx] = dists[i, j]
                idx += 1
        
        # Boundary constraints
        for i in range(n):
            x, y = centers[i]
            # r <= x
            A_ub[idx, i] = 1.0
            b_ub[idx] = x
            idx += 1
            # r <= 1-x
            A_ub[idx, i] = 1.0
            b_ub[idx] = 1.0 - x
            idx += 1
            # r <= y
            A_ub[idx, i] = 1.0
            b_ub[idx] = y
            idx += 1
            # r <= 1-y
            A_ub[idx, i] = 1.0
            b_ub[idx] = 1.0 - y
            idx += 1
            
        bounds = [(0, None) for _ in range(n)]
        
        try:
            # Solve LP
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
            if res.success:
                radii = res.x
            else:
                # Fallback if LP fails (rare)
                continue
        except:
            continue
            
        # Step B: Compute Forces to update centers
        forces = np.zeros((n, 2))
        
        # Tolerance for "tight" constraints
        tol = 1e-5
        
        # Calculate forces based on tight constraints
        # We look for constraints that are active (binding)
        
        # 1. Pairwise forces
        for i in range(n):
            for j in range(i + 1, n):
                d = dists[i, j]
                sum_r = radii[i] + radii[j]
                
                # If circles are touching (or very close)
                if sum_r > d - tol:
                    # Repulsive force: push i away from j, j away from i
                    if d > 1e-9:
                        vec = centers[i] - centers[j]
                        unit_vec = vec / d
                        # Force magnitude proportional to how tight/sum of radii
                        # Simple heuristic: push with strength proportional to radii sum
                        strength = (radii[i] + radii[j]) * 0.5
                        forces[i] += unit_vec * strength
                        forces[j] -= unit_vec * strength

        # 2. Boundary forces
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            
            # If touching left wall (x - r approx 0) -> push right
            if r > x - tol:
                forces[i, 0] += 1.0
            # If touching right wall (x + r approx 1) -> push left
            if r > (1.0 - x) - tol:
                forces[i, 0] -= 1.0
            # If touching bottom wall (y - r approx 0) -> push up
            if r > y - tol:
                forces[i, 1] += 1.0
            # If touching top wall (y + r approx 1) -> push down
            if r > (1.0 - y) - tol:
                forces[i, 1] -= 1.0

        # Update centers
        # Adaptive step size: decrease as we converge
        step_size = 0.05 * (1.0 - step / iterations)
        
        # Add small random jitter to escape local minima
        jitter = np.random.randn(n, 2) * 0.001 * (step < 100)
        
        centers += step_size * forces + jitter
        
        # Clip to valid range [0, 1]
        centers = np.clip(centers, 1e-9, 1.0 - 1e-9)

    return centers, radii, np.sum(radii)
