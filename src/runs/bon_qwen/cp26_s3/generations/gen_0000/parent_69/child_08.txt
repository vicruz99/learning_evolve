import numpy as np
from scipy.optimize import minimize


def objective_func(x, n, penalty):
    """Objective: maximize sum of radii with penalties for violations."""
    c = x[:2 * n].reshape(n, 2)
    r = x[2 * n:]

    obj = -np.sum(r)

    # Penalty for boundary violations
    for i in range(n):
        violation = 0.0
        if c[i, 0] - r[i] < 0:
            violation += (c[i, 0] - r[i]) ** 2
        if c[i, 0] + r[i] > 1:
            violation += (c[i, 0] + r[i] - 1) ** 2
        if c[i, 1] - r[i] < 0:
            violation += (c[i, 1] - r[i]) ** 2
        if c[i, 1] + r[i] > 1:
            violation += (c[i, 1] + r[i] - 1) ** 2
        obj += penalty * violation

    # Penalty for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            diff = c[i] - c[j]
            dist_sq = diff[0] ** 2 + diff[1] ** 2
            dist = np.sqrt(dist_sq) if dist_sq > 1e-20 else 1e-10
            overlap = r[i] + r[j] - dist
            if overlap > 0:
                obj += penalty * overlap ** 2

    return obj


def hex_init(n):
    """Initialize with hexagonal grid pattern."""
    centers = np.zeros((n, 2))
    r = 0.075
    d = 2 * r
    rs = d * np.sqrt(3) / 2

    idx = 0
    for row in range(7):
        if row % 2 == 0:
            num = 5
            sx = r
        else:
            num = 6
            sx = r + d / 2

        for col in range(num):
            if idx >= n:
                break
            centers[idx, 0] = sx + col * d
            centers[idx, 1] = r + row * rs
            idx += 1
        if idx >= n:
            break

    return centers


def random_init(n, seed):
    """Initialize with random positions."""
    np.random.seed(seed)
    centers = np.random.uniform(0.15, 0.85, (n, 2))
    return centers


def grid_init(n):
    """Initialize with square grid pattern."""
    centers = np.zeros((n, 2))
    idx = 0
    # 5x5 grid for 25 circles
    for i in range(5):
        for j in range(5):
            centers[idx, 0] = 0.1 + j * 0.2
            centers[idx, 1] = 0.1 + i * 0.2
            idx += 1
    # 26th circle at center
    centers[25, 0] = 0.5
    centers[25, 1] = 0.5
    return centers


def staggered_init(n):
    """Initialize with staggered rows optimized for square."""
    centers = np.zeros((n, 2))
    # Try to fit circles with diameter ~0.2
    # Rows at y = 0.1, 0.3, 0.5, 0.7, 0.9
    row_y = [0.1, 0.3, 0.5, 0.7, 0.9]
    row_counts = [5, 6, 5, 6, 4]
    row_offsets = [0.1, 0.2, 0.1, 0.2, 0.1]

    idx = 0
    for row_idx, (y, count, offset) in enumerate(zip(row_y, row_counts, row_offsets)):
        for col in range(count):
            if idx >= n:
                break
            x = offset + col * 0.2
            centers[idx, 0] = x
            centers[idx, 1] = y
            idx += 1
        if idx >= n:
            break

    return centers


def optimize_from_init(centers_init, radii_init, n, penalty):
    """Run optimization from given initial state."""
    x0 = np.concatenate([centers_init.flatten(), radii_init])

    bounds = [(0.0, 1.0)] * (2 * n) + [(1e-6, 0.5)] * n

    result = minimize(
        objective_func,
        x0,
        args=(n, penalty),
        method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': 20000, 'ftol': 1e-16, 'gtol': 1e-12}
    )

    centers_opt = result.x[:2 * n].reshape(n, 2)
    radii_opt = result.x[2 * n:]

    return centers_opt, radii_opt


def enforce_constraints(centers, radii, n):
    """Ensure all constraints are satisfied."""
    # Clamp radii based on boundary distance
    for i in range(n):
        max_r = min(
            centers[i, 0],
            1 - centers[i, 0],
            centers[i, 1],
            1 - centers[i, 1]
        )
        radii[i] = min(radii[i], max(max_r, 1e-8))

    # Iteratively reduce radii to resolve overlaps
    for _ in range(100):
        max_reduction = 0
        for i in range(n):
            for j in range(i + 1, n):
                diff = centers[i] - centers[j]
                dist = np.sqrt(diff[0] ** 2 + diff[1] ** 2)
                if dist < radii[i] + radii[j]:
                    excess = radii[i] + radii[j] - dist
                    reduction = excess / 2 + 1e-8
                    if reduction > max_reduction:
                        max_reduction = reduction
                    radii[i] -= reduction / 2
                    radii[j] -= reduction / 2

        if max_reduction < 1e-10:
            break

    # Ensure non-negative radii
    radii = np.maximum(radii, 1e-8)

    return centers, radii


def validate_quick(centers, radii, n):
    """Quick validation without printing."""
    for i in range(n):
        if centers[i, 0] - radii[i] < -1e-12 or centers[i, 0] + radii[i] > 1 + 1e-12:
            return False
        if centers[i, 1] - radii[i] < -1e-12 or centers[i, 1] + radii[i] > 1 + 1e-12:
            return False

    for i in range(n):
        for j in range(i + 1, n):
            diff = centers[i] - centers[j]
            dist = np.sqrt(diff[0] ** 2 + diff[1] ** 2)
            if dist < radii[i] + radii[j] - 1e-12:
                return False

    return True
