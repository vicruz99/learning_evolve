import numpy as np
from scipy.optimize import minimize


def smooth_penalty(val, threshold, c=1.0, alpha=10.0):
    diff = (threshold - val) * alpha
    return -c * np.log(np.cosh(diff)) / c


def run_packing():
    n = 26

    # Phase 1: Initialize with hexagonal pattern
    r_init = 0.08
    centers_init = []

    row = 0
    while len(centers_init) < n:
        y = row * np.sqrt(3) * r_init + r_init
        if y > 1.2:
            break
        col = 0
        while col * 2 * r_init + r_init <= 1.2 and len(centers_init) < n:
            x = col * 2 * r_init + r_init
            if row % 2 == 1:
                x += r_init
            if x >= 0 and x <= 1.2 and y >= 0 and y <= 1.2:
                centers_init.append([x, y])
            col += 1
        row += 1

    centers_init = np.array(centers_init[:n])
    radii_init = np.ones(n) * r_init

    # Flatten to 1D array
    x0_base = np.zeros(3 * n)
    for i in range(n):
        x0_base[3 * i] = centers_init[i, 0]
        x0_base[3 * i + 1] = centers_init[i, 1]
        x0_base[3 * i + 2] = radii_init[i]

    def objective_with_penalties(x, c_pen=10.0, alpha_pen=20.0):
        centers_opt = x[:2 * n].reshape(n, 2)
        radii_opt = x[2 * n:]

        # Primary objective: maximize sum of radii
        obj = -np.sum(radii_opt)

        # Boundary penalties: ri - xi <= 0, xi + ri - 1 <= 0, ri - yi <= 0, yi + ri - 1 <= 0, -ri <= 0
        for i in range(n):
            xi, yi, ri = centers_opt[i, 0], centers_opt[i, 1], radii_opt[i]
            obj += smooth_penalty(xi, ri, c_pen, alpha_pen)
            obj += smooth_penalty(1 - (xi + ri), 0, c_pen, alpha_pen)
            obj += smooth_penalty(yi, ri, c_pen, alpha_pen)
            obj += smooth_penalty(1 - (yi + ri), 0, c_pen, alpha_pen)
            obj += smooth_penalty(ri, 0, c_pen, alpha_pen)

        # Overlap penalties: dist >= ri + rj
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers_opt[i, 0] - centers_opt[j, 0]
                dy = centers_opt[i, 1] - centers_opt[j, 1]
                dist = np.sqrt(dx * dx + dy * dy + 1e-16)
                min_dist = radii_opt[i] + radii_opt[j]
                obj += smooth_penalty(dist, min_dist, c_pen, alpha_pen)

        return obj

    bounds = [(0.001, 0.999)] * (2 * n) + [(0.001, 0.45)] * n

    best_x = None
    best_score = float('inf')

    # Multiple restarts with different seeds and initial conditions
    for restart in range(15):
        np.random.seed(100 + restart)

        # Create varied initial conditions
        if restart == 0:
            x_start = x0_base.copy()
        elif restart < 8:
            # Perturb positions
            x_start = x0_base.copy()
            perturbation = np.random.normal(0, 0.02, 3 * n)
            x_start += perturbation
            x_start = np.clip(x_start, 0.01, 0.99)
            x_start[2::3] = np.clip(x_start[2::3], 0.02, 0.4)
        else:
            # Try different radius configurations
            x_start = x0_base.copy()
            # Some circles larger, some smaller
            radius_perturbation = np.random.normal(0, 0.03, n)
            x_start[2::3] = np.clip(x_start[2::3] + radius_perturbation, 0.02, 0.4)
            x_start[:2 * n] += np.random.normal(0, 0.01, 2 * n)
            x_start[:2 * n] = np.clip(x_start[:2 * n], 0.01, 0.99)

        # Use increasing penalty parameters
        c_pen = 5.0 + restart * 2.0
        alpha_pen = 15.0 + restart * 5.0

        result = minimize(
            objective_with_penalties,
            x_start,
            method='L-BFGS-B',
            args=(c_pen, alpha_pen),
            bounds=bounds,
            options={'maxiter': 8000, 'ftol': 1e-14, 'gtol': 1e-10}
        )

        if result.fun < best_score:
            best_score = result.fun
            best_x = result.x.copy()

    # Extract solution
    centers = best_x[:2 * n].reshape(n, 2)
    radii = best_x[2 * n:]

    # Final cleanup to ensure validity
    radii = np.maximum(radii, 0.001)

    # Ensure boundary constraints
    for i in range(n):
        max_r = min(centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])
        radii[i] = min(radii[i], max_r)
        radii[i] = max(radii[i], 0.001)
        centers[i, 0] = np.clip(centers[i, 0], radii[i], 1 - radii[i])
        centers[i, 1] = np.clip(centers[i, 1], radii[i], 1 - radii[i])

    # Fix any remaining overlaps by slightly reducing radii
    for iteration in range(100):
        max_overlap = 0
        max_pair = None
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                overlap = radii[i] + radii[j] - dist
                if overlap > max_overlap:
                    max_overlap = overlap
                    max_pair = (i, j)

        if max_overlap <= 1e-10:
            break

        i, j = max_pair
        reduction = max_overlap / 2 + 1e-8
        radii[i] = max(radii[i] - reduction, 0.001)
        radii[j] = max(radii[j] - reduction, 0.001)

    sum_radii = np.sum(radii)

    return centers, radii, sum_radii
