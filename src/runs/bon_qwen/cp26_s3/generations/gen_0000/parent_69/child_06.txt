    n = centers.shape[0]

    # x - r >= 0  =>  r - x <= 0
    # x + r <= 1  =>  x + r - 1 <= 0
    # y - r >= 0
    # y + r <= 1
    
    # Softplus-like or quadratic penalty
    # Using quadratic: max(0, violation)^2
    
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        
        # Left
        viol = r - x
        if viol > 0: penalty += viol**2
        # Right
        viol = x + r - 1.0
        if viol > 0: penalty += viol**2
        # Bottom
        viol = r - y
        if viol > 0: penalty += viol**2
        # Top
        viol = y + r - 1.0
        if viol > 0: penalty += viol**2

    for x0 in inits:
        try:
            res = minimize(objective, x0, args=(n_circles,), method='L-BFGS-B', 
                           bounds=bounds, options={'maxiter': 2000, 'ftol': 1e-12, 'gtol': 1e-8})
            
            pts = res.x.reshape((n_circles, 3))
            centers = pts[:, :2]
            radii = pts[:, 2]
            
            # Strict validation
            valid = True
            # Check radii non-negative
            if np.any(radii < 0):
                valid = False
            
            # Check boundaries
            if valid:
                for i in range(n_circles):
                    x, y = centers[i]
                    r = radii[i]
                    if x - r < -1e-9 or x + r > 1.0 + 1e-9 or \
                       y - r < -1e-9 or y + r > 1.0 + 1e-9:
                        valid = False
                        break
            
            # Check overlaps
            if valid:
                for i in range(n_circles):
                    for j in range(i + 1, n_circles):
                        dx = centers[i, 0] - centers[j, 0]
                        dy = centers[i, 1] - centers[j, 1]
                        dist = math.sqrt(dx*dx + dy*dy)
                        if dist < radii[i] + radii[j] - 1e-9:
                            valid = False
                            break
                    if not valid:
                        break
            
            if valid:
                current_sum = np.sum(radii)
                if current_sum > best_sum:
                    best_sum = current_sum
                    best_centers = centers.copy()
                    best_radii = radii.copy()
                    
        except Exception:
            continue

    # Fallback if no valid solution found (should not happen with good inits)
    if best_centers is None:
        centers = np.random.rand(26, 2)
        radii = np.full(26, 0.01)
        best_centers = centers
        best_radii = radii
        best_sum = np.sum(radii)

    return best_centers, best_radii, best_sum
