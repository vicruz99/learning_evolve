import math

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    Uses a penalty-based optimization method starting from a hexagonal lattice.

    def objective_with_penalty(vars_flat, lambda_pen=1000.0):
        # Reshape variables

        centers = vars_flat[0::3].reshape(-1, 1) # x coords
        centers = np.hstack([centers, vars_flat[1::3].reshape(-1, 1)]) # x, y
        radii = vars_flat[2::3]

        obj_val = -np.sum(radii)

        # x - r >= 0 => r - x <= 0
        # x + r <= 1 => x + r - 1 <= 0
        # y - r >= 0 => r - y <= 0
        # y + r <= 1 => y + r - 1 <= 0

        # Using element-wise max for violation
        # Violation is amount by which constraint is broken

        # Constraint: x <= 1 - r -> x + r <= 1 -> violation max(0, x + r - 1)
        v_x_hi = np.maximum(0, centers[:, 0] + radii - 1.0)
        # Constraint: y >= r -> violation max(0, r - y)

        # dist(c_i, c_j) >= r_i + r_j
        # violation max(0, r_i + r_j - dist)

        # centers shape (n, 2)
        # radii shape (n,)

        # Using broadcasting

        # Sum squared violations (divide by 2 because symmetric)
        penalty += np.sum(violation**2) / 2.0
        
        return obj_val + lambda_pen * penalty
