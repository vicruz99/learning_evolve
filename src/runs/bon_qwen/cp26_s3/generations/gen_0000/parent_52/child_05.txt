import numpy as np
import scipy.optimize as opt

def solve_radii(centers):
    """
    Solves for optimal radii given fixed centers using Linear Programming.
    Maximizes sum(r_i) subject to:
    1. r_i <= min(x_i, 1-x_i, y_i, 1-y_i)
    2. r_i + r_j <= distance(centers[i], centers[j])
    """
    n = centers.shape[0]
    c = -np.ones(n) # Maximize sum(r) => Minimize -sum(r)
    
    # Bounds for each radius: 0 <= r_i <= margin to boundary
    x, y = centers[:, 0], centers[:, 1]
    upper_bounds = np.minimum.reduce([x, 1 - x, y, 1 - y])
    bounds = [(0, u) for u in upper_bounds]
    
    # Pairwise constraints: r_i + r_j <= d_ij
    A_ub = []
    b_ub = []
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist > 1e-9:
                row = np.zeros(n)
                row[i] = 1
                row[j] = 1
                A_ub.append(row)
                b_ub.append(dist)
    
    if A_ub:
        A_ub = np.array(A_ub)
        b_ub = np.array(b_ub)
        res = opt.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
        else:
            # Fallback if LP fails (e.g., infeasible due to numerical issues)
            return np.zeros(n)
    else:
        return np.zeros(n)

def evaluate_packing(centers):
    """Computes the sum of radii for a given set of centers."""
    radii = solve_radii(centers)
    return np.sum(radii), radii

def compute_displacement(centers, radii):
    """
    Computes a displacement vector for each center to improve the sum of radii.
    Uses a force-directed approach based on active constraints.
    """
    n = centers.shape[0]
    forces = np.zeros_like(centers)
    
    # 1. Pairwise repulsion for active constraints
    for i in range(n):
        for j in range(i + 1, n):
            vec = centers[i] - centers[j]
            dist = np.linalg.norm(vec)
            if dist < 1e-9: continue
            
            # Check if constraint is active (touching)
            if radii[i] + radii[j] >= dist - 1e-5:
                # Normalize direction
                direction = vec / dist
                forces[i] += direction
                forces[j] -= direction
                
    # 2. Boundary repulsion
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        
        # Push away from left/right boundaries if touching
        if x - r <= 1e-5:
            forces[i, 0] += 1.0
        if x + r >= 1.0 - 1e-5:
            forces[i, 0] -= 1.0
            
        # Push away from bottom/top boundaries if touching
        if y - r <= 1e-5:
            forces[i, 1] += 1.0
        if y + r >= 1.0 - 1e-5:
            forces[i, 1] -= 1.0
            
    return forces

def local_optimize(centers_init):
    """Runs a local optimization loop to improve the packing."""
    centers = centers_init.copy()
    radii = solve_radii(centers)
    current_sum = np.sum(radii)
    
    # Adaptive step size parameters
    step_size = 0.01
    cooling = 0.99
    
    for _ in range(300): # Max iterations
        displacement = compute_displacement(centers, radii)
        
        # Normalize forces to prevent huge jumps
        norms = np.linalg.norm(displacement, axis=1, keepdims=True)
        norms[norms < 1e-9] = 1
        normalized_disp = displacement / norms
        
        # Update centers
        new_centers = centers + step_size * normalized_disp
        
        # Keep centers within valid bounds (with a small margin)
        new_centers = np.clip(new_centers, 1e-4, 1.0 - 1e-4)
        
        # Evaluate new configuration
        new_radii = solve_radii(new_centers)
        new_sum = np.sum(new_radii)
        
        # Accept if improved
        if new_sum > current_sum:
            centers = new_centers
            radii = new_radii
            current_sum = new_sum
            step_size *= 1.01 # Increase step size slightly on success
        else:
            step_size *= cooling # Decrease step size
            
        if step_size < 1e-6:
            break
            
    return centers, radii, current_sum

def objective(x, n):
    """Objective function for differential evolution (minimize negative sum of radii)."""
    centers = x.reshape((n, 2))
    s, _ = evaluate_packing(centers)
    return -s

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # 1. Generate a good initial guess: Perturbed 5x5 Grid + extra circles
    # Base grid for 25 circles
    grid_coords = []
    for i in range(5):
        for j in range(5):
            grid_coords.append([0.1 + i * 0.2, 0.1 + j * 0.2])
    
    # Add 26th circle in a gap (center of a 2x2 block of grid)
    # e.g. at 0.3, 0.3
    grid_coords.append([0.3, 0.3])
    
    centers_init = np.array(grid_coords)
    
    # 2. Run local optimization on the grid start
    centers, radii, best_sum = local_optimize(centers_init)
    
    # 3. Run Differential Evolution to find global optimum
    # We map 2D centers to 1D array for scipy
    bounds = [(0, 1) for _ in range(2 * n)]
    
    # We use a population size that balances exploration and speed
    try:
        res = opt.differential_evolution(
            func=lambda x: objective(x, n),
            bounds=bounds,
            seed=42,
            maxiter=200,
            popsize=20,
            tol=1e-5,
            mutation=(0.5, 1.5),
            recombination=0.9
        )
        if res.success:
            centers_opt = res.x.reshape((n, 2))
            radii_opt = solve_radii(centers_opt)
            sum_opt = np.sum(radii_opt)
            
            if sum_opt > best_sum:
                centers = centers_opt
                radii = radii_opt
                best_sum = sum_opt
    except Exception as e:
        print(f"DE failed: {e}")
        
    # 4. Final local refinement on the best found configuration
    centers, radii, best_sum = local_optimize(centers)
    
    return centers, radii, best_sum
