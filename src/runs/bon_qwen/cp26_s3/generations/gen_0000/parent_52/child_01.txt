import numpy as np
from scipy.optimize import minimize

def run_packing():
    """
    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    
    Returns:
        tuple: (centers, radii, sum_radii)
    """
    n = 26
    
    # 1. Initialize a staggered grid (hexagonal-like) layout
    # This provides a dense, non-overlapping starting configuration
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.05  # Start with small radii
    
    # Distribution: 6 rows with counts [5, 4, 5, 4, 5, 3] = 26 circles
    row_counts = [5, 4, 5, 4, 5, 3]
    idx = 0
    y_pos = 0.1
    
    for i, count in enumerate(row_counts):
        # Stagger x-coordinates for adjacent rows
        if i % 2 == 0: # Even rows (0, 2, 4) -> 5 circles
            # Positions: 0.1, 0.3, 0.5, 0.7, 0.9
            x_vals = np.linspace(0.1, 0.9, count)
        else: # Odd rows (1, 3, 5) -> 4 or 3 circles
            if count == 4:
                # Positions: 0.2, 0.4, 0.6, 0.8
                x_vals = np.linspace(0.2, 0.8, count)
            elif count == 3:
                # Positions: 0.3, 0.5, 0.7
                x_vals = np.linspace(0.3, 0.7, count)
            else:
                x_vals = np.linspace(0.1, 0.9, count)
        
        for x in x_vals:
            centers[idx] = [x, y_pos]
            idx += 1
        y_pos += 0.15 # Move down to next row
        
    # 2. Define objective and constraints for the optimizer
    # Variables vector: [x1, y1, ..., xn, yn, r1, ..., rn]
    x0 = np.concatenate([centers.flatten(), radii])
    
    def objective(vars):
        # Minimize negative sum of radii (equivalent to maximizing sum)
        radii = vars[n*2:]
        return -np.sum(radii)
    
    def constraint_boundary(vars):
        # Circles must be inside [0,1]x[0,1]
        centers = vars[:n*2].reshape(n, 2)
        radii = vars[n*2:]
        cons = []
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            # x - r >= 0, 1 - (x + r) >= 0, same for y
            cons.extend([x - r, 1 - (x + r), y - r, 1 - (y + r)])
        return np.array(cons)
    
    def constraint_overlap(vars):
        # Distance between centers >= sum of radii
        # Formulated as: dist^2 - (r1 + r2)^2 >= 0
        centers = vars[:n*2].reshape(n, 2)
        radii = vars[n*2:]
        cons = []
        for i in range(n):
            for j in range(i + 1, n):
                dist_sq = np.sum((centers[i] - centers[j]) ** 2)
                sum_r = radii[i] + radii[j]
                cons.append(dist_sq - sum_r ** 2)
        return np.array(cons)
    
    def constraint_non_neg(vars):
        # Radii must be non-negative
        return vars[n*2:]

    constraints = [
        {'type': 'ineq', 'fun': constraint_boundary},
        {'type': 'ineq', 'fun': constraint_overlap},
        {'type': 'ineq', 'fun': constraint_non_neg}
    ]

    # Bounds: x, y in [0, 1], r in [0, 0.5]
    bounds = []
    for i in range(n):
        bounds.extend([(0.0, 1.0), (0.0, 1.0)]) # x, y bounds
        bounds.append((0.0, 0.5))               # r bounds

    # 3. Run Optimization
    res = minimize(
        objective, 
        x0, 
        method='SLSQP', 
        bounds=bounds, 
        constraints=constraints,
        options={'maxiter': 1000, 'ftol': 1e-12, 'disp': False}
    )
    
    # 4. Extract results
    centers_opt = res.x[:n*2].reshape(n, 2)
    radii_opt = res.x[n*2:]
    total_sum = np.sum(radii_opt)
    
    # Basic post-check for numerical stability (clamp small negatives to 0)
    radii_opt = np.maximum(radii_opt, 0)
    
    return centers_opt, radii_opt, total_sum

def validate_packing(centers, radii):

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False

        if radii[i] < 0: return False
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            return False
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False
    return True
