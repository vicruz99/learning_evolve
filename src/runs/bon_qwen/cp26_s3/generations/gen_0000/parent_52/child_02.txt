from scipy.optimize import minimize
from functools import partial

def pair_constraint(v, i, j):

    xj, yj, rj = v[3*j], v[3*j+1], v[3*j+2]
    return (xi - xj)**2 + (yi - yj)**2 - (ri + rj)**2

def left_constraint(v, i):
    return v[3*i] - v[3*i+2]

def right_constraint(v, i):
    return 1.0 - v[3*i] - v[3*i+2]

def bottom_constraint(v, i):
    return v[3*i+1] - v[3*i+2]

def top_constraint(v, i):
    return 1.0 - v[3*i+1] - v[3*i+2]

def objective(v):
