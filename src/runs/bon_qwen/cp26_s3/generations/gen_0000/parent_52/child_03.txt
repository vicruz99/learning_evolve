import numpy as np

def solve_radii_lp(centers):
    """
    Solves the Linear Programming problem to maximize sum of radii
    given fixed centers.
    
    Args:
        centers: np.array of shape (n, 2)
        
    Returns:
        radii: np.array of shape (n)
    """
    n = centers.shape[0]
    
    # Objective: maximize sum(r) => minimize -sum(r)
    c_obj = -np.ones(n)
    
    # Bounds for radii: 0 <= r_i <= min(dist to boundaries)
    # r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    bounds = []
    for i in range(n):
        x, y = centers[i]
        max_r = min(x, 1 - x, y, 1 - y)
        # Ensure non-negative bound
        if max_r < 0:
            max_r = 0
        bounds.append((0, max_r))
        
    # Constraints: r_i + r_j <= distance(c_i, c_j)
    # A_ub @ r <= b_ub
    # We need to construct A_ub matrix. 
    # However, scipy linprog is faster with sparse matrices if n is large, 
    # but for n=26 dense is fine.
    # Number of pair constraints: n*(n-1)/2
    num_pairs = n * (n - 1) // 2
    
    # Preallocate matrices
    A_ub = np.zeros((num_pairs, n))
    b_ub = np.zeros(num_pairs)
    
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dist
            idx += 1
            
    # Solve LP
    # method='highs' is usually robust and fast
    res = scipy.optimize.linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return res.x
    else:
        # Fallback: return zeros or small radii if LP fails
        return np.zeros(n)
