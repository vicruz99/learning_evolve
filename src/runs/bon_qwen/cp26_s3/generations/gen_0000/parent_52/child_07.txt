import numpy as np
from scipy.optimize import minimize

N = 26

def objective(vars):
    """Objective function: minimize negative sum of radii"""
    return -np.sum(vars[2*N:])

def constraints(vars):
    """Inequality constraints: distance and boundary"""
    centers = vars[:2*N].reshape(N, 2)
    radii = vars[2*N:]

    # Pairwise distance constraints: d^2 - (r_i + r_j)^2 >= 0
    dx = centers[:, 0:1] - centers[:, 0:1].T
    dy = centers[:, 1:2] - centers[:, 1:2].T
    d2 = dx**2 + dy**2
    rs = radii[:, None] + radii[None, :]
    con_dist = d2 - rs**2
    con_dist = con_dist[np.triu_indices(N, k=1)]

    # Boundary constraints: x - r >= 0, 1 - x - r >= 0, etc.
    con_x = centers[:, 0] - radii
    con_x1 = 1.0 - centers[:, 0] - radii
    con_y = centers[:, 1] - radii
    con_y1 = 1.0 - centers[:, 1] - radii

    return np.concatenate([con_dist, con_x, con_x1, con_y, con_y1])

def run_packing():
    best_vars = None
    best_obj = -np.inf
    
    # Bounds: centers in [0,1], radii in [1e-5, 0.5]
    bounds = [(0.0, 1.0)]*(2*N) + [(1e-5, 0.5)]*N
    cons = {'type': 'ineq', 'fun': constraints}
    
    inits = []
    
    # 1. Grid initialization
    c_grid = []
    for i in range(5):
        for j in range(5):
            c_grid.append([0.1 + i*0.2, 0.1 + j*0.2])
    c_grid.append([0.5, 0.5])
    c_grid = np.array(c_grid[:N])
    inits.append(np.concatenate([c_grid.flatten(), np.full(N, 0.02)]))
    
    # 2. Hexagonal initialization
    c_hex = []
    x, y = 0.1, 0.1
    row = 0
    while len(c_hex) < N and y <= 0.9:
        x = 0.1 if row % 2 == 0 else 0.15
        while x <= 0.9 and len(c_hex) < N:
            c_hex.append([x, y])
            x += 0.15
        y += 0.13
        row += 1
    c_hex = np.array(c_hex[:N])
    inits.append(np.concatenate([c_hex.flatten(), np.full(N, 0.02)]))
    
    # 3. Random initializations
    for seed in range(4):
        np.random.seed(seed)
        c_rand = np.random.uniform(0.15, 0.85, (N, 2))
        inits.append(np.concatenate([c_rand.flatten(), np.full(N, 0.01)]))
        
    # Optimize from each start
    for x0 in inits:
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 4000, 'ftol': 1e-10, 'disp': False})
            if res.success and res.fun < best_obj:
                best_obj = res.fun
                best_vars = res.x.copy()
        except Exception:
            continue

    # Fallback
    if best_vars is None:
        best_vars = inits[0]

    centers = best_vars[:2*N].reshape(N, 2)
    radii = best_vars[2*N:]

    # Post-optimization safety scaling to strictly satisfy validator tolerances
    min_ratio = 1.0
    for i in range(N):
        cx, cy = centers[i]
        r = radii[i]
        r_bound = min(cx, 1.0 - cx, cy, 1.0 - cy)
        if r > r_bound:
            min_ratio = min(min_ratio, r_bound / r)
            
    for i in range(N):
        for j in range(i + 1, N):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            if dist < radii[i] + radii[j]:
                min_ratio = min(min_ratio, dist / (radii[i] + radii[j]))

    # Apply scaling but preserve optimization gains if already valid
    radii *= max(min_ratio, 0.995)

    return centers, radii, float(np.sum(radii))
