import numpy as np
from scipy.optimize import linprog

def solve_radii_for_centers(centers):
    """
    Computes the maximal radii for a fixed set of centers using Linear Programming.
    Maximizes sum(r_i) subject to boundary and non-overlap constraints.
    """
    n = centers.shape[0]
    # Objective: maximize sum(r) => minimize -sum(r)
    c_obj = -np.ones(n)
    
    # Constraints: 
    # 1. r_i <= min(x_i, 1-x_i, y_i, 1-y_i)  (4 per circle)
    # 2. r_i + r_j <= ||c_i - c_j||          (n*(n-1)/2 pairs)
    n_constraints = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((n_constraints, n))
    b_ub = np.zeros(n_constraints)
    
    idx = 0
    for i in range(n):
        x, y = centers[i]
        r_max = min(x, 1.0 - x, y, 1.0 - y)
        r_max = max(0.0, r_max)  # Ensure non-negative bound
        
        # 4 boundary constraints per circle
        for _ in range(4):
            A_ub[idx, i] = 1.0
            b_ub[idx] = r_max
            idx += 1
            
    # Pairwise non-overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.hypot(dx, dy)
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dist
            idx += 1
            
    bounds = [(0, None)] * n
    # Use HiGHS solver for speed and robustness
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return -res.fun, -res.x
    else:
        # Fallback in case of solver issues (highly unlikely with valid geometry)
        return 0.0, np.zeros(n)

def run_packing():
    n = 26
    centers = np.zeros((n, 2))
    
    # Initialize with a hexagonal-like pattern for good distribution
    rows = [6, 5, 6, 5, 4]  # Total 26 circles
    dx = 1.0 / 7.0
    dy = 1.0 / 5.0
    
    idx = 0
    y = 0.0
    for r_idx, count in enumerate(rows):
        row_width = (count - 1) * dx
        x_start = (1.0 - row_width) / 2.0
        # Offset alternating rows for hexagonal packing
        if r_idx % 2 == 1:
            x_start += dx / 2.0
            
        for c_idx in range(count):
            if idx < n:
                centers[idx] = [x_start + c_idx * dx, y]
                idx += 1
        y += dy
        
    best_centers = centers.copy()
    best_sum, _ = solve_radii_for_centers(best_centers)
    
    rng = np.random.default_rng(42)
    step = 0.015
    
    # Hill climbing local search on center positions
    for epoch in range(30):
        improved = False
        for _ in range(150):
            i = rng.integers(n)
            pert = rng.uniform(-step, step, 2)
            new_pos = best_centers[i] + pert
            new_pos = np.clip(new_pos, 0.0, 1.0)  # Keep within unit square
            
            old_pos = best_centers[i]
            best_centers[i] = new_pos
            
            curr_sum, _ = solve_radii_for_centers(best_centers)
            if curr_sum > best_sum + 1e-8:
                best_sum = curr_sum
                improved = True
            else:
                best_centers[i] = old_pos
                
        if not improved:
            step *= 0.85  # Reduce step size on stagnation
            if step < 1e-4:
                break
                
    # Final exact computation of radii for the optimized centers
    _, final_radii = solve_radii_for_centers(best_centers)
    return best_centers, final_radii, best_sum
