import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def objective(x):
    # Maximize sum of radii => minimize negative sum
    return -np.sum(x[52:])

def constraint_fun(x):
    n = N_CIRCLES
    c = x[:52].reshape((n, 2))
    r = x[52:]
    
    # Boundary constraints: c_i - r_i >= 0, 1 - c_i - r_i >= 0, r_i >= 0
    # Returns 5 constraints per circle
    con = np.hstack([c[:, 0] - r, 1.0 - c[:, 0] - r,
                     c[:, 1] - r, 1.0 - c[:, 1] - r, r])
                     
    # Overlap constraints: ||c_i - c_j||^2 >= (r_i + r_j)^2
    # Vectorized computation using broadcasting
    c_diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
    d2 = np.sum(c_diff**2, axis=2)
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    con = np.append(con, d2[mask] - r_sum[mask]**2)
    return con

def run_packing():
    n = N_CIRCLES
    
    # Initial feasible configuration: grid layout with small radii
    centers = np.array([[0.2 + 0.6*i/4.0, 0.2 + 0.6*j/4.0] for i in range(5) for j in range(5)])
    centers = np.vstack([centers, [0.5, 0.5]])  # 26th circle
    radii = np.ones(n) * 0.05
    
    x0 = np.hstack([centers.flatten(), radii])
    
    cons = {'type': 'ineq', 'fun': constraint_fun}
    bounds = [(0.0, 1.0)] * 52 + [(0.0, 0.5)] * n
    
    # Optimize
    res = minimize(objective, x0, method='SLSQP', bounds=bounds, 
                   constraints=cons, options={'maxiter': 2000, 'ftol': 1e-10})
                   
    c_opt = res.x[:52].reshape((n, 2))
    r_opt = res.x[52:]
    
    # Ensure strict numerical feasibility
    r_opt = np.clip(r_opt, 1e-9, 0.5)
    c_opt[:, 0] = np.clip(c_opt[:, 0], r_opt, 1.0 - r_opt)
    c_opt[:, 1] = np.clip(c_opt[:, 1], r_opt, 1.0 - r_opt)
    
    return c_opt, r_opt, np.sum(r_opt)
