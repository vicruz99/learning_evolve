import numpy as np
from scipy.optimize import minimize
from functools import partial

def c_x1(v, i): return v[3*i] - v[3*i+2]
def c_x2(v, i): return 1.0 - v[3*i] - v[3*i+2]
def c_y1(v, i): return v[3*i+1] - v[3*i+2]
def c_y2(v, i): return 1.0 - v[3*i+1] - v[3*i+2]
def c_pair(v, i, j):
    dx = v[3*i] - v[3*j]
    dy = v[3*i+1] - v[3*j+1]
    dr = v[3*i+2] + v[3*j+2]
    return dx*dx + dy*dy - dr*dr

def run_packing():
    np.random.seed(42)
    n = 26
    best_sum_r = 0.0
    best_x = None
    
    # Build constraints list
    cons = []
    for i in range(n):
        cons.append({'type': 'ineq', 'fun': partial(c_x1, i=i)})
        cons.append({'type': 'ineq', 'fun': partial(c_x2, i=i)})
        cons.append({'type': 'ineq', 'fun': partial(c_y1, i=i)})
        cons.append({'type': 'ineq', 'fun': partial(c_y2, i=i)})
    for i in range(n):
        for j in range(i+1, n):
            cons.append({'type': 'ineq', 'fun': partial(c_pair, i=i, j=j)})
            
    bnds = [(0.0, 1.0)] * (3*n)
    
    def obj(v):
        return -np.sum(v[2::3])
        
    # Generate diverse initial configurations
    configs = []
    
    # 1. Hexagonal lattice
    c1 = np.zeros((n, 2))
    idx = 0
    y = 0.12
    row = 0
    while idx < n:
        x = 0.12 if row % 2 == 0 else 0.25
        while idx < n and x <= 0.98:
            c1[idx] = [x, y]
            x += 0.33
            idx += 1
        y += 0.28
        row += 1
    configs.append(c1)
    
    # 2. Random distribution
    c2 = np.random.rand(n, 2) * 0.8 + 0.1
    configs.append(c2)
    
    # 3. Vogel spiral
    c3 = np.zeros((n, 2))
    for i in range(n):
        t = i * 2.39996322972865332
        r = np.sqrt(i/n) * 0.45
        c3[i] = [0.5 + r*np.cos(t), 0.5 + r*np.sin(t)]
    configs.append(c3)
    
    # Optimization loop
    for c_init in configs:
        x0 = np.zeros(3*n)
        for i in range(n):
            x0[3*i] = c_init[i, 0]
            x0[3*i+1] = c_init[i, 1]
            x0[3*i+2] = 0.02  # Start with small radii to ensure feasibility
            
        res = minimize(obj, x0, method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': 1000, 'ftol': 1e-11, 'disp': False})
        if -res.fun > best_sum_r:
            best_sum_r = -res.fun
            best_x = res.x.copy()
            
        # Local refinement with decreasing noise
        for step in range(5):
            if best_x is None:
                break
            noise = 0.01 * (0.6**step)
            xp = best_x.copy()
            xp[0::3] += np.random.randn(n) * noise
            xp[1::3] += np.random.randn(n) * noise
            xp[0::3] = np.clip(xp[0::3], 0.05, 0.95)
            xp[1::3] = np.clip(xp[1::3], 0.05, 0.95)
            
            res = minimize(obj, xp, method='SLSQP', bounds=bnds, constraints=cons,
                           options={'maxiter': 600, 'ftol': 1e-11, 'disp': False})
            if -res.fun > best_sum_r:
                best_sum_r = -res.fun
                best_x = res.x.copy()
                
    # Extract results
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        centers[i] = best_x[3*i:3*i+2]
        radii[i] = best_x[3*i+2]
        
    return centers, radii, np.sum(radii)
