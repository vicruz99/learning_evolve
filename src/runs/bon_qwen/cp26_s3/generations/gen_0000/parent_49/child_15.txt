import numpy as np
from scipy.optimize import linprog
import math

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    np.random.seed(42)

    # 1. Initialization: Perturbed Hexagonal Grid
    centers = np.zeros((n, 2))
    # Hexagonal packing density is high; we start with a dense hex grid
    r_init = 0.11
    dx, dy = 2 * r_init, math.sqrt(3) * r_init
    idx = 0
    row = 0
    while idx < n:
        col = 0
        while idx < n and col < 6:
            x = 0.05 + col * dx + (row % 2) * (dx / 2)
            y = 0.05 + row * dy
            if 0 <= x <= 1 and 0 <= y <= 1:
                centers[idx] = [x + np.random.uniform(-0.005, 0.005), 
                                y + np.random.uniform(-0.005, 0.005)]
                idx += 1
            col += 1
        row += 1
    
    # If we have fewer than 26 due to bounds, fill randomly
    if idx < n:
        while idx < n:
            centers[idx] = np.random.rand(2)
            idx += 1
            
    radii = np.zeros(n)
    
    # 2. Iterative Optimization
    alpha = 0.15 # Learning rate
    best_sum_r = 0
    best_centers = centers.copy()
    best_radii = radii.copy()
    
    for iteration in range(300):
        # Solve LP to find optimal radii for current centers
        # Objective: maximize sum(r_i) => minimize -sum(r_i)
        c_obj = np.ones(n) * -1
        
        # Constraints: A_ub * r <= b_ub
        A_ub = []
        b_ub = []
        
        # Boundary constraints for each circle
        for i in range(n):
            # r_i <= x_i
            row_r = np.zeros(n)
            row_r[i] = 1.0
            A_ub.append(row_r)
            b_ub.append(centers[i, 0])
            
            # r_i <= 1 - x_i
            row_r = np.zeros(n)
            row_r[i] = 1.0
            A_ub.append(row_r)
            b_ub.append(1 - centers[i, 0])
            
            # r_i <= y_i
            row_r = np.zeros(n)
            row_r[i] = 1.0
            A_ub.append(row_r)
            b_ub.append(centers[i, 1])
            
            # r_i <= 1 - y_i
            row_r = np.zeros(n)
            row_r[i] = 1.0
            A_ub.append(row_r)
            b_ub.append(1 - centers[i, 1])
            
        # Pairwise non-overlap constraints: r_i + r_j <= dist
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.linalg.norm(centers[i] - centers[j])
                row_r = np.zeros(n)
                row_r[i] = 1.0
                row_r[j] = 1.0
                A_ub.append(row_r)
                b_ub.append(dist)
                
        A_ub = np.array(A_ub)
        b_ub = np.array(b_ub)
        
        bounds = [(0, None) for _ in range(n)]
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
            if not res.success:
                continue
                
            radii = res.x
            sum_r = np.sum(radii)
            
            if sum_r > best_sum_r:
                best_sum_r = sum_r
                best_centers = centers.copy()
                best_radii = radii.copy()
            
            # Calculate forces based on marginals
            forces = np.zeros((n, 2))
            
            # Process boundary marginals
            # The marginals for boundary constraints are at indices 4*i to 4*i+3
            for i in range(n):
                # Left wall: r_i <= x_i (index 4*i)
                m = res.ineqlin.marginals[4 * i]
                if m < 0:
                    forces[i, 0] += -m
                    
                # Right wall: r_i <= 1 - x_i (index 4*i + 1)
                m = res.ineqlin.marginals[4 * i + 1]
                if m < 0:
                    forces[i, 0] -= -m
                    
                # Bottom wall: r_i <= y_i (index 4*i + 2)
                m = res.ineqlin.marginals[4 * i + 2]
                if m < 0:
                    forces[i, 1] += -m
                    
                # Top wall: r_i <= 1 - y_i (index 4*i + 3)
                m = res.ineqlin.marginals[4 * i + 3]
                if m < 0:
                    forces[i, 1] -= -m
                    
            # Process pairwise marginals
            pair_idx = 4 * n
            for i in range(n):
                for j in range(i + 1, n):
                    m = res.ineqlin.marginals[pair_idx]
                    if m < 0:
                        w = -m
                        dist = np.linalg.norm(centers[i] - centers[j])
                        if dist > 1e-9:
                            direction = (centers[i] - centers[j]) / dist
                            forces[i] += w * direction
                            forces[j] -= w * direction
                    pair_idx += 1

            centers += alpha * forces
            # Clip to valid range to prevent numerical drift
            centers = np.clip(centers, 1e-4, 1 - 1e-4)
            
            # Reduce step size for convergence
            alpha *= 0.99
            
        except Exception:
            continue

    # Final validation and adjustment
    best_radii = np.maximum(best_radii, 0)
    best_centers = np.clip(best_centers, 0, 1)
    
    return best_centers, best_radii, np.sum(best_radii)
