import numpy as np
from scipy.optimize import minimize

def compute_objective(v, n, lam):
    """Objective function to maximize sum of radii with penalty constraints."""
    c = v[:2*n].reshape(n, 2)
    r = v[2*n:]
    obj = -np.sum(r)
    
    # Boundary penalties: ensure circles stay within [0, 1]x[0, 1]
    for i in range(n):
        obj += lam * np.maximum(0.0, r[i] - c[i, 0])**2
        obj += lam * np.maximum(0.0, c[i, 0] + r[i] - 1.0)**2
        obj += lam * np.maximum(0.0, r[i] - c[i, 1])**2
        obj += lam * np.maximum(0.0, c[i, 1] + r[i] - 1.0)**2
        
    # Overlap penalties: ensure non-overlapping circles
    for i in range(n):
        for j in range(i + 1, n):
            dx = c[i, 0] - c[j, 0]
            dy = c[i, 1] - c[j, 1]
            dist = np.sqrt(dx*dx + dy*dy)
            overlap = r[i] + r[j] - dist
            if overlap > 0.0:
                obj += lam * overlap**2
                
    return obj
