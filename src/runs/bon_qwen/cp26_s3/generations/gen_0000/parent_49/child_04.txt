import numpy as np
from scipy.optimize import minimize

def run_packing():
    n = 26
    np.random.seed(42)
    
    # Precompute indices for pairwise constraints to speed up evaluation
    triu_idx = np.triu_indices(n, k=1)
    
    def constraint_fun(v):
        cen = v[:2*n].reshape(-1, 2)
        rad = v[2*n:]
        
        # Boundary constraints: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
        b_con = np.concatenate([
            cen[:, 0] - rad,
            1.0 - cen[:, 0] - rad,
            cen[:, 1] - rad,
            1.0 - cen[:, 1] - rad
        ])
        
        # Pairwise non-overlap constraints: dist(i,j) - r_i - r_j >= 0
        diff = cen[:, np.newaxis, :] - cen[np.newaxis, :, :]
        dists = np.sqrt(diff[:,:,0]**2 + diff[:,:,1]**2)
        p_con = dists[triu_idx] - rad[triu_idx[0]] - rad[triu_idx[1]]
        
        return np.concatenate([b_con, p_con])

    constraints = [{'type': 'ineq', 'fun': constraint_fun}]
    bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n
    
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Generate diverse initial configurations
    inits = []
    
    # 1. Hexagonal lattice
    c1 = np.zeros((n, 2))
    k = 0
    for r in range(6):
        for c in range(6):
            if k >= n: break
            x = 0.1 + c * 0.18 + (0.09 if r % 2 else 0.0)
            y = 0.1 + r * 0.18 * np.sqrt(3)
            c1[k] = [x, y]
            k += 1
    inits.append(c1)
    
    # 2. 5x5 grid + center
    c2 = np.zeros((n, 2))
    k = 0
    for i in range(5):
        for j in range(5):
            c2[k] = [0.1 + i * 0.2, 0.1 + j * 0.2]
            k += 1
    c2[25] = [0.5, 0.5]
    inits.append(c2)
    
    # 3. Perturbed hexagonal
    inits.append(c1 + np.random.uniform(-0.02, 0.02, c1.shape))
    
    # Ensure all initial positions are strictly inside the square
    inits = [np.clip(cfg, 0.05, 0.95) for cfg in inits]
    
    for cfg in inits:
        # Randomize initial radii slightly to break symmetry and avoid local minima
        x0 = np.concatenate([cfg.ravel(), np.random.uniform(0.04, 0.06, n)])
        
        res = minimize(
            lambda v: -np.sum(v[2*n:]), 
            x0, 
            method='SLSQP',
            bounds=bounds, 
            constraints=constraints,
            options={'maxiter': 2000, 'ftol': 1e-12}
        )
        
        if res.success:
            s = np.sum(res.x[2*n:])
            if s > best_sum:
                best_sum = s
                best_centers = res.x[:2*n].reshape(-1, 2)
                best_radii = res.x[2*n:]
                
    return best_centers, best_radii, float(best_sum)
