import numpy as np
from scipy.optimize import minimize


def create_hex_initial(n):
    """Create hexagonal initial arrangement."""
    centers = np.zeros((n, 2))
    rows = [6, 5, 6, 5, 4]
    y_spacing = 0.18
    x_spacing = 0.17
    idx = 0
    for row_i, count in enumerate(rows):
        y = 0.12 + row_i * y_spacing
        x_start = 0.12 if row_i % 2 == 0 else 0.12 + x_spacing / 2
        for col in range(count):
            centers[idx, 0] = x_start + col * x_spacing
            centers[idx, 1] = y
            idx += 1
    return centers


def create_grid_initial(n):
    """Create grid-like initial arrangement."""
    centers = np.zeros((n, 2))
    rows = [5, 5, 5, 5, 4, 2]
    idx = 0
    for row_i, count in enumerate(rows):
        y = 0.1 + row_i * 0.18
        x_start = 0.1
        x_spacing = 0.17
        for col in range(count):
            centers[idx, 0] = x_start + col * x_spacing
            centers[idx, 1] = y
            idx += 1
    return centers


def create_corner_initial(n):
    """Create arrangement with emphasis on corners."""
    centers = np.zeros((n, 2))
    radii_vals = np.zeros(n)
    
    # Corner circles (larger)
    idx = 0
    corners = [[0.15, 0.15], [0.85, 0.15], [0.15, 0.85], [0.85, 0.85]]
    for c in corners:
        centers[idx] = c
        radii_vals[idx] = 0.08
        idx += 1
    
    # Edge circles
    edge_positions = [
        [0.5, 0.12], [0.5, 0.88],
        [0.12, 0.5], [0.88, 0.5],
        [0.3, 0.12], [0.7, 0.12],
        [0.3, 0.88], [0.7, 0.88],
        [0.12, 0.3], [0.12, 0.7],
        [0.88, 0.3], [0.88, 0.7],
    ]
    for ep in edge_positions:
        centers[idx] = ep
        radii_vals[idx] = 0.06
        idx += 1
    
    # Center circles
    center_positions = [
        [0.5, 0.5],

    ]
    for cp in center_positions:
        if idx < n:
            centers[idx] = cp
            radii_vals[idx] = 0.05
            idx += 1
    
    return centers, radii_vals


def constraints_vec(xv, n):
    """Compute all constraint values as array."""
    num_bc = 5 * n  # 5 boundary constraints per circle
    num_oc = n * (n - 1) // 2  # non-overlap constraints
    cons = np.zeros(num_bc + num_oc)
    
    idx = 0
    for i in range(n):
        x = xv[3 * i]
        y = xv[3 * i + 1]
        r = xv[3 * i + 2]
        cons[idx] = x - r
        idx += 1
        cons[idx] = 1.0 - x - r
        idx += 1
        cons[idx] = y - r
        idx += 1
        cons[idx] = 1.0 - y - r
        idx += 1
        cons[idx] = r - 0.0001
        idx += 1
    
    for i in range(n):
        for j in range(i + 1, n):
            dx = xv[3 * i] - xv[3 * j]
            dy = xv[3 * i + 1] - xv[3 * j + 1]
            dist = np.sqrt(dx * dx + dy * dy)
            cons[idx] = dist - xv[3 * i + 2] - xv[3 * j + 2]
            idx += 1
    
    return cons


def penalized_obj(xv, n, penalty):
    """Objective with penalty for constraint violations."""
    obj = -np.sum(xv[2::3])
    cons = constraints_vec(xv, n)
    violations = np.maximum(-cons, 0.0)
    penalty_sum = np.sum(violations ** 2)
    return obj + penalty * penalty_sum


def gradient_penalized_obj(xv, n, penalty):
    """Compute objective and gradient with penalty."""
    obj = penalized_obj(xv, n, penalty)
    eps = 1e-7
    grad = np.zeros(3 * n)
    for k in range(3 * n):
        xv_plus = xv.copy()
        xv_minus = xv.copy()
        xv_plus[k] += eps
        xv_minus[k] -= eps
        grad[k] = (penalized_obj(xv_plus, n, penalty) - penalized_obj(xv_minus, n, penalty)) / (2 * eps)
    return obj, grad


def force_based_packing(n, iterations=8000):
    """Force-based simulation to find good packing."""
    centers = create_hex_initial(n).copy()
    radii = np.ones(n) * 0.03
    
    dt = 0.02
    radius_growth = 1e-5
    force_scale = 15.0
    
    for it in range(iterations):
        forces = np.zeros((n, 2))
        
        # Repulsive forces between circles
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                if dist < 1e-10:
                    dist = 1e-10
                    dx = 0.01
                    dy = 0.01
                overlap = radii[i] + radii[j] - dist
                if overlap > 0:
                    f = overlap / dist
                    forces[i, 0] += f * dx
                    forces[i, 1] += f * dy
                    forces[j, 0] -= f * dx
                    forces[j, 1] -= f * dy
        
        # Boundary repulsion
        for i in range(n):
            x, y, r = centers[i, 0], centers[i, 1], radii[i]
            margin = max(0, r - x)
            forces[i, 0] += margin * force_scale
            margin = max(0, x + r - 1.0)
            forces[i, 0] -= margin * force_scale
            margin = max(0, r - y)
            forces[i, 1] += margin * force_scale
            margin = max(0, y + r - 1.0)
            forces[i, 1] -= margin * force_scale
        
        # Update centers
        centers += dt * forces
        
        # Clip to valid range
        centers[:, 0] = np.clip(centers[:, 0], 0.001, 0.999)
        centers[:, 1] = np.clip(centers[:, 1], 0.001, 0.999)
        
        # Reduce dt over time for stability
        if it > iterations // 2:
            dt *= 0.9998
        
        # Grow radii slowly
        if it < iterations * 0.9:
            radii += radius_growth
    
    # Final adjustment: reduce radii slightly to ensure feasibility
    for i in range(n):
        r = radii[i]
        x, y = centers[i, 0], centers[i, 1]
        max_r = min(x, 1.0 - x, y, 1.0 - y)
        if r > max_r:
            radii[i] = max_r * 0.999
    
    return centers, radii


def centers_radii_to_xv(centers, radii):
    """Convert centers and radii to optimization variable vector."""
    n = len(radii)
    xv = np.zeros(3 * n)
    for i in range(n):
        xv[3 * i] = centers[i, 0]
        xv[3 * i + 1] = centers[i, 1]
        xv[3 * i + 2] = radii[i]
    return xv


def xv_to_centers_radii(xv, n):
    """Convert optimization variable vector to centers and radii."""
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        centers[i, 0] = xv[3 * i]
        centers[i, 1] = xv[3 * i + 1]
        radii[i] = xv[3 * i + 2]
    return centers, radii


def refine_with_optimization(centers, radii, n):
    """Refine packing using gradient-based optimization."""
    xv = centers_radii_to_xv(centers, radii)
    
    best_xv = xv.copy()
    best_sum = np.sum(radii)
    
    penalties = [500, 2000, 5000, 10000, 20000, 50000]
    
    for pen in penalties:
        result = minimize(
            penalized_obj,
            best_xv,
            args=(n, pen),
            method='L-BFGS-B',
            bounds=[(0.001, 0.999)] * (2 * n) + [(0.0001, 0.5)] * n,
            options={'maxiter': 5000, 'ftol': 1e-15}
        )
        if result.success or result.fun < penalized_obj(best_xv, n, pen):
            best_xv = result.x.copy()
            r_temp = best_xv[2::3]
            if np.sum(r_temp) > best_sum:
                best_sum = np.sum(r_temp)
    
    centers_out, radii_out = xv_to_centers_radii(best_xv, n)
    
    # Ensure radii are non-negative
    radii_out = np.maximum(radii_out, 0.0001)
    
    return centers_out, radii_out
