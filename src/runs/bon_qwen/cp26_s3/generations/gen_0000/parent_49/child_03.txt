import numpy as np
from scipy.optimize import minimize

def compute_penalty(vars_flat, n, r):
    """
    Computes the penalty for overlaps and boundary violations.
    Designed to be smooth enough for L-BFGS-B while strictly penalizing constraints.
    """
    centers = vars_flat.reshape((n, 2))
    
    # Pairwise distances
    c1 = centers[:, np.newaxis, :]
    c2 = centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum((c1 - c2)**2, axis=2))
    
    # Overlap penalty: penalize when dist < 2r
    overlaps = np.maximum(0, 2*r - dists)
    penalty = np.sum(overlaps**2)
    
    # Boundary penalties: ensure r <= x,y <= 1-r
    p_x = np.sum(np.maximum(0, r - centers[:, 0])**2) + np.sum(np.maximum(0, centers[:, 0] - (1-r))**2)
    p_y = np.sum(np.maximum(0, r - centers[:, 1])**2) + np.sum(np.maximum(0, centers[:, 1] - (1-r))**2)
    penalty += p_x + p_y
    
    return penalty

def run_packing():
    n = 26
    best_sum = 0.0
    best_out = (None, None, 0.0)
    
    # Run multiple optimization starts to avoid local minima
    for seed in range(6):
        np.random.seed(42 + seed)
        
        # Initialize with a staggered hexagonal pattern
        centers = np.zeros((n, 2))
        idx = 0
        row_counts = [6, 5, 6, 5, 4]  # Sums to 26
        y = 0.15
        dy = 0.8 / 4.0
        for count in row_counts:
            xs = np.linspace(0.15, 0.85, count)
            for x in xs:
                if idx < n:
                    centers[idx] = [x, y]
                    idx += 1
            y += dy
        
        # Add controlled noise to break symmetry
        centers += np.random.uniform(-0.03, 0.03, centers.shape)
        
        r = 0.06
        step = 0.002
        
        # Continuation method: grow radius while optimizing positions
        for _ in range(400):
            res = minimize(compute_penalty, centers.flatten(), args=(n, r), 
                           method='L-BFGS-B', bounds=[(0.0, 1.0)]*(2*n), 
                           options={'maxiter': 1000, 'ftol': 1e-12})
            centers = res.x.reshape((n, 2))
            
            # Check validity for current r
            dists = np.sqrt(np.sum((centers[:, np.newaxis, :] - centers[np.newaxis, :, :])**2, axis=2))
            np.fill_diagonal(dists, 2*r)
            min_d = np.min(dists[np.triu_indices(n, k=1)])
            
            if min_d >= 2*r - 1e-7:
                r += step
            else:
                step *= 0.9
                if step < 1e-6:
                    break
        
        # Post-processing: ensure strict validity
        centers = np.clip(centers, r, 1-r)
        dists = np.sqrt(np.sum((centers[:, np.newaxis, :] - centers[np.newaxis, :, :])**2, axis=2))
        np.fill_diagonal(dists, 2*r)
        min_d = np.min(dists[np.triu_indices(n, k=1)])
        
        # If any overlap remains (even tiny), reduce r slightly to satisfy constraints
        if min_d < 2*r:
            r = min_d / 2.0
            
        sum_r = n * r
        if sum_r > best_sum:
            best_sum = sum_r
            best_out = (centers.copy(), np.full(n, r), sum_r)
            
    return best_out
