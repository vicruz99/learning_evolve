def solve_packing():
    # Get initial guess
    centers, radii = get_initial_packing()
    
    # Flatten variables for optimization: [x1, y1, r1, x2, y2, r2, ...]
    x0 = np.concatenate([centers.flatten(), radii])
    
    n = 26
    
    def objective(vars):
        # Minimize negative sum of radii
        r = vars[5::3] # Radii are at indices 2, 5, 8...
        return -np.sum(r)

    def constraints_factory():
        cons = []
        
        # 1. Non-negative radii
        # r_i >= 0
        # Handled by bounds usually, but let's add explicit for clarity or rely on bounds
        
        # 2. Boundary constraints
        # x_i - r_i >= 0  => r_i - x_i <= 0
        # x_i + r_i <= 1  => x_i + r_i - 1 <= 0
        # y_i - r_i >= 0
        # y_i + r_i <= 1
        
        for i in range(n):
            # Indices: x=3i, y=3i+1, r=3i+2
            idx_x = 3 * i
            idx_y = 3 * i + 1
            idx_r = 3 * i + 2
            
            # x - r >= 0
            def make_bound_x_minus_r(i, idx_x, idx_r):
                return lambda v: v[idx_x] - v[idx_r]
            cons.append({'type': 'ineq', 'fun': make_bound_x_minus_r(i, idx_x, idx_r)})
            
            # 1 - x - r >= 0
            def make_bound_1_minus_x_r(i, idx_x, idx_r):
                return lambda v: 1.0 - v[idx_x] - v[idx_r]
            cons.append({'type': 'ineq', 'fun': make_bound_1_minus_x_r(i, idx_x, idx_r)})
            
            # y - r >= 0
            def make_bound_y_minus_r(i, idx_y, idx_r):
                return lambda v: v[idx_y] - v[idx_r]
            cons.append({'type': 'ineq', 'fun': make_bound_y_minus_r(i, idx_y, idx_r)})
            
            # 1 - y - r >= 0
            def make_bound_1_minus_y_r(i, idx_y, idx_r):
                return lambda v: 1.0 - v[idx_y] - v[idx_r]
            cons.append({'type': 'ineq', 'fun': make_bound_1_minus_y_r(i, idx_y, idx_r)})

        # 3. Pairwise separation
        # dist(i, j) >= r_i + r_j
        # dist^2 >= (r_i + r_j)^2
        # (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
        
        for i in range(n):
            for j in range(i + 1, n):
                idx_xi, idx_yi, idx_ri = 3*i, 3*i+1, 3*i+2
                idx_xj, idx_yj, idx_rj = 3*j, 3*j+1, 3*j+2
                
                def make_sep(i, j, idx_xi, idx_yi, idx_ri, idx_xj, idx_yj, idx_rj):
                    def fun(v):
                        xi, yi, ri = v[idx_xi], v[idx_yi], v[idx_ri]
                        xj, yj, rj = v[idx_xj], v[idx_yj], v[idx_rj]
                        dist_sq = (xi - xj)**2 + (yi - yj)**2
                        r_sum_sq = (ri + rj)**2
                        return dist_sq - r_sum_sq
                    return fun
                cons.append({'type': 'ineq', 'fun': make_sep(i, j, idx_xi, idx_yi, idx_ri, idx_xj, idx_yj, idx_rj)})
                
        return cons

    cons = constraints_factory()
    
    # Bounds for variables
    # x, y in [0, 1]
    # r in [0, 0.5] (max radius is 0.5)
    bounds = []
    for _ in range(n):
        bounds.append((0, 1)) # x
        bounds.append((0, 1)) # y
        bounds.append((0, 0.5)) # r

    # Solve
    # Use SLSQP
    try:
        res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, 
                       options={'maxiter': 1000, 'ftol': 1e-12})
        
        if res.success:
            final_vars = res.x
            final_centers = final_vars[:n*2].reshape((n, 2))
            final_radii = final_vars[n*2:]
            return final_centers, final_radii, -res.fun
        else:
            # If optimization fails, return initial guess (which should be valid)
            # But initial guess might not be optimal.
            # Let's try to return initial anyway.
            return centers, radii, np.sum(radii)
            
    except Exception as e:
        # Fallback
        return centers, radii, np.sum(radii)

def run_packing():
    centers, radii, sum_radii = solve_packing()
    return centers, radii, sum_radii
