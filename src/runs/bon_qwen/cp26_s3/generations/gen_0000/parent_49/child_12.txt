import numpy as np
from scipy.optimize import minimize

N = 26

def objective(v):
    # Maximize sum of radii -> minimize negative sum
    return -np.sum(v[2*N:])

def constraint_func(v):
    # Returns constraint values that must be >= 0 for SLSQP 'ineq'
    cx = v[:N]
    cy = v[N:2*N]
    r = v[2*N:]
    cons = []
    
    # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    for i in range(N):
        for j in range(i+1, N):
            dx = cx[i] - cx[j]
            dy = cy[i] - cy[j]
            dist_sq = dx*dx + dy*dy
            cons.append(dist_sq - (r[i] + r[j])**2)
            
        # Boundary constraints: x - r >= 0, 1 - x - r >= 0, etc.
        cons.append(cx[i] - r[i])
        cons.append(1.0 - cx[i] - r[i])
        cons.append(cy[i] - r[i])
        cons.append(1.0 - cy[i] - r[i])
        
    return np.array(cons)

def get_initial_guess(seed):
    np.random.seed(seed)
    # Start with a dense 5x5 grid-like arrangement
    x_grid = np.linspace(0.12, 0.88, 5)
    y_grid = np.linspace(0.12, 0.88, 5)
    pts = []
    for xi in x_grid:
        for yi in y_grid:
            pts.append([xi, yi])
            if len(pts) >= N:
                break
        if len(pts) >= N:
            break
            
    # Fill remaining if needed (should be exactly 25 from grid, add 1)
    while len(pts) < N:
        pts.append([np.random.uniform(0.2, 0.8), np.random.uniform(0.2, 0.8)])
        
    cx_init = np.array([p[0] for p in pts])
    cy_init = np.array([p[1] for p in pts])
    r_init = np.full(N, 0.06)
    
    # Add controlled noise to break symmetry and help optimizer explore
    cx_init += np.random.uniform(-0.02, 0.02, N)
    cy_init += np.random.uniform(-0.02, 0.02, N)
    cx_init = np.clip(cx_init, 0.05, 0.95)
    cy_init = np.clip(cy_init, 0.05, 0.95)
    
    return np.concatenate([cx_init, cy_init, r_init])

def run_packing():
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Bounds for variables: centers in [0,1], radii in [eps, 0.5]
    bounds = [(0.0, 1.0)] * (2*N) + [(1e-6, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraint_func}
    
    # Run multiple restarts to avoid poor local optima
    for seed in range(10):
        v0 = get_initial_guess(seed)
        res = minimize(objective, v0, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 3000, 'ftol': 1e-10, 'disp': False})
        
        # Verify feasibility with strict tolerance
        con_vals = constraint_func(res.x)
        if np.min(con_vals) >= -1e-8:
            curr_sum = np.sum(res.x[2*N:])
            if curr_sum > best_sum:
                best_sum = curr_sum
                best_centers = np.column_stack([res.x[:N], res.x[N:2*N]])
                best_radii = res.x[2*N:]
                
    # Fallback to initial guess if optimization fails unexpectedly
    if best_centers is None:
        v0 = get_initial_guess(0)
        best_centers = np.column_stack([v0[:N], v0[N:2*N]])
        best_radii = v0[2*N:]
        best_sum = np.sum(best_radii)
        
    return best_centers, best_radii, best_sum
