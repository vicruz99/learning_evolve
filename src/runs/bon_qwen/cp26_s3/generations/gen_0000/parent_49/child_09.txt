from scipy.optimize import minimize

    return -v[52]

def constraint_func(v):
    """
    Returns array of inequality constraints (must be >= 0):

    - Non-overlap constraints: ||c_i - c_j||^2 >= 4r^2

    r = v[52]
    centers = v[:52].reshape(n, 2)
    num_con = n * 4 + n * (n - 1) // 2
    con = np.empty(num_con)
    idx = 0
    for i in range(n):
        con[idx] = centers[i, 0] - r
        idx += 1

        idx += 1
        con[idx] = centers[i, 1] - r
        idx += 1
        con[idx] = 1.0 - centers[i, 1] - r
        idx += 1
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            con[idx] = dx*dx + dy*dy - 4.0 * r * r
            idx += 1
    return con

            pts.append([start_x + j, y])

    cons = {'type': 'ineq', 'fun': constraint_func}

    return opt_centers, opt_radii, np.sum(opt_radii)
