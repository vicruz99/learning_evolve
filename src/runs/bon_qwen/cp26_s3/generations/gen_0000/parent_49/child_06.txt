To maximize the sum of radii for 26 circles in a unit square, we formulate this as a constrained nonlinear optimization problem. The objective is to maximize $\sum r_i$, equivalent to minimizing $-\sum r_i$. The constraints include boundary conditions ($0 \le r_i \le x_i \le 1-r_i$, etc.) and non-overlap conditions ($(x_i-x_j)^2 + (y_i-y_j)^2 \ge (r_i+r_j)^2$).

Strategy:
1. **Initialization**: Start with a hexagonal lattice packing, which is known to be dense. This provides a geometrically sound starting point close to a local optimum, reducing the search space for the optimizer.

3. **Constraint Handling**: All $5N$ boundary constraints and $N(N-1)/2$ pairwise distance constraints are computed in a single vectorized function. This avoids closures and lambda functions as required.

import numpy as np
from scipy.optimize import minimize

N = 26

def compute_constraints(vars):
    """Compute all inequality constraints as a single vector."""
    centers = vars[:2*N].reshape(N, 2)
    radii = vars[2*N:]
    
    con = []
    
    # Boundary constraints: 0 <= x-r, r <= x <= 1-r, etc.
    con.append(centers[:, 0] - radii)
    con.append(1.0 - centers[:, 0] - radii)
    con.append(centers[:, 1] - radii)
    con.append(1.0 - centers[:, 1] - radii)
    con.append(radii)
    
    # Pairwise non-overlap constraints: dist^2 >= (r_i + r_j)^2
    # Vectorized computation for all pairs
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist_sq = np.sum(diff**2, axis=2)
    
    # Extract upper triangle indices to avoid duplicates and self-pairs
    mask = np.triu(np.ones((N, N)), k=1).astype(bool)
    r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    
    con.append(dist_sq[mask] - r_sum[mask]**2)
    
    return np.concatenate(con)

def obj(vars):
    """Objective function: negative sum of radii."""
    return -np.sum(vars[2*N:])

def run_packing():
    # 1. Initialization: Hexagonal packing pattern
    r0 = 0.09
    centers = np.zeros((N, 2))
    radii = np.full(N, r0)
    
    idx = 0
    row = 0
    while idx < N:
        col = 0
        while idx < N:
            # Hexagonal offset for odd rows
            xc = r0 + col * 2 * r0 + (r0 if row % 2 == 1 else 0)
            yc = r0 + row * r0 * np.sqrt(3)
            
            if xc <= 1 - r0 and yc <= 1 - r0:
                centers[idx] = [xc, yc]
                idx += 1
            else:
                break
            col += 1
        row += 1
        
    x0 = np.concatenate([centers.flatten(), radii])
    
    # 2. Bounds and Constraints
    bnds = [(0.0, 1.0) for _ in range(2*N)]
    bnds += [(0.0, 0.5) for _ in range(N)]
    
    cons = {'type': 'ineq', 'fun': compute_constraints}
    
    # 3. Optimization
    try:
        res = minimize(obj, x0, method='SLSQP', bounds=bnds, constraints=cons, 
                       options={'maxiter': 5000, 'ftol': 1e-10, 'disp': False})
        best_centers = res.x[:2*N].reshape(N, 2)
        best_radii = res.x[2*N:]
    except Exception:
        best_centers = centers
        best_radii = radii
        
    # 4. Safety margin to ensure strict validity against 1e-12 tolerance
    best_radii *= 0.9999
    
    return best_centers, best_radii, np.sum(best_radii)
