import numpy as np
from scipy.optimize import minimize

def compute_penalty(centers, r):
    """Compute squared penalty for boundary and overlap constraints."""
    n = centers.shape[0]
    penalty = 0.0
    
    # Boundary constraints
    diff_l = r - centers[:, 0]
    mask_l = diff_l > 0
    if np.any(mask_l):
        penalty += np.sum(diff_l[mask_l]**2)
        
    diff_r = centers[:, 0] - (1.0 - r)
    mask_r = diff_r > 0
    if np.any(mask_r):
        penalty += np.sum(diff_r[mask_r]**2)
        
    diff_b = r - centers[:, 1]
    mask_b = diff_b > 0
    if np.any(mask_b):
        penalty += np.sum(diff_b[mask_b]**2)
        
    diff_t = centers[:, 1] - (1.0 - r)
    mask_t = diff_t > 0
    if np.any(mask_t):
        penalty += np.sum(diff_t[mask_t]**2)
        
    # Pairwise constraints
    c1 = centers[:, np.newaxis, :]
    c2 = centers[np.newaxis, :, :]
    diff = c1 - c2
    dist_sq = np.sum(diff**2, axis=2)
    
    idx = np.triu_indices(n, 1)
    dist_sq_ij = dist_sq[idx]
    min_dist = 2.0 * r
    
    dist = np.sqrt(np.maximum(dist_sq_ij, 1e-12))
    diff_dist = min_dist - dist
    mask_p = diff_dist > 0
    if np.any(mask_p):
        penalty += np.sum(diff_dist[mask_p]**2)
        
    return penalty

def objective_func(pos, centers_shape, r_val):
    """Wrapper for scipy optimizer."""
    centers = pos.reshape(centers_shape)
    return compute_penalty(centers, r_val)
