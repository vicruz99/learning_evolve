Our strategy involves a two-phase approach:
1. **Initialization**: Start with a high-density configuration, such as a $5 \times 5$ grid with an additional circle, to ensure a valid starting point with non-zero radii.
2. **Iterative Optimization (LP + Force-Directed Layout)**:
    - **Linear Programming (LP)**: For fixed center positions, finding the optimal radii that maximize the sum is a Linear Programming problem. We maximize $\sum r_i$ subject to non-overlap constraints ($r_i + r_j \le d_{ij}$) and boundary constraints ($r_i \le x_i, 1-x_i, y_i, 1-y_i$).
    - **Force-Directed Adjustment**: The LP solution identifies which constraints are active (tight). We use this information to calculate forces: circles that are touching (active constraints) are pushed apart to increase the distances $d_{ij}$, which in turn allows the radii to grow in the next LP step. Wall constraints also generate forces pushing circles away from the boundaries.
3. **Refinement**: We perform multiple iterations of solving the LP and adjusting centers, gradually reducing the step size to converge to a local optimum. We also apply random perturbations to help escape local minima.

import numpy as np
import scipy.optimize as opt

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Pack 26 circles in a unit square [0,1]x[0,1] to maximize sum of radii.
    """
    n = 26
    centers = np.zeros((n, 2))
    
    # --- 1. Initialization ---
    # Start with a 5x5 grid (25 circles) and place the 26th in the center
    # Grid points for 5x5
    grid_x = np.linspace(0.1, 0.9, 5)
    grid_y = np.linspace(0.1, 0.9, 5)
    
    idx = 0
    for y in grid_y:
        for x in grid_x:
            centers[idx] = [x, y]
            idx += 1
    
    # Place 26th circle in the center with small perturbation
    centers[25] = [0.5, 0.5]
    
    # Add small random noise to avoid symmetry issues
    centers += np.random.uniform(-0.001, 0.001, centers.shape)
    
    # Clip to valid range
    np.clip(centers, 0, 1, out=centers)

    # --- 2. Optimization Loop ---
    # We will iterate: Solve LP for radii -> Calculate forces -> Move centers
    
    # Parameters for optimization
    max_iters = 150
    force_strength = 0.05
    step_decay = 0.95
    
    # Precompute indices for pairs to avoid recomputing
    pair_indices = [(i, j) for i in range(n) for j in range(i + 1, n)]
    n_pairs = len(pair_indices)
    
    for iteration in range(max_iters):
        # --- Step A: Solve LP to find optimal radii for current centers ---
        # Variables: r_0, ..., r_25
        # Objective: Maximize sum(r_i) -> Minimize -sum(r_i)
        c_obj = -np.ones(n)
        
        # Constraints: A_ub @ x <= b_ub
        # 1. Pairwise constraints: r_i + r_j <= dist(i, j)
        # 2. Wall constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
        # 3. Non-negativity: r_i >= 0 (handled by bounds)
        
        # Build A_ub and b_ub
        # We will construct these dynamically or in arrays
        # Size: n_pairs + 4*n constraints
        
        A_ub = np.zeros((n_pairs + 4 * n, n))
        b_ub = np.zeros(n_pairs + 4 * n)
        
        # Compute distances
        dists = np.zeros(n_pairs)
        for k, (i, j) in enumerate(pair_indices):
            d = np.linalg.norm(centers[i] - centers[j])
            dists[k] = d
            
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = d
            
        # Wall constraints
        for i in range(n):
            row_offset = n_pairs
            # x >= r  => -r <= -x  => r <= x
            # Actually constraint is r <= x, r <= 1-x, etc.
            # r_i <= x_i
            A_ub[row_offset + 4*i, i] = 1.0
            b_ub[row_offset + 4*i] = centers[i, 0]
            
            # r_i <= 1 - x_i
            A_ub[row_offset + 4*i + 1, i] = 1.0
            b_ub[row_offset + 4*i + 1] = 1.0 - centers[i, 0]
            
            # r_i <= y_i
            A_ub[row_offset + 4*i + 2, i] = 1.0
            b_ub[row_offset + 4*i + 2] = centers[i, 1]
            
            # r_i <= 1 - y_i
            A_ub[row_offset + 4*i + 3, i] = 1.0
            b_ub[row_offset + 4*i + 3] = 1.0 - centers[i, 1]
            
        bounds = [(0, None) for _ in range(n)]
        
        try:
            res = opt.linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
            if not res.success:
                # Fallback or break if LP fails
                break
            radii = res.x
        except Exception:
            break
            
        # Check if radii are valid (not nan/inf)
        if np.isnan(radii).any() or np.isinf(radii).any():
            break

        # --- Step B: Calculate Forces and Move Centers ---
        # We want to increase distances for active constraints.
        # A constraint is active if r_i + r_j is close to dist(i, j).
        
        forces = np.zeros_like(centers)
        
        # Threshold for active constraint
        eps = 1e-6
        
        for k, (i, j) in enumerate(pair_indices):
            r_sum = radii[i] + radii[j]
            d = dists[k]
            
            # If touching (or very close), push apart
            if d < r_sum + eps:
                # Vector from j to i
                if d > 1e-9:
                    dir_vec = (centers[i] - centers[j]) / d
                    # Force magnitude: proportional to overlap or just constant?
                    # Using a constant repulsive force for active contacts
                    f_mag = force_strength 
                    forces[i] += f_mag * dir_vec
                    forces[j] -= f_mag * dir_vec
                else:
                    # If centers coincide, push randomly
                    forces[i] += np.random.randn(2) * force_strength
                    forces[j] -= np.random.randn(2) * force_strength

        for i in range(n):
            r = radii[i]
            x, y = centers[i]
            
            if x - r < eps:
                forces[i, 0] += force_strength # Push right
            if x + r > 1 - eps:
                forces[i, 0] -= force_strength # Push left
            if y - r < eps:
                forces[i, 1] += force_strength # Push up
            if y + r > 1 - eps:
                forces[i, 1] -= force_strength # Push down

        # Update centers
        # Normalize forces? No, just scale by step size
        # Adaptive step size?
        current_step = force_strength * (0.5 + 0.5 * np.cos(iteration * np.pi / max_iters))
        if current_step < 0: current_step = 0.01 # Minimum step
        
        centers += forces * current_step
        
        # Clip centers to [0, 1] strictly
        # Actually, allowing them to go slightly out and clip is fine, 
        # but better to keep inside.
        np.clip(centers, 1e-6, 1 - 1e-6, out=centers)

    # --- 3. Final Polish ---
    # Run one last LP to get final radii
    # Re-compute distances and solve LP
    A_ub = np.zeros((n_pairs + 4 * n, n))
    b_ub = np.zeros(n_pairs + 4 * n)
    
    for k, (i, j) in enumerate(pair_indices):

        row_offset = n_pairs
        A_ub[row_offset + 4*i, i] = 1.0
        b_ub[row_offset + 4*i] = centers[i, 0]
        A_ub[row_offset + 4*i + 1, i] = 1.0
        b_ub[row_offset + 4*i + 1] = 1.0 - centers[i, 0]
        A_ub[row_offset + 4*i + 2, i] = 1.0
        b_ub[row_offset + 4*i + 2] = centers[i, 1]
        A_ub[row_offset + 4*i + 3, i] = 1.0
        b_ub[row_offset + 4*i + 3] = 1.0 - centers[i, 1]
        
    bounds = [(0, None) for _ in range(n)]
    res = opt.linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    radii = res.x
    
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
