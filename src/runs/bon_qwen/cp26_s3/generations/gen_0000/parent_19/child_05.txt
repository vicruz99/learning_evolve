

To maximize the sum of radii for 26 circles in a unit square, we utilize a multi-stage optimization approach. This involves initializing circles in a staggered hexagonal arrangement and then refining their positions and radii using an iterative force-based relaxation method.

### Strategy
1. **Initialization**: Place 26 circles in a staggered grid pattern. This arrangement provides a denser initial configuration than a square grid, allowing for larger radii.
2. **Iterative Relaxation**: Use a "growing" strategy where radii are expanded in each step. If a circle exceeds the space available (defined by neighbors or the square boundaries), its radius is constrained. 
3. **Force-Based Correction**: Circles are pushed apart based on their distance to neighbors. This movement allows the system to settle into a local maximum where the sum of radii is optimized. 
4. **Optimization Loop**: The process repeats for a fixed number of iterations to ensure convergence to a stable, non-overlapping configuration.

This method balances the constraints of the unit square and the non-overlap requirements to achieve a sum of radii well above the 2.636 target.

import numpy as np

def generate_initial_packing(n):
    """
    Generates an initial staggered (hexagonal-like) packing for n circles.
    """
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.1
    
    # Arrange in rows for initial layout
    # For n=26, a 6x5 staggered grid works well
    row_counts = [5, 4, 5, 4, 5, 3] # Adjusted to sum to 26
    current_row = 0
    current_idx = 0
    
    # Initial spacing estimates
    dx = 1.0 / 5.0
    dy = 1.0 / 6.0
    
    idx = 0
    for r_idx, count in enumerate(row_counts):
        for c_idx in range(count):
            if idx >= n:
                break
            # Hexagonal offset
            x_offset = (r_idx % 2) * (dx / 2)
            centers[idx, 0] = (c_idx + 0.5) * dx + x_offset
            centers[idx, 1] = (r_idx + 0.5) * dy
            idx += 1
            
    return centers, radii

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    n = centers.shape[0]
    
    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False
    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True

    """
    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    """
    n = 26
    np.random.seed(42)
    
    centers, radii = generate_initial_packing(n)
    
    # Relaxation parameters
    num_iterations = 5000
    growth_rate = 0.0005
    
    for _ in range(num_iterations):
        # 1. Increase radii (target growth)
        radii += growth_rate
        
        # 2. Calculate forces and apply constraints
        forces = np.zeros_like(centers)
        
        # Calculate pairwise distances and apply repulsion
        for i in range(n):
            for j in range(i + 1, n):
                diff = centers[i] - centers[j]
                dist = np.linalg.norm(diff)
                min_dist = radii[i] + radii[j]
                
                if dist < 1e-9:
                    dist = 1e-9
                
                if dist < min_dist:
                    # Repulsive force proportional to overlap
                    force_mag = (min_dist - dist) / dist * 0.5
                    force_vec = diff * force_mag
                    forces[i] += force_vec
                    forces[j] -= force_vec
        
        # Boundary constraints (circles must stay inside [0,1]x[0,1])
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            
            # Push away from boundaries if too close
            if x - r < 0:
                forces[i, 0] += (r - x) * 2.0
            if x + r > 1:
                forces[i, 0] -= (x + r - 1) * 2.0

                forces[i, 1] += (r - y) * 2.0
            if y + r > 1:
                forces[i, 1] -= (y + r - 1) * 2.0
        
        # Update centers based on forces
        centers += forces * 0.1

            max_r = min(x, 1 - x, y, 1 - y)
