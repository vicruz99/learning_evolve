import numpy as np
from scipy.optimize import linprog

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    
    Strategy:
    1. Use a force-directed growth simulation to find a good arrangement of centers.
       - Starts with small circles, expands them while resolving overlaps via repulsive forces.
       - Runs multiple times with different random seeds to find a good local optimum.
    2. Once centers are fixed, use Linear Programming (LP) to optimize the radii.
       - This maximizes sum(radii) subject to distance and boundary constraints.
       - This step allows radii to be unequal, potentially filling small gaps.
    """
    n = 26
    best_sum_radii = -1.0
    best_centers = None
    best_radii = None
    
    # Parameters for the force simulation
    num_restarts = 20
    initial_radius = 0.02
    radius_growth = 0.0003
    max_iters = 3000
    dt = 0.08  # Step size for center movement
    k_repulse = 1.0  # Repulsion strength
    
    for seed in range(num_restarts):
        np.random.seed(seed * 1337)
        
        # Initialize centers randomly within a safe margin
        # We leave a margin to allow initial growth
        centers = np.random.uniform(0.15, 0.85, size=(n, 2))
        
        r = initial_radius
        
        for _ in range(max_iters):
            forces = np.zeros_like(centers)
            
            # Calculate pairwise repulsive forces
            # Vectorized operations for performance
            # Centers shape: (n, 2)
            # Diffs shape: (n, n, 2)
            diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
            dists = np.linalg.norm(diffs, axis=2)
            
            # Create a mask for pairs (i, j) where i < j to avoid double counting and self-interaction
            # But vectorized loop over all pairs is easier to code, then adjust
            # We iterate explicitly for clarity and control
            
            for i in range(n):
                for j in range(i + 1, n):
                    d = dists[i, j]
                    if d < 2 * r and d > 1e-9:
                        # Overlap exists
                        overlap = 2 * r - d
                        # Force vector
                        direction = diffs[i, j] / d
                        force_vec = k_repulse * overlap * direction
                        forces[i] += force_vec
                        forces[j] -= force_vec
            
            # Boundary forces
            # Push centers away from walls if they violate r <= coord <= 1-r
            # This acts as a penalty/force to keep them valid
            for i in range(n):
                x, y = centers[i]
                # Left wall
                if x < r:
                    forces[i, 0] += k_repulse * (r - x)
                # Right wall
                if x > 1 - r:
                    forces[i, 0] -= k_repulse * (x - (1 - r))
                # Bottom wall
                if y < r:
                    forces[i, 1] += k_repulse * (r - y)
                # Top wall
                if y > 1 - r:
                    forces[i, 1] -= k_repulse * (y - (1 - r))
            
            # Update centers
            centers += dt * forces
            
            # Hard clamp to valid region [r, 1-r]
            # This ensures the current radius r is always valid for the positions
            centers[:, 0] = np.clip(centers[:, 0], r, 1 - r)
            centers[:, 1] = np.clip(centers[:, 1], r, 1 - r)
            
            # Grow radius
            r += radius_growth
            
            # Safety break if radius exceeds theoretical max (approx 0.12)
            if r > 0.12:
                break
        
        # After simulation, we have a valid configuration of equal circles with radius approx 'r'.
        # However, 'r' might be slightly inconsistent due to discrete steps.
        # We use Linear Programming to find the optimal radii for these fixed centers.
        # This maximizes sum(radii) and handles potential gaps.
        
        try:
            optimal_radii = solve_optimal_radii(centers)
            current_sum = np.sum(optimal_radii)
            
            if current_sum > best_sum_radii:
                best_sum_radii = current_sum
                best_centers = centers.copy()
                best_radii = optimal_radii.copy()
        except Exception:
            continue

    # Final validation and slight adjustment if needed
    if best_centers is None:
        # Fallback to simple grid if everything fails
        best_centers = np.array([(0.1 + 0.2*i, 0.1 + 0.2*j) for i in range(5) for j in range(5)] + [(0.5, 0.5)])
        best_centers = best_centers[:26]
        best_radii = np.full(26, 0.01)
        best_sum_radii = 0.26

    # Ensure radii are valid with respect to centers (LP should guarantee this, but just in case)
    # LP constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    # And r_i + r_j <= dist.
    # If for some reason LP fails or numeric issues, clip radii.
    for i in range(n):
        r = best_radii[i]
        x, y = best_centers[i]
        max_r_boundary = min(x, 1-x, y, 1-y)
        if r > max_r_boundary + 1e-9:
            best_radii[i] = max_r_boundary

    # Final check for overlaps (redundant if LP solved correctly, but good for safety)
    # If overlaps exist due to precision, shrink radii slightly
    valid = True
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(best_centers[i] - best_centers[j])
            req_dist = best_radii[i] + best_radii[j]
            if dist < req_dist - 1e-9:
                # Scale down radii involved to fix overlap
                # Simple heuristic: reduce both
                overlap = req_dist - dist
                best_radii[i] -= overlap / 2
                best_radii[j] -= overlap / 2
                valid = False
    
    # Recalculate sum
    final_sum = np.sum(best_radii)
    
    return (best_centers, best_radii, final_sum)

def solve_optimal_radii(centers):
    """
    Solves the Linear Programming problem to maximize sum of radii
    given fixed centers.
    
    Maximize: sum(r_i)
    Subject to:
      r_i >= 0
      r_i <= x_i
      r_i <= 1 - x_i
      r_i <= y_i
      r_i <= 1 - y_i
      r_i + r_j <= dist(i, j) for all i < j
    """
    n = centers.shape[0]
    
    # Objective: minimize -sum(r_i) => maximize sum(r_i)
    c = -np.ones(n)
    
    # Constraints in form A_ub * r <= b_ub
    
    A_ub = []
    b_ub = []
    
    # Boundary constraints:
    # r_i <= x_i  => r_i <= x_i
    # r_i <= 1 - x_i => r_i <= 1 - x_i
    # r_i <= y_i
    # r_i <= 1 - y_i
    
    for i in range(n):
        x, y = centers[i]
        
        # r_i <= x
        row = np.zeros(n)
        row[i] = 1.0
        A_ub.append(row)
        b_ub.append(x)
        
        # r_i <= 1 - x
        row = np.zeros(n)
        row[i] = 1.0
        A_ub.append(row)
        b_ub.append(1.0 - x)
        
        # r_i <= y
        row = np.zeros(n)
        row[i] = 1.0
        A_ub.append(row)
        b_ub.append(y)
        
        # r_i <= 1 - y
        row = np.zeros(n)
        row[i] = 1.0
        A_ub.append(row)
        b_ub.append(1.0 - y)
        
    # Pairwise constraints: r_i + r_j <= dist_ij
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dist)
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    # Bounds for r_i: r_i >= 0
    bounds = [(0, None) for _ in range(n)]
    
    # Solve LP
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return -res.x # Negate back because we minimized -sum
    else:
        raise Exception("LP optimization failed")
