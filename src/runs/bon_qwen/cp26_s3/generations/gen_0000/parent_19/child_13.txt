import numpy as np
from scipy.optimize import minimize
from functools import partial
import math

N = 26


def objective(vars):
    radii = vars[2 * N:]
    return -np.sum(radii)


def boundary_x_lower(v, i):
    return v[2 * i] - v[2 * N + i]


def boundary_x_upper(v, i):
    return 1.0 - v[2 * i] - v[2 * N + i]


def boundary_y_lower(v, i):
    return v[2 * i + 1] - v[2 * N + i]


def boundary_y_upper(v, i):
    return 1.0 - v[2 * i + 1] - v[2 * N + i]


def non_overlap(v, i, j):
    dx = v[2 * i] - v[2 * j]
    dy = v[2 * i + 1] - v[2 * j + 1]
    dr = v[2 * N + i] + v[2 * N + j]
    return dx * dx + dy * dy - dr * dr


def hexagonal_init(r):
    rows = [6, 5, 6, 5, 4]
    centers = []
    y = r
    for i, n_in_row in enumerate(rows):
        x_start = r
        if i % 2 == 1:
            x_start = 2 * r
        for j in range(n_in_row):
            centers.append([x_start + j * 2 * r, y])
        y += math.sqrt(3) * r
    centers = np.array(centers[:N])
    radii = np.full(N, r)
    return centers, radii


def perturbed_hexagonal_init(r, scale=0.5):
    centers, radii = hexagonal_init(r)
    perturbation = np.random.uniform(-scale * r, scale * r, centers.shape)
    centers = centers + perturbation
    centers = np.clip(centers, r + 1e-6, 1 - r - 1e-6)
    return centers, radii


def random_init():
    centers = np.random.uniform(0.15, 0.85, (N, 2))
    radii = np.full(N, 0.04)
    return centers, radii


def build_constraints():
    constraints = []
    for i in range(N):
        constraints.append({'type': 'ineq', 'fun': partial(boundary_x_lower, i=i)})
        constraints.append({'type': 'ineq', 'fun': partial(boundary_x_upper, i=i)})
        constraints.append({'type': 'ineq', 'fun': partial(boundary_y_lower, i=i)})
        constraints.append({'type': 'ineq', 'fun': partial(boundary_y_upper, i=i)})
    for i in range(N):
        for j in range(i + 1, N):
            constraints.append({'type': 'ineq', 'fun': partial(non_overlap, i=i, j=j)})
    return constraints


def get_bounds():
    bounds = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    return bounds


def optimize_from_init(centers, radii, constraints, bounds, maxiter=3000):
    vars0 = np.concatenate([centers.flatten(), radii])
    result = minimize(
        objective,
        vars0,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': maxiter, 'ftol': 1e-15, 'disp': False}
    )
    return result


def refine_packing(centers, radii, constraints, bounds):
    for _ in range(3):
        vars0 = np.concatenate([centers.flatten(), radii])
        result = optimize_from_init(centers, radii, constraints, bounds, maxiter=2000)
        centers = result.x[:2 * N].reshape(N, 2)
        radii = result.x[2 * N:]
    return centers, radii


def run_packing():
    constraints = build_constraints()
    bounds = get_bounds()

    best_sum = -1.0
    best_centers = None
    best_radii = None

    init_configs = [
        hexagonal_init(0.06),
        hexagonal_init(0.07),
        hexagonal_init(0.08),
        hexagonal_init(0.09),
    ]

    for seed in range(15):
        np.random.seed(42 + seed)
        if seed < 4:
            centers, radii = init_configs[seed]
        else:
            r_base = 0.06 + np.random.uniform(-0.01, 0.03)
            centers, radii = perturbed_hexagonal_init(r_base, scale=0.3)

        # Small random perturbation for diversity
        centers = centers + np.random.uniform(-0.005, 0.005, centers.shape)
        radii = np.maximum(radii + np.random.uniform(-0.003, 0.003, N), 0.01)

        result = optimize_from_init(centers, radii, constraints, bounds, maxiter=5000)

        sum_r = -result.fun
        if sum_r > best_sum:
            best_sum = sum_r
            best_centers = result.x[:2 * N].reshape(N, 2)
            best_radii = result.x[2 * N:]

    # Additional refinement on best solution
    best_centers, best_radii = refine_packing(best_centers, best_radii, constraints, bounds)

    # Final optimization pass
    vars0 = np.concatenate([best_centers.flatten(), best_radii])
    result = optimize_from_init(best_centers, best_radii, constraints, bounds, maxiter=10000)

    best_centers = result.x[:2 * N].reshape(N, 2)
    best_radii = result.x[2 * N:]

    # Ensure radii are non-negative
    best_radii = np.maximum(best_radii, 0.0)

    # Clamp centers to valid range
    for i in range(N):
        r = best_radii[i]
        best_centers[i, 0] = np.clip(best_centers[i, 0], r, 1.0 - r)
        best_centers[i, 1] = np.clip(best_centers[i, 1], r, 1.0 - r)

    final_sum = np.sum(best_radii)

    return best_centers, best_radii, final_sum
