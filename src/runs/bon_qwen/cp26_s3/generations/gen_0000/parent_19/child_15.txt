import numpy as np
import scipy.optimize
import math

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    Pack 26 circles in a unit square to maximize sum of radii.
    Uses scipy.optimize to find a local optimum starting from a hexagonal lattice configuration.
    """

    # --- Helper functions ---
    
    def distance(p1, p2):
        return np.sqrt(np.sum((p1 - p2) ** 2))

    def objective(vars_flat):
        # vars_flat contains [x1, y1, r1, x2, y2, r2, ...]
        centers = vars_flat[0::3].reshape(-1, 2)
        radii = vars_flat[1::3]
        # We want to maximize sum of radii, so minimize negative sum
        return -np.sum(radii)

    def constraint_boundary(vars_flat):
        # Returns array of constraints >= 0
        # x >= r, 1-x >= r => x <= 1-r => 1-x-r >= 0
        # y >= r, 1-y >= r => y <= 1-r => 1-y-r >= 0
        centers = vars_flat[0::3].reshape(-1, 2)
        radii = vars_flat[1::3]
        
        constraints = []
        for i in range(n_circles):
            x, y = centers[i]
            r = radii[i]
            # x - r >= 0
            constraints.append(x - r)
            # 1 - x - r >= 0
            constraints.append(1.0 - x - r)
            # y - r >= 0
            constraints.append(y - r)
            # 1 - y - r >= 0
            constraints.append(1.0 - y - r)
        return np.array(constraints)

    def constraint_non_overlap(vars_flat):
        # Returns array of constraints >= 0

    v_spacing = math.sqrt(3) * r_guess
    
    centers_init = []
    radii_init = []

    row = 0
    while current_idx < n_circles:
        # Determine number of circles in this row
        # In hex packing, rows alternate.

        # Odd rows (1, 3, ...): centers at x = 2r, 4r, 6r... (shifted by r)
