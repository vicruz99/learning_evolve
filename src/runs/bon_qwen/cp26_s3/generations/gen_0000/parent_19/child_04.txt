import numpy as np
from scipy.optimize import minimize


N = 26


def objective_with_penalty(x, penalty_scale=1e6):
    cs = x[:2 * N].reshape(N, 2)
    rs = x[2 * N:]

    obj = -np.sum(rs)
    penalty = 0.0

    # Boundary penalty
    x_coords = cs[:, 0]
    y_coords = cs[:, 1]

    for i in range(N):
        r = rs[i]
        xc = x_coords[i]
        yc = y_coords[i]

        if xc - r < 0:
            penalty += penalty_scale * (xc - r) ** 2
        if xc + r > 1:
            penalty += penalty_scale * (xc + r - 1) ** 2
        if yc - r < 0:
            penalty += penalty_scale * (yc - r) ** 2
        if yc + r > 1:
            penalty += penalty_scale * (yc + r - 1) ** 2

    # Overlap penalty (vectorized)
    dists = np.sqrt(((cs[:, np.newaxis, :] - cs[np.newaxis, :, :]) ** 2).sum(axis=2))
    np.fill_diagonal(dists, np.inf)

    rs_sum = rs[:, np.newaxis] + rs[np.newaxis, :]
    overlaps = np.maximum(0, rs_sum - dists)
    penalty += penalty_scale * np.sum(overlaps ** 2)

    # Radius non-negativity
    neg_radii = rs[rs < 0]
    if len(neg_radii) > 0:
        penalty += penalty_scale * np.sum(neg_radii ** 2)

    return obj + penalty


def create_hexagonal_init():
    centers = np.zeros((N, 2))
    radii = np.ones(N) * 0.06

    idx = 0
    spacing = 0.14
    for row in range(6):
        for col in range(5):
            if idx >= N:
                break
            x = 0.12 + col * spacing + (spacing / 2 if row % 2 == 1 else 0)
            y = 0.12 + row * spacing * 0.866
            if x <= 0.9 and y <= 0.9:
                centers[idx] = [x, y]
                idx += 1

    for i in range(idx, N):
        centers[i] = [0.5, 0.5]

    return centers, radii


def create_random_init(seed):
    np.random.seed(seed)
    centers = np.random.uniform(0.15, 0.85, (N, 2))
    radii = np.ones(N) * 0.05
    return centers, radii


def optimize_from_start(centers, radii, penalty_scale=1e6):
    x0 = np.concatenate([centers.flatten(), radii])
    bounds = [(0.005, 0.995)] * (2 * N) + [(0.001, 0.35)] * N

    result = minimize(
        objective_with_penalty,
        x0,
        args=(penalty_scale,),
        method="L-BFGS-B",
        bounds=bounds,
        options={"maxiter": 100000, "ftol": 1e-15, "gtol": 1e-10},
    )

    centers_opt = result.x[:2 * N].reshape(N, 2)
    radii_opt = result.x[2 * N:]
    radii_opt = np.maximum(radii_opt, 0)

    return centers_opt, radii_opt

        best_sum = s
        best_centers = c_opt.copy()
        best_radii = r_opt.copy()

    # Try multiple random initializations
    for seed in range(20):
        centers, radii = create_random_init(seed)
        c_opt, r_opt = optimize_from_start(centers, radii)
        s = np.sum(r_opt)
        if s > best_sum:
            best_sum = s
            best_centers = c_opt.copy()
            best_radii = r_opt.copy()
