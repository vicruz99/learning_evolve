def compute_overlap_energy(centers, r):
    """
    Computes the sum of squared overlaps between circles and boundaries.
    Overlap between circles i and j is max(0, 2r - distance)^2.
    Overlap with boundary is max(0, r - min_dist_to_edge)^2.
    """
    n = centers.shape[0]
    energy = 0.0
    
    # Circle-circle overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            overlap = 2 * r - dist
            if overlap > 0:
                energy += overlap ** 2
                
    # Boundary overlaps
    # Centers must be in [r, 1-r]
    # If x < r, overlap is (r - x)
    # If x > 1-r, overlap is (x - (1-r))
    for i in range(n):
        x, y = centers[i]
        # Left
        if x < r:
            energy += (r - x) ** 2
        # Right
        if x > 1 - r:
            energy += (x - (1 - r)) ** 2
        # Bottom
        if y < r:
            energy += (r - y) ** 2
        # Top
        if y > 1 - r:
            energy += (y - (1 - r)) ** 2
            
    return energy
