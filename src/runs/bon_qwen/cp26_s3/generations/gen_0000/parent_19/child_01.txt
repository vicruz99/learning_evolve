import numpy as np
from scipy.optimize import minimize, linprog

def evaluate_packing(centers):
    """
    Given a set of centers, solve an LP to find the maximum radii
    such that circles do not overlap and stay inside [0,1]x[0,1].
    """
    n = centers.shape[0]
    
    # Variables for the LP: r_0, r_1, ..., r_{n-1}
    # We want to maximize sum(r_i), which is equivalent to minimizing -sum(r_i)
    c_obj = -np.ones(n)
    
    # Inequality constraints matrix: A_ub @ r <= b_ub
    # We collect all constraints here
    A_ub_rows = []
    b_ub_vals = []
    
    # 1. Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        x, y = centers[i]
        # r_i <= x
        row = np.zeros(n)
        row[i] = 1
        A_ub_rows.append(row)
        b_ub_vals.append(x)
        
        # r_i <= 1 - x
        row = np.zeros(n)
        row[i] = 1
        A_ub_rows.append(row)
        b_ub_vals.append(1 - x)
        
        # r_i <= y
        row = np.zeros(n)
        row[i] = 1
        A_ub_rows.append(row)
        b_ub_vals.append(y)
        
        # r_i <= 1 - y
        row = np.zeros(n)
        row[i] = 1
        A_ub_rows.append(row)
        b_ub_vals.append(1 - y)
        
    # 2. Non-overlap constraints: r_i + r_j <= dist(centers[i], centers[j])
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            row = np.zeros(n)
            row[i] = 1
            row[j] = 1
            A_ub_rows.append(row)
            b_ub_vals.append(dist)
            
    A_ub = np.array(A_ub_rows)
    b_ub = np.array(b_ub_vals)
    
    # Bounds for radii: r_i >= 0
    bounds = [(0, None) for _ in range(n)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            optimal_radii = res.x
            return -res.fun, optimal_radii # Return sum of radii
        else:
            # If LP fails, return 0 and small radii (fallback)
            return 0.0, np.zeros(n)
    except Exception:
        return 0.0, np.zeros(n)

def objective_function(centers_flat):
    """
    Objective function for the center optimizer.
    Minimizes negative sum of radii.
    """
    n = 26
    centers = centers_flat.reshape((n, 2))
    sum_radii, _ = evaluate_packing(centers)
    return -sum_radii

def generate_hex_grid(n):
    """Generates a hexagonal grid of n points in [0,1]x[0,1]"""
    centers = np.zeros((n, 2))
    idx = 0
    
    # Approximate spacing for 26 circles
    # 5 rows of roughly 5 circles
    row_height = 0.18
    col_width = 0.18
    
    y = 0.1
    shift = 0.0
    
    while idx < n:
        # Determine how many fit in this row
        x = 0.1
        row_centers = []
        while x <= 0.9 and idx < n:
            centers[idx] = [x + shift, y]
            row_centers.append(x + shift)
            idx += 1
            x += col_width
        
        y += row_height
        shift = col_width / 2 if shift == 0 else 0 # Alternate shift
        
    return centers

def run_packing():
    n = 26
    
    # Helper to run optimization
    def run_optimizer(init_centers, seed=None):
        if seed is not None:
            np.random.seed(seed)
            
        # Flatten centers for optimizer
        x0 = init_centers.flatten()

        bounds = [(0, 1) for _ in range(2 * n)]

        # Nelder-Mead is derivative-free and robust for non-smooth landscapes
        result = minimize(
            objective_function, 
            x0, 
            method='Nelder-Mead', 
            bounds=bounds,
            options={'maxiter': 2000, 'xatol': 1e-4, 'fatol': 1e-4}
        )
        
        if result.success:
            best_centers = result.x.reshape((n, 2))
            _, best_radii = evaluate_packing(best_centers)
            return best_centers, best_radii

            # Return initial if failed (unlikely)
            _, radii = evaluate_packing(init_centers)
            return init_centers, radii

    best_sum = -1.0

    print("Optimizing Hex Grid...")

            if final_radii[i] > max_r + 1e-9:

    return final_centers, final_radii, np.sum(final_radii)
