import numpy as np
from scipy.optimize import linprog

def solve_radii_lp(centers, n):
    """Solve LP for optimal radii given fixed centers."""
    c = -np.ones(n)
    A = []
    b = []
    
    for i in range(n):
        x, y = centers[i]
        row = np.zeros(n)
        row[i] = 1
        for bound in [x, 1 - x, y, 1 - y]:
            A.append(row.copy())
            b.append(bound)
    
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            row = np.zeros(n)
            row[i] = 1
            row[j] = 1
            A.append(row)
            b.append(dist)
    
    A = np.array(A)
    b = np.array(b)
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    
    if result.success:
        return result.x, -result.fun
    return np.zeros(n), 0.0

def validate_packing(centers, radii):
    """Validate the packing."""
    n = centers.shape[0]
    if np.isnan(centers).any() or np.isnan(radii).any():
        return False
    for i in range(n):
        if radii[i] < 0:
            return False
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            return False
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False
    return True

def generate_hex_configs(n, seed_base=0):
    """Generate hexagonal-like configurations with perturbations."""
    configs = []
    
    # Different row patterns that sum to 26
    patterns = [
        [5, 5, 5, 5, 6],
        [6, 5, 5, 5, 5],
        [4, 5, 6, 5, 6],
        [5, 6, 5, 6, 4],
        [6, 6, 6, 4, 4],
        [4, 4, 4, 4, 4, 6],
        [5, 5, 4, 5, 7],
        [6, 4, 6, 5, 5],
        [5, 7, 4, 5, 5],
        [4, 6, 4, 6, 6],
        [6, 5, 6, 4, 5],
        [5, 4, 6, 5, 6],
        [6, 6, 4, 5, 5],
        [5, 5, 5, 6, 5],
        [4, 5, 5, 6, 6],
    ]
    
    for pat in patterns:
        centers = []
        for row in range(len(pat)):
            nc = pat[row]
            y = 0.10 + row * 0.17
            offset = 0.085 if row % 2 == 1 else 0.0
            for col in range(nc):
                x = 0.09 + col * 0.17 + offset
                centers.append([x, y])
        
        if len(centers) == n:
            configs.append(np.array(centers))
    
    # Perturbed versions
    for seed in range(50):
        np.random.seed(seed_base + seed)
        for config in configs[:10]:  # Perturb first 10 patterns
            perturbed = config + np.random.randn(n, 2) * 0.03
            perturbed = np.clip(perturbed, 0.02, 0.98)
            configs.append(perturbed)
    
    return configs

def run_packing():
    n = 26
    np.random.seed(42)
    
    best_centers = None
    best_radii = None
    best_sum = 0.0
    
    # Try many random configurations
    for trial in range(800):
        np.random.seed(trial * 7 + 3)
        centers = np.random.rand(n, 2)
        radii, s = solve_radii_lp(centers, n)
        if s > best_sum:
            best_centers = centers.copy()
            best_radii = radii.copy()
            best_sum = s
    
    # Try structured hexagonal configurations
    configs = generate_hex_configs(n)
    for config in configs:
        radii, s = solve_radii_lp(config, n)
        if s > best_sum:
            best_centers = config.copy()
            best_radii = radii.copy()
            best_sum = s
    
    # Try concentrated patterns: place some circles near corners/edges
    for seed in range(100):
        np.random.seed(seed * 13 + 7)
        centers = np.zeros((n, 2))
        
        # 4 near corners
        for i in range(4):
            if i == 0:
                centers[i] = [0.1 + np.random.rand() * 0.05, 0.1 + np.random.rand() * 0.05]
            elif i == 1:
                centers[i] = [0.9 - np.random.rand() * 0.05, 0.1 + np.random.rand() * 0.05]
            elif i == 2:
                centers[i] = [0.1 + np.random.rand() * 0.05, 0.9 - np.random.rand() * 0.05]
            else:
                centers[i] = [0.9 - np.random.rand() * 0.05, 0.9 - np.random.rand() * 0.05]
        
        # 8 along edges
        for i in range(4, 12):
            side = i % 4
            t = np.random.rand()
            if side == 0:
                centers[i] = [t, 0.05 + np.random.rand() * 0.05]
            elif side == 1:
                centers[i] = [t, 0.95 - np.random.rand() * 0.05]
            elif side == 2:
                centers[i] = [0.05 + np.random.rand() * 0.05, t]
            else:
                centers[i] = [0.95 - np.random.rand() * 0.05, t]
        
        # Remaining in center
        for i in range(12, n):
            centers[i] = 0.3 + np.random.rand(2) * 0.4
        
        radii, s = solve_radii_lp(centers, n)
        if s > best_sum:
            best_centers = centers.copy()
            best_radii = radii.copy()
            best_sum = s
    
    # Fine-tune best configuration with local search
    if best_centers is not None:
        for iteration in range(200):
            np.random.seed(iteration * 17 + 11)
            improved = False
            
            # Try perturbing each circle individually
            for i in range(n):
                for _ in range(20):
                    trial_centers = best_centers.copy()
                    trial_centers[i] += np.random.randn(2) * 0.01
                    trial_centers[i] = np.clip(trial_centers[i], 0.01, 0.99)
                    
                    radii, s = solve_radii_lp(trial_centers, n)
                    if s > best_sum:
                        best_centers = trial_centers.copy()
                        best_radii = radii.copy()
                        best_sum = s
                        improved = True
            
            if not improved:
                # Try larger perturbations
                for _ in range(5):
                    trial_centers = best_centers + np.random.randn(n, 2) * 0.005
                    trial_centers = np.clip(trial_centers, 0.01, 0.99)
                    radii, s = solve_radii_lp(trial_centers, n)
                    if s > best_sum:
                        best_centers = trial_centers.copy()
                        best_radii = radii.copy()
                        best_sum = s
    
    # Ensure radii are positive
    if best_radii is not None:
        best_radii = np.maximum(best_radii, 1e-10)
    
    # Final validation
    if best_centers is not None:
        if not validate_packing(best_centers, best_radii):
            # Try to fix by slightly reducing radii
            for i in range(n):
                max_r = min(best_centers[i, 0], 1 - best_centers[i, 0],
                           best_centers[i, 1], 1 - best_centers[i, 1])
                best_radii[i] = min(best_radii[i], max_r)
    
    return best_centers, best_radii, best_sum
