import numpy as np
import math
from scipy.optimize import minimize

def run_packing():
    """
    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    """
    n_circles = 26
    best_centers = None
    best_radii = None
    best_sum_r = 0.0
    best_valid = False

    # Penalty weight for constraints
    penalty_weight = 1000.0

    def compute_objective_and_penalty(params, centers, radii):
        """
        Computes the objective (negative sum of radii) and penalty terms.
        """
        n = n_circles
        # Reshape params if needed, but here we assume params is (n, 3)
        # x, y, r
        x = params[:, 0]
        y = params[:, 1]
        r = params[:, 2]

        # Objective: minimize -sum(r) => maximize sum(r)
        obj = -np.sum(r)

        penalty = 0.0
        
        # Boundary penalties
        # r <= x <= 1-r  =>  x >= r  AND  x <= 1-r
        # If x < r, penalty (r - x)^2
        # If x > 1-r, penalty (x - (1-r))^2
        
        # Vectorized boundary checks
        # Left/Right boundaries
        # Condition: x - r >= 0  => violation if x - r < 0 => max(0, r - x)
        left_viol = np.maximum(0, r - x)
        right_viol = np.maximum(0, x - (1.0 - r))
        
        # Top/Bottom boundaries
        bottom_viol = np.maximum(0, r - y)
        top_viol = np.maximum(0, y - (1.0 - r))
        
        boundary_penalty = np.sum(left_viol**2) + np.sum(right_viol**2) + \
                           np.sum(bottom_viol**2) + np.sum(top_viol**2)
        
        # Overlap penalties
        # dist >= r_i + r_j
        # violation = (r_i + r_j) - dist
        # We sum max(0, violation)^2
        
        # Compute pairwise distances efficiently
        # Centers shape (n, 2)
        centers = np.column_stack((x, y))
        
        # Difference matrix
        diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :] # (n, n, 2)
        dist_sq = np.sum(diff**2, axis=2) # (n, n)
        dist = np.sqrt(dist_sq)
        
        # Radii sum matrix
        r_sum = r[:, np.newaxis] + r[np.newaxis, :]
        
        # Violation matrix
        # Only consider i < j to avoid double counting and self
        violations = np.maximum(0, r_sum - dist)
        
        # Sum of squared violations for upper triangle
        # Use triu to get upper triangle
        triu_mask = np.triu(np.ones((n, n), dtype=bool), k=1)
        overlap_penalty = np.sum(violations[triu_mask]**2)
        
        total_penalty = penalty_weight * (boundary_penalty + overlap_penalty)
        
        return obj + total_penalty, overlap_penalty, boundary_penalty

    def get_valid_solution(params):
        """
        Checks if params yields a valid packing and returns centers, radii, sum_r.
        """
        centers = params[:, :2]
        radii = params[:, 2]
        
        # Check for NaN or negative radii
        if np.any(np.isnan(centers)) or np.any(np.isnan(radii)):
            return None, None, 0.0
        if np.any(radii < -1e-12):
            return None, None, 0.0
            
        # Check boundaries strictly (with tolerance)
        for i in range(n_circles):
            xc, yc = centers[i]
            r = radii[i]
            if xc - r < -1e-9 or xc + r > 1 + 1e-9 or yc - r < -1e-9 or yc + r > 1 + 1e-9:
                return None, None, 0.0
                
        # Check overlaps
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                dist = np.sqrt((centers[i,0] - centers[j,0])**2 + (centers[i,1] - centers[j,1])**2)
                if dist < radii[i] + radii[j] - 1e-9:
                    return None, None, 0.0
        
        return centers, radii, np.sum(radii)

    # Try multiple initial configurations
    # 1. Grid initialization
    # 2. Random initialization
    
    initial_guesses = []
    
    # Grid 5x5 plus one extra
    # 25 circles in grid, 1 in a gap or random
    grid_centers = []
    for i in range(5):
        for j in range(5):
            grid_centers.append([0.1 + 0.2 * i, 0.1 + 0.2 * j])
    # Add one more circle, maybe at center of a gap or random
    # Gap at (0.2, 0.2) relative to grid lines? 
    # Centers are at 0.1, 0.3... Gaps at 0.2.
    # But 0.2 is occupied if we consider full grid? No, 0.2 is between 0.1 and 0.3.
    # Let's add at (0.2, 0.2)
    grid_centers.append([0.2, 0.2])
    # Trim to 26
    grid_centers = grid_centers[:n_circles]
    
    # Convert to array
    init_grid = np.zeros((n_circles, 3))
    init_grid[:, :2] = grid_centers
    init_grid[:, 2] = 0.05 # Small initial radius
    
    initial_guesses.append(init_grid)
    
    # Random initialization
    for _ in range(5):
        rand_centers = np.random.uniform(0.2, 0.8, size=(n_circles, 2))
        rand_params = np.zeros((n_circles, 3))
        rand_params[:, :2] = rand_centers
        rand_params[:, 2] = 0.05
        initial_guesses.append(rand_params)

    # Optimization loop
    for guess in initial_guesses:
        # Flatten for scipy
        x0 = guess.flatten()
        
        # Bounds: x,y in [0,1], r in [0, 0.5]
        bounds = []
        for _ in range(n_circles):
            bounds.append((0.0, 1.0)) # x
            bounds.append((0.0, 1.0)) # y
            bounds.append((0.0, 0.5)) # r
        
        # Run optimization
        # L-BFGS-B is good for bounded problems
        try:
            res = minimize(compute_objective_and_penalty, x0, args=(None, None), 
                           method='L-BFGS-B', bounds=bounds, 
                           options={'maxiter': 2000, 'ftol': 1e-9, 'gtol': 1e-6})
            
            if res.success or res.nit > 50:
                # Reshape result
                sol_params = res.x.reshape((n_circles, 3))
                
                # Validate and check sum
                # Note: The optimizer might have slightly violated constraints if penalty wasn't high enough
                # But with high penalty weight, it should be close.
                # We enforce validity strictly in get_valid_solution.
                
                centers, radii, sum_r = get_valid_solution(sol_params)
                
                if centers is not None and sum_r > best_sum_r:
                    best_sum_r = sum_r
                    best_centers = centers
                    best_radii = radii
                    best_valid = True
        except Exception as e:
            print(f"Optimization failed: {e}")
            continue

    # Fallback if optimization failed or didn't find valid solution
    if not best_valid:
        # Simple valid fallback: 26 small circles
        centers = np.random.uniform(0.2, 0.8, size=(n_circles, 2))
        radii = np.full(n_circles, 0.05)
        best_centers = centers
        best_radii = radii
        best_sum_r = np.sum(radii)
        # Make sure they don't overlap (unlikely with r=0.05 and random in 0.2-0.8)
        # But let's just return something valid if possible.
        # Actually, the random init with r=0.05 is likely valid.
        
    return best_centers, best_radii, best_sum_r
