import numpy as np
from scipy.optimize import minimize
import math

# Global constants
N_CIRCLES = 26

def get_min_radius(centers):
    """
    Calculates the maximum possible equal radius for the given centers.
    Returns the radius r such that all circles of radius r fit inside [0,1]x[0,1]
    and do not overlap.
    """
    n = centers.shape[0]
    min_r = float('inf')
    
    # Boundary constraints
    # A circle of radius r at (x,y) is inside if r <= x, r <= 1-x, r <= y, r <= 1-y
    for i in range(n):
        x, y = centers[i]
        min_r = min(min_r, x, 1-x, y, 1-y)
        
    # Inter-circle constraints
    # Two circles of radius r at i, j don't overlap if dist >= 2r => r <= dist/2
    for i in range(n):
        for j in range(i+1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            min_r = min(min_r, dist / 2.0)
    return min_r

def objective_func(centers_flat):
    """
    Objective to minimize: negative of min radius.
    Minimizing -r is equivalent to maximizing r.
    """
    centers = centers_flat.reshape(-1, 2)
    return -get_min_radius(centers)

def constraint_func(centers_flat):
    """
    Constraints: centers must be in [0, 1].
    Returns array of values that must be >= 0.
    """
    centers = centers_flat.reshape(-1, 2)
    vals = []
    for i in range(N_CIRCLES):
        vals.append(centers[i, 0])      # x >= 0
        vals.append(1 - centers[i, 0])  # x <= 1
        vals.append(centers[i, 1])      # y >= 0
        vals.append(1 - centers[i, 1])  # y <= 1
    return np.array(vals)

                cy += np.random.normal(0, 0.01)

                centers[idx] = [cx, cy]
                idx += 1

                forces[j] -= f_vec

            if x > 0.95: forces[i, 0] -= wall_k * (x - 0.95)
            if y < 0.05: forces[i, 1] += wall_k * (0.05 - y)
            if y > 0.95: forces[i, 1] -= wall_k * (y - 0.95)

            if idx < N_CIRCLES:
                # Center of cell
                cx = x_step * (c + 0.5)
                cy = y_step * (r + 0.5)
                # Add small random perturbation to break symmetry
                cx += np.random.normal(0, 0.005)
                cy += np.random.normal(0, 0.005)
                # Clip to [0,1]
                cx = np.clip(cx, 0.01, 0.99)
                cy = np.clip(cy, 0.01, 0.99)
                centers[idx] = [cx, cy]
                idx += 1

    # 2. Optimization
    bounds = [(0, 1)] * (N_CIRCLES * 2)
    cons = {'type': 'ineq', 'fun': constraint_func}
    x0 = centers.flatten()

    # Run optimizer to maximize min radius (equal radii strategy)
    res = minimize(objective_func, x0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': 5000, 'ftol': 1e-10})

    best_centers = res.x.reshape(-1, 2)
    best_r = -res.fun

    # 3. Force-directed refinement
    # This helps to escape local minima where circles are locked in a suboptimal grid-like structure.
    # We simulate repulsive forces to push circles apart, effectively increasing the minimum distance.
    current_centers = best_centers.copy()
    
    # Simulation parameters
    dt = 0.002
    damping = 0.95
    repulsion_k = 1.0
    wall_k = 20.0 # Stronger wall repulsion to keep them inside
    steps = 2000
    
    velocities = np.zeros_like(current_centers)
    
    for _ in range(steps):
        forces = np.zeros_like(current_centers)
        
        # Inter-circle repulsion (inverse square law)
        for i in range(N_CIRCLES):
            for j in range(i + 1, N_CIRCLES):
                diff = current_centers[i] - current_centers[j]
                dist = np.sqrt(np.sum(diff**2))
                if dist < 1e-6: dist = 1e-6
                
                # Force magnitude
                f_mag = repulsion_k / (dist * dist)
                f_vec = (diff / dist) * f_mag
                
                forces[i] += f_vec
                forces[j] -= f_vec
        
        # Wall repulsion
        for i in range(N_CIRCLES):
            x, y = current_centers[i]
            # Push back if too close to walls
            if x < 0.05: forces[i, 0] += wall_k * (0.05 - x)
            if x > 0.95: forces[i, 0] -= wall_k * (x - 0.95)
            if y < 0.05: forces[i, 1] += wall_k * (0.05 - y)
            if y > 0.95: forces[i, 1] -= wall_k * (y - 0.95)
            
        # Update
        velocities = velocities * damping + forces * dt
        current_centers += velocities * dt
        
        # Hard constraints to keep inside [0,1]
        current_centers = np.clip(current_centers, 0.001, 0.999)

    # 4. Final Polish with Optimizer
    # The force simulation might have moved circles to a configuration with larger min distance
    # Run optimizer again to fine-tune positions
    res2 = minimize(objective_func, current_centers.flatten(), method='SLSQP', bounds=bounds, constraints=cons,
                    options={'maxiter': 5000, 'ftol': 1e-10})
    
    final_centers = res2.x.reshape(-1, 2)
    final_r = -res2.fun
    
    # Fallback if optimization failed (rare)
    if final_r < 0.05:
        final_r = 0.05
        idx = 0
        for r_idx in range(6):
            for c_idx in range(5):
                if idx < N_CIRCLES:
                    final_centers[idx] = [0.1 + 0.15*c_idx, 0.1 + 0.15*r_idx]
                    idx += 1

    # Construct radii array (equal radii)
    radii = np.full(N_CIRCLES, final_r)
    sum_radii = np.sum(radii)
    
    return final_centers, radii, sum_radii
