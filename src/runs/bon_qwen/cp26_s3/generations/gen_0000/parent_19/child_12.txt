import numpy as np
from scipy.optimize import minimize

def evaluate_constraints(vars, n):
    """
    Evaluates inequality constraints for the circle packing problem.
    Returns an array where all elements must be >= 0.
    """
    centers = vars[:2*n].reshape(n, 2)
    radii = vars[2*n:]
    
    con = []
    # Boundary constraints: circles must stay inside [0,1]x[0,1]
    con.append(centers[:, 0] - radii)
    con.append(1.0 - centers[:, 0] - radii)
    con.append(centers[:, 1] - radii)
    con.append(1.0 - centers[:, 1] - radii)
    
    # Pairwise non-overlap constraints using squared distances
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dist_sq = dx**2 + dy**2
    r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    pair_con = dist_sq - r_sum**2
    
    # Extract upper triangular part to avoid duplicates and self-comparison
    idx = np.triu_indices(n, k=1)
    con.append(pair_con[idx])
    
    return np.concatenate(con)

def neg_sum_radii(vars, n):
    """Objective function: negative sum of radii (to be minimized)"""
    return -np.sum(vars[2*n:])

def run_packing():
    n = 26
    dim = 3 * n
    best_vars = None
    best_obj = -np.inf
    
    # Variable bounds: centers in [0,1], radii in [0, 0.5]
    bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n
    
    initial_configs = []
    
    # Configuration 1: Structured grid layout
    grid_centers = []
    for i in range(5):
        for j in range(5):
            grid_centers.append([0.12 + 0.17*i, 0.12 + 0.17*j])
    while len(grid_centers) < n:
        grid_centers.append([0.5, 0.5])
    grid_centers = np.array(grid_centers[:n])
    initial_configs.append(np.concatenate([grid_centers.flatten(), np.full(n, 0.06)]))
    
    # Configurations 2-6: Diverse random feasible starting points
    for seed in range(100, 105):
        np.random.seed(seed)
        rc = np.random.rand(n, 2) * 0.8 + 0.1
        initial_configs.append(np.concatenate([rc.flatten(), np.full(n, 0.05)]))
        
    constraint_dict = {'type': 'ineq', 'fun': evaluate_constraints, 'args': (n,)}
    
    # Phase 1: Multiple restarts from different initial configurations
    for v_init in initial_configs:
        res = minimize(neg_sum_radii, v_init, args=(n,), method='SLSQP', bounds=bounds,
                       constraints=constraint_dict, options={'maxiter': 3000, 'ftol': 1e-12, 'disp': False})
        if -res.fun > best_obj:
            best_obj = -res.fun
            best_vars = res.x
            
    # Phase 2: Perturbation refinement around the best solution
    if best_vars is not None:
        for i in range(10):
            pert = best_vars + np.random.randn(dim) * 1e-3
            # Enforce bounds after perturbation
            pert[:2*n] = np.clip(pert[:2*n], 0.0, 1.0)
            pert[2*n:] = np.clip(pert[2*n:], 0.0, 0.5)
            
            res = minimize(neg_sum_radii, pert, args=(n,), method='SLSQP', bounds=bounds,
                           constraints=constraint_dict, options={'maxiter': 2000, 'ftol': 1e-12, 'disp': False})
            if -res.fun > best_obj:
                best_obj = -res.fun
                best_vars = res.x
                
    centers = best_vars[:2*n].reshape(n, 2)
    radii = best_vars[2*n:]
    return centers, radii, best_obj
