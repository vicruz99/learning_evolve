import numpy as np

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    Uses a force-directed simulation with iterative radius expansion.
    """
    n_circles = 26
    np.random.seed(42) # For reproducibility

    # 1. Initialization: Hexagonal Grid Pattern
    # We try to fit circles in a hexagonal arrangement.
    # Estimate radius to start with. For 26 circles, r ~ 0.1 is a good guess.
    # We start smaller and let them grow.
    
    centers = np.zeros((n_circles, 2))
    radii = np.zeros(n_circles)
    
    # Create a hexagonal grid of points
    # We will generate enough points and select 26, or place them directly.
    # A 6x6 grid is 36 points, we can subset.
    
    # Let's try to place them in a pattern that covers the square well.
    # 5 rows of 5 is 25. We need 1 more.
    # Let's try a layout with 6 rows, varying number of circles.
    # Target: dense packing.
    
    # Initial radius guess
    r_init = 0.05
    
    idx = 0
    # Try to generate points in a hexagonal lattice
    # Spacing 2*r_init
    dx = 2 * r_init * 1.1 # slightly less dense to allow growth
    dy = dx * np.sqrt(3) / 2
    
    y = r_init
    row = 0
    while idx < n_circles:
        # Number of circles in this row?
        # Adjust x offset for hexagonal shift
        x_start = r_init
        if row % 2 == 1:
            x_start += r_init # shift by r_init (which is dx/2 approx if dx=2r)
            # Actually shift by dx/2
            x_start = r_init + dx/2
            
        x = x_start
        while x <= 1 - r_init:
            if idx < n_circles:
                centers[idx, 0] = x
                centers[idx, 1] = y
                radii[idx] = r_init
                idx += 1
            x += dx
        y += dy
        row += 1

    # If we didn't get enough (unlikely with this logic), fill with random
    if idx < n_circles:
        for i in range(idx, n_circles):
            centers[i] = np.random.uniform(0.2, 0.8, 2)
            radii[i] = r_init

    # 2. Simulation / Optimization Loop
    # We will try to expand radii and resolve conflicts.
    
    # Parameters
    max_iter = 3000
    growth_rate = 1.0005 # Expand radii by this factor each step
    resolution_passes = 50 # How many times to resolve overlaps per growth step
    
    # To allow different radii, we could grow them differently, 
    # but uniform growth is a good heuristic for maximizing sum.
    # However, to maximize sum, maybe growing smaller circles faster helps?
    # Let's stick to uniform or slightly biased towards smaller ones.
    
    for step in range(max_iter):
        # Grow radii
        # We multiply radii by a factor > 1. 
        # To prevent explosion, we can cap or just let resolution handle it.
        radii *= growth_rate
        
        # Resolve overlaps and boundary constraints
        for _ in range(resolution_passes):
            # Boundary constraints
            for i in range(n_circles):
                r = radii[i]
                cx, cy = centers[i]
                
                # Clamp to [r, 1-r]
                if cx < r:
                    centers[i, 0] = r
                elif cx > 1 - r:
                    centers[i, 0] = 1 - r
                
                if cy < r:
                    centers[i, 1] = r
                elif cy > 1 - r:
                    centers[i, 1] = 1 - r
            
            # Overlap resolution
            # Check all pairs
            # To speed up, we could use a grid, but n=26 is small.
            for i in range(n_circles):
                for j in range(i + 1, n_circles):
                    dist_vec = centers[i] - centers[j]
                    dist = np.linalg.norm(dist_vec)
                    min_dist = radii[i] + radii[j]
                    
                    if dist < min_dist:
                        # Overlap detected
                        # Move circles apart
                        if dist == 0:
                            # Avoid division by zero, push in random direction
                            dir_vec = np.random.uniform(-1, 1, 2)
                            dir_vec /= np.linalg.norm(dir_vec)
                        else:
                            dir_vec = dist_vec / dist
                        
                        # Amount to move apart
                        overlap = min_dist - dist
                        # Move each by half the overlap
                        move_vec = dir_vec * (overlap / 2)
                        
                        centers[i] += move_vec
                        centers[j] -= move_vec

    # Final check and clamping
    for i in range(n_circles):
        r = radii[i]
        centers[i, 0] = np.clip(centers[i, 0], r, 1 - r)
        centers[i, 1] = np.clip(centers[i, 1], r, 1 - r)

    # Calculate sum of radii
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
