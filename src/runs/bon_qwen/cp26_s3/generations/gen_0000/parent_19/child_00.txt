import numpy as np
from scipy.optimize import minimize


N = 26


def compute_overlap_violations(centers, radii):
    """Compute sum of squared overlap violations."""
    n = len(radii)
    violation = 0.0
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[j, 0] - centers[i, 0]
            dy = centers[j, 1] - centers[i, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            overlap = radii[i] + radii[j] - dist
            if overlap > 0:
                violation += overlap * overlap
    return violation


def resolve_overlaps_iterative(centers, radii, n, num_iters=200):
    """Push overlapping circles apart iteratively."""
    for iteration in range(num_iters):
        moved = False
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[j, 0] - centers[i, 0]
                dy = centers[j, 1] - centers[i, 1]
                dist_sq = dx * dx + dy * dy
                dist = np.sqrt(dist_sq) if dist_sq > 1e-20 else 1e-10
                
                min_dist = radii[i] + radii[j]
                if dist < min_dist:
                    overlap = min_dist - dist
                    scale = 0.5
                    inv_dist = 1.0 / dist
                    centers[i, 0] -= dx * inv_dist * overlap * scale
                    centers[i, 1] -= dy * inv_dist * overlap * scale
                    centers[j, 0] += dx * inv_dist * overlap * scale
                    centers[j, 1] += dy * inv_dist * overlap * scale
                    moved = True
        
        # Enforce boundary constraints
        for i in range(n):
            centers[i, 0] = max(radii[i], min(1.0 - radii[i], centers[i, 0]))
            centers[i, 1] = max(radii[i], min(1.0 - radii[i], centers[i, 1]))
        
        if not moved:
            break
    return centers


def objective_slsqp(x):
    """Objective: maximize sum of radii (minimize negative)."""
    return -np.sum(x[2 * N:])


def make_bound_constraint_x_min(i):
    """Factory for x >= r constraint for circle i."""
    def constraint(x):
        return x[2 * i] - x[2 * N + i]
    return constraint


def make_bound_constraint_x_max(i):
    """Factory for x + r <= 1 constraint for circle i."""
    def constraint(x):
        return 1.0 - x[2 * i] - x[2 * N + i]
    return constraint


def make_bound_constraint_y_min(i):
    """Factory for y >= r constraint for circle i."""
    def constraint(x):
        return x[2 * i + 1] - x[2 * N + i]
    return constraint


def make_bound_constraint_y_max(i):
    """Factory for y + r <= 1 constraint for circle i."""
    def constraint(x):
        return 1.0 - x[2 * i + 1] - x[2 * N + i]
    return constraint


def make_overlap_constraint(i, j):
    """Factory for non-overlap constraint between circles i and j."""
    def constraint(x):
        ci_x = x[2 * i]
        ci_y = x[2 * i + 1]
        cj_x = x[2 * j]
        cj_y = x[2 * j + 1]
        ri = x[2 * N + i]
        rj = x[2 * N + j]
        dx = cj_x - ci_x
        dy = cj_y - ci_y
        dist = np.sqrt(dx * dx + dy * dy)
        return dist - ri - rj
    return constraint


def hexagonal_initial_placement(r_init):
    """Create initial hexagonal lattice placement of N circles."""
    n = N
    centers = np.zeros((n, 2))
    radii = np.full(n, r_init)
    
    idx = 0
    row = 0
    max_rows = 8
    while idx < n and row < max_rows:
        col = 0
        max_cols = 8
        while col < max_cols and idx < n:
            x = r_init + col * 2 * r_init + (r_init if row % 2 == 1 else 0)
            y = r_init + row * r_init * np.sqrt(3)
            if x + r_init <= 1.0 and y + r_init <= 1.0:
                centers[idx] = [x, y]
                idx += 1
            col += 1
        row += 1
    
    return centers[:n], radii[:n]


def run_packing():

    # Phase 1: Create initial hexagonal placement

            radii[i] = min(radii[i], centers[i, 1])

            best_radii = radii.copy()

    radii = best_radii

    x0 = np.concatenate([centers.flatten(), radii])
    
    constraints_list = []

        constraints_list.append({'type': 'ineq', 'fun': make_bound_constraint_x_min(i)})
        constraints_list.append({'type': 'ineq', 'fun': make_bound_constraint_x_max(i)})
        constraints_list.append({'type': 'ineq', 'fun': make_bound_constraint_y_min(i)})

                'ftol': 1e-14,

            }

                        continue

                        continue
                    
                    # Check overlaps

    for i in range(n):
        # Find maximum possible radius for circle i given current positions
        max_r = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])

            dist = np.sqrt(dx * dx + dy * dy)
            max_r = min(max_r, dist - radii[j])
        max_r = max(0, max_r)
        radii[i] = max_r
    
    # Final cleanup to ensure validity
    for i in range(n):
        centers[i, 0] = np.clip(centers[i, 0], radii[i], 1.0 - radii[i])
