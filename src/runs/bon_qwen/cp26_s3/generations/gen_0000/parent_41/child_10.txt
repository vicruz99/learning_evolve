def compute_overlap_penalty(centers, radii):
    """Compute total squared overlap for all pairs and boundaries."""
    n = len(radii)
    penalty = 0.0
    for i in range(n):
        # Boundary constraints
        x, y = centers[i]
        r = radii[i]
        penalty += max(0, r - x)**2
        penalty += max(0, r - (1 - x))**2
        penalty += max(0, r - y)**2
        penalty += max(0, r - (1 - y))**2

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            overlap = radii[i] + radii[j] - dist
            if overlap > 0:
                penalty += overlap**2
    return penalty

            })
