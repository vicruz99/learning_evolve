import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    """
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    # 1. Initialization: Hexagonal Lattice
    # Distribute 26 circles into 6 rows (5, 4, 5, 4, 5, 3) to maximize density
    row_counts = [5, 4, 5, 4, 5, 3]
    row_idx = 0
    for i in range(n):
        r = row_counts[row_idx]
        col_idx = (i - sum(row_counts[:row_idx])) # 0-based index in row
        y = 0.5 + row_idx * (np.sqrt(3)/2) # Staggered y
        # Shift x by 0.5 if row index is odd to create hexagonal pattern
        x_offset = 0.5 if (row_idx % 2 != 0) else 0
        # Normalize x to center the row
        x = (col_idx + 0.5 + x_offset) / r
        
        centers[i] = [x, y]
        row_idx += (1 if col_idx == r - 1 else 0)

    # Initial small radius to ensure starting validity
    radii[:] = 0.01
    
    # 2. Optimization: Iterative "Grow and Relax"
    # We try to increase radii, then optimize positions to remove overlaps.
    for step in range(100):
        # Grow radii
        growth_factor = 1.0 + (0.02 / (step + 1.0))
        radii *= growth_factor
        
        # Flatten centers for optimizer (x1, y1, x2, y2...)
        x0 = centers.flatten()
        
        # Define bounds [r, 1-r] for each coordinate
        bounds = []
        for _ in range(n):
            bounds.append((radii[0], 1 - radii[0])) # x
            bounds.append((radii[0], 1 - radii[0])) # y
            
        # Optimization Objective: Minimize Overlap Penalty
        # If dist < r1 + r2, penalty is quadratic.
        def objective(x):
            curr_centers = x.reshape(-1, 2)
            penalty = 0.0
            
            # Overlap penalty (smoothed)
            for i in range(n):
                r_i = radii[i]
                x_i, y_i = curr_centers[i]
                
                # Boundary penalty
                if x_i < r_i: penalty += 1000 * (r_i - x_i)**2
                if x_i > 1 - r_i: penalty += 1000 * (x_i - (1 - r_i))**2
                if y_i < r_i: penalty += 1000 * (r_i - y_i)**2
                if y_i > 1 - r_i: penalty += 1000 * (y_i - (1 - r_i))**2
                
                # Inter-circle overlap penalty
                for j in range(i + 1, n):
                    r_j = radii[j]
                    dx = curr_centers[i, 0] - curr_centers[j, 0]
                    dy = curr_centers[i, 1] - curr_centers[j, 1]
                    dist = np.sqrt(dx*dx + dy*dy)
                    min_dist = r_i + r_j
                    if dist < min_dist:
                        overlap = min_dist - dist
                        penalty += 1000 * overlap**2
            return penalty

        result = minimize(objective, x0, method='L-BFGS-B', bounds=bounds, 
                         options={'maxiter': 100, 'ftol': 1e-9})
        
        if result.fun < 1e-6:
            centers = result.x.reshape(-1, 2)
        else:
            # If optimizer fails to resolve overlap, revert radii growth
            # This indicates we hit the density limit
            radii /= growth_factor
            break

    # Calculate final sum
    sum_radii = np.sum(radii)
    
    # Final verification and cleaning
    # Ensure centers are valid (NaN check)
    if np.isnan(centers).any() or np.isnan(radii).any():
        # Fallback to a known valid grid if optimization fails
        centers = np.zeros((n, 2))
        radii = np.zeros(n)
        k = 0
        for i in range(5):
            for j in range(5):
                if k < 25:
                    centers[k] = [0.1 + i * 0.2, 0.1 + j * 0.2]
                    radii[k] = 0.1
                    k += 1
        # Place 26th circle in a small gap (if possible) or reduce size
        radii[25] = 0.01
        centers[25] = [0.5, 0.5] # Placeholder
        sum_radii = np.sum(radii)

    return centers, radii, sum_radii
