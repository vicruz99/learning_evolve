import numpy as np
from scipy.optimize import minimize

def compute_objective(x, n_circles, penalty_weight):
    """
    Computes the objective function: negative sum of radii + penalty for violations.
    """
    # Reshape flat array to (N, 3) -> [x, y, r]
    p = x.reshape(n_circles, 3)
    centers = p[:, :2]
    radii = p[:, 2]

    # 1. Overlap Penalty
    # Compute pairwise distances squared
    # diff shape: (N, N, 2)
    diff = centers[:, None, :] - centers[None, :, :]
    dist_sq = np.sum(diff**2, axis=2)
    # Avoid division by zero or sqrt issues, though dist_sq >= 0
    dist = np.sqrt(dist_sq + 1e-12)
    
    # Sum of radii matrix
    sum_radii = radii[:, None] + radii[None, :]
    
    # Overlap amount: positive means overlap
    overlap = sum_radii - dist
    
    # We only care about i < j. Use a mask.
    mask = np.triu(np.ones((n_circles, n_circles), dtype=bool), k=1)
    
    # Squared penalty for overlaps
    penalty_overlap = np.sum(np.maximum(0, overlap[mask]) ** 2)

    # 2. Boundary Penalty
    # Constraints: r <= x <= 1-r  =>  x - r >= 0  AND  1 - x - r >= 0
    # Violation: max(0, r - x) and max(0, x + r - 1)
    
    viol_left = np.maximum(0, radii - centers[:, 0])
    viol_right = np.maximum(0, centers[:, 0] + radii - 1)
    viol_bottom = np.maximum(0, radii - centers[:, 1])
    viol_top = np.maximum(0, centers[:, 1] + radii - 1)
    
    penalty_bound = np.sum(viol_left**2 + viol_right**2 + viol_bottom**2 + viol_top**2)

    total_penalty = penalty_overlap + penalty_bound
    
    # Objective: minimize (-sum_radii + penalty_weight * total_penalty)
    # We want to maximize sum_radii, so we minimize negative sum.
    obj = -np.sum(radii) + penalty_weight * total_penalty
    
    return obj

def get_hexagonal_init(n_circles):
    """
    Generates a valid initial packing using a hexagonal grid pattern.
    """
    # Estimate grid size to fit n_circles
    # Approximate area per circle in hex packing: sqrt(3)/2 * (2r)^2 = 2*sqrt(3)*r^2
    # But for initialization, just place them in a rectangle.
    # We want a starting radius around 0.05-0.08 to be safe and dense.
    
    # Try to fit in a 6x5 grid (30 slots)
    cols = 6
    rows = 5
    
    # Generate grid points with spacing
    # Width 1, 6 circles -> spacing approx 1/7? 
    # Let's use spacing 0.15 to allow room for expansion
    x_coords = np.linspace(0.15, 0.85, cols)
    y_coords = np.linspace(0.15, 0.85, rows)
    
    points = []
    for y in y_coords:
        for x in x_coords:
            points.append([x, y])
            if len(points) >= n_circles:
                break
        if len(points) >= n_circles:
            break
            
    # If not enough points (unlikely with 6x5=30), add random ones
    while len(points) < n_circles:
        points.append([np.random.rand(), np.random.rand()])
        
    points = np.array(points[:n_circles])
    
    # Initial radii: small but non-zero
    radii = np.full(n_circles, 0.05)
    
    # Combine to shape (N, 3)
    init_arr = np.zeros((n_circles, 3))
    init_arr[:, 0] = points[:, 0]
    init_arr[:, 1] = points[:, 1]
    init_arr[:, 2] = radii
    
    return init_arr.flatten()

def run_packing():
    n_circles = 26
    penalty_weight = 5000 # High weight to enforce constraints
    
    best_obj = np.inf
    best_x = None
    
    # Run multiple trials to avoid local minima
    n_trials = 20
    
    for trial in range(n_trials):
        # Generate initialization
        # Add small perturbation to avoid exact grid lock-in
        init_x = get_hexagonal_init(n_circles)
        # Perturb positions slightly
        perturbation = np.random.normal(0, 0.02, size=init_x.shape)
        # Perturb radii slightly
        perturbation[-1::3] *= 0 # Don't perturb radii too much initially or keep positive
        init_x += perturbation
        
        # Ensure initial radii are positive
        init_x[2::3] = np.maximum(init_x[2::3], 0.01)
        
        # Bounds: x in [0, 1], y in [0, 1], r in [0, 0.5]
        bounds = []
        for _ in range(n_circles):
            bounds.append((0.0, 1.0)) # x
            bounds.append((0.0, 1.0)) # y
            bounds.append((0.0, 0.5)) # r
            
        # Optimization
        res = minimize(
            compute_objective,
            init_x,
            args=(n_circles, penalty_weight),
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 1000, 'ftol': 1e-9}
        )
        
        if res.fun < best_obj:
            best_obj = res.fun
            best_x = res.x.copy()
            
    # Extract result
    if best_x is None:
        # Fallback if optimization failed
        best_x = get_hexagonal_init(n_circles)
        
    result_p = best_x.reshape(n_circles, 3)
    centers = result_p[:, :2]
    radii = result_p[:, 2]
    
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii

if __name__ == "__main__":
    centers, radii, sum_radii = run_packing()
    print(f"Sum of radii: {sum_radii}")
