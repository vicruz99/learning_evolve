import numpy as np
import scipy.optimize as opt
from scipy.spatial.distance import pdist, squareform

def solve_radii_lp(centers):
    """
    Solve the LP to find the maximum sum of radii for fixed centers.
    """
    n = centers.shape[0]
    
    # 1. Prepare Linear Programming Constraints: A_ub @ r <= b_ub
    # We want to Maximize sum(r) => Minimize -sum(r)
    c_obj = np.ones(n) * -1.0
    
    # Initialize constraint matrix and vector
    n_constraints = 0
    # Pairs: n*(n-1)/2
    n_pairs = n * (n - 1) // 2
    # Walls: 4 per circle
    n_walls = 4 * n
    n_constraints = n_pairs + n_walls
    
    A_ub = np.zeros((n_constraints, n))
    b_ub = np.zeros(n_constraints)
    
    row = 0
    
    # Distance constraints (r_i + r_j <= dist_ij)
    # Using pdist for efficiency
    dist_matrix = squareform(pdist(centers))
    
    # Iterate over upper triangle
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[row, i] = 1.0
            A_ub[row, j] = 1.0
            b_ub[row] = dist_matrix[i, j]
            row += 1
            
    # Wall constraints (r_i <= x, r_i <= 1-x, r_i <= y, r_i <= 1-y)
    for i in range(n):
        x, y = centers[i]
        
        # x - r >= 0 => r <= x
        A_ub[row, i] = 1.0
        b_ub[row] = x
        row += 1
        
        # x + r <= 1 => r <= 1 - x
        A_ub[row, i] = 1.0
        b_ub[row] = 1.0 - x
        row += 1
        
        # y - r >= 0 => r <= y
        A_ub[row, i] = 1.0
        b_ub[row] = y
        row += 1
        
        # y + r <= 1 => r <= 1 - y
        A_ub[row, i] = 1.0
        b_ub[row] = 1.0 - y
        row += 1
        
    # Bounds: r_i >= 0
    bounds = [(0, None) for _ in range(n)]
    
    # Solve
    try:
        res = opt.linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return -res.fun, -res.x # Return sum and radii (negated because we minimized)
        else:
            return 0.0, np.zeros(n)
    except Exception:
        return 0.0, np.zeros(n)

def objective_function(centers_flat):
    """
    Objective function for the optimizer. 
    Returns negative sum of radii (since we minimize).
    """
    centers = centers_flat.reshape((26, 2))
    sum_radii, radii = solve_radii_lp(centers)
    return -sum_radii

def run_packing():

    n_circles = 26

    def generate_hex_grid(r_guess):
        # Approximate radius for 26 circles
        # Area approx: 26 * pi * r^2 <= 0.9 (density) => r approx 0.1
        # Let's try to pack them loosely first
        centers = []
        y = r_guess
        while y < 1.0 - r_guess + 1e-9:
            x = r_guess
            offset = 0
            if len(centers) > 0 and (len(centers) // 5) % 2 == 1:
                offset = r_guess # Shift row
                x += r_guess
            while x < 1.0 - r_guess + 1e-9:
                centers.append([x, y])
                x += 2 * r_guess
            y += r_guess * np.sqrt(3)
        # Trim or pad to 26
        if len(centers) > n_circles:
            return np.array(centers[:n_circles])
        else:
            # Fill remaining with small random offsets
            while len(centers) < n_circles:
                centers.append([np.random.rand(), np.random.rand()])
            return np.array(centers)

    # We will run differential evolution multiple times with different seeds

    best_centers = None
    best_radii = None

    lb = 0.01
    ub = 0.99
    bounds_opt = [(lb, ub) for _ in range(n_circles * 2)]

    # Hex allows slightly tighter.

    hex_centers = np.clip(hex_centers, lb, ub)
    
    initial_guesses = [hex_centers.flatten()]
    
    # Strategy 2: Random starts near center
    for _ in range(5):
        centers = np.random.uniform(lb + 0.05, ub - 0.05, (n_circles, 2))
        initial_guesses.append(centers.flatten())

            best_centers = local_res.x.reshape((n_circles, 2))

    except:
        pass

    # Final validation and return
    # The radii returned by solve_radii_lp are the optimal ones for the best_centers
    # Ensure radii are non-negative (LP ensures this, but float precision)
    best_radii = np.maximum(best_radii, 0.0)
    
    # If the solution is invalid according to strict validation (numerical issues),
    # we might need to shrink radii slightly. But LP guarantees constraints.
    # The validate function allows 1e-12 tolerance.
    
    return best_centers, best_radii, best_score
