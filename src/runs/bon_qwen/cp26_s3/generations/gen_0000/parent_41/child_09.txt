

To maximize the sum of radii for 26 circles in a unit square, we employ a two-stage optimization strategy. First, we treat the sum of radii as an objective that can be computed efficiently via a Linear Program (LP) for any fixed configuration of centers. Then, we use a hill-climbing algorithm with random restarts to find the optimal centers.

### Strategy
1. **LP-based Radius Calculation**: For any fixed set of centers, the maximum possible radii are found by solving an LP that maximizes $\sum r_i$ subject to non-overlap ($r_i + r_j \le \|c_i - c_j\|$) and boundary ($r_i \le \text{dist}(c_i, \partial)$) constraints.
2. **Stochastic Center Optimization**: We initialize centers in a grid and use iterative perturbations. At each step, we calculate the optimal radii and move centers to increase distances between circles that are currently "touching" (tight constraints), effectively "relaxing" the packing to allow for larger radii.
3. **Constraint Enforcement**: We ensure centers remain within the valid range $[0, 1]$ and that radii are non-negative.

```python
import numpy as np
from scipy.optimize import linprog

def solve_radii(centers):
    """Solve the LP to find maximum sum of radii for fixed centers."""
    n = centers.shape[0]
    c_obj = -np.ones(n)  # Minimize negative sum of radii

    # Construct constraints A_ub * r <= b_ub
    m_pairs = n * (n - 1) // 2
    m_bound = n * 4
    A_ub = np.zeros((m_pairs + m_bound, n))
    b_ub = np.zeros(m_pairs + m_bound)
    
    idx = 0
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dist
            idx += 1
            
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        x, y = centers[i]
        A_ub[idx, i] = 1.0; b_ub[idx] = x; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - x; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = y; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - y; idx += 1

    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    return res.x if res.success else np.zeros(n)

def compute_forces(centers, radii, n):
    """Compute repulsive forces based on active constraints."""
    forces = np.zeros_like(centers)
    threshold = 1e-5
    
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            if radii[i] + radii[j] > dist - threshold:
                vec = centers[i] - centers[j]
                if dist > 1e-8:
                    force_mag = (radii[i] + radii[j] - dist) * 10.0
                    norm_vec = vec / dist
                    forces[i] += force_mag * norm_vec
                    forces[j] -= force_mag * norm_vec
    
    # Boundary repulsion
    for i in range(n):
        x, y = centers[i]
        if radii[i] > x - threshold: forces[i, 0] += (radii[i] - x) * 10.0
        if radii[i] > (1.0 - x) - threshold: forces[i, 0] -= (radii[i] - (1.0 - x)) * 10.0
        if radii[i] > y - threshold: forces[i, 1] += (radii[i] - y) * 10.0
        if radii[i] > (1.0 - y) - threshold: forces[i, 1] -= (radii[i] - (1.0 - y)) * 10.0
    
    return forces

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    best_sum = 0
    best_centers = None
    n = 26

    for restart in range(15):
        # Grid initialization with jitter
        centers = np.random.uniform(0.1, 0.9, (n, 2))
        step = 0.05
        decay = 0.95
        temperature = 1.0
        
        for iteration in range(150):
            radii = solve_radii(centers)
            current_sum = np.sum(radii)
            
            if current_sum > best_sum:
                best_sum = current_sum
                best_centers = centers.copy()

            # Force-directed update
            forces = compute_forces(centers, radii, n)
            centers += step * forces * temperature
            
            # Clip to valid area (leaving a small margin)
            centers = np.clip(centers, 0.01, 0.99)
            step *= decay
            
            if iteration > 100:
                temperature *= 0.9

    radii = solve_radii(best_centers)
    return best_centers, radii, np.sum(radii)
```
