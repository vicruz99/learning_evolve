def initialize_hexagonal(n=26):
    """Initialize centers in a hexagonal lattice pattern."""
    centers = np.zeros((n, 2))
    idx = 0
    row = 0
    col = 0

        x = col * spacing * np.sqrt(3)
        y = row * spacing * 1.5

            x += spacing * np.sqrt(3) / 2
        centers[idx] = [0.5 + x * 0.3, 0.5 + y * 0.3]
        idx += 1
        col += 1

            col = 0
            row += 1
    return centers

def solve_radii_lp(centers, n=26):
    """Given centers, find optimal radii via linear programming."""
    # Maximize sum(r_i) subject to:
    # r_i <= min(x_i, 1-x_i, y_i, 1-y_i) for all i (boundary)
    # r_i + r_j <= ||c_i - c_j|| for all i < j (non-overlap)
    # r_i >= 0
    
    c_obj = -np.ones(n)  # Minimize -sum(r_i) = maximize sum(r_i)
    
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= max_r_i
    for i in range(n):
        x, y = centers[i]
        max_r = min(x, 1 - x, y, 1 - y)
        max_r = max(max_r, 1e-10)  # Ensure positive
        row = np.zeros(n)
        row[i] = 1.0
        A_ub.append(row)
        b_ub.append(max_r)
    
    # Non-overlap constraints: r_i + r_j <= dist(i,j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dist)
    
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    # Ensure all b_ub are non-negative
    b_ub = np.maximum(b_ub, 1e-10)
    
    bounds = [(0, None) for _ in range(n)]
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success and result.fun > -1e10:
        radii = result.x
        # Ensure non-negative
        radii = np.maximum(radii, 0)
        return radii
    else:
        # Fallback: small uniform radii
        return np.full(n, 0.02)
