import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # --- 1. Initialization ---
    # Start with a hexagonal lattice pattern for better packing density potential.
    # Initial radius small enough to fit many circles, allowing optimizer to grow them.
    r_init = 0.06
    spacing = 2.1 * r_init  # Slight overlap allowed in init or just tight packing
    
    centers = []
    y = r_init
    row_idx = 0
    while y <= 1 - r_init:
        x = r_init
        # Shift every other row for hexagonal packing
        if row_idx % 2 == 1:
            x += spacing / 2
        
        while x <= 1 - r_init:
            centers.append([x, y])
            x += spacing
        
        y += spacing * np.sqrt(3) / 2
        row_idx += 1
        
    centers = np.array(centers)
    
    # If we generated more than 26, select the first 26 (which are well-distributed)
    if len(centers) > n:
        centers = centers[:n]
    elif len(centers) < n:
        # Fallback: fill with random valid positions if grid was too sparse
        while len(centers) < n:
            c = np.random.rand(2) * (1 - 2*r_init) + r_init
            centers = np.vstack([centers, c])
            
    radii = np.full(n, r_init)
    
    # Flatten to 1D array for optimizer: [x1, y1, r1, x2, y2, r2, ...]
    x0 = np.zeros(3 * n)
    for i in range(n):
        x0[3*i] = centers[i, 0]
        x0[3*i+1] = centers[i, 1]
        x0[3*i+2] = radii[i]
        
    # --- 2. Optimization Setup ---
    # Bounds: x, y in [0, 1], r >= 0 (max possible radius is 0.5)
    bounds = []
    for _ in range(n):
        bounds.extend([
            (0.0, 1.0), # x
            (0.0, 1.0), # y
            (0.0, 0.5)  # r
        ])
        
    # Constraints
    constraints = []
    
    # Boundary constraints:
    # x >= r, x + r <= 1, y >= r, y + r <= 1
    # Formulated as fun(x) >= 0
    for i in range(n):
        idx = 3*i
        xi, yi, ri = idx, idx+1, idx+2
        
        # x - r >= 0
        constraints.append({'type': 'ineq', 'fun': lambda v, idx=idx, ri=ri: v[idx] - v[ri]})
        # 1 - x - r >= 0
        constraints.append({'type': 'ineq', 'fun': lambda v, idx=idx, ri=ri: 1.0 - v[idx] - v[ri]})
        # y - r >= 0
        constraints.append({'type': 'ineq', 'fun': lambda v, idx=yi, ri=ri: v[idx] - v[ri]})
        # 1 - y - r >= 0
        constraints.append({'type': 'ineq', 'fun': lambda v, idx=yi, ri=ri: 1.0 - v[idx] - v[ri]})
        
    # Non-overlap constraints:
    # dist^2 >= (r_i + r_j)^2  =>  dist^2 - (r_i + r_j)^2 >= 0
    for i in range(n):
        for j in range(i + 1, n):
            idx_i = 3*i
            idx_j = 3*j
            
            xi, yi, ri = idx_i, idx_i+1, idx_i+2
            xj, yj, rj = idx_j, idx_j+1, idx_j+2
            
            def overlap_constraint(v, ri=ri, rj=rj, xi=xi, yi=yi, xj=xj, yj=yj):
                dx = v[xi] - v[xj]
                dy = v[yi] - v[yj]
                dr = v[ri] + v[rj]
                return (dx*dx + dy*dy) - (dr*dr)
            
            constraints.append({'type': 'ineq', 'fun': overlap_constraint})
            
    # Objective function: Maximize sum of radii
    def objective(v):
        radii_vals = v[2::3]
        return -np.sum(radii_vals)
        
    # --- 3. Run Optimization ---
    # Use SLSQP which supports bounds and inequality constraints
    try:
        res = minimize(
            objective, 
            x0, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=constraints, 
            options={'maxiter': 2000, 'ftol': 1e-12, 'disp': False}
        )
        
        if res.success:
            final_centers = np.zeros((n, 2))
            final_radii = np.zeros(n)
            sum_r = 0.0
            
            for i in range(n):
                final_centers[i, 0] = res.x[3*i]
                final_centers[i, 1] = res.x[3*i+1]
                final_radii[i] = res.x[3*i+2]
                sum_r += final_radii[i]
            
            return final_centers, final_radii, sum_r
        else:
            # If optimization fails, return the initial valid configuration
            # (though it likely won't fail with this setup)
            print("Optimization did not converge, returning initial config.")
            return centers, radii, np.sum(radii)
            
    except Exception as e:
        print(f"Optimization error: {e}")
        return centers, radii, np.sum(radii)

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    import numpy as np
    n = centers.shape[0]

    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True
