from scipy.optimize import minimize

    radii_init = []

        else:

            centers_init.append([x, y])
            radii_init.append(r_init)
            x += 2 * r_init
        
        y += row_spacing
        row_idx += 1

        radii = np.array(temp_radii)

        return -r_sum # Minimize negative sum => Maximize sum

        cons = {'type': 'ineq', 'fun': lambda v, i=i: v[3*i] - v[3*i+2]}

        cons = {'type': 'ineq', 'fun': lambda v, i=i: 1 - v[3*i] - v[3*i+2]}

    total_sum = np.sum(opt_radii)
    
    # Return
    return opt_centers, opt_radii, total_sum
