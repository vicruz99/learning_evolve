import scipy.optimize as opt

def compute_pairwise_dist(v, i, j):
    c_i = v[2*i:2*i+2]
    c_j = v[2*j:2*j+2]
    return np.linalg.norm(c_i - c_j)

def constraint_pairwise(v, i, j):
    """Non-overlap constraint: distance >= r_i + r_j"""
    return compute_pairwise_dist(v, i, j) - (v[52+i] + v[52+j])

def constraint_boundary_lower(v, i, d):
    """Constraint: x >= r or y >= r"""
    return v[2*i+d] - v[52+i]

def constraint_boundary_upper(v, i, d):
    """Constraint: 1 - x >= r or 1 - y >= r"""
    return 1.0 - v[2*i+d] - v[52+i]

    return -np.sum(v[52:])

            centers[idx] = [x, y]
            idx += 1

    centers[idx] = [0.5, 0.5]

    growth_factor = 1.003

        radii *= growth_factor

                        f = overlap * diff / d
                        forces[i] += f
                        forces[j] -= f
