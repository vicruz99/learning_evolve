import numpy as np
import scipy.optimize as opt

# Global constant for number of circles
N = 26

def objective(v):
    """
    Objective function to minimize negative sum of radii.
    v is the variable vector of size 3*N: [x1, y1, r1, x2, y2, r2, ...]
    """
    radii = v[2*N:]
    return -np.sum(radii)

def inequality_constraints(v):
    """
    Inequality constraints function.
    Returns an array of values that must be >= 0.
    Constraints:
    1. x_i >= r_i
    2. y_i >= r_i
    3. x_i <= 1 - r_i  => 1 - x_i - r_i >= 0
    4. y_i <= 1 - r_i  => 1 - y_i - r_i >= 0
    5. (x_i - x_j)^2 + (y_i - y_j)^2 >= (r_i + r_j)^2
    """
    centers = v[:2*N].reshape((N, 2))
    radii = v[2*N:]
    
    # Boundary constraints
    c1 = centers[:, 0] - radii       # x >= r
    c2 = centers[:, 1] - radii       # y >= r
    c3 = 1 - centers[:, 0] - radii   # 1 - x >= r
    c4 = 1 - centers[:, 1] - radii   # 1 - y >= r
    
    # Overlap constraints
    # Vectorized distance calculation
    # centers shape (N, 2)
    # Add dimensions for broadcasting: (N, 1, 2) and (1, N, 2)
    c_diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist_sq = np.sum(c_diff**2, axis=2) # Shape (N, N)
    
    # Radii sum squared
    r_sum = radii[:, np.newaxis] + radii[np.newaxis, :] # Shape (N, N)
    r_sum_sq = r_sum**2
    
    # Constraint: dist^2 - (r_i + r_j)^2 >= 0
    diff_sq = dist_sq - r_sum_sq
    
    # Extract upper triangular part (i < j) to avoid duplicates and self-check
    idx = np.triu_indices(N, k=1)
    c5 = diff_sq[idx]
    
    # Concatenate all constraints
    return np.concatenate([c1, c2, c3, c4, c5])

def get_initial_guess(use_hex, r_init):

    Generate an initial guess vector.
    """
    pts = []
    if use_hex:
        # Hexagonal lattice initialization
        d = 2 * r_init

        max_c = int(1 / d) + 5

                x = r_init + c * d + (r % 2) * (d / 2)
                y = r_init + r * h

    else:

        for _ in range(N + 50):
            x = np.random.uniform(r_init, 1 - r_init)
            y = np.random.uniform(r_init, 1 - r_init)
            pts.append((x, y))
            
    # Take first N points
    pts = pts[:N]
    
    # If we don't have enough points (unlikely), fill with center
    while len(pts) < N:
        pts.append((0.5, 0.5))
        
    # Construct variable vector
    v = np.zeros(3 * N)
    for i, (x, y) in enumerate(pts):
        v[3*i] = x
        v[3*i+1] = y
        v[3*i+2] = r_init
    return v

def run_packing():

    Run the packing optimization.
    Returns (centers, radii, sum_radii).
    """
    # Variable bounds: x, y in [0, 1], r in [0, 0.5]

            res = opt.minimize(

    # We only need constraints for unique pairs (i, j) with i < j
    # to avoid duplicates and self-comparison.
    idx = np.triu_indices(N, k=1)
    c5 = diff_sq[idx]
    
    # Concatenate all constraints into a single array
    return np.concatenate([c1, c2, c3, c4, c5])

def get_initial_guess(use_hex, r_init):
    """
    Generate an initial guess vector for the optimizer.
    """
    pts = []
    if use_hex:
        # Hexagonal lattice initialization for better packing start
        d = 2 * r_init
        h = d * np.sqrt(3) / 2
        max_r = int(1 / h) + 5
        max_c = int(1 / d) + 5
        for r in range(max_r):
            for c in range(max_c):
                x = r_init + c * d + (r % 2) * (d / 2)
                y = r_init + r * h
                if x <= 1 - r_init and y <= 1 - r_init:
                    pts.append((x, y))
    else:
        # Random initialization
        for _ in range(N + 50):
            x = np.random.uniform(r_init, 1 - r_init)
            y = np.random.uniform(r_init, 1 - r_init)
            pts.append((x, y))
            
    pts = pts[:N]
    # Fill if necessary (should not happen with reasonable r_init)
    while len(pts) < N:
        pts.append((0.5, 0.5))
        
    v = np.zeros(3 * N)
    for i, (x, y) in enumerate(pts):
        v[3*i] = x
        v[3*i+1] = y
        v[3*i+2] = r_init
    return v

def run_packing():
    """
    Solves the circle packing problem to maximize sum of radii.
    Returns (centers, radii, sum_radii).
    """
    # Bounds for variables: x, y in [0, 1], r in [0, 0.5]
    bounds = [(0, 1)] * (2 * N) + [(0, 0.5)] * N
    constraints = {'type': 'ineq', 'fun': inequality_constraints}
    
    best_v = None
    best_sum_r = -1.0
    
    # Try multiple initialization strategies to find a good local optimum
    strategies = [
        (True, 0.08),   # Hex lattice, larger radius
        (True, 0.06),   # Hex lattice, smaller radius
        (False, 0.02),  # Random, small radius
        (False, 0.01),  # Random, very small radius
    ]
    
    for use_hex, r_init in strategies:
        v0 = get_initial_guess(use_hex, r_init)
        try:
            # Run optimization using SLSQP
            res = opt.minimize(
                objective, 
                v0, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=constraints,
                options={'maxiter': 2000, 'ftol': 1e-12}
            )
            
            # Check if the solution satisfies constraints within tolerance
            cons_vals = inequality_constraints(res.x)
            min_cons = np.min(cons_vals)
            
            # Tolerance for constraint violation
            if min_cons >= -1e-7:
                current_sum_r = -res.fun
                if current_sum_r > best_sum_r:
                    best_sum_r = current_sum_r
                    best_v = res.x.copy()
        except Exception:
            continue
            
    # Fallback solution if optimization fails to find a valid packing
    if best_v is None:
        v = np.zeros(3 * N)
        r_fallback = 0.02
        idx = 0
        for row in range(6):
            for col in range(5):
                if idx < N:
                    x = 0.1 + col * 0.2
                    y = 0.1 + row * 0.2
                    x = max(r_fallback, min(1 - r_fallback, x))
                    y = max(r_fallback, min(1 - r_fallback, y))
                    v[3*idx] = x
                    v[3*idx+1] = y
                    v[3*idx+2] = r_fallback
                    idx += 1
        best_v = v
        best_sum_r = np.sum(v[2*N:])

    centers = best_v[:2*N].reshape((N, 2))
    radii = best_v[2*N:]
    
    # Ensure radii are non-negative
    radii = np.maximum(radii, 0.0)
    
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
