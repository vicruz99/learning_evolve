import numpy as np
from scipy.optimize import minimize

def compute_penalty(v, mu, n):
    """
    Computes the objective function: -sum(radii) + penalty for violations.
    v: flattened array [x0, y0, r0, x1, y1, r1, ...]
    mu: penalty weight
    n: number of circles
    """
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    
    # Boundary penalties: ensure circles stay inside [0,1]x[0,1]
    pen = np.sum(np.maximum(0, r - x)**2)
    pen += np.sum(np.maximum(0, r - (1 - x))**2)
    pen += np.sum(np.maximum(0, r - y)**2)
    pen += np.sum(np.maximum(0, r - (1 - y))**2)
    
    # Overlap penalties: ensure dist >= r_i + r_j
    dx = x[:, None] - x
    dy = y[:, None] - y
    dist_mat = np.sqrt(dx**2 + dy**2)
    sum_r_mat = r[:, None] + r
    
    # Only consider upper triangle pairs (i < j)
    i_idx, j_idx = np.triu_indices(n, 1)
    overlaps = np.maximum(0, sum_r_mat[i_idx, j_idx] - dist_mat[i_idx, j_idx])
    pen += np.sum(overlaps**2)
    
    return -np.sum(r) + mu * pen

    for trial in range(10):
