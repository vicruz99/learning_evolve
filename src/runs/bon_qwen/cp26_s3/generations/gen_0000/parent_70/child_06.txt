def compute_cost_and_grad(params, n):
    """
    Computes the cost function and its gradient.
    params: array of size 3*n (x, y, r for each circle)
    """
    centers = params[:2*n].reshape((n, 2))
    radii = params[2*n:]
    
    # Objective: - sum(radii)
    cost = -np.sum(radii)
    grad_cost = np.zeros_like(params)
    grad_cost[2*n:] = -1.0
    
    # Penalty weights
    # We want constraints to be hard. 
    # Overlap: dist < r1 + r2
    # Boundary: r > x or r > 1-x etc.
    
    lam_overlap = 1000.0
    lam_bound = 1000.0
    
    # Overlap Penalty
    # P_overlap = sum_{i<j} max(0, (r_i + r_j)^2 - dist_ij^2)^2
    # Using squared distance to avoid sqrt
    
    # Vectorized pairwise distance squared
    # centers shape (n, 2)
    # diff shape (n, n, 2)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist_sq = np.sum(diff**2, axis=2)
    
    # Radii sum matrix
    r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    
    # Violation: (r_i + r_j)^2 - dist_sq
    # We only care if > 0
    violation = (r_sum**2 - dist_sq)
    # Mask for upper triangle (i < j)
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    violation_active = violation[mask]
    
    # Only positive violations matter
    active_mask = violation_active > 0
    active_viol = violation_active[active_mask]
    
    penalty_overlap = np.sum(active_viol**2)
    cost += lam_overlap * penalty_overlap
    
    # Gradient of overlap penalty
    # d/d(r_i) [ (r_i+r_j)^2 - dist^2 ]^2 = 2 * viol * 2*(r_i+r_j)
    # d/d(x_i) [ ... ] = 2 * viol * (-2*(x_i-x_j))
    # Note: dist_sq derivative w.r.t x_i is 2(x_i - x_j)
    # So derivative of (r_sum^2 - dist_sq) w.r.t x_i is -2(x_i-x_j)
    
    # We need to sum contributions from all neighbors for each circle
    
    # Let's compute gradients efficiently
    # For each pair (i, j) with i < j and active violation
    
    # Indices of active pairs
    rows, cols = np.where(mask & (violation > 0))
    
    if len(rows) > 0:
        viol_vals = violation[rows, cols] # Shape (M,)
        r_sums = r_sum[rows, cols]
        
        # Term derivative factor: 2 * viol * lam_overlap
        # Gradient contribution to cost is + lam_overlap * deriv
        
        # w.r.t radii
        # deriv_r_i = 2 * viol * 2 * (r_i + r_j) = 4 * viol * (r_i + r_j)
        # deriv_r_j = same
        
        # w.r.t x_i
        # deriv_x_i = 2 * viol * (-2 * (x_i - x_j)) = -4 * viol * (x_i - x_j)
        # deriv_x_j = 2 * viol * (-2 * (x_j - x_i)) = -4 * viol * (x_j - x_i)
        #            = 4 * viol * (x_i - x_j)
        
        # w.r.t y_i similar
        
        grad_factor = 4.0 * lam_overlap * viol_vals
        
        # Accumulate to grad_cost
        # radii indices are 2*n + i
        for k in range(len(rows)):
            i = rows[k]
            j = cols[k]
            g = grad_factor[k]
            
            # Radii
            grad_cost[2*n + i] += g * r_sums[k]
            grad_cost[2*n + j] += g * r_sums[k]
            
            # Centers
            # x: index i, y: index i+1 in flat centers? No, centers are flattened first.
            # params: [x0, x1... xn-1, y0, y1... yn-1, r0... rn-1]?
            # Wait, my reshape was params[:2*n].reshape((n, 2)).
            # This means params[0:n] are x, params[n:2n] are y.
            # Let's verify reshape order.
            # default C-order: row 0 (x0, y0), row 1 (x1, y1)...
            # So params[0] = x0, params[1] = y0, params[2] = x1, params[3] = y1.
            
            # So x_i is at 2*i, y_i is at 2*i+1.
            
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            
            grad_cost[2*i]   += -g * dx
            grad_cost[2*i+1] += -g * dy
            grad_cost[2*j]   +=  g * dx
            grad_cost[2*j+1] +=  g * dy

    # Boundary Penalty
    # P_bound = sum_i [ max(0, r_i - x_i)^2 + max(0, r_i - (1-x_i))^2 + ... ]
    # Note: x_i >= r_i => x_i - r_i >= 0. Violation r_i - x_i > 0.
    # Term: 0.5 * max(0, r_i - x_i)^2 ? 
    # In cost I used sum(viol^2). Let's stick to sum(viol^2) without 0.5 to match derivative logic?
    # Actually derivative of z^2 is 2z.
    # If P = z^2, grad = 2z * dz/dparam.
    
    # Violations
    # r > x  => viol = r - x
    # r > 1-x => viol = r - (1-x) = r + x - 1
    # r > y  => viol = r - y
    # r > 1-y => viol = r + y - 1
    
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        
        # Left wall
        v = r - x
        if v > 0:
            p = v**2
            cost += lam_bound * p
            # grad w.r.t x: 2*v * (-1)
            # grad w.r.t r: 2*v * (1)
            grad_cost[2*i]   += lam_bound * 2 * v * (-1)
            grad_cost[2*n+i] += lam_bound * 2 * v * (1)
            
        # Right wall
        v = r - (1 - x)
        if v > 0:
            p = v**2
            cost += lam_bound * p
            # grad w.r.t x: 2*v * (1) (since d/dx (r+x-1) = 1)
            # grad w.r.t r: 2*v * (1)
            grad_cost[2*i]   += lam_bound * 2 * v * (1)
            grad_cost[2*n+i] += lam_bound * 2 * v * (1)
            
        # Bottom wall
        v = r - y
        if v > 0:
            p = v**2
            cost += lam_bound * p
            # grad w.r.t y: 2*v * (-1)
            # grad w.r.t r: 2*v * (1)
            grad_cost[2*i+1] += lam_bound * 2 * v * (-1)
            grad_cost[2*n+i] += lam_bound * 2 * v * (1)
            
        # Top wall
        v = r - (1 - y)
        if v > 0:
            p = v**2
            cost += lam_bound * p
            # grad w.r.t y: 2*v * (1)
            # grad w.r.t r: 2*v * (1)
            grad_cost[2*i+1] += lam_bound * 2 * v * (1)
            grad_cost[2*n+i] += lam_bound * 2 * v * (1)
            
    return cost, grad_cost
