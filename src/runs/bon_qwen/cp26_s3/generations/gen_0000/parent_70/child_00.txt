import numpy as np
from scipy.optimize import minimize
import math

def objective_func(params):
    """Objective function to maximize sum of radii (minimize negative sum)."""
    radii = params[2::3]
    return -np.sum(radii)

def constraint_func(params):
    """Constraint function for SLSQP. Returns array of values >= 0."""
    n = 26
    centers = np.column_stack((params[0::3], params[1::3]))
    radii = params[2::3]
    
    cons = []
    
    # Boundary constraints: circles must be inside [0,1]x[0,1]
    # x >= r, y >= r, x <= 1-r, y <= 1-r
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        cons.append(x - r)       # >= 0
        cons.append(y - r)       # >= 0
        cons.append(1.0 - x - r) # >= 0
        cons.append(1.0 - y - r) # >= 0
        
    # Pairwise non-overlap constraints: dist >= r_i + r_j
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = math.sqrt(dx * dx + dy * dy)
            cons.append(dist - radii[i] - radii[j])
            
    return np.array(cons)

def run_packing():
    np.random.seed(42)
    n = 26
    
    # Initialization: Hexagonal packing pattern
    centers_init = []
    r_init = 0.085
    y = r_init
    row = 0
    while len(centers_init) < n and y < 1.0 - r_init:
        x = r_init
        if row % 2 == 1:
            x += r_init  # Staggered row
        while x < 1.0 - r_init and len(centers_init) < n:
            centers_init.append([x, y])
            x += 2 * r_init
        y += math.sqrt(3) * r_init
        row += 1
        
    # Fill any remaining slots with random points if hex pattern was sparse
    while len(centers_init) < n:
        centers_init.append([np.random.rand(), np.random.rand()])
        
    centers_init = np.array(centers_init[:n])
    
    # Flatten into optimization vector: [x1, y1, r1, ..., xn, yn, rn]
    params0 = np.zeros(3 * n)
    params0[0::3] = centers_init[:, 0]
    params0[1::3] = centers_init[:, 1]
    params0[2::3] = np.ones(n) * 0.05  # Start with small valid radii
    
    # Bounds for variables
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)] * n + [(0.0, 0.5)] * n
    
    # Run optimization
    res = minimize(
        fun=objective_func,
        x0=params0,
        bounds=bounds,
        constraints={'type': 'ineq', 'fun': constraint_func},
        method='SLSQP',
        options={'maxiter': 2000, 'ftol': 1e-15, 'disp': False}
    )
    
    best_params = res.x
    centers = np.column_stack((best_params[0::3], best_params[1::3]))
    radii = best_params[2::3]
    
    # Ensure non-negative radii
    radii = np.maximum(radii, 0.0)
    
    # Calculate final sum
    total_sum = np.sum(radii)
    
    return centers, radii, total_sum
