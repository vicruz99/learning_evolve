def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True

def objective(vars_vec, n):
    """
    Objective function to minimize (negative sum of radii).
    """
    radii = vars_vec[2::3]
    return -np.sum(radii)

def constraint_boundary(vars_vec, n):
    """
    Constraints: circles inside [0,1]x[0,1]
    x_i >= r_i  => x_i - r_i >= 0
    x_i <= 1-r_i => 1 - x_i - r_i >= 0
    y_i >= r_i  => y_i - r_i >= 0
    y_i <= 1-r_i => 1 - y_i - r_i >= 0
    """
    cons = []
    for i in range(n):
        x = vars_vec[3*i]
        y = vars_vec[3*i+1]
        r = vars_vec[3*i+2]
        
        cons.append(x - r)
        cons.append(1 - x - r)
        cons.append(y - r)
        cons.append(1 - y - r)
    return np.array(cons)

def constraint_overlap(vars_vec, n):
    """
    Constraints: non-overlap
    dist_ij >= r_i + r_j
    (x_i - x_j)^2 + (y_i - y_j)^2 >= (r_i + r_j)^2
    """
    cons = []
    for i in range(n):
        for j in range(i + 1, n):
            x_i, y_i, r_i = vars_vec[3*i], vars_vec[3*i+1], vars_vec[3*i+2]
            x_j, y_j, r_j = vars_vec[3*j], vars_vec[3*j+1], vars_vec[3*j+2]
            
            dist_sq = (x_i - x_j)**2 + (y_i - y_j)**2
            r_sum_sq = (r_i + r_j)**2
            
            cons.append(dist_sq - r_sum_sq)
    return np.array(cons)
