def get_constraints(v):
    n = len(v) // 3
    c = []
    for i in range(n):
        x = v[3*i]
        y = v[3*i+1]
        r = v[3*i+2]
        
        # Boundary constraints: circle inside [0,1]x[0,1]
        # x >= r
        c.append(x - r)
        # 1 - x >= r
        c.append(1.0 - x - r)
        # y >= r
        c.append(y - r)
        # 1 - y >= r
        c.append(1.0 - y - r)
        
        # Pairwise non-overlap constraints
        # dist(i, j) >= r_i + r_j
        # dist^2 >= (r_i + r_j)^2
        for j in range(i + 1, n):

            yj = v[3*j+1]
            rj = v[3*j+2]

            dy = y - yj
