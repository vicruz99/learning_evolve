def solve_radii_lp(centers):
    """
    Given fixed centers, solve the LP to find the optimal radii that maximize sum(r_i)
    subject to non-overlap and boundary constraints.
    """
    n = len(centers)
    
    # Variables: r_0, r_1, ..., r_{n-1}
    # Objective: Maximize sum(r_i) => Minimize -sum(r_i)
    c_obj = -np.ones(n)
    
    # Constraints:
    # 1. r_i >= 0 (Handled by bounds)
    # 2. r_i <= x_i
    # 3. r_i <= 1 - x_i
    # 4. r_i <= y_i
    # 5. r_i <= 1 - y_i
    # 6. r_i + r_j <= dist(i, j) for all i < j
    
    # We use the standard form for linprog: 
    # Minimize c^T x
    # subject to A_ub x <= b_ub
    #            A_eq x = b_eq
    #            bounds
    
    A_ub = []
    b_ub = []
    
    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        # r_i <= x
        row = np.zeros(n)
        row[i] = 1
        A_ub.append(row)
        b_ub.append(x)
        
        # r_i <= 1 - x
        row = np.zeros(n)
        row[i] = 1
        A_ub.append(row)
        b_ub.append(1 - x)
        
        # r_i <= y
        row = np.zeros(n)
        row[i] = 1
        A_ub.append(row)
        b_ub.append(y)
        
        # r_i <= 1 - y
        row = np.zeros(n)
        row[i] = 1
        A_ub.append(row)
        b_ub.append(1 - y)
        
    # Pairwise constraints: r_i + r_j <= dist
    # Precompute distances
    dists = np.zeros((n, n))
    for i in range(n):
        for j in range(i + 1, n):
            d = np.linalg.norm(centers[i] - centers[j])
            dists[i, j] = d
            dists[j, i] = d
            
            row = np.zeros(n)
            row[i] = 1
            row[j] = 1
            A_ub.append(row)
            b_ub.append(d)
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
        else:
            # Fallback to a safe small radius if LP fails
            return np.full(n, 0.001)
    except Exception:
        return np.full(n, 0.001)
