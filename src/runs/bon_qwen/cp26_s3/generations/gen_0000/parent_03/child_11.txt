import numpy as np
from scipy.optimize import minimize

def objective(v):
    return -np.sum(v[2::3])

def constr_x_r(v, i):
    return v[3*i] - v[3*i+2]

def constr_1_x_r(v, i):
    return 1.0 - v[3*i] - v[3*i+2]

def constr_y_r(v, i):
    return v[3*i+1] - v[3*i+2]

def constr_1_y_r(v, i):
    return 1.0 - v[3*i+1] - v[3*i+2]

def constr_overlap(v, i, j):
    dx = v[3*i] - v[3*j]
    dy = v[3*i+1] - v[3*j+1]
    dist = np.sqrt(dx*dx + dy*dy + 1e-12)
    return dist - (v[3*i+2] + v[3*j+2])

def run_packing():
    N = 26
    v0 = np.zeros(3 * N)
    idx = 0
    
    # Hexagonal lattice initialization for efficient packing
    y = 0.12
    dy = 0.155
    dx = 0.175
    shift = 0.0
    
    while idx < N:
        x = 0.12 + shift
        while x <= 0.88:
            if idx < N:
                v0[3*idx] = x
                v0[3*idx+1] = y
                v0[3*idx+2] = 0.05
                idx += 1
            x += dx
        y += dy
        shift = dx - shift
        
    bounds = [(0.0, 1.0)] * N + [(0.0, 1.0)] * N + [(0.0, 0.5)] * N
    
    constraints = []
    for i in range(N):
        constraints.append({'type': 'ineq', 'fun': constr_x_r, 'args': (i,)})
        constraints.append({'type': 'ineq', 'fun': constr_1_x_r, 'args': (i,)})
        constraints.append({'type': 'ineq', 'fun': constr_y_r, 'args': (i,)})
        constraints.append({'type': 'ineq', 'fun': constr_1_y_r, 'args': (i,)})
        
    for i in range(N):
        for j in range(i+1, N):
            constraints.append({'type': 'ineq', 'fun': constr_overlap, 'args': (i, j)})
            
    res = minimize(objective, v0, method='SLSQP', bounds=bounds, constraints=constraints, 
                   options={'maxiter': 5000, 'ftol': 1e-12, 'disp': False})
                   
    centers = res.x[:2*N].reshape(N, 2)
    radii = res.x[2*N:]
    radii = np.maximum(radii, 0.0)
    
    return centers, radii, np.sum(radii)
