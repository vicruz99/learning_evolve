import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26
OVERLAP_PAIRS = np.triu_indices(N_CIRCLES, k=1)

def objective_func(x):
    """Negative sum of radii to be minimized."""
    return -np.sum(x[2*N_CIRCLES:])

def constraint_func(x):
    """Returns constraint values >= 0 for feasibility."""
    centers = x[:2*N_CIRCLES].reshape(N_CIRCLES, 2)
    radii = x[2*N_CIRCLES:]
    
    con = []
    # Boundary constraints
    con.append(centers[:, 0] - radii)
    con.append(1.0 - centers[:, 0] - radii)
    con.append(centers[:, 1] - radii)
    con.append(1.0 - centers[:, 1] - radii)
    
    # Overlap constraints
    diffs = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    rad_sums = radii[:, None] + radii[None, :]
    con.append(dists[OVERLAP_PAIRS] - rad_sums[OVERLAP_PAIRS])
    
    return np.concatenate(con)

def get_initial_guess(seed):
    """Generates a perturbed triangular lattice configuration."""
    np.random.seed(seed)
    pts = []
    dx = 0.22
    dy = dx * np.sqrt(3) / 2
    
    for i in range(10):
        y = 0.12 + i * dy
        if y > 0.88: break
        offset = (i % 2) * (dx / 2)
        x = 0.12 + offset
        while x < 0.88:
            pts.append([x, y])
            x += dx
            
    np.random.shuffle(pts)
    centers = np.array(pts[:N_CIRCLES])
    centers += np.random.randn(N_CIRCLES, 2) * 0.008
    centers = np.clip(centers, 0.06, 0.94)
    radii = np.full(N_CIRCLES, 0.05)
    return np.concatenate([centers.flatten(), radii])

def run_packing():
    bounds = [(0.0, 1.0)] * (2*N_CIRCLES) + [(0.0, 0.5)] * N_CIRCLES
    constr = {'type': 'ineq', 'fun': constraint_func}
    
    best_x = None
    best_val = np.inf
    
    # Multi-start optimization
    for seed in range(20):
        x0 = get_initial_guess(seed)
        try:
            res = minimize(objective_func, x0, method='SLSQP', bounds=bounds, 
                           constraints=[constr], options={'maxiter': 2500, 'ftol': 1e-10})
            if res.fun < best_val:
                best_val = res.fun
                best_x = res.x
        except Exception:
            continue
            
    if best_x is None:
        best_x = get_initial_guess(0)
        
    centers = best_x[:2*N_CIRCLES].reshape(N_CIRCLES, 2)
    radii = best_x[2*N_CIRCLES:]
    
    # Ensure strict boundary compliance within numerical tolerance
    for i in range(N_CIRCLES):
        cx, cy = centers[i]
        r = radii[i]
        centers[i, 0] = np.clip(cx, r, 1.0 - r)
        centers[i, 1] = np.clip(cy, r, 1.0 - r)
        
    return centers, radii, -best_val
