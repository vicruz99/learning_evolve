import numpy as np
from scipy.optimize import minimize


def compute_penalty(centers, radii, n):
    """Compute squared penalty for constraint violations."""
    penalty = 0.0
    # Boundary violations
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < 0:
            penalty += (x - r) ** 2
        if x + r > 1:
            penalty += (x + r - 1) ** 2
        if y - r < 0:
            penalty += (y - r) ** 2
        if y + r > 1:
            penalty += (y + r - 1) ** 2
    # Overlap violations
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j]:
                penalty += (radii[i] + radii[j] - dist) ** 2
    return penalty


def objective_with_penalty(x, n, penalty_weight):
    """Objective: minimize negative sum of radii plus penalty for violations."""
    centers = x[: 2 * n].reshape(n, 2)
    radii = x[2 * n :]
    obj = -np.sum(radii)
    penalty = compute_penalty(centers, radii, n)
    return obj + penalty_weight * penalty


def init_hexagonal(n, seed):
    """Initialize circles in a hexagonal grid pattern with noise."""
    np.random.seed(seed)
    centers = np.zeros((n, 2))
    # 6 rows: 5+4+5+4+5+3 = 26
    rows = 6
    cols_per_row = [5, 4, 5, 4, 5, 3]
    idx = 0
    for row in range(rows):
        count = cols_per_row[row]
        spacing_x = 1.0 / (count + 1)
        for col in range(count):
            x = spacing_x * (col + 1)
            y = (row + 0.5) / rows
            if row % 2 == 1:
                x += spacing_x / 2.0
            # Add controlled noise
            x += np.random.uniform(-0.015, 0.015)
            y += np.random.uniform(-0.015, 0.015)
            centers[idx] = [np.clip(x, 0.01, 0.99), np.clip(y, 0.01, 0.99)]
            idx += 1
    radii = np.ones(n) * 0.085
    return centers, radii


def validate(centers, radii, n):
    """Check if packing is valid."""
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-9 or x + r > 1 + 1e-9 or y - r < -1e-9 or y + r > 1 + 1e-9:
            return False
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-9:
                return False
    return True


def optimize_from_init(centers, radii, n):
    """Run optimization with annealing penalty weights."""
    x0 = np.concatenate([centers.flatten(), radii])
    penalty_weights = [200, 800, 2000, 5000, 10000, 20000]
    for pw in penalty_weights:
        def obj(x, pw=pw):
            return objective_with_penalty(x, n, pw)
        bounds = [(0, 1)] * (2 * n) + [(0.001, 0.5)] * n
        result = minimize(
            obj, x0, method="L-BFGS-B", bounds=bounds,
            options={"maxiter": 3000, "ftol": 1e-15, "gtol": 1e-10}
        )
        x0 = result.x.copy()
    return x0[: 2 * n].reshape(n, 2), x0[2 * n :]


def run_packing() -> tuple:
    n = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None

    for trial in range(30):
        centers_init, radii_init = init_hexagonal(n, trial)
        centers_opt, radii_opt = optimize_from_init(centers_init, radii_init, n)
        current_sum = np.sum(radii_opt)
        if validate(centers_opt, radii_opt, n) and current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers_opt.copy()
            best_radii = radii_opt.copy()

    # Ensure we have a valid result
    if best_centers is None:
        # Fallback: use last trial regardless of validity check edge cases
        centers_opt, radii_opt = optimize_from_init(*init_hexagonal(n, 0), n)
        best_centers = centers_opt
        best_radii = radii_opt
        best_sum = np.sum(radii_opt)

    return best_centers, best_radii, best_sum
