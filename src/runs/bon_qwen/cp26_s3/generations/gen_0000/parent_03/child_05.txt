import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    Uses a penalty-based optimization approach starting from a grid-like initialization.
    """
    n = 26
    
    # Objective function to minimize: -(sum of radii) + penalty for violations
    def objective(vars, n):
        centers = np.zeros((n, 2))
        radii = np.zeros(n)
        for i in range(n):
            centers[i] = vars[3*i], vars[3*i+1]
            radii[i] = vars[3*i+2]
        
        # Primary objective: maximize sum of radii (minimize negative sum)
        score = -np.sum(radii)
        
        penalty_factor = 1000.0
        penalty = 0.0
        
        # Boundary constraints: circle must be inside [0, 1]x[0, 1]
        # x - r >= 0  => violation if x < r
        # x + r <= 1  => violation if x > 1 - r
        # same for y
        
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            
            # Left boundary
            viol = x - r
            if viol < 0:
                penalty += penalty_factor * viol**2
            
            # Right boundary
            viol = (x + r) - 1
            if viol > 0:
                penalty += penalty_factor * viol**2
                
            # Bottom boundary
            viol = y - r
            if viol < 0:
                penalty += penalty_factor * viol**2
            
            # Top boundary
            viol = (y + r) - 1
            if viol > 0:
                penalty += penalty_factor * viol**2
        
        # Overlap constraints: distance between centers >= sum of radii
        # dist - (r_i + r_j) >= 0
        
        # Vectorized distance calculation for efficiency
        # centers shape (n, 2)
        # Compute pairwise squared distances
        # Using broadcasting
        diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
        dist_sq = np.sum(diff**2, axis=2)
        dist = np.sqrt(dist_sq)
        
        # Sum of radii matrix
        r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
        
        # Violation matrix: where dist < r_sum
        violations = r_sum - dist
        
        # We only care about violations > 0 (overlap)
        # And only for i < j (upper triangle) to avoid double counting, 
        # but summing all squares is fine, just scales penalty.
        # Actually, summing all is easier.
        
        # Mask for upper triangle (i < j) to avoid double counting and self-check
        # Self check is dist=0, r_sum=2r > 0, so violation = 2r. 
        # Wait, self-check is not a constraint. 
        # We must exclude diagonal.
        
        mask = np.triu(np.ones((n, n), dtype=bool), k=1)
        valid_violations = violations[mask]
        
        # Penalty for overlaps
        overlaps = valid_violations[valid_violations > 0]
        if len(overlaps) > 0:
            penalty += penalty_factor * np.sum(overlaps**2)
            
        return score + penalty

    # Initialization
    # Start with a 5x5 grid (25 circles) radius 0.1
    # Add 1 circle in a gap
    centers_init = []
    radii_init = []
    
    # Grid coordinates for 5x5
    # x in {0.1, 0.3, 0.5, 0.7, 0.9}
    grid_x = [0.1, 0.3, 0.5, 0.7, 0.9]
    grid_y = [0.1, 0.3, 0.5, 0.7, 0.9]
    
    count = 0
    for y in grid_y:
        for x in grid_x:
            centers_init.append([x, y])
            radii_init.append(0.1)
            count += 1
            if count == 26:
                break
        if count == 26:
            break
            
    # If we need more circles (we stopped at 25 in grid, need 26)
    # Actually the loop above fills 25. We need 1 more.
    # Let's find a spot with low overlap. Center of a hole?
    # Holes are at (0.2, 0.2), (0.2, 0.4) etc.
    # (0.2, 0.2) is surrounded by (0.1, 0.1), (0.3, 0.1), (0.1, 0.3), (0.3, 0.3).
    # Dist to (0.1, 0.1) is sqrt(0.01+0.01) approx 0.141.
    # Sum of radii 0.2. Overlap 0.059.
    # This is a reasonable starting point.
    
    if len(centers_init) < n:
        # Add remaining circles in random holes or random spots
        # Try to place in gaps first
        potential_spots = [
            [0.2, 0.2], [0.2, 0.4], [0.2, 0.6], [0.2, 0.8],
            [0.4, 0.2], [0.4, 0.4], [0.4, 0.6], [0.4, 0.8],
            [0.6, 0.2], [0.6, 0.4], [0.6, 0.6], [0.6, 0.8],
            [0.8, 0.2], [0.8, 0.4], [0.8, 0.6], [0.8, 0.8],
            [0.5, 0.2], [0.5, 0.8] # edge centers
        ]
        
        for spot in potential_spots:
            if len(centers_init) >= n:
                break
            # Check if spot is too close to existing?
            # Just add it. Optimizer will fix it.
            # Smaller radius to start
            centers_init.append(spot)
            radii_init.append(0.05)
            
        # Fill remaining randomly if needed
        while len(centers_init) < n:
            centers_init.append([np.random.uniform(0, 1), np.random.uniform(0, 1)])
            radii_init.append(0.01)

    # Flatten to array
    vars_init = np.array([])
    for i in range(n):
        vars_init = np.append(vars_init, [centers_init[i][0], centers_init[i][1], radii_init[i]])
    
    # Bounds for optimization
    # x, y in [0, 1], r in [0, 0.5]
    bounds = []
    for _ in range(n):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 0.5)) # r

    # Optimization
    # Use L-BFGS-B for bound constraints
    # Multiple restarts might help, but let's try one robust run first.
    # We can try to perturb initial vars slightly to escape symmetry if needed.
    np.random.seed(42) # For reproducibility
    
    # Try to optimize
    res = minimize(objective, vars_init, args=(n,), method='L-BFGS-B', bounds=bounds, 
                   options={'maxiter': 2000, 'ftol': 1e-12, 'gtol': 1e-8})
    
    # Check if we improved significantly. If not, maybe try another start?
    # But for this task, one good run is likely sufficient given the initialization.
    # Let's do a second pass with random perturbation to ensure global optimum exploration.
    
    # Perturb and optimize again
    vars_perturbed = vars_init.copy() + np.random.normal(0, 0.01, size=vars_init.shape)
    # Clip perturbed vars to bounds
    # x, y clip to [0,1], r clip to [0, 0.5]
    # vars structure: [x0, y0, r0, x1, ...]
    for i in range(0, len(vars_perturbed), 3):
        vars_perturbed[i] = np.clip(vars_perturbed[i], 0, 1)
        vars_perturbed[i+1] = np.clip(vars_perturbed[i+1], 0, 1)
        vars_perturbed[i+2] = np.clip(vars_perturbed[i+2], 0, 0.5)
        
    res2 = minimize(objective, vars_perturbed, args=(n,), method='L-BFGS-B', bounds=bounds, 
                    options={'maxiter': 2000, 'ftol': 1e-12, 'gtol': 1e-8})
    
    # Select best result
    if res.fun < res2.fun: # Lower fun means higher sum radii (since fun = -sum + penalty)
        best_vars = res.x
    else:
        best_vars = res2.x
        
    # Extract results
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        centers[i] = best_vars[3*i], best_vars[3*i+1]
        radii[i] = best_vars[3*i+2]
        
    # Ensure radii are non-negative (just in case)
    radii = np.maximum(radii, 0.0)
    
    # Final validation check (optional but good for debugging mentally)
    # Not needed for return
    
    return centers, radii, float(np.sum(radii))
