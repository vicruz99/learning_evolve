import numpy as np
from scipy.optimize import minimize

def compute_objective(vars, n, penalty_weight):
    """
    Objective function: maximize sum of radii with overlap penalty.
    vars: flattened array [x1, y1, ..., xn, yn, r1, ..., rn]
    """
    centers = vars[:2*n].reshape(n, 2)
    radii = vars[2*n:]
    
    # Compute pairwise Euclidean distances
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    
    # Overlap amount: positive when circles intersect
    overlaps = radii[:, np.newaxis] + radii[np.newaxis, :] - dists
    
    # Consider only upper triangle to avoid double counting
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    overlaps = np.where(mask, overlaps, 0.0)
    
    # Quadratic penalty for overlaps
    penalty = penalty_weight * np.sum(np.maximum(0.0, overlaps)**2)
    
    # We minimize negative sum of radii + penalty
    return -np.sum(radii) + penalty

def run_packing():
    n = 26
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Variable bounds: centers in [0,1], radii in [1e-5, 0.5]
    bounds = [(0.0, 1.0)] * (2 * n) + [(1e-5, 0.5)] * n
    
    # Adaptive penalty schedule
    penalty_weights = [1000, 5000, 10000, 50000, 100000]
    
    # Multiple restarts to ensure global optimum
    for seed in range(10):
        np.random.seed(seed)
        
        # Initialize with hex-like grid pattern
        init_centers_list = []
        row_counts = [6, 5, 6, 5, 4]
        y_step = 1.0 / 6.0
        for i, count in enumerate(row_counts):
            y = (i + 1) * y_step
            x_step = 1.0 / (count + 1)
            for j in range(count):
                x = (j + 1) * x_step
                # Shift odd rows for hexagonal packing
                if i % 2 == 1:
                    x += x_step / 2.0
                init_centers_list.append([x, y])
                
        init_centers = np.array(init_centers_list[:n])
        # Add controlled noise to break symmetry and aid optimization
        init_centers += np.random.uniform(-0.02, 0.02, size=init_centers.shape)
        init_centers = np.clip(init_centers, 0.05, 0.95)
        
        init_radii = np.full(n, 0.07)
        current_vars = np.concatenate([init_centers.ravel(), init_radii])
        
        # Iterative optimization with increasing penalty
        for pw in penalty_weights:
            res = minimize(compute_objective, current_vars, 
                           args=(n, pw), method='L-BFGS-B', 
                           bounds=bounds, 
                           options={'maxiter': 2000, 'ftol': 1e-15, 'gtol': 1e-8})
            current_vars = res.x
            
        centers_opt = current_vars[:2*n].reshape(n, 2)
        radii_opt = current_vars[2*n:]
        current_sum = np.sum(radii_opt)
        
        # Verify feasibility against tolerance
        diffs = centers_opt[:, np.newaxis, :] - centers_opt[np.newaxis, :, :]
        dists = np.sqrt(np.sum(diffs**2, axis=2))
        overlaps = radii_opt[:, np.newaxis] + radii_opt[np.newaxis, :] - dists
        overlaps[np.tril_indices(n)] = 0
        max_overlap = np.max(np.maximum(0.0, overlaps))
        
        # Accept if overlap is negligible and improves best sum
        if max_overlap < 1e-4 and current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers_opt.copy()
            best_radii = radii_opt.copy()
            
    # Fallback in case of failure (should not happen with this setup)
    if best_centers is None:
        best_centers = np.random.rand(n, 2) * 0.8 + 0.1
        best_radii = np.full(n, 0.05)
        best_sum = np.sum(best_radii)
        
    return best_centers, best_radii, best_sum
