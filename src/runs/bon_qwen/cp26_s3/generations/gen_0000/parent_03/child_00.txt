    """Smooth approximation of max(0, violation)^2"""
    # Using a smooth hinge loss approximation
    return 0.5 * (violation + np.sqrt(violation**2 + 4 * sigma))**2 / 2 - 0.5 * sigma + 0.5 * sigma
