from scipy.optimize import minimize


def hexagonal_init(n=26):
    """Initialize circles in a hexagonal pattern."""
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # 5 rows: 6, 5, 6, 5, 4 = 26 circles
    row_counts = [6, 5, 6, 5, 4]
    r_init = 0.09
    y_spacing = 0.19
    x_spacing = 0.17
    
    idx = 0
    for row in range(5):
        y = r_init + row * y_spacing
        num_cols = row_counts[row]
        
        if row % 2 == 0:
            x_start = r_init
        else:
            x_start = r_init + x_spacing / 2
        
        for col in range(num_cols):
            if idx >= n:
                break
            centers[idx, 0] = x_start + col * x_spacing
            centers[idx, 1] = y
            radii[idx] = r_init
            idx += 1
            if idx >= n:
                break
        if idx >= n:
            break
    
    return centers, radii


def fix_violations(centers, radii, tol=1e-10):
    """Fix boundary and overlap violations by reducing radii."""
    n = len(radii)
    centers = centers.copy()
    radii = radii.copy()
    
    # Fix boundary violations
    for i in range(n):
        max_r = min(centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])
        radii[i] = min(radii[i], max(0, max_r))
    
    # Fix overlap violations iteratively
    for iteration in range(2000):
        max_overlap = 0
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                sum_r = radii[i] + radii[j]
                if dist < sum_r - tol:
                    overlap = sum_r - dist
                    max_overlap = max(max_overlap, overlap)
                    # Reduce proportionally to radius size
                    if sum_r > 0:
                        scale_i = radii[i] / sum_r
                        scale_j = radii[j] / sum_r

                        scale_i = 0.5
                        scale_j = 0.5
                    radii[i] = max(0, radii[i] - overlap * scale_i)
                    radii[j] = max(0, radii[j] - overlap * scale_j)
        
        if max_overlap < tol:
            break
    
    return centers, radii


def penalty_objective(x, n, penalty_weight):
    """Compute objective with penalties for constraint violations."""
    c = x[:2 * n].reshape(n, 2)
    r = x[2 * n:]
    
    obj = -np.sum(r)
    
    # Boundary penalties
    for i in range(n):
        for coord in range(2):
            val = c[i, coord]
            if val < r[i]:
                obj += penalty_weight * (val - r[i]) ** 2
            if val > 1 - r[i]:
                obj += penalty_weight * (val - (1 - r[i])) ** 2
    
    # Overlap penalties
    for i in range(n):
        for j in range(i + 1, n):
            dx = c[i, 0] - c[j, 0]
            dy = c[i, 1] - c[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            sum_r = r[i] + r[j]
            if dist < sum_r:
                obj += penalty_weight * (dist - sum_r) ** 2
    
    return obj
