

<strategy>
The problem is a non-convex optimization task: maximizing the sum of radii of 26 circles in a unit square. 
My strategy uses an alternating optimization approach combined with sensitivity analysis from Linear Programming:
1. **Alternating Optimization**: Fix circle centers, solve for optimal radii. Then use sensitivity information to move centers to create more space, repeat.
2. **LP for Radii**: For fixed centers, maximizing $\sum r_i$ subject to non-overlap ($r_i + r_j \le d_{ij}$) and boundary ($r_i \le \text{dist to edge}$) constraints is a Linear Program. This gives the exact optimal radii for any configuration efficiently.
3. **Dual-Guided Repulsion**: The dual variables (marginals) from the LP indicate which constraints are most "binding". A high marginal for $r_i \le x_i$ means moving circle $i$ rightward would significantly increase the total sum of radii. We use these marginals to compute a force vector for each center, pushing circles away from tight constraints (boundaries and other circles). This acts like a physics-based repulsion but guided by actual optimization sensitivity.
4. **Initialization & Restart**: Start with multiple configurations (hexagonal lattice, grid, random) to avoid poor local optima. Apply iterative force-based movement with decaying learning rate and periodic jitter.
5. **Validation**: The LP guarantees strict feasibility. The final output is verified to meet all constraints within numerical tolerance.

import numpy as np
from scipy.optimize import linprog
from scipy.spatial.distance import cdist

def compute_optimal_radii_and_forces(centers):
    """
    Solves the LP to maximize sum of radii for fixed centers.
    Returns optimal radii and force vectors derived from LP dual variables.
    """
    n = centers.shape[0]
    c_obj = -np.ones(n)  # Minimize negative sum of radii
    
    A_rows = []
    b_vals = []
    constraint_types = []
    constraint_ids = []
    
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        x, y = centers[i]
        row = np.zeros(n)
        row[i] = 1.0
        
        # r_i <= x
        A_rows.append(row.copy())
        b_vals.append(x)
        constraint_types.append('bx')
        constraint_ids.append(i)
        
        # r_i <= 1 - x
        A_rows.append(row.copy())
        b_vals.append(1.0 - x)
        constraint_types.append('bnx')
        constraint_ids.append(i)
        
        # r_i <= y
        A_rows.append(row.copy())
        b_vals.append(y)
        constraint_types.append('by')
        constraint_ids.append(i)
        
        # r_i <= 1 - y
        A_rows.append(row.copy())
        b_vals.append(1.0 - y)
        constraint_types.append('bny')
        constraint_ids.append(i)
        
    # Pairwise non-overlap constraints: r_i + r_j <= d_ij
    dists = cdist(centers, centers)
    np.fill_diagonal(dists, 0.0)
    
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_rows.append(row)
            b_vals.append(dists[i, j])
            constraint_types.append('pair')
            constraint_ids.append((i, j))
            
    A_ub = np.array(A_rows)
    b_ub = np.array(b_vals)
    bounds = [(0.0, None) for _ in range(n)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs', options={'disp': False})
        if not res.success:
            return np.full(n, 1e-4), np.zeros((n, 2))
            
        radii = res.x
        marginals = np.abs(res.ineqlin.marginals)
    except Exception:
        return np.full(n, 1e-4), np.zeros((n, 2))
        
    forces = np.zeros((n, 2))
    
    for k in range(len(marginals)):
        m = marginals[k]
        if m < 1e-10:
            continue
            
        ctype = constraint_types[k]
        cid = constraint_ids[k]
        
        if ctype == 'bx':
            forces[cid, 0] += m
        elif ctype == 'bnx':
            forces[cid, 0] -= m
        elif ctype == 'by':
            forces[cid, 1] += m
        elif ctype == 'bny':
            forces[cid, 1] -= m
        elif ctype == 'pair':
            i, j = cid
            v = centers[i] - centers[j]
            d = np.linalg.norm(v)
            if d > 1e-8:
                direction = v / d
                forces[i] += direction * m
                forces[j] -= direction * m
                
    return radii, forces

def generate_hex_init(n):
    """Generates a hexagonal lattice initialization for n circles."""
    centers = []
    # Layout: 6, 5, 6, 5, 4 rows sums to 26
    row_counts = [6, 5, 6, 5, 4]
    y_step = 0.14
    x_step = 0.16
    y_base = 0.12
    
    for i, count in enumerate(row_counts):
        y = y_base + i * y_step
        x_start = 0.5 - (count - 1) * x_step / 2
        for j in range(count):
            x = x_start + j * x_step
            if i % 2 == 1:
                x += x_step / 2
            centers.append([x, y])
            if len(centers) >= n:
                break
        if len(centers) >= n:
            break
    return np.array(centers[:n])

def generate_grid_init(n):
    """Generates a grid initialization."""
    centers = []
    for i in range(5):
        for j in range(5):
            centers.append([0.1 + j * 0.2, 0.1 + i * 0.2])
    if len(centers) < n:
        centers.append([0.5, 0.5])
    return np.array(centers[:n])

def run_packing():
    np.random.seed(42)
    n = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    initial_configs = [
        generate_hex_init(n),
        generate_grid_init(n),
        np.random.rand(n, 2),
        np.random.rand(n, 2),
        np.random.rand(n, 2)
    ]
    
    for init in initial_configs:
        centers = init.copy()
        for step in range(400):
            radii, forces = compute_optimal_radii_and_forces(centers)
            current_sum = np.sum(radii)
            
            if current_sum > best_sum:
                best_sum = current_sum
                best_centers = centers.copy()
                best_radii = radii.copy()
                
            # Learning rate decay
            lr = 0.008 / (1.0 + step / 60.0)
            centers += forces * lr
            
            # Periodic jitter to escape local minima
            if step % 40 == 0:
                noise_scale = 0.025 * max(0.0, 1.0 - step / 150.0)
                centers += np.random.randn(n, 2) * noise_scale
                
            # Keep centers strictly inside to avoid numerical boundary issues
            centers[:, 0] = np.clip(centers[:, 0], 0.015, 0.985)
            centers[:, 1] = np.clip(centers[:, 1], 0.015, 0.985)
            
    # Final LP solve on best configuration to ensure radii are maximized
    final_radii, _ = compute_optimal_radii_and_forces(best_centers)
    
    return best_centers, final_radii, np.sum(final_radii)
