import numpy as np
from scipy.optimize import minimize

def compute_packing_metrics(centers):
    """
    Computes the maximum valid radii for given centers.
    Ensures circles are within [0,1]x[0,1] and non-overlapping.
    """
    n = centers.shape[0]
    radii = np.full(n, 1.0)
    
    # Boundary constraints: distance to nearest edge
    for i in range(n):
        x, y = centers[i]
        radii[i] = min(x, 1.0 - x, y, 1.0 - y)
        
    # Pairwise constraints: distance to nearest neighbor
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            r_lim = dist * 0.5
            if r_lim < radii[i]:
                radii[i] = r_lim
            if r_lim < radii[j]:
                radii[j] = r_lim
                
    return radii

def objective_function(x):
    """Objective to minimize: negative sum of radii."""
    centers = x.reshape(-1, 2)
    radii = compute_packing_metrics(centers)
    return -np.sum(radii)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n_circles = 26
    best_sum = -np.inf
    best_centers = None
    best_radii = None
    
    # Configuration parameters
    n_trials = 40
    np.random.seed(42)
    seeds = np.random.randint(0, 10000, size=n_trials)
    
    for seed in seeds:
        rng = np.random.RandomState(seed)
        
        # --- Hexagonal Lattice Initialization ---
        pts = []
        dx = 0.22
        dy = dx * np.sqrt(3) / 2.0
        for j in range(8):
            for i in range(8):
                x = i * dx + (j % 2) * dx / 2.0
                y = j * dy
                if 0.06 <= x <= 0.94 and 0.06 <= y <= 0.94:
                    pts.append([x, y])
                    if len(pts) >= n_circles:
                        break
            if len(pts) >= n_circles:
                break
                
        # Fallback if hex grid doesn't yield enough points (rare)
        if len(pts) < n_circles:
            pts = [rng.uniform(0.15, 0.85, 2) for _ in range(n_circles)]
            
        centers = np.array(pts[:n_circles])
        centers += rng.uniform(-0.015, 0.015, size=centers.shape)
        centers = np.clip(centers, 0.02, 0.98)
        
        x0 = centers.flatten()
        
        # --- Hybrid Optimization ---
        # 1. Nelder-Mead: Robust for non-smooth objectives
        res_nm = minimize(objective_function, x0, method='Nelder-Mead', 
                          options={'maxiter': 2000, 'xatol': 1e-9, 'fatol': 1e-9})
        
        # 2. Powell: Fast local refinement
        res_pw = minimize(objective_function, res_nm.x, method='Powell', 
                          options={'maxiter': 500, 'xtol': 1e-9, 'ftol': 1e-9})
        
        current_sum = -res_pw.fun
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = res_pw.x.reshape(-1, 2)
            best_radii = compute_packing_metrics(best_centers)
            
    # --- Final Validation & Safety Clipping ---
    # Ensure strict interior placement to satisfy validator tolerance
    best_centers = np.clip(best_centers, 1e-8, 1.0 - 1e-8)
    best_radii = compute_packing_metrics(best_centers)
    final_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, final_sum
