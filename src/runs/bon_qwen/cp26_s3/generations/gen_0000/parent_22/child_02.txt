import numpy as np

def run_packing():
    np.random.seed(42)
    n = 26
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.08)
    
    # Initialize positions in a grid with small random perturbation
    for i in range(n):
        centers[i] = [(i % 5) * 0.2 + 0.1, (i // 5) * 0.2 + 0.1]
    centers += np.random.randn(*centers.shape) * 0.002
    
    # Simulation parameters
    max_iter = 15000
    wall_k = 1000.0
    rep_k = 2000.0
    damping = 0.85
    growth = 1.0001
    
    velocities = np.zeros_like(centers)
    
    for _ in range(max_iter):
        forces = np.zeros_like(centers)
        
        # Boundary forces: push inward if too close to edges
        for i in range(n):
            r = radii[i]
            if centers[i, 0] < r:
                forces[i, 0] += wall_k * (r - centers[i, 0])
            if centers[i, 0] > 1 - r:
                forces[i, 0] -= wall_k * (centers[i, 0] - (1 - r))
            if centers[i, 1] < r:
                forces[i, 1] += wall_k * (r - centers[i, 1])
            if centers[i, 1] > 1 - r:
                forces[i, 1] -= wall_k * (centers[i, 1] - (1 - r))
                
        # Repulsion forces between circles
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                d2 = dx * dx + dy * dy
                dist = np.sqrt(d2)
                sum_r = radii[i] + radii[j]
                
                if dist < sum_r:
                    overlap = sum_r - dist
                    f = rep_k * overlap / (dist + 1e-9)
                    fx = f * dx / dist
                    fy = f * dy / dist
                    forces[i, 0] += fx
                    forces[i, 1] += fy
                    forces[j, 0] -= fx
                    forces[j, 1] -= fy
                    
        # Update velocities and positions
        velocities = damping * velocities + forces
        centers += velocities
        centers = np.clip(centers, 0, 1)
        
        # Gradually increase radii to maximize sum
        radii *= growth
        
    # Apply a small safety margin to ensure strict validity
    radii *= 0.995
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
