import numpy as np
from scipy.optimize import minimize

def compute_loss(params, N, mu):
    """
    Compute the negative sum of radii plus penalty terms for constraint violations.
    """
    centers = params[:2*N].reshape(N, 2)
    radii = params[2*N:]
    
    # Objective: maximize sum of radii -> minimize negative sum
    loss = -np.sum(radii)
    
    # Boundary penalties: circle must be within [0,1]^2
    # Violations occur if x < r, x > 1-r, y < r, y > 1-r
    v_x1 = np.maximum(0.0, radii - centers[:, 0])
    v_x2 = np.maximum(0.0, centers[:, 0] + radii - 1.0)
    v_y1 = np.maximum(0.0, radii - centers[:, 1])
    v_y2 = np.maximum(0.0, centers[:, 1] + radii - 1.0)
    loss += mu * np.sum(v_x1**2 + v_x2**2 + v_y1**2 + v_y2**2)
    
    # Overlap penalties: distance between centers must be >= sum of radii
    c1 = centers[:, np.newaxis, :]
    c2 = centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum((c1 - c2)**2, axis=2))
    r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    
    # Only consider upper triangle to avoid double counting
    v = np.maximum(0.0, r_sum - dists)
    mask = np.triu(np.ones((N, N), dtype=bool), k=1)
    loss += mu * np.sum((v * mask)**2)
    
    return loss

def run_packing():
    N = 26
    
    # 1. Initial configuration: Hexagonal lattice pattern
    init_centers = np.zeros((N, 2))
    idx = 0
    y = 0.1
    while y < 0.95 and idx < N:
        x = 0.1
        # Hexagonal offset for alternating rows
        offset = 0.05 if int((y - 0.1) / 0.15) % 2 == 1 else 0.0
        x += offset
        while x < 0.95 and idx < N:
            init_centers[idx] = [x, y]
            idx += 1
            x += 0.2
        y += 0.15
    # Fill any remaining positions if the grid was too sparse
    for k in range(idx, N):
        init_centers[k] = [0.2 + 0.15 * (k % 5), 0.2 + 0.15 * (k // 5)]
        
    init_radii = np.ones(N) * 0.08
    base_params = np.concatenate([init_centers.flatten(), init_radii])
    
    best_params = base_params.copy()
    best_sum_r = -1.0
    
    # Bounds for optimizer: centers in [0.001, 0.999], radii in [0.001, 0.5]
    bounds = [(0.001, 0.999)] * (2 * N) + [(0.001, 0.5)] * N
    
    # 2. Multi-start optimization with adaptive penalty
    for seed in range(7):
        rng = np.random.default_rng(seed)
        # Perturb initial guess
        curr_params = base_params + rng.normal(0, 0.008, size=base_params.shape)
        curr_params = np.clip(curr_params, 0.01, 0.4)
        curr_params[:2*N] = np.clip(curr_params[:2*N], 0.02, 0.98)
        curr_params[2*N:] = np.clip(curr_params[2*N:], 0.02, 0.3)
        
        # Increase penalty strength gradually
        for mu in [100, 500, 2000, 10000]:
            res = minimize(compute_loss, curr_params, args=(N, mu), method='L-BFGS-B', 
                           bounds=bounds, options={'maxiter': 2000, 'ftol': 1e-14, 'gtol': 1e-12, 'disp': False})
            curr_params = res.x
            
        curr_radii = curr_params[2*N:]
        curr_sum = np.sum(curr_radii)
        if curr_sum > best_sum_r:
            best_sum_r = curr_sum
            best_params = curr_params.copy()
            
    centers = best_params[:2*N].reshape(N, 2)
    radii = best_params[2*N:]
    
    # 3. Deterministic correction phase to guarantee strict validity
    # Push overlapping circles apart along their connecting line
    for _ in range(300):
        max_push = 0.0
        for i in range(N):
            for j in range(i + 1, N):
                d = np.sqrt(np.sum((centers[i] - centers[j])**2))
                r_sum = radii[i] + radii[j]
                if d < r_sum - 1e-10 and d > 1e-12:
                    push = (r_sum - d) / 2.0
                    dir_v = (centers[i] - centers[j]) / d
                    centers[i] += dir_v * push
                    centers[j] -= dir_v * push
                    max_push = push
        if max_push < 1e-11:
            break
            
    # Enforce boundary constraints strictly
    for i in range(N):
        r = radii[i]
        centers[i, 0] = np.clip(centers[i, 0], r, 1.0 - r)
        centers[i, 1] = np.clip(centers[i, 1], r, 1.0 - r)
        
    return centers, radii, np.sum(radii)
