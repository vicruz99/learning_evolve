def get_packing_penalty(vars, n=26):
    """
    Computes the objective value: -sum(radii) + penalty for constraint violations.
    vars: array of shape (n, 3) -> [x, y, r]
    """
    x = vars[:, 0]
    y = vars[:, 1]
    r = vars[:, 2]
    
    # Objective: maximize sum of radii -> minimize negative sum
    obj_val = -np.sum(r)
    
    penalty_weight = 5000.0
    
    # Boundary constraints:
    # Circle i must be inside [0,1]x[0,1].
    # r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    # Violation is max(0, r - bound)
    
    # Vectorized boundary checks
    # x, y, r are shape (n,)
    
    viol_x_neg = np.maximum(0, r - x)
    viol_x_pos = np.maximum(0, r - (1 - x))
    viol_y_neg = np.maximum(0, r - y)
    viol_y_pos = np.maximum(0, r - (1 - y))
    
    boundary_penalty = np.sum(viol_x_neg**2) + np.sum(viol_x_pos**2) + \
                       np.sum(viol_y_neg**2) + np.sum(viol_y_pos**2)
    
    # Overlap constraints:
    # r_i + r_j <= distance(i, j)
    # Violation is max(0, r_i + r_j - dist(i, j))
    
    # Compute pairwise distances
    # x[:, None] creates (n, 1), x[None, :] creates (1, n) -> broadcasting (n, n)
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist = np.sqrt(dx**2 + dy**2)
    
    # Compute r_i + r_j
    r_sum = r[:, None] + r[None, :]
    
    # Overlap amount (positive means overlap)
    # We only care about i < j to avoid double counting and self-interaction
    # dist[i, i] = 0, r_sum[i, i] = 2*r_i. This would be a huge violation if not masked.
    # We set diagonal to 0 overlap contribution or handle mask.
    np.fill_diagonal(dist, np.inf) # Distance to self is infinite, so no overlap
    
    overlap = r_sum - dist
    
    # We only sum upper triangle (i < j)
    # np.triu_indices(n, k=1) gives indices for upper triangle
    triu_indices = np.triu_indices(n, k=1)
    overlap_upper = overlap[triu_indices]
    
    overlap_violation = np.maximum(0, overlap_upper)
    overlap_penalty = np.sum(overlap_violation**2)
    
    total_penalty = penalty_weight * (boundary_penalty + overlap_penalty)
    
    return obj_val + total_penalty

    count = 0

                count += 1
            else:

def generate_hex_start(n=26):
    """Generate a hexagonal lattice starting configuration."""
    centers = []
    # Hex spacing roughly 0.2
    s = 0.21
    dy = s * np.sqrt(3) / 2
    
    y = s / 2 # Start with some margin
    row = 0
    
    while len(centers) < n:
        # Offset every other row
        offset = (s / 2) if row % 2 == 1 else 0
        x = s / 2 + offset
        
        while x < 1 - s/2 and len(centers) < n:
            centers.append([x, y, 0.09])
            x += s
        
        y += dy
        row += 1
        
    # Trim or pad if necessary
    return np.array(centers[:n])
