import numpy as np
import scipy.optimize as opt

def run_packing():
    n = 26
    
    # Helper function to generate initial hexagonal packing
    def generate_initial_centers(n_circles, r_init=0.09):
        centers = []
        # Hexagonal lattice parameters
        row_spacing = r_init * np.sqrt(3)
        col_spacing = 2 * r_init
        
        y = r_init
        row_idx = 0
        
        while len(centers) < n_circles:
            # Determine shift for this row
            shift = r_init if row_idx % 2 == 1 else 0
            
            x = r_init + shift
            while x + r_init <= 1.0 + 1e-9:
                centers.append([x, y])
                if len(centers) >= n_circles:
                    break
                x += col_spacing
            
            y += row_spacing
            row_idx += 1
            
        return np.array(centers[:n_circles])

    # Objective function: maximize sum of radii (minimize negative sum)
    def objective(vars):
        x = vars[:n]
        y = vars[n:2*n]
        r = vars[2*n:]
        return -np.sum(r)

    # Constraints
    constraints = []

    # Boundary constraints: 0 <= x - r, 1 >= x + r, etc.
    # x - r >= 0  =>  r - x <= 0
    # x + r <= 1  =>  x + r - 1 <= 0
    # y - r >= 0  =>  r - y <= 0
    # y + r <= 1  =>  y + r - 1 <= 0
    # r >= 0      =>  -r <= 0

    # We can group these or define them individually. 
    # For SLSQP, we pass a list of dictionaries.
    
    # 1. x - r >= 0  (lower bound x)
    constraints.append({'type': 'ineq', 'fun': lambda v: v[:n] - v[2*n:]})
    # 2. 1 - x - r >= 0  (upper bound x)
    constraints.append({'type': 'ineq', 'fun': lambda v: 1 - v[:n] - v[2*n:]})
    # 3. y - r >= 0  (lower bound y)
    constraints.append({'type': 'ineq', 'fun': lambda v: v[n:2*n] - v[2*n:]})
    # 4. 1 - y - r >= 0  (upper bound y)
    constraints.append({'type': 'ineq', 'fun': lambda v: 1 - v[n:2*n] - v[2*n:]})
    # 5. r >= 0
    constraints.append({'type': 'ineq', 'fun': lambda v: v[2*n:]})

    # 6. Non-overlap constraints: dist^2 >= (r_i + r_j)^2
    # (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
    # This is computationally expensive to define as separate constraints for SLSQP 
    # inside the loop, but necessary for accuracy.
    # We will define a single constraint function that returns an array of violations.
    # However, SLSQP accepts a function returning an array for 'ineq'.
    
    def non_overlap_constraint(v):
        x = v[:n]
        y = v[n:2*n]
        r = v[2*n:]
        violations = np.full((n * (n - 1) // 2,), 0.0)
        idx = 0
        for i in range(n):
            for j in range(i + 1, n):
                dist_sq = (x[i] - x[j])**2 + (y[i] - y[j])**2
                min_dist_sq = (r[i] + r[j])**2
                violations[idx] = dist_sq - min_dist_sq
                idx += 1
        return violations

    constraints.append({'type': 'ineq', 'fun': non_overlap_constraint})

    # Bounds for variables
    # x, y in [0, 1], r in [0, 0.5]
    bounds = [(0, 1)] * n + [(0, 1)] * n + [(0, 0.5)] * n

    best_sum_radii = -1.0
    best_vars = None

    # Run optimization multiple times with slight perturbations to avoid local minima
    for trial in range(5):
        # Generate initial guess
        init_centers = generate_initial_centers(n, r_init=0.09)
        
        # Add small random noise
        noise = np.random.uniform(-0.005, 0.005, size=init_centers.shape)
        init_centers = np.clip(init_centers + noise, 0.01, 0.99)
        
        init_radii = np.full(n, 0.08) # Start with a safe valid radius
        
        # Flatten initial variables
        x0 = np.concatenate([init_centers[:, 0], init_centers[:, 1], init_radii])
        
        try:
            res = opt.minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints, 
                               options={'maxiter': 500, 'ftol': 1e-9})
            
            if res.success and -res.fun > best_sum_radii:
                # Verify validity manually just in case
                x_sol = res.x[:n]
                y_sol = res.x[n:2*n]
                r_sol = res.x[2*n:]
                
                # Quick check for obvious invalidity
                valid = True
                if np.any(r_sol < -1e-9) or np.any(x_sol < -1e-9) or np.any(x_sol > 1+1e-9):
                    valid = False
                
                if valid:
                    best_sum_radii = -res.fun
                    best_vars = res.x.copy()
                    
        except Exception as e:
            print(f"Optimization failed in trial {trial}: {e}")
            continue

    if best_vars is None:
        # Fallback to a simple valid configuration if optimization fails
        centers = generate_initial_centers(n, r_init=0.05)
        radii = np.full(n, 0.05)
        best_vars = np.concatenate([centers[:, 0], centers[:, 1], radii])
        best_sum_radii = np.sum(radii)

    # Extract final solution
    final_centers = np.column_stack((best_vars[:n], best_vars[n:2*n]))
    final_radii = best_vars[2*n:]
    final_sum = np.sum(final_radii)
    
    # Ensure radii are non-negative (clip small negatives)
    final_radii = np.maximum(final_radii, 0.0)

    return final_centers, final_radii, final_sum
