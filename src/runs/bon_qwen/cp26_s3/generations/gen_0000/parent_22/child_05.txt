import numpy as np
from scipy.optimize import minimize

def compute_penalty(centers, radii, W=1e6):
    """
    Computes the penalty for boundary violations and circle overlaps.
    Uses squared max functions to ensure differentiability.
    """
    n = centers.shape[0]
    pen = 0.0
    
    # Boundary penalties: circles must stay within [0, 1]
    # x - r >= 0  =>  r - x <= 0
    pen += W * np.sum(np.maximum(0, radii - centers[:, 0])**2)
    # x + r <= 1  =>  r - (1 - x) <= 0
    pen += W * np.sum(np.maximum(0, radii - (1 - centers[:, 0]))**2)
    # y - r >= 0
    pen += W * np.sum(np.maximum(0, radii - centers[:, 1])**2)
    # y + r <= 1
    pen += W * np.sum(np.maximum(0, radii - (1 - centers[:, 1]))**2)

    # Overlap penalties: distance >= r_i + r_j
    ii, jj = np.triu_indices(n, k=1)
    diff = centers[ii] - centers[jj]
    dists = np.sqrt(np.sum(diff**2, axis=1))
    overlaps = np.maximum(0, radii[ii] + radii[jj] - dists)
    pen += W * np.sum(overlaps**2)
    
    return pen

def objective(vars):
    """
    Objective function to minimize: negative sum of radii + penalties.
    """
    n = 26
    centers = vars[:2*n].reshape(n, 2)
    radii = vars[2*n:]
    return -np.sum(radii) + compute_penalty(centers, radii)

def run_packing():
    n = 26
    
    # 1. Initialization: Perturbed grid for good spatial distribution

    cx, cy = np.meshgrid(cols, rows)

    # Add small noise to break symmetry and escape grid local minima
    init_centers += np.random.uniform(-0.02, 0.02, size=init_centers.shape)
    init_centers = np.clip(init_centers, 0.0, 1.0)
    init_radii = np.full(n, 0.04)
    
    x0 = np.concatenate([init_centers.ravel(), init_radii])
    
    # 2. Bounds: coordinates in [0, 1], radii in [0, 0.5]
    bounds = [(0.0, 1.0) for _ in range(2*n)] + [(0.0, 0.5) for _ in range(n)]
    
    # 3. Optimization
    res = minimize(
        objective, 
        x0, 
        method='L-BFGS-B', 
        bounds=bounds, 
        options={'maxiter': 3000, 'ftol': 1e-12, 'gtol': 1e-8}
    )

    centers_opt = res.x[:2*n].reshape(n, 2)
    radii_opt = res.x[2*n:]
    
    # Ensure strict non-negativity and clip to bounds for safety
    radii_opt = np.maximum(radii_opt, 0.0)
    centers_opt = np.clip(centers_opt, 0.0, 1.0)
    
    sum_radii = float(np.sum(radii_opt))
    return centers_opt, radii_opt, sum_radii
