import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def compute_objective(vars):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(vars[2::3])

def compute_jac_objective(vars):
    """Analytical Jacobian for the objective function."""
    jac = np.zeros_like(vars)
    jac[2::3] = -1.0
    return jac

def compute_constraints_bounds(vars):
    """Boundary constraints: circles must be inside [0,1]^2."""
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    return np.concatenate([
        x - r,
        1.0 - x - r,
        y - r,
        1.0 - y - r
    ])

def compute_constraints_overlap(vars):
    """Non-overlap constraints: pairwise distance >= sum of radii."""
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    n_constraints = N_CIRCLES * (N_CIRCLES - 1) // 2
    res = np.empty(n_constraints)
    idx = 0
    for i in range(N_CIRCLES):
        xi, yi, ri = x[i], y[i], r[i]
        for j in range(i + 1, N_CIRCLES):
            dx = xi - x[j]
            dy = yi - y[j]
            r_sum = ri + r[j]
            res[idx] = dx*dx + dy*dy - r_sum*r_sum
            idx += 1
    return res

def get_bounds():
    """Variable bounds: x,y in [0,1], r in [1e-8, 0.5]."""
    bounds = []
    for _ in range(N_CIRCLES):
        bounds.extend([(0.0, 1.0), (0.0, 1.0), (1e-8, 0.5)])
    return bounds

def get_initial_guess(seed):
    """Generate a feasible initial configuration using a perturbed hexagonal lattice."""
    rng = np.random.default_rng(seed)
    pts = []
    rows, cols = 6, 5
    sx, sy = 1.0 / (cols + 1), 1.0 / (rows + 1)
    
    for j in range(rows):
        for i in range(cols):
            if len(pts) >= N_CIRCLES:
                break
            ox = sx * 0.5 if j % 2 == 1 else 0.0
            pts.append(((i + 1 + ox) * sx, (j + 1) * sy))
        if len(pts) >= N_CIRCLES:
            break

    xs = np.array([p[0] for p in pts[:N_CIRCLES]]) + rng.uniform(-0.03, 0.03, N_CIRCLES)
    ys = np.array([p[1] for p in pts[:N_CIRCLES]]) + rng.uniform(-0.03, 0.03, N_CIRCLES)
    xs = np.clip(xs, 0.05, 0.95)
    ys = np.clip(ys, 0.05, 0.95)
    rs = np.full(N_CIRCLES, 0.04)

    vars = np.empty(3 * N_CIRCLES)
    vars[0::3] = xs
    vars[1::3] = ys
    vars[2::3] = rs
    return vars

def run_packing():
    """Main function to solve the packing problem."""
    bounds = get_bounds()
    constraints = [
        {'type': 'ineq', 'fun': compute_constraints_bounds},
        {'type': 'ineq', 'fun': compute_constraints_overlap}
    ]
    
    best_vars = None
    best_sum_r = -1.0

    # Run multiple optimizations with different seeds to escape local minima
    for seed in range(10):
        x0 = get_initial_guess(seed)
        try:
            res = minimize(
                compute_objective, 
                x0, 
                method='SLSQP', 
                jac=compute_jac_objective,
                bounds=bounds, 
                constraints=constraints,
                options={'maxiter': 2000, 'ftol': 1e-12, 'disp': False}
            )
            
            if res.success:
                curr_sum_r = -res.fun
                # Verify constraints are satisfied within numerical tolerance
                if (np.all(compute_constraints_bounds(res.x) >= -1e-9) and 
                    np.all(compute_constraints_overlap(res.x) >= -1e-9)):
                    if curr_sum_r > best_sum_r:
                        best_sum_r = curr_sum_r
                        best_vars = res.x.copy()
        except Exception:
            continue

    # Fallback if optimization fails
    if best_vars is None:
        best_vars = get_initial_guess(0)

    # Extract and clean results
    centers = np.column_stack((best_vars[0::3], best_vars[1::3]))
    radii = best_vars[2::3]
    radii = np.maximum(radii, 0.0)
    
    # Ensure strict boundary compliance
    for i in range(N_CIRCLES):
        r = radii[i]
        centers[i, 0] = np.clip(centers[i, 0], r, 1.0 - r)
        centers[i, 1] = np.clip(centers[i, 1], r, 1.0 - r)

    return centers, radii, float(np.sum(radii))
