    n = centers.shape[0]

        if y < -tolerance or y > 1 + tolerance: return False
        if x - r < -tolerance or x + r > 1 + tolerance: return False
        if y - r < -tolerance or y + r > 1 + tolerance: return False

        for j in range(i + 1, n):
