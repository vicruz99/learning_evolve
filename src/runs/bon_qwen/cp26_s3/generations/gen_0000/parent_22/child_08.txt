import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Number of restarts to find a good local optimum
    n_restarts = 30
    
    # Penalty weight
    lambda_penalty = 10000.0
    
    for _ in range(n_restarts):
        # --- Initialization ---
        # Generate a hexagonal grid pattern
        pts = []
        # Estimated radius for 26 circles is around 0.1
        # Spacing for hex grid
        r_est = 0.1
        dy = r_est * np.sqrt(3) # Vertical spacing between rows
        dx = 2 * r_est         # Horizontal spacing
        
        # Fill grid
        y = r_est + 0.01 # Start slightly inside
        row = 0
        while y < 1.0 - 0.01:
            x = r_est + 0.01 if row % 2 == 0 else r_est + 0.01 + dx/2
            while x < 1.0 - 0.01:
                pts.append([x, y])
                x += dx
            y += dy
            row += 1
            
        # If grid didn't produce enough points, fill with random
        if len(pts) < n:
            while len(pts) < n:
                pts.append([np.random.uniform(0.1, 0.9), np.random.uniform(0.1, 0.9)])
        
        # Take exactly n points
        centers_init = np.array(pts[:n])
        
        # Add random perturbation to break symmetry and escape local minima
        noise_scale = 0.05
        centers_init += np.random.normal(0, noise_scale, centers_init.shape)
        # Clip to valid range for centers [0.05, 0.95] to keep initial radii valid
        centers_init = np.clip(centers_init, 0.05, 0.95)
        
        # Initial radii small to ensure valid start
        radii_init = np.ones(n) * 0.02
        
        x0 = np.concatenate([centers_init.flatten(), radii_init])
        
        # --- Bounds ---
        # Centers must be in [0, 1], Radii in [0, 0.5]
        bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
        
        # --- Objective Function ---
        def objective(vars):
            c = vars[:2*n].reshape(n, 2)
            r = vars[2*n:]
            
            # We want to maximize sum(r), so minimize -sum(r)
            val = -np.sum(r)
            
            penalty = 0.0
            
            # 1. Boundary Penalties
            # Circle i must satisfy: r <= x <= 1-r  =>  x-r >= 0  and  1-x-r >= 0
            # Violations: max(0, r - x) and max(0, x + r - 1)
            # We use squared penalty for smooth gradients near boundary
            
            # Vectorized boundary checks
            # Left wall: x >= r -> violation r - x
            v_left = np.maximum(0, r - c[:, 0])
            # Right wall: x <= 1-r -> violation x + r - 1
            v_right = np.maximum(0, c[:, 0] + r - 1.0)
            # Bottom wall: y >= r -> violation r - y
            v_bottom = np.maximum(0, r - c[:, 1])
            # Top wall: y <= 1-r -> violation y + r - 1
            v_top = np.maximum(0, c[:, 1] + r - 1.0)
            
            pen_boundary = np.sum(v_left**2 + v_right**2 + v_bottom**2 + v_top**2)
            penalty += pen_boundary
            
            # 2. Overlap Penalties
            # Circle i and j must satisfy: dist >= r_i + r_j
            # Violation: max(0, r_i + r_j - dist)
            
            # O(N^2) loop, N=26 is small enough
            for i in range(n):
                ri = r[i]
                xi = c[i, 0]
                yi = c[i, 1]
                for j in range(i + 1, n):
                    # Calculate distance squared first to avoid sqrt if not needed?
                    # But we need dist for violation
                    dx = xi - c[j, 0]
                    dy = yi - c[j, 1]
                    dist = np.hypot(dx, dy)
                    
                    sum_r = ri + r[j]
                    if sum_r > dist:
                        penalty += (sum_r - dist)**2
            
            return val + lambda_penalty * penalty

        # --- Optimization ---
        try:
            res = minimize(objective, x0, method='L-BFGS-B', bounds=bounds, 
                           options={'maxiter': 1000, 'ftol': 1e-12, 'gtol': 1e-8})
            
            centers_cand = res.x[:2*n].reshape(n, 2)
            radii_cand = res.x[2*n:]
            
            # --- Validation ---
            # Check if solution is valid
            is_valid = True
            
            # Check boundary
            for i in range(n):
                x, y = centers_cand[i]
                r = radii_cand[i]
                if x - r < -1e-9 or x + r > 1 + 1e-9 or y - r < -1e-9 or y + r > 1 + 1e-9:
                    is_valid = False
                    break
            
            # Check overlap
            if is_valid:
                for i in range(n):
                    xi, yi = centers_cand[i]
                    ri = radii_cand[i]
                    for j in range(i + 1, n):
                        xj, yj = centers_cand[j]
                        rj = radii_cand[j]
                        dist = np.hypot(xi - xj, yi - yj)
                        if dist < ri + rj - 1e-9:
                            is_valid = False
                            break
                    if not is_valid:
                        break
            
            current_sum = np.sum(radii_cand)
            
            if is_valid and current_sum > best_sum:
                best_sum = current_sum
                best_centers = centers_cand.copy()
                best_radii = radii_cand.copy()
                
        except Exception as e:
            # Skip failed iterations
            pass

    # Fallback if no valid solution found (should not happen with good init)
    if best_centers is None:
        # Generate a trivial valid solution: small circles in a grid
        centers = np.zeros((n, 2))
        radii = np.zeros(n)
        side = int(np.ceil(np.sqrt(n)))
        spacing = 1.0 / (side + 1)
        count = 0
        for i in range(1, side + 1):
            for j in range(1, side + 1):
                if count < n:
                    centers[count] = [i * spacing, j * spacing]
                    radii[count] = spacing / 2 * 0.9 # Safe radius
                    count += 1
        # Fill remaining if any
        while count < n:
             centers[count] = [0.5, 0.5] # Will overlap, but radius 0
             radii[count] = 0.0
             count += 1
        best_centers = centers
        best_radii = radii
        best_sum = np.sum(radii)

    return best_centers, best_radii, best_sum
