import numpy as np
from scipy.optimize import minimize, differential_evolution


def compute_max_equal_radius(centers):
    n = centers.shape[0]
    r = 1.0
    for i in range(n):
        x, y = centers[i]
        r = min(r, x, 1.0 - x, y, 1.0 - y)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            r = min(r, dist / 2.0)
    return r


def obj_equal(centers_flat):
    centers = centers_flat.reshape(-1, 2)
    return -compute_max_equal_radius(centers)


def generate_hex(arrangement, n=26):
    r = 0.05
    centers = []
    y = r
    for i, cnt in enumerate(arrangement):
        if cnt <= 1:
            xs = [0.5]
        else:
            if i % 2 == 0:
                xs = [r + j * 2 * r for j in range(cnt)]
            else:
                xs = [2 * r + j * 2 * r for j in range(cnt)]
        for x in xs:
            centers.append([x, y])
        y += r * np.sqrt(3)
    centers = np.array(centers[:n])
    return centers


def scale_fit(centers, margin=0.05):
    mn = centers.min(0) - margin
    mx = centers.max(0) + margin
    w = mx - mn
    s = 1.0 / max(w[0], w[1])
    c = (centers - mn) * s
    c[:, 0] += 0.5 - c[:, 0].mean()
    c[:, 1] += 0.5 - c[:, 1].mean()
    return c


def try_equal_optimization(arrangement, n=26):
    centers = scale_fit(generate_hex(arrangement, n))
    res = minimize(obj_equal, centers.flatten(), method='Nelder-Mead',
                   options={'maxiter': 150000, 'xatol': 1e-12, 'fatol': 1e-14})
    opt_centers = res.x.reshape(-1, 2)
    opt_r = compute_max_equal_radius(opt_centers)
    opt_sum = n * opt_r
    return opt_centers, np.full(n, opt_r), opt_sum


def try_de_optimization(n=26, seed=0):
    rng = np.random.RandomState(seed)
    bounds = [(0.05, 0.95)] * (2 * n)

    def de_obj(cef):
        return obj_equal(cef)

    result = differential_evolution(de_obj, bounds, seed=seed, maxiter=5000,
                                     popsize=30, mutation=(0.5, 1.5),
                                     recombination=0.9, atol=1e-12,
                                     workers=1, updating='immediate',
                                     polish=True)
    opt_centers = result.x.reshape(-1, 2)
    opt_r = compute_max_equal_radius(opt_centers)
    opt_sum = n * opt_r
    return opt_centers, np.full(n, opt_r), opt_sum


def grow_unequal(centers, radii, n=26, max_iter=5000):
    centers = centers.copy()
    radii = radii.copy()

    for it in range(max_iter):
        radii += 1e-5

        resolved = False
        for _ in range(300):
            any_overlap = False
            for i in range(n):
                for j in range(i + 1, n):
                    dx = centers[j, 0] - centers[i, 0]
                    dy = centers[j, 1] - centers[i, 1]
                    d = np.sqrt(dx * dx + dy * dy)
                    md = radii[i] + radii[j]
                    if d < md and d > 1e-14:
                        any_overlap = True
                        f = (md - d) / d * 0.5
                        centers[i, 0] -= dx * f
                        centers[i, 1] -= dy * f
                        centers[j, 0] += dx * f
                        centers[j, 1] += dy * f

            for i in range(n):
                for k in range(2):
                    if centers[i, k] - radii[i] < 0:
                        centers[i, k] = radii[i]
                    if centers[i, k] + radii[i] > 1.0:
                        centers[i, k] = 1.0 - radii[i]

            if not any_overlap:
                resolved = True
                break

        if not resolved:
            radii -= 2e-5
            break

    return centers, radii


def grow_unequal_with_perturbation(centers, radii, n=26, seed=0):
    rng = np.random.RandomState(seed)
    centers = centers.copy()
    radii = radii.copy()

    # Perturb radii to break symmetry
    perturbation = rng.uniform(-0.002, 0.002, size=n)
    radii += perturbation
    radii = np.maximum(radii, 0.01)

    return grow_unequal(centers, radii, n, max_iter=8000)
