import numpy as np
from scipy.optimize import minimize

def run_packing():
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    Returns centers, radii, and sum_radii.
    """
    n_circles = 26
    
    # 1. Initialization
    # Create a 5x5 grid for the first 25 circles
    # and place the 26th circle in a gap.
    x_coords = np.array([0.1, 0.3, 0.5, 0.7, 0.9])
    y_coords = np.array([0.1, 0.3, 0.5, 0.7, 0.9])
    
    centers = np.zeros((n_circles, 2))
    radii = np.zeros(n_circles)
    
    idx = 0
    # Fill 5x5 grid
    for i in range(5):
        for j in range(5):
            centers[idx] = [x_coords[j], y_coords[i]]
            radii[idx] = 0.09  # Initial radius, slightly less than max possible 0.1
            idx += 1
            
    # Place 26th circle in a void between (0.1,0.1), (0.3,0.1), etc.
    # Center approx (0.2, 0.2)
    centers[idx] = [0.2, 0.2]
    radii[idx] = 0.05  # Small radius to fit in gap initially
    
    # Convert to 1D vector for optimizer: [x1, y1, r1, x2, y2, r2, ...]
    # Order: x, y, r for each circle
    initial_vars = np.zeros(3 * n_circles)
    for i in range(n_circles):
        initial_vars[3*i] = centers[i, 0]
        initial_vars[3*i+1] = centers[i, 1]
        initial_vars[3*i+2] = radii[i]

    # 2. Define Objective and Constraints
    
    def objective(vars_1d):
        # Maximize sum of radii -> Minimize negative sum
        return -np.sum(vars_1d[2::3])

    def constraints(vars_1d):
        constraints_list = []
        
        # Reshape to 2D for easier access
        # cols: 0=x, 1=y, 2=r
        data = vars_1d.reshape((n_circles, 3))
        xs = data[:, 0]
        ys = data[:, 1]
        rs = data[:, 2]
        
        # Boundary constraints: r <= x <= 1-r  =>  x - r >= 0,  1 - r - x >= 0
        # y - r >= 0,  1 - r - y >= 0
        
        # Vectorized boundary constraints
        # x >= r
        constraints_list.append(xs - rs)
        # x <= 1 - r  =>  x + r <= 1
        constraints_list.append(1.0 - xs - rs)
        # y >= r
        constraints_list.append(ys - rs)
        # y <= 1 - r
        constraints_list.append(1.0 - ys - rs)
        
        # Non-overlap constraints: dist(i,j) >= r_i + r_j
        # sqrt((xi-xj)^2 + (yi-yj)^2) >= ri + rj
        # To avoid sqrt in constraint function (which can be noisy), 
        # we can square both sides, but need to be careful with signs.
        # Since radii >= 0 and dist >= 0, D^2 >= (r_i+r_j)^2 is valid.
        # However, for SLSQP, smooth constraints are preferred.
        # Let's stick to distance >= sum_radii.
        
        # We will generate constraints for all pairs i < j
        # This is O(N^2), but N=26 is small enough.
        
        # Construct matrices for vectorized distance calculation
        # X[i] - X[j]
        # Using broadcasting
        # xs[:, None] - xs[None, :] gives matrix of differences
        
        dx = xs[:, None] - xs[None, :]
        dy = ys[:, None] - ys[None, :]
        d2 = dx**2 + dy**2
        dists = np.sqrt(d2)
        
        # Sum of radii matrix
        sum_rs = rs[:, None] + rs[None, :]
        
        # Constraint: dists - sum_rs >= 0
        # We only need upper triangle (i < j) to avoid duplicates and self (0)
        # Flatten the matrix and select upper triangle
        flat_constraints = dists - sum_rs
        
        # Mask for upper triangle
        mask = np.triu(np.ones((n_circles, n_circles), dtype=bool), k=1)
        upper_tri_constraints = flat_constraints[mask]
        
        constraints_list.append(upper_tri_constraints)
        
        # Concatenate all constraints into a single array
        return np.concatenate(constraints_list)

    # Bounds for variables
    # x in [0, 1], y in [0, 1], r in [0, 0.5]
    bounds = []
    for i in range(n_circles):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((1e-6, 0.5)) # r (keep strictly positive to avoid log issues if any, though not used)

    # 3. Optimization
    # Run optimization
    result = minimize(objective, initial_vars, method='SLSQP', bounds=bounds, 
                      constraints={'type': 'ineq', 'fun': constraints},
                      options={'maxiter': 1000, 'ftol': 1e-9})
    
    # 4. Post-processing
    best_vars = result.x
    best_centers = np.zeros((n_circles, 2))
    best_radii = np.zeros(n_circles)
    
    for i in range(n_circles):
        best_centers[i, 0] = best_vars[3*i]
        best_centers[i, 1] = best_vars[3*i+1]
        best_radii[i] = best_vars[3*i+2]
        
    sum_radii = np.sum(best_radii)
    
    return best_centers, best_radii, sum_radii
