import numpy as np
import scipy.optimize as opt
import math

def solve_radii_lp(centers):
    """
    Given fixed centers, solve for radii that maximize sum of radii.
    Constraints:
    1. r_i >= 0
    2. r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i (boundary)
    3. r_i + r_j <= dist(i, j) (non-overlap)
    
    This is an LP problem: Maximize sum(r_i)
    """
    n = centers.shape[0]
    
    # Objective: Maximize sum(r) => Minimize -sum(r)
    c_obj = -np.ones(n)
    
    # Inequality constraints A_ub * r <= b_ub
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    # Actually, these are upper bounds on variables. 
    # linprog handles bounds directly.
    
    bounds = []
    for i in range(n):
        x, y = centers[i]
        max_r = min(x, 1 - x, y, 1 - y)
        # Radius cannot be negative
        bounds.append((0, max_r))
        
    # Pairwise distance constraints: r_i + r_j <= dist_ij
    # This translates to 1*r_i + 1*r_j <= dist_ij
    # We can add these as rows in A_ub
    
    # Precompute distances
    # dist_matrix[i, j]
    dist_matrix = np.linalg.norm(centers[:, np.newaxis] - centers[np.newaxis, :], axis=2)
    
    # We only need i < j constraints
    # Construct sparse or dense matrix. For n=26, dense is fine.
    # Number of constraints: n*(n-1)/2
    num_constraints = n * (n - 1) // 2
    
    A_ub = np.zeros((num_constraints, n))
    b_ub = np.zeros(num_constraints)
    
    row_idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[row_idx, i] = 1.0
            A_ub[row_idx, j] = 1.0
            b_ub[row_idx] = dist_matrix[i, j]
            row_idx += 1
            
    try:
        res = opt.linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
        else:
            # Fallback if LP fails (shouldn't happen usually)
            return np.zeros(n)
    except Exception:
        return np.zeros(n)

def get_sum_radii(centers):
    """Helper to get sum of radii for a given configuration"""
    radii = solve_radii_lp(centers)
    return np.sum(radii), radii
