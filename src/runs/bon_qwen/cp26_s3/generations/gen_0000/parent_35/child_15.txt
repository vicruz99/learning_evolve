import numpy as np
from scipy.optimize import minimize

def packing_objective(vars, n, w):
    """
    Objective function for circle packing optimization.
    Minimizes negative sum of radii plus penalty for constraint violations.
    """
    xs = vars[:n]
    ys = vars[n:2*n]
    rs = vars[2*n:]
    
    # Primary goal: maximize sum of radii
    score = -np.sum(rs)
    penalty = 0.0
    
    # Boundary constraints: circles must stay within [0,1]x[0,1]
    for i in range(n):
        if rs[i] > xs[i]:
            penalty += w * (rs[i] - xs[i])**2
        if rs[i] > 1.0 - xs[i]:
            penalty += w * (rs[i] - (1.0 - xs[i]))**2
        if rs[i] > ys[i]:
            penalty += w * (rs[i] - ys[i])**2
        if rs[i] > 1.0 - ys[i]:
            penalty += w * (rs[i] - (1.0 - ys[i]))**2
            
    # Non-overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(xs[i] - xs[j], ys[i] - ys[j])
            overlap = rs[i] + rs[j] - dist
            if overlap > 0.0:
                penalty += w * overlap**2
                
    return score + penalty
