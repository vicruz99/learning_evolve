import numpy as np
from scipy.optimize import minimize

def compute_objective(vars, lam):
    """
    Compute the objective function: negative sum of radii plus penalty for overlaps and boundary violations.
    vars: array of shape (78,) containing [x1, y1, ..., x26, y26, r1, ..., r26]
    lam: penalty weight
    """
    n = 26
    centers = vars[:n*2].reshape(n, 2)
    radii = vars[n*2:]
    
    penalty = 0.0
    
    # Vectorized pairwise distance computation
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dist, np.inf)
    
    r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    overlap = r_sum - dist
    mask = overlap > 0
    if np.any(mask):
        penalty += lam * np.sum(overlap[mask]**2)
        
    # Boundary violation penalty
    d_min = np.minimum(centers, 1 - centers)
    dist_bound = np.min(d_min, axis=1)
    bound_viol = radii - dist_bound
    if np.any(bound_viol > 0):
        penalty += lam * np.sum(bound_viol**2)
        
    return -np.sum(radii) + penalty

def run_packing():
    np.random.seed(42)
    n = 26
    
    # Initialize centers in a slightly perturbed grid pattern
    centers = np.zeros((n, 2))
    idx = 0
    for i in range(5):
        for j in range(5):
            centers[idx] = [0.07 + i*0.19, 0.07 + j*0.19]
            idx += 1
    centers[idx] = [0.5, 0.5] # 26th circle in the center
    
    # Initial small radii
    radii = np.full(n, 0.02)
    
    # Force-directed layout to spread circles and grow radii
    dt = 0.01
    steps = 5000
    
    for step in range(steps):
        forces = np.zeros_like(centers)
        
        # Compute repulsive forces between overlapping circles
        for i in range(n):
            for j in range(i+1, n):
                d = np.linalg.norm(centers[i] - centers[j])
                if d < 1e-9:
                    d = 1e-9
                    diff = np.random.randn(2) * 1e-5
                else:
                    diff = centers[i] - centers[j]
                
                r_sum = radii[i] + radii[j]
                if d < r_sum:
                    # Force proportional to overlap
                    f = (diff / d) * (r_sum - d)
                    forces[i] += f
                    forces[j] -= f
                    
        # Update centers
        centers += forces * dt
        
        # Hard clamp to boundaries to maintain feasibility during simulation
        for i in range(n):
            r = radii[i]
            centers[i, 0] = max(r, min(1-r, centers[i, 0]))
            centers[i, 1] = max(r, min(1-r, centers[i, 1]))
            
        # Grow radii multiplicatively
        radii *= 1.00005
        
    # Refine using scipy.optimize with increasing penalty weight
    x0 = np.concatenate([centers.flatten(), radii])
    bounds = [(0, 1)]*52 + [(0, 0.5)]*26
    
    lam = 10.0
    for _ in range(4):
        res = minimize(compute_objective, x0, args=(lam,), bounds=bounds, 
                       method='L-BFGS-B', options={'maxiter': 1500, 'ftol': 1e-12})
        x0 = res.x
        lam *= 4.0
        
    centers_final = x0[:52].reshape(26, 2)
    radii_final = x0[52:]
    
    # Post-processing: scale down radii uniformly to strictly satisfy validation constraints
    scale = 1.0
    while True:
        valid = True
        r_test = radii_final * scale
        
        # Check boundary constraints
        for i in range(n):
            if r_test[i] < 0:
                valid = False
                break
            if (centers_final[i, 0] - r_test[i] < -1e-12 or 
                centers_final[i, 0] + r_test[i] > 1 + 1e-12 or
                centers_final[i, 1] - r_test[i] < -1e-12 or 
                centers_final[i, 1] + r_test[i] > 1 + 1e-12):
                valid = False
                break
                
        if valid:
            # Check pairwise overlap constraints
            for i in range(n):
                for j in range(i+1, n):
                    d = np.linalg.norm(centers_final[i] - centers_final[j])
                    if d < r_test[i] + r_test[j] - 1e-12:
                        valid = False
                        break
                if not valid:
                    break
                    
        if valid:
            radii_final = r_test
            break
            
        scale *= 0.995
        if scale < 0.5:
            # Fallback to ensure valid output
            radii_final *= scale
            break
            
    return centers_final, radii_final, np.sum(radii_final)
