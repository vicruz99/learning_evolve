import numpy as np
from scipy.optimize import minimize

N = 26

def objective_func(z):
    """Minimize negative sum of radii to maximize total radius."""
    return -np.sum(z[2::3])

def constraint_func(z):
    """
    Returns constraint values. All must be >= 0 for feasibility.
    Order: [x-r, 1-x-r, y-r, 1-y-r, pairwise_dist^2 - (r_i+r_j)^2]
    """
    x = z[0::3]
    y = z[1::3]
    r = z[2::3]

    # Boundary constraints (4 per circle)
    c_boundary = np.concatenate([
        x - r,
        1.0 - x - r,
        y - r,
        1.0 - y - r
    ])

    # Pairwise non-overlap constraints
    # Vectorized computation for efficiency
    X = x[:, None]
    Y = y[:, None]
    R = r[:, None]

    dist_sq = (X - X.T)**2 + (Y - Y.T)**2
    r_sum_sq = (R + R.T)**2

    # Extract upper triangular part (i < j)
    triu_idx = np.triu_indices(N, k=1)
    c_pairwise = dist_sq[triu_idx] - r_sum_sq[triu_idx]

    return np.concatenate([c_boundary, c_pairwise])

def run_packing():
    # 1. Initial Configuration: Perturbed hexagonal grid
    # Hexagonal packing is dense and provides a strong starting point
    xs = np.linspace(0.2, 0.8, 6)
    ys = np.linspace(0.2, 0.8, 5)
    centers = []
    count = 0
    for j in range(5):
        offset = (xs[1] - xs[0]) / 2.0 if j % 2 == 1 else 0.0
        for i in range(6):
            if count < N:
                centers.append([xs[i] + offset, ys[j]])
                count += 1
                
    centers = np.array(centers[:N])
    z0 = np.zeros(3 * N)
    z0[0::3] = centers[:, 0]
    z0[1::3] = centers[:, 1]
    z0[2::3] = 0.04  # Small initial radius to ensure feasibility
    
    # 2. Bounds and Constraints Setup
    bnds = [(0.0, 1.0)] * N + [(0.0, 1.0)] * N + [(0.0, 0.25)] * N
    cons = {'type': 'ineq', 'fun': constraint_func}
    
    # 3. Optimization
    res = minimize(
        objective_func,
        z0,
        method='SLSQP',
        bounds=bnds,
        constraints=cons,
        options={'maxiter': 5000, 'ftol': 1e-12, 'disp': False}
    )
    
    z_opt = res.x
    centers_opt = np.column_stack((z_opt[0::3], z_opt[1::3]))
    radii_opt = z_opt[2::3]
    
    # 4. Feasibility Check & Correction
    # Guarantee strict satisfaction of constraints within numerical tolerance
    min_gap = 1.0
    for i in range(N):
        x, y, r = centers_opt[i, 0], centers_opt[i, 1], radii_opt[i]
        min_gap = min(min_gap, x - r, 1.0 - x - r, y - r, 1.0 - y - r)
        
    for i in range(N):
        for j in range(i + 1, N):
            dx = centers_opt[i, 0] - centers_opt[j, 0]
            dy = centers_opt[i, 1] - centers_opt[j, 1]
            dist = np.sqrt(dx*dx + dy*dy)
            min_gap = min(min_gap, dist - radii_opt[i] - radii_opt[j])
            
    if min_gap < 0:
        # Slightly shrink radii to resolve marginal violations
        safe_scale = 1.0 + min_gap / (np.mean(radii_opt) + 1e-9)
        radii_opt *= max(safe_scale, 0.999999)
        
    sum_radii = float(np.sum(radii_opt))
    return centers_opt, radii_opt, sum_radii
