import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def _initial_guess(seed):
    """Generate a perturbed grid layout as initial guess."""
    rng = np.random.RandomState(seed)
    xs = np.linspace(0.15, 0.85, 6)
    ys = np.linspace(0.15, 0.85, 5)
    coords = []
    for y in ys:
        for x in xs:
            coords.append([x, y])
            if len(coords) >= N_CIRCLES:
                break
        if len(coords) >= N_CIRCLES:
            break
    coords = np.array(coords[:N_CIRCLES])
    # Add random noise to break symmetry and avoid grid locking
    coords += rng.uniform(-0.03, 0.03, size=coords.shape)
    coords = np.clip(coords, 0.05, 0.95)
    r_init = np.full(N_CIRCLES, 0.04)
    return np.concatenate([coords[:, 0], coords[:, 1], r_init])

def _objective(vars):
    """Objective: maximize sum of radii -> minimize negative sum."""
    return -np.sum(vars[2*N_CIRCLES:3*N_CIRCLES])

def _constraints(vars):
    """Define inequality constraints: boundaries and non-overlap."""
    x = vars[:N_CIRCLES]
    y = vars[N_CIRCLES:2*N_CIRCLES]
    r = vars[2*N_CIRCLES:3*N_CIRCLES]

    cons = []
    # Boundary constraints: circle inside [0,1]x[0,1]
    cons.extend(x - r)
    cons.extend(1.0 - x - r)
    cons.extend(y - r)
    cons.extend(1.0 - y - r)

    # Non-overlap constraints: dist(i,j) >= r_i + r_j
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist = np.sqrt(dx**2 + dy**2)
    sum_r = r[:, None] + r[None, :]

    # Only upper triangle needed due to symmetry
    idx_i, idx_j = np.triu_indices(N_CIRCLES, k=1)
    overlap = dist[idx_i, idx_j] - sum_r[idx_i, idx_j]
    cons.extend(overlap)

    return np.array(cons)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Run optimization to pack 26 circles in a unit square."""
    best_vars = None
    best_val = np.inf
    
    # Bounds for variables: x,y in [0,1], r in [1e-6, 0.5]
    bounds = [(0.0, 1.0)] * N_CIRCLES + [(0.0, 1.0)] * N_CIRCLES + [(1e-6, 0.5)] * N_CIRCLES
    cons = {'type': 'ineq', 'fun': _constraints}

    # Multiple restarts to find global optimum
    for i in range(50):
        v0 = _initial_guess(seed=i)
        res = minimize(_objective, v0, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 1500, 'ftol': 1e-10, 'disp': False})
        if res.success and res.fun < best_val:
            best_val = res.fun
            best_vars = res.x.copy()

    # Fallback if optimization fails unexpectedly
    if best_vars is None:
        best_vars = _initial_guess(seed=0)

    x = best_vars[:N_CIRCLES]
    y = best_vars[N_CIRCLES:2*N_CIRCLES]
    r = best_vars[2*N_CIRCLES:3*N_CIRCLES]
    
    # Ensure strict non-negativity and valid range
    r = np.maximum(r, 1e-9)
    centers = np.column_stack((x, y))
    
    return centers, r, float(np.sum(r))
