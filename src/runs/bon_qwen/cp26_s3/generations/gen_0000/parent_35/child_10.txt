import numpy as np
from scipy.optimize import minimize


def objective_func(x, n):
    radii = x[2 * n:]
    return -np.sum(radii)


def non_overlap_constraint(x, args):
    i, j, n = args
    xi, yi = x[2 * i], x[2 * i + 1]
    xj, yj = x[2 * j], x[2 * j + 1]
    ri, rj = x[2 * n + i], x[2 * n + j]
    dist = np.sqrt((xi - xj) ** 2 + (yi - yj) ** 2)
    return dist - ri - rj


def left_bound_constraint(x, args):
    i, n = args
    return x[2 * i] - x[2 * n + i]


def right_bound_constraint(x, args):
    i, n = args
    return 1.0 - x[2 * i] - x[2 * n + i]


def bottom_bound_constraint(x, args):
    i, n = args
    return x[2 * i + 1] - x[2 * n + i]


def top_bound_constraint(x, args):
    i, n = args
    return 1.0 - x[2 * i + 1] - x[2 * n + i]


def build_constraints(n):
    constraints = []
    for i in range(n):
        for j in range(i + 1, n):
            constraints.append(
                {
                    "type": "ineq",
                    "fun": non_overlap_constraint,
                    "args": (i, j, n),
                }
            )
    for i in range(n):
        constraints.append(
            {"type": "ineq", "fun": left_bound_constraint, "args": (i, n)}
        )
        constraints.append(
            {"type": "ineq", "fun": right_bound_constraint, "args": (i, n)}
        )
        constraints.append(
            {"type": "ineq", "fun": bottom_bound_constraint, "args": (i, n)}
        )
        constraints.append(
            {"type": "ineq", "fun": top_bound_constraint, "args": (i, n)}
        )
    return constraints


def hexagonal_init(n, spacing_scale=1.0):
    centers = []
    row_configs = [6, 5, 6, 5, 4]
    y_spacing = 0.18 * spacing_scale
    x_spacing = 0.17 * spacing_scale
    y = 0.13
    for i, size in enumerate(row_configs):
        x_start = (1.0 - (size - 1) * x_spacing) / 2.0
        offset = (x_spacing / 2.0) * (i % 2)
        for j in range(size):
            if len(centers) >= n:
                break
            x = x_start + j * x_spacing + offset
            centers.append([x, y])
        y += y_spacing * np.sqrt(3.0) / 2.0
    centers = centers[:n]
    while len(centers) < n:
        centers.append([0.5, 0.5])
    centers = np.array(centers)
    radii = np.full(n, 0.03)
    return np.concatenate([centers.flatten(), radii])


def physics_optimize(centers, radii, n, iterations=3000):
    for iteration in range(iterations):
        forces = np.zeros((n, 2))
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[j, 0] - centers[i, 0]
                dy = centers[j, 1] - centers[i, 1]
                dist_sq = dx * dx + dy * dy
                dist = np.sqrt(max(dist_sq, 1e-20))
                target_dist = radii[i] + radii[j] + 0.005
                if dist < target_dist:
                    fx = dx / dist
                    fy = dy / dist
                    force_mag = (target_dist - dist) * 80.0
                    forces[i, 0] -= fx * force_mag
                    forces[i, 1] -= fy * force_mag
                    forces[j, 0] += fx * force_mag
                    forces[j, 1] += fy * force_mag
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < 0.015:
                forces[i, 0] += (0.015 - (x - r)) * 150.0
            if x + r > 0.985:
                forces[i, 0] -= (x + r - 0.985) * 150.0
            if y - r < 0.015:
                forces[i, 1] += (0.015 - (y - r)) * 150.0
            if y + r > 0.985:
                forces[i, 1] -= (y + r - 0.985) * 150.0
        centers += forces * 0.003
        centers = np.clip(centers, 0.02, 0.98)
        for i in range(n):
            max_r = min(
                centers[i, 0],
                1.0 - centers[i, 0],
                centers[i, 1],
                1.0 - centers[i, 1],
            )
            radii[i] = min(radii[i] * 1.0002, max_r)
    return centers, radii


def optimize_from_initial(x0, n, constraints, bounds):
    result = minimize(
        objective_func,
        x0,
        args=(n,),
        method="SLSQP",
        bounds=bounds,
        constraints=constraints,
        options={"maxiter": 8000, "ftol": 1e-14},
    )
    return result


def run_packing():
    n = 26
    constraints = build_constraints(n)
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n

    best_sum = -np.inf
    best_x = None

    # Try multiple initial configurations
    for trial in range(8):
        if trial == 0:
            x0 = hexagonal_init(n, 1.0)
        elif trial == 1:
            x0 = hexagonal_init(n, 1.2)
        elif trial == 2:
            x0 = hexagonal_init(n, 0.9)
        else:
            np.random.seed(trial * 17)
            centers = np.random.uniform(0.15, 0.85, (n, 2))
            radii = np.full(n, 0.025)
            x0 = np.concatenate([centers.flatten(), radii])

        # Physics pre-optimization
        centers_phys = x0[: 2 * n].reshape(n, 2).copy()
        radii_phys = x0[2 * n:].copy()
        centers_phys, radii_phys = physics_optimize(
            centers_phys, radii_phys, n, iterations=2000
        )
        x0_phys = np.concatenate([centers_phys.flatten(), radii_phys])

        # SLSQP refinement
        result = optimize_from_initial(x0_phys, n, constraints, bounds)

        if result.success:
            current_sum = -result.fun
            if current_sum > best_sum:
                best_sum = current_sum
                best_x = result.x.copy()

    if best_x is None:
        x0 = hexagonal_init(n, 1.0)
        best_x = x0

    centers = best_x[: 2 * n].reshape(n, 2)
    radii = best_x[2 * n:]

    # Ensure radii respect boundaries
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        max_r = min(x, 1.0 - x, y, 1.0 - y)
        radii[i] = max(0.0, min(r, max_r))

    # Final cleanup: ensure no overlaps (clipping radii if needed)
    for iteration in range(50):
        changed = False
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[j, 0] - centers[i, 0]
                dy = centers[j, 1] - centers[i, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                if dist < radii[i] + radii[j] - 1e-12:
                    excess = radii[i] + radii[j] - dist
                    reduction = excess / 2.0
                    radii[i] = max(0.0, radii[i] - reduction)
                    radii[j] = max(0.0, radii[j] - reduction)
                    changed = True
        if not changed:
            break

    # Re-check boundaries after overlap resolution
    for i in range(n):
        x, y = centers[i]
        max_r = min(x, 1.0 - x, y, 1.0 - y)
        radii[i] = max(0.0, min(radii[i], max_r))

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
