import numpy as np
from scipy.optimize import minimize
import warnings
warnings.filterwarnings('ignore')

N_CIRCLES = 26

def compute_constraints(vars):
    """Compute all inequality constraints: g(vars) >= 0"""
    n = N_CIRCLES
    pts = vars.reshape((n, 3))
    x, y, r = pts[:, 0], pts[:, 1], pts[:, 2]
    
    c = []
    # Boundary constraints: x - r >= 0, 1 - x - r >= 0, etc.
    c.append(x - r)
    c.append(1 - x - r)
    c.append(y - r)
    c.append(1 - y - r)
    
    # Overlap constraints: dist_ij - r_i - r_j >= 0
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist = np.sqrt(dx**2 + dy**2)
    r_sum = r[:, None] + r[None, :]
    
    i_u, j_u = np.triu_indices(n, k=1)
    c.append(dist[i_u, j_u] - r_sum[i_u, j_u])
    
    return np.concatenate(c)

def objective(vars):
    """Objective: minimize negative sum of radii"""
    n = N_CIRCLES
    return -np.sum(vars[2*n:])

def generate_initial_config(n, r=0.095):
    """Generate a hexagonal grid configuration as a warm start"""
    centers = []
    dy = r * np.sqrt(3)
    y = r
    row_idx = 0
    while y <= 1 - r:
        x_start = r if row_idx % 2 == 0 else 2 * r
        x = x_start
        while x <= 1 - r:
            if len(centers) < n:
                centers.append([x, y])
            x += 2 * r
        y += dy
        row_idx += 1
    return np.array(centers[:n]), np.full(n, r)

def run_packing():
    n = N_CIRCLES
    centers, radii = generate_initial_config(n, r=0.095)
    x0 = np.concatenate([centers.flatten(), radii])
    
    # Bounds for variables: centers in [0,1], radii in [0, 0.5]
    bounds = [(0, 1)] * (2 * n) + [(0, 0.5)] * n
    
    cons = {'type': 'ineq', 'fun': compute_constraints}
    
    # Run constrained optimization
    res = minimize(objective, x0, method='SLSQP', bounds=bounds, 
                   constraints=cons, options={'maxiter': 3000, 'ftol': 1e-12})
    
    final_vars = res.x if res.success else x0
    final_centers = final_vars[:2*n].reshape((n, 2))
    final_radii = final_vars[2*n:]
    
    # Safety clamping
    final_centers = np.clip(final_centers, 0, 1)
    final_radii = np.maximum(final_radii, 0)
    
    # Deterministic repair for numerical violations
    for i in range(n):
        for j in range(i + 1, n):
            d = np.sqrt(np.sum((final_centers[i] - final_centers[j])**2))
            req = final_radii[i] + final_radii[j]
            if d < req - 1e-9:
                scale = d / req
                final_radii[i] *= scale
                final_radii[j] *= scale
        for k in range(2):
            if final_centers[i, k] < final_radii[i] - 1e-9:
                final_radii[i] = final_centers[i, k]
            if final_centers[i, k] > 1 - final_radii[i] - 1e-9:
                final_radii[i] = 1 - final_centers[i, k]
                
    return final_centers, final_radii, np.sum(final_radii)
