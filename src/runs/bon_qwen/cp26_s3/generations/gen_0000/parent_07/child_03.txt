import numpy as np

def compute_repulsion(centers, radii):
    """Compute pairwise repulsive forces proportional to overlap."""
    n = centers.shape[0]
    # Pairwise differences: shape (n, n, 2)
    diff = centers[:, None, :] - centers[None, :, :]
    # Pairwise distances: shape (n, n)
    dist = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dist, np.inf)
    
    # Overlap calculation
    min_dist = radii[:, None] + radii[None, :]
    overlap = np.maximum(min_dist - dist, 0.0)
    
    # Force magnitude: overlap / distance (0 if no overlap or dist=0)
    safe_dist = np.where(dist > 1e-9, dist, 1.0)
    force_mag = np.where(overlap > 0, overlap / safe_dist, 0.0)
    
    # Unit direction vectors
    dir_unit = np.where(dist > 1e-9, diff / safe_dist[:, :, None], 0.0)
    
    # Accumulate forces: forces[i] = sum_j (mag_ij * dir_ij)
    forces = np.einsum('ij,ijk->ik', force_mag, dir_unit)
    return forces

def clamp_to_boundary(centers, radii):
    """Ensure all circles remain inside the unit square."""
    r_col = radii[:, None]
    centers[:, 0] = np.clip(centers[:, 0], r_col[:, 0], 1.0 - r_col[:, 0])
    centers[:, 1] = np.clip(centers[:, 1], r_col[:, 0], 1.0 - r_col[:, 0])

def optimize_packing(n, seed):
    """Run force-directed expansion simulation for a single seed."""
    np.random.seed(seed)
    
    # Hexagonal-ish grid initialization with margin
    cols = int(np.ceil(np.sqrt(n)))
    rows = int(np.ceil(n / cols))
    centers = np.zeros((n, 2))
    idx = 0
    margin = 0.12
    dx = (1.0 - 2 * margin) / (cols + 1e-9)
    dy = (1.0 - 2 * margin) / (rows + 1e-9)
    
    for r in range(rows):
        for c in range(cols):
            if idx < n:
                x = margin + c * dx + (0.5 * dx if r % 2 == 1 else 0.0)
                y = margin + r * dy
                centers[idx] = [x, y]
                idx += 1
                
    # Add initial noise
    centers += np.random.randn(n, 2) * 0.03
    radii = np.full(n, 0.015)
    
    dt = 0.6
    expand_rate = 1.0015
    max_iters = 2500
    relax_steps = 40
    
    for step in range(max_iters):
        # Expand radii
        radii *= expand_rate
        
        # Adaptive time step (cooling)
        current_dt = dt * (0.9995 ** step)
        
        # Positional relaxation
        for _ in range(relax_steps):
            forces = compute_repulsion(centers, radii)
            centers += current_dt * forces
            clamp_to_boundary(centers, radii)
            
        # Periodic shake to escape local minima
        if step > 0 and step % 400 == 0:
            centers += np.random.randn(n, 2) * 0.008
            clamp_to_boundary(centers, radii)
            
    # Final high-precision relaxation with fixed radii
    for _ in range(3000):
        forces = compute_repulsion(centers, radii)
        centers += 0.15 * forces
        clamp_to_boundary(centers, radii)
        
    return centers, radii

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Find and return the optimal packing configuration."""
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Run multiple restarts to find the best local optimum
    for seed in range(7):
        centers, radii = optimize_packing(26, seed)
        current_sum = np.sum(radii)
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers.copy()
            best_radii = radii.copy()
            
    # Guarantee strict validity by scaling down if overlaps/boundary violations exist
    min_scale = 1.0
    
    # Check pairwise overlaps
    for i in range(26):
        for j in range(i + 1, 26):
            dist = np.sqrt(np.sum((best_centers[i] - best_centers[j])**2))
            req = best_radii[i] + best_radii[j]
            if req > 1e-12:
                ratio = dist / req
                if ratio < min_scale:
                    min_scale = ratio
                    
    # Check boundary distances
    for i in range(26):
        r = best_radii[i]
        if r > 1e-12:
            dist_wall = min(best_centers[i, 0], 1.0 - best_centers[i, 0], 
                            best_centers[i, 1], 1.0 - best_centers[i, 1])
            ratio = dist_wall / r
            if ratio < min_scale:
                min_scale = ratio
                
    # Apply scaling with a small safety margin
    if min_scale < 1.0:
        best_radii *= min_scale * (1.0 - 1e-7)
        best_sum = np.sum(best_radii)
        
    return best_centers, best_radii, best_sum
