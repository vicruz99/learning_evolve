import numpy as np
import math
from scipy.optimize import minimize

def check_constraints(centers, radii):
    """
    Returns a penalty value. 0 if valid, >0 if violations exist.
    """
    n = centers.shape[0]
    penalty = 0.0
    
    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        # Penalties for being outside [0,1]
        if x - r < 0: penalty += (x - r)**2
        if x + r > 1: penalty += (x + r - 1)**2
        if y - r < 0: penalty += (y - r)**2
        if y + r > 1: penalty += (y + r - 1)**2
        
    # Overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            dist_sq = np.sum((centers[i] - centers[j])**2)
            dist = math.sqrt(dist_sq) if dist_sq > 0 else 0.0
            r_sum = radii[i] + radii[j]
            if dist < r_sum:
                penalty += (r_sum - dist)**2
                
    return penalty
