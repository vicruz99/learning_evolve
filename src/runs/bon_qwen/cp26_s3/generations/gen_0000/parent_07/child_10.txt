import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    N = 26
    
    def objective(vars):
        # Maximize r, so minimize -r
        # vars[-1] is the radius r
        return -vars[-1]
    
    def boundary_constraints(vars):
        coords = vars[:-1].reshape(N, 2)
        r = vars[-1]
        x = coords[:, 0]
        y = coords[:, 1]
        
        # Returns a vector where all elements must be >= 0
        # x - r >= 0
        # 1 - x - r >= 0
        # y - r >= 0
        # 1 - y - r >= 0
        b1 = x - r
        b2 = 1.0 - x - r
        b3 = y - r
        b4 = 1.0 - y - r
        return np.concatenate([b1, b2, b3, b4])
    
    def pairwise_constraints(vars):
        coords = vars[:-1].reshape(N, 2)
        r = vars[-1]
        
        # Compute squared Euclidean distances between all pairs
        # Using broadcasting for efficiency
        # dist_sq[i, j] = ||p_i - p_j||^2
        diff = coords[:, np.newaxis, :] - coords[np.newaxis, :, :]
        dist_sq = np.sum(diff**2, axis=2)
        
        # We only need upper triangle (i < j)
        indices = np.triu_indices(N, k=1)
        distances = dist_sq[indices]
        
        # Constraint: dist^2 - 4r^2 >= 0
        return distances - 4.0 * (r**2)
    
    # Define bounds for variables
    # x, y in [0, 1], r in [0, 1]
    bounds = [(0, 1)] * (2 * N) + [(0, 1)]
    
    best_sum_radii = -1.0
    best_centers = np.zeros((N, 2))
    best_radii = np.zeros(N)
    
    # Multi-start optimization to avoid local minima
    for seed in range(20):
        # Initialize centers randomly
        np.random.seed(seed)
        initial_centers = np.random.rand(N, 2) * 0.8 + 0.1
        initial_r = 0.05
        
        x0 = np.concatenate([initial_centers.flatten(), [initial_r]])
        
        # Define constraints for the solver
        constraints = [
            {'type': 'ineq', 'fun': boundary_constraints},
            {'type': 'ineq', 'fun': pairwise_constraints}
        ]
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, 
                           constraints=constraints, options={'maxiter': 1000, 'ftol': 1e-8})
            
            if res.success:
                current_r = res.x[-1]
                current_sum = current_r * N
                
                if current_sum > best_sum_radii:
                    best_sum_radii = current_sum
                    best_centers = res.x[:2*N].reshape(N, 2)
                    best_radii = np.full(N, current_r)
        except Exception:
            continue
            
    return best_centers, best_radii, best_sum_radii
