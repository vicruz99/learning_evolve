import numpy as np
import scipy.optimize as opt

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    def get_loss(params):

        # Reshape to (n, 3)
        pts = params.reshape(n_circles, 3)
        centers = pts[:, :2]
        radii = pts[:, 2]

        # but here we just check)

        loss = -np.sum(radii)
        
        # Penalty for overlapping circles
        # dist >= r_i + r_j
        # (dist)^2 >= (r_i + r_j)^2
        # We can penalize if (r_i + r_j)^2 - dist^2 > 0
        
        # Vectorized distance matrix
        # dist_sq[i, j] = (x_i - x_j)^2 + (y_i - y_j)^2
        diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
        dist_sq = np.sum(diff**2, axis=2)
        
        # r_sum_sq[i, j] = (r_i + r_j)^2
        r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
        r_sum_sq = r_sum**2
        
        # Constraint: dist_sq >= r_sum_sq
        # Violation = r_sum_sq - dist_sq
        violation = r_sum_sq - dist_sq
        np.fill_diagonal(violation, 0) # Ignore self
        
        # Quadratic penalty for overlap
        # Only penalize positive violations (overlaps)
        overlap = np.maximum(0, violation)
        penalty = np.sum(overlap**2)

        # x + r <= 1 => x + r - 1 <= 0

        # y - r < 0
