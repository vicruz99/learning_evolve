import numpy as np

def compute_forces(centers, radii, k_rep=100.0, k_wall=100.0):
    """
    Compute repulsive forces between circles and from boundaries.
    """
    N = len(radii)
    forces = np.zeros_like(centers)
    
    # Boundary forces: push circles inside the square
    for i in range(N):
        for d in range(2):
            if centers[i, d] < radii[i]:
                forces[i, d] += k_wall * (radii[i] - centers[i, d])
            elif centers[i, d] > 1.0 - radii[i]:
                forces[i, d] -= k_wall * (centers[i, d] - (1.0 - radii[i]))
                
    # Pairwise repulsion forces: push overlapping circles apart
    for i in range(N):
        for j in range(i + 1, N):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist_sq = dx*dx + dy*dy
            min_dist = radii[i] + radii[j]
            
            if dist_sq < min_dist * min_dist and dist_sq > 1e-14:
                dist = np.sqrt(dist_sq)
                overlap = min_dist - dist
                # Force proportional to overlap amount
                force_mag = k_rep * overlap
                fx = force_mag * dx / dist
                fy = force_mag * dy / dist
                forces[i, 0] += fx
                forces[i, 1] += fy
                forces[j, 0] -= fx
                forces[j, 1] -= fy
                
    return forces

def run_packing():
    N = 26
    best_centers = None

    best_sum_r = -1.0
    
    # Run multiple optimizations with different seeds to escape local minima
    for seed in range(10):

        # Initialize positions randomly within a safe inner region
        centers = np.random.rand(N, 2) * 0.8 + 0.1
        radii = np.full(N, 0.005)
        
        lr = 0.0015
        growth_rate = 0.00012
        k_rep = 60.0
        k_wall = 60.0
        
        # Main growth and relaxation loop
        for step in range(2500):
            # Grow radii uniformly
            radii += growth_rate
            
            # Compute forces based on current configuration
            forces = compute_forces(centers, radii, k_rep, k_wall)
            
            # Update positions
            centers += lr * forces
            
            # Enforce boundary constraints strictly
            centers = np.clip(centers, radii[:, None], 1.0 - radii[:, None])
            
            # Gradually reduce growth rate to allow the system to settle
            if step > 1000:
                growth_rate *= 0.9995
                
        # Final high-stiffness relaxation to resolve any remaining overlaps
        for _ in range(800):

            centers += 0.0005 * forces
            centers = np.clip(centers, radii[:, None], 1.0 - radii[:, None])
            
        current_sum = np.sum(radii)
        if current_sum > best_sum_r:
            best_sum_r = current_sum
            best_centers = centers.copy()
            best_radii = radii.copy()
            
    return best_centers, best_radii, np.sum(best_radii)
