import numpy as np
from scipy.optimize import minimize

def compute_penalty(centers_flat, r, n):
    """
    Computes the penalty for boundary violations and overlaps.
    Lower penalty is better. Penalty ~ 0 implies a valid packing.
    """
    centers = centers_flat.reshape(n, 2)
    penalty = 0.0
    
    # Boundary constraints: circles must be within [0,1]x[0,1]
    # x >= r, x <= 1-r, y >= r, y <= 1-r
    left_viol = np.maximum(0, r - centers[:, 0])
    right_viol = np.maximum(0, centers[:, 0] - (1 - r))
    bottom_viol = np.maximum(0, r - centers[:, 1])
    top_viol = np.maximum(0, centers[:, 1] - (1 - r))
    
    penalty += np.sum(left_viol**2 + right_viol**2 + bottom_viol**2 + top_viol**2)
    
    # Overlap constraints: distance between centers >= 2r
    # Compute pairwise distances efficiently
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    # Mask for upper triangle to consider each pair once
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    overlaps = np.maximum(0, 2 * r - dists[mask])
    penalty += np.sum(overlaps**2)
    
    return penalty

    tol = 1e-9
    
    # Perform binary search
    for step in range(40):
        r_target = (low + high) / 2
        found_valid = False
        
        # Multiple restarts to escape local minima
        for trial in range(6):
            curr_centers = best_centers.copy()

            if trial > 0:
                noise = np.random.uniform(-0.02, 0.02, curr_centers.shape)
                curr_centers += noise

            try:

                res = minimize(compute_penalty, curr_centers.flatten(),

                    found_valid = True
                    break

                continue
                
        if not found_valid:
            high = r_target
            
        if high - low < tol:
            break
            
    # Final radii array
    radii = np.full(n, best_r)
    sum_radii = np.sum(radii)
    
    return best_centers, radii, sum_radii
