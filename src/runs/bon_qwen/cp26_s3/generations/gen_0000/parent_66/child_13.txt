from scipy.optimize import minimize

def constraint_overlap(x, i, j):

    xj, yj = x[2*j], x[2*j+1]

def constraint_wall_x_min(x, i):
    return x[2*i] - x[52+i]

def constraint_wall_x_max(x, i):
    return 1.0 - x[2*i] - x[52+i]

def constraint_wall_y_min(x, i):
    return x[2*i+1] - x[52+i]

def constraint_wall_y_max(x, i):

        x = col * 0.2 + shift

        col += 1
        if col > 6:
            col = 0
            row += 1
            
    xs = np.array([p[0] for p in pts])

    return centers, radii, float(np.sum(radii))
