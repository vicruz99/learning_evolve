To maximize the sum of radii for 26 circles in a unit square, we formulate the problem as a constrained nonlinear optimization. 
Key insights:
1. **Parameterization**: Instead of optimizing $(x, y, r)$ directly with boundary constraints $r \le x \le 1-r$, we use a transformation $x = r + (1-2r)u_x$ with $u_x \in [0,1]$. This automatically satisfies boundary constraints for any $r \in [0, 0.5]$, simplifying the optimization landscape.
2. **Objective**: Maximize $\sum r_i$ is equivalent to minimizing $-\sum r_i$.
3. **Constraints**: Only pairwise non-overlap constraints remain: $\|c_i - c_j\|^2 \ge (r_i + r_j)^2$.
4. **Solver**: `scipy.optimize.minimize` with SLSQP handles inequality constraints efficiently. We use multiple random restarts with perturbed initial guesses to escape local minima.
5. **Initialization**: Start with a slightly perturbed 5x5 grid ($r \approx 0.09$), which is feasible and close to the optimal region.
6. **Validation**: Post-optimization, we strictly verify non-overlap and boundary conditions before accepting a solution.

import numpy as np
from scipy.optimize import minimize

def decode(Z, n=26):
    """Transforms optimization variables to centers and radii."""
    Z_mat = Z.reshape(-1, 3)
    u_x = np.clip(Z_mat[:, 0], 0.0, 1.0)
    u_y = np.clip(Z_mat[:, 1], 0.0, 1.0)
    r = np.clip(Z_mat[:, 2], 1e-5, 0.5)
    
    # Boundary-safe transformation
    x = r + (1.0 - 2.0 * r) * u_x
    y = r + (1.0 - 2.0 * r) * u_y
    return np.column_stack((x, y)), r

def objective(Z):
    """Objective function: maximize sum of radii."""
    _, radii = decode(Z)
    return -np.sum(radii)

def constraint_func(Z):
    """Pairwise non-overlap constraints."""
    n = 26
    centers, radii = decode(Z, n)
    c = np.empty(n * (n - 1) // 2)
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            d2 = np.sum((centers[i] - centers[j])**2)
            c[idx] = d2 - (radii[i] + radii[j])**2
            idx += 1
    return c

def run_packing():
    np.random.seed(42)
    n = 26
    
    # Initial configuration: 5x5 grid + 1, slightly perturbed
    Z0 = np.zeros((n, 3))
    for k in range(n):
        row = k // 5
        col = k % 5
        Z0[k, 0] = (col + 0.5) / 5.0
        Z0[k, 1] = (row + 0.5) / 5.0
        Z0[k, 2] = 0.09 + 0.005 * np.random.randn()

    # Bounds: u_x, u_y in [0,1], r in [0, 0.2]
    bounds = [(0.0, 1.0) for _ in range(2 * n)] + [(0.0, 0.2) for _ in range(n)]
    cons = {'type': 'ineq', 'fun': constraint_func}

    best_Z = Z0.flatten().copy()
    best_sum = np.sum(best_Z[2::3])

    # Multiple restarts to escape local optima
    for trial in range(8):
        if trial == 0:
            x0 = best_Z.copy()
        else:
            x0 = best_Z.copy()
            x0[:2 * n] += np.random.randn(2 * n) * 0.04
            x0[2 * n:] += np.random.randn(n) * 0.008
            x0[:2 * n] = np.clip(x0[:2 * n], 0.0, 1.0)
            x0[2 * n:] = np.clip(x0[2 * n:], 0.05, 0.2)

        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 2000, 'ftol': 1e-9})
            
            centers, radii = decode(res.x)
            
            # Strict feasibility check
            feasible = True
            for i in range(n):
                for j in range(i + 1, n):
                    dist = np.linalg.norm(centers[i] - centers[j])
                    if dist < radii[i] + radii[j] - 1e-7:
                        feasible = False
                        break
                if not feasible:
                    break
            
            if feasible:
                current_sum = np.sum(radii)
                if current_sum > best_sum:
                    best_sum = current_sum
                    best_Z = res.x.copy()
        except Exception:
            continue

    final_centers, final_radii = decode(best_Z)
    return final_centers, final_radii, np.sum(final_radii)
