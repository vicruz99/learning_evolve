import numpy as np
from scipy.optimize import minimize

def compute_forces(coords, radii, k_rep=20.0, k_bnd=20.0):
    """Compute repulsive forces between circles and boundaries."""
    n = len(coords)
    forces = np.zeros_like(coords)
    
    # Pairwise repulsion
    diffs = coords[:, np.newaxis, :] - coords[np.newaxis, :, :]
    dists_sq = np.sum(diffs**2, axis=2)
    dists = np.sqrt(np.maximum(dists_sq, 1e-12))
    normals = diffs / dists[:, :, np.newaxis]
    
    r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    overlap = np.maximum(r_sum - dists, 0)
    
    # Accumulate forces: F = k * overlap * direction
    forces += np.sum(k_rep * overlap[:, :, np.newaxis] * normals, axis=1)
    
    # Boundary repulsion
    forces[:, 0] += k_bnd * np.maximum(radii - coords[:, 0], 0)
    forces[:, 0] -= k_bnd * np.maximum(coords[:, 0] + radii - 1.0, 0)
    forces[:, 1] += k_bnd * np.maximum(radii - coords[:, 1], 0)
    forces[:, 1] -= k_bnd * np.maximum(coords[:, 1] + radii - 1.0, 0)
    
    return forces

def build_objective(n):
    """Returns the optimization objective function with penalty terms."""
    def func(x):
        c = x[:n*2].reshape(n, 2)
        r = x[n*2:]
        
        # Primary objective: maximize sum of radii
        val = -np.sum(r)
        
        # Penalty for pairwise overlaps
        diffs = c[:, np.newaxis, :] - c[np.newaxis, :, :]
        dists = np.sqrt(np.sum(diffs**2, axis=2))
        r_sum = r[:, np.newaxis] + r[np.newaxis, :]
        p_overlap = np.sum(np.maximum(r_sum - dists, 0)**2)
        
        # Penalty for boundary violations
        p_bnd = (np.sum(np.maximum(r - c[:,0], 0)**2) +
                 np.sum(np.maximum(c[:,0] + r - 1, 0)**2) +
                 np.sum(np.maximum(r - c[:,1], 0)**2) +
                 np.sum(np.maximum(c[:,1] + r - 1, 0)**2))
        
        # High penalty weight to enforce constraints strictly
        return val + 1000.0 * (p_overlap + p_bnd)
    return func

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # 1. Hexagonal Lattice Initialization
    coords = []
    r0 = 0.09
    y = r0
    # Row pattern sums to 26: 6+5+6+5+4
    pattern = [6, 5, 6, 5, 4]
    for row, nc in enumerate(pattern):
        offset = r0 if row % 2 == 1 else 0.0
        for i in range(nc):
            x = r0 + offset + i * 2 * r0
            coords.append([x, y])
        y += np.sqrt(3) * r0
        
    coords = np.array(coords)
    radii = np.full(n, r0)
    
    # 2. Grow & Relax Heuristic
    for step in range(120):
        # Relax positions for current radii
        for _ in range(500):
            forces = compute_forces(coords, radii)
            coords += 0.0025 * forces
            coords = np.clip(coords, 0.0, 1.0)
        
        # Gradually increase radii
        if step % 5 == 0:
            radii += 0.0012
            
    # 3. Numerical Fine-Tuning
    x0 = np.concatenate([coords.flatten(), radii])
    obj_func = build_objective(n)
    
    res = minimize(obj_func, x0, method='L-BFGS-B', 
                   options={'ftol': 1e-10, 'maxiter': 3000, 'iprint': -1})
    
    final_coords = res.x[:n*2].reshape(n, 2)
    final_radii = res.x[n*2:]
    
    # Ensure radii are strictly positive
    final_radii = np.maximum(final_radii, 1e-6)
    
    return final_coords, final_radii, float(np.sum(final_radii))
