3.  **Stage 2 (Unequal Radii)**: Using the result from Stage 1 as a starting point, relax the equal radii constraint to allow independent radii $r_i$. Maximize $\sum r_i$ subject to generalized non-overlap constraints ($dist \ge r_i + r_j$).

import numpy as np
import scipy.optimize as opt

def get_initial_positions(n=26):
    """
    Generates an initial valid packing configuration using a staggered grid.
    Returns centers (n, 2) and radii (n,).
    """
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.05 # Start with small radius
    
    # Heuristic layout: 5 rows with alternating counts
    # Target 26 circles. 
    # Layout: 6, 5, 6, 5, 4 circles in 5 rows.
    # This mimics hexagonal packing structure.
    
    rows_config = [6, 5, 6, 5, 4]
    y_positions = [0.2, 0.4, 0.6, 0.8, 0.5] # Roughly spaced
    
    idx = 0
    for r_idx, count in enumerate(rows_config):
        y_val = y_positions[r_idx]
        # X positions spread within [0.1, 0.9]
        # To fit 'count' circles, we need spacing
        # Available width approx 0.8
        # Spacing = 0.8 / (count - 1) if count > 1 else 0
        if count > 1:
            step = 0.8 / (count - 1)
            # Center the row
            start_x = 0.5 - (step * (count - 1)) / 2
            xs = start_x + np.arange(count) * step
        else:
            xs = np.array([0.5])
            
        for k in range(count):
            if idx < n:
                centers[idx, 0] = xs[k]
                centers[idx, 1] = y_val
                idx += 1
                
    # Fill remaining if any (should be exact for 26)
    if idx < n:
        for i in range(idx, n):
            centers[i, 0] = 0.5
            centers[i, 1] = 0.5 + (i - idx) * 0.01
            
    return centers, radii

def objective_equal(params, n):
    """Objective for equal radii: minimize -r"""
    # params layout: x1, y1, ..., xn, yn, r
    # Indices 0..2n-1 are coords, 2n is r
    r = params[2*n]
    return -r

def objective_unequal(params, n):
    """Objective for unequal radii: minimize sum(-r_i)"""
    # params layout: x1, y1, r1, x2, y2, r2, ...
    # Stride 3
    r_sum = 0.0
    for i in range(n):
        r_sum += params[3*i + 2]
    return -r_sum

def constraints_equal(params, n):
    """Constraints for equal radii optimization"""
    constraints_list = []
    
    # Extract variables
    # x: 0..n-1, y: n..2n-1, r: 2n
    x = params[0:n]
    y = params[n:2*n]
    r = params[2*n]
    
    # 1. Boundary constraints: r <= x <= 1-r  =>  x - r >= 0,  1 - x - r >= 0
    # Using >= 0 form for SLSQP
    
    # x - r >= 0
    for i in range(n):
        constraints_list.append({'type': 'ineq', 'fun': lambda p, i=i: p[i] - p[2*n]})
    # 1 - x - r >= 0
    for i in range(n):
        constraints_list.append({'type': 'ineq', 'fun': lambda p, i=i: 1.0 - p[i] - p[2*n]})
        
    # y - r >= 0
    for i in range(n):
        constraints_list.append({'type': 'ineq', 'fun': lambda p, i=i: p[n+i] - p[2*n]})
    # 1 - y - r >= 0
    for i in range(n):
        constraints_list.append({'type': 'ineq', 'fun': lambda p, i=i: 1.0 - p[n+i] - p[2*n]})
        
    # 2. Non-negativity of r
    constraints_list.append({'type': 'ineq', 'fun': lambda p: p[2*n]})
    
    # 3. Distance constraints: dist^2 >= (2r)^2
    # (x_i - x_j)^2 + (y_i - y_j)^2 - 4r^2 >= 0
    for i in range(n):
        for j in range(i + 1, n):
            def dist_con(p, i=i, j=j):
                dx = p[i] - p[j]
                dy = p[n+i] - p[n+j]
                r_val = p[2*n]
                return dx**2 + dy**2 - 4.0 * (r_val**2)
            constraints_list.append({'type': 'ineq', 'fun': dist_con})
            
    return constraints_list

def constraints_unequal(params, n):
    """Constraints for unequal radii optimization"""
    constraints_list = []
    
    # 1. Boundary constraints for each circle
    # Circle i: x_i, y_i, r_i at indices 3i, 3i+1, 3i+2
    
    for i in range(n):
        idx_x = 3*i
        idx_y = 3*i + 1
        idx_r = 3*i + 2
        
        # x - r >= 0
        constraints_list.append({'type': 'ineq', 'fun': lambda p, ix=idx_x, ir=idx_r: p[ix] - p[ir]})
        # 1 - x - r >= 0
        constraints_list.append({'type': 'ineq', 'fun': lambda p, ix=idx_x, ir=idx_r: 1.0 - p[ix] - p[ir]})
        # y - r >= 0
        constraints_list.append({'type': 'ineq', 'fun': lambda p, iy=idx_y, ir=idx_r: p[iy] - p[ir]})
        # 1 - y - r >= 0
        constraints_list.append({'type': 'ineq', 'fun': lambda p, iy=idx_y, ir=idx_r: 1.0 - p[iy] - p[ir]})
        
        # r >= 0
        constraints_list.append({'type': 'ineq', 'fun': lambda p, ir=idx_r: p[ir]})

    # 2. Distance constraints
    for i in range(n):
        for j in range(i + 1, n):
            ix_i, iy_i, ir_i = 3*i, 3*i+1, 3*i+2
            ix_j, iy_j, ir_j = 3*j, 3*j+1, 3*j+2
            
            def dist_con_unequal(p, ii=i, jj=j):
                ix_i, iy_i, ir_i = 3*ii, 3*ii+1, 3*ii+2
                ix_j, iy_j, ir_j = 3*jj, 3*jj+1, 3*jj+2
                dx = p[ix_i] - p[ix_j]
                dy = p[iy_i] - p[iy_j]
                r_sum = p[ir_i] + p[ir_j]
                return dx**2 + dy**2 - r_sum**2
            
            constraints_list.append({'type': 'ineq', 'fun': dist_con_unequal})
            
    return constraints_list

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # Helper to convert params to centers/radii for equal radii stage
    def unpack_equal(p):
        centers = np.column_stack((p[:n], p[n:2*n]))
        radii = np.full(n, p[2*n])
        return centers, radii

    # Helper to convert params to centers/radii for unequal stage
    def unpack_unequal(p):
        centers = np.zeros((n, 2))
        radii = np.zeros(n)
        for i in range(n):
            centers[i, 0] = p[3*i]
            centers[i, 1] = p[3*i+1]
            radii[i] = p[3*i+2]
        return centers, radii

    best_sum_radii = -np.inf
    best_centers = None
    best_radii = None
    
    # Try multiple random restarts/permutations to avoid local minima
    num_restarts = 5
    
    for seed in range(num_restarts):
        # 1. Initialize
        centers_init, radii_init = get_initial_positions(n)
        
        # Add small random perturbation to break symmetry
        if seed > 0:
            centers_init += np.random.uniform(-0.02, 0.02, centers_init.shape)
            # Clip to valid range roughly
            centers_init = np.clip(centers_init, 0.05, 0.95)
            
        # --- Stage 1: Equal Radii Optimization ---
        # Variables: x(26), y(26), r(1)
        x0_equal = np.concatenate([centers_init[:, 0], centers_init[:, 1], [0.08]])
        
        # Bounds: x, y in [0, 1], r in [0, 0.5]
        bounds_equal = [(0, 1)] * (2*n) + [(0, 0.5)]
        
        try:
            res_equal = opt.minimize(
                objective_equal,
                x0_equal,
                args=(n,),
                method='SLSQP',
                bounds=bounds_equal,
                constraints=constraints_equal(x0_equal, n),
                options={'maxiter': 200, 'ftol': 1e-9}
            )
            
            if res_equal.success:
                c_eq, r_eq = unpack_equal(res_equal.x)
                sum_eq = np.sum(r_eq)
                
                # --- Stage 2: Unequal Radii Optimization ---
                # Initialize unequal params from equal solution
                # We might want to perturb radii slightly or just start there
                # But equal solution is a valid point.
                
                x0_unequal = np.zeros(3*n)
                for i in range(n):
                    x0_unequal[3*i] = c_eq[i, 0]
                    x0_unequal[3*i+1] = c_eq[i, 1]
                    x0_unequal[3*i+2] = r_eq[i]
                
                bounds_unequal = [(0, 1), (0, 1), (0, 0.5)] * n
                
                try:
                    res_unequal = opt.minimize(
                        objective_unequal,
                        x0_unequal,
                        args=(n,),
                        method='SLSQP',
                        bounds=bounds_unequal,
                        constraints=constraints_unequal(x0_unequal, n),
                        options={'maxiter': 500, 'ftol': 1e-10}
                    )
                    
                    if res_unequal.success:
                        c_uneq, r_uneq = unpack_unequal(res_unequal.x)
                        sum_uneq = np.sum(r_uneq)
                        
                        # Check validity manually just in case numerical issues
                        # The optimizer should respect constraints, but tight tolerances might slip
                        # We can try to repair or just accept.
                        
                        if sum_uneq > best_sum_radii:
                            # Validate before storing
                            # Simple check
                            valid = True
                            # Check bounds

                                if r_uneq[i] < -1e-9: valid = False
                                if c_uneq[i,0] < r_uneq[i] - 1e-9 or c_uneq[i,0] > 1 - r_uneq[i] + 1e-9: valid = False
                                if c_uneq[i,1] < r_uneq[i] - 1e-9 or c_uneq[i,1] > 1 - r_uneq[i] + 1e-9: valid = False
                            
                            if valid:
                                best_sum_radii = sum_uneq
                                best_centers = c_uneq.copy()
                                best_radii = r_uneq.copy()
                                
                except Exception as e:
                    pass # Stage 2 failed, keep Stage 1 result if better?
                    # Actually Stage 1 result is valid.
                    # Let's check Stage 1 sum
                    if sum_eq > best_sum_radii:
                        best_sum_radii = sum_eq
                        best_centers = c_eq.copy()
                        best_radii = r_eq.copy()

        except Exception as e:
            pass

    return best_centers, best_radii, best_sum_radii
