def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    # Distribute circles into rows to fit in a 1x1 square.
    # 5 rows of 5 circles and 1 row of 1 circle = 26 circles.
    # Hexagonal vertical spacing allows tighter packing than square grid.

        y_center = 0.5 + i * (np.sqrt(3) / 2) * 0.15 # rough scaling
        
        # Horizontal positions
        # Shift odd rows by r to create hexagonal pattern
        x_start = 0.5
        spacing = 2 * r_est
        
        for j in range(count):
            idx = total_circles
            total_circles += 1
            
            # Adjust x based on row parity for staggering
            if i % 2 == 1:

            else:
                x = x_start - (count-1)*spacing/2 + j*spacing

            centers[idx] = [x, y_center]
            radii[idx] = r_est
            
    # Normalize centers to be within [0.1, 0.9] to ensure initial validity

        """
        Parameters: [x1, y1, r1, x2, y2, r2, ...]
        Returns: -sum(radii) (to minimize)

        # If any radius is negative or NaN, penalize heavily

        x = params[0::3]
        y = params[1::3]
        r = params[2::3]

        c.extend((1 - x) - r)

        c.extend((1 - y) - r)

                c.append(dist_sq - sum_r**2)
                
        return np.array(c)
