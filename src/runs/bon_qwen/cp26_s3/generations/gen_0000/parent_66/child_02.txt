import numpy as np
from scipy.optimize import minimize

def _get_circles(vars, n):
    """Extract centers and radii from the optimization variable vector."""
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        centers[i] = vars[3*i:3*i+2]
        radii[i] = vars[3*i+2]
    return centers, radii

def _objective(vars, n):
    """Objective function: maximize sum of radii (minimize negative sum)."""
    _, radii = _get_circles(vars, n)
    return -np.sum(radii)

def _constraints(vars, n):
    """
    Returns a 1D array of constraint values.
    SLSQP requires fun(x) >= 0.
    """
    centers, radii = _get_circles(vars, n)
    con = []
    
    # Boundary constraints: x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
    con.append(centers[:, 0] - radii)
    con.append(1.0 - centers[:, 0] - radii)
    con.append(centers[:, 1] - radii)
    con.append(1.0 - centers[:, 1] - radii)
    
    # Pairwise non-overlap constraints: dist^2 - (r_i + r_j)^2 >= 0
    for i in range(n):
        for j in range(i + 1, n):
            d2 = np.sum((centers[i] - centers[j])**2)
            r_sum = radii[i] + radii[j]
            con.append(d2 - r_sum**2)
            
    return np.concatenate(con)

def run_packing():

            vars0[3*i+2] = radii_init[i]

        best_sum = np.sum(best_radii)
        
    return best_centers, best_radii, best_sum
