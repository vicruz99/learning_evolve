import numpy as np
from scipy.optimize import minimize

def obj_func(v, n):
    """Objective: maximize sum of radii (minimize negative sum)"""
    return -np.sum(v[2::3])

def make_x_ge_r(i):
    def fun(v):
        return v[3*i] - v[3*i+2]
    return fun

def make_1_minus_x_ge_r(i):
    def fun(v):
        return 1 - v[3*i] - v[3*i+2]
    return fun

def make_y_ge_r(i):
    def fun(v):
        return v[3*i+1] - v[3*i+2]
    return fun

def make_1_minus_y_ge_r(i):
    def fun(v):
        return 1 - v[3*i+1] - v[3*i+2]
    return fun

def make_dist_constraint(i, j):
    def fun(v):
        dx = v[3*i] - v[3*j]
        dy = v[3*i+1] - v[3*j+1]
        dr = v[3*i+2] + v[3*j+2]
        return dx*dx + dy*dy - dr*dr
    return fun

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    n = 26
    best_result = None
    best_sum = -np.inf
    
    # Try multiple initial configurations to avoid local minima
    for trial in range(5):
        xs = []
        ys = []
        if trial < 3:
            # Perturbed square grid + center circle
            for r in range(5):
                for c in range(5):
                    xs.append(0.1 + c * 0.2 + np.random.uniform(-0.015, 0.015))
                    ys.append(0.1 + r * 0.2 + np.random.uniform(-0.015, 0.015))
            xs.append(0.5 + np.random.uniform(-0.1, 0.1))
            ys.append(0.5 + np.random.uniform(-0.1, 0.1))
        else:
            # Perturbed hexagonal arrangement
            xs = []
            ys = []
            count = 0
            for row in range(6):
                num_in_row = 5 if row % 2 == 0 else 4
                offset = 0.13 if row % 2 == 1 else 0.0
                for c in range(num_in_row):
                    xs.append(offset + c * 0.22 + np.random.uniform(-0.01, 0.01))
                    ys.append(0.07 + row * 0.18 + np.random.uniform(-0.01, 0.01))
                    count += 1
                    if count == 26:
                        break
                if count == 26:
                    break
        
        x0 = np.zeros(3 * n)
        for i in range(n):
            x0[3*i] = np.clip(xs[i], 0.06, 0.94)
            x0[3*i+1] = np.clip(ys[i], 0.06, 0.94)
            # Slightly randomized initial radii to break symmetry
            x0[3*i+2] = 0.09 + np.random.uniform(-0.005, 0.005)
            
        bounds = [(0, 1)] * n + [(0, 1)] * n + [(0, 1)] * n
        
        cons = []
        for i in range(n):
            cons.append({'type': 'ineq', 'fun': make_x_ge_r(i)})
            cons.append({'type': 'ineq', 'fun': make_1_minus_x_ge_r(i)})
            cons.append({'type': 'ineq', 'fun': make_y_ge_r(i)})
            cons.append({'type': 'ineq', 'fun': make_1_minus_y_ge_r(i)})
            
        for i in range(n):
            for j in range(i+1, n):
                cons.append({'type': 'ineq', 'fun': make_dist_constraint(i, j)})
                
        try:
            res = minimize(obj_func, x0, args=(n,), method='SLSQP', bounds=bounds, 
                           constraints=cons, options={'maxiter': 4000, 'ftol': 1e-9})
            if -res.fun > best_sum:
                best_sum = -res.fun
                best_result = res
        except Exception:
            continue
            
    if best_result is None:
        # Fallback to dense grid packing
        centers = np.column_stack([np.linspace(0.1, 0.9, 5).repeat(5), 
                                   np.tile(np.linspace(0.1, 0.9, 5), 5)])
        centers = np.vstack([centers, [0.5, 0.5]])
        radii = np.full(26, 0.09)
        return centers, radii, np.sum(radii)

    centers = np.column_stack((best_result.x[0::3], best_result.x[1::3]))
    radii = best_result.x[2::3]
    
    # Ensure radii are strictly non-negative
    radii = np.maximum(radii, 1e-6)
    
    return centers, radii, float(np.sum(radii))
