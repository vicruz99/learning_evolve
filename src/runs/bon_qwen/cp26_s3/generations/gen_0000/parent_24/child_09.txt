    radii = np.full(n, 0.04)

                idx += 1
        y += np.sqrt(3) * r0
        row_idx += 1

    best_centers = centers.copy()

    for step in range(5000):

        dr = 0.00012 * np.exp(-step / 400)
        radii += dr

        for _ in range(80):
            # Enforce boundary constraints
            for i in range(n):
                r = radii[i]
                centers[i, 0] = np.clip(centers[i, 0], r, 1.0 - r)
                centers[i, 1] = np.clip(centers[i, 1], r, 1.0 - r)

                centers[i, 1] = np.clip(centers[i, 1], r, 1.0 - r)
