import numpy as np
from scipy.optimize import minimize

def compute_constraints(x, n):
    centers = x[:2*n].reshape(n, 2)
    radii = x[2*n:]
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    c1 = centers[:, 0] - radii
    c2 = 1.0 - centers[:, 0] - radii
    c3 = centers[:, 1] - radii
    c4 = 1.0 - centers[:, 1] - radii
    
    # Overlap constraints: d_ij^2 >= (r_i + r_j)^2
    diff = centers[:, None, :] - centers[None, :, :]
    dist_sq = np.sum(diff**2, axis=2)
    r_sum = radii[:, None] + radii[None, :]
    overlap = dist_sq - r_sum**2
    
    triu_idx = np.triu_indices(n, k=1)
    overlap_triu = overlap[triu_idx]
    
    return np.concatenate([c1, c2, c3, c4, overlap_triu])

def objective_func(x, n):
    return -np.sum(x[2*n:])

def run_packing():
    n = 26
    # Initialize centers in a compact hexagonal pattern
    row_counts = [6, 5, 6, 5, 4]
    centers_init = []
    y = 0.15
    dy = 0.16

        if len(centers_init) >= n:
            break
        x_start = 0.15 + (idx % 2) * 0.085

            if len(centers_init) >= n:
                break
            x = x_start + i * 0.17
            centers_init.append([x, y])
        y += dy

    radii_init = np.full(n, 0.08) + np.arange(n) * 0.0005
    x0 = np.concatenate([centers_init.ravel(), radii_init])
    
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
    
    cons = {'type': 'ineq', 'fun': compute_constraints, 'args': (n,)}
    
    res = minimize(objective_func, x0, args=(n,), bounds=bounds, constraints=cons, method='SLSQP', options={'maxiter': 5000, 'ftol': 1e-12, 'disp': False})
    
    centers = res.x[:2*n].reshape(n, 2)
    radii = res.x[2*n:]

    radii = np.maximum(radii, 1e-9)
    centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
    centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)
    
    # Iterative projection to resolve tiny numerical overlaps
    for _ in range(20):
        changed = False
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))

                    centers[i] -= dir_vec * overlap * 0.501
                    centers[j] += dir_vec * overlap * 0.501
                    changed = True
        if not changed:
            break
            
    return centers, radii, np.sum(radii)
