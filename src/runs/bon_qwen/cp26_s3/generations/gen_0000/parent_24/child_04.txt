1. Random points distributed in the interior.
2. A hexagonal lattice pattern, which naturally approximates high-density packings.
3. Each run starts with small feasible radii to satisfy initial constraints.

import numpy as np
from scipy.optimize import minimize

N = 26

def compute_obj(z):
    """Objective: maximize sum of radii => minimize negative sum."""
    return -np.sum(z[2::3])

def compute_cons(z):
    """
    Computes inequality constraints >= 0:
    - Boundary: center >= radius, 1 - center >= radius
    - Non-overlap: distance_ij >= radius_i + radius_j
    """
    xi = z[0::3]
    yi = z[1::3]
    ri = z[2::3]
    
    c = []
    # Boundary constraints
    c.extend(ri - xi)
    c.extend(ri - (1.0 - xi))
    c.extend(ri - yi)
    c.extend(ri - (1.0 - yi))
    
    # Pairwise distance constraints
    dx = xi[:, None] - xi[None, :]
    dy = yi[:, None] - yi[None, :]
    dist = np.sqrt(dx**2 + dy**2)
    r_sum = ri[:, None] + ri[None, :]
    
    # Extract upper triangle to avoid duplicates and self-constraints
    mask = np.triu(np.ones((N, N), dtype=bool), k=1)
    violations = r_sum - dist
    violations = violations[mask]
    # We need dist - r_sum >= 0  =>  -(r_sum - dist) >= 0
    c.extend(-violations)
    
    return np.array(c)

def run_packing():
    best_sum_r = 0.0
    best_z = None
    
    # Variable bounds: x, y in [0, 1], r in [0, 0.5]
    bounds = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    
    starts = []
    np.random.seed(42)
    
    # Random initial configurations
    for _ in range(6):
        xi = np.random.uniform(0.15, 0.85, N)
        yi = np.random.uniform(0.15, 0.85, N)
        ri = np.full(N, 0.08)
        starts.append(np.concatenate([xi, yi, ri]))
        
    # Hexagonal lattice initial configuration
    pts = []
    spacing = 0.16
    for r_idx in range(8):
        y = 0.08 + r_idx * spacing * np.sqrt(3) / 2
        offset = (r_idx % 2) * spacing * 0.5
        for c_idx in range(8):
            x = 0.08 + offset + c_idx * spacing
            if 0 <= x <= 1 and 0 <= y <= 1:
                pts.append((x, y))
            if len(pts) >= N:
                break
        if len(pts) >= N:
            break
            
    if len(pts) >= N:
        xs, ys = zip(*pts[:N])
        ri = np.full(N, 0.09)
        starts.append(np.concatenate([np.array(xs), np.array(ys), ri]))
        
    # Run optimization from each start
    for z0 in starts:
        try:
            res = minimize(
                compute_obj,
                z0,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'ineq', 'fun': compute_cons},
                options={'maxiter': 2000, 'ftol': 1e-10, 'disp': False}
            )
            if res.success:
                cur_sum = np.sum(res.x[2::3])
                cons_vals = compute_cons(res.x)
                # Accept if constraints are satisfied within tolerance and improves sum
                if cons_vals.min() >= -1e-7 and cur_sum > best_sum_r:
                    best_sum_r = cur_sum
                    best_z = res.x.copy()
        except Exception:
            pass

    # Fallback if all runs failed
    if best_z is None:
        res = minimize(compute_obj, starts[0], method='SLSQP', bounds=bounds, 
                       constraints={'type': 'ineq', 'fun': compute_cons}, options={'maxiter': 1000})
        best_z = res.x
        
    centers = np.column_stack((best_z[0::3], best_z[1::3]))
    radii = best_z[2::3]
    
    # Final validation correction for numerical tolerance
    cons_vals = compute_cons(best_z)
    min_v = cons_vals.min()
    if min_v < 0:
        scale = 1.0 + min_v / (np.mean(radii) + 1e-9)
        if scale < 1.0:
            radii *= scale
            best_sum_r = np.sum(radii)
            
    return centers, radii, best_sum_r
