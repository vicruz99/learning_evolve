import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def get_initial_guess():
    row_counts = [6, 5, 6, 5, 4]
    pts = []
    for i, cnt in enumerate(row_counts):
        for j in range(cnt):
            pts.append([j + (i % 2) * 0.5, i * np.sqrt(3) / 2])
    pts = np.array(pts)
    mins = pts.min(axis=0)
    maxs = pts.max(axis=0)
    ranges = maxs - mins
    return (pts - mins) / ranges * 0.8 + 0.1

def objective(vars):
    return -vars[-1]

def constraints(vars):
    r = vars[-1]
    C = vars[:-1].reshape(-1, 2)
    con = []
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    con.extend(C[:, 0] - r)
    con.extend(1.0 - C[:, 0] - r)
    con.extend(C[:, 1] - r)
    con.extend(1.0 - C[:, 1] - r)
    
    # Pairwise distance constraints: ||Ci - Cj||^2 >= 4r^2
    diff = C[:, np.newaxis, :] - C[np.newaxis, :, :]
    dist_sq = np.sum(diff**2, axis=2)
    idx = np.triu_indices(N_CIRCLES, k=1)
    con.extend(dist_sq[idx] - 4.0 * r**2)
    
    return np.array(con)

def run_packing():
    pts0 = get_initial_guess()
    bounds = [(0.0, 1.0)] * (2 * N_CIRCLES) + [(0.0, 0.5)]
    
    best_res = None
    best_r = -np.inf
    
    for seed in range(5):
        np.random.seed(seed)
        x0 = np.concatenate([pts0.flatten(), [0.04]])
        noise = np.random.randn(2 * N_CIRCLES) * 0.02
        x0[:-1] += noise
        x0 = np.clip(x0[:-1].reshape(-1, 2), 0.05, 0.95).flatten()
        x0 = np.concatenate([x0, [0.04]])
        
        res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                       constraints={'type': 'ineq', 'fun': constraints},
                       options={'maxiter': 5000, 'ftol': 1e-10, 'disp': False})
        
        if res.success:
            r_val = res.x[-1]
            if r_val > best_r:
                best_r = r_val
                best_res = res
                
    if best_res is not None:
        C = best_res.x[:-1].reshape(-1, 2)
        r = best_res.x[-1]
    else:
        C = pts0
        r = 0.04
        
    radii = np.full(N_CIRCLES, r)
    
    # Post-processing to ensure strict validity within tolerances
    for _ in range(10):
        moved = False
        for i in range(N_CIRCLES):
            for j in range(i + 1, N_CIRCLES):
                dist = np.sqrt(np.sum((C[i] - C[j])**2))
                min_dist = radii[i] + radii[j]
                if dist < min_dist - 1e-10:
                    vec = C[i] - C[j]
                    if dist > 1e-12:
                        shift = (min_dist - dist) / 2.0 + 1e-7
                        C[i] += vec / dist * shift
                        C[j] -= vec / dist * shift
                        moved = True
            for dim in [0, 1]:
                if C[i, dim] < radii[i] - 1e-10:
                    C[i, dim] = radii[i]
                    moved = True
                if C[i, dim] > 1.0 - radii[i] + 1e-10:
                    C[i, dim] = 1.0 - radii[i]
                    moved = True
        if not moved:
            break
            
    return C, radii, np.sum(radii)
