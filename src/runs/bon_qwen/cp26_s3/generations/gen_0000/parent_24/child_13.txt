import numpy as np
import warnings
from scipy.optimize import minimize

warnings.filterwarnings("ignore")

def compute_loss_and_grad(z, n, lam):
    """
    Computes the loss and gradient for the circle packing problem.
    z: flattened array of [x1, y1, r1, x2, y2, r2, ...]
    """
    centers = z[:2*n].reshape(n, 2)
    radii = z[2*n:]

    loss = -np.sum(radii)
    grad = np.zeros_like(z)
    grad[2*n:] = -1.0  # Gradient of -sum(radii) w.r.t radii
    grad_c = grad[:2*n].reshape(n, 2)

    # Pairwise overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            diff = centers[i] - centers[j]
            dist_sq = np.dot(diff, diff)
            dist = np.sqrt(dist_sq)
            if dist < 1e-12:
                dist = 1e-12
            
            overlap = radii[i] + radii[j] - dist
            if overlap > 0.0:
                loss += lam * overlap * overlap
                factor = 2.0 * lam * overlap / dist
                
                # Gradient w.r.t centers: push apart
                grad_c[i] -= factor * diff
                grad_c[j] += factor * diff
                
                # Gradient w.r.t radii: penalize large radii on overlap
                grad[2*n + i] += 2.0 * lam * overlap
                grad[2*n + j] += 2.0 * lam * overlap

    # Boundary constraints
    for i in range(n):
        xi, yi = centers[i]
        ri = radii[i]

        # x lower bound
        v = ri - xi
        if v > 0.0:
            loss += lam * v * v
            grad_c[i, 0] -= 2.0 * lam * v
            grad[2*n + i] += 2.0 * lam * v
            
        # x upper bound
        v = xi + ri - 1.0
        if v > 0.0:
            loss += lam * v * v
            grad_c[i, 0] += 2.0 * lam * v
            grad[2*n + i] += 2.0 * lam * v
            
        # y lower bound
        v = ri - yi
        if v > 0.0:
            loss += lam * v * v
            grad_c[i, 1] -= 2.0 * lam * v
            grad[2*n + i] += 2.0 * lam * v
            
        # y upper bound
        v = yi + ri - 1.0
        if v > 0.0:
            loss += lam * v * v
            grad_c[i, 1] += 2.0 * lam * v
            grad[2*n + i] += 2.0 * lam * v

    return loss, grad

def run_packing():
    n = 26
    
    # Structured initial placement: spread points in a grid
    xs = np.linspace(0.15, 0.85, 6)
    ys = np.linspace(0.15, 0.85, 5)
    init_centers = np.array([(x, y) for x in xs for y in ys])[:n]
    init_radii = np.full(n, 0.05)
    
    # Combine and add deterministic jitter to break symmetries
    z0 = np.concatenate([init_centers.ravel(), init_radii])
    z0 += np.arange(len(z0)) * 0.0001
    
    # Bounds: centers in [0,1], radii in [0, 0.5]
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
    
    z_best = z0.copy()
    lam = 10.0
    
    # Sequential penalty method: gradually tighten constraints
    for step in range(10):
        lam *= 3.5
        res = minimize(compute_loss_and_grad, z_best, args=(n, lam),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': 800, 'ftol': 1e-12, 'gtol': 1e-9})
        z_best = res.x
        
    # Extract results
    centers = z_best[:2*n].reshape(n, 2)
    radii = z_best[2*n:]
    
    # Ensure strict non-negativity and validity
    radii = np.maximum(radii, 0.0)
    total_sum = float(np.sum(radii))
    
    return centers, radii, total_sum
