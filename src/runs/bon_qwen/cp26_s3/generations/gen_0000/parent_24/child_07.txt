import numpy as np
from scipy.optimize import minimize

def compute_objective(z, n):
    """
    Computes the barrier objective function for circle packing.
    z is a flattened array: [x0, y0, r0, x1, y1, r1, ...]
    """
    xs = z[0::3]
    ys = z[1::3]
    rs = z[2::3]
    
    # Objective: minimize negative sum of radii
    obj = -np.sum(rs)
    mu = 50.0
    
    # Pairwise non-overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt((xs[i] - xs[j])**2 + (ys[i] - ys[j])**2)
            gap = dist - rs[i] - rs[j]
            # Logarithmic barrier term
            safe_gap = max(gap, 1e-9)
            obj -= mu * np.log(safe_gap)
            # Quadratic penalty for active violations to guide solver back to feasibility
            if gap < 0:
                obj += 1000.0 * (gap ** 2)
                
    # Boundary constraints
    for i in range(n):
        # Margins to four walls
        gaps = [xs[i] - rs[i], 1.0 - xs[i] - rs[i], 
                ys[i] - rs[i], 1.0 - ys[i] - rs[i]]
        for g in gaps:
            safe_g = max(g, 1e-9)
            obj -= mu * np.log(safe_g)
            if g < 0:
                obj += 1000.0 * (g ** 2)
            
    return obj

def run_packing():

    best_radii = None

    for seed in range(8):

                                 for i in range(k) for j in range(k) if i * k + j < n])

        noise = np.random.randn(n, 2) * 0.02
        init_centers = np.clip(base_centers + noise, 0.05, 0.95)
        init_radii = np.full(n, 0.025)
        
        x0 = np.concatenate([init_centers.T, init_radii])
        
        # Bounds: centers in [0,1], radii in [0, 0.4]
        bounds = [(0.0, 1.0) for _ in range(2 * n)] + [(0.0, 0.4) for _ in range(n)]

            res = minimize(compute_objective, x0, args=(n,), method='L-BFGS-B',

                    best_centers = res.x[0:2*n].reshape((n, 2))
                    best_radii = rs.copy()
        except Exception:
            continue
            
    if best_centers is None:
        return np.zeros((n, 2)), np.zeros(n), 0.0

    ys = best_centers[:, 1]
    rs = best_radii

    for _ in range(20):
        overlap_found = False
        for i in range(n):
            for j in range(i + 1, n):
                d = np.sqrt((xs[i] - xs[j])**2 + (ys[i] - ys[j])**2)
                if d < rs[i] + rs[j] - 1e-8:

                    break

                break

    return best_centers, rs, np.sum(rs)
