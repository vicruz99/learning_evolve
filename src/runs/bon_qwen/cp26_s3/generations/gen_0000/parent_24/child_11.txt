To solve the problem of packing 26 circles in a unit square to maximize the sum of radii, we formulate it as a non-linear programming (NLP) problem. We aim to maximize $\sum_{i=1}^{26} r_i$ subject to boundary constraints ($r_i \leq x_i \leq 1-r_i$, $r_i \leq y_i \leq 1-r_i$) and non-overlap constraints ($\|c_i - c_j\| \geq r_i + r_j$).

Strategy:
1. **Formulation**: Define variables as a flat vector $[x_1, y_1, r_1, \dots, x_{26}, y_{26}, r_{26}]$. The objective is to minimize $-\sum r_i$.
2. **Constraints**: Explicitly define inequality constraints for boundaries and pairwise squared distances. Using squared distances avoids non-smooth square roots and is differentiable everywhere.
3. **Optimization**: Use `scipy.optimize.minimize` with the SLSQP method, which handles non-linear inequality constraints efficiently. We provide analytical gradients (Jacobian) for both the objective and constraints to speed up convergence and improve accuracy.
4. **Initialization**: Since the problem is non-convex, local optima exist. We run the optimizer multiple times with different initial configurations:
   - A structured $5 \times 5$ grid.
   - A hexagonal lattice (denser packing).
   - Random perturbations.
   This increases the probability of finding the global optimum or a high-quality local optimum.
5. **Validation**: The constraints are enforced strictly during optimization. The final result is extracted and returned.

The target sum of radii is ~2.636. With $N=26$, average radius $\approx 0.101$. A hexagonal packing density suggests this is achievable. The optimizer will expand radii from a small initial value until constraints are tight.

```python
import numpy as np
from scipy.optimize import minimize

def objective(var, N):
    # var is structured as [x0, y0, r0, x1, y1, r1, ...]
    # We want to maximize sum of radii, so minimize negative sum
    radii = var[2::3]
    return -np.sum(radii)

def objective_jac(var, N):
    jac = np.zeros(3 * N)
    # Gradient w.r.t radii is -1, others 0
    jac[2::3] = -1.0
    return jac

def constraints(var, N):
    c = []
    # Boundary constraints for each circle
    for i in range(N):
        xi, yi, ri = var[3*i], var[3*i+1], var[3*i+2]
        c.append(xi - ri)          # x - r >= 0
        c.append(1.0 - xi - ri)    # 1 - x - r >= 0
        c.append(yi - ri)          # y - r >= 0
        c.append(1.0 - yi - ri)    # 1 - y - r >= 0
    
    # Non-overlap constraints for each pair
    # Using squared distance to avoid square root: dist^2 >= (r1+r2)^2
    for i in range(N):
        for j in range(i + 1, N):
            xi, yi, ri = var[3*i], var[3*i+1], var[3*i+2]
            xj, yj, rj = var[3*j], var[3*j+1], var[3*j+2]
            dx = xi - xj
            dy = yi - yj
            dr = ri + rj
            c.append(dx*dx + dy*dy - dr*dr)
            
    return np.array(c)

def jacobian(var, N):
    M = 4 * N + N * (N - 1) // 2
    jac = np.zeros((M, 3 * N))
    
    # Boundary constraints Jacobian
    for i in range(N):
        b = 4 * i
        # x - r >= 0
        jac[b, 3*i] = 1.0; jac[b, 3*i+2] = -1.0
        # 1 - x - r >= 0
        jac[b+1, 3*i] = -1.0; jac[b+1, 3*i+2] = -1.0
        # y - r >= 0
        jac[b+2, 3*i+1] = 1.0; jac[b+2, 3*i+2] = -1.0
        # 1 - y - r >= 0
        jac[b+3, 3*i+1] = -1.0; jac[b+3, 3*i+2] = -1.0

    # Non-overlap constraints Jacobian
    idx = 4 * N
    for i in range(N):
        for j in range(i + 1, N):
            xi, yi, ri = var[3*i], var[3*i+1], var[3*i+2]
            xj, yj, rj = var[3*j], var[3*j+1], var[3*j+2]
            dx = xi - xj
            dy = yi - yj
            dr = ri + rj

            # Gradient of (x_i-x_j)^2 + (y_i-y_j)^2 - (r_i+r_j)^2
            jac[idx, 3*i] = 2*dx; jac[idx, 3*i+1] = 2*dy; jac[idx, 3*i+2] = -2*dr
            jac[idx, 3*j] = -2*dx; jac[idx, 3*j+1] = -2*dy; jac[idx, 3*j+2] = -2*dr
            idx += 1
            
    return jac

def get_initial_guesses(N):
    guesses = []
    
    # 1. Structured Grid (5x5)
    var = np.zeros(3 * N)
    x_grid = np.linspace(0.15, 0.85, 5)
    y_grid = np.linspace(0.15, 0.85, 5)
    idx = 0
    for x in x_grid:
        for y in y_grid:
            if idx < N:
                var[3*idx] = x
                var[3*idx+1] = y
                var[3*idx+2] = 0.01
                idx += 1
    while idx < N:
        var[3*idx] = np.random.uniform(0.2, 0.8)
        var[3*idx+1] = np.random.uniform(0.2, 0.8)
        var[3*idx+2] = 0.01
        idx += 1
    guesses.append(var)
    
    # 2. Hexagonal Lattice (Denser)
    var = np.zeros(3 * N)
    idx = 0
    s = 0.2 # Spacing
    for r in range(6):
        for c in range(6):
            if idx >= N: break
            x = c * s + 0.15 + (r % 2) * s / 2
            y = r * s * np.sqrt(3) / 2 + 0.15
            var[3*idx] = x
            var[3*idx+1] = y
            var[3*idx+2] = 0.01
            idx += 1
    while idx < N:
        var[3*idx] = np.random.uniform(0.2, 0.8)
        var[3*idx+1] = np.random.uniform(0.2, 0.8)
        var[3*idx+2] = 0.01
        idx += 1
    guesses.append(var)

    # 3. Random Initializations
    for _ in range(3):
        var = np.zeros(3 * N)
        var[0::3] = np.random.uniform(0.1, 0.9, N)
        var[1::3] = np.random.uniform(0.1, 0.9, N)
        var[2::3] = 0.005 # Small initial radius to ensure feasibility
        guesses.append(var)
        
    return guesses

def run_packing():
    N = 26
    bounds = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints, 'jac': jacobian, 'args': (N,)}
    
    best_var = None
    best_obj = -np.inf
    
    guesses = get_initial_guesses(N)
    
    for var0 in guesses:
        try:
            res = minimize(objective, var0, args=(N,),
                          method='SLSQP', jac=objective_jac,
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 3000, 'ftol': 1e-12})
            # res.fun is negative sum of radii
            if res.fun < best_obj:
                best_var = res.x
                best_obj = res.fun
        except Exception:
            continue
            
    if best_var is None:
        # Fallback in case all fail
        var0 = get_initial_guesses(N)[0]
        res = minimize(objective, var0, args=(N,), method='SLSQP', jac=objective_jac, bounds=bounds, constraints=cons)
        best_var = res.x
        
    centers = best_var[:2*N].reshape((N, 2))
    radii = best_var[2::3]
    return centers, radii, np.sum(radii)
```
