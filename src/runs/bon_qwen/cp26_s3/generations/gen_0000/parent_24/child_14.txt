        x0[3*i] = centers[i, 0]
        x0[3*i+1] = centers[i, 1]
        x0[3*i+2] = radii[i]
        
    best_res = None
    best_val = -np.inf
    
    # 2. Optimization
    # Use SLSQP method which handles nonlinear constraints.
    # Run multiple restarts with perturbed initial conditions to escape local optima.
    np.random.seed(42) # Fixed seed for reproducibility
    
    for trial in range(5):
        # Perturb the initial guess
        current_x0 = x0.copy()
        current_x0[0::3] += np.random.uniform(-0.05, 0.05, n)
        current_x0[1::3] += np.random.uniform(-0.05, 0.05, n)
        current_x0[2::3] += np.random.uniform(-0.01, 0.01, n)
        
        # Ensure variables stay within bounds for the optimizer start
        current_x0[0::3] = np.clip(current_x0[0::3], 0.0, 1.0)
        current_x0[1::3] = np.clip(current_x0[1::3], 0.0, 1.0)
        current_x0[2::3] = np.clip(current_x0[2::3], 0.0, 0.5)
        
        try:
            res = minimize(objective, current_x0, method='SLSQP', bounds=bounds,
                           constraints={'type': 'ineq', 'fun': get_constraints},
                           options={'maxiter': 2000, 'ftol': 1e-12, 'disp': False})
            
            # Objective is negative sum of radii
            current_sum = -res.fun
            if current_sum > best_val:
                best_val = current_sum
                best_res = res
        except Exception:
            continue
            
    # 3. Extract results
    if best_res is not None:
        final_x = best_res.x
    else:
        final_x = x0
        
    final_centers = np.column_stack((final_x[0::3], final_x[1::3]))
    final_radii = final_x[2::3]
    
    # 4. Post-processing to ensure strict validity
    # Enforce boundary constraints strictly
    for i in range(n):
        x, y = final_centers[i]
        r = final_radii[i]
        max_r = min(x, 1.0 - x, y, 1.0 - y)
        if r > max_r:
            final_radii[i] = max_r
            
    # Resolve any minor overlaps due to numerical tolerance
    # Iterate a few times to propagate corrections
    for _ in range(10):
        overlap_found = False
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((final_centers[i] - final_centers[j])**2))
                r_sum = final_radii[i] + final_radii[j]
                # Check for overlap with tolerance
                if dist < r_sum - 1e-12:
                    overlap_found = True
                    excess = r_sum - dist
                    # Reduce radii to eliminate overlap
                    reduction = excess / 2.0
                    final_radii[i] -= reduction
                    final_radii[j] -= reduction
                    final_radii[i] = max(0.0, final_radii[i])
                    final_radii[j] = max(0.0, final_radii[j])
        if not overlap_found:
            break

    return final_centers, final_radii, np.sum(final_radii)
