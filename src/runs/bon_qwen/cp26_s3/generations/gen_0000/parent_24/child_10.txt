**Strategy:**
1. **Initialization:** Create multiple diverse starting configurations (staggered hexagonal grid, random placement, perturbed grid) to avoid poor local minima.
2. **Penalty Method:** Transform the constrained problem into an unconstrained one using a penalty function. The objective becomes:
   $F(\mathbf{x}) = -\sum r_i + \lambda \left( \sum_{\text{boundary}} \max(0, \text{violation})^2 + \sum_{i<j} \max(0, r_i+r_j-d_{ij})^2 \right)$
   We minimize $F$ using L-BFGS-B, which efficiently handles box constraints.
3. **Continuation:** Start with a small penalty coefficient $\lambda$ to find a rough feasible layout, then gradually increase $\lambda$ (5 → 20 → 100 → 500 → 2000) to tightly enforce constraints while maximizing radii.
4. **Feasibility Projection:** After optimization, project the result onto the exact feasible set by clamping radii to boundary limits and iteratively scaling overlapping pairs to exactly touch. This guarantees the output passes validation.
5. **Vectorization:** Pairwise distance and overlap calculations are fully vectorized with NumPy for speed, keeping each objective evaluation fast enough for the optimizer.

import numpy as np
from scipy.optimize import minimize

def evaluate_objective(vars, n, penalty):
    """
    Computes the objective function: -sum(radii) + penalty * constraint_violations
    """
    c = vars[:2*n].reshape(n, 2)
    r = vars[2*n:]
    obj = -np.sum(r)
    pen = 0.0

    # Boundary constraints: circle must be inside [0,1]x[0,1]
    pen += np.sum(np.maximum(0, r - c[:, 0])**2)
    pen += np.sum(np.maximum(0, r - (1 - c[:, 0]))**2)
    pen += np.sum(np.maximum(0, r - c[:, 1])**2)
    pen += np.sum(np.maximum(0, r - (1 - c[:, 1]))**2)

    # Pairwise non-overlap constraints
    dx = c[:, None, 0] - c[None, :, 0]
    dy = c[:, None, 1] - c[None, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    
    overlaps = r[:, None] + r[None, :] - dists
    np.fill_diagonal(overlaps, 0.0)
    # Sum squared positive overlaps (divide by 2 to avoid double counting symmetric pairs)
    pen += np.sum(np.maximum(0, overlaps)**2) * 0.5

    return obj + penalty * pen

def project_to_feasible(centers, radii, n):
    """
    Projects an approximate solution onto the strictly feasible set.
    """
    c = centers.copy()
    r = radii.copy()
    
    # Enforce boundary constraints
    for i in range(n):
        r[i] = min(r[i], c[i, 0], 1 - c[i, 0], c[i, 1], 1 - c[i, 1])
        
    # Iteratively resolve pairwise overlaps
    for _ in range(5):
        for i in range(n):
            for j in range(i + 1, n):
                d = np.hypot(c[i, 0] - c[j, 0], c[i, 1] - c[j, 1])
                if r[i] + r[j] > d:
                    max_r = d / 2
                    r[i] = min(r[i], max_r)
                    r[j] = min(r[j], max_r)
                    
    return c, r

def run_packing():
    n = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None

    # Generate diverse initial configurations
    starts = []

    # Config 1: Staggered hexagonal-like grid
    c1 = []
    rows = [6, 5, 6, 5, 4]
    for r_idx, count in enumerate(rows):
        y = 0.12 + r_idx * 0.18
        xs = np.linspace(0.12, 0.88, count)
        if r_idx % 2 == 1:
            xs = xs * 0.85 + 0.08
        for x in xs:
            c1.append([x, y])
    c1 = np.array(c1[:n])
    starts.append(np.concatenate([c1.flatten(), np.full(n, 0.09)]))

    # Config 2: Random initialization
    np.random.seed(42)
    c2 = np.random.uniform(0.15, 0.85, (n, 2))
    starts.append(np.concatenate([c2.flatten(), np.full(n, 0.07)]))

    # Config 3: Perturbed grid
    c3 = c1.copy()
    c3 += np.random.uniform(-0.02, 0.02, c3.shape)
    starts.append(np.concatenate([c3.flatten(), np.full(n, 0.085)]))

    # Bounds: centers in [0,1], radii in [0, 0.5]
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n

    for x0 in starts:
        x_opt = x0.copy()
        # Continuation method: gradually increase penalty strength
        for p in [5, 20, 100, 500, 2000]:
            res = minimize(evaluate_objective, x_opt, args=(n, p), method='L-BFGS-B',
                           bounds=bounds, options={'maxiter': 3000, 'ftol': 1e-14, 'gtol': 1e-12})
            x_opt = res.x

        c_opt = x_opt[:2*n].reshape(n, 2)
        r_opt = x_opt[2*n:]
        
        # Ensure strict feasibility
        c_opt, r_opt = project_to_feasible(c_opt, r_opt, n)
        curr_sum = np.sum(r_opt)

        if curr_sum > best_sum:
            best_sum = curr_sum
            best_centers = c_opt.copy()
            best_radii = r_opt.copy()

    return best_centers, best_radii, best_sum
