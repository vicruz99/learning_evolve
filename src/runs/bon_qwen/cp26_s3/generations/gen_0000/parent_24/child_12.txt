import numpy as np
from scipy.optimize import minimize


def bound_con_low(x, i, n, dim):
    """Constraint: coordinate - radius >= 0"""
    return x[2 * i + dim] - x[n * 2 + i]


def bound_con_high(x, i, n, dim):
    """Constraint: 1 - coordinate - radius >= 0"""
    return 1.0 - x[2 * i + dim] - x[n * 2 + i]


def radius_con(x, i, n):
    """Constraint: radius >= 0"""
    return x[n * 2 + i]


def overlap_con(x, i, j, n):
    """Constraint: distance between centers >= sum of radii"""
    d2 = (x[2 * i] - x[2 * j]) ** 2 + (x[2 * i + 1] - x[2 * j + 1]) ** 2
    return np.sqrt(d2) - x[n * 2 + i] - x[n * 2 + j]


def objective(x, n):
    """Objective: maximize sum of radii (minimize negative)"""
    return -np.sum(x[n * 2:])


def create_constraints(n):
    """Create all constraints for the optimization problem"""
    cons = []
    for i in range(n):
        cons.append({'type': 'ineq', 'fun': bound_con_low, 'args': (i, n, 0)})
        cons.append({'type': 'ineq', 'fun': bound_con_high, 'args': (i, n, 0)})
        cons.append({'type': 'ineq', 'fun': bound_con_low, 'args': (i, n, 1)})
        cons.append({'type': 'ineq', 'fun': bound_con_high, 'args': (i, n, 1)})
        cons.append({'type': 'ineq', 'fun': radius_con, 'args': (i, n)})
    for i in range(n):
        for j in range(i + 1, n):
            cons.append({'type': 'ineq', 'fun': overlap_con, 'args': (i, j, n)})
    return cons


def make_grid_init(n):

    centers = np.zeros((n, 2))
    radii = np.zeros(n)

            centers[idx, 0] = 0.1 + i * 0.2

            idx += 1

    return np.concatenate([centers.flatten(), radii])


def make_hex_init(n):
    """Initialize with hexagonal arrangement"""
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    row_counts = [5, 6, 5, 6, 4]
    r = 0.075
    y = r + 0.03
    idx = 0
    for row_i, count in enumerate(row_counts):
        total_width = count * 2 * r
        x_start = (1.0 - total_width) / 2.0
        if row_i % 2 == 1:
            x_start += r
        for col in range(count):
            centers[idx, 0] = x_start + col * 2 * r
            centers[idx, 1] = y
            radii[idx] = r
            idx += 1
        y += r * np.sqrt(3) + 0.01
    return np.concatenate([centers.flatten(), radii])


def make_uniform_init(n, seed):
    """Initialize with random positions and small radii"""
    np.random.seed(seed)
    centers = np.random.rand(n, 2)
    radii = np.ones(n) * 0.02
    return np.concatenate([centers.flatten(), radii])


def make_staggered_init(n):

    radii = np.zeros(n)

            idx += 1

    return np.concatenate([centers.flatten(), radii])


def refine_with_penalty(x0, n, cons):
    """Refine solution using penalty-based optimization"""
    best_x = x0.copy()
    best_val = objective(x0, n)
    
    for penalty in [100, 500, 1000, 5000, 10000]:
        def pen_obj(x):
            centers = x[:n * 2].reshape(n, 2)
            radii = x[n * 2:]
            obj = -np.sum(radii)
            
            # Boundary penalties
            for i in range(n):
                v = centers[i, 0] - radii[i]
                if v < 0:
                    obj += penalty * v * v
                v = 1.0 - centers[i, 0] - radii[i]
                if v < 0:
                    obj += penalty * v * v
                v = centers[i, 1] - radii[i]
                if v < 0:
                    obj += penalty * v * v
                v = 1.0 - centers[i, 1] - radii[i]
                if v < 0:
                    obj += penalty * v * v
                if radii[i] < 0:
                    obj += penalty * radii[i] * radii[i]
            
            # Overlap penalties
            for i in range(n):
                for j in range(i + 1, n):
                    dist = np.sqrt((centers[i, 0] - centers[j, 0]) ** 2 + 
                                   (centers[i, 1] - centers[j, 1]) ** 2)
                    v = dist - radii[i] - radii[j]
                    if v < 0:
                        obj += penalty * v * v
            
            return obj
        
        try:
            result = minimize(pen_obj, best_x, method='L-BFGS-B',
                            bounds=[(0, 1)] * (n * 2) + [(0, 0.5)] * n,
                            options={'maxiter': 500, 'ftol': 1e-15})
            if result.fun < best_val:
                best_x = result.x
                best_val = result.fun
        except Exception:
            pass
    
    return best_x
