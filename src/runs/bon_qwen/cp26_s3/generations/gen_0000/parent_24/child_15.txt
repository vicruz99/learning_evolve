import numpy as np
from scipy.optimize import minimize

def init_hexagonal(pattern):
    n = sum(pattern)
    n_rows = len(pattern)
    max_cols = max(pattern)
    r_h = 1.0 / (2.0 * max_cols)
    if n_rows > 1:
        r_v = 1.0 / ((n_rows - 1) * np.sqrt(3) + 2)
    else:
        r_v = 0.5
    r_est = min(r_h, r_v) * 0.95
    centers = []
    for row_idx, n_cols in enumerate(pattern):
        y = r_est + row_idx * np.sqrt(3) * r_est
        offset = r_est if row_idx % 2 == 1 else 0.0
        total_width = (n_cols - 1) * 2 * r_est
        available = 1 - 2 * r_est - 2 * offset
        if available < 0:
            available = 0
        x_start = r_est + offset + max(0, available - total_width) / 2.0
        for col in range(n_cols):
            x = x_start + col * 2 * r_est
            centers.append([x, y])
    centers = np.array(centers)
    radii = np.full(n, r_est)
    return centers, radii


def obj_func(x, n):
    radii = x[2 * n:]
    return -np.sum(radii)


def constr_func(x, n):
    centers = x[:2 * n].reshape(n, 2)
    radii = x[2 * n:]
    c = []
    for i in range(n):
        c.append(x[2 * n + i])
        c.append(centers[i, 0] - radii[i])
        c.append(1 - centers[i, 0] - radii[i])
        c.append(centers[i, 1] - radii[i])
        c.append(1 - centers[i, 1] - radii[i])
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy + 1e-16)
            c.append(dist - radii[i] - radii[j])
    return np.array(c)


def force_relax(centers, radii, n, n_iter=200, strength=0.5):
    centers = centers.copy().astype(float)
    radii = radii.copy().astype(float)
    dt = 0.01
    for iteration in range(n_iter):
        forces = np.zeros((n, 2))
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[j, 0] - centers[i, 0]
                dy = centers[j, 1] - centers[i, 1]
                dist = np.sqrt(dx * dx + dy * dy + 1e-16)
                min_dist = radii[i] + radii[j]
                if dist < min_dist:
                    overlap = min_dist - dist
                    fx = (dx / dist) * overlap * strength
                    fy = (dy / dist) * overlap * strength
                    forces[i, 0] -= fx
                    forces[i, 1] -= fy
                    forces[j, 0] += fx
                    forces[j, 1] += fy
            r = radii[i]
            if centers[i, 0] < r:
                forces[i, 0] += (r - centers[i, 0]) * strength
            if centers[i, 0] > 1 - r:
                forces[i, 0] -= (centers[i, 0] - (1 - r)) * strength
            if centers[i, 1] < r:
                forces[i, 1] += (r - centers[i, 1]) * strength
            if centers[i, 1] > 1 - r:
                forces[i, 1] -= (centers[i, 1] - (1 - r)) * strength
        centers += forces * dt
        for i in range(n):
            r = radii[i]
            centers[i, 0] = np.clip(centers[i, 0], r, 1 - r)
            centers[i, 1] = np.clip(centers[i, 1], r, 1 - r)
        decay = 0.995
        strength *= decay
        dt *= 0.998
    return centers


def grow_radii(centers, radii, n, n_iter=50):
    centers = centers.copy().astype(float)
    radii = radii.copy().astype(float)
    for _ in range(n_iter):
        min_gap = float('inf')
        for i in range(n):
            if radii[i] < centers[i, 0] - 1e-8:
                gap = centers[i, 0] - radii[i]
                min_gap = min(min_gap, gap)
            if radii[i] < (1 - centers[i, 0]) - 1e-8:
                gap = 1 - centers[i, 0] - radii[i]
                min_gap = min(min_gap, gap)
            if radii[i] < centers[i, 1] - 1e-8:
                gap = centers[i, 1] - radii[i]

        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy + 1e-16)
                gap = dist - radii[i] - radii[j]
                min_gap = min(min_gap, gap)

            radii += growth
        for i in range(n):
            r = radii[i]
            radii[i] = min(r, centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])
            radii[i] = max(radii[i], 0)
    return radii


def slsqp_optimize(centers, radii, n):
    x0 = np.concatenate([centers.flatten(), radii])
    bounds = [(0, 1)] * (2 * n) + [(1e-6, 0.5)] * n
    cons = {'type': 'ineq', 'fun': lambda x: constr_func(x, n)}
    result = minimize(obj_func, x0, args=(n,), method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 500, 'ftol': 1e-12})
    if result.success or result.fun < -0.1:
        opt_centers = result.x[:2 * n].reshape(n, 2)
        opt_radii = result.x[2 * n:]
        opt_sum = np.sum(opt_radii)
        return opt_centers, opt_radii, opt_sum
    return None


def optimize_from_init(centers, radii, n):

    opt_result = slsqp_optimize(centers, radii, n)
    if opt_result is not None:
        return opt_result
    return None
