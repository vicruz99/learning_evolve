import numpy as np
from scipy.optimize import minimize

def compute_objective(vars, n, C):
    """
    Computes the objective function: -sum(radii) + penalty for violations.
    vars: flattened array [x1, y1, r1, x2, y2, r2, ...]
    """
    cx = vars[0::3]
    cy = vars[1::3]
    cr = vars[2::3]
    
    sum_r = np.sum(cr)
    pen = 0.0
    
    # Boundary penalties: circles must stay within [0,1]
    # x < r  =>  r - x > 0
    pen += np.sum(np.maximum(cr - cx, 0)**2)
    # x > 1-r  =>  x - (1-r) > 0
    pen += np.sum(np.maximum(cx - (1 - cr), 0)**2)
    # y < r
    pen += np.sum(np.maximum(cr - cy, 0)**2)
    # y > 1-r
    pen += np.sum(np.maximum(cy - (1 - cr), 0)**2)
    
    # Pairwise overlap penalties
    # Using broadcasting for efficiency
    cx_mat = cx[:, np.newaxis] - cx[np.newaxis, :]
    cy_mat = cy[:, np.newaxis] - cy[np.newaxis, :]
    dist_mat = np.sqrt(cx_mat**2 + cy_mat**2)
    
    cr_sum = cr[:, np.newaxis] + cr[np.newaxis, :]
    overlap_mat = cr_sum - dist_mat
    
    # Zero out diagonal and lower triangle to count each pair once
    np.fill_diagonal(overlap_mat, 0.0)
    overlap_mat = np.triu(overlap_mat, k=1)
    
    pen += np.sum(np.maximum(overlap_mat, 0)**2)
        
    return -sum_r + C * pen

def optimize_packing(init_vars, n, C):
    """
    Runs L-BFGS-B optimization to maximize sum of radii.
    """
    bounds = [(0.0, 1.0), (0.0, 1.0), (1e-5, 0.5)] * n
    res = minimize(compute_objective, init_vars, args=(n, C), method='L-BFGS-B', 
                   bounds=bounds, options={'maxiter': 2000, 'ftol': 1e-10, 'gtol': 1e-8})
    return res.x

def run_packing():
    n = 26
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Run multiple restarts with different random seeds
    for seed in range(5):
        # 1. Generate hexagonal lattice initialization
        pts = []
        row = 0
        while len(pts) < n:
            y = row * np.sqrt(3)/2
            x_offset = 0.5 if row % 2 == 1 else 0.0
            col = 0
            while len(pts) < n:
                x = col + x_offset
                pts.append([x, y])
                col += 1
            row += 1
        pts = np.array(pts[:n])
        
        # Normalize and center in [0,1] with margin
        mx, my = pts.max(axis=0)
        scale = 0.75 / max(mx, my)
        pts = pts * scale + (1.0 - scale) / 2.0
        
        # Add small random noise to break symmetry
        rng = np.random.RandomState(seed)
        pts += rng.normal(0, 0.005, pts.shape)
        pts = np.clip(pts, 0.05, 0.95)
        
        # Initialize variables
        init_radii = np.full(n, 0.08)
        init_vars = np.zeros(3*n)
        init_vars[0::3] = pts[:, 0]
        init_vars[1::3] = pts[:, 1]
        init_vars[2::3] = init_radii
        
        # 2. Optimize
        vars_opt = optimize_packing(init_vars, n, C=8000.0)
        
        centers = np.zeros((n, 2))
        radii = np.zeros(n)
        centers[:, 0] = vars_opt[0::3]
        centers[:, 1] = vars_opt[1::3]
        radii = vars_opt[2::3]
        
        # 3. Post-process to ensure strict validity
        is_valid = True
        for i in range(n):
            for j in range(i+1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                if dist < radii[i] + radii[j] - 1e-10:
                    is_valid = False
            for coord in centers[i]:
                if coord < radii[i] - 1e-10 or coord > 1 - radii[i] + 1e-10:
                    is_valid = False
                    
        if not is_valid:
            # Uniformly scale down radii slightly to eliminate tiny numerical violations
            radii *= 0.99999
            
        current_sum = np.sum(radii)
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers.copy()
            best_radii = radii.copy()
            
    return best_centers, best_radii, best_sum
