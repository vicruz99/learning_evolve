import numpy as np
import random
from scipy.optimize import linprog

def get_optimal_radii(centers):
    """
    Solves the Linear Programming problem to find the maximum radii
    for a fixed set of centers.
    """
    n = centers.shape[0]
    # Objective: Maximize sum(r_i) -> Minimize -sum(r_i)
    c_obj = -np.ones(n)
    
    A_ub = []
    b_ub = []
    
    # Constraints: r_i <= distance to boundaries
    # r_i <= x_i
    # r_i <= 1 - x_i
    # r_i <= y_i
    # r_i <= 1 - y_i
    for i in range(n):
        # Row vector with 1 at index i, 0 elsewhere
        row = np.zeros(n)
        row[i] = 1.0
        
        # r_i <= x_i
        A_ub.append(row)
        b_ub.append(centers[i, 0])
        
        # r_i <= 1 - x_i
        A_ub.append(row)
        b_ub.append(1.0 - centers[i, 0])
        
        # r_i <= y_i
        A_ub.append(row)
        b_ub.append(centers[i, 1])
        
        # r_i <= 1 - y_i
        A_ub.append(row)
        b_ub.append(1.0 - centers[i, 1])
        
    # Constraints: r_i + r_j <= distance(c_i, c_j)
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            b_ub.append(dist)
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
        else:
            return None, 0.0
    except Exception:
        return None, 0.0

def run_packing():
    n = 26
    best_sum = 0.0

    best_radii = None

                row += 1

            centers += np.random.uniform(-0.01, 0.01, size=centers.shape)
        else:
            # Random initialization inside [0.1, 0.9]
            centers = np.random.uniform(0.1, 0.9, size=(n, 2))
        
        # Optimization loop
        current_centers = centers.copy()
        
        for iteration in range(iterations_per_restart):
            radii, current_sum = get_optimal_radii(current_centers)

                break

                best_centers = current_centers.copy()
                best_radii = radii.copy()

                    forces[i, 1] += 0.05
                # Top

                    forces[i, 1] -= 0.05
