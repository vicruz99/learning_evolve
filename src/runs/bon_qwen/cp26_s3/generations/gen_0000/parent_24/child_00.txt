import numpy as np
from scipy.optimize import minimize


def make_overlap_constraint(i, j, n):
    """Create a non-overlap constraint function for circles i and j."""
    def constraint(x):
        dx = x[2 * i] - x[2 * j]
        dy = x[2 * i + 1] - x[2 * j + 1]
        dist = np.sqrt(dx * dx + dy * dy)
        r_sum = x[2 * n + i] + x[2 * n + j]
        return dist - r_sum
    return constraint


def make_boundary_constraint_left(i, n):
    """x >= r for circle i."""
    def constraint(x):
        return x[2 * i] - x[2 * n + i]
    return constraint


def make_boundary_constraint_right(i, n):
    """x + r <= 1 for circle i."""
    def constraint(x):
        return 1.0 - x[2 * i] - x[2 * n + i]
    return constraint


def make_boundary_constraint_bottom(i, n):
    """y >= r for circle i."""
    def constraint(x):
        return x[2 * i + 1] - x[2 * n + i]
    return constraint


def make_boundary_constraint_top(i, n):
    """y + r <= 1 for circle i."""
    def constraint(x):
        return 1.0 - x[2 * i + 1] - x[2 * n + i]
    return constraint


def make_radius_nonneg(i, n):
    """r >= 0 for circle i."""
    def constraint(x):
        return x[2 * n + i]
    return constraint


def build_constraints(n):
    """Build all constraints for the optimization."""
    constraints = []
    
    # Boundary constraints for each circle
    for i in range(n):
        constraints.append({'type': 'ineq', 'fun': make_boundary_constraint_left(i, n)})
        constraints.append({'type': 'ineq', 'fun': make_boundary_constraint_right(i, n)})
        constraints.append({'type': 'ineq', 'fun': make_boundary_constraint_bottom(i, n)})
        constraints.append({'type': 'ineq', 'fun': make_boundary_constraint_top(i, n)})
        constraints.append({'type': 'ineq', 'fun': make_radius_nonneg(i, n)})
    
    # Non-overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({'type': 'ineq', 'fun': make_overlap_constraint(i, j, n)})
    
    return constraints

def random_initialization(n, seed=42):
    """Create random initialization."""
    np.random.seed(seed)
    centers = np.random.uniform(0.15, 0.85, size=(n, 2))
    return [centers]


def optimize_packing(centers_init, radii_init, n, constraints, bounds):
    """Run optimization with given initial conditions."""
    x0 = np.concatenate([centers_init.flatten(), radii_init])
    
    def objective(x):
        return -np.sum(x[2 * n:])
    
    try:
        result = minimize(
            objective,
            x0,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={
                'maxiter': 3000,
                'ftol': 1e-14,
                'disp': False
            }
        )
        
        if result.success or result.fun < 0:
            centers_opt = result.x[:2 * n].reshape(n, 2)
            radii_opt = result.x[2 * n:]
            return centers_opt, radii_opt, np.sum(radii_opt), result.success
    except Exception:
        pass
    
    return None, None, 0, False
