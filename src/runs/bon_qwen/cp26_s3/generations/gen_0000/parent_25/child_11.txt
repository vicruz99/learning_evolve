import numpy as np
from scipy.optimize import minimize

def objective(v):
    """Objective: minimize negative sum of radii (equivalent to maximizing sum)"""
    return -np.sum(v[2::3])

def boundary_con(v):
    """Boundary constraints: circles must stay within [0,1]x[0,1]"""
    n = v.shape[0] // 3
    xs = v[0::3]
    ys = v[1::3]
    rs = v[2::3]
    return np.concatenate([
        xs - rs,          # x >= r
        1.0 - xs - rs,    # x + r <= 1
        ys - rs,          # y >= r
        1.0 - ys - rs     # y + r <= 1
    ])

def pairwise_con(v):
    """Non-overlap constraints: squared distance >= squared sum of radii"""
    n = v.shape[0] // 3
    xs = v[0::3]
    ys = v[1::3]
    rs = v[2::3]
    cons = []
    for i in range(n):
        xi, yi, ri = xs[i], ys[i], rs[i]
        for j in range(i + 1, n):
            d2 = (xi - xs[j])**2 + **(yi - ys[j])2
            sum_r2 = (ri + rs[j])**2
            cons.append(d2 - sum_r2)
    return np.array(cons)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # 1. Initial configuration: hexagonal-like grid for high packing density
    xs, ys = [], []
    y_vals = np.linspace(0.2, 0.8, 5)
    for i, y in enumerate(y_vals):
        n_pts = 6 if i == 0 else 5
        x_vals = np.linspace(0.1, 0.9, n_pts)
        # Shift odd rows for hexagonal packing
        if i % 2 == 1:
            x_vals = (x_vals[:-1] + x_vals[1:]) / 2.0
        xs.extend(x_vals)
        ys.extend([y] * n_pts)
        
    centers0 = np.column_stack((xs, ys))
    r0 = 0.08 * np.ones(n)
    
    # Flatten to optimization vector: [x0, y0, r0, x1, y1, r1, ...]
    v0 = np.empty(3 * n)
    for i in range(n):
        v0[3*i] = centers0[i, 0]
        v0[3*i+1] = centers0[i, 1]
        v0[3*i+2] = r0[i]
        
    # Variable bounds
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * n
    
    # Constraints setup
    constraints = [
        {'type': 'ineq', 'fun': boundary_con},
        {'type': 'ineq', 'fun': pairwise_con}
    ]
    
    # 2. Optimization
    res = minimize(objective, v0, method='SLSQP', bounds=bounds, constraints=constraints,
                   options={'maxiter': 2000, 'ftol': 1e-12, 'disp': False})
                   
    v_opt = res.x
    
    # 3. Extract and clean results
    centers = np.column_stack((v_opt[0::3], v_opt[1::3]))
    radii = v_opt[2::3]
    
    # Handle negligible numerical drift
    radii = np.maximum(radii, 0.0)
    centers = np.clip(centers, 0.0, 1.0)
    
    sum_radii = float(np.sum(radii))
    
    return centers, radii, sum_radii
