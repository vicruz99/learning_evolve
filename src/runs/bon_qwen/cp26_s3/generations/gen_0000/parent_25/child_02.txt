def objective_func(vars, n):
    """Objective function to minimize: -sum(radii)"""

    radii = vars[2::3]
    return -np.sum(radii)

def inequality_constraints(vars, n):
    """Returns array of constraint values. We want g(vars) >= 0.
    scipy minimize expects constraints as dictionaries or functions returning >= 0 (for 'eq') or >= 0 (for 'ineq' depending on definition).
    Actually, 'ineq' in scipy.optimize.minimize means constraint fun(x) >= 0.
    
    Constraints:
    1. x_i - r_i >= 0
    2. 1 - x_i - r_i >= 0
    3. y_i - r_i >= 0
    4. 1 - y_i - r_i >= 0
    5. dist_ij - (r_i + r_j) >= 0
    """
    centers = vars[:2*n].reshape(n, 2)
    radii = vars[2*n:]
    
    cons = []
    
    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        cons.append(x - r)
        cons.append(1.0 - x - r)
        cons.append(y - r)
        cons.append(1.0 - y - r)
        
    # Overlap constraints
    # Vectorized calculation
    # dist^2 - (r_i + r_j)^2 >= 0
    # To avoid sqrt, we can check squared distances, but the constraint is linear in r for LP, quadratic here.
    # Actually, sqrt(dist^2) >= r_i + r_j.
    # dist >= r_i + r_j.
    
    # Let's compute dist matrix
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.linalg.norm(diff, axis=2)
    
    # We only need upper triangle
    for i in range(n):
        for j in range(i + 1, n):
            cons.append(dists[i, j] - (radii[i] + radii[j]))
            
    return np.array(cons)
