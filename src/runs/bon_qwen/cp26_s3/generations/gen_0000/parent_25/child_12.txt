import numpy as np
from scipy.optimize import minimize

def objective(x, n):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints(x, n):
    """
    Inequality constraints g(x) >= 0.
    Returns an array of constraint values.
    """
    n_con = 5 * n + n * (n - 1) // 2
    con = np.empty(n_con)
    idx = 0
    
    # Boundary and positive radius constraints
    for i in range(n):
        r = x[3*i + 2]
        con[idx] = x[3*i] - r           # x_i - r_i >= 0
        idx += 1
        con[idx] = 1.0 - x[3*i] - r     # 1 - x_i - r_i >= 0
        idx += 1
        con[idx] = x[3*i + 1] - r       # y_i - r_i >= 0
        idx += 1
        con[idx] = 1.0 - x[3*i + 1] - r # 1 - y_i - r_i >= 0
        idx += 1
        con[idx] = r - 1e-5             # r_i >= epsilon
        idx += 1

    # Pairwise non-overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            dx = x[3*i] - x[3*j]
            dy = x[3*i + 1] - x[3*j + 1]
            rs = x[3*i + 2] + x[3*j + 2]
            con[idx] = dx*dx + dy*dy - rs*rs # dist^2 - (r_i+r_j)^2 >= 0
            idx += 1
            
    return con

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None

    # Bounds: x,y in [0,1], r in [1e-5, 0.5]
    bounds = [(0.0, 1.0), (0.0, 1.0), (1e-5, 0.5)] * n
    cons = {'type': 'ineq', 'fun': constraints, 'args': (n,)}

    # Run multiple trials with different initializations to escape local minima
    for trial in range(6):
        rng = np.random.default_rng(trial * 12345 + 42)
        
        # Start with a reasonable radius estimate
        r_init = 0.055 + trial * 0.005
        
        # Hexagonal-ish grid initialization
        cols = int(np.ceil(np.sqrt(n)))
        rows = int(np.ceil(n / cols))
        xs = np.linspace(r_init + 0.03, 1.0 - r_init - 0.03, cols)
        ys = np.linspace(r_init + 0.03, 1.0 - r_init - 0.03, rows)
        X, Y = np.meshgrid(xs, ys)
        centers = np.column_stack((X.ravel(), Y.ravel()))[:n]
        radii = np.full(n, r_init)

        # Add perturbation
        centers += rng.normal(0, 0.008, centers.shape)
        centers = np.clip(centers, r_init, 1.0 - r_init)

        x0 = np.concatenate([centers.flatten(), radii])

        res = minimize(
            objective, x0, args=(n,),
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 2500, 'ftol': 1e-9, 'disp': False}
        )

        current_sum = -res.fun
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = res.x[:2*n].reshape(n, 2).copy()
            best_radii = res.x[2*n:].copy()

    # Apply a tiny shrink to guarantee validation passes despite numerical precision limits
    shrink_factor = 1.0 - 1e-8
    best_radii *= shrink_factor
    final_sum = float(np.sum(best_radii))

    return best_centers, best_radii, final_sum
