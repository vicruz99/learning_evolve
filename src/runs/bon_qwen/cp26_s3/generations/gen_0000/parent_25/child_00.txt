import numpy as np
from scipy.optimize import minimize


def softplus(x, beta=200.0):
    """Smooth approximation of max(0, x) that is differentiable everywhere."""
    if x > 10 / beta:
        return float(x)
    if x < -10 / beta:
        return 0.0
    return float(np.log1p(np.exp(beta * x))) / beta


def compute_objective(x, n, penalty_weight=10000.0):
    """
    Compute objective: negative sum of radii plus smooth penalty for violations.
    
    Args:
        x: array of shape (2*n + n,) = (3*n,) with centers and radii
        n: number of circles
        penalty_weight: weight for constraint violation penalties
    
    Returns:
        objective value (to minimize)
    """
    centers = x[:2 * n].reshape((n, 2))
    radii = x[2 * n:]
    
    # Negative sum of radii (we want to maximize sum)
    obj = -np.sum(radii)
    
    penalty = 0.0
    
    # Boundary penalties
    for i in range(n):
        for d in range(2):
            # Violation if center[d] - radius < 0
            v1 = -centers[i, d] + radii[i]
            if v1 > 0:
                penalty += v1 * v1
            # Violation if center[d] + radius > 1
            v2 = centers[i, d] + radii[i] - 1.0
            if v2 > 0:
                penalty += v2 * v2
    
    # Overlap penalties
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            overlap = radii[i] + radii[j] - dist
            if overlap > 0:
                penalty += overlap * overlap
    
    return obj + penalty_weight * penalty


def compute_forces(centers, radii, n):
    """
    Compute repulsion forces to resolve overlaps and boundary violations.
    
    Args:
        centers: array of shape (n, 2)
        radii: array of shape (n,)
        n: number of circles
    
    Returns:
        forces: array of shape (n, 2)
    """
    forces = np.zeros((n, 2))
    
    # Pair repulsion forces
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist_sq = dx * dx + dy * dy
            dist = np.sqrt(dist_sq)
            
            min_dist = radii[i] + radii[j]
            if dist < min_dist and dist > 1e-10:
                overlap = min_dist - dist
                force_mag = overlap / (dist * dist + 1e-10)
                forces[i, 0] += force_mag * dx
                forces[i, 1] += force_mag * dy
                forces[j, 0] -= force_mag * dx
                forces[j, 1] -= force_mag * dy
    
    # Boundary repulsion forces
    for i in range(n):
        for d in range(2):
            if centers[i, d] - radii[i] < 0:
                forces[i, d] += (radii[i] - centers[i, d]) * 50.0
            if centers[i, d] + radii[i] > 1.0:
                forces[i, d] -= (centers[i, d] + radii[i] - 1.0) * 50.0
    
    return forces


def initialize_hexagonal(n, r_init=0.08):
    """
    Initialize circle centers in a hexagonal grid pattern.
    
    Args:
        n: number of circles
        r_init: initial radius estimate for spacing
    
    Returns:
        centers: array of shape (n, 2)
    """
    centers = np.zeros((n, 2))
    idx = 0
    
    for row in range(10):
        for col in range(6):
            if idx >= n:
                break
            
            if row % 2 == 0:
                x = r_init + col * 2 * r_init
            else:
                x = 2 * r_init + col * 2 * r_init
            y = r_init + row * r_init * np.sqrt(3)
            
            if x + r_init <= 1.0 and y + r_init <= 1.0:
                centers[idx] = [x, y]
                idx += 1
        if idx >= n:
            break
    
    return centers[:n]


def initialize_random(n, seed=42):
    """
    Initialize circle centers at random positions.
    
    Args:
        n: number of circles
        seed: random seed
    
    Returns:
        centers: array of shape (n, 2)
    """
    rng = np.random.RandomState(seed)
    centers = rng.uniform(0.1, 0.9, size=(n, 2))
    return centers


def force_directed_optimization(centers, radii, n, n_iters=20000):
    """
    Run force-directed optimization to resolve overlaps.
    
    Args:
        centers: array of shape (n, 2)
        radii: array of shape (n,)
        n: number of circles
        n_iters: number of iterations
    
    Returns:
        optimized centers
    """
    centers = centers.copy()
    
    for iteration in range(n_iters):
        forces = compute_forces(centers, radii, n)
        
        # Adaptive learning rate
        lr = 1e-3 / (1.0 + iteration * 5e-5)
        
        centers += lr * forces
        
        # Enforce boundary constraints
        for i in range(n):
            for d in range(2):
                centers[i, d] = max(radii[i], min(1.0 - radii[i], centers[i, d]))
    
    return centers


def grow_radii(centers, radii, n, max_iters=50000):
    """
    Iteratively grow radii while maintaining valid positions.
    
    Args:
        centers: array of shape (n, 2)
        radii: array of shape (n,)
        n: number of circles
        max_iters: maximum iterations
    
    Returns:
        updated centers and radii
    """
    centers = centers.copy()
    radii = radii.copy()
    
    for iteration in range(max_iters):
        # Compute minimum margin for growth
        min_margin = 1.0
        
        for i in range(n):
            # Check boundaries
            for d in range(2):
                min_margin = min(min_margin, centers[i, d] - radii[i])
                min_margin = min(min_margin, 1.0 - centers[i, d] - radii[i])
            
            # Check overlaps
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                min_margin = min(min_margin, dist - radii[i] - radii[j])
        
        if min_margin > 1e-8:
            # Grow all radii uniformly
            dr = min(0.0005, min_margin / 2.0)
            radii += dr
        else:
            # Try to resolve overlaps by moving circles
            forces = compute_forces(centers, radii, n)
            lr = 1e-4 / (1.0 + iteration * 1e-5)
            centers += lr * forces
            
            # Enforce boundaries
            for i in range(n):
                for d in range(2):
                    centers[i, d] = max(radii[i], min(1.0 - radii[i], centers[i, d]))
    
    return centers, radii


def fine_tune(centers, radii, n):
    """
    Fine-tune positions and radii using L-BFGS-B optimization.
    
    Args:
        centers: array of shape (n, 2)
        radii: array of shape (n,)
        n: number of circles
    
    Returns:
        optimized centers and radii
    """
    x0 = np.concatenate([centers.flatten(), radii])
    
    bounds = []
    for _ in range(2 * n):
        bounds.append((0.0001, 0.9999))
    for _ in range(n):
        bounds.append((0.0001, 0.5))
    
    result = minimize(
        compute_objective,
        x0,
        args=(n, 50000.0),
        method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': 50000, 'ftol': 1e-16, 'gtol': 1e-14}
    )
    
    centers_opt = result.x[:2 * n].reshape((n, 2))
    radii_opt = result.x[2 * n:]
    
    return centers_opt, radii_opt


def compute_true_radii(centers, n):
    """
    Compute the true maximum radius for each circle given fixed positions.
    
    Args:
        centers: array of shape (n, 2)
        n: number of circles
    
    Returns:
        radii: array of shape (n,)
    """
    radii = np.zeros(n)
    
    for i in range(n):
        max_r = min(centers[i, 0], 1.0 - centers[i, 0],
                    centers[i, 1], 1.0 - centers[i, 1])
        
        for j in range(n):
            if i != j:
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                max_r = min(max_r, dist)
        
        radii[i] = max(0.0, max_r / 2.0)
    
    return radii


def run_packing():
    """
    Pack 26 circles in a unit square to maximize sum of radii.
    
    Returns:
        centers: np.array of shape (26, 2)
        radii: np.array of shape (26,)
        sum_radii: float
    """
    n = 26
    
    best_centers = None
    best_radii = None
    best_sum = 0.0
    
    # Try multiple initializations
    initializations = []
    
    # Hexagonal initialization with different parameters
    for r_init in [0.07, 0.08, 0.09, 0.10]:
        initializations.append(initialize_hexagonal(n, r_init))
    
    # Random initializations
    for seed in [42, 123, 456, 789]:
        initializations.append(initialize_random(n, seed))
    
    for init_idx, centers in enumerate(initializations):
        # Start with small radii
        radii = np.full(n, 0.01)
        
        # Phase 1: Force-directed relaxation to resolve any initial overlaps
        centers = force_directed_optimization(centers, radii, n, n_iters=5000)
        
        # Phase 2: Grow radii iteratively
        centers, radii = grow_radii(centers, radii, n, max_iters=30000)
        
        # Phase 3: Force-directed relaxation again
        centers = force_directed_optimization(centers, radii, n, n_iters=10000)
        
        # Phase 4: Fine-tune with L-BFGS-B
        centers, radii = fine_tune(centers, radii, n)
        
        # Phase 5: Compute true maximum radii for final positions
        # This ensures no overlaps by definition
        true_radii = compute_true_radii(centers, n)
        
        sum_radii = np.sum(true_radii)
        
        if sum_radii > best_sum:
            best_sum = sum_radii
            best_centers = centers.copy()
            best_radii = true_radii.copy()
    
    # Additional fine-tuning on best solution
    centers, radii = fine_tune(best_centers, best_radii, n)
    radii = compute_true_radii(centers, n)
    sum_radii = np.sum(radii)
    
    # If this improved, use it
    if sum_radii > best_sum:
        best_centers = centers
        best_radii = radii
        best_sum = sum_radii
    
    return best_centers, best_radii, best_sum
