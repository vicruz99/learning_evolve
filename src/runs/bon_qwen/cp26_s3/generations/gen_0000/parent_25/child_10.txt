import numpy as np
from scipy.optimize import minimize

def obj_func(x):
    return -np.sum(x[2::3])

def bnd_x_min(x, i):
    return x[3*i] - x[3*i+2]

def bnd_x_max(x, i):
    return 1.0 - x[3*i] - x[3*i+2]

def bnd_y_min(x, i):
    return x[3*i+1] - x[3*i+2]

def bnd_y_max(x, i):
    return 1.0 - x[3*i+1] - x[3*i+2]

def pair_dist(x, i, j):
    dx = x[3*i] - x[3*j]
    dy = x[3*i+1] - x[3*j+1]
    dr = x[3*i+2] + x[3*j+2]
    return dx*dx + dy*dy - dr*dr

def run_packing():
    N = 26
    x0 = np.zeros(N * 3)
    idx = 0
    for i in range(5):
        for j in range(5):
            x0[idx] = 0.125 + 0.2 * i
            x0[idx+1] = 0.125 + 0.2 * j
            x0[idx+2] = 0.1
            idx += 1
    x0[idx] = 0.5
    x0[idx+1] = 0.5
    x0[idx+2] = 0.05
    
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N
    
    constraints = []
    for i in range(N):
        constraints.append({'type': 'ineq', 'fun': bnd_x_min, 'args': (i,)})
        constraints.append({'type': 'ineq', 'fun': bnd_x_max, 'args': (i,)})
        constraints.append({'type': 'ineq', 'fun': bnd_y_min, 'args': (i,)})
        constraints.append({'type': 'ineq', 'fun': bnd_y_max, 'args': (i,)})
        
    for i in range(N):
        for j in range(i + 1, N):
            constraints.append({'type': 'ineq', 'fun': pair_dist, 'args': (i, j)})
            
    res = minimize(obj_func, x0, method='SLSQP', bounds=bounds, constraints=constraints, options={'maxiter': 1000, 'ftol': 1e-12})
    
    centers = res.x.reshape(N, 3)[:, :2]
    radii = res.x.reshape(N, 3)[:, 2]
    
    radii = np.maximum(radii, 0.0)
    centers = np.clip(centers, 0.0, 1.0)
    
    return centers, radii, float(np.sum(radii))
