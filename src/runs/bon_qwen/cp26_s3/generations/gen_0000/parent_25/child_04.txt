import numpy as np

def get_max_radius(centers):
    """
    Calculates the maximum radius r such that circles with given centers
    do not overlap and stay within the unit square.
    Assumes all circles have the same radius r.
    """
    n = centers.shape[0]
    if n == 0:
        return 0.0
    
    # Distance to walls
    # For a circle at (x, y) to be inside [0,1]^2 with radius r:
    # r <= x <= 1-r  =>  r <= x and r <= 1-x
    # r <= y <= 1-r  =>  r <= y and r <= 1-y
    # So r <= min(x, 1-x, y, 1-y)
    
    x = centers[:, 0]
    y = centers[:, 1]
    
    dists_to_walls = np.array([x, 1 - x, y, 1 - y])
    min_dist_to_wall = np.min(dists_to_walls)
    
    # Distance between circles
    # For circles i and j with radius r:
    # dist(i, j) >= 2r  =>  r <= dist(i, j) / 2
    
    # Compute pairwise distances
    # Using broadcasting
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    # Ignore diagonal (distance to self)
    np.fill_diagonal(dists, np.inf)
    min_dist_between = np.min(dists) / 2.0
    
    return min(min_dist_to_wall, min_dist_between)
