import numpy as np
from scipy.optimize import minimize

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    n = centers.shape[0]
    if np.isnan(centers).any() or np.isnan(radii).any():
        return False
    
    for i in range(n):
        if radii[i] < 0:
            return False
        
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            return False

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False
    return True

def get_initial_valid_packing(n, seed):
    rng = np.random.RandomState(seed)
    centers = rng.uniform(0.05, 0.95, (n, 2))
    radii = np.full(n, 0.01)
    
    # Iterative expansion to find a valid initial state
    for _ in range(50):
        for i in range(n):
            max_r = 1.0
            # Boundary constraints
            max_r = min(max_r, centers[i, 0], 1 - centers[i, 0], 
                             centers[i, 1], 1 - centers[i, 1])
            # Overlap constraints
            for j in range(n):
                if i != j:
                    dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                    max_r = min(max_r, dist - radii[j])
            
            if max_r > radii[i]:
                radii[i] = max(radii[i] * 1.05, radii[i] + 0.001)
            elif max_r < radii[i]:
                radii[i] = max_r

        # A small random jitter helps escape local traps
        centers += rng.normal(0, 0.001, centers.shape)
        centers = np.clip(centers, 0.01, 0.99)
        
    return centers, radii

def objective(params, n):
    # params = [x1, y1, r1, x2, y2, r2, ...]
    # But let's keep centers and radii separate for easier constraint handling
    # Here we use a flat array: first 2*n are centers, next n are radii
    radii = params[2*n:]
    return -np.sum(radii)

            constraints.append({'type': 'ineq', 'fun': lambda p, i=i: p[x_idx] - p[r_idx]})
            # 1 - x - r >= 0
            constraints.append({'type': 'ineq', 'fun': lambda p, i=i: 1.0 - p[x_idx] - p[r_idx]})
