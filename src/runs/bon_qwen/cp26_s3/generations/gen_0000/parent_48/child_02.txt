To maximize the sum of radii for 26 circles in a unit square, we employ a hybrid optimization strategy combining Linear Programming (LP) and local center refinement.

1. **Inner Optimization (LP):** For any fixed set of centers, the problem of finding radii that maximize the sum is a Linear Program. The constraints are linear ($r_i + r_j \le d_{ij}$ and $r_i \le \text{boundary}$), and the objective is linear ($\max \sum r_i$). This solver guarantees the optimal radii for a specific configuration.
2. **Outer Optimization (Force-Directed):** The LP solver identifies "tight" constraints (pairs where circles are touching). We use a physics-based heuristic to move centers apart from these tight pairs, effectively increasing distances ($d_{ij}$) and allowing the radii to grow in the next iteration.
3. **Initialization:** We start with a hexagonal grid layout, which is known for high packing density, and then refine positions to fill gaps and optimize radii.
4. **Execution:** We iterate between solving for radii and nudging centers. Finally, we return the validated configuration.

import numpy as np
from scipy.optimize import linprog

def get_boundary_limits(centers):
    """Calculates the max radius allowed by the unit square boundaries for each center."""
    # Limits based on x and y coords: min(x, 1-x, y, 1-y)
    x_lim = np.minimum(centers[:, 0], 1 - centers[:, 0])
    y_lim = np.minimum(centers[:, 1], 1 - centers[:, 1])
    return np.minimum(x_lim, y_lim)

def solve_radii(centers):
    """
    Solves the LP to find optimal radii for fixed centers.
    Maximizes sum(r_i) subject to:
      r_i + r_j <= dist(i, j)
      r_i <= boundary_limit(i)
      r_i >= 0
    """
    n = centers.shape[0]
    
    # Calculate pairwise distances
    dist_matrix = np.zeros((n, n))
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist_matrix = np.sqrt(np.sum(diff**2, axis=2))
    
    # Boundary limits
    b_lim = get_boundary_limits(centers)
    
    # LP Setup: Minimize -sum(r_i)
    c_obj = np.ones(n) * -1.0
    
    # Inequality constraints: A_ub @ r <= b_ub
    # 1. r_i <= b_lim_i  -> A_row has 1 at col i
    # 2. r_i + r_j <= dist_ij -> A_row has 1 at col i, 1 at col j
    
    n_constraints = n + n * (n - 1) // 2
    A_ub = np.zeros((n_constraints, n))
    b_ub = np.zeros(n_constraints)
    
    row_idx = 0
    
    # Boundary constraints
    for i in range(n):
        A_ub[row_idx, i] = 1.0
        b_ub[row_idx] = b_lim[i]
        row_idx += 1
        
    # Pairwise constraints
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[row_idx, i] = 1.0
            A_ub[row_idx, j] = 1.0
            b_ub[row_idx] = dist_matrix[i, j]
            row_idx += 1
            
    # Bounds for r_i: [0, None]
    bounds = [(0, None) for _ in range(n)]
    
    # Solve LP
    # Using high tolerance to ensure precision
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        radii = res.x
        return radii, res.fun * -1 # Return radii and sum
    else:
        # Fallback if LP fails (should be rare)
        return np.zeros(n), 0.0

def optimize_packing():
    """
    Iterative optimization:
    1. Solve LP for radii.
    2. Identify tight pairs (constraints where r_i + r_j ~ dist).
    3. Move centers to increase distance for tight pairs.
    4. Repeat.
    """
    n = 26
    # Initialize centers in a hexagonal grid pattern
    centers = np.zeros((n, 2))
    
    # Parameters for hexagonal grid
    # Approximate spacing to fit 26 circles
    # 5x5 grid is 25 circles. We'll add one.
    # Let's try a dense hex grid.
    # Spacing dx = 0.2, dy = 0.2 * sqrt(3)/2 approx 0.1732
    
    # We will place centers and let the optimizer fix it.
    # Simple grid start
    idx = 0
    for r in range(6):
        for c in range(5):
            if idx >= n: break
            x = 0.15 + c * 0.18
            y = 0.15 + r * 0.16
            # Shift every other row for hex pattern
            if r % 2 == 1:
                x += 0.09
            centers[idx] = [x, y]
            idx += 1
            
    # Perturb slightly to break symmetry
    centers += np.random.uniform(-0.01, 0.01, centers.shape)
    
    # Run iterations
    num_iterations = 200
    learning_rate = 0.005
    
    best_sum_radii = 0
    best_centers = centers.copy()
    best_radii = np.zeros(n)
    
    # Solve initial radii
    radii, current_sum = solve_radii(centers)
    best_sum_radii = current_sum
    best_radii = radii.copy()
    best_centers = centers.copy()
    
    for iter_idx in range(num_iterations):
        # Solve LP to get optimal radii for current centers
        radii, current_sum = solve_radii(centers)
        
        if current_sum > best_sum_radii:
            best_sum_radii = current_sum
            best_radii = radii.copy()
            best_centers = centers.copy()
            
        # Calculate forces to move centers apart
        # We want to maximize distances between circles with tight constraints
        forces = np.zeros_like(centers)
        
        dist_matrix = np.zeros((n, n))
        diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
        dist_matrix = np.sqrt(np.sum(diff**2, axis=2))
        np.fill_diagonal(dist_matrix, np.inf) # Avoid self
        
        # Calculate pressure from tight pairs
        # A pair is tight if dist < r_i + r_j + tolerance
        # We add a force proportional to the overlap or how close it is to limit
        # Actually, since LP ensures r_i + r_j <= dist, the "tightness" is determined
        # by the dual variables or just by checking if constraint is active.
        # A simple heuristic: push away if dist is close to sum of radii.
        
        # We use the radii obtained from LP.
        # If r_i + r_j is very close to dist(i,j), it's a binding constraint.
        # Increasing dist(i,j) allows increasing r_i + r_j.
        
        tightness_threshold = 1e-4
        interaction_strength = 0.5 # Factor for force magnitude
        
        for i in range(n):
            for j in range(i + 1, n):
                d = dist_matrix[i, j]
                r_sum = radii[i] + radii[j]
                
                # Calculate how "active" this constraint is
                # If r_sum is close to d, we need to increase d.
                # Slack = d - r_sum
                slack = d - r_sum
                
                # We want to push i and j apart if slack is small.
                # Force magnitude inversely proportional to slack (but clipped)
                if slack < 0.1: # Only interact if somewhat close
                    # Force direction: vector from j to i
                    vec = centers[i] - centers[j]
                    norm = np.linalg.norm(vec)
                    if norm > 1e-9:
                        force_vec = vec / norm
                        
                        # Heuristic force strength
                        # If slack is 0, force is max. If slack is large, force is 0.
                        strength = max(0, (0.1 - slack)) * interaction_strength
                        
                        forces[i] += strength * force_vec
                        forces[j] -= strength * force_vec

        # Also add repulsion from boundaries to keep centers valid
        # Centers must stay in [0,1]x[0,1]
        # If a center is too close to boundary, push it in?
        # Actually, LP handles boundary constraints on radii.
        # But centers themselves must be in [0,1].
        # The forces above might push centers out.
        # We clamp centers to [0,1] and maybe add boundary repulsion.
        
        # Apply forces
        centers += learning_rate * forces
        
        # Project back to [0, 1]x[0, 1]
        centers = np.clip(centers, 1e-5, 1 - 1e-5)
        
    # Final solve to get exact radii for best centers
    final_radii, final_sum = solve_radii(best_centers)
    
    return best_centers, final_radii, final_sum

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    centers, radii, sum_radii = optimize_packing()
    return centers, radii, sum_radii
