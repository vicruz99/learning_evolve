import numpy as np

def run_packing():
    """
    Packs 26 circles in a unit square [0,1]x[0,1] to maximize the sum of radii.
    Uses a force-directed relaxation with iterative radius growth.
    """
    n = 26
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Constants
    stiffness = 50.0       # Spring constant for repulsion
    dt = 1e-4              # Time step for integration
    relax_iter = 100       # Relaxation iterations per growth step
    growth_step = 0.0005   # Radius increment per step
    max_growth_steps = 3000
    
    def compute_forces(centers, radii):
        """Compute repulsive forces between circles and walls."""
        forces = np.zeros_like(centers)
        
        # Pairwise repulsion
        for i in range(n):
            for j in range(i + 1, n):
                diff = centers[i] - centers[j]
                dist = np.hypot(diff[0], diff[1])
                min_dist = radii[i] + radii[j]
                
                if dist < min_dist and dist > 1e-9:
                    overlap = min_dist - dist
                    # Force magnitude proportional to overlap
                    f_mag = overlap * stiffness
                    dir_vec = diff / dist
                    forces[i] += dir_vec * f_mag
                    forces[j] -= dir_vec * f_mag
                elif dist <= 1e-9:
                    # Prevent division by zero, push randomly if centers coincide
                    forces[i] += np.random.rand(2) * stiffness
                    forces[j] -= np.random.rand(2) * stiffness
        
        # Wall repulsion
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            
            # Left wall
            if x < r:
                forces[i, 0] += (r - x) * stiffness
            # Right wall
            if x > 1 - r:
                forces[i, 0] -= (x - (1 - r)) * stiffness
            # Bottom wall
            if y < r:
                forces[i, 1] += (r - y) * stiffness
            # Top wall
            if y > 1 - r:
                forces[i, 1] -= (y - (1 - r)) * stiffness
                
        return forces

    def relax(centers, radii, n_iter=relax_iter):
        """Relax positions to minimize overlaps."""
        for _ in range(n_iter):
            forces = compute_forces(centers, radii)
            centers += forces * dt
            # Clip to ensure they stay within bounds during simulation
            centers = np.clip(centers, 1e-9, 1 - 1e-9)
        return centers

    def is_valid(centers, radii, tol=1e-7):
        """Check if packing is valid."""
        # Boundary checks
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < -tol or x + r > 1 + tol or y - r < -tol or y + r > 1 + tol:
                return False
        
        # Overlap checks
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                if dist < radii[i] + radii[j] - tol:
                    return False
        return True

    # Run multiple trials with different seeds
    for seed in range(30):
        np.random.seed(seed)
        
        # Initialization: Hexagonal grid pattern
        # Try to fit roughly 5x5 or similar
        init_centers = []
        r_init = 0.05
        
        # Generate hex grid points
        for row in range(6):
            for col in range(6):
                if len(init_centers) >= n:
                    break
                # Hexagonal offset
                x = col * 2 * r_init + (row % 2) * r_init
                y = row * np.sqrt(3) * r_init
                init_centers.append([x, y])
        
        if len(init_centers) < n:
            # Fallback to random if grid is small (unlikely)
            init_centers = np.random.rand(n, 2)
        else:
            init_centers = np.array(init_centers[:n])
        
        # Normalize/Scale initial positions to be inside [0.1, 0.9] roughly
        xmin, ymin = np.min(init_centers, axis=0)
        xmax, ymax = np.max(init_centers, axis=0)
        w = xmax - xmin
        h = ymax - ymin
        if w > 0 and h > 0:
            init_centers = (init_centers - [xmin, ymin]) / [w, h] * 0.8 + 0.1
        
        centers = init_centers.copy()
        radii = np.full(n, 0.01) # Start small
        
        # Growth phase
        for _ in range(max_growth_steps):
            radii += growth_step
            
            # Relax to accommodate larger radii
            centers = relax(centers, radii)
            
            # Optional: Add small random noise to escape local minima occasionally
            if np.random.rand() < 0.01:
                centers += np.random.randn(n, 2) * 1e-4
                centers = np.clip(centers, 0, 1)

        # Post-optimization: If overlaps remain, shrink radii slightly to make valid
        # Binary search or linear reduction to find max valid radii for this configuration
        if not is_valid(centers, radii):
            # Reduce radii until valid
            reduction_step = 0.001
            while not is_valid(centers, radii) and radii[0] > 0:
                radii -= reduction_step
                centers = relax(centers, radii, n_iter=20)
        
        # Final relaxation with current radii
        centers = relax(centers, radii, n_iter=500)
        
        # Validate and record best
        # We might still have tiny overlaps due to numerical issues,

        while not is_valid(centers, radii):
            radii *= 0.999
            centers = relax(centers, radii, n_iter=50)
            
        current_sum = np.sum(radii)
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers.copy()

    return best_centers, best_radii, best_sum
