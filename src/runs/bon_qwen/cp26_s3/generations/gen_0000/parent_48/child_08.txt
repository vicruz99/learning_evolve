def get_constraints(x, n):
    """
    Returns the constraint values for the optimization.
    Constraints must be >= 0.
    x is a flattened array: [x_0, y_0, ..., x_n, y_n, r_0, ..., r_n]
    """
    centers = x[:2*n].reshape(n, 2)
    radii = x[2*n:]
    
    constraints_list = []
    
    # Boundary constraints:
    # x - r >= 0
    constraints_list.append(centers[:, 0] - radii)
    # 1 - x - r >= 0
    constraints_list.append(1 - centers[:, 0] - radii)
    # y - r >= 0
    constraints_list.append(centers[:, 1] - radii)
    # 1 - y - r >= 0
    constraints_list.append(1 - centers[:, 1] - radii)
    
    # Overlap constraints:
    # dist^2 - (r_i + r_j)^2 >= 0
    for i in range(n):

            r_sum_sq = (radii[i] + radii[j])**2
            constraints_list.append(dist_sq - r_sum_sq)
            
    return np.concatenate(constraints_list)
