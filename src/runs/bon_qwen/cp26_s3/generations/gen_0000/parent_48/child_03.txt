import numpy as np
from scipy.optimize import minimize

def compute_penalty(centers, radii):
    n = len(radii)
    pen = 0.0
    # Boundary constraints: circles must stay within [0, 1]
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        pen += max(0, r - x)**2 + max(0, x + r - 1)**2 + max(0, r - y)**2 + max(0, y + r - 1)**2
    # Overlap constraints: distance between centers must be >= sum of radii
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.hypot(dx, dy)
            min_dist = radii[i] + radii[j]
            pen += max(0, min_dist - dist)**2
    return pen

def packing_objective(v, n_circles):
    # Extract variables: first 2n are centers, last is common radius
    centers = v[:2*n_circles].reshape(n_circles, 2)
    r = v[2*n_circles]
    radii = np.full(n_circles, r)
    pen = compute_penalty(centers, radii)
    # Objective: maximize r, penalize constraint violations heavily
    return -r + 10000.0 * pen

def run_packing():
