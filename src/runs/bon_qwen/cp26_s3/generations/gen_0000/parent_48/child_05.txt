import numpy as np
from scipy.optimize import minimize

def objective(params, n):
    r = params[2::3]
    return -np.sum(r)

def all_constraints(params, n):
    num_cons = 4 * n + n * (n - 1) // 2
    c = np.zeros(num_cons)
    idx = 0
    for i in range(n):
        p = 3 * i
        # x - r >= 0
        c[idx] = params[p] - params[p + 2]
        # 1 - x - r >= 0
        c[idx + 1] = 1.0 - params[p] - params[p + 2]
        # y - r >= 0
        c[idx + 2] = params[p + 1] - params[p + 2]
        # 1 - y - r >= 0
        c[idx + 3] = 1.0 - params[p + 1] - params[p + 2]
        idx += 4
        
    for i in range(n):
        for j in range(i + 1, n):
            pi, pj = 3 * i, 3 * j
            dx = params[pi] - params[pj]
            dy = params[pi + 1] - params[pj + 1]
            dist = np.sqrt(dx * dx + dy * dy + 1e-14)
            c[idx] = dist - params[pi + 2] - params[pj + 2]
            idx += 1
    return c

def run_packing():
    n = 26
    best_params = None
    best_val = -np.inf
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * n
    cons = {'type': 'ineq', 'fun': all_constraints, 'args': (n,)}
    
    # Multi-start optimization to escape local minima
    for seed in range(10):
        rng = np.random.default_rng(seed)
        
        centers = np.zeros((n, 2))
        # Initialize 25 circles on a 5x5 grid
        for k in range(25):
            centers[k, 0] = (k % 5 + 0.5) / 5.0
            centers[k, 1] = (k // 5 + 0.5) / 5.0
        # Place 26th circle in a corner region
        centers[25, 0] = 0.15
        centers[25, 1] = 0.15
        
        # Add controlled perturbation
        centers += rng.uniform(-0.04, 0.04, size=centers.shape)
        centers = np.clip(centers, 0.05, 0.95)
        
        r_init = np.full(n, 0.02)
        params = np.zeros(3 * n)
        for i in range(n):
            params[3*i] = centers[i, 0]
            params[3*i+1] = centers[i, 1]
            params[3*i+2] = r_init[i]
            
        try:
            res = minimize(objective, params, args=(n,), method='SLSQP', 
                           bounds=bounds, constraints=cons, 
                           options={'ftol': 1e-9, 'maxiter': 2000})
            if -res.fun > best_val:
                best_val = -res.fun
                best_params = res.x.copy()
        except Exception:
            continue
            
    centers_out = np.zeros((n, 2))
    radii_out = np.zeros(n)
    for i in range(n):
        centers_out[i, 0] = best_params[3*i]
        centers_out[i, 1] = best_params[3*i+1]
        radii_out[i] = np.maximum(best_params[3*i+2], 0.0)
        
    sum_r = np.sum(radii_out)
    return centers_out, radii_out, float(sum_r)
