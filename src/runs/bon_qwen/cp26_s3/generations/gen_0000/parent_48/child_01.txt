

<strategy>

   - Box constraints: $0 \le x_i, y_i \le 1$
   - Boundary constraints: $r_i \le x_i, y_i \le 1 - r_i$
   - Non-overlap constraints: $(x_i - x_j)^2 + (y_i - y_j)^2 \ge (r_i + r_j)^2$ for all $i < j$
2. **Solver**: Use `scipy.optimize.minimize` with the SLSQP algorithm, which efficiently handles non-linear inequality constraints. The objective function minimizes the negative sum of radii.
3. **Initialization**: Generate diverse starting points to avoid poor local optima:

   - A hexagonal lattice pattern (theoretically densest for equal circles)
   - Multiple randomized configurations with controlled spread
4. **Constraints & Bounds**: Vectorize constraint evaluations for performance. Enforce $[0,1]$ bounds on all variables. The constraint functions return arrays representing all inequalities simultaneously.

import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def get_indices():
    xs = np.arange(0, N_CIRCLES * 3, 3)
    ys = xs + 1
    rs = xs + 2
    return xs, ys, rs

def objective(vars_flat):
    _, _, rs = get_indices()
    return -np.sum(vars_flat[rs])

def constraint_boundary(vars_flat):
    xs, ys, rs = get_indices()
    con = np.empty(N_CIRCLES * 4)
    con[0::4] = vars_flat[xs] - vars_flat[rs]
    con[1::4] = 1.0 - vars_flat[rs] - vars_flat[xs]
    con[2::4] = vars_flat[ys] - vars_flat[rs]
    con[3::4] = 1.0 - vars_flat[rs] - vars_flat[ys]
    return con

def constraint_overlap(vars_flat):
    xs, ys, rs = get_indices()
    n_con = N_CIRCLES * (N_CIRCLES - 1) // 2
    con = np.empty(n_con)
    idx = 0
    for i in range(N_CIRCLES):
        xi, yi, ri = xs[i], ys[i], rs[i]
        for j in range(i + 1, N_CIRCLES):
            xj, yj, rj = xs[j], ys[j], rs[j]
            dx = vars_flat[xi] - vars_flat[xj]
            dy = vars_flat[yi] - vars_flat[yj]
            dr = vars_flat[ri] + vars_flat[rj]
            con[idx] = dx*dx + dy*dy - dr*dr
            idx += 1
    return con

def generate_initial_guess(seed):
    rng = np.random.RandomState(seed)
    vars_flat = np.zeros(N_CIRCLES * 3)
    xs, ys, rs = get_indices()
    
    if seed == 0:
        points = np.array([(i*0.2 + 0.1, j*0.2 + 0.1) for i in range(5) for j in range(5)])
        points = np.vstack([points, [0.5, 0.5]])
        vars_flat[xs] = points[:, 0]
        vars_flat[ys] = points[:, 1]
    elif seed == 1:
        r0 = 0.1
        points = []
        y = r0
        row = 0
        while y < 1 - r0:
            x = r0 + row * 0.5 * r0
            while x < 1 - r0:
                if len(points) < N_CIRCLES:
                    points.append([x, y])
                x += 1.5 * r0
            y += np.sqrt(3)/2 * r0
            row += 1
        while len(points) < N_CIRCLES:
            points.append(rng.uniform(r0, 1-r0, 2))
        vars_flat[xs] = np.array(points)[:, 0]
        vars_flat[ys] = np.array(points)[:, 1]
    else:
        vars_flat[xs] = rng.uniform(0.15, 0.85, N_CIRCLES)
        vars_flat[ys] = rng.uniform(0.15, 0.85, N_CIRCLES)
        
    vars_flat[rs] = 0.08
    return vars_flat

def run_packing():
    bounds = [(0, 1)] * (2 * N_CIRCLES) + [(0, 1)] * N_CIRCLES
    constraints = [
        {'type': 'ineq', 'fun': constraint_boundary},
        {'type': 'ineq', 'fun': constraint_overlap}
    ]
    
    best_val = -np.inf
    best_sol = None
    
    seeds = [0, 1, 2, 3, 4]
    for seed in seeds:
        x0 = generate_initial_guess(seed)
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, 
                           constraints=constraints, options={'maxiter': 3000, 'ftol': 1e-10, 'disp': False})
            if -res.fun > best_val:
                best_val = -res.fun
                best_sol = res.x.copy()
        except Exception:
            continue
            
    if best_sol is None:
        x0 = generate_initial_guess(0)
        res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints)
        best_sol = res.x
        best_val = -res.fun

    centers = best_sol[:2*N_CIRCLES].reshape(N_CIRCLES, 2)
    radii = best_sol[2*N_CIRCLES:]
    
    # Numerical safety clamping
    centers = np.clip(centers, 0.0, 1.0)
    radii = np.maximum(radii, 0.0)
    
    return centers, radii, np.sum(radii)
