import numpy as np
from scipy.optimize import minimize

def compute_objective(vars, n, lam):
    """
    Computes the objective: -sum(radii) + penalty for constraint violations.
    vars: 1D array of length 3n [x0, y0, ..., xn-1, yn-1, r0, ..., rn-1]
    n: number of circles
    lam: penalty multiplier
    """
    centers = vars[:2*n].reshape(n, 2)
    radii = vars[2*n:]

    obj = -np.sum(radii)
    pen = 0.0

    # Boundary penalties
    # Circles must satisfy: r <= x <= 1-r and r <= y <= 1-r
    x_viol = np.maximum(0.0, radii - centers[:, 0])
    x_viol += np.maximum(0.0, centers[:, 0] - (1.0 - radii))
    y_viol = np.maximum(0.0, radii - centers[:, 1])
    y_viol += np.maximum(0.0, centers[:, 1] - (1.0 - radii))
    pen += np.sum(x_viol**2 + y_viol**2)

    # Overlap penalties
    # Circles must satisfy: dist(c_i, c_j) >= r_i + r_j
    dx = centers[:, 0][:, None] - centers[:, 0][None, :]
    dy = centers[:, 1][:, None] - centers[:, 1][None, :]
    dists = np.sqrt(dx**2 + dy**2)
    sum_r = radii[:, None] + radii[None, :]

    violations = np.maximum(0.0, sum_r - dists)
    # Sum only upper triangle to count each pair once
    violations_upper = np.triu(violations, k=1)
    pen += np.sum(violations_upper**2)

    return obj + lam * pen

def generate_initial_config(n):
    """
    Generates a valid initial hexagonal packing configuration.
    Returns centers (n, 2) and radii (n,).
    """
    r = 0.09
    centers = np.zeros((n, 2))
    radii = np.full(n, r)

    idx = 0
    y = r
    row_idx = 0
    while idx < n:
        if row_idx % 2 == 0:
            num = min(5, n - idx)
            for k in range(num):
                centers[idx, 0] = r + 2*r*k
                centers[idx, 1] = y
                idx += 1
        else:
            num = min(4, n - idx)
            for k in range(num):
                centers[idx, 0] = 2*r + 2*r*k
                centers[idx, 1] = y
                idx += 1
        y += r * np.sqrt(3)
        row_idx += 1
        
    return centers, radii

def run_packing():
    n = 26
    np.random.seed(42) # For reproducibility
    
    centers, radii = generate_initial_config(n)
    
    # Add small random perturbation to break symmetry and help optimization
    centers += np.random.uniform(-0.005, 0.005, centers.shape)
    # Clip to ensure initial validity is roughly maintained
    centers = np.clip(centers, radii, 1.0 - radii)

    # Flatten to 1D array for optimizer: [x0, y0, ..., x25, y25, r0, ..., r25]
    x0 = np.zeros(3 * n)
    x0[:2*n] = centers.flatten()
    x0[2*n:] = radii

    # Define bounds
    bounds = [(0.0, 1.0)] * (2*n) + [(1e-5, 0.5)] * n

    # Optimization with increasing penalty parameter
    lambdas = [10, 100, 1000, 5000, 10000]
    current_x = x0.copy()

    for lam in lambdas:
        res = minimize(compute_objective, current_x, args=(n, lam), 
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': 1000, 'ftol': 1e-12, 'gtol': 1e-12, 'disp': False})
        current_x = res.x

    # Extract results
    final_centers = current_x[:2*n].reshape(n, 2)
    final_radii = current_x[2*n:]
    sum_radii = np.sum(final_radii)

    return final_centers, final_radii, float(sum_radii)
