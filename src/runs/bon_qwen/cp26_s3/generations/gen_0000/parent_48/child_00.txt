    centers = np.array(centers[:n])
    
    # Initialize radii
    radii = np.full(n, 0.05)
    
    # Flatten variables for optimization: [x1, y1, r1, x2, y2, r2, ...]
    x0 = np.zeros(3 * n)
    for i in range(n):
        x0[3*i] = centers[i, 0]
        x0[3*i+1] = centers[i, 1]
        x0[3*i+2] = radii[i]
        
    # Bounds: x, y in [0, 1], r >= 0
    bounds = []
    for i in range(n):
        bounds.append((0, 1)) # x
        bounds.append((0, 1)) # y
        bounds.append((0, None)) # r (upper bound effectively 0.5)

    def objective(vars_vec):
        current_centers = np.zeros((n, 2))
        current_radii = np.zeros(n)
        for i in range(n):
            current_centers[i] = [vars_vec[3*i], vars_vec[3*i+1]]
            current_radii[i] = vars_vec[3*i+2]
            
        # Objective: Minimize negative sum of radii
        score = -np.sum(current_radii)
        
        # Penalty for overlaps
        penalty = 0.0
        penalty_weight = 1000.0 # High weight to enforce constraints
        
        # Check boundary violations
        for i in range(n):
            x, y, r = vars_vec[3*i], vars_vec[3*i+1], vars_vec[3*i+2]

            if x - r < 0: penalty += penalty_weight * (x - r)**2
            if x + r > 1: penalty += penalty_weight * (x + r - 1)**2
            # Top/Bottom
            if y - r < 0: penalty += penalty_weight * (y - r)**2
            if y + r > 1: penalty += penalty_weight * (y + r - 1)**2
            
        # Check overlaps
        for i in range(n):

                r_i = current_radii[i]
                r_j = current_radii[j]
                dist = np.sqrt(np.sum((current_centers[i] - current_centers[j])**2))
                if dist < r_i + r_j:
                    overlap = r_i + r_j - dist
                    penalty += penalty_weight * overlap**2
                    
        return score + penalty

    # Run optimization
    # Using L-BFGS-B is efficient for bounded problems
    result = minimize(objective, x0, method='L-BFGS-B', bounds=bounds, options={'ftol': 1e-9, 'gtol': 1e-6, 'maxiter': 1000})

    final_radii = np.maximum(final_radii, 1e-9)
    
    sum_radii = np.sum(final_radii)
    
    return final_centers, final_radii, sum_radii
