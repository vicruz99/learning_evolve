import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    best_sum_r = 0.0
    best_centers = np.zeros((n, 2))
    best_radii = np.zeros(n)

    # 1. Generate Initial Hexagonal Grid Configuration
    # A hexagonal arrangement is denser than a square grid.
    # We arrange points in rows, shifting odd rows to create a honeycomb structure.
    # We adjust spacing to ensure initial non-overlap.
    # Target: fit 26 points. 6 rows with distribution 4, 5, 4, 5, 4, 4 works well.
    row_counts = [4, 5, 4, 5, 4, 4]
    centers_init = []
    y_spacing = 0.18  # Vertical spacing for hex lattice
    x_spacing = 0.17  # Horizontal spacing
    
    current_y = 0.15
    for r_idx, count in enumerate(row_counts):
        # Shift x for odd rows (1, 3, 5...) to nest in gaps
        x_offset = 0.0
        if r_idx % 2 == 1:
            x_offset = x_spacing / 2.0
        
        # Distribute points in the current row within [x_margin, 1-x_margin]
        # Available width is 1.0. We leave margin for radii.
        # Let's distribute them evenly in [0.1, 0.9]
        if count > 0:
            x_positions = np.linspace(0.1, 0.9, count)
            # Add offset
            x_positions = x_positions + x_offset
            # Clip to valid range [0.05, 0.95]
            x_positions = np.clip(x_positions, 0.05, 0.95)
            
            for x in x_positions:
                centers_init.append([x, current_y])
        current_y += y_spacing
        
    # If we have fewer/more than 26, trim or pad (though 26 is target)
    centers_init = np.array(centers_init[:n])
    if len(centers_init) < n:
        # Fill remaining with random valid points if needed
        extra = n - len(centers_init)
        random_pts = np.random.rand(extra, 2) * 0.8 + 0.1
        centers_init = np.vstack([centers_init, random_pts])
        
    initial_radii = np.full(n, 0.05)

    # 2. Define Objective and Constraints for Optimizer
    def objective(vars_flat):
        # vars_flat contains [x1, y1, r1, x2, y2, r2, ...]
        centers = vars_flat[:2*n].reshape((n, 2))
        radii = vars_flat[2*n:]
        return -np.sum(radii) # Negative because we minimize

    def boundary_constraints(vars_flat):
        radii = vars_flat[2*n:]
        centers = vars_flat[:2*n].reshape((n, 2))
        cons = []
        for i in range(n):
            r = radii[i]
            x, y = centers[i]
            # r <= x, r <= 1-x, r <= y, r <= 1-y  =>  r - x <= 0, etc.
            cons.append(r - x)
            cons.append(r - (1 - x))
            cons.append(r - y)
            cons.append(r - (1 - y))
            # r >= 0
            cons.append(-r)
        return cons

    def overlap_constraints(vars_flat):
        centers = vars_flat[:2*n].reshape((n, 2))
        radii = vars_flat[2*n:]
        cons = []
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                # dist >= r_i + r_j  =>  r_i + r_j - dist <= 0
                cons.append(radii[i] + radii[j] - dist)
        return cons

    # 3. Optimization Loop with Multiple Restarts
    # We perform several optimizations to avoid local minima.
    # We use SLSQP which handles bounds and non-linear constraints well.
    
    num_restarts = 5
    bounds = []
    for _ in range(2 * n): # x, y coordinates
        bounds.append((0.0, 1.0))
    for _ in range(n): # radii
        bounds.append((0.0, 0.5))

    for k in range(num_restarts):
        # Perturb initial guess slightly for diversity
        if k == 0:
            guess_centers = centers_init.copy()
        else:
            # Random walk perturbation
            guess_centers = centers_init + np.random.normal(0, 0.05, centers_init.shape)
            guess_centers = np.clip(guess_centers, 0.01, 0.99)
            
        guess_radii = initial_radii.copy() + np.random.normal(0, 0.01, n)
        guess_radii = np.clip(guess_radii, 0.001, 0.5)
        
        x0 = np.concatenate([guess_centers.flatten(), guess_radii])
        
        # Define constraints for scipy
        constraints = []
        
        # Boundary constraints (inequality >= 0 form for scipy usually, but here we use generic)
        # scipy.minimize constraints: fun(x) >= 0 for 'ineq'
        # Our derived constraints were <= 0. So we negate them.
        # Actually, easier to define fun(x) >= 0
        # x - r >= 0 => x - r >= 0
        # 1-x - r >= 0
        # y - r >= 0
        # 1-y - r >= 0
        # r >= 0
        # dist - (r1+r2) >= 0
        
        c_boundary = {'type': 'ineq', 'fun': lambda v: [v[2*i] - v[2*n+i] for i in range(n)]} # x - r
        constraints.append(c_boundary)
        constraints.append({'type': 'ineq', 'fun': lambda v: [(1.0 - v[2*i]) - v[2*n+i] for i in range(n)]}) # 1-x - r
        constraints.append({'type': 'ineq', 'fun': lambda v: [v[2*i+1] - v[2*n+i] for i in range(n)]}) # y - r
        constraints.append({'type': 'ineq', 'fun': lambda v: [(1.0 - v[2*i+1]) - v[2*n+i] for i in range(n)]}) # 1-y - r
        constraints.append({'type': 'ineq', 'fun': lambda v: [v[2*n+i] for i in range(n)]}) # r >= 0

        # Overlap constraints: dist >= r_i + r_j
        # We create a list of constraint functions. 
        # Since lambda captures variable i/j by reference, we need to be careful or use default args.
        # But defining a function inside the loop for each pair is costly.
        # Better to define one function that returns vector of all constraints.
        
        def overlap_vec(v):
            c = v[:2*n].reshape((n, 2))
            r = v[2*n:]
            cons = []
            for i in range(n):
                for j in range(i + 1, n):
                    d = np.sqrt(np.sum((c[i] - c[j])**2))
                    cons.append(d - (r[i] + r[j]))
            return cons

        constraints.append({'type': 'ineq', 'fun': overlap_vec})

        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints, 
                           options={'maxiter': 200, 'ftol': 1e-8})
            
            if res.success:
                current_sum = -res.fun
                if current_sum > best_sum_r:
                    best_sum_r = current_sum
                    best_centers = res.x[:2*n].reshape((n, 2))
                    best_radii = res.x[2*n:]
        except Exception as e:
            print(f"Optimization failed in iteration {k}: {e}")
            continue

    # Final validation and cleanup (just in case)
    # Ensure radii are strictly positive if possible, though 0 is allowed.
    # The problem allows nonnegative radii.
    
    return best_centers, best_radii, best_sum_r
