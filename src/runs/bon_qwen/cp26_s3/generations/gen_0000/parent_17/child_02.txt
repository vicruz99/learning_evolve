import numpy as np
from scipy.optimize import minimize

def generate_hexagonal_initial_positions(n):
    """
    Generates initial positions for n circles based on a hexagonal lattice.
    """
    # We create a grid of points based on hexagonal packing logic
    # Rows of points, staggered
    points = []
    row = 0
    # Estimate spacing to fit roughly in a square, then we will scale/center
    # Just generate enough points and pick n
    while len(points) < n:
        # For even rows (0, 2, ...), x starts at 0. For odd rows (1, 3, ...), x starts at 0.5
        # Actually standard hex lattice:
        # y = row * sqrt(3)/2
        # x = col + (row % 2) * 0.5
        
        # We iterate cols until we have enough points or run out of "reasonable" range
        # But simpler: just fill a bounding box
        # Let's generate points in a 6x6 grid area with hex spacing
        
        y_coord = row * np.sqrt(3) / 2.0
        start_x = 0.5 if row % 2 == 1 else 0.0
        
        col = 0
        while col < 6: # Limit columns
            x_coord = col + start_x
            points.append([x_coord, y_coord])
            col += 1
        row += 1
        
    # Take the first n points
    points = np.array(points[:n])
    
    # Normalize to [0, 1] x [0, 1]
    # Center the cluster and scale to fit within [0, 1] with some margin
    # Actually, just scaling min-max to [0, 1] might put them on edges.
    # Let's find bounding box and center it.
    
    if points.size == 0:
        return np.random.rand(n, 2)
        
    min_coords = points.min(axis=0)
    max_coords = points.max(axis=0)
    center = (min_coords + max_coords) / 2
    size = max_coords - min_coords
    
    # Scale to fit in 0.8 x 0.8 centered at 0.5
    scale = 0.8 / max(size)
    normalized_points = (points - center) * scale + 0.5
    
    return normalized_points

def run_packing():
    n = 26
    
    # 1. Initialize centers
    centers = generate_hexagonal_initial_positions(n)
    
    # 2. Define variables: [x1, y1, ..., x26, y26, r]
    # Total 26*2 + 1 = 53 variables
    # Initial guess: centers + small radius
    x0 = np.concatenate([centers.flatten(), [0.1]]) 
    
    # 3. Define objective function
    # We want to maximize r, so we minimize -r
    def objective(vars):
        r = vars[-1]
        return -r
    
    # 4. Define constraints
    # Constraints:
    # 1. r >= 0 (bounds)
    # 2. x_i >= r
    # 3. x_i <= 1 - r
    # 4. y_i >= r
    # 5. y_i <= 1 - r
    # 6. dist(c_i, c_j) >= 2r  => (x_i - x_j)^2 + (y_i - y_j)^2 - 4r^2 >= 0
    
    def constraints(vars):
        c_list = []
        r = vars[-1]
        # Extract centers
        # vars[0::2] are x, vars[1::2] are y
        # But vars is flat. 
        # x_i is at index 2*i, y_i at 2*i+1
        xs = vars[0:2*n:2]
        ys = vars[1:2*n:2]
        
        # Boundary constraints
        for i in range(n):
            c_list.append(xs[i] - r)
            c_list.append(1 - xs[i] - r)
            c_list.append(ys[i] - r)
            c_list.append(1 - ys[i] - r)
            
        # Overlap constraints
        for i in range(n):
            for j in range(i + 1, n):
                dx = xs[i] - xs[j]
                dy = ys[i] - ys[j]
                dist_sq = dx*dx + dy*dy
                c_list.append(dist_sq - 4*r*r)
        
        return np.array(c_list)

    # 5. Bounds
    # x, y in [0, 1], r in [0, 1]
    bounds = []
    for i in range(n):
        bounds.append((0, 1)) # x_i
        bounds.append((0, 1)) # y_i
    bounds.append((0, 1))     # r
    
    # 6. Optimization
    # Use SLSQP which supports constraints and bounds
    # We might need to run it a few times or use a good start.
    # The initial r=0.1 might be feasible or not.
    # If infeasible, solver might struggle. 
    # Let's try to start with a feasible r. 
    # With grid spacing ~0.2, r=0.05 is safe.
    
    # Re-initialize with safe radius if needed, but 0.1 is likely close.
    # To be safe, let's reduce initial r slightly.
    x0[-1] = 0.05
    
    # Define constraint structure for SLSQP
    cons = ({'type': 'ineq', 'fun': constraints})
    
    # Run optimization
    try:
        res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, 
                       options={'maxiter': 1000, 'ftol': 1e-9})
    except Exception as e:
        # Fallback to a simple grid solution if optimization fails
        print(f"Optimization failed: {e}")
        # Fallback: 5x5 grid + 1
        centers_fallback = np.array([(i*0.2 + 0.1, j*0.2 + 0.1) for j in range(5) for i in range(5)] + [(0.05, 0.5)])
        radii_fallback = np.full(26, 0.08) # Safe radius
        return centers_fallback, radii_fallback, np.sum(radii_fallback)

    # Extract results
    final_centers = np.column_stack((res.x[0:2*n:2], res.x[1:2*n:2]))
    final_r = res.x[-1]
    final_radii = np.full(n, final_r)
    
    # Check validity
    # The solver might slightly violate constraints due to tolerance, 
    # but we trust the output for now. 
    # However, to be absolutely safe against numerical noise in the validation function:
    # We can clamp radii slightly if needed, but usually optimal solution is tight.
    
    # Just to be safe, if any constraint is violated by tiny amount, we might shrink r slightly?
    # But the objective maximizes r, so it should be tight.
    
    sum_radii = np.sum(final_radii)
    
    return final_centers, final_radii, sum_radii
