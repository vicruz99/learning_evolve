    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.

    best_radii = None

            # Place 26th circle in the center gap or random

        # Flatten variables for optimization: [x1, y1, r1, x2, y2, r2, ...]

        bounds = []

            # Reshape
            c = vars_flat.reshape(-1, 3)
            centers_opt = c[:, :2]
            radii_opt = c[:, 2]
            
            # Objective: Maximize sum of radii => Minimize negative sum
            sum_r = -np.sum(radii_opt)
            
            # Penalties for constraints
            penalty = 0.0
            alpha = 100.0 # Penalty weight
            
            # Boundary constraints
            # Circle i must be inside [0,1]x[0,1]
            # x - r >= 0 => r - x <= 0
            # x + r <= 1 => r - (1-x) <= 0
            # y - r >= 0 => r - y <= 0
            # y + r <= 1 => r - (1-y) <= 0
            for i in range(n_circles):
                r = radii_opt[i]
                x = centers_opt[i, 0]
                y = centers_opt[i, 1]
                
                # Soft penalty using ReLU-like function for smoothness if possible,

                # Using max(0, violation)^2
                v1 = r - x
                v2 = r - (1.0 - x)
                v3 = r - y
                v4 = r - (1.0 - y)
                
                if v1 > 0: penalty += alpha * v1**2
                if v2 > 0: penalty += alpha * v2**2
                if v3 > 0: penalty += alpha * v3**2
                if v4 > 0: penalty += alpha * v4**2
                
            # Overlap constraints

            for i in range(n_circles):

                    dist = np.sqrt(np.sum((centers_opt[i] - centers_opt[j])**2))
                    r_sum = radii_opt[i] + radii_opt[j]
                    overlap = r_sum - dist
                    if overlap > 0:

            return sum_r + penalty

                    if dist < final_radii[i] + final_radii[j] - 1e-7:

                        break
                if not is_valid: break
