import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    Returns centers, radii, and the sum of radii.
    """
    n = 26
    
    # Objective function: minimize negative sum of radii
    def objective(vars):
        return -np.sum(vars[2::3])

    # Helper to create constraint function for a specific configuration
    def get_constraint_fun(n):
        def constr(vars):
            cons = []
            # Boundary constraints for each circle
            for i in range(n):
                x, y, r = vars[3*i], vars[3*i+1], vars[3*i+2]
                cons.append(x - r)      # x - r >= 0
                cons.append(1 - x - r)  # x + r <= 1
                cons.append(y - r)      # y - r >= 0
                cons.append(1 - y - r)  # y + r <= 1
            
            # Non-overlap constraints
            for i in range(n):
                for j in range(i + 1, n):
                    xi, yi, ri = vars[3*i], vars[3*i+1], vars[3*i+2]
                    xj, yj, rj = vars[3*j], vars[3*j+1], vars[3*j+2]
                    
                    # Distance squared minus (sum of radii)^2 >= 0
                    dist_sq = (xi - xj)**2 + (yi - yj)**2
                    sum_r_sq = (ri + rj)**2
                    cons.append(dist_sq - sum_r_sq)
            return np.array(cons)
        return constr

    # Constraint dictionary for the optimizer
    constraint = {'type': 'ineq', 'fun': get_constraint_fun(n)}
    
    # Bounds: x, y in [0, 1], r in [0, 0.5]
    bounds = [(0, 1), (0, 1), (0, 0.5)] * n
    
    best_sum = -np.inf
    best_centers = None
    best_radii = None
    
    # Try multiple initializations to avoid local minima
    # We try a hexagonal grid and several random spreads
    num_restarts = 10
    np.random.seed(42)

    for attempt in range(num_restarts):
        vars_init = np.zeros(3 * n)
        
        if attempt == 0:
            # Initialization 1: Hexagonal lattice
            # We place points in rows with vertical spacing sqrt(3)/2 * spacing
            # Adjust spacing to fit within [0,1] with margins
            count = 0
            row_y = 0.08
            while count < n:
                row_points = 0
                # Determine x-start based on row parity for hex staggering
                x_start = 0.08 if int(row_y / 0.15) % 2 == 0 else 0.18
                
                row_x = x_start
                while row_x <= 0.92 and count < n:
                    vars_init[3*count] = row_x
                    vars_init[3*count+1] = row_y
                    vars_init[3*count+2] = 0.05 # Initial radius
                    count += 1
                    row_points += 1
                    row_x += 0.2 # Horizontal spacing
                
                # Adjust vertical spacing based on number of points in row to pack tightly
                # Standard hex vertical spacing is roughly 0.1732 * horizontal_spacing
                row_y += 0.16 
        else:
            # Initialization 2+: Random jittered grid
            # Create a 5x5 grid and jitter, plus extra points
            indices = np.random.choice(range(25), n - 5, replace=False)
            grid_points = []
            for i in range(5):
                for j in range(5):
                    grid_points.append([0.1 + i*0.2, 0.1 + j*0.2])
            
            selected_points = [grid_points[k] for k in indices]
            
            # Add remaining points randomly in gaps
            remaining = n - len(selected_points)
            for _ in range(remaining):
                px = np.random.uniform(0.05, 0.95)
                py = np.random.uniform(0.05, 0.95)
                selected_points.append([px, py])
                
            for k in range(n):
                vars_init[3*k] = selected_points[k][0]
                vars_init[3*k+1] = selected_points[k][1]
                vars_init[3*k+2] = 0.03 # Smaller initial radius for random

        # Run optimization
        try:
            res = minimize(objective, vars_init, method='SLSQP', bounds=bounds, 
                           constraints=constraint, options={'maxiter': 2000, 'ftol': 1e-10})
            
            if res.success:
                current_sum = -res.fun
                if current_sum > best_sum:
                    best_sum = current_sum
                    
                    # Extract and validate
                    centers = np.array([[res.x[3*i], res.x[3*i+1]] for i in range(n)])
                    radii = np.array([res.x[3*i+2] for i in range(n)])
                    
                    # Manual validation check (optional but safe)
                    if validate_packing(centers, radii):
                        best_centers = centers
                        best_radii = radii
        except Exception:
            continue

    # Fallback if optimization failed completely (should not happen)
    if best_centers is None:
        # Simple grid fallback
        centers = np.zeros((n, 2))
        radii = np.full(n, 0.05)
        for i in range(n):
            centers[i, 0] = 0.1 + (i % 5) * 0.18
            centers[i, 1] = 0.1 + (i // 5) * 0.18
        best_centers = centers
        best_radii = radii
        best_sum = np.sum(radii)

    return best_centers, best_radii, best_sum

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    n = centers.shape[0]
    if np.isnan(centers).any() or np.isnan(radii).any():
        return False
    for i in range(n):
        if radii[i] < 0: return False
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            return False
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False
    return True
