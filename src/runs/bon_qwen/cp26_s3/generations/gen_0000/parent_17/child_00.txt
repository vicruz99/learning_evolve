import numpy as np
from scipy.optimize import minimize

N = 26
# Precompute mask for pairwise separation constraints to avoid recomputing
SEP_MASK = np.triu(np.ones((N, N), dtype=bool), k=1)

def objective(vars):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(vars[52:])

def constraints(vars):
    """Returns an array of constraint values. All must be >= 0."""
    # Reshape variables into centers (N, 2) and radii (N,)
    c = vars[:52].reshape(N, 2)
    r = vars[52:]
    
    # Boundary constraints: x - r >= 0, 1 - x - r >= 0, same for y
    b1 = c[:, 0] - r
    b2 = 1 - c[:, 0] - r
    b3 = c[:, 1] - r
    b4 = 1 - c[:, 1] - r
    
    # Separation constraints: dist^2 - (r_i + r_j)^2 >= 0
    # Vectorized pairwise differences
    diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
    dist2 = np.sum(diff**2, axis=2)
    rs = r[:, np.newaxis] + r[np.newaxis, :]
    sep = dist2 - rs**2
    
    # Extract upper triangle (unique pairs)
    return np.concatenate([b1, b2, b3, b4, sep[SEP_MASK]])

def get_initial_guess():
    """Generates a dense grid initialization for faster convergence."""
    ys = np.linspace(0.15, 0.85, 6)
    xs = np.linspace(0.15, 0.85, 6)
    coords = []
    for y in ys:
        for x in xs:
            coords.append([x, y])
            if len(coords) >= 26:
                break
        if len(coords) >= 26:
            break
            
    centers = np.array(coords)
    radii = np.ones(26) * 0.05
    vars0 = np.zeros(78)
    vars0[:52] = centers.flatten()
    vars0[52:] = radii
    return vars0

def run_packing():
    """Runs the optimization and returns the best valid packing found."""
    bounds = [(0, 1)] * 52 + [(0, 0.5)] * 26
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_res = None
    best_sum_r = -np.inf
    
    # Multiple restarts to escape local minima
    for _ in range(6):
        x0 = get_initial_guess()
        # Small random perturbation
        x0 += np.random.uniform(-0.005, 0.005, size=78)
        # Ensure initial guess respects basic bounds
        x0 = np.clip(x0, 0, 1)
        x0[52:] = np.clip(x0[52:], 0.001, 0.5)
        
        res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 3000, 'ftol': 1e-10, 'disp': False})
        
        if res.success:
            sum_r = np.sum(res.x[52:])
            if sum_r > best_sum_r:
                best_sum_r = sum_r
                best_res = res
                
    # Fallback if no run succeeded
    if best_res is None:
        best_res = minimize(objective, get_initial_guess(), method='SLSQP', bounds=bounds, constraints=cons)
        
    centers = best_res.x[:52].reshape(N, 2)
    radii = best_res.x[52:]
    
    # Post-optimization safety check against 1e-12 validation tolerance
    # If any pair is extremely close, slightly shrink radii to guarantee validity
    dists = np.linalg.norm(centers[:, np.newaxis, :] - centers[np.newaxis, :, :], axis=2)
    r_sums = radii[:, np.newaxis] + radii[np.newaxis, :]
    
    if np.any(dists < r_sums - 1e-13):
        radii *= 0.9999999
        
    return centers, radii, float(np.sum(radii))
