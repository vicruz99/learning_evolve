def run_packing():

    def objective(variables):
        # variables are [x1, y1, r1, x2, y2, r2, ...]
        # We want to maximize sum of radii, so minimize negative sum
        radii = variables[2::3]
        return -np.sum(radii)

    def boundary_constraints(variables):
        # x >= r, x <= 1-r => r <= x <= 1-r
        # y >= r, y <= 1-r => r <= y <= 1-r
        # Equivalent to:
        # x - r >= 0
        # 1 - x - r >= 0
        # y - r >= 0
        # 1 - y - r >= 0
        
        constraints = []
        for i in range(n):
            idx = 3 * i
            x, y, r = variables[idx], variables[idx+1], variables[idx+2]
            constraints.append(x - r)
            constraints.append(1 - x - r)
            constraints.append(y - r)
            constraints.append(1 - y - r)
        return np.array(constraints)

    def overlap_constraints(variables):
        # dist(i, j) >= r_i + r_j
        # sqrt((xi-xj)^2 + (yi-yj)^2) - (ri + rj) >= 0
        constraints = []
        for i in range(n):
            for j in range(i + 1, n):
                idx_i, idx_j = 3 * i, 3 * j
                xi, yi, ri = variables[idx_i], variables[idx_i+1], variables[idx_i+2]
                xj, yj, rj = variables[idx_j], variables[idx_j+1], variables[idx_j+2]
                
                dist = np.sqrt((xi - xj)**2 + (yi - yj)**2)
                constraints.append(dist - (ri + rj))
        return np.array(constraints)

    def get_constraints(variables):
        b = boundary_constraints(variables)
        o = overlap_constraints(variables)
        return np.concatenate([b, o])

    # Helper to create constraint dict for scipy

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square

    Args:
        centers: np.array of shape (n, 2) with (x, y) coordinates
        radii: np.array of shape (n) with radius of each circle

    Returns:
        True if valid, False otherwise
    """
    import numpy as np
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True
