from scipy.optimize import minimize

N_CIRCLES = 26

def eval_constraints(v):

    vals = []
    
    # Boundary constraints
    for i in range(n):

        vals.append(v[ix] - v[ir])
        vals.append(1.0 - v[ix] - v[ir])
        vals.append(v[iy] - v[ir])
        vals.append(1.0 - v[iy] - v[ir])
        
    # Overlap constraints
    for i in range(n):
        for j in range(i + 1, n):

            ix_j, iy_j, ir_j = j, n + j, 2 * n + j

            dy = v[iy_i] - v[iy_j]
            dr = v[ir_i] + v[ir_j]
            
            vals.append(dx * dx + dy * dy - dr * dr)
            
    return np.array(vals)

def constraint_wrapper(v):

    return eval_constraints(v)
