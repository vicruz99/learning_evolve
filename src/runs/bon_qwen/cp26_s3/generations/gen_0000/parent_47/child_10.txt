import numpy as np
from scipy.optimize import minimize


def create_hexagonal_packing(row_sizes, r):
    """Create a hexagonal packing with given row sizes and radius."""
    n = sum(row_sizes)
    centers = np.zeros((n, 2))
    radii = np.ones(n) * r
    idx = 0
    for row_idx, row_size in enumerate(row_sizes):
        y = r + row_idx * r * np.sqrt(3.0)
        x_offset = r if row_idx % 2 == 0 else 2.0 * r
        for col_idx in range(row_size):
            centers[idx, 0] = x_offset + col_idx * 2.0 * r
            centers[idx, 1] = y
            idx += 1
    return centers, radii


def compute_forces(centers, radii, n, strength=10.0, boundary_strength=50.0):
    """Compute repulsive forces between circles and boundary forces."""
    forces = np.zeros_like(centers)
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist_sq = dx * dx + dy * dy
            dist = np.sqrt(dist_sq) if dist_sq > 1e-16 else 1e-16
            min_dist = radii[i] + radii[j]
            if dist < min_dist + 0.05:
                overlap = min_dist - dist
                if overlap > 0:
                    fx = dx / dist * overlap * strength
                    fy = dy / dist * overlap * strength
                    forces[i, 0] += fx
                    forces[i, 1] += fy
                    forces[j, 0] -= fx
                    forces[j, 1] -= fy
        for c in [0, 1]:
            if centers[i, c] - radii[i] < 0:
                forces[i, c] += (radii[i] - centers[i, c]) * boundary_strength
            if centers[i, c] + radii[i] > 1.0:
                forces[i, c] -= (centers[i, c] + radii[i] - 1.0) * boundary_strength
    return forces


def iterative_expand(centers, radii, n, max_iter=3000, expansion_rate=1.0005):
    """Iteratively expand radii while maintaining validity."""
    for it in range(max_iter):
        radii *= expansion_rate
        forces = compute_forces(centers, radii, n)
        step = 0.5 / (1.0 + 0.01 * it)
        centers += step * forces
        for i in range(n):
            for c in [0, 1]:
                if centers[i, c] - radii[i] < 0:
                    centers[i, c] = radii[i]
                if centers[i, c] + radii[i] > 1.0:
                    centers[i, c] = 1.0 - radii[i]
        if it % 500 == 499:
            expansion_rate = max(1.0001, expansion_rate * 0.95)
    return centers, radii


def add_perturbations(centers, radii, n, scale=0.01):
    """Add small random perturbations."""
    perturbed_centers = centers.copy()
    for i in range(n):
        for c in [0, 1]:
            perturbed_centers[i, c] += np.random.uniform(-scale, scale)
            perturbed_centers[i, c] = max(radii[i], min(1.0 - radii[i], perturbed_centers[i, c]))
    return perturbed_centers


def fix_validity(centers, radii, n, max_push=100):
    """Fix any remaining overlaps by pushing circles apart."""
    for _ in range(max_push):
        any_violation = False
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                min_dist = radii[i] + radii[j]
                if dist < min_dist - 1e-12:
                    any_violation = True
                    if dist < 1e-16:
                        dx = 0.001
                        dy = 0.001
                        dist = np.sqrt(dx * dx + dy * dy)
                    overlap = min_dist - dist
                    for c in [0, 1]:
                        delta = (dx if c == 0 else dy) / dist * overlap * 0.5
                        centers[i, c] += delta
                        centers[j, c] -= delta
        if not any_violation:
            break
    for i in range(n):
        for c in [0, 1]:
            if centers[i, c] - radii[i] < 0:
                centers[i, c] = radii[i]
            if centers[i, c] + radii[i] > 1.0:
                centers[i, c] = 1.0 - radii[i]
    return centers


def objective_func(v, n=26):
    """Objective: negative sum of radii."""
    return -np.sum(v[2::3])


def make_constraints(n=26):
    """Create constraint dictionaries for SLSQP."""
    cons = []
    for i in range(n):
        cons.append({'type': 'ineq', 'fun': lambda v, i=i: v[3*i] - v[3*i+2]})
        cons.append({'type': 'ineq', 'fun': lambda v, i=i: 1.0 - v[3*i] - v[3*i+2]})
        cons.append({'type': 'ineq', 'fun': lambda v, i=i: v[3*i+1] - v[3*i+2]})
        cons.append({'type': 'ineq', 'fun': lambda v, i=i: 1.0 - v[3*i+1] - v[3*i+2]})
        cons.append({'type': 'ineq', 'fun': lambda v, i=i: v[3*i+2]})
    for i in range(n):
        for j in range(i + 1, n):
            cons.append({'type': 'ineq', 'fun': lambda v, i=i, j=j:
                np.sqrt((v[3*i] - v[3*j])**2 + (v[3*i+1] - v[3*j+1])**2) - v[3*i+2] - v[3*j+2]})
    return cons


def slsqp_refine(centers, radii, n=26):
    """Refine packing using SLSQP optimizer."""
    v0 = np.zeros(3 * n)
    for i in range(n):
        v0[3*i] = centers[i, 0]
        v0[3*i+1] = centers[i, 1]
        v0[3*i+2] = radii[i]
    
    cons = make_constraints(n)
    bounds = [(0, 1)] * (2 * n) + [(0, 0.5)] * n
    
    result = minimize(objective_func, v0, method='SLSQP', constraints=cons,
                      bounds=bounds, options={'maxiter': 1500, 'ftol': 1e-14})
    
    centers_opt = np.array([[result.x[3*i], result.x[3*i+1]] for i in range(n)])
    radii_opt = np.array([result.x[3*i+2] for i in range(n)])
    return centers_opt, radii_opt, -result.fun


def try_packing(row_sizes, seed=42):
    """Try a specific row structure."""
    np.random.seed(seed)
    n = 26
    r_init = 1.0 / 13.0
    centers, radii = create_hexagonal_packing(row_sizes, r_init)
    centers, radii = iterative_expand(centers, radii, n, max_iter=2000, expansion_rate=1.0008)
    centers = fix_validity(centers, radii, n)
    centers_opt, radii_opt, s = slsqp_refine(centers, radii, n)
    return centers_opt, radii_opt, s
