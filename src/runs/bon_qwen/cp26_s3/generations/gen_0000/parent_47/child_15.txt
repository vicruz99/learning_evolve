2. **Target analysis**: Target sum is 2.636, implying average radius ~0.1014. A 5×5 grid gives r=0.1 for 25 circles (sum=2.5). We need to fit a 26th circle, requiring some optimization.

import numpy as np
from scipy.optimize import minimize

N = 26

def objective_func(x):
    """Minimize negative sum of radii (equivalent to maximizing sum of radii)."""
    s = 0.0
    for i in range(N):
        s += x[3 * i + 2]
    return -s

def constraints_func(x):
    """Return array of all constraint values. All must be >= 0 for feasibility."""
    result = []
    
    # Boundary constraints for each circle
    for i in range(N):
        # x_i >= r_i
        result.append(x[3 * i] - x[3 * i + 2])
        # x_i + r_i <= 1
        result.append(1.0 - x[3 * i] - x[3 * i + 2])
        # y_i >= r_i
        result.append(x[3 * i + 1] - x[3 * i + 2])
        # y_i + r_i <= 1
        result.append(1.0 - x[3 * i + 1] - x[3 * i + 2])
    
    # Non-overlap constraints for each pair
    for i in range(N):
        for j in range(i + 1, N):
            dx = x[3 * i] - x[3 * j]
            dy = x[3 * i + 1] - x[3 * j + 1]
            ri = x[3 * i + 2]
            rj = x[3 * j + 2]
            dist_sq = dx * dx + dy * dy
            min_dist_sq = (ri + rj) * (ri + rj)
            result.append(dist_sq - min_dist_sq)
    
    return np.array(result)

def build_bounds():
    """Build bounds for all variables."""
    bounds = []
    for i in range(N):
        bounds.append((0.0, 1.0))  # x_i
        bounds.append((0.0, 1.0))  # y_i
        bounds.append((0.0, 0.5))  # r_i
    return bounds

def init_to_flat(x0):
    """Convert centers/radii to flat optimization vector."""
    flat = np.zeros(3 * N)
    for i in range(N):
        flat[3 * i] = x0[i, 0]
        flat[3 * i + 1] = x0[i, 1]
        flat[3 * i + 2] = x0[i, 2] if x0.shape[1] == 3 else x0[i]
    return flat

def flat_to_output(flat):
    """Convert flat vector back to centers and radii."""
    centers = np.zeros((N, 2))
    radii = np.zeros(N)
    for i in range(N):
        centers[i, 0] = flat[3 * i]
        centers[i, 1] = flat[3 * i + 1]
        radii[i] = flat[3 * i + 2]
    return centers, radii

def make_hexagonal_init(r_base, row_counts):
    """Create hexagonal grid initialization with specified row counts."""
    centers = np.zeros((N, 2))
    radii = np.full(N, r_base)
    
    count = 0
    row_height = r_base * np.sqrt(3)
    
    for row_idx, ncols in enumerate(row_counts):
        if count + ncols > N:
            ncols = N - count
        if ncols <= 0:
            break
        
        y_base = r_base + row_idx * row_height
        offset = r_base if row_idx % 2 == 1 else 0.0
        
        # Center the row horizontally
        total_width = (ncols - 1) * 2 * r_base
        x_start = (1.0 - total_width) / 2.0 if ncols > 1 else 0.5
        
        for col in range(ncols):
            x = x_start + col * 2 * r_base + offset
            y = y_base
            if x < 0:
                x = 0.01
            if x > 1:
                x = 0.99
            if y < 0:
                y = 0.01
            if y > 1:
                y = 0.99
            centers[count, 0] = x
            centers[count, 1] = y
            count += 1
        
        if count >= N:
            break
    
    return centers, radii

def make_random_init():
    """Create random initialization."""
    centers = np.zeros((N, 2))
    radii = np.full(N, 0.08)
    
    np.random.seed(42)
    for i in range(N):
        centers[i, 0] = np.random.uniform(0.15, 0.85)
        centers[i, 1] = np.random.uniform(0.15, 0.85)
    
    return centers, radii

def make_grid_init():
    """Create grid-like initialization."""
    centers = np.zeros((N, 2))
    radii = np.full(N, 0.085)
    
    # Try 6x5 grid pattern
    rows = 6
    cols = 5
    count = 0
    
    x_spacing = 1.0 / (cols + 1)
    y_spacing = 1.0 / (rows + 1)
    
    for row in range(rows):
        for col in range(cols):
            if count >= N:
                break
            centers[count, 0] = (col + 1) * x_spacing
            centers[count, 1] = (row + 1) * y_spacing
            count += 1
        if count >= N:
            break
    
    return centers, radii

def make_corner_fill_init():
    """Initialize with circles in corners and edges, then fill interior."""
    centers = np.zeros((N, 2))
    radii = np.full(N, 0.09)
    
    # Place 4 circles in corners
    centers[0] = [0.1, 0.1]
    centers[1] = [0.9, 0.1]
    centers[2] = [0.1, 0.9]
    centers[3] = [0.9, 0.9]
    
    # Place circles along edges
    count = 4
    edge_positions = [
        [0.3, 0.1], [0.5, 0.1], [0.7, 0.1],
        [0.3, 0.9], [0.5, 0.9], [0.7, 0.9],
        [0.1, 0.3], [0.1, 0.5], [0.1, 0.7],
        [0.9, 0.3], [0.9, 0.5], [0.9, 0.7],
    ]
    for pos in edge_positions:
        if count >= N:
            break
        centers[count] = pos
        count += 1
    
    # Fill interior
    interior_positions = [
        [0.3, 0.3], [0.5, 0.3], [0.7, 0.3],
        [0.3, 0.5], [0.5, 0.5], [0.7, 0.5],
        [0.3, 0.7], [0.5, 0.7], [0.7, 0.7],
    ]
    for pos in interior_positions:
        if count >= N:
            break
        centers[count] = pos
        count += 1
    
    return centers, radii

def try_optimization(x0, max_iter=500):
    """Run SLSQP optimization from given starting point."""
    flat_x0 = np.zeros(3 * N)
    if x0.shape[1] == 2:
        for i in range(N):
            flat_x0[3 * i] = x0[i, 0]
            flat_x0[3 * i + 1] = x0[i, 1]
            flat_x0[3 * i + 2] = 0.085
    else:
        for i in range(N):
            flat_x0[3 * i] = x0[i, 0]
            flat_x0[3 * i + 1] = x0[i, 1]
            flat_x0[3 * i + 2] = x0[i, 2]
    
    bounds = build_bounds()
    constraint = {'type': 'ineq', 'fun': constraints_func}
    
    result = minimize(
        objective_func,
        flat_x0,
        method='SLSQP',
        bounds=bounds,
        constraints=constraint,
        options={'maxiter': max_iter, 'ftol': 1e-14, 'disp': False}
    )
    
    return result.x

def try_optimization_trust(x0):
    """Try trust-constr method."""
    flat_x0 = np.zeros(3 * N)
    if x0.shape[1] == 2:
        for i in range(N):
            flat_x0[3 * i] = x0[i, 0]
            flat_x0[3 * i + 1] = x0[i, 1]
            flat_x0[3 * i + 2] = 0.085
    else:
        for i in range(N):
            flat_x0[3 * i] = x0[i, 0]
            flat_x0[3 * i + 1] = x0[i, 1]
            flat_x0[3 * i + 2] = x0[i, 2]
    
    bounds = build_bounds()
    
    from scipy.optimize import NonlinearConstraint
    nl_constraint = NonlinearConstraint(constraints_func, 0, np.inf)
    
    result = minimize(
        objective_func,
        flat_x0,
        method='trust-constr',
        bounds=bounds,
        constraints=[nl_constraint],
        options={'maxiter': 300, 'verbose': 0}
    )
    
    return result.x

def perturb_init(centers, radii, scale=0.02):
    """Add small random perturbation to initialization."""
    new_centers = centers.copy()
    new_radii = radii.copy()
    np.random.seed(None)
    for i in range(N):
        new_centers[i, 0] += np.random.uniform(-scale, scale)
        new_centers[i, 1] += np.random.uniform(-scale, scale)
        new_radii[i] += np.random.uniform(-scale * 0.5, scale * 0.5)
        # Clamp
        new_centers[i, 0] = np.clip(new_centers[i, 0], 0.01, 0.99)
        new_centers[i, 1] = np.clip(new_centers[i, 1], 0.01, 0.99)
        new_radii[i] = np.clip(new_radii[i], 0.01, 0.4)
    return new_centers, new_radii

def run_packing():
    """Main function to find optimal packing of 26 circles in unit square."""
    best_sum = -1.0
    best_flat = None
    
    # Generate various initial configurations
    inits = []
    
    # Hexagonal with different row distributions
    row_configs = [
        [5, 5, 5, 5, 6],
        [5, 6, 5, 6, 4],
        [4, 5, 5, 5, 4, 3],
        [5, 4, 5, 4, 5, 3],
        [6, 5, 5, 5, 5],
        [4, 4, 4, 4, 4, 4, 2],
        [5, 5, 5, 4, 4, 3],
        [6, 6, 5, 5, 4],
        [4, 5, 6, 5, 4, 2],
        [5, 5, 4, 5, 5, 2],
    ]
    
    for r in [0.085, 0.09, 0.095, 0.098, 0.10]:
        for rc in row_configs:
            total = sum(rc)
            if total < N:
                # Pad with extra circles in last row
                rc = list(rc)
                rc[-1] += N - total
            elif total > N:
                # Trim from last row
                rc = list(rc)
                rc[-1] -= (total - N)
            
            try:
                c, rr = make_hexagonal_init(r, rc)
                inits.append((c, rr))
            except:
                pass
    
    # Grid init
    inits.append(make_grid_init())
    
    # Corner fill init
    inits.append(make_corner_fill_init())
    
    # Random init
    np.random.seed(123)
    inits.append(make_random_init())
    
    # More random with different seeds
    for seed in [1, 2, 3, 5, 7, 11]:
        np.random.seed(seed)
        c = np.random.uniform(0.2, 0.8, (N, 2))
        r = np.full(N, 0.085)
        inits.append((c, r))
    
    # Perturbed versions of hexagonal inits
    for c, r in inits[:5]:
        for _ in range(3):
            cp, rp = perturb_init(c, r, scale=0.01)
            inits.append((cp, rp))
    
    # Try each initialization
    for idx, (centers, radii) in enumerate(inits):
        # Build flat vector with proper radii
        if radii.shape == (N, 2):
            rad = np.full(N, 0.085)
        else:
            rad = radii
        
        x0 = np.zeros(3 * N)
        for i in range(N):
            x0[3 * i] = centers[i, 0]
            x0[3 * i + 1] = centers[i, 1]
            x0[3 * i + 2] = rad[i]
        
        bounds = build_bounds()
        constraint = {'type': 'ineq', 'fun': constraints_func}
        
        # Try SLSQP
        try:
            result = minimize(
                objective_func,
                x0,
                method='SLSQP',
                bounds=bounds,
                constraints=constraint,
                options={'maxiter': 800, 'ftol': 1e-14, 'disp': False}
            )
            
            current_sum = -result.fun
            if current_sum > best_sum:
                best_sum = current_sum
                best_flat = result.x.copy()
        except:
            pass
    
    # Refine best solution with additional optimization rounds
    if best_flat is not None:
        for round_num in range(3):
            x0 = best_flat.copy()
            # Add tiny perturbation
            for i in range(3 * N):
                x0[i] += np.random.uniform(-1e-6, 1e-6)
            
            bounds = build_bounds()
            constraint = {'type': 'ineq', 'fun': constraints_func}
            
            try:
                result = minimize(
                    objective_func,
                    x0,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=constraint,
                    options={'maxiter': 1000, 'ftol': 1e-14, 'disp': False}
                )
                
                current_sum = -result.fun
                if current_sum > best_sum:
                    best_sum = current_sum
                    best_flat = result.x.copy()
            except:
                pass
    
    # Extract final solution
    centers = np.zeros((N, 2))
    radii = np.zeros(N)
    for i in range(N):
        centers[i, 0] = best_flat[3 * i]
        centers[i, 1] = best_flat[3 * i + 1]
        radii[i] = best_flat[3 * i + 2]
    
    # Final validation and correction
    # Ensure no negative radii
    radii = np.maximum(radii, 1e-10)
    
    # Ensure centers are within bounds considering radii
    for i in range(N):
        centers[i, 0] = np.clip(centers[i, 0], radii[i], 1.0 - radii[i])
        centers[i, 1] = np.clip(centers[i, 1], radii[i], 1.0 - radii[i])
    
    final_sum = np.sum(radii)
    
    return centers, radii, final_sum
