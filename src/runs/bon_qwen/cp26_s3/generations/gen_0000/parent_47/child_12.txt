from scipy.optimize import minimize

def _try_pack(n, r):
    best_loss = float('inf')
    best_centers = None
    
    # Generate initial configurations to avoid local minima
    inits = [_structured_init(n, r)]
    for _ in range(4):
        inits.append(np.random.rand(n, 2) * (1 - 2*r) + r)
        
    for init_centers in inits:
        res = minimize(_objective, init_centers.flatten(), 
                       args=(n, r), method='L-BFGS-B',
                       bounds=[(r, 1-r)] * (2*n),
                       options={'maxiter': 3000, 'ftol': 1e-12, 'gtol': 1e-12})
        
        if res.fun < best_loss:
            best_loss = res.fun

def _structured_init(n, r):
    # Generate points in a hexagonal lattice pattern
    pts = []
    for i in range(8):
        for j in range(8):
            x = i + 0.5 * (j % 2)
            y = j * np.sqrt(3)/2
            pts.append([x, y])
            if len(pts) >= n:
                break
        if len(pts) >= n:
            break
            
    pts = np.array(pts[:n])
    pts -= pts.min(axis=0)
    rng = pts.max(axis=0) - pts.min(axis=0)
    rng[rng == 0] = 1.0
    pts = (pts - pts.min(axis=0)) / rng
    # Scale and shift to fit within [r, 1-r]
    pts = pts * (1 - 2*r) + r
    return pts

def _objective(centers_flat, n, r):
    centers = centers_flat.reshape(n, 2)
    loss = 0.0
    
    # Boundary penalty
    for i in range(n):
        for d in range(2):
            val = centers[i, d]
            if val < r:
                loss += (r - val)**2
            elif val > 1-r:
                loss += (val - (1-r))**2
                
    # Overlap penalty
    for i in range(n):
        for j in range(i+1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx*dx + dy*dy)
            if dist < 2*r:
                loss += (2*r - dist)**2
                
    return loss
