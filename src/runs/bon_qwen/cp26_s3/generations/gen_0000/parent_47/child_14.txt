def objective(x_vec, n):

        # We want x - r >= 0. In scipy minimize, bounds or constraints <= 0.
        # Let's use simple bounds for x,y,r to restrict search space, 
        # and use constraints for overlaps.
        # But for boundaries, we can just clamp or use constraints.
        # Using constraints:
        # r - x <= 0
        # x + r - 1 <= 0
        # r - y <= 0

        constraints.append(r - x)
        constraints.append(x + r - 1)
        constraints.append(r - y)
        constraints.append(y + r - 1)
        
    return np.array(constraints)

def overlap_constraints(x_vec, n):
