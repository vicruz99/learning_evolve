def compute_constraints_and_objective(params, n=26):
    """
    Helper function to compute objective and constraints for scipy optimizer.
    params: flat array of [x1, y1, r1, x2, y2, r2, ...]
    """
    centers = params[:2*n].reshape(n, 2)
    radii = params[2*n:]
    
    # Objective: Maximize sum of radii -> Minimize negative sum
    obj = -np.sum(radii)
    
    constraints = []
    
    # Boundary constraints: x >= r, x <= 1-r, y >= r, y <= 1-r
    # Equivalent to: x - r >= 0, 1 - x - r >= 0, etc.

        x, y = centers[i]
        r = radii[i]
        constraints.append(x - r)
        constraints.append(1 - x - r)
        constraints.append(y - r)
        constraints.append(1 - y - r)
        
    # Overlap constraints: dist(i, j) >= r_i + r_j
    # dist^2 >= (r_i + r_j)^2
    # (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
    for i in range(n):

            dist_sq = dx*dx + dy*dy
            r_sum = radii[i] + radii[j]
            constraints.append(dist_sq - r_sum*r_sum)
            
    return obj, np.array(constraints)
