import numpy as np
import math
from scipy.optimize import minimize


def compute_forces(centers, radii, n):
    """Compute repulsive forces between circles and boundary forces."""
    forces = np.zeros((n, 2))

    for i in range(n):
        x, y = centers[i]
        r = radii[i]

        if x - r < 0:
            forces[i, 0] += (r - x) * 500
        if x + r > 1:
            forces[i, 0] -= (x + r - 1) * 500
        if y - r < 0:
            forces[i, 1] += (r - y) * 500
        if y + r > 1:
            forces[i, 1] -= (y + r - 1) * 500

    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[j, 0] - centers[i, 0]
            dy = centers[j, 1] - centers[i, 1]
            dist_sq = dx * dx + dy * dy
            if dist_sq < 1e-20:
                dist = 1e-10
                dx = 1e-10
            else:
                dist = math.sqrt(dist_sq)
            min_dist = radii[i] + radii[j]
            if dist < min_dist:
                overlap = min_dist - dist
                nx = dx / dist
                ny = dy / dist
                force = overlap * 1000
                forces[i, 0] -= nx * force
                forces[i, 1] -= ny * force
                forces[j, 0] += nx * force
                forces[j, 1] += ny * force

    return forces


def hexagonal_init(n, row_counts, rng_seed=0):
    """Initialize circles in a hexagonal pattern with given row counts."""
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.08

    num_rows = len(row_counts)
    y_spacing = 1.0 / (num_rows + 1)
    max_count = max(row_counts)
    x_spacing = 1.0 / (max_count + 1)

    idx = 0
    for row_idx, count in enumerate(row_counts):
        y = (row_idx + 1) * y_spacing
        x_offset = (max_count - count) * x_spacing / 2
        for j in range(count):
            x = x_offset + (j + 0.5) * x_spacing
            centers[idx] = [x, y]
            radii[idx] = min(x_spacing / 2, y_spacing * 0.866 / 2)
            idx += 1

    # Small random perturbation
    rng = np.random.RandomState(rng_seed)
    centers += rng.randn(n, 2) * 0.003
    radii += rng.randn(n) * 0.002
    radii = np.maximum(radii, 0.01)

    return centers, radii


def force_optimize(centers, radii, n, max_iter=150000):
    """Optimize packing using force-based relaxation with radius expansion."""
    for iteration in range(max_iter):
        # Slowly expand all radii
        expansion_rate = 0.000015 / (1 + iteration * 0.0003)
        radii += expansion_rate

        forces = compute_forces(centers, radii, n)

        # Adaptive damping
        damping = 0.0005 / (1 + iteration * 0.00005)
        centers += forces * damping

        # Clamp centers to valid range
        centers[:, 0] = np.clip(centers[:, 0], 0, 1)
        centers[:, 1] = np.clip(centers[:, 1], 0, 1)

    return centers, radii


def objective_neg_sum(x, n):
    """Objective: maximize sum of radii (negated for minimization)."""
    return -np.sum(x[2 * n:])


def constraint_all(x, n):
    """All constraints: boundary, overlap, non-negative radii."""
    centers = x[:2 * n].reshape(n, 2)
    radii = x[2 * n:]

    num_boundary = 4 * n
    num_overlap = n * (n - 1) // 2
    num_constraints = num_boundary + num_overlap + n

    c = np.zeros(num_constraints)
    idx = 0

    # Boundary constraints
    for i in range(n):
        c[idx] = centers[i, 0] - radii[i]
        idx += 1
        c[idx] = 1.0 - centers[i, 0] - radii[i]
        idx += 1
        c[idx] = centers[i, 1] - radii[i]
        idx += 1
        c[idx] = 1.0 - centers[i, 1] - radii[i]
        idx += 1

    # Overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            c[idx] = math.sqrt(dx * dx + dy * dy) - radii[i] - radii[j]
            idx += 1

    # Non-negative radii
    for i in range(n):
        c[idx] = radii[i]
        idx += 1

    return c


def scipy_refine(centers, radii, n):
    """Refine packing using scipy SLSQP optimizer."""
    x0 = np.concatenate([centers.flatten(), radii])

    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n

    constraint_dict = {
        'type': 'ineq',
        'fun': constraint_all,
        'args': (n,)
    }

    obj_dict = {
        'fun': objective_neg_sum,
        'args': (n,)
    }

    result = minimize(
        obj_dict, x0, method='SLSQP', bounds=bounds,
        constraints=constraint_dict,
        options={'maxiter': 2000, 'ftol': 1e-14, 'disp': False}
    )

    centers_opt = result.x[:2 * n].reshape(n, 2)
    radii_opt = result.x[2 * n:]

    return centers_opt, radii_opt


def validate_packing_internal(centers, radii, n):
    """Internal validation check."""
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-9 or x + r > 1 + 1e-9 or y - r < -1e-9 or y + r > 1 + 1e-9:
            return False
        if r < -1e-9:
            return False

    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = math.sqrt(dx * dx + dy * dy)
            if dist < radii[i] + radii[j] - 1e-9:
                return False

    return True
