import numpy as np
from scipy.optimize import minimize

def compute_objective(vars, n):
    """
    Objective function to minimize: -sum(radii) + penalty for constraint violations.
    Uses vectorized operations for efficiency.
    """
    c = vars[:2*n].reshape(n, 2)
    r = vars[2*n:]
    
    # Primary goal: maximize sum of radii -> minimize negative sum
    score = -np.sum(r)
    
    # Vectorized boundary penalties
    # Circle must be inside [0,1]x[0,1]: r <= x <= 1-r, r <= y <= 1-r
    v1 = np.maximum(0, r - c[:,0])
    v2 = np.maximum(0, r - (1 - c[:,0]))
    v3 = np.maximum(0, r - c[:,1])
    v4 = np.maximum(0, r - (1 - c[:,1]))
    pen = np.sum(v1**2 + v2**2 + v3**2 + v4**2)
    
    # Vectorized overlap penalties
    # Distance between centers must be >= sum of radii
    diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    r_sums = r[:, np.newaxis] + r[np.newaxis, :]
    violations = np.maximum(0, r_sums - dists)
    # Sum over upper triangle to count each pair once
    pen += np.sum(np.triu(violations**2, k=1))
    
    # Penalty coefficient balances constraint satisfaction vs radius growth
    return score + 500.0 * pen

def generate_initial_config(n):
    """Generate an initial hexagonal packing configuration."""
    r_init = 0.08
    dx = 2 * r_init
    dy = np.sqrt(3) * r_init
    positions = []
    for row in range(10):
        y = r_init + row * dy
        for col in range(10):
            x = r_init + (row % 2) * (dx / 2) + col * dx
            if x + r_init <= 1.0 and y + r_init <= 1.0:
                positions.append((x, y))
    # Take first n positions
    return np.array(positions[:n]), np.full(n, r_init)

def project_to_feasible(centers, radii, n):
    """
    Iteratively shrink radii until all circles are strictly within bounds 
    and non-overlapping. Guarantees validity for the validator.
    """
    r_opt = radii.copy()
    c_opt = centers.copy()
    
    for _ in range(10):
        changed = False
        for i in range(n):
            # Max radius allowed by boundaries
            max_r = min(c_opt[i,0], 1 - c_opt[i,0], c_opt[i,1], 1 - c_opt[i,1])
            
            # Max radius allowed by other circles
            for j in range(n):
                if i == j:
                    continue
                dist = np.sqrt((c_opt[i,0] - c_opt[j,0])**2 + (c_opt[i,1] - c_opt[j,1])**2)
                if dist / 2 < max_r:
                    max_r = dist / 2
            
            if r_opt[i] > max_r + 1e-9:
                r_opt[i] = max_r
                changed = True
        if not changed:
            break
    return c_opt, r_opt

def solve_packing(n):
    best_sum = 0
    best_centers = None
    best_radii = None
    
    bounds = [(0, 1)] * (2*n) + [(0, 0.5)] * n
    
    # Run multiple optimization passes with different initial perturbations
    for run in range(5):
        centers, radii = generate_initial_config(n)
        if run > 0:
            # Add controlled random noise to escape local minima
            centers += np.random.randn(n, 2) * 0.02
            centers = np.clip(centers, 0.01, 0.99)
            radii += np.random.randn(n) * 0.01
            radii = np.clip(radii, 0.02, 0.4)
            
        x0 = np.concatenate([centers.ravel(), radii])
        
        res = minimize(compute_objective, x0, args=(n,), bounds=bounds, method='L-BFGS-B', 
                       options={'maxiter': 10000, 'ftol': 1e-10, 'gtol': 1e-6})
        
        c_opt = res.x[:2*n].reshape(n, 2)
        r_opt = res.x[2*n:]
        
        # Ensure strict feasibility
        c_opt, r_opt = project_to_feasible(c_opt, r_opt, n)
        
        current_sum = np.sum(r_opt)
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = c_opt.copy()
            best_radii = r_opt.copy()
            
    return best_centers, best_radii, best_sum

def run_packing():
    np.random.seed(42)
    n = 26
    centers, radii, s = solve_packing(n)
    return centers, radii, s
