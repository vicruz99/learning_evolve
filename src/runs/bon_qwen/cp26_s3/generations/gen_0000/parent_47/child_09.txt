import numpy as np
from scipy.optimize import minimize

def generate_initial_config(n):
    """Generate a hexagonal grid initial placement for n circles."""
    pts = []
    r_grid = 0.09  # Spacing parameter
    dy = r_grid * np.sqrt(3)
    row = 0
    while len(pts) < n:
        y = r_grid + row * dy
        if y + r_grid > 1.0:
            break
        offset = r_grid * 1.5 if row % 2 == 1 else r_grid
        col = 0
        while len(pts) < n:
            x = offset + col * 2 * r_grid
            if x + r_grid > 1.0:
                break
            pts.append([x, y])
            col += 1
        row += 1
    
    pts = np.array(pts[:n])
    # Center the configuration in the unit square
    min_c = pts.min(axis=0)
    max_c = pts.max(axis=0)
    center = (min_c + max_c) / 2.0
    pts = pts - center + 0.5
    
    # Keep strictly inside to guarantee initial feasibility
    pts = np.clip(pts, 0.1, 0.9)
    radii = np.full(n, 0.07)
    return pts, radii

def objective_func(x, n):
    """Negative sum of radii (to be minimized)."""
    return -np.sum(x[2*n:])

def constraint_nonoverlap_func(x, n, idx):
    """Constraint: squared distance >= squared sum of radii for all pairs."""
    c = x[:2*n].reshape(n, 2)
    r = x[2*n:]
    c_diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
    dist_sq = np.sum(c_diff**2, axis=2)
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    return (dist_sq - r_sum**2)[idx]

def constraint_bound_func(x, n):
    """Constraint: circles inside [0,1]x[0,1]."""
    c = x[:2*n].reshape(n, 2)
    r = x[2*n:]
    return np.concatenate([
        c[:, 0] - r,
        1.0 - c[:, 0] - r,
        c[:, 1] - r,
        1.0 - c[:, 1] - r
    ])

def run_packing():
    n = 26
    centers_init, radii_init = generate_initial_config(n)
    x0 = np.concatenate([centers_init.ravel(), radii_init])
    
    bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n
    idx = np.triu_indices(n, k=1)
    
    constraints = [
        {'type': 'ineq', 'fun': constraint_nonoverlap_func, 'args': (n, idx)},
        {'type': 'ineq', 'fun': constraint_bound_func, 'args': (n,)}
    ]
    
    res = minimize(objective_func, x0, method='SLSQP', bounds=bounds,
                   constraints=constraints, args=(n,),
                   options={'maxiter': 2000, 'ftol': 1e-12})
    
    x_opt = res.x
    centers = x_opt[:2*n].reshape(n, 2)
    radii = x_opt[2*n:]
    
    # Ensure non-negative radii
    radii = np.maximum(radii, 1e-9)
    centers = np.clip(centers, 0.0, 1.0)
    
    # Project centers to respect boundary constraints exactly
    for i in range(n):
        centers[i, 0] = np.clip(centers[i, 0], radii[i], 1.0 - radii[i])
        centers[i, 1] = np.clip(centers[i, 1], radii[i], 1.0 - radii[i])
        
    # Safety check: if numerical precision caused overlaps, shrink slightly
    valid = True
    for i in range(n):
        for j in range(i+1, n):
            d = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
            if d < radii[i] + radii[j] - 1e-7:
                valid = False
                break
        if not valid: 
            break
            
    if not valid:
        radii *= 0.95
        for i in range(n):
            centers[i, 0] = np.clip(centers[i, 0], radii[i], 1.0 - radii[i])
            centers[i, 1] = np.clip(centers[i, 1], radii[i], 1.0 - radii[i])
            
    return centers, radii, np.sum(radii)
