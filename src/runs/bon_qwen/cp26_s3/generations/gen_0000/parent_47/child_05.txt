import numpy as np
from scipy.optimize import minimize

def compute_objective(vars, n, P):
    """
    Computes the penalized objective function.
    vars: flattened array [x1, y1, ..., xn, yn, r1, ..., rn]
    n: number of circles
    P: penalty coefficient
    """
    # Base objective: minimize negative sum of radii
    obj = -np.sum(vars[2*n:])
    
    # Boundary constraint penalties
    for i in range(n):
        xi = vars[2*i]
        yi = vars[2*i+1]
        ri = vars[2*n+i]
        
        if xi - ri < 0.0:
            obj += P * (xi - ri)**2
        if 1.0 - xi - ri < 0.0:
            obj += P * (1.0 - xi - ri)**2
        if yi - ri < 0.0:
            obj += P * (yi - ri)**2
        if 1.0 - yi - ri < 0.0:
            obj += P * (1.0 - yi - ri)**2
            
    # Pairwise non-overlap penalties
    for i in range(n):
        xi = vars[2*i]
        yi = vars[2*i+1]
        ri = vars[2*n+i]
        for j in range(i+1, n):
            xj = vars[2*j]
            yj = vars[2*j+1]
            rj = vars[2*n+j]
            
            dx = xi - xj
            dy = yi - yj
            dist = np.sqrt(dx*dx + dy*dy)
            overlap = (ri + rj) - dist
            
            if overlap > 0.0:
                obj += P * overlap**2
                
    return obj

def objective_wrapper(vars, n, P):
    """Wrapper to match scipy.optimize signature"""
    return compute_objective(vars, n, P)

def hex_grid_init(n):
    """Generates a hexagonal lattice initialization for n circles"""
    positions = []
    row_y = 0.13
    while len(positions) < n:
        row_x = 0.13
        offset = 0.0 if len(positions) % 2 == 0 else 0.11
        while row_x <= 0.87 and len(positions) < n:
            positions.append((row_x + offset, row_y))
            row_x += 0.23
        row_y += 0.19
    return np.array(positions[:n])

def run_packing():
    n = 26
    best_vars = None
    best_sum = -np.inf
    
    # Box constraints: coordinates in [0, 1], radii in [0, 0.5]
    bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n

    penalties = [50.0, 200.0, 1000.0, 5000.0]

        if current_sum > best_sum:
            best_sum = current_sum
            best_vars = current_vars.copy()
            
    centers = best_vars[:2*n].reshape(n, 2)
    radii = best_vars[2*n:]

    # 1. Enforce boundaries
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        radii[i] = min(r, x, 1.0 - x, y, 1.0 - y)

    changed = True
    while changed:
        changed = False
