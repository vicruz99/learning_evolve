import numpy as np
from scipy.optimize import minimize

def get_cost(centers, r):
    """
    Calculates the cost (squared overlaps + squared boundary violations) for a given set of centers and radius r.
    """
    n = centers.shape[0]
    cost = 0.0
    
    # Pairwise overlaps: sum(max(0, 2r - dist)^2)
    # Vectorized calculation for efficiency
    # Compute all pairwise distances
    # Using broadcasting: (n, 1, 2) - (1, n, 2) -> (n, n, 2)
    # This might be memory intensive for large n, but n=26 is small.
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.linalg.norm(diff, axis=2)
    
    # We only need upper triangle to avoid double counting and self-distance
    # Create a mask or just sum all and divide by 2 (diagonal is 0, so ok)
    # Actually, dists[i,j] is symmetric.
    # Sum of squares over all i != j is 2 * sum_{i<j}.
    # Overlap condition: dist < 2r.
    
    # Mask for overlaps
    overlaps = np.maximum(0, 2 * r - dists)
    pair_cost = np.sum(overlaps**2) / 2.0
    
    # Boundary violations
    # x >= r, x <= 1-r, y >= r, y <= 1-r
    x = centers[:, 0]
    y = centers[:, 1]
    
    # Violations: max(0, r - x) + max(0, x - (1-r)) + ...
    v_x_left = np.maximum(0, r - x)
    v_x_right = np.maximum(0, x - (1 - r))
    v_y_bottom = np.maximum(0, r - y)
    v_y_top = np.maximum(0, y - (1 - r))
    
    boundary_cost = np.sum(v_x_left**2 + v_x_right**2 + v_y_bottom**2 + v_y_top**2)
    
    return pair_cost + boundary_cost
