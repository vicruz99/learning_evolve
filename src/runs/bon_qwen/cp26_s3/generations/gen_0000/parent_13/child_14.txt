from scipy.optimize import minimize, differential_evolution, basinhopping
import cvxpy as cp


def solve_radii_lp(centers):
    """Given fixed centers, find optimal radii via LP."""
    n = centers.shape[0]
    r = cp.Variable(n)
    constraints = []

    for i in range(n):
        x, y = centers[i]
        constraints.append(r[i] <= x)
        constraints.append(r[i] <= 1.0 - x)
        constraints.append(r[i] <= y)
        constraints.append(r[i] <= 1.0 - y)
        constraints.append(r[i] >= 0)

    for i in range(n):
        for j in range(i + 1, n):
            d = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            constraints.append(r[i] + r[j] <= d)

    prob = cp.Problem(cp.Maximize(cp.sum(r)), constraints)
    prob.solve(solver=cp.ECOS, verbose=False)

    if r.value is None:
        return np.zeros(n)
    return np.array(r.value)
