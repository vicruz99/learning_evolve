import numpy as np
from scipy.optimize import minimize, linprog
import time

def get_max_radii_sum(centers):
    """
    Solves the LP to find the maximum sum of radii for a fixed set of centers.
    """
    n = centers.shape[0]
    c = -np.ones(n) # Maximize sum(r) <=> Minimize -sum(r)
    
    # Number of constraints: 4*n (boundaries) + n*(n-1)/2 (pairs)
    num_constraints = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_constraints, n))
    b_ub = np.zeros(num_constraints)
    
    row = 0
    
    # Boundary constraints: r_i <= x, r_i <= 1-x, r_i <= y, r_i <= 1-y
    for i in range(n):
        x, y = centers[i]
        # r_i <= x
        A_ub[row, i] = 1.0
        b_ub[row] = x
        row += 1
        # r_i <= 1-x
        A_ub[row, i] = 1.0
        b_ub[row] = 1.0 - x
        row += 1
        # r_i <= y
        A_ub[row, i] = 1.0
        b_ub[row] = y
        row += 1
        # r_i <= 1-y
        A_ub[row, i] = 1.0
        b_ub[row] = 1.0 - y
        row += 1
        
    # Pairwise constraints: r_i + r_j <= dist_ij
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt((centers[i, 0] - centers[j, 0])**2 + (centers[i, 1] - centers[j, 1])**2)
            A_ub[row, i] = 1.0
            A_ub[row, j] = 1.0
            b_ub[row] = dist
            row += 1
            
    bounds = [(0, None) for _ in range(n)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return -res.fun, res.x
    except Exception:
        pass
    return 0.0, np.zeros(n)

def objective(centers_flat):
    centers = centers_flat.reshape(-1, 2)
    # Ensure centers are within [0,1] for the LP to be well-behaved, 
    # though the optimizer should naturally keep them there.
    centers = np.clip(centers, 0, 1)
    val, _ = get_max_radii_sum(centers)
    return -val

def get_initial_centers():
    n = 26
    # Generate a hexagonal lattice pattern
    centers = []
    rows = 6
    points_per_row = [5, 4, 5, 4, 5, 3] # Sum = 26
    # Adjust widths/heights to fit in [0,1]
    
    # Simple scaling for hexagonal packing
    # Horizontal spacing
    w = 1.0 / (max(points_per_row) + 0.5) 
    h = np.sqrt(3) / 2 * w
    
    # Vertical offset to center
    y_curr = h / 2
    for i, count in enumerate(points_per_row):
        # Center the row horizontally
        row_width = (count - 1) * w
        x_start = (1.0 - row_width) / 2
        
        if i % 2 == 1: # Shift odd rows
            x_start += w / 2
            
        for j in range(count):
            x = x_start + j * w
            y = y_curr
            # Keep within bounds roughly
            if 0 <= x <= 1 and 0 <= y <= 1:
                centers.append([x, y])
        
        y_curr += h
        if len(centers) >= n:
            break
            
    centers = np.array(centers[:n])
    # If we didn't get 26, pad with random points (unlikely with this config)
    while len(centers) < n:
        centers = np.vstack([centers, [np.random.rand(), np.random.rand()]])
        
    return centers

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    best_centers = None
    best_radii = None
    
    # Generate multiple initial seeds for robust optimization
    seeds = []
    base_centers = get_initial_centers()

    for _ in range(5):

        seeds.append(perturbed.flatten())

    final_sum, final_radii = get_max_radii_sum(best_centers)
