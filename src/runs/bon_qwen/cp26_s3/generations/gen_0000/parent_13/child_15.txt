

<strategy>
To maximize the sum of radii of 26 circles packed in a unit square, we transform the problem into optimizing the positions of the circle centers. For any fixed set of centers, the maximum valid radius for each circle is determined by its distance to the nearest neighbor and its distance to the square boundaries. Specifically, `r_i = min( min_{j≠i} ||p_i - p_j||/2, dist(p_i, boundary) )`. The objective becomes finding 26 points in `[0,1]²` that maximize the sum of these derived radii.

This is a non-smooth, non-convex optimization problem with 52 variables. We will use a robust derivative-free optimizer (`Powell` method from `scipy.optimize`) which handles non-smooth landscapes well. To avoid poor local optima, we will run multiple restarts with diverse initial configurations:
1. A structured 5×5 grid + center point.
2. A hexagonal-inspired layout.
3. Several randomized configurations.

A penalty term is added to the objective function to strongly discourage centers from leaving the unit square, keeping the optimization unconstrained internally while satisfying the problem constraints. After optimization, we clamp centers to `[0,1]`, recompute exact radii, and return the best valid configuration found.

```python
import numpy as np
from scipy.optimize import minimize

N = 26

def compute_packing(centers):
    """
    Given circle centers, compute the maximum valid radii and their sum.
    """
    # Compute pairwise distances: shape (N, N)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    # Ignore self-distances
    np.fill_diagonal(dists, np.inf)
    
    # Minimum distance to any other circle center
    min_dists = np.min(dists, axis=1)
    
    # Minimum distance to square boundaries
    # Distance to left/right: x, 1-x. Distance to bottom/top: y, 1-y.
    dist_bounds = np.minimum(centers, 1.0 - centers)
    min_bounds = np.min(dist_bounds, axis=1)
    
    # Radius is limited by half the distance to nearest neighbor and boundary distance
    radii = np.minimum(min_dists / 2.0, min_bounds)
    radii = np.maximum(radii, 0.0)  # Ensure non-negative
    
    return radii, np.sum(radii)

def objective(x):
    """
    Objective function for optimization: negative sum of radii + boundary penalty.
    """
    centers = x.reshape(N, 2)
    _, s = compute_packing(centers)
    
    # Strong penalty if any center is outside [0, 1]
    bounds_val = np.minimum(centers, 1.0 - centers)
    outside = bounds_val < 0
    penalty = 0.0
    if np.any(outside):
        penalty = 1e6 * np.sum(bounds_val[outside]**2)
        
    return -s + penalty

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Prepare diverse initial configurations
    starts = []
    
    # 1. 5x5 Grid + 1 center point (25 + 1 = 26)
    grid_pts = []
    gs = np.linspace(0.12, 0.88, 5)
    for x in gs:
        for y in gs:
            grid_pts.append([x, y])
    grid_pts.append([0.5, 0.5])
    starts.append(np.array(grid_pts).flatten())
    
    # 2. Hexagonal-ish lattice start
    pts_hex = []
    y_pos = 0.12
    while len(pts_hex) < 26:
        x_pos = 0.12
        while x_pos <= 0.88 and len(pts_hex) < 26:
            pts_hex.append([x_pos, y_pos])
            x_pos += 0.18
        y_pos += 0.155
    starts.append(np.array(pts_hex[:26]).flatten())
    
    # 3. Random starts
    for _ in range(4):
        starts.append(np.random.rand(N * 2))
        
    # Optimization settings
    opt_kwargs = {
        'method': 'Powell',
        'options': {
            'maxiter': 2000,
            'ftol': 1e-10,
            'xtol': 1e-9
        }
    }
    
    for x0 in starts:
        try:
            res = minimize(objective, x0, **opt_kwargs)
            
            # Extract and validate centers
            cur_centers = res.x.reshape(N, 2)
            
            # Strictly enforce [0, 1] bounds
            cur_centers = np.clip(cur_centers, 0.0, 1.0)
            
            # Recompute radii exactly for the clipped centers
            cur_radii, cur_sum = compute_packing(cur_centers)
            
            # Check validity (no NaN, radii >= 0, inside bounds)
            if not np.isnan(cur_sum) and np.all(cur_radii >= 0):
                if cur_sum > best_sum:
                    best_sum = cur_sum
                    best_centers = cur_centers
                    best_radii = cur_radii
        except Exception:
            continue
            
    # Fallback in case optimization fails completely
    if best_centers is None:
        best_centers = np.random.rand(N, 2) * 0.8 + 0.1
        best_radii, best_sum = compute_packing(best_centers)
        
    return best_centers, best_radii, float(best_sum)
```
