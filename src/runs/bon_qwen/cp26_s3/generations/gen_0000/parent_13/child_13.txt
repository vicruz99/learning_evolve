import numpy as np
from scipy.optimize import minimize

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    n = centers.shape[0]
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False
    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False
    return True

def run_packing():
    n = 26
    rng = np.random.default_rng(42)

    # Function to generate initial configurations
    def generate_initial_configurations():

        radii_grid = np.full(n, 0.1)

        for i in range(25):
            centers_grid[i] = coords[i]
        
        # 26th circle in center gap (approx)
        centers_grid[25] = [0.5, 0.5]
        radii_grid[25] = 0.01 # Small radius to start
        
        configs.append((centers_grid.copy(), radii_grid.copy()))
        
        # 3. Hexagonal packing attempt
        centers_hex = np.zeros((n, 2))
        radii_hex = np.full(n, 0.09) # Slightly less than 0.1
        idx = 0
        for row in range(7): # 7 rows to fit 26
            n_circles = 5 if row % 2 == 0 else 4 # Alternating 5, 4
            if idx + n_circles > n:
                n_circles = n - idx
            y = 0.1 + row * 0.13 # Spacing roughly sqrt(3)/2 * 0.2 + margin
            for col in range(n_circles):
                x = 0.1 + col * 0.2 + (0.1 if row % 2 != 0 else 0.0)
                if idx < n:
                    centers_hex[idx] = [x, y]
                    idx += 1
        configs.append((centers_hex.copy(), radii_hex.copy()))
        
        return configs

    def check_validity(centers, radii):
        # Boundary check
        if np.any(radii < 1e-5) or np.any(radii > 0.5):
            return False
        # Overlap check
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x < r + 1e-9 or x > 1 - r - 1e-9 or y < r + 1e-9 or y > 1 - r - 1e-9:
                return False
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist < radii[i] + radii[j] - 1e-9:
                    return False
        return True

    best_centers = None
    best_radii = None
    best_sum = -1.0

    configs = generate_initial_configurations()

    for c_idx, (centers_init, radii_init) in enumerate(configs):
        # Local Optimization using scipy
        # Flatten variables: [x1, y1, r1, x2, y2, r2, ...]
        x0 = np.zeros(3 * n)
        for i in range(n):
            x0[3*i] = centers_init[i, 0]
            x0[3*i+1] = centers_init[i, 1]
            x0[3*i+2] = radii_init[i]

        def objective(x):
            # Maximize sum of radii => Minimize negative sum
            current_radii = x[2::3]
            return -np.sum(current_radii)

        def constraints_func(x):
            cons = []
            # Boundary constraints: r <= x <= 1-r  =>  x - r >= 0, 1 - x - r >= 0
            # We express as g(x) >= 0

                xi = x[3*i]
                yi = x[3*i+1]
                ri = x[3*i+2]
                # x >= r
                cons.append({'type': 'ineq', 'fun': lambda x, idx=i: x[3*idx] - x[3*idx+2]})
                # 1 - x >= r
                cons.append({'type': 'ineq', 'fun': lambda x, idx=i: 1 - x[3*idx] - x[3*idx+2]})
                # y >= r
                cons.append({'type': 'ineq', 'fun': lambda x, idx=i: x[3*idx+1] - x[3*idx+2]})
                # 1 - y >= r
                cons.append({'type': 'ineq', 'fun': lambda x, idx=i: 1 - x[3*idx+1] - x[3*idx+2]})

                cons.append({'type': 'ineq', 'fun': lambda x, idx=i: x[3*idx+2]})

            # dist^2 - (r1+r2)^2 >= 0

                    cons.append({
                        'type': 'ineq',

        # Bounds
        bounds = [(0.0, 1.0) for _ in range(2*n)] + [(0.0, 0.5) for _ in range(n)]

            if res.success or res.fun < -best_sum:
                current_sum = -res.fun

                        best_centers[i, 0] = res.x[3*i]
                        best_centers[i, 1] = res.x[3*i+1]
                        best_radii[i] = res.x[3*i+2]
                    best_sum = current_sum
        except Exception:
            continue

    # Final validation and cleanup
    if best_centers is not None:
        # Small correction to ensure strict validity within tolerance
        # (The optimizer might hit boundary exactly)
        if not validate_packing(best_centers, best_radii):
