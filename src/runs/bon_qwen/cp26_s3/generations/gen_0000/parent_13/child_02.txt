import numpy as np
from scipy.optimize import minimize

def objective(vars, n, penalty):
    """
    Compute the objective value: negative sum of radii plus penalty terms for violations.
    """
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    # Primary objective: maximize sum of radii
    val = -np.sum(r)
    
    # Pairwise overlap penalties
    cx = x[:, np.newaxis] - x[np.newaxis, :]
    cy = y[:, np.newaxis] - y[np.newaxis, :]
    d = np.sqrt(cx**2 + cy**2)
    r_s = r[:, np.newaxis] + r[np.newaxis, :]
    
    # Only consider upper triangle to avoid double counting and self-pairs
    v = np.maximum(0, r_s - d)
    v = np.triu(v, k=1)
    val += penalty * np.sum(v**2)
    
    # Boundary violation penalties
    val += penalty * np.sum(np.maximum(0, r - x)**2)
    val += penalty * np.sum(np.maximum(0, r - (1.0 - x))**2)
    val += penalty * np.sum(np.maximum(0, r - y)**2)
    val += penalty * np.sum(np.maximum(0, r - (1.0 - y))**2)
    
    return val

def generate_hex_init(n):

    r = 0.1

        counts[-1] += 1

            cx.append(x_start)
            cy.append(y_pos)
            x_start += 2*r
        y_pos += np.sqrt(3)*r
        
    return np.array(cx), np.array(cy)

def run_packing():
