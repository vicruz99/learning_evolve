from scipy.optimize import minimize

def run_packing():
    n_circles = 26
    
    # 1. Initialization
    # 5x5 grid + 1 circle
    centers = np.zeros((n_circles, 2))
    radii = np.zeros(n_circles)
    
    grid_x = np.array([0.1, 0.3, 0.5, 0.7, 0.9])
    grid_y = np.array([0.1, 0.3, 0.5, 0.7, 0.9])
    
    idx = 0
    for y in grid_y:
        for x in grid_x:
            centers[idx] = [x, y]
            idx += 1
    # 26th circle
    centers[25] = [0.2, 0.2]
    
    r_init = 0.05 
    radii[:] = r_init
    
    # Add small noise to break symmetry
    np.random.seed(42)
    centers += np.random.uniform(-0.001, 0.001, centers.shape)
    # Clamp to keep inside reasonable bounds
    centers = np.clip(centers, r_init, 1-r_init)
    
    # 2. Force-based Expansion (Jiggling)
    # Helps to find a locally dense packing before optimization
    n_steps = 1000
    dt = 0.005
    k_rep = 20.0
    k_wall = 50.0
    
    for step in range(n_steps):
        # Grow radii slowly
        radii *= 1.00005 
        
        forces = np.zeros_like(centers)
        
        # Inter-circle repulsion
        # Vectorized computation could be faster, but loops are okay for 26
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                diff = centers[i] - centers[j]
                dist_sq = np.dot(diff, diff)
                if dist_sq < 1e-12:
                    dist = 1e-6
                    diff = np.array([1e-6, 0.0]) # Dummy direction
                else:
                    dist = np.sqrt(dist_sq)
                
                min_dist = radii[i] + radii[j]
                if dist < min_dist:
                    overlap = min_dist - dist
                    # Force proportional to overlap
                    f = k_rep * overlap
                    fx = f * diff[0] / dist
                    fy = f * diff[1] / dist
                    forces[i, 0] += fx
                    forces[i, 1] += fy
                    forces[j, 0] -= fx
                    forces[j, 1] -= fy
        
        # Wall repulsion
        for i in range(n_circles):
            x, y = centers[i]
            r = radii[i]
            
            # Left
            if x < r:
                forces[i, 0] += k_wall * (r - x)
            # Right
            elif x > 1 - r:
                forces[i, 0] -= k_wall * (x - (1 - r))
            # Bottom
            if y < r:
                forces[i, 1] += k_wall * (r - y)
            # Top
            elif y > 1 - r:
                forces[i, 1] -= k_wall * (y - (1 - r))
        
        centers += dt * forces
        
        # Clamp centers to [0, 1]
        # But actually, we should clamp to [r, 1-r] ideally, 
        # but r is changing.

        centers = np.clip(centers, 0, 1)
        
        # Ensure centers respect current radii bounds roughly
        # If x < r, force pushes right.
        # But if we clamp x to 0, and r > 0, we are outside.
        # The next step's wall force will be huge.
        # It's okay.

    # 3. Optimization
    # Map to (r, u, v) parameters
    # x = r + (1-2r)u
    
    def get_uv(centers, radii):
        u = np.zeros(n_circles)
        v = np.zeros(n_circles)
        for i in range(n_circles):
            denom = 1 - 2 * radii[i]
            if denom < 1e-9:
                u[i] = 0.5
                v[i] = 0.5
            else:
                u[i] = (centers[i, 0] - radii[i]) / denom
                v[i] = (centers[i, 1] - radii[i]) / denom
        return u, v

    def get_centers(radii, u, v):
        centers = np.zeros((n_circles, 2))
        for i in range(n_circles):
            denom = 1 - 2 * radii[i]
            centers[i, 0] = radii[i] + denom * u[i]
            centers[i, 1] = radii[i] + denom * v[i]
        return centers

    u_curr, v_curr = get_uv(centers, radii)
    
    # Clip radii and u,v to valid ranges
    radii = np.clip(radii, 0, 0.49)
    u_curr = np.clip(u_curr, 0, 1)
    v_curr = np.clip(v_curr, 0, 1)
    
    x0 = np.concatenate([radii, u_curr, v_curr])
    
    bounds = []
    for _ in range(n_circles):
        bounds.append((0.0, 0.5)) # r
        bounds.append((0.0, 1.0)) # u
        bounds.append((0.0, 1.0)) # v
        
    def objective(vars):
        r = vars[:n_circles]
        u = vars[n_circles:2*n_circles]
        v = vars[2*n_circles:3*n_circles]
        
        centers = get_centers(r, u, v)

        # Penalty for overlaps
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                c1 = centers[i]
                c2 = centers[j]
                dx = c1[0] - c2[0]
                dy = c1[1] - c2[1]
                dist = np.sqrt(dx*dx + dy*dy)
                min_dist = r[i] + r[j]
                if dist < min_dist:
                    overlap = min_dist - dist
                    penalty += overlap**2

        # numerical errors might occur or if r>0.5)
        # Actually parameterization x = r + (1-2r)u guarantees x in [r, 1-r] if u in [0,1]
        # So boundary penalty not strictly needed if bounds on u,v are respected.
        # But optimizer might violate bounds? L-BFGS-B respects bounds.
        
        return -np.sum(r) + 10000.0 * penalty

    try:

        x_opt = res.x

    v_opt = x_opt[2*n_circles:3*n_circles]

    # Ensure radii are non-negative
    r_opt = np.maximum(r_opt, 0)
    
    # Re-calculate centers
    centers_opt = get_centers(r_opt, u_opt, v_opt)
    
    # Post-processing: shrink radii if overlaps exist (safety)
    # Check overlaps
    for _ in range(50):
        max_overlap = 0
        pair = (-1, -1)
        for i in range(n_circles):
            for j in range(i + 1, n_circles):

                sum_r = r_opt[i] + r_opt[j]
                if dist < sum_r:
                    overlap = sum_r - dist
                    if overlap > max_overlap:
                        max_overlap = overlap
                        pair = (i, j)

            reduction = max_overlap / 2
            r_opt[pair[0]] = max(0, r_opt[pair[0]] - reduction)
            r_opt[pair[1]] = max(0, r_opt[pair[1]] - reduction)
            centers_opt = get_centers(r_opt, u_opt, v_opt)
        else:
            break
            
    sum_radii = np.sum(r_opt)
    
    return centers_opt, r_opt, sum_radii

    """
    Convert centers and radii to parameterization (u, v).
    x = r + (1 - 2r) * u
    y = r + (1 - 2r) * v
    This ensures x, y stay within [r, 1-r] if u, v in [0, 1].
    """
    n = len(radii)
    u = np.zeros(n)
    v = np.zeros(n)
    for i in range(n):
        denom = 1 - 2 * radii[i]

            u[i] = (centers[i, 0] - radii[i]) / denom
