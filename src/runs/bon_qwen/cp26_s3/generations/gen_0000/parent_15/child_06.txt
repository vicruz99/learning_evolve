import numpy as np
import scipy.optimize

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square

    Args:
        centers: np.array of shape (n, 2) with (x, y) coordinates
        radii: np.array of shape (n) with radius of each circle

    Returns:
        True if valid, False otherwise
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True

def generate_hex_grid(n=26):
    """Generate an initial hexagonal packing layout for n circles."""
    centers = []
    r = 0.105  # Initial estimate for radius
    row_counts = [5, 5, 5, 5, 6]  # Distribution of 26 circles
    
    row_y = r
    for count in row_counts:
        row_width = 1.0 - 2 * r
        if count > 1:
            dx = row_width / (count - 1)
        else:
            dx = 0
            
        # Shift every other row
        shift = dx / 2 if len(centers) > 0 and (len(centers) // 5) % 2 != 0 else 0
        
        row_x = r - shift
        for _ in range(count):
            centers.append([row_x, row_y])
            row_x += dx
        row_y += r * np.sqrt(3)  # Vertical step for hex packing
            
    return np.array(centers)

def run_packing():

    best_radii = None

        r = vars[2::3]
        return -np.sum(r)

    # Constraints: 
    # 1. x >= r
    # 2. x <= 1 - r
    # 3. y >= r
    # 4. y <= 1 - r
    # 5. r >= 0 (handled by bounds or separate constraint if needed)
    # 6. dist_ij >= r_i + r_j
    
    def constraints(vars):
        n = len(vars) // 3
        c_x = vars[0::3]
        c_y = vars[1::3]
        r = vars[2::3]
        
        con = []
        # Boundary constraints
        con.append(c_x - r)       # x - r >= 0
        con.append(1 - c_x - r)   # x + r <= 1
        con.append(c_y - r)       # y - r >= 0
        con.append(1 - c_y - r)   # y + r <= 1
        
        # Overlap constraints
        for i in range(n):
            for j in range(i + 1, n):
                dx = c_x[i] - c_x[j]
                dy = c_y[i] - c_y[j]
                dist = np.sqrt(dx**2 + dy**2)
                con.append(dist - (r[i] + r[j]))
        
        return con

    # Run optimization from multiple starts
    initial_configs = []
    base_grid = generate_hex_grid(N)
