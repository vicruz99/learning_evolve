import numpy as np
from scipy.optimize import minimize, linprog

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    n = centers.shape[0]
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False
    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False
    return True

def get_optimal_radii_and_sum(centers):
    """
    Solves the LP to find optimal radii for fixed centers.
    Maximize sum(r) subject to r_i + r_j <= dist(i,j) and boundary constraints.
    Returns (radii, sum_radii).
    """
    n = centers.shape[0]
    
    # 1. Boundary constraints for radii
    # r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    # This simplifies to r_i <= min(x_i, 1-x_i, y_i, 1-y_i)
    ub_r = np.array([min(c[0], 1-c[0], c[1], 1-c[1]) for c in centers])
    
    # 2. Pairwise distance constraints
    # r_i + r_j <= dist_ij
    # We construct A_ub and b_ub for linprog
    # linprog minimizes c^T x. We want to maximize sum(r), so we minimize -sum(r).
    c_obj = -np.ones(n)
    
    # Inequalities: A_ub @ x <= b_ub
    # We need r_i + r_j <= dist_ij
    # There are n*(n-1)/2 pairs
    num_pairs = n * (n - 1) // 2
    A_ub = np.zeros((num_pairs, n))
    b_ub = np.zeros(num_pairs)
    
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            # Constraint: 1*r_i + 1*r_j <= dist
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = dist
            idx += 1
            
    # Bounds for radii: 0 <= r_i <= ub_r[i]
    bounds = [(0, ub_r[i]) for i in range(n)]
    
    # Solve LP
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        optimal_radii = res.x
        sum_radii = -res.fun
        return optimal_radii, sum_radii
    else:
        # Fallback if LP fails (should not happen for valid centers)
        return np.zeros(n), 0.0

def objective_function(center_coords):
    """
    Objective function for the optimizer.
    Takes flattened center coordinates, reshapes, solves LP, returns negative sum of radii.
    """
    n = 26
    centers = center_coords.reshape((n, 2))
    radii, sum_radii = get_optimal_radii_and_sum(centers)
    # We want to maximize sum_radii, so minimize -sum_radii
    return -sum_radii

    hex_coords = []

    row = 0
    count = 0
