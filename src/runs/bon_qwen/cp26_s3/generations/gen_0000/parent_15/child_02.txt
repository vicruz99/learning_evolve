import numpy as np
from scipy.optimize import minimize
import math

def generate_initial_configurations(n_circles=26, num_configs=5):
    """
    Generates multiple initial configurations using hexagonal lattice packing.
    Returns a list of (centers, radii) tuples.
    """
    configurations = []
    
    # Hexagonal packing parameters
    # We generate a grid of points and filter/select n_circles
    
    # Generate a dense hexagonal grid covering the square
    # Spacing dx = 2r, dy = sqrt(3)r. 
    # We don't know r yet, so we generate points assuming a small r 
    # or just a geometric grid and scale later.
    # Let's use a fixed geometric grid and then scale/fit.
    
    rows = 10
    cols = 10
    points = []
    
    # Generate points in a hexagonal pattern
    # Row 0: y=0, x=0, 2, 4...
    # Row 1: y=sqrt(3), x=1, 3, 5...
    for i in range(rows):
        for j in range(cols):
            x = j * 2.0
            y = i * math.sqrt(3)
            if i % 2 != 0:
                x += 1.0 # Shift for odd rows
            points.append([x, y])
            
    points = np.array(points)
    
    # We need to fit these into [0,1]x[0,1].
    # We will create multiple configurations by shifting/rotating or just selecting subsets
    # But a better way is to scale the grid to fit the square and pick n_circles.
    
    # Let's try a few scaling factors and offsets to create diversity
    np.random.seed(42)
    
    for k in range(num_configs):
        # Random offset to perturb the grid slightly
        offset = np.random.uniform(-0.1, 0.1, 2)
        
        # Scale factor to roughly fit n_circles. 
        # Area of square is 1. Area per circle approx pi * r^2.
        # Hexagonal density ~ 0.9069. 
        # 26 * pi * r^2 ~ 0.9 -> r ~ 0.105.
        # Grid spacing 2r ~ 0.21.
        # Let's scale our base grid (spacing 2) by factor s such that 2s ~ 0.21 -> s ~ 0.105.
        
        scale = 0.105 
        # Randomly vary scale slightly
        scale *= np.random.uniform(0.95, 1.05)
        
        pts = (points * scale) + offset
        
        # Filter points inside [0,1]x[0,1] with some margin for radius
        # Margin approx r. Let's say 0.15.
        mask = (pts[:, 0] >= 0.15) & (pts[:, 0] <= 0.85) & \
               (pts[:, 1] >= 0.15) & (pts[:, 1] <= 0.85)
        
        valid_pts = pts[mask]
        
        if len(valid_pts) < n_circles:
            # If not enough points, try a different scale or just take all and clamp
            # Fallback: generate random points if grid fails
            valid_pts = np.random.uniform(0.2, 0.8, (n_circles, 2))
        else:
            # Select n_circles points. To maximize density, pick those closest to center or random?
            # Random selection from valid set is fine for initialization.
            indices = np.random.choice(len(valid_pts), n_circles, replace=False)
            valid_pts = valid_pts[indices]
            
        centers = valid_pts
        radii = np.full(n_circles, 0.05) # Start with small radius
        configurations.append((centers, radii))
        
    return configurations

    best_sum = -1.0
    best_centers = None
    best_radii = None

    configs = generate_initial_configurations(n_circles, num_configs=10)
    
    def objective(variables):
        # variables is 1D array of shape (3 * n_circles)
        # Order: x0, y0, r0, x1, y1, r1, ...
        r = variables[2::3]
        return -np.sum(r)

    def boundary_constraints(variables):
        # r_i <= x_i <= 1 - r_i  => x_i - r_i >= 0, 1 - x_i - r_i >= 0
        # Same for y
        cons = []
        for i in range(n_circles):
            idx = 3 * i
            x = variables[idx]
            y = variables[idx+1]
            r = variables[idx+2]
            
            cons.append(x - r)           # x >= r
            cons.append(1.0 - x - r)     # 1 - x >= r
            cons.append(y - r)           # y >= r
            cons.append(1.0 - y - r)     # 1 - y >= r
            cons.append(r)               # r >= 0
        return cons

    def non_overlap_constraints(variables):
        cons = []
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                idx_i = 3 * i
                idx_j = 3 * j
                
                xi, yi, ri = variables[idx_i], variables[idx_i+1], variables[idx_i+2]
                xj, yj, rj = variables[idx_j], variables[idx_j+1], variables[idx_j+2]
                
                # Distance squared >= (ri + rj)^2
                dist_sq = (xi - xj)**2 + (yi - yj)**2
                sum_r_sq = (ri + rj)**2
                
                cons.append(dist_sq - sum_r_sq)
        return cons

    # Combine constraints
    # scipy.optimize.minimize expects a list of constraint dictionaries
    # or a single function returning array for 'ineq' type?
    # 'SLSQP' supports list of dicts or a single dict with fun returning array.
    
    # Using a single function returning array for efficiency
    def constraints_fun(variables):
        b_cons = boundary_constraints(variables)
        o_cons = non_overlap_constraints(variables)
        return np.array(b_cons + o_cons)

    constraint = {'type': 'ineq', 'fun': constraints_fun}

    bounds = []

            variables[3*i+2] = radii[i]

                           bounds=bounds, constraints=constraint,
