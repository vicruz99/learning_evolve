import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    Returns (centers, radii, sum_radii)
    """
    n = 26
    best_sum_r = 0.0
    best_centers = np.zeros((n, 2))
    best_radii = np.zeros(n)

    def generate_hex_init(r_guess=0.1):
        # Generate a grid of points in a hexagonal lattice
        # Spacing 2*r_guess
        pts = []
        y_step = r_guess * np.sqrt(3)
        x_step = 2 * r_guess
        
        y = 0
        while y + r_guess < 1.0:
            x = 0
            shift = (int(y / y_step) % 2) * r_guess
            while x + r_guess < 1.0:
                pts.append((x + shift + r_guess, y + r_guess))
                x += x_step
            y += y_step
        return pts

    for _ in range(10):
        # 1. Initialization
        pts = generate_hex_init(0.1)
        if len(pts) < n:
            pts += [(np.random.rand(), np.random.rand()) for _ in range(n - len(pts))]
        np.random.shuffle(pts)
        init_centers = np.array(pts[:n])

        # 2. Variable vector: [x1, y1, r1, x2, y2, r2, ...]
        x0 = np.zeros(n * 3)
        for i in range(n):
            x0[3*i] = init_centers[i, 0]
            x0[3*i+1] = init_centers[i, 1]
            x0[3*i+2] = 0.1 # Initial radius guess

        # 3. Objective: Maximize sum of radii -> Minimize negative sum
        def objective(x_vars):
            return -sum(x_vars[3*i+2] for i in range(n))

        # 4. Constraints
        constraints = []

        # Boundary constraints: 0 <= x-r, x+r <= 1, 0 <= y-r, y+r <= 1
        for i in range(n):
            x_idx, y_idx, r_idx = 3*i, 3*i+1, 3*i+2
            constraints.append({'type': 'ineq', 'fun': lambda v, i=i: v[x_idx] - v[r_idx]})
            constraints.append({'type': 'ineq', 'fun': lambda v, i=i: 1.0 - v[x_idx] - v[r_idx]})
            constraints.append({'type': 'ineq', 'fun': lambda v, i=i: v[y_idx] - v[r_idx]})
            constraints.append({'type': 'ineq', 'fun': lambda v, i=i: 1.0 - v[y_idx] - v[r_idx]})

        # Non-overlap constraints: dist >= r1 + r2
        for i in range(n):
            for j in range(i + 1, n):
                idx_i, idx_j = 3*i, 3*j
                def dist_constraint(v, i=i, j=j):
                    xi, yi, ri = v[idx_i], v[idx_i+1], v[idx_i+2]
                    xj, yj, rj = v[idx_j], v[idx_j+1], v[idx_j+2]
                    d = np.sqrt((xi - xj)**2 + (yi - yj)**2)
                    return d - (ri + rj)
                constraints.append({'type': 'ineq', 'fun': dist_constraint})

        # 5. Bounds
        bounds = [(0, 1)] * n + [(0, 1)] * n + [(0, 0.5)] * n

        # 6. Optimization
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints,
                           options={'maxiter': 500, 'ftol': 1e-9})
            
            if res.success or res.fun < -best_sum_r:
                current_sum = -res.fun
                if current_sum > best_sum_r:
                    best_sum_r = current_sum
                    for i in range(n):
                        best_centers[i] = [res.x[3*i], res.x[3*i+1]]
                        best_radii[i] = res.x[3*i+2]
        except Exception:
            continue

    return best_centers, best_radii, best_sum_r
