import numpy as np
from scipy.optimize import minimize, differential_evolution
import scipy

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26

    # 1. Initial Hexagonal Grid Configuration
    # We arrange circles in a staggered pattern to utilize space efficiently.
    # We select row counts that sum to 26. 
    # Pattern: 5, 5, 5, 5, 5, 1 (Total 26)
    # This creates a dense block with one extra circle.
    
    row_counts = [5, 5, 5, 5, 5, 1]
    centers_init = []
    y_pos = 0.1 # Initial guess for y
    spacing_y = 0.15 # Initial guess for vertical spacing
    spacing_x = 0.2 # Initial guess for horizontal spacing
    
    for row_idx, count in enumerate(row_counts):
        shift = (spacing_x / 2) if row_idx % 2 == 1 else 0
        x_start = 0.1 + shift
        # Adjust spacing to fit 'count' circles in width 1
        current_spacing_x = (1.0 - 2 * 0.1) / count if count > 1 else 1.0
        for col_idx in range(count):
            x = x_start + col_idx * current_spacing_x
            centers_init.append([x, y_pos])
        
        if count > 0:
            y_pos += spacing_y * 1.2 # Slightly spaced out initially

    # Convert to numpy array
    centers_init = np.array(centers_init)
    radii_init = np.full(n, 0.05) # Small initial radius to ensure validity

    # Ensure at least 26 points if logic failed
    while centers_init.shape[0] < n:
        centers_init = np.vstack([centers_init, [0.5, 0.5]])

    # 2. Global Optimization (Differential Evolution)
    # We optimize positions and radii. 
    # Parameter vector: [x1, y1, r1, x2, y2, r2, ...]
    
    def objective(p):
        centers = p[:2*n].reshape(-1, 2)
        radii = p[2*n:]
        score = -np.sum(radii)
        
        penalty = 0.0
        
        # Boundary penalty
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if r < 0:
                penalty += 1000 * (1 + abs(r))
            
            # Soft constraints for boundaries
            margin = 1e-4
            if x - r < margin: penalty += 100 * (margin - (x - r))**2
            if x + r > 1 - margin: penalty += 100 * ((x + r) - (1 - margin))**2
            if y - r < margin: penalty += 100 * (margin - (y - r))**2
            if y + r > 1 - margin: penalty += 100 * ((y + r) - (1 - margin))**2
            
        # Overlap penalty
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                sum_r = radii[i] + radii[j]
                if dist < sum_r:
                    penalty += 1000 * (sum_r - dist)**2
                    
        return score + penalty

    # Bounds for DE: x, y in [0, 1], r in [0, 0.5]
    bounds = [(0, 1)] * 2*n + [(0, 0.5)] * n

    # Note: DE can be slow, so we limit iterations or use a seed.
    # For 78 variables, 1000 iterations might be heavy, but necessary for global search.
    # We use a smaller population to keep it fast.
    
    try:
        res_de = differential_evolution(objective, bounds, popsize=20, maxiter=500, 
                                        seed=42, tol=1e-6, workers=1, polishing=False)
        best_params = res_de.x
    except Exception:
        # Fallback if DE fails
        best_params = np.concatenate([centers_init.flatten(), radii_init])

    centers_global = best_params[:2*n].reshape(-1, 2)
    radii_global = best_params[2*n:]

    # Use the global solution as a starting point for a precise local optimizer.
    # We define explicit constraints.
    
    def obj_local(p):
        return -np.sum(p[2*n:])

    def con_bounds(p):
        # Returns array of violations (should be >= 0)
        # x >= r, x <= 1-r => r - x <= 0, x + r - 1 <= 0
        # Actually easier: r <= x, r <= 1-x, r <= y, r <= 1-y
        # Constraints: x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
        centers = p[:2*n].reshape(-1, 2)
        radii = p[2*n:]
        res = []
        for i in range(n):
            res.append(centers[i, 0] - radii[i])
            res.append(1 - centers[i, 0] - radii[i])
            res.append(centers[i, 1] - radii[i])
            res.append(1 - centers[i, 1] - radii[i])
        return np.array(res)

    def con_overlap(p):
        centers = p[:2*n].reshape(-1, 2)
        radii = p[2*n:]
        res = []
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                res.append(dist - (radii[i] + radii[j]))
        return np.array(res)

    # Define constraints dict
    constraints = [
        {'type': 'ineq', 'fun': con_bounds},
        {'type': 'ineq', 'fun': con_overlap}
    ]
    
    # Bounds for local opt: x,y in [0,1], r in [0, 0.5]
    local_bounds = [(0, 1)] * 2*n + [(1e-6, 0.5)] * n
    
    # Run local optimization
    # Try a few perturbations to escape local minima
    best_sol = None
    best_score = -np.inf
    
    for _ in range(5): # 5 restarts
        # Perturb initial guess slightly
        p0 = best_params.copy()
        p0 += np.random.normal(0, 0.001, p0.shape)
        
        res = minimize(obj_local, p0, method='SLSQP', bounds=local_bounds, 
                       constraints=constraints, options={'maxiter': 2000, 'ftol': 1e-12})
        
        if res.success and res.fun < best_score: # fun is negative sum
             # Validate strictly
             centers_cand = res.x[:2*n].reshape(-1, 2)
             radii_cand = res.x[2*n:]

                             valid = False
                             break
                     if not valid: break
             
             if valid:
                 current_sum = np.sum(radii_cand)
                 if current_sum > best_score:
                     best_score = current_sum
                     best_sol = (centers_cand, radii_cand)

    if best_sol is None:
        # Fallback to the global result if local failed
        centers_out = centers_global
        radii_out = radii_global
    else:
        centers_out, radii_out = best_sol

    sum_radii = np.sum(radii_out)
    return centers_out, radii_out, sum_radii
