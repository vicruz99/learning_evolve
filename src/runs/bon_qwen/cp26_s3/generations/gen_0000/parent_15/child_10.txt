import numpy as np
from scipy.optimize import linprog, minimize

def get_sum_radii(centers):
    """
    Solves the LP to find the optimal radii for fixed centers.
    """
    n = centers.shape[0]
    if n == 0:
        return 0.0, np.array([])

    # LP variables: radii r_0 ... r_{n-1}
    # Objective: maximize sum(r) -> minimize -sum(r)
    c_obj = -np.ones(n)

    # Bounds for radii: [0, distance_to_nearest_wall]
    r_bounds = []
    for i in range(n):
        x, y = centers[i]
        max_r = min(x, 1.0 - x, y, 1.0 - y)
        if max_r < 0: max_r = 0
        r_bounds.append((0, max_r))

    # Inequality constraints: r_i + r_j <= dist(i, j)
    # We only need to add constraints for pairs that are close enough to be tight.
    # If dist > 1.0, the constraint is redundant because r_i + r_j <= 0.5 + 0.5 = 1.0
    A_ub_rows = []
    b_ub_list = []
    
    # Precompute pairwise distances for efficient constraint building
    # Using broadcasting for speed
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.linalg.norm(diff, axis=2)
    
    # Upper triangle indices
    i_indices, j_indices = np.triu_indices(n, k=1)
    
    for idx in range(len(i_indices)):
        i, j = i_indices[idx], j_indices[idx]
        d = dists[i, j]
        if d < 1.0: # Optimization: ignore redundant constraints
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub_rows.append(row)
            b_ub_list.append(d)

    if A_ub_rows:
        A_ub = np.array(A_ub_rows)
        b_ub = np.array(b_ub_list)
    else:
        A_ub = None
        b_ub = None

    # Solve LP
    # method='highs' is generally faster and more robust
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=r_bounds, method='highs')

    if res.success:
        # res.fun is the minimized value (-sum), so sum is -res.fun
        return -res.fun, res.x
    else:
        # Fallback if LP fails (should not happen with valid inputs)
        return 0.0, np.zeros(n)

def run_packing():
    n = 26
    
    # Initialization: Hexagonal lattice pattern
    centers = np.zeros((n, 2))
    count = 0
    
    # Estimate initial radius to set spacing
    # Packing density heuristic
    est_r = 0.12
    spacing_x = 2 * est_r
    spacing_y = np.sqrt(3) * est_r
    
    y = est_r # Start slightly inside boundary
    row = 0
    while count < n:
        x = est_r
        # Shift every other row for hexagonal packing
        if row % 2 == 1:
            x += spacing_x / 2
        
        while x <= 1.0 - est_r and count < n:
            centers[count] = [x, y]
            count += 1
            x += spacing_x
        
        y += spacing_y
        row += 1
    
    # If we didn't fill enough (unlikely with these params), pad with random
    if count < n:
        for i in range(count, n):
            centers[i] = [np.random.uniform(0.1, 0.9), np.random.uniform(0.1, 0.9)]

    # Optimization
    # We want to maximize sum of radii, so we minimize negative sum
    def objective(pos):
        c = pos.reshape((n, 2))
        val, _ = get_sum_radii(c)
        return -val

    bounds = [(0, 1)] * (2 * n)
    
    # Use Powell method for derivative-free optimization with bounds
    # Multiple restarts can help, but one good run with Powell is usually effective
    # for this dimensionality if initialized well.
    
    x0 = centers.flatten()
    
    # Run optimization
    # ftol is tolerance for function value change
    res = minimize(objective, x0, method='Powell', bounds=bounds, 
                   options={'maxiter': 2000, 'ftol': 1e-9, 'xtol': 1e-9})
    
    final_centers = res.x.reshape((n, 2))
    
    # Compute final radii and sum
    sum_radii, radii = get_sum_radii(final_centers)
    
    return final_centers, radii, sum_radii
