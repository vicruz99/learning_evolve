from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26

    # We pack rows with vertical spacing sqrt(3)/2 * diameter = r*sqrt(3)
    # and horizontal spacing 2r.

    r_init = 0.10
    centers = []
    r = r_init
    h = r * np.sqrt(3)

    y = r

        x = r
        if row_idx % 2 == 1:
            x += r # Shift by r for hexagonal offset
        
        while x + r <= 1.0 and len(centers) < n:
            centers.append([x, y])
            x += 2 * r
        
        y += h
        row_idx += 1
            
    centers = np.array(centers[:n])
    radii = np.full(n, r_init)
    
    # --- 2. Optimization Setup ---
    def objective(params):
        # We minimize the negative sum of radii to maximize it
        r = params[n:]
        return -np.sum(r)

        # Constraints: x >= r, x <= 1-r, y >= r, y <= 1-r

        c4 = 1.0 - xy[:, 1] - r
        return np.concatenate([c1, c2, c3, c4])

        {'type': 'ineq', 'fun': constraint_overlap}
    ]
