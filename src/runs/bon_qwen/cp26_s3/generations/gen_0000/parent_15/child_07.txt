# 5 rows, varying columns, hexagonal-ish shift

    [0.0, 0.0], [0.2, 0.0], [0.4, 0.0], [0.6, 0.0], [0.8, 0.0], [1.0, 0.0],

    [0.0, 1.0], [0.2, 1.0], [0.4, 1.0], [0.6, 1.0]
])

def energy_function(pos, r, n):
    """
    Calculates the sum of squared overlaps for circles of radius r.
    pos: flattened array of centers (x1, y1, x2, y2, ...)
    r: radius
    n: number of circles
    """
    pts = pos.reshape(n, 2)
    # Distance matrix
    diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
    dists = np.linalg.norm(diff, axis=2)
    
    # Lower triangle mask (excluding diagonal)
    mask = np.tril(np.ones((n, n), dtype=bool), -1)
    d_lower = dists[mask]
    
    # Overlap penalty
    overlaps = np.maximum(0.0, 2.0 * r - d_lower)
    return np.sum(overlaps**2)

def check_r(r, n, template):
    """
    Checks if n circles of radius r can fit in the unit square.
    Returns (True, centers) if feasible, else (False, None).
    """
    low = r

        return False, None

# Global template for 26 circles
# Arranged in 5 rows with a hexagonal-like shift to promote dense packing.
# Row counts: 6, 5, 6, 5, 4 = 26 circles.
# Coordinates are normalized to [0, 1] x [0, 1].
TEMPLATE_POINTS = np.array([
    [0.0, 0.0], [0.2, 0.0], [0.4, 0.0], [0.6, 0.0], [0.8, 0.0], [1.0, 0.0],
    [0.1, 0.25], [0.3, 0.25], [0.5, 0.25], [0.7, 0.25], [0.9, 0.25],
    [0.0, 0.5], [0.2, 0.5], [0.4, 0.5], [0.6, 0.5], [0.8, 0.5], [1.0, 0.5],
    [0.1, 0.75], [0.3, 0.75], [0.5, 0.75], [0.7, 0.75], [0.9, 0.75],
    [0.0, 1.0], [0.2, 1.0], [0.4, 1.0], [0.6, 1.0]
])

def energy_function(pos, r, n):
    """
    Calculates the sum of squared overlaps for n circles of radius r.
    pos: flattened array of centers (x1, y1, x2, y2, ...) of length 2*n.
    r: radius of circles.
    n: number of circles.
    """
    pts = pos.reshape(n, 2)
    
    # Compute pairwise distance matrix
    # diff shape: (n, n, 2)
    diff = pts[:, np.newaxis, :] - pts[np.newaxis, :, :]
    dists = np.linalg.norm(diff, axis=2)
    
    # Extract lower triangle distances (excluding diagonal)
    mask = np.tril(np.ones((n, n), dtype=bool), -1)
    d_lower = dists[mask]
    
    # Calculate overlap: max(0, 2r - distance)
    # If distance < 2r, there is overlap.
    overlaps = np.maximum(0.0, 2.0 * r - d_lower)
    
    # Sum of squared overlaps
    return np.sum(overlaps**2)

def check_r(r, n, template):
    """
    Attempts to pack n circles of radius r in the unit square.
    Returns (True, centers) if a valid configuration is found, else (False, None).
    """
    low = r
    high = 1.0 - r
    
    # If radius is too large to fit even a single circle diameter
    if low > high:
        return False, None
    
    # Bounds for each coordinate (x, y) must be within [r, 1-r]
    bounds = [(low, high)] * (2 * n)
    
    # Initialize positions using the template scaled to the feasible region
    # Template points are in [0, 1]. Map to [low, high].
    init_pts = low + (high - low) * template
    
    # Create candidate initializations
    candidates = []
    
    # 1. Template with slight random noise to break symmetry

    candidates.append(init_pts_noisy.flatten())
    
    # 2. Fully random initialization
    rand_pts = np.random.uniform(low, high, size=(n, 2))
    candidates.append(rand_pts.flatten())
    
    best_energy = float('inf')
    best_pos = None
    
    # Try to minimize energy for each candidate
    for x0 in candidates:
        try:
            res = scipy.optimize.minimize(
                energy_function, 
                x0, 
                args=(r, n), 
                method='L-BFGS-B', 
                bounds=bounds,
                options={'maxiter': 2000, 'ftol': 1e-15}

            if res.fun < best_energy:

            pass
