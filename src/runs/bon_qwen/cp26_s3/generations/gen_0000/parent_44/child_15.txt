def get_optimal_radii_for_centers(centers):
    """
    Given fixed centers, solve the LP to find radii that maximize sum of radii.
    Constraints:
    1. r_i + r_j <= distance(c_i, c_j) for all i < j
    2. r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    3. r_i >= 0
    Objective: Maximize sum(r_i)
    """
    n = centers.shape[0]
    
    # Variables: r_0, r_1, ..., r_{n-1}
    # Objective: max sum(r) => min -sum(r)
    c_obj = -np.ones(n)
    
    # Inequality constraints: A_ub * r <= b_ub
    
    # We will build A_ub and b_ub
    # Constraints are of form r_i + r_j <= d_ij
    # and r_i <= boundary_limit
    
    num_constraints = 0
    # Pre-allocate roughly
    # n boundary constraints + n*(n-1)/2 pairwise constraints
    num_constraints = n + n * (n - 1) // 2
    A_ub = np.zeros((num_constraints, n))
    b_ub = np.zeros(num_constraints)
    
    idx = 0
    
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        x, y = centers[i]
        limit = min(x, 1 - x, y, 1 - y)
        # r_i <= limit  => 1*r_i <= limit
        A_ub[idx, i] = 1.0
        b_ub[idx] = limit
        idx += 1
        
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    # Calculate distances
    # pdist returns condensed distance matrix
    dists = pdist(centers, metric='euclidean')
    # squareform converts to dense matrix for easier indexing
    dist_matrix = squareform(dists)
    
    for i in range(n):
        for j in range(i + 1, n):
            d_ij = dist_matrix[i, j]
            # r_i + r_j <= d_ij
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = d_ij
            idx += 1
            
    # Bounds for r_i: 0 <= r_i
    bounds = [(0, None) for _ in range(n)]
    
    # Solve LP
    # method='highs' is usually the best/fastest in recent scipy
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return -res.fun, res.x # Return sum and radii
        else:
            # Fallback or error handling
            return 0, np.zeros(n)
    except Exception:
        return 0, np.zeros(n)
