import numpy as np
from scipy.optimize import minimize

def make_initial_config(n, row_counts):
    """Create an initial hexagonal-like configuration."""
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.02
    idx = 0
    for row in range(len(row_counts)):
        count = row_counts[row]
        offset = 0.08 if row % 2 == 1 else 0.0
        for col in range(count):
            centers[idx, 0] = 0.06 + offset + col * 0.14
            centers[idx, 1] = 0.06 + row * 0.14
            idx += 1
    return centers, radii


def iterative_growth(centers, radii, n, max_iterations=80000):
    """Grow circles iteratively and resolve overlaps."""
    centers = centers.copy()
    radii = radii.copy()
    
    for iteration in range(max_iterations):
        # Adaptive growth rate
        base_growth = 3e-4
        if iteration > 40000:
            base_growth = 1e-4
        if iteration > 60000:
            base_growth = 5e-5
        
        radii += base_growth
        
        # Resolve overlaps with multiple passes
        passes = 200
        if iteration > 60000:
            passes = 300
        
        for _ in range(passes):
            # Circle-circle overlap resolution
            for i in range(n):
                for j in range(i + 1, n):
                    dx = centers[j, 0] - centers[i, 0]
                    dy = centers[j, 1] - centers[i, 1]
                    dist = np.sqrt(dx * dx + dy * dy)
                    min_dist = radii[i] + radii[j]
                    
                    if dist < min_dist:
                        if dist < 1e-12:
                            # Nearly coincident: random perturbation
                            eps = 0.002
                            centers[i, 0] -= eps
                            centers[i, 1] += eps * 0.5
                            centers[j, 0] += eps
                            centers[j, 1] -= eps * 0.5
                        else:
                            overlap = min_dist - dist
                            nx = dx / dist
                            ny = dy / dist
                            # Push apart, weighted by radii
                            total_r = radii[i] + radii[j]
                            wi = radii[j] / total_r if total_r > 0 else 0.5
                            wj = radii[i] / total_r if total_r > 0 else 0.5
                            centers[i, 0] -= nx * overlap * wi
                            centers[i, 1] -= ny * overlap * wi
                            centers[j, 0] += nx * overlap * wj
                            centers[j, 1] += ny * overlap * wj
            
            # Boundary constraint enforcement
            for i in range(n):
                r = radii[i]
                centers[i, 0] = max(r, min(1 - r, centers[i, 0]))
                centers[i, 1] = max(r, min(1 - r, centers[i, 1]))
        
        # Check convergence: if all circles are touching constraints, reduce radii slightly and stop
        all_touching = True
        for i in range(n):
            touching = False
            # Check boundary
            if abs(centers[i, 0] - radii[i]) < 0.001:
                touching = True
            if abs(centers[i, 0] - (1 - radii[i])) < 0.001:
                touching = True
            if abs(centers[i, 1] - radii[i]) < 0.001:
                touching = True
            if abs(centers[i, 1] - (1 - radii[i])) < 0.001:
                touching = True
            # Check neighbors
            for j in range(n):
                if i != j:
                    dist = np.sqrt((centers[i, 0] - centers[j, 0]) ** 2 + 
                                   (centers[i, 1] - centers[j, 1]) ** 2)
                    if dist < radii[i] + radii[j] + 0.001:
                        touching = True
                        break
            if not touching:
                all_touching = False
                break
        
        if all_touching and iteration > 10000:
            # Reached equilibrium, shrink slightly to ensure validity
            radii -= base_growth * 5
            break
    
    # Final shrink to ensure strict validity
    radii -= 1e-4
    
    return centers, radii


def local_optimization(centers, radii, n):
    """Fine-tune with scipy optimization."""
    x0 = np.concatenate([radii, centers.flatten()])
    
    def penalty_objective(x):
        r = x[:n]
        c = x[n:].reshape((n, 2))
        
        obj = -np.sum(r)
        penalty = 0.0
        
        # Boundary penalties
        for i in range(n):
            if r[i] > c[i, 0]:
                penalty += (r[i] - c[i, 0]) ** 2
            if r[i] + c[i, 0] > 1:
                penalty += (r[i] + c[i, 0] - 1) ** 2
            if r[i] > c[i, 1]:
                penalty += (r[i] - c[i, 1]) ** 2
            if r[i] + c[i, 1] > 1:
                penalty += (r[i] + c[i, 1] - 1) ** 2
        
        # Overlap penalties
        for i in range(n):
            for j in range(i + 1, n):
                dx = c[i, 0] - c[j, 0]
                dy = c[i, 1] - c[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                ov = r[i] + r[j] - dist
                if ov > 0:
                    penalty += ov ** 2
        
        # Negative radius penalty
        for i in range(n):
            if r[i] < 0:
                penalty += r[i] ** 2 * 1000
        
        return obj + 1e6 * penalty
    
    bounds = []
    for i in range(n):
        bounds.append((0, 0.5))
    for i in range(2 * n):
        bounds.append((0, 1))
    
    try:
        result = minimize(penalty_objective, x0, method='L-BFGS-B', bounds=bounds,
                         options={'maxiter': 30000, 'ftol': 1e-12, 'gtol': 1e-10})
        
        if result.success or result.nit > 0:
            r = result.x[:n]
            c = result.x[n:].reshape((n, 2))
            
            # Validate and fix
            for _ in range(500):
                changed = False
                for i in range(n):
                    r[i] = min(r[i], c[i, 0], 1 - c[i, 0], c[i, 1], 1 - c[i, 1])
                    r[i] = max(r[i], 0)
                    c[i, 0] = max(r[i], min(1 - r[i], c[i, 0]))
                    c[i, 1] = max(r[i], min(1 - r[i], c[i, 1]))
                
                # Check overlaps and shrink if needed
                for i in range(n):
                    for j in range(i + 1, n):
                        dist = np.sqrt((c[i, 0] - c[j, 0]) ** 2 + (c[i, 1] - c[j, 1]) ** 2)
                        if dist < r[i] + r[j] - 1e-10:
                            shrink = (r[i] + r[j] - dist) * 0.5 + 1e-6
                            r[i] -= shrink
                            r[j] -= shrink
                            changed = True
                            r[i] = max(r[i], 0)
                            r[j] = max(r[j], 0)
                
                if not changed:
                    break
            
            return c, r
    except Exception:
        pass
    
    return centers, radii
