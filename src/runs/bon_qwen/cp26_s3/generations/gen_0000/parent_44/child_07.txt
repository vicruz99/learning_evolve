import numpy as np
from scipy.optimize import linprog, differential_evolution

def get_optimal_radii(centers):
    """
    Solves the LP to maximize sum of radii given circle centers.
    """
    n = len(centers)
    
    # Prepare coefficients for objective: max sum(r_i) => min -sum(r_i)
    c = -np.ones(n)
    
    # Constraints: r_i + r_j <= dist(i, j)
    # And r_i <= boundary_distance(i)
    # Also r_i >= 0 (handled by bounds)
    
    A_ub = []
    b_ub = []
    
    # Pairwise distance constraints
    for i in range(n):
        for j in range(i + 1, n):
            # Distance between center i and j
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.hypot(dx, dy)
            
            # Constraint: r_i + r_j <= dist
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dist)
            
    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        # r_i <= x, r_i <= 1-x, r_i <= y, r_i <= 1-y
        boundary_lim = min(x, 1 - x, y, 1 - y)
        
        row = np.zeros(n)
        row[i] = 1.0
        A_ub.append(row)
        b_ub.append(boundary_lim)
        
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    # Bounds for variables: r_i >= 0. Upper bound can be 0.5 (max possible)
    bounds = [(0, 0.5) for _ in range(n)]
    
    # Solve LP
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return -res.fun, res.x
    else:
        # Fallback if LP fails, though unlikely with valid bounds
        return 0.0, np.zeros(n)
