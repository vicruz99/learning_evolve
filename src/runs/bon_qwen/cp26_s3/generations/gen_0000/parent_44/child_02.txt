from scipy.spatial.distance import cdist

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    best_centers = np.zeros((n, 2))
    best_radii = np.zeros(n)
    best_sum = -1.0

    for restart in range(15):
        # 1. Initialization: Grid + perturbation

        side = int(np.ceil(np.sqrt(n)))
        pad = 0.12
        x_grid = np.linspace(pad, 1 - pad, side)
        y_grid = np.linspace(pad, 1 - pad, side)
        for i in range(n):
            r_idx = i // side
            c_idx = i % side
            centers[i] = [x_grid[c_idx], y_grid[r_idx]]
        
        centers += np.random.uniform(-0.02, 0.02, (n, 2))
        centers = np.clip(centers, 0.05, 0.95)
        radii = np.full(n, 0.03)

        # 2. Optimization dynamics
        lr_pos = 0.008
        lr_rad = 0.15
        noise = 0.008
        decay = 0.9985

        for step in range(3000):
            dists = cdist(centers, centers)
            np.fill_diagonal(dists, np.inf)

            forces = np.zeros_like(centers)
            margin = 0.05 * max(0.0, 1.0 - step / 3000)
            stiffness = 30.0

            # Compute repulsion forces
            for i in range(n):
                for j in range(i + 1, n):
                    d = dists[i, j]
                    limit = radii[i] + radii[j] + margin
                    if d < limit:
                        if d > 1e-8:
                            push = (limit - d) * stiffness
                            dir_vec = (centers[i] - centers[j]) / d
                            forces[i] += dir_vec * push
                            forces[j] -= dir_vec * push
                        else:
                            rand_dir = np.random.randn(2)
                            forces[i] += rand_dir
                            forces[j] -= rand_dir

            # Update positions
            centers += lr_pos * forces
            centers += np.random.randn(n, 2) * noise
            centers = np.clip(centers, 0.005, 0.995)

            # Update radii towards maximum possible
            for i in range(n):
                r_bound = min(centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])

                r_target = min(r_bound, r_circles)
                if r_target < 0.0:
                    r_target = 0.0
                radii[i] += lr_rad * (r_target - radii[i])

            lr_pos *= decay
            noise *= decay

        # 3. Strict projection to guarantee validity
        for _ in range(50):
            radii_copy = radii.copy()
            dists_final = cdist(centers, centers)
            np.fill_diagonal(dists_final, np.inf)

                    r_new = 0.0
                radii[i] = r_new

        # 4. Track best result
        current_sum = np.sum(radii)
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers.copy()
            best_radii = radii.copy()

    return best_centers, best_radii, best_sum
