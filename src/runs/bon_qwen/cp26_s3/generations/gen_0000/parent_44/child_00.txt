def get_optimal_radii(centers):
    """
    Given fixed centers, calculate the maximum possible radii using Linear Programming.
    Maximizes sum(r_i) subject to r_i + r_j <= dist(i, j) and boundary constraints.
    """
    n = centers.shape[0]
    
    # Variables: r_0, r_1, ..., r_{n-1}
    # Objective: minimize -sum(r_i)
    c = -np.ones(n)
    
    # Constraints: A_ub * r <= b_ub
    # 1. Non-negativity: r_i >= 0  =>  -r_i <= 0
    # 2. Boundary: r_i <= x_i, r_i <= 1-x_i, etc.
    # 3. Pairwise: r_i + r_j <= dist_ij
    
    # We will build A_ub and b_ub
    constraints = []
    
    # Boundary constraints
    # r_i <= x_i  => 1*r_i <= x_i
    # r_i <= 1-x_i => 1*r_i <= 1-x_i
    # r_i <= y_i
    # r_i <= 1-y_i
    for i in range(n):
        x, y = centers[i]
        # r_i <= x
        row = np.zeros(n)
        row[i] = 1.0
        constraints.append((row, x))
        # r_i <= 1-x
        row = np.zeros(n)
        row[i] = 1.0
        constraints.append((row, 1.0 - x))
        # r_i <= y
        row = np.zeros(n)
        row[i] = 1.0
        constraints.append((row, y))

        row = np.zeros(n)

            # Fallback if LP fails (shouldn't happen)
