import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26
DIM = 3 * N_CIRCLES  # x, y, r for each circle


def objective(x):
    """Maximize sum of radii = minimize negative sum."""
    return -np.sum(x[2::3])


def objective_jac(x):
    """Analytical gradient of objective."""
    jac = np.zeros(DIM)
    jac[2::3] = -1.0
    return jac


# ---- Constraint functions (analytical) ----

def overlap_con(x, i, j):
    dx = x[3 * i] - x[3 * j]
    dy = x[3 * i + 1] - x[3 * j + 1]
    dist = np.sqrt(dx * dx + dy * dy)
    return dist - x[3 * i + 2] - x[3 * j + 2]


def overlap_con_jac(x, i, j):
    jac = np.zeros(DIM)
    dx = x[3 * i] - x[3 * j]
    dy = x[3 * i + 1] - x[3 * j + 1]
    dist = np.sqrt(dx * dx + dy * dy)
    if dist > 1e-15:
        jac[3 * i] = dx / dist
        jac[3 * i + 1] = dy / dist
        jac[3 * j] = -dx / dist
        jac[3 * j + 1] = -dy / dist
    jac[3 * i + 2] = -1.0
    jac[3 * j + 2] = -1.0
    return jac


def left_con(x, i):
    return x[3 * i] - x[3 * i + 2]


def left_con_jac(x, i):
    jac = np.zeros(DIM)
    jac[3 * i] = 1.0
    jac[3 * i + 2] = -1.0
    return jac


def right_con(x, i):
    return 1.0 - x[3 * i] - x[3 * i + 2]


def right_con_jac(x, i):
    jac = np.zeros(DIM)
    jac[3 * i] = -1.0
    jac[3 * i + 2] = -1.0
    return jac


def bottom_con(x, i):
    return x[3 * i + 1] - x[3 * i + 2]


def bottom_con_jac(x, i):
    jac = np.zeros(DIM)
    jac[3 * i + 1] = 1.0
    jac[3 * i + 2] = -1.0
    return jac


def top_con(x, i):
    return 1.0 - x[3 * i + 1] - x[3 * i + 2]


def top_con_jac(x, i):
    jac = np.zeros(DIM)
    jac[3 * i + 1] = -1.0
    jac[3 * i + 2] = -1.0
    return jac


def make_bounds():
    """Bounds: x,y in [0,1], r in [0, 0.5]."""
    b = [(0.0, 1.0)] * (2 * N_CIRCLES) + [(1e-8, 0.5)] * N_CIRCLES
    return b


def build_constraints():
    """Build all constraint dicts for SLSQP."""
    cons = []
    # Overlap constraints
    for i in range(N_CIRCLES):
        for j in range(i + 1, N_CIRCLES):
            cons.append({
                'type': 'ineq',
                'fun': overlap_con,
                'jac': overlap_con_jac,
                'args': (i, j)
            })
    # Boundary constraints
    for i in range(N_CIRCLES):
        cons.append({'type': 'ineq', 'fun': left_con, 'jac': left_con_jac, 'args': (i,)})
        cons.append({'type': 'ineq', 'fun': right_con, 'jac': right_con_jac, 'args': (i,)})
        cons.append({'type': 'ineq', 'fun': bottom_con, 'jac': bottom_con_jac, 'args': (i,)})
        cons.append({'type': 'ineq', 'fun': top_con, 'jac': top_con_jac, 'args': (i,)})
    return cons


def flatten_init(centers, radii):
    """Flatten centers and radii into optimization vector."""
    x = np.zeros(DIM)
    for i in range(N_CIRCLES):
        x[3 * i] = centers[i, 0]
        x[3 * i + 1] = centers[i, 1]
        x[3 * i + 2] = radii[i]
    return x


def extract_solution(x):
    """Extract centers and radii from optimization vector."""
    centers = np.column_stack((x[0::3], x[1::3]))
    radii = x[2::3]
    return centers, radii


def make_hex_init(n, perturb=0.0, seed=0):
    """Create hexagonal lattice initialization with optional perturbation."""
    centers = np.zeros((n, 2))
    # Pattern: 5,5,5,5,5,1 = 26
    rows = [5, 5, 5, 5, 5, 1]
    n_rows = len(rows)
    
    x_base = 1.0 / (5 + 1.0)
    y_base = 1.0 / (n_rows + 1.0)
    
    idx = 0
    for row_idx, ncols in enumerate(rows):
        x_offset = x_base * 0.433 if row_idx % 2 == 1 else 0.0
        for col_idx in range(ncols):
            centers[idx, 0] = x_base * (col_idx + 1) + x_offset
            centers[idx, 1] = y_base * (row_idx + 1)
            idx += 1
    
    # Clip to valid range
    centers = np.clip(centers, 0.01, 0.99)
    
    if perturb > 0:
        rng = np.random.RandomState(seed)
        centers += rng.uniform(-perturb, perturb, centers.shape)
        centers = np.clip(centers, 0.02, 0.98)
    
    return centers


def make_grid_init(n):
    """Alternative: near-grid initialization."""
    centers = np.zeros((n, 2))
    # 5x5 grid = 25, plus 1 in a corner gap
    x_step = 1.0 / 6.0
    y_step = 1.0 / 6.0
    idx = 0
    for row in range(5):
        for col in range(5):
            centers[idx, 0] = x_step * (col + 1)
            centers[idx, 1] = y_step * (row + 1)
            idx += 1
    # 26th circle in upper-right area
    centers[25, 0] = x_step * 5.5
    centers[25, 1] = y_step * 5.5
    return centers


def make_init_variant(n, variant, seed=0):
    """Create various initialization patterns."""
    if variant == 0:
        return make_hex_init(n, perturb=0.0, seed=seed)
    elif variant == 1:
        return make_hex_init(n, perturb=0.02, seed=seed)
    elif variant == 2:
        return make_hex_init(n, perturb=0.04, seed=seed)
    elif variant == 3:
        return make_grid_init(n)
    elif variant == 4:
        # Different hex pattern: 5,5,5,5,4,2
        centers = np.zeros((n, 2))
        rows = [5, 5, 5, 5, 4, 2]
        n_rows = len(rows)
        x_base = 1.0 / 6.0
        y_base = 1.0 / (n_rows + 1.0)
        idx = 0
        for row_idx, ncols in enumerate(rows):
            x_offset = x_base * 0.433 if row_idx % 2 == 1 else 0.0
            for col_idx in range(ncols):
                centers[idx, 0] = x_base * (col_idx + 1) + x_offset
                centers[idx, 1] = y_base * (row_idx + 1)
                idx += 1
        centers = np.clip(centers, 0.01, 0.99)
        return centers
    elif variant == 5:
        # Another variant: start with known good packing
        centers = np.zeros((n, 2))
        r_approx = 0.10
        # Place in hexagonal rows
        row_configs = [(0, 5), (1, 5), (2, 5), (3, 5), (4, 5), (5, 1)]
        idx = 0
        for row_info in row_configs:
            row_idx = row_info[0]
            ncols = row_info[1]
            y = (row_idx + 0.5) * (1.0 / 6.0) * 1.1
            x_start = 0.5 * r_approx * 1.732 if row_idx % 2 == 1 else 0.5 * r_approx
            for col in range(ncols):
                x = x_start + col * 2.0 * r_approx * 1.0
                centers[idx] = [x, y]
                idx += 1
        # Normalize
        cx = (centers[:, 0] - centers[:, 0].min()) / (centers[:, 0].max() - centers[:, 0].min())
        cy = (centers[:, 1] - centers[:, 1].min()) / (centers[:, 1].max() - centers[:, 1].min())
        centers[:, 0] = cx * 0.8 + 0.1
        centers[:, 1] = cy * 0.8 + 0.1
        return centers
    else:
        return make_hex_init(n, perturb=0.03, seed=seed)


def run_optimization(x0, constraints, bounds, maxiter=2000):
    """Run SLSQP optimization."""
    result = minimize(
        objective,
        x0,
        jac=objective_jac,
        bounds=bounds,
        constraints=constraints,
        method='SLSQP',
        options={
            'maxiter': maxiter,
            'ftol': 1e-14,
            'disp': False
        }
    )
    return result


def validate_and_fix(x):
    """Validate solution and fix minor constraint violations."""
    centers = x[0::3].reshape(-1, 2)
    radii = x[2::3]
    
    # Ensure radii are positive
    radii = np.maximum(radii, 1e-10)
    
    # Push centers inside if needed
    for i in range(N_CIRCLES):
        r = radii[i]
        centers[i, 0] = np.clip(centers[i, 0], r, 1.0 - r)
        centers[i, 1] = np.clip(centers[i, 1], r, 1.0 - r)
    
    # Rebuild x vector
    x_fixed = np.zeros(DIM)
    for i in range(N_CIRCLES):
        x_fixed[3 * i] = centers[i, 0]
        x_fixed[3 * i + 1] = centers[i, 1]
        x_fixed[3 * i + 2] = radii[i]
    
    return x_fixed


def local_push_apart(centers, radii, n_steps=50):
    """Local search: push overlapping circles apart."""
    for step in range(n_steps):
        moved = False
        for i in range(N_CIRCLES):
            for j in range(i + 1, N_CIRCLES):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                min_dist = radii[i] + radii[j]
                if dist < min_dist and dist > 1e-12:
                    # Push apart
                    overlap = (min_dist - dist) * 0.5 * (1.0 + step * 0.01)
                    nx = dx / dist
                    ny = dy / dist
                    centers[i, 0] += nx * overlap
                    centers[i, 1] += ny * overlap
                    centers[j, 0] -= nx * overlap
                    centers[j, 1] -= ny * overlap
                    moved = True
        
        # Enforce boundaries
        for i in range(N_CIRCLES):
            r = radii[i]
            centers[i, 0] = max(r, min(1.0 - r, centers[i, 0]))
            centers[i, 1] = max(r, min(1.0 - r, centers[i, 1]))
        
        if not moved:
            break
    
    return centers, radii


def run_packing():
    """Main packing function. Returns (centers, radii, sum_radii)."""
    
    bounds = make_bounds()
    constraints = build_constraints()
    
    best_sum = -np.inf
    best_x = None
    
    # Try multiple initialization variants
    for variant in range(6):
        for seed in [0, 1, 42]:
            try:
                centers_init = make_init_variant(N_CIRCLES, variant, seed=seed)
                r_init = 0.065 + variant * 0.005
                radii_init = np.ones(N_CIRCLES) * r_init
                
                # Pre-optimize: push apart and inflate
                centers_pre, radii_pre = local_push_apart(centers_init.copy(), radii_init.copy(), 20)
                
                x0 = flatten_init(centers_pre, radii_pre)
                
                result = run_optimization(x0, constraints, bounds, maxiter=3000)
                
                if result.success or True:  # Accept even non-successful results
                    x_sol = validate_and_fix(result.x)
                    current_sum = -objective(x_sol)
                    
                    if current_sum > best_sum:
                        best_sum = current_sum
                        best_x = x_sol.copy()
                        
            except Exception:
                continue
    
    # Final refinement from best solution
    if best_x is not None:
        # Run additional refinement passes
        for refine in range(3):
            try:
                result = run_optimization(best_x.copy(), constraints, bounds, maxiter=5000)
                x_sol = validate_and_fix(result.x)
                current_sum = -objective(x_sol)
                if current_sum > best_sum:
                    best_sum = current_sum
                    best_x = x_sol.copy()
            except Exception:
                break
    
    # If all failed, fallback to simple hex
    if best_x is None:
        centers = make_hex_init(N_CIRCLES)
        radii = np.ones(N_CIRCLES) * 0.08
        best_x = flatten_init(centers, radii)
        best_sum = np.sum(radii)
    
    centers_opt, radii_opt = extract_solution(best_x)
    
    # Final cleanup: ensure validity
    radii_opt = np.maximum(radii_opt, 1e-10)
    for i in range(N_CIRCLES):
        r = radii_opt[i]
        centers_opt[i, 0] = np.clip(centers_opt[i, 0], r + 1e-12, 1.0 - r - 1e-12)
        centers_opt[i, 1] = np.clip(centers_opt[i, 1], r + 1e-12, 1.0 - r - 1e-12)
    
    final_sum = np.sum(radii_opt)
    
    return centers_opt, radii_opt, final_sum
