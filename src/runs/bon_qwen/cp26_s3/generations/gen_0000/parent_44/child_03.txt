import numpy as np

def compute_gradients(pos, rad):
    """
    Computes forces on positions and gradients on radii for the penalty function.
    Returns (penalty, forces_on_pos, grad_wrt_rad)
    """
    n = pos.shape[0]
    forces = np.zeros_like(pos)
    grad_r = np.zeros_like(rad)
    penalty = 0.0
    
    # Pairwise interaction gradients
    for i in range(n):
        for j in range(i + 1, n):
            dx = pos[i, 0] - pos[j, 0]
            dy = pos[i, 1] - pos[j, 1]
            d2 = dx*dx + dy*dy
            d = np.sqrt(d2) if d2 > 1e-12 else 1e-6
            r_sum = rad[i] + rad[j]
            
            if d < r_sum:
                gap = r_sum - d
                penalty += gap**2
                # Force direction: push apart
                f = -2.0 * gap / d
                fx, fy = f * dx, f * dy
                forces[i, 0] += fx
                forces[i, 1] += fy
                forces[j, 0] -= fx
                forces[j, 1] -= fy
                # Gradient w.r.t radii: penalty increases if radii increase
                grad_r[i] += 2.0 * gap
                grad_r[j] += 2.0 * gap
                
        # Boundary interaction gradients
        r = rad[i]
        # Left boundary: x < r => violation is r - x
        if pos[i, 0] < r:
            gap = r - pos[i, 0]
            penalty += gap**2
            forces[i, 0] += 2.0 * gap  # Push right
            grad_r[i] += 2.0 * gap     # Shrink radius
        # Right boundary: x + r > 1 => violation is x + r - 1
        if pos[i, 0] > 1.0 - r:
            gap = pos[i, 0] + r - 1.0
            penalty += gap**2
            forces[i, 0] -= 2.0 * gap  # Push left
            grad_r[i] += 2.0 * gap     # Shrink radius
        # Bottom boundary: y < r => violation is r - y
        if pos[i, 1] < r:
            gap = r - pos[i, 1]
            penalty += gap**2
            forces[i, 1] += 2.0 * gap  # Push up
            grad_r[i] += 2.0 * gap     # Shrink radius
        # Top boundary: y + r > 1 => violation is y + r - 1
        if pos[i, 1] > 1.0 - r:
            gap = pos[i, 1] + r - 1.0
            penalty += gap**2
            forces[i, 1] -= 2.0 * gap  # Push down
            grad_r[i] += 2.0 * gap     # Shrink radius
            
    return penalty, forces, grad_r

    pos *= scale

    np.clip(pos, 0.05, 0.95, out=pos)
    
    rad = np.full(n, 0.05)

    lam = 10.0
    max_iter = 6000
    
    for it in range(max_iter):
        penalty, forces, grad_r = compute_gradients(pos, rad)
        
        # Update positions to resolve overlaps/boundaries
        pos -= lr_pos * forces
        np.clip(pos, 1e-7, 1.0 - 1e-7, out=pos)

        # Objective gradient is -1. Constraint gradient is grad_r.

        rad -= lr_rad * grad_total
        np.clip(rad, 1e-7, None, out=rad)
        
        # Adapt penalty weight
        if penalty > 1e-5:
            lam *= 1.03
        else:
            lam *= 0.97

        lr_pos *= 0.9998
        lr_rad *= 0.9998

    margin = 1e-8
