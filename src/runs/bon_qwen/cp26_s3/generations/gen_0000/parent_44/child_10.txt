import numpy as np

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Pack 26 circles in a unit square to maximize the sum of radii.
    Uses a force-directed simulation with gradually expanding circles.
    """
    n = 26
    np.random.seed(42)
    
    # 1. Initial configuration: slightly perturbed grid
    x_grid = np.linspace(0.05, 0.95, 6)
    y_grid = np.linspace(0.05, 0.95, 6)
    xx, yy = np.meshgrid(x_grid, y_grid)
    points = np.vstack([xx.ravel(), yy.ravel()]).T
    
    centers = points[:n].copy()
    # Add small random perturbation to break symmetry
    centers += np.random.randn(n, 2) * 0.005
    centers = np.clip(centers, 0.01, 0.99)
    
    # 2. Simulation parameters
    r = 0.01
    v = np.zeros((n, 2))
    dt = 0.4
    damping = 0.85
    growth_rate = 2.8e-6
    num_steps = 70000
    
    k_wall = 150.0
    k_rep = 250.0
    
    best_r = r
    best_centers = centers.copy()
    
    # Pre-allocate arrays for performance
    F = np.zeros((n, 2))
    diff = np.zeros((n, n, 2))
    dists = np.zeros((n, n))
    overlaps = np.zeros((n, n))
    dirs = np.zeros((n, n, 2))
    
    for step in range(num_steps):
        # Gradually grow circles
        r += growth_rate
        
        # Compute relative positions
        diff[:] = centers[:, None] - centers[None, :]
        dists[:] = np.linalg.norm(diff, axis=2)
        np.fill_diagonal(dists, np.inf)
        
        # Reset forces
        F[:] = 0.0
        
        # Wall repulsion forces
        for i in range(n):
            # X boundaries
            if centers[i, 0] < r:
                F[i, 0] += k_wall * (r - centers[i, 0])
            elif centers[i, 0] > 1 - r:
                F[i, 0] -= k_wall * (centers[i, 0] - (1 - r))
            # Y boundaries
            if centers[i, 1] < r:
                F[i, 1] += k_wall * (r - centers[i, 1])
            elif centers[i, 1] > 1 - r:
                F[i, 1] -= k_wall * (centers[i, 1] - (1 - r))
                
        # Inter-circle repulsion
        overlaps[:] = np.maximum(2 * r - dists, 0.0)
        norms = dists.copy()
        norms[norms < 1e-10] = 1e-10
        dirs[:] = diff / norms[:, :, np.newaxis]
        
        # Vectorized force accumulation
        F += np.sum(dirs * overlaps[:, :, np.newaxis] * k_rep, axis=1)
        
        # Update dynamics
        v = v * damping + F * 0.1
        centers += v * dt
        
        # Hard boundary clamping
        centers = np.clip(centers, r, 1.0 - r)
        
        # Track best valid configuration
        # Tolerance 1e-8 is safe relative to validation's 1e-12
        if np.min(dists) >= 2 * r - 1e-8:
            if r > best_r:
                best_r = r
                best_centers = centers.copy()
                
    radii = np.full(n, best_r)
    return best_centers, radii, best_r * n
