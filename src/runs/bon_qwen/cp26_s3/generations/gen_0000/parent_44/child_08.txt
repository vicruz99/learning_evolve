

To maximize the sum of radii for 26 circles in a unit square, we must transition from a standard equal-radius packing to an unequal configuration. While equal circles in a $5 \times 5$ grid yield a sum of 2.5, allowing circles to vary in size enables us to utilize the gaps between circles (interstitial spaces) and corners more efficiently. This approach, combined with a hexagonal lattice initialization for higher density, typically achieves a sum of radii exceeding 2.6.

The strategy involves:
1.  **Initialization:** Placing circles in a dense hexagonal lattice pattern. This maximizes local density and provides a better starting geometry than a square grid.
2.  **Optimization:** Using a numerical optimization method (L-BFGS-B) to maximize the sum of radii. This is achieved by minimizing a loss function that includes a negative sum of radii (to maximize) and a high penalty for constraint violations (overlaps or boundary breaches).
3.  **Refinement:** Iteratively adjusting the penalty weight and running multiple optimization passes with different initializations to escape local minima and find a robust, high-value configuration.

```python
import numpy as np
import math
from scipy.optimize import minimize

def get_circle_params(params, n):
    """Reshape flat parameters array into centers and radii."""
    centers = params[:2 * n].reshape(n, 2)
    radii = params[2 * n:]
    return centers, radii

def compute_loss(params, n, penalty_weight):
    """
    Compute the objective function: -sum(radii) + penalty for constraints.
    """
    centers, radii = get_circle_params(params, n)
    loss = -np.sum(radii)
    
    # Boundary constraints penalty
    # Circle i is inside if: r <= x <= 1-r and r <= y <= 1-r
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        
        # Left boundary
        dist_left = x - r
        if dist_left < 0:
            loss += penalty_weight * (dist_left ** 2)
        
        # Right boundary
        dist_right = (1 - x) - r
        if dist_right < 0:
            loss += penalty_weight * (dist_right ** 2)
            
        # Bottom boundary
        dist_bottom = y - r
        if dist_bottom < 0:
            loss += penalty_weight * (dist_bottom ** 2)
            
        # Top boundary
        dist_top = (1 - y) - r
        if dist_top < 0:
            loss += penalty_weight * (dist_top ** 2)

    # Overlap constraints penalty
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = math.sqrt(dx * dx + dy * dy)
            min_dist = radii[i] + radii[j]
            
            if dist < min_dist:
                overlap = min_dist - dist
                loss += penalty_weight * (overlap ** 2)

    return loss

def get_hexagonal_initialization(n):
    """
    Initialize centers in a hexagonal lattice pattern.
    """
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.05) # Small initial radius
    
    # Hexagonal packing parameters
    # Spacing in x: 2r (approx), spacing in y: sqrt(3)r (approx)
    # We place them in a grid and scale/shift to fit [0,1]x[0,1]
    
    rows = int(np.ceil(np.sqrt(n * math.sqrt(3) / 2)))
    cols = int(np.ceil(n / rows))
    
    # Generate points
    points = []
    idx = 0
    for r_idx in range(rows):
        for c_idx in range(cols):
            if idx >= n:
                break
            x = c_idx * 1.0 + (0.5 if r_idx % 2 == 1 else 0)
            y = r_idx * math.sqrt(3) / 2
            points.append([x, y])
            idx += 1
            
    points = np.array(points[:n])
    
    # Normalize to fit in unit square with some padding
    min_p = points.min(axis=0)
    max_p = points.max(axis=0)
    span = max_p - min_p
    
    # Avoid division by zero
    if span[0] == 0: span[0] = 1
    if span[1] == 0: span[1] = 1
    
    # Scale to 80% of square size to leave room for expansion
    scale = 0.8
    centers = (points - min_p) / span * scale
    centers += (1 - scale) / 2
    
    return centers, radii

def run_packing():
    n = 26
    best_sum_radii = 0
    best_centers = None
    best_radii = None
    
    # Try multiple optimization runs with different penalties and seeds
    penalties = [1000, 5000, 10000]
    n_iterations = 5000
    
    for p_weight in penalties:
        # Generate random variations of the hexagonal lattice
        for _ in range(5):
            centers, radii = get_hexagonal_initialization(n)
            
            # Add slight random noise to break symmetry
            noise = np.random.uniform(-0.02, 0.02, size=centers.shape)
            centers = np.clip(centers + noise, 0, 1)
            
            # Initial parameters array
            params = np.concatenate([centers.flatten(), radii])
            
            # Bounds for x, y in [0, 1], r in [0, 0.5]
            bounds = [(0, 1) for _ in range(2 * n)]
            bounds += [(0, 0.5) for _ in range(n)]
            
            # Optimization
            res = minimize(
                compute_loss,
                params,
                args=(n, p_weight),
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': n_iterations, 'ftol': 1e-12, 'gtol': 1e-12}
            )
            
            # Check result
            curr_centers, curr_radii = get_circle_params(res.x, n)
            curr_sum = np.sum(curr_radii)
            
            # Basic validation check inside optimization loop (optional, but good for debugging)
            # We rely on the loss function to guide us, but we need to ensure validity for the final return
            
            # Simple validity check function (local copy)
            is_valid = True
            for i in range(n):
                x, y = curr_centers[i]
                r = curr_radii[i]
                if x < r or x > 1 - r or y < r or y > 1 - r:
                    is_valid = False
                    break
                for j in range(i + 1, n):
                    dx = curr_centers[i, 0] - curr_centers[j, 0]
                    dy = curr_centers[i, 1] - curr_centers[j, 1]
                    dist = math.sqrt(dx*dx + dy*dy)
                    if dist < curr_radii[i] + curr_radii[j] - 1e-7: # Tolerance
                        is_valid = False
                        break
                if not is_valid:
                    break
            
            # If valid and better sum, update best
            if is_valid and curr_sum > best_sum_radii:
                best_sum_radii = curr_sum
                best_centers = curr_centers.copy()
                best_radii = curr_radii.copy()

    # If no valid solution found (unlikely with high penalty), fallback to a known valid grid
    if best_centers is None:
        # Fallback: 5x5 grid + 1 small circle
        centers = np.array([[0.1 + 0.2*i, 0.1 + 0.2*j] for i in range(5) for j in range(5)])
        radii = np.full(25, 0.1)
        # Add 26th circle
        centers = np.vstack([centers, [0.5, 0.5]]) # This might overlap, need to adjust
        # Better fallback: just small random circles
        centers = np.random.rand(n, 2) * 0.8 + 0.1
        radii = np.full(n, 0.01)
        best_centers = centers
        best_radii = radii
        best_sum_radii = np.sum(radii)

    return best_centers, best_radii, best_sum_radii
```
