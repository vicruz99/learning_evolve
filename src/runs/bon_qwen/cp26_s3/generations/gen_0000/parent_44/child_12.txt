import numpy as np

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    
    Strategy:
    1. Initialize centers in a staggered hexagonal pattern (5-4-5-4-5-3 rows).
    2. Scale the pattern to fit within the unit square.
    3. Run a simulated annealing local optimizer to maximize the minimum separation 
       between circles and the boundaries.
    4. Set all radii to the maximum uniform radius allowed by the optimized geometry.
    """
    np.random.seed(42)
    n = 26
    
    # 1. Hexagonal Initialization
    # Pattern: 5, 4, 5, 4, 5, 3 circles per row
    rows_config = [5, 4, 5, 4, 5, 3]
    centers = []
    y = 0
    for i, n_circles in enumerate(rows_config):
        # Row spacing for hexagonal packing
        y = i * np.sqrt(3)
        # Horizontal shift for staggered rows
        x_start = 0.5 if (i % 2 == 1) else 0
        x = x_start
        for _ in range(n_circles):
            centers.append([x, y])
            x += 1
    
    centers = np.array(centers)
    
    # 2. Scale to fit in [0, 1] x [0, 1]
    x_min, y_min = centers.min(axis=0)
    x_max, y_max = centers.max(axis=0)
    width = x_max - x_min
    height = y_max - y_min
    
    # Scale factor to fit within 1x1 (adding a small buffer for centering)
    scale = 1.0 / max(width, height)
    centers = (centers - np.array([x_min, y_min])) * scale
    centers += (1.0 - scale * np.array([width, height])) / 2 # Center in square
    
    # 3. Local Optimization (Simulated Annealing)
    # Objective: Maximize the minimum distance (gap) between circles and boundaries.
    # The radius will eventually be half of this min distance.
    
    best_centers = centers.copy()
    
    def get_min_dist(c):
        # Distance to walls
        d_walls = np.min([c.min(axis=0), 1.0 - c.max(axis=0)], axis=0)
        min_wall = np.min(d_walls)
        
        # Distance between circles (pairwise)
        # Efficient pairwise distance calculation
        diffs = c[:, np.newaxis, :] - c[np.newaxis, :, :]
        dists = np.sqrt(np.sum(diffs**2, axis=2))
        # Ignore self-distances (diagonal)
        np.fill_diagonal(dists, np.inf)
        min_circle = np.min(dists)
        
        return min(min_wall, 0.5 * min_circle)

    best_dist = get_min_dist(centers)
    temp = 0.02
    n_iter = 5000
    
    for _ in range(n_iter):
        # Select a random circle to move
        idx = np.random.randint(n)
        new_c = centers.copy()
        # Perturbation magnitude proportional to temperature
        new_c[idx] += np.random.normal(0, temp, 2)
        
        # Ensure stays inside [0, 1]
        new_c[idx] = np.clip(new_c[idx], 1e-5, 1 - 1e-5)
        
        current_dist = get_min_dist(new_c)
        delta = current_dist - best_dist
        
        # Acceptance probability
        if delta > 0 or np.random.random() < np.exp(delta / temp):
            centers = new_c
            if current_dist > best_dist:
                best_dist = current_dist
                best_centers = centers.copy()
        
        # Cool down
        temp *= 0.995
        
    # 4. Finalize Radii
    # Use the best centers found
    final_centers = best_centers
    # Calculate the tightest constraint for uniform radii
    min_dist = get_min_dist(final_centers)
    radii = np.full(n, min_dist)
    sum_radii = float(np.sum(radii))
    
    return final_centers, radii, sum_radii
