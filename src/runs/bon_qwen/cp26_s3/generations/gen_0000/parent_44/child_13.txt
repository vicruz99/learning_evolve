from scipy.optimize import minimize

N_CIRCLES = 26

def compute_objective(x):

    return -np.sum(x[2*N_CIRCLES:])

def compute_constraints(x):
    n = N_CIRCLES
    c = x[:2*n].reshape((n, 2))
    r = x[2*n:]
    cons = []
    
    # Boundary constraints: circles must stay inside [0,1]x[0,1]
    cons.extend(c[:, 0] - r)
    cons.extend(1.0 - c[:, 0] - r)
    cons.extend(c[:, 1] - r)
    cons.extend(1.0 - c[:, 1] - r)
    
    # Pairwise non-overlap constraints: distance >= sum of radii
    for i in range(n):
        for j in range(i+1, n):
            d = np.sqrt(np.sum((c[i] - c[j])**2))
            cons.append(d - (r[i] + r[j]))
            
    return np.array(cons)

def run_packing():

            break
