import numpy as np
import scipy.optimize as opt

def calculate_objective(params, n_circles):
    """
    Calculates the objective value to be minimized (negative sum of radii + penalties).
    """
    centers = params[:2 * n_circles].reshape(n_circles, 2)
    radii = params[2 * n_circles:]

    # Penalty for radii being too small or negative
    radii_penalty = np.maximum(0, -radii + 1e-6)
    
    # Boundary penalties
    boundary_penalty = 0.0
    for i in range(n_circles):
        x, y = centers[i]
        r = radii[i]
        if x < r: boundary_penalty += (x - r) ** 2
        if y < r: boundary_penalty += (y - r) ** 2
        if x > 1 - r: boundary_penalty += (x - (1 - r)) ** 2
        if y > 1 - r: boundary_penalty += (y - (1 - r)) ** 2

    # Overlap penalties
    overlap_penalty = 0.0
    for i in range(n_circles):
        for j in range(i + 1, n_circles):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            min_dist = radii[i] + radii[j]
            if dist < min_dist:
                overlap_penalty += (dist - min_dist) ** 2

    # Objective: Maximize sum of radii => Minimize negative sum
    # We use a very high weight for penalties to enforce constraints
    objective = -np.sum(radii) + 1000 * (radii_penalty + boundary_penalty + overlap_penalty)
    return objective
