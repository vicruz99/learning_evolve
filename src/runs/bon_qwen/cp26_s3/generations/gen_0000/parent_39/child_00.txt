N_CIRCLES = 26

def objective(vars):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(vars[2::3])

def gradient(vars):
    """Gradient of the objective function."""
    grad = np.zeros_like(vars)
    # d/dr (-r) = -1
    grad[2::3] = -1.0
    return grad

def boundary_constraints(vars):
    """
    Constraints: x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
    """
    n = N_CIRCLES
    vals = []
    for i in range(n):
        x = vars[3*i]
        y = vars[3*i+1]
        r = vars[3*i+2]
        vals.extend([x - r, 1 - x - r, y - r, 1 - y - r])
    return np.array(vals)

def boundary_constraints_jac(vars):
    """Jacobian of boundary constraints."""
    n = N_CIRCLES
    jac = np.zeros((n * 4, n * 3))
    for i in range(n):
        base = i * 4
        x_idx, y_idx, r_idx = 3*i, 3*i+1, 3*i+2
        # x - r >= 0
        jac[base, x_idx] = 1.0; jac[base, r_idx] = -1.0
        # 1 - x - r >= 0
        jac[base+1, x_idx] = -1.0; jac[base+1, r_idx] = -1.0
        # y - r >= 0
        jac[base+2, y_idx] = 1.0; jac[base+2, r_idx] = -1.0
        # 1 - y - r >= 0
        jac[base+3, y_idx] = -1.0; jac[base+3, r_idx] = -1.0
    return jac

def non_overlap_constraints(vars):
    """
    Constraints: dist_sq >= (ri + rj)^2  =>  dist_sq - (ri + rj)^2 >= 0
    """
    n = N_CIRCLES
    vals = []
    for i in range(n):
        xi, yi, ri = vars[3*i], vars[3*i+1], vars[3*i+2]
        for j in range(i + 1, n):
            xj, yj, rj = vars[3*j], vars[3*j+1], vars[3*j+2]
            dx = xi - xj
            dy = yi - yj
            dist_sq = dx*dx + dy*dy
            r_sum = ri + rj
            vals.append(dist_sq - r_sum*r_sum)
    return np.array(vals)

def non_overlap_constraints_jac(vars):
    """Jacobian of non-overlap constraints."""
    n = N_CIRCLES
    m = n * (n - 1) // 2
    jac = np.zeros((m, n * 3))
    idx = 0
    for i in range(n):
        xi, yi, ri = vars[3*i], vars[3*i+1], vars[3*i+2]
        col_i_x, col_i_y, col_i_r = 3*i, 3*i+1, 3*i+2
        for j in range(i + 1, n):
            xj, yj, rj = vars[3*j], vars[3*j+1], vars[3*j+2]
            dx = xi - xj
            dy = yi - yj
            r_sum = ri + rj
            
            col_j_x, col_j_y, col_j_r = 3*j, 3*j+1, 3*j+2
            
            # Derivatives of (xi-xj)^2 + (yi-yj)^2 - (ri+rj)^2
            # w.r.t xi: 2(xi-xj) = 2dx
            # w.r.t yi: 2(yi-yj) = 2dy
            # w.r.t xj: -2(xi-xj) = -2dx
            # w.r.t yj: -2(yi-yj) = -2dy
            # w.r.t ri: -2(ri+rj)
            # w.r.t rj: -2(ri+rj)
            
            jac[idx, col_i_x] = 2.0 * dx
            jac[idx, col_i_y] = 2.0 * dy
            jac[idx, col_j_x] = -2.0 * dx
            jac[idx, col_j_y] = -2.0 * dy
            jac[idx, col_i_r] = -2.0 * r_sum
            jac[idx, col_j_r] = -2.0 * r_sum
            idx += 1
    return jac

def run_packing():

        radii_list = []
