import numpy as np
from scipy.optimize import minimize


def make_hexagonal_init(n, r, shifted=True):
    """Create hexagonal lattice initial configuration."""
    centers = []
    y_spacing = r * np.sqrt(3)
    
    row = 0
    while len(centers) < n:
        y = r + row * y_spacing * 2
        if y + r > 1 + 1e-6:
            break
        x_start = r
        if row % 2 == 1 and shifted:
            x_start = r + r  # shift by 2r for hexagonal
        col = 0
        while len(centers) < n:
            x = x_start + col * 2 * r
            if x + r > 1 + 1e-6:
                break
            centers.append([x, y])
            col += 1
        row += 1
    
    return np.array(centers[:n])


def make_random_init(n, r):
    """Create random initial configuration with equal radii."""
    centers = []
    attempts = 0
    while len(centers) < n and attempts < 10000:
        x = np.random.uniform(r, 1 - r)
        y = np.random.uniform(r, 1 - r)
        valid = True
        for cx, cy in centers:
            if np.sqrt((x-cx)**2 + (y-cy)**2) < 2*r + 1e-6:
                valid = False
                break
        if valid:
            centers.append([x, y])
        attempts += 1
    
    # Fill remaining if needed
    while len(centers) < n:
        centers.append([np.random.uniform(r, 1-r), np.random.uniform(r, 1-r)])
    
    return np.array(centers[:n])


def optimize_packing(centers, radii, n, max_iter=5000):
    """Optimize packing using SLSQP."""
    
    def objective(x):
        return -np.sum(x[2::3])
    
    def constraints(x):
        cons = []
        for i in range(n):
            cons.extend([
                x[3*i] - x[3*i+2],
                1 - x[3*i] - x[3*i+2],
                x[3*i+1] - x[3*i+2],
                1 - x[3*i+1] - x[3*i+2]
            ])
        
        for i in range(n):
            for j in range(i+1, n):
                dx = x[3*i] - x[3*j]
                dy = x[3*i+1] - x[3*j+1]
                dist_sq = dx*dx + dy*dy
                min_dist_sq = (x[3*i+2] + x[3*j+2])**2
                cons.append(dist_sq - min_dist_sq)
        
        return np.array(cons)
    
    x0 = np.zeros(3*n)
    for i in range(n):
        x0[3*i] = centers[i, 0]
        x0[3*i+1] = centers[i, 1]
        x0[3*i+2] = radii[i]
    
    bounds = [(0, 1), (0, 1), (1e-8, 0.5)] * n
    cons = {'type': 'ineq', 'fun': constraints}
    
    try:
        result = minimize(objective, x0, method='SLSQP', bounds=bounds,
                         constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-15, 'disp': False})
        return result.x, result.success
    except Exception:
        return x0, False


def relax_and_expand(centers, radii, n, steps=300):
    """Iteratively expand circles while relaxing positions."""
    n = len(centers)
    centers = centers.copy()
    radii = radii.copy()
    
    for step in range(steps):
        # Compute minimum gap
        min_gap = float('inf')
        
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            gap = min(x - r, 1 - x - r, y - r, 1 - y - r)
            min_gap = min(min_gap, gap)
        
        for i in range(n):
            for j in range(i+1, n):
                dx = centers[j, 0] - centers[i, 0]
                dy = centers[j, 1] - centers[i, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                gap = dist - radii[i] - radii[j]
                min_gap = min(min_gap, gap)
        
        # Expand radii
        expand_rate = 0.3
        if min_gap > 1e-8:
            radii += min_gap * expand_rate
        
        # Relax positions with forces
        lr = 0.005 / (1 + step * 0.005)
        for _ in range(50):
            forces = np.zeros((n, 2))
            
            for i in range(n):
                for j in range(i+1, n):
                    dx = centers[j, 0] - centers[i, 0]
                    dy = centers[j, 1] - centers[i, 1]
                    dist = np.sqrt(dx*dx + dy*dy) + 1e-10
                    min_dist = radii[i] + radii[j]
                    
                    if dist < min_dist:
                        overlap = min_dist - dist
                        fx = overlap * dx / dist
                        fy = overlap * dy / dist
                        forces[i, 0] -= fx
                        forces[j, 0] += fx
                        forces[i, 1] -= fy
                        forces[j, 1] += fy
            
            for i in range(n):
                x, y = centers[i]
                r = radii[i]
                if x - r < 0:
                    forces[i, 0] += max(0, -(x - r)) * 10
                if x + r > 1:
                    forces[i, 0] -= max(0, x + r - 1) * 10
                if y - r < 0:
                    forces[i, 1] += max(0, -(y - r)) * 10
                if y + r > 1:
                    forces[i, 1] -= max(0, y + r - 1) * 10
            
            centers += forces * lr
            lr *= 0.98
    
    return centers, radii
