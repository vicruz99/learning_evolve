import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    
    Returns:
        centers: np.array of shape (26, 2)
        radii: np.array of shape (26,)
        sum_radii: float
    """
    n = 26
    best_sum_radii = -1.0
    best_centers = None
    best_radii = None
    
    # Helper function to generate constraints
    def make_constraints(centers, radii):
        constraints = []
        
        # Boundary constraints: x-r >= 0, x+r <= 1, y-r >= 0, y+r <= 1
        # Flattened index mapping: 0..2n-1 are coords, 2n..3n-1 are radii
        # But here we just define the logic for the bounds and the optimizer handles bounds separately?
        # Actually, SLSQP supports bounds. Constraints are for non-linear relationships.
        
        # Overlap constraints
        for i in range(n):
            for j in range(i + 1, n):
                def overlap_constraint(vars, i=i, j=j):
                    # vars layout: [x0, y0, r0, x1, y1, r1, ...]
                    xi, yi, ri = vars[3*i], vars[3*i+1], vars[3*i+2]
                    xj, yj, rj = vars[3*j], vars[3*j+1], vars[3*j+2]
                    dist_sq = (xi - xj)**2 + (yi - yj)**2
                    sum_r = ri + rj
                    return dist_sq - sum_r**2  # Must be >= 0
                
                constraints.append({'type': 'ineq', 'fun': overlap_constraint})
        
        return constraints

    # Objective function: minimize -sum(radii)
    def objective(vars):
        radii = vars[2::3] # Extract radii
        return -np.sum(radii)

    # Define bounds
    # x in [0, 1], y in [0, 1], r in [0, 0.5]
    bounds = []
    for _ in range(n):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 0.5)) # r

    # We will run multiple optimizations from different starting points
    # to escape local optima.
    
    best_result = None
    
    for attempt in range(10):
        # Generate initial grid positions

        rows = 5

        count = 0
        for r in range(rows):
            for c in range(cols):
                if count < n:
                    cx = (c + 1) * dx
                    cy = (r + 1) * dy
                    centers_init.append([cx, cy])
                    count += 1
        
        # Add some random perturbation to avoid symmetry traps
        noise = np.random.uniform(-0.02, 0.02, size=(n, 2))
        centers_init = np.array(centers_init) + noise
        # Clip to stay within bounds (with margin for radius)
        centers_init = np.clip(centers_init, 0.05, 0.95)
        
        # Initial small radius
        radii_init = np.full(n, 0.05)

        x0 = np.zeros(3 * n)
        for i in range(n):
            x0[3*i] = centers_init[i, 0]
            x0[3*i+1] = centers_init[i, 1]
            x0[3*i+2] = radii_init[i]
            
        constraints = make_constraints(None, None) # Logic is inside lambda closure
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints,
                           options={'maxiter': 200, 'ftol': 1e-9})
            
            if res.success:
                sum_r = -res.fun
                if sum_r > best_sum_radii:
                    best_sum_radii = sum_r
                    # Extract results
                    best_centers = res.x[0::3].reshape(n, 2)
                    best_radii = res.x[2::3]
        except Exception as e:
            pass

    # If optimization failed to find anything, fallback to a valid grid
    if best_centers is None:

        rows = 5
        dx = 1.0 / (cols + 1)
        dy = 1.0 / (rows + 1)
        centers = np.zeros((n, 2))

        count = 0
        for r in range(rows):
            for c in range(cols):
                if count < n:
                    centers[count] = [(c + 1) * dx, (r + 1) * dy]
                    count += 1
        best_centers = centers
        best_radii = radii

    # Final validation and slight adjustment if needed (clipping)
    # Ensure radii are strictly positive
    best_radii = np.maximum(best_radii, 1e-9)

        if y - r < 0: best_centers[i, 1] = r
        if y + r > 1: best_centers[i, 1] = 1 - r

    return best_centers, best_radii, best_sum_radii
