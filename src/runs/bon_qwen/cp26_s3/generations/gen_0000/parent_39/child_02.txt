import numpy as np
from scipy.optimize import minimize

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square

    Args:
        centers: np.array of shape (n, 2) with (x, y) coordinates
        radii: np.array of shape (n) with radius of each circle

    Returns:
        True if valid, False otherwise
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True

def get_constraints(n):
    """
    Define constraints for the optimization problem.
    Constraints:
    1. x_i >= r_i  => x_i - r_i >= 0
    2. x_i <= 1 - r_i => 1 - x_i - r_i >= 0
    3. y_i >= r_i  => y_i - r_i >= 0
    4. y_i <= 1 - r_i => 1 - y_i - r_i >= 0
    5. dist(i, j) >= r_i + r_j => dist(i, j)^2 - (r_i + r_j)^2 >= 0
    """
    constraints = []

    # Boundary constraints
    for i in range(n):
        # x_i - r_i >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, idx=i: v[3 * idx] - v[3 * idx + 2]
        })
        # 1 - x_i - r_i >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, idx=i: 1.0 - v[3 * idx] - v[3 * idx + 2]
        })
        # y_i - r_i >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, idx=i: v[3 * idx + 1] - v[3 * idx + 2]
        })
        # 1 - y_i - r_i >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, idx=i: 1.0 - v[3 * idx + 1] - v[3 * idx + 2]
        })

    # Overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            def overlap_dist(v, i=i, j=j):
                xi, yi, ri = v[3*i], v[3*i+1], v[3*i+2]
                xj, yj, rj = v[3*j], v[3*j+1], v[3*j+2]
                dx = xi - xj
                dy = yi - yj
                dist_sq = dx*dx + dy*dy
                r_sum = ri + rj
                return dist_sq - r_sum * r_sum
            
            constraints.append({
                'type': 'ineq',
                'fun': overlap_dist
            })

    return constraints

def objective(v, n):
    """
    Objective function: Minimize negative sum of radii.
    Variables layout: [x1, y1, r1, x2, y2, r2, ...]
    """
    radii = v[2::3]
    return -np.sum(radii)

def run_optimization(init_v, n, constraints):
    """
    Run the optimizer with a given initial configuration.
    """
    # Bounds for variables
    # x, y in [0, 1], r in [0, 0.5]
    bounds = []
    for _ in range(n):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 0.5)) # r

    try:
        res = minimize(
            objective,
            init_v,
            args=(n,),
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 200, 'ftol': 1e-9}
        )
        return res
    except Exception:
        return None

def generate_hexagonal_init(n, r_init=0.08):
    """
    Generate a hexagonal grid initialization for n circles.
    """
    centers = []
    r = r_init
    dx = 2 * r
    dy = np.sqrt(3) * r
    
    y = r
    row_idx = 0
    while len(centers) < n:
        row_width = 1.0 - 2 * r
        num_in_row = int(row_width / dx) + 1
        # Adjust number in row to not exceed n
        if len(centers) + num_in_row > n:
            num_in_row = n - len(centers)
        
        x_start = (1.0 - (num_in_row - 1) * dx) / 2.0
        
        # Shift odd rows
        offset = (dx / 2.0) if row_idx % 2 == 1 else 0.0
        
        row_centers = []
        for k in range(num_in_row):
            x = x_start + k * dx + offset
            # If shifted, might go out of bounds slightly, clamp or just add
            if x < r: x = r
            if x > 1 - r: x = 1 - r
            row_centers.append([x, y])
        
        centers.extend(row_centers)
        y += dy
        row_idx += 1
        
    radii = np.full(n, r)
    return centers, radii
