import numpy as np
from scipy.optimize import minimize

N = 26

def objective(vars):
    r = vars[2::3]
    return -np.sum(r)

def constraints(vars):
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    # Boundary constraints: circles must stay inside [0,1]x[0,1]
    # Formulated as: x - r >= 0, y - r >= 0, 1 - x - r >= 0, 1 - y - r >= 0
    con = np.concatenate([
        x - r,
        y - r,
        1.0 - x - r,
        1.0 - y - r
    ])
    
    # Overlap constraints: distance^2 >= (r1 + r2)^2
    overlaps = []
    for i in range(N):
        for j in range(i + 1, N):
            dx = x[i] - x[j]
            dy = y[i] - y[j]
            overlaps.append(dx*dx + dy*dy - (r[i] + r[j])**2)
            
    return np.concatenate([con, overlaps])

def run_packing():
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    for seed in range(12):
        try:
            np.random.seed(seed)
            
            # Initial layout: staggered grid approximating hexagonal packing
            centers = np.zeros((N, 2))
            idx = 0
            for row in range(6):
                for col in range(5):
                    if idx >= N: break
                    # Hexagonal shift pattern
                    cx = col * 0.175 + 0.12 + (row % 2) * 0.0875
                    cy = row * 0.175 + 0.12
                    centers[idx] = [cx, cy]
                    idx += 1
                if idx >= N: break
                
            # Small random perturbation to break symmetry and escape trivial local minima
            centers += np.random.randn(N, 2) * 0.005
            centers = np.clip(centers, 0.05, 0.95)
            
            r_init = 0.045
            radii = np.full(N, r_init)
            
            # Flatten variables: [x1, x2, ..., xN, y1, y2, ..., yN, r1, r2, ..., rN]
            x0 = np.hstack([centers[:, 0], centers[:, 1], radii])
            
            # Bounds: x,y in [0,1], r in [0.01, 0.5]
            bounds = [(0.0, 1.0)] * N + [(0.0, 1.0)] * N + [(0.01, 0.5)] * N
            
            cons = {'type': 'ineq', 'fun': constraints}
            
            res = minimize(objective, x0, method='SLSQP', 
                           bounds=bounds, constraints=cons, 
                           options={'maxiter': 8000, 'ftol': 1e-10, 'disp': False})
            
            if res.success:
                x_opt = res.x
                c_opt = np.column_stack((x_opt[0::3], x_opt[1::3]))
                r_opt = x_opt[2::3]
                
                # Strict post-optimization validation
                valid = True
                if np.any(r_opt < 0.005):
                    valid = False
                if np.any(c_opt[:, 0] < -1e-9) or np.any(c_opt[:, 0] > 1.0 + 1e-9):
                    valid = False
                if np.any(c_opt[:, 1] < -1e-9) or np.any(c_opt[:, 1] > 1.0 + 1e-9):
                    valid = False
                    
                for i in range(N):
                    if c_opt[i, 0] - r_opt[i] < -1e-7 or c_opt[i, 0] + r_opt[i] > 1.0 + 1e-7:
                        valid = False
                    if c_opt[i, 1] - r_opt[i] < -1e-7 or c_opt[i, 1] + r_opt[i] > 1.0 + 1e-7:
                        valid = False
                for i in range(N):
                    for j in range(i+1, N):
                        dist = np.hypot(c_opt[i,0]-c_opt[j,0], c_opt[i,1]-c_opt[j,1])
                        if dist < r_opt[i] + r_opt[j] - 1e-7:
                            valid = False
                if not valid:
                    continue
                    
                current_sum = np.sum(r_opt)
                if current_sum > best_sum:
                    best_sum = current_sum
                    best_centers = c_opt.copy()
                    best_radii = r_opt.copy()
        except Exception:
            continue
            
    # Fallback configuration (guaranteed valid)
    if best_centers is None:
        best_centers = np.array([[i%6*0.16+0.1, i//6*0.16+0.1] for i in range(N)])
        best_radii = np.full(N, 0.05)
        best_sum = np.sum(best_radii)
        
    return best_centers, best_radii, best_sum
