import numpy as np
from scipy.optimize import minimize
import math

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    Pack 26 circles in a unit square to maximize the sum of radii.
    Uses scipy.optimize to refine an initial hexagonal packing.
    """

    # We estimate a radius slightly less than 0.1 to start safely inside bounds.
    r_init = 0.095
    
    centers = np.zeros((n, 2))
    radii = np.full(n, r_init)

    # Hexagonal packing: rows of circles shifted by radius horizontally
    y = r_init
    row_count = 0
    while idx < n:
        # Determine x start for this row

        if row_count % 2 == 0:
            x_start = r_init
        else:
            x_start = r_init + r_init # Shifted by radius (distance between centers in gap is r horizontally? No, 2r spacing implies shift r)
            # Wait, in hex packing, horizontal distance between adjacent centers is 2r.
            # Vertical distance is sqrt(3)r.
            # If row 1 is at x = r, 3r, 5r...
            # Row 2 nests in between, so x = 2r, 4r, 6r...
            # Shift is r.
        
        x = x_start
        row_circles = 0
        
        # Fill row
        while x + r_init <= 1.0 + 1e-9 and idx < n:
            centers[idx] = [x, y]
            idx += 1
            row_circles += 1
            x += 2 * r_init
            
        y += r_init * math.sqrt(3)
        row_count += 1

    # Ensure we filled all circles (loop might stop early if width/height limits reached)

            # Fallback: place in empty spots or just grid

            gy = (grid_pos // 6 + 1) * 0.15 + 0.05
            # Clamp and adjust

            gy = max(0.1, min(0.9, gy))
            centers[i] = [gx, gy]

    # Flatten variables for optimizer: [x1, y1, r1, x2, y2, r2, ...]

    for i in range(n):
        x0[3*i] = centers[i, 0]
        x0[3*i+1] = centers[i, 1]
        x0[3*i+2] = radii[i]

    def objective(vars_flat):
        # Maximize sum of radii -> Minimize negative sum
        r_sum = 0.0
        for i in range(n):
            r_sum += vars_flat[3*i+2]
        return -r_sum

    def constraints(vars_flat):
        cons = []
        for i in range(n):
            xi = vars_flat[3*i]
            yi = vars_flat[3*i+1]
            ri = vars_flat[3*i+2]

            cons.append({'type': 'ineq', 'fun': lambda v, idx=i: v[3*idx+1] - v[3*idx+2]}) # y - r >= 0
            cons.append({'type': 'ineq', 'fun': lambda v, idx=i: 1.0 - v[3*idx+2] - v[3*idx+1]}) # 1 - r - y >= 0
            cons.append({'type': 'ineq', 'fun': lambda v, idx=i: v[3*idx+2]}) # r >= 0

        # (xi-xj)^2 + (yi-yj)^2 >= (ri+rj)^2

            for j in range(i + 1, n):
                xi, yi, ri = vars_flat[3*i], vars_flat[3*i+1], vars_flat[3*i+2]

                # Constraint: dist - (ri + rj) >= 0

                    dist = math.sqrt((xi - xj)**2 + (yi - yj)**2)

        print(f"Optimization failed: {e}")
        x_opt = x0

    # --- 3. Extract and Validate ---
    final_centers = np.zeros((n, 2))

            final_centers[i, 0] = r

            final_centers[i, 1] = r
        elif final_centers[i, 1] > 1 - r:
            final_centers[i, 1] = 1 - r
