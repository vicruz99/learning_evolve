    p0_grid = np.zeros(3*n)
    for i in range(n):
        p0_grid[3*i] = points_grid[i][0]
        p0_grid[3*i+1] = points_grid[i][1]
        p0_grid[3*i+2] = r_grid
    inits.append(p0_grid)
    
    # Run optimization
    for p0 in inits:
        try:
            res = opt.minimize(objective, p0, method='SLSQP', bounds=bounds, 
                               constraints=cons, options={'maxiter': 10000, 'ftol': 1e-12})
            
            cx = res.x[0::3]
            cy = res.x[1::3]
            r_vals = res.x[2::3]
            centers = np.column_stack((cx, cy))
            
            if validate_packing(centers, r_vals):
                s = np.sum(r_vals)
                if s > best_sum:
                    best_sum = s
                    best_centers = centers.copy()
                    best_radii = r_vals.copy()
        except Exception:
            pass
            
    if best_centers is None:
        centers = np.tile([0.5, 0.5], (n, 1))
        radii = np.full(n, 0.01)
        best_sum = 0.26
    
    return best_centers, best_radii, float(best_sum)
