def objective(vars, n, penalty_weight=1e6):
    centers = vars[:2*n].reshape((n, 2))
    radii = vars[2*n:]
    
    sum_radii = np.sum(radii)
    
    penalty = 0.0
    
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            min_dist = radii[i] + radii[j]
            overlap = min_dist - dist
            if overlap > 0:
                penalty += penalty_weight * overlap * overlap
    
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        v1 = r - x
        if v1 > 0:
            penalty += penalty_weight * v1 * v1
        v2 = x + r - 1.0
        if v2 > 0:
            penalty += penalty_weight * v2 * v2
        v3 = r - y
        if v3 > 0:
            penalty += penalty_weight * v3 * v3
        v4 = y + r - 1.0
        if v4 > 0:
            penalty += penalty_weight * v4 * v4
    
    return -sum_radii + penalty

                x_positions = [0.5]

        centers.append([0.5 + (len(centers) - 25) * 0.05, 0.5])
    
    return np.array(centers[:n])

        size=(n, 2)
    )
