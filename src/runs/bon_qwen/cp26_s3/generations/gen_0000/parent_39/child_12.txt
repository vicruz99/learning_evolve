import numpy as np
from scipy.optimize import minimize

def run_packing():
    """
    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    """
    n = 26
    max_iterations = 100
    
    # Objective function: Maximize sum of radii (minimize negative sum)
    def objective(vars_flat):
        radii = vars_flat[2::3]
        return -np.sum(radii)

    # Constraints for SLSQP
    # Type 'ineq' implies func(x) >= 0
    
    def wall_constraints(vars_flat):
        # Constraints: x - r >= 0, 1 - x - r >= 0, etc.
        cons_val = []
        for i in range(n):
            xi = vars_flat[3 * i]
            yi = vars_flat[3 * i + 1]
            ri = vars_flat[3 * i + 2]
            cons_val.append(xi - ri)
            cons_val.append(1.0 - xi - ri)
            cons_val.append(yi - ri)
            cons_val.append(1.0 - yi - ri)
        return np.array(cons_val)

    def pair_constraints(vars_flat):
        # Constraints: dist^2 - (ri + rj)^2 >= 0
        cons_val = []
        for i in range(n):
            xi = vars_flat[3 * i]
            yi = vars_flat[3 * i + 1]
            ri = vars_flat[3 * i + 2]
            for j in range(i + 1, n):
                xj = vars_flat[3 * j]
                yj = vars_flat[3 * j + 1]
                rj = vars_flat[3 * j + 2]
                dist_sq = (xi - xj)**2 + (yi - yj)**2
                cons_val.append(dist_sq - (ri + rj)**2)
        return np.array(cons_val)

    constraints = [
        {'type': 'ineq', 'fun': wall_constraints},
        {'type': 'ineq', 'fun': pair_constraints}
    ]

    # Bounds for variables: x, y in [0, 1], r in [0, 0.5]
    bounds = [(0, 1), (0, 1), (0, 0.5)] * n

    best_result = None
    best_score = -np.inf

    # Try multiple initializations to avoid local minima
    initializations = []
    
    # 1. Grid layout (approx 5x5)
    pts = []
    for row in range(5):
        for col in range(6):
            if len(pts) >= 26: break
            pts.append((0.1 + col * 0.16, 0.1 + row * 0.18))
        if len(pts) >= 26: break
    
    vars_grid = np.zeros(3 * n)
    for i, (x, y) in enumerate(pts):
        vars_grid[3*i] = x
        vars_grid[3*i+1] = y
        vars_grid[3*i+2] = 0.01 # Small initial radius
    initializations.append(vars_grid)

    # 2. Random layout
    for _ in range(4):
        vars_rand = np.random.uniform(0.1, 0.9, 3 * n)
        vars_rand[2::3] = 0.01
        initializations.append(vars_rand)

    # 3. Hexagonal-like perturbation
    vars_hex = np.zeros(3 * n)
    idx = 0
    for r in range(6):
        for c in range(5):
            if idx >= 26: break
            x = 0.1 + c * 0.2 + (0.1 if r % 2 == 1 else 0)
            y = 0.1 + r * 0.17
            vars_hex[3*idx] = x
            vars_hex[3*idx+1] = y
            vars_hex[3*idx+2] = 0.01
            idx += 1
    initializations.append(vars_hex)

    # Optimization loop
    for i, x0 in enumerate(initializations):
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, 
                           constraints=constraints, options={'maxiter': max_iterations, 'ftol': 1e-9})
            
            if res.success and -res.fun > best_score:
                best_score = -res.fun
                best_result = res.x
        except Exception:
            continue

    if best_result is None:
        # Fallback to a safe valid packing if optimization fails
        centers = np.zeros((26, 2))
        radii = np.zeros(26)
        for i in range(26):
            x = 0.05 + (i % 6) * 0.15
            y = 0.05 + (i // 6) * 0.18
            centers[i] = [x, y]
            radii[i] = 0.05
        return centers, radii, np.sum(radii)

    # Extract results
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        centers[i, 0] = best_result[3 * i]
        centers[i, 1] = best_result[3 * i + 1]
        radii[i] = best_result[3 * i + 2]

    return centers, radii, np.sum(radii)
