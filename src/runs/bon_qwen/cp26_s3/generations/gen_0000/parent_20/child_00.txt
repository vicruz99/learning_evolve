import numpy as np
from scipy.optimize import minimize

def _get_initial_config(n):
    """Generate a deterministic hexagonal initial configuration."""
    pts = []
    y = 0.08
    while y < 1.0 and len(pts) < n:
        x = 0.08
        is_shifted = int(round((y - 0.08) / 0.14)) % 2 == 1
        if is_shifted:
            x = 0.20
        while x < 1.0 and len(pts) < n:
            pts.append([x, y])
            x += 0.22
        y += 0.14 * np.sqrt(3)
        
    pts = np.array(pts)
    if len(pts) < n:
        # Fallback random if grid didn't yield enough (unlikely)
        pts = np.random.rand(n, 2)
        
    # Compute a safe initial radius to guarantee feasibility
    dists = np.sqrt(np.sum((pts[:, None, :] - pts[None, :, :])**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    min_d = np.min(dists)
    r_init = 0.35 * min_d / 2.0
    
    return pts[:n], r_init

def _obj_func(v):
    """Negative sum of radii."""
    return -np.sum(v[2::3])

def _constr_fun(v):
    """Constraint function values (must be >= 0 for SLSQP ineq)."""
    n = len(v) // 3
    c = []
    # Boundary constraints
    for i in range(n):
        xi, yi, ri = v[3*i], v[3*i+1], v[3*i+2]
        c.append(1.0 - xi - ri)  # x + r <= 1
        c.append(xi - ri)        # x >= r
        c.append(1.0 - yi - ri)  # y + r <= 1
        c.append(yi - ri)        # y >= r
    # Overlap constraints
    for i in range(n):
        for j in range(i+1, n):
            xi, yi, ri = v[3*i], v[3*i+1], v[3*i+2]
            xj, yj, rj = v[3*j], v[3*j+1], v[3*j+2]
            dx, dy = xi - xj, yi - yj
            c.append(dx*dx + dy*dy - (ri + rj)**2)
    return np.array(c)

def _constr_jac(v):
    """Analytical Jacobian of constraints."""
    n = len(v) // 3
    n_c = 4*n + n*(n-1)//2
    jac = np.zeros((n_c, 3*n))
    row = 0
    
    # Boundary Jacobians
    for i in range(n):
        # 1 - x - r
        jac[row, 3*i] = -1.0; jac[row, 3*i+2] = -1.0; row += 1
        # x - r
        jac[row, 3*i] = 1.0; jac[row, 3*i+2] = -1.0; row += 1
        # 1 - y - r
        jac[row, 3*i+1] = -1.0; jac[row, 3*i+2] = -1.0; row += 1
        # y - r
        jac[row, 3*i+1] = 1.0; jac[row, 3*i+2] = -1.0; row += 1
        
    # Overlap Jacobians: d/dv (dist^2 - (r1+r2)^2)
    for i in range(n):
        for j in range(i+1, n):
            xi, yi, ri = v[3*i], v[3*i+1], v[3*i+2]
            xj, yj, rj = v[3*j], v[3*j+1], v[3*j+2]
            dx, dy = xi - xj, yi - yj
            
            jac[row, 3*i] = 2*dx
            jac[row, 3*j] = -2*dx
            jac[row, 3*i+1] = 2*dy
            jac[row, 3*j+1] = -2*dy
            jac[row, 3*i+2] = -2*(ri + rj)
            jac[row, 3*j+2] = -2*(ri + rj)
            row += 1
    return jac

def run_packing():
    n = 26
    centers_init, r_init = _get_initial_config(n)
    
    v0 = np.zeros(3*n)
    v0[0::3] = centers_init[:, 0]
    v0[1::3] = centers_init[:, 1]
    v0[2::3] = r_init
    
    bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n
    cons = {'type': 'ineq', 'fun': _constr_fun, 'jac': _constr_jac}
    
    res = minimize(_obj_func, v0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False})
                   
    best_v = res.x
    centers = np.column_stack((best_v[0::3], best_v[1::3]))
    radii = best_v[2::3]
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
