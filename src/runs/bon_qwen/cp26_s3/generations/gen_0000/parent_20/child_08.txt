import numpy as np
from scipy.optimize import minimize
from functools import partial


def boundary_x_lower(x, i):
    return x[3 * i] - x[3 * i + 2]


def boundary_x_upper(x, i):
    return 1 - x[3 * i] - x[3 * i + 2]


def boundary_y_lower(x, i):
    return x[3 * i + 1] - x[3 * i + 2]


def boundary_y_upper(x, i):
    return 1 - x[3 * i + 1] - x[3 * i + 2]


def radius_nonneg(x, i):
    return x[3 * i + 2]


def non_overlap(x, i, j):
    dx = x[3 * i] - x[3 * j]
    dy = x[3 * i + 1] - x[3 * j + 1]
    return np.sqrt(dx * dx + dy * dy) - x[3 * i + 2] - x[3 * j + 2]


def make_initial_hexagonal(n, r_init):
    centers = []
    for row in range(5):
        y = r_init + row * r_init * np.sqrt(3)
        for col in range(5):
            x = r_init + col * 2 * r_init
            if row % 2 == 1:
                x += r_init
            centers.append([x, y])
    centers.append([0.5, 1 - r_init])
    centers = np.array(centers[:n])
    radii = np.full(n, r_init)
    return centers, radii


def make_initial_grid(n, r_init):
    centers = []
    for row in range(5):
        for col in range(5):
            x = r_init + col * (1 - 2 * r_init) / 4
            y = r_init + row * (1 - 2 * r_init) / 4
            centers.append([x, y])
    centers.append([1 - r_init, 1 - r_init])
    centers = np.array(centers[:n])
    radii = np.full(n, r_init)
    return centers, radii


def make_initial_staggered(n, r_init):
    centers = []
    # Row pattern: 5, 5, 5, 5, 5, 1
    rows = [5, 5, 5, 5, 5, 1]
    y_pos = 0
    for row_idx, count in enumerate(rows):
        if count == 0:
            continue
        y = r_init + y_pos
        x_start = r_init + (5 - count) * r_init if row_idx % 2 == 1 else r_init
        for col in range(count):
            x = x_start + col * 2 * r_init
            centers.append([x, y])
        y_pos += r_init * np.sqrt(3)
    
    centers = np.array(centers[:n])
    radii = np.full(n, r_init)
    return centers, radii


def optimize_from_initial(centers_init, radii_init, n, max_iter=2000):
    x0 = np.zeros(3 * n)
    for i in range(n):
        x0[3 * i] = centers_init[i, 0]
        x0[3 * i + 1] = centers_init[i, 1]
        x0[3 * i + 2] = radii_init[i]
    
    bounds = []
    for i in range(n):
        bounds.append((0, 1))
        bounds.append((0, 1))
        bounds.append((0, 0.5))
    
    def objective(x):
        return -np.sum(x[2::3])
    
    constraints = []
    
    for i in range(n):
        constraints.append({'type': 'ineq', 'fun': partial(boundary_x_lower, i=i)})
        constraints.append({'type': 'ineq', 'fun': partial(boundary_x_upper, i=i)})
        constraints.append({'type': 'ineq', 'fun': partial(boundary_y_lower, i=i)})
        constraints.append({'type': 'ineq', 'fun': partial(boundary_y_upper, i=i)})
        constraints.append({'type': 'ineq', 'fun': partial(radius_nonneg, i=i)})
    
    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({'type': 'ineq', 'fun': partial(non_overlap, i=i, j=j)})
    
    result = minimize(
        objective, x0, method='SLSQP', bounds=bounds,
        constraints=constraints, options={'maxiter': max_iter, 'ftol': 1e-14, 'disp': False}
    )
    
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        centers[i, 0] = result.x[3 * i]
        centers[i, 1] = result.x[3 * i + 1]
        radii[i] = result.x[3 * i + 2]
    
    return centers, radii, -result.fun


def run_packing():
    n = 26
