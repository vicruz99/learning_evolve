def compute_penalty(centers, radii):
    """Compute total constraint violation penalty"""
    n = centers.shape[0]
    penalty = 0.0
    
    # Overlap penalties between all pairs
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = math.sqrt(dx * dx + dy * dy)
            r_sum = radii[i] + radii[j]
            violation = r_sum - dist
            if violation > 0:
                penalty += violation * violation
    
    # Boundary penalties
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < 0:
            penalty += (r - x) * (r - x)
        if x + r > 1:
            penalty += (x + r - 1) * (x + r - 1)
        if y - r < 0:
            penalty += (r - y) * (r - y)
        if y + r > 1:
            penalty += (y + r - 1) * (y + r - 1)
    
    return penalty


def objective_wrapper(x, n):
    """Objective function: maximize sum of radii via penalty method"""
    centers = x[:2 * n].reshape(n, 2)
    radii = x[2 * n:]
    
    # Clip radii to be non-negative
    radii = np.maximum(radii, 0.0)
    
    obj = -np.sum(radii)
    penalty = compute_penalty(centers, radii)
    obj += 500.0 * penalty
    
    return obj

        bounds.append((0.0, 1.0))
    for _ in range(n):
        bounds.append((0.0, 0.5))
    
    try:
        result = minimize(
            objective_wrapper,
            x0,
            args=(n,),
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': max_iterations, 'ftol': 1e-15, 'maxfun': 15000}
        )
        return result
    except Exception:
        return None


def validate_packing(centers, radii):
    """Validate packing"""
    n = centers.shape[0]
    
    if np.isnan(centers).any():
        return False
    if np.isnan(radii).any():
        return False
    
    for i in range(n):
        if radii[i] < 0:
            return False
        elif np.isnan(radii[i]):
            return False
    
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            return False
    
    for i in range(n):
        for j in range(i + 1, n):
            dist = math.sqrt((centers[i, 0] - centers[j, 0])**2 + (centers[i, 1] - centers[j, 1])**2)
            if dist < radii[i] + radii[j] - 1e-12:
                return False
    
    return True


def shrink_to_valid(centers, radii):
    """Shrink circles slightly to ensure validity"""
    n = centers.shape[0]
    radii = np.copy(radii)
    centers = np.copy(centers)
    
    for i in range(n):
        r = radii[i]
        centers[i, 0] = max(r, min(1.0 - r, centers[i, 0]))
        centers[i, 1] = max(r, min(1.0 - r, centers[i, 1]))
    
    for iteration in range(200):
        max_violation = 0
        for i in range(n):
            for j in range(i + 1, n):
                dist = math.sqrt((centers[i, 0] - centers[j, 0])**2 + (centers[i, 1] - centers[j, 1])**2)
                r_sum = radii[i] + radii[j]
                if dist < r_sum - 1e-10:
                    violation = r_sum - dist
                    max_violation = max(max_violation, violation)
                    if radii[i] < radii[j]:
                        radii[i] = max(0, radii[i] - violation * 0.5)
                    else:
                        radii[j] = max(0, radii[j] - violation * 0.5)
        
        if max_violation < 1e-10:
            break
    
    return centers, radii


def local_refinement(centers, radii, n):
    """Perform local refinement"""
    centers = np.copy(centers)
    radii = np.copy(radii)
    
    for iteration in range(100):
        x0 = centers.flatten()
        r_fixed = radii.copy()
        
        def pos_obj(x):
            c = x.reshape(n, 2)
            penalty = 0.0
            for i in range(n):
                for j in range(i + 1, n):
                    dist = math.sqrt((c[i,0]-c[j,0])**2 + (c[i,1]-c[j,1])**2)
                    r_sum = r_fixed[i] + r_fixed[j]
                    if dist < r_sum:
                        penalty += (r_sum - dist) ** 2
            for i in range(n):
                r = r_fixed[i]
                if c[i,0] < r:
                    penalty += (r - c[i,0]) ** 2
                if c[i,0] > 1.0 - r:
                    penalty += (c[i,0] - (1.0 - r)) ** 2
                if c[i,1] < r:
                    penalty += (r - c[i,1]) ** 2
                if c[i,1] > 1.0 - r:
                    penalty += (c[i,1] - (1.0 - r)) ** 2
            return penalty
        
        bounds = [(0, 1)] * (2 * n)
        try:
            result = minimize(pos_obj, x0, method='L-BFGS-B', bounds=bounds, 
                           options={'maxiter': 1000, 'ftol': 1e-18})
            centers = result.x.reshape(n, 2)
        except:
            pass
        
        can_increase = True
        for i in range(n):
            for j in range(i + 1, n):
                dist = math.sqrt((centers[i,0]-centers[j,0])**2 + (centers[i,1]-centers[j,1])**2)
                if dist < radii[i] + radii[j] + 1e-8:
                    can_increase = False
                    break
            if not can_increase:
                break
        
        for i in range(n):
            if centers[i,0] - radii[i] < 1e-8 or centers[i,0] + radii[i] > 1.0 - 1e-8:
                can_increase = False
            if centers[i,1] - radii[i] < 1e-8 or centers[i,1] + radii[i] > 1.0 - 1e-8:
                can_increase = False
        
        if can_increase:
            radii *= 1.002
        else:
            break
    
    return centers, radii


def create_optimal_hexagonal(n):
    """Create optimal hexagonal packing"""
    centers = np.zeros((n, 2))
    rows = [5, 6, 5, 6, 4]
    n_rows = len(rows)
    r = 0.095
    
    idx = 0
    for row_i, nc in enumerate(rows):
        y = r + row_i * (1.0 - 2.0 * r) / (n_rows - 1)
        offset = r if row_i % 2 == 1 else 0.0
        
        for j in range(nc):
            x = r + offset + j * 2.0 * r
            if x > 1.0 - r:
                available = 1.0 - 2.0 * r
                actual_width = offset + (nc - 1) * 2.0 * r
                if actual_width > 0.001:
                    scale = available / actual_width
                    x = r + (offset + j * 2.0 * r) * scale
            centers[idx] = [x, y]
            idx += 1
    
    radii = np.full(n, r)
    return centers, radii


def run_packing():
    """Main packing function"""
    n = 26
    best_centers = None
    best_radii = None
    best_sum = 0.0
    
    initial_configs = []
    
    c, r = create_hexagonal_56564(n)
    initial_configs.append((c, r))
    
    c, r = create_hexagonal_65654(n)
    initial_configs.append((c, r))
    
    c, r = create_hexagonal_5row_equal(n)
    initial_configs.append((c, r))
    
    c, r = create_grid_perturbed(n)
    initial_configs.append((c, r))
    
    for seed in [42, 123, 456, 789]:
        c, r = create_random_config(n, seed)
        initial_configs.append((c, r))
    
    c, r = create_4row_config(n)
    initial_configs.append((c, r))
    
    c, r = create_6row_config(n)
    initial_configs.append((c, r))
    
    for centers0, radii0 in initial_configs:
        result = run_single_optimization(centers0, radii0, n, max_iterations=10000)
        
        if result is not None:
            x_opt = result.x
            centers_opt = x_opt[:2*n].reshape(n, 2)
            radii_opt = x_opt[2*n:]
            radii_opt = np.maximum(radii_opt, 0.0)
            
            centers_opt, radii_opt = shrink_to_valid(centers_opt, radii_opt)
            
            if validate_packing(centers_opt, radii_opt):
                s = np.sum(radii_opt)
                if s > best_sum:
                    best_sum = s
                    best_centers = centers_opt.copy()
                    best_radii = radii_opt.copy()
    
    if best_centers is not None:
        best_centers, best_radii = local_refinement(best_centers, best_radii, n)
        best_sum = np.sum(best_radii)
    
    if best_centers is None or best_sum < 2.0:
        best_centers, best_radii = create_optimal_hexagonal(n)
        best_sum = np.sum(best_radii)
    
    return best_centers, best_radii, best_sum
