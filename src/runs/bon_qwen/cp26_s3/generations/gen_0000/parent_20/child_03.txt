To pack 26 circles in a unit square and maximize the sum of radii, we address the geometric constraints and packing density. A standard $5 \times 5$ grid of equal circles allows a radius of $r = 0.1$, yielding a sum of $2.5$. To reach the target of $2.636$, we need an average radius of approximately $0.1014$. 

The strategy involves:
1.  **Hexagonal Initialization**: Generating a hexagonal lattice configuration, which is denser than a square grid, to serve as the starting point for optimization.
2.  **Penalty-Based Optimization**: Using `scipy.optimize.minimize` with a Nelder-Mead algorithm to optimize the coordinates and radii. The objective function maximizes the sum of radii while penalizing violations of the unit square boundaries and inter-circle overlap.
3.  **Refinement**: Applying a secondary refinement phase with an L-BFGS-B solver to fine-tune the configuration and ensure strict adherence to boundary constraints.
4.  **Iterative Improvement**: Running the optimization with multiple initial perturbations to escape local minima and select the configuration with the highest sum of radii.

import numpy as np
from scipy.optimize import minimize
import math

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    n = centers.shape[0]

    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True

def generate_hexagonal_lattice(n, seed=42):
    """Generates an initial configuration of n circles in a hexagonal pattern."""
    rng = np.random.default_rng(seed)
    centers = []
    
    # Estimate initial spacing
    # For n=26, a 5x5-ish grid is expected. Spacing ~ 0.2
    s = 0.2 
    y_offset = s * math.sqrt(3) / 2
    
    i = 0
    row = 0
    while i < n:
        x_offset = (s / 2) if row % 2 != 0 else 0
        # Number of circles in this row
        # Adjust width to fit in [0, 1]
        max_circles_x = math.floor(1 / s) + 1
        
        for col in range(max_circles_x):
            if i < n:
                x = x_offset + col * s
                y = row * y_offset
                
                # Add small random perturbation to break symmetry and help optimization
                x += rng.uniform(-0.02, 0.02)
                y += rng.uniform(-0.02, 0.02)
                
                # Clamp to valid range [0, 1] with margin
                x = np.clip(x, 0.05, 0.95)
                y = np.clip(y, 0.05, 0.95)
                
                centers.append([x, y])
                i += 1
            else:
                break
        row += 1
        if len(centers) > 0 and centers[-1][1] > 0.9:
            break
            
    centers = np.array(centers[:n])
    # Initial small radii to ensure feasibility
    radii = np.full(n, 0.01)
    return centers, radii

def objective_function(params, n_circles):
    """Objective function to maximize sum of radii with penalties for constraints."""
    # params: [x0, y0, r0, x1, y1, r1, ...]
    # Reshape to (n, 3)
    data = params.reshape(-1, 3)
    centers = data[:, :2]
    radii = data[:, 2]
    
    # Objective: Minimize negative sum of radii
    obj = -np.sum(radii)
    
    # Penalty weight
    W = 1000.0
    
    # Boundary penalties
    # Circle must be inside [0, 1]
    # x - r >= 0 => r - x <= 0
    # x + r <= 1 => x + r - 1 <= 0
    # y - r >= 0
    # y + r <= 1
    
    for i in range(n_circles):
        x, y, r = centers[i, 0], centers[i, 1], radii[i]
        
        # Violations
        v1 = max(0, r - x)
        v2 = max(0, x + r - 1)
        v3 = max(0, r - y)
        v4 = max(0, y + r - 1)
        
        obj += W * (v1**2 + v2**2 + v3**2 + v4**2)
        
    # Overlap penalties
    for i in range(n_circles):
        for j in range(i + 1, n_circles):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = math.sqrt(dx*dx + dy*dy)
            r_sum = radii[i] + radii[j]
            
            if dist < r_sum:
                overlap = r_sum - dist
                obj += W * overlap**2
                
    return obj

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n_circles = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Try multiple seeds to find a better local optimum
    seeds = [42, 123, 456, 789, 1001]
    
    for seed in seeds:
        # 1. Generate initial configuration
        centers, radii = generate_hexagonal_lattice(n_circles, seed=seed)
        
        # Flatten parameters: x, y, r for each circle
        # Interleaved structure for easier reshaping in objective
        # But scipy works better with flat array. Let's use [x0, y0, r0, x1, y1, r1...]
        initial_params = np.zeros(n_circles * 3)
        for i in range(n_circles):
            initial_params[3*i] = centers[i, 0]
            initial_params[3*i+1] = centers[i, 1]
            initial_params[3*i+2] = radii[i]
            
        # 2. Optimization with Nelder-Mead (handles non-smooth penalties well)
        # Maximize sum of radii => Minimize -sum(radii)
        result_nm = minimize(
            objective_function, 
            initial_params, 
            args=(n_circles,), 
            method='Nelder-Mead', 
            options={'maxiter': 2000, 'xatol': 1e-6, 'fatol': 1e-6}
        )
        
        # 3. Refine with L-BFGS-B (better convergence, uses gradients)
        # We need to clamp variables to reasonable bounds to help
        bounds = []
        for _ in range(n_circles):
            bounds.extend([(0, 1), (0, 1), (0, 0.5)]) # x, y in [0,1], r in [0, 0.5]
            
        result_lbfgs = minimize(
            objective_function,
            result_nm.x,
            args=(n_circles,),
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 1000, 'ftol': 1e-8}
        )
        
        # Extract results
        params = result_lbfgs.x
        curr_centers = params.reshape(-1, 3)[:, :2]
        curr_radii = params.reshape(-1, 3)[:, 2]
        
        # Calculate sum of radii
        curr_sum = np.sum(curr_radii)
        
        # Validate and keep best
        # We need to ensure strict validity for the final return, but validation is slow.
        # The penalty function should have minimized violations.
        # Let's do a quick check for validity roughly
        valid = True
        for i in range(n_circles):
            x, y, r = curr_centers[i, 0], curr_centers[i, 1], curr_radii[i]
            if r < 0: valid = False
            if x < r or x > 1-r or y < r or y > 1-y: valid = False # Approx check
            
        # Check overlaps roughly
        for i in range(n_circles):
            for j in range(i+1, n_circles):
                dist = np.linalg.norm(curr_centers[i] - curr_centers[j])
                if dist < curr_radii[i] + curr_radii[j] - 1e-9:
                    valid = False
        
        if valid and curr_sum > best_sum:
            best_sum = curr_sum
            best_centers = curr_centers.copy()
            best_radii = curr_radii.copy()
            
    return best_centers, best_radii, best_sum

# Function to be called by the validator environment
if __name__ == "__main__":
    centers, radii, sum_radii = run_packing()
    
    # Final validation check
    if validate_packing(centers, radii):
        print(f"Valid packing found. Sum of radii: {sum_radii}")
    else:
        print("Packing validation failed.")
