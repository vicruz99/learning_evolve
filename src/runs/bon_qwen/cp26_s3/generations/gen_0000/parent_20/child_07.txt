import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    """
    n = 26
    
    # --- Helper Functions for Optimization ---

    def compute_distances(centers):
        """Compute pairwise distances matrix for given centers."""
        diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
        return np.sqrt(np.sum(diffs ** 2, axis=2))

    def get_objective_and_jac(vars_flat):
        """Objective: Maximize sum of radii (minimize negative sum)."""
        centers = vars_flat[:2 * n].reshape(n, 2)
        radii = vars_flat[2 * n:]
        
        obj = -np.sum(radii)
        
        # Gradient of objective: 0 for centers, -1 for radii
        grad = np.zeros_like(vars_flat)
        grad[2 * n:] = -1.0
        
        return obj, grad

    def get_constraint_func(vars_flat, grad=False):
        """
        Returns constraint values and optionally their Jacobians.
        Constraints:
        1. dist(i, j) >= r_i + r_j  => dist - r_i - r_j >= 0
        2. x_i >= r_i               => x_i - r_i >= 0
        3. x_i <= 1 - r_i           => 1 - x_i - r_i >= 0
        4. y_i >= r_i               => y_i - r_i >= 0
        5. y_i <= 1 - r_i           => 1 - y_i - r_i >= 0
        """
        centers = vars_flat[:2 * n].reshape(n, 2)
        radii = vars_flat[2 * n:]
        
        dists = compute_distances(centers)
        
        # Initialize constraints array
        # Pairwise constraints: n*(n-1)/2
        # Boundary constraints: 4 * n
        n_pairs = n * (n - 1) // 2
        n_bound = 4 * n
        con_vals = np.zeros(n_pairs + n_bound)
        
        idx = 0
        # Pairwise constraints
        for i in range(n):
            for j in range(i + 1, n):
                con_vals[idx] = dists[i, j] - radii[i] - radii[j]
                idx += 1
        
        # Boundary constraints
        x = centers[:, 0]
        y = centers[:, 1]
        con_vals[idx:idx+n] = x - radii
        con_vals[idx+n:idx+2*n] = 1 - x - radii
        con_vals[idx+2*n:idx+3*n] = y - radii
        con_vals[idx+3*n:idx+4*n] = 1 - y - radii
        
        if not grad:
            return con_vals

        # Compute Jacobian if requested
        jac = np.zeros((n_pairs + n_bound, 3 * n))
        
        # Pairwise constraints Jacobian
        # Variables order: x0, y0, r0, x1, y1, r1, ...
        idx = 0
        for i in range(n):
            for j in range(i + 1, n):
                d = dists[i, j]
                # Avoid division by zero
                if d > 1e-12:
                    dx = (centers[i, 0] - centers[j, 0]) / d
                    dy = (centers[i, 1] - centers[j, 1]) / d
                    
                    # Derivatives w.r.t xi, yi
                    jac[idx, 3*i] = dx
                    jac[idx, 3*i + 1] = dy
                    
                    # Derivatives w.r.t xj, yj
                    jac[idx, 3*j] = -dx
                    jac[idx, 3*j + 1] = -dy
                
                # Derivatives w.r.t ri, rj
                jac[idx, 3*i + 2] = -1.0
                jac[idx, 3*j + 2] = -1.0
                
                idx += 1
        
        # Boundary constraints Jacobian
        # x - r >= 0
        for k in range(n):
            row = idx + k
            jac[row, 3*k] = 1.0   # dx
            jac[row, 3*k + 2] = -1.0 # dr
        
        # 1 - x - r >= 0
        for k in range(n):
            row = idx + n + k
            jac[row, 3*k] = -1.0  # dx
            jac[row, 3*k + 2] = -1.0 # dr
            
        # y - r >= 0
        for k in range(n):
            row = idx + 2*n + k
            jac[row, 3*k + 1] = 1.0   # dy
            jac[row, 3*k + 2] = -1.0  # dr
            
        # 1 - y - r >= 0
        for k in range(n):
            row = idx + 3*n + k
            jac[row, 3*k + 1] = -1.0  # dy
            jac[row, 3*k + 2] = -1.0  # dr
            
        return con_vals, jac

    # --- Initialization ---
    # Use a hexagonal packing pattern to start close to a dense configuration
    centers_init = np.zeros((n, 2))
    radii_init = np.full(n, 0.09) # Start with a safe radius
    
    # Generate hexagonal grid points
    row_height = np.sqrt(3) / 2 * 2 * radii_init[0]
    count = 0
    row = 0
    y = radii_init[0]
    
    while count < n:
        x = radii_init[0]
        if row % 2 == 1:
            x += radii_init[0] # Shift for odd rows
            
        col = 0
        while x <= 1 - radii_init[0] and count < n:
            centers_init[count, 0] = x
            centers_init[count, 1] = y
            count += 1
            x += 2 * radii_init[0]
            col += 1
        y += row_height
        row += 1

    # Flatten variables: [x0, y0, r0, x1, y1, r1, ...]
    x0 = np.concatenate([centers_init.ravel(), radii_init])
    
    # Bounds: x, y in [0, 1], r in [0, 0.5]
    bounds = [(0, 1)] * (2 * n) + [(1e-6, 0.5)] * n
    
    # Define constraints object for scipy
    cons = {'type': 'ineq', 'fun': get_constraint_func, 'jac': True}
    
    # Optimization
    result = minimize(
        get_objective_and_jac, 
        x0, 
        method='SLSQP', 
        bounds=bounds, 
        constraints=cons, 
        options={'maxiter': 2000, 'ftol': 1e-9}
    )
    
    # Extract results
    centers_opt = result.x[:2 * n].reshape(n, 2)
    radii_opt = result.x[2 * n:]
    
    # Clip small negative radii due to numerical precision
    radii_opt = np.maximum(radii_opt, 0.0)
    
    sum_radii = np.sum(radii_opt)
    
    return centers_opt, radii_opt, sum_radii
