2. **Mathematical Formulation**: Frame the problem as a Nonlinear Programming (NLP) task. 
   - **Objective**: Maximize $\sum_{i=1}^{26} r_i$.
   - **Constraints**: 
     - Boundary: $r_i \le x_i \le 1-r_i$ and $r_i \le y_i \le 1-r_i$ for all $i$.
     - Non-overlap: $\|c_i - c_j\| \ge r_i + r_j$ for all $i < j$.
3. **Optimization Algorithm**: Use `scipy.optimize.minimize` with the `SLSQP` method, which is robust for problems with nonlinear inequality constraints. Variables include all centers and radii simultaneously, allowing the optimizer to adjust sizes and positions cooperatively.
4. **Robustness**: Run the optimization from 5 different perturbed initial states. This mitigates the risk of converging to poor local optima, which is common in non-convex geometric packing problems.

import numpy as np
from scipy.optimize import minimize

def get_initial_config(n):
    """Generate an initial hexagonal lattice configuration for n circles."""
    pts = []
    row = 0
    while len(pts) < n:
        y = row * np.sqrt(3)/2
        x_start = (row % 2) * 0.5
        col = 0
        while len(pts) < n:
            pts.append([x_start + col, y])
            col += 1
        row += 1
    pts = np.array(pts[:n])
    # Normalize to unit square with margin to ensure initial feasibility
    pts -= pts.min(0)
    pts /= pts.max(0)
    pts = pts * 0.6 + 0.2
    r_init = np.full(n, 0.08)
    return np.concatenate([pts.flatten(), r_init])

def obj_func(vars):
    """Objective: maximize sum of radii (minimize negative sum)."""
    n = 26
    return -np.sum(vars[2*n:])

def con_func(vars):
    """Inequality constraints: boundaries and non-overlap."""
    n = 26
    X = vars[:2*n].reshape(n, 2)
    R = vars[2*n:]
    cons = []
    # Boundary constraints: r <= x <= 1-r and r <= y <= 1-r
    cons.append(X[:, 0] - R)
    cons.append(1 - R - X[:, 0])
    cons.append(X[:, 1] - R)
    cons.append(1 - R - X[:, 1])
    
    # Pairwise non-overlap: dist_sq >= (r_i + r_j)^2
    diff = X[:, np.newaxis, :] - X[np.newaxis, :, :]
    dist_sq = np.sum(diff**2, axis=2)
    rad_sum = R[:, np.newaxis] + R[np.newaxis, :]
    pairwise = dist_sq - rad_sum**2
    
    # Flatten upper triangle to avoid duplicates and self-comparisons
    idx = np.triu_indices(n, k=1)
    cons.append(pairwise[idx])
    
    return np.concatenate(cons)

def run_packing():
    n = 26
    best_centers = None
    best_radii = None
    best_sum = -1.0
    
    # Multiple restarts to escape local optima and ensure robust convergence
    for seed in range(5):
        np.random.seed(seed)
        x0 = get_initial_config(n)
        x0 += np.random.normal(0, 0.02, size=x0.shape)
        x0 = np.clip(x0, 0.05, 0.95)
        
        bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n
        cons = {'type': 'ineq', 'fun': con_func}
        
        try:
            res = minimize(obj_func, x0, method='SLSQP', bounds=bounds, constraints=cons, 
                           options={'maxiter': 3000, 'ftol': 1e-9, 'disp': False})
            
            centers = res.x[:2*n].reshape(n, 2)
            radii = res.x[2*n:]
            curr_sum = np.sum(radii)
            
            if curr_sum > best_sum:
                best_centers = centers
                best_radii = radii
                best_sum = curr_sum
        except Exception:
            pass
            
    return best_centers, best_radii, best_sum
