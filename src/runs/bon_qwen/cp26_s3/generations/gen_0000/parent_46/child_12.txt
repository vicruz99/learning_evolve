import numpy as np
from scipy.optimize import minimize

def compute_violation(params, n):
    """Compute squared violation of boundary and pairwise constraints."""
    centers = params[:2*n].reshape(n, 2)
    radii = params[2*n:]
    
    # Boundary distances: left, right, bottom, top
    left = centers[:, 0]
    right = 1.0 - centers[:, 0]
    bottom = centers[:, 1]
    top = 1.0 - centers[:, 1]
    b_dists = np.column_stack([left, right, bottom, top])
    
    # Penalty: max(0, r - dist)^2
    diff_b = radii[:, np.newaxis] - b_dists
    pen_b = np.sum(np.maximum(diff_b, 0.0)**2)
    
    # Pairwise distances
    c1 = centers[:, np.newaxis, :]
    c2 = centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum((c1 - c2)**2, axis=2))
    
    r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    diff_p = r_sum - dists
    
    # Upper triangle mask to count each pair once
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    pen_p = np.sum(np.maximum(diff_p, 0.0)**2 * mask)
    
    return pen_b + pen_p

def phase1_obj(p, r_fixed, n):
    """Objective for Phase 1: minimize overlap with fixed radii."""
    params = np.concatenate([p, r_fixed])
    return compute_violation(params, n)

def objective(p, n, W):
    """Objective for Phase 2: maximize sum of radii with penalty for violations."""
    viol = compute_violation(p, n)
    return -np.sum(p[2*n:]) + W * viol

def get_hex_seed(n):
    """Generate a hexagonal lattice seed configuration."""
    pts = []
    y = 0.15
    while len(pts) < n:
        x = 0.15
        offset = (len(pts) % 2) * 0.08
        while x + 0.1 < 0.95 and len(pts) < n:
            pts.append([x + offset, y])
            x += 0.18
        y += 0.15
    return np.array(pts[:n])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    n = 26
    best_params = None
    best_val = -np.inf
    W = 15000.0  # Penalty weight
    
    # Prepare diverse initial seeds
    seeds = []
    for _ in range(6):
        centers = np.random.rand(n, 2) * 0.6 + 0.2
        seeds.append(centers)
    seeds.append(get_hex_seed(n))
    
    for centers_init in seeds:
        radii_init = np.full(n, 0.04)
        
        # Phase 1: Spread centers to eliminate initial overlaps
        res1 = minimize(phase1_obj, centers_init.flatten(), args=(radii_init, n), 
                        method='L-BFGS-B', bounds=[(0.0, 1.0)] * (2*n), 
                        options={'maxiter': 2000, 'ftol': 1e-10})
        centers_opt = res1.x.reshape(n, 2)
        
        # Phase 2: Optimize both centers and radii
        x0 = np.concatenate([centers_opt.flatten(), radii_init])
        bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n
        res2 = minimize(objective, x0, args=(n, W), method='L-BFGS-B',
                        bounds=bounds, options={'maxiter': 5000, 'ftol': 1e-12})
        
        current_sum = np.sum(res2.x[2*n:])
        viol = compute_violation(res2.x, n)
        
        # Accept if sufficiently feasible and better
        if viol < 1e-8 and current_sum > best_val:
            best_val = current_sum
            best_params = res2.x
            
    # Fallback safety (should not trigger with proper optimization)
    if best_params is None:
        centers = np.random.rand(n, 2)
        radii = np.full(n, 0.01)
        best_params = np.concatenate([centers.flatten(), radii])
        
    final_centers = best_params[:2*n].reshape(n, 2)
    final_radii = best_params[2*n:]
    
    return final_centers, final_radii, float(np.sum(final_radii))
