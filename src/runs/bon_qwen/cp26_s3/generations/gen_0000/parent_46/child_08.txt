import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # 1. Initialize centers in a hexagonal grid pattern
    # This provides a highly efficient starting configuration for packing.
    centers = np.zeros((n, 2))
    idx = 0
    
    # Create a hexagonal grid layout.
    # Rows of 5 and 4 circles to accommodate 26 circles.
    # We shift odd rows by r horizontally (approx 0.1) to nest them.
    row_pattern = [5, 4, 5, 4, 5, 3] # 5+4+5+4+5+3 = 26
    
    # Approximate row spacing for hex packing: sqrt(3) * 2r. 
    # Using a base spacing of 0.2 for now to separate them.
    dy = 0.25 
    dx = 0.2
    y = 0.125 # Initial y offset
    
    for r_idx, count in enumerate(row_pattern):
        x_start = 0.125
        if r_idx % 2 == 1: # Staggered row
            x_start += 0.1
        x = x_start
        for _ in range(count):
            if idx < n:
                centers[idx, 0] = x
                centers[idx, 1] = y
                idx += 1
            x += dx
        y += dy

    # Flatten centers for optimizer
    x0_centers = centers.flatten()
    # Initialize radii small to ensure no initial violations
    x0_radii = np.full(n, 0.05)
    x0 = np.concatenate([x0_centers, x0_radii])
    
    # 2. Define Objective and Penalty
    def objective(vars_vec):
        centers = vars_vec[:2*n].reshape((n, 2))
        radii = vars_vec[2*n:]
        
        # Negative sum of radii (minimizing -sum is maximizing sum)
        obj = -np.sum(radii)
        penalty = 0.0
        lambda_p = 1e5 # Penalty weight
        
        # Boundary penalties: Circle must be inside [0,1]x[0,1]
        # Constraint: r <= x <= 1-r  =>  x-r >= 0 and 1-(x+r) >= 0
        # Penalty for x-r < 0
        for i in range(n):
            xi, yi = centers[i]
            ri = radii[i]
            
            # Left boundary
            val = xi - ri
            if val < 0: penalty += lambda_p * (val ** 2)
            # Right boundary
            val = 1.0 - (xi + ri)
            if val < 0: penalty += lambda_p * (val ** 2)
            # Bottom boundary
            val = yi - ri
            if val < 0: penalty += lambda_p * (val ** 2)
            # Top boundary
            val = 1.0 - (yi + ri)
            if val < 0: penalty += lambda_p * (val ** 2)
            
        # Overlap penalties: dist >= r_i + r_j
        # We check squared distance to avoid sqrt in loop, but for penalty logic we use linear gap
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                val = dist - (radii[i] + radii[j])
                if val < 0:
                    penalty += lambda_p * (val ** 2)
                    
        return obj + penalty

    # 3. Bounds for optimization
    # x, y in [0, 1], r in [0, 0.5]
    bnds = [(0, 1)] * (2 * n) + [(0, 0.5)] * n

    # 4. Run Optimization
    result = minimize(objective, x0, method='L-BFGS-B', bounds=bnds, 
                      options={'ftol': 1e-9, 'maxiter': 2000})
    
    # 5. Extract and Finalize
    best_centers = result.x[:2*n].reshape((n, 2))
    best_radii = result.x[2*n:]
    
    # Safety margin adjustment to strictly satisfy 1e-12 tolerance
    # Check for any tight overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((best_centers[i] - best_centers[j]) ** 2))
            sum_r = best_radii[i] + best_radii[j]
            if dist < sum_r:
                # Reduce radii slightly to resolve overlap
                factor = dist / sum_r
                best_radii[i] *= factor
                best_radii[j] *= factor

    # Ensure boundary constraints with a tiny buffer
    for i in range(n):
        x, y = best_centers[i]
        r = best_radii[i]
        # Adjust if touching boundary
        min_dist = min(x, 1-x, y, 1-y)
        if r > min_dist:
            best_radii[i] = min_dist

    total_sum = np.sum(best_radii)
    return best_centers, best_radii, total_sum

if __name__ == "__main__":
    # Run the packing
    centers, radii, sum_r = run_packing()

    def validate_packing(centers, radii):

        Validate that circles don't overlap and are inside the unit square
