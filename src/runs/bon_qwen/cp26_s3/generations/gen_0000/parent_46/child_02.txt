- **Objective**: Maximize $\sum r_i$ (minimize $-\sum r_i$).
- **Constraints**: 
  1. Boundary: $r_i \le x_i \le 1-r_i$ and $r_i \le y_i \le 1-r_i$
  2. Non-overlap: $\sqrt{(x_i-x_j)^2 + (y_i-y_j)^2} \ge r_i + r_j$ for all $i<j$

import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def get_initial_guess():
    """Generate a feasible initial configuration with small random perturbations."""
    x0 = np.zeros(3 * N_CIRCLES)
    cols = 6
    rows = 5
    xs = np.linspace(0.15, 0.85, cols)
    ys = np.linspace(0.15, 0.85, rows)
    
    idx = 0
    for i in range(N_CIRCLES):
        r_idx = i // cols
        c_idx = i % cols
        if r_idx >= rows:
            break
        x0[idx] = xs[c_idx]
        x0[idx+1] = ys[r_idx]
        x0[idx+2] = 0.05
        idx += 1
        
    # Add controlled noise to break symmetry and explore different local optima
    x0[:2*N_CIRCLES] += np.random.uniform(-0.02, 0.02, size=2*N_CIRCLES)
    x0[:2*N_CIRCLES] = np.clip(x0[:2*N_CIRCLES], 0.05, 0.95)
    x0[2*N_CIRCLES:] = 0.05
    return x0

def objective(vars):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(vars[2*N_CIRCLES:])

def constraints(vars):
    """Constraint function: boundary and non-overlap inequalities (>= 0)."""
    centers = vars[:2*N_CIRCLES].reshape(N_CIRCLES, 2)
    radii = vars[2*N_CIRCLES:]
    n_con = 4*N_CIRCLES + N_CIRCLES*(N_CIRCLES-1)//2
    c_arr = np.zeros(n_con)
    
    idx = 0
    # Left, Right, Bottom, Top boundaries
    c_arr[idx:idx+N_CIRCLES] = centers[:, 0] - radii
    idx += N_CIRCLES
    c_arr[idx:idx+N_CIRCLES] = 1.0 - centers[:, 0] - radii
    idx += N_CIRCLES
    c_arr[idx:idx+N_CIRCLES] = centers[:, 1] - radii
    idx += N_CIRCLES
    c_arr[idx:idx+N_CIRCLES] = 1.0 - centers[:, 1] - radii
    idx += N_CIRCLES
    
    # Pairwise non-overlap
    for i in range(N_CIRCLES):
        for j in range(i + 1, N_CIRCLES):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx*dx + dy*dy)
            c_arr[idx] = dist - radii[i] - radii[j]
            idx += 1
    return c_arr

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Run the optimization to pack 26 circles and return centers, radii, and sum of radii."""
    np.random.seed(42)
    bounds = [(0, 1)]*N_CIRCLES + [(0, 1)]*N_CIRCLES + [(0, 0.5)]*N_CIRCLES
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_x = None
    best_val = -np.inf
    
    # Multiple restarts to find a better local optimum
    for _ in range(5):
        x0 = get_initial_guess()
        res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 2000, 'ftol': 1e-10, 'disp': False})
        if res.fun < best_val:
            best_val = res.fun
            best_x = res.x
            
    centers = best_x[:2*N_CIRCLES].reshape(N_CIRCLES, 2)
    radii = best_x[2*N_CIRCLES:]
    
    # Apply tiny uniform scaling to guarantee strict feasibility within validator tolerance
    # This handles floating-point accumulation errors without meaningfully reducing the objective
    final_radii = radii * 0.9999
    return centers, final_radii, float(np.sum(final_radii))
