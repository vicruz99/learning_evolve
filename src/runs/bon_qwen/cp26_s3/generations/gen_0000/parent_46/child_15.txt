import numpy as np
from scipy.optimize import minimize

def run_packing():
    """
    Pack 26 circles in a unit square to maximize the sum of radii.
    """
    n = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Helper to compute max valid radii given centers
    def update_radii(centers, radii):
        new_radii = radii.copy()
        for i in range(n):
            # Distance to walls
            max_r = min(centers[i, 0], 1.0 - centers[i, 0], 
                        centers[i, 1], 1.0 - centers[i, 1])
            # Distance to other circles
            for j in range(n):
                if i != j:
                    d = np.sqrt(np.sum((centers[i] - centers[j])**2))
                    max_r = min(max_r, d - radii[j])
            new_radii[i] = max_r
        return new_radii

    # Objective for position optimization: minimize overlap penalty
    def position_penalty(c_flat, radii):
        c = c_flat.reshape(n, 2)
        pen = 0.0
        for i in range(n):
            for j in range(i + 1, n):
                d = np.sqrt(np.sum((c[i] - c[j])**2))
                req = radii[i] + radii[j] + 1e-6
                if d < req:
                    pen += (req - d) ** 2
            # Wall penalties
            for k in range(2):
                if c[i, k] < radii[i] + 1e-6:
                    pen += (radii[i] - c[i, k]) ** 2
                if c[i, k] > 1.0 - radii[i] - 1e-6:
                    pen += (radii[i] - (1.0 - c[i, k])) ** 2
        return pen

    # Try multiple initial configurations
    for seed in range(8):
        np.random.seed(seed)
        
        # Initialize with a hexagonal-like pattern
        centers = []
        row_counts = [6, 5, 6, 5, 4]  # Sums to 26
        y = 0.12
        for i, cnt in enumerate(row_counts):
            x = 0.10 + (i % 2) * 0.06
            for _ in range(cnt):
                centers.append([x, y])
                x += 0.14
            y += 0.17
            
        centers = np.array(centers)
        # Add controlled random noise to break symmetry
        centers += np.random.randn(n, 2) * 0.025
        centers = np.clip(centers, 0.05, 0.95)
        
        radii = np.ones(n) * 0.09
        
        # Iterative optimization
        for _ in range(60):
            # 1. Expand radii
            radii = update_radii(centers, radii)
            
            # 2. Optimize positions to relieve congestion
            try:
                res = minimize(position_penalty, centers.flatten(), 
                               args=(radii,), 
                               method='BFGS', 
                               options={'maxiter': 100, 'ftol': 1e-9})
                centers = res.x.reshape(n, 2)
            except:
                pass
                
            # 3. Slight random perturbation to escape local minima
            if _ % 20 == 0:
                centers += np.random.randn(n, 2) * 0.005
                centers = np.clip(centers, 0.01, 0.99)

        # Final radius update after position optimization
        radii = update_radii(centers, radii)
        
        # Validate and record if better
        current_sum = np.sum(radii)
        if current_sum > best_sum:
            # Strict validation pass
            valid = True
            for i in range(n):
                for j in range(i + 1, n):
                    d = np.sqrt(np.sum((centers[i] - centers[j])**2))

                        break

                        valid = False
                        break
                if not valid: break
                
            if valid:
                best_sum = current_sum
                best_centers = centers.copy()
                best_radii = radii.copy()

    # Fallback if all failed (unlikely)
    if best_centers is None:
        return np.zeros((n, 2)), np.ones(n)*0.01, 0.26

    return best_centers, best_radii, best_sum
