import numpy as np
from scipy.optimize import minimize
import math

def run_packing():

    # --- Helper Functions ---

    def get_constraints(centers, radii):
        """Calculate constraint values: [boundary constraints, non-overlap constraints]"""
        c_x, c_y = centers[:, 0], centers[:, 1]
        r = radii
        
        # Boundary constraints: x-r >= 0, 1-(x+r) >= 0, etc.
        # We want these to be >= 0
        bounds = np.array([
            c_x - r,
            1 - (c_x + r),
            c_y - r,
            1 - (c_y + r)
        ])
        
        # Non-overlap constraints: dist^2 - (r_i + r_j)^2 >= 0
        overlaps = []
        for i in range(n):
            for j in range(i + 1, n):
                dx = c_x[i] - c_x[j]
                dy = c_y[i] - c_y[j]
                dist_sq = dx*dx + dy*dy
                r_sum = r[i] + r[j]
                overlaps.append(dist_sq - r_sum*r_sum)
        
        return np.concatenate([bounds.flatten(), overlaps])

    def objective(variables):
        # Maximize sum of radii -> Minimize -sum(radii)
        centers = variables[:n*2].reshape(n, 2)

        return -np.sum(radii)
