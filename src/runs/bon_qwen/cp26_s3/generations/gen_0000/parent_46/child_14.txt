import numpy as np
from scipy.optimize import minimize

def run_packing():
    n = 26
    
    # ==========================================
    # Phase 1: Growing Circles Simulation
    # ==========================================
    np.random.seed(42)
    centers = np.random.rand(n, 2) * 0.8 + 0.1
    radii = np.ones(n) * 0.05
    velocities = np.zeros((n, 2))
    
    dt = 0.03
    damping = 0.90
    repulsion_strength = 45.0
    boundary_strength = 70.0
    
    max_r_target = 0.118
    steps_per_increase = 250
    r_increment = 0.00025
    
    current_r = 0.05
    total_iter = 0
    max_total_iter = 5000
    
    while total_iter < max_total_iter:
        radii[:] = min(current_r, max_r_target)
        
        forces = np.zeros_like(centers)
        
        # Compute repulsive forces between all pairs
        for i in range(n):
            for j in range(i + 1, n):
                diff = centers[i] - centers[j]
                dist = np.linalg.norm(diff)
                req_dist = radii[i] + radii[j]
                
                if dist < req_dist and dist > 1e-9:
                    overlap = req_dist - dist
                    force_mag = repulsion_strength * overlap / dist
                    forces[i] += diff * force_mag
                    forces[j] -= diff * force_mag
                    
        # Boundary forces
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < 0: forces[i, 0] += boundary_strength * (r - x)
            if x + r > 1: forces[i, 0] -= boundary_strength * (x + r - 1)
            if y - r < 0: forces[i, 1] += boundary_strength * (r - y)
            if y + r > 1: forces[i, 1] -= boundary_strength * (y + r - 1)
            
        # Update dynamics
        velocities += forces * dt
        velocities *= damping
        centers += velocities * dt
        
        total_iter += 1
        
        if total_iter % steps_per_increase == 0:
            current_r += r_increment
            
    # ==========================================
    # Phase 2: Local Constrained Optimization
    # ==========================================
    x0 = np.hstack([centers.ravel(), radii])
    
    def objective(x):
        return -np.sum(x[2*n:])
        
    def constraint_margins(x):
        c = x[:2*n].reshape(n, 2)
        r = x[2*n:]
        margins = []
        
        # Boundary constraints: c_i >= r_i, 1 - c_i >= r_i
        for i in range(n):
            margins.append(c[i, 0] - r[i])
            margins.append(1.0 - c[i, 0] - r[i])
            margins.append(c[i, 1] - r[i])
            margins.append(1.0 - c[i, 1] - r[i])
            
        # Pairwise constraints: ||c_i - c_j|| >= r_i + r_j
        for i in range(n):
            for j in range(i + 1, n):
                d = np.linalg.norm(c[i] - c[j])
                margins.append(d - r[i] - r[j])
                
        return np.array(margins)
        
    cons = {'type': 'ineq', 'fun': constraint_margins}
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
    
    res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': 600, 'ftol': 1e-10, 'disp': False})
                   
    if res.success:
        final_centers = res.x[:2*n].reshape(n, 2)
        final_radii = res.x[2*n:]
    else:
        # Fallback to simulation state if optimization fails
        final_centers = centers
        final_radii = radii
        
    # Final safety adjustments
    final_radii = np.maximum(final_radii, 0.0)
    final_centers = np.clip(final_centers, 1e-12, 1.0 - 1e-12)
    
    return final_centers, final_radii, float(np.sum(final_radii))
