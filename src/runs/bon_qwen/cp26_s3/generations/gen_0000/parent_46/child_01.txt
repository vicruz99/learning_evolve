import numpy as np
from scipy.optimize import minimize

def obj_func(v):
    return -np.sum(v[2::3])

def con_bound_x_min(v, i):
    return v[3*i] - v[3*i+2]

def con_bound_x_max(v, i):
    return 1.0 - v[3*i] - v[3*i+2]

def con_bound_y_min(v, i):
    return v[3*i+1] - v[3*i+2]

def con_bound_y_max(v, i):
    return 1.0 - v[3*i+1] - v[3*i+2]

def con_overlap(v, i, j):
    dx = v[3*i] - v[3*j]
    dy = v[3*i+1] - v[3*j+1]
    return np.sqrt(dx*dx + dy*dy) - v[3*i+2] - v[3*j+2]

def run_packing():
    n = 26
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.001, 1.0)] * n
    
    constraints = []
    for i in range(n):
        constraints.append({'type': 'ineq', 'fun': con_bound_x_min, 'args': (i,)})
        constraints.append({'type': 'ineq', 'fun': con_bound_x_max, 'args': (i,)})
        constraints.append({'type': 'ineq', 'fun': con_bound_y_min, 'args': (i,)})
        constraints.append({'type': 'ineq', 'fun': con_bound_y_max, 'args': (i,)})
        
    for i in range(n):
        for j in range(i+1, n):
            constraints.append({'type': 'ineq', 'fun': con_overlap, 'args': (i, j)})
            
    best_sum = -np.inf
    best_v = None
    
    # Structured initial guess (hexagonal-ish grid)
    v0 = np.zeros(3*n)
    idx = 0
    y = 0.12
    while idx < n:
        count = 5 if idx % 2 == 0 else 4
        if idx + count > n:
            count = n - idx
        x_start = 0.12 if idx % 2 == 0 else 0.22
        for k in range(count):
            v0[3*idx] = x_start + k * 0.18
            v0[3*idx+1] = y
            v0[3*idx+2] = 0.06
            idx += 1
        y += 0.15
        
    candidates = [v0]
    for seed in range(5):
        np.random.seed(seed)
        v_rand = np.random.rand(3*n)
        v_rand[::3] *= 0.9
        v_rand[1::3] *= 0.9
        v_rand[2::3] *= 0.08
        candidates.append(v_rand)
        
    for v_init in candidates:
        res = minimize(obj_func, v_init, method='SLSQP', bounds=bounds, 
                       constraints=constraints, options={'maxiter': 5000, 'ftol': 1e-12})
        if -res.fun > best_sum:
            best_sum = -res.fun
            best_v = res.x
            
    # Final polish with tighter tolerance
    res = minimize(obj_func, best_v, method='SLSQP', bounds=bounds, 
                   constraints=constraints, options={'maxiter': 5000, 'ftol': 1e-14})
    best_v = res.x
    best_sum = -res.fun
    
    centers = best_v.reshape(n, 3)[:, :2]
    radii = best_v.reshape(n, 3)[:, 2]
    return centers, radii, best_sum
