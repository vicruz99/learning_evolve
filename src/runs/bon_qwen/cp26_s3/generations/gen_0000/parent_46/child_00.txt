import numpy as np
from scipy.optimize import minimize


def compute_penalty(centers, radii, n, weight):
    """Compute total penalty for constraint violations."""
    penalty = 0.0
    for i in range(n):
        xi, yi = centers[i]
        ri = radii[i]
        for margin in [xi - ri, 1.0 - xi - ri, yi - ri, 1.0 - yi - ri]:
            if margin < 0.0:
                penalty += weight * margin * margin
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy + 1e-15)
            req = radii[i] + radii[j]
            gap = dist - req
            if gap < 0.0:
                penalty += weight * gap * gap
    return penalty


def objective(x, n, weight):
    """Objective: negative sum of radii plus penalty for violations."""
    radii = x[:n]
    centers = x[n:].reshape(n, 2)
    f = -np.sum(radii)
    f += compute_penalty(centers, radii, n, weight)
    return f


def init_config_style1(n):
    """Hexagonal pattern: 6-6-6-6-2 arrangement."""
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.075
    count = 0
    rows = [6, 6, 6, 6, 2]
    spacing = 0.165
    for row_idx, ncols in enumerate(rows):
        row_width = (ncols - 1) * spacing
        start_x = (1.0 - row_width) / 2.0
        y = 0.1 + row_idx * spacing * np.sqrt(3.0) / 2.0
        if row_idx % 2 == 1:
            start_x += spacing / 2.0
        for col in range(ncols):
            if count < n:
                x = start_x + col * spacing
                centers[count] = [x, y]
                count += 1
    return centers, radii

def optimize_run(centers, radii, n, weight, maxiter=20000):
    """Run optimization from given starting point."""
    x0 = np.concatenate([radii, centers.flatten()])
    bounds = [(1e-8, 0.5)] * n + [(0.0, 1.0)] * (2 * n)
    
    result = minimize(
        objective, x0, args=(n, weight), method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': maxiter, 'ftol': 1e-16}
    )
    
    r_opt = np.maximum(result.x[:n], 1e-10)
    c_opt = result.x[n:].reshape(n, 2)
    
    for i in range(n):
        c_opt[i, 0] = np.clip(c_opt[i, 0], r_opt[i], 1.0 - r_opt[i])
        c_opt[i, 1] = np.clip(c_opt[i, 1], r_opt[i], 1.0 - r_opt[i])
    
    return c_opt, r_opt
