import numpy as np
from scipy.optimize import minimize

def get_initial_config(n, method='grid'):
    """Generate initial circle positions and radii."""
    centers = []
    if method == 'grid':
        for i in range(5):
            for j in range(5):
                centers.append([0.1 + 0.2*i, 0.1 + 0.2*j])
        centers.append([0.5, 0.5])
    else:
        s = 0.12
        for row in range(7):
            for col in range(7):
                x = col * s + (row % 2) * s / 2
                y = row * s * np.sqrt(3) / 2
                if 0 <= x <= 1 and 0 <= y <= 1:
                    centers.append([x, y])
            if len(centers) >= n:
                break
    centers = centers[:n]
    radii = [0.08] * n
    return centers, radii

def objective(vars, n, penalty):
    """Objective function: maximize sum of radii with quadratic penalties for violations."""
    c = vars.reshape(n, 3)
    r = c[:, 2]
    pos = c[:, :2]
    
    pen = 0.0
    # Boundary penalties: ensure r <= x, r <= 1-x, r <= y, r <= 1-y
    pen += np.sum(np.maximum(0, r - pos[:, 0])**2)
    pen += np.sum(np.maximum(0, r - (1.0 - pos[:, 0]))**2)
    pen += np.sum(np.maximum(0, r - pos[:, 1])**2)
    pen += np.sum(np.maximum(0, r - (1.0 - pos[:, 1]))**2)
    
    # Overlap penalties: ensure dist >= r_i + r_j
    diffs = pos[:, np.newaxis, :] - pos[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    violations = np.maximum(0, r_sum - dists)
    pen += np.sum(violations[mask]**2)
    
    return -np.sum(r) + penalty * pen

def run_packing():
    """Optimize circle packing to maximize sum of radii."""
    n = 26
    best_sum = -1.0
    best_x = None
    
    # Try multiple initial configurations
    for method in ['grid', 'hex']:
        centers, radii = get_initial_config(n, method)
        x0 = np.zeros(3*n)
        for i in range(n):
            x0[3*i] = centers[i][0]
            x0[3*i+1] = centers[i][1]
            x0[3*i+2] = radii[i]
            
        bounds = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * n
        
        # Stage 1: Explore with lower penalty
        res1 = minimize(objective, x0, args=(n, 100.0), bounds=bounds, method='L-BFGS-B', 
                        options={'maxiter': 2000, 'ftol': 1e-12})
        # Stage 2: Refine with higher penalty to enforce constraints
        res2 = minimize(objective, res1.x, args=(n, 5000.0), bounds=bounds, method='L-BFGS-B',

        final_c = res2.x.reshape(n, 3)
        r_out = final_c[:, 2]
        if np.sum(r_out) > best_sum:
            best_sum = np.sum(r_out)
            best_x = res2.x
            
    final_c = best_x.reshape(n, 3)
    centers_out = final_c[:, :2]
    radii_out = final_c[:, 2]
    
    # Strict boundary enforcement
    for i in range(n):
        x, y, r = centers_out[i, 0], centers_out[i, 1], radii_out[i]
        r = min(r, x, 1.0-x, y, 1.0-y)
        radii_out[i] = r
        
    # Iterative shrink to guarantee non-overlap within validator tolerance
    for _ in range(30):
        changed = False
        for i in range(n):
            for j in range(i+1, n):

                    radii_out[i] *= np.sqrt(scale)
                    radii_out[j] *= np.sqrt(scale)
                    changed = True
        if not changed:
            break
            
    return centers_out, radii_out, np.sum(radii_out)
