import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    """
    n = 26
    
    # 1. Initialize centers in a hexagonal grid pattern
    # Pattern: 6, 5, 6, 5, 4 circles per row
    centers = []
    radii_guess = []
    
    row_counts = [6, 5, 6, 5, 4]
    y_pos = 0.1  # Start slightly above 0
    
    for r_idx, count in enumerate(row_counts):
        # Hexagonal offset
        if r_idx % 2 == 0:
            x_start = 0.1
        else:
            x_start = 0.1 + 0.15 # Shift right for staggered row
        
        # Calculate spacing to fit 'count' circles in [0.1, 0.9] roughly
        # We want them evenly distributed
        # Available width approx 0.8 (from 0.1 to 0.9)
        # But we can use full width [0,1]
        # Let's just place them evenly
        
        # Spacing between centers
        # If we have 'count' circles, we have 'count-1' gaps
        # Total width 1.0. Centers should be at r, ..., 1-r.
        # Let's just use a fixed spacing heuristic first.
        spacing = 1.0 / (count + 1) 
        
        current_x = spacing
        for _ in range(count):
            centers.append([current_x, y_pos])
            # Initial radius guess: half the minimum distance to neighbors or walls
            # Roughly spacing/2
            radii_guess.append(spacing / 2.5) 
            current_x += spacing
        
        # Increase y for next row (vertical spacing in hex is sqrt(3)/2 * horizontal spacing approx)
        # Horizontal spacing was approx 1/7 ~ 0.14. 
        # Vertical step approx 0.14 * 0.866 ~ 0.12
        y_pos += 0.15

    centers = np.array(centers)
    
    # 2. Refine initial radii to be valid (no overlap)
    # Calculate max radius for each circle based on current positions
    initial_radii = np.zeros(n)
    for i in range(n):
        max_r = 10.0
        # Distance to walls
        max_r = min(max_r, centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])
        # Distance to other centers
        for j in range(n):
            if i == j: continue
            dist = np.linalg.norm(centers[i] - centers[j])
            max_r = min(max_r, dist / 2)
        initial_radii[i] = max_r * 0.95 # Slight safety margin

    # 3. Optimization Setup
    # Variables: [x1, y1, r1, x2, y2, r2, ..., x26, y26, r26]
    x0 = np.zeros(n * 3)
    for i in range(n):
        x0[3*i] = centers[i, 0]
        x0[3*i+1] = centers[i, 1]
        x0[3*i+2] = initial_radii[i]

    # Objective: Minimize negative sum of radii
    def objective(vars):
        return -np.sum(vars[2::3])

    # Constraints
    constraints = []
    
    # Boundary constraints: 0 <= x <= 1, 0 <= y <= 1, and radius fit
    # x - r >= 0  =>  x - r >= 0
    # 1 - x - r >= 0
    # y - r >= 0
    # 1 - y - r >= 0
    # r >= 0
    
    # We can use bounds for r >= 0, x, y in [0,1] maybe? 
    # But x, y constraints are coupled with r.
    # Better to use inequality constraints.
    
    # Helper to add constraint
    def add_constraint(func, name):
        constraints.append({'type': 'ineq', 'fun': func, 'args': ()})

    # Boundary constraints
    for i in range(n):
        def make_c_x_low(i):
            return lambda vars: vars[3*i] - vars[3*i+2] # x - r >= 0
        add_constraint(make_c_x_low(i), f'x_low_{i}')
        
        def make_c_x_high(i):
            return lambda vars: 1.0 - vars[3*i] - vars[3*i+2] # 1 - x - r >= 0
        add_constraint(make_c_x_high(i), f'x_high_{i}')
        
        def make_c_y_low(i):
            return lambda vars: vars[3*i+1] - vars[3*i+2] # y - r >= 0
        add_constraint(make_c_y_low(i), f'y_low_{i}')
        
        def make_c_y_high(i):
            return lambda vars: 1.0 - vars[3*i+1] - vars[3*i+2] # 1 - y - r >= 0
        add_constraint(make_c_y_high(i), f'y_high_{i}')
        
        def make_c_r_pos(i):
            return lambda vars: vars[3*i+2] # r >= 0
        add_constraint(make_c_r_pos(i), f'r_pos_{i}')

    # Pairwise non-overlap constraints
    # dist(i, j) >= r_i + r_j
    # sqrt(...) - r_i - r_j >= 0
    # To avoid sqrt, we can use squared: (dist)^2 - (r_i + r_j)^2 >= 0
    # But (r_i + r_j)^2 is convex? No.
    # SLSQP handles general constraints. Let's use distance directly or squared.
    # Squared is safer numerically if dist is large, but dist is fine.
    # However, derivative of sqrt is singular at 0. 
    # But dist should never be 0 if r > 0.
    
    for i in range(n):
        for j in range(i + 1, n):
            def make_c_pair(i, j):
                # vars indices
                # x_i = 3*i, y_i = 3*i+1, r_i = 3*i+2

                    xi, yi, ri = vars[3*i], vars[3*i+1], vars[3*i+2]
                    xj, yj, rj = vars[3*j], vars[3*j+1], vars[3*j+2]
                    
                    dx = xi - xj
                    dy = yi - yj
                    dist = np.sqrt(dx*dx + dy*dy)
                    return dist - (ri + rj)
                return constraint_func
            add_constraint(make_c_pair(i, j), f'pair_{i}_{j}')

    # Bounds for variables to help solver
    # x, y in [0, 1], r in [0, 0.5]
    bounds = []
    for _ in range(n):
        bounds.extend([(0, 1), (0, 1), (0, 0.5)])

    # Run optimization
    # SLSQP is good for constrained nonlinear optimization
    result = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints, 
                      options={'maxiter': 1000, 'ftol': 1e-9, 'disp': False})

    # Extract result
    best_vars = result.x
    best_centers = np.zeros((n, 2))
    best_radii = np.zeros(n)
    
    for i in range(n):
        best_centers[i, 0] = best_vars[3*i]
        best_centers[i, 1] = best_vars[3*i+1]
        best_radii[i] = best_vars[3*i+2]

    sum_radii = np.sum(best_radii)
    
    return best_centers, best_radii, sum_radii
