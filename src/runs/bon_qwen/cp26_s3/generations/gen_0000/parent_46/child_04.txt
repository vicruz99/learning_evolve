import numpy as np
from scipy.optimize import minimize

def run_packing():
    """
    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    """
    n = 26
    
    # --- Initialization ---
    # We initialize centers in a hexagonal grid pattern.
    # This provides a dense, structured starting point.
    
    # Parameters for hexagonal grid generation
    rows = 6
    cols = 6
    s = 0.18  # Horizontal spacing
    h = s * np.sqrt(3) / 2  # Vertical spacing for hexagonal packing
    
    points = []
    for i in range(rows):
        for j in range(cols):
            x = j * s + (i % 2) * (s / 2)
            y = i * h
            # Normalize/Scale to fit in [0, 1] with some margin
            # We map the generated grid to the unit square
            # The grid generated above starts at (0,0).
            # We want points roughly in [0.1, 0.9] to allow space for radii.
            if 0 <= x <= 1 and 0 <= y <= 1:
                points.append([x, y])
    
    # We might have more or fewer than 26 points.
    # Let's select 26 points that are well-distributed or just take the first 26 
    # if we have enough, otherwise pad.
    # Actually, with rows=6, cols=6, s=0.18, we get roughly 30 points.
    
    if len(points) < n:
        # Fallback to a regular grid if hex grid was too sparse
        ys = np.linspace(0.1, 0.9, 5)
        xs = np.linspace(0.1, 0.9, 5)
        grid_points = []
        for y in ys:
            for x in xs:
                grid_points.append([x, y])
        # Add extra points if needed
        while len(grid_points) < n:
            grid_points.append([0.5, 0.5]) # Dummy, will be moved
        points = grid_points

    # Select exactly n points
    # To avoid clustering, we can sort or randomize, but taking a subset of a dense grid is okay.
    # Let's just take the first n valid points from our generated list.
    # If we have more, we might want to pick a subset that covers the area better.
    # A simple deterministic selection:
    selected_points = points[:n]
    
    centers = np.array(selected_points)
    
    # Ensure centers are within [0, 1]
    centers = np.clip(centers, 0.0, 1.0)
    
    # Initial radii
    # Small enough to not overlap initially given the grid spacing
    initial_radii = np.full(n, 0.04)
    
    # Construct initial parameter vector
    # Structure: [x1, ..., xn, y1, ..., yn, r1, ..., rn]
    x_init = centers[:, 0]
    y_init = centers[:, 1]
    r_init = initial_radii
    
    x0 = np.concatenate([x_init, y_init, r_init])
    
    # Bounds for variables
    # x, y in [0, 1]
    # r in [0, 0.5] (max possible radius is 0.5)
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
    
    # --- Objective Function ---
    def objective(params):
        x = params[0:n]
        y = params[n:2*n]
        r = params[2*n:3*n]
        
        # We want to maximize sum(r), so minimize -sum(r)
        loss = -np.sum(r)
        
        # Penalty coefficient
        # Needs to be large enough to enforce constraints
        penalty_weight = 1000.0
        
        # 1. Boundary constraints
        # r <= x  => x - r >= 0
        violation = r - x
        loss += penalty_weight * np.sum(np.maximum(violation, 0)**2)
        
        # r <= 1 - x => x + r <= 1
        violation = x + r - 1.0
        loss += penalty_weight * np.sum(np.maximum(violation, 0)**2)
        
        # r <= y
        violation = r - y
        loss += penalty_weight * np.sum(np.maximum(violation, 0)**2)
        
        # r <= 1 - y => y + r <= 1
        violation = y + r - 1.0
        loss += penalty_weight * np.sum(np.maximum(violation, 0)**2)
        
        # 2. Non-negativity of radii (handled by bounds, but safety check)
        violation = -r
        loss += penalty_weight * np.sum(np.maximum(violation, 0)**2)
        
        # 3. Non-overlap constraints
        # dist(i, j) >= r_i + r_j
        # dist^2 >= (r_i + r_j)^2
        # Vectorized calculation of pairwise distances
        
        # Compute pairwise differences
        # x_diff[i, j] = x[i] - x[j]
        X_diff = x[:, None] - x[None, :]
        Y_diff = y[:, None] - y[None, :]
        
        # Squared distances
        D2 = X_diff**2 + Y_diff**2
        D = np.sqrt(D2 + 1e-12) # Add epsilon for numerical stability
        
        # Sum of radii matrix
        R_sum = r[:, None] + r[None, :]
        
        # Violation: r_i + r_j - dist > 0
        # We only care about pairs i < j, but summing all and halving is fine 
        # except for diagonal (i=i).
        # Diagonal: r_i + r_i - 0 = 2*r_i. This is not a constraint violation 
        # (a circle doesn't overlap itself).
        # So we mask out the diagonal.
        
        violations = R_sum - D
        # Zero out diagonal and lower triangle (since symmetric)
        # Actually, summing max(0, violation)^2 over all pairs counts each overlap twice.
        # We can just sum all and it doesn't affect the optimum, just the gradient magnitude.
        # But we must ensure diagonal is 0.
        
        # Create mask for upper triangle (excluding diagonal)
        mask = np.triu(np.ones((n, n), dtype=bool), k=1)
        violations = np.where(mask, violations, 0.0)
        
        loss += penalty_weight * np.sum(np.maximum(violations, 0)**2)
        
        return loss

    # --- Optimization ---
    # Use L-BFGS-B for bound constrained optimization
    # maxiter can be increased if needed
    res = minimize(objective, x0, method='L-BFGS-B', bounds=bounds, 
                   options={'maxiter': 1000, 'ftol': 1e-9, 'gtol': 1e-6})
    
    # Extract solution
    x_opt = res.x[0:n]
    y_opt = res.x[n:2*n]
    r_opt = res.x[2*n:3*n]
    
    centers_opt = np.column_stack((x_opt, y_opt))
    radii_opt = r_opt
    
    # Ensure radii are non-negative (clip just in case)
    radii_opt = np.maximum(radii_opt, 0.0)
    
    # Clip centers to [0, 1] just in case
    centers_opt = np.clip(centers_opt, 0.0, 1.0)
    
    sum_radii = np.sum(radii_opt)
    
    return centers_opt, radii_opt, sum_radii
