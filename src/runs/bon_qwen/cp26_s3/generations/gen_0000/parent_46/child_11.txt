def make_initial_config(n, row_counts, config_idx, trial):
    centers = np.zeros((n, 2))
    idx = 0
    
    np.random.seed(config_idx * 100 + trial * 10)
    jitter = 0.005 * np.random.randn(1)
    
    r_guess = 0.095
    vertical_spacing = r_guess * np.sqrt(3)
    
    for row in range(len(row_counts)):
        count = row_counts[row]
        y = r_guess + row * vertical_spacing
        offset = r_guess if row % 2 == 1 else 0
        
        total_width = (count - 1) * 2 * r_guess
        x_start = (1 - total_width) / 2 + offset
        
        for col in range(count):
            if idx < n:
                x = x_start + col * 2 * r_guess + jitter
                y_actual = y + jitter
                centers[idx] = [np.clip(x, 0, 1), np.clip(y_actual, 0, 1)]
                idx += 1
    
    return centers


def optimize_single(n, row_counts, config_idx, trial):
    centers = make_initial_config(n, row_counts, config_idx, trial)
    radii = np.ones(n) * 0.06
    
    best_centers = centers.copy()
    best_radii = radii.copy()

    # Main optimization loop
    for outer in range(80):
        for step in range(3000):
            lr = 0.004 * np.exp(-step / 800)
            forces = np.zeros((n, 2))
            
            # Inter-circle repulsion
            for i in range(n):
                for j in range(i + 1, n):
                    diff = centers[i] - centers[j]
                    dist_sq = np.dot(diff, diff)
                    dist = np.sqrt(dist_sq)
                    if dist < 1e-12:
                        diff = np.random.randn(2) * 0.005
                        dist = np.sqrt(np.dot(diff, diff))

                        overlap = max(0.0, min_dist - dist)
                        force_mag = overlap / (dist + 1e-12) * 50.0
                        if dist < min_dist:
                            force_mag *= 10.0
                        forces[i] += force_mag * diff
                        forces[j] -= force_mag * diff
            
            # Boundary repulsion
            for i in range(n):
                for k in [0, 1]:
                    r = radii[i]
                    if centers[i][k] < r:
                        forces[i][k] += 100.0 * (r - centers[i][k])
                    if centers[i][k] > 1 - r:
                        forces[i][k] -= 100.0 * (centers[i][k] - (1 - r))
            
            centers += lr * forces
            
            # Clip to valid range
            for i in range(n):
                r = radii[i]
                centers[i, 0] = np.clip(centers[i, 0], r, 1 - r)
                centers[i, 1] = np.clip(centers[i, 1], r, 1 - r)
            
            # Gradually increase radii
            if step % 3 == 0:
                radii *= 1.00003
        
        current_sum = np.sum(radii)
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers.copy()
            best_radii = radii.copy()
    
    return best_centers, best_radii
