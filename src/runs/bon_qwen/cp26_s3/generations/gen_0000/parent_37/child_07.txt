import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def objective(vars):
    """Objective function to maximize sum of radii (minimize negative sum)"""
    return -np.sum(vars[2::3])

def constraint_func(vars):
    """
    Computes constraint values. All values must be >= 0.
    Handles boundary and non-overlap constraints efficiently.
    """
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    N = N_CIRCLES
    
    n_con = 4*N + N*(N-1)//2
    con = np.empty(n_con)
    
    idx = 0
    # x >= r
    con[idx:idx+N] = x - r; idx += N
    # x + r <= 1  => 1 - x - r >= 0
    con[idx:idx+N] = 1.0 - x - r; idx += N
    # y >= r
    con[idx:idx+N] = y - r; idx += N
    # y + r <= 1  => 1 - y - r >= 0
    con[idx:idx+N] = 1.0 - y - r; idx += N
    
    # (xi - xj)^2 + (yi - yj)^2 >= (ri + rj)^2
    for i in range(N):
        for j in range(i+1, N):
            dx = x[i] - x[j]
            dy = y[i] - y[j]
            con[idx] = dx*dx + dy*dy - (r[i] + r[j])**2
            idx += 1
    return con

def get_initial_vars(seed):
    """Generates initial configuration using hexagonal grid + perturbation"""
    rng = np.random.RandomState(seed)
    centers = []
    y = 0.1
    row = 0
    while len(centers) < N_CIRCLES:
        row_len = 6 if row % 2 == 0 else 5
        x_start = 0.1 if row % 2 == 0 else 0.15
        for k in range(row_len):
            if len(centers) >= N_CIRCLES:
                break
            cx = x_start + k * 0.15
            if cx + 0.1 < 1.0:
                centers.append((cx, y))
        y += 0.17
        row += 1
        
    while len(centers) < N_CIRCLES:
        centers.append((rng.rand(), rng.rand()))
        
    centers = np.array(centers[:N_CIRCLES])
    vars = np.zeros(3*N_CIRCLES)
    vars[0::3] = centers[:, 0]
    vars[1::3] = centers[:, 1]
    vars[2::3] = 0.02
    return vars

def run_packing():
    bounds = [(0.0, 1.0)] * (2*N_CIRCLES) + [(1e-4, 0.5)] * N_CIRCLES
    cons = {'type': 'ineq', 'fun': constraint_func}
    
    best_x = None
    best_obj = -np.inf
    
    # Multiple restarts to escape local optima and find high-quality packing
    for seed in range(40):
        x0 = get_initial_vars(seed)
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 1000, 'ftol': 1e-12, 'disp': False})
            curr_obj = objective(res.x)
            if curr_obj < best_obj:
                best_obj = curr_obj
                best_x = res.x
        except Exception:
            continue
            
    if best_x is None:
        # Fallback to a simple valid grid if optimization fails
        centers = np.zeros((N_CIRCLES, 2))
        radii = np.full(N_CIRCLES, 0.01)
        return centers, radii, np.sum(radii)
        
    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = best_x[2::3]
    return centers, radii, -best_obj
