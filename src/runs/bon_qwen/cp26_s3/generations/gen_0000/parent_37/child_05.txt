import numpy as np
from scipy.optimize import linprog
import math

def solve_radii_lp(centers):
    """
    Solves the linear programming problem to find the maximum sum of radii
    for a fixed set of centers.
    Constraints:
    1. r_i >= 0
    2. r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    3. r_i + r_j <= dist(centers[i], centers[j])
    """
    n = centers.shape[0]
    
    # Objective: maximize sum(r_i) => minimize -sum(r_i)
    c = np.zeros(n)
    c[:] = -1.0

    # A_ub * x <= b_ub
    A_ub = []
    b_ub = []

    # 1. Boundary constraints (4 per circle)
    for i in range(n):
        x, y = centers[i]
        # r_i <= x
        row = np.zeros(n)
        row[i] = 1.0
        A_ub.append(row)
        b_ub.append(x)
        
        # r_i <= 1-x
        row = np.zeros(n)
        row[i] = 1.0
        A_ub.append(row)
        b_ub.append(1.0 - x)
        
        # r_i <= y
        row = np.zeros(n)
        row[i] = 1.0
        A_ub.append(row)
        b_ub.append(y)
        
        # r_i <= 1-y
        row = np.zeros(n)
        row[i] = 1.0
        A_ub.append(row)
        b_ub.append(1.0 - y)

    # 2. Non-overlap constraints (r_i + r_j <= dist)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dist)

    # 3. Non-negativity constraints r_i >= 0
    # linprog handles bounds as (lower, upper)
    bounds = [(0, None)] * n

    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
        else:
            # Fallback if LP fails
            return np.full(n, 0.01)
    except Exception:
        return np.full(n, 0.01)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # 1. Initialization: 5x5 grid for 25 circles, 1 in center
    centers = np.zeros((n, 2))
    
    # 5x5 grid positions
    # Spacing 1.0/6 to leave room? No, let's use 0.1 based grid initially
    # We'll use a slightly compressed grid to allow expansion
    step = 1.0 / 6.0 
    grid_centers = []
    for r in range(5):
        for c in range(5):
            grid_centers.append([step * (c + 1), step * (r + 1)])
            
    centers[:25] = grid_centers
    centers[25] = [0.5, 0.5] # 26th circle in the middle

    # Initial small radii
    radii = np.full(n, 0.02)

    # Optimization loop
    num_iterations = 400
    expansion_rate = 1.002
    
    for _ in range(num_iterations):
        # Solve for optimal radii given current centers
        current_radii = solve_radii_lp(centers)
        
        # Expand radii slightly to force centers to move
        radii = current_radii * expansion_rate
        
        # Force-directed layout adjustment
        # Apply forces to resolve overlaps and push away from walls
        for i in range(n):
            fx, fy = 0.0, 0.0
            
            # Repulsion from other circles
            for j in range(n):
                if i == j:
                    continue
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = math.sqrt(dx*dx + dy*dy)
                
                if dist < 1e-9:
                    dist = 1e-9
                    dx, dy = 0.1, 0.1 # Random push if identical
                
                # Target distance is sum of radii
                target_dist = radii[i] + radii[j]
                
                if dist < target_dist:
                    # Overlap penalty force
                    # Force proportional to overlap, inversely proportional to distance
                    overlap = target_dist - dist
                    force_mag = overlap * 50.0 / dist # Strong repulsion
                    
                    fx += (dx / dist) * force_mag
                    fy += (dy / dist) * force_mag
            
            # Repulsion from walls
            r = radii[i]
            # Left wall
            if centers[i][0] < r:
                fx += (r - centers[i][0]) * 100.0
            # Right wall
            if centers[i][0] > 1.0 - r:
                fx -= (centers[i][0] - (1.0 - r)) * 100.0
            # Bottom wall
            if centers[i][1] < r:
                fy += (r - centers[i][1]) * 100.0
            # Top wall
            if centers[i][1] > 1.0 - r:
                fy -= (centers[i][1] - (1.0 - r)) * 100.0
                
            # Update center
            # Learning rate / step size
            step_size = 0.01
            centers[i][0] += fx * step_size
            centers[i][1] += fy * step_size
            
            # Clamp to valid range [0, 1] roughly, though radii constraints handle it
            centers[i][0] = max(0.001, min(0.999, centers[i][0]))
            centers[i][1] = max(0.001, min(0.999, centers[i][1]))

    # Final clean-up: Solve LP one last time for the final centers
    # to get the exact max radii
    final_radii = solve_radii_lp(centers)
    
    # Clamp radii to ensure they are strictly non-negative
    final_radii = np.maximum(final_radii, 0.0)
    
    # Ensure centers are valid (just in case)
    # The LP ensures r <= dist_to_wall, so center +/- r is within [0,1]
    
    sum_radii = float(np.sum(final_radii))
    
    return centers, final_radii, sum_radii
