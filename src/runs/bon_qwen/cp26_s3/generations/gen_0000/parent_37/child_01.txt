import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Solves the circle packing problem for 26 circles in a unit square.
    Returns (centers, radii, sum_radii).
    """
    n = 26
    
    # 1. Generate an initial hexagonal grid layout
    # This approximates the densest packing (hexagonal)
    # We aim for 5 rows with counts: 6, 5, 6, 5, 4 (Total 26)
    # Initial guess for spacing to fit 26 items roughly in unit square
    initial_spacing = 0.22 
    row_height = initial_spacing * np.sqrt(3) / 2
    
    centers_list = []
    
    # Row configurations: (number of circles, horizontal offset)
    # Offsets of 0 or spacing/2 create the staggered hexagonal effect
    rows_config = [
        (6, 0.0),
        (5, initial_spacing / 2),
        (6, 0.0),
        (5, initial_spacing / 2),
        (4, 0.0)
    ]
    
    y_pos = 0.1 # Start slightly above bottom
    
    for count, offset in rows_config:
        if y_pos + initial_spacing > 1.0:
            break # Stop if out of bounds
        x_pos = 0.1 + offset
        for _ in range(count):
            centers_list.append([x_pos, y_pos])
            x_pos += initial_spacing
        y_pos += row_height

    centers = np.array(centers_list[:n]) # Ensure exactly 26
    
    # 2. Define the penalty function
    # We minimize boundary violations and overlaps.
    # r is fixed during optimization; we try to find valid centers for a given r.
    def compute_violations(centers_flat, r, n):
        c = centers_flat.reshape(-1, 2)
        
        # Boundary Violations: x >= r, x <= 1-r, y >= r, y <= 1-r
        # Penalty is squared distance from the valid region
        v_x_low = np.maximum(0, r - c[:, 0])
        v_x_high = np.maximum(0, c[:, 0] - (1 - r))
        v_y_low = np.maximum(0, r - c[:, 1])
        v_y_high = np.maximum(0, c[:, 1] - (1 - r))
        
        boundary_pen = np.sum(v_x_low**2) + np.sum(v_x_high**2) + \
                       np.sum(v_y_low**2) + np.sum(v_y_high**2)
        
        # Overlap Violations: dist(i,j) >= 2r
        # We calculate pairwise distances using broadcasting
        diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        
        # Upper triangle mask to avoid double counting and self-distance
        mask = np.triu(np.ones((n, n), dtype=bool), k=1)
        pair_dists = dists[mask]
        
        # Penalty if distance is less than 2r
        overlap_pen = np.sum(np.maximum(0, 2 * r - pair_dists)**2)
        
        return boundary_pen + overlap_pen

    # 3. Optimization Loop to maximize r
    # Start with a safe radius and increase until we hit the limit
    current_r = 0.08
    best_centers = centers.copy()
    
    # Tolerance for validity check
    tol = 1e-7
    
    # We iterate, increasing r and optimizing positions
    # This is essentially a bisection or incremental search on the feasible radius
    for step in range(100): # Max iterations for radius search
        try:
            # Try to optimize positions for current_r
            # Using L-BFGS-B which is efficient for bounds/constraints implicitly via penalty
            res = minimize(
                compute_violations, 
                best_centers.flatten(), 
                args=(current_r, n),
                method='L-BFGS-B',
                options={'ftol': 1e-12, 'gtol': 1e-10, 'maxiter': 500}
            )
            
            violations = res.fun
            
            if violations < tol:
                # Success: Current r is valid. Try larger r.
                best_centers = res.x.reshape(-1, 2)
                current_r += 0.0005 # Increment r
                
                # Dynamic step size adjustment could be added, but fixed small step works
                # To speed up, we could double the step if successful
            else:
                # Failure: Current r is too large (or local minimum trap).
                # We have found our limit for this topology.
                break
                
        except Exception:
            break

    # Final cleanup and scaling check
    # If we stopped at a valid r, we might want to do one last binary search refinement
    # between (current_r - step) and current_r to find exact max.
    
    valid_r = current_r - 0.0005 # The last successful r
    # However, the loop logic above: if success, r increases. 
    # If it breaks, current_r was the one that failed. 
    # So best_centers correspond to the state before the last failed attempt?
    # Actually, if 'res' was good, we updated best_centers. 
    # If 'res' was bad, we broke. So best_centers is valid for the r used in that failed attempt?
    # No, if res.fun > tol, best_centers is NOT valid for current_r.
    # So valid_r is the previous step's r.
    
    # Let's perform a final refinement using binary search on r around the found range
    r_low = valid_r
    r_high = current_r
    final_centers = best_centers.copy()
    
    # Refine radius to high precision
    for _ in range(20):
        r_mid = (r_low + r_high) / 2.0
        
        try:
            # Optimize for r_mid starting from current best_centers
            res = minimize(
                compute_violations,
                final_centers.flatten(),
                args=(r_mid, n),
                method='L-BFGS-B',
                options={'ftol': 1e-12, 'maxiter': 500}
            )
            
            if res.fun < tol:
                r_low = r_mid
                final_centers = res.x.reshape(-1, 2)
            else:
                r_high = r_mid
        except:
            r_high = r_mid
            
    final_r = r_low
    
    # Construct output
    radii = np.full(n, final_r)
    sum_radii = np.sum(radii)
    
    return final_centers, radii, sum_radii
