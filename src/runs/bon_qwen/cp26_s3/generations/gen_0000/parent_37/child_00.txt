import numpy as np

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    Uses a simulated annealing approach to find positions that maximize the minimum 
    distance to boundaries and other circles.
    """
    np.random.seed(42)
    N = 26
    
    # --- 1. Initialization ---
    # Create a 6x5 grid to generate candidate points
    x_grid = np.linspace(0, 1, 6)
    y_grid = np.linspace(0, 1, 5)
    
    points = []
    for y in y_grid:
        for x in x_grid:
            points.append([x, y])
    
    points = np.array(points)
    
    # Remove 4 points from the corners to get exactly 26 points
    # and keep them well-distributed
    indices_to_remove = [0, -1, 5, 24] 
    indices_to_keep = [i for i in range(len(points)) if i not in indices_to_remove]
    centers = points[indices_to_keep].copy()
    
    # Add small random noise to break symmetry and grid alignment
    centers += np.random.uniform(-0.05, 0.05, centers.shape)
    
    # Clip to valid range [0, 1]
    centers = np.clip(centers, 0, 1)

    # --- 2. Simulated Annealing Optimization ---
    # Objective: Maximize the minimum distance (to boundary or neighbor)
    # This distance determines the maximum possible equal radius.
    
    def evaluate_min_dist(positions):
        """
        Calculates the minimum distance to boundaries and between circles.
        Returns the minimum distance found.
        """
        min_dist = 1.0
        
        # Distance to boundaries
        # For a circle at (x,y) with radius r, we need x >= r, 1-x >= r, etc.
        # So r <= min(x, 1-x, y, 1-y).
        dist_bound = np.min([
            positions[:, 0], 
            1 - positions[:, 0], 
            positions[:, 1], 
            1 - positions[:, 1]
        ])
        min_dist = min(min_dist, dist_bound)
        
        # Distance between circles
        # We need dist(i, j) >= 2r. So r <= dist(i, j) / 2.
        # We look at squared distances to avoid sqrt in loop, but need actual dist for result.
        # Efficiently compute pairwise distances
        diff = positions[:, np.newaxis, :] - positions[np.newaxis, :, :]
        dists_sq = np.sum(diff**2, axis=2)
        
        # Mask out diagonal (self-distances) and lower triangle
        np.fill_diagonal(dists_sq, np.inf)
        dists = np.sqrt(dists_sq)
        
        # Find minimum distance between distinct pairs
        current_min_pair_dist = np.min(dists)
        min_dist = min(min_dist, current_min_pair_dist / 2.0)
        
        return min_dist

    # Initial evaluation
    current_centers = centers.copy()
    current_min_dist = evaluate_min_dist(current_centers)
    best_centers = current_centers.copy()
    best_min_dist = current_min_dist
    
    # Annealing parameters
    temperature = 0.02
    decay = 0.9995
    min_temp = 1e-7
    
    iterations = 0
    max_iterations = 50000
    
    while temperature > min_temp and iterations < max_iterations:
        iterations += 1
        
        # Propose a move: pick a random circle and move it slightly
        i = np.random.randint(N)
        step_size = temperature * 0.5
        new_center = current_centers[i].copy()
        new_center[0] += np.random.uniform(-step_size, step_size)
        new_center[1] += np.random.uniform(-step_size, step_size)
        
        # Keep within bounds (simple reflection or clamp)
        new_center = np.clip(new_center, 0, 1)
        
        # Evaluate new state
        trial_centers = current_centers.copy()
        trial_centers[i] = new_center
        trial_min_dist = evaluate_min_dist(trial_centers)
        
        # Acceptance criterion
        delta = trial_min_dist - current_min_dist
        
        if delta > 0 or np.random.rand() < np.exp(delta / temperature):
            current_centers = trial_centers
            current_min_dist = trial_min_dist
            
            if current_min_dist > best_min_dist:
                best_min_dist = current_min_dist
                best_centers = current_centers.copy()
        
        # Cool down
        temperature *= decay

    # --- 3. Finalize Packing ---
    # Use the best centers found
    centers = best_centers
    # Recalculate the precise maximum radius for these centers
    r_opt = evaluate_min_dist(centers)
    
    # Ensure radii are strictly positive and valid
    r_opt = max(r_opt, 1e-6)
    radii = np.full(N, r_opt)
    
    # Final validation check and correction if any center is too close to edge due to float errors
    # Adjust centers if necessary to strictly satisfy r_opt <= x <= 1-r_opt
    # This is a safety clamp
    centers[:, 0] = np.clip(centers[:, 0], r_opt, 1 - r_opt)
    centers[:, 1] = np.clip(centers[:, 1], r_opt, 1 - r_opt)
    
    # Re-eval radius after clamping just in case
    # (Clamping shouldn't reduce min_dist if logic is sound, but safe to check)
    r_opt = evaluate_min_dist(centers)
    radii = np.full(N, r_opt)
    
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
