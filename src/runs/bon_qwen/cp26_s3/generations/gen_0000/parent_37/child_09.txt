import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def get_initial_centers():
    n = N_CIRCLES
    # Hex grid generation
    s = 0.15
    h = s * np.sqrt(3) / 2
    points = []
    for r in range(10):
        for c in range(10):
            x = c * s + (r % 2) * (s / 2)
            y = r * h
            if 0 <= x <= 1 and 0 <= y <= 1:
                points.append([x, y])
    points = np.array(points)
    
    if len(points) < n:
        # Fallback grid
        pts_x = np.linspace(0, 1, 6)
        pts_y = np.linspace(0, 1, 6)
        grid = []
        for y in pts_y:
            for x in pts_x:
                grid.append([x, y])
        points = np.array(grid)
        
    num_pts = len(points)
    if num_pts >= n:
        indices = np.linspace(0, num_pts - 1, n, dtype=int)
        return points[indices]
    else:
        return points[:n]

def objective(v):
    # v is flat array [x0, y0, r0, x1, y1, r1, ...]
    # Radii are at indices 2, 5, 8...
    return -np.sum(v[2::3])

def constraints_func(v):
    n = N_CIRCLES
    xs = v[0::3]
    ys = v[1::3]
    rs = v[2::3]
    
    cons = []
    
    # Boundary constraints
    cons.append(xs - rs)
    cons.append(1.0 - xs - rs)
    cons.append(ys - rs)
    cons.append(1.0 - ys - rs)
    
    # Overlap constraints
    # Iterate all pairs
    # To avoid creating a list of tuples every time, we can loop
    # But list of tuples is fast to create?
    # Actually, we can just loop indices.
    
    overlap_cons = np.empty(n * (n - 1) // 2)
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            dx = xs[i] - xs[j]
            dy = ys[i] - ys[j]
            dist_sq = dx*dx + dy*dy
            r_sum = rs[i] + rs[j]
            overlap_cons[idx] = dist_sq - r_sum*r_sum
            idx += 1
            
    cons.append(overlap_cons)
    return np.concatenate(cons)

def run_packing():
    n = N_CIRCLES
    centers_init = get_initial_centers()
    
    # Estimate initial radius
    min_dist = 1.0
    for i in range(n):
        for j in range(i + 1, n):
            d = np.linalg.norm(centers_init[i] - centers_init[j])
            if d < min_dist:
                min_dist = d
    for i in range(n):
        d_bound = min(centers_init[i,0], 1-centers_init[i,0], centers_init[i,1], 1-centers_init[i,1])
        if d_bound < min_dist:
            min_dist = d_bound
            
    r_init = max(0.01, min_dist / 2.5)
    
    x0 = np.zeros(3 * n)
    for i in range(n):
        x0[3*i] = centers_init[i, 0]
        x0[3*i+1] = centers_init[i, 1]
        x0[3*i+2] = r_init
        
    bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n
    
    try:
        res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                       constraints={'type': 'ineq', 'fun': constraints_func},
                       options={'maxiter': 500, 'ftol': 1e-12, 'disp': False})
        
        if np.isnan(res.x).any():
            raise ValueError("Optimization produced NaN")
            
        centers = np.zeros((n, 2))

            centers[i, 1] = res.x[3*i+1]
            radii[i] = res.x[3*i+2]
            
    except Exception:
        centers = centers_init
        radii = np.full(n, r_init)
        
    return centers, radii, np.sum(radii)

        # Select n points evenly distributed from the generated grid
        # This helps in distributing circles across the square
        indices = np.linspace(0, num_pts - 1, n, dtype=int)
        return points[indices]
    else:
        return points[:n]

def objective(v):
    """
    Objective function to maximize sum of radii.
    v is a 1D array: [x0, y0, r0, x1, y1, r1, ..., x25, y25, r25]
    """
    # Radii are at indices 2, 5, 8, ... (step 3, starting at 2)
    # We minimize the negative sum to maximize the sum
    return -np.sum(v[2::3])

def constraints_func(v):
    """
    Computes inequality constraints.
    Returns a 1D array of constraint values, all must be >= 0.
    """
    n = N_CIRCLES
    xs = v[0::3]
    ys = v[1::3]
    rs = v[2::3]
    
    cons = []
    
    # 1. Boundary Constraints
    # Circle i must be inside [0,1]x[0,1]
    # x_i >= r_i  => x_i - r_i >= 0
    cons.append(xs - rs)
    # x_i <= 1 - r_i => 1 - x_i - r_i >= 0
    cons.append(1.0 - xs - rs)
    # y_i >= r_i => y_i - r_i >= 0
    cons.append(ys - rs)
    # y_i <= 1 - r_i => 1 - y_i - r_i >= 0
    cons.append(1.0 - ys - rs)
    
    # 2. Non-overlap Constraints
    # dist(i, j)^2 >= (r_i + r_j)^2
    # dist(i, j)^2 - (r_i + r_j)^2 >= 0
    
    num_pairs = n * (n - 1) // 2
    overlap_cons = np.empty(num_pairs)
    idx = 0
    
    # Iterate over all unique pairs (i, j) with i < j
    for i in range(n):
        xi = xs[i]
        yi = ys[i]
        ri = rs[i]
        for j in range(i + 1, n):
            dx = xi - xs[j]
            dy = yi - ys[j]
            dist_sq = dx * dx + dy * dy
            r_sum = ri + rs[j]
            overlap_cons[idx] = dist_sq - r_sum * r_sum
            idx += 1
            
    cons.append(overlap_cons)
    
    # Concatenate all constraint arrays into a single 1D array
    return np.concatenate(cons)

def run_packing():
    """
    Main function to solve the circle packing problem.
    Returns:
        centers: np.array of shape (26, 2)
        radii: np.array of shape (26,)
        sum_radii: float
    """
    n = N_CIRCLES
    
    # 1. Generate Initial Configuration
    centers_init = get_initial_centers()
    
    # Estimate a safe initial radius based on minimum distance between points
    min_dist = 1.0
    
    # Check pairwise distances
    for i in range(n):
        xi, yi = centers_init[i]
        for j in range(i + 1, n):
            xj, yj = centers_init[j]
            d_sq = (xi - xj)**2 + (yi - yj)**2
            if d_sq < min_dist:
                min_dist = d_sq
    min_dist = np.sqrt(min_dist)
    
    # Check distance to boundaries
    for i in range(n):
        xi, yi = centers_init[i]
        d_bound = min(xi, 1.0 - xi, yi, 1.0 - yi)
        if d_bound < min_dist:
            min_dist = d_bound
            
    # Initialize radii to a fraction of the minimum distance
    r_init = max(0.001, min_dist / 2.5)
    
    # 2. Setup Optimization
    # Flatten variables: [x0, y0, r0, x1, y1, r1, ...]
    x0 = np.zeros(3 * n)
    for i in range(n):
        x0[3*i] = centers_init[i, 0]
        x0[3*i+1] = centers_init[i, 1]
        x0[3*i+2] = r_init
        
    # Bounds: x, y in [0, 1], r in [0, 0.5]
    bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n
    
    try:
        # Run SLSQP optimizer
        res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                       constraints={'type': 'ineq', 'fun': constraints_func},
                       options={'maxiter': 500, 'ftol': 1e-12, 'disp': False})
        
        # Check for NaN in result
        if np.isnan(res.x).any():
            raise ValueError("Optimization produced NaN values")
            
        # Extract optimized centers and radii
        centers = np.zeros((n, 2))
        radii = np.zeros(n)
        for i in range(n):
            centers[i, 0] = res.x[3*i]
            centers[i, 1] = res.x[3*i+1]
            radii[i] = res.x[3*i+2]
            
    except Exception:
        # Fallback to initial configuration if optimization fails
        centers = centers_init
        radii = np.full(n, r_init)
        
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
