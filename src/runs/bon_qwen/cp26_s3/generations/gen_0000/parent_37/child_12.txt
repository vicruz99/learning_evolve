Constraints include:
1. Boundary conditions: Each circle must lie within $[0, 1] \times [0, 1]$, i.e., $r_i \le x_i \le 1-r_i$ and $r_i \le y_i \le 1-r_i$.
2. Non-overlap conditions: For every pair $i, j$, the Euclidean distance between centers must be at least $r_i + r_j$, i.e., $(x_i-x_j)^2 + (y_i-y_j)^2 \ge (r_i+r_j)^2$.
3. Non-negativity: $r_i \ge 0$.

import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def objective(vars_):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(vars_[2::3])

def constraints_fun(vars_):
    """
    Returns array of constraint values. All must be >= 0.
    Constraints:
    - 5 per circle: x-r, 1-x-r, y-r, 1-y-r, r
    - 1 per pair: dist^2 - (r_i+r_j)^2
    """
    n = N_CIRCLES
    pts = vars_.reshape((n, 3))
    n_constr = 5 * n + n * (n - 1) // 2
    c = np.empty(n_constr)
    idx = 0
    for i in range(n):
        x, y, r = pts[i]
        c[idx] = x - r
        c[idx+1] = 1.0 - x - r
        c[idx+2] = y - r
        c[idx+3] = 1.0 - y - r
        c[idx+4] = r
        idx += 5
    for i in range(n):
        for j in range(i + 1, n):
            xi, yi, ri = pts[i]
            xj, yj, rj = pts[j]
            c[idx] = (xi - xj)**2 + (yi - yj)**2 - (ri + rj)**2
            idx += 1
    return c

def get_init(seed, n):
    """Generate initial configuration using perturbed hexagonal grid."""
    rng = np.random.RandomState(seed)
    pts = np.zeros((n, 3))
    idx = 0
    y = 0.12
    while idx < n:
        x = 0.12
        while x < 0.88:
            if idx < n:
                pts[idx] = [x + rng.uniform(-0.02, 0.02), 
                            y + rng.uniform(-0.02, 0.02), 
                            0.09]
                idx += 1
            x += 0.18
        y += 0.155
    return pts.flatten()

def run_packing():
    np.random.seed(42)
    n = N_CIRCLES
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * n
    
    best_sum_r = 0.0
    best_vars = None
    
    cons = {'type': 'ineq', 'fun': constraints_fun}
    
    # Run optimizations from multiple initial seeds
    for seed in range(8):
        init_vars = get_init(seed, n)
        
        res = minimize(objective, init_vars, method='SLSQP',
                       bounds=bounds,
                       constraints=cons,
                       options={'maxiter': 1500, 'ftol': 1e-10, 'disp': False})
        
        if res.success or res.nit > 100:
            c_vals = constraints_fun(res.x)
            if np.min(c_vals) >= -1e-5:
                s_r = -res.fun
                if s_r > best_sum_r:
                    best_sum_r = s_r
                    best_vars = res.x.copy()
                    
        # Perturb best solution found so far and try again
        if best_vars is not None:
            pert = best_vars + np.random.uniform(-0.05, 0.05, 78)
            np.clip(pert, 0.0, 1.0, out=pert)
            pert[2::3] = np.clip(pert[2::3], 0.01, 0.2)
            
            res2 = minimize(objective, pert, method='SLSQP',
                            bounds=bounds,
                            constraints=cons,
                            options={'maxiter': 1000, 'ftol': 1e-10, 'disp': False})
            if res2.success or res2.nit > 100:
                c_vals2 = constraints_fun(res2.x)
                if np.min(c_vals2) >= -1e-5:
                    s_r2 = -res2.fun
                    if s_r2 > best_sum_r:
                        best_sum_r = s_r2
                        best_vars = res2.x.copy()

    if best_vars is not None:
        pts = best_vars.reshape((n, 3))
        centers = pts[:, :2].copy()
        radii = pts[:, 2].copy()
        
        # Apply safety margin to guarantee strict feasibility
        radii -= 1e-7
        radii = np.maximum(radii, 1e-9)
        
        return centers, radii, float(np.sum(radii))
    else:
        # Fallback uniform grid
        centers = np.zeros((n, 2))
        radii = np.full(n, 0.09)
        idx = 0
        for i in range(5):
            for j in range(5):
                if idx < n:
                    centers[idx] = [0.1 + 0.18*i, 0.1 + 0.18*j]
                    idx += 1
        return centers, radii, float(np.sum(radii))
