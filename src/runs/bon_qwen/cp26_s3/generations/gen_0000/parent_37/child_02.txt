import numpy as np
from scipy.optimize import minimize

def compute_energy_and_grad(params, n, lam):
    centers = params[:2*n].reshape(n, 2)
    radii = params[2*n:]
    
    energy = -np.sum(radii)
    grad = np.zeros_like(params)
    grad[2*n:] = -1.0
    
    # Boundary penalty terms and gradients
    for i in range(n):
        x, y, r = centers[i, 0], centers[i, 1], radii[i]
        
        v1 = r - x
        if v1 > 0:
            energy += lam * v1**2
            grad[i*2] -= 2 * lam * v1
            grad[2*n+i] += 2 * lam * v1
            
        v2 = r - (1.0 - x)
        if v2 > 0:
            energy += lam * v2**2
            grad[i*2] += 2 * lam * v2
            grad[2*n+i] += 2 * lam * v2
            
        v3 = r - y
        if v3 > 0:
            energy += lam * v3**2
            grad[i*2+1] -= 2 * lam * v3
            grad[2*n+i] += 2 * lam * v3
            
        v4 = r - (1.0 - y)
        if v4 > 0:
            energy += lam * v4**2
            grad[i*2+1] += 2 * lam * v4
            grad[2*n+i] += 2 * lam * v4

    # Overlap penalty terms and gradients
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx*dx + dy*dy)
            if dist < 1e-12:
                dist = 1e-12
                
            v = radii[i] + radii[j] - dist
            if v > 0:
                energy += lam * v**2
                gx = -dx / dist
                gy = -dy / dist
                
                grad[i*2] += 2 * lam * v * gx
                grad[i*2+1] += 2 * lam * v * gy
                grad[2*n+i] += 2 * lam * v
                
                grad[j*2] += 2 * lam * v * (-gx)
                grad[j*2+1] += 2 * lam * v * (-gy)
                grad[2*n+j] += 2 * lam * v
                
    return energy, grad

def project_radii(centers, radii, n):
    radii = radii.copy()
    for _ in range(10):
        changed = False
        for i in range(n):
            max_r = min(centers[i, 0], 1.0 - centers[i, 0], 
                        centers[i, 1], 1.0 - centers[i, 1])
            for j in range(n):
                if i == j:
                    continue
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                max_r = min(max_r, dist - radii[j])
            if max_r < radii[i] - 1e-9:
                radii[i] = max(max_r, 0.0)
                changed = True
        if not changed:
            break
    return radii

def local_search(init_centers, init_radii, n=26):
    params = np.concatenate([init_centers.flatten(), init_radii])
    bounds = [(0.0, 1.0) for _ in range(2*n)] + [(1e-6, 0.5) for _ in range(n)]
    
    lam = 5.0
    params = params.copy()
    for _ in range(15):
        res = minimize(compute_energy_and_grad, params, args=(n, lam),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-10})
        params = res.x
        lam *= 2.0
        
    centers = params[:2*n].reshape(n, 2)
    radii = params[2*n:]
    return centers, radii, np.sum(radii)

def run_packing():
    n = 26
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    inits = []
    
    # 1. Structured Grid Initialization
    c_grid, r_grid = [], []
    for i in range(5):
        for j in range(5):
            if len(c_grid) < n:
                c_grid.append([0.1 + i*0.18, 0.1 + j*0.18])
                r_grid.append(0.05)
    while len(c_grid) < n:
        c_grid.append([0.5, 0.5])
        r_grid.append(0.05)
    inits.append((np.array(c_grid), np.array(r_grid)))
    
    # 2. Hexagonal Lattice Initialization
    c_hex, r_hex = [], []
    r0 = 0.08
    row = 0
    while len(c_hex) < n:
        col = 0
        while len(c_hex) < n:
            x = (col + 0.5 + (0.5 if row % 2 else 0)) * r0 * 1.6 + 0.05
            y = row * r0 * 1.4 + 0.05
            if x < 0.95 and y < 0.95:
                c_hex.append([x, y])
                r_hex.append(r0)
            col += 1
        row += 1
    inits.append((np.array(c_hex), np.array(r_hex)))
    
    # 3. Random Initializations with different seeds
    np.random.seed(42)
    inits.append((np.random.uniform(0.2, 0.8, (n, 2)), np.ones(n) * 0.05))
    np.random.seed(123)
    inits.append((np.random.uniform(0.2, 0.8, (n, 2)), np.ones(n) * 0.06))
    np.random.seed(777)
    inits.append((np.random.uniform(0.15, 0.85, (n, 2)), np.ones(n) * 0.07))
    np.random.seed(999)
    inits.append((np.random.uniform(0.1, 0.9, (n, 2)), np.ones(n) * 0.04))

    for c0, r0 in inits:
        try:
            centers, radii, s = local_search(c0, r0, n)
            radii = project_radii(centers, radii, n)
            s = np.sum(radii)
            if s > best_sum:
                best_sum = s
                best_centers = centers.copy()
                best_radii = radii.copy()
        except Exception:
            pass
            
    return best_centers, best_radii, best_sum
