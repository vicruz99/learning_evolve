import numpy as np
from scipy.optimize import minimize


def penalized_objective(x, n, penalty):
    """Objective: maximize sum of radii with penalties for constraint violations."""
    obj = -np.sum(x[3 * n:])

    for i in range(n):
        r = x[3 * n + i]
        x_i = x[3 * i]
        y_i = x[3 * i + 1]

        v1 = max(0.0, r - x_i)
        v2 = max(0.0, r - (1.0 - x_i))
        v3 = max(0.0, r - y_i)
        v4 = max(0.0, r - (1.0 - y_i))
        obj += penalty * (v1 * v1 + v2 * v2 + v3 * v3 + v4 * v4)

    for i in range(n):
        for j in range(i + 1, n):
            dx = x[3 * i] - x[3 * j]
            dy = x[3 * i + 1] - x[3 * j + 1]
            dist_sq = dx * dx + dy * dy
            if dist_sq < 1e-16:
                dist = 1e-8
            else:
                dist = np.sqrt(dist_sq)
            r_sum = x[3 * n + i] + x[3 * n + j]

            v = max(0.0, r_sum - dist)
            obj += penalty * v * v

    return obj


def initialize_hexagonal(n):
    """Initialize circles in a staggered hexagonal pattern."""
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.04

    idx = 0
    for row in range(5):
        if row < 4:
            n_in_row = 5
        else:
            n_in_row = 6
        for col in range(n_in_row):
            if idx >= n:
                break
            x = 0.1 + col * 0.2 + (0.1 if row % 2 == 1 else 0.0)
            y = 0.1 + row * 0.2
            centers[idx] = [x, y]
            idx += 1

    return centers, radii


def initialize_grid(n):
    """Initialize circles in a regular grid pattern."""
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.035

    idx = 0
    for row in range(6):
        for col in range(5):
            if idx >= n:
                break
            centers[idx] = [0.15 + col * 0.18, 0.1 + row * 0.16]
            idx += 1

    return centers, radii


def optimize_from_initial(x0, n, bounds):
    """Run iterative penalty optimization from a given initial point."""
    penalty = 50.0
    x_current = x0.copy()

    for iteration in range(20):
        result = minimize(
            penalized_objective,
            x_current,
            args=(n, penalty),
            method="L-BFGS-B",
            bounds=bounds,
            options={"maxiter": 3000, "ftol": 1e-16, "gtol": 1e-10},
        )
        x_current = result.x.copy()
        penalty *= 1.5

    return x_current


def run_packing():

        best_sum = sum_grid

    for seed in range(5):
