import numpy as np
from scipy.optimize import differential_evolution

def compute_objective(params):
    """
    Objective function for differential evolution.
    Minimizes negative sum of radii + penalty for constraint violations.
    """
    n = 26
    xs = params[0::3]
    ys = params[1::3]
    rs = params[2::3]
    
    # Primary objective: maximize sum of radii
    obj = -np.sum(rs)
    
    # Boundary violations: circle must be inside [0, 1] x [0, 1]
    v = np.maximum(0, rs - xs)
    v += np.maximum(0, xs + rs - 1)
    v += np.maximum(0, rs - ys)
    v += np.maximum(0, ys + rs - 1)
    
    # Overlap violations: distance between centers >= sum of radii
    # Vectorized pairwise distance calculation
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dist = np.sqrt(dx**2 + dy**2)
    r_sum = rs[:, None] + rs[None, :]
    
    viol = r_sum - dist
    np.fill_diagonal(viol, 0)
    v += np.maximum(0, viol).sum() / 2.0
    
    # Large penalty coefficient ensures feasibility is prioritized
    return obj + 1000.0 * v**2

def run_packing():
    """
    Runs the optimization and returns the best valid packing found.
    """
    n = 26
    bounds = []
    for i in range(n):
        bounds.append((0.0, 1.0))  # x bounds
        bounds.append((0.0, 1.0))  # y bounds
        bounds.append((0.0, 0.3))  # r bounds (conservative upper limit)
        
    # Global optimization using Differential Evolution
    result = differential_evolution(
        compute_objective, 
        bounds, 
        popsize=20, 
        maxiter=1000, 
        seed=42, 
        tol=1e-7,
        polish=True
    )
    
    params = result.x
    centers = np.column_stack((params[0::3], params[1::3]))
    radii = params[2::3]
    
    # Strict feasibility correction
    max_viol = 0.0
    
    # Check boundaries
    max_viol = max(max_viol, np.max(np.maximum(0, radii - centers[:,0])))
    max_viol = max(max_viol, np.max(np.maximum(0, centers[:,0] + radii - 1)))
    max_viol = max(max_viol, np.max(np.maximum(0, radii - centers[:,1])))
    max_viol = max(max_viol, np.max(np.maximum(0, centers[:,1] + radii - 1)))
    
    # Check overlaps
    for i in range(n):
        for j in range(i+1, n):
            d = np.sqrt((centers[i,0]-centers[j,0])**2 + (centers[i,1]-centers[j,1])**2)
            v = radii[i] + radii[j] - d
            if v > max_viol:
                max_viol = v
                
    # If any violation exists, uniformly shrink radii to satisfy constraints
    if max_viol > 1e-9:
        radii -= max_viol
        
    return centers, radii, np.sum(radii)
