def calculate_sum_radii(centers):
    """
    Calculates the sum of the maximum possible radii for a given set of centers.
    """
    n = centers.shape[0]
    radii = np.zeros(n)
    
    for i in range(n):
        x, y = centers[i]
        # Distance to boundaries
        d_bound = min(x, 1 - x, y, 1 - y)
        # Radius cannot be negative (if center is outside, though optimizer should keep inside)
        if d_bound < 0:
            d_bound = 0
        
        # Distance to other circles (half the Euclidean distance)
        d_min = np.inf

        r = min(d_bound, d_min / 2.0)
        radii[i] = r
        
    return -np.sum(radii), radii # Return negative sum for minimization
