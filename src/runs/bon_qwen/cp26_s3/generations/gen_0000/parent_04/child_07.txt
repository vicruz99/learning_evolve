

<strategy>
To maximize the sum of radii for 26 circles in a unit square, we treat this as a constrained nonlinear optimization problem. Since the objective is linear in radii and constraints are quadratic/linear, a direct solver might struggle with non-convexity. Instead, we use a **penalty method** combined with `L-BFGS-B`, which efficiently handles bound constraints and smooth objective functions.

**Strategy:**
1. **Initialization:** Generate multiple diverse starting configurations to avoid poor local minima:
   - A 5x5 grid with one extra circle.
   - A hexagonal lattice approximation (higher packing density).
   - Random placements with reasonable initial radii.
2. **Objective & Penalty:** Minimize $-\sum r_i + \lambda \cdot P(z)$, where $P(z)$ penalizes boundary violations and circle overlaps. We use squared penalties for smoothness.
3. **Penalty Annealing:** Iteratively increase the penalty weight $\lambda$ (geometric progression). This forces the solver to strictly respect constraints while maximizing radii.
4. **Vectorization:** Compute overlaps and boundary checks using NumPy broadcasting for performance.
5. **Repair:** Apply a final safety shrink if minor numerical violations remain.
6. **Selection:** Run optimization from each initialization and keep the configuration with the highest valid sum of radii.
</strategy>

import numpy as np
from scipy.optimize import minimize

def compute_penalty(z):
    """Computes the squared penalty for constraint violations."""
    n = len(z) // 3
    X = z[0::3]
    Y = z[1::3]
    R = z[2::3]
    
    pen = 0.0
    
    # Boundary penalties: circle must be inside [0,1]^2
    pen += np.sum(np.maximum(R - X, 0.0)**2)
    pen += np.sum(np.maximum(X + R - 1.0, 0.0)**2)
    pen += np.sum(np.maximum(R - Y, 0.0)**2)
    pen += np.sum(np.maximum(Y + R - 1.0, 0.0)**2)
    
    # Overlap penalties: distance between centers >= sum of radii
    # Vectorized computation for all pairs
    DX = X[:, None] - X[None, :]
    DY = Y[:, None] - Y[None, :]
    D = np.sqrt(DX**2 + DY**2)
    O = R[:, None] + R[None, :] - D
    np.fill_diagonal(O, 0.0) # Ignore self-interaction
    pen += np.sum(np.maximum(O, 0.0)**2)
    
    return pen

def objective(z, penalty_factor):
    """Objective function: maximize sum of radii minus penalty."""
    n = len(z) // 3
    R = z[2::3]
    return -np.sum(R) + penalty_factor * compute_penalty(z)

def run_packing():
    n = 26
    best_z = None
    best_sum = 0.0
    
    # Bounds for x, y in [0,1] and r in [1e-4, 0.5]
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)] * n + [(1e-4, 0.5)] * n
    
    initializations = []
    
    # 1. Grid pattern (5x5 + 1 center)
    z_grid = np.zeros(3 * n)
    coords = np.linspace(0.125, 0.875, 5)
    for i in range(5):
        for j in range(5):
            idx = i * 5 + j
            z_grid[3 * idx] = coords[i]
            z_grid[3 * idx + 1] = coords[j]
            z_grid[3 * idx + 2] = 0.08
    z_grid[3 * 25] = 0.5
    z_grid[3 * 25 + 1] = 0.5
    z_grid[3 * 25 + 2] = 0.08
    initializations.append(z_grid)
    
    # 2. Hexagonal-like pattern
    z_hex = np.zeros(3 * n)
    row_counts = [5, 6, 5, 6, 4]
    y_start = 0.14
    y_step = 0.185
    x_start = 0.10
    x_step = 0.20
    idx = 0
    for r, cnt in enumerate(row_counts):
        y = y_start + r * y_step
        for c in range(cnt):
            x = x_start + c * x_step + (0.10 if r % 2 == 1 else 0.0)
            z_hex[3 * idx] = np.clip(x, 0.0, 1.0)
            z_hex[3 * idx + 1] = np.clip(y, 0.0, 1.0)
            z_hex[3 * idx + 2] = 0.085
            idx += 1
    initializations.append(z_hex)
    
    # 3. Random initializations
    for _ in range(4):
        z_rand = np.random.uniform(0.15, 0.85, 3 * n)
        z_rand[2::3] = 0.075
        initializations.append(z_rand)
        
    # Optimization loop
    for z0 in initializations:
        z = z0.copy()
        penalty = 30.0
        
        # Anneal penalty to enforce constraints strictly
        for step in range(6):
            res = minimize(objective, z, args=(penalty,), method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 4000, 'ftol': 1e-14, 'gtol': 1e-9})
            z = res.x
            penalty *= 3.0
            
        # Safety repair for numerical tolerance
        while compute_penalty(z) > 1e-5:
            z[2::3] *= 0.98
            if z[2::3].min() < 1e-4:
                break
                
        current_sum = np.sum(z[2::3])
        if current_sum > best_sum:
            best_sum = current_sum
            best_z = z.copy()
            
    centers = np.column_stack([best_z[0::3], best_z[1::3]])
    radii = best_z[2::3]
    
    return centers, radii, np.sum(radii)
