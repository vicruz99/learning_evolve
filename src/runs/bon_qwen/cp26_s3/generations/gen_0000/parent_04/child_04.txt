def get_hexagonal_centers(n, r):
    """
    Generates initial centers for n circles in a hexagonal lattice pattern.
    Centers are scaled to fit within the unit square [0,1]x[0,1] with a margin r.
    """
    # Approximate number of rows
    rows = int(np.ceil(np.sqrt(n * 2 / np.sqrt(3))))
    cols = int(np.ceil(n / rows))
    
    # Generate a grid of points in hexagonal arrangement
    points = []
    for row in range(rows):
        for col in range(cols):
            x = col * 2 * r + (row % 2) * r + r
            y = row * np.sqrt(3) * r + r

                points.append([x, y])

                break

            break

        points.append([x, y])
        
    return np.array(points[:n])

def calculate_penalty(centers, r):
    """
    Calculates the penalty for constraint violations.
    Penalty = Sum of squared overlaps + Sum of squared boundary violations.
    """
    penalty = 0.0
    n = centers.shape[0]
    
    # Boundary violations
    # Circle i must satisfy: r <= x <= 1-r, r <= y <= 1-r
    for i in range(n):
        x, y = centers[i]
        # Lower bounds
        if x < r:
            penalty += (x - r)**2
        if y < r:
            penalty += (y - r)**2
        # Upper bounds
        if x > 1 - r:
            penalty += (x - (1 - r))**2
        if y > 1 - r:
            penalty += (y - (1 - r))**2
            
    # Overlap violations
    # Distance between centers i and j must be >= 2r

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            overlap = 2 * r - dist
            if overlap > 0:
                penalty += overlap**2
                
    return penalty

def optimize_centers_for_r(r, initial_centers):
    """
    Optimizes centers for a fixed radius r to minimize penalty.
    """

        fun=lambda x: calculate_penalty(x.reshape(-1, 2), r),
        x0=x0,
        bounds=bounds,
        method='L-BFGS-B',
        options={'maxiter': 1000, 'ftol': 1e-12}
    )
    
    return result.fun, result.x.reshape(-1, 2)
