The decision variables are the centers $(x_i, y_i)$ and radii $r_i$ for each circle. The objective is to maximize $\sum r_i$. The constraints ensure circles remain within $[0,1]^2$ and do not overlap.

Key insights:
1. **Non-convex Constraints**: Pairwise non-overlap $(x_i-x_j)^2 + (y_i-y_j)^2 \ge (r_i+r_j)^2$ creates a non-convex feasible region. Gradient-based methods like SLSQP work well if provided with analytical gradients and good initial guesses.
2. **Initialization Strategies**: A single random start often converges to suboptimal local maxima (e.g., small circles). We use multiple diverse initial configurations:

   - Perturbed 5x5 grid

   - Corner-focused arrangements
3. **Analytical Jacobians**: Providing exact gradients for pairwise constraints significantly speeds up convergence and improves accuracy compared to finite-difference approximations.
4. **Post-processing**: After optimization, we verify constraints. If numerical tolerance causes slight violations, we uniformly scale down radii to guarantee validity before returning.

import numpy as np
from scipy.optimize import minimize

def pairwise_constraint_fun(var, i, j, N):
    idx_xi, idx_yi, idx_ri = 3*i, 3*i+1, 2*N+i
    idx_xj, idx_yj, idx_rj = 3*j, 3*j+1, 2*N+j
    xi, yi, ri = var[idx_xi], var[idx_yi], var[idx_ri]
    xj, yj, rj = var[idx_xj], var[idx_yj], var[idx_rj]
    return (xi-xj)**2 + (yi-yj)**2 - (ri+rj)**2

def pairwise_constraint_jac(var, i, j, N):
    jv = np.zeros(3*N)
    idx_xi, idx_yi, idx_ri = 3*i, 3*i+1, 2*N+i
    idx_xj, idx_yj, idx_rj = 3*j, 3*j+1, 2*N+j
    xi, yi, ri = var[idx_xi], var[idx_yi], var[idx_ri]
    xj, yj, rj = var[idx_xj], var[idx_yj], var[idx_rj]
    dx, dy = xi-xj, yi-yj
    r_sum = ri+rj
    jv[idx_xi] = 2*dx
    jv[idx_yi] = 2*dy
    jv[idx_xj] = -2*dx
    jv[idx_yj] = -2*dy
    jv[idx_ri] = -2*r_sum
    jv[idx_rj] = -2*r_sum
    return jv

def boundary_constraint_fun(var, i, N, side):
    idx_r = 2*N + i
    if side == 0: return var[3*i] - var[idx_r]
    if side == 1: return var[3*i+1] - var[idx_r]
    if side == 2: return 1.0 - var[3*i] - var[idx_r]
    return 1.0 - var[3*i+1] - var[idx_r]

def get_init_radii(centers, N):
    radii = np.zeros(N)
    for i in range(N):
        d_bound = min(centers[i,0], centers[i,1], 1-centers[i,0], 1-centers[i,1])
        dists = np.sqrt(np.sum((centers - centers[i])**2, axis=1))
        dists[i] = np.inf
        radii[i] = min(d_bound, np.min(dists)/2.0)
    return radii * 0.98

def run_packing():
    np.random.seed(42)
    N = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    def objective(var):
        return -np.sum(var[2*N:])
        
    cons = []
    
    # Boundary constraints
    for i in range(N):
        for side in range(4):
            cons.append({
                'type': 'ineq',
                'fun': lambda v, i=i, s=side: boundary_constraint_fun(v, i, N, s)
            })
            
    # Pairwise non-overlap constraints with analytical jacobian
    for i in range(N):
        for j in range(i+1, N):
            cons.append({
                'type': 'ineq',
                'fun': lambda v, i=i, j=j: pairwise_constraint_fun(v, i, j, N),
                'jac': lambda v, i=i, j=j: pairwise_constraint_jac(v, i, j, N)
            })
            
    bounds = [(0, 1)]*(2*N) + [(0, 0.5)]*N
    
    strategies = []
    
    # 1. Random initializations
    for _ in range(3):
        c = np.random.uniform(0.1, 0.9, (N, 2))
        strategies.append(np.concatenate([c.ravel(), get_init_radii(c, N)]))
        
    # 2. Grid-based initializations
    for _ in range(3):
        pts = np.array([[x, y] for y in np.linspace(0.1, 0.9, 5) for x in np.linspace(0.1, 0.9, 5)])
        pts = np.vstack([pts, [[0.5, 0.5]]])
        c = pts[:N] + np.random.uniform(-0.04, 0.04, (N, 2))
        c = np.clip(c, 0.05, 0.95)
        strategies.append(np.concatenate([c.ravel(), get_init_radii(c, N)]))
        
    # 3. Hexagonal lattice initializations
    for _ in range(3):
        pts = []
        y = 0.12
        row = 0
        while y < 0.9 and len(pts) < N:
            x = 0.12 if row % 2 == 0 else 0.17
            while x < 0.9 and len(pts) < N:
                pts.append([x, y])
                x += 0.22
            y += 0.19
            row += 1
        c = np.array(pts) + np.random.uniform(-0.02, 0.02, (N, 2))
        c = np.clip(c, 0.05, 0.95)
        strategies.append(np.concatenate([c.ravel(), get_init_radii(c, N)]))
        
    # 4. Corner-focused initializations
    for _ in range(3):
        c = np.array([[0.15, 0.15], [0.85, 0.15], [0.15, 0.85], [0.85, 0.85]])
        rest = np.random.uniform(0.3, 0.7, (N-4, 2))
        c = np.vstack([c, rest])
        c += np.random.uniform(-0.02, 0.02, (N, 2))
        c = np.clip(c, 0.05, 0.95)
        strategies.append(np.concatenate([c.ravel(), get_init_radii(c, N)]))
        
    for x0 in strategies:
        res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 2000, 'ftol': 1e-13, 'disp': False})
        cur_sum = -res.fun
        if cur_sum > best_sum:
            best_sum = cur_sum
            best_centers = res.x[:2*N].reshape(N, 2)
            best_radii = res.x[2*N:]
            
    if best_centers is None:
        centers = np.random.uniform(0.2, 0.8, (N, 2))
        radii = np.full(N, 0.05)
        return centers, radii, 0.0
        
    # Safety validation & correction
    valid = True
    for i in range(N):
        x, y, r = best_centers[i,0], best_centers[i,1], best_radii[i]
        if r > x or r > y or r > 1-x or r > 1-y:
            valid = False; break
    if valid:
        for i in range(N):
            for j in range(i+1, N):
                dx = best_centers[i,0] - best_centers[j,0]
                dy = best_centers[i,1] - best_centers[j,1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < best_radii[i] + best_radii[j]:
                    valid = False; break
            if not valid: break
            
    if not valid:
        scale = 1.0
        for _ in range(50):
            valid = True
            for i in range(N):
                r = best_radii[i] * scale
                if r > best_centers[i,0] or r > best_centers[i,1] or r > 1-best_centers[i,0] or r > 1-best_centers[i,1]:
                    valid = False; break
            if not valid: break
            if valid:
                for i in range(N):
                    for j in range(i+1, N):
                        dx = best_centers[i,0] - best_centers[j,0]
                        dy = best_centers[i,1] - best_centers[j,1]
                        dist = np.sqrt(dx*dx + dy*dy)
                        if dist < (best_radii[i] + best_radii[j]) * scale:
                            valid = False; break
                    if not valid: break
            if valid:
                break
            scale *= 0.95
        best_radii *= scale
        best_sum = np.sum(best_radii)
        
    return best_centers, best_radii, best_sum
