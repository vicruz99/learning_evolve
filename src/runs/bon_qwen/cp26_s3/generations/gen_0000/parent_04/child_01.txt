

To maximize the sum of radii for 26 circles in a unit square, we use an equal-radius assumption where $\sum r_i = 26r$. This transforms the problem into maximizing the radius $r$ while satisfying non-overlap and boundary constraints. 

The strategy involves:
1.  **Initialization**: Placing 26 centers on a uniform $6 \times 5$ grid to ensure a high-quality starting point.
2.  **Iterative Expansion**: Gradually increasing the target radius $r$ from a small value up to a high target.
3.  **Energy Minimization**: For each $r$ increment, applying a repulsive force-based relaxation to push overlapping circles apart and away from the boundaries.
4.  **Validation**: Returning the highest valid radius configuration found, with a slight margin to account for numerical precision.

```python
import numpy as np
from scipy.optimize import minimize

def run_packing():
    n = 26
    
    # 1. Initialize centers on a 6x5 grid (taking first 26)
    x_grid = np.linspace(0.15, 0.85, 6)
    y_grid = np.linspace(0.15, 0.85, 5)
    centers = np.empty((30, 2))
    idx = 0
    for y in y_grid:
        for x in x_grid:
            if idx < n:
                centers[idx] = [x, y]
                idx += 1
    centers = centers[:n]
    
    # 2. Iterative expansion and relaxation
    r_current = 0.05
    r_target = 0.11
    steps_per_incr = 100
    
    while r_current < r_target:
        r_current += 0.0002
        
        for _ in range(steps_per_incr):
            # Calculate forces
            forces = np.zeros_like(centers)
            
            # Inter-circle repulsion
            for i in range(n):
                for j in range(i + 1, n):
                    diff = centers[i] - centers[j]
                    dist = np.linalg.norm(diff)
                    if dist < 2 * r_current and dist > 1e-9:
                        overlap = 2 * r_current - dist
                        # Force magnitude proportional to overlap
                        force = (diff / dist) * overlap * 50.0
                        forces[i] += force
                        forces[j] -= force
            
            # Boundary repulsion
            for i in range(n):
                x, y = centers[i]
                # Left
                if x < r_current:
                    forces[i, 0] += (r_current - x) * 200.0
                # Right
                if x > 1 - r_current:
                    forces[i, 0] -= (x - (1 - r_current)) * 200.0
                # Bottom
                if y < r_current:
                    forces[i, 1] += (r_current - y) * 200.0
                # Top
                if y > 1 - r_current:
                    forces[i, 1] -= (y - (1 - r_current)) * 200.0
            
            # Update centers (learning rate 0.001)
            centers += forces * 0.001
            # Ensure centers stay within [0, 1]
            centers = np.clip(centers, 0, 1)
            
            # Reduce learning rate over time to settle
            if r_current > 0.08:
                centers += forces * 0.0005

    # Final optimization using scipy to polish the solution
    def objective(v):
        c = v[:2*n].reshape((n, 2))
        r = v[2*n]
        penalty = 0.0
        # Boundary penalty
        for i in range(n):
            x, y = c[i]
            if x < r: penalty += (r - x)**2 * 1000
            if x > 1 - r: penalty += (x - (1 - r))**2 * 1000
            if y < r: penalty += (r - y)**2 * 1000
            if y > 1 - r: penalty += (y - (1 - r))**2 * 1000
        # Overlap penalty
        for i in range(n):
            for j in range(i + 1, n):
                d = np.linalg.norm(c[i] - c[j])
                if d < 2 * r:
                    penalty += (2 * r - d)**2 * 1000
        return -r + penalty

    x0 = np.concatenate([centers.flatten(), [r_current]])
    bounds = [(0, 1)] * (2 * n) + [(0, 0.5)]
    res = minimize(objective, x0, method='L-BFGS-B', bounds=bounds, options={'maxiter': 5000})
    
    opt_centers = res.x[:2*n].reshape((n, 2))
    opt_r = res.x[2*n]
    
    # Safety margin for numerical validation
    final_r = max(0.01, opt_r - 1e-5)
    radii = np.full(n, final_r)
    
    return opt_centers, radii, np.sum(radii)
```
