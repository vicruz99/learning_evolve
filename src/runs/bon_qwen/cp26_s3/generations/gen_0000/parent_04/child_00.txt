        np.fill_diagonal(dists_mat, np.inf)
        min_neighbor_dists = np.min(dists_mat, axis=1)
        
        radii_new = np.minimum(min_bounds, min_neighbor_dists / 2.0)
        
        new_sum = np.sum(radii_new)
        
        if new_sum > current_sum + 1e-9:
            current_sum = new_sum
            # Accept
        else:
            # Reject
            centers[i] = old_pos
            
        step_size *= 0.99995

    # Final computation of radii
    radii = np.zeros(n)
    # Recompute properly
    bounds = np.column_stack([
        centers[:, 0], 
        1.0 - centers[:, 0], 
        centers[:, 1], 
        1.0 - centers[:, 1]
    ])
    min_bounds = np.min(bounds, axis=1)
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists_mat = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists_mat, np.inf)
    min_neighbor_dists = np.min(dists_mat, axis=1)
    
    radii = np.minimum(min_bounds, min_neighbor_dists / 2.0)
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
