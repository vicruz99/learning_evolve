import numpy as np

def get_initial_guess(n, seed=None):
    """
    Generates an initial guess for centers and radii.
    Uses a grid-like structure perturbed by noise to help with packing.
    """
    if seed is not None:
        np.random.seed(seed)
    
    # Start with a grid arrangement
    # Approximate square root of n to determine grid size
    rows = int(np.ceil(np.sqrt(n)))
    cols = rows
    
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    idx = 0
    for r in range(rows):
        for c in range(cols):
            if idx >= n:
                break
            # Place centers roughly in a grid, shifted to allow space
            # Using a 0.1 offset from edges to start safely inside
            x = 0.1 + c * 0.8 / max(cols - 1, 1) if cols > 1 else 0.5
            y = 0.1 + r * 0.8 / max(rows - 1, 1) if rows > 1 else 0.5
            
            # Add some random perturbation
            x += np.random.uniform(-0.05, 0.05)
            y += np.random.uniform(-0.05, 0.05)
            
            centers[idx] = [x, y]
            # Initial radius small to avoid immediate heavy penalty
            radii[idx] = 0.02 
            idx += 1
            
    return centers, radii

def compute_penalty(centers, radii, alpha=1e6):
    """
    Computes the penalty for overlaps and boundary violations.
    """
    n = centers.shape[0]
    penalty = 0.0
    
    # 1. Overlap Penalty
    # Calculate pairwise distances
    # Vectorized distance calculation for performance
    # diffs shape: (n, n, 2)
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists_sq = np.sum(diffs**2, axis=2)
    
    # We only care about lower triangle (i > j) to avoid double counting
    # and self-distance
    i_indices, j_indices = np.tril_indices(n, k=-1)
    
    dists = np.sqrt(dists_sq[i_indices, j_indices])
    r_sums = radii[i_indices] + radii[j_indices]
    
    # Overlap amount: (r_i + r_j) - dist
    overlaps = np.maximum(0.0, r_sums - dists)
    penalty += np.sum(overlaps**2)
    
    # 2. Boundary Penalty
    # Circles must be within [0, 1] x [0, 1]
    # Constraints: x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
    # Violation is max(0, r - x), etc.
    
    # x >= r  => r - x <= 0
    pen_x_min = np.maximum(0.0, radii - centers[:, 0])
    # x <= 1 - r => r - (1 - x) <= 0 => r + x - 1 <= 0
    pen_x_max = np.maximum(0.0, radii - (1.0 - centers[:, 0]))
    # y >= r
    pen_y_min = np.maximum(0.0, radii - centers[:, 1])
    # y <= 1 - r
    pen_y_max = np.maximum(0.0, radii - (1.0 - centers[:, 1]))
    
    boundary_penalty = (np.sum(pen_x_min**2) + np.sum(pen_x_max**2) + 
                        np.sum(pen_y_min**2) + np.sum(pen_y_max**2))
    
    penalty += boundary_penalty
    
    return alpha * penalty

def objective_function(vars_flat, n=26, alpha=1e6):
    """
    Objective function to minimize: -sum(radii) + penalty
    """
    centers = np.reshape(vars_flat[:2*n], (n, 2))
    radii = vars_flat[2*n:]
    
    # Primary objective: Maximize sum of radii (minimize negative sum)
    obj = -np.sum(radii)
    
    # Add penalty for infeasibility
    obj += compute_penalty(centers, radii, alpha)
    
    return obj
