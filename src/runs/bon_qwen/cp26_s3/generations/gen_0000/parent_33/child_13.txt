import numpy as np
from scipy.optimize import minimize

def _objective(v):
    """Objective function: maximize sum of radii -> minimize negative sum."""
    # Radii start at index 52 (2 * 26)
    return -np.sum(v[52:])

def _constraints(v):
    """Constraint function: boundary and non-overlap conditions."""
    n = 26
    vals = []
    # Boundary constraints
    for i in range(n):
        xi, yi = v[2*i], v[2*i+1]
        ri = v[52+i]
        vals.extend([xi - ri, 1.0 - xi - ri, yi - ri, 1.0 - yi - ri])
    
    # Non-overlap constraints: distance^2 >= (r_i + r_j)^2
    for i in range(n):
        for j in range(i+1, n):
            d2 = (v[2*i] - v[2*j])**2 + (v[2*i+1] - v[2*j+1])**2
            r_sum = v[52+i] + v[52+j]
            vals.append(d2 - r_sum**2)
            
    return np.array(vals)

def run_packing():
    n = 26
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Bounds: centers in [0,1], radii in [0, 0.5]
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
    constraint_def = {'type': 'ineq', 'fun': _constraints}
    
    # Try multiple seeds to escape local minima
    for seed in range(8):
        np.random.seed(seed)
        
        # Initial layout: Perturbed 5x5 grid + 1 extra circle
        centers = np.zeros((n, 2))
        idx = 0
        for i in range(5):
            for j in range(5):
                if idx < n:
                    centers[idx] = [0.1 + i * 0.2, 0.1 + j * 0.2]
                    idx += 1
        if idx < n:
            centers[idx:] = np.random.uniform(0.1, 0.9, (n - idx, 2))
            
        # Add small random noise to break symmetry and aid optimization
        centers += np.random.uniform(-0.015, 0.015, centers.shape)
        
        # Initial radii
        radii = np.full(n, 0.05)
        params0 = np.concatenate([centers.ravel(), radii])
        
        try:
            # Run optimization
            res = minimize(
                _objective,
                params0,
                method='SLSQP',
                bounds=bounds,
                constraints=constraint_def,
                options={'maxiter': 3000, 'ftol': 1e-10}
            )
            
            if res.success or (np.isfinite(res.fun) and res.fun < -2.0):
                c_opt = res.x[:2*n].reshape(n, 2)
                r_opt = res.x[2*n:]
                
                # Validate packing

                    s = np.sum(r_opt)
                    if s > best_sum:
                        best_sum = s
                        best_centers = c_opt.copy()
                        best_radii = r_opt.copy()
        except Exception:
            continue
            
    # Fallback initialization if optimization fails (unlikely)
    if best_centers is None:
        best_centers = np.random.uniform(0.1, 0.9, (n, 2))
        best_radii = np.full(n, 0.01)
        best_sum = n * 0.01
        
    return best_centers, best_radii, best_sum
