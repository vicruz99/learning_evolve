I'll use a two-phase optimization approach:

1. **LP-based inner optimization**: For fixed centers, the maximum radii can be found via Linear Programming (LP). This gives the optimal radii for any given center configuration.

2. **Center optimization**: I optimize the 26 center positions using multiple methods:
   - Nelder-Mead (robust to non-smooth objective)
   - L-BFGS-B (fast local refinement)
   - Multiple restarts from diverse initializations (grid, hexagonal, random)

3. **Penalty-based refinement**: After LP-based optimization, I'll do a second pass with a smooth penalty-based objective to further improve.

4. **Initialization strategies**: Grid patterns (various spacings), hexagonal patterns, and random seeds to explore the solution space thoroughly.

The key insight: for fixed centers, finding optimal radii is a linear program. The outer problem optimizes centers to maximize this LP objective.

import numpy as np
from scipy.optimize import minimize, linprog


def compute_max_radii_lp(centers, n):
    """Given fixed centers, find maximum radii using LP.
    
    Maximizes sum of radii subject to:
    - r_i + r_j <= ||c_i - c_j|| for all i < j (non-overlap)
    - r_i <= min(x_i, 1-x_i, y_i, 1-y_i) for all i (boundary)
    - r_i >= 0 for all i
    """
    c_obj = -np.ones(n)
    
    A_ub_rows = []
    b_ub_vals = []
    
    # Pairwise non-overlap: r_i + r_j <= dist(i,j)
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub_rows.append(row)
            b_ub_vals.append(dist)
    
    # Boundary constraints: r_i <= dist to each wall
    for i in range(n):
        wall_dists = [
            centers[i, 0],
            1.0 - centers[i, 0],
            centers[i, 1],
            1.0 - centers[i, 1]
        ]
        for val in wall_dists:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub_rows.append(row)
            b_ub_vals.append(max(val, 1e-12))
    
    A_ub = np.array(A_ub_rows)
    b_ub = np.array(b_ub_vals)
    
    bounds = [(0.0, None) for _ in range(n)]
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success and result.fun is not None:
        return result.x
    else:
        # Fallback: compute radii manually
        return np.full(n, 0.01)


def neg_total_radii(params, n):
    """Objective function: negative sum of radii for given centers."""
    centers = params.reshape((n, 2))
    radii = compute_max_radii_lp(centers, n)
    return -np.sum(radii)


def objective_penalty(params, n, pw):
    """Smooth penalized objective for centers + radii."""
    centers = params[:2 * n].reshape((n, 2))
    radii = np.maximum(params[2 * n:], 1e-10)
    
    obj = -np.sum(radii)
    penalty = 0.0
    
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy + 1e-15)
            violation = radii[i] + radii[j] - dist
            if violation > 0:
                penalty += pw * violation * violation
    
    for i in range(n):
        r = radii[i]
        x = centers[i, 0]
        y = centers[i, 1]
        
        v = x - r
        if v < 0:
            penalty += pw * v * v
        v = x + r - 1.0
        if v > 0:
            penalty += pw * v * v
        v = y - r
        if v < 0:
            penalty += pw * v * v
        v = y + r - 1.0
        if v > 0:
            penalty += pw * v * v
    
    return obj + penalty


def make_grid_init(n, spacing):
    """Create grid-based initialization."""
    centers = np.zeros((n, 2))
    idx = 0
    y = spacing
    while y <= 1.0 + 1e-9 and idx < n:
        x = spacing
        while x <= 1.0 + 1e-9 and idx < n:
            centers[idx, 0] = np.clip(x, 0.01, 0.99)
            centers[idx, 1] = np.clip(y, 0.01, 0.99)
            idx += 1
            x += spacing
        y += spacing
    return centers[:n]


def make_hex_init(n):
    """Create hexagonal initialization."""
    centers = np.zeros((n, 2))
    idx = 0
    
    # Hexagonal packing: alternating rows
    y_step = np.sqrt(3.0) / 2.0 * 0.45
    x_step = 0.40
    
    y = 0.50
    offset = False
    while idx < n and y <= 1.0 + 1e-9:
        start_x = 0.55 if offset else 0.35
        x = start_x
        while idx < n and x <= 1.0 + 1e-9:
            centers[idx, 0] = np.clip(x, 0.05, 0.95)
            centers[idx, 1] = np.clip(y, 0.05, 0.95)
            idx += 1
            x += x_step
        offset = not offset
        y += y_step
    
    return centers[:n]


def make_random_init(n, seed):
    """Create random initialization."""
    rng = np.random.RandomState(seed)
    centers = rng.uniform(0.15, 0.85, size=(n, 2))
    return centers


def make_dense_hex_init(n):
    """Create a denser hexagonal pattern targeting ~26 circles."""
    centers = np.zeros((n, 2))
    idx = 0
    
    # Target spacing for ~26 circles
    y_step = 0.28
    x_step = 0.32
    
    y = 0.18
    offset = False
    while idx < n and y <= 1.0 + 1e-9:
        if offset:
            start_x = 0.33
        else:
            start_x = 0.17
        x = start_x
        while idx < n and x <= 1.0 + 1e-9:
            centers[idx, 0] = np.clip(x, 0.05, 0.95)
            centers[idx, 1] = np.clip(y, 0.05, 0.95)
            idx += 1
            x += x_step
        offset = not offset
        y += y_step
    
    return centers[:n]


def init_g033(n):
    return make_grid_init(n, 0.33)

def init_g035(n):
    return make_grid_init(n, 0.35)

def init_g037(n):
    return make_grid_init(n, 0.37)

def init_g040(n):
    return make_grid_init(n, 0.40)

def init_g042(n):
    return make_grid_init(n, 0.42)

def init_hex1(n):
    return make_hex_init(n)

def init_hex2(n):
    return make_dense_hex_init(n)

def init_r42(n):
    return make_random_init(n, 42)

def init_r123(n):
    return make_random_init(n, 123)

def init_r7(n):
    return make_random_init(n, 7)

def init_r999(n):
    return make_random_init(n, 999)


def optimize_centers_nelder(centers_init, n, maxiter=5000):
    """Optimize centers using Nelder-Mead."""
    params_init = centers_init.flatten()
    bounds = [(0.02, 0.98)] * (2 * n)
    
    # Clip to bounds
    params_init = np.clip(params_init, 0.02, 0.98)
    
    result = minimize(neg_total_radii, params_init, args=(n,),
                      method='Nelder-Mead',
                      options={'maxiter': maxiter, 'xatol': 1e-8, 'fatol': 1e-12, 'adapt': True})
    
    centers = np.clip(result.x.reshape((n, 2)), 0.01, 0.99)
    return centers


def optimize_centers_lbfgsb(centers_init, n, maxiter=10000):
    """Optimize centers using L-BFGS-B."""
    params_init = centers_init.flatten()
    bounds = [(0.01, 0.99)] * (2 * n)
    params_init = np.clip(params_init, 0.01, 0.99)
    
    result = minimize(neg_total_radii, params_init, args=(n,),
                      bounds=bounds, method='L-BFGS-B',
                      options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-8})
    
    centers = np.clip(result.x.reshape((n, 2)), 0.01, 0.99)
    return centers


def penalty_optimization(centers_init, n, pw=50000, maxiter=15000):
    """Refine using penalty-based smooth optimization."""
    radii_init = compute_max_radii_lp(centers_init, n)
    
    params_init = np.concatenate([centers_init.flatten(), radii_init])
    
    bounds = [(0.01, 0.99)] * (2 * n) + [(0.001, 0.5)] * n
    params_init = np.clip(params_init[:2*n], 0.01, 0.99)
    params_init[2*n:] = np.clip(params_init[2*n:], 0.001, 0.5)
    
    def obj_func(p):
        return objective_penalty(p, n, pw)
    
    result = minimize(obj_func, params_init, bounds=bounds,
                      method='L-BFGS-B',
                      options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10})
    
    centers = np.clip(result.x[:2*n].reshape((n, 2)), 0.001, 0.999)
    radii = np.maximum(result.x[2*n:], 0.0001)
    
    return centers, radii


def perturb_and_optimize(centers, n, scale=0.02, seed=0):
    """Perturb centers slightly and re-optimize."""
    rng = np.random.RandomState(seed)
    perturbed = centers + rng.uniform(-scale, scale, size=centers.shape)
    perturbed = np.clip(perturbed, 0.05, 0.95)
    
    # Quick Nelder-Mead refinement
    result = optimize_centers_nelder(perturbed, n, maxiter=2000)
    return result


def run_packing():
    n = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    init_funcs = [
        init_g033, init_g035, init_g037, init_g040, init_g042,
        init_hex1, init_hex2,
        init_r42, init_r123, init_r7, init_r999,
    ]
    
    print("Phase 1: LP-based center optimization")
    
    for init_fn in init_funcs:
        try:
            centers_init = init_fn(n)
            
            # Try Nelder-Mead first (robust)
            centers_nm = optimize_centers_nelder(centers_init, n, maxiter=8000)
            radii_nm = compute_max_radii_lp(centers_nm, n)
            total_nm = np.sum(radii_nm)
            
            if total_nm > best_sum:
                best_sum = total_nm
                best_centers = centers_nm.copy()
                best_radii = radii_nm.copy()
            
            # Then try L-BFGS-B from same start (faster convergence)
            centers_lb = optimize_centers_lbfgsb(centers_nm, n, maxiter=5000)
            radii_lb = compute_max_radii_lp(centers_lb, n)
            total_lb = np.sum(radii_lb)
            
            if total_lb > best_sum:
                best_sum = total_lb
                best_centers = centers_lb.copy()
                best_radii = radii_lb.copy()
                
        except Exception as e:
            pass
    
    print(f"After Phase 1: sum = {best_sum:.6f}")
    
    if best_centers is not None:
        print("Phase 2: Perturbation search")
        # Perturbation search
        for seed in range(20):
            try:
                perturbed = perturb_and_optimize(best_centers, n, scale=0.015, seed=seed)
                radii_p = compute_max_radii_lp(perturbed, n)
                total_p = np.sum(radii_p)
                
                if total_p > best_sum:
                    best_sum = total_p
                    best_centers = perturbed.copy()
                    best_radii = radii_p.copy()
            except Exception:
                continue
        
        print(f"After Phase 2: sum = {best_sum:.6f}")
        
        print("Phase 3: Penalty-based refinement")
        # Penalty-based refinement
        for pw in [10000, 50000, 100000]:
            try:
                centers_ref, radii_ref = penalty_optimization(best_centers, n, pw=pw, maxiter=10000)
                # Verify with LP
                radii_check = compute_max_radii_lp(centers_ref, n)
                total_check = np.sum(radii_check)
                
                if total_check > best_sum:
                    best_sum = total_check
                    best_centers = centers_ref.copy()
                    best_radii = radii_check.copy()
            except Exception:
                continue
        
        print(f"After Phase 3: sum = {best_sum:.6f}")
    
    return best_centers, best_radii, best_sum
