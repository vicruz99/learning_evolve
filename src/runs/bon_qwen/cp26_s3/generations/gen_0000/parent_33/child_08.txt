Since equal circles pack most efficiently in a dense lattice, we initialize the configuration with a hexagonal grid. We then optimize the coordinates $(x_i, y_i)$ and radii $r_i$ using the `scipy.optimize.minimize` solver with the 'SLSQP' method, which effectively handles the non-convex inequality constraints for non-overlap and boundary containment.

<strategy>
1.  **Initialization**: Create a hexagonal grid pattern for 26 circles with a small initial radius ($r \approx 0.095$) to ensure a valid, dense starting point.
2.  **Formulation**: Define variables for all centers and radii. The objective is to minimize the negative sum of radii (equivalent to maximizing the sum).
3.  **Constraints**: 
    -   **Non-overlap**: For every pair of circles, the squared distance between centers must be at least the square of the sum of their radii: $\|c_i - c_j\|^2 \ge (r_i + r_j)^2$.
    -   **Boundary**: Each circle must be within the unit square: $r_i \le x_i \le 1-r_i$ and $r_i \le y_i \le 1-r_i$.
4.  **Optimization**: Run the SLSQP solver with high precision tolerances.
5.  **Finalization**: Extract the optimized parameters, enforce strict boundary conditions (clipping), and return the final configuration.

import numpy as np
from scipy.optimize import minimize

# Global constant for the number of circles
N_CIRCLES = 26

def objective(vars):
    """
    Objective function: Minimize negative sum of radii (maximize sum of radii).
    vars is a flattened array of [x1, y1, r1, x2, y2, r2, ...]
    """
    # Radii are at indices 2, 5, 8, ... (step of 3)
    return -np.sum(vars[2::3])

def pairwise_constraints(vars):
    """
    Calculates pairwise overlap constraints.
    Constraint: (r_i + r_j)^2 - dist^2 <= 0
    """
    n = N_CIRCLES
    c = vars.reshape(n, 3)
    xs = c[:, 0]
    ys = c[:, 1]
    rs = c[:, 2]
    violations = []
    for i in range(n):
        for j in range(i + 1, n):
            dx = xs[i] - xs[j]
            dy = ys[i] - ys[j]
            dist_sq = dx*dx + dy*dy
            r_sum = rs[i] + rs[j]
            # We want (r_sum)^2 <= dist_sq, so (r_sum)^2 - dist_sq <= 0
            violations.append(r_sum*r_sum - dist_sq)
    return np.array(violations)

def boundary_constraints(vars):
    """
    Calculates boundary constraints.
    Constraints: r_i - x_i <= 0, x_i + r_i - 1 <= 0, etc.
    """
    n = N_CIRCLES
    c = vars.reshape(n, 3)
    xs = c[:, 0]
    ys = c[:, 1]
    rs = c[:, 2]
    v = []
    for i in range(n):
        v.append(rs[i] - xs[i])          # r - x <= 0
        v.append(xs[i] + rs[i] - 1.0)    # x + r - 1 <= 0
        v.append(rs[i] - ys[i])          # r - y <= 0
        v.append(ys[i] + rs[i] - 1.0)    # y + r - 1 <= 0
    return np.array(v)

def pair_con_func(vars):
    """Wrapper for SLSQP inequality constraints (expects g(x) >= 0)"""
    return -pairwise_constraints(vars)

def bound_con_func(vars):
    """Wrapper for SLSQP inequality constraints (expects g(x) >= 0)"""
    return -boundary_constraints(vars)

def run_packing():
    """
    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    Returns: (centers, radii, sum_radii)
    """
    n = N_CIRCLES
    
    # --- Initialization ---
    # Start with a hexagonal grid pattern for dense packing
    r_init = 0.095 
    centers = []
    y = r_init
    row_idx = 0
    
    # Generate hexagonal points
    while len(centers) < n:
        # Even rows start at r, odd rows shifted by r (start at 2r)
        x_start = r_init if row_idx % 2 == 0 else 2 * r_init
        x = x_start
        while x <= 1 - r_init:
            centers.append([x, y])
            x += 2 * r_init
        y += r_init * np.sqrt(3)
        row_idx += 1
        
    centers = np.array(centers[:n])
    radii = np.full(n, r_init)
    
    # Flatten variables: [x1, y1, r1, x2, y2, r2, ...]
    x0 = np.zeros(3 * n)
    for i in range(n):
        x0[3*i] = centers[i, 0]
        x0[3*i+1] = centers[i, 1]
        x0[3*i+2] = radii[i]
    
    # Bounds for variables
    bounds = []
    for i in range(n):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 0.5)) # r (cannot be larger than 0.5 in unit square)

    constraints = [
        {'type': 'ineq', 'fun': pair_con_func},
        {'type': 'ineq', 'fun': bound_con_func}
    ]

    # --- Optimization ---
    try:
        # SLSQP solver with high tolerance
        res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints, 
                       options={'maxiter': 2000, 'ftol': 1e-14, 'disp': False})
        x_opt = res.x
    except Exception:
        # Fallback to initial configuration if optimization fails
        x_opt = x0

    # --- Post-processing ---
    c_flat = x_opt.reshape(n, 3)
    centers_final = c_flat[:, :2]
    radii_final = c_flat[:, 2]
    
    # Ensure radii are non-negative
    radii_final = np.maximum(radii_final, 0)
    
    # Final strict boundary adjustment
    for i in range(n):
        r = radii_final[i]
        x, y = centers_final[i]
        # Project center into valid box [r, 1-r] x [r, 1-r]
        if r < 0.5:
            centers_final[i, 0] = np.clip(x, r, 1 - r)
            centers_final[i, 1] = np.clip(y, r, 1 - r)
        else:
            # Fallback if radius is unexpectedly large
            centers_final[i, 0] = 0.5
            centers_final[i, 1] = 0.5
            radii_final[i] = 0.5

    sum_radii = np.sum(radii_final)
    return centers_final, radii_final, sum_radii
