import scipy.optimize

def calculate_loss(theta, n, mu):

    radii = theta[2*n:]
    
    # Objective: minimize -sum(radii) + penalty for overlaps
    loss = -np.sum(radii)
    
    # Compute pairwise distances efficiently
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2) + 1e-16)

    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    active = overlaps[mask]
    pos = np.maximum(0.0, active)
    
    # Quadratic penalty for overlaps
    return float(loss + mu * np.sum(pos**2))
