import numpy as np

def run_packing():
    """
    Packs 26 circles in a unit square [0,1]x[0,1] to maximize the sum of radii.
    Uses a hexagonal initialization followed by an iterative expand-and-push solver.
    """
    n = 26
    
    # --- 1. Initialization ---
    # We arrange 26 circles in a staggered 6-row pattern (5, 4, 5, 4, 5, 3)
    # to approximate a hexagonal lattice, which is efficient for space usage.
    
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.08) # Initial radius, safe for 6 rows
    
    row_counts = [5, 4, 5, 4, 5, 3]
    idx = 0
    
    # Calculate vertical spacing (sqrt(3)/2 * diameter)
    dy = np.sqrt(3) * 0.15 
    
    y_pos = 0.15 # Starting Y for the first row
    
    for i, count in enumerate(row_counts):
        # Center the row horizontally based on the number of circles
        # Total width of 'count' circles with spacing 2r (diameter 0.16)
        # We want to fit them in [0, 1]. 
        # Width occupied = count * 0.16. 
        # Start X = (1 - count * 0.16) / 2 + 0.08 (center of first circle)
        
        circle_diam = 0.16
        row_width = count * circle_diam
        start_x = (1.0 - row_width) / 2.0 + 0.08
        
        for j in range(count):
            if idx < n:
                centers[idx, 0] = start_x + j * circle_diam
                centers[idx, 1] = y_pos
                idx += 1
        
        # Offset Y for next row
        y_pos += dy
        # Offset X for next row to create staggering
        start_x += circle_diam / 2.0

    # --- 2. Optimization: Expand and Push ---
    
    max_iter = 3000
    # Initial growth rate per step
    growth_rate = 0.0004 
    
    for step in range(max_iter):
        # Adaptive growth: slow down as we approach jammed state
        current_growth = growth_rate * (1.0 - step / (2 * max_iter))
        if current_growth < 1e-6:
            current_growth = 1e-6
            
        # 2a. Expand Radii
        radii *= (1.0 + current_growth)
        
        # 2b. Collision Resolution (Multiple passes)
        # We iterate multiple times per growth step to resolve chain reactions
        for _ in range(15):
            # Resolve pairwise overlaps
            for i in range(n):
                for j in range(i + 1, n):
                    dx = centers[j, 0] - centers[i, 0]
                    dy = centers[j, 1] - centers[i, 1]
                    dist_sq = dx*dx + dy*dy
                    dist = np.sqrt(dist_sq)
                    
                    min_dist = radii[i] + radii[j]
                    
                    if dist < min_dist:
                        # Overlap detected
                        if dist < 1e-9:
                            # Handle coincident centers by random push
                            dx = np.random.randn()
                            dy = np.random.randn()
                            dist = np.sqrt(dx*dx + dy*dy)
                            if dist < 1e-9: dist = 1e-9
                        
                        # Vector pointing from i to j
                        ux = dx / dist
                        uy = dy / dist
                        
                        # Amount of overlap
                        overlap = min_dist - dist
                        
                        # Push apart: move each by half the overlap (plus epsilon)
                        move = (overlap * 0.5) + 1e-7
                        
                        centers[i, 0] -= ux * move
                        centers[i, 1] -= uy * move
                        centers[j, 0] += ux * move
                        centers[j, 1] += uy * move

            # 2c. Boundary Constraints
            # Ensure all circles are within [0, 1]x[0, 1]
            for i in range(n):
                r = radii[i]
                # Clamp x
                if centers[i, 0] < r:
                    centers[i, 0] = r
                elif centers[i, 0] > 1.0 - r:
                    centers[i, 0] = 1.0 - r
                
                # Clamp y
                if centers[i, 1] < r:
                    centers[i, 1] = r
                elif centers[i, 1] > 1.0 - r:
                    centers[i, 1] = 1.0 - r

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
