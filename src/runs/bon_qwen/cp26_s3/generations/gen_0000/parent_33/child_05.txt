from scipy.optimize import minimize

def get_initial_config():

    pts = []
    # Row structure that closely matches optimal hexagonal packing topology
    row_counts = [6, 5, 6, 5, 4]
    r_init = 0.06
    dx = 2.0 * r_init
    dy = np.sqrt(3.0) * r_init
    
    current_y = r_init + 0.01
    for idx, count in enumerate(row_counts):
        row_width = (count - 1) * dx
        start_x = (1.0 - row_width) / 2.0
        # Shift every other row for hexagonal packing
        if idx % 2 == 1:
            start_x += dx / 2.0
            
        for _ in range(count):
            pts.append([start_x, current_y])
            start_x += dx
        current_y += dy
        
    centers = np.array(pts)
    # Ensure initial guess is strictly inside the square
    centers = np.clip(centers, 0.05, 0.95)
    return centers, r_init

def evaluate_constraints(vars):
    """Evaluate inequality constraints for SLSQP."""
    n = 26
    x = vars[:n]
    y = vars[n:2*n]
    r = vars[2*n]
    
    c = []
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    for i in range(n):
        c.append(x[i] - r)
        c.append(1.0 - x[i] - r)
        c.append(y[i] - r)
        c.append(1.0 - y[i] - r)
        
    # Non-overlap constraints: distance >= 2r
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt((x[i] - x[j])**2 + (y[i] - y[j])**2)
            c.append(dist - 2.0 * r)
            
    return np.array(c)

def objective(vars):
    """Objective function: maximize radius r (minimize -r)."""
    return -vars[2*26]

def project_to_valid(centers, r_guess):
    """Ensure the packing is strictly valid by clipping and computing feasible radius."""
    n = centers.shape[0]
    # Clip centers to valid boundary region
    centers = np.clip(centers, r_guess, 1.0 - r_guess)
    
    # Compute minimum distance to boundaries and between circles
    min_dist = 1.0
    for i in range(n):
        # Distance to boundaries
        d_bound = min(centers[i, 0], 1.0 - centers[i, 0], 
                      centers[i, 1], 1.0 - centers[i, 1])
        min_dist = min(min_dist, d_bound)
        
        # Distance to other circles
        for j in range(i + 1, n):
            d = np.sqrt(np.sum((centers[i] - centers[j])**2))
            min_dist = min(min_dist, d)
            
    # The largest valid radius is half the minimum separation
    return min_dist / 2.0
