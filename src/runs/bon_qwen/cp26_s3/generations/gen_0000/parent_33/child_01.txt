import numpy as np
from scipy.optimize import minimize


def compute_overlap_penalty(centers, radii, n=26):
    """Compute total overlap between all circle pairs."""
    penalty = 0.0
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            required = radii[i] + radii[j]
            if dist < required:
                penalty += (required - dist) ** 2
    return penalty


def compute_boundary_penalty(centers, radii, n=26):
    """Compute penalty for circles outside the unit square."""
    penalty = 0.0
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if r < 0:
            penalty += r * r
        if x - r < 0:
            penalty += (x - r) ** 2
        if x + r > 1:
            penalty += (x + r - 1) ** 2
        if y - r < 0:
            penalty += (y - r) ** 2
        if y + r > 1:
            penalty += (y + r - 1) ** 2
    return penalty


def objective_function(params, n=26):
    """Objective: maximize sum of radii (minimize negative sum) with penalties."""
    centers = params[:2 * n].reshape(n, 2)
    radii = params[2 * n:]
    
    # Main objective: maximize sum of radii
    obj = -np.sum(radii)
    
    # Add penalties for violations
    obj += 50000 * compute_overlap_penalty(centers, radii, n)
    obj += 50000 * compute_boundary_penalty(centers, radii, n)
    
    # Penalty for negative radii
    for i in range(n):
        if radii[i] < 0:
            obj += 100000 * radii[i] * radii[i]
    
    return obj


def force_directed_optimize(centers, radii, n=26, iterations=5000, lr=0.01):
    """Force-directed optimization to resolve overlaps and expand radii."""
    for iteration in range(iterations):
        forces = np.zeros_like(centers)
        radius_grads = np.zeros(n)
        
        # Repulsion forces between circles
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                if dist < 1e-12:
                    dist = 1e-12
                    dx = 1e-12
                    dy = 0
                
                overlap = radii[i] + radii[j] - dist
                if overlap > 0:
                    force_strength = overlap * 50
                    fx = dx / dist * force_strength
                    fy = dy / dist * force_strength
                    forces[i, 0] += fx
                    forces[i, 1] += fy
                    forces[j, 0] -= fx
                    forces[j, 1] -= fy
        
        # Expansion forces (try to increase radii)
        for i in range(n):
            # Boundary limits
            max_r_boundary = min(centers[i, 0], 1 - centers[i, 0], 
                                  centers[i, 1], 1 - centers[i, 1])
            
            # Neighbor limits
            max_r_neighbor = max_r_boundary
            for j in range(n):
                if i != j:
                    dx = centers[i, 0] - centers[j, 0]
                    dy = centers[i, 1] - centers[j, 1]
                    dist = np.sqrt(dx * dx + dy * dy)
                    max_r_neighbor = min(max_r_neighbor, dist - radii[j])
            
            if max_r_neighbor > radii[i] + 1e-10:
                radius_grads[i] = 1.0
            elif max_r_neighbor < radii[i] - 1e-10:
                radius_grads[i] = -1.0
        
        # Update positions
        centers = centers + forces * lr
        
        # Update radii
        radii = radii + radius_grads * lr * 0.1
        
        # Ensure radii are non-negative
        radii = np.maximum(radii, 1e-6)
        
        # Clamp centers to valid range
        for i in range(n):
            centers[i, 0] = np.clip(centers[i, 0], radii[i] + 1e-10, 1 - radii[i] - 1e-10)
            centers[i, 1] = np.clip(centers[i, 1], radii[i] + 1e-10, 1 - radii[i] - 1e-10)
    
    return centers, radii


def init_grid_plus_gap():
    """Initialize with 5x5 grid plus gap-filling circle."""
    n = 26
    r = 0.098
    s = (1 - 2 * r) / 4
    centers = np.zeros((25, 2))
    radii = np.ones(25) * r
    
    idx = 0
    for i in range(5):
        for j in range(5):
            centers[idx, 0] = r + i * s
            centers[idx, 1] = r + j * s
            idx += 1
    
    # Add gap-filling circle
    gap_center = np.array([r + s / 2, r + s / 2])
    gap_radius = s / np.sqrt(2) - r
    
    centers = np.vstack([centers, gap_center.reshape(1, 2)])
    radii = np.append(radii, gap_radius)
    
    return centers, radii


def init_hexagonal():
    """Initialize with hexagonal-like arrangement."""
    n = 26
    r = 0.082
    centers = np.zeros((n, 2))
    radii = np.ones(n) * r
    
    # 5 rows: 6, 4, 6, 4, 6 = 26
    row_counts = [6, 4, 6, 4, 6]
    idx = 0
    spacing_x = 2 * r
    spacing_y = np.sqrt(3) * r
    
    # Center the arrangement
    for row_idx, count in enumerate(row_counts):
        offset_x = spacing_x / 2 if row_idx % 2 == 1 else 0
        total_width = (count - 1) * spacing_x + 2 * r
        start_x = (1 - total_width) / 2
        
        for col in range(count):
            x = start_x + offset_x + col * spacing_x
            y = r + row_idx * spacing_y
            if idx < n:
                centers[idx, 0] = x
                centers[idx, 1] = y
                idx += 1
    
    return centers, radii


def init_random_spread():
    """Initialize with well-spread random positions."""
    n = 26
    np.random.seed(42)
    
    # Place circles in a roughly uniform grid pattern with some randomness
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.08
    
    for i in range(n):
        row = i // 6
        col = i % 6
        base_x = (col + 0.5) / 6 + np.random.uniform(-0.02, 0.02)
        base_y = (row + 0.5) / 5 + np.random.uniform(-0.02, 0.02)
        centers[i, 0] = np.clip(base_x, 0.1, 0.9)
        centers[i, 1] = np.clip(base_y, 0.1, 0.9)
    
    return centers, radii


def local_optimize(centers, radii, n=26, maxiter=500):
    """Local optimization using scipy's SLSQP."""
    initial_params = np.concatenate([centers.flatten(), radii])
    
    # Bounds: centers in [0.001, 0.999], radii >= 0.001
    bounds = []
    for i in range(n):
        bounds.append((0.001, 0.999))  # x
        bounds.append((0.001, 0.999))  # y
    for i in range(n):
        bounds.append((0.001, 0.5))  # radius
    
    result = minimize(
        objective_function,
        initial_params,
        args=(n,),
        method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': maxiter, 'ftol': 1e-15}
    )
    
    optimized_centers = result.x[:2 * n].reshape(n, 2)
    optimized_radii = result.x[2 * n:]
    
    return optimized_centers, optimized_radii


def clamp_and_fix(centers, radii, n=26):
    """Clamp circles to be within bounds and fix overlaps."""
    # Clamp radii based on position
    for i in range(n):
        r = min(centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])
        radii[i] = max(radii[i], min(radii[i], r - 1e-12))
    
    # Clamp centers based on radius
    for i in range(n):
        centers[i, 0] = np.clip(centers[i, 0], radii[i] + 1e-12, 1 - radii[i] - 1e-12)
        centers[i, 1] = np.clip(centers[i, 1], radii[i] + 1e-12, 1 - radii[i] - 1e-12)
    
    return centers, radii
