def compute_objective(vars, n, lam):
    """
    Objective function to maximize sum of radii with penalty for violations.
    vars: flat array [x1, y1, r1, x2, y2, r2, ...]
    n: number of circles
    lam: penalty weight
    """
    xs = vars[0::3]
    ys = vars[1::3]
    rs = vars[2::3]
    
    pen = 0.0
    
    # Boundary penalties: ensure circles stay within [0,1]x[0,1]
    pen += np.sum(np.maximum(0, rs - xs)**2)
    pen += np.sum(np.maximum(0, xs + rs - 1)**2)
    pen += np.sum(np.maximum(0, rs - ys)**2)
    pen += np.sum(np.maximum(0, ys + rs - 1)**2)
    
    # Overlap penalties: ensure distance >= r_i + r_j
    for i in range(n):
        for j in range(i+1, n):
            d = np.hypot(xs[i]-xs[j], ys[i]-ys[j])
            dist_sum = rs[i] + rs[j]
            if dist_sum > d:
                pen += (dist_sum - d)**2
                
    # We want to maximize sum(r_i), so minimize negative sum
    return -np.sum(rs) + lam * pen
