import numpy as np
from scipy.optimize import differential_evolution, minimize

def compute_cost(params):
    """
    Objective function for circle packing optimization.
    Maximizes common radius while penalizing constraint violations.
    """
    n = 26
    centers = params[:n*2].reshape((n, 2))
    r = params[n*2]
    
    if r <= 0:
        return 1e9
        
    penalty = 0.0
    x = centers[:, 0]
    y = centers[:, 1]
    
    # Boundary penalties: circles must stay within [0,1]x[0,1]
    penalty += np.sum(np.maximum(0.0, r - x)**2)
    penalty += np.sum(np.maximum(0.0, r - (1.0 - x))**2)
    penalty += np.sum(np.maximum(0.0, r - y)**2)
    penalty += np.sum(np.maximum(0.0, r - (1.0 - y))**2)
    
    # Overlap penalties: distance between centers must be >= 2r
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist = np.sqrt(np.sum(diff**2, axis=2))
    i, j = np.triu_indices(n, k=1)
    overlaps = np.maximum(0.0, 2.0 * r - dist[i, j])
    penalty += np.sum(overlaps**2)
    
    # Objective: maximize sum of radii (n*r) <=> minimize -n*r + weighted_penalty
    return -n * r + 10000.0 * penalty

def run_packing():
    n = 26
    # Bounds: x,y in [0,1], radius in [0.01, 0.2]
    bounds = [(0.0, 1.0)] * (n * 2) + [(0.01, 0.2)]
    
    # Stage 1: Global optimization with Differential Evolution
    result_de = differential_evolution(
        compute_cost,
        bounds,
        maxiter=2000,
        popsize=15,
        tol=1e-8,
        mutation=(0.5, 1.0),
        recombination=0.7,
        seed=42,
        polish=False
    )
    
    # Stage 2: Local refinement with L-BFGS-B
    res_local = minimize(
        compute_cost,
        result_de.x,
        method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': 3000, 'ftol': 1e-12}
    )
    
    # Extract optimized parameters
    centers = res_local.x[:n*2].reshape((n, 2))
    r = res_local.x[n*2]
    radii = np.full(n, r)
    
    # Stage 3: Feasibility verification and correction
    min_gap = 1.0
    for i in range(n):
        xc, yc = centers[i]
        rad = radii[i]
        min_gap = min(min_gap, xc - rad, 1.0 - xc - rad, yc - rad, 1.0 - yc - rad)
    for i in range(n):
        for j in range(i+1, n):
            d = np.sqrt(np.sum((centers[i] - centers[j])**2))
            min_gap = min(min_gap, d - radii[i] - radii[j])
            
    # If numerical overlaps exist, scale down radii slightly to guarantee validity
    if min_gap < -1e-5:
        scale = max(0.0, 1.0 + min_gap * 0.95)
        radii *= scale
        r = radii[0]
        
    # Final safety clamping
    centers = np.clip(centers, 0.0, 1.0)
    radii = np.clip(radii, 0.0, 0.5)
    
    return centers, radii, np.sum(radii)
