import numpy as np
from scipy.optimize import minimize

def compute_objective(vars, n, w):
    """
    Computes the penalized objective for circle packing.
    Minimizes negative sum of radii while penalizing constraint violations.
    """
    x = vars[:n]
    y = vars[n:2*n]
    r = vars[2*n:]
    
    # Primary objective: maximize sum of radii
    obj = -np.sum(r)
    
    # Boundary constraint penalties: r <= coord, r <= 1-coord
    vals = np.concatenate([x - r, 1 - x - r, y - r, 1 - y - r])
    mask = vals < 0
    if np.any(mask):
        obj += w * np.sum(vals[mask]**2)
        
    # Overlap constraint penalties: dist >= r_i + r_j
    X = x[:, None] - x[None, :]
    Y = y[:, None] - y[None, :]
    R_sum = r[:, None] + r[None, :]
    D = np.sqrt(X**2 + Y**2)
    
    # Extract upper triangle to avoid double counting and self-comparison
    vals = np.triu(D - R_sum, k=1).flatten()
    mask = vals < 0
    if np.any(mask):
        obj += w * np.sum(vals[mask]**2)
        
    return obj

def project_to_feasible(centers, radii, n):
    """
    Iteratively projects radii down to strictly satisfy non-overlap and boundary constraints.
    """
    for _ in range(100):
        max_change = 0.0
        for i in range(n):
            # Boundary limits
            r_limit = min(centers[i, 0], 1 - centers[i, 0], 
                          centers[i, 1], 1 - centers[i, 1])
            # Overlap limits
            for j in range(n):
                if i == j:
                    continue
                d = np.hypot(centers[i, 0] - centers[j, 0], 
                             centers[i, 1] - centers[j, 1])
                r_limit = min(r_limit, d - radii[j])
            
            r_limit = max(r_limit, 0.0)
            if radii[i] > r_limit + 1e-13:
                change = radii[i] - r_limit
                radii[i] = r_limit
                max_change = max(max_change, change)
                
        if max_change < 1e-14:
            break
    return centers, radii

def run_packing():
    n = 26
    
    # 1. Initialization: 5x5 grid + center circle
    pts = []
    for i in range(5):
        for j in range(5):
            pts.append([0.1 + 0.2 * i, 0.1 + 0.2 * j])
    pts.append([0.5, 0.5])
    pts = np.array(pts)
    
    # Add controlled jitter to break symmetry and avoid degenerate local minima
    np.random.seed(42)
    pts += np.random.uniform(-0.01, 0.01, size=pts.shape)
    pts = np.clip(pts, 0.05, 0.95)
    
    r_init = np.full(n, 0.09) + np.random.uniform(-0.005, 0.005, n)
    r_init = np.clip(r_init, 0.01, 0.2)
    
    x0 = np.concatenate([pts.ravel(), r_init])
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
    
    # 2. Optimization Stage 1: Moderate penalty for rapid exploration
    res1 = minimize(compute_objective, x0, args=(n, 1000.0), 
                    method='L-BFGS-B', bounds=bounds, options={'maxiter': 2000})
    
    # 3. Optimization Stage 2: High penalty for precision and feasibility
    res2 = minimize(compute_objective, res1.x, args=(n, 10000.0), 
                    method='L-BFGS-B', bounds=bounds, options={'maxiter': 3000})
    
    x_opt = res2.x
    centers = x_opt[:2 * n].reshape((n, 2))
    radii = x_opt[2 * n:]
    
    # 4. Feasibility Projection: Guarantee strict compliance with validator tolerances
    centers, radii = project_to_feasible(centers, radii, n)
    
    total_sum = np.sum(radii)
    return centers, radii, total_sum
