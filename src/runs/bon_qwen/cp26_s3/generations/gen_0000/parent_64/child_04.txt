    -   **Gradient Calculation**: We compute the gradient of this objective function. For each circle, we identify the active "bottleneck" constraint (either a wall or a neighbor) and calculate the subgradient.

        -   If constrained by a neighbor $j$, the gradient points directly away from $j$ (unit vector scaled by $0.5$).

import numpy as np

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes the placement of 26 circles in a unit square to maximize the sum of radii.
    """
    n = 26
    
    # --- 1. Initialization: Hexagonal Lattice ---
    centers = np.zeros((n, 2))
    
    # Target approximate radius for initialization
    r_init = 0.05
    
    # Define row pattern: 6, 5, 6, 5, 4 circles to sum to 26
    row_counts = [6, 5, 6, 5, 4]
    
    current_idx = 0
    sqrt3 = np.sqrt(3)
    
    for i, count in enumerate(row_counts):
        # Vertical spacing for hexagonal packing: sqrt(3) * r
        y = r_init + i * sqrt3 * r_init
        
        # Horizontal positioning
        # Even rows (0, 2, 4) are unshifted (starting at r_init)
        # Odd rows (1, 3) are shifted by r_init to nest in gaps
        if i % 2 == 0:
            x_start = r_init
        else:
            x_start = 2 * r_init
        
        # Horizontal spacing is 2*r
        step = 2 * r_init
        
        for k in range(count):
            if current_idx < n:
                x = x_start + k * step
                centers[current_idx] = [x, y]
                current_idx += 1

    # --- 2. Optimization: Gradient Ascent ---
    # We maximize sum of radii by moving centers away from constraints.
    
    # Decay schedule for learning rate
    lr_start = 0.05
    lr_end = 1e-6
    steps = 4000
    
    for step in range(steps):
        # Cosine decay for learning rate
        progress = step / steps
        lr = lr_end + 0.5 * (lr_start - lr_end) * (1 + np.cos(np.pi * progress))
        
        # Compute gradients for each circle
        grad = np.zeros_like(centers)
        
        # Precompute pairwise distances for efficiency
        # diff[i, j] = centers[i] - centers[j]
        diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        np.fill_diagonal(dists, np.inf) # Ignore self-distance
        
        for i in range(n):
            # 1. Check boundary constraints
            x, y = centers[i]
            
            # Distances to 4 walls
            d_left = x
            d_right = 1.0 - x
            d_down = y
            d_up = 1.0 - y
            
            # Find minimum boundary distance and corresponding gradient
            min_bd = min(d_left, d_right, d_down, d_up)
            bd_grad = np.zeros(2)
            
            if min_bd == d_left:
                bd_grad[0] = 1.0 # Push right
            elif min_bd == d_right:
                bd_grad[0] = -1.0 # Push left
            elif min_bd == d_down:
                bd_grad[1] = 1.0 # Push up
            else: # d_up
                bd_grad[1] = -1.0 # Push down
            
            # 2. Check neighbor constraints
            # We need the nearest neighbor distance
            # dists[i] contains distances to all other circles
            min_neighbor_dist = np.min(dists[i])
            neighbor_idx = np.argmin(dists[i])
            
            # The constraint is r_i + r_j <= dist_ij. 
            # For equal radii optimization, r_i = dist_ij / 2.
            # Gradient of (dist_ij / 2) w.r.t center i is 0.5 * (c_i - c_j) / dist_ij
            
            if min_neighbor_dist < 1.0: # Sanity check, though always true inside square
                # Unit vector from j to i
                direction = diff[i, neighbor_idx] / min_neighbor_dist
                neighbor_grad = 0.5 * direction
            
            # 3. Determine active bottleneck
            # The radius is min(boundary_dist, neighbor_dist / 2)
            # We pick the gradient of the tighter constraint
            
            if min_bd < min_neighbor_dist / 2:
                grad[i] = bd_grad
            else:
                grad[i] = neighbor_grad
        
        # Apply gradient update
        centers += lr * grad
        
        # Clamp centers to valid range to prevent numerical drift
        # (Though gradient should theoretically keep them inside, clamping is safer)
        centers = np.clip(centers, 1e-4, 1.0 - 1e-4)

    # --- 3. Final Radius Calculation ---
    # Recalculate radii based on the optimized centers to ensure strict validity
    radii = np.ones(n) * np.inf
    
    # Constraint from boundaries
    for i in range(n):
        x, y = centers[i]
        r_bound = min(x, 1.0 - x, y, 1.0 - y)
        radii[i] = r_bound
        
    # Constraint from neighbors
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            # Each circle can have at most half the distance between them
            r_pair = dist / 2.0
            if r_pair < radii[i]:
                radii[i] = r_pair
            if r_pair < radii[j]:
                radii[j] = r_pair

    # Ensure non-negative (should be positive given initialization)
    radii = np.maximum(radii, 0.0)
    
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
