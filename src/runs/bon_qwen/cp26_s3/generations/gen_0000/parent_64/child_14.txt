To maximize the sum of radii for 26 circles in a unit square, we treat this as a non-linear optimization problem. The objective is to maximize $\sum r_i$ subject to boundary constraints ($0 \le x_i \pm r_i \le 1$, etc.) and non-overlap constraints ($\|c_i - c_j\| \ge r_i + r_j$).

Strategy:
1. **Non-Linear Programming (NLP)**: We use `scipy.optimize.minimize` with the `SLSQP` method. This solver handles non-linear objectives and constraints efficiently. We define the variables as a flattened vector of centers and radii.
2. **Multiple Initializations**: To avoid local minima, we initialize the optimization from several structured configurations:
   - A hexagonal lattice (dense packing structure).
   - A perturbed grid layout.
   - A random layout.
3. **Linear Programming (LP) Refinement**: After finding an optimal center layout with NLP, we fix the centers and solve a Linear Program to find the exact maximum sum of radii for those fixed positions. This guarantees the radii are optimal for the given geometry and handles the boundary/overlap constraints linearly.
4. **Constraints Handling**: We implement constraint functions that return non-negative values for feasible points. Boundary constraints ensure circles stay within $[0,1]^2$. Overlap constraints ensure distance between centers is at least the sum of radii.
5. **Robustness**: We handle potential solver failures and ensure the final output meets validation criteria.

```python
import numpy as np
from scipy.optimize import minimize, linprog

N_CIRCLES = 26

def objective_func(v):
    """Objective function: maximize sum of radii (minimize negative sum)."""
    # radii are at indices 2, 5, 8, ...
    return -np.sum(v[2::3])

def constraint_func(v):
    """Constraint function: returns values >= 0 for feasible configurations."""
    n = N_CIRCLES
    # Boundary constraints: 4 per circle (x-r, 1-x-r, y-r, 1-y-r)
    c_bound = np.zeros(4 * n)
    for i in range(n):
        idx = 3 * i
        xi, yi, ri = v[idx], v[idx+1], v[idx+2]
        c_bound[4*i]   = xi - ri
        c_bound[4*i+1] = 1.0 - xi - ri
        c_bound[4*i+2] = yi - ri
        c_bound[4*i+3] = 1.0 - yi - ri
        
    # Overlap constraints: dist >= r_i + r_j
    m = n * (n - 1) // 2
    c_overlap = np.zeros(m)
    k = 0
    for i in range(n):
        for j in range(i + 1, n):
            idx_i, idx_j = 3 * i, 3 * j
            dx = v[idx_i] - v[idx_j]
            dy = v[idx_i+1] - v[idx_j+1]
            dist = np.sqrt(dx*dx + dy*dy)
            c_overlap[k] = dist - v[idx_i+2] - v[idx_j+2]
            k += 1
            
    return np.concatenate([c_bound, c_overlap])

def run_packing():
    n = N_CIRCLES
    best_sum = -np.inf
    best_x = None
    
    # Generate multiple initial configurations to escape local minima
    starts = []
    
    # 1. Hexagonal packing (dense structure)
    r0 = 0.08
    centers_hex = []
    row = 0
    while len(centers_hex) < n:
        y = row * r0 * np.sqrt(3) + r0
        x_offset = r0 if row % 2 == 1 else 0
        x = r0 + x_offset
        while x <= 1.0 - r0 and len(centers_hex) < n:
            centers_hex.append([x, y])
            x += 2 * r0
        row += 1
    centers_hex = np.array(centers_hex[:n])
    v_hex = np.hstack([centers_hex.ravel(), np.ones(n) * r0])
    starts.append(v_hex)
    
    # 2. Perturbed grid
    rng_grid = np.random.RandomState(42)
    grid = np.meshgrid(np.linspace(0.2, 0.8, 6), np.linspace(0.2, 0.8, 5))
    pts = np.vstack([grid[0].ravel(), grid[1].ravel()]).T[:n]
    pts += rng_grid.uniform(-0.05, 0.05, pts.shape)
    pts = np.clip(pts, 0.1, 0.9)
    v_grid = np.hstack([pts.ravel(), np.ones(n) * 0.05])
    starts.append(v_grid)
    
    # 3. Random initialization
    rng_rand = np.random.RandomState(123)
    c_rand = rng_rand.uniform(0.15, 0.85, (n, 2))
    v_rand = np.hstack([c_rand.ravel(), np.ones(n) * 0.04])
    starts.append(v_rand)
    
    # Bounds for variables: x, y in [0, 1], r in [1e-6, 0.5]
    bounds = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * n
    cons = {'type': 'ineq', 'fun': constraint_func}
    
    # Run optimization for each start
    for v_init in starts:
        try:
            res = minimize(objective_func, v_init, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 1000, 'ftol': 1e-9})
            if -res.fun > best_sum:
                best_sum = -res.fun
                best_x = res.x.copy()
        except Exception:
            continue
            
    if best_x is None:
        # Fallback to simple valid configuration
        centers = np.tile([0.5, 0.5], (n, 1))
        radii = np.ones(n) * 0.01
        return centers, radii, sum(radii)
        
    centers = best_x[:2*n].reshape(n, 2)
    radii = best_x[2*n:]
    
    # LP Refinement: Fix centers, maximize sum of radii linearly
    # Maximize sum(r_i) <=> Minimize -sum(r_i)
    c_obj = -np.ones(n)
    A_ub = []
    b_ub = []
    
    for i in range(n):
        x, y = centers[i]
        # r_i <= x
        row = np.zeros(n)
        row[i] = 1.0
        A_ub.append(row)
        b_ub.append(x)
        # r_i <= 1-x
        row = np.zeros(n)
        row[i] = 1.0
        A_ub.append(row)
        b_ub.append(1.0 - x)
        # r_i <= y
        row = np.zeros(n)
        row[i] = 1.0
        A_ub.append(row)
        b_ub.append(y)
        # r_i <= 1-y
        row = np.zeros(n)
        row[i] = 1.0
        A_ub.append(row)
        b_ub.append(1.0 - y)
        
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dist)
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds_lp = [(0, None)] * n
    
    try:
        # Use HiGHS solver if available, else default
        res_lp = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_lp, method='highs')
        if res_lp.success:
            radii = res_lp.x
            best_sum = -res_lp.fun
    except Exception:
        # Fallback to interior point if highs fails
        try:
            res_lp = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_lp, method='interior-point')
            if res_lp.success:
                radii = res_lp.x
                best_sum = -res_lp.fun
        except Exception:
            pass # Keep NLP result
            
    return centers, radii, best_sum
```
