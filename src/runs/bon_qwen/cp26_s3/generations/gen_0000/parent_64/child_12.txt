import numpy as np
import scipy.optimize
import math

def run_packing():
    n = 26
    
    # 1. Initialize centers using a hexagonal grid pattern
    # This provides a good starting point for dense packing
    centers = np.zeros((n, 2))
    
    # Hexagonal packing constants
    # We want to fit n points. 
    # Let's arrange them in rows.
    # Row i has points at y = i * sqrt(3)/2 * s
    # x = j * s + (i % 2) * s/2
    
    # Estimate spacing s to fit in [0, 1]
    # Approx area per circle ~ s^2 * sqrt(3)/2. 
    # 26 * s^2 * sqrt(3)/2 ~ 1 => s ~ sqrt(2/(26*sqrt(3))) ~ 0.215
    # But we need to fit inside square.
    # Let's just generate a dense hex grid and scale/center it.
    
    points = []
    # Try a grid of size approx 5x5 to 6x6
    for row in range(7):
        y_base = row * math.sqrt(3) / 2
        for col in range(7):
            x_val = col + (row % 2) * 0.5
            points.append([x_val, y_base])
            
    points = np.array(points)
    
    # Select first 26 points? Or pick best subset?
    # Just taking the first 26 from the dense grid might be skewed.
    # Let's pick points that are most central or just take a chunk.
    # Better: Just take the first 26.
    initial_centers = points[:n].copy()
    
    # Normalize to fit in [0, 1] x [0, 1] with some margin
    # Find bounding box
    min_c = initial_centers.min(axis=0)
    max_c = initial_centers.max(axis=0)
    width = max_c - min_c
    
    # Scale to fit in [0.05, 0.95] initially to avoid immediate boundary hits
    # Actually, let's just scale to [0, 1]
    initial_centers = (initial_centers - min_c) / width
    # Center in the square
    # current range is [0, 1]. 
    # Let's scale down slightly to 0.9 and center
    initial_centers = initial_centers * 0.9 + 0.05
    
    centers = initial_centers

    # Helper function to solve LP for radii given centers
    def solve_radii(c):
        # Variables: r_0, ..., r_25
        # Maximize sum(r_i) => Minimize -sum(r_i)
        c_obj = -np.ones(n)
        
        # Constraints: A_ub * r <= b_ub
        # 1. Boundary constraints
        # r_i <= x_i  => 1*r_i <= x_i
        # r_i <= 1-x_i => 1*r_i <= 1-x_i
        # r_i <= y_i
        # r_i <= 1-y_i
        # 2. Pairwise constraints
        # r_i + r_j <= dist(i, j)
        
        # We will construct the constraint matrix
        # Number of boundary constraints: 4 * n
        # Number of pair constraints: n*(n-1)/2
        
        # Instead of building huge matrix explicitly (sparse would be better but dense is okay for n=26)
        # Let's use scipy.optimize.linprog
        
        # Bounds for r_i: (0, None)
        bounds = [(0, None) for _ in range(n)]
        
        # We can pass constraints as a list of linear constraints?
        # linprog accepts A_ub and b_ub.
        
        # Let's build A_ub and b_ub
        # This might be slow if done repeatedly, but n=26 is small.
        
        n_constraints = 4 * n + n * (n - 1) // 2
        A = np.zeros((n_constraints, n))
        b = np.zeros(n_constraints)
        
        idx = 0
        
        # Boundary constraints
        for i in range(n):
            x, y = c[i]
            # r_i <= x
            A[idx, i] = 1
            b[idx] = x
            idx += 1
            # r_i <= 1 - x
            A[idx, i] = 1
            b[idx] = 1 - x
            idx += 1
            # r_i <= y
            A[idx, i] = 1
            b[idx] = y
            idx += 1
            # r_i <= 1 - y
            A[idx, i] = 1
            b[idx] = 1 - y
            idx += 1
            
        # Pairwise constraints
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((c[i] - c[j])**2))
                # r_i + r_j <= dist
                A[idx, i] = 1
                A[idx, j] = 1
                b[idx] = dist
                idx += 1
                
        # Solve
        res = scipy.optimize.linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        
        if res.success:
            return res.x, -res.fun
        else:
            # Fallback: small radii
            return np.full(n, 1e-5), 26e-5

    # 2. Optimization Loop
    # We will iteratively update centers to maximize sum of radii
    # Strategy: Compute radii, identify tight constraints, move centers away from constraints.
    
    best_sum_radii = 0
    best_radii = np.zeros(n)
    best_centers = centers.copy()
    
    # Parameters for update
    learning_rate = 0.05
    decay = 0.95
    
    # Run multiple restarts or just one long optimization?
    # Let's do one long optimization with cooling learning rate.
    
    for step in range(500):
        radii, current_sum = solve_radii(centers)

            best_sum_radii = current_sum
            best_radii = radii.copy()
            best_centers = centers.copy()

        # 1. Boundary forces
        # If r_i is close to boundary limit, push center away
        # r_i <= x_i  => if r_i ~ x_i, push x positive
        # r_i <= 1-x_i => if r_i ~ 1-x_i, push x negative
        # etc.
        
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            
            # Left wall
            if x - r < 1e-4:
                forces[i, 0] += 1.0 * (r - (x - 1e-4)) # Push right
            # Right wall
            if x + r > 1 - 1e-4:
                forces[i, 0] -= 1.0 * ((x + r) - (1 - 1e-4)) # Push left
            # Bottom wall
            if y - r < 1e-4:
                forces[i, 1] += 1.0 * (r - (y - 1e-4)) # Push up
            # Top wall
            if y + r > 1 - 1e-4:
                forces[i, 1] -= 1.0 * ((y + r) - (1 - 1e-4)) # Push down
                
        # 2. Pairwise forces
        # If r_i + r_j ~ dist(i, j), push apart
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                r_sum = radii[i] + radii[j]
                
                if dist > 1e-9:
                    # Overlap or close
                    if r_sum > dist - 1e-4:
                        # Vector from j to i
                        v = centers[i] - centers[j]
                        v_norm = dist
                        # Force magnitude proportional to violation
                        violation = r_sum - dist + 1e-4 # small buffer
                        force_mag = violation 
                        forces[i] += force_mag * v / v_norm
                        forces[j] -= force_mag * v / v_norm

        # Update centers
        # Clip forces to prevent huge jumps
        max_force = 0.1
        forces = np.clip(forces, -max_force, max_force)
        
        centers += learning_rate * forces
        
        # Project back to [0, 1]
        centers = np.clip(centers, 1e-6, 1 - 1e-6)
        
        # Decay learning rate
        learning_rate *= decay

    # Final refinement: Local search on best_centers
    # Try random perturbations
    current_centers = best_centers.copy()
    current_radii, current_val = solve_radii(current_centers)
    
    for _ in range(1000):
        # Pick random circle to move
        idx = np.random.randint(n)
        # Random direction
        dx = (np.random.rand(2) - 0.5) * 0.05
        new_centers = current_centers.copy()
        new_centers[idx] += dx
        new_centers = np.clip(new_centers, 1e-6, 1 - 1e-6)
        
        new_radii, new_val = solve_radii(new_centers)
        
        if new_val > current_val:
            current_centers = new_centers
            current_radii = new_radii
            current_val = new_val
            if current_val > best_sum_radii:
                best_sum_radii = current_val
                best_radii = current_radii
                best_centers = current_centers
    
    return best_centers, best_radii, float(best_sum_radii)
