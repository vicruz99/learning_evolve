To maximize the sum of radii for 26 circles in a unit square, we formulate this as a continuous optimization problem. The objective is to maximize $\sum r_i$ subject to:
1. Boundary constraints: $r_i \le x_i \le 1-r_i$ and $r_i \le y_i \le 1-r_i$
2. Non-overlap constraints: $(x_i-x_j)^2 + (y_i-y_j)^2 \ge (r_i+r_j)^2$ for all $i < j$

This is a non-convex problem but can be efficiently solved using Sequential Least Squares Programming (SLSQP) from `scipy.optimize`. We initialize with a feasible configuration (random centers with small equal radii) and let the solver push centers apart while expanding radii. import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def compute_objective(x):
    """Negative sum of radii to enable minimization."""
    r = x[2*N_CIRCLES:]
    return -np.sum(r)

def compute_constraints(x):
    """Returns inequality constraints: boundary and non-overlap."""
    n = N_CIRCLES
    c = x[:2*n].reshape(n, 2)
    r = x[2*n:]
    
    con = []
    # Boundary constraints: r <= coord <= 1-r
    con.append(c[:, 0] - r)
    con.append(1.0 - c[:, 0] - r)
    con.append(c[:, 1] - r)
    con.append(1.0 - c[:, 1] - r)
    
    # Non-overlap constraints: dist^2 >= (r1 + r2)^2
    diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
    dist2 = np.sum(diff**2, axis=2)
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    con.append(dist2[mask] - r_sum[mask]**2)
    
    return np.concatenate(con)

def run_packing():
    n = N_CIRCLES
    np.random.seed(42)
    
    # Feasible initialization: small circles spread in the center region
    centers = np.random.uniform(0.2, 0.8, size=(n, 2))
    radii = np.full(n, 0.05)
    x0 = np.concatenate([centers.flatten(), radii])
    
    cons = {'type': 'ineq', 'fun': compute_constraints}
    bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n
    
    # Optimize using SLSQP
    res = minimize(compute_objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': 2000, 'ftol': 1e-10, 'disp': False})
                   
    centers_opt = res.x[:2*n].reshape(n, 2)
    radii_opt = res.x[2*n:]
    
    # Apply safety margin to strictly satisfy validation tolerances
    radii_opt = np.maximum(radii_opt, 0.0) * 0.999
    centers_opt = np.clip(centers_opt, 0.0, 1.0)
    
    return centers_opt, radii_opt, np.sum(radii_opt)
