import numpy as np
from scipy.optimize import minimize

N = 26

def get_initial_params():
    # Hexagonal lattice initialization for dense packing
    rows_config = [6, 5, 6, 5, 4]  # Sums to 26
    pts = []
    s = 0.2
    for r in range(5):
        n_c = rows_config[r]
        y = r * s * np.sqrt(3) / 2.0
        for c in range(n_c):
            x = c * s + (r % 2) * s / 2.0
            pts.append([x, y])
            
    pts = np.array(pts)
    # Normalize to center in [0.1, 0.9] to leave margin for growth
    pts -= pts.min(axis=0)
    pts /= pts.max(axis=0)
    pts = pts * 0.8 + 0.1
    
    xs = pts[:, 0]
    ys = pts[:, 1]
    rs = np.full(N, 0.015)
    
    params = np.empty(3 * N)
    params[0::3] = xs
    params[1::3] = ys
    params[2::3] = rs
    return params

def objective(params):
    # Maximize sum of radii -> minimize negative sum
    return -np.sum(params[2::3])

def constraint_fun(params):
    xs = params[0::3]
    ys = params[1::3]
    rs = params[2::3]
    
    n_constraints = 4*N + N*(N-1)//2
    slacks = np.empty(n_constraints)
    k = 0
    
    # Boundary constraints: 0 <= x-r, r+x <= 1, 0 <= y-r, r+y <= 1
    slacks[k:k+N] = xs - rs; k+=N
    slacks[k:k+N] = 1.0 - xs - rs; k+=N
    slacks[k:k+N] = ys - rs; k+=N
    slacks[k:k+N] = 1.0 - ys - rs; k+=N
    
    # Pairwise distance constraints: ||p_i - p_j||^2 >= (r_i + r_j)^2
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    dist_sq = dx*dx + dy*dy
    rad_sq = dr*dr
    upper = np.triu_indices(N, 1)
    slacks[k:] = dist_sq[upper] - rad_sq[upper]
    
    return slacks

def run_packing():
    np.random.seed(42)
    bounds = [(0.0, 1.0)] * N + [(0.0, 1.0)] * N + [(0.0001, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraint_fun}
    
    best_res = None
    best_obj = -np.inf
    
    # Multiple restarts to find global optimum
    for trial in range(4):
        x0 = get_initial_params()
        if trial > 0:
            x0 += np.random.normal(0, 0.03, size=x0.shape)
            x0[:2*N] = np.clip(x0[:2*N], 0.0, 1.0)
            x0[2*N:] = np.clip(x0[2*N:], 0.0001, 0.5)
            
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 2500, 'ftol': 1e-10, 'disp': False})
            if res.fun < best_obj:
                best_obj = res.fun
                best_res = res
        except Exception:
            continue
            
    if best_res is None:
        res = minimize(objective, get_initial_params(), method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 1000})
        best_res = res
        
    params = best_res.x
    centers = np.column_stack((params[0::3], params[1::3]))
    radii = params[2::3]
    return centers, radii, np.sum(radii)
