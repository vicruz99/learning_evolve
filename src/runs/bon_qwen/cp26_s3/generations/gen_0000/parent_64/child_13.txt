import numpy as np
from scipy.optimize import minimize

def objective_fn(vars, n):
    """Objective function: minimize negative sum of radii (equivalent to maximizing sum)"""
    return -np.sum(vars[2*n:])

def constraint_fn(vars, n):
    """Constraint function: returns array of constraint violations (must be >= 0)"""
    xs = vars[:n]
    ys = vars[n:2*n]
    rs = vars[2*n:]
    
    # Boundary constraints: x >= r, x + r <= 1, y >= r, y + r <= 1
    c1 = xs - rs
    c2 = 1.0 - xs - rs
    c3 = ys - rs
    c4 = 1.0 - ys - rs
    
    # Pairwise non-overlap constraints: dist(i, j) >= r_i + r_j
    # Compute distance matrix efficiently
    diff_xs = xs[:, None] - xs[None, :]
    diff_ys = ys[:, None] - ys[None, :]
    dist_matrix = np.sqrt(diff_xs**2 + diff_ys**2)
    rs_sum = rs[:, None] + rs[None, :]
    
    # Extract upper triangle indices for unique pairs
    ii, jj = np.triu_indices(n, k=1)
    c5 = dist_matrix[ii, jj] - rs_sum[ii, jj]
    
    return np.concatenate([c1, c2, c3, c4, c5])

def run_packing():
    n = 26
    best_sum_r = 0.0
    best_centers = None
    best_radii = None
    
    # Constraint dictionary for SLSQP
    cons_dict = {'type': 'ineq', 'fun': constraint_fn, 'args': (n,)}
    
    # Bounds for variables: x, y in [0, 1], r in [0, 0.5]
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
    
    # Multiple restarts to avoid local minima
    for seed in range(20):
        np.random.seed(seed)
        
        # Initial guess: random centers inside [0.1, 0.9], small uniform radii
        centers_init = np.random.uniform(0.1, 0.9, (n, 2))
        radii_init = np.ones(n) * 0.05
        x0 = np.concatenate([centers_init[:, 0], centers_init[:, 1], radii_init])
        
        try:
            res = minimize(objective_fn, x0, args=(n,), method='SLSQP', 
                           constraints=cons_dict, bounds=bounds,
                           options={'maxiter': 2000, 'ftol': 1e-12, 'disp': False})
            
            if -res.fun > best_sum_r:
                best_sum_r = -res.fun
                best_centers = np.column_stack((res.x[:n], res.x[n:2*n]))
                best_radii = res.x[2*n:]
        except Exception:
            continue
            
    return best_centers, best_radii, best_sum_r
