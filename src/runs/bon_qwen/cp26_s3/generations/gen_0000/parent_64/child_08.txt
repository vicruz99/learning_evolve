import numpy as np
from scipy.optimize import minimize

def objective(params, n_circles, mu):
    """
    Penalized objective function to minimize.
    Minimizes -sum(radii) + penalty for boundary/overlap violations.
    """
    X = params[:n_circles]
    Y = params[n_circles:2*n_circles]
    R = params[2*n_circles:]
    
    cost = -np.sum(R)
    
    # Boundary penalties: circles must stay inside [0,1]x[0,1]
    cost += mu * np.sum(np.maximum(0, R - X)**2 + 
                        np.maximum(0, R - (1 - X))**2 +
                        np.maximum(0, R - Y)**2 + 
                        np.maximum(0, R - (1 - Y))**2)
                        
    # Overlap penalties: distance between centers must be >= sum of radii
    DX = X[:, None] - X[None, :]
    DY = Y[:, None] - Y[None, :]
    R_sum = R[:, None] + R[None, :]
    
    D = np.sqrt(DX**2 + DY**2)
    np.fill_diagonal(D, np.inf)  # Ignore self-distance
    
    Overlap = np.maximum(0, R_sum - D)
    cost += mu * np.sum(Overlap**2)
    
    return cost

def run_packing():
    n_circles = 26
    # Bounds: x,y in [0,1], radii in [0.001, 0.5]
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.001, 0.5)] * n_circles
    
    best_params = None
    best_sum_r = -np.inf
    
    # Run multiple optimizations from different random seeds
    for seed in range(10):
        rng = np.random.default_rng(seed)
        # Initialize centers in the middle region, small radii
        centers = rng.uniform(0.2, 0.8, (n_circles, 2))
        radii = np.full(n_circles, 0.03)
        params = np.concatenate([centers.ravel(), radii])
        
        current_params = params
        # Continuation: gradually increase penalty strength to refine solution
        for mu in [200, 2000, 10000, 50000]:
            res = minimize(objective, current_params, args=(n_circles, mu), 
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 4000, 'ftol': 1e-15, 'gtol': 1e-12})
            current_params = res.x
            
        if current_params is not None:
            sum_r = np.sum(current_params[2*n_circles:])
            if sum_r > best_sum_r:
                best_sum_r = sum_r
                best_params = current_params
                
    centers = best_params[:2*n_circles].reshape((n_circles, 2))
    radii = best_params[2*n_circles:].copy()
    
    # Final safety projection: uniformly shrink radii until all constraints are strictly satisfied
    for _ in range(30):
        max_viol = 1e-13
        for i in range(n_circles):
            # Check boundaries
            max_viol = max(max_viol, 
                           radii[i] - centers[i][0],

                           radii[i] - centers[i][1],
                           radii[i] - (1.0 - centers[i][1]))
            # Check overlaps

                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                max_viol = max(max_viol, radii[i] + radii[j] - dist)
                
        if max_viol <= 1e-13:
            break
            
        # Scale down radii slightly to reduce violation
        scale_factor = 1.0 - (max_viol / (np.sum(radii) + 1e-9))
        radii *= max(scale_factor, 0.99)
        
    best_sum_r = np.sum(radii)
    return centers, radii, float(best_sum_r)
