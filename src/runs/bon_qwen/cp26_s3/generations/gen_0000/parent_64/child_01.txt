import numpy as np

def run_packing():
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    Uses a force-directed growing circles simulation.
    """
    np.random.seed(42)
    n = 26
    
    # 1. Initialization: Perturbed grid
    # Create a 6x5 grid (30 points), take first 26, add small noise
    xs = np.linspace(0.12, 0.88, 6)
    ys = np.linspace(0.12, 0.88, 5)
    grid = np.array([(x, y) for x in xs for y in ys])
    centers = grid[:n].copy() + np.random.normal(0, 0.015, grid[:n].shape)
    centers = np.clip(centers, 0.05, 0.95)
    
    radii = np.full(n, 0.04)
    
    # Simulation parameters
    lr = 0.008
    k_rep = 8.0       # Repulsion stiffness
    k_spread = 0.3    # Long-range spreading stiffness
    growth_factor = 1.0015
    max_iters = 500
    steps_per_iter = 40
    
    # 2. Growing & Relaxation Loop
    for it in range(max_iters):
        radii *= growth_factor
        
        for _ in range(steps_per_iter):
            forces = np.zeros_like(centers)
            
            # Vectorized pairwise differences and distances
            diff = centers[:, None, :] - centers[None, :, :]
            dists = np.sqrt(np.sum(diff**2, axis=2) + 1e-9)
            sum_r = radii[:, None] + radii[None, :]
            
            # Overlap repulsion (active only when circles intersect)
            overlap = sum_r - dists
            mask = overlap > 0
            rep_force_mag = k_rep * np.where(mask, overlap, 0.0)
            dir_vec = diff / (dists[:, :, None] + 1e-9)
            forces += np.sum(dir_vec * rep_force_mag[:, :, None], axis=1)
            
            # Boundary repulsion
            bx = radii - centers[:, 0]
            forces[:, 0] += k_rep * np.where(bx > 0, bx, 0.0)
            bx2 = centers[:, 0] + radii - 1.0
            forces[:, 0] -= k_rep * np.where(bx2 > 0, bx2, 0.0)
            by = radii - centers[:, 1]
            forces[:, 1] += k_rep * np.where(by > 0, by, 0.0)
            by2 = centers[:, 1] + radii - 1.0
            forces[:, 1] -= k_rep * np.where(by2 > 0, by2, 0.0)
            
            # Long-range spreading to avoid grid collapse & encourage hexagonal packing
            spread_force = k_spread / (dists**2 + 0.5)
            forces += np.sum(dir_vec * spread_force[:, :, None], axis=1)
            
            # Update positions
            centers += lr * forces
            lr *= 0.999 # Slow decay to allow settling
            centers = np.clip(centers, 0.002, 0.998)
            
        # Check feasibility penalty
        pen = np.sum(np.maximum(overlap, 0.0)**2)
        pen += np.sum(np.maximum(bx, 0.0)**2) + np.sum(np.maximum(bx2, 0.0)**2) + \
               np.sum(np.maximum(by, 0.0)**2) + np.sum(np.maximum(by2, 0.0)**2)
               
        if pen > 0.04:
            # Exceeded packing capacity, backtrack and stop
            radii /= growth_factor
            break
            
    # 3. Extract strictly valid maximal radii for final centers
    final_radii = np.full(n, 1.0)
    for i in range(n):
        min_dist = min(centers[i, 0], 1.0 - centers[i, 0], 
                       centers[i, 1], 1.0 - centers[i, 1])
        for j in range(n):
            if i != j:
                d = np.sqrt(np.sum((centers[i] - centers[j])**2))
                if d < min_dist:
                    min_dist = d
        final_radii[i] = min_dist / 2.0
        
    return centers, final_radii, np.sum(final_radii)
