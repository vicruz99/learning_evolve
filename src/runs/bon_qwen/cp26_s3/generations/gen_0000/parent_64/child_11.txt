import numpy as np
from scipy.optimize import differential_evolution, minimize

def compute_sum_radii(centers):
    """
    Computes the maximum valid radii for a given set of centers.
    Radius is limited by boundaries and nearest neighbor distances.
    """
    n = centers.shape[0]
    # Boundary constraints: distance to [0,1] walls
    rb = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                    np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    
    # Pairwise distance constraints
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    # Nearest neighbor constraint: r_i <= d_ij / 2
    rc = np.min(dists, axis=1) / 2.0
    
    # Take the tightest constraint for each circle
    radii = np.minimum(rb, rc)
    return radii

def objective(x):
    """
    Objective function for optimization. We minimize the negative sum of radii.
    """
    centers = x.reshape(-1, 2)
    radii = compute_sum_radii(centers)
    return -np.sum(radii)

def run_packing():
    n = 26
    # Bounds for x and y coordinates of each circle
    bounds = [(0.0, 1.0)] * (2 * n)
    
    # Stage 1: Global optimization with Differential Evolution
    de_result = differential_evolution(
        objective,
        bounds,
        popsize=30,
        maxiter=1200,
        mutation=(0.5, 1.5),
        recombination=0.7,
        seed=42,
        tol=1e-7,
        atol=1e-7,
        polish=False,  # Handled separately for better control
        workers=1
    )
    
    # Stage 2: Local refinement with Nelder-Mead
    # Nelder-Mead is derivative-free and robust to the non-smooth objective
    local_result = minimize(
        objective,
        de_result.x,
        method='Nelder-Mead',
        options={'xatol': 1e-8, 'fatol': 1e-9, 'maxiter': 5000}
    )
    
    # Select the best configuration found
    best_x = local_result.x if local_result.fun < de_result.fun else de_result.x
    centers = best_x.reshape(-1, 2)
    radii = compute_sum_radii(centers)
    
    return centers, radii, np.sum(radii)
