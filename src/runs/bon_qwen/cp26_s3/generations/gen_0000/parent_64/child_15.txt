import numpy as np
from scipy.optimize import minimize

def get_constraint_values(v, n):
    """Computes constraint values for SLSQP optimizer."""
    centers = v[:2*n].reshape(n, 2)
    radii = v[2*n:]
    
    num_con = 4*n + n*(n-1)//2
    con = np.empty(num_con)
    
    idx = 0
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    con[idx:idx+n] = centers[:, 0] - radii; idx += n
    con[idx:idx+n] = 1.0 - centers[:, 0] - radii; idx += n
    con[idx:idx+n] = centers[:, 1] - radii; idx += n
    con[idx:idx+n] = 1.0 - centers[:, 1] - radii; idx += n
    
    # Overlap constraints: dist^2 >= (r1 + r2)^2
    for i in range(n):
        ri = radii[i]
        ci = centers[i]
        for j in range(i+1, n):
            rj = radii[j]
            cj = centers[j]
            dx = ci[0] - cj[0]
            dy = ci[1] - cj[1]
            con[idx] = dx*dx + dy*dy - (ri + rj)**2
            idx += 1
    return con

def get_objective(v, n):
    """Objective function: negative sum of radii."""
    radii = v[2*n:]
    return -np.sum(radii)

def run_packing():
    n = 26
    best_sum = 0.0
    best_centers = np.zeros((n, 2))
    best_radii = np.zeros(n)
    
    # Run optimization with multiple seeds to find global optimum
    for seed in range(8):
        np.random.seed(seed)
        
        # Initialize centers in a staggered grid pattern
        centers_init = np.zeros((n, 2))
        idx = 0
        for row in range(6):
            cols = 5
            if idx + cols > n:
                cols = n - idx
            y = 0.08 + row * 0.16
            for c in range(cols):
                x = 0.08 + c * 0.16 + (0.08 if row % 2 == 1 else 0)
                centers_init[idx] = [x, y]
                idx += 1
                
        # Add random perturbation
        centers_init += np.random.uniform(-0.015, 0.015, centers_init.shape)
        centers_init = np.clip(centers_init, 0.05, 0.95)
        
        # Initialize radii slightly smaller to start feasible
        radii_init = np.ones(n) * 0.06
        radii_init += np.random.uniform(-0.005, 0.005, n)
        radii_init = np.clip(radii_init, 0.01, 0.1)
        
        v0 = np.concatenate([centers_init.flatten(), radii_init])
        
        bounds = [(0, 1)] * (2*n) + [(1e-6, 0.5)] * n
        cons = {'type': 'ineq', 'fun': get_constraint_values, 'args': (n,)}
        
        res = minimize(get_objective, v0, args=(n,), method='SLSQP', bounds=bounds, constraints=cons, 
                       options={'maxiter': 2000, 'ftol': 1e-10, 'disp': False})
        
        curr_sum = -res.fun
        if curr_sum > best_sum:
            best_sum = curr_sum
            best_centers = res.x[:2*n].reshape(n, 2)
            best_radii = res.x[2*n:]
            
    return best_centers, best_radii, best_sum
