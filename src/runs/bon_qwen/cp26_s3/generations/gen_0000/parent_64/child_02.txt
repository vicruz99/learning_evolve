

<strategy>
For this circle packing problem, I'll use a multi-phase optimization approach:

1. **Initialization**: Start with circles arranged in a hexagonal grid pattern (6-5-6-5-4 per row), which is known to be efficient for dense packing.

2. **Phase 1 - Force-directed layout**: Use a repulsion-based force model to push circles apart while gradually growing radii. This helps escape bad local configurations early.

3. **Phase 2 - Coordinate descent**: Alternate between:
   - Fixing radii and optimizing centers (using force-directed with simulated annealing)
   - Fixing centers and computing maximum feasible radii for each circle

4. **Phase 3 - Fine optimization**: Use scipy.optimize with the SLSQP method and explicit constraints to polish the solution. This handles the non-convex constraints more carefully.

Key insights:
- The constraint (xi-xj)² + (yi-yj)² >= (ri+rj)² is used for non-overlap
- Boundary constraints: ri <= xi,yi <= 1-ri
- Using functools.partial to create constraint functions without lambdas
- Multiple restarts with different initial configurations to avoid local optima
- The optimal solution for equal circles would have r ≈ 0.1, giving sum ≈ 2.6

import numpy as np
from scipy.optimize import minimize
from functools import partial
import math


def objective(params, n):
    """Minimize negative sum of radii"""
    return -np.sum(params[2 * n:])


def constraint_boundary_x_lower(params, n, i):
    return params[2 * i] - params[2 * n + i]


def constraint_boundary_x_upper(params, n, i):
    return 1.0 - params[2 * i] - params[2 * n + i]


def constraint_boundary_y_lower(params, n, i):
    return params[2 * i + 1] - params[2 * n + i]


def constraint_boundary_y_upper(params, n, i):
    return 1.0 - params[2 * i + 1] - params[2 * n + i]


def constraint_overlap(params, n, i, j):
    xi = params[2 * i]
    yi = params[2 * i + 1]
    xj = params[2 * j]
    yj = params[2 * j + 1]
    ri = params[2 * n + i]
    rj = params[2 * n + j]
    dx = xi - xj
    dy = yi - yj
    return dx * dx + dy * dy - (ri + rj) * (ri + rj)


def init_hexagonal(n, variant=0):
    """Initialize circles in a hexagonal grid pattern"""
    centers_list = []
    
    if variant == 0:
        rows_config = [(6, 0.0), (5, 0.07), (6, 0.0), (5, 0.07), (4, 0.0)]
    elif variant == 1:
        rows_config = [(5, 0.07), (6, 0.0), (5, 0.07), (6, 0.0), (4, 0.07)]
    elif variant == 2:
        rows_config = [(4, 0.14), (6, 0.0), (5, 0.07), (6, 0.0), (5, 0.07)]
    else:
        rows_config = [(6, 0.0), (5, 0.07), (6, 0.0), (5, 0.07), (4, 0.0)]
    
    s = 0.13
    for row_idx, (count, x_off) in enumerate(rows_config):
        y = 0.07 + row_idx * s * math.sqrt(3) / 2
        for col_idx in range(count):
            x = 0.06 + x_off + col_idx * s
            centers_list.append([x, y])
    
    return np.array(centers_list)


def force_directed_step(centers, radii, n, step_size=0.002, temperature=0.001):
    """One step of force-directed optimization"""
    forces = np.zeros_like(centers)
    
    # Repulsion between circles
    for i in range(n):
        for j in range(i + 1, n):
            diff = centers[j] - centers[i]
            dist = math.sqrt(diff[0] * diff[0] + diff[1] * diff[1])
            if dist < 1e-12:
                dist = 1e-12
                diff = np.array([np.random.rand() * 0.01, np.random.rand() * 0.01])
            
            min_dist = radii[i] + radii[j]
            if dist < min_dist + 0.005:
                excess = min_dist - dist
                force_mag = excess * excess / (dist + 0.001)
                direction = diff / dist
                forces[i] -= force_mag * direction
                forces[j] += force_mag * direction
    
    # Boundary repulsion
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x < r:
            forces[i, 0] += (r - x) * 100
        if x > 1.0 - r:
            forces[i, 0] -= (x - (1.0 - r)) * 100
        if y < r:
            forces[i, 1] += (r - y) * 100
        if y > 1.0 - r:
            forces[i, 1] -= (y - (1.0 - r)) * 100
    
    centers += step_size * forces + temperature * (np.random.rand(n, 2) - 0.5) * 0.001
    centers = np.clip(centers, 0.0001, 0.9999)
    return centers


def compute_max_radii(centers, radii, n):
    """Compute maximum feasible radius for each circle given current centers"""
    new_radii = np.copy(radii)
    for i in range(n):
        max_r = min(
            centers[i, 0],
            1.0 - centers[i, 0],
            centers[i, 1],
            1.0 - centers[i, 1]
        )
        for j in range(n):
            if i != j:
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = math.sqrt(dx * dx + dy * dy)
                max_r = min(max_r, dist - radii[j])
        new_radii[i] = max(0.0001, max_r)
    return new_radii


def check_feasibility(centers, radii, n):
    """Check if current configuration is feasible"""
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x < r - 1e-10 or x > 1.0 - r + 1e-10:
            return False
        if y < r - 1e-10 or y > 1.0 - r + 1e-10:
            return False
    
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = math.sqrt(dx * dx + dy * dy)
            if dist < radii[i] + radii[j] - 1e-10:
                return False
    return True


def shrink_to_feasible(centers, radii, n, iterations=50):
    """Shrink radii until configuration is feasible"""
    for _ in range(iterations):
        if check_feasibility(centers, radii, n):
            break
        max_violation = 0.0
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            max_violation = max(max_violation, r - x, x - (1.0 - r), r - y, y - (1.0 - r))
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = math.sqrt(dx * dx + dy * dy)
                max_violation = max(max_violation, radii[i] + radii[j] - dist)
        
        if max_violation > 0:
            shrink_factor = 1.0 - max_violation * 0.5
            radii *= shrink_factor
    
    return radii


def run_packing():
    n = 26
    
    best_centers = None
    best_radii = None
    best_sum = 0.0
    
    for variant in range(3):
        centers = init_hexagonal(n, variant)
        radii = np.full(n, 0.065)
        
        # Phase 1: Force-directed with annealing
        temperature = 0.0005
        cooling_rate = 0.9998
        
        for iteration in range(8000):
            step_size = max(0.0005, temperature)
            centers = force_directed_step(centers, radii, n, step_size, temperature)
            
            # Compute max radii
            new_radii = compute_max_radii(centers, radii, n)
            
            # Gradually increase radii
            growth = 1.0 + 0.00003
            radii = np.minimum(new_radii, radii * growth)
            
            temperature *= cooling_rate
            if temperature < 1e-8:
                temperature = 1e-8
        
        # Phase 2: Coordinate descent
        for cd_iter in range(200):
            # Grow radii
            for i in range(n):
                max_r = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
                for j in range(n):
                    if i != j:
                        dx = centers[i, 0] - centers[j, 0]
                        dy = centers[i, 1] - centers[j, 1]
                        dist = math.sqrt(dx * dx + dy * dy)
                        max_r = min(max_r, dist - radii[j])
                radii[i] = max(0.0001, max_r)
            
            # Force-directed adjustment
            centers = force_directed_step(centers, radii, n, 0.001, 0.0001)
        
        # Phase 3: scipy optimization
        params0 = np.concatenate([centers.flatten(), radii])
        
        bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
        
        cons = []
        for i in range(n):
            cons.append({'type': 'ineq', 'fun': partial(constraint_boundary_x_lower, n=n, i=i)})
            cons.append({'type': 'ineq', 'fun': partial(constraint_boundary_x_upper, n=n, i=i)})
            cons.append({'type': 'ineq', 'fun': partial(constraint_boundary_y_lower, n=n, i=i)})
            cons.append({'type': 'ineq', 'fun': partial(constraint_boundary_y_upper, n=n, i=i)})
        
        for i in range(n):
            for j in range(i + 1, n):
                cons.append({'type': 'ineq', 'fun': partial(constraint_overlap, n=n, i=i, j=j)})
        
        result = minimize(
            partial(objective, n=n),
            params0,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 500, 'ftol': 1e-14}
        )
        
        opt_centers = result.x[:2 * n].reshape(n, 2)
        opt_radii = result.x[2 * n:]
        
        # Ensure feasibility
        opt_radii = shrink_to_feasible(opt_centers, opt_radii, n)
        
        # Ensure boundaries
        for i in range(n):
            r = opt_radii[i]
            opt_centers[i, 0] = max(r + 1e-12, min(1.0 - r - 1e-12, opt_centers[i, 0]))
            opt_centers[i, 1] = max(r + 1e-12, min(1.0 - r - 1e-12, opt_centers[i, 1]))
        
        current_sum = np.sum(opt_radii)
        
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = opt_centers.copy()
            best_radii = opt_radii.copy()
    
    # Final shrink to ensure feasibility with tolerance
    best_radii = shrink_to_feasible(best_centers, best_radii, n, iterations=100)
    
    return best_centers, best_radii, np.sum(best_radii)
