import numpy as np
from scipy.optimize import minimize
import itertools

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    # Check for NaN values
    if np.isnan(centers).any():
        return False
    if np.isnan(radii).any():
        return False

    n = centers.shape[0]

    # Check if radii are nonnegative
    for i in range(n):
        if radii[i] < 0:
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False

    return True

def get_bounds(n):
    """
    Return bounds for variables [x1, y1, r1, ..., xn, yn, rn]
    """
    b = []
    for _ in range(n):
        # x in [0, 1]
        b.append((0.0, 1.0))
        # y in [0, 1]
        b.append((0.0, 1.0))
        # r in [0, 0.5] (safe upper bound)
        b.append((0.0, 0.5))
    return b

def get_constraints(n):
    """
    Return constraint dict for scipy.optimize.minimize
    Constraints are of the form c(x) >= 0.
    """
    cons = []
    
    # 1. Boundary constraints for each circle i
    # x_i - r_i >= 0
    # 1 - x_i - r_i >= 0
    # y_i - r_i >= 0
    # 1 - y_i - r_i >= 0
    
    # We define a function that computes all boundary constraints at once for efficiency
    # But scipy prefers individual constraint dicts or a single function returning array.
    # Let's define a function for boundary constraints.
    
    def bound_constraints(x):
        c = []
        for i in range(n):
            idx = 3 * i
            xi, yi, ri = x[idx], x[idx+1], x[idx+2]
            c.append(xi - ri)
            c.append(1.0 - xi - ri)
            c.append(yi - ri)
            c.append(1.0 - yi - ri)
        return np.array(c)

    cons.append({
        'type': 'ineq',
        'fun': bound_constraints
    })

    # 2. Non-overlap constraints for each pair (i, j)
    # (xi - xj)^2 + (yi - yj)^2 - (ri + rj)^2 >= 0
    
    def overlap_constraints(x):
        c = []
        for i in range(n):
            idx_i = 3 * i
            xi, yi, ri = x[idx_i], x[idx_i+1], x[idx_i+2]
            for j in range(i + 1, n):
                idx_j = 3 * j
                xj, yj, rj = x[idx_j], x[idx_j+1], x[idx_j+2]
                dist_sq = (xi - xj)**2 + (yi - yj)**2
                sum_r = ri + rj
                c.append(dist_sq - sum_r**2)
        return np.array(c)

    cons.append({
        'type': 'ineq',
        'fun': overlap_constraints
    })

    return cons

def generate_hexagonal_packing(n):
    """
    Generate initial guesses based on hexagonal lattice arrangements.
    Returns a list of (x0, sum_radii) tuples.
    """
    candidates = []
    
    # Try number of rows from 1 to roughly sqrt(n) + buffer
    # For 26 circles, rows between 1 and 8 is reasonable.
    max_rows = 10 
    
    for m in range(1, max_rows + 1):
        # Distribute n circles into m rows
        # We want row lengths to be balanced
        base_len = n // m
        rem = n % m
        
        # Rows 0 to rem-1 have length base_len + 1
        # Rows rem to m-1 have length base_len
        row_lengths = [base_len + 1] * rem + [base_len] * (m - rem)
        
        # Generate coordinates in a "radius=1" space (diameter 2)
        # Even rows (index 0, 2, ...): x = 0, 2, 4, ...
        # Odd rows (index 1, 3, ...): x = 1, 3, 5, ...
        # y = row_index * sqrt(3)
        
        points = []
        for r_idx, length in enumerate(row_lengths):
            y = r_idx * np.sqrt(3)
            for c_idx in range(length):
                if r_idx % 2 == 0:
                    x = 2 * c_idx
                else:
                    x = 2 * c_idx + 1
                points.append((x, y))
        
        if len(points) != n:
            continue
            
        xs = [p[0] for p in points]
        ys = [p[1] for p in points]
        
        # Calculate bounding box span
        x_min, x_max = min(xs), max(xs)
        y_min, y_max = min(ys), max(ys)
        
        # Width and Height of the pattern
        W = x_max - x_min
        H = y_max - y_min
        
        # To fit in [0,1] with radius r, we need:
        # r * W <= 1 - 2r  => r(W+2) <= 1 => r <= 1/(W+2)
        # r * H <= 1 - 2r  => r(H+2) <= 1 => r <= 1/(H+2)
        # Also we need to center it.
        # If we scale by s (so r=s), and shift to center.
        # Scaled width s*W. Available width 1-2s.
        # So s = 1 / (max(W, H) + 2) is the limiting factor if we center perfectly.
        
        span = max(W, H)
        if span < 0: continue # Should not happen
        
        r_opt = 1.0 / (span + 2.0)
        
        # Construct the initial solution
        # Scale points by r_opt and center them
        # Center of pattern in abstract space
        cx = (x_min + x_max) / 2.0
        cy = (y_min + y_max) / 2.0
        
        centers = np.zeros((n, 2))
        radii = np.full(n, r_opt)
        
        for i, (px, py) in enumerate(points):
            # Shift to center relative to bounding box center, then scale?
            # Actually, simpler:
            # We want final centers x' in [r, 1-r].
            # Let's map [x_min, x_max] to [r, 1-r] linearly?
            # No, scaling is linear.
            # x' = r * (x - cx) + 0.5
            # Check: if x = x_min, x' = r(x_min - (x_min+x_max)/2) + 0.5
            #       = r(-W/2) + 0.5 = 0.5 - rW/2.
            # We need x' >= r.
            # 0.5 - rW/2 >= r => 0.5 >= r(W/2 + 1) => r <= 0.5 / (W/2 + 1) = 1/(W+2).
            # Matches.
            
            x_abs = px - cx
            y_abs = py - cy
            centers[i, 0] = 0.5 + r_opt * x_abs
            centers[i, 1] = 0.5 + r_opt * y_abs
            
        sum_r = np.sum(radii)
        
        # Flatten centers and radii for scipy
        x0 = np.array([v for center in centers for v in center] + list(radii))
        
        candidates.append((x0, sum_r))

    return candidates

def run_packing():
    n = 26
    
    # 1. Generate initial guesses
    candidates = generate_hexagonal_packing(n)
    
    if not candidates:
        # Fallback to small random circles if generation fails (unlikely)
        centers = np.random.rand(n, 2)
        radii = np.full(n, 0.01)
        x0 = np.array([v for center in centers for v in center] + list(radii))
        candidates = [(x0, np.sum(radii))]
    
    # Sort candidates by sum of radii descending
    candidates.sort(key=lambda x: x[1], reverse=True)
    
    bounds = get_bounds(n)
    constraints = get_constraints(n)
    
    best_result = None
    best_sum_r = -1.0
    
    # Try top candidates
    # Also add slight random perturbations to the best one to escape local minima
    trials = []
    for i, (x0, sr) in enumerate(candidates):
        if i == 0:
            # Add perturbed versions of the best
            for _ in range(5):
                perturbed = x0.copy()
                # Perturb positions slightly
                perturbed[:2*n] += np.random.normal(0, 0.01, 2*n)
                # Perturb radii slightly
                perturbed[2*n:] += np.random.normal(0, 0.001, n)
                trials.append(perturbed)
        trials.append(x0)
        # Limit total trials to save time
        if len(trials) >= 10:
            break
            
    for x0 in trials:
        try:
            # Minimize negative sum of radii
            # Variables: [x1, y1, r1, ..., xn, yn, rn]
            # Objective: -sum(radii) = -sum(x[2:3n+1] with step 3)
            
            def objective(x):
                # Radii are at indices 2, 5, 8, ...
                radii = x[2::3]
                return -np.sum(radii)
            
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints,
                           options={'maxiter': 1000, 'ftol': 1e-9})
            
            if res.success or res.fun < 0: # Check if optimization made progress or succeeded
                final_radii = res.x[2::3]
                sum_r = np.sum(final_radii)
                
                if sum_r > best_sum_r:
                    # Validate the result to be sure
                    final_centers = np.zeros((n, 2))
                    for i in range(n):
                        final_centers[i] = res.x[3*i : 3*i+2]
                    
                    if validate_packing(final_centers, final_radii):
                        best_sum_r = sum_r
                        best_result = (final_centers, final_radii, sum_r)
        except Exception as e:
            print(f"Optimization failed: {e}")
            continue
            
    if best_result is None:
        # Fallback: return the best initial guess
        best_x0 = candidates[0][0]
        centers = np.array([[best_x0[3*i], best_x0[3*i+1]] for i in range(n)])
        radii = np.array([best_x0[3*i+2] for i in range(n)])
        sum_r = np.sum(radii)
        best_result = (centers, radii, sum_r)

    return best_result

# Run the packing function to get the solution
if __name__ == "__main__":
    centers, radii, sum_radii = run_packing()
    print(f"Sum of radii: {sum_radii}")
    print(f"Validation: {validate_packing(centers, radii)}")
