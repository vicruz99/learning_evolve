import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Returns a valid packing of 26 circles in a unit square maximizing the sum of radii.
    """
    n_circles = 26
    best_sum = 0.0
    best_centers = np.zeros((n_circles, 2))
    best_radii = np.zeros(n_circles)

    def get_initial_configs():
        configs = []
        # 1. Hexagonal lattice (dense packing)
        configs.append(create_hexagonal_config(n_circles))
        # 2. Shifted hexagonal lattice
        configs.append(create_shifted_hexagonal_config(n_circles))
        # 3. Grid with random perturbation
        configs.append(create_perturbed_grid_config(n_circles))
        return configs

    def create_hexagonal_config(n):
        centers = np.zeros((n, 2))
        radii = np.full(n, 0.095) # Initial guess slightly below 0.1
        # 6 rows: 5, 5, 5, 5, 4, 2 (sum=26)
        row_counts = [5, 5, 5, 5, 4, 2]
        r_init = 0.095
        h = r_init * np.sqrt(3)
        current_idx = 0
        for r_idx, count in enumerate(row_counts):
            y = r_init + r_idx * h
            if r_idx % 2 == 0:
                # Align left
                x_start = r_init
            else:
                # Shift right by r
                x_start = 2 * r_init
            
            for c in range(count):
                if current_idx < n:
                    x = x_start + c * 2 * r_init
                    # Clamp to unit square just in case
                    x = np.clip(x, r_init, 1 - r_init)
                    y = np.clip(y, r_init, 1 - r_init)
                    centers[current_idx] = [x, y]
                    current_idx += 1
        return centers, radii

    def create_shifted_hexagonal_config(n):
        centers, radii = create_hexagonal_config(n)
        # Shift y-axis
        centers[:, 1] = centers[:, 1] + 0.05
        centers[:, 1] = np.clip(centers[:, 1], radii, 1 - radii)
        return centers, radii

    def create_perturbed_grid_config(n):
        centers = np.zeros((n, 2))
        radii = np.full(n, 0.09)
        # Approximate 5x6 grid but only 26 points
        # 5 cols, 6 rows would be 30. 
        # Let's do 5 cols, 5 rows (25) + 1
        idx = 0
        for r in range(5):
            for c in range(5):
                if idx < n:
                    centers[idx] = [0.1 + c * 0.2, 0.1 + r * 0.2]
                    idx += 1
        if idx < n:
            centers[idx] = [0.5, 0.5] # Add in center
        # Perturb
        centers += np.random.uniform(-0.02, 0.02, centers.shape)
        # Clamp
        centers = np.clip(centers, radii, 1 - radii)
        return centers, radii

    def objective(vars):
        # Maximize sum of radii => Minimize -sum(radii)
        r = vars[2::3]
        return -np.sum(r)

    def get_constraints():
        cons = []
        # Boundary constraints
        for i in range(n_circles):
            # x - r >= 0
            cons.append({'type': 'ineq', 'fun': lambda v, i=i: v[3*i] - v[3*i+2]})
            # x + r <= 1
            cons.append({'type': 'ineq', 'fun': lambda v, i=i: 1 - v[3*i] - v[3*i+2]})
            # y - r >= 0
            cons.append({'type': 'ineq', 'fun': lambda v, i=i: v[3*i+1] - v[3*i+2]})
            # y + r <= 1
            cons.append({'type': 'ineq', 'fun': lambda v, i=i: 1 - v[3*i+1] - v[3*i+2]})
            # r >= 0 (implicitly handled by bounds usually, but added for safety)
            cons.append({'type': 'ineq', 'fun': lambda v, i=i: v[3*i+2]})

        # Non-overlap constraints: dist >= r_i + r_j
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                cons.append({
                    'type': 'ineq',
                    'fun': lambda v, i=i, j=j: 
                        np.sqrt((v[3*i] - v[3*j])**2 + (v[3*i+1] - v[3*j+1])**2) - (v[3*i+2] + v[3*j+2])
                })
        return cons

    constraints = get_constraints()
    bounds = [(0, 1), (0, 1), (0, 0.5)] * n_circles

    configs = get_initial_configs()
    
    for centers, radii in configs:
        # Flatten to 1D array for optimizer: [x1, y1, r1, x2, y2, r2, ...]
        x0 = np.zeros(3 * n_circles)
        for i in range(n_circles):
            x0[3*i] = centers[i, 0]
            x0[3*i+1] = centers[i, 1]
            x0[3*i+2] = radii[i]

        # Optimization
        res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints, 
                       options={'maxiter': 1000, 'ftol': 1e-12})

        if res.success:
            current_sum = -res.fun
            if current_sum > best_sum:
                # Extract results
                c = np.array([res.x[3*i:3*i+2] for i in range(n_circles)])
                r = np.array([res.x[3*i+2] for i in range(n_circles)])
                
                # Post-processing to ensure strict validity (clamp)
                r = np.maximum(r, 0)
                c[:, 0] = np.clip(c[:, 0], r, 1-r)
                c[:, 1] = np.clip(c[:, 1], r, 1-r)
                
                # If after clamping radius is larger than available space, reduce radius
                # This handles numerical issues where center might be too close to edge
                for i in range(n_circles):
                    margin_x = min(c[i,0], 1-c[i,0])
                    margin_y = min(c[i,1], 1-c[i,1])
                    r[i] = min(r[i], margin_x, margin_y)
                
                best_sum = np.sum(r)
                best_centers = c
                best_radii = r

    return best_centers, best_radii, best_sum
