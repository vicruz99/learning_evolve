    if np.any(np.isnan(final_radii)) or np.any(final_radii < 0):
        # Fallback to uniform radius if issues arise
        final_radii = np.full(n_circles, best_uniform_radius)
        sum_radii = best_uniform_radius * n_circles

    return best_centers, final_radii, sum_radii
