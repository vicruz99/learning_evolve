import numpy as np

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    n = centers.shape[0]
    if np.isnan(centers).any():
        return False
    if np.isnan(radii).any():
        return False
    for i in range(n):
        if radii[i] < 0:
            return False
        elif np.isnan(radii[i]):
            return False
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            return False
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False
    return True

def get_hex_grid_positions(n):

def objective_function(params, n_circles):
    """
    Objective function to minimize.
    We want to maximize sum of radii, so we minimize negative sum of radii.
    We add penalties for overlaps and boundary violations.
    """
    # Unpack parameters
    # params layout: [x1, y1, r1, x2, y2, r2, ...]
    centers = params[0::3].reshape(n_circles, 2)
    radii = params[1::3]
    
    # 1. Penalty for boundaries
    # Distance to left wall: x - r
    # Distance to right wall: 1 - (x + r)
    # Distance to bottom wall: y - r
    # Distance to top wall: 1 - (y + r)
    # Penalty is quadratic on the violation
    left_dist = centers[:, 0] - radii
    right_dist = 1.0 - (centers[:, 0] + radii)
    bottom_dist = centers[:, 1] - radii
    top_dist = 1.0 - (centers[:, 1] + radii)
    
    # Violations are when dist < 0
    boundary_penalty = 0
    for d in [left_dist, right_dist, bottom_dist, top_dist]:
        violations = d[d < 0]
        boundary_penalty += np.sum(violations ** 2) * 1000
    
    # 2. Penalty for overlaps
    # Distance between centers i and j should be >= r_i + r_j
    overlap_penalty = 0
    for i in range(n_circles):
        for j in range(i + 1, n_circles):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            min_dist = radii[i] + radii[j]
            if dist < min_dist:
                overlap_penalty += (min_dist - dist) ** 2 * 1000
                
    # 3. Negative sum of radii (we want to maximize this)
    # We use a weight to balance the objective vs penalties.
    # However, to ensure validity, we can use a very high penalty.
    # Alternatively, we can optimize for a fixed radius configuration.
    # Here we optimize both centers and radii.
    
    # Note: To ensure the optimizer finds large radii, we start with a reasonable radius.
    # The objective is -sum(radii).
    return -np.sum(radii) + boundary_penalty + overlap_penalty
