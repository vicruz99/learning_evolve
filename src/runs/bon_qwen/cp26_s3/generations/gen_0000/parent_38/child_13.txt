

To maximize the sum of radii for 26 circles in a unit square, we employ a "Grow and Pack" strategy. This approach iteratively increases circle radii while using a force-directed solver to maintain non-overlap and boundary constraints.

### Strategy

1.  **Initialization:** 26 circles are placed on a dense hexagonal grid. Hexagonal packing provides a higher density baseline than a square grid, which is crucial when starting with larger radii.
2.  **Optimization Loop:** We use a `repulsion_solver` to adjust circle centers, treating overlaps and boundary violations as repulsive forces.
3.  **Growth Cycle:** Radii are increased incrementally. If the solver fails to resolve overlaps after a growth step, the growth rate is reduced (annealing) to maintain stability.
4.  **Final Refinement:** Once the radii stabilize, a secondary solver maximizes the sum of radii by minimizing boundary and overlap penalties with a high weight on "validity."

### Python Program

```python
import numpy as np
from scipy.optimize import minimize
import math

def calculate_distances_and_penalties(centers, radii):
    """Calculates all pairwise distances, overlap penalties, and boundary penalties."""
    n = centers.shape[0]
    distances = np.zeros((n, n))
    overlaps = np.zeros(n)
    boundary_penalties = 0.0
    
    # Pairwise distances and overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            d = math.sqrt(dx * dx + dy * dy)
            distances[i, j] = d
            distances[j, i] = d
            
            overlap = radii[i] + radii[j] - d
            if overlap > 1e-8:
                overlaps[i] += overlap
                overlaps[j] += overlap
    
    # Boundary penalties
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x < r:
            boundary_penalties += (r - x)**2
        if x > 1 - r:
            boundary_penalties += (x - (1 - r))**2
        if y < r:
            boundary_penalties += (r - y)**2
        if y > 1 - r:
            boundary_penalties += (y - (1 - r))**2
            
    return distances, overlaps, boundary_penalties

def repulsion_solver(centers, radii, max_iter=500):
    """Moves centers to resolve overlaps and boundary violations."""
    n = centers.shape[0]
    alpha = 0.05  # Movement step size
    
    for _ in range(max_iter):
        max_overlap = 0
        for i in range(n):
            if centers[i, 0] < radii[i]:
                max_overlap = max(max_overlap, radii[i] - centers[i, 0])
            if centers[i, 0] > 1 - radii[i]:
                max_overlap = max(max_overlap, centers[i, 0] - (1 - radii[i]))
            if centers[i, 1] < radii[i]:
                max_overlap = max(max_overlap, radii[i] - centers[i, 1])
            if centers[i, 1] > 1 - radii[i]:
                max_overlap = max(max_overlap, centers[i, 1] - (1 - radii[i]))
            
            for j in range(i + 1, n):
                d = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if d < radii[i] + radii[j] - 1e-10:
                    max_overlap = max(max_overlap, radii[i] + radii[j] - d)
        
        if max_overlap < 1e-6:
            break
        
        forces = np.zeros_like(centers)
        # Overlap repulsion
        for i in range(n):
            for j in range(i + 1, n):
                d = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if d < radii[i] + radii[j] - 1e-10:
                    push = (radii[i] + radii[j] - d) * 0.5
                    dx = centers[i, 0] - centers[j, 0]
                    dy = centers[i, 1] - centers[j, 1]
                    if d > 1e-10:
                        nx, ny = dx / d, dy / d
                        forces[i, 0] += nx * push
                        forces[i, 1] += ny * push
                        forces[j, 0] -= nx * push
                        forces[j, 1] -= ny * push
        
        # Boundary repulsion
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x < r: forces[i, 0] += (r - x) * 2.0
            if x > 1 - r: forces[i, 0] -= (x - (1 - r)) * 2.0
            if y < r: forces[i, 1] += (r - y) * 2.0
            if y > 1 - r: forces[i, 1] -= (y - (1 - r)) * 2.0
        
        centers += forces * alpha
        # Clamp centers to prevent explosion
        centers = np.clip(centers, 0, 1)
        
    return centers

def objective_function(x, radii):
    """Objective for scipy: minimize overlap and boundary penalties."""
    n = len(radii)
    centers = x.reshape((n, 2))
    total_penalty = 0.0
    
    # Boundary penalties
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x < r: total_penalty += (r - x) ** 2
        if x > 1 - r: total_penalty += (x - (1 - r)) ** 2
        if y < r: total_penalty += (r - y) ** 2
        if y > 1 - r: total_penalty += (y - (1 - r)) ** 2
            
    # Pairwise overlap penalties
    for i in range(n):
        for j in range(i + 1, n):
            d = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if d < radii[i] + radii[j] - 1e-10:
                total_penalty += (radii[i] + radii[j] - d) ** 2
                
    return total_penalty

def run_packing():
    n = 26
    # 1. Initialization: Hexagonal Grid
    centers = np.zeros((n, 2))
    idx = 0
    for row in range(6):
        y = 0.1 + row * 0.15
        for col in range(5):
            x = 0.1 + col * 0.2 + (0.1 if row % 2 else 0)
            if idx < n and x <= 0.9 and y <= 0.9:
                centers[idx] = [x, y]
                idx += 1
    
    radii = np.full(n, 0.05)
    
    # 2. Grow and Pack Loop
    current_radius = 0.05
    for step in range(300):
        # Solve for positions
        centers = repulsion_solver(centers, radii, max_iter=200)
        
        # Check validity
        valid = True
        for i in range(n):
            if centers[i, 0] < radii[i] - 1e-6 or centers[i, 0] > 1 - radii[i] + 1e-6:
                valid = False
            if centers[i, 1] < radii[i] - 1e-6 or centers[i, 1] > 1 - radii[i] + 1e-6:
                valid = False
            for j in range(i + 1, n):
                d = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if d < radii[i] + radii[j] - 1e-6:
                    valid = False
                    break
        
        if valid:
            # Increase radii
            radii += 0.0005
        else:
            # Anneal growth if stuck
            radii += 0.00005
            
        # Limit max radius for numerical stability in this phase
        if radii[0] > 0.11:
            break
            
    # 3. Final Refinement using scipy
    # Optimize centers for the current radii to ensure maximum validity
    x0 = centers.flatten()
    res = minimize(objective_function, x0, args=(radii,), method='L-BFGS-B', 
                   bounds=[(0, 1)] * (2 * n))
    centers = res.x.reshape((n, 2))
    
    # 4. Validate and return
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii

if __name__ == "__main__":
    centers, radii, sum_radii = run_packing()
    print(f"Final Sum of Radii: {sum_radii:.5f}")
    print(f"Average Radius: {np.mean(radii):.5f}")
```
