import numpy as np
import scipy.optimize as opt

def run_packing():
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    Returns (centers, radii, sum_radii).
    """
    n = 26
    best_centers = None
    best_radii = None
    max_sum = -1.0

    # Helper to extract variables
    def get_vars(p):
        # p is shape (3*n,)
        # x, y, r
        x = p[0::3]
        y = p[1::3]
        r = p[2::3]
        return x, y, r

    # Objective: maximize sum of radii -> minimize negative sum
    def objective(p):
        _, _, r = get_vars(p)
        return -np.sum(r)

    # Constraints
    def constraints(p):
        x, y, r = get_vars(p)
        
        # 1. Boundary constraints
        # x - r >= 0
        # 1 - x - r >= 0
        # y - r >= 0
        # 1 - y - r >= 0
        c_bounds = np.concatenate([
            x - r,
            1.0 - x - r,
            y - r,
            1.0 - y - r
        ])
        
        # 2. Distance constraints (non-overlap)
        # (xi - xj)^2 + (yi - yj)^2 >= (ri + rj)^2
        # We need to compute this for all pairs i < j
        # Vectorized computation
        
        # Coordinates matrix (n, 2)
        coords = np.column_stack((x, y))
        
        # Distance matrix
        # D_ij = ||coords_i - coords_j||
        # Using broadcasting
        diff = coords[:, np.newaxis, :] - coords[np.newaxis, :, :]
        dist_sq = np.sum(diff**2, axis=2)
        
        # Sum of radii matrix
        sum_r = r[:, np.newaxis] + r[np.newaxis, :]
        
        # Constraint value: dist_sq - sum_r^2 >= 0
        # We only need upper triangle to avoid duplicates and self-interaction
        mask = np.triu(np.ones((n, n), dtype=bool), k=1)
        dist_vals = np.sqrt(np.maximum(dist_sq, 0)) # Just to be safe, though dist_sq >= 0
        sum_r_vals = sum_r
        
        # We need dist >= sum_r  => dist^2 >= sum_r^2
        # But numerical stability suggests checking dist - sum_r >= 0
        # However, derivative of dist^2 is simpler. Let's use dist^2 - sum_r^2 >= 0
        # But wait, if dist < sum_r, dist^2 - sum_r^2 is negative.
        # So constraint is dist_sq - sum_r_vals**2 >= 0
        
        # Extract only upper triangle
        # dist_sq is (n, n), sum_r_vals is (n, n)
        # constraint_vals = dist_sq[mask] - sum_r_vals[mask]**2
        
        # To handle potential negative dist_sq due to precision (unlikely), clip
        safe_dist_sq = np.maximum(dist_sq, 0.0)
        constraint_vals = safe_dist_sq[mask] - sum_r_vals[mask]**2
        
        # Combine all constraints
        all_cons = np.concatenate([c_bounds, constraint_vals])
        return all_cons

    # Bounds for variables
    # x, y in [0, 1] (actually tighter, but constraints handle it)
    # r in [0, 0.5]
    bounds = [(0.0, 1.0)] * (3 * n)
    # Tighten bounds for r slightly? No, let constraints handle it.
    # But r cannot be > 0.5.
    for i in range(0, 3*n, 3):
        bounds[i+2] = (0.0, 0.5)

    # Function to run optimization from a given initial point
    def run_opt(p0):
        # Normalize p0 to be within bounds
        p0 = np.array(p0, dtype=float)
        
        # Ensure r is positive
        for i in range(0, len(p0), 3):
            if p0[i+2] < 0:
                p0[i+2] = 1e-5
        
        cons = {'type': 'ineq', 'fun': constraints}
        
        try:
            res = opt.minimize(
                objective, 
                p0, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 2000, 'ftol': 1e-12}
            )
            if res.success or res.fun < -max_sum: # Check if we found a better sum
                x, y, r = get_vars(res.x)
                s = np.sum(r)

                    # Validate constraints manually to be sure
                    valid = True
                    # Check bounds
                    if np.any(r < -1e-9): valid = False
                    if np.any(x < r - 1e-9) or np.any(x > 1 - r + 1e-9): valid = False
                    if np.any(y < r - 1e-9) or np.any(y > 1 - r + 1e-9): valid = False
                    
                    if valid:
                        # Check overlaps
                        coords = np.column_stack((x, y))
                        dists = np.sqrt(np.sum((coords[:, None] - coords[None, :])**2, axis=2))
                        sum_rs = r[:, None] + r[None, :]
                        # Check strict inequality with tolerance
                        if np.any(dists < sum_rs - 1e-9):
                            valid = False
                        
                        return res.x, s, valid
            return None, max_sum, False
        except Exception:
            return None, max_sum, False

    # Try to fit 26 with slightly different geometry
    # Just perturb the first hex solution

    p0_hex_perturbed = np.clip(p0_hex_perturbed, 0.01, 0.99)
    # Reset r
    for i in range(0, len(p0_hex_perturbed), 3):
        p0_hex_perturbed[i+2] = 0.08
    initial_points.append(p0_hex_perturbed)

    # Run optimization for each
    best_p = None
    for p0 in initial_points:
        res_p, s, valid = run_opt(p0)
        if valid and s > max_sum:
            max_sum = s
            best_p = res_p

    if best_p is not None:
        x, y, r = get_vars(best_p)
        centers = np.column_stack((x, y))
        radii = r
        return centers, radii, np.sum(radii)
    else:
        # Fallback: return a valid simple packing
        # 25 circles in 5x5 grid r=0.1, 1 small circle? 
        # Actually 25 circles r=0.1 fit. 26th needs to be tiny or we shrink all.
        # Let's shrink all to fit 26.
        # 26 circles, equal radius.
        # Approx r = 1 / (2 * sqrt(26)) ? No.
        # Let's just return the hex result even if sum is lower, or fallback.
        # The optimizer should have found something.
        # If not, construct a safe fallback.
        
        # Fallback: 5 rows of 5, 1 row of 1?
        # Just grid 6x5 reduced
        pts = []
        r_fall = 0.07
        for i in range(6):

                if len(pts) >= 26: break

        # Simple fallback: 26 circles of radius 0.075

        # But valid.

        centers_fb = []
        radii_fb = []
        count = 0
        # 5 columns, 6 rows
