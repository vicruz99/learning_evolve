import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n_circles = 26
    
    # 1. Initial Configuration: Hexagonal-like staggered layout
    # Rows: 5, 5, 5, 6, 5 (Total 26)
    # This pattern is known to be dense for this count in a square.
    initial_radii = np.full(n_circles, 0.095)
    centers = np.zeros((n_circles, 2))
    
    idx = 0
    row_counts = [5, 5, 5, 6, 5]
    y_step = 0.175 # Approximate hexagonal vertical spacing
    y_pos = 0.15   # Initial Y offset
    
    for count in row_counts:
        # Calculate X spacing to fit 'count' circles in [0, 1]
        # Margins on sides = (1 - (count-1)*dx) / 2 ?
        # Actually, we want to center them.
        # Width occupied by centers = (count - 1) * dx
        # Total width 1. 
        # Let's just space them evenly with some padding.
        
        # Estimate diameter based on count to fit in width 1
        # d = 1.0 / count is max diameter if touching boundaries and each other?
        # No, 5 circles of d=0.2 fits perfectly.
        
        dx = 1.0 / count 
        # Offset for staggered rows
        if (row_counts.index(count) % 2 == 1) and count < 6:
             x_offset = dx / 2
        else:
             x_offset = 0
             
        # Adjust x positions to ensure they are within bounds and roughly spaced
        # Better approach: place centers at (i+0.5)/count
        # But for 6 circles in a row, they need to be tighter.
        
        # Let's place centers evenly
        # For count circles, centers at: r + i*(2r)? No.
        # Just use (i + 0.5) / count for uniform distribution, 
        # but shift odd rows.
        
        current_x = np.linspace(0.05, 0.95, count) # Initial rough x
        
        # Apply stagger shift
        if (idx < 5): # Row 0
            x_pos = current_x
        elif (idx < 10): # Row 1
            x_pos = current_x + 0.05 # Stagger
        elif (idx < 15): # Row 2
            x_pos = current_x
        elif (idx < 21): # Row 3 (6 circles)
            # 6 circles need tighter spacing or smaller radius.
            # Let's use linspace
            x_pos = np.linspace(0.05, 0.95, 6)
        else: # Row 4
            x_pos = current_x + 0.05
            
        # Ensure x is within [0, 1]
        x_pos = np.clip(x_pos, 0.01, 0.99)
        
        centers[idx:idx+count, 0] = x_pos
        centers[idx:idx+count, 1] = y_pos
        
        idx += count
        y_pos += y_step

    # 2. Optimization Setup
    # Variables: [r1, r2, ..., r26, x1, y1, x2, y2, ..., x26, y26]
    # Total 3 * 26 = 78 variables
    
    x0 = np.zeros(3 * n_circles)
    x0[0::3] = initial_radii
    x0[1::3] = centers[:, 0]
    x0[2::3] = centers[:, 1]
    
    # Bounds
    bounds = []
    for i in range(n_circles):
        bounds.append((1e-6, 0.5))      # Radius
        bounds.append((0.0, 1.0))       # x
        bounds.append((0.0, 1.0))       # y
        
    # Constraints
    constraints = []
    
    # Boundary Constraints: x - r >= 0, x + r <= 1, etc.
    for i in range(n_circles):
        r_idx = i
        x_idx = n_circles + i
        y_idx = 2 * n_circles + i
        
        # x - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: v[x_idx] - v[r_idx]
        })
        # 1 - x - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: 1.0 - v[x_idx] - v[r_idx]
        })
        # y - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: v[y_idx] - v[r_idx]
        })
        # 1 - y - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: 1.0 - v[y_idx] - v[r_idx]
        })
        
    # Non-overlap Constraints: dist >= r_i + r_j
    for i in range(n_circles):
        for j in range(i + 1, n_circles):
            ri_idx = i
            rj_idx = j
            xi_idx = n_circles + i
            yi_idx = 2 * n_circles + i
            xj_idx = n_circles + j
            yj_idx = 2 * n_circles + j
            
            def non_overlap(v, i=i, j=j):
                xi, yi = v[xi_idx], v[yi_idx]
                xj, yj = v[xj_idx], v[yj_idx]
                ri, rj = v[ri_idx], v[rj_idx]
                dist_sq = (xi - xj)**2 + (yi - yj)**2
                sum_r = ri + rj
                # We want dist >= sum_r  =>  dist^2 >= sum_r^2  => dist^2 - sum_r^2 >= 0
                return dist_sq - sum_r**2
            
            constraints.append({
                'type': 'ineq',
                'fun': non_overlap
            })

    # Objective: Maximize sum of radii => Minimize -sum(radii)
    def objective(v):
        return -np.sum(v[0:n_circles])

    # Run optimization
    # SLSQP handles non-linear constraints well
    res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints, 
                   options={'maxiter': 2000, 'ftol': 1e-10})
    
    if res.success:
        final_radii = res.x[0:n_circles]
        final_centers = np.column_stack((res.x[n_circles:2*n_circles], 
                                         res.x[2*n_circles:3*n_circles]))
        sum_radii = np.sum(final_radii)
        
        # Final cleanup to ensure strict validity (handle numerical noise)
        # Clip radii slightly if they touch boundary exactly to avoid float errors in validation
        # Although validation allows 1e-12, being safe is good.
        
        return final_centers, final_radii, sum_radii
    else:
        # Fallback to initial config if optimization fails
        sum_radii = np.sum(initial_radii)
        return centers, initial_radii, sum_radii
