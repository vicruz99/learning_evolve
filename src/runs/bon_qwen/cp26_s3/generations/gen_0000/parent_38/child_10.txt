import numpy as np
from scipy.optimize import minimize

N = 26

def objective_fn(x):
    """Objective function: maximize sum of radii (minimize negative sum)"""
    return -np.sum(x[2*N:])

def constraint_fn(x):
    """
    Inequality constraints >= 0:
    1. Boundary constraints: x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
    2. Non-overlap: dist^2 - (r1 + r2)^2 >= 0
    """
    c = x[:2*N].reshape(N, 2)
    r = x[2*N:]
    con_vals = []
    
    # Boundary constraints
    con_vals.extend(c[:, 0] - r)
    con_vals.extend(1.0 - c[:, 0] - r)
    con_vals.extend(c[:, 1] - r)
    con_vals.extend(1.0 - c[:, 1] - r)
    
    # Pairwise non-overlap constraints
    # Compute all pairwise squared distances
    c_diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
    dist_sq = np.sum(c_diff**2, axis=2)
    
    # Compute all pairwise radius sums
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    
    # Only need upper triangle pairs (i < j)
    tri_idx = np.triu_indices(N, k=1)
    con_vals.extend(dist_sq[tri_idx] - r_sum[tri_idx]**2)
    
    return np.array(con_vals)

def generate_initial_config():
    """Generates a hexagonal initial layout scaled to fit safely inside the unit square"""
    centers0 = []
    # Row pattern for 26 circles
    row_counts = [6, 5, 6, 5, 4]
    dummy_r = 0.1
    y_step = np.sqrt(3) * dummy_r
    x_step = 2.0 * dummy_r
    y_base = dummy_r
    
    for r_idx, count in enumerate(row_counts):
        y = y_base + r_idx * y_step
        x_start = dummy_r + (r_idx % 2) * (dummy_r / 2)
        for c in range(count):
            x = x_start + c * x_step
            centers0.append([x, y])
            
    centers0 = np.array(centers0)
    
    # Normalize positions to lie strictly within [0.1, 0.9] to guarantee initial feasibility
    c_min = centers0.min(axis=0)
    c_max = centers0.max(axis=0)
    centers0 = (centers0 - c_min) / (c_max - c_min) * 0.8 + 0.1
    
    r0 = np.full(N, 0.07)
    return centers0, r0

def run_packing():
    # Initial feasible configuration
    centers0, r0 = generate_initial_config()
    x0 = np.concatenate([centers0.flatten(), r0])
    
    # Variable bounds: centers in [0,1], radii in [0, 0.5]
    bounds = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    
    # Setup optimization problem
    constraints = {'type': 'ineq', 'fun': constraint_fn}
    
    # Run SLSQP optimizer
    res = minimize(objective_fn, x0, method='SLSQP', bounds=bounds, constraints=constraints,
                   options={'maxiter': 600, 'ftol': 1e-12, 'disp': False})
                   
    # Extract optimized parameters
    centers_opt = res.x[:2*N].reshape(N, 2)
    radii_opt = np.maximum(res.x[2*N:], 0.0)
    
    return centers_opt, radii_opt, np.sum(radii_opt)
