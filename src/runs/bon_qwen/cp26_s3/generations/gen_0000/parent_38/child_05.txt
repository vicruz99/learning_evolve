import numpy as np
from scipy.optimize import minimize, linprog
import functools


def constraint_x_min(x, i, n):
    return x[2 * i] - x[2 * n + i]


def constraint_x_max(x, i, n):
    return 1 - x[2 * i] - x[2 * n + i]


def constraint_y_min(x, i, n):
    return x[2 * i + 1] - x[2 * n + i]


def constraint_y_max(x, i, n):
    return 1 - x[2 * i + 1] - x[2 * n + i]


def constraint_nonoverlap(x, i, j, n):
    dx = x[2 * i] - x[2 * j]
    dy = x[2 * i + 1] - x[2 * j + 1]
    ri = x[2 * n + i]
    rj = x[2 * n + j]
    return dx * dx + dy * dy - (ri + rj) * (ri + rj)


def objective_neg_sum_radii(x, n):
    return -np.sum(x[2 * n:])


def initialize_hexagonal(n):
    centers = []
    rows = int(np.ceil(np.sqrt(2 * n / np.sqrt(3))))
    if rows < 2:
        rows = 2
    count = 0
    y_spacing = 1.0 / (rows + 1)
    for row in range(rows):
        remaining = n - count
        rows_left = rows - row
        n_in_row = max(1, int(np.ceil(remaining * 1.0 / rows_left)))
        if count + n_in_row > n:
            n_in_row = n - count
        if n_in_row <= 0:
            continue
        x_spacing = 1.0 / (n_in_row + 1)
        x_offset = x_spacing / 2.0 if row % 2 == 1 else 0
        for col in range(n_in_row):
            x = x_offset + (col + 1) * x_spacing
            y = (row + 1) * y_spacing
            centers.append([x, y])
            count += 1
            if count >= n:
                break
        if count >= n:
            break
    centers = np.array(centers[:n])
    return centers


def initialize_dense_hexagonal(n):
    centers = []
    rows = int(np.ceil(np.sqrt(2.0 * n / np.sqrt(3))))
    if rows < 2:
        rows = 2
    count = 0
    for row in range(rows):
        remaining = n - count
        rows_left = rows - row
        n_in_row = max(1, int(np.ceil(remaining * 1.0 / rows_left)))
        if count + n_in_row > n:
            n_in_row = n - count
        if n_in_row <= 0:
            continue
        x_positions = np.linspace(0.05, 0.95, n_in_row)
        if row % 2 == 1:
            x_positions = x_positions * 0.95 + 0.025
        y = (row + 1.0) / (rows + 1)
        for x in x_positions:
            centers.append([x, y])
            count += 1
            if count >= n:
                break
        if count >= n:
            break
    centers = np.array(centers[:n])
    return centers


def solve_radii_lp(centers, n):
    c_matrix = np.zeros((n, n))
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            c_matrix[i, j] = dist
            c_matrix[j, i] = dist
    c_obj = np.ones(n) * (-1)
    A_ub_boundary = np.zeros((4 * n, n))
    b_ub_boundary = np.zeros(4 * n)
    for i in range(n):
        A_ub_boundary[4 * i, i] = -1
        b_ub_boundary[4 * i] = -centers[i, 0]
        A_ub_boundary[4 * i + 1, i] = -1
        b_ub_boundary[4 * i + 1] = -(1 - centers[i, 0])
        A_ub_boundary[4 * i + 2, i] = -1
        b_ub_boundary[4 * i + 2] = -centers[i, 1]
        A_ub_boundary[4 * i + 3, i] = -1
        b_ub_boundary[4 * i + 3] = -(1 - centers[i, 1])
    n_pairs = n * (n - 1) // 2
    A_ub_overlap = np.zeros((n_pairs, n))
    b_ub_overlap = np.zeros(n_pairs)
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub_overlap[idx, i] = 1
            A_ub_overlap[idx, j] = 1
            b_ub_overlap[idx] = c_matrix[i, j]
            idx += 1
    A_ub = np.vstack([A_ub_boundary, A_ub_overlap])
    b_ub = np.concatenate([b_ub_boundary, b_ub_overlap])
    bounds = [(0, None)] * n
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return np.maximum(result.x, 1e-10)
    else:
        return np.ones(n) * 0.05


def objective_penalty(x, n, penalty):
    centers = x[:2 * n].reshape(n, 2)
    radii = x[2 * n:]
    obj = -np.sum(radii)
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            min_dist = radii[i] + radii[j]
            if dist < min_dist:
                violation = min_dist - dist
                obj += penalty * violation * violation
    for i in range(n):
        r = radii[i]
        cx, cy = centers[i, 0], centers[i, 1]
        if cx < r:
            obj += penalty * (r - cx) ** 2
        if cx > 1 - r:
            obj += penalty * (cx - (1 - r)) ** 2
        if cy < r:
            obj += penalty * (r - cy) ** 2
        if cy > 1 - r:
            obj += penalty * (cy - (1 - r)) ** 2
    return obj


def optimize_penalty(centers, radii, n):
    x0 = np.concatenate([centers.flatten(), radii])
    bounds = []
    for i in range(n):
        bounds.append((0, 1))
        bounds.append((0, 1))
        bounds.append((0, 0.5))
    x = x0.copy()
    penalty = 100
    for iteration in range(30):
        obj = functools.partial(objective_penalty, n=n, penalty=penalty)
        result = minimize(obj, x, method='L-BFGS-B', bounds=bounds,
                         options={'maxiter': 2000, 'ftol': 1e-15})
        x = result.x
        penalty = penalty * 3
    centers_opt = x[:2 * n].reshape(n, 2)
    radii_opt = x[2 * n:]
    return centers_opt, np.maximum(radii_opt, 1e-10)

def optimize_slqp(centers, radii, n):
    x0 = np.concatenate([centers.flatten(), radii])
    bounds = []
    for i in range(n):
        bounds.append((0.001, 0.999))
        bounds.append((0.001, 0.999))
        bounds.append((0.001, 0.5))
    constraints = []
    for i in range(n):
        constraints.append({'type': 'ineq', 'fun': functools.partial(constraint_x_min, i=i, n=n)})
        constraints.append({'type': 'ineq', 'fun': functools.partial(constraint_x_max, i=i, n=n)})
        constraints.append({'type': 'ineq', 'fun': functools.partial(constraint_y_min, i=i, n=n)})
        constraints.append({'type': 'ineq', 'fun': functools.partial(constraint_y_max, i=i, n=n)})
    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({'type': 'ineq', 'fun': functools.partial(constraint_nonoverlap, i=i, j=j, n=n)})
    obj = functools.partial(objective_neg_sum_radii, n=n)
    result = minimize(obj, x0, method='SLSQP', bounds=bounds, constraints=constraints,
                     options={'maxiter': 20000, 'ftol': 1e-15})
    centers_opt = result.x[:2 * n].reshape(n, 2)
    radii_opt = np.maximum(result.x[2 * n:], 1e-10)
    return centers_opt, radii_opt


def project_feasible(centers, radii, n):
    centers = centers.copy()
    radii = radii.copy()
    for i in range(n):
        r = radii[i]
        centers[i, 0] = np.clip(centers[i, 0], r, 1 - r)
        centers[i, 1] = np.clip(centers[i, 1], r, 1 - r)
    for iteration in range(50):
        max_reduction = 0
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.linalg.norm(centers[i] - centers[j])
                min_dist = radii[i] + radii[j]
                if dist < min_dist - 1e-12:
                    reduction = (min_dist - dist) / 2 + 1e-6
                    radii[i] = max(radii[i] - reduction / 2, 1e-10)
                    radii[j] = max(radii[j] - reduction / 2, 1e-10)
                    max_reduction = max(max_reduction, reduction)
        if max_reduction < 1e-12:
            break
    return centers, radii
