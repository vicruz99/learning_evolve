import numpy as np
import math

def run_packing():
    """
    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    """
    # Constants
    N = 26
    SQRT_3 = math.sqrt(3)
    
    # 1. Initialization
    # Create a hexagonal grid arrangement.
    # Pattern: 6, 5, 6, 5, 4 circles in 5 rows.
    row_counts = [6, 5, 6, 5, 4]
    
    centers = np.zeros((N, 2))
    radii = np.zeros(N)
    current_r = 0.04  # Initial radius, small enough to fit comfortably
    
    # Calculate bounding box for initial placement to ensure uniformity
    # We want to pack them somewhat centrally before expansion starts
    # But placing them in their relative lattice structure is better.
    
    idx = 0
    # Determine spacing based on initial radius
    dx = 2 * current_r
    dy = current_r * SQRT_3
    
    # Center the grid within the square roughly
    # Estimate width and height of the grid
    # Max width is for row with 6 circles: 5 * 2r = 10r
    # Height is 4 rows gap * dy = 4 * 1.732r = 6.928r
    # We place them such that they are roughly centered, but with room to grow
    
    # Let's place the first circle at (current_r, current_r)
    # Actually, let's just place them and the optimizer will move them.
    
    # Better initialization: place them in a dense cluster in the center or spread out?
    # Spread out is safer to avoid local minima, but dense is good for finding the tight packing.
    # Let's try to place them in a valid configuration.
    
    # Re-evaluating row counts to be safer: 5, 6, 5, 6, 4?
    # Let's stick to 6, 5, 6, 5, 4.
    # To fit 6 circles with radius r, width needed is 12r.
    # If r=0.04, width=0.48. Fits easily.
    
    y_pos = current_r
    for row_idx, count in enumerate(row_counts):
        # Offset for staggered rows
        x_offset = 0
        if row_idx % 2 == 1: # Stagger odd rows (1, 3)
             # Shift by radius to nest into gaps?
             # Standard hex: shift by r. But here spacing is 2r.
             # If row above has 6 circles (spacing 2r), row below with 5 should be shifted by r.
             # But our counts are 6, 5, 6...
             # Row 0: 6 circles. Centers at x = r, 3r, 5r...
             # Row 1: 5 circles. Should be at x = 2r, 4r... (shifted by r)
             x_offset = current_r 
        
        # Center the row horizontally?
        # Or just align left?
        # Let's align left at current_r + x_offset
        start_x = current_r + x_offset
        
        for col in range(count):
            x = start_x + col * (2 * current_r)
            centers[idx] = [x, y_pos]
            radii[idx] = current_r
            idx += 1
        
        y_pos += dy
        
    # 2. Optimization Loop
    # We will run a simple gradient ascent / force-directed simulation
    
    num_iterations = 5000
    dt = 1.0
    friction = 0.9
    
    # Growth rate for radii
    growth_rate = 0.0001

    # but for N=26, nested loop is 26*25/2 = 325 checks, very fast.

    velocities = np.zeros((N, 2))
    
    for step in range(num_iterations):
        # Decay growth rate over time
        current_growth = growth_rate * (1.0 - step / num_iterations)
        if current_growth < 1e-6:
            current_growth = 1e-6
            
        # Increase radii
        radii += current_growth
        
        # Reset forces
        forces = np.zeros((N, 2))
        
        # Check boundary constraints and apply forces
        for i in range(N):
            x, y = centers[i]
            r = radii[i]
            
            # Wall constraints
            # If x - r < 0, push right
            if x - r < 0:
                forces[i, 0] += (r - x) * 10.0 # Strong repulsion
            # If x + r > 1, push left
            if x + r > 1:
                forces[i, 0] -= (x + r - 1) * 10.0
            # If y - r < 0, push up
            if y - r < 0:
                forces[i, 1] += (r - y) * 10.0
            # If y + r > 1, push down
            if y + r > 1:
                forces[i, 1] -= (y + r - 1) * 10.0

        # Check inter-circle overlaps
        for i in range(N):
            for j in range(i + 1, N):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist_sq = dx*dx + dy*dy
                dist = math.sqrt(dist_sq)
                
                min_dist = radii[i] + radii[j]
                
                if dist < min_dist and dist > 1e-9:
                    # Overlap detected
                    # Repulsive force proportional to overlap depth
                    overlap = min_dist - dist
                    # Force direction is along the line connecting centers
                    # Unit vector from j to i
                    fx = (dx / dist) * overlap * 5.0
                    fy = (dy / dist) * overlap * 5.0
                    
                    forces[i, 0] += fx
                    forces[i, 1] += fy
                    forces[j, 0] -= fx
                    forces[j, 1] -= fy
        
        # Update velocities and positions
        velocities += forces * dt
        velocities *= friction
        centers += velocities * dt

            r = radii[i]

            y = max(r, min(1.0 - r, y))
            centers[i] = [x, y]

            r = radii[i]
            x = max(r, min(1.0 - r, x))
            y = max(r, min(1.0 - r, y))
            centers[i] = [x, y]

    # Final check to ensure strict validity
    # If any overlap exists, shrink the overlapping circles slightly until valid.
    # This is a fallback.
    
    # Re-validate and fix
    changed = True
    iterations = 0

        iterations += 1

                    new_r_j = radii[j] - reduction
                    # Ensure non-negative
                    if new_r_i < 0: new_r_i = 0
                    if new_r_j < 0: new_r_j = 0

                    # If shrinking doesn't fix boundary, we have bigger issues.
                    # But usually shrinking helps boundary too.
                    
                    # To keep sum high, maybe shrink less?
                    # But we must be valid.
                    # Let's just reduce the sum_r to dist.
                    # Distribute reduction proportional to current radii?
                    # Or just equal.
                    
                    # Better: reduce the pair sum to match distance.
                    # Keep ratio?
                    ratio_i = radii[i] / sum_r
                    ratio_j = radii[j] / sum_r
                    
                    target_sum = dist
                    radii[i] = target_sum * ratio_i
                    radii[j] = target_sum * ratio_j

        # Fix boundaries
        for i in range(N):
            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:

                max_r_x = min(x, 1-x)
                max_r_y = min(y, 1-y)

                if new_r < radii[i]:
                    radii[i] = new_r
                    changed = True
                    if radii[i] < 0: radii[i] = 0

    sum_radii = np.sum(radii)
    return centers, radii, float(sum_radii)
