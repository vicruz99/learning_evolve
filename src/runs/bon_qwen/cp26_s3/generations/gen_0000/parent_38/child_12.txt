import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Packs 26 circles in a unit square [0,1]x[0,1] to maximize the sum of radii.
    
    Returns:
        centers: np.array of shape (26, 2)
        radii: np.array of shape (26,)
        sum_radii: float
    """
    N = 26
    
    # 1. Generate Initial Configuration (Hexagonal Lattice)
    # We aim for a layout that fits well. 
    # A 6-row hexagonal packing is a good starting point for N=26.
    # Rows will have counts: 5, 5, 5, 5, 4, 2 (Sum = 26)
    # This arrangement allows for a denser packing than 5x5.
    
    centers = []
    radii = []
    
    # Estimated radius based on hexagonal packing height constraints for 6 rows
    # Height approx: 2*r + (rows-1)*sqrt(3)*r <= 1
    # r * (2 + 5*1.732) <= 1 => r * 10.66 <= 1 => r approx 0.093
    # However, with 26 circles, we might squeeze more. Let's start with r=0.095
    r_init = 0.095
    row_counts = [5, 5, 5, 5, 4, 2]
    
    current_circle_idx = 0
    
    for row_idx, count in enumerate(row_counts):
        # Vertical position of the row
        # Offset every other row by r horizontally
        y = r_init + row_idx * np.sqrt(3) * r_init
        
        # Horizontal start offset
        if row_idx % 2 == 0:
            x_start = r_init
        else:
            x_start = r_init + r_init # Shift by r for nesting
        
        for col_idx in range(count):
            x = x_start + col_idx * 2 * r_init
            centers.append([x, y])
            radii.append(r_init)
            
            if current_circle_idx + 1 >= N:
                break
            current_circle_idx += 1
        if current_circle_idx + 1 >= N:
            break
            
    # Trim or pad if necessary (though logic above should produce exactly N)
    centers = np.array(centers[:N])
    radii = np.array(radii[:N])
    
    # 2. Optimization
    # Variables: [x1, y1, r1, x2, y2, r2, ...]
    # Total variables: 3 * N = 78
    
    def flatten(vars_dict):
        """Flatten variables into 1D array for optimizer"""
        arr = np.zeros(3 * N)
        for i in range(N):
            arr[3*i] = vars_dict['x'][i]
            arr[3*i+1] = vars_dict['y'][i]
            arr[3*i+2] = vars_dict['r'][i]
        return arr

    def unflatten(arr):
        """Unflatten 1D array to variables"""
        x = arr[0::3]
        y = arr[1::3]
        r = arr[2::3]
        return x, y, r

    def objective(vars_arr):
        """Objective: Maximize sum of radii (Minimize negative sum)"""
        _, _, r = unflatten(vars_arr)
        return -np.sum(r)

    def constraints_func(vars_arr):
        """Returns list of constraint values (must be >= 0)"""
        x, y, r = unflatten(vars_arr)
        cons = []
        
        # Boundary constraints: 0 <= x-r, x+r <= 1, etc.
        # x - r >= 0
        cons.extend(x - r)
        # 1 - x - r >= 0
        cons.extend(1 - x - r)
        # y - r >= 0
        cons.extend(y - r)
        # 1 - y - r >= 0
        cons.extend(1 - y - r)
        
        # Non-overlap constraints: dist^2 >= (r_i + r_j)^2
        # dist^2 - (r_i + r_j)^2 >= 0
        for i in range(N):
            for j in range(i + 1, N):
                dx = x[i] - x[j]
                dy = y[i] - y[j]
                dist_sq = dx*dx + dy*dy
                r_sum = r[i] + r[j]
                cons.append(dist_sq - r_sum*r_sum)
                
        return np.array(cons)

    # Initial guess for optimizer
    x0 = flatten({'x': centers[:, 0], 'y': centers[:, 1], 'r': radii})
    
    # Bounds for variables: x, y in [0, 1], r >= 0
    # Note: r <= 0.5 implicitly handled by boundary constraints, but good to bound
    bounds = []
    for _ in range(N):
        bounds.extend([
            (0, 1), # x
            (0, 1), # y
            (0, 0.5), # r
        ])
    
    # Define constraints for SLSQP
    # SLSQP expects a dictionary or a list of dictionaries
    # For efficiency with many constraints, we can pass a single constraint function
    # However, SLSQP in scipy usually handles list of dicts or a single dict with 'fun' returning array.
    # We'll use a single constraint object.
    
    constraint = {
        'type': 'ineq',
        'fun': constraints_func
    }
    
    # Run Optimization
    # We might need multiple restarts or a robust method.
    # SLSQP is generally good for this scale.
    
    res = minimize(
        objective,
        x0,
        method='SLSQP',
        bounds=bounds,
        constraints=[constraint],
        options={'ftol': 1e-9, 'maxiter': 1000, 'disp': False}
    )
    
    if res.success:
        best_x, best_y, best_r = unflatten(res.x)
    else:
        # If optimization fails, fallback to initial configuration (might be valid)
        # Or try to repair by scaling down radii slightly
        best_x = centers[:, 0]
        best_y = centers[:, 1]
        best_r = radii
        
        # Simple repair: if overlaps, shrink radii slightly
        # This is a fallback to ensure validity
        for _ in range(100):
            overlap = False
            for i in range(N):
                for j in range(i+1, N):
                    d = np.sqrt((best_x[i]-best_x[j])**2 + (best_y[i]-best_y[j])**2)
                    if d < best_r[i] + best_r[j] - 1e-9:
                        # Shrink both slightly
                        factor = 0.999
                        best_r[i] *= factor
                        best_r[j] *= factor
                        overlap = True
            if not overlap:
                break
    
    # Ensure non-negative radii
    best_r = np.maximum(best_r, 0)
    
    final_centers = np.column_stack((best_x, best_y))
    final_radii = best_r
    final_sum = np.sum(final_radii)
    
    return final_centers, final_radii, final_sum
