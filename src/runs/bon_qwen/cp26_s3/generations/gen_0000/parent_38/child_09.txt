import numpy as np
from scipy.optimize import minimize


def compute_objective(x, n, penalty_weight):
    """Compute objective: negative sum of radii plus penalty for constraint violations."""
    centers = x[:2 * n].reshape(n, 2)
    radii = x[2 * n:]
    
    penalty = 0.0
    
    # Boundary violation penalties
    for i in range(n):
        r = radii[i]
        xi, yi = centers[i]
        penalty += max(0.0, r - xi) ** 2
        penalty += max(0.0, r - (1.0 - xi)) ** 2
        penalty += max(0.0, r - yi) ** 2
        penalty += max(0.0, r - (1.0 - yi)) ** 2
    
    # Overlap violation penalties
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            min_dist = radii[i] + radii[j]
            if dist < min_dist:
                penalty += (min_dist - dist) ** 2
    
    return -np.sum(radii) + penalty_weight * penalty


def force_directed_phase(centers, radii, n, iterations, dt, growth_rate, repulsion_strength):
    """Run force-directed layout to grow circles and resolve overlaps."""
    for _ in range(iterations):
        forces = np.zeros_like(centers)
        
        # Repulsion between overlapping circles
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[j, 0] - centers[i, 0]
                dy = centers[j, 1] - centers[i, 1]
                dist_sq = dx * dx + dy * dy
                dist = np.sqrt(dist_sq) if dist_sq > 1e-20 else 1e-10
                min_dist = radii[i] + radii[j]
                if dist < min_dist:
                    overlap = min_dist - dist
                    force_mag = repulsion_strength * overlap / dist
                    fx = force_mag * dx
                    fy = force_mag * dy
                    forces[i, 0] -= fx
                    forces[i, 1] -= fy
                    forces[j, 0] += fx
                    forces[j, 1] += fy
        
        # Boundary repulsion
        for i in range(n):
            r = radii[i]
            if centers[i, 0] < r:
                forces[i, 0] += repulsion_strength * (r - centers[i, 0])
            if centers[i, 0] > 1.0 - r:
                forces[i, 0] -= repulsion_strength * (centers[i, 0] - (1.0 - r))
            if centers[i, 1] < r:
                forces[i, 1] += repulsion_strength * (r - centers[i, 1])
            if centers[i, 1] > 1.0 - r:
                forces[i, 1] -= repulsion_strength * (centers[i, 1] - (1.0 - r))
        
        # Update positions
        centers += dt * forces
        
        # Clamp to valid region
        for i in range(n):
            r = radii[i]
            centers[i, 0] = max(r, min(1.0 - r, centers[i, 0]))
            centers[i, 1] = max(r, min(1.0 - r, centers[i, 1]))
        
        # Grow circles
        radii += growth_rate
    
    return centers, radii


def make_hexagonal_init(n, row_counts, init_r):
    """Create hexagonal initial arrangement."""
    centers = np.zeros((n, 2))
    radii = np.full(n, init_r)
    
    idx = 0
    for row, count in enumerate(row_counts):
        y = init_r + row * init_r * np.sqrt(3)
        if count > 1:
            spacing = (1.0 - 2.0 * init_r) / (count - 1)
        else:
            spacing = 0.0
        
        if row % 2 == 0:
            x_start = init_r
        else:
            x_start = init_r + spacing / 2.0
        
        for col in range(count):
            centers[idx, 0] = x_start + col * spacing
            centers[idx, 1] = y
            idx += 1
    
    return centers, radii


def make_grid_init(n, init_r):
    """Create grid initial arrangement."""
    centers = np.zeros((n, 2))
    radii = np.full(n, init_r)
    
    cols = int(np.ceil(np.sqrt(n)))
    rows = int(np.ceil(n / cols))
    
    if cols > 1:
        x_spacing = (1.0 - 2.0 * init_r) / (cols - 1)
    else:
        x_spacing = 0.0
    if rows > 1:
        y_spacing = (1.0 - 2.0 * init_r) / (rows - 1)
    else:
        y_spacing = 0.0
    
    idx = 0
    for row in range(rows):
        for col in range(cols):
            if idx >= n:
                break
            centers[idx, 0] = init_r + col * x_spacing
            centers[idx, 1] = init_r + row * y_spacing
            idx += 1
    
    return centers, radii


def optimize_from(centers, radii, n, penalty_weights):
    """Run scipy optimization from given initial state."""
    best_sum = -np.inf
    best_centers = centers.copy()
    best_radii = radii.copy()
    
    for pw in penalty_weights:
        obj_func = lambda x, w=pw: compute_objective(x, n, w)
        
        x0 = np.concatenate([centers.flatten(), radii])
        bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
        
        try:
            result = minimize(obj_func, x0, method='L-BFGS-B', bounds=bounds,
                              options={'maxiter': 100000, 'ftol': 1e-15, 'gtol': 1e-12})
            
            centers_opt = result.x[:2 * n].reshape(n, 2)
            radii_opt = result.x[2 * n:]
            radii_opt = np.maximum(radii_opt, 0.0)
            
            current_sum = np.sum(radii_opt)
            if current_sum > best_sum:
                best_sum = current_sum
                best_centers = centers_opt.copy()
                best_radii = radii_opt.copy()
        except Exception:
            pass
    
    return best_centers, best_radii, best_sum


def correct_solution(centers, radii, n, iterations=500):
    """Post-process to ensure validity and try to improve."""
    centers = centers.copy()
    radii = radii.copy()
    
    for _ in range(iterations):
        # Resolve overlaps by pushing circles apart
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[j, 0] - centers[i, 0]
                dy = centers[j, 1] - centers[i, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                min_dist = radii[i] + radii[j]
                if dist < min_dist and dist > 1e-12:
                    overlap = min_dist - dist
                    push = overlap / (2.0 * dist)
                    centers[i, 0] -= push * dx
                    centers[i, 1] -= push * dy
                    centers[j, 0] += push * dx
                    centers[j, 1] += push * dy
        
        # Clamp to bounds
        for i in range(n):
            r = radii[i]
            centers[i, 0] = max(r, min(1.0 - r, centers[i, 0]))
            centers[i, 1] = max(r, min(1.0 - r, centers[i, 1]))
    
    return centers, radii


def run_packing():
    n = 26
    
    best_sum = -np.inf
    best_centers = None
    best_radii = None
    
    # Different initialization strategies
    init_configs = [
        # Hexagonal: 6+5+6+5+4 = 26
        make_hexagonal_init(n, [6, 5, 6, 5, 4], 0.07),
        make_hexagonal_init(n, [5, 6, 5, 6, 4], 0.07),
        make_hexagonal_init(n, [6, 5, 6, 5, 4], 0.08),
        make_hexagonal_init(n, [5, 6, 5, 6, 4], 0.08),
        # Grid
        make_grid_init(n, 0.07),
        make_grid_init(n, 0.08),
        # Alternative hexagonal patterns
        make_hexagonal_init(n, [6, 6, 5, 5, 4], 0.07),
        make_hexagonal_init(n, [4, 5, 6, 5, 6], 0.07),
    ]
    
    penalty_weights = [5000, 10000, 20000, 50000]
    
    for init_centers, init_radii in init_configs:
        # Phase 1: Force-directed relaxation
        fd_centers, fd_radii = force_directed_phase(
            init_centers.copy(), init_radii.copy(), n,
            iterations=30000, dt=0.003, growth_rate=0.000015, repulsion_strength=15.0
        )
        
        # Phase 2: scipy optimization
        opt_centers, opt_radii, opt_sum = optimize_from(fd_centers, fd_radii, n, penalty_weights)
        
        # Phase 3: Post-processing
        opt_centers, opt_radii = correct_solution(opt_centers, opt_radii, n, iterations=200)
        opt_radii = np.maximum(opt_radii, 0.0)
        
        # Verify and re-optimize if needed
        current_sum = np.sum(opt_radii)
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = opt_centers.copy()
            best_radii = opt_radii.copy()
    
    # Final optimization pass on best solution
    best_centers, best_radii, best_sum = optimize_from(best_centers, best_radii, n, [10000, 50000, 100000])
    best_centers, best_radii = correct_solution(best_centers, best_radii, n, iterations=300)
    best_radii = np.maximum(best_radii, 0.0)
    best_sum = np.sum(best_radii)
    
    # Ensure centers are valid
    for i in range(n):
        r = best_radii[i]
        best_centers[i, 0] = max(r + 1e-10, min(1.0 - r - 1e-10, best_centers[i, 0]))
        best_centers[i, 1] = max(r + 1e-10, min(1.0 - r - 1e-10, best_centers[i, 1]))
    
    return best_centers, best_radii, best_sum
