import numpy as np
from scipy.optimize import minimize

def run_packing():
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    """
    np.random.seed(42)
    n_circles = 26
    target_n = 30  # Start with more circles to ensure coverage

    # 1. Generate Initial Hexagonal Lattice
    # This provides a dense starting configuration.
    centers = []
    r_init = 0.08
    # Estimate grid size
    for r in range(6):
        for c in range(6):
            x = c * 1.5 * (2 * r_init)
            y = r * np.sqrt(3) * r_init
            if c % 2 == 1:
                x += 1.5 * r_init
            if x + r_init <= 1.0 and y + r_init <= 1.0:
                centers.append([x, y])
    
    # Pad or truncate to target_n
    while len(centers) < target_n:
        centers.append([0.5, 0.5]) # Add dummy centers if short
    centers = np.array(centers[:target_n])
    radii = np.full(target_n, r_init)

    # 2. Define Objective and Constraints
    def objective(x_flat):
        # x_flat contains [x1, y1, r1, x2, y2, r2, ...]
        radii = x_flat[2::3]
        # Maximize sum of radii => Minimize negative sum
        return -np.sum(radii)

    def constraint_inside(x_flat):
        # 0 <= x - r and x + r <= 1 (same for y)
        # x >= r => x - r >= 0
        # x <= 1 - r => x + r <= 1
        constraints = []
        for i in range(target_n):
            idx = i * 3
            x, y, r = x_flat[idx], x_flat[idx+1], x_flat[idx+2]
            constraints.append(x - r)       # left
            constraints.append(y - r)       # bottom
            constraints.append(1 - x - r)   # right
            constraints.append(1 - y - r)   # top
        return np.array(constraints)

    def constraint_no_overlap(x_flat):
        constraints = []
        for i in range(target_n):
            for j in range(i + 1, target_n):
                idx_i = i * 3
                idx_j = j * 3
                x1, y1, r1 = x_flat[idx_i], x_flat[idx_i+1], x_flat[idx_i+2]
                x2, y2, r2 = x_flat[idx_j], x_flat[idx_j+1], x_flat[idx_j+2]
                
                dist = np.sqrt((x1 - x2)**2 + (y1 - y2)**2)
                constraints.append(dist - r1 - r2)
        return np.array(constraints)

    # 3. Setup Optimization
    x0 = np.concatenate([centers.flatten(), radii])
    
    bounds = []
    for _ in range(target_n):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((1e-5, 0.5)) # r (small lower bound, 0.5 upper bound)

    cons_inside = {'type': 'ineq', 'fun': constraint_inside}
    cons_overlap = {'type': 'ineq', 'fun': constraint_no_overlap}
    
    # 4. Run Optimization (SLSQP)
    try:
        res = minimize(objective, x0, method='SLSQP', bounds=bounds, 
                       constraints=[cons_inside, cons_overlap],
                       options={'ftol': 1e-9, 'maxiter': 1000, 'disp': False})
        x_opt = res.x
    except Exception:
        x_opt = x0

    # 5. Post-process to exactly 26 circles
    # We have 30 circles. We want to keep the 26 largest.
    radii_all = x_opt[2::3]
    centers_all = np.array([x_opt[i*3 : i*3+2] for i in range(target_n)])
    
    # Sort by radius descending
    sorted_indices = np.argsort(radii_all)[::-1]
    final_indices = sorted_indices[:n_circles]
    
    final_centers = centers_all[final_indices]
    final_radii = radii_all[final_indices]
    
    # Final check and normalization if needed (though solver should handle it)
    sum_radii = np.sum(final_radii)
    
    return final_centers, final_radii, sum_radii

if __name__ == "__main__":
    centers, radii, s = run_packing()
    print(f"Sum of radii: {s}")
