import numpy as np
from scipy.optimize import linprog, minimize

def solve_radii_lp(centers):
    """
    Solves the Linear Programming problem to find optimal radii for fixed centers.
    Maximizes sum(r_i) subject to non-overlap and boundary constraints.
    """
    N = centers.shape[0]
    
    # Objective: maximize sum(r_i) -> minimize -sum(r_i)
    c_obj = -np.ones(N)
    
    A_ub = []
    b_ub = []
    
    # Wall constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(N):
        x, y = centers[i]
        # r_i <= x
        row = np.zeros(N)
        row[i] = 1.0
        A_ub.append(row)
        b_ub.append(x)
        
        # r_i <= 1 - x
        row = np.zeros(N)
        row[i] = 1.0
        A_ub.append(row)
        b_ub.append(1.0 - x)
        
        # r_i <= y
        row = np.zeros(N)
        row[i] = 1.0
        A_ub.append(row)
        b_ub.append(y)
        
        # r_i <= 1 - y
        row = np.zeros(N)
        row[i] = 1.0
        A_ub.append(row)
        b_ub.append(1.0 - y)
        
    # Pairwise constraints: r_i + r_j <= d_ij
    # Compute distances
    dists = np.zeros((N, N))
    for i in range(N):
        for j in range(i + 1, N):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            d = np.hypot(dx, dy)
            dists[i, j] = d
            dists[j, i] = d
            
            # Constraint: r_i + r_j <= d
            row = np.zeros(N)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(d)
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    # Bounds for radii: [0, 0.5]
    bounds = [(0, 0.5) for _ in range(N)]
    
    # Solve LP
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return -res.fun, res.x
        else:
            return 0.0, np.zeros(N)
    except Exception:
        return 0.0, np.zeros(N)

def get_objective(centers_flat):
    """Objective function for optimization: negative sum of radii."""
    centers = centers_flat.reshape(26, 2)
    # Ensure centers are within bounds for stability, though LP handles it
    centers = np.clip(centers, 0, 1)
    val, _ = solve_radii_lp(centers)
    return -val

def generate_hex_grid(n_points):
    """Generates a hexagonal grid of points inside [0,1]x[0,1]."""
    points = []
    # Approximate number of points per row
    # Area ~ n * (sqrt(3)/2 * s^2). s ~ 1/sqrt(n). 
    # Let's just fill with a reasonable spacing
    spacing = 0.2
    dy = spacing * np.sqrt(3) / 2
    
    y = 0.1
    while y <= 0.9:
        x = 0.1
        offset = 0
        if len(points) % 2 == 1: # Offset every other row
             offset = spacing / 2
        
        while x <= 0.9:
            points.append([x + offset, y])
            if len(points) >= n_points:
                break
            x += spacing
        if len(points) >= n_points:
            break
        y += dy
        
    if len(points) < n_points:
        # Fallback to random if not enough
        pts = np.random.uniform(0.1, 0.9, (n_points, 2))
        return pts
    
    return np.array(points[:n_points])

def run_packing():
    N = 26
    best_sum = 0
    best_centers = None
    best_radii = None
    
    # Initial configuration: Hexagonal grid
    init_centers = generate_hex_grid(N)
    
    # Optimization using Nelder-Mead (good for non-smooth objectives from LP active sets)
    # We run multiple restarts with different seeds/initial perturbations
    best_sol = None
    best_obj = np.inf
    
    for seed in range(5):
        # Perturb initial centers slightly
        current_centers = init_centers.copy()
        if seed > 0:
            noise = np.random.uniform(-0.05, 0.05, current_centers.shape)
            current_centers += noise
            current_centers = np.clip(current_centers, 0.01, 0.99)
            
        # Flatten for optimizer
        x0 = current_centers.flatten()
        
        # Bounds for centers
        bounds = [(0, 1) for _ in range(2 * N)]
        
        # Optimize
        # Nelder-Mead doesn't use bounds directly in all versions, but we can clip in objective
        # Or use SLSQP/L-BFGS-B. Nelder-Mead is robust.
        res = minimize(get_objective, x0, method='Nelder-Mead', 
                       options={'maxiter': 5000, 'xatol': 1e-6, 'fatol': 1e-6})
        
        if res.fun < best_obj:
            best_obj = res.fun
            best_sol = res.x
            
    # Extract best solution
    if best_sol is not None:
        final_centers = best_sol.reshape(N, 2)
        # Final LP solve to get radii
        sum_r, final_radii = solve_radii_lp(final_centers)
        
        # Validate and return
        return final_centers, final_radii, sum_r
    else:
        # Fallback
        return init_centers, np.zeros(N), 0.0
