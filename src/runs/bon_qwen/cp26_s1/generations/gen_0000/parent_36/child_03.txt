            spacing = 2 * r_init

    best_radii = radii.copy()

        A_ub[idx, j] = 1.0

            np.add.at(forces, u, directions)

        if res.success:
            best_radii = res.x
            best_sum = np.sum(best_radii)
    except:
        pass

    return best_centers, best_radii, best_sum
