import numpy as np
from scipy.optimize import minimize

def compute_overlap_energy(params, R, n):
    """
    Computes the squared overlap energy for n circles of radius R.
    Energy is zero if and only if no circles overlap.
    """
    centers = params.reshape(n, 2)
    # Compute all pairwise distances efficiently
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    # Extract lower triangle to avoid duplicates and self-distances
    i, j = np.tril_indices(n, -1)
    d = dists[i, j]
    
    # Overlap penalty: (2R - d)^2 when d < 2R, else 0
    overlap = 2*R - d
    mask = overlap > 0
    return np.sum((overlap[mask])**2)

def get_initial_positions(n, R):
    """
    Generates initial circle positions using a hexagonal lattice pattern
    perturbed by noise, filling up to n circles.
    """
    centers = []
    row = 0
    while len(centers) < n:
        y = R + row * R * np.sqrt(3)
        if y + R > 1.0:
            break
        col = 0
        while len(centers) < n:
            shift = R if row % 2 == 1 else 0.0
            x = R + shift + col * 2*R
            if x + R > 1.0:
                break
            centers.append([x, y])
            col += 1
        row += 1
        
    # Fill remaining spots randomly if lattice didn't reach n
    while len(centers) < n:
        centers.append([np.random.uniform(R, 1-R), np.random.uniform(R, 1-R)])
        
    arr = np.array(centers).flatten()
    # Add small perturbation to break symmetry and avoid flat basins
    arr += np.random.normal(0, R*0.05, arr.shape)
    arr = np.clip(arr, R, 1-R)
    return arr

def run_packing():
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    Returns (centers, radii, sum_radii).
    """
    n = 26
    lo, hi = 0.05, 0.15
    best_params = None
    best_R = lo

    # Binary search for the maximum feasible radius
    for _ in range(40):
        mid = (lo + hi) / 2
        found = False
        
        # Multiple restarts to escape local minima
        for _ in range(10):
            x0 = get_initial_positions(n, mid)
            bounds = [(mid, 1-mid)] * (2*n)
            
            res = minimize(
                compute_overlap_energy, 
                x0, 
                args=(mid, n), 
                method='L-BFGS-B', 
                bounds=bounds, 
                options={'maxiter': 1500, 'ftol': 1e-12}
            )
            
            # If overlap energy is negligible, the packing is valid
            if res.fun < 1e-7:
                found = True
                best_params = res.x
                break
                
        if found:
            best_R = mid
            lo = mid
        else:
            hi = mid

    centers = best_params.reshape(n, 2)
    # Slight shrinkage guarantees strict validity against float precision limits
    radii = np.full(n, best_R * 0.9999)
    
    return centers, radii, np.sum(radii)
