        # x = r + (1 - 2r) * u
        # y = r + (1 - 2r) * v
        x = r + (1 - 2 * r) * u
        y = r + (1 - 2 * r) * v
        centers = np.column_stack((x, y))
        return centers, r

    def objective(vars):
        centers, r = decode_variables(vars)
        
        # Main objective: maximize sum of radii (minimize negative)
        val = -np.sum(r)

        # Overlap condition: dist^2 < (r_i + r_j)^2
