import numpy as np
from scipy.optimize import linprog
import time


def solve_radii_lp(centers):
    """Given fixed centers, solve LP to maximize sum of radii."""
    n = centers.shape[0]
    c = -np.ones(n)
    
    A_list = []
    b_list = []
    
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        x, y = centers[i]
        for bound in [x, 1.0 - x, y, 1.0 - y]:
            row = np.zeros(n)
            row[i] = 1.0
            A_list.append(row)
            b_list.append(bound)
    
    # Non-overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i][0] - centers[j][0]
            dy = centers[i][1] - centers[j][1]
            dist = np.sqrt(dx * dx + dy * dy)
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_list.append(row)
            b_list.append(dist)
    
    A_ub = np.array(A_list)
    b_ub = np.array(b_list)
    
    bounds = [(0.0, None) for _ in range(n)]
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        radii = np.maximum(result.x, 0.0)
        return radii, -result.fun
    else:
        r = np.full(n, 0.01)
        return r, float(sum(r))


def generate_hexagonal_centers(n):
    """Generate hexagonal grid centers for n circles."""
    centers = []
    
    # Estimate spacing for n circles
    # For hexagonal packing, each circle effectively occupies area ~ pi*r^2 / 0.9069
    # With sum of radii target ~2.636, avg radius ~0.101, spacing ~0.2
    spacing = 0.18
    
    y = spacing * 0.5
    row = 0
    while len(centers) < n:
        x_offset = 0.0 if row % 2 == 0 else spacing * np.sqrt(3) * 0.5
        x = spacing * 0.5
        while x + x_offset <= 1.0 and len(centers) < n:
            cx = x + x_offset
            cy = y
            if 0 <= cx <= 1 and 0 <= cy <= 1:
                centers.append([cx, cy])
            x += spacing * np.sqrt(3)
        y += spacing
        row += 1
    
    return np.array(centers[:n])


def generate_grid_centers(n):
    """Generate grid centers for n circles."""
    rows = int(np.ceil(np.sqrt(n)))
    cols = int(np.ceil(float(n) / rows))
    
    margin_x = 0.05
    margin_y = 0.05
    available_x = 1.0 - 2 * margin_x
    available_y = 1.0 - 2 * margin_y
    
    spacing_x = available_x / (cols - 1) if cols > 1 else 0
    spacing_y = available_y / (rows - 1) if rows > 1 else 0
    
    centers = []
    for i in range(rows):
        for j in range(cols):
            if len(centers) < n:
                x = margin_x + j * spacing_x
                y = margin_y + i * spacing_y
                centers.append([x, y])
    
    return np.array(centers[:n])


def generate_random_centers(n):
    """Generate random centers."""
    return np.random.uniform(0.08, 0.92, (n, 2))


def simulated_annealing(centers, n, iterations=8000, init_temp=0.03):
    """Run simulated annealing to optimize center positions."""
    centers = centers.copy()
    radii, current_sum = solve_radii_lp(centers)
    
    best_centers = centers.copy()
    best_radii = radii.copy()
    best_sum = current_sum
    
    temp = init_temp
    cooling_rate = 0.9995
    
    for iteration in range(iterations):
        i = np.random.randint(n)
        
        dx = np.random.uniform(-temp, temp)
        dy = np.random.uniform(-temp, temp)
        
        new_centers = centers.copy()
        new_centers[i][0] = np.clip(new_centers[i][0] + dx, 0.0, 1.0)
        new_centers[i][1] = np.clip(new_centers[i][1] + dy, 0.0, 1.0)
        
        new_radii, new_sum = solve_radii_lp(new_centers)
        
        if new_sum > best_sum:
            best_sum = new_sum
            best_centers = new_centers.copy()
            best_radii = new_radii.copy()
        
        if new_sum > current_sum:
            centers = new_centers.copy()
            current_sum = new_sum
        else:
            diff = new_sum - current_sum
            if temp > 1e-10 and np.random.random() < np.exp(diff / temp):
                centers = new_centers.copy()
                current_sum = new_sum
        
        temp *= cooling_rate
    
    return best_centers, best_radii, best_sum


def coordinate_refine(centers, n, max_iter=500):
    """Refine centers using coordinate-wise 8-direction search."""
    centers = centers.copy()
    radii, current_sum = solve_radii_lp(centers)
    
    best_centers = centers.copy()
    best_radii = radii.copy()
    best_sum = current_sum
    
    step = 0.02
    shrink = 0.9
    
    for iteration in range(max_iter):
        improved = False
        
        for i in range(n):
            best_local = current_sum
            best_pos = centers[i].copy()
            
            for dx in [-step, 0.0, step]:
                for dy in [-step, 0.0, step]:
                    if dx == 0.0 and dy == 0.0:
                        continue
                    
                    new_centers = centers.copy()
                    new_centers[i][0] = np.clip(centers[i][0] + dx, 0.0, 1.0)
                    new_centers[i][1] = np.clip(centers[i][1] + dy, 0.0, 1.0)
                    
                    new_radii, new_sum = solve_radii_lp(new_centers)
                    
                    if new_sum > best_local:
                        best_local = new_sum
                        best_pos = new_centers[i].copy()
            
            if best_local > current_sum:
                centers[i] = best_pos
                current_sum = best_local
                improved = True
        
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers.copy()
            best_radii = solve_radii_lp(centers)[0]
        
        if not improved:
            step *= shrink
            if step < 1e-7:
                break
    
    return best_centers, best_radii, best_sum
