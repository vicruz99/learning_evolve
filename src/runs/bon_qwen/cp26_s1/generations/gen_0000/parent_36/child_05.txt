import numpy as np
from scipy.optimize import minimize

def _objective(vars):
    """Objective function: maximize the radius r."""
    return -vars[-1]

def _constraints(vars):
    """Constraint function: returns array of constraint values (must be >= 0)."""
    N = 26
    centers = vars[:-1].reshape(N, 2)
    r = vars[-1]
    
    # Number of constraints: 4 boundaries per circle + pairwise non-overlap
    n_cons = 4 * N + N * (N - 1) // 2
    vals = np.empty(n_cons)
    idx = 0
    
    # Boundary constraints: x >= r, x <= 1-r, y >= r, y <= 1-r
    for i in range(N):
        vals[idx] = centers[i, 0] - r
        vals[idx+1] = 1.0 - centers[i, 0] - r
        vals[idx+2] = centers[i, 1] - r
        vals[idx+3] = 1.0 - centers[i, 1] - r
        idx += 4
        
    # Non-overlap constraints: ||c_i - c_j||^2 >= (2r)^2
    for i in range(N):
        for j in range(i + 1, N):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            vals[idx] = dx*dx + dy*dy - 4.0 * r * r
            idx += 1
            
    return vals

def run_packing():
    N = 26
    best_r = 0.0
    best_centers = None

    # Run multiple optimizations with different initializations
    for seed in range(40):
        np.random.seed(seed)
        
        # Hexagonal lattice initialization
        cx = np.zeros((N, 2))
        idx = 0
        row = 0
        while idx < N:
            y = 0.1 + row * 0.16
            col = 0
            while idx < N:
                x = 0.1 + col * 0.14 + (row % 2) * 0.07
                if x < 0.9 and y < 0.9:
                    cx[idx] = [x, y]
                    idx += 1
                col += 1
            row += 1
        
        # Add small random jitter to break symmetry and help escape local minima
        cx += np.random.uniform(-0.015, 0.015, (N, 2))
        cx = np.clip(cx, 0.1, 0.9)
        
        # Initial variables: centers flattened + initial radius
        x0 = np.concatenate([cx.ravel(), [0.045]])
        
        cons = {'type': 'ineq', 'fun': _constraints}
        bounds = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)]
        
        try:
            res = minimize(_objective, x0, method='SLSQP', 
                          bounds=bounds, constraints=cons, 
                          options={'maxiter': 3000, 'ftol': 1e-10, 'disp': False})
            
            if not np.isnan(res.x).any():
                current_r = res.x[-1]
                if current_r > best_r:
                    # Verify constraints are satisfied within numerical tolerance
                    cons_vals = _constraints(res.x)
                    if np.all(cons_vals >= -1e-6):
                        best_r = current_r
                        best_centers = res.x[:-1].reshape(N, 2)
        except Exception:
            continue

    # Fallback to simple grid if optimization fails entirely
    if best_centers is None:
        cx = np.zeros((N, 2))
        idx = 0
        for i in range(5):
            for j in range(5):
                if idx < N:
                    cx[idx] = [0.1 + j * 0.2, 0.1 + i * 0.2]
                    idx += 1
        best_centers = cx
        best_r = 0.1
        
    centers = best_centers
    radii = np.full(N, best_r)
    
    # Final adjustment to guarantee strict validity against the checker
    # Compute the actual maximum radius allowed by the geometry of the found centers
    min_sep = float('inf')
    for i in range(N):
        for j in range(i + 1, N):
            d = np.linalg.norm(centers[i] - centers[j])
            if d < min_sep:
                min_sep = d
    for i in range(N):
        d_b = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        if d_b < min_sep:
            min_sep = d_b
            
    valid_r = min_sep / 2.0
    if valid_r < radii[0]:
        radii[:] = valid_r
        
    return centers, radii, np.sum(radii)
