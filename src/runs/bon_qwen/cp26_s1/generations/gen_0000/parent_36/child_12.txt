import numpy as np
from scipy.optimize import minimize

def compute_constraints(v, n, margin=1e-9):
    """
    Computes all inequality constraints for the packing problem.
    v: optimization variables [x1, y1, r1, ..., xn, yn, rn]
    n: number of circles
    margin: small safety margin to ensure strict feasibility
    Returns array of constraint values that must be >= 0
    """
    centers = v[:2*n].reshape(n, 2)
    radii = v[2*n:]
    
    cons = []
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    cons.append(centers[:, 0] - radii + margin)
    cons.append(1.0 - centers[:, 0] - radii + margin)
    cons.append(centers[:, 1] - radii + margin)
    cons.append(1.0 - centers[:, 1] - radii + margin)
    
    # Pairwise non-overlap constraints: dist >= r1 + r2
    # Compute pairwise distances efficiently
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.linalg.norm(diff, axis=2)
    r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    
    # Only need upper triangle (i < j)
    triu_idx = np.triu_indices(n, k=1)
    cons.append(dists[triu_idx] - r_sum[triu_idx] + margin)
    
    return np.concatenate(cons)

def run_packing():
    n = 26
    
    def objective(v):
        # We maximize sum of radii, so minimize negative sum
        return -np.sum(v[2*n:])
        
    # Bounds: coordinates in [0,1], radii in [0, 0.5]
    bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n
    
    best_v = None
    best_sum_r = -np.inf
    
    # Constraint dictionary for SLSQP
    cons = {'type': 'ineq', 'fun': lambda v: compute_constraints(v, n)}
    
    # Generate multiple starting configurations to explore solution space
    starts = []
    
    # 1. Hexagonal lattice pattern (denser packing)
    pts_hex = []
    rows = [5, 6, 6, 5, 4]
    dx, dy = 0.16, 0.138
    for r_idx, cnt in enumerate(rows):
        for c_idx in range(cnt):
            x = 0.15 + c_idx * dx + (r_idx % 2) * (dx / 2)
            y = 0.15 + r_idx * dy
            pts_hex.append([x, y])
    starts.append(np.array(pts_hex))
    
    # 2. Grid pattern
    pts_grid = []
    k = 5
    step = 0.7 / k
    for i in range(k):
        for j in range(k):
            if len(pts_grid) < n:
                pts_grid.append([0.15 + i * step, 0.15 + j * step])
    starts.append(np.array(pts_grid))
    
    # 3. Random centered layout
    np.random.seed(42)
    starts.append(np.random.uniform(0.2, 0.8, (n, 2)))
    
    # Optimize from each start
    for pts in starts:
        v0 = np.zeros(3*n)
        v0[:2*n] = pts.flatten()
        v0[2*n:] = 0.05  # Initial feasible radius
        
        res = minimize(objective, v0, method='SLSQP', bounds=bounds, 
                       constraints=cons, options={'maxiter': 2000, 'ftol': 1e-12})
                       
        sum_r = -res.fun
        if sum_r > best_sum_r:
            best_sum_r = sum_r
            best_v = res.x.copy()
            
    centers = best_v[:2*n].reshape(n, 2)
    radii = best_v[2*n:]
    
    return centers, radii, best_sum_r
