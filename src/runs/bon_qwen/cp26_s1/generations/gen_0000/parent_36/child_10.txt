To maximize the sum of radii for 26 circles in a unit square, we decouple the problem into two parts:
1. **Radius Optimization (Convex)**: For any fixed set of centers, finding the optimal radii is a Linear Programming problem. We maximize $\sum r_i$ subject to $r_i + r_j \leq \|c_i - c_j\|_2$ (non-overlap) and $r_i \leq \text{dist}(c_i, \text{boundary})$. This guarantees mathematically optimal radii for any given layout.
2. **Center Optimization (Non-convex)**: We use `scipy.optimize.differential_evolution` to search the configuration space of the 52 center coordinates. The fitness function is the negative optimal sum of radii from the LP.
3. **Initialization**: We start with a structured grid ($5 \times 5$ plus one extra) which yields an initial sum of $\approx 2.6$. This provides a strong starting point for the global optimizer to push towards the target of $2.636$.
4. **Constraints Handling**: The LP inherently satisfies all overlap and boundary constraints. The optimizer clips centers to $[0,1]$. This approach is robust, avoids complex penalty tuning, and leverages modern linear solvers (`HiGHS`) for speed.

```python
import numpy as np
import scipy.optimize as opt

N_CIRCLES = 26

def get_max_sum_radii(centers):
    """Solve LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    # Objective: minimize -sum(r) => maximize sum(r)
    c_obj = np.ones(n) * -1.0
    
    n_pairs = n * (n - 1) // 2
    n_bound = 4 * n
    A_ub = np.zeros((n_pairs + n_bound, n))
    b_ub = np.zeros(n_pairs + n_bound)
    
    idx = 0
    # Pairwise non-overlap constraints: r_i + r_j <= dist_ij
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dist
            idx += 1
            
        # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
        x, y = centers[i]
        A_ub[idx, i] = 1.0; b_ub[idx] = x; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - x; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = y; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1.0 - y; idx += 1
        
    bounds = [(0, None) for _ in range(n)]
    # Use HiGHS solver for speed and reliability
    res = opt.linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs', options={'disp': False})
    if res.success:
        return -res.fun, res.x
    return 0.0, np.zeros(n)

def objective(x_flat):
    """Objective function for differential evolution."""
    c = np.clip(x_flat.reshape((N_CIRCLES, 2)), 0.0, 1.0)
    s, _ = get_max_sum_radii(c)
    return -s

def run_packing():
    n = N_CIRCLES
    # Initial guess: 5x5 grid + 1 extra circle
    # Provides a valid starting sum ~ 2.6
    centers = []
    for i in range(5):
        for j in range(5):
            centers.append([0.1 + 0.2*j, 0.1 + 0.2*i])
    centers.append([0.5, 0.05])
    centers = np.array(centers[:n])
    
    bounds = [(0.0, 1.0) for _ in range(2 * n)]
    
    # Differential Evolution optimizes center positions
    result = opt.differential_evolution(
        objective, bounds, 
        popsize=15, maxiter=1000, tol=1e-7, seed=42, 
        polish=True, workers=1
    )
    
    best_centers = np.clip(result.x.reshape((n, 2)), 0.0, 1.0)
    best_sum, best_radii = get_max_sum_radii(best_centers)
    
    return best_centers, best_radii, best_sum
```
