import numpy as np
import scipy.optimize as opt
import math

def solve_radii_for_centers(centers):
    """
    Given fixed centers, solve the LP to maximize sum of radii.
    Constraints:
    1. r_i >= 0
    2. r_i <= distance to boundary (x, 1-x, y, 1-y)
    3. r_i + r_j <= distance(c_i, c_j)
    """
    n = len(centers)
    # Variables: r_0, r_1, ..., r_{n-1}
    # Objective: Maximize sum(r_i) -> Minimize -sum(r_i)
    c_obj = -np.ones(n)

    # Inequality constraints: A_ub @ x <= b_ub
    # We need to construct matrices for:
    # 1. -r_i <= 0  (r_i >= 0) -> handled by bounds usually, but linprog standard form is <=
    #    Actually bounds argument in linprog handles lower bounds.
    # 2. r_i <= bound_i  => r_i <= b_i
    # 3. r_i + r_j <= d_ij

    # Bounds for r_i: [0, max_possible_radius]
    # Max possible radius for circle i is distance to nearest wall
    max_radii = np.array([
        min(c[0], 1 - c[0], c[1], 1 - c[1]) for c in centers
    ])
    
    # Lower bounds 0, upper bounds max_radii
    bounds = [(0, r) for r in max_radii]

    # Constraints r_i + r_j <= d_ij
    # We only need to consider pairs that are close, but for safety and correctness
    # we can include all pairs. However, 26 circles -> 325 pairs.
    # Matrix size 325 x 26 is small.
    
    rows = []
    vals = []
    dists = []
    
    # Precompute distances
    dists = np.zeros((n, n))
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            dists[i, j] = dist
            dists[j, i] = dist

    # Build A_ub and b_ub
    # Constraint: r_i + r_j <= dist[i, j]
    # Row for pair (i, j): [0... 1 ... 1 ... 0] <= dist[i, j]
    
    # Since we have bounds for r_i <= max_radii, we don't strictly need to add 
    # r_i <= max_radii to A_ub, linprog handles bounds.
    # But we must add r_i + r_j <= dist constraints.
    
    A_ub = []
    b_ub = []
    
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)

    # Solve
    res = opt.linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return res.x
    else:
        # Fallback: small radii
        return np.ones(n) * 1e-6
