import numpy as np
from scipy.optimize import minimize

def objective_fun(v, n):
    """Objective: maximize sum of radii (minimize negative sum)."""
    return -np.sum(v[2*n:])

def constraint_fun(v, n, triu_idx):
    """
    Returns inequality constraints (must be >= 0):
    - 4*n boundary constraints
    - n*(n-1)/2 non-overlap constraints
    """
    c = v[:2*n].reshape(n, 2)
    r = v[2*n:]
    
    # Boundary constraints
    b_con = np.concatenate([
        c[:, 0] - r,
        1.0 - (c[:, 0] + r),
        c[:, 1] - r,
        1.0 - (c[:, 1] + r)
    ])
    
    # Overlap constraints (squared distance >= squared radius sum)
    diff = c[triu_idx[0]] - c[triu_idx[1]]
    dist2 = np.sum(diff**2, axis=1)
    rad_sum = r[triu_idx[0]] + r[triu_idx[1]]
    o_con = dist2 - rad_sum**2
    
    return np.concatenate([b_con, o_con])

def run_packing():
    n = 26
    triu_idx = np.triu_indices(n, k=1)
    best_sum = -np.inf
    best_sol = None
    
    # Bounds: centers in [0,1], radii in [1e-5, 0.5]
    bounds = [(0.0, 1.0) for _ in range(2*n)] + [(1e-5, 0.5) for _ in range(n)]
    
    # Multiple restarts to find global optimum
    for trial in range(8):
        if trial == 0:
            # Structured hexagonal-like grid initialization
            pts = np.array([( (i%6)*0.16 + 0.12, (i//6)*0.16 + 0.12) for i in range(n)])
        else:
            # Random initialization within safe region
            pts = np.random.uniform(0.15, 0.85, size=(n, 2))
            
        # Initial radii with slight perturbation to break symmetry
        r_init = np.full(n, 0.06) + np.random.uniform(-0.005, 0.005, n)
        r_init = np.clip(r_init, 1e-5, 0.4)
        x0 = np.hstack([pts.flatten(), r_init])
        
        try:
            res = minimize(objective_fun, x0, args=(n,), method='SLSQP',
                           bounds=bounds,
                           constraints={'type': 'ineq', 'fun': constraint_fun, 'args': (n, triu_idx)},
                           options={'maxiter': 5000, 'ftol': 1e-10, 'disp': False})
            if -res.fun > best_sum:
                best_sum = -res.fun
                best_sol = res.x
        except Exception:
            pass
            
    # Fallback if optimization fails
    if best_sol is None:
        best_sol = np.hstack([np.random.uniform(0.2, 0.8, 2*n), np.full(n, 0.05)])
        
    centers = best_sol[:2*n].reshape(n, 2)
    radii = best_sol[2*n:]
    
    # Post-processing to guarantee strict validity
    radii = np.maximum(radii, 1e-6)
    centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
    centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)
    
    return centers, radii, np.sum(radii)
