def get_overlap_penalty(centers, radii):

    Calculates the sum of squared overlaps for all pairs and boundaries.
    """
    n = centers.shape[0]
    penalty = 0.0
    
    # Pairwise overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            min_dist = radii[i] + radii[j]
            if dist < min_dist:
                overlap = min_dist - dist
                penalty += overlap ** 2
                
    # Boundary overlaps
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        
        # Check boundaries x=0, x=1, y=0, y=1
        if x < r:
            penalty += (r - x) ** 2
        if x > 1 - r:
            penalty += (x - (1 - r)) ** 2
        if y < r:
            penalty += (r - y) ** 2
        if y > 1 - r:
            penalty += ((1 - y) - r) ** 2
            
    return penalty
