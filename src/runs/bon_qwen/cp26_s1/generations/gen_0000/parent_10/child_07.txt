import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    """
    n = 26
    
    # Optimization Objective: Minimize negative sum of radii
    def objective(vars):
        # vars layout: x0, y0, r0, x1, y1, r1, ...
        radii = vars[2::3]
        return -np.sum(radii)

    # Constraints for SLSQP (must return >= 0)
    def get_constraints_and_jac():
        constraints = []

        # 1. Boundary Constraints
        # For each circle, ensure it is within [0, 1]
        # x - r >= 0  => r - x <= 0 (Wait, SLSQP expects g(x) >= 0 or g(x) <= 0? 
        # SLSQP expects 'type': 'ineq' meaning fun(x) >= 0)
        
        for i in range(n):
            idx_x = 3 * i
            idx_y = 3 * i + 1
            idx_r = 3 * i + 2
            
            # x - r >= 0
            def c_lower_x(v, idx_x=idx_x, idx_r=idx_r):
                return v[idx_x] - v[idx_r]
            constraints.append({'type': 'ineq', 'fun': c_lower_x})
            
            # 1 - x - r >= 0
            def c_upper_x(v, idx_x=idx_x, idx_r=idx_r):
                return 1.0 - v[idx_x] - v[idx_r]
            constraints.append({'type': 'ineq', 'fun': c_upper_x})
            
            # y - r >= 0
            def c_lower_y(v, idx_y=idx_y, idx_r=idx_r):
                return v[idx_y] - v[idx_r]
            constraints.append({'type': 'ineq', 'fun': c_lower_y})
            
            # 1 - y - r >= 0
            def c_upper_y(v, idx_y=idx_y, idx_r=idx_r):
                return 1.0 - v[idx_y] - v[idx_r]
            constraints.append({'type': 'ineq', 'fun': c_upper_y})

            # r >= 0
            def c_r_pos(v, idx_r=idx_r):
                return v[idx_r]
            constraints.append({'type': 'ineq', 'fun': c_r_pos})

        # 2. Non-overlap Constraints
        # dist^2 - (r_i + r_j)^2 >= 0
        for i in range(n):
            idx_xi = 3 * i
            idx_yi = 3 * i + 1
            idx_ri = 3 * i + 2
            
            for j in range(i + 1, n):
                idx_xj = 3 * j
                idx_yj = 3 * j + 1
                idx_rj = 3 * j + 2
                
                def c_overlap(v, xi=idx_xi, yi=idx_yi, ri=idx_ri, 
                                      xj=idx_xj, yj=idx_yj, rj=idx_rj):
                    dx = v[xi] - v[xj]
                    dy = v[yi] - v[yj]
                    dist_sq = dx*dx + dy*dy
                    r_sum = v[ri] + v[rj]
                    return dist_sq - r_sum**2
                constraints.append({'type': 'ineq', 'fun': c_overlap})
        
        return constraints

    constraints = get_constraints_and_jac()

        r_init = 0.08 # Start with a valid small radius

        # Row 2: 5 circles

            {'count': 5, 'offset_x': 0.0},
            {'count': 4, 'offset_x': 0.1}, # Shifted
            {'count': 5, 'offset_x': 0.0},
            {'count': 4, 'offset_x': 0.1},
            {'count': 5, 'offset_x': 0.0},
            {'count': 3, 'offset_x': 0.2}, # Centered roughly

            y = y_coords[row_idx]

                xs = [0.5]

                centers.append([x, y, r_init])

        return np.array(centers).flatten()

        return np.array(centers).flatten()

        centers = []
        r_init = 0.03

        ("Grid", get_grid_init()),
        ("Random", get_random_init()),
        ("Random", get_random_init()),

        centers[i, 0] = vars[3 * i]
        centers[i, 1] = vars[3 * i + 1]
        radii[i] = vars[3 * i + 2]

    # Ensure non-negative radii and valid bounds (clamping for safety)
    radii = np.maximum(radii, 0)
    # Clamp centers to be valid given radii
    for i in range(n):
        r = radii[i]
        centers[i, 0] = np.clip(centers[i, 0], r, 1 - r)
        centers[i, 1] = np.clip(centers[i, 1], r, 1 - r)

    return centers, radii, sum(radii)
