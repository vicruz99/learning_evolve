import numpy as np
from scipy.optimize import minimize

def compute_penalty_and_objective(vars, alpha, triu_idx):
    """
    Computes the objective function: -sum(radii) + penalty for constraint violations.
    Vectorized for performance.
    """
    N = 26
    centers = vars[:N*2].reshape(N, 2)
    radii = vars[N*2:]
    
    # Pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    # Sum of radii for pairs
    r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    
    # Overlap penalty (upper triangle only)
    overlap = np.maximum(0, r_sum[triu_idx] - dists[triu_idx])
    penalty = alpha * np.sum(overlap**2)
    
    # Boundary penalties
    # x >= r  => r - x <= 0
    penalty += alpha * np.sum(np.maximum(0, radii - centers[:, 0])**2)
    # x <= 1-r => x + r - 1 <= 0
    penalty += alpha * np.sum(np.maximum(0, centers[:, 0] + radii - 1.0)**2)
    # y >= r
    penalty += alpha * np.sum(np.maximum(0, radii - centers[:, 1])**2)
    # y <= 1-r
    penalty += alpha * np.sum(np.maximum(0, centers[:, 1] + radii - 1.0)**2)
    
    # Objective: maximize sum of radii => minimize negative sum
    return -np.sum(radii) + penalty

def run_packing() -> tuple:
    N = 26
    triu_idx = np.triu_indices(N, k=1)
    bounds = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N
    n_restarts = 60
    best_sum_r = -np.inf
    best_vars = None
    alpha = 2000.0
    
    for restart in range(n_restarts):
        np.random.seed(restart * 1337 + 42)
        
        # Initial configuration: scattered with moderate radii
        init_centers = 0.15 + 0.7 * np.random.rand(N, 2)
        init_r = 0.07 + 0.05 * np.random.rand(N)
        x0 = np.concatenate([init_centers.flatten(), init_r])
        
        res = minimize(compute_penalty_and_objective, x0, args=(alpha, triu_idx),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': 4000, 'ftol': 1e-12, 'gtol': 1e-12})
        
        # Extract results
        c_opt = res.x[:52].reshape(N, 2)
        r_opt = res.x[52:]
        
        # Compute exact minimum slack to ensure strict validity
        diff = c_opt[:, None, :] - c_opt[None, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        r_sum = r_opt[:, None] + r_opt[None, :]
        
        pair_slack = dists[triu_idx] - r_sum[triu_idx]
        b_slack = np.concatenate([c_opt[:,0]-r_opt, 1-c_opt[:,0]-r_opt,
                                  c_opt[:,1]-r_opt, 1-c_opt[:,1]-r_opt])
        min_slack = min(np.min(pair_slack), np.min(b_slack))
        
        # Fix any violations by uniformly shrinking radii to guarantee validity
        if min_slack < -1e-9:
            reduction = -min_slack + 1e-6
            r_opt = np.maximum(0, r_opt - reduction)
            
        current_sum = np.sum(r_opt)
        if current_sum > best_sum_r:
            best_sum_r = current_sum
            best_vars = np.concatenate([c_opt.flatten(), r_opt]).copy()
            
    final_centers = best_vars[:52].reshape(N, 2)
    final_radii = best_vars[52:]
    
    return final_centers, final_radii, float(best_sum_r)
