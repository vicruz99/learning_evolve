import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def objective(x):
    """Objective: minimize negative sum of radii (equivalent to maximizing sum of radii)"""
    return -np.sum(x[2::3])

def constraint_fun(x):
    """Returns array of inequality constraints >= 0"""
    n = N_CIRCLES
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    # Boundary and non-negative radius constraints
    c = np.concatenate([
        cx - r,               # x - r >= 0
        1.0 - (cx + r),       # x + r <= 1
        cy - r,               # y - r >= 0
        1.0 - (cy + r),       # y + r <= 1
        r                     # r >= 0
    ])
    
    # Non-overlap constraints: dist^2 >= (r_i + r_j)^2
    i, j = np.triu_indices(n, k=1)
    dx = cx[i] - cx[j]
    dy = cy[i] - cy[j]
    dist_sq = dx*dx + dy*dy
    r_sum = r[i] + r[j]
    
    c_overlap = dist_sq - r_sum*r_sum
    return np.concatenate([c, c_overlap])

def get_initial_guess(perturb_scale=0.0, seed=42):
    """Generates a valid hexagonal lattice packing, optionally perturbed"""
    np.random.seed(seed)
    r0 = 0.095
    pts = []
    y = r0
    row = 0
    while y + r0 <= 1.0:
        x_start = r0 if row % 2 == 0 else 2*r0
        x = x_start
        while x + r0 <= 1.0:
            pts.append([x, y])
            x += 2*r0
        y += np.sqrt(3) * r0
        row += 1
        
    pts = np.array(pts[:N_CIRCLES])
    r_init = np.full(N_CIRCLES, 0.085)
    
    if perturb_scale > 0:
        pts += np.random.uniform(-perturb_scale, perturb_scale, pts.shape)
        pts = np.clip(pts, 0.01, 0.99)
        
    x0 = np.zeros(3 * N_CIRCLES)
    x0[0::3] = pts[:, 0]
    x0[1::3] = pts[:, 1]
    x0[2::3] = r_init
    return x0

def run_packing():
    # Define variable bounds: cx, cy in [0, 1], r in [0, 0.5]
    bounds = []
    for _ in range(N_CIRCLES):
        bounds.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
        
    cons = {'type': 'ineq', 'fun': constraint_fun}
    
    best_sum_r = 0.0
    best_centers = None
    best_radii = None
    
    # Multi-start optimization to escape local optima
    seeds = [42, 123, 456, 789]
    scales = [0.0, 0.015, 0.03]
    
    for seed in seeds:
        for scale in scales:
            x0 = get_initial_guess(scale, seed)
            try:
                res = minimize(objective, x0, method='SLSQP', constraints=cons,
                               bounds=bounds, options={'maxiter': 2500, 'ftol': 1e-10})
                curr_r = res.x[2::3]
                curr_sum = np.sum(curr_r)
                if curr_sum > best_sum_r:
                    best_sum_r = curr_sum
                    best_centers = np.column_stack((res.x[0::3], res.x[1::3]))
                    best_radii = curr_r
            except Exception:
                continue
                
    # Fallback if optimization fails completely
    if best_centers is None:
        x0 = get_initial_guess(0.0, 42)
        best_centers = np.column_stack((x0[0::3], x0[1::3]))
        best_radii = x0[2::3]
        best_sum_r = np.sum(best_radii)
        
    return best_centers, best_radii, float(best_sum_r)
