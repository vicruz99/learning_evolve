from scipy.optimize import minimize

def compute_violation(centers, radii):

    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    r_sum = radii[:, None] + radii[None, :]
    
    overlap = np.maximum(0, r_sum - dists)
    np.fill_diagonal(overlap, 0)
    max_overlap = np.max(overlap)
    
    x, y = centers[:, 0], centers[:, 1]
    max_bound = max(np.max(np.maximum(0, radii - x)),
                    np.max(np.maximum(0, x - (1 - radii))),
                    np.max(np.maximum(0, radii - y)),
                    np.max(np.maximum(0, y - (1 - radii))),
                    np.max(np.maximum(0, -radii)))
    return max(max_overlap, max_bound)

def penalty_obj(x, mu):
