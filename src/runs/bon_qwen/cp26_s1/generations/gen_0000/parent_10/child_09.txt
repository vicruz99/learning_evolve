import numpy as np
from scipy.optimize import minimize, NonlinearConstraint

def _constraint_fun(vars):
    """Compute all pairwise and boundary constraints."""
    n = 26
    c = vars[:2*n].reshape(n, 2)
    r = vars[2*n:]
    
    # Pairwise non-overlap: dist_sq >= (r_i + r_j)^2
    diffs = c[:, np.newaxis, :] - c[np.newaxis, :, :]
    dist_sq = np.sum(diffs**2, axis=2)
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    cons_pair = dist_sq[mask] - r_sum[mask]**2
    
    # Boundary containment: c - r >= 0 and 1 - c - r >= 0
    cons_bound = np.concatenate([
        c[:, 0] - r,
        1.0 - c[:, 0] - r,
        c[:, 1] - r,
        1.0 - c[:, 1] - r
    ])
    
    return np.concatenate([cons_pair, cons_bound])

def _objective(vars):
    """Maximize sum of radii by minimizing negative sum."""
    return -np.sum(vars[52:])

def _generate_hex_centers(n):
    """Generate initial centers in a perturbed hexagonal pattern."""
    centers = []
    r_approx = 0.1
    dx = 2.0 * r_approx
    dy = np.sqrt(3.0) * r_approx
    
    row = 0
    col = 0
    while len(centers) < n:
        y = 0.05 + row * dy
        if y > 0.95:
            row += 1
            col = 0
            continue
            
        x = 0.05 + col * dx + (0.5 * dx if row % 2 else 0.0)
        if x <= 0.95:
            centers.append([x, y])
            col += 1
        else:
            row += 1
            col = 0
            
    centers = np.array(centers[:n])
    if len(centers) < n:
        pad = np.random.uniform(0.1, 0.9, (n - len(centers), 2))
        centers = np.vstack([centers, pad])
    return centers

def run_packing():
    n = 26
    np.random.seed(42)
    
    # 1. Initialize with hexagonal packing pattern + small jitter
    centers_init = _generate_hex_centers(n)
    centers_init += np.random.uniform(-0.01, 0.01, centers_init.shape)
    centers_init = np.clip(centers_init, 0.1, 0.9)
    
    radii_init = np.full(n, 0.06)
    x0 = np.concatenate([centers_init.flatten(), radii_init])
    
    # 2. Define bounds and constraints
    bounds = [(0.0, 1.0) for _ in range(2*n)] + [(0.0, 0.5) for _ in range(n)]
    nl_con = NonlinearConstraint(_constraint_fun, 0, np.inf)
    
    # 3. Optimize using trust-constr method
    res = minimize(_objective, x0, method='trust-constr', bounds=bounds, 
                   constraints=nl_con, options={'maxiter': 2500})
                   
    c_final = res.x[:2*n].reshape(n, 2)
    r_final = res.x[2*n:]
    
    # 4. Post-process to guarantee strict validity
    r_final = np.maximum(r_final, 1e-7)
    for i in range(n):
        c_final[i, 0] = np.clip(c_final[i, 0], r_final[i], 1.0 - r_final[i])
        c_final[i, 1] = np.clip(c_final[i, 1], r_final[i], 1.0 - r_final[i])
        
    return c_final, r_final, float(np.sum(r_final))
