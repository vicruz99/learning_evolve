To maximize the sum of radii for 26 circles in a unit square, we treat this as a constrained non-convex optimization problem. The objective is to maximize $\sum r_i$, which is equivalent to minimizing $-\sum r_i$. Constraints include boundary conditions ($r_i \le x_i \le 1-r_i$, $r_i \le y_i \le 1-r_i$) and non-overlap ($\text{dist}_{ij} \ge r_i + r_j$).

Strategy:
1. **Penalty Method**: Transform constraints into penalty terms added to the objective function. This allows the use of unconstrained gradient-based optimizers (`L-BFGS-B`) which handle the smooth penalty landscape efficiently.
2. **Annealing Schedule**: Start with a small penalty multiplier `mu` to allow the optimizer to explore the configuration space freely, then gradually increase `mu` (doubling each step) to strictly enforce constraints as the solution converges.
3. **Multiple Initializations**: Circle packing has many local minima. We initialize from different structured layouts (hexagonal lattice, grid) and random seeds, running the optimization schedule on each, and keep the best result.
4. **Vectorization**: Compute pairwise distances and penalties using NumPy matrix operations for speed.
5. **Final Projection**: After optimization, strictly clamp radii to satisfy boundary constraints exactly, ensuring validity per the checker's tolerance.

import numpy as np
from scipy.optimize import minimize

def get_loss(vars, n, mu):
    """Compute objective + penalty for constraints"""
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    # Objective: maximize sum of radii -> minimize -sum(r)
    loss = -np.sum(r)
    
    # Boundary penalties: r <= x, r <= 1-x, r <= y, r <= 1-y, r >= 0
    loss += mu * np.sum(np.maximum(0, r - x)**2)
    loss += mu * np.sum(np.maximum(0, r - (1 - x))**2)
    loss += mu * np.sum(np.maximum(0, r - y)**2)
    loss += mu * np.sum(np.maximum(0, r - (1 - y))**2)
    loss += mu * np.sum(np.maximum(0, -r)**2)
    
    # Overlap penalties: dist(i,j) >= r_i + r_j
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist = np.sqrt(dx**2 + dy**2 + 1e-12)
    r_sum = r[:, None] + r[None, :]
    overlap = r_sum - dist
    
    # Only consider upper triangle to avoid double counting
    tri = np.triu_indices(n, k=1)
    loss += mu * np.sum(np.maximum(0, overlap[tri])**2)
    
    return loss

def initialize_hex(n=26):
    """Hexagonal lattice initialization"""
    rows = [5, 5, 6, 5, 5]
    pos = []
    r0 = 0.05
    dx = 2.1 * r0
    dy = 1.8 * r0
    y = 0.15
    for i, count in enumerate(rows):
        x = 0.15 + (i % 2) * dx / 2
        for _ in range(count):
            pos.append([x, y, r0])
            x += dx
        y += dy
    return np.array(pos).flatten()

def initialize_grid(n=26):
    """Uniform grid initialization"""
    pos = []
    r0 = 0.04
    cols = 6
    rows = (n + cols - 1) // cols
    dx = 1.0 / (cols + 1)
    dy = 1.0 / (rows + 1)
    idx = 0
    for i in range(rows):
        for j in range(cols):
            if idx < n:
                pos.append([dx * (j + 1), dy * (i + 1), r0])
                idx += 1
    return np.array(pos).flatten()

def initialize_random(n=26, seed=42):
    """Random initialization"""
    rng = np.random.RandomState(seed)
    x = rng.rand(n) * 0.6 + 0.2
    y = rng.rand(n) * 0.6 + 0.2
    r = np.full(n, 0.03)
    return np.column_stack((x, y, r)).flatten()

def optimize_from(init_vars, n=26):
    """Run penalty annealing optimization from initial configuration"""
    vars = init_vars.copy()
    mu = 10.0
    # Gradually increase penalty to enforce constraints
    for _ in range(18):
        res = minimize(get_loss, vars, args=(n, mu), method='L-BFGS-B', 
                       options={'maxiter': 3000, 'ftol': 1e-12})
        vars = res.x
        mu *= 2.0
    return vars

def run_packing():
    n = 26
    best_vars = None
    best_loss = np.inf
    
    # Try multiple starting configurations to escape local minima
    inits = [
        initialize_hex(n),
        initialize_grid(n),
        initialize_random(n, seed=1),
        initialize_random(n, seed=2),
        initialize_random(n, seed=3)
    ]
    
    for init in inits:
        vars = optimize_from(init, n)
        # Evaluate with high mu to verify constraint satisfaction
        loss = get_loss(vars, n, 100000.0)
        if loss < best_loss:
            best_loss = loss
            best_vars = vars.copy()
            
    # Extract centers and radii
    centers = np.column_stack((best_vars[0::3], best_vars[1::3]))
    radii = best_vars[2::3]
    
    # Ensure non-negative radii
    radii = np.maximum(radii, 0.0)
    
    # Strict projection to guarantee boundary constraints are met
    for i in range(n):
        max_r = min(centers[i,0], 1-centers[i,0], centers[i,1], 1-centers[i,1], radii[i])
        if max_r < radii[i]:
            radii[i] = max_r
            
    return centers, radii, np.sum(radii)
