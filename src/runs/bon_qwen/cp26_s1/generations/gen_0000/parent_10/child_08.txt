import numpy as np
from scipy.optimize import minimize

def compute_violations(vars, n):
    """Compute sum of squared constraint violations for boundary and pairwise overlaps."""
    cx = vars[0::3]
    cy = vars[1::3]
    r = vars[2::3]
    
    # Boundary violations: circle must stay inside [0,1]x[0,1]
    v = np.maximum(0.0, r - cx)**2
    v += np.maximum(0.0, r - (1.0 - cx))**2
    v += np.maximum(0.0, r - cy)**2
    v += np.maximum(0.0, r - (1.0 - cy))**2
    viol = np.sum(v)
    
    # Pairwise overlap violations
    # Vectorized distance calculation
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dist = np.hypot(dx, dy)
    sum_r = r[:, None] + r[None, :]
    overlap = np.maximum(0.0, sum_r - dist)
    
    # Only consider each pair once (upper triangle)
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    viol += np.sum(overlap[mask]**2)
    
    return viol

def obj_with_penalty(vars, mu, n):
    """Objective: minimize negative sum of radii + penalty for constraint violations."""
    return -np.sum(vars[2::3]) + mu * compute_violations(vars, n)

def run_packing():
    n = 26
    best_sum = -1.0
    best_centers = None
    best_radii = None
    rng = np.random.RandomState(42)
    
    # Try multiple initial configurations to avoid poor local minima
    for seed in range(8):
        cx_init = rng.uniform(0.2, 0.8, n)
        cy_init = rng.uniform(0.2, 0.8, n)
        r_init = np.full(n, 0.09)
        
        vars = np.zeros(3 * n)
        vars[0::3] = cx_init
        vars[1::3] = cy_init
        vars[2::3] = r_init
        
        bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n
        
        current_vars = vars.copy()
        # Annealing schedule: slowly increase penalty weight
        for mu in [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 50.0, 100.0, 200.0]:
            res = minimize(obj_with_penalty, current_vars, args=(mu, n), 
                           method='L-BFGS-B', bounds=bounds,
                           options={'ftol': 1e-10, 'gtol': 1e-8, 'maxiter': 2000})
            current_vars = res.x
            
            # Inject small noise at higher mu to escape local minima
            if mu > 10.0:
                noise = rng.uniform(-0.005, 0.005, 3*n)
                current_vars = current_vars + noise
                
        # Check feasibility of the final configuration
        viol = compute_violations(current_vars, n)
        if viol < 1e-4:
            s = -np.sum(current_vars[2::3])
            if s > best_sum:
                best_sum = s
                best_centers = current_vars.reshape(-1, 3)[:, :2].copy()
                best_radii = current_vars.reshape(-1, 3)[:, 2].copy()
                
    return best_centers, best_radii, best_sum
