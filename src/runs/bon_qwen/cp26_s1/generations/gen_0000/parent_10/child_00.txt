This is a 26-circle packing problem in a unit square. I need to maximize the sum of radii.

**Key insights:**
1. For equal circles, a hexagonal packing is most efficient (density π/(2√3) ≈ 0.9069)
2. For 26 equal circles, theoretical max sum ≈ 2.73, but boundary effects reduce this
3. With unequal circles, we can sometimes do better by having larger circles in corners/edges

**Strategy:**
1. Use alternating optimization: fix positions → solve LP for radii → optimize positions → repeat
2. The LP for radii given fixed centers is exact and efficient
3. Position optimization uses L-BFGS-B with a penalty-based objective
4. Try multiple initial configurations (hexagonal, grid, random) with different seeds
5. Add a refinement step with direct SLSQP optimization on all variables together
6. Use multiple restarts to avoid local optima

**Implementation details:**
- All helper functions are top-level (no closures, no lambdas)
- LP solver (scipy.linprog) for exact radius computation given positions
- L-BFGS-B for position optimization with smooth penalty objective
- SLSQP for final joint optimization of positions and radii
- Multiple random seeds and patterns for robustness

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.spatial.distance import cdist


def solve_radii_lp(centers, n):
    """Solve LP to find optimal radii given fixed centers.
    
    Maximize sum(r_i)
    Subject to:
        r_i + r_j <= ||c_i - c_j|| for all i < j
        0 <= r_i <= min(c_ix, c_iy, 1-c_ix, 1-c_iy)
    """
    # Compute pairwise distances
    d_ij = cdist(centers, centers, 'euclidean')
    
    # Compute max radius for each circle from boundary
    c_i = np.zeros(n)
    for i in range(n):
        c_i[i] = min(centers[i, 0], centers[i, 1], 1 - centers[i, 0], 1 - centers[i, 1])
        c_i[i] = max(c_i[i], 1e-12)
    
    # LP: min -sum(r_i)
    c_obj = -np.ones(n)
    
    # Constraints: r_i + r_j <= d_ij for i < j
    A_ub = []
    b_ub = []
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(d_ij[i, j])
    
    if len(A_ub) > 0:
        A_ub = np.array(A_ub)
        b_ub = np.array(b_ub)
    else:
        A_ub = np.zeros((0, n))
        b_ub = np.zeros(0)
    
    bounds = [(0.0, max(c_i[i], 1e-12)) for i in range(n)]
    
    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success and not np.any(np.isnan(result.x)):
            return result.x
    except Exception:
        pass
    
    return np.zeros(n)


def center_penalty(vars_flat, radii, n):
    """Compute penalty for constraint violations."""
    centers = vars_flat.reshape(n, 2)
    penalty = 0.0
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        # Boundary violations (squared)
        if x < r:
            penalty += (r - x) ** 2
        if x > 1 - r:
            penalty += (x + r - 1) ** 2
        if y < r:
            penalty += (r - y) ** 2
        if y > 1 - r:
            penalty += (y + r - 1) ** 2
        # Overlap violations
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist_sq = dx * dx + dy * dy
            dist = np.sqrt(dist_sq) if dist_sq > 1e-20 else 1e-10
            min_dist = radii[i] + radii[j]
            if dist < min_dist:
                penalty += (min_dist - dist) ** 2
    return penalty


def center_gradient(vars_flat, radii, n):
    """Compute gradient of penalty."""
    centers = vars_flat.reshape(n, 2)
    grad = np.zeros_like(vars_flat)
    
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        
        # Boundary gradients
        if x < r:
            grad[2 * i] += 2 * (x - r)
        if x > 1 - r:
            grad[2 * i] += 2 * (x + r - 1)
        if y < r:
            grad[2 * i + 1] += 2 * (y - r)
        if y > 1 - r:
            grad[2 * i + 1] += 2 * (y + r - 1)
        
        # Overlap gradients
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist_sq = dx * dx + dy * dy
            dist = np.sqrt(dist_sq) if dist_sq > 1e-20 else 1e-10
            min_dist = radii[i] + radii[j]
            if dist < min_dist and dist > 1e-10:
                factor = 2 * (min_dist - dist) / dist
                grad[2 * i] -= factor * dx
                grad[2 * i + 1] -= factor * dy
                grad[2 * j] += factor * dx
                grad[2 * j + 1] += factor * dy
    
    return grad


def optimize_centers_lbfgsb(centers, radii, n, n_iter=200):
    """Optimize centers using L-BFGS-B."""
    def objective(vars_flat):
        return center_penalty(vars_flat, radii, n)
    
    def gradient(vars_flat):
        return center_gradient(vars_flat, radii, n)
    
    bounds_list = [(0.001, 0.999)] * (2 * n)
    
    result = minimize(objective, centers.flatten(), jac=gradient, method='L-BFGS-B',
                      bounds=bounds_list,
                      options={'maxiter': n_iter, 'ftol': 1e-15, 'gtol': 1e-12})
    
    return result.x.reshape(n, 2)


def make_hex_initial(n, spacing_x=None, spacing_y=None, offset_x=0, offset_y=0):
    """Create hexagonal initial configuration."""
    centers = np.zeros((n, 2))
    idx = 0
    
    if spacing_x is None:
        spacing_x = 0.22
    if spacing_y is None:
        spacing_y = spacing_x * np.sqrt(3) / 2
    
    for row in range(7):
        n_cols = 5 if row % 2 == 0 else 4
        for col in range(n_cols):
            if idx < n:
                x = offset_x + col * spacing_x + (spacing_x / 2 if row % 2 == 1 else 0)
                y = offset_y + row * spacing_y
                centers[idx] = [x, y]
                idx += 1
    
    return centers


def make_grid_initial(n, cols=5, rows=6):
    """Create grid initial configuration."""
    centers = np.zeros((n, 2))
    idx = 0
    dx = 0.8 / max(cols - 1, 1)
    dy = 0.8 / max(rows - 1, 1)
    
    for row in range(rows):
        for col in range(cols):
            if idx < n:
                x = 0.1 + col * dx
                y = 0.1 + row * dy
                centers[idx] = [x, y]
                idx += 1
    
    return centers


def make_corner_edge_initial(n):
    """Create configuration with larger circles in corners and edges."""
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.03
    
    # 4 corner circles
    centers[0] = [0.12, 0.12]
    centers[1] = [0.88, 0.12]
    centers[2] = [0.12, 0.88]
    centers[3] = [0.88, 0.88]
    
    # 8 edge circles
    edge_positions = [
        [0.37, 0.12], [0.63, 0.12],  # bottom
        [0.37, 0.88], [0.63, 0.88],  # top
        [0.12, 0.37], [0.12, 0.63],  # left
        [0.88, 0.37], [0.88, 0.63],  # right
    ]
    for i, pos in enumerate(edge_positions):
        centers[4 + i] = pos
    
    # 14 interior circles in grid
    idx = 12
    positions = [
        [0.37, 0.37], [0.63, 0.37],
        [0.37, 0.63], [0.63, 0.63],
        [0.5, 0.25], [0.5, 0.75],
        [0.25, 0.5], [0.75, 0.5],
        [0.37, 0.5], [0.63, 0.5],
        [0.5, 0.37], [0.5, 0.63],
        [0.3, 0.3], [0.7, 0.7],
    ]
    for i, pos in enumerate(positions):
        if idx + i < n:
            centers[idx + i] = pos
    
    return centers


def alternating_optimize(centers, n, n_iter=80, verbose=False):
    """Run alternating optimization: LP for radii, L-BFGS for centers."""
    radii = np.ones(n) * 0.02
    
    for iteration in range(n_iter):
        # Solve LP for radii
        radii = solve_radii_lp(centers, n)
        
        # Optimize centers
        centers = optimize_centers_lbfgsb(centers, radii, n, n_iter=100)
        
        # Solve LP again
        radii = solve_radii_lp(centers, n)
        
        if verbose and iteration % 20 == 0:
            print(f"  Alt opt iter {iteration}: sum={np.sum(radii):.6f}")
    
    return centers, radii


def direct_slscp(centers, radii, n, max_iter=2000):
    """Direct optimization using SLSQP on all variables."""
    vars0 = np.concatenate([centers.flatten(), radii])
    
    def objective(vars_flat):
        return -np.sum(vars_flat[2::3])
    
    # Build constraints
    constraints = []
    
    # Non-negative radii
    for i in range(n):
        constraints.append({
            'type': 'ineq',
            'fun': radius_nonneg,
            'args': (i,)
        })
    
    # Boundary constraints: x >= r, y >= r, 1-x >= r, 1-y >= r
    for i in range(n):
        for ctype in range(4):
            constraints.append({
                'type': 'ineq',
                'fun': boundary_constraint,
                'args': (i, ctype)
            })
    
    # Non-overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({
                'type': 'ineq',
                'fun': overlap_constraint,
                'args': (i, j)
            })
    
    # Bounds
    bounds = []
    for i in range(n):
        bounds.extend([(0.001, 0.999), (0.001, 0.999), (0.0, 0.5)])
    
    try:
        result = minimize(objective, vars0, method='SLSQP', bounds=bounds,
                          constraints=constraints,
                          options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
        
        if not np.any(np.isnan(result.x)):
            opt_centers = result.x[:2 * n].reshape(n, 2)
            opt_radii = result.x[2 * n:]
            return opt_centers, opt_radii
    except Exception:
        pass
    
    return centers, radii


def radius_nonneg(vars_flat, i):
    return vars_flat[3 * i + 2]


def boundary_constraint(vars_flat, i, ctype):
    x = vars_flat[3 * i]
    y = vars_flat[3 * i + 1]
    r = vars_flat[3 * i + 2]
    if ctype == 0:
        return x - r
    elif ctype == 1:
        return y - r
    elif ctype == 2:
        return 1 - x - r
    else:
        return 1 - y - r


def overlap_constraint(vars_flat, i, j):
    x1, y1, r1 = vars_flat[3 * i], vars_flat[3 * i + 1], vars_flat[3 * i + 2]
    x2, y2, r2 = vars_flat[3 * j], vars_flat[3 * j + 1], vars_flat[3 * j + 2]
    dx = x1 - x2
    dy = y1 - y2
    return dx * dx + dy * dy - (r1 + r2) ** 2


def simulated_annealing_centers(centers, n, n_iter=5000, temp_init=0.05):
    """Simulated annealing for center positions."""
    best_centers = centers.copy()
    best_radii = solve_radii_lp(centers, n)
    best_sum = np.sum(best_radii)
    
    current_centers = centers.copy()
    current_radii = best_radii.copy()
    current_sum = best_sum
    
    for iteration in range(n_iter):
        # Cooling schedule
        temp = temp_init * (0.999) ** iteration
        
        # Perturb a random circle
        i = np.random.randint(n)
        perturbation = np.random.randn(2) * temp
        new_centers = current_centers.copy()
        new_centers[i] += perturbation
        new_centers[i] = np.clip(new_centers[i], 0.01, 0.99)
        
        new_radii = solve_radii_lp(new_centers, n)
        new_sum = np.sum(new_radii)
        
        # Accept or reject
        delta = new_sum - current_sum
        if delta > 0 or np.random.rand() < np.exp(delta / max(temp, 1e-10)):
            current_centers = new_centers
            current_radii = new_radii
            current_sum = new_sum
        
        if current_sum > best_sum:
            best_centers = current_centers.copy()
            best_radii = current_radii.copy()
            best_sum = current_sum
    
    return best_centers, best_radii


def perturb_centers(centers, magnitude=0.02):
    """Add random perturbation to centers."""
    new_centers = centers.copy()
    new_centers += np.random.randn(*centers.shape) * magnitude
    new_centers = np.clip(new_centers, 0.02, 0.98)
    return new_centers

    c = make_corner_edge_initial(n)
    initial_configs.append(c)
    
    # Try each initial configuration
    for config_idx, centers in enumerate(initial_configs):
        # Run alternating optimization
        opt_centers, opt_radii = alternating_optimize(centers, n, n_iter=60)
        s = np.sum(opt_radii)
        
        if s > best_sum:
            best_sum = s

            best_radii = opt_radii.copy()
        
        # Try simulated annealing refinement
        sa_centers, sa_radii = simulated_annealing_centers(opt_centers, n, n_iter=2000, temp_init=0.03)

        if s > best_sum:
            best_sum = s

            best_radii = sa_radii.copy()
    
    # Refine best with direct SLSQP
    slscp_centers, slscp_radii = direct_slscp(best_centers, best_radii, n, max_iter=3000)
    s = np.sum(slscp_radii)
    if s > best_sum:
