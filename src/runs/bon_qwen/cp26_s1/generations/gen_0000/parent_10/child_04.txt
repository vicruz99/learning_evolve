To maximize the sum of radii for 26 circles in a unit square, we decompose the problem into two parts:
1. **Given centers, find optimal radii**: This is a Linear Programming (LP) problem. For fixed centers, the constraints $r_i + r_j \leq \|C_i - C_j\|$ and $r_i \leq \text{dist}(C_i, \text{boundary})$ are linear in $r$. We maximize $\sum r_i$ subject to these constraints.
2. **Optimize centers**: The LP objective value as a function of centers is non-smooth and non-convex. We use Simulated Annealing (SA) to explore the center configuration space, accepting moves that increase the LP sum of radii. This is followed by a local hill-climbing phase to polish the solution.

This approach efficiently handles the non-convex geometry by leveraging LP for exact radius computation and SA for robust global search.

import numpy as np
from scipy.optimize import linprog

def solve_radii(centers):
    """
    Solves the LP to find the maximum sum of radii for fixed centers.
    Returns (radii, sum_radii).
    """
    n = centers.shape[0]
    # Objective: maximize sum(r_i) => minimize -sum(r_i)
    c = -np.ones(n)
    
    # Compute pairwise distances
    diff = centers[:, None] - centers[None, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    # Indices for upper triangle (pairwise constraints)
    i, j = np.triu_indices(n, k=1)
    n_pairs = len(i)
    
    # Constraints matrix A_ub * x <= b_ub
    # 1. Pairwise: r_i + r_j <= d_ij
    # 2. Boundaries: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    A_ub = np.zeros((n_pairs + 4*n, n))
    b_ub = np.zeros(n_pairs + 4*n)
    
    # Fill pairwise constraints
    row_idx = np.arange(n_pairs)
    A_ub[row_idx, i] = 1
    A_ub[row_idx, j] = 1
    b_ub[:n_pairs] = dists[i, j]
    
    # Fill boundary constraints
    # r_i <= x_i
    A_ub[n_pairs : n_pairs+n, np.arange(n)] = 1
    b_ub[n_pairs : n_pairs+n] = centers[:, 0]
    # r_i <= 1-x_i
    A_ub[n_pairs+n : n_pairs+2*n, np.arange(n)] = 1
    b_ub[n_pairs+n : n_pairs+2*n] = 1 - centers[:, 0]
    # r_i <= y_i
    A_ub[n_pairs+2*n : n_pairs+3*n, np.arange(n)] = 1
    b_ub[n_pairs+2*n : n_pairs+3*n] = centers[:, 1]
    # r_i <= 1-y_i
    A_ub[n_pairs+3*n : n_pairs+4*n, np.arange(n)] = 1
    b_ub[n_pairs+3*n : n_pairs+4*n] = 1 - centers[:, 1]
    
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return res.x, -res.fun
    else:
        # Fallback for numerical issues
        return np.full(n, 1e-5), 0.0

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    N = 26
    np.random.seed(42)
    # Initialize centers randomly
    centers = np.random.rand(N, 2)
    
    # Simulated Annealing parameters
    T = 0.1
    decay = 0.999
    moves = 0
    max_moves = 12000
    
    best_centers = centers.copy()
    _, best_score = solve_radii(best_centers)
    current_score = best_score
    
    # SA Loop: Optimize centers to maximize sum of radii
    while T > 1e-6 and moves < max_moves:
        i = np.random.randint(N)
        step = max(0.005, T)
        new_centers = centers.copy()
        new_centers[i] += np.random.uniform(-step, step, 2)
        new_centers[i] = np.clip(new_centers[i], 0, 1)
        
        _, new_score = solve_radii(new_centers)
        
        delta = new_score - current_score
        if delta > 0 or np.random.rand() < np.exp(delta / T):
            centers = new_centers
            current_score = new_score
            if current_score > best_score:
                best_score = current_score
                best_centers = centers.copy()
        T *= decay
        moves += 1
        
    # Local hill climbing to polish the solution
    for _ in range(5000):
        i = np.random.randint(N)
        step = 0.001
        new_centers = best_centers.copy()
        new_centers[i] += np.random.uniform(-step, step, 2)
        new_centers[i] = np.clip(new_centers[i], 0, 1)
        
        _, new_score = solve_radii(new_centers)
        if new_score > best_score:
            best_score = new_score
            best_centers = new_centers
            
    best_radii, final_score = solve_radii(best_centers)
    return best_centers, best_radii, final_score
