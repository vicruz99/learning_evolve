import numpy as np
from scipy.optimize import minimize

def compute_cost(vars, n, penalty, mask):
    """
    Computes the objective function: negative sum of radii + penalty for constraint violations.
    """
    centers = vars[:2*n].reshape(n, 2)
    radii = vars[2*n:]
    
    # Objective: maximize sum of radii => minimize negative sum
    cost = -np.sum(radii)
    
    # Overlap penalty: (r_i + r_j)^2 >= dist^2
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist_sq = np.sum(diff**2, axis=2)
    
    r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    
    # Calculate violation for upper triangle pairs
    viol = np.zeros_like(dist_sq)
    viol[mask] = np.maximum(0.0, r_sum[mask]**2 - dist_sq[mask])
    cost += penalty * np.sum(viol**2)
    
    # Boundary penalty: circles must be within [0, 1]
    # x - r >= 0  =>  r - x <= 0
    b1 = np.maximum(0.0, radii - centers[:, 0])
    # 1 - x - r >= 0  =>  r + x - 1 <= 0
    b2 = np.maximum(0.0, radii + centers[:, 0] - 1.0)
    # y - r >= 0
    b3 = np.maximum(0.0, radii - centers[:, 1])
    # 1 - y - r >= 0
    b4 = np.maximum(0.0, radii + centers[:, 1] - 1.0)
    
    cost += penalty * (np.sum(b1**2) + np.sum(b2**2) + np.sum(b3**2) + np.sum(b4**2))
    
    return cost

def get_initial_guess(n):
    """
    Generates an initial hexagonal-like arrangement for n circles.
    """
    centers = []
    # Distribute 26 circles in staggered rows
    row_counts = [6, 5, 6, 5, 4]

    curr_sum = sum(row_counts)
    if curr_sum < n:
        i = 0
        while curr_sum < n:
            row_counts[i] += 1
            curr_sum += 1
            i = (i + 1) % len(row_counts)
    elif curr_sum > n:
        i = 0
        while curr_sum > n:
            row_counts[i] -= 1
            curr_sum -= 1
            i = (i + 1) % len(row_counts)

    y_step = (1 - 2 * y_margin) / (len(row_counts) + 1)
    
    idx = 0
    for i, count in enumerate(row_counts):
        y = y_margin + (i + 1) * y_step
        # Stagger odd rows
        shift = (i % 2) * 0.05
        x_start = 0.05 + shift
        x_end = 0.95 - shift
        
        for j in range(count):
            if idx >= n:
                break
            if count == 1:
                x = 0.5
            else:
                x_step = (x_end - x_start) / (count - 1)
                x = x_start + j * x_step
            centers.append([x, y])
            idx += 1
            
    return np.array(centers)

def run_packing():
    n = 26
    # Precompute mask for upper triangle to avoid reallocation in loop
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    
    # Bounds for variables: x, y in [0, 1], r in [1e-7, 0.5]
    bounds = [(0.0, 1.0)] * (2 * n) + [(1e-7, 0.5)] * n
    
    best_centers = None
    best_radii = None
    best_sum = -1.0
    
    # Multiple restarts to avoid local minima
    for seed in range(5):
        # Initialize
        centers0 = get_initial_guess(n)
        # Add small perturbation
        centers0 += np.random.randn(*centers0.shape) * 0.02
        centers0 = np.clip(centers0, 0.01, 0.99)
        
        radii0 = np.full(n, 0.09) # Start with reasonable radii
        vars0 = np.concatenate([centers0.flatten(), radii0])
        
        current_vars = vars0.copy()

        for penalty in [100.0, 1000.0, 5000.0]:

            current_vars = res.x
