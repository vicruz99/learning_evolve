To maximize the sum of radii for 26 circles in a unit square, we treat this as a non-linear optimization problem. Since equal circles often provide a good baseline for density, we initialize the centers using a perturbed hexagonal lattice, which is geometrically more efficient than a square grid. We then employ a dual optimization strategy:
1.  **Linear Programming (LP) for Radii:** For any fixed set of centers, the problem of finding optimal radii is a linear program (maximize $\sum r_i$ subject to non-overlap and boundary constraints). This is solved efficiently at each step.
2.  **Force-Directed Movement for Centers:** Based on the "tight" (active) constraints identified by the LP, we apply repulsive forces between touching circles and repulsive forces from the boundaries. If a circle is limited by a wall, its center is pushed away from that wall; if two circles are touching, they are pushed apart.
3.  **Iterative Relaxation:** We iterate between solving the LP and moving the centers, gradually increasing the sum of radii until convergence. This approach avoids the non-convexity of a direct optimization over all variables simultaneously.

import numpy as np
import scipy.optimize as opt

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    np.random.seed(42)

    # 1. Initialize centers with a perturbed hexagonal lattice
    centers = np.zeros((n, 2))
    count = 0
    # Hexagonal spacing
    row_y = 0.05
    col_x = 0.05
    while count < n:
        # Even row
        if (row_y // 0.15) % 2 == 0:
            x = col_x
        else:
            x = col_x + 0.1  # Shift for hexagonal pattern
        
        while x < 0.95 and count < n:
            centers[count] = [x, row_y]
            count += 1
            x += 0.2
        row_y += 0.15
    
    # Add random jitter to avoid symmetric locking
    centers += np.random.uniform(-0.001, 0.001, centers.shape)
    # Ensure centers are within bounds initially
    centers = np.clip(centers, 1e-4, 1 - 1e-4)

    # 2. Optimization Loop
    for iteration in range(2000):
        # Solve LP to find optimal radii for current centers
        c = opt.linprog.c
        A_ub = []
        b_ub = []
        
        # Objective: maximize sum(r) -> minimize -sum(r)
        c = np.ones(n) * -1.0

        # Constraints: r_i + r_j <= dist(i, j)
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                row = np.zeros(n)
                row[i] = 1.0
                row[j] = 1.0
                A_ub.append(row)
                b_ub.append(dist)

        # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
        for i in range(n):
            x, y = centers[i]
            A_ub.append(np.zeros(n) if False else np.eye(n)[i]) # r_i <= x_i
            b_ub.append(x)
            A_ub.append(np.eye(n)[i]) # r_i <= 1-x_i
            b_ub.append(1 - x)
            A_ub.append(np.eye(n)[i]) # r_i <= y_i
            b_ub.append(y)
            A_ub.append(np.eye(n)[i]) # r_i <= 1-y_i
            b_ub.append(1 - y)

        A_ub = np.array(A_ub)
        b_ub = np.array(b_ub)
        bounds = [(0, None) for _ in range(n)]

        res = opt.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        radii = np.array(res.x)

        # 3. Move centers to relax active constraints
        learning_rate = 0.0005 * (1 - iteration / 3000.0) # Decay learning rate
        forces = np.zeros_like(centers)

        # Check circle-circle contacts
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                # If touching (within a small margin)
                if dist < radii[i] + radii[j] + 1e-4:
                    dir_vec = (centers[i] - centers[j]) / (dist + 1e-8)
                    # Repulsive force
                    forces[i] += dir_vec * (radii[i] + radii[j] - dist + 1e-4)
                    forces[j] -= dir_vec * (radii[i] + radii[j] - dist + 1e-4)

        # Check boundary contacts
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            # If touching left wall (x - r approx 0)
            if x < r + 1e-4:
                forces[i, 0] += (r - x + 1e-4) * 0.5
            # If touching right wall (1 - x - r approx 0)
            if (1 - x) < r + 1e-4:
                forces[i, 0] -= (r - (1 - x) + 1e-4) * 0.5
            # If touching bottom wall (y - r approx 0)
            if y < r + 1e-4:
                forces[i, 1] += (r - y + 1e-4) * 0.5
            # If touching top wall (1 - y - r approx 0)
            if (1 - y) < r + 1e-4:
                forces[i, 1] -= (r - (1 - y) + 1e-4) * 0.5

        centers += forces * learning_rate
        # Keep centers inside square
        centers = np.clip(centers, 1e-5, 1 - 1e-5)

    # Final validation to ensure no NaN or negative radii
    radii = np.maximum(radii, 0.0)
    sum_radii = np.sum(radii)

    return centers, radii, float(sum_radii)

if __name__ == "__main__":
    c, r, s = run_packing()
    print(f"Sum of radii: {s}")
