import numpy as np
from scipy.optimize import minimize

def compute_overlap_penalty(centers, r, n):
    """
    Computes a smooth penalty based on circle overlaps.
    Returns 0 if no overlaps, positive value otherwise.
    """
    c = centers.reshape(n, 2)
    penalty = 0.0
    for i in range(n):
        for j in range(i + 1, n):
            dx = c[i, 0] - c[j, 0]
            dy = c[i, 1] - c[j, 1]
            dist = np.sqrt(dx*dx + dy*dy)
            overlap = 2.0 * r - dist
            if overlap > 0:
                penalty += overlap * overlap
    return penalty

def get_hex_initial(n, r):
    """
    Generates an initial configuration based on a hexagonal lattice.
    Falls back to random positions if the lattice doesn't yield n points.
    """
    pts = []
    y = r
    row_idx = 0
    while len(pts) < n:
        x = r
        # Hexagonal shift: alternate rows are offset by r horizontally
        shift = r if row_idx % 2 == 1 else 0.0
        while len(pts) < n:
            cx = x + shift
            cy = y
            # Check if point fits within the valid region [r, 1-r]
            if cx + r <= 1.0 and cy + r <= 1.0:
                pts.append([cx, cy])
            x += 2.0 * r
        y += np.sqrt(3) * r
        row_idx += 1
        
    # Pad with random valid points if necessary
    while len(pts) < n:
        pts.append([np.random.uniform(r, 1-r), np.random.uniform(r, 1-r)])
    return np.array(pts[:n])

    best_centers = None

    for _ in range(30):
        mid = (low + high) / 2.0
        found = False
        
        # Try multiple initial configurations to escape local minima
        for _ in range(5):
            if np.random.random() < 0.5:
                x0 = get_hex_initial(n, mid).flatten()
            else:
                x0 = np.random.uniform(mid, 1-mid, size=(n, 2)).flatten()
                
            # Bounds ensure circles stay inside [0,1]x[0,1]
            bounds = [(mid, 1-mid)] * (2*n)
            
            try:
                res = minimize(compute_overlap_penalty, x0, args=(mid, n),
                               method='L-BFGS-B', bounds=bounds,
                               options={'maxiter': 3000, 'ftol': 1e-12})
                
                # If penalty is negligible, the configuration is valid
                if res.fun < 1e-7:
                    best_r = mid
                    best_centers = res.x.reshape(n, 2)
                    found = True
                    break
            except Exception:
                continue
                
        if found:
            low = mid
        else:
            high = mid
            
    radii = np.full(n, best_r)
    return best_centers, radii, float(np.sum(radii))
