

Looking at this circle packing problem, I need to pack 26 circles in a unit square to maximize sum of radii (target: 2.636).

## Strategy

1. **Penalty-based optimization**: Use a continuous relaxation where constraint violations are penalized, allowing gradient-based optimization to work smoothly.

2. **Two-phase optimization**: 
   - Phase 1: Differential evolution for global search to avoid local minima
   - Phase 2: L-BFGS-B for fine-grained local refinement

3. **Multiple random restarts** with different seeds to increase chance of finding high-quality solution

4. **Hexagonal initial arrangement**: Hexagonal packing is known to be optimal for equal circles. Starting from this gives better convergence.

5. **Vectorized constraint evaluation** for computational efficiency in the objective function.

6. **Iterative penalty tightening**: Start with moderate penalty lambda, then increase it to enforce constraints more strictly.

The key insight is that maximizing sum of radii (rather than minimum radius) allows unequal circle sizes - larger circles in favorable positions with smaller circles filling gaps.

```python
import numpy as np
from scipy.optimize import differential_evolution, minimize


def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26

    def objective(params):
        centers = params[:2 * n].reshape(n, 2)
        radii = params[2 * n:]

        if np.any(radii < 0):
            return 1e10

        score = np.sum(radii)

        x = centers[:, 0]
        y = centers[:, 1]

        # Boundary penalties
        penalty = np.sum(np.maximum(0, radii - x) ** 2)
        penalty += np.sum(np.maximum(0, x + radii - 1) ** 2)
        penalty += np.sum(np.maximum(0, radii - y) ** 2)
        penalty += np.sum(np.maximum(0, y + radii - 1) ** 2)

        # Overlap penalties (vectorized)
        diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
        dists = np.sqrt(np.sum(diff ** 2, axis=2))
        r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
        mask = np.tril(np.ones((n, n)), k=-1)
        violations = np.maximum(0, r_sum - dists) * mask
        penalty += np.sum(violations ** 2)

        lam = 8000
        return -score + lam * penalty

    def make_hex_initial():
        centers = np.zeros((n, 2))
        radii = np.ones(n) * 0.095
        idx = 0
        for row in range(6):
            for col in range(5):
                if idx >= n:
                    break
                x = 0.08 + col * 0.175
                y = 0.08 + row * 0.16
                if row % 2 == 1:
                    x += 0.0875
                x = np.clip(x, 0.05, 0.95)
                y = np.clip(y, 0.05, 0.95)
                centers[idx, 0] = x
                centers[idx, 1] = y
                idx += 1
        return centers, radii

    def optimize_from(start_params):
        # Local refinement with increasing penalty
        for lam_mult in [1, 2, 5, 10]:
            def obj_wrapped(params):
                centers = params[:2 * n].reshape(n, 2)
                radii = params[2 * n:]
                if np.any(radii < 0):
                    return 1e10
                score = np.sum(radii)
                x = centers[:, 0]
                y = centers[:, 1]
                penalty = np.sum(np.maximum(0, radii - x) ** 2)
                penalty += np.sum(np.maximum(0, x + radii - 1) ** 2)
                penalty += np.sum(np.maximum(0, radii - y) ** 2)
                penalty += np.sum(np.maximum(0, y + radii - 1) ** 2)
                diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
                dists = np.sqrt(np.sum(diff ** 2, axis=2))
                r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
                mask = np.tril(np.ones((n, n)), k=-1)
                violations = np.maximum(0, r_sum - dists) * mask
                penalty += np.sum(violations ** 2)
                return -score + (8000 * lam_mult) * penalty

            result = minimize(
                obj_wrapped, start_params, method="L-BFGS-B", bounds=bounds,
                options={"maxiter": 15000, "ftol": 1e-12}
            )
            start_params = result.x
        return start_params

    bounds = [(0.001, 0.999)] * (2 * n) + [(0.001, 0.16)] * n

    best_x = None
    best_obj = float("inf")

    # Run differential evolution with multiple seeds
    for seed in range(15):
        # Create slightly varied initial guess
        centers_init, radii_init = make_hex_initial()
        centers_init += np.random.default_rng(seed).normal(0, 0.01, centers_init.shape)
        radii_init += np.random.default_rng(seed).normal(0, 0.005, n)
        radii_init = np.clip(radii_init, 0.02, 0.14)

        initial_guess = np.concatenate([centers_init.flatten(), radii_init])

        result_de = differential_evolution(
            objective, bounds, maxiter=400, popsize=12, seed=seed,
            initial_guess=initial_guess, mutation=(0.5, 1.5), recombination=0.9
        )

        # Local refinement
        refined = optimize_from(result_de.x)
        refined_obj = objective(refined)

        if refined_obj < best_obj:
            best_obj = refined_obj
            best_x = refined.copy()

    centers = best_x[:2 * n].reshape(n, 2)
    radii = np.maximum(best_x[2 * n:], 0)

    # Final safety clip to ensure validity
    for i in range(n):
        r = radii[i]
        centers[i, 0] = np.clip(centers[i, 0], r, 1 - r)
        centers[i, 1] = np.clip(centers[i, 1], r, 1 - r)

    return centers, radii, float(np.sum(radii))
