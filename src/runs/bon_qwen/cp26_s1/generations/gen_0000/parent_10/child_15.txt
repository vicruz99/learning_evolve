import numpy as np
from scipy.optimize import minimize


def compute_constraints(centers, radii, n):
    """Compute all constraint values for the optimization."""
    cons = []
    
    # Boundary constraints: each circle must be inside [0,1]^2
    for i in range(n):
        cons.append(centers[i, 0] - radii[i])
        cons.append(1.0 - centers[i, 0] - radii[i])
        cons.append(centers[i, 1] - radii[i])
        cons.append(1.0 - centers[i, 1] - radii[i])
    
    # Non-overlap constraints: distance >= sum of radii
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            cons.append(dist - radii[i] - radii[j])
    
    return np.array(cons)


def hexagonal_init(n):
    """Initialize circles in hexagonal pattern."""
    centers = np.zeros((n, 2))
    idx = 0
    row = 0
    base_spacing = 0.175
    while idx < n:
        cols = 6 if row % 2 == 0 else 5
        if idx + cols > n:
            cols = n - idx
        for col in range(cols):
            shift = base_spacing * 0.5 if row % 2 == 1 else 0.0
            x = 0.09 + col * base_spacing + shift
            y = 0.09 + row * base_spacing * np.sqrt(3.0) / 2.0
            centers[idx] = [x, y]
            idx += 1
        row += 1
    return centers


def grid_init(n):

            x = 0.08 + col * 0.18
            y = 0.08 + row * 0.19
            centers[idx] = [x, y]
            idx += 1
    return centers

def optimize_packing(n, centers_init, radii_init, max_iter=8000):
    """Run SLSQP optimization to maximize sum of radii."""
    x0 = np.concatenate([centers_init.flatten(), radii_init])
    
    def objective(x):
        return -np.sum(x[2 * n:])

        c = x[:2 * n].reshape(n, 2)
        r = x[2 * n:]
        return compute_constraints(c, r, n)
    
    bounds = [(0.0, 1.0)] * (2 * n) + [(1e-8, 0.5)] * n
    
    result = minimize(
        objective,
        x0,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'ineq', 'fun': constraint_fun},

    )
    
    centers_opt = result.x[:2 * n].reshape(n, 2)
    radii_opt = result.x[2 * n:]
    
    # Ensure radii are non-negative
    radii_opt = np.maximum(radii_opt, 1e-10)
    
    return centers_opt, radii_opt, np.sum(radii_opt)
