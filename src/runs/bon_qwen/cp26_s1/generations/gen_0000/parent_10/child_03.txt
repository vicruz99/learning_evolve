import numpy as np
from scipy.optimize import minimize
from functools import partial

def circle_overlap_constraint(x, n, i, j):
    ci = x[2*i:2*i+2]
    cj = x[2*j:2*j+2]
    ri = x[2*n + i]
    rj = x[2*n + j]
    dist_sq = np.sum((ci - cj)**2)
    return dist_sq - (ri + rj)**2

def left_constraint(x, n, i):
    return x[2*i] - x[2*n + i]

def right_constraint(x, n, i):
    return 1 - x[2*i] - x[2*n + i]

def bottom_constraint(x, n, i):
    return x[2*i + 1] - x[2*n + i]

def top_constraint(x, n, i):
    return 1 - x[2*i + 1] - x[2*n + i]

def radius_constraint(x, n, i):
    return x[2*n + i]

def negative_sum_radii(x, n):
    return -np.sum(x[2*n:])

def build_constraints(n):
    constraints = []
    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({
                'type': 'ineq',
                'fun': partial(circle_overlap_constraint, n=n, i=i, j=j)
            })
    for i in range(n):
        constraints.append({'type': 'ineq', 'fun': partial(left_constraint, n=n, i=i)})
        constraints.append({'type': 'ineq', 'fun': partial(right_constraint, n=n, i=i)})
        constraints.append({'type': 'ineq', 'fun': partial(bottom_constraint, n=n, i=i)})
        constraints.append({'type': 'ineq', 'fun': partial(top_constraint, n=n, i=i)})
        constraints.append({'type': 'ineq', 'fun': partial(radius_constraint, n=n, i=i)})
    return constraints

def hexagonal_init(n, r0):
    centers = []
    idx = 0
    row = 0
    while idx < n and row < 20:
        col = 0
        while col < 20 and idx < n:
            x = r0 + col * 2 * r0 + (row % 2) * r0
            y = r0 + row * np.sqrt(3) * r0
            if x + r0 <= 1.0 and y + r0 <= 1.0:
                centers.append([x, y])
                idx += 1
            col += 1
        row += 1
    return np.array(centers[:n])

def staggered_grid_init(n, r0):
    centers = []
    cols = int(np.ceil(np.sqrt(n * 1.5)))
    rows = int(np.ceil(n / cols))
    for row in range(rows):
        for col in range(cols):
            if len(centers) >= n:
                break
            offset = (row % 2) * r0
            x = r0 + col * 2 * r0 + offset
            y = r0 + row * np.sqrt(3) * r0
            if x + r0 <= 1.0 and y + r0 <= 1.0:
                centers.append([x, y])
        if len(centers) >= n:
            break
    return np.array(centers[:n])

def random_init(n, r0):
    centers = np.zeros((n, 2))
    for i in range(n):
        centers[i, 0] = r0 + np.random.rand() * (1 - 2 * r0)
        centers[i, 1] = r0 + np.random.rand() * (1 - 2 * r0)
    return centers

def is_valid_packing(n, x):
    centers = x[:2*n].reshape(n, 2)
    radii = x[2*n:]
    for i in range(n):
        if radii[i] < -1e-10:
            return False
        cx, cy = centers[i]
        r = radii[i]
        if cx - r < -1e-6 or cx + r > 1 + 1e-6:
            return False
        if cy - r < -1e-6 or cy + r > 1 + 1e-6:
            return False
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            if dist < radii[i] + radii[j] - 1e-6:
                return False
    return True

def optimize_from_initial(n, constraints, x0, max_iter=3000):
    result = minimize(negative_sum_radii, x0, args=(n,),
                     method='SLSQP', constraints=constraints,
                     options={'maxiter': max_iter, 'ftol': 1e-14, 'disp': False})
    return result

def run_packing():

                        if is_valid_packing(n, x_fixed):

                                best_obj = result.fun
                                best_result = result

            continue
