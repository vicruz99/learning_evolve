import numpy as np
from scipy.optimize import minimize

def run_packing():
    n = 26
    
    def constr(v):
        c = v[:2*n].reshape(n, 2)
        r = v[2*n:]
        m = 4*n + n*(n-1)//2
        vals = np.empty(m)
        idx = 0
        for i in range(n):
            vals[idx] = c[i,0] - r[i]
            vals[idx+1] = 1 - c[i,0] - r[i]
            vals[idx+2] = c[i,1] - r[i]
            vals[idx+3] = 1 - c[i,1] - r[i]
            idx += 4
        for i in range(n):
            for j in range(i+1, n):
                dx = c[i,0] - c[j,0]
                dy = c[i,1] - c[j,1]
                vals[idx] = np.sqrt(dx*dx + dy*dy) - r[i] - r[j]
                idx += 1
        return vals
        
    cons = {'type': 'ineq', 'fun': constr}
    bounds = [(0, 1) for _ in range(2*n)] + [(0, 0.5) for _ in range(n)]
    
    def optimize_from(v0):
        res = minimize(lambda v: -np.sum(v[2*n:]), v0, method='SLSQP', bounds=bounds, 
                       constraints=cons, options={'maxiter': 1000, 'ftol': 1e-9, 'disp': False})
        return res.x, np.sum(res.x[2*n:])
        
    best_v = None
    best_sum = 0.0
    guesses = []
    
    # 1. Hexagonal lattice
    c_hex = []
    r_init = 0.08
    y = r_init + 0.001
    while len(c_hex) < n:
        x = r_init + 0.001
        shift = r_init if int((y - r_init) / (r_init * np.sqrt(3))) % 2 == 1 else 0
        while x + r_init <= 1.0 and len(c_hex) < n:
            c_hex.append([x + shift, y])
            x += 2 * r_init
        y += r_init * np.sqrt(3)
    guesses.append(np.concatenate([np.array(c_hex[:n]).flatten(), np.full(n, 0.05)]))
    
    # 2. Uniform grid
    c_grid = []
    for i in range(6):
        for j in range(5):
            if len(c_grid) < n:
                c_grid.append([0.1 + i*0.15, 0.1 + j*0.18])
    guesses.append(np.concatenate([np.array(c_grid[:n]).flatten(), np.full(n, 0.04)]))
    
    # 3. Random
    np.random.seed(123)
    c_rand = np.random.uniform(0.15, 0.85, (n, 2))
    guesses.append(np.concatenate([c_rand.flatten(), np.full(n, 0.04)]))
    
    for v0 in guesses:
        try:
            v_opt, s = optimize_from(v0)
            if s > best_sum:
                best_sum = s
                best_v = v_opt
        except Exception:
            continue
            
    if best_v is None:
        best_v = guesses[0]
        
    return best_v[:2*n].reshape(n, 2), best_v[2*n:], np.sum(best_v[2*n:])
