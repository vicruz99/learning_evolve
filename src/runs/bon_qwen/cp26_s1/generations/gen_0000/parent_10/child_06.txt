import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # --- Phase 1: Initialization ---
    # We initialize with a hexagonal-like arrangement to provide a dense starting configuration.
    # 26 circles: approx 5 rows. Pattern 5, 5, 6, 5, 5 (26 circles).
    # We place them in a grid and add some jitter to break symmetry and avoid saddle points.
    
    centers_init = []
    count = 0
    # Parameters for a rough hexagonal grid fit in [0,1]
    x_spacing = 1.0 / 6.0
    y_spacing = 1.0 / 5.0
    
    # Pattern: 5, 5, 6, 5, 5
    row_counts = [5, 5, 6, 5, 5]
    
    rng = np.random.RandomState(123)
    
    for r_idx, count_circles in enumerate(row_counts):
        y_pos = (r_idx + 0.5) * y_spacing
        # Stagger rows slightly for hexagonal effect
        x_offset = 0.5 * x_spacing if (r_idx % 2 == 1) else 0.0
        
        # Distribute 'count_circles' in the row
        row_spacing = 1.0 / (count_circles + 1) # Leave some margin
        
        for c_idx in range(count_circles):
            # Base grid position
            x_base = (c_idx + 1) * row_spacing + x_offset
            
            # Add small jitter
            jitter = rng.uniform(-0.02, 0.02, 2)
            cx = np.clip(x_base + jitter[0], 0.1, 0.9)
            cy = np.clip(y_pos + jitter[1], 0.1, 0.9)
            
            centers_init.append([cx, cy])
            
            if count >= n:
                break
        if count >= n:
            break
            
    centers_init = np.array(centers_init[:n])
    
    # Initial radius: small enough to be non-overlapping
    radii_init = np.ones(n) * 0.05
    
    # --- Phase 2: Optimization ---
    
    def objective(x_vars):
        # x_vars contains [x1, y1, r1, x2, y2, r2, ...]
        # We want to maximize sum(r), so minimize -sum(r)
        radii = x_vars[2::3]
        return -np.sum(radii)

    def constraint_boundary(x_vars):
        # Constraints: x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
        # Returns array of values >= 0
        constraints = np.empty(n * 4)
        for i in range(n):
            x_idx = 3 * i
            y_idx = 3 * i + 1
            r_idx = 3 * i + 2
            
            cx = x_vars[x_idx]
            cy = x_vars[y_idx]
            cr = x_vars[r_idx]
            
            constraints[4 * i]     = cx - cr       # x - r >= 0
            constraints[4 * i + 1] = 1.0 - cx - cr # 1 - x - r >= 0
            constraints[4 * i + 2] = cy - cr       # y - r >= 0
            constraints[4 * i + 3] = 1.0 - cy - cr # 1 - y - r >= 0
        return constraints

    def constraint_overlap(x_vars):
        # Constraint: dist(c_i, c_j) >= r_i + r_j
        # dist^2 >= (r_i + r_j)^2
        # dist^2 - (r_i + r_j)^2 >= 0
        constraints = []
        for i in range(n):
            for j in range(i + 1, n):
                idx_i = 3 * i
                idx_j = 3 * j
                
                dx = x_vars[idx_i] - x_vars[idx_j]
                dy = x_vars[idx_i + 1] - x_vars[idx_j + 1]
                dr = x_vars[idx_i + 2] + x_vars[idx_j + 2]
                
                val = dx*dx + dy*dy - dr*dr
                constraints.append(val)
        return np.array(constraints)

    # Flatten initial state
    x0 = np.zeros(3 * n)
    for i in range(n):
        x0[3 * i]     = centers_init[i, 0]
        x0[3 * i + 1] = centers_init[i, 1]
        x0[3 * i + 2] = radii_init[i]

    # Bounds: x, y in [0, 1], r in [0, 0.5]
    bounds = []
    for i in range(n):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 0.5)) # r (max possible radius is 0.5)

    cons = [
        {'type': 'ineq', 'fun': constraint_boundary},
        {'type': 'ineq', 'fun': constraint_overlap}
    ]

    # Run Optimization
    # SLSQP is good for constrained problems
    result = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, 
                      options={'maxiter': 500, 'ftol': 1e-9})

    # Extract result
    opt_vars = result.x
    opt_centers = np.array([[opt_vars[3*i], opt_vars[3*i+1]] for i in range(n)])
    opt_radii = np.array([opt_vars[3*i+2] for i in range(n)])

    # --- Phase 3: Boundary Pushing (Refinement) ---
    # Sometimes optimizers leave circles floating near boundaries. 
    # We can try to expand them slightly by moving them into corners/edges.
    # This is a simple greedy local search.
    
    # Sort circles by radius descending to try to expand largest ones first?
    # Or just iterate.
    # We check if a circle can be moved to touch a boundary and expand.
    
    improved = True
    while improved:
        improved = False
        for i in range(n):
            cx, cy = opt_centers[i]
            r = opt_radii[i]
            
            # Potential maximum radius based on boundaries
            max_r_bound = min(cx, 1.0 - cx, cy, 1.0 - cy)
            
            # If there is slack against boundary
            if max_r_bound > r + 1e-8:
                # Try to expand and move
                # Move center to touch boundary that limits size?
                # Actually, just center the circle in the available boundary box?
                # Better: move center so it touches the closest wall?
                
                # Let's try moving center to expand r.
                # New r would be limited by other circles too.
                # Simple heuristic: Move center towards the wall that is closest to center?
                # No, move center so that (cx - r) = 0 or (cx + r) = 1 etc.
                
                # Let's check distance to boundaries
                d_left = cx - r
                d_right = 1.0 - (cx + r)
                d_down = cy - r
                d_up = 1.0 - (cy + r)
                
                min_d = min(d_left, d_right, d_down, d_up)
                
                if min_d > 1e-8:
                    # Move circle to touch that boundary
                    if d_left == min_d:
                        new_r = cx # Touch left wall
                        # Check if this r is valid against neighbors
                        # If valid, apply
                    elif d_right == min_d:
                        new_r = 1.0 - cx
                    elif d_down == min_d:
                        new_r = cy
                    else:
                        new_r = 1.0 - cy
                    
                    # Check if new_r is feasible with neighbors
                    feasible = True
                    for j in range(n):
                        if i == j: continue
                        dist = np.sqrt(np.sum((opt_centers[i] - opt_centers[j])**2))
                        if dist < new_r + opt_radii[j] - 1e-12:
                            feasible = False
                            break
                    
                    if feasible:
                        opt_radii[i] = new_r
                        improved = True

    sum_radii = np.sum(opt_radii)
    
    # Validate internally
    # (Optional but good for debugging)
    # assert validate_packing(opt_centers, opt_radii)
    
    return opt_centers, opt_radii, sum_radii
