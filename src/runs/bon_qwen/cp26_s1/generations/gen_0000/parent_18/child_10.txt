import numpy as np
import scipy.optimize as opt

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True

def objective(vars_flat, n):
    """
    Objective function: minimize negative sum of radii.
    vars_flat contains [x1, y1, r1, x2, y2, r2, ...]
    """
    radii = vars_flat[2::3]
    return -np.sum(radii)

def get_constraints(n):
    """
    Generate constraints for the optimization problem.
    """
    constraints = []

    # Boundary constraints:
    # x_i - r_i >= 0  => r_i - x_i <= 0
    # x_i + r_i <= 1  => r_i + x_i - 1 <= 0
    # y_i - r_i >= 0  => r_i - y_i <= 0
    # y_i + r_i <= 1  => r_i + y_i - 1 <= 0
    
    for i in range(n):
        # indices in vars_flat: 3*i, 3*i+1, 3*i+2
        idx_x = 3 * i
        idx_y = 3 * i + 1
        idx_r = 3 * i + 2

        # r - x <= 0
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: v[idx_x] - v[idx_r]}) # x - r >= 0
        # x + r - 1 <= 0
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: 1.0 - (v[idx_x] + v[idx_r])}) # 1 - (x+r) >= 0
        # r - y <= 0
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: v[idx_y] - v[idx_r]}) # y - r >= 0
        # y + r - 1 <= 0
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: 1.0 - (v[idx_y] + v[idx_r])}) # 1 - (y+r) >= 0

    # Non-overlap constraints:
    # (x_i - x_j)^2 + (y_i - y_j)^2 >= (r_i + r_j)^2
    # Distance squared - (r_i + r_j)^2 >= 0
    
    for i in range(n):
        for j in range(i + 1, n):
            idx_xi = 3 * i
            idx_yi = 3 * i + 1
            idx_ri = 3 * i + 2
            idx_xj = 3 * j
            idx_yj = 3 * j + 1
            idx_rj = 3 * j + 2

            def dist_constraint(v, i=i, j=j):
                xi, yi, ri = v[idx_xi], v[idx_yi], v[idx_ri]
                xj, yj, rj = v[idx_xj], v[idx_yj], v[idx_rj]
                dist_sq = (xi - xj)**2 + (yi - yj)**2
                sum_r_sq = (ri + rj)**2
                return dist_sq - sum_r_sq

            constraints.append({'type': 'ineq', 'fun': dist_constraint})

    return constraints

def get_bounds(n):
    """
    Variable bounds: x, y in [0, 1], r in [0, 0.5]
    """
    bounds = []
    for _ in range(n):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((1e-6, 0.5)) # r (lower bound > 0 to avoid trivial solution)
    return bounds

def generate_grid_initialization(n):
    """
    Initialize circles in a grid pattern.
    """
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Grid size
    rows = int(np.ceil(np.sqrt(n)))
    cols = rows
    
    # Spacing
    step_x = 1.0 / (cols + 1)
    step_y = 1.0 / (rows + 1)
    
    r_est = min(step_x, step_y) / 2.0
    
    idx = 0
    for r in range(rows):
        for c in range(cols):
            if idx < n:
                x = (c + 1) * step_x
                y = (r + 1) * step_y
                centers[idx] = [x, y]
                radii[idx] = r_est
                idx += 1
    return centers, radii
