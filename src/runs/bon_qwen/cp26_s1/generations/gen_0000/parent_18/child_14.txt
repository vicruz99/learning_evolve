import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Finds a packing of 26 circles in a unit square to maximize the sum of radii.
    """
    n = 26
    
    # Binary search parameters for the equal radius r
    low = 0.01
    high = 0.15  # Upper bound estimate based on density
    best_r = 0.0
    best_centers = None
    
    # Optimization parameters
    max_iter = 100
    num_restarts = 8
    
    for _ in range(max_iter):
        mid = (low + high) / 2
        
        # If radius is too large for any circle to exist
        if mid > 0.5:
            high = mid
            continue
            
        feasible = False
        current_centers = None
        
        # Attempt to find a valid configuration for radius 'mid'
        for attempt in range(num_restarts):
            # Generate initial guess
            if attempt == 0:
                # Try Hexagonal initialization for better convergence
                init_centers = generate_hexagonal_guess(n, mid)
                if init_centers is None:
                    # Fallback if hex grid doesn't generate enough points
                    init_centers = generate_random_guess(n, mid)
            else:
                # Random initialization for diversity
                init_centers = generate_random_guess(n, mid)
                
            # Setup optimization bounds (must be inside [r, 1-r])
            bounds = [(mid, 1 - mid) for _ in range(2 * n)]
            
            # Objective function: Minimize squared overlap
            # Overlap for pair (i, j) is max(0, 2r - dist)^2
            def objective(x_flat):
                centers = x_flat.reshape((n, 2))
                penalty = 0.0
                
                # Compute pairwise distances and overlaps
                # Using broadcasting for efficiency
                # dist_sq = (x_i - x_j)^2 + (y_i - y_j)^2
                # We only need to compute for i < j to avoid duplication
                
                # Flattened centers for broadcasting
                # Shape (n, 2)
                c = centers
                # Compute distance matrix
                # dist[i, j] = sqrt((c[i]-c[j])**2)
                # Optimization: sum of squared violations
                # violation = (2r - dist)^2 if dist < 2r else 0
                
                # Vectorized distance calculation
                # (n, 1, 2) - (1, n, 2) -> (n, n, 2)
                diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
                dist_sq = np.sum(diff**2, axis=2)
                dist = np.sqrt(dist_sq)
                
                # Identify pairs with distance < 2r
                # Mask for upper triangle to avoid double counting and self
                mask = np.triu(np.ones((n, n), dtype=bool), k=1)
                min_dist = 2 * mid
                violation = np.maximum(0, min_dist - dist)
                penalty = np.sum((violation[mask])**2)
                
                return penalty

            try:
                res = minimize(
                    objective, 
                    init_centers.flatten(), 
                    method='L-BFGS-B', 
                    bounds=bounds, 
                    options={'ftol': 1e-12, 'maxiter': 500}
                )

                    feasible = True
                    current_centers = res.x.reshape((n, 2))

            best_centers = current_centers
        else:

    if best_centers is None:

        best_r = 0.1
        best_centers = generate_random_guess(n, best_r)
        
    radii = np.full(n, best_r)
    sum_radii = np.sum(radii)
    
    return best_centers, radii, sum_radii

def generate_hexagonal_guess(n, r):
    """
    Attempts to place n circles in a hexagonal pattern.
    """
    centers = []
    # Hexagonal packing:
    # Rows at y = r + k * sqrt(3)*r
    # x offset alternates between r (even rows) and 2r (odd rows) relative to start?
    # Actually, standard hex:
    # Row 0 (y=r): x = r, 3r, 5r ...
    # Row 1 (y=r + sqrt(3)r): x = 2r, 4r, 6r ...
    
    k = 0
    while len(centers) < n:
        y = r + k * np.sqrt(3) * r
        if y > 1 - r:
            break # Row too high
            
        shift = (k % 2) * r # 0 for even (starts at r), r for odd (starts at 2r)
        # Wait, if shift is 0, x starts at r. If shift is r, x starts at r+r = 2r.
        # Correct.
        
        x_start = r + shift
        m = 0
        while True:
            x = x_start + 2 * r * m
            if x > 1 - r:
                break
            centers.append([x, y])
            if len(centers) >= n:
                break
            m += 1
        k += 1
        
    if len(centers) < n:
        return None
    return np.array(centers[:n])

def generate_random_guess(n, r):
    """
    Generates random centers within bounds [r, 1-r].
    """
    low = r
    high = 1 - r
    if low > high:
        return None
    return np.random.uniform(low, high, size=(n, 2))
