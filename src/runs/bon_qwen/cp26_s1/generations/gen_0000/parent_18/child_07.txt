I will use a multi-stage approach:
1. **Force-directed initialization**: Start with circles on a hexagonal grid, then use repulsive forces to resolve overlaps while gradually increasing radii. This produces a valid, reasonably dense packing.
2. **Scipy polishing**: Use L-BFGS-B with a smooth penalty formulation to maximize sum of radii while maintaining constraints.
3. **Multiple restarts**: Run several optimizations from different initial configurations to avoid local minima and find a high-quality solution.
4. **Validation**: Ensure the final packing passes the validation check.

The key insight is that force-directed simulation naturally handles the non-convex geometry of circle packing, while gradient-based optimization can fine-tune the result. The penalty method converts hard constraints into soft penalties in the objective function, allowing scipy to optimize freely.

Key design decisions:
- Strong repulsive forces proportional to overlap amount
- Boundary forces proportional to boundary violation
- Gradual radius growth to explore larger configurations
- Cooling schedule to converge to a stable packing
- Multiple random restarts to escape local minima
- Final validation to ensure constraints are satisfied

import numpy as np
from scipy.optimize import minimize


def compute_penalty(centers, radii, n):
    """Compute smooth penalty for all constraint violations."""
    penalty = 0.0

    # Boundary violations
    for i in range(n):
        x, y = centers[i]
        r = radii[i]

        for val in [x - r, 1.0 - x - r, y - r, 1.0 - y - r]:
            if val < 0:
                penalty += 1000.0 * val * val

        if r < 0:
            penalty += 10000.0 * r * r

    # Overlap violations
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i][0] - centers[j][0]
            dy = centers[i][1] - centers[j][1]
            dist = np.sqrt(dx * dx + dy * dy + 1e-12)
            min_dist = radii[i] + radii[j]
            if dist < min_dist:
                penalty += 1000.0 * (min_dist - dist) * (min_dist - dist)

    return penalty


def objective(params, n):
    """Objective: maximize sum of radii (minimize negative sum) plus penalty."""
    centers = params[:2 * n].reshape(n, 2)
    radii = params[2 * n:]
    return -np.sum(radii) + compute_penalty(centers, radii, n)


def force_directed_pack(n, centers, radii, n_steps=8000):
    """Run force-directed simulation to pack circles."""
    dt = 0.08

    for step in range(n_steps):
        forces = np.zeros((n, 2))

        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[j][0] - centers[i][0]
                dy = centers[j][1] - centers[i][1]
                dist = np.sqrt(dx * dx + dy * dy + 1e-12)
                min_dist = radii[i] + radii[j]

                if dist < min_dist:
                    overlap = min_dist - dist
                    force_mag = overlap / dist * 30.0
                    fx = force_mag * dx
                    fy = force_mag * dy
                    forces[i, 0] -= fx
                    forces[i, 1] -= fy
                    forces[j, 0] += fx
                    forces[j, 1] += fy

            # Boundary forces
            x, y = centers[i]
            r = radii[i]

            if x - r < 0:
                forces[i, 0] += (x - r) * 60.0
            if x + r > 1.0:
                forces[i, 0] -= (x + r - 1.0) * 60.0
            if y - r < 0:
                forces[i, 1] += (y - r) * 60.0
            if y + r > 1.0:
                forces[i, 1] -= (y + r - 1.0) * 60.0

        centers += dt * forces
        centers = np.clip(centers, 0.0, 1.0)

        # Gradually increase radii
        radii *= 1.00008

        # Cooling
        dt *= 0.99996

    return centers, radii


def hexagonal_init(n, radius):
    """Initialize circles on a hexagonal grid."""
    centers = np.zeros((n, 2))
    radii = np.ones(n) * radius

    spacing = radius * 2.2
    idx = 0
    for row in range(12):
        for col in range(12):
            if idx >= n:
                return centers, radii
            x = 0.1 + col * spacing
            y = 0.1 + row * spacing * np.sqrt(3.0) / 2.0
            if col % 2 == 1:
                y += spacing * np.sqrt(3.0) / 4.0
            if 0.02 <= x <= 0.98 and 0.02 <= y <= 0.98:
                centers[idx] = [x, y]
                idx += 1

    return centers, radii


def random_init(n):
    """Initialize circles randomly."""
    centers = np.random.uniform(0.1, 0.9, (n, 2))
    radii = np.ones(n) * np.random.uniform(0.06, 0.10)
    return centers, radii


def optimize_from_initial(centers, radii, n):
    """Polish a packing using scipy L-BFGS-B."""
    params = np.concatenate([centers.flatten(), radii])

    result = minimize(
        objective,
        params,
        args=(n,),
        method='L-BFGS-B',
        bounds=[(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n,
        options={'maxiter': 20000, 'ftol': 1e-14, 'gtol': 1e-10}
    )

    centers_opt = result.x[:2 * n].reshape(n, 2)
    radii_opt = result.x[2 * n:]
    return centers_opt, radii_opt


def is_valid(centers, radii, n):
    """Check if packing is valid (no overlaps, inside boundaries)."""
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-9 or x + r > 1.0 + 1e-9:
            return False
        if y - r < -1e-9 or y + r > 1.0 + 1e-9:
            return False
        if r < -1e-9:
            return False

    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i][0] - centers[j][0]
            dy = centers[i][1] - centers[j][1]
            dist = np.sqrt(dx * dx + dy * dy)
            if dist < radii[i] + radii[j] - 1e-9:
                return False

    return True


def run_packing():
    n = 26

    best_centers = None
    best_radii = None
    best_sum = 0.0

    # Strategy 1: Hexagonal initialization with force-directed + polish
    for trial in range(3):
        centers, radii = hexagonal_init(n, 0.09)
        centers, radii = force_directed_pack(n, centers, radii, n_steps=6000)
        centers, radii = optimize_from_initial(centers, radii, n)
        penalty = compute_penalty(centers, radii, n)
        if penalty < 0.1:
            s = np.sum(radii)
            if s > best_sum:
                best_centers = centers.copy()
                best_radii = radii.copy()
                best_sum = s

    # Strategy 2: Random restarts with force-directed + polish
    for trial in range(25):
        centers, radii = random_init(n)
        centers, radii = force_directed_pack(n, centers, radii, n_steps=5000)
        centers, radii = optimize_from_initial(centers, radii, n)
        penalty = compute_penalty(centers, radii, n)
        if penalty < 0.5:
            s = np.sum(radii)
            if s > best_sum:
                best_centers = centers.copy()
                best_radii = radii.copy()
                best_sum = s

    # Strategy 3: Grid-based initialization variations
    for seed_val in range(5):
        np.random.seed(seed_val + 42)
        centers = np.zeros((n, 2))
        radii = np.ones(n) * 0.085

        cols = 6
        rows = 5
        idx = 0
        for i in range(cols):
            for j in range(rows):
                if idx >= n:
                    break
                base_x = 0.08 + i * 0.84 / max(cols - 1, 1)
                base_y = 0.08 + j * 0.84 / max(rows - 1, 1)
                jitter_x = np.random.uniform(-0.03, 0.03)
                jitter_y = np.random.uniform(-0.03, 0.03)
                cx = base_x + jitter_x
                cy = base_y + jitter_y
                cx = np.clip(cx, 0.05, 0.95)
                cy = np.clip(cy, 0.05, 0.95)
                centers[idx] = [cx, cy]
                radii[idx] = 0.07 + np.random.uniform(0, 0.04)
                idx += 1
            if idx >= n:
                break

        centers, radii = force_directed_pack(n, centers, radii, n_steps=5000)
        centers, radii = optimize_from_initial(centers, radii, n)
        penalty = compute_penalty(centers, radii, n)
        if penalty < 0.5:
            s = np.sum(radii)
            if s > best_sum:
                best_centers = centers.copy()
                best_radii = radii.copy()
                best_sum = s

    # Final validation and safety
    if best_centers is not None and is_valid(best_centers, best_radii, n):
        return best_centers, best_radii, np.sum(best_radii)

    # Fallback: re-optimize from best found
    if best_centers is not None:
        centers, radii = optimize_from_initial(best_centers, best_radii, n)
        if is_valid(centers, radii, n):
            return centers, radii, np.sum(radii)

    # Last resort: deterministic grid
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.08
    cols = 6
    rows = 5
    idx = 0
    for i in range(cols):
        for j in range(rows):
            if idx >= n:
                break
            centers[idx] = [0.08 + i * 0.84 / max(cols - 1, 1), 0.08 + j * 0.84 / max(rows - 1, 1)]
            idx += 1
        if idx >= n:
            break
    return centers, radii, np.sum(radii)
