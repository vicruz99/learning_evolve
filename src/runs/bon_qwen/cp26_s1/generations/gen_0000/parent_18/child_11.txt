def compute_overlaps(centers, radii):
    """
    Compute overlap energy: sum of (r_i + r_j - dist_ij)^2 for overlapping pairs.
    """
    n = centers.shape[0]
    energy = 0.0
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            min_dist = radii[i] + radii[j]
            if dist < min_dist:
                energy += (min_dist - dist) ** 2
    return energy

def boundary_penalty(centers, radii):
    """
    Compute penalty for circles being outside [0,1] x [0,1].
    """
    penalty = 0.0
    for i in range(centers.shape[0]):
        x, y = centers[i]
        r = radii[i]
        # Left
        if x - r < 0: penalty += (x - r) ** 2
        # Right
        if x + r > 1: penalty += (x + r - 1) ** 2
        # Bottom
        if y - r < 0: penalty += (y - r) ** 2
        # Top
        if y + r > 1: penalty += (y + r - 1) ** 2
    return penalty

def force_directed_packing(centers, radii, steps=2000, initial_step=0.05):
    """
    Iteratively expand radii and resolve overlaps using forces.
    """
    n = len(radii)
    # Increase radii gradually
    # We'll try to increase the sum of radii by growing them all
    # But since we want to maximize sum, let's just grow them as much as possible
    
    current_radii = radii.copy()
    
    # We will run a loop where we try to increase radii slightly and relax
    # However, a simpler way is to just run force-directed with a target expansion
    
    # Let's implement a loop that increases radius factor
    for k in range(100): # 100 expansion phases
        # Try to increase radii by a small factor or additive
        # If we just scale, we might hit local optima.
        # Let's add a small amount to all radii
        increment = 0.0002
        current_radii += increment
        
        # Relax positions to accommodate new radii
        step_size = initial_step * (0.99 ** k) # Decay step size
        if step_size < 0.0001: step_size = 0.0001
        
        for _ in range(100): # 100 relaxation steps per expansion
            forces = np.zeros_like(centers)
            
            # Inter-circle repulsion
            for i in range(n):
                for j in range(i + 1, n):
                    diff = centers[i] - centers[j]
                    dist = np.linalg.norm(diff)
                    min_dist = current_radii[i] + current_radii[j]
                    
                    if dist < min_dist:
                        # Repulsive force proportional to overlap
                        overlap = min_dist - dist
                        if dist > 1e-9:
                            direction = diff / dist
                            forces[i] += overlap * direction
                            forces[j] -= overlap * direction
                        else:
                            # Random push if coincident
                            rand_dir = np.random.randn(2)
                            rand_dir /= np.linalg.norm(rand_dir)
                            forces[i] += rand_dir
                            forces[j] -= rand_dir
            
            # Boundary repulsion
            for i in range(n):
                x, y = centers[i]
                r = current_radii[i]
                
                # Left wall
                if x - r < 0:
                    forces[i, 0] += (r - x) * 10.0
                # Right wall
                if x + r > 1:
                    forces[i, 0] -= (x + r - 1) * 10.0
                # Bottom wall
                if y - r < 0:
                    forces[i, 1] += (r - y) * 10.0
                # Top wall
                if y + r > 1:
                    forces[i, 1] -= (y + r - 1) * 10.0
            
            # Apply forces
            centers += step_size * forces
            
            # Strictly clamp to valid range to prevent escape
            # Center must be in [r, 1-r]
            for i in range(n):
                r = current_radii[i]
                centers[i, 0] = np.clip(centers[i, 0], r, 1 - r)
                centers[i, 1] = np.clip(centers[i, 1], r, 1 - r)

    return centers, current_radii
