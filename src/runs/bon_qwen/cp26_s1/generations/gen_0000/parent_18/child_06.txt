3. Hexagonal lattice arrangements are near-optimal for circle packing

2. Use SLSQP optimization to maximize sum of radii with boundary and non-overlap constraints

import numpy as np
from scipy.optimize import minimize, differential_evolution

N_CIRCLES = 26

def objective(params):
    """Maximize sum of radii (minimize negative)."""
    radii = params[2 * N_CIRCLES:]
    return -np.sum(radii)

def constraints(params):
    """Boundary and non-overlap constraints."""
    centers = params[:2 * N_CIRCLES].reshape(N_CIRCLES, 2)
    radii = params[2 * N_CIRCLES:]
    
    c = []
    # Boundary constraints
    for i in range(N_CIRCLES):
        c.append(centers[i, 0] - radii[i])
        c.append(1 - centers[i, 0] - radii[i])
        c.append(centers[i, 1] - radii[i])
        c.append(1 - centers[i, 1] - radii[i])
    
    # Non-overlap constraints
    for i in range(N_CIRCLES):
        for j in range(i + 1, N_CIRCLES):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            c.append(dist - radii[i] - radii[j])
    
    return np.array(c)

def init_hex():
    """Initialize in hexagonal pattern: 6-5-6-5-4 rows."""
    centers = np.zeros((N_CIRCLES, 2))
    radii = np.full(N_CIRCLES, 0.085)
    
    rows = [6, 5, 6, 5, 4]
    k = 0
    num_rows = len(rows)
    
    for r in range(num_rows):
        y = 0.08 + r * (0.84 / (num_rows - 1))
        shift = 0.1 if r % 2 else 0
        count = rows[r]
        if count > 1:
            x_spacing = (0.84 - 2 * shift) / (count - 1)
        else:
            x_spacing = 0
        for cc in range(count):
            x = 0.08 + shift + cc * x_spacing
            centers[k] = [x, y]
            k += 1
    
    return centers, radii

def init_hex_variant(rows_cfg):
    """Initialize in hexagonal pattern with given row configuration."""
    centers = np.zeros((N_CIRCLES, 2))
    radii = np.full(N_CIRCLES, 0.085)
    
    k = 0
    num_rows = len(rows_cfg)
    
    for r in range(num_rows):
        y = 0.08 + r * (0.84 / (num_rows - 1))
        shift = 0.1 if r % 2 else 0
        count = rows_cfg[r]
        if count > 1:
            x_spacing = (0.84 - 2 * shift) / (count - 1)
        else:
            x_spacing = 0
        for cc in range(count):
            x = 0.08 + shift + cc * x_spacing
            centers[k] = [x, y]
            k += 1
    
    return centers, radii

def run_optimization(params0, bounds, cons, max_iter=15000):
    """Run SLSQP optimization and return result."""
    try:
        res = minimize(
            objective, params0, method='SLSQP', bounds=bounds,
            constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-15}
        )
        return res
    except Exception:
        return None

def penalty_objective(params):
    """Objective with penalty for constraint violations."""
    centers = params[:2 * N_CIRCLES].reshape(N_CIRCLES, 2)
    radii = params[2 * N_CIRCLES:]
    
    obj = -np.sum(radii)
    penalty = 0
    alpha = 1e8
    
    # Boundary penalties
    for i in range(N_CIRCLES):
        penalty += alpha * max(0, radii[i] - centers[i, 0]) ** 2
        penalty += alpha * max(0, radii[i] - (1 - centers[i, 0])) ** 2
        penalty += alpha * max(0, radii[i] - centers[i, 1]) ** 2
        penalty += alpha * max(0, radii[i] - (1 - centers[i, 1])) ** 2
    
    # Overlap penalties
    for i in range(N_CIRCLES):
        for j in range(i + 1, N_CIRCLES):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            violation = radii[i] + radii[j] - dist
            penalty += alpha * max(0, violation) ** 2
    
    return obj + penalty

def check_validity(centers, radii):
    """Check if packing is valid."""
    n = centers.shape[0]
    
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-10 or x + r > 1 + 1e-10:
            return False
        if y - r < -1e-10 or y + r > 1 + 1e-10:
            return False
        if r < 0:
            return False
    
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-10:
                return False
    
    return True

def repair_packing(centers, radii):
    """Attempt to repair an invalid packing by slightly reducing radii."""
    n = centers.shape[0]
    radii = radii.copy()
    centers = centers.copy()
    
    # Fix boundary violations
    for i in range(n):
        max_r = min(centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])
        if radii[i] > max_r:
            radii[i] = max(max_r - 1e-10, 0)
    
    # Fix overlap violations iteratively
    for _ in range(100):
        changed = False
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist < radii[i] + radii[j] - 1e-12:
                    # Reduce the smaller radius
                    new_sum = dist - 1e-10
                    if new_sum < 0:
                        new_sum = 0
                    if radii[i] < radii[j]:
                        radii[i] = max(new_sum * radii[i] / max(radii[i] + radii[j], 1e-15), 0)
                    else:
                        radii[j] = max(new_sum * radii[j] / max(radii[i] + radii[j], 1e-15), 0)
                    changed = True
        if not changed:
            break
    
    return centers, radii
