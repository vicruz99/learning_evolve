The constraints are linear: $r_i + r_j \leq \text{distance}(c_i, c_j)$ for non-overlap, and $r_i \leq \text{distance}(c_i, \text{boundary})$ for containment. import numpy as np
from scipy.optimize import linprog, differential_evolution
import warnings

warnings.filterwarnings('ignore')

def solve_radii_lp(centers):
    """
    Given fixed centers, solve the LP to maximize sum of radii.
    
    Args:
        centers: np.array of shape (n, 2)
        
    Returns:
        radii: np.array of shape (n) with optimal radii
        sum_radii: float
    """
    n = centers.shape[0]
    
    # Objective: maximize sum(r_i) -> minimize -sum(r_i)
    c_obj = -np.ones(n)
    
    # Inequality constraints A_ub @ x <= b_ub
    # Constraints: r_i + r_j <= dist(i, j)
    # And r_i <= dist(i, boundary)
    
    # Calculate pairwise distances
    # Using broadcasting for distance matrix
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.linalg.norm(diff, axis=2)
    
    # Create A_ub matrix for pairwise constraints
    # We only need i < j to avoid redundancy, but for simplicity we can build all
    # Actually, standard form is Ax <= b.
    # r_i + r_j <= d_ij
    
    n_pairs = 0
    for i in range(n):
        for j in range(i + 1, n):
            n_pairs += 1
            
    A_ub = np.zeros((n_pairs + n, n))
    b_ub = np.zeros(n_pairs + n)
    
    idx = 0
    for i in range(n):
        # Boundary constraints
        # r_i <= min(x_i, 1-x_i, y_i, 1-y_i)
        x, y = centers[i]
        max_r_boundary = min(x, 1 - x, y, 1 - y)
        
        # Constraint: 1 * r_i <= max_r_boundary
        A_ub[n_pairs, i] = 1.0
        b_ub[n_pairs] = max_r_boundary
        
        for j in range(i + 1, n):
            # Constraint: r_i + r_j <= dist(i, j)
            d = dists[i, j]
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = d
            idx += 1
            
    # Bounds for radii: r_i >= 0
    bounds = [(0, None) for _ in range(n)]
    
    # Solve LP
    # method='highs' is efficient
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        radii = res.x
        return radii, -res.fun
    else:
        # Fallback if LP fails (shouldn't happen for valid centers)
        return np.zeros(n), 0.0

def objective_function(center_flat):
    """
    Objective function for differential_evolution.
    Input is a flat array of centers.
    """
    n = 26
    centers = center_flat.reshape((n, 2))
    
    # Clip centers to be strictly inside to avoid boundary issues, 
    # though LP handles boundary constraints. 
    # Keeping them inside [0,1] is good practice.
    # However, LP handles r <= dist_to_boundary, so even if center is on edge, r=0.
    # Differential evolution might explore outside [0,1]. 
    # We can penalize or clip. Clipping centers to [0,1] is safe because 
    # moving a center closer to boundary only reduces feasible radius.
    # Wait, if center is outside, dist to boundary is negative? 
    # min(x, 1-x) would be negative.
    # LP constraint r <= negative number with r>=0 implies r=0.
    # So it's safe.
    
    radii, sum_r = solve_radii_lp(centers)
    return -sum_r  # Minimize negative sum
