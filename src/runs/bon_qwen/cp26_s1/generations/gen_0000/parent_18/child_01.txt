from scipy.optimize import minimize


def compute_penalty(centers, radii, lam, n):
    """Compute constraint violation penalty."""
    penalty = 0.0

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        penalty += lam * max(0, r - x)**2
        penalty += lam * max(0, r - (1 - x))**2
        penalty += lam * max(0, r - y)**2
        penalty += lam * max(0, r - (1 - y))**2
        penalty += lam * max(0, -r)**2
    
    # Overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx**2 + dy**2)
            overlap = radii[i] + radii[j] - dist
            if overlap > 0:
                penalty += lam * overlap**2
    return penalty

    return centers, radii
