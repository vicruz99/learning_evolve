import numpy as np
from scipy.optimize import minimize


def initialize_hexagonal(n):
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.05

    # Rows: 6, 5, 6, 5, 4 = 26 circles
    row_counts = [6, 5, 6, 5, 4]
    idx = 0
    num_rows = len(row_counts)
    y_spacing = 1.0 / (num_rows + 1)

    for row_i, count in enumerate(row_counts):
        y = (row_i + 1) * y_spacing
        x_spacing = 1.0 / (count + 1)
        offset = 0.5 * x_spacing if row_i % 2 == 1 else 0
        for col_i in range(count):
            if idx < n:
                centers[idx, 0] = (col_i + 1) * x_spacing + offset
                centers[idx, 1] = y
                idx += 1

    return centers, radii


def initialize_hexagonal_perturbed(n, seed):
    np.random.seed(seed)
    centers, radii = initialize_hexagonal(n)

    # Add small random perturbation
    perturbation = np.random.randn(n, 2) * 0.02
    centers = centers + perturbation
    centers[:, 0] = np.clip(centers[:, 0], 0.01, 0.99)
    centers[:, 1] = np.clip(centers[:, 1], 0.01, 0.99)

    return centers, radii


def force_based_relaxation(centers, radii, n):
    centers = centers.copy()
    radii = radii.copy()

    max_iter = 8000
    dt = 0.008
    base_growth = 1.00008

    for iteration in range(max_iter):
        forces = np.zeros((n, 2))
        max_overlap = 0.0

        # Inter-circle repulsion
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist_sq = dx * dx + dy * dy
                dist = np.sqrt(dist_sq) if dist_sq > 1e-15 else 1e-15
                target_dist = radii[i] + radii[j]

                if dist < target_dist:
                    overlap = target_dist - dist
                    max_overlap = max(max_overlap, overlap)
                    force_mag = overlap * 50.0 / (1.0 + overlap * 10.0)
                    inv_dist = 1.0 / dist
                    fx = force_mag * dx * inv_dist
                    fy = force_mag * dy * inv_dist
                    forces[i, 0] += fx
                    forces[i, 1] += fy
                    forces[j, 0] -= fx
                    forces[j, 1] -= fy

        # Wall repulsion (strong forces to keep circles inside)
        for i in range(n):
            # Left wall
            pen = radii[i] - centers[i, 0]
            if pen > 0:
                forces[i, 0] += pen * 150.0
            # Right wall
            pen = centers[i, 0] + radii[i] - 1.0
            if pen > 0:
                forces[i, 0] -= pen * 150.0
            # Bottom wall
            pen = radii[i] - centers[i, 1]
            if pen > 0:
                forces[i, 1] += pen * 150.0
            # Top wall
            pen = centers[i, 1] + radii[i] - 1.0
            if pen > 0:
                forces[i, 1] -= pen * 150.0

        # Adaptive time step based on overlap
        effective_dt = dt * (1.0 / (1.0 + max_overlap * 200.0))
        centers[:, 0] += forces[:, 0] * effective_dt
        centers[:, 1] += forces[:, 1] * effective_dt

        # Hard boundary enforcement
        for i in range(n):
            centers[i, 0] = max(radii[i], min(1.0 - radii[i], centers[i, 0]))
            centers[i, 1] = max(radii[i], min(1.0 - radii[i], centers[i, 1]))

        # Gradually grow radii (slow down towards end)
        progress = iteration / max_iter
        growth = base_growth * (1.0 - progress * 0.7)
        radii *= growth

    return centers, radii


def objective_func(x, n):
    radii = x[2 * n:]
    return -np.sum(radii)


def boundary_x_min(x, i, n):
    return x[2 * i] - x[2 * n + i]


def boundary_x_max(x, i, n):
    return 1.0 - x[2 * i] - x[2 * n + i]


def boundary_y_min(x, i, n):
    return x[2 * i + 1] - x[2 * n + i]


def boundary_y_max(x, i, n):
    return 1.0 - x[2 * i + 1] - x[2 * n + i]


def nonoverlap_constraint(x, i, j, n):
    dx = x[2 * i] - x[2 * j]
    dy = x[2 * i + 1] - x[2 * j + 1]
    dist = np.sqrt(dx * dx + dy * dy)
    return dist - x[2 * n + i] - x[2 * n + j]


def scipy_optimize(centers, radii, n):
    x0 = np.concatenate([centers.flatten(), radii])

    bounds = []
    for _ in range(n):
        bounds.append((0.0, 1.0))
        bounds.append((0.0, 1.0))
        bounds.append((1e-8, 0.5))

    constraints = []
    for i in range(n):
        constraints.append({'type': 'ineq', 'fun': boundary_x_min, 'args': (i, n)})
        constraints.append({'type': 'ineq', 'fun': boundary_x_max, 'args': (i, n)})
        constraints.append({'type': 'ineq', 'fun': boundary_y_min, 'args': (i, n)})
        constraints.append({'type': 'ineq', 'fun': boundary_y_max, 'args': (i, n)})

    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({'type': 'ineq', 'fun': nonoverlap_constraint, 'args': (i, j, n)})

    result = minimize(
        objective_func, x0, args=(n,),
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 1500, 'ftol': 1e-15, 'disp': False}
    )

    centers_out = result.x[:2 * n].reshape(n, 2)
    radii_out = result.x[2 * n:]
    return centers_out, radii_out
