def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    n = centers.shape[0]
    if np.isnan(centers).any():
        return False
    if np.isnan(radii).any():
        return False
    
    for i in range(n):
        if radii[i] < -1e-9: # Allow tiny negative due to float error
            return False
        
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-9 or x + r > 1 + 1e-9 or y - r < -1e-9 or y + r > 1 + 1e-9:
            return False

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-9:
                return False
    return True

def solve_radii_lp(centers):
    """
    Given fixed centers, solve LP to maximize sum of radii.
    Returns radii and dual values for constraints.
    """
    n = centers.shape[0]
    r = cp.Variable(n, nonneg=True)
    
    # Objective: Maximize sum of radii
    objective = cp.Maximize(cp.sum(r))
    
    constraints = []

    dists = np.zeros((n, n))

        for j in range(i + 1, n):
            dists[i, j] = np.linalg.norm(centers[i] - centers[j])
            dists[j, i] = dists[i, j]

        x, y = centers[i]
        boundary_limits[i] = min(x, 1 - x, y, 1 - y)

    # Constraints
    # 1. Boundary constraints: r_i <= limit_i
    # We track duals for these to compute gradient w.r.t centers
    # Actually, for gradient ascent on centers, we need duals of r_i <= x_i etc.
    # But cvxpy doesn't easily expose duals for variable bounds or simple linear constraints 
    # in the same way as inequality constraints.
    # We can formulate r_i <= limit_i as a linear constraint.
    
    # Let's use explicit constraints to get duals.
    # r_i + 0 <= limit_i
    # But limit_i depends on center.
    
    # To get gradients, we rely on the envelope theorem.
    # We need duals for:
    # 1. r_i + r_j <= d_ij
    # 2. r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    
    # Let's add them as explicit constraints.
    duals_overlap = []
    duals_boundary = []
    
    # Pairwise constraints
    for i in range(n):
        for j in range(i + 1, n):
            const = cp.Constraint(r[i] + r[j] <= dists[i, j])
            constraints.append(const)
            duals_overlap.append((i, j, const))

    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        # r_i <= x_i
        const1 = cp.Constraint(r[i] <= x)
        constraints.append(const1)
        duals_boundary.append((i, 'x', const1))
        
        # r_i <= 1 - x_i  => r_i + x_i <= 1
        const2 = cp.Constraint(r[i] + x <= 1) # Note: x is param, treated as const in LP
        constraints.append(const2)
        duals_boundary.append((i, '1-x', const2))
        
        # r_i <= y_i
        const3 = cp.Constraint(r[i] <= y)
        constraints.append(const3)
        duals_boundary.append((i, 'y', const3))
        
        # r_i <= 1 - y_i
        const4 = cp.Constraint(r[i] + y <= 1)
        constraints.append(const4)
        duals_boundary.append((i, '1-y', const4))

    problem = cp.Problem(objective, constraints)
    try:
        problem.solve(solver=cp.ECOS, verbose=False) # ECOS is good for LP
    except:
        # Fallback
        problem.solve(solver=cp.SCS, verbose=False)
        
    if problem.status not in ["optimal", "optimal_inaccurate"]:
        return None, None, 0
        
    radii = r.value
    total_sum = cp.sum(r).value
    
    # Extract duals
    # Dual variable is const.dual_value
    # For constraint f(x) <= 0, dual is >= 0.
    # Our constraints are linear.
    
    # Map duals
    dual_overlap = {}
    for (i, j, const) in duals_overlap:
        dual_overlap[(i, j)] = const.dual_value
        
    dual_boundary = {}
    for (i, type_str, const) in duals_boundary:
        dual_boundary[(i, type_str)] = const.dual_value
        
    return radii, {'overlap': dual_overlap, 'boundary': dual_boundary}, total_sum

                    grad_centers[j] -= lam * direction # Symmetric for j

        # r_i <= x_i. Dual mu_x.
        # If we increase x_i, constraint relaxes.

            grad_centers[i, 0] += dx - d1x
