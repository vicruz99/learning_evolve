import numpy as np
from scipy.optimize import minimize

def obj_equal_r(v):
    """Objective: maximize the common radius r (minimize -r)"""
    return -v[-1]

def constr_bound_x(v, i):
    """Constraint: x_i >= r"""
    return v[2*i] - v[-1]

def constr_bound_x1(v, i):
    """Constraint: x_i + r <= 1"""
    return 1.0 - v[2*i] - v[-1]

def constr_bound_y(v, i):
    """Constraint: y_i >= r"""
    return v[2*i+1] - v[-1]

def constr_bound_y1(v, i):
    """Constraint: y_i + r <= 1"""
    return 1.0 - v[2*i+1] - v[-1]

def constr_overlap(v, i, j):
    """Constraint: distance(i, j) >= 2r"""
    xi, yi = v[2*i], v[2*i+1]
    xj, yj = v[2*j], v[2*j+1]
    r = v[-1]
    return (xi - xj)**2 + (yi - yj)**2 - (2*r)**2

def run_packing():
    np.random.seed(42)
    n = 26
    
    # --- Phase 1: Force-Directed Simulation ---
    # Initialize in a hexagonal pattern for high initial density
    centers = np.zeros((n, 2))
    idx = 0
    dy = 0.13
    dx = dy * np.sqrt(3)
    for row in range(12):
        y = 0.1 + row * dy
        if y > 0.9:
            break
        for col in range(12):
            offset = (dx / 2) if row % 2 else 0
            x = 0.1 + col * dx + offset
            if x > 0.9:
                break
            centers[idx] = [x, y]
            idx += 1
            if idx == n:
                break
        if idx == n:
            break
            
    sim_steps = 6000
    r_start = 0.04
    r_growth = 0.00005
    dt = 0.03
    k_rep = 8.0
    k_bound = 15.0
    
    for step in range(sim_steps):
        r_sim = r_start + step * r_growth
        forces = np.zeros((n, 2))
        
        # Pairwise repulsion
        for i in range(n):
            for j in range(i + 1, n):
                diff = centers[i] - centers[j]
                dist_sq = np.dot(diff, diff)
                if dist_sq < 1e-12:
                    continue
                dist = np.sqrt(dist_sq)
                if dist < 2 * r_sim:
                    repulse_mag = k_rep * (2 * r_sim - dist)
                    forces[i] += diff * repulse_mag / dist
                    forces[j] -= diff * repulse_mag / dist
                    
        # Boundary repulsion
        for i in range(n):
            if centers[i, 0] < r_sim:
                forces[i, 0] += (r_sim - centers[i, 0]) * k_bound
            elif centers[i, 0] > 1 - r_sim:
                forces[i, 0] -= (centers[i, 0] - (1 - r_sim)) * k_bound
            if centers[i, 1] < r_sim:
                forces[i, 1] += (r_sim - centers[i, 1]) * k_bound
            elif centers[i, 1] > 1 - r_sim:
                forces[i, 1] -= (centers[i, 1] - (1 - r_sim)) * k_bound
                
        centers += forces * dt
        centers = np.clip(centers, 0.0, 1.0)
        
    # --- Phase 2: SLSQP Local Optimization ---
    v0 = np.zeros(2 * n + 1)
    v0[:2*n] = centers.flatten()
    v0[-1] = r_sim
    
    cons = []
    for i in range(n):
        cons.append({'type': 'ineq', 'fun': constr_bound_x, 'args': (i,)})
        cons.append({'type': 'ineq', 'fun': constr_bound_x1, 'args': (i,)})
        cons.append({'type': 'ineq', 'fun': constr_bound_y, 'args': (i,)})
        cons.append({'type': 'ineq', 'fun': constr_bound_y1, 'args': (i,)})
    for i in range(n):
        for j in range(i + 1, n):
            cons.append({'type': 'ineq', 'fun': constr_overlap, 'args': (i, j)})
            
    bounds = [(0.0, 1.0) for _ in range(2 * n)]
    bounds.append((0.0, 0.5))
    
    res = minimize(obj_equal_r, v0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': 800, 'ftol': 1e-11})
                   
    best_r = res.x[-1]
    best_centers = res.x[:2*n].reshape(n, 2)
    
    # Fallback to simulation result if optimizer underperforms
    if best_r < r_sim * 0.95:
        best_r = r_sim
        best_centers = centers
        
    # Tiny shrink for numerical safety in validation
    best_r = max(0.0, best_r - 1e-10)
    
    radii = np.full(n, best_r)
    return best_centers, radii, best_r * n
