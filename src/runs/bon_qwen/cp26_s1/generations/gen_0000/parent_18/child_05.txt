def objective_and_constraints(centers_flat, n):
    """
    Computes negative sum of radii (to minimize) and constraints.
    Variables: [x1, y1, r1, x2, y2, r2, ...]
    """
    centers = centers_flat[:2*n].reshape(n, 2)
    radii = centers_flat[2*n:]
    
    # Objective: minimize -sum(radii)
    obj = -np.sum(radii)
    
    # We will handle constraints via a penalty in the loss function or bounds
    # But for scipy.minimize with SLSQP, we can define constraints.
    # However, defining 26*26 constraints is heavy.
    # Let's use a penalty method in the objective for robustness.
    
    penalty = 0.0
    alpha = 1000.0 # Penalty weight
    
    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        
        # x - r >= 0  => r - x <= 0
        if r - x > 0: penalty += alpha * (r - x)**2
        # x + r <= 1  => x + r - 1 <= 0
        if x + r - 1 > 0: penalty += alpha * (x + r - 1)**2
        # y - r >= 0
        if r - y > 0: penalty += alpha * (r - y)**2
        # y + r <= 1
        if y + r - 1 > 0: penalty += alpha * (y + r - 1)**2
        
        # r >= 0
        if r < 0: penalty += alpha * r**2

    # Overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            sum_r = radii[i] + radii[j]
            if dist < sum_r:
                penalty += alpha * (sum_r - dist)**2
                
    return obj + penalty
