def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square

    Args:
        centers: np.array of shape (n, 2) with (x, y) coordinates
        radii: np.array of shape (n) with radius of each circle

    Returns:
        True if valid, False otherwise
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))

                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True

def objective_func(vars, n):
    """
    Objective function to minimize: -sum(radii) + penalty
    vars: [x1, y1, r1, x2, y2, r2, ...]
    """
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    for i in range(n):
        centers[i] = vars[3*i : 3*i + 2]
        radii[i] = vars[3*i + 2]
    
    # Objective: Maximize sum of radii -> Minimize -sum(radii)
    cost = -np.sum(radii)
    
    # Penalty for boundary violations
    # x - r >= 0 => r - x <= 0
    # x + r <= 1 => x + r - 1 <= 0
    # y - r >= 0 => r - y <= 0
    # y + r <= 1 => y + r - 1 <= 0
    boundary_penalties = np.maximum(0, radii - centers[:, 0]) + \
                         np.maximum(0, centers[:, 0] + radii - 1) + \
                         np.maximum(0, radii - centers[:, 1]) + \
                         np.maximum(0, centers[:, 1] + radii - 1)
    cost += 100.0 * np.sum(boundary_penalties ** 2)
    
    # Penalty for overlaps
    # dist >= r1 + r2 => r1 + r2 - dist <= 0
    # Vectorized pairwise distance calculation
    # centers: (n, 2)
    # diff: (n, n, 2)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.linalg.norm(diff, axis=2)
    
    # r_sum: (n, n)
    r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    
    # Overlap amount (positive if overlapping)
    overlap = r_sum - dists
    
    # Only consider upper triangle to avoid double counting and self

    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    overlap_vals = overlap[mask]
    
    # Penalty: sum of squared positive overlaps
    positive_overlap = np.maximum(0, overlap_vals)
    cost += 100.0 * np.sum(positive_overlap ** 2)
    
    return cost
