def init_positions(n, rng):
    centers = np.zeros((n, 2))
    idx = 0

    # Hexagonal lattice with random offset
    spacing = 0.2
    offset_x = rng.uniform(0.02, 0.1)
    offset_y = rng.uniform(0.02, 0.1)

    row = 0
    while idx < n:
        for col in range(12):
            if idx >= n:
                break
            x = col * spacing + (row % 2) * spacing / 2.0 + offset_x + 0.04
            y = row * spacing * np.sqrt(3.0) / 2.0 + offset_y + 0.04

            if x < 1.0 - 0.03 and y < 1.0 - 0.03:
                centers[idx] = [x, y]
                idx += 1
        row += 1

    return centers


def compute_repulsion_forces(centers, radii, n):
    forces = np.zeros_like(centers)

    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[j, 0] - centers[i, 0]
            dy = centers[j, 1] - centers[i, 1]
            dist_sq = dx * dx + dy * dy
            dist = np.sqrt(dist_sq) if dist_sq > 1e-14 else 1e-7

            min_dist = radii[i] + radii[j]
            if dist < min_dist:
                overlap = min_dist - dist
                if dist > 1e-8:
                    fx = overlap * dx / dist
                    fy = overlap * dy / dist
                    forces[i, 0] -= fx
                    forces[i, 1] -= fy
                    forces[j, 0] += fx
                    forces[j, 1] += fy

    return forces


def enforce_boundaries(centers, radii, n):
    for i in range(n):
        centers[i, 0] = max(radii[i], min(1.0 - radii[i], centers[i, 0]))
        centers[i, 1] = max(radii[i], min(1.0 - radii[i], centers[i, 1]))


def grow_independent(centers, radii, n):
    """Let each circle grow to its maximum possible radius."""
    for i in range(n):
        max_r = min(
            centers[i, 0],
            1.0 - centers[i, 0],
            centers[i, 1],
            1.0 - centers[i, 1]
        )
        for j in range(n):
            if i != j:
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy + 1e-14)
                max_r = min(max_r, dist - radii[j])
        radii[i] = max(max_r, 0.0) - 1e-11


def final_refinement(centers, radii, n):
    """Additional refinement after selecting best trial."""
    for _ in range(15):
        grow_independent(centers, radii, n)
        lr = 0.001
        for _ in range(1000):
            forces = compute_repulsion_forces(centers, radii, n)
            centers += forces * lr
            enforce_boundaries(centers, radii, n)
        grow_independent(centers, radii, n)

    return centers, radii
