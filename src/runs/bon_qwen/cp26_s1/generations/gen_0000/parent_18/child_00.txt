import numpy as np

def compute_min_dist(pts):
    """Compute the minimum distance between any pair of points or any point to the boundary."""
    n = pts.shape[0]
    min_d = 2.0
    for i in range(n):
        xi, yi = pts[i]
        d = min(xi, 1.0 - xi, yi, 1.0 - yi)
        if d < min_d:
            min_d = d
        for j in range(i + 1, n):
            dx = pts[i, 0] - pts[j, 0]
            dy = pts[i, 1] - pts[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            if dist < min_d:
                min_d = dist
    return min_d
