from scipy.optimize import minimize

def obj_func(vars):

def cons_boundary(vars):
    """Boundary constraints: circles must stay within [0,1]x[0,1]"""
    c = vars[:52].reshape((26, 2))
    r = vars[52:]
    return np.concatenate([
        c[:, 0] - r,
        1.0 - c[:, 0] - r,
        c[:, 1] - r,
        1.0 - c[:, 1] - r
    ])

    c = vars[:52].reshape((26, 2))
    r = vars[52:]
