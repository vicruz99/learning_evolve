import numpy as np
from scipy.optimize import linprog


def compute_optimal_radii(centers):
    """Compute optimal radii for given centers using linear programming."""
    n = centers.shape[0]
    
    # Objective: maximize sum of radii (minimize negative sum)
    c = -np.ones(n)
    
    A_rows = []
    b_vals = []
    
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        x, y = centers[i]
        for bound in [x, 1 - x, y, 1 - y]:
            row = np.zeros(n)
            row[i] = 1
            A_rows.append(row)
            b_vals.append(bound)
    
    # Pairwise non-overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            row = np.zeros(n)
            row[i] = 1
            row[j] = 1
            A_rows.append(row)
            b_vals.append(dist)
    
    A = np.array(A_rows)
    b = np.array(b_vals)
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    
    if result.success:
        return result.x
    else:
        # Fallback: compute simple radii
        radii = np.full(n, 0.05)
        for i in range(n):
            x, y = centers[i]
            radii[i] = min(x, 1 - x, y, 1 - y, 0.1)
        return radii


def generate_hexagonal_config(n, rows_per_side, row_counts):
    """Generate a hexagonal-like initial configuration."""
    centers = np.zeros((n, 2))
    idx = 0
    
    y_spacing = 1.0 / (rows_per_side + 1)
    
    for row_idx, count in enumerate(row_counts):
        if count == 0:
            continue
        y = (row_idx + 1) * y_spacing
        
        if row_idx % 2 == 0:
            x_start = 0.08
        else:
            x_start = 0.08 + 0.5 * (1.0 - 2 * 0.08) / max(count, 1)
        
        if count > 1:
            x_spacing = (1.0 - 2 * 0.08) / (count - 1)
        else:
            x_spacing = 0.5
        
        for col in range(count):
            if idx >= n:
                break
            if row_idx % 2 == 0:
                x = x_start + col * x_spacing
            else:
                x = x_start + col * x_spacing
            centers[idx] = [x, y]
            idx += 1
    
    # Fill remaining positions
    while idx < n:
        centers[idx] = [0.5, 0.5]
        idx += 1
    
    return centers

def coordinate_descent(centers, n, max_iters=80, step_sizes=None):
    """Optimize positions using coordinate descent."""
    if step_sizes is None:
        step_sizes = [0.02, 0.01, 0.005, 0.002, 0.001]
    
    centers = centers.copy()
    radii = compute_optimal_radii(centers)
    best_sum = np.sum(radii)
    
    for step_size in step_sizes:
        for iteration in range(max_iters):
            improved = False
            
            for i in range(n):
                current_pos = centers[i].copy()
                current_sum = best_sum
                
                # Try perturbations in a grid pattern
                perturbations = []
                for dx_frac in [-1, 0, 1]:
                    for dy_frac in [-1, 0, 1]:
                        if dx_frac == 0 and dy_frac == 0:
                            continue
                        perturbations.append([dx_frac * step_size, dy_frac * step_size])
                
                # Also try random perturbations
                for _ in range(4):
                    angle = np.random.uniform(0, 2 * np.pi)
                    dist = np.random.uniform(0, step_size)
                    perturbations.append([dist * np.cos(angle), dist * np.sin(angle)])
                
                for dx, dy in perturbations:
                    new_pos = current_pos + np.array([dx, dy])
                    new_pos = np.clip(new_pos, 0.001, 0.999)
                    
                    temp_centers = centers.copy()
                    temp_centers[i] = new_pos
                    
                    new_radii = compute_optimal_radii(temp_centers)
                    new_sum = np.sum(new_radii)
                    
                    if new_sum > current_sum:
                        current_sum = new_sum
                        current_pos = new_pos.copy()
                        improved = True
                
                centers[i] = current_pos
            
            radii = compute_optimal_radii(centers)
            new_sum = np.sum(radii)
            
            if new_sum > best_sum:
                best_sum = new_sum
            elif not improved:
                break
    
    radii = compute_optimal_radii(centers)
    return centers, radii, np.sum(radii)


def validate_and_fix(centers, radii, n):
    """Validate and fix any issues with the packing."""
    # Ensure centers are in [0, 1]
    centers = np.clip(centers, 0, 1)
    
    # Ensure radii are non-negative
    radii = np.maximum(radii, 0)
    
    # Check and fix overlaps by reducing radii
    for _ in range(10):
        fixed = False
        for i in range(n):
            x, y = centers[i]
            # Boundary constraints
            radii[i] = min(radii[i], x, 1 - x, y, 1 - y)
            
            for j in range(n):
                if i == j:
                    continue
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                max_r = (dist - radii[j]) / 2 if dist > radii[j] else 0
                if radii[i] > max_r:
                    radii[i] = max(0, max_r)
                    fixed = True
        if not fixed:
            break
    
    return centers, radii
