import numpy as np
from scipy.optimize import minimize

def calculate_distances(centers):
    """
    Calculates pairwise squared distances between centers.
    centers: numpy array of shape (n, 2)
    Returns: numpy array of shape (n, n) with squared distances
    """
    # Using broadcasting to compute all pairwise distances
    # diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    # dist_sq = np.sum(diff**2, axis=2)
    
    # More memory efficient way for large n, though n=26 is small
    n = centers.shape[0]
    dist_sq = np.zeros((n, n))
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            d_sq = dx*dx + dy*dy
            dist_sq[i, j] = d_sq
            dist_sq[j, i] = d_sq
    return dist_sq

def objective_function(vars, n_circles):
    """
    Objective function to minimize: -sum(radii) + penalty for violations.
    vars: array of shape (3 * n_circles)
    Layout: [x1, x2, ..., xn, y1, y2, ..., yn, r1, r2, ..., rn]
    """
    # Unpack variables
    x = vars[0:n_circles]
    y = vars[n_circles:2*n_circles]
    r = vars[2*n_circles:3*n_circles]
    
    # Sum of radii (we want to maximize this, so minimize negative)
    sum_r = np.sum(r)
    loss = -sum_r
    
    # Penalty parameters
    penalty_weight = 1000.0
    
    # 1. Boundary constraints
    # Circle i must satisfy: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    # Violation is max(0, r_i - x_i), etc.
    # We square the violation to make it smooth and penalize heavily.
    
    # x >= r  => r - x <= 0. Violation: max(0, r - x)
    viol_x_left = np.maximum(0, r - x)
    # x <= 1 - r => r + x <= 1 => r - (1-x) <= 0. Violation: max(0, r - 1 + x)
    viol_x_right = np.maximum(0, r - 1 + x)
    # y >= r
    viol_y_bottom = np.maximum(0, r - y)
    # y <= 1 - r
    viol_y_top = np.maximum(0, r - 1 + y)
    
    boundary_penalty = (np.sum(viol_x_left**2) + 
                        np.sum(viol_x_right**2) + 
                        np.sum(viol_y_bottom**2) + 
                        np.sum(viol_y_top**2))
    
    loss += penalty_weight * boundary_penalty
    
    # 2. Overlap constraints
    # For all i < j: dist(i, j) >= r_i + r_j
    # Violation: max(0, r_i + r_j - dist(i, j))
    
    # Construct centers array for distance calculation
    centers = np.column_stack((x, y))
    dist_sq = calculate_distances(centers)
    # Take square root, but avoid sqrt if possible? No, constraint is linear in r but dist is sqrt.
    # dist = r_i + r_j is correct.
    
    # We only need upper triangle
    overlap_penalty = 0.0
    for i in range(n_circles):
        # radii sum with all j > i
        r_sum_i = r[i] + r[i+1:] # vector of r_i + r_j for j > i
        
        # distances to all j > i
        dists_i = np.sqrt(dist_sq[i, i+1:])
        
        # violation
        viol = np.maximum(0, r_sum_i - dists_i)
        overlap_penalty += np.sum(viol**2)
        
        # Also need to check j < i? No, loop covers pairs once.
        
    loss += penalty_weight * overlap_penalty
    
    return loss
