import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def compute_objective(vars_):
    # We maximize sum of radii, so we minimize the negative sum
    return -np.sum(vars_[2::3])

def compute_constraints(vars_):
    # Total constraints: 4 boundary per circle + pairwise non-overlap
    n_constraints = 4 * N_CIRCLES + N_CIRCLES * (N_CIRCLES - 1) // 2
    cons = np.empty(n_constraints)
    idx = 0
    
    # Boundary constraints: x >= r, x + r <= 1, y >= r, y + r <= 1
    for i in range(N_CIRCLES):
        x, y, r = vars_[3*i], vars_[3*i+1], vars_[3*i+2]
        cons[idx] = x - r
        cons[idx+1] = 1.0 - x - r
        cons[idx+2] = y - r
        cons[idx+3] = 1.0 - y - r
        idx += 4
        
    # Non-overlap constraints: dist^2 >= (r_i + r_j)^2
    for i in range(N_CIRCLES):
        xi, yi, ri = vars_[3*i], vars_[3*i+1], vars_[3*i+2]
        for j in range(i+1, N_CIRCLES):
            xj, yj, rj = vars_[3*j], vars_[3*j+1], vars_[3*j+2]
            dx = xi - xj
            dy = yi - yj
            dist_sq = dx*dx + dy*dy
            sum_r = ri + rj
            cons[idx] = dist_sq - sum_r*sum_r
            idx += 1
    return cons

def generate_initial_config(seed):
    np.random.seed(seed)
    centers = []
    r_init = 0.09
    # Hexagonal-like row distribution for 26 circles
    row_counts = [5, 6, 5, 6, 4]
    
    for row_idx, count in enumerate(row_counts):
        # Vertical spacing based on hexagonal packing
        y = r_init + row_idx * np.sqrt(3) * r_init
        y += np.random.uniform(-0.03, 0.03)
        
        # Evenly space centers horizontally
        if count > 1:
            x_coords = np.linspace(r_init, 1 - r_init, count)
        else:
            x_coords = np.array([0.5])
            
        # Shift odd rows to create staggered/hexagonal pattern
        if row_idx % 2 == 1:
            spacing = (1 - 2*r_init) / (count - 1) if count > 1 else 0
            x_coords = x_coords + spacing / 2
            
        # Add random perturbation to break symmetry and aid optimization
        x_coords += np.random.uniform(-0.02, 0.02, count)
        centers.extend(zip(x_coords, [y]*count))
        
    centers = np.array(centers)
    radii = np.full(N_CIRCLES, r_init)
    
    # Flatten to 1D array [x0, y0, r0, x1, y1, r1, ...]
    x0 = np.zeros(3 * N_CIRCLES)
    for i in range(N_CIRCLES):
        x0[3*i] = np.clip(centers[i, 0], 0.001, 0.999)
        x0[3*i+1] = np.clip(centers[i, 1], 0.001, 0.999)
        x0[3*i+2] = radii[i]
    return x0

def run_packing():
    # Bounds for x, y in [0, 1] and r in [0, 0.5]
    bounds = [(0.0, 1.0)] * (2*N_CIRCLES) + [(0.0, 0.5)] * N_CIRCLES
    
    best_sum = -np.inf
    best_x = None
    
    # Run multiple restarts to escape local minima
    for trial in range(10):
        x0 = generate_initial_config(seed=trial * 42)
        try:
            res = minimize(
                compute_objective,
                x0,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'ineq', 'fun': compute_constraints},
                options={'maxiter': 5000, 'ftol': 1e-10}
            )
            current_sum = np.sum(res.x[2::3])
            if current_sum > best_sum:
                best_sum = current_sum
                best_x = res.x
        except Exception:
            continue
            
    # Fallback if all trials fail (highly unlikely)
    if best_x is None:
        best_x = generate_initial_config(seed=0)
        best_sum = np.sum(best_x[2::3])
        
    # Unpack results
    final_centers = np.zeros((N_CIRCLES, 2))
    final_radii = np.zeros(N_CIRCLES)
    for i in range(N_CIRCLES):
        final_centers[i, 0] = best_x[3*i]
        final_centers[i, 1] = best_x[3*i+1]
        final_radii[i] = best_x[3*i+2]
        
    return final_centers, final_radii, best_sum
