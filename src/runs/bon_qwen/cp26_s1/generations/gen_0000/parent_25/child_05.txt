import numpy as np
from scipy.optimize import minimize

def get_distance_matrix(centers):
    """Calculate pairwise Euclidean distances between centers."""
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    return np.sqrt(np.sum(diff ** 2, axis=2))

def hexagonal_grid(n_circles):
    """
    Generate initial centers and radii for a hexagonal packing.
    Attempts to pack n_circles into a hexagonal lattice within the unit square.
    """
    # Approximate rows based on sqrt(n)
    n_rows = int(np.ceil(np.sqrt(n_circles * 2 / np.sqrt(3))))
    
    # Try to fit circles in a hexagonal pattern
    # Estimate a radius that fits n_rows
    # Height constraint: 2*r + (n_rows-1)*r*sqrt(3) <= 1
    # r * (2 + (n_rows-1)*sqrt(3)) <= 1
    height_factor = 2 + (n_rows - 1) * np.sqrt(3)
    r_est = 1.0 / height_factor
    
    # Adjust radius based on width constraint for the row with most circles
    # This is a rough estimate; we will optimize later
    r = r_est * 0.95 
    
    centers = []
    for i in range(n_rows):
        # y coordinate
        y = r + i * r * np.sqrt(3)
        # Determine number of circles in this row
        # Width constraint: (n_cols - 1) * 2r + 2r <= 1 => n_cols <= 1/(2r)
        # Offset for odd/even rows
        offset = r if i % 2 == 1 else 0
        x_start = r + offset
        # Max x
        max_x = 1 - r
        
        current_x = x_start
        while current_x <= max_x:
            centers.append([current_x, y])
            current_x += 2 * r
            if len(centers) >= n_circles:
                break
        if len(centers) >= n_circles:
            break

        centers.append([0.5 + 0.1 * (np.random.rand() - 0.5), 
                        0.5 + 0.1 * (np.random.rand() - 0.5)])
        
    # If we have more (unlikely with loop break), trim
    centers = centers[:n_circles]
    
    radii = np.full(n_circles, r)
    return np.array(centers), radii

def objective(vars, n):
    """Objective function: minimize negative sum of radii."""
    radii = vars[2*n:]
    return -np.sum(radii)

def constraints_builder(n):
    """Build a list of constraints for the optimizer."""
    constraints = []
    
    # 1. Boundary constraints: x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
    # Flattened vars: [x1, x2, ..., xn, y1, ..., yn, r1, ..., rn]
    # Indices: 
    # x: 0 to n-1
    # y: n to 2n-1
    # r: 2n to 3n-1
    
    for i in range(n):
        idx_x = i
        idx_y = n + i
        idx_r = 2 * n + i
        
        # x - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, ix=idx_x, ir=idx_r: v[ix] - v[ir]
        })
        # 1 - x - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, ix=idx_x, ir=idx_r: 1.0 - v[ix] - v[ir]
        })
        # y - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, iy=idx_y, ir=idx_r: v[iy] - v[ir]
        })
        # 1 - y - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, iy=idx_y, ir=idx_r: 1.0 - v[iy] - v[ir]
        })
        
        # Non-negative radii
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, ir=idx_r: v[ir]
        })

    # 2. Non-overlap constraints: dist(i, j) >= r_i + r_j
    # (x_i - x_j)^2 + (y_i - y_j)^2 >= (r_i + r_j)^2
    for i in range(n):
        for j in range(i + 1, n):
            idx_xi, idx_yi, idx_ri = i, n + i, 2 * n + i
            idx_xj, idx_yj, idx_rj = j, n + j, 2 * n + j
            
            def non_overlap(v, xi=idx_xi, yi=idx_yi, ri=idx_ri, 
                                   xj=idx_xj, yj=idx_yj, rj=idx_rj):
                dx = v[xi] - v[xj]
                dy = v[yi] - v[yj]
                dist_sq = dx*dx + dy*dy
                r_sum = v[ri] + v[rj]
                return dist_sq - r_sum*r_sum

            constraints.append({
                'type': 'ineq',
                'fun': non_overlap
            })
            
    return constraints
