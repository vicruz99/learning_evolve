import numpy as np
import scipy.optimize as opt

def get_repulsive_force(p1, p2, r1, r2):
    """Calculate repulsive force between two overlapping circles."""
    diff = p1 - p2
    dist = np.linalg.norm(diff)
    if dist == 0:
        return np.array([1.0, 0.0]), 1.0 # Avoid division by zero
    if dist < r1 + r2:
        # Force magnitude proportional to overlap
        overlap = r1 + r2 - dist
        force_vec = diff / dist
        return force_vec, overlap
    return np.zeros(2), 0.0

def validate_and_fix(centers, radii):
    """Ensure constraints are met and fix minor violations."""
    n = len(radii)
    for i in range(n):
        r = radii[i]
        # Clip to boundary
        centers[i, 0] = np.clip(centers[i, 0], r, 1.0 - r)
        centers[i, 1] = np.clip(centers[i, 1], r, 1.0 - r)
    return centers

def objective_and_penalty(vars_array, n):
    """
    Objective function for optimization.
    We want to maximize sum of radii.
    Constraints: non-overlap and boundary.
    We minimize: -sum(radii) + penalty(overlaps) + penalty(boundaries)
    """
    centers = vars_array[:2*n].reshape(n, 2)
    radii = vars_array[2*n:]
    
    # Penalty parameters
    C = 1000.0 
    
    # 1. Boundary penalties
    # x - r >= 0  => r - x <= 0
    # x + r <= 1  => x + r - 1 <= 0
    # same for y
    
    penalty = 0.0
    
    # Left/Right boundaries
    for i in range(n):
        r = radii[i]
        x, y = centers[i]
        
        # Left: x >= r => r - x <= 0
        if r - x > 0:
            penalty += C * (r - x)**2
        # Right: x <= 1-r => x + r - 1 <= 0
        if x + r - 1 > 0:
            penalty += C * (x + r - 1)**2
        # Bottom: y >= r => r - y <= 0
        if r - y > 0:
            penalty += C * (r - y)**2
        # Top: y <= 1-r => y + r - 1 <= 0
        if y + r - 1 > 0:
            penalty += C * (y + r - 1)**2
            
    # 2. Overlap penalties
    # dist >= r_i + r_j => r_i + r_j - dist <= 0
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            sum_r = radii[i] + radii[j]
            if dist < sum_r:
                overlap = sum_r - dist
                penalty += C * overlap**2

    # Objective: maximize sum of radii -> minimize negative sum
    obj = -np.sum(radii) + penalty
    
    return obj

        growth_rate = 0.0005

        temp = 1.0
