import numpy as np
from scipy.optimize import minimize

def run_packing():
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    Returns centers (26, 2), radii (26,), and sum_radii.
    """
    n_circles = 26
    best_sum_radii = 0.0
    best_centers = None
    best_radii = None

    # Helper to generate hexagonal lattice points
    def generate_hex_points(num_points):
        points = []
        # Approximate number of rows/cols needed
        # Hexagonal packing density suggests roughly sqrt(N) items per dimension
        # For 26, maybe 5 or 6 rows.
        rows = 0
        cols = 0
        # Estimate
        side = np.ceil(np.sqrt(num_points * 2 / np.sqrt(3)))
        rows = int(side) + 1
        cols = int(side) + 1
        
        y = 0
        row_idx = 0
        while len(points) < num_points:
            x = 0
            shift = (0.5 if row_idx % 2 == 1 else 0)
            col_idx = 0
            while len(points) < num_points:
                points.append([x + shift, y])
                x += 1
                col_idx += 1
                # Adjust number of cols per row to fit shape
                if col_idx > cols + (1 if row_idx % 2 == 1 else 0):
                    break
            y += np.sqrt(3) / 2
            row_idx += 1
            if row_idx > rows + 2:
                break
        return np.array(points[:num_points])

    # Initial configuration generation
    def get_initial_config():
        # Generate a hexagonal pattern
        pts = generate_hex_points(n_circles)
        
        # Normalize to fit in [0, 1] x [0, 1] with some padding
        if len(pts) > 0:
            min_p = pts.min(axis=0)
            max_p = pts.max(axis=0)
            extent = max_p - min_p
            if extent[0] > 0 and extent[1] > 0:
                scale = 0.8 / max(extent) # Scale to fit in 0.8 range
                pts = (pts - min_p) * scale + (1 - scale * extent) / 2 # Center
            else:
                pts = np.random.rand(n_circles, 2) * 0.6 + 0.2
        
        # Add some random noise to break symmetry
        pts += np.random.randn(n_circles, 2) * 0.02
        pts = np.clip(pts, 0.05, 0.95)
        
        # Initial radii
        # Estimate based on spacing. If points are roughly uniform, 
        # density ~ n_circles. Area per circle ~ 1/n_circles. 
        # r ~ sqrt(1/(pi*n)) ~ 0.11. Let's start small.
        radii = np.full(n_circles, 0.02)
        
        return pts, radii

    def objective_with_penalty(params, penalty_weight):
        # params is 1D array of size 3*n
        # [x1, y1, r1, x2, y2, r2, ...]
        
        centers = np.zeros((n_circles, 2))
        radii = np.zeros(n_circles)
        
        for i in range(n_circles):
            centers[i, 0] = params[3 * i]
            centers[i, 1] = params[3 * i + 1]
            radii[i] = params[3 * i + 2]
            
        # Clip radii to be non-negative (though bounds handle this, for safety)
        radii = np.maximum(radii, 0.0)
        
        # Update params with clipped radii for return (optional but good practice)
        for i in range(n_circles):
            params[3 * i + 2] = radii[i]
            
        sum_radii = np.sum(radii)
        
        # Penalty terms
        penalty = 0.0
        
        # 1. Boundary constraints
        # Circle i must be inside [0,1]x[0,1]
        # r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
        for i in range(n_circles):
            x, y = centers[i]
            r = radii[i]
            
            if r > x: penalty += (r - x)**2
            if r > 1 - x: penalty += (r - (1 - x))**2
            if r > y: penalty += (r - y)**2
            if r > 1 - y: penalty += (r - (1 - y))**2
            
        # 2. Overlap constraints
        # dist(c_i, c_j) >= r_i + r_j
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                dist_sq = (centers[i, 0] - centers[j, 0])**2 + (centers[i, 1] - centers[j, 1])**2
                dist = np.sqrt(dist_sq)
                min_dist = radii[i] + radii[j]
                if dist < min_dist:
                    penetration = min_dist - dist
                    penalty += penetration**2
        
        return -sum_radii + penalty_weight * penalty

    # Bounds for variables
    # x, y in [0, 1], r in [0, 0.5]
    bounds = [(0, 1), (0, 1), (0, 0.5)] * n_circles

    num_restarts = 10
    penalty_weights = [100.0, 500.0, 1000.0, 5000.0]

            x0[3 * i + 1] = centers_init[i, 1]
            x0[3 * i + 2] = radii_init[i]
            
        current_params = x0
        
        # Sequential penalty optimization
        for weight in penalty_weights:
            # Define objective closure for current weight
            def obj(p):
                return objective_with_penalty(p, weight)
            
            # Optimize
            # Using L-BFGS-B for bound constraints
            result = minimize(obj, current_params, method='L-BFGS-B', bounds=bounds, 
                              options={'maxiter': 500, 'ftol': 1e-9, 'gtol': 1e-6})
            
            if result.success or result.nit == result.options['maxiter']:
                current_params = result.x
            # If failed, we still take the result as best guess
        
        # Extract result
        final_centers = np.zeros((n_circles, 2))
        final_radii = np.zeros(n_circles)
        for i in range(n_circles):
            final_centers[i, 0] = current_params[3 * i]
            final_centers[i, 1] = current_params[3 * i + 1]
            final_radii[i] = max(0, current_params[3 * i + 2])
            
        # Validate result strictly

                    is_valid = False
                    break
            
            if is_valid:
                for i in range(n_circles):
                    for j in range(i + 1, n_circles):
                        dist = np.sqrt(np.sum((final_centers[i] - final_centers[j])**2))
                        if dist < final_radii[i] + final_radii[j] - 1e-7:
                            is_valid = False
                            break
                    if not is_valid: break
        
        if is_valid:
            current_sum = np.sum(final_radii)
            if current_sum > best_sum_radii:
                best_sum_radii = current_sum
                best_centers = final_centers.copy()
                best_radii = final_radii.copy()

    # If no valid solution found (unlikely), return a fallback grid
    if best_centers is None:
        best_centers = np.zeros((n_circles, 2))
        best_radii = np.zeros(n_circles)

                    best_radii[idx] = 0.09
                    idx += 1
                else:
                    break
            if idx >= n_circles: break
        best_sum_radii = np.sum(best_radii)

    return best_centers, best_radii, best_sum_radii
