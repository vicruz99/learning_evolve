import numpy as np
import scipy.optimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    """
    N = 26
    
    # --- 1. Initialization ---
    # Create a hexagonal-like initialization to start with a dense packing.
    # We aim for an initial radius around 0.1.
    
    centers = np.zeros((N, 2))
    radii = np.zeros(N)
    
    # Hexagonal lattice parameters
    # Assume initial radius r0 = 0.09 to fit easily
    r0 = 0.09
    row_height = r0 * np.sqrt(3)
    
    idx = 0
    row = 0
    
    # Generate points in a hexagonal pattern
    # We iterate rows and fill them with circles
    y = r0
    while idx < N:
        # Determine x-offset for this row
        # Even rows (0, 2, ...) start at x=r0, odd rows (1, 3, ...) start at x=2*r0 (shifted)
        # Actually, standard hex packing offset is r0 (center to center horizontal dist is 2r0)
        # Wait, distance between (0,0) and (r0, sqrt(3)r0) is 2r0.
        # So x-shift is r0.
        
        x_offset = 0.0 if row % 2 == 0 else r0
        x = r0 + x_offset
        
        while x + r0 <= 1.0 + 1e-9:
            if idx < N:
                centers[idx] = [x, y]
                radii[idx] = r0
                idx += 1
            x += 2 * r0
        
        y += row_height
        row += 1
        
        # If we can't fit more in this row height, break or adjust?
        # With r0=0.09, we should fit 26 easily.
        # 5 circles in row (width 0.9), 4 circles in next.
        # 5+4+5+4+5+4 = 27 circles in 6 rows.
        # So we will fill 26.

    # If for some reason we didn't fill (unlikely), fill remaining randomly
    while idx < N:
        centers[idx] = [np.random.rand(), np.random.rand()]
        radii[idx] = 0.01
        idx += 1

    # --- 2. Optimization ---
    
    # Flatten parameters for scipy
    # Vector: [x1, y1, r1, x2, y2, r2, ...]
    x0 = np.zeros(3 * N)
    for i in range(N):
        x0[3*i] = centers[i, 0]
        x0[3*i+1] = centers[i, 1]
        x0[3*i+2] = radii[i]

    # Bounds for variables: x, y in [0, 1], r in [0, 0.5]
    bounds = []
    for _ in range(N):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 0.5)) # r

    # Objective function with penalty
    # We want to minimize -sum(radii) + penalties
    penalty_weight = 1000.0 # High weight to enforce constraints
    
    def objective(params):
        # Unpack
        c = params.reshape((N, 3))
        xs = c[:, 0]
        ys = c[:, 1]
        rs = c[:, 2]
        
        # Objective: maximize sum of radii -> minimize -sum
        obj = -np.sum(rs)
        
        # Penalty 1: Boundary violations
        # x - r >= 0  => r - x <= 0
        # 1 - x - r >= 0 => r + x - 1 <= 0
        # Same for y
        
        # Vectorized boundary penalties
        # Left: r - x
        p_left = np.maximum(0, rs - xs)
        # Right: r + x - 1
        p_right = np.maximum(0, rs + xs - 1.0)
        # Bottom: r - y
        p_bottom = np.maximum(0, rs - ys)
        # Top: r + y - 1
        p_top = np.maximum(0, rs + ys - 1.0)
        
        boundary_penalty = np.sum(p_left**2 + p_right**2 + p_bottom**2 + p_top**2)
        
        # Penalty 2: Overlaps
        # dist >= r_i + r_j
        # dist^2 >= (r_i + r_j)^2
        # We penalize if dist < r_i + r_j
        
        # Compute pairwise distances
        # Using broadcasting: (N, 1, 2) - (1, N, 2) -> (N, N, 2)
        # This might be memory heavy for large N, but N=26 is small.
        c_xy = c[:, :2] # (N, 2)
        diff = c_xy[:, np.newaxis, :] - c_xy[np.newaxis, :, :] # (N, N, 2)
        dist_sq = np.sum(diff**2, axis=2) # (N, N)
        dist = np.sqrt(np.maximum(dist_sq, 0)) # avoid nan
        
        # Radii sums matrix
        rs_sum = rs[:, np.newaxis] + rs[np.newaxis, :] # (N, N)
        
        # Overlap violation: r_i + r_j - dist
        violation = rs_sum - dist
        
        # Zero out diagonal and lower triangle (upper triangle is symmetric)
        # We only care about i < j
        np.fill_diagonal(violation, 0)
        # Keep only upper triangle
        violation = np.triu(violation, k=1)
        
        # Penalty is sum of squares of positive violations
        overlap_penalty = np.sum(np.maximum(0, violation)**2)
        
        return obj + penalty_weight * (boundary_penalty + overlap_penalty)

    # Run optimization
    # L-BFGS-B is suitable for box constraints
    res = scipy.optimize.minimize(
        objective, 
        x0, 
        method='L-BFGS-B', 
        bounds=bounds,
        options={'maxiter': 2000, 'ftol': 1e-12, 'gtol': 1e-8}
    )
    
    # Extract solution
    best_params = res.x
    best_centers = best_params.reshape((N, 3))[:, :2]
    best_radii = best_params.reshape((N, 3))[:, 2]
    
    # Post-processing: Ensure validity strictly if needed, 
    # though the penalty method usually keeps us valid enough.
    # Just in case, clip radii to be non-negative and centers to be within bounds
    best_radii = np.maximum(best_radii, 0)
    best_centers[:, 0] = np.clip(best_centers[:, 0], 0, 1)
    best_centers[:, 1] = np.clip(best_centers[:, 1], 0, 1)
    
    # If any radius is huge (invalid), clamp it (should not happen with penalty)
    # A circle of radius > 0.5 is impossible in unit square unless center is adjusted,
    # but bounds on r were 0.5.
    
    sum_radii = np.sum(best_radii)
    
    return best_centers, best_radii, sum_radii
