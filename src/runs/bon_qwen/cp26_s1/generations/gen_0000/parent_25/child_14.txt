import numpy as np
from scipy.optimize import minimize
import math


def compute_overlap_penalty(centers, radii):
    n = len(radii)
    penalty = 0.0
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = math.sqrt(dx * dx + dy * dy)
            overlap = radii[i] + radii[j] - dist
            if overlap > 0:
                penalty += overlap * overlap
    return penalty


def compute_boundary_penalty(centers, radii):
    n = len(radii)
    penalty = 0.0
    for i in range(n):
        x = centers[i, 0]
        y = centers[i, 1]
        r = radii[i]
        if x - r < 0:
            penalty += (x - r) * (x - r)
        if x + r > 1:
            penalty += (x + r - 1) * (x + r - 1)
        if y - r < 0:
            penalty += (y - r) * (y - r)
        if y + r > 1:
            penalty += (y + r - 1) * (y + r - 1)
    return penalty


def objective_fn(vars_array, n, penalty_weight):
    centers = vars_array[:2 * n].reshape(n, 2)
    radii = np.maximum(vars_array[2 * n:], 0)
    sum_radii = -np.sum(radii)
    overlap_p = compute_overlap_penalty(centers, radii)
    boundary_p = compute_boundary_penalty(centers, radii)
    return sum_radii + penalty_weight * (overlap_p + boundary_p)


class ObjWrapper:
    def __init__(self, n_val, pw_val):
        self.n_val = n_val
        self.pw_val = pw_val

    def __call__(self, vars_array):
        return objective_fn(vars_array, self.n_val, self.pw_val)


def init_hexagonal(n, r_init):
    centers = np.zeros((n, 2))
    radii = np.ones(n) * r_init
    idx = 0
    row = 0
    while idx < n:
        if row % 2 == 0:
            n_in_row = min(6, n - idx)
        else:
            n_in_row = min(5, n - idx)
        for col in range(n_in_row):
            if row % 2 == 0:
                x = r_init + col * 2 * r_init
            else:
                x = 2 * r_init + col * 2 * r_init
            y = r_init + row * r_init * math.sqrt(3)
            centers[idx, 0] = x
            centers[idx, 1] = y
            idx += 1
        row += 1
    return centers, radii


def init_square_grid(n, r_init):
    centers = np.zeros((n, 2))
    radii = np.ones(n) * r_init
    idx = 0
    for i in range(5):
        for j in range(5):
            if idx < n:
                centers[idx, 0] = r_init + i * 2 * r_init
                centers[idx, 1] = r_init + j * 2 * r_init
                idx += 1
    if idx < n:
        centers[idx, 0] = 0.5
        centers[idx, 1] = 0.5
        idx += 1
    return centers, radii

        row += 1
    return centers, radii

def optimize_packing(n, centers_init, radii_init, seed_val):
    np.random.seed(seed_val)
    x0 = np.concatenate([centers_init.flatten(), radii_init])
    
    bounds_list = []
    for i in range(2 * n):
        bounds_list.append((0, 1))
    for i in range(n):
        bounds_list.append((0, 0.5))
    
    x_opt = x0.copy()
    
    penalty_weights = [50, 200, 800, 3000, 10000, 50000]
    for pw in penalty_weights:
        wrapper = ObjWrapper(n, pw)
        result = minimize(wrapper, x_opt, method='L-BFGS-B', bounds=bounds_list,
                         options={'maxiter': 2000, 'ftol': 1e-18, 'gtol': 1e-12})
        x_opt = result.x.copy()
    
    centers_opt = x_opt[:2 * n].reshape(n, 2)
    radii_opt = np.maximum(x_opt[2 * n:], 1e-10)
    
    centers_opt, radii_opt = enforce_constraints(centers_opt, radii_opt, n)
    
    return centers_opt, radii_opt


def perturb_and_optimize(n, centers, radii, seed_val):

    centers_p = centers.copy() + np.random.randn(n, 2) * 0.005

    radii_p = np.maximum(radii_p, 0.001)
    
    for i in range(n):

        centers_p[i, 1] = np.clip(centers_p[i, 1], radii_p[i], 1 - radii_p[i])
    
    centers_opt, radii_opt = optimize_packing(n, centers_p, radii_p, seed_val)
    return centers_opt, radii_opt
