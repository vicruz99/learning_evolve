import numpy as np
from scipy.optimize import minimize
from functools import partial
import math
import random


def obj_func(x, n):
    radii = x[2*n:]
    return -np.sum(radii)


def constraint_x_min(x, n, i):
    return x[2*i] - x[2*n + i]


def constraint_x_max(x, n, i):
    return 1.0 - x[2*i] - x[2*n + i]


def constraint_y_min(x, n, i):
    return x[2*i + 1] - x[2*n + i]


def constraint_y_max(x, n, i):
    return 1.0 - x[2*i + 1] - x[2*n + i]


def constraint_overlap(x, n, i, j):
    dx = x[2*i] - x[2*j]
    dy = x[2*i+1] - x[2*j+1]
    dist_sq = dx*dx + dy*dy
    dist = math.sqrt(dist_sq) if dist_sq > 1e-20 else 1e-10
    return dist - x[2*n+i] - x[2*n+j]


def constraint_radius_pos(x, n, i):
    return x[2*n + i]


def create_constraints(n):
    constraints = []
    for i in range(n):
        constraints.append({'type': 'ineq', 'fun': partial(constraint_x_min, n=n, i=i)})
        constraints.append({'type': 'ineq', 'fun': partial(constraint_x_max, n=n, i=i)})
        constraints.append({'type': 'ineq', 'fun': partial(constraint_y_min, n=n, i=i)})
        constraints.append({'type': 'ineq', 'fun': partial(constraint_y_max, n=n, i=i)})
        constraints.append({'type': 'ineq', 'fun': partial(constraint_radius_pos, n=n, i=i)})
    
    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({'type': 'ineq', 'fun': partial(constraint_overlap, n=n, i=i, j=j)})
    
    return constraints


def force_based_sim(n, centers_init, radii_init, steps=3000):
    centers = centers_init.copy()
    radii = radii_init.copy()
    lr = 0.02
    
    for step in range(steps):
        forces = np.zeros_like(centers)
        
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist_sq = dx*dx + dy*dy
                dist = math.sqrt(dist_sq) if dist_sq > 1e-20 else 1e-10
                min_dist = radii[i] + radii[j]
                
                if dist < min_dist:
                    force_mag = (min_dist - dist) / dist * 2.0
                    forces[i, 0] += dx * force_mag
                    forces[i, 1] += dy * force_mag
                    forces[j, 0] -= dx * force_mag
                    forces[j, 1] -= dy * force_mag
        
        centers += lr * forces
        
        for i in range(n):
            centers[i, 0] = np.clip(centers[i, 0], radii[i], 1 - radii[i])
            centers[i, 1] = np.clip(centers[i, 1], radii[i], 1 - radii[i])
        
        radii += 0.00008
        lr *= 0.9993
    
    return centers, radii


def make_hex_config(n):
    centers_list = []
    
    row_configs = [(5, 0.0), (6, 0.5/6), (5, 0.0), (6, 0.5/6), (4, 0.0)]
    num_rows = len(row_configs)
    
    for row_idx, (num_circles, x_off) in enumerate(row_configs):
        y = (row_idx + 0.5) / num_rows
        for col in range(num_circles):
            x = (col + 0.5) / num_circles + x_off
            x = np.clip(x, 0.05, 0.95)
            centers_list.append([x, y])
    
    centers_list = centers_list[:n]
    while len(centers_list) < n:
        centers_list.append([0.5, 0.5])
    
    return np.array(centers_list[:n]), np.ones(n) * 0.065


def make_random_config(n, seed=42):
    rng = random.Random(seed)
    centers = np.zeros((n, 2))
    for i in range(n):
        centers[i, 0] = rng.random()
        centers[i, 1] = rng.random()
    radii = np.ones(n) * 0.05
    return centers, radii


def make_corner_config(n):
    centers_list = []
    
    # 4 corners
    corners = [[0.15, 0.15], [0.85, 0.15], [0.15, 0.85], [0.85, 0.85]]
    centers_list.extend(corners)
    
    # Edge positions
    edges = [
        [0.5, 0.08], [0.25, 0.06], [0.75, 0.06],
        [0.08, 0.5], [0.06, 0.25], [0.06, 0.75],
        [0.92, 0.5], [0.94, 0.25], [0.94, 0.75],
        [0.5, 0.92], [0.25, 0.94], [0.75, 0.94],
    ]
    centers_list.extend(edges[:n - 4])
    
    # Fill remaining with interior points
    interior = [
        [0.3, 0.3], [0.7, 0.3], [0.3, 0.7], [0.7, 0.7],
        [0.5, 0.5], [0.35, 0.5], [0.65, 0.5], [0.5, 0.35], [0.5, 0.65],
        [0.4, 0.4], [0.6, 0.4], [0.4, 0.6], [0.6, 0.6],
        [0.2, 0.5], [0.8, 0.5], [0.5, 0.2], [0.5, 0.8],
    ]
    while len(centers_list) < n:
        centers_list.append(interior[len(centers_list) - 16])
    
    centers = np.array(centers_list[:n])
    radii = np.ones(n) * 0.07
    return centers, radii


def make_dense_hex_config(n):
    centers_list = []
    
    # Denser hexagonal pattern
    r_est = 0.08
    dy = math.sqrt(3) * r_est
    rows = int(1.0 / dy) + 1
    
    for row in range(rows):
        y = r_est + row * dy
        if y + r_est > 1:
            break
        x_off = r_est if row % 2 == 1 else r_est
        x = x_off
        while x + r_est <= 1:
            centers_list.append([x, y])
            x += 2 * r_est
    
    centers_list = centers_list[:n]
    while len(centers_list) < n:
        centers_list.append([0.5, 0.5])
    
    return np.array(centers_list[:n]), np.ones(n) * 0.06


def optimize_from(centers, radii, n, constraints, bounds, max_iter=8000):
    x0 = np.concatenate([centers.flatten(), radii])
    
    result = minimize(obj_func, x0, method='SLSQP',
                     bounds=bounds, constraints=constraints,
                     args=(n,),
                     options={'maxiter': max_iter, 'ftol': 1e-14})
    
    return result.x[:2*n].reshape(n, 2), result.x[2*n:], np.sum(result.x[2*n:])


def run_packing():
    n = 26
    constraints = create_constraints(n)
    bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n
    
    best_centers = None
    best_radii = None
    best_sum = -1.0
    
    configs = [
        ("hex", make_hex_config(n)),
        ("corner", make_corner_config(n)),
        ("dense_hex", make_dense_hex_config(n)),
    ]
    
    # Add random configs with different seeds
    for seed in [42, 123, 456, 789, 1024]:
        configs.append((f"rand_{seed}", make_random_config(n, seed)))
    
    for name, (centers, radii) in configs:
        # Phase 1: Force-based simulation
        c_sim, r_sim = force_based_sim(n, centers.copy(), radii.copy(), steps=2000)
        
        # Phase 2: SLSQP optimization
        c_opt, r_opt, s_opt = optimize_from(c_sim, r_sim, n, constraints, bounds, max_iter=8000)
        
        if s_opt > best_sum:
            best_sum = s_opt
            best_centers = c_opt.copy()
            best_radii = r_opt.copy()
    
    # Additional refinement: run SLSQP again from the best solution with smaller tolerance
    c_final, r_final, s_final = optimize_from(best_centers, best_radii, n, constraints, bounds, max_iter=12000)
    
    return c_final, r_final, s_final
