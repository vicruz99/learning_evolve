import numpy as np
from scipy.optimize import minimize, NonlinearConstraint, Bounds

def compute_obj(vars_):
    """Objective function: minimize negative sum of radii"""
    return -np.sum(vars_[2::3])

def compute_constraints(vars_):
    """Compute all inequality constraints as a single array"""
    n = 26
    x = vars_[::3]
    y = vars_[1::3]
    r = vars_[2::3]
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    cons_bound = np.concatenate([
        x - r,
        1.0 - x - r,
        y - r,
        1.0 - y - r
    ])
    
    # Pairwise non-overlap constraints: dist^2 >= (r_i + r_j)^2
    # Vectorized computation for all pairs
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    r_sum = r[:, None] + r[None, :]
    
    # Extract upper triangular part (i < j)
    row_idx, col_idx = np.triu_indices(n, k=1)
    cons_pair = (dx[row_idx, col_idx]**2 + dy[row_idx, col_idx]**2) - (r_sum[row_idx, col_idx]**2)
    
    return np.concatenate([cons_bound, cons_pair])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # Initialize centers in a hexagonal lattice pattern
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.05
    
    idx = 0
    # Row structure: 6, 5, 6, 5, 4 circles total 26
    row_counts = [6, 5, 6, 5, 4]
    y_base = 0.15
    dy = 0.16
    
    for r_idx, count in enumerate(row_counts):
        y = y_base + r_idx * dy
        # Hexagonal offset for alternating rows
        offset = 0.05 if r_idx % 2 == 1 else 0.0
        for c_idx in range(count):
            x = 0.1 + c_idx * 0.13 + offset
            centers[idx] = [x, y]
            idx += 1
            
    # Flatten variables: [x0, y0, r0, x1, y1, r1, ...]
    x0 = np.zeros(3 * n)
    x0[::3] = centers[:, 0]
    x0[1::3] = centers[:, 1]
    x0[2::3] = radii
    
    # Bounds: x,y in [0,1], r in [0, 0.5]
    lb = [0.0] * (2 * n) + [0.0] * n
    ub = [1.0] * (2 * n) + [0.5] * n
    bounds = Bounds(lb, ub)
    
    # Nonlinear constraints: all >= 0
    constraint = NonlinearConstraint(compute_constraints, 0.0, np.inf)
    
    # Optimization options
    options = {
        'maxiter': 5000,
        'verbose': 0,
        'gtol': 1e-10
    }
    
    # Run optimization
    res = minimize(compute_obj, x0, method='trust-constr', bounds=bounds, 
                   constraints=[constraint], options=options)
                   
    final_centers = res.x[:2*n].reshape((n, 2))
    final_radii = res.x[2*n:]
    return final_centers, final_radii, np.sum(final_radii)
