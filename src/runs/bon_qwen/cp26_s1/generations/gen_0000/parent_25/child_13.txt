import numpy as np
from scipy.optimize import minimize

def _objective(vars, n):
    # Maximize sum of radii -> minimize negative sum
    # vars layout: [x0, y0, r0, x1, y1, r1, ...]
    return -np.sum(vars[2*n:])

def _constraint_fun(vars, n):
    c = vars.reshape((n, 3))
    cons = []
    
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    # Equivalent to: x - r >= 0, 1 - x - r >= 0, etc.
    cons.extend(c[:, 0] - c[:, 2])
    cons.extend(1.0 - c[:, 0] - c[:, 2])
    cons.extend(c[:, 1] - c[:, 2])
    cons.extend(1.0 - c[:, 1] - c[:, 2])
    
    # Non-overlap constraints: dist^2 >= (r_i + r_j)^2
    for i in range(n):
        for j in range(i + 1, n):
            dx = c[i, 0] - c[j, 0]
            dy = c[i, 1] - c[j, 1]
            dr = c[i, 2] + c[j, 2]
            cons.append(dx**2 + dy**2 - dr**2)
            
    return np.array(cons)

def run_packing():
    n = 26
    
    # Initial hexagonal configuration for a high-quality starting point
    r0 = 0.098
    centers = []
    radii = []
    y = r0
    row = 0
    while len(centers) < n:
        x = r0 + (row % 2) * r0
        while x + r0 <= 1.0 and len(centers) < n:
            centers.append([x, y])
            radii.append(r0)
            x += 2 * r0
        y += np.sqrt(3) * r0
        row += 1
        
    centers = np.array(centers[:n])
    radii = np.array(radii[:n])
    
    x0 = np.concatenate([centers.flatten(), radii])
    
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
    
    constr = {'type': 'ineq', 'fun': _constraint_fun, 'args': (n,)}
    
    # Use SLSQP for constrained optimization
    res = minimize(_objective, x0, args=(n,), method='SLSQP', bounds=bounds, constraints=constr,
                   options={'maxiter': 5000, 'ftol': 1e-12, 'disp': False})
                   
    best_vars = res.x
    best_centers = best_vars[:2*n].reshape((n, 2))
    best_radii = best_vars[2*n:]
    
    # Post-processing to ensure strict validity within numerical tolerance
    best_radii = np.maximum(best_radii, 0.0)
    best_centers = np.clip(best_centers, 0.0, 1.0)
    
    # Enforce boundary constraints strictly
    for i in range(n):
        margin = 1e-9
        r = best_radii[i]
        x, y = best_centers[i]
        max_r = min(x, 1.0 - x, y, 1.0 - y)
        if r > max_r - margin:
            best_radii[i] = max(0.0, max_r - margin)
            
    return best_centers, best_radii, float(np.sum(best_radii))
