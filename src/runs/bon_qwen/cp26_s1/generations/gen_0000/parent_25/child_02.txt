import numpy as np
from scipy.optimize import minimize, linprog


def compute_optimal_radii(centers):
    """Given circle centers, compute optimal radii via LP to maximize sum of radii."""
    n = centers.shape[0]

    # Objective: minimize -sum(r_i)
    c = -np.ones(n)

    A_ub_list = []
    b_ub_list = []

    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        x, y = centers[i]
        row = np.zeros(n)
        row[i] = 1
        # r_i <= x
        A_ub_list.append(row.copy())
        b_ub_list.append(x)
        # r_i <= 1 - x
        A_ub_list.append(row.copy())
        b_ub_list.append(1 - x)
        # r_i <= y
        A_ub_list.append(row.copy())
        b_ub_list.append(y)
        # r_i <= 1 - y
        A_ub_list.append(row.copy())
        b_ub_list.append(1 - y)

    # Non-overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            row = np.zeros(n)
            row[i] = 1
            row[j] = 1
            A_ub_list.append(row)
            b_ub_list.append(dist)

    A_ub = np.array(A_ub_list)
    b_ub = np.array(b_ub_list)

    bounds = [(0, None) for _ in range(n)]

    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')

    if result.success:
        return result.x
    else:
        return np.zeros(n)


def objective(params, n):
    """Objective function: negative sum of optimal radii."""
    centers = params.reshape(n, 2)
    radii = compute_optimal_radii(centers)
    return -np.sum(radii)


def init_hexagonal_lattice(n):
    """Initialize centers in a hexagonal lattice pattern with 6-5-6-5-4 rows."""
    row_counts = [6, 5, 6, 5, 4]
    centers = []

    # Approximate radius for spacing
    r_approx = 0.1
    h_spacing = 2 * r_approx  # horizontal spacing
    v_spacing = np.sqrt(3) * r_approx  # vertical spacing

    y = r_approx + 0.02  # starting y position

    for row_idx, num_circles in enumerate(row_counts):
        # Calculate x positions for this row
        total_width = (num_circles - 1) * h_spacing
        x_start = (1 - total_width) / 2

        # Offset alternating rows
        if row_idx % 2 == 1:
            x_start += h_spacing / 2

        for i in range(num_circles):
            x = x_start + i * h_spacing
            centers.append([x, y])

        y += v_spacing

    return np.array(centers[:n])


def run_packing():
    """Pack 26 circles in unit square to maximize sum of radii."""
    n = 26

    best_obj = np.inf
    best_centers = None

    # Try multiple initial configurations
    for trial in range(5):
        if trial == 0:
            # Hexagonal lattice
            centers_init = init_hexagonal_lattice(n)
        else:
            # Perturbed hexagonal
            centers_init = init_hexagonal_lattice(n).copy()
            np.random.seed(trial * 42)
            centers_init += np.random.normal(0, 0.02, size=centers_init.shape)
            # Clip to valid range
            centers_init = np.clip(centers_init, 0.02, 0.98)

        # Optimize
        result = minimize(
            objective,
            centers_init.flatten(),
            args=(n,),
            method='Nelder-Mead',
            options={
                'maxiter': 50000,
                'xatol': 1e-8,
                'fatol': 1e-10,
                'adaptive': True
            }
        )

        if result.fun < best_obj:
            best_obj = result.fun
            best_centers = result.x.reshape(n, 2)

    # Compute final radii
    radii = compute_optimal_radii(best_centers)
    total = np.sum(radii)

    # Safety: clip any tiny negative radii from numerical errors
    radii = np.maximum(radii, 0)

    return best_centers, radii, np.sum(radii)
