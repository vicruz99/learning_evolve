While equal circles are a strong candidate, a general approach maximizing the sum of radii directly allows for potentially better packing configurations by utilizing gaps more effectively.

The strategy is:
1.  **Initialization**: Generate 26 initial positions using a Sobol sequence (quasi-random) to ensure a low-discrepancy, well-distributed spread across the square. Radii are initialized to a small value.
2.  **Optimization**: Use `scipy.optimize.minimize` with the 'SLSQP' method.
    *   **Objective**: Maximize the sum of radii (minimize the negative sum).
    *   **Variables**: A 78-dimensional vector containing $(x_1, y_1, r_1, \dots, x_{26}, y_{26}, r_{26})$.
    *   **Constraints**: 
        *   Boundary constraints ensuring each circle is within $[0, 1]$.
        *   Non-overlap constraints ensuring the distance between any pair of centers is at least the sum of their radii.
        *   Non-negativity of radii.
3.  **Robustness**: Multiple restarts with different initial configurations are performed to escape local optima and find the best possible sum.

import numpy as np
from scipy.optimize import minimize
from numpy.random import SeedSequence, BitGenerator, Generator

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    Returns (centers, radii, sum_radii).
    """
    n = 26
    best_sum = -1.0
    best_result = None
    
    # Function to generate initial positions (Sobol-like distribution)
    def generate_init(seed_val):
        rng = Generator(BitGenerator('mt19937', SeedSequence(seed_val)))
        # Use a simple grid perturbation or random start
        # Let's try a grid start with random jitter for robustness
        # 5x5 grid is 25. We need 26. 
        # Let's place 26 points somewhat uniformly.
        
        # Base grid 6x6 is 36 points. We can select 26 or just distribute.
        # Better: Sobol sequence
        try:
            from scipy.stats import qmc
            sampler = qmc.Sobol(d=2)
            points = sampler.random(n, seed=seed_val)
            # Transform to unit square
            centers = points
        except:
            # Fallback to random
            centers = rng.random((n, 2))
        
        # Initialize radii small to ensure validity
        radii = np.full(n, 0.05)
        
        # Vectorize: [x1, y1, r1, x2, y2, r2, ...]
        x0 = np.zeros(n * 3)
        for i in range(n):
            x0[3*i] = centers[i, 0]
            x0[3*i+1] = centers[i, 1]
            x0[3*i+2] = radii[i]
            
        return x0

    # Objective function: maximize sum of radii -> minimize negative sum
    def objective(x):
        radii = x[2::3]
        return -np.sum(radii)

    # Constraints
    # 1. Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    # 2. Non-overlap: dist >= r_i + r_j
    
    def get_constraints(x):
        constraints = []
        
        # Extract variables
        xs = x[0::3]
        ys = x[1::3]
        rs = x[2::3]
        
        # Boundary constraints for each circle
        for i in range(n):
            # x - r >= 0
            constraints.append({'type': 'ineq', 'fun': lambda x, i=i: x[3*i] - x[3*i+2]})
            # x + r <= 1 => 1 - (x + r) >= 0
            constraints.append({'type': 'ineq', 'fun': lambda x, i=i: 1.0 - (x[3*i] + x[3*i+2])})
            # y - r >= 0
            constraints.append({'type': 'ineq', 'fun': lambda x, i=i: x[3*i+1] - x[3*i+2]})
            # y + r <= 1 => 1 - (y + r) >= 0
            constraints.append({'type': 'ineq', 'fun': lambda x, i=i: 1.0 - (x[3*i+1] + x[3*i+2])})
            
        # Non-overlap constraints for pairs
        # To avoid creating thousands of lambda closures, we can define them inside the loop 
        # or use a vectorized approach, but for N=26, simple list is manageable.
        # However, passing i, j in lambda needs default args.
        for i in range(n):
            for j in range(i + 1, n):
                # dist^2 >= (r_i + r_j)^2
                # dist^2 - (r_i + r_j)^2 >= 0
                # (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
                idx_i_x, idx_i_y, idx_i_r = 3*i, 3*i+1, 3*i+2
                idx_j_x, idx_j_y, idx_j_r = 3*j, 3*j+1, 3*j+2
                
                def dist_constraint(x, i=i, j=j):
                    xi, yi, ri = x[3*i], x[3*i+1], x[3*i+2]
                    xj, yj, rj = x[3*j], x[3*j+1], x[3*j+2]
                    dist_sq = (xi - xj)**2 + (yi - yj)**2
                    sum_r_sq = (ri + rj)**2
                    return dist_sq - sum_r_sq
                
                constraints.append({'type': 'ineq', 'fun': dist_constraint})
                
        return constraints

    # Run optimization with multiple restarts
    seeds = [42, 123, 456, 789, 1024, 2048, 3333, 5555, 7777, 9999]
    
    for seed in seeds:
        x0 = generate_init(seed)
        
        # Get constraints for this run (structure is same, but lambdas capture loop vars)
        # Note: Re-evaluating constraints list every iteration is expensive if done inside optimizer callback,
        # but we pass the list to minimize.
        # However, creating 400+ constraints objects is fine.
        cons = get_constraints(x0)
        
        # Bounds: x, y in [0, 1], r in [0, 0.5] (max possible radius)
        bounds = []
        for i in range(n):
            bounds.extend([(0, 1), (0, 1), (0, 0.5)])
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, 
                           options={'ftol': 1e-9, 'maxiter': 500, 'disp': False})
            
            if res.success or res.fun < -best_sum: # res.fun is negative sum
                current_sum = -res.fun
                if current_sum > best_sum:
                    best_sum = current_sum
                    best_result = res.x
        except Exception:
            continue

    # If no result found (should not happen), fallback
    if best_result is None:
        best_result = generate_init(0)
        best_sum = np.sum(best_result[2::3])

    # Extract final solution
    centers = np.column_stack((best_result[0::3], best_result[1::3]))
    radii = best_result[2::3]
    sum_radii = np.sum(radii)
    
    # Final validation and cleaning (clip tiny negative radii due to numerical errors)
    radii = np.maximum(radii, 0.0)
    
    # Ensure strict validity by a tiny margin if needed, but optimizer should handle it.
    # Just return the result.
    return centers, radii, sum_radii
