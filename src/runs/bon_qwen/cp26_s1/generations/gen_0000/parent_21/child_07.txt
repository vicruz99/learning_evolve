import numpy as np
from scipy.optimize import minimize

def compute_penalty(centers, radii):
    """
    Computes the squared overlap penalty for circle-circle and circle-boundary constraints.
    Uses vectorized operations for speed.
    """
    n = len(radii)
    penalty = 0.0
    
    # Circle-circle overlaps
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    reqs = radii[:, np.newaxis] + radii[np.newaxis, :]
    overlaps = np.maximum(0.0, reqs - dists)
    penalty += np.sum(overlaps**2)
    
    # Boundary overlaps
    xb, yb = centers[:, 0], centers[:, 1]
    penalty += np.sum(np.maximum(0.0, radii - xb)**2)
    penalty += np.sum(np.maximum(0.0, xb - (1.0 - radii))**2)
    penalty += np.sum(np.maximum(0.0, radii - yb)**2)
    penalty += np.sum(np.maximum(0.0, yb - (1.0 - radii))**2)
        
    return penalty

def _objective(x_flat, n, radii):
    """
    Objective function for scipy.optimize.minimize.
    Reshapes flat vector to centers and computes penalty.
    """
    centers = x_flat.reshape((n, 2))
    return compute_penalty(centers, radii)

                idx += 1
                
    radii = np.full(n, r_init)
    
    best_centers = centers.copy()
    best_radii = radii.copy()

        x0 = centers.flatten()
        bounds = []

            r = radii[i]
            bounds.extend([(r, 1.0 - r), (r, 1.0 - r)])
            
        # Optimize positions to minimize overlap penalty
        res = minimize(_objective, x0, args=(n, radii), method='L-BFGS-B',

            r = radii[i]

            centers[i, 1] = np.clip(centers[i, 1], r, 1.0 - r)

            best_centers = centers.copy()
            best_radii = radii.copy()
            best_sum = np.sum(radii)

            # Re-optimize once to recover a valid state

                r = radii[i]
