def generate_hexagonal(n, row_configs):
    centers = np.zeros((n, 2))
    idx = 0
    
    for row_idx, (count, offset_type) in enumerate(row_configs):
        # Distribute rows evenly
        y = 0.08 + row_idx * 0.18
        
        if offset_type == 0:
            x_start = 0.08
            spacing = 0.22
        else:
            x_start = 0.08 + spacing * 0.5
            spacing = 0.22
        
        for col in range(count):
            x = x_start + col * spacing
            centers[idx] = [x, y]
            idx += 1
    
    return centers


def force_directed_optimize(centers, radii, n, iterations=30000):
    centers = centers.copy()
    radii = radii.copy()
    
    for iteration in range(iterations):
        forces = np.zeros((n, 2))
        
        # Compute repulsive forces between overlapping circles
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist_sq = dx * dx + dy * dy
                dist = math.sqrt(dist_sq) if dist_sq > 1e-15 else 1e-15
                min_dist = radii[i] + radii[j]
                
                if dist < min_dist:
                    overlap = min_dist - dist
                    force_mag = overlap / dist
                    fx = dx * force_mag * 0.5
                    fy = dy * force_mag * 0.5
                    forces[i][0] += fx
                    forces[i][1] += fy
                    forces[j][0] -= fx
                    forces[j][1] -= fy
        
        # Boundary forces
        for i in range(n):
            for d in range(2):
                if centers[i][d] - radii[i] < 0:
                    forces[i][d] += 2.0 * (radii[i] - centers[i][d])
                if centers[i][d] + radii[i] > 1.0:
                    forces[i][d] -= 2.0 * (centers[i][d] + radii[i] - 1.0)
        
        # Learning rate with decay
        lr = 0.005 / (1.0 + iteration * 0.0001)
        centers = centers + forces * lr
        
        # Clamp positions
        for i in range(n):
            for d in range(2):
                centers[i][d] = max(radii[i], min(1.0 - radii[i], centers[i][d]))
        
        # Try to increase radii uniformly
        if iteration % 10 == 0:
            min_gap = compute_min_gap(centers, radii, n)

                radii = radii + increment
            else:
                # Still try small increments
                radii = radii + 1e-7
                # Fix any violations by adjusting positions
                centers = fix_violations(centers, radii, n, max_iter=500)
    
    return centers, radii


def compute_min_gap(centers, radii, n):
    min_gap = 1e10
    for i in range(n):
        for d in range(2):
            gap = min(centers[i][d] - radii[i], 1.0 - centers[i][d] - radii[i])
            min_gap = min(min_gap, gap)
    
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i][0] - centers[j][0]
            dy = centers[i][1] - centers[j][1]
            dist = math.sqrt(dx * dx + dy * dy)
            gap = dist - radii[i] - radii[j]
            min_gap = min(min_gap, gap)
    
    return min_gap


def fix_violations(centers, radii, n, max_iter=500):
    centers = centers.copy()
    for iteration in range(max_iter):
        max_violation = 0
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = math.sqrt(dx * dx + dy * dy)
                min_dist = radii[i] + radii[j]
                if dist < min_dist and dist > 1e-15:
                    violation = min_dist - dist
                    max_violation = max(max_violation, violation)
                    # Push apart
                    factor = (violation + 1e-8) / (2 * dist)
                    centers[i][0] += dx * factor
                    centers[i][1] += dy * factor
                    centers[j][0] -= dx * factor
                    centers[j][1] -= dy * factor
    
    # Clamp to boundaries
    for i in range(n):
        for d in range(2):
            centers[i][d] = max(radii[i], min(1.0 - radii[i], centers[i][d]))
    
    return centers


def scipy_objective(vars, n):
    radii = vars[2 * n:]
    return -np.sum(radii)


def scipy_constraints(vars, n):
    centers = vars[:2 * n].reshape(n, 2)
    radii = vars[2 * n:]
    
    result = []
    
    # Distance constraints: ||c_i - c_j||^2 >= (r_i + r_j)^2
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i][0] - centers[j][0]
            dy = centers[i][1] - centers[j][1]
            result.append(dx * dx + dy * dy - (radii[i] + radii[j]) ** 2)
    
    # Boundary constraints
    for i in range(n):
        for d in range(2):
            result.append(centers[i][d] - radii[i])
            result.append(1.0 - centers[i][d] - radii[i])
    
    # Non-negative radii
    for i in range(n):
        result.append(radii[i])
    
    return np.array(result)


def scipy_refine(centers, radii, n, max_trials=3):
    centers = centers.copy()
    radii = radii.copy()
    
    best_centers = centers
    best_radii = radii
    best_sum = np.sum(radii)
    
    for trial in range(max_trials):
        x0 = np.concatenate([centers.flatten(), radii])
        
        # Bounds: centers in [0,1], radii in [0, 0.5]
        bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
        
        cons = {'type': 'ineq', 'fun': lambda v, nn=n: scipy_constraints(v, nn)}
        
        try:
            result = minimize(
                lambda v, nn=n: scipy_objective(v, nn),
                x0,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 500, 'ftol': 1e-12, 'disp': False}
            )
            
            if result.success or result.fun < -best_sum:
                centers_opt = result.x[:2 * n].reshape(n, 2)
                radii_opt = result.x[2 * n:]
                sum_opt = np.sum(radii_opt)
                
                # Validate
                if np.all(radii_opt >= 0) and np.all(centers_opt >= 0) and np.all(centers_opt <= 1):
                    # Check validity
                    valid = True
                    for i in range(n):
                        for d in range(2):
                            if centers_opt[i][d] - radii_opt[i] < -1e-10 or centers_opt[i][d] + radii_opt[i] > 1 + 1e-10:
                                valid = False
                                break
                        if not valid:
                            break
                    
                    if valid and sum_opt > best_sum:
                        best_centers = centers_opt
                        best_radii = radii_opt
                        best_sum = sum_opt
                        centers = centers_opt
                        radii = radii_opt
        except Exception:
            pass
    
    return best_centers, best_radii
