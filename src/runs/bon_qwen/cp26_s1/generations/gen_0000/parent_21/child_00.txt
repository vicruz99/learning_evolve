import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import math
import random


def hexagonal_init(n, seed=42, base_r=0.09):
    """Initialize circles in a hexagonal pattern"""
    rng = random.Random(seed)
    r = base_r
    d = 2 * r
    h = math.sqrt(3) * r
    
    centers = []
    for row in range(20):
        for col in range(20):
            if len(centers) >= n:
                break
            x = r + col * d + (row % 2) * (d / 2)
            y = r + row * h
            if x + r <= 1.0 and y + r <= 1.0:
                centers.append([x + rng.uniform(-0.002, 0.002), y + rng.uniform(-0.002, 0.002)])
        if len(centers) >= n:
            break
    
    if len(centers) < n:
        for i in range(len(centers), n):
            centers.append([rng.uniform(0.1, 0.9), rng.uniform(0.1, 0.9)])
    
    return np.array(centers[:n])


def grid_init(n, rows, cols, seed=42):
    """Initialize circles in a grid pattern"""
    rng = random.Random(seed)
    centers = []
    for i in range(rows):
        for j in range(cols):
            if len(centers) >= n:
                break
            if cols > 1:
                x = 0.5 + (j - (cols - 1) / 2) * (1.0 / (cols - 1))
            else:
                x = 0.5
            if rows > 1:
                y = 0.5 + (i - (rows - 1) / 2) * (1.0 / (rows - 1))
            else:
                y = 0.5
            centers.append([x + rng.uniform(-0.01, 0.01), y + rng.uniform(-0.01, 0.01)])
        if len(centers) >= n:
            break
    while len(centers) < n:
        centers.append([rng.uniform(0.1, 0.9), rng.uniform(0.1, 0.9)])
    return np.array(centers[:n])


def compute_optimal_radii(centers, tol=1e-9):
    """Given fixed centers, compute optimal radii using LP"""
    n = len(centers)
    r = cp.Variable(n)
    
    constraints = []
    
    for i in range(n):
        x, y = centers[i]
        constraints.append(r[i] <= x + tol)
        constraints.append(r[i] <= 1 - x + tol)
        constraints.append(r[i] <= y + tol)
        constraints.append(r[i] <= 1 - y + tol)
        constraints.append(r[i] >= 0)
    
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = math.sqrt(dx * dx + dy * dy)
            constraints.append(r[i] + r[j] <= dist + tol)
    
    objective = cp.Maximize(cp.sum(r))
    problem = cp.Problem(objective, constraints)
    try:
        problem.solve(solver=cp.ECOS, verbose=False)
        if problem.status in ['optimal', 'optimal_inaccurate']:
            return np.maximum(r.value, 0)
        else:
            return np.ones(n) * 0.01
    except Exception:
        return np.ones(n) * 0.01


def push_circles_apart(centers, radii, steps=500, step_size=0.05):
    """Push circles apart to reduce overlaps using force-based method"""
    n = len(centers)
    centers = centers.copy().astype(float)
    radii = radii.astype(float)
    
    for step in range(steps):
        forces = np.zeros_like(centers)
        max_overlap = 0
        
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            
            if x - r < 0:
                forces[i, 0] += 50.0 * (r - x)
            if x + r > 1:
                forces[i, 0] -= 50.0 * (x + r - 1)
            if y - r < 0:
                forces[i, 1] += 50.0 * (r - y)
            if y + r > 1:
                forces[i, 1] -= 50.0 * (y + r - 1)
        
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[j, 0] - centers[i, 0]
                dy = centers[j, 1] - centers[i, 1]
                dist_sq = dx * dx + dy * dy
                dist = math.sqrt(dist_sq) if dist_sq > 1e-20 else 1e-10
                
                min_dist = radii[i] + radii[j]
                if dist < min_dist:
                    overlap = min_dist - dist
                    max_overlap = max(max_overlap, overlap)
                    fx = (dx / dist) * overlap
                    fy = (dy / dist) * overlap
                    forces[i, 0] -= fx
                    forces[i, 1] -= fy
                    forces[j, 0] += fx
                    forces[j, 1] += fy
        
        decay = max(0.1, 1.0 - step / steps)
        centers += step_size * decay * forces
        centers[:, 0] = np.clip(centers[:, 0], 0.005, 0.995)
        centers[:, 1] = np.clip(centers[:, 1], 0.005, 0.995)
        
        if max_overlap < 1e-8:
            break
    
    return centers


def expand_circles_toward_boundaries(centers, radii, iterations=200):
    """Push circles toward boundaries to allow larger radii"""
    n = len(centers)
    centers = centers.copy()

            r = radii[i]
            
            # Push toward nearest boundary
            d_left = x
            d_right = 1 - x
            d_bottom = y
            d_top = 1 - y
            d_min = min(d_left, d_right, d_bottom, d_top)
            
            # If circle is not touching a boundary, push it toward the nearest one
            if d_min > r + 0.005:
                strength = 2.0 * (d_min - r - 0.005)
                if d_min == d_left:
                    forces[i, 0] -= strength
                elif d_min == d_right:
                    forces[i, 0] += strength
                elif d_min == d_bottom:
                    forces[i, 1] -= strength
                else:
                    forces[i, 1] += strength
        
        # Also separate overlapping circles
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[j, 0] - centers[i, 0]
                dy = centers[j, 1] - centers[i, 1]
                dist = math.sqrt(dx * dx + dy * dy)
                min_dist = radii[i] + radii[j]
                
                if dist < min_dist + 0.001 and dist > 1e-10:
                    repel = (min_dist + 0.001 - dist) / dist
                    forces[i, 0] -= dx * repel * 3.0
                    forces[i, 1] -= dy * repel * 3.0
                    forces[j, 0] += dx * repel * 3.0
                    forces[j, 1] += dy * repel * 3.0
        
        centers += 0.02 * forces
        centers[:, 0] = np.clip(centers[:, 0], 0.01, 0.99)
        centers[:, 1] = np.clip(centers[:, 1], 0.01, 0.99)
    
    return centers, radii


def alternating_optimize(centers_init, max_iter=80, push_steps=300):
    """Alternating optimization of positions and radii"""
    n = len(centers_init)
    centers = centers_init.copy()
    
    for iteration in range(max_iter):
        radii = compute_optimal_radii(centers)
        centers = push_circles_apart(centers, radii, steps=push_steps, step_size=0.03)
        radii = compute_optimal_radii(centers)
        
        if iteration % 20 == 0:
            print(f"  AltOpt iter {iteration}: sum_r = {np.sum(radii):.6f}, min_r = {np.min(radii):.6f}, max_r = {np.max(radii):.6f}")
    
    return centers, radii
