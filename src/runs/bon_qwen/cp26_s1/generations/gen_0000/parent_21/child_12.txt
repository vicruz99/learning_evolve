import numpy as np
from scipy.optimize import minimize

def objective_func(x):
    """Objective: maximize sum of radii (minimize negative sum)."""
    return -np.sum(x[2::3])

def get_constraints(x):
    """Compute all boundary and non-overlap constraints."""
    n = 26
    xs = x[0:2*n:3]
    ys = x[1:2*n:3]
    rs = x[2:2*n:3]
    
    # Boundary constraints: x >= r, x <= 1-r, y >= r, y <= 1-r
    c1 = xs - rs
    c2 = 1.0 - xs - rs
    c3 = ys - rs
    c4 = 1.0 - ys - rs
    
    # Pairwise non-overlap constraints: dist^2 >= (r_i + r_j)^2
    diff_x = xs[:, None] - xs[None, :]
    diff_y = ys[:, None] - ys[None, :]
    dist2 = diff_x**2 + diff_y**2
    
    r_sum = rs[:, None] + rs[None, :]
    r_sum2 = r_sum**2
    
    # Extract upper triangle to avoid duplicates and self-comparisons
    i, j = np.triu_indices(n, k=1)
    c5 = dist2[i, j] - r_sum2[i, j]
    
    return np.concatenate([c1, c2, c3, c4, c5])

def run_packing():
    n = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Bounds for x, y, r
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
    cons = {'type': 'ineq', 'fun': get_constraints}
    
    starts = []
    
    # 1. Random initializations
    for s in range(12):
        np.random.seed(s + 42)
        cx = np.random.uniform(0.05, 0.95, (n, 2))
        x0 = np.empty(3 * n)
        x0[0::3] = cx[:, 0]
        x0[1::3] = cx[:, 1]
        x0[2::3] = 0.015  # Small initial radius ensures no overlaps
        starts.append(x0)
        
    # 2. Grid initialization (5x5 layout, first 26)
    gs = np.linspace(0.08, 0.92, 5)
    gx, gy = np.meshgrid(gs, gs)
    cg = np.column_stack([gx.ravel(), gy.ravel()])[:26]
    x0_grid = np.empty(3 * n)
    x0_grid[0::3] = cg[:, 0]
    x0_grid[1::3] = cg[:, 1]
    x0_grid[2::3] = 0.035
    starts.append(x0_grid)

    # Optimization loop
    for x0 in starts:
        res = minimize(objective_func, x0, method='SLSQP', 
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 4000, 'ftol': 1e-11, 'disp': False})
                       
        # Verify constraints are satisfied within tolerance
        cvals = get_constraints(res.x)
        if cvals.min() >= -1e-5:
            cur_sum = np.sum(res.x[2::3])
            if cur_sum > best_sum:
                best_sum = cur_sum
                best_centers = np.column_stack([res.x[0::3], res.x[1::3]])
                best_radii = res.x[2::3]
                
    # Fallback if optimizer fails completely (highly unlikely)
    if best_centers is None:
        best_centers = cg
        best_radii = np.full(n, 0.04)
        best_sum = np.sum(best_radii)
        
    return best_centers, best_radii, best_sum
