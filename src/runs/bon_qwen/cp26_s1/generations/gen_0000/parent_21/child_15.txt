import numpy as np
from scipy.optimize import minimize


def compute_penalty_objective(x, n, penalty_weight):
    """Compute objective with penalty for constraint violations."""
    c = x[:2 * n].reshape((n, 2))
    r = x[2 * n:]

    obj = -np.sum(r)
    penalty = 0.0

    # Non-overlap penalty
    for i in range(n):
        for j in range(i + 1, n):
            dx = c[i, 0] - c[j, 0]
            dy = c[i, 1] - c[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            violation = r[i] + r[j] - dist
            if violation > 1e-12:
                penalty += violation * violation

    # Boundary penalties
    for i in range(n):
        for k in range(2):
            left_viol = r[i] - c[i, k]
            if left_viol > 1e-12:
                penalty += left_viol * left_viol
            right_viol = c[i, k] + r[i] - 1.0
            if right_viol > 1e-12:
                penalty += right_viol * right_viol

    # Radius non-negativity
    for i in range(n):
        if r[i] < -1e-12:
            penalty += (r[i] + 1e-12) * (r[i] + 1e-12)

    return obj + penalty_weight * penalty


def generate_hexagonal_init(n, spacing_factor=1.0):
    """Generate initial hexagonal configuration."""
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.07 * spacing_factor

    idx = 0
    base_spacing = 0.18
    spacing_x = base_spacing * spacing_factor
    spacing_y = spacing_x * np.sqrt(3) / 2

    for i in range(7):
        for j in range(7):
            if idx >= n:
                break
            x = 0.12 + j * spacing_x + (i % 2) * spacing_x / 2
            y = 0.12 + i * spacing_y
            if x < 0.88 and y < 0.88:
                centers[idx, 0] = x
                centers[idx, 1] = y
                idx += 1
        if idx >= n:
            break

    # Fill remaining positions
    while idx < n:
        row = idx // 6
        col = idx % 6
        centers[idx, 0] = 0.2 + col * 0.12
        centers[idx, 1] = 0.2 + row * 0.12
        idx += 1

    return centers, radii


def optimize_from_initial(x0, n):
    """Run optimization from initial point with increasing penalties."""
    x = x0.copy()
    bounds = [(0.005, 0.995)] * (2 * n) + [(0.005, 0.49)] * n

    # Phase 1: Moderate penalty to get feasible region
    for pw in [50, 100, 200, 500]:
        result = minimize(
            compute_penalty_objective,
            x,
            args=(n, pw),
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 3000, 'ftol': 1e-15, 'gtol': 1e-10}
        )
        x = result.x

    # Phase 2: High penalty for precision
    for pw in [1000, 2000, 5000]:
        result = minimize(
            compute_penalty_objective,
            x,
            args=(n, pw),
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 5000, 'ftol': 1e-15, 'gtol': 1e-10}
        )
        x = result.x

    return x


def cleanup_solution(centers, radii, n):
    """Post-process to ensure validity."""
    radii = radii.copy()
    centers = centers.copy()

    # Ensure non-negative radii
    radii = np.maximum(radii, 1e-10)

    # Iterative cleanup: reduce radii to fix violations
    for iteration in range(50):
        max_violation = 0

        # Check non-overlap
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                required_dist = radii[i] + radii[j]
                if dist < required_dist - 1e-12:
                    # Reduce the larger radius
                    if radii[i] >= radii[j]:
                        radii[i] = (dist - radii[j]) * 0.999
                    else:
                        radii[j] = (dist - radii[i]) * 0.999
                    max_violation = max(max_violation, required_dist - dist)

        # Check boundaries
        for i in range(n):
            for k in range(2):
                if centers[i, k] - radii[i] < -1e-12:
                    radii[i] = centers[i, k] - 1e-12
                    max_violation = max(max_violation, 1e-12)
                if centers[i, k] + radii[i] > 1 + 1e-12:
                    radii[i] = 1 - centers[i, k] - 1e-12
                    max_violation = max(max_violation, 1e-12)

        if max_violation < 1e-14:
            break

    # Final safety: small uniform reduction if needed
    radii = np.maximum(radii, 0)

    return centers, radii


def run_packing() -> tuple:
