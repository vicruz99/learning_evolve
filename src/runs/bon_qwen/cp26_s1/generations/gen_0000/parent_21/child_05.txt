import numpy as np
from scipy.optimize import minimize
import math

N_CIRCLES = 26

def objective(x):
    n = N_CIRCLES
    return -np.sum(x[2 * n:])

def constraints(x):
    n = N_CIRCLES
    c = x[:2 * n].reshape(n, 2)
    r = x[2 * n:]
    
    m = 4 * n + n * (n - 1) // 2
    result = np.empty(m)
    k = 0
    
    for i in range(n):
        result[k] = c[i, 0] - r[i]
        k += 1
        result[k] = 1.0 - c[i, 0] - r[i]
        k += 1
        result[k] = c[i, 1] - r[i]
        k += 1
        result[k] = 1.0 - c[i, 1] - r[i]
        k += 1
    
    for i in range(n):
        for j in range(i + 1, n):
            dx = c[i, 0] - c[j, 0]
            dy = c[i, 1] - c[j, 1]
            dist = math.sqrt(dx * dx + dy * dy)
            result[k] = dist - r[i] - r[j]
            k += 1
    
    return result

def make_hex_config(r_init=0.101):
    n = N_CIRCLES
    centers = np.zeros((n, 2))
    radii = np.ones(n) * r_init
    
    idx = 0
    row_counts = [6, 5, 6, 5, 4]
    for row, count in enumerate(row_counts):
        y = r_init + row * r_init * math.sqrt(3)
        if row % 2 == 0:
            x_start = r_init
        else:
            x_start = 2.0 * r_init
        for col in range(count):
            x = x_start + col * 2.0 * r_init
            centers[idx] = [x, y]
            idx += 1
    
    return centers, radii

def make_random_config(n=26, seed=0):
    rng = np.random.RandomState(seed)
    centers = rng.rand(n, 2)
    radii = np.ones(n) * 0.08
    return centers, radii

def make_grid_config(n=26, r_init=0.101):
    centers = np.zeros((n, 2))
    radii = np.ones(n) * r_init
    
    cols = 6
    rows = 5
    idx = 0
    
    for row in range(rows):
        for col in range(cols):
            if idx >= n:
                break
            y = r_init + row * 2.0 * r_init * 0.95
            x = r_init + col * 2.0 * r_init
            if row % 2 == 1:
                x += r_init
            centers[idx] = [x, y]
            idx += 1
    
    return centers, radii

def run_packing():
    n = N_CIRCLES
    
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    bounds = [(0.0, 1.0)] * (2 * n) + [(1e-8, 0.5)] * n
    constraint = {'type': 'ineq', 'fun': constraints}
    
    # Try multiple initial configurations
    initial_configs = []
    
    # Hexagonal configurations with different starting radii
    for r_init in [0.095, 0.100, 0.105, 0.108]:
        centers, radii = make_hex_config(r_init)
        initial_configs.append((centers, radii, f"hex_{r_init}"))
    
    # Random configurations
    for seed in range(5):
        centers, radii = make_random_config(n, seed)
        initial_configs.append((centers, radii, f"rand_{seed}"))
    
    # Grid configurations
    for r_init in [0.095, 0.100, 0.105]:
        centers, radii = make_grid_config(n, r_init)
        initial_configs.append((centers, radii, f"grid_{r_init}"))
    
    for centers_init, radii_init, name in initial_configs:
        # Perturb each initial config multiple times
        for trial in range(3):
            centers = centers_init.copy()
            radii = radii_init.copy()
            
            if trial > 0:
                np.random.seed(trial + hash(name) % 1000)
                centers += np.random.randn(n, 2) * 0.003 * trial
                radii += np.random.randn(n) * 0.001 * trial
                radii = np.maximum(radii, 0.005)
            
            # Clip to valid range
            centers = np.clip(centers, 0.01, 0.99)
            radii = np.clip(radii, 1e-6, 0.4)
            
            x0 = np.concatenate([centers.ravel(), radii])
            
            try:
                result = minimize(objective, x0, method='SLSQP', bounds=bounds,
                                constraints=constraint, 
                                options={'maxiter': 3000, 'ftol': 1e-14, 'disp': False})
                
                centers_opt = result.x[:2 * n].reshape(n, 2)
                radii_opt = result.x[2 * n:]
                s = np.sum(radii_opt)
                
                if s > best_sum:
                    best_sum = s
                    best_centers = centers_opt.copy()
                    best_radii = radii_opt.copy()
            except Exception:
                pass
    
    # Refine best solution with more iterations
    if best_centers is not None:
        x0 = np.concatenate([best_centers.ravel(), best_radii])
        try:
            result = minimize(objective, x0, method='SLSQP', bounds=bounds,
                            constraints=constraint,

                best_sum = s
                best_centers = centers_opt.copy()
                best_radii = radii_opt.copy()
        except Exception:
            pass
    
    # Ensure radii are non-negative
    best_radii = np.maximum(best_radii, 0.0)
    
    return best_centers, best_radii, best_sum
