import numpy as np
from scipy.optimize import minimize
import math

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Pack 26 circles in a unit square to maximize the sum of radii.
    """
    N = 26
    
    # --- Step 1: Initialization (Hexagonal Pattern) ---
    # Arrange circles in alternating rows to utilize space efficiently.
    centers = np.zeros((N, 2))
    radii = np.full(N, 0.09) # Start with a feasible radius
    
    # 5 rows with 6, 5, 6, 5, 4 circles sums to 26.
    counts = [6, 5, 6, 5, 4]
    row_y = 0.15 # Starting y for the first row
    row_idx = 0
    circle_idx = 0
    
    for r, count in enumerate(counts):
        # Shift x for alternating rows in hexagonal packing
        x_offset = 0.15 if r % 2 == 1 else 0.15
        # Adjust offset to center the row better within [0, 1]
        if r % 2 == 1:
            x_offset = 0.22 
        else:
            x_offset = 0.13
            
        for c in range(count):
            x = x_offset + c * 0.20
            y = row_y + r * 0.17
            
            # If circle goes out of bounds, shrink radius or clamp
            if circle_idx < N:
                centers[circle_idx] = [x, y]
                # Slightly vary radii to help optimization escape symmetry
                radii[circle_idx] = 0.09 + (circle_idx % 3) * 0.005
                circle_idx += 1
        
        row_y += 0.01 # Move to next row position logic is handled in loop
    
    # --- Step 2: Optimization ---
    # We use a penalty method to handle constraints during optimization.
    # Objective: Maximize sum(radii) - Penalty for overlaps/boundaries.
    
    def objective_and_constraints(params):
        # params: flat array of [x0, y0, r0, x1, y1, r1, ...]
        centers_opt = params[:2*N].reshape((N, 2))
        radii_opt = params[2*N:]
        
        # Ensure radii are positive
        radii_opt = np.maximum(radii_opt, 1e-6)
        
        sum_r = np.sum(radii_opt)
        penalty = 0.0
        
        # Boundary constraints: 0 <= x-r, x+r <= 1, 0 <= y-r, y+r <= 1
        # We want centers to be at least r away from boundaries
        for i in range(N):
            x, y = centers_opt[i]
            r = radii_opt[i]
            
            # Violation of boundaries
            v_left = max(0, r - x)
            v_right = max(0, (x + r) - 1)
            v_bottom = max(0, r - y)
            v_top = max(0, (y + r) - 1)
            
            penalty += 100 * (v_left**2 + v_right**2 + v_bottom**2 + v_top**2)
            
        # Overlap constraints: dist(i, j) >= r_i + r_j
        # Iterate over all pairs
        for i in range(N):
            for j in range(i + 1, N):
                dx = centers_opt[i, 0] - centers_opt[j, 0]
                dy = centers_opt[i, 1] - centers_opt[j, 1]
                dist = math.sqrt(dx*dx + dy*dy)
                min_dist = radii_opt[i] + radii_opt[j]
                
                if dist < min_dist:
                    overlap = min_dist - dist
                    penalty += 500 * (overlap**2)
                    
        return -sum_r + penalty

    # Prepare initial parameters
    init_params = np.concatenate([centers.flatten(), radii])
    
    # Bounds for parameters
    # x, y in [0, 1], r in [0, 0.5]
    bounds = []
    for _ in range(N):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 0.5)) # r
        
    # Run optimization
    # Using L-BFGS-B for bound handling
    result = minimize(objective_and_constraints, init_params, method='L-BFGS-B', bounds=bounds, 
                      options={'maxiter': 2000, 'ftol': 1e-12})
    
    # Extract final parameters
    final_centers = result.x[:2*N].reshape((N, 2))
    final_radii = result.x[2*N:]
    
    # Post-processing: Ensure strict validity
    # If there are tiny violations due to numerical error, shrink radii slightly
    for i in range(N):
        x, y = final_centers[i]
        r = final_radii[i]
        # Clamp to boundaries
        max_r_x = min(x, 1-x)
        max_r_y = min(y, 1-y)
        final_radii[i] = min(r, max_r_x, max_r_y)
        
    # Final check for overlaps and correct them if necessary
    # Simple iterative shrinking for overlaps
    for _ in range(100):
        overlap_found = False
        for i in range(N):
            for j in range(i+1, N):
                dx = final_centers[i, 0] - final_centers[j, 0]
                dy = final_centers[i, 1] - final_centers[j, 1]
                dist = math.sqrt(dx*dx + dy*dy)
                sum_r_ij = final_radii[i] + final_radii[j]
                if dist < sum_r_ij - 1e-9:
                    # Reduce the larger radius to resolve overlap
                    if final_radii[i] > final_radii[j]:
                        final_radii[i] -= (sum_r_ij - dist) / 2.0
                    else:
                        final_radii[j] -= (sum_r_ij - dist) / 2.0
                    overlap_found = True
        if not overlap_found:
            break
            
    # Ensure non-negative radii
    final_radii = np.maximum(final_radii, 0.0)
    
    total_sum = np.sum(final_radii)
    
    return final_centers, final_radii, total_sum
