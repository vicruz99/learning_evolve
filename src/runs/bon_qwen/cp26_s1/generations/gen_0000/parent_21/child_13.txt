

<strategy>
This is a circle packing problem to maximize sum of radii for 26 circles in a unit square.

**Key Observations:**
1. A 5×5 grid of equal circles with r=0.1 gives sum=2.5, but we need 26 circles and target is 2.636
2. Average radius needed: ~0.1014 per circle
3. Simple grid won't fit 26 equal circles at that radius

**Strategy:**
1. Use scipy.optimize.minimize with SLSQP method to directly optimize centers and radii
2. Formulate as: maximize Σr_i subject to boundary and non-overlap constraints
3. Use squared distance constraints for numerical stability: ||c_i - c_j||² ≥ (r_i + r_j)²
4. Start from a perturbed 5×5 grid + 1 extra circle configuration
5. Try multiple initial configurations to avoid local optima
6. Use a single constraint function returning a vector (all constraints at once)
7. Apply small random perturbations to initial grid to help optimization escape symmetry
8. After optimization, verify and clean up the solution

**Implementation Details:**
- 78 variables: 26×2 for centers + 26 for radii
- ~429 constraints: 26×4 boundary + C(26,2) non-overlap + 26 non-negative radii
- Use bounds to constrain centers to [0,1] and radii to [0, 0.5]
</strategy>

```python
import numpy as np
from scipy.optimize import minimize

N = 26


def objective(x):
    """Objective: minimize negative sum of radii (equivalent to maximizing sum)."""
    return -np.sum(x[2 * N :])


def constraint_vec(x):
    """
    Compute all constraint violations.
    Returns array where each element >= 0 means constraint is satisfied.
    """
    n = N
    c = []
    # Boundary constraints for each circle
    for i in range(n):
        c.append(x[2 * i] - x[2 * n + i])  # x_i >= r_i
        c.append(1.0 - x[2 * i] - x[2 * n + i])  # x_i + r_i <= 1
        c.append(x[2 * i + 1] - x[2 * n + i])  # y_i >= r_i
        c.append(1.0 - x[2 * i + 1] - x[2 * n + i])  # y_i + r_i <= 1

    # Non-overlap constraints using squared distances
    for i in range(n):
        for j in range(i + 1, n):
            dx = x[2 * i] - x[2 * j]
            dy = x[2 * i + 1] - x[2 * j + 1]
            d2 = dx * dx + dy * dy
            rs = x[2 * n + i] + x[2 * n + j]
            c.append(d2 - rs * rs)

    # Non-negative radii
    for i in range(n):
        c.append(x[2 * n + i])

    return np.array(c)


def make_initial_grid(perturbation_seed=0):
    """Create initial configuration: perturbed 5x5 grid + 1 extra circle."""
    rng = np.random.RandomState(perturbation_seed)
    centers = []
    for i in range(5):
        for j in range(5):
            cx = 0.1 + 0.2 * i + rng.uniform(-0.005, 0.005)
            cy = 0.1 + 0.2 * j + rng.uniform(-0.005, 0.005)
            centers.append([cx, cy])
    # Extra circle placed in a gap
    centers.append([0.2 + rng.uniform(-0.01, 0.01), 0.2 + rng.uniform(-0.01, 0.01)])
    centers = np.array(centers)
    # Start with slightly smaller radii to ensure feasibility
    radii = np.array([0.085] * 25 + [0.02])
    return centers, radii


def make_hex_initial(perturbation_seed=0):
    """Create hexagonal-ish initial configuration."""
    rng = np.random.RandomState(perturbation_seed + 100)
    centers = []
    # Try 5 rows with varying numbers: 5, 5, 6, 5, 5 = 26
    row_counts = [5, 5, 6, 5, 5]
    row_y = 0.12
    for row, count in enumerate(row_counts):
        offset = (count - 5) * 0.1
        row_x = 0.1 + offset
        for col in range(count):
            cx = row_x + 0.2 * col + rng.uniform(-0.005, 0.005)
            cy = row_y + rng.uniform(-0.005, 0.005)
            cx = np.clip(cx, 0.05, 0.95)
            cy = np.clip(cy, 0.05, 0.95)
            centers.append([cx, cy])
        row_y += 0.22 + rng.uniform(-0.01, 0.01)
    centers = np.array(centers[:N])
    radii = np.array([0.08] * N)
    return centers, radii


def make_corner_heavy_initial(perturbation_seed=0):
    """Place larger circles near corners and smaller in center."""
    rng = np.random.RandomState(perturbation_seed + 200)
    centers = []
    radii = []
    # 4 large corner circles
    for cx, cy in [(0.15, 0.15), (0.85, 0.15), (0.15, 0.85), (0.85, 0.85)]:
        centers.append([cx + rng.uniform(-0.01, 0.01), cy + rng.uniform(-0.01, 0.01)])
        radii.append(0.13)
    # Remaining 22 circles in a grid-like pattern
    for i in range(4):
        for j in range(4):
            cx = 0.3 + 0.12 * i + rng.uniform(-0.005, 0.005)
            cy = 0.3 + 0.12 * j + rng.uniform(-0.005, 0.005)
            centers.append([cx, cy])
            radii.append(0.07)
    # Fill remaining spots
    remaining = N - len(centers)
    for _ in range(remaining):
        cx = rng.uniform(0.1, 0.9)
        cy = rng.uniform(0.1, 0.9)
        centers.append([cx, cy])
        radii.append(0.05)
    centers = np.array(centers[:N])
    radii = np.array(radii[:N])
    return centers, radii


def optimize_packing(x0):
    """Run SLSQP optimization from given starting point."""
    bounds = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
    constraints = [{"type": "ineq", "fun": constraint_vec}]

    result = minimize(
        objective,
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=constraints,
        options={"maxiter": 15000, "ftol": 1e-15, "disp": False},
    )
    return result


def extract_and_clean(result):
    """Extract centers, radii from optimization result and clean up."""
    centers = result.x[: 2 * N].reshape(N, 2)
    radii = result.x[2 * N :]
    radii = np.maximum(radii, 0.0)
    centers[:, 0] = np.clip(centers[:, 0], 0.0, 1.0)
    centers[:, 1] = np.clip(centers[:, 1], 0.0, 1.0)
    return centers, radii, np.sum(radii)


def iterative_refinement(centers, radii, max_iters=100):
    """
    Iteratively try to expand radii for circles that have room,
    while maintaining feasibility.
    """
    for iteration in range(max_iters):
        improved = False
        # Sort circles by radius (smallest first) to give them chance to grow
        indices = np.argsort(radii)
        for idx in indices:
            # Try to increase radius
            best_r = radii[idx]
            for step in np.linspace(0.001, 0.01, 10):
                test_r = radii[idx] + step
                # Check feasibility with this larger radius
                feasible = True
                cx, cy = centers[idx]
                if cx - test_r < -1e-10 or cx + test_r > 1 + 1e-10:
                    feasible = False
                if cy - test_r < -1e-10 or cy + test_r > 1 + 1e-10:
                    feasible = False
                if feasible:
                    for other in range(N):
                        if other == idx:
                            continue
                        dx = cx - centers[other, 0]
                        dy = cy - centers[other, 1]
                        dist = np.sqrt(dx * dx + dy * dy)
                        if dist < test_r + radii[other] - 1e-10:
                            feasible = False
                            break
                if feasible:
                    best_r = test_r
                else:
                    break
            if best_r > radii[idx] + 1e-8:
                radii[idx] = best_r
                improved = True
        if not improved:
            break
    return centers, radii, np.sum(radii)


def run_packing():
    """Main function to solve the circle packing problem."""
    best_centers = None
    best_radii = None
    best_sum = 0.0

    # Try multiple initial configurations
    initial_configs = []
    for seed in range(20):
        c, r = make_initial_grid(seed)
        initial_configs.append((c, r))

    for seed in range(5):
        c, r = make_hex_initial(seed)
        initial_configs.append((c, r))

    for seed in range(3):
        c, r = make_corner_heavy_initial(seed)
        initial_configs.append((c, r))

    # Optimize from each configuration
    for c, r in initial_configs:
        x0 = np.concatenate([c.ravel(), r])
        result = optimize_packing(x0)
        opt_c, opt_r, opt_sum = extract_and_clean(result)

        # Run iterative refinement
        opt_c, opt_r, opt_sum = iterative_refinement(opt_c, opt_r, max_iters=50)

        if opt_sum > best_sum:
            best_sum = opt_sum
            best_centers = opt_c.copy()
            best_radii = opt_r.copy()

    # Also try a second round of optimization using the best solution as starting point
    if best_centers is not None:
        x0 = np.concatenate([best_centers.ravel(), best_radii])
        result = optimize_packing(x0)
        opt_c, opt_r, opt_sum = extract_and_clean(result)
        opt_c, opt_r, opt_sum = iterative_refinement(opt_c, opt_r, max_iters=50)

        if opt_sum > best_sum:
            best_sum = opt_sum
            best_centers = opt_c.copy()
            best_radii = opt_r.copy()

    # Third round with slight perturbation
    if best_centers is not None:
        rng = np.random.RandomState(42)
        perturbed_c = best_centers + rng.uniform(-0.002, 0.002, (N, 2))
        perturbed_c = np.clip(perturbed_c, 0.01, 0.99)
        perturbed_r = best_radii * (1 + rng.uniform(-0.02, 0.02, N))
        perturbed_r = np.maximum(perturbed_r, 0.001)

        x0 = np.concatenate([perturbed_c.ravel(), perturbed_r])
        result = optimize_packing(x0)
        opt_c, opt_r, opt_sum = extract_and_clean(result)
        opt_c, opt_r, opt_sum = iterative_refinement(opt_c, opt_r, max_iters=50)

        if opt_sum > best_sum:
            best_sum = opt_sum
            best_centers = opt_c.copy()
            best_radii = opt_r.copy()

    # Final small perturbation and optimization
    if best_centers is not None:
        rng = np.random.RandomState(99)
        perturbed_c = best_centers + rng.uniform(-0.001, 0.001, (N, 2))
        perturbed_c = np.clip(perturbed_c, 0.01, 0.99)
        perturbed_r = best_radii * (1 + rng.uniform(-0.01, 0.01, N))
        perturbed_r = np.maximum(perturbed_r, 0.001)

        x0 = np.concatenate([perturbed_c.ravel(), perturbed_r])
        result = optimize_packing(x0)
        opt_c, opt_r, opt_sum = extract_and_clean(result)
        opt_c, opt_r, opt_sum = iterative_refinement(opt_c, opt_r, max_iters=50)

        if opt_sum > best_sum:
            best_sum = opt_sum
            best_centers = opt_c.copy()
            best_radii = opt_r.copy()

    return best_centers, best_radii, best_sum
```
