import numpy as np
from scipy.optimize import minimize

def calculate_penalty(centers, r):
    """
    Calculates the penalty for a given configuration and radius.
    Penalty comes from overlaps between circles.
    Boundary constraints are handled by bounds in the optimizer.
    """
    N = centers.shape[0]
    # Compute all pairwise distances
    # centers: (N, 2)
    # diffs: (N, N, 2)
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    
    # Set diagonal to infinity to ignore self-overlap
    np.fill_diagonal(dists, np.inf)
    
    # Circle-circle overlap penalty
    # We want dist >= 2*r
    circle_overlaps = np.maximum(0.0, 2*r - dists)
    return np.sum(circle_overlaps**2)

def objective_function(x, r, N):
    """
    Wrapper for scipy.optimize.minimize.
    """
    centers = x.reshape(N, 2)
    return calculate_penalty(centers, r)

def get_hex_grid_guess(N, r):
    """
    Generates a hexagonal grid of N points, normalized to [0,1],
    then scaled to [r, 1-r].
    """
    pts = []
    # Generate a dense set of hexagonal points
    for row in range(30):
        for col in range(30):
            x = col * 1.0
            y = row * np.sqrt(3)/2
            if row % 2 == 1:
                x += 0.5
            
            pts.append([x, y])
            if len(pts) >= N:
                break
        if len(pts) >= N:
            break
            
    pts = np.array(pts[:N])
    
    # Normalize to [0, 1]
    if pts.size > 0:
        pts = pts - pts.min(axis=0)
        max_val = pts.max(axis=0)
        # Avoid division by zero
        max_val[max_val == 0] = 1.0
        pts = pts / max_val
        
    # Scale to [r, 1-r]
    pts = pts * (1.0 - 2.0 * r) + r
    
    return pts

def run_packing():
    N = 26
    
    # Binary search for maximum r
    r_lo = 0.01
    r_hi = 0.20 # Safe upper bound
    best_r = 0.0
    best_centers = None
    
    # Number of binary search iterations
    num_iter = 25
    
    for _ in range(num_iter):
        r_mid = (r_lo + r_hi) / 2.0
        
        # Generate initial guess
        initial_guess = get_hex_grid_guess(N, r_mid)
        
        # Bounds for coordinates
        bounds = [(r_mid, 1.0 - r_mid)] * (2 * N)
        
        # Optimization
        # We use multiple restarts to avoid local minima
        best_penalty = float('inf')
        best_result_centers = None
        
        # Flatten initial guess for scipy
        flat_guess = initial_guess.flatten()
        
        # Restart 1: Hex grid
        res = minimize(
            objective_function,
            flat_guess,
            args=(r_mid, N),
            method='L-BFGS-B',
            bounds=bounds,
            options={'ftol': 1e-15, 'maxiter': 2000}
        )
        if res.fun < best_penalty:
            best_penalty = res.fun
            best_result_centers = np.array(res.x).reshape(N, 2)
            
        # Restart 2: Random perturbation
        noise = np.random.normal(0, 0.05 * (1.0 - 2.0 * r_mid), size=initial_guess.shape)
        flat_guess_noisy = (initial_guess + noise).flatten()
        # Clip to bounds
        flat_guess_noisy = np.clip(flat_guess_noisy, r_mid, 1.0 - r_mid)
        
        res = minimize(
            objective_function,
            flat_guess_noisy,
            args=(r_mid, N),
            method='L-BFGS-B',
            bounds=bounds,
            options={'ftol': 1e-15, 'maxiter': 2000}
        )
        if res.fun < best_penalty:
            best_penalty = res.fun
            best_result_centers = np.array(res.x).reshape(N, 2)

        # Feasibility threshold
        # The penalty is sum of squared overlaps.
        # We want individual overlaps to be very small.
        if best_penalty < 1e-8:
            best_r = r_mid
            best_centers = best_result_centers
            r_lo = r_mid
        else:
            r_hi = r_mid

    if best_centers is None:
        # Fallback if optimization failed
        best_centers = get_hex_grid_guess(N, 0.1)
        best_r = 0.1
        
    radii = np.array([best_r] * N)
    return best_centers, radii, best_r * N
