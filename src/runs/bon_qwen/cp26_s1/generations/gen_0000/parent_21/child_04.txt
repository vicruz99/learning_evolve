import numpy as np
from scipy.optimize import minimize

N = 26

def objective(v):
    """Objective function: negative sum of radii (to maximize sum)."""
    radii = v[2::3]
    return -np.sum(radii)

def constraints(v):
    """Constraint function: returns an array of values that must be >= 0."""
    res = []
    X = v[0::3]
    Y = v[1::3]
    R = v[2::3]
    
    # Boundary constraints: circles inside [0,1]^2
    res.extend(X - R)
    res.extend(1.0 - X - R)
    res.extend(Y - R)
    res.extend(1.0 - Y - R)
    # Radii must be positive
    res.extend(R - 0.001)
    
    # Pairwise non-overlap constraints
    for i in range(N):
        xi, yi, ri = X[i], Y[i], R[i]
        for j in range(i + 1, N):
            dx = xi - X[j]
            dy = yi - Y[j]
            dist2 = dx*dx + dy*dy
            rad_sum = ri + R[j]
            res.append(dist2 - rad_sum**2)
            
    return np.array(res)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # 1. Initialize with a hexagonal lattice
    centers = []
    sx, sy = 0.22, 0.22 * np.sqrt(3) / 2
    row_counts = [5, 4, 5, 4, 5, 3]  # Sums to 26
    
    row_idx = 0
    for i in range(6):
        for j in range(row_counts[i]):
            if len(centers) >= N:
                break
            x = sx/2 + j * sx + (i%2)*sx/2
            y = sy/2 + i * sy
            if x < 0.95 and y < 0.95:
                centers.append([x, y])
        if len(centers) >= N:
            break
            
    centers = np.array(centers[:N])
    radii = np.full(N, 0.09)
    
    # Flatten variables: [x1, y1, r1, x2, y2, r2, ...]
    x0 = np.zeros(3 * N)
    for i in range(N):
        x0[3*i] = centers[i, 0]
        x0[3*i+1] = centers[i, 1]
        x0[3*i+2] = radii[i]
        
    bounds = [(0.0, 1.0) for _ in range(3*N)]
    cons = {'type': 'ineq', 'fun': constraints}
    
    # 2. Optimize
    res = minimize(
        objective, 
        x0, 
        bounds=bounds, 
        constraints=cons, 
        method='SLSQP', 
        options={'maxiter': 2000, 'ftol': 1e-10}
    )
    
    # 3. Extract results
    final_centers = np.zeros((N, 2))
    final_radii = np.zeros(N)
    for i in range(N):
        final_centers[i, 0] = res.x[3*i]
        final_centers[i, 1] = res.x[3*i+1]
        final_radii[i] = res.x[3*i+2]
        
    # 4. Validation and safe shrinkage to ensure strict compliance
    for _ in range(100):
        valid = True
        # Check boundaries
        for i in range(N):
            if (final_centers[i,0] - final_radii[i] < -1e-7 or 
                final_centers[i,0] + final_radii[i] > 1.0 + 1e-7 or
                final_centers[i,1] - final_radii[i] < -1e-7 or 
                final_centers[i,1] + final_radii[i] > 1.0 + 1e-7):
                valid = False
                break
        if not valid:
            final_radii *= 0.998
            continue
            
        # Check overlaps
        for i in range(N):
            for j in range(i+1, N):
                dx = final_centers[i,0] - final_centers[j,0]
                dy = final_centers[i,1] - final_centers[j,1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < final_radii[i] + final_radii[j] - 1e-7:
                    valid = False
                    break
            if not valid:
                break
                
        if not valid:
            final_radii *= 0.998
        else:
            break
            
    return final_centers, final_radii, float(np.sum(final_radii))
