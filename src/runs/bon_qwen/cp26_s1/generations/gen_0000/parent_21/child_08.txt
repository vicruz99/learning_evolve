import numpy as np
from scipy.optimize import linprog, differential_evolution

def solve_radii(centers):
    """
    Given fixed centers, solve the LP to find radii that maximize sum(r_i)
    while satisfying boundary and non-overlap constraints.
    """
    n = centers.shape[0]
    
    # Pairwise distances
    dists = np.linalg.norm(centers[:, None] - centers[None, :], axis=2)
    
    # Boundary limits for each circle
    x, y = centers[:, 0], centers[:, 1]
    b_upper = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
    
    # Construct LP matrices: A_ub @ r <= b_ub
    num_constraints = n + n * (n - 1) // 2
    A_ub = np.zeros((num_constraints, n))
    b_ub = np.zeros(num_constraints)
    
    # Boundary constraints: r_i <= b_upper[i]
    for i in range(n):
        A_ub[i, i] = 1.0
        b_ub[i] = b_upper[i]
        
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    idx = n
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1
            
    # Objective: maximize sum(r_i) => minimize -sum(r_i)
    c_obj = -np.ones(n)
    bounds = [(0.0, None)] * n  # r_i >= 0
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return -res.fun, res.x
    else:
        return 0.0, np.zeros(n)

def objective(center_flat):
    """Objective function for differential evolution: negative sum of optimal radii."""
    centers = center_flat.reshape(-1, 2)
    sum_r, _ = solve_radii(centers)
    return -sum_r

def run_packing():
    n = 26
    # Bounds for each center coordinate: x, y in [0, 1]
    bounds = [(0.0, 1.0)] * (2 * n)
    
    # Global optimization of centers
    result = differential_evolution(
        objective,
        bounds,
        popsize=12,
        maxiter=300,
        tol=1e-7,
        seed=42,
        workers=1,
        mutation=(0.5, 1.0),
        recombination=0.9,
        polish=True  # Refine best solution with local optimizer
    )
    
    centers = result.x.reshape(-1, 2)
    
    # Final LP solve to get exact radii for the optimized centers
    sum_r, radii = solve_radii(centers)
    
    return centers, radii, float(sum_r)
