import numpy as np
from scipy.optimize import minimize

def compute_objective(vars, n, penalty_weight):
    centers = vars[:2*n].reshape(n, 2)
    radii = vars[2*n:]
    
    obj = -np.sum(radii)
    p = penalty_weight
    
    # Boundary constraints
    for i in range(n):
        r = radii[i]
        x, y = centers[i]
        if x < r:
            obj += p * (r - x)**2
        if x > 1 - r:
            obj += p * (x + r - 1)**2
        if y < r:
            obj += p * (r - y)**2
        if y > 1 - r:
            obj += p * (y + r - 1)**2
            
    # Pairwise non-overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            if dist < radii[i] + radii[j]:
                obj += p * (radii[i] + radii[j] - dist)**2
    return obj

def get_max_radii(centers):
    n = centers.shape[0]
    radii = np.zeros(n)
    
    # Initialize with maximum possible radius constrained by boundaries
    for i in range(n):
        x, y = centers[i]
        radii[i] = min(x, 1 - x, y, 1 - y)
        
    # Iteratively enforce pairwise constraints: r_i + r_j <= dist_ij
    for _ in range(60):
        changed = False
        for i in range(n):
            new_r = radii[i]
            for j in range(n):
                if i == j:
                    continue
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                limit = dist - radii[j]
                if limit < new_r:
                    new_r = limit
            if new_r < radii[i] - 1e-10:
                radii[i] = new_r
                changed = True
        if not changed:
            break
    return np.maximum(radii, 0.0)

            idx += 1
    centers[idx] = [0.5 + np.random.normal(0, 0.02), 0.5 + np.random.normal(0, 0.02)]
    
    initial_radii = get_max_radii(centers)
    x0 = np.concatenate([centers.ravel(), initial_radii])

    bounds = [(0.001, 0.999)] * (2 * n) + [(0.001, 0.5)] * n

    penalty = 100.0
    
    # Progressive penalty method to gradually enforce constraints
    for _ in range(5):

    return final_centers, final_radii, float(np.sum(final_radii))
