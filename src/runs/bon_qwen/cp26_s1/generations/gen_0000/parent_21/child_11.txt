def calculate_energy(centers, radii, penalty_weight=10000.0):
    """
    Calculates the objective function value to be minimized.
    Objective: Maximize sum of radii -> Minimize -sum(radii).
    Constraints enforced via penalty functions.
    
    Args:
        centers: np.array of shape (N, 2)
        radii: np.array of shape (N,)
        penalty_weight: Weight for constraint violations
    
    Returns:
        energy: float
    """
    n = centers.shape[0]
    
    # Objective: minimize negative sum of radii
    obj = -np.sum(radii)
    
    # 1. Boundary Penalties
    # Circles must be inside [0, 1] x [0, 1]
    # x - r >= 0  => violation max(0, r - x)
    # x + r <= 1  => violation max(0, r + x - 1)
    # y - r >= 0
    # y + r <= 1
    
    x = centers[:, 0]
    y = centers[:, 1]
    r = radii
    
    # Boundary violations
    # Using np.maximum to create a smooth-ish penalty (squared)
    pen_x_left = np.maximum(r - x, 0)
    pen_x_right = np.maximum(r + x - 1, 0)
    pen_y_bottom = np.maximum(r - y, 0)
    pen_y_top = np.maximum(r + y - 1, 0)
    
    boundary_penalty = np.sum(pen_x_left**2 + pen_x_right**2 + 
                              pen_y_bottom**2 + pen_y_top**2)
    
    # 2. Overlap Penalties
    # Distance between centers must be >= sum of radii
    # dist >= r_i + r_j  => violation max(0, r_i + r_j - dist)
    
    # Compute pairwise distances efficiently
    # Centers shape: (N, 2)
    # diff shape: (N, N, 2)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    # Sum of radii matrix
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    
    # Overlap amount (positive means overlap)
    overlap = r_sum - dists
    
    # Ignore diagonal (distance to self is 0, r+r > 0, but we don't care)
    # Actually dists[i,i] = 0, r_sum[i,i] = 2r[i]. Overlap is 2r[i].
    # We should zero out diagonal.
    np.fill_diagonal(overlap, 0)
    
    # Keep only positive overlaps
    positive_overlap = np.maximum(overlap, 0)
    
    # Sum of squared overlaps
    overlap_penalty = np.sum(positive_overlap**2)
    
    # Total Energy
    energy = obj + penalty_weight * (boundary_penalty + overlap_penalty)
    
    return energy
