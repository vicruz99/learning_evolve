import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    
    Returns:
        centers: np.array of shape (26, 2)
        radii: np.array of shape (26,)
        sum_radii: float
    """
    n = 26
    
    # 1. Initialize positions in a hexagonal lattice pattern
    # We aim to fit 26 circles. A 5-row staggered pattern works well.
    # Row counts: 5, 6, 5, 6, 4 -> Sum = 26.
    # This pattern is denser than a simple grid.
    
    centers = []
    radii = []
    
    # Estimated initial radius based on fitting 5 rows of ~5 circles
    # Height constraint: 2r + 4 * sqrt(3)*r <= 1 => r(2 + 4*1.732) <= 1 => r <= 1/8.928 ~= 0.112
    # Width constraint for 6 circles: 6 * 2r <= 1 => r <= 0.0833 (if aligned)
    # With staggering, effective width is slightly less, but let's start conservative.
    r_init = 0.09 
    
    row_y = r_init
    row_height = np.sqrt(3) * r_init
    
    # Define row configurations (count, shift)
    # Shift 0: aligned with y-axis (x = r, 3r, 5r...)
    # Shift 1: shifted by r (x = 2r, 4r, 6r...)
    rows_config = [
        (5, 0),
        (6, 1),
        (5, 0),
        (6, 1),
        (4, 0)
    ]
    
    for count, shift in rows_config:
        for k in range(count):
            # x coordinate: start at r_init + k*2*r_init + shift*r_init
            x = r_init + k * 2 * r_init + shift * r_init
            y = row_y
            centers.append([x, y])
            radii.append(r_init)
        
        row_y += row_height
        
    centers = np.array(centers)
    radii = np.array(radii)
    
    # 2. Optimization Setup
    # Variables: [x1, y1, r1, x2, y2, r2, ...]
    # Total 78 variables.
    
    # We map 1D array to centers and radii
    # indices: 0,1 -> c0; 2 -> r0; 3,4 -> c1; 5 -> r1 ...
    # Actually, easier to store as [x, y, r] for each circle.
    # Shape (26, 3) -> flatten.
    
    x0 = np.zeros(n * 3)
    for i in range(n):
        x0[3*i] = centers[i, 0]
        x0[3*i+1] = centers[i, 1]
        x0[3*i+2] = radii[i]
        
    # Bounds: x in [0, 1], y in [0, 1], r in [0, 0.5]
    bounds = []
    for i in range(n):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((1e-6, 0.5)) # r (small epsilon to avoid 0)
        
    def objective(vars_flat):
        # Maximize sum of radii -> Minimize negative sum
        radii_curr = vars_flat[2::3]
        return -np.sum(radii_curr)
        
    def gradient(vars_flat):
        grad = np.zeros_like(vars_flat)
        # Gradient of -sum(r) w.r.t r is -1
        grad[2::3] = -1.0
        return grad

    # Penalty function for constraints
    # Overlap: dist >= r_i + r_j
    # Boundary: r <= x <= 1-r => x-r >= 0, x+r <= 1 => r <= x, r <= 1-x
    # y-r >= 0, y+r <= 1 => r <= y, r <= 1-y
    
    def constraint_violation(vars_flat):
        centers_curr = vars_flat[:-1].reshape(n, 3) # Wait, vars_flat is flat
        # Re-extract
        c = vars_flat.reshape(n, 3)
        x = c[:, 0]
        y = c[:, 1]
        r = c[:, 2]
        
        violation = 0.0
        
        # 1. Boundary constraints
        # x - r >= 0
        diff = x - r
        diff = diff[diff < 0]
        if len(diff) > 0:
            violation += np.sum(diff**2)
            
        # x + r <= 1 => 1 - x - r >= 0
        diff = 1.0 - x - r
        diff = diff[diff < 0]
        if len(diff) > 0:
            violation += np.sum(diff**2)
            
        # y - r >= 0
        diff = y - r
        diff = diff[diff < 0]
        if len(diff) > 0:
            violation += np.sum(diff**2)
            
        # y + r <= 1
        diff = 1.0 - y - r
        diff = diff[diff < 0]
        if len(diff) > 0:
            violation += np.sum(diff**2)
            
        # 2. Overlap constraints
        # dist^2 >= (r_i + r_j)^2
        # To avoid sqrt, check dist^2 - (r_i+r_j)^2 >= 0
        # We iterate all pairs
        for i in range(n):
            for j in range(i + 1, n):
                dx = x[i] - x[j]
                dy = y[i] - y[j]
                dist_sq = dx*dx + dy*dy
                min_dist = r[i] + r[j]
                if dist_sq < min_dist*min_dist:
                    # Penalty proportional to overlap squared?
                    # Overlap amount = min_dist - sqrt(dist_sq)
                    # But sqrt is costly. Let's use a simple penalty.
                    # If dist < min_dist, penalty = (min_dist - dist)^2
                    dist = np.sqrt(dist_sq)
                    overlap = min_dist - dist
                    if overlap > 0:
                        violation += overlap**2
                        
        return violation

    # We can use a simple penalty method or augmented Lagrangian.
    # Since we want a robust solution, let's use the 'SLSQP' method which handles constraints directly?
    # But defining 300+ constraints in SLSQP is verbose.
    # Instead, we add the penalty to the objective function.
    # Objective = -sum(r) + lambda * violation.
    # We need to find a lambda large enough.
    
    # Let's try a simple iterative penalty approach or just a large lambda.
    # Since it's a local optimization, starting with a valid config helps.
    
    # Let's wrap objective to include penalty
    penalty_weight = 100.0
    
    def penalized_objective(vars_flat):
        obj_val = objective(vars_flat)
        viol = constraint_violation(vars_flat)
        return obj_val + penalty_weight * viol
        
    def penalized_gradient(vars_flat):
        # Numerical gradient for the whole thing? 
        # Or approximate. 
        # Let's rely on scipy's numerical differentiation for the penalty part if we don't provide grad.
        # But providing grad for -sum(r) is easy.
        # Let's just not provide gradient for the penalized objective to avoid errors, 
        # or compute it numerically. 
        # scipy.minimize can do finite difference.
        return None # Let scipy handle it

    # Actually, providing the exact gradient for the main term is good.
    # But the penalty term gradient is complex.
    # Let's stick to default numerical differentiation for the penalized objective.
    
    # Optimization
    # Using L-BFGS-B as it respects bounds.
    
    result = minimize(penalized_objective, x0, method='L-BFGS-B', bounds=bounds, 
                      options={'maxiter': 1000, 'ftol': 1e-12, 'gtol': 1e-12})
    
    # Extract result
    c_opt = result.x.reshape(n, 3)
    centers_opt = c_opt[:, :2]
    radii_opt = c_opt[:, 2]
    
    # Final check and cleanup
    # If optimization failed to satisfy constraints perfectly, we might need to clamp.
    # But with penalty, it should be close.
    
    # Verify validity (internal check, not required by return but good for sanity)
    # The validate_packing function will be run externally.
    
    sum_r = np.sum(radii_opt)
    
    return centers_opt, radii_opt, sum_r
