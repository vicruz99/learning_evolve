import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26

    # 1. Initialization: Staggered Hexagonal Grid
    # Distribute 26 circles into rows to fit tightly.
    # Pattern: 5, 6, 5, 6, 4 (Sum = 26)
    row_counts = [5, 6, 5, 6, 4]
    
    # Initial parameters
    r_init = 0.09
    centers = []
    radii = np.full(n, r_init)

    current_idx = 0
    for r_idx, count in enumerate(row_counts):
        y = (r_idx + 1) * 2 * r_init  # Vertical spacing
        # Stagger rows with even index (1, 3...) to fit between previous
        x_start = 2 * r_init if r_idx % 2 == 1 else r_init
        
        for i in range(count):
            if current_idx < n:
                x = x_start + i * 2 * r_init
                centers.append([x, y])
                current_idx += 1

    centers = np.array(centers[:n])
    radii = np.array([r_init] * n)

    # 2. Optimization Function
    # We will optimize variables [x_0, y_0, r_0, ..., x_25, y_25, r_25]
    # Objective: Maximize sum(radii) -> Minimize -sum(radii)
    
    def objective(vars_):
        r = vars_[2::3]
        return -np.sum(r)

    # Constraints for boundary and non-overlap
    cons = []
    
    # Boundary Constraints
    for i in range(n):
        x, y, r = vars_[3*i], vars_[3*i+1], vars_[3*i+2]
        # x >= r  => r - x <= 0
        cons.append({'type': 'ineq', 'fun': lambda v, idx=i: v[3*idx] - v[3*idx+2]})
        # x + r <= 1 => 1 - (x + r) >= 0
        cons.append({'type': 'ineq', 'fun': lambda v, idx=i: 1 - (v[3*idx] + v[3*idx+2])})
        # y >= r
        cons.append({'type': 'ineq', 'fun': lambda v, idx=i: v[3*idx+1] - v[3*idx+2]})
        # y + r <= 1
        cons.append({'type': 'ineq', 'fun': lambda v, idx=i: 1 - (v[3*idx+1] + v[3*idx+2])})
        
    # Non-overlap Constraints
    for i in range(n):
        for j in range(i + 1, n):
            def dist_constraint(v, i=i, j=j):
                x1, y1, r1 = v[3*i], v[3*i+1], v[3*i+2]
                x2, y2, r2 = v[3*j], v[3*j+1], v[3*j+2]
                dist_sq = (x1 - x2)**2 + (y1 - y2)**2
                min_dist = r1 + r2
                return dist_sq - min_dist**2

            cons.append({'type': 'ineq', 'fun': dist_constraint})

    # 3. Run Optimization
    # SLSQP is generally good for this type of problem
    x0 = np.concatenate([centers.T.flatten(), radii])
    
    # Bounds for variables (x, y in [0, 1], r >= 0)
    bounds = []
    for i in range(n):
        bounds.append((0, 1)) # x
        bounds.append((0, 1)) # y
        bounds.append((0, 1)) # r

    result = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, 
                      options={'maxiter': 1000, 'ftol': 1e-9})

    # Extract results
    final_centers = np.zeros((n, 2))
    final_radii = np.zeros(n)
    for i in range(n):
        final_centers[i] = [result.x[3*i], result.x[3*i+1]]
        final_radii[i] = result.x[3*i+2]

    return final_centers, final_radii, np.sum(final_radii)
