import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # Set seed for reproducibility
    np.random.seed(42)

    # --- Initialization ---
    # Hexagonal-ish grid initialization for better packing density
    centers = np.zeros((n, 2))
    row_lengths = [6, 5, 6, 5, 4] # Sum = 26
    
    idx = 0
    # y-coordinates for 5 rows, evenly spaced with margin
    y_coords = np.linspace(0.15, 0.85, 5)
    
    for i, length in enumerate(row_lengths):
        # x-coordinates for the current row
        # For shifted rows (odd index in hex grid), we offset x
        if i % 2 == 1:
            # Shifted row
            x_coords = np.linspace(0.08 + 0.083, 0.92 - 0.083, length)
        else:
            x_coords = np.linspace(0.08, 0.92, length)
            
        centers[idx : idx + length, 0] = x_coords
        centers[idx : idx + length, 1] = y_coords[i]
        idx += length

    # Initial small radii to avoid immediate massive penalties
    radii = np.ones(n) * 0.05
    
    # Concatenate centers and radii into a single parameter vector
    params_init = np.concatenate([centers.flatten(), radii])
    
    # --- Bounds ---
    # x, y in [0, 1], r in [0, 0.5]
    bounds = []
    for _ in range(n):
        bounds.extend([(0, 1), (0, 1)]) # x, y bounds
        bounds.append((0, 0.5))         # r bounds

    # --- Objective Function ---
    def objective(p):
        c = p[:2*n].reshape(n, 2)
        r = p[2*n:]
        
        # 1. Boundary Penalties
        # x - r >= 0  =>  penalty if r > x
        # x + r <= 1  =>  penalty if x + r > 1
        # Same for y
        pen_x_low = np.maximum(0, r - c[:, 0])**2
        pen_y_low = np.maximum(0, r - c[:, 1])**2
        pen_x_high = np.maximum(0, c[:, 0] + r - 1)**2
        pen_y_high = np.maximum(0, c[:, 1] + r - 1)**2
        
        pen_boundary = np.sum(pen_x_low + pen_y_low + pen_x_high + pen_y_high)
        
        # 2. Overlap Penalties
        # Distance between centers >= sum of radii
        # We compute pairwise distances using broadcasting
        # dists[i, j] = ||c_i - c_j||
        dists = np.linalg.norm(c[:, None, :] - c[None, :, :], axis=2)
        r_sum = r[:, None] + r[None, :]
        
        # Overlap occurs if r_sum > dists
        overlap = r_sum - dists
        
        # We only care about upper triangle (i < j) to avoid double counting
        # and self-comparisons.
        mask = np.triu(np.ones((n, n), dtype=bool), k=1)
        pen_overlap = np.sum((np.maximum(0, overlap)**2) * mask)
        
        # Total loss: negative sum of radii + weighted penalties
        # High weight ensures constraints are prioritized
        return -np.sum(r) + 10000 * (pen_boundary + pen_overlap)

    # --- Optimization ---
    res = minimize(
        objective, 
        params_init, 
        method='L-BFGS-B', 
        bounds=bounds, 
        options={'maxiter': 5000, 'ftol': 1e-12, 'gtol': 1e-12}
    )
    
    best_centers = res.x[:2*n].reshape(n, 2)
    best_radii = res.x[2*n:]
    
    # --- Post-processing / Validation Safety ---
    # Check for NaNs
    if np.isnan(best_centers).any() or np.isnan(best_radii).any():
        # Fallback to a very safe small packing if optimization fails
        best_centers = np.random.rand(n, 2) * 0.5 + 0.25
        best_radii = np.ones(n) * 0.01

    # Ensure radii are non-negative
    best_radii = np.maximum(best_radii, 0)
    
    # Calculate sum
    sum_radii = float(np.sum(best_radii))
    
    return best_centers, best_radii, sum_radii
