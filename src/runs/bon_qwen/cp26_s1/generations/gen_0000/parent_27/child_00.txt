import numpy as np
from scipy.optimize import minimize

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square

    Args:
        centers: np.array of shape (n, 2) with (x, y) coordinates
        radii: np.array of shape (n) with radius of each circle

    Returns:
        True if valid, False otherwise
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True

def hexagonal_initial_guess(n_circles):
    # Define a hexagonal layout pattern.
    # We want to distribute n_circles in rows.
    # For 26 circles, a 5-6-5-6-4 or 6-5-6-5-4 pattern works well.
    # Let's try to generate a pattern and then center/scale it.
    
    # Pattern: number of circles per row
    rows = [6, 5, 6, 5, 4]
    assert sum(rows) == n_circles, f"Pattern sum {sum(rows)} != {n_circles}"
    
    points = []
    y_current = 0
    dy = np.sqrt(3) # Relative vertical spacing for unit horizontal spacing 2
    
    for r_idx, count in enumerate(rows):
        # Shift x for alternating rows in hexagonal packing
        x_offset = 1.0 if r_idx % 2 == 1 else 0.0
        
        # Generate points for this row
        # Centers at x = x_offset + 0, 2, 4, ...
        # To center the row, we can compute range. 
        # A row of k circles with spacing 2 has width 2*(k-1).
        # We want to center this row around x=0? Or just generate and center later.
        
        # Let's generate relative coordinates
        xs = x_offset + 2.0 * np.arange(count)
        
        # To make it somewhat centered within the row logic, we don't strictly need to center yet
        # but we will center the whole set later.
        
        for x in xs:
            points.append([x, y_current])
            
        y_current += dy
        
    points = np.array(points)
    
    # Center and Scale
    # Find bounding box
    min_x, min_y = np.min(points, axis=0)
    max_x, max_y = np.max(points, axis=0)
    
    width = max_x - min_x
    height = max_y - min_y
    
    # We want to fit this into a square, leaving some margin for the radii
    # Initial radius guess. If spacing is 2, diameter is 2.
    # Scale factor s such that width*s <= 1.
    # But we also need space for radii.
    # Let's scale to fit in [0.1, 0.9] roughly to allow expansion.
    target_width = 0.8
    target_height = 0.8
    
    scale_x = target_width / width if width > 0 else 1
    scale_y = target_height / height if height > 0 else 1
    scale = min(scale_x, scale_y)
    
    points = (points - np.array([min_x, min_y])) * scale
    
    # Center in [0,1]
    min_x, min_y = np.min(points, axis=0)
    max_x, max_y = np.max(points, axis=0)
    current_w = max_x - min_x
    current_h = max_y - min_y
    
    shift_x = (1.0 - current_w) / 2.0 - min_x
    shift_y = (1.0 - current_h) / 2.0 - min_y
    
    points = points + np.array([shift_x, shift_y])
    
    # Initial radii
    # Since we scaled to fit in 0.8 width, and original spacing was 2 (diameter 2),
    # the new diameter is roughly 2 * scale.
    # radius = scale.
    # But we might have compressed it more than needed if aspect ratio differs.
    # Let's estimate radius based on min distance.
    dists = []
    for i in range(len(points)):
        for j in range(i+1, len(points)):
            dist = np.linalg.norm(points[i] - points[j])
            if dist < 1e-6: continue
            dists.append(dist)
    min_dist = np.min(dists)
    initial_r = min_dist / 2.0 * 0.9 # 90% of half-distance to allow growth
    
    radii = np.full(n_circles, initial_r)
    
    return points, radii

    centers, radii = hexagonal_initial_guess(n)
    
    # Variables: [x1, y1, r1, x2, y2, r2, ..., x26, y26, r26]

            constraints.append(ri - xi)
            # x + r - 1 <= 0

            # r - y <= 0

            # y + r - 1 <= 0
            constraints.append(yi + ri - 1.0)

        final_x = x0

    final_c = final_x.reshape((n, 3))
    centers_out = final_c[:, :2]
    radii_out = final_c[:, 2]

    radii_out = np.maximum(radii_out, 0)
