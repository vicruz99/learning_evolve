import numpy as np
from scipy.optimize import minimize

def get_hex_grid_candidates(n, scale=1.0):
    """
    Generates candidate centers based on a hexagonal lattice.
    scale controls the density/spacing.
    """
    points = []
    # Try to cover the unit square with a hex grid
    # Vertical spacing h = sqrt(3)/2 * 2*scale ~ 1.732 * scale
    # Horizontal spacing 2*scale
    # We want roughly n points.
    # Area of square 1. Area per hex point ~ sqrt(3)/2 * (2r)^2? 
    # Let's just generate a dense grid and filter.
    
    # Heuristic spacing
    # For r ~ 0.1, diameter 0.2. 
    # Let's use a base spacing of 0.25 to ensure we get enough points
    spacing = 0.25 
    h = spacing * np.sqrt(3) / 2
    
    y = spacing # Start with some margin
    while y <= 1.0 - spacing:
        x = spacing
        # Shift every other row
        shift = (int((y - spacing) / h) % 2) * (spacing / 2)
        current_x = spacing + shift
        while current_x <= 1.0 - spacing:
            points.append((current_x, y))
            current_x += spacing
        y += h
    
    # If we don't have enough points, refine spacing
    while len(points) < n:
        spacing *= 0.8 # Make grid denser
        h = spacing * np.sqrt(3) / 2
        points = []
        y = spacing 
        row_idx = 0
        while y <= 1.0 - spacing:
            shift = (row_idx % 2) * (spacing / 2)
            current_x = spacing + shift
            while current_x <= 1.0 - spacing:
                points.append((current_x, y))
                current_x += spacing
            y += h
            row_idx += 1
            
    # Convert to numpy array
    pts = np.array(points)
    
    # If we have more than n, select n points.
    # Ideally, we want points distributed.
    if len(pts) > n:
        # Sort by x then y? Or just take first n?
        # Taking first n from generation order usually keeps distribution
        pts = pts[:n]
        
    return pts

def energy_function(centers, r):
    """
    Calculates the overlap energy for a given radius r and centers.
    Lower energy means less overlap.
    """
    n = centers.shape[0]
    energy = 0.0
    
    # Pairwise overlaps
    # Vectorized computation might be faster, but for n=26, loops are fine
    # or we can do it block-wise.
    # Let's do a simple loop for clarity and low overhead
    
    # Boundary violations
    # x - r >= 0 => r - x <= 0. Violation max(0, r - x)
    # 1 - x - r >= 0 => x + r - 1 <= 0. Violation max(0, x + r - 1)
    
    # Boundary penalty
    dx_neg = np.maximum(0, r - centers[:, 0])
    dx_pos = np.maximum(0, centers[:, 0] + r - 1)
    dy_neg = np.maximum(0, r - centers[:, 1])
    dy_pos = np.maximum(0, centers[:, 1] + r - 1)
    
    energy += np.sum(dx_neg**2) + np.sum(dx_pos**2) + np.sum(dy_neg**2) + np.sum(dy_pos**2)
    
    # Pairwise penalty
    # dist^2 >= (2r)^2 => (2r)^2 - dist^2 <= 0
    # Violation max(0, 2r - dist)
    # To avoid sqrt, we can check (2r)^2 - dist_sq, but max(0, 2r - dist) is smoother?
    # Actually max(0, 4r^2 - dist_sq) is not smooth at 0 either?
    # Let's use max(0, 2r - dist)^2.
    
    # Compute all pairwise distances
    # centers shape (n, 2)
    # diff shape (n, n, 2)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    # Lower triangle
    mask = np.tril(np.ones((n, n), dtype=bool), k=-1)
    pairwise_dists = dists[mask]
    
    violations = np.maximum(0, 2 * r - pairwise_dists)
    energy += np.sum(violations**2)
    
    return energy

def solve_packing_for_radius(r):
    """
    Tries to find a valid packing for 26 circles of radius r.
    Returns (success, centers)
    """
    n = 26
    
    # Initial guess
    init_centers = get_hex_grid_candidates(n)
    # If we got fewer than n, pad with random
    if init_centers.shape[0] < n:
        pad = np.random.rand(n - init_centers.shape[0], 2) * 0.5 + 0.25
        init_centers = np.vstack([init_centers, pad])
    
    # Bounds for centers: [r, 1-r] x [r, 1-r]
    # But since we optimize centers inside the function with penalty, 
    # we can use loose bounds [0, 1].
    bounds = [(0, 1)] * (2 * n)
    
    # Flatten centers for optimizer
    x0 = init_centers.flatten()
    
    # Optimization
    # We want to minimize energy
    def objective(x_flat):
        c = x_flat.reshape((n, 2))
        return energy_function(c, r)
    
    # Use L-BFGS-B for bound constrained optimization
    # It handles bounds well, but our bounds are just [0,1].
    # The energy function handles the 'r' margin via penalty.
    
    res = minimize(objective, x0, method='L-BFGS-B', bounds=bounds, 
                   options={'ftol': 1e-15, 'gtol': 1e-10, 'maxiter': 1000})
    
    # Check if solution is valid (energy close to 0)
    # Threshold depends on r. For r=0.1, overlaps are small.
    # Energy is sum of squared violations.
    # If energy < 1e-6, we consider it valid.
    
    best_centers = res.x.reshape((n, 2))
    
    # Verify energy
    final_energy = energy_function(best_centers, r)
    
    # Allow small numerical error
    is_valid = final_energy < 1e-6
    
    return is_valid, best_centers, final_energy
