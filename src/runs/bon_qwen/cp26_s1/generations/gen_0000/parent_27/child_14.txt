import numpy as np
from scipy.optimize import minimize


def compute_objective(x, n):
    """Compute negative sum of radii."""
    return -np.sum(x[2 * n:])


def compute_constraints(x, n):
    """Compute all constraint values."""
    c = x[:2 * n].reshape((n, 2))
    r = x[2 * n:]
    
    num_boundary = 4 * n
    num_overlap = n * (n - 1) // 2
    num_total = num_boundary + num_overlap
    cons = np.zeros(num_total)
    
    idx = 0
    # Boundary constraints
    for i in range(n):
        cons[idx] = c[i, 0] - r[i]
        idx += 1
        cons[idx] = 1 - c[i, 0] - r[i]
        idx += 1
        cons[idx] = c[i, 1] - r[i]
        idx += 1
        cons[idx] = 1 - c[i, 1] - r[i]
        idx += 1
    
    # Non-overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            d = np.sqrt(np.sum((c[i] - c[j]) ** 2) + 1e-16)
            cons[idx] = d - r[i] - r[j]
            idx += 1
    
    return cons


def build_bounds(n):
    """Build variable bounds."""
    bounds = []
    for i in range(2 * n):
        bounds.append((0, 1))
    for i in range(n):
        bounds.append((1e-9, 0.5))
    return bounds


def init_hexagonal(n, trial):
    """Initialize circles in hexagonal lattice pattern."""
    np.random.seed(trial * 77 + 13)
    
    # Different row configurations for 26 circles
    configs = [
        [6, 5, 6, 5, 4],
        [5, 6, 5, 6, 4],
        [6, 6, 5, 5, 4],
        [5, 5, 5, 6, 5],
        [6, 5, 5, 5, 5],
        [4, 6, 6, 6, 4],
        [5, 6, 6, 5, 4],
        [6, 6, 6, 4, 4],
    ]
    
    rows = configs[trial % len(configs)]
    
    r0 = 0.08 + trial * 0.001
    centers = np.zeros((n, 2))
    radii = np.full(n, r0)
    
    idx = 0
    for ri, count in enumerate(rows):
        y = r0 + ri * r0 * np.sqrt(3)
        row_width = (count - 1) * 2 * r0
        x_start = (1 - row_width) / 2
        offset = r0 if ri % 2 == 1 else 0
        x_start += offset
        for ci in range(count):
            x = x_start + ci * 2 * r0
            centers[idx] = [x, y]
            idx += 1
    
    # Add perturbation
    centers += np.random.randn(n, 2) * 0.003
    
    # Ensure valid initial configuration
    for i in range(n):
        max_r = min(centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])
        radii[i] = min(r0, max(max_r, 0.001))
        centers[i, 0] = np.clip(centers[i, 0], radii[i], 1 - radii[i])
        centers[i, 1] = np.clip(centers[i, 1], radii[i], 1 - radii[i])
    
    return centers, radii


def optimize_packing(n, init_centers, init_radii):
    """Optimize packing using SLSQP."""
    x0 = np.concatenate([init_centers.flatten(), init_radii])
    
    obj_func = lambda x: compute_objective(x, n)
    constr_func = lambda x: compute_constraints(x, n)
    
    cons = {'type': 'ineq', 'fun': constr_func}
    bounds = build_bounds(n)
    
    try:
        result = minimize(
            obj_func, x0, method='SLSQP', bounds=bounds,
            constraints=cons,
            options={'maxiter': 50000, 'ftol': 1e-16, 'disp': False}
        )
        centers = result.x[:2 * n].reshape((n, 2))
        radii = result.x[2 * n:]
        return True, centers, radii, np.sum(radii)
    except Exception:
        return False, init_centers, init_radii, np.sum(init_radii)


def adjust_validity(n, centers, radii):
    """Adjust configuration to ensure validity."""
    # Clip radii to boundary
    for i in range(n):
        max_r = min(
            centers[i, 0],
            1 - centers[i, 0],
            centers[i, 1],
            1 - centers[i, 1]
        )
        radii[i] = max(max_r, 0)
        centers[i, 0] = np.clip(centers[i, 0], radii[i], 1 - radii[i])
        centers[i, 1] = np.clip(centers[i, 1], radii[i], 1 - radii[i])
    
    # Iteratively resolve overlaps
    for iteration in range(500):
        overlap_found = False
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                d = np.sqrt(dx * dx + dy * dy)
                if d < radii[i] + radii[j]:
                    overlap_found = True
                    excess = (radii[i] + radii[j] - d) / 2 + 1e-10
                    radii[i] = max(0, radii[i] - excess)
                    radii[j] = max(0, radii[j] - excess)
        if not overlap_found:
            break
    
    # Final boundary check
    for i in range(n):
        max_r = min(
            centers[i, 0],
            1 - centers[i, 0],
            centers[i, 1],
            1 - centers[i, 1]
        )
        radii[i] = max(max_r, 0)
    
    return centers, radii


def run_packing():

    best_sum = 0.0
    best_centers = None
    best_radii = None

    num_trials = 25
    for trial in range(num_trials):

    if best_centers is None:

        best_sum = np.sum(best_radii)
    
    # Post-process for validity
    best_centers, best_radii = adjust_validity(n, best_centers, best_radii)
    
    return best_centers, best_radii, np.sum(best_radii)
