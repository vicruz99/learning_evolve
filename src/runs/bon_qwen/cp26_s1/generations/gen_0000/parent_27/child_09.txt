<strategy>
1. **Initial Configuration**: Start with hexagonal packing arrangements (5 rows with varying counts like 6-5-5-5-5), which is denser than square packing.

2. **Optimization Approach**: Use SLSQP (Sequential Least Squares Programming) from scipy to maximize sum of radii subject to:
   - Boundary constraints: each circle must fit inside [0,1]×[0,1]
   - Overlap constraints: distance between centers >= sum of radii

3. **Multiple Restarts**: Try different initial hexagonal configurations (varying spacing, row distributions) to escape local minima.

4. **Parameter Structure**: Concatenate centers (2*n) and radii (n) into a single parameter vector for scipy optimization.

5. **Key Insight**: The optimal solution likely has non-uniform radii - circles near corners/edges can be larger than interior ones. The optimizer will discover this naturally.
</strategy>

import numpy as np
from scipy.optimize import minimize


class CirclePackingProblem:
    """Optimization problem for circle packing in unit square."""

    def __init__(self, n):
        self.n = n

    def objective(self, params):
        """Maximize sum of radii (minimize negative sum)."""
        radii = params[2 * self.n:]
        return -np.sum(radii)

    def get_bounds(self):
        """Bounds for all parameters."""
        bounds = []
        for _ in range(self.n):
            bounds.append((0.0, 1.0))  # x coordinate
            bounds.append((0.0, 1.0))  # y coordinate
        for _ in range(self.n):
            bounds.append((0.0, 0.5))  # radius (max 0.5)
        return bounds

    def get_constraints(self):
        """All constraints for the optimization."""
        constraints = []

        # Boundary constraints for each circle
        for i in range(self.n):
            constraints.append({'type': 'ineq', 'fun': self._c_x_min, 'args': (i,)})
            constraints.append({'type': 'ineq', 'fun': self._c_x_max, 'args': (i,)})
            constraints.append({'type': 'ineq', 'fun': self._c_y_min, 'args': (i,)})
            constraints.append({'type': 'ineq', 'fun': self._c_y_max, 'args': (i,)})

        # Overlap constraints for each pair
        for i in range(self.n):
            for j in range(i + 1, self.n):
                constraints.append({'type': 'ineq', 'fun': self._c_overlap, 'args': (i, j)})

        return constraints

    def _c_x_min(self, params, i):
        return params[2 * i] - params[2 * self.n + i]

    def _c_x_max(self, params, i):
        return 1.0 - params[2 * i] - params[2 * self.n + i]

    def _c_y_min(self, params, i):
        return params[2 * i + 1] - params[2 * self.n + i]

    def _c_y_max(self, params, i):
        return 1.0 - params[2 * i + 1] - params[2 * self.n + i]

    def _c_overlap(self, params, i, j):
        dx = params[2 * i] - params[2 * j]
        dy = params[2 * i + 1] - params[2 * j + 1]
        dist = np.sqrt(dx * dx + dy * dy)
        r_sum = params[2 * self.n + i] + params[2 * self.n + j]
        return dist - r_sum

    def solve(self, x0):
        """Solve the optimization problem."""
        result = minimize(
            self.objective, x0, method='SLSQP',
            bounds=self.get_bounds(),
            constraints=self.get_constraints(),
            options={
                'maxiter': 5000,
                'ftol': 1e-14,
                'disp': False
            }
        )
        return result


def generate_hexagonal_packing(n, row_counts=None, spacing=0.16, y_spacing=0.18, offset_x=0.0, offset_y=0.0):
    """Generate initial hexagonal-like arrangement of circles."""
    if row_counts is None:
        row_counts = [6, 5, 5, 5, 5]

    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.04

    idx = 0
    for row_i, count in enumerate(row_counts):
        y = 0.08 + offset_y + row_i * y_spacing
        x_start = (1.0 - (count - 1) * spacing) / 2.0 + offset_x
        if row_i % 2 == 1:
            x_start += spacing / 2.0
        for col in range(count):
            centers[idx] = [x_start + col * spacing, y]
            idx += 1

    return centers, radii


def generate_grid_packing(n, rows=5, cols=5):
    """Generate a grid arrangement with one extra circle."""
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.04

    idx = 0
    for r in range(rows):
        for c in range(cols):
            if idx < n:
                x = (c + 0.5) / cols
                y = (r + 0.5) / rows
                centers[idx] = [x, y]
                idx += 1

    return centers, radii


def perturb_packing(centers, radii, magnitude=0.02):
    """Add small random perturbation to packing."""
    n = len(radii)
    new_centers = centers.copy()
    for i in range(n):
        dx = np.random.uniform(-magnitude, magnitude)
        dy = np.random.uniform(-magnitude, magnitude)
        new_centers[i, 0] = np.clip(new_centers[i, 0] + dx, radii[i], 1 - radii[i])
        new_centers[i, 1] = np.clip(new_centers[i, 1] + dy, radii[i], 1 - radii[i])
    return new_centers, radii


def run_packing():
    n = 26
    best_sum = -1.0
    best_centers = None
    best_radii = None

    # Define various initial configurations to try
    configs = []

    # Hexagonal-like packings with different row distributions
    hex_configs = [
        [6, 5, 5, 5, 5],
        [5, 6, 5, 5, 5],
        [5, 5, 6, 5, 5],
        [5, 5, 5, 6, 5],
        [5, 5, 5, 5, 6],
        [6, 6, 5, 5, 4],
        [4, 5, 5, 5, 7],
        [5, 5, 5, 4, 7],
        [7, 5, 5, 5, 4],
    ]

    spacings = [0.15, 0.16, 0.165, 0.17, 0.18]
    y_spacings = [0.17, 0.18, 0.185, 0.19, 0.20]

    for row_counts in hex_configs:
        if sum(row_counts) != n:
            continue
        for spacing in spacings:
            for y_spacing in y_spacings:
                configs.append({
                    'type': 'hex',
                    'row_counts': row_counts,
                    'spacing': spacing,
                    'y_spacing': y_spacing
                })

    # Try configurations (limit to reasonable number)
    np.random.seed(42)
    max_attempts = min(len(configs), 20)
    np.random.shuffle(configs)

    for attempt_idx in range(max_attempts):
        config = configs[attempt_idx]

        if config['type'] == 'hex':
            centers, radii = generate_hexagonal_packing(
                n,
                config['row_counts'],
                config['spacing'],
                config['y_spacing']
            )
        else:
            centers, radii = generate_grid_packing(n)

        # Add small perturbation
        centers, radii = perturb_packing(centers, radii, magnitude=0.005)

        x0 = np.concatenate([centers.flatten(), radii])

        problem = CirclePackingProblem(n)
        result = problem.solve(x0)

        if result.fun is not None:
            sum_radii = -result.fun
            if sum_radii > best_sum:
                best_sum = sum_radii
                best_centers = result.x[:2 * n].reshape(n, 2)
                best_radii = result.x[2 * n:].copy()

    # Refinement: take best solution and optimize again with perturbed starts
    if best_centers is not None:
        for refinement in range(10):
            np.random.seed(100 + refinement)
            perturbed_centers, _ = perturb_packing(best_centers, best_radii, magnitude=0.003)
            x0 = np.concatenate([perturbed_centers.flatten(), best_radii])

            problem = CirclePackingProblem(n)
            result = problem.solve(x0)

            if result.fun is not None:
                sum_radii = -result.fun
                if sum_radii > best_sum:
                    best_sum = sum_radii
                    best_centers = result.x[:2 * n].reshape(n, 2)
                    best_radii = result.x[2 * n:].copy()

    # Final clean-up: ensure validity
    if best_centers is None:
        centers, radii = generate_hexagonal_packing(n)
        best_centers = centers
        best_radii = radii
        best_sum = np.sum(radii)

    # Sort by radius for consistency
    order = np.argsort(best_radii)
    best_centers = best_centers[order]
    best_radii = best_radii[order]
    best_sum = np.sum(best_radii)

    return best_centers, best_radii, best_sum
