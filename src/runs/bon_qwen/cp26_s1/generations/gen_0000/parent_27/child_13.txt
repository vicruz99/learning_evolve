import numpy as np
from scipy.optimize import minimize

def compute_penalty(v, n):
    """Compute squared penalty for constraint violations."""
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    
    penalty = 0.0
    # Boundary constraints: circles must stay within [0,1]
    penalty += np.sum(np.maximum(0.0, r - x)**2)
    penalty += np.sum(np.maximum(0.0, r - (1.0 - x))**2)
    penalty += np.sum(np.maximum(0.0, r - y)**2)
    penalty += np.sum(np.maximum(0.0, r - (1.0 - y))**2)
    
    # Overlap constraints: distance between centers >= sum of radii
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist = np.sqrt(dx**2 + dy**2)
    np.fill_diagonal(dist, 1.0)  # Prevent self-comparison
    
    r_sum = r[:, None] + r[None, :]
    overlap = np.maximum(0.0, r_sum - dist)
    
    # Only count upper triangle to avoid double counting
    triu_mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    penalty += np.sum(overlap[triu_mask]**2)
    
    return penalty

def objective(v, n, lambda_pen):
    """Objective: minimize negative sum of radii + penalty."""
    r = v[2::3]
    pen = compute_penalty(v, n)
    return -np.sum(r) + lambda_pen * pen

def generate_hex_start(n):
    """Generate initial configuration based on hexagonal lattice."""
    v = np.zeros(3 * n)
    idx = 0
    row = 0
    while idx < n:
        cols = 6 if row % 2 == 0 else 5
        for col in range(cols):
            if idx >= n: break
            x = (col + 0.5) / 6.0
            if row % 2 == 1:
                x += 1.0 / 12.0
            y = (row + 0.5) / 5.0
            v[3*idx] = np.clip(x, 0.15, 0.85)
            v[3*idx+1] = np.clip(y, 0.15, 0.85)
            v[3*idx+2] = 0.045
            idx += 1
        row += 1
    return v

def generate_grid_start(n):
    """Generate initial configuration based on rectangular grid."""
    v = np.zeros(3 * n)
    idx = 0
    for i in range(6):
        for j in range(5):
            if idx >= n: break
            v[3*idx] = (i + 0.5) / 6.0
            v[3*idx+1] = (j + 0.5) / 5.0
            v[3*idx+2] = 0.045
            idx += 1
    return v

def run_packing():
    n = 26
    # Bounds: x,y in [0,1], r in [0, 0.5]
    bounds = [(0.0, 1.0)] * (3 * n)
    for i in range(n):
        bounds[2*i + 2] = (0.0, 0.5)
        
    # Prepare initial guesses
    starts = [generate_hex_start(n), generate_grid_start(n)]
    np.random.seed(2025)
    v_rand = np.zeros(3*n)
    for i in range(n):
        v_rand[3*i] = np.random.uniform(0.2, 0.8)
        v_rand[3*i+1] = np.random.uniform(0.2, 0.8)
        v_rand[3*i+2] = 0.035
    starts.append(v_rand)
    
    best_v = None
    best_sum_r = -1.0
    # Continuation schedule for penalty weight
    lambdas = [500, 2000, 10000, 50000, 200000]
    
    for v_init in starts:
        v = v_init.copy()
        for lam in lambdas:
            res = minimize(
                objective, v, args=(n, lam), method='L-BFGS-B', bounds=bounds,
                options={'maxiter': 3000, 'ftol': 1e-13, 'gtol': 1e-10}
            )
            v = res.x
        
        pen = compute_penalty(v, n)
        # Accept if constraints are sufficiently satisfied
        if pen < 1e-8:
            s_r = np.sum(v[2::3])
            if s_r > best_sum_r:
                best_sum_r = s_r
                best_v = v.copy()
                
    if best_v is None:
        best_v = starts[0].copy()
        
    # High-precision polishing pass
    v = best_v.copy()
    for lam in [500000, 2000000, 5000000]:
        res = minimize(
            objective, v, args=(n, lam), method='L-BFGS-B', bounds=bounds,
            options={'maxiter': 5000, 'ftol': 1e-15, 'gtol': 1e-12}
        )
