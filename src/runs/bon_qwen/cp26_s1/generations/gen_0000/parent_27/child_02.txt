import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26


def obj_func(params):
    n = N_CIRCLES
    return -np.sum(params[2 * n:])


def constr_x_min(params, i, n):
    return params[2 * i] - params[2 * n + i]


def constr_x_max(params, i, n):
    return 1.0 - params[2 * i] - params[2 * n + i]


def constr_y_min(params, i, n):
    return params[2 * i + 1] - params[2 * n + i]


def constr_y_max(params, i, n):
    return 1.0 - params[2 * i + 1] - params[2 * n + i]


def constr_overlap(params, i, j, n):
    x1 = params[2 * i]
    y1 = params[2 * i + 1]
    x2 = params[2 * j]
    y2 = params[2 * j + 1]
    r1 = params[2 * n + i]
    r2 = params[2 * n + j]
    return (x1 - x2) ** 2 + (y1 - y2) ** 2 - (r1 + r2) ** 2


def constr_r_pos(params, i, n):
    return params[2 * n + i]


def hexagonal_init(n, r):
    centers = []
    count = 0
    row = 0
    while count < n and row < 15:
        offset = (row % 2) * r
        col = 0
        while count < n and col < 15:
            x = col * 2 * r + r + offset
            y = row * r * np.sqrt(3) + r
            if x + r <= 1.0 and y + r <= 1.0 and x - r >= 0 and y - r >= 0:
                centers.append([x, y])
                count += 1
            col += 1
        row += 1
    while len(centers) < n:
        centers.append([0.5, 0.5])
    return np.array(centers[:n])


def grid_init(n, rows, cols):
    centers = []
    for i in range(n):
        r = i // cols
        c = i % cols
        x = (c + 0.5) / cols
        y = (r + 0.5) / rows
        centers.append([x, y])
    return np.array(centers[:n])


def build_constraints(n):
    cons = []
    for i in range(n):
        cons.append({"type": "ineq", "fun": constr_x_min, "args": (i, n)})
        cons.append({"type": "ineq", "fun": constr_x_max, "args": (i, n)})
        cons.append({"type": "ineq", "fun": constr_y_min, "args": (i, n)})
        cons.append({"type": "ineq", "fun": constr_y_max, "args": (i, n)})
        cons.append({"type": "ineq", "fun": constr_r_pos, "args": (i, n)})
    for i in range(n):
        for j in range(i + 1, n):
            cons.append({"type": "ineq", "fun": constr_overlap, "args": (i, j, n)})
    return cons


def build_bounds(n):
    bounds = [(0.001, 0.999) for _ in range(2 * n)]
    bounds += [(0.001, 0.5) for _ in range(n)]
    return bounds


def iterative_relaxation(centers, radii, n, iterations=1000):
    centers = centers.copy()
    radii = radii.copy()
    
    for _ in range(iterations):
        for i in range(n):
            max_r = min(
                centers[i, 0], 1 - centers[i, 0],
                centers[i, 1], 1 - centers[i, 1]
            )
            for j in range(n):
                if i != j:
                    dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                    max_r = min(max_r, dist - radii[j])
            radii[i] = max(0.0001, max_r)
        
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[j, 0] - centers[i, 0]
                dy = centers[j, 1] - centers[i, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                required = radii[i] + radii[j]
                if dist < required and dist > 1e-12:
                    push = (required - dist) * 0.3
                    nx = dx / dist
                    ny = dy / dist
                    centers[i, 0] -= push * nx
                    centers[i, 1] -= push * ny
                    centers[j, 0] += push * nx
                    centers[j, 1] += push * ny
        
        for i in range(n):
            centers[i, 0] = np.clip(centers[i, 0], radii[i], 1 - radii[i])
            centers[i, 1] = np.clip(centers[i, 1], radii[i], 1 - radii[i])
    
    return centers, radii


def run_packing():
    n = N_CIRCLES
    best_centers = None
    best_radii = None
    best_sum = -1.0
    cons = build_constraints(n)
    bounds = build_bounds(n)

    init_configs = []
    
    for r_init in [0.07, 0.075, 0.08, 0.085, 0.09, 0.095, 0.10]:
        c = hexagonal_init(n, r_init)
        for i in range(min(n, len(c))):
            c[i, 0] = np.clip(c[i, 0], 0.01, 0.99)
            c[i, 1] = np.clip(c[i, 1], 0.01, 0.99)
        radii = np.full(n, r_init * 0.8)
        params = np.concatenate([c.ravel(), radii])
        init_configs.append(params)
    
    for r_init in [0.08, 0.09, 0.10]:
        np.random.seed(42)
        c = hexagonal_init(n, r_init)
        c += np.random.randn(n, 2) * 0.015
        c[:, 0] = np.clip(c[:, 0], 0.02, 0.98)
        c[:, 1] = np.clip(c[:, 1], 0.02, 0.98)
        radii = np.full(n, r_init * 0.75)
        params = np.concatenate([c.ravel(), radii])
        init_configs.append(params)

    for rows, cols in [(5, 6), (6, 5)]:
        c = grid_init(n, rows, cols)
        c[:, 0] = np.clip(c[:, 0], 0.02, 0.98)
        c[:, 1] = np.clip(c[:, 1], 0.02, 0.98)
        radii = np.full(n, 0.07)
        params = np.concatenate([c.ravel(), radii])
        init_configs.append(params)

    for idx, params0 in enumerate(init_configs):
        try:
            result = minimize(
                obj_func,
                params0.copy(),
                method="SLSQP",
                bounds=bounds,
                constraints=cons,
                options={"maxiter": 5000, "ftol": 1e-13}
            )
            current_sum = -result.fun
            if current_sum > best_sum:
                best_sum = current_sum
                best_centers = result.x[: 2 * n].reshape(n, 2)
                best_radii = result.x[2 * n:].copy()
        except Exception:
            continue

    if best_centers is None:
        r = 0.08
        best_centers = hexagonal_init(n, r)
        best_radii = np.full(n, r)

    best_centers, best_radii = iterative_relaxation(best_centers, best_radii, n, iterations=2000)

    best_centers = np.round(best_centers, 12)
    best_radii = np.round(best_radii, 12)

    total = np.sum(best_radii)
    return best_centers, best_radii, float(total)
