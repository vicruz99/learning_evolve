import numpy as np
from scipy.optimize import minimize

def compute_penalty_and_grad(vars, n):
    """Compute smooth quadratic penalty and its gradient for circle packing constraints."""
    centers = vars[:2*n].reshape(n, 2)
    radii = vars[2*n:]
    
    penalty = 0.0
    grad = np.zeros_like(vars)
    
    # Inter-circle non-overlap constraints
    for i in range(n):
        for j in range(i+1, n):
            dx = centers[i,0] - centers[j,0]
            dy = centers[i,1] - centers[j,1]
            d_sq = dx*dx + dy*dy
            d = np.sqrt(d_sq)
            
            # Prevent division by zero; if exactly overlapping, assign arbitrary small separation direction
            if d < 1e-8:
                d = 1e-8
                dx = 0.0
                dy = 0.0
            
            v = radii[i] + radii[j] - d
            if v > 0:
                penalty += v**2
                factor = 2 * v
                inv_d = 1.0 / d
                
                # Gradients for radii
                grad[2*n + i] += factor
                grad[2*n + j] += factor
                
                # Gradients for centers (repulsion forces)
                grad[2*i] -= factor * dx * inv_d
                grad[2*i+1] -= factor * dy * inv_d
                grad[2*j] += factor * dx * inv_d
                grad[2*j+1] += factor * dy * inv_d
                
    # Boundary constraints
    for i in range(n):
        # x >= r
        v = radii[i] - centers[i,0]
        if v > 0:
            penalty += v**2
            factor = 2 * v
            grad[2*n + i] += factor
            grad[2*i] -= factor
            
        # x + r <= 1
        v = centers[i,0] + radii[i] - 1
        if v > 0:
            penalty += v**2
            factor = 2 * v
            grad[2*n + i] += factor
            grad[2*i] += factor
            
        # y >= r
        v = radii[i] - centers[i,1]
        if v > 0:
            penalty += v**2
            factor = 2 * v
            grad[2*n + i] += factor
            grad[2*i+1] -= factor
            
        # y + r <= 1
        v = centers[i,1] + radii[i] - 1
        if v > 0:
            penalty += v**2
            factor = 2 * v
            grad[2*n + i] += factor
            grad[2*i+1] += factor
            
    return penalty, grad

def solve_from_init(n, init_centers, init_radii, penalty_weight=10000.0):
    """Run L-BFGS-B optimization from a given initialization."""
    init_vars = np.concatenate([init_centers.ravel(), init_radii])
    
    def obj_wrapper(v):
        p, _ = compute_penalty_and_grad(v, n)
        return -np.sum(v[2*n:]) + penalty_weight * p
        
    def grad_wrapper(v):
        _, g = compute_penalty_and_grad(v, n)
        obj_grad = np.zeros_like(v)
        obj_grad[2*n:] = -1.0
        return obj_grad + penalty_weight * g

    bounds = [(0.0, 1.0)] * (2*n) + [(1e-6, 0.5)] * n
    
    res = minimize(obj_wrapper, init_vars, jac=grad_wrapper, bounds=bounds, 
                   method='L-BFGS-B', options={'maxiter': 1500, 'ftol': 1e-10, 'gtol': 1e-6})
    
    final_centers = res.x[:2*n].reshape(n, 2)
    final_radii = res.x[2*n:]
    return final_centers, final_radii
