import numpy as np
from scipy.optimize import minimize

def generate_initial_centers(n_circles=26):
    """
    Generates a hexagonal-like grid of centers for n_circles inside [0,1]x[0,1].
    """
    centers = []
    # Try to form a hexagonal packing layout
    # Row counts: 5, 6, 5, 6, 4 sums to 26.
    row_counts = [5, 6, 5, 6, 4]
    
    # Vertical spacing approximation for hexagonal packing
    # We need to fit 5 rows. Let's estimate height to scale later or just place them.
    # Approx row height ~ 0.2
    y_positions = np.linspace(0.15, 0.85, len(row_counts))
    
    for i, count in enumerate(row_counts):
        y = y_positions[i]
        # Horizontal spacing ~ 0.2. Offset every other row.
        # Total width 1.0.
        # If count=6, step is 1/7 approx 0.14.
        # If count=5, step is 1/6 approx 0.16.
        # Let's just distribute them evenly in [0,1] with padding
        
        # Center the row in x
        row_x = np.linspace(0.1, 0.9, count)
        
        # Offset every other row (index 1, 3)
        if i % 2 == 1:
            shift = (row_x[1] - row_x[0]) / 2
            row_x = row_x + shift
            # Clamp to [0,1] roughly, though we just want initial spread
            row_x = np.clip(row_x, 0.05, 0.95)
        
        for x in row_x:
            centers.append([x, y])
            
    return np.array(centers[:n_circles])

def objective(vars, n):
    """
    Objective function to minimize: -sum(radii)
    vars: [x1, y1, r1, ..., xn, yn, rn]
    """
    radii = vars[2::3]
    return -np.sum(radii)

def constraints_factory(centers, radii, n):
    """
    Generates constraint functions for non-overlap and boundaries.
    """
    cons = []
    
    # Boundary constraints
    for i in range(n):
        x = centers[i, 0]
        y = centers[i, 1]
        r = radii[i]
        
        # r <= x
        cons.append({'type': 'ineq', 'fun': lambda v, idx=i: v[2*idx + 2] - v[3*idx]}) # r - x <= 0 ? No, ineq means fun >= 0. 
        # Wait, scipy constraint: fun(x) >= 0 for 'ineq'.
        # We want x - r >= 0 => x >= r.
        # So fun = x - r.
        
        cons.append({'type': 'ineq', 'fun': lambda v, idx=i: v[3*idx] - v[3*idx + 2]}) # x - r >= 0
        cons.append({'type': 'ineq', 'fun': lambda v, idx=i: v[3*idx] - 1 + v[3*idx + 2]}) # 1 - (x + r) >= 0 => x + r <= 1
        cons.append({'type': 'ineq', 'fun': lambda v, idx=i: v[3*idx + 1] - v[3*idx + 2]}) # y - r >= 0
        cons.append({'type': 'ineq', 'fun': lambda v, idx=i: v[3*idx + 1] - 1 + v[3*idx + 2]}) # y + r <= 1

    # Non-overlap constraints: dist >= r_i + r_j
    for i in range(n):
        for j in range(i + 1, n):
            cons.append({'type': 'ineq', 'fun': lambda v, i=i, j=j: 
                         np.sqrt((v[3*i] - v[3*j])**2 + (v[3*i+1] - v[3*j+1])**2) - (v[3*i+2] + v[3*j+2])
                        })
    return cons

    num_trials = 15
