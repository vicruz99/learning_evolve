import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    Packs 26 circles in a unit square to maximize the sum of radii.
    """
    n_circles = 26

    centers = np.zeros((n_circles, 2))

    # Row 1: 5 circles (shifted)
    # Row 2: 5 circles
    # Row 3: 5 circles (shifted)

    # Layout: 5 rows. 
    # Rows 0, 2, 4 (even) have 5 circles.
    # Rows 1, 3 (odd) have 5 circles.
    # Total 25. We need 26.
    # Let's put 6 circles in one row? 
    # If we shift rows, we can fit 6 in some rows if radius is small enough initially.
    # Let's just generate a grid and pick 26 points.
    
    # Generate a hexagonal grid of points
    points = []
    # Spacing parameters for initial grid (loose)
    h_spacing = 0.25
    v_spacing = h_spacing * np.sqrt(3) / 2
    
    row_idx = 0
    y = 0.1 # Start near bottom
    
    while len(points) < n_circles:
        # Determine number of circles in this row
        # Shift for odd rows
        x_start = 0.1 + (row_idx % 2) * (h_spacing / 2)
        
        # How many fit?
        x = x_start
        while x <= 0.9 and len(points) < n_circles:
            points.append([x, y])
            x += h_spacing
        
        y += v_spacing
        row_idx += 1
        
    # If we generated more than needed, truncate (should not happen with loop condition)
    # If less (unlikely with these params), pad with random
    while len(points) < n_circles:
        points.append([np.random.rand(), np.random.rand()])
        
    centers = np.array(points[:n_circles])
    
    # 2. Define constraints and objective for optimization
    
    # Variables vector: [x1, y1, r1, x2, y2, r2, ...]
    # Total variables: 3 * n_circles
    
    def objective(vars):
        # Minimize negative sum of radii
        radii = vars[2::3]
        return -np.sum(radii)

    # Helper to get indices
    def get_center(i):
        return (vars[i*3], vars[i*3+1])
    def get_radius(i):
        return vars[i*3+2]

    # We will use SLSQP which takes constraints as dicts.
    # However, defining 429 constraints explicitly in a list might be heavy.
    # We can use a penalty approach or define constraint function that returns array.
    # SLSQP supports inequality constraints g(x) >= 0.
    
    # Constraints:
    # 1. Boundary: x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
    # 2. Overlap: dist(i,j) - r_i - r_j >= 0
    
    # To make it efficient, we can define a single constraint function that returns an array of violations.
    # But scipy expects constraints to be defined with 'fun' returning value >= 0.
    # We can return an array for 'ineq' constraints.
    
    def constraints(vars):
        vals = []
        
        # Boundary constraints
        for i in range(n_circles):
            x, y, r = vars[i*3], vars[i*3+1], vars[i*3+2]
            # x - r >= 0
            vals.append(x - r)
            # 1 - x - r >= 0 => x + r <= 1
            vals.append(1 - x - r)
            # y - r >= 0
            vals.append(y - r)
            # 1 - y - r >= 0
            vals.append(1 - y - r)
            
        # Overlap constraints
        # To reduce computational cost, we might skip some checks if far apart,
        # but for 26 circles, 325 checks is manageable.
        for i in range(n_circles):
            xi, yi, ri = vars[i*3], vars[i*3+1], vars[i*3+2]
            for j in range(i + 1, n_circles):
                xj, yj, rj = vars[j*3], vars[j*3+1], vars[j*3+2]
                dist = np.sqrt((xi - xj)**2 + (yi - yj)**2)
                vals.append(dist - ri - rj)
                
        return np.array(vals)

    # Convert initial state to vector
    x0 = np.zeros(3 * n_circles)
    for i in range(n_circles):
        x0[i*3] = centers[i, 0]
        x0[i*3+1] = centers[i, 1]

    bounds = []

        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 0.5)) # r
        
    # Define constraint dictionary
    cons = ({'type': 'ineq', 'fun': constraints})
    
    # Run optimization
    # Using SLSQP. Might need multiple restarts or good initialization.
    # The initial hexagonal grid should be good.
    
    try:
        res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, 
                       options={'maxiter': 1000, 'ftol': 1e-9, 'disp': False})
        if res.success or res.fun < -2.0: # If we get a decent sum
            final_vars = res.x
        else:
            # Fallback to initial if optimization fails (unlikely with SLSQP on this scale)
            final_vars = x0
    except Exception:
        final_vars = x0
        
    # Extract results
    final_centers = np.zeros((n_circles, 2))
    final_radii = np.zeros(n_circles)
    
    for i in range(n_circles):
        final_centers[i, 0] = final_vars[i*3]
        final_centers[i, 1] = final_vars[i*3+1]
        final_radii[i] = final_vars[i*3+2]
        
    sum_radii = np.sum(final_radii)
    
    return final_centers, final_radii, sum_radii
