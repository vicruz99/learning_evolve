import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Pack 26 circles in a unit square to maximize sum of radii.
    """
    N = 26
    best_sum_r = 0.0
    best_centers = None
    best_radii = None
    
    # Number of random restarts to find a good global optimum
    num_restarts = 10
    
    for restart in range(num_restarts):
        # --- Initialization ---
        # Create a grid-like initialization to spread circles out
        # We aim for roughly sqrt(26) ~ 5.1 circles per side
        cols = 5
        rows = int(np.ceil(N / cols))
        
        cx = np.zeros(N)
        cy = np.zeros(N)
        
        for i in range(N):
            r_idx = i // cols
            c_idx = i % cols
            
            # Place centers in a grid within a slightly smaller box to allow for radii
            # Box [0.1, 0.9] x [0.1, 0.9]
            cx[i] = 0.1 + (c_idx + 0.5) * 0.8 / cols
            cy[i] = 0.1 + (r_idx + 0.5) * 0.8 / rows
            
        # Add random perturbation to break symmetry
        np.random.seed(42 + restart) # Fixed seed for reproducibility within restarts
        cx += np.random.uniform(-0.02, 0.02, N)
        cy += np.random.uniform(-0.02, 0.02, N)
        
        # Clip to valid range [0.1, 0.9]
        cx = np.clip(cx, 0.1, 0.9)
        cy = np.clip(cy, 0.1, 0.9)
        
        # Initial radii: small enough to be feasible, but large enough to give optimizer headroom
        # 0.05 is safe since min distance in grid is roughly 0.16
        r0 = np.full(N, 0.04)
        
        # Flatten variables: [x1, y1, r1, x2, y2, r2, ...]
        x0 = np.zeros(3 * N)
        x0[0::3] = cx
        x0[1::3] = cy
        x0[2::3] = r0
        
        # --- Bounds ---
        # x, y in [0, 1], r >= 0. 
        # Practically r <= 0.5, but let's set loose upper bound or None. 
        # SLSQP handles bounds well.
        bounds = []
        for _ in range(N):
            bounds.append((0.0, 1.0)) # x
            bounds.append((0.0, 1.0)) # y
            bounds.append((0.0, 0.5)) # r (max possible radius in unit square is 0.5)
            
        # --- Objective Function ---
        def objective(vars_vec):
            r = vars_vec[2::3]
            return -np.sum(r) # Minimize negative sum
        
        # --- Constraints ---
        # Inequality constraints g(vars) >= 0
        # 1. Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
        # 2. Overlap constraints: dist^2 >= (r_i + r_j)^2
        
        def constraint_func(vars_vec):
            x = vars_vec[0::3]
            y = vars_vec[1::3]
            r = vars_vec[2::3]
            
            cons_list = []
            
            # Boundary constraints
            # x - r >= 0
            cons_list.extend(x - r)
            # 1 - x - r >= 0
            cons_list.extend(1.0 - x - r)
            # y - r >= 0
            cons_list.extend(y - r)
            # 1 - y - r >= 0
            cons_list.extend(1.0 - y - r)
            
            # Overlap constraints (squared distance)
            # (xi - xj)^2 + (yi - yj)^2 - (ri + rj)^2 >= 0
            # To be efficient, we can vectorize this loop slightly or just loop
            # N=26 is small, loop is fine.
            
            for i in range(N):
                for j in range(i + 1, N):
                    dist_sq = (x[i] - x[j])**2 + (y[i] - y[j])**2
                    sum_r = r[i] + r[j]
                    cons_val = dist_sq - sum_r**2
                    cons_list.append(cons_val)
                    
            return np.array(cons_list)

        # Define constraint dictionary for SLSQP
        # We can pass a single function that returns an array
        constraints = ({'type': 'ineq', 'fun': constraint_func})
        
        # --- Optimization ---
        try:
            res = minimize(
                objective,
                x0,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 1000, 'ftol': 1e-9, 'disp': False}
            )

                # Check if the solution is actually valid (tolerances)

                    current_radii[i] = res.x[3*i+2]
