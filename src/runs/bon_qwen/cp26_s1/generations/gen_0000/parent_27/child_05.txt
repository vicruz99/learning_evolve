import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # 1. Generate Initial Configuration (Hexagonal Grid)
    # We create a hexagonal packing pattern to ensure a dense, valid start.
    # We will generate enough points and select the first 26.
    
    # Hexagonal grid parameters
    # Row 0: 5 circles, Row 1: 4 circles, etc. to fit roughly in square
    # We will generate a set of points and use them.
    
    points = []
    r_guess = 0.1 # Initial guess for radius

    # Spacing in x: 2*r, in y: sqrt(3)*r
    # Shift odd rows by r in x

    # Let's use a simple grid generation that covers the square
    # 6 rows, approx 4-5 circles per row
    
    rows_count = 7
    cols_count = 5
    
    for row in range(rows_count):
        for col in range(cols_count):
            # Hexagonal offset
            x = (col + 0.5) * 0.2 + (row % 2) * 0.1
            y = row * 0.1732 + 0.1 # 0.1732 approx sqrt(3)/10
            
            # Check bounds roughly to keep inside
            if 0.1 <= x <= 0.9 and 0.1 <= y <= 0.9:
                points.append([x, y])

    # We need exactly 26.
    if len(points) >= 26:

    else:

        np.random.seed(42)
        init_centers = np.random.rand(26, 2) * 0.8 + 0.1

    # Initialize radii to a small value to ensure feasibility
    init_radii = np.ones(n) * 0.05

    # To strictly satisfy boundary constraints x in [r, 1-r], we parameterize:
    # x = r + (1 - 2r) * u
    # y = r + (1 - 2r) * v
    # where u, v in [0, 1].
    # This ensures the circle is always inside the unit square.
    
    # Calculate initial u, v
    # u = (x - r) / (1 - 2r)
    # Need to handle r=0.5 edge case, but r starts small.
    denom = 1 - 2 * init_radii
    # Avoid division by zero
    denom = np.clip(denom, 1e-9, 1.0)
    
    init_u = (init_centers[:, 0] - init_radii) / denom
    init_v = (init_centers[:, 1] - init_radii) / denom
    
    # Clip u, v to [0, 1]
    init_u = np.clip(init_u, 0, 1)
    init_v = np.clip(init_v, 0, 1)

    x0 = np.concatenate([init_radii, init_u, init_v])

    def objective(vars_vec):
        # Maximize sum of radii -> Minimize -sum(radii)
        return -np.sum(vars_vec[:n])

    def constraint_func(vars_vec):
        rs = vars_vec[:n]
        us = vars_vec[n:2*n]
        vs = vars_vec[2*n:3*n]
        
        # Reconstruct centers
        # x = r + (1 - 2r) * u
        # y = r + (1 - 2r) * v
        # To avoid numerical issues if r is close to 0.5, we can clamp or just compute.
        # Since r <= 0.5, 1-2r >= 0.
        
        factor = 1.0 - 2.0 * rs
        xs = rs + factor * us
        ys = rs + factor * vs
        
        # Vectorized distance calculation
        # dx[i, j] = xs[i] - xs[j]
        dx = xs[:, None] - xs[None, :]
        dy = ys[:, None] - ys[None, :]
        dist_sq = dx**2 + dy**2
        
        # Required distance squared (r_i + r_j)^2
        dr = rs[:, None] + rs[None, :]

        diff = dist_sq - min_dist_sq
        
        # We only need constraints for i < j (upper triangle)
        # Use tril/triu mask
        mask = np.triu(np.ones((n, n), dtype=bool), k=1)
        return diff[mask]

        temp_v = (temp_centers[:, 1] - temp_radii) / denom
