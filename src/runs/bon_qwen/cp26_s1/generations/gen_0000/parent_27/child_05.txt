import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def boundary_constraints(v):
    """Returns array of boundary constraint values: must be >= 0"""
    cons = np.zeros(4 * N_CIRCLES)
    for i in range(N_CIRCLES):
        idx = 3 * i
        cons[4 * i] = v[idx] - v[idx + 2]           
        cons[4 * i + 1] = 1.0 - v[idx] - v[idx + 2] 
        cons[4 * i + 2] = v[idx + 1] - v[idx + 2]   
        cons[4 * i + 3] = 1.0 - v[idx + 1] - v[idx + 2] 
    return cons

def overlap_constraints(v):
    """Returns array of non-overlap constraint values: must be >= 0"""
    n_constr = N_CIRCLES * (N_CIRCLES - 1) // 2
    cons = np.zeros(n_constr)
    k = 0
    for i in range(N_CIRCLES):
        for j in range(i + 1, N_CIRCLES):
            idx_i = 3 * i
            idx_j = 3 * j
            dx = v[idx_i] - v[idx_j]
            dy = v[idx_i + 1] - v[idx_j + 1]
            dr = v[idx_i + 2] + v[idx_j + 2]
            cons[k] = dx*dx + dy*dy - dr*dr
            k += 1
    return cons

def objective(v):
    """Negative sum of radii (to be minimized)"""
    return -np.sum(v[2::3])

def run_packing():
    np.random.seed(42)
    n = N_CIRCLES
    
    bounds = [(0.0, 1.0), (0.0, 1.0), (1e-4, 0.5)] * n
    constraints = [
        {'type': 'ineq', 'fun': boundary_constraints},
        {'type': 'ineq', 'fun': overlap_constraints}
    ]
    
    best_v = None
    best_obj = -np.inf
    
    # Multiple restarts to escape local minima
    for restart in range(8):
        centers_init = []
        rows = [5, 5, 6, 5, 5]
        dy = 0.18 + 0.01 * np.random.rand()
        dx = 0.19 + 0.01 * np.random.rand()
        y_base = 0.5 + 0.02 * np.random.randn()
        x_base = 0.5 + 0.02 * np.random.randn()
        
        for i, count in enumerate(rows):
            y = y_base + (i - 2) * dy * np.sqrt(3)/2
            x_start = x_base - (count - 1) * dx / 2
            for j in range(count):
                cx = x_start + j * dx + 0.005 * np.random.randn()
                cy = y + 0.005 * np.random.randn()
                centers_init.append([np.clip(cx, 0.1, 0.9), np.clip(cy, 0.1, 0.9)])
                
        centers_init = np.array(centers_init)
        r_init = 0.065 + 0.005 * np.random.rand()
        
        x0 = np.zeros(3 * n)
        for i in range(n):
            x0[3*i] = centers_init[i,0]
            x0[3*i+1] = centers_init[i,1]
            x0[3*i+2] = r_init
            
        res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints, 
                       options={'maxiter': 3000, 'ftol': 1e-12, 'disp': False})
        
        curr_obj = -res.fun
        if curr_obj > best_obj:
            best_obj = curr_obj
            best_v = res.x
            
    v_opt = best_v if best_v is not None else x0
    centers_opt = np.column_stack((v_opt[0::3], v_opt[1::3]))
    radii_opt = v_opt[2::3]
    
    # Ensure strict boundary compliance
    for i in range(n):
        for k in range(2):
            if centers_opt[i,k] < radii_opt[i]:
                centers_opt[i,k] = radii_opt[i]
            elif centers_opt[i,k] > 1.0 - radii_opt[i]:
                centers_opt[i,k] = 1.0 - radii_opt[i]
                
    return centers_opt, radii_opt, np.sum(radii_opt)
