import numpy as np

def run_packing():
    n = 26
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Run multiple times with different seeds to explore configuration space
    for seed in range(40):
        rng = np.random.default_rng(seed)
        
        # Initialize centers randomly, kept away from boundaries initially
        centers = rng.uniform(0.1, 0.9, size=(n, 2))
        radii = np.ones(n) * 0.015
        
        perturbation_base = 0.008
        steps = 50000
        
        for step in range(steps):
            min_clearance = 1.0
            
            # Compute pairwise clearances
            for i in range(n):
                for j in range(i + 1, n):
                    dx = centers[i, 0] - centers[j, 0]
                    dy = centers[i, 1] - centers[j, 1]
                    dist = np.hypot(dx, dy)
                    c = dist - radii[i] - radii[j]
                    if c < min_clearance:
                        min_clearance = c
                            
            # Compute boundary clearances
            for i in range(n):
                c_left = centers[i, 0] - radii[i]
                c_right = 1.0 - centers[i, 0] - radii[i]
                c_down = centers[i, 1] - radii[i]
                c_up = 1.0 - centers[i, 1] - radii[i]
                c_bound = min(c_left, c_right, c_down, c_up)
                if c_bound < min_clearance:
                    min_clearance = c_bound
            
            # Growth step
            if min_clearance > 1e-9:
                # Increase radius conservatively. 
                # Pairwise clearance shrinks by 2*delta, boundary by delta.
                # Taking min_clearance/2 satisfies both safely.
                delta = min_clearance * 0.5
                radii += delta
            else:
                # Jammed: perturb to escape local optimum
                # Cooling schedule for perturbations
                cooling = 1.0 / (1.0 + step / 3000)
                noise = rng.normal(0, perturbation_base * cooling, size=centers.shape)
                centers += noise
                # Strictly enforce boundaries after perturbation
                np.clip(centers, radii[:, None], 1.0 - radii[:, None], out=centers)
                
            cur_sum = np.sum(radii)
            if cur_sum > best_sum:
                best_sum = cur_sum
                best_centers = centers.copy()
                best_radii = radii.copy()
                
    # Final safety clamp to guarantee validity within tolerance
    best_centers = np.clip(best_centers, best_radii[:, None], 1.0 - best_radii[:, None])
    
    return best_centers, best_radii, best_sum
