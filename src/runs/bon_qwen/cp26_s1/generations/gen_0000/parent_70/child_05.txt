import numpy as np
from scipy.optimize import minimize

def compute_objective(x, n_circles, lam):
    """
    Computes the objective function: -sum(radii) + penalty * violations.
    x: flattened array of [x1, y1, r1, x2, y2, r2, ...]
    """
    obj = 0.0
    penalty = 0.0
    
    for i in range(n_circles):
        idx = 3 * i
        rx = x[idx + 2]
        obj -= rx  # Maximizing sum(r) <=> Minimizing -sum(r)
        
        xi, yi = x[idx], x[idx + 1]
        
        # Boundary constraint violations
        # x - r >= 0  =>  r - x <= 0
        # x + r <= 1  =>  x + r - 1 <= 0
        # y - r >= 0  =>  r - y <= 0
        # y + r <= 1  =>  y + r - 1 <= 0
        boundary_violations = [
            rx - xi,
            xi + rx - 1.0,
            rx - yi,
            yi + rx - 1.0
        ]
        for viol in boundary_violations:
            if viol > 0:
                penalty += 0.5 * viol**2
                
        # Overlap constraint violations
        for j in range(i + 1, n_circles):
            jdx = 3 * j
            rj = x[jdx + 2]
            dx = x[idx] - x[jdx]
            dy = x[idx + 1] - x[jdx + 1]
            # Add epsilon to prevent NaN gradients when circles coincide
            dist = np.sqrt(dx*dx + dy*dy + 1e-12)
            viol = rx + rj - dist
            if viol > 0:
                penalty += 0.5 * viol**2
                
    return obj + lam * penalty

def run_packing():
    n = 26
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Bounds: x,y in [0,1], r in [0, 0.5]
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
    
    n_restarts = 25
    
    for _ in range(n_restarts):
        # Random initial positions inside [0.15, 0.85] to keep them away from edges initially
        centers = np.random.uniform(0.15, 0.85, (n, 2))
        # Start with small radii to ensure initial feasibility
        radii = np.full(n, 0.005)
        x0 = np.concatenate([centers.ravel(), radii])
        
        # Sequential penalty method
        lam = 10.0
        current_x = x0
        for _ in range(6):
            res = minimize(compute_objective, current_x, args=(n, lam), 
                           method='L-BFGS-B', bounds=bounds, 
                           options={'maxiter': 4000, 'ftol': 1e-13, 'gtol': 1e-12})
            current_x = res.x
            lam *= 5.0
            
        centers_curr = current_x[:2*n].reshape((n, 2))
        radii_curr = current_x[2*n:]
        
        # Ensure non-negative radii (L-BFGS-B should handle bounds, but safety check)
        radii_curr = np.maximum(radii_curr, 0.0)
        
        current_sum = np.sum(radii_curr)
        
        # Track best solution
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers_curr.copy()
            best_radii = radii_curr.copy()
            
    # Final cleanup: sort circles by radius (optional, but nice for output)
    idx_sort = np.argsort(best_radii)[::-1]
    best_centers = best_centers[idx_sort]
    best_radii = best_radii[idx_sort]
    
    return best_centers, best_radii, best_sum
