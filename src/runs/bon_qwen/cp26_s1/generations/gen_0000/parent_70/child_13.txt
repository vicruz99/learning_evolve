import numpy as np
from scipy.optimize import linprog, minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes the placement of 26 circles in a unit square to maximize the sum of radii.
    """
    n = 26

    def evaluate_centers(centers):
        """Solves the LP to find max sum of radii for fixed centers."""
        # Bounds for radii: [0, 0.5]
        bounds = [(0, 0.5) for _ in range(n)]
        
        # Objective: maximize sum(r) -> minimize -sum(r)
        c = np.ones(n) * -1
        
        # Constraints A_ub * r <= b_ub
        # 1. Boundary constraints: r_i <= dist(center_i, boundary)
        A_ub = []
        b_ub = []
        
        for i in range(n):
            x, y = centers[i]
            row = np.zeros(n)
            row[i] = 1.0
            
            # r_i <= x
            A_ub.append(row)
            b_ub.append(x)
            # r_i <= 1-x
            A_ub.append(row)
            b_ub.append(1.0 - x)
            # r_i <= y
            A_ub.append(row)
            b_ub.append(y)
            # r_i <= 1-y
            A_ub.append(row)
            b_ub.append(1.0 - y)
            
        # 2. Pairwise constraints: r_i + r_j <= dist(i, j)
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                row = np.zeros(n)
                row[i] = 1.0
                row[j] = 1.0
                A_ub.append(row)
                b_ub.append(dist)
                
        A_ub = np.array(A_ub)
        b_ub = np.array(b_ub)
        
        # Solve LP
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        
        if res.success:
            return -res.fun, res.x
        else:
            return 0.0, np.zeros(n)

    def generate_hexagonal_packing():

        centers = []
        # Configuration: 5 rows with counts 5, 6, 5, 5, 5

        row_counts = [5, 6, 5, 5, 5]

            x_start = r_est

            # To simulate hex packing, we can shift rows.
            # But for robustness, let's just center them in [0,1]
            # If count=6, width is 5 gaps. Span = 5*2r. Center at 0.5.
            # Start = 0.5 - (count-1)*r
            # But we need to fit in [r, 1-r].
            
            # Let's distribute x uniformly in [0.05, 0.95] for a good spread
            x_coords = np.linspace(0.05, 0.95, count)
            
            for x in x_coords:
                centers.append([x, y])
                
        return np.array(centers)

    best_sum = 0.0
    best_centers = None
    best_radii = None

    # Phase 1: Optimize from a Hexagonal start
    initial_centers = generate_hexagonal_packing()
    
    # Flatten centers for optimizer
    x0 = initial_centers.flatten()
    
    # Define objective for minimize (negate sum)
    def objective(x_flat):
        c_mat = x_flat.reshape((n, 2))
        s, _ = evaluate_centers(c_mat)
        return -s # Minimize negative sum

    # Run Nelder-Mead optimization
    # Maxiter can be high, but we want speed. 
    res = minimize(objective, x0, method='Nelder-Mead', options={'maxiter': 5000, 'xatol': 1e-6, 'fatol': 1e-6})
    
    curr_sum = -res.fun
    curr_centers = res.x.reshape((n, 2))
    curr_sum_val, curr_radii = evaluate_centers(curr_centers)
    
    if curr_sum_val > best_sum:
        best_sum = curr_sum_val
        best_centers = curr_centers.copy()
        best_radii = curr_radii.copy()

    # Phase 2: Randomized restarts from the best found so far to escape local optima
    # We perturb the best centers slightly and re-optimize
    rng = np.random.default_rng(42)
    for _ in range(5):

        perturbed_centers = best_centers + perturbation

        perturbed_centers = np.clip(perturbed_centers, 0.0, 1.0)

        res_pert = minimize(objective, x0_pert, method='Nelder-Mead', options={'maxiter': 2000})

        p_centers = res_pert.x.reshape((n, 2))
