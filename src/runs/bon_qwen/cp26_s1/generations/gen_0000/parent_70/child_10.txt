import numpy as np
from scipy.optimize import minimize

N = 26

def obj_equal(vars):
    return -vars[2*N]

def cons_equal(vars):
    centers = vars[:2*N].reshape(N, 2)
    r = vars[2*N]
    con = []
    # Boundary constraints: x >= r, x <= 1-r, y >= r, y <= 1-r
    con.extend(centers[:, 0] - r)
    con.extend(1 - centers[:, 0] - r)
    con.extend(centers[:, 1] - r)
    con.extend(1 - centers[:, 1] - r)
    # Overlap constraints: dist^2 >= (2r)^2
    for i in range(N):
        for j in range(i+1, N):
            dist_sq = np.sum((centers[i] - centers[j])**2)
            con.append(dist_sq - (2*r)**2)
    return np.array(con)

def obj_free(vars):
    return -np.sum(vars[2*N:])

def cons_free(vars):
    centers = vars[:2*N].reshape(N, 2)
    radii = vars[2*N:]
    con = []
    # Boundary constraints
    con.extend(centers[:, 0] - radii)
    con.extend(1 - centers[:, 0] - radii)
    con.extend(centers[:, 1] - radii)
    con.extend(1 - centers[:, 1] - radii)
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    for i in range(N):
        for j in range(i+1, N):
            dist_sq = np.sum((centers[i] - centers[j])**2)
            con.append(dist_sq - (radii[i] + radii[j])**2)
    return np.array(con)

def run_packing():
    best_val = -np.inf
    best_centers = None
    best_radii = None
    
    seeds = [42, 123, 456, 789, 101, 202, 315, 420]
    
    for seed in seeds:
        np.random.seed(seed)
        # Initialize with small radii and well-separated centers to ensure feasibility
        init_centers = np.random.rand(N, 2) * 0.7 + 0.15
        r_init = 0.04
        x0 = np.hstack([init_centers.flatten(), r_init])
        
        bounds_eq = [(0, 1)]*(2*N) + [(0, 0.5)]
        cons_eq = {'type': 'ineq', 'fun': cons_equal}
        
        res_eq = minimize(obj_equal, x0, method='SLSQP', bounds=bounds_eq, constraints=cons_eq, 
                          options={'maxiter': 1000, 'ftol': 1e-12, 'iprint': -1})
                          
        r_opt = res_eq.x[-1]
        c_opt = res_eq.x[:2*N].reshape(N, 2)
        
        # Phase 2: Optimize with free radii
        x0_free = np.hstack([c_opt.flatten(), np.full(N, r_opt)])
        bounds_free = [(0, 1)]*(2*N) + [(0, 0.5)]*N
        cons_fr = {'type': 'ineq', 'fun': cons_free}
        
        res_free = minimize(obj_free, x0_free, method='SLSQP', bounds=bounds_free, constraints=cons_fr,
                            options={'maxiter': 1000, 'ftol': 1e-12, 'iprint': -1})
                            
        val = np.sum(res_free.x[2*N:])
        if val > best_val:
            best_val = val
            best_centers = res_free.x[:2*N].reshape(N, 2)
            best_radii = res_free.x[2*N:]
            
    return best_centers, best_radii, best_val
