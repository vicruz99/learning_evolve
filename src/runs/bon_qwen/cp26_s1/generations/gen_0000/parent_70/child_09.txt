import numpy as np
from scipy.optimize import minimize
import random

# Global constant for number of circles
N_CIRCLES = 26

def run_packing():
    """
    Solves the circle packing problem for 26 circles in a unit square.
    Returns centers, radii, and sum of radii.
    """
    best_sum_radii = -1.0
    best_centers = None
    best_radii = None
    
    # Define bounds for variables: x, y in [0, 1], r in [0, 0.5]
    # Variable order: x0, y0, r0, x1, y1, r1, ...
    bounds = []
    for _ in range(N_CIRCLES):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 0.5)) # r (radius cannot exceed 0.5 in unit square)

    # Objective function: minimize negative sum of radii
    def objective(params):
        # params is flattened array [x0, y0, r0, x1, y1, r1, ...]
        r = params[2::3]
        return -np.sum(r)

    # Constraints list for SLSQP
    # type 'ineq' means fun(x) >= 0
    
    # 1. Boundary Constraints:
    # x_i >= r_i  =>  x_i - r_i >= 0
    # x_i <= 1-r_i => 1 - x_i - r_i >= 0
    # y_i >= r_i  =>  y_i - r_i >= 0
    # y_i <= 1-r_i => 1 - y_i - r_i >= 0
    
    # 2. Non-overlap Constraints:
    # dist(i, j) >= r_i + r_j  =>  dist^2 - (r_i + r_j)^2 >= 0
    
    # We construct these constraints dynamically inside the loop or pre-define them.
    # Since N is small (26), we can define them as lambdas or closures, 
    # but the prompt asks for top-level helper functions and no closures from nesting if possible,
    # though closures for constraints are standard in scipy. 
    # To be safe and clean, we will pass a single constraint function that checks all, 
    # or list them. Listing them is safer for SLSQP.
    
    # Let's create a list of constraint dictionaries.
    constraints = []

    # Boundary constraints
    for i in range(N_CIRCLES):
        idx_x = 3 * i
        idx_y = 3 * i + 1
        idx_r = 3 * i + 2
        
        # x >= r
        constraints.append({'type': 'ineq', 'fun': lambda p, i=i: p[idx_x] - p[idx_r]})
        # 1-x >= r
        constraints.append({'type': 'ineq', 'fun': lambda p, i=i: 1.0 - p[idx_x] - p[idx_r]})
        # y >= r
        constraints.append({'type': 'ineq', 'fun': lambda p, i=i: p[idx_y] - p[idx_r]})
        # 1-y >= r
        constraints.append({'type': 'ineq', 'fun': lambda p, i=i: 1.0 - p[idx_y] - p[idx_r]})

    # Overlap constraints
    # We only need to check i < j
    for i in range(N_CIRCLES):
        for j in range(i + 1, N_CIRCLES):
            idx_xi = 3 * i
            idx_yi = 3 * i + 1
            idx_ri = 3 * i + 2
            
            idx_xj = 3 * j
            idx_yj = 3 * j + 1
            idx_rj = 3 * j + 2
            
            def overlap_constraint(p, i=i, j=j):
                dx = p[idx_xi] - p[idx_xj]
                dy = p[idx_yi] - p[idx_yj]
                r_sum = p[idx_ri] + p[idx_rj]
                return (dx**2 + dy**2) - (r_sum**2)
            
            constraints.append({'type': 'ineq', 'fun': overlap_constraint})

    # Helper to generate initial parameters
    def generate_initial_params(strategy='hex'):
        centers = np.zeros((N_CIRCLES, 2))
        
        if strategy == 'hex':
            # Hexagonal grid packing initialization
            # We want to fit circles roughly. Target radius ~0.1.
            # Spacing ~0.2.
            pts = []
            y = 0.1
            row = 0
            # Try to fill the square
            while len(pts) < N_CIRCLES:
                # Alternate x offset for hexagonal packing
                x_offset = 0.0 if row % 2 == 0 else 0.1
                x = 0.1 + x_offset
                while x < 0.95 and len(pts) < N_CIRCLES:
                    pts.append([x, y])
                    x += 0.2
                y += 0.17 # approx 0.1 * sqrt(3)
                row += 1
            
            centers[:len(pts)] = pts[:N_CIRCLES]
            # Fill remaining if any (though loop should cover)
            for k in range(len(pts), N_CIRCLES):
                centers[k] = [0.5, 0.5]
                
        else:
            # Random initialization
            # Spread points to avoid immediate huge overlaps
            centers = np.random.rand(N_CIRCLES, 2)
            # Scale to keep some margin
            centers = 0.1 + 0.8 * centers
            
        radii = np.full(N_CIRCLES, 0.05) # Start with small radius
        
        # Flatten to params array
        params = np.empty(N_CIRCLES * 3)
        params[0::3] = centers[:, 0]
        params[1::3] = centers[:, 1]
        params[2::3] = radii
        return params

    # Run optimization multiple times
    num_restarts = 30
    
    for k in range(num_restarts):
        # Decide strategy
        strategy = 'hex' if k < 20 else 'random'
        x0 = generate_initial_params(strategy)
        
        # Add a tiny bit of noise to hex grid to break symmetry
        if strategy == 'hex':
            noise = np.random.normal(0, 0.01, size=x0.shape)
            # Ensure bounds aren't violated by noise too much (clamping later in bounds)
            # Just add to centers, keep radii fixed
            x0[0::3] += noise[0::3]
            x0[1::3] += noise[1::3]
            # Clamp radii to positive
            x0[2::3] = np.maximum(x0[2::3], 0.01)

        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints,
                           options={'maxiter': 1000, 'ftol': 1e-9})
            
            if res.success or res.fun < -best_sum_radii + 1e-5: # Check if found better sum
                # Extract results
                centers_opt = res.x[0::3].reshape((N_CIRCLES, 2))
                radii_opt = res.x[2::3]
                
                # Verify validity manually just in case (SLSQP might allow slight violation)
                # But we rely on the validator provided in the prompt for final check.
                # However, we should ensure radii are non-negative.
                radii_opt = np.maximum(radii_opt, 0.0)
                
                current_sum = np.sum(radii_opt)
                
                if current_sum > best_sum_radii:
                    best_sum_radii = current_sum
                    best_centers = centers_opt
                    best_radii = radii_opt
                    
        except Exception:
            continue

    # Final cleanup: ensure no negative radii or NaNs
    if best_radii is not None:
        best_radii = np.maximum(best_radii, 0.0)
        best_radii = np.nan_to_num(best_radii, nan=0.0)
        best_centers = np.nan_to_num(best_centers, nan=0.0)
        
        # Re-calculate sum
        final_sum = np.sum(best_radii)
    else:
        # Fallback if everything fails (should not happen)
        best_centers = np.zeros((N_CIRCLES, 2))
        best_radii = np.zeros(N_CIRCLES)
        final_sum = 0.0

    return best_centers, best_radii, final_sum
