import numpy as np
import scipy.optimize as opt

def get_objective(vars, n_circles):
    """
    Objective function: minimize negative sum of radii.
    vars layout: [x0, y0, r0, x1, y1, r1, ...]
    """
    radii = vars[2::3]
    return -np.sum(radii)

def get_constraints(n_circles):
    """
    Generates the list of constraint dictionaries for scipy.optimize.
    """
    constraints = []
    
    # Boundary constraints for each circle
    # r <= x <= 1-r  =>  x - r >= 0  AND  1 - x - r >= 0
    # r <= y <= 1-r  =>  y - r >= 0  AND  1 - y - r >= 0
    # Also r >= 0
    
    for i in range(n_circles):
        idx_x = i * 3
        idx_y = i * 3 + 1
        idx_r = i * 3 + 2
        
        # x - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: v[idx_x] - v[idx_r]
        })
        # 1 - x - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: 1.0 - v[idx_x] - v[idx_r]
        })
        # y - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: v[idx_y] - v[idx_r]
        })
        # 1 - y - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: 1.0 - v[idx_y] - v[idx_r]
        })
        # r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: v[idx_r]
        })

    # Pairwise non-overlap constraints
    # dist^2 - (r_i + r_j)^2 >= 0
    for i in range(n_circles):
        for j in range(i + 1, n_circles):
            idx_xi = i * 3
            idx_yi = i * 3 + 1
            idx_ri = i * 3 + 2
            idx_xj = j * 3
            idx_yj = j * 3 + 1
            idx_rj = j * 3 + 2
            
            def non_overlap(v, i=i, j=j):
                xi, yi, ri = v[idx_xi], v[idx_yi], v[idx_ri]
                xj, yj, rj = v[idx_xj], v[idx_yj], v[idx_rj]
                dist_sq = (xi - xj)**2 + (yi - yj)**2
                r_sum_sq = (ri + rj)**2
                return dist_sq - r_sum_sq

            constraints.append({
                'type': 'ineq',
                'fun': non_overlap
            })
            
    return constraints

def generate_hex_packing(n, radius):
    """
    Generates centers for a hexagonal packing of n circles with given radius.
    Returns a numpy array of shape (n, 3) with [x, y, r].
    """
    centers = []
    
    # Hexagonal lattice parameters
    # Row height: sqrt(3)/2 * diameter = sqrt(3) * radius
    row_height = np.sqrt(3) * radius
    # Horizontal spacing: diameter = 2 * radius
    col_spacing = 2 * radius
    # Shift for odd rows: radius
    shift = radius
    
    y = radius
    row_idx = 0
    while len(centers) < n:
        # Determine starting x for this row
        x_start = radius if row_idx % 2 == 0 else 2 * radius
        
        x = x_start
        while True:
            # Check if circle fits in width
            if x + radius > 1.0 + 1e-9:
                break
            
            # Check if circle fits in height
            if y + radius > 1.0 + 1e-9:
                break
            
            centers.append([x, y, radius])
            
            if len(centers) == n:
                break
            
            x += col_spacing
        
        if len(centers) == n:
            break
            
        y += row_height
        row_idx += 1
        
        # Safety break to avoid infinite loop if radius is too large
        if y > 1.0 + 2 * radius:
            break
            
    return np.array(centers)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    # We use SLSQP. Bounds can be used to help, but constraints handle the logic.

    return centers, radii, float(sum_radii)
