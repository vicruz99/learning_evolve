- $r \le x_i, y_i \le 1-r$ for all $i$
- $r > 0$

import numpy as np
from scipy.optimize import minimize

def objective_func(x, n):
    """Objective: maximize radius r (minimize -r)"""
    return -x[-1]

def constraint_dist(x, n, idx_i, idx_j):
    """Non-overlap constraints: dist(c_i, c_j) >= 2r"""
    c = x[:2*n].reshape(n, 2)
    r = x[-1]
    diff = c[idx_i] - c[idx_j]
    dist = np.sqrt(np.sum(diff**2, axis=1))
    return dist - 2*r

def constraint_bound(x, n):
    """Boundary constraints: r <= c_i <= 1-r"""
    c = x[:2*n].reshape(n, 2)
    r = x[-1]
    return np.concatenate([c[:, 0] - r, 1 - (c[:, 0] + r),
                           c[:, 1] - r, 1 - (c[:, 1] + r)])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    idx_i, idx_j = np.triu_indices(n, k=1)
    
    # Initial hexagonal configuration
    r_init = 0.085
    centers = []
    rows_counts = [6, 5, 6, 5, 4]
    row_idx = 0
    for count in rows_counts:
        for col in range(count):
            x = col * 2 * r_init + (row_idx % 2) * r_init + r_init
            y = row_idx * np.sqrt(3) * r_init + r_init
            centers.append([x, y])
        row_idx += 1
    centers = np.array(centers[:n])
    
    # Slight random perturbation to break symmetry
    centers += np.random.uniform(-0.002, 0.002, centers.shape)
    centers = np.clip(centers, r_init, 1 - r_init)
    
    x0 = np.concatenate([centers.flatten(), [r_init]])
    
    cons = ({'type': 'ineq', 'fun': constraint_dist, 'args': (n, idx_i, idx_j)},
            {'type': 'ineq', 'fun': constraint_bound, 'args': (n,)})
            
    bnds = [(0.0, 1.0)] * (2*n) + [(0.01, 0.5)]
    
    res = minimize(objective_func, x0, method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': 5000, 'ftol': 1e-12, 'disp': False})
                   
    if res.success:
        x_opt = res.x
    else:
        # Fallback to initial if optimization fails
        x_opt = x0
        
    centers_opt = x_opt[:2*n].reshape(n, 2)
    r_opt = x_opt[-1]
    radii_opt = np.full(n, r_opt)
    
    # Strictly enforce boundaries to handle numerical precision
    centers_opt = np.clip(centers_opt, r_opt + 1e-9, 1.0 - r_opt - 1e-9)
    
    return centers_opt, radii_opt, float(np.sum(radii_opt))
