import numpy as np
import scipy.optimize as opt

def generate_hex_init(n_circles, rng):
    """Generates an initial hexagonal packing configuration."""
    centers = []
    # Try to fit circles with a base spacing of 0.2 (r=0.1)
    r_start = 0.09
    spacing = 2 * r_start
    row_height = spacing * np.sqrt(3) / 2
    
    y = r_start
    while True:
        x_start = r_start
        # Shift alternate rows
        if len(centers) % 2 == 1:
            x_start += spacing / 2
            
        x = x_start
        while x <= 1 - r_start:
            centers.append([x, y])
            x += spacing
            if len(centers) >= n_circles:
                break
        y += row_height
        if len(centers) >= n_circles or y > 1 - r_start:
            break
            
    # Pad or trim to exactly n_circles
    if len(centers) < n_circles:
        while len(centers) < n_circles:
            centers.append(rng.uniform(r_start, 1 - r_start, 2))
    else:
        centers = centers[:n_circles]
        
    return np.array(centers)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    best_sum_r = -1.0
    best_centers = None
    best_radii = None
    
    rng = np.random.default_rng(42)
    
    # Optimization function
    def solve_from_init(init_centers, init_r):
        # Variable vector: [x1, y1, ..., x26, y26, r]
        x0 = np.concatenate([init_centers.flatten(), [init_r]])
        
        # Bounds for centers [0, 1] and r [0, 0.5]
        bounds = [(0, 1)] * (2 * n) + [(0, 0.5)]
        
        # Constraints
        cons = []
        
        # 1. Boundary constraints: x - r >= 0, 1 - x - r >= 0 (same for y)
        # These ensure circles are inside the square
        for i in range(n):
            # x_i - r >= 0
            cons.append({'type': 'ineq', 'fun': lambda x, i=i: x[2*i] - x[2*n]})
            # 1 - x_i - r >= 0
            cons.append({'type': 'ineq', 'fun': lambda x, i=i: 1 - x[2*i] - x[2*n]})
            # y_i - r >= 0
            cons.append({'type': 'ineq', 'fun': lambda x, i=i: x[2*i + 1] - x[2*n]})
            # 1 - y_i - r >= 0
            cons.append({'type': 'ineq', 'fun': lambda x, i=i: 1 - x[2*i + 1] - x[2*n]})
            
        # 2. Non-overlap constraints: ||c_i - c_j||^2 >= (2r)^2
        # We use a subset of constraints or all? All is safer but slower.
        # For 26 circles, 325 constraints. Let's include all for correctness.
        for i in range(n):
            for j in range(i + 1, n):
                cons.append({
                    'type': 'ineq',
                    'fun': lambda x, i=i, j=j: np.sum((x[2*i:2*i+2] - x[2*j:2*j+2])**2) - 4 * x[2*n]**2
                })
        
        # Objective: Maximize r -> Minimize -r
        def objective(x):
            return -x[2*n]
            
        try:
            res = opt.minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, 
                               options={'maxiter': 500, 'ftol': 1e-9})
            if res.success and res.fun < 0:
                r = res.x[2*n]
                centers = res.x[:2*n].reshape(-1, 2)
                sum_r = 26 * r
                return centers, np.full(n, r), sum_r
        except Exception:
            pass
        return None, None, -1.0

    # Attempt 1: Hexagonal grid initialization
    init_centers = generate_hex_init(n, rng)
    c, r, s = solve_from_init(init_centers, 0.09)
    if s > best_sum_r:
        best_sum_r = s
        best_centers = c
        best_radii = r

    # Attempt 2: Random initialization
    init_centers = rng.uniform(0.1, 0.9, (n, 2))
    c, r, s = solve_from_init(init_centers, 0.08)
    if s > best_sum_r:
        best_sum_r = s
        best_centers = c
        best_radii = r

    # Attempt 3: Random initialization with different seed/center
    rng2 = np.random.default_rng(123)
    init_centers = rng2.uniform(0.1, 0.9, (n, 2))
    c, r, s = solve_from_init(init_centers, 0.08)
    if s > best_sum_r:
        best_sum_r = s
        best_centers = c
        best_radii = r
        
    # Ensure valid result
    if best_centers is None:
        # Fallback to grid
        best_centers = generate_hex_init(n, np.random.default_rng(0))
        best_radii = np.full(n, 0.05)
        best_sum_r = np.sum(best_radii)

    return best_centers, best_radii, float(best_sum_r)
