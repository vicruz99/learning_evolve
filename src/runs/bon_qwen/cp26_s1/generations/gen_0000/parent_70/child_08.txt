import numpy as np
from scipy.optimize import minimize
import warnings

# Suppress convergence warnings for cleaner output
warnings.filterwarnings('ignore', category=UserWarning)

def obj_and_grad(params, n, W):
    """Computes objective and gradient for the circle packing problem."""
    x = params[0::3]
    y = params[1::3]
    r = params[2::3]
    
    obj = -np.sum(r)
    
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist = np.sqrt(dx**2 + dy**2 + 1e-12)
    
    # Ignore self-distances
    np.fill_diagonal(dist, 1e9)
    
    r_sum = r[:, None] + r[None, :]
    viol = np.maximum(0.0, r_sum - dist)
    
    penalty = np.sum(viol**2)
    
    grad = np.zeros_like(params)
    grad[2::3] = -1.0
    
    dx_dist = dx / dist
    dy_dist = dy / dist
    
    diag_idx = np.arange(n)
    dx_dist[diag_idx, diag_idx] = 0.0
    dy_dist[diag_idx, diag_idx] = 0.0
    
    pair_grad_r = 2.0 * W * np.sum(viol, axis=1)
    pair_grad_x = -2.0 * W * np.sum(viol * dx_dist, axis=1)
    pair_grad_y = -2.0 * W * np.sum(viol * dy_dist, axis=1)
    
    grad[2::3] += pair_grad_r
    grad[0::3] += pair_grad_x
    grad[1::3] += pair_grad_y
    
    # Left boundary
    v_l = np.maximum(0.0, r - x)
    penalty += np.sum(v_l**2)
    grad[2::3] += 2.0 * W * v_l
    grad[0::3] -= 2.0 * W * v_l
    
    # Right boundary
    v_r = np.maximum(0.0, x + r - 1.0)
    penalty += np.sum(v_r**2)
    grad[2::3] += 2.0 * W * v_r
    grad[0::3] += 2.0 * W * v_r
    
    # Bottom boundary
    v_b = np.maximum(0.0, r - y)
    penalty += np.sum(v_b**2)
    grad[2::3] += 2.0 * W * v_b
    grad[1::3] -= 2.0 * W * v_b
    
    # Top boundary
    v_t = np.maximum(0.0, y + r - 1.0)
    penalty += np.sum(v_t**2)
    grad[2::3] += 2.0 * W * v_t
    grad[1::3] += 2.0 * W * v_t
    
    return obj + W * penalty, grad

def get_obj(params, n, W):
    return obj_and_grad(params, n, W)[0]

def get_grad(params, n, W):
    return obj_and_grad(params, n, W)[1]

def run_packing() -> tuple:
    n = 26
    best_score = -np.inf
    best_centers = None
    best_radii = None
    
    bounds = [(0.0, 1.0)]*n + [(0.0, 1.0)]*n + [(0.0, 0.5)]*n
    W_steps = [50.0, 500.0, 2000.0]
    
    for trial in range(20):
        rng = np.random.default_rng(trial * 97 + 11)
        
        # Mix of grid-like and random initializations
        if trial < 6:
            pts = []
            for i in range(5):
                for j in range(5):
                    pts.append((0.1 + i*0.18, 0.1 + j*0.18))
            pts.append((0.5, 0.5))
            x = np.array([p[0] for p in pts])
            y = np.array([p[1] for p in pts])
            x += rng.normal(0, 0.03, n)
            y += rng.normal(0, 0.03, n)
        else:
            x = rng.uniform(0.15, 0.85, n)
            y = rng.uniform(0.15, 0.85, n)
            
        r = np.full(n, 0.03)
        params = np.concatenate([x, y, r])
        
        # Initial clipping
        params = np.clip(params, [0.01]*n + [0.01]*n + [0.01]*n, [0.99]*n + [0.99]*n + [0.5]*n)
        
        # Continuation optimization
        for W in W_steps:
            # Small perturbation to escape local minima
            params += rng.normal(0, 0.002, size=params.shape)
            params = np.clip(params, [0.005]*n + [0.005]*n + [0.005]*n, [0.995]*n + [0.995]*n + [0.5]*n)
            
            res = minimize(get_obj, params, args=(n, W), jac=get_grad, method='L-BFGS-B', 
                           bounds=bounds, options={'maxiter': 600, 'ftol': 1e-10, 'gtol': 1e-6})
            params = res.x
            
        cur_r = params[2::3]
        score = np.sum(cur_r)
        
        if score > best_score:
            best_score = score
            best_centers = np.column_stack((params[0::3], params[1::3]))
            best_radii = cur_r
            
    # Final post-processing to guarantee strict boundary compliance
    best_radii = np.maximum(best_radii, 1e-6)
    best_centers[:, 0] = np.clip(best_centers[:, 0], best_radii, 1.0 - best_radii)
    best_centers[:, 1] = np.clip(best_centers[:, 1], best_radii, 1.0 - best_radii)
    
    return best_centers, best_radii, np.sum(best_radii)
