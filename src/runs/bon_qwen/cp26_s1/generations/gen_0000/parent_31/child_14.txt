import numpy as np

def compute_radii(centers):
    """Compute the maximum valid radius for each circle given fixed centers."""
    n = centers.shape[0]
    radii = np.zeros(n)
    for i in range(n):
        # Distance to boundaries
        r = min(centers[i, 0], 1.0 - centers[i, 0],
                centers[i, 1], 1.0 - centers[i, 1])
        # Distance to other circles
        for j in range(n):
            if i != j:
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                d = np.sqrt(dx*dx + dy*dy)
                if d / 2.0 < r:
                    r = d / 2.0
        radii[i] = r
    return radii
