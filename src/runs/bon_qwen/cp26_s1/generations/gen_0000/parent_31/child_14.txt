import numpy as np
from scipy.optimize import minimize

def hexagonal_initial(n=26):
    """Create a hexagonal lattice arrangement for n circles."""
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.095
    
    r_est = 0.095
    h = r_est * np.sqrt(3)
    w = 2 * r_est
    
    idx = 0
    row = 0
    y = r_est + row * h
    while idx < n and y + r_est <= 1.0:
        if row % 2 == 0:
            x_start = r_est
        else:
            x_start = r_est + w / 2
        
        x = x_start
        while x + r_est <= 1.0 and idx < n:
            centers[idx, 0] = x
            centers[idx, 1] = y
            idx += 1
            x += w
        row += 1
        y = r_est + row * h
    
    while idx < n:
        centers[idx, 0] = 0.5 + 0.05 * np.cos(2 * np.pi * idx / n)
        centers[idx, 1] = 0.5 + 0.05 * np.sin(2 * np.pi * idx / n)
        radii[idx] = 0.05
        idx += 1
    
    return centers, radii

def grid_initial(n=26):
    """Create a grid arrangement."""
    cols = 6
    rows = 5
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.08
    
    idx = 0
    for r in range(rows):
        for c in range(cols):
            if idx >= n:
                break
            x = 0.5 + c * 0.15 - 2 * 0.15
            y = 0.5 + r * 0.2 - 2 * 0.2
            centers[idx, 0] = x
            centers[idx, 1] = y
            idx += 1
        if idx >= n:
            break
    
    while idx < n:
        centers[idx, 0] = 0.5
        centers[idx, 1] = 0.5
        radii[idx] = 0.03
        idx += 1
    
    return centers, radii

def random_initial(n=26):
    """Create a random arrangement."""
    centers = np.random.rand(n, 2) * 0.6 + 0.2
    radii = np.ones(n) * 0.07
    return centers, radii

def penalty_objective(params, n):
    """Compute negative sum of radii plus penalty for constraint violations."""
    centers = params[:2*n].reshape(n, 2)
    radii = params[2*n:]
    
    radii = np.maximum(radii, 1e-10)
    
    obj = -np.sum(radii)
    
    w_boundary = 5000.0
    w_overlap = 8000.0
    
    penalty = 0.0
    
    for i in range(n):
        x = centers[i, 0]
        y = centers[i, 1]
        r = radii[i]
        
        if x - r < 0:
            penalty += w_boundary * (x - r) ** 2
        if x + r > 1:
            penalty += w_boundary * (x + r - 1) ** 2
        if y - r < 0:
            penalty += w_boundary * (y - r) ** 2
        if y + r > 1:
            penalty += w_boundary * (y + r - 1) ** 2
    
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist_sq = dx * dx + dy * dy
            dist = np.sqrt(dist_sq + 1e-12)
            overlap = radii[i] + radii[j] - dist
            if overlap > 0:
                penalty += w_overlap * overlap ** 2
    
    return obj + penalty

def post_process(centers, radii, n):
    """Ensure packing validity by clamping radii."""
    radii = radii.copy()
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        r = min(r, x, 1.0 - x, y, 1.0 - y)
        for j in range(n):
            if i != j:
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                r = min(r, dist - radii[j] + 1e-12)
        radii[i] = max(0.0, r)
    return radii

def optimize_once(n, centers, radii, seed):
    """Run one optimization from given initial configuration."""
    np.random.seed(seed)
    
    params = np.concatenate([centers.flatten(), radii])
    bounds = [(0.01, 0.99)] * (2 * n) + [(1e-6, 0.5)] * n
    
    result = minimize(penalty_objective, params, args=(n,), method='L-BFGS-B', bounds=bounds,
                     options={'maxiter': 15000, 'ftol': 1e-15, 'gtol': 1e-12})
    
    centers_opt = result.x[:2 * n].reshape(n, 2)
    radii_opt = np.maximum(result.x[2 * n:], 1e-6)
    
    return centers_opt, radii_opt, result.fun

def perturbation_search(n, centers, radii, num_trials=50):
    """Local search with random perturbations."""
    best_centers = centers.copy()
    best_radii = radii.copy()
    best_sum = np.sum(radii)
    
    for trial in range(num_trials):
        c_trial = centers.copy()
        r_trial = radii.copy()
        
        # Perturb centers
        c_trial += np.random.randn(n, 2) * 0.005
        c_trial = np.clip(c_trial, 0.01, 0.99)
        
        # Perturb radii slightly
        r_trial += np.random.randn(n) * 0.003
        r_trial = np.maximum(r_trial, 1e-6)
        
        params = np.concatenate([c_trial.flatten(), r_trial])
        bounds = [(0.01, 0.99)] * (2 * n) + [(1e-6, 0.5)] * n
        
        result = minimize(penalty_objective, params, args=(n,), method='L-BFGS-B', bounds=bounds,
                         options={'maxiter': 5000, 'ftol': 1e-14})
        
        c_opt = result.x[:2 * n].reshape(n, 2)
        r_opt = np.maximum(result.x[2 * n:], 1e-6)
        
        r_opt = post_process(c_opt, r_opt, n)
        current_sum = np.sum(r_opt)
        
        if current_sum > best_sum:
            best_centers = c_opt.copy()
            best_radii = r_opt.copy()
            best_sum = current_sum
    
    return best_centers, best_radii, best_sum

def run_packing():
    n = 26
    
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Stage 1: Multiple initial configurations with optimization
    for seed in range(80):

        else:
            centers, radii = random_initial(n)

        radii_opt = post_process(centers_opt, radii_opt, n)
        current_sum = np.sum(radii_opt)
        
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers_opt.copy()
            best_radii = radii_opt.copy()
    
    # Stage 2: Refinement with perturbation search around best solution
    if best_centers is not None:
        best_centers, best_radii, best_sum = perturbation_search(n, best_centers, best_radii, 100)
    
    # Stage 3: Final fine-tuning optimization
    if best_centers is not None:
        params = np.concatenate([best_centers.flatten(), best_radii])
        bounds = [(0.001, 0.999)] * (2 * n) + [(1e-6, 0.5)] * n
        
        result = minimize(penalty_objective, params, args=(n,), method='L-BFGS-B', bounds=bounds,
                         options={'maxiter': 30000, 'ftol': 1e-16, 'gtol': 1e-13})
        
        centers_final = result.x[:2 * n].reshape(n, 2)
        radii_final = np.maximum(result.x[2 * n:], 1e-6)
        radii_final = post_process(centers_final, radii_final, n)
        
        final_sum = np.sum(radii_final)
        if final_sum > best_sum:
            best_sum = final_sum
            best_centers = centers_final
            best_radii = radii_final
    
    if best_centers is None:
        best_centers, best_radii = hexagonal_initial(n)
        best_radii = post_process(best_centers, best_radii, n)
        best_sum = np.sum(best_radii)
    
    return best_centers, best_radii, float(best_sum)
