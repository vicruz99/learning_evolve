The problem asks to pack 26 circles in a unit square to maximize the sum of their radii. This is a constrained optimization problem where the variables are the centers $(x_i, y_i)$ and radii $r_i$ of the 26 circles. The constraints are non-overlap (distance between centers $\ge$ sum of radii) and boundary containment (centers $\pm$ radii within $[0,1]$).

A robust approach for this type of problem is to decouple the variables. We can optimize the centers first, and for any fixed set of centers, the optimal radii can be found by solving a Linear Programming (LP) problem.
1.  **LP for Radii**: Given centers, we want to maximize $\sum r_i$ subject to:
    *   $r_i \ge 0$
    *   $r_i \le x_i, r_i \le 1-x_i, r_i \le y_i, r_i \le 1-y_i$ (Boundary)
    *   $r_i + r_j \le \sqrt{(x_i-x_j)^2 + (y_i-y_j)^2}$ (Non-overlap)
    This LP is small (26 variables, ~400 constraints) and can be solved very quickly using `scipy.optimize.linprog`.

2.  **Optimization of Centers**: We define an objective function $f(c_1, \dots, c_{26})$ that returns the negative of the maximum sum of radii for those centers. We then use a derivative-free optimizer like `Nelder-Mead` (via `scipy.optimize.minimize`) to find the centers that minimize this objective (maximize sum of radii).

3.  **Initialization**: To avoid poor local minima, we use multiple initialization strategies:
    *   A grid-based initialization (5x5 grid plus one extra circle), which provides a structured starting point close to high-density packings.
    *   Multiple random restarts to explore different configurations.

4.  **Implementation Details**:
    *   We precompute the sparse constraint matrix $A_{ub}$ for the LP, as its structure depends only on the number of circles, not their positions. Only the right-hand side vector $b_{ub}$ changes with centers.
    *   We use the `highs` method in `linprog` for speed.
    *   We ensure all helper functions are top-level to comply with the "no closures" rule.

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix

def prepare_constraints(n):
    """
    Precompute the sparse constraint matrix A_ub for the LP.
    Constraints:
    1. Boundary: 4 per circle (r_i <= x, 1-x, y, 1-y)
    2. Pairwise: 1 per pair (r_i + r_j <= dist)
    """
    n_bound = 4 * n
    n_pairs = n * (n - 1) // 2
    n_constraints = n_bound + n_pairs
    
    rows = []
    cols = []
    data = []
    
    # Boundary constraints
    # Each circle i has 4 constraints involving only r_i
    for i in range(n):
        # r_i <= x_i
        rows.append(4*i); cols.append(i); data.append(1.0)
        # r_i <= 1-x_i
        rows.append(4*i+1); cols.append(i); data.append(1.0)
        # r_i <= y_i
        rows.append(4*i+2); cols.append(i); data.append(1.0)
        # r_i <= 1-y_i
        rows.append(4*i+3); cols.append(i); data.append(1.0)
        
    # Pairwise constraints
    # r_i + r_j <= dist_ij
    row_idx = n_bound
    for i in range(n):
        for j in range(i + 1, n):
            rows.append(row_idx); cols.append(i); data.append(1.0)
            rows.append(row_idx); cols.append(j); data.append(1.0)
            row_idx += 1
            
    A_ub = csr_matrix((data, (rows, cols)), shape=(n_constraints, n))
    return A_ub

def get_b_ub(centers, n):
    """
    Compute the right-hand side vector b_ub for the LP given current centers.
    """
    n_bound = 4 * n
    n_pairs = n * (n - 1) // 2
    b_ub = np.zeros(n_bound + n_pairs)
    
    # Boundary constraints RHS
    for i in range(n):
        x, y = centers[i]
        b_ub[4*i] = x
        b_ub[4*i+1] = 1-x
        b_ub[4*i+2] = y
        b_ub[4*i+3] = 1-y
        
    # Pairwise constraints RHS
    row_idx = n_bound
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            b_ub[row_idx] = dist
            row_idx += 1
    return b_ub

def solve_lp_radii(centers, A_ub):
    """
    Solve the LP to find radii that maximize sum for fixed centers.
    Returns (sum_radii, radii_array).
    """
    n = len(centers)
    # Objective: minimize -sum(r) => maximize sum(r)
    c_obj = -np.ones(n)
    
    b_ub = get_b_ub(centers, n)
    
    # Bounds: 0 <= r_i
    bounds = [(0, None) for _ in range(n)]
    
    # Solve LP
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return -res.fun, res.x
        else:
            # Fallback if LP fails (unlikely with feasible r=0)
            return 0, np.zeros(n)
    except Exception:
        return 0, np.zeros(n)

def objective(centers_flat, n, A_ub):
    """
    Objective function for scipy.optimize.minimize.
    Minimizes the negative sum of radii.
    """
    centers = centers_flat.reshape(n, 2)
    # Clip centers to [0,1] to ensure valid boundary logic
    centers = np.clip(centers, 0, 1)
    
    val, _ = solve_lp_radii(centers, A_ub)
    return -val

def run_packing():
    n = 26
    A_ub = prepare_constraints(n)
    
    best_sum = -np.inf
    best_centers = None
    
    # Helper to record best result
    def update_best(res, current_sum):
        nonlocal best_sum, best_centers
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = res.x.reshape(n, 2)

    # --- Initialization 1: Grid-based ---
    # 5x5 grid covers 25 circles, add 1 in a gap
    x_pts = np.linspace(0.1, 0.9, 5)
    y_pts = np.linspace(0.1, 0.9, 5)
    grid = np.array([(x, y) for x in x_pts for y in y_pts])
    extra = np.array([[0.5, 0.05]])
    init1 = np.vstack([grid, extra])
    np.random.shuffle(init1)
    
    x0 = init1.flatten()
    res = minimize(objective, x0, args=(n, A_ub), method='Nelder-Mead', 
                   options={'maxiter': 1000, 'xatol': 1e-6, 'fatol': 1e-6})
    update_best(res, -res.fun)

    # --- Initialization 2: Random Restarts ---
    np.random.seed(42)
    for seed in range(10):
        init_random = np.random.rand(n, 2)
        x0 = init_random.flatten()
        res = minimize(objective, x0, args=(n, A_ub), method='Nelder-Mead', 
                       options={'maxiter': 1000, 'xatol': 1e-6, 'fatol': 1e-6})
        update_best(res, -res.fun)
        
    # --- Final Radii Calculation ---
    if best_centers is None:
        best_centers = np.random.rand(n, 2)
    else:
        best_centers = np.clip(best_centers, 0, 1)
        
    final_sum, final_radii = solve_lp_radii(best_centers, A_ub)
    
    # Ensure non-negative radii
    final_radii = np.maximum(final_radii, 0)
    
    return best_centers, final_radii, final_sum
