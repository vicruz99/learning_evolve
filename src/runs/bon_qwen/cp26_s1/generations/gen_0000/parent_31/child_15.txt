To maximize the sum of radii for 26 circles in a unit square, we treat the problem as an optimization of circle centers. While the radii can vary, their optimal values for any fixed set of centers are determined by the nearest neighbors and the boundaries of the square. 

The strategy involves a two-phase iterative process:
1.  **Initialization:** Start with a deterministic hexagonal lattice arrangement of 26 centers, which is a near-optimal dense packing configuration.
2.  **Iterative Optimization:** We alternate between solving a Linear Programming (LP) problem to find the maximum possible radii for the current centers and applying "repulsive forces" to move the centers.
    *   **LP Step:** For the current positions, we calculate the maximum radius for each circle $r_i$ such that $r_i + r_j \le \text{dist}(i, j)$ and the circles remain within the square. This is solved efficiently using `scipy.optimize.linprog`.
    *   **Force Step:** We identify the "tight" constraints (where circles or boundaries are limiting the radius) and push the centers apart to relax these constraints. We also add a small random perturbation to avoid getting stuck in local minima.
3.  **Finalization:** After several iterations, we re-solve the LP for the best set of centers and return the result.

This approach decouples the geometric constraints from the optimization of the sum, allowing for a robust and computationally efficient search for a high-quality packing.

import numpy as np
from scipy.optimize import linprog

def get_boundary_limits(centers):
    """Calculate the maximum radius allowed by the square boundary for each center."""
    x, y = centers[:, 0], centers[:, 1]
    left = x
    right = 1.0 - x
    bottom = y
    top = 1.0 - y
    return np.minimum(np.minimum(left, right), np.minimum(bottom, top))

def solve_radii(centers):
    """
    Solve for the optimal radii given fixed centers using Linear Programming.
    Maximizes sum(r_i) subject to r_i + r_j <= dist(i,j) and r_i <= boundary_dist(i).
    """
    n = centers.shape[0]
    
    # Objective: Maximize sum(r_i) -> Minimize -sum(r_i)
    c = -np.ones(n)
    
    # Prepare inequality constraints A_ub @ r <= b_ub
    # Constraint types:
    # 1. Pairwise: r_i + r_j <= dist_ij
    # 2. Boundary: r_i <= dist_to_boundary_i
    
    num_pairs = n * (n - 1) // 2
    num_constraints = num_pairs + n
    
    A_ub = np.zeros((num_constraints, n))
    b_ub = np.zeros(num_constraints)
    
    # Pairwise constraints
    row_idx = 0
    # Compute pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    dists = np.triu(dists, k=1) # Upper triangle
    
    count = 0
    for i in range(n):
        for j in range(i + 1, n):
            d = dists[i, j]
            # Avoid zero distance causing issues, though unlikely with good init
            if d < 1e-9:
                d = 1e-9
            A_ub[row_idx, i] = 1.0
            A_ub[row_idx, j] = 1.0
            b_ub[row_idx] = d
            row_idx += 1
            count += 1
            
    # Boundary constraints
    boundary_limits = get_boundary_limits(centers)
    for i in range(n):
        A_ub[row_idx, i] = 1.0
        b_ub[row_idx] = boundary_limits[i]
        row_idx += 1
        
    # Bounds for radii: r_i >= 0
    bounds = [(0, None) for _ in range(n)]
    
    # Solve LP
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return res.x, -res.fun
    else:
        # Fallback if LP fails (should not happen)
        return np.zeros(n), 0.0

def compute_forces(centers, radii):
    """
    Compute repulsive forces based on tight constraints.
    Returns a force vector for each center.
    """
    n = centers.shape[0]
    forces = np.zeros_like(centers)
    
    # Identify tight constraints
    # Pairwise: r_i + r_j approx dist(i, j)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    dists = np.triu(dists, k=1)
    
    # Tolerance for "tight" constraint
    tol = 1e-4
    
    for i in range(n):
        # Boundary forces
        xi, yi = centers[i]
        ri = radii[i]
        
        # Distance to boundaries
        d_left = xi - ri
        d_right = (1.0 - xi) - ri
        d_bottom = yi - ri
        d_top = (1.0 - yi) - ri
        
        # If close to boundary, push away
        force_mag = 1.0
        if d_left < tol: forces[i, 0] += force_mag
        if d_right < tol: forces[i, 0] -= force_mag
        if d_bottom < tol: forces[i, 1] += force_mag
        if d_top < tol: forces[i, 1] -= force_mag
        
        # Pairwise forces
        for j in range(i + 1, n):
            rj = radii[j]
            d = dists[i, j]
            # If touching
            if d < (ri + rj) + tol:
                # Repulsive force direction: i away from j
                # Vector from j to i
                if d > 1e-9:
                    dir_vec = (centers[i] - centers[j]) / d
                    forces[i] += dir_vec
                    forces[j] -= dir_vec # Newton's 3rd law
    
    return forces

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Number of circles
    n = 26
    
    # 1. Initialization: Hexagonal Grid
    centers = np.zeros((n, 2))
    idx = 0
    
    # Approximate spacing for 26 circles in unit square
    # Area ~ 1, 26 circles -> r ~ 0.1. Spacing ~ 0.2
    # Let's try a grid of rows
    # 5 rows: 6, 5, 6, 5, 4 circles = 26
    rows_config = [6, 5, 6, 5, 4]
    
    y_spacing = 1.0 / (len(rows_config) + 1)
    
    for r_idx, count in enumerate(rows_config):
        y = (r_idx + 1) * y_spacing
        # X spacing based on count
        x_spacing = 1.0 / (count + 1)
        for c_idx in range(count):
            x = (c_idx + 1) * x_spacing
            # Offset for hexagonal packing
            if r_idx % 2 == 1:
                x += x_spacing / 2
            
            # Keep within bounds
            x = np.clip(x, 0.05, 0.95)
            y = np.clip(y, 0.05, 0.95)
            
            centers[idx, 0] = x
            centers[idx, 1] = y
            idx += 1

    # 2. Optimization Loop
    best_sum = 0.0
    best_centers = centers.copy()
    best_radii = np.zeros(n)
    
    # Parameters
    iterations = 200
    step_size = 0.05
    noise_scale = 0.02
    
    rng = np.random.default_rng(42)
    
    current_centers = centers.copy()
    
    for it in range(iterations):
        # Solve for radii
        radii, current_sum = solve_radii(current_centers)
        
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = current_centers.copy()
            best_radii = radii.copy()
        
        # Compute forces
        forces = compute_forces(current_centers, radii)
        
        # Add random noise to escape local minima
        noise = rng.normal(0, noise_scale, size=current_centers.shape)
        
        # Update centers
        current_centers += step_size * forces + noise
        
        # Project back to valid range [0, 1] (loosely, to keep centers inside)
        # Actually, centers must be in [0,1] for the circle to be valid if r>=0.
        # The LP handles r <= boundary, but if center is at 1.1, boundary limit is negative -> r=0.
        # Better to keep centers inside.
        current_centers = np.clip(current_centers, 0.01, 0.99)
        
        # Annealing: reduce step size and noise
        step_size *= 0.99
        noise_scale *= 0.99

    # Final refinement on best centers found
    # Run a few more precise LP solves if needed, but the loop should have found good spots.
    # Let's re-solve radii for the best centers to be sure.
    final_radii, final_sum = solve_radii(best_centers)
    
    return best_centers, final_radii, final_sum
