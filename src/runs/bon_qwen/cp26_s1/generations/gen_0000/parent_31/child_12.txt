def _make_overlap_constraint(i, j):
    """Creates a constraint function for non-overlapping circles i and j."""
    idx_i = 3 * i
    idx_j = 3 * j
    def constraint(v):
        dx = v[idx_i] - v[idx_j]
        dy = v[idx_i + 1] - v[idx_j + 1]
        dist = np.sqrt(dx**2 + dy**2)
        return dist - v[idx_i + 2] - v[idx_j + 2]
    return constraint

def _make_boundary_constraint(i, side):
    """Creates a boundary constraint for circle i. side: 'left', 'right', 'bottom', 'top'"""
    idx = 3 * i

        return lambda v: v[idx] - v[idx + 2]  # x >= r
