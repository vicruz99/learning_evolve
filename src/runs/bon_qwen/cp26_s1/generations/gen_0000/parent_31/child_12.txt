import numpy as np
from scipy.optimize import minimize
import warnings

warnings.filterwarnings('ignore')

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    num_runs = 10
    
    best_centers = None
    best_radii = None
    best_sum = -1.0

    # Define constraints for scipy.optimize
    # Variables: x_0, y_0, r_0, x_1, y_1, r_1, ...
    # Index mapping: 3*i, 3*i+1, 3*i+2

    def get_bounds(n):
        bounds = []
        for _ in range(n):
            bounds.append((0.0, 1.0)) # x
            bounds.append((0.0, 1.0)) # y
            bounds.append((0.0, 0.5)) # r (max possible radius in unit square is 0.5)
        return bounds

    def objective(vars):
        # Maximize sum of radii -> Minimize -sum(radii)
        r = vars[2::3]
        return -np.sum(r)

    def create_constraints(n):
        constraints = []
        
        # 1. Boundary constraints: r <= x <= 1-r  =>  x - r >= 0, 1 - x - r >= 0
        # 2. Boundary constraints: r <= y <= 1-r  =>  y - r >= 0, 1 - y - r >= 0
        for i in range(n):
            idx_x, idx_y, idx_r = 3*i, 3*i+1, 3*i+2
            
            # x - r >= 0
            constraints.append({'type': 'ineq', 'fun': lambda v, i=i: v[idx_x] - v[idx_r]})
            # 1 - x - r >= 0
            constraints.append({'type': 'ineq', 'fun': lambda v, i=i: 1.0 - v[idx_x] - v[idx_r]})
            # y - r >= 0
            constraints.append({'type': 'ineq', 'fun': lambda v, i=i: v[idx_y] - v[idx_r]})
            # 1 - y - r >= 0
            constraints.append({'type': 'ineq', 'fun': lambda v, i=i: 1.0 - v[idx_y] - v[idx_r]})

        # 3. Non-overlap constraints: dist^2 >= (r_i + r_j)^2
        # (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
        for i in range(n):
            for j in range(i + 1, n):
                idx_xi, idx_yi, idx_ri = 3*i, 3*i+1, 3*i+2
                idx_xj, idx_yj, idx_rj = 3*j, 3*j+1, 3*j+2
                
                def non_overlap(v, i=i, j=j, 
                               xi=3*i, yi=3*i+1, ri=3*i+2,
                               xj=3*j, yj=3*j+1, rj=3*j+2):
                    dist_sq = (v[xi] - v[xj])**2 + (v[yi] - v[yj])**2
                    sum_r_sq = (v[ri] + v[rj])**2
                    return dist_sq - sum_r_sq
                
                constraints.append({'type': 'ineq', 'fun': non_overlap})
                
        return constraints

    constraints = create_constraints(n)
    bounds = get_bounds(n)

    # Hexagonal packing initialization
    def init_hex(n, scale=0.085):
        centers = []
        r = scale
        # Approximate rows
        row_idx = 0
        col_idx = 0
        
        # Try to fit in a roughly hexagonal pattern
        # Vertical spacing sqrt(3)/2 * diameter ~ 0.866 * 2r
        # Horizontal spacing 2r
        
        dy = np.sqrt(3) * r
        dx = 2 * r
        
        x_start = r
        y_start = r
        
        count = 0
        while count < n:
            x = x_start + col_idx * dx
            y = y_start + row_idx * dy
            
            # Shift odd rows
            if row_idx % 2 == 1:
                x += dx / 2
            
            if x <= 1 - r and y <= 1 - r:
                centers.append([x, y, r])
                count += 1
                col_idx += 1
            else:
                row_idx += 1
                col_idx = 0
        
        return np.array(centers).flatten()

    for run in range(num_runs):
        # Perturb initial configuration slightly for diversity
        init_vec = init_hex(n, scale=0.09 + run * 0.001)
        noise = np.random.normal(0, 0.005, init_vec.shape)
        # Ensure radii are not negative due to noise
        noise[2::3] = np.abs(noise[2::3])
        x0 = init_vec + noise
        
        # Clip bounds for x0
        x0 = np.clip(x0, 0, 1)
        x0[2::3] = np.clip(x0[2::3], 0, 0.5)

        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, 
                           constraints=constraints, options={'maxiter': 1000, 'ftol': 1e-9})
            
            if res.success:
                centers_res = res.x[0::3], res.x[1::3]
                radii_res = res.x[2::3]
                current_sum = np.sum(radii_res)
                
                # Validate inside the loop to discard invalid numerical results
                if validate_packing(np.column_stack(centers_res), radii_res):
                    if current_sum > best_sum:
                        best_sum = current_sum
                        best_centers = np.column_stack(centers_res)
                        best_radii = radii_res
        except Exception:
            continue

    # Fallback if optimization fails to produce valid result (should not happen with good init)
    if best_centers is None:
        # Simple grid fallback
        best_centers = np.zeros((n, 2))
        best_radii = np.full(n, 0.05)
        best_sum = n * 0.05
        # Fill grid
        cols = 5
        rows = 6
        step_x = 1.0 / (cols + 1)
        step_y = 1.0 / (rows + 1)
        idx = 0
        for r in range(rows):
            for c in range(cols):
                if idx < n:
                    best_centers[idx] = [step_x * (c + 1), step_y * (r + 1)]
                    idx += 1

    # Final validation
    if not validate_packing(best_centers, best_radii):
        # If somehow invalid, return safe empty-like or grid (though unlikely given checks)
        best_sum = 0.0

    return best_centers, best_radii, best_sum

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    n = centers.shape[0]
    if np.isnan(centers).any():
        return False
    if np.isnan(radii).any():
        return False
    for i in range(n):
        if radii[i] < 0:
            return False
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            return False
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False
    return True
