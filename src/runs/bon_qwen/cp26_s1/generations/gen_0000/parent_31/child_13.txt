import numpy as np
import math
from scipy.optimize import minimize

# --- Helper Functions ---

def calculate_pairwise_dist_matrix(centers):
    """Calculate distance matrix for centers"""
    n = centers.shape[0]
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist_matrix = np.sqrt(np.sum(diff**2, axis=2))
    return dist_matrix

def hexagonal_init(n, square_size=1.0):
    """
    Initialize centers in a hexagonal grid pattern.
    n: number of circles
    """
    centers = []
    y = 0.0
    while len(centers) < n:
        x = 0.0
        while len(centers) < n and x < square_size:
            centers.append([x, y])
            x += math.sqrt(3)/2
        y += 0.5
    centers = np.array(centers[:n])
    # Randomly perturb slightly to break symmetry and avoid singularities
    centers += np.random.uniform(-0.05, 0.05, centers.shape)
    # Clip to valid range for optimization start
    centers = np.clip(centers, 0.05, 0.95)
    return centers

def random_init(n, square_size=1.0):
    """Initialize centers randomly within a safe margin"""
    return np.random.uniform(0.1, 0.9, (n, 2))

def optimize_packing(centers):
    """
    Optimize radii and centers to maximize sum of radii.
    """
    n = centers.shape[0]
    
    # Initialize variables: [x1, y1, r1, x2, y2, r2, ...]
    # We fix the centers initially or allow them to move. 
    # To be safe, we allow centers to move but penalize boundary heavily.
    # Initial radii based on safe minimum distance
    r_init = np.full(n, 0.01)
    
    # Flatten state: centers + radii
    x0 = np.concatenate([centers.flatten(), r_init])
    
    def objective(x):
        radii = x[2::3]
        return -np.sum(radii) # Maximize sum of radii

    def constraints_func(x):
        c_list = []
        centers = x[:2*n].reshape(n, 2)
        radii = x[2*n:]
        
        # Boundary constraints: r <= x, r <= 1-x, r <= y, r <= 1-y
        # r - x <= 0
        for i in range(n):
            c_list.append({'type': 'ineq', 'fun': lambda x, i=i: x[3*i] - x[3*i+2]}) # x - r >= 0 -> r - x <= 0 (Inequality type usually >= 0)
            # Actually scipy ineq means fun(x) >= 0. So x - r >= 0.
            
            c_list.append({'type': 'ineq', 'fun': lambda x, i=i: (1.0 - x[3*i]) - x[3*i+2]}) # 1-x - r >= 0
            c_list.append({'type': 'ineq', 'fun': lambda x, i=i: x[3*i+1] - x[3*i+2]}) # y - r >= 0
            c_list.append({'type': 'ineq', 'fun': lambda x, i=i: (1.0 - x[3*i+1]) - x[3*i+2]}) # 1-y - r >= 0
            
            # Non-negative radius
            c_list.append({'type': 'ineq', 'fun': lambda x, i=i: x[3*i+2]})

        # Overlap constraints: dist >= r_i + r_j  =>  dist - (r_i + r_j) >= 0
        # To avoid O(n^2) constraints which are slow, we only check close pairs?
        # But we don't know close pairs. However, 26 circles is small enough.
        # Let's use a vectorized approach or just loop.
        for i in range(n):
            for j in range(i + 1, n):
                xi, yi = x[3*i], x[3*i+1]
                xj, yj = x[3*j], x[3*j+1]
                ri, rj = x[3*i+2], x[3*j+2]
                dist = math.sqrt((xi - xj)**2 + (yi - yj)**2)
                # Constraint: dist - ri - rj >= 0
                c_list.append({'type': 'ineq', 'fun': lambda x, i=i, j=j: math.sqrt((x[3*i]-x[3*j])**2 + (x[3*i+1]-x[3*j+1])**2) - x[3*i+2] - x[3*j+2]})
        
        return c_list

    # Constraints
    bounds = [(0.0, 1.0) for _ in range(2*n)] # Centers
    bounds += [(0.0, 0.5) for _ in range(n)]   # Radii

    # Run optimization
    try:
        res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints_func(x0), options={'maxiter': 200, 'ftol': 1e-9})
        if res.success:
            return res.x, -res.fun
        else:
            # Return best found even if not converged
            return x0, -objective(x0)
    except Exception as e:
        print(f"Optimization failed: {e}")
        return x0, 0.0

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    n = centers.shape[0]
    if np.isnan(centers).any() or np.isnan(radii).any():
        return False
    for i in range(n):
        if radii[i] < 0:
            return False
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            return False
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False
    return True

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    best_centers = None
    best_radii = None
    max_sum = 0.0

        hexagonal_init(n),
        random_init(n),
        hexagonal_init(n) + np.random.uniform(-0.1, 0.1, (n, 2))
    ]
    
    for init_centers in initializations:
        # Ensure valid start
        init_centers = np.clip(init_centers, 0.01, 0.99)
        
        result_x, current_sum = optimize_packing(init_centers)
        
        c_res = result_x[:2*n].reshape(n, 2)
        r_res = result_x[2*n:]
        
        # Validate
        if validate_packing(c_res, r_res):
            if current_sum > max_sum:
                max_sum = current_sum
                best_centers = c_res
                best_radii = r_res

    # If optimization failed to find valid, fallback to 5x5 grid + 1 small

        centers = []
        for r in range(5):

        max_sum = np.sum(best_radii)

        best_centers = np.array(centers)
        best_radii = np.array([0.1]*25 + [0.041])
        max_sum = np.sum(best_radii)

    return best_centers, best_radii, max_sum
