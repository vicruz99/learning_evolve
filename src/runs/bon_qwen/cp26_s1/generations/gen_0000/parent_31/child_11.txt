import numpy as np
from scipy.optimize import minimize


class CircleBoundaryConstraint:
    """Constraints for single circle: boundary and non-negativity."""
    def __init__(self, idx, ctype):
        self.idx = idx
        self.ctype = ctype

    def __call__(self, v):
        i = self.idx
        if self.ctype == 0:
            return v[3 * i] - v[3 * i + 2]
        elif self.ctype == 1:
            return 1.0 - v[3 * i] - v[3 * i + 2]
        elif self.ctype == 2:
            return v[3 * i + 1] - v[3 * i + 2]
        elif self.ctype == 3:
            return 1.0 - v[3 * i + 1] - v[3 * i + 2]
        else:
            return v[3 * i + 2]


class OverlapConstraint:
    """Non-overlap constraint between two circles."""
    def __init__(self, i, j):
        self.i = i
        self.j = j

    def __call__(self, v):
        dx = v[3 * self.i] - v[3 * self.j]
        dy = v[3 * self.i + 1] - v[3 * self.j + 1]
        d = np.sqrt(dx * dx + dy * dy)
        return d - v[3 * self.i + 2] - v[3 * self.j + 2]


def objective(v):
    """Negative sum of radii (to minimize)."""
    n = 26
    s = 0.0
    for i in range(n):
        s += v[3 * i + 2]
    return -s


def build_constraints(n):
    """Build all constraints for the optimization."""
    constr = []
    for i in range(n):
        for ct in range(5):
            constr.append({'type': 'ineq', 'fun': CircleBoundaryConstraint(i, ct)})
    for i in range(n):
        for j in range(i + 1, n):
            constr.append({'type': 'ineq', 'fun': OverlapConstraint(i, j)})
    return constr


def hex_init(n, pattern_variant=0):
    """Initialize circles in hexagonal pattern."""
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.095)

    idx = 0
    row = 0
    spacing_x = 0.18
    spacing_y = 0.156

    if pattern_variant == 0:
        base_x = 0.09
        base_y = 0.09
    elif pattern_variant == 1:
        base_x = 0.1
        base_y = 0.12
    else:
        base_x = 0.08
        base_y = 0.08

    while idx < n:
        for col in range(7):
            if idx >= n:
                break
            x = base_x + col * spacing_x + (row % 2) * spacing_x * 0.5
            y = base_y + row * spacing_y
            if x >= 0 and x <= 1 and y >= 0 and y <= 1:
                centers[idx] = [x, y]
                idx += 1
        row += 1

    return centers, radii


def grid_init(n, spacing=0.18, offset=0.1):
    """Initialize circles in a grid pattern."""
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.085)

    idx = 0
    for row in range(6):
        for col in range(5):
            if idx >= n:
                break
            x = offset + col * spacing
            y = offset + row * spacing
            centers[idx] = [x, y]
            idx += 1

    return centers, radii


def force_refine(centers, radii, n, iterations=3000):
    """Force-based refinement to improve packing."""
    centers = centers.copy()
    radii = radii.copy()

    for iteration in range(iterations):
        forces = np.zeros_like(centers)

        # Compute repulsion forces
        for i in range(n):
            # Boundary forces
            for d in range(2):
                if centers[i, d] - radii[i] < 0.002:
                    forces[i, d] += 80.0 * (0.002 - (centers[i, d] - radii[i]))
                if centers[i, d] + radii[i] > 0.998:
                    forces[i, d] -= 80.0 * ((centers[i, d] + radii[i]) - 0.998)

            # Repulsion from other circles
            for j in range(i + 1, n):
                diff = centers[i] - centers[j]
                dist = np.sqrt(np.sum(diff * diff)) + 1e-12
                min_dist = radii[i] + radii[j] + 0.001

                if dist < min_dist:
                    repulsion_strength = (min_dist - dist) / (dist + 1e-12)
                    force_mag = repulsion_strength * 150.0
                    forces[i] += diff * force_mag
                    forces[j] -= diff * force_mag

        # Update positions
        lr = 0.008 / (1.0 + iteration * 0.0008)
        centers += forces * lr

        # Update radii: try to grow each circle to max possible
        for i in range(n):
            max_r = min(centers[i, 0], 1.0 - centers[i, 0],
                        centers[i, 1], 1.0 - centers[i, 1])
            for j in range(n):
                if i != j:
                    dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                    available = dist - radii[j]
                    if available < max_r:

            target_r = max(1e-8, max_r)

        # Clamp positions

            centers[i, 1] = max(radii[i] + 1e-9, min(1.0 - radii[i] - 1e-9, centers[i, 1]))
