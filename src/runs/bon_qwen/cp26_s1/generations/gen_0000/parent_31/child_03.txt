

<strategy>
1.  **Equal Radius Relaxation**: Use a binary search to find the maximum radius $r$ for which 26 equal circles can be packed into the unit square. For each candidate $r$, run a force-based relaxation algorithm with multiple random initializations to resolve overlaps and settle the circles into a dense configuration.
2.  **Linear Programming (LP) for Radii Optimization**: Once a valid center configuration is found from the equal-radius packing, fix the centers and solve a Linear Programming problem to maximize the sum of radii $\sum r_i$. This allows circles to adjust their individual radii to exploit boundary gaps and asymmetries, potentially increasing the total sum beyond the equal-radius limit.
3.  **Constraints**: The LP constraints will enforce non-overlap ($r_i + r_j \le \text{dist}_{ij}$) and boundary containment ($r_i \le x_i, r_i \le 1-x_i$, etc.).
4.  **Implementation**: Define top-level helper functions for relaxation and LP solving. The main `run_packing` function orchestrates the binary search and LP.
</strategy>

import numpy as np
from scipy.optimize import linprog

def relax_packing(centers, radii, steps=5000, dt=0.005):
    """
    Performs force-directed relaxation to minimize overlaps for given radii.
    """
    n = centers.shape[0]
    pos = centers.copy()
    r = radii
    
    vel = np.zeros_like(pos)
    
    for step in range(steps):
        forces = np.zeros_like(pos)
        
        # Pairwise repulsion
        for i in range(n):
            for j in range(i + 1, n):
                diff = pos[i] - pos[j]
                dist_sq = np.dot(diff, diff)
                if dist_sq < 1e-14:
                    continue
                dist = np.sqrt(dist_sq)
                min_dist = r[i] + r[j]
                
                if dist < min_dist:
                    overlap = min_dist - dist
                    # Force direction: push i away from j
                    f_vec = (diff / dist) * overlap
                    forces[i] += f_vec
                    forces[j] -= f_vec
                    
        # Boundary forces
        for i in range(n):
            # Left wall
            if pos[i, 0] < r[i]:
                forces[i, 0] += (r[i] - pos[i, 0])
            # Right wall
            if pos[i, 0] > 1 - r[i]:
                forces[i, 0] -= (pos[i, 0] - (1 - r[i]))
            # Bottom wall
            if pos[i, 1] < r[i]:
                forces[i, 1] += (r[i] - pos[i, 1])
            # Top wall
            if pos[i, 1] > 1 - r[i]:
                forces[i, 1] -= (pos[i, 1] - (1 - r[i]))
                
        # Integrate
        vel += forces * dt
        vel *= 0.95 # Damping
        pos += vel * dt
        
        # Hard clamp to prevent numerical drift outside [0,1]
        pos = np.clip(pos, 0, 1)
        
    return pos

def try_pack_equal(n, r, num_starts=10):
    """
    Attempts to pack n circles of equal radius r.
    Returns (valid, centers) or (False, None).
    """
    for _ in range(num_starts):
        # Random initial positions within valid range [r, 1-r]
        margin = r
        centers = np.random.uniform(margin, 1-margin, size=(n, 2))
        radii = np.full(n, r)
        
        # Relax positions
        final_centers = relax_packing(centers, radii, steps=5000)
        
        # Validate final configuration
        valid = True
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((final_centers[i] - final_centers[j]) ** 2))
                if dist < 2 * r - 1e-9:
                    valid = False
                    break
            if not valid:
                break
                
            # Check boundaries
            if final_centers[i, 0] < r - 1e-9 or final_centers[i, 0] > 1 - r + 1e-9:
                valid = False
                break
            if final_centers[i, 1] < r - 1e-9 or final_centers[i, 1] > 1 - r + 1e-9:
                valid = False
                break
                
        if valid:
            return True, final_centers
            
    return False, None

def optimize_radii_lp(centers):
    """
    Given fixed centers, maximize sum of radii using Linear Programming.
    """
    n = centers.shape[0]
    
    # Objective: maximize sum(r_i) => minimize -sum(r_i)
    c = -np.ones(n)
    
    # Constraints A_ub * r <= b_ub
    A_ub = []
    b_ub = []
    
    # Pairwise constraints: r_i + r_j <= dist_ij
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dist)
            
    # Boundary constraints:
    # r_i <= x_i
    # r_i <= 1 - x_i
    # r_i <= y_i
    # r_i <= 1 - y_i
    
    for i in range(n):
        row = np.zeros(n)
        row[i] = 1.0
        
        # r_i <= x_i
        A_ub.append(row)
        b_ub.append(centers[i, 0])
        
        # r_i <= 1 - x_i
        A_ub.append(row)
        b_ub.append(1.0 - centers[i, 0])
        
        # r_i <= y_i
        A_ub.append(row)
        b_ub.append(centers[i, 1])
        
        # r_i <= 1 - y_i
        A_ub.append(row)
        b_ub.append(1.0 - centers[i, 1])
        
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    # Bounds: r_i >= 0
    bounds = [(0, None)] * n
    
    # Solve LP
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return res.x
    else:
        # Fallback: return small equal radii if LP fails
        return np.full(n, 0.05)

def run_packing():
    n = 26
    
    # Binary search for the maximum equal radius
    low = 0.0
    high = 0.12 # Upper bound estimate
    best_r = 0.0
    best_centers = None
    
    # 25 iterations of binary search
    for _ in range(25):
        mid = (low + high) / 2.0
        valid, centers = try_pack_equal(n, mid, num_starts=5)
        
        if valid:
            low = mid
            best_r = mid
            best_centers = centers
        else:
            high = mid
            
    if best_centers is None:
        # Fallback: generate a simple valid packing
        # e.g. grid
        centers = np.zeros((n, 2))
        radii = np.full(n, 0.05)
        # Just place them in a grid pattern roughly
        row = int(np.sqrt(n))
        for i in range(n):
            r_idx = i // row
            c_idx = i % row
            x = 0.1 + c_idx * 0.15
            y = 0.1 + r_idx * 0.15
            centers[i] = [x, y]
        best_centers = centers
        best_radii = radii
        return centers, radii, np.sum(radii)
        
    # Optimize radii for the found centers using LP
    final_radii = optimize_radii_lp(best_centers)
    
    return best_centers, final_radii, np.sum(final_radii)
