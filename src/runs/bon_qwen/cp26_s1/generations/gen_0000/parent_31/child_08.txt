import numpy as np
from scipy.optimize import minimize

def objective_func(vars):
    """Objective: maximize sum of radii => minimize negative sum."""
    return -np.sum(vars[2::3])

def all_constraints(vars):
    """Compute all boundary and non-overlap constraints in a vectorized manner."""
    n = 26
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    # Boundary constraints: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0, r >= 0
    c1 = x - r
    c2 = 1.0 - x - r
    c3 = y - r
    c4 = 1.0 - y - r
    c5 = r
    
    # Overlap constraints: d^2 - (ri + rj)^2 >= 0 for all i < j
    X, Y, R = x[:, None], y[:, None], r[:, None]
    d_sq = (X - X.T)**2 + (Y - Y.T)**2
    r_sum_sq = (R + R.T)**2
    
    idx = np.triu_indices(n, k=1)
    c6 = d_sq[idx] - r_sum_sq[idx]
    
    return np.concatenate([c1, c2, c3, c4, c5, c6])

def get_initial_vars():
    """Generate initial variables from a hexagonal lattice with random perturbation."""
    n = 26
    pts = []
    cols, rows = 6, 5
    for r_idx in range(rows):
        for c_idx in range(cols):
            if len(pts) >= n:
                break
            x = c_idx * np.sqrt(3) / 2.0 + (r_idx % 2) * np.sqrt(3) / 4.0
            y = r_idx * 1.5
            pts.append([x, y])
        if len(pts) >= n:
            break
            
    pts = np.array(pts[:n])
    # Normalize to [0.1, 0.9]
    pts -= pts.min(axis=0)
    span = pts.max(axis=0) - pts.min(axis=0)
    pts = 0.1 + (pts / span) * 0.8
    
    vars = np.zeros(3 * n)
    vars[0::3] = pts[:, 0]
    vars[1::3] = pts[:, 1]
    vars[2::3] = 0.05  # Initial small radii
    
    # Add noise to break symmetry and help escape bad local minima
    vars[:2*n] += np.random.normal(0, 0.02, (2*n))
    vars[:2*n] = np.clip(vars[:2*n], 0.05, 0.95)
    return vars

def run_packing():
    n = 26
    np.random.seed(42)  # For reproducibility
    
    best_vars = None
    best_sum_r = -np.inf
    
    cons = {'type': 'ineq', 'fun': all_constraints}
    
    # Run multiple optimization attempts with restarts
    for attempt in range(20):
        x0 = get_initial_vars()
        res = minimize(objective_func, x0, method='SLSQP', 
                       constraints=cons, options={'maxiter': 5000, 'ftol': 1e-12, 'disp': False})
        
        curr_sum_r = -res.fun
        con_vals = all_constraints(res.x)
        min_con = np.min(con_vals)
        
        # Accept if sufficiently feasible and better than current best
        if min_con > -1e-5:
            if curr_sum_r > best_sum_r:
                best_sum_r = curr_sum_r
                best_vars = res.x.copy()
                
        # Perturb best solution for next attempt to explore nearby regions
        if best_vars is not None:
            x0 = best_vars + np.random.normal(0, 0.005, best_vars.shape)
            x0[:2*n] = np.clip(x0[:2*n], 0.01, 0.99)
            x0[2*n:] = np.maximum(x0[2*n:], 0.0)

    centers = best_vars[:2*n].reshape(n, 2)
    radii = best_vars[2*n:]
    
    # Final safety adjustment: if any constraint is slightly violated, shrink radii
    min_c = np.min(all_constraints(best_vars))
    if min_c < 0:
        shrink_factor = 1.0 - (abs(min_c) / 0.05) - 1e-7
        radii = radii * max(shrink_factor, 0.0)
        
    return centers, radii, np.sum(radii)
