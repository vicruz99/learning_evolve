import numpy as np
from scipy.optimize import minimize
import math

def _compute_raw(params, n, penalty_weight):
    """
    Computes the objective value and gradient for the packing problem.
    """
    centers = params[:2*n].reshape(n, 2)
    radii = params[2*n:]
    
    obj_val = -np.sum(radii)
    penalty = 0.0
    grad = np.zeros_like(params)
    grad[2*n:] = -1.0
    
    pw2 = 2.0 * penalty_weight
    
    # Boundary constraints gradients
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        idx = 2*i
        idx_r = 2*n + i
        
        # x >= r  => violation = max(0, r - x)
        vx = r - x
        if vx > 0:
            penalty += vx**2
            grad[idx] -= pw2 * vx
            grad[idx_r] += pw2 * vx
            
        # x <= 1 - r => violation = max(0, x + r - 1)
        vx = x + r - 1.0
        if vx > 0:
            penalty += vx**2
            grad[idx] += pw2 * vx
            grad[idx_r] += pw2 * vx
            
        # y >= r
        vy = r - y
        if vy > 0:
            penalty += vy**2
            grad[idx+1] -= pw2 * vy
            grad[idx_r] += pw2 * vy
            
        # y <= 1 - r
        vy = y + r - 1.0
        if vy > 0:
            penalty += vy**2
            grad[idx+1] += pw2 * vy
            grad[idx_r] += pw2 * vy
            
    # Overlap constraints gradients
    for i in range(n):
        xi, yi = centers[i]
        ri = radii[i]
        idx_i = 2*i
        idx_ri = 2*n + i
        
        for j in range(i + 1, n):
            xj, yj = centers[j]
            rj = radii[j]
            idx_j = 2*j
            idx_rj = 2*n + j
            
            dx = xi - xj
            dy = yi - yj
            dist_sq = dx*dx + dy*dy
            dist = math.sqrt(dist_sq) if dist_sq > 0 else 1e-12
            
            overlap = ri + rj - dist
            if overlap > 0:
                penalty += overlap**2
                if dist > 1e-12:
                    d_dist_dx = dx / dist
                    d_dist_dy = dy / dist
                else:
                    d_dist_dx = 0.0
                    d_dist_dy = 0.0
                    
                grad[idx_i] -= pw2 * overlap * d_dist_dx
                grad[idx_i+1] -= pw2 * overlap * d_dist_dy
                grad[idx_j] += pw2 * overlap * d_dist_dx
                grad[idx_j+1] += pw2 * overlap * d_dist_dy
                
                grad[idx_ri] += pw2 * overlap
                grad[idx_rj] += pw2 * overlap
                
    return obj_val + penalty_weight * penalty, grad

def compute_obj_val(params, n, w):
    val, _ = _compute_raw(params, n, w)
    return val

def compute_obj_grad(params, n, w):
    _, grad = _compute_raw(params, n, w)
    return grad

def run_single_optimization(c0, r0, n):
    params = np.concatenate([c0.flatten(), r0])
    bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n
    
    # Progressive penalty annealing to escape local minima and enforce constraints
    weights = [5.0, 50.0, 500.0, 2000.0]
    current_params = params.copy()
    
    for w in weights:
        res = minimize(
            fun=compute_obj_val,
            x0=current_params,
            method='L-BFGS-B',
            jac=compute_obj_grad,
            args=(n, w),
            bounds=bounds,
            options={'maxiter': 1000, 'ftol': 1e-15, 'gtol': 1e-12, 'disp': False}
        )
        current_params = res.x
        
    centers = current_params[:2*n].reshape(n, 2)
    radii = current_params[2*n:]
    
    # Final projection to strictly satisfy constraints
    radii = np.maximum(radii, 0.0)
    for i in range(n):
        centers[i, 0] = np.clip(centers[i, 0], radii[i], 1.0 - radii[i])
        centers[i, 1] = np.clip(centers[i, 1], radii[i], 1.0 - radii[i])
        
    return centers, radii, np.sum(radii)

def run_packing():
    n = 26
    best_sum = -1.0
    best_c = None
    best_r = None
    
    # Generate diverse initial configurations
    configs = []
    
    # 1. Hexagonal lattice pattern
    c1 = np.zeros((n, 2))
    idx = 0
    row = 0
    y = 0.07
    while idx < n:
        x = 0.07 if row % 2 == 1 else 0.14
        while x < 0.93 and idx < n:
            c1[idx] = [x, y]
            idx += 1
            x += 0.17
        y += 0.14
        row += 1
    configs.append((c1, np.full(n, 0.06)))
    
    # 2. Uniform grid
    c2 = np.zeros((n, 2))
    idx = 0
    for y in np.linspace(0.1, 0.9, 6):
        for x in np.linspace(0.1, 0.9, 5):
            if idx < n:
                c2[idx] = [x, y]
                idx += 1
    configs.append((c2, np.full(n, 0.05)))
    
    # 3. Randomized layout 1
    np.random.seed(42)
    c3 = np.random.rand(n, 2) * 0.6 + 0.2
    configs.append((c3, np.full(n, 0.05)))
    
    # 4. Randomized layout 2
    np.random.seed(123)
    c4 = np.random.rand(n, 2) * 0.5 + 0.25
    configs.append((c4, np.full(n, 0.06)))
    
    # Optimize from each starting point
    for c0, r0 in configs:
        try:
            c_opt, r_opt, s_opt = run_single_optimization(c0, r0, n)
            # Basic validity check
            valid = np.all(r_opt >= 0)
            if valid and s_opt > best_sum:
                best_sum = s_opt
                best_c = c_opt.copy()
                best_r = r_opt.copy()
        except Exception:
            continue
            
    return best_c, best_r, best_sum
