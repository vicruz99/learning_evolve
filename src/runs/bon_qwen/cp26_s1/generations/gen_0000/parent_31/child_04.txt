import numpy as np
from scipy.optimize import minimize

def compute_loss(params, n, penalty_weight):
    """
    Compute the objective function: -sum(radii) + penalty terms for constraint violations.
    """
    centers = params[:2*n].reshape(n, 2)
    radii = params[2*n:]
    
    loss = -np.sum(radii)
    
    # Boundary penalties
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if r < 0.0:
            r = 0.0
            
        # Left wall
        val = r - x
        if val > 0.0:
            loss += penalty_weight * val**2
        # Right wall
        val = r - (1.0 - x)
        if val > 0.0:
            loss += penalty_weight * val**2
        # Bottom wall
        val = r - y
        if val > 0.0:
            loss += penalty_weight * val**2
        # Top wall
        val = r - (1.0 - y)
        if val > 0.0:
            loss += penalty_weight * val**2
            
    # Pairwise overlap penalties
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.hypot(dx, dy)
            overlap = radii[i] + radii[j] - dist
            if overlap > 0.0:
                loss += penalty_weight * overlap**2
                
    return loss

def run_packing() -> tuple:
    n = 26
    np.random.seed(42)

            centers[idx, 1] = y
            idx += 1
            
    # Add slight perturbation to break symmetry
    centers += np.random.uniform(-0.01, 0.01, centers.shape)
    centers = np.clip(centers, 0.02, 0.98)
    
    params = np.concatenate([centers.flatten(), radii])
    
    # Loose bounds to let penalty function handle exact limits
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n

    current_params = params.copy()
    penalty_weights = [500, 2000, 8000, 30000]
    
    for pw in penalty_weights:
        res = minimize(
            compute_loss, 
            current_params, 
            args=(n, pw), 
            method='L-BFGS-B', 
            bounds=bounds, 
            options={'maxiter': 5000, 'ftol': 1e-9}
        )
        current_params = res.x.copy()

    centers = current_params[:2*n].reshape(n, 2)
    radii = current_params[2*n:]
    radii = np.maximum(radii, 0.0)

    max_viol = 0.0
    for i in range(n):
        x, y = centers[i]
        r = radii[i]

                max_viol = max(max_viol, -dist_wall)

        for j in range(i + 1, n):
