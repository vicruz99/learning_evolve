import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def generate_initial_guess(n):
    """Generate a hexagonal lattice configuration as initial guess."""
    dx = 0.21
    dy = dx * np.sqrt(3) / 2.0
    pts = []
    y = 0.0
    row = 0
    while len(pts) < n:
        x = dx / 2.0 if row % 2 == 1 else 0.0
        while x < 1.0 and len(pts) < n:
            pts.append([x, y])
            x += dx
        y += dy
        row += 1
        
    pts = np.array(pts[:n])
    # Normalize and center within [0,1] with a safety margin
    mins = pts.min(axis=0)
    maxs = pts.max(axis=0)
    span = maxs - mins
    target_span = 0.85
    scale = target_span / span.max()
    pts = (pts - mins) * scale + (1.0 - target_span) / 2.0
    
    radii = np.full(n, 0.06)
    return pts, radii

def objective(vars):
    """Objective: minimize negative sum of radii (equivalent to maximizing sum)."""
    return -np.sum(vars[2::3])

def constraints(vars, n):
    """Vectorized inequality constraints >= 0."""
    pts = vars.reshape(n, 3)
    xx, yy, rr = pts[:, 0], pts[:, 1], pts[:, 2]
    
    # Boundary and non-negativity constraints
    c1 = xx - rr
    c2 = 1.0 - xx - rr
    c3 = yy - rr
    c4 = 1.0 - yy - rr
    c5 = rr
    
    # Overlap constraints: dist^2 - (r1 + r2)^2 >= 0
    dx = xx[:, None] - xx[None, :]
    dy = yy[:, None] - yy[None, :]
    dr = rr[:, None] + rr[None, :]
    dist2 = dx**2 + dy**2
    rad2 = dr**2
    
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    c6 = (dist2 - rad2)[mask]
    
    return np.concatenate([c1, c2, c3, c4, c5, c6])

def constraint_fun(vars):
    """Wrapper for constraint function to avoid closures/lambdas."""
    return constraints(vars, N_CIRCLES)

def run_packing():
    """Run optimization to pack N_CIRCLES in unit square maximizing sum of radii."""
    n = N_CIRCLES
    pts0, r0 = generate_initial_guess(n)
    
    # Flatten to optimization variable array: [x1, y1, r1, x2, y2, r2, ...]
    x0 = np.zeros(3 * n)
    x0[0::3] = pts0[:, 0]
    x0[1::3] = pts0[:, 1]
    x0[2::3] = r0
    
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * n
    
    res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                   constraints={'type': 'ineq', 'fun': constraint_fun},
                   options={'maxiter': 2000, 'ftol': 1e-9, 'disp': False})
    
    # Extract results
    centers = res.x[:2*n].reshape(n, 2)
    radii = res.x[2*n:]
    
    # Post-process to guarantee strict feasibility
    radii = np.maximum(radii, 0.0)
    centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
    centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)
    
    total_radius = np.sum(radii)
    return centers, radii, total_radius
