import numpy as np
from scipy.optimize import linprog, minimize

def solve_radii(centers):
    """
    Solves for the optimal radii given a set of centers using Linear Programming.
    Maximizes sum(r_i) subject to non-overlap and boundary constraints.
    """
    n = len(centers)
    
    # 1. Setup LP Coefficients
    c = -np.ones(n)  # We minimize negative sum to maximize sum of radii
    
    # 2. Setup Constraints (A_ub @ r <= b_ub)
    n_constraints = 0
    for i in range(n):
        for j in range(i + 1, n):
            n_constraints += 1
    for i in range(n):
        n_constraints += 4 # 4 boundary constraints per circle (x, 1-x, y, 1-y)
        
    A_ub = np.zeros((n_constraints, n))
    b_ub = np.zeros(n_constraints)
    
    row = 0
    
    # Pairwise non-overlap constraints: r_i + r_j <= dist(c_i, c_j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            A_ub[row, i] = 1
            A_ub[row, j] = 1
            b_ub[row] = dist
            row += 1
            
    # Boundary constraints: r_i <= x, r_i <= 1-x, r_i <= y, r_i <= 1-y
    for i in range(n):
        x, y = centers[i]
        A_ub[row, i] = 1; b_ub[row] = x; row += 1
        A_ub[row, i] = 1; b_ub[row] = 1 - x; row += 1
        A_ub[row, i] = 1; b_ub[row] = y; row += 1
        A_ub[row, i] = 1; b_ub[row] = 1 - y; row += 1
        
    # 3. Bounds for radii (r_i >= 0)
    bounds = [(0, None) for _ in range(n)]
    
    # 4. Solve LP
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if not res.success:
        return np.zeros(n), 0.0
        
    radii = np.maximum(res.x, 0) # Ensure non-negative due to numerical noise
    return radii, np.sum(radii)

def objective_func(coords, n=26):
    """Objective function for optimization: negative sum of radii."""
    centers = coords.reshape((n, 2))
    _, sum_r = solve_radii(centers)
    return -sum_r

def run_packing():
    n = 26
    best_sum = 0

                if x <= 0.95: coords.append([x, y])
        coords.append([0.5, 0.5])
        return np.array(coords[:n])

            centers = get_hex_grid()

        # Local optimization of centers
        res = minimize(
            objective_func, 
            centers.flatten(), 
            args=(n,), 
            method='Nelder-Mead', 
            options={'maxiter': 2000, 'xatol': 1e-5, 'fatol': 1e-5}
        )
        
        current_sum = -res.fun
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = res.x.reshape((n, 2))
