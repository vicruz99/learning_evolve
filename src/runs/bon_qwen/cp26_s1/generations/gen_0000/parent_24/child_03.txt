import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Pack 26 circles in a unit square [0,1]x[0,1] to maximize the sum of radii.
    Uses a penalty-based optimization approach starting from a hexagonal lattice.
    """
    n_circles = 26
    
    # 1. Initialization: Hexagonal lattice
    # Start with a reasonable radius guess and fill rows
    r_init = 0.10
    centers = []
    
    y = r_init
    row = 0
    while True:
        x_start = r_init if row % 2 == 0 else r_init + r_init # Shift for hexagonal
        # Ensure x_start is valid
        if x_start > 1.0 - r_init:
            break
            
        while x_start <= 1.0 - r_init:
            centers.append([x_start, y])
            if len(centers) >= n_circles:
                break
            x_start += 2 * r_init
        else:
            y += np.sqrt(3) * r_init
            row += 1
            if y + r_init > 1.0:
                break
            continue
        break
        
    # Convert to numpy array and ensure we have exactly n_circles
    centers = np.array(centers)
    if len(centers) < n_circles:
        # Fallback to random if lattice failed (unlikely)
        centers = np.random.rand(n_circles, 2) * 0.5 + 0.25
    else:
        centers = centers[:n_circles]
        
    # Initial radii
    radii = np.full(n_circles, r_init)

    def objective(x):
        # x contains [x0, y0, r0, x1, y1, r1, ...]
        sum_radii = 0
        penalty = 0
        
        centers = x[:n_circles*2].reshape(n_circles, 2)
        radii = x[n_circles*2:]
        
        # Add radii to sum (we want to maximize, so minimize negative)
        sum_radii = -np.sum(radii)
        
        # Boundary constraints penalty
        for i in range(n_circles):
            x, y = centers[i]
            r = radii[i]
            
            # Left/Right
            if x - r < 0: penalty += 1000 * (x - r)**2
            if x + r > 1: penalty += 1000 * (x + r - 1)**2
            # Top/Bottom
            if y - r < 0: penalty += 1000 * (y - r)**2
            if y + r > 1: penalty += 1000 * (y + r - 1)**2
            
            # Radius positivity
            if r < 0: penalty += 1000 * r**2
            
        # Overlap constraints penalty
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                r_sum = radii[i] + radii[j]
                if dist < r_sum:
                    penalty += 1000 * (r_sum - dist)**2
                    
        return sum_radii + penalty

    # Initial guess vector
    x0 = np.concatenate([centers.flatten(), radii])
    
    # Bounds: x, y in [0, 1], r in [0, 1]
    bounds = []
    for i in range(n_circles):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 1.0)) # r

    # Optimization
    result = minimize(objective, x0, method='L-BFGS-B', bounds=bounds, 
                      options={'maxiter': 5000, 'ftol': 1e-12, 'gtol': 1e-8})
    
    if result.success or result.fun < 100: # Check if a valid-ish solution was found
        final_centers = result.x[:n_circles*2].reshape(n_circles, 2)
        final_radii = result.x[n_circles*2:]
    else:
        # Fallback to initial if optimization fails
        final_centers = centers
        final_radii = radii
        
    # Clip radii to be non-negative and ensure centers are valid
    final_radii = np.maximum(final_radii, 0)
    final_centers = np.clip(final_centers, 0, 1)
    
    total_sum = np.sum(final_radii)
    
    return final_centers, final_radii, total_sum

if __name__ == "__main__":
    centers, radii, total_sum = run_packing()
    print(f"Sum of radii: {total_sum}")
