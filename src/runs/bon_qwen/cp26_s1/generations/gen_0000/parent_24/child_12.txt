import numpy as np

def solve_radii(centers):
    """
    Given centers, solve the LP to find radii that maximize sum of radii.
    Maximize sum(r_i)
    Subject to:
      r_i + r_j <= dist(c_i, c_j)
      r_i <= x_i, 1-x_i, y_i, 1-y_i
      r_i >= 0
    """
    n = len(centers)
    num_vars = n
    
    # Objective: minimize -sum(r_i)
    c_obj = -np.ones(num_vars)
    
    # Constraints matrix A_ub @ r <= b_ub
    # We will build lists and convert to array
    
    A_ub = []
    b_ub = []
    
    # Pairwise constraints: r_i + r_j <= dist
    # dist >= r_i + r_j  =>  r_i + r_j <= dist
    # This is an upper bound constraint.
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            row = np.zeros(num_vars)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dist)
            
    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        dist_left = x
        dist_right = 1.0 - x
        dist_down = y
        dist_up = 1.0 - y
        
        # r_i <= dist_left
        row = np.zeros(num_vars)
        row[i] = 1.0
        A_ub.append(row)
        b_ub.append(dist_left)
        
        # r_i <= dist_right
        row = np.zeros(num_vars)
        row[i] = 1.0
        A_ub.append(row)
        b_ub.append(dist_right)
        
        # r_i <= dist_down
        row = np.zeros(num_vars)
        row[i] = 1.0
        A_ub.append(row)
        b_ub.append(dist_down)
        
        # r_i <= dist_up
        row = np.zeros(num_vars)
        row[i] = 1.0
        A_ub.append(row)
        b_ub.append(dist_up)

    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    # Bounds for r_i: [0, infinity)
    bounds = [(0, None) for _ in range(num_vars)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return -res.fun, res.x # sum_radii, radii
        else:
            # Fallback: 0 radii
            return 0.0, np.zeros(n)
    except Exception:
        return 0.0, np.zeros(n)
