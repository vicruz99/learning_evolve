from scipy.optimize import minimize


def compute_forces(centers, radii, n):
    """Compute repulsive forces between circles and with boundaries."""
    forces = np.zeros_like(centers)

    for i in range(n):
        for j in range(i + 1, n):
            diff = centers[i] - centers[j]
            dist_sq = np.sum(diff * diff)
            dist = np.sqrt(dist_sq)
            min_dist = radii[i] + radii[j] + 1e-10

            if dist < min_dist and dist > 1e-12:
                overlap = min_dist - dist
                force_mag = overlap / dist
                forces[i, 0] += diff[0] * force_mag
                forces[i, 1] += diff[1] * force_mag
                forces[j, 0] -= diff[0] * force_mag
                forces[j, 1] -= diff[1] * force_mag

        for d in range(2):
            if centers[i, d] < radii[i] + 1e-10:
                forces[i, d] += (radii[i] + 1e-10 - centers[i, d]) * 50
            if centers[i, d] > 1 - radii[i] - 1e-10:
                forces[i, d] -= (centers[i, d] - (1 - radii[i] - 1e-10)) * 50

    return forces


def grow_radii(centers, radii, n, growth_rate):
    """Grow radii as much as possible given current positions."""
    for i in range(n):
        max_r = min(centers[i, 0], 1 - centers[i, 0],
                   centers[i, 1], 1 - centers[i, 1])
        for j in range(n):
            if i != j:
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                max_r = min(max_r, dist - radii[j] - 1e-10)
        radii[i] = min(radii[i] + growth_rate, max(0, max_r - 1e-11))


def force_directed_optimize(centers, radii, n, num_iterations, growth_rate):

    for iteration in range(num_iterations):
        forces = compute_forces(centers, radii, n)

    return centers, radii
