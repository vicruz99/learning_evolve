def compute_overlap_energy(centers, radii):
    """
    Computes a penalty energy for overlaps and boundary violations.
    Returns a scalar value. 0 means valid.

    energy = 0.0
    
    # Boundary violations

        x, y = centers[i]
        r = radii[i]
        # Distance to walls
        d_left = x - r
        d_right = 1 - x - r
        d_down = y - r
        d_up = 1 - y - r
        
        for d in [d_left, d_right, d_down, d_up]:
            if d < 0:
                energy += d**2
                
    # Circle overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            sum_r = radii[i] + radii[j]
            if dist < sum_r:
                overlap = sum_r - dist
                energy += overlap**2
                
    return energy
