import numpy as np
from scipy.optimize import linprog, differential_evolution

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26

    def get_max_sum_radii(centers_flat: np.ndarray) -> tuple[float, np.ndarray]:
        """
        For a given set of centers, solve the LP to find maximum radii.
        Returns (sum_radii, radii).
        """
        centers = centers_flat.reshape(n, 2)
        
        # Maximize sum(r_i) is equivalent to minimizing -sum(r_i)
        c_obj = -np.ones(n)
        
        A_ub = []
        b_ub = []
        
        # Boundary constraints: r_i <= dist to boundaries
        # x, y, 1-x, 1-y
        for i in range(n):
            x, y = centers[i]
            row = np.zeros(n)
            row[i] = 1.0
            
            # r_i <= x
            A_ub.append(row)
            b_ub.append(x)
            # r_i <= 1 - x
            A_ub.append(row)
            b_ub.append(1.0 - x)
            # r_i <= y
            A_ub.append(row)
            b_ub.append(y)
            # r_i <= 1 - y
            A_ub.append(row)
            b_ub.append(1.0 - y)
            
        # Pairwise constraints: r_i + r_j <= distance(i, j)
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                
                row = np.zeros(n)
                row[i] = 1.0
                row[j] = 1.0
                A_ub.append(row)
                b_ub.append(dist)
        
        A_ub = np.array(A_ub)
        b_ub = np.array(b_ub)
        
        bounds = [(0, 1.0)] * n
        
        # Use 'highs' solver for speed
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
            if res.success:
                return -res.fun, res.x
            else:
                # Fallback or error handling
                return 0.0, np.zeros(n)
        except Exception:
            return 0.0, np.zeros(n)

    def objective(centers_flat: np.ndarray) -> float:
        s, _ = get_max_sum_radii(centers_flat)
        # DE minimizes, so we return negative sum
        return -s

    bounds_centers = [(0, 1)] * (2 * n)
    
    # Generate diverse initial populations
    # 1. Random
    pop1 = np.random.uniform(0, 1, (10, 2 * n))
    
    # 2. Grid-like structure (5x5 + 1) perturbed
    grid_pop = []
    base_coords = [0.1, 0.3, 0.5, 0.7, 0.9]
    points = []
    for x in base_coords:
        for y in base_coords:
            points.append([x, y])
    # Add 26th point in a gap
    points.append([0.2, 0.2])
    pts_arr = np.array(points).flatten()
    
    for _ in range(5):
        noise = np.random.uniform(-0.05, 0.05, 2 * n)
        perturbed = np.clip(pts_arr + noise, 0, 1)
        grid_pop.append(perturbed)
    pop2 = np.array(grid_pop)
    
    # 3. Hexagonal-like structure
    hex_pop = []
    pts_hex = []
    y_row = 0.1
    while y_row < 0.9:
        x_col = 0.1
        row_points = []
        while x_col < 0.9:
            pts_hex.append([x_col, y_row])
            x_col += 0.2
        y_row += 0.1732 # sqrt(3)/2 * 0.2
    # Fill up to 26
    while len(pts_hex) < 26:
        pts_hex.append([0.5, 0.5])
    pts_hex = np.array(pts_hex[:26]).flatten()
    
    for _ in range(5):
        noise = np.random.uniform(-0.05, 0.05, 2 * n)
        perturbed = np.clip(pts_hex + noise, 0, 1)
        hex_pop.append(perturbed)
    pop3 = np.array(hex_pop)
    
    # Combine populations
    all_pops = np.vstack([pop1, pop2, pop3])
    
    # Run Differential Evolution
    result = differential_evolution(
        objective, 
        bounds_centers, 
        seed=42, 
        maxiter=300, 
        popsize=all_pops, 
        tol=1e-8, 
        mutation=(0.5, 1.5),
        recombination=0.7,
        polish=False # Manual refinement if needed, but DE with polish can be slow
    )
    
    best_centers = result.x.reshape(n, 2)
    
    # Get final radii for the best centers
    _, radii = get_max_sum_radii(best_centers)
    sum_radii = np.sum(radii)
    
    # Optional: Run a quick local refinement (Nelder-Mead) if time permits, 
    # but for 52D it might be risky. DE with polish is usually sufficient.
    # Given the complexity, we rely on the robust DE result.
    
    return best_centers, radii, sum_radii

    import numpy as np
    def validate_packing(centers, radii):
        n = centers.shape[0]
        if np.isnan(centers).any() or np.isnan(radii).any(): return False
        for i in range(n):
            if radii[i] < 0: return False
            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                return False
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist < radii[i] + radii[j] - 1e-12:
                    return False
        return True
    
    print(f"Valid: {validate_packing(centers, radii)}")
