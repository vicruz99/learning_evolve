import numpy as np
import scipy.optimize
import random

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Pack 26 circles in a unit square [0,1]x[0,1] to maximize the sum of radii.
    """
    n_circles = 26
    best_sum_radii = 0.0
    best_centers = None
    best_radii = None

    # Helper to validate and compute sum
    def get_sum(centers, radii):
        return np.sum(radii)

    # Simulation parameters
    num_iterations = 2000
    expansion_step = 0.0005
    repulsion_strength = 0.5
    boundary_push_strength = 2.0
    
    # Optimization parameters
    opt_maxiter = 1000
    lambda_penalty = 1000.0

    def compute_forces_and_update(centers, radii, step):
        """
        Apply expansion and repulsion forces.
        """
        forces = np.zeros_like(centers)
        
        # 1. Inter-circle repulsion
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                vec = centers[i] - centers[j]
                dist = np.linalg.norm(vec)
                min_dist = radii[i] + radii[j]
                
                if dist < min_dist and dist > 1e-9:
                    overlap = min_dist - dist
                    # Repulsive force proportional to overlap
                    # Normalize vector
                    dir_vec = vec / dist
                    # Push apart
                    push = overlap * repulsion_strength
                    forces[i] += dir_vec * push
                    forces[j] -= dir_vec * push
                elif dist < 1e-9:
                    # Handle exact overlap by random push
                    random_dir = np.random.randn(2)
                    random_dir /= np.linalg.norm(random_dir)
                    forces[i] += random_dir * 0.1
                    forces[j] -= random_dir * 0.1

        # 2. Boundary repulsion
        for i in range(n_circles):
            r = radii[i]
            x, y = centers[i]
            
            # Left wall
            if x - r < 0:
                forces[i, 0] += boundary_push_strength * (r - x)
            # Right wall
            if x + r > 1:
                forces[i, 0] -= boundary_push_strength * (x + r - 1)
            # Bottom wall
            if y - r < 0:
                forces[i, 1] += boundary_push_strength * (r - y)
            # Top wall
            if y + r > 1:
                forces[i, 1] -= boundary_push_strength * (y + r - 1)

        # Update centers
        centers += forces * step
        
        # Clamp centers to valid range based on current radii to prevent drifting out
        for i in range(n_circles):
            r = radii[i]
            centers[i, 0] = np.clip(centers[i, 0], r, 1 - r)
            centers[i, 1] = np.clip(centers[i, 1], r, 1 - r)
            
        return centers

    def penalty_function(params, n):
        """
        Penalty function for scipy optimization.
        Minimizes -sum(radii) + lambda * violations.
        """
        centers = params[:2*n].reshape(n, 2)
        radii = params[2*n:]
        
        # Ensure non-negative radii
        radii = np.maximum(radii, 0)
        
        penalty = 0.0
        
        # Boundary penalties
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            # Left
            if x - r < 0: penalty += lambda_penalty * (x - r)**2
            # Right
            if x + r > 1: penalty += lambda_penalty * (x + r - 1)**2
            # Bottom
            if y - r < 0: penalty += lambda_penalty * (y - r)**2
            # Top
            if y + r > 1: penalty += lambda_penalty * (y + r - 1)**2
            
        # Overlap penalties
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.linalg.norm(centers[i] - centers[j])
                r_sum = radii[i] + radii[j]
                if dist < r_sum:
                    penalty += lambda_penalty * (dist - r_sum)**2
                    
        return -np.sum(radii) + penalty

    def run_optimization(initial_centers, initial_radii):
        """
        Run local optimization starting from given state.
        """
        n = len(initial_radii)
        # Flatten parameters: [x1, y1, ..., xn, yn, r1, ..., rn]
        x0 = np.concatenate([initial_centers.flatten(), initial_radii])
        
        # Bounds: x, y in [0, 1], r in [0, 0.5]
        bounds = []
        for _ in range(n):
            bounds.extend([(0, 1), (0, 1)])
        for _ in range(n):
            bounds.append((0, 0.5))
            
        try:
            res = scipy.optimize.minimize(
                penalty_function,
                x0,
                args=(n,),
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': opt_maxiter, 'ftol': 1e-9}
            )
            
            opt_centers = res.x[:2*n].reshape(n, 2)
            opt_radii = res.x[2*n:]
            
            # Clean up radii (ensure non-negative)
            opt_radii = np.maximum(opt_radii, 0)
            
            # Final clamping to ensure strict validity
            for i in range(n):
                r = opt_radii[i]
                opt_centers[i, 0] = np.clip(opt_centers[i, 0], r, 1 - r)
                opt_centers[i, 1] = np.clip(opt_centers[i, 1], r, 1 - r)
                
            return opt_centers, opt_radii
            
        except Exception:
            return initial_centers, initial_radii

    def try_packing():
        nonlocal best_sum_radii, best_centers, best_radii
        
        # Initialize centers in a hexagonal-like pattern
        # 5 rows roughly
        centers = np.zeros((n_circles, 2))
        radii = np.full(n_circles, 0.01) # Small initial radius
        
        idx = 0
        # Approximate hexagonal grid
        row = 0
        while idx < n_circles:
            # Determine number of circles in this row
            # Alternating 5, 6, 5, 6... or similar to fill 26
            # Let's just place them sequentially with offset
            cols = 5
            if row % 2 == 1:
                cols = 6 # Shifted row might fit more? 
                # Actually width is 1. 6 circles of diam 0.2 needs 1.0. 
                # With radius 0.01, it's fine.
            
            # Calculate positions
            # y spacing
            y_spacing = 1.1 * 0.2 # slightly compressed
            y = 0.1 + row * y_spacing * np.sqrt(3)/2
            
            # x spacing
            x_start = 0.1
            if row % 2 == 1:
                x_start += 0.1 # Shift
            
            # Fit circles in row
            current_row_count = 0
            c_idx = idx
            while c_idx < n_circles and current_row_count < cols:
                x = x_start + current_row_count * 0.2
                if x < 1.0 and y < 1.0:
                    centers[c_idx] = [x, y]
                    c_idx += 1
                    current_row_count += 1
                else:
                    break
            idx = c_idx
            row += 1
            
        # If not all filled, fill remaining randomly (should not happen with logic above if careful)
        # But simple grid fill is safer
        if idx < n_circles:
            # Fallback to random for remaining
            for k in range(idx, n_circles):
                centers[k] = [np.random.random(), np.random.random()]

        # Simulation Phase
        for t in range(num_iterations):
            # Expand radii
            radii += expansion_step
            
            # Resolve collisions and update positions
            # Run multiple sub-steps for stability
            for _ in range(10):
                centers = compute_forces_and_update(centers, radii, 0.01)
                
            # Add small random perturbation to escape local minima
            if t % 100 == 0:
                centers += np.random.normal(0, 0.005, centers.shape)
                # Re-clamp
                for i in range(n_circles):
                    r = radii[i]
                    centers[i, 0] = np.clip(centers[i, 0], r, 1 - r)
                    centers[i, 1] = np.clip(centers[i, 1], r, 1 - r)

        # Optimization Phase
        centers, radii = run_optimization(centers, radii)
        
        current_sum = np.sum(radii)
        if current_sum > best_sum_radii:
            best_sum_radii = current_sum
            best_centers = centers.copy()
            best_radii = radii.copy()

    # Run multiple trials
    num_trials = 10
    for _ in range(num_trials):
        try_packing()

    return best_centers, best_radii, best_sum_radii
