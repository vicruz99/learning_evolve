def dist(p1, p2):
    return math.sqrt((p1[0] - p2[0])**2 + (p1[1] - p2[1])**2)

def check_validity(centers, radii):
    """Check if a configuration is valid without detailed error messages for speed."""
    n = len(centers)
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-9 or x + r > 1 + 1e-9 or y - r < -1e-9 or y + r > 1 + 1e-9:
            return False
        for j in range(i + 1, n):
            d = dist(centers[i], centers[j])
            if d < radii[i] + radii[j] - 1e-9:
                return False
    return True

def get_overlap_energy(centers, radii):

    n = len(centers)
    energy = 0.0
    for i in range(n):

            min_dist = radii[i] + radii[j]

                energy += penetration**2 * 1000
    return energy

def force_directed_relaxation(centers, radii, steps=2000):

def generate_hexagonal_packing(n):
    """Generate a hexagonal lattice packing."""
    centers = []
    row = 0
    count = 0
    y = 0.1
    while count < n:
        x_start = 0.1 if row % 2 == 0 else 0.2

        while x <= 0.9 and count < n:
            centers.append([x, y])
            x += 0.2
            count += 1
        y += 0.1732 # sqrt(3)/2 * 0.2
        row += 1
    return np.array(centers[:n])

def generate_grid_packing(n):
    """Generate a grid packing."""
    cols = math.ceil(math.sqrt(n))
    rows = math.ceil(n / cols)
    centers = []

            if len(centers) < n:
                x = 0.1 + j * 0.2
                y = 0.1 + i * 0.2
                centers.append([x, y])
    return np.array(centers[:n])
