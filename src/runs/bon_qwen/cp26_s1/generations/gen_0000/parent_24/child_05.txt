import numpy as np
from scipy.optimize import minimize


def hexagonal_initial(n, row_configs):
    """Generate a hexagonal packing initial configuration."""
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.085)
    idx = 0
    for row in range(len(row_configs)):
        num = row_configs[row]
        y = 0.085 + row * 0.085 * np.sqrt(3)
        for col in range(num):
            x = 0.085 + col * 2 * 0.085 + (0.085 if row % 2 == 1 else 0)
            centers[idx] = [x, y]
            idx += 1
    return centers, radii


def grid_initial(n, rows, cols):
    """Generate a grid-based initial configuration."""
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.08)
    idx = 0
    for r in range(rows):
        for c in range(cols):
            if idx >= n:
                break
            x = 0.5 / cols + c * (1 - 1.0 / cols)
            y = 0.5 / rows + r * (1 - 1.0 / rows)
            centers[idx] = [x, y]
            idx += 1
    return centers, radii


def random_initial(n, r0=0.05):
    """Generate a random initial configuration with small circles."""
    centers = np.random.uniform(r0, 1 - r0, (n, 2))
    radii = np.full(n, r0)
    return centers, radii


def force_directed_optimize(centers, radii, n, max_iter=5000, verbose=False):
    """Optimize packing using force-directed simulation."""
    centers = centers.copy()
    radii = radii.copy()
    
    lr = 0.005
    lr_decay = 0.9995
    radii_growth = 0.00005
    
    for iteration in range(max_iter):
        forces = np.zeros_like(centers)
        
        # Repulsion between overlapping circles
        for i in range(n):
            for j in range(i + 1, n):
                diff = centers[i] - centers[j]
                dist = np.sqrt(np.sum(diff ** 2))
                min_dist = radii[i] + radii[j]
                if dist < min_dist and dist > 1e-12:
                    repulsion = (min_dist - dist) / dist
                    forces[i] += diff * repulsion * 5
                    forces[j] -= diff * repulsion * 5
                elif dist < min_dist * 1.5 and dist > 1e-12:
                    # Gentle repulsion even for near-touching circles
                    repulsion = (min_dist * 1.5 - dist) / (dist * 1.5) * 0.1
                    forces[i] += diff * repulsion
                    forces[j] -= diff * repulsion
        
        # Boundary repulsion
        for i in range(n):
            if centers[i, 0] - radii[i] < 0:
                forces[i, 0] += (radii[i] - centers[i, 0]) * 10
            if centers[i, 0] + radii[i] > 1:
                forces[i, 0] -= (centers[i, 0] + radii[i] - 1) * 10
            if centers[i, 1] - radii[i] < 0:
                forces[i, 1] += (radii[i] - centers[i, 1]) * 10
            if centers[i, 1] + radii[i] > 1:
                forces[i, 1] -= (centers[i, 1] + radii[i] - 1) * 10
        
        # Update centers
        centers += forces * lr
        centers = np.clip(centers, 1e-8, 1 - 1e-8)
        
        # Grow radii
        radii += radii_growth
        
        # Shrink radii if violating constraints
        for i in range(n):
            radii[i] = min(radii[i], centers[i, 0])
            radii[i] = min(radii[i], 1 - centers[i, 0])
            radii[i] = min(radii[i], centers[i, 1])
            radii[i] = min(radii[i], 1 - centers[i, 1])
        
        # Check overlap and shrink if needed
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist < radii[i] + radii[j]:
                    excess = (radii[i] + radii[j] - dist) / 2
                    radii[i] = max(radii[i] - excess * 0.5, 1e-6)
                    radii[j] = max(radii[j] - excess * 0.5, 1e-6)
        
        lr *= lr_decay
        
        if iteration % 500 == 0 and verbose:
            print(f"Iter {iteration}: sum_radii = {np.sum(radii):.6f}")
    
    return centers, radii


def penalty_objective(x, n):
    """Objective function with penalties for constraint violations."""
    centers = x[:2 * n].reshape(n, 2)
    radii = x[2 * n:]
    
    obj = -np.sum(radii)
    
    # Boundary penalty
    for i in range(n):
        xi, yi = centers[i]
        ri = radii[i]
        obj += 1e6 * max(0, ri - xi) ** 2
        obj += 1e6 * max(0, xi + ri - 1) ** 2
        obj += 1e6 * max(0, ri - yi) ** 2
        obj += 1e6 * max(0, yi + ri - 1) ** 2
    
    # Overlap penalty
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            violation = radii[i] + radii[j] - dist
            obj += 1e6 * max(0, violation) ** 2
    
    return obj


def scipy_refine(centers, radii, n, max_iter=2000):
    """Refine packing using scipy optimizer with penalty method."""
    x0 = np.concatenate([centers.flatten(), radii])
    
    bounds = [(0, 1)] * (2 * n) + [(1e-8, 0.5)] * n
    
    result = minimize(
        penalty_objective,
        x0,
        args=(n,),
        method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-10}
    )
    
    x_opt = result.x
    centers_opt = x_opt[:2 * n].reshape(n, 2)
    radii_opt = x_opt[2 * n:]
    
    return centers_opt, radii_opt, result.fun


def ensure_feasibility(centers, radii, n, tolerance=1e-10):
    """Ensure the packing is strictly feasible by adjusting radii if needed."""
    radii = radii.copy()
    
    # Boundary constraints
    for i in range(n):
        ri = min(radii[i], centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])
        radii[i] = max(ri, 0)
    
    # Overlap constraints - iterative adjustment
    for _ in range(100):
        max_violation = 0
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                required = radii[i] + radii[j]
                if dist < required - tolerance:
                    excess = (required - dist) / 2 + tolerance
                    radii[i] = max(radii[i] - excess, 0)
                    radii[j] = max(radii[j] - excess, 0)
                    max_violation = max(max_violation, excess)
        if max_violation < tolerance:
            break
    
    return radii


def try_configuration(row_configs, n, label=""):
    """Try a specific row configuration and return best result."""
    centers, radii = hexagonal_initial(n, row_configs)
    
    # Phase 1: Force-directed optimization
    centers, radii = force_directed_optimize(centers, radii, n, max_iter=8000)
    
    # Phase 2: Scipy refinement
    centers, radii, obj_val = scipy_refine(centers, radii, n, max_iter=3000)
    
    # Phase 3: Ensure feasibility
    radii = ensure_feasibility(centers, radii, n)
    
    sum_radii = np.sum(radii)
    if label:
        print(f"Config {label}: sum_radii = {sum_radii:.6f}, penalty = {obj_val:.6f}")
    
    return centers, radii, sum_radii


def try_with_perturbation(centers, radii, n, num_trials=5):
    """Try optimization with random perturbations."""
    best_sum = -1
    best_centers = centers.copy()
    best_radii = radii.copy()
    
    for trial in range(num_trials):
        trial_centers = centers + np.random.normal(0, 0.01, centers.shape)
        trial_centers = np.clip(trial_centers, 0.01, 0.99)
        trial_radii = radii * (0.9 + np.random.random(n) * 0.2)
        
        trial_centers, trial_radii = force_directed_optimize(trial_centers, trial_radii, n, max_iter=5000)
        trial_centers, trial_radii, _ = scipy_refine(trial_centers, trial_radii, n, max_iter=2000)
        trial_radii = ensure_feasibility(trial_centers, trial_radii, n)
        
        trial_sum = np.sum(trial_radii)
        if trial_sum > best_sum:
            best_sum = trial_sum
            best_centers = trial_centers.copy()
            best_radii = trial_radii.copy()
    
    return best_centers, best_radii, best_sum
