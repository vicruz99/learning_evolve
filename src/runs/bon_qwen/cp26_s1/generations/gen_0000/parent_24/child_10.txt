To solve the problem of packing 26 circles in a unit square to maximize the sum of radii, I will use a nested optimization approach:

1.  **Problem Decomposition**: 
    *   **Outer Optimization**: Optimize the coordinates $(x_i, y_i)$ of the 26 circle centers.
    *   **Inner Optimization**: For any fixed set of centers, calculate the maximum possible radii. This is a Linear Programming (LP) problem because the constraints ($r_i + r_j \le d_{ij}$ and boundary limits) are linear in $r_i$.

2.  **Inner Optimization (LP)**:
    *   Given centers, the maximum radius for each circle is constrained by the distance to other circles and the boundaries of the square.
    *   We formulate an LP: Maximize $\sum r_i$ subject to $r_i + r_j \le \text{dist}(c_i, c_j)$ for all pairs, and $0 \le r_i \le \text{dist}(c_i, \text{boundary})$.
    *   This LP is solved efficiently using `scipy.optimize.linprog` with the 'highs' method.

3.  **Outer Optimization**:
    *   Since the objective function (sum of radii) depends non-smoothly on centers, I will use `scipy.optimize.differential_evolution`. This is a global optimization algorithm well-suited for this type of problem.
    *   **Initialization**: I will initialize the search with a structured grid (5x6) to ensure the circles are spread out, avoiding poor local minima where circles overlap heavily.
    *   **Search Space**: Centers are constrained to the unit square $[0, 1]^2$.

import numpy as np
import scipy.optimize as opt
import scipy.sparse as sp

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # Function to solve the Linear Programming problem for radii given fixed centers
    def get_max_radii(centers):
        # centers shape (n, 2)
        
        # Calculate distances between all pairs
        # dist_matrix[i, j] = distance between center i and j
        dist_matrix = np.linalg.norm(centers[:, np.newaxis, :] - centers[np.newaxis, :, :], axis=2)
        
        # Prepare LP constraints for r_i + r_j <= dist_ij
        # We only need i < j to avoid duplicates and self-constraints
        pairs_i, pairs_j = np.triu_indices(n, k=1)
        num_constraints = len(pairs_i)
        
        # A_ub * r <= b_ub
        # Each row has 1 at index i and 1 at index j
        row_indices = np.concatenate([pairs_i, pairs_j])
        col_indices = np.concatenate([np.arange(num_constraints), np.arange(num_constraints)])
        data = np.ones(2 * num_constraints)
        
        A_ub = sp.csr_matrix((data, (row_indices, col_indices)), shape=(num_constraints, n))
        b_ub = dist_matrix[pairs_i, pairs_j]
        
        # Variable bounds: 0 <= r_i <= distance to nearest boundary
        # Boundary limit for circle i is min(x, 1-x, y, 1-y)
        x = centers[:, 0]
        y = centers[:, 1]
        b_i = np.minimum(np.minimum(x, 1 - x), np.minimum(y, 1 - y))
        
        # Ensure bounds are non-negative to avoid infeasibility if center is on edge
        b_i = np.maximum(b_i, 0.0)
        
        bounds = [(0, b) for b in b_i]
        
        # Objective: maximize sum(r_i) => minimize -sum(r_i)
        c_obj = -np.ones(n)
        
        try:
            # Solve LP
            res = opt.linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
            
            if res.success:
                return res.x
            else:
                # If LP fails (shouldn't happen with valid bounds), return small radii
                return np.full(n, 1e-9)
        except Exception:
            return np.full(n, 1e-9)

    # Objective function for the outer optimizer
    # Input is a flat array of 52 coordinates [x1, y1, x2, y2, ..., x26, y26]
    def objective(center_flat):
        centers = center_flat.reshape(n, 2)
        
        # Clip centers to valid range [0, 1] to ensure boundary constraints make sense
        # Although DE bounds handle this, numerical safety is good.
        centers = np.clip(centers, 0.0, 1.0)
        
        radii = get_max_radii(centers)
        
        # We want to maximize sum of radii, so minimize negative sum
        return -np.sum(radii)

    # Initial guess: Hexagonal-like grid pattern
    # 6 columns, 5 rows = 30 points. We take 26.
    # Spacing roughly 0.2
    x_coords = np.linspace(0.1, 0.9, 6)
    y_coords = np.linspace(0.1, 0.9, 5)
    
    grid_points = []
    for y in y_coords:
        for x in x_coords:
            grid_points.append([x, y])
            
    # We have 30 points, take first 26
    initial_centers = np.array(grid_points[:n])
    initial_flat = initial_centers.flatten()
    
    # Bounds for DE: centers must be in [0, 1]
    bounds_de = [(0.0, 1.0)] * (2 * n)
    
    # Run Differential Evolution
    # popsize controls number of candidates, maxiter controls iterations
    # tol controls convergence threshold
    result = opt.differential_evolution(
        objective, 
        bounds_de, 
        initial_guess=initial_flat,
        popsize=20, 
        maxiter=1000, 
        tol=1e-6, 
        mutation=(0.5, 1.0), 
        recombination=0.7,
        seed=42,
        polish=True
    )
    
    best_centers = result.x.reshape(n, 2)
    best_radii = get_max_radii(best_centers)
    sum_radii = np.sum(best_radii)
    
    return best_centers, best_radii, sum_radii
