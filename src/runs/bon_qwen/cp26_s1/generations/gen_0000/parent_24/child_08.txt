To maximize the sum of radii for 26 circles in a unit square, we treat this as a continuous non-linear optimization problem. The objective is to maximize $\sum r_i$ subject to non-overlap and boundary constraints. 

We will use an **augmented Lagrangian / sequential penalty method** combined with the **BFGS** quasi-Newton optimizer. This approach is highly effective for packing problems because:
1. It transforms hard constraints into smooth penalty terms that the optimizer can handle efficiently.
2. BFGS converges rapidly for smooth objectives and provides good gradient approximations.
3. We iteratively increase the penalty coefficient $\lambda$, forcing the solution towards feasibility while continuously growing the radii.
4. Multiple random restarts ensure we escape poor local minima.

The constraints are formulated as:
- Boundary: $r_i \le x_i$, $r_i \le 1-x_i$, $r_i \le y_i$, $r_i \le 1-y_i$
- Non-overlap: $(x_i-x_j)^2 + (y_i-y_j)^2 \ge (r_i+r_j)^2$

Violations are squared and summed into the objective function. After optimization, a conservative scaling step ensures strict validity within the allowed numerical tolerance.

import numpy as np
from scipy.optimize import minimize

def compute_penalty(x, N, lam):
    """
    Computes the penalized objective function.
    x: flattened array [x1, y1, ..., xN, yN, r1, ..., rN]
    N: number of circles
    lam: penalty coefficient
    """
    centers = x[:2*N].reshape(N, 2)
    radii = x[2*N:]
    
    # Objective: maximize sum of radii => minimize negative sum
    obj = -np.sum(radii)
    
    # Boundary violations: radii must fit within [0,1]
    v = np.maximum(0, radii - centers[:, 0])**2
    v += np.maximum(0, radii - (1 - centers[:, 0]))**2
    v += np.maximum(0, radii - centers[:, 1])**2
    v += np.maximum(0, radii - (1 - centers[:, 1]))**2
    
    # Pairwise non-overlap violations (squared distance constraint for smoothness)
    dx = centers[:, 0, None] - centers[:, 0, None].T
    dy = centers[:, 1, None] - centers[:, 1, None].T
    dist_sq = dx**2 + dy**2
    r_sum = radii[:, None] + radii[None, :]
    r_sum_sq = r_sum**2
    
    i_idx, j_idx = np.triu_indices(N, k=1)
    pair_v = np.maximum(0, r_sum_sq[i_idx, j_idx] - dist_sq[i_idx, j_idx])**2
    
    return obj + lam * (np.sum(v) + np.sum(pair_v))

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    N = 26
    best_centers = np.zeros((N, 2))
    best_radii = np.zeros(N)
    best_sum = -np.inf
    
    # Configuration parameters
    restarts = 6
    lambda_iters = 18
    initial_lambda = 150.0
    
    for seed in range(restarts):
        # 1. Initialize centers and radii
        if seed == 0:
            # Hexagonal grid initialization (good for dense packing)
            cx, cy = [], []
            for i in range(5):
                y = 0.12 + i * 0.19
                row_n = 6 if i % 2 == 0 else 5
                for j in range(row_n):
                    if len(cx) >= N: break
                    x = 0.12 + j * 0.19
                    cx.append(x)
                    cy.append(y)
            cx = np.array(cx[:N])
            cy = np.array(cy[:N])
            while len(cx) < N:
                cx = np.append(cx, 0.5)
                cy = np.append(cy, 0.5)
        else:
            # Random initialization with boundary padding
            rng = np.random.RandomState(seed + 42)
            cx = rng.rand(N) * 0.6 + 0.2
            cy = rng.rand(N) * 0.6 + 0.2
            
        r_init = np.full(N, 0.025)
        x = np.concatenate([cx, cy, r_init])
        
        # 2. Sequential penalty optimization
        lam = initial_lambda
        for _ in range(lambda_iters):
            res = minimize(compute_penalty, x, args=(N, lam), method='BFGS', 
                           options={'maxiter': 400, 'ftol': 1e-13})
            x = res.x
            
            # Check feasibility to potentially early-exit or adjust lambda growth
            centers_tmp = x[:2*N].reshape(N, 2)
            radii_tmp = x[2*N:]
            
            # Quick violation check
            v_bound = np.maximum(0, radii_tmp - centers_tmp[:, 0])
            v_bound = np.maximum(v_bound, np.maximum(0, radii_tmp - (1 - centers_tmp[:, 0])))
            v_bound = np.maximum(v_bound, np.maximum(0, radii_tmp - centers_tmp[:, 1]))
            v_bound = np.maximum(v_bound, np.maximum(0, radii_tmp - (1 - centers_tmp[:, 1])))
            
            dx = centers_tmp[:, 0, None] - centers_tmp[:, 0, None].T
            dy = centers_tmp[:, 1, None] - centers_tmp[:, 1, None].T
            dist = np.sqrt(dx**2 + dy**2)
            r_sum = radii_tmp[:, None] + radii_tmp[None, :]
            i_idx, j_idx = np.triu_indices(N, k=1)
            v_pair = np.maximum(0, r_sum[i_idx, j_idx] - dist[i_idx, j_idx])
            
            max_viol = max(np.max(v_bound), np.max(v_pair))
            if max_viol < 1e-7:
                break
            lam *= 2.5 # Aggressive increase once close to feasible
            
        # 3. Post-process and validate
        centers = x[:2*N].reshape(N, 2)
        radii = x[2*N:]
        
        # Ensure strict validity by shrinking slightly if any violation exists
        v_bound = np.maximum(0, radii - centers[:, 0])
        v_bound = np.maximum(v_bound, np.maximum(0, radii - (1 - centers[:, 0])))
        v_bound = np.maximum(v_bound, np.maximum(0, radii - centers[:, 1]))
        v_bound = np.maximum(v_bound, np.maximum(0, radii - (1 - centers[:, 1])))
        
        dx = centers[:, 0, None] - centers[:, 0, None].T
        dy = centers[:, 1, None] - centers[:, 1, None].T
        dist = np.sqrt(dx**2 + dy**2)
        r_sum = radii[:, None] + radii[None, :]
        i_idx, j_idx = np.triu_indices(N, k=1)
        v_pair = np.maximum(0, r_sum[i_idx, j_idx] - dist[i_idx, j_idx])
        
        max_viol = max(np.max(v_bound), np.max(v_pair))
        if max_viol > 1e-10:
            # Scale down radii to guarantee feasibility
            scale = 1.0 - max_viol - 1e-8
            radii *= max(scale, 0.1)
        
        # Clamp to physical bounds
        radii = np.minimum(radii, centers[:, 0])
        radii = np.minimum(radii, 1 - centers[:, 0])
        radii = np.minimum(radii, centers[:, 1])
        radii = np.minimum(radii, 1 - centers[:, 1])
        radii = np.maximum(radii, 0.0)
        
        current_sum = np.sum(radii)
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers.copy()
            best_radii = radii.copy()
            
    return best_centers, best_radii, float(best_sum)
