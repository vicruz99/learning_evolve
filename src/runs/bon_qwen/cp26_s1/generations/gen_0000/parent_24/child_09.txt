

Looking at this circle packing problem, I need to maximize the sum of 26 circle radii within a unit square. Let me reason through an optimal strategy.

<strategy>
1. **Initialization**: Start with a hexagonal grid pattern - this is the densest 2D packing arrangement. Place ~26 circles in rows with alternating offsets.

2. **Optimization approach**: Use a multi-stage optimization:
   - Stage 1: Penalty-based optimization using scipy's Nelder-Mead to find good positions and radii simultaneously
   - Stage 2: Force-based simulation to refine - repulsive forces between overlapping circles, with iterative radius growth
   - Stage 3: Final greedy radius maximization where each circle expands to its maximum possible size given fixed positions of others

3. **Key considerations**:
   - Corner/edge circles can be larger than interior ones
   - Hexagonal spacing (vertical: r√3/2 between rows) is more efficient than square grid
   - Multiple restarts help escape local optima
   - The penalty weight must be high enough to enforce constraints but not dominate the objective

4. **Target analysis**: For 26 equal circles in hexagonal packing, r ≈ 0.097, sum ≈ 2.52. With unequal circles optimized, we should approach the 2.636 target.
</strategy>

import numpy as np
from scipy.optimize import minimize


def run_packing():
    n = 26
    np.random.seed(42)

    # Stage 1: Initialize with hexagonal pattern
    centers, radii = initialize_hexagonal(n)

    # Stage 2: Penalty-based optimization with multiple restarts
    centers, radii = optimize_with_penalty(centers, radii, n)

    # Stage 3: Force-based refinement
    centers, radii = force_refinement(centers, radii, n)

    # Stage 4: Final greedy radius maximization
    radii = maximize_radii(centers, radii, n)

    # Clip centers to valid range and ensure non-negative radii
    centers = np.clip(centers, 1e-6, 1 - 1e-6)
    radii = np.maximum(radii, 1e-6)

    return centers, radii, float(np.sum(radii))


def initialize_hexagonal(n):
    """Initialize circles in a hexagonal grid pattern."""
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.04

    idx = 0
    spacing_x = 0.18
    spacing_y = 0.15 * np.sqrt(3) / 2  # hexagonal vertical spacing
    margin = 0.08

    for row in range(7):
        for col in range(5):
            if idx >= n:
                break
            x = margin + col * spacing_x
            y = margin + row * spacing_y
            if row % 2 == 1:
                x += spacing_x / 2
            centers[idx] = [np.clip(x, 0.05, 0.95), np.clip(y, 0.05, 0.95)]
            idx += 1

    return centers, radii


def objective_function(x, n, weight=5000.0):
    """Objective: maximize sum of radii with penalty for constraint violations."""
    c = x[:2 * n].reshape(n, 2)
    r = np.maximum(x[2 * n:], 1e-6)

    penalty = 0.0

    # Boundary constraint penalties
    for i in range(n):
        xi, yi = c[i]
        ri = r[i]
        # Left boundary
        if xi - ri < 0:
            penalty += weight * (xi - ri) ** 2
        # Right boundary
        if xi + ri > 1:
            penalty += weight * (xi + ri - 1) ** 2
        # Bottom boundary
        if yi - ri < 0:
            penalty += weight * (yi - ri) ** 2
        # Top boundary
        if yi + ri > 1:
            penalty += weight * (yi + ri - 1) ** 2

    # Overlap constraint penalties
    for i in range(n):
        for j in range(i + 1, n):
            dx = c[i, 0] - c[j, 0]
            dy = c[i, 1] - c[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            overlap = r[i] + r[j] - dist
            if overlap > 0:
                penalty += weight * overlap ** 2

    return -np.sum(r) + penalty


def optimize_with_penalty(centers, radii, n):
    """Optimize using penalty-based method with multiple restarts."""
    x0 = np.concatenate([centers.flatten(), radii])

    best_x = x0.copy()
    best_obj = float('inf')

    # Multiple restarts with different seeds
    for restart in range(15):
        np.random.seed(42 + restart)
        x_start = x0.copy()

        # Add small random perturbation for diversity
        if restart > 0:
            x_start += np.random.randn(len(x_start)) * 0.015

        # Clip to reasonable bounds
        x_start[:2 * n] = np.clip(x_start[:2 * n], 0.02, 0.98)
        x_start[2 * n:] = np.maximum(x_start[2 * n:], 0.01)

        # Run Nelder-Mead optimization
        result = minimize(
            objective_function,
            x_start,
            args=(n, 5000.0),
            method='Nelder-Mead',
            options={'maxiter': 150000, 'xatol': 1e-10, 'fatol': 1e-10}
        )

        if result.fun < best_obj:
            best_obj = result.fun
            best_x = result.x.copy()

        # Also try with different penalty weights
        for w in [3000.0, 8000.0]:
            result2 = minimize(
                objective_function,
                x_start,
                args=(n, w),
                method='Nelder-Mead',
                options={'maxiter': 100000, 'xatol': 1e-10, 'fatol': 1e-10}
            )
            if result2.fun < best_obj:
                best_obj = result2.fun
                best_x = result2.x.copy()

    centers = best_x[:2 * n].reshape(n, 2)
    radii = np.maximum(best_x[2 * n:], 1e-6)

    return centers, radii


def force_refinement(centers, radii, n):
    """Refine packing using force-based simulation."""
    centers = centers.copy()
    radii = radii.copy()

    # Gradually increase radii while maintaining validity
    num_iterations = 20000

    for iteration in range(num_iterations):
        cooling = 1.0 - iteration / num_iterations
        step_size = 0.005 * cooling

        for i in range(n):
            # Compute repulsive forces from overlapping circles
            fx, fy = 0.0, 0.0

            for j in range(n):
                if i == j:
                    continue
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                if dist < 1e-10:
                    dist = 1e-10

                desired_dist = radii[i] + radii[j] + 0.001
                if dist < desired_dist:
                    force = (desired_dist - dist) / dist
                    fx += force * dx
                    fy += force * dy

            # Also consider boundary forces
            x, y = centers[i]
            r = radii[i]

            # Boundary repulsion
            if x - r < 0.01:
                fx += (0.01 - (x - r)) * 10
            if x + r > 0.99:
                fx -= ((x + r) - 0.99) * 10
            if y - r < 0.01:
                fy += (0.01 - (y - r)) * 10
            if y + r > 0.99:
                fy -= ((y + r) - 0.99) * 10

            # Apply forces
            centers[i, 0] = np.clip(centers[i, 0] + fx * step_size, 0.01, 0.99)
            centers[i, 1] = np.clip(centers[i, 1] + fy * step_size, 0.01, 0.99)

        # Periodically grow radii
        if iteration % 100 == 0:
            for i in range(n):
                max_r = float('inf')
                for j in range(n):
                    if i != j:
                        dist = np.sqrt((centers[i, 0] - centers[j, 0]) ** 2 +
                                       (centers[i, 1] - centers[j, 1]) ** 2)
                        max_r = min(max_r, dist - radii[j] - 1e-8)
                max_r = min(max_r, centers[i, 0], 1 - centers[i, 0],
                            centers[i, 1], 1 - centers[i, 1])
                # Gradually increase radius toward max
                target_r = max(0, max_r)
                radii[i] = radii[i] * 0.95 + target_r * 0.05

    return centers, radii


def maximize_radii(centers, radii, n):
    """Greedy final radius maximization."""
    radii = radii.copy()

    # Multiple passes to converge
    for _ in range(10):
        for i in range(n):
            max_r = float('inf')
            for j in range(n):
                if i != j:
                    dist = np.sqrt((centers[i, 0] - centers[j, 0]) ** 2 +
                                   (centers[i, 1] - centers[j, 1]) ** 2)
                    max_r = min(max_r, dist - radii[j])
            max_r = min(max_r, centers[i, 0], 1 - centers[i, 0],
                        centers[i, 1], 1 - centers[i, 1])
            radii[i] = max(1e-6, max_r)

    return radii
