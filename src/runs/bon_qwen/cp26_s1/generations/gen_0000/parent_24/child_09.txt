import numpy as np
from scipy.optimize import minimize


def hex_init(n):
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.06)
    idx = 0
    r = 0.06
    row_spacing = np.sqrt(3) * r
    for row in range(8):
        n_in_row = 4 if row % 2 == 0 else 4
        offset = r if row % 2 == 1 else 0
        for col in range(n_in_row):
            if idx >= n:
                break
            x = (col + 1) * 2 * r + offset + 0.1
            y = (row + 1) * row_spacing + 0.1
            centers[idx] = [np.clip(x, r, 1 - r), np.clip(y, r, 1 - r)]
            idx += 1
        if idx >= n:
            break
    return centers, radii


def grid_init(n):
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.06)
    row = 0
    col = 0
    for i in range(n):
        x = (col + 0.5) / 6 * 0.9 + 0.05
        y = (row + 0.5) / 6 * 0.9 + 0.05
        centers[i] = [np.clip(x, 0.06, 0.94), np.clip(y, 0.06, 0.94)]
        col += 1
        if col >= 5:
            col = 0
            row += 1
    return centers, radii


def random_init(n):
    centers = np.random.uniform(0.1, 0.9, (n, 2))
    radii = np.full(n, 0.03)
    return centers, radii


def force_optimize(centers, radii, n, iterations=50000):
    centers = centers.copy()
    radii = radii.copy()
    for it in range(iterations):
        forces = np.zeros_like(centers)
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                min_dist = radii[i] + radii[j]
                if dist < min_dist and dist > 1e-10:
                    overlap = min_dist - dist
                    force = overlap / dist
                    forces[i, 0] += dx * force * 3
                    forces[i, 1] += dy * force * 3
                    forces[j, 0] -= dx * force * 3
                    forces[j, 1] -= dy * force * 3
            for d in range(2):
                if centers[i, d] - radii[i] < 0.0001:
                    forces[i, d] += (radii[i] - centers[i, d] + 0.0001) * 5
                if centers[i, d] + radii[i] > 0.9999:
                    forces[i, d] -= (centers[i, d] + radii[i] - 0.9999) * 5
        lr = 0.005 * (1.0 - it / iterations * 0.9)
        centers += forces * lr
        growth = 0.00001 * (1.0 - it / iterations * 0.8)
        radii += growth
        for i in range(n):
            radii[i] = max(radii[i], 0.001)
            centers[i, 0] = np.clip(centers[i, 0], radii[i], 1 - radii[i])
            centers[i, 1] = np.clip(centers[i, 1], radii[i], 1 - radii[i])
    return centers, radii


def objective_func(x, n):
    total = 0
    for i in range(n):
        total -= x[3 * i + 2]
    return total


def constraint_boundary_i_x_low(x, i):
    return x[3 * i] - x[3 * i + 2]


def constraint_boundary_i_x_high(x, i):
    return 1 - x[3 * i] - x[3 * i + 2]


def constraint_boundary_i_y_low(x, i):
    return x[3 * i + 1] - x[3 * i + 2]


def constraint_boundary_i_y_high(x, i):
    return 1 - x[3 * i + 1] - x[3 * i + 2]


def constraint_radius_i(x, i):
    return x[3 * i + 2]


def constraint_overlap_ij(x, i, j):
    dx = x[3 * i] - x[3 * j]
    dy = x[3 * i + 1] - x[3 * j + 1]
    dist = np.sqrt(dx * dx + dy * dy)
    return dist - x[3 * i + 2] - x[3 * j + 2]


def build_constraints(n):
    constraints = []
    for i in range(n):
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: constraint_boundary_i_x_low(x, i)})
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: constraint_boundary_i_x_high(x, i)})
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: constraint_boundary_i_y_low(x, i)})
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: constraint_boundary_i_y_high(x, i)})
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: constraint_radius_i(x, i)})
    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({'type': 'ineq', 'fun': lambda x, i=i, j=j: constraint_overlap_ij(x, i, j)})
    return constraints


def scipy_refine(centers, radii, n):
    x0 = np.zeros(n * 3)
    for i in range(n):
        x0[3 * i] = centers[i, 0]
        x0[3 * i + 1] = centers[i, 1]
        x0[3 * i + 2] = radii[i]
    constraints = build_constraints(n)
    result = minimize(
        lambda x: objective_func(x, n),
        x0,
        method='SLSQP',
        constraints=constraints,
        options={'maxiter': 500, 'ftol': 1e-12}
    )
    centers_opt = np.array([[result.x[3 * i], result.x[3 * i + 1]] for i in range(n)])
    radii_opt = np.array([result.x[3 * i + 2] for i in range(n)])
    return centers_opt, radii_opt


def run_packing():
    n = 26
    best_centers = None
    best_radii = None
    best_sum = 0
    inits = [
        hex_init(n),
        grid_init(n),
    ]
    for _ in range(8):
        inits.append(random_init(n))
    for centers, radii in inits:
        centers, radii = force_optimize(centers, radii, n, iterations=50000)
        s = np.sum(radii)
        if s > best_sum:
            best_sum = s
            best_centers = centers.copy()
            best_radii = radii.copy()
    best_centers, best_radii = scipy_refine(best_centers, best_radii, n)
    return best_centers, best_radii, float(np.sum(best_radii))
