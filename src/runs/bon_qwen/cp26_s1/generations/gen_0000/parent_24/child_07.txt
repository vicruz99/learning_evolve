import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    Uses numerical optimization (SLSQP).
    """
    n = 26
    best_sum_radii = -np.inf
    best_centers = None
    best_radii = None

    # Helper function to generate a hexagonal lattice initialization
    def get_hexagonal_init(seed_offset=0):
        rng = np.random.default_rng(42 + seed_offset)
        
        # Start with a reasonable radius estimate
        # 25 circles in 5x5 grid have r=0.1. 
        # Hexagonal packing allows slightly denser packing, but we have 26 circles.
        # Let's start with r=0.08 to ensure feasibility and allow growth.
        r_init = 0.08
        
        centers = []
        y = r_init
        row = 0
        while y < 1.0 - r_init:
            # Hexagonal shift
            shift = r_init if row % 2 == 1 else 0.0
            x = r_init + shift
            
            while x < 1.0 - r_init:
                centers.append([x, y])
                x += 2 * r_init
            y += r_init * np.sqrt(3)
            row += 1
        
        # If we have fewer points than needed, fill with random points
        # If we have more, truncate
        if len(centers) < n:
            while len(centers) < n:
                cx = rng.uniform(r_init, 1.0 - r_init)
                cy = rng.uniform(r_init, 1.0 - r_init)
                centers.append([cx, cy])
        
        centers = centers[:n]
        
        # Add some random perturbation to avoid symmetry issues
        noise_scale = 0.01
        centers = np.array(centers) + rng.uniform(-noise_scale, noise_scale, size=centers.shape)
        # Clip to valid range
        centers = np.clip(centers, r_init, 1.0 - r_init)
        
        # Initial radii: small enough to be valid
        radii = np.ones(n) * r_init * 0.8
        
        return centers, radii

    def objective(vars):
        # vars is flattened [x1, y1, r1, x2, y2, r2, ...]
        # r are at indices 2, 5, 8, ... i.e., 2::3
        r = vars[2::3]
        return -np.sum(r)

        # Extract centers and radii
        # centers shape (26, 2)
        # radii shape (26,)
        centers = np.column_stack((vars[0::3], vars[1::3]))
        r = vars[2::3]

        # diff shape (26, 26, 2)
        diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))

        r_sum = r[:, np.newaxis] + r[np.newaxis, :]
        
        # Constraint: dist >= r_i + r_j  =>  dist - (r_i + r_j) >= 0
        # We only care about upper triangle to avoid duplicates
        mask = np.triu(np.ones((n, n), dtype=bool), k=1)
        vals = dists[mask] - r_sum[mask]
        
        return vals

    def constraints_boundary(vars):
        centers = np.column_stack((vars[0::3], vars[1::3]))
        r = vars[2::3]
        
        # x >= r, 1-x >= r, y >= r, 1-y >= r
        # => x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
        c1 = centers[:, 0] - r
        c2 = 1.0 - centers[:, 0] - r
        c3 = centers[:, 1] - r
        c4 = 1.0 - centers[:, 1] - r
        
        return np.concatenate([c1, c2, c3, c4])

    # Define constraints for scipy
    # 'ineq' type means fun(x) >= 0
    cons = [
        {'type': 'ineq', 'fun': constraints_non_overlap},
        {'type': 'ineq', 'fun': constraints_boundary}
    ]
    
    # Bounds for variables

    bounds = []

        bounds.extend([(0, 1), (0, 1), (0, 0.5)])

        x0[0::3] = centers_init[:, 0]
        x0[1::3] = centers_init[:, 1]
        x0[2::3] = radii_init
        
        try:
            result = minimize(
                objective, 
                x0, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 1000, 'ftol': 1e-9, 'disp': False}
            )

                    best_radii = result.x[2::3]
        except Exception:
            continue

    # If optimization failed to find anything better than initial guess (unlikely given logic)
    if best_centers is None:

        radii = []

                centers.append([0.1 + i*0.2, 0.1 + j*0.2])

        # Actually, let's just return the best found
