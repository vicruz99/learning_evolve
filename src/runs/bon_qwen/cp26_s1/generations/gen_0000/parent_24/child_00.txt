import numpy as np

def compute_forces(centers, radii):
    """Compute repulsive forces to resolve overlaps and boundary violations."""
    n = centers.shape[0]
    forces = np.zeros_like(centers)
    k_rep = 20.0
    k_wall = 40.0
    
    for i in range(n):
        # Wall repulsion
        for d in range(2):
            if centers[i, d] < radii[i]:
                forces[i, d] += k_wall * (radii[i] - centers[i, d])
            elif centers[i, d] > 1.0 - radii[i]:
                forces[i, d] -= k_wall * (centers[i, d] - (1.0 - radii[i]))
                
        # Pairwise circle repulsion
        for j in range(i + 1, n):
            diff = centers[i] - centers[j]
            dist_sq = np.sum(diff**2)
            dist = np.sqrt(dist_sq)
            
            if dist < 1e-10:
                continue  # Skip identical positions; jitter prevents this
                
            min_dist = radii[i] + radii[j]
            if dist < min_dist:
                # Repulsive force proportional to overlap distance
                force_mag = k_rep * (min_dist - dist) / dist
                forces[i] += diff * force_mag
                forces[j] -= diff * force_mag
    return forces

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Evaluate multiple initial configurations to escape poor local optima
    for seed in range(12):
        rng = np.random.default_rng(seed * 17 + 3)
        
        if seed == 0:
            # Hexagonal-inspired grid initialization for dense packing
            pts = []
            for r in range(5):
                for c in range(6):
                    if len(pts) < n:
                        x = 0.12 + c * 0.13
                        y = 0.12 + r * 0.17
                        if r % 2 == 1:
                            x += 0.065
                        pts.append([x, y])
            centers = np.array(pts)
        else:
            centers = rng.random((n, 2)) * 0.8 + 0.1
            
        radii = np.full(n, 0.025)
        
        grow_rate = 0.0018
        dt = 0.06
        tol = 1e-7
        
        local_best_sum = 0.0
        local_best_c = centers.copy()
        local_best_r = radii.copy()
        
        for step in range(4000):
            # Expand all circles uniformly to maximize sum of radii
            radii += grow_rate
            
            # Apply repulsive forces to resolve collisions
            forces = compute_forces(centers, radii)
            centers += forces * dt
            
            # Constrain centers to unit square
            centers = np.clip(centers, 0.0, 1.0)
            
            # Periodic perturbation to help escape local minima
            if step % 400 == 0:
                centers += rng.normal(0, 0.003, size=(n, 2))
                centers = np.clip(centers, 0.0, 1.0)
            
            # Validate current state
            valid = True
            for i in range(n):
                r = radii[i]
                x, y = centers[i]
                if x < r - tol or x > 1.0 - r + tol or y < r - tol or y > 1.0 - r + tol:
                    valid = False
                    break
                    
            if valid:
                for i in range(n):
                    for j in range(i + 1, n):
                        dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                        if dist < radii[i] + radii[j] - tol:
                            valid = False
                            break
                    if not valid:
                        break
                        
            if valid:
                current_sum = np.sum(radii)
                if current_sum > local_best_sum:
                    local_best_sum = current_sum
                    local_best_c = centers.copy()
                    local_best_r = radii.copy()
            
            # Cool down parameters for convergence
            if step % 150 == 0:
                grow_rate *= 0.975
                dt *= 0.995
                
        if local_best_sum > best_sum:
            best_sum = local_best_sum
            best_centers = local_best_c
            best_radii = local_best_r

    # Final precision safeguard
    # Ensure strict feasibility by slightly shrinking radii if numerical artifacts exist
    final_tol = 1e-10
    valid = True
    for i in range(n):
        r = best_radii[i]
        x, y = best_centers[i]
        if x < r - final_tol or x > 1.0 - r + final_tol or y < r - final_tol or y > 1.0 - r + final_tol:
            valid = False
            break
    if valid:
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((best_centers[i] - best_centers[j])**2))
                if dist < best_radii[i] + best_radii[j] - final_tol:
                    valid = False
                    break
            if not valid:
                break
                
    if not valid:
        # Minimal scaling to guarantee validation passes
        scale = 0.999
        while not valid:
            best_radii *= scale
            valid = True
            for i in range(n):
                r = best_radii[i]
                x, y = best_centers[i]
                if x < r - final_tol or x > 1.0 - r + final_tol or y < r - final_tol or y > 1.0 - r + final_tol:
                    valid = False
                    break
            if valid:
                for i in range(n):
                    for j in range(i + 1, n):
                        dist = np.sqrt(np.sum((best_centers[i] - best_centers[j])**2))
                        if dist < best_radii[i] + best_radii[j] - final_tol:
                            valid = False
                            break
                    if not valid:
                        break
            scale *= 0.999
            if scale < 0.95:
                break # Fallback safety

    return best_centers, best_radii, float(np.sum(best_radii))
