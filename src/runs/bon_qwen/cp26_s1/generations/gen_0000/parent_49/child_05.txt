import numpy as np
from scipy.optimize import minimize

def compute_overlap(vars, n):
    """Compute non-overlap constraints: dist(i,j) >= r_i + r_j"""
    c = vars[:2*n].reshape(n, 2)
    r = vars[2*n:]
    dists = np.linalg.norm(c[:, None, :] - c[None, :, :], axis=2)
    np.fill_diagonal(dists, np.inf)
    idx = np.triu_indices(n, k=1)
    return dists[idx] - (r[:, None] + r[None, :])[idx]

def compute_boundary(vars, n):
    """Compute boundary constraints: circle inside [0,1]^2"""
    c = vars[:2*n].reshape(n, 2)
    r = vars[2*n:]
    dists_to_bound = np.minimum(np.minimum(c[:, 0], 1.0 - c[:, 0]), 
                                np.minimum(c[:, 1], 1.0 - c[:, 1]))
    return dists_to_bound - r

    np.random.seed(42)

    row, col, idx = 0, 0, 0

        y = row * 0.18
        if 0 <= x <= 1 and 0 <= y <= 1:
            centers[idx] = [x, y]
            col += 1
        if col > 5 or x > 1:
            row += 1
            col = 0
        idx += 1
        
    # 2. Force-directed expansion simulation
    # Grows radii while resolving overlaps via repulsive forces
    lr, penalty, growth, dt = 0.008, 100.0, 0.002, 0.01
    for step in range(8000):
        radii += growth * dt
        
        diffs = centers[:, None, :] - centers[None, :, :]
        dists = np.linalg.norm(diffs, axis=2)
        np.fill_diagonal(dists, np.inf)
        
        r_sum = radii[:, None] + radii[None, :]
        overlaps = np.maximum(0, r_sum - dists)
        dirs = diffs / (dists[:, :, None] + 1e-8)
        forces = np.sum(overlaps[:, :, None] * penalty * dirs, axis=1)
        
        # Boundary repulsion
        forces[:, 0] += penalty * np.maximum(0, radii - centers[:, 0])
        forces[:, 0] -= penalty * np.maximum(0, centers[:, 0] + radii - 1)
        forces[:, 1] += penalty * np.maximum(0, radii - centers[:, 1])
        forces[:, 1] -= penalty * np.maximum(0, centers[:, 1] + radii - 1)
        
        centers -= forces * lr * dt
        centers = np.clip(centers, 0.0, 1.0)

            lr *= 0.9
            penalty *= 1.1
            growth *= 0.9

    dists = np.linalg.norm(centers[:, None, :] - centers[None, :, :], axis=2)

        radii -= max_ov / 2

    init_vars = np.concatenate([centers.flatten(), radii])
    bounds = [(0, 1)] * 2*n + [(0, 0.5)] * n

        {'type': 'ineq', 'fun': compute_overlap, 'args': (n,)},
        {'type': 'ineq', 'fun': compute_boundary, 'args': (n,)}

        return -np.sum(vars[2*n:])

                   constraints=constraints, options={'maxiter': 2000, 'ftol': 1e-9})

    final_radii = res.x[2*n:]

    final_centers = np.clip(final_centers, 0.0, 1.0)
    final_radii = np.maximum(0, final_radii)
