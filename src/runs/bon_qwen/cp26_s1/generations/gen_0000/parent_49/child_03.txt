import numpy as np
from scipy.optimize import minimize
import math

def compute_penalty(params, n, penalty_weight):
    centers = params[:2 * n].reshape(n, 2)
    radii = params[2 * n:]
    penalty = 0.0

    for i in range(n):
        x, y = centers[i]
        ri = radii[i]
        if ri < 0:
            penalty += penalty_weight * ri * ri
        if x - ri < 0:
            violation = -(x - ri)
            penalty += penalty_weight * violation * violation
        if x + ri > 1:
            violation = (x + ri - 1)
            penalty += penalty_weight * violation * violation
        if y - ri < 0:
            violation = -(y - ri)
            penalty += penalty_weight * violation * violation
        if y + ri > 1:
            violation = (y + ri - 1)
            penalty += penalty_weight * violation * violation

    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i][0] - centers[j][0]
            dy = centers[i][1] - centers[j][1]
            dist = math.sqrt(dx * dx + dy * dy)
            overlap = radii[i] + radii[j] - dist
            if overlap > 0:
                penalty += penalty_weight * overlap * overlap

    return penalty


def objective_with_penalty(params, n, penalty_weight):
    radii = params[2 * n:]
    return -np.sum(radii) + compute_penalty(params, n, penalty_weight)


def generate_grid_config(n, grid_x, grid_y):
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    idx = 0
    r = 0.1
    for i in range(grid_x):
        for j in range(grid_y):
            if idx < n:
                centers[idx] = [r + i * 2 * r, r + j * 2 * r]
                radii[idx] = r
                idx += 1

    while idx < n:
        centers[idx] = [0.5, 0.5]
        radii[idx] = 0.01
        idx += 1

    return centers, radii


def generate_hex_config(n, rows, cols_list):
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    idx = 0
    r = 0.085
    for row in range(rows):
        y = r + row * math.sqrt(3) * r
        cols = cols_list[row]
        offset = r if row % 2 == 1 else 0
        for col in range(cols):
            if idx < n:
                x = r + col * 2 * r + offset
                centers[idx] = [x, y]
                radii[idx] = r
                idx += 1

    while idx < n:
        centers[idx] = [0.5, 0.5]
        radii[idx] = 0.01
        idx += 1

    return centers, radii


def perturb_config(centers, radii, magnitude):
    centers = centers.copy()
    rng = np.random.RandomState(42)
    centers += rng.uniform(-magnitude, magnitude, centers.shape)
    centers = np.clip(centers, 0.01, 0.99)
    return centers, radii


def optimize_positions(centers, radii, n, max_iter=2000, lr=0.005):
    centers = centers.copy()
    radii = radii.copy()

    for iteration in range(max_iter):
        grad = np.zeros_like(centers)

        for i in range(n):
            for j in range(i + 1, n):
                diff = centers[i] - centers[j]
                dist_sq = np.sum(diff * diff)
                dist = math.sqrt(dist_sq)
                min_dist = radii[i] + radii[j]
                if dist < min_dist + 0.005 and dist > 1e-10:
                    force = (min_dist - dist + 0.005) / dist * diff
                    grad[i] += force
                    grad[j] -= force

        for i in range(n):
            x, y = centers[i]
            ri = radii[i]
            if x < ri + 0.002:
                grad[i][0] += (ri + 0.002 - x)
            if x > 1 - ri - 0.002:
                grad[i][0] -= (x - (1 - ri - 0.002))
            if y < ri + 0.002:
                grad[i][1] += (ri + 0.002 - y)
            if y > 1 - ri - 0.002:
                grad[i][1] -= (y - (1 - ri - 0.002))

        centers += lr * grad
        centers = np.clip(centers, 0.001, 0.999)
        lr *= 0.9995

    return centers


def optimize_radii(centers, radii, n):
    radii = radii.copy()

    for _ in range(200):
        new_radii = radii.copy()
        for i in range(n):
            r_boundary = min(centers[i][0], 1 - centers[i][0], centers[i][1], 1 - centers[i][1])
            r_circles = r_boundary
            for j in range(n):
                if i != j:
                    dx = centers[i][0] - centers[j][0]
                    dy = centers[i][1] - centers[j][1]
                    dist = math.sqrt(dx * dx + dy * dy)
                    max_r = dist - radii[j]
                    if max_r < r_circles:
                        r_circles = max_r
            new_radii[i] = max(0.001, r_circles)

        if np.max(np.abs(new_radii - radii)) < 1e-12:
            break
        radii = new_radii

    return radii


def alternating_opt(centers, radii, n, outer_iters=30):
    for _ in range(outer_iters):
        centers = optimize_positions(centers, radii, n)
        radii = optimize_radii(centers, radii, n)
    return centers, radii, np.sum(radii)


def optimize_with_scipy(centers, radii, n, max_iter=200000):
    initial_params = np.concatenate([centers.flatten(), radii])
    penalty_weight = 50000

    def obj(params):
        return objective_with_penalty(params, n, penalty_weight)

    result = minimize(obj, initial_params, method='Nelder-Mead',
                      options={'maxiter': max_iter, 'xatol': 1e-12, 'fatol': 1e-12, 'adaptive': True})

    final_centers = result.x[:2 * n].reshape(n, 2)
    final_radii = result.x[2 * n:]
    final_radii = np.maximum(final_radii, 0.0001)
    final_centers = np.clip(final_centers, 0.001, 0.999)

    return final_centers, final_radii, np.sum(final_radii)


def ensure_validity(centers, radii, n):
    centers = centers.copy()
    radii = radii.copy()

    for _ in range(100):
        changed = False
        for i in range(n):
            x, y = centers[i]
            ri = radii[i]
            r_boundary = min(x, 1 - x, y, 1 - y)
            if ri > r_boundary:
                radii[i] = max(0.0001, r_boundary)
                changed = True

        for _ in range(50):
            changed_inner = False
            for i in range(n):
                for j in range(i + 1, n):
                    dx = centers[i][0] - centers[j][0]
                    dy = centers[i][1] - centers[j][1]
                    dist = math.sqrt(dx * dx + dy * dy)
                    total = radii[i] + radii[j]
                    if total > dist + 1e-12:
                        reduction = (total - dist - 1e-12) / 2
                        radii[i] = max(0.0001, radii[i] - reduction / 2)
                        radii[j] = max(0.0001, radii[j] - reduction / 2)
                        changed_inner = True
            if not changed_inner:
                break

        if not changed:
            break

    return centers, radii
