import numpy as np
from scipy.optimize import minimize

def objective(v, n):
    return -np.sum(v[2*n:])

def constraint_x_ge_r(v, i, n):
    return v[i] - v[2*n+i]

def constraint_x_le_1mr(v, i, n):
    return 1.0 - v[i] - v[2*n+i]

def constraint_y_ge_r(v, i, n):
    return v[n+i] - v[2*n+i]

def constraint_y_le_1mr(v, i, n):
    return 1.0 - v[n+i] - v[2*n+i]

def constraint_pair(v, i, j, n):
    dx = v[i] - v[j]
    dy = v[n+i] - v[n+j]
    return np.sqrt(dx*dx + dy*dy) - v[2*n+i] - v[2*n+j]

        while x < 0.9 and len(centers) < n:

            x += 0.18
            if len(centers) < n:
                centers.append([x + 0.09, y])
                x += 0.18
        y += 0.156
        
    centers = np.array(centers[:n])
    # Add slight deterministic perturbation to break symmetry
    centers = centers + np.random.uniform(-0.02, 0.02, size=centers.shape)
    centers = np.clip(centers, 0.05, 0.95)
    
    radii_init = np.full(n, 0.08)
    
    x0 = centers[:, 0]
    y0 = centers[:, 1]
    r0 = radii_init
    vars0 = np.concatenate([x0, y0, r0])
    
    bounds = [(0.0, 1.0)]*n + [(0.0, 1.0)]*n + [(0.0, 0.5)]*n
    
    cons = []
    for i in range(n):
        cons.append({'type': 'ineq', 'fun': constraint_x_ge_r, 'args': (i, n)})
        cons.append({'type': 'ineq', 'fun': constraint_x_le_1mr, 'args': (i, n)})
        cons.append({'type': 'ineq', 'fun': constraint_y_ge_r, 'args': (i, n)})
        cons.append({'type': 'ineq', 'fun': constraint_y_le_1mr, 'args': (i, n)})
        
    for i in range(n):
        for j in range(i+1, n):
            cons.append({'type': 'ineq', 'fun': constraint_pair, 'args': (i, j, n)})
            
    res = minimize(objective, vars0, method='SLSQP', bounds=bounds, constraints=cons, 
                   options={'maxiter': 2000, 'ftol': 1e-9, 'disp': False})
    
    cx = res.x[:n]
    cy = res.x[n:2*n]
    cr = res.x[2*n:]
    
    centers_out = np.column_stack([cx, cy])
    radii_out = cr.copy()
    
    # Post-processing to ensure strict validity per validation function tolerances
    for i in range(n):
        r = radii_out[i]
        x, y = centers_out[i]
        r = min(r, x, 1.0-x, y, 1.0-y)
        radii_out[i] = max(r, 1e-7)
        
    # Resolve any remaining numerical overlaps by shrinking radii
    for _ in range(500):
        overlap_found = False
        for i in range(n):
            for j in range(i+1, n):

                    radii_out[j] -= diff

                    radii_out[j] = max(radii_out[j], 1e-7)
                    overlap_found = True

            break
            
    return centers_out, radii_out, np.sum(radii_out)
