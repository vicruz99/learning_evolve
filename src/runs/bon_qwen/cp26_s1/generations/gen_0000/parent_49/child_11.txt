import numpy as np

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square

    Args:
        centers: np.array of shape (n, 2) with (x, y) coordinates
        radii: np.array of shape (n) with radius of each circle

    Returns:
        True if valid, False otherwise
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    Optimizes the packing of 26 circles in a unit square to maximize sum of radii.
    Uses a force-directed simulation with inflation and perturbation.
    """
    n = 26
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Configuration for the simulation
    num_attempts = 15
    simulation_steps = 3000
    growth_rate = 0.00015
    repulsion_strength = 50.0
    damping = 0.1
    
    for attempt in range(num_attempts):
        # 1. Initialization
        # Place centers somewhat uniformly but with randomness
        # Using a perturbed grid helps initial packing
        centers = np.random.rand(n, 2) * 0.8 + 0.1
        # Ensure initial separation roughly
        radii = np.full(n, 0.02) 
        
        # 2. Simulation Loop
        for step in range(simulation_steps):
            # Increase radii (inflation)
            # We grow them slowly to allow rearrangement
            radii += growth_rate
            
            # Compute forces
            forces = np.zeros_like(centers)
            
            # Pairwise repulsion
            for i in range(n):
                for j in range(i + 1, n):
                    diff = centers[i] - centers[j]
                    dist_sq = np.dot(diff, diff)
                    dist = np.sqrt(dist_sq)
                    
                    if dist < 1e-9:
                        dist = 1e-9
                        diff = np.array([1e-4, 1e-4]) # random perturbation direction if coincident
                    
                    req_dist = radii[i] + radii[j]
                    
                    if dist < req_dist:
                        overlap = req_dist - dist
                        # Force proportional to overlap
                        # Using a slightly stronger force for larger overlaps
                        f_mag = repulsion_strength * overlap
                        direction = diff / dist
                        forces[i] += f_mag * direction
                        forces[j] -= f_mag * direction
            
            # Wall repulsion
            for i in range(n):
                r = radii[i]
                x, y = centers[i]
                
                # Left wall
                if x < r:
                    forces[i, 0] += repulsion_strength * (r - x)
                # Right wall
                if x > 1 - r:
                    forces[i, 0] -= repulsion_strength * (x - (1 - r))
                # Bottom wall
                if y < r:
                    forces[i, 1] += repulsion_strength * (r - y)
                # Top wall
                if y > 1 - r:
                    forces[i, 1] -= repulsion_strength * (y - (1 - r))
            
            # Update positions with damping
            centers += damping * forces
            
            # Clamp centers to [0, 1] roughly to prevent explosion, 
            # though wall forces should handle it. 
            # Strict clamping ensures numerical stability.
            np.clip(centers[:, 0], 0.0, 1.0, out=centers[:, 0])
            np.clip(centers[:, 1], 0.0, 1.0, out=centers[:, 1])
            
            # Perturbation to escape local minima
            # Shake more frequently at the beginning, less later
            if step % 50 == 0 and step > 0:
                noise_scale = 0.02 * (1.0 - step / simulation_steps) + 0.001
                centers += np.random.randn(n, 2) * noise_scale
                
                # Occasionally shrink radii slightly to unjam
                if step % 200 == 0:
                    radii *= 0.95

        # 3. Post-processing / Hard Projection
        # The simulation leaves small overlaps; we need to strictly satisfy constraints.
        # Iterative projection
        
        # First, clip to boundaries based on radii
        for i in range(n):
            r = radii[i]
            centers[i, 0] = np.clip(centers[i, 0], r, 1 - r)
            centers[i, 1] = np.clip(centers[i, 1], r, 1 - r)

        for _ in range(200):
            overlap_found = False

                    diff = centers[i] - centers[j]
                    dist = np.linalg.norm(diff)
                    req = radii[i] + radii[j]
                    
                    if dist < req and dist > 1e-9:
                        overlap_found = True
                        # Push apart equally
                        shift = (req - dist) / 2.0
                        direction = diff / dist
                        centers[i] += direction * shift
                        centers[j] -= direction * shift
            
            # Re-clip after moves
            for i in range(n):
                r = radii[i]
                centers[i, 0] = np.clip(centers[i, 0], r, 1 - r)
                centers[i, 1] = np.clip(centers[i, 1], r, 1 - r)
            
            if not overlap_found:
                break

        # Final validity check and update best
        # We need a local check function or just trust the projection
        # Let's do a quick check
        is_valid = True
        # Check bounds
        for i in range(n):
            if centers[i,0] < radii[i] - 1e-9 or centers[i,0] > 1 - radii[i] + 1e-9 or \
               centers[i,1] < radii[i] - 1e-9 or centers[i,1] > 1 - radii[i] + 1e-9:
                is_valid = False
                break
        if is_valid:
            for i in range(n):
                for j in range(i + 1, n):
                    dist = np.linalg.norm(centers[i] - centers[j])
                    if dist < radii[i] + radii[j] - 1e-9:
                        is_valid = False
                        break
                if not is_valid: break
        
        if is_valid:
            current_sum = np.sum(radii)
            if current_sum > best_sum:
                best_sum = current_sum
                best_centers = centers.copy()
                best_radii = radii.copy()

    # Fallback if no valid packing found (should not happen with this logic)
    if best_centers is None:
        best_centers = np.random.rand(n, 2) * 0.8 + 0.1
        best_radii = np.full(n, 0.01)
        best_sum = np.sum(best_radii)

    return best_centers, best_radii, float(best_sum)
