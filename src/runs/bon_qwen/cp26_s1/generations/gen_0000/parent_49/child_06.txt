import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def penalty_term(x):
    """Compute squared violation penalties for boundaries and overlaps."""
    centers = x[:2*N_CIRCLES].reshape(N_CIRCLES, 2)
    radii = x[2*N_CIRCLES:]
    p = 0.0
    
    # Boundary penalties
    for i in range(N_CIRCLES):
        cx, cy = centers[i]
        r = radii[i]
        if cx < r: p += (cx - r)**2
        if cx > 1 - r: p += (cx - (1 - r))**2
        if cy < r: p += (cy - r)**2
        if cy > 1 - r: p += (cy - (1 - r))**2
        
    # Overlap penalties
    for i in range(N_CIRCLES):
        for j in range(i + 1, N_CIRCLES):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.hypot(dx, dy)
            sum_r = radii[i] + radii[j]
            if dist < sum_r:
                p += (dist - sum_r)**2
    return p

def objective(x):
    """Objective to minimize: negative sum of radii + penalty."""
    radii = x[2*N_CIRCLES:]
    mu = 5000.0
    return -np.sum(radii) + mu * penalty_term(x)

def make_feasible(centers, radii):
    """Shrink radii slightly until all constraints are strictly satisfied."""
    while True:
        valid = True
        for i in range(N_CIRCLES):
            cx, cy = centers[i]
            r = radii[i]
            if cx - r < -1e-9 or cx + r > 1 + 1e-9 or cy - r < -1e-9 or cy + r > 1 + 1e-9:
                valid = False
                break
        if valid:
            for i in range(N_CIRCLES):
                for j in range(i + 1, N_CIRCLES):
                    dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                    if dist < radii[i] + radii[j] - 1e-9:
                        valid = False
                        break
                if not valid:
                    break
        if valid:
            return centers, radii
        radii = radii * 0.995

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    bounds = [(0.0, 1.0)] * (2 * N_CIRCLES) + [(0.0, 0.25)] * N_CIRCLES
    
    best_centers = None
    best_radii = None
    best_sum = -1.0
    
    # Generate diverse initial configurations
    inits = []
    
    # 1. Hexagonal lattice approximation
    pts = []
    y = 0.12
    while len(pts) < N_CIRCLES:
        x = 0.12
        shift = 0.08 if (len(pts) // 5) % 2 == 1 else 0.0
        while len(pts) < N_CIRCLES and x < 0.9:
            pts.append([x + shift, y])
            x += 0.16
        y += 0.14
    inits.append(np.array(pts[:N_CIRCLES]))
    
    # 2. Random spread
    np.random.seed(2024)
    inits.append(np.random.rand(N_CIRCLES, 2) * 0.7 + 0.15)
    
    # 3. Uniform grid subset
    gs = np.linspace(0.15, 0.85, 6)
    pts3 = [[gx, gy] for gx in gs for gy in gs]
    inits.append(np.array(pts3[:N_CIRCLES]))
    
    for init_c in inits:
        init_r = np.full(N_CIRCLES, 0.06)
        x0 = np.concatenate([init_c.flatten(), init_r])
        
        # First optimization run
        res = minimize(objective, x0, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': 2000, 'ftol': 1e-12})
        
        c = res.x[:2*N_CIRCLES].reshape(N_CIRCLES, 2)
        r = res.x[2*N_CIRCLES:]
        c, r = make_feasible(c, r)
        
        if np.sum(r) > best_sum:
            best_sum = np.sum(r)
            best_centers = c.copy()
            best_radii = r.copy()
            
        # Continuation method: push radii up and re-optimize to escape local minima
        r_push = best_radii * 1.05
        x0_push = np.concatenate([best_centers.flatten(), r_push])
        res2 = minimize(objective, x0_push, method='L-BFGS-B',
                        bounds=bounds, options={'maxiter': 1500, 'ftol': 1e-12})
        
        c2 = res2.x[:2*N_CIRCLES].reshape(N_CIRCLES, 2)
        r2 = res2.x[2*N_CIRCLES:]
        c2, r2 = make_feasible(c2, r2)
        
        if np.sum(r2) > best_sum:
            best_sum = np.sum(r2)
            best_centers = c2.copy()
            best_radii = r2.copy()
            
    return best_centers, best_radii, best_sum
