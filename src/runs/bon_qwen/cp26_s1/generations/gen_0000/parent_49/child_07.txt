import numpy as np
import scipy.optimize as opt

def calculate_max_radii(centers):
    """
    Calculates the maximum valid radii for a set of centers.
    Uses the heuristic r_i = 0.5 * min(dist to boundary, min dist to other centers).
    This guarantees a valid non-overlapping packing.
    """
    n = centers.shape[0]
    radii = np.zeros(n)
    
    for i in range(n):
        x, y = centers[i]
        # Distance to boundaries
        dist_boundary = min(x, 1 - x, y, 1 - y)
        
        # Distance to other centers
        dist_centers = np.sqrt(np.sum((centers - centers[i])**2, axis=1))
        dist_centers[i] = np.inf # Ignore self
        min_dist_center = np.min(dist_centers)
        
        # The radius is limited by half the distance to the nearest obstacle (boundary or circle)
        # If we set r_i = 0.5 * dist, then r_i + r_j <= 0.5*d_ij + 0.5*d_ij = d_ij
        radii[i] = 0.5 * min(dist_boundary, min_dist_center)
        
    return radii

    return -np.sum(radii)
