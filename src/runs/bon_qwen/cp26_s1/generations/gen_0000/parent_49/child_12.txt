import numpy as np
from scipy.optimize import minimize

def obj_func(v, n):
    return -np.sum(v[2*n:])

def cons_x_min(v, i, n):
    return v[2*i] - v[2*n+i]

def cons_x_max(v, i, n):
    return 1.0 - v[2*i] - v[2*n+i]

def cons_y_min(v, i, n):
    return v[2*i+1] - v[2*n+i]

def cons_y_max(v, i, n):
    return 1.0 - v[2*i+1] - v[2*n+i]

def cons_dist(v, i, j, n):
    dx = v[2*i] - v[2*j]
    dy = v[2*i+1] - v[2*j+1]
    r_sum = v[2*n+i] + v[2*n+j]
    return dx*dx + dy*dy - r_sum*r_sum

def run_packing():
    n = 26
    best_sum = 0.0
    best_centers = np.zeros((n, 2))
    best_radii = np.zeros(n)
    
    # Generate multiple starting configurations to escape local optima
    configs = []
    
    # 1. Hexagonal grid (optimal for infinite plane, excellent base for square)
    c1 = np.zeros((n, 2))
    idx = 0
    row = 0
    while idx < n:
        y = 0.05 + row * 0.1732
        if y > 0.95: break
        offset = 0.0866 if row % 2 == 1 else 0.0
        x = 0.05 + offset
        while x <= 0.95 and idx < n:
            c1[idx] = [x, y]
            idx += 1
            x += 0.1732
        row += 1
    for i in range(idx, n):
        c1[i] = [0.5 + (i-idx)*0.01, 0.5]
    configs.append(c1)
    
    # 2. Perturbed hex grid (helps find asymmetric boundary optimizations)
    np.random.seed(42)
    c2 = c1.copy() + np.random.uniform(-0.02, 0.02, size=c1.shape)
    configs.append(c2)
    
    # 3. 5x5 grid + center (alternative structure)
    c3 = np.zeros((n, 2))
    idx = 0
    for i in range(5):
        for j in range(5):
            c3[idx] = [0.1 + i*0.2, 0.1 + j*0.2]
            idx += 1
    c3[25] = [0.5, 0.5]
    configs.append(c3)
    
    for cfg in configs:
        radii_init = np.ones(n) * 0.06
        x0 = np.concatenate([cfg.flatten(), radii_init])
        
        cons = []
        for i in range(n):
            cons.append({'type': 'ineq', 'fun': cons_x_min, 'args': (i, n)})
            cons.append({'type': 'ineq', 'fun': cons_x_max, 'args': (i, n)})
            cons.append({'type': 'ineq', 'fun': cons_y_min, 'args': (i, n)})
            cons.append({'type': 'ineq', 'fun': cons_y_max, 'args': (i, n)})

            for j in range(i+1, n):
                cons.append({'type': 'ineq', 'fun': cons_dist, 'args': (i, j, n)})
                
        bnds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n
        
        try:
            res = minimize(obj_func, x0, args=(n,), method='SLSQP', bounds=bnds, constraints=cons, options={'maxiter': 3000, 'ftol': 1e-14})
            if -res.fun > best_sum:
                best_sum = -res.fun
                best_centers = res.x[:2*n].reshape(n, 2)
                best_radii = res.x[2*n:]
        except Exception:
            continue
            
    # Final validity check and clipping to guarantee constraints are met
    for i in range(n):
        r = best_radii[i]
        best_centers[i, 0] = np.clip(best_centers[i, 0], r, 1.0 - r)
        best_centers[i, 1] = np.clip(best_centers[i, 1], r, 1.0 - r)
        best_radii[i] = max(0.0, best_radii[i])
        
    return best_centers, best_radii, np.sum(best_radii)
