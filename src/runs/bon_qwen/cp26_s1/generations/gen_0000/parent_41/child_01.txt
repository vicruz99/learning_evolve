import numpy as np
from scipy.optimize import minimize

def compute_loss(vars, n, pen):
    """
    Computes the objective function: negative sum of radii + penalty for constraint violations.
    """
    centers = vars[:2*n].reshape(n, 2)
    radii = vars[2*n:]
    
    # Primary objective: maximize sum of radii
    loss = -np.sum(radii)
    
    # --- Boundary Violations ---
    x = centers[:, 0]
    y = centers[:, 1]
    r = radii
    
    # Squared max(0, violation) for smooth-ish penalty landscape
    v_bound = (np.maximum(0, r - x)**2 + 
               np.maximum(0, r - (1 - x))**2 + 
               np.maximum(0, r - y)**2 + 
               np.maximum(0, r - (1 - y))**2)
               
    # --- Pairwise Distance Violations ---
    # Compute distance matrix efficiently
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    dist = np.sqrt(dx**2 + dy**2)
    
    # Required minimum distance matrix
    min_dist = r[:, np.newaxis] + r[np.newaxis, :]
    
    # Penalty for overlapping circles
    v_pair = np.maximum(0, min_dist - dist)**2
    np.fill_diagonal(v_pair, 0.0)  # No self-interaction
    
    return loss + pen * (np.sum(v_bound) + np.sum(v_pair))


def run_packing():
    n = 26
    best_sum_r = -1.0
    best_vars = None
    
    # Multiple restarts to escape local minima
    np.random.seed(42)
    for _ in range(6):
        # Random initialization inside a safe margin
        centers = np.random.uniform(0.15, 0.85, (n, 2))
        radii = np.full(n, 0.03)
        v0 = np.concatenate([centers.flatten(), radii])
        
        # Bounds: centers in [0,1], radii in [1e-6, 0.5]
        bounds = [(0, 1)] * (2 * n) + [(1e-6, 0.5)] * n
        
        v = v0.copy()
        pen = 100.0
        
        # Adaptive penalty schedule
        for step in range(45):
            res = minimize(compute_loss, v, args=(n, pen), method='L-BFGS-B',
                           bounds=bounds, options={'maxiter': 1000, 'ftol': 1e-15})
            v = res.x
            pen *= 1.35
            
        # Evaluate true objective after optimization
        c_curr = v[:2*n].reshape(n, 2)
        r_curr = v[2*n:]
        current_sum = np.sum(r_curr)
        
        if current_sum > best_sum_r:
            best_sum_r = current_sum
            best_vars = v.copy()
            
    centers_opt = best_vars[:2*n].reshape(n, 2)
    radii_opt = best_vars[2*n:]
    
    # Safety post-processing: ensure strict validity within validation tolerance
    scale = 1.0
    max_iterations = 1000
    for _ in range(max_iterations):
        valid = True
        x = centers_opt[:, 0]
        y = centers_opt[:, 1]
        r = radii_opt
        
        # Check boundaries
        if np.any(x - r < -1e-10) or np.any(x + r > 1 + 1e-10) or \
           np.any(y - r < -1e-10) or np.any(y + r > 1 + 1e-10):
            valid = False
        else:
            # Check overlaps
            dx = centers_opt[:, 0, np.newaxis] - centers_opt[np.newaxis, :, 0]
            dy = centers_opt[:, 1, np.newaxis] - centers_opt[np.newaxis, :, 1]
            dist = np.sqrt(dx**2 + dy**2)
            min_dist = r[:, np.newaxis] + r[np.newaxis, :]
            if np.any(np.triu(min_dist - dist, k=1) > 1e-10):
                valid = False
                
        if valid:
            break
        scale *= 0.9998
        radii_opt *= scale
        
    return centers_opt, radii_opt, np.sum(radii_opt)
