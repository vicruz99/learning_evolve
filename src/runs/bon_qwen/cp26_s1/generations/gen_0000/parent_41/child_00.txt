import numpy as np
from scipy.optimize import minimize

def generate_initial_centers(n=26):

    centers = []
    
    # Hexagonal packing row heights
    # We need 26 circles. A 5-4-5-4-5-4 pattern sums to 27 (one too many).
    # Let's try 5-4-5-4-4-4 (sum 26) or similar.
    # Actually, let's just generate a grid of points and take the first 26.
    # Or a specific hex pattern.
    
    # Let's try 6 rows.
    rows = [5, 4, 5, 4, 4, 4] # Sum = 26
    # Check width/height scaling later.
    
    # Calculate approximate radius to fit in unit square for scaling
    # This is just for initial placement, optimizer will adjust.
    # Width for 5 circles ~ 10r, Height for 6 rows ~ 10r.
    # r approx 0.1.
    
    y = 0.1 # Initial bottom y
    row_idx = 0
    
    for count in rows:
        # x spacing 0.2 (approx 2r)
        # Center the row
        row_width = (count - 1) * 0.2
        x_start = (1.0 - row_width) / 2.0
        
        for i in range(count):
            x = x_start + i * 0.2
            centers.append([x, y])
        
        # Next row: shift x by 0.1 (hex shift), y by 0.1732 (sqrt(3)/2 * 0.2)
        y += 0.1732
        row_idx += 1
        
    return np.array(centers)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    # scipy minimizes, so we minimize negative sum.

    def objective(x_flat):
        radii = x_flat[2::3] # Extract radii: indices 2, 5, 8...
        # Objective: -sum(radii)
        obj = -np.sum(radii)
        
        # Regularization: penalize deviation from initial radii
        # This helps the solver stay in a "packed" region
        reg = 0.1 * np.sum((radii - r_init)**2)
        
        return obj + reg

    for i in range(n):

        constraints.append({
            'type': 'ineq',
            'fun': lambda x, i=i: x[i*3] - x[i*3 + 2]
        })
        # 1 - x - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda x, i=i: 1.0 - x[i*3] - x[i*3 + 2]
        })
        # y - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda x, i=i: x[i*3 + 1] - x[i*3 + 2]
        })

        constraints.append({
            'type': 'ineq',
            'fun': lambda x, i=i: 1.0 - x[i*3 + 1] - x[i*3 + 2]
        })

            constraints.append({

    return centers, radii, sum_radii
