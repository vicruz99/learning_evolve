import numpy as np
from scipy.optimize import minimize

N = 26

def get_hex_init():
    """Initialize circles in a hexagonal lattice pattern."""
    pts = []
    d = 0.21
    y = 0.08
    row = 0
    k = 0
    while k < N:
        x = 0.08 if row % 2 == 0 else 0.08 + d / 2
        while x < 0.92 and k < N:
            pts.append([x, y])
            x += d
            k += 1
        y += d * np.sqrt(3) / 2
        row += 1
    return np.array(pts)

def get_grid_init():
    """Initialize circles in a uniform grid pattern."""
    nc = 6
    nr = 5
    pts = np.zeros((N, 2))
    for i in range(N):
        c = i % nc
        r = i // nc
        pts[i, 0] = (c + 0.5) / nc
        pts[i, 1] = (r + 0.5) / nr
    return pts

def objective(vars):
    """Objective function: maximize sum of radii => minimize negative sum."""
    return -np.sum(vars[2::3])

def constraints(vars):
    """
    Evaluate all inequality constraints: g(vars) >= 0.
    Includes boundary constraints and pairwise non-overlap constraints.
    """
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    # Boundary constraints
    b = np.concatenate([
        x - r,
        1.0 - x - r,
        y - r,
        1.0 - y - r
    ])
    
    # Pairwise non-overlap constraints using broadcasting
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    d2 = dx**2 + dy**2
    
    rs = r[:, None] + r[None, :]
    diff = d2 - rs**2
    
    # Extract upper triangular part to avoid duplicates
    triu_idx = np.triu_indices(N, k=1)
    p = diff[triu_idx]
    
    return np.concatenate([b, p])

def run_packing():
    """
    Solves the circle packing optimization problem.
    Returns (centers, radii, sum_radii)
    """
    # Bounds for x, y in [0,1] and r in [0, 0.5]
    bounds = [(0.0, 1.0)] * (N * 2) + [(0.0, 0.5)] * N
    constr_dict = {'type': 'ineq', 'fun': constraints}
    
    # Multiple starting points to escape local minima
    inits = [get_hex_init(), get_grid_init()]
    
    rng = np.random.RandomState(42)
    rand_init = rng.rand(N, 2)
    inits.append(rand_init)
    
    best_sol = None
    best_sum = -1.0
    
    for init_pts in inits:
        # Start with small feasible radii
        r_init = np.full(N, 0.04)
        x0 = np.zeros(3 * N)
        x0[0::3] = init_pts[:, 0]
        x0[1::3] = init_pts[:, 1]
        x0[2::3] = r_init
        
        # Run optimization
        res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                       constraints=constr_dict, options={'maxiter': 1500, 'ftol': 1e-12})
        
        # Track best valid solution
        current_sum = -res.fun
        if current_sum > best_sum:
            best_sum = current_sum
            best_sol = res.x
            
    # Extract and clean results
    centers = np.zeros((N, 2))
    centers[:, 0] = best_sol[0::3]
    centers[:, 1] = best_sol[1::3]
    radii = best_sol[2::3]
    
    # Clamp to ensure strict validity against numerical drift
    radii = np.maximum(radii, 0.0)
    centers[:, 0] = np.clip(centers[:, 0], 0.0, 1.0)
    centers[:, 1] = np.clip(centers[:, 1], 0.0, 1.0)
    
    return centers, radii, best_sum
