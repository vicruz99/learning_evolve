import numpy as np
from scipy.optimize import minimize
from scipy.optimize import linprog

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    """
    n_circles = 26
    np.random.seed(42)

    # Stage 1: Initialize centers using a perturbed hexagonal lattice
    # Hexagonal packing has higher density than square packing
    rows = 6
    cols = 5
    centers = []
    
    # Hexagonal spacing parameters
    # We want to fit roughly 5-6 circles across and 4-5 vertically
    # Try to fit 26 circles in a staggered grid
    # Layout: 5, 4, 5, 4, 5, 3? No, let's just generate a dense cloud and let optimization handle it.
    # A simple hexagonal grid for 26 points
    
    x_spacing = 0.22
    y_spacing = 0.22 * np.sqrt(3)
    
    count = 0
    for r in range(8): # Enough rows
        for c in range(8): # Enough cols
            if count >= n_circles:
                break
            x = 0.1 + c * x_spacing
            y = 0.1 + r * y_spacing
            # Stagger odd rows
            if r % 2 == 1:
                x += x_spacing / 2
            if 0 <= x <= 1 and 0 <= y <= 1:
                centers.append([x, y])
                count += 1
        if count >= n_circles:
            break
            
    centers = np.array(centers[:n_circles])
    
    # Add random jitter to break symmetries
    centers += np.random.uniform(-0.01, 0.01, centers.shape)
    centers = np.clip(centers, 0.01, 0.99)

    def solve_radii_lp(c_list):
        """
        Solves LP to maximize sum of radii for fixed centers.
        Variables: r_0 ... r_{n-1}
        Maximize sum(r_i)
        Subject to:
          r_i <= x_i
          r_i <= 1 - x_i
          r_i <= y_i
          r_i <= 1 - y_i
          r_i + r_j <= dist(i, j)
          r_i >= 0
        """
        n = len(c_list)
        
        # Objective: maximize sum(r) => minimize -sum(r)
        c_obj = -np.ones(n)
        
        # Inequality constraints A_ub @ r <= b_ub
        # r_i <= x_i  =>  r_i <= x_i
        # r_i <= 1-x_i => r_i <= 1-x_i
        # ...
        # r_i + r_j <= dist
        
        # We will build A_ub and b_ub
        # Number of constraints: 4*n (bounds) + n*(n-1)/2 (distances)
        
        # To optimize, we can just use the bounds and distances.
        # However, building the full matrix for n=26 is large but manageable.
        # 4*26 + 325 = 429 constraints. 26 variables.
        
        A_ub = []
        b_ub = []
        
        # Boundary constraints
        for i in range(n):
            x, y = c_list[i]
            # r_i <= x
            row = np.zeros(n)
            row[i] = 1
            A_ub.append(row)
            b_ub.append(x)
            # r_i <= 1-x
            row = np.zeros(n)
            row[i] = 1
            A_ub.append(row)
            b_ub.append(1-x)
            # r_i <= y
            row = np.zeros(n)
            row[i] = 1
            A_ub.append(row)
            b_ub.append(y)
            # r_i <= 1-y
            row = np.zeros(n)
            row[i] = 1
            A_ub.append(row)
            b_ub.append(1-y)
            
        # Distance constraints
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.linalg.norm(c_list[i] - c_list[j])
                row = np.zeros(n)
                row[i] = 1
                row[j] = 1
                A_ub.append(row)
                b_ub.append(dist)
                
        A_ub = np.array(A_ub)
        b_ub = np.array(b_ub)
        
        # Bounds for r_i: [0, None]
        bounds = [(0, None) for _ in range(n)]
        
        # Solve LP
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        
        if res.success:
            return res.x
        else:
            # Fallback: simple projection
            r = np.ones(n) * 0.001
            # Just return small radii to avoid failure
            return r

    # Stage 2: Optimize centers
    # We want to maximize the sum of radii returned by the LP.
    # Since linprog is not differentiable everywhere, we use a direct search or gradient-free method.
    # However, for speed, we can use a repulsive force heuristic or Nelder-Mead on a subsample?
    # Nelder-Mead on 52 dimensions is slow.
    # Let's try a simpler iterative relaxation:
    # Compute radii, then move centers apart based on tight constraints.
    
    # Better approach for code execution time: 
    # Use a few iterations of "expansion".
    
    current_centers = centers.copy()
    
    # Helper to get current objective
    def get_obj(cents):
        r = solve_radii_lp(cents)
        return -np.sum(r), r # Return negative for minimization

    # Simple iterative improvement:
    # 1. Get radii
    # 2. Identify pairs that are touching (dist approx r_i + r_j)
    # 3. Push them apart slightly
    # This is essentially a local search.
    
    for step in range(20): # 20 iterations of refinement
        r = solve_radii_lp(current_centers)
        if np.any(np.isnan(r)):
            break
            
        # Calculate forces
        forces = np.zeros_like(current_centers)
        n = len(current_centers)
        
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.linalg.norm(current_centers[i] - current_centers[j])
                sum_r = r[i] + r[j]
                if dist < sum_r + 1e-6:
                    # Overlap or touching, push apart
                    if dist > 0:
                        direction = (current_centers[i] - current_centers[j]) / dist
                    else:
                        direction = np.random.randn(2)
                    # Force magnitude proportional to overlap
                    overlap = max(0, sum_r - dist)
                    # Also add a repulsion if very close to limit to explore space
                    repulsion = 0.001 
                    force_mag = overlap + repulsion
                    forces[i] += direction * force_mag
                    forces[j] -= direction * force_mag
            
            # Boundary forces: push away from walls if touching
            x, y = current_centers[i]
            rx = r[i]
            if x - rx < 1e-6: forces[i, 0] += 0.001
            if x + rx > 1 - 1e-6: forces[i, 0] -= 0.001
            if y - rx < 1e-6: forces[i, 1] += 0.001
            if y + rx > 1 - 1e-6: forces[i, 1] -= 0.001

        # Update centers
        current_centers += forces * 0.5 # Step size
        current_centers = np.clip(current_centers, 0.01, 0.99)

    # Final LP solve to get valid radii
    final_radii = solve_radii_lp(current_centers)
    
    # Clip any negative or NaN radii just in case
    final_radii = np.maximum(final_radii, 0)
    
    # Validate and return
    # Check if valid, if not, fallback to grid
    try:
        if validate_packing(current_centers, final_radii):
            return current_centers, final_radii, np.sum(final_radii)
    except:
        pass

    # Fallback: 5x5 grid with one small circle
    # This ensures we return *something* valid

    radii_fb = np.ones(25) * 0.1
    # 26th circle in center gap? 
    # Gap at (0.2, 0.2) is center of (0.1,0.1)... no.
    # Center of square (0.5, 0.5) is occupied.
    # Let's put 26th at (0.2, 0.3) with small radius
    centers_fb = np.vstack([centers_fb, [0.25, 0.25]]) # Just a guess
    radii_fb = np.append(radii_fb, 0.01)
    
    # Re-adjust to make valid
    # Simple valid packing: 6x5 grid of very small circles
    centers_fb = np.array([[i*0.2+0.1, j*0.16+0.08] for i in range(5) for j in range(6)])
    radii_fb = np.ones(30) * 0.04 # 30 circles, take 26
    centers_fb = centers_fb[:26]
    radii_fb = radii_fb[:26]
    
    return centers_fb, radii_fb, np.sum(radii_fb)

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square

    Args:
        centers: np.array of shape (n, 2) with (x, y) coordinates
        radii: np.array of shape (n) with radius of each circle

    Returns:
        True if valid, False otherwise
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True

# Run the packing
