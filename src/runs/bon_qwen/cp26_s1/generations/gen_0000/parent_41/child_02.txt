import numpy as np

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square

    Args:
        centers: np.array of shape (n, 2) with (x, y) coordinates
        radii: np.array of shape (n) with radius of each circle

    Returns:
        True if valid, False otherwise
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True

def get_constraints(centers, radii):
    """
    Generates a list of constraint dictionaries for scipy.optimize
    """
    n = centers.shape[0]
    constraints = []

    # Boundary constraints: x - r >= 0, 1 - (x + r) >= 0, same for y
    for i in range(n):
        # x - r >= 0
        def constr_x_min(x_var, idx=i):
            return x_var[idx*2] - x_var[n + idx]
        constraints.append({'type': 'ineq', 'fun': constr_x_min})

        # 1 - (x + r) >= 0
        def constr_x_max(x_var, idx=i):
            return 1.0 - (x_var[idx*2] + x_var[n + idx])
        constraints.append({'type': 'ineq', 'fun': constr_x_max})

        # y - r >= 0
        def constr_y_min(x_var, idx=i):
            return x_var[idx*2 + 1] - x_var[n + idx]
        constraints.append({'type': 'ineq', 'fun': constr_y_min})

        # 1 - (y + r) >= 0
        def constr_y_max(x_var, idx=i):
            return 1.0 - (x_var[idx*2 + 1] + x_var[n + idx])
        constraints.append({'type': 'ineq', 'fun': constr_y_max})

        # r >= 0
        def constr_r_nonneg(x_var, idx=i):
            return x_var[n + idx]
        constraints.append({'type': 'ineq', 'fun': constr_r_nonneg})

    # Non-overlap constraints: dist >= r_i + r_j
    for i in range(n):
        for j in range(i + 1, n):
            def constr_overlap(x_var, i=i, j=j):
                xi, yi = x_var[i*2], x_var[i*2+1]
                xj, yj = x_var[j*2], x_var[j*2+1]
                ri, rj = x_var[n+i], x_var[n+j]
                dist_sq = (xi - xj)**2 + (yi - yj)**2
                # We want dist >= ri + rj  =>  dist^2 >= (ri + rj)^2
                # But dist^2 - (ri+rj)^2 can be tricky numerically if dist is small.
                # However, for inequality constraint fun >= 0, this works.
                return dist_sq - (ri + rj)**2
            constraints.append({'type': 'ineq', 'fun': constr_overlap})

    return constraints
