import numpy as np
from scipy.optimize import minimize

def run_packing():
    """
    Packs 26 circles in a unit square [0,1]x[0,1] to maximize sum of radii.
    """
    np.random.seed(42)
    n = 26

    # ---------------------------------------------------------
    # 1. Initialization: Hybrid Hexagonal Layout
    # ---------------------------------------------------------
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.095)
    
    idx = 0
    # Row parameters for a dense 5-row arrangement
    rows = [5, 5, 6, 5, 5] 
    y_coords = [0.1, 0.275, 0.45, 0.625, 0.8]
    
    for i, count in enumerate(rows):
        if idx >= n: break
        # Calculate row x-spacing
        width = 1.0
        gap = width / (count + 1) 
        x_start = gap
        for j in range(count):
            if idx >= n: break
            x = x_start + j * (2 * gap)
            # Add slight jitter to prevent perfect symmetry from trapping optimizer
            centers[idx, 0] = x + (np.random.rand() - 0.5) * 0.005
            centers[idx, 1] = y_coords[i] + (np.random.rand() - 0.5) * 0.005
            idx += 1

    # Fill any remaining if needed (though 26 should be covered)
    while idx < n:
        centers[idx] = np.random.rand(2)
        idx += 1

    # ---------------------------------------------------------
    # 2. Heuristic Optimization: Repulsive Dynamics
    # ---------------------------------------------------------
    dt = 0.05
    growth = 0.0005
    
    # Run dynamics for a fixed number of steps
    for step in range(1500):
        # Increase radii
        radii += growth
        
        # Calculate forces
        forces = np.zeros_like(centers)
        
        # Inter-circle repulsion
        # O(N^2) is fine for N=26
        for i in range(n):
            for j in range(i + 1, n):
                diff = centers[i] - centers[j]
                dist_sq = np.dot(diff, diff)
                dist = np.sqrt(dist_sq)
                
                min_dist = radii[i] + radii[j]
                if dist < min_dist and dist > 1e-9:
                    # Repulsive force
                    overlap = min_dist - dist
                    # Force magnitude proportional to overlap, inversely proportional to distance
                    f_mag = (overlap ** 2) / (dist + 1e-9)
                    dir_norm = diff / dist
                    forces[i] += dir_norm * f_mag
                    forces[j] -= dir_norm * f_mag

        # Boundary repulsion
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            
            if x - r < 0: forces[i, 0] += (r - x) * 10
            if x + r > 1: forces[i, 0] -= (x + r - 1) * 10
            if y - r < 0: forces[i, 1] += (r - y) * 10
            if y + r > 1: forces[i, 1] -= (y + r - 1) * 10

        # Update positions
        centers += dt * forces
        dt *= 0.99 # Decay step size
        growth *= 0.99 # Decay growth rate

    # ---------------------------------------------------------
    # 3. Numerical Refinement: Augmented Lagrangian
    # ---------------------------------------------------------
    def objective(variables):
        r_vec = variables[:n]
        c_vec = variables[n:]
        
        # Negative sum of radii
        return -np.sum(r_vec)

    def constraint_penalty(variables):
        r_vec = variables[:n]
        c_vec = variables[n:].reshape((n, 2))
        penalty = 0.0
        
        # Boundary constraints
        for i in range(n):
            r = r_vec[i]
            x, y = c_vec[i]
            # Lower bounds
            penalty += max(0, r - x)**2
            penalty += max(0, r - y)**2
            # Upper bounds
            penalty += max(0, x + r - 1)**2
            penalty += max(0, y + r - 1)**2
            
        # Overlap constraints
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((c_vec[i] - c_vec[j])**2))
                min_dist = r_vec[i] + r_vec[j]
                if dist < min_dist:
                    penalty += (min_dist - dist)**2
                    
        return penalty

    # Combine objective and penalty
    def combined_function(variables, lambda_pen):
        return objective(variables) + lambda_pen * constraint_penalty(variables)

    # Flatten initial state
    x0 = np.concatenate([radii, centers.flatten()])
    
    # Optimization parameters
    bounds_r = [(0, 0.5) for _ in range(n)]
    bounds_c = [(0, 1) for _ in range(2 * n)]
    bounds = bounds_r + bounds_c

    # Run optimization with increasing penalty
    current_penalty = 10.0
    best_res = minimize(combined_function, x0, args=(current_penalty,), method='L-BFGS-B', bounds=bounds)
    
    # Refine with higher penalty
    for i in range(2):
        current_penalty *= 5
        best_res = minimize(combined_function, best_res.x, args=(current_penalty,), method='L-BFGS-B', bounds=bounds)

    # Extract final results
    final_r = best_res.x[:n]
    final_c = best_res.x[n:].reshape((n, 2))
    
    # Final validation and NaN handling
    if np.isnan(final_r).any() or np.isnan(final_c).any():
        # Fallback to heuristic result if optimization fails
        final_r = radii
        final_c = centers

    return final_c, final_r, float(np.sum(final_r))

import numpy as np
def validate_packing(centers, radii):
    n = centers.shape[0]
    if np.isnan(centers).any() or np.isnan(radii).any(): return False
    for i in range(n):
        if radii[i] < 0: return False
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            return False
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False
    return True

is_valid = validate_packing(final_centers, final_radii)
