from scipy.optimize import minimize, basinhopping, differential_evolution

def get_max_radii(centers):
    """
    Calculates the maximum valid radius for each circle given fixed centers.
    Radius is limited by the distance to other circles and the unit square boundaries.
    """
    n = centers.shape[0]
    radii = np.full(n, np.inf)

    # Boundary constraints: circle must be inside [0,1]x[0,1]
    # r <= x, r <= 1-x, r <= y, r <= 1-y
    radii = np.minimum(radii, centers[:, 0])
    radii = np.minimum(radii, 1.0 - centers[:, 0])
    radii = np.minimum(radii, centers[:, 1])
    radii = np.minimum(radii, 1.0 - centers[:, 1])

    # Overlap constraints: r_i + r_j <= dist(i, j) => r_i <= dist(i, j) - r_j
    # To find the max r_i, we iterate. For equal radii approximation:
    # r <= dist(i, j) / 2
    # For precise max radii with fixed centers, we solve the system or use a simple bound.
    # A tight bound for r_i is min(dist(i,j) - r_j).
    # However, for a static score function, we can assume the "bottleneck" is half distance.
    # A more accurate approach for fixed centers is to assume neighbors have some radius.
    # Let's use the half-distance as a robust upper bound.
    for i in range(n):
        for j in range(n):
            if i == j: continue
            dist = np.linalg.norm(centers[i] - centers[j])
            radii[i] = min(radii[i], dist / 2.0)
            
    return radii

    # We want to maximize sum of radii.
