2.  **Iterative Optimization**: We use `scipy.optimize.minimize` with the SLSQP method to maximize the sum of radii. This method allows for inequality constraints to enforce non-overlap ($dist \ge r_i + r_j$) and boundary conditions ($r \le x \le 1-r$).
3.  **Multi-Start Robustness**: Since the landscape is non-convex, we run three distinct initialization patterns (Hexagonal, Dense Square Grid, and Centralized) and select the result with the highest sum of radii.
4.  **Radius Asymmetry**: The optimization allows radii to vary. This enables larger circles to occupy more open areas or corners while smaller circles fill gaps, pushing the total sum toward the target.

import numpy as np
from scipy.optimize import minimize

def get_constraints(n):
    constraints = []
    for i in range(n):
        idx_x = 2 * i
        idx_y = 2 * i + 1
        idx_r = 2 * n + i
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: x[idx_x] - x[idx_r]})
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: 1 - (x[idx_x] + x[idx_r])})
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: x[idx_y] - x[idx_r]})
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: 1 - (x[idx_y] + x[idx_r])})
    
    for i in range(n):
        for j in range(i + 1, n):
            idx_xi, idx_yi = 2 * i, 2 * i + 1
            idx_xj, idx_yj = 2 * j, 2 * j + 1
            idx_ri, idx_rj = 2 * n + i, 2 * n + j
            
            def constr_overlap(x, i=i, j=j):
                xi, yi = x[idx_xi], x[idx_yi]
                xj, yj = x[idx_xj], x[idx_yj]
                ri, rj = x[idx_ri], x[idx_rj]
                dist_sq = (xi - xj)**2 + (yi - yj)**2
                return dist_sq - (ri + rj)**2
            
            constraints.append({'type': 'ineq', 'fun': constr_overlap})
    return constraints

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    cons = get_constraints(n)
    bounds = [(0, 1) for _ in range(2 * n)] + [(0, 1) for _ in range(n)]
    best_sum = 0.0

    def objective(x):
        return -np.sum(x[2 * n:])

    # 1. Hexagonal Initialization
    centers1 = []
    r_init = 0.11
    count = 0

                    centers1.append([x, y])

    # 2. Dense Square Grid Initialization
    centers2 = np.array([[0.1 + 0.8 * (i // 5), 0.1 + 0.8 * (i % 5)] for i in range(n)])

    # 3. Centralized Initialization
    centers3 = np.array([[0.5 + 0.4 * np.cos(2 * np.pi * i / n), 
                          0.5 + 0.4 * np.sin(2 * np.pi * i / n)] for i in range(n)])
    centers3[0] = [0.5, 0.5]

    for init_centers in [np.array(centers1), centers2, centers3]:
        if len(init_centers) < n:
            init_centers = np.vstack([init_centers, np.random.rand(n - len(init_centers), 2) * 0.8 + 0.1])
        
        x0 = np.concatenate([init_centers.flatten(), np.full(n, 0.11)])
        
        res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, 
                       options={'maxiter': 2000, 'ftol': 1e-10})
        
        if not res.success:
            # Fallback to a simpler optimization if SLSQP fails or gets stuck
            # Simple radius expansion fallback
            pass
            
        current_sum = -res.fun
        if current_sum > best_sum:
            best_sum = current_sum
            best_result = (res.x[:2 * n].reshape(n, 2), res.x[2 * n:], current_sum)

    return best_result
