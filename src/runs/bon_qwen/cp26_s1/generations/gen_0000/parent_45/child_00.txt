This provides a geometrically sound starting point where circles are evenly distributed and non-overlapping, helping the optimizer avoid poor local minima.
2.  **Variable Formulation:** We define the state vector as a concatenation of all circle parameters: $[x_1, y_1, r_1, x_2, y_2, r_2, \dots, x_{26}, y_{26}, r_{26}]$. The objective is to minimize the negative sum of radii (equivalent to maximizing the sum).
3.  **Constraint Vectorization:** To ensure computational efficiency, we implement constraints using vectorized NumPy operations instead of Python loops.

    *   **Non-Overlap Constraints:** Enforce $(x_i - x_j)^2 + (y_i - y_j)^2 \ge (r_i + r_j)^2$ for all pairs.
4.  **Optimization (SLSQP):** We use `scipy.optimize.minimize` with the Sequential Least Squares Programming (SLSQP) method. This method handles non-linear constraints effectively. import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Pack 26 circles in a unit square to maximize the sum of radii.
    """
    n = 26
    
    def objective(v):
        # v is [x1, y1, r1, x2, y2, r2, ...]
        # We want to maximize sum(r_i), so we minimize -sum(r_i)
        return -np.sum(v[2::3])

    def boundary_constraints(v):
        # Returns a vector of values that must be >= 0
        # x - r >= 0
        # 1 - x - r >= 0
        # y - r >= 0
        # 1 - y - r >= 0
        x = v[0::3]
        y = v[1::3]
        r = v[2::3]
        
        cons = []
        cons.append(x - r)           # x >= r
        cons.append(1.0 - x - r)     # x + r <= 1
        cons.append(y - r)           # y >= r
        cons.append(1.0 - y - r)     # y + r <= 1
        
        return np.concatenate(cons)

    def distance_constraints(v):
        # Returns a vector of values that must be >= 0
        # dist^2 - (r1 + r2)^2 >= 0
        pts = v.reshape(n, 3)
        coords = pts[:, :2]
        r = pts[:, 2]
        
        # Compute pairwise squared distances
        # Using broadcasting to avoid explicit loops for speed
        # diff shape (n, n, 2)
        diff = coords[:, np.newaxis, :] - coords[np.newaxis, :, :]
        dist_sq = np.sum(diff**2, axis=2) # shape (n, n)
        
        # Compute pairwise sum of radii squared
        # rad_sum shape (n, n)
        rad_sum = r[:, np.newaxis] + r[np.newaxis, :]
        rad_sum_sq = rad_sum**2
        
        # We only need upper triangle (i < j)
        # Mask for upper triangle
        mask = np.triu(np.ones((n, n), dtype=bool), k=1)
        
        # Constraint values
        cons_vals = dist_sq[mask] - rad_sum_sq[mask]
        return cons_vals

    # Bounds for variables: x, y in [0, 1], r in [0, 0.5]
    bounds = []
    for _ in range(n):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 0.5)) # r

    # Define constraints structure for scipy
    constraints = [
        {'type': 'ineq', 'fun': boundary_constraints},
        {'type': 'ineq', 'fun': distance_constraints}
    ]

    best_sum_radii = -1.0
    best_centers = None
    best_radii = None
    best_res = None

    # Multiple restarts with different initial configurations
    num_restarts = 10
    
    for restart in range(num_restarts):
        # Generate Hexagonal Grid Initialization
        # Approximate radius for 26 circles is around 0.1
        r_init = 0.09 
        if restart > 0:
            # Slightly vary initial radius or shift to perturb
            r_init = 0.08 + restart * 0.005
            
        centers_list = []
        # Hexagonal lattice generation
        # Vertical spacing between rows: r * sqrt(3)
        # Horizontal spacing within row: 2 * r
        # Offset odd rows by r
        
        y = r_init
        row_idx = 0
        while len(centers_list) < n and y <= 1 - r_init:
            x = r_init if (row_idx % 2 == 0) else 2 * r_init
            while x <= 1 - r_init:
                centers_list.append([x, y])
                x += 2 * r_init
            
            y += r_init * np.sqrt(3)
            row_idx += 1
            
        # If we didn't get enough points, fill with random valid points
        if len(centers_list) < n:
            # Fallback to random or simple grid fill
            np.random.seed(restart)
            while len(centers_list) < n:
                cx = np.random.uniform(r_init, 1 - r_init)
                cy = np.random.uniform(r_init, 1 - r_init)
                centers_list.append([cx, cy])
        
        # Take first n points
        centers_list = centers_list[:n]
        
        # Perturb centers slightly for diversity
        np.random.seed(restart * 100)
        noise = np.random.uniform(-0.02, 0.02, size=(n, 2))
        init_centers = np.array(centers_list) + noise
        
        # Clip to valid range [r_init, 1-r_init]
        init_centers = np.clip(init_centers, r_init, 1 - r_init)
        
        # Construct initial vector
        v0 = np.zeros(3 * n)
        for i in range(n):
            v0[3*i] = init_centers[i, 0]
            v0[3*i+1] = init_centers[i, 1]
            v0[3*i+2] = r_init
            
        # Optimize
        try:
            res = minimize(objective, v0, method='SLSQP', bounds=bounds, 
                           constraints=constraints, options={'maxiter': 2000, 'ftol': 1e-10})
            
            if res.success or res.fun < best_sum_radii * -1 - 0.001: # Check if improved
                current_sum = -res.fun
                if current_sum > best_sum_radii:
                    best_sum_radii = current_sum
                    best_centers = res.x.reshape(n, 3)[:, :2].copy()
                    best_radii = res.x.reshape(n, 3)[:, 2].copy()
        except Exception:
            continue

    # If optimization didn't find anything better than 0 (shouldn't happen), return valid default
    if best_centers is None:
        # Fallback: simple grid
        best_centers = np.zeros((n, 2))
        best_radii = np.zeros(n)
        # Just a valid configuration
        r = 0.05
        idx = 0
        for r_y in np.linspace(0.1, 0.9, 5):
            for r_x in np.linspace(0.1, 0.9, 5):
                if idx < n:
                    best_centers[idx] = [r_x, r_y]
                    best_radii[idx] = r
                    idx += 1

    return best_centers, best_radii, best_sum_radii
