import numpy as np

def compute_forces(centers, radii, N):
    """Compute repulsive forces between circles and boundary forces."""
    force = np.zeros_like(centers)
    for i in range(N):
        for j in range(i + 1, N):
            diff = centers[i] - centers[j]
            d = np.sqrt(np.sum(diff ** 2))
            if d < 1e-7:
                continue
            overlap = radii[i] + radii[j] - d
            if overlap > 0:
                f = overlap * diff / d
                force[i] += f
                force[j] -= f
        
        # Boundary repulsion
        for ax in range(2):
            if centers[i, ax] < radii[i]:
                force[i, ax] += radii[i] - centers[i, ax]
            if centers[i, ax] > 1.0 - radii[i]:
                force[i, ax] -= centers[i, ax] - (1.0 - radii[i])
    return force

def update_radii(centers, radii, N):
    """Update each circle's radius to be the maximum feasible given current positions."""
    for i in range(N):
        # Distance to boundaries
        mr = min(centers[i, 0], 1.0 - centers[i, 0], 
                 centers[i, 1], 1.0 - centers[i, 1])
        # Distance to other circles
        for j in range(N):
            if i != j:
                d = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                mr = min(mr, d - radii[j])
        radii[i] = max(0.0, mr)
    return radii
