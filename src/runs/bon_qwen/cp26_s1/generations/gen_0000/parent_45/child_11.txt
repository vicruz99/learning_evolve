import numpy as np
from scipy.optimize import minimize, differential_evolution
import math

N_CIRCLES = 26
PENALTY = 1e6

def compute_violation(centers, radii):
    n = centers.shape[0]
    violation = 0.0
    
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if r > x:
            violation += (r - x) ** 2
        if r > (1 - x):
            violation += (r - (1 - x)) ** 2
        if r > y:
            violation += (r - y) ** 2
        if r > (1 - y):
            violation += (r - (1 - y)) ** 2
    
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist_sq = dx * dx + dy * dy
            min_dist = radii[i] + radii[j]
            if dist_sq < min_dist * min_dist:
                dist = math.sqrt(dist_sq)
                violation += (min_dist - dist) ** 2
    
    return violation

def objective(vars):
    centers = vars[:2 * N_CIRCLES].reshape(N_CIRCLES, 2)
    radii = vars[2 * N_CIRCLES:]
    violation = compute_violation(centers, radii)
    return -np.sum(radii) + PENALTY * violation

def init_hexagonal():
    n = N_CIRCLES
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.095)
    
    row_counts = [5, 6, 5, 6, 4]
    idx = 0
    for row in range(5):
        width = row_counts[row]
        y = 0.09 + row * 0.185
        x_start = 0.09 + (row % 2) * 0.105
        for col in range(width):
            if idx < n:
                centers[idx] = [x_start + col * 0.195, y]
                idx += 1
    
    return centers, radii

def init_random(seed):
    np.random.seed(seed)
    n = N_CIRCLES
    centers = np.random.uniform(0.12, 0.88, (n, 2))
    radii = np.full(n, 0.09)
    return centers, radii

def optimize_from_init(centers, radii, max_iter=20000):
    n = N_CIRCLES
    x0 = np.concatenate([centers.flatten(), radii])
    
    bounds = [(0.005, 0.995)] * (2 * n) + [(0.005, 0.5)] * n
    
    result = minimize(
        objective,
        x0,
        bounds=bounds,
        method='L-BFGS-B',
        options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-10}
    )
    
    centers_opt = result.x[:2 * n].reshape(n, 2)
    radii_opt = result.x[2 * n:]
    
    # Clip radii to ensure non-negativity
    radii_opt = np.maximum(radii_opt, 1e-8)
    
    return centers_opt, radii_opt, np.sum(radii_opt)

def de_objective(vars):
    return objective(vars)

def run_packing():

    best_sum = -1
    best_centers = None
    best_radii = None

        radii_de = result_de.x[2 * n:]

        centers_polished, radii_polished, sum_polished = optimize_from_init(centers_de, radii_de)
