import numpy as np
from scipy.optimize import minimize

def compute_penalty(centers, radii, n):
    """Compute smooth penalty for overlaps and boundary violations."""
    diffs = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    dists[np.diag_indices_from(dists)] = 2.0
    overlaps = radii[:, None] + radii[None, :] - dists
    pen = np.sum(np.maximum(overlaps, 0)**2)
    
    pen += np.sum(np.maximum(radii - centers[:, 0], 0)**2)
    pen += np.sum(np.maximum(radii - (1.0 - centers[:, 0]), 0)**2)
    pen += np.sum(np.maximum(radii - centers[:, 1], 0)**2)
    pen += np.sum(np.maximum(radii - (1.0 - centers[:, 1]), 0)**2)
    return pen

def objective_function(vars, n):
    """Objective: maximize sum of radii with penalty for violations."""
    centers = vars[:n*2].reshape(n, 2)
    radii = vars[n*2:]
    radii = np.maximum(radii, 1e-5)
    return -np.sum(radii) + 1e4 * compute_penalty(centers, radii, n)

def run_packing():
    np.random.seed(42)
    n = 26
    
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    for trial in range(5):
        # 1. Initialize on perturbed hexagonal grid
        centers = np.zeros((n, 2))
        idx = 0
        spacing = 0.22
        for i in range(6):
            for j in range(6):
                if idx >= n:
                    break
                x = 0.1 + i * spacing + (j % 2) * spacing / 2.0
                y = 0.1 + j * spacing * 0.866
                if x < 1.0 and y < 1.0:
                    centers[idx] = [x + np.random.uniform(-0.005, 0.005), 
                                    y + np.random.uniform(-0.005, 0.005)]
                    idx += 1
            if idx >= n:
                break
                
        centers = np.clip(centers, 0.01, 0.99)
        radii = np.full(n, 0.08) + np.random.uniform(-0.005, 0.005, n)
        radii = np.maximum(radii, 0.01)
        
        # 2. Force-Directed Relaxation with radius expansion
        dt = 0.02
        k_rep = 500.0
        expansion = 0.001
        
        for step in range(1500):
            diffs = centers[:, None, :] - centers[None, :, :]
            dists = np.sqrt(np.sum(diffs**2, axis=2))
            dists[np.diag_indices_from(dists)] = 2.0
            
            overlaps = radii[:, None] + radii[None, :] - dists
            overlaps = np.maximum(overlaps, 0)
            
            inv_dists = 1.0 / dists
            inv_dists[np.diag_indices_from(inv_dists)] = 0.0
            dirs = diffs * inv_dists[:, :, None]
            
            forces = k_rep * np.sum(overlaps[:, :, None] * dirs, axis=1)
            
            forces[:, 0] += k_rep * np.maximum(radii - centers[:, 0], 0)
            forces[:, 0] -= k_rep * np.maximum(centers[:, 0] + radii - 1.0, 0)
            forces[:, 1] += k_rep * np.maximum(radii - centers[:, 1], 0)
            forces[:, 1] -= k_rep * np.maximum(centers[:, 1] + radii - 1.0, 0)
            
            centers += forces * dt
            centers = np.clip(centers, 0.001, 0.999)
            
            # Decaying expansion pressure
            radii += expansion * max(0.0, 1.0 - step / 1000.0)
            
        # 3. Gradient-Based Refinement
        x0 = np.concatenate([centers.flatten(), radii])
        try:
            res = minimize(objective_function, x0, args=(n,), 
                           method='L-BFGS-B', options={'maxiter': 3000, 'ftol': 1e-12})
            centers = res.x[:n*2].reshape(n, 2)
            radii = res.x[n*2:]
        except Exception:
            pass
            
        # 4. Final Boundary Compliance
        for i in range(n):
            r = radii[i]
            centers[i, 0] = np.clip(centers[i, 0], r, 1.0 - r)
            centers[i, 1] = np.clip(centers[i, 1], r, 1.0 - r)
            
        # Check validity and update best
        pen = compute_penalty(centers, radii, n)
        if pen < 1e-6:
            s = np.sum(radii)
            if s > best_sum:
                best_sum = s
                best_centers = centers.copy()
                best_radii = radii.copy()
                
    # Fallback if no valid config found (should not happen with robust init)
    if best_centers is None:
        best_centers = centers
        best_radii = radii
        best_sum = np.sum(radii)
        
    return best_centers, best_radii, best_sum
