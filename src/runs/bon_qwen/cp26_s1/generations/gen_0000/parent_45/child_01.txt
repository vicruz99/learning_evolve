import numpy as np
from scipy.optimize import minimize

def compute_penalty(centers_flat, r):
    """Compute penalty for boundary violations and overlaps for fixed radius r."""
    centers = centers_flat.reshape(26, 2)
    n = 26
    
    # Boundary penalties: ensure r <= x <= 1-r and r <= y <= 1-r
    x, y = centers[:, 0], centers[:, 1]
    p = np.sum(np.maximum(0, r - x)**2)
    p += np.sum(np.maximum(0, r - y)**2)
    p += np.sum(np.maximum(0, x - (1 - r))**2)
    p += np.sum(np.maximum(0, y - (1 - r))**2)
    
    # Overlap penalties: ensure dist(i, j) >= 2r
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist = np.sqrt(np.sum(diff**2, axis=2))
    # Ignore self-distance by making diagonal large
    dist = dist + np.eye(n) * 2 * r
    
    overlap = np.maximum(0, 2 * r - dist)
    p += np.sum(overlap**2)
    
    return p

def optimize_for_radius(r, trial_seed=0):
    """Optimize centers for a fixed radius r using multiple random restarts."""
    n = 26
    best_viol = float('inf')
    best_pos = None
    
    for t in range(25):
        np.random.seed(42 + trial_seed + t)
        pts = []
        y = 0.05
        # Generate hexagonal grid points
        while len(pts) < n:
            x = 0.05
            shift = (int(y / 0.08) % 2) * 0.06
            while x < 0.95 and len(pts) < n:
                pts.append([x + shift, y])
                x += 0.12
            y += 0.08 * np.sqrt(3)/2
            
        centers = np.array(pts[:n])
        # Add random perturbation to break symmetry
        centers += np.random.uniform(-0.015, 0.015, centers.shape)
        centers = np.clip(centers, 0.02, 0.98)
        
        # Optimize centers
        res = minimize(
            compute_penalty,
            centers.flatten(),
            args=(r,),
            method='L-BFGS-B',
            bounds=[(0.0, 1.0)]*(2*n),
            options={'maxiter': 500}
        )
        
        if res.fun < best_viol:
            best_viol = res.fun
            best_pos = res.x.copy()
            
    return best_pos.reshape(26, 2), best_viol

def run_packing():
    n = 26
    r_low, r_high = 0.095, 0.108
    
    # Bisection to find maximum feasible equal radius
    for _ in range(50):
        r_mid = (r_low + r_high) / 2
        _, viol = optimize_for_radius(r_mid, trial_seed=int(r_mid * 10000))
        if viol < 1e-8:
            r_low = r_mid
        else:
            r_high = r_mid
            
    final_r = r_low
    final_centers, _ = optimize_for_radius(final_r, trial_seed=12345)
    final_radii = np.full(n, final_r)
    
    # Shrink slightly to guarantee strict validity against floating point errors
    final_radii *= 0.99995
    final_sum = float(np.sum(final_radii))
    
    return final_centers, final_radii, final_sum
