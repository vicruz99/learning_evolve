import numpy as np
from scipy.optimize import linprog, differential_evolution

def compute_sum_radii(centers):
    """
    Given fixed centers, solve an LP to find maximum radii satisfying 
    non-overlap and boundary constraints. Returns (sum_radii, radii_array).
    """
    n = centers.shape[0]
    # Maximum radius allowed by boundaries for each circle
    b_ub = np.min(np.vstack([
        centers[:, 0], 1 - centers[:, 0],
        centers[:, 1], 1 - centers[:, 1]
    ]), axis=0)
    b_ub = np.maximum(b_ub, 0.0)
    
    # Build LP constraint matrix: A_ub @ r <= b_ub
    # Constraints: r_i + r_j <= d_ij  (for all pairs)
    #              r_i <= b_i         (for all i)
    n_pairs = n * (n - 1) // 2
    A_ub = np.zeros((n_pairs + n, n))
    b_ineq = np.zeros(n_pairs + n)
    
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            d = np.sqrt(dx*dx + dy*dy)
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ineq[idx] = d
            idx += 1
            
    for i in range(n):
        A_ub[idx, i] = 1.0
        b_ineq[idx] = b_ub[i]
        idx += 1
        
    # Maximize sum(r) => minimize -sum(r)
    res = linprog(c=-np.ones(n), A_ub=A_ub, b_ub=b_ineq, 
                  bounds=[(0, None)]*n, method='highs')
    
    if res.success:
        return -res.fun, res.x
    # Fallback if LP fails (should not happen with feasible centers)
    return 0.0, np.zeros(n)

def objective(x):
    """Objective function for optimization: negative sum of radii."""
    centers = x.reshape((26, 2))
    s, _ = compute_sum_radii(centers)
    return -s

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    bounds = [(0.0, 1.0)] * (2 * n)
    
    # Global optimization to find optimal center positions
    de_res = differential_evolution(
        objective, bounds, maxiter=400, popsize=15, seed=42, 
        tol=1e-8, workers=1, polish=False
    )
    
    centers = de_res.x.reshape((26, 2))
    sum_r, radii = compute_sum_radii(centers)
    
    # Apply a tiny safety margin to radii to strictly satisfy validator tolerance
    # (LP solutions are mathematically tight, but float arithmetic can be exact)
    radii = np.maximum(radii - 1e-9, 0.0)
    
    return centers, radii, float(sum_r)
