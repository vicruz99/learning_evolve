import math


def hexagonal_initial(n, offset_x=0.0, offset_y=0.0):
    """Generate hexagonal grid configuration for n circles."""
    centers = []
    r = 0.088
    
    row = 0
    y = r + offset_y
    while len(centers) < n:
        if row % 2 == 0:
            x = r + offset_x
        else:
            x = 2 * r + offset_x
        
        while x + r <= 1.001 and len(centers) < n:
            centers.append([x, y])
            x += 2 * r
        
        if not centers or centers[-1][1] != y:
            y += math.sqrt(3) * r
        row += 1
    
    centers = np.array(centers[:n])
    radii = np.ones(n) * r
    return centers, radii


def grid_initial(n):
    """Generate regular grid configuration."""
    rows = int(math.ceil(math.sqrt(n)))
    cols = int(math.ceil(n / rows))
    
    centers = []
    dx = 1.0 / cols
    dy = 1.0 / rows
    r = min(dx, dy) * 0.45
    
    count = 0
    for i in range(rows):
        for j in range(cols):
            if count >= n:
                break
            x = dx * (j + 0.5)
            y = dy * (i + 0.5)
            centers.append([x, y])
            count += 1
        if count >= n:
            break
    
    centers = np.array(centers[:n])
    radii = np.ones(n) * r
    return centers, radii

def compute_objective(v, n):
    """Objective: negative sum of radii."""
    return -np.sum(v[2 * n:])


def compute_constraints(v, n):
    """Compute all inequality constraints (must be >= 0)."""
    centers = v[:2 * n].reshape(n, 2)
    radii = v[2 * n:]
    
    m = 4 * n + n * (n - 1) // 2
    constraints = np.empty(m)
    
    idx = 0
    # Boundary constraints
    for i in range(n):
        constraints[idx] = centers[i, 0] - radii[i]
        idx += 1
        constraints[idx] = 1.0 - centers[i, 0] - radii[i]
        idx += 1
        constraints[idx] = centers[i, 1] - radii[i]
        idx += 1
        constraints[idx] = 1.0 - centers[i, 1] - radii[i]
        idx += 1
    
    # Overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist_sq = dx * dx + dy * dy
            sum_r = radii[i] + radii[j]
            constraints[idx] = dist_sq - sum_r * sum_r
            idx += 1
    
    return constraints


def do_optimize(centers, radii, n, max_iter=3000):
    """Run SLSQP optimization."""
    x0 = np.concatenate([centers.flatten(), radii])
    
    lb = np.zeros(3 * n)
    ub = np.ones(3 * n)
    ub[2 * n:] = 0.5
    
    bounds = list(zip(lb, ub))
    
    cons = {'type': 'ineq', 'fun': lambda v: compute_constraints(v, n)}
    
    result = minimize(
        compute_objective,
        x0,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={
            'maxiter': max_iter,
            'ftol': 1e-14,
            'disp': False,
            'iprint': -1,
        }
    )
    
    centers_opt = result.x[:2 * n].reshape(n, 2)
    radii_opt = result.x[2 * n:]
    
    # Ensure non-negative radii
    radii_opt = np.maximum(radii_opt, 1e-10)
    
    return centers_opt, radii_opt, result


def fix_boundary(centers, radii, n):
    """Ensure circles are inside the unit square."""
    for i in range(n):
        r = radii[i]
        x, y = centers[i]
        r = min(r, x, 1.0 - x, y, 1.0 - y, 0.5)
        r = max(r, 0.0)
        radii[i] = r
        centers[i, 0] = max(r, min(1.0 - r, centers[i, 0]))
        centers[i, 1] = max(r, min(1.0 - r, centers[i, 1]))
    return centers, radii


def fix_overlaps(centers, radii, n, max_iter=200):
    """Resolve overlaps by shrinking radii proportionally."""

        max_violation = 0
        max_pair = (0, 1)

                    violation = sum_r - dist
                    if violation > max_violation:
                        max_violation = violation
                        max_pair = (i, j)
        
        if max_violation < 1e-12:
            break
        
        i, j = max_pair
        total = radii[i] + radii[j]
        if total < 1e-15:
            break
        
        # Shrink both circles proportionally
        shrink = max_violation / 2.0
        radii[i] = max(1e-10, radii[i] - shrink)
        radii[j] = max(1e-10, radii[j] - shrink)
        
        # Update center positions for boundary

            centers[k, 0] = max(r, min(1.0 - r, centers[k, 0]))
            centers[k, 1] = max(r, min(1.0 - r, centers[k, 1]))
    
    return centers, radii


def refine_with_expansion(centers, radii, n):
