1.  **Initialize** 26 circles with small radii and random positions within the unit square to ensure a valid starting configuration.
2.  **Optimization Setup**: Formulate the problem as minimizing a penalty function. The objective is to maximize the sum of radii (equivalent to minimizing $-\sum r_i$).
3.  **Penalty Terms**: Add penalty terms for:
    *   **Overlaps**: Squared excess of sum of radii over Euclidean distance between centers.
    *   **Boundary Violations**: Squared excess of radius over distance to the nearest wall.
4.  **Solver**: Use `scipy.optimize.minimize` with the `L-BFGS-B` method, which supports box constraints (centers in $[0,1]$, radii $\ge 0$).
5.  **Robustness**: Run the optimization multiple times with different initial conditions to avoid local minima.
6.  **Post-Processing**: After optimization, enforce strict validity by clamping centers to ensure circles stay within the square and slightly reducing radii if any numerical overlaps remain.

import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    # Define bounds for optimization
    # x, y in [0, 1], r in [0, 0.5]
    bounds = []
    for _ in range(n):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 0.5)) # r

    def objective(vars):
        # Unpack variables: [x1, y1, r1, x2, y2, r2, ...]
        x = vars[0::3]
        y = vars[1::3]
        r = vars[2::3]
        
        # Primary objective: maximize sum of radii -> minimize -sum(r)
        obj_val = -np.sum(r)
        
        # Calculate pairwise distances and radius sums
        # Centers matrix (n, 2)
        centers = np.column_stack((x, y))
        
        # Distance matrix
        # dist_sq[i, j] = (x_i - x_j)^2 + (y_i - y_j)^2
        diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
        dist_sq = np.sum(diff**2, axis=2)
        
        # Sum of radii matrix
        r_sum = r[:, np.newaxis] + r[np.newaxis, :]
        r_sum_sq = r_sum**2
        
        # Identify overlap violations
        # Overlap occurs if dist < r_i + r_j  <=>  dist^2 < (r_i + r_j)^2
        # We look at the lower triangle (i > j) to avoid double counting and self-checks
        i_idx, j_idx = np.tril_indices(n, -1)
        
        violations_sq = r_sum_sq[i_idx, j_idx] - dist_sq[i_idx, j_idx]
        # We only penalize if violation > 0
        overlap_penalty = np.sum(np.maximum(0, violations_sq)**2)
        
        # Boundary violations
        # Circle i must satisfy: r <= x <= 1-r  and  r <= y <= 1-r
        # Violations: r - x > 0, x + r - 1 > 0, etc.
        b_left = np.maximum(0, r - x)
        b_right = np.maximum(0, x + r - 1)
        b_bottom = np.maximum(0, r - y)
        b_top = np.maximum(0, y + r - 1)
        
        boundary_penalty = np.sum(b_left**2 + b_right**2 + b_bottom**2 + b_top**2)
        
        # Combine penalties with a high weight to enforce constraints strictly
        # The weight needs to be high enough so that the optimizer prefers valid packings
        # with slightly smaller radii over invalid packings with larger radii.
        penalty_weight = 5000.0
        
        return obj_val + penalty_weight * (overlap_penalty + boundary_penalty)

    best_score = np.inf
    best_vars = None
    
    # Perform multiple restarts to find a better global optimum
    num_restarts = 15
    
    for k in range(num_restarts):
        # Initialize with random positions and small radii
        # Starting with small radii helps the optimizer grow them into the available space
        init_centers = np.random.rand(n, 2) * 0.8 + 0.1 # Keep away from edges initially
        init_radii = np.ones(n) * 0.01
        
        init_vars = np.zeros(3 * n)
        init_vars[0::3] = init_centers[:, 0]
        init_vars[1::3] = init_centers[:, 1]
        init_vars[2::3] = init_radii
        
        # Run optimizer
        try:
            res = minimize(objective, init_vars, method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 3000, 'ftol': 1e-12, 'gtol': 1e-9})
            
            if res.fun < best_score:
                best_score = res.fun
                best_vars = res.x
        except Exception:
            continue
            
    if best_vars is None:
        # Fallback to simple grid if optimization failed completely
        centers = np.zeros((n, 2))
        radii = np.zeros(n)
        # Simple 5x5 grid + 1
        idx = 0
        for r_row in range(5):
            for c_col in range(5):
                if idx < n:
                    centers[idx] = [0.1 + c_col * 0.2, 0.1 + r_row * 0.2]
                    radii[idx] = 0.09
                    idx += 1
        # Add last one
        if idx < n:
            centers[idx] = [0.5, 0.5]
            radii[idx] = 0.01
            idx += 1
        while idx < n:
             centers[idx] = [0.5, 0.5]
             radii[idx] = 0.001
             idx += 1
             
    else:
        x_opt = best_vars[0::3]
        y_opt = best_vars[1::3]
        r_opt = best_vars[2::3]
        
        centers = np.column_stack((x_opt, y_opt))
        radii = r_opt
        
        # Post-processing to ensure strict validity
        # 1. Clamp centers to ensure circles are within [0,1]x[0,1]
        for i in range(n):
            r = radii[i]
            cx, cy = centers[i]
            
            # Ensure center is at least r distance from walls
            if cx - r < 0: cx = r
            if cx + r > 1: cx = 1 - r
            if cy - r < 0: cy = r
            if cy + r > 1: cy = 1 - r
            centers[i] = [cx, cy]
            
        # 2. Resolve overlaps by shrinking radii if necessary
        # This ensures the returned packing is valid.
        # We iterate to resolve cascading overlaps.
        for _ in range(10): # Iterate a few times
            max_overlap = 0
            overlap_indices = (-1, -1)
            
            for i in range(n):
                for j in range(i + 1, n):
                    dist = np.sqrt((centers[i, 0] - centers[j, 0])**2 + (centers[i, 1] - centers[j, 1])**2)
                    required_dist = radii[i] + radii[j]
                    if dist < required_dist:
                        overlap = required_dist - dist
                        if overlap > max_overlap:
                            max_overlap = overlap
                            overlap_indices = (i, j)
            
            if max_overlap > 1e-9:
                i, j = overlap_indices
                # Shrink both radii to just touch
                # We want radii[i] + radii[j] = dist
                # Distribute the reduction proportionally or equally?
                # To preserve sum as much as possible, we just set sum = dist.
                # But we can't increase radii.
                # Current sum S = radii[i] + radii[j]. New sum S' = dist.
                # Loss = S - dist.
                # We can reduce one or both. Let's reduce both proportionally to current size?
                # Or just reduce the larger one? 
                # Actually, simply setting radii[i] += delta1, radii[j] += delta2 is not allowed.
                # We must reduce.
                # Let's reduce the larger radius to maintain balance? 
                # Or just reduce both by half the overlap.
                
                reduction = max_overlap / 2.0
                # Check if reduction is valid (radii non-negative)
                # Since radii are positive and overlap is small relative to radius usually, this is safe.
                # But strictly:
                available_i = radii[i]
                available_j = radii[j]
                
                # We need to reduce sum by max_overlap.
                # Reduce proportional to radius
                total_r = radii[i] + radii[j]
                if total_r > 0:
                    r_i_reduction = max_overlap * (radii[i] / total_r)
                    r_j_reduction = max_overlap * (radii[j] / total_r)
                else:
                    r_i_reduction = 0
                    r_j_reduction = 0
                
                radii[i] -= r_i_reduction
                radii[j] -= r_j_reduction
                
                # Ensure non-negative
                radii[i] = max(0.0, radii[i])
                radii[j] = max(0.0, radii[j])
            else:
                break

    # Ensure radii are non-negative
    radii = np.maximum(radii, 0.0)
    
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii

if __name__ == "__main__":
    centers, radii, sum_r = run_packing()
