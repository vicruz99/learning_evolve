import numpy as np
from scipy.optimize import minimize

def hexagonal_initial(n, r_start=0.035):
    """Initialize circles in a hexagonal pattern with 6,5,6,5,4 arrangement"""
    centers = np.zeros((n, 2))
    radii = np.ones(n) * r_start
    
    rows = [6, 5, 6, 5, 4]
    idx = 0
    for row_idx, num_circles in enumerate(rows):
        y = r_start + row_idx * r_start * np.sqrt(3)
        for col_idx in range(num_circles):
            x = r_start + col_idx * 2 * r_start
            if row_idx % 2 == 1:
                x += r_start
            centers[idx] = [x, y]
            idx += 1
    
    return centers, radii


def force_optimize(centers, radii, n, num_iterations=80000):
    """Force-based optimization with growing circles"""
    centers = centers.copy().astype(float)
    radii = radii.copy().astype(float)
    
    for it in range(num_iterations):
        forces = np.zeros((n, 2))
        growth_potential = np.zeros(n)
        
        # Repulsion between circles
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy) + 1e-12
                min_dist = radii[i] + radii[j]
                
                if dist < min_dist:
                    overlap = min_dist - dist
                    f = overlap * 50.0 / (dist + 1e-12)
                    fx = f * dx / (dist + 1e-12)
                    fy = f * dy / (dist + 1e-12)
                    
                    forces[i, 0] += fx
                    forces[i, 1] += fy
                    forces[j, 0] -= fx
                    forces[j, 1] -= fy
                else:
                    # Track free space for growth
                    gap = (dist - min_dist) / 2.0
                    growth_potential[i] += gap
                    growth_potential[j] += gap
        
        # Boundary repulsion
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            margin_left = x - r
            margin_right = 1 - (x + r)
            margin_bottom = y - r
            margin_top = 1 - (y + r)
            
            if margin_left < 0:
                forces[i, 0] += margin_left * 100
            if margin_right < 0:
                forces[i, 0] -= margin_right * 100
            if margin_bottom < 0:
                forces[i, 1] += margin_bottom * 100
            if margin_top < 0:
                forces[i, 1] -= margin_top * 100
        
        # Update positions with adaptive step size
        step = 0.008 / (1 + it / 15000)
        centers = centers + step * forces
        
        # Clamp to valid region
        for i in range(n):
            centers[i, 0] = max(radii[i] + 1e-12, min(1 - radii[i] - 1e-12, centers[i, 0]))
            centers[i, 1] = max(radii[i] + 1e-12, min(1 - radii[i] - 1e-12, centers[i, 1]))
        
        # Gradually grow radii
        if it % 20 == 0:
            growth_rate = 0.00002 / (1 + it / 20000)
            radii = radii + growth_rate
            
            # Enforce boundary constraints on radii
            for i in range(n):
                x, y = centers[i]
                max_r = min(x, 1 - x, y, 1 - y) - 1e-12
                if radii[i] > max_r:
                    radii[i] = max_r
        
        # Occasionally perturb to escape local minima
        if it % 5000 == 0 and it > 0:
            perturbation = 0.001 * np.random.randn(n, 2)
            centers = centers + perturbation
            for i in range(n):
                centers[i, 0] = max(radii[i], min(1 - radii[i], centers[i, 0]))
                centers[i, 1] = max(radii[i], min(1 - radii[i], centers[i, 1]))
    
    return centers, radii


def unequal_radius_optimize(centers, radii, n, num_iterations=50000):
    """Allow radii to vary - grow interior circles more, shrink boundary circles if needed"""
    centers = centers.copy().astype(float)
    radii = radii.copy().astype(float)
    
    for it in range(num_iterations):
        forces = np.zeros((n, 2))
        radius_forces = np.zeros(n)
        
        # Circle-circle interactions
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy) + 1e-12
                min_dist = radii[i] + radii[j]

                    overlap = min_dist - dist
                    f = overlap * 30.0 / (dist + 1e-12)
                    fx = f * dx / (dist + 1e-12)
                    fy = f * dy / (dist + 1e-12)
                    
                    forces[i, 0] += fx
                    forces[i, 1] += fy
                    forces[j, 0] -= fx
                    forces[j, 1] -= fy
                else:
                    # Free space encourages growth
                    gap = (dist - min_dist) / 2.0
                    radius_forces[i] += gap
                    radius_forces[j] += gap
        
        # Boundary interactions - circles near boundary can't grow as much
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            
            # Distance to nearest boundary
            d_left = x - r
            d_right = 1 - (x + r)
            d_bottom = y - r
            d_top = 1 - (y + r)
            
            min_boundary = min(d_left, d_right, d_bottom, d_top)
            
            # Growth potential based on boundary distance
            if min_boundary > 0.02:
                radius_forces[i] += 0.5  # Interior circles can grow more
            
            # Boundary repulsion
            if d_left < 0:
                forces[i, 0] += d_left * 100
            if d_right < 0:
                forces[i, 0] -= d_right * 100
            if d_bottom < 0:
                forces[i, 1] += d_bottom * 100
            if d_top < 0:
                forces[i, 1] -= d_top * 100
        
        # Update
        step_pos = 0.005 / (1 + it / 10000)
        step_rad = 0.0003 / (1 + it / 10000)
        
        centers = centers + step_pos * forces
        radii = radii + step_rad * np.clip(radius_forces, 0, 1)
        
        # Enforce constraints
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            centers[i, 0] = max(r + 1e-12, min(1 - r - 1e-12, x))
            centers[i, 1] = max(r + 1e-12, min(1 - r - 1e-12, y))
            max_r = min(centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])
            radii[i] = min(radii[i], max_r)
            radii[i] = max(radii[i], 1e-6)
    
    return centers, radii


def fine_tune_scipy(centers, radii, n):
    """Fine-tune using scipy L-BFGS-B with penalty method"""
    x0 = np.concatenate([centers.flatten(), radii])
    
    def objective(x):
        c = x[:2 * n].reshape((n, 2))
        r = x[2 * n:]
        
        # Maximize sum of radii (minimize negative)
        val = -np.sum(r)
        
        # Overlap penalties
        for i in range(n):
            for j in range(i + 1, n):
                dx = c[i, 0] - c[j, 0]
                dy = c[i, 1] - c[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                min_dist = r[i] + r[j]
                if dist < min_dist:
                    val += 50000 * (min_dist - dist) ** 2
        
        # Boundary penalties
        for i in range(n):
            x, y = c[i]
            ri = r[i]
            val += 50000 * max(0, ri - x) ** 2
            val += 50000 * max(0, x + ri - 1) ** 2
            val += 50000 * max(0, ri - y) ** 2
            val += 50000 * max(0, y + ri - 1) ** 2
        
        # Radius positivity penalty
        val += 1000 * sum(max(0, -ri) ** 2 for ri in r)
        
        return val
    
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
    
    result = minimize(
        objective, x0, bounds=bounds, method='L-BFGS-B',
        options={'maxiter': 20000, 'ftol': 1e-15, 'gtol': 1e-10}
    )
    
    c_opt = result.x[:2 * n].reshape((n, 2))
    r_opt = result.x[2 * n:]
    
    # Clip radii to ensure validity
    for i in range(n):
        x, y = c_opt[i]
        max_r = min(x, 1 - x, y, 1 - y)
        r_opt[i] = min(r_opt[i], max(0, max_r - 1e-10))
        c_opt[i, 0] = max(r_opt[i], min(1 - r_opt[i], c_opt[i, 0]))
        c_opt[i, 1] = max(r_opt[i], min(1 - r_opt[i], c_opt[i, 1]))
    
    return c_opt, r_opt


def validate_and_fix(centers, radii, n):

        max_r = min(x, 1 - x, y, 1 - y)
