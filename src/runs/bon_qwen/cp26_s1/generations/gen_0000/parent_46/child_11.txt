import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # Helper to create a hexagonal packing initialization
    def hex_init(n_circles):
        centers = []
        r_init = 0.08 # Start with a safe small radius
        row_idx = 0
        while len(centers) < n_circles:
            y = r_init + row_idx * r_init * np.sqrt(3)
            if y + r_init > 1.0:
                break
            if row_idx % 2 == 0:
                x_start = r_init
            else:
                x_start = r_init * 2
            col_idx = 0
            while True:
                x = x_start + col_idx * 2 * r_init
                if x + r_init > 1.0:
                    break
                centers.append([x, y])
                col_idx += 1
                if len(centers) >= n_circles:
                    break
            row_idx += 1
        return np.array(centers[:n_circles]), np.full(n_circles, r_init)

    # Helper to create a grid initialization
    def grid_init(n_circles):
        centers = []
        r_init = 0.09
        # Approximate sqrt layout
        cols = int(np.ceil(np.sqrt(n_circles)))
        rows = int(np.ceil(n_circles / cols))
        step_x = 1.0 / (cols + 1)
        step_y = 1.0 / (rows + 1)
        for r in range(rows):
            for c in range(cols):
                if len(centers) >= n_circles:
                    break
                centers.append([step_x * (c + 1), step_y * (r + 1)])
            if len(centers) >= n_circles:
                break
        return np.array(centers[:n_circles]), np.full(n_circles, r_init)

    def objective_and_penalty(vars):
        # vars is flattened: [x1, y1, r1, x2, y2, r2, ...]
        # Actually, we can separate radii and centers for cleaner constraints, 
        # but a flat vector is easier for scipy.
        # Let's structure as: centers (2*n) + radii (n)
        centers = np.array(vars[:2*n]).reshape(n, 2)
        radii = np.array(vars[2*n:])
        
        # Objective: Maximize sum of radii -> Minimize -sum
        obj = -np.sum(radii)
        
        penalty = 0.0
        
        # Boundary penalties
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if r < 0:
                penalty += 1000 * (r ** 2) # Push r to be positive
            else:
                # Distance to boundaries
                dist_left = x - r
                dist_right = 1.0 - x - r
                dist_down = y - r
                dist_up = 1.0 - y - r
                
                for d in [dist_left, dist_right, dist_down, dist_up]:
                    if d < 0:
                        penalty += 100 * d ** 2 # Quadratic penalty for violation
        
        # Overlap penalties
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.linalg.norm(centers[i] - centers[j])
                r_sum = radii[i] + radii[j]
                if dist < r_sum:
                    gap = r_sum - dist
                    penalty += 100 * gap ** 2
        
        return obj + penalty

    best_obj = float('-inf')
    best_centers = None
    best_radii = None

    # Try multiple initializations
    inits = [hex_init(n), grid_init(n)]
    
    # Also try some random inits near the center
    for _ in range(2):
        centers = np.random.uniform(0.1, 0.9, (n, 2))
        radii = np.full(n, 0.05)
        inits.append((centers, radii))

    for init_centers, init_radii in inits:
        # Flatten variables
        x0 = np.hstack([init_centers.flatten(), init_radii])
        
        # Bounds for variables
        # Centers: [0, 1]
        # Radii: [0, 0.5]
        bounds = []
        for i in range(2 * n):
            bounds.append((0.0, 1.0)) # x, y
        for i in range(n):
            bounds.append((0.0, 0.5)) # r

        try:
            res = minimize(objective_and_penalty, x0, method='L-BFGS-B', bounds=bounds, 
                           options={'maxiter': 1000, 'ftol': 1e-9})
            
            if res.success or res.nit > 0:
                centers = res.x[:2*n].reshape(n, 2)
                radii = res.x[2*n:]
                current_sum = np.sum(radii)
                
                # Verify constraints manually to be sure
                valid = True
                for i in range(n):
                    x, y = centers[i]
                    r = radii[i]
                    if r < 0: valid = False
                    if x < r - 1e-6 or x > 1 - r + 1e-6 or y < r - 1e-6 or y > 1 - r + 1e-6:
                        valid = False
                
                for i in range(n):
                    for j in range(i + 1, n):
                        d = np.linalg.norm(centers[i] - centers[j])
                        if d < radii[i] + radii[j] - 1e-6:
                            valid = False
                
                if valid and current_sum > best_obj:
                    best_obj = current_sum
                    best_centers = centers.copy()
                    best_radii = radii.copy()
                    
        except Exception:
            continue

    # Fallback to a simple valid solution if optimization fails
    if best_centers is None or best_obj < 2.5:
        best_centers, best_radii = grid_init(n)
        best_obj = np.sum(best_radii)

    # Final clean-up: ensure radii are non-negative and centers are valid
    # A simple scaling down if needed to strictly satisfy constraints (though penalty should handle it)
    # But let's just return the best found.
    
    return best_centers, best_radii, best_obj
