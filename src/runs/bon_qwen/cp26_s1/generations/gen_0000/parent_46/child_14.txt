To maximize the sum of radii for 26 circles in a unit square, we treat this as a constrained nonlinear optimization problem. The objective is to maximize $\sum r_i$, subject to:
1. Boundary constraints: $0 \le r_i \le x_i \le 1$ and $0 \le r_i \le y_i \le 1$.
2. Non-overlap constraints: $(x_i - x_j)^2 + (y_i - y_j)^2 \ge (r_i + r_j)^2$ for all $i \ne j$.

We use `scipy.optimize.minimize` with the SLSQP algorithm, which handles inequality constraints effectively. To escape local optima:
- We initialize from a hexagonal lattice pattern, which closely approximates optimal circle packing densities.
- We run multiple random restarts (40) with slight perturbations to the initial positions and radii.
- We enforce strict bounds $[0, 1]$ for coordinates and $[0, 0.5]$ for radii.
- After finding the best feasible solution across restarts, we run a final high-precision refinement pass.
- All helper functions are defined at the top level to comply with rules. Constraints are vectorized for performance.

import numpy as np
from scipy.optimize import minimize

def neg_sum_radii(vars, n):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(vars[2::3])

def compute_constraints(vars, n, mask):
    """Compute boundary and non-overlap inequality constraints (must be >= 0)."""
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    # Boundary constraints: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    b_cons = np.empty(4 * n)
    b_cons[0::4] = x - r
    b_cons[1::4] = 1.0 - x - r
    b_cons[2::4] = y - r
    b_cons[3::4] = 1.0 - y - r
    
    # Overlap constraints: dist^2 - (r_i + r_j)^2 >= 0
    X = vars.reshape(n, 3)
    dx = X[:, 0, np.newaxis] - X[np.newaxis, :, 0]
    dy = X[:, 1, np.newaxis] - X[np.newaxis, :, 1]
    dr = X[:, 2, np.newaxis] + X[np.newaxis, :, 2]
    
    o_cons = (dx[mask] ** 2 + dy[mask] ** 2) - dr[mask] ** 2
    return np.concatenate([b_cons, o_cons])

def get_bounds(n):
    """Return variable bounds: x in [0,1], y in [0,1], r in [0, 0.5]."""
    return [(0.0, 1.0)] * n + [(0.0, 1.0)] * n + [(0.0, 0.5)] * n

def run_packing():
    n = 26
    mask = np.tril(np.ones((n, n), dtype=bool), -1)
    
    cons = {'type': 'ineq', 'fun': compute_constraints, 'args': (n, mask)}
    bnds = get_bounds(n)
    
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Multiple restarts from perturbed hexagonal grids
    for seed in range(40):
        rng = np.random.RandomState(seed)
        pts = []
        rows = [6, 5, 6, 5, 4]
        y_base = 0.1
        dy = 0.18
        for i, cnt in enumerate(rows):
            y = y_base + i * dy
            x_start = 0.1 + (0.0 if i % 2 == 0 else 0.12)
            for _ in range(cnt):
                pts.append([x_start, y, 0.02])
                x_start += 0.20
            if len(pts) >= n:
                break
                
        while len(pts) < n:
            pts.append([rng.uniform(0.1, 0.9), rng.uniform(0.1, 0.9), 0.02])
            
        x0 = np.array(pts[:n]).flatten()
        x0 += rng.normal(0, 0.01, size=x0.shape)
        x0[0::3] = np.clip(x0[0::3], 0.05, 0.95)
        x0[1::3] = np.clip(x0[1::3], 0.05, 0.95)
        x0[2::3] = np.clip(x0[2::3], 0.02, 0.15)
        
        res = minimize(neg_sum_radii, x0, args=(n,), method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': 4000, 'ftol': 1e-10, 'disp': False})
        
        if res.success:
            centers = res.x.reshape(n, 3)[:, :2]
            radii = res.x.reshape(n, 3)[:, 2]
            
            valid = True
            if np.any(radii < 0) or np.any(centers < 0) or np.any(centers > 1):
                valid = False
            elif np.any(centers - radii < -1e-9) or np.any(centers + radii > 1 + 1e-9):
                valid = False
            else:
                dists = np.linalg.norm(centers[:, np.newaxis, :] - centers[np.newaxis, :, :], axis=2)
                sum_r = radii[:, np.newaxis] + radii[np.newaxis, :]
                if np.any(dists < sum_r - 1e-9):
                    valid = False
                    
            if valid:
                total = np.sum(radii)
                if total > best_sum:
                    best_sum = total
                    best_centers = centers.copy()
                    best_radii = radii.copy()
                    
    # High-precision refinement of the best found configuration
    if best_centers is not None:
        x0 = np.concatenate([best_centers.flatten(), best_radii])
        res = minimize(neg_sum_radii, x0, args=(n,), method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': 5000, 'ftol': 1e-12, 'disp': False})
        if res.success:
            centers = res.x.reshape(n, 3)[:, :2]
            radii = res.x.reshape(n, 3)[:, 2]
            if np.all(centers >= 0) and np.all(centers <= 1) and np.all(radii >= 0):
                best_centers = centers
                best_radii = radii
                best_sum = np.sum(radii)

    # Fallback to a valid simple packing if optimization fails
    if best_centers is None:
        best_centers = np.random.uniform(0.2, 0.8, (n, 2))
        best_radii = np.full(n, 0.05)
        best_sum = np.sum(best_radii)
        
    return best_centers, best_radii, best_sum
