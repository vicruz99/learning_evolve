def generate_hexagonal_packing(n):
    """Generate an initial hexagonal packing for n circles."""
    # Estimate radius based on area
    r_est = 0.1
    centers = []
    radii = []
    
    # Hexagonal lattice parameters
    dx = 2 * r_est
    dy = math.sqrt(3) * r_est
    
    y = r_est
    while len(centers) < n:
        x = r_est
        # Alternate row shifts
        shift = 0 if int((y - r_est) / dy) % 2 == 0 else r_est
        x += shift
        
        while x <= 1 - r_est and len(centers) < n:
            centers.append([x, y])
            radii.append(r_est)
            x += dx
        y += dy
        
    # Convert to numpy arrays
    centers = np.array(centers[:n])
    radii = np.array(radii[:n])
    return centers, radii

def optimize_packing(initial_centers, initial_radii):
    """Optimize positions and radii to maximize sum of radii."""
    n = len(initial_radii)
    x0 = np.concatenate([initial_centers.flatten(), initial_radii])
    
    # Constraints
    constraints = []
    
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    # x_i - r_i >= 0  =>  x_i - r_i >= 0
    # 1 - x_i - r_i >= 0

        # x constraints
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: x[2*i] - x[2*n + i]}) # x_i - r_i
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: 1.0 - x[2*i] - x[2*n + i]}) # 1 - x_i - r_i
        
        # y constraints
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: x[2*i + 1] - x[2*n + i]}) # y_i - r_i
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: 1.0 - x[2*i + 1] - x[2*n + i]}) # 1 - y_i - r_i
        
        # Non-negative radius
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: x[2*n + i]})

    # Non-overlap constraints: dist_ij >= r_i + r_j
    # (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({
                'type': 'ineq',

            })

    def objective(x):
        return -np.sum(x[2*n:])

            final_radii = res.x[2*n:]
            return final_centers, final_radii
    except Exception:
        pass
        
    return initial_centers, initial_radii

def ensure_validity(centers, radii):
    """
    Adjust centers and radii to ensure strict validity within numerical tolerance.
    """
    n = len(radii)
    # Shrink radii slightly to avoid boundary overlaps due to precision
    # and ensure non-overlap

        r = radii[i]
        # Max possible radius constrained by boundaries
        max_r_bound = min(x, 1-x, y, 1-y)
        if r > max_r_bound:
            radii[i] = max_r_bound - tolerance
            # Recalculate if needed, but for small adjustments this is enough
            
    # 2. Enforce non-overlap
    # Simple iterative shrink
    for _ in range(50):
        overlap_found = False
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = math.sqrt(dx*dx + dy*dy)
                req_dist = radii[i] + radii[j]
                
                if dist < req_dist - tolerance:
                    # Overlap detected, shrink radii

                    radii[i] -= reduction
                    radii[j] -= reduction
                    radii[i] = max(0, radii[i])
                    radii[j] = max(0, radii[j])
                    overlap_found = True
        if not overlap_found:
            break
            
    return centers, radii
