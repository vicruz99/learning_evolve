import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist


def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square

    Args:
        centers: np.array of shape (n, 2) with (x, y) coordinates
        radii: np.array of shape (n) with radius of each circle

    Returns:
        True if valid, False otherwise
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True


def force_directed_packing(n, seed=42, steps=80000):
    """
    Force-directed circle packing with growing radii.
    Returns centers and radii arrays.
    """
    rng = np.random.RandomState(seed)
    centers = rng.rand(n, 2) * 0.6 + 0.2
    radii = np.ones(n) * 0.02
    velocities = np.zeros((n, 2))
    
    dt = 0.002
    growth_rate = 2e-7
    repulsion_k = 800
    boundary_k = 400
    damping = 0.85
    
    for step in range(steps):
        # Grow radii
        growth = growth_rate * (1 + 0.0001 * step)
        radii += growth * np.ones(n)
        
        # Compute repulsive forces
        diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
        dists_sq = np.sum(diffs**2, axis=2)
        dists_sq = np.maximum(dists_sq, 1e-16)
        dists = np.sqrt(dists_sq)
        
        # Only consider pairs that are close (overlap or near-overlap)
        threshold = radii[:, np.newaxis] + radii[np.newaxis, :] + 0.05
        close_mask = dists < threshold
        
        # Repulsive forces
        repulsion = np.zeros((n, n, 2))
        np.fill_diagonal(close_mask, False)
        
        # Vectorized force computation for close pairs
        for i in range(n):
            mask = close_mask[i]
            if not np.any(mask):
                continue
            j_indices = np.where(mask)[0]
            diff_i = diffs[i, j_indices]
            dist_i = dists[i, j_indices]
            radii_sum = radii[i] + radii[j_indices]
            
            # Force magnitude proportional to overlap
            overlap = np.maximum(0, radii_sum - dist_i)
            force_mag = repulsion_k * overlap / (dist_i + 1e-10)
            direction = diff_i / (dist_i[:, np.newaxis] + 1e-10)
            
            force = force_mag[:, np.newaxis] * direction
            repulsion[i, j_indices] = force
            repulsion[j_indices, i] = -force
        
        forces = np.sum(repulsion, axis=1)
        
        # Boundary forces
        for i in range(n):
            r = radii[i]
            cx, cy = centers[i]
            
            if cx - r < 0:
                forces[i, 0] += boundary_k * (r - cx)
            if cx + r > 1:
                forces[i, 0] -= boundary_k * (cx + r - 1)
            if cy - r < 0:
                forces[i, 1] += boundary_k * (r - cy)
            if cy + r > 1:
                forces[i, 1] -= boundary_k * (cy + r - 1)
        
        # Update velocities and positions
        velocities = velocities * damping + forces * dt
        centers += velocities * dt
        
        # Clip centers to stay in bounds
        centers = np.clip(centers, 0.001, 0.999)
    
    return centers, radii


def hexagonal_initial(n, seed=42):
    """
    Create an initial configuration using hexagonal-like arrangement.
    """
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.1
    
    # Try to fit circles in a hexagonal pattern
    # 5 rows with varying numbers of circles
    row_configs = [
        (5, 0.1, 0.1),       # 5 circles at y=0.1
        (5, 0.2, 0.1 + 0.1*np.sqrt(3)),  # 5 circles shifted
        (6, 0.1, 0.1 + 2*0.1*np.sqrt(3)),  # 6 circles
        (5, 0.2, 0.1 + 3*0.1*np.sqrt(3)),  # 5 circles shifted
        (5, 0.1, 0.1 + 4*0.1*np.sqrt(3)),  # 5 circles
    ]
    
    idx = 0
    for count, x_start, y in row_configs:
        if idx >= n:
            break
        for col in range(count):
            if idx >= n:
                break
            x = x_start + col * 0.2
            if x < 0 or x > 1:
                continue
            centers[idx] = [x, y]
            idx += 1
    
    # Fill remaining circles
    while idx < n:
        centers[idx] = [0.2 + (idx % 5) * 0.2, 0.2 + (idx // 5) * 0.2]
        idx += 1
    
    return centers, radii


def grid_initial(n):
    """
    Create a 5x5 grid plus one extra circle.
    """
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.1
    
    idx = 0
    for i in range(5):
        for j in range(5):
            centers[idx] = [0.1 + i * 0.2, 0.1 + j * 0.2]
            idx += 1
    
    # Extra circle in a gap
    centers[idx] = [0.2, 0.2]
    radii[idx] = 0.04
    
    return centers, radii


def optimize_with_slsqp(centers, radii, n=26, maxiter=2000):
    """
    Use SLSQP to optimize the sum of radii with explicit constraints.
    """
    x0 = np.concatenate([radii, centers.flatten()])
    
    def objective(x):
        return -np.sum(x[:n])
    
    constraints = []
    
    # Boundary constraints for each circle
    for i in range(n):
        # x_i - r_i >= 0
        def make_bc_xlow(i):
            def bc(x):
                return x[n + 2*i] - x[i]
            return bc
        constraints.append({'type': 'ineq', 'fun': make_bc_xlow(i)})
        
        # 1 - x_i - r_i >= 0
        def make_bc_xhigh(i):
            def bc(x):
                return 1 - x[n + 2*i] - x[i]
            return bc
        constraints.append({'type': 'ineq', 'fun': make_bc_xhigh(i)})
        
        # y_i - r_i >= 0
        def make_bc_ylow(i):
            def bc(x):
                return x[n + 2*i + 1] - x[i]
            return bc
        constraints.append({'type': 'ineq', 'fun': make_bc_ylow(i)})
        
        # 1 - y_i - r_i >= 0
        def make_bc_yhigh(i):
            def bc(x):
                return 1 - x[n + 2*i + 1] - x[i]
            return bc
        constraints.append({'type': 'ineq', 'fun': make_bc_yhigh(i)})
        
        # r_i >= 0
        def make_bc_rpos(i):
            def bc(x):
                return x[i]
            return bc
        constraints.append({'type': 'ineq', 'fun': make_bc_rpos(i)})
    
    # Non-overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            def make_overlap(i, j):
                def oc(x):
                    dx = x[n + 2*i] - x[n + 2*j]
                    dy = x[n + 2*i + 1] - x[n + 2*j + 1]
                    dist = np.sqrt(dx**2 + dy**2)
                    return dist - x[i] - x[j]
                return oc
            constraints.append({'type': 'ineq', 'fun': make_overlap(i, j)})
    
    bounds = [(0, 0.5) for _ in range(n)] + [(0, 1) for _ in range(2*n)]
    
    result = minimize(
        objective, x0, method='SLSQP',
        bounds=bounds, constraints=constraints,
        options={'maxiter': maxiter, 'ftol': 1e-14, 'disp': False}
    )
    
    return result.x[:n], result.x[n:].reshape((n, 2))


def penalized_optimization(centers, radii, n=26, steps=5000):
    """
    Use gradient-based optimization with penalty method.
    """
    from scipy.optimize import minimize
    
    x0 = np.concatenate([radii, centers.flatten()])
    
    def penalized_objective(x, penalty=5000):
        radii = x[:n]
        centers = x[n:].reshape((n, 2))
        
        score = -np.sum(radii)
        
        # Boundary penalties (smooth)
        for i in range(n):
            r = radii[i]
            cx, cy = centers[i]
            
            # x - r >= 0
            violation = max(0, -(cx - r))
            score += penalty * violation**2
            
            # 1 - x - r >= 0
            violation = max(0, -(1 - cx - r))
            score += penalty * violation**2
            
            # y - r >= 0
            violation = max(0, -(cy - r))
            score += penalty * violation**2
            
            # 1 - y - r >= 0
            violation = max(0, -(1 - cy - r))
            score += penalty * violation**2
            
            # r >= 0
            violation = max(0, -r)
            score += penalty * violation**2
        
        # Overlap penalties (smooth)
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                overlap = radii[i] + radii[j] - dist
                violation = max(0, overlap)
                score += penalty * violation**2
        
        return score
    
    bounds = [(0, 0.5) for _ in range(n)] + [(0.001, 0.999) for _ in range(2*n)]
    
    best_score = float('inf')
    best_x = x0
    
    # Multiple restarts from nearby points
    for restart in range(5):
        x_init = x0.copy()
        x_init[:n] += np.random.randn(n) * 0.002
        x_init[:n] = np.clip(x_init[:n], 0.01, 0.3)
        x_init[n:] += np.random.randn(2*n) * 0.005
        x_init[n:] = np.clip(x_init[n:], 0.01, 0.99)
        
        result = minimize(
            penalized_objective, x_init, method='L-BFGS-B',
            bounds=bounds, options={'maxiter': steps, 'ftol': 1e-16}
        )
        
        if result.fun < best_score:
            best_score = result.fun
            best_x = result.x
    
    radii_opt = best_x[:n]
    centers_opt = best_x[n:].reshape((n, 2))
    
    # Clip to valid range
    radii_opt = np.clip(radii_opt, 0, 0.5)
    centers_opt = np.clip(centers_opt, 0, 1)
    
    return radii_opt, centers_opt
