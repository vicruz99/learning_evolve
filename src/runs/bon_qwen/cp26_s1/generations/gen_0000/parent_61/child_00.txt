To maximize the sum of radii for 26 circles in a unit square, we formulate this as a constrained non-linear optimization problem. Directly handling 325 pairwise non-overlap constraints plus boundary constraints is computationally heavy for standard solvers. Instead, we use a **smooth penalty method** combined with **L-BFGS-B**.

Key steps:
1. **Penalty Formulation**: Convert hard constraints into a smooth penalty term added to the objective function. The objective becomes: $\min (-\sum r_i + \lambda \cdot \text{Penalty})$, where Penalty sums squared violations of boundary and overlap constraints. Using squared `max(0, violation)` ensures C1 continuity, suitable for gradient-based optimizers.
2. **Multi-Start Strategy**: The problem is highly non-convex with many local minima. We run the optimizer from 30 different initial configurations:
   - 15 hexagonal lattice starts (geometrically inspired, high chance of finding dense packings)
   - 15 random starts (explore diverse regions)
3. **Refinement Phase**: After identifying the best solution from multi-starts, we run a final high-precision optimization pass on that configuration to squeeze out remaining slack.
4. **Bounds & Stability**: L-BFGS-B naturally handles box constraints ($0 \le x,y \le 1$, $0 \le r \le 0.5$), keeping the search within feasible physical limits. A high penalty weight ($\lambda = 5000$) ensures constraints are tightly respected.

This approach is robust, requires no manual Jacobian implementation, and efficiently navigates the solution space to reach or exceed the target sum of radii.

import numpy as np
from scipy.optimize import minimize

def compute_penalty_and_neg_sum(params, n, pairs):
    """Computes the negative sum of radii plus a smooth penalty for constraint violations."""
    centers = params[:2*n].reshape(n, 2)
    radii = params[2*n:]
    
    pen = 0.0
    # Boundary constraint penalties: circle must be inside [0,1]x[0,1]
    # x - r >= 0
    pen += np.sum(np.maximum(0.0, radii - centers[:, 0])**2)
    # 1 - x - r >= 0
    pen += np.sum(np.maximum(0.0, radii - (1.0 - centers[:, 0]))**2)
    # y - r >= 0
    pen += np.sum(np.maximum(0.0, radii - centers[:, 1])**2)
    # 1 - y - r >= 0
    pen += np.sum(np.maximum(0.0, radii - (1.0 - centers[:, 1]))**2)
    
    # Overlap constraint penalties: distance >= r_i + r_j
    for i, j in pairs:
        dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
        overlap = radii[i] + radii[j] - dist
        if overlap > 0.0:
            pen += overlap**2
            
    # High penalty weight ensures constraints are respected
    return -np.sum(radii) + 5000.0 * pen

def generate_initial_guess(n, method, seed):
    """Generates an initial configuration of centers and radii."""
    np.random.seed(seed)
    if method == 'hex':
        pts = []
        y = 0.12
        while len(pts) < n:
            x = 0.12
            row_shift = (np.floor(y / 0.18) % 2) * 0.09
            while x <= 0.9 and len(pts) < n:
                pts.append([x + row_shift, y])
                x += 0.18
            y += 0.15
        pts = np.array(pts[:n])
        pts += np.random.randn(n, 2) * 0.01
        pts = np.clip(pts, 0.05, 0.95)
        r = np.ones(n) * 0.11
    else:
        pts = np.random.uniform(0.1, 0.9, (n, 2))
        r = np.ones(n) * 0.06
    return np.concatenate([pts.ravel(), r])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
    
    best_sum = -np.inf
    best_centers = None
    best_radii = None
    
    # Box constraints for L-BFGS-B
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
    
    # Multi-start optimization
    for i in range(30):
        method = 'hex' if i < 15 else 'random'
        x0 = generate_initial_guess(n, method, seed=42 + i)
        
        def objective(p):
            return compute_penalty_and_neg_sum(p, n, pairs)
            
        try:
            res = minimize(objective, x0, method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 800, 'ftol': 1e-12, 'gtol': 1e-9})
            
            # Accept solution if successful or sufficiently optimized
            if res.success or (res.nit > 100 and res.fun < 0):
                c = res.x[:2*n].reshape(n, 2)
                r = res.x[2*n:]
                s = np.sum(r)
                if s > best_sum:
                    best_sum = s
                    best_centers = c.copy()
                    best_radii = r.copy()
        except Exception:
            continue
            
    # Final high-precision refinement on the best found configuration
    if best_centers is not None:
        x0_refine = np.concatenate([best_centers.ravel(), best_radii])
        def objective_refine(p):
            return compute_penalty_and_neg_sum(p, n, pairs)
            
        res_ref = minimize(objective_refine, x0_refine, method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 1200, 'ftol': 1e-14, 'gtol': 1e-10})
        
        # Update if refinement improved the result
        if res_ref.fun < compute_penalty_and_neg_sum(x0_refine, n, pairs):
            best_centers = res_ref.x[:2*n].reshape(n, 2)
            best_radii = res_ref.x[2*n:]
            best_sum = np.sum(best_radii)
            
    return best_centers, best_radii, float(best_sum)
