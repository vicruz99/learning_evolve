import numpy as np
from scipy.optimize import minimize

def compute_radii(centers):
    """
    Compute the maximum possible radii for fixed centers.
    Uses iterative relaxation to satisfy r_i + r_j <= dist(i, j).
    """
    n = centers.shape[0]
    # Initialize radii based on wall constraints
    r = np.zeros(n)
    for i in range(n):
        x, y = centers[i]
        r[i] = min(x, 1.0 - x, y, 1.0 - y)
    
    # Iterative relaxation to satisfy inter-circle constraints
    # r_i + r_j <= dist(i, j) => r_i <= dist(i, j) - r_j
    # We iterate until convergence
    for _ in range(100): # Sufficient iterations for 26 circles
        r_new = r.copy()
        for i in range(n):
            # Check against all other circles
            min_lim = r[i]
            for j in range(n):
                if i == j: continue
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                lim = dist - r[j]
                if lim < min_lim:
                    min_lim = lim
            r_new[i] = max(0.0, min_lim)
        # Check convergence
        if np.max(np.abs(r_new - r)) < 1e-7:
            break
        r = r_new
    return r

from scipy.optimize import minimize, linprog

def get_optimal_radii(centers):
    """
    Calculates the maximum sum of radii for a fixed set of centers
    by solving a Linear Programming problem.
    """
    n = centers.shape[0]
    x = centers[:, 0]
    y = centers[:, 1]
    
    # Upper bounds for radii imposed by walls
    # r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    w = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))

    # though LP might fail with negative bounds.
    w = np.maximum(w, 0.0)
    
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    # Indices for upper triangle
    i_idx, j_idx = np.triu_indices(n, k=1)
    num_pairs = len(i_idx)
    
    # Distances
    diff = centers[i_idx] - centers[j_idx]
    dists = np.sqrt(np.sum(diff**2, axis=1))

    # r_i + r_j <= dist_ij  =>  [0..1..1..0] @ r <= dist
    A_ub = np.zeros((num_pairs, n))
