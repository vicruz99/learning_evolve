import numpy as np

def compute_forces(centers, radii, stiffness=80.0, wall_stiffness=300.0):
    """
    Compute repulsive forces between circles and with walls.
    Forces push centers apart to resolve overlaps and boundary violations.
    """
    n = centers.shape[0]
    forces = np.zeros_like(centers)
    
    # Circle-Circle repulsion
    for i in range(n):
        for j in range(i + 1, n):
            diff = centers[j] - centers[i]
            dist_sq = diff[0]**2 + diff[1]**2
            dist = np.sqrt(dist_sq) if dist_sq > 1e-14 else 1e-7
            
            min_dist = radii[i] + radii[j]
            if dist < min_dist:
                overlap = min_dist - dist
                norm_diff = diff / dist
                f = stiffness * overlap * norm_diff
                forces[i] -= f
                forces[j] += f
                
    # Wall repulsion
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < 0:
            forces[i, 0] += wall_stiffness * (r - x)
        if x + r > 1:
            forces[i, 0] -= wall_stiffness * (x + r - 1)
        if y - r < 0:
            forces[i, 1] += wall_stiffness * (r - y)
        if y + r > 1:
            forces[i, 1] -= wall_stiffness * (y + r - 1)
            
    return forces
