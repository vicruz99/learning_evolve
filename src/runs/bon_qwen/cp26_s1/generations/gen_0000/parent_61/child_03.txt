import numpy as np


def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None

    for restart in range(10):
        centers, radii = initialize_hexagonal(n, restart)
        centers, radii = force_directed_optimize(centers, radii, n)
        centers, radii = refine_with_scipy(centers, radii, n)
        
        current_sum = np.sum(radii)
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers.copy()
            best_radii = radii.copy()

    return best_centers, best_radii, best_sum


def initialize_hexagonal(n, seed):
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.065
    
    np.random.seed(seed)
    
    idx = 0
    for row in range(6):
        n_cols = 5
        for col in range(n_cols):
            if idx >= n:
                break
            x = (col + 0.5 * (row % 2)) * 0.19 + 0.095
            y = row * 0.185 + 0.12
            centers[idx] = [x, y]
            idx += 1
    
    centers += np.random.uniform(-0.015, 0.015, size=(n, 2))
    centers = np.clip(centers, 0.05, 0.95)
    
    return centers, radii


def compute_forces(centers, radii, n):
    forces = np.zeros_like(centers)
    
    for i in range(n):
        for j in range(i + 1, n):
            diff = centers[i] - centers[j]
            dist_sq = np.sum(diff ** 2)
            dist = np.sqrt(dist_sq)
            min_dist = radii[i] + radii[j]
            
            if dist < min_dist and dist > 1e-12:
                repulsion_strength = (min_dist - dist) / dist
                forces[i] += diff * repulsion_strength
                forces[j] -= diff * repulsion_strength
    
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        
        if x - r < 0.01:
            forces[i, 0] += (0.01 - (x - r)) * 50
        if x + r > 0.99:
            forces[i, 0] -= ((x + r) - 0.99) * 50
        if y - r < 0.01:
            forces[i, 1] += (0.01 - (y - r)) * 50
        if y + r > 0.99:
            forces[i, 1] -= ((y + r) - 0.99) * 50
    
    return forces


def force_directed_optimize(centers, radii, n):
    max_iter = 4000
    
    for iteration in range(max_iter):
        forces = compute_forces(centers, radii, n)
        
        progress = iteration / max_iter
        lr = 0.015 * max(0.05, 1.0 - progress)
        
        centers += forces * lr
        
        for i in range(n):
            x, y = centers[i]
            max_r_from_boundary = min(x, 1 - x, y, 1 - y)
            
            for j in range(n):
                if i != j:
                    diff = centers[i] - centers[j]
                    dist = np.sqrt(np.sum(diff ** 2))
                    max_r_from_neighbor = dist - radii[j]
                    max_r_from_boundary = min(max_r_from_boundary, max_r_from_neighbor)
            
            max_r = min(max_r_from_boundary, 0.5)
            radii[i] = min(radii[i] * 1.0003, max(0, max_r - 1e-10))
        
        centers = np.clip(centers, 0.001, 0.999)
    
    return centers, radii


def refine_with_scipy(centers, radii, n):
    from scipy.optimize import minimize
    
    vars_init = np.concatenate([centers.flatten(), radii])
    
    cons = []
    for i in range(n):
        for j in range(i + 1, n):
            cons.append({
                'type': 'ineq',
                'fun': nonoverlap_constraint,
                'args': (n, i, j)
            })
        cons.append({
            'type': 'ineq',
            'fun': boundary_left,
            'args': (n, i)
        })
        cons.append({
            'type': 'ineq',
            'fun': boundary_right,
            'args': (n, i)
        })
        cons.append({
            'type': 'ineq',
            'fun': boundary_bottom,
            'args': (n, i)
        })
        cons.append({
            'type': 'ineq',
            'fun': boundary_top,
            'args': (n, i)
        })
    
    bounds = []
    for i in range(n):
        bounds.append((0.001, 0.999))
        bounds.append((0.001, 0.999))
    for i in range(n):
        bounds.append((0.001, 0.5))
    
    result = minimize(
        objective, vars_init, args=(n,),
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 300, 'ftol': 1e-12}
    )
    
    if result.success or result.nit > 0:
        vars_opt = result.x
    else:
        vars_opt = vars_init
    
    opt_centers = vars_opt[:2 * n].reshape(n, 2)
    opt_radii = vars_opt[2 * n:]
    
    opt_centers = np.clip(opt_centers, 0.001, 0.999)
    opt_radii = np.clip(opt_radii, 0.001, 0.5)
    
    for i in range(n):
        x, y = opt_centers[i]
        max_r = min(x, 1 - x, y, 1 - y) - 1e-12
        opt_radii[i] = min(opt_radii[i], max_r)
    
    return opt_centers, opt_radii


def objective(vars, n):
    radii = vars[2 * n:]
    return -np.sum(radii)


def nonoverlap_constraint(vars, n, i, j):
    c1x = vars[2 * i]
    c1y = vars[2 * i + 1]
    c2x = vars[2 * j]
    c2y = vars[2 * j + 1]
    r1 = vars[2 * n + i]
    r2 = vars[2 * n + j]
    
    dx = c1x - c2x
    dy = c1y - c2y
    dist = np.sqrt(dx * dx + dy * dy)
    
    return dist - r1 - r2


def boundary_left(vars, n, i):
    x = vars[2 * i]
    r = vars[2 * n + i]
    return x - r


def boundary_right(vars, n, i):
    x = vars[2 * i]
    r = vars[2 * n + i]
    return 1.0 - x - r


def boundary_bottom(vars, n, i):
    y = vars[2 * i + 1]
    r = vars[2 * n + i]
    return y - r


def boundary_top(vars, n, i):
    y = vars[2 * i + 1]
    r = vars[2 * n + i]
    return 1.0 - y - r
