import numpy as np
from scipy.optimize import minimize


N_CIRCLES = 26


def objective_func(x):
    """Objective: maximize sum of radii (minimize negative sum)."""
    radii = x[2 * N_CIRCLES:]
    return -np.sum(radii)


def constraint_func(x):
    """All inequality constraints: boundary and non-overlap."""
    n = N_CIRCLES
    centers = x[:2 * n].reshape(n, 2)
    radii = x[2 * n:]
    constraints = []

    # Boundary constraints for each circle
    for i in range(n):
        constraints.append(centers[i, 0] - radii[i])
        constraints.append(1.0 - centers[i, 0] - radii[i])
        constraints.append(centers[i, 1] - radii[i])
        constraints.append(1.0 - centers[i, 1] - radii[i])

    # Non-overlap constraints for each pair
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            constraints.append(dist - radii[i] - radii[j])

    return np.array(constraints)


def make_bounds():
    """Create variable bounds."""
    bounds = []
    for _ in range(2 * N_CIRCLES):
        bounds.append((0.0, 1.0))
    for _ in range(N_CIRCLES):
        bounds.append((0.0, 0.5))
    return bounds


def init_5x5_plus_center():
    """Initialize with 5x5 grid plus one circle in center gap."""
    centers = np.zeros((N_CIRCLES, 2))
    idx = 0
    for i in range(5):
        for j in range(5):
            centers[idx] = [0.1 + j * 0.2, 0.1 + i * 0.2]
            idx += 1
    # 26th circle in a gap between grid circles
    centers[25] = [0.2, 0.2]
    radii = np.ones(N_CIRCLES) * 0.07
    return centers, radii


def init_5x5_plus_various_gaps():
    """Initialize with 5x5 grid and 26th circle in various gaps."""
    inits = []
    gap_positions = [
        [0.2, 0.2], [0.4, 0.2], [0.6, 0.2], [0.8, 0.2],
        [0.2, 0.4], [0.4, 0.4], [0.6, 0.4], [0.8, 0.4],
        [0.2, 0.6], [0.4, 0.6], [0.6, 0.6], [0.8, 0.6],
        [0.2, 0.8], [0.4, 0.8], [0.6, 0.8], [0.8, 0.8],
        [0.3, 0.3], [0.5, 0.5], [0.7, 0.7], [0.5, 0.3]
    ]
    for gap in gap_positions:
        centers = np.zeros((N_CIRCLES, 2))
        idx = 0
        for i in range(5):
            for j in range(5):
                centers[idx] = [0.1 + j * 0.2, 0.1 + i * 0.2]
                idx += 1
        centers[25] = gap
        radii = np.ones(N_CIRCLES) * 0.07
        inits.append((centers, radii))
    return inits


def init_hexagonal_like():
    """Initialize with hexagonal-like arrangement."""
    inits = []
    # Various row count patterns summing to 26
    row_patterns = [
        [6, 5, 6, 5, 4],
        [5, 6, 5, 6, 4],
        [5, 5, 6, 5, 5],
        [4, 6, 6, 6, 4],
        [5, 5, 5, 5, 6],
        [6, 6, 5, 5, 4],
    ]
    for pattern in row_patterns:
        centers = np.zeros((N_CIRCLES, 2))
        idx = 0
        num_rows = len(pattern)
        for row in range(num_rows):
            count = pattern[row]
            for col in range(count):
                # Hexagonal offset: alternating rows shifted by half spacing
                if row % 2 == 0:
                    x = (col + 0.5) / count
                else:
                    x = col / (count - 1) if count > 1 else 0.5
                y = (row + 0.5) / num_rows
                centers[idx] = [x, y]
                idx += 1
        radii = np.ones(N_CIRCLES) * 0.06
        inits.append((centers, radii))
    return inits


def init_perturbed_grid(rng_seed=42):
    """Initialize with perturbed 5x5 grid."""
    inits = []
    for seed in [42, 123, 456, 789, 1000]:
        rng = np.random.RandomState(seed)
        centers = np.zeros((N_CIRCLES, 2))
        idx = 0
        for i in range(5):
            for j in range(5):
                centers[idx] = [0.1 + j * 0.2 + rng.uniform(-0.02, 0.02),
                                0.1 + i * 0.2 + rng.uniform(-0.02, 0.02)]
                idx += 1
        centers[25] = [0.5 + rng.uniform(-0.1, 0.1), 0.5 + rng.uniform(-0.1, 0.1)]
        radii = np.ones(N_CIRCLES) * 0.07 + rng.uniform(-0.01, 0.01, N_CIRCLES)
        radii = np.maximum(radii, 0.02)
        inits.append((centers, radii))
    return inits


def init_corner_heavy():

    radii = np.zeros(N_CIRCLES)
    idx = 0
    # Corner circles slightly larger

    for pos in corner_positions:
        centers[idx] = pos
        radii[idx] = 0.15
        idx += 1
    # Remaining circles in grid-like fashion
    for i in range(4):
        for j in range(4):
            if idx >= N_CIRCLES:
                break
            centers[idx] = [0.2 + j * 0.2, 0.2 + i * 0.2]
            radii[idx] = 0.07
            idx += 1
    while idx < N_CIRCLES:
        centers[idx] = [0.5, 0.5]
        radii[idx] = 0.05
        idx += 1
    return [(centers, radii)]


def init_staggered_rows():
    """Initialize with staggered rows (hexagonal packing pattern)."""
    centers = np.zeros((N_CIRCLES, 2))
    radii = np.ones(N_CIRCLES) * 0.07
    idx = 0
    # 5 rows: 6,5,6,5,4

    for row in range(5):
        count = row_counts[row]
        for col in range(count):
            if row % 2 == 0:
                x = (col + 0.5) / 6.0
            else:
                x = (col + 0.5) / 5.0
            y = (row + 0.5) / 5.0
            centers[idx] = [x, y]
            idx += 1
    return [(centers, radii)]


def optimize_from_init(centers, radii):
    """Run optimization from a given initialization."""
    n = N_CIRCLES
    x0 = np.concatenate([centers.flatten(), radii])
    cons = {'type': 'ineq', 'fun': constraint_func}
    bounds = make_bounds()

    try:
        res = minimize(
            objective_func,
            x0,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 5000, 'ftol': 1e-15, 'disp': False}
        )
        return res
    except Exception:
        return None


def clamp_solution(centers, radii):
    """Clamp solution to ensure validity."""
    n = N_CIRCLES
    for i in range(n):
        radii[i] = min(radii[i], centers[i, 0], 1 - centers[i, 0],
                        centers[i, 1], 1 - centers[i, 1])
        radii[i] = max(radii[i], 0)
        centers[i, 0] = max(radii[i], min(1 - radii[i], centers[i, 0]))
        centers[i, 1] = max(radii[i], min(1 - radii[i], centers[i, 1]))

    # Fix overlaps by reducing radii
    for _ in range(10):
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                if dist < radii[i] + radii[j] - 1e-12:
                    overlap = (radii[i] + radii[j] - dist) / 2 + 1e-10
                    radii[i] = max(0, radii[i] - overlap)
                    radii[j] = max(0, radii[j] - overlap)
                    centers[i, 0] = max(radii[i], min(1 - radii[i], centers[i, 0]))
                    centers[i, 1] = max(radii[i], min(1 - radii[i], centers[i, 1]))
                    centers[j, 0] = max(radii[j], min(1 - radii[j], centers[j, 0]))
                    centers[j, 1] = max(radii[j], min(1 - radii[j], centers[j, 1]))
    return centers, radii


def run_packing():
    """Main function to pack 26 circles in unit square."""
    n = N_CIRCLES
    best_sum = 0.0
    best_centers = np.zeros((n, 2))
    best_radii = np.zeros(n)

    # Collect all initializations
