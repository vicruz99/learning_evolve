import numpy as np
from scipy.optimize import minimize
import math

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    Packs 26 circles in a unit square to maximize the sum of radii.
    
    Returns:
        centers: np.array of shape (26, 2)
        radii: np.array of shape (26,)
        sum_radii: float

    stiffness = 2000.0      # Strong repulsion to resolve overlaps quickly
    learning_rate = 0.02    # Step size for position updates

    max_iters = 3000

        diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        np.fill_diagonal(dists, np.inf)  # Ignore self-distance

        overlaps = r_sum_matrix - dists

        if total_overlap < 1e-7:
            radii += growth_step

            if step > 2000:
                radii += growth_step * 0.1 
        else:
            # If stuck with high overlap, ensure radii don't grow
            pass

        forces = np.zeros((n, 2))
        
        # Pairwise repulsion
        for i in range(n):
            for j in range(i + 1, n):

                dy = centers[i, 1] - centers[j, 1]
                d_sq = dx*dx + dy*dy
                d = math.sqrt(d_sq)
                
                r_i, r_j = radii[i], radii[j]
                
                if d < r_i + r_j:
                    # Overlap detected
                    # Force proportional to overlap depth
                    # Use a soft potential to avoid numerical instability at d=0
                    if d < 1e-9: d = 1e-9
                    
                    overlap_depth = (r_i + r_j) - d
                    # Repulsion force magnitude

                    # Direction: i away from j
                    fx = force_mag * dx / d
                    fy = force_mag * dy / d
                    
                    forces[i, 0] += fx
                    forces[i, 1] += fy
                    forces[j, 0] -= fx
                    forces[j, 1] -= fy

            r = radii[i]

                forces[i, 0] -= stiffness * (x - (1 - r))
            
            # Top/Bottom

                forces[i, 1] += stiffness * (r - y)
            elif y > 1 - r:
                forces[i, 1] -= stiffness * (y - (1 - r))

    final_radii = np.maximum(final_radii, 0)
    
    sum_radii = np.sum(final_radii)
    
    return final_centers, final_radii, sum_radii
