For packing problems of this type, the optimal configuration typically consists of circles with equal radii, as equal radii minimize the total area for a fixed sum of radii, allowing them to fit more densely. The target sum of 2.636 implies an optimal radius of approximately $r \approx 0.1014$.

2. **Expanding Circles Simulation**: Start with a small radius $r=0.01$. import numpy as np
from scipy.optimize import minimize

def compute_min_dist(centers):
    """Compute the minimum distance between any two circles or to boundaries."""
    n = centers.shape[0]
    md = 1.0
    for i in range(n):
        # Distance to boundaries
        md = min(md, centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])
        # Distance to other circles
        for j in range(i + 1, n):
            md = min(md, np.linalg.norm(centers[i] - centers[j]))
    return md

def neg_min_dist_obj(x, n):
    """Objective function for optimization: negative of minimum distance."""
    centers = x.reshape(-1, 2)
    return -compute_min_dist(centers)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # 1. Initialize centers in a hexagonal pattern
    rows = [6, 5, 6, 5, 4]
    s = 0.2  # horizontal spacing
    h = s * np.sqrt(3) / 2  # vertical spacing
    
    pts = []
    for r_idx, count in enumerate(rows):
        for c_idx in range(count):
            x = c_idx * s + (0.5 * s if r_idx % 2 == 1 else 0)
            y = r_idx * h
            pts.append([x, y])
    pts = np.array(pts)
    
    # Center and scale pattern to fit initially with margins
    min_pt = pts.min(axis=0)
    max_pt = pts.max(axis=0)
    centers = (pts - min_pt) / (max_pt - min_pt) * 0.8 + 0.1
    
    # 2. Expanding circles simulation
    r = 0.01
    lr = 1e-3
    
    for step in range(30000):
        forces = np.zeros_like(centers)
        for i in range(n):
            for j in range(i + 1, n):
                diff = centers[i] - centers[j]
                dist = np.linalg.norm(diff)
                if dist < 2 * r and dist > 1e-7:
                    force_mag = (2 * r - dist) * 50.0
                    forces[i] += diff * force_mag / dist
                    forces[j] -= diff * force_mag / dist
            
            for dim in range(2):
                if centers[i, dim] < r:
                    forces[i, dim] += (r - centers[i, dim]) * 50.0
                elif centers[i, dim] > 1 - r:
                    forces[i, dim] += (1 - r - centers[i, dim]) * 50.0
                    
        centers += forces * lr
        centers = np.clip(centers, r, 1 - r)
        
        if step % 20 == 0:
            r += 5e-6
        if step % 500 == 0:
            lr *= 0.95
            
    # 3. Polish with local optimization
    res = minimize(neg_min_dist_obj, centers.flatten(), args=(n,), 
                   method='Nelder-Mead', options={'maxiter': 15000, 'xatol': 1e-7, 'fatol': 1e-7})
    centers = res.x.reshape(-1, 2)
    
    # 4. Compute final valid radius
    md = compute_min_dist(centers)
    r_final = md / 2
    radii = np.full(n, r_final)
    
    return centers, radii, np.sum(radii)
