import numpy as np
import scipy.optimize as opt

def run_packing():
    """
    Returns the optimal packing of 26 circles in a unit square.
    """
    n_circles = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None

    # Function to solve for optimal radii given fixed centers using LP
    def get_optimal_radii(centers):
        # Variables: r_0, ..., r_25
        # Objective: Maximize sum(r_i) => Minimize -sum(r_i)
        c_obj = -np.ones(n_circles)

        # Constraints: A_ub @ r <= b_ub
        # 1. Boundary constraints: r_i <= x_i, r_i <= 1-x_i, etc.
        #    r_i - x_i <= 0  =>  r_i <= x_i
        #    -r_i <= 0 (non-negativity handled by bounds usually, but linprog defaults to 0)
        #    Actually linprog bounds (0, None) handles r >= 0.
        
        A_ub = []
        b_ub = []

        # Boundary constraints: r_i <= dist to wall
        # r_i <= x_i
        # -r_i <= -x_i  (Not needed if we just set upper bounds, but linprob uses inequalities)
        # Let's use bounds for variables to handle 0 <= r_i <= min_walls
        # And inequalities for r_i + r_j <= d_ij
        
        # We will construct the inequality matrix for r_i + r_j <= d_ij
        # There are n*(n-1)/2 such constraints.
        
        num_constraints = n_circles * (n_circles - 1) // 2
        A_ub = np.zeros((num_constraints, n_circles))
        b_ub = np.zeros(num_constraints)
        
        idx = 0
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                A_ub[idx, i] = 1.0
                A_ub[idx, j] = 1.0
                b_ub[idx] = dist
                idx += 1
        
        # Bounds for each radius: 0 <= r_i <= min(x_i, 1-x_i, y_i, 1-y_i)
        bounds = []
        for i in range(n_circles):
            x, y = centers[i]
            max_r = min(x, 1 - x, y, 1 - y)
            # Ensure non-negative max_r (numerical safety)
            if max_r < 0: max_r = 0
            bounds.append((0, max_r))
            
        try:
            res = opt.linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
            if res.success:
                return res.fun * -1, res.x # Return sum (negative of min result) and radii
            else:
                return 0.0, np.zeros(n_circles)
        except Exception:
            return 0.0, np.zeros(n_circles)

    # Objective function for center optimization
    def objective(center_params):
        centers = center_params.reshape((n_circles, 2))
        s, _ = get_optimal_radii(centers)
        return -s # Minimize negative sum

            row_x = 0.1 + offset
            while row_x <= 0.9:
                if len(pts) < n_circles:
                    pts.append([row_x, row_y])
                row_x += dx
            row_y += dy
        
        # If we have too few or too many, adjust or randomize slightly

        if len(pts) < n_circles:

            pts = pts[:n_circles]
            
        np.random.shuffle(pts)
        return np.array(pts[:n_circles])

    best_global_sum = 0.0
    best_global_config = None

    # Run optimization with multiple restarts
    for restart in range(10):
        # Initialize centers
        # Use a structured grid + noise
        rng = np.random.default_rng(restart * 42)

        try:

    # Final calculation of radii for best centers
    if best_global_config is not None:

    else:
        # Fallback
        final_radii = np.zeros(26)
        best_global_config = np.zeros((26, 2))

    return best_global_config, final_radii, float(best_global_sum)
