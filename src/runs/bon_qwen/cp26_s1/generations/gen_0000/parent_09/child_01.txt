import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Pack 26 circles in a unit square to maximize the sum of radii.
    
    Returns:
        centers: np.array of shape (26, 2) with (x, y) coordinates
        radii: np.array of shape (26) with radius of each circle
        sum_radii: float, the sum of the radii
    """
    n_circles = 26
    
    # 1. Initialization: Hexagonal lattice pattern
    # Estimate spacing for N=26 in unit square
    # Density approx: N * pi * r^2 ~ 1 * packing_efficiency
    # Let's try a grid that fits 26 points
    rows = 6
    # Distribution of circles per row to get 26
    # 4, 5, 4, 5, 4, 4 = 26
    row_counts = [4, 5, 4, 5, 4, 4]
    
    centers = []
    y_step = 1.0 / (rows + 1) # Simple spacing
    
    current_y = y_step
    for r_idx, count in enumerate(row_counts):
        x_step = 1.0 / (count + 1)
        # Offset every other row for hexagonal pattern
        offset = x_step / 2 if r_idx % 2 == 1 else 0
        for c in range(count):
            x = (c + 1) * x_step + offset
            y = current_y
            # Clamp to [0.1, 0.9] to avoid walls initially
            x = np.clip(x, 0.1, 0.9)
            y = np.clip(y, 0.1, 0.9)
            centers.append([x, y])
        
        current_y += y_step
    
    centers = np.array(centers)
    
    # Initial radii: small value
    radii_init = np.full(n_circles, 0.05)
    
    # Combine variables: x, y, r for each circle
    # Shape: (26, 3) -> flatten to 1D vector for optimizer
    x0 = np.zeros(n_circles * 3)
    for i in range(n_circles):
        x0[3*i] = centers[i, 0]
        x0[3*i+1] = centers[i, 1]
        x0[3*i+2] = radii_init[i]
        
    # Bounds: x, y in [0, 1], r in [0, 0.5]
    bounds = []
    for i in range(n_circles):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 0.5)) # r

    def objective(vars_flat):
        # Maximize sum of radii -> Minimize negative sum
        r = vars_flat[2::3]
        return -np.sum(r)

    def get_centers_radii(vars_flat):
        c = vars_flat.reshape(-1, 3)
        return c[:, :2], c[:, 2]

    def constraint_wall(vars_flat):
        # Returns vector of constraints >= 0
        c, r = get_centers_radii(vars_flat)
        # x - r >= 0 => r - x <= 0 (not this form)
        # We need c(x) >= 0 for minimize constraints usually?
        # scipy minimize constraints: fun(x) >= 0
        # x - r >= 0
        # 1 - x - r >= 0
        # y - r >= 0
        # 1 - y - r >= 0
        
        cons = []
        # Left
        cons.append(c[:, 0] - r)
        # Right
        cons.append(1.0 - c[:, 0] - r)
        # Bottom
        cons.append(c[:, 1] - r)
        # Top
        cons.append(1.0 - c[:, 1] - r)
        return np.concatenate(cons)

    def constraint_no_overlap(vars_flat):
        # dist(i, j) >= r_i + r_j
        # dist - (r_i + r_j) >= 0
        c, r = get_centers_radii(vars_flat)
        n = c.shape[0]
        constraints = []
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((c[i] - c[j])**2))
                constraints.append(dist - (r[i] + r[j]))
        return np.array(constraints)

    # Define constraints for scipy
    # Inequality constraints: fun(x) >= 0
    constr_wall = {
        'type': 'ineq',
        'fun': constraint_wall,
        'jac': '2-point' # Use numerical gradient
    }
    
    constr_overlap = {
        'type': 'ineq',
        'fun': constraint_no_overlap,
        'jac': '2-point'
    }
    
    constraints = [constr_wall, constr_overlap]

    # Run optimization
    # SLSQP is suitable for non-linear constraints
    res = minimize(objective, x0, method='SLSQP', bounds=bounds, 
                   constraints=constraints, options={'maxiter': 500, 'ftol': 1e-8})
    
    # Extract results
    final_vars = res.x
    final_centers = final_vars.reshape(-1, 3)[:, :2]
    final_radii = final_vars.reshape(-1, 3)[:, 2]
    
    # Ensure radii are non-negative and valid (numerical safety)
    final_radii = np.maximum(final_radii, 0.0)
    
    sum_radii = np.sum(final_radii)
    
    return final_centers, final_radii, sum_radii
