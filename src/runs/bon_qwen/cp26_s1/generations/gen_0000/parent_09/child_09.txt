import numpy as np
from scipy.optimize import minimize

N = 26

def objective(v):
    return -np.sum(v[2::3])

def constraint_func(v):
    cons = []
    for i in range(N):
        xi = v[3*i]
        yi = v[3*i+1]
        ri = v[3*i+2]
        # Boundary constraints: circle must be inside [0,1]x[0,1]
        cons.append(xi - ri)
        cons.append(1.0 - xi - ri)
        cons.append(yi - ri)
        cons.append(1.0 - yi - ri)
        # Non-overlap constraints: distance between centers >= sum of radii
        for j in range(i + 1, N):
            xj = v[3*j]
            yj = v[3*j+1]
            rj = v[3*j+2]
            dx = xi - xj
            dy = yi - yj
            dist = np.sqrt(dx*dx + dy*dy)
            cons.append(dist - ri - rj)
    return np.array(cons)

def run_packing():
    np.random.seed(42)
    
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N
    con = {'type': 'ineq', 'fun': constraint_func}

    # Hexagonal lattice initialization for fast convergence
    v0 = np.zeros(3 * N)
    idx = 0
    row_counts = [6, 5, 6, 5, 4]
    dx = 0.16
    dy = dx * np.sqrt(3) / 2
    y_start = 0.15

    for r_idx, count in enumerate(row_counts):
        row_width = (count - 1) * dx
        start_x = (1.0 - row_width) / 2.0
        if r_idx % 2 == 1:
            start_x += dx / 2.0
        y_pos = y_start + r_idx * dy
        for c in range(count):
            x_pos = start_x + c * dx
            v0[3*idx] = np.clip(x_pos, 0.05, 0.95)
            v0[3*idx+1] = np.clip(y_pos, 0.05, 0.95)
            v0[3*idx+2] = 0.06
            idx += 1

    best_v = v0.copy()
    best_val = objective(v0)

    # Multiple restarts with jitter to escape local optima
    for _ in range(5):
        v_start = best_v.copy()
        noise = np.random.normal(0, 0.02, v_start.shape)
        v_start += noise
        v_start[::3] = np.clip(v_start[::3], 0.01, 0.99)
        v_start[1::3] = np.clip(v_start[1::3], 0.01, 0.99)
        v_start[2::3] = np.clip(v_start[2::3], 0.01, 0.4)

        res = minimize(objective, v_start, method='SLSQP', bounds=bounds,
                       constraints=con, options={'maxiter': 2000, 'ftol': 1e-10})
        if res.fun < best_val:
            best_v = res.x
            best_val = res.fun

    centers = best_v[:3*N:3].reshape(N, 2)
    radii = best_v[2::3]
    
    # Post-processing to ensure strict feasibility and clean numerical values
    radii = np.maximum(radii, 1e-9)
    centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
    centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)

    sum_r = np.sum(radii)
    return centers, radii, sum_r
