import numpy as np
from scipy.optimize import minimize


def objective(x, n):
    """Maximize sum of radii (minimize negative sum)"""
    return -np.sum(x[2 * n:])


def constraints_vector(x, n):
    """Compute all constraint values (all must be >= 0 for feasibility)"""
    result = []

    # Boundary constraints for each circle
    for i in range(n):
        xi = x[2 * i]
        yi = x[2 * i + 1]
        ri = x[2 * n + i]

        # x >= r
        result.append(xi - ri)
        # x + r <= 1
        result.append(1.0 - xi - ri)
        # y >= r
        result.append(yi - ri)
        # y + r <= 1
        result.append(1.0 - yi - ri)

    # Non-overlap constraints for each pair
    for i in range(n):
        xi = x[2 * i]
        yi = x[2 * i + 1]
        ri = x[2 * n + i]
        for j in range(i + 1, n):
            xj = x[2 * j]
            yj = x[2 * j + 1]
            rj = x[2 * n + j]

            dx = xi - xj
            dy = yi - yj
            dist = np.sqrt(dx * dx + dy * dy)
            result.append(dist - ri - rj)

    return np.array(result)


def generate_hexagonal_init(n, r_base):
    """Generate hexagonal lattice initial configuration"""
    # 5 rows for 26 circles: 6+5+6+5+4 = 26
    row_counts = [6, 5, 6, 5, 4]
    centers = []
    radii = []

    for row in range(len(row_counts)):
        y = r_base + row * r_base * np.sqrt(3)
        x_offset = (row % 2) * r_base
        for col in range(row_counts[row]):
            x = r_base + x_offset + col * 2 * r_base
            centers.append([x, y])
            radii.append(r_base)

    return np.array(centers), np.array(radii)


def generate_grid_init(n, r_base):
    """Generate grid-based initial configuration"""
    # Try 5x5 grid plus 1 extra
    centers = []
    radii = []

    # 5x5 grid
    x_positions = np.linspace(r_base, 1.0 - r_base, 5)
    y_positions = np.linspace(r_base, 1.0 - r_base, 5)

    for yi in y_positions:
        for xi in x_positions:
            centers.append([xi, yi])
            radii.append(r_base)

    # Add 6th circle in a row with smaller radius
    extra_r = r_base * 0.7
    centers.append([r_base, (y_positions[0] + y_positions[1]) / 2])
    radii.append(extra_r)

    return np.array(centers[:n]), np.array(radii[:n])


def penalty_objective(x, n, lam):
    """Penalty-based objective for more robust optimization"""
    centers = x[:2 * n].reshape(n, 2)
    radii = x[2 * n:]

    penalty = 0.0

    # Boundary penalties
    for i in range(n):
        xi = centers[i, 0]
        yi = centers[i, 1]
        ri = radii[i]

        if ri < 0:
            penalty += lam * ri ** 2
        if xi - ri < 0:
            penalty += lam * (xi - ri) ** 2
        if xi + ri > 1:
            penalty += lam * (xi + ri - 1) ** 2
        if yi - ri < 0:
            penalty += lam * (yi - ri) ** 2
        if yi + ri > 1:
            penalty += lam * (yi + ri - 1) ** 2

    # Overlap penalties
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            min_dist = radii[i] + radii[j]
            if dist < min_dist:
                penalty += lam * (min_dist - dist) ** 2

    return -np.sum(radii) + penalty


def optimize_with_constraints(n, x0, max_iter=30000):
    """Optimize using constraint-based method"""
    bounds = [(0.0, 1.0)] * (2 * n) + [(1e-10, 0.5)] * n

    constraint = {
        'type': 'ineq',
        'fun': constraints_vector,
        'args': (n,)
    }

    result = minimize(
        objective,
        x0,
        args=(n,),
        bounds=bounds,
        constraints=constraint,
        method='SLSQP',
        options={'maxiter': max_iter, 'ftol': 1e-16, 'disp': False}
    )

    return result


def optimize_with_penalty(n, x0, lam):
    """Optimize using penalty method"""
    bounds = [(0.0, 1.0)] * (2 * n) + [(1e-10, 0.5)] * n

    result = minimize(
        penalty_objective,
        x0,
        args=(n, lam),
        bounds=bounds,
        method='L-BFGS-B',
        options={'maxiter': 20000, 'ftol': 1e-16}
    )

    return result


def refine_packing(n, centers, radii):
    """Refine packing by alternating position and radius optimization"""
    best_sum = np.sum(radii)
    best_centers = centers.copy()
    best_radii = radii.copy()

    for iteration in range(5):
        # Perturb and reoptimize
        x0 = np.concatenate([best_centers.flatten(), best_radii])

        # Add small perturbation
        perturbation = np.random.normal(0, 0.005, size=len(x0))
        x0_perturbed = x0 + perturbation
        x0_perturbed[:2 * n] = np.clip(x0_perturbed[:2 * n], 0.01, 0.99)
        x0_perturbed[2 * n:] = np.maximum(x0_perturbed[2 * n:], 1e-6)

        result = optimize_with_constraints(n, x0_perturbed, max_iter=20000)

        current_sum = -result.fun
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = result.x[:2 * n].reshape(n, 2)
            best_radii = result.x[2 * n:]

    return best_centers, best_radii, best_sum
