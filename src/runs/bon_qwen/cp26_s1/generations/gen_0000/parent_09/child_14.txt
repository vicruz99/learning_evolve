from scipy.optimize import minimize

def objective(x):
    n = 26
    radii = x[2*n:]
    return -np.sum(radii)

def boundary_constraint_fun(x):
    n = 26
    centers = x[:2*n].reshape(n, 2)
    radii = x[2*n:]
    
    cons = []
    for i in range(n):
        cons.append(centers[i, 0] - radii[i])
        cons.append(1.0 - centers[i, 0] - radii[i])
        cons.append(centers[i, 1] - radii[i])
        cons.append(1.0 - centers[i, 1] - radii[i])
    return np.array(cons)

def non_overlap_constraint_fun(x):
    n = 26
    centers = x[:2*n].reshape(n, 2)
    radii = x[2*n:]
    
    cons = []
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist_sq = dx * dx + dy * dy
            min_dist_sq = (radii[i] + radii[j]) ** 2
            cons.append(dist_sq - min_dist_sq)
    return np.array(cons)

def make_hexagonal_initial():
    n = 26
    x0 = np.zeros(3 * n)
    r_init = 0.04
    row_counts = [5, 6, 5, 6, 4]
    idx = 0
    for row, count in enumerate(row_counts):
        y = r_init + row * r_init * math.sqrt(3)
        for col in range(count):
            x = r_init + (col + (row % 2) * 0.5) * 2 * r_init
            x0[2 * idx] = x
            x0[2 * idx + 1] = y
            x0[2 * n + idx] = r_init
            idx += 1
    return x0

def optimize_from(x0):
    n = 26
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
    
    constraints = [
        {"type": "ineq", "fun": boundary_constraint_fun},
        {"type": "ineq", "fun": non_overlap_constraint_fun}
    ]
    
    try:
        result = minimize(
            objective, x0, method="SLSQP", bounds=bounds,
            constraints=constraints,
            options={"maxiter": 3000, "ftol": 1e-14, "disp": False}
        )
        return result
    except Exception:
        return None

def extract_result(result, n=26):
    if result is None:
        return None, None, -1.0
    
    centers = result.x[:2*n].reshape(n, 2)
    radii = result.x[2*n:]
    
    radii = np.maximum(radii, 0.0)
    centers[:, 0] = np.clip(centers[:, 0], 0.0, 1.0)
    centers[:, 1] = np.clip(centers[:, 1], 0.0, 1.0)
    
    for i in range(n):
        r = radii[i]
        centers[i, 0] = np.clip(centers[i, 0], r, 1.0 - r)
        centers[i, 1] = np.clip(centers[i, 1], r, 1.0 - r)
    
    return centers, radii, np.sum(radii)
