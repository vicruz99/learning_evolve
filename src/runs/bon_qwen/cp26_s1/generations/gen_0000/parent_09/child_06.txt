import numpy as np
from scipy.optimize import minimize

def compute_cost(vars, n, lam):
    centers = vars[:2*n].reshape(n, 2)
    radii = vars[2*n:]
    
    # Objective: maximize sum of radii -> minimize negative sum
    obj = -np.sum(radii)
    
    # Boundary penalty: circles must stay within [0,1]x[0,1]
    pen = 0.0
    pen += np.sum(np.maximum(0.0, radii - centers[:, 0])**2)
    pen += np.sum(np.maximum(0.0, radii - (1.0 - centers[:, 0]))**2)
    pen += np.sum(np.maximum(0.0, radii - centers[:, 1])**2)
    pen += np.sum(np.maximum(0.0, radii - (1.0 - centers[:, 1]))**2)
    
    # Overlap penalty: distance between centers must be >= sum of radii
    c_diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist = np.sqrt(np.sum(c_diff**2, axis=2))
    r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    
    # Consider only upper triangle to avoid double counting and self-interaction
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    violation = r_sum - dist
    pen += np.sum(np.maximum(0.0, violation[mask])**2)
    
    return obj + lam * pen

def run_packing():
    np.random.seed(42)
    n = 26
    
    # Initialize with a hexagonal lattice layout
    r_init = 0.09
    h = r_init * np.sqrt(3)
    coords = []
    for i in range(6):
        y = r_init + i * h
        if y + r_init > 1.0:
            break

        x = x_start
        while x + r_init <= 1.0:
            coords.append([x, y, r_init])
            x += 2 * r_init
        if len(coords) >= n:
            break
            
    # Fallback padding if hex layout generated fewer than n circles
    while len(coords) < n:
        coords.append([0.5, 0.5, 0.05])
        
    init_pts = np.array(coords[:n])
    x0 = np.concatenate([init_pts[:, :2].flatten(), init_pts[:, 2]])
    
    # Bounds: centers in [0,1], radii in [1e-6, 0.5]
    bounds = [(0.0, 1.0) for _ in range(2*n)] + [(1e-6, 0.5) for _ in range(n)]
    
    x_curr = x0.copy()
    
    # Continuation method: gradually increase penalty weight to enforce constraints
    for lam in [10.0, 50.0, 200.0, 1000.0, 5000.0]:
        # Small noise to help escape local minima, decaying with lambda
        noise = np.random.randn(len(x_curr)) * 0.002 * (10.0 / lam)
        x_curr = x_curr + noise
        
        res = minimize(compute_cost, x_curr, args=(n, lam), method='L-BFGS-B', 
                       bounds=bounds, options={'ftol': 1e-12, 'gtol': 1e-8, 'maxiter': 3000})
        x_curr = res.x
        
    centers = x_curr[:2*n].reshape(n, 2)
    radii = x_curr[2*n:]
    return centers, radii, np.sum(radii)
