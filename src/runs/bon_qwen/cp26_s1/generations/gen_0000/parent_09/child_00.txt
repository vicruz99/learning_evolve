import numpy as np
from scipy.optimize import minimize

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    n = centers.shape[0]
    if np.isnan(centers).any() or np.isnan(radii).any():
        return False
    for i in range(n):
        if radii[i] < 0: return False
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            return False
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False
    return True

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    best_sum = 0.0
    best_centers = np.zeros((n, 2))
    best_radii = np.zeros(n)

    for trial in range(20):
        # Random seed for each trial
        seed = trial * 123 + 456
        rng = np.random.RandomState(seed)
        
        # 1. Initialize centers: Randomly place points, then expand radii to boundaries initially
        # Or use a grid-based initialization for better starting geometry
        # Using a perturbed grid approach:
        # Place 5x5 grid plus 1 in center
        centers_init = np.zeros((n, 2))
        radii_init = np.ones(n) * 0.1
        
        # Fill 5x5 grid
        count = 0
        for r in range(5):
            for c in range(5):
                if count < n:
                    centers_init[count] = [0.1 + c * 0.2 + rng.uniform(-0.01, 0.01), 
                                           0.1 + r * 0.2 + rng.uniform(-0.01, 0.01)]
                    count += 1
        # Fill remaining circles in gaps if needed (though n=26 fits in 5x5+1 logic roughly)
        while count < n:
            centers_init[count] = [rng.uniform(0.1, 0.9), rng.uniform(0.1, 0.9)]
            count += 1
            
        # Initial radii
        radii_init = np.ones(n) * 0.09  # Start slightly smaller to allow optimization

        # 2. Define optimization variables
        # We optimize [x0, y0, r0, x1, y1, r1, ..., x25, y25, r25]
        x0 = np.zeros(3 * n)
        for i in range(n):
            x0[3*i] = centers_init[i, 0]
            x0[3*i+1] = centers_init[i, 1]
            x0[3*i+2] = radii_init[i]

        # 3. Constraints
        # Bounds: x, y in [0, 1], r in [0, 1]
        # Actually bounds can be tighter: r <= 0.5, but SLSQP handles [0, 1] fine
        bounds = [(0, 1)] * (2 * n) + [(0, 1)] * n 

        cons = []

        # Non-overlap constraints: dist^2 >= (r_i + r_j)^2
        # (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
        for i in range(n):
            for j in range(i + 1, n):
                idx_i_x, idx_i_y, idx_i_r = 3*i, 3*i+1, 3*i+2
                idx_j_x, idx_j_y, idx_j_r = 3*j, 3*j+1, 3*j+2
                
                def constraint_overlap(vars, i=i, j=j):
                    xi, yi, ri = vars[idx_i_x], vars[idx_i_y], vars[idx_i_r]
                    xj, yj, rj = vars[idx_j_x], vars[idx_j_y], vars[idx_j_r]
                    dist_sq = (xi - xj)**2 + (yi - yj)**2
                    return dist_sq - (ri + rj)**2

                cons.append({'type': 'ineq', 'fun': constraint_overlap})

        # Boundary constraints:
        # x - r >= 0  => r - x <= 0 (handled by 'leq' or 'ineq' of -val)
        # x + r <= 1  => x + r - 1 <= 0
        # same for y
        # SLSQP 'ineq' expects function >= 0.
        # So: x - r >= 0
        #    1 - x - r >= 0
        
        for i in range(n):
            idx_x, idx_y, idx_r = 3*i, 3*i+1, 3*i+2
            
            # x - r >= 0
            def left_boundary(vars, i=i):
                return vars[3*i] - vars[3*i+2]
            cons.append({'type': 'ineq', 'fun': left_boundary})
            
            # 1 - x - r >= 0
            def right_boundary(vars, i=i):
                return 1.0 - vars[3*i] - vars[3*i+2]
            cons.append({'type': 'ineq', 'fun': right_boundary})

            # y - r >= 0
            def bottom_boundary(vars, i=i):
                return vars[3*i+1] - vars[3*i+2]
            cons.append({'type': 'ineq', 'fun': bottom_boundary})

            # 1 - y - r >= 0
            def top_boundary(vars, i=i):
                return 1.0 - vars[3*i+1] - vars[3*i+2]
            cons.append({'type': 'ineq', 'fun': top_boundary})

        # Objective: Maximize sum(r_i) -> Minimize -sum(r_i)
        def objective(vars):
            return -np.sum(vars[2::3]) # Sum of radii

        # 4. Optimize
        try:
            # Options for convergence
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, 
                           options={'maxiter': 500, 'ftol': 1e-9})
            
            if res.success:
                opt_vals = res.x
                centers = np.zeros((n, 2))
                radii = np.zeros(n)
                for i in range(n):
                    centers[i, 0] = opt_vals[3*i]
                    centers[i, 1] = opt_vals[3*i+1]
                    radii[i] = opt_vals[3*i+2]
                
                # Validate
                if validate_packing(centers, radii):
                    current_sum = np.sum(radii)
                    if current_sum > best_sum:
                        best_sum = current_sum
                        best_centers = centers
                        best_radii = radii
        except Exception:
            continue

    return best_centers, best_radii, best_sum
