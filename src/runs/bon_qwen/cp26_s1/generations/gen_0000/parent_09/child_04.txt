from scipy.optimize import minimize, linprog

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    n = centers.shape[0]
    if np.isnan(centers).any() or np.isnan(radii).any():
        return False

        if radii[i] < 0:
            return False
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            return False

        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False
    return True

def compute_optimal_radii(centers):
    """
    Given fixed centers, solve the LP to find the optimal radii 
    that maximize sum(radii) without overlapping.
    """
    n = centers.shape[0]
    
    # Calculate boundary distances for each center
    # r_i <= min(x, 1-x, y, 1-y)
    x = centers[:, 0]
    y = centers[:, 1]
    max_r_boundary = np.minimum(np.minimum(x, 1 - x), np.minimum(y, 1 - y))
    
    # Calculate pairwise distances
    # dist(i, j) = ||c_i - c_j||
    # Constraint: r_i + r_j <= dist(i, j)
    # Matrix form: A_ub @ r <= b_ub
    
    # We have n variables (radii).
    # Constraints:
    # 1. r_i <= max_r_boundary[i]  (n constraints)
    # 2. r_i + r_j <= dist(i, j)   (n*(n-1)/2 constraints)
    # 3. r_i >= 0                  (handled by bounds)

    num_constraints = n + n * (n - 1) // 2
    A_ub = np.zeros((num_constraints, n))
    b_ub = np.zeros(num_constraints)
    
    # Boundary constraints: 1 * r_i <= max_r_boundary[i]
    for i in range(n):
        A_ub[i, i] = 1.0
        b_ub[i] = max_r_boundary[i]
    
    # Overlap constraints: r_i + r_j <= dist(i, j)
    constraint_idx = n
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            A_ub[constraint_idx, i] = 1.0
            A_ub[constraint_idx, j] = 1.0
            b_ub[constraint_idx] = dist
            constraint_idx += 1
            
    # Objective: maximize sum(radii) -> minimize -sum(radii)
    c_obj = -np.ones(n)
    
    # Bounds: 0 <= r_i <= 1 (upper bound 1 is loose but safe)
    bounds = [(0, 1)] * n
    
    # Solve LP
    # Method 'highs' is robust and fast for dense problems
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
        else:
            # Fallback to small radii if LP fails (rare)
            return np.full(n, 0.01)
    except Exception:
        return np.full(n, 0.01)

        return 1e6
        
    return -np.sum(radii)
