import numpy as np
from scipy.optimize import minimize

def objective(x, n):
    """Minimize negative sum of radii -> Maximize sum of radii"""
    return -np.sum(x[2::3])

def constr_bounds(x, n):
    """Boundary constraints: circles must stay inside [0,1]^2"""
    x_c = x[0::3]
    y_c = x[1::3]
    r = x[2::3]
    
    # x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
    return np.concatenate([
        x_c - r,
        1.0 - x_c - r,
        y_c - r,
        1.0 - y_c - r
    ])

def constr_overlap(x, n):
    """Non-overlap constraints: dist_sq >= (r_i + r_j)^2"""
    X = x[0::3]
    Y = x[1::3]
    R = x[2::3]
    
    # Pairwise differences using broadcasting
    dx = X[:, None] - X[None, :]
    dy = Y[:, None] - Y[None, :]
    r_sum = R[:, None] + R[None, :]
    
    dist_sq = dx**2 + dy**2
    r_sq = r_sum**2
    
    # Extract upper triangular part (i < j)
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    return (dist_sq - r_sq)[mask]

def run_packing():
    n = 26
    np.random.seed(42)
    
    # Generate initial grid layout
    pts = []
    for i in range(5):
        for j in range(5):
            pts.append([(j + 0.5) / 5.0, (i + 0.5) / 5.0])
    pts.append([0.5, 0.5])  # 26th circle
    pts = np.array(pts[:n])
    
    # Add controlled jitter to break symmetry and escape trivial configurations
    pts += np.random.uniform(-0.015, 0.015, size=pts.shape)
    pts = np.clip(pts, 0.15, 0.85)
    
    # Flatten initial guess: [x1, y1, r1, x2, y2, r2, ...]
    x0 = np.zeros(3 * n)
    x0[0::3] = pts[:, 0]
    x0[1::3] = pts[:, 1]
    x0[2::3] = 0.035 * np.ones(n)
    
    # Variable bounds
    bounds = [(0.0, 1.0) for _ in range(2 * n)] + [(1e-6, 0.5) for _ in range(n)]
    
    # Constraints setup
    cons = [
        {'type': 'ineq', 'fun': constr_bounds, 'args': (n,)},
        {'type': 'ineq', 'fun': constr_overlap, 'args': (n,)}
    ]
    
    best_x = None
    best_val = np.inf
    
    # Multi-start optimization to avoid local minima
    n_trials = 6
    for trial in range(n_trials):
        x_init = x0.copy()
        if trial > 0:
            # Perturb initial guess for subsequent trials
            x_init += np.random.normal(0, 0.025, size=3 * n)
            x_init = np.clip(x_init, 0.0, 1.0)
            x_init[2::3] = np.maximum(x_init[2::3], 0.02)
            
        try:
            res = minimize(
                objective, 
                x_init, 
                args=(n,), 
                method='SLSQP', 
                bounds=bounds,
                constraints=cons, 
                options={'maxiter': 600, 'ftol': 1e-13, 'disp': False}
            )
            
            if res.fun < best_val:
                best_val = res.fun
                best_x = res.x
        except Exception:
            continue
            
    if best_x is None:
        best_x = x0
        
    centers = np.column_stack((best_x[0::3], best_x[1::3]))
    radii = np.maximum(best_x[2::3], 0.0)
    
    return centers, radii, np.sum(radii)
