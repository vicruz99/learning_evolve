import numpy as np
import scipy.optimize as opt
import math

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Pack 26 circles in a unit square to maximize sum of radii.
    Returns (centers, radii, sum_radii).
    """
    n_circles = 26
    
    # Helper to flatten/unflatten variables
    def unpack(x):
        # x is size 78: [x1, y1, r1, x2, y2, r2, ...]
        centers = np.zeros((n_circles, 2))
        radii = np.zeros(n_circles)
        for i in range(n_circles):
            centers[i, 0] = x[3 * i]
            centers[i, 1] = x[3 * i + 1]
            radii[i] = x[3 * i + 2]
        return centers, radii

    def pack(centers, radii):
        x = np.zeros(3 * n_circles)
        for i in range(n_circles):
            x[3 * i] = centers[i, 0]
            x[3 * i + 1] = centers[i, 1]
            x[3 * i + 2] = radii[i]
        return x

    # Objective function: Maximize sum of radii -> Minimize negative sum
    def objective(x):
        _, radii = unpack(x)
        return -np.sum(radii)

    # Constraints
    def get_constraints():
        cons = []
        
        # Boundary constraints for each circle
        for i in range(n_circles):
            # x >= r  => x - r >= 0
            cons.append({'type': 'ineq', 'fun': lambda x, i=i: x[3 * i] - x[3 * i + 2]})
            # x <= 1 - r => 1 - r - x >= 0
            cons.append({'type': 'ineq', 'fun': lambda x, i=i: 1.0 - x[3 * i + 2] - x[3 * i]})
            # y >= r => y - r >= 0
            cons.append({'type': 'ineq', 'fun': lambda x, i=i: x[3 * i + 1] - x[3 * i + 2]})
            # y <= 1 - r => 1 - r - y >= 0
            cons.append({'type': 'ineq', 'fun': lambda x, i=i: 1.0 - x[3 * i + 2] - x[3 * i + 1]})
            
            # Non-negative radius (already handled by bounds, but good for safety)
            # r >= 0
            # cons.append({'type': 'ineq', 'fun': lambda x, i=i: x[3 * i + 2]})

        # Non-overlap constraints: dist >= r1 + r2
        # dist^2 >= (r1 + r2)^2
        # (x1-x2)^2 + (y1-y2)^2 - (r1+r2)^2 >= 0
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                cons.append({
                    'type': 'ineq',
                    'fun': lambda x, i=i, j=j: 
                        (x[3 * i] - x[3 * j])**2 + (x[3 * i + 1] - x[3 * j + 1])**2 - 
                        (x[3 * i + 2] + x[3 * j + 2])**2
                })
        return cons

    constraints = get_constraints()
    
    # Bounds for variables: x, y in [0, 1], r in [0, 0.5]
    bounds = []
    for i in range(n_circles):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((1e-6, 0.5)) # r (strictly positive)

    best_x = None
    best_val = -np.inf
    
    # Strategy: Multiple restarts with different initial configurations
    
    # 1. Grid-based initialization (5x5 + 1)
    def init_grid():
        centers = np.zeros((n_circles, 2))
        radii = np.ones(n_circles) * 0.05 # Initial small radius
        
        # Try to place in a grid
        # 5x5 grid points
        grid_points = []
        for r in range(5):
            for c in range(5):
                # Centers spaced by 0.2, starting at 0.1
                # But we have 26 circles.
                grid_points.append((0.1 + c * 0.2, 0.1 + r * 0.2))
        
        # We have 25 points. Add one more.
        # Place the 26th in the center of a cell or edge
        grid_points.append((0.5, 0.5))
        
        # Shuffle to make it more random? No, order doesn't matter for final state much
        # But let's just take first 26
        for i in range(n_circles):
            centers[i] = grid_points[i]
            
        # Scale radii to fit roughly
        # Max radius in 5x5 grid is 0.1. With 26 circles, maybe 0.08?
        radii = np.ones(n_circles) * 0.06 
        
        return pack(centers, radii)

    # 2. Hexagonal-like initialization
    def init_hex():
        centers = np.zeros((n_circles, 2))
        radii = np.ones(n_circles) * 0.05
        
        # Hexagonal packing approximation
        # Rows offset
        idx = 0
        y = 0.1
        row = 0
        while idx < n_circles and y <= 0.9:
            # Determine x step
            x_step = 0.2
            if row % 2 == 1:
                x_start = 0.2 # Offset
            else:
                x_start = 0.1
            
            x = x_start
            while x <= 0.9 and idx < n_circles:
                centers[idx] = [x, y]
                idx += 1
                x += x_step
            
            y += 0.15 # Vertical step for hex packing approx
            row += 1
        
        # If we didn't fill, fill remaining randomly
        while idx < n_circles:
            cx = np.random.uniform(0.1, 0.9)
            cy = np.random.uniform(0.1, 0.9)
            centers[idx] = [cx, cy]
            idx += 1
            
        radii = np.ones(n_circles) * 0.06
        return pack(centers, radii)

    # 3. Random initialization
    def init_random():
        centers = np.random.uniform(0.1, 0.9, (n_circles, 2))
        radii = np.ones(n_circles) * 0.05
        return pack(centers, radii)

    initial_guesses = [init_grid(), init_hex(), init_random(), init_random(), init_random()]
    
    for i, x0 in enumerate(initial_guesses):
        try:
            # Run optimization
            res = opt.minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints,
                               options={'ftol': 1e-9, 'maxiter': 1000, 'disp': False})

                        valid = False
                        break
                
                if valid:
                    # Overlap check

                        for m in range(k + 1, n_circles):
                            dist = np.linalg.norm(centers_opt[k] - centers_opt[m])
                            if dist < radii_opt[k] + radii_opt[m] - 1e-9:
                                valid = False
                                break
                        if not valid: break
                
                if valid:
                    current_sum = np.sum(radii_opt)
                    if current_sum > best_val:
                        best_val = current_sum
                        best_x = res.x.copy()
                        # print(f"New best: {current_sum}")

        except Exception as e:
            pass

    if best_x is not None:
        centers, radii = unpack(best_x)
    else:
        # Fallback to grid if optimization failed
        centers, radii = unpack(init_grid())

    # Post-optimization shrinkage to ensure strict validity with tolerance
    # The solver might be on the edge. Shrink radii slightly.
    # However, we want to maximize sum. The solver should have pushed to limit.
    # The validation function allows 1e-12 error.
    # Let's just return the result.
    
    sum_radii = np.sum(radii)
    return centers, radii, float(sum_radii)
