import numpy as np
from scipy.optimize import minimize, linprog

def get_optimal_radii(centers):
    """
    Solves the Linear Program to find the optimal radii for a fixed set of centers.
    Maximizes sum(r_i) subject to packing and boundary constraints.
    """
    n = centers.shape[0]
    
    # Objective: maximize sum(r_i) => minimize -sum(r_i)
    c_obj = np.ones(n)
    
    # Bounds for radii: r_i >= 0
    bounds = [(0, None) for _ in range(n)]
    
    # Inequality constraints: A_ub @ r <= b_ub
    constraints_A = []
    constraints_b = []
    
    # Boundary constraints:
    # r_i <= x_i
    # r_i <= 1 - x_i
    # r_i <= y_i
    # r_i <= 1 - y_i
    for i in range(n):
        x, y = centers[i]
        
        # r_i <= x_i  =>  1*r_i <= x_i
        row = np.zeros(n)
        row[i] = 1
        constraints_A.append(row)
        constraints_b.append(x)
        
        # r_i <= 1 - x_i  =>  1*r_i <= 1 - x_i
        row = np.zeros(n)
        row[i] = 1
        constraints_A.append(row)
        constraints_b.append(1 - x)
        
        # r_i <= y_i  =>  1*r_i <= y_i
        row = np.zeros(n)
        row[i] = 1
        constraints_A.append(row)
        constraints_b.append(y)
        
        # r_i <= 1 - y_i  =>  1*r_i <= 1 - y_i
        row = np.zeros(n)
        row[i] = 1
        constraints_A.append(row)
        constraints_b.append(1 - y)
        
    # Pairwise distance constraints:
    # r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            row = np.zeros(n)
            row[i] = 1
            row[j] = 1
            constraints_A.append(row)
            constraints_b.append(dist)
            
    A_ub = np.array(constraints_A)
    b_ub = np.array(constraints_b)
    
    # Solve LP
    res = linprog(-c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return np.sum(res.x), res.x
    else:
        # Fallback to zero if LP fails (should not happen with valid centers)
        return 0.0, np.zeros(n)

def objective_function(center_flat):
    """
    Objective function for the optimizer.
    Input: flat array of 52 coordinates
    Output: negative sum of radii (to be minimized)
    """
    centers = center_flat.reshape(-1, 2)
    sum_radii, _ = get_optimal_radii(centers)
    return -sum_radii

def run_packing():
    # 1. Initialization: Hexagonal Lattice
    # We need to place 26 points. A hexagonal lattice is a good start.
    # We approximate a radius of 0.1 to fit them initially.
    n = 26
    r_init = 0.1
    
    # Hexagonal grid spacing
    dx = 2 * r_init
    dy = np.sqrt(3) * r_init
    
    centers_list = []
    x = 0.1 # Start with margin
    y = 0.1
    
    # Generate points in a zig-zag pattern
    row = 0
    count = 0
    while count < n:
        # X positions for current row (offset every other row)
        offset = 0.5 * dx if row % 2 == 1 else 0
        x_pos = 0.1 + offset
        
        # Add points in this row until we hit the right boundary
        while x_pos + r_init <= 1.0 and count < n:
            centers_list.append([x_pos, y])
            count += 1
            x_pos += dx
        
        y += dy
        row += 1
    
    centers = np.array(centers_list[:n])
    
    # 2. Optimization
    # Use Nelder-Mead to optimize centers to maximize the sum of radii
    # Initial guess flattened
    x0 = centers.flatten()
    
    # Bounds for centers [0, 1]
    bounds = [(0, 1) for _ in range(2 * n)]
    
    # Run optimization
    # maxiter might need to be high for convergence
    res = minimize(objective_function, x0, method='Nelder-Mead', 
                   options={'maxiter': 5000, 'xatol': 1e-7, 'fatol': 1e-9})
    
    optimal_centers = res.x.reshape(-1, 2)
    optimal_sum, optimal_radii = get_optimal_radii(optimal_centers)
    
    return optimal_centers, optimal_radii, optimal_sum
