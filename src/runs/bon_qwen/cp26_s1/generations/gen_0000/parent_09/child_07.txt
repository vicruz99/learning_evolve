2. **Optimization Formulation**: Frame the problem as a constrained nonlinear optimization: minimize $-\sum r_i$ subject to boundary constraints ($0 \le x \pm r \le 1$, $0 \le y \pm r \le 1$) and pairwise non-overlap constraints ($(x_i-x_j)^2 + (y_i-y_j)^2 \ge (r_i+r_j)^2$).
3. **Solver Selection**: Use `scipy.optimize.minimize` with the `SLSQP` algorithm, which efficiently handles bound and nonlinear inequality constraints. 
4. **Efficiency & Robustness**: 
   - Vectorize constraint evaluation into two functions (boundary and non-overlap) returning arrays, reducing Python overhead.
   - Pass `N` via the `args` parameter to avoid closures and lambdas, adhering to the rules.
   - Set high iteration limits and tight tolerances for precision.
   - Post-process results to clamp centers/radii strictly within bounds, handling any 1e-12 numerical drift.

```python
import numpy as np
from scipy.optimize import minimize

# Global constant to avoid closures and match top-level helper requirement
N_CIRCLES = 26

def _objective(vars, N):
    """Negative sum of radii (to be minimized)"""
    return -sum(vars[3*i+2] for i in range(N))

def _boundary_constraints(vars, N):
    """Circles must stay within [0,1]x[0,1]"""
    vals = np.zeros(4 * N)
    idx = 0
    for i in range(N):
        vals[idx] = vars[3*i] - vars[3*i+2]          # x - r >= 0
        vals[idx+1] = 1.0 - vars[3*i] - vars[3*i+2]  # 1 - x - r >= 0
        vals[idx+2] = vars[3*i+1] - vars[3*i+2]      # y - r >= 0
        vals[idx+3] = 1.0 - vars[3*i+1] - vars[3*i+2]# 1 - y - r >= 0
        idx += 4
    return vals

def _non_overlap_constraints(vars, N):
    """Pairwise distance squared must be >= (r_i + r_j)^2"""
    num_constraints = N * (N - 1) // 2
    vals = np.zeros(num_constraints)
    idx = 0
    for i in range(N):
        for j in range(i + 1, N):
            dx = vars[3*i] - vars[3*j]
            dy = vars[3*i+1] - vars[3*j+1]
            r_sum = vars[3*i+2] + vars[3*j+2]
            vals[idx] = dx**2 + dy**2 - r_sum**2
            idx += 1
    return vals

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    N = N_CIRCLES
    
    # 1. Structured grid initialization for robust convergence
    x_grid = np.linspace(0.12, 0.88, 6)
    y_grid = np.linspace(0.12, 0.88, 5)
    centers = np.array([[x, y] for x in x_grid for y in y_grid])[:N]
    radii = np.full(N, 0.05)
    
    # 2. Flatten variables: [x0, y0, r0, x1, y1, r1, ...]
    x0 = np.zeros(3 * N)
    for i in range(N):
        x0[3*i] = centers[i, 0]
        x0[3*i+1] = centers[i, 1]
        x0[3*i+2] = radii[i]
        
    # Bounds: centers in [0,1], radii in [epsilon, 0.5]
    bounds = [(0.0, 1.0)] * (2*N) + [(1e-7, 0.5)] * N
    
    # 3. Define constraints with args to avoid closures/lambdas
    constraints = [
        {'type': 'ineq', 'fun': _boundary_constraints, 'args': (N,)},
        {'type': 'ineq', 'fun': _non_overlap_constraints, 'args': (N,)}
    ]
    
    # 4. Run SLSQP optimization
    try:
        res = minimize(
            _objective, x0, method='SLSQP', bounds=bounds, 
            constraints=constraints, args=(N,),
            options={'maxiter': 5000, 'ftol': 1e-12, 'disp': False}
        )
        opt_vars = res.x
    except Exception:
        opt_vars = x0
        
    # 5. Extract and enforce strict feasibility
    final_centers = np.array([[opt_vars[3*i], opt_vars[3*i+1]] for i in range(N)])
    final_radii = np.array([opt_vars[3*i+2] for i in range(N)])
    
    # Clamp to handle potential 1e-12 numerical drift from solver
    final_radii = np.maximum(final_radii, 1e-9)
    final_centers = np.clip(final_centers, final_radii[:, None], 1.0 - final_radii[:, None])
    
    return final_centers, final_radii, float(np.sum(final_radii))
```
