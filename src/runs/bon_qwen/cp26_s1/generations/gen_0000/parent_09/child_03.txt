import numpy as np
from scipy.optimize import minimize
import math


def objective(vars):
    """Objective function: negative sum of radii (to be minimized)."""
    return -np.sum(vars[2::3])


def constraint_func(vars, n):
    """
    Compute all constraint values.
    Returns array where each element >= 0 means constraint satisfied.
    """
    centers = np.column_stack((vars[0::3], vars[1::3]))
    radii = vars[2::3]

    num_constraints = 4 * n + n * (n - 1) // 2 + n
    constraints = np.zeros(num_constraints)

    idx = 0

    # Boundary constraints: x >= r, x <= 1-r, y >= r, y <= 1-r
    for i in range(n):
        constraints[idx] = centers[i, 0] - radii[i]
        idx += 1
        constraints[idx] = 1.0 - centers[i, 0] - radii[i]
        idx += 1
        constraints[idx] = centers[i, 1] - radii[i]
        idx += 1
        constraints[idx] = 1.0 - centers[i, 1] - radii[i]
        idx += 1

    # Pairwise non-overlap: dist(i,j) >= r_i + r_j
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = math.sqrt(dx * dx + dy * dy)
            constraints[idx] = dist - radii[i] - radii[j]
            idx += 1

    # Non-negativity of radii
    for i in range(n):
        constraints[idx] = radii[i]
        idx += 1

    return constraints


def constraint_jac(vars, n):
    """Jacobian of constraint function for faster optimization."""
    centers = np.column_stack((vars[0::3], vars[1::3]))
    radii = vars[2::3]

    num_constraints = 4 * n + n * (n - 1) // 2 + n
    num_vars = 3 * n
    jac = np.zeros((num_constraints, num_vars))

    idx = 0

    # Boundary constraints Jacobian
    for i in range(n):
        # x_i - r_i >= 0
        jac[idx, 3 * i] = 1.0
        jac[idx, 3 * i + 2] = -1.0
        idx += 1
        # 1 - x_i - r_i >= 0
        jac[idx, 3 * i] = -1.0
        jac[idx, 3 * i + 2] = -1.0
        idx += 1
        # y_i - r_i >= 0
        jac[idx, 3 * i + 1] = 1.0
        jac[idx, 3 * i + 2] = -1.0
        idx += 1
        # 1 - y_i - r_i >= 0
        jac[idx, 3 * i + 1] = -1.0
        jac[idx, 3 * i + 2] = -1.0
        idx += 1

    # Pairwise constraints Jacobian
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = math.sqrt(dx * dx + dy * dy)
            if dist > 1e-15:
                jac[idx, 3 * i] = dx / dist
                jac[idx, 3 * i + 1] = dy / dist
                jac[idx, 3 * i + 2] = -1.0
                jac[idx, 3 * j] = -dx / dist
                jac[idx, 3 * j + 1] = -dy / dist
                jac[idx, 3 * j + 2] = -1.0
            idx += 1

    # Non-negativity Jacobian
    for i in range(n):
        jac[idx, 3 * i + 2] = 1.0
        idx += 1

    return jac


def init_hexagonal(n, seed):
    """Initialize circles in hexagonal pattern with random jitter."""
    rng = np.random.RandomState(seed)
    positions = []

    spacing = 0.17
    row = 0
    while len(positions) < n:
        for col in range(15):
            if len(positions) >= n:
                break
            x = col * spacing * math.sqrt(3) + (0.5 if row % 2 == 1 else 0) * spacing * math.sqrt(3)
            y = row * spacing * math.sqrt(3)

            if -0.1 <= x <= 1.1 and -0.1 <= y <= 1.1:
                x_jitter = x + rng.uniform(-0.003, 0.003)
                y_jitter = y + rng.uniform(-0.003, 0.003)
                positions.append([x_jitter, y_jitter])
        row += 1

    centers = np.array(positions[:n])
    centers = np.clip(centers, 0.02, 0.98)
    radii = np.ones(n) * 0.008
    return centers, radii


def init_grid(n, seed):
    """Initialize circles in grid pattern with random jitter."""
    rng = np.random.RandomState(seed)
    positions = []

    # Approximate grid layout
    rows = 6
    cols = 5
    dx = 1.0 / (cols + 1)
    dy = 1.0 / (rows + 1)

    count = 0
    for r in range(rows):
        for c in range(cols):
            if count >= n:
                break
            x = (c + 1) * dx + rng.uniform(-0.01, 0.01)
            y = (r + 1) * dy + rng.uniform(-0.01, 0.01)
            x = max(0.02, min(0.98, x))
            y = max(0.02, min(0.98, y))
            positions.append([x, y])
            count += 1
        if count >= n:
            break

    centers = np.array(positions[:n])
    radii = np.ones(n) * 0.008
    return centers, radii


def grow_and_repel(centers, radii, n, num_iters=2000):
    """
    Physics-based optimization: grow circles and repel on overlap.
    Returns updated centers and radii.
    """
    centers = centers.copy()
    radii = radii.copy()

    growth_rate = 0.0005
    damping = 0.95

    for iteration in range(num_iters):
        # Grow all circles
        for i in range(n):
            radii[i] += growth_rate

        # Handle overlaps with repulsion
        for _ in range(3):  # Multiple passes per iteration
            for i in range(n):
                for j in range(i + 1, n):
                    dx = centers[i, 0] - centers[j, 0]
                    dy = centers[i, 1] - centers[j, 1]
                    dist_sq = dx * dx + dy * dy
                    dist = math.sqrt(dist_sq) if dist_sq > 1e-20 else 1e-10

                    min_dist = radii[i] + radii[j]
                    if dist < min_dist:
                        overlap = min_dist - dist
                        # Push apart proportional to radii
                        total_r = radii[i] + radii[j]
                        if total_r > 0 and dist > 1e-15:
                            force_i = overlap * radii[j] / total_r
                            force_j = overlap * radii[i] / total_r
                            dx_norm = dx / dist
                            dy_norm = dy / dist
                            centers[i, 0] += force_i * dx_norm * damping
                            centers[i, 1] += force_i * dy_norm * damping
                            centers[j, 0] -= force_j * dx_norm * damping
                            centers[j, 1] -= force_j * dy_norm * damping

        # Enforce boundary constraints
        for i in range(n):
            if centers[i, 0] < radii[i]:
                centers[i, 0] = radii[i]
            if centers[i, 0] > 1.0 - radii[i]:
                centers[i, 0] = 1.0 - radii[i]
            if centers[i, 1] < radii[i]:
                centers[i, 1] = radii[i]
            if centers[i, 1] > 1.0 - radii[i]:
                centers[i, 1] = 1.0 - radii[i]

        # Gradually reduce growth rate
        if iteration > 500:
            growth_rate *= 0.999

    # Final shrink to ensure validity
    for i in range(n):
        radii[i] = max(radii[i], 0)

    return centers, radii


def validate_quick(centers, radii, n):
    """Quick validation of packing validity."""
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if r < -1e-10:
            return False
        if x < r - 1e-10 or x > 1.0 - r + 1e-10:
            return False
        if y < r - 1e-10 or y > 1.0 - r + 1e-10:
            return False

    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = math.sqrt(dx * dx + dy * dy)
            if dist < radii[i] + radii[j] - 1e-10:
                return False

    return True


def run_packing():

            centers, radii = init_grid(n, seed)

        # Flatten to 1D array for optimization
        vars_init = np.zeros(3 * n)
        for i in range(n):
            vars_init[3 * i] = centers[i, 0]
            vars_init[3 * i + 1] = centers[i, 1]
            vars_init[3 * i + 2] = radii[i]

        # Bounds
        bounds = []
        for i in range(n):
            bounds.append((0.0, 1.0))   # x
            bounds.append((0.0, 1.0))   # y
            bounds.append((0.0, 0.5))   # r

        # Constraints
        cons = {
            'type': 'ineq',
            'fun': constraint_func,
            'jac': constraint_jac,
            'args': (n,)
        }

        # SLSQP optimization
        result = minimize(
            objective,
            vars_init,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,

        for i in range(n):
            vars_refined[3 * i] = centers_refined[i, 0]
