import numpy as np
from scipy.optimize import minimize

def objective(v):
    """Objective function: minimize negative sum of radii (equivalent to maximizing sum)"""
    # v is structured as [x1, y1, r1, x2, y2, r2, ...]
    radii = v[2::3]
    return -np.sum(radii)

def constraint_vec(v):
    """Vectorized constraint function for SLSQP"""
    n = len(v) // 3
    xs = v[0::3]
    ys = v[1::3]
    rs = v[2::3]
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    boundary_constraints = np.concatenate([
        xs - rs,
        1.0 - xs - rs,
        ys - rs,
        1.0 - ys - rs
    ])
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2 for all i < j
    X = xs[:, None]
    Y = ys[:, None]
    R = rs[:, None]
    
    dx = X - X.T
    dy = Y - Y.T
    dist_sq = dx**2 + dy**2
    r_sum_sq = (R + R.T)**2
    
    # Extract upper triangle constraints (i < j)
    idx = np.triu_indices(n, k=1)
    overlap_constraints = dist_sq[idx] - r_sum_sq[idx]
    
    return np.concatenate([boundary_constraints, overlap_constraints])

def run_packing():
