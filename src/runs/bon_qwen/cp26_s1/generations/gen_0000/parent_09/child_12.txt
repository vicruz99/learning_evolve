To pack 26 circles in a unit square and maximize the sum of radii, we aim for a high-density packing arrangement. While equal radii are a strong baseline, allowing variable radii provides flexibility to exploit gaps.

1.  **Initialization**:
    *   A hexagonal lattice provides the densest packing in 2D. We generate a hexagonal grid of points and select the 26 most central points to form a compact cluster.
    *   These points are scaled and shifted to fit within the unit square $[0,1]^2$.
    *   Initial radii are set to a small safe value to ensure a valid starting state for the optimizer.

2.  **Optimization**:
    *   We use `scipy.optimize.minimize` with the 'SLSQP' method to maximize the sum of radii (equivalent to minimizing the negative sum).
    *   **Constraints**:
        *   **Boundary**: $x_i \ge r_i$, $x_i \le 1 - r_i$, $y_i \ge r_i$, $y_i \le 1 - r_i$.
        *   **Non-overlap**: Distance between any two circles must be at least the sum of their radii. We use a squared distance constraint $(x_i - x_j)^2 + (y_i - y_j)^2 \ge (r_i + r_j)^2$ for numerical stability.
    *   **Bounds**: Radii must be non-negative; centers must be within the square (though boundary constraints are tighter).
    *   **Multiple Restarts**: Since the problem is non-convex, we run the optimizer from multiple randomized starting positions (perturbations of the hex grid) to avoid local minima and find a globally better solution.

3.  **Validation**:
    *   The final configuration is checked against the validation logic to ensure no overlaps and boundary violations.

import numpy as np
from scipy.optimize import minimize
import random

def generate_hex_grid(n_points, scale=1.0, seed=0):
    """
    Generate a compact hexagonal grid of points.
    Selects n_points closest to the center of a generated lattice.
    """
    rng = random.Random(seed)
    
    # Generate a large enough grid
    # We need a hexagonal pattern.
    # x = j * spacing + (i % 2) * spacing / 2
    # y = i * spacing * sqrt(3) / 2
    
    spacing = 1.0
    # Estimate grid size needed for 26 points
    # Area ~ n * sqrt(3)/2 * spacing^2. 
    # Just generate a generous box
    range_limit = 10 
    points = []
    
    for i in range(-range_limit, range_limit + 1):
        for j in range(-range_limit, range_limit + 1):
            # Hex offset
            x = j * spacing + (i % 2) * spacing / 2.0
            y = i * spacing * np.sqrt(3) / 2.0
            points.append([x, y])
            
    points = np.array(points)
    
    # Sort by distance from origin
    dists = np.linalg.norm(points, axis=1)
    sorted_indices = np.argsort(dists)
    
    # Pick closest n_points
    selected_points = points[sorted_indices[:n_points]]
    
    # Center and scale to fit roughly in [0,1]
    # Find bounding box
    min_coords = selected_points.min(axis=0)
    max_coords = selected_points.max(axis=0)
    
    # We want to fit into [0,1]. 
    # Add some padding to ensure they are strictly inside for optimizer start
    # But actually, just centering is fine.
    # Let's scale such that the bounding box fits in [0.1, 0.9] initially?
    # Or just fit to [0,1] and let optimizer handle it.
    
    # Center the cluster at (0.5, 0.5)
    center = selected_points.mean(axis=0)
    centered_points = selected_points - center
    
    # Scale to fit in [0,1] with some margin
    max_dim = np.max(np.abs(centered_points))
    if max_dim > 0:
        # Scale to fit in range [-0.4, 0.4] so total width 0.8, centered at 0.5 -> [0.1, 0.9]
        scale_factor = 0.4 / max_dim
        centered_points *= scale_factor
    
    final_points = centered_points + 0.5
    
    # Add small random perturbation to break symmetry
    noise = rng.uniform(-0.001, 0.001, size=final_points.shape)
    final_points += noise
    
    # Clip to valid range [0,1]
    final_points = np.clip(final_points, 0.0, 1.0)
    
    return final_points

def run_packing() -> tuple:
    n = 26
    
    # Best solution found so far
    best_sum_radii = -1.0
    best_centers = None
    best_radii = None
    
    # Try multiple restarts
    n_restarts = 5
    
    for restart in range(n_restarts):
        # Generate initial centers
        centers_init = generate_hex_grid(n, seed=restart*100)
        
        # Initialize radii small
        # Estimate a safe radius based on min distance
        # But for optimizer, small is safer to start valid
        radii_init = np.full(n, 0.01)
        
        # Stack variables: [x0, y0, r0, x1, y1, r1, ...]
        # Order: centers (n*2), then radii (n) ?
        # Or interleaved? Interleaved might be easier for indexing.
        # Let's use: [x0, y0, r0, x1, y1, r1, ...]
        # Shape (3*n,)
        
        x0 = centers_init[:, 0]
        y0 = centers_init[:, 1]
        r0 = radii_init
        
        x0 = np.concatenate([x0, y0, r0]) # Wait, this is wrong.
        # x0 has shape (n,). y0 (n,). r0 (n,).
        # We want a vector of length 3n.
        
        vars_init = np.zeros(3 * n)
        for i in range(n):
            vars_init[3*i] = x0[i]
            vars_init[3*i+1] = y0[i]
            vars_init[3*i+2] = r0[i]
            
        # Bounds
        # x, y in [0, 1]
        # r >= 0
        bounds = []
        for i in range(n):
            bounds.append((0.0, 1.0)) # x
            bounds.append((0.0, 1.0)) # y
            bounds.append((0.0, 1.0)) # r (upper bound 1.0 is safe)
            
        # Objective: minimize -sum(radii)
        def objective(vars_vec):
            sum_r = 0.0
            for i in range(n):
                sum_r += vars_vec[3*i + 2]
            return -sum_r
            
        # Constraints
        cons = []
        
        # 1. Boundary constraints
        # r <= x, r <= 1-x, r <= y, r <= 1-y
        # => x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
        for i in range(n):
            idx_x = 3*i
            idx_y = 3*i + 1
            idx_r = 3*i + 2
            
            # x - r >= 0
            cons.append({
                'type': 'ineq',
                'fun': lambda v, i=i: v[3*i] - v[3*i+2]
            })
            # 1 - x - r >= 0
            cons.append({
                'type': 'ineq',
                'fun': lambda v, i=i: 1.0 - v[3*i] - v[3*i+2]
            })
            # y - r >= 0
            cons.append({
                'type': 'ineq',
                'fun': lambda v, i=i: v[3*i+1] - v[3*i+2]
            })
            # 1 - y - r >= 0
            cons.append({
                'type': 'ineq',
                'fun': lambda v, i=i: 1.0 - v[3*i+1] - v[3*i+2]
            })
            
        # 2. Non-overlap constraints
        # dist_ij^2 >= (r_i + r_j)^2
        # (xi - xj)^2 + (yi - yj)^2 - (ri + rj)^2 >= 0
        # Note: This constraint is non-convex, but SLSQP handles local constraints.
        # To help convergence, maybe use squared distance directly.
        
        for i in range(n):
            for j in range(i + 1, n):
                idx_xi, idx_yi, idx_ri = 3*i, 3*i+1, 3*i+2
                idx_xj, idx_yj, idx_rj = 3*j, 3*j+1, 3*j+2
                
                def make_constraint(i, j):
                    def constraint_func(v):
                        dx = v[idx_xi] - v[idx_xj]
                        dy = v[idx_yi] - v[idx_yj]
                        r_sum = v[idx_ri] + v[idx_rj]
                        return (dx*dx + dy*dy) - (r_sum*r_sum)
                    return constraint_func

                cons.append({
                    'type': 'ineq',
                    'fun': make_constraint(i, j)
                })
        
        # Run optimization
        try:
            res = minimize(objective, vars_init, method='SLSQP', bounds=bounds, constraints=cons, 
                           options={'maxiter': 1000, 'ftol': 1e-9})
            
            if res.success:
                sum_r = -res.fun
                if sum_r > best_sum_radii:
                    best_sum_radii = sum_r
                    # Extract centers and radii
                    final_vars = res.x
                    final_centers = np.zeros((n, 2))
                    final_radii = np.zeros(n)
                    for k in range(n):
                        final_centers[k, 0] = final_vars[3*k]
                        final_centers[k, 1] = final_vars[3*k+1]
                        final_radii[k] = final_vars[3*k+2]
                    
                    best_centers = final_centers
                    best_radii = final_radii
                    
        except Exception as e:
            print(f"Optimization failed in restart {restart}: {e}")
            continue

    # Fallback if nothing worked (should not happen with valid init)
    if best_centers is None:
        # Return a simple grid
        step = 1.0 / 6.0
        best_centers = np.array([[i*step, j*step] for i in range(1,6) for j in range(1,6)] + [[0.5, 0.5]])
        best_radii = np.full(26, 0.05)
        best_sum_radii = np.sum(best_radii)

    return best_centers, best_radii, best_sum_radii
