import numpy as np
from scipy.optimize import minimize
from functools import partial


def obj_func(vars):
    """Objective: minimize negative sum of radii"""
    return -np.sum(vars[2::3])


def constr_x_lo(vars, i):
    """x_i - r_i >= 0"""
    return vars[3 * i] - vars[3 * i + 2]


def constr_x_hi(vars, i):
    """1 - x_i - r_i >= 0"""
    return 1 - vars[3 * i] - vars[3 * i + 2]


def constr_y_lo(vars, i):
    """y_i - r_i >= 0"""
    return vars[3 * i + 1] - vars[3 * i + 2]


def constr_y_hi(vars, i):
    """1 - y_i - r_i >= 0"""
    return 1 - vars[3 * i + 1] - vars[3 * i + 2]


def constr_radius_pos(vars, i):
    """r_i >= 0.001 (small positive lower bound)"""
    return vars[3 * i + 2] - 1e-6


def constr_overlap_sq(vars, i, j):
    """||c_i - c_j||^2 - (r_i + r_j)^2 >= 0
    Using squared form avoids sqrt singularity at coincident centers."""
    dx = vars[3 * i] - vars[3 * j]
    dy = vars[3 * i + 1] - vars[3 * j + 1]
    dist_sq = dx * dx + dy * dy
    r_sum = vars[3 * i + 2] + vars[3 * j + 2]
    return dist_sq - r_sum * r_sum


def make_hex_packing(n, rows_config, r_guess):
    """Create hexagonal-like packing with given row configuration"""
    num_rows = len(rows_config)
    centers_list = []
    radii_list = []

    for row_i, num_c in enumerate(rows_config):
        # Vertical spacing for hexagonal: sqrt(3)/2 * diameter
        # We scale to fit in unit square
        y = r_guess + row_i * (1 - 2 * r_guess) / max(num_rows - 1, 1)

        # Horizontal offset for staggered rows
        if row_i % 2 == 1:
            x_offset = r_guess * 0.5
        else:
            x_offset = 0

        if num_c > 1:
            x_spacing = (1 - 2 * r_guess - x_offset) / (num_c - 1)
        else:
            x_spacing = 0
            x_offset = 0.5 - r_guess

        for col in range(num_c):
            x = r_guess + x_offset + col * x_spacing
            centers_list.append([x, y])
            radii_list.append(r_guess)

    return np.array(centers_list), np.array(radii_list)


def validate_no_overlap(centers, radii):
    """Quick check if packing is valid"""
    n = len(centers)
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-10 or x + r > 1 + 1e-10 or y - r < -1e-10 or y + r > 1 + 1e-10:
            return False
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist_sq = dx * dx + dy * dy
            r_sum = radii[i] + radii[j]
            if dist_sq < r_sum * r_sum - 1e-10:
                return False
    return True


def build_constraints(n):
    """Build all constraints for the optimization"""
    cons = []
    for i in range(n):
        cons.append({'type': 'ineq', 'fun': partial(constr_x_lo, i=i)})
        cons.append({'type': 'ineq', 'fun': partial(constr_x_hi, i=i)})
        cons.append({'type': 'ineq', 'fun': partial(constr_y_lo, i=i)})
        cons.append({'type': 'ineq', 'fun': partial(constr_y_hi, i=i)})
        cons.append({'type': 'ineq', 'fun': partial(constr_radius_pos, i=i)})

    for i in range(n):
        for j in range(i + 1, n):
            cons.append({'type': 'ineq', 'fun': partial(constr_overlap_sq, i=i, j=j)})

    return cons


def optimize_from_initial(n, centers, radii, cons, max_iter=3000):
    """Run optimization from a given initial configuration"""
    x0 = np.zeros(3 * n)
    for i in range(n):
        x0[3 * i] = centers[i, 0]
        x0[3 * i + 1] = centers[i, 1]
        x0[3 * i + 2] = radii[i]

    result = minimize(
        obj_func,
        x0,
        method='SLSQP',
        constraints=cons,
        options={'maxiter': max_iter, 'ftol': 1e-15, 'disp': False},
    )

    opt_centers = np.zeros((n, 2))
    opt_radii = np.zeros(n)
    for i in range(n):
        opt_centers[i, 0] = result.x[3 * i]
        opt_centers[i, 1] = result.x[3 * i + 1]
        opt_radii[i] = max(1e-8, result.x[3 * i + 2])

    return opt_centers, opt_radii, -result.fun
