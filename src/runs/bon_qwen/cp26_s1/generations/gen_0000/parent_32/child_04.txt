import numpy as np
from scipy.optimize import minimize

def generate_hexagonal_init(n):
    """
    Generates an initial valid configuration of n circles using a hexagonal lattice.
    """
    # Target 26 circles. A 5-row hexagonal arrangement works well.
    # Row lengths: 6, 5, 6, 5, 4 sum to 26.
    # This fits the aspect ratio of the square reasonably well.
    row_lengths = [6, 5, 6, 5, 4]
    
    # Calculate approximate spacing to fit in unit square
    # We want a small initial radius so they don't overlap initially.
    # Let's target a scale where they fit loosely first.
    # A scaling factor for the lattice.
    
    centers = []
    r_init = 0.02 # Small initial radius
    
    # Lattice parameters
    # Vertical distance between rows
    h_row = 0.15 
    # Horizontal distance between centers in a row
    h_col = 0.12
    
    # Shift for alternating rows
    shift_x = h_col / 2.0
    
    y_pos = 0.1 # Start y
    
    current_n = 0
    for i, length in enumerate(row_lengths):
        x_start = 0.1
        if i % 2 == 1:
            x_start += shift_x
        
        for j in range(length):
            centers.append([x_start + j * h_col, y_pos])
            current_n += 1
            if current_n == n:
                break
        if current_n == n:
            break
        y_pos += h_row
        
    centers = np.array(centers[:n])
    radii = np.full(n, r_init)
    
    # Ensure validity by shrinking radii if overlaps exist in init (unlikely with small r)
    # Just to be safe, check nearest neighbor
    # (Skipped for brevity as r=0.02 is small enough for the grid spacing used)
    
    return centers, radii

def run_packing():
    n = 26
    
    # Objective function: maximize sum of radii => minimize negative sum
    def objective(v):
        # Radii are at indices 2, 5, 8, ...
        radii = v[2::3]
        return -np.sum(radii)

    # Constraint factory for boundaries
    # Returns function that computes constraint value >= 0
    def make_boundary_constraints():
        cons = []
        for i in range(n):
            idx_x = 3 * i
            idx_y = 3 * i + 1
            idx_r = 3 * i + 2
            
            # x - r >= 0
            cons.append({'type': 'ineq', 'fun': lambda v, i=i: v[3*i] - v[3*i+2]})
            # 1 - x - r >= 0  => x + r <= 1
            cons.append({'type': 'ineq', 'fun': lambda v, i=i: 1 - v[3*i] - v[3*i+2]})
            # y - r >= 0
            cons.append({'type': 'ineq', 'fun': lambda v, i=i: v[3*i+1] - v[3*i+2]})
            # 1 - y - r >= 0
            cons.append({'type': 'ineq', 'fun': lambda v, i=i: 1 - v[3*i+1] - v[3*i+2]})
        return cons

    # Constraint factory for non-overlap
    # dist^2 >= (r_i + r_j)^2
    def make_overlap_constraints():
        cons = []
        for i in range(n):
            for j in range(i + 1, n):
                idx_xi, idx_yi, idx_ri = 3*i, 3*i+1, 3*i+2
                idx_xj, idx_yj, idx_rj = 3*j, 3*j+1, 3*j+2
                
                # We define a function that captures i and j
                # Avoiding lambda closure issues by using default arguments or a factory
                def constraint_func(v, ii=i, jj=j):
                    xi, yi, ri = v[3*ii], v[3*ii+1], v[3*ii+2]
                    xj, yj, rj = v[3*jj], v[3*jj+1], v[3*jj+2]
                    dist_sq = (xi - xj)**2 + (yi - yj)**2
                    sum_r_sq = (ri + rj)**2
                    return dist_sq - sum_r_sq
                
                cons.append({'type': 'ineq', 'fun': constraint_func})
        return cons

    bounds = []
    for _ in range(n):
        # x, y in [0, 1], r in [0, 0.5]
        bounds.extend([(0, 1), (0, 1), (0, 0.5)])

    boundary_cons = make_boundary_constraints()
    overlap_cons = make_overlap_constraints()
    all_constraints = boundary_cons + overlap_cons

    # Generate initial guess
    centers_init, radii_init = generate_hexagonal_init(n)
    x0 = np.concatenate([centers_init.flatten(), radii_init])
    
    best_result = None
    best_val = -np.inf

    # Run optimization multiple times from slightly different starts
    # to avoid local optima
    num_restarts = 5
    
    for k in range(num_restarts):
        # Perturb initial guess slightly
        if k > 0:
            x0_perturbed = x0 + np.random.normal(0, 0.01, size=x0.shape)
            # Clip to keep roughly valid
            x0_perturbed[:2*n] = np.clip(x0_perturbed[:2*n], 0, 1)
            x0_perturbed[2*n:] = np.clip(x0_perturbed[2*n:], 0.01, 0.2)
        else:
            x0_perturbed = x0

        try:
            res = minimize(objective, x0_perturbed, method='SLSQP', 
                           bounds=bounds, constraints=all_constraints,
                           options={'maxiter': 500, 'ftol': 1e-9})
            
            if res.success and -res.fun > best_val:
                best_val = -res.fun
                best_result = res
        except Exception:
            continue

    if best_result is not None:
        centers_opt = best_result.x[:2*n].reshape(n, 2)
        radii_opt = best_result.x[2*n:]
    else:
        # Fallback to initial if optimization failed completely
        centers_opt = centers_init
        radii_opt = radii_init

    sum_radii = np.sum(radii_opt)
    
    return centers_opt, radii_opt, sum_radii
