import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    
    Returns:
        centers: np.array of shape (26, 2) with (x, y) coordinates
        radii: np.array of shape (26,) with radius of each circle
        sum_radii: float, the sum of all radii
    """
    n_circles = 26
    
    # --- 1. Initialization: Hexagonal Grid ---
    # Generate a hexagonal grid pattern for initial placement
    # We estimate a grid size to fit 26 circles
    # Approximate spacing for N=26 in 1x1 square: sqrt(1/26) ~ 0.2
    # Hexagonal packing is denser, so we can start with a slightly larger spacing or just fit them.
    
    points = []
    # 5 rows pattern: 6, 5, 6, 5, 4 = 26 circles
    row_counts = [6, 5, 6, 5, 4]
    spacing = 0.21 # Initial guess for spacing (diameter approx 0.42, radius 0.21)
    # Vertical spacing for hex is spacing * sqrt(3)/2
    
    y = 0.5 # Start y roughly in middle
    # Adjust y to center the grid
    total_height = (len(row_counts) - 1) * spacing * np.sqrt(3)/2
    start_y = (1 - total_height) / 2
    
    current_y = start_y
    
    for r_idx, count in enumerate(row_counts):
        # Horizontal offset for hexagonal packing
        # Even rows (0, 2, 4) aligned left? Let's shift odd rows
        offset = spacing / 2 if r_idx % 2 == 1 else 0
        
        # Center the row
        row_width = (count - 1) * spacing
        start_x = (1 - row_width) / 2 + offset
        
        # If the row with offset goes out of bounds, adjust or just clip later
        # For initialization, we just want points roughly distributed
        
        for c_idx in range(count):
            x = start_x + c_idx * spacing
            points.append([x, current_y])
        current_y += spacing * np.sqrt(3) / 2
        
    centers_init = np.array(points[:n_circles]) # Ensure we have 26
    
    # Initial radii: small enough to not overlap significantly but large enough to guide
    # Based on spacing ~ 0.21, radius ~ 0.1
    radii_init = np.full(n_circles, 0.09)
    
    # Combine variables: [x1, y1, x2, y2, ..., r1, r2, ...]
    # Actually easier to keep [x1, y1, r1, x2, y2, r2...] or separate?
    # Let's flatten to [x1, y1, ..., xn, yn, r1, r2, ..., rn]
    x0 = np.concatenate([centers_init.flatten(), radii_init])
    
    # --- 2. Optimization ---
    
    def objective(vars_flat):
        """Objective to minimize: Negative sum of radii"""
        radii = vars_flat[2*n_circles:]
        return -np.sum(radii)

    def constraint_overlap(vars_flat, i, j):
        """Constraint: dist(ci, cj) >= ri + rj"""
        ci = np.array([vars_flat[2*i], vars_flat[2*i+1]])
        cj = np.array([vars_flat[2*j], vars_flat[2*j+1]])
        ri = vars_flat[2*n_circles + i]
        rj = vars_flat[2*n_circles + j]
        dist = np.linalg.norm(ci - cj)
        return dist - (ri + rj)

    def constraint_boundary_x(vars_flat, i):
        """Constraint: ri <= xi and ri <= 1 - xi"""
        xi = vars_flat[2*i]
        ri = vars_flat[2*n_circles + i]
        return [xi - ri, (1 - xi) - ri]

    def constraint_boundary_y(vars_flat, i):
        """Constraint: ri <= yi and ri <= 1 - yi"""
        yi = vars_flat[2*i + 1]
        ri = vars_flat[2*n_circles + i]
        return [yi - ri, (1 - yi) - ri]

    # Define constraints for SLSQP
    # Type 'ineq' means function value >= 0
    
    constraints = []
    
    # Overlap constraints
    for i in range(n_circles):
        for j in range(i + 1, n_circles):
            constraints.append({
                'type': 'ineq',
                'fun': lambda v, i=i, j=j: constraint_overlap(v, i, j)
            })
            
    # Boundary constraints
    for i in range(n_circles):
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: constraint_boundary_x(v, i)[0], # xi - ri >= 0
        })
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: constraint_boundary_x(v, i)[1], # 1-xi - ri >= 0
        })
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: constraint_boundary_y(v, i)[0], # yi - ri >= 0
        })
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: constraint_boundary_y(v, i)[1], # 1-yi - ri >= 0
        })

    # Bounds: coordinates in [0, 1], radii >= 0
    # x, y bounds
    bounds_coords = [(0, 1)] * (2 * n_circles)
    # radius bounds
    bounds_radii = [(0, 0.5)] * n_circles # Radius cannot be > 0.5 in unit square
    
    bounds = bounds_coords + bounds_radii

    # Run optimization
    # SLSQP is a good choice for constrained non-linear optimization
    res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints,
                   options={'maxiter': 1000, 'ftol': 1e-9, 'disp': False})
    
    # Extract results
    optimal_vars = res.x
    centers = optimal_vars[:2*n_circles].reshape(n_circles, 2)
    radii = optimal_vars[2*n_circles:]
    
    # Post-processing: Clean up numerical noise
    # Ensure radii are non-negative (solver should handle this, but just in case)
    radii = np.maximum(radii, 0)
    
    # Ensure centers are within bounds (solver should handle this)
    centers = np.clip(centers, 0, 1)
    
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
