import numpy as np

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square

    Args:
        centers: np.array of shape (n, 2) with (x, y) coordinates
        radii: np.array of shape (n) with radius of each circle

    Returns:
        True if valid, False otherwise
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True

def initialize_packing_hexagonal(n, rotation_angle=0):

    r_init = r_est * 0.9

    dx = 2 * r_init
    dy = np.sqrt(3) * r_init

    row = 0
    col = 0

    # Hexagonal packing: row y increases by dy.
    # Odd rows shifted by dx/2.

        y = r_init + row * dy
        if y + r_init > 1.0 + 1e-4:
            row += 1
            continue

            x_start = r_init
        else:
            x_start = r_init + dx / 2
            
        col = 0
        x = x_start

            centers.append([x, y])
            x += dx
            col += 1
        row += 1
        
    centers = np.array(centers[:n])
    radii = np.ones(n) * r_init
    
    # Apply rotation if specified

        sin_a = np.sin(angle_rad)

        rotated_y = centers_shifted[:, 0] * sin_a + centers_shifted[:, 1] * cos_a
        centers = np.column_stack([rotated_x, rotated_y]) + [c_x, c_y]
        
        # After rotation, centers might be outside. We just return them,

        # But radii might be too large for the new orientation.
        # Let's scale down radii slightly to ensure validity initially if needed.
