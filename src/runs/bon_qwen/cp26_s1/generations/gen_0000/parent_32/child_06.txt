

Looking at this circle packing problem, I need to maximize the sum of radii for 26 circles in a unit square.

<strategy>
My strategy involves several key components:

1. **Hexagonal initial configuration**: Start with circles arranged in a hexagonal grid pattern, which is known to be optimal for dense circle packings. I'll use 5 rows with alternating 5 and 6 circles per row.

2. **Iterative radius computation**: Given fixed center positions, compute the maximum feasible radius for each circle by considering both boundary constraints (distance to square edges) and non-overlap constraints (distance to other circles).

3. **Multi-phase optimization**:
   - Phase 1: Use gradient-based optimization (L-BFGS-B) to find optimal center positions
   - Phase 2: Use iterative local search (perturbation) to escape local optima
   - Phase 3: Use penalty-based optimization with SLSQP for fine-tuning

4. **Multiple restarts**: Try different initial configurations and penalty parameters, keeping the best result.

5. **Final radius verification**: After optimization, compute exact feasible radii to ensure validity.

The key insight is that for any fixed set of centers, the problem of finding maximum radii reduces to checking boundary distances and pairwise distances. The optimization then focuses on finding optimal center positions.

import numpy as np
from scipy.optimize import minimize


def compute_radii(positions, n, max_iter=300):
    """Given fixed positions, compute maximum feasible radii iteratively."""
    r = np.ones(n) * 0.5
    for iteration in range(max_iter):
        r_new = np.zeros(n)
        for i in range(n):
            r_i = min(positions[i, 0], 1.0 - positions[i, 0],
                      positions[i, 1], 1.0 - positions[i, 1])
            for j in range(n):
                if i != j:
                    dx = positions[i, 0] - positions[j, 0]
                    dy = positions[i, 1] - positions[j, 1]
                    dist = np.sqrt(dx * dx + dy * dy)
                    r_i = min(r_i, dist - r[j])
            r_new[i] = max(r_i, 0.0)
        if np.max(np.abs(r_new - r)) < 1e-12:
            break
        r = r_new
    return r


def objective_penalty(x, n, penalty):
    """Objective with penalty for constraint violations."""
    pos = x[:2 * n].reshape(n, 2)
    r = x[2 * n:]
    
    obj = -np.sum(r)
    
    for i in range(n):
        x_i, y_i = pos[i]
        r_i = r[i]
        violations = [
            r_i - (1.0 - x_i),
            r_i - x_i,
            r_i - (1.0 - y_i),
            r_i - y_i
        ]
        for v in violations:
            if v > 0:
                obj += penalty * v * v
    
    for i in range(n):
        for j in range(i + 1, n):
            dx = pos[i, 0] - pos[j, 0]
            dy = pos[i, 1] - pos[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            overlap = r[i] + r[j] - dist
            if overlap > 0:
                obj += penalty * overlap * overlap
    
    return obj


def objective_positions(x, n):
    """Objective to maximize sum of radii given positions."""
    pos = x.reshape(n, 2)
    r = compute_radii(pos, n)
    return -np.sum(r)


def make_initial_hexagonal(n, r_est):
    """Create initial hexagonal grid configuration."""
    positions = np.zeros((n, 2))
    circles_per_row = [5, 6, 5, 6, 4]
    
    idx = 0
    for row, count in enumerate(circles_per_row):
        y = row * np.sqrt(3) * r_est + r_est
        x_offset = r_est if row % 2 == 1 else 0.0
        for col in range(count):
            x = col * 2 * r_est + r_est + x_offset
            if idx < n:
                positions[idx, 0] = x
                positions[idx, 1] = y
                idx += 1
    
    positions = np.clip(positions, 0.001, 0.999)
    return positions


def local_search(positions, n, step=0.005, max_improvement_iters=50):
    """Improve positions via local perturbation search."""
    radii = compute_radii(positions, n)
    current_sum = np.sum(radii)
    improved = True
    
    for iteration in range(max_improvement_iters):
        if not improved:
            break
        improved = False
        for i in range(n):
            for dx, dy in [(step, 0), (-step, 0), (0, step), (0, -step),
                           (step * 0.707, step * 0.707), (-step * 0.707, step * 0.707),
                           (step * 0.707, -step * 0.707), (-step * 0.707, -step * 0.707)]:
                new_positions = positions.copy()
                new_positions[i, 0] = np.clip(positions[i, 0] + dx, 0.001, 0.999)
                new_positions[i, 1] = np.clip(positions[i, 1] + dy, 0.001, 0.999)
                new_radii = compute_radii(new_positions, n)
                new_sum = np.sum(new_radii)
                if new_sum > current_sum + 1e-8:
                    positions = new_positions
                    radii = new_radii
                    current_sum = new_sum
                    improved = True
    
    return positions, radii, current_sum


def run_packing():
    n = 26
    
    best_positions = None
    best_radii = None
    best_sum = -1.0
    
    for r_est in [0.08, 0.085, 0.09, 0.095, 0.10]:
        positions = make_initial_hexagonal(n, r_est)
        radii = compute_radii(positions, n)
        current_sum = np.sum(radii)
        
        if current_sum > best_sum:
            best_sum = current_sum
            best_positions = positions.copy()
            best_radii = radii.copy()
        
        # Phase 1: Gradient-based optimization
        x0 = positions.flatten()
        bounds = [(0.001, 0.999)] * (2 * n)
        
        res = minimize(objective_positions, x0, method='L-BFGS-B',
                      bounds=bounds, options={'maxiter': 3000, 'ftol': 1e-12})
        positions_opt = res.x.reshape(n, 2)
        radii_opt = compute_radii(positions_opt, n)
        sum_opt = np.sum(radii_opt)
        
        if sum_opt > best_sum:
            best_sum = sum_opt
            best_positions = positions_opt.copy()
            best_radii = radii_opt.copy()
        
        # Phase 2: Local search
        positions_ls, radii_ls, sum_ls = local_search(positions_opt, n, step=0.005)
        if sum_ls > best_sum:
            best_sum = sum_ls
            best_positions = positions_ls.copy()
            best_radii = radii_ls.copy()
    
    # Phase 3: Fine-tuning with penalty optimization
    for penalty_factor in [1e3, 1e4, 1e5, 1e6, 1e7]:
        x0 = np.concatenate([best_positions.flatten(), best_radii])
        
        x_bounds = []
        for _ in range(n):
            x_bounds.extend([(0.001, 0.999), (0.001, 0.999)])
        for _ in range(n):
            x_bounds.append((0.001, 0.5))
        
        res = minimize(objective_penalty, x0, method='SLSQP',
                      args=(n, penalty_factor),
                      bounds=x_bounds,
                      options={'maxiter': 2000, 'ftol': 1e-12})
        
        positions_pen = res.x[:2 * n].reshape(n, 2)
        radii_pen = compute_radii(positions_pen, n)
        sum_pen = np.sum(radii_pen)
        
        if sum_pen > best_sum:
            best_sum = sum_pen
            best_positions = positions_pen.copy()
            best_radii = radii_pen.copy()
        
        # Additional local search after penalty optimization
        positions_ls, radii_ls, sum_ls = local_search(positions_pen, n, step=0.003)
        if sum_ls > best_sum:
            best_sum = sum_ls
            best_positions = positions_ls.copy()
            best_radii = radii_ls.copy()
    
    # Final refinement: multiple rounds of local search with decreasing step
    for step_size in [0.003, 0.001, 0.0005, 0.0002]:
        best_positions, best_radii, best_sum = local_search(
            best_positions, n, step=step_size, max_improvement_iters=100)
    
    # Final verification of radii
    best_radii = compute_radii(best_positions, n)
    final_sum = np.sum(best_radii)
    
    return best_positions, best_radii, float(final_sum)
