import numpy as np
from scipy.optimize import minimize

def smooth_plus(x):
    """Smooth approximation of max(0, x), returns 0 when x <= 0"""
    return np.where(x > 0, x, 0.0)

def compute_penalty(centers, radii, n):
    """Compute total penalty for constraint violations"""
    penalty = 0.0
    
    # Boundary violations
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        penalty += smooth_plus(r - x) ** 2
        penalty += smooth_plus(x + r - 1.0) ** 2
        penalty += smooth_plus(r - y) ** 2
        penalty += smooth_plus(y + r - 1.0) ** 2
    
    # Overlap violations
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            penalty += smooth_plus(radii[i] + radii[j] - dist) ** 2
    
    return penalty

def objective(params, n, penalty_weight):
    centers = params[:2 * n].reshape(n, 2)
    radii = params[2 * n:]
    score = np.sum(radii)
    penalty = compute_penalty(centers, radii, n)
    return -score + penalty_weight * penalty

def generate_hexagonal_initial(n):
    """Generate initial hexagonal arrangement"""
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.03)
    idx = 0
    
    row_counts = [5, 6, 5, 6, 4]
    for row_idx, count in enumerate(row_counts):
        y = (row_idx + 0.5) / 5.0
        for col_idx in range(count):
            if idx < n:
                x = (col_idx + 0.5) / count
                centers[idx] = np.array([x, y])
                idx += 1
    
    return centers, radii

def generate_grid_initial(n):

    centers = np.zeros((n, 2))
    radii = np.full(n, 0.03)
    idx = 0

        y = (row + 0.5) / 5.0
        count = 6 if row % 2 == 0 else 5

                idx += 1
    
    return centers, radii

def expand_and_relax(centers, radii, n, steps=2000):
    """Force-directed expansion: expand circles and push apart overlaps"""
    for step in range(steps):
        # Expand each circle to max possible size
        for i in range(n):
            max_r = min(centers[i, 0], 1.0 - centers[i, 0],
                        centers[i, 1], 1.0 - centers[i, 1])
            for j in range(n):
                if i != j:
                    dx = centers[i, 0] - centers[j, 0]
                    dy = centers[i, 1] - centers[j, 1]
                    dist = np.sqrt(dx * dx + dy * dy)
                    max_r = min(max_r, dist - radii[j])
            radii[i] = max(0.0, max_r)
        
        # Push apart overlapping circles
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                min_dist = radii[i] + radii[j]
                if dist < min_dist and dist > 1e-12:
                    direction = np.array([dx / dist, dy / dist])
                    overlap = (min_dist - dist) / 2.0
                    centers[i] += direction * overlap
                    centers[j] -= direction * overlap
    
    return centers, radii
