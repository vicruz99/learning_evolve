from scipy.optimize import minimize

N_CIRCLES = 26

def boundary_constraints(x):
    """Ensure circles are within [0,1]x[0,1]."""
    centers = np.column_stack((x[0::3], x[1::3]))
    radii = x[2::3]
    return np.concatenate([
        centers[:, 0] - radii,
        1 - centers[:, 0] - radii,
        centers[:, 1] - radii,
        1 - centers[:, 1] - radii,
    ])


def overlap_constraints(x):
    """Ensure no circle overlaps: distance >= sum of radii."""
    centers = np.column_stack((x[0::3], x[1::3]))

    result = np.empty(n * (n - 1) // 2)

            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)

            idx += 1

def resolve_collisions(centers, radii, n, max_iter=300):

def make_hexagonal(n, r0, seed=0):

    state = np.random.RandomState(seed)
    centers = []
    y = r0
    row = 0
    while len(centers) < n:

            x_start = r0
        else:
            x_start = 2 * r0

                break
            x = x_start + c * 2 * r0

        y += r0 * np.sqrt(3)
        row += 1

    centers += 0.003 * state.randn(n, 2)
    radii += 0.002 * state.randn(n)
    radii = np.maximum(radii, 0.02)
