import numpy as np
from scipy.optimize import linprog, minimize

N = 26

def solve_radii(centers):
    """Solves the LP to find maximum sum of radii for fixed centers."""
    n = centers.shape[0]
    c = -np.ones(n)
    
    # Preallocate constraint matrices
    # Constraints: r_i <= boundary_i  (n constraints)
    #              r_i + r_j <= dist_ij (n*(n-1)/2 constraints)
    n_constraints = n + n * (n - 1) // 2
    A_ub = np.zeros((n_constraints, n))
    b_ub = np.zeros(n_constraints)
    
    idx = 0
    for i in range(n):
        x, y = centers[i]
        b_i = min(x, 1.0 - x, y, 1.0 - y)
        A_ub[idx, i] = 1.0
        b_ub[idx] = b_i
        idx += 1
        
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = np.hypot(centers[i, 0] - centers[j, 0], 
                                 centers[i, 1] - centers[j, 1])
            idx += 1
            
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    except Exception:
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='interior-point')
        except Exception:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='simplex')
            
    if res.success:
        return -res.fun, res.x
    return 0.0, np.zeros(n)

def objective(centers_flat):
    """Objective function to minimize: negative sum of radii."""
    centers = np.clip(centers_flat.reshape(N, 2), 1e-5, 1.0 - 1e-5)
    s, _ = solve_radii(centers)
    return -s

def run_packing():
    best_val = -np.inf
    best_centers = None
    best_radii = None
    
    def optimize_from(init_centers):
        nonlocal best_val, best_centers, best_radii
        res = minimize(objective, init_centers.flatten(), method='Powell', 
                       options={'maxiter': 1500, 'ftol': 1e-9})
        if -res.fun > best_val:
            best_val = -res.fun
            best_centers = res.x.reshape(N, 2)
            _, best_radii = solve_radii(best_centers)

    # 1. Random restarts to explore configuration space
    for _ in range(8):
        init_centers = np.random.rand(N, 2)
        optimize_from(init_centers)
        
    # 2. Grid initialization (5x5 + 1) with perturbation
    gx = np.linspace(0.12, 0.88, 5)
    gy = np.linspace(0.12, 0.88, 5)
    pts = np.array([(x, y) for x in gx for y in gy])
    pts = np.vstack([pts, [0.5, 0.05]])
    init_centers = pts + np.random.normal(0, 0.02, pts.shape)
    optimize_from(init_centers)
    
    # 3. Hexagonal-like initialization for dense packing
    pts = []
    y = 0.15
    while len(pts) < N:
        x = 0.15 if len(pts) % 2 == 0 else 0.28
        while x < 0.85 and len(pts) < N:
            pts.append([x, y])
            x += 0.22
        y += 0.26
    pts = np.array(pts[:N]) + np.random.normal(0, 0.015, (N, 2))
    optimize_from(pts)
    
    # Final verification and strict clipping
    best_centers = np.clip(best_centers, 0.0, 1.0)
    _, final_radii = solve_radii(best_centers)
    final_sum = np.sum(final_radii)
    
    return best_centers, final_radii, final_sum
