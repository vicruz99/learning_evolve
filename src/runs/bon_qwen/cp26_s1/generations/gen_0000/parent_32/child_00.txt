import numpy as np
from scipy.optimize import minimize, linprog
from scipy.spatial.distance import pdist, squareform

def get_hex_grid(n):
    """Generates a hexagonal grid of n points in a unit square."""
    s = 0.14 # Approximate spacing for 26 points
    points = []
    y = 0
    row = 0
    while len(points) < n:
        x = 0
        if row % 2 == 1:
            x = s / 2
        while x <= 1.0:
            points.append([x, y])
            x += s
        y += s * np.sqrt(3) / 2
        row += 1
    return np.array(points[:n])

def repulsion_init(centers, iterations=50):
    """Simple force-directed repulsion to separate points."""
    pts = centers.copy().astype(float)
    n = pts.shape[0]
    for _ in range(iterations):
        forces = np.zeros_like(pts)
        for i in range(n):
            for j in range(i + 1, n):
                vec = pts[j] - pts[i]
                dist = np.linalg.norm(vec)
                if dist < 0.05 and dist > 1e-6:
                    repel = (0.05 - dist) / dist
                    forces[i] -= repel * vec
                    forces[j] += repel * vec
        pts += forces * 0.1
        pts = np.clip(pts, 0.01, 0.99)
    return pts

def solve_radii_lp(centers):
    """Solves the LP to find radii maximizing sum for fixed centers."""
    n = centers.shape[0]
    dists_to_bound = np.minimum(
        np.minimum(centers[:, 0], 1 - centers[:, 0]),
        np.minimum(centers[:, 1], 1 - centers[:, 1])
    )
    dist_matrix = squareform(pdist(centers))

    c_obj = np.ones(n) * -1
    num_pairwise = n * (n - 1) // 2
    num_constraints = n + num_pairwise
    
    A_ub = np.zeros((num_constraints, n))
    b_ub = np.zeros(num_constraints)
    
    # Boundary constraints: r_i <= dist
    for i in range(n):
        A_ub[i, i] = 1.0
        b_ub[i] = dists_to_bound[i]
        
    # Pairwise constraints: r_i + r_j <= dist
    idx = n
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dist_matrix[i, j]
            idx += 1
            
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return -res.fun, res.x
    except Exception:
        pass
    return 0.0, np.zeros(n)

def objective(centers_flat):
    centers = centers_flat.reshape(-1, 2)
    sum_r, _ = solve_radii_lp(centers)
    return -sum_r

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    # 1. Initialize
    init_centers = get_hex_grid(n)
    init_centers = repulsion_init(init_centers)
    
    # 2. Optimize
    bounds_opt = [(0.0, 1.0)] * (n * 2)
    res = minimize(objective, init_centers.flatten(), method='L-BFGS-B', bounds=bounds_opt, options={'maxiter': 2000})
    
    final_centers = res.x.reshape(-1, 2)
    sum_r, final_radii = solve_radii_lp(final_centers)
    
    return final_centers, final_radii, sum_r
