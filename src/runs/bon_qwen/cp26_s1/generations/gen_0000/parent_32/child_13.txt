from scipy.optimize import minimize

def initialize_hexagonal(n, seed, arr_idx):
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    r0 = 0.09
    dx = 2.0 * r0
    dy = np.sqrt(3.0) * r0

    # Different row arrangements for diversity
    if arr_idx == 0:
        row_counts = [5, 5, 5, 5, 4, 2]
    elif arr_idx == 1:
        row_counts = [6, 5, 6, 5, 4]
    else:
        row_counts = [5, 4, 5, 5, 5, 2]

    idx = 0
    for row_num, count in enumerate(row_counts):
        y = r0 + row_num * dy
        for col in range(count):
            x = r0 + col * dx
            if row_num % 2 == 1:
                x += dx / 2.0
            centers[idx] = [x, y]
            radii[idx] = r0
            idx += 1

    # Add randomness based on seed
    rng = np.random.RandomState(seed * 1000 + arr_idx * 100)
    centers += rng.uniform(-0.01, 0.01, size=centers.shape)
    centers = np.clip(centers, 0.02, 0.98)

    return centers, radii


def force_directed_pre_optimize(centers, radii, n, iterations=300):
    for t in range(iterations):
        forces = np.zeros_like(centers)

        for i in range(n):
            for j in range(i + 1, n):
                diff = centers[j] - centers[i]
                dist_sq = np.sum(diff ** 2)
                if dist_sq < 1e-20:
                    dist_sq = 1e-20
                dist = np.sqrt(dist_sq)
                dir_vec = diff / dist

                overlap = radii[i] + radii[j] - dist
                if overlap > 0:
                    force_mag = overlap * 10.0
                    forces[i] -= force_mag * dir_vec
                    forces[j] += force_mag * dir_vec

        # Boundary repulsion
        for i in range(n):
            for dim in range(2):
                if centers[i, dim] - radii[i] < 0:
                    forces[i, dim] += 5.0 * (radii[i] - centers[i, dim])
                if centers[i, dim] + radii[i] > 1.0:
                    forces[i, dim] -= 5.0 * (centers[i, dim] + radii[i] - 1.0)

        learning_rate = 0.005 / (1.0 + t * 0.005)
        centers += learning_rate * forces

        for i in range(n):
            centers[i, 0] = np.clip(centers[i, 0], radii[i] + 1e-6, 1.0 - radii[i] - 1e-6)
            centers[i, 1] = np.clip(centers[i, 1], radii[i] + 1e-6, 1.0 - radii[i] - 1e-6)

        # Gradually increase radii
        if t % 5 == 0:
            radii += 0.00008

    return centers, radii


def objective_func(x, n):
    return -np.sum(x[2::3])


def boundary_constraint_x_min(x, i):
    return x[3 * i] - x[3 * i + 2]


def boundary_constraint_x_max(x, i):
    return 1.0 - x[3 * i] - x[3 * i + 2]


def boundary_constraint_y_min(x, i):
    return x[3 * i + 1] - x[3 * i + 2]


def boundary_constraint_y_max(x, i):
    return 1.0 - x[3 * i + 1] - x[3 * i + 2]


def non_overlap_constraint(x, i, j):
    xi = x[3 * i]
    yi = x[3 * i + 1]
    ri = x[3 * i + 2]
    xj = x[3 * j]
    yj = x[3 * j + 1]
    rj = x[3 * j + 2]
    dist_sq = (xi - xj) ** 2 + (yi - yj) ** 2
    return dist_sq - (ri + rj) ** 2


def make_constraints(n):
    constraints = []

    for i in range(n):
        constraints.append({'type': 'ineq', 'fun': boundary_constraint_x_min, 'args': (i,)})
        constraints.append({'type': 'ineq', 'fun': boundary_constraint_x_max, 'args': (i,)})
        constraints.append({'type': 'ineq', 'fun': boundary_constraint_y_min, 'args': (i,)})
        constraints.append({'type': 'ineq', 'fun': boundary_constraint_y_max, 'args': (i,)})

    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({'type': 'ineq', 'fun': non_overlap_constraint, 'args': (i, j)})

    return constraints


def scipy_optimize(centers, radii, n):
    x0 = np.concatenate([centers.ravel(), radii])

    constraints = make_constraints(n)

    bounds = []
    for i in range(2 * n):
        bounds.append((0.0, 1.0))
    for i in range(n):
        bounds.append((0.0, 0.5))

    result = minimize(
        objective_func,
        x0,
        args=(n,),
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 5000, 'ftol': 1e-14, 'disp': False}
    )

    centers_opt = result.x[:2 * n].reshape((n, 2))
    radii_opt = result.x[2 * n:]
    radii_opt = np.maximum(radii_opt, 0.0)

    # Post-process: ensure validity by clipping if needed
    for i in range(n):
        r = radii_opt[i]
        centers_opt[i, 0] = np.clip(centers_opt[i, 0], r + 1e-10, 1.0 - r - 1e-10)
        centers_opt[i, 1] = np.clip(centers_opt[i, 1], r + 1e-10, 1.0 - r - 1e-10)

    sum_radii = np.sum(radii_opt)
    return centers_opt, radii_opt, sum_radii
