def calculate_cost(centers, radii):
    """
    Calculates the sum of radii minus penalty for overlaps and boundary violations.
    """
    n = centers.shape[0]
    sum_radii = np.sum(radii)
    penalty = 0.0
    eps = 1e-6

    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < 0: penalty += 100 * max(0, -(x - r))
        if x + r > 1: penalty += 100 * max(0, x + r - 1)
        if y - r < 0: penalty += 100 * max(0, -(y - r))
        if y + r > 1: penalty += 100 * max(0, y + r - 1)

    return -(sum_radii - penalty)

def hexagonal_lattice(n, r):
    """
    Generates a hexagonal lattice of n circles with radius r.
    """
    rows = []
    row_idx = 0
    remaining = n
    # Determine row sizes for a dense packing
    # For n=26, a 6-5-6-5-4 split is efficient
    row_sizes = [6, 5, 6, 5, 4]
    while remaining > 0 and row_idx < len(row_sizes):
        count = min(row_sizes[row_idx], remaining)
        rows.append(count)
        remaining -= count
        row_idx += 1
    
    centers = []
    y = r
    row_num = 0

            x = x_start + i * 2 * r
            centers.append([x, y])

        row_num += 1
        
    return np.array(centers)

def optimize_packing(initial_centers, initial_radii):

    Refines the positions and radii using L-BFGS-B.

    centers[:, 0] = np.clip(centers[:, 0], 0, 1)
