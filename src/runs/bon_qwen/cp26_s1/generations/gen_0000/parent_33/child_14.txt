

<strategy>
1. **Problem Formulation**: Frame the packing problem as a nonlinear optimization task. We want to maximize the sum of radii $\sum r_i$ subject to boundary constraints ($r_i \leq x_i, y_i \leq 1-r_i$) and non-overlap constraints ($(x_i-x_j)^2 + (y_i-y_j)^2 \geq (r_i+r_j)^2$).
2. **Initialization**: Generate an initial configuration using a hexagonal lattice pattern, which is known to be the densest packing in 2D. Scale and center these points within the unit square, assigning equal initial radii. Add small random perturbations to break symmetry and help the optimizer explore the space.
3. **Optimization Method**: Use `scipy.optimize.minimize` with the SLSQP algorithm. SLSQP handles bound and nonlinear inequality constraints effectively. The squared distance formulation avoids non-smooth square roots, improving convergence.
4. **Multiple Restarts**: Run the optimization from several slightly different initial configurations to escape local optima and find the global (or near-global) maximum sum of radii.
5. **Validation & Selection**: Keep track of the best feasible solution found across all attempts. The target sum of 2.636 corresponds to an average radius of ~0.1014, which is within reach of a dense hexagonal arrangement of 26 circles.

import numpy as np
from scipy.optimize import minimize

def run_packing():
    n = 26
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    np.random.seed(42)
    
    for attempt in range(6):
        # 1. Initialize with a hexagonal lattice
        pts = []
        dy = 0.14
        dx = dy * np.sqrt(3)
        y = 0.1
        row = 0
        # Generate more points than needed to ensure coverage
        while len(pts) < n + 5:
            for i in range(6):
                x = 0.1 + i * dx + (dx / 2 if row % 2 else 0)
                pts.append([x, y])
            y += dy
            row += 1
        pts = np.array(pts[:n])
        
        # 2. Center and scale to fit roughly in [0.1, 0.9]
        pts -= pts.mean(axis=0)
        scale = 0.35 / np.max(np.abs(pts))
        pts *= scale
        pts += 0.5
        
        # Initial radii
        r_init = np.full(n, 0.08)
        
        # Flatten to optimization vector [x0, y0, r0, x1, y1, r1, ...]
        x0 = np.zeros(3 * n)
        x0[0::3] = pts[:, 0]
        x0[1::3] = pts[:, 1]
        x0[2::3] = r_init
        
        # 3. Add random perturbation to break symmetry
        x0[0::3] += np.random.uniform(-0.02, 0.02, n)
        x0[1::3] += np.random.uniform(-0.02, 0.02, n)
        x0[0::3] = np.clip(x0[0::3], 0.05, 0.95)
        x0[1::3] = np.clip(x0[1::3], 0.05, 0.95)
        
        bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)] * n + [(0.0, 0.5)] * n
        
        # 4. Optimize
        try:
            res = minimize(_objective, x0, bounds=bounds, constraints={'type': 'ineq', 'fun': _constraints},
                           method='SLSQP', options={'ftol': 1e-12, 'maxiter': 5000, 'disp': False})
            
            # Check if constraints are satisfied within tolerance
            cons_vals = _constraints(res.x)
            if np.min(cons_vals) >= -1e-7:
                centers = res.x[0::3].reshape(n, 2)
                radii = res.x[2::3]
                s = np.sum(radii)
                if s > best_sum:
                    best_sum = s
                    best_centers = centers
                    best_radii = radii
        except Exception:
            continue
            
    # Fallback if optimization fails completely (highly unlikely)
    if best_centers is None:
        centers = np.array([(0.1 + i*0.18, 0.1 + j*0.18) for j in range(5) for i in range(5)])[:n]
        best_centers = centers
        best_radii = np.full(n, 0.08)
        best_sum = np.sum(best_radii)
        
    return best_centers, best_radii, best_sum

def _objective(x):
    # Maximize sum of radii => minimize negative sum
    return -np.sum(x[2::3])

def _constraints(x):
    n = len(x) // 3
    c = []
    # Boundary constraints: r <= x,y <= 1-r
    for i in range(n):
        cx, cy, r = x[3*i], x[3*i+1], x[3*i+2]
        c.extend([cx - r, 1.0 - cx - r, cy - r, 1.0 - cy - r])
        
    # Non-overlap constraints: dist^2 >= (r1 + r2)^2
    for i in range(n):
        for j in range(i + 1, n):
            dx = x[3*i] - x[3*j]
            dy = x[3*i+1] - x[3*j+1]
            r_sum = x[3*i+2] + x[3*j+2]
            c.append(dx*dx + dy*dy - r_sum*r_sum)
            
    return np.array(c)
