import numpy as np
from scipy.optimize import minimize
import itertools

def run_packing():

    n = 26

    # Function to generate initial hexagonal packing
    def get_initial_guess():
        centers = np.zeros((n, 2))
        radii = np.zeros(n)
        
        # Configuration for 26 circles: 5, 4, 5, 4, 5, 3 (Total 26)
        # This arrangement balances width and height constraints well.
        rows_config = [5, 4, 5, 4, 5, 3]
        r_guess = 0.09 # Initial small radius to ensure feasibility
        
        r = 0
        y_idx = 0
        
        for row_idx, count in enumerate(rows_config):
            # Hexagonal vertical spacing
            y_coord = r_guess + row_idx * (r_guess * np.sqrt(3))
            
            # Horizontal shift for alternating rows
            if row_idx % 2 == 0:
                x_start = r_guess
            else:
                x_start = 2 * r_guess # Shifted to fit in gaps
            
            for col_idx in range(count):
                x_coord = x_start + col_idx * (2 * r_guess)
                centers[r] = [x_coord, y_coord]
                radii[r] = r_guess
                r += 1
                
            y_idx += 1
            
        return centers, radii

    # Objective function: Maximize sum of radii -> Minimize negative sum
    def objective(x):
        # x contains [x1, y1, r1, x2, y2, r2, ...]
        radii = x[2::3]
        return -np.sum(radii)

    # Constraint function for boundaries
    def boundary_constraints(x):
        constraints = []
        centers = np.array([[x[i], x[i+1]] for i in range(0, len(x), 3)])
        radii = np.array([x[i+2] for i in range(0, len(x), 3)])
        
        for i in range(n):
            # x - r >= 0
            constraints.append(centers[i, 0] - radii[i])
            # 1 - x - r >= 0
            constraints.append(1.0 - centers[i, 0] - radii[i])
            # y - r >= 0
            constraints.append(centers[i, 1] - radii[i])
            # 1 - y - r >= 0
            constraints.append(1.0 - centers[i, 1] - radii[i])
            
        return np.array(constraints)

    def overlap_constraints(x):
        constraints = []
        centers = np.array([[x[i], x[i+1]] for i in range(0, len(x), 3)])

                dist_sq = np.sum((centers[i] - centers[j]) ** 2)
                sum_radii = radii[i] + radii[j]
                # dist^2 >= (r_i + r_j)^2  => dist^2 - (r_i + r_j)^2 >= 0
                constraints.append(dist_sq - sum_radii**2)
                
        return np.array(constraints)

    # Combine constraints for SLSQP
    def constraints_factory():
        # Boundary constraints
        bc = {'type': 'ineq', 'fun': boundary_constraints}
        # Overlap constraints
        oc = {'type': 'ineq', 'fun': overlap_constraints}
        return [bc, oc]

    # There are 3 variables per circle
    bounds = [(0, 1), (0, 1), (0, 0.5)] * n

            x0[3*i] = centers[i, 0]
            x0[3*i + 1] = centers[i, 1]

        result = minimize(

            x0, 
            method='SLSQP', 
            bounds=bounds,
