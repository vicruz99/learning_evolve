from scipy.optimize import minimize


def bound_constraint_x_low(p, i):
    return p[3 * i] - p[3 * i + 2]


def bound_constraint_x_high(p, i):
    return 1.0 - p[3 * i] - p[3 * i + 2]


def bound_constraint_y_low(p, i):
    return p[3 * i + 1] - p[3 * i + 2]


def bound_constraint_y_high(p, i):
    return 1.0 - p[3 * i + 1] - p[3 * i + 2]


def overlap_constraint(p, i, j):
    ci = np.array([p[3 * i], p[3 * i + 1]])
    cj = np.array([p[3 * j], p[3 * j + 1]])
    ri = p[3 * i + 2]
    rj = p[3 * j + 2]
    dist = np.sqrt(np.sum((ci - cj) ** 2))
    return dist - ri - rj


def build_constraints(n):
    constraints = []
    for i in range(n):
        constraints.append({'type': 'ineq', 'fun': lambda p, i=i: bound_constraint_x_low(p, i)})
        constraints.append({'type': 'ineq', 'fun': lambda p, i=i: bound_constraint_x_high(p, i)})
        constraints.append({'type': 'ineq', 'fun': lambda p, i=i: bound_constraint_y_low(p, i)})
        constraints.append({'type': 'ineq', 'fun': lambda p, i=i: bound_constraint_y_high(p, i)})
    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({'type': 'ineq', 'fun': lambda p, i=i, j=j: overlap_constraint(p, i, j)})
    return constraints


def make_hexagonal_init(r, row_counts):
    n = sum(row_counts)
    centers = np.zeros((n, 2))
    radii = np.full(n, r)
    idx = 0
    for row in range(len(row_counts)):
        y = r + row * r * np.sqrt(3)
        count = row_counts[row]
        offset = r if row % 2 == 1 else 0.0
        for col in range(count):
            x = r + offset + col * 2 * r
            centers[idx] = [x, y]
            idx += 1
    return centers, radii


def make_grid_init(r, rows, cols, extra_offset=None):
    n = rows * cols

    centers = np.zeros((n, 2))
    radii = np.full(n, r)

            centers[idx] = [r + j * 2 * r, r + i * 2 * r]
            idx += 1
    if extra_offset is not None:
        centers[idx] = extra_offset
    return centers, radii


def try_optimization(n, centers_init, radii_init, max_iter=8000):
    params_init = np.concatenate([centers_init.flatten(), radii_init])
    constraints = build_constraints(n)
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n

    result = minimize(
        lambda p: -np.sum(p[2 * n:]),
        params_init,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': max_iter, 'ftol': 1e-15, 'disp': False}
    )

    sum_radii = np.sum(result.x[2 * n:])
    centers_opt = result.x[:2 * n].reshape((n, 2))
    radii_opt = result.x[2 * n:]
    return centers_opt, radii_opt, sum_radii


def project_to_feasible(centers, radii, n):
    """Project solution to feasible region by clipping and shrinking overlapping circles."""
    radii = np.maximum(radii, 0)
    for iteration in range(50):
        changed = False
        # Boundary projection
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < 0:
                r = x
                changed = True
            if x + r > 1:
                r = 1 - x
                changed = True
            if y - r < 0:
                r = y
                changed = True
            if y + r > 1:
                r = 1 - y
                changed = True
            radii[i] = max(r, 0)

        # Overlap resolution
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[j][0] - centers[i][0]
                dy = centers[j][1] - centers[i][1]
                dist = np.sqrt(dx * dx + dy * dy)

                        shrink_j = overlap * radii[j] / r_total

                        changed = True
                    # Push apart

                    ny = dy / dist

                    centers[i][1] -= ny * overlap * 0.5
                    centers[j][0] += nx * overlap * 0.5
                    centers[j][1] += ny * overlap * 0.5
        if not changed:
            break
    radii = np.maximum(radii, 0)
    return centers, radii
