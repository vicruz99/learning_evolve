import numpy as np
from scipy.optimize import linprog, minimize


N = 26


def compute_optimal_radii(centers):
    """Given fixed center positions, compute optimal radii via linear programming.
    
    Maximize sum(r_i) subject to:
    - r_i + r_j <= distance(i,j) for all pairs (non-overlap)
    - r_i <= min(x_i, 1-x_i, y_i, 1-y_i) (boundary)
    - r_i >= 0
    """
    n = centers.shape[0]
    
    # Objective: minimize negative sum of radii
    c_obj = -np.ones(n)
    
    A_ub_list = []
    b_ub_list = []
    
    # Non-overlap constraints: r_i + r_j <= dist(i,j)
    for i in range(n):
        for j in range(i + 1, n):
            diff = centers[i] - centers[j]
            dist = np.sqrt(np.sum(diff * diff))
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub_list.append(row)
            b_ub_list.append(dist)
    
    # Boundary constraints: r_i <= boundary distance
    for i in range(n):
        x, y = centers[i]
        max_r = min(x, 1.0 - x, y, 1.0 - y)
        row = np.zeros(n)
        row[i] = 1.0
        A_ub_list.append(row)
        b_ub_list.append(max_r)
    
    A_ub = np.array(A_ub_list)
    b_ub = np.array(b_ub_list)
    
    bounds = [(0.0, None)] * n
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        return result.x
    # Fallback: small uniform radii
    return np.ones(n) * 0.01


def neg_sum_optimal_radii(centers_flat):
    """Objective function for position optimization."""
    centers = centers_flat.reshape((N, 2))
    radii = compute_optimal_radii(centers)
    return -np.sum(radii)


def init_hexagonal_variant(variant):
    """Generate hexagonal grid with different row arrangements."""
    centers = np.zeros((N, 2))
    
    # Different arrangements summing to 26
    arrangements = {
        0: [5, 6, 5, 6, 4],
        1: [6, 5, 6, 5, 4],
        2: [5, 5, 6, 5, 5],
        3: [4, 6, 6, 6, 4],
        4: [6, 6, 5, 5, 4],
        5: [5, 6, 6, 5, 4],
    }
    
    arr = arrangements.get(variant % len(arrangements), [5, 6, 5, 6, 4])
    
    n_rows = len(arr)
    y_step = 0.8 / (n_rows - 1) if n_rows > 1 else 0.5
    
    idx = 0
    for row in range(n_rows):
        count = arr[row]
        y = 0.1 + row * y_step
        # Alternate offset for hexagonal pattern
        x_start = 0.06 if row % 2 == 1 else 0.02
        x_step = 0.88 / (count - 1) if count > 1 else 0.5
        
        for col in range(count):
            x = x_start + col * x_step
            centers[idx] = [x, y]
            idx += 1
    
    return centers


def init_dense_grid():
    """Initialize with a denser grid-like pattern."""
    centers = np.zeros((N, 2))
    idx = 0
    
    # Try to pack circles in a 6x5 pattern with some missing
    positions = []
    for row in range(5):
        y = 0.1 + row * 0.2
        n_in_row = 6 if row % 2 == 0 else 5
        x_start = 0.05 if row % 2 == 0 else 0.12
        for col in range(n_in_row):
            x = x_start + col * 0.16
            positions.append([x, y])
    
    # Take first N positions
    for i in range(N):
        centers[i] = positions[i] if i < len(positions) else [0.5, 0.5]
    
    return centers


def init_corner_heavy():
    """Place circles with emphasis on corners."""
    centers = np.zeros((N, 2))
    
    # Place 4 circles in corners
    corners = [[0.1, 0.1], [0.9, 0.1], [0.1, 0.9], [0.9, 0.9]]
    for i, corner in enumerate(corners):
        centers[i] = corner
    
    # Fill rest in hexagonal pattern
    idx = 4
    for row in range(5):
        y = 0.2 + row * 0.15
        x_start = 0.15 if row % 2 == 1 else 0.1
        for col in range(5):
            if idx >= N:
                break
            x = x_start + col * 0.18
            centers[idx] = [x, y]
            idx += 1
    
    return centers


def simulated_annealing_positions(centers_init, n_iterations=30000):
    """Optimize positions using simulated annealing."""
    centers = centers_init.copy()
    best_centers = centers.copy()
    best_radii = compute_optimal_radii(centers)
    best_sum = np.sum(best_radii)
    
    np.random.seed(123)
    
    temperature = 0.05
    min_temp = 1e-6
    
    for it in range(n_iterations):
        temp = temperature * max(0.001, 1.0 - it / n_iterations)
        
        # Pick a circle to move
        i = np.random.randint(N)
        
        # Step size decreases with temperature
        step = 0.05 * max(0.01, temp)
        delta = np.random.normal(0, step, 2)
        
        new_centers = centers.copy()
        new_centers[i] = np.clip(centers[i] + delta, 0.01, 0.99)
        
        new_radii = compute_optimal_radii(new_centers)
        new_sum = np.sum(new_radii)
        
        # Acceptance criterion
        if new_sum > best_sum:
            centers = new_centers.copy()
            best_centers = new_centers.copy()
            best_radii = new_radii.copy()
            best_sum = new_sum
        elif np.random.random() < np.exp((new_sum - best_sum) / max(temp, 1e-10)):
            centers = new_centers.copy()
    
    return best_centers, best_radii, best_sum


def local_refine_positions(centers_init):
    """Use scipy.optimize to refine positions."""
    centers_flat = centers_init.flatten()
    
    bounds = [(0.01, 0.99)] * (2 * N)
    
    best_sum = 0
    best_centers = None
    
    # Try COBYLA
    try:
        result = minimize(neg_sum_optimal_radii, centers_flat, method='COBYLA',
                         bounds=bounds,
                         options={'maxiter': 8000, 'rhobeg': 0.005, 'catol': 1e-10})
        
        opt_centers = result.x.reshape((N, 2))
        opt_radii = compute_optimal_radii(opt_centers)
        opt_sum = np.sum(opt_radii)
        
        if opt_sum > best_sum:
            best_sum = opt_sum
            best_centers = opt_centers.copy()
    except Exception:
        pass
    
    # Try Nelder-Mead
    try:
        result2 = minimize(neg_sum_optimal_radii, centers_flat, method='Nelder-Mead',
                          options={'maxiter': 20000, 'xatol': 1e-8, 'fatol': 1e-10})
        
        opt_centers2 = result2.x.reshape((N, 2))
        opt_radii2 = compute_optimal_radii(opt_centers2)
        opt_sum2 = np.sum(opt_radii2)
        
        if opt_sum2 > best_sum:
            best_sum = opt_sum2
            best_centers = opt_centers2.copy()
    except Exception:
        pass
    
    if best_centers is None:
        best_centers = centers_init.copy()
    
    return best_centers, compute_optimal_radii(best_centers), best_sum


def greedy_expand(centers, radii):
    """Try to expand circles that have room."""
    centers = centers.copy()
    radii = radii.copy()
    changed = True
    
    while changed:
        changed = False
        for i in range(N):
            # Maximum possible radius for circle i
            max_r = min(centers[i, 0], 1.0 - centers[i, 0],
                       centers[i, 1], 1.0 - centers[i, 1])
            
            for j in range(N):
                if i != j:
                    diff = centers[i] - centers[j]
                    dist = np.sqrt(np.sum(diff * diff))
                    max_r = min(max_r, dist - radii[j])
            
            max_r = max(0, max_r)
            
            if max_r > radii[i] + 1e-10:
                radii[i] = max_r
                changed = True
    
    return centers, radii


def run_packing():
    """Main function to pack 26 circles in unit square."""
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Generate multiple initial configurations
    initial_configs = []
    for variant in range(6):
        initial_configs.append(init_hexagonal_variant(variant))
    initial_configs.append(init_dense_grid())
    initial_configs.append(init_corner_heavy())
    
    # Add some random configurations
    np.random.seed(42)
    for _ in range(5):
        random_centers = np.random.uniform(0.1, 0.9, (N, 2))
        initial_configs.append(random_centers)
    
    # For each initial configuration, try optimization
    for config_idx, centers_init in enumerate(initial_configs):
        # First compute baseline radii
        radii_baseline = compute_optimal_radii(centers_init)
        sum_baseline = np.sum(radii_baseline)
        
        if sum_baseline > best_sum:
            best_sum = sum_baseline
            best_centers = centers_init.copy()
            best_radii = radii_baseline.copy()
        
        # Try simulated annealing (short version for speed)
        sa_centers, sa_radii, sa_sum = simulated_annealing_positions(
            centers_init.copy(), n_iterations=15000
        )
        if sa_sum > best_sum:
            best_sum = sa_sum
            best_centers = sa_centers.copy()
            best_radii = sa_radii.copy()
        
        # Try scipy local refinement
        opt_centers, opt_radii, opt_sum = local_refine_positions(sa_centers)
        if opt_sum > best_sum:
            best_sum = opt_sum
            best_centers = opt_centers.copy()
            best_radii = opt_radii.copy()
    
    # Final greedy expansion pass
    best_centers, best_radii = greedy_expand(best_centers, best_radii)
    
    # Verify and return
    final_sum = np.sum(best_radii)
    
    # Safety: ensure radii are non-negative
    best_radii = np.maximum(best_radii, 0.0)
    
    return best_centers, best_radii, final_sum
