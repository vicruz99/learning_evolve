def validate_and_fix(centers, radii):
    """
    Ensures validity by shrinking radii if necessary.
    """
    n = len(radii)
    for i in range(n):

        r = min(centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])
        radii[i] = min(radii[i], r - 1e-9)
        
        # Overlap constraints
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))

                # Reduce both radii slightly to resolve overlap

    # Ensure non-negative
    radii = np.maximum(radii, 0)
    return centers, radii
