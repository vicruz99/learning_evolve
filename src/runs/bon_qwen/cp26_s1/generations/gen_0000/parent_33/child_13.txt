import numpy as np
from scipy.optimize import minimize
import itertools

def run_packing():
    np.random.seed(42)
    n = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None

    def objective(centers):
        # Solve LP to find max radii for fixed centers
        # Maximize sum(r_i)
        # Subject to:
        # r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
        # r_i + r_j <= sqrt((x_i-x_j)^2 + (y_i-y_j)^2)
        
        x = centers[:, 0]
        y = centers[:, 1]
        
        # Distance matrix
        dists = np.zeros((n, n))
        for i in range(n):
            for j in range(n):
                dists[i, j] = np.sqrt((x[i] - x[j])**2 + (y[i] - y[j])**2)
        
        # Constraints: A_ub @ r <= b_ub
        # r_i <= boundary
        A_ub = []
        b_ub = []
        
        for i in range(n):
            # r_i <= x_i
            row = np.zeros(n)
            row[i] = 1
            A_ub.append(row)
            b_ub.append(x[i])
            
            # r_i <= 1-x_i
            row = np.zeros(n)
            row[i] = 1
            A_ub.append(row)
            b_ub.append(1 - x[i])
            
            # r_i <= y_i
            row = np.zeros(n)
            row[i] = 1
            A_ub.append(row)
            b_ub.append(y[i])
            
            # r_i <= 1-y_i
            row = np.zeros(n)
            row[i] = 1
            A_ub.append(row)
            b_ub.append(1 - y[i])
            
            # r_i + r_j <= dist(i, j) for j > i
            for j in range(i + 1, n):
                row = np.zeros(n)
                row[i] = 1
                row[j] = 1
                A_ub.append(row)
                b_ub.append(dists[i, j])
        
        A_ub = np.array(A_ub)
        b_ub = np.array(b_ub)
        c = -np.ones(n) # Maximize sum -> Minimize -sum
        
        # Using scipy linear programming
        from scipy.optimize import linprog
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
        
        if res.success:
            return -res.fun, res.x
        else:
            return 0, np.zeros(n)

    def get_sum_radii(centers):
        s, r = objective(centers)
        return s

    # 1. Hexagonal Lattice Initialization
    centers = np.zeros((n, 2))
    r_guess = 0.102
    # Try to fit 5 rows: 6, 5, 6, 5, 4 or similar
    counts = [6, 5, 6, 5, 4]
    idx = 0
    for row_idx, count in enumerate(counts):
        y = r_guess + row_idx * r_guess * np.sqrt(3)
        # Shift every other row
        x_start = r_guess if row_idx % 2 == 0 else 2 * r_guess
        for k in range(count):
            if idx < n:
                centers[idx, 0] = x_start + k * 2 * r_guess
                centers[idx, 1] = y
                idx += 1
    
    # Normalize to fit in [0, 1]
    min_x, min_y = np.min(centers, axis=0)
    max_x, max_y = np.max(centers, axis=0)
    centers = (centers - min_x) / (max_x - min_x) * 0.9 + 0.05
    centers[:, 1] = (centers[:, 1] - min_y) / (max_y - min_y) * 0.9 + 0.05
    
    # Optimize
    res = minimize(lambda x: -get_sum_radii(x.reshape(-1, 2))[0], 
                   centers.flatten(), 
                   method='SLSQP', 
                   bounds=[(0, 1)] * (2 * n), 
                   options={'maxiter': 200, 'ftol': 1e-8})
    
    if res.fun < 0:
        curr_sum, curr_radii = -res.fun, objective(res.x.reshape(-1, 2))[1]
        if curr_sum > best_sum:
            best_sum = curr_sum
            best_centers = res.x.reshape(-1, 2)
            best_radii = curr_radii

    # 2. Random Initializations
    for _ in range(15):
        curr_centers = np.random.rand(n, 2) * 0.8 + 0.1 # Keep away from edges initially
        res = minimize(lambda x: -get_sum_radii(x.reshape(-1, 2))[0], 
                       curr_centers.flatten(), 
                       method='SLSQP', 
                       bounds=[(0, 1)] * (2 * n), 
                       options={'maxiter': 150, 'ftol': 1e-9})
        
        if res.fun < 0:
            curr_sum, curr_radii = -res.fun, objective(res.x.reshape(-1, 2))[1]
            if curr_sum > best_sum:
                best_sum = curr_sum
                best_centers = res.x.reshape(-1, 2)
                best_radii = curr_radii

    # 3. Perturbation of Best Result
    for _ in range(5):
        perturbed = best_centers + np.random.randn(n, 2) * 0.02
        perturbed = np.clip(perturbed, 0.001, 0.999)
        
        res = minimize(lambda x: -get_sum_radii(x.reshape(-1, 2))[0], 
                       perturbed.flatten(),

                best_centers = res.x.reshape(-1, 2)
                best_radii = curr_radii

    # Final cleanup to ensure validity

        best_radii = r
        
    return best_centers, best_radii, best_sum
