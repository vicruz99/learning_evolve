   - Non-overlap: $r_i + r_j \le \|c_i - c_j\|$ for all $i < j$
   - Boundary: $r_i \le \min(x_i, 1-x_i, y_i, 1-y_i)$
   - Non-negativity: $r_i \ge 0$
   This LP is solved efficiently using `scipy.optimize.linprog` with the HiGHS solver. The constraint matrix is precomputed since it only depends on circle indices.

2. **Outer Problem (Optimize Centers)**: We optimize the 52 center coordinates using `scipy.optimize.minimize` with the `L-BFGS-B` algorithm. import numpy as np
from scipy.optimize import linprog, minimize

N = 26
N_PAIRS = N * (N - 1) // 2
N_CONST = N_PAIRS + N

# Precompute LP structure (constant across calls)
A_LP = np.zeros((N_CONST, N))
c_LP = -np.ones(N)
bounds_LP = [(0.0, None)] * N

idx = 0
for i in range(N):
    for j in range(i + 1, N):
        A_LP[idx, i] = 1.0
        A_LP[idx, j] = 1.0
        idx += 1
for i in range(N):
    A_LP[idx, i] = 1.0
    idx += 1

def solve_radii(centers):
    """Solve the inner LP to find optimal radii for fixed centers."""
    b_ub = np.empty(N_CONST)
    # Pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    b_ub[:N_PAIRS] = np.triu(dists, k=1).flatten()
    # Boundary constraints
    b_ub[N_PAIRS:] = np.minimum(
        np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
        np.minimum(centers[:, 1], 1.0 - centers[:, 1])
    )
    
    res = linprog(c_LP, A_ub=A_LP, b_ub=b_ub, bounds=bounds_LP, method='highs')
    if res.success:
        return -res.fun, res.x
    return 0.0, np.zeros(N)

def objective(x):
    """Outer optimization objective: negative sum of optimal radii."""
    centers = x.reshape((N, 2))
    val, _ = solve_radii(centers)
    return -val

def run_packing():
    """Optimize circle positions and radii to maximize sum of radii."""
    best_sum = 0.0
    best_c = None
    best_r = None
    bounds_outer = [(1e-4, 1.0 - 1e-4)] * (2 * N)
    
    inits = []
    # 1. Hexagonal lattice initialization
    pts = []
    s = 0.18
    for r in range(6):
        y = 0.05 + r * s * np.sqrt(3)/2
        for c in range(5):
            x = 0.05 + c * s + (0.5 if r % 2 else 0) * s
            if 0.01 < x < 0.99 and 0.01 < y < 0.99:
                pts.append([x, y])
                if len(pts) >= 26: break
        if len(pts) >= 26: break
    while len(pts) < 26: pts.append(np.random.rand(2)*0.8+0.1)
    inits.append(np.array(pts).flatten())
    
    # 2. Random restarts
    for _ in range(10):
        inits.append((np.random.rand(N, 2)*0.8 + 0.1).flatten())
        
    # 3. Grid initialization
    g = []
    for i in range(5):
        for j in range(5):
            g.append([0.1+i*0.18, 0.1+j*0.18])
    g.append([0.5, 0.5])
    inits.append(np.array(g).flatten())
    
    for x0 in inits:
        try:
            res = minimize(objective, x0, method='L-BFGS-B', bounds=bounds_outer,
                           options={'maxiter': 3000, 'ftol': 1e-10})
            if -res.fun > best_sum:
                best_sum = -res.fun
                best_c = res.x.reshape((N, 2))
                _, best_r = solve_radii(best_c)
        except:
            continue
            
    return best_c, best_r, np.sum(best_r)
