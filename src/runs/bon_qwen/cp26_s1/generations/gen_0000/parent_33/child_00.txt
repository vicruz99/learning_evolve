import numpy as np
from scipy.optimize import minimize
import math


def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # Initial hexagonal grid placement with small radius for feasibility
    r_init = 0.05
    h = 2 * r_init
    v = r_init * math.sqrt(3)
    
    centers_list = []
    y = r_init
    row_pattern = [5, 4, 5, 4, 5, 3]  # Total: 26 circles
    for row in range(6):
        ncols = row_pattern[row]
        x_start = r_init if row % 2 == 0 else r_init + h / 2
        for col in range(ncols):
            x = x_start + col * h
            if len(centers_list) < n:
                centers_list.append([x, y])
        y += v
    
    while len(centers_list) < n:
        centers_list.append([0.5, 0.5])
    
    centers_init = np.array(centers_list[:n])
    
    # Flatten to optimization variable vector: [x1,y1,r1, x2,y2,r2, ..., x26,y26,r26]
    x0 = np.zeros(3 * n)
    for i in range(n):
        x0[3 * i] = centers_init[i, 0]
        x0[3 * i + 1] = centers_init[i, 1]
        x0[3 * i + 2] = r_init
    
    # Variable bounds
    bounds = [(0.0, 1.0)] * (3 * n)
    for i in range(n):
        bounds[3 * i + 2] = (0.0, 0.5)
    
    # Objective: maximize sum of radii (minimize negative)
    def obj(x):
        return -np.sum(x[2::3])
    
    # Constraints
    cons = []
    
    # Boundary constraints for each circle
    for i in range(n):
        cons.append({'type': 'ineq', 'fun': lambda x, i=i: x[3 * i] - x[3 * i + 2]})

        cons.append({'type': 'ineq', 'fun': lambda x, i=i: 1.0 - x[3 * i] - x[3 * i + 2]})
        cons.append({'type': 'ineq', 'fun': lambda x, i=i: 1.0 - x[3 * i + 1] - x[3 * i + 2]})
    
    # Non-overlap constraints for each pair

                'fun': lambda x, i=i, j=j: 
                    math.sqrt((x[3 * i] - x[3 * j]) ** 2 + (x[3 * i + 1] - x[3 * j + 1]) ** 2) 
                    - x[3 * i + 2] - x[3 * j + 2]
            })
    
    # Run optimization
    result = minimize(obj, x0, method='SLSQP', bounds=bounds, constraints=cons,
                     options={'maxiter': 3000, 'ftol': 1e-12})

    centers_opt = result.x[:2 * n].reshape(n, 2)

        centers_opt[i, 0] = np.clip(centers_opt[i, 0], r, 1.0 - r)
        centers_opt[i, 1] = np.clip(centers_opt[i, 1], r, 1.0 - r)
    
    sum_radii = np.sum(radii_opt)
    
    return centers_opt, radii_opt, float(sum_radii)
