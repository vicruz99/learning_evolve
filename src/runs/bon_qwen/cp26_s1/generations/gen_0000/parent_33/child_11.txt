import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    Uses a two-phase approach: 1) Equal radii optimization, 2) Unequal radii refinement.
    """
    N = 26

    def solve_packing(equal_radii: bool, initial_centers: np.ndarray, initial_radii: np.ndarray) -> tuple:
        """
        Optimizes centers and radii using SLSQP.
        """
        n_vars = 2 * N + (1 if equal_radii else N)
        
        def flatten(c, r):
            if equal_radii:
                return np.hstack([c.flatten(), [r[0]]])
            return np.hstack([c.flatten(), r])

        def unflatten(z):
            if equal_radii:
                r = np.full(N, z[-1])
                return z[:-1].reshape(N, 2), r
            r = z[2*N:]
            return z[:2*N].reshape(N, 2), r

        def objective(z):
            _, r = unflatten(z)
            return -np.sum(r)

        def constraints(z):
            c, r = unflatten(z)
            cons = []
            # Boundary constraints
            for i in range(N):
                x, y = c[i]
                ri = r[i]
                cons.append(x - ri)
                cons.append(1 - x - ri)
                cons.append(y - ri)
                cons.append(1 - y - ri)
            
            # Pairwise non-overlap
            for i in range(N):
                for j in range(i + 1, N):
                    dx = c[i, 0] - c[j, 0]
                    dy = c[i, 1] - c[j, 1]
                    dist_sq = dx*dx + dy*dy
                    sum_r = r[i] + r[j]
                    cons.append(dist_sq - sum_r*sum_r)
            return np.array(cons)

        # Initial bounds
        bounds = [(0.0, 1.0)] * (2 * N)
        if equal_radii:
            bounds.append((0.0, 0.5))
        else:
            bounds += [(0.0, 0.5)] * N

        x0 = flatten(initial_centers, initial_radii)
        
        result = minimize(objective, x0, method='SLSQP', bounds=bounds, 
                          constraints={'type': 'ineq', 'fun': constraints},
                          options={'maxiter': 1000, 'ftol': 1e-9})
        
        if result.success:
            final_c, final_r = unflatten(result.x)
            return final_c, final_r, -result.fun
        return initial_centers, initial_radii, np.sum(initial_radii)

    # --- Phase 1: Equal Radii with Hexagonal Initialization ---
    # Estimate target radius for hexagonal packing
    # Density ~ 0.9069. Area = 26 * pi * r^2. 1 * 0.9069 ~ 26 * pi * r^2 => r ~ 0.105
    # Use slightly smaller to ensure valid start
    r0 = 0.095
    
    # Hexagonal pattern generation
    centers = []
    r_y = np.sqrt(3) * r0
    x_offset = 0
    y_offset = r0
    
    for row in range(6): # 6 rows to fit 26 circles
        num_in_row = 5 if row % 2 == 0 else 5 # 5 circles per row
        if len(centers) + num_in_row > 26:
            num_in_row = 26 - len(centers)
            
        for k in range(num_in_row):
            x = r0 + k * (2 * r0) + x_offset
            y = y_offset + row * r_y
            centers.append([x, y])
        x_offset += r0
    
    centers = np.array(centers[:N])
    radii = np.full(N, r0)
    
    # Run Phase 1
    centers, radii, sum_r = solve_packing(equal_radii=True, initial_centers=centers, initial_radii=radii)
    
    # --- Phase 2: Unequal Radii Refinement ---
    # Use the result from Phase 1 to allow independent radii
    centers, radii, sum_r = solve_packing(equal_radii=False, initial_centers=centers, initial_radii=radii)
    
    # Post-validation and cleanup
    # Ensure radii are non-negative and centers are valid
    radii = np.maximum(radii, 0)
    # Clip centers slightly to ensure they are strictly inside if on edge
    # (SLSQP should handle bounds, but just in case)
    
    return centers, radii, float(sum_r)

# Helper to ensure top-level execution if needed, though run_packing is the entry point
if __name__ == "__main__":
    c, r, s = run_packing()
