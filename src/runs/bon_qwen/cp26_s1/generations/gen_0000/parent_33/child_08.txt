def get_constraints(centers, radii):
    """
    Returns a list of constraint dictionaries for scipy.optimize.minimize
    """
    n = len(radii)
    constraints = []
    
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    # Equivalent to: x - r >= 0, x + r <= 1, etc.
    for i in range(n):
        # x_i - r_i >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: v[2*i] - v[2*i + 2]
        })
        # 1 - x_i - r_i >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: 1 - v[2*i] - v[2*i + 2]
        })
        # y_i - r_i >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: v[2*i + 1] - v[2*i + 2]
        })
        # 1 - y_i - r_i >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: 1 - v[2*i + 1] - v[2*i + 2]
        })
        
    # Non-overlap constraints: dist >= r_i + r_j
    # (x_i - x_j)^2 + (y_i - y_j)^2 >= (r_i + r_j)^2
    # Constraint: dist_sq - (r_i + r_j)^2 >= 0
    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({
                'type': 'ineq',
                'fun': lambda v, i=i, j=j: (
                    (v[2*i] - v[2*j])**2 + (v[2*i + 1] - v[2*j + 1])**2 - 
                    (v[2*i + 2] + v[2*j + 2])**2
                )
            })
            
    return constraints

def objective_function(v, n):
    """
    Objective: Maximize sum of radii.
    v is flattened array [x0, y0, r0, x1, y1, r1, ...]

    # Let's define structure: v[3*i] = x, v[3*i+1] = y, v[3*i+2] = r
    radii = v[2::3]
    return -np.sum(radii)
