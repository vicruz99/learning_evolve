from scipy.optimize import minimize


def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

        centers = np.zeros((n, 2))
        radii = np.ones(n) * 0.09

        idx = 0
        row = 0
        col = 0
        spacing = 0.11
        while idx < n:
            x = 0.08 + col * spacing
            y = 0.08 + row * spacing * np.sqrt(3) / 2
            if row % 2 == 1:
                x += spacing / 2
            if 0.02 <= x <= 0.98 and 0.02 <= y <= 0.98 and idx < n:
                centers[idx] = [x, y]
                idx += 1
            col += 1
            if col > 9:
                col = 0
                row += 1

        return centers, radii

    def random_init():
        """Initialize circles with random perturbation from hexagonal."""
        centers, radii = hexagonal_init()
        centers += np.random.randn(n, 2) * 0.02
        radii += np.random.randn(n) * 0.01
        radii = np.maximum(radii, 0.02)
        return centers, radii

    def compute_objective(x, n, lam):
        """Compute objective: negative sum of radii plus penalty for violations."""
        centers = x[:2 * n].reshape(n, 2)
        radii = x[2 * n:]

        sum_r = np.sum(radii)

        violations = 0.0

        # Boundary violations
        for i in range(n):
            cx, cy = centers[i]
            r = radii[i]
            if r - cx > 0:
                violations += (r - cx) ** 2
            if cx + r - 1 > 0:
                violations += (cx + r - 1) ** 2
            if r - cy > 0:
                violations += (r - cy) ** 2
            if cy + r - 1 > 0:
                violations += (cy + r - 1) ** 2

        # Overlap violations
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                min_dist = radii[i] + radii[j]
                if min_dist - dist > 0:
                    violations += (min_dist - dist) ** 2

        return -sum_r + lam * violations

    def compute_objective_grad(x, n, lam):
        """Compute objective and gradient together."""
        centers = x[:2 * n].reshape(n, 2)
        radii = x[2 * n:]

        grad = np.zeros_like(x)
        grad[2 * n:] = -np.ones(n)  # d/d(r_i) of -sum(r_i) = -1

        violations = 0.0

        # Boundary gradients
        for i in range(n):
            cx, cy = centers[i]
            r = radii[i]

            # Left boundary: r - cx > 0
            if r - cx > 0:
                v = r - cx
                violations += v ** 2
                grad[2 * i] += lam * (-2 * v)       # d/d(cx)
                grad[2 * n + i] += lam * (2 * v)   # d/d(r_i)

            # Right boundary: cx + r - 1 > 0
            if cx + r - 1 > 0:
                v = cx + r - 1
                violations += v ** 2
                grad[2 * i] += lam * (2 * v)        # d/d(cx)
                grad[2 * n + i] += lam * (2 * v)    # d/d(r_i)

            if r - cy > 0:
                v = r - cy
                violations += v ** 2
                grad[2 * i + 1] += lam * (-2 * v)   # d/d(cy)
                grad[2 * n + i] += lam * (2 * v)    # d/d(r_i)

            # Top boundary: cy + r - 1 > 0
            if cy + r - 1 > 0:
                v = cy + r - 1
                violations += v ** 2
                grad[2 * i + 1] += lam * (2 * v)    # d/d(cy)
                grad[2 * n + i] += lam * (2 * v)    # d/d(r_i)

        # Overlap gradients
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                min_dist = radii[i] + radii[j]

                if min_dist - dist > 0 and dist > 1e-10:
                    v = min_dist - dist
                    violations += v ** 2

                    # Gradient w.r.t centers[i]
                    grad[2 * i] += lam * (-2 * v) * (-dx / dist)
                    grad[2 * i + 1] += lam * (-2 * v) * (-dy / dist)

                    # Gradient w.r.t centers[j]
                    grad[2 * j] += lam * (-2 * v) * (dx / dist)
                    grad[2 * j + 1] += lam * (-2 * v) * (dy / dist)

                    # Gradient w.r.t radii[i] and radii[j]
                    grad[2 * n + i] += lam * (2 * v)
                    grad[2 * n + j] += lam * (2 * v)

        obj = -np.sum(radii) + lam * violations
        return obj, grad

    best_sum = 0
    best_centers = None
    best_radii = None

        x0 = np.concatenate([centers.flatten(), radii])

                return compute_objective_grad(x, n, lam)

            result = minimize(
                obj_grad,
                x_current,
                bounds=bounds,
                method='L-BFGS-B',

                    else:
                        best_radii[j] = (dist - best_radii[i]) / 2
                    violated = True

            break

    final_sum = np.sum(best_radii)
    return best_centers, best_radii, final_sum
