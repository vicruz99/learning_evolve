import numpy as np
from scipy.optimize import minimize

def compute_energy(centers, r):
    """
    Calculates the energy of the system based on overlaps and boundary violations.
    Energy is defined as the sum of squared overlaps.
    """
    n = centers.shape[0]
    energy = 0.0
    
    # Check inter-circle overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            overlap = 2 * r - dist
            if overlap > 0:
                energy += overlap ** 2
                
    # Check boundary violations
    for i in range(n):
        x, y = centers[i]
        if x < r:
            energy += (r - x) ** 2
        if x > 1 - r:
            energy += (x - (1 - r)) ** 2
        if y < r:
            energy += (r - y) ** 2
        if y > 1 - r:
            energy += (y - (1 - r)) ** 2
            
    return energy

    n = 26

    r = x[n*2]
    
    # We add a penalty for energy to ensure constraints are respected
    return -r + 10.0 * compute_energy(centers, r)
