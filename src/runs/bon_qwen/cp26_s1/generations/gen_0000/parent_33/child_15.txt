def get_boundary_constraints(centers):
    """
    Calculates the maximum possible radius for each circle based on boundaries.
    r_i <= min(x_i, 1-x_i, y_i, 1-y_i)
    """
    x = centers[:, 0]
    y = centers[:, 1]
    bounds = np.minimum(np.minimum(x, 1 - x), np.minimum(y, 1 - y))
    return bounds

def solve_radii_lp(centers):
    """
    Solves the Linear Programming problem to maximize sum of radii given fixed centers.
    Maximize sum(r_i)
    Subject to:
      r_i + r_j <= ||c_i - c_j||  for all i < j
      r_i <= boundary_limit_i      for all i
      r_i >= 0
    """
    n = centers.shape[0]
    
    # Calculate pairwise distances
    dists = dist.pdist(centers, metric='euclidean')
    
    # Boundary constraints upper bounds
    b_ub = get_boundary_constraints(centers)
    
    # Objective: Minimize -sum(r_i)
    c_obj = -np.ones(n)
    
    # Inequality constraints A_ub * r <= b_ub
    # We need to construct the matrix for pairwise constraints
    # Number of pairwise constraints: n*(n-1)/2
    num_pairs = n * (n - 1) // 2
    
    # Using sparse matrix construction would be more efficient, but for n=26, dense is okay
    # However, creating a dense matrix of size (num_pairs + 4*n, n) might be memory heavy but manageable.
    # num_pairs approx 325. 4*n = 104. Total rows ~ 430. Cols 26.
    
    A_ub = []
    b_ub_vec = []
    
    # Pairwise constraints: r_i + r_j <= d_ij
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub_vec.append(dists[idx])
            idx += 1
            
    # Boundary constraints: r_i <= bound_i
    # This is equivalent to 1*r_i <= bound_i
    # We can just add rows to A_ub or handle via bounds in linprog
    # scipy.optimize.linprog supports bounds. Let's use bounds for r_i >= 0 and r_i <= b_ub.
    
    # Construct A_ub and b_ub for pairwise only
    A_ub = np.array(A_ub)
    b_ub_vec = np.array(b_ub_vec)
    
    # Bounds for variables: (0, boundary_limit)
    bounds = [(0, b) for b in b_ub]
    
    try:
        # Use high-s method or simplex? 'highs' is default in newer scipy and very fast.
        res = opt.linprog(c_obj, A_ub=A_ub, b_ub=b_ub_vec, bounds=bounds, method='highs')
        
        if res.success:
            radii = res.x
            return radii, -res.fun
        else:
            # Fallback if LP fails: just set radii to small values
            radii = np.full(n, 1e-6)
            return radii, 0.0
    except Exception:
        radii = np.full(n, 1e-6)
        return radii, 0.0

def objective_function(center_params, n):
    """
    Objective function for center optimization.
    Returns negative sum of radii (since we want to maximize sum).
    """
    centers = center_params.reshape((n, 2))
    
    # Ensure centers are within [0, 1]
    # If they go out, penalize or clamp. Clamping is safer for continuity.
    centers = np.clip(centers, 0, 1)
    
    # Solve for radii
    radii, sum_radii = solve_radii_lp(centers)
    
    # We want to maximize sum_radii, so return negative
    return -sum_radii
