import numpy as np
from scipy.optimize import minimize

def get_penalty(centers, radii):
    """
    Calculates a penalty for constraint violations.
    Returns 0 if the configuration is valid.
    """
    penalty = 0.0
    
    # Boundary constraints
    # r_i <= x_i <= 1-r_i  =>  x_i - r_i >= 0 AND 1 - x_i - r_i >= 0
    # Same for y
    boundary_violations = 0.0
    for i in range(len(radii)):
        r = radii[i]
        x, y = centers[i]
        
        # Left and Right walls
        if x - r < 0: boundary_violations += (r - x)**2
        if x + r > 1: boundary_violations += (x + r - 1)**2
        
        # Bottom and Top walls
        if y - r < 0: boundary_violations += (r - y)**2
        if y + r > 1: boundary_violations += (y + r - 1)**2
        
    penalty += boundary_violations
    
    # Overlap constraints
    # dist >= r_i + r_j
    overlap_violations = 0.0
    for i in range(len(radii)):
        for j in range(i + 1, len(radii)):
            dist = np.linalg.norm(centers[i] - centers[j])
            r_sum = radii[i] + radii[j]
            if dist < r_sum:
                overlap_violations += (r_sum - dist)**2
                
    penalty += overlap_violations
    
    return penalty

def objective_function(params, n):
    """
    Minimizes negative sum of radii + penalty.
    params: [x1, y1, r1, x2, y2, r2, ...]
    """
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    for i in range(n):
        centers[i] = params[3*i : 3*i+2]
        radii[i] = params[3*i+2]
        
    sum_radii = np.sum(radii)
    pen = get_penalty(centers, radii)
    
    return -sum_radii + 1000.0 * pen

def generate_hexagonal_grid(n, square_size=1.0, radius=0.1):
    """
    Generates a hexagonal lattice packing inside the unit square.
    """
    points = []
    # Hexagonal packing row spacing
    row_height = radius * np.sqrt(3)
    
    # Determine max rows
    # 2*radius for first and last row thickness
    max_y = square_size - radius

        # Standard hex: even rows aligned, odd rows shifted by radius
        is_shifted = (y > radius + 1e-6) # Simplistic check, better to use counter
        
        # Let's use a counter for rows
        row_idx = int((y - radius) / row_height)
        is_shifted = (row_idx % 2 == 1)
        
        start_x = radius + (radius if is_shifted else 0)
        # Width available
        end_x = square_size - radius
        
        x = start_x
        while x <= end_x + 1e-9:
            points.append([x, y])
            x += 2 * radius
        
        y += row_height
        if y > square_size + 1e-9:
            break
            
    return points

def create_valid_config(n, points, r_base):

    Selects n points from the grid and assigns radii.

    # If we have more points than needed, select a dense subset
    # If fewer, we might need to adjust, but for 26 and radius 0.1,

                pts.append([x, y])
        points = pts
        
    # Take first n points (they are ordered by rows)
    selected_points = points[:n]
    
    # Flatten to params: [x1, y1, r1, ...]

    params = []
    for p in selected_points:
        params.append(p[0])
        params.append(p[1])
        params.append(r_base)
        
    return np.array(params)
