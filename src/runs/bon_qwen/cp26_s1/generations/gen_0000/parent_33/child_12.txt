import numpy as np
from scipy.optimize import minimize
from functools import partial


def init_hexagonal_lattice(n, seed=None):
    """Initialize circles in a hexagonal lattice pattern."""
    if seed is not None:
        np.random.seed(seed)
    
    # Use a staggered hexagonal arrangement
    # 6 rows, varying columns per row to get 26 circles
    row_counts = [4, 5, 4, 5, 4, 4]  # sums to 26
    
    centers = []
    radii = []
    
    # Initial spacing estimate
    d = 0.18
    r_init = d / 2.0
    spacing_y = d * np.sqrt(3) / 2
    
    count = 0
    for row in range(6):
        num_in_row = row_counts[row]
        # Horizontal spacing for this row
        spacing_x = (1 - 2 * r_init) / (num_in_row - 1 + 1) if num_in_row > 1 else 1
        for col in range(num_in_row):
            x = (col + 0.5) * spacing_x
            y = (row + 0.5) * spacing_y / max(1, 5)
            # Add stagger for odd rows
            if row % 2 == 1:
                x += spacing_x / 2
            if count < n:
                centers.append([x, y])
                radii.append(r_init)
                count += 1
            if count >= n:
                break
        if count >= n:
            break
    
    return np.array(centers), np.array(radii[:n])


def init_uniform_grid(n, seed=None):
    """Initialize with a more uniform distribution."""
    if seed is not None:
        np.random.seed(seed)
    
    centers = []
    radii = []
    
    # Arrange in roughly hexagonal rows
    num_rows = 5
    base_spacing_x = 1.0 / (num_rows + 1)
    base_spacing_y = base_spacing_x * np.sqrt(3) / 2
    
    count = 0
    for row in range(num_rows):
        cols_in_row = 6 if row % 2 == 0 else 5
        for col in range(cols_in_row):
            if count >= n:
                break
            x = (col + 1) * base_spacing_x
            y = (row + 1) * base_spacing_y
            if row % 2 == 1:
                x += base_spacing_x / 2
            centers.append([x, y])
            radii.append(0.06)
            count += 1
        if count >= n:
            break
    
    return np.array(centers), np.array(radii[:n])


def objective_func(x, n):
    """Objective: minimize negative sum of radii."""
    radii = x[2::3]
    return -np.sum(radii)


def boundary_constraint_func(x, n, i, ctype):
    """Boundary constraint: circle i inside unit square.
    ctype: 0=x-r>=0, 1=1-x-r>=0, 2=y-r>=0, 3=1-y-r>=0
    """
    xi = x[3 * i]
    yi = x[3 * i + 1]
    ri = x[3 * i + 2]
    if ctype == 0:
        return xi - ri
    elif ctype == 1:
        return 1 - xi - ri
    elif ctype == 2:
        return yi - ri
    elif ctype == 3:
        return 1 - yi - ri
    return 0


def overlap_constraint_func(x, n, i, j):
    """Non-overlap constraint: squared distance >= (r_i + r_j)^2."""
    dx = x[3 * i] - x[3 * j]
    dy = x[3 * i + 1] - x[3 * j + 1]
    rsum = x[3 * i + 2] + x[3 * j + 2]
    return dx * dx + dy * dy - rsum * rsum


def boundary_jacobian(x, n, i, ctype):
    """Jacobian for boundary constraint."""
    jac = np.zeros(3 * n)
    idx_base = 3 * i
    if ctype == 0:
        jac[idx_base] = 1.0
        jac[idx_base + 2] = -1.0
    elif ctype == 1:
        jac[idx_base] = -1.0
        jac[idx_base + 2] = -1.0
    elif ctype == 2:
        jac[idx_base + 1] = 1.0
        jac[idx_base + 2] = -1.0
    elif ctype == 3:
        jac[idx_base + 1] = -1.0
        jac[idx_base + 2] = -1.0
    return jac


def overlap_jacobian(x, n, i, j):
    """Jacobian for overlap constraint."""
    jac = np.zeros(3 * n)
    dx = x[3 * i] - x[3 * j]
    dy = x[3 * i + 1] - x[3 * j + 1]
    rsum = x[3 * i + 2] + x[3 * j + 2]
    
    idx_i = 3 * i
    idx_j = 3 * j
    
    jac[idx_i] = 2 * dx
    jac[idx_i + 1] = 2 * dy
    jac[idx_i + 2] = -2 * rsum
    
    jac[idx_j] = -2 * dx
    jac[idx_j + 1] = -2 * dy
    jac[idx_j + 2] = -2 * rsum
    
    return jac


def run_optimization(x0, n):
    """Run the SLSQP optimization."""
    bounds = []
    for i in range(n):
        bounds.append((0.0, 1.0))   # x
        bounds.append((0.0, 1.0))   # y
        bounds.append((0.0, 0.5))   # r
    
    constraints = []
    
    # Add boundary constraints with Jacobians
    for i in range(n):
        for ctype in range(4):
            constraints.append({
                'type': 'ineq',
                'fun': partial(boundary_constraint_func, n=n, i=i, ctype=ctype),
                'jac': partial(boundary_jacobian, n=n, i=i, ctype=ctype)
            })
    
    # Add overlap constraints with Jacobians
    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({
                'type': 'ineq',
                'fun': partial(overlap_constraint_func, n=n, i=i, j=j),
                'jac': partial(overlap_jacobian, n=n, i=i, j=j)
            })
    
    result = minimize(
        partial(objective_func, n=n),
        x0,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={
            'maxiter': 3000,
            'ftol': 1e-12,
            'disp': False
        }
    )
    
    return result


def unpack_result(x, n):
    """Extract centers and radii from optimization variable."""
    centers = x[:2 * n].reshape(n, 2)
    radii = x[2 * n:]
    return centers, radii


def pack_inputs(centers, radii, n):
    """Convert centers and radii to flat optimization variable."""
    x = np.zeros(3 * n)
    for i in range(n):
        x[3 * i] = centers[i, 0]
        x[3 * i + 1] = centers[i, 1]
        x[3 * i + 2] = radii[i]
    return x


def perturb_centers(centers, scale=0.02):
    """Add random perturbation to centers."""
    perturbed = centers.copy()
    n = centers.shape[0]
    noise_x = np.random.uniform(-scale, scale, n)
    noise_y = np.random.uniform(-scale, scale, n)
    perturbed[:, 0] = np.clip(centers[:, 0] + noise_x, 0.05, 0.95)
    perturbed[:, 1] = np.clip(centers[:, 1] + noise_y, 0.05, 0.95)
    return perturbed


def run_packing():

    n = 26
    best_sum = 0.0

    for idx, (centers_init, radii_init) in enumerate(initializations):
        x0 = pack_inputs(centers_init, radii_init, n)
        
        result = run_optimization(x0, n)
        
        centers_opt, radii_opt = unpack_result(result.x, n)
        current_sum = np.sum(radii_opt)
        
        # Check validity
        valid = True
        for i in range(n):
            x, y = centers_opt[i]
            r = radii_opt[i]
            if r < 0 or x < -0.001 or y < -0.001 or x > 1.001 or y > 1.001:
                valid = False
                break
            if x - r < -0.001 or x + r > 1.001 or y - r < -0.001 or y + r > 1.001:
                valid = False
                break
        
        if valid:
            # Check overlaps
            for i in range(n):
                for j in range(i + 1, n):
                    dist = np.sqrt(
                        (centers_opt[i, 0] - centers_opt[j, 0])**2 +
                        (centers_opt[i, 1] - centers_opt[j, 1])**2
                    )
                    if dist < radii_opt[i] + radii_opt[j] - 1e-10:
                        valid = False
                        break
                if not valid:
                    break
        
        if valid and current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers_opt.copy()
            best_radii = radii_opt.copy()

        x0 = pack_inputs(best_centers, best_radii, n)
