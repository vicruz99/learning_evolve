import numpy as np
from scipy.optimize import minimize

def compute_objective(v, n, mu):
    """
    Computes the objective: -sum(radii) + penalty for violations.
    Uses squared distances and penalties for smooth optimization.
    """
    xs = v[0::3]
    ys = v[1::3]
    rs = v[2::3]
    
    # Boundary violations: circles must stay inside [0,1]^2
    # r <= x <= 1-r  =>  violations: r - x > 0  or  x + r - 1 > 0
    v1 = np.maximum(0.0, rs - xs)
    v2 = np.maximum(0.0, xs + rs - 1.0)
    v3 = np.maximum(0.0, rs - ys)
    v4 = np.maximum(0.0, ys + rs - 1.0)
    boundary_pen = np.sum(v1**2 + v2**2 + v3**2 + v4**2)
    
    # Pairwise overlap violations using squared distances for smoothness
    diff_x = xs[:, None] - xs[None, :]
    diff_y = ys[:, None] - ys[None, :]
    dist_sq = diff_x**2 + diff_y**2
    
    sum_r = rs[:, None] + rs[None, :]
    # Constraint: dist >= r_i + r_j  <=>  dist^2 >= (r_i + r_j)^2
    overlap_sq = np.maximum(0.0, sum_r**2 - dist_sq)
    
    # Sum penalty only for i < j
    triu_idx = np.triu_indices(n, k=1)
    overlap_pen = np.sum(overlap_sq[triu_idx]**2)
    
    return -np.sum(rs) + mu * (boundary_pen + overlap_pen)

def run_packing():
    np.random.seed(42)
    n = 26
    best_val = -np.inf
    best_centers = None
    best_radii = None
    
    # Bounds: x in [0,1], y in [0,1], r in [1e-6, 0.5]
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)] * n + [(1e-6, 0.5)] * n
    mu = 10000.0  # Penalty weight
    
    for trial in range(10):
        # Generate initial hexagonal grid layout
        pts = []
        d = 0.18
        y_step = d * np.sqrt(3) / 2
        row = 0
        while len(pts) < n:
            for col in range(6):
                if len(pts) >= n:
                    break
                x = col * d + (row % 2) * d/2 + 0.05
                y = row * y_step + 0.05
                if x < 0.95 and y < 0.95:
                    pts.append((x, y))
            row += 1
        pts = pts[:n]
        
        # Add controlled noise to break symmetry
        noise = np.random.uniform(-0.02, 0.02, size=(n, 2))
        pts = [(max(0.02, min(0.98, p[0] + noise[i][0])), 
                max(0.02, min(0.98, p[1] + noise[i][1]))) for i, p in enumerate(pts)]
        
        r_init = 0.08 + np.random.uniform(0, 0.03)
        
        v0 = np.zeros(3*n)
        for i in range(n):
            v0[3*i] = pts[i][0]
            v0[3*i+1] = pts[i][1]
            v0[3*i+2] = r_init
            
        # Optimize
        res = minimize(compute_objective, v0, args=(n, mu), method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': 4000, 'ftol': 1e-15})
        
        rs_opt = res.x[2::3]
        # Recover penalty magnitude from objective value
        # obj = -sum(rs) + mu * pen  =>  mu * pen = obj + sum(rs)
        penalty_magnitude = res.fun + np.sum(rs_opt)
        
        # Accept if penalty is negligible (valid packing)
        if penalty_magnitude < 1e-4:
            val = np.sum(rs_opt)
            if val > best_val:
                best_val = val
                best_centers = np.column_stack((res.x[0::3], res.x[1::3]))
                best_radii = rs_opt
                
    # Fallback to last result if strict validity check failed (should not happen with high mu)
    if best_centers is None:
        best_centers = np.column_stack((res.x[0::3], res.x[1::3]))
        best_radii = res.x[2::3]
        best_val = np.sum(best_radii)
        
    return best_centers, best_radii, best_val
