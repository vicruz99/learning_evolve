import numpy as np
from scipy.optimize import minimize

def objective(vars):
    """Objective function: maximize sum of radii (minimize negative sum)"""
    n = 26
    return -np.sum(vars[2*n:])

def compute_constraints(vars):
    """Compute all inequality constraints: must be >= 0"""
    n = 26
    centers = vars[:2*n].reshape(n, 2)
    radii = vars[2*n:]
    nc = 4*n + n*(n-1)//2
    c = np.empty(nc)
    
    # Boundary constraints: x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
    c[:4*n:4] = centers[:, 0] - radii
    c[1:4*n:4] = 1.0 - centers[:, 0] - radii
    c[2:4*n:4] = centers[:, 1] - radii
    c[3:4*n:4] = 1.0 - centers[:, 1] - radii
    
    # Pairwise non-overlap constraints: dist(i,j) - (r_i + r_j) >= 0
    idx = 4*n
    for i in range(n):
        diff = centers[i] - centers[i+1:]
        dists = np.sqrt(np.sum(diff**2, axis=1))
        c[idx:idx+n-i-1] = dists - radii[i] - radii[i+1:]
        idx += n - i - 1
    return c

def run_packing():
    n = 26
    best_sum = -1.0
    best_vars = None
    
    inits = []
    # Multiple random starts with feasible initial radii
    for seed in range(12):
        rng = np.random.default_rng(seed)
        centers = rng.uniform(0.05, 0.95, size=(n, 2))
        radii = np.full(n, 0.035)
        inits.append(np.concatenate([centers.ravel(), radii]))
        
    # Structured grid starts (5x5 + 1) with small perturbations
    grid_pts = np.array([[i*0.2 + 0.1, j*0.2 + 0.1] for i in range(5) for j in range(5)])
    grid_pts = np.vstack([grid_pts, [[0.5, 0.5]]])
    for k in range(8):
        rng = np.random.default_rng(k + 100)
        centers = grid_pts[:n] + rng.normal(0, 0.012, size=(n, 2))
        centers = np.clip(centers, 0.05, 0.95)
        radii = np.full(n, 0.055)
        inits.append(np.concatenate([centers.ravel(), radii]))

    bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n
    cons = {'type': 'ineq', 'fun': compute_constraints}
    
    for init in inits:
        try:
            res = minimize(objective, init, method='SLSQP', bounds=bounds, constraints=cons, 
                           options={'maxiter': 1500, 'ftol': 1e-12, 'disp': False})
            if res.success:
                curr_sum = np.sum(res.x[2*n:])
                if curr_sum > best_sum:
                    c_val = compute_constraints(res.x)
                    # Accept if constraints are satisfied within reasonable numerical tolerance
                    if np.min(c_val) >= -1e-8:
                        best_sum = curr_sum
                        best_vars = res.x
        except Exception:
            continue
            
    if best_vars is None:
        # Fallback to first initialization if optimization fails completely
        best_vars = inits[0]
        best_sum = np.sum(inits[0][2*n:])
        
    centers = best_vars[:2*n].reshape(n, 2)
    radii = best_vars[2*n:]
    
    # Post-processing: ensure strict validity within validator's 1e-12 tolerance
    # Iteratively enforce constraints with a tiny safety margin
    margin = 1e-11
    valid_radii = np.copy(radii)
    for _ in range(3):
        for i in range(n):
            r = min(centers[i, 0] - margin, 1.0 - centers[i, 0] - margin, 
                    centers[i, 1] - margin, 1.0 - centers[i, 1] - margin)
            for j in range(i+1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                r = min(r, dist - valid_radii[j] - margin)
            valid_radii[i] = max(0.0, r)
            
    final_centers = centers
    final_radii = valid_radii
    final_sum = float(np.sum(final_radii))
    
    return final_centers, final_radii, final_sum
