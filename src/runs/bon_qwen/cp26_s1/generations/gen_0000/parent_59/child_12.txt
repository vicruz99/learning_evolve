import numpy as np
from scipy.optimize import minimize
import scipy

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    num_rows = 5
    
    # 1. Initialize centers based on a dense hexagonal pattern (5-6-5-6-4 layout)
    # This layout is geometrically optimal for 26 circles to maximize radius
    rows = [5, 6, 5, 6, 4]
    centers = []
    current_y = 0.1
    row_h = np.sqrt(3) * 0.1
    
    for i, count in enumerate(rows):
        # Calculate spacing to fit circles perfectly in width
        # Total width = 1, spacing d = (1 - 2*r) / (count - 1)
        # For initialization, assume r=0.1
        if count > 1:
            d = (1 - 0.2) / (count - 1)
        else:
            d = 0
        start_x = 0.1
        # Stagger shifted rows
        if i % 2 == 1:
            start_x = 0.1 + d / 2
        
        for j in range(count):
            centers.append([start_x + j * d, current_y])
        
        if i < num_rows - 1:
            current_y += row_h
            
    centers = np.array(centers)
    radii = np.full(n, 0.1)

    # 2. Optimization variables: 26 radii + 26 x-coords + 26 y-coords = 78 vars
    # Structure: [r1..r26, x1..x26, y1..y26]
    initial_vars = np.concatenate([radii, centers[:, 0], centers[:, 1]])
    bounds = [(0, 1)] * n + # radii
             [(0, 1)] * n + # x
             [(0, 1)] * n   # y

    def objective(params):
        # Maximize sum of radii => Minimize negative sum
        return -np.sum(params[:n])

    def constraints_func(params):
        radii = params[:n]
        cx = params[n:2*n]
        cy = params[2*n:]
        
        # 1. Boundary constraints: cx - r >= 0, cx + r <= 1, cy - r >= 0, cy + r <= 1
        bounds_con = []
        for i in range(n):
            r = radii[i]
            bounds_con.extend([
                cx[i] - r, 1 - (cx[i] + r),
                cy[i] - r, 1 - (cy[i] + r)
            ])
        
        # 2. Non-overlap constraints: dist(c1, c2) >= r1 + r2
        overlap_con = []
        for i in range(n):
            for j in range(i + 1, n):
                dx = cx[i] - cx[j]
                dy = cy[i] - cy[j]
                dist = np.sqrt(dx**2 + dy**2)
                overlap_con.append(dist - (radii[i] + radii[j]))
                
        return bounds_con + overlap_con

    # Use SLSQP for handling bounds and non-linear constraints
    constraints = [{'type': 'ineq', 'fun': constraints_func}]
    
    # Try to find the optimum. Nelder-Mead is robust for non-smooth functions.
    # We minimize negative sum of radii.
    # Using a penalty approach in objective to help with strict constraints if necessary,
    # but SLSQP with constraints is usually better for this geometry.
    
    res = minimize(objective, initial_vars, method='SLSQP', bounds=bounds, 
                   constraints=constraints, options={'maxiter': 2000, 'ftol': 1e-9})
    
    # If SLSQP fails or gets stuck, the penalty method below serves as a robust backup
    # We define a separate helper for penalty minimization if needed, but let's stick to SLSQP first.
    
    final_r = res.x[:n]
    final_cx = res.x[n:2*n]
    final_cy = res.x[2*n:]
    final_centers = np.column_stack((final_cx, final_cy))
    
    # Ensure no NaNs
    if np.isnan(final_r).any() or np.isnan(final_centers).any():
        return np.zeros((n, 2)), np.zeros(n), 0.0

    sum_r = np.sum(final_r)
    
    return final_centers, final_r, sum_r
