import numpy as np
from scipy.optimize import minimize

def generate_initial_layout(n):
    """
    Generates an initial layout of n circles using a hexagonal lattice pattern.
    """
    centers = []
    radii = []
    
    # Hexagonal packing parameters
    r_init = 0.07  # Start with a safe radius
    spacing_x = 2 * r_init
    spacing_y = np.sqrt(3) * r_init
    
    # Try to fit into the square, roughly 6 rows x 5 cols or similar
    row = 0
    col = 0
    count = 0
    
    # We generate points in a hexagonal pattern
    # Adjust start positions to be centered in the square
    y_start = 0.1
    x_start = 0.1
    
    while count < n:
        # Hexagonal offset for alternating rows
        x_offset = 0 if row % 2 == 0 else r_init
        
        x = x_start + col * spacing_x + x_offset
        y = y_start + row * spacing_y
        
        # Check if point is within a reasonable range before adding
        # We allow it to be slightly outside [0,1] for the optimizer to pull it in, 
        # but better to keep it close.
        if 0 <= x <= 1 and 0 <= y <= 1:
            centers.append([x, y])
            radii.append(r_init)
            count += 1
            col += 1
        else:
            col += 1
            # If we overshoot x, move to next row
            if x > 1.1:
                col = 0
                row += 1
    
    # If we didn't get enough circles (unlikely with these params), pad with random
    while len(centers) < n:
        centers.append([np.random.rand(), np.random.rand()])
        radii.append(r_init)
        
    centers = np.array(centers[:n])
    radii = np.array(radii[:n])
    return centers, radii

def constraints_func(vars_flat, n):

    Defines the constraints for the optimization problem.
    vars_flat: flattened array of [x1, y1, r1, x2, y2, r2, ...]
    """

    radii = np.zeros(n)
    
    for i in range(n):
        idx = 3 * i
        centers[i] = vars_flat[idx:idx+2]
        radii[i] = vars_flat[idx+2]
        
    constraints = []
    
    # 1. Boundary constraints: x +/- r in [0, 1], y +/- r in [0, 1]
    for i in range(n):
        # x - r >= 0
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: v[3*i] - v[3*i+2]})
        # 1 - (x + r) >= 0
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: 1 - (v[3*i] + v[3*i+2])})
        # y - r >= 0
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: v[3*i+1] - v[3*i+2]})
        # 1 - (y + r) >= 0
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: 1 - (v[3*i+1] + v[3*i+2])})
        
    # 2. Non-overlap constraints: dist(i, j) >= r_i + r_j
    # Equivalent to: dist^2 - (r_i + r_j)^2 >= 0
    for i in range(n):
        for j in range(i + 1, n):
            idx_i = 3 * i
            idx_j = 3 * j
            
            def overlap_con(v, i=i, j=j, idx_i=idx_i, idx_j=idx_j):
                x1, y1, r1 = v[idx_i], v[idx_i+1], v[idx_i+2]
                x2, y2, r2 = v[idx_j], v[idx_j+1], v[idx_j+2]
                dist_sq = (x1 - x2)**2 + (y1 - y2)**2
                sum_r = r1 + r2
                return dist_sq - sum_r**2
            
            constraints.append({'type': 'ineq', 'fun': overlap_con})
            
    return constraints

def objective_func(vars_flat):
    """
    Objective: Maximize sum of radii => Minimize negative sum of radii.
    """
    n = len(vars_flat) // 3
    radii = vars_flat[2::3]
    return -np.sum(radii)
