def objective_function(vars_flat, n, penalty_weight):
    x = vars_flat[0::3]
    y = vars_flat[1::3]
    r = vars_flat[2::3]
    
    # Objective: minimize -sum(r)
    obj_val = -np.sum(r)
    
    penalty = 0.0
    
    # Boundary penalties
    # x >= r => r - x <= 0
    penalty += np.sum(np.maximum(0.0, r - x)**2)
    # x + r <= 1 => x + r - 1 <= 0
    penalty += np.sum(np.maximum(0.0, x + r - 1.0)**2)
    # y >= r => r - y <= 0
    penalty += np.sum(np.maximum(0.0, r - y)**2)
    # y + r <= 1 => y + r - 1 <= 0
    penalty += np.sum(np.maximum(0.0, y + r - 1.0)**2)
    
    # Overlap penalties
    for i in range(n):
        xi, yi, ri = x[i], y[i], r[i]
        for j in range(i + 1, n):
            dx = xi - x[j]
            dy = yi - y[j]
            dist = math.sqrt(dx*dx + dy*dy)
            overlap = ri + r[j] - dist
            if overlap > 0:
                penalty += overlap * overlap
                
    return obj_val + penalty_weight * penalty

import numpy as np
from scipy.optimize import minimize
import math

def objective_function(vars_flat, n, penalty_weight):
    """
    Computes the objective value for the optimization.
    Objective: Minimize -sum(radii) + penalty_weight * (boundary_violations + overlap_violations).
    """
    x = vars_flat[0::3]
    y = vars_flat[1::3]
    r = vars_flat[2::3]
    
    # We want to maximize sum(r), so we minimize -sum(r)
    obj_val = -np.sum(r)
    
    penalty = 0.0
    
    # Boundary penalties
    # Constraint: r <= x <= 1 - r  =>  x - r >= 0, 1 - x - r >= 0
    # Violations: max(0, r - x), max(0, x + r - 1)
    penalty += np.sum(np.maximum(0.0, r - x)**2)
    penalty += np.sum(np.maximum(0.0, x + r - 1.0)**2)
    
    # Constraint: r <= y <= 1 - r
    penalty += np.sum(np.maximum(0.0, r - y)**2)
    penalty += np.sum(np.maximum(0.0, y + r - 1.0)**2)
    
    # Overlap penalties
    # Constraint: dist(i, j) >= r_i + r_j
    # Violation: max(0, r_i + r_j - dist)
    for i in range(n):
        xi, yi, ri = x[i], y[i], r[i]
        for j in range(i + 1, n):
            dx = xi - x[j]
            dy = yi - y[j]
            dist = math.sqrt(dx*dx + dy*dy)
            overlap = ri + r[j] - dist
            if overlap > 0:
                penalty += overlap * overlap
                
    return obj_val + penalty_weight * penalty

                # k <= (1 - r_est) / (2r_est)
