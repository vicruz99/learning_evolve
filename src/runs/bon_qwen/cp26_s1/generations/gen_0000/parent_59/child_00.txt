import numpy as np
from scipy.optimize import minimize

def compute_loss(pos, n, radii):
    """Compute the sum of squared overlaps for N circles."""
    c = pos.reshape(-1, 2)
    # Pairwise squared distances
    diff = np.sum((c[:, None, :] - c[None, :, :])**2, axis=2)
    d = np.sqrt(diff + 1e-15)
    r_sum = radii[:, None] + radii[None, :]
    overlap = r_sum - d
    np.fill_diagonal(overlap, 0.0)
    pos_overlap = np.maximum(0.0, overlap)
    return np.sum(pos_overlap**2)

def run_packing():
    n = 26
    np.random.seed(42)
    
    # 1. Hexagonal initialization
    r_start = 0.085
    centers_list = []
    y = r_start
    row = 0
    while y <= 1.0 - r_start:
        x = r_start
        shift = (row % 2) * r_start
        while x <= 1.0 - r_start:
            centers_list.append([x + shift, y])
            x += 2.0 * r_start
        y += np.sqrt(3.0) * r_start
        row += 1
        
    centers = np.array(centers_list[:n])
    if len(centers) < n:
        diff = n - len(centers)
        extra = np.random.uniform([r_start, r_start], [1-r_start, 1-r_start], size=(diff, 2))
        centers = np.vstack([centers, extra])
        
    radii = np.full(n, r_start)
    
    # 2. Iterative expansion & relaxation
    for it in range(1500):
        # Bounds ensure circles stay strictly inside [0,1]^2
        bnds = []
        for i in range(n):
            lo = radii[i]
            hi = 1.0 - radii[i]
            bnds.append((lo, hi))
            bnds.append((lo, hi))
            
        res = minimize(compute_loss, centers.flatten(), args=(n, radii), 
                       method='L-BFGS-B', bounds=bnds, 
                       options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-7})
        centers = res.x.reshape(-1, 2)
        
        # 3. Adaptive scaling based on optimization success
        if res.fun < 1e-8:
            scale = 1.0025
        elif res.fun < 1e-4:
            scale = 1.001
        else:
            scale = 1.0002
            
        radii *= scale
        
        # Safety break slightly above expected optimum (~0.1014)
        if radii[0] > 0.1025:
            break
            
    return centers, radii, np.sum(radii)
