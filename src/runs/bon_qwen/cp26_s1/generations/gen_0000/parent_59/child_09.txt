import numpy as np
import math
import scipy.optimize
import scipy.spatial

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    n = centers.shape[0]
    if np.isnan(centers).any() or np.isnan(radii).any():
        return False
    for i in range(n):
        if radii[i] < 0:
            return False
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            return False
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False
    return True

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    # 1. Initialize in a perturbed hexagonal grid
    k = int(math.ceil(math.sqrt(n * math.sqrt(3) / 2)))
    spacing = 1.0 / (k + 1)
    idx = 0
    for row in range(k + 2):
        for col in range(k + 2):
            if idx >= n:
                break
            x = (col + 0.5 + (row % 2) * 0.5) * spacing
            y = (row + 0.5) * spacing * math.sqrt(3) / 2
            # Clamp into [0.05, 0.95] for safety margin
            centers[idx, 0] = np.clip(x, 0.05, 0.95) + np.random.uniform(-0.01, 0.01)
            centers[idx, 1] = np.clip(y, 0.05, 0.95) + np.random.uniform(-0.01, 0.01)
            idx += 1
    
    # 2. Iterative optimization loop
    for iteration in range(5000):
        # Solve LP to maximize sum of radii for fixed centers
        dist_matrix = scipy.spatial.distance.cdist(centers, centers)
        dist_lower = np.tril(dist_matrix, -1)
        flat_dist = dist_lower.flatten()
        
        # Constraints for LP: A_ub * r <= b_ub
        # Pairwise: r_i + r_j <= dist_ij
        # Boundary: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
        
        n_ineq = len(flat_dist) + 4 * n
        A_ub = np.zeros((n_ineq, n))
        b_ub = np.zeros(n_ineq)
        
        # Pairwise constraints
        row_idx = 0
        for i in range(n):
            for j in range(i + 1, n):
                A_ub[row_idx, i] = 1
                A_ub[row_idx, j] = 1
                b_ub[row_idx] = dist_matrix[i, j]
                row_idx += 1
        
        # Boundary constraints
        for i in range(n):
            A_ub[row_idx, i] = 1; b_ub[row_idx] = centers[i, 0]; row_idx += 1
            A_ub[row_idx, i] = 1; b_ub[row_idx] = 1 - centers[i, 0]; row_idx += 1
            A_ub[row_idx, i] = 1; b_ub[row_idx] = centers[i, 1]; row_idx += 1
            A_ub[row_idx, i] = 1; b_ub[row_idx] = 1 - centers[i, 1]; row_idx += 1
            
        # Objective: max sum(r) -> min -sum(r)
        c_obj = -np.ones(n)
        
        try:
            res = scipy.optimize.linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
            if res.success:
                radii = np.maximum(res.x, 1e-9)
                current_sum = np.sum(radii)
            else:
                # Fallback if LP fails
                radii.fill(0.01)
                current_sum = n * 0.01
        except Exception:
            radii.fill(0.01)
            current_sum = n * 0.01

        # Calculate forces based on constraint tightness
        forces = np.zeros((n, 2))
        for i in range(n):
            for j in range(i + 1, n):
                d = dist_matrix[i, j]
                if d > 1e-12:
                    overlap = radii[i] + radii[j] - d
                    if overlap > 0:
                        force_mag = overlap * 5.0 / (iteration + 10)
                        f = force_mag * (centers[i] - centers[j]) / d
                        forces[i] += f
                        forces[j] -= f
                        # Push centers apart to allow larger radii in next LP step
                        push = overlap * 0.5
                        centers[i] -= push * (centers[i] - centers[j]) / d
                        centers[j] += push * (centers[i] - centers[j]) / d
            
            # Boundary repulsion
            if centers[i, 0] - radii[i] < 0:
                centers[i, 0] = radii[i] + 1e-4
            if centers[i, 0] + radii[i] > 1:
                centers[i, 0] = 1 - radii[i] - 1e-4
            if centers[i, 1] - radii[i] < 0:
                centers[i, 1] = radii[i] + 1e-4
            if centers[i, 1] + radii[i] > 1:
                centers[i, 1] = 1 - radii[i] - 1e-4

    # Final clamp to [0,1]
    centers[:, 0] = np.clip(centers[:, 0], 1e-9, 1 - 1e-9)
    centers[:, 1] = np.clip(centers[:, 1], 1e-9, 1 - 1e-9)
    
    # Final radius recalculation
    radii[:] = np.min([
        np.min(np.diff(np.sort(np.append(np.append(centers[:, 0], 1), 0)))),
        np.min(np.diff(np.sort(np.append(np.append(centers[:, 1], 1), 0)))),
        np.min(dist_matrix[np.tril_indices(n, -1)]) / 2
    ], axis=0)
    radii = np.clip(radii, 1e-9, 0.5)

    # Ensure final validity
    if not validate_packing(centers, radii):
        # Fallback to simple grid
        grid = np.linspace(0.1, 0.9, 5)
        c, r = np.meshgrid(grid, grid)
        centers = np.vstack([c.flatten(), r.flatten()]).T[:n]
        radii = np.full(n, 0.09)

    return centers, radii, float(np.sum(radii))
