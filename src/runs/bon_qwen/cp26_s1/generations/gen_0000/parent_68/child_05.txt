We approach this as a constrained nonlinear optimization problem. The objective is to maximize the sum of radii $\sum r_i$, which is equivalent to minimizing $-\sum r_i$. The constraints ensure each circle lies within the unit square $[0,1]^2$ and that no two circles overlap (distance between centers $\ge r_i + r_j$). 

Given the non-convex nature of circle packing, standard gradient-based solvers can get trapped in local minima. To mitigate this, we use `scipy.optimize.minimize` with the SLSQP algorithm, running multiple trials from different initial configurations. We initialize with a perturbed 5x5 grid layout, which is a good starting point for $N=26$. We strictly avoid lambda functions and closures by using top-level classes for constraint evaluation. Analytical gradients for the objective function are supplied to improve solver performance. The bounds and inequality constraints are formulated directly to match the problem statement.

```python
import numpy as np
from scipy.optimize import minimize

# --- Constraint Classes (Top-level, no closures, no lambdas) ---
class XLowerBound:
    def __init__(self, i):
        self.i = i
    def __call__(self, v):
        return v[3*self.i] - v[3*self.i+2]

class XUpperBound:
    def __init__(self, i):
        self.i = i
    def __call__(self, v):
        return 1.0 - v[3*self.i] - v[3*self.i+2]

class YLowerBound:
    def __init__(self, i):
        self.i = i
    def __call__(self, v):
        return v[3*self.i+1] - v[3*self.i+2]

class YUpperBound:
    def __init__(self, i):
        self.i = i
    def __call__(self, v):
        return 1.0 - v[3*self.i+1] - v[3*self.i+2]

class PairDistance:
    def __init__(self, i, j):
        self.i = i
        self.j = j
    def __call__(self, v):
        xi, yi, ri = 3*self.i, 3*self.i+1, 3*self.i+2
        xj, yj, rj = 3*self.j, 3*self.j+1, 3*self.j+2
        dx = v[xi] - v[xj]
        dy = v[yi] - v[yj]
        dr = v[ri] + v[rj]
        return dx*dx + dy*dy - dr*dr

def run_packing():
    n = 26
    # Variable layout: [x0, y0, r0, x1, y1, r1, ...]
    n_vars = 3 * n
    
    # Bounds: centers in [0, 1], radii in [1e-6, 0.5]
    bounds = [(0.0, 1.0)] * (2 * n) + [(1e-6, 0.5)] * n
    
    # Build constraints
    cons = []
    for i in range(n):
        cons.append({'type': 'ineq', 'fun': XLowerBound(i)})
        cons.append({'type': 'ineq', 'fun': XUpperBound(i)})
        cons.append({'type': 'ineq', 'fun': YLowerBound(i)})
        cons.append({'type': 'ineq', 'fun': YUpperBound(i)})
        
    for i in range(n):
        for j in range(i + 1, n):
            cons.append({'type': 'ineq', 'fun': PairDistance(i, j)})
            
    # Objective and its analytical gradient
    def obj(v):
        return -np.sum(v[2::3])
        
    jac_obj = np.zeros(n_vars)
    jac_obj[2::3] = -1.0
    
    best_val = -np.inf
    best_v = None
    
    # Multiple restarts to escape local minima
    for trial in range(12):
        v0 = np.zeros(n_vars)
        idx = 0
        
        # Initialize with a grid pattern
        for r in range(5):
            for c in range(5):
                if idx < n:
                    v0[3*idx] = 0.1 + c * 0.2
                    v0[3*idx+1] = 0.1 + r * 0.2
                    v0[3*idx+2] = 0.04
                    idx += 1
                    
        # Place remaining circles near center with noise
        for rem in range(idx, n):
            v0[3*rem] = 0.5 + 0.08 * np.random.randn()
            v0[3*rem+1] = 0.5 + 0.08 * np.random.randn()
            v0[3*rem+2] = 0.04
            
        # Ensure initial feasibility bounds
        v0[:2*n] = np.clip(v0[:2*n], 0.02, 0.98)
        
        res = minimize(obj, v0, method='SLSQP', jac=jac_obj, bounds=bounds, constraints=cons,
                       options={'maxiter': 2000, 'ftol': 1e-10, 'disp': False})
                       
        if not np.isnan(res.fun) and -res.fun > best_val:
            best_val = -res.fun
            best_v = res.x
            
    # Extract results
    centers = best_v[:2*n].reshape((n, 2))
    radii = best_v[2*n:]
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
```
