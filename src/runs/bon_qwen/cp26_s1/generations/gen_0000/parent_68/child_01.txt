import numpy as np
from scipy.optimize import minimize


def initialize_hexagonal(n, seed=0):
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.08
    
    row_counts = [6, 5, 6, 5, 4]
    idx = 0
    spacing = 0.17
    
    rng = np.random.RandomState(seed)
    
    for row, count in enumerate(row_counts):
        for col in range(count):
            offset = spacing / 2 if row % 2 == 1 else 0
            x = 0.5 + col * spacing - (count - 1) * spacing / 2 + offset
            y = 0.12 + row * spacing * np.sqrt(3) / 2
            if idx < n:
                centers[idx, 0] = np.clip(x + rng.uniform(-0.01, 0.01), 0.05, 0.95)
                centers[idx, 1] = np.clip(y + rng.uniform(-0.01, 0.01), 0.05, 0.95)
                idx += 1
    
    return centers, radii


def force_optimize(centers, radii, n, steps=1500):
    centers = centers.copy()
    radii = radii.copy()
    
    for step in range(steps):
        forces = np.zeros((n, 2))
        
        for i in range(n):
            for j in range(i + 1, n):
                diff = centers[i] - centers[j]
                dist_sq = np.sum(diff ** 2)
                dist = np.sqrt(dist_sq) if dist_sq > 1e-10 else 1e-5
                desired_dist = radii[i] + radii[j] + 0.008
                if dist < desired_dist:
                    strength = (desired_dist - dist) / (dist + 1e-6) * 8.0
                    forces[i, 0] += strength * diff[0] / dist
                    forces[i, 1] += strength * diff[1] / dist
                    forces[j, 0] -= strength * diff[0] / dist
                    forces[j, 1] -= strength * diff[1] / dist
        
        for i in range(n):
            for axis in range(2):
                margin = radii[i] + 0.003
                if centers[i, axis] < margin:
                    forces[i, axis] += (margin - centers[i, axis]) * 20.0
                if centers[i, axis] > 1.0 - margin:
                    forces[i, axis] -= (centers[i, axis] - (1.0 - margin)) * 20.0
        
        lr = 0.012 * max(0.08, 1.0 - step / steps)
        centers += forces * lr
        centers = np.clip(centers, 0.015, 0.985)
    
    return centers, radii


def grow_radii(centers, radii, n, steps=150):
    radii = radii.copy()
    
    for step in range(steps):
        min_gap = 1.0
        
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                gap = dist - radii[i] - radii[j]
                if gap < min_gap:
                    min_gap = gap
            
            for axis in range(2):
                gap_left = centers[i, axis] - radii[i]
                gap_right = 1.0 - centers[i, axis] - radii[i]
                if gap_left < min_gap:
                    min_gap = gap_left
                if gap_right < min_gap:
                    min_gap = gap_right
        
        if min_gap > 0.0015:
            increase = min_gap * 0.18
            radii += increase
    
    return radii


def objective_penalty(vars, n):
    centers = vars[:2 * n].reshape(n, 2)
    radii = vars[2 * n:]
    
    obj = -np.sum(radii)
    
    penalty = 0.0
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            overlap = radii[i] + radii[j] - dist
            if overlap > 0:
                penalty += overlap ** 2
        
        for axis in range(2):
            if centers[i, axis] < radii[i]:
                penalty += (radii[i] - centers[i, axis]) ** 2
            if centers[i, axis] > 1.0 - radii[i]:
                penalty += (centers[i, axis] - (1.0 - radii[i])) ** 2
        
        if radii[i] < 0.0005:
            penalty += (radii[i] - 0.0005) ** 2 * 50.0
    
    return obj + 1500.0 * penalty


def run_packing():
    n = 26
    
    best_centers = None
    best_radii = None
    best_sum = 0.0
    
    for trial in range(12):
        centers, radii = initialize_hexagonal(n, seed=trial * 7 + 3)
        
        for cycle in range(5):
            centers, radii = force_optimize(centers, radii, n, steps=1200)
            radii = grow_radii(centers, radii, n, steps=100)
        
        vars_init = np.concatenate([centers.ravel(), radii])
        bounds = [(0.02, 0.98)] * (2 * n) + [(0.001, 0.5)] * n
        
        result = minimize(
            objective_penalty,
            vars_init,
            args=(n,),
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 3000, 'ftol': 1e-14, 'gtol': 1e-10}
        )
        
        centers_opt = result.x[:2 * n].reshape(n, 2)
        radii_opt = result.x[2 * n:]
        
        current_sum = np.sum(radii_opt)
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers_opt.copy()
            best_radii = radii_opt.copy()
    
    return best_centers, best_radii, best_sum
