The optimal packing for equal circles (the "equal circle packing problem") is a good approximation, where the radius is determined by the densest arrangement (hexagonal) or boundary constraints. However, since radii can vary, we optimize all variables ($x, y, r$).

    *   **Objective**: Minimize $-\sum r_i$ (equivalent to maximizing sum of radii).
    *   **Constraints**:
        *   Boundary constraints: $r_i \le x_i \le 1-r_i$ and $r_i \le y_i \le 1-r_i$.

3.  **Robustness**: Run the optimizer multiple times from different random initializations (or perturbations) to find the best local optimum. Select the configuration with the highest sum of radii.

import numpy as np
from scipy.optimize import minimize

def run_packing():
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    Returns (centers, radii, sum_radii).
    """
    n = 26
    
    # Helper function to create a grid-like initialization
    def get_initial_guess(seed=None):
        rng = np.random.RandomState(seed)
        # Create a grid of points
        # Try to fit 26 points. A 5x5 grid has 25. 
        # We can use a perturbed hexagonal or just a dense grid selection.
        # Let's try a 6x5 grid (30 points) and pick 26, or just place 26 on a 5x6 grid.
        # A 5x6 grid: 5 cols, 6 rows.
        # x in [0.1, 0.9], y in [0.1, 0.9]
        
        # Generate 26 points on a grid structure to ensure they are spread out initially
        # Using a 5x5 grid plus one extra, or just random grid sampling
        
        # Let's create a set of candidate grid points
        xs = np.linspace(0.15, 0.85, 6) # 6 points
        ys = np.linspace(0.15, 0.85, 5) # 5 points
        grid_points = np.array([[x, y] for x in xs for y in ys])
        
        # We need 26 points.
        # Shuffle and pick 26
        indices = rng.permutation(len(grid_points))[:n]
        centers = grid_points[indices].copy()
        
        # Add small random noise
        centers += rng.uniform(-0.05, 0.05, size=centers.shape)
        
        # Clip to valid range [0.1, 0.9] roughly
        centers = np.clip(centers, 0.05, 0.95)
        
        # Initial radii: small enough to not overlap significantly
        # Grid spacing is roughly 0.17. So radius 0.04 should be safe.
        radii = np.full(n, 0.04)
        
        return centers, radii

    def objective(params):
        # params is flattened [x1, y1, r1, x2, y2, r2, ...]
        # We want to maximize sum of radii -> minimize negative sum
        radii = params[2::3]
        return -np.sum(radii)

    def get_constraints():
        constraints = []
        
        # Boundary constraints:
        # x_i >= r_i  => x_i - r_i >= 0
        # x_i <= 1 - r_i => 1 - x_i - r_i >= 0
        # y_i >= r_i => y_i - r_i >= 0
        # y_i <= 1 - r_i => 1 - y_i - r_i >= 0
        
        # Helper to extract variables
        def extract_vars(params):
            # params shape (3*n,)
            # centers (n, 2), radii (n,)
            # But scipy passes 1D array
            centers = np.zeros((n, 2))
            radii = np.zeros(n)
            for i in range(n):
                centers[i, 0] = params[3*i]
                centers[i, 1] = params[3*i+1]
                radii[i] = params[3*i+2]
            return centers, radii

        # We can define constraint functions that take params
        # However, creating 4*n + n*(n-1)/2 functions might be slow to define dynamically
        # But n=26 is small enough.
        
        # Actually, it's better to pass a function that computes all constraints at once?
        # SLSQP accepts a list of dicts.
        
        for i in range(n):
            idx_x = 3*i
            idx_y = 3*i + 1
            idx_r = 3*i + 2
            
            # x >= r
            constraints.append({
                'type': 'ineq',
                'fun': lambda p, i=i: p[idx_x] - p[idx_r]
            })
            # x <= 1 - r  => 1 - x - r >= 0
            constraints.append({
                'type': 'ineq',
                'fun': lambda p, i=i: 1.0 - p[idx_x] - p[idx_r]
            })
            # y >= r
            constraints.append({
                'type': 'ineq',
                'fun': lambda p, i=i: p[idx_y] - p[idx_r]
            })
            # y <= 1 - r
            constraints.append({
                'type': 'ineq',
                'fun': lambda p, i=i: 1.0 - p[idx_y] - p[idx_r]
            })

        # Non-overlap constraints:
        # (x_i - x_j)^2 + (y_i - y_j)^2 >= (r_i + r_j)^2
        # => dist_sq - (r_i + r_j)^2 >= 0
        
        for i in range(n):
            for j in range(i + 1, n):
                idx_xi = 3*i
                idx_yi = 3*i + 1
                idx_ri = 3*i + 2
                idx_xj = 3*j
                idx_yj = 3*j + 1
                idx_rj = 3*j + 2
                
                constraints.append({
                    'type': 'ineq',
                    'fun': lambda p, i=i, j=j: 
                        (p[idx_xi] - p[idx_xj])**2 + (p[idx_yi] - p[idx_yj])**2 - (p[idx_ri] + p[idx_rj])**2
                })
        
        return constraints

    best_sum = -np.inf
    best_centers = None
    best_radii = None

    constraints = get_constraints()
    bounds = [(0, 1), (0, 1), (0, 0.5)] * n # x, y in [0,1], r in [0, 0.5]

    # Try multiple random seeds to avoid local minima
    for seed in range(10):
        centers, radii = get_initial_guess(seed)
        x0 = np.zeros(3 * n)
        for i in range(n):
            x0[3*i] = centers[i, 0]
            x0[3*i+1] = centers[i, 1]
            x0[3*i+2] = radii[i]

        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints, 
                           options={'maxiter': 1000, 'ftol': 1e-9})
            
            if res.success:
                current_sum = -res.fun
                if current_sum > best_sum:
                    best_sum = current_sum
                    best_centers = res.x[:3*n].reshape(n, 3)[:, :2]
                    best_radii = res.x[:3*n].reshape(n, 3)[:, 2]
        except Exception as e:
            # If optimization fails, continue to next seed
            continue

    # Fallback if no optimization succeeded (unlikely given the logic)
    if best_centers is None:
        # Return a valid default (grid)
        centers, radii = get_initial_guess(42)
        best_centers = centers
        best_radii = radii
        best_sum = np.sum(radii)

    # Ensure radii are non-negative
    best_radii = np.maximum(best_radii, 0.0)
    
    return best_centers, best_radii, best_sum
