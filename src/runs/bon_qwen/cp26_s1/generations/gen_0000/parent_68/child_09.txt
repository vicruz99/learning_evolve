import numpy as np
from scipy.optimize import minimize

def run_packing():
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    """
    n_circles = 26
    
    # Strategy: Initialize with a hexagonal packing pattern
    # Pattern: 5, 4, 5, 4, 5, 3 circles in 6 rows
    row_counts = [5, 4, 5, 4, 5, 3]
    
    # Estimate initial radius based on height constraint for 6 rows
    # Height = 2r + (rows-1)*sqrt(3)*r <= 1
    # r * (2 + 5*sqrt(3)) <= 1 => r <= 1 / (2 + 8.66) ≈ 0.094
    # However, we can optimize positions, so we start with a safe guess.
    initial_r = 0.1
    
    centers = []
    radii = np.zeros(n_circles)
    
    # Build initial hexagonal layout
    current_idx = 0
    for row_idx, count in enumerate(row_counts):
        y = initial_r + row_idx * np.sqrt(3) * initial_r
        
        # Horizontal shift for odd rows to create hexagonal packing
        if row_idx % 2 == 1:
            x_start = 2 * initial_r
        else:
            x_start = initial_r
            
        for col_idx in range(count):
            x = x_start + col_idx * 2 * initial_r
            centers.append([x, y])
            radii[current_idx] = initial_r
            current_idx += 1
            
    centers = np.array(centers)

    def objective(x):
        # x contains [centers_flat, radii]
        # We want to maximize sum of radii, so we minimize negative sum
        radii_vec = x[2*n_circles:]
        return -np.sum(radii_vec)

    def constraints_factory():
        constraints = []
        
        # Boundary constraints: x - r >= 0, x + r <= 1, y - r >= 0, y + r <= 1
        for i in range(n_circles):
            ci = [2*i, 2*i + 1]
            ri = 2*n_circles + i
            
            # x_i - r_i >= 0
            constraints.append({
                'type': 'ineq',
                'fun': lambda x, idx=ci[0], r_idx=ri: x[idx] - x[r_idx]
            })
            # 1 - (x_i + r_i) >= 0
            constraints.append({
                'type': 'ineq',
                'fun': lambda x, idx=ci[0], r_idx=ri: 1.0 - x[idx] - x[r_idx]
            })
            # y_i - r_i >= 0
            constraints.append({
                'type': 'ineq',
                'fun': lambda x, idx=ci[1], r_idx=ri: x[idx] - x[r_idx]
            })
            # 1 - (y_i + r_i) >= 0
            constraints.append({
                'type': 'ineq',
                'fun': lambda x, idx=ci[1], r_idx=ri: 1.0 - x[idx] - x[r_idx]
            })
            
        # Non-overlap constraints: dist_ij >= r_i + r_j
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                ci = [2*i, 2*i + 1]
                cj = [2*j, 2*j + 1]
                ri = 2*n_circles + i
                rj = 2*n_circles + j
                
                def dist_constraint(x, idx_i=ci, idx_j=cj, r_i=ri, r_j=rj):
                    dx = x[idx_i[0]] - x[idx_j[0]]
                    dy = x[idx_i[1]] - x[idx_j[1]]
                    dist = np.sqrt(dx*dx + dy*dy)
                    return dist - x[r_i] - x[r_j]

                constraints.append({
                    'type': 'ineq',
                    'fun': dist_constraint
                })
        return constraints

    # Flatten initial guess
    x0 = np.concatenate([centers.flatten(), radii])
    
    # Bounds for variables: x, y in [0, 1], r in [0, 0.5]
    bounds = []
    for _ in range(n_circles):
        bounds.append((0, 1)) # x
        bounds.append((0, 1)) # y
    for _ in range(n_circles):
        bounds.append((0, 0.5)) # r

    cons = constraints_factory()
    
    best_result = None
    best_sum = -np.inf
    
    # Run optimization with multiple random restarts to find global optimum
    np.random.seed(42)
    n_restarts = 5
    
    for _ in range(n_restarts):
        # Perturb initial positions slightly
        x_perturbed = x0.copy()
        # Add small noise to centers
        noise = np.random.uniform(-0.05, 0.05, size=2*n_circles)
        x_perturbed[:2*n_circles] += noise
        # Clip to bounds
        x_perturbed[:2*n_circles] = np.clip(x_perturbed[:2*n_circles], 0, 1)
        # Reset radii to initial guess
        x_perturbed[2*n_circles:] = initial_r

        try:
            res = minimize(objective, x_perturbed, method='SLSQP', bounds=bounds, 
                           constraints=cons, options={'maxiter': 1000, 'ftol': 1e-9})
            
            if res.success:
                current_sum = -res.fun
                if current_sum > best_sum:
                    best_sum = current_sum
                    best_result = res
        except Exception:
            continue

    if best_result is not None:
        best_centers = best_result.x[:2*n_circles].reshape(-1, 2)
        best_radii = best_result.x[2*n_circles:]
    else:
        # Fallback to initial configuration if optimization fails
        best_centers = centers
        best_radii = radii

    # Final validation and cleanup to ensure strict compliance
    # Adjust radii down slightly if there are any numerical overlaps
    # (Though the optimizer should have handled this)
    
    return best_centers, best_radii, np.sum(best_radii)
