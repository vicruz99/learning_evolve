import numpy as np
from scipy.optimize import minimize


def obj_fun(x, n):
    return -np.sum(x[2 * n :])


def bound_x_min(x, n, i):
    return x[2 * i] - x[2 * n + i]


def bound_x_max(x, n, i):
    return 1.0 - x[2 * i] - x[2 * n + i]


def bound_y_min(x, n, i):
    return x[2 * i + 1] - x[2 * n + i]


def bound_y_max(x, n, i):
    return 1.0 - x[2 * i + 1] - x[2 * n + i]


def no_overlap(x, n, i, j):
    dx = x[2 * i] - x[2 * j]
    dy = x[2 * i + 1] - x[2 * j + 1]
    dist = np.sqrt(dx * dx + dy * dy)
    return dist - x[2 * n + i] - x[2 * n + j]


def init_hex(n, spacing_x=0.14, spacing_y=0.12):
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.035
    idx = 0
    row = 0
    cols_per_row = [6, 5, 6, 5, 4]
    for row_idx in range(len(cols_per_row)):
        ncols = cols_per_row[row_idx]
        for col in range(ncols):
            if idx >= n:
                break
            x = 0.06 + col * spacing_x + (spacing_x / 2 if row_idx % 2 == 1 else 0)
            y = 0.06 + row_idx * spacing_y
            centers[idx] = [x, y]
            idx += 1
        if idx >= n:
            break
    return centers, radii


def expand_circles(centers, radii, n, steps=8000):
    centers = centers.copy()
    radii = radii.copy()
    for step in range(steps):
        radii *= 1.00008
        for _ in range(30):
            for i in range(n):
                for j in range(i + 1, n):
                    dx = centers[j, 0] - centers[i, 0]
                    dy = centers[j, 1] - centers[i, 1]
                    dist = np.sqrt(dx * dx + dy * dy)
                    if dist < radii[i] + radii[j] and dist > 1e-12:
                        overlap = radii[i] + radii[j] - dist
                        dx_n = dx / dist
                        dy_n = dy / dist
                        centers[i, 0] -= dx_n * overlap / 2
                        centers[i, 1] -= dy_n * overlap / 2
                        centers[j, 0] += dx_n * overlap / 2
                        centers[j, 1] += dy_n * overlap / 2
            for i in range(n):
                centers[i, 0] = max(radii[i], min(1 - radii[i], centers[i, 0]))
                centers[i, 1] = max(radii[i], min(1 - radii[i], centers[i, 1]))
    return centers, radii


def build_constraints(n):
    cons = []
    for i in range(n):
        cons.append({"type": "ineq", "fun": bound_x_min, "args": (n, i)})
        cons.append({"type": "ineq", "fun": bound_x_max, "args": (n, i)})
        cons.append({"type": "ineq", "fun": bound_y_min, "args": (n, i)})
        cons.append({"type": "ineq", "fun": bound_y_max, "args": (n, i)})
    for i in range(n):
        for j in range(i + 1, n):
            cons.append({"type": "ineq", "fun": no_overlap, "args": (n, i, j)})
    return cons


def optimize_from(centers, radii, n, cons, bounds):
    x0 = np.concatenate([centers.flatten(), radii])
    result = minimize(
        obj_fun,
        x0,
        args=(n,),
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": 5000, "ftol": 1e-14},
    )
    centers_opt = result.x[: 2 * n].reshape((n, 2))
    radii_opt = result.x[2 * n :]
    return centers_opt, radii_opt


def run_packing():
    n = 26
    bounds = [(0.0, 1.0)] * (2 * n) + [(1e-8, 0.5)] * n
    cons = build_constraints(n)

    best_sum = -1.0
    best_centers = None
    best_radii = None

    for trial in range(5):

        if trial > 0:

        current_sum = np.sum(radii_opt)
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers_opt.copy()
            best_radii = radii_opt.copy()

    # Final refinement with best result
    best_centers, best_radii = optimize_from(best_centers, best_radii, n, cons, bounds)

    return best_centers, best_radii, np.sum(best_radii)
