import numpy as np
import scipy.optimize as opt

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    best_sum = -1.0
    best_centers = None
    best_radii = None

    # Define row configurations for hexagonal packing
    # Config 1: 5, 6, 5, 6, 4
    row_counts_1 = [5, 6, 5, 6, 4]
    # Config 2: 6, 5, 6, 5, 4
    row_counts_2 = [6, 5, 6, 5, 4]

    row_configs = [row_counts_1, row_counts_2]

    def get_hex_centers(row_counts, scale=1.0):
        centers = []
        y_curr = 0.0
        # Vertical spacing for hex grid (assuming unit distance between neighbors)
        # dy = sqrt(3)/2
        dy = np.sqrt(3) / 2.0
        
        for i, count in enumerate(row_counts):
            row_y = y_curr + 0.5 # Start y at 0.5 to center vertically roughly
            # Calculate row x positions to center the row
            total_width = (count - 1) * 1.0
            start_x = (1.0 - total_width) / 2.0
            
            # Stagger odd rows
            if i % 2 == 1:
                start_x += 0.5
            
            for j in range(count):
                x = start_x + j * 1.0
                centers.append([x, row_y])
            
            y_curr += dy

        # Normalize to unit square [0, 1]
        # Calculate bounding box of generated centers
        cx = np.array(centers)
        min_x, min_y = np.min(cx[:, 0]), np.min(cx[:, 1])
        max_x, max_y = np.max(cx[:, 0]), np.max(cx[:, 1])
        
        span_x = max_x - min_x
        span_y = max_y - min_y
        
        # Scale to fit with a margin (scale factor < 1 to ensure feasibility)
        margin = 0.1 # Leave 0.1 radius margin initially
        max_possible_x = (1.0 - 2 * margin) / span_x
        max_possible_y = (1.0 - 2 * margin) / span_y
        s = min(max_possible_x, max_possible_y)
        
        # Scale and translate
        centers = (cx - np.array([min_x, min_y])) * s + np.array([margin, margin])
        return centers

    def objective(vars_flat):
        # vars_flat: [x1, y1, r1, x2, y2, r2, ...]
        # Maximize sum of radii -> Minimize negative sum
        r = vars_flat[2::3]
        return -np.sum(r)

    def constraints(vars_flat):
        cons = []
        centers = vars_flat[:52].reshape((n, 2))
        radii = vars_flat[52:]
        
        # Boundary constraints: x - r >= 0, 1 - x - r >= 0, etc.
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            cons.append({'type': 'ineq', 'fun': lambda v, i=i: v[3*i] - v[3*i+2]}) # x - r >= 0
            cons.append({'type': 'ineq', 'fun': lambda v, i=i: 1 - v[3*i] - v[3*i+2]}) # 1 - x - r >= 0
            cons.append({'type': 'ineq', 'fun': lambda v, i=i: v[3*i+1] - v[3*i+2]}) # y - r >= 0
            cons.append({'type': 'ineq', 'fun': lambda v, i=i: 1 - v[3*i+1] - v[3*i+2]}) # 1 - y - r >= 0
            
        # Overlap constraints
        for i in range(n):
            for j in range(i + 1, n):
                def dist_sq(v, i=i, j=j):
                    c1 = v[3*i:3*i+2]
                    c2 = v[3*j:3*j+2]
                    return np.sum((c1 - c2)**2)
                
                def sum_r(v, i=i, j=j):
                    return v[3*i+2] + v[3*j+2]

                cons.append({'type': 'ineq', 'fun': lambda v, i=i, j=j: dist_sq(v) - (sum_r(v))**2})
        
        return cons

    # Run optimization for multiple starts
    for rc in row_configs:
        for seed in range(5):
            # Generate initial centers
            init_centers = get_hex_centers(rc)
            
            # Initial radii (conservative estimate)
            min_dist = np.inf

                for j in range(i + 1, len(init_centers)):
                    d = np.linalg.norm(init_centers[i] - init_centers[j])
                    min_dist = min(min_dist, d)
            
            init_radii = np.full(n, min_dist / 2.0 * 0.9) # Slightly smaller to ensure validity

            # Flatten to 1D array
            x0 = np.zeros(3 * n)
            x0[:52] = init_centers.flatten()
            x0[52:] = init_radii

            # Bounds for variables
            bounds = []
            for _ in range(n):
                bounds.extend([(0, 1), (0, 1), (0, 0.5)])

            try:
                res = opt.minimize(
                    objective, 
                    x0, 
                    method='SLSQP', 
                    bounds=bounds, 
                    constraints=constraints(x0), # SLSQP can take list of dicts
                    options={'maxiter': 200, 'ftol': 1e-9}
                )
                
                if res.success:
                    r_final = res.x[52:]
                    c_final = res.x[:52].reshape(n, 2)
                    s = np.sum(r_final)
                    if s > best_sum:
                        best_sum = s
                        best_centers = c_final
                        best_radii = r_final

                continue

    # Fallback / Final check and slight perturbation if needed

        best_sum = np.sum(best_radii)

    return best_centers, best_radii, best_sum
