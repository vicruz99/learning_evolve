**Phase 1: Configuration Optimization (Maximin Distance)**
We first assume equal radii $r$. Maximizing the sum of radii ($26r$) is equivalent to finding the largest possible radius $r$ for 26 identical circles. This is a "maximin" problem: finding center coordinates $C$ that maximize the minimum distance between any two circles ($2r$) and between any circle and the boundary ($r$).
We solve this by defining a penalty function for constraint violations (overlaps and boundary breaches) and minimizing the negative of the radius using `scipy.optimize`. To ensure a robust result, we perform multiple optimization runs with different random initial configurations and select the best one.

**Phase 2: Unequal Radius Refinement (Linear Programming)**
Once an optimal set of centers is found, we relax the equal-radius constraint. We solve a Linear Programming (LP) problem to maximize the sum of radii $\sum r_i$ subject to the linear distance constraints ($r_i + r_j \le \text{dist}(c_i, c_j)$) and boundary constraints. import numpy as np
from scipy.optimize import minimize
import scipy.spatial.distance as dist

def compute_initial_configs(n_points, n_configs):
    configs = []
    for _ in range(n_configs):
        x = np.random.uniform(0.1, 0.9, size=n_points)
        y = np.random.uniform(0.1, 0.9, size=n_points)
        configs.append((x, y))
    return configs

def get_packing_score(coords, r):
    n = len(coords)
    x, y = coords[:, 0], coords[:, 1]
    
    # Boundary violations
    r_bound = np.min([x, 1-x, y, 1-y])
    
    # Inter-circle distances
    dists = dist.pdist(coords)
    min_dist = np.min(dists)
    r_inter = min_dist / 2.0
    
    return min(r_bound, r_inter)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # Phase 1: Find optimal centers for equal radii
    best_coords = None
    best_r = 0.0

            r = vars_[-1]
            penalty = 0.0
            coords = vars_[:-1].reshape(n, 2)
            
            # Boundary penalties
            bounds = np.min(np.concatenate([coords, 1 - coords]), axis=1)
            violations_b = np.maximum(0, r - bounds)
            penalty += np.sum(violations_b**2)
            
            # Overlap penalties

            violations_o = np.maximum(0, 2*r - dists)
            penalty += np.sum(violations_o**2)
            
            return -r + 1000 * penalty

        bounds = [(0, 1)] * (2 * n) + [0, 0.5]
        
        res = minimize(objective, np.concatenate([x0, y0, [0.05]]), 
                       method='L-BFGS-B', bounds=bounds, options={'ftol': 1e-12, 'maxiter': 2000})
        
        if res.success:
            current_r = res.x[-1]
            current_coords = res.x[:-1].reshape(n, 2)
            # Verify validity
            score = get_packing_score(current_coords, current_r)
            if score > best_r:
                best_r = score
                best_coords = current_coords.copy()

    # Phase 2: Refine radii using Linear Programming
    # Maximize sum(r_i) subject to r_i + r_j <= dist_ij and r_i <= dist(c_i, wall)
    n_vars = n
    c = -np.ones(n_vars) # Minimize -sum(r)
    
    A_ub = []
    b_ub = []
    
    # Inter-circle constraints
    dists = dist.pdist(best_coords)
    for idx, d in enumerate(dists):
        row = np.zeros(n_vars)
        # Map flat index to i, j

        # We can just iterate i, j again to match or use the mapping logic
        # Mapping: for i < j, index = i*(N-1) - i*(i+1)/2 + (j-i-1)
        # Easier: just re-iterate
        pass

    # Re-iterate for constraints construction
    for i in range(n):
        for j in range(i + 1, n):
            d = np.linalg.norm(best_coords[i] - best_coords[j])
            row = np.zeros(n)
            row[i] = 1
            row[j] = 1
            A_ub.append(row)
            b_ub.append(d)
            
        # Boundary constraints
        row = np.zeros(n)
        row[i] = 1
        # dist to nearest boundary
        wall_dist = min(best_coords[i,0], 1-best_coords[i,0], best_coords[i,1], 1-best_coords[i,1])
        A_ub.append(row)

        res_lp = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
        
        if res_lp.success:
            final_radii = res_lp.x
            sum_radii = np.sum(final_radii)
            return best_coords, final_radii, sum_radii
    except Exception:
        pass

    # Fallback to equal radii
    final_radii = np.full(n, best_r)
