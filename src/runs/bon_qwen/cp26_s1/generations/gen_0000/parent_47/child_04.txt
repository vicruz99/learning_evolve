def solve_radii_lp(centers):
    """
    Given fixed centers, solve the LP to maximize sum of radii.
    Maximize sum(r_i)
    Subject to:
      r_i + r_j <= dist(i, j) for all i < j
      r_i <= x_i
      r_i <= 1 - x_i
      r_i <= y_i
      r_i <= 1 - y_i
      r_i >= 0
    """
    n = centers.shape[0]
    
    # Objective: maximize sum(r) => minimize -sum(r)
    c_obj = -np.ones(n)
    
    # Constraints matrix A_ub x <= b_ub
    constraints = []
    bounds_list = []
    
    for i in range(n):
        x, y = centers[i]
        # Boundary constraints
        # r_i <= x
        # r_i <= 1-x
        # r_i <= y
        # r_i <= 1-y
        # r_i >= 0
        
        # Bounds for r_i
        max_r_i = min(x, 1-x, y, 1-y)
        if max_r_i < 0:
            max_r_i = 0
        bounds_list.append((0, max_r_i))
        
        # Pairwise constraints r_i + r_j <= d_ij
        for j in range(i + 1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            # Constraint: r_i + r_j <= dist
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            constraints.append((row, dist))
            
    if not constraints:
        return np.array([b[1] for b in bounds_list]) # Just max possible by bounds
    
    A_ub = np.array([c[0] for c in constraints])
    b_ub = np.array([c[1] for c in constraints])
    
    # Solve LP
    res = scipy.optimize.linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_list, method='highs')
    
    if res.success:
        return res.x
    else:
        # Fallback: return bounds if LP fails (shouldn't happen often)
        return np.array([b[1] for b in bounds_list])
