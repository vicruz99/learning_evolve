import numpy as np
from scipy.optimize import minimize

def dist(p1, p2):
    return np.sqrt(np.sum((p1 - p2) ** 2))

def boundary_dist(c, r):
    """
    Returns the violation of boundary constraints.
    Positive if violating, 0 otherwise.
    """
    x, y = c
    # Circle must be inside [0, 1]x[0, 1]
    # x - r >= 0, x + r <= 1, y - r >= 0, y + r <= 1
    violations = 0.0
    if x - r < 0: violations += (r - x)**2
    if x + r > 1: violations += (x + r - 1)**2
    if y - r < 0: violations += (r - y)**2
    if y + r > 1: violations += (y + r - 1)**2
    return violations

def objective(z, n_circles):
    """
    Objective function to minimize:
    - Sum of radii + Penalty for overlaps and boundary violations.
    """
    # Reshape z into centers (n, 2) and radii (n,)
    # z contains [x1, y1, r1, x2, y2, r2, ...]
    centers = z[:3 * n_circles].reshape(n_circles, 3)[:, :2]
    radii = z[:3 * n_circles].reshape(n_circles, 3)[:, 2]
    
    # Note: z structure is interleaved: x1, y1, r1, x2, y2, r2...
    # Actually, let's flatten as [x1, y1, r1, x2, y2, r2, ...]
    # So z[0] is x1, z[1] is y1, z[2] is r1, z[3] is x2...
    
    centers = np.zeros((n_circles, 2))
    radii = np.zeros(n_circles)
    
    for i in range(n_circles):
        idx = i * 3
        centers[i, 0] = z[idx]
        centers[i, 1] = z[idx+1]
        radii[i] = z[idx+2]
        
    # Objective: Minimize -sum(radii)
    obj = -np.sum(radii)
    
    # Penalty weight
    mu = 1000.0
    
    # Boundary penalties
    for i in range(n_circles):
        obj += mu * boundary_dist(centers[i], radii[i])
        
    # Overlap penalties
    for i in range(n_circles):
        for j in range(i + 1, n_circles):
            d = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            overlap = radii[i] + radii[j] - d
            if overlap > 0:
                obj += mu * overlap ** 2
                
    return obj

def generate_initial_packing(n):

    Generate a hexagonal packing configuration for n circles.
    """
    # Estimate radius for hexagonal packing in unit square
    # Density ~ 0.9069. Area = n * pi * r^2 <= 0.9069
    # r_est = sqrt(0.9069 / (n * pi))
    # For n=26, r_est approx 0.105
    r_est = 0.1
    
    # Hexagonal spacing
    dx = 2 * r_est
    dy = np.sqrt(3) * r_est
    
    centers = []
    radii = []
    
    row = 0
    while len(centers) < n:
        col = 0
        while len(centers) < n:
            # Offset every other row
            if row % 2 == 1:
                x = r_est + col * dx + dx / 2
            else:
                x = r_est + col * dx
            
            y = r_est + row * dy
            
            if x + r_est <= 1.0 and y + r_est <= 1.0:
                centers.append([x, y])
                radii.append(r_est)
                if len(centers) == n:
                    break
            col += 1
        row += 1
        
    return np.array(centers), np.array(radii)
