import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def objective(x):
    radii = x[2 * N_CIRCLES:]
    return -np.sum(radii)


def boundary_constraint_left(x, i):
    return x[2 * i] - x[2 * N_CIRCLES + i]


def boundary_constraint_right(x, i):
    return 1 - x[2 * i] - x[2 * N_CIRCLES + i]


def boundary_constraint_bottom(x, i):
    return x[2 * i + 1] - x[2 * N_CIRCLES + i]


def boundary_constraint_top(x, i):
    return 1 - x[2 * i + 1] - x[2 * N_CIRCLES + i]


def boundary_constraint_radius(x, i):
    return x[2 * N_CIRCLES + i]


def nonoverlap_constraint(x, i, j):
    xi, yi = x[2 * i], x[2 * i + 1]
    xj, yj = x[2 * j], x[2 * j + 1]
    ri, rj = x[2 * N_CIRCLES + i], x[2 * N_CIRCLES + j]
    dist_sq = (xi - xj) ** 2 + (yi - yj) ** 2
    sum_r = ri + rj
    return dist_sq - sum_r ** 2


def init_hexagonal(n):
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.07
    
    spacing = 0.16
    row_spacing = spacing * np.sqrt(3) / 2
    
    idx = 0
    row = 0
    while idx < n:
        cols = 6 if row % 2 == 0 else 5
        for col in range(cols):
            if idx >= n:
                break
            x = 0.06 + col * spacing + (row % 2) * spacing / 2
            y = 0.06 + row * row_spacing
            centers[idx] = [x, y]
            radii[idx] = 0.07
            idx += 1
        row += 1
    
    return centers, radii


def init_smart(n):
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.08
    
    idx = 0
    
    corners = [[0.1, 0.1], [0.9, 0.1], [0.1, 0.9], [0.9, 0.9]]
    for c in corners:
        if idx < n:
            centers[idx] = list(c)
            radii[idx] = 0.11
            idx += 1
    
    edge_positions = []
    for i in range(2, 5):
        t = i / 5.0
        edge_positions.append([t, 0.09])
        edge_positions.append([t, 0.91])
        edge_positions.append([0.09, t])
        edge_positions.append([0.91, t])
    
    for e in edge_positions:
        if idx < n:
            centers[idx] = list(e)
            radii[idx] = 0.08
            idx += 1
    
    spacing = 0.11
    row_spacing = spacing * np.sqrt(3) / 2
    row = 0
    count = 0
    while count < n - idx:
        cols = 5 if row % 2 == 0 else 4
        for col in range(cols):
            if count >= n - idx:
                break
            x = 0.22 + col * spacing + (row % 2) * spacing / 2
            y = 0.22 + row * row_spacing
            centers[idx + count] = [x, y]
            radii[idx + count] = 0.06
            count += 1
        row += 1
    
    return centers, radii


def init_uniform_grid(n):
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.06
    
    grid_size = 6
    step = 0.8 / (grid_size - 1)
    
    idx = 0
    for i in range(grid_size):
        for j in range(grid_size):
            if idx >= n:
                break
            x = 0.1 + i * step
            y = 0.1 + j * step
            centers[idx] = [x, y]
            idx += 1
        if idx >= n:
            break
    
    return centers, radii


def build_constraints():
    constraints = []
    
    for i in range(N_CIRCLES):
        constraints.append({'type': 'ineq', 'fun': boundary_constraint_left, 'args': (i,)})
        constraints.append({'type': 'ineq', 'fun': boundary_constraint_right, 'args': (i,)})
        constraints.append({'type': 'ineq', 'fun': boundary_constraint_bottom, 'args': (i,)})
        constraints.append({'type': 'ineq', 'fun': boundary_constraint_top, 'args': (i,)})
        constraints.append({'type': 'ineq', 'fun': boundary_constraint_radius, 'args': (i,)})
    
    for i in range(N_CIRCLES):
        for j in range(i + 1, N_CIRCLES):
            constraints.append({'type': 'ineq', 'fun': nonoverlap_constraint, 'args': (i, j)})
    
    return constraints


def build_bounds():
    bounds = []
    for i in range(N_CIRCLES):
        bounds.append((0.001, 0.999))
        bounds.append((0.001, 0.999))
    bounds += [(0.001, 0.49) for _ in range(N_CIRCLES)]
    return bounds


def optimize_from_init(centers, radii, maxiter=3000):
    x0 = np.concatenate([centers.flatten(), radii])
    constraints = build_constraints()
    bounds = build_bounds()
    
    result = minimize(
        objective, 
        x0, 
        method='SLSQP', 
        constraints=constraints,
        bounds=bounds, 
        options={'maxiter': maxiter, 'ftol': 1e-14, 'disp': False}
    )
    
    centers_opt = result.x[:2 * N_CIRCLES].reshape(N_CIRCLES, 2)
    radii_opt = result.x[2 * N_CIRCLES:]
    
    return centers_opt, radii_opt, -result.fun


def run_packing():
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    constraints = build_constraints()
    bounds = build_bounds()
    
    init_functions = [init_hexagonal, init_smart, init_uniform_grid]
    
    for seed in range(30):
        np.random.seed(seed)
        
        for init_func in init_functions:
            centers, radii = init_func(N_CIRCLES)
            
            centers_opt, radii_opt, sum_r = optimize_from_init(centers, radii, maxiter=4000)
            
            if sum_r > best_sum:
                best_sum = sum_r
                best_centers = centers_opt.copy()
                best_radii = radii_opt.copy()
            
            centers_pert = centers.copy() + np.random.randn(N_CIRCLES, 2) * 0.04
            centers_pert = np.clip(centers_pert, 0.05, 0.95)
            radii_pert = radii.copy() + np.random.randn(N_CIRCLES) * 0.02
            radii_pert = np.clip(radii_pert, 0.02, 0.3)
            
            centers_opt2, radii_opt2, sum_r2 = optimize_from_init(centers_pert, radii_pert, maxiter=4000)
            
            if sum_r2 > best_sum:
                best_sum = sum_r2
                best_centers = centers_opt2.copy()
                best_radii = radii_opt2.copy()
        
        if best_centers is not None and seed > 0:
            x_best = np.concatenate([best_centers.flatten(), best_radii])
            x_pert = x_best + np.random.randn(len(x_best)) * 0.01
            x_pert[:2 * N_CIRCLES] = np.clip(x_pert[:2 * N_CIRCLES].reshape(N_CIRCLES, 2), 0.05, 0.95).flatten()
            x_pert[2 * N_CIRCLES:] = np.clip(x_pert[2 * N_CIRCLES:], 0.02, 0.3)
            
            result = minimize(
                objective,
                x_pert,
                method='SLSQP',
                constraints=constraints,
                bounds=bounds,
                options={'maxiter': 4000, 'ftol': 1e-14, 'disp': False}
            )
            
            centers_opt = result.x[:2 * N_CIRCLES].reshape(N_CIRCLES, 2)
            radii_opt = result.x[2 * N_CIRCLES:]
            sum_r = -result.fun
            
            if sum_r > best_sum:
                best_sum = sum_r
                best_centers = centers_opt.copy()
                best_radii = radii_opt.copy()
    
    if best_centers is None:
        centers, radii = init_hexagonal(N_CIRCLES)
        best_centers, best_radii, best_sum = optimize_from_init(centers, radii)
    
    return best_centers, best_radii, best_sum
