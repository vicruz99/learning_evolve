import numpy as np
from scipy.optimize import minimize

N = 26


def compute_violation(x):
    """Compute total constraint violation for configuration x."""
    centers = x[:2 * N].reshape(N, 2)
    radii = x[2 * N:]
    v = 0.0

    for i in range(N):
        cx, cy = centers[i]
        r = radii[i]
        # Boundary violations
        if cx < r:
            v += (cx - r) ** 2
        if cx > 1 - r:
            v += (cx - (1 - r)) ** 2
        if cy < r:
            v += (cy - r) ** 2
        if cy > 1 - r:
            v += (cy - (1 - r)) ** 2
        if r < 0:
            v += r ** 2

    # Overlap violations
    for i in range(N):
        for j in range(i + 1, N):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            min_dist = radii[i] + radii[j]
            if dist < min_dist:
                v += (dist - min_dist) ** 2

    return v


def objective_with_penalty(x, penalty):
    """Maximize sum of radii minus penalty times violation."""
    radii = x[2 * N:]
    return -np.sum(radii) - penalty * compute_violation(x)


def make_objective(penalty):
    """Factory to create objective function with given penalty."""
    def obj(x):
        return objective_with_penalty(x, penalty)
    return obj


def hexagonal_init(r_init):
    """Create hexagonal lattice initial configuration."""
    r = r_init
    centers = []
    row = 0
    while len(centers) < N:
        col = 0
        while col * 2 * r + 2 * r <= 1.01 and len(centers) < N:
            x = col * 2 * r + r
            y = row * r * np.sqrt(3) + r
            if row % 2 == 1:
                x += r
            if x + r <= 1.01 and y + r <= 1.01 and x >= -0.01 and y >= -0.01:
                centers.append([x, y])
            col += 1
        row += 1
        if row * r * np.sqrt(3) + 2 * r > 1.03:
            break

    centers = np.array(centers[:N])
    while centers.shape[0] < N:
        centers = np.vstack([centers, np.array([[0.5, 0.5]])])
    centers = centers[:N]
    radii = np.full(N, r)
    return centers, radii


def random_init():
    """Random initial configuration."""
    centers = np.random.uniform(0.15, 0.85, (N, 2))
    radii = np.full(N, 0.06)
    return centers, radii


def sparse_init():
    """Sparse initial configuration with some large and some small circles."""
    centers = []
    radii = []
    # Place some larger circles in corners
    corner_positions = [[0.2, 0.2], [0.8, 0.2], [0.2, 0.8], [0.8, 0.8], [0.5, 0.5]]
    for pos in corner_positions:
        centers.append(pos)
        radii.append(0.12)
    # Fill rest randomly
    while len(centers) < N:
        c = np.random.uniform(0.1, 0.9, 2)
        centers.append(c)
        radii.append(0.05)
    return np.array(centers[:N]), np.array(radii[:N])


def optimize_sequential(x0, n_phases=5):
    """Run sequential penalty optimization."""
    best_x = x0.copy()
    penalties = [50, 200, 1000, 5000, 20000, 100000, 500000]

    for idx in range(min(n_phases, len(penalties))):
        penalty = penalties[idx]
        obj = make_objective(penalty)

        bounds = []
        for i in range(N):
            bounds.append((0.0, 1.0))
            bounds.append((0.0, 1.0))
            bounds.append((1e-10, 0.5))

        result = minimize(obj, best_x, method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': 3000, 'ftol': 1e-16, 'gtol': 1e-10})
        best_x = result.x

    return best_x


def fix_small_violations(centers, radii, max_iter=100):
    """Post-process to fix small constraint violations."""
    centers = centers.copy()
    radii = radii.copy()

    for iteration in range(max_iter):
        # Fix boundary violations
        for i in range(N):
            r = radii[i]
            centers[i, 0] = max(r, min(1 - r, centers[i, 0]))
            centers[i, 1] = max(r, min(1 - r, centers[i, 1]))

        # Fix overlaps by shrinking smaller circle
        overlap_found = False
        for i in range(N):
            for j in range(i + 1, N):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                min_dist = radii[i] + radii[j]
                if dist < min_dist - 1e-12:
                    overlap = min_dist - dist
                    # Push centers apart
                    if dist > 1e-12:
                        nx = dx / dist
                        ny = dy / dist
                        push = overlap / 2 + 1e-10
                        centers[i, 0] += nx * push
                        centers[i, 1] += ny * push
                        centers[j, 0] -= nx * push
                        centers[j, 1] -= ny * push
                    # Also shrink radii if still overlapping
                    if dist < min_dist - 1e-12:
                        shrink = (min_dist - dist) / 2 + 1e-10
                        if radii[i] < radii[j]:
                            radii[i] = max(radii[i] - shrink, 1e-10)
                        else:
                            radii[j] = max(radii[j] - shrink, 1e-10)
                    overlap_found = True

        # Re-clamp to boundaries after pushing
        for i in range(N):
            r = radii[i]
            centers[i, 0] = max(r, min(1 - r, centers[i, 0]))
            centers[i, 1] = max(r, min(1 - r, centers[i, 1]))

        if not overlap_found:
            break

    return centers, radii


def try_configuration(x0, seed=None):
    """Try to optimize from a given initial configuration."""
    if seed is not None:
        np.random.seed(seed)

    x_opt = optimize_sequential(x0)
    centers = x_opt[:2 * N].reshape(N, 2)
    radii = x_opt[2 * N:]

    # Post-process
    centers, radii = fix_small_violations(centers, radii)

    # Ensure valid
    for i in range(N):
        r = radii[i]
        centers[i, 0] = max(r + 1e-10, min(1 - r - 1e-10, centers[i, 0]))
        centers[i, 1] = max(r + 1e-10, min(1 - r - 1e-10, centers[i, 1]))
        radii[i] = max(radii[i], 1e-10)

    viol = compute_violation(x_opt)
    return centers, radii, np.sum(radii), viol


def run_packing():

            best_centers = centers.copy()
            best_radii = radii.copy()
            best_sum = s
            best_viol = viol

        x_perturbed = x0 + noise
        # Clip to bounds

            x_perturbed[2 * i] = max(0, min(1, x_perturbed[2 * i]))
            x_perturbed[2 * i + 1] = max(0, min(1, x_perturbed[2 * i + 1]))
            x_perturbed[2 * N + i] = max(1e-10, min(0.5, x_perturbed[2 * N + i]))

        centers, radii, s, viol = try_configuration(x_perturbed)
