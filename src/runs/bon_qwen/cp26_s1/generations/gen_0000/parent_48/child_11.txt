def calculate_repulsive_forces(centers, radii, k_repulse=100.0, softness=1e-3):
    """
    Calculate repulsive forces between circles to resolve overlaps.
    """
    n = len(centers)
    forces = np.zeros_like(centers)
    
    # Vectorized distance calculation
    # Shape (n, n, 2)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist_sq = np.sum(diff**2, axis=2)
    
    # Avoid self-interaction and zero division
    np.fill_diagonal(dist_sq, np.inf)
    dist = np.sqrt(dist_sq)
    
    # Overlap amount: (r_i + r_j) - dist. Positive if overlapping.
    radii_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    overlap = radii_sum - dist
    
    # We only care about positive overlaps (penalty) or close contacts
    # Force magnitude proportional to overlap
    # F = k * overlap * direction
    # Direction is diff / dist
    
    # Mask for active interactions (avoid division by zero and infinite dist)
    mask = (overlap > 0) & (dist < 1.0) # Only consider pairs that are overlapping or close
    
    # Calculate forces
    # Force on i from j is repulsive (push away)
    # F_ij = k * overlap * (center_i - center_j) / dist
    # Since diff[i,j] = center_i - center_j
    
    # Normalize diff vector (careful with dist=0)
    dir_vec = diff / (dist[:, :, np.newaxis] + softness)
    
    # Force magnitude
    force_mag = k_repulse * np.maximum(overlap, 0)
    
    # Apply force to forces array
    # Sum forces over j for each i
    # forces[i] += sum_j (force_mag[i,j] * dir_vec[i,j])
    
    # Only apply where mask is true to save compute/avoid noise
    force_mag = np.where(mask, force_mag, 0.0)
    
    # Compute vector forces
    vec_forces = force_mag[:, :, np.newaxis] * dir_vec
    
    # Sum over the other circles (axis 1)
    total_force = np.sum(vec_forces, axis=1)
    
    return total_force
