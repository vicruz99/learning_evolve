import numpy as np

def solve_radii_lp(centers):
    """
    Solves the Linear Programming problem to find the maximum radii
    for a fixed set of centers.
    Maximize sum(r_i) subject to r_i + r_j <= dist(i, j) and boundary constraints.
    """
    n = centers.shape[0]
    
    # Objective: Maximize sum(r_i) => Minimize -sum(r_i)
    c = -np.ones(n)
    
    # Inequalities A_ub @ r <= b_ub
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= x, r_i <= 1-x, r_i <= y, r_i <= 1-y
    for i in range(n):
        x, y = centers[i]
        # r_i <= x
        row = np.zeros(n)
        row[i] = 1.0
        A_ub.append(row)
        b_ub.append(x)
        
        # r_i <= 1 - x
        A_ub.append(row)
        b_ub.append(1.0 - x)
        
        # r_i <= y
        A_ub.append(row)
        b_ub.append(y)
        
        # r_i <= 1 - y
        A_ub.append(row)
        b_ub.append(1.0 - y)
        
    # Distance constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dist)

    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    # Bounds for radii: r_i >= 0
    bounds = [(0, None) for _ in range(n)]
    
    try:
        res = opt.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, res.fun * -1 # Return radii and sum
        else:
            # Fallback if LP fails
            radii = np.zeros(n)
            return radii, 0.0
    except Exception:
        return np.zeros(n), 0.0
