def generate_hexagonal_grid(n, size=1.0):
    """Generates a hexagonal grid initialization for n points in a square."""
    points = []
    rows = int(np.ceil(np.sqrt(n * np.sqrt(3) / 2)))

    return np.array(points[:n])

def generate_grid(n, size=1.0):

def objective_smooth(points_flat, n, smoothness=100.0):

    Smooth approximation of -min_distance using Log-Sum-Exp.
    Minimizing this maximizes the minimum distance.

        return -np.max(min_dists) # Fallback
