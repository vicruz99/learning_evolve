import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    np.random.seed(42)

    # 1. Initialization: Hexagonal lattice pattern (4, 5, 4, 5, 4, 4)
    # This pattern is denser than a square grid and fits well within the square.
    centers = []
    r_init = 0.095
    dy = r_init * np.sqrt(3)
    
    # Generate centers for a 6-row staggered pattern
    rows = [5, 4, 5, 4, 5, 4] 
    for row_idx, num_circles in enumerate(rows):
        y = r_init + row_idx * dy
        # Shift odd rows (1, 3, 5) by r_init to center circles in gaps
        x_offset = r_init if row_idx % 2 == 1 else r_init
        
        # Calculate x spacing to fit 'num_circles' in width 1
        # Available width is 1 - 2*r_init. 
        # If num_circles > 1, spacing dx = (1 - 2*r_init) / (num_circles - 1)
        # If num_circles = 1, center at 0.5
        
        if num_circles == 1:
            centers.append([0.5, y])
        else:
            dx = (1 - 2 * r_init) / (num_circles - 1)
            for col in range(num_circles):
                x = x_offset + col * dx
                # Adjust offset slightly to center the row if needed, 
                # though simple logic works for the solver to start
                centers.append([x, y])

    centers = np.array(centers[:n]) # Trim to exactly 26
    radii = np.full(n, r_init)

    # 2. Optimization
    # We define a penalty function to handle constraints.
    # Minimize: -Sum(radii) + Penalty(overlap)
    
    def objective(x_vec):
        x_vec = np.array(x_vec)
        c = x_vec[:2*n].reshape(n, 2)
        r = x_vec[2*n:]
        
        # Sum of radii (negated for minimization)
        obj = -np.sum(r)
        
        # Penalty for overlaps
        # dist < r_i + r_j
        penalty = 0.0
        # Vectorized overlap check
        # c_i - c_j
        diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
        dist = np.sqrt(np.sum(diff**2, axis=2))
        
        # Lower triangle mask
        mask = np.tril(np.ones((n, n)), k=-1)
        
        # Constraints: r_i + r_j - dist <= 0
        # Violation is max(0, r_i + r_j - dist)
        sum_radii = r[:, np.newaxis] + r[np.newaxis, :]
        violation = np.maximum(0, sum_radii - dist)
        
        penalty = np.sum(mask * violation**2)
        
        # Penalty for boundaries (x - r < 0 etc)
        # We rely on bounds for simple bounds, but soft penalty for safety
        # Penalty for r < 0
        penalty += np.sum(np.maximum(0, -r)**2)
        
        return obj + 1000 * penalty

    # Flattened vector: [x1, y1, ..., x26, y26, r1, ..., r26]
    x0 = np.hstack([centers.flatten(), radii])
    
    # Bounds: x, y in [0, 1], r in [0, 0.5]
    bnds = []
    for _ in range(n):
        bnds.append((0.0, 1.0)) # x
        bnds.append((0.0, 1.0)) # y
    for _ in range(n):
        bnds.append((0.0, 0.5)) # r

    # Run optimization
    res = minimize(objective, x0, method='L-BFGS-B', bounds=bnds, 
                   options={'maxiter': 2000, 'ftol': 1e-9})

    final_centers = res.x[:2*n].reshape(n, 2)
    final_radii = res.x[2*n:]
    
    # Ensure non-negative radii
    final_radii = np.maximum(final_radii, 0)
    
    sum_radii = np.sum(final_radii)
    
    return final_centers, final_radii, sum_radii
