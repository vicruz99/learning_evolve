

Looking at this circle packing problem, I need to maximize the sum of radii for 26 circles in a unit square. The target is 2.636.

<strategy>
1. **Initial Configuration**: Start with a hexagonal lattice arrangement which is more space-efficient than a square grid. For 26 circles, I'll use 5 rows with alternating counts (e.g., 6, 5, 6, 5, 4) in a hexagonal pattern.

2. **Optimization Approach**: Use SLSQP with constraints:
   - Boundary constraints: circles must stay within [0,1]²
   - Non-overlap: distance between centers ≥ sum of radii
   - Maximize sum of radii (minimize negative sum)

3. **Multiple Restarts**: Try many different initial configurations with random perturbations to escape local optima.

4. **Key Insight**: The constraint ||c_i - c_j||² ≥ (r_i + r_j)² is differentiable and works better numerically than using square roots.

5. **Refinement Strategy**: 
   - Use squared distance constraints for better numerical stability
   - Add small epsilon to avoid division by zero in gradients
   - Run multiple optimization trials with different seeds
   - Keep track of best valid solution found

6. **Validation**: After optimization, clip any slight numerical violations to ensure the solution passes validation.
</strategy>

```python
import numpy as np
from scipy.optimize import minimize
import math


def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    best_centers = None
    best_radii = None
    best_sum = 0.0

    # Try many initial configurations
    for seed in range(200):
        centers, radii = _init_hexagonal(n, seed)
        centers, radii = _optimize(centers, radii, n)
        
        # Fix minor numerical violations
        centers, radii = _fix_violations(centers, radii, n)
        
        # Check validity
        if _is_valid(centers, radii, n):
            s = np.sum(radii)
            if s > best_sum:
                best_sum = s
                best_centers = centers.copy()
                best_radii = radii.copy()

    if best_centers is None:
        # Fallback
        best_centers, best_radii = _init_hexagonal(n, 42)
        best_sum = np.sum(best_radii)

    return best_centers, best_radii, best_sum


def _init_hexagonal(n, seed):
    np.random.seed(seed)
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    idx = 0
    row_height = 0.175
    col_width = 0.17
    
    for row in range(6):
        y = 0.12 + row * row_height
        offset = col_width / 2.0 if row % 2 == 1 else 0.0
        x_start = 0.10 + offset
        
        n_in_row = 6 if row % 2 == 0 else 4
        if idx + n_in_row > n:
            n_in_row = n - idx
        
        for col in range(n_in_row):
            if idx < n:
                centers[idx] = [x_start + col * col_width, y]
                radii[idx] = 0.075
                idx += 1
        if idx >= n:
            break

    # Add small random perturbation
    centers += np.random.randn(n, 2) * 0.015
    centers = np.clip(centers, 0.08, 0.92)

    return centers, radii


def _optimize(centers, radii, n):
    x0 = np.concatenate([centers.flatten(), radii])

    def objective(x):
        return -np.sum(x[2 * n:])

    def constraints_func(x):
        c = x[:2 * n].reshape((n, 2))
        r = x[2 * n:]
        cons = []

        for i in range(n):
            # Boundary constraints
            cons.append(c[i, 0] - r[i])
            cons.append(1.0 - c[i, 0] - r[i])
            cons.append(c[i, 1] - r[i])
            cons.append(1.0 - c[i, 1] - r[i])
            cons.append(r[i] - 1e-12)

        for i in range(n):
            for j in range(i + 1, n):
                dx = c[i, 0] - c[j, 0]
                dy = c[i, 1] - c[j, 1]
                dist_sq = dx * dx + dy * dy
                sum_r = r[i] + r[j]
                # Use squared formulation: dist^2 >= (r_i + r_j)^2
                cons.append(dist_sq - sum_r * sum_r)

        return np.array(cons)

    bounds = []
    for _ in range(n):
        bounds.extend([(0.0, 1.0), (0.0, 1.0)])
        bounds.append((1e-12, 0.5))

    result = minimize(
        objective,
        x0,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'ineq', 'fun': constraints_func},
        options={'maxiter': 8000, 'ftol': 1e-14, 'disp': False}
    )

    opt_centers = result.x[:2 * n].reshape((n, 2))
    opt_radii = result.x[2 * n:]

    return opt_centers, opt_radii


def _fix_violations(centers, radii, n):
    """Fix minor numerical violations."""
    centers = centers.copy()
    radii = radii.copy()

    # Fix boundary violations
    for i in range(n):
        r = radii[i]
        # Clip center to valid range given radius
        centers[i, 0] = np.clip(centers[i, 0], r + 1e-12, 1.0 - r - 1e-12)
        centers[i, 1] = np.clip(centers[i, 1], r + 1e-12, 1.0 - r - 1e-12)
        # If center is invalid even after clipping, reduce radius
        if centers[i, 0] > 1.0 - r - 1e-12 or centers[i, 0] < r + 1e-12:
            max_r = min(centers[i, 0], 1.0 - centers[i, 0])
            radii[i] = max(max_r - 1e-12, 1e-12)
        if centers[i, 1] > 1.0 - radii[i] - 1e-12 or centers[i, 1] < radii[i] + 1e-12:
            max_r = min(centers[i, 1], 1.0 - centers[i, 1])
            radii[i] = max(max_r - 1e-12, 1e-12)

    # Fix overlap violations by shrinking radii
    for _ in range(50):
        max_violation = 0
        worst_i, worst_j = -1, -1
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = math.sqrt(dx * dx + dy * dy)
                violation = radii[i] + radii[j] - dist
                if violation > max_violation:
                    max_violation = violation
                    worst_i, worst_j = i, j
        
        if max_violation < 1e-12:
            break
        
        # Reduce both radii proportionally
        if worst_i >= 0:
            radii[worst_i] -= max_violation * 0.51
            radii[worst_j] -= max_violation * 0.51
            radii[worst_i] = max(radii[worst_i], 1e-12)
            radii[worst_j] = max(radii[worst_j], 1e-12)

    return centers, radii


def _is_valid(centers, radii, n):
    for i in range(n):
        if radii[i] < 0:
            return False
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-10 or x + r > 1 + 1e-10:
            return False
        if y - r < -1e-10 or y + r > 1 + 1e-10:
            return False

    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = math.sqrt(dx * dx + dy * dy)
            if dist < radii[i] + radii[j] - 1e-10:
                return False

    return True
```
