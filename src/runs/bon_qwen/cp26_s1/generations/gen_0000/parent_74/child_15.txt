import numpy as np
from scipy.optimize import minimize

def compute_loss(v, n, penalty_weight):
    """
    Computes the negative sum of radii plus penalty terms for boundary and overlap violations.
    """
    centers = v[0:3*n].reshape(n, 3)[:, :2]
    radii = v[2::3]
    
    # Objective: maximize sum of radii => minimize negative sum
    loss = -np.sum(radii)
    
    # Boundary penalties
    for i in range(n):
        x, y, r = centers[i, 0], centers[i, 1], radii[i]
        if x - r < 0.0:
            loss += penalty_weight * (x - r)**2
        if x + r > 1.0:
            loss += penalty_weight * (x + r - 1.0)**2
        if y - r < 0.0:
            loss += penalty_weight * (y - r)**2
        if y + r > 1.0:
            loss += penalty_weight * (y + r - 1.0)**2
            
    # Overlap penalties
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist_sq = dx*dx + dy*dy
            min_dist = radii[i] + radii[j]
            # Penalty applied if squared distance is less than squared minimum required distance
            if dist_sq < min_dist**2:
                loss += penalty_weight * (min_dist**2 - dist_sq)**2
                
    return loss

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    centers_init = np.array(centers_init)
    radii_init = np.ones(n) * r_init
    
    # Flatten to optimization vector: [x0, y0, r0, x1, y1, r1, ...]

    # Optimize
    res = minimize(compute_loss, x0, args=(n, 5000.0), method='L-BFGS-B', bounds=bounds,

    return c, r, float(np.sum(r))
