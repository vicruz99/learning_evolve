        *   **Boundary Forces**: If a circle is constrained by a boundary wall, the dual variable indicates the benefit of moving the center away from that wall. We apply forces directed towards the center of the square.
    *   **Movement**: We update the centers based on these forces, clamping them to stay within the $[0, 1] \times [0, 1]$ square.

import numpy as np
from scipy.optimize import linprog

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # --- 1. Initialization ---
    # Place centers in a hexagonal grid pattern to start with a good packing density.
    # We aim for roughly 5-6 rows and 5-6 columns.
    centers = np.zeros((n, 2))
    
    # Layout strategy: 5 rows. 
    # Rows will have varying numbers of circles to fit 26.
    # Pattern: 5, 5, 6, 5, 5 (Total 26)
    # Or just a dense grid and let optimization fix it.
    # Let's try a 6x5 grid (30 points) and keep first 26.
    
    # Grid parameters
    cols = 6
    rows = 5
    
    # We want points distributed in [0.1, 0.9] roughly
    xs = np.linspace(0.15, 0.85, cols)
    ys = np.linspace(0.15, 0.85, rows)
    
    points = []
    for y in ys:
        for x in xs:
            points.append([x, y])
            
    # Take first 26 points
    centers = np.array(points[:n])
    
    # Small random perturbation to break symmetry
    centers += np.random.uniform(-0.02, 0.02, size=centers.shape)
    # Clamp to valid range
    centers = np.clip(centers, 0.01, 0.99)

    # --- 2. Optimization Loop ---
    # We will iterate to improve the sum of radii.
    num_iterations = 60
    step_size = 0.005 # Initial step size for center updates
    
    # Pre-calculate indices for pair constraints to speed up loop
    pair_indices = []
    for i in range(n):
        for j in range(i + 1, n):
            pair_indices.append((i, j))
    num_pairs = len(pair_indices)
    
    # Boundary constraints indices (4 per circle)
    # Constraint 0: r_i <= x_i
    # Constraint 1: r_i <= 1-x_i
    # Constraint 2: r_i <= y_i
    # Constraint 3: r_i <= 1-y_i
    num_boundary_constraints = n * 4
    total_constraints = num_pairs + num_boundary_constraints
    
    # Optimization loop
    for iteration in range(num_iterations):
        # --- Solve LP for Radii ---
        # Objective: Maximize sum(r) => Minimize -sum(r)
        c_obj = np.ones(n) * -1.0
        
        # Construct A_ub and b_ub
        # Shape of A_ub: (total_constraints, n)
        A_ub = np.zeros((total_constraints, n))
        b_ub = np.zeros(total_constraints)
        
        idx = 0
        
        # 1. Pair constraints: r_i + r_j <= dist(i, j)
        for i, j in pair_indices:
            dist = np.linalg.norm(centers[i] - centers[j])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dist
            idx += 1
            
        # 2. Boundary constraints
        for i in range(n):
            x, y = centers[i]
            
            # r_i <= x
            A_ub[idx, i] = 1.0
            b_ub[idx] = x
            idx += 1
            
            # r_i <= 1 - x
            A_ub[idx, i] = 1.0
            b_ub[idx] = 1.0 - x
            idx += 1
            
            # r_i <= y
            A_ub[idx, i] = 1.0
            b_ub[idx] = y
            idx += 1
            
            # r_i <= 1 - y
            A_ub[idx, i] = 1.0
            b_ub[idx] = 1.0 - y
            idx += 1
            
        bounds = [(0, None)] * n
        
        # Solve LP
        # method='highs' is efficient
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        except:
            break
            
        if not res.success:
            # If LP fails, we might have numerical issues, break or continue with old radii
            # For stability, let's just break
            break
            
        radii = res.x
        current_sum = np.sum(radii)

        force = np.zeros_like(centers)

        try:

                duals = res.marginals.ineqlin
            elif hasattr(res, 'ineqlin') and hasattr(res.ineqlin, 'marginals'):
                duals = res.ineqlin.marginals
            else:
                duals = None
        except:
            duals = None
            
        if duals is not None and len(duals) == total_constraints:
            # Pair forces
            idx = 0
            for i, j in pair_indices:

                lam = duals[idx]

                # Let's assume positive means "tight".

                        direction /= dist

                        # So we want to increase distance.

                idx += 1

                    force[i, 0] += lam_x

                if lam_1mx > 1e-7:

                            direction = vec / dist
                            force[i] += direction
                            force[j] -= direction

        centers = np.clip(centers, 0.0, 1.0)

        pass # Keep previous radii if fails

    # Calculate sum

    return centers, radii, sum_radii
