2. **Gradient Ascent with Soft Constraints**: We will implement a custom optimization routine using `scipy.optimize`. 
   - **Objective**: Maximize the sum of the radii ($\sum r_i$).
   - **Soft Constraints**: To handle the non-overlap and boundary conditions, we define a penalty function.

     - *Boundary Penalty*: If a circle protrudes past the unit square edges, a penalty is applied.
   
3. **Optimization Loop**: We use the L-BFGS-B algorithm to simultaneously optimize the $(x, y)$ centers and the radii $r$ of all 26 circles. The solver will push the circles apart and grow their radii until they form a "jammed" state where they are touching neighbors and the walls, maximizing the total sum of radii.

import numpy as np
from scipy.optimize import minimize

def generate_hex_grid(n):
    """
    Generates initial centers for n circles in a hexagonal pattern.
    """
    centers = []
    rows = int(np.ceil(np.sqrt(n)))
    cols = int(np.ceil(n / rows))
    
    # Estimated size based on hexagonal packing
    # Width ~ (cols - 0.5) * 2r, Height ~ (rows - 1) * sqrt(3)r + 2r
    # We pick a small safe radius for initialization
    r_init = 0.08 
    spacing_x = 2.0 * r_init
    spacing_y = np.sqrt(3) * r_init
    
    y = r_init + 0.05
    count = 0
    while count < n:
        x = r_init + 0.05
        row_len = 0
        while x <= 1.0 - r_init - 0.05 and count < n:
            centers.append([x, y])
            x += spacing_x
            row_len += 1
            count += 1
        # Shift every other row
        x_offset = spacing_x / 2.0
        if y + spacing_y + r_init < 1.0 and count < n:
             y += spacing_y
             centers.append([x_offset + r_init, y])
             x = x_offset + r_init + spacing_x
             while x <= 1.0 - r_init - 0.05 and count < n:
                 centers.append([x, y])
                 x += spacing_x
                 count += 1
                 y += spacing_y

    return np.array(centers[:n])

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    n = centers.shape[0]
    if np.isnan(centers).any() or np.isnan(radii).any():
        return False

    for i in range(n):
        if radii[i] < 0:
            return False
    
    # Check boundaries
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-6 or x + r > 1 + 1e-6 or y - r < -1e-6 or y + r > 1 + 1e-6:
            return False

    # Check overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-6:
                return False

    return True

def objective_and_grad(vars, n):
    """
    Calculates the objective (negative sum of radii) and gradients.
    Includes penalties for overlaps and boundary violations.
    """
    # Unpack variables
    # Order: x1, y1, r1, x2, y2, r2, ...
    centers = np.array(vars[:2*n]).reshape((n, 2))
    radii = np.array(vars[2*n:])

    # Objective: Minimize - sum(radii)
    obj = -np.sum(radii)
    
    # Initialize gradients
    grad = np.zeros_like(vars)
    
    # Gradient of -sum(radii) with respect to radii is -1
    grad[2*n:] = -1.0
    
    # Penalty constants
    penalty_weight = 100.0

    # 1. Boundary Penalties
    # x >= r  => r - x <= 0
    # x <= 1-r => x + r - 1 <= 0
    # y >= r  => r - y <= 0
    # y <= 1-r => y + r - 1 <= 0
    
    for i in range(n):
        idx = i
        x, y = centers[i]
        r = radii[i]
        
        # x < r
        if r > x:
            val = r - x
            penalty = val ** 2
            obj += penalty_weight * penalty
            # grad[r] += 2 * val
            grad[2*n + idx] += 2 * penalty_weight * val
            # grad[x] -= 2 * val
            grad[idx] -= 2 * penalty_weight * val
            
        # x > 1 - r
        if x + r > 1:
            val = x + r - 1
            penalty = val ** 2
            obj += penalty_weight * penalty
            # grad[r] += 2 * val
            grad[2*n + idx] += 2 * penalty_weight * val
            # grad[x] += 2 * val
            grad[idx] += 2 * penalty_weight * val

        # y < r
        if r > y:
            val = r - y
            penalty = val ** 2
            obj += penalty_weight * penalty
            # grad[r] += 2 * val
            grad[2*n + idx] += 2 * penalty_weight * val
            # grad[y] -= 2 * val
            grad[n + idx] -= 2 * penalty_weight * val
            
        # y > 1 - r
        if y + r > 1:
            val = y + r - 1
            penalty = val ** 2
            obj += penalty_weight * penalty
            # grad[r] += 2 * val
            grad[2*n + idx] += 2 * penalty_weight * val
            # grad[y] += 2 * val
            grad[n + idx] += 2 * penalty_weight * val

    # 2. Overlap Penalties
    for i in range(n):
        for j in range(i + 1, n):
            c_i = centers[i]
            c_j = centers[j]
            r_i = radii[i]
            r_j = radii[j]
            
            diff = c_i - c_j
            dist = np.sqrt(np.sum(diff**2))
            
            sum_radii = r_i + r_j
            
            if dist < sum_radii:
                # Overlap amount
                overlap = sum_radii - dist
                # Penalty: overlap^2
                # Derivative of overlap^2 wrt dist is -2*overlap
                # Derivative of dist wrt x_i is diff/dist
                # Derivative of sum_radii wrt r_i is 1
                
                # Gradient contribution
                g_factor = 2 * penalty_weight * overlap
                
                # d/d(dist) term
                if dist > 1e-9:
                    g_dist = -g_factor / dist * diff
                    
                    # Centers gradients
                    grad[i] += g_dist[0]
                    grad[n + i] += g_dist[1]
                    
                    grad[j] -= g_dist[0]
                    grad[n + j] -= g_dist[1]
                else:
                    # Parallel case, push apart arbitrarily
                    pass

                # Radii gradients
                grad[2*n + i] += g_factor
                grad[2*n + j] += g_factor

    return obj, grad

def run_packing():
    n = 26
    
    # 1. Initialize positions using a hexagonal grid
    centers = generate_hex_grid(n)
    
    # Initialize radii slightly small to be valid
    radii = np.full(n, 0.05)
    
    # Combine into optimization vector
    # [x1, ..., xn, y1, ..., yn, r1, ..., rn]
    # Or interleaved: x1, y1, r1... let's stick to the structure in objective
    # vars = [x1, y1, r1, x2, y2, r2 ...] is easier for loops but let's use blocks
    # vars[0:n] = x, vars[n:2n] = y, vars[2n:] = r
    
    x_vec = centers[:, 0]
    y_vec = centers[:, 1]
    
    x0 = np.concatenate([x_vec, y_vec, radii])
    
    # Bounds: x, y in [0, 1], r in [0, 0.5]
    bounds = []
    for i in range(n):
        bounds.append((0, 1)) # x
    for i in range(n):
        bounds.append((0, 1)) # y
    for i in range(n):
        bounds.append((0, 0.5)) # r
        
    # Optimization
    # Using L-BFGS-B for bound constraints
    res = minimize(
        objective_and_grad, 
        x0, 
        args=(n,), 
        method='L-BFGS-B', 
        bounds=bounds,
        options={'maxiter': 1000, 'ftol': 1e-12, 'gtol': 1e-8}
    )
    
    # Extract results
    final_vars = res.x
    final_x = final_vars[:n]
    final_y = final_vars[n:2*n]
    final_r = final_vars[2*n:]
    
    final_centers = np.column_stack([final_x, final_y])
    
    # Validation check (for safety/clipping if needed)
    # Although optimization should handle it, sometimes numerical noise happens.
    # We ensure strict validity by slightly shrinking if any violation detected in logic above
    # (The penalty method is soft, so we enforce hard constraints post-hoc if necessary)
    
    # Post-processing to ensure strict non-overlap and boundaries
    # This is a simple iterative shrink step
    for _ in range(50):
        valid = True
        for i in range(n):
            # Boundary check
            x, y = final_centers[i]
            r = final_r[i]
            min_dist_boundary = min(x, 1-x, y, 1-y)
            if r > min_dist_boundary:
                final_r[i] = min_dist_boundary
                valid = False
        
        if not valid:
            continue
            
        for i in range(n):
            for j in range(i+1, n):
                dist = np.sqrt(np.sum((final_centers[i] - final_centers[j])**2))
                sum_r = final_r[i] + final_r[j]
                if dist < sum_r:
                    # Reduce radii proportionally or equally to fix overlap
                    excess = sum_r - dist
                    # Reduce both by half the excess
                    reduction = excess / 2.0
                    final_r[i] -= reduction
                    final_r[j] -= reduction
                    valid = False
                    
        if valid:
            break
            
    # Ensure non-negative radii
    final_r = np.maximum(final_r, 0)
    
    sum_radii = np.sum(final_r)
    
    return final_centers, final_r, sum_radii
