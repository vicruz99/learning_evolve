import numpy as np
from scipy.optimize import minimize

def run_packing():
    n = 26
    num_restarts = 20
    
    best_sum = -1.0
    best_centers = None
    best_radii = None

    # Function to compute initial positions on a grid/hex pattern to start reasonably close
    def get_initial_guess(n):
        # Create a grid of points
        # sqrt(26) approx 5.1, so 6x5 grid is good
        # But we can just do random for robustness, or perturbed grid
        # Let's use a perturbed grid
        rows = 6
        cols = 5 # 30 points, we pick 26
        pts = []
        for r in range(rows):
            for c in range(cols):
                pts.append([c * (1.0/cols) + (1.0/(2*cols)), r * (1.0/rows) + (1.0/(2*rows))])
        
        pts = np.array(pts[:n])
        # Add small jitter
        pts += np.random.normal(0, 0.02, pts.shape)
        return pts

    for seed in range(num_restarts):
        np.random.seed(seed)
        
        # Initialize centers
        centers_init = get_initial_guess(n)
        
        # Initialize radii small
        radii_init = np.ones(n) * 0.05
        
        # Variable vector: [x1, y1, r1, x2, y2, r2, ...]
        x0 = np.zeros(3 * n)
        for i in range(n):
            x0[3*i] = centers_init[i, 0]
            x0[3*i+1] = centers_init[i, 1]
            x0[3*i+2] = radii_init[i]

        # Define objective: Maximize sum of radii -> Minimize negative sum
        def objective(x_vars):
            # Sum of radii is at indices 2, 5, 8, ...
            return -np.sum(x_vars[2::3])

        # Define constraints
        constraints = []

        # 1. Boundary Constraints: 
        # r <= x <= 1-r  =>  x - r >= 0, 1 - x - r >= 0
        # r <= y <= 1-y  =>  y - r >= 0, 1 - y - r >= 0
        for i in range(n):
            idx_x = 3*i
            idx_y = 3*i + 1
            idx_r = 3*i + 2
            
            # x - r >= 0
            constraints.append({
                'type': 'ineq',
                'fun': lambda x, i=i: x[idx_x] - x[idx_r]
            })
            # 1 - x - r >= 0
            constraints.append({
                'type': 'ineq',
                'fun': lambda x, i=i: 1.0 - x[idx_x] - x[idx_r]
            })
            # y - r >= 0
            constraints.append({
                'type': 'ineq',
                'fun': lambda x, i=i: x[idx_y] - x[idx_r]
            })
            # 1 - y - r >= 0
            constraints.append({
                'type': 'ineq',
                'fun': lambda x, i=i: 1.0 - x[idx_y] - x[idx_r]
            })

        # 2. Non-overlap Constraints
        # dist_ij >= r_i + r_j
        # sqrt((xi-xj)^2 + (yi-yj)^2) - (ri + rj) >= 0
        # To avoid sqrt derivative issues or complexity, we can use squared form?
        # But (ri+rj)^2 is not linear. 
        # SLSQP handles non-linear constraints.
        # Let's use the distance formulation.
        for i in range(n):
            for j in range(i + 1, n):
                idx_xi, idx_yi, idx_ri = 3*i, 3*i+1, 3*i+2
                idx_xj, idx_yj, idx_rj = 3*j, 3*j+1, 3*j+2
                
                def dist_constraint(x, i=i, j=j):
                    xi, yi, ri = x[idx_xi], x[idx_yi], x[idx_ri]
                    xj, yj, rj = x[idx_xj], x[idx_yj], x[idx_rj]
                    dist = np.sqrt((xi - xj)**2 + (yi - yj)**2)
                    return dist - (ri + rj)
                
                constraints.append({
                    'type': 'ineq',
                    'fun': dist_constraint
                })

        # Bounds for variables
        # x, y in [0, 1], r >= 0. 
        # Upper bound for r is 0.5 (fits in square)
        bounds = []
        for i in range(n):
            bounds.append((0, 1)) # x
            bounds.append((0, 1)) # y
            bounds.append((0, 0.5)) # r

        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints, 
                           options={'maxiter': 1000, 'ftol': 1e-9})
            
            if res.success:
                current_sum = -res.fun # Objective was negative sum
                if current_sum > best_sum:
                    best_sum = current_sum
                    best_centers = np.zeros((n, 2))
                    best_radii = np.zeros(n)
                    for i in range(n):
                        best_centers[i, 0] = res.x[3*i]
                        best_centers[i, 1] = res.x[3*i+1]
                        best_radii[i] = res.x[3*i+2]
        except Exception:
            continue

    return best_centers, best_radii, best_sum
