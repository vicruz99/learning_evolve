import numpy as np

def simulate_packing(centers, radii, steps=4000):
    """Iteratively grow radii and resolve overlaps using repulsive forces."""
    for step in range(steps):
        diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
        dists = np.linalg.norm(diffs, axis=2)
        np.fill_diagonal(dists, np.inf)
        
        r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
        overlaps = np.maximum(r_sum - dists, 0)
        
        # Direction vectors (safe division)
        safe_dists = np.where(dists < 1e-9, 1.0, dists)
        dirs = diffs / safe_dists[:, :, np.newaxis]
        
        # Compute repulsive forces
        forces = np.zeros_like(centers)
        # F_i = sum_j (overlap_ij * dir_ij)
        forces += np.sum(overlaps[:, :, np.newaxis] * dirs, axis=1)
        
        # Boundary repulsion (strong push back into square)
        wall_force = 10.0
        for axis in range(2):
            # Left / Bottom
            mask_low = centers[:, axis] < radii
            forces[mask_low, axis] += wall_force * (radii[mask_low] - centers[mask_low, axis])
            # Right / Top
            mask_high = centers[:, axis] > 1 - radii
            forces[mask_high, axis] -= wall_force * (centers[mask_high, axis] - (1 - radii[mask_high]))
            
        # Update positions
        lr = 0.06 * (1 - step / steps)
        centers += lr * forces
        centers = np.clip(centers, radii[:, np.newaxis], 1 - radii[:, np.newaxis])
        
        # Grow radii if system is stable (no significant overlaps)
        if np.max(overlaps) < 1e-7:
            radii += 2e-6
        else:
            # Cool down growth if stuck
            if step % 200 == 0:
                radii *= 0.995
                
    return centers, radii

def compute_max_radii(centers):

    radii = np.full(26, 0.08)
    for _ in range(2000):

            # Max radius constrained by boundaries
            max_r = min(centers[i, 0], 1 - centers[i, 0],

            for j in range(26):

                    max_r = min(max_r, (d - radii[j]) / 2.0)
            radii[i] = max(1e-8, max_r)
    return radii
