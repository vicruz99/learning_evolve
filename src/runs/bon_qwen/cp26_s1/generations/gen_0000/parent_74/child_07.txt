import numpy as np
from scipy.optimize import minimize

# Number of circles to pack
N_CIRCLES = 26

def compute_penalty_and_grad(params):
    """
    Computes the penalty term and its gradient for the objective function.
    Penalty includes boundary violations and overlap violations.
    
    Args:
        params: np.array of shape (3*N_CIRCLES,) containing [x1, y1, r1, x2, y2, r2, ...]
        
    Returns:
        penalty: float, sum of squared violations
        grad: np.array, gradient of the penalty with respect to params
    """
    n = N_CIRCLES
    centers = params[:2*n].reshape(n, 2)
    radii = params[2*n:]
    
    penalty = 0.0
    grad = np.zeros_like(params)
    
    # 1. Boundary Violations
    # Constraints: r <= x, r <= 1-x, r <= y, r <= 1-y
    # Violations: v = max(0, r - x), etc.
    
    # X-axis boundaries
    # v = max(0, r - x) -> deriv wrt r: sign(v), wrt x: -sign(v)
    v_x1 = radii - centers[:, 0] # r - x
    idx_x1 = v_x1 > 0
    penalty += np.sum(v_x1[idx_x1]**2)
    # Gradient
    grad[2*n + np.where(idx_x1)[0]] += 2 * v_x1[idx_x1] # dr
    grad[np.where(idx_x1)[0]] -= 2 * v_x1[idx_x1]       # dx
    
    v_x2 = radii - (1 - centers[:, 0]) # r - (1-x) = r + x - 1
    idx_x2 = v_x2 > 0
    penalty += np.sum(v_x2[idx_x2]**2)
    grad[2*n + np.where(idx_x2)[0]] += 2 * v_x2[idx_x2] # dr
    grad[np.where(idx_x2)[0]] += 2 * v_x2[idx_x2]        # dx
    
    # Y-axis boundaries
    v_y1 = radii - centers[:, 1] # r - y
    idx_y1 = v_y1 > 0
    penalty += np.sum(v_y1[idx_y1]**2)
    grad[2*n + np.where(idx_y1)[0]] += 2 * v_y1[idx_y1] # dr
    grad[n + np.where(idx_y1)[0]] -= 2 * v_y1[idx_y1]   # dy
    
    v_y2 = radii - (1 - centers[:, 1]) # r - (1-y) = r + y - 1
    idx_y2 = v_y2 > 0
    penalty += np.sum(v_y2[idx_y2]**2)
    grad[2*n + np.where(idx_y2)[0]] += 2 * v_y2[idx_y2] # dr
    grad[n + np.where(idx_y2)[0]] += 2 * v_y2[idx_y2]    # dy
    
    # 2. Overlap Violations
    # Constraint: dist(i, j) >= r_i + r_j
    # Violation: v_ij = max(0, r_i + r_j - dist(i, j))
    
    # Compute pairwise distances
    # diff shape (n, n, 2)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    # Avoid division by zero on diagonal, though diagonal violation is ignored
    dists[np.eye(n, dtype=bool)] = 1.0 
    
    r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    
    # Violations matrix (symmetric)
    violations = r_sum - dists
    
    # We only consider upper triangle to avoid double counting
    # Mask for upper triangle
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    
    # Filter positive violations
    pos_violations = np.maximum(0, violations)
    active_violations = pos_violations * mask
    
    penalty += np.sum(active_violations**2)
    
    # Gradients for active violations
    # v_ij = r_i + r_j - d_ij
    # d/d(r_i) v_ij^2 = 2 * v_ij * 1
    # d/d(x_i) v_ij^2 = 2 * v_ij * (- (x_i - x_j) / d_ij)
    
    # Identify active pairs
    i_indices, j_indices = np.where(mask)
    
    # Filter for positive violations
    active_vals = active_violations[i_indices, j_indices]
    active_mask = active_vals > 0
    
    active_i = i_indices[active_mask]
    active_j = j_indices[active_mask]
    active_v = active_vals[active_mask]
    
    # Distances for active pairs
    d_active = dists[active_i, active_j]
    
    # Gradient components
    # For r_i and r_j
    grad[2*n + active_i] += 2 * active_v
    grad[2*n + active_j] += 2 * active_v
    
    # For x and y
    # Deriv of dist wrt x_i is (x_i - x_j)/d
    # Deriv of v wrt x_i is - (x_i - x_j)/d
    # Deriv of v^2 wrt x_i is 2*v * (- (x_i - x_j)/d)
    
    dx = centers[active_i, 0] - centers[active_j, 0]
    dy = centers[active_i, 1] - centers[active_j, 1]
    
    factor = -2 * active_v / d_active
    
    grad[active_i] += factor * dx # dx_i
    grad[n + active_i] += factor * dy # dy_i
    
    grad[active_j] -= factor * dx # dx_j (since diff is i-j, deriv wrt x_j is -dx/d)
    grad[n + active_j] -= factor * dy # dy_j
    
    return penalty, grad

def objective(params):
    """
    Objective function: Minimize -sum(radii) + lambda * penalty
    """
    n = N_CIRCLES
    radii = params[2*n:]
    
    # Penalty weight
    # High weight to enforce constraints
    penalty_weight = 5000.0
    
    penalty, grad_pen = compute_penalty_and_grad(params)
    
    obj_val = -np.sum(radii) + penalty_weight * penalty
    
    # Gradient of objective
    # d/dr_i (-r_i) = -1
    # d/dx_i, d/dy_i = 0
    grad = np.zeros_like(params)
    grad[2*n:] -= 1.0
    grad += penalty_weight * grad_pen
    
    return obj_val, grad

def get_bounds():
    """Returns bounds for L-BFGS-B"""
    bounds = []
    for _ in range(N_CIRCLES):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 0.5)) # r
    return bounds

def generate_initial_params():
    """
    Generates initial parameters with a hexagonal-like grid distribution.
    """
    n = N_CIRCLES
    centers = np.zeros((n, 2))
    
    # Create a grid to distribute points
    # We want roughly sqrt(n) points per side
    # For n=26, maybe 6x5 grid (30 spots) and pick 26
    rows = 6
    cols = 5
    
    pts = []
    for r in range(rows):
        for c in range(cols):
            if len(pts) >= n:
                break
            # Hexagonal shift for odd rows
            shift = 0.0
            if r % 2 == 1:
                shift = 1.0 / (2 * cols)
            
            # Uniform distribution in [0, 1]
            # Using slight margin to avoid boundary issues initially
            x = (c + 0.5 + shift) / cols
            y = (r + 0.5) / rows
            
            # Clamp to [0, 1]
            x = np.clip(x, 0, 1)
            y = np.clip(y, 0, 1)
            
            pts.append([x, y])
        if len(pts) >= n:
            break
            
    centers = np.array(pts[:n])
    
    # Add small random noise
    np.random.seed(42) # Fixed seed for reproducibility in one run, but we will vary in loop
    centers += np.random.uniform(-0.01, 0.01, centers.shape)
    centers = np.clip(centers, 0.01, 0.99)
    
    # Initialize radii small
    radii = np.full(n, 0.05)
    
    params = np.zeros(3 * n)
    params[:2*n] = centers.flatten()
    params[2*n:] = radii
    
    return params

def check_validity(centers, radii):
    """
    Checks if the packing is valid according to problem constraints.
    """
    n = centers.shape[0]
    # Check boundaries
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x < r - 1e-9 or x > 1 - r + 1e-9 or y < r - 1e-9 or y > 1 - r + 1e-9:
            return False
    
    # Check overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-9:
                return False
    return True

def run_packing():
    """
    Main function to run the packing optimization.
    Returns centers, radii, sum_radii.
    """
    best_sum_radii = 0.0
    best_centers = None
    best_radii = None
    
    bounds = get_bounds()
    
    # Number of restarts
    num_restarts = 10
    
    for k in range(num_restarts):
        # Generate initial params with different noise seed if possible
        # Since generate_initial_params uses fixed seed 42, let's modify it slightly
        # Or just call it and rely on the fact that we want to explore.
        # Actually, let's create a new seed inside or pass it.
        # But helper functions must be top level.
        # I'll just use the function and maybe vary the noise manually here?
        # No, I can't modify generate_initial_params easily without args.
        # Let's just call it. The noise is small.
        # To get diversity, I can add different offsets.
        
        base_params = generate_initial_params()
        
        # Add larger random perturbation for diversity
        np.random.seed(k + 100)
        noise = np.random.uniform(-0.05, 0.05, base_params.shape)
        # Keep radii positive and smallish
        noise[2*N_CIRCLES:] *= 0.01 
        params = base_params + noise
        params = np.clip(params, 0.001, 0.999) # Ensure within bounds roughly
        
        # Run optimization
        res = minimize(objective, params, method='L-BFGS-B', bounds=bounds, 
                       options={'maxiter': 1000, 'ftol': 1e-9})
        
        # Extract results
        opt_params = res.x
        opt_centers = opt_params[:2*N_CIRCLES].reshape(N_CIRCLES, 2)
        opt_radii = opt_params[2*N_CIRCLES:]
        
        # Check validity
        if check_validity(opt_centers, opt_radii):
            current_sum = np.sum(opt_radii)
            if current_sum > best_sum_radii:
                best_sum_radii = current_sum
                best_centers = opt_centers.copy()
                best_radii = opt_radii.copy()
    
    # If no valid packing found (unlikely with penalty), fallback to a simple grid
    if best_centers is None:
        # Fallback: 5x5 grid + 1
        centers = np.zeros((N_CIRCLES, 2))
        r = 0.05 # Safe radius
        idx = 0
        for r_idx in range(6):
            for c_idx in range(5):
                if idx < N_CIRCLES:
                    centers[idx] = [ (c_idx + 0.5)/5, (r_idx + 0.5)/6 ]
                    idx += 1
        best_centers = centers
        best_radii = np.full(N_CIRCLES, r)
        best_sum_radii = np.sum(best_radii)

    return best_centers, best_radii, best_sum_radii
