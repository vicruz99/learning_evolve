def init_hex_grid(n, seed):
    centers = np.zeros((n, 2))
    idx = 0
    row = 0
    y_start = 0.08 + seed * 0.01
    y = y_start
    while idx < n and y < 0.93:
        x_off = 0.08 if row % 2 == 0 else 0.17
        x = x_off
        while idx < n and x < 0.93:
            centers[idx] = [x, y]
            idx += 1
            x += 0.20
        y += 0.173
        row += 1
    return centers


def growth_method(n):
    centers = init_hex_grid(n, 0)
    radii = np.full(n, 0.04)

        for i in range(n):
            for j in range(i + 1, n):
                diff = centers[i] - centers[j]
                d = np.sqrt(np.sum(diff ** 2))
                if d < 1e-10:
                    d = 1e-10
                want = radii[i] + radii[j] + 0.005
                if d < want:
                    f = (want - d) / d * diff
                    forces[i] += f
                    forces[j] -= f
            # Boundary repulsion

                    forces[i][d] += 0.5
                if centers[i][d] > 1 - radii[i] - 0.002:
                    forces[i][d] -= 0.5

        centers += forces * 0.02
        centers = np.clip(centers, 0, 1)

        # Grow radii slowly

            radii = np.minimum(radii * 1.003, 0.25)

    return centers, radii
