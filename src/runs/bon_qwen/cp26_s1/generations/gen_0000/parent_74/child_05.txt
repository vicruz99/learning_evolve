import numpy as np
from scipy.optimize import linprog, minimize

def run_packing():
    # Constants
    N = 26
    
    # Objective: Minimize negative sum of radii (Maximize sum)
    c_obj = -np.ones(N)

        # centers shape: (26, 2)
        # Construct constraints for LP: A_ub @ r <= b_ub
        
        m = 0
        # Pre-allocate arrays for performance
        n_constraints = 4 * N + (N * (N - 1)) // 2
        A_ub = np.zeros((n_constraints, N))
        b_ub = np.zeros(n_constraints)
        
        # 1. Boundary constraints (r_i <= min(x, 1-x, y, 1-y))
        for i in range(N):
            x, y = centers[i]
            min_dist = min(x, 1.0 - x, y, 1.0 - y)
            # r_i <= min_dist
            A_ub[m, i] = 1.0
            b_ub[m] = min_dist
            m += 1
            
        # 2. Pairwise non-overlap constraints (r_i + r_j <= dist_ij)
        dists = np.zeros((N, N))
        for i in range(N):
            for j in range(i + 1, N):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                d = np.sqrt(dx*dx + dy*dy)
                
                # r_i + r_j <= d
                A_ub[m, i] = 1.0
                A_ub[m, j] = 1.0
                b_ub[m] = d
                m += 1
                
        # Bounds for r_i (r_i >= 0)
        bounds = [(0, None) for _ in range(N)]
        
        # Solve LP
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        
        if res.success:
            return res.fun, -res.x # Return negative sum and radii
        else:
            # Fallback to 0 if LP fails
            return 0.0, np.zeros(N)

        centers = centers_flat.reshape((N, 2))

        return neg_sum

            # Approximate diameter 0.2
