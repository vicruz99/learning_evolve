from scipy.spatial.distance import cdist
import random

def compute_loss(params, lam):

    Computes the objective function: maximize radius r while penalizing overlaps and boundary violations.
    params: array of shape (53,) -> [x1, y1, ..., x26, y26, r]
    lam: penalty weight

    r = params[52]

    dists = cdist(centers, centers)

    loss += np.sum(np.maximum(0.0, r - centers[:, 1])**2)
    loss += np.sum(np.maximum(0.0, centers[:, 1] - (1.0 - r))**2)
    
    # Objective: maximize r (minimize -r) + penalty
    return -r + lam * loss

def optimize_packing(init_centers, r_init, lam_start):

    Optimizes centers and radius using homotopy continuation.
    """
