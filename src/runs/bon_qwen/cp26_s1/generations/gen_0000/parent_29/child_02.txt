import numpy as np
from scipy.optimize import differential_evolution, minimize

def compute_negative_sum_radii(params):
    """
    Computes the negative sum of radii for a given configuration of circle centers.
    Radii are determined by the minimum distance to the boundary and half the minimum 
    distance to any other circle center.
    """
    centers = params.reshape(26, 2)
    
    # Distance to the four boundaries of the unit square
    d_bound = np.minimum(
        np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
        np.minimum(centers[:, 1], 1.0 - centers[:, 1])
    )
    
    # Pairwise distances between all circles
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    min_dists = np.min(dists, axis=1)
    
    # Radius of each circle is constrained by boundaries and neighbor distances
    r = np.minimum(d_bound, min_dists / 2.0)
    
    # Return negative sum because optimizers minimize by default
    return -np.sum(r)

    Optimizes the placement of 26 circles in a unit square to maximize the sum of radii.

        compute_negative_sum_radii,

        maxiter=1500,
