import numpy as np
import scipy.optimize

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square

    Args:
        centers: np.array of shape (n, 2) with (x, y) coordinates
        radii: np.array of shape (n) with radius of each circle

    Returns:
        True if valid, False otherwise
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True

def generate_initial_configs(n_circles=26):
    configs = []
    
    # Config 1: Adaptive Hexagonal / Brick-wall Pattern
    # 5 rows of 5, 1 row of 1 (or distributed)
    # Let's do 5, 5, 5, 5, 4, 2 = 26
    rows_counts = [5, 5, 5, 5, 4, 2]
    centers = []
    radii = []
    
    # Estimated radius for initialization
    r_init = 0.095
    
    y_step = 1.0 / (len(rows_counts) + 1)
    
    for r_idx, count in enumerate(rows_counts):
        y = (r_idx + 1) * y_step
        # Stagger x positions for odd/even rows
        offset = 0.0
        if r_idx % 2 == 1:
            offset = 0.05  # Small shift for brick wall effect
        
        x_start = r_init + offset
        spacing = (1.0 - 2 * r_init - offset) / (count - 1) if count > 1 else 0
        
        for c_idx in range(count):
            if count == 1:
                x = 0.5
            else:
                x = x_start + c_idx * spacing
            centers.append([x, y])
            radii.append(r_init)
            
    configs.append((np.array(centers), np.array(radii)))

    # Config 2: Randomized perturbation of Config 1
    c1_centers, c1_radii = configs[0]
    random_centers = c1_centers + np.random.normal(0, 0.01, size=c1_centers.shape)
    random_radii = c1_radii + np.random.normal(0, 0.005, size=c1_radii.shape)
    # Clip to ensure validity roughly
    random_centers = np.clip(random_centers, 0.01, 0.99)
    random_radii = np.maximum(random_radii, 0.01)
    configs.append((random_centers, random_radii))

    # Config 3: Grid-like but optimized
    # 5x5 grid + 1 in center
    grid_centers = []
    grid_radii = []
    step = 0.2
    for i in range(5):
        for j in range(5):
            grid_centers.append([step * (i + 0.5), step * (j + 0.5)])
            grid_radii.append(0.1)
    # Add 26th circle? 5x5 is 25.
    # Let's just add a small one or perturb. 
    # Actually 5x5 grid with r=0.1 is tight. 
    # Let's use a slightly smaller r and add one.
    r_grid = 0.09
    gc = []
    gr = []
    for i in range(5):
        for j in range(5):
            gc.append([r_grid + i * (2*r_grid), r_grid + j * (2*r_grid)])
            gr.append(r_grid)
    gc.append([0.5, 0.5])
    gr.append(0.05)
    configs.append((np.array(gc), np.array(gr)))

    return configs

def objective(x_vec, n):
    # x_vec contains [x0, y0, x1, y1, ..., rn-1]
    # Structure: x[0], x[1], ..., x[n-1], y[0], ..., y[n-1], r[0], ..., r[n-1]
    # Actually easier: flat list of x_i, y_i, r_i
    
    # Reconstruct
    # vars: x0, x1... xn-1, y0...yn-1, r0...rn-1
    # Length 3n
    xs = x_vec[:n]
    ys = x_vec[n:2*n]
    rs = x_vec[2*n:]
    
    return -np.sum(rs)

def constraint_boundary(x_vec, n):
    xs = x_vec[:n]
    ys = x_vec[n:2*n]
    rs = x_vec[2*n:]
    
    constraints = []
    for i in range(n):
        # x - r >= 0
        constraints.append(xs[i] - rs[i])
        # x + r <= 1 => r - (1-x) <= 0 => 1 - x - r >= 0
        constraints.append(1 - xs[i] - rs[i])
        # y - r >= 0
        constraints.append(ys[i] - rs[i])
        # y + r <= 1
        constraints.append(1 - ys[i] - rs[i])
    return np.array(constraints)

def constraint_overlap(x_vec, n):
    xs = x_vec[:n]
    ys = x_vec[n:2*n]
    rs = x_vec[2*n:]
    
    constraints = []
    for i in range(n):
        for j in range(i + 1, n):
            dist_sq = (xs[i] - xs[j])**2 + (ys[i] - ys[j])**2
            sum_r = rs[i] + rs[j]
            # dist >= sum_r => dist^2 >= sum_r^2
            # We want dist^2 - sum_r^2 >= 0
            # However, dist^2 can be large, sum_r^2 small.
            # Using linearized or direct? Direct is better for scipy if smooth.
            # dist - sum_r >= 0 is non-smooth at 0.
            # Let's use dist^2 - sum_r^2 >= 0.
            constraints.append(dist_sq - sum_r**2)
    return np.array(constraints)

def run_packing():
    n = 26
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    configs = generate_initial_configs(n)
    
    # Bounds for variables
    # x, y in [0, 1]
    # r >= 0 (effectively bounded by 0.5)
    bounds = []
    for i in range(n):
        bounds.extend([(0, 1), (0, 1)]) # x, y
        bounds.append((0, 0.5))          # r

    for idx, (init_centers, init_radii) in enumerate(configs):
        print(f"Running optimization from config {idx + 1}...")
        
        # Flatten initial vector
        x0 = np.concatenate([init_centers[:, 0], init_centers[:, 1], init_radii])
        
        # Constraints
        cons = []
        cons.append({'type': 'ineq', 'fun': constraint_boundary, 'args': (n,)})
        cons.append({'type': 'ineq', 'fun': constraint_overlap, 'args': (n,)})
        
        # Optimization
        try:
            res = scipy.optimize.minimize(
                objective,
                x0,
                method='SLSQP',
                args=(n,),
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 1000, 'ftol': 1e-9}
            )
            
            if res.success:
                curr_sum = -res.fun

                    # Extract results
                    best_xs = res.x[:n]
                    best_ys = res.x[n:2*n]
                    best_rs = res.x[2*n:]
                    
                    best_sum = curr_sum
                    best_centers = np.column_stack([best_xs, best_ys])
                    best_radii = best_rs
                    print(f"  New best sum: {best_sum:.4f}")
            else:
                print(f"  Optimization failed: {res.message}")
                
        except Exception as e:
            print(f"  Exception in optimization: {e}")

    if best_sum < 2.636:
