import numpy as np
from scipy.optimize import minimize

def compute_loss(params, n, lam):
    """
    Computes the objective function: -sum(radii) + lambda * penalty.
    Penalty enforces boundary and non-overlap constraints.
    """
    centers = params[:2*n].reshape(n, 2)
    radii = params[2*n:]
    
    # Primary objective: maximize sum of radii
    loss = -np.sum(radii)
    
    # Boundary constraint penalties:
    # Circle must be inside [0,1]x[0,1]
    # r <= x <= 1-r  =>  r - x <= 0  and  x + r - 1 <= 0
    penalty = 0.0
    penalty += np.sum(np.maximum(0, radii - centers[:, 0])**2)
    penalty += np.sum(np.maximum(0, centers[:, 0] + radii - 1)**2)
    penalty += np.sum(np.maximum(0, radii - centers[:, 1])**2)
    penalty += np.sum(np.maximum(0, centers[:, 1] + radii - 1)**2)
    
    # Overlap constraint penalties:
    # dist^2 >= (r_i + r_j)^2  =>  (r_i + r_j)^2 - dist^2 <= 0
    c_diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist_sq = np.sum(c_diff**2, axis=2)
    r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    violations = (r_sum**2 - dist_sq)
    
    # Only consider pairs i < j to avoid double counting
    penalty += np.sum(np.maximum(0, violations[np.triu_indices(n, k=1)])**2)
    
    return loss + lam * penalty

def run_packing():
    n = 26
    best_params = None
    best_sum = -np.inf
    
    # Try multiple random restarts to escape local minima
    seeds = [42, 137, 256, 399, 512]
    
    for seed in seeds:
        np.random.seed(seed)
        
        # 1. Generate initial hexagonal layout
        pts = []
        r0 = 0.05
        k = 0
        while len(pts) < n:
            l = 0
            y = r0 * (1 + k * np.sqrt(3))
            if y + r0 > 1.0:
                k += 1
                continue
            while True:
                x = r0 * (1 + 2*l + (k % 2))
                if x + r0 > 1.0:
                    break
                pts.append([x, y])
                l += 1
                if len(pts) == n:
                    break
            k += 1
            
        pts = np.array(pts[:n])
        # Add small random perturbation to break grid symmetry
        pts += np.random.randn(*pts.shape) * 0.015
        pts = np.clip(pts, 0.05, 0.95)
        
        init_radii = np.ones(n) * 0.06
        params = np.concatenate([pts.flatten(), init_radii])
        
        # Variable bounds
        bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n
        
        # 2. Iterative penalty method optimization
        lam = 100.0
        for _ in range(18):
            res = minimize(compute_loss, params, args=(n, lam), 
                           method='L-BFGS-B', bounds=bounds,
                           options={'ftol': 1e-12, 'gtol': 1e-8, 'maxiter': 500})
            params = res.x
            
            # Check constraint violation to decide if lambda should increase
            c = params[:2*n].reshape(n, 2)
            r = params[2*n:]
            viol = np.sum(np.maximum(0, r - c[:, 0])**2)
            viol += np.sum(np.maximum(0, c[:, 0] + r - 1)**2)
            viol += np.sum(np.maximum(0, r - c[:, 1])**2)
            viol += np.sum(np.maximum(0, c[:, 1] + r - 1)**2)
            
            c_diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
            dist_sq = np.sum(c_diff**2, axis=2)
            r_sum = r[:, np.newaxis] + r[np.newaxis, :]
            viol += np.sum(np.maximum(0, (r_sum**2 - dist_sq)[np.triu_indices(n, k=1)])**2)
            
            if viol < 1e-6:
                break
            lam *= 4.0
            
        current_sum = np.sum(params[2*n:])
        if current_sum > best_sum:
            best_sum = current_sum
            best_params = params.copy()
            
    centers = best_params[:2*n].reshape(n, 2)
    radii = best_params[2*n:]
    
    return centers, radii, best_sum
