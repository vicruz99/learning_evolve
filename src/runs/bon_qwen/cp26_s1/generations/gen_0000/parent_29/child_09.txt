import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Solves the circle packing problem for 26 circles in a unit square.
    Returns centers, radii, and the sum of radii.
    """
    n = 26
    # Total number of variables: 3 per circle (x, y, r)
    dim = 3 * n
    
    # Initial guess: Hexagonal lattice arrangement
    # Estimated radius for 26 circles is around 0.1. 
    # We start with a slightly smaller radius to ensure validity.
    r_init = 0.09
    points = []
    
    # Generate hexagonal lattice points
    # Row spacing: r * sqrt(3), Col spacing: 2 * r
    # We generate a grid and filter points that fit well
    y_val = r_init
    while y_val + r_init <= 1.0:
        x_val = r_init
        # Offset every other row
        offset = r_init if len(points) % 2 == 1 else 0
        if offset > 0: x_val += offset
        
        while x_val + r_init <= 1.0:
            points.append([x_val, y_val])
            x_val += 2 * r_init
        y_val += r_init * np.sqrt(3)
    
    # If we have more points than needed, take the first n
    # If fewer, we might need to adjust, but for n=26 this should be sufficient
    # with r_init=0.09. Let's verify count.
    # With r=0.09, width=1, approx 5-6 points per row. Height=1, approx 5-6 rows.
    # Should be plenty.
    
    if len(points) < n:
        # Fallback to random if lattice is sparse (unlikely)
        np.random.seed(42)
        centers_guess = np.random.rand(n, 2) * 0.8 + 0.1
    else:
        centers_guess = np.array(points[:n])
        
    # Initialize radii
    radii_guess = np.full(n, r_init)
    
    # Flatten for optimizer
    x0 = np.hstack([centers_guess.flatten(), radii_guess])
    
    # Define bounds: x, y in [0, 1], r in [0, 0.5]
    bounds = []
    for _ in range(n):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 0.5)) # r

    # Penalty coefficient
    penalty_k = 10000.0

    def objective(x_vars):
        # Reshape
        centers = x_vars[:2*n].reshape((n, 2))
        radii = x_vars[2*n:]
        
        # Objective: Maximize sum of radii => Minimize negative sum
        obj = -np.sum(radii)
        
        # Penalty for overlaps
        # dist >= r_i + r_j  =>  dist^2 >= (r_i + r_j)^2
        # We penalize if dist^2 < (r_i + r_j)^2
        # To avoid sqrt, we check squared distance.
        # But penalty on squared difference is smoother?
        # Let's use linear penalty on violation of dist - (ri+rj)
        
        penalty = 0.0
        
        # Boundary penalties
        # x - r >= 0, 1 - x - r >= 0, etc.
        # Violation = max(0, r - x), max(0, r - (1-x))
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            
            # Left wall
            viol = max(0.0, r - x)
            penalty += viol**2
            # Right wall
            viol = max(0.0, r - (1.0 - x))
            penalty += viol**2
            # Bottom wall
            viol = max(0.0, r - y)
            penalty += viol**2
            # Top wall
            viol = max(0.0, r - (1.0 - y))
            penalty += viol**2

        # Overlap penalties
        for i in range(n):
            for j in range(i + 1, n):
                dist_sq = np.sum((centers[i] - centers[j])**2)
                dist = np.sqrt(dist_sq)
                sum_r = radii[i] + radii[j]
                
                if dist < sum_r:
                    # Violation amount
                    viol = sum_r - dist
                    penalty += viol**2
        
        return obj + penalty_k * penalty

    # Run optimization
    # L-BFGS-B is good for bound-constrained problems, but we moved constraints to penalty.
    # However, bounds are still useful for box constraints.
    res = minimize(objective, x0, method='L-BFGS-B', bounds=bounds, options={'maxiter': 10000, 'ftol': 1e-12})
    
    # Extract results
    centers_opt = res.x[:2*n].reshape((n, 2))
    radii_opt = res.x[2*n:]
    
    # Post-processing: 
    # The optimizer might leave tiny violations. 
    # We can clip radii to satisfy constraints strictly if needed, 
    # but the penalty method usually handles it well.
    # Let's ensure non-negative radii.
    radii_opt = np.maximum(radii_opt, 0.0)
    
    # Re-validate and fix tiny boundary issues if any
    # Although the penalty should keep them valid, numerical noise might occur.
    # We can do a simple projection step.
    for i in range(n):
        x, y = centers_opt[i]
        r = radii_opt[i]
        
        # Project center to be at least r away from walls
        x = np.clip(x, r, 1.0 - r)
        y = np.clip(y, r, 1.0 - r)
        centers_opt[i] = [x, y]

    sum_radii = np.sum(radii_opt)
    
    return centers_opt, radii_opt, sum_radii
