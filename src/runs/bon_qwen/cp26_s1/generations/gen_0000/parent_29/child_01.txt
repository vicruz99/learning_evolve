import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    # Initial guess radius (approximate upper bound for 26 circles)
    r_low = 0.0
    r_high = 0.15 
    tol = 1e-6
    max_iters = 40
    
    # Grid initialization for centers
    def get_init_centers(r):
        centers = np.zeros((n, 2))
        # 6x5 grid layout logic adapted for 26 circles
        # 5 rows of 5 = 25, add 1 more
        idx = 0
        # Place 5 rows
        for row in range(5):
            for col in range(5):
                if idx < n:
                    x = r + (col * (1 - 2*r)) / 4.0
                    y = r + (row * (1 - 2*r)) / 4.0
                    centers[idx] = [x, y]
                    idx += 1
        # Place the 26th circle
        if idx < n:
            x = r + (2.5 * (1 - 2*r)) / 4.0
            y = r + (4.5 * (1 - 2*r)) / 4.0
            centers[idx] = [x, y]
            idx += 1
        return centers

    # Penalty function for overlaps
    def overlap_penalty(centers_flat, r):
        centers = centers_flat.reshape(n, 2)
        penalty = 0.0
        # Pairwise overlap
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                if dist < 2 * r:
                    overlap = 2 * r - dist
                    penalty += overlap**2
        return penalty

    # Objective function for scipy (minimizing penalty)
    def objective(centers_flat, r):
        return overlap_penalty(centers_flat, r)

    best_centers = None
    best_r = 0.0

    for _ in range(max_iters):
        if r_high - r_low < tol:
            break
            
        r_mid = (r_low + r_high) / 2
        
        # Initialization for optimization
        # Use best known centers if available, scaled/adjusted if necessary, 
        # or regenerate if r changed significantly. 
        # For simplicity, we use a grid init for the first step and 
        # carry over the best valid centers for subsequent steps.
        if best_centers is not None:
            x0 = best_centers.flatten()
        else:
            x0 = get_init_centers(r_mid).flatten()

        # Bounds: each coordinate must be in [r_mid, 1-r_mid]
        bounds = [(r_mid, 1 - r_mid)] * (2 * n)

        # Optimize centers
        # L-BFGS-B handles box constraints well
        res = minimize(objective, x0, args=(r_mid,), method='L-BFGS-B', bounds=bounds, 
                       options={'ftol': 1e-10, 'gtol': 1e-10, 'maxiter': 1000})

        if res.fun < 1e-9:
            # Feasible radius found
            best_r = r_mid
            best_centers = res.x.reshape(n, 2)
            r_low = r_mid
        else:
            # Not feasible
            r_high = r_mid

    # If we found a valid packing, output it
    if best_centers is not None:
        centers = best_centers
        radii = np.full(n, best_r)
        sum_radii = np.sum(radii)
    else:
        # Fallback to a valid small packing
        centers = get_init_centers(0.05)
        radii = np.full(n, 0.05)
        sum_radii = np.sum(radii)

    return centers, radii, sum_radii
