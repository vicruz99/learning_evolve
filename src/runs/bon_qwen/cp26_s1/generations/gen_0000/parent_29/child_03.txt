import numpy as np
from scipy.optimize import minimize


def objective(params, n):
    radii = params[2::3]
    return -np.sum(radii)


def constraint_x_ge_r(params, i):
    return params[3 * i] - params[3 * i + 2]


def constraint_x_le_1mr(params, i):
    return 1.0 - params[3 * i] - params[3 * i + 2]


def constraint_y_ge_r(params, i):
    return params[3 * i + 1] - params[3 * i + 2]


def constraint_y_le_1mr(params, i):
    return 1.0 - params[3 * i + 1] - params[3 * i + 2]


def constraint_overlap(params, i, j):
    xi, yi, ri = params[3 * i], params[3 * i + 1], params[3 * i + 2]
    xj, yj, rj = params[3 * j], params[3 * j + 1], params[3 * j + 2]
    dist_sq = (xi - xj) ** 2 + (yi - yj) ** 2
    min_dist_sq = (ri + rj) ** 2
    return dist_sq - min_dist_sq


def initialize_hexagonal(n):
    r = 0.07
    params = np.zeros(3 * n)
    idx = 0

    row_y = r
    row_num = 0

    while idx < n:
        if row_num % 2 == 0:
            x_start = r
            spacing = 2 * r
            max_cols = int((1.0 - 2 * r) / spacing) + 1
            cols = min(max_cols, n - idx)
            for c in range(cols):
                x = x_start + c * spacing
                params[3 * idx] = x
                params[3 * idx + 1] = row_y
                params[3 * idx + 2] = r
                idx += 1
        else:
            x_start = 2 * r
            spacing = 2 * r
            max_cols = int((1.0 - 2 * r) / spacing) + 1
            cols = min(max_cols, n - idx)
            for c in range(cols):
                x = x_start + c * spacing
                params[3 * idx] = x
                params[3 * idx + 1] = row_y
                params[3 * idx + 2] = r
                idx += 1
        row_y += np.sqrt(3) * r
        row_num += 1

    return params


def force_directed_refinement(n, centers, radii, iterations=5000):
    lr = 0.005
    repulsion_strength = 100.0
    expansion_force = 0.5

    for it in range(iterations):
        decay = max(0.1, 1.0 - it / iterations)
        current_lr = lr * decay

        fx = np.zeros(n)
        fy = np.zeros(n)
        fr = np.zeros(n)

        for i in range(n):
            fr[i] = expansion_force
            x, y, r = centers[i, 0], centers[i, 1], radii[i]

            if x - r < 0.001:
                fx[i] -= repulsion_strength * (0.001 - (x - r))
            if x + r > 0.999:
                fx[i] += repulsion_strength * ((x + r) - 0.999)
            if y - r < 0.001:
                fy[i] -= repulsion_strength * (0.001 - (y - r))
            if y + r > 0.999:
                fy[i] += repulsion_strength * ((y + r) - 0.999)

            for j in range(i + 1, n):
                dx = centers[j, 0] - centers[i, 0]
                dy = centers[j, 1] - centers[i, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                min_dist = radii[i] + radii[j]

                if dist < min_dist and dist > 1e-10:
                    overlap = min_dist - dist
                    force_magnitude = repulsion_strength * overlap
                    fx[i] -= force_magnitude * dx / dist
                    fy[i] -= force_magnitude * dy / dist
                    fx[j] += force_magnitude * dx / dist
                    fy[j] += force_magnitude * dy / dist
                    fr[i] -= overlap * 2.0
                    fr[j] -= overlap * 2.0

        for i in range(n):
            centers[i, 0] += current_lr * fx[i]
            centers[i, 1] += current_lr * fy[i]
            radii[i] += current_lr * fr[i]
            radii[i] = max(1e-8, min(0.49, radii[i]))
            centers[i, 0] = max(1e-6, min(0.999, centers[i, 0]))
            centers[i, 1] = max(1e-6, min(0.999, centers[i, 1]))

    # Clamp to ensure feasibility
    for i in range(n):
        radii[i] = min(radii[i], centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        radii[i] = max(1e-8, radii[i])

    return centers, radii


def optimize_with_slsqp(params0, n):
    cons = []
    for i in range(n):
        cons.append({'type': 'ineq', 'fun': constraint_x_ge_r, 'args': (i,)})
        cons.append({'type': 'ineq', 'fun': constraint_x_le_1mr, 'args': (i,)})
        cons.append({'type': 'ineq', 'fun': constraint_y_ge_r, 'args': (i,)})
        cons.append({'type': 'ineq', 'fun': constraint_y_le_1mr, 'args': (i,)})

    for i in range(n):
        for j in range(i + 1, n):
            cons.append({'type': 'ineq', 'fun': constraint_overlap, 'args': (i, j)})

    bounds = []
    for i in range(n):
        bounds.append((0.0, 1.0))
        bounds.append((0.0, 1.0))
        bounds.append((0.0, 0.5))

    result = minimize(objective, params0, args=(n,), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': 3000, 'ftol': 1e-15})

    return result.x


def perturb_params(params, scale=0.02):
    perturbed = params.copy()
    n = len(params) // 3
    for i in range(n):
        perturbed[3 * i] += np.random.uniform(-scale, scale)
        perturbed[3 * i + 1] += np.random.uniform(-scale, scale)
        perturbed[3 * i + 2] += np.random.uniform(-scale, scale)
    # Clamp
    for i in range(n):
        perturbed[3 * i] = max(0.01, min(0.99, perturbed[3 * i]))
        perturbed[3 * i + 1] = max(0.01, min(0.99, perturbed[3 * i + 1]))
        perturbed[3 * i + 2] = max(0.001, min(0.49, perturbed[3 * i + 2]))
    return perturbed


def run_packing():
    n = 26

    params0 = initialize_hexagonal(n)

    best_params = params0.copy()
    best_sum = -np.inf

    # First optimization run
    params_opt = optimize_with_slsqp(params0, n)
    current_sum = np.sum(params_opt[2::3])
    if current_sum > best_sum:
        best_sum = current_sum
        best_params = params_opt.copy()

    # Try perturbed restarts
    for restart in range(3):
        np.random.seed(42 + restart)
        perturbed = perturb_params(best_params, scale=0.01)
        params_opt = optimize_with_slsqp(perturbed, n)
        current_sum = np.sum(params_opt[2::3])
        if current_sum > best_sum:
            best_sum = current_sum
            best_params = params_opt.copy()

    # Force-directed refinement
    centers = np.array([[best_params[3 * i], best_params[3 * i + 1]] for i in range(n)])
    radii = np.array([best_params[3 * i + 2] for i in range(n)])

    centers, radii = force_directed_refinement(n, centers, radii, iterations=3000)

    # Final SLSQP pass on force-directed result
    params_fd = np.zeros(3 * n)
    for i in range(n):
        params_fd[3 * i] = centers[i, 0]
        params_fd[3 * i + 1] = centers[i, 1]
        params_fd[3 * i + 2] = radii[i]

    params_opt = optimize_with_slsqp(params_fd, n)
    current_sum = np.sum(params_opt[2::3])
    if current_sum > best_sum:
        best_sum = current_sum
        best_params = params_opt.copy()

    centers = np.array([[best_params[3 * i], best_params[3 * i + 1]] for i in range(n)])
    radii = np.array([best_params[3 * i + 2] for i in range(n)])

    # Final force-directed pass
    centers, radii = force_directed_refinement(n, centers, radii, iterations=2000)

    # Ensure all radii are positive
    radii = np.maximum(radii, 1e-8)

    total_sum = np.sum(radii)

    return centers, radii, total_sum
