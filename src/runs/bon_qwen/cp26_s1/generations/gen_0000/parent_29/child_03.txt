

Looking at this circle packing problem, I need to pack 26 circles in a unit square to maximize the sum of radii, targeting 2.636.

<strategy>
1. **Initial Configuration**: Create hexagonal packings as they're the densest circle arrangement. I'll generate multiple variants (different row counts, shifts) to explore the space.

2. **Core Optimization Strategy - Alternating Optimization**:
   - **LP for Radii**: Given fixed positions, the optimal radii form a Linear Program (maximize sum of radii subject to boundary and non-overlap constraints). This is exact and efficient.
   - **Gradient Optimization for Positions**: Given fixed radii, use BFGS to minimize an overlap penalty function. This is non-convex but effective for local refinement.

3. **Scaling Technique**: After optimization converges, gradually scale up all radii while re-optimizing positions. This pushes the solution toward the true optimum.

4. **Multiple Restart Configurations**: Generate diverse initial packings:
   - Hexagonal with different row counts
   - Corner-focused arrangements
   - Random perturbations
   
5. **Validation**: Ensure all constraints are satisfied before returning results.

The key insight is that unequal radii outperform equal radii because corner/edge circles can be larger, while inner circles fill gaps. The LP naturally discovers this when positions are fixed.
</strategy>

import numpy as np
from scipy.optimize import minimize, linprog


def initial_hexagonal_packing(n, r=0.08, shift_pattern=0):
    """Create initial hexagonal packing configuration with equal radii."""
    centers = np.zeros((n, 2))
    idx = 0
    row = 0
    col = 0
    while idx < n:
        x = r + col * 2 * r
        y = r + row * np.sqrt(3) * r
        if row % 2 == 1:
            x += r + shift_pattern * r * 0.1
        
        if x + r <= 1 and y + r <= 1:
            centers[idx] = [x, y]
            idx += 1
            col += 1
        else:
            row += 1
            col = 0
    
    radii = np.ones(n) * r
    return centers, radii


def initial_corner_packing(n):
    """Create initial packing with emphasis on corners."""
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.06
    
    # Place 4 circles near corners
    corner_positions = [[0.15, 0.15], [0.85, 0.15], [0.15, 0.85], [0.85, 0.85]]
    corner_radii = [0.12, 0.12, 0.12, 0.12]
    
    for i in range(4):
        centers[i] = corner_positions[i]
        radii[i] = corner_radii[i]
    
    # Fill remaining with hexagonal pattern
    r_fill = 0.06
    idx = 4
    row = 0
    col = 0
    while idx < n:
        x = r_fill + col * 2 * r_fill
        y = r_fill + row * np.sqrt(3) * r_fill
        if row % 2 == 1:
            x += r_fill
        
        if x + r_fill <= 1 and y + r_fill <= 1:
            centers[idx] = [x, y]
            idx += 1
            col += 1
        else:
            row += 1
            col = 0
    
    return centers, radii


def compute_overlap_penalty(centers, radii):
    """Compute penalty for overlapping circles and boundary violations."""
    n = len(radii)
    penalty = 0.0
    
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < 0:
            penalty += (x - r) ** 2
        if x + r > 1:
            penalty += (x + r - 1) ** 2
        if y - r < 0:
            penalty += (y - r) ** 2
        if y + r > 1:
            penalty += (y + r - 1) ** 2
    
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            overlap = radii[i] + radii[j] - dist
            if overlap > 0:
                penalty += overlap ** 2
    
    return penalty


def optimize_positions_bfgs(centers, radii, n):
    """Optimize positions given fixed radii using BFGS."""
    def objective(vars):
        c = vars.reshape((n, 2))
        return compute_overlap_penalty(c, radii)
    
    result = minimize(objective, centers.flatten(), method='BFGS',
                     options={'maxiter': 10000, 'gtol': 1e-12})
    return result.x.reshape((n, 2))


def optimize_positions_cg(centers, radii, n):
    """Optimize positions given fixed radii using conjugate gradient."""
    def objective(vars):
        c = vars.reshape((n, 2))
        return compute_overlap_penalty(c, radii)
    
    result = minimize(objective, centers.flatten(), method='CG',
                     options={'maxiter': 20000, 'gtol': 1e-12})
    return result.x.reshape((n, 2))


def optimize_radii_lp(centers, n, r_min=0.005):
    """Optimize radii given fixed positions using LP."""
    c = -np.ones(n)
    
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        x, y = centers[i]
        for bound_val in [x, 1 - x, y, 1 - y]:
            row = [0.0] * n
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(bound_val)
    
    # Non-overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            row = [0.0] * n
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dist)
    
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    bounds = [(r_min, None) for _ in range(n)]
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        return result.x
    else:
        return np.ones(n) * r_min


def optimize_positions_force(centers, radii, n, steps=500):
    """Optimize positions using force-directed method."""
    centers = centers.copy()
    
    for step in range(steps):
        forces = np.zeros((n, 2))
        
        # Repulsion forces between overlapping circles
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[j, 0] - centers[i, 0]
                dy = centers[j, 1] - centers[i, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                
                if dist > 1e-12:
                    overlap = radii[i] + radii[j] - dist
                    if overlap > 0:
                        fx = overlap * dx / dist
                        fy = overlap * dy / dist
                        forces[i] -= np.array([fx, fy])
                        forces[j] += np.array([fx, fy])
        
        # Boundary forces
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < 0:
                forces[i, 0] += (r - x) * 10
            if x + r > 1:
                forces[i, 0] -= (x + r - 1) * 10
            if y - r < 0:
                forces[i, 1] += (r - y) * 10
            if y + r > 1:
                forces[i, 1] -= (y + r - 1) * 10
        
        # Apply forces with decaying learning rate
        lr = 0.05 / (1 + step * 0.01)
        centers += forces * lr
        centers = np.clip(centers, 0.001, 0.999)
    
    return centers


def refine_packing(centers, radii, n):
    """Refine a packing through alternating optimization."""
    for iteration in range(150):
        # Optimize radii with LP
        radii = optimize_radii_lp(centers, n, r_min=0.005)
        
        # Optimize positions with BFGS
        centers = optimize_positions_bfgs(centers, radii, n)
        
        # Try to scale up radii
        scale = 1.0005
        radii_scaled = radii * scale
        centers_opt = optimize_positions_bfgs(centers, radii_scaled, n)
        
        penalty = compute_overlap_penalty(centers_opt, radii_scaled)
        if penalty < 1e-12:
            centers = centers_opt
            radii = radii_scaled
        else:
            # Try smaller scale
            for s in [1.0003, 1.0001, 1.00005]:
                radii_test = radii * s
                centers_test = optimize_positions_bfgs(centers, radii_test, n)
                pen_test = compute_overlap_penalty(centers_test, radii_test)
                if pen_test < 1e-12:
                    centers = centers_test
                    radii = radii_test
                    break
            else:
                break
    
    # Final force-directed refinement
    centers = optimize_positions_force(centers, radii, n, steps=300)
    radii = optimize_radii_lp(centers, n, r_min=0.005)
    
    return centers, radii


def run_packing():
    n = 26
    
    best_centers = None
    best_radii = None
    best_sum = 0.0
    
    # Generate diverse initial configurations
    initial_configs = []
    
    # Hexagonal packings with different parameters
    for r_init in [0.07, 0.08, 0.09, 0.10]:
        for shift in [0, 1]:
            c, r = initial_hexagonal_packing(n, r=r_init, shift_pattern=shift)
            initial_configs.append((c, r))
    
    # Corner-focused packing
    c, r = initial_corner_packing(n)
    initial_configs.append((c, r))
    
    # Random perturbed hexagonal
    for _ in range(5):
        c, r = initial_hexagonal_packing(n, r=0.085)
        c += np.random.randn(n, 2) * 0.02
        c = np.clip(c, 0.05, 0.95)
        initial_configs.append((c, r))
    
    # Optimize each configuration
    for idx, (centers, radii) in enumerate(initial_configs):
        centers_opt, radii_opt = refine_packing(centers.copy(), radii.copy(), n)
        
        current_sum = np.sum(radii_opt)
        
        # Validate
        valid = True
        for i in range(n):
            x, y = centers_opt[i]
            r = radii_opt[i]
            if x - r < -1e-10 or x + r > 1 + 1e-10 or y - r < -1e-10 or y + r > 1 + 1e-10:
                valid = False
                break
            if r < 0:
                valid = False
                break
        
        for i in range(n):
            if not valid:
                break
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers_opt[i] - centers_opt[j]) ** 2))
                if dist < radii_opt[i] + radii_opt[j] - 1e-10:
                    valid = False
                    break
        
        if valid and current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers_opt.copy()
            best_radii = radii_opt.copy()
    
    # Additional scaling attempts on best solution
    if best_centers is not None:
        for _ in range(50):
            scale = 1.0001
            radii_test = best_radii * scale
            centers_test = optimize_positions_bfgs(best_centers, radii_test, n)
            penalty = compute_overlap_penalty(centers_test, radii_test)
            if penalty < 1e-12:
                best_centers = centers_test
                best_radii = radii_test
                best_sum = np.sum(best_radii)
            else:
                break
    
    return best_centers, best_radii, best_sum
