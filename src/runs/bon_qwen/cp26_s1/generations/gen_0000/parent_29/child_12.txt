import numpy as np
from scipy.optimize import minimize

def get_loss(centers, radii, penalty_strength=100.0):
    """
    Calculates the loss function: -sum(radii) + penalty for violations.
    We want to minimize this.
    """
    n = centers.shape[0]
    obj = -np.sum(radii)
    penalty = 0.0
    
    # Boundary constraints
    # x - r >= 0  => r - x <= 0
    # x + r <= 1  => x + r - 1 <= 0
    # same for y
    
    # Check x bounds
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        
        # Boundary violations
        if x - r < 0: penalty += penalty_strength * (-(x - r))**2
        if x + r > 1: penalty += penalty_strength * (x + r - 1)**2
        if y - r < 0: penalty += penalty_strength * (-(y - r))**2
        if y + r > 1: penalty += penalty_strength * (y + r - 1)**2
        
        # Overlap constraints
        for j in range(i + 1, n):
            dist_sq = (centers[i][0] - centers[j][0])**2 + (centers[i][1] - centers[j][1])**2
            dist = np.sqrt(dist_sq)
            r_sum = radii[i] + radii[j]
            if dist < r_sum:
                violation = r_sum - dist
                penalty += penalty_strength * violation**2
                
    return obj + penalty
