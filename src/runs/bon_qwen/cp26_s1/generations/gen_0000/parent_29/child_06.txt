def compute_cost_and_gradient(vars_vec, n_circles, penalty_weight=100.0):

    Cost = -sum(radii) + penalty_weight * (overlap_penalty + boundary_penalty)
    """

    state = vars_vec.reshape((n_circles, 3))

    cost = -np.sum(radii)

    # y - r >= 0
    # y + r <= 1
    
    # We use a squared hinge loss: max(0, violation)^2
    # Violation 1: r - x > 0. Val = r - x.
    # Violation 2: x + r - 1 > 0. Val = x + r - 1.
    
    v1 = np.maximum(0, radii - centers[:, 0]) # r - x
    v2 = np.maximum(0, centers[:, 0] + radii - 1.0) # x + r - 1
    v3 = np.maximum(0, radii - centers[:, 1]) # r - y
    v4 = np.maximum(0, centers[:, 1] + radii - 1.0) # y + r - 1
    
    boundary_penalty = np.sum(v1**2 + v2**2 + v3**2 + v4**2)

    # Direction vector: -diff * inv_dists

    grad[:, 1] += grad_xy[:, 1]
