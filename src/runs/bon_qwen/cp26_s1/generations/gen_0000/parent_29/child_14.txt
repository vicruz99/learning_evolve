import numpy as np
from scipy.optimize import minimize
import warnings
warnings.filterwarnings('ignore')


def objective(vars):
    radii = vars[2::3]
    return -np.sum(radii)


def boundary_x_ge_r(vars, i):
    return vars[3*i] - vars[3*i+2]


def boundary_x_le_1r(vars, i):
    return 1.0 - vars[3*i] - vars[3*i+2]


def boundary_y_ge_r(vars, i):
    return vars[3*i+1] - vars[3*i+2]


def boundary_y_le_1r(vars, i):
    return 1.0 - vars[3*i+1] - vars[3*i+2]


def non_overlap(vars, i, j):
    dx = vars[3*i] - vars[3*j]
    dy = vars[3*i+1] - vars[3*j+1]
    dr = vars[3*i+2] + vars[3*j+2]
    return dx*dx + dy*dy - dr*dr


def build_constraints(n):
    cons = []
    for i in range(n):
        cons.append({'type': 'ineq', 'fun': boundary_x_ge_r, 'args': (i,)})
        cons.append({'type': 'ineq', 'fun': boundary_x_le_1r, 'args': (i,)})
        cons.append({'type': 'ineq', 'fun': boundary_y_ge_r, 'args': (i,)})
        cons.append({'type': 'ineq', 'fun': boundary_y_le_1r, 'args': (i,)})
    
    for i in range(n):
        for j in range(i+1, n):
            cons.append({'type': 'ineq', 'fun': non_overlap, 'args': (i, j)})
    
    return cons


def initialize_hexagonal(n, r=0.08):
    centers = np.zeros((n, 2))
    radii = np.ones(n) * r
    
    idx = 0
    y = r + 0.01
    row = 0
    while idx < n:
        offset = r + 0.01 if row % 2 == 1 else 0
        x = r + 0.01 + offset
        while x + r <= 1.0 - 0.01 and idx < n:
            centers[idx] = [x, y]
            radii[idx] = r
            idx += 1
            x += 2 * r + 0.02
        y += np.sqrt(3) * r + 0.01
        row += 1
    
    for i in range(n):
        centers[i, 0] = np.clip(centers[i, 0], radii[i] + 1e-6, 1.0 - radii[i] - 1e-6)
        centers[i, 1] = np.clip(centers[i, 1], radii[i] + 1e-6, 1.0 - radii[i] - 1e-6)
    
    return centers, radii

def force_directed_packing(n, init_r=0.06, iterations=5000):
    rng = np.random.RandomState(123)
    centers = np.zeros((n, 2))
    radii = np.ones(n) * init_r
    
    row_y = init_r + 0.02
    row = 0
    idx = 0
    while idx < n:
        offset = init_r + 0.02 if row % 2 == 1 else 0
        x = init_r + 0.02 + offset
        while x + init_r <= 1.0 - 0.02 and idx < n:
            centers[idx] = [x + rng.uniform(-0.005, 0.005), row_y + rng.uniform(-0.005, 0.005)]
            idx += 1
            x += 2 * init_r + 0.04
        row_y += np.sqrt(3) * init_r + 0.02
        row += 1
    
    growth_rate = 0.0003
    for it in range(iterations):
        forces = np.zeros((n, 2))
        
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[j, 0] - centers[i, 0]
                dy = centers[j, 1] - centers[i, 1]
                dist = np.sqrt(dx*dx + dy*dy) + 1e-12
                desired = radii[i] + radii[j] + 0.005
                
                if dist < desired:
                    strength = (desired - dist) / dist * 0.5
                    fx = dx * strength
                    fy = dy * strength
                    forces[i, 0] -= fx
                    forces[i, 1] -= fy
                    forces[j, 0] += fx
                    forces[j, 1] += fy
        
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            margin = 0.002
            if x - r < margin:
                forces[i, 0] += 0.5 * (margin + r - x)
            if x + r > 1.0 - margin:
                forces[i, 0] -= 0.5 * (x + r - 1.0 + margin)
            if y - r < margin:
                forces[i, 1] += 0.5 * (margin + r - y)
            if y + r > 1.0 - margin:
                forces[i, 1] -= 0.5 * (y + r - 1.0 + margin)
        
        step = max(0.001, 0.05 / (1 + it * 0.001))
        centers += step * forces
        centers[:, 0] = np.clip(centers[:, 0], 0.001, 0.999)
        centers[:, 1] = np.clip(centers[:, 1], 0.001, 0.999)
        
        radii += growth_rate / (1 + it * 0.002)
    
    return centers, radii


def optimize_packing_from(centers, radii, n, constraints, max_iter=2000):
    vars0 = np.zeros(3 * n)
    vars0[0::3] = centers[:, 0]
    vars0[1::3] = centers[:, 1]
    vars0[2::3] = radii
    
    bounds = []
    for i in range(n):
        bounds.append((0.0, 1.0))
        bounds.append((0.0, 1.0))
        bounds.append((1e-8, 0.5))
    
    result = minimize(objective, vars0, method='SLSQP',
                     bounds=bounds, constraints=constraints,
                     options={'maxiter': max_iter, 'ftol': 1e-15})
    
    final_centers = np.column_stack([result.x[0::3], result.x[1::3]])
    final_radii = result.x[2::3]
    final_radii = np.clip(final_radii, 1e-8, None)
    
    return final_centers, final_radii

            break
        
        for i in range(n):
