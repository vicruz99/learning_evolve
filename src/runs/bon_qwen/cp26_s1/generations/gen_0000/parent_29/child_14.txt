import numpy as np
from scipy.optimize import minimize

def compute_constraints(vars, n):
    """Computes boundary and non-overlap constraints for SLSQP."""
    m = 4 * n + n * (n - 1) // 2
    cons = np.empty(m)
    idx = 0
    
    # Boundary constraints: x >= r, x+r <= 1, y >= r, y+r <= 1
    for i in range(n):
        x, y, r = vars[3 * i], vars[3 * i + 1], vars[3 * i + 2]
        cons[idx] = x - r
        cons[idx + 1] = 1.0 - x - r
        cons[idx + 2] = y - r
        cons[idx + 3] = 1.0 - y - r
        idx += 4
        
        xi, yi, ri = x, y, r
        # Overlap constraints: dist^2 >= (ri + rj)^2
        for j in range(i + 1, n):
            xj, yj, rj = vars[3 * j], vars[3 * j + 1], vars[3 * j + 2]
            dx, dy = xi - xj, yi - yj
            dist_sq = dx * dx + dy * dy
            sum_r = ri + rj
            cons[idx] = dist_sq - sum_r * sum_r
            idx += 1
    return cons

def objective(vars, n):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(vars[2::3])

def run_packing():
    n = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Constraint dictionary
    cons = {'type': 'ineq', 'fun': compute_constraints, 'args': (n,)}
    # Variable bounds: x,y in [0,1], r in [0, 0.5]
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)] * n + [(0.0, 0.5)] * n
    
    for restart in range(20):
        if restart == 0:
            # Hexagonal lattice initialization for better convergence
            x, y, r = [], [], []
            row = 0
            while len(x) < n:
                y_val = 0.15 + row * 0.23
                if y_val > 0.85: break
                shift = (row % 2) * 0.115
                col = 0
                while len(x) < n:
                    x_val = 0.15 + col * 0.23 + shift
                    if x_val > 0.85: break
                    x.append(x_val)
                    y.append(y_val)
                    r.append(0.07)
                    col += 1
                row += 1
        else:
            # Random initialization with small radii
            x = np.random.uniform(0.15, 0.85, n)
            y = np.random.uniform(0.15, 0.85, n)
            r = np.ones(n) * 0.04
            
        vars0 = np.zeros(3 * n)
        for i in range(n):
            vars0[3 * i] = x[i]
            vars0[3 * i + 1] = y[i]
            vars0[3 * i + 2] = r[i]
            
        try:
            res = minimize(objective, vars0, args=(n,), method='SLSQP', 
                          bounds=bounds, constraints=cons, 
                          options={'maxiter': 2000, 'ftol': 1e-10})
            
            c = np.column_stack((res.x[0::3], res.x[1::3]))
            rad = res.x[2::3]
            
            # Strict internal validation
            valid = True
            if np.any(rad < -1e-7): 
                valid = False
            if np.any(c[:, 0] - rad < -1e-7) or np.any(c[:, 0] + rad > 1 + 1e-7): 
                valid = False
            if np.any(c[:, 1] - rad < -1e-7) or np.any(c[:, 1] + rad > 1 + 1e-7): 
                valid = False
                
            if valid:
                for i in range(n):
                    for j in range(i + 1, n):
                        dist = np.hypot(c[i, 0] - c[j, 0], c[i, 1] - c[j, 1])
                        if dist < rad[i] + rad[j] - 1e-7:
                            valid = False
                            break
                    if not valid: break
                    
            if valid:
                s = np.sum(rad)
                if s > best_sum:
                    best_sum = s
                    best_centers = c.copy()
                    best_radii = rad.copy()
                    
        except Exception:
            continue
            
    # Fallback valid packing (5x5 grid + 1) in case optimization fails
    if best_centers is None:
        centers = np.zeros((n, 2))
        radii = np.zeros(n)
        for i in range(n):
            row = i // 6
            col = i % 6
            centers[i, 0] = 0.08 + col * 0.16
            centers[i, 1] = 0.08 + row * 0.16
            radii[i] = 0.06
        best_centers = centers
        best_radii = radii
        best_sum = np.sum(radii)
        
    return best_centers, best_radii, best_sum
