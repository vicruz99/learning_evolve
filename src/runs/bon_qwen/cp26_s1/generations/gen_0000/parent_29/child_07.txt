import numpy as np
from scipy.optimize import minimize

def objective(v):
    n = 26
    return -np.sum(v[2*n:])

def boundary_ineq(v):
    n = 26
    cons = np.empty(4 * n)
    for i in range(n):
        idx = 3 * i
        x, y, r = v[idx], v[idx+1], v[idx+2]
        cons[4*i] = x - r
        cons[4*i+1] = 1.0 - x - r
        cons[4*i+2] = y - r
        cons[4*i+3] = 1.0 - y - r
    return cons

def overlap_ineq(v):
    n = 26
    m = n * (n - 1) // 2
    cons = np.empty(m)
    k = 0
    for i in range(n):
        idx_i = 3 * i
        xi, yi, ri = v[idx_i], v[idx_i+1], v[idx_i+2]
        for j in range(i + 1, n):
            idx_j = 3 * j
            xj, yj, rj = v[idx_j], v[idx_j+1], v[idx_j+2]
            dist2 = (xi - xj)**2 + (yi - yj)**2
            cons[k] = dist2 - (ri + rj)**2
            k += 1
    return cons

def run_packing():
    n = 26
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * n
    
    constraints = [
        {'type': 'ineq', 'fun': boundary_ineq},
        {'type': 'ineq', 'fun': overlap_ineq}
    ]
    
    best_sum = 0.0
    best_centers = np.zeros((n, 2))
    best_radii = np.zeros(n)
    
    starts = []
    
    # Config 1: Dense grid perturbed
    c1 = []
    for i in range(5):
        for j in range(5):
            c1.append([0.1 + i*0.2, 0.1 + j*0.2])
    c1.append([0.5, 0.5])
    r1 = np.ones(n) * 0.095
    starts.append((np.array(c1), r1))
    
    # Config 2: Hexagonal arrangement
    c2 = []
    r2 = np.zeros(n)
    y = 0.1
    cnt = 0
    while cnt < n:
        x = 0.1
        while x <= 0.9 and cnt < n:
            c2.append([x, y])
            r2[cnt] = 0.095
            x += 0.1732
            cnt += 1
        y += 0.1
        if y > 0.9: break
    while len(c2) < n:
        c2.append([0.5, 0.5])
        r2[len(c2)-1] = 0.01
    starts.append((np.array(c2[:n]), r2[:n]))
    
    # Config 3: Random seeded
    np.random.seed(42)
    c3 = []
    r3 = np.zeros(n)
    for i in range(n):
        c3.append([np.random.uniform(0.1, 0.9), np.random.uniform(0.1, 0.9)])
        r3[i] = 0.08
    starts.append((np.array(c3), r3))
    
    for centers, radii in starts:
        x0 = np.concatenate([centers.flatten(), radii])
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, 
                           constraints=constraints, options={'maxiter': 8000, 'ftol': 1e-12, 'disp': False})
            cur_sum = -res.fun
            if cur_sum > best_sum:
                best_sum = cur_sum
                best_centers = res.x[:2*n].reshape(n, 2)
                best_radii = res.x[2*n:]
        except Exception:
            continue
            
    return best_centers, best_radii, best_sum
