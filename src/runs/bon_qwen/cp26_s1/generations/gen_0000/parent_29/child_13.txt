from scipy.optimize import minimize

    # Distribute 26 circles into 6 rows: 4, 5, 5, 5, 5, 2

        span = (count - 1) * 2 * initial_r
        x_start = (1 - span) / 2 + initial_r  # Center of first circle

        shift = initial_r if row_idx % 2 == 1 else 0

                x = x_start + col_idx * 2 * initial_r + shift
                # Clamp to valid range [initial_r, 1-initial_r] just in case
                x = np.clip(x, initial_r, 1 - initial_r)
                centers.append([x, y_pos])
        y_pos += initial_r * np.sqrt(3)

        return -np.sum(radii)

        # But we flatten as [x1, y1, r1, x2, y2, r2, ...]
        # Actually, let's use shape (n, 3) for convenience in constraints
        # But optimize expects 1D vector.
        # x_vec indices: 0,1,2 (c1), 3,4,5 (c2)...
        # i-th circle: idx = 3*i
        
        # We'll use a list of constraint dictionaries for SLSQP
        pass # Handled below

    # We will pack variables as [x1, y1, ..., xn, yn, r1, r2, ..., rn]

        x = x_vec[:n]
        y = x_vec[n:2*n]
        r = x_vec[2*n:]
        return x, y, r

        x, y, r = extract_state(x_vec)
        # For every pair (i, j) with i < j
        # (xi-xj)^2 + (yi-yj)^2 >= (ri+rj)^2
        # => (xi-xj)^2 + (yi-yj)^2 - (ri+rj)^2 >= 0

            for j in range(i + 1, n):
                dx = x[i] - x[j]
                dy = y[i] - y[j]
                dr = r[i] + r[j]
                constraints.append(dx**2 + dy**2 - dr**2)
        return np.array(constraints)
