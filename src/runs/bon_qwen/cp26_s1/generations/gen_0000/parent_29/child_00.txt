from scipy.optimize import minimize, linprog
from itertools import combinations

    Pack 26 circles in a unit square to maximize the sum of radii.
    
    Returns:
        centers: np.ndarray of shape (26, 2)
        radii: np.ndarray of shape (26,)
        sum_radii: float

    n = 26
    
    # Precompute pair indices for constraints
    # We need constraints r_i + r_j <= dist(i, j) for all pairs i < j
    pairs = list(combinations(range(n), 2))
    num_pairs = len(pairs)
    
    # Construct the constraint matrix A_ub for the LP.
    # A_ub @ r <= b_ub
    # Each row corresponds to a pair (i, j) with 1 at col i and 1 at col j.
    # This matrix is constant regardless of center positions.
    A_ub_pairs = np.zeros((num_pairs, n))
    for k, (i, j) in enumerate(pairs):
        A_ub_pairs[k, i] = 1.0
        A_ub_pairs[k, j] = 1.0
        
    # Objective vector for LP: maximize sum(r) => minimize -sum(r)
    c_obj_lp = -np.ones(n)

    def solve_lp_for_centers(centers):
        """
        Given fixed centers, solve LP to find optimal radii.
        Returns sum of radii, or -1e9 if infeasible.
        """
        # 1. Compute distances between all pairs
        # Centers shape (n, 2)
        # dists[i, j] is distance between center i and j
        # Using broadcasting
        diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        
        # Extract distances for the specific pairs we have constraints for
        b_ub_pairs = np.array([dists[i, j] for i, j in pairs])
        
        # 2. Compute boundary constraints for each circle
        # r_i <= min(x, 1-x, y, 1-y)
        x = centers[:, 0]
        y = centers[:, 1]
        
        # Distance to boundaries
        d_left = x
        d_right = 1 - x
        d_down = y
        d_up = 1 - y
        
        # Max radius allowed by boundaries
        r_bounds_max = np.minimum(np.minimum(d_left, d_right), np.minimum(d_down, d_up))
        
        # Bounds for LP variables: 0 <= r_i <= r_bounds_max
        # If center is outside [0,1], r_bounds_max might be negative, making LP infeasible.
        # We clip to 0 to avoid infeasibility if centers drift slightly out, 
        # but optimizer bounds should prevent this.
        r_bounds_max = np.maximum(r_bounds_max, 0.0)
        
        bounds = [(0.0, r) for r in r_bounds_max]
        
        # 3. Solve LP
        # Maximize sum(r_i) subject to:
        # r_i + r_j <= dist(i, j)
        # 0 <= r_i <= r_bounds_max
        res = linprog(c_obj_lp, A_ub=A_ub_pairs, b_ub=b_ub_pairs, bounds=bounds, method='highs')
        
        if res.success:
            # res.fun is -sum(r), so we return sum(r)
            return -res.fun, res.x
        else:
            # If LP fails, return a penalty
            return -1e9, np.zeros(n)

    def objective(centers_flat):
        """
        Objective function for center optimization.
        Returns negative sum of radii (to minimize).
        """
        centers = centers_flat.reshape((n, 2))
        sum_r, _ = solve_lp_for_centers(centers)
        return -sum_r
