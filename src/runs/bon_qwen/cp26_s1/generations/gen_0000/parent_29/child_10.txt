def get_hex_grid(n):
    """Generate a hexagonal grid initialization for n points."""
    points = []
    y = 0.05
    while len(points) < n:
        x = 0.05

            points.append([x, y])
            if len(points) == n:
                break
            x += 0.15
        y += 0.13
        if len(points) == n:
            break
    return np.array(points[:n])
