import numpy as np
from scipy.optimize import minimize

# Global constant for number of circles
N = 26

def get_initial_packing(n=26, seed=0):

    Generates a valid initial packing of n circles using a greedy max-min distance heuristic.
    """
    np.random.seed(seed)
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Place first circle at center
    centers[0] = [0.5, 0.5]
    radii[0] = 0.4  # Large initial radius

        best_val = -1

        samples = np.random.rand(2000, 2)

        diff = samples[:, np.newaxis, :] - centers[:i, np.newaxis, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        min_dists = np.min(dists, axis=1)
        
        # Distance to boundaries
        dist_to_left = samples[:, 0]
        dist_to_right = 1 - samples[:, 0]
        dist_to_bottom = samples[:, 1]
        dist_to_top = 1 - samples[:, 1]
        dist_to_boundary = np.min(np.array([dist_to_left, dist_to_right, dist_to_bottom, dist_to_top]), axis=0)
        
        # The feasible radius is limited by half the distance to nearest neighbor and distance to boundary
        feasible_r = np.minimum(min_dists / 2, dist_to_boundary)
        
        # Select the position that allows the largest radius
        idx = np.argmax(feasible_r)
        best_pos = samples[idx]
        best_r = feasible_r[idx]

        radii[i] = best_r
        
    return centers, radii

def objective(v):
    """
    Objective function: Minimize negative sum of radii.
    v is a flat array [x1, y1, r1, x2, y2, r2, ...]
    """
    radii = v[2::3]
    return -np.sum(radii)

def boundary_constraints(v):
    """
    Boundary constraints:
    x - r >= 0
    1 - x - r >= 0
    y - r >= 0
    1 - y - r >= 0
    Returns a flat array of constraint values.
    """
    C = v.reshape(-1, 3)
    x = C[:, 0]
    y = C[:, 1]
    r = C[:, 2]
    
    c1 = x - r
    c2 = 1 - x - r
    c3 = y - r
    c4 = 1 - y - r
    
    return np.concatenate([c1, c2, c3, c4])

def overlap_constraints(v):
    """
    Overlap constraints:
    dist(c_i, c_j)^2 - (r_i + r_j)^2 >= 0 for all i < j
    Returns a flat array of constraint values.
    """
    C = v.reshape(-1, 3)
    X = C[:, 0:1]
    Y = C[:, 1:2]
    R = C[:, 2:3]
    
    # Compute pairwise differences and sums
    DX = X - X.T
    DY = Y - Y.T
    DR = R + R.T
    
    DistSq = DX**2 + DY**2
    SumR = DR**2
    
    # Get indices for upper triangle (i < j)
    idx = np.triu_indices(N, k=1)
    violations = DistSq[idx] - SumR[idx]
    
    return violations

def run_packing():

    Main function to run the optimization.

        {'type': 'ineq', 'fun': overlap_constraints}
    ]
