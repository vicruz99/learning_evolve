import numpy as np
import scipy.optimize as opt

def solve_radii(centers):
    """
    Solves the LP to maximize sum of radii given fixed centers.
    Returns radii, sum_radii, and dual variables (marginals).
    """
    n = centers.shape[0]
    # Distance matrix
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    # LP Constraints: A_ub * r <= b_ub
    # 1. Pairwise: r_i + r_j <= dist_ij
    # 2. Boundary: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    
    n_constraints = (n * (n - 1)) // 2 + 4 * n
    A_ub = np.zeros((n_constraints, n))
    b_ub = np.zeros(n_constraints)
    
    idx = 0
    # Pairwise constraints
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub[idx] = row
            b_ub[idx] = dists[i, j]
            idx += 1
            
    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        # r_i <= x
        row = np.zeros(n)
        row[i] = 1.0
        A_ub[idx] = row
        b_ub[idx] = x
        idx += 1
        # r_i <= 1-x
        A_ub[idx] = row
        b_ub[idx] = 1.0 - x
        idx += 1
        # r_i <= y
        A_ub[idx] = row
        b_ub[idx] = y
        idx += 1
        # r_i <= 1-y
        A_ub[idx] = row
        b_ub[idx] = 1.0 - y
        idx += 1

    # Objective: Maximize sum(r_i) => Minimize -sum(r_i)
    c = -np.ones(n)
    bounds = [(0, None) for _ in range(n)]
    
    try:
        # Use 'highs' method as it is efficient and supports marginals
        res = opt.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        
        if res.success:
            radii = res.x
            sum_radii = np.sum(radii)
            # Marginals (dual variables)
            # In scipy linprog with highs, marginals.ineqlin corresponds to duals for A_ub x <= b_ub
            # The duals y satisfy y >= 0 and represent the shadow price.
            # Specifically, increasing b_ub by epsilon increases the objective value (min -sum r) by -epsilon * y?
            # Let's check sign convention. 
            # linprog minimizes c^T x. 
            # If we increase b (relax constraint), min value decreases (becomes more negative).
            # So derivative of min_obj wrt b is -y (where y >= 0).
            # But we care about max sum r = -min_obj.
            # Derivative of max_sum wrt b is y.
            # So marginal is exactly the rate of change of sum of radii wrt RHS.
            marginals = res.marginals.ineqlin
            return radii, sum_radii, marginals, dists
        else:
            # Fallback if LP fails (should not happen for valid geometry)
            return np.zeros(n), 0.0, np.zeros(n_constraints), dists
    except Exception:
        return np.zeros(n), 0.0, np.zeros(n_constraints), dists

def compute_gradients(centers, radii, marginals):
    """
    Computes gradient of sum of radii with respect to centers.
    Uses dual variables from LP.
    """
    n = centers.shape[0]
    grad = np.zeros_like(centers)
    
    # Parse marginals
    # First N(N-1)/2 are pairwise
    # Next 4N are boundaries
    
    n_pairs = (n * (n - 1)) // 2
    pair_marginals = marginals[:n_pairs]
    bound_marginals = marginals[n_pairs:]
    
    # Indices for pairs
    # We need to map linear index back to (i, j)
    # But easier to just loop or vectorize
    
    # Let's reconstruct indices
    pairs = []
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            pairs.append((i, j))
            idx += 1
            
    # Vectorized gradient calculation for pairs
    # grad[i] += lambda_ij * (x_i - x_j) / dist_ij
    # grad[j] += lambda_ij * (x_j - x_i) / dist_ij
    
    # We need dists to normalize
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    # Handle division by zero
    dists = np.maximum(dists, 1e-9)
    
    # Normalize difference vectors
    # unit_vec[i, j] = (centers[i] - centers[j]) / dists[i, j]
    unit_vec = diff / dists[:, :, np.newaxis]
    
    # We only care about upper triangle for pairs (i < j)
    # But unit_vec[i, j] = - unit_vec[j, i]
    
    # Extract upper triangle indices
    i_idx, j_idx = np.triu_indices(n, k=1)
    
    # Get marginals for these pairs
    # The order in marginals matches the loop: (0,1), (0,2)... (0,n-1), (1,2)...
    # np.triu_indices with k=1 produces exactly this order if we iterate i then j?
    # Actually triu_indices returns indices in row-major order?
    # Let's verify or just use a loop for safety, n=26 is small.
    
    # Using loop for clarity and correctness of index mapping
    m_idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            lam = pair_marginals[m_idx]
            if lam > 1e-9: # Only process active constraints
                vec = (centers[i] - centers[j]) / dists[i, j]
                grad[i] += lam * vec
                grad[j] -= lam * vec
            m_idx += 1
            
    # Boundary constraints
    # 4 constraints per circle
    # 0: r <= x (Left)  -> grad += mu * (1, 0)
    # 1: r <= 1-x (Right) -> grad += mu * (-1, 0)  (since RHS is 1-x, deriv wrt x is -1)
    # 2: r <= y (Bottom) -> grad += mu * (0, 1)
    # 3: r <= 1-y (Top) -> grad += mu * (0, -1)
    
    for i in range(n):
        base = i * 4
        mu_left = bound_marginals[base]
        mu_right = bound_marginals[base + 1]
        mu_bottom = bound_marginals[base + 2]
        mu_top = bound_marginals[base + 3]
        
        grad[i, 0] += mu_left * 1.0 + mu_right * (-1.0)
        grad[i, 1] += mu_bottom * 1.0 + mu_top * (-1.0)
        
    return grad
