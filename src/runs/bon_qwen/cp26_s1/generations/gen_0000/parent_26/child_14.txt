import numpy as np
from scipy.optimize import minimize
import math

def run_packing():
    n = 26
    
    def get_constraints(n):
        cons = []
        for i in range(n):
            # Boundary: x >= r
            cons.append({'type': 'ineq', 'fun': lambda v, i=i: v[3*i] - v[3*i+2]})
            # Boundary: x <= 1 - r
            cons.append({'type': 'ineq', 'fun': lambda v, i=i: (1 - v[3*i]) - v[3*i+2]})
            # Boundary: y >= r
            cons.append({'type': 'ineq', 'fun': lambda v, i=i: v[3*i+1] - v[3*i+2]})
            # Boundary: y <= 1 - r
            cons.append({'type': 'ineq', 'fun': lambda v, i=i: (1 - v[3*i+1]) - v[3*i+2]})
            
        for i in range(n):
            for j in range(i + 1, n):
                # Non-overlap: dist >= r_i + r_j
                cons.append({
                    'type': 'ineq',
                    'fun': lambda v, i=i, j=j: 
                        (v[3*i] - v[3*j])**2 + (v[3*i+1] - v[3*j+1])**2 - (v[3*i+2] + v[3*j+2])**2
                })
        return cons

    def objective(v):
        return -sum(v[3*i+2] for i in range(n))

    bounds = [(0, 1)] * (2 * n) + [(0, 0.5)] * n

    def hex_init():
        centers = np.zeros((n, 2))
        r = 0.1
        y = r
        row = 0
        count = 0
        while count < n:
            x = r
            cols_in_row = 5 if row % 2 == 0 else 4
            for _ in range(cols_in_row):
                if count < n:
                    centers[count] = [x, y]
                    count += 1
                x += 2 * r
            y += math.sqrt(3) * r
            row += 1
        return centers

    def grid_init():
        centers = np.zeros((n, 2))
        pts = []
        step = 0.2
        for i in range(5):
            for j in range(5):
                pts.append([0.1 + i * step, 0.1 + j * step])
        if len(pts) < n:
            pts.append([0.5, 0.5])
        return np.array(pts[:n])

    def random_init():
        return np.random.rand(n, 2) * 0.8 + 0.1

    def optimize(centers):
        v0 = np.zeros(3 * n)
        for i in range(n):
            v0[3*i] = centers[i, 0]
            v0[3*i+1] = centers[i, 1]
            v0[3*i+2] = 0.05
        cons = get_constraints(n)
        res = minimize(objective, v0, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 2000})
        return res.x, res.fun

    best_sum = -1.0
    best_centers = np.zeros((n, 2))
    best_radii = np.zeros(n)

    for init_func in [hex_init, grid_init, random_init, random_init, random_init]:
        centers = init_func()
        # Quick relaxation to remove overlaps and increase radii
        for _ in range(50):
            dr = 0.001
            for i in range(n):
                centers[i, 2] = min(0.5, centers[i, 2] + dr)
                # This is a simplified repulsion step, but we'll rely on scipy for exactness
                # We just want to ensure initial radii are somewhat large
                pass
        
        # Reset radii to small value for optimizer to find true max
        v0 = np.zeros(3 * n)
        for i in range(n):
            v0[3*i] = centers[i, 0]
            v0[3*i+1] = centers[i, 1]
            v0[3*i+2] = 0.01

        res = minimize(objective, v0, method='SLSQP', bounds=bounds, constraints=get_constraints(n), options={'maxiter': 3000})
        
        if res.success and -res.fun > best_sum:
            best_sum = -res.fun
            best_centers = np.zeros((n, 2))
            best_radii = np.zeros(n)
            for i in range(n):
                best_centers[i] = [res.x[3*i], res.x[3*i+1]]
                best_radii[i] = res.x[3*i+2]

    return best_centers, best_radii, best_sum

if __name__ == "__main__":
    c, r, s = run_packing()
    print(f"Sum of radii: {s}")
    print(centers=c, radii=r)
