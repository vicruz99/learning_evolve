def is_valid(centers, radii):
    """Check if the packing satisfies all geometric constraints."""
    n = centers.shape[0]
    # Boundary checks
    if np.any(centers[:, 0] < radii - 1e-9) or np.any(centers[:, 0] > 1 - radii + 1e-9):
        return False

        return False

    diff = centers[:, None, :] - centers[None, :, :]

    reqs = radii[:, None] + radii[None, :]
    
    # Ignore self-interaction
    mask = ~np.eye(n, dtype=bool)
    return np.all(dists[mask] >= reqs[mask] - 1e-9)

def resolve(centers, radii):

    for _ in range(60):
        changed = False

        centers[:, 0] = np.clip(centers[:, 0], radii, 1 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1 - radii)
