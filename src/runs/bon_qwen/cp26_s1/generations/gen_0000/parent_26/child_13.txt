import numpy as np
from scipy.optimize import minimize

def compute_energy(vars, n, mu):
    """
    Computes the energy function: negative sum of radii + penalty for overlaps and boundary violations.
    vars: 1D array [x1, ..., xn, y1, ..., yn, r1, ..., rn]
    n: number of circles
    mu: penalty weight
    """
    xs = vars[:n]
    ys = vars[n:2*n]
    rs = vars[2*n:]
    
    # Objective: maximize sum of radii
    energy = -np.sum(rs)
    
    # Pairwise distances
    pts = np.column_stack([xs, ys])
    diff = pts[:, None, :] - pts[None, :, :]
    dist_sq = np.sum(diff**2, axis=2)
    dist = np.sqrt(dist_sq + 1e-12)
    
    rs_sum = rs[:, None] + rs[None, :]
    overlap = rs_sum - dist
    
    # Mask for upper triangle to count each pair once
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    overlap_upper = np.where(mask, overlap, 0.0)
    overlap_penalty = np.sum(np.maximum(0.0, overlap_upper)**2)
    
    # Boundary penalties: r <= x <= 1-r, r <= y <= 1-r
    b1 = np.maximum(0.0, rs - xs)
    b2 = np.maximum(0.0, xs + rs - 1.0)
    b3 = np.maximum(0.0, rs - ys)
    b4 = np.maximum(0.0, ys + rs - 1.0)
    bound_penalty = np.sum(b1**2 + b2**2 + b3**2 + b4**2)
    
    energy += mu * (overlap_penalty + bound_penalty)
    return energy

def run_packing():
    n = 26
    best_sum_r = 0.0
    best_centers = None
    best_radii = None
    
    np.random.seed(42) # For reproducibility
    
    # Multiple restarts with hexagonal initialization and noise
    for restart in range(8):
        r_init = 0.095
        centers = np.zeros((n, 2))
        idx = 0
        y = r_init
        row = 0
        while idx < n:
            num_in_row = int((1 - 2*r_init) / (2*r_init)) + 1
            if row % 2 == 1:
                num_in_row -= 1
            if num_in_row < 1: num_in_row = 1
            if idx + num_in_row > n:
                num_in_row = n - idx
                
            x_start = (1 - (num_in_row - 1) * 2 * r_init) / 2.0
            for k in range(num_in_row):
                centers[idx] = [x_start + k * 2 * r_init, y]
                idx += 1
            y += r_init * np.sqrt(3)
            row += 1
            
        # Add noise to escape symmetric local minima
        centers += np.random.normal(0, 0.02, centers.shape)
        centers = np.clip(centers, 0.05, 0.95)
        
        x0 = np.hstack([centers.flatten(), np.full(n, r_init)])
        bounds = [(0.0, 1.0)] * (2*n) + [(1e-5, 0.5)] * n
        
        mu = 100.0
        current_vars = x0.copy()
        
        # Continuation method: gradually increase penalty weight
        for step in range(15):
            res = minimize(compute_energy, current_vars, args=(n, mu), 
                           method='L-BFGS-B', bounds=bounds,
                           options={'ftol': 1e-12, 'gtol': 1e-8, 'maxiter': 300})
            current_vars = res.x
            mu *= 3.0
            
        xs = current_vars[:n]
        ys = current_vars[n:2*n]
        rs = current_vars[2*n:]
        
        current_sum = np.sum(rs)
        if current_sum > best_sum_r:
            best_sum_r = current_sum
            best_centers = np.column_stack([xs, ys])
            best_radii = rs
            
    return best_centers, best_radii, best_sum_r
