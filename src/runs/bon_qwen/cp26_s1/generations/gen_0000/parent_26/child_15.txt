# Constants
N_CIRCLES = 26
NUM_RESTARTS = 10
MAX_ITER = 1000

def get_centers_radii(vars_arr):
    """
    Helper to unpack variables array into centers and radii.
    vars_arr shape: (78,) -> first 26 are x, next 26 are y, last 26 are r.
    """
    x = vars_arr[:N_CIRCLES]
    y = vars_arr[N_CIRCLES:2*N_CIRCLES]
    r = vars_arr[2*N_CIRCLES:]
    centers = np.column_stack((x, y))
    return centers, r

def objective(vars_arr):
    """
    Objective function: Minimize negative sum of radii.

    _, r = get_centers_radii(vars_arr)
    return -np.sum(r)
