import numpy as np
from scipy.optimize import minimize, linprog
import math
import functools


def compute_optimal_radii(centers, n):
    """Compute optimal radii for given centers using linear programming."""
    c = -np.ones(n)

    A_ub = []
    b_ub = []

    # Non-overlap constraints: r_i + r_j <= dist(i,j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = math.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            row = np.zeros(n)
            row[i] = 1
            row[j] = 1
            A_ub.append(row)
            b_ub.append(dist)

    # Boundary constraints: r_i <= min(x_i, 1-x_i, y_i, 1-y_i)
    for i in range(n):
        max_r = min(centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])
        row = np.zeros(n)
        row[i] = 1
        A_ub.append(row)
        b_ub.append(max(0, max_r))

    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)

    bounds = [(0, None) for _ in range(n)]

    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')

    if result.success:
        return result.x
    else:
        return np.full(n, 0.01)


def objective_func(centers_flat, n):
    """Negative sum of optimal radii for given centers."""
    centers_opt = centers_flat.reshape((n, 2))
    radii = compute_optimal_radii(centers_opt, n)
    return -np.sum(radii)


def hexagonal_config(n, spacing, offset_x=0.1, offset_y=0.1, cols_per_row=6):
    """Generate hexagonal lattice configuration."""
    centers = np.zeros((n, 2))
    row = 0
    col = 0
    idx = 0
    while idx < n:
        x = offset_x + col * spacing + (row % 2) * spacing / 2
        y = offset_y + row * spacing * math.sqrt(3) / 2
        if x > 0.95 and col > 0:
            col = 0
            row += 1
            continue
        centers[idx] = [x, y]
        idx += 1
        col += 1
        if col >= cols_per_row:
            col = 0
            row += 1
    return centers


def grid_config(n, spacing_x, spacing_y, offset_x=0.08, offset_y=0.08, cols=6):
    """Generate grid configuration."""
    centers = np.zeros((n, 2))
    row = 0
    col = 0
    idx = 0
    while idx < n:
        x = offset_x + col * spacing_x
        y = offset_y + row * spacing_y
        centers[idx] = [x, y]
        idx += 1
        col += 1
        if col >= cols:
            col = 0
            row += 1
    return centers


def random_config(n, low=0.05, high=0.95):
    """Generate random configuration."""
    return np.random.uniform(low, high, (n, 2))


def refine_radii(centers, radii, n):
    """Iteratively refine radii to ensure consistency."""
    for iteration in range(200):
        updated = False
        for i in range(n):
            max_r = min(centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])
            for j in range(n):
                if i != j:
                    dist = math.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                    max_r = min(max_r, dist - radii[j])
            max_r = max(0, max_r)
            if abs(max_r - radii[i]) > 1e-10:
                radii[i] = max_r
                updated = True
        if not updated:
            break
    return radii


def clamp_to_valid(centers, radii, n):
    """Clamp centers and radii to ensure boundary constraints."""
    centers = centers.copy()
    radii = radii.copy()
    for i in range(n):
        radii[i] = max(0, min(centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1]))
        centers[i, 0] = max(radii[i], min(1 - radii[i], centers[i, 0]))
        centers[i, 1] = max(radii[i], min(1 - radii[i], centers[i, 1]))
    return centers, radii


def run_packing():
    n = 26

    best_sum = 0
    best_centers = np.zeros((n, 2))
    best_radii = np.zeros(n)

    # Generate diverse initial configurations
    configs = []

    # Hexagonal lattices with various spacings
    for spacing in [0.14, 0.15, 0.16, 0.17, 0.18, 0.19, 0.20, 0.21, 0.22]:
        configs.append(hexagonal_config(n, spacing, offset_x=0.08, offset_y=0.08, cols_per_row=6))
        configs.append(hexagonal_config(n, spacing, offset_x=0.10, offset_y=0.10, cols_per_row=5))

    # Grid arrangements
    for sx, sy in [(0.16, 0.18), (0.18, 0.16), (0.17, 0.17), (0.19, 0.19), (0.20, 0.18)]:
        configs.append(grid_config(n, sx, sy, offset_x=0.06, offset_y=0.06, cols=6))
        configs.append(grid_config(n, sx, sy, offset_x=0.08, offset_y=0.08, cols=5))

    # Random configurations
    for _ in range(8):
        configs.append(random_config(n, 0.05, 0.95))

    # Optimize each configuration
    for trial, init_centers in enumerate(configs):
        # Ensure centers are in valid range
        init_centers = np.clip(init_centers, 0.01, 0.99)

        result = minimize(
            objective_func,
            init_centers.flatten(),
            args=(n,),
            method='Nelder-Mead',
            options={
                'maxiter': 30000,
                'xatol': 1e-12,
                'fatol': 1e-12,
                'adaptive': True
            }
        )

        centers_opt = result.x.reshape((n, 2))
        radii = compute_optimal_radii(centers_opt, n)

        # Refine radii
        radii = refine_radii(centers_opt, radii, n)

        # Clamp to valid
        centers_opt, radii = clamp_to_valid(centers_opt, radii, n)

        current_sum = np.sum(radii)

        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers_opt.copy()
            best_radii = radii.copy()

    # Final refinement
    best_centers, best_radii = clamp_to_valid(best_centers, best_radii, n)
    best_radii = refine_radii(best_centers, best_radii, n)
    best_centers, best_radii = clamp_to_valid(best_centers, best_radii, n)

    final_sum = np.sum(best_radii)

    return best_centers, best_radii, final_sum
