import numpy as np
from scipy.optimize import minimize


def hexagonal_init_65555(n):
    """Hexagonal arrangement with rows of 6,5,5,5,5 = 26 circles"""
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    row_counts = [6, 5, 5, 5, 5]
    idx = 0
    r_init = 0.085
    h = 2 * r_init
    v = r_init * np.sqrt(3)
    for row in range(len(row_counts)):
        count = row_counts[row]
        y = r_init + row * v
        x0 = r_init + (row % 2) * h / 2
        for col in range(count):
            x = x0 + col * h
            centers[idx] = [x, y]
            radii[idx] = r_init
            idx += 1
    max_ext = np.max(centers + radii[:, None])
    if max_ext > 0:
        scale = 0.95 / max_ext
        centers *= scale
        radii *= scale
    return centers, radii

def grid_init(n):
    """Grid-based initial arrangement"""
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    grid_n = 5
    spacing = 1.0 / (grid_n + 1)
    idx = 0
    for i in range(grid_n):
        for j in range(grid_n):
            if idx < n:
                centers[idx] = [(i + 1) * spacing, (j + 1) * spacing]
                radii[idx] = spacing / 2 * 0.9
                idx += 1
    return centers, radii

def varied_size_init(n):

    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    row_counts = [6, 5, 5, 5, 5]
    idx = 0
    r_base = 0.08
    h = 2 * r_base
    v = r_base * np.sqrt(3)
    for row in range(len(row_counts)):
        count = row_counts[row]
        y = r_base + row * v
        x0 = r_base + (row % 2) * h / 2
        for col in range(count):
            x = x0 + col * h
            centers[idx] = [x, y]
            size_factor = 0.8 + 0.4 * np.sin(col * 0.5) * np.cos(row * 0.7)
            radii[idx] = r_base * size_factor
            idx += 1
    max_ext = np.max(centers + radii[:, None])
    if max_ext > 0:
        scale = 0.95 / max_ext
        centers *= scale
        radii *= scale
    return centers, radii


def compute_violations(params, n):
    """Compute total constraint violation (squared)"""
    violations = 0.0
    for i in range(n):
        x = params[3 * i]
        y = params[3 * i + 1]
        r = params[3 * i + 2]
        if x < r:
            violations += (r - x) ** 2
        if x > 1 - r:
            violations += (x + r - 1) ** 2
        if y < r:
            violations += (r - y) ** 2
        if y > 1 - r:
            violations += (y + r - 1) ** 2
        if r < 0:
            violations += r ** 2

    for i in range(n):
        for j in range(i + 1, n):
            xi = params[3 * i]
            yi = params[3 * i + 1]
            xj = params[3 * j]
            yj = params[3 * j + 1]
            ri = params[3 * i + 2]
            rj = params[3 * j + 2]
            dx = xi - xj
            dy = yi - yj
            dist = np.sqrt(dx * dx + dy * dy)
            overlap = ri + rj - dist
            if overlap > 0:
                violations += overlap ** 2

    return violations


def penalized_objective(params, n, weight):
    """Objective: maximize sum of radii with penalty for violations"""
    return -np.sum(params[2::3]) + weight * compute_violations(params, n)


def optimize_from(x0, n, bounds):
    """Run progressive penalty optimization from initial point"""
    current = x0.copy()
    weights = [5, 20, 100, 500, 2000, 10000, 50000]
    for weight in weights:
        result = minimize(
            penalized_objective,
            current,
            args=(n, weight),
            bounds=bounds,
            method="L-BFGS-B",
            options={"maxiter": 800, "ftol": 1e-15, "gtol": 1e-12},
        )
        current = result.x
    return current
