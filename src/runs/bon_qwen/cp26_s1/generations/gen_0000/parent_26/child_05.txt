import numpy as np
from scipy.optimize import minimize

N = 26

def objective(x):
    return -np.sum(x[2*N:])

def constraints(x):
    pts = x[:2*N].reshape(N, 2)
    r = x[2*N:]
    c = []
    
    # Boundary constraints: x - r >= 0, 1 - x - r >= 0, same for y
    c.append(pts[:, 0] - r)
    c.append(1.0 - pts[:, 0] - r)
    c.append(pts[:, 1] - r)
    c.append(1.0 - pts[:, 1] - r)
    
    # Pairwise non-overlap constraints
    diff = pts[:, None, :] - pts[None, :, :]
    d2 = np.sum(diff**2, axis=2)
    r_sum = r[:, None] + r[None, :]
    idx = np.triu_indices(N, k=1)
    c.append(d2[idx] - r_sum[idx]**2)
    
    return np.concatenate(c)

def run_packing():
    np.random.seed(42)
    
    bounds = [(0, 1)] * (2*N) + [(0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_val = -np.inf
    best_x = None
    
    for trial in range(50):
        if trial == 0:
            # Structured grid initialization
            grid = np.linspace(0.15, 0.85, 5)
            pts0 = np.array([(x, y) for x in grid for y in grid]).flatten()[:2*N]
            r0 = np.ones(N) * 0.095
        else:
            # Random initialization with varied radii
            pts0 = np.random.uniform(0.1, 0.9, size=(2*N,))
            r0 = np.random.uniform(0.06, 0.16, size=N)
            
        x0 = np.concatenate([pts0, r0])
        
        res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'ftol': 1e-10, 'maxiter': 2000, 'disp': False})
                       
        val = -res.fun
        if val > best_val:
            best_val = val
            best_x = res.x.copy()
            
    if best_x is None:
        best_x = x0
        
    centers = best_x[:2*N].reshape(N, 2)
    radii = best_x[2*N:]
    
    # Safety margin to strictly satisfy validation tolerances
    radii *= 0.9999
    centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
    centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)
    
    return centers, radii, np.sum(radii)
