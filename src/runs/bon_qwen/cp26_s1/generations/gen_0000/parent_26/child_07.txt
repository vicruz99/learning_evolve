import numpy as np
from scipy.optimize import minimize


def objective(vars_, n):
    """Negative sum of radii (we minimize, so negating gives maximization)."""
    return -np.sum(vars_[2::3])


def constraints_func(vars_, n):
    """Compute all constraint values >= 0."""
    c = []
    for i in range(n):
        x, y, r = vars_[3 * i], vars_[3 * i + 1], vars_[3 * i + 2]
        c.append(x - r)
        c.append(1.0 - x - r)
        c.append(y - r)
        c.append(1.0 - y - r)
    for i in range(n):
        for j in range(i + 1, n):
            dx = vars_[3 * i] - vars_[3 * j]
            dy = vars_[3 * i + 1] - vars_[3 * j + 1]
            dist_sq = dx * dx + dy * dy
            sr = vars_[3 * i + 2] + vars_[3 * j + 2]
            c.append(dist_sq - sr * sr)
    return np.array(c)


def make_hex_config(r0, row_counts):
    """Create a hexagonal grid configuration with given row counts."""
    s = 2.0 * r0
    centers = []
    for row_idx, count in enumerate(row_counts):
        y = r0 + row_idx * s * np.sqrt(3.0) / 2.0
        x_start = 0.5 - (count - 1) * s / 2.0
        for col in range(count):
            centers.append([x_start + col * s, y])
    return centers


def make_hex_offset_config(r0, row_counts):
    """Create hexagonal grid with alternating row offsets."""
    s = 2.0 * r0
    centers = []
    for row_idx, count in enumerate(row_counts):
        y = r0 + row_idx * s * np.sqrt(3.0) / 2.0
        offset = 0.0 if row_idx % 2 == 0 else s / 2.0
        x_start = 0.5 - (count - 1) * s / 2.0 + offset
        for col in range(count):
            centers.append([x_start + col * s, y])
    return centers


def make_random_config(n, r0, seed):
    """Create a random configuration with some spacing."""
    np.random.seed(seed)
    centers = []
    for _ in range(n):
        x = np.random.uniform(r0 + 0.01, 1.0 - r0 - 0.01)
        y = np.random.uniform(r0 + 0.01, 1.0 - r0 - 0.01)
        centers.append([x, y])
    return centers


def refine_iterative(centers, radii, n, iterations=500):
    """Refine packing using iterative radius update and position adjustment."""
    for iteration in range(iterations):
        # Compute max possible radii given current positions
        new_radii = np.zeros(n)
        for i in range(n):
            max_r = 1.0
            max_r = min(max_r, centers[i, 0], 1.0 - centers[i, 0],
                        centers[i, 1], 1.0 - centers[i, 1])
            for j in range(n):
                if i == j:
                    continue
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                max_r = min(max_r, dist - radii[j])
            new_radii[i] = max(0.0001, max_r)
        radii = new_radii

        # Apply repulsion forces to move circles apart
        learning_rate = 0.005 * (1.0 - iteration / iterations)
        for i in range(n):
            fx, fy = 0.0, 0.0
            for j in range(n):
                if i == j:
                    continue
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                if dist < 1e-10:
                    continue
                min_dist = radii[i] + radii[j]
                if dist < min_dist + 0.005:
                    force = (min_dist + 0.005 - dist) / dist * learning_rate
                    fx += dx * force
                    fy += dy * force

            # Boundary forces
            margin = 0.002
            if centers[i, 0] - radii[i] < margin:
                fx += 0.01
            if centers[i, 0] + radii[i] > 1.0 - margin:
                fx -= 0.01
            if centers[i, 1] - radii[i] < margin:
                fy += 0.01
            if centers[i, 1] + radii[i] > 1.0 - margin:
                fy -= 0.01

            centers[i, 0] += fx
            centers[i, 1] += fy

            # Clamp to valid range
            centers[i, 0] = max(radii[i], min(1.0 - radii[i], centers[i, 0]))
            centers[i, 1] = max(radii[i], min(1.0 - radii[i], centers[i, 1]))

    # Final radius recalculation
    for i in range(n):
        max_r = min(centers[i, 0], 1.0 - centers[i, 0],
                    centers[i, 1], 1.0 - centers[i, 1])
        for j in range(n):
            if i == j:
                continue
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            max_r = min(max_r, dist - radii[j])
        radii[i] = max(0.0, max_r)

    return centers, radii


def repair_constraints(centers, radii, n):
    """Repair any constraint violations by slightly reducing radii."""
    for _ in range(100):
        max_violation = 0.0
        for i in range(n):
            x, y = centers[i, 0], centers[i, 1]
            r = radii[i]
            if x - r < 0:
                radii[i] = x
            if x + r > 1:
                radii[i] = 1.0 - x
            if y - r < 0:
                radii[i] = y
            if y + r > 1:
                radii[i] = 1.0 - y

        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                sr = radii[i] + radii[j]
                if dist < sr:
                    scale = dist / sr
                    radii[i] *= np.sqrt(scale)
                    radii[j] *= np.sqrt(scale)
                    max_violation = 1

        if max_violation == 0:
            break

    return centers, radii
