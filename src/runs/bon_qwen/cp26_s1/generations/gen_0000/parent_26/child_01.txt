import numpy as np
from scipy.optimize import minimize, linprog

N_CIRCLES = 26

def objective(vars):
    """Objective to minimize: negative sum of radii"""
    return -np.sum(vars[2::3])

def constraints(vars):
    """Vectorized constraint function for SLSQP"""
    pts = vars.reshape((N_CIRCLES, 3))
    con = []
    
    # Boundary and non-negativity constraints per circle
    for i in range(N_CIRCLES):
        x, y, r = pts[i]
        con.extend([x - r, 1.0 - x - r, y - r, 1.0 - y - r, r])
        
    # Pairwise non-overlap constraints
    xi = pts[:, 0, None] - pts[None, :, 0]
    yi = pts[:, 1, None] - pts[None, :, 1]
    ri = pts[:, 2, None] + pts[None, :, 2]
    
    D = np.sqrt(xi**2 + yi**2)
    triu_idx = np.triu_indices(N_CIRCLES, k=1)
    con.extend(D[triu_idx] - ri[triu_idx])
    
    return np.array(con)

def get_max_radii(centers):
    """LP to find maximal feasible radii for fixed centers"""
    n = centers.shape[0]
    D = np.sqrt(((centers[:, None, :] - centers[None, :, :])**2).sum(axis=2))
    
    c = -np.ones(n)
    n_pairs = n * (n - 1) // 2
    n_bounds = n * 4
    A_ub = np.zeros((n_pairs + n_bounds, n))
    b_ub = np.zeros(n_pairs + n_bounds)
    k = 0
    
    # Pairwise constraints: r_i + r_j <= d_ij
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[k, i] = 1.0
            A_ub[k, j] = 1.0
            b_ub[k] = D[i, j]
            k += 1
            
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        x, y = centers[i]
        A_ub[k, i] = 1.0; b_ub[k] = x; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = 1.0 - x; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = y; k += 1
        A_ub[k, i] = 1.0; b_ub[k] = 1.0 - y; k += 1
        
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return np.zeros(n)

def generate_init(method, seed=42):
    """Generate initial center positions"""
    np.random.seed(seed)
    if method == 'hex':
        pts = []
        y = 0.1
        row = 0
        while len(pts) < N_CIRCLES:
            x = 0.1 if row % 2 == 0 else 0.15
            while x < 1.0 and len(pts) < N_CIRCLES:
                pts.append([x, y])
                x += 0.15
            y += 0.15 * np.sqrt(3) / 2
            row += 1
        return np.array(pts)
    else:
        # Random initialization with margin
        return np.random.rand(N_CIRCLES, 2) * 0.8 + 0.1

def run_packing():
    best_sum = 0.0
    best_vars = None
    
    # Multi-restart optimization
    for meth in ['hex', 'random']:
        for sd in range(15):
            centers0 = generate_init(meth, sd)
            vars0 = np.zeros(N_CIRCLES * 3)
            for i in range(N_CIRCLES):
                vars0[i * 3] = centers0[i, 0]
                vars0[i * 3 + 1] = centers0[i, 1]
                vars0[i * 3 + 2] = 0.05  # Initial radius
                
            bounds = [(0, 1), (0, 1), (0, 0.5)] * N_CIRCLES
            cons = {'type': 'ineq', 'fun': constraints}
            
            try:
                res = minimize(
                    objective, 
                    vars0, 
                    method='SLSQP', 
                    bounds=bounds, 
                    constraints=cons, 
                    options={'maxiter': 800, 'ftol': 1e-13}
                )
                curr_sum = -res.fun
                if curr_sum > best_sum:
                    best_sum = curr_sum
                    best_vars = res.x
            except Exception:
                pass
                
    if best_vars is None:
        best_vars = np.zeros(N_CIRCLES * 3)
        
    # Extract centers and refine radii with LP
    centers_opt = best_vars.reshape((N_CIRCLES, 3))[:, :2]
    radii_opt = get_max_radii(centers_opt)
    
    return centers_opt, radii_opt, float(np.sum(radii_opt))
