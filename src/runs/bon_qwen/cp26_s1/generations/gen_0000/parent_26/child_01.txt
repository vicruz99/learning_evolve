import numpy as np
from scipy.optimize import minimize


def compute_objective(x, n, penalty_weight):
    """Compute objective value with penalty terms for constraints."""
    centers = x[:2*n].reshape(n, 2)
    radii = x[2*n:]

    cx = centers[:, 0]
    cy = centers[:, 1]
    r = radii

    # Maximize sum of radii (negated for minimization)
    val = -np.sum(r)

    # Boundary penalty: circles must be within [0,1]x[0,1]
    # x - r >= 0, x + r <= 1, y - r >= 0, y + r <= 1
    val += penalty_weight * np.sum(np.maximum(0, r - cx)**2)
    val += penalty_weight * np.sum(np.maximum(0, r - (1 - cx))**2)
    val += penalty_weight * np.sum(np.maximum(0, r - cy)**2)
    val += penalty_weight * np.sum(np.maximum(0, r - (1 - cy))**2)

    # Overlap penalty: dist(i,j) >= r_i + r_j
    # Vectorized computation
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dist, np.inf)  # Ignore self-distance

    overlap = r[:, np.newaxis] + r[np.newaxis, :] - dist
    val += penalty_weight * np.sum(np.maximum(0, overlap)**2)

    return val


def hexagonal_initialize(n, rng):
    """Initialize circles in a hexagonal grid pattern."""
    x0 = np.zeros(3*n)
    idx = 0

    # 6 rows with counts summing to 26
    row_counts = [5, 5, 5, 5, 5, 1]

    for row_i, count in enumerate(row_counts):
        y_base = 0.12 + row_i * 0.165
        for col in range(count):
            x_base = 0.1 + col * 0.18
            if row_i % 2 == 1:
                x_base += 0.09

            x0[3*idx] = x_base + rng.randn() * 0.02
            x0[3*idx+1] = y_base + rng.randn() * 0.02
            x0[3*idx+2] = 0.08 + rng.randn() * 0.015
            x0[3*idx+2] = max(0.02, min(0.3, x0[3*idx+2]))

            idx += 1

    return x0


def random_initialize(n, rng):
    """Initialize circles at random positions with small radii."""
    x0 = rng.rand(3*n)
    x0[2*n:] *= 0.05  # Small initial radii
    return x0


def optimize_single_run(x0, n, penalty_weight, max_iter=20000):
    """Run a single optimization from given starting point."""
    bounds = [(0, 1)] * (2*n) + [(1e-8, 0.5)] * n

    result = minimize(
        compute_objective,
        x0,
        args=(n, penalty_weight),
        method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-10}
    )
    return result


def refine_optimization(x0, n):
    """Multi-stage optimization with increasing penalty weights."""
    current_x = x0.copy()

    # Stage 1: Low penalty, coarse search
    for pw in [100, 500, 1000, 2000, 5000]:
        result = optimize_single_run(current_x, n, pw, max_iter=10000)
        current_x = result.x

    return current_x


def run_packing():
    """Main function to pack 26 circles in unit square."""
    n = 26

    best_x = None
    best_obj = np.inf

    # Try many different initial configurations
    for seed in range(60):
        rng = np.random.RandomState(seed)

        # Alternate between hexagonal and random initialization
        if seed < 40:
            x0 = hexagonal_initialize(n, rng)
        else:
            x0 = random_initialize(n, rng)

        # Multi-stage optimization
        optimized_x = refine_optimization(x0, n)

        # Final high-precision optimization
        result = optimize_single_run(optimized_x, n, 10000, max_iter=30000)

        if result.fun < best_obj:
            best_obj = result.fun
            best_x = result.x.copy()

    # Extract centers and radii
    centers = best_x[:2*n].reshape(n, 2)
    radii = best_x[2*n:]

    # Ensure radii are positive
    radii = np.maximum(radii, 1e-10)

    # Ensure centers are valid (clip if needed due to numerical issues)
    centers[:, 0] = np.clip(centers[:, 0], radii, 1 - radii)
    centers[:, 1] = np.clip(centers[:, 1], radii, 1 - radii)

    sum_radii = np.sum(radii)

    return centers, radii, sum_radii
