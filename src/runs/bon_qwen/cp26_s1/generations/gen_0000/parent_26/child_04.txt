import numpy as np
from scipy.optimize import minimize

def compute_objective(params, n):
    """
    Computes the loss function: -sum(radii) + penalty * (boundary_violations^2 + overlap_violations^2)
    """
    centers = params[:2*n].reshape(n, 2)
    radii = params[2*n:]
    
    loss = 0.0
    mu = 15000.0
    
    # Objective: maximize sum of radii -> minimize negative sum
    loss -= np.sum(radii)
    
    # Boundary penalties: r <= x, r <= 1-x, r <= y, r <= 1-y
    for i in range(n):
        x, y, r = centers[i, 0], centers[i, 1], radii[i]
        for val in [x, 1-x, y, 1-y]:
            viol = r - val
            if viol > 0:
                loss += mu * viol**2
                
    # Overlap penalties: r_i + r_j <= dist(i,j)
    for i in range(n):
        for j in range(i+1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx*dx + dy*dy) if (dx*dx + dy*dy) > 0 else 1e-12
            sum_r = radii[i] + radii[j]
            viol = sum_r - dist
            if viol > 0:
                loss += mu * viol**2
                
    return loss

def compute_gradient(params, n):
    """
    Computes the analytical gradient of the loss function.
    """
    centers = params[:2*n].reshape(n, 2)
    radii = params[2*n:]
    grad = np.zeros_like(params)
    mu = 15000.0
    
    # Gradient of objective: d(-sum(r))/dr_i = -1
    grad[2*n:] -= 1.0
    
    # Boundary gradients
    for i in range(n):
        x, y, r = centers[i, 0], centers[i, 1], radii[i]
        idx_r = 2*n + i
        idx_x = 2*i
        idx_y = 2*i + 1
        
        # val, d(val)/dx, d(val)/dy
        constraints = [(x, 1, 0), (1-x, -1, 0), (y, 0, 1), (1-y, 0, -1)]
        for val, d_val_dx, d_val_dy in constraints:
            viol = r - val
            if viol > 0:
                factor = 2 * mu * viol
                grad[idx_r] += factor * 1.0          # d(viol)/dr = 1
                grad[idx_x] += factor * (-d_val_dx)  # d(viol)/dx = -d(val)/dx
                grad[idx_y] += factor * (-d_val_dy)  # d(viol)/dy = -d(val)/dy

    # Overlap gradients
    for i in range(n):
        for j in range(i+1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist_sq = dx*dx + dy*dy
            dist = np.sqrt(dist_sq) if dist_sq > 1e-24 else 1e-12
            sum_r = radii[i] + radii[j]
            viol = sum_r - dist
            if viol > 0:
                factor = 2 * mu * viol
                grad[2*n + i] += factor
                grad[2*n + j] += factor
                
                if dist > 1e-12:
                    ndx = -dx / dist
                    ndy = -dy / dist
                    grad[2*i]     += factor * ndx
                    grad[2*i+1]   += factor * ndy
                    grad[2*j]     += factor * (-ndx)
                    grad[2*j+1]   += factor * (-ndy)
                    
    return grad

def run_packing():
    n = 26
    best_loss = np.inf
    best_params = None
    
    # Prepare bounds
    bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n
    
    # Run optimization from multiple starts to avoid local minima
    seeds = [42, 123, 777]
    for seed in seeds:
        np.random.seed(seed)
        # Base layout: 5x5 grid + 1 center
        coords = []
        for i in range(5):
            for j in range(5):
                coords.append([0.1 + i*0.2, 0.1 + j*0.2])
        coords.append([0.5, 0.5])
        
        coords = np.array(coords)
        # Add noise and shuffle
        coords += np.random.randn(n, 2) * 0.008
        indices = np.random.permutation(n)
        coords = coords[indices]
        np.clip(coords, 0.02, 0.98, out=coords)
        
        radii_init = np.ones(n) * 0.085
        params_init = np.concatenate([coords.flatten(), radii_init])
        
        res = minimize(compute_objective, params_init, args=(n,), jac=compute_gradient,
                       method='L-BFGS-B', bounds=bounds, 
                       options={'ftol': 1e-12, 'gtol': 1e-8, 'maxiter': 3000})
        
        if res.fun < best_loss:
            best_loss = res.fun
            best_params = res.x.copy()
            
    # Extract result
    opt_centers = best_params[:2*n].reshape(n, 2)
    opt_radii = best_params[2*n:]
    
    # Safety scaling to guarantee strict validity within validator tolerance
    # Check for any minor violations and shrink uniformly if necessary
    violation = False
    max_viol = 0.0
    for i in range(n):
        x, y, r = opt_centers[i, 0], opt_centers[i, 1], opt_radii[i]
        for val in [x, 1-x, y, 1-y]:
            v = r - val
            if v > max_viol: max_viol = v
        for j in range(i+1, n):
            dx = opt_centers[i, 0] - opt_centers[j, 0]
            dy = opt_centers[i, 1] - opt_centers[j, 1]
            dist = np.sqrt(dx*dx + dy*dy)
            v = (opt_radii[i] + opt_radii[j]) - dist
            if v > max_viol: max_viol = v
            
    if max_viol > 1e-10:
        # Scale radii down slightly to clear violations
        scale = 1.0 - (max_viol + 1e-9) * 0.5
        opt_radii *= max(0.0, scale)
        
    # Ensure non-negative
    opt_radii = np.maximum(opt_radii, 0.0)
    
    sum_radii = float(np.sum(opt_radii))
    return opt_centers, opt_radii, sum_radii
