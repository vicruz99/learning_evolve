import numpy as np
import scipy.optimize as opt

def generate_initial_guess(n_circles=26):
    """
    Generates an initial feasible configuration of centers and radii.
    Uses a perturbed grid layout to ensure circles are spread out.
    """
    centers = []
    radii = []
    
    # Use a grid approach. 5x5 = 25 circles.
    # We need 26. We can add one in a gap or just use a denser grid and remove some.
    # Let's try to place them on a 6x5 grid (30 spots) and pick 26, 
    # or just a simple loop.
    
    # A simple way to get a spread out initial guess:
    # Distribute points roughly uniformly.
    
    # Let's try a 5x5 grid + 1 extra point
    # Grid points:
    grid_x = [0.1, 0.3, 0.5, 0.7, 0.9]
    grid_y = [0.1, 0.3, 0.5, 0.7, 0.9]
    
    count = 0
    for x in grid_x:
        for y in grid_y:
            if count < 25:
                centers.append([x, y])
                radii.append(0.05) # Small initial radius
                count += 1
    
    # 26th circle
    # Place it in a gap, e.g., center of square (0.5, 0.5) is occupied by grid?
    # Grid has (0.5, 0.5).
    # Let's place it at (0.2, 0.2) roughly, but (0.2, 0.2) is gap between (0.1,0.1), (0.1,0.3)...
    # Distance to (0.1, 0.1) is sqrt(0.01+0.01) ~ 0.141.
    # If neighbor r=0.05, max r ~ 0.09.
    centers.append([0.2, 0.2])
    radii.append(0.04)
    
    return np.array(centers), np.array(radii)

def objective(variables, n_circles):
    """
    Objective function: minimize negative sum of radii.
    Variables layout: [x1, y1, r1, x2, y2, r2, ...]
    """
    radii = variables[2::3] # Every 3rd element starting from index 2
    return -np.sum(radii)

def constraints_builder(n_circles):
    """
    Builds the list of constraints for the optimizer.
    """
    cons = []
    
    # Boundary constraints:
    # x_i >= r_i  => x_i - r_i >= 0
    # x_i + r_i <= 1 => 1 - x_i - r_i >= 0
    # y_i >= r_i  => y_i - r_i >= 0
    # y_i + r_i <= 1 => 1 - y_i - r_i >= 0
    
    for i in range(n_circles):
        idx_x = i * 3
        idx_y = i * 3 + 1
        idx_r = i * 3 + 2
        
        # x - r >= 0
        cons.append({
            'type': 'ineq',
            'fun': lambda v, i=i: v[idx_x] - v[idx_r]
        })
        # 1 - x - r >= 0
        cons.append({
            'type': 'ineq',
            'fun': lambda v, i=i: 1.0 - v[idx_x] - v[idx_r]
        })
        # y - r >= 0
        cons.append({
            'type': 'ineq',
            'fun': lambda v, i=i: v[idx_y] - v[idx_r]
        })
        # 1 - y - r >= 0
        cons.append({
            'type': 'ineq',
            'fun': lambda v, i=i: 1.0 - v[idx_y] - v[idx_r]
        })

    # Non-overlap constraints:
    # (x_i - x_j)^2 + (y_i - y_j)^2 >= (r_i + r_j)^2
    # => (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
    
    for i in range(n_circles):
        for j in range(i + 1, n_circles):
            idx_xi = i * 3
            idx_yi = i * 3 + 1
            idx_ri = i * 3 + 2
            
            idx_xj = j * 3
            idx_yj = j * 3 + 1
            idx_rj = j * 3 + 2
            
            def non_overlap(v, i=i, j=j):
                dx = v[idx_xi] - v[idx_xj]
                dy = v[idx_yi] - v[idx_yj]
                dist_sq = dx*dx + dy*dy
                r_sum = v[idx_ri] + v[idx_rj]
                return dist_sq - r_sum*r_sum

            cons.append({
                'type': 'ineq',
                'fun': non_overlap
            })
            
    return cons

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n_circles = 26
    
    # 1. Initial Guess
    centers_init, radii_init = generate_initial_guess(n_circles)
    
    # Flatten into a single vector [x1, y1, r1, x2, y2, r2, ...]
    x0 = []
    for i in range(n_circles):
        x0.append(centers_init[i, 0])
        x0.append(centers_init[i, 1])
        x0.append(radii_init[i])
    x0 = np.array(x0)
    
    # 2. Bounds
    # x, y in [0, 1], r in [0, 0.5] (radius cannot be > 0.5)
    bounds = []
    for i in range(n_circles):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 0.5)) # r
    
    # 3. Constraints
    cons = constraints_builder(n_circles)
    
    # 4. Optimization
    # Using SLSQP
    try:
        result = opt.minimize(
            fun=objective,
            x0=x0,
            args=(n_circles,),
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 500, 'ftol': 1e-9, 'disp': False}
        )
        
        if result.success:
            optimal_vars = result.x
        else:
            # Fallback to best found even if not successful (might be close)
            optimal_vars = result.x
    except Exception:
        # In case of crash, return initial guess (though unlikely)
        optimal_vars = x0

    # 5. Extract results
    centers = np.zeros((n_circles, 2))
    radii = np.zeros(n_circles)
    
    for i in range(n_circles):
        centers[i, 0] = optimal_vars[i * 3]
        centers[i, 1] = optimal_vars[i * 3 + 1]
        radii[i] = optimal_vars[i * 3 + 2]
        
    sum_radii = np.sum(radii)
    
    # Just in case numerical issues made radii slightly negative (though bounds prevent it)
    radii = np.maximum(radii, 0.0)
    
    # Re-validate roughly to ensure sanity (optional but good practice)
    # The validation function provided in prompt is strict, so we rely on solver constraints.
    
    return centers, radii, float(sum_radii)
