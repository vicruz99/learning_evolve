def solve_lp_for_radii(centers):
    """
    Given fixed centers, solve LP to find optimal radii that maximize sum.
    Returns (sum_radii, radii).
    """

    c = -np.ones(n)

    # 1. Boundary constraints: r_i <= dist to walls
    # 2. Non-overlap constraints: r_i + r_j <= dist(i, j)

    A_ub = np.zeros((n_constraints, n))
    b_ub = np.zeros(n_constraints)

        row_idx += 1

        b_ub[row_idx] = dists[3]
