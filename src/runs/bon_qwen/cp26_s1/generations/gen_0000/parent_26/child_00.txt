import numpy as np
from scipy.optimize import minimize

N = 26

def compute_loss(vars, N):
    """
    Computes a penalized objective: -sum(radii) + penalty for overlaps and boundary violations.
    """
    centers = vars[:2*N].reshape(N, 2)
    radii = vars[2*N:]
    
    # Boundary penalties
    b_pen = 0.0
    for i in range(N):
        x, y = centers[i]
        r = radii[i]
        if r > x: b_pen += (r - x)**2
        if r > 1.0 - x: b_pen += (r - (1.0 - x))**2
        if r > y: b_pen += (r - y)**2
        if r > 1.0 - y: b_pen += (r - (1.0 - y))**2
        
    # Overlap penalties (vectorized)
    diff = centers[:, None, :] - centers[None, :, :]
    dist = np.sqrt(np.sum(diff**2, axis=2))
    sum_r = radii[:, None] + radii[None, :]
    overlap = sum_r - dist
    
    # Only penalize positive overlaps
    o_pen = np.sum(np.maximum(overlap, 0.0)**2)
    
    # High weight to strongly discourage violations
    penalty_weight = 1000.0
    return -np.sum(radii) + penalty_weight * (b_pen + o_pen)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Deterministic seed for reproducibility
    np.random.seed(42)
    
    # Initial layout: 5x5 grid + 1 extra circle in a gap
    centers = []
    for i in range(5):
        for j in range(5):
            centers.append([0.1 + i*0.2, 0.1 + j*0.2])
    centers.append([0.5, 0.95])
    centers = np.array(centers)
    
    # Break symmetry with small perturbation
    centers += np.random.uniform(-0.005, 0.005, centers.shape)
    np.clip(centers, 0.0, 1.0, out=centers)
    
    # Initial radii slightly below grid capacity to allow optimization to expand them
    radii = np.full(N, 0.09)
    x0 = np.concatenate([centers.flatten(), radii])
    
    # Variable bounds: x,y in [0,1], r in [0, 0.5]
    bounds = [(0, 1)] * (2*N) + [(0, 0.5)] * N
    
    # Optimize
    res = minimize(
        compute_loss, 
        x0, 
        args=(N,), 
        method='L-BFGS-B', 
        bounds=bounds,
        options={'maxiter': 3000, 'ftol': 1e-13, 'gtol': 1e-10}
    )
    
    if res.success:
        c_opt = res.x[:2*N].reshape(N, 2)
        r_opt = res.x[2*N:]
    else:
        # Fallback to perturbed initial guess
        c_opt = centers
        r_opt = radii
        
    # Post-processing: Strictly enforce validity by clamping radii to geometric limits
    r_orig = r_opt.copy()
    r_final = np.zeros(N)
    for i in range(N):
        max_r = 0.5
        # Boundary limits
        max_r = min(max_r, c_opt[i,0], 1.0 - c_opt[i,0], c_opt[i,1], 1.0 - c_opt[i,1])
        # Overlap limits with all other circles
        for j in range(N):
            if i == j: continue
            d = np.sqrt(np.sum((c_opt[i] - c_opt[j])**2))
            # Ensure non-negative limit
            max_r = min(max_r, max(0.0, d - r_orig[j]))
            
        r_final[i] = min(r_orig[i], max_r)
        
    total_sum = float(np.sum(r_final))
    return c_opt, r_final, total_sum
