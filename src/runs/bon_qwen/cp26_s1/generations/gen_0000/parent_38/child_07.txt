    # 1 - x - r >= 0 => r - (1-x) <= 0
    # Same for y

    # y >= r => r - y <= 0
    # y <= 1-r => r - (1-y) <= 0

def objective(params):

    Objective function to minimize.
    We want to maximize sum of radii, so we minimize -sum(radii).
    We add a penalty term for constraint violations (boundaries and overlaps).
    """

    # params structure: [x1, y1, x2, y2, ..., x26, y26, r1, r2, ..., r26]
    centers = params[:2*n].reshape((n, 2))
    radii = params[2*n:]

    # Circles must be inside [0, 1] x [0, 1].

    # x >= r  => r - x <= 0
    # x <= 1-r => r - (1-x) <= 0
    # y >= r => r - y <= 0

    penalty += np.sum(np.maximum(0, v_bottom)**2)
    
    v_top = radii - (1.0 - centers[:, 1])
    penalty += np.sum(np.maximum(0, v_top)**2)
    
    # 2. Overlap penalties
    # Distance between centers must be >= sum of radii
    # dist_ij >= r_i + r_j  =>  r_i + r_j - dist_ij <= 0
    
    # Compute pairwise distances efficiently
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]

    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
