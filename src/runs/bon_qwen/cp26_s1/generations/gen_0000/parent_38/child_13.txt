import numpy as np
from scipy.optimize import minimize


def compute_penalty(centers, radii):
    """Compute penalty for constraint violations."""
    n = len(radii)
    penalty = 0.0

    # Boundary violations
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        penalty += max(0, r - x) ** 2
        penalty += max(0, r - y) ** 2
        penalty += max(0, x + r - 1) ** 2
        penalty += max(0, y + r - 1) ** 2

    # Overlaps between circles
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            overlap = radii[i] + radii[j] - dist
            if overlap > 0:
                penalty += overlap ** 2

    return penalty


def project_to_feasible(centers, radii):
    """Project solution to ensure all constraints are satisfied."""
    n = len(radii)
    radii = radii.copy()

    # Clip radii to satisfy boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        r = max(0, r)
        r = min(r, x, y, 1 - x, 1 - y)
        radii[i] = r

    # Iteratively resolve overlaps by reducing radii
    for _ in range(20):
        changed = False
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                overlap = radii[i] + radii[j] - dist
                if overlap > 1e-10:
                    # Reduce both radii proportionally
                    total_r = radii[i] + radii[j]
                    if total_r > 0:
                        scale = dist / total_r
                        radii[i] *= scale
                        radii[j] *= scale
                    changed = True
        if not changed:
            break

    return centers, radii


def optimize_packing(centers_init, radii_init, n):
    """Optimize packing using penalty method with homotopy."""
    # Flatten parameters into single array
    x0 = np.zeros(3 * n)
    for i in range(n):
        x0[3 * i] = centers_init[i, 0]
        x0[3 * i + 1] = centers_init[i, 1]
        x0[3 * i + 2] = radii_init[i]

    # Homotopy: gradually increase penalty to enforce constraints
    penalty_params = [5, 50, 500, 5000, 50000, 500000]

    x_current = x0.copy()

    for lam in penalty_params:
        def objective(x, penalty_scale=lam):
            c = x[:2 * n].reshape(n, 2)
            r = x[2 * n:]
            return -np.sum(r) + penalty_scale * compute_penalty(c, r)

        bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n

        result = minimize(
            objective,
            x_current,
            method="L-BFGS-B",
            bounds=bounds,
            options={"maxiter": 2000, "ftol": 1e-15, "gtol": 1e-10},
        )
        x_current = result.x.copy()

    # Extract optimized parameters
    centers_opt = x_current[:2 * n].reshape(n, 2)
    radii_opt = x_current[2 * n:]

    # Ensure strict feasibility
    centers_opt, radii_opt = project_to_feasible(centers_opt, radii_opt)

    return centers_opt, radii_opt, np.sum(radii_opt)


def create_hexagonal_init(n):
    """Create hexagonal lattice initialization."""
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.09)

    # Pattern: 6, 5, 6, 5, 4 = 26 circles
    row_counts = [6, 5, 6, 5, 4]
    idx = 0

    for row in range(5):
        count = row_counts[row]
        y = 0.12 + row * 0.18
        for col in range(count):
            x = 0.12 + col * 0.16
            if row % 2 == 1:
                x += 0.08
            centers[idx] = [x, y]
            idx += 1

    return centers, radii


def create_grid_init(n):
    """Create grid-like initialization."""
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.09)

    idx = 0
    for row in range(5):
        for col in range(6):
            if idx < n:
                centers[idx] = [0.1 + col * 0.15, 0.1 + row * 0.2]
                idx += 1

    return centers, radii


def create_random_init(n, seed=42):
    """Create random initialization."""
    rng = np.random.RandomState(seed)
    centers = rng.uniform(0.1, 0.9, size=(n, 2))
    radii = np.full(n, 0.06)
    return centers, radii
