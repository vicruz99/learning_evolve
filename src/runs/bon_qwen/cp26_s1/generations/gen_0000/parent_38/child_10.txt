                dy = centers[j, 1] - centers[i, 1]

                    fx = (dx / dist) * f
                    fy = (dy / dist) * f

                    forces[i, 1] -= fy
                    forces[j, 0] += fx
                    forces[j, 1] += fy

            r = radii[i]

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Pack 26 circles in a unit square [0,1]x[0,1] to maximize the sum of radii.
    Uses a force-directed inflation algorithm to find a dense packing.
    """
    n = 26
    rng = np.random.RandomState(42)

    # Initialization: 5x5 grid perturbed + 1 in center
    # This provides a dense starting configuration
    centers = np.zeros((n, 2))

            y = 0.1 + i * 0.2 + rng.uniform(-0.01, 0.01)
            x = np.clip(x, 0.05, 0.95)
            y = np.clip(y, 0.05, 0.95)
            grid.append([x, y])
    
    # Add 26th circle at center (will be pushed out)
    grid.append([0.5, 0.5])
    
    centers = np.array(grid[:n])
    # Start with small radii to ensure initial validity
    radii[:] = 0.01 

    # Optimization parameters
    iterations = 5000
    # Initial growth rate for radii, decays over time
    growth = 5.0e-3 
    force_k = 100.0       # Strength of repulsion between circles
    boundary_k = 200.0    # Strength of repulsion from boundaries
    dt = 1.0              # Time step for integration
    
    best_sum = -1.0
    best_c = centers.copy()
    best_r = radii.copy()

    for step in range(iterations):
        # 1. Grow radii to maximize sum
        radii += growth
        if step % 200 == 0:
            growth *= 0.99 # Slow down growth rate

        forces = np.zeros_like(centers)
        overlap_error = 0.0
        
        # 2. Calculate repulsive forces from overlaps
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[j, 0] - centers[i, 0]
                dy = centers[j, 1] - centers[i, 1]
                dist_sq = dx*dx + dy*dy
                dist = np.sqrt(dist_sq) if dist_sq > 1e-12 else 1e-12
                
                min_dist = radii[i] + radii[j]
                
                if dist < min_dist:
                    overlap = min_dist - dist
                    # Repulsive force proportional to overlap depth
                    f = force_k * overlap
                    fx = (dx / dist) * f
                    fy = (dy / dist) * f
                    
                    forces[i, 0] -= fx
                    forces[i, 1] -= fy
                    forces[j, 0] += fx
                    forces[j, 1] += fy
                    overlap_error += overlap

        # 3. Calculate repulsive forces from boundaries
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            
            # Left boundary
            if x < r:
                f = boundary_k * (r - x)
                forces[i, 0] += f
                overlap_error += (r - x)
            # Right boundary
            if x > 1 - r:
                f = boundary_k * (x - (1 - r))
                forces[i, 0] -= f
                overlap_error += (x - (1 - r))
            # Bottom boundary
            if y < r:
                f = boundary_k * (r - y)
                forces[i, 1] += f
                overlap_error += (r - y)
            # Top boundary
            if y > 1 - r:
                f = boundary_k * (y - (1 - r))
                forces[i, 1] -= f
                overlap_error += (y - (1 - r))

        # 4. Update centers
        max_move = 0.02
        moves = forces * dt * 0.001
        max_m = np.max(np.abs(moves))
        if max_m > max_move:
            moves *= (max_move / max_m)
            
        centers += moves

            if is_valid(best_c, test_r):
                best_r = test_r
                lo = mid
            else:
                hi = mid
    
    return best_c, best_r, float(np.sum(best_r))
