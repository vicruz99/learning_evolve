import numpy as np
import scipy.optimize as opt

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Packs 26 circles in a unit square [0,1]x[0,1] to maximize the sum of radii.
    """
    n_circles = 26
    
    # 1. Initialization: Hexagonal Lattice
    # We want to pack 26 circles. A hexagonal lattice is efficient.
    # We estimate a spacing s such that we can fit roughly 26 circles.
    # Area per circle approx 1/26. Radius approx sqrt(1/(26*pi)) ~ 0.11.
    # Diameter ~ 0.22. Spacing s ~ 0.22.
    
    # Let's try to generate points on a hexagonal grid.
    # x_ij = i * s + (j % 2) * s / 2
    # y_ij = j * s * sqrt(3) / 2
    
    s = 0.22  # Initial guess for spacing
    centers_init = []
    
    # Generate a sufficient number of points and pick those inside [0,1]x[0,1]
    # We need at least 26 points.
    # Let's iterate over a grid large enough to cover the square.
    rows = int(1 / (s * np.sqrt(3) / 2)) + 5
    cols = int(1 / s) + 5
    
    for j in range(rows):
        for i in range(cols):
            x = i * s + (j % 2) * s / 2
            y = j * s * np.sqrt(3) / 2
            if 0 <= x <= 1 and 0 <= y <= 1:
                centers_init.append([x, y])
                
    # Sort points by distance to center (0.5, 0.5) to pick the most central ones
    # or just take the first 26. Central ones usually pack better.
    center_point = np.array([0.5, 0.5])
    centers_init_arr = np.array(centers_init)
    dists_to_center = np.linalg.norm(centers_init_arr - center_point, axis=1)
    sorted_indices = np.argsort(dists_to_center)
    
    # Select 26 points
    if len(centers_init_arr) >= n_circles:
        initial_centers = centers_init_arr[sorted_indices[:n_circles]]
    else:
        # Fallback to random if grid failed (unlikely with s=0.22)
        np.random.seed(42)
        initial_centers = np.random.rand(n_circles, 2)

    # Initial radii: small value to ensure valid start
    initial_radii = np.ones(n_circles) * 0.01
    
    # 2. Optimization
    # Variables: [x1, y1, ..., x26, y26, r1, ..., r26]
    # Total 3 * 26 = 78 variables.
    
    def objective(vars):
        # vars shape (78,)
        # First 52 are centers, last 26 are radii
        radii = vars[52:]
        return -np.sum(radii) # Minimize negative sum -> Maximize sum

    # Constraints
    constraints = []
    
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    # Also x_i, y_i in [0,1] handled by bounds, but r_i <= x_i is non-linear relative to x
    # Actually, simpler: x_i - r_i >= 0 => 0 <= x_i - r_i
    # We can add linear constraints for boundaries if we treat x, y, r as variables.
    # But x, y are variables.
    # Constraint: x_i - r_i >= 0
    # In scipy, inequality constraints are func(x) >= 0.
    
    for i in range(n_circles):
        x_idx = 2 * i
        y_idx = 2 * i + 1
        r_idx = 52 + i
        
        # x >= r
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, ix=x_idx, ir=r_idx: v[ix] - v[ir]
        })
        # 1 - x >= r  =>  1 - x - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, ix=x_idx, ir=r_idx: 1.0 - v[ix] - v[ir]
        })
        # y >= r
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, iy=y_idx, ir=r_idx: v[iy] - v[ir]
        })
        # 1 - y >= r
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, iy=y_idx, ir=r_idx: 1.0 - v[iy] - v[ir]
        })
        # r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, ir=r_idx: v[ir]
        })

    # Non-overlap constraints: dist(i,j)^2 >= (r_i + r_j)^2
    # dist^2 - (r_i + r_j)^2 >= 0
    for i in range(n_circles):
        for j in range(i + 1, n_circles):
            xi, yi = 2*i, 2*i+1
            xj, yj = 2*j, 2*j+1
            ri, rj = 52+i, 52+j
            
            def overlap_con(v, ix=xi, iy=yi, jx=xj, jy=yj, ir=ri, jr=rj):
                dx = v[ix] - v[jx]
                dy = v[iy] - v[jy]
                dist_sq = dx*dx + dy*dy
                r_sum = v[ir] + v[jr]
                return dist_sq - r_sum*r_sum
            
            constraints.append({
                'type': 'ineq',
                'fun': overlap_con
            })

    # Bounds for variables
    # x, y in [0, 1]
    # r in [0, 0.5]
    bounds = []
    for i in range(2 * n_circles):
        bounds.append((0.0, 1.0))
    for i in range(n_circles):
        bounds.append((0.0, 0.5))

    # Initial guess vector
    x0 = np.concatenate([initial_centers.flatten(), initial_radii])

    # Run optimization
    # SLSQP is good for non-linear constraints
    try:
        res = opt.minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints,
                           options={'maxiter': 1000, 'ftol': 1e-9})
        final_vars = res.x
    except Exception as e:
        # Fallback if optimization fails
        final_vars = x0

    # Extract results
    centers = final_vars[:52].reshape(n_circles, 2)
    radii = final_vars[52:]
    
    # Clamp radii to non-negative just in case
    radii = np.maximum(radii, 0)
    
    # Ensure centers are valid (inside [0,1])
    centers = np.clip(centers, 0, 1)
    
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
