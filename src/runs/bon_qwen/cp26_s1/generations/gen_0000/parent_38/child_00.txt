def get_optimal_radii(centers):
    """
    Given fixed centers, solve the LP to maximize sum of radii.
    Variables: r_0, ..., r_25
    Maximize sum(r_i)
    Subject to:
      r_i + r_j <= distance(c_i, c_j) for all i < j
      r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
      r_i >= 0
    """
    n = centers.shape[0]
    
    # Objective: minimize -sum(r) -> c = [-1, -1, ..., -1]
    c = -np.ones(n)
    
    # Constraints matrix A_ub and vector b_ub for A_ub @ r <= b_ub
    # We will build lists of rows and then stack them
    A_rows = []
    b_rows = []
    
    # 1. Pairwise distance constraints: r_i + r_j <= dist_ij
    # Precompute distances
    # dist matrix
    dists = np.zeros((n, n))
    for i in range(n):
        for j in range(i + 1, n):
            d = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            dists[i, j] = d
            dists[j, i] = d
            
            # Constraint: 1*r_i + 1*r_j <= d
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_rows.append(row)
            b_rows.append(d)
            
    # 2. Boundary constraints
    for i in range(n):
        x, y = centers[i]

        row = np.zeros(n)
        row[i] = 1.0
        A_rows.append(row)
