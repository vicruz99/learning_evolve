def compute_energy_grad(centers, r):

    energy = 0.0

    dist_sq = np.sum(diff**2, axis=2)
    dist = np.sqrt(dist_sq + 1e-12)
    np.fill_diagonal(dist, np.inf)
    
    overlap = np.maximum(0, 2*r - dist)
    energy = np.sum(overlap**2)
    
    mask = overlap > 0

        dist_ij = dist[i_idx, j_idx]
        ov_ij = overlap[i_idx, j_idx]

        np.add.at(grad, i_idx, factor[:, None] * diff_ij)
        np.add.at(grad, j_idx, -factor[:, None] * diff_ij)

def optimize_pack(centers, r, lr=0.04, max_iter=3000, tol=1e-10):
    """Momentum-based gradient descent to minimize overlap energy."""
    centers = centers.copy().reshape(-1, 2)
    v = np.zeros_like(centers)
    mu = 0.85
    for t in range(max_iter):
        e, g = compute_energy_grad(centers, r)

            return centers, e
        current_lr = lr / (1 + t * 0.0008)
        v = mu * v - current_lr * g
        centers += v
    return centers, compute_energy_grad(centers, r)[0]
