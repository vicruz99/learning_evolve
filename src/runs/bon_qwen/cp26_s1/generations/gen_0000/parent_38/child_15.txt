from scipy.optimize import minimize

def run_packing():

    Solves the circle packing problem for 26 circles in a unit square.
    Uses a hexagonal lattice initialization followed by SLSQP optimization 
    to maximize the sum of radii.
    """

    # Construct a hexagonal pattern to provide a good initial guess
    centers = []
    r_init = 0.05 # Initial radius to avoid immediate overlap

    dy = r_init * np.sqrt(3)

    y_start = r_init
    
    current_y = y_start

        offset = dx / 2 if i % 2 == 1 else 0
        
        # Calculate total width of this row to center it if needed, 
        # but for now just placing them. We will let the optimizer move them.
        for j in range(count):
            x = x_start + j * dx + offset
            y = current_y
            centers.append([x, y])
        
        current_y += dy
    
    centers = np.array(centers)
    radii = np.full(n, r_init)

        return -sum(vars[3 * i + 2] for i in range(n))

                return dist_sq - r_sum_sq
            
            constraints.append({'type': 'ineq', 'fun': non_overlap})

        # r in [0, 1] (loose upper bound)
        bounds.append((0, 1))

    # SLSQP is suitable for constrained non-linear optimization
