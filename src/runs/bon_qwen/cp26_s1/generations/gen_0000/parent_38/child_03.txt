from scipy.optimize import minimize

def compute_penalty_and_obj(vars, n):
    """Compute objective and penalty for the packing problem."""
    centers = vars[:2*n].reshape(n, 2)
    radii = vars[2*n:]
    radii = np.maximum(radii, 1e-6)
    
    penalty = 0.0
    obj = -np.sum(radii)
    
    # Boundary penalties
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < 0:
            penalty += 1e4 * (x - r)**2
        if x + r > 1:
            penalty += 1e4 * (x + r - 1)**2
        if y - r < 0:
            penalty += 1e4 * (y - r)**2
        if y + r > 1:
            penalty += 1e4 * (y + r - 1)**2
            
    # Overlap penalties
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            d = np.sqrt(dx*dx + dy*dy + 1e-12)
            if d < radii[i] + radii[j]:
                diff = radii[i] + radii[j] - d
                penalty += 1e4 * diff**2
                
    return obj + penalty

def objective_wrapper(vars, n):

    return compute_penalty_and_obj(vars, n)

    centers = np.array(centers[:n])

    return np.concatenate([centers.flatten(), radii])

    radii = radii.copy()
