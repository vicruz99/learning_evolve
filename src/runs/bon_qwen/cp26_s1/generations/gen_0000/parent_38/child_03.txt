import numpy as np
from scipy.optimize import minimize

def calculate_loss(positions, n):
    """
    Calculate the loss function for the optimization.
    Positions vector format: [x1, y1, r1, x2, y2, r2, ...]
    """
    centers = positions[:2*n].reshape(n, 2)
    radii = positions[2*n:]
    
    loss = -np.sum(radii)  # We want to maximize sum of radii
    
    penalty_weight = 100.0
    penalty = 0.0
    
    # Boundary constraints
    # x - r >= 0  => r - x <= 0
    # x + r <= 1  => x + r - 1 <= 0
    # Same for y
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        
        # Left boundary
        if r - x > 0:
            penalty += (r - x)**2
        # Right boundary
        if x + r - 1 > 0:
            penalty += (x + r - 1)**2
        # Bottom boundary
        if r - y > 0:
            penalty += (r - y)**2
        # Top boundary
        if y + r - 1 > 0:
            penalty += (y + r - 1)**2
            
    # Overlap constraints
    # dist(i, j) >= r_i + r_j  => r_i + r_j - dist(i, j) <= 0
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            overlap = radii[i] + radii[j] - dist
            if overlap > 0:
                penalty += overlap**2
                
    return loss + penalty_weight * penalty

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        return False
    if np.isnan(radii).any():
        return False

    # Check if radii are nonnegative
    for i in range(n):
        if radii[i] < -1e-9: # Allow tiny negative due to float precision
             return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-9 or x + r > 1 + 1e-9 or y - r < -1e-9 or y + r > 1 + 1e-9:
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-9:
                return False

    return True
