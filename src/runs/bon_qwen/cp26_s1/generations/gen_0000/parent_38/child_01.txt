def init_hexagonal(n):
    """Initialize circles in a hexagonal pattern"""
    centers = []
    r = 0.075
    y = r
    row = 0
    while len(centers) < n:

        x = x_start
        while x <= 1 - r and len(centers) < n:
            centers.append([x, y])
            x += 2 * r
        y += r * math.sqrt(3)
        row += 1

    centers = np.array(centers[:n])
    # Add small random perturbation
    centers += np.random.normal(0, 0.005, centers.shape)
    centers = np.clip(centers, 0.01, 0.99)
    radii = np.full(n, 0.05)

    return centers, radii


def init_grid(n):

    rows = 5

    x_spacing = 0.9 / (cols - 1)
    y_spacing = 0.9 / (rows - 1)

                y = 0.05 + i * y_spacing

    centers = np.array(centers[:n])
    radii = np.full(n, 0.05)
    return centers, radii


def init_random_scatter(n):
    """Initialize circles at random positions"""
    centers = np.random.uniform(0.1, 0.9, (n, 2))
    radii = np.full(n, 0.04)
    return centers, radii


def apply_repulsion(centers, radii, n, lr=0.01, iters=10):
    """Apply repulsion forces to resolve overlaps"""
    centers = centers.copy()

    for _ in range(iters):
        forces = np.zeros_like(centers)

        # Circle-circle repulsion
        for i in range(n):
            for j in range(i + 1, n):
                diff = centers[i] - centers[j]
                dist_sq = np.dot(diff, diff)
                dist = math.sqrt(dist_sq) if dist_sq > 1e-16 else 1e-10
                min_dist = radii[i] + radii[j]

                if dist < min_dist:
                    overlap = min_dist - dist
                    # Force proportional to overlap, inversely proportional to distance
                    force_mag = overlap * 10.0
                    if dist > 1e-10:
                        forces[i] += force_mag * diff / dist
                        forces[j] -= force_mag * diff / dist

        # Boundary repulsion (stronger to keep circles well inside)
        for i in range(n):
            x, y = centers[i]
            r = radii[i]

            if x - r < 0:
                forces[i, 0] += (r - x) * 50
            if x + r > 1:
                forces[i, 0] -= (x + r - 1) * 50
            if y - r < 0:
                forces[i, 1] += (r - y) * 50
            if y + r > 1:
                forces[i, 1] -= (y + r - 1) * 50

        centers += lr * forces
        centers = np.clip(centers, 1e-6, 1 - 1e-6)

    return centers


def final_refinement(centers, radii, n):

def compute_max_valid_radii(centers, n):
    """Given fixed centers, compute maximum valid radii via iterative relaxation"""
    radii = np.zeros(n)

    # Initialize with boundary limits
    for i in range(n):
        x, y = centers[i]
        radii[i] = min(x, 1 - x, y, 1 - y)

    # Iteratively enforce pairwise constraints
    for _ in range(200):
        old_sum = np.sum(radii)
        for i in range(n):
            x, y = centers[i]
            r_max = min(x, 1 - x, y, 1 - y)
            for j in range(n):
                if i != j:
                    dist = np.linalg.norm(centers[i] - centers[j])
                    r_max = min(r_max, dist - radii[j])
            radii[i] = max(0, r_max)
        new_sum = np.sum(radii)
        if abs(new_sum - old_sum) < 1e-12:
            break

    return radii


def obj_func(x, n):
    """Objective: maximize sum of radii (negate for minimization)"""
    radii = x[2::3]
    penalty = 0.0
    weight = 1000.0

    centers = np.column_stack((x[0::3], x[1::3]))

    # Boundary penalty
    for i in range(n):
        x_coord, y_coord = centers[i]
        r = radii[i]
        if x_coord - r < 0:
            penalty += weight * (r - x_coord) ** 2
        if x_coord + r > 1:
            penalty += weight * (x_coord + r - 1) ** 2
        if y_coord - r < 0:
            penalty += weight * (r - y_coord) ** 2
        if y_coord + r > 1:
            penalty += weight * (y_coord + r - 1) ** 2

    # Overlap penalty
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            min_dist = radii[i] + radii[j]
            if dist < min_dist:
                penalty += weight * (min_dist - dist) ** 2

    return -np.sum(radii) + penalty


def is_valid(centers, radii, n):
    """Check if packing is valid"""
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-10 or x + r > 1 + 1e-10 or y - r < -1e-10 or y + r > 1 + 1e-10:
            return False
        if r < 0:
            return False

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            if dist < radii[i] + radii[j] - 1e-10:
                return False

    return True
