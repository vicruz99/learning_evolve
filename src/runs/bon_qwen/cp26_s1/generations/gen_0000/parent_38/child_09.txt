import numpy as np
from scipy.optimize import minimize

def run_packing():
    """
    Optimizes the packing of 26 circles in a unit square to maximize sum of radii.
    """
    n_circles = 26
    best_sum_radii = -1.0
    best_centers = None
    best_radii = None
    
    # Penalty weight multiplier for constraints
    # We will increase this dynamically or use a high static value
    penalty_weight = 1000.0 
    
    # Number of restarts to find a good global optimum
    n_restarts = 50

    for restart in range(n_restarts):
        # 1. Initialization
        # Try to initialize in a grid-like structure to start dense
        # A 5x6 grid has 30 spots, we pick 26. 
        # Or just random uniform in safe zone.
        
        rng = np.random.default_rng(restart + 123)
        
        # Random initialization in the center to avoid boundary issues initially
        # but radii will be small.
        centers = rng.uniform(0.2, 0.8, size=(n_circles, 2))
        radii = np.full(n_circles, 0.05)
        
        # Flatten to 1D array for optimizer: [x1, y1, r1, x2, y2, r2, ...]
        x0 = np.concatenate([centers.flatten(), radii])
        
        # 2. Define Objective Function with Penalties
        def objective(vars):
            # Unflatten variables
            centers_opt = vars[:2 * n_circles].reshape(n_circles, 2)
            radii_opt = vars[2 * n_circles:]
            
            # Primary Objective: Maximize sum of radii => Minimize negative sum
            cost = -np.sum(radii_opt)
            
            # Penalties
            p = 0.0
            
            # Boundary Constraints: r <= x <= 1-r, r <= y <= 1-r
            # Equivalent to: r - x <= 0, x + r - 1 <= 0, r - y <= 0, y + r - 1 <= 0
            for i in range(n_circles):
                x, y = centers_opt[i]
                r = radii_opt[i]
                
                # Left
                if r > x:
                    p += penalty_weight * (r - x)**2
                # Right
                if x + r > 1.0:
                    p += penalty_weight * (x + r - 1.0)**2
                # Bottom
                if r > y:
                    p += penalty_weight * (r - y)**2
                # Top
                if y + r > 1.0:
                    p += penalty_weight * (y + r - 1.0)**2
            
            # Overlap Constraints: dist >= r_i + r_j
            # dist^2 >= (r_i + r_j)^2
            # Penalty if dist < r_i + r_j
            # Using squared penalty for smoothness
            
            # Optimization: only check pairs if we want speed, but N=26 is small enough for all pairs
            # 26*25/2 = 325 pairs. Very fast.
            
            # Vectorized overlap check
            # Centers shape (N, 2)
            # Radii shape (N,)
            
            # Compute pairwise distances
            # diff = centers_opt[:, np.newaxis, :] - centers_opt[np.newaxis, :, :] # (N, N, 2)
            # dists = np.sqrt(np.sum(diff**2, axis=2)) # (N, N)
            
            # To avoid self-distance and double counting, we can use upper triangle
            # But simple loop is robust and clear for N=26
            
            for i in range(n_circles):
                r_i = radii_opt[i]
                c_i = centers_opt[i]
                for j in range(i + 1, n_circles):
                    r_j = radii_opt[j]
                    c_j = centers_opt[j]
                    
                    dist = np.sqrt(np.sum((c_i - c_j)**2))
                    sum_r = r_i + r_j
                    
                    if dist < sum_r:
                        overlap = sum_r - dist
                        p += penalty_weight * overlap**2
            
            return cost + p

        # 3. Bounds
        # x, y in [0, 1], r in [0, 0.5] (r can't be > 0.5)
        bounds = []
        for i in range(n_circles):
            bounds.extend([(0, 1), (0, 1)]) # x, y
            bounds.append((0, 0.5))         # r

        # 4. Optimization
        # Use L-BFGS-B for box constraints
        res = minimize(objective, x0, method='L-BFGS-B', bounds=bounds, 
                       options={'ftol': 1e-9, 'gtol': 1e-6, 'maxiter': 5000})
        
        # 5. Extract results
        centers_res = res.x[:2 * n_circles].reshape(n_circles, 2)
        radii_res = res.x[2 * n_circles:]
        
        # Filter out negative radii or tiny ones due to noise, though bounds prevent this
        # Ensure radii are non-negative (clipping just in case)
        radii_res = np.maximum(radii_res, 0.0)
        
        current_sum = np.sum(radii_res)
        
        # Check validity roughly (penalty should be low)
        # The objective value is -sum + penalty. 
        # If penalty is high, sum is inflated.
        # We want to maximize true sum of a valid packing.
        # Let's check if the configuration is actually valid with a small tolerance.
        
        is_valid = True
        # Check boundaries
        for i in range(n_circles):
            x, y = centers_res[i]
            r = radii_res[i]
            if x - r < -1e-5 or x + r > 1 + 1e-5 or y - r < -1e-5 or y + r > 1 + 1e-5:
                is_valid = False
                break
        
        if is_valid:
            # Check overlaps
            for i in range(n_circles):
                for j in range(i + 1, n_circles):
                    dist = np.sqrt(np.sum((centers_res[i] - centers_res[j])**2))
                    if dist < radii_res[i] + radii_res[j] - 1e-5:
                        is_valid = False
                        break
                if not is_valid:
                    break
        
        if is_valid and current_sum > best_sum_radii:
            best_sum_radii = current_sum
            best_centers = centers_res.copy()
            best_radii = radii_res.copy()

    # If no valid packing found (unlikely with this approach), return empty/zero
    if best_centers is None:
        return np.zeros((n_circles, 2)), np.zeros(n_circles), 0.0

    return best_centers, best_radii, best_sum_radii
