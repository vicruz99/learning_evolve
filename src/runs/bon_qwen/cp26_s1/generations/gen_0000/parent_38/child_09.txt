import numpy as np
from scipy.optimize import minimize

def left_constraint(x, i):
    return x[3*i] - x[3*i+2]

def right_constraint(x, i):
    return 1.0 - x[3*i] - x[3*i+2]

def bottom_constraint(x, i):
    return x[3*i+1] - x[3*i+2]

def top_constraint(x, i):
    return 1.0 - x[3*i+1] - x[3*i+2]

def overlap_constraint(x, i, j):
    dx = x[3*i] - x[3*j]
    dy = x[3*i+1] - x[3*j+1]
    dist = np.sqrt(dx*dx + dy*dy)
    return dist - x[3*i+2] - x[3*j+2]

def objective(x):
    radii = x[2::3]
    return -np.sum(radii)

def force_directed_optimize(centers, radii, n, iterations=2000):
    """Force-directed pre-optimization to find a good starting configuration."""
    for iteration in range(iterations):
        lr = 0.002 / (1.0 + iteration * 0.0005)
        
        # Grow radii gradually
        for i in range(n):
            max_r = min(centers[i, 0], 1.0 - centers[i, 0],
                       centers[i, 1], 1.0 - centers[i, 1])
            for j in range(n):
                if i != j:
                    d = np.sqrt(np.sum((centers[i] - centers[j])**2))
                    max_r = min(max_r, d - radii[j])
            radii[i] = min(radii[i] * 1.0003, max(max_r * 0.999, 0.0))
        
        # Compute repulsion forces
        forces = np.zeros((n, 2))
        for i in range(n):
            # Boundary repulsion
            for d in [0, 1]:
                gap_left = centers[i, d] - radii[i]
                gap_right = 1.0 - centers[i, d] - radii[i]
                
                if gap_left < 0.04:
                    forces[i, d] += 2.0 / (gap_left + 0.01)
                if gap_right < 0.04:
                    forces[i, d] -= 2.0 / (gap_right + 0.01)
            
            # Circle repulsion
            for j in range(i + 1, n):
                diff = centers[i] - centers[j]
                dist = np.sqrt(np.sum(diff**2))
                min_dist = radii[i] + radii[j]
                
                if dist < min_dist:
                    overlap = min_dist - dist
                    force_mag = overlap * 15.0 / (dist + 0.001)
                    direction = diff / (dist + 0.001)
                    forces[i] += force_mag * direction
                    forces[j] -= force_mag * direction
        
        centers += lr * forces
        
        # Enforce boundary constraints
        for i in range(n):
            centers[i, 0] = np.clip(centers[i, 0], radii[i], 1.0 - radii[i])
            centers[i, 1] = np.clip(centers[i, 1], radii[i], 1.0 - radii[i])
    
    return centers, radii

def initialize_hexagonal(n, spacing=0.17, y_spacing=None):
    """Initialize circles in a hexagonal pattern."""
    if y_spacing is None:
        y_spacing = spacing * np.sqrt(3) / 2
    
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.07
    
    idx = 0
    for row in range(6):
        y = 0.05 + row * y_spacing
        offset = spacing / 2.0 if row % 2 == 1 else 0.0
        for col in range(7):
            if idx >= n:
                break
            x = offset + col * spacing
            if 0 <= x <= 1:
                centers[idx] = [x, y]
                idx += 1
    
    while idx < n:
        centers[idx] = [0.5, 0.5]
        idx += 1
    
    return centers, radii

def initialize_grid(n, spacing=0.16):
    """Initialize circles in a grid pattern."""
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.07
    
    idx = 0
    for row in range(6):
        for col in range(7):
            if idx >= n:
                break
            x = spacing * col + spacing / 2
            y = spacing * row + spacing / 2
            if x <= 1.0 and y <= 1.0:
                centers[idx] = [x, y]
                idx += 1
    
    while idx < n:
        centers[idx] = [0.5, 0.5]
        idx += 1
    
    return centers, radii

def validate_and_fix(centers, radii, n):
    """Validate and fix any constraint violations."""
    # Fix boundary violations
    for i in range(n):
        r = radii[i]
        centers[i, 0] = np.clip(centers[i, 0], r, 1.0 - r)
        centers[i, 1] = np.clip(centers[i, 1], r, 1.0 - r)
    
    # Fix overlaps by reducing radii
    for iteration in range(100):
        max_overlap = 0
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                overlap = radii[i] + radii[j] - dist
                if overlap > max_overlap:
                    max_overlap = overlap
        
        if max_overlap < 1e-10:
            break
        
        # Reduce both radii slightly
        for i in range(n):
            radii[i] = max(radii[i] * 0.999, 0.001)
    
    return centers, radii
