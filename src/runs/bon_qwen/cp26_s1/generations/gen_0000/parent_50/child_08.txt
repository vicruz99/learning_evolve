import numpy as np
from scipy.optimize import minimize

N = 26

def constraint_func(v):
    """Computes all inequality constraints as a single array."""
    vals = []
    for i in range(N):
        # x >= r
        vals.append(v[2*i] - v[2*N + i])
        # x <= 1 - r
        vals.append(1.0 - (v[2*i] + v[2*N + i]))
        # y >= r
        vals.append(v[2*i+1] - v[2*N + i])
        # y <= 1 - r
        vals.append(1.0 - (v[2*i+1] + v[2*N + i]))
        
    for i in range(N):
        for j in range(i + 1, N):
            dx = v[2*i] - v[2*j]
            dy = v[2*i+1] - v[2*j+1]
            dr = v[2*N+i] + v[2*N+j]
            vals.append(dx*dx + dy*dy - dr*dr)
    return np.array(vals)

def objective_func(v):
    """Negative sum of radii for minimization."""
    return -np.sum(v[2*N:])

def run_packing():
    bounds = [(0.0, 1.0)] * (2 * N) + [(1e-6, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraint_func}
    
    best_v = None
    best_val = -np.inf
    
    # Generate diverse starting points
    starts = []
    
    # 1. Random starts with different seeds
    for seed in range(4):
        rng = np.random.RandomState(seed)
        x = rng.rand(N, 2) * 0.8 + 0.1
        r = np.full(N, 0.05)
        starts.append(np.concatenate([x.flatten(), r]))
        
    # 2. Square grid start (5x5 grid + 1 center)
    grid_x = np.tile(np.linspace(0.1, 0.9, 5), 5)
    grid_y = np.repeat(np.linspace(0.1, 0.9, 5), 5)
    grid_pts = np.column_stack((grid_x, grid_y))
    grid_pts = np.vstack([grid_pts, [0.5, 0.5]])
    grid_pts = grid_pts[:N]
    r_grid = np.full(N, 0.08)
    starts.append(np.concatenate([grid_pts.flatten(), r_grid]))
    
    # 3. Hexagonal staggered start
    hex_pts = []
    r_init = 0.09
    y = r_init
    counts = [6, 5, 6, 5, 4]
    for c in counts:
        x = r_init + (1.0 - 2.0 * r_init) * np.linspace(0, 1, c)
        if len(hex_pts) % 2 == 1:
            spacing = (1.0 - 2.0 * r_init) / (c - 1) if c > 1 else 0
            x = x + spacing / 2
        hex_pts.extend(np.column_stack((x, np.full(c, y))))
        y += r_init * np.sqrt(3)
    hex_pts = np.array(hex_pts[:N])
    r_hex = np.full(N, r_init)
    starts.append(np.concatenate([hex_pts.flatten(), r_hex]))
    
    # Run optimization from each start
    for v0 in starts:
        try:
            res = minimize(objective_func, v0, method='SLSQP', bounds=bounds, 
                          constraints=cons, options={'maxiter': 2000, 'ftol': 1e-10})
            if res.fun < best_val:
                best_val = res.fun
                best_v = res.x
        except Exception:
            continue
            
    if best_v is None:
        # Fallback to valid grid configuration
        centers = grid_pts
        radii = np.full(N, 0.05)
        return centers, radii, np.sum(radii)
        
    centers = np.reshape(best_v[:2*N], (N, 2))
    radii = best_v[2*N:]
    
    # Ensure strict validity within tolerance
    radii = np.clip(radii, 1e-9, 0.5)
    centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
    centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)
    
    # Slight shrink to guarantee non-overlap and boundary constraints pass numerical checks
    radii *= 0.9999999
    
    return centers, radii, np.sum(radii)
