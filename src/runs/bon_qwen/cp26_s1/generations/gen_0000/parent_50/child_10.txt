import numpy as np
from scipy.optimize import minimize

def compute_loss(vars, n, mu):
    """
    Computes the objective: -sum(radii) + penalty for constraint violations.
    vars: flattened array [x1, y1, ..., xn, yn, r1, ..., rn]
    n: number of circles
    mu: penalty weight
    """
    centers = vars[:2 * n].reshape((n, 2))
    radii = vars[2 * n:]
    
    # Objective: minimize negative sum of radii
    loss = -np.sum(radii)
    
    # Boundary penalties: ensure circles stay inside [0,1]x[0,1]
    pen = 0.0
    pen += np.sum(np.maximum(0, radii - centers[:, 0])**2)          # left
    pen += np.sum(np.maximum(0, radii - (1 - centers[:, 0]))**2)    # right
    pen += np.sum(np.maximum(0, radii - centers[:, 1])**2)          # bottom
    pen += np.sum(np.maximum(0, radii - (1 - centers[:, 1]))**2)    # top
    
    # Overlap penalties: ensure dist(i,j) >= r_i + r_j
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    r_sums = radii[:, np.newaxis] + radii[np.newaxis, :]
    
    i_idx, j_idx = np.triu_indices(n, k=1)
    violations = np.maximum(0, r_sums[i_idx, j_idx] - dists[i_idx, j_idx])
    pen += np.sum(violations**2)
    
    return loss + mu * pen

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    best_vars = None
    best_loss = float('inf')
    
    # Bounds: coordinates in [0,1], radii in [0, 0.5]
    bounds = [(0, 1)] * (2 * n) + [(0, 0.5)] * n
    
    # Run multiple trials with different initializations
    for trial in range(20):
        # Generate initialization
        if trial % 3 == 0:

            grid = np.array([(i % 5) / 5.0 + 0.1, (i // 5) / 5.0 + 0.1] for i in range(25))
            grid = np.vstack([grid, [0.5, 0.5]])

        elif trial % 3 == 1:
            # Random positions, moderate radii
            c = np.random.rand(2 * n)
            r = np.random.rand(n) * 0.06 + 0.04
        else:
            # Hexagonal-ish layout
            c = np.zeros((2 * n))
            r = np.ones(n) * 0.085
            idx = 0
            for row in range(5):
                y = 0.12 + row * 0.175

                        c[2*idx] = x + np.random.randn()*0.01
                        c[2*idx+1] = y + np.random.randn()*0.01
                        r[idx] += np.random.randn()*0.005

            # Fill remaining if any
            while idx < 26:
                c[2*idx] = 0.5 + np.random.randn()*0.1
                c[2*idx+1] = 0.5 + np.random.randn()*0.1
                r[idx] = 0.05 + np.random.rand()*0.02
                idx += 1
            
        vars_init = np.concatenate([c, r])
        
        # Multi-stage penalty optimization
        current_vars = vars_init
        for mu in [50.0, 500.0, 3000.0]:
            res = minimize(

                current_vars,
                args=(n, mu),
                method='L-BFGS-B',

    radii = best_vars[2 * n:]

    centers = np.clip(centers, 0.0, 1.0)
    radii = np.maximum(radii, 0.0)
    
    return centers, radii, float(np.sum(radii))
