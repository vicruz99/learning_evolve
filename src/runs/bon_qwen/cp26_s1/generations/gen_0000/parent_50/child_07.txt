import numpy as np
from scipy.optimize import minimize
import math

def compute_objective(vars, n, mu):
    centers = vars[:2*n].reshape((n, 2))
    radii = vars[2*n:]
    
    sum_r = np.sum(radii)
    penalty = 0.0
    
    # Circle-circle overlap penalty
    for i in range(n):
        for j in range(i+1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = math.hypot(dx, dy)
            overlap = radii[i] + radii[j] - dist
            if overlap > 0:
                penalty += mu * overlap**2
                
    # Boundary containment penalty
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < 0: penalty += mu * (x - r)**2
        if x + r > 1: penalty += mu * (x + r - 1)**2
        if y - r < 0: penalty += mu * (y - r)**2
        if y + r > 1: penalty += mu * (y + r - 1)**2
        
    return -sum_r + penalty

def run_packing():

        res1 = minimize(compute_objective, vars_init, args=(n, 5000.0), bounds=bounds, method='L-BFGS-B',

            best_obj = res2.fun
            best_vars = res2.x.copy()

    radii_opt = best_vars[2*n:]

    max_violation = 0.0
    for i in range(n):
        for j in range(i+1, n):

            if viol > max_violation:

    return centers_opt, radii_opt, sum_r
