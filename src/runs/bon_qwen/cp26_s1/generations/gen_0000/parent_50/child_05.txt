import numpy as np
import math
from scipy.optimize import minimize

def run_packing():
    n = 26
    
    # 1. Initialize centers and radii
    # Start with a 5x5 grid and place the 26th circle in the center
    centers = []
    # 5x5 grid with spacing 0.2
    x_coords = np.linspace(0.1, 0.9, 5)
    y_coords = np.linspace(0.1, 0.9, 5)
    
    for x in x_coords:
        for y in y_coords:
            centers.append([x, y])
    
    # Add the 26th circle at the center (0.5, 0.5)
    centers.append([0.5, 0.5])
    centers = np.array(centers)
    
    # Initial small radii
    radii = np.full(n, 0.001)
    
    # 2. Grow and Relax Loop
    grow_step = 0.0005
    max_grow_iterations = 10000
    relax_iterations = 50
    force_strength = 10.0
    boundary_strength = 5.0
    
    for grow_iter in range(max_grow_iterations):
        # Increase radii
        radii += grow_step
        
        # Relaxation phase to resolve overlaps
        for _ in range(relax_iterations):
            forces = np.zeros_like(centers)
            
            # Calculate inter-circle forces
            for i in range(n):
                for j in range(i + 1, n):
                    diff = centers[i] - centers[j]
                    dist = np.linalg.norm(diff)
                    req_dist = radii[i] + radii[j]
                    
                    if dist < 1e-9:
                        dist = 1e-9
                        diff = [np.random.rand(), np.random.rand()]
                        
                    overlap = req_dist - dist
                    
                    if overlap > 0:
                        # Repulsive force
                        direction = diff / dist
                        force_magnitude = force_strength * overlap
                        
                        forces[i] += direction * force_magnitude
                        forces[j] -= direction * force_magnitude
            
            # Calculate boundary forces
            for i in range(n):
                x, y = centers[i]
                r = radii[i]
                
                # Left wall
                if x - r < 0:
                    forces[i, 0] += boundary_strength * (r - x)
                # Right wall
                if x + r > 1:
                    forces[i, 0] -= boundary_strength * (x + r - 1)
                # Bottom wall
                if y - r < 0:
                    forces[i, 1] += boundary_strength * (r - y)
                # Top wall
                if y + r > 1:
                    forces[i, 1] -= boundary_strength * (y + r - 1)
            
            # Update positions
            centers += 0.1 * forces
            # Clamp centers to ensure they don't fly out during relaxation
            centers = np.clip(centers, 0, 1)
            
        # Check for significant overlaps to decide if we should stop or continue
        # If overlaps are too high, we might need to backtrack or just continue with higher forces
        # For this simulation, we just continue, hoping the forces resolve them.
        
        # Safety break if radii are getting too large or we've done enough iterations
        if np.sum(radii) > 2.8:
            break
            
    # 3. Final Polish using SciPy to maximize sum of radii locally
    # We fix the positions roughly and try to maximize radii, or jointly optimize
    # Given the simulation is usually quite good, we can run a local optimizer on the whole system
    # Variables: x1, y1, r1, ..., x26, y26, r26
    # Objective: Maximize sum(r) -> Minimize -sum(r)
    # Constraints: r_i >= 0, boundary, non-overlap
    
    # Flatten state for scipy
    x0 = np.concatenate([centers.ravel(), radii])
    
    def objective(state):
        r = state[52:]
        return -np.sum(r)
    
    constraints = []
    
    # Boundary constraints
    for i in range(n):
        idx_x = 2 * i
        idx_y = 2 * i + 1
        idx_r = 52 + i
        
        # x - r >= 0
        constraints.append({'type': 'ineq', 'fun': lambda s, i=i: s[idx_x] - s[idx_r]})
        # 1 - x - r >= 0
        constraints.append({'type': 'ineq', 'fun': lambda s, i=i: 1 - s[idx_x] - s[idx_r]})
        # y - r >= 0
        constraints.append({'type': 'ineq', 'fun': lambda s, i=i: s[idx_y] - s[idx_r]})
        # 1 - y - r >= 0
        constraints.append({'type': 'ineq', 'fun': lambda s, i=i: 1 - s[idx_y] - s[idx_r]})
        # r >= 0
        constraints.append({'type': 'ineq', 'fun': lambda s, i=i: s[idx_r]})
        
    # Overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            idx_xi, idx_yi = 2 * i, 2 * i + 1
            idx_xj, idx_yj = 2 * j, 2 * j + 1
            idx_ri, idx_rj = 52 + i, 52 + j
            
            def dist_constr(s, i=i, j=j, idx_xi=idx_xi, idx_yi=idx_yi, idx_xj=idx_xj, idx_yj=idx_yj, idx_ri=idx_ri, idx_rj=idx_rj):
                dx = s[idx_xi] - s[idx_xj]
                dy = s[idx_yi] - s[idx_yj]
                dist_sq = dx**2 + dy**2
                sum_r = s[idx_ri] + s[idx_rj]
                return dist_sq - sum_r**2
            
            constraints.append({'type': 'ineq', 'fun': dist_constr})
            
    # Run local optimizer
    # Use L-BFGS-B or SLSQP. SLSQP handles non-linear constraints better but is slower.
    # L-BFGS-B only handles bounds. We have general constraints. So SLSQP.
    try:
        res = minimize(objective, x0, method='SLSQP', constraints=constraints, options={'maxiter': 100, 'ftol': 1e-9})
        if res.success:
            x_opt = res.x
            centers = x_opt[:52].reshape(n, 2)
            radii = x_opt[52:]
    except Exception:
        pass # Fallback to simulation result
        
    # Ensure validity and non-negative radii
    radii = np.maximum(radii, 0.0)
    
    # 4. Sort circles by x, then y for consistent output
    sort_indices = np.lexsort((centers[:, 1], centers[:, 0]))
    centers = centers[sort_indices]
    radii = radii[sort_indices]
    
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
