import numpy as np
from scipy.optimize import minimize

def _penalized_obj(z, mu, n):
    """Objective function with penalty for constraint violations."""
    obj = -np.sum(z[2::3])
    pen = 0.0
    for i in range(n):
        xi, yi, ri = z[3*i], z[3*i+1], z[3*i+2]
        for j in range(i+1, n):
            xj, yj, rj = z[3*j], z[3*j+1], z[3*j+2]
            d = np.sqrt((xi-xj)**2 + (yi-yj)**2 + 1e-12)
            overlap = ri + rj - d
            if overlap > 0:
                pen += mu * overlap**2
        for val in [xi - ri, xi + ri - 1, yi - ri, yi + ri - 1]:
            if val < 0:
                pen += mu * val**2
    return obj + pen

        z0[3*i+2] = radii[i]

    for _ in range(25):
        res = minimize(_penalized_obj, z, args=(mu, n), method='L-BFGS-B',

        z = res.x
        mu *= 3.0
        
    final_centers = z[:2*n].reshape(n, 2)
