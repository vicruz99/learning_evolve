import numpy as np
from scipy.optimize import minimize

def compute_objective(vars, N, PEN):
    """
    Computes the objective function: negative sum of radii + penalty for violations.
    """
    c = vars[:2*N].reshape(N, 2)
    r = vars[2*N:]
    
    val = -np.sum(r)
    
    x, y = c[:, 0], c[:, 1]
    
    # Boundary penalties: ensure circles stay within [0, 1]x[0, 1]
    val += PEN * np.sum(np.maximum(0, r - x)**2)
    val += PEN * np.sum(np.maximum(0, x + r - 1)**2)
    val += PEN * np.sum(np.maximum(0, r - y)**2)
    val += PEN * np.sum(np.maximum(0, y + r - 1)**2)
    
    # Overlap penalties: ensure distance >= sum of radii
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist = np.sqrt(dx**2 + dy**2)
    np.fill_diagonal(dist, np.inf)  # Ignore self-distance
    
    r_sum = r[:, None] + r[None, :]
    val += PEN * np.sum(np.maximum(0, r_sum - dist)**2)
    
    return val

def run_packing():
