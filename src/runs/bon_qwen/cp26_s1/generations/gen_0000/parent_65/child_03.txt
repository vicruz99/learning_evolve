import numpy as np

def init_hex_packing(n):
    """Generate an initial hexagonal packing configuration for n circles."""
    centers = np.zeros((n, 2))
    r_init = 0.12
    idx = 0
    y = r_init
    row = 0
    while idx < n:
        x = r_init + row * r_init
        while x <= 1 - r_init and idx < n:
            centers[idx] = [x, y]
            idx += 1
            x += 2 * r_init
        y += r_init * np.sqrt(3)/2
        row += 1
    return centers[:n]

def compute_radii(centers):
    """Compute maximal radii and pairwise distances for given centers."""
    n = centers.shape[0]
    xb = np.minimum(centers[:, 0], 1 - centers[:, 0])
    yb = np.minimum(centers[:, 1], 1 - centers[:, 1])
    db = np.minimum(xb, yb)

    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)

    r_pairs = dists / 2.0
    radii = np.minimum(db, np.min(r_pairs, axis=1))
    return radii, dists, db, r_pairs, xb, yb

def run_packing():

    n = 26
    best_centers = None

    # Try multiple initializations to avoid poor local optima
    for trial in range(4):

            centers = init_hex_packing(n)

            centers = np.clip(centers, 0.01, 0.99)
        else:
            centers = np.random.rand(n, 2) * 0.8 + 0.1

        lr = 0.015
        for step in range(5000):
            radii, dists, db, r_pairs, xb, yb = compute_radii(centers)
            curr_sum = radii.sum()
            
            if curr_sum > best_sum:
                best_sum = curr_sum
                best_centers = centers.copy()

            grad = np.zeros_like(centers)
            eps = 1e-5

            for i in range(n):
                gi = np.zeros(2)
                cnt = 0

                # Boundary constraints
                if abs(db[i] - radii[i]) < eps:
                    if xb[i] < yb[i] - eps:
                        gi[0] += 1.0 if centers[i, 0] < 0.5 else -1.0
                        cnt += 1
                    elif yb[i] < xb[i] - eps:
                        gi[1] += 1.0 if centers[i, 1] < 0.5 else -1.0
                        cnt += 1
                    else:
                        gi[0] += 1.0 if centers[i, 0] < 0.5 else -1.0
                        gi[1] += 1.0 if centers[i, 1] < 0.5 else -1.0
                        cnt += 2

                # Neighbor constraints affecting r_i
                for j in range(n):
                    if i == j:
                        continue

                        if dists[i, j] < 1e-8:
                            continue
                        gi -= (centers[i] - centers[j]) / (2 * dists[i, j])
                        cnt += 1

                if cnt > 0:
                    grad[i] += gi / cnt

                # Contribution to other radii r_k from moving c_i

                    if i == k:
                        continue

                        if dists[k, i] < 1e-8:
                            continue
                        grad[i] += (centers[k] - centers[i]) / (2 * dists[k, i])

    return best_centers, final_radii, final_radii.sum()
