import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def objective_func(x):
    """Objective: minimize negative sum of radii (equivalent to maximizing sum of radii)"""
    return -np.sum(x[2::3])

def constraint_func(x):
    """Returns array of inequality constraints: all values must be >= 0"""
    c = []
    # Boundary constraints for each circle
    for i in range(N_CIRCLES):
        idx = 3 * i
        xi, yi, ri = x[idx], x[idx + 1], x[idx + 2]
        c.append(xi - ri)           # Left boundary
        c.append(1.0 - xi - ri)     # Right boundary
        c.append(yi - ri)           # Bottom boundary
        c.append(1.0 - yi - ri)     # Top boundary
        
    # Non-overlap constraints for each pair
    for i in range(N_CIRCLES):
        for j in range(i + 1, N_CIRCLES):
            idx_i, idx_j = 3 * i, 3 * j
            dx = x[idx_i] - x[idx_j]
            dy = x[idx_i + 1] - x[idx_j + 1]
            r_sum = x[idx_i + 2] + x[idx_j + 2]
            c.append(dx * dx + dy * dy - r_sum * r_sum)
            
    return np.array(c)

def run_packing():
    np.random.seed(42)
    best_sum_r = 0.0
    best_x = None
    
    # Multiple restarts to avoid local minima
    for trial in range(15):
        # Initialize centers away from edges, with small feasible radii
        centers = np.random.uniform(0.1, 0.9, size=(N_CIRCLES, 2))
        radii = np.full(N_CIRCLES, 0.02)
        x0 = np.concatenate([centers.flatten(), radii])
        
        # Bounds: coordinates in [0, 1], radii in [1e-6, 1]
        bounds = [(0.0, 1.0)] * (2 * N_CIRCLES) + [(1e-6, 1.0)] * N_CIRCLES
        cons = {'type': 'ineq', 'fun': constraint_func}
        
        # SLSQP solver for smooth non-linear programming
        res = minimize(objective_func, x0, method='SLSQP', bounds=bounds, 
                       constraints=cons, options={'maxiter': 3000, 'ftol': 1e-10})
                       
        current_sum = -res.fun
        if current_sum > best_sum_r:
            best_sum_r = current_sum
            best_x = res.x
            
    centers = best_x[:2 * N_CIRCLES].reshape(N_CIRCLES, 2)
    radii = best_x[2 * N_CIRCLES:]
    return centers, radii, best_sum_r
