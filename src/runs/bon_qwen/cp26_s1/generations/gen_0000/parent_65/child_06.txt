import numpy as np
from scipy.optimize import linprog, differential_evolution
import math

def solve_lp(centers):
    """
    Solves the Linear Programming problem to maximize the sum of radii
    given fixed center coordinates.
    
    Args:
        centers: np.array of shape (n, 2)
        
    Returns:
        sum_radii: float
        radii: np.array of shape (n)
    """
    n = centers.shape[0]
    if n == 0:
        return 0.0, np.array([])

    # Objective: Maximize sum(r_i) => Minimize -sum(r_i)
    c = -np.ones(n)

    # Variable bounds: 0 <= r_i <= distance_to_boundary
    # We compute the max possible radius based on boundaries first
    bounds = []
    for x, y in centers:
        dist_to_boundary = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(0.0, dist_to_boundary)))

    # Pairwise constraints: r_i + r_j <= distance(i, j)
    # r_i + r_j - distance(i, j) <= 0
    A_ub = []
    b_ub = []
    
    # Precompute squared distances to avoid sqrt in loop if not needed, 
    # but we need distance for b_ub.
    dists = np.linalg.norm(centers[:, np.newaxis] - centers[np.newaxis, :], axis=2)
    
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])

    if len(A_ub) == 0:
        # No pairwise constraints (n=1)
        res = linprog(c, bounds=bounds, method='highs')
    else:
        A_ub = np.array(A_ub)
        b_ub = np.array(b_ub)
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')

    if res.success:
        return -res.fun, res.x
    else:
        # Fallback: return 0 or current bounds
        return 0.0, np.array([b[1] for b in bounds])

def objective_func(flat_centers):
    """
    Objective function for the optimizer. Minimizes negative sum of radii.
    """
    n = 26
    centers = flat_centers.reshape(n, 2)
    
    # Clip to ensure centers are within [0, 1]
    centers = np.clip(centers, 0.0, 1.0)
    
    sum_radii, radii = solve_lp(centers)
    
    # Return negative sum for minimization
    return -sum_radii

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    """
    n = 26

    centers_init = []

    # Hex spacing vertical: diam * sqrt(3)/2 ~ 0.18
    
    # Try to place rows
    # Row 0: y=0.105
    # Row 1: y=0.105 + 0.182 = 0.287
    # ...
    
    # A simpler approach: Fill a grid and perturb or just use a dense grid
    # and let the optimizer spread them.
    
    # Let's create a 6x5 grid (30 points) and pick 26 best? 
    # Or just a 5x6 grid.
    # Let's try to fit as many as possible in a hex pattern.
    
    r_est = 0.11
    dx = 2 * r_est
    dy = dx * math.sqrt(3) / 2
    
    y = r_est
    row_idx = 0
    while y + r_est <= 1.0:
        x = r_est
        # Offset every other row

        offset = dx / 2.0
        if row_idx % 2 == 1:
            x += offset
        
        while x + r_est <= 1.0:
            centers_init.append([x, y])
            if len(centers_init) == n:
                break
            x += dx
        if len(centers_init) == n:
            break
        y += dy
        row_idx += 1

    # If we didn't fill 26, fill remaining with random points or grid points
    # But with r_est=0.11, we should get close.

    if len(centers_init) < n:

            centers_init.append([np.random.rand(), np.random.rand()])

    bounds = [(0, 1)] * (2 * n)
    
    # Differential Evolution is robust for global optimization

    # popsize * dim * maxiter.

    result = differential_evolution(
        objective_func,

        seed=42,

        recombination=0.7,

    )
