3. **Continuation**: Start with a moderate penalty parameter $\lambda$ and gradually increase it. This guides the optimizer from a loose configuration to a tightly packed feasible one without getting stuck in poor local minima.
4. **Multi-start Initialization**: To escape local minima, I'll run the optimization from 3 different initial configurations:

import numpy as np
from scipy.optimize import minimize

def compute_objective(vars, lam, n):
    val = -np.sum(vars[2::3])
    for i in range(n):
        x = vars[3*i]
        y = vars[3*i+1]
        r = vars[3*i+2]
        if x < r:
            val += lam * (r - x)**2
        elif x > 1 - r:
            val += lam * (x - (1 - r))**2
        if y < r:
            val += lam * (r - y)**2
        elif y > 1 - r:
            val += lam * (y - (1 - r))**2
    for i in range(n):
        xi = vars[3*i]
        yi = vars[3*i+1]
        ri = vars[3*i+2]
        for j in range(i+1, n):
            xj = vars[3*j]
            yj = vars[3*j+1]
            rj = vars[3*j+2]
            dist = np.sqrt((xi - xj)**2 + (yi - yj)**2 + 1e-12)
            if dist < ri + rj:
                val += lam * (ri + rj - dist)**2
    return val

def check_validity(centers, radii, n):
    for i in range(n):
        x = centers[i, 0]
        y = centers[i, 1]
        r = radii[i]
        if x - r < -1e-7 or x + r > 1 + 1e-7:
            return False
        if y - r < -1e-7 or y + r > 1 + 1e-7:
            return False
    for i in range(n):
        for j in range(i+1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            d = np.sqrt(dx*dx + dy*dy)
            if d < radii[i] + radii[j] - 1e-7:
                return False
    return True

def run_optimization(x0, n):
    bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n
    lam = 50.0
    for _ in range(40):
        res = minimize(compute_objective, x0, args=(lam, n), method='L-BFGS-B', 
                       bounds=bounds, options={'ftol': 1e-9, 'gtol': 1e-7, 'maxiter': 1000})
        x0 = res.x
        pen = compute_objective(x0, lam, n) + np.sum(x0[2::3])
        if pen < 1e-5:
            break
        lam *= 1.5
    return x0

def run_packing():
    n = 26
    
    # Init 1: Grid
    x_base = np.linspace(0.1, 0.9, 5)
    y_base = np.linspace(0.1, 0.9, 5)
    init1 = np.zeros(3*n)
    idx = 0
    for y in y_base:
        for x in x_base:
            init1[3*idx] = x
            init1[3*idx+1] = y
            init1[3*idx+2] = 0.08
            idx += 1
    init1[3*25] = 0.05
    init1[3*25+1] = 0.5
    init1[3*25+2] = 0.08
    
    # Init 2: Random
    np.random.seed(123)
    init2 = np.zeros(3*n)
    init2[:2*n] = np.random.rand(2*n)
    init2[2::3] = 0.05
    
    # Init 3: Another layout
    init3 = np.zeros(3*n)
    xs = np.linspace(0.08, 0.92, 6)
    ys = np.linspace(0.125, 0.875, 4)
    idx = 0
    for y in ys:
        for x in xs:
            init3[3*idx] = x
            init3[3*idx+1] = y
            init3[3*idx+2] = 0.07
            idx += 1
    init3[3*24] = 0.02; init3[3*24+1] = 0.02; init3[3*24+2] = 0.06
    init3[3*25] = 0.98; init3[3*25+1] = 0.98; init3[3*25+2] = 0.06
    
    best_x = None
    best_sum = -1.0
    
    for x0 in [init1, init2, init3]:
        x_opt = run_optimization(x0, n)
        s = np.sum(x_opt[2::3])
        if s > best_sum:
            best_sum = s
            best_x = x_opt.copy()
            
    centers = best_x.reshape(n, 3)[:, :2]
    radii = best_x.reshape(n, 3)[:, 2]
    
    if not check_validity(centers, radii, n):
        scale = 1.0
        while not check_validity(centers, radii * scale, n) and scale > 0.95:
            scale *= 0.99
        radii *= scale
        
    return centers, radii, np.sum(radii)
