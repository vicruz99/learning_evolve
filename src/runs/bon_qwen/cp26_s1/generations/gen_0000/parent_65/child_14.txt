import numpy as np
from scipy.optimize import minimize
import itertools

N_CIRCLES = 26


def compute_violation_penalty(params, weight=5000.0):
    """Compute penalty for constraint violations."""
    n = N_CIRCLES
    centers = params[:2 * n].reshape(n, 2)
    radii = params[2 * n:]

    penalty = 0.0

    # Boundary violations
    for i in range(n):
        x, y = centers[i]
        r = radii[i]

        violation = max(0.0, r - x)
        penalty += weight * violation * violation

        violation = max(0.0, x + r - 1.0)
        penalty += weight * violation * violation

        violation = max(0.0, r - y)
        penalty += weight * violation * violation

        violation = max(0.0, y + r - 1.0)
        penalty += weight * violation * violation

    # Overlap violations
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            violation = max(0.0, radii[i] + radii[j] - dist)
            penalty += weight * violation * violation

    return penalty


def objective_with_penalty(params):
    """Objective: maximize sum of radii with penalty for violations."""
    n = N_CIRCLES
    radii = params[2 * n:]
    penalty = compute_violation_penalty(params, weight=10000.0)
    return -np.sum(radii) + penalty


def compute_constraints(params):
    """Compute all inequality constraints for SLSQP."""
    n = N_CIRCLES
    centers = params[:2 * n].reshape(n, 2)
    radii = params[2 * n:]

    constraints = []

    # Boundary constraints
    for i in range(n):
        constraints.append(centers[i, 0] - radii[i])
        constraints.append(1.0 - centers[i, 0] - radii[i])
        constraints.append(centers[i, 1] - radii[i])
        constraints.append(1.0 - centers[i, 1] - radii[i])

    # Non-overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            constraints.append(dist - radii[i] - radii[j])

    return np.array(constraints)


def compute_objective(params):
    """Negative sum of radii for minimization."""
    n = N_CIRCLES
    return -np.sum(params[2 * n:])


def get_bounds():
    """Create bounds for all optimization variables."""
    n = N_CIRCLES
    bounds = []
    for i in range(n):
        bounds.extend([(0.0, 1.0), (0.0, 1.0)])
        bounds.append((0.0, 0.5))
    return bounds


def init_hexagonal():
    """Initialize circles in a hexagonal lattice pattern."""
    n = N_CIRCLES
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.075)

    idx = 0
    spacing = 0.17
    row = 0
    while idx < n:
        col = 0
        while idx < n:
            x = 0.04 + col * spacing
            y = 0.04 + row * spacing * 0.86602540378
            if row % 2 == 1:
                x += spacing / 2.0
            if x < 0.96 and y < 0.96:
                centers[idx] = [x, y]
                idx += 1
            col += 1
        row += 1

    while idx < n:
        centers[idx] = [0.5, 0.5 + idx * 0.005]
        idx += 1

    return centers, radii


def init_hexagonal_v2():
    """Alternative hexagonal initialization with different spacing."""
    n = N_CIRCLES
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.08)

    idx = 0
    spacing = 0.165
    row = 0
    while idx < n:
        col = 0
        while idx < n:
            x = 0.03 + col * spacing
            y = 0.03 + row * spacing * 0.86602540378
            if row % 2 == 1:
                x += spacing / 2.0
            if x <= 0.97 and y <= 0.97:
                centers[idx] = [x, y]
                idx += 1
            col += 1
        row += 1

    while idx < n:
        centers[idx] = [0.5, 0.5]
        idx += 1

    return centers, radii


def init_square_grid():
    """Initialize circles in a square grid pattern."""
    n = N_CIRCLES
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.06)

    spacing = 0.18
    idx = 0
    for row in range(6):
        for col in range(5):
            if idx < n:
                x = 0.05 + col * spacing
                y = 0.05 + row * spacing
                centers[idx] = [x, y]
                idx += 1

    while idx < n:
        centers[idx] = [0.5, 0.5]
        idx += 1

    return centers, radii


def init_stratified(seed=None):
    """Initialize with stratified random sampling."""
    n = N_CIRCLES
    if seed is not None:
        np.random.seed(seed)
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.05)

    grid_size = 6
    spacing = 1.0 / grid_size

    idx = 0
    for i in range(grid_size):
        for j in range(grid_size):
            if idx < n:
                x = (i + 0.5) * spacing + np.random.uniform(-0.3, 0.3) * spacing
                y = (j + 0.5) * spacing + np.random.uniform(-0.3, 0.3) * spacing
                x = np.clip(x, 0.08, 0.92)
                y = np.clip(y, 0.08, 0.92)
                centers[idx] = [x, y]
                idx += 1

    while idx < n:
        centers[idx] = [np.random.uniform(0.1, 0.9), np.random.uniform(0.1, 0.9)]
        idx += 1

    return centers, radii


def init_random(seed=None):
    """Initialize with random positions."""
    n = N_CIRCLES
    if seed is not None:
        np.random.seed(seed)
    centers = np.random.uniform(0.12, 0.88, (n, 2))
    radii = np.full(n, 0.04)
    return centers, radii


def init_corner_optimized():
    """Initialize with circles optimized for corner usage."""
    n = N_CIRCLES
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.07)

    while idx < n:
        centers[idx] = [0.5, 0.5]
        idx += 1

def check_validity(centers, radii):
    """Check if packing satisfies all constraints."""
    n = N_CIRCLES

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if r < 0:
            return False
        if x - r < -1e-10 or x + r > 1.0 + 1e-10:
            return False
        if y - r < -1e-10 or y + r > 1.0 + 1e-10:
            return False

    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            if dist < radii[i] + radii[j] - 1e-10:
                return False

    return True


def optimize_penalty(params0):
    """Optimize using penalty method with L-BFGS-B."""
    bounds = get_bounds()

    result = minimize(
        objective_with_penalty,
        params0,
        method='L-BFGS-B',
        bounds=bounds,
        options={
            'maxiter': 5000,
            'ftol': 1e-15,
            'gtol': 1e-12
        }
    )

    return result.x


def optimize_constrained(params0):
    """Optimize using SLSQP with explicit constraints."""
    bounds = get_bounds()
    constraint_dict = {
        'type': 'ineq',
        'fun': compute_constraints
    }

    result = minimize(
        compute_objective,
        params0,
        method='SLSQP',
        bounds=bounds,
        constraints=constraint_dict,
        options={
            'maxiter': 2000,
            'ftol': 1e-14,
            'disp': False
        }
    )

    return result.x


def run_packing():
