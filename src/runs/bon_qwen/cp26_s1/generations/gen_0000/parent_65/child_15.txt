import numpy as np
from scipy.optimize import differential_evolution

N_CIRCLES = 26

def evaluate_min_dist(centers):
    """
    Computes the minimum distance between all pairs of centers 
    and between centers and the unit square boundaries.
    """
    # Pairwise distances using broadcasting
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    min_pair = np.min(dists)
    
    # Distance to square boundaries [0,1]x[0,1]
    d_bound = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                         np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    min_bnd = np.min(d_bound)
    
    return min(min_pair, min_bnd)

def de_objective(x):
    """
    Objective function for differential evolution.
    Minimizes the negative of the minimum distance.
    """
    centers = x.reshape((N_CIRCLES, 2))
    return -evaluate_min_dist(centers)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Bounds for each coordinate: [0, 1]
    bounds = [(0.0, 1.0) for _ in range(2 * N_CIRCLES)]
    
    best_x = None
    best_val = -1.0
    
    # Run Differential Evolution multiple times to ensure global convergence
    for _ in range(5):
        try:
            res = differential_evolution(
                de_objective, 
                bounds, 
                popsize=20, 
                maxiter=400, 
                mutation=(0.5, 1.5), 
                recombination=0.7,
                seed=None, 
                workers=1,
                tol=1e-6,
                atol=1e-6,
                polish=True
            )
            if -res.fun > best_val:
                best_val = -res.fun
                best_x = res.x
        except Exception:
            continue
            
    if best_x is not None:
        centers = best_x.reshape((N_CIRCLES, 2))
    else:
        # Fallback to uniform random initialization if optimization fails
        centers = np.random.uniform(0.1, 0.9, (N_CIRCLES, 2))
        
    # Final evaluation and radius assignment
    min_d = evaluate_min_dist(centers)
    r = min_d / 2.0
    
    # Strictly enforce boundaries for numerical safety
    centers = np.clip(centers, r, 1.0 - r)
    
    radii = np.full(N_CIRCLES, r)
    total_sum = float(N_CIRCLES * r)
    
    return centers, radii, total_sum
