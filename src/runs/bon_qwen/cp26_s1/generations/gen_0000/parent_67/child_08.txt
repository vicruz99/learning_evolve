def calculate_objective(vars):
    xs = vars[0::3]
    ys = vars[1::3]
    rs = vars[2::3]
    
    sum_r = np.sum(rs)
    
    # Boundary penalties
    # x >= r -> r - x <= 0
    # x <= 1-r -> x + r - 1 <= 0
    # Same for y
    
    # Vectorized boundary check
    # x - r
    lower_x = xs - rs
    upper_x = xs + rs - 1.0
    lower_y = ys - rs
    upper_y = ys + rs - 1.0
    
    # Penalty is sum of squares of positive violations
    penalty_boundary = 0.0
    if np.any(lower_x < 0): penalty_boundary += np.sum(lower_x[lower_x < 0]**2)
    if np.any(upper_x > 0): penalty_boundary += np.sum(upper_x[upper_x > 0]**2)
    if np.any(lower_y < 0): penalty_boundary += np.sum(lower_y[lower_y < 0]**2)
    if np.any(upper_y > 0): penalty_boundary += np.sum(upper_y[upper_y > 0]**2)
    
    # Overlap penalties
    # Distance matrix
    # Centers shape (N, 2)
    centers = np.column_stack((xs, ys))
    # Diff matrix (N, N, 2)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist = np.sqrt(np.sum(diff**2, axis=2))
    
    # Radii sum matrix (N, N)
    radii_sum = rs[:, np.newaxis] + rs[np.newaxis, :]
    
    # Violations where radii_sum > dist
    violations = radii_sum - dist
    # Only upper triangle to avoid double counting (and diagonal is 0)
    # But diagonal is dist=0, req=2r > 0, so violation.
    # We must exclude diagonal (i==j)
    np.fill_diagonal(violations, 0)
    
    # Take only positive violations
    overlap_penalty = violations * (violations > 0)
    penalty_overlap = np.sum(overlap_penalty**2)
    
    # Combine
    # We sum penalty_overlap, but note that we counted each pair twice (i,j) and (j,i)
    # Since we squared and summed, it's fine as long as consistent.
    # Actually, (r_i+r_j - d_ij)^2 is same for (j,i).
    # So summing all entries counts each pair twice.
    # This is fine for optimization, just scales the gradient.
    
    weight_boundary = 10000.0
    weight_overlap = 10000.0
    
    return -sum_r + weight_boundary * penalty_boundary + weight_overlap * penalty_overlap

        centers_opt[i, 0] = np.clip(x, r, 1.0 - r)
        centers_opt[i, 1] = np.clip(y, r, 1.0 - r)

    sum_radii = np.sum(radii_opt)
    
    return centers_opt, radii_opt, float(sum_radii)
