def get_centers_radii(z, n):
    """
    Decode the optimization variables z into centers and radii.
    z = [r_1...r_n, d_x_1...d_x_n, d_y_1...d_y_n]
    """
    r = z[:n]
    dx = z[n:2*n]
    dy = z[2*n:3*n]
    
    # Reconstruction: x = r + dx * (1 - 2r)
    # This ensures r <= x <= 1-r when 0 <= dx <= 1
    x = r + dx * (1 - 2 * r)
    y = r + dy * (1 - 2 * r)
    
    centers = np.vstack([x, y]).T
    radii = r
    return centers, radii

def constraint_overlap_func(args):
    """
    Factory function for overlap constraints.
    Returns a function that evaluates the constraint for a specific pair (i, j).
    Constraint: dist^2 - (r_i + r_j)^2 >= 0
    """
    i, j, n = args
    
    def fun(z):
        r = z[:n]
        dx = z[n:2*n]
        dy = z[2*n:3*n]
        
        ri, rj = r[i], r[j]
        
        xi = ri + dx[i] * (1 - 2 * ri)
        yi = ri + dy[i] * (1 - 2 * ri)
        
        xj = rj + dx[j] * (1 - 2 * rj)
        yj = rj + dy[j] * (1 - 2 * rj)
        
        dist_sq = (xi - xj)**2 + (yi - yj)**2
        min_dist_sq = (ri + rj)**2
        
        return dist_sq - min_dist_sq
        
    return fun
