import numpy as np
from scipy.optimize import linprog, minimize

def solve_radii(centers):
    """
    Solves the LP to find optimal radii for a given set of centers.
    Maximizes sum(r_i) subject to:
    r_i >= 0
    r_i <= boundary distances
    r_i + r_j <= distance(centers[i], centers[j])
    """
    n = centers.shape[0]
    c = -np.ones(n)  # Maximize sum(r) -> Minimize -sum(r)
    
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        x, y = centers[i]
        bounds = [x, 1 - x, y, 1 - y]
        for b in bounds:
            row = np.zeros(n)
            row[i] = 1
            A_ub.append(row)
            b_ub.append(b)
            
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            row = np.zeros(n)
            row[i] = 1
            row[j] = 1
            A_ub.append(row)
            b_ub.append(dist)
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    # Bounds for radii: r_i >= 0
    bounds = [(0, None) for _ in range(n)]
    
    # Solve LP
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return res.x
    else:
        # Fallback if LP fails (shouldn't happen with valid centers)
        return np.zeros(n)

def objective(centers_flat):
    """Objective function for the optimizer: returns negative sum of radii."""
    n = 26
    centers = centers_flat.reshape((n, 2))
    
    # Ensure centers are within [0, 1] to avoid invalid LP constraints
    # The optimizer should handle bounds, but clamping helps numerical stability
    centers = np.clip(centers, 1e-6, 1 - 1e-6)
    
    radii = solve_radii(centers)
    return -np.sum(radii)

def generate_hex_centers(n, r_init=0.1):
    """Generate a hexagonal grid of n centers."""
    centers = []
    row = 0
    while len(centers) < n:
        y = (row + 0.5) * r_init * np.sqrt(3)
        if y > 1.0: break
        
        col = 0
        while True:
            x = (col + 0.5) * 2 * r_init + (0.5 * r_init if row % 2 == 1 else 0)
            if x > 1.0: break
            
            if len(centers) < n:
                centers.append([x, y])
            col += 1
        row += 1
    return np.array(centers)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    bounds = [(0, 1) for _ in range(2 * n)]
    
    best_centers = None
    best_radii = None
    best_sum = -np.inf
    
    # Strategy: Multiple restarts from hexagonal grids with random perturbations
    # and some fully random starts.
    for trial in range(20):
        if trial < 15:
            # Hexagonal grid start with noise
            # Estimate initial radius to fit 26 circles roughly
            # Area approx: 26 * pi * r^2 approx 0.9 * 1 => r approx 0.105
            # But boundary effects reduce this. Start with 0.1
            c = generate_hex_centers(n, r_init=0.1)
            # Add small noise
            noise = np.random.normal(0, 0.02, c.shape)
            c = np.clip(c + noise, 0.01, 0.99)
        else:
            # Random start
            c = np.random.rand(n, 2)
        
        x0 = c.flatten()
        
        try:
            # Nelder-Mead is good for non-smooth functions
            res = minimize(objective, x0, method='Nelder-Mead', 
                           bounds=bounds, 
                           options={'xatol': 1e-6, 'fatol': 1e-6, 'maxiter': 1000})
            
            if res.fun < best_sum:
                best_sum = res.fun
                best_centers = res.x.reshape((n, 2))
        except Exception:
            continue

    # Solve for final radii with best centers
    radii = solve_radii(best_centers)
    sum_radii = np.sum(radii)
    
    return best_centers, radii, sum_radii

# Run the function to get the result
centers, radii, sum_radii = run_packing()

# Validation check (internal)
def validate_packing(centers, radii):
    n = centers.shape[0]
    if np.isnan(centers).any() or np.isnan(radii).any():
        return False
    for i in range(n):
        if radii[i] < 0:
            return False
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            return False
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False
    return True

is_valid = validate_packing(centers, radii)
print(f"Valid: {is_valid}, Sum of radii: {sum_radii}")
