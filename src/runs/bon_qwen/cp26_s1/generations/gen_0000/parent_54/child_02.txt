import numpy as np
from scipy.optimize import differential_evolution
from scipy.spatial.distance import cdist

def objective(centers_flat):
    """
    Computes the negative sum of maximum possible radii for given centers.
    Minimizing this is equivalent to maximizing the sum of radii.
    """
    centers = centers_flat.reshape(26, 2)
    
    # Minimum distance to boundaries
    d_bound = np.min(np.stack([
        centers[:, 0], 1.0 - centers[:, 0],
        centers[:, 1], 1.0 - centers[:, 1]
    ], axis=0), axis=0)
    
    # Minimum distance to other circles
    dists = cdist(centers, centers)
    np.fill_diagonal(dists, np.inf)
    d_circ = np.min(dists, axis=1)
    
    # Valid radius is half the limiting distance
    radii = np.minimum(d_bound, d_circ) / 2.0
    return -np.sum(radii)

def run_packing():
    # Optimization bounds for 26 circles (x, y) in [0, 1]
    bounds = [(0.0, 1.0)] * 52
    
    # Differential Evolution to find optimal centers
    result = differential_evolution(
        objective, 
        bounds, 
        popsize=15, 
        maxiter=200, 
        seed=42, 
        polish=False,
        tol=1e-9,
        mutation=(0.5, 1.5),
        recombination=0.7
    )
    
    centers = result.x.reshape(26, 2)
    
    # Recompute exact maximal radii for the optimized centers
    d_bound = np.min(np.stack([
        centers[:, 0], 1.0 - centers[:, 0],
        centers[:, 1], 1.0 - centers[:, 1]
    ], axis=0), axis=0)
    
    dists = cdist(centers, centers)
    np.fill_diagonal(dists, np.inf)
    d_circ = np.min(dists, axis=1)
    
    radii = np.minimum(d_bound, d_circ) / 2.0
    
    # Slight reduction to safely pass validation tolerance
    radii -= 1e-10
    sum_radii = np.sum(radii)
    
    # Fallback to a structured grid if optimization underperforms (unlikely)
    if sum_radii < 2.55:
        grid_pts = np.array([
            [0.1 + i*0.2, 0.1 + j*0.2] for i in range(5) for j in range(5)
        ])
        centers = np.vstack([grid_pts, [0.5, 0.5]])
        
        d_bound = np.min(np.stack([
            centers[:, 0], 1.0 - centers[:, 0],
            centers[:, 1], 1.0 - centers[:, 1]
        ], axis=0), axis=0)
        
        dists = cdist(centers, centers)
        np.fill_diagonal(dists, np.inf)
        d_circ = np.min(dists, axis=1)
        
        radii = np.minimum(d_bound, d_circ) / 2.0 - 1e-10
        sum_radii = np.sum(radii)
        
    return centers, radii, sum_radii
