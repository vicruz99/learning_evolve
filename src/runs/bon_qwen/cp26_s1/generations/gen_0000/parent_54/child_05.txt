import numpy as np
from scipy.optimize import linprog, minimize
from scipy.spatial.distance import pdist, squareform

def compute_max_radii(centers):
    n = centers.shape[0]
    dists = squareform(pdist(centers))
    
    c_obj = -np.ones(n)
    
    m = n * (n - 1) // 2
    A_ub = np.zeros((m, n))
    b_ub = np.zeros(m)
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1
            
    bounds = []
    for i in range(n):
        x, y = centers[i]
        lim = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(0.0, lim)))
        
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    except ValueError:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='interior-point')
        
    if res.success:
        return -res.fun, -res.x
    return 0.0, np.zeros(n)

def objective(x):
    n = 26
    centers = x.reshape(n, 2)
    centers = np.clip(centers, 1e-5, 1.0 - 1e-5)
    val, _ = compute_max_radii(centers)
    return -val

def run_packing():
    n = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    inits = []
    
    # Hexagonal grid initialization
    pts = []
    rows, cols = 5, 6
    for i in range(rows):
        for j in range(cols):
            if len(pts) >= n: 
                break
            x = 0.1 + j * 0.8 / (cols - 1) + (0.4 / (cols - 1) if i % 2 else 0)
            y = 0.1 + i * 0.8 / (rows - 1)
            pts.append([x, y])
        if len(pts) >= n: 
            break
    inits.append(np.array(pts).flatten())
    
    # Random initializations to escape local minima
    for _ in range(5):
        inits.append((np.random.rand(2 * n) * 0.8 + 0.1).flatten())
        
    bounds = [(0.0, 1.0)] * (2 * n)
    
    for x0 in inits:
        # Robust basin search
        res = minimize(objective, x0, method='Nelder-Mead', 
                       options={'maxiter': 1000, 'xatol': 1e-7, 'fatol': 1e-7})
        
        # Precise local refinement respecting boundaries
        res2 = minimize(objective, res.x, method='L-BFGS-B', bounds=bounds,
                        options={'maxiter': 1500, 'ftol': 1e-9})
                        
        if -res2.fun > best_sum:
            best_sum = -res2.fun
            best_centers = res2.x.reshape(n, 2)
            
    _, final_radii = compute_max_radii(best_centers)
    return best_centers, final_radii, best_sum
