import numpy as np
from scipy.optimize import linprog

def get_optimal_radii_and_forces(centers):
    """
    Solves the LP to find max sum of radii for fixed centers
    and computes the gradient forces on centers.
    """
    n = centers.shape[0]
    
    # Objective: maximize sum(r_i) => minimize -sum(r_i)
    c_obj = -np.ones(n)
    
    # Constraints:
    # 1. r_i + r_j <= distance(c_i, c_j)
    # 2. r_i <= x_i
    # 3. r_i <= 1 - x_i
    # 4. r_i <= y_i
    # 5. r_i <= 1 - y_i
    
    # We will build A_ub and b_ub.
    # To track forces, we need to know which constraint corresponds to which pair/bound.
    # We can store this info or just iterate and reconstruct.
    # Iterating is safer.
    
    num_constraints = (n * (n - 1)) // 2 + 4 * n
    A_ub = np.zeros((num_constraints, n))
    b_ub = np.zeros(num_constraints)
    
    constraint_idx = 0
    
    # Pairwise constraints
    pair_constraints = [] # Store (i, j) for each row
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub[constraint_idx] = row
            
            dist = np.linalg.norm(centers[i] - centers[j])
            b_ub[constraint_idx] = dist
            
            pair_constraints.append((i, j))
            constraint_idx += 1
            
    # Boundary constraints
    # Store type: 'x_low', 'x_high', 'y_low', 'y_high'
    bound_constraints = [] 
    for i in range(n):
        x, y = centers[i]
        
        # r_i <= x  (x_low)
        row = np.zeros(n); row[i] = 1.0
        A_ub[constraint_idx] = row
        b_ub[constraint_idx] = x
        bound_constraints.append(('x_low', i))
        constraint_idx += 1
        
        # r_i <= 1 - x (x_high)
        row = np.zeros(n); row[i] = 1.0
        A_ub[constraint_idx] = row
        b_ub[constraint_idx] = 1.0 - x
        bound_constraints.append(('x_high', i))
        constraint_idx += 1
        
        # r_i <= y (y_low)
        row = np.zeros(n); row[i] = 1.0
        A_ub[constraint_idx] = row
        b_ub[constraint_idx] = y
        bound_constraints.append(('y_low', i))
        constraint_idx += 1
        
        # r_i <= 1 - y (y_high)
        row = np.zeros(n); row[i] = 1.0
        A_ub[constraint_idx] = row
        b_ub[constraint_idx] = 1.0 - y
        bound_constraints.append(('y_high', i))
        constraint_idx += 1
        
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    except Exception:
        return None, None, None
        
    if not res.success:
        return None, None, None
        
    radii = res.x
    sum_radii = -res.fun
    
    # Compute forces
    # Forces are gradient of sum_radii w.r.t centers
    # sum_radii depends on centers via RHS of constraints (b_ub)
    # d(sum_radii)/d(center_k) = sum over constraints m of (marginal_m * d(b_m)/d(center_k))
    
    forces = np.zeros((n, 2))
    marginals = res.ineqlin.marginals
    
    # Process pairwise constraints
    # Marginals indices 0 to num_pairs-1
    for idx, (i, j) in enumerate(pair_constraints):
        lam = marginals[idx]
        if lam > 1e-9: # Active constraint
            # Constraint: r_i + r_j <= ||c_i - c_j||
            # Increasing ||c_i - c_j|| relaxes constraint.
            # Gradient of distance w.r.t c_i is (c_i - c_j) / ||c_i - c_j||
            # Gradient w.r.t c_j is (c_j - c_i) / ||c_i - c_j||
            
            vec = centers[i] - centers[j]
            dist = np.linalg.norm(vec)
            if dist > 1e-9:
                unit_vec = vec / dist
                # Force on i: +lam * unit_vec (push i away from j)
                forces[i, 0] += lam * unit_vec[0]
                forces[i, 1] += lam * unit_vec[1]
                # Force on j: +lam * (-unit_vec) (push j away from i)
                forces[j, 0] -= lam * unit_vec[0]
                forces[j, 1] -= lam * unit_vec[1]
                
    # Process boundary constraints
    # Marginals indices num_pairs to end
    pair_count = len(pair_constraints)
    for idx, (b_type, i) in enumerate(bound_constraints):
        lam = marginals[pair_count + idx]
        if lam > 1e-9:
            # Constraint r_i <= bound
            # Increasing bound relaxes constraint.
            # Force direction is gradient of bound w.r.t center i.
            
            if b_type == 'x_low': # r <= x
                # d(x)/dx = 1
                forces[i, 0] += lam
            elif b_type == 'x_high': # r <= 1-x
                # d(1-x)/dx = -1
                forces[i, 0] -= lam
            elif b_type == 'y_low': # r <= y
                # d(y)/dy = 1
                forces[i, 1] += lam
            elif b_type == 'y_high': # r <= 1-y
                # d(1-y)/dy = -1
                forces[i, 1] -= lam
                
    return radii, sum_radii, forces

```python
import numpy as np
from scipy.optimize import linprog

def get_optimal_radii_and_forces(centers):
    """
    Solves the LP to find max sum of radii for fixed centers
    and computes the gradient forces on centers based on dual variables.
    """
    n = centers.shape[0]
    if n == 0:
        return np.array([]), 0.0, np.array([])

    # Objective: maximize sum(r_i) => minimize -sum(r_i)
    c_obj = -np.ones(n)
    
    # Constraints:
    # 1. r_i + r_j <= distance(c_i, c_j) for all i < j
    # 2. r_i <= x_i
    # 3. r_i <= 1 - x_i
    # 4. r_i <= y_i
    # 5. r_i <= 1 - y_i
    
    # We build A_ub and b_ub.
    # To compute forces, we need to know which constraint corresponds to which pair/bound.
    # We will store metadata.
    
    num_pairs = (n * (n - 1)) // 2
    num_bounds = 4 * n
    num_constraints = num_pairs + num_bounds
    
    A_ub = np.zeros((num_constraints, n))
    b_ub = np.zeros(num_constraints)
    
    # Metadata for force calculation
    # Store (type, index1, index2_or_type, value_for_grad)
    # For pairs: ('pair', i, j, None)
    # For bounds: ('bound', i, type_str, grad_factor)
    # grad_factor: derivative of RHS w.r.t center coordinate.
    # x: (1, 0), 1-x: (-1, 0), y: (0, 1), 1-y: (0, -1)
    
    constraint_meta = []
    
    idx = 0
    # Pairwise constraints
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub[idx] = row
            
            dist = np.linalg.norm(centers[i] - centers[j])
            b_ub[idx] = dist
            
            constraint_meta.append(('pair', i, j, None))
            idx += 1
            
    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        
        # r_i <= x  (RHS = x, grad_x = 1, grad_y = 0)
        row = np.zeros(n); row[i] = 1.0
        A_ub[idx] = row
        b_ub[idx] = x
        constraint_meta.append(('bound', i, 'x_low', (1.0, 0.0)))
        idx += 1
        
        # r_i <= 1 - x (RHS = 1-x, grad_x = -1, grad_y = 0)
        row = np.zeros(n); row[i] = 1.0
        A_ub[idx] = row
        b_ub[idx] = 1.0 - x
        constraint_meta.append(('bound', i, 'x_high', (-1.0, 0.0)))
        idx += 1
        
        # r_i <= y (RHS = y, grad_x = 0, grad_y = 1)
        row = np.zeros(n); row[i] = 1.0
        A_ub[idx] = row
        b_ub[idx] = y
        constraint_meta.append(('bound', i, 'y_low', (0.0, 1.0)))
        idx += 1
        
        # r_i <= 1 - y (RHS = 1-y, grad_x = 0, grad_y = -1)
        row = np.zeros(n); row[i] = 1.0
        A_ub[idx] = row
        b_ub[idx] = 1.0 - y
        constraint_meta.append(('bound', i, 'y_high', (0.0, -1.0)))
        idx += 1
        
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    except Exception:
        return None, None, None
        
    if not res.success:
        return None, None, None
        
    radii = res.x
    sum_radii = -res.fun
    
    # Compute forces
    # Force on center k is gradient of sum_radii w.r.t center_k
    # sum_radii = -objective_value
    # d(sum_radii)/d(center) = - d(objective)/d(center)
    # d(objective)/d(center) = sum(marginal_m * d(b_m)/d(center))
    # So Force = - sum(marginal_m * grad_RHS_m)
    # Note: linprog marginals for A_ub x <= b_ub are typically <= 0 for minimization.
    # But we just follow the formula.
    
    forces = np.zeros((n, 2))
    marginals = res.ineqlin.marginals
    
    for idx, meta in enumerate(constraint_meta):
        lam = marginals[idx]
        if abs(lam) < 1e-9:
            continue
            
        # Coefficient for force is -lam
        coeff = -lam
        
        if meta[0] == 'pair':
            _, i, j, _ = meta
            vec = centers[i] - centers[j]
            dist = np.linalg.norm(vec)
            if dist > 1e-9:
                unit_vec = vec / dist
                # Gradient of distance ||c_i - c_j|| w.r.t c_i is unit_vec
                # Gradient w.r.t c_j is -unit_vec
                # Force contribution: coeff * grad
                forces[i, 0] += coeff * unit_vec[0]
                forces[i, 1] += coeff * unit_vec[1]
                forces[j, 0] += coeff * (-unit_vec[0])
                forces[j, 1] += coeff * (-unit_vec[1])
        else:
            # Bound
            _, i, btype, grad = meta
            gx, gy = grad
            forces[i, 0] += coeff * gx
            forces[i, 1] += coeff * gy
            
    return radii, sum_radii, forces
