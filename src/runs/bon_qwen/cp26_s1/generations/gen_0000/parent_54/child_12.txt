import numpy as np
from scipy.optimize import minimize


def run_packing():
    n = 26

    best_sum = 0.0
    best_centers = None
    best_radii = None

    for trial in range(20):
        # Create hexagonal pattern with varying parameters
        centers = []
        base_dx = 0.215
        dx = base_dx + (trial % 8) * 0.004
        dy = dx * np.sqrt(3) / 2

        base_y = 0.115 + (trial % 7) * 0.009
        base_x = 0.115 + (trial // 7) * 0.005
        y = base_y
        row = 0
        while len(centers) < n:
            x = base_x + (dx / 2 if row % 2 == 1 else 0)
            while x <= 0.88 and len(centers) < n:
                centers.append([x, y])
                x += dx
            y += dy
            row += 1

        centers = np.array(centers[:n])
        radii = np.full(n, 0.015)

        x0 = np.concatenate([centers.flatten(), radii])

        def objective(x):
            return -np.sum(x[2 * n:])

        def cone_constraints(x):
            c = x[:2 * n].reshape(n, 2)
            r = x[2 * n:]
            out = []
            for i in range(n):
                for j in range(i + 1, n):
                    d2 = np.sum((c[i] - c[j]) ** 2)
                    s = r[i] + r[j]
                    out.append(d2 - s * s + 1e-13)
            return np.array(out)

        def box_constraints(x):
            c = x[:2 * n].reshape(n, 2)
            r = x[2 * n:]
            out = []
            for i in range(n):
                out.append(c[i, 0] - r[i])
                out.append(1.0 - c[i, 0] - r[i])
                out.append(c[i, 1] - r[i])
                out.append(1.0 - c[i, 1] - r[i])
                out.append(r[i] - 1e-7)
            return np.array(out)

        bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
        constraints = [
            {"type": "ineq", "fun": cone_constraints},
            {"type": "ineq", "fun": box_constraints}
        ]

        try:
            res = minimize(
                objective, x0, method="SLSQP", bounds=bounds,
                constraints=constraints,
                options={"maxiter": 2000, "ftol": 1e-16, "disp": False}
            )

            if res.fun < 0:
                c_opt = res.x[:2 * n].reshape(n, 2)
                r_opt = res.x[2 * n:]
                s = np.sum(r_opt)

                # Quick validity check
                valid = True
                for i in range(n):
                    if r_opt[i] < 1e-7:
                        valid = False
                        break
                    if c_opt[i, 0] < r_opt[i] - 1e-10 or c_opt[i, 0] > 1.0 - r_opt[i] + 1e-10:
                        valid = False
                        break
                    if c_opt[i, 1] < r_opt[i] - 1e-10 or c_opt[i, 1] > 1.0 - r_opt[i] + 1e-10:
                        valid = False
                        break
                if valid:
                    for i in range(n):
                        for j in range(i + 1, n):
                            dist = np.sqrt(np.sum((c_opt[i] - c_opt[j]) ** 2))
                            if dist < r_opt[i] + r_opt[j] - 1e-10:
                                valid = False
                                break
                        if not valid:
                            break
                if valid and s > best_sum:
                    best_sum = s
                    best_centers = c_opt.copy()
                    best_radii = r_opt.copy()
        except Exception:
            pass

    if best_centers is None:
        # Fallback: manual hexagonal placement with reasonable radii
        centers = []
        dx = 0.22
        dy = dx * np.sqrt(3) / 2
        y = 0.12
        row = 0
        while len(centers) < n:
            x = 0.12 + (dx / 2 if row % 2 == 1 else 0)
            while x <= 0.88 and len(centers) < n:
                centers.append([x, y])
                x += dx
            y += dy
            row += 1
        centers = np.array(centers[:n])
        radii = np.full(n, 0.095)
        best_centers = centers
        best_radii = radii
        best_sum = np.sum(radii)

    return best_centers, best_radii, best_sum
