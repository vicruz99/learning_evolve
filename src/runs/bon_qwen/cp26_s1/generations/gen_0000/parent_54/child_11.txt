import numpy as np
from scipy.optimize import minimize

def compute_obj(vars, n):
    """Objective function: minimize (-sum_radii + penalty_for_overlaps)"""
    c = vars[:2*n].reshape(-1, 2)
    r = vars[2*n:]
    pen = 0.0
    for i in range(n):
        for j in range(i+1, n):
            d = np.sqrt(np.sum((c[i] - c[j])**2))
            if d < r[i] + r[j]:
                pen += (r[i] + r[j] - d)**2
        ri = r[i]
        x, y = c[i]
        pen += max(0, ri - x)**2 + max(0, x - (1 - ri))**2
        pen += max(0, ri - y)**2 + max(0, y - (1 - ri))**2
    return -np.sum(r) + 5000.0 * pen

def run_force_phase(centers, radii, n, iters=6000):
    """Grow circles using force-directed relaxation"""
    alpha = 0.15
    beta = 0.0004
    for it in range(iters):
        forces = np.zeros_like(centers)
        overlap = 0.0
        
        for i in range(n):
            for j in range(i+1, n):
                diff = centers[i] - centers[j]
                dist = np.sqrt(np.sum(diff**2))
                if dist < 1e-9:
                    dist = 1e-9
                    diff = np.random.rand(2) * 1e-5
                req = radii[i] + radii[j]
                if dist < req:
                    push = diff / dist * (req - dist)
                    forces[i] += push
                    forces[j] -= push
                    overlap += (req - dist)**2
                    
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x < r:
                forces[i, 0] += r - x
                overlap += (r - x)**2
            if x > 1 - r:
                forces[i, 0] -= x - (1 - r)
                overlap += (x - (1 - r))**2
            if y < r:
                forces[i, 1] += r - y
                overlap += (r - y)**2
            if y > 1 - r:
                forces[i, 1] -= y - (1 - r)
                overlap += (y - (1 - r))**2
                
        centers += forces * alpha
        centers = np.clip(centers, 0, 1)
        
        if overlap < 1e-6:
            radii += beta
            beta *= 0.999
            alpha *= 0.999
            
        if it % 500 == 0:
            centers += np.random.randn(*centers.shape) * 0.0002
            
    return centers, radii

def run_packing():
    n = 26
    best_obj = 1e9
    best_vars = None
    
    for seed in range(4):
        np.random.seed(seed)
        # Hexagonal initial arrangement
        centers = []
        rows = [6, 5, 6, 5, 4]
        y = 0.25
        for cnt in rows:
            x = 0.25
            for i in range(cnt):
                centers.append([x + i * 0.12, y])
            y += 0.15
        centers = np.array(centers[:n])
        centers += np.random.randn(*centers.shape) * 0.01
        radii = np.full(n, 0.01)
        
        # Phase 1: Force-directed growth
        centers, radii = run_force_phase(centers, radii, n, iters=6000)
        
        # Phase 2: Gradient-based polish
        vars0 = np.concatenate([centers.flatten(), radii])
        bounds = [(0, 1)] * (2 * n) + [(0, 0.5)] * n
        res = minimize(compute_obj, vars0, args=(n,), method='L-BFGS-B', 
                       bounds=bounds, options={'maxiter': 5000, 'ftol': 1e-12})
        
        if res.fun < best_obj:
            best_obj = res.fun
            best_vars = res.x
            
    final_c = best_vars[:2*n].reshape(-1, 2)
    final_r = best_vars[2*n:]
    
    # Final clipping to guarantee boundary constraints
    for i in range(n):
        r = final_r[i]
        final_c[i, 0] = np.clip(final_c[i, 0], r, 1 - r)
        final_c[i, 1] = np.clip(final_c[i, 1], r, 1 - r)
        
    return final_c, final_r, np.sum(final_r)
