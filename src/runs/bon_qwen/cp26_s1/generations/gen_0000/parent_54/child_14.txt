def get_overlap_energy(centers, radii):
    """
    Calculates the energy based on overlaps and boundary violations.
    Lower energy is better. Energy = 0 implies a valid packing.
    """
    n = len(radii)
    energy = 0.0
    
    # 1. Pairwise Overlap Penalty
    # Vectorized distance calculation
    # centers shape: (n, 2)
    # diff shape: (n, n, 2)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    # radii_sum shape: (n, n)

    positive_overlap = np.maximum(0, overlap[mask])
    
    # Squared penalty
    energy += np.sum(positive_overlap**2) * 1000 # High weight for overlap
    
    # 2. Boundary Violation Penalty
    # Centers must be within [r, 1-r]
    x = centers[:, 0]
    y = centers[:, 1]
    r = radii
    
    # Left wall: x >= r  => r - x <= 0. Penalty if r > x
    viol_left = np.maximum(0, r - x)
    # Right wall: x <= 1-r => x + r - 1 <= 0. Penalty if x + r > 1
    viol_right = np.maximum(0, x + r - 1)
    # Bottom wall: y >= r => r - y <= 0. Penalty if r > y
    viol_bottom = np.maximum(0, r - y)
    # Top wall: y <= 1-r => y + r - 1 <= 0. Penalty if y + r > 1
    viol_top = np.maximum(0, y + r - 1)
    
    boundary_violation = np.sum(viol_left**2) + np.sum(viol_right**2) + \
                         np.sum(viol_bottom**2) + np.sum(viol_top**2)
    
    energy += boundary_violation * 1000 # High weight for boundary
    
    return energy
