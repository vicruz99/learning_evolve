import numpy as np
from scipy.optimize import minimize

# Helper functions to access variables
# Variables array layout: [x1, y1, r1, x2, y2, r2, ..., x26, y26, r26]
def get_x(vars, i):
    return vars[3 * i]

def get_y(vars, i):
    return vars[3 * i + 1]

def get_r(vars, i):
    return vars[3 * i + 2]

# Top-level constraint functions
def constraint_left(vars, i):
    """Circle i must be to the right of the left boundary."""
    return get_x(vars, i) - get_r(vars, i)

def constraint_right(vars, i):
    """Circle i must be to the left of the right boundary."""
    return 1.0 - get_x(vars, i) - get_r(vars, i)

def constraint_bottom(vars, i):
    """Circle i must be above the bottom boundary."""
    return get_y(vars, i) - get_r(vars, i)

def constraint_top(vars, i):
    """Circle i must be below the top boundary."""
    return 1.0 - get_y(vars, i) - get_r(vars, i)

def constraint_overlap(vars, i, j):
    """Circles i and j must not overlap."""
    xi = get_x(vars, i)
    yi = get_y(vars, i)
    ri = get_r(vars, i)
    
    xj = get_x(vars, j)
    yj = get_y(vars, j)
    rj = get_r(vars, j)
    
    dist = np.sqrt((xi - xj)**2 + (yi - yj)**2)
    return dist - ri - rj

def objective(vars):
    """Objective: Maximize sum of radii -> Minimize negative sum."""

    return -np.sum(vars[2::3])
