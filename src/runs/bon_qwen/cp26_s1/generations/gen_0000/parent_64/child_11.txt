import numpy as np
import scipy.optimize as opt
import math

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    """
    N = 26
    LAMBDA = 10000.0  # Penalty weight for constraints
    
    # --- Initialization ---
    # Generate a hexagonal packing pattern as a starting point.
    # We start with a small radius to ensure validity, then the optimizer will expand them.
    r_init = 0.08
    centers = []
    radii = []
    
    # Hexagonal packing parameters
    # Vertical spacing between rows
    dy = math.sqrt(3) * r_init
    # Horizontal spacing within a row
    dx = 2 * r_init
    
    row = 0
    count = 0
    while count < N:
        y = r_init + row * dy
        
        # Check if row fits in Y bounds (rough check)
        if y - r_init < -0.01 or y + r_init > 1.01:
            row += 1
            continue
            
        # Determine x start based on row parity (staggered)
        if row % 2 == 0:
            x_start = r_init
        else:
            x_start = 2 * r_init
            
        x = x_start
        while x <= 1.0 - r_init and count < N:
            centers.append([x, y])
            radii.append(r_init)
            count += 1
            x += dx
        row += 1
        
    # Convert to numpy arrays
    centers = np.array(centers)
    radii = np.array(radii)
    
    # If we didn't get 26 circles (should not happen with r=0.08), fallback to grid
    if len(centers) < N:
        centers = []
        radii = []
        r_grid = 0.05
        for i in range(5):
            for j in range(6):
                if len(centers) < N:
                    centers.append([r_grid + i * (1 - 2*r_grid)/4, r_grid + j * (1 - 2*r_grid)/5])
                    radii.append(r_grid)
        centers = np.array(centers)
        radii = np.array(radii)

    # Flatten for optimizer
    # Vector: [x0, y0, r0, x1, y1, r1, ...]
    x0 = np.hstack([centers.flatten(), radii])
    
    # Bounds: x, y in [0, 1], r in [0, 0.5]
    bounds = []
    for _ in range(N):
        bounds.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
        
    def loss_function(vars):
        """
        Objective: Minimize -sum(radii) + penalty
        """
        # Reshape
        centers = vars[:2*N].reshape(N, 2)
        radii = vars[2*N:]
        
        # Base objective: negative sum of radii
        obj = -np.sum(radii)
        
        # Penalty term
        penalty = 0.0
        
        # Boundary penalties
        # x - r >= 0  => x - r
        # x + r <= 1  => 1 - x - r
        # y - r >= 0  => y - r
        # y + r <= 1  => 1 - y - r
        
        # Vectorized boundary checks
        # Left
        violations = radii - centers[:, 0]
        penalty += np.sum(np.maximum(0, violations)**2)
        # Right
        violations = radii - (1.0 - centers[:, 0])
        penalty += np.sum(np.maximum(0, violations)**2)
        # Bottom
        violations = radii - centers[:, 1]
        penalty += np.sum(np.maximum(0, violations)**2)
        # Top
        violations = radii - (1.0 - centers[:, 1])
        penalty += np.sum(np.maximum(0, violations)**2)
        
        # Overlap penalties
        # dist >= r_i + r_j
        # We iterate pairs. Since N=26, N(N-1)/2 = 325 pairs.
        # Vectorized approach might be memory heavy but N is small.
        # dist matrix
        diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :] # (N, N, 2)
        dists = np.sqrt(np.sum(diff**2, axis=2)) # (N, N)
        
        # Radius sum matrix
        rad_sum = radii[:, np.newaxis] + radii[np.newaxis, :] # (N, N)
        
        # Overlap amount
        overlap = rad_sum - dists
        
        # Triangular mask to count each pair once
        tri_mask = np.triu(np.ones((N, N), dtype=bool), k=1)
        overlap_upper = overlap[tri_mask]
        
        penalty += np.sum(np.maximum(0, overlap_upper)**2)
        
        return obj + LAMBDA * penalty

    # Run optimization
    # Maxiter should be sufficient to converge
    res = opt.minimize(
        loss_function, 
        x0, 
        method='L-BFGS-B', 
        bounds=bounds,
        options={'maxiter': 2000, 'ftol': 1e-12, 'gtol': 1e-8}
    )
    
    # Extract results
    final_vars = res.x
    final_centers = final_vars[:2*N].reshape(N, 2)
    final_radii = final_vars[2*N:]
    
    # Validation and correction
    # The optimizer might leave tiny violations due to numerical precision or local minima
    # We can try to shrink radii slightly if needed, but usually high lambda handles it.
    # Let's check validity manually and clamp if necessary.
    
    # Check for negative radii
    final_radii = np.maximum(final_radii, 0.0)
    
    # Ensure centers are within bounds (clip)
    final_centers = np.clip(final_centers, 0.0, 1.0)
    
    # Final sum
    sum_radii = np.sum(final_radii)
    
    return (final_centers, final_radii, sum_radii)
