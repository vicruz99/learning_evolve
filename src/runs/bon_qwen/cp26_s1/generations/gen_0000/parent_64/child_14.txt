import numpy as np
from scipy.optimize import linprog, minimize

def compute_sum_radii(centers):
    """
    Solve LP to find maximum radii for fixed centers.
    Returns (sum_radii, radii_array)
    """
    n = centers.shape[0]
    c_obj = -np.ones(n)  # Minimize negative sum => maximize sum
    A_ub = []
    b_ub = []

    # Pairwise non-overlap constraints: r_i + r_j <= d_ij
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1
            row[j] = 1
            A_ub.append(row)
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            b_ub.append(np.sqrt(dx*dx + dy*dy))

    # Boundary constraints: r_i <= dist to each wall
    for i in range(n):
        x = centers[i, 0]
        y = centers[i, 1]
        for dist in (x, 1 - x, y, 1 - y):
            row = np.zeros(n)
            row[i] = 1
            A_ub.append(row)
            b_ub.append(max(0.0, dist))

    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None)] * n

    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return -res.fun, res.x
    return 0.0, np.zeros(n)

def objective(centers_flat):
    """Objective function for scipy optimizer (minimizes negative sum of radii)"""
    centers = centers_flat.reshape(26, 2)
    val, _ = compute_sum_radii(centers)
    return -val

def run_packing():
    n = 26
    best_val = 0.0
    best_c = None
    best_r = None

    # Multiple restarts to escape local optima
    for seed in range(20):
        np.random.seed(seed)
        init_centers = np.random.rand(n, 2) * 0.7 + 0.15

        # Use hexagonal grid for every 4th seed to bias towards dense packing
        if seed % 4 == 0:
            s = 0.18
            pts = []
            for r in range(5):
                for c in range(5):
                    pts.append([c * s + (r % 2) * s / 2, r * s * np.sqrt(3) / 2 + 0.05])
            if len(pts) >= n:
                init_centers = np.array(pts[:n])

        # Powell method handles non-smooth objectives well
        res = minimize(objective, init_centers.flatten(), method='Powell',
                       options={'maxiter': 500, 'ftol': 1e-10, 'xtol': 1e-10})

        if -res.fun > best_val:
            best_val = -res.fun
            best_c = res.x.reshape(n, 2)
            _, best_r = compute_sum_radii(best_c)

    # Ensure strict validity by recomputing radii on the final optimized centers
    final_sum, final_r = compute_sum_radii(best_c)
    return best_c, final_r, final_sum
