import numpy as np
from scipy.optimize import minimize

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square

    Args:
        centers: np.array of shape (n, 2) with (x, y) coordinates
        radii: np.array of shape (n) with radius of each circle

    Returns:
        True if valid, False otherwise
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True

def get_constraints_and_jac(vars, n):
    """
    Helper to generate constraints for the optimizer.
    vars: array of [x0, y0, r0, x1, y1, r1, ...]
    n: number of circles
    """
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

        centers[i, 0] = vars[3*i]
        centers[i, 1] = vars[3*i + 1]

    constraints = []
    
    # 1. Boundary constraints: r <= x <= 1-r  => x - r >= 0, 1 - x - r >= 0
    # 2. Boundary constraints: r <= y <= 1-r  => y - r >= 0, 1 - y - r >= 0
    # 3. Non-overlap: dist >= r_i + r_j
    # 4. Radii non-negative: r >= 0 (handled by bounds mostly, but added for safety if needed, though bounds are preferred)
    
    # We will use 'ineq' constraints (func(x) >= 0)
    
    # Boundary constraints
    # x - r >= 0
    for i in range(n):
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, idx=i: v[3*idx] - v[3*idx + 2],
            'jac': lambda v, idx=i: np.zeros_like(v) if False else _jac_boundary_x_minus_r(v, idx) # Placeholder
        })
        # 1 - x - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, idx=i: 1.0 - v[3*idx] - v[3*idx + 2],
            'jac': lambda v, idx=i: _jac_boundary_1_minus_x_minus_r(v, idx)
        })
        # y - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, idx=i: v[3*idx + 1] - v[3*idx + 2],
            'jac': lambda v, idx=i: _jac_boundary_y_minus_r(v, idx)
        })
        # 1 - y - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, idx=i: 1.0 - v[3*idx + 1] - v[3*idx + 2],
            'jac': lambda v, idx=i: _jac_boundary_1_minus_y_minus_r(v, idx)
        })
        
    # Overlap constraints
    # dist^2 >= (ri + rj)^2  =>  dist^2 - (ri + rj)^2 >= 0
    # To avoid sqrt in constraint function, use squared distance.
    # However, Jacobian of dist^2 - (ri+rj)^2 is simpler.
    # Constraint: (xi-xj)^2 + (yi-yj)^2 - (ri+rj)^2 >= 0
    
    for i in range(n):
        for j in range(i + 1, n):
            idx_i = 3*i
            idx_j = 3*j
            
            def constr_func(v, i=i, j=j):
                xi, yi, ri = v[3*i], v[3*i+1], v[3*i+2]
                xj, yj, rj = v[3*j], v[3*j+1], v[3*j+2]
                return (xi-xj)**2 + (yi-yj)**2 - (ri+rj)**2
            
            def constr_jac(v, i=i, j=j):
                grad = np.zeros_like(v)
                xi, yi, ri = v[3*i], v[3*i+1], v[3*i+2]
                xj, yj, rj = v[3*j], v[3*j+1], v[3*j+2]
                
                # d/dxi: 2(xi - xj)
                grad[3*i] = 2 * (xi - xj)
                # d/dyi: 2(yi - yj)
                grad[3*i + 1] = 2 * (yi - yj)
                # d/dri: -2(ri + rj)
                grad[3*i + 2] = -2 * (ri + rj)
                
                # d/dxj: -2(xi - xj)
                grad[3*j] = -2 * (xi - xj)
                # d/dyj: -2(yi - yj)
                grad[3*j + 1] = -2 * (yi - yj)
                # d/drj: -2(ri + rj)
                grad[3*j + 2] = -2 * (ri + rj)
                return grad

            constraints.append({
                'type': 'ineq',
                'fun': constr_func,
                'jac': constr_jac
            })

    return constraints

def _jac_boundary_x_minus_r(v, idx):

def objective(vars):
    """Maximize sum of radii, so minimize negative sum."""
    n = len(vars) // 3
    sum_r = 0
    for i in range(n):
        sum_r += vars[3*i + 2]
    return -sum_r
