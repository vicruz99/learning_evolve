import numpy as np
from scipy.optimize import minimize

def compute_objective(params, N=26, lam=5000.0):
    """
    Computes the objective function: -sum(r_i) + penalty for overlaps.
    params: array of shape (3*N,) where [u_1, v_1, r_1, u_2, v_2, r_2, ...]
    """
    u = params[0::3]
    v = params[1::3]
    r = params[2::3]
    
    # Reconstruct centers using boundary-adaptive parameterization
    # x = r + (1-2r)*u  => ensures r <= x <= 1-r
    # y = r + (1-2r)*v  => ensures r <= y <= 1-r
    x = r + (1.0 - 2.0 * r) * u
    y = r + (1.0 - 2.0 * r) * v
    
    # Objective: maximize sum of radii -> minimize -sum(r)
    obj = -np.sum(r)
    
    # Penalty for overlaps
    # Compute pairwise distances vectorized
    X = x[:, None]
    Y = y[:, None]
    R = r[:, None]
    
    dx = X - X.T
    dy = Y - Y.T
    dist_sq = dx**2 + dy**2
    dist = np.sqrt(dist_sq)
    
    # Overlap matrix: r_i + r_j - dist_ij
    r_sum = R + R.T
    overlap = r_sum - dist
    
    # Only consider upper triangle (i < j)
    mask = np.triu(np.ones((N, N), dtype=bool), k=1)
    pos_overlap = np.maximum(overlap * mask, 0.0)
    
    penalty = np.sum(pos_overlap**2)
    
    return obj + lam * penalty

def run_packing():

    best_centers = None
    best_radii = None
    
    # Bounds: u, v in [0, 1], r in [eps, 0.5]

        params = np.zeros(3 * N)

                    break

            params[3 * i + 1] = np.clip(v, 0.0, 1.0)
            params[3 * i + 2] = r

    for init_params in inits:

        u = res.x[0::3]
        v = res.x[1::3]
