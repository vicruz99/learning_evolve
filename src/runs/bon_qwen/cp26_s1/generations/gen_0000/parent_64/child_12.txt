import numpy as np
from scipy.optimize import minimize

def compute_cost(v, n=26):
    """
    Computes the negative sum of radii plus a penalty for constraint violations.
    v: flattened array of [x1, y1, r1, x2, y2, r2, ...]
    """
    centers = np.array(v[:2*n]).reshape(n, 2)
    radii = np.array(v[2*n:])
    
    cost = -np.sum(radii)
    penalty_factor = 1000.0
    penalty = 0.0

    # Boundary constraints: x >= r, x <= 1-r, y >= r, y <= 1-r
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        
        # Check if r is negative (should be prevented by logic, but good for robustness)
        if r < 0:
            penalty += penalty_factor * r**2
            continue
            
        violations = [
            r - x,       # x < r
            r - (1 - x), # x > 1-r
            r - y,       # y < r
            r - (1 - y)  # y > 1-r
        ]
        for v_val in violations:
            if v_val > 0:
                penalty += v_val**2

    # Overlap constraints: dist >= r1 + r2
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            min_dist = radii[i] + radii[j]
            overlap = min_dist - dist
            if overlap > 0:
                penalty += overlap**2

    return cost + penalty_factor * penalty
