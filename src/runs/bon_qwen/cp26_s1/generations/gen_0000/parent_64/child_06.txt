from scipy.optimize import minimize
import math
import random

def initialize_hexagonal(n, radius=0.085):
    """Initialize circles in hexagonal pattern."""
    centers = np.zeros((n, 2))
    radii = np.full(n, radius)
    
    idx = 0
    y = radius
    row = 0
    
    while idx < n:
        num_in_row = min(n - idx, 6)
        if idx + num_in_row > n:
            num_in_row = n - idx
        
        total_width = (num_in_row - 1) * 2 * radius + 2 * radius
        x_start = (1 - total_width) / 2
        
        if row % 2 == 1:
            x_start += radius
        
        for col in range(num_in_row):
            if idx >= n:
                break
            x = x_start + col * 2 * radius
            centers[idx, 0] = x
            centers[idx, 1] = y
            idx += 1
        
        y += math.sqrt(3) * radius
        row += 1
    
    return centers, radii

def initialize_grid(n, rows=None, cols=None):
    """Initialize circles in grid pattern."""
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    if rows is None or cols is None:
        cols = int(math.ceil(math.sqrt(n)))
        rows = int(math.ceil(n / cols))
    
    spacing_x = 1.0 / (cols + 1)
    spacing_y = 1.0 / (rows + 1)
    
    idx = 0
    for r in range(rows):
        for c in range(cols):
            if idx >= n:
                break
            x = (c + 1) * spacing_x
            y = (r + 1) * spacing_y
            centers[idx, 0] = x
            centers[idx, 1] = y
            r_est = min(spacing_x, spacing_y) / 2 * 0.9
            radii[idx] = r_est
            idx += 1
        if idx >= n:
            break
    
    return centers, radii

def initialize_random(n):

def check_validity(centers, radii, n):
    """Check if configuration is valid."""
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if r < 0 or x - r < -1e-9 or x + r > 1 + 1e-9 or y - r < -1e-9 or y + r > 1 + 1e-9:
            return False
    
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-9:
                return False
    return True

def fix_violations(centers, radii, n):

        r = radii[i]
        x, y = centers[i]

        max_r_y = min(y, 1 - y)
