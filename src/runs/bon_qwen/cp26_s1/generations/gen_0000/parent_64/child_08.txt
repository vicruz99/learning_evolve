from scipy.optimize import minimize

def calculate_overlap_penalty(centers, radii, n):
    """Calculates a penalty score for overlaps and boundary violations."""
    penalty = 0.0
    overlap_count = 0
    n = len(radii)
    
    # 1. Check Inter-circle Overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            min_dist = radii[i] + radii[j]
            if dist < min_dist:
                violation = min_dist - dist
                # Quadratic penalty for overlaps to strongly discourage them
                penalty += 100.0 * (violation ** 2)
                overlap_count += 1
                
    # 2. Check Boundary Violations
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        # x - r >= 0  =>  r - x <= 0
        # x + r <= 1  =>  x + r - 1 <= 0
        # y - r >= 0
        # y + r <= 1
        
        violations = [
            radii[i] - x,
            x + radii[i] - 1.0,
            radii[i] - y,
            y + radii[i] - 1.0
        ]
        for v in violations:
            if v > 0:
                penalty += 100.0 * (v ** 2)

    return penalty, overlap_count

def objective_function(params, n):
    """Objective function to minimize: Negative sum of radii + Penalty."""
    # Reshape params into centers and radii
    # params structure: [x1, y1, r1, x2, y2, r2, ...]

    radii = params[2::3]
    
    sum_radii = np.sum(radii)

    # We want to maximize sum_radii, so we minimize -sum_radii
    # We add the penalty to force valid constraints
    return -sum_radii + penalty
