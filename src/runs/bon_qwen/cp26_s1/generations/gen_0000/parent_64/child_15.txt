Key aspects of the strategy:
1. **Parameterization**: Each circle $i$ is represented by variables $(x_i, y_i, r_i)$. This gives $26 \times 3 = 78$ optimization variables.
2. **Initialization**: A 5x5 grid layout plus one center circle provides a structured, feasible starting point. This mimics a high-density square packing which serves as a good baseline.
3. **Penalty Method**: Instead of explicit constraint handling (which can be fragile with many constraints), we use a penalty function added to the objective. The penalty grows quadratically with constraint violations (boundary exceedance and inter-circle overlaps). This turns it into a bound-constrained unconstrained optimization problem.

5. **Post-Processing**: After optimization, we strictly enforce feasibility by clipping radii to boundary limits and iteratively shrinking overlapping circles. A small tolerance buffer (`1e-9`) is added to guarantee the validation function's `1e-12` tolerance is comfortably satisfied.

import numpy as np
from scipy.optimize import minimize

def compute_penalty(centers, radii, penalty_weight):
    """Compute penalty for boundary violations and circle overlaps."""
    n = centers.shape[0]
    pen = 0.0
    
    # Boundary penalties: circles must stay within [0, 1]
    pen += np.sum(np.maximum(0, -centers[:, 0] + radii)**2)
    pen += np.sum(np.maximum(0, centers[:, 0] - 1 + radii)**2)
    pen += np.sum(np.maximum(0, -centers[:, 1] + radii)**2)
    pen += np.sum(np.maximum(0, centers[:, 1] - 1 + radii)**2)
    
    # Overlap penalties: distance between centers must be >= sum of radii
    dx = centers[:, 0][:, None] - centers[:, 0][None, :]
    dy = centers[:, 1][:, None] - centers[:, 1][None, :]
    dist = np.sqrt(dx**2 + dy**2)
    
    r_sum = radii[:, None] + radii[None, :]
    iu, ju = np.triu_indices(n, k=1)
    overlaps = np.maximum(0, r_sum[iu, ju] - dist[iu, ju])
    pen += penalty_weight * np.sum(overlaps**2)
    
    return pen

def objective(var, n, penalty_weight):
    """Objective function: minimize negative sum of radii plus penalties."""
    centers = var[:2*n].reshape(n, 2)
    radii = var[2*n:]
    return -np.sum(radii) + compute_penalty(centers, radii, penalty_weight)

def run_packing():
    """Pack 26 circles in a unit square to maximize sum of radii."""
    n = 26
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Base initialization: 5x5 grid + center point
    base_x = np.linspace(0.1, 0.9, 5)
    base_y = np.linspace(0.1, 0.9, 5)
    init_centers = np.array([[x, y] for x in base_x for y in base_y])
    init_centers = np.vstack([init_centers, [0.5, 0.5]])
    
    np.random.seed(42)
    
    # Multi-start optimization to escape poor local minima
    for trial in range(15):
        centers = init_centers.copy()
        centers += np.random.normal(0, 0.02, centers.shape)
        radii = np.full(n, 0.08 + np.random.normal(0, 0.01, n))
        radii = np.clip(radii, 0.01, 0.4)
        
        var0 = np.concatenate([centers.flatten(), radii])
        bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n
        
        res = minimize(objective, var0, args=(n, 5000), method='L-BFGS-B', 
                       bounds=bounds, options={'maxiter': 2000, 'ftol': 1e-15})
        
        c_opt = res.x[:2*n].reshape(n, 2)
        r_opt = res.x[2*n:]
        
        # Strictly enforce boundary constraints
        for i in range(n):
            r_opt[i] = min(r_opt[i], c_opt[i,0], 1-c_opt[i,0], c_opt[i,1], 1-c_opt[i,1])
            
        # Strictly enforce non-overlap constraints
        for i in range(n):
            for j in range(i+1, n):
                d = np.sqrt(np.sum((c_opt[i] - c_opt[j])**2))
                if d < r_opt[i] + r_opt[j]:
                    shrink = (r_opt[i] + r_opt[j] - d) * 0.5 + 1e-9
                    r_opt[i] = max(0.0, r_opt[i] - shrink)
                    r_opt[j] = max(0.0, r_opt[j] - shrink)
                    
        s = np.sum(r_opt)
        if s > best_sum:
            best_sum = s
            best_centers = c_opt.copy()
            best_radii = r_opt.copy()
            
    return best_centers, best_radii, float(best_sum)
