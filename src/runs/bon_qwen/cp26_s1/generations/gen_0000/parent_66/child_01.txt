def initialize_hexagonal(n, variant):
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.09)

    # Different hexagonal arrangements
    row_counts = [
        [6, 5, 6, 5, 4],
        [5, 6, 5, 6, 4],
        [5, 5, 6, 5, 5],
        [6, 6, 6, 5, 3],
        [4, 6, 6, 6, 4],
        [5, 5, 5, 5, 6],
        [6, 4, 6, 6, 4],
        [5, 6, 4, 6, 5],
    ]

    rc = row_counts[variant % len(row_counts)]

    idx = 0
    row = 0
    r_est = 0.09
    while row < len(rc) and idx < n:
        num_in_row = rc[row]
        y = r_est + row * r_est * math.sqrt(3)
        x_start = r_est + (r_est if row % 2 == 1 else 0)
        spacing = (1 - 2 * r_est) / (num_in_row - 1) if num_in_row > 1 else 0
        for col in range(num_in_row):
            if idx >= n:
                break
            x = x_start + col * spacing
            centers[idx] = [x, y]
            idx += 1
        row += 1

    # Fill remaining if any
    while idx < n:
        centers[idx] = [0.5, 0.5]
        idx += 1

    # Add small perturbation based on variant
    centers += np.random.RandomState(variant).randn(n, 2) * 0.003

    return centers, radii


def initialize_grid(n, seed):
    np.random.seed(seed)
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.085)

    # Grid initialization with some randomness
    rows = 6
    cols = 5
    idx = 0
    for i in range(rows):
        for j in range(cols):
            if idx >= n:
                break
            x = 0.1 + j * 0.18 + np.random.uniform(-0.02, 0.02)
            y = 0.1 + i * 0.16 + np.random.uniform(-0.02, 0.02)
            x = np.clip(x, 0.05, 0.95)
            y = np.clip(y, 0.05, 0.95)
            centers[idx] = [x, y]
            idx += 1

    while idx < n:
        centers[idx] = [np.random.uniform(0.1, 0.9), np.random.uniform(0.1, 0.9)]
        idx += 1

    return centers, radii


def initialize_random(n, seed):
    np.random.seed(seed)
    centers = np.random.uniform(0.1, 0.9, (n, 2))
    radii = np.full(n, 0.07)
    return centers, radii


def optimize_packing(x0, n):
    bounds = [(0.001, 0.999)] * (2 * n) + [(0.001, 0.4)] * n

    result = minimize(
        penalty_objective,
        x0,
        args=(n,),
        method="L-BFGS-B",
        bounds=bounds,
        options={"maxiter": 100000, "ftol": 1e-15, "gtol": 1e-10},
    )

    return result


def penalty_objective(x, n):
    centers = x[: 2 * n].reshape(n, 2)
    radii = x[2 * n :]

    # Objective: maximize sum of radii (negate for minimization)
    obj = -np.sum(radii)

    # Penalty weight
    W = 5e6

    # Boundary constraint penalties
    for i in range(n):
        x_i = centers[i, 0]
        y_i = centers[i, 1]
        r_i = radii[i]

        # x_i >= r_i
        v1 = r_i - x_i
        if v1 > 0:
            obj += W * v1 * v1

        # x_i + r_i <= 1
        v2 = x_i + r_i - 1.0
        if v2 > 0:
            obj += W * v2 * v2

        # y_i >= r_i
        v3 = r_i - y_i
        if v3 > 0:
            obj += W * v3 * v3

        # y_i + r_i <= 1
        v4 = y_i + r_i - 1.0
        if v4 > 0:
            obj += W * v4 * v4

        # r_i >= 0
        if r_i < 0:
            obj += W * r_i * r_i

    # Overlap constraint penalties
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = math.sqrt(dx * dx + dy * dy)
            min_dist = radii[i] + radii[j]

            if dist < min_dist:
                obj += W * (min_dist - dist) * (min_dist - dist)

    return obj


def is_valid(centers, radii, n):
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-10 or x + r > 1 + 1e-10:
            return False
        if y - r < -1e-10 or y + r > 1 + 1e-10:
            return False
        if r < 0:
            return False

    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = math.sqrt(dx * dx + dy * dy)
            if dist < radii[i] + radii[j] - 1e-10:
                return False

    return True
