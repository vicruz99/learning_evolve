

Looking at this circle packing problem, I need to pack 26 circles in a unit square to maximize the sum of radii. The target is 2.636, which corresponds to an average radius of ~0.1014. From known circle packing literature, the optimal packing of 26 equal circles has radius ~0.1015, giving sum ≈ 2.639.

<strategy>
1. **Initialization**: Place circles in a hexagonal grid pattern, which is the densest packing pattern in 2D. For 26 circles, use row distributions like [6,5,6,5,4] with alternating offsets.

2. **Optimization Approach**: Use penalty-based optimization with L-BFGS-B. The objective maximizes sum of radii while penalizing boundary violations, overlaps, and negative radii. The penalty method converts constrained optimization into unconstrained form.

3. **Multiple Restarts**: Run optimization from 30 different initial configurations (varying row distributions and spacing) to escape local optima.

4. **Validation**: After each optimization, validate the solution meets all constraints before accepting it.

5. **Key Insight**: The hexagonal arrangement naturally leads to near-optimal configurations. The penalty weight must be high enough to enforce constraints but not so high as to cause numerical issues.

6. **Improvement**: After finding a valid configuration, attempt to grow all radii slightly and re-optimize positions to squeeze out more total radius.
</strategy>

import numpy as np
from scipy.optimize import minimize


def compute_penalty(params, n):
    """Compute penalty for constraint violations."""
    centers = params[:2 * n].reshape((n, 2))
    radii = params[2 * n:]

    penalty = 0.0

    for i in range(n):
        x = centers[i, 0]
        y = centers[i, 1]
        r = radii[i]

        # Boundary violations: circle must be inside [0,1]x[0,1]
        if x < r:
            penalty += (r - x) ** 2
        if x > 1 - r:
            penalty += (x - (1 - r)) ** 2
        if y < r:
            penalty += (r - y) ** 2
        if y > 1 - r:
            penalty += (y - (1 - r)) ** 2

    # Overlap penalties
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            min_dist = radii[i] + radii[j]
            if dist < min_dist:
                penalty += (min_dist - dist) ** 2

    # Non-negative radius penalties
    for i in range(n):
        if radii[i] < 0:
            penalty += radii[i] ** 2

    return penalty


def objective(params, n, weight):
    """Objective: minimize negative sum of radii plus weighted penalty."""
    radii = params[2 * n:]
    return -np.sum(radii) + weight * compute_penalty(params, n)


def init_hex(n, s):
    """Initialize circles in hexagonal grid pattern with seed s."""
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.085
    np.random.seed(s)

    # Different row distributions summing to 26
    configs = [
        [6, 5, 6, 5, 4],
        [5, 6, 5, 5, 5],
        [6, 6, 5, 5, 4],
        [5, 5, 6, 5, 5],
        [4, 6, 6, 5, 5],
        [6, 5, 5, 5, 5],
        [5, 5, 5, 6, 5],
        [5, 6, 6, 5, 4],
    ]
    
    cfg_idx = s % len(configs)
    rows = configs[cfg_idx]
    
    # Vary spacing slightly
    rs = 0.168 + (s % 5) * 0.003
    cs = 0.158 + (s % 5) * 0.002

    idx = 0
    for ri, cnt in enumerate(rows):
        y = 0.085 + ri * rs
        off = cs / 2 if ri % 2 else 0
        for ci in range(cnt):
            x = 0.085 + off + ci * cs
            centers[idx] = [x, y]
            idx += 1

    return centers, radii


def optimize(ci, ri, n, w=1e5):
    """Optimize circle packing from initial configuration."""
    p = np.concatenate([ci.flatten(), ri])
    res = minimize(
        objective, p, args=(n, w), method="L-BFGS-B",
        options={"maxiter": 30000, "ftol": 1e-15, "gtol": 1e-12, "history_size": 50}
    )
    c = res.x[:2 * n].reshape((n, 2))
    r = np.maximum(res.x[2 * n:], 0)
    return c, r


def valid(c, r, n):
    """Check if packing is valid."""
    for i in range(n):
        if r[i] < 0:
            return False
        if c[i, 0] - r[i] < -1e-11 or c[i, 0] + r[i] > 1 + 1e-11:
            return False
        if c[i, 1] - r[i] < -1e-11 or c[i, 1] + r[i] > 1 + 1e-11:
            return False
    for i in range(n):
        for j in range(i + 1, n):
            d = np.sqrt(np.sum((c[i] - c[j]) ** 2))
            if d < r[i] + r[j] - 1e-11:
                return False
    return True


def run_packing():
    """Pack 26 circles in unit square to maximize sum of radii."""
    n = 26
    bs = -1
    bc = br = None

    # Try multiple initial configurations
    for s in range(30):
        ci, ri = init_hex(n, s)
        c, r = optimize(ci, ri, n)
        sr = np.sum(r)
        if valid(c, r, n) and sr > bs:
            bs = sr
            bc, br = c.copy(), r.copy()

    # Fallback if no valid solution found
    if bc is None:
        ci, ri = init_hex(n, 0)
        bc, br = optimize(ci, ri, n)
        bs = np.sum(br)

    # Attempt to grow circles further from best solution
    # Gradually increase radii and re-optimize
    for _ in range(5):
        c, r = bc.copy(), br.copy()
        factor = 1.005
        r *= factor
        c, r = optimize(c, r, n, w=1e6)
        sr = np.sum(r)
        if valid(c, r, n) and sr > bs:
            bs = sr
            bc, br = c.copy(), r.copy()

    return bc, br, bs
