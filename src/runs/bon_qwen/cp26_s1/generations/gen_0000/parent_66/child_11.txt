def check_overlap_dist(v, i, j, n):
    """Helper to compute overlap constraint violation."""
    xi, yi, ri = v[3*i], v[3*i+1], v[3*i+2]
    xj, yj, rj = v[3*j], v[3*j+1], v[3*j+2]
    # Constraint: dist^2 - (ri + rj)^2 >= 0
    dist_sq = (xi - xj)**2 + (yi - yj)**2
    sum_r = ri + rj
    return dist_sq - sum_r**2
