import numpy as np
import scipy.optimize as opt

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    """
    n = 26
    best_centers = None
    best_radii = None
    best_sum = 0.0

    # Number of optimization restarts to avoid local minima
    n_restarts = 5
    n_iterations_per_restart = 2

    for restart in range(n_restarts):
        # --- Initialization: Perturbed Hexagonal Pattern ---
        centers = np.zeros((n, 2))
        radii = np.full(n, 0.05) # Start with small radii to ensure feasibility
        
        # Distribution of circles per row to approximate hex packing
        # 5 + 6 + 5 + 6 + 4 = 26
        row_counts = [5, 6, 5, 6, 4]
        current_idx = 0
        
        # Vertical spacing for hex packing (offset rows)
        # Height = 2r + (num_rows-1) * sqrt(3)r. 
        # We'll distribute y coordinates evenly first and let optimizer fix it.
        y_step = 1.0 / (len(row_counts) + 1)
        
        for row_idx, count in enumerate(row_counts):
            # Calculate y position for this row
            # Alternate y to create hexagonal nesting effect
            if row_idx % 2 == 0:
                y_pos = (row_idx + 1) * y_step
            else:
                y_pos = (row_idx + 1) * y_step + 0.5 * y_step * 0.1 # slight jitter
            
            # Calculate x positions
            # Distribute count circles across the width
            # Leave margin for radius
            if count > 0:
                # Horizontal spacing
                if row_idx % 2 == 1:
                    # Offset x for hex pattern
                    x_start = 0.5 / count + 0.01 * (restart + 1)
                else:
                    x_start = 0.5 / count

                for k in range(count):
                    x_pos = x_start + k * (1.0 - 2*x_start) / max(1, count - 1) if count > 1 else 0.5
                    # Apply slight random perturbation
                    x_pos += np.random.uniform(-0.02, 0.02)
                    y_pos += np.random.uniform(-0.02, 0.02)
                    
                    # Clip to valid range
                    centers[current_idx, 0] = np.clip(x_pos, 0.05, 0.95)
                    centers[current_idx, 1] = np.clip(y_pos, 0.05, 0.95)
                    current_idx += 1
        
        # Flatten variables for optimizer: [x1, y1, r1, x2, y2, r2, ...]
        x0 = np.zeros(3 * n)
        for i in range(n):
            x0[3*i] = centers[i, 0]
            x0[3*i+1] = centers[i, 1]
            x0[3*i+2] = 0.1 # Initial guess for radius

        # --- Optimization ---
        for _ in range(n_iterations_per_restart):
            try:
                # Define bounds: x, y in [0, 1], r in [0, 0.5]
                bounds = [(0, 1)] * 2 + [(0, 0.5)]
                bounds = (bounds * n)

                # Define constraints
                constraints = []
                
                # 1. Boundary constraints for each circle
                # x - r >= 0  => r - x <= 0
                # 1 - x - r >= 0 => r + x - 1 <= 0
                # y - r >= 0
                # 1 - y - r >= 0
                for i in range(n):
                    # Left
                    constraints.append({
                        'type': 'ineq',
                        'fun': lambda v, i=i: v[3*i] - v[3*i+2]
                    })
                    # Right
                    constraints.append({
                        'type': 'ineq',
                        'fun': lambda v, i=i: 1 - v[3*i] - v[3*i+2]
                    })
                    # Bottom
                    constraints.append({
                        'type': 'ineq',
                        'fun': lambda v, i=i: v[3*i+1] - v[3*i+2]
                    })
                    # Top
                    constraints.append({
                        'type': 'ineq',
                        'fun': lambda v, i=i: 1 - v[3*i+1] - v[3*i+2]
                    })

                # 2. Non-overlap constraints
                # dist(i, j) >= r_i + r_j
                # sqrt((xi-xj)^2 + (yi-yj)^2) - ri - rj >= 0
                for i in range(n):
                    for j in range(i + 1, n):
                        constraints.append({
                            'type': 'ineq',
                            'fun': lambda v, i=i, j=j: \
                                np.sqrt((v[3*i] - v[3*j])**2 + (v[3*i+1] - v[3*j+1])**2) - (v[3*i+2] + v[3*j+2])
                        })

                # Objective: Minimize negative sum of radii
                def objective(v):
                    return -sum(v[3*i+2] for i in range(n))

                # Run SLSQP
                result = opt.minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints,
                                      options={'ftol': 1e-9, 'maxiter': 500, 'disp': False})

                if result.success or result.fun < -best_sum:
                    current_sum = -result.fun
                    if current_sum > best_sum:
                        best_sum = current_sum

                        best_radii = np.array([result.x[3*i+2] for i in range(n)])

                # If optimization fails, continue to next iteration
                continue

    # Final validation check

    radii = best_radii

        radii = np.full(n, 0.01)
        best_sum = sum(radii)

    return centers, radii, float(best_sum)

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    n = centers.shape[0]
    if np.isnan(centers).any() or np.isnan(radii).any():
        return False
    for i in range(n):
        if radii[i] < 0: return False
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            return False
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False
    return True

# Run the packing function
if __name__ == "__main__":

    print(f"Valid packing: {validate_packing(centers, radii)}")
