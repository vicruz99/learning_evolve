import numpy as np
from scipy.optimize import minimize

def obj_func(v):
    # Minimize negative sum of radii (indices 2, 5, 8, ...)
    return -np.sum(v[2::3])

def constr_x_ge_r(v, xi, ri):
    return v[xi] - v[ri]

def constr_1_minus_x_minus_r(v, xi, ri):
    return 1.0 - v[xi] - v[ri]

def constr_y_ge_r(v, yi, ri):
    return v[yi] - v[ri]

def constr_1_minus_y_minus_r(v, yi, ri):
    return 1.0 - v[yi] - v[ri]

def constr_dist_sq(v, xi, yi, ri, xj, yj, rj):
    # Squared distance constraint: ||p_i - p_j||^2 >= (r_i + r_j)^2
    return (v[xi] - v[xj])**2 + (v[yi] - v[yj])**2 - (v[ri] + v[rj])**2

def run_packing():
    n = 26
    r_init = 0.08
    dx = 2 * r_init
    dy = np.sqrt(3) * r_init
    
    centers = []
    radii = []
    
    # Hexagonal row pattern summing to 26
    row_counts = [5, 4, 5, 4, 5, 3]
    
    for row_idx, count in enumerate(row_counts):
        y = r_init + row_idx * dy
        total_width = (count - 1) * dx
        margin = (1.0 - total_width) / 2
        x_start = margin
        
        for col in range(count):
            centers.append([x_start + col * dx, y])
            radii.append(r_init)
            
    centers = np.array(centers)
    radii = np.array(radii)
    x0 = np.concatenate([centers.flatten(), radii])
    
    # Bounds: x,y in [0,1], r in [1e-6, 0.5]
    bounds = [(0.0, 1.0) for _ in range(2*n)] + [(1e-6, 0.5) for _ in range(n)]
    
    constraints = []
    for i in range(n):
        xi, yi, ri = 3*i, 3*i+1, 3*i+2
        constraints.append({'type': 'ineq', 'fun': constr_x_ge_r, 'args': (xi, ri)})
        constraints.append({'type': 'ineq', 'fun': constr_1_minus_x_minus_r, 'args': (xi, ri)})
        constraints.append({'type': 'ineq', 'fun': constr_y_ge_r, 'args': (yi, ri)})
        constraints.append({'type': 'ineq', 'fun': constr_1_minus_y_minus_r, 'args': (yi, ri)})
        
    for i in range(n):
        for j in range(i+1, n):
            xi, yi, ri = 3*i, 3*i+1, 3*i+2
            xj, yj, rj = 3*j, 3*j+1, 3*j+2
            constraints.append({'type': 'ineq', 'fun': constr_dist_sq, 'args': (xi, yi, ri, xj, yj, rj)})
            
    res = minimize(obj_func, x0, method='SLSQP', bounds=bounds, constraints=constraints, 
                   options={'maxiter': 2000, 'ftol': 1e-10, 'disp': False})
                   
    opt_centers = res.x[:2*n].reshape((n, 2))
    opt_radii = res.x[2*n:]
    
    # Safety clamp for radii
    opt_radii = np.maximum(opt_radii, 0.0)
    
    return opt_centers, opt_radii, np.sum(opt_radii)
