import numpy as np

def compute_max_overlap(centers, radii):
    """Compute the maximum overlap between circles or with boundaries."""
    n = centers.shape[0]
    max_ov = 0.0
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            d = np.sqrt(dx*dx + dy*dy)
            ov = radii[i] + radii[j] - d
            if ov > max_ov:
                max_ov = ov
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < 0:
            ov = r - x
            if ov > max_ov: max_ov = ov
        if x + r > 1:
            ov = x + r - 1
            if ov > max_ov: max_ov = ov
        if y - r < 0:
            ov = r - y
            if ov > max_ov: max_ov = ov
        if y + r > 1:
            ov = y + r - 1
            if ov > max_ov: max_ov = ov
    return max_ov

def relax_packing(centers, radii, n_iter=150, alpha=0.03):
    """Vectorized force-based relaxation to resolve overlaps."""
    n = centers.shape[0]
    for _ in range(n_iter):
        diffs = centers[:, None, :] - centers[None, :, :]
        dists_sq = np.sum(diffs**2, axis=2)
        dists = np.sqrt(dists_sq + 1e-12)
        np.fill_diagonal(dists, 1.0)
        
        overlaps = radii[:, None] + radii[None, :] - dists
        overlaps = np.maximum(overlaps, 0)
        
        inv_dists = np.where(dists > 0, 1.0 / dists, 0.0)
        directions = diffs * inv_dists[:, :, None]
        forces = np.sum(overlaps[:, :, None] * directions, axis=1)
        
        fx = np.zeros(n)
        fy = np.zeros(n)
        mask_l = centers[:, 0] < radii
        fx[mask_l] = radii[mask_l] - centers[mask_l, 0]
        mask_r = centers[:, 0] > 1 - radii
        fx[mask_r] = (1 - radii[mask_r]) - centers[mask_r, 0]
        mask_b = centers[:, 1] < radii
        fy[mask_b] = radii[mask_b] - centers[mask_b, 1]
        mask_t = centers[:, 1] > 1 - radii
        fy[mask_t] = (1 - radii[mask_t]) - centers[mask_t, 1]
        
        forces[:, 0] += fx
        forces[:, 1] += fy
        
        centers += alpha * forces
        centers[:, 0] = np.clip(centers[:, 0], radii, 1 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1 - radii)
    return centers

def run_packing():
    """Pack 26 circles in a unit square to maximize sum of radii."""
    n = 26
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    for seed in range(5):
        np.random.seed(seed)
        # Hexagonal-like initialization
        centers = np.zeros((n, 2))
        idx = 0
        for row in range(6):
            for col in range(5):
                if idx >= n: break
                x = (col + 0.5 + (0.5 if row % 2 else 0)) / 5.0
                y = (row + 0.5) / 6.0
                centers[idx] = [x, y]
                idx += 1
                
        # Add small random perturbation to break symmetry
        centers += np.random.uniform(-0.015, 0.015, centers.shape)
        centers[:, 0] = np.clip(centers[:, 0], 0.05, 0.95)
        centers[:, 1] = np.clip(centers[:, 1], 0.05, 0.95)
        
        radii = np.full(n, 0.05)
        
        # Iterative expansion: grow radii and relax positions
        for step in range(2000):
            radii[:] += 0.000025
            centers = relax_packing(centers, radii, n_iter=100, alpha=0.03)
            
        # Verify and adjust if necessary
        max_ov = compute_max_overlap(centers, radii)
        if max_ov > 1e-9:
            # Uniformly scale down radii to eliminate remaining overlaps
            scale = 1.0 - max_ov / (np.sum(radii) + 1e-9)
            radii *= scale
            
        current_sum = np.sum(radii)
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers.copy()
            best_radii = radii.copy()
            
    return best_centers, best_radii, float(best_sum)
