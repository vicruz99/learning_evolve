import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Returns centers, radii, and sum_radii for a valid packing of 26 circles.
    """
    n = 26
    
    # Initial guess: 25 circles in a grid, 1 circle in a gap
    r_base = 0.1
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # 25 circles in a 5x5 grid
    indices = 0
    for i in range(5):
        for j in range(5):
            centers[indices] = [0.1 + i * 0.2, 0.1 + j * 0.2]
            radii[indices] = r_base
            indices += 1
            
    # 26th circle in the first gap (0.2, 0.2)
    centers[25] = [0.2, 0.2]
    radii[25] = 0.04
    
    # Optimization variables: [x1, y1, r1, ..., x26, y26, r26]
    # Flattened array of size 3 * n
    x0 = np.hstack([centers.flatten(), radii])
    
    # Constraints
    constraints = []
    
    # 1. Boundary constraints: x - r >= 0, 1 - x - r >= 0, etc.
    # x_i - r_i >= 0  =>  x_i - r_i >= 0
    # 1 - x_i - r_i >= 0
    # y_i - r_i >= 0
    # 1 - y_i - r_i >= 0
    
    def bound_constraint_x_neg(var):
        return var[0::3] - var[6::3] # x - r
    def bound_constraint_x_pos(var):
        return 1 - var[0::3] - var[6::3] # 1 - x - r
    def bound_constraint_y_neg(var):
        return var[1::3] - var[6::3] # y - r
    def bound_constraint_y_pos(var):
        return 1 - var[1::3] - var[6::3] # 1 - y - r
    
    constraints.append({'type': 'ineq', 'fun': bound_constraint_x_neg})
    constraints.append({'type': 'ineq', 'fun': bound_constraint_x_pos})
    constraints.append({'type': 'ineq', 'fun': bound_constraint_y_neg})
    constraints.append({'type': 'ineq', 'fun': bound_constraint_y_pos})
    
    # 2. Non-overlap constraints: dist(i, j) >= r_i + r_j
    # dist^2 >= (r_i + r_j)^2
    # (xi-xj)^2 + (yi-yj)^2 - (ri+rj)^2 >= 0
    
    def overlap_constraints(var):
        vals = []
        xs = var[0::3]
        ys = var[1::3]
        rs = var[2::3]
        for i in range(n):
            for j in range(i + 1, n):
                dx = xs[i] - xs[j]
                dy = ys[i] - ys[j]
                dr = rs[i] + rs[j]
                vals.append(dx*dx + dy*dy - dr*dr)
        return np.array(vals)
        
    constraints.append({'type': 'ineq', 'fun': overlap_constraints})
    
    # 3. Non-negative radii
    def radius_nonneg(var):
        return var[2::3]
        
    constraints.append({'type': 'ineq', 'fun': radius_nonneg})

    # Objective: Maximize sum of radii => Minimize -sum(radii)
    def objective(var):
        return -np.sum(var[2::3])

    # Run optimization
    # Using SLSQP as it handles constraints well
    # Bounds can help guide the search, though constraints are explicit
    bounds = [(0, 1) if k % 3 == 0 or k % 3 == 1 else (0, 1) for k in range(3 * n)]
    
    res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints, 
                   options={'ftol': 1e-12, 'maxiter': 2000, 'disp': False})
    
    # Extract results
    opt_centers = res.x[0::3].reshape(-1, 2)
    opt_radii = res.x[2::3]
    
    # Ensure no negative radii due to numerical noise
    opt_radii = np.maximum(opt_radii, 0)
    
    sum_radii = np.sum(opt_radii)
    
    return opt_centers, opt_radii, sum_radii
