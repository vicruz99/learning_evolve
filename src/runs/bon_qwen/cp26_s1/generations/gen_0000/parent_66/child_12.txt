This is a circle packing problem where we need to maximize the sum of radii of 26 circles in a unit square. Key insights:

1. **Sum of radii vs equal radii**: Unlike equal-circle packing, we want varying sizes. Large circles in corners/edges and smaller ones in gaps maximize the sum.

2. **Optimal structure**: 
   - 4 large circles in corners can each have radius ~0.22-0.25
   - 8-10 medium circles along edges can have radius ~0.08-0.12
   - 10-14 smaller circles filling interior gaps with radius ~0.03-0.08

3. **Optimization approach**: 
   - Use a two-phase method: (a) compute optimal radii for fixed centers, (b) optimize centers via local search
   - Start with structured initial placement (corners → edges → interior)
   - Iteratively improve by perturbing each circle's position and accepting moves that increase its radius
   - Use decreasing step sizes for convergence
   - Try multiple restarts from different configurations

4. **Key constraint**: For fixed centers, radii satisfy r_i = min(boundary_distances, min_j(dist_ij - r_j)), solved iteratively.

5. **Local search improvement**: For each circle, try random perturbations; accept if it increases that circle's achievable radius. This greedily improves the configuration.

import numpy as np


def compute_radii(centers, n, max_iter=3000):
    """Compute maximum radii for given fixed centers using iterative relaxation."""
    radii = np.full(n, 1e-10)
    
    for _ in range(max_iter):
        new_radii = np.zeros(n)
        for i in range(n):
            r_max = min(centers[i, 0], 1 - centers[i, 0],
                        centers[i, 1], 1 - centers[i, 1])
            for j in range(n):
                if i != j:
                    dist = np.hypot(centers[i, 0] - centers[j, 0],
                                    centers[i, 1] - centers[j, 1])
                    r_max = min(r_max, dist - radii[j])
            new_radii[i] = max(0.0, r_max)
        diff = np.max(np.abs(new_radii - radii))
        radii = new_radii
        if diff < 1e-15:
            break
    return radii


def evaluate_sum(centers, n):
    """Evaluate the sum of radii for given centers."""
    radii = compute_radii(centers, n)
    return np.sum(radii), radii


def local_search(centers, n, max_iter=200):
    """Improve centers via local search."""
    radii = compute_radii(centers, n)
    
    for iteration in range(max_iter):
        radii = compute_radii(centers, n)
        improved = False
        step_base = 0.02 * max(0.01, 1 - iteration / max_iter)
        
        for i in range(n):
            best_r = radii[i]
            best_c = centers[i].copy()
            
            # Try multiple perturbations with varying step sizes
            for trial in range(150):
                step = step_base * (1 + trial * 0.01)
                perturbation = np.random.randn(2) * step
                new_c = np.clip(centers[i] + perturbation, 1e-7, 1 - 1e-7)
                
                # Compute achievable radius at new position
                r_max = min(new_c[0], 1 - new_c[0], new_c[1], 1 - new_c[1])
                for j in range(n):
                    if i != j:
                        dist = np.hypot(new_c[0] - centers[j, 0],
                                        new_c[1] - centers[j, 1])
                        r_max = min(r_max, dist - radii[j])
                
                if r_max > best_r + 1e-12:
                    best_r = r_max
                    best_c = new_c.copy()
            
            if np.hypot(best_c[0] - centers[i, 0], best_c[1] - centers[i, 1]) > 1e-10:
                centers[i] = best_c
                improved = True
        
        if not improved and iteration > 50:
            break
    
    radii = compute_radii(centers, n)
    return centers, radii


def make_structured_init(n, corner_r=0.22):
    """Create a structured initial placement."""
    centers = np.zeros((n, 2))
    
    # 4 corner circles
    centers[0] = [corner_r, corner_r]
    centers[1] = [1 - corner_r, corner_r]
    centers[2] = [corner_r, 1 - corner_r]
    centers[3] = [1 - corner_r, 1 - corner_r]
    
    # 8 edge circles
    edge_off = corner_r + 0.025
    centers[4] = [0.5, edge_off]
    centers[5] = [edge_off, 0.5]
    centers[6] = [1 - edge_off, 0.5]
    centers[7] = [0.5, 1 - edge_off]
    centers[8] = [0.3, edge_off]
    centers[9] = [0.7, edge_off]
    centers[10] = [edge_off, 0.3]
    centers[11] = [edge_off, 0.7]
    
    # Interior circles (14 remaining)
    interior_pts = [
        (0.5, 0.5),
        (0.38, 0.38), (0.62, 0.38), (0.38, 0.62), (0.62, 0.62),
        (0.5, 0.3), (0.5, 0.7), (0.3, 0.5), (0.7, 0.5),
        (0.28, 0.28), (0.72, 0.28), (0.28, 0.72), (0.72, 0.72),
        (0.5, 0.2),
    ]
    for idx, (x, y) in enumerate(interior_pts):
        if 12 + idx < n:
            centers[12 + idx] = [x, y]
    
    return centers


def make_hex_grid_init(n):
    """Create a hexagonal grid initial placement."""
    centers = np.zeros((n, 2))
    spacing_x = 0.18
    spacing_y = spacing_x * np.sqrt(3) / 2
    
    idx = 0
    row = 0
    while idx < n and row < 6:
        col = 0
        while idx < n and col < 6:
            x = 0.08 + col * spacing_x + (row % 2) * spacing_x / 2
            y = 0.08 + row * spacing_y
            if 0 < x < 1 and 0 < y < 1:
                centers[idx] = [x, y]
                idx += 1
            col += 1
        row += 1
    
    # Fill remaining
    while idx < n:
        centers[idx] = [np.random.rand() * 0.6 + 0.2, np.random.rand() * 0.6 + 0.2]
        idx += 1
    
    return centers


def run_packing():
    n = 26
    np.random.seed(42)
    
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Strategy 1: Structured init with local search
    for corner_r in [0.20, 0.22, 0.23, 0.24, 0.25]:
        for seed in range(3):
            np.random.seed(seed)
            centers = make_structured_init(n, corner_r).copy()
            
            # Add small random perturbation
            centers += np.random.randn(n, 2) * 0.005
            centers = np.clip(centers, 0.01, 0.99)
            
            centers_opt, radii_opt = local_search(centers, n, max_iter=150)
            s = np.sum(radii_opt)
            
            if s > best_sum:
                best_sum = s
                best_centers = centers_opt.copy()
                best_radii = radii_opt.copy()
    
    # Strategy 2: Hex grid init
    for seed in range(5):
        np.random.seed(seed)
        centers = make_hex_grid_init(n).copy()
        centers += np.random.randn(n, 2) * 0.01
        centers = np.clip(centers, 0.01, 0.99)
        
        centers_opt, radii_opt = local_search(centers, n, max_iter=150)
        s = np.sum(radii_opt)
        
        if s > best_sum:
            best_sum = s
            best_centers = centers_opt.copy()
            best_radii = radii_opt.copy()
    
    # Strategy 3: Multiple random restarts
    for seed in range(10):
        np.random.seed(100 + seed)
        centers = np.random.rand(n, 2) * 0.7 + 0.15
        
        centers_opt, radii_opt = local_search(centers, n, max_iter=100)
        s = np.sum(radii_opt)
        
        if s > best_sum:
            best_sum = s
            best_centers = centers_opt.copy()
            best_radii = radii_opt.copy()
    
    # Final polish: run local search again on best solution
    best_centers, best_radii = local_search(best_centers, n, max_iter=200)
    best_sum = np.sum(best_radii)
    
    return best_centers, best_radii, best_sum
