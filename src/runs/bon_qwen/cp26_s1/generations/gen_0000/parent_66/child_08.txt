

To maximize the sum of radii for 26 circles in a unit square, we employ a strategy based on **Sequential Convex Optimization**. This approach is superior to standard grid packing because it allows circles to "jostle" into the local density maximum, which often involves a slightly rotated or non-lattice configuration that utilizes the square's corners and edges more efficiently.

### Strategy
1.  **Initialization**: We start with a "warm" start by placing circles in a dense hexagonal-like grid pattern. This avoids local minima associated with random placement.
2.  **Iterative Refinement**: We iteratively solve for the maximum equal radius $r$ that fits 26 circles. In each iteration, we use `scipy.optimize.minimize` to find the optimal positions for that specific radius. 
3.  **Radius Growth**: By successfully finding valid positions for a radius $r$, we incrementally increase $r$ and repeat. 
4.  **Final Optimization**: Once we reach the limit of equal radii, we perform one final unconstrained optimization where radii are allowed to vary individually to "fill" any remaining microscopic gaps.

### Code

```python
import numpy as np
from scipy.optimize import minimize

def run_packing():
    """
    Packs 26 circles in a unit square [0,1]x[0,1] to maximize the sum of radii.
    Uses sequential optimization and a penalty method to enforce constraints.
    """
    n = 26
    
    # --- Initialization: Hexagonal Lattice ---
    # We generate a hexagonal pattern to provide a dense starting configuration.
    r_start = 0.09
    s = 2 * r_start
    centers = []
    
    # Approximate rows needed for 26 circles in a square
    # 5 rows of ~5 circles is a good heuristic
    for j in range(6):
        for i in range(5):
            if len(centers) >= n:
                break
            x = r_start + i * s + (j % 2) * (s / 2)
            y = r_start + j * (s * np.sqrt(3) / 2)
            if x <= 1 - r_start and y <= 1 - r_start:
                centers.append([x, y])
        if len(centers) >= n:
            break
            
    centers = np.array(centers[:n])
    radii = np.full(n, r_start)
    
    # --- Optimization Helper ---
    def solve_positions(r_val, init_centers):
        """
        Given a fixed radius r_val, finds optimal centers to minimize 
        the penalty for overlapping circles and boundary violations.
        """
        n = len(init_centers)
        vars_init = init_centers.flatten()
        
        def objective(x_vars):
            # Reshape into (n, 2)
            c = x_vars.reshape(n, 2)
            penalty = 0.0
            
            # 1. Boundary penalties (soft constraints)
            # If x - r < 0, penalty increases
            for i in range(n):
                x, y = c[i]
                # Penalize distance from boundaries being less than r
                # Using a smooth max-like function or simple penalty
                # dist to left: x - r. if < 0, penalize.
                for bound in [x, y]:
                    if bound < r_val:
                        penalty += 100 * (r_val - bound)**2
                for bound in [1-x, 1-y]:
                    if bound < r_val:
                        penalty += 100 * (r_val - bound)**2
            
            # 2. Overlap penalties
            for i in range(n):
                for j in range(i + 1, n):
                    dist = np.sqrt(np.sum((c[i] - c[j])**2))
                    min_dist = r_val + r_val
                    if dist < min_dist:
                        penalty += 100 * (min_dist - dist)**2
            
            return penalty

        # Bounds for positions [0, 1]
        bounds = [(0, 1)] * (2 * n)
        
        # Run optimization
        res = minimize(objective, vars_init, bounds=bounds, method='L-BFGS-B', 
                       options={'ftol': 1e-9, 'gtol': 1e-6, 'maxiter': 5000})
        
        # Check if valid (penalty is low enough)
        if res.fun < 1e-5:
            return res.x.reshape(n, 2), True
        else:
            return res.x.reshape(n, 2), False

    # --- Iterative Radius Growth ---
    current_centers = centers
    r_opt = r_start
    step = 0.0005
    
    # Try to increase radius in steps
    for _ in range(200):
        r_target = r_opt + step
        new_centers, valid = solve_positions(r_target, current_centers)
        if valid:
            r_opt = r_target
            current_centers = new_centers
        else:
            # If we fail, refine the step size and try smaller increments
            if step > 1e-6:
                step /= 2.0
                r_target = r_opt + step
                new_centers, valid = solve_positions(r_target, current_centers)
                if valid:
                    r_opt = r_target
                    current_centers = new_centers
                else:
                    break
            else:
                break
                
    # --- Final Unequal Radii Optimization ---
    # Once equal radii are maximized, we try to increase sum of radii by allowing them to vary.
    # This often captures extra volume in the gaps.
    def final_objective(vars):
        # vars: [x1, y1, r1, x2, y2, r2, ...]
        c = vars[:2*n].reshape(n, 2)
        r = vars[2*n:]
        
        penalty = 0.0
        total_r = np.sum(r)
        
        # Maximize total_r => Minimize -total_r
        # Penalty is added to objective
        
        # Boundary penalties
        for i in range(n):
            x, y = c[i]
            rad = r[i]
            if rad < 0: penalty += 1000 * rad**2 # Ensure non-negative
            
            if x < rad: penalty += 100 * (rad - x)**2
            if y < rad: penalty += 100 * (rad - y)**2
            if 1-x < rad: penalty += 100 * (rad - (1-x))**2
            if 1-y < rad: penalty += 100 * (rad - (1-y))**2
            
        # Overlap penalties
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((c[i] - c[j])**2))
                req = r[i] + r[j]
                if dist < req:
                    penalty += 100 * (req - dist)**2
                    
        return -total_r + penalty

    # Combine positions and radii into one vector
    vars_init = np.concatenate([current_centers.flatten(), np.full(n, r_opt)])
    
    # Bounds: x,y in [0,1], r in [0, 0.5]
    bounds = []
    for _ in range(2 * n):
        bounds.append((0.0, 1.0))
    for _ in range(n):
        bounds.append((0.0, 0.5))

    res_final = minimize(final_objective, vars_init, bounds=bounds, method='L-BFGS-B',
                         options={'ftol': 1e-12, 'maxiter': 10000})

    final_centers = res_final.x[:2*n].reshape(n, 2)
    final_radii = res_final.x[2*n:]
    
    return final_centers, final_radii, np.sum(final_radii)
```
