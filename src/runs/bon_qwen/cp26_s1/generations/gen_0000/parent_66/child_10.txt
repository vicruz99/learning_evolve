import numpy as np
from scipy.optimize import minimize, linprog

def objective_func(x):
    """Objective: maximize sum of radii (minimize negative sum)"""
    radii = x[2::3]
    return -np.sum(radii)

def pair_dist_constraint(x, i, j):
    """Constraint: squared distance between centers >= squared sum of radii"""
    xi, yi, ri = x[3*i], x[3*i+1], x[3*i+2]
    xj, yj, rj = x[3*j], x[3*j+1], x[3*j+2]
    return (xi - xj)**2 + (yi - yj)**2 - (ri + rj)**2

def boundary_constraints(x):
    """Constraints: circles must be inside the unit square"""
    n = len(x) // 3
    out = np.empty(4 * n)
    for i in range(n):
        xi, yi, ri = x[3*i], x[3*i+1], x[3*i+2]
        out[4*i] = xi - ri          # left
        out[4*i+1] = 1.0 - xi - ri  # right
        out[4*i+2] = yi - ri        # bottom
        out[4*i+3] = 1.0 - yi - ri  # top
    return out

def solve_radii_lp(centers):
    """Solve LP to find optimal radii for fixed centers ensuring strict feasibility"""
    n = centers.shape[0]
    c = -np.ones(n)  # Maximize sum radii => minimize negative sum
    
    n_pairs = n * (n - 1) // 2
    n_bounds = n
    A_ub = np.zeros((n_pairs + n_bounds, n))
    b_ub = np.zeros(n_pairs + n_bounds)
    
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt((centers[i, 0] - centers[j, 0])**2 + (centers[i, 1] - centers[j, 1])**2)
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dist
            idx += 1
            
    for i in range(n):
        b_i = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        A_ub[idx, i] = 2.0
        b_ub[idx] = b_i
        idx += 1
        
    bounds_r = [(0.0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method='highs')
    
    if res.success:
        return res.x
    else:
        # Fallback to safe small radii if LP fails
        return np.full(n, 0.01)

def run_packing():

    best_x = None

        points = points[:n]
