import numpy as np
from scipy.optimize import minimize

def compute_objective(v, n, lam):
    """Computes the objective value: -r + lambda * penalty."""
    c = v[:2*n].reshape(n, 2)
    r = v[2*n]
    
    # Vectorized pairwise squared distances
    dx = c[:, 0][:, None] - c[:, 0][None, :]
    dy = c[:, 1][:, None] - c[:, 1][None, :]
    d2 = dx**2 + dy**2
    
    # Extract upper triangle indices to avoid duplicates and self-distance
    tri = np.triu_indices(n, k=1)
    d = np.sqrt(d2[tri] + 1e-12)
    
    pen = 0.0
    # Penalty for circle-circle overlaps
    diff = 2*r - d
    mask = diff > 0
    pen += np.sum(diff[mask]**2)
    
    # Penalty for boundary violations
    diff_x1 = r - c[:, 0]
    mask_x1 = diff_x1 > 0
    pen += np.sum(diff_x1[mask_x1]**2)
    diff_x2 = c[:, 0] - (1 - r)
    mask_x2 = diff_x2 > 0
    pen += np.sum(diff_x2[mask_x2]**2)
    diff_y1 = r - c[:, 1]
    mask_y1 = diff_y1 > 0
    pen += np.sum(diff_y1[mask_y1]**2)
    diff_y2 = c[:, 1] - (1 - r)
    mask_y2 = diff_y2 > 0
    pen += np.sum(diff_y2[mask_y2]**2)
    
    return -r + lam * pen

def run_optimization(n, init_centers, lam_start=5.0):
    """Runs the penalty-continuation optimization from given initial centers."""
    v0 = np.concatenate([init_centers.flatten(), [0.05]])
    bounds = [(0, 1)] * (2*n) + [(0, 0.5)]
    
    lam = lam_start
    v_curr = v0.copy()
    
    # Continuation method: gradually increase penalty weight
    for _ in range(20):
        res = minimize(compute_objective, v_curr, args=(n, lam), method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': 2000, 'ftol': 1e-12, 'gtol': 1e-12})
        v_curr = res.x
        lam *= 1.3
        
    c = v_curr[:2*n].reshape(n, 2)
    r = v_curr[2*n]
    return c, r

def run_packing():
    """Main function to find and return the optimal packing configuration."""
    n = 26
    best_sum = 0.0
    best_c = None
    best_r = None
    
    # Prepare diverse initial configurations
    inits = []
    
    # 1. Hexagonal lattice pattern
    hex_c = []
    for i in range(6):
        for j in range(5):
            if len(hex_c) < n:
                x = 0.12 + j * 0.17 + (0.085 if i % 2 else 0.0)
                y = 0.12 + i * 0.16
                hex_c.append([x, y])
    inits.append(np.array(hex_c[:n]))
    
    # 2. Square grid 5x5 + center
    grid_c = []
    for i in range(5):
        for j in range(5):
            grid_c.append([0.1 + j*0.2, 0.1 + i*0.2])
    grid_c.append([0.5, 0.5])
    inits.append(np.array(grid_c[:n]))
    
    # 3. Deterministic random placement
    np.random.seed(42)
    inits.append(np.random.uniform(0.15, 0.85, (n, 2)))
    
    # Optimize from each initialization
    for init_c in inits:
        c, r = run_optimization(n, init_c)
        
        # Compute the true maximum feasible radius for these positions
        min_d = 1.0
        for i in range(n):
            for j in range(i+1, n):
                d = np.sqrt(np.sum((c[i]-c[j])**2))
                if d < min_d:
                    min_d = d
            min_d = min(min_d, c[i,0], 1-c[i,0], c[i,1], 1-c[i,1])
            
        r_true = min_d / 2.0
        
        # Strict validity check
        valid = True
        for i in range(n):
            if c[i,0] < r_true - 1e-6 or c[i,0] > 1 - r_true + 1e-6:
                valid = False
            if c[i,1] < r_true - 1e-6 or c[i,1] > 1 - r_true + 1e-6:
                valid = False
            for j in range(i+1, n):
                d = np.sqrt(np.sum((c[i]-c[j])**2))
                if d < 2*r_true - 1e-6:
                    valid = False
                    
        if valid:
            s = n * r_true
            if s > best_sum:
                best_sum = s
                best_c = c.copy()
                best_r = np.full(n, r_true)
                
    return best_c, best_r, best_sum
