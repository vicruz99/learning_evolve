import numpy as np
from scipy.optimize import minimize
import math

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Pack 26 circles in a unit square [0,1]x[0,1] to maximize the sum of radii.
    Returns (centers, radii, sum_radii).
    """
    n = 26
    
    # --- Stage 1: Initialization ---
    # Initialize centers in a hexagonal pattern for better packing density
    centers_init = generate_hexagonal_init(n)
    
    # Set initial radii small enough to avoid overlap in the initial layout
    # A radius of 0.05 is safe for the generated hexagonal pattern
    radii_init = np.full(n, 0.05)
    
    # --- Stage 2: Optimization ---
    # Objective: Maximize sum of radii -> Minimize negative sum of radii
    # We optimize 26 (x, y) coordinates and 26 radii = 78 variables
    x0 = np.concatenate([centers_init.flatten(), radii_init])
    
    # Bounds: 
    # x, y in [0, 1]
    # r in [0, 1] (technically could be tighter, but [0,1] is safe)
    bounds = []
    for _ in range(n):
        bounds.extend([(0.0, 1.0), (0.0, 1.0)]) # x, y bounds
        bounds.append((0.0, 1.0))                # r bounds

    # Define constraints
    constraints = []
    
    # Boundary constraints: r <= x <= 1-r and r <= y <= 1-r
    # 1. r <= x  => x - r >= 0
    # 2. x <= 1-r => 1 - x - r >= 0
    # 3. r <= y  => y - r >= 0
    # 4. y <= 1-r => 1 - y - r >= 0
    def make_boundary_constraints(idx, var_offset):
        # var_offset is the start index of x_i in the flat vector
        c1 = {'type': 'ineq', 'fun': lambda vars, i=idx: vars[var_offset + 0] - vars[var_offset + 2 * n]}
        c2 = {'type': 'ineq', 'fun': lambda vars, i=idx: 1.0 - vars[var_offset + 0] - vars[var_offset + 2 * n]}
        c3 = {'type': 'ineq', 'fun': lambda vars, i=idx: vars[var_offset + 1] - vars[var_offset + 2 * n]}
        c4 = {'type': 'ineq', 'fun': lambda vars, i=idx: 1.0 - vars[var_offset + 1] - vars[var_offset + 2 * n]}
        return [c1, c2, c3, c4]

    for i in range(n):
        var_offset = i * 3 # x, y, r structure in flat array
        constraints.extend(make_boundary_constraints(i, var_offset))

    # Overlap constraints: dist(i, j) >= r_i + r_j
    # dist^2 >= (r_i + r_j)^2
    # (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
    for i in range(n):
        for j in range(i + 1, n):
            vi = i * 3
            vj = j * 3
            ri_idx = vi + 2
            rj_idx = vj + 2
            
            # We need to capture i and j in the lambda
            def make_overlap_con(i, j, vi, vj, ri, rj):
                def func(vars):
                    xi, yi = vars[vi], vars[vi + 1]
                    xj, yj = vars[vj], vars[vj + 1]
                    ri_val, rj_val = vars[ri], vars[rj]
                    dist_sq = (xi - xj)**2 + (yi - yj)**2
                    min_dist_sq = (ri_val + rj_val)**2
                    return dist_sq - min_dist_sq
                return {'type': 'ineq', 'fun': func}
            
            constraints.append(make_overlap_con(i, j, vi, vj, ri_idx, rj_idx))

    # Objective function: minimize -sum(radii)
    def objective(vars):
        # radii are at indices 2, 5, 8, ... (every 3rd starting at 2)
        # Or simply slice if structured, but here we iterate
        total_r = 0.0
        for i in range(n):
            total_r += vars[i * 3 + 2]
        return -total_r

    # Run optimization
    # SLSQP is suitable for bound-constrained and inequality-constrained problems
    result = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints, 
                      options={'maxiter': 2000, 'ftol': 1e-9})

    # Extract results
    if result.success or result.fun < 0: # Check if we found a valid non-zero sum
        final_centers = np.array([[result.x[i * 3], result.x[i * 3 + 1]] for i in range(n)])
        final_radii = np.array([result.x[i * 3 + 2] for i in range(n)])
    else:
        # Fallback to initial state if optimization fails completely (unlikely)
        final_centers = centers_init
        final_radii = radii_init

    # Post-processing: Clip tiny radii or negative values if any numerical noise
    final_radii = np.maximum(final_radii, 0.0)
    
    sum_radii = np.sum(final_radii)
    
    # Ensure return types match expected np.ndarray
    final_centers = np.asarray(final_centers)
    final_radii = np.asarray(final_radii)

    return final_centers, final_radii, sum_radii

def generate_hexagonal_init(n):
    """
    Generates initial centers for n circles in a hexagonal pattern within [0,1]x[0,1].
    """
    centers = []
    # Estimate grid dimensions. sqrt(26) ~ 5.1. 
    # A 5-row hexagonal packing works well.
    # Pattern of circles per row: 6, 5, 6, 5, 4 (Sum = 26)
    # Or 5, 6, 5, 6, 4? Let's try to center them.
    # Row counts for 26 items in hex packing:
    # If we use 5 rows, counts could be [5, 6, 5, 6, 4] -> 26
    # Or [6, 5, 6, 5, 4] -> 26
    # Let's use [6, 5, 6, 5, 4] starting from bottom.
    
    row_counts = [6, 5, 6, 5, 4]
    
    # Vertical spacing for hex packing: sqrt(3)/2 * diameter. 
    # We don't know diameter yet, but we can assume a scale.
    # Let's assume diameter approx 0.2 for initialization.
    r_est = 0.05 
    dy = math.sqrt(3) * r_est
    dx = 2 * r_est
    
    current_y = r_est
    
    for row_idx, count in enumerate(row_counts):
        # Horizontal shift for alternating rows
        shift = r_est if (row_idx % 2 == 1) else 0.0
        
        # Calculate total width of this row to center it
        # Width = (count - 1) * dx
        total_width = (count - 1) * dx
        start_x = (1.0 - total_width) / 2.0 - shift # Adjust for shift? 
        # Actually, easier to just place them and they will relax.
        # Let's just place them starting from a centered position.
        
        # Recalculate start_x to be roughly centered in [0,1] considering shift
        # If shifted, the block of circles is offset.
        # Let's just place x_i = r_est + i*dx + shift
        # And clamp or scale later. The optimizer will fix it.
        
        row_centers = []
        for k in range(count):
            x = r_est + k * dx + shift
            y = current_y
            # Clamp to [0, 1] roughly
            x = np.clip(x, 0.01, 0.99)
            y = np.clip(y, 0.01, 0.99)
            row_centers.append([x, y])
        centers.extend(row_centers)
        current_y += dy
        
    # If we didn't get exactly n (logic check), pad or trim
    # My counts sum to 26, so it should be exact.
    return np.array(centers[:n])

# Helper for validation (read-only, do not modify this) - included for context but not used in solution
def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    import numpy as np
    n = centers.shape[0]
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False
    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False
    return True
