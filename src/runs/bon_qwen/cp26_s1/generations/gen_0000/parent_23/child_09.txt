from scipy.optimize import minimize

        # 5 rows.

        # Row 3: 5
        # Row 4: 5 (offset)
        # Row 5: 5
        # Row 6: 1 (small) - maybe not.

        centers = np.random.rand(n, 2) * 0.8 + 0.1
        radii = np.ones(n) * 0.05 # Small start to ensure no overlap
        
        # Perturb to look like a grid
        for i in range(n):
            r = i // 6
            c = i % 6
            # 6 columns, ~4.33 rows.
            # Map to square
            cx = (c + 0.5) / 6.0
            cy = (r + 0.5) / (n/6.0)
            centers[i] = [cx, cy]
            
        return centers, radii

    def objective(x):
        # x contains [x1, y1, r1, x2, y2, r2, ...]
        # We want to maximize sum of radii, so minimize negative sum.
        # Extract radii: every 3rd variable starting from index 2
        r_vals = x[2::3]
        return -np.sum(r_vals)

    def constraints_factory(n_circles):
        cons = []
        # Boundary constraints
        # For each circle i:
        # x_i >= r_i  => x_i - r_i >= 0
        # x_i <= 1 - r_i => 1 - x_i - r_i >= 0
        # y_i >= r_i
        # y_i <= 1 - r_i
        
        # We will add these as nonlinear constraints using scipy's constraint format
        # However, with 78 variables and many constraints, function calls are expensive.
        # We can implement a vectorized penalty or just use bounds and simple constraints?
        # SLSQP supports bounds and nonlinear constraints.
        
        # Bounds:
        # x_i, y_i in [0, 1]
        # r_i >= 0
        # Actually, if r_i > 0.5, it can't fit. But bounds [0, 1] are safe.
        
        # Let's define bounds separately
        bounds = []
        for i in range(n_circles):
            bounds.append((0, 1)) # x
            bounds.append((0, 1)) # y
            bounds.append((0, 0.5)) # r (radius can't be > 0.5)
            
        # Nonlinear constraints for non-overlap and boundary inclusion relative to r
        # It's easier to enforce x_i - r_i >= 0 etc as nonlinear constraints 
        # because bounds on x_i don't know about r_i.
        
        def boundary_constraints(x):
            vals = []
            for i in range(n_circles):
                idx = i * 3
                xi, yi, ri = x[idx], x[idx+1], x[idx+2]
                # x >= r
                vals.append(xi - ri)
                # x <= 1 - r => 1 - x - r >= 0
                vals.append(1 - xi - ri)
                # y >= r
                vals.append(yi - ri)
                # y <= 1 - r
                vals.append(1 - yi - ri)
            return np.array(vals)

            vals = []
            for i in range(n_circles):
                for j in range(i + 1, n_circles):

                    idx_j = j * 3
                    xi, yi, ri = x[idx_i], x[idx_i+1], x[idx_i+2]

                    min_dist = ri + rj
                    # Constraint: dist_sq >= min_dist^2
                    # i.e., dist_sq - min_dist^2 >= 0
                    # Numerical stability: use a small slack? 
                    # SLSQP handles >= 0.
                    # dist_sq - (ri+rj)^2 >= 0
                    vals.append(dist_sq - (ri + rj)**2)
            return np.array(vals)

        return bounds, boundary_constraints, overlap_constraints

        def combined_constraints(x):
            bc = bc_func(x)
            oc = oc_func(x)
            return np.concatenate((bc, oc))

        cons = {'type': 'ineq', 'fun': combined_constraints}
        
        # Optimize
        try:
            res = minimize(
                objective, 
                x0, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons, 
                options={'maxiter': 200, 'ftol': 1e-9}
            )
            
            if res.success or (res.fun < best_score and not np.isnan(res.fun)):
                best_score = res.fun
                best_sol = res.x
        except Exception as e:
            continue

    if best_sol is None:

        radii_final[i] = best_sol[i*3+2]
        
    sum_radii = np.sum(radii_final)
    
    return centers_final, radii_final, sum_radii
