import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Pack 26 circles in a unit square to maximize the sum of radii.
    """
    n_circles = 26
    
    # 1. Generate Initial Configuration (Hexagonal Lattice)
    # We start with a valid packing of equal circles with radius r_init.
    # A hexagonal packing is denser than square grid.
    r_init = 0.09
    
    centers = []
    radii = []
    
    # Hexagonal grid parameters
    # Vertical spacing: r * sqrt(3)
    # Horizontal spacing: 2 * r
    # Odd rows shifted by r
    
    dy = r_init * np.sqrt(3)
    dx = 2 * r_init
    
    # Generate rows
    y = r_init
    row_idx = 0
    
    while y + r_init <= 1.0 + 1e-9:
        x_start = r_init + (row_idx % 2) * r_init
        
        x = x_start
        while x + r_init <= 1.0 + 1e-9:
            centers.append([x, y])
            radii.append(r_init)
            
            if len(centers) >= n_circles:
                break
            
            x += dx
        
        if len(centers) >= n_circles:
            break
            
        y += dy
        row_idx += 1
        
    # If we somehow generated fewer (unlikely with these params), pad or truncate
    if len(centers) > n_circles:
        centers = centers[:n_circles]
        radii = radii[:n_circles]
    
    centers = np.array(centers)
    radii = np.array(radii)
    
    # Ensure we have exactly 26 circles
    assert len(centers) == 26, f"Initial config has {len(centers)} circles"

    # 2. Optimization Setup
    # Variables: [x1, y1, r1, x2, y2, r2, ..., x26, y26, r26]
    # Total 3 * 26 = 78 variables.
    
    x0 = np.zeros(3 * n_circles)
    for i in range(n_circles):
        x0[3*i] = centers[i, 0]
        x0[3*i+1] = centers[i, 1]
        x0[3*i+2] = radii[i]
        
    # Bounds: x, y in [0, 1], r in [0, 0.5] (loose upper bound)
    # Actually tighter bound for r is 0.5, but x,y bounds depend on r.
    # We'll use simple bounds for variables and let constraints handle the rest.
    # x, y in [0, 1], r in [0, 0.5]
    bounds = [(0, 1), (0, 1), (0, 0.5)] * n_circles
    
    # Constraints
    constraints = []
    
    # Boundary Constraints
    # r <= x <= 1-r  =>  x - r >= 0  and  1 - x - r >= 0
    # r <= y <= 1-r  =>  y - r >= 0  and  1 - y - r >= 0
    
    for i in range(n_circles):
        idx_x = 3 * i
        idx_y = 3 * i + 1
        idx_r = 3 * i + 2
        
        # x - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: v[idx_x] - v[idx_r]
        })
        # 1 - x - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: 1.0 - v[idx_x] - v[idx_r]
        })
        # y - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: v[idx_y] - v[idx_r]
        })
        # 1 - y - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: 1.0 - v[idx_y] - v[idx_r]
        })
        
    # Overlap Constraints
    # dist >= r_i + r_j
    # dist^2 >= (r_i + r_j)^2
    # (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
    
    for i in range(n_circles):
        for j in range(i + 1, n_circles):
            idx_xi = 3 * i
            idx_yi = 3 * i + 1
            idx_ri = 3 * i + 2
            idx_xj = 3 * j
            idx_yj = 3 * j + 1
            idx_rj = 3 * j + 2
            
            constraints.append({
                'type': 'ineq',
                'fun': lambda v, i=i, j=j: (
                    (v[idx_xi] - v[idx_xj])**2 + (v[idx_yi] - v[idx_yj])**2 
                    - (v[idx_ri] + v[idx_rj])**2
                )
            })

    # Objective: Maximize sum of radii => Minimize -sum(radii)
    def objective(v):
        total_r = 0.0
        for i in range(n_circles):
            total_r += v[3 * i + 2]
        return -total_r

    # Run Optimization
    # SLSQP is good for constrained non-linear optimization
    result = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints, 
                      options={'maxiter': 1000, 'ftol': 1e-9})
    
    # Extract results
    final_centers = np.zeros((n_circles, 2))
    final_radii = np.zeros(n_circles)
    
    if result.success or result.fun > -10: # Check if we got a reasonable result
        for i in range(n_circles):
            final_centers[i, 0] = result.x[3 * i]
            final_centers[i, 1] = result.x[3 * i + 1]
            final_radii[i] = result.x[3 * i + 2]
    else:
        # Fallback to initial configuration if optimization fails
        final_centers = centers
        final_radii = radii

    sum_radii = np.sum(final_radii)
    
    # Validate before returning (optional but good practice, though prompt says validation function is read-only)
    # We assume the optimizer respected constraints closely.
    
    return final_centers, final_radii, sum_radii
