3.  **Vectorized Computation**: To handle the $O(N^2)$ pairwise interactions efficiently, we use NumPy vectorization for distance and force calculations.

import numpy as np

def run_packing():
    np.random.seed(42)
    N = 26
    
    # --- Initialization ---
    # Hexagonal grid initialization for high packing density
    centers = np.zeros((N, 2))
    idx = 0
    # Approximate spacing for 26 circles
    dx, dy = 0.15, 0.15 * np.sqrt(3)
    for row in range(10):
        for col in range(6):
            if idx >= N:
                break
            x = 0.1 + col * dx + (row % 2) * (dx / 2)
            y = 0.1 + row * dy
            if x < 1.0 and y < 1.0:
                centers[idx] = [x, y]
                idx += 1
    # Randomize slightly to break symmetry
    centers += np.random.uniform(-0.02, 0.02, centers.shape)
    centers = np.clip(centers, 0.05, 0.95)
    
    r = 0.05
    margin = 1e-5
    lr_pos = 0.05
    expand_step = 2e-5
    
    # --- Optimization Loop ---
    for step in range(8000):
        # Vectorized distance matrix
        diff = centers[:, None, :] - centers[None, :, :]
        dist = np.sqrt(np.sum(diff**2, axis=2))
        np.fill_diagonal(dist, np.inf)
        
        # Compute repulsive forces
        # Force magnitude: proportional to overlap distance
        overlap = np.maximum(0, 2 * r - dist - margin)
        # Avoid division by zero
        safe_dist = np.where(dist < 1e-9, 1e-9, dist)
        force_mag = overlap / safe_dist
        force_dir = diff / safe_dist[:, :, None]
        
        # Sum forces (using Einstein summation for efficiency)
        forces = np.einsum('ijk,ijk->ik', force_mag, force_dir)
        
        # Boundary repulsion
        for i in range(N):
            x, y = centers[i]
            # Left
            if x < r + margin: forces[i, 0] += (r + margin - x) * 50
            # Right
            if x > 1 - r - margin: forces[i, 0] -= (x - (1 - r - margin)) * 50
            # Bottom
            if y < r + margin: forces[i, 1] += (r + margin - y) * 50
            # Top
            if y > 1 - r - margin: forces[i, 1] -= (y - (1 - r - margin)) * 50
            
        # Update positions
        centers += lr_pos * forces
        centers = np.clip(centers, r + margin, 1 - r - margin)
        
        # Expand radius
        r += expand_step
        
        # Adaptive step size control based on overlap severity
        min_d = np.min(dist)
        if min_d < 2 * r - margin - 0.01:
            r -= expand_step * 2 # Reverse if too much overlap
            expand_step *= 0.95
            lr_pos *= 0.95
        elif step > 100 and expand_step < 5e-6:
            expand_step *= 1.002 # Slowly increase step if stable

    # --- Final Validation & Adjustment ---
    # Compute actual minimum distance to set a safe radius
    diff_final = centers[:, None, :] - centers[None, :, :]
    dist_final = np.sqrt(np.sum(diff_final**2, axis=2))
    np.fill_diagonal(dist_final, np.inf)
    
    min_pair_dist = np.min(dist_final)
    
    # Safety shrink factor to guarantee strict inequality in validator
    safe_r = (min_pair_dist / 2) - 1e-4
    
    # Boundary check
    dist_to_boundary = np.minimum(np.minimum(centers[:, 0], 1 - centers[:, 0]),
                                  np.minimum(centers[:, 1], 1 - centers[:, 1]))
    min_boundary = np.min(dist_to_boundary)
    
    if min_boundary - 1e-4 < safe_r:
        safe_r = min_boundary - 1e-4
        
    # Ensure non-negative
    safe_r = max(1e-5, safe_r)
    
    radii = np.full(N, safe_r)
    total_sum = np.sum(radii)
    
    # Final clamp to ensure strict boundary compliance
    centers = np.clip(centers, safe_r, 1 - safe_r)
    
    return centers, radii, total_sum
