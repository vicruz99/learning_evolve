import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Bounds: centers in [0,1], radii in [0, 0.5]
    bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n
    
    # Objective: maximize sum of radii => minimize negative sum
    def objective(vars):
        return -np.sum(vars[2*n:])

    # Define constraints
    def get_constraints(n):
        cons = []
        # Boundary constraints for each circle
        for i in range(n):
            xi = 2*i
            yi = 2*i + 1
            ri = 2*n + i
            # x - r >= 0
            cons.append({'type': 'ineq', 'fun': lambda v, i=i: v[2*i] - v[2*n+i]})
            # 1 - x - r >= 0
            cons.append({'type': 'ineq', 'fun': lambda v, i=i: 1.0 - v[2*i] - v[2*n+i]})
            # y - r >= 0
            cons.append({'type': 'ineq', 'fun': lambda v, i=i: v[2*i+1] - v[2*n+i]})
            # 1 - y - r >= 0
            cons.append({'type': 'ineq', 'fun': lambda v, i=i: 1.0 - v[2*i+1] - v[2*n+i]})
            
        # Pairwise non-overlap constraints
        def pairwise(v):
            c = v[:2*n].reshape(n, 2)
            r = v[2*n:]
            vals = []
            for i in range(n):
                for j in range(i+1, n):
                    d = np.hypot(c[i,0]-c[j,0], c[i,1]-c[j,1])
                    vals.append(d - r[i] - r[j])
            return np.array(vals)
        cons.append({'type': 'ineq', 'fun': pairwise})
        return cons

    constraints = get_constraints(n)
    
    # Initialization strategies
    inits = []
    
    # 1. Multiple random starts
    np.random.seed(42)
    for _ in range(15):
        c = 0.05 + 0.9 * np.random.rand(n, 2)
        r = np.full(n, 0.01)
        inits.append(np.concatenate([c.ravel(), r]))
        
    # 2. Grid-based start (5x5 plus one)
    grid_pts = []
    for i in range(5):
        for j in range(5):
            grid_pts.append([i/4.0, j/4.0])
    grid_pts.append([0.5, 0.5]) # 26th point
    grid_arr = np.array(grid_pts[:26])
    grid_arr += np.random.normal(0, 0.02, size=grid_arr.shape)
    grid_arr = np.clip(grid_arr, 0.05, 0.95)

    hex_pts = []
    for row in range(6):
        y = (row + 0.5) / 6.0

            if len(hex_pts) >= 26: break

            hex_pts.append([x, y])
    hex_arr = np.array(hex_pts)
    hex_arr += np.random.normal(0, 0.005, size=hex_arr.shape)
    inits.append(np.concatenate([hex_arr.ravel(), np.full(n, 0.02)]))

    for x0 in inits:
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, 
                           constraints=constraints, options={'maxiter': 5000, 'ftol': 1e-11})
            if res.success and -res.fun > best_sum:
                best_sum = -res.fun
                best_centers = res.x[:2*n].reshape(n, 2)
                best_radii = res.x[2*n:]
        except Exception:
            pass
            
    # Fallback if optimization completely fails
    if best_centers is None:
        best_centers = np.random.rand(n, 2) * 0.8 + 0.1
        best_radii = np.full(n, 0.01)
        best_sum = 0.26

    return best_centers, best_radii, float(best_sum)
