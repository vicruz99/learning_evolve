import numpy as np
from scipy.optimize import basinhopping, minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    Optimizes the packing of 26 circles in a unit square to maximize sum of radii.
    Assumes equal radii for optimal density.
    """
    n_circles = 26
    
    # 1. Initialization: Hexagonal Grid
    # Estimate radius for initialization based on density
    # Density approx 0.9. Area per circle = 1/26.
    # pi * r^2 = 1/26 * 0.9 => r ~ 0.105
    # Spacing 2r ~ 0.21
    init_centers = []
    x, y = 0, 0
    # Generate a hexagonal pattern
    # We generate more points than needed and select the best fitting subset or just place them
    # A simple grid generation:
    rows = 7
    cols = 6
    dx = 1.0 / (cols - 0.5) # spacing
    dy = 1.0 / (rows - 0.5)

    # We'll use a perturbation of a grid
    rng = np.random.RandomState(42)
    centers = rng.rand(n_circles, 2)
    
    # Better initialization: Hexagonal lattice points centered and scaled
    # Hex lattice vectors
    # We want to fill the square.
    # Let's just use a dense random perturbation of a grid
    # Grid 6x5 = 30 points, take 26?
    grid_x = np.linspace(0.1, 0.9, 6)
    grid_y = np.linspace(0.1, 0.9, 5)
    gx, gy = np.meshgrid(grid_x, grid_y)
    grid_points = np.vstack([gx.ravel(), gy.ravel()]).T
    
    # Shuffle and take 26
    rng.shuffle(grid_points)
    initial_centers = grid_points[:n_circles]

    # 2. Objective Function for Optimization
    # We want to maximize the minimum distance between circles and boundaries.
    # The feasible radius r is min(dist_boundary, dist_pair/2).
    # We maximize r, so we minimize -r.
    
    def get_max_radius(centers_flat):
        centers = centers_flat.reshape(-1, 2)
        
        # Distance to boundaries: min(x, 1-x, y, 1-y)
        dists_to_boundary = np.array([
            np.min([c[0], 1 - c[0], c[1], 1 - c[1]]) for c in centers
        ])
        min_dist_boundary = np.min(dists_to_boundary)
        
        # Distance between pairs
        # Compute all pairwise distances
        # Using broadcasting for speed
        diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        
        # Mask out diagonal (distance 0)
        np.fill_diagonal(dists, np.inf)
        min_dist_pair = np.min(dists)
        
        # The limiting factor for radius r is:
        # r <= dist_to_boundary
        # 2r <= dist_between_pairs => r <= dist_between_pairs / 2
        r_boundary = min_dist_boundary
        r_pair = min_dist_pair / 2.0
        
        return min(r_boundary, r_pair)

    def objective(centers_flat):
        r = get_max_radius(centers_flat)
        return -r # Minimize negative radius

    # 3. Optimization
    # Bounds for centers are [0, 1]
    bounds = [(0, 1)] * (n_circles * 2)
    
    # Flatten initial centers
    x0 = initial_centers.flatten()
    
    # Use Basinhopping for global optimization
    # Nelder-Mead is used as the local minimizer as it handles non-smooth functions well
    minimizer_kwargs = {
        "method": "Nelder-Mead",
        "bounds": bounds,
        "options": {'xatol': 1e-5, 'fatol': 1e-5, 'maxiter': 200}
    }
    
    # Run optimization
    # niter=50 should be sufficient for N=26
    ret = basinhopping(objective, x0, minimizer_kwargs=minimizer_kwargs, niter=100, stepsize=0.05)
    
    optimal_centers = ret.x.reshape(-1, 2)
    optimal_radius = -ret.fun
    
    # 4. Verification and Adjustment
    # Ensure valid packing
    # Recalculate exact radius to be safe
    final_radius = get_max_radius(optimal_centers)
    radii = np.full(n_circles, final_radius)
    sum_radii = np.sum(radii)
    
    return optimal_centers, radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]
    if np.isnan(centers).any() or np.isnan(radii).any():
        return False
    for i in range(n):
        if radii[i] < 0:
            return False
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            return False
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False
    return True
