import scipy.optimize as opt
import math

    1. Linear Programming to maximize radii for fixed centers.
    2. Force-directed repulsion to optimize centers for future radius expansion.

    num_iterations = 2000
    cooling_rate = 0.995

    # 5x5 grid has 25 circles. We add one slightly perturbed.
    centers = np.zeros((n, 2))
    r_init = 0.09  # Slightly less than 0.1 to fit 25 in 5x5

            idx += 1

    centers[idx, 0] = 0.5 + 0.05
    centers[idx, 1] = 0.5 + 0.05

    radii = np.full(n, r_init)

    temperature = 1.0

        # Step 1: Optimize Radii using Linear Programming
        # Objective: Maximize sum(r_i) => Minimize sum(-r_i)
        # Constraints:
        # 1. r_i + r_j <= distance(c_i, c_j)  for all i < j
        # 2. r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
        # 3. r_i >= 0

        m_pairs = n * (n - 1) // 2
        m_bound = 4 * n
        A_ub = np.zeros((m_pairs + m_bound, n))
        b_ub = np.zeros(m_pairs + m_bound)

                row_idx += 1

            x, y = centers[i]
            dists = [x, 1-x, y, 1-y]

                A_ub[row_idx, i] = 1.0
                b_ub[row_idx] = d
                row_idx += 1
