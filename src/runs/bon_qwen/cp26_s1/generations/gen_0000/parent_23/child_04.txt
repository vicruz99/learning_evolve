def generate_hex_grid_points(n_points, margin=0.01):

    # Rough estimate of rows/cols based on n_points
    # For 26, roughly 5x5 or 6x5
    rows = int(np.ceil(np.sqrt(n_points * np.sqrt(3)/2))) + 2
    cols = int(np.ceil(np.sqrt(n_points * 2/np.sqrt(3)))) + 2
    
    # Adjust grid density to fit in [0,1]
    # Horizontal spacing 2, vertical sqrt(3) in unit circle packing
    # Scale later
    xs = []
    for r in range(rows):
        row_xs = []
        for c in range(cols):
            x = 2 * c + (r % 2)
            y = np.sqrt(3) * r
            row_xs.append((x, y))
        xs.extend(row_xs)
    
    xs = np.array(xs)
    if len(xs) == 0:
        return np.zeros((n_points, 2))
    
    # Normalize to fit in [0,1]
    min_x, min_y = np.min(xs, axis=0)
    max_x, max_y = np.max(xs, axis=0)
    
    # Center and scale
    cx, cy = (min_x + max_x)/2, (min_y + max_y)/2
    xs_centered = xs - [cx, cy]
    
    width = max_x - min_x
    height = max_y - min_y
    scale = 0.95 / max(width, height) # 0.95 to leave some room
    
    xs_scaled = xs_centered * scale + 0.5
    
    # Select n_points closest to the center of the square to get a compact cluster
    dists = np.sum((xs_scaled - 0.5)**2, axis=1)
    indices = np.argsort(dists)[:n_points]
    return xs_scaled[indices]
