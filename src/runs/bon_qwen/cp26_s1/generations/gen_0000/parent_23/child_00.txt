import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def init_hex(n):
    """Initialize circles in a hexagonal pattern scaled to fit the unit square."""
    centers = []
    s = 0.18
    row = 0
    placed = 0
    while placed < n and row < 10:
        y = row * s * np.sqrt(3) / 2
        offset = s * 0.5 if row % 2 == 1 else 0.0
        x = offset
        while placed < n:
            if x < 1.1:
                centers.append([x, y])
                placed += 1
            else:
                break
            x += s
        row += 1
    centers = np.array(centers)
    margin = 0.02
    x_min, y_min = centers.min(axis=0)
    x_max, y_max = centers.max(axis=0)
    w = x_max - x_min + 1e-9
    h = y_max - y_min + 1e-9
    scale = min((1 - 2 * margin) / w, (1 - 2 * margin) / h)
    centers -= [x_min, y_min]
    centers *= scale
    centers += margin
    radii = np.full(n, 0.05)
    return centers, radii

def objective(x):
    return -np.sum(x[2 * N_CIRCLES:])

def objective_jac(x):
    jac = np.zeros(len(x))
    jac[2 * N_CIRCLES:] = -1.0
    return jac

def constraints_fun(x):
    M = 4 * N_CIRCLES + N_CIRCLES * (N_CIRCLES - 1) // 2
    c = np.empty(M)
    idx = 0
    for i in range(N_CIRCLES):
        r = x[2 * N_CIRCLES + i]
        c[idx] = x[2 * i] - r; idx += 1
        c[idx] = 1.0 - x[2 * i] - r; idx += 1
        c[idx] = x[2 * i + 1] - r; idx += 1
        c[idx] = 1.0 - x[2 * i + 1] - r; idx += 1
    for i in range(N_CIRCLES):
        for j in range(i + 1, N_CIRCLES):
            dx = x[2 * i] - x[2 * j]
            dy = x[2 * i + 1] - x[2 * j + 1]
            c[idx] = np.hypot(dx, dy) - x[2 * N_CIRCLES + i] - x[2 * N_CIRCLES + j]
            idx += 1
    return c

def constraints_jac(x):
    M = 4 * N_CIRCLES + N_CIRCLES * (N_CIRCLES - 1) // 2
    nv = 2 * 3 * N_CIRCLES
    jac = np.zeros((M, nv))
    idx = 0
    for i in range(N_CIRCLES):
        jac[idx, 2 * i] = 1.0; jac[idx, 2 * N_CIRCLES + i] = -1.0; idx += 1
        jac[idx, 2 * i] = -1.0; jac[idx, 2 * N_CIRCLES + i] = -1.0; idx += 1
        jac[idx, 2 * i + 1] = 1.0; jac[idx, 2 * N_CIRCLES + i] = -1.0; idx += 1
        jac[idx, 2 * i + 1] = -1.0; jac[idx, 2 * N_CIRCLES + i] = -1.0; idx += 1
    for i in range(N_CIRCLES):
        for j in range(i + 1, N_CIRCLES):
            dx = x[2 * i] - x[2 * j]
            dy = x[2 * i + 1] - x[2 * j + 1]
            dist = np.hypot(dx, dy)
            inv_d = 1.0 / dist if dist > 1e-12 else 0.0
            jac[idx, 2 * i] = dx * inv_d
            jac[idx, 2 * i + 1] = dy * inv_d
            jac[idx, 2 * j] = -dx * inv_d
            jac[idx, 2 * j + 1] = -dy * inv_d
            jac[idx, 2 * N_CIRCLES + i] = -1.0
            jac[idx, 2 * N_CIRCLES + j] = -1.0
            idx += 1
    return jac

def force_packing_fallback():
    n = N_CIRCLES
    centers, radii = init_hex(n)
    dt = 0.001
    alpha = 1.0001
    
    for _ in range(5000):
        radii *= alpha

            elif x > 1 - r: forces[i, 0] -= (x - (1 - r)) * 10.0

                    forces[i, 1] += fy

def run_packing():
    centers, radii = init_hex(N_CIRCLES)
    x0 = np.concatenate([centers.flatten(), radii])
    
    bounds = [(0.0, 1.0)] * (2 * N_CIRCLES) + [(0.0, None)] * N_CIRCLES
    cons = ({'type': 'ineq', 'fun': constraints_fun, 'jac': constraints_jac})
    
    try:
        res = minimize(objective, x0, method='SLSQP', jac=objective_jac, bounds=bounds,
                       constraints=cons, options={'maxiter': 3000, 'ftol': 1e-12, 'disp': False})
                       
        if -res.fun > 2.5:
            final_centers = res.x[:2 * N_CIRCLES].reshape(N_CIRCLES, 2)
            final_radii = res.x[2 * N_CIRCLES:]
            final_radii = np.maximum(final_radii, 0.0)
            # Final safety clamp
            for i in range(N_CIRCLES):
                final_centers[i, 0] = np.clip(final_centers[i, 0], final_radii[i], 1.0 - final_radii[i])
                final_centers[i, 1] = np.clip(final_centers[i, 1], final_radii[i], 1.0 - final_radii[i])
            return final_centers, final_radii, np.sum(final_radii)
    except Exception:
        pass
        
    return force_packing_fallback()
