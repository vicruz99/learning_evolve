from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    # Initial hexagonal packing heuristic
    # Parameters based on estimated max radius for 26 circles
    r_init = 0.105
    x_coords = []
    y_coords = []
    row = 0

            x = 0.5 + (col - 2) * 2 * r_init * np.cos(np.pi/6)

            y = 0.5 + (row - 2) * 2 * r_init * np.sin(np.pi/6)

            if 0 <= x <= 1 and 0 <= y <= 1:
                x_coords.append(x)
                y_coords.append(y)

                    break
        row += 1
        if row > 5: break

    while len(x_coords) < n:
        x_coords.append(0.5)
        y_coords.append(0.5)

    r_init_arr = np.full(n, r_init)

    x0 = np.concatenate([x_init, y_init, r_init_arr])
    
    def objective(vars):
        return -np.sum(vars[52:])

    def constraints(vars):
        constraints_list = []
        centers = vars[:52].reshape(n, 2)
        radii = vars[52:]

        # Boundary constraints
        for i in range(n):
            constraints_list.append(centers[i, 0] - radii[i])       # x >= r
            constraints_list.append(1 - centers[i, 0] - radii[i])   # 1-x >= r
            constraints_list.append(centers[i, 1] - radii[i])       # y >= r
            constraints_list.append(1 - centers[i, 1] - radii[i])   # 1-y >= r

                dist_sq = np.sum((centers[i] - centers[j])**2)
                sum_r = radii[i] + radii[j]
                # dist >= r_i + r_j  =>  dist^2 >= (r_i + r_j)^2
                constraints_list.append(dist_sq - sum_r**2)
                
        return constraints_list

    bounds = [(0, 1) for _ in range(2 * n)] + [(0, 0.5) for _ in range(n)]
    
    # Run optimization
    res = minimize(

        x0, 
        method='SLSQP', 
        bounds=bounds,
        constraints={'type': 'ineq', 'fun': constraints},
        options={'maxiter': 1000, 'disp': False}

        centers[i, 1] = np.clip(centers[i, 1], r, 1 - r)

    sum_radii = float(np.sum(radii))
    return centers, radii, sum_radii
