import math

def run_packing():

    Packs 26 circles in a unit square to maximize the sum of radii.
    Strategy: 5x5 grid of radius 0.1 + 1 larger circle in the central gap.
    """
    # Radius for the 25 grid circles
    r_grid = 0.1
    centers = []
    radii = []
    
    # Generate 5x5 grid
    for i in range(5):
        for j in range(5):
            x = 0.1 + 0.2 * i
            y = 0.1 + 0.2 * j
            centers.append([x, y])
            radii.append(r_grid)

    # Distance to nearest grid center is sqrt(0.08). 
    # Max radius = distance - r_grid. Subtract epsilon for numerical safety.

    centers.append([0.5, 0.5])
    radii.append(r_center)
    
    centers = np.array(centers)
    radii = np.array(radii)
    total_sum = float(np.sum(radii))
    
    return centers, radii, total_sum
