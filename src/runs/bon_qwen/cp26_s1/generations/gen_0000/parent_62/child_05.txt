import numpy as np
from scipy.optimize import minimize

def calculate_distance_and_overlap(center1, center2, r1, r2):
    """Helper to calculate squared distance and overlap constraint."""
    dx = center1[0] - center2[0]
    dy = center1[1] - center2[1]
    return dx*dx + dy*dy - (r1 + r2)**2

def objective_function(vars):
    """Objective: Maximize sum of radii (minimize negative sum)."""
    radii = vars[2::3]
    return -np.sum(radii)

def boundary_constraints(vars, n_circles):
    """Returns array of constraint values (must be >= 0)."""
    constraints = []
    for i in range(n_circles):
        x = vars[3*i]
        y = vars[3*i+1]
        r = vars[3*i+2]
        # Left, Right, Bottom, Top
        constraints.append(x - r)
        constraints.append(1.0 - x - r)
        constraints.append(y - r)
        constraints.append(1.0 - y - r)
    return np.array(constraints)

def overlap_constraints(vars, n_circles):
    """Returns array of constraint values for pairwise non-overlap."""
    constraints = []
    for i in range(n_circles):
        for j in range(i + 1, n_circles):
            ci = np.array([vars[3*i], vars[3*i+1]])
            cj = np.array([vars[3*j], vars[3*j+1]])
            ri = vars[3*i+2]
            rj = vars[3*j+2]
            dist_sq = np.sum((ci - cj)**2)
            min_dist_sq = (ri + rj)**2
            constraints.append(dist_sq - min_dist_sq)
    return np.array(constraints)

def hexagonal_init(n_circles):
    """Generates a hexagonal lattice initialization for n circles."""
    # Target radius estimate for 26 circles is approx 0.101
    # Start with a slightly smaller radius to ensure validity
    r_init = 0.095
    dx = 2.0 * r_init
    dy = np.sqrt(3.0) * r_init
    
    centers = []
    i, j = 0, 0
    while len(centers) < n_circles:
        # Hex grid offset
        x = 0.05 + (j * dx + (i % 2) * r_init)
        y = 0.05 + (i * dy)
        
        if 0 <= x <= 1 and 0 <= y <= 1:
            centers.append([x, y, r_init])
        
        j += 1
        if j * dx > 0.9: # Width limit check for next row
            i += 1
            j = 0
            
    return np.array(centers[:n_circles]).flatten()

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n_circles = 26
    best_sum = -np.inf
    best_vars = None

    # Define optimization bounds [x, y, r]
    bounds = []
    for _ in range(n_circles):
        bounds.extend([(0, 1), (0, 1), (1e-6, 0.5)])

    # Define constraint structures for SLSQP
    b_cons = {'type': 'ineq', 'fun': lambda v: boundary_constraints(v, n_circles)}
    o_cons = {'type': 'ineq', 'fun': lambda v: overlap_constraints(v, n_circles)}
    cons = [b_cons, o_cons]

    # Multi-start optimization
    for _ in range(15):
        # Use hexagonal init with slight random perturbation
        x0 = hexagonal_init(n_circles)
        # Add noise to help escape local minima
        noise = np.random.uniform(-0.01, 0.01, len(x0))
        x0 += noise
        # Ensure initial radii are within bounds
        x0[2::3] = np.clip(x0[2::3], 0.05, 0.5)
        
        res = minimize(objective_function, x0, method='SLSQP', bounds=bounds, constraints=cons, options={'ftol': 1e-9, 'maxiter': 2000})
        
        if res.success or res.fun < best_sum: # Note: fun is negative sum, so lower is better
            curr_sum = -res.fun
            if curr_sum > best_sum:
                best_sum = curr_sum
                best_vars = res.x

    # Extract final results
    centers = np.zeros((n_circles, 2))
    radii = np.zeros(n_circles)
    for i in range(n_circles):
        centers[i] = [best_vars[3*i], best_vars[3*i+1]]
        radii[i] = best_vars[3*i+2]

    return centers, radii, float(best_sum)

    c, r, s = run_packing()
    print(f"Sum of radii: {s}")
