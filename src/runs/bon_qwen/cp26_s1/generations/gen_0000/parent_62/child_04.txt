import numpy as np
import scipy.optimize as opt

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square

    Args:
        centers: np.array of shape (n, 2) with (x, y) coordinates
        radii: np.array of shape (n) with radius of each circle

    Returns:
        True if valid, False otherwise
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True

def get_bounds(n):
    """
    Returns bounds for each variable.
    Variables order: x1, y1, r1, x2, y2, r2, ...
    x, y in [0, 1], r in [0, 0.5]
    """
    bnds = []
    for _ in range(n):
        bnds.append((0.0, 1.0)) # x
        bnds.append((0.0, 1.0)) # y
        bnds.append((0.0, 0.5)) # r
    return bnds

def create_constraints(n):
    """
    Creates constraint dictionaries for SLSQP.
    1. Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    2. Non-overlap constraints: dist^2 >= (r_i + r_j)^2
    """
    constraints = []

    # Boundary constraints
    for i in range(n):
        idx_x = i * 3
        idx_y = i * 3 + 1
        idx_r = i * 3 + 2
        
        # x - r >= 0  => x >= r
        def bound_x_ge_r(vars, i=i):
            return vars[i*3] - vars[i*3+2]
        constraints.append({'type': 'ineq', 'fun': bound_x_ge_r})

        # 1 - x - r >= 0 => x + r <= 1
        def bound_x_le_1(vars, i=i):
            return 1.0 - vars[i*3] - vars[i*3+2]
        constraints.append({'type': 'ineq', 'fun': bound_x_le_1})

        # y - r >= 0
        def bound_y_ge_r(vars, i=i):
            return vars[i*3+1] - vars[i*3+2]
        constraints.append({'type': 'ineq', 'fun': bound_y_ge_r})

        # 1 - y - r >= 0
        def bound_y_le_1(vars, i=i):
            return 1.0 - vars[i*3+1] - vars[i*3+2]
        constraints.append({'type': 'ineq', 'fun': bound_y_le_1})

    # Non-overlap constraints
    # (x_i - x_j)^2 + (y_i - y_j)^2 >= (r_i + r_j)^2
    # => (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
    for i in range(n):
        for j in range(i + 1, n):
            idx_xi, idx_yi, idx_ri = i*3, i*3+1, i*3+2
            idx_xj, idx_yj, idx_rj = j*3, j*3+1, j*3+2
            
            def dist_constraint(vars, i=i, j=j):
                dx = vars[idx_xi] - vars[idx_xj]
                dy = vars[idx_yi] - vars[idx_yj]
                dr = vars[idx_ri] + vars[idx_rj]
                return dx*dx + dy*dy - dr*dr
            
            constraints.append({'type': 'ineq', 'fun': dist_constraint})

    return constraints

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # Initial Configuration: Hexagonal-ish grid
    # Place 25 circles in a 5x5 grid
    # Place 1 circle in a gap
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    init_r = 0.05 # Start with small feasible radius
    
    idx = 0
    # 5x5 grid
    grid_spacing = 0.2
    for row in range(5):
        for col in range(5):
            centers[idx, 0] = 0.1 + col * grid_spacing
            centers[idx, 1] = 0.1 + row * grid_spacing
            radii[idx] = init_r
            idx += 1
            
    # 26th circle in a gap, e.g., between (0.1, 0.1) and (0.3, 0.3)
    # Center of gap is roughly (0.2, 0.2)
    centers[25, 0] = 0.2
    centers[25, 1] = 0.2
    radii[25] = init_r
    
    # Flatten variables: x1, y1, r1, x2, y2, r2, ...
    x0 = np.zeros(3 * n)
    for i in range(n):
        x0[i*3] = centers[i, 0]
        x0[i*3+1] = centers[i, 1]
        x0[i*3+2] = radii[i]
        
    # Bounds and Constraints
    bnds = get_bounds(n)
    cons = create_constraints(n)
    
    # Objective: minimize -sum(radii)
    def objective(vars):
        s = 0.0
        for i in range(n):
            s += vars[i*3 + 2]
        return -s

    # Run Optimization
    # SLSQP is suitable for this type of problem
    res = opt.minimize(objective, x0, method='SLSQP', bounds=bnds, constraints=cons, 
                       options={'maxiter': 1000, 'ftol': 1e-9})
    
    # Extract results
    final_vars = res.x
    final_centers = np.zeros((n, 2))
    final_radii = np.zeros(n)
    
    total_sum = 0.0
    for i in range(n):
        final_centers[i, 0] = final_vars[i*3]
        final_centers[i, 1] = final_vars[i*3+1]
        final_radii[i] = final_vars[i*3+2]
        total_sum += final_radii[i]
        
    # Sanity check (optional, but good for debugging)
    # if not validate_packing(final_centers, final_radii):
    #     print("WARNING: Validation failed!")
        
    return (final_centers, final_radii, total_sum)

if __name__ == "__main__":
    c, r, s = run_packing()
    print(f"Sum of radii: {s}")
    print(f"Min radius: {np.min(r)}")
    print(f"Max radius: {np.max(r)}")
    print(validate_packing(c, r))
