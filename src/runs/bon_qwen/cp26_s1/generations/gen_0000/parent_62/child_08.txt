import numpy as np
from scipy.optimize import minimize


def penalty_function(x, n):
    """Compute negative sum of radii plus penalty for constraint violations."""
    centers = x[:2 * n].reshape((n, 2))
    radii = x[2 * n:]

    obj = -np.sum(radii)
    penalty = 0.0

    # Boundary penalties
    for i in range(n):
        cx, cy = centers[i]
        r = radii[i]
        # x >= r
        if cx < r:
            penalty += 500.0 * (r - cx) ** 2
        # x <= 1 - r
        if cx > 1 - r:
            penalty += 500.0 * (cx - (1 - r)) ** 2
        # y >= r
        if cy < r:
            penalty += 500.0 * (r - cy) ** 2
        # y <= 1 - r
        if cy > 1 - r:
            penalty += 500.0 * (cy - (1 - r)) ** 2

    # Overlap penalties
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            min_dist = radii[i] + radii[j]
            if dist < min_dist:
                penalty += 500.0 * (min_dist - dist) ** 2

    return obj + penalty


def make_initial_placement(n, row_counts, perturbation=0.0):
    """Create initial hexagonal-like placement."""
    centers = np.zeros((n, 2))
    idx = 0

    y_spacing = 0.20
    x_spacing = 0.17

    for row_idx, count in enumerate(row_counts):
        y = 0.08 + row_idx * y_spacing
        offset = x_spacing * 0.5 if row_idx % 2 == 1 else 0
        for col_idx in range(count):
            x = 0.05 + col_idx * x_spacing + offset
            centers[idx, 0] = max(0.02, min(0.98, x))
            centers[idx, 1] = max(0.02, min(0.98, y))
            idx += 1

    if perturbation > 0:
        centers += np.random.randn(n, 2) * perturbation
        centers = np.clip(centers, 0.02, 0.98)

    radii = np.ones(n) * 0.04
    return centers, radii


def post_process(centers, radii, n):
    """Enforce hard constraints after optimization."""
    # Clamp radii to boundary limits
    for i in range(n):
        max_r = min(
            centers[i, 0],
            1 - centers[i, 0],
            centers[i, 1],
            1 - centers[i, 1]
        )
        radii[i] = min(radii[i], max(0, max_r - 1e-10))

    # Fix overlaps by iteratively reducing radii
    for iteration in range(300):
        changed = False
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                total_r = radii[i] + radii[j]
                if total_r > dist - 1e-10:
                    excess = total_r - dist + 1e-10
                    if total_r > 1e-15:
                        # Reduce proportionally to current radii
                        radii[i] = max(0, radii[i] - excess * radii[i] / total_r)
                        radii[j] = max(0, radii[j] - excess * radii[j] / total_r)
                        changed = True
                    else:
                        radii[i] = max(0, (dist - 1e-10) / 2)
                        radii[j] = max(0, (dist - 1e-10) / 2)
                        changed = True
        if not changed:
            break

    # Re-clamp to boundaries after overlap fixing
    for i in range(n):
        max_r = min(
            centers[i, 0],
            1 - centers[i, 0],
            centers[i, 1],
            1 - centers[i, 1]
        )
        radii[i] = min(radii[i], max(0, max_r - 1e-10))

    return centers, radii

                lambda x: penalty_function(x, n),
