import numpy as np
from scipy.optimize import minimize

def objective_func(params, n):
    centers = params[:2*n].reshape(n, 2)
    radii = params[2*n:]
    
    loss = -np.sum(radii)
    
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.hypot(dx, dy)
            overlap = radii[i] + radii[j] - dist
            if overlap > 0:
                loss += 500.0 * overlap**2
                
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if r - x > 0: loss += 500.0 * (r - x)**2
        if x + r - 1 > 0: loss += 500.0 * (x + r - 1)**2
        if r - y > 0: loss += 500.0 * (r - y)**2
        if y + r - 1 > 0: loss += 500.0 * (y + r - 1)**2
        
    return loss

def run_packing():
    n = 26
    best_val = -1.0
    best_centers = None
    best_radii = None
    
    configs = []
    # Random configurations
    for _ in range(10):
        centers = np.random.rand(n, 2) * 0.6 + 0.2
        radii = np.ones(n) * 0.05
        configs.append(np.concatenate([centers.flatten(), radii]))
        
    # Hexagonal-ish configuration
    centers_hex = []
    r_init = 0.1
    for row in range(6):
        for col in range(6):
            if len(centers_hex) >= n: break
            x = 0.05 + col * 0.22 + (row % 2) * 0.11
            y = 0.05 + row * 0.19
            if 0 <= x <= 1 and 0 <= y <= 1:
                centers_hex.append([x, y])
    if len(centers_hex) < n:
        for _ in range(n - len(centers_hex)):
            centers_hex.append(np.random.rand(2) * 0.6 + 0.2)
    configs.append(np.concatenate([np.array(centers_hex[:n]).flatten(), np.ones(n) * r_init]))
    
    bounds = [(0.0, 1.0)] * (2 * n) + [(1e-4, 0.5)] * n
    
    for x0 in configs:
        res = minimize(objective_func, x0, args=(n,), method='L-BFGS-B', bounds=bounds, options={'maxiter': 5000, 'ftol': 1e-10})
        
        c_opt = res.x[:2*n].reshape(n, 2)
        r_opt = res.x[2*n:]
        
        valid = True
        for i in range(n):
            for j in range(i + 1, n):
                d = np.hypot(c_opt[i, 0] - c_opt[j, 0], c_opt[i, 1] - c_opt[j, 1])
                if d < r_opt[i] + r_opt[j] - 1e-12:
                    valid = False
                    break
            if not valid: break
            
        if valid:
            for i in range(n):
                x, y = c_opt[i]
                r = r_opt[i]
                if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                    valid = False
                    break
                    
        if valid:
            curr_sum = np.sum(r_opt)
            if curr_sum > best_val:
                best_val = curr_sum
                best_centers = c_opt.copy()
                best_radii = r_opt.copy()
        else:
            factor = 0.98
            r_fix = r_opt * factor
            valid_fix = True
            for i in range(n):
                for j in range(i + 1, n):
                    d = np.hypot(c_opt[i, 0] - c_opt[j, 0], c_opt[i, 1] - c_opt[j, 1])
                    if d < r_fix[i] + r_fix[j] - 1e-12:
                        valid_fix = False
                        break
                if not valid_fix: break
            if not valid_fix:
                r_fix = r_opt * 0.95
                
            if np.sum(r_fix) > best_val:
                best_val = np.sum(r_fix)
                best_centers = c_opt.copy()
                best_radii = r_fix.copy()
                
    return best_centers, best_radii, best_val
