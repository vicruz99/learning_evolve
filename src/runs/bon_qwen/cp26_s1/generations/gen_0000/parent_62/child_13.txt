def compute_repulsion_forces(centers, radii):

    n = centers.shape[0]
    forces = np.zeros_like(centers)
    
    # Pairwise repulsion
    for i in range(n):
        for j in range(i + 1, n):
            diff = centers[i] - centers[j]
            dist = np.linalg.norm(diff)
            min_dist = radii[i] + radii[j]
            
            if dist < min_dist and dist > 1e-12:
                # Overlap detected, push apart
                overlap = min_dist - dist
                direction = diff / dist
                # Split force equally
                forces[i] += direction * overlap * 0.5
                forces[j] -= direction * overlap * 0.5
                
    # Boundary forces
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        
        # Left wall

        # Right wall

            forces[i, 0] -= ((x + r) - 1) * 2.0

            forces[i, 1] += (0 - (y - r)) * 2.0
        # Top wall
        if y + r > 1:
