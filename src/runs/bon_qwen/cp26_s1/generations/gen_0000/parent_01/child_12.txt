def generate_initial_hexagonal(n, seed):

    rng_state = np.random.get_state()

    r = 0.088
    centers = []
    
    row = 0
    while len(centers) < n:
        y = r + row * r * np.sqrt(3.0)

            break

            x = r
        else:
            x = 2.0 * r
        while x + r <= 1.0 and len(centers) < n:
            centers.append([x, y])
            x += 2.0 * r
        row += 1
    
    # Add small random perturbations to escape symmetry

        centers[i][0] += (np.random.random() - 0.5) * 0.005
        centers[i][1] += (np.random.random() - 0.5) * 0.005

        centers.append([0.5, 0.5])

    return np.array(centers[:n]), np.ones(n) * r

            penalty += alpha * diff * diff
        # Right boundary: x <= 1 - r
        if x > 1.0 - r:
            diff = x + r - 1.0
            penalty += alpha * diff * diff
        # Bottom boundary: y >= r
        if y < r:
            diff = r - y
            penalty += alpha * diff * diff
        # Top boundary: y <= 1 - r
        if y > 1.0 - r:
            diff = y + r - 1.0
            penalty += alpha * diff * diff
    
    # Non-overlap constraint penalties

                    diff = min_dist - dist
                    penalty += alpha * diff * diff

                    penalty += alpha * min_dist * min_dist

def is_valid_packing(centers, radii, n):
    """Check if packing is valid (no NaN, non-negative radii, inside square, no overlaps)."""
    if np.isnan(centers).any() or np.isnan(radii).any():
        return False

        if radii[i] < 0:
            return False

        y = centers[i, 1]
        r = radii[i]
        
        # Boundary check
        if x - r < -1e-10 or x + r > 1.0 + 1e-10:
            return False

            return False
    
    # Overlap check
    for i in range(n):

            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            if dist < radii[i] + radii[j] - 1e-10:
                return False
    
    return True
