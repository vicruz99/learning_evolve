from scipy.optimize import minimize

def obj_func(v):
    n = 26
    return -np.sum(v[2*n:])

def con_func(v):

    centers = v[:2*n].reshape((n, 2))
    radii = v[2*n:]
    
    # Boundary constraints: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    bc = np.concatenate([
        centers[:, 0] - radii,
        1.0 - centers[:, 0] - radii,
        centers[:, 1] - radii,
        1.0 - centers[:, 1] - radii
    ])
    
    # Pairwise non-overlap constraints: dist^2 >= (r_i + r_j)^2
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]

    r2 = (radii[:, np.newaxis] + radii[np.newaxis, :])**2
    
    idx = np.triu_indices(n, k=1)
    pc = d2[idx] - r2[idx]
    
    return np.concatenate([bc, pc])
