import numpy as np
from scipy.optimize import minimize

def get_hex_centers(n):
    """Generate initial centers in a hexagonal grid pattern."""
    cols = int(np.ceil(np.sqrt(n * np.sqrt(3) / 2)))
    rows = int(np.ceil(n / cols))
    
    centers = []

        for c in range(cols):

                break

            if r % 2 == 1:
                x += 1.0 / (2 * cols)

            centers.append([x, y])
        if len(centers) >= n:
            break
            
    return np.array(centers[:n])

def constraints_builder(x, y, r, centers, radii):

    n = len(radii)
    cons = []
    
    # Boundary constraints
    for i in range(n):
        cons.append({'type': 'ineq', 'fun': lambda v, i=i: v[3*i] - v[3*i+2]}) # x - r >= 0
        cons.append({'type': 'ineq', 'fun': lambda v, i=i: 1 - v[3*i] - v[3*i+2]}) # 1 - x - r >= 0
        cons.append({'type': 'ineq', 'fun': lambda v, i=i: v[3*i+1] - v[3*i+2]}) # y - r >= 0
        cons.append({'type': 'ineq', 'fun': lambda v, i=i: 1 - v[3*i+1] - v[3*i+2]}) # 1 - y - r >= 0

            def dist_con(v, i=i, j=j):

                return np.sqrt(d_sq) - (ri + rj)
            cons.append({'type': 'ineq', 'fun': dist_con})

    init_centers = get_hex_centers(n)
