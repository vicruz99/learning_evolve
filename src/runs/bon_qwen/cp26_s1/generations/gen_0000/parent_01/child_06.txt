**Strategy:**
1. **Penalty Method Formulation:** We convert the constrained problem into an unconstrained one using a quadratic penalty function. The objective becomes minimizing $-\sum r_i + \mu \cdot P(\text{centers}, \text{radii})$, where $P$ penalizes boundary violations and inter-circle overlaps. As $\mu$ increases, the optimizer is forced to satisfy constraints strictly while pushing radii to their maximum feasible values.
2. **Iterative Annealing:** We start with a moderate penalty weight and gradually increase it geometrically. This allows the optimizer to first explore large-radius configurations, then systematically resolve overlaps by moving centers apart or reducing radii locally, finally converging to a tightly packed, valid configuration.
3. **Multi-Start Initialization:** Since the landscape has many local minima, we run the optimization from multiple starting points. One starts with a hexagonal lattice (highly efficient for circle packing), while others use randomized placements to escape symmetric traps.
4. **Robust Numerical Solver:** `scipy.optimize.minimize` with `L-BFGS-B` is used. It handles box constraints efficiently and converges quickly for this scale ($N=26$).

import numpy as np
from scipy.optimize import minimize

def compute_objective(p, n, mu):
    """
    Computes the penalized objective: -sum(radii) + mu * penalty
    Penalty accounts for boundary violations and circle overlaps.
    """
    c = p[:2*n].reshape(n, 2)
    r = p[2*n:]
    
    obj = -np.sum(r)
    penalty = 0.0
    
    for i in range(n):
        xi, yi, ri = c[i, 0], c[i, 1], r[i]
        
        # Boundary constraints
        if xi < ri:
            penalty += (ri - xi)**2
        if xi > 1.0 - ri:
            penalty += (xi + ri - 1.0)**2
        if yi < ri:
            penalty += (ri - yi)**2
        if yi > 1.0 - ri:
            penalty += (yi + ri - 1.0)**2
            
        for j in range(i + 1, n):
            dx = xi - c[j, 0]
            dy = yi - c[j, 1]
            dist = np.sqrt(dx*dx + dy*dy)
            if dist < ri + r[j]:
                penalty += (ri + r[j] - dist)**2
                
    return obj + mu * penalty

def run_packing():
    n = 26
    best_params = None
    best_sum_r = -1.0
    
    # Define bounds for [x, y, r] for each circle
    bounds_list = []
    for _ in range(n):
        bounds_list.extend([(0.001, 0.999), (0.001, 0.999), (0.001, 0.5)])
        
    # Multi-start optimization
    for seed in range(6):
        np.random.seed(seed)
        
        # Initialize centers
        centers = np.random.uniform(0.2, 0.8, size=(n, 2))
        if seed == 0:
            # Hexagonal lattice initialization for better starting density
            cnt = 0
            y = 0.15
            while cnt < n:
                x = 0.15
                while x < 0.9 and cnt < n:
                    centers[cnt] = [x, y]
                    cnt += 1
                    x += 0.15
                y += 0.13
                if cnt < n:
                    centers[cnt] = [x + 0.075, y]
                    cnt += 1
                    
        radii = np.full(n, 0.08)
        
        params = np.zeros(n * 3)
        for i in range(n):
            params[3*i] = centers[i, 0]
            params[3*i+1] = centers[i, 1]
            params[3*i+2] = radii[i]
            
        current_params = params.copy()
        mu = 50.0
        
        # Iterative penalty annealing
        for step in range(40):
            res = minimize(compute_objective, current_params, args=(n, mu), 
                           method='L-BFGS-B', bounds=bounds_list,
                           options={'maxiter': 200, 'ftol': 1e-14, 'gtol': 1e-14, 'disp': False})
            current_params = res.x
            mu *= 1.5
            
        r_val = current_params[2*n:]
        s = np.sum(r_val)
        if s > best_sum_r:
            best_sum_r = s
            best_params = current_params.copy()
            
    centers_opt = best_params[:2*n].reshape(n, 2)
    radii_opt = best_params[2*n:]
    
    # Safety check and scaling to guarantee strict validity
    max_viol = 0.0
    for i in range(n):
        ri = radii_opt[i]
        cx, cy = centers_opt[i, 0], centers_opt[i, 1]
        max_viol = max(max_viol, ri - cx, cx + ri - 1.0, ri - cy, cy + ri - 1.0)
        for j in range(i + 1, n):
            d = np.sqrt((cx - centers_opt[j, 0])**2 + (cy - centers_opt[j, 1])**2)
            max_viol = max(max_viol, ri + radii_opt[j] - d)
            
    if max_viol > 1e-12:
        scale = 1.0 - (max_viol + 1e-13)
        radii_opt *= scale
        for i in range(n):
            ri = radii_opt[i]
            centers_opt[i, 0] = max(ri, min(1.0 - ri, centers_opt[i, 0]))
            centers_opt[i, 1] = max(ri, min(1.0 - ri, centers_opt[i, 1]))

    return centers_opt, radii_opt, float(np.sum(radii_opt))
