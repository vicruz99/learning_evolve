

To maximize the sum of radii for 26 circles in a unit square, we use a numerical optimization approach. This involves initializing the circles in a dense configuration and iteratively optimizing their positions and radii using the L-BFGS-B algorithm with a smooth penalty function. The penalty function handles boundary constraints and circle-circle overlaps, allowing the solver to push the system toward a valid, high-sum configuration.

### Strategy
1.  **Initialization:** Generate an initial grid configuration of 26 circles with small radii (0.02) to ensure the solver starts within a valid state.
2.  **Penalty Function:** Define a composite objective function: `minimize -Sum(Radii) + Penalty`. 
    *   **Overlap Penalty:** Penalizes when `r_i + r_j > distance(center_i, center_j)`.
    *   **Boundary Penalty:** Penalizes circles that fall outside the unit square `[0, 1]`.
3.  **Optimization:** Use `scipy.optimize.minimize` with the `L-BFGS-B` method. This method is efficient for handling bound constraints and provides analytical gradients for the smooth penalty terms to ensure rapid convergence.
4.  **Robustness:** To mitigate local minima, the optimizer runs multiple times from slightly perturbed initial configurations and returns the best valid result.

```python
import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    """
    n = 26
    
    def objective(vars):
        centers = vars[:2 * n].reshape(n, 2)
        radii = vars[2 * n:]
        x, y = centers[:, 0], centers[:, 1]
        r = radii

        # Smooth boundary penalties
        pen = np.sum(np.maximum(0, r - x)**2) + np.sum(np.maximum(0, r - y)**2)
        pen += np.sum(np.maximum(0, x + r - 1)**2) + np.sum(np.maximum(0, y + r - 1)**2)

        # Smooth overlap penalties
        diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
        dist = np.sqrt(np.sum(diff**2, axis=2))
        np.fill_diagonal(dist, 1e-12)
        r_sum = r[:, np.newaxis] + r[np.newaxis, :]
        overlap = np.maximum(0, r_sum - dist)
        pen += np.sum(np.triu(overlap**2, k=1))

        return -np.sum(r) + pen

    def gradient(vars):
        centers = vars[:2 * n].reshape(n, 2)
        radii = vars[2 * n:]
        x, y = centers[:, 0], centers[:, 1]
        r = radii

        grad_c = np.zeros_like(centers)
        grad_r = np.zeros_like(radii)

        # Boundary gradients
        grad_c[np.where(r > x), 0] -= 2 * (r[x < r] - x[x < r])
        grad_r[r > x] += 2 * (r[r > x] - x[r > x])
        grad_c[np.where(r > y), 1] -= 2 * (r[y < r] - y[y < r])
        grad_r[r > y] += 2 * (r[r > y] - y[r > y])

        val_xr = x + r - 1; val_yr = y + r - 1
        grad_c[np.where(val_xr > 0), 0] += 2 * val_xr[val_xr > 0]
        grad_r[val_xr > 0] += 2 * val_xr[val_xr > 0]
        grad_c[np.where(val_yr > 0), 1] += 2 * val_yr[val_yr > 0]
        grad_r[val_yr > 0] += 2 * val_yr[val_yr > 0]

        # Overlap gradients
        diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
        dist = np.sqrt(np.sum(diff**2, axis=2))
        np.fill_diagonal(dist, 1e-12)
        overlap = np.maximum(0, r[:, np.newaxis] + r[np.newaxis, :] - dist)
        np.fill_diagonal(overlap, 0)
        
        grad_r += 2 * np.sum(overlap, axis=1)
        unit_vec = np.where(dist > 1e-12, diff / dist[:, :, np.newaxis], 0)
        # Vectorized application of gradient on centers
        force = 2 * overlap[:, :, np.newaxis] * unit_vec
        grad_c += np.sum(force, axis=1)

        return np.concatenate([grad_c.flatten(), grad_r])

    best_sum = 0
    best_c, best_r = None, None
    rng = np.random.RandomState(42)

    # Generate initial configurations
    inits = []
    for _ in range(10):
        c = np.zeros((n, 2))
        r = np.ones(n) * 0.02
        # Grid-like placement
        rows, cols = 6, 5
        dx, dy = 0.2, 0.2
        for i in range(n):
            c[i, 0] = 0.1 + (i % cols) * dx + rng.uniform(-0.02, 0.02)
            c[i, 1] = 0.1 + (i // cols) * dy + rng.uniform(-0.02, 0.02)
        inits.append(np.concatenate([c.flatten(), r]))

    for vars0 in inits:
        res = minimize(objective, vars0, jac=gradient, method='L-BFGS-B', 
                       bounds=[(0, 1)] * (2 * n) + [(0, 0.5)] * n, options={'maxiter': 2000})
        
        c = res.x[:2 * n].reshape(n, 2)
        r = res.x[2 * n:]
        
        # Final validation and clamping
        for i in range(n):
            c[i, 0] = np.clip(c[i, 0], r[i], 1 - r[i])
            c[i, 1] = np.clip(c[i, 1], r[i], 1 - r[i])

        s = np.sum(r)
        if s > best_sum:
            best_sum = s
            best_c, best_r = c, r

    return best_c, best_r, best_sum
```
