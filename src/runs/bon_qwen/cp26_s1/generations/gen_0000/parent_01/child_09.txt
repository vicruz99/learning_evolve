import numpy as np
from scipy.optimize import minimize

def run_packing():
    n = 26
    np.random.seed(42)
    
    # 1. Initialize with a hexagonal lattice pattern
    centers = np.zeros((n, 2))
    idx = 0
    for r in range(6):
        for c in range(6):
            if idx >= n: break
            centers[idx, 0] = c + (r % 2) * 0.5
            centers[idx, 1] = r * np.sqrt(3) / 2
            idx += 1
            
    # Normalize to fit inside [0.05, 0.95] initially to avoid boundary traps
    cx, cy = centers[:, 0], centers[:, 1]
    centers[:, 0] = 0.05 + 0.9 * (cx - cx.min()) / (cx.max() - cx.min() + 1e-12)
    centers[:, 1] = 0.05 + 0.9 * (cy - cy.min()) / (cy.max() - cy.min() + 1e-12)
    
    radii = np.full(n, 0.05)
    penalty = 100.0
    
    # 2. Sequential penalty optimization
    for step in range(40):
        x0 = np.concatenate([centers.flatten(), radii])
        
        def objective(vars):
            c = vars[:2*n].reshape(n, 2)
            rad = vars[2*n:]
            p = 0.0
            # Boundary penalties
            for i in range(n):
                if c[i,0] < rad[i]: p += (rad[i] - c[i,0])**2
                if c[i,0] > 1-rad[i]: p += (c[i,0] - (1-rad[i]))**2
                if c[i,1] < rad[i]: p += (rad[i] - c[i,1])**2
                if c[i,1] > 1-rad[i]: p += (c[i,1] - (1-rad[i]))**2
            # Overlap penalties
            for i in range(n):
                for j in range(i+1, n):
                    d = np.sqrt(np.sum((c[i] - c[j])**2) + 1e-12)
                    if d < rad[i] + rad[j]:
                        p += (rad[i] + rad[j] - d)**2
            return -np.sum(rad) + penalty * p
            
        bnds = [(0.0, 1.0)]*(2*n) + [(0.0, 0.5)]*n
        res = minimize(objective, x0, method='L-BFGS-B', bounds=bnds, 
                       options={'maxiter': 500, 'ftol': 1e-10, 'gtol': 1e-7})
        
        centers = res.x[:2*n].reshape(n, 2)
        radii = res.x[2*n:]
        
        # Check constraint satisfaction
        penalty_term = 0.0
        for i in range(n):
             if centers[i,0] < radii[i]: penalty_term += (radii[i] - centers[i,0])**2
             if centers[i,0] > 1-radii[i]: penalty_term += (centers[i,0] - (1-radii[i]))**2
             if centers[i,1] < radii[i]: penalty_term += (radii[i] - centers[i,1])**2
             if centers[i,1] > 1-radii[i]: penalty_term += (centers[i,1] - (1-radii[i]))**2
        for i in range(n):
            for j in range(i+1, n):
                d = np.sqrt(np.sum((centers[i] - centers[j])**2))
                if d < radii[i] + radii[j]:
                    penalty_term += (radii[i] + radii[j] - d)**2
                    
        if penalty_term < 1e-4:
            radii = np.clip(radii * 1.005, 0, 0.5) # Grow radii if constraints satisfied
            penalty *= 1.5
        else:
            penalty *= 1.2
            
    # 3. Final validation and repair to guarantee validity
    centers = np.clip(centers, 0, 1)
    for i in range(n):
        radii[i] = min(radii[i], centers[i,0], 1-centers[i,0], centers[i,1], 1-centers[i,1])
        
    # Resolve any remaining numerical overlaps
    for _ in range(20):
        changed = False
        for i in range(n):
            for j in range(i+1, n):
                d = np.sqrt(np.sum((centers[i] - centers[j])**2))
                req = radii[i] + radii[j]
                if d < req - 1e-7:
                    scale = d / req
                    radii[i] *= scale
                    radii[j] *= scale
                    changed = True
        if not changed:
            break
            
    return centers, radii, float(np.sum(radii))
