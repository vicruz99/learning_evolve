import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Solves the circle packing problem for 26 circles in a unit square.
    Maximizes the sum of radii by optimizing positions and a common radius.
    """
    n = 26
    # We optimize for equal radii first as a strong heuristic.
    # Variables: [x_0, y_0, x_1, y_1, ..., x_25, y_25, r]

    # and provide a good starting point for the optimizer.
    centers = np.zeros((n, 2))

    initial_r = 0.08

    idx = 0
    y = initial_r
    while idx < n:
        x = initial_r
        row_count = 0
        while x <= 1 - initial_r and idx < n:
            centers[idx] = [x, y]
            idx += 1
            row_count += 1
            x += 2 * initial_r # Move to next circle in row
        
        # Shift next row for hexagonal packing
        y += initial_r * np.sqrt(3)
        if y + initial_r > 1:
            y = 1 - initial_r # Push back if overflow
            # Reset x to fill remaining if needed, though logic above handles count

        r = vars[-1]
        return -n * r

                return vars[2*i] - vars[r_idx]
            return con
        constraints.append({'type': 'ineq', 'fun': make_con_x_min(i, 2*n)})

        def make_con_x_max(i, r_idx):
            def con(vars):
                return 1.0 - vars[2*i] - vars[r_idx]
            return con
        constraints.append({'type': 'ineq', 'fun': make_con_x_max(i, 2*n)})
        
        # y_i >= r => y_i - r >= 0
        # Index of y_i is 2*i + 1
        def make_con_y_min(i, r_idx):
            def con(vars):
                return vars[2*i+1] - vars[r_idx]
            return con
        constraints.append({'type': 'ineq', 'fun': make_con_y_min(i, 2*n)})
        
        # 1 - y_i - r >= 0
        def make_con_y_max(i, r_idx):
            def con(vars):
                return 1.0 - vars[2*i+1] - vars[r_idx]
            return con
        constraints.append({'type': 'ineq', 'fun': make_con_y_max(i, 2*n)})
