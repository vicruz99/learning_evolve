To maximize the sum of radii for 26 circles in a unit square, we formulate the problem as a constrained nonlinear optimization task. The objective is to maximize $\sum r_i$, subject to:
1. Circle containment: $r_i \leq x_i \leq 1-r_i$ and $r_i \leq y_i \leq 1-r_i$
2. Non-overlap: $(x_i-x_j)^2 + (y_i-y_j)^2 \geq (r_i+r_j)^2$ for all $i < j$

We use `scipy.optimize.minimize` with the `SLSQP` method, which handles inequality constraints effectively. To avoid local minima and ensure robustness:
- We initialize from multiple starting configurations: a hexagonal lattice pattern (naturally efficient for packing) and several grid-based layouts with small initial radii to guarantee feasibility.
- Constraints are implemented as top-level functions returning arrays, adhering to the no-lambda and no-closure rules.
- Post-processing clips centers to boundaries and applies a tiny uniform shrink if numerical precision causes marginal overlaps, ensuring the validation function passes strictly.
- The optimization runs multiple times with different seeds, keeping the best valid configuration.

```python
import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def compute_overlap_constraints(p):
    """Returns an array of non-overlap constraint values: dist^2 - (r_i+r_j)^2 >= 0"""
    p = p.reshape(N_CIRCLES, 3)
    res = np.empty(N_CIRCLES * (N_CIRCLES - 1) // 2)
    idx = 0
    for i in range(N_CIRCLES):
        for j in range(i + 1, N_CIRCLES):
            dx = p[i, 0] - p[j, 0]
            dy = p[i, 1] - p[j, 1]
            rs = p[i, 2] + p[j, 2]
            res[idx] = dx**2 + dy**2 - rs**2
            idx += 1
    return res

def compute_box_constraints(p):
    """Returns an array of boundary constraint values: x-r >= 0, 1-x-r >= 0, etc."""
    p = p.reshape(N_CIRCLES, 3)
    res = np.empty(4 * N_CIRCLES)
    for i in range(N_CIRCLES):
        res[4 * i] = p[i, 0] - p[i, 2]
        res[4 * i + 1] = 1.0 - p[i, 0] - p[i, 2]
        res[4 * i + 2] = p[i, 1] - p[i, 2]
        res[4 * i + 3] = 1.0 - p[i, 1] - p[i, 2]
    return res

def objective_function(p):
    """Negative sum of radii (for minimization)"""
    return -np.sum(p[2::3])

def run_packing():
    # Bounds for x, y in [0, 1] and r in [1e-5, 0.5]
    bounds = [(0.0, 1.0)] * (N_CIRCLES * 3)
    for i in range(N_CIRCLES):
        bounds[3 * i + 2] = (1e-5, 0.5)

    constraints = [
        {'type': 'ineq', 'fun': compute_overlap_constraints},
        {'type': 'ineq', 'fun': compute_box_constraints}
    ]

    best_sum = -1.0
    best_centers = None
    best_radii = None

    # Initial Guess 1: Hexagonal lattice pattern
    r_init = 0.09
    count = 0
    x0_hex = np.zeros(N_CIRCLES * 3)
    row_counts = [6, 5, 6, 5, 4]
    for r_idx, nc in enumerate(row_counts):
        for k in range(nc):
            if count >= N_CIRCLES:
                break
            x = (k + 0.5) * 2 * r_init + (0.5 if r_idx % 2 == 1 else 0) * 2 * r_init
            y = (r_idx + 0.5) * 2 * r_init * np.sqrt(3) / 2
            x0_hex[3 * count] = x
            x0_hex[3 * count + 1] = y
            x0_hex[3 * count + 2] = r_init
            count += 1
    # Normalize to [0.1, 0.9]
    xs = x0_hex[0::3]
    ys = x0_hex[1::3]
    x0_hex[0::3] = (xs - np.min(xs)) / (np.max(xs) - np.min(xs) + 1e-6) * 0.8 + 0.1
    x0_hex[1::3] = (ys - np.min(ys)) / (np.max(ys) - np.min(ys) + 1e-6) * 0.8 + 0.1
    initial_guesses = [x0_hex]

    # Initial Guesses 2: Structured grid layouts with small radii (guaranteed feasible)
    for seed in [42, 123, 456]:
        np.random.seed(seed)
        x0_rand = np.zeros(N_CIRCLES * 3)
        idx = 0
        # Place in a loose grid to avoid initial overlaps
        for i in range(6):
            for j in range(5):
                if idx >= N_CIRCLES:
                    break
                x0_rand[3 * idx] = 0.12 + i * 0.14
                x0_rand[3 * idx + 1] = 0.12 + j * 0.14
                x0_rand[3 * idx + 2] = 0.035
                idx += 1
        # Add slight random perturbation
        for k in range(N_CIRCLES * 3):
            x0_rand[k] += np.random.uniform(-0.02, 0.02)
        x0_rand = np.clip(x0_rand, 0.05, 0.95)
        initial_guesses.append(x0_rand)

    # Run optimization from each start
    for x0 in initial_guesses:
        try:
            res = minimize(objective_function, x0, method='SLSQP', bounds=bounds,
                           constraints=constraints, options={'maxiter': 2000, 'ftol': 1e-10, 'disp': False})
            if -res.fun > best_sum:
                best_sum = -res.fun
                best_centers = res.x[:2 * N_CIRCLES].reshape(N_CIRCLES, 2)
                best_radii = res.x[2 * N_CIRCLES:]
        except Exception:
            continue

    if best_centers is None:
        # Fallback to hex configuration
        best_centers = x0_hex[:2 * N_CIRCLES].reshape(N_CIRCLES, 2)
        best_radii = x0_hex[2 * N_CIRCLES:]
        best_sum = np.sum(best_radii)

    # Post-processing for numerical safety
    for i in range(N_CIRCLES):
        best_centers[i, 0] = np.clip(best_centers[i, 0], best_radii[i], 1.0 - best_radii[i])
        best_centers[i, 1] = np.clip(best_centers[i, 1], best_radii[i], 1.0 - best_radii[i])

    # Verify and slightly shrink if any constraint is marginally violated
    min_gap = 1.0
    for i in range(N_CIRCLES):
        for j in range(i + 1, N_CIRCLES):
            dist = np.sqrt(np.sum((best_centers[i] - best_centers[j])**2))
            gap = dist - best_radii[i] - best_radii[j]
            if gap < min_gap:
                min_gap = gap
                
    if min_gap < -1e-9:
        shrink_factor = 1.0 + min_gap / (2.0 * np.mean(best_radii) + 1e-9)
        best_radii *= shrink_factor
        for i in range(N_CIRCLES):
            best_centers[i, 0] = np.clip(best_centers[i, 0], best_radii[i], 1.0 - best_radii[i])
            best_centers[i, 1] = np.clip(best_centers[i, 1], best_radii[i], 1.0 - best_radii[i])
            
    best_sum = np.sum(best_radii)
    return best_centers, best_radii, best_sum
```
