import numpy as np
from scipy.optimize import minimize

def run_packing():

    Packs 26 circles in a unit square to maximize the sum of radii.
    Returns: (centers, radii, sum_radii)
    """
    N = 26

    # A staggered grid provides a good initial spread
    np.random.seed(42)
    centers = np.zeros((N, 2))

    row_counts = [6, 5, 6, 5, 4]
    y_coords = np.linspace(0.15, 0.85, len(row_counts))
    
    idx = 0
    for r_idx, y in enumerate(y_coords):
        n_cols = row_counts[r_idx]
        # Calculate starting x to center the row
        x_start = (1.0 - (n_cols - 1) * 0.2) / 2.0
        if n_cols == 1:
            x_start = 0.5
        # Offset odd rows for hexagonal arrangement
        if r_idx % 2 == 1:
            x_start += 0.1
            
        for c in range(n_cols):
            x = x_start + c * 0.2
            centers[idx] = [x, y]
            idx += 1
            
    # Add small random perturbation to break symmetry
    centers += np.random.uniform(-0.005, 0.005, centers.shape)
    
    # Initial radii
    radii = np.full(N, 0.04)
    
    # 2. Iterative Optimization Loop
    growth_rate = 0.002
    max_iters = 200
    
    for it in range(max_iters):
        # Bounds for positions: [r, 1-r] for both x and y
        # Flattened bounds to match scipy's expected format
        bnds = [(max(0.0, radii[i]), min(1.0, 1.0 - radii[i])) for _ in range(2) for i in range(N)]
        c0 = centers.flatten()

        def obj(c_flat):
            c = c_flat.reshape(N, 2)
            # Pairwise differences
            diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
            dist = np.sqrt(np.sum(diff**2, axis=2))
            # Minimum required distance
            min_dist = radii[:, np.newaxis] + radii[np.newaxis, :]
            overlap = min_dist - dist
            # Zero out self-overlaps and negative overlaps (no collision)
            np.fill_diagonal(overlap, 0)
            overlap = np.maximum(overlap, 0)
            return np.sum(overlap**2)
        
        # Optimize positions
        res = minimize(obj, c0, method='L-BFGS-B', bounds=bnds, 
                       options={'maxiter': 300, 'ftol': 1e-10, 'gtol': 1e-8})
        centers = res.x.reshape(N, 2)

        radii += growth_rate
        
        # Adjust growth rate based on residual overlap
        if res.fun < 1e-8:

        else:
            growth_rate *= 0.92

        radii = np.clip(radii, 0, 0.49)
        
        # If growth becomes negligible, stop early
        if growth_rate < 1e-6:
            break
            
    # 3. Post-processing and Validation
    # Strictly enforce boundary constraints
    centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
    centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)

    # This handles any numerical precision issues from the optimizer
    valid = False
    shrink_factor = 1.0
    while not valid and shrink_factor > 0.999:
        shrink_factor -= 0.0001
        r_test = radii * shrink_factor
        valid = True
        # Check boundaries
        if np.any(centers[:, 0] - r_test < -1e-12) or np.any(centers[:, 0] + r_test > 1 + 1e-12):
            valid = False
        if np.any(centers[:, 1] - r_test < -1e-12) or np.any(centers[:, 1] + r_test > 1 + 1e-12):

            # Check overlaps

                valid = False

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
