import numpy as np
from scipy.optimize import minimize

def compute_penalty(z, n):
    """
    Computes the squared penalty for constraint violations.
    Constraints:
    1. r_i <= x_i
    2. r_i <= 1 - x_i
    3. r_i <= y_i
    4. r_i <= 1 - y_i
    5. r_i + r_j <= distance(i, j)
    """
    X = z[0::3]
    Y = z[1::3]
    R = z[2::3]

    # Boundary constraints
    p = np.sum(np.maximum(0.0, R - X)**2)
    p += np.sum(np.maximum(0.0, R - (1.0 - X))**2)
    p += np.sum(np.maximum(0.0, R - Y)**2)
    p += np.sum(np.maximum(0.0, R - (1.0 - Y))**2)

    # Pairwise distance constraints
    # Vectorized computation of distance matrix
    DX = X[:, None] - X[None, :]
    DY = Y[:, None] - Y[None, :]
    D = np.sqrt(DX**2 + DY**2)
    
    # Sum of radii matrix
    SR = R[:, None] + R[None, :]
    
    # Violation: max(0, SR - D)^2
    diff = SR - D
    
    # Only consider upper triangle to avoid double counting and self-interaction
    triu = np.triu_indices(n, k=1)
    p += np.sum(np.maximum(0.0, diff[triu])**2)
    
    return p

def combined_objective(z, n, rho):
    """Objective function: minimize negative sum of radii + penalty term"""
    return -np.sum(z[2::3]) + rho * compute_penalty(z, n)

def run_packing():
    """
    Optimizes circle packing in a unit square to maximize sum of radii.
    Uses a penalty method with L-BFGS-B and multiple restarts.
    """
    n = 26
    bounds = [(0, 1)] * n + [(0, 1)] * n + [(0, 0.5)] * n
    
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Run optimization with multiple seeds to escape local minima
    for seed in range(5):
        np.random.seed(seed)
        
        # Initialize centers in a hexagonal-like grid pattern with small random noise
        centers = np.zeros((n, 2))
        idx = 0
        for i in range(5):
            # Alternate row lengths to approximate hexagonal packing
            num_cols = 6 if i % 2 == 0 else 5
            y_base = (i + 0.5) / 5.0
            for j in range(num_cols):
                if idx >= n:
                    break
                x_base = (j + 0.5 + (0.5 if i % 2 == 1 else 0)) / 6.0
                # Add slight jitter to break symmetry and explore space
                centers[idx] = [
                    np.clip(x_base + np.random.normal(0, 0.02), 0.05, 0.95),
                    np.clip(y_base + np.random.normal(0, 0.02), 0.05, 0.95)
                ]
                idx += 1
            if idx >= n:
                break
        
        radii = np.full(n, 0.04)
        
        # Flatten to optimization vector z = [x1, y1, r1, x2, y2, r2, ...]
        z0 = np.zeros(3 * n)
        z0[0::3] = centers[:, 0]
        z0[1::3] = centers[:, 1]
        z0[2::3] = radii
        
        z = z0.copy()
        rho = 50.0
        
        # Iterative penalty increase
        for step in range(20):
            res = minimize(combined_objective, z, args=(n, rho),
                           method='L-BFGS-B', bounds=bounds,
                           options={'ftol': 1e-14, 'gtol': 1e-12, 'maxiter': 1500})
            z = res.x
            rho *= 2.0
            
            # Early stopping if constraints are tightly satisfied
            if compute_penalty(z, n) < 1e-10:
                break
        
        # Projection step: Adjust radii to strictly satisfy constraints based on optimized centers.
        # This guarantees validity and often increases the sum of radii because the penalty
        # method may keep circles slightly separated to avoid the penalty cost.
        X, Y = z[0::3], z[1::3]
        R_proj = np.zeros(n)
        for i in range(n):
            # Max radius allowed by boundaries
            r_max = min(X[i], 1.0 - X[i], Y[i], 1.0 - Y[i])
            
            # Max radius allowed by nearest neighbors
            dists = np.sqrt((X[i] - X)**2 + (Y[i] - Y)**2)
            dists[i] = np.inf  # Ignore self-distance
            r_max = min(r_max, np.min(dists) / 2.0)
            
            R_proj[i] = r_max
            
        curr_sum = np.sum(R_proj)
        if curr_sum > best_sum:
            best_sum = curr_sum
            best_centers = np.column_stack((X, Y))
            best_radii = R_proj
            
    return best_centers, best_radii, best_sum
