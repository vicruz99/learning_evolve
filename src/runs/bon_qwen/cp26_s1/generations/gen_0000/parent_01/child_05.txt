import numpy as np
import scipy.optimize as opt
import math

class BoundaryConstraint:
    def __init__(self, idx, dim, upper_bound=True):
        self.idx = idx
        self.dim = dim # 0 for x, 1 for y
        self.upper_bound = upper_bound

    def __call__(self, x):
        i = self.idx
        dim = self.dim
        val = x[3*i + dim]
        r = x[3*i + 2]
        if self.upper_bound:
            return 1.0 - val - r
        else:
            return val - r

class OverlapConstraint:
    def __init__(self, i, j):
        self.i = i
        self.j = j

    def __call__(self, x):
        xi = x[3*self.i]
        yi = x[3*self.i + 1]
        ri = x[3*self.i + 2]
        
        xj = x[3*self.j]
        yj = x[3*self.j + 1]
        rj = x[3*self.j + 2]
        
        # Use squared distance to avoid sqrt

    2. x_i <= 1 - r_i
    3. y_i >= r_i
    4. y_i <= 1 - r_i

    constraints.append(centers[:, 1] - radii)
    # 4. y <= 1 - r => 1 - y - r >= 0
    constraints.append(1.0 - centers[:, 1] - radii)
