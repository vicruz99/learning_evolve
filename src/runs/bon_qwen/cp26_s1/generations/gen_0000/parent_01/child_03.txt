import numpy as np
from scipy.optimize import minimize
import math

# Global constants and precomputed data
N_CIRCLES = 26
PAIRS = []
for i in range(N_CIRCLES):
    for j in range(i + 1, N_CIRCLES):
        PAIRS.append((i, j))

def objective(p):
    """
    Objective function to minimize: -sum(radii).
    p structure: [x0, y0, ..., x_n, y_n, r0, ..., r_n]
    """
    radii = p[2*N_CIRCLES:]
    return -np.sum(radii)

def constraint_func(p):
    """
    Constraint function.
    Returns a vector of values that must be >= 0.
    Constraints:
    1. Boundary: x >= r, 1-x >= r, y >= r, 1-y >= r
    2. Overlap: dist(i, j) >= r_i + r_j
    3. Non-negative radii: r >= 0
    """
    centers = p[:2*N_CIRCLES].reshape(N_CIRCLES, 2)
    radii = p[2*N_CIRCLES:]
    
    vals = []
    
    # Boundary and non-negative radius constraints
    for i in range(N_CIRCLES):
        # x - r >= 0
        vals.append(centers[i, 0] - radii[i])
        # 1 - x - r >= 0
        vals.append(1.0 - centers[i, 0] - radii[i])
        # y - r >= 0
        vals.append(centers[i, 1] - radii[i])
        # 1 - y - r >= 0
        vals.append(1.0 - centers[i, 1] - radii[i])
        # r >= 0
        vals.append(radii[i])
        
    # Overlap constraints
    for i, j in PAIRS:
        # dist^2 = (x_i-x_j)^2 + (y_i-y_j)^2
        dx = centers[i, 0] - centers[j, 0]
        dy = centers[i, 1] - centers[j, 1]
        dist = math.sqrt(dx*dx + dy*dy)
        sum_r = radii[i] + radii[j]
        vals.append(dist - sum_r)
        
    return np.array(vals)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Pack 26 circles in a unit square to maximize sum of radii.
    Returns (centers, radii, sum_radii).
    """
    n = N_CIRCLES
    
    # Bounds for variables
    # x, y in [0, 1]
    # r in [0, 0.5]
    bounds = []
    for _ in range(2 * n):
        bounds.append((0.0, 1.0))
    for _ in range(n):
        bounds.append((0.0, 0.5))

    # Constraint definition
    cons = {'type': 'ineq', 'fun': constraint_func}
    
    # Generate initial guesses
    guesses = []

            idx += 1
    centers_g1 = np.vstack([centers_g1, [0.5, 0.5]])
    radii_g1 = np.ones(26) * r_g1
    p0_g1 = np.hstack([centers_g1.flatten(), radii_g1])
    guesses.append(p0_g1)

    # Guess 2: Random
    np.random.seed(42)
    c_rand = np.random.rand(n, 2) * 0.6 + 0.2
    r_rand = np.ones(n) * 0.05
    p0_rand = np.hstack([c_rand.flatten(), r_rand])
    guesses.append(p0_rand)

    # Guess 3: Hexagonal Packing
    c_hex = []
    r_h = 0.09
    # Try to fit in 6 rows
    for row in range(6):
        y = r_h + row * r_h * math.sqrt(3)
        if row % 2 == 0:
            # 5 circles
            for col in range(5):
                x = r_h + col * 2 * r_h
                c_hex.append([x, y])
        else:
            # 4 circles (shifted)
            for col in range(4):
                x = 2 * r_h + col * 2 * r_h
                c_hex.append([x, y])
    # We have 27 circles, take first 26
    c_hex = np.array(c_hex[:26])
    # Shift to center of square
    c_hex -= 0.5
    c_hex += 0.5
    # Ensure bounds
    c_hex = np.clip(c_hex, 0.01, 0.99)
    r_hex = np.ones(26) * 0.09
    p0_hex = np.hstack([c_hex.flatten(), r_hex])
    guesses.append(p0_hex)

    best_p = None
    best_obj_val = -np.inf # We want to maximize sum, so minimize -sum. Lower is better for objective.
    # Wait, objective is -sum. So we want to minimize it (make it very negative).
    # So best_obj_val should be -inf initially.
    
    for idx, p0 in enumerate(guesses):
        try:
            # Run optimization
            res = minimize(objective, p0, method='SLSQP', bounds=bounds, constraints=cons,

        final_radii = guesses[0][2*n:]

    return final_centers, final_radii, sum_r
