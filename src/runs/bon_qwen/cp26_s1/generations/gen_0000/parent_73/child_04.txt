import numpy as np

def run_packing():
    np.random.seed(42)
    
    # --- 1. Initialization ---
    n = 26
    # Generate a hexagonal grid of points to serve as initial centers
    # This creates a dense packing structure
    r_init = 0.08
    centers = []
    
    # Hex grid parameters
    dx = 2 * r_init
    dy = np.sqrt(3) * r_init
    
    # Generate rows
    y = r_init
    row = 0
    while True:
        # Offset for odd rows
        x_start = 0 if row % 2 == 0 else r_init
        x = x_start
        while x <= 1:
            centers.append([x, y])
            x += dx
            if len(centers) >= n:
                break
        if len(centers) >= n:
            break
        y += dy
        row += 1
        
    # Ensure we have exactly n centers, pad or trim if necessary
    centers = np.array(centers[:n])
    
    # If we have fewer than n, generate some random ones (fallback, though hex should cover)
    while len(centers) < n:
        centers = np.vstack([centers, [np.random.rand(), np.random.rand()]])
        
    centers = centers[:n]
    radii = np.full(n, r_init)
    
    # --- 2. Optimization Loop ---
    # We will run a simulation to expand radii and resolve overlaps
    
    max_iter = 3000
    step_size = 0.01
    expand_rate = 0.0005
    
    for iteration in range(max_iter):
        # Decay step size for precision
        current_step = step_size * (1.0 - iteration / max_iter)
        
        forces = np.zeros_like(centers)
        
        # Compute repulsion forces between circles
        for i in range(n):
            for j in range(i + 1, n):
                dist_vec = centers[i] - centers[j]
                dist = np.linalg.norm(dist_vec)
                min_dist = radii[i] + radii[j]
                
                if dist < min_dist and dist > 1e-9:
                    # Overlap: push apart
                    # Force magnitude proportional to overlap
                    overlap = min_dist - dist
                    force_mag = overlap * 10.0 # Stiff spring constant
                    force_vec = (dist_vec / dist) * force_mag
                    forces[i] += force_vec
                    forces[j] -= force_vec
                elif dist < 1e-9:
                    # Very close, push in random direction
                    rand_dir = np.random.randn(2)
                    forces[i] += rand_dir
                    forces[j] -= rand_dir
        
        # Compute boundary repulsion forces
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            
            # Left boundary
            if x - r < 0:
                forces[i, 0] += abs(x - r) * 20.0
            # Right boundary
            if x + r > 1:
                forces[i, 0] -= abs(x + r - 1) * 20.0
            # Bottom boundary
            if y - r < 0:
                forces[i, 1] += abs(y - r) * 20.0
            # Top boundary
            if y + r > 1:
                forces[i, 1] -= abs(y + r - 1) * 20.0
                
        # Update centers based on forces
        centers += forces * current_step
        
        # Clamp centers to [0, 1] to prevent wild excursions, though forces should handle it
        centers = np.clip(centers, 0, 1)
        
        # Expand radii
        # We only expand if the system is relatively stable (low forces/overlaps)
        # A simple heuristic: expand slowly. The repulsion will push circles apart to accommodate.
        # If they are jammed, forces will be high and centers won't move much, but radii keep growing?
        # Actually, if radii grow and centers can't move, overlaps increase, forces increase, centers move.
        # If centers hit boundaries, they can't move, so overlaps persist.
        # To handle this, we can scale expansion based on overlap status.
        
        # Check max overlap to decide expansion
        max_overlap = 0
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.linalg.norm(centers[i] - centers[j])
                overlap = (radii[i] + radii[j]) - dist
                if overlap > max_overlap:
                    max_overlap = overlap
            
            # Boundary overlap check
            x, y = centers[i]
            r = radii[i]
            b_overlap = max(r - x, x + r - 1, r - y, y + r - 1, 0)
            if b_overlap > max_overlap:
                max_overlap = b_overlap
                
        # If overlaps are small, expand. If large, expand slower or not at all.
        # This acts as a control loop.
        if max_overlap < 0.001:
            radii += expand_rate * (1.0 + 0.5 * np.sin(iteration * 0.1)) # Slight oscillation to help unjam
        else:
            radii += expand_rate * 0.1 # Slow expansion to let forces resolve
            
        # Prevent negative radii
        radii = np.maximum(radii, 0.0)

    # --- 3. Final Validation and Cleanup ---
    # A final strict check and adjustment if necessary
    # We can run a few more iterations with very small steps and no expansion to settle
    for _ in range(500):
        forces = np.zeros_like(centers)
        for i in range(n):
            for j in range(i + 1, n):
                dist_vec = centers[i] - centers[j]
                dist = np.linalg.norm(dist_vec)
                min_dist = radii[i] + radii[j]
                if dist < min_dist and dist > 1e-9:
                    overlap = min_dist - dist
                    force_vec = (dist_vec / dist) * overlap
                    forces[i] += force_vec
                    forces[j] -= force_vec
        
        # Boundary forces
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < 0: forces[i, 0] += abs(x - r)
            if x + r > 1: forces[i, 0] -= abs(x + r - 1)
            if y - r < 0: forces[i, 1] += abs(y - r)
            if y + r > 1: forces[i, 1] -= abs(y + r - 1)
            
        centers += forces * 0.001
        centers = np.clip(centers, 0, 1)

    # Calculate sum of radii
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
