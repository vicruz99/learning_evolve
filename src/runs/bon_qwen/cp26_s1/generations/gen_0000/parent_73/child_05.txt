import numpy as np
from scipy.optimize import minimize

def objective(vars, n):
    """Objective function: maximize sum of radii => minimize negative sum."""
    return -np.sum(vars[2*n:])

def constraint_boundary(vars, n):
    """Constraints: circles must be inside [0,1]^2."""
    c = vars[:2*n].reshape(n, 2)
    r = vars[2*n:]
    # x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
    return np.concatenate([
        c[:, 0] - r,
        1.0 - c[:, 0] - r,
        c[:, 1] - r,
        1.0 - c[:, 1] - r
    ])

def constraint_overlap(vars, n):
    """Constraints: circles must not overlap. dist^2 >= (r_i + r_j)^2"""
    c = vars[:2*n].reshape(n, 2)
    r = vars[2*n:]
    
    # Vectorized pairwise distance squared
    diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
    dist_sq = np.sum(diff**2, axis=2)
    
    # Vectorized sum of radii squared
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    
    # Constraint value: dist^2 - (r_i + r_j)^2 >= 0
    vals = dist_sq - r_sum**2
    
    # Extract upper triangle to avoid duplicate and self constraints
    idx = np.triu_indices(n, k=1)
    return vals[idx]

def run_packing():
    np.random.seed(42)
    n = 26
    best_sum = -1.0
    best_sol = None
    
    # Prepare multiple initial configurations for robustness
    inits = []
    
    # Init 1: Uniform grid subset
    x1 = np.linspace(0.15, 0.85, 6)
    y1 = np.linspace(0.15, 0.85, 5)
    xx, yy = np.meshgrid(x1, y1)
    inits.append(np.vstack([xx.ravel(), yy.ravel()]).T[:n])
    
    # Init 2: Shifted grid subset
    x2 = np.linspace(0.12, 0.88, 6)
    y2 = np.linspace(0.12, 0.88, 5)
    xx, yy = np.meshgrid(x2, y2)
    pts = np.vstack([xx.ravel(), yy.ravel()]).T
    idx = np.random.choice(len(pts), n, replace=False)
    inits.append(pts[idx])
    
    # Init 3: Hexagonal-like perturbation
    pts3 = inits[0].copy()
    pts3[::2, 0] += 0.05
    inits.append(pts3)

    for centers_init in inits:
        radii_init = np.full(n, 0.04)  # Small initial radius to ensure feasibility
        x0 = np.concatenate([centers_init.ravel(), radii_init])
        
        cons = [
            {'type': 'ineq', 'fun': constraint_boundary, 'args': (n,)},
            {'type': 'ineq', 'fun': constraint_overlap, 'args': (n,)}
        ]
        bounds = [(0.0, 0.5)] * (2*n) + [(0.0, 0.5)] * n
        
        try:
            res = minimize(
                objective, 
                x0, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons, 
                args=(n,),
                options={'maxiter': 3000, 'ftol': 1e-12, 'disp': False}
            )
            
            if res.success:
                current_sum = -res.fun
                if current_sum > best_sum:
                    best_sum = current_sum
                    best_sol = res.x
        except Exception:
            continue
            
    if best_sol is not None:
        best_centers = best_sol[:2*n].reshape(n, 2)
        best_radii = best_sol[2*n:]
        return best_centers, best_radii, float(np.sum(best_radii))
    else:
        # Fallback to a simple valid packing if optimization fails unexpectedly
        centers = np.zeros((n, 2))
        radii = np.full(n, 0.01)
        for i in range(n):
            centers[i] = [(i % 5 + 0.5) / 5.0, (i // 5 + 0.5) / 5.0]
        return centers, radii, float(np.sum(radii))
