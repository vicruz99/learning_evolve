import numpy as np
import scipy.optimize as opt

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    Pack 26 circles in a unit square [0,1]x[0,1] to maximize the sum of radii.
    """

    num_vars = n * 3

    # We iterate to find a grid density that provides at least 26 points.

    dx = 0.22 
    dy = dx * np.sqrt(3) / 2
    
    # Generate grid
    # Range roughly 0 to 1
    rows = int(1 / dy) + 2
    cols = int(1 / dx) + 2

        for i in range(cols):
            x = i * dx + (j % 2) * (dx / 2)
            y = j * dy
            # Check if inside [0, 1] x [0, 1] with some margin for radius
            if 0 <= x <= 1 and 0 <= y <= 1:
                points.append([x, y])

    # If we have more, we can select a subset. A simple heuristic is to pick points
    # that are well distributed. However, just taking the first 26 from a dense scan
    # might be okay if the scan order is spatial.

    # But we might have too many or too few.
    
    if len(points) < 26:

        points = selected_points
