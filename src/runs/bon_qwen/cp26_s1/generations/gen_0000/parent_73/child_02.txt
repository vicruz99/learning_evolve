The objective function minimizes `-sum(radii) + W * penalty`, where `penalty` quantifies boundary violations and inter-circle overlaps squared. import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26
PENALTY_WEIGHT = 10000.0

def compute_penalty(vars):
    centers = vars[:2*N_CIRCLES].reshape(N_CIRCLES, 2)
    radii = vars[2*N_CIRCLES:]
    
    penalty = 0.0
    
    # Boundary constraints: circles must stay within [0,1]x[0,1]
    for i in range(N_CIRCLES):
        x, y = centers[i]
        r = radii[i]
        if x - r < 0: penalty += (x - r)**2
        if x + r > 1: penalty += (x + r - 1)**2
        if y - r < 0: penalty += (y - r)**2
        if y + r > 1: penalty += (y + r - 1)**2
        
    # Overlap constraints: distance between centers >= sum of radii
    for i in range(N_CIRCLES):
        for j in range(i + 1, N_CIRCLES):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.hypot(dx, dy)
            min_dist = radii[i] + radii[j]
            if dist < min_dist:
                penalty += (min_dist - dist)**2
                
    return penalty

def objective(vars):
    # Maximize sum of radii <=> minimize negative sum
    # Large penalty weight ensures feasibility is prioritized
    return -np.sum(vars[2*N_CIRCLES:]) + PENALTY_WEIGHT * compute_penalty(vars)

def run_packing():
    dim = 2 * N_CIRCLES
    total_vars = dim + N_CIRCLES
    # Bounds: centers in [0,1], radii in [1e-6, 0.5]
    bounds = [(0.0, 1.0)] * dim + [(1e-6, 0.5)] * N_CIRCLES
    
    best_vars = None
    best_score = float('inf')
    
    # Strategy 1: Multiple random starts to escape poor local minima
    for seed in range(40):
        np.random.seed(seed)
        x0 = np.random.rand(total_vars)
        x0[:dim] = x0[:dim] * 0.8 + 0.1  # Keep centers initially away from boundaries
        x0[dim:] = 0.08                   # Start with moderate radii
        
        res = minimize(objective, x0, method='L-BFGS-B', bounds=bounds,
                       options={'ftol': 1e-10, 'maxiter': 2000})
        
        if res.fun < best_score:
            best_score = res.fun
            best_vars = res.x.copy()
            
    # Strategy 2: Hexagonal grid initialization (high packing density heuristic)
    centers_init = []
    idx = 0
    row = 0
    while idx < N_CIRCLES:
        col = 0
        while col < 5 and idx < N_CIRCLES:
            y = 0.05 + row * 0.15
            x = 0.1 + col * 0.18 + (0.09 if row % 2 == 1 else 0.0)
            centers_init.append([x, y])
            idx += 1
            col += 1
        row += 1
        
    x0_grid = np.zeros(total_vars)
    x0_grid[:dim] = np.array(centers_init).flatten()
    x0_grid[dim:] = 0.08
    
    res_grid = minimize(objective, x0_grid, method='L-BFGS-B', bounds=bounds,
                        options={'ftol': 1e-10, 'maxiter': 3000})
    if res_grid.fun < best_score:
        best_score = res_grid.fun
        best_vars = res_grid.x.copy()
        
    # Strategy 3: Local refinement / basin hopping around the best solution found
    if best_vars is not None:
        for _ in range(30):
            x0_pert = best_vars.copy()
            x0_pert[:dim] += np.random.randn(dim) * 0.01
            x0_pert[dim:] += np.random.randn(N_CIRCLES) * 0.002
            
            res_pert = minimize(objective, x0_pert, method='L-BFGS-B', bounds=bounds,
                                options={'ftol': 1e-10, 'maxiter': 1000})
            if res_pert.fun < best_score:
                best_score = res_pert.fun
                best_vars = res_pert.x.copy()
                
    centers = best_vars[:dim].reshape(N_CIRCLES, 2)
    radii = best_vars[dim:]
    
    return centers, radii, np.sum(radii)
