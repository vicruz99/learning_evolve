import numpy as np
from scipy.optimize import minimize

def objective(vars):
    # Maximize sum of radii <=> Minimize negative sum
    # vars layout: [x1, y1, r1, x2, y2, r2, ...]
    return -np.sum(vars[2::3])

def constraint_fun(vars):
    n = 26
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    # Total constraints: 4 boundaries per circle + pairwise distance constraints
    n_constraints = 4 * n + n * (n - 1) // 2
    c = np.empty(n_constraints)
    idx = 0
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    c[idx:idx+n] = x - r; idx += n
    c[idx:idx+n] = 1.0 - x - r; idx += n
    c[idx:idx+n] = y - r; idx += n
    c[idx:idx+n] = 1.0 - y - r; idx += n
    
    # Pairwise non-overlap: (xi-xj)^2 + (yi-yj)^2 >= (ri+rj)^2
    i_idx, j_idx = np.triu_indices(n, k=1)
    dx = x[i_idx] - x[j_idx]
    dy = y[i_idx] - y[j_idx]
    dr = r[i_idx] + r[j_idx]
    c[idx:] = dx*dx + dy*dy - dr*dr
    
    return c

def run_packing():
    n = 26
    np.random.seed(42)
    
    # Initial layout: 5x5 grid + 1 center point, slightly perturbed to break symmetry
    grid_x = np.linspace(0.1, 0.9, 5)
    grid_y = np.linspace(0.1, 0.9, 5)
    centers = np.array([(x, y) for x in grid_x for y in grid_y])
    centers = np.vstack([centers, [0.5, 0.5]])
    
    # Add random jitter to avoid symmetry traps and aid convergence
    centers += np.random.uniform(-0.02, 0.02, centers.shape)
    centers = np.clip(centers, 0.05, 0.95)
    
    # Initial radii (feasible with respect to initial centers)
    radii = np.full(n, 0.06)
    
    # Flatten variables: [x1, y1, r1, x2, y2, r2, ...]
    x0 = np.concatenate([centers.ravel(), radii])
    
    # Bounds for variables
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
    
    # Constraints definition (SLSQP expects ineq: fun(vars) >= 0)
    cons = {'type': 'ineq', 'fun': constraint_fun}
    
    # Run optimization
    res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': 5000, 'ftol': 1e-12, 'disp': False})
                   
    if res.success:
        final_centers = res.x[:2*n].reshape(n, 2)
        final_radii = res.x[2*n:]
    else:
        final_centers = centers
        final_radii = radii
        
    # Ensure radii are strictly non-negative
    final_radii = np.maximum(final_radii, 0.0)
    
    return final_centers, final_radii, np.sum(final_radii)
