import numpy as np
from scipy.optimize import minimize
import math


N_CIRCLES = 26


def penalty_objective(vars, penalty_weight):
    """Objective with penalty for constraint violations."""
    n = N_CIRCLES
    obj = -np.sum(vars[2::3])
    
    for i in range(n):
        x = vars[3 * i]
        y = vars[3 * i + 1]
        r = vars[3 * i + 2]
        
        v1 = max(0, r - x)
        v2 = max(0, x + r - 1)
        v3 = max(0, r - y)
        v4 = max(0, y + r - 1)
        obj -= penalty_weight * (v1 * v1 + v2 * v2 + v3 * v3 + v4 * v4)
    
    for i in range(n):
        for j in range(i + 1, n):
            dx = vars[3 * i] - vars[3 * j]
            dy = vars[3 * i + 1] - vars[3 * j + 1]
            dist = math.sqrt(dx * dx + dy * dy)
            r_sum = vars[3 * i + 2] + vars[3 * j + 2]
            overlap = max(0, r_sum - dist)
            obj -= penalty_weight * overlap * overlap
    
    return obj


def hexagonal_initial_guess(r_base=0.09):
    """Create hexagonal packing initial guess with rows 6+5+6+5+4=26."""
    centers = []
    radii = []
    rows = [6, 5, 6, 5, 4]
    
    y = r_base
    for row_idx, count in enumerate(rows):
        if count == 0:
            continue
        if row_idx % 2 == 0:
            x_positions = np.linspace(r_base, 1 - r_base, count)
        else:
            x_positions = np.linspace(r_base + (1 - 2 * r_base) / (count + 1),
                                      1 - r_base - (1 - 2 * r_base) / (count + 1), count)
        for x in x_positions:
            centers.append([x, y])
            radii.append(r_base)
        y += r_base * math.sqrt(3)
    
    return np.array(centers), np.array(radii)


def perturbed_hexagonal_guess(r_base=0.09, seed=42):
    """Create perturbed hexagonal packing."""
    np.random.seed(seed)
    centers, radii = hexagonal_initial_guess(r_base)
    centers = centers + np.random.normal(0, 0.005, centers.shape)
    centers = np.clip(centers, 0.01, 0.99)
    radii = radii + np.random.normal(0, 0.003, len(radii))
    radii = np.clip(radii, 0.01, 0.3)
    return centers, radii


def to_flat(centers, radii):
    """Convert centers and radii to flat optimization vector."""
    n = len(radii)
    result = np.zeros(3 * n)
    for i in range(n):
        result[3 * i] = centers[i, 0]
        result[3 * i + 1] = centers[i, 1]
        result[3 * i + 2] = radii[i]
    return result


def from_flat(vars):
    """Convert flat optimization vector to centers and radii."""
    n = len(vars) // 3
    centers = np.array([[vars[3 * i], vars[3 * i + 1]] for i in range(n)])
    radii = vars[2::3]
    return centers, radii


def get_bounds(n):
    """Get variable bounds for optimization."""
    bounds = []
    for i in range(n):
        bounds.append((0.001, 0.999))
        bounds.append((0.001, 0.999))
        bounds.append((1e-6, 0.49))
    return bounds


def optimize_from_initial(centers, radii, max_penalty_iterations=15):
    """Optimize packing starting from given initial configuration."""
    n = N_CIRCLES
    vars = to_flat(centers, radii)
    bounds = get_bounds(n)
    
    penalty = 500
    for iteration in range(max_penalty_iterations):
        current_penalty = penalty
        
        def objective(v, p=current_penalty):
            return penalty_objective(v, p)
        
        result = minimize(objective, vars, method='L-BFGS-B', bounds=bounds,
                         options={'maxiter': 2000, 'ftol': 1e-18, 'gtol': 1e-12})
        vars = result.x
        penalty = min(penalty * 3, 500000)
    
    return from_flat(vars)


def check_validity(centers, radii):
    """Check if packing is valid."""
    n = len(radii)
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-10 or x + r > 1 + 1e-10 or y - r < -1e-10 or y + r > 1 + 1e-10:
            return False
    
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = math.sqrt(dx * dx + dy * dy)
            if dist < radii[i] + radii[j] - 1e-10:
                return False
    
    return True


def refine_with_sqp(centers, radii):
    """Refine using SLSQP with explicit constraints."""
    n = N_CIRCLES
    vars = to_flat(centers, radii)
    
    def obj(v):
        return -np.sum(v[2::3])
    
    def con_func(v):
        cons = []
        for i in range(n):
            x, y, r = v[3 * i], v[3 * i + 1], v[3 * i + 2]
            cons.append(x - r)
            cons.append(1 - x - r)
            cons.append(y - r)
            cons.append(1 - y - r)
        
        for i in range(n):
            for j in range(i + 1, n):
                dx = v[3 * i] - v[3 * j]
                dy = v[3 * i + 1] - v[3 * j + 1]
                dist_sq = dx * dx + dy * dy
                r_sum = v[3 * i + 2] + v[3 * j + 2]
                cons.append(dist_sq - r_sum * r_sum)
        return np.array(cons)
    
    constraint = {'type': 'ineq', 'fun': con_func}
    bounds = get_bounds(n)
    
    result = minimize(obj, vars, method='SLSQP', constraints=[constraint],
                     bounds=bounds, options={'maxiter': 3000, 'ftol': 1e-15})
    
    return from_flat(result.x)
