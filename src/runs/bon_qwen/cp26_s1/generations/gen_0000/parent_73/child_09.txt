import numpy as np
import scipy.optimize as opt

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # 1. Initialization: Hexagonal Lattice Pattern
    # We arrange 26 circles in 5 rows: 6, 5, 6, 5, 4
    rows_counts = [6, 5, 6, 5, 4]
    centers = []
    radii = []
    
    # Hexagonal spacing parameters
    # If radius is r, horizontal step is 2r, vertical step is r*sqrt(3)
    # We'll build the geometry first and scale later
    r_unit = 1.0
    dx = 2 * r_unit
    dy = r_unit * np.sqrt(3)
    
    current_y = r_unit
    row_idx = 0
    
    for count in rows_counts:
        # Center the row horizontally
        row_width = (count - 1) * dx
        start_x = (1 - row_width) / 2
        
        # For odd rows (1, 3, ...), shift horizontally by dx/2 (r_unit)
        if row_idx % 2 == 1:
            start_x += r_unit
            
        for i in range(count):
            centers.append([start_x + i * dx, current_y])
            radii.append(r_unit)
            
        current_y += dy
        row_idx += 1
        
    centers = np.array(centers)
    radii = np.array(radii)
    
    # 2. Scale to fit in unit square
    # Find max extent and scale down
    max_x = np.max(centers[:, 0]) + np.max(radii)
    max_y = np.max(centers[:, 1]) + np.max(radii)
    scale = 1.0 / max(max_x, max_y)
    
    centers *= scale
    radii *= scale
    
    # 3. Optimization
    # We will maximize sum(radii) subject to non-overlap and boundary constraints.
    # We run a few iterations of optimization with perturbations to escape local minima.
    
    best_centers = centers.copy()
    best_radii = radii.copy()
    best_sum = np.sum(best_radii)
    
    for _ in range(10): # Multiple restarts/perturbations
        # Add slight noise to current best to explore neighborhood
        if _ > 0:
            noise = np.random.normal(0, 0.01, best_centers.shape)
            centers_opt = best_centers + noise
            radii_opt = best_radii * (1 + np.random.normal(0, 0.02, n))
        else:
            centers_opt = best_centers.copy()
            radii_opt = best_radii.copy()

        # Vectorize variables for optimization
        # Variables: [x1, y1, r1, x2, y2, r2, ...]
        # Total 3*n variables
        x0 = np.zeros(3 * n)
        for i in range(n):
            x0[3*i] = centers_opt[i, 0]
            x0[3*i+1] = centers_opt[i, 1]
            x0[3*i+2] = radii_opt[i]
            
        # Bounds for each variable: x in [0,1], y in [0,1], r in [0, 0.5]
        bounds = [(0, 1)] * n + [(0, 1)] * n + [(0, 0.5)] * n
        
        def objective(vars_vec):
            # We want to maximize sum of radii, so minimize negative sum
            return -np.sum(vars_vec[2::3])
            
        def constraints(vars_vec):
            cons = []
            x = vars_vec[0::3]
            y = vars_vec[1::3]
            r = vars_vec[2::3]
            
            # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
            # Which implies x - r >= 0, 1 - x - r >= 0, etc.
            # We can just add linear constraints or check in loop. 
            # SLSQP supports linear constraints, but let's stick to nonlinear for generality or just simple bounds.
            # Actually, simple bounds are handled by 'bounds' argument. 
            # But we need r <= x <= 1-r. 
            # Bounds [0,1] for x is not enough if r is large.
            # Let's rely on nonlinear constraints for boundary.
            
            # Circle containment: r <= x <= 1-r  =>  r - x <= 0  and  x + r - 1 <= 0
            for i in range(n):
                cons.append(r[i] - x[i]) # x >= r
                cons.append(x[i] + r[i] - 1) # x <= 1-r
                cons.append(r[i] - y[i]) # y >= r
                cons.append(y[i] + r[i] - 1) # y <= 1-r
                
            # Non-overlap: dist^2 >= (r_i + r_j)^2
            # dist^2 - (r_i + r_j)^2 >= 0
            for i in range(n):
                for j in range(i + 1, n):
                    dx_ij = x[i] - x[j]
                    dy_ij = y[i] - y[j]
                    dist_sq = dx_ij**2 + dy_ij**2
                    r_sum_sq = (r[i] + r[j])**2
                    cons.append(dist_sq - r_sum_sq)
            
            return cons

        # Using SLSQP
        try:
            res = opt.minimize(objective, x0, method='SLSQP', bounds=bounds, 
                               constraints={'type': 'ineq', 'fun': constraints},
                               options={'maxiter': 500, 'ftol': 1e-9})
            
            if res.success or res.fun < -best_sum:
                new_sum = -res.fun
                if new_sum > best_sum:
                    best_sum = new_sum
                    best_centers = np.column_stack((res.x[0::3], res.x[1::3]))
                    best_radii = res.x[2::3]
        except Exception:
            continue

    # Final validation check and return
    return best_centers, best_radii, best_sum
