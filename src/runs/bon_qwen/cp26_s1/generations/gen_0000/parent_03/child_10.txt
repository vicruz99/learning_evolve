import scipy.optimize as opt

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    Solves the circle packing problem for 26 circles in a unit square.
    Returns (centers, radii, sum_radii).
    """
    n = 26
    best_sum = -1.0
    best_centers = None
    best_radii = None

    # Helper to calculate constraints and objective for optimization
    def problem_func(x):
        # x contains 26 radii followed by 52 coordinates (x1, y1, x2, y2, ...)
        # Order: [r0, r1, ..., r25, x0, y0, x1, y1, ..., x25, y25]
        
        radii = x[:n]
        coords = x[n:].reshape(n, 2)
        
        # Objective: Minimize negative sum of radii
        return -np.sum(radii)

    def constraints_func(x):
        radii = x[:n]
        coords = x[n:].reshape(n, 2)
        c_list = []
        
        # 1. Boundary constraints: r <= x, r <= 1-x, r <= y, r <= 1-y
        # Equivalent to: x - r >= 0, 1 - x - r >= 0, etc.
        for i in range(n):
            c_list.append(coords[i, 0] - radii[i])       # x - r >= 0
            c_list.append(1.0 - coords[i, 0] - radii[i]) # 1 - x - r >= 0
            c_list.append(coords[i, 1] - radii[i])       # y - r >= 0
            c_list.append(1.0 - coords[i, 1] - radii[i]) # 1 - y - r >= 0

        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((coords[i] - coords[j]) ** 2))
                c_list.append(dist - radii[i] - radii[j])

    return best_centers, best_radii, best_sum
