import numpy as np
from scipy.optimize import linprog, minimize

def generate_hex_lattice(n, w=1.0, h=1.0):
    """Generates a hexagonal lattice of n points."""
    points = []
    rows = int(np.ceil(np.sqrt(n * 2 / np.sqrt(3))))
    for i in range(rows):
        y = (i + 1) * h / (rows + 1)
        x_offset = (i % 2) * 0.5
        cols = int(np.ceil(np.sqrt(n / 1.5)))
        for j in range(cols):
            x = (j + 1 + x_offset) * w / (cols + 1)
            if x <= 1.0 and y <= 1.0:
                points.append([x, y])
            if len(points) >= n:
                break
        if len(points) >= n:
            break
    return np.array(points[:n])

def generate_grid_perturbed(n, w=1.0, h=1.0):
    """Generates a grid-like layout and perturbs it."""
    cols = int(np.ceil(np.sqrt(n)))
    rows = int(np.ceil(n / cols))

            if len(points) < n:

                points.append([x, y])
    pts = np.array(points)
    pts += np.random.uniform(-0.02, 0.02, pts.shape)
    pts = np.clip(pts, 0.0, 1.0)
    return pts

def compute_max_radii(centers):

    A_ub = []
    b_ub = []

        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1
            row[j] = 1
            A_ub.append(row)
            b_ub.append(dists[i, j])

        r_max = min(centers[i, 0], 1 - centers[i, 0],

        A_ub.append(row)
        b_ub.append(r_max)

    if res.success:
        return -res.fun, -res.x
    return 0.0, np.zeros(n)

def validate_and_adjust(centers, radii):
    """Ensures strict validity by shrinking radii slightly if needed."""
    n = centers.shape[0]
    adjusted_radii = radii.copy()
    
    # Boundary adjustment
    for i in range(n):
        min_dist_to_edge = min(centers[i, 0], 1 - centers[i, 0], 
                               centers[i, 1], 1 - centers[i, 1])
        if adjusted_radii[i] > min_dist_to_edge - 1e-7:
            adjusted_radii[i] = max(0, min_dist_to_edge - 1e-7)
            
    # Overlap adjustment
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(centers[i] - centers[j])

                # Reduce both radii by half the overlap
                adjusted_radii[i] -= overlap / 2
                adjusted_radii[j] -= overlap / 2
                
    return adjusted_radii
