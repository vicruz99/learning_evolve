import numpy as np
from scipy.optimize import minimize

def compute_loss_and_grad(vars, n, penalty_weight):
    """
    Computes the objective function value and its gradient for the packing problem.
    Objective: Minimize -sum(radii) + penalty_weight * sum(violations^2)
    """
    centers = vars[:n * 2].reshape(n, 2)
    radii = vars[n * 2:]

    # Objective: negative sum of radii
    loss = -np.sum(radii)

    # Gradient initialization
    grad = np.zeros_like(vars)
    grad[n * 2:] = -1.0  # d(loss)/d(radii) from -sum(radii)

    # Helper to calculate smooth max(x)^2 and its derivative
    def smooth_penalty(val):
        if val > 0:
            return val**2, 2 * val
        return 0.0, 0.0

    # 1. Boundary Constraints
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        
        # Constraints: r <= x, r <= 1-x, r <= y, r <= 1-y
        # Violations: r - x > 0, etc.
        boundaries = [r - x, r - (1 - x), r - y, r - (1 - y)]
        
        for b in boundaries:
            p, gp = smooth_penalty(b)
            loss += penalty_weight * p
            # d(loss)/d(r) += weight * 2b
            grad[n * 2 + i] += penalty_weight * gp
            # d(loss)/d(x) or d(y)
            if b == r - x:
                grad[2 * i] -= penalty_weight * gp
            elif b == r - (1 - x):
                grad[2 * i] += penalty_weight * gp
            elif b == r - y:
                grad[2 * i + 1] -= penalty_weight * gp
            elif b == r - (1 - y):
                grad[2 * i + 1] += penalty_weight * gp

    # 2. Overlap Constraints
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist_sq = dx**2 + dy**2
            dist = np.sqrt(dist_sq + 1e-12) # Small epsilon to avoid div by zero
            
            r_sum = radii[i] + radii[j]
            violation = r_sum - dist
            
            p, gp = smooth_penalty(violation)
            loss += penalty_weight * p
            
            # Gradients for radii
            grad[n * 2 + i] += penalty_weight * gp
            grad[n * 2 + j] += penalty_weight * gp
            
            # Gradients for centers (chain rule through dist)
            # d(violation)/d(x_i) = -d(dist)/d(x_i) = -(dx / dist)
            # d(loss)/d(x_i) = weight * gp * (-dx/dist)
            factor = -penalty_weight * gp / dist
            grad[2 * i] += factor * dx
            grad[2 * j] -= factor * dx
            grad[2 * i + 1] += factor * dy
            grad[2 * j + 1] -= factor * dy

    return loss, grad

def run_packing():
    n = 26
    
    # We will run multiple restarts to find a good global configuration
    best_loss = np.inf
    best_vars = None
    
    num_restarts = 10
    penalty_weight = 1000.0
    
    for _ in range(num_restarts):
        # 1. Initialize Centers (Hexagonal Lattice)
        centers_init = []
        # Hex grid parameters
        s = 0.18  # Spacing
        x_off = s / 2
        y_off = s * np.sqrt(3) / 2
        
        y = 0
        while y < 1.0:
            x = x_off if int(y / y_off) % 2 == 1 else 0
            while x < 1.0:
                centers_init.append([x, y])
                x += s
            y += y_off
            
        centers_init = np.array(centers_init)
        
        # Shuffle and pick 26, or just take first 26 if we have enough
        np.random.shuffle(centers_init)
        if len(centers_init) < n:
            # Fallback: random placement if grid too sparse
            centers_init = np.random.uniform(0, 1, (n, 2))
        else:
            centers_init = centers_init[:n]
            
        # 2. Initialize Radii
        radii_init = np.full(n, 0.05)
        
        # 3. Flatten to 1D array for optimizer
        vars_init = np.concatenate([centers_init.flatten(), radii_init])
        
        # 4. Optimization
        bounds = [(0, 1)] * (n * 2) + [(0, 0.5)] * n
        
        res = minimize(
            compute_loss_and_grad,
            vars_init,
            args=(n, penalty_weight),
            method='L-BFGS-B',
            jac=True,
            bounds=bounds,
            options={'maxiter': 2000, 'ftol': 1e-12}
        )
        
        # Check if this solution is better
        if res.fun < best_loss:
            # Check if it's actually a valid packing (penalty should be near zero)
            # We check the sum of radii directly from the result
            curr_radii = res.x[n * 2:]
            curr_sum = np.sum(curr_radii)
            if curr_sum > 2.0: # Basic sanity check to avoid stuck local minima with tiny radii
                best_loss = res.fun
                best_vars = res.x

    # 5. Extract best solution
    if best_vars is not None:
        centers = best_vars[:n * 2].reshape(n, 2)
        radii = best_vars[n * 2:]
        
        # Final validation pass to ensure no NaN or negative radii
        radii = np.maximum(radii, 0)
        sum_radii = np.sum(radii)
        return centers, radii, sum_radii
    
    # Fallback if optimization failed completely
    centers = np.random.uniform(0.1, 0.9, (n, 2))
    radii = np.full(n, 0.01)
    return centers, radii, np.sum(radii)
