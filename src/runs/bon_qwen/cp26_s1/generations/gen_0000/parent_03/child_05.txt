def repulsion_force(centers, radii, walls=True):
    """
    Calculates repulsive forces to resolve overlaps.
    Returns forces to add to centers.
    """
    n = centers.shape[0]
    forces = np.zeros_like(centers)
    
    # Pairwise repulsion
    # Vectorized calculation for performance
    # diff[i, j] = centers[i] - centers[j]
    # To save memory and time, we can iterate or use broadcasting if n is small
    # n=26 is small enough for O(n^2)
    
    for i in range(n):
        for j in range(i + 1, n):
            diff = centers[i] - centers[j]
            dist = np.linalg.norm(diff)
            min_dist = radii[i] + radii[j]
            
            if dist < min_dist and dist > 1e-9:
                # Overlap amount
                overlap = min_dist - dist
                # Force direction: away from each other
                # Normalize diff vector
                direction = diff / dist
                # Apply force proportional to overlap
                # We split the movement between the two circles
                force_mag = overlap 
                forces[i] += direction * force_mag * 0.5
                forces[j] -= direction * force_mag * 0.5
            elif dist <= 1e-9:
                # Prevent division by zero, push randomly
                random_vec = np.random.randn(2)
                forces[i] += random_vec
                forces[j] -= random_vec

    if walls:
        for i in range(n):
            r = radii[i]
            x, y = centers[i]
            # Wall repulsion
            if x - r < 0:
                forces[i, 0] += (r - x) * 2.0 # Push right
            if x + r > 1:
                forces[i, 0] -= (x + r - 1) * 2.0 # Push left

                forces[i, 1] += (r - y) * 2.0 # Push up
            if y + r > 1:
                forces[i, 1] -= (y + r - 1) * 2.0 # Push down
                
    return forces
