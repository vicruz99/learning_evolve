import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def compute_constraints(x):
    """
    Computes inequality constraints. Returns an array where all elements must be >= 0.
    """
    n = N_CIRCLES
    centers = x[:n*2].reshape(n, 2)
    radii = x[n*2:]

    # Boundary constraints: center - radius >= 0, 1 - center - radius >= 0
    c_min = centers - radii[:, None]
    c_max = np.ones((n, 2)) - centers - radii[:, None]
    boundaries = np.concatenate([c_min.ravel(), c_max.ravel()])

    # Overlap constraints: dist(i,j) - (r[i] + r[j]) >= 0
    # Vectorized pairwise distance calculation
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    # Extract upper triangle indices for unique pairs (i < j)
    idx = np.triu_indices(n, k=1)
    dists_pairs = dists[idx]
    r_sum_pairs = radii[idx[0]] + radii[idx[1]]
    overlaps = dists_pairs - r_sum_pairs

    return np.concatenate([boundaries, overlaps])

def compute_objective(x):
    """
    Objective function to minimize: negative sum of radii.
    """
    n = N_CIRCLES
    radii = x[n*2:]
    return -np.sum(radii)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = N_CIRCLES
    
    # Initialize with a 5x5 grid + 1 center circle.
    # This layout provides a very strong baseline (~2.82 sum of radii)
    # and naturally guides the optimizer towards a high-density configuration.
    pts = []
    for i in range(5):
        for j in range(5):
            pts.append([0.2 + i * 0.2, 0.2 + j * 0.2])
    pts.append([0.5, 0.5])
    centers = np.array(pts)

    # Add small deterministic perturbation to break symmetry
    # This prevents the optimizer from getting stuck in symmetric local minima
    centers[0, 0] += 0.002
    centers[1, 1] -= 0.001
    centers[-1, 0] += 0.001

    radii = np.ones(n) * 0.02
    x0 = np.concatenate([centers.ravel(), radii])

    # Bounds: centers in [0,1], radii in [epsilon, 0.5]
    bounds = [(0.0, 1.0)] * (n * 2) + [(1e-4, 0.5)] * n
    
    cons = {'type': 'ineq', 'fun': compute_constraints}

    # SLSQP handles the nonlinear constraints efficiently
    res = minimize(compute_objective, x0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'ftol': 1e-9, 'maxiter': 4000})

    final_centers = res.x[:n*2].reshape(n, 2)
    final_radii = res.x[n*2:]

    # Post-processing to ensure strict validity within numerical tolerance
    final_radii = np.maximum(final_radii, 0.0)
    final_centers = np.clip(final_centers, final_radii[:, None], 1.0 - final_radii[:, None])

    return final_centers, final_radii, float(np.sum(final_radii))
