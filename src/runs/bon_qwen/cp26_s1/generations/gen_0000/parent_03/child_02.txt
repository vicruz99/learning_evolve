import numpy as np
from scipy.optimize import minimize

def compute_loss_and_grad(params, n, mu):
    """
    Computes the objective function and its analytical gradient.
    Objective: -sum(radii) + mu * penalty(overlaps, boundary_violations)
    """
    centers = params[:2*n].reshape(n, 2)
    radii = params[2*n:]
    
    # Base objective: minimize negative sum of radii
    loss = -np.sum(radii)
    grad = np.zeros_like(params)
    grad[2*n:] = -1.0
    
    # Pairwise distances and overlaps
    diff = centers[:, None, :] - centers[None, :, :]          # Shape (n, n, 2)
    dist = np.sqrt(np.sum(diff**2, axis=2))                   # Shape (n, n)
    np.fill_diagonal(dist, 1.0)                               # Avoid division by zero on diagonal
    
    overlap = radii[:, None] + radii[None, :] - dist          # Shape (n, n)
    mask = overlap > 0
    active_overlap = np.where(mask, overlap, 0.0)
    
    loss += mu * np.sum(active_overlap**2)
    
    # Gradient contribution from pairwise overlaps
    force_mag = 2 * mu * active_overlap / np.maximum(dist, 1e-9)
    grad_centers = np.einsum('ij,ijk->ik', force_mag, diff)
    grad[:2*n].reshape(n, 2) += grad_centers
    
    grad[2*n:] += np.sum(2 * mu * active_overlap, axis=1)
    
    # Boundary violations
    vx1 = radii - centers[:, 0]          # x < r
    vx2 = centers[:, 0] + radii - 1.0    # x + r > 1
    vy1 = radii - centers[:, 1]          # y < r
    vy2 = centers[:, 1] + radii - 1.0    # y + r > 1
    
    bx1 = np.where(vx1 > 0, vx1, 0.0)
    bx2 = np.where(vx2 > 0, vx2, 0.0)
    by1 = np.where(vy1 > 0, vy1, 0.0)
    by2 = np.where(vy2 > 0, vy2, 0.0)
    
    loss += mu * np.sum(bx1**2 + bx2**2 + by1**2 + by2**2)
    
    # Gradient contribution from boundaries
    grad[2*n:] += 2 * mu * (bx1 + bx2 + by1 + by2)
    grad[:2*n:2] += 2 * mu * (-bx1 + bx2)  # x coordinates
    grad[1:2*n:2] += 2 * mu * (-by1 + by2) # y coordinates
    
    return loss, grad

def init_config(n=26):
    """Generates a hexagonal grid initialization scaled to fit in [0,1]^2."""
    pts = []
    s = 0.14
    for i in range(8):
        for j in range(8):
            x = j * s + (i % 2) * s / 2
            y = i * s * np.sqrt(3) / 2
            pts.append([x, y])
    np.random.shuffle(pts)
    pts = np.array(pts[:n])
    
    margin = 0.05
    pts = pts * (1 - 2*margin) + margin
    radii = np.ones(n) * 0.04
    return np.concatenate([pts.flatten(), radii])

def run_packing():

    n = 26
    mu = 50000.0
    
    best_loss = np.inf
    best_params = None

    for trial in range(12):
        params0 = init_config(n)
        params0 += np.random.randn(len(params0)) * 0.005

        try:
            res = minimize(compute_loss_and_grad, params0, args=(n, mu),
                           method='L-BFGS-B', bounds=bounds, jac=True,
                           options={'maxiter': 2000, 'ftol': 1e-13})
            if res.fun < best_loss:
                best_loss = res.fun
                best_params = res.x.copy()
        except Exception:
            continue
            
    if best_params is None:
        # Fallback to a valid sparse packing if optimization completely fails
        centers = init_config(n)[:2*n].reshape(n, 2)
        radii = np.ones(n) * 0.02
        return centers, radii, np.sum(radii)

    dist = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dist, 1.0)

    max_ov = np.max(np.maximum(overlap, 0.0))

        np.max(radii - centers[:, 0]),
        np.max(centers[:, 0] + radii - 1.0),

    )

    if shrink > 0:
        radii -= shrink
        radii = np.maximum(radii, 1e-6)
        
    sum_r = np.sum(radii)
    return centers, radii, sum_r
