def solve_lp_radii(centers):
    """
    Given centers, solve the LP to maximize sum of radii.
    Constraints:
    1. r_i + r_j <= dist(centers[i], centers[j]) for all i < j
    2. r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    3. r_i >= 0
    
    Variables: r_0, ..., r_25
    """
    n = centers.shape[0]
    
    # Objective: maximize sum(r) -> minimize -sum(r)
    c_obj = -np.ones(n)
    
    # Constraints: A_ub * r <= b_ub
    # We will construct A_ub and b_ub
    
    # Number of constraints:
    # Pairwise: n*(n-1)/2
    # Boundary: 4 * n
    
    n_pairs = n * (n - 1) // 2
    n_boundary = 4 * n
    n_constraints = n_pairs + n_boundary
    
    A_ub = np.zeros((n_constraints, n))
    b_ub = np.zeros(n_constraints)
    
    idx = 0
    # Pairwise constraints: r_i + r_j <= dist
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dist
            idx += 1
            
    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        # r_i <= x
        A_ub[idx, i] = 1.0
        b_ub[idx] = x
        idx += 1
        # r_i <= 1-x
        A_ub[idx, i] = 1.0
        b_ub[idx] = 1 - x
        idx += 1
        # r_i <= y
        A_ub[idx, i] = 1.0
        b_ub[idx] = y
        idx += 1
        # r_i <= 1-y
        A_ub[idx, i] = 1.0
        b_ub[idx] = 1 - y
        idx += 1

    # Bounds for r: [0, infinity)
    bounds = [(0, None)] * n
    
    # Solve LP
    # Method 'highs' is robust and fast
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        radii = res.x

    else:
        # Fallback or error handling
        # Return zeros if LP fails
        return np.zeros(n), 0.0
