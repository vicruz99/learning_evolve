import numpy as np
from scipy.optimize import linprog, minimize

def solve_radii_for_centers(centers):
    """Solve LP to find maximum sum of radii for fixed centers."""
    n = centers.shape[0]
    
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    b_bnd = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                       np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    b_bnd = np.maximum(b_bnd, 0.0)
    
    # Pairwise distance constraints: r_i + r_j <= dist_ij
    n_pairs = n * (n - 1) // 2
    A_ub = np.zeros((n_pairs + n, n))
    b_ub = np.zeros(n_pairs + n)
    
    idx = 0
    for i in range(n):
        xi, yi = centers[i]
        for j in range(i + 1, n):
            xj, yj = centers[j]
            dist = np.sqrt((xi - xj)**2 + (yi - yj)**2)
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dist
            idx += 1
            
    for i in range(n):
        A_ub[idx, i] = 1.0
        b_ub[idx] = b_bnd[i]
        idx += 1
        
    # Objective: maximize sum(r) -> minimize -sum(r)
    c_obj = -np.ones(n)
    bounds = [(0.0, None)] * n
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return -res.fun, res.x
    else:
        return 0.0, np.zeros(n)

def objective_neg_sum_radii(x_flat):
    """Objective function for the optimizer: negative of max sum of radii."""
    centers = x_flat.reshape(-1, 2)
    # Keep centers strictly inside to avoid degenerate LP bases
    centers = np.clip(centers, 1e-6, 1.0 - 1e-6)
    val, _ = solve_radii_for_centers(centers)
    return -val

def run_packing():
    np.random.seed(42)
    n = 26
    
    # Initialize with a hexagonal lattice pattern
    init_pos = []
    step_x = 0.17
    step_y = step_x * np.sqrt(3) / 2.0
    
    for i in range(7):
        for j in range(5):
            if len(init_pos) >= 26:
                break
            x = 0.06 + i * step_x
            y = 0.06 + j * step_y + (0.5 * step_y if i % 2 == 1 else 0.0)
            if 0.0 <= x <= 1.0 and 0.0 <= y <= 1.0:
                init_pos.append([x, y])
                
    while len(init_pos) < 26:
        init_pos.append([np.random.uniform(0.15, 0.85), np.random.uniform(0.15, 0.85)])
        
    init_centers = np.array(init_pos[:26])
    # Shuffle to avoid symmetric trapping
    np.random.shuffle(init_centers)
    
    # Optimize centers
    res_opt = minimize(objective_neg_sum_radii, init_centers.flatten(), method='Powell',
                       options={'maxiter': 2500, 'ftol': 1e-12, 'xtol': 1e-12})
    
    best_centers = res_opt.x.reshape((n, 2))
    best_centers = np.clip(best_centers, 0.0, 1.0)
    
    # Compute final optimal radii for the optimized centers
    _, best_radii = solve_radii_for_centers(best_centers)
    best_radii = np.maximum(best_radii, 0.0)
    
    sum_r = np.sum(best_radii)
    return best_centers, best_radii, sum_r
