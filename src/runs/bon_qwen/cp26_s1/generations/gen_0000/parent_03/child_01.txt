import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Solves the circle packing problem for 26 circles in a unit square.
    Returns the best valid (centers, radii, sum_radii) found.
    """
    N = 26
    best_solution = None
    best_score = -np.inf

    # Vectorized Constraint Function for SLSQP
    # Returns a vector where each element must be >= 0
    def constraint_function(vars):
        # Reshape 1D array to (N, 3) -> [x, y, r]
        centers = vars.reshape(N, 3)[:, :2]
        radii = vars.reshape(N, 3)[:, 2]
        
        constraints = []

        # 1. Boundary Constraints (x >= r, x <= 1-r, y >= r, y <= 1-r)
        # Converted to: x - r >= 0, 1 - x - r >= 0, etc.
        constraints.append(centers[:, 0] - radii)
        constraints.append(1.0 - centers[:, 0] - radii)
        constraints.append(centers[:, 1] - radii)
        constraints.append(1.0 - centers[:, 1] - radii)

        # 2. Non-overlap Constraints (dist^2 >= (r1+r2)^2)
        # Compute pairwise squared distances efficiently
        # Shape: (N, N)
        dx = centers[:, 0][:, None] - centers[:, 0][None, :]
        dy = centers[:, 1][:, None] - centers[:, 1][None, :]
        dist_sq = dx**2 + dy**2

        # Compute squared sum of radii
        r_sum = radii[:, None] + radii[None, :]
        r_sum_sq = r_sum**2

        # Extract upper triangle (i < j) to avoid duplicates and self-checks
        mask = np.triu(np.ones((N, N), dtype=bool), k=1)
        overlap_constraints = dist_sq[mask] - r_sum_sq[mask]
        constraints.append(overlap_constraints)

        return np.concatenate(constraints)

    def objective(vars):
        # Maximize sum of radii => Minimize negative sum
        return -np.sum(vars.reshape(N, 3)[:, 2])

    # Define constraint structure for scipy
    cons = {'type': 'ineq', 'fun': constraint_function}
    
    # Bounds for variables: x, y in [0, 1], r in [0, 1]
    # Optimization will tighten these via constraints
    bounds = [(0, 1)] * (3 * N)

    # --- Generate Initial Guesses ---
    initial_guesses = []

    # Guess 1: Perturbed 5x5 Grid (25 circles) + 1 extra
    # A 5x5 grid is optimal for 25. We add a 26th and let optimizer rearrange.
    g1 = []
    for i in range(5):
        for j in range(5):
            g1.append([0.1 + 0.2 * i, 0.1 + 0.2 * j, 0.09])
    # Add 26th circle in a gap or center
    g1.append([0.5, 0.5, 0.05]) 
    initial_guesses.append(g1)

    # Guess 2: Hexagonal Lattice Approximation
    # Rows of 6, 5, 6, 5, 4 circles
    # Vertical spacing ~ sqrt(3)/2 * 2r
    g2 = []
    r_est = 0.08
    rows = [6, 5, 6, 5, 4]
    y_pos = 0.1
    for r in rows:
        x_start = 0.1
        # Spacing between centers
        spacing = 0.15
        for k in range(r):
            x_pos = x_start + k * spacing
            # Shift every other row for hex pattern
            if len(g2) > 0: # approximate check
                 pass 
            
            # Just place them, optimization will fix positions
            g2.append([x_pos, y_pos, r_est])
        y_pos += spacing * 0.8 # Rough vertical step
    # Pad or trim to 26 if needed
    while len(g2) < N:
        g2.append([0.5, 0.5, r_est])
    g2 = g2[:N]
    initial_guesses.append(g2)

    # Guess 3: Random Cloud
    for _ in range(3): # Try a few random seeds
        rng = np.random.default_rng(_)
        centers = rng.uniform(0.1, 0.9, size=(N, 2))
        radii = np.ones(N) * 0.05
        guess = []
        for i in range(N):
            guess.append([centers[i, 0], centers[i, 1], radii[i]])
        initial_guesses.append(guess)

    # --- Optimization Loop ---
    for guess in initial_guesses:
        x0 = np.array(guess).flatten()
        
        try:
            res = minimize(
                objective, 
                x0, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 2000, 'ftol': 1e-10}
            )
            
            if res.success:
                current_score = -res.fun
                if current_score > best_score:
                    best_score = current_score
                    # Reconstruct solution
                    sol_vars = res.x.reshape(N, 3)
                    best_solution = (sol_vars[:, :2], sol_vars[:, 2])
        except Exception:
            continue

    # --- Final Validation and Adjustment ---
    if best_solution is None:
        # Fallback to a trivial valid solution if optimization fails
        # 26 small circles
        centers = np.linspace(0.1, 0.9, 26).reshape(-1, 1) # 1D line
        centers = np.hstack([centers, centers]) # Fill Y
        radii = np.ones(26) * 0.01
        best_solution = (centers, radii)

    centers, radii = best_solution
    
    # Safety shrink: If optimization hit the boundary exactly, numerical errors might trigger overlap check.
    # We shrink radii by a tiny epsilon to be safe, but the validator allows 1e-12.
    # SLSQP usually respects constraints well. 
    # We can try to verify and repair if needed.
    
    # Verify and adjust
    valid = True
    # Simple check logic inline
    for i in range(N):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            valid = False
            break
        for j in range(i + 1, N):
            dist = np.sqrt((x - centers[j, 0])**2 + (y - centers[j, 1])**2)
            if dist < r + radii[j] - 1e-12:
                valid = False
                break
    
    if not valid:
        # Fallback scaling
        radii = radii * 0.999
        # Re-center if out of bounds (rare)
        centers = np.clip(centers, radii[:, None], 1 - radii[:, None])

    return centers, radii, np.sum(radii)

if __name__ == "__main__":
    # Run the function to get results
    c, r, s = run_packing()
    print(f"Optimal Sum of Radii: {s}")
    print(f"Centers shape: {c.shape}, Radii shape: {r.shape}")
