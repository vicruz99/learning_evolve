import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Pack 26 circles in a unit square to maximize the sum of radii.
    
    Returns:
        centers: np.array of shape (26, 2)
        radii: np.array of shape (26,)
        sum_radii: float
    """
    n_circles = 26
    
    # --- 1. Initialization ---
    # Estimate optimal radius for hexagonal packing density ~0.9
    # Area of square = 1. Area of circles = n * pi * r^2.
    # 0.9 = 26 * pi * r^2 => r ≈ 0.105
    r_est = np.sqrt(0.9 / (n_circles * np.pi))
    
    # Generate points on a hexagonal lattice
    # We want to fit 26 points.
    # Spacing in hex lattice: horizontal 2r, vertical sqrt(3)r.
    # Let's generate a grid of potential centers and pick 26 that are well-distributed.
    # Or simply generate a hex lattice and clip/adjust.
    
    centers_init = []
    
    # Hexagonal grid parameters
    # Let's try to fit points with spacing approx 2*r_est
    # But to allow optimizer to expand, we can start slightly tighter or looser.
    # Let's start with a spacing that allows no overlap for r_est.
    dx = 2.0 * r_est
    dy = np.sqrt(3.0) * r_est
    
    # Generate points
    y = 0.0
    row_idx = 0
    while True:
        # Offset x for odd rows
        x_offset = (dx / 2.0) if (row_idx % 2 == 1) else 0.0
        x = 0.0
        while True:
            cx = x + x_offset
            cy = y
            # Check if inside [0,1] with some margin
            if cx >= 0 and cx <= 1 and cy >= 0 and cy <= 1:
                centers_init.append([cx, cy])
                if len(centers_init) >= n_circles:
                    break
            x += dx
            if x + x_offset > 1.5: # Stop generating x points
                break
        if len(centers_init) >= n_circles:
            break
        y += dy
        row_idx += 1
        if y > 1.5:
            break
            
    # If we didn't get enough points (unlikely with r_est ~ 0.1), fill with random
    if len(centers_init) < n_circles:
        for _ in range(n_circles - len(centers_init)):
            centers_init.append([np.random.rand(), np.random.rand()])
    
    # Trim to exactly n_circles
    centers_init = centers_init[:n_circles]
    
    # Flatten variables: [x1, y1, r1, x2, y2, r2, ...]
    # Initial radii: start with r_est
    x0 = []
    for c in centers_init:
        x0.extend([c[0], c[1], r_est])
    x0 = np.array(x0)
    
    # --- 2. Optimization ---
    
    # Bounds: x, y in [0, 1], r in [0, 0.5]
    bounds = []
    for _ in range(n_circles):
        bounds.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
    
    # Penalty weight
    # High weight to enforce constraints strictly
    alpha = 1000.0 
    
    def objective(vars):
        # Unpack
        centers = np.zeros((n_circles, 2))
        radii = np.zeros(n_circles)
        for i in range(n_circles):
            idx = i * 3
            centers[i] = [vars[idx], vars[idx+1]]
            radii[i] = vars[idx+2]
            
        # Objective: Maximize sum of radii -> Minimize negative sum
        obj_val = -np.sum(radii)
        
        # Penalty for overlaps
        penalty = 0.0
        for i in range(n_circles):
            r_i = radii[i]
            c_i = centers[i]
            
            # Boundary penalties
            # x - r >= 0 => r - x <= 0. Violation if r > x.
            if r_i > centers[i, 0]:
                penalty += alpha * (r_i - centers[i, 0])**2
            # x + r <= 1 => x + r - 1 <= 0. Violation if x + r > 1.
            if centers[i, 0] + r_i > 1.0:
                penalty += alpha * (centers[i, 0] + r_i - 1.0)**2
            # y - r >= 0
            if r_i > centers[i, 1]:
                penalty += alpha * (r_i - centers[i, 1])**2
            # y + r <= 1
            if centers[i, 1] + r_i > 1.0:
                penalty += alpha * (centers[i, 1] + r_i - 1.0)**2
                
            # Overlap penalties with subsequent circles
            for j in range(i + 1, n_circles):
                r_j = radii[j]
                c_j = centers[j]
                dist = np.sqrt(np.sum((c_i - c_j)**2))
                sum_r = r_i + r_j
                if dist < sum_r:
                    overlap = sum_r - dist
                    penalty += alpha * overlap**2
                    
        return obj_val + penalty

    # Run optimizer
    # L-BFGS-B is good for bound-constrained problems
    # We might need to run it a few times or with different seeds if it gets stuck,
    # but a good initial guess usually helps.
    
    result = minimize(
        objective, 
        x0, 
        method='L-BFGS-B', 
        bounds=bounds,
        options={'ftol': 1e-9, 'gtol': 1e-6, 'maxiter': 5000, 'disp': False}
    )
    
    # Extract solution
    optimal_vars = result.x
    centers = np.zeros((n_circles, 2))
    radii = np.zeros(n_circles)
    for i in range(n_circles):
        idx = i * 3
        centers[i] = [optimal_vars[idx], optimal_vars[idx+1]]
        radii[i] = optimal_vars[idx+2]
        
    sum_radii = np.sum(radii)
    
    # --- 3. Validation and Cleanup ---
    # Due to numerical noise, radii might be slightly negative or constraints slightly violated.
    # Enforce non-negative radii.
    radii = np.maximum(radii, 0.0)

    # However, L-BFGS-B with high alpha should be close.

    valid = True
    max_violation = 0.0

    for i in range(n_circles):
        x, y = centers[i]
        r = radii[i]
        # Project to valid range
        # x must be in [r, 1-r]
        min_x = r
        max_x = 1.0 - r
        if x < min_x: x = min_x
        if x > max_x: x = max_x
        min_y = r
        max_y = 1.0 - r
        if y < min_y: y = min_y
        if y > max_y: y = max_y
        centers[i] = [x, y]

                    radii[j] -= overlap / 2.0

            break
            
    # Final clamp radii to 0
    radii = np.maximum(radii, 0.0)
    
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
