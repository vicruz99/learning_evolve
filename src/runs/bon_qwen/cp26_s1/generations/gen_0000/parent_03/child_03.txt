import numpy as np
from scipy.optimize import minimize
import random

def get_distance_matrix(centers):
    """Helper to compute pairwise distances."""
    # Centers shape (n, 2)
    # dist^2 matrix
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    return np.sqrt(np.sum(diff**2, axis=2))

def generate_hexagonal_initial_guess(n, w=1.0, h=1.0):
    """
    Generates an initial guess for centers based on a hexagonal lattice.
    """
    centers = []
    # Approximate radius for hexagonal packing of 26 circles.
    # Density ~ 0.9069. Area ~ 1. 
    # 26 * pi * r^2 / 0.9069 <= 1 => r approx 0.105
    # But boundary effects reduce this. Start with r_guess = 0.1
    r_guess = 0.1
    
    # Hexagonal spacing
    dx = 2 * r_guess
    dy = r_guess * np.sqrt(3)
    
    y = r_guess
    row = 0
    while y + r_guess <= h:
        x = r_guess
        # Shift every other row
        if row % 2 == 1:
            x += r_guess
        
        while x + r_guess <= w:
            centers.append([x, y])
            x += dx
            if len(centers) >= n:
                break
        if len(centers) >= n:
            break
        y += dy
        row += 1
        
    # If we have fewer than n, add random points inside valid area
    # If we have more, truncate
    centers = centers[:n]
    
    # Pad with random points if needed (though hex grid usually gives enough)
    while len(centers) < n:
        cx = random.uniform(r_guess, 1 - r_guess)
        cy = random.uniform(r_guess, 1 - r_guess)
        centers.append([cx, cy])
        
    return np.array(centers), r_guess

def solve_packing(n_circles=26):
    best_sum_radii = -1.0
    best_centers = None
    best_radii = None
    
    # Try multiple restarts to avoid local minima
    n_restarts = 10
    
    for _ in range(n_restarts):
        # 1. Initialize
        init_centers, r_init = generate_hexagonal_initial_guess(n_circles)
        
        # Add some noise to break symmetry
        noise = np.random.normal(0, 0.01, init_centers.shape)
        init_centers = np.clip(init_centers + noise, r_init, 1 - r_init)
        
        # Variables: [x1, y1, r1, x2, y2, r2, ...]
        # Or separate arrays? scipy minimizes a 1D array.
        # Let's flatten: [x1, y1, r1, ..., xn, yn, rn]
        # Total variables: 3 * n_circles
        
        x0 = np.zeros(3 * n_circles)
        for i in range(n_circles):
            x0[3*i] = init_centers[i, 0]
            x0[3*i+1] = init_centers[i, 1]
            x0[3*i+2] = r_init
            
        # Bounds: x, y in [0, 1], r in [0, 1]
        # Actually tighter bounds: r <= 0.5, x,y depend on r but for simplicity [0,1]
        # Better: r <= 0.5
        bounds = []
        for _ in range(n_circles):
            bounds.append((0.0, 1.0)) # x
            bounds.append((0.0, 1.0)) # y
            bounds.append((0.0, 0.5)) # r
        
        def objective(vars):
            # vars is 1D array
            sum_r = 0
            for i in range(n_circles):
                sum_r += vars[3*i + 2]
            return -sum_r # Minimize negative sum

        # Constraints
        constraints = []
        
        # 1. Boundary constraints: r <= x <= 1-r  =>  x - r >= 0, x + r <= 1
        # y - r >= 0, y + r <= 1
        for i in range(n_circles):
            idx_x = 3*i
            idx_y = 3*i+1
            idx_r = 3*i+2
            
            # x - r >= 0
            constraints.append({
                'type': 'ineq',
                'fun': lambda v, i=i: v[idx_x] - v[idx_r]
            })
            # 1 - x - r >= 0
            constraints.append({
                'type': 'ineq',
                'fun': lambda v, i=i: 1.0 - v[idx_x] - v[idx_r]
            })
            # y - r >= 0
            constraints.append({
                'type': 'ineq',
                'fun': lambda v, i=i: v[idx_y] - v[idx_r]
            })
            # 1 - y - r >= 0
            constraints.append({
                'type': 'ineq',
                'fun': lambda v, i=i: 1.0 - v[idx_y] - v[idx_r]
            })

        # 2. Non-overlap constraints: dist >= r_i + r_j
        # (xi-xj)^2 + (yi-yj)^2 >= (ri+rj)^2
        # (xi-xj)^2 + (yi-yj)^2 - (ri+rj)^2 >= 0

            for j in range(i + 1, n_circles):
                idx_xi = 3*i
                idx_yi = 3*i+1
                idx_ri = 3*i+2
                idx_xj = 3*j
                idx_yj = 3*j+1
                idx_rj = 3*j+2
                
                def make_overlap_func(i, j):
                    def fun(v):
                        xi, yi, ri = v[3*i], v[3*i+1], v[3*i+2]
                        xj, yj, rj = v[3*j], v[3*j+1], v[3*j+2]
                        dist_sq = (xi - xj)**2 + (yi - yj)**2
                        rad_sum_sq = (ri + rj)**2
                        return dist_sq - rad_sum_sq
                    return fun

                constraints.append({
                    'type': 'ineq',
                    'fun': make_overlap_func(i, j)
                })

        # Run optimization
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints, 
                           options={'maxiter': 500, 'ftol': 1e-9})

                # Check validity roughly (optimizers can be messy with constraints)
                current_sum = -res.fun

                # but we trust the constraints mostly. 
                # However, SLSQP allows small violations.
                # Let's clamp radii to be valid based on positions to be safe.
                
                centers_out = np.zeros((n_circles, 2))
                radii_out = np.zeros(n_circles)
                
                valid = True
                for i in range(n_circles):
                    cx = res.x[3*i]
                    cy = res.x[3*i+1]
                    cr = res.x[3*i+2]
                    
                    # Enforce bounds strictly
                    cr = min(cr, cx, 1-cx, cy, 1-cy)
                    cr = max(cr, 0)
                    
                    centers_out[i] = [cx, cy]
                    radii_out[i] = cr
                
                # Final validation check (loose)
                # If radii became negative or too small, discard?
                # Actually, just take the result.
                
                if valid and current_sum > best_sum_radii:
                    best_sum_radii = current_sum
                    best_centers = centers_out
                    best_radii = radii_out
                    
        except Exception as e:
            continue

    return best_centers, best_radii, best_sum_radii
