import scipy.optimize as opt

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    """
    n = 26

    # Use a 5x5 grid + 1 center as a robust starting point for 26 circles.
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 5
        col = i % 5
        # Space them out in a 0.9 range to leave some initial buffer

    def get_cost(x, penalty_weight):

        x: flattened array of [x1, y1, r1, ..., x26, y26, r26]
        Returns cost to minimize.
        """
        # Reshape variables
        centers_opt = x[:2*n].reshape(n, 2)
        radii_opt = x[2*n:]
        
        # Boundary penalties: push centers away from walls if r > dist_to_wall
        # Penalty is (violation)^2
        penalty = 0.0
        
        # Boundary constraints
        # r <= x <= 1-r  =>  x >= r and x <= 1-r
        for i in range(n):
            x_c, y_c = centers_opt[i]
            r = radii_opt[i]

            if x_c < r:
                penalty += penalty_weight * (r - x_c)**2
            # Distance to right wall
            if x_c > 1 - r:
                penalty += penalty_weight * (x_c - (1 - r))**2
            # Distance to bottom wall
            if y_c < r:
                penalty += penalty_weight * (r - y_c)**2
            # Distance to top wall
            if y_c > 1 - r:
                penalty += penalty_weight * (y_c - (1 - r))**2
