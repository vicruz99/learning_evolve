import numpy as np
from scipy.optimize import linprog

def compute_optimal_radii(centers):
    """
    Given fixed centers, compute radii that maximize sum(radii)
    subject to boundary and non-overlap constraints using LP.
    """
    N = centers.shape[0]
    
    # Boundary constraints: r_i <= min(x_i, 1-x_i, y_i, 1-y_i)
    bounds_i = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]), 
                          np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    
    # LP: minimize -sum(r_i)  => maximize sum(r_i)
    c = np.ones(N)
    
    # Constraints matrix and vector
    # Number of constraints: N (boundary) + N(N-1)/2 (pairwise)
    num_constraints = N + (N * (N - 1)) // 2
    A_ub = np.zeros((num_constraints, N))
    b_ub = np.zeros(num_constraints)
    
    row = 0
    # Boundary constraints: 1 * r_i <= bounds_i[i]
    for i in range(N):
        A_ub[row, i] = 1.0
        b_ub[row] = bounds_i[i]
        row += 1
        
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    for i in range(N):
        for j in range(i + 1, N):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            A_ub[row, i] = 1.0
            A_ub[row, j] = 1.0
            b_ub[row] = dist
            row += 1
            
    bounds = [(0, 1.0)] * N
    
    # Solve LP
    res = linprog(-c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return -res.fun, res.x
    else:
        # Fallback: conservative radii
        fallback_r = bounds_i * 0.4
        return np.sum(fallback_r), fallback_r
