def compute_radii(centers, n):
    """Compute radii for each circle based on current positions."""
    radii = np.zeros(n)
    for i in range(n):
        # Distance to boundaries
        min_d = min(centers[i, 0], 1.0 - centers[i, 0],
                    centers[i, 1], 1.0 - centers[i, 1])
        # Distance to other circles
        for j in range(n):
            if i != j:
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                d = np.sqrt(dx * dx + dy * dy)
                if d < min_d:
                    min_d = d
        radii[i] = min_d / 2.0
    return radii

    idx = 0

    spacing_y = 1.0 / (rows + 1)

            if idx >= n:
                break

            if row % 2 == 1:
                x += spacing_x / 2.0
            centers[idx, 0] = x
            centers[idx, 1] = y
            idx += 1
    
    return centers
