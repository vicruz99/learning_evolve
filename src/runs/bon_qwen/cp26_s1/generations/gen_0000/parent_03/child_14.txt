import numpy as np
import scipy.optimize as opt

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # 1. Generate Initial Centers using Hexagonal Grid
    centers_init = generate_hex_centers(n)
    
    # 2. Optimize Centers to maximize Sum of Radii
    # We maximize f(C) = max sum(r_i) s.t. distance constraints
    # This is equivalent to minimizing -f(C)
    
    # Bounds for centers: [0, 1] for each x and y
    bounds = [(0, 1)] * (2 * n)
    
    # Optimization function wrapper
    def objective(coords):
        centers = coords.reshape((n, 2))
        # Ensure centers are within bounds to avoid issues in distance calc
        # though optimizer handles bounds, clipping for safety
        centers = np.clip(centers, 1e-9, 1 - 1e-9)
        return -get_max_radii_sum(centers, n)
    
    # Run optimization
    # Powell method is good for derivative-free optimization of this type
    res = opt.minimize(objective, centers_init.flatten(), method='Powell', bounds=bounds, 
                       options={'maxiter': 1000, 'ftol': 1e-12, 'xtol': 1e-12})
    
    best_centers = res.x.reshape((n, 2))
    
    # 3. Compute Final Radii for the optimized centers
    best_radii = get_radii_array(best_centers, n)
    sum_radii = np.sum(best_radii)
    
    # Final check to ensure strict validity (handling any tiny numerical violations)
    # Although the LP ensures constraints, numerical noise might exist.
    # We can slightly shrink radii if needed, but usually not necessary.
    
    return best_centers, best_radii, sum_radii

def generate_hex_centers(n):
    """Generates n points in a hexagonal grid pattern inside [0,1]x[0,1]."""
    centers = []
    # Estimate spacing to fit n circles roughly
    # Area of hex cell ~ sqrt(3) * s^2 where s is spacing? 
    # Actually distance between neighbors is s. Area per point ~ s^2 * sqrt(3)/2?
    # Let's just generate a grid and scale/trim.
    
    # Generate a larger grid of points
    pts = []
    s = 0.1 # Initial step size, we will filter and select
    
    # We want to cover the square. 
    # y coordinates: j * h, where h = s * sqrt(3)/2
    # x coordinates: i * s + (j%2)*s/2
    
    h = s * np.sqrt(3) / 2
    
    y = 0
    row = 0
    while len(pts) < n + 10: # Generate a few extra to pick best or just fill
        x = 0
        if row % 2 == 1:
            x = s / 2
        
        while x <= 1.0:
            pts.append([x, y])
            x += s
        
        y += h
        row += 1
        
    # Select first n points
    # Ideally, we want points centered in the square.
    # The generated points start at 0. 
    # Let's center them.
    
    if len(pts) >= n:
        # Just take the first n? 
        # Better to take points that are well distributed.
        # Since we generated row by row from bottom-left, it's okay.
        # But let's shift them to be centered if they are biased.
        # Actually, optimization will fix positions. 
        # Just ensuring they are inside [0,1] is key.
        selected = np.array(pts[:n])
    else:
        # If we didn't generate enough, reduce spacing
        selected = np.array(pts) # Fallback
        
    return selected

def get_max_radii_sum(centers, n):
    """
    Solves LP to maximize sum of radii for fixed centers.
    Returns the max sum.
    """
    # Variables: r_0, ..., r_{n-1}
    # Objective: min -sum(r_i)
    c_obj = np.ones(n) * -1
    
    # Constraints:
    # 1. r_i + r_j <= dist(c_i, c_j) for all i < j
    # 2. r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    # 3. r_i >= 0 (bounds)
    
    m = 0
    constraints_list = []
    
    # Precompute distances
    dists = np.linalg.norm(centers[:, np.newaxis, :] - centers[np.newaxis, :, :], axis=2)
    
    # Pairwise constraints
    for i in range(n):
        for j in range(i + 1, n):
            # r_i + r_j <= dist_ij
            # 1*r_i + 1*r_j <= dist_ij
            row = np.zeros(n)
            row[i] = 1
            row[j] = 1
            constraints_list.append(row)
            m += 1
            
    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        
        # r_i <= x
        row = np.zeros(n)
        row[i] = 1
        constraints_list.append(row)
        m += 1
        
        # r_i <= 1-x
        row = np.zeros(n)
        row[i] = 1
        constraints_list.append(row)
        m += 1
        
        # r_i <= y
        row = np.zeros(n)
        row[i] = 1
        constraints_list.append(row)
        m += 1
        
        # r_i <= 1-y
        row = np.zeros(n)
        row[i] = 1
        constraints_list.append(row)
        m += 1

    # Build A_ub and b_ub
    A_ub = np.array(constraints_list)
    b_ub = np.zeros(len(constraints_list))
    
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            b_ub[idx] = dists[i, j]
            idx += 1
            
    for i in range(n):
        x, y = centers[i]
        b_ub[idx] = x
        idx += 1
        b_ub[idx] = 1 - x
        idx += 1
        b_ub[idx] = y
        idx += 1
        b_ub[idx] = 1 - y
        idx += 1
        
    # Bounds for r: [0, inf)
    r_bounds = [(0, None)] * n
    
    # Solve LP
    # Use 'highs' solver if available, else default
    try:
        res = opt.linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=r_bounds, method='highs')
    except:
        res = opt.linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=r_bounds)
        
    if res.success:
        return -res.fun
    else:
        # Fallback or error
        return 0.0

def get_radii_array(centers, n):
    """
    Returns the array of radii for the given centers.
    Re-solves the LP to get the values.
    """
    c_obj = np.ones(n) * -1

    for i in range(n):
        x, y = centers[i]
        row = np.zeros(n); row[i] = 1; constraints_list.append(row)
        row = np.zeros(n); row[i] = 1; constraints_list.append(row)
        row = np.zeros(n); row[i] = 1; constraints_list.append(row)
        row = np.zeros(n); row[i] = 1; constraints_list.append(row)
        
    A_ub = np.array(constraints_list)
    b_ub = np.zeros(len(constraints_list))
    
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            b_ub[idx] = dists[i, j]
            idx += 1
    for i in range(n):
        x, y = centers[i]
        b_ub[idx] = x; idx += 1
        b_ub[idx] = 1 - x; idx += 1
        b_ub[idx] = y; idx += 1
        b_ub[idx] = 1 - y; idx += 1
        
    r_bounds = [(0, None)] * n
    
    try:
        res = opt.linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=r_bounds, method='highs')
    except:
        res = opt.linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=r_bounds)
        
    if res.success:
        return res.x
    else:
        return np.zeros(n)

if __name__ == "__main__":
    centers, radii, sum_r = run_packing()
    print(f"Sum of radii: {sum_r}")
