from scipy.optimize import minimize
from scipy.spatial.distance import pdist

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    def get_initial_guess(seed):
        rng = np.random.default_rng(seed)
        
        # Hexagonal layout generation
        # Rows: 6, 5, 6, 5, 4
        rows_cfg = [
            (6, 0), (5, 1), (6, 0), (5, 1), (4, 0)
        ]

        y_current = 0

                x = k * 2 + shift

            y_current += dy

        min_x, min_y = np.min(pts_raw, axis=0)
        max_x, max_y = np.max(pts_raw, axis=0)
        
        width = max_x - min_x
        height = max_y - min_y
        
        # Target fit in [0.1, 0.9] -> size 0.8
        margin = 0.1
        target_size = 1.0 - 2 * margin

        center_y = (max_y + min_y) / 2

        pts_scaled += noise

        if n > 1:

            min_dist = np.min(dists)
        else:
            min_dist = 1.0

            pts_scaled[:, 0] - margin,
            (1 - margin) - pts_scaled[:, 0],
            pts_scaled[:, 1] - margin,
            (1 - margin) - pts_scaled[:, 1]
