import numpy as np
import math
from scipy.optimize import minimize

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square

    Args:
        centers: np.array of shape (n, 2) with (x, y) coordinates
        radii: np.array of shape (n) with radius of each circle

    Returns:
        True if valid, False otherwise
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True

def get_initial_positions(n):
    """Generates a hexagonal-like grid of positions."""
    positions = []
    if n <= 0: return np.array(positions)
    
    # Approximate rows and cols
    # Area per circle approx sqrt(3)/2 * (2r)^2 ? No, just estimate density.
    # For r=0.1, diameter 0.2. ~25 circles.
    
    # Hexagonal packing generation
    # Row spacing sqrt(3)/2 * diameter = sqrt(3) * r
    # Let's assume r ~ 0.1
    row_spacing = 0.1732
    col_spacing = 0.2
    
    y = 0.1
    while len(positions) < n:
        x = 0.1
        row_offset = (len(positions) // 2) * (col_spacing / 2) if len(positions) % 2 == 1 else 0
        while x < 1.0 and len(positions) < n:
            positions.append([x + row_offset, y])
            x += col_spacing
        y += row_spacing
        
    return np.array(positions[:n])

def objective_func(params, n):
    """
    Objective function to minimize: -Sum of radii + Penalty for violations
    """
    centers = params[:n*2].reshape(n, 2)
    radii = params[n*2:]
    
    # Objective: Maximize sum of radii -> Minimize -sum(radii)
    obj_val = -np.sum(radii)
    
    penalty_weight = 10.0
    total_penalty = 0.0
    
    # Boundary penalties
    # x - r >= 0 => r - x <= 0
    # x + r <= 1 => x + r - 1 <= 0
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        
        # Soft penalties for boundaries
        if x - r < 0: total_penalty += (x - r)**2
        if x + r > 1: total_penalty += (x + r - 1)**2
        if y - r < 0: total_penalty += (y - r)**2
        if y + r > 1: total_penalty += (y + r - 1)**2
        
    # Overlap penalties
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            sum_radii = radii[i] + radii[j]
            if dist < sum_radii:
                total_penalty += (dist - sum_radii)**2
                
    return obj_val + penalty_weight * total_penalty

def run_packing():
    n = 26
    best_centers = None
    best_radii = None
    best_score = -np.inf
    
    # Try multiple initial configurations
    # Config 1: Hexagonal grid
    initial_centers = get_initial_positions(n)
    # Config 2: Random
    np.random.seed(42)
    random_centers = np.random.rand(n, 2)
    
    configs = [initial_centers, random_centers]
    
    for seed in range(10): # Multiple restarts
        np.random.seed(seed)
        if seed < 2:
            # Use pre-defined configs for first few
            centers = configs[seed % 2].copy()
        else:
            # Perturbed random
            centers = np.random.rand(n, 2)
            
        # Initial radii: start small to ensure validity, then grow
        # Or start with estimated valid radius
        r_init = 0.05 
        
        # Pack tightly initially to find valid centers? 
        # Actually, let's just start with small radii and let optimizer grow them.
        
        x0 = np.concatenate([centers.flatten(), np.ones(n) * r_init])
        
        # Bounds for variables:
        # x, y in [0, 1]
        # r in [0, 0.5]
        bounds = []
        for _ in range(n):
            bounds.extend([(0, 1), (0, 1)]) # x, y
            bounds.append((0, 0.5))          # r
            
        try:
            res = minimize(objective_func, x0, args=(n,), method='L-BFGS-B', bounds=bounds, 
                           options={'maxiter': 2000, 'ftol': 1e-9})
            
            if res.success or res.fun < 0: # res.fun is negative sum + penalty
                curr_centers = res.x[:n*2].reshape(n, 2)
                curr_radii = res.x[n*2:]
                
                # Validate
                if validate_packing(curr_centers, curr_radii):
                    current_score = np.sum(curr_radii)
                    if current_score > best_score:
                        best_score = current_score
                        best_centers = curr_centers.copy()
                        best_radii = curr_radii.copy()
        except:
            pass

    # Fallback to simple grid if optimization fails (unlikely)
    if best_centers is None:
        best_centers = get_initial_positions(n)
        best_radii = np.full(n, 0.1)

    return best_centers, best_radii, np.sum(best_radii)
