import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Penalty weight for constraints
    # High weight ensures constraints are satisfied if possible
    PENALTY_WEIGHT = 1000.0 

    def objective(vars):
        centers = vars[:2*n].reshape((n, 2))
        radii = vars[2*n:]
        
        # Objective: Minimize negative sum of radii
        val = -np.sum(radii)
        
        # --- Penalty Terms ---
        penalty = 0.0
        
        # 1. Boundary Constraints
        # Circles must be inside [0,1]x[0,1]
        # x - r >= 0  =>  r - x <= 0
        # 1 - x - r >= 0 => r - (1-x) <= 0
        # y - r >= 0
        # 1 - y - r >= 0
        
        # Vectorized calculation
        # Left wall: r <= x => violation max(0, r - x)
        v_left = np.maximum(0, radii - centers[:, 0])
        # Right wall: r <= 1-x => violation max(0, r - (1-x))
        v_right = np.maximum(0, radii - (1 - centers[:, 0]))
        # Bottom wall: r <= y => violation max(0, r - y)
        v_bottom = np.maximum(0, radii - centers[:, 1])
        # Top wall: r <= 1-y => violation max(0, r - (1-y))
        v_top = np.maximum(0, radii - (1 - centers[:, 1]))
        
        penalty += np.sum(v_left**2) + np.sum(v_right**2) + np.sum(v_bottom**2) + np.sum(v_top**2)
        
        # 2. Non-overlap Constraints
        # dist(i, j) >= r_i + r_j  =>  r_i + r_j - dist(i, j) <= 0
        # Compute pairwise distances
        # Centers shape (n, 2)
        # Differences shape (n, n, 2)
        diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
        dists = np.sqrt(np.sum(diffs**2, axis=2))
        
        # Sum of radii matrix shape (n, n)
        r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
        
        # Violation matrix
        # We only care about i < j, but squaring and summing all is fine (symmetric)
        # Diagonal is 0 (r_i + r_i - 0 = 2r_i > 0?), wait dist is 0 on diagonal.
        # But constraint is for i != j.
        # Let's mask diagonal or just ignore since we iterate carefully or handle it.
        # Actually, dist(i,i) = 0, r_i + r_i = 2r_i. Violation 2r_i.
        # This would penalize radii being non-zero!

        mask = np.triu(np.ones((n, n), dtype=bool), k=1)
        
        violations = np.maximum(0, r_sum - dists)
        penalty += np.sum(violations[mask]**2)
        
        return val + PENALTY_WEIGHT * penalty

            # x, y in [0, 1]
            bounds.append((0.0, 1.0))
            bounds.append((0.0, 1.0))
            # r in [0, 0.5] (radius cannot be larger than 0.5 in unit square)
            bounds.append((0.0, 0.5))
        return tuple(bounds)

        centers_init = np.random.uniform(0.1, 0.9, size=(n, 2))
        radii_init = np.ones(n) * 0.05 # Start small
        
        vars_init = np.concatenate([centers_init.flatten(), radii_init])

                           options={'maxiter': 2000, 'ftol': 1e-10, 'gtol': 1e-8})

                # Extract solution
                centers_cand = res.x[:2*n].reshape((n, 2))
                radii_cand = res.x[2*n:]

                valid = True

                        r = radii_cand[i]

                            break

                            valid = False
                            break

                                    valid = False
                                    break
                            if not valid:
                                break
