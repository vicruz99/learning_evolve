import numpy as np

def run_packing():
    """
    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    Uses Linear Programming (CVXPY) to find optimal radii for fixed centers, 
    and gradient ascent on centers based on LP dual variables to improve packing density.
    """
    try:
        import cvxpy as cp
    except ImportError:
        # Fallback implementation if cvxpy is not available
        # Returns a basic grid packing
        centers = np.zeros((26, 2))
        radii = np.zeros(26)
        idx = 0
        # 5x5 grid
        for i in range(5):
            for j in range(5):
                centers[idx] = [0.1 + i*0.2, 0.1 + j*0.2]
                radii[idx] = 0.09
                idx += 1
        # 26th circle
        centers[25] = [0.2, 0.2]
        radii[25] = 0.04
        return centers, radii, float(np.sum(radii))

    n_circles = 26
    np.random.seed(42)
    
    # --- Initialization ---
    # Start with a 5x5 grid plus an extra point, then spread them out using repulsion
    pts = []
    for i in range(5):
        for j in range(5):
            pts.append([0.1 + i*0.2, 0.1 + j*0.2])
    pts.append([0.2, 0.2])
    centers = np.array(pts)
    
    # Repulsion step to ensure points are well-distributed
    for _ in range(150):
        forces = np.zeros_like(centers)
        for i in range(n_circles):
            for j in range(i+1, n_circles):
                diff = centers[i] - centers[j]
                dist = np.linalg.norm(diff)
                if dist < 0.15 and dist > 1e-6:
                    forces[i] += diff / dist
                    forces[j] -= diff / dist
            # Repel from walls
            for k in range(2):
                if centers[i, k] < 0.05: forces[i, k] += 1.0
                if centers[i, k] > 0.95: forces[i, k] -= 1.0
        centers += forces * 0.01
        centers = np.clip(centers, 0.05, 0.95)

    # --- Optimization Loop ---
    max_iter = 80
    lr = 0.05
    
    for it in range(max_iter):
        r = cp.Variable(n_circles, nonneg=True)
        constraints = []
        wall_cons = []
        pair_cons = []
        
        # Wall constraints: r_i <= x, r_i <= 1-x, r_i <= y, r_i <= 1-y
        for i in range(n_circles):
            x, y = centers[i]
            c1 = cp.Constraint(r[i] <= x)
            c2 = cp.Constraint(r[i] <= 1 - x)
            c3 = cp.Constraint(r[i] <= y)
            c4 = cp.Constraint(r[i] <= 1 - y)
            constraints.extend([c1, c2, c3, c4])
            wall_cons.append([c1, c2, c3, c4])
            
        # Overlap constraints: r_i + r_j <= dist(i, j)
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                c = cp.Constraint(r[i] + r[j] <= dist)
                constraints.append(c)
                pair_cons.append({'i': i, 'j': j, 'c': c})
        
        objective = cp.Maximize(cp.sum(r))
        prob = cp.Problem(objective, constraints)
        
        try:
            # Solve LP with high precision
            prob.solve(solver=cp.ECOS, eps_abs=1e-9, eps_rel=1e-9)
        except Exception:
            break
            
        if prob.status not in ["optimal", "optimal_inaccurate"]:
            break
            
        radii = np.array(r.value)
        if not np.all(np.isfinite(radii)):
            break
            
        # Compute forces on centers based on dual variables (gradients)
        forces = np.zeros((n_circles, 2))
        
        # Wall duals: moving center away from active wall constraint increases radius sum
        for i in range(n_circles):
            c1, c2, c3, c4 = wall_cons[i]
            try:
                mu_x_p = c1.dual_value if c1.dual_value is not None else 0.0
                mu_x_n = c2.dual_value if c2.dual_value is not None else 0.0
                mu_y_p = c3.dual_value if c3.dual_value is not None else 0.0
                mu_y_n = c4.dual_value if c4.dual_value is not None else 0.0
            except:
                mu_x_p = mu_x_n = mu_y_p = mu_y_n = 0.0
            
            # Force direction: +x if touching left wall (x=0), -x if touching right wall (x=1)
            forces[i, 0] += (mu_x_p - mu_x_n)
            forces[i, 1] += (mu_y_p - mu_y_n)
            
        # Pair duals: separating touching circles increases radius sum
        for item in pair_cons:
            i, j = item['i'], item['j']
            try:

                    forces[i] += mu * u
                    forces[j] -= mu * u

        centers = np.clip(centers, 0.001, 0.999)

    # Solve one last time with fixed centers to get precise radii
    r_var = cp.Variable(n_circles, nonneg=True)

            constraints_final.append(r_var[i] + r_var[j] <= d)

    radii = np.array(r_var.value) if r_var.value is not None else np.zeros(n_circles)
    sum_radii = float(np.sum(radii))
    
    return centers, radii, sum_radii
