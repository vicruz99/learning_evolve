import numpy as np
from scipy.optimize import linprog

def solve_radii_lp(centers, n):
    """
    Solve the Linear Programming problem to maximize sum of radii
    given fixed centers.
    
    Constraints:
    1. r_i >= 0
    2. r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i (Boundaries)
    3. r_i + r_j <= distance(i, j) (Non-overlap)
    """
    # Objective: Maximize sum(r_i) -> Minimize -sum(r_i)
    c_obj = np.ones(n) * -1.0
    
    # Inequality constraints A_ub @ r <= b_ub
    A_ub = []
    b_ub = []
    
    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        # r_i <= x_i
        row_x = np.zeros(n)
        row_x[i] = 1.0
        A_ub.append(row_x)
        b_ub.append(x)
        
        # r_i <= 1 - x_i
        A_ub.append(row_x)
        b_ub.append(1.0 - x)
        
        # r_i <= y_i
        A_ub.append(row_x)
        b_ub.append(y)
        
        # r_i <= 1 - y_i
        A_ub.append(row_x)
        b_ub.append(1.0 - y)
        
    # Non-overlap constraints: r_i + r_j <= dist
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            row_pair = np.zeros(n)
            row_pair[i] = 1.0
            row_pair[j] = 1.0
            A_ub.append(row_pair)
            b_ub.append(dist)
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    # Bounds for r_i: [0, None]
    bounds = [(0, None) for _ in range(n)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return -res.fun, res.x # Return max sum and radii
        else:
            # Fallback: return small radii if LP fails
            return 0.0, np.zeros(n)
    except Exception:
        return 0.0, np.zeros(n)
