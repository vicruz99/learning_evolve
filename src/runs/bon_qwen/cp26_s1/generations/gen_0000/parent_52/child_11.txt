import numpy as np
from scipy.optimize import minimize

N = 26


def objective(x):
    """Minimize negative sum of radii (equivalent to maximizing sum of radii)"""
    return -np.sum(x[2 * N :])


def boundary_constraints(x):
    """Boundary constraints: each circle must be inside [0,1]^2"""
    centers = x[: 2 * N].reshape(N, 2)
    radii = x[2 * N :]
    cons = np.empty(4 * N)
    for i in range(N):
        cons[4 * i] = centers[i, 0] - radii[i]
        cons[4 * i + 1] = 1.0 - centers[i, 0] - radii[i]
        cons[4 * i + 2] = centers[i, 1] - radii[i]
        cons[4 * i + 3] = 1.0 - centers[i, 1] - radii[i]
    return cons


def nonoverlap_constraints(x):
    """Non-overlap constraints: distance between centers >= sum of radii"""
    centers = x[: 2 * N].reshape(N, 2)
    radii = x[2 * N :]
    ncons = N * (N - 1) // 2
    cons = np.empty(ncons)
    k = 0
    for i in range(N):
        for j in range(i + 1, N):
            d2 = np.sum((centers[i] - centers[j]) ** 2)
            sr = radii[i] + radii[j]
            cons[k] = d2 - sr * sr
            k += 1
    return cons


def init_hexagonal_packing():
    """Initialize circles in a hexagonal grid pattern"""
    centers = np.zeros((N, 2))
    radii = np.full(N, 0.09)

    idx = 0
    row = 0
    r_init = 0.09
    h = np.sqrt(3) * r_init  # vertical spacing
    d = 2 * r_init  # horizontal spacing

    while idx < N:
        n_cols = 5
        offset = (row % 2) * r_init
        for col in range(n_cols):
            if idx >= N:
                break
            x = r_init + col * d + offset
            y = r_init + row * h
            if x <= 1.0 - r_init and y <= 1.0 - r_init:
                centers[idx] = [x, y]
                idx += 1
        row += 1

    # Fill any remaining circles with small perturbations
    if idx < N:
        for i in range(idx, N):
            centers[i] = [
                0.5 + 0.15 * np.cos(2 * np.pi * i / 6),
                0.5 + 0.15 * np.sin(2 * np.pi * i / 6),
            ]

    return centers, radii


def init_grid_packing():
    """Initialize circles in a slightly perturbed grid pattern"""
    centers = np.zeros((N, 2))
    radii = np.full(N, 0.085)

    # 5x5 grid with one extra
    idx = 0

        for col in range(5):
            if idx >= N:
                break
            x = 0.1 + col * 0.18
            y = 0.1 + row * 0.18
            centers[idx] = [x, y]
            idx += 1
        if idx >= N:
            break

        centers[idx] = [0.5, 0.92]
        idx += 1

        centers[idx] = [0.5 + 0.05 * idx, 0.5]
        idx += 1

    return centers, radii

    cons = [

    best_centers = None
    best_radii = None
    best_sum = 0.0

    # Try multiple initial configurations
    init_funcs = [init_hexagonal_packing, init_grid_packing]
    all_starts = []

    for init_func in init_funcs:
        c, r = init_func()
        all_starts.append(np.concatenate([c.ravel(), r]))

    if best_centers is not None:

    return best_centers, best_radii, best_sum
