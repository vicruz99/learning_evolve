1. Boundary violations (circles crossing x=0, x=1, y=0, y=1)
2. Pairwise overlaps (distance between centers < 2*r)

import numpy as np
from scipy.optimize import minimize

def packing_cost(params, n):
    """
    Computes the negative sum of radii plus penalty terms for constraint violations.
    """
    centers = params[:2 * n].reshape(n, 2)
    r = params[2 * n]
    
    # Base objective: maximize sum of radii => minimize -n*r
    cost = -n * r
    
    # Boundary violation penalties
    for i in range(n):
        x, y = centers[i]
        cost += 100000.0 * max(0, r - x) ** 2
        cost += 100000.0 * max(0, r - y) ** 2
        cost += 100000.0 * max(0, x + r - 1) ** 2
        cost += 100000.0 * max(0, y + r - 1) ** 2
        
    # Pairwise overlap penalties
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
            overlap = 2 * r - dist
            if overlap > 0:
                cost += 1000000.0 * overlap ** 2
                
    return cost

def run_packing():
    np.random.seed(42)
    n = 26
    best_res = None
    best_cost = np.inf
    
    # Multiple restarts with structured hexagonal initialization + jitter
    for restart in range(12):
        x = np.empty(n)
        y = np.empty(n)
        idx = 0
        
        # Hexagonal-ish grid pattern to cover the square efficiently
        for row in range(6):
            ncols = 6 if row % 2 == 0 else 5
            for col in range(ncols):
                if idx >= n:
                    break
                x[idx] = 0.5 + (col - (ncols - 1) / 2) * 0.22
                y[idx] = 0.5 + (row - 2.5) * 0.19
                idx += 1
                
        # Add controlled randomness to escape symmetric local minima
        x += np.random.uniform(-0.025, 0.025, n)
        y += np.random.uniform(-0.025, 0.025, n)
        
        # Initial parameter vector: [x1, y1, ..., x26, y26, r]
        x0 = np.concatenate([np.column_stack((x, y)).flatten(), [0.09]])
        bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)]
        
        res = minimize(
            packing_cost, 
            x0, 
            args=(n,), 
            method='L-BFGS-B', 
            bounds=bounds,
            options={'maxiter': 5000, 'ftol': 1e-12, 'gtol': 1e-10}
        )
        
        if res.fun < best_cost:
            best_cost = res.fun
            best_res = res
            
    centers = best_res.x[:2 * n].reshape(n, 2)
    r = best_res.x[2 * n]
    radii = np.full(n, r)
    
    return centers, radii, float(np.sum(radii))
