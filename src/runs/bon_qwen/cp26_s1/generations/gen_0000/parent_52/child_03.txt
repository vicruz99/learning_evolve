import numpy as np
from scipy.optimize import minimize

def compute_objective(x, n):
    """Minimize negative sum of radii to maximize sum of radii."""
    return -np.sum(x[2*n:])

def compute_constraints(x, n, triu_idx):
    """Compute all boundary and non-overlap constraints as >= 0."""
    c = x[:2*n].reshape(n, 2)
    r = x[2*n:]
    
    # Boundary constraints: c - r >= 0, 1 - c - r >= 0
    b1 = c - r[:, np.newaxis]
    b2 = 1.0 - c - r[:, np.newaxis]
    bounds_flat = np.concatenate([b1.flatten(), b2.flatten()])
    
    # Overlap constraints: dist(i,j) - (r_i + r_j) >= 0
    diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
    dist = np.sqrt(np.sum(diff**2, axis=2))
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    
    # Extract only upper triangle constraints using precomputed indices
    overlaps_flat = dist[triu_idx[0], triu_idx[1]] - r_sum[triu_idx[0], triu_idx[1]]
    
    return np.concatenate([bounds_flat, overlaps_flat])

def run_packing():
    n = 26
    # Precompute indices for pairwise constraints to speed up solver evaluations
    triu_idx = np.triu_indices(n, k=1)
    
    # Initialization: Hexagonal grid pattern
    r_init = 0.06
    centers = []
    radii = []
    idx = 0
    for row in range(6):
        for col in range(6):
            if idx >= n:
                break
            x = r_init + col * 2 * r_init + (row % 2) * r_init
            y = r_init + row * np.sqrt(3) * r_init
            if x + r_init <= 1.0 and y + r_init <= 1.0:
                centers.append([x, y])
                radii.append(r_init)
                idx += 1
        if idx >= n:
            break
            
    # Fill remaining slots if grid doesn't cover all 26
    while len(centers) < n:
        centers.append([0.5, 0.5])
        radii.append(0.01)
        
    centers = np.array(centers[:n])
    radii = np.array(radii[:n])
    
    # Flatten variables: [x0, y0, x1, y1, ..., r0, r1, ...]
    x0 = np.concatenate([centers.flatten(), radii])
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
    
    constraint = {'type': 'ineq', 'fun': compute_constraints}
    
    # Run SLSQP optimizer
    res = minimize(
        compute_objective, 
        x0, 
        method='SLSQP', 
        bounds=bounds, 
        constraints=constraint, 
        args=(n, triu_idx),
        options={'maxiter': 1500, 'ftol': 1e-12, 'disp': False}
    )
                   
    final_centers = res.x[:2*n].reshape(n, 2)
    final_radii = res.x[2*n:]
    
    # Ensure physical bounds
    final_radii = np.maximum(final_radii, 0.0)
    final_centers = np.clip(final_centers, 0.0, 1.0)
    
    # Safety scaling to guarantee strict validity against numerical precision limits
    scale = 1.0
    for _ in range(100):
        c = final_centers
        r = final_radii * scale
        # Check boundaries
        if np.any(c - r[:, None] < -1e-9) or np.any(1.0 - c - r[:, None] < -1e-9):
            scale *= 0.99
            continue
        # Check overlaps
        diff = c[:, None, :] - c[None, :, :]
        dist = np.sqrt(np.sum(diff**2, axis=2))
        r_sum = r[:, None] + r[None, :]
        if np.any(dist - r_sum < -1e-9):
            scale *= 0.99
            continue
        break
    final_radii *= scale
    
    sum_r = float(np.sum(final_radii))
    return final_centers, final_radii, sum_r
