import numpy as np
from scipy.optimize import minimize
import math

def run_packing():
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    Uses a hexagonal lattice initialization followed by SLSQP optimization.
    """
    n_circles = 26
    
    # --- 1. Initialization ---
    # We start with a hexagonal packing layout.
    # We estimate a starting radius. For N=26, r ~ 0.1 is a good guess.
    # Hexagonal packing has row spacing of sqrt(3)/2 * diameter = sqrt(3) * r.
    # Horizontal spacing is 2 * r.
    
    # Let's try to fit them in rows. 
    # 26 circles. Maybe 6 rows? 4, 5, 4, 5, 4, 4 (sum 26) or similar.
    # Or just place them on a grid and let the optimizer fix it.
    
    # A simple grid approach for initialization:
    # 5x5 grid is 25 circles. We need 1 more.
    # Let's try a 5x6 grid (30 spots) and pick the best 26? 
    # Or just a dense random perturbation of a grid.
    
    # Let's generate a hexagonal lattice points list
    points = []
    r_init = 0.10 # Initial guess
    diam = 2 * r_init
    
    # Try to fit rows
    # Row height = sqrt(3)/2 * diam
    row_h = math.sqrt(3)/2 * diam
    
    y = r_init
    row_idx = 0
    while y < 1.0 and len(points) < n_circles:
        # Determine x offset for this row (staggered)
        x_offset = (row_idx % 2) * (diam / 2.0)
        x = r_init + x_offset
        
        # Fill row
        while x < 1.0 and len(points) < n_circles:
            points.append((x, y))
            x += diam
        y += row_h
        row_idx += 1
    
    # If we don't have 26 points (unlikely with r=0.1, but safe check), add them randomly
    if len(points) < n_circles:
        # Fill remaining
        for _ in range(n_circles - len(points)):
            points.append((0.5, 0.5)) # Fallback, optimizer will move them

    # Convert to numpy arrays
    centers = np.array(points[:n_circles])
    radii = np.full(n_circles, r_init)
    
    # Ensure initial feasibility roughly (clipping to bounds)
    # Though optimizer handles constraints, good start helps.
    
    # --- 2. Optimization ---
    
    # Variables: x1, y1, r1, x2, y2, r2, ...
    # Total variables: 3 * n_circles
    x0 = np.hstack([centers.ravel(), radii])
    
    # Bounds for variables
    # x, y in [0, 1]
    # r in [0, 0.5] (max radius in unit square)
    bounds = []
    for _ in range(n_circles):
        bounds.extend([
            (0.0, 1.0), # x
            (0.0, 1.0), # y
            (0.0, 0.5)  # r
        ])
    
    def objective(vars):
        # Maximize sum of radii => Minimize negative sum
        r = vars[2::3]
        return -np.sum(r)
    
    # Constraints
    constraints = []
    
    # 1. Boundary constraints
    # For each circle i: r <= x <= 1-r  =>  x - r >= 0, 1 - x - r >= 0
    # Same for y
    for i in range(n_circles):
        idx_x = 3 * i
        idx_y = 3 * i + 1
        idx_r = 3 * i + 2
        
        # x >= r  => x - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: v[idx_x] - v[idx_r]
        })
        # 1 - x - r >= 0 => x + r <= 1
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: 1.0 - v[idx_x] - v[idx_r]
        })
        # y >= r
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: v[idx_y] - v[idx_r]
        })
        # 1 - y - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: 1.0 - v[idx_y] - v[idx_r]
        })
        
    # 2. Non-overlap constraints
    # dist(i, j) >= r_i + r_j
    # dist^2 >= (r_i + r_j)^2
    # (xi - xj)^2 + (yi - yj)^2 - (ri + rj)^2 >= 0
    for i in range(n_circles):
        for j in range(i + 1, n_circles):
            idx_xi, idx_yi, idx_ri = 3 * i, 3 * i + 1, 3 * i + 2
            idx_xj, idx_yj, idx_rj = 3 * j, 3 * j + 1, 3 * j + 2
            
            def no_overlap(v, i=i, j=j):
                xi, yi, ri = v[idx_xi], v[idx_yi], v[idx_ri]
                xj, yj, rj = v[idx_xj], v[idx_yj], v[idx_rj]
                dist_sq = (xi - xj)**2 + (yi - yj)**2
                sum_r = ri + rj
                return dist_sq - sum_r**2
            
            constraints.append({
                'type': 'ineq',
                'fun': no_overlap
            })
            
    # Run optimization
    # Using SLSQP. It might take a few seconds for 78 variables and ~400 constraints.
    # We can try to limit iterations or use a robust method.
    try:
        res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints, 
                       options={'maxiter': 1000, 'ftol': 1e-9})
        x_opt = res.x
    except Exception:
        # Fallback if optimization fails
        x_opt = x0

    # Extract results
    centers_opt = x_opt[:2*n_circles].reshape(n_circles, 2)
    radii_opt = x_opt[2*n_circles:]
    
    # Final validation and clamping just in case
    # Sometimes numerical errors leave radii slightly negative or out of bounds
    # But constraints should handle it. 
    # Let's ensure radii are non-negative.
    radii_opt = np.maximum(radii_opt, 0.0)
    
    # Re-check boundaries strictly to be safe
    for i in range(n_circles):
        x, y = centers_opt[i]
        r = radii_opt[i]
        # Clamp center to be at least r away from boundary? 
        # Actually, the constraint was x >= r. 
        # If optimizer found x < r, it violated constraint.
        # We assume it respects it.
        
    sum_radii = np.sum(radii_opt)
    
    return centers_opt, radii_opt, sum_radii
