import numpy as np
from scipy.optimize import minimize
import math

def run_packing():
    """
    Optimize the packing of 26 circles in a unit square to maximize sum of radii.
    Returns (centers, radii, sum_radii).
    """
    n = 26
    
    # Initialize centers in a 5x5 grid + 1, slightly perturbed
    # 5x5 grid centers at 0.1, 0.3, 0.5, 0.7, 0.9
    # This gives radius 0.1. We want to expand.
    # Let's create a 5x5 grid and add one in a gap or just random valid positions
    # A 5x5 grid has 25 points. We need 26.
    # Let's place 26 points roughly uniformly or in a hexagonal pattern.
    
    # Hexagonal packing initialization is often better for density
    # Let's try to fit a block of hexagonal points
    # Row spacing sqrt(3)/2 * 2r ~ 1.732r
    # Horizontal spacing 2r
    
    # Let's start with a random valid configuration or a dense grid
    # A 6x5 grid? No.
    # Let's just use a 5x5 grid and place the 26th circle in the center of a gap, 
    # then optimize everything.
    
    centers = []
    # 5x5 grid
    step = 0.2
    offset = 0.1
    for i in range(5):
        for j in range(5):
            centers.append([offset + i * step, offset + j * step])
    
    # 26th circle: place it at a center of a square gap, e.g., (0.2, 0.2) is taken? 
    # Grid points are (0.1, 0.1), (0.3, 0.1)...
    # Gap center is (0.2, 0.2). Distance to (0.1, 0.1) is sqrt(0.02) ~ 0.141.
    # If radii are 0.1, sum is 0.2. 0.141 < 0.2, so overlap.
    # But we will shrink/expand during optimization.
    # Let's place it at (0.2, 0.2) initially with a small radius.
    centers.append([0.2, 0.2])
    
    centers = np.array(centers)
    radii = np.full(n, 0.1) # Start with 0.1, will adjust
    
    # Objective: Maximize sum(radii) -> Minimize -sum(radii)
    # Constraints handled via penalty in objective function
    
    def objective(params):
        # params: [x0, y0, r0, x1, y1, r1, ...]
        # Shape: 3 * n
        centers_opt = params[:2*n].reshape(n, 2)
        radii_opt = params[2*n:]
        
        # Penalty for negative radii
        penalty = 0.0
        if np.any(radii_opt < 0):
            return 1e9 # Big penalty
        
        # Sum of radii to maximize
        sum_r = np.sum(radii_opt)
        
        # Boundary constraints: x-r >= 0, x+r <= 1, y-r >= 0, y+r <= 1
        # Equivalent to r <= x, r <= 1-x, r <= y, r <= 1-y
        # Penalty if violated
        dist_to_bounds = np.min([
            centers_opt[:, 0], 
            1 - centers_opt[:, 0], 
            centers_opt[:, 1], 
            1 - centers_opt[:, 1]
        ], axis=0)
        
        boundary_violation = np.maximum(0, radii_opt - dist_to_bounds)
        penalty += np.sum(boundary_violation ** 2) * 1e4
        
        # Overlap constraints: dist(i, j) >= r_i + r_j
        # Penalty if dist < r_i + r_j
        # Calculate all pairwise distances
        # This is O(N^2)
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers_opt[i] - centers_opt[j]) ** 2))
                required_dist = radii_opt[i] + radii_opt[j]
                if dist < required_dist:
                    overlap = required_dist - dist
                    penalty += overlap ** 2 * 1e5
                    
        return -sum_r + penalty

    # Initial parameters
    x0 = np.concatenate([centers.flatten(), radii])
    
    # Bounds: 
    # x, y in [0, 1]
    # r in [0, 1]
    bounds = []
    for _ in range(n):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 1.0)) # r
    
    # Optimization
    # Use L-BFGS-B as it supports bounds
    result = minimize(objective, x0, method='L-BFGS-B', bounds=bounds, 
                      options={'ftol': 1e-9, 'maxiter': 1000})
    
    # Extract results
    best_params = result.x
    best_centers = best_params[:2*n].reshape(n, 2)
    best_radii = best_params[2*n:]
    
    # Post-processing: ensure non-negative radii and clip to valid range if needed
    # Due to penalty function, radii might be slightly negative or > 1, but should be fine.
    best_radii = np.maximum(best_radii, 0.0)
    
    # Refine centers to be strictly inside based on radii (clamping)
    # This is a safety step, though optimization should have handled it.
    # Actually, if we clamp, we might introduce overlaps, but it's better for validity.
    # The penalty function pushes radii down if boundaries are hit, so radii should be consistent.
    
    # Calculate sum
    total_sum = np.sum(best_radii)
    
    return best_centers, best_radii, total_sum
