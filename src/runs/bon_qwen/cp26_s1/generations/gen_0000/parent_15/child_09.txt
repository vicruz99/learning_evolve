def solve_radii_lp(centers):
    """
    Solves for the maximum sum of radii given fixed center positions using Linear Programming.
    
    Args:
        centers: np.array of shape (n, 2)
        
    Returns:
        radii: np.array of shape (n)
        sum_radii: float
    """
    n = centers.shape[0]
    
    # Objective: maximize sum(r) => minimize -sum(r)
    c = -np.ones(n)
    
    # Build constraints matrix A_ub and vector b_ub
    # Constraints: r_i + r_j <= dist(i,j), and r_i <= dist_to_boundary(i)
    
    A_ub_rows = []
    b_ub_vals = []
    
    # Pairwise non-overlap constraints
    for i in range(n):

            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub_rows.append(row)
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            b_ub_vals.append(dist)
            
    # Boundary constraints (left, right, bottom, top)
    for i in range(n):
        x, y = centers[i]

        row = np.zeros(n); row[i] = 1.0
