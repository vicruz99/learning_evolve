def calculate_violations(centers, radii):

    n = len(radii)

        r = radii[i]
        
        # Left
        if x < r:
            violations += (x - r)**2

            violations += (x + r - 1)**2
        # Bottom
        if y < r:
            violations += (y - r)**2

            min_dist = radii[i] + radii[j]
            if dist < min_dist:
                violations += (min_dist - dist)**2
                
    return violations
