def compute_optimal_radii(centers):
    """Given fixed centers, find optimal radii using LP.
    
    Maximizes sum(r_i) subject to:
    - r_i <= min(x_i, 1-x_i, y_i, 1-y_i) for boundary
    - r_i + r_j <= dist(c_i, c_j) for non-overlap
    - r_i >= 0
    """
    n_c = centers.shape[0]
    c_obj = np.ones(n_c) * (-1)
    
    A_list = []
    b_list = []
    
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n_c):
        row = np.zeros(n_c)
        row[i] = 1
        for val in [centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1]]:
            A_list.append(row.copy())
            b_list.append(max(0, val))
    
    # Pairwise constraints: r_i + r_j <= dist(c_i, c_j)
    for i in range(n_c):
        for j in range(i + 1, n_c):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            row = np.zeros(n_c)
            row[i] = 1
            row[j] = 1
            A_list.append(row)
            b_list.append(max(0, dist))
    
    A = np.array(A_list)
    b = np.array(b_list)
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)] * n_c, method='highs')
        if res.success and -res.fun > 1e-10:
            return -res.fun, res.x
    except Exception:
        pass
    
    # Fallback
    radii = np.ones(n_c) * 0.01
    return np.sum(radii), radii


def create_hexagonal_init(n, config, spacing, base_x, base_y, offset_x=0, offset_y=0):
    """Create a hexagonal grid initialization."""
    centers = np.zeros((n, 2))
    idx = 0
    
    for row_i, row_size in enumerate(config):
        y = base_y + offset_y + row_i * spacing * np.sqrt(3) / 2

        for col_i in range(row_size):
            x = base_x + x_offset + col_i * spacing
            # Clamp to valid range
            x = np.clip(x, 0.01, 0.99)
            y = np.clip(y, 0.01, 0.99)
            centers[idx] = [x, y]
            idx += 1
    
    return centers
