import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    def compute_energy(centers, r):
        """Compute overlap and boundary violation energy."""
        # centers: (n, 2)
        # Pairwise distances using broadcasting
        diff = centers[:, None, :] - centers[None, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        
        # Extract upper triangle (unique pairs)
        mask = np.triu(np.ones((n, n), dtype=bool), k=1)
        pair_dists = dists[mask]
        
        # Overlap penalty
        overlap = np.maximum(0.0, 2.0 * r - pair_dists)
        e = np.sum(overlap**2)
        
        # Boundary penalties: center must be in [r, 1-r]
        e += np.sum(np.maximum(0.0, r - centers[:, 0])**2)
        e += np.sum(np.maximum(0.0, centers[:, 0] - (1.0 - r))**2)
        e += np.sum(np.maximum(0.0, r - centers[:, 1])**2)
        e += np.sum(np.maximum(0.0, centers[:, 1] - (1.0 - r))**2)
        return e

    def objective(vars, r):
        centers = vars.reshape((n, 2))
        return compute_energy(centers, r)

    # Deterministic seed for reproducibility
    np.random.seed(42)
    
    # Initialize with a 5x5 grid plus the center
    grid = np.linspace(0.2, 0.8, 5)
    centers = np.array([[x, y] for x in grid for y in grid])
    centers = np.vstack([centers, [0.5, 0.5]])
    centers += np.random.uniform(-0.02, 0.02, centers.shape)
    centers = np.clip(centers, 0.1, 0.9)
    
    best_centers = centers.copy()
    best_r = 0.10
    
    # Binary search for the maximum feasible radius
    r_low, r_high = 0.10, 0.115
    for _ in range(40):
        r_mid = (r_low + r_high) / 2.0
        
        # Primary optimization
        res = minimize(objective, best_centers.flatten(), args=(r_mid,),
                       method='L-BFGS-B', bounds=[(0.0, 1.0)]*(2*n),
                       options={'maxiter': 1000, 'ftol': 1e-12, 'gtol': 1e-8})
                       
        if res.fun < 1e-8:
            r_low = r_mid
            best_centers = res.x.reshape((n, 2))
            best_r = r_mid
        else:
            # Escape local minima with perturbation
            pert = best_centers + np.random.uniform(-0.015, 0.015, best_centers.shape)
            pert = np.clip(pert, 0.05, 0.95)
            res2 = minimize(objective, pert.flatten(), args=(r_mid,),
                            method='L-BFGS-B', bounds=[(0.0, 1.0)]*(2*n),
                            options={'maxiter': 500})
            if res2.fun < 1e-8:
                r_low = r_mid
                best_centers = res2.x.reshape((n, 2))
                best_r = r_mid
            else:
                r_high = r_mid
                
    radii = np.full(n, best_r)
    return best_centers, radii, float(best_r * n)
