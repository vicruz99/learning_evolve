def calculate_min_dist_and_wall(centers, width=1.0, height=1.0):
    """
    Calculates the minimum pairwise distance and minimum distance to wall.
    Returns min_dist (between centers) and min_wall_dist (from center to boundary).
    Actually for circle packing with radius r, we need:
    pairwise dist >= 2r
    center dist to wall >= r
    So we track d_pair = min pairwise dist
    and d_wall = min center-to-wall dist.
    Then 2r <= d_pair and r <= d_wall => r <= min(d_pair/2, d_wall).
    Let's return a metric 'bottleneck' = min(d_pair/2, d_wall).
    """
    min_pair_dist = float('inf')
    min_wall_dist = float('inf')
    
    n = centers.shape[0]
    
    # Pairwise distances
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, ] - centers[j, 1]
            dist = math.sqrt(dx*dx + dy*dy)
            if dist < min_pair_dist:
                min_pair_dist = dist
                
        # Wall distances
        x, y = centers[i]
        dist_left = x
        dist_right = width - x
        dist_down = y
        dist_up = height - y
        d_wall = min(dist_left, dist_right, dist_down, dist_up)
        if d_wall < min_wall_dist:
            min_wall_dist = d_wall
            
    return min_pair_dist, min_wall_dist
