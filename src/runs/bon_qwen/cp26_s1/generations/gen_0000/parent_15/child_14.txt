def generate_hexagonal_grid(n_circles, width=1.0, height=1.0):
    """
    Generates initial positions for circles in a hexagonal grid pattern.
    """

    # Approximate density for hex packing ~ 0.9. Area ~ 0.9.
    # n * pi * r^2 ~ 0.9 => r ~ sqrt(0.9 / (n*pi))

                col += 1

                break
        row += 1
        
    return np.array(centers[:n_circles]), r_init

def get_variables(centers, radii):
    """
    Transforms physical variables (centers, radii) to optimization variables (u, v, r).
    x = r + (1-2r)u => u = (x - r) / (1 - 2r)

            v = (centers[i, 1] - r) / denom
        vars[3*i] = u
        vars[3*i+1] = v
        vars[3*i+2] = r
    return vars

def get_physical(vars):

        radii[i] = r
        centers[i, 0] = r + (1 - 2 * r) * u
        centers[i, 1] = r + (1 - 2 * r) * v
    return centers, radii

def objective(vars):
    """
    Objective function: minimize negative sum of radii.
    """
    _, radii = get_physical(vars)
    return -np.sum(radii)

def constraint_overlap(vars, i, j):
    """
    Defines the non-overlap constraint for pair (i, j).
    (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
    """
    centers, radii = get_physical(vars)
    dx = centers[i, 0] - centers[j, 0]
    dy = centers[i, 1] - centers[j, 1]
    dist_sq = dx**2 + dy**2
    r_sum = radii[i] + radii[j]
    return dist_sq - r_sum**2

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square

    Args:
        centers: np.array of shape (n, 2) with (x, y) coordinates
        radii: np.array of shape (n) with radius of each circle

    Returns:
        True if valid, False otherwise
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True
