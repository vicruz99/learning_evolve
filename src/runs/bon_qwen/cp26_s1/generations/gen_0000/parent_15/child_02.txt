import numpy as np
from scipy.optimize import linprog

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    n = centers.shape[0]

    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True

def calculate_max_radii(centers):
    """Solves LP to find radii maximizing sum for fixed centers."""
    n = centers.shape[0]
    c_obj = np.ones(n) * -1
    
    A_ub = []
    b_ub = []

    # Pairwise constraints: r_i + r_j <= dist
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            row = np.zeros(n)
            row[i] = 1
            row[j] = 1
            A_ub.append(row)
            b_ub.append(dist)

    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, etc.
    for i in range(n):
        x, y = centers[i]
        
        row = np.zeros(n); row[i] = 1
        A_ub.append(row); b_ub.append(x)
        
        row = np.zeros(n); row[i] = 1
        A_ub.append(row); b_ub.append(1 - x)
        
        row = np.zeros(n); row[i] = 1
        A_ub.append(row); b_ub.append(y)
        
        row = np.zeros(n); row[i] = 1
        A_ub.append(row); b_ub.append(1 - y)

    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    bounds = [(0, None) for _ in range(n)]

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return -res.fun, -res.x
        else:
            return 0.0, np.zeros(n)
    except Exception:
        return 0.0, np.zeros(n)

def repulsion_refine(centers, radii, steps=200):
    """Refines positions to resolve overlaps and maximize radii."""
    n = centers.shape[0]
    k_rep = 10.0
    k_attr = 0.1
    lr = 0.005

    current_centers = centers.copy()
    current_radii = radii.copy()

    for _ in range(steps):
        forces = np.zeros_like(current_centers)
        
        for i in range(n):
            # Boundary forces
            r = current_radii[i]
            if current_centers[i, 0] < r: forces[i, 0] += k_attr * (r - current_centers[i, 0])
            elif current_centers[i, 0] > 1 - r: forces[i, 0] -= k_attr * (current_centers[i, 0] - (1 - r))
            
            if current_centers[i, 1] < r: forces[i, 1] += k_attr * (r - current_centers[i, 1])
            elif current_centers[i, 1] > 1 - r: forces[i, 1] -= k_attr * (current_centers[i, 1] - (1 - r))

            for j in range(i + 1, n):
                vec = current_centers[i] - current_centers[j]
                dist = np.linalg.norm(vec)
                req = current_radii[i] + current_radii[j]
                
                if dist < req + 1e-6:
                    overlap = req - dist
                    if dist > 1e-9:
                        f = k_rep * (overlap / dist) * vec
                    else:
                        f = k_rep * 0.1 * (np.random.rand(2) - 0.5)
                    forces[i] += f
                    forces[j] -= f
        
        current_centers += lr * forces
        current_centers = np.clip(current_centers, 1e-6, 1 - 1e-6)
        
        # Slightly shrink radii if overlaps persist, grow if free
        # Recalculate optimal radii for these centers occasionally or just enforce
        # For simplicity, enforce min radius
        for i in range(n):
             current_radii[i] = np.clip(current_radii[i], 0, 1)

    return current_centers, current_radii
