import numpy as np
import scipy.optimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    
    Returns:
        centers: np.array of shape (26, 2)
        radii: np.array of shape (26,)
        sum_radii: float
    """
    n = 26
    
    # --- 1. Initialization ---
    # Start with a hexagonal packing pattern to provide a good initial guess.
    # We place circles in rows.
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.05) # Small initial radius to ensure no overlaps
    
    # Parameters for hexagonal grid
    # We want to fit 26 circles. 
    # Let's try 5 rows. 
    # Row lengths: 5, 4, 5, 4, 5, 3? Sum = 26.
    # Or 6, 5, 5, 5, 5?
    # Let's just fill a dense grid.
    
    idx = 0
    row = 0
    while idx < n:
        # Estimate radius for spacing
        r_curr = 0.05
        # Vertical spacing for hex grid is sqrt(3)/2 * diameter = sqrt(3) * r
        # But since r is small, we can just pack them tight
        y = r_curr + row * np.sqrt(3) * r_curr
        
        # Horizontal spacing is 2 * r
        # Shift alternate rows by r
        x_start = r_curr
        if row % 2 == 1:
            x_start += r_curr
            
        # How many circles fit in this row?
        # Width available is 1 - 2*r_curr
        # Spacing 2*r_curr
        # Number of circles approx (1 - 2r) / 2r + 1 = 1/2r - 1 + 1 = 1/2r = 10 for r=0.05
        # Let's just place as many as needed or fit
        
        count_in_row = 0
        x = x_start
        while x <= 1 - r_curr and idx < n:
            centers[idx] = [x, y]
            idx += 1
            count_in_row += 1
            x += 2 * r_curr
        
        row += 1
        
    # If we didn't fill 26 (unlikely with r=0.05), fill randomly
    while idx < n:
        # Random position away from others? Just random in center
        centers[idx] = [0.5, 0.5] 
        idx += 1

    # Flatten variables for optimizer: [x1, y1, r1, x2, y2, r2, ...]
    # Initial guess
    x0 = np.zeros(3 * n)
    for i in range(n):
        x0[3*i] = centers[i, 0]
        x0[3*i + 1] = centers[i, 1]
        x0[3*i + 2] = radii[i]
        
    # Bounds: x, y in [0, 1], r in [0, 0.5]
    bounds = []
    for _ in range(n):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 0.5)) # r

    # --- 2. Constraints Definition ---
    def constraints_func(vars_flat):
        centers_opt = vars_flat[:2*n].reshape(n, 2)
        radii_opt = vars_flat[2*n:]
        
        constraints = []
        
        # Boundary constraints
        # x - r >= 0
        constraints.append(centers_opt[:, 0] - radii_opt)
        # 1 - x - r >= 0
        constraints.append(1.0 - centers_opt[:, 0] - radii_opt)
        # y - r >= 0
        constraints.append(centers_opt[:, 1] - radii_opt)
        # 1 - y - r >= 0
        constraints.append(1.0 - centers_opt[:, 1] - radii_opt)
        
        # Overlap constraints: dist^2 - (r_i + r_j)^2 >= 0
        # We compute all pairs
        # To be efficient, we can vectorize, but for n=26, loop is fine or broadcasting
        # Broadcasting:
        # c_i - c_j shape (n, n, 2)
        # diff = centers_opt[:, np.newaxis, :] - centers_opt[np.newaxis, :, :]
        # dist_sq = np.sum(diff**2, axis=2) # shape (n, n)
        # r_sum_sq = (radii_opt[:, np.newaxis] + radii_opt[np.newaxis, :])**2
        # constraint_matrix = dist_sq - r_sum_sq
        # We only need upper triangle (i < j)
        
        # Efficient vectorized calculation
        diff = centers_opt[:, np.newaxis, :] - centers_opt[np.newaxis, :, :]
        dist_sq = np.sum(diff**2, axis=2)
        r_sum = radii_opt[:, np.newaxis] + radii_opt[np.newaxis, :]
        overlap_constraints = dist_sq - r_sum**2
        
        # Extract upper triangle
        triu_indices = np.triu_indices(n, k=1)
        overlap_vals = overlap_constraints[triu_indices]
        constraints.append(overlap_vals)
        
        return np.concatenate(constraints)

    cons = {'type': 'ineq', 'fun': constraints_func}

    # --- 3. Optimization ---
    def objective(vars_flat):
        # Maximize sum of radii => Minimize negative sum
        r = vars_flat[2*n:]
        return -np.sum(r)

    # Run optimization
    # SLSQP is suitable for this non-linear constrained problem
    res = scipy.optimize.minimize(
        objective,
        x0,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'ftol': 1e-9, 'maxiter': 1000, 'disp': False}
    )
    
    # Extract results
    best_centers = res.x[:2*n].reshape(n, 2)
    best_radii = res.x[2*n:]
    
    # Clip radii to non-negative just in case of numerical noise
    best_radii = np.maximum(best_radii, 0.0)
    
    sum_radii = np.sum(best_radii)
    
    return best_centers, best_radii, sum_radii
