import numpy as np

def create_hex_grid(n, width=1.0, height=1.0):
    """Generates initial centers for a hexagonal packing of n circles."""
    # Estimate radius for n circles in a square to set spacing
    # Area approx: n * pi * r^2 ~ 0.9 * 1.  r ~ sqrt(0.9 / (n*pi))
    r_est = np.sqrt(0.9 / (n * np.pi))
    d = 2 * r_est
    rows = int(np.ceil(np.sqrt(n * 2 / np.sqrt(3))))
    
    centers = []
    y = r_est # Start with radius margin
    row_idx = 0
    while len(centers) < n:
        if row_idx % 2 == 0:
            x = r_est
            # Shift even rows
            x_start = r_est
            num_cols = int((width - 2*r_est) / d) + 1
            for k in range(num_cols):
                if len(centers) >= n: break
                cx = x_start + k * d
                if cx + r_est <= width + 1e-9:
                    centers.append([cx, y])
        else:
            x = r_est + d/2
            # Shift odd rows
            x_start = r_est + d/2
            num_cols = int((width - 2*r_est) / d) + 1
            for k in range(num_cols):
                if len(centers) >= n: break
                cx = x_start + k * d
                if cx + r_est <= width + 1e-9:
                    centers.append([cx, y])
        
        y += d * np.sqrt(3) / 2
        row_idx += 1
        
    return np.array(centers[:n])

def objective_and_constraints(vars, n=26):
    """
    Computes the objective (negative sum of radii) and constraints.
    vars structure: [x1, y1, r1, x2, y2, r2, ..., x26, y26, r26]
    """
    centers = vars[:2*n].reshape(n, 2)
    radii = vars[2*n:]
    
    # Objective: maximize sum of radii => minimize negative sum
    obj = -np.sum(radii)
    
    constraints_list = []
    
    # 1. Boundary constraints: r <= x <= 1-r => r - x <= 0 and x + r - 1 <= 0
    # For each circle i:
    # r_i - x_i <= 0
    # r_i - y_i <= 0
    # x_i + r_i - 1 <= 0
    # y_i + r_i - 1 <= 0
    for i in range(n):
        cx, cy = centers[i]
        r = radii[i]
        constraints_list.append(r - cx)       # Left boundary
        constraints_list.append(r - cy)       # Bottom boundary
        constraints_list.append(cx + r - 1)   # Right boundary
        constraints_list.append(cy + r - 1)   # Top boundary
        
    # 2. Non-overlap constraints: dist(c_i, c_j) >= r_i + r_j
    # dist^2 >= (r_i + r_j)^2
    # (r_i + r_j)^2 - dist^2 <= 0
    # Using squared distance to avoid sqrt and keep differentiability smooth (mostly)
    # However, (r_i+r_j)^2 - dist^2 <= 0 is equivalent to dist >= r_i + r_j for non-negative r
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist_sq = dx*dx + dy*dy
            r_sum = radii[i] + radii[j]
            # Constraint: r_sum^2 - dist_sq <= 0
            constraints_list.append(r_sum**2 - dist_sq)
            
    return obj, constraints_list

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square

    Args:
        centers: np.array of shape (n, 2) with (x, y) coordinates
        radii: np.array of shape (n) with radius of each circle

    Returns:
        True if valid, False otherwise
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True
