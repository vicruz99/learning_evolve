import numpy as np
from scipy.optimize import minimize

def compute_penalized_objective(vars, n, P):
    """
    Computes the negative sum of radii plus penalty terms for constraint violations.
    vars: 1D array of length 2n + n containing [x0, y0, ..., xn, yn, r0, ..., rn]
    """
    pos = vars[:2*n].reshape(n, 2)
    rad = vars[2*n:]
    
    # Objective: maximize sum of radii => minimize negative sum
    obj = -np.sum(rad)
    
    # Boundary violations: circle must stay inside [0,1]x[0,1]
    b1 = rad - pos[:, 0]          # x - r >= 0  => violation if r - x > 0
    b2 = pos[:, 0] + rad - 1.0    # x + r <= 1 => violation if x + r - 1 > 0
    b3 = rad - pos[:, 1]          # y - r >= 0
    b4 = pos[:, 1] + rad - 1.0    # y + r <= 1
    b5 = -rad                     # r >= 0     => violation if -r > 0
    
    for b in [b1, b2, b3, b4, b5]:
        obj += P * np.sum(np.maximum(b, 0.0)**2)
        
    # Overlap violations: distance between centers >= sum of radii
    diff = pos[:, np.newaxis, :] - pos[np.newaxis, :, :]
    dist_mat = np.sqrt(np.sum(diff**2, axis=2))
    rad_sum_mat = rad[:, np.newaxis] + rad[np.newaxis, :]
    overlap = rad_sum_mat - dist_mat
    
    # Only consider upper triangle to avoid double counting and self-pairing
    i_idx, j_idx = np.triu_indices(n, 1)
    ov_upper = overlap[i_idx, j_idx]
    obj += P * np.sum(np.maximum(ov_upper, 0.0)**2)
        
    return obj

def run_packing():
    n = 26
    # Box constraints for optimization: x,y in [0,1], r in [0, 0.5]
    bounds = [(0, 1)] * (2*n) + [(0, 0.5)] * n

    best_sum = -1.0
    
    # Multiple random restarts to escape local minima
    for restart in range(3):
        np.random.seed(restart * 100 + 7)
        # Initialize positions in the central region to avoid immediate boundary conflicts
        pos = np.random.rand(n, 2) * 0.6 + 0.2
        rad = np.full(n, 0.05)  # Start with small radii
        x0 = np.concatenate([pos.flatten(), rad])
        
        # Sequential penalty increase (annealing-like)
        P = 10
        for stage in range(5):
            res = minimize(compute_penalized_objective, x0, args=(n, P), method='L-BFGS-B',
                           bounds=bounds, options={'maxiter': 500, 'ftol': 1e-10, 'gtol': 1e-8})
            x0 = res.x
            P *= 10
            
        # Final high-precision refinement stage
        res = minimize(compute_penalized_objective, x0, args=(n, 100000), method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': 1500, 'ftol': 1e-13, 'gtol': 1e-11})
        x0 = res.x
        
        final_pos = x0[:2*n].reshape(n, 2)
        final_rad = np.maximum(x0[2*n:], 0.0)
        s = np.sum(final_rad)
        
        # Validate against constraints with a small safety margin
        valid = True
        # Boundary check
        if np.any(final_pos[:, 0] - final_rad < -1e-9) or np.any(final_pos[:, 0] + final_rad > 1 + 1e-9) or \
           np.any(final_pos[:, 1] - final_rad < -1e-9) or np.any(final_pos[:, 1] + final_rad > 1 + 1e-9):
            valid = False
        # Overlap check
        if valid:
            dist_mat = np.sqrt(np.sum((final_pos[:, np.newaxis, :] - final_pos[np.newaxis, :, :])**2, axis=2))

            best_sum = s
            best_result = (final_pos, final_rad)

        best_result = (final_pos, final_rad)
