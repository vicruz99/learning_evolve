import numpy as np
from scipy.optimize import minimize
import math

def compute_objective(v, n=26, C=10000.0):
    """
    Objective function to minimize.
    Minimizes -sum(radii) + C * penalty for constraint violations.
    """
    centers = v[:2 * n].reshape(n, 2)
    radii = v[2 * n:]
    
    # Boundary penalties: circles must stay inside [0,1]x[0,1]
    # r <= x, r <= 1-x, r <= y, r <= 1-y
    p_bound = (
        np.sum(np.maximum(0, radii - centers[:, 0])**2) +
        np.sum(np.maximum(0, radii - (1 - centers[:, 0]))**2) +
        np.sum(np.maximum(0, radii - centers[:, 1])**2) +
        np.sum(np.maximum(0, radii - (1 - centers[:, 1]))**2)
    )
    
    # Overlap penalties: dist >= r_i + r_j
    # Vectorized pairwise distance computation
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist = np.sqrt(np.sum(diff**2, axis=2))
    
    # We only care about pairs i < j
    triu_indices = np.triu_indices(n, k=1)
    dists = dist[triu_indices]
    r_sums = radii[triu_indices[0]] + radii[triu_indices[1]]
    
    p_overlap = np.sum(np.maximum(0, r_sums - dists)**2)
    
    total_penalty = p_bound + p_overlap
    return -np.sum(radii) + C * total_penalty
