from scipy.optimize import minimize

CIRCLE_COUNT = 26

def objective(params):
    """Minimize negative sum of radii."""
    radii = params[2::3]
    return -np.sum(radii)

def get_boundary_constraints(params):
    """Vectorized boundary constraints: circles inside [0,1]^2."""
    n = CIRCLE_COUNT
    xs = params[0::3]
    ys = params[1::3]
    rs = params[2::3]
    
    # Constraints: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    c1 = xs - rs
    c2 = 1.0 - xs - rs
    c3 = ys - rs
    c4 = 1.0 - ys - rs
    
    return np.concatenate([c1, c2, c3, c4])

def get_overlap_constraints(params):
    """Vectorized pairwise non-overlap constraints."""
    n = CIRCLE_COUNT
    xs = params[0::3]
    ys = params[1::3]
    rs = params[2::3]

    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    # Constraint: dist^2 - (r_i + r_j)^2 >= 0
    diff = dx**2 + dy**2 - dr**2
    
    # Return only upper triangle (i < j) to avoid duplicates and self-comparison
    return np.triu(diff, k=1).flatten()

def run_packing():
