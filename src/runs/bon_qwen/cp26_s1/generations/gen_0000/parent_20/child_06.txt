## Strategy for Maximizing Sum of Radii for 26 Circles in Unit Square

### Problem Analysis
We need to pack 26 circles in [0,1]² to maximize Σri. The target is 2.636, implying an average radius of ~0.101.

For equal circles in hexagonal packing, the theoretical upper bound is ~2.74 (via area argument: Σπri² ≈ π/(2√3), maximize Σri). With boundary effects and unequal circles, 2.636 is achievable.

### Key Insights
1. **Hexagonal lattice** is the densest infinite packing (density π/(2√3) ≈ 0.907)
2. **Corner/edge circles can be larger** - boundary effects allow larger radii near corners
3. **Non-convex optimization** - constraint ||ci-cj|| ≥ ri+rj is non-convex, requiring careful handling
4. **Multiple phases needed** - initialization → coarse optimization → fine-tuning

### Approach
**Phase 1: Smart Initialization**
- Hexagonal lattice with 5 rows (6,5,6,5,4 circles)
- Radii sized to fill available space

**Phase 2: Force-Directed Relaxation**
- Repulsion forces when circles overlap
- Growth forces proportional to local free space
- Boundary forces keep circles inside
- Gradual cooling (learning rate decay)

**Phase 3: SLSQP Refinement**
- Use scipy's SLSQP with analytic gradients
- Constraint: d² - (ri+rj)² ≥ 0 (avoids sqrt, has smooth gradient)
- Boundary constraints: xi ≥ ri, xi ≤ 1-ri, etc.

**Phase 4: Multiple Restarts**
- Try different perturbed initializations
- Keep best solution

### Technical Details
- Squared-distance constraints for smoother optimization
- Analytic gradients for all constraints
- functools.partial for constraint callbacks (no lambdas)
- Gradient clipping and bounds for stability

import numpy as np
from scipy.optimize import minimize
from functools import partial


N_CIRCLES = 26


def obj(x):
    """Objective: maximize sum of radii (minimize negative)."""
    return -np.sum(x[2 * N_CIRCLES:])


def obj_grad(x):
    """Gradient of objective."""
    grad = np.zeros(3 * N_CIRCLES)
    grad[2 * N_CIRCLES:] = -1.0
    return grad


def no_overlap_sq(i, j, x):
    """Squared distance constraint: d^2 - (ri+rj)^2 >= 0"""
    xi, yi = x[2 * i], x[2 * i + 1]
    xj, yj = x[2 * j], x[2 * j + 1]
    ri, rj = x[2 * N_CIRCLES + i], x[2 * N_CIRCLES + j]
    return (xi - xj) ** 2 + (yi - yj) ** 2 - (ri + rj) ** 2


def no_overlap_sq_grad(i, j, x):
    """Gradient of squared distance constraint."""
    grad = np.zeros(3 * N_CIRCLES)
    xi, yi = x[2 * i], x[2 * i + 1]
    xj, yj = x[2 * j], x[2 * j + 1]
    ri, rj = x[2 * N_CIRCLES + i], x[2 * N_CIRCLES + j]
    dx = xi - xj
    dy = yi - yj
    rs = ri + rj
    grad[2 * i] = 2.0 * dx
    grad[2 * i + 1] = 2.0 * dy
    grad[2 * j] = -2.0 * dx
    grad[2 * j + 1] = -2.0 * dy
    grad[2 * N_CIRCLES + i] = -2.0 * rs
    grad[2 * N_CIRCLES + j] = -2.0 * rs
    return grad


def bound_x_lo(i, x):
    """xi >= ri"""
    return x[2 * i] - x[2 * N_CIRCLES + i]


def bound_x_lo_grad(i, x):
    grad = np.zeros(3 * N_CIRCLES)
    grad[2 * i] = 1.0
    grad[2 * N_CIRCLES + i] = -1.0
    return grad


def bound_x_hi(i, x):
    """xi <= 1 - ri"""
    return 1.0 - x[2 * i] - x[2 * N_CIRCLES + i]


def bound_x_hi_grad(i, x):
    grad = np.zeros(3 * N_CIRCLES)
    grad[2 * i] = -1.0
    grad[2 * N_CIRCLES + i] = -1.0
    return grad


def bound_y_lo(i, x):
    """yi >= ri"""
    return x[2 * i + 1] - x[2 * N_CIRCLES + i]


def bound_y_lo_grad(i, x):
    grad = np.zeros(3 * N_CIRCLES)
    grad[2 * i + 1] = 1.0
    grad[2 * N_CIRCLES + i] = -1.0
    return grad


def bound_y_hi(i, x):
    """yi <= 1 - ri"""
    return 1.0 - x[2 * i + 1] - x[2 * N_CIRCLES + i]


def bound_y_hi_grad(i, x):
    grad = np.zeros(3 * N_CIRCLES)
    grad[2 * i + 1] = -1.0
    grad[2 * N_CIRCLES + i] = -1.0
    return grad


def make_bounds():
    """Create bounds for all variables."""
    bounds = []
    for _ in range(2 * N_CIRCLES):
        bounds.append((0.0, 1.0))
    for _ in range(N_CIRCLES):
        bounds.append((0.001, 0.5))
    return bounds


def init_hexagonal():
    """Initialize circles in a hexagonal lattice pattern."""
    centers = np.zeros((N_CIRCLES, 2))
    radii = np.ones(N_CIRCLES) * 0.08

    idx = 0
    # 5 rows: 6, 5, 6, 5, 4 = 26 circles
    row_counts = [6, 5, 6, 5, 4]
    y = 0.11
    x_base = 0.11
    dx = 0.155
    dy = 0.14 * np.sqrt(3) / 2.0

    for row_i, nc in enumerate(row_counts):
        x = x_base + (row_i % 2) * dx / 2.0
        for _ in range(nc):
            centers[idx] = [x, y]
            x += dx
            idx += 1
        y += dy

    # Adjust radii based on position (larger near corners/edges)
    for i in range(N_CIRCLES):
        x, y = centers[i]
        dist_to_corner = min(
            np.sqrt(x ** 2 + y ** 2),
            np.sqrt((1 - x) ** 2 + y ** 2),
            np.sqrt(x ** 2 + (1 - y) ** 2),
            np.sqrt((1 - x) ** 2 + (1 - y) ** 2)
        )
        dist_to_edge = min(x, 1 - x, y, 1 - y)
        # Corner/edge circles get larger
        radii[i] = 0.06 + 0.04 * max(0, 1 - dist_to_corner * 3) + 0.02 * max(0, 1 - dist_to_edge * 4)

    return centers, radii


def force_directed(centers, radii, iterations=15000):
    """Force-directed optimization with growth."""
    n = N_CIRCLES
    centers = centers.copy()
    radii = radii.copy()

    for it in range(iterations):
        forces = np.zeros((n, 2))
        growth = np.zeros(n)

        # Compute growth potential (min separation minus current radius)
        for i in range(n):
            min_sep = 10.0
            for j in range(n):
                if i != j:
                    d = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                    min_sep = min(min_sep, d - radii[j])
            min_sep = min(min_sep, centers[i, 0], 1 - centers[i, 0])
            min_sep = min(min_sep, centers[i, 1], 1 - centers[i, 1])
            growth[i] = max(0.0, min_sep - radii[i])

        # Repulsion forces for overlapping/near circles
        for i in range(n):
            for j in range(i + 1, n):
                diff = centers[i] - centers[j]
                d = np.sqrt(np.sum(diff ** 2))
                target = radii[i] + radii[j]
                if d < target + 0.02 and d > 1e-10:
                    overlap = max(0, target + 0.01 - d)
                    f = overlap * diff / d / max(d, 0.001)
                    forces[i] += f
                    forces[j] -= f

        # Boundary forces
        for i in range(n):
            r = radii[i]
            if centers[i, 0] < r + 0.001:
                forces[i, 0] += (r + 0.001 - centers[i, 0]) * 50.0
            if centers[i, 0] > 1 - r - 0.001:
                forces[i, 0] -= (centers[i, 0] - (1 - r - 0.001)) * 50.0
            if centers[i, 1] < r + 0.001:
                forces[i, 1] += (r + 0.001 - centers[i, 1]) * 50.0
            if centers[i, 1] > 1 - r - 0.001:
                forces[i, 1] -= (centers[i, 1] - (1 - r - 0.001)) * 50.0

        # Update with cooling schedule
        lr = 0.05 / (1.0 + it * 0.0003)
        centers += forces * lr * 2.0
        radii += growth * lr * 0.5

        # Enforce bounds
        radii = np.maximum(radii, 0.005)

    return centers, radii


def build_constraints():
    """Build all constraints for SLSQP."""
    cons = []
    n = N_CIRCLES

    # Non-overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            cons.append({
                'type': 'ineq',
                'fun': partial(no_overlap_sq, i, j),
                'jac': partial(no_overlap_sq_grad, i, j)
            })

    # Boundary constraints
    for i in range(n):
        cons.append({'type': 'ineq', 'fun': partial(bound_x_lo, i), 'jac': partial(bound_x_lo_grad, i)})
        cons.append({'type': 'ineq', 'fun': partial(bound_x_hi, i), 'jac': partial(bound_x_hi_grad, i)})
        cons.append({'type': 'ineq', 'fun': partial(bound_y_lo, i), 'jac': partial(bound_y_lo_grad, i)})
        cons.append({'type': 'ineq', 'fun': partial(bound_y_hi, i), 'jac': partial(bound_y_hi_grad, i)})

    return cons


def slsqp_optimize(x0, constraints, bounds):
    """Run SLSQP optimization."""
    result = minimize(
        obj,
        x0,
        method='SLSQP',
        jac=obj_grad,
        bounds=bounds,
        constraints=constraints,
        options={
            'maxiter': 3000,
            'ftol': 1e-14,
            'disp': False
        }
    )
    return result


def perturbation_init(centers, radii, scale=0.05):
    """Create perturbed initialization."""
    n = N_CIRCLES
    pc = centers.copy()
    pr = radii.copy()
    for i in range(n):
        pc[i, 0] += np.random.uniform(-scale, scale)
        pc[i, 1] += np.random.uniform(-scale, scale)
        pc[i, 0] = np.clip(pc[i, 0], 0, 1)
        pc[i, 1] = np.clip(pc[i, 1], 0, 1)
        pr[i] += np.random.uniform(-scale * 0.5, scale * 0.5)
        pr[i] = max(pr[i], 0.01)
    return pc, pr


def run_packing():
    """Main packing function."""
    n = N_CIRCLES
    constraints = build_constraints()
    bounds = make_bounds()

    best_sum = -1.0
    best_centers = None
    best_radii = None

    # Strategy 1: Hexagonal initialization + force-directed + SLSQP
    centers, radii = init_hexagonal()
    centers, radii = force_directed(centers, radii, iterations=12000)

    # Ensure initial feasibility for SLSQP
    radii = np.minimum(radii, np.minimum(centers[:, 0], 1 - centers[:, 0]))
    radii = np.minimum(radii, np.minimum(centers[:, 1], 1 - centers[:, 1]))
    radii = np.maximum(radii, 0.005)

    # Reduce radii slightly to ensure no overlap
    for i in range(3):
        for j in range(i + 1, n):
            d = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if d < radii[i] + radii[j] - 1e-8:
                excess = radii[i] + radii[j] - d
                r_adj = excess / 2.0 + 1e-6
                radii[i] = max(radii[i] - r_adj, 0.005)
                radii[j] = max(radii[j] - r_adj, 0.005)

    x0 = np.concatenate([centers.flatten(), radii])
    result = slsqp_optimize(x0, constraints, bounds)

    if result.success or result.fun < 0:
        co = result.x[:2 * n].reshape(n, 2)
        ro = result.x[2 * n:]
        current_sum = np.sum(ro)
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = co.copy()
            best_radii = ro.copy()

    # Strategy 2: Multiple perturbed restarts
    for restart in range(5):
        pc, pr = perturbation_init(best_centers, best_radii, scale=0.03)
        pc, pr = force_directed(pc, pr, iterations=8000)

        # Ensure feasibility
        pr = np.minimum(pr, np.minimum(pc[:, 0], 1 - pc[:, 0]))
        pr = np.minimum(pr, np.minimum(pc[:, 1], 1 - pc[:, 1]))
        pr = np.maximum(pr, 0.005)

        for i in range(3):
            for j in range(i + 1, n):
                d = np.sqrt(np.sum((pc[i] - pc[j]) ** 2))
                if d < pr[i] + pr[j] - 1e-8:
                    excess = pr[i] + pr[j] - d
                    r_adj = excess / 2.0 + 1e-6
                    pr[i] = max(pr[i] - r_adj, 0.005)
                    pr[j] = max(pr[j] - r_adj, 0.005)

        result = slsqp_optimize(x0, constraints, bounds)

    radii3 = np.minimum(radii3, np.minimum(centers3[:, 0], 1 - centers3[:, 0]))
    radii3 = np.minimum(radii3, np.minimum(centers3[:, 1], 1 - centers3[:, 1]))
    radii3 = np.maximum(radii3, 0.005)
