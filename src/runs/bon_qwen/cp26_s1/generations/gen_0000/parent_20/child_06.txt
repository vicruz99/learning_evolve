import numpy as np
from scipy.optimize import minimize
import random

def _compute_loss(params, n):
    xs = params[0::3]
    ys = params[1::3]
    rs = params[2::3]
    
    # Objective: maximize sum of radii => minimize negative sum
    obj = -np.sum(rs)
    
    penalty = 0.0
    weight = 5000.0
    
    # Overlap penalty: sum of squared overlaps
    for i in range(n):
        xi, yi, ri = xs[i], ys[i], rs[i]
        for j in range(i+1, n):
            dist = np.hypot(xi - xs[j], yi - ys[j])
            overlap = ri + rs[j] - dist
            if overlap > 0.0:
                penalty += overlap**2
                
    # Boundary penalty: sum of squared violations
    for i in range(n):
        xi, yi, ri = xs[i], ys[i], rs[i]
        if xi < ri: penalty += (xi - ri)**2
        if xi > 1.0 - ri: penalty += (xi + ri - 1.0)**2
        if yi < ri: penalty += (yi - ri)**2
        if yi > 1.0 - ri: penalty += (yi + ri - 1.0)**2
        
    return obj + weight * penalty

def _generate_initial_packing(n):
    r_init = 0.03
    centers = []
    attempts = 0
    # Try to place circles randomly without overlap
    while len(centers) < n and attempts < 20000:
        x = random.uniform(r_init, 1.0 - r_init)
        y = random.uniform(r_init, 1.0 - r_init)
        valid = True

            if (x - cx)**2 + (y - cy)**2 < (r_init + r_init)**2:
                valid = False
                break
        if valid:
            centers.append((x, y))
        attempts += 1
        
    if len(centers) < n:
        # Fallback to hexagonal grid
        centers = []
        k = 0
        row = 0
        while k < n:
            y = 0.15 + row * 0.1732
            if y > 0.85:
                y = 0.85
            offset = 0.0866 if row % 2 == 1 else 0.0
            x = 0.1 + offset
            while x <= 0.9 and k < n:
                centers.append((x, y))
                k += 1
                x += 0.1732
            row += 1
    return centers

def run_packing():
    n = 26
    best_params = None
    best_sum_r = -np.inf
    
    # Multiple restarts to avoid local minima
    for _ in range(10):
        centers = _generate_initial_packing(n)
        r_init = 0.06
        params0 = np.array([val for c in centers for val in [c[0], c[1], r_init]])

            bounds[2 + 3*i] = (0.0, 0.5)

        current_sum = np.sum(rs)

        penalty_residual = res.fun + current_sum

            continue

            best_sum_r = current_sum
            best_params = res.x.copy()
            
    xs = best_params[0::3]
    ys = best_params[1::3]
    rs = best_params[2::3]
    
    centers = np.column_stack((xs, ys))
    return centers, rs, best_sum_r
