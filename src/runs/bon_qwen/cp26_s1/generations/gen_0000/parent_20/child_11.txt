import numpy as np
from scipy.optimize import linprog, minimize

def run_packing():
    # Constants
    N = 26
    
    # 1. Precompute LP Constraint Matrix A_ub
    # Variables: r_0, ..., r_25
    # Constraints:
    # 1. r_i <= x_i
    # 2. r_i <= 1 - x_i
    # 3. r_i <= y_i
    # 4. r_i <= 1 - y_i
    # 5. r_i + r_j <= dist(c_i, c_j) for all i < j
    
    # Structure of A_ub is constant.
    # Rows 0 to N-1: r_i <= x_i  (coeff 1 at i)
    # Rows N to 2N-1: r_i <= 1-x_i (coeff 1 at i)
    # Rows 2N to 3N-1: r_i <= y_i (coeff 1 at i)
    # Rows 3N to 4N-1: r_i <= 1-y_i (coeff 1 at i)
    # Rows 4N to end: r_i + r_j <= dist (coeff 1 at i, 1 at j)
    
    num_boundary_rows = 4 * N
    num_pairs = N * (N - 1) // 2
    total_rows = num_boundary_rows + num_pairs
    
    A_ub = np.zeros((total_rows, N))
    pairs = []
    
    # Fill boundary constraints
    for i in range(N):
        A_ub[i, i] = 1.0          # r_i <= x_i
        A_ub[N + i, i] = 1.0      # r_i <= 1-x_i
        A_ub[2*N + i, i] = 1.0    # r_i <= y_i
        A_ub[3*N + i, i] = 1.0    # r_i <= 1-y_i
        
    # Fill distance constraints
    row_idx = 4 * N
    for i in range(N):
        for j in range(i + 1, N):
            A_ub[row_idx, i] = 1.0
            A_ub[row_idx, j] = 1.0
            pairs.append((i, j))
            row_idx += 1
            
    c_obj = -np.ones(N) # Maximize sum(r) => Minimize -sum(r)
    bounds_lp = [(0, None)] * N

    def objective(centers_flat):
        centers = centers_flat.reshape((N, 2))
        
        # Compute RHS vector b_ub
        b_ub = np.zeros(total_rows)
        
        # Boundary constraints RHS
        # x coords: centers[:, 0]
        # y coords: centers[:, 1]
        
        # r_i <= x_i
        b_ub[:N] = centers[:, 0]
        # r_i <= 1-x_i
        b_ub[N:2*N] = 1.0 - centers[:, 0]
        # r_i <= y_i
        b_ub[2*N:3*N] = centers[:, 1]
        # r_i <= 1-y_i
        b_ub[3*N:4*N] = 1.0 - centers[:, 1]
        
        # Distance constraints RHS
        idx = 4 * N
        for i, j in pairs:
            # Euclidean distance
            diff = centers[i] - centers[j]
            dist = np.sqrt(np.dot(diff, diff))
            b_ub[idx] = dist
            idx += 1
            
        try:
            # Solve LP
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_lp, method='highs')
            if res.success:
                # Return negative of objective value (since we minimized -sum)
                return -res.fun
            else:
                # If LP fails (shouldn't happen with r=0 feasible), return 0
                return 0.0
        except Exception:
            return 0.0

    # 2. Generate Initial Configuration (Hexagonal Grid)
    # We want to pack 26 points.
    # A hexagonal lattice allows higher density.
    # Let's try to fit a grid.
    
    centers_init = np.zeros((N, 2))
    
    # Heuristic for hex grid placement
    # Approx spacing to fit 26 circles.
    # Area per circle ~ 1/26. r ~ 0.1. Diameter ~ 0.2.
    # Spacing ~ 0.2.
    
    # Let's construct rows.
    # Row height = spacing * sqrt(3)/2
    # Let spacing = 0.21
    spacing = 0.21
    h = spacing * np.sqrt(3) / 2
    
    pts = []
    y = spacing # Start with margin
    row_idx = 0
    
    while y <= 1.0 - spacing and len(pts) < N:
        x = spacing
        shift = spacing / 2.0 if (row_idx % 2 == 1) else 0.0
        
        while x <= 1.0 - spacing and len(pts) < N:
            pts.append([x + shift, y])
            x += spacing
            if len(pts) == N:
                break
        y += h
        row_idx += 1
        
    # If we have fewer than N points (unlikely with this spacing), fill randomly
    while len(pts) < N:
        pts.append([np.random.rand(), np.random.rand()])
        
    centers_init = np.array(pts[:N])
    
    # Ensure centers are within valid range [0, 1]
    # (The grid generation above respects margins, but let's clamp just in case)
    centers_init = np.clip(centers_init, 0, 1)

    # 3. Optimization
    # Flatten centers for optimizer
    x0 = centers_init.flatten()
    bounds_opt = [(0, 1)] * (N * 2)
    
    # Use Nelder-Mead as it doesn't require gradient and handles non-smooth functions reasonably well
    # We can also try Powell.
    # To improve chance of global optimum, we might run multiple times or use Differential Evolution,
    # but for a single run, a strong local optimizer from a good start is efficient.
    
    # Let's try a multi-start or just a robust run.
    # Since time is a factor, let's do one solid optimization run.
    # Nelder-Mead is derivative-free.
    
    try:
        res_opt = minimize(objective, x0, method='Nelder-Mead', 
                           options={'maxiter': 5000, 'xatol': 1e-6, 'fatol': 1e-8})
        best_centers_flat = res_opt.x
    except Exception:
        # Fallback to initial if optimization fails
        best_centers_flat = x0
        
    best_centers = best_centers_flat.reshape((N, 2))
    
    # 4. Final Calculation of Radii
    # We need to solve the LP one last time to get the exact radii for the optimized centers
    centers = best_centers
    b_final = np.zeros(total_rows)
    
    b_final[:N] = centers[:, 0]
    b_final[N:2*N] = 1.0 - centers[:, 0]
    b_final[2*N:3*N] = centers[:, 1]
    b_final[3*N:4*N] = 1.0 - centers[:, 1]
    
    idx = 4 * N
    for i, j in pairs:
        diff = centers[i] - centers[j]
        dist = np.sqrt(np.dot(diff, diff))
        b_final[idx] = dist
        idx += 1
        
    res_final = linprog(c_obj, A_ub=A_ub, b_ub=b_final, bounds=bounds_lp, method='highs')
    
    if res_final.success:
        radii = -res_final.x # x is maximized sum, but linprog minimizes, so res_final.x are radii?
        # Wait, linprog minimizes c^T x. c = -1. So it minimizes -sum(r).
        # The solution x contains the values of r.
        # Since c was -1, minimizing -sum is maximizing sum.
        # The variable x in linprog result corresponds to r.
        radii = res_final.x
        
        # Validate non-negativity (should be handled by bounds)
        radii = np.maximum(radii, 0)
        
        sum_radii = np.sum(radii)
        
        return best_centers, radii, sum_radii
    else:
        # Fallback if LP fails (very unlikely)
        return best_centers, np.zeros(N), 0.0
