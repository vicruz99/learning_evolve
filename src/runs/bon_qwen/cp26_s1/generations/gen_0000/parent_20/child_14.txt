import numpy as np
import scipy.optimize as opt

def get_initial_configuration():
    """
    Generates an initial hexagonal packing configuration for 26 circles.
    We aim for a layout that fits 26 circles with a reasonable radius.
    A 5-4-5-4-5-3 row distribution is a good candidate for hexagonal packing.
    """
    centers = []
    # We will estimate a radius r such that 5 circles fit in width 1.
    # 10r <= 1 => r <= 0.1.
    # Let's start with r = 0.095 to be safe, then optimizer will expand.
    r_start = 0.095
    
    # Row parameters
    # Row 1: 5 circles
    # Row 2: 4 circles (offset)
    # Row 3: 5 circles
    # Row 4: 4 circles (offset)
    # Row 5: 5 circles
    # Row 6: 3 circles (offset) -> Total 26
    
    rows_config = [5, 4, 5, 4, 5, 3]
    
    y_curr = r_start
    
    for i, count in enumerate(rows_config):
        # In hexagonal packing, even rows (index 1, 3, 5) are shifted by r horizontally
        # relative to odd rows.
        if i % 2 == 0:
            # Odd rows (0, 2, 4): aligned with left wall at r
            x_start = r_start
        else:
            # Even rows (1, 3, 5): shifted by r
            # To center them or keep them inside, we need to be careful.
            # If row 0 has centers at r, 3r, 5r, 7r, 9r.
            # Row 1 (shifted) centers at 2r, 4r, 6r, 8r.
            # This fits perfectly if width allows.
            x_start = 2 * r_start
            
        for j in range(count):
            x = x_start + j * (2 * r_start)
            centers.append([x, y_curr])
            
        # Move to next row
        # Vertical spacing for hexagonal packing is sqrt(3) * r
        y_curr += np.sqrt(3) * r_start
        
    return np.array(centers), r_start

def objective(vars):
    """
    Objective function: Negative sum of radii.
    vars structure: [x1, y1, r1, x2, y2, r2, ...]
    """
    n = len(vars) // 3
    radii = np.array(vars[2::3])
    return -np.sum(radii)

def get_constraints(n):
    """
    Returns a list of constraint dictionaries for scipy.optimize.
    Constraints:
    1. Box constraints (handled by bounds)
    2. Radius non-negativity (handled by bounds)
    3. Boundary containment: r <= x <= 1-r  =>  x - r >= 0, 1 - x - r >= 0
    4. Boundary containment: r <= y <= 1-r  =>  y - r >= 0, 1 - y - r >= 0
    5. Non-overlap: dist_ij >= ri + rj  =>  dist_ij - (ri + rj) >= 0
    """
    constraints = []
    
    # Helper to extract vars
    # x = vars[0::3], y = vars[1::3], r = vars[2::3]
    
    # We will define constraint functions that return values >= 0
    
    # 1. Boundary containment constraints
    # x_i - r_i >= 0
    # 1 - x_i - r_i >= 0
    # y_i - r_i >= 0
    # 1 - y_i - r_i >= 0
    
    # To avoid creating 4*n functions, we can vectorize or create a generic lambda,
    # but rule says no lambda/closures from nesting? 
    # Actually, standard closure is allowed if defined inside function? 
    # "Don't use any lambda functions." -> I will use explicit functions or lists.
    # However, scipy expects callable. I can return a list of dictionaries with 'fun' key.
    # I can define helper functions inside get_constraints.
    
    # Since I cannot use lambdas, I will create a list of constraint dicts 
    # where 'fun' is a regular function defined in the scope.
    # But defining 100+ functions dynamically is messy.
    # Alternative: Create one big constraint function that checks all boundaries and returns vector?
    # SLSQP supports vector constraints? 
    # "fun" can return an array. If array > 0, constraints satisfied.
    
    def boundary_constraints(vars):
        res = []
        # vars layout: x1, y1, r1, x2, y2, r2...
        # Indices: x=0, 3, 6...; y=1, 4, 7...; r=2, 5, 8...
        # Actually vars is flat.
        # x_i is at index 3*i
        # y_i is at index 3*i + 1
        # r_i is at index 3*i + 2
        
        for i in range(n):
            x = vars[3*i]
            y = vars[3*i + 1]
            r = vars[3*i + 2]
            
            # x >= r
            res.append(x - r)
            # x <= 1 - r  =>  1 - x - r >= 0
            res.append(1.0 - x - r)
            # y >= r
            res.append(y - r)
            # y <= 1 - r  =>  1 - y - r >= 0
            res.append(1.0 - y - r)
        return np.array(res)

    # 2. Non-overlap constraints
    # dist(i, j) - (r_i + r_j) >= 0
    # (x_i - x_j)^2 + (y_i - y_j)^2 >= (r_i + r_j)^2
    # sqrt(...) >= r_i + r_j
    # It's safer to use the squared form for differentiability? 
    # sqrt is non-differentiable at 0, but circles shouldn't be at same pos.
    # Actually, constraint g(x) >= 0.
    # g = dist - sum_r.
    
    def nonoverlap_constraints(vars):
        res = []
        # Extract arrays
        xs = np.array(vars[0::3])
        ys = np.array(vars[1::3])
        rs = np.array(vars[2::3])
        
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt((xs[i] - xs[j])**2 + (ys[i] - ys[j])**2)
                sum_r = rs[i] + rs[j]
                res.append(dist - sum_r)
        return np.array(res)

    constraints.append({'type': 'ineq', 'fun': boundary_constraints})
    constraints.append({'type': 'ineq', 'fun': nonoverlap_constraints})
    
    return constraints

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # 1. Get initial configuration
    centers_init, r_init = get_initial_configuration()
    
    # 2. Flatten variables
    # Format: [x1, y1, r1, x2, y2, r2, ...]
    x0 = np.zeros(3 * n)
    for i in range(n):
        x0[3*i]   = centers_init[i, 0]
        x0[3*i + 1] = centers_init[i, 1]
        x0[3*i + 2] = r_init
        
    # 3. Bounds
    # x: [0, 1], y: [0, 1], r: [0, 0.5]
    bounds = []
    for _ in range(n):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 0.5)) # r
        
    # 4. Constraints
    constraints = get_constraints(n)
    
    # 5. Optimization
    # Using SLSQP
    try:
        res = opt.minimize(
            objective, 
            x0, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=constraints,
            options={'ftol': 1e-9, 'maxiter': 1000, 'disp': False}
        )
        
        if res.success:
            optimal_vars = res.x
        else:
            # Fallback if optimization fails, try to return initial or best found
            # Or run a second pass with different method?
            # Let's try to use the result anyway, it might be feasible
            optimal_vars = res.x
            
    except Exception as e:
        # In case of any error, return initial config
        # Reshape x0 back
        optimal_vars = x0
        
    # 6. Extract results
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    for i in range(n):
        centers[i, 0] = optimal_vars[3*i]
        centers[i, 1] = optimal_vars[3*i + 1]
        radii[i] = optimal_vars[3*i + 2]
        
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
