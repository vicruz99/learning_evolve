import numpy as np
from scipy.optimize import minimize


def compute_penalty(centers, radii, n):
    """Compute penalty for constraint violations."""
    penalty = 0.0
    weight = 2000.0

    # Boundary violations
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        for boundary, coord in [(0.0, x), (1.0, x), (0.0, y), (1.0, y)]:
            if boundary == 0.0:
                violation = r - coord
            else:
                violation = coord + r - 1.0
            if violation > 0.0:
                penalty += weight * violation * violation

    # Overlap violations
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            min_dist = radii[i] + radii[j]
            violation = min_dist - dist
            if violation > 0.0:
                penalty += weight * violation * violation

    return penalty


def pack_objective(params, n):
    """Objective function: minimize negative sum of radii plus penalties."""
    centers = params[: 2 * n].reshape(n, 2)
    radii = params[2 * n:]
    radii = np.maximum(radii, 1e-10)

    obj_val = -np.sum(radii)
    penalty = compute_penalty(centers, radii, n)

    return obj_val + penalty


def is_valid_packing(centers, radii, n):
    """Check if packing satisfies all constraints."""
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-10 or x + r > 1.0 + 1e-10:
            return False
        if y - r < -1e-10 or y + r > 1.0 + 1e-10:
            return False

    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            if dist < radii[i] + radii[j] - 1e-10:
                return False

    return True


def make_hex_config(n, pattern):
    """Create hexagonal grid configuration from row pattern."""
    centers_list = []
    r = 0.095

    y_pos = r
    for row_idx, count in enumerate(pattern):
        if len(centers_list) >= n:
            break

        offset = r * 0.5 if row_idx % 2 == 1 else 0.0
        x_start = r + offset
        x_end = 1.0 - r - offset

        remaining = n - len(centers_list)
        actual_count = min(count, remaining)

        if actual_count <= 0:
            continue

        if actual_count == 1:
            xs = [(x_start + x_end) / 2.0]
        else:
            xs = np.linspace(x_start, x_end, actual_count)

        for x in xs:
            centers_list.append([x, y_pos])

        y_pos += r * np.sqrt(3.0) * 0.95

    centers = np.array(centers_list[:n])
    radii = np.full(n, r)

    return centers, radii


def make_grid_config(n):
    """Create square grid configuration."""
    # 5x5 grid = 25, plus 1 extra
    centers_list = []
    r = 0.095

    for row in range(5):
        for col in range(5):
            if len(centers_list) >= n:
                break
            x = r + col * (1.0 - 2.0 * r) / 4.0
            y = r + row * (1.0 - 2.0 * r) / 4.0
            centers_list.append([x, y])
        if len(centers_list) >= n:
            break

    # Add extra circle in center if needed
    if len(centers_list) < n:
        centers_list.append([0.5, 0.5])

    centers = np.array(centers_list[:n])
    radii = np.full(n, r)

    return centers, radii


def perturb_config(centers, radii, seed):
    """Add random perturbation to configuration."""
    rng = np.random.RandomState(seed)
    c_pert = centers + rng.randn(*centers.shape) * 0.015
    r_pert = radii + rng.randn(len(radii)) * 0.005
    c_pert = np.clip(c_pert, 0.05, 0.95)
    r_pert = np.clip(r_pert, 0.02, 0.25)
    return c_pert, r_pert


def optimize_config(centers, radii, n):
    """Optimize a single configuration."""
    params = np.concatenate([centers.ravel(), radii])

    bounds = []
    for _ in range(n):
        bounds.append((0.0, 1.0))  # x
        bounds.append((0.0, 1.0))  # y
    for _ in range(n):
        bounds.append((0.001, 0.5))  # radius

    result = minimize(
        pack_objective,
        params,
        args=(n,),
        method="L-BFGS-B",
        bounds=bounds,
        options={"maxiter": 5000, "ftol": 1e-15, "gtol": 1e-10},
    )

    c_opt = result.x[: 2 * n].reshape(n, 2)
    r_opt = result.x[2 * n:]

    return c_opt, r_opt


def run_packing():
    """Main function to pack 26 circles in unit square."""
    n = 26

    best_sum = 0.0
    best_centers = np.zeros((n, 2))
    best_radii = np.full(n, 0.05)

    # Generate diverse initial configurations
    initial_configs = []

    # Various hexagonal patterns summing to >= 26
    hex_patterns = [
        [5, 4, 5, 4, 5, 3],
        [5, 5, 5, 5, 4, 2],
        [6, 5, 5, 5, 4, 1],
        [5, 4, 5, 4, 6, 2],
        [4, 5, 4, 5, 4, 4],
        [5, 5, 4, 5, 5, 2],
        [5, 3, 5, 5, 5, 3],
        [6, 6, 6, 6, 2],
    ]

    for pattern in hex_patterns:
        if sum(pattern) >= n:
            c, r = make_hex_config(n, pattern)
            initial_configs.append((c, r))

    # Grid configuration
    c, r = make_grid_config(n)
    initial_configs.append((c, r))

    # Perturbed versions of top configs
    for i, (c_base, r_base) in enumerate(initial_configs[:4]):
        for trial in range(6):
            seed = i * 100 + trial + 7
            c_pert, r_pert = perturb_config(c_base, r_base, seed)
            initial_configs.append((c_pert, r_pert))

    # Optimize each configuration
    for c_init, r_init in initial_configs:
        c_opt, r_opt = optimize_config(c_init, r_init, n)

        if is_valid_packing(c_opt, r_opt, n):
            current_sum = np.sum(r_opt)
            if current_sum > best_sum:
                best_sum = current_sum
                best_centers = c_opt.copy()
                best_radii = r_opt.copy()

    # If no valid packing found, return reasonable fallback
    if best_sum < 0.1:
        best_centers, best_radii = make_hex_config(n, [5, 4, 5, 4, 5, 3])
        best_sum = np.sum(best_radii)

    return best_centers, best_radii, best_sum
