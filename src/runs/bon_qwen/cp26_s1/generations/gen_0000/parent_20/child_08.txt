import numpy as np
from scipy.optimize import minimize

def compute_constraints(z):
    """
    Computes all inequality constraints g(z) >= 0.
    z is a flat array: [x0, x1, ..., x25, y0, ..., y25, r0, ..., r25]
    """
    n = 26
    x = z[0:n]
    y = z[n:2*n]
    r = z[2*n:]
    
    num_boundary = 4 * n
    num_pairwise = n * (n - 1) // 2
    cons = np.empty(num_boundary + num_pairwise)
    
    idx = 0
    for i in range(n):
        # Circle inside square constraints
        cons[idx] = x[i] - r[i]
        idx += 1
        cons[idx] = 1.0 - x[i] - r[i]
        idx += 1
        cons[idx] = y[i] - r[i]
        idx += 1
        cons[idx] = 1.0 - y[i] - r[i]
        idx += 1
        
    for i in range(n):
        for j in range(i + 1, n):
            # Non-overlap constraints
            dx = x[i] - x[j]
            dy = y[i] - y[j]
            cons[idx] = dx * dx + dy * dy - (r[i] + r[j]) ** 2
            idx += 1
            
    return cons

def obj_func(z):
    """Objective: maximize sum of radii -> minimize negative sum."""
    return -np.sum(z[52:])

def run_packing():
    n = 26
    best_z = None
    best_score = -np.inf
    
    # Try multiple initializations to avoid poor local minima
    for seed in range(5):
        rng = np.random.RandomState(seed)
        
        # Generate hexagonal-ish initial positions
        pts = []
        for row in range(7):
            for col in range(5):
                if len(pts) >= n:
                    break
                # Staggered grid coordinates
                px = col * 0.20 + (row % 2) * 0.10 + 0.08 + rng.uniform(-0.02, 0.02)
                py = row * 0.16 + 0.08 + rng.uniform(-0.02, 0.02)
                if 0.05 <= px <= 0.95 and 0.05 <= py <= 0.95:
                    pts.append([px, py])
        # Fill remaining if grid didn't yield enough
        while len(pts) < n:
            pts.append([rng.uniform(0.2, 0.8), rng.uniform(0.2, 0.8)])
        pts = np.array(pts[:n])
        
        # Compute safe initial radii (fraction of min clearance)
        r_init = np.zeros(n)
        for i in range(n):
            d_min = min(pts[i, 0], 1.0 - pts[i, 0], pts[i, 1], 1.0 - pts[i, 1])
            for j in range(n):
                if i != j:
                    d = np.sqrt(np.sum((pts[i] - pts[j]) ** 2))
                    if d < d_min:
                        d_min = d
            r_init[i] = d_min * 0.40  # Strictly feasible start
            
        # Flatten to optimization variable vector
        z0 = np.zeros(3 * n)
        z0[0:n] = pts[:, 0]
        z0[n:2*n] = pts[:, 1]
        z0[2*n:] = r_init
        
        bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)] * n + [(0.0, 0.5)] * n
        cons = {'type': 'ineq', 'fun': compute_constraints}
        
        try:
            res = minimize(obj_func, z0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 4000, 'ftol': 1e-10, 'disp': False})
            
            # Verify constraints are satisfied within tolerance
            if np.all(compute_constraints(res.x) > -1e-5):
                score = -res.fun
                if score > best_score:
                    best_score = score
                    best_z = res.x
        except Exception:
            continue
            
    # Fallback to initial config if optimization completely fails (unlikely)
    if best_z is None:
        best_z = z0
        best_score = -obj_func(z0)
        
    centers = np.column_stack((best_z[0:n], best_z[n:2*n]))
    radii = best_z[2*n:]
    return centers, radii, np.sum(radii)
