import numpy as np
from scipy.optimize import minimize

def evaluate(vars, n, mu):
    """Compute objective and constraint violations."""
    C = vars.reshape((n, 3))
    x, y, r = C[:, 0], C[:, 1], C[:, 2]
    
    loss = -np.sum(r)
    
    # Pairwise constraints: distance >= r_i + r_j
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist = np.hypot(dx, dy)
    req = r[:, None] + r[None, :]
    
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    diff = req - dist
    diff = np.maximum(0, diff)
    pair_pen = np.sum((diff[mask])**2)
    
    # Boundary constraints
    bound_pen = np.sum(np.maximum(0, r - x)**2 + 
                       np.maximum(0, r - (1 - x))**2 + 
                       np.maximum(0, r - y)**2 + 
                       np.maximum(0, r - (1 - y))**2)
                       
    total_pen = pair_pen + bound_pen
    loss += mu * total_pen
    return loss, total_pen

def obj_func(vars, n, mu):
    """Wrapper for minimize."""
    loss, _ = evaluate(vars, n, mu)
    return loss

def run_packing():
    n = 26
    best_loss = np.inf
    best_vars = None
    best_sum_r = 0.0
    
    bounds = [(0, 1)] * (2*n) + [(0, 0.5)] * n
    
    for seed in range(30):
        np.random.seed(seed)
        
        init_vars = np.zeros((n, 3))
        
        # Diverse initialization strategies to escape local minima
        if seed < 10:
            init_vars[:, 0] = np.random.rand(n)
            init_vars[:, 1] = np.random.rand(n)
            init_vars[:, 2] = 0.05
        elif seed < 20:
            idx = 0
            for row in range(5):
                y = 0.15 + row * 0.18
                x_start = 0.12 if row % 2 == 1 else 0.08
                for col in range(6):
                    x = x_start + col * 0.16
                    if idx < n:
                        init_vars[idx] = [x, y, 0.06]
                        idx += 1
        else:
            init_vars[:, 0] = np.random.uniform(0.05, 0.95, n)
            init_vars[:, 1] = np.random.uniform(0.05, 0.95, n)
            init_vars[:, 2] = 0.07
            
        mu = 1000.0
        try:
            res = minimize(obj_func, init_vars.ravel(), method='L-BFGS-B', bounds=bounds, 
                           args=(n, mu), options={'maxiter': 500, 'ftol': 1e-10, 'gtol': 1e-8})
            
            if not np.isnan(res.fun):
                final_loss, final_viol = evaluate(res.x, n, mu)
                if final_loss < best_loss:
                    best_loss = final_loss
                    best_vars = res.x.copy()
                    best_sum_r = np.sum(best_vars.reshape((n, 3))[:, 2])
        except Exception:
            continue
            
    centers = best_vars.reshape((n, 3))[:, :2]
    radii = best_vars.reshape((n, 3))[:, 2]
    
    # Post-processing to strictly enforce constraints numerically
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        # Ensure inside unit square
        r = min(r, x, 1-x, y, 1-y)
        radii[i] = max(r, 0.0)
        
    return centers, radii, float(np.sum(radii))
