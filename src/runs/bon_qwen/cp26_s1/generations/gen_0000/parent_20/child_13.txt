```python
import numpy as np
import scipy.optimize as opt

def compute_penalty(centers, r):
    n = centers.shape[0]
    penalty = 0.0
    
    # Vectorized overlap penalty
    # Compute pairwise distances
    # centers is (N, 2)
    # diff shape (N, N, 2)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    # Upper triangle indices
    # We only care about i < j
    # dists is symmetric.
    # We can sum over upper triangle.
    # Mask for upper triangle
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    upper_dists = dists[mask]
    
    min_dist = 2 * r
    violations = np.maximum(0, min_dist - upper_dists)
    penalty = np.sum(violations**2)
    
    # Boundary penalty
    # x >= r => r - x <= 0. Violation if x < r. Amount r - x.
    # x <= 1-r => x - (1-r) <= 0. Violation if x > 1-r. Amount x - (1-r).
    
    x = centers[:, 0]
    y = centers[:, 1]
    
    # x < r
    vx = np.maximum(0, r - x)
    penalty += np.sum(vx**2)
    
    # x > 1 - r
    vx = np.maximum(0, x - (1 - r))
    penalty += np.sum(vx**2)
    
    # y < r
    vy = np.maximum(0, r - y)
    penalty += np.sum(vy**2)
    
    # y > 1 - r
    vy = np.maximum(0, y - (1 - r))
    penalty += np.sum(vy**2)
    
    return penalty

def objective_with_args(v, args):
    centers = np.array(v).reshape(args['n'], 2)
    return compute_penalty(centers, args['r'])
```
