import numpy as np
import scipy.optimize as opt

def calculate_penalty(centers, radii):
    """
    Calculates the penalty for overlaps and boundary violations.
    Returns a scalar penalty value.
    """
    n = centers.shape[0]
    penalty = 0.0
    
    # Boundary penalty
    # We want x >= r, x <= 1-r, y >= r, y <= 1-r
    # Violation = max(0, r - x) + max(0, r - (1-x)) ...
    # Using a smooth approximation or simple max
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        
        # Left wall
        if x - r < 0:
            penalty += (x - r)**2
        # Right wall
        if x + r > 1:
            penalty += (x + r - 1)**2
        # Bottom wall
        if y - r < 0:
            penalty += (y - r)**2
        # Top wall
        if y + r > 1:
            penalty += (y + r - 1)**2

    # Overlap penalty
    # We want dist >= r_i + r_j
    # Violation if dist < r_i + r_j
    # Penalty = max(0, r_i + r_j - dist)^2
    # To avoid sqrt in derivative issues, we can work with squared distances?
    # But radius sum is linear. Let's stick to Euclidean distance.
    
    # Vectorized overlap calculation for speed
    # centers shape (n, 2)
    # dist matrix
    # Using broadcasting
    c_diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :] # (n, n, 2)
    dists = np.sqrt(np.sum(c_diff**2, axis=2)) # (n, n)
    
    # radii sum matrix
    r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    
    # Overlap amount (positive if overlapping)
    overlap = r_sum - dists
    
    # We only care about upper triangle to avoid double counting and self
    # Triangular mask
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    overlaps = overlap[mask]
    
    # Sum of squared positive overlaps
    positive_overlaps = overlaps[overlaps > 0]
    if len(positive_overlaps) > 0:
        penalty += np.sum(positive_overlaps**2)
        
    return penalty

def objective_function(params, n_circles):
    """
    Objective function to minimize.
    params contains [x_0, y_0, r_0, x_1, y_1, r_1, ...]
    We minimize -sum(radii) + penalty.
    """
    centers = params[0::3].reshape(n_circles, 2)
    radii = params[1::3] # Wait, indexing might be tricky.
    # Let's restructure params to [x0, y0, r0, x1, y1, r1...]
    # Actually, reshaping is easier if we keep structure.
    # params is 1D array.
    
    centers = params.reshape(n_circles, 3)[:, :2]
    radii = params.reshape(n_circles, 3)[:, 2]
    
    # Ensure radii are non-negative for penalty calculation stability, 
    # though bounds will handle it.
    radii = np.maximum(radii, 1e-6)
    
    penalty = calculate_penalty(centers, radii)
    
    # Objective: Maximize sum of radii -> Minimize -sum(radii)
    # We add a large weight to penalty to enforce constraints.
    # Weight needs to be high.
    weight = 10000.0
    
    cost = -np.sum(radii) + weight * penalty
    return cost

    # Best solution found
