import numpy as np
from scipy.optimize import differential_evolution, minimize

def get_radii(centers):
    """
    Computes the maximum valid radius for each circle given their centers.
    Radii are limited by the unit square boundaries and pairwise distances.
    """
    n = centers.shape[0]
    radii = np.ones(n)
    
    # Constraint: Inside unit square [0,1]x[0,1]
    x = centers[:, 0]
    y = centers[:, 1]
    radii = np.minimum(radii, np.minimum(x, 1.0 - x))
    radii = np.minimum(radii, np.minimum(y, 1.0 - y))
    
    # Constraint: Non-overlapping
    # Compute pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    # Each circle radius is at most half the distance to the nearest neighbor
    min_dists = np.min(dists, axis=1) / 2.0
    radii = np.minimum(radii, min_dists)
    
    return radii

def objective(params):
    """Objective function to minimize (negative sum of radii)"""
    centers = params.reshape(-1, 2)
    radii = get_radii(centers)
    return -np.sum(radii)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    n_vars = n * 2
    
    # Define bounds for each coordinate (0 to 1)
    bounds = [(0, 1)] * n_vars
    
    # Global optimization using Differential Evolution
    # popsize=30 and maxiter=1000 provide a good balance for this dimensionality
    result_global = differential_evolution(
        objective, 
        bounds, 
        seed=42, 
        popsize=30, 
        maxiter=1000, 
        tol=1e-7, 
        polish=False
    )
    
    # Local refinement using Nelder-Mead on the best global result
    result_local = minimize(
        objective, 
        result_global.x, 
        method='Nelder-Mead', 
        options={'xatol': 1e-7, 'fatol': 1e-7, 'maxiter': 1000}
    )
    
    # Extract best centers
    best_centers = result_local.x.reshape(-1, 2)
    
    # Compute final radii based on optimized centers
    best_radii = get_radii(best_centers)
    
    # Calculate sum of radii
    sum_radii = float(np.sum(best_radii))
    
    return best_centers, best_radii, sum_radii
