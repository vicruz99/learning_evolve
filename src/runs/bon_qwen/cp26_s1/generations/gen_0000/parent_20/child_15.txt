import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def objective(v):
    """Objective function: minimize negative sum of radii"""
    return -np.sum(v[2::3])

def get_boundary_constraints(v):
    """Compute boundary constraints: circles must be inside [0,1]x[0,1]"""
    cx = v[0::3]
    cy = v[1::3]
    cr = v[2::3]
    
    # x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
    c1 = cx - cr
    c2 = 1.0 - cx - cr
    c3 = cy - cr
    c4 = 1.0 - cy - cr
    
    return np.concatenate([c1, c2, c3, c4])

def get_dist_constraints(v):
    """Compute pairwise non-overlap constraints: distance >= r_i + r_j"""
    cx = v[0::3]
    cy = v[1::3]
    cr = v[2::3]
    
    # Vectorized pairwise distance calculation
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dr = cr[:, None] + cr[None, :]
    
    dist = np.sqrt(dx*dx + dy*dy)
    margins = dist - dr
    
    # Extract upper triangle (i < j)
    mask = np.triu(np.ones((N_CIRCLES, N_CIRCLES), dtype=bool), k=1)
    return margins[mask]

def get_all_constraints(v):
    """Combine all constraints into a single vector"""
    bc = get_boundary_constraints(v)
    dc = get_dist_constraints(v)
    return np.concatenate([bc, dc])

def run_packing():
    """
    Optimizes circle packing in a unit square to maximize sum of radii.
    Returns (centers, radii, sum_radii)
    """
    # Initialize with a hexagonal lattice pattern
    centers = np.zeros((N_CIRCLES, 2))
    radii = np.full(N_CIRCLES, 0.09)
    
    row_counts = [6, 5, 6, 5, 4]
    idx = 0
    y = 0.12
    dx = 0.165
    dy = dx * np.sqrt(3.0) / 2.0
    
    for r_idx, count in enumerate(row_counts):
        x_start = 0.165 if r_idx % 2 == 1 else 0.165 - dx / 2.0
        for c in range(count):
            centers[idx] = [x_start + c * dx, y]
            idx += 1
        y += dy
        
    # Add small random perturbation to avoid symmetry traps
    centers += np.random.uniform(-0.005, 0.005, size=centers.shape)
    
    # Flatten variables: [x1, y1, r1, x2, y2, r2, ...]
    x0 = np.zeros(3 * N_CIRCLES)
    for i in range(N_CIRCLES):
        x0[3 * i] = centers[i, 0]
        x0[3 * i + 1] = centers[i, 1]
        x0[3 * i + 2] = radii[i]
        
    # Bounds for variables
    bounds = [(0.0, 1.0)] * N_CIRCLES + [(0.0, 1.0)] * N_CIRCLES + [(0.0, 0.5)] * N_CIRCLES
    
    # Constraint definition
    cons = {'type': 'ineq', 'fun': get_all_constraints}
    
    # Run optimization
    res = minimize(objective, x0, bounds=bounds, constraints=cons, method='SLSQP',
                   options={'maxiter': 5000, 'ftol': 1e-12, 'disp': False})
                   
    best_x = res.x
    best_centers = best_x[:2 * N_CIRCLES].reshape((N_CIRCLES, 2))
    best_radii = best_x[2 * N_CIRCLES:]
    
    # Ensure non-negative radii (numerical safety)
    best_radii = np.maximum(best_radii, 0.0)
    
    # Slight safety clamp to strictly satisfy validation tolerances
    best_radii *= 0.999999
    
    return best_centers, best_radii, float(np.sum(best_radii))
