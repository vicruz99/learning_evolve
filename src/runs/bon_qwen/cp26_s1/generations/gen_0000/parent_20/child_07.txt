N = 26

def objective(x):
    """Maximize sum of radii = minimize negative sum."""
    return -np.sum(x[2 * N:])


def grad_objective(x):
    """Gradient of objective."""
    g = np.zeros_like(x)
    g[2 * N:] = -1.0
    return g


def boundary_constraints(x):
    """All boundary and non-negativity constraints."""
    result = np.zeros(5 * N)
    for i in range(N):
        result[5 * i] = x[2 * i] - x[2 * N + i]  # x >= r
        result[5 * i + 1] = 1.0 - x[2 * i] - x[2 * N + i]  # x + r <= 1
        result[5 * i + 2] = x[2 * i + 1] - x[2 * N + i]  # y >= r
        result[5 * i + 3] = 1.0 - x[2 * i + 1] - x[2 * N + i]  # y + r <= 1
        result[5 * i + 4] = x[2 * N + i]  # r >= 0
    return result


def boundary_constraints_jac(x):
    """Jacobian of boundary constraints."""
    jac = np.zeros((5 * N, 3 * N))
    for i in range(N):
        jac[5 * i, 2 * i] = 1.0
        jac[5 * i, 2 * N + i] = -1.0
        jac[5 * i + 1, 2 * i] = -1.0
        jac[5 * i + 1, 2 * N + i] = -1.0
        jac[5 * i + 2, 2 * i + 1] = 1.0
        jac[5 * i + 2, 2 * N + i] = -1.0
        jac[5 * i + 3, 2 * i + 1] = -1.0
        jac[5 * i + 3, 2 * N + i] = -1.0
        jac[5 * i + 4, 2 * N + i] = 1.0
    return jac


def nonoverlap_constraints(x):
    """All pairwise non-overlap constraints."""
    num_pairs = N * (N - 1) // 2
    result = np.zeros(num_pairs)
    idx = 0
    for i in range(N):
        for j in range(i + 1, N):

            xj, yj = x[2 * j], x[2 * j + 1]
            ri, rj = x[2 * N + i], x[2 * N + j]
            dx = xi - xj
            dy = yi - yj
            dist = np.sqrt(dx * dx + dy * dy)
            result[idx] = dist - ri - rj
            idx += 1
    return result


def nonoverlap_constraints_jac(x):
    """Jacobian of non-overlap constraints."""
    num_pairs = N * (N - 1) // 2
    jac = np.zeros((num_pairs, 3 * N))
    idx = 0
    for i in range(N):
        for j in range(i + 1, N):
            xi, yi = x[2 * i], x[2 * i + 1]

            dy = yi - yj
            dist = np.sqrt(dx * dx + dy * dy) + 1e-15
            ndx = dx / dist
            ndy = dy / dist
            jac[idx, 2 * i] = ndx
            jac[idx, 2 * i + 1] = ndy
            jac[idx, 2 * j] = -ndx
            jac[idx, 2 * j + 1] = -ndy
            jac[idx, 2 * N + i] = -1.0
            jac[idx, 2 * N + j] = -1.0
            idx += 1
    return jac


def create_bounds():
    """Create bounds for optimization variables."""
    bounds = []
    for _ in range(2 * N):
        bounds.append((0.0, 1.0))
    for _ in range(N):
        bounds.append((0.0, 0.5))
    return bounds


def create_constraints():
    """Create constraint dictionaries."""
    return [
        {'type': 'ineq', 'fun': boundary_constraints, 'jac': boundary_constraints_jac},
        {'type': 'ineq', 'fun': nonoverlap_constraints, 'jac': nonoverlap_constraints_jac}
    ]

                break

def resolve_overlaps(centers, radii, max_iter=200):
    """Resolve overlaps by pushing circles apart, then project to bounds."""
    centers = centers.copy()
    radii = radii.copy()
    
    for iteration in range(max_iter):
        max_overlap = 0
        for i in range(N):
            for j in range(i + 1, N):
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = np.sqrt(dx * dx + dy * dy) + 1e-15
                required = radii[i] + radii[j]
                if dist < required:
                    overlap = required - dist
                    max_overlap = max(max_overlap, overlap)
                    nx = dx / dist
                    ny = dy / dist
                    shift = overlap / 2.0
                    centers[i][0] += nx * shift
                    centers[i][1] += ny * shift
                    centers[j][0] -= nx * shift
                    centers[j][1] -= ny * shift
        
        # Project to unit square
        for i in range(N):
            centers[i][0] = np.clip(centers[i][0], radii[i], 1.0 - radii[i])
            centers[i][1] = np.clip(centers[i][1], radii[i], 1.0 - radii[i])
        
        if max_overlap < 1e-10:
            break
    
    return centers, radii
