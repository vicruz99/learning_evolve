import numpy as np

def run_packing():
    """
    Optimizes the packing of 26 circles in a unit square to maximize sum of radii.
    Uses a repulsive force simulation with radial inflation.
    """
    n = 26
    
    # --- Initialization ---
    # Start with a 5x5 grid pattern (25 circles) + 1 random circle to get 26.
    # This provides a dense, structured starting point.
    coords = []
    x_vals = np.linspace(0.1, 0.9, 5)
    y_vals = np.linspace(0.1, 0.9, 5)
    
    for y in y_vals:
        for x in x_vals:
            coords.append([x, y])
    
    # We have 25 coordinates. Add one more at a random valid position.
    coords.append([0.5, 0.5]) 
    
    centers = np.array(coords, dtype=float)
    
    # Add slight random perturbation to avoid symmetry locking
    np.random.seed(123)
    centers += np.random.randn(n, 2) * 0.02
    centers = np.clip(centers, 0.1, 0.9)
    
    # Initial radii: small enough to avoid immediate overlap
    radii = np.ones(n) * 0.04
    
    # --- Optimization Parameters ---
    max_iter = 3000
    lr_pos = 0.05       # Learning rate for position updates
    lr_rad_base = 0.002 # Base learning rate for radius growth
    
    # --- Simulation Loop ---
    for step in range(max_iter):
        # Decay radius learning rate to settle into a stable high-value packing
        lr_rad = lr_rad_base * (1.0 - step / max_iter)
        if lr_rad < 1e-6:
            lr_rad = 1e-6
            
        # --- 1. Compute Pairwise Distances and Overlaps ---
        # Vectorized distance calculation
        # diffs: (n, n, 2)
        diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
        dists = np.linalg.norm(diffs, axis=2)
        
        # Mask to ignore self-distance (diagonal)
        mask = np.ones((n, n), dtype=bool)
        np.fill_diagonal(mask, False)
        
        # Calculate overlap: (r_i + r_j) - dist
        # Positive value means overlap
        r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
        overlaps = r_sum - dists
        
        # Only consider positive overlaps
        overlaps_pos = np.maximum(0.0, overlaps)
        
        # --- 2. Compute Repulsive Forces on Positions ---
        # Force direction is along the vector between centers
        # F = overlap * (diff / dist)
        # Handle division by zero
        safe_dists = np.where(dists > 1e-8, dists, 1e-8)
        force_vec = overlaps_pos[:, :, np.newaxis] * (diffs / safe_dists[:, :, np.newaxis])
        
        # Sum forces exerted by all other circles on each circle
        pos_forces = np.sum(force_vec, axis=1)

        # Force to push circle away from walls if it violates r_i <= x <= 1-r_i
        wall_forces = np.zeros_like(centers)

        viol = radii - centers[:, 0]
        wall_forces[:, 0] += np.maximum(0.0, viol)
        
        # Right wall (x > 1-r)
        viol = centers[:, 0] - (1.0 - radii)
        wall_forces[:, 0] -= np.maximum(0.0, viol)
        
        # Bottom wall (y < r)
        viol = radii - centers[:, 1]
        wall_forces[:, 1] += np.maximum(0.0, viol)
        
        # Top wall (y > 1-r)
        viol = centers[:, 1] - (1.0 - radii)
        wall_forces[:, 1] -= np.maximum(0.0, viol)
        
        total_pos_forces = pos_forces + wall_forces
        
        # --- 4. Update Positions ---
        centers += lr_pos * total_pos_forces
        # Clamp centers to ensure they don't fly out numerically
        # Note: The forces should keep them within [r, 1-r], but clamping to [0, 1] is safe
        centers = np.clip(centers, 0.0, 1.0)
        
        # --- 5. Update Radii ---
        # We want to maximize sum of radii.
        # Strategy: Grow radii, but if overlaps are high, the positional forces will fight it.
        # To make it more explicit, we can reduce radius growth proportional to overlap pressure.
        
        # Calculate total overlap pressure on each circle (sum of overlaps with others)
        overlap_pressure = np.sum(overlaps_pos, axis=1)
        
        # Update rule: r_i += growth - penalty * pressure
        # This creates a balance. If pressure is high, growth stops or radii shrink.
        penalty_factor = 0.1 
        radii_update = lr_rad * (1.0 - penalty_factor * overlap_pressure)
        
        # Ensure radii don't become negative
        radii_update = np.maximum(radii_update, 0.0)
        
        radii += radii_update

    # 1. Ensure centers allow for radius

    for i in range(n):
        r = radii[i]
        x, y = centers[i]
        
        # Clamp x
        if x < r:
            x = r
        elif x > 1 - r:
            x = 1 - r
            
        # Clamp y
        if y < r:
            y = r
        elif y > 1 - r:
            y = 1 - r
            
        centers[i] = [x, y]
        
    # 2. Resolve remaining tiny overlaps by slightly reducing radii if needed
    # Check pairwise
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            required_dist = radii[i] + radii[j]
            if dist < required_dist - 1e-12:
                overlap = required_dist - dist
                # Reduce radii slightly to resolve
                # Distribute reduction equally
                reduction = overlap / 2.0 + 1e-7
                radii[i] = max(0, radii[i] - reduction)
                radii[j] = max(0, radii[j] - reduction)
                
                # After reducing radii, re-check boundary constraints
                # (Reducing radius makes boundary constraints easier to satisfy, so no action needed)

    # 3. Final boundary check
    for i in range(n):
        r = radii[i]
        x, y = centers[i]
        # Ensure strict boundary
        x = max(r, min(1 - r, x))
        y = max(r, min(1 - r, y))
        centers[i] = [x, y]

    sum_radii = np.sum(radii)
    
    return centers, radii, float(sum_radii)
