6. **Validation**: Apply a small safety epsilon to radii to guarantee strict non-overlap within numerical tolerance.

import numpy as np

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Pack 26 circles in a unit square to maximize the sum of radii.
    Uses a force-directed expansion algorithm with multiple restarts.
    """
    np.random.seed(42)
    N = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Parameters for the expansion optimizer
    NUM_TRIALS = 80
    MAX_STEPS = 4000
    INIT_LR = 0.06
    LR_DECAY = 0.9992
    DELTA_R = 0.00012
    REPULSION_SCALE = 2.5
    BOUNDARY_SCALE = 5.0
    EPSILON = 1e-12

    for trial in range(NUM_TRIALS):
        # 1. Initialization
        if trial < 20:
            # Hexagonal-ish grid initialization
            pts = []
            y = 0.2
            row = 0
            while len(pts) < N:
                x_start = 0.15 if row % 2 == 1 else 0.2
                x = x_start
                while x <= 0.85 and len(pts) < N:
                    pts.append([x, y])
                    x += 0.18
                y += 0.14
                row += 1
            pts = np.array(pts[:N])
            # Add slight noise to break symmetry
            pts += np.random.randn(*pts.shape) * 0.015
        else:
            # Random initialization
            pts = np.random.rand(N, 2) * 0.6 + 0.2
            
        pts = np.clip(pts, 0.01, 0.99)
        
        # 2. Expansion & Repulsion Loop
        r = 0.005
        lr = INIT_LR
        
        for step in range(MAX_STEPS):
            r += DELTA_R
            forces = np.zeros_like(pts)
            
            # Boundary forces
            mask_left = pts[:, 0] < r
            mask_right = pts[:, 0] > 1.0 - r
            mask_down = pts[:, 1] < r
            mask_up = pts[:, 1] > 1.0 - r
            
            forces[mask_left, 0] += (r - pts[mask_left, 0]) * BOUNDARY_SCALE
            forces[mask_right, 0] -= (pts[mask_right, 0] - (1.0 - r)) * BOUNDARY_SCALE
            forces[mask_down, 1] += (r - pts[mask_down, 1]) * BOUNDARY_SCALE
            forces[mask_up, 1] -= (pts[mask_up, 1] - (1.0 - r)) * BOUNDARY_SCALE
            
            # Pairwise repulsion
            diff = pts[:, None] - pts[None, :]
            dist = np.sqrt(np.sum(diff**2, axis=2))
            np.fill_diagonal(dist, 1.0)  # Prevent division by zero
            
            overlap = 2.0 * r - dist
            overlap = np.maximum(overlap, 0.0)
            
            safe_dist = np.where(dist > 1e-7, dist, 1.0)
            direction = diff / safe_dist[:, :, None]
            
            repulsion = direction * (overlap[:, :, None] * REPULSION_SCALE)
            forces += repulsion.sum(axis=1)
            
            # Update positions
            pts += lr * forces
            pts = np.clip(pts, 0.0, 1.0)
            
            # Decay learning rate to stabilize
            lr *= LR_DECAY
            
            # If system is very stable, we can skip some steps or continue expanding
            # The loop naturally handles stability via force magnitude
            
        # 3. Evaluate Packing Quality
        # Compute pairwise distances
        diff = pts[:, None] - pts[None, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        np.fill_diagonal(dists, 2.0)
        
        min_pair_dist = dists.min()
        r_pair = min_pair_dist / 2.0
        
        # Compute boundary clearance
        r_bound = min(pts[:, 0].min(), 1.0 - pts[:, 0].max(),
                      pts[:, 1].min(), 1.0 - pts[:, 1].max())
        
        # The actual feasible equal radius
        r_actual = min(r_pair, r_bound)
        
        # Apply safety epsilon to guarantee validation passes
        r_actual *= (1.0 - EPSILON)
        
        current_sum = 26.0 * r_actual
        
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = pts.copy()
            best_radii = np.full(N, r_actual)
            
        # Early termination if we exceed a high threshold
        if best_sum > 2.75:
            break
            
    return best_centers, best_radii, best_sum
