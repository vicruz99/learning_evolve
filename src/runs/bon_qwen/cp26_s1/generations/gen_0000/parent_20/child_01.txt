from scipy.optimize import minimize

def objective_function(v, n, beta):
    c = np.column_stack((v[0::3], v[1::3]))
    r = v[2::3]
    val = -np.sum(r)
    
    for i in range(n):
        for j in range(i + 1, n):
            d = np.sqrt(np.sum((c[i] - c[j]) ** 2))
            if d < r[i] + r[j]:
                val += beta * (r[i] + r[j] - d) ** 2
        for lim in [c[i, 0], 1.0 - c[i, 0], c[i, 1], 1.0 - c[i, 1]]:
            if r[i] > lim:
                val += beta * (r[i] - lim) ** 2
    return val
