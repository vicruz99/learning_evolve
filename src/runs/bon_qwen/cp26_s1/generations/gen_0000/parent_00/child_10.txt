import numpy as np
import scipy.optimize as opt
import math

def get_constraints(centers, radii):
    """
    Helper to compute constraint values for SLSQP.
    Returns a list of constraint dicts.
    Constraints are of form fun(x) >= 0.
    """
    n = centers.shape[0]
    constraints = []

    # Boundary constraints: x - r >= 0, x + r <= 1, etc.
    # x - r >= 0  =>  x - r >= 0
    # x + r <= 1  =>  1 - (x + r) >= 0
    for i in range(n):
        idx_x = 3 * i
        idx_y = 3 * i + 1
        idx_r = 3 * i + 2
        
        # x - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda x, i=i: x[idx_x] - x[idx_r]
        })
        # 1 - x - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda x, i=i: 1.0 - x[idx_x] - x[idx_r]
        })
        # y - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda x, i=i: x[idx_y] - x[idx_r]
        })
        # 1 - y - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda x, i=i: 1.0 - x[idx_y] - x[idx_r]
        })

    # Overlap constraints: dist^2 - (r_i + r_j)^2 >= 0
    for i in range(n):
        for j in range(i + 1, n):
            idx_xi, idx_yi, idx_ri = 3*i, 3*i+1, 3*i+2
            idx_xj, idx_yj, idx_rj = 3*j, 3*j+1, 3*j+2
            
            def overlap_con(x, i=i, j=j):
                xi, yi, ri = x[idx_xi], x[idx_yi], x[idx_ri]
                xj, yj, rj = x[idx_xj], x[idx_yj], x[idx_rj]
                dist_sq = (xi - xj)**2 + (yi - yj)**2
                sum_r = ri + rj
                return dist_sq - sum_r**2

            constraints.append({
                'type': 'ineq',
                'fun': overlap_con
            })
            
    return constraints

def objective(x):
    """
    Objective function to minimize: -sum(radii)
    """
    radii = x[2::3]
    return -np.sum(radii)

def run_packing():
    n = 26
    num_vars = n * 3
    
    # Bounds: x, y in [0, 1], r in [0, 0.5] (tighter upper bound for r is 0.5)
    # Actually r can be at most 0.5.
    bounds = []
    for i in range(n):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 0.5)) # r

    best_val = -np.inf
    best_x = None

    # We will try a few initial configurations
    # Config 1: Hexagonal packing attempt
    # Config 2: Random valid packing
    
    # Helper to create initial guess
    def create_initial_guess(method='hex'):
        x0 = np.zeros(num_vars)
        if method == 'hex':
            # Try to pack in hexagonal lattice
            # Estimate radius to fit 26 circles. 
            # Area approx 1. Density ~0.9. 26 * pi * r^2 ~ 0.9 => r ~ 0.105
            # But boundary effects reduce this. Let's start with r=0.09 to be safe.
            r_start = 0.09
            
            # Rows
            # Height of hex row is sqrt(3)*r
            # Width of row with k circles is (k-1)*2r + 2r = 2kr? No, span is (k-1)*2r.
            # Extent is 2r + (k-1)*2r = 2kr.
            # We need 26 circles.
            # Maybe 5 rows? 6, 5, 5, 5, 5? 
            # Let's just place them in a grid-like hex pattern and let optimizer fix.
            
            row_counts = [6, 5, 5, 5, 5] # Sum 26
            # Adjust r_start to fit 6 circles in width? 12*r <= 1 => r <= 0.0833.
            # If we use r=0.08, it's safe.
            r_start = 0.08
            
            current_idx = 0
            y_pos = r_start
            for row_idx, count in enumerate(row_counts):
                # Offset for hex
                offset = r_start if row_idx % 2 == 1 else 0.0
                # Calculate width occupied
                # Width needed = (count-1)*2*r + 2*r = count * 2r? 
                # Actually centers span (count-1)*2r.
                # Start x at r_start + offset.
                # But if offset is r, first center is at r + r = 2r?
                # Standard hex: Row 0: centers at r, 3r, 5r...
                # Row 1: centers at 2r, 4r, 6r... (shifted by r)
                
                # Let's align to boundaries properly.
                # For a row of 'count' circles, we want to center them or align left.
                # Let's distribute them evenly in [r, 1-r]
                
                # To allow optimization to work, just place them validly.
                # If count is large, we might violate width with fixed r.
                # But with r=0.08, 6 circles width = 6*0.16 = 0.96. Fits in 1.0.
                
                for col in range(count):
                    if row_idx % 2 == 0:
                        # Even row: 0, 2r, 4r...
                        # x = r + col * 2r
                        x_pos = r_start + col * 2 * r_start
                    else:
                        # Odd row: r, 3r, 5r... (shifted by r)
                        # But wait, if we shift by r, first is at 2r?
                        # If we want to pack tightly, maybe shift by r.
                        # x = r_start + r_start + col * 2 * r_start = 2r + 2cr?
                        # Actually offset is r_start.
                        x_pos = r_start + offset + col * 2 * r_start
                    
                    # Check if x_pos is valid?
                    # With r=0.08, 6 circles:
                    # Row 0: 0.08, 0.24, 0.40, 0.56, 0.72, 0.88. (Last + r = 0.96 <= 1). OK.
                    # Row 1 (offset 0.08): 0.16, 0.32, 0.48, 0.64, 0.80. (5 circles). OK.
                    
                    if current_idx < n:
                        x0[3*current_idx] = x_pos
                        x0[3*current_idx+1] = y_pos
                        x0[3*current_idx+2] = r_start
                        current_idx += 1
                
                # Vertical spacing sqrt(3)*r
                y_pos += math.sqrt(3) * r_start
        
        elif method == 'random':
            # Random valid packing with small radius
            r_start = 0.05
            for i in range(n):
                x0[3*i] = np.random.uniform(r_start, 1-r_start)
                x0[3*i+1] = np.random.uniform(r_start, 1-r_start)
                x0[3*i+2] = r_start

        return x0

    # Define constraints only once? No, need to create them.
    # But creating 400+ constraints objects every time is slow?
    # Actually we pass them to minimize.
    # Let's define them inside the loop or once.
    # SLSQP needs the constraints list.
    
    # We can define a function that returns constraints
    # But `minimize` expects a list of dicts.
    
    constraints_list = []
    # Boundary constraints
    for i in range(n):
        idx_x = 3 * i
        idx_y = 3 * i + 1
        idx_r = 3 * i + 2
        
        constraints_list.append({
            'type': 'ineq',
            'fun': lambda x, i=i: x[idx_x] - x[idx_r]
        })
        constraints_list.append({
            'type': 'ineq',
            'fun': lambda x, i=i: 1.0 - x[idx_x] - x[idx_r]
        })
        constraints_list.append({
            'type': 'ineq',
            'fun': lambda x, i=i: x[idx_y] - x[idx_r]
        })
        constraints_list.append({
            'type': 'ineq',
            'fun': lambda x, i=i: 1.0 - x[idx_y] - x[idx_r]
        })

    # Overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            idx_xi, idx_yi, idx_ri = 3*i, 3*i+1, 3*i+2
            idx_xj, idx_yj, idx_rj = 3*j, 3*j+1, 3*j+2
            
            constraints_list.append({
                'type': 'ineq',
                'fun': (lambda x, i=i, j=j: (x[idx_xi] - x[idx_xj])**2 + (x[idx_yi] - x[idx_yj])**2 - (x[idx_ri] + x[idx_rj])**2)
            })

    # Run optimization multiple times
    methods_to_try = ['hex', 'random', 'random']
    
    for method in methods_to_try:
        print(f"Trying method: {method}")
        x0 = create_initial_guess(method)
        
        try:
            res = opt.minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints_list, 
                               options={'maxiter': 1000, 'ftol': 1e-9})
            
            current_sum = -res.fun
            if current_sum > best_val:
                best_val = current_sum
                best_x = res.x.copy()
                print(f"New best sum: {best_val}")
        except Exception as e:
            print(f"Optimization failed with {method}: {e}")

    if best_x is None:
        # Fallback to a simple grid
        # 5x5 grid r=0.1, but we need 26. 
        # Just put 26 circles with small radius
        x0 = create_initial_guess('hex')
        # If optimization failed completely, return the initial guess (valid?)
        # Initial guess r=0.08 is valid.
        best_x = x0
        best_val = -objective(x0)

    # Extract results
    centers = np.array([[best_x[3*i], best_x[3*i+1]] for i in range(n)])
    radii = np.array([best_x[3*i+2] for i in range(n)])
    
    # Post-processing: Check if we can slightly expand radii if not tight
    # The optimizer should have found the max.
    # However, numerical noise might keep it slightly valid.
    
    # Final validation check (optional but good for debugging)
    # validate_packing(centers, radii)
    
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
