import numpy as np
from scipy.optimize import linprog, differential_evolution

def compute_max_radii(centers):
    """
    Solves the LP to find the maximum radii for a fixed set of centers.
    Maximizes sum(r_i) subject to non-overlap and boundary constraints.
    """
    n = centers.shape[0]
    
    # Objective: maximize sum(r_i) -> minimize -sum(r_i)
    c_obj = -np.ones(n)
    
    # Inequality constraints: A_ub * r <= b_ub
    # We collect constraints for boundaries and pairwise distances
    
    n_boundary_constraints = 4 * n
    n_pairwise_constraints = n * (n - 1) // 2
    total_constraints = n_boundary_constraints + n_pairwise_constraints
    
    A_ub = np.zeros((total_constraints, n))
    b_ub = np.zeros(total_constraints)
    
    idx = 0
    
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        x, y = centers[i]
        
        # r_i <= x_i
        A_ub[idx, i] = 1.0
        b_ub[idx] = x
        idx += 1
        
        # r_i <= 1 - x_i
        A_ub[idx, i] = 1.0
        b_ub[idx] = 1.0 - x
        idx += 1
        
        # r_i <= y_i
        A_ub[idx, i] = 1.0
        b_ub[idx] = y
        idx += 1
        
        # r_i <= 1 - y_i
        A_ub[idx, i] = 1.0
        b_ub[idx] = 1.0 - y
        idx += 1
        
    # Pairwise constraints: r_i + r_j <= distance(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dist
            idx += 1
            
    # Bounds for radii: r_i >= 0
    bounds = [(0, None) for _ in range(n)]
    
    # Solve LP
    # Using 'highs' solver for robustness
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return res.x
    else:
        # Fallback to small radii if LP fails (should not happen with valid centers)
        return np.full(n, 1e-6)

def evaluate_packing(centers_flat):
    """
    Objective function for optimization.
    Takes flat array of centers, returns negative sum of radii.
    """
    n = 26
    centers = centers_flat.reshape((n, 2))
    
    # Clip centers to [0, 1] to avoid invalid constraints in LP (though DE bounds handle this)
    centers = np.clip(centers, 0, 1)
    
    radii = compute_max_radii(centers)
    
    # If radii are effectively zero, return a bad score
    if np.any(radii <= 1e-6):
        return 0.0 
        
    return -np.sum(radii)
