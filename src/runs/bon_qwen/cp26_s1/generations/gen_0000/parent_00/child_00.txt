    *   To ensure exactly 26 circles fit within the $[0,1] \times [0,1]$ boundaries, I will generate a grid of points and select the 26 points closest to the center. This centers the cluster and handles boundary clipping naturally.
    *   Set an initial radius small enough to guarantee no overlaps (e.g., 0.05).

2.  **Optimization**:
    *   Use `scipy.optimize.minimize` with the `SLSQP` method to maximize the sum of radii.
    *   **Objective**: Maximize $\sum r_i$, which is equivalent to minimizing $-\sum r_i$.
    *   **Constraints**:
        *   **Boundary Constraints**: Ensure each circle is inside the square ($r_i \le x_i \le 1-r_i$ and $r_i \le y_i \le 1-r_i$).
        *   **Non-Overlap Constraints**: Ensure the distance between any two circle centers is at least the sum of their radii ($dist(c_i, c_j) \ge r_i + r_j$).
    *   **Variables**: The optimizer will adjust the 26 centers ($x, y$) and 26 radii ($r$) simultaneously.
    *   **Bounds**: Centers are bounded in $[0,1]$ and radii in $[0, 0.5]$.

import numpy as np
from scipy.optimize import minimize

def run_packing():
    n = 26
    
    # 1. Generate initial positions (Hexagonal lattice)
    # Create a grid of points with hexagonal spacing
    # We create a superset of points and pick the best 26
    points = []
    # Estimate spacing to fit roughly 26 points in 1x1
    # Area per point ~ 1/26 ~ 0.038. Hexagonal area factor ~ sqrt(3)/2 * d^2
    # Rough guess for spacing
    spacing = 0.15 
    
    # Generate points in a hexagonal grid
    for row in range(15):
        for col in range(15):
            x = col * spacing * 2
            y = row * spacing * np.sqrt(3)
            if col % 2 == 1:
                y += spacing * np.sqrt(3) / 2
            points.append([x, y])
            
    points = np.array(points)
    
    # Select 26 points closest to the center (0.5, 0.5) to center the packing
    center = np.array([0.5, 0.5])
    dists_to_center = np.linalg.norm(points - center, axis=1)
    indices = np.argsort(dists_to_center)[:n]
    initial_centers = points[indices]
    
    # Initial radius (small enough to not overlap)
    initial_radii = np.ones(n) * 0.05
    
    # 2. Define objective and constraints for optimization
    # We optimize variables: [x1, y1, r1, x2, y2, r2, ...]
    # Total variables = 3 * n = 78
    
    def objective(vars_flat):
        # vars_flat is shape (3*n,)
        # Structure: [x1, y1, r1, x2, y2, r2, ...]
        centers = vars_flat[0:2*n].reshape((n, 2))
        radii = vars_flat[2*n:3*n]
        # Maximize sum of radii => Minimize negative sum
        return -np.sum(radii)

    def constraint_boundary(vars_flat):
        centers = vars_flat[0:2*n].reshape((n, 2))
        radii = vars_flat[2*n:3*n]
        # x - r >= 0 => r - x <= 0
        # x + r <= 1 => x + r - 1 <= 0
        # Same for y
        
        cons = []
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            cons.append(r - x)       # x >= r
            cons.append(y - r)       # y >= r
            cons.append(x + r - 1)   # x + r <= 1
            cons.append(y + r - 1)   # y + r <= 1
        return cons

    def constraint_overlap(vars_flat):
        centers = vars_flat[0:2*n].reshape((n, 2))
        radii = vars_flat[2*n:3*n]
        
        cons = []
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.linalg.norm(centers[i] - centers[j])
                # dist >= r_i + r_j  =>  r_i + r_j - dist <= 0
                cons.append(radii[i] + radii[j] - dist)
        return cons

    # Flatten initial state
    x0 = np.concatenate([initial_centers.flatten(), initial_radii])
    
    # Bounds: centers in [0, 1], radii in [0, 0.5]
    bounds = []
    for i in range(n):
        bounds.append((0, 1)) # x
        bounds.append((0, 1)) # y
        bounds.append((1e-6, 0.5)) # r (lower bound slightly > 0)

    # Constraints definition for SLSQP
    # Inequality constraints: g(x) >= 0. 
    # Our functions return values <= 0 for valid. 
    # So we need to negate them or use 'ineq' with reversed logic.
    # scipy 'ineq' means g(x) >= 0.
    # My constraints: val <= 0. So pass -val.
    
    cons_list = []
    
    # Boundary constraints
    # We can pass a function that returns an array of constraint values
    def boundary_ineq(vars_flat):
        vals = constraint_boundary(vars_flat)
        return -np.array(vals) # Flip sign for >= 0

    def overlap_ineq(vars_flat):
        vals = constraint_overlap(vars_flat)
        return -np.array(vals) # Flip sign for >= 0

    cons = {
        'type': 'ineq', 
        'fun': boundary_ineq,
        'jac': '2-point' # numerical gradient
    }
    
    # Overlap constraint is large (325 constraints), might be slow with analytical jac.
    # We'll try without jac first or use 2-point.
    cons_overlap = {
        'type': 'ineq', 
        'fun': overlap_ineq
    }

    # Run optimization multiple times with slight perturbations to find global max
    best_sum_radii = -np.inf
    best_result = None

    # We will run a few iterations. 
    # To speed up, we might just run one robust optimization.
    # SLSQP is good for medium problems.
    
    # Let's try 3 random seeds (permutations of initial order doesn't matter much, 
    # but slight noise helps)
    
    for _ in range(3):
        # Add small noise to centers
        noise_centers = initial_centers + np.random.normal(0, 0.01, initial_centers.shape)
        # Clamp noise
        noise_centers = np.clip(noise_centers, 0.05, 0.95)
        x0_noisy = np.concatenate([noise_centers.flatten(), initial_radii])
        
        try:
            res = minimize(
                objective, 
                x0_noisy, 
                method='SLSQP', 
                bounds=bounds,
                constraints=[cons, cons_overlap],
                options={'ftol': 1e-9, 'maxiter': 1000, 'disp': False}
            )
            
            if res.success:
                current_sum = -res.fun
                if current_sum > best_sum_radii:
                    best_sum_radii = current_sum
                    best_result = res
        except Exception:
            continue

    if best_result is None:
        # Fallback to initial
        best_result = type('obj', (object,), {'x': x0})()
        best_sum_radii = -objective(x0)

    # Extract solution
    final_vars = best_result.x
    final_centers = final_vars[0:2*n].reshape((n, 2))
    final_radii = final_vars[2*n:3*n]
    final_sum = np.sum(final_radii)

    # Final validation and clipping to ensure strict compliance
    # Sometimes optimization ends on boundary, slight nudge might be needed?
    # The constraints should handle it.
    
    return final_centers, final_radii, final_sum
