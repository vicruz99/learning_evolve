    c_obj = -np.ones(n)
    A_ub = []
    b_ub = []
    
    # Pairwise non-overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            d = np.linalg.norm(centers[i] - centers[j])
            row = np.zeros(n)
            row[i] = 1
            row[j] = 1

            b_ub.append(d)
    
    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        row = np.zeros(n)
        row[i] = 1
        for val in [x, 1 - x, y, 1 - y]:
            A_ub.append(row.copy())
            b_ub.append(max(0, val))

    b_ub = np.array(b_ub)

    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')

                    r_max = min(r_max, d - radii[j])
            radii[i] = max(0.0005, r_max)

        # Compute repulsion forces for overlapping/nearby circles
        forces = np.zeros_like(centers)
        for i in range(n):
            for j in range(i + 1, n):
                diff = centers[i] - centers[j]
                d = np.linalg.norm(diff)
                if d < 1e-10:
                    continue
                overlap = radii[i] + radii[j] - d
                if overlap > -0.03:
                    f = max(0, overlap + 0.03) / d
                    forces[i] += diff * f
                    forces[j] -= diff * f

        centers += lr * forces
        centers = np.clip(centers, 0.002, 0.998)

    return centers


def run_packing():
    n = N_CIRCLES

    # Multiple row configurations to explore different packings
    row_configs = [
        [6, 5, 6, 5, 4],
        [5, 6, 5, 6, 4],
        [7, 6, 6, 7],
        [6, 6, 6, 6, 2],
        [5, 5, 5, 5, 6],
        [4, 6, 6, 6, 4],
        [6, 4, 6, 4, 6],
        [5, 7, 5, 9],
        [8, 6, 6, 6],
        [7, 7, 7, 5],
        [6, 7, 6, 7],
        [5, 8, 5, 8],
    ]

    best_sum = -1
    best_centers = None
    best_radii = None

    for row_config in row_configs:
        if sum(row_config) != n:
            continue

        for seed in range(20):
            centers = init_hexagonal(n, seed, row_config)
            opt_centers = expand_and_optimize(centers, n, steps=500)
            s, radii = solve_radii_lp(opt_centers)

            if s > best_sum:
                best_sum = s
                best_centers = opt_centers.copy()
                best_radii = radii.copy()

            # Try perturbation and re-optimize
            perturbed = opt_centers + np.random.uniform(-0.015, 0.015, opt_centers.shape)
            perturbed = np.clip(perturbed, 0.01, 0.99)
            opt_centers2 = expand_and_optimize(perturbed, n, steps=300)
            s2, radii2 = solve_radii_lp(opt_centers2)

            if s2 > best_sum:
                best_sum = s2
                best_centers = opt_centers2.copy()
                best_radii = radii2.copy()

    return best_centers, best_radii, best_sum
