def compute_radii(centers):
    """Compute the maximum possible radius for each circle given their centers."""
    # Boundary constraints
    r = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                   np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    
    # Inter-circle constraints
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    r = np.minimum(r, dists.min(axis=1) / 2.0)
    return r
