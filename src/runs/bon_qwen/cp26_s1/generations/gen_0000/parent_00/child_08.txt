import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def get_initial_guess():
    """Generate a hexagonal lattice initial placement scaled to the unit square."""
    pts = []
    row = 0
    while len(pts) < N_CIRCLES:
        for col in range(6):
            x = col * np.sqrt(3) / 2.0
            y = row * 1.5
            if col % 2 == 1:
                y += 0.75
            pts.append([x, y])
        row += 1
        
    xs = np.array([p[0] for p in pts])
    ys = np.array([p[1] for p in pts])
    
    # Normalize coordinates to [0.1, 0.9] to start strictly inside the square
    for v in [xs, ys]:
        mn, mx = v.min(), v.max()
        v[:] = (v - mn) / (mx - mn) * 0.8 + 0.1
        
    # Compute minimum distance between any two points
    min_dist = np.inf
    for i in range(N_CIRCLES):
        for j in range(i+1, N_CIRCLES):
            d = np.sqrt((xs[i]-xs[j])**2 + (ys[i]-ys[j])**2)
            if d < min_dist:
                min_dist = d
                
    # Set initial radius safely below half the minimum distance
    r_init = min_dist / 3.0
    
    x0 = np.zeros(3 * N_CIRCLES)
    for i in range(N_CIRCLES):
        x0[3*i] = xs[i]
        x0[3*i+1] = ys[i]
        x0[3*i+2] = r_init
    return x0

def objective_func(z):
    """Objective: maximize sum of radii -> minimize negative sum."""
    return -np.sum(z[2::3])

def constraint_func(z):
    """
    Returns an array of inequality constraint values (must be >= 0).
    Constraints:
    1. Circles inside unit square boundaries
    2. Radii non-negative
    3. No overlaps (squared distance >= sum of radii squared)
    """
    centers = np.column_stack((z[0::3], z[1::3]))
    radii = z[2::3]
    val = []
    
    # Boundary constraints: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    val.extend(centers[:, 0] - radii)
    val.extend(1.0 - centers[:, 0] - radii)
    val.extend(centers[:, 1] - radii)
    val.extend(1.0 - centers[:, 1] - radii)
    
    # Non-negative radii
    val.extend(radii)
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    for i in range(N_CIRCLES):
        for j in range(i+1, N_CIRCLES):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            val.append(dx**2 + dy**2 - (radii[i] + radii[j])**2)
            
    return np.array(val)

def run_packing():
    """Optimize circle packing to maximize sum of radii."""
    x0 = get_initial_guess()
    bounds = [(0.0, 1.0)] * (2 * N_CIRCLES) + [(0.0, 0.5)] * N_CIRCLES
    cons = {'type': 'ineq', 'fun': constraint_func}
    
    res = minimize(objective_func, x0, method='SLSQP', bounds=bounds, constraints=cons, 
                   options={'maxiter': 2000, 'ftol': 1e-10, 'disp': False})
                   
    centers = np.column_stack((res.x[0::3], res.x[1::3]))
    radii = np.maximum(res.x[2::3], 0.0)  # Clamp to avoid tiny negative values from numerical noise
    return centers, radii, np.sum(radii)
