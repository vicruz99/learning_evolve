import numpy as np
from scipy.optimize import minimize, linprog


def make_hex_config(n):
    centers = np.zeros((n, 2))
    idx = 0
    row_cols = [5, 5, 5, 5, 6]
    row_y = np.linspace(0.12, 0.88, 5)

    for row in range(5):
        nc = row_cols[row]
        if nc > 1:
            dx = 0.8 / (nc - 1)
        else:
            dx = 0
        offset = 0.1 + (row % 2) * dx * 0.5
        for col in range(nc):
            x = offset + col * dx
            y = row_y[row]
            centers[idx] = [x, y]
            idx += 1

    radii = np.ones(n) * 0.06
    return centers, radii


def obj_func(x, n):
    return -np.sum(x[2 * n :])


def no_overlap(x, n, i, j):
    xi, yi = x[2 * i], x[2 * i + 1]
    xj, yj = x[2 * j], x[2 * j + 1]
    ri, rj = x[2 * n + i], x[2 * n + j]
    dx = xi - xj
    dy = yi - yj
    return dx * dx + dy * dy - (ri + rj) ** 2


def bound_left(x, n, i):
    return x[2 * i] - x[2 * n + i]


def bound_right(x, n, i):
    return 1 - x[2 * i] - x[2 * n + i]


def bound_bottom(x, n, i):
    return x[2 * i + 1] - x[2 * n + i]


def bound_top(x, n, i):
    return 1 - x[2 * i + 1] - x[2 * n + i]


def pos_radius(x, n, i):
    return x[2 * n + i]


def make_bounds(n):
    b = []
    for i in range(2 * n):
        b.append((0, 1))
    for i in range(n):
        b.append((1e-8, 0.5))
    return b


def make_constraints(n):
    cons = []
    for i in range(n):
        for j in range(i + 1, n):
            cons.append({"type": "ineq", "fun": no_overlap, "args": (n, i, j)})
        cons.append({"type": "ineq", "fun": bound_left, "args": (n, i)})
        cons.append({"type": "ineq", "fun": bound_right, "args": (n, i)})
        cons.append({"type": "ineq", "fun": bound_bottom, "args": (n, i)})
        cons.append({"type": "ineq", "fun": bound_top, "args": (n, i)})
        cons.append({"type": "ineq", "fun": pos_radius, "args": (n, i)})
    return cons


def optimize_radii_lp(centers, n):
    c = -np.ones(n)
    A_ub_rows = []
    b_ub_vals = []

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            row = np.zeros(n)
            row[i] = 1
            row[j] = 1
            A_ub_rows.append(row)
            b_ub_vals.append(dist)

    for i in range(n):
        for bv in [
            centers[i, 0],
            1 - centers[i, 0],
            centers[i, 1],
            1 - centers[i, 1],
        ]:
            row = np.zeros(n)
            row[i] = 1
            A_ub_rows.append(row)
            b_ub_vals.append(bv)

    A_ub = np.array(A_ub_rows)
    b_ub = np.array(b_ub_vals)
    bounds = [(0, None)] * n

    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method="highs")

    if result.success:
        return result.x
    return np.ones(n) * 0.02


def run_packing():
    n = 26

    best_sum = 0
    best_c = None
    best_r = None

    for trial in range(8):
        np.random.seed(trial * 13 + 3)

        centers, radii = make_hex_config(n)
        centers += np.random.uniform(-0.04, 0.04, (n, 2))
        centers = np.clip(centers, 0.05, 0.95)

        for iteration in range(30):
            radii = optimize_radii_lp(centers, n)

            x0 = np.concatenate([centers.flatten(), radii])

            result = minimize(
                obj_func,
                x0,
                args=(n,),
                method="SLSQP",
                bounds=make_bounds(n),
                constraints=make_constraints(n),
                options={"maxiter": 600, "ftol": 1e-14},
            )

            centers = result.x[: 2 * n].reshape(n, 2)
            radii = result.x[2 * n :]
            centers = np.clip(centers, radii[np.newaxis, :], 1 - radii[np.newaxis, :])
            radii = np.clip(radii, 1e-8, 0.5)

        current_sum = np.sum(radii)
        if current_sum > best_sum:
            best_sum = current_sum
            best_c = centers.copy()
            best_r = radii.copy()

    # Final refinement
    x0 = np.concatenate([best_c.flatten(), best_r])
    result = minimize(
        obj_func,
        x0,
        args=(n,),
        method="SLSQP",
        bounds=make_bounds(n),
        constraints=make_constraints(n),
        options={"maxiter": 3000, "ftol": 1e-15},
    )
    best_c = result.x[: 2 * n].reshape(n, 2)
    best_r = result.x[2 * n :]
    best_c = np.clip(best_c, best_r[np.newaxis, :], 1 - best_r[np.newaxis, :])
    best_r = np.clip(best_r, 1e-8, 0.5)

    return best_c, best_r, np.sum(best_r)
