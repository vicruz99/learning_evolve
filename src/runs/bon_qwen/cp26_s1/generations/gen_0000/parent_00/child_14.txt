import numpy as np
from scipy.optimize import minimize

def run_packing():
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    
    Returns:
        centers: np.array of shape (26, 2)
        radii: np.array of shape (26,)
        sum_radii: float
    """
    N = 26
    
    # 1. Initialize a hexagonal-like grid layout
    # Row sizes: 5, 4, 5, 4, 5, 3 (Sum = 26)
    row_sizes = [5, 4, 5, 4, 5, 3]
    
    centers = []
    current_y = 0.15  # Initial Y offset
    
    for i, size in enumerate(row_sizes):
        # Shift even rows (0-indexed) by 0.5 units relative to circle spacing
        # Spacing is roughly 0.2, so shift is 0.1
        if i % 2 == 1:
            current_x = 0.15 
        else:
            current_x = 0.05
            
        for _ in range(size):
            centers.append([current_x, current_y])
            current_x += 0.205 # Slightly more than 2*0.1 radius
            
        current_y += 0.175 # Vertical spacing approx sqrt(3)/2 * d
    
    centers = np.array(centers)
    radii = np.full(N, 0.10) # Start with optimistic radius
    
    best_sum = -np.inf
    best_result = None

    # 2. Optimization loop with multiple random restarts
    for restart in range(10):
        # Slight perturbation to avoid local minima in grid
        x0 = np.hstack((centers, radii)) + np.random.randn(78) * 0.005
        
        # Boundary limits for variables
        # x, y in [0, 1], r >= 0
        bounds = [(0, 1)] * N + [(0, 1)] * N + [(0, 0.5)] * N
        
        def objective(x_vars):
            r = x_vars[52:]
            return -np.sum(r) # Maximize sum of radii

        def constraints(x_vars):
            c = x_vars.reshape(N, 3)
            cons = []
            
            # Boundary constraints: r <= x <= 1-r  =>  x-r>=0, 1-x-r>=0
            for i in range(N):
                xi, yi, ri = c[i]
                cons.append(xi - ri)
                cons.append(1.0 - xi - ri)
                cons.append(yi - ri)
                cons.append(1.0 - yi - ri)
            
            # Non-overlap constraints: dist_ij >= ri + rj
            # Check all pairs
            for i in range(N):
                for j in range(i + 1, N):
                    dx = c[i, 0] - c[j, 0]
                    dy = c[i, 1] - c[j, 1]
                    dist = np.sqrt(dx**2 + dy**2)
                    cons.append(dist - c[i, 2] - c[j, 2])
            
            return np.array(cons)

        cons_obj = {'type': 'ineq', 'fun': constraints}
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, 
                           constraints=cons_obj, options={'maxiter': 500, 'disp': False})
            
            if res.success or res.fun < -2.636: # Found a better packing
                final_vars = res.x
                final_centers = final_vars[:52].reshape(N, 2)
                final_radii = final_vars[52:]
                final_sum = np.sum(final_radii)
                
                if final_sum > best_sum:
                    best_sum = final_sum
                    best_result = (final_centers, final_radii)
                    
        except Exception:
            continue

    if best_result is not None:
        return best_result[0], best_result[1], best_sum
    else:
        # Fallback to the initial guess if optimization fails completely
        return centers, radii, np.sum(radii)
