import numpy as np
import scipy.optimize as opt
import math

def generate_hexagonal_initial_guess(n_circles):
    """
    Generates an initial guess for circle centers and radii based on a hexagonal packing.
    """
    # We want to fit n_circles into a unit square.
    # Hexagonal packing has rows with spacing h = r * sqrt(3).
    # Horizontal spacing is 2r.
    # Let's estimate a radius r. 
    # Area ~ n * pi * r^2 / density. 
    # For n=26, r approx 0.09 - 0.10.
    
    # Let's try to construct rows.
    # We can try to fit as many rows as possible.
    # If we have 6 rows, we need to distribute 26 circles.
    # 26 / 6 = 4.33. So rows will have 4 or 5 circles.
    # Pattern: 5, 4, 5, 4, 5, 3 sums to 26.
    # Or 5, 5, 5, 5, 4, 2? 
    # A balanced pattern is better. 5, 4, 5, 4, 5, 3 is somewhat balanced.
    # Or 6 rows of roughly 4-5.
    # Let's try 5, 4, 5, 4, 5, 3.
    
    row_counts = [5, 4, 5, 4, 5, 3]
    
    # Estimate radius to fit 5 circles in width 1.
    # 5 circles of radius r need width 2r + 4*(2r) ? No.
    # 5 circles centered at x.
    # If aligned, x coords: r, r+2r, r+4r, r+6r, r+8r.
    # Max x = r + 8r = 9r. Must be <= 1. So r <= 1/9 = 0.111.
    # But for hexagonal, rows are shifted.
    # Row with 4 circles (offset by r) fits in width 1-r - (r + 3*2r + r)?
    # Let's just use a reasonable r like 0.09.
    
    r_est = 0.09
    centers = []
    
    # Calculate y coordinates for rows
    # Row height step = r * sqrt(3)
    dy = r_est * math.sqrt(3)
    y_start = r_est # First row center y
    
    # We need to fit 6 rows.
    # Last row center y = y_start + 5 * dy
    # y_end = r + 5*r*sqrt(3) + r <= 1 ?
    # r(1 + 5*sqrt(3) + 1) <= 1 ?
    # r(2 + 8.66) <= 1 => 10.66r <= 1 => r <= 0.0938.
    # So r=0.09 is safe.
    
    current_y = y_start
    total_circles = 0
    
    # We need to center the rows vertically and horizontally if possible, 
    # but simple top-left packing is fine for initialization.
    # Let's shift to center vertically.
    # Total height occupied by rows: 5 * dy.
    # Available space for centers: [r, 1-r]. Height 1-2r.
    # We can distribute rows evenly in [r, 1-r].
    
    y_coords = np.linspace(r_est, 1 - r_est, len(row_counts))
    
    for i, count in enumerate(row_counts):
        y = y_coords[i]
        # Determine x coordinates
        # For odd rows (index 0, 2, 4), count is 5. 
        # For even rows (index 1, 3, 5), count is 4 or 3.
        # Let's just pack them centered.
        
        # Width required for 'count' circles:
        # (count - 1) * 2r + 2r = count * 2r ?
        # Actually span is from r to 1-r.
        # If we place k circles, spacing 2r.
        # Total span = (k-1)*2r.
        # We can center this span in [0, 1].
        # Start x = (1 - (count-1)*2r) / 2.
        
        span = (count - 1) * 2 * r_est
        start_x = (1.0 - span) / 2.0
        
        for j in range(count):
            x = start_x + j * 2 * r_est
            centers.append([x, y])
            total_circles += 1
            
    centers = np.array(centers)
    radii = np.full(n_circles, r_est)
    
    # If we didn't generate exactly n_circles, pad or trim?
    # With [5, 4, 5, 4, 5, 3] sum is 26.
    assert len(centers) == n_circles, f"Generated {len(centers)} circles, expected {n_circles}"
    
    return centers, radii

def solve_packing(centers_init, radii_init):
    """
    Optimizes circle packing given an initial guess.
    """
    n = len(radii_init)
    
    # Variables: [x1, y1, r1, x2, y2, r2, ...]
    # Total 3n variables.
    x0 = np.zeros(3 * n)
    for i in range(n):
        x0[3*i] = centers_init[i, 0]
        x0[3*i+1] = centers_init[i, 1]
        x0[3*i+2] = radii_init[i]
        
    # Bounds for variables
    # x, y in [0, 1]
    # r >= 0
    # Also implicitly r <= x, r <= 1-x etc handled by constraints.
    # But we can set bounds [0, 1] for x,y and [0, 0.5] for r.
    bounds = []
    for i in range(n):
        bounds.append((0, 1)) # x
        bounds.append((0, 1)) # y
        bounds.append((0, 0.5)) # r
        
    # Constraints
    constraints = []
    
    # 1. Boundary constraints:
    # r_i <= x_i  => x_i - r_i >= 0
    # r_i <= 1-x_i => 1 - x_i - r_i >= 0
    # r_i <= y_i => y_i - r_i >= 0
    # r_i <= 1-y_i => 1 - y_i - r_i >= 0
    
    # We can define a function for boundary constraints
    def boundary_constraints(vars):
        c = []
        for i in range(n):
            x, y, r = vars[3*i], vars[3*i+1], vars[3*i+2]
            c.append(x - r)
            c.append(1 - x - r)
            c.append(y - r)
            c.append(1 - y - r)
        return c
    
    # SLSQP requires constraints in form func(x) >= 0 or = 0 or <= 0?
    # type='ineq' means func(x) >= 0.
    constraints.append({'type': 'ineq', 'fun': boundary_constraints})
    
    # 2. Non-overlap constraints
    # dist(i, j) >= r_i + r_j
    # sqrt((xi-xj)^2 + (yi-yj)^2) - ri - rj >= 0
    
    def overlap_constraints(vars):
        c = []
        for i in range(n):
            for j in range(i + 1, n):
                xi, yi, ri = vars[3*i], vars[3*i+1], vars[3*i+2]
                xj, yj, rj = vars[3*j], vars[3*j+1], vars[3*j+2]
                
                dist = math.sqrt((xi - xj)**2 + (yi - yj)**2)
                c.append(dist - ri - rj)
        return c
        
    # This creates ~325 constraints. Might be slow inside the loop?
    # But SLSQP evaluates them.
    # To speed up, we can use vectorization if possible, but loop is simple.
    constraints.append({'type': 'ineq', 'fun': overlap_constraints})
    
    # Objective: Maximize sum of radii => Minimize -sum(r)
    def objective(vars):
        total_r = 0.0
        for i in range(n):
            total_r += vars[3*i+2]
        return -total_r
        
    # Run optimization
    # We might run it multiple times with slight perturbations to avoid local minima
    best_vars = x0
    best_val = objective(x0)
    
    # Try a few random restarts or perturbations
    # But with good init, one run might suffice. 
    # Let's try 3 runs with small noise.
    
    for trial in range(3):
        # Perturb initial guess slightly
        if trial > 0:
            noise_scale = 0.005
            current_x0 = x0 + np.random.normal(0, noise_scale, size=x0.shape)
            # Clip to valid range roughly
            # x, y should stay within [0, 1]
            current_x0[0::3] = np.clip(current_x0[0::3], 0, 1)
            current_x0[1::3] = np.clip(current_x0[1::3], 0, 1)
            current_x0[2::3] = np.clip(current_x0[2::3], 0, 0.5)
        else:
            current_x0 = x0
            
        try:
            res = opt.minimize(objective, current_x0, method='SLSQP', bounds=bounds, constraints=constraints, options={'maxiter': 1000, 'ftol': 1e-9})
            if res.success and res.fun < best_val:
                best_val = res.fun
                best_vars = res.x
        except Exception:
            pass

    # Extract results
    centers = best_vars[0::3].reshape(-1, 2)
    radii = best_vars[2::3]
    
    # Post-processing: Ensure strict feasibility if needed?
    # The optimizer tries to satisfy constraints.
    # However, due to numerical tolerance, we might have tiny violations.
    # We can shrink radii slightly to ensure validity.
    # But the validation function allows 1e-12.
    
    return centers, radii, -best_val

def run_packing():
    n = 26
    # Generate initial guess
    centers_init, radii_init = generate_hexagonal_initial_guess(n)
    
    # Run optimization
    centers, radii, sum_r = solve_packing(centers_init, radii_init)
    
    return centers, radii, sum_r
