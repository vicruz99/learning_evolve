2. For fixed centers, the maximum radii can be found by solving a Linear Program (LP). The constraints are $r_i \le x_i, 1-x_i, y_i, 1-y_i$ and $r_i + r_j \le \text{dist}(c_i, c_j)$.
3. Define an objective function that takes center coordinates, solves the LP, and returns the negative sum of radii (for minimization).
4. Use a derivative-free optimizer (Nelder-Mead) with multiple starting points initialized on a perturbed hexagonal grid. This balances global exploration (via multiple starts) and local exploitation (via Nelder-Mead).
5. Clip centers to [0, 1] within the objective to handle unconstrained optimizer steps.
6. Return the best centers, corresponding radii from the final LP solve, and the sum of radii.

import numpy as np
from scipy.optimize import linprog, minimize

def compute_sum_radii(centers):
    """
    Solves an LP to find the maximum sum of radii for fixed centers.
    Constraints: circles inside [0,1]^2 and non-overlapping.
    """
    n = centers.shape[0]
    # Objective: maximize sum(r_i) -> minimize -sum(r_i)
    c_obj = np.ones(n) * -1.0
    
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        x, y = centers[i]
        row = np.zeros(n)
        row[i] = 1.0
        A_ub.extend([row, row, row, row])
        b_ub.extend([x, 1.0 - x, y, 1.0 - y])
        
    # Pairwise non-overlap constraints: r_i + r_j <= dist(i, j)
    dists = np.sqrt(((centers[:, None] - centers[None, :])**2).sum(axis=2))
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0, None)] * n
    
    # Use HiGHS solver for speed
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return -res.fun, res.x
    return 0.0, np.zeros(n)

def objective(centers_flat):
    """
    Objective function for optimization: negative sum of radii.
    """
    centers = centers_flat.reshape(-1, 2)
    # Ensure centers stay within bounds
    centers = np.clip(centers, 0.0, 1.0)
    s, _ = compute_sum_radii(centers)
    return -s

def run_packing():
    """
    Optimizes circle centers to maximize the sum of radii.
    Returns (centers, radii, sum_radii).
    """
    np.random.seed(42)
    n = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Initialize with a hexagonal grid pattern
    rows = 5
    cols = 6
    y_coords = np.linspace(0.15, 0.85, rows)
    x_coords = np.linspace(0.1, 0.9, cols)
    
    initial_centers = []
    idx = 0
    for i in range(rows):
        offset = 0.15 if i % 2 == 1 else 0.0
        for x in x_coords:
            if idx < n:
                initial_centers.append([x + offset, y_coords[i]])
                idx += 1
    initial_centers = np.array(initial_centers[:n])
    
    # Generate multiple starting points with perturbations
    starts = [initial_centers]
    for _ in range(8):
        p = initial_centers.copy()
        p += np.random.randn(*p.shape) * 0.05
        p = np.clip(p, 0.05, 0.95)
        starts.append(p)
        
    # Optimize from each starting point
    for start in starts:
        res = minimize(objective, start.flatten(), method='Nelder-Mead',
                       options={'maxiter': 3000, 'xatol': 1e-9, 'fatol': 1e-10})
        s = -res.fun
        if s > best_sum:
            best_sum = s
            best_centers = res.x.reshape(-1, 2)
            
    # Compute final radii for the best configuration
    _, best_radii = compute_sum_radii(best_centers)
    return best_centers, best_radii, best_sum
