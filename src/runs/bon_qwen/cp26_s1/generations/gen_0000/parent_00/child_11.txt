import numpy as np
import scipy.optimize as opt

def compute_objective_and_gradient(vars, lam, n):
    """
    Computes the objective function value and its gradient.
    Objective: Maximize sum(r) => Minimize -sum(r) + lam * penalty.
    Penalty: sum of squared overlaps.
    
    Variables structure:
    vars[0:n] -> radii (r)
    vars[n:2n] -> normalized x coords (u)
    vars[2n:3n] -> normalized y coords (v)
    
    Transformation:
    x = r + u * (1 - 2r)
    y = r + v * (1 - 2r)
    """
    r = vars[:n]
    u = vars[n:2*n]
    v = vars[2*n:3*n]
    
    # Coordinates
    x = r + u * (1 - 2 * r)
    y = r + v * (1 - 2 * r)
    
    obj = -np.sum(r)
    penalty = 0.0
    
    # Precompute centers for penalty
    centers = np.column_stack((x, y))
    
    # Compute pairwise distances and penalties
    # To avoid O(N^2) python loops if possible, but N=26 is small.
    # Using vectorized operations for speed.
    
    # Centers shape (n, 2)
    # Diffs shape (n, n, 2)
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    
    # Radii sums shape (n, n)
    r_sums = r[:, np.newaxis] + r[np.newaxis, :]
    
    # Overlap matrix (symmetric)
    # overlap = r_sum - dist
    overlaps = r_sums - dists
    
    # We only care about positive overlaps (violations)
    # And only unique pairs (i < j)
    # Upper triangle indices
    i_upper, j_upper = np.triu_indices(n, k=1)
    
    active_overlaps = overlaps[i_upper, j_upper]
    
    # Penalty term
    penalty = np.sum(np.maximum(active_overlaps, 0)**2)
    
    obj += lam * penalty
    
    # Gradient computation
    # d(obj)/d(r_k) = -1 + lam * d(penalty)/d(r_k)
    # d(obj)/d(u_k) = lam * d(penalty)/d(u_k)
    # d(obj)/d(v_k) = lam * d(penalty)/d(v_k)
    
    grad = np.zeros_like(vars)
    
    # Gradient contribution from -sum(r)
    grad[:n] -= 1.0
    
    # Gradient of penalty wrt centers and radii
    # Let P = sum_{i<j} max(0, r_i+r_j - d_{ij})^2
    # Let g_{ij} = 2 * max(0, r_i+r_j - d_{ij})
    # If overlap > 0, dP/dr_i += g_{ij} * (1 - d_{ij}/d_{ij} * dr_i/d... wait)
    
    # Derivatives of overlap w.r.t variables:
    # overlap_{ij} = r_i + r_j - sqrt((x_i-x_j)^2 + (y_i-y_j)^2)
    # d(overlap)/dr_i = 1
    # d(overlap)/dx_i = -(x_i - x_j) / d_{ij} = -n_ix
    # d(overlap)/dy_i = -(y_i - y_j) / d_{ij} = -n_iy
    
    # Variables: x_i = r_i + u_i(1-2r_i)
    # dx_i/dr_i = 1 - 2u_i
    # dx_i/du_i = 1 - 2r_i
    # Same for y and v.
    
    # Chain rule for gradient accumulation
    # grad_penalty_r[i] = sum_{j!=i} g_{ij} * (1 + dx_i/dr_i * (-n_ix) + dy_i/dr_i * (-n_iy))
    # grad_penalty_u[i] = sum_{j!=i} g_{ij} * (dx_i/du_i * (-n_ix) + dy_i/du_i * (-n_iy))
    
    # Compute unit vectors for active pairs
    # Only compute for pairs with overlap > 0
    mask = active_overlaps > 0
    i_active = i_upper[mask]
    j_active = j_upper[mask]
    
    # Number of active pairs
    m = len(i_active)
    if m > 0:
        # Get overlaps for active pairs
        olaps = active_overlaps[mask]
        g = 2.0 * olaps # gradient multiplier for square
        
        # Distances for active pairs
        d_ij = dists[i_active, j_active]
        # Avoid division by zero
        d_safe = np.where(d_ij < 1e-9, 1e-9, d_ij)
        
        # Normal vectors
        dx_ij = x[i_active] - x[j_active]
        dy_ij = y[i_active] - y[j_active]
        
        n_x = dx_ij / d_safe
        n_y = dy_ij / d_safe
        
        # Derivatives of coords wrt vars
        # dx/dr = 1 - 2u, dx/du = 1 - 2r
        # dy/dr = 1 - 2v, dy/dv = 1 - 2r
        
        # Accumulate gradients
        # For each active pair (i, j), update grad for i and j
        
        # Precompute terms for i
        dr_i_term = 1.0 - (1 - 2*u[i_active]) * n_x - (1 - 2*v[i_active]) * n_y
        du_i_term = -(1 - 2*r[i_active]) * n_x
        dv_i_term = -(1 - 2*r[i_active]) * n_y
        
        # For j
        dr_j_term = 1.0 + (1 - 2*u[j_active]) * n_x + (1 - 2*v[j_active]) * n_y # Note signs: d(x_j-x_i)/dx_j is opposite
        # Wait, derivative of overlap w.r.t x_j:
        # overlap = ... - sqrt((x_i-x_j)^2...)
        # d/dx_j (-dist) = - (- (x_i-x_j)/dist) = (x_i-x_j)/dist = -n_x
        # Actually, dist = sqrt((x_i-x_j)^2). d(dist)/dx_j = -(x_i-x_j)/dist = -n_x?
        # Let's recheck.
        # d_ij = ||c_i - c_j||
        # d(d_ij)/dx_i = (x_i-x_j)/d_ij = n_x
        # d(d_ij)/dx_j = -(x_i-x_j)/d_ij = -n_x
        # overlap = r_i + r_j - d_ij
        # d(overlap)/dx_i = -n_x
        # d(overlap)/dx_j = n_x
        
        # My previous n_x was (x_i - x_j)/d. So d(overlap)/dx_i = -n_x. Correct.
        # So for j, d(overlap)/dx_j = n_x.
        
        # Recalculate j terms with correct signs
        dr_j_term = 1.0 - (1 - 2*u[j_active]) * (-n_x) - (1 - 2*v[j_active]) * (-n_y)
        # = 1.0 + (1 - 2u) n_x + (1 - 2v) n_y
        du_j_term = (1 - 2*r[j_active]) * n_x
        dv_j_term = (1 - 2*r[j_active]) * n_y
        
        # Accumulate
        np.add.at(grad[:n], i_active, lam * g * dr_i_term)
        np.add.at(grad[n:2*n], i_active, lam * g * du_i_term)
        np.add.at(grad[2*n:3*n], i_active, lam * g * dv_i_term)
        
        np.add.at(grad[:n], j_active, lam * g * dr_j_term)
        np.add.at(grad[n:2*n], j_active, lam * g * du_j_term)
        np.add.at(grad[2*n:3*n], j_active, lam * g * dv_j_term)

    return obj, grad
