def compute_overlap_penalty(centers, radii, weight=100.0):

    Computes a smooth penalty for overlapping circles and boundary violations.
    Penalty is 0 if valid, positive if violated.
    """
    n = centers.shape[0]
    penalty = 0.0

    # Circle i must satisfy: r_i <= x_i <= 1-r_i  =>  x_i - r_i >= 0, 1 - x_i - r_i >= 0
    # Violation: max(0, -(x - r))

        r = radii[i]

            penalty += weight * (x + r - 1)**2

            penalty += weight * (y - r)**2

    # Overlap penalty
    # dist >= r_i + r_j  =>  dist - (r_i + r_j) >= 0
    # Violation: max(0, r_i + r_j - dist)
