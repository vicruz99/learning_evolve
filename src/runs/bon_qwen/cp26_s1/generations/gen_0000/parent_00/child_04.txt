import numpy as np

def compute_violations(pos, rad):
    """Compute the maximum violation of constraints."""
    n = len(rad)
    max_v = 0.0
    
    # Boundary violations
    d_b = np.minimum(np.minimum(pos[:,0], 1-pos[:,0]), np.minimum(pos[:,1], 1-pos[:,1]))
    v_b = np.maximum(rad - d_b, 0)
    max_v = max(max_v, np.max(v_b))
    
    # Pairwise overlap violations
    diff = pos[:, None, :] - pos[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    r_sum = rad[:, None] + rad[None, :]
    v_p = np.maximum(r_sum - dists, 0)
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    max_v = max(max_v, np.max(v_p[mask]))
    
    return max_v

def run_packing():
    np.random.seed(42)
    n = 26
    best_sum = 0.0
    best_pos = None
    best_rad = None
    
    num_restarts = 10
    
    for restart in range(num_restarts):
        # Initialize positions
        if restart == 0:
            # Hexagonal-ish grid initialization
            rows, cols = 5, 6
            pos = np.array([[i * 0.18 + (j % 2) * 0.09, j * 0.19] for j in range(rows) for i in range(cols)])[:n]
        else:
            pos = np.random.uniform(0.15, 0.85, (n, 2))
            
        rad = np.ones(n) * 0.02
        
        dt = 0.005
        k_rep = 1000.0
        k_bound = 1000.0
        base_growth = 0.0008
        n_steps = 40000
        
        for step in range(n_steps):
            # Decrease growth rate over time to stabilize
            growth = base_growth * max(0.0, 1.0 - step / n_steps)
            
            # Vectorized pairwise interactions
            diff = pos[:, None, :] - pos[None, :, :]
            dists = np.sqrt(np.sum(diff**2, axis=2))
            dists = np.where(dists < 1e-9, 1e-9, dists)
            
            r_sum = rad[:, None] + rad[None, :]
            overlap = np.maximum(r_sum - dists, 0)
            
            dir_vec = diff / dists[:, :, None]
            repulsion = overlap * k_rep * dir_vec
            
            # Sum forces on each circle
            forces = np.sum(repulsion, axis=1)
            
            # Boundary forces
            bx = np.where(pos[:,0] < rad, k_bound*(rad - pos[:,0]), 0.0) 
            bx -= np.where(pos[:,0] > 1-rad, k_bound*(pos[:,0] - (1-rad)), 0.0)
            by = np.where(pos[:,1] < rad, k_bound*(rad - pos[:,1]), 0.0) 
            by -= np.where(pos[:,1] > 1-rad, k_bound*(pos[:,1] - (1-rad)), 0.0)
            
            forces[:,0] += bx
            forces[:,1] += by
            
            # Update positions
            pos += dt * forces
            pos = np.clip(pos, 1e-5, 1-1e-5)
            
            # Grow radii
            rad += growth
            
        # Enforce validity by scaling down radii if necessary
        mv = compute_violations(pos, rad)
        if mv > 1e-9:
            scale = 1.0 - mv * 1.2
            rad *= scale
            
        curr_sum = np.sum(rad)
        if curr_sum > best_sum:
            best_sum = curr_sum
            best_pos = pos.copy()
            best_rad = rad.copy()
            
    # Final strict validity check
    mv = compute_violations(best_pos, best_rad)
    if mv > 1e-9:
        scale = 1.0 - mv * 1.1
        best_rad *= scale
        
    return best_pos, best_rad, float(np.sum(best_rad))
