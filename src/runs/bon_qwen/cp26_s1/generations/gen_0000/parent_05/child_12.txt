import numpy as np
from scipy.optimize import minimize


def compute_violation(centers, radii):
    """Compute total constraint violation (boundary + overlap)."""
    n = centers.shape[0]
    v = 0.0
    for i in range(n):
        r = radii[i]
        x, y = centers[i]
        v += max(0, r - x) ** 2
        v += max(0, r - (1 - x)) ** 2
        v += max(0, r - y) ** 2
        v += max(0, r - (1 - y)) ** 2
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            v += max(0, radii[i] + radii[j] - dist) ** 2
    return v


def penalty_objective(vars, n, penalty):
    """Objective: maximize sum of radii with penalty for constraint violations."""
    centers = vars[:2 * n].reshape(n, 2)
    radii = vars[2 * n:]
    radii = np.maximum(radii, 0)
    return -np.sum(radii) + penalty * compute_violation(centers, radii)


def force_based_relaxation(centers, radii, n, iterations=1500):

                dy = centers[j, 1] - centers[i, 1]

                    continue

                    fy = force * dy
