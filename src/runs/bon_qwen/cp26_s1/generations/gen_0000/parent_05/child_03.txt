def compute_penalized_objective(x, n, penalty_weight):
    centers = x[:2*n].reshape(n, 2)
    radii = x[2*n:]
    
    obj = -np.sum(radii)
    
    for i in range(n):
        r = radii[i]
        xi, yi = centers[i]
        
        # Boundary penalties
        if xi - r < 0:
            obj += penalty_weight * (xi - r)**2
        if xi + r > 1:
            obj += penalty_weight * (xi + r - 1)**2
        if yi - r < 0:
            obj += penalty_weight * (yi - r)**2
        if yi + r > 1:
            obj += penalty_weight * (yi + r - 1)**2
        
        # Radius non-negativity
        if r < 0:
            obj += penalty_weight * r**2
    
    # Overlap penalties
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt((centers[i,0] - centers[j,0])**2 + (centers[i,1] - centers[j,1])**2)
            gap = dist - radii[i] - radii[j]
            if gap < 0:
                obj += penalty_weight * gap**2
    
    return obj

def optimize_packing(centers, radii, n, max_iterations=5000):
    """Optimize packing using penalty method."""
    x0 = np.concatenate([centers.flatten(), radii])
    
    # Bounds: x in [0,1], y in [0,1], r in [0, 0.5]
    bounds = []
    for i in range(n):
        bounds.append((0, 1))  # x
        bounds.append((0, 1))  # y
    for i in range(n):
        bounds.append((1e-8, 0.5))  # r
    
    best_x = x0.copy()
    best_obj = compute_penalized_objective(x0, n, 1000)
    
    # Progressive penalty approach
    for stage in range(5):
        penalty = 100 * (10 ** stage)
        
        result = minimize(
            lambda x: compute_penalized_objective(x, n, penalty),
            best_x,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': max_iterations, 'ftol': 1e-12, 'gtol': 1e-10}
        )
        
        if result.fun < best_obj:
            best_obj = result.fun
            best_x = result.x.copy()
    
    return best_x
