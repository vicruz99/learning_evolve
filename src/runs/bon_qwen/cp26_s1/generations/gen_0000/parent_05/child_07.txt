import numpy as np
from scipy.optimize import minimize
import math
import random


def compute_penalty(params, n):
    """Compute smooth penalty for constraint violations."""
    penalty = 0.0

    for i in range(n):
        x = params[3 * i]
        y = params[3 * i + 1]
        r = params[3 * i + 2]

        # Boundary constraints - smooth penalties
        if x - r < 0:
            penalty += (x - r) ** 2
        if x + r > 1:
            penalty += (x + r - 1) ** 2
        if y - r < 0:
            penalty += (y - r) ** 2
        if y + r > 1:
            penalty += (y + r - 1) ** 2
        if r < 0:
            penalty += r ** 2

    # Overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            x1 = params[3 * i]
            y1 = params[3 * i + 1]
            r1 = params[3 * i + 2]
            x2 = params[3 * j]
            y2 = params[3 * j + 1]
            r2 = params[3 * j + 2]

            dx = x1 - x2
            dy = y1 - y2
            dist_sq = dx * dx + dy * dy
            dist = math.sqrt(dist_sq) if dist_sq > 1e-20 else 1e-10

            min_dist = r1 + r2
            if dist < min_dist:
                penalty += (min_dist - dist) ** 2

    return penalty


def compute_negative_sum(params, n):
    """Compute negative sum of radii."""
    return -np.sum(params[2::3])


def make_penalized_objective(n, weight):
    """Create a penalized objective function."""
    def objective(params):
        obj = compute_negative_sum(params, n)
        pen = compute_penalty(params, n)
        return obj + weight * pen
    return objective


def create_hex_grid(n, r_spacing, offset_x=0, offset_y=0):
    """Create hexagonal grid of centers."""
    centers = []
    row = 0
    while len(centers) < n + 20:
        col = 0
        while len(centers) < n + 20:
            x = col * math.sqrt(3) * r_spacing + r_spacing + offset_x
            y = row * 1.5 * r_spacing + r_spacing + offset_y

            if row % 2 == 1:
                x += math.sqrt(3) * r_spacing / 2

            if x - r_spacing >= -0.01 and x + r_spacing <= 1.01 and y - r_spacing >= -0.01 and y + r_spacing <= 1.01:
                centers.append([x, y])

            col += 1

        row += 1

    return np.array(centers[:n])


def create_random_config(n, r_base, seed):
    """Create a random initial configuration."""
    rng = np.random.RandomState(seed)
    centers = rng.uniform(r_base, 1 - r_base, (n, 2))
    return centers


def create_corner_config(n, r_base, seed):
    """Create config biased toward corners and edges."""
    rng = np.random.RandomState(seed)
    centers = []

    # Place some circles in corners
    corner_candidates = [
        [r_base, r_base],
        [1 - r_base, r_base],
        [r_base, 1 - r_base],
        [1 - r_base, 1 - r_base]
    ]

    for c in corner_candidates:
        if len(centers) < n:
            centers.append(c)

    # Place some along edges
    edge_count = n - len(centers)
    for _ in range(edge_count):
        side = rng.randint(0, 4)
        if side == 0:  # bottom
            centers.append([rng.uniform(r_base, 1 - r_base), r_base])
        elif side == 1:  # top
            centers.append([rng.uniform(r_base, 1 - r_base), 1 - r_base])
        elif side == 2:  # left
            centers.append([r_base, rng.uniform(r_base, 1 - r_base)])
        else:  # right
            centers.append([1 - r_base, rng.uniform(r_base, 1 - r_base)])

    return np.array(centers[:n])


def build_params(centers, radii):
    """Build parameter array from centers and radii."""
    n = len(radii)
    params = np.zeros(3 * n)
    for i in range(n):
        params[3 * i] = centers[i, 0]
        params[3 * i + 1] = centers[i, 1]
        params[3 * i + 2] = radii[i]
    return params


def extract_results(params, n):
    """Extract centers and radii from params."""
    centers = params[:2 * n].reshape(n, 2)
    radii = params[2 * n:]
    return centers, radii


def optimize_from_config(params, n, max_iter=5000):
    """Optimize from a given initial configuration."""
    current_params = params.copy()

    # Progressive penalty optimization
    for weight in [500, 2000, 8000, 20000]:
        obj_func = make_penalized_objective(n, weight)
        bounds = [(0.01, 0.99) for _ in range(2 * n)] + [(0.001, 0.49) for _ in range(n)]

        result = minimize(
            obj_func,
            current_params,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-8}
        )
        current_params = result.x

    return current_params


def local_refine(centers, radii, n):
    """Local refinement: try to increase individual radii and adjust positions."""
    radii = radii.copy()
    centers = centers.copy()

    for iteration in range(500):
        improved = False

        for i in range(n):
            # Try to increase radius
            best_r = radii[i]
            best_x, best_y = centers[i, 0], centers[i, 1]

            # Try increasing radius
            for delta_r in [0.001, 0.002, 0.005, 0.01]:
                test_r = radii[i] + delta_r
                if test_r > 0.49:
                    continue

                # Check if current position works with larger radius
                valid = True

                # Boundary check
                if centers[i, 0] - test_r < 0 or centers[i, 0] + test_r > 1:
                    valid = False
                if centers[i, 1] - test_r < 0 or centers[i, 1] + test_r > 1:
                    valid = False

                if valid:
                    # Check overlaps
                    for j in range(n):
                        if i == j:
                            continue
                        dx = centers[i, 0] - centers[j, 0]
                        dy = centers[i, 1] - centers[j, 1]
                        dist = math.sqrt(dx * dx + dy * dy)
                        if dist < test_r + radii[j] - 1e-12:
                            valid = False
                            break

                if valid:
                    best_r = test_r

            if best_r > radii[i] + 1e-10:
                radii[i] = best_r
                improved = True

            # Try small position adjustments to allow larger radius
            for _ in range(20):
                for dx in [-0.005, -0.002, -0.001, 0.001, 0.002, 0.005]:
                    for dy in [-0.005, -0.002, -0.001, 0.001, 0.002, 0.005]:
                        new_x = centers[i, 0] + dx
                        new_y = centers[i, 1] + dy

                        if new_x - radii[i] < 0 or new_x + radii[i] > 1:
                            continue
                        if new_y - radii[i] < 0 or new_y + radii[i] > 1:
                            continue

                        valid = True
                        for j in range(n):
                            if i == j:
                                continue
                            ddx = new_x - centers[j, 0]
                            ddy = new_y - centers[j, 1]
                            dist = math.sqrt(ddx * ddx + ddy * ddy)
                            if dist < radii[i] + radii[j] - 1e-12:
                                valid = False
                                break

                        if valid:
                            centers[i, 0] = new_x
                            centers[i, 1] = new_y
                            break
                    else:
                        continue
                    break

    return centers, radii
