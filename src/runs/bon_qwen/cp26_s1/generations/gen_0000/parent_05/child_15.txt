

To maximize the sum of radii for 26 circles in a unit square, we can use a numerical optimization approach that balances the trade-off between expanding the circles and resolving overlaps. Given the target sum of 2.636, a uniform hexagonal-like arrangement is likely optimal. 

We will use `scipy.optimize.minimize` with the SLSQP method. This solver handles the non-linear constraints required for non-overlapping circles and boundary conditions. To ensure a high-quality result, we initialize the centers in a perturbed hexagonal lattice, which provides a much better starting point than a square grid for 26 circles.

### Optimization Strategy
1.  **Initial Configuration**: We create a hexagonal packing of 26 circles. This is denser than a square grid. We calculate an initial radius that fits this configuration within the unit square and apply small random perturbations to prevent the optimizer from getting stuck in a local minimum.
2.  **Variables**: We optimize the $(x, y)$ coordinates and the radius $r$ for each of the 26 circles, totaling 78 variables.
3.  **Constraints**:
    *   **Non-overlap**: The distance between any two centers must be at least the sum of their radii.
    *   **Boundary**: Each circle must be fully contained within $[0, 1] \times [0, 1]$.
4.  **Optimization**: We minimize the negative sum of radii. SLSQP is used as it is efficient for constrained non-linear problems.

import numpy as np
from scipy.optimize import minimize

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square

    Args:
        centers: np.array of shape (n, 2) with (x, y) coordinates
        radii: np.array of shape (n) with radius of each circle

    Returns:
        True if valid, False otherwise
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    np.random.seed(42)

    # Initial hexagonal packing
    rows = 6
    circles_per_row = [5, 4, 5, 4, 5, 3]
    centers = []
    for i, num in enumerate(circles_per_row):
        y = 0.1 + i * 0.175
        x_start = 0.1 + (0.5 if i % 2 == 1 else 0)
        for j in range(num):
            x = x_start + j * 0.2
            centers.append([x, y])
    centers = np.array(centers)
    radii = np.full(n, 0.09)

    # Perturb
    centers += np.random.uniform(-0.02, 0.02, centers.shape)
    radii += np.random.uniform(-0.01, 0.01, n)
    radii = np.maximum(radii, 0.01)

    # Bounds
    bounds = [(0.01, 0.5)] * (2 * n) + [(0.01, 0.5)] * n

    def objective(x):
        return -np.sum(x[2*n:])

    def constraint_overlap(x):
        dists = np.linalg.norm(centers[None, :, :] - centers[:, None, :], axis=2)
        sums_r = x[2*n:]
        sums_r_matrix = sums_r[None, :] + sums_r[:, None]
        return dists - sums_r_matrix

    def constraint_boundary(x):
        c = x[:2*n].reshape(-1, 2)
        r = x[2*n:]
        return np.concatenate([c[:, 0] - r, 1 - c[:, 0] - r, c[:, 1] - r, 1 - c[:, 1] - r])

    cons = [
        {'type': 'ineq', 'fun': lambda x: constraint_overlap(x).flatten()},
        {'type': 'ineq', 'fun': constraint_boundary}
    ]

    res = minimize(objective, np.concatenate([centers.flatten(), radii]), method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 2000, 'ftol': 1e-12})

    final_centers = res.x[:2*n].reshape(-1, 2)
    final_radii = res.x[2*n:]
    sum_radii = np.sum(final_radii)

    return final_centers, final_radii, sum_radii
