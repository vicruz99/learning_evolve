import numpy as np

from scipy.optimize import minimize

def run_packing():
    n = 26
    
    # --- 1. Initialization ---
    # Initialize centers in a hexagonal grid pattern.
    # This provides a dense packing starting point.
    
    centers = np.zeros((n, 2))
    r_init = 0.1
    row_h = r_init * math.sqrt(3)
    col_w = 2 * r_init
    
    idx = 0
    row = 0
    while idx < n:
        # y coordinate for the row
        y = r_init + row * row_h
        
        # x offset for odd rows to create hexagonal shift
        # Even rows (0, 2, ...): start at r_init (0.1)
        # Odd rows (1, 3, ...): start at r_init + shift (0.2)
        # The shift is r_init (0.1) which is half of col_w (0.2)
        shift = r_init if row % 2 == 1 else 0.0
        start_x = r_init + shift
        
        x = start_x
        while x + r_init <= 1.0 + 1e-9 and idx < n:
            centers[idx, 0] = x
            centers[idx, 1] = y
            idx += 1
            x += col_w
        
        row += 1
        # Safety break to prevent infinite loop if logic fails
        if row > 20: 
            break

    # Fill any remaining slots if needed (unlikely)
    while idx < n:
        centers[idx, 0] = np.random.uniform(0, 1)
        centers[idx, 1] = np.random.uniform(0, 1)
        idx += 1
        
    # Initial radii: start small to ensure feasibility
    radii = np.full(n, 0.05)
    
    # Flatten variables: [x1, y1, r1, x2, y2, r2, ...]
    x0 = np.concatenate([centers.ravel(), radii])
    
    # --- 2. Optimization ---
    # We maximize sum(radii) subject to non-overlap and boundary constraints.
    # We use a penalty method with L-BFGS-B.
    # We run multiple passes with increasing penalty to ensure strict feasibility.
    
    current_x = x0
    
    for iteration in range(3):
        # Penalty weight increases with iteration
        p_weight = 100.0 * (10 ** iteration)
        
        def objective(vars):
            c = vars[:2*n].reshape(n, 2)
            r = vars[2*n:]
            
            # Objective: minimize -sum(radii)
            obj = -np.sum(r)
            
            # Boundary penalties
            # Constraints: 
            # r - x <= 0  (left)
            # x + r - 1 <= 0 (right)
            # r - y <= 0 (bottom)
            # y + r - 1 <= 0 (top)
            
            # Penalty is sum of squares of positive violations
            b1 = np.maximum(0, r - c[:, 0])
            b2 = np.maximum(0, c[:, 0] + r - 1.0)
            b3 = np.maximum(0, r - c[:, 1])
            b4 = np.maximum(0, c[:, 1] + r - 1.0)

            overlaps = np.maximum(0, r_sums[upper_idx] - dists[upper_idx])
            overlap_pen = np.sum(overlaps**2)
            
            obj += p_weight * (boundary_pen + overlap_pen)
            return obj
