def hexagonal_init(n):
    """Initialize with hexagonal packing arrangement."""
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    r_init = 0.08
    h = np.sqrt(3) * r_init
    w = 2 * r_init

    row_counts = [6, 5, 6, 5, 4]
    idx = 0
    for row in range(5):
        y = r_init + row * h
        offset = r_init if row % 2 == 1 else 0
        for col in range(row_counts[row]):
            x = r_init + offset + col * w
            centers[idx] = [x, y]
            radii[idx] = r_init
            idx += 1

    return centers, radii


def hexagonal_init_scaled(n, scale):
    """Initialize with scaled hexagonal packing."""
    r_init = 0.08 * scale
    h = np.sqrt(3) * r_init
    w = 2 * r_init

    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    row_counts = [6, 5, 6, 5, 4]
    idx = 0
    for row in range(5):
        y = r_init + row * h
        offset = r_init if row % 2 == 1 else 0
        for col in range(row_counts[row]):
            x = r_init + offset + col * w
            x = np.clip(x, r_init, 1 - r_init)
            y = np.clip(y, r_init, 1 - r_init)
            centers[idx] = [x, y]
            radii[idx] = r_init
            idx += 1

    return centers, radii


def structured_random_init(n):
    """Initialize with structured random positions."""
    centers = np.zeros((n, 2))
    np.random.seed(42)

    # Create a grid-like structure with jitter
    row_counts = [6, 5, 6, 5, 4]
    idx = 0
    for row in range(5):
        y = 0.12 + row * 0.18
        offset = 0.09 if row % 2 == 1 else 0
        for col in range(row_counts[row]):
            x = 0.12 + offset + col * 0.18
            x += np.random.normal(0, 0.02)
            y += np.random.normal(0, 0.02)
            x = np.clip(x, 0.05, 0.95)
            y = np.clip(y, 0.05, 0.95)
            centers[idx] = [x, y]
            idx += 1

    return centers


def optimize_radii_lp(centers, n):
    """Maximize sum of radii given fixed positions using linear programming."""
    # Objective: maximize sum(r_i) => minimize -sum(r_i)
    c = -np.ones(n)

    A_ub = []
    b_ub = []

    # Non-overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dist)

    # Boundary constraints: r_i <= min(x_i, 1-x_i, y_i, 1-y_i)
    for i in range(n):
        x, y = centers[i]
        max_r = min(x, 1 - x, y, 1 - y)
        row = np.zeros(n)
        row[i] = 1.0
        A_ub.append(row)
        b_ub.append(max_r)

    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, None) for _ in range(n)]

    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')

    if result.success:
        return result.x
    else:
        # Fallback: return zero radii
        return np.zeros(n)


def optimize_positions(centers, radii, n, max_iter=500, lr=0.005):
    """Optimize positions given fixed radii using repulsive forces."""
    centers = centers.copy()

    for iteration in range(max_iter):
        forces = np.zeros((n, 2))
        total_overlap = 0.0

        # Repulsive forces for overlapping or close circles
        for i in range(n):
            for j in range(i + 1, n):
                diff = centers[i] - centers[j]
                dist = np.sqrt(np.sum(diff ** 2))
                min_dist = radii[i] + radii[j]
                if dist < min_dist and dist > 1e-10:
                    overlap = min_dist - dist
                    total_overlap += overlap
                    force_mag = overlap * 200.0
                    force = force_mag * diff / dist
                    forces[i] += force
                    forces[j] -= force
                elif dist < min_dist * 1.5 and dist > 1e-10:
                    # Gentle repulsion even for non-overlapping close circles
                    force_mag = 0.1 * diff / dist
                    forces[i] += force_mag
                    forces[j] -= force_mag

        # Boundary forces
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            margin = 0.0
            if x - r < margin:
                forces[i, 0] += (margin - (x - r)) * 500.0
            if x + r > 1 - margin:
                forces[i, 0] -= ((x + r) - (1 - margin)) * 500.0
            if y - r < margin:
                forces[i, 1] += (margin - (y - r)) * 500.0
            if y + r > 1 - margin:
                forces[i, 1] -= ((y + r) - (1 - margin)) * 500.0

        # Apply forces with adaptive learning rate
        current_lr = lr * (1.0 + 0.5 * np.exp(-iteration / 100))
        centers += current_lr * forces

        # Clip to feasible region
        for i in range(n):
            centers[i, 0] = np.clip(centers[i, 0], radii[i] + 1e-10, 1 - radii[i] - 1e-10)
            centers[i, 1] = np.clip(centers[i, 1], radii[i] + 1e-10, 1 - radii[i] - 1e-10)

        if total_overlap < 1e-12:
            break

    return centers


def optimize_alternating(centers, radii, n, n_iter=300):
    """Alternate between radius optimization and position optimization."""
    for iteration in range(n_iter):
        # Optimize radii given positions
        new_radii = optimize_radii_lp(centers, n)

        # Check if radii changed significantly
        if np.max(np.abs(new_radii - radii)) < 1e-10:
            break

        radii = new_radii

        # Optimize positions given radii
        centers = optimize_positions(centers, radii, n, max_iter=200, lr=0.003)

    # Final radius optimization
    radii = optimize_radii_lp(centers, n)

    return centers, radii


def is_valid(centers, radii, n):
    """Check if the packing is valid."""
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if r < -1e-12:
            return False
        if x - r < -1e-10 or x + r > 1 + 1e-10:
            return False
        if y - r < -1e-10 or y + r > 1 + 1e-10:
            return False

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-10:
                return False

    return True
