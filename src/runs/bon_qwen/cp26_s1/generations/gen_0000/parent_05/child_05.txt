import numpy as np
from scipy.optimize import minimize

def calculate_distance_matrix(centers):
    """Calculates the Euclidean distance matrix between all pairs of centers."""
    centers = np.asarray(centers)
    n = centers.shape[0]
    dists = np.zeros((n, n))
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            dists[i, j] = dist
            dists[j, i] = dist
    return dists

def get_constraints(centers, radii):
    """
    Returns the list of constraint dictionaries for the optimizer.
    Constraints are formatted as c(x) >= 0.
    """
    centers = np.asarray(centers)
    radii = np.asarray(radii)
    n = len(radii)
    
    cons = []
    
    # Boundary constraints: r <= x <= 1-r and r <= y <= 1-r
    for i in range(n):
        cons.append({'type': 'ineq', 'fun': lambda v, i=i: v[2*n + i] - v[2*i]})
        cons.append({'type': 'ineq', 'fun': lambda v, i=i: v[2*i] + v[2*n + i] - 1.0})
        cons.append({'type': 'ineq', 'fun': lambda v, i=i: v[2*n + i] - v[2*i + 1]})
        cons.append({'type': 'ineq', 'fun': lambda v, i=i: v[2*i + 1] + v[2*n + i] - 1.0})
        cons.append({'type': 'ineq', 'fun': lambda v, i=i: v[2*n + i]}) # r >= 0

    # Pairwise non-overlap constraints: dist >= r_i + r_j
    dist_matrix = calculate_distance_matrix(centers)
    for i in range(n):
        for j in range(i + 1, n):
            d_ij = dist_matrix[i, j]
            # If initial distance is very small, we might need a different approach, 
            # but for valid init it should be fine.
            cons.append({
                'type': 'ineq',
                'fun': lambda v, i=i, j=j, d=d_ij: d - (v[2*n + i] + v[2*n + j])
            })
            
    return cons

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes the packing of 26 circles in a unit square to maximize sum of radii.
    """
    n = 26
    np.random.seed(42)

    # 1. Initialization: Hexagonal Lattice
    # We aim for a radius around 0.101 (target sum 2.636 / 26)
    r_init = 0.10
    centers = []
    
    # Try to fit 26 circles in a hexagonal pattern. 
    # A 5x5 grid with hexagonal spacing often works well.
    # 5 rows, with alternating counts of 5 and 5 (shifted) might be too wide.
    # Let's create a dense grid and prune/select.
    
    rows = 6
    cols = 5
    # Hexagonal spacing
    dx = 2 * r_init
    dy = np.sqrt(3) * r_init
    
    y_pos = r_init
    while len(centers) < n:
        x_pos = r_init
        # Shift every other row
        if (len(centers) // cols) % 2 == 1:
            x_pos += dx / 2
            
        while x_pos + r_init <= 1.0 and len(centers) < n:
            centers.append([x_pos, y_pos])
            x_pos += dx
        
        y_pos += dy
        if y_pos + r_init > 1.0:
            break
            
    # If we didn't get 26, or if we exceeded, we adjust. 
    # With r=0.1, 5 cols * 2r = 1.0 width. 6 rows height approx 0.9.
    # This should yield 30 points. We just take the first 26.
    if len(centers) < n:
        # Fallback to random if grid failed
        centers = np.random.uniform(0.2, 0.8, (n, 2))
    
    centers = np.array(centers[:n])
    
    # Randomize slightly to break symmetry and help optimization
    centers += np.random.normal(0, 0.01, centers.shape)
    # Clip to bounds
    centers = np.clip(centers, 0.05, 0.95)
    
    radii = np.ones(n) * r_init
    
    # Combine variables: [x1, y1, ..., x26, y26, r1, ..., r26]
    x0 = np.concatenate([centers.flatten(), radii])
    
    # 2. Define Objective: Maximize sum of radii -> Minimize -sum(radii)
    def objective(x):
        radii_vec = x[2*n:]
        return -np.sum(radii_vec)
    
    # 3. Define Constraints
    # We define them dynamically inside the optimization loop or pass them.
    # However, SLSQP needs a list of constraint dicts.
    # The distance depends on x (centers), so we must update the constraint functions 
    # or make them dynamic. SLSQP accepts functions.
    
    # To handle the distance constraint which depends on x, we define a function 
    # that extracts current centers and checks all pairs.
    
    def all_constraints(x):
        c = []
        curr_centers = x[:2*n].reshape(n, 2)
        curr_radii = x[2*n:]
        
        # Boundary
        for i in range(n):
            c.append(curr_radii[i] - curr_centers[i, 0]) # x >= r
            c.append(curr_centers[i, 0] + curr_radii[i] - 1.0) # x <= 1-r -> 1 - x >= r -> x + r <= 1
            c.append(curr_radii[i] - curr_centers[i, 1]) # y >= r
            c.append(curr_centers[i, 1] + curr_radii[i] - 1.0) # y <= 1-r
            c.append(curr_radii[i]) # r >= 0
            
        # Overlap
        dist_matrix = calculate_distance_matrix(curr_centers)
        for i in range(n):
            for j in range(i + 1, n):
                d = dist_matrix[i, j]
                c.append(d - (curr_radii[i] + curr_radii[j]))
                
        return np.array(c)

    # SLSQP needs constraints in a specific format: {'type': 'ineq', 'fun': ...}
    # Since 'fun' returns a vector, we can return the whole vector.
    cons = {'type': 'ineq', 'fun': all_constraints}
    
    # 4. Optimization
    # Use bounds to help
    # x, y in [0, 1], r in [0, 0.5]
    bounds = [(0, 1)] * (2 * n) + [(0, 0.5)] * n
    
    # Run optimization
    # tol=1e-8 for high precision
    result = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, 
                      options={'maxiter': 5000, 'ftol': 1e-12, 'disp': False})
    
    if result.success:
        final_centers = result.x[:2*n].reshape(n, 2)
        final_radii = result.x[2*n:]
    else:
        # Fallback if optimization fails
        final_centers = centers
        final_radii = radii
        
    # Ensure non-negative radii and valid bounds for final output
    final_radii = np.maximum(final_radii, 0)
    final_centers = np.clip(final_centers, final_radii.reshape(-1, 1), 1 - final_radii.reshape(-1, 1))
    
    sum_radii = np.sum(final_radii)
    
    return final_centers, final_radii, sum_radii
