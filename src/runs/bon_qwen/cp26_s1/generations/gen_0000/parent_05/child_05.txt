import numpy as np
from scipy.optimize import minimize

def compute_objective(v):
    """Objective function: negative sum of radii (for minimization)"""
    return -np.sum(v[52:])

def compute_constraints(v):
    """Computes all inequality constraints: boundaries and non-overlaps"""
    n = 26
    centers = v[:52].reshape(n, 2)
    radii = v[52:]
    
    # 1. Boundary constraints: 4*n
    # x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
    c = np.zeros(4 * n)
    c[:n] = centers[:, 0] - radii
    c[n:2*n] = 1.0 - centers[:, 0] - radii
    c[2*n:3*n] = centers[:, 1] - radii
    c[3*n:4*n] = 1.0 - centers[:, 1] - radii
    
    # 2. Overlap constraints: n*(n-1)/2
    # (xi-xj)^2 + (yi-yj)^2 - (ri+rj)^2 >= 0
    cx = centers[:, 0]
    cy = centers[:, 1]
    
    # Vectorized pairwise distances and radius sums
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dist_sq = dx**2 + dy**2
    sum_r = radii[:, None] + radii[None, :]
    
    # Extract strict upper triangle to avoid duplicates and self-pairs
    mask = np.triu(np.ones((n, n)), k=1).astype(bool)
    overlap_constraints = dist_sq[mask] - sum_r[mask]**2
    
    return np.concatenate([c, overlap_constraints])

def generate_initial_guess(seed):
    """Generates a structured hexagonal lattice initialization with random perturbation"""
    rng = np.random.default_rng(seed)
    n = 26
    r0 = 0.095
    centers = np.zeros((n, 2))
    idx = 0
    
    # Build hexagonal rows
    for row in range(8):
        y = r0 + row * np.sqrt(3) * r0
        cols = 5 if row % 2 == 0 else 4
        if idx + cols > n:
            cols = n - idx
        shift = (row % 2) * r0
        for k in range(cols):
            centers[idx] = [r0 + k * 2 * r0 + shift, y]
            idx += 1
            
    # Normalize positions to fit comfortably within [0.02, 0.98]
    for d in range(2):
        min_val, max_val = centers[:, d].min(), centers[:, d].max()
        span = max_val - min_val
        if span > 1e-9:
            centers[:, d] = 0.02 + (centers[:, d] - min_val) * 0.96 / span
            
    # Add small random perturbation to break symmetry and avoid flat basins
    centers += rng.uniform(-0.015, 0.015, centers.shape)
    centers = np.clip(centers, 0.01, 0.99)
    
    initial_radii = np.full(n, 0.085)
    return np.concatenate([centers.flatten(), initial_radii])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Problem setup
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
    constraints = {'type': 'ineq', 'fun': compute_constraints}
    
    # Multi-restart optimization to ensure global optimum is found
    seeds = [42, 123, 456, 789, 1024]
    for seed in seeds:
        try:
            x0 = generate_initial_guess(seed)
            res = minimize(
                compute_objective,
                x0,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 1000, 'ftol': 1e-12, 'disp': False}
            )
            
            current_sum = -res.fun
            if current_sum > best_sum:
                best_sum = current_sum
                best_centers = res.x[:2*n].reshape(n, 2)
                best_radii = res.x[2*n:]
        except Exception:
            continue
            
    # Safety clamping and final extraction
    if best_centers is not None:
        best_radii = np.maximum(best_radii, 0.0)
        return best_centers, best_radii, float(np.sum(best_radii))
        
    # Fallback (should not be reached given robust initialization)
    return np.zeros((n, 2)), np.zeros(n), 0.0
