from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    Pack 26 circles in a unit square [0,1]x[0,1] to maximize the sum of radii.
    
    Returns:
        centers: np.array of shape (26, 2) with (x, y) coordinates
        radii: np.array of shape (26,) with radius of each circle
        sum_radii: float sum of radii

    best_sum = -1.0
    best_centers = None
    best_radii = None

        # Let's use structure: centers (n,2), radii (n,) -> flattened

        y = variables[n:2*n]
        r = variables[2*n:]
        
        return -np.sum(r)

        # Boundary constraints: 0 <= x - r, x + r <= 1, etc.
        # We will handle bounds in the optimizer, but let's add explicit constraints if needed.
        # Actually, bounds are easier.
        pass
        
        # Non-overlap constraints: (xi - xj)^2 + (yi - yj)^2 >= (ri + rj)^2
        for i in range(n):
            for j in range(i + 1, n):
                def dist_constraint(idx1=i, idx2=j):
                    x1, y1, r1 = variables[2*idx1], variables[n+2*idx1], variables[2*n+idx1]
                    x2, y2, r2 = variables[2*idx2], variables[n+2*idx2], variables[2*n+idx2]
                    # Constraint: d^2 - (r1+r2)^2 >= 0
                    return (x1 - x2)**2 + (y1 - y2)**2 - (r1 + r2)**2
                cons.append({'type': 'ineq', 'fun': dist_constraint})
        return cons

    # Bounds: x, y in [0, 1], r in [0, 0.5] (theoretically max radius is 0.5)
    # Tighter bound for r could be 0.2 for most, but 0.5 is safe.
    bounds = [(0, 1)] * n + [(0, 1)] * n + [(0, 0.5)] * n

    # Optimization function
    def optimize_packing(initial_vars, n):
        try:
            cons = []
            for i in range(n):
                for j in range(i + 1, n):
                    def dist_constraint(idx1=i, idx2=j, vars=initial_vars): # Closure capture issue?
                        # We need to access current variables, not initial
                        # scipy passes current variables to 'fun'
                        return lambda variables: (variables[2*idx1] - variables[2*idx2])**2 + \
                                                 (variables[n+2*idx1] - variables[n+2*idx2])**2 - \
                                                 (variables[2*n+idx1] + variables[2*n+idx2])**2

                        return constraint_func
                    cons.append({'type': 'ineq', 'fun': make_constraint(i, j)})

        best_radii = radii_grid

    return best_centers, best_radii, best_sum
