import numpy as np
import scipy.optimize as opt

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Solves the circle packing problem for 26 circles in a unit square.
    Returns centers, radii, and sum of radii.
    """
    n = 26
    best_sum = -1.0
    best_centers = None
    best_radii = None

    # Seed for reproducibility of randomness
    np.random.seed(42)

    def create_seeds():
        seeds = []
        
        # 1. Random distribution
        centers = np.random.rand(n, 2)
        centers = 0.1 + 0.8 * centers
        seeds.append(centers)

        # 2. Grid (6x5 subset)
        grid = np.linspace(0.1, 0.9, 6)
        cx = []
        cy = []
        for x in grid:
            for y in grid:
                cx.append(x)
                cy.append(y)
                if len(cx) >= n:
                    break
            if len(cx) >= n:
                break
        centers = np.array([cx[:n], cy[:n]]).T
        seeds.append(centers)

        # 3. Hexagonal-ish
        cx, cy = [], []
        y_curr = 0.1
        row = 0
        while len(cx) < n:
            if row % 2 == 0:
                xs = np.arange(0.1, 0.901, 0.2)
            else:
                xs = np.arange(0.2, 0.901, 0.2)
            for x in xs:
                if len(cx) < n:
                    cx.append(x)
                    cy.append(y_curr)
                else:
                    break
            y_curr += 0.173 # approx sqrt(3)/2 * 0.2
            row += 1
        centers = np.array([cx[:n], cy[:n]]).T
        seeds.append(centers)

        # 4-10. Perturbed versions of above
        for i in range(7):
            base = seeds[i % 3]
            pert = np.random.rand(n, 2) * 0.05
            c = base + pert - 0.025
            c = np.clip(c, 0.05, 0.95)
            seeds.append(c)
            
        return seeds

    seeds = create_seeds()

    def get_max_radii(centers):
        radii = np.zeros(n)
        for i in range(n):
            # Boundary constraints
            r_max = min(centers[i, 0], 1.0 - centers[i, 0], 
                        centers[i, 1], 1.0 - centers[i, 1])
            
            # Overlap constraints with other circles
            # r_i + r_j <= dist
            # r_i <= dist - r_j
            # We compute the tightest bound assuming other circles are at their max possible size?
            # A conservative estimate: r_i <= 0.5 * dist to any other center
            for j in range(n):
                if i == j: continue
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                # To avoid division by zero or negative, clamp dist
                if dist > 1e-6:
                    r_max = min(r_max, dist / 2.0)
            radii[i] = r_max
        return radii

    def objective(x_vars):
        # x_vars contains centers (n, 2) flattened
        centers = x_vars.reshape((n, 2))
        radii = get_max_radii(centers)
        # Return negative sum for minimization
        return -np.sum(radii)

    for seed_centers in seeds:
        # Optimization loop
        current_centers = seed_centers.copy()
        
        # Phase 1: Coordinate Ascent to grow radii
        for _ in range(15):
            # 1. Calculate radii based on current centers
            current_radii = get_max_radii(current_centers)
            
            # 2. Optimize centers to maximize sum of radii
            # We treat radii as dependent variables in the objective
            result = opt.minimize(
                objective,
                current_centers.flatten(),
                method='L-BFGS-B',
                bounds=[(0.01, 0.99)] * (2 * n),
                options={'maxiter': 200, 'ftol': 1e-12}
            )
            
            new_centers = result.x.reshape((n, 2))
            
            # Update current centers if improved
            new_radii = get_max_radii(new_centers)
            if np.sum(new_radii) > np.sum(current_radii):
                current_centers = new_centers
            else:
                # If optimization didn't find better, keep current
                pass
        
        # Final calculation for this run
        final_radii = get_max_radii(current_centers)
        current_sum = np.sum(final_radii)
        
        # Validation check
        is_valid = True
        for i in range(n):
            x, y = current_centers[i]
            r = final_radii[i]
            if r < 0 or x - r < -1e-9 or x + r > 1 + 1e-9 or y - r < -1e-9 or y + r > 1 + 1e-9:
                is_valid = False
                break
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((current_centers[i] - current_centers[j])**2))
                if dist < final_radii[i] + final_radii[j] - 1e-9:
                    is_valid = False
                    break
            if not is_valid: break
        
        if is_valid and current_sum > best_sum:
            best_sum = current_sum
            best_centers = current_centers
            best_radii = final_radii

    # Fallback if optimization failed completely (should not happen with valid seeds)
    if best_centers is None:
        centers = np.random.rand(n, 2) * 0.8 + 0.1
        radii = np.full(n, 0.04)
        best_sum = np.sum(radii)
        best_centers = centers
        best_radii = radii

    return best_centers, best_radii, best_sum
