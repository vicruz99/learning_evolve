def compute_forces(centers, radii, n):
    """
    Computes repulsive forces between circles and boundaries.
    Returns forces array of shape (n, 2).
    """
    forces = np.zeros((n, 2))
    k_overlap = 10.0
    k_wall = 100.0
    
    # Vectorized distance calculation
    # centers shape (n, 2)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :] # (n, n, 2)
    dists = np.sqrt(np.sum(diff**2, axis=2)) # (n, n)
    
    # Avoid division by zero for self
    np.fill_diagonal(dists, np.inf)
    
    # Radii sums
    r_sums = radii[:, np.newaxis] + radii[np.newaxis, :]
    
    # Overlap condition: dist < r1 + r2
    overlap = r_sums - dists
    
    # Only consider positive overlaps
    valid_overlap = np.maximum(0, overlap)
    
    # Force magnitude
    force_mags = k_overlap * valid_overlap
    
    # Force direction (unit vector)
    # Handle dists close to 0 to avoid NaN
    safe_dists = np.where(dists < 1e-9, 1e-9, dists)
    directions = diff / safe_dists[:, :, np.newaxis]
    
    # Apply forces: F_ij acts on i, -F_ij on j
    # Sum over j for each i
    # forces[i] += sum_j (force_mag[i,j] * direction[i,j])
    # Note: direction[i,j] points from j to i.
    # If we want to push i away from j, force on i is in direction (i - j).
    # diff[i,j] = C_i - C_j. So directions are correct.
    
    # We need to sum over the second axis (j)
    # forces shape (n, 2)
    # term shape (n, n, 2)
    force_vectors = force_mags[:, :, np.newaxis] * directions
    
    # Sum over neighbors
    forces += np.sum(force_vectors, axis=1)
    
    # Boundary forces
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        
        # Left wall
        if x - r < 0:
            forces[i, 0] += k_wall * (r - x) # Push right
        # Right wall
        if x + r > 1:
            forces[i, 0] -= k_wall * (x + r - 1) # Push left

        # Top wall
        if y + r > 1:
            forces[i, 1] -= k_wall * (y + r - 1) # Push down
            
    return forces
