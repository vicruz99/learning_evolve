import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def generate_initial_guess(n):
    """Generates a hexagonal lattice packing as an initial guess."""
    # Estimate radius for a hexagonal packing of n circles in a unit square
    # Area per circle in hex packing is 2*sqrt(3)*r^2
    # n * 2*sqrt(3)*r^2 approx 1  =>  r approx sqrt(1 / (2*sqrt(3)*n))
    r_est = np.sqrt(1.0 / (2.0 * np.sqrt(3.0) * n))
    
    # Place circles in a hexagonal grid
    centers = []
    y = r_est
    row = 0
    while y + r_est <= 1.0:
        x = r_est
        if row % 2 == 1:
            x += r_est  # Shift odd rows horizontally
        while x + r_est <= 1.0:
            centers.append([x, y])
            x += 2.0 * r_est
        y += np.sqrt(3.0) * r_est
        row += 1
        
    # If we have fewer than n points from the grid, fill the rest or truncate
    centers = np.array(centers)
    if len(centers) < n:
        # Fallback to a simple grid if lattice doesn't fit enough (rare for this setup)
        cols = int(np.ceil(np.sqrt(n)))
        rows = int(np.ceil(n / cols))
        temp_centers = np.zeros((n, 2))
        for i in range(n):
            temp_centers[i, 0] = (i % cols + 0.5) / cols
            temp_centers[i, 1] = (i // cols + 0.5) / rows
        centers = temp_centers[:n]
    else:
        centers = centers[:n]
        
    # Use a slightly smaller radius initially to ensure feasibility
    radii = np.full(n, r_est * 0.9)
    return centers, radii

def compute_objective(vars_array, n):
    """Objective function: negative sum of radii."""
    radii = vars_array[2 * n:]
    return -np.sum(radii)

def compute_constraint_vector(vars_array, n):
    """
    Computes all inequality constraints g(x) >= 0.
    Includes boundary and pairwise non-overlap constraints.
    """
    centers = vars_array[:2 * n].reshape(n, 2)
    radii = vars_array[2 * n:]
    
    constraints = []
    
    # 1. Boundary constraints: 4 per circle
    # x - r >= 0
    constraints.extend(centers[:, 0] - radii)
    # 1 - x - r >= 0
    constraints.extend(1.0 - centers[:, 0] - radii)
    # y - r >= 0
    constraints.extend(centers[:, 1] - radii)
    # 1 - y - r >= 0
    constraints.extend(1.0 - centers[:, 1] - radii)
    
    # 2. Non-overlap constraints: dist_ij >= r_i + r_j
    # Vectorized computation of all pairwise distances
    # diff shape: (n, n, 2)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff ** 2, axis=2))
    
    # Sum of radii matrix
    r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    
    # We only need constraints for i < j (upper triangle)
    # Extract upper triangle elements
    i_indices, j_indices = np.triu_indices(n, k=1)
    overlaps = dists[i_indices, j_indices] - r_sum[i_indices, j_indices]
    constraints.extend(overlaps)
    
    return np.array(constraints)

def run_packing() -> tuple:
    """
    Runs the optimization to pack 26 circles in a unit square.
    Returns (centers, radii, sum_radii).
    """
    n = N_CIRCLES
    
    # 1. Initial Guess
    centers_init, radii_init = generate_initial_guess(n)
    x0 = np.concatenate([centers_init.flatten(), radii_init])
    
    # 2. Bounds
    # x, y in [0, 1], r in [1e-6, 0.2]
    bounds = []
    for _ in range(2 * n):
        bounds.append((0.0, 1.0))
    for _ in range(n):
        bounds.append((1e-6, 0.2))
        
    # 3. Constraints
    # SLSQP expects a function that returns an array of constraint values
    def constraint_func(vars_array):
        return compute_constraint_vector(vars_array, n)
        
    constraints = {'type': 'ineq', 'fun': constraint_func}
    
    # 4. Optimization
    # SLSQP is suitable for this type of constrained non-linear problem
    result = minimize(
        compute_objective,
        x0,
        args=(n,),
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 2000, 'ftol': 1e-15, 'disp': False}
    )
    
    if result.success or np.all(compute_constraint_vector(result.x, n) >= -1e-9):
        final_vars = result.x
    else:
        # If optimization failed or constraints are violated, fallback to initial
        final_vars = x0
        
    final_centers = final_vars[:2 * n].reshape(n, 2)
    final_radii = final_vars[2 * n:]
    total_sum = np.sum(final_radii)
    
    return final_centers, final_radii, total_sum
