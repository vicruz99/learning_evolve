import numpy as np
from scipy.optimize import linprog


def solve_radii_lp(centers, n):
    """Given fixed centers, solve LP to find optimal radii maximizing sum of radii."""
    c = -np.ones(n)

    n_pairs = n * (n - 1) // 2
    n_constraints = n_pairs + n

    A_ub = np.zeros((n_constraints, n))
    b_ub = np.zeros(n_constraints)

    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dist
            idx += 1

    for i in range(n):
        x, y = centers[i]
        max_r = min(x, 1.0 - x, y, 1.0 - y)
        A_ub[idx, i] = 1.0
        b_ub[idx] = max_r
        idx += 1

    bounds = [(0.0, None)] * n

    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')

    if result.success and np.all(result.x >= -1e-10):
        return np.maximum(result.x, 0.0)
    else:
        return np.full(n, 0.01)


def optimize_positions_force(centers, radii, n, steps=1000, lr_init=0.008):
    """Force-based position optimization with gradual cooling."""
    centers = centers.copy()
    lr = lr_init

    for step in range(steps):
        forces = np.zeros((n, 2))

        for i in range(n):
            fx, fy = 0.0, 0.0
            x, y = centers[i]
            r = radii[i]

            # Boundary repulsion
            margin = 0.002
            if x - r < margin:
                fx += 2000.0 * (margin - (x - r))
            if x + r > 1.0 - margin:
                fx -= 2000.0 * (x + r - (1.0 - margin))
            if y - r < margin:
                fy += 2000.0 * (margin - (y - r))
            if y + r > 1.0 - margin:
                fy -= 2000.0 * (y + r - (1.0 - margin))

            # Inter-circle repulsion
            for j in range(n):
                if i == j:
                    continue
                dx = x - centers[j, 0]
                dy = y - centers[j, 1]
                dist_sq = dx * dx + dy * dy
                dist = np.sqrt(dist_sq) if dist_sq > 1e-20 else 1e-10
                min_dist = r + radii[j]
                if dist < min_dist:
                    overlap = min_dist - dist
                    fx += 800.0 * overlap * dx / dist
                    fy += 800.0 * overlap * dy / dist

            forces[i, 0] = fx
            forces[i, 1] = fy

        centers += lr * forces
        centers = np.clip(centers, 0.001, 0.999)
        lr *= 0.9997

    return centers

def hexagonal_init(n, offset_x=0.1, offset_y=0.1, spacing_x=0.17, spacing_y=0.15):
    """Initialize circles in a hexagonal grid pattern."""
    centers = np.zeros((n, 2))
    idx = 0

    for row in range(8):
        for col in range(7):
            if idx >= n:
                break
            x = offset_x + col * spacing_x + (row % 2) * (spacing_x * 0.5)
            y = offset_y + row * spacing_y
            centers[idx, 0] = np.clip(x, 0.05, 0.95)
            centers[idx, 1] = np.clip(y, 0.05, 0.95)
            idx += 1
        if idx >= n:
            break

    return centers


def random_init(n, seed=42):
    """Initialize circles at random positions."""
    rng = np.random.RandomState(seed)
    centers = rng.uniform(0.1, 0.9, (n, 2))
    return centers


def corner_biased_init(n, seed=42):
    """Initialize with circles biased toward corners for larger radii."""
    rng = np.random.RandomState(seed)
    centers = np.zeros((n, 2))

    # Place some circles near corners
    corners = [[0.1, 0.1], [0.9, 0.1], [0.1, 0.9], [0.9, 0.9]]
    idx = 0

    for cx, cy in corners:
        for _ in range(3):
            if idx >= n:
                break
            centers[idx, 0] = cx + rng.uniform(-0.08, 0.08)
            centers[idx, 1] = cy + rng.uniform(-0.08, 0.08)
            centers[idx, 0] = np.clip(centers[idx, 0], 0.05, 0.95)
            centers[idx, 1] = np.clip(centers[idx, 1], 0.05, 0.95)
            idx += 1

    # Fill remaining with random
    for _ in range(n - idx):
        centers[idx, 0] = rng.uniform(0.1, 0.9)
        centers[idx, 1] = rng.uniform(0.1, 0.9)
        idx += 1

    return centers
