1. For any fixed set of centers, finding the optimal radii that maximize $\sum r_i$ subject to non-overlap and boundary constraints is a Linear Programming problem. The constraints are linear in $r_i$: $r_i + r_j \le \text{dist}(c_i, c_j)$ and $r_i \le \text{dist}(c_i, \text{boundary})$.

3. We use `scipy.optimize.minimize` with the `Powell` method (derivative-free, robust for non-smooth objectives) starting from multiple perturbed hexagonal-like grid configurations to avoid local optima.
4. Post-processing ensures numerical strictness against the validator's `1e-12` tolerance by slightly scaling radii down if any constraint is borderline violated.

import numpy as np
from scipy.optimize import minimize, linprog

def _compute_max_sum_radii(centers):
    """Solve LP to find optimal radii for fixed centers maximizing sum(r)."""
    n = centers.shape[0]
    
    # Maximum possible radius for each circle based on boundaries
    lims = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]), 
                      np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    
    # Pairwise distances vectorized
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    # LP: max sum(r) s.t. r_i + r_j <= dists[i,j], r_i <= lims[i], r >= 0
    n_pairs = n * (n - 1) // 2
    A_ub = np.zeros((n_pairs + n, n))
    b_ub = np.zeros(n_pairs + n)
    
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1
            
    for i in range(n):
        A_ub[idx, i] = 1.0
        b_ub[idx] = lims[i]
        idx += 1
        
    res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    if res.success:
        return -res.fun, res.x
    return 0.0, np.zeros(n)

def _objective(centers_flat):
    """Objective function for minimizer: negative sum of optimal radii."""
    centers = centers_flat.reshape(-1, 2)
    val, _ = _compute_max_sum_radii(centers)
    return -val

def run_packing():
    np.random.seed(42)
    n = 26
    best_sum = -1.0
    best_centers = None
    c_bounds = [(0.0, 1.0)] * (2 * n)
    
    # Multiple restarts from perturbed grids to escape local optima
    for seed in range(20):
        np.random.seed(seed)
        
        # Hexagonal-ish grid initialization
        x = np.linspace(0.1, 0.9, 6)
        y = np.linspace(0.1, 0.9, 5)
        cx, cy = np.meshgrid(x, y)
        c_grid = np.vstack([cx.ravel(), cy.ravel()]).T
        
        # Shuffle and select n points
        idx = np.random.permutation(c_grid.shape[0])[:n]
        c_init = c_grid[idx]
        
        # Add controlled noise
        c_init += np.random.uniform(-0.03, 0.03, c_init.shape)
        c_init = np.clip(c_init, 0.01, 0.99)
        
        # Optimize centers
        res = minimize(_objective, c_init.ravel(), method='Powell', 
                       bounds=c_bounds, options={'maxiter': 250, 'ftol': 1e-9})
        
        if -res.fun > best_sum:
            best_sum = -res.fun
            best_centers = res.x.reshape(-1, 2)
            
    # Compute final optimal radii
    _, best_radii = _compute_max_sum_radii(best_centers)
    
    # Numerical safety: ensure strict satisfaction of validator tolerance
    scale = 1.0
    for i in range(n):
        for j in range(i + 1, n):
            d = np.sqrt(np.sum((best_centers[i] - best_centers[j])**2))
            req = best_radii[i] + best_radii[j]
            if req > d - 1e-13 and req > 0:
                s = (d + 1e-14) / req
                if s < scale:
                    scale = s
    best_radii *= scale
    best_radii = np.maximum(best_radii, 0.0)
    
    # Boundary safety
    for i in range(n):
        lim = np.min(np.array([best_centers[i,0], 1-best_centers[i,0], 
                               best_centers[i,1], 1-best_centers[i,1]]))
        if best_radii[i] > lim - 1e-13:
            best_radii[i] = lim - 1e-14
            
    return best_centers, best_radii, float(np.sum(best_radii))
