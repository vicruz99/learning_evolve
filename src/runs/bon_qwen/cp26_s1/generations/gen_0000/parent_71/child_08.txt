def solve_radii_for_centers(centers):
    """
    Given fixed centers, solve for optimal radii to maximize sum of radii.
    Uses Linear Programming.
    """
    n = centers.shape[0]
    
    # Objective: maximize sum(r) => minimize -sum(r)
    c_obj = -np.ones(n)
    
    # Bounds for radii: 0 <= r_i <= 0.5 (since square is unit size)
    # Actually r_i <= 0.5 is implied by constraints, but safe bound.
    bounds = [(0, 0.5) for _ in range(n)]
    
    # Inequality constraints A_ub @ r <= b_ub
    constraints_A = []
    constraints_b = []
    
    # 1. Boundary constraints
    # r_i <= x_i
    # r_i <= 1 - x_i
    # r_i <= y_i
    # r_i <= 1 - y_i
    for i in range(n):
        x, y = centers[i]
        
        # r_i <= x
        row = np.zeros(n)
        row[i] = 1.0
        constraints_A.append(row)
        constraints_b.append(x)
        
        # r_i <= 1 - x
        row = np.zeros(n)
        row[i] = 1.0
        constraints_A.append(row)
        constraints_b.append(1.0 - x)
        
        # r_i <= y
        row = np.zeros(n)
        row[i] = 1.0
        constraints_A.append(row)
        constraints_b.append(y)
        
        # r_i <= 1 - y
        row = np.zeros(n)
        row[i] = 1.0
        constraints_A.append(row)
        constraints_b.append(1.0 - y)
        
    # 2. Pairwise constraints: r_i + r_j <= dist(i, j)
    # dist(i, j) = sqrt((xi-xj)^2 + (yi-yj)^2)

        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            constraints_A.append(row)
            constraints_b.append(dist)
            
    A_ub = np.array(constraints_A)
    b_ub = np.array(constraints_b)
    
    try:
        # Use HiGHS solver if available, it's fast
        res = opt.linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')

            radii = res.x

            radii = np.maximum(radii, 0)
            return radii, -res.fun
        else:
            # Fallback or return zeros if LP fails
            return np.zeros(n), 0.0
    except Exception:
        return np.zeros(n), 0.0
