from scipy.optimize import minimize

N = 26

def objective(vars):
    """Objective function to minimize (negative sum of radii)."""
    radii = vars[2::3]
    return -np.sum(radii)

def constraint_func(vars):
    """Constraint function: pairwise distances must be >= sum of radii."""
    xs = vars[0::3]
    ys = vars[1::3]
    rs = vars[2::3]
    
    # Compute pairwise distances
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dists = np.sqrt(dx**2 + dy**2)

    sum_r = rs[:, None] + rs[None, :]
    
    # Slack: dist - (ri + rj) >= 0
    slacks = dists - sum_r
    
    # Ignore self-distance constraints
    np.fill_diagonal(slacks, np.inf)
    
    # Return only upper triangle constraints to reduce redundancy
    return slacks[np.triu_indices(N, k=1)]
