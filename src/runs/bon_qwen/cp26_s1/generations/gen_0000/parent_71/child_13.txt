import numpy as np
import math
from scipy.optimize import minimize, linprog


def compute_max_radii_lp(centers, n):
    """Compute maximum sum of radii for given centers using LP."""
    c_obj = -np.ones(n)
    
    A_ub = []
    b_ub = []
    
    for i in range(n):
        b_i = min(centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])
        row = np.zeros(n)
        row[i] = 1
        A_ub.append(row)
        b_ub.append(b_i)
    
    for i in range(n):
        for j in range(i + 1, n):
            d = math.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            row = np.zeros(n)
            row[i] = 1
            row[j] = 1
            A_ub.append(row)
            b_ub.append(d)
    
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0, None) for _ in range(n)]
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        return -result.fun, -result.x
    else:
        return 0, np.zeros(n)


def compute_max_radii_iterative(centers, n, max_iter=500):
    """Compute maximum feasible radii for given centers iteratively."""
    radii = np.ones(n) * 0.001
    for _ in range(max_iter):
        new_radii = np.zeros(n)
        for i in range(n):
            r = min(centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])
            for j in range(n):
                if i != j:
                    d = math.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                    r = min(r, d - radii[j])
            new_radii[i] = max(1e-12, r)
        if np.max(np.abs(new_radii - radii)) < 1e-14:
            break
        radii = new_radii
    return radii


def objective_with_penalty(x, n, pw):
    """Objective function with penalty for constraint violations."""
    centers = x[:2 * n].reshape(n, 2)
    radii = x[2 * n:]
    radii = np.maximum(radii, 1e-12)
    
    penalty = 0.0
    for i in range(n):
        cx, cy = centers[i]
        r = radii[i]
        if cx - r < 0:
            penalty += (cx - r) ** 2
        if 1 - cx - r < 0:
            penalty += (1 - cx - r) ** 2
        if cy - r < 0:
            penalty += (cy - r) ** 2
        if 1 - cy - r < 0:
            penalty += (1 - cy - r) ** 2
    
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = math.sqrt(dx * dx + dy * dy)
            overlap = radii[i] + radii[j] - dist
            if overlap > 0:
                penalty += overlap ** 2
    
    return -np.sum(radii) + pw * penalty


def objective_lp(centers_flat, n):
    """Objective using LP-based radius computation."""
    centers = centers_flat.reshape(n, 2)
    _, radii = compute_max_radii_lp(centers, n)
    return -np.sum(radii)


def run_packing():
    n = 26
    
    best_sum = 0
    best_centers = None
    best_radii = None
    
    num_trials = 25
    
    for trial in range(num_trials):
        np.random.seed(trial * 7 + 3)
        
        centers_list = []
        y = 0.04 + trial * 0.002
        row = 0
        while len(centers_list) < n:
            x_offset = 0.05 if row % 2 == 1 else 0
            x = 0.04 + x_offset + trial * 0.001
            while len(centers_list) < n and x < 0.96:
                cx = x + np.random.uniform(-0.008, 0.008)
                cy = y + np.random.uniform(-0.008, 0.008)
                cx = max(0.01, min(0.99, cx))
                cy = max(0.01, min(0.99, cy))
                centers_list.append([cx, cy])
                x += 0.105
            y += 0.085
            row += 1
        
        centers_init = np.array(centers_list[:n])
        radii_init = np.ones(n) * 0.005
        
        x0 = np.concatenate([centers_init.flatten(), radii_init])
        bounds_opt = [(0.002, 0.998)] * (2 * n) + [(1e-12, 0.45)] * n
        
        x = x0.copy()
        pw = 80
        for phase in range(10):
            def obj(x, penalty=pw):
                return objective_with_penalty(x, n, penalty)
            
            result = minimize(obj, x, method='L-BFGS-B', bounds=bounds_opt,
                              options={'maxiter': 8000, 'ftol': 1e-16, 'gtol': 1e-12})
            x = result.x
            pw *= 4
        
        c_opt = x[:2 * n].reshape(n, 2)
        
        _, r_lp = compute_max_radii_lp(c_opt, n)

        sum_iter = np.sum(r_iter)
        
        if sum_lp > sum_iter:
            r_opt = r_lp
            current_sum = sum_lp
        else:
            r_opt = r_iter
            current_sum = sum_iter
        
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = c_opt.copy()
            best_radii = r_opt.copy()

    final_radii = compute_max_radii_iterative(best_centers, n, max_iter=1000)

        best_sum = np.sum(final_radii)
    else:
        best_radii = refined_radii
        best_sum = np.sum(refined_radii)
    
    best_radii = np.maximum(best_radii, 0)
    best_centers = np.clip(best_centers, 0, 1)
    
    return best_centers, best_radii, best_sum
