import numpy as np

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square

    Args:
        centers: np.array of shape (n, 2) with (x, y) coordinates
        radii: np.array of shape (n) with radius of each circle

    Returns:
        True if valid, False otherwise
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True

def compute_energy_and_gradient(centers, radii):
    """
    Computes the penalty energy and the gradient with respect to centers.
    Energy penalizes overlaps between circles and violations of the unit square boundaries.
    """
    n = centers.shape[0]
    energy = 0.0
    gradients = np.zeros_like(centers)

    # Circle-Circle overlaps
    # Vectorized computation for performance
    # dist_sq[i, j] = |c_i - c_j|^2
    # Using broadcasting to compute pairwise differences
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :] # (n, n, 2)
    dist_sq = np.sum(diff**2, axis=2) # (n, n)
    dist = np.sqrt(dist_sq) # (n, n)
    
    # Radii sum matrix
    r_sum = radii[:, np.newaxis] + radii[np.newaxis, :] # (n, n)
    
    # Overlap condition: dist < r_sum
    overlap = r_sum - dist
    mask = overlap > 0
    
    # Penalty: overlap^2
    # We only consider upper triangle to avoid double counting, but for gradient 
    # it's easier to compute for all and divide or handle carefully.
    # Let's compute energy for all pairs and divide by 2.
    energy_overlap = np.sum(overlap[mask]**2)
    energy += energy_overlap / 2.0

    # Gradient for circle-circle
    # d/d(c_i) [ (r_i + r_j - dist_ij)^2 ] 
    # = 2 * (r_i + r_j - dist_ij) * (-1) * (c_i - c_j) / dist_ij
    # = -2 * overlap_ij * diff_ij / dist_ij
    # Force is in direction of -gradient (to minimize energy), so we add:
    # 2 * overlap_ij * diff_ij / dist_ij
    
    # Avoid division by zero
    dist_safe = np.where(dist < 1e-9, 1e-9, dist)
    
    # Gradient contribution
    # Term: 2 * overlap * diff / dist
    grad_term = 2.0 * overlap * (diff / dist_safe[:, :, np.newaxis])
    # Apply mask to ignore non-overlaps (where overlap <= 0)
    grad_term[~mask] = 0.0
    
    # Sum gradients for each circle
    # grad_term[i, j] is the force on i due to j
    # Total gradient on i is sum over j of grad_term[i, j]
    gradients += np.sum(grad_term, axis=1)

    # Boundary violations
    # x bounds: 0 <= x <= 1. Circle radius r.
    # Constraint: x >= r => r - x <= 0. Penalty: max(0, r - x)^2
    # Constraint: x <= 1 - r => x + r - 1 <= 0. Penalty: max(0, x + r - 1)^2
    
    # Left wall (x=0)
    violation_left_x = radii - centers[:, 0]
    mask_left_x = violation_left_x > 0
    energy += np.sum(violation_left_x[mask_left_x]**2)
    
    # Gradient for x: d/dx [ (r - x)^2 ] = 2(r - x)(-1) = -2(r - x)
    # We want to move x to increase it (away from wall), so force is +2(r-x)
    # But gradient descent moves in -gradient direction.
    # Wait, we defined energy. We want to minimize energy.
    # Gradient of energy w.r.t x is -2(r-x).
    # Gradient descent step: x = x - alpha * (-2(r-x)) = x + 2*alpha*(r-x).
    # So the "force" to add to position is 2(r-x).
    force_left_x = 2.0 * violation_left_x
    force_left_x[~mask_left_x] = 0.0
    gradients[:, 0] += force_left_x

    # Right wall (x=1)
    violation_right_x = centers[:, 0] + radii - 1.0
    mask_right_x = violation_right_x > 0
    energy += np.sum(violation_right_x[mask_right_x]**2)
    
    # Gradient for x: d/dx [ (x + r - 1)^2 ] = 2(x + r - 1)(1) = 2(x + r - 1)
    # We want to move x to decrease it. Force should be negative.
    # Gradient is positive. Descent step x = x - alpha * grad => x decreases.
    # So we add gradient to the accumulator for the update rule x -= alpha * grad.
    # Wait, in the circle-circle part, I derived "force to add".
    # Let's stick to standard gradient descent: x_new = x_old - alpha * grad_x.
    # grad_x = 2 * violation_right_x.
    # So we add this to gradients array.
    grad_right_x = 2.0 * violation_right_x
    grad_right_x[~mask_right_x] = 0.0
    gradients[:, 0] += grad_right_x

    # Bottom wall (y=0)
    violation_bottom_y = radii - centers[:, 1]
    mask_bottom_y = violation_bottom_y > 0
    energy += np.sum(violation_bottom_y[mask_bottom_y]**2)
    
    force_bottom_y = 2.0 * violation_bottom_y
    force_bottom_y[~mask_bottom_y] = 0.0
    gradients[:, 1] += force_bottom_y

    # Top wall (y=1)
    violation_top_y = centers[:, 1] + radii - 1.0
    mask_top_y = violation_top_y > 0
    energy += np.sum(violation_top_y[mask_top_y]**2)
    
    grad_top_y = 2.0 * violation_top_y
    grad_top_y[~mask_top_y] = 0.0
    gradients[:, 1] += grad_top_y

    return energy, gradients

        r_scale *= 1.0005
