from scipy.optimize import minimize


def make_hexagonal_initial(n, n_rows=5):
    """Create initial hexagonal grid placement."""
    centers = np.zeros((n, 2))
    idx = 0
    
    # Distribute circles across rows
    circles_per_row = [0] * n_rows
    remaining = n
    for r in range(n_rows):
        if r == 0:
            circles_per_row[r] = min(6, remaining)
        else:
            base = 5
            if remaining <= base:
                circles_per_row[r] = remaining
            else:
                circles_per_row[r] = base
            remaining -= circles_per_row[r]
    
    # Adjust if needed
    while remaining > 0:
        for r in range(n_rows - 1, -1, -1):

                circles_per_row[r] += 1
                remaining -= 1

    y_spacing = 1.0 / (n_rows + 1)

        num_in_row = circles_per_row[row]
        y = (row + 1) * y_spacing
        x_spacing = 1.0 / (num_in_row + 1)

            if row % 2 == 1 and num_in_row > 0:
                x = (col + 1.5) * x_spacing

            if idx < n:
                centers[idx] = [x, y]
                idx += 1

            break
    
    return centers


def compute_penalty_term(x, n):
    """Compute penalty for constraint violations."""
    penalty = 0.0
    for i in range(n):
        xi, yi, ri = x[3*i], x[3*i+1], x[3*i+2]
        # Boundary: x - r >= 0
        violation = max(0.0, -(xi - ri))
        penalty += violation * violation
        # Boundary: x + r <= 1
        violation = max(0.0, xi + ri - 1.0)
        penalty += violation * violation
        # Boundary: y - r >= 0
        violation = max(0.0, -(yi - ri))
        penalty += violation * violation
        # Boundary: y + r <= 1
        violation = max(0.0, yi + ri - 1.0)
        penalty += violation * violation
    
    # Overlap penalties
    for i in range(n):
        for j in range(i + 1, n):
            xi, yi, ri = x[3*i], x[3*i+1], x[3*i+2]
            xj, yj, rj = x[3*j], x[3*j+1], x[3*j+2]
            dx = xi - xj
            dy = yi - yj
            dist = np.sqrt(dx * dx + dy * dy)
            if dist < ri + rj:
                violation = ri + rj - dist
                penalty += violation * violation
    
    return penalty


def objective_penalized(x, penalty_weight):
    """Objective: maximize sum of radii with penalty for violations."""
    n = len(x) // 3
    sum_radii = sum(x[3*i + 2] for i in range(n))
    penalty = compute_penalty_term(x, n)
    return -sum_radii + penalty_weight * penalty


def force_based_refinement(centers, radii, n, iterations=500):
    """Refine packing using repulsive forces."""
    centers = centers.copy()
    radii = radii.copy()
    
    for it in range(iterations):
        step = 0.005 / (1.0 + it * 0.002)
        forces = np.zeros((n, 2))
        
        for i in range(n):
            for j in range(i + 1, n):
                diff = centers[j] - centers[i]
                dist = np.sqrt(np.sum(diff * diff))
                min_dist = radii[i] + radii[j]
                if dist < min_dist and dist > 1e-10:
                    overlap = min_dist - dist
                    force = diff / dist * overlap * 50.0
                    forces[i] -= force
                    forces[j] += force
                elif dist < min_dist * 1.1 and dist > 1e-10:
                    force = diff / dist * 0.001 * 10.0
                    forces[i] -= force
                    forces[j] += force
        
        # Boundary forces
        for i in range(n):
            if centers[i, 0] - radii[i] < 0:
                forces[i, 0] += (-(centers[i, 0] - radii[i])) * 100.0
            if centers[i, 0] + radii[i] > 1:
                forces[i, 0] -= (centers[i, 0] + radii[i] - 1.0) * 100.0
            if centers[i, 1] - radii[i] < 0:
                forces[i, 1] += (-(centers[i, 1] - radii[i])) * 100.0
            if centers[i, 1] + radii[i] > 1:
                forces[i, 1] -= (centers[i, 1] + radii[i] - 1.0) * 100.0
        
        centers += forces * step
        
        # Clip to valid range
        for i in range(n):
            r = radii[i]
            centers[i, 0] = max(r, min(1.0 - r, centers[i, 0]))
            centers[i, 1] = max(r, min(1.0 - r, centers[i, 1]))

                max_r = min(centers[i, 0], 1.0 - centers[i, 0],

                        d = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))

                if max_r > radii[i]:
                    radii[i] = max(radii[i], radii[i] * 1.002)
                    radii[i] = min(radii[i], max_r * 0.999)
    
    return centers, radii


def run_optimization_from(centers_init, n):
    """Run optimization from given initial centers."""
    radii_init = np.ones(n) * 0.025
    x0 = np.zeros(3 * n)
    for i in range(n):
        x0[3*i] = centers_init[i, 0]
        x0[3*i+1] = centers_init[i, 1]
        x0[3*i+2] = radii_init[i]
    
    bounds = [(0.002, 0.998)] * (2*n) + [(0.002, 0.5)] * n
    
    best_x = x0.copy()
    best_sum_r = sum(best_x[3*i+2] for i in range(n))
    
    # Penalty annealing with multiple stages
    penalties = [50, 200, 1000, 5000, 20000, 100000, 500000, 2000000]
    
    for pw in penalties:
        def obj(x):
            return objective_penalized(x, pw)
        result = minimize(obj, best_x, method='L-BFGS-B', bounds=bounds,
                         options={'maxiter': 2000, 'ftol': 1e-15, 'gtol': 1e-12})
        best_x = result.x
    
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        centers[i] = [best_x[3*i], best_x[3*i+1]]
        radii[i] = best_x[3*i+2]
    
    s = sum(radii)
    return centers, radii, s
