import numpy as np
from scipy.optimize import minimize

def objective(x, n):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(x[2*n:])

def constraint_func(x, n):
    """
    Compute all inequality constraints as a single 1D array.
    Returns values that must be >= 0.
    """
    c = x[:2*n].reshape(n, 2)
    r = x[2*n:]
    
    # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    # Broadcasting to compute all pairwise differences at once
    dx = c[:, 0, None] - c[:, 0]
    dy = c[:, 1, None] - c[:, 1]
    dist_sq = dx*dx + dy*dy
    
    r_sum = r[:, None] + r[None, :]
    
    # Extract only upper triangle (i < j) to avoid duplicates and self-constraints
    tri = np.triu_indices(n, k=1)
    pair_cons = dist_sq[tri] - r_sum[tri]**2
    
    # Boundary constraints: c >= r  and  1 - c >= r
    b1 = c[:, 0] - r
    b2 = 1.0 - c[:, 0] - r
    b3 = c[:, 1] - r
    b4 = 1.0 - c[:, 1] - r
    
    return np.concatenate([pair_cons, b1, b2, b3, b4])

def run_packing():
    """
    Solve the circle packing problem for 26 circles in a unit square.
    Returns (centers, radii, sum_radii).
    """
    n = 26
    best_sum = -np.inf
    best_centers = None
    best_radii = None
    
    # Multiple restarts with deterministic seeds for reproducibility
    seeds = [42, 123, 456, 789, 101112, 131415, 161718, 192021, 222324, 252627]
    
    for seed in seeds:
        rng = np.random.RandomState(seed)
        
        # Initialize centers in the interior to avoid immediate boundary conflicts
        centers = rng.uniform(0.1, 0.9, (n, 2))
        radii = np.full(n, 0.05)
        x0 = np.concatenate([centers.flatten(), radii])
        
        bounds = [(0, 1)] * (2 * n) + [(0, 0.5)] * n
        
        try:
            res = minimize(
                func=objective,
                x0=x0,
                args=(n,),
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'ineq', 'fun': constraint_func, 'args': (n,)},
                options={'maxiter': 1000, 'ftol': 1e-10, 'disp': False}
            )
            
            cur_sum = -res.fun
            if cur_sum > best_sum:
                c = res.x[:2*n].reshape(n, 2)
                r = res.x[2*n:]
                # Quick validity check
                if np.all(r >= 0) and np.all(c >= 0) and np.all(c <= 1):
                    best_sum = cur_sum
                    best_centers = c.copy()
                    best_radii = r.copy()
        except Exception:
            continue
            
    # Fallback in case optimization fails entirely
    if best_centers is None:
        best_centers = np.tile([0.5, 0.5], (n, 1)) + np.array([[i*0.01, i*0.02] for i in range(n)])
        best_radii = np.full(n, 0.02)
        best_sum = np.sum(best_radii)
        
    return best_centers, best_radii, best_sum
