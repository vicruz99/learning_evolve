import numpy as np
from scipy.optimize import linprog, differential_evolution

def solve_radii(centers):
    """
    Solves the LP to maximize sum of radii for a fixed set of centers.
    
    Args:
        centers: np.array of shape (n, 2)
        
    Returns:
        radii: np.array of shape (n)
        sum_radii: float
    """
    n = centers.shape[0]
    
    # Objective: Maximize sum(r) => Minimize -sum(r)
    c_obj = np.ones(n)
    
    # Constraints: A_ub @ r <= b_ub
    # 1. Pairwise non-overlap: r_i + r_j <= dist(i, j)
    # 2. Boundary constraints: r_i <= min(x_i, 1-x_i, y_i, 1-y_i)
    
    n_pairs = n * (n - 1) // 2
    n_bounds = 4 * n
    
    n_constraints = n_pairs + n_bounds
    A_ub = np.zeros((n_constraints, n))
    b_ub = np.zeros(n_constraints)
    
    row_idx = 0
    
    # Pairwise constraints
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            A_ub[row_idx, i] = 1.0
            A_ub[row_idx, j] = 1.0
            b_ub[row_idx] = dist
            row_idx += 1
            
    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        bounds = [x, 1 - x, y, 1 - y]
        for b in bounds:
            A_ub[row_idx, i] = 1.0
            b_ub[row_idx] = b
            row_idx += 1
            
    # Bounds for variables (r_i >= 0)
    bounds = [(0, None) for _ in range(n)]
    
    # Solve LP
    res = linprog(-c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        radii = res.x
        return radii, -res.fun
    else:
        # Fallback: return small radii if LP fails
        return np.zeros(n), 0.0

def get_sum_radii(centers_flat):
    """
    Objective function for optimization.
    Minimizes negative sum of radii.
    """
    centers = centers_flat.reshape((26, 2))
    radii, sum_r = solve_radii(centers)
    return -sum_r

def run_packing():
    n_circles = 26
    
    # Bounds for each coordinate (0 to 1)
    bounds = [(0.0, 1.0) for _ in range(n_circles * 2)]
    
    # Initial guess: A slightly perturbed grid
    # 5x5 grid is 25 points. We need 26.
    # Let's generate 26 points in a rough grid pattern
    rng = np.random.default_rng(42)
    
    # Create a 5x5 grid
    x_grid = np.linspace(0.1, 0.9, 5)
    y_grid = np.linspace(0.1, 0.9, 5)
    points = []
    for y in y_grid:
        for x in x_grid:
            points.append([x, y])
    
    # Add one more point in a gap, e.g., center of a cell
    # Grid cells are 0.2 wide. Center of cell (0,0) is (0.2, 0.2) relative to 0.1
    # Absolute center (0.2, 0.2)
    points.append([0.2, 0.2])
    
    initial_centers = np.array(points[:26])
    
    # Add some noise to avoid exact symmetry which might confuse optimizer
    initial_centers += rng.normal(0, 0.01, initial_centers.shape)
    initial_centers = np.clip(initial_centers, 0, 1)
    
    # Run Differential Evolution
    # seed for reproducibility
    res = differential_evolution(
        get_sum_radii,
        bounds,
        seed=42,
        maxiter=200,
        popsize=20,
        mutation=(0.5, 1.5),
        recombination=0.9,
        tol=1e-7,
        atol=1e-7,
        initial_guess=initial_centers.flatten()
    )
    
    best_centers = res.x.reshape((n_circles, 2))
    
    # Final calculation of radii
    final_radii, final_sum = solve_radii(best_centers)
    
    return best_centers, final_radii, final_sum

if __name__ == "__main__":
    centers, radii, s_r = run_packing()
    print(f"Sum of radii: {s_r}")
