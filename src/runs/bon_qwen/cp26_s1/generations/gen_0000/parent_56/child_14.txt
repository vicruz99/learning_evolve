import numpy as np
from scipy.optimize import minimize

def resolve_overlaps(centers, radii, n, n_inner=150):
    """Resolve overlaps using repulsion forces between circles."""
    for _ in range(n_inner):
        moved = False
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[j][0] - centers[i][0]
                dy = centers[j][1] - centers[i][1]
                dist_sq = dx * dx + dy * dy
                min_dist = radii[i] + radii[j]
                
                if dist_sq < min_dist * min_dist and dist_sq > 1e-10:
                    dist = np.sqrt(dist_sq)
                    overlap = min_dist - dist
                    nx, ny = dx / dist, dy / dist
                    
                    centers[i][0] -= nx * overlap * 0.5
                    centers[i][1] -= ny * overlap * 0.5
                    centers[j][0] += nx * overlap * 0.5
                    centers[j][1] += ny * overlap * 0.5
                    moved = True
        
        # Clamp to valid bounds
        for i in range(n):
            for dim in range(2):
                centers[i][dim] = max(radii[i], min(1 - radii[i], centers[i][dim]))
        
        if not moved:
            break
    return centers


def grow_and_optimize(centers, radii, n, max_iter=60000, initial_growth=0.0007):
    """Iteratively grow circles and resolve overlaps."""
    growth = initial_growth
    
    for iteration in range(max_iter):
        radii += growth
        
        # Resolve overlaps with repulsion
        moved = False
        for _ in range(200):
            inner_moved = False
            for i in range(n):
                for j in range(i + 1, n):
                    dx = centers[j][0] - centers[i][0]
                    dy = centers[j][1] - centers[i][1]
                    dist_sq = dx * dx + dy * dy
                    min_dist = radii[i] + radii[j]
                    
                    if dist_sq < min_dist * min_dist and dist_sq > 1e-10:
                        dist = np.sqrt(dist_sq)
                        overlap = min_dist - dist
                        nx, ny = dx / dist, dy / dist
                        
                        # Smaller circles move more (inverse radius weighting)
                        total_w = (1.0 / max(radii[i], 1e-8)) + (1.0 / max(radii[j], 1e-8))
                        w_i = (1.0 / max(radii[i], 1e-8)) / total_w
                        w_j = (1.0 / max(radii[j], 1e-8)) / total_w
                        
                        centers[i][0] -= nx * overlap * w_i * 0.5
                        centers[i][1] -= ny * overlap * w_i * 0.5
                        centers[j][0] += nx * overlap * w_j * 0.5
                        centers[j][1] += ny * overlap * w_j * 0.5
                        inner_moved = True
            
            for i in range(n):
                for dim in range(2):
                    centers[i][dim] = max(radii[i], min(1 - radii[i], centers[i][dim]))
            
            if not inner_moved:
                break
            moved = True
        
        # Decay growth rate
        growth *= 0.99995
    
    return centers, radii


def fine_tune_individual(centers, radii, n, n_attempts=500):
    """Try to grow individual circles one at a time."""
    for attempt in range(n_attempts):
        # Random order
        order = np.random.permutation(n)
        for i in order:
            old_r = radii[i]
            radii[i] += 0.002
            
            # Resolve overlaps involving circle i
            for _ in range(80):
                moved = False
                for j in range(n):
                    if i == j:
                        continue
                    dx = centers[j][0] - centers[i][0]
                    dy = centers[j][1] - centers[i][1]
                    dist_sq = dx * dx + dy * dy

                        overlap = min_dist - dist
                        nx, ny = dx / dist, dy / dist

                        centers[i][1] -= ny * overlap * 0.7
                        centers[j][0] += nx * overlap * 0.3
                        centers[j][1] += ny * overlap * 0.3
                        moved = True
                
                for dim in range(2):
                    centers[i][dim] = max(radii[i], min(1 - radii[i], centers[i][dim]))
                
                if not moved:
                    break
            
            # Check validity

            if not valid:

def gradient_optimize(centers, radii, n):
    """Use scipy L-BFGS-B with penalty method for gradient-based optimization."""
    
    def objective_and_penalty(vars_flat):
        c = vars_flat[:2 * n].reshape((n, 2))
        r = vars_flat[2 * n:]
        
        # Objective: negative sum of radii (minimize)
        obj = -np.sum(r)
        
        # Penalty for constraint violations
        penalty = 0.0
        penalty_weight = 500.0
        
        # Boundary penalties
        for i in range(n):
            for dim in range(2):
                if c[i][dim] - r[i] < 0:
                    penalty += penalty_weight * (r[i] - c[i][dim]) ** 2
                if c[i][dim] + r[i] > 1:
                    penalty += penalty_weight * (c[i][dim] + r[i] - 1) ** 2
        
        # Overlap penalties
        for i in range(n):
            for j in range(i + 1, n):
                dx = c[i][0] - c[j][0]
                dy = c[i][1] - c[j][1]
                dist = np.sqrt(dx * dx + dy * dy)
                if dist < r[i] + r[j]:
                    penalty += penalty_weight * (r[i] + r[j] - dist) ** 2
        
        return obj + penalty
    
    vars_flat = np.concatenate([centers.flatten(), radii])
    
    bounds = [(0, 1)] * (2 * n) + [(0, 0.5)] * n
    
    result = minimize(
        objective_and_penalty,
        vars_flat,
        method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': 5000, 'ftol': 1e-12}
    )
    
    new_centers = result.x[:2 * n].reshape((n, 2))
    new_radii = result.x[2 * n:]
    
    # Ensure validity
    for i in range(n):
        for dim in range(2):
            new_centers[i][dim] = max(new_radii[i], min(1 - new_radii[i], new_centers[i][dim]))
    
    return new_centers, new_radii


def run_packing() -> tuple:
    n = 26
    
    best_sum = 0.0

    best_radii = None
    
    # Try multiple initial configurations
    for trial in range(60):

        # Initialize with strategic hexagonal pattern

        row_spacing = 0.14

            for col in range(cols_in_row):
                if idx >= n:
                    break
                x = col * col_spacing + offset + col_spacing / 2
                y = row * row_spacing + row_spacing / 2
                
                # Clip to valid range
                x = min(max(x, 0.05), 0.95)
                y = min(max(y, 0.05), 0.95)
                
                centers[idx] = [x, y]
                idx += 1
            if idx >= n:
                break
        
        # Add small random perturbation
        centers += np.random.uniform(-0.03, 0.03, (n, 2))
        centers = np.clip(centers, 0.05, 0.95)
        
        # Phase 1: Growing circles with repulsion
        centers, radii = grow_and_optimize(centers, radii, n, max_iter=50000, initial_growth=0.0006)
        
        # Phase 2: Fine-tune individual circles
        fine_tune_individual(centers, radii, n, n_attempts=400)
        
        # Phase 3: Gradient-based optimization
        centers, radii = gradient_optimize(centers, radii, n)
        
        # Phase 4: Final cleanup - resolve any remaining overlaps
        resolve_overlaps(centers, radii, n, n_inner=300)
        
        # Validate and compute sum
        valid = True
        for i in range(n):
            for dim in range(2):
                if centers[i][dim] - radii[i] < -1e-10 or centers[i][dim] + radii[i] > 1 + 1e-10:
                    valid = False
            if radii[i] < 0:
                valid = False
        
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[j][0] - centers[i][0]
                dy = centers[j][1] - centers[i][1]
                dist = np.sqrt(dx * dx + dy * dy)
                if dist < radii[i] + radii[j] - 1e-10:

    # Final validation
    for i in range(n):

            best_centers[i][dim] = max(best_radii[i], min(1 - best_radii[i], best_centers[i][dim]))
    
    return best_centers, best_radii, np.sum(best_radii)
