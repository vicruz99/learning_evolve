1. **Radius Optimization (LP)**: For fixed center positions, finding the maximum sum of radii is a Linear Programming problem. The constraints are linear: $r_i \le \text{dist}(c_i, \text{boundary})$ and $r_i + r_j \le \text{dist}(c_i, c_j)$. This gives the exact optimal radii for any configuration.
2. **Center Optimization (Force-Directed Layout)**: The sum of radii is a non-smooth, non-convex function of centers. To navigate this landscape, we use a physics-based repulsion model. Circles repel each other with a force inversely proportional to the square of their distance. We also apply boundary repulsion to keep centers inside the square.

By iterating between these two steps with a decaying step size and random jitter to escape local minima, we effectively perform a coordinate ascent on the sum of radii. import numpy as np
from scipy.optimize import linprog

def compute_optimal_radii(centers):
    """
    Solves the LP to find maximum sum of radii for fixed center positions.
    Returns (radii_array, sum_radii).
    """
    n = centers.shape[0]
    # Distance from center to the nearest boundary
    dist_to_edge = np.minimum(
        np.minimum(centers[:, 0], 1.0 - centers[:, 0]), 
        np.minimum(centers[:, 1], 1.0 - centers[:, 1])
    )
    r_ub = dist_to_edge
    
    # Pairwise constraints: r_i + r_j <= distance(i, j)
    m = n * (n - 1) // 2
    A_ub = np.zeros((m, n))
    b_ub = np.zeros(m)
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            diff = centers[i] - centers[j]
            dist = np.sqrt(np.sum(diff**2))
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dist
            idx += 1
            
    # Objective: max sum(r)  =>  min -sum(r)
    c_obj = -np.ones(n)
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=(0, r_ub), method='highs')
    
    if res.success:
        return res.x, -res.fun
    return np.zeros(n), 0.0

def force_directed_optimization(centers, n_steps=800):
    """
    Iteratively repels centers to find configurations that allow larger radii.
    """
    n = centers.shape[0]
    curr_centers = centers.copy()
    step_size = 0.015
    
    for step in range(n_steps):
        radii, _ = compute_optimal_radii(curr_centers)
        
        forces = np.zeros_like(curr_centers)
        for i in range(n):
            for j in range(i + 1, n):
                diff = curr_centers[i] - curr_centers[j]
                dist = np.sqrt(np.sum(diff**2))
                if dist < 1e-9:
                    dist = 1e-9
                    diff = np.array([1e-9, 0.0])
                
                # Inverse-square repulsion force
                f_mag = 1.0 / (dist**2)
                f_vec = (diff / dist) * f_mag
                forces[i] += f_vec
                forces[j] -= f_vec
                
            # Strong boundary repulsion to keep centers strictly inside
            for d in range(2):
                val = curr_centers[i, d]
                if val < 0.02:
                    forces[i, d] += (0.02 - val) * 200.0
                elif val > 0.98:
                    forces[i, d] -= (val - 0.98) * 200.0
                    
        # Update positions
        curr_centers += forces * step_size
        curr_centers = np.clip(curr_centers, 0.001, 0.999)
        
        # Annealing schedule
        step_size *= 0.996
        
        # Periodic jitter to escape local minima
        if step % 150 == 0 and step > 0:
            curr_centers += np.random.normal(0, 0.003, size=curr_centers.shape)
            curr_centers = np.clip(curr_centers, 0.001, 0.999)
            
    return curr_centers

def run_packing():
    n = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    seeds = []
    
    # Seed 1: Hexagonal grid arrangement
    pts = []
    for r in range(7):
        for c in range(4):
            if len(pts) >= n: break
            x = 0.12 + c * 0.22 + (0.11 if r % 2 else 0.0)
            y = 0.12 + r * 0.14
            pts.append([x, y])
    seeds.append(np.array(pts[:n]))
    
    # Seed 2 & 3: Randomized dense placements
    np.random.seed(42)
    seeds.append(np.random.rand(n, 2) * 0.8 + 0.1)
    np.random.seed(123)
    seeds.append(np.random.rand(n, 2) * 0.6 + 0.2)

    for seed in seeds:
        opt_centers = force_directed_optimization(seed, n_steps=800)
        radii, curr_sum = compute_optimal_radii(opt_centers)
        
        if curr_sum > best_sum:
            best_sum = curr_sum
            best_centers = opt_centers
            best_radii = radii
            
    # Final verification step (recompute to ensure consistency)
    final_radii, final_sum = compute_optimal_radii(best_centers)
    return best_centers, final_radii, final_sum
