import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # 1. Initialization: Hexagonal Lattice
    # Target average radius based on density estimates: ~0.101
    # We start slightly conservative (0.095) to ensure a valid starting point 
    # for the optimizer, which will then push them to the maximum.
    r_init = 0.095
    
    centers = np.zeros((n, 2))
    radii = np.full(n, r_init)
    
    # Fill a hexagonal pattern
    # Row spacing (y): sqrt(3)/2 * 2r = sqrt(3)*r
    # Column spacing (x): 2r
    # Offset alternate rows by r
    row_height = np.sqrt(3) * r_init
    col_width = 2 * r_init
    
    count = 0
    row_idx = 0
    
    # Generate points
    while count < n:
        # Start of row
        y_start = r_init + row_idx * row_height
        # Offset for odd rows
        x_offset = r_init if row_idx % 2 == 0 else r_init + col_width / 2
        
        # How many fit in this row?
        # Center x = x_offset + k * col_width
        # Max x = 1 - r_init
        # x_offset + k * col_width <= 1 - r_init
        # k <= (1 - 2*r_init - x_offset + r_init) / col_width 
        # Simplified: we just place them and let optimizer move them if needed,
        # but staying within [0,1] is best.
        
        # Estimate number of columns for this row
        # Available width approx 1.0
        # Spacing 2r. Max cols approx 1 / (2r) + 1
        
        # Let's just iterate x
        x = x_offset
        while x <= 1.0 - r_init and count < n:
            centers[count] = [x, y_start]
            count += 1
            x += col_width
        
        row_idx += 1
        
        # Safety break to prevent infinite loop if r_init is too small/large logic fails
        if row_idx > 20: 
            break
            
    # If we somehow have fewer than 26 (due to boundary clipping), fill remaining randomly
    if count < n:
        for i in range(count, n):
            centers[i] = [np.random.uniform(0.1, 0.9), np.random.uniform(0.1, 0.9)]

    # Flatten variables for scipy: [x1, y1, r1, x2, y2, r2, ...]
    x0 = np.concatenate([centers.flatten(), radii])
    
    # Bounds for each variable
    # x, y in [0, 1], r in [0, 0.5]
    bounds = []
    for _ in range(n):
        bounds.append((0, 1)) # x
        bounds.append((0, 1)) # y
        bounds.append((0, 0.5)) # r
        
    # Constraints
    constraints = []
    
    # 1. Boundary constraints: x >= r, x <= 1-r, y >= r, y <= 1-r
    # Equivalent to: x - r >= 0, 1 - x - r >= 0, etc.
    for i in range(n):
        idx_x = 3 * i
        idx_y = 3 * i + 1
        idx_r = 3 * i + 2
        
        # x - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: v[idx_x] - v[idx_r]
        })
        # 1 - x - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: 1 - v[idx_x] - v[idx_r]
        })
        # y - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: v[idx_y] - v[idx_r]
        })
        # 1 - y - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: 1 - v[idx_y] - v[idx_r]
        })

    # 2. Non-overlap constraints: dist(i,j)^2 - (ri + rj)^2 >= 0
    # To avoid sqrt, we use squared distance.
    # (xi - xj)^2 + (yi - yj)^2 >= (ri + rj)^2
    for i in range(n):
        for j in range(i + 1, n):
            idx_xi, idx_yi, idx_ri = 3*i, 3*i+1, 3*i+2
            idx_xj, idx_yj, idx_rj = 3*j, 3*j+1, 3*j+2
            
            constraints.append({
                'type': 'ineq',
                'fun': lambda v, i=i, j=j, idx_xi=idx_xi, idx_yi=idx_yi, idx_ri=idx_ri, \
                             idx_xj=idx_xj, idx_yj=idx_yj, idx_rj=idx_rj: \
                             (v[idx_xi] - v[idx_xj])**2 + (v[idx_yi] - v[idx_yj])**2 - (v[idx_ri] + v[idx_rj])**2
            })

    # Objective: Maximize sum of radii -> Minimize negative sum
    def objective(v):
        radii_vals = v[2::3]
        return -np.sum(radii_vals)

    # Run optimization
    # SLSQP is robust for inequality constraints
    result = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints, 
                      options={'maxiter': 1000, 'ftol': 1e-9})
    
    # Extract results
    if result.success:
        optimal_v = result.x
    else:
        # Fallback to initial if optimization fails (unlikely)
        optimal_v = x0
        
    opt_centers = np.zeros((n, 2))
    opt_radii = np.zeros(n)
    
    for i in range(n):
        opt_centers[i, 0] = optimal_v[3*i]
        opt_centers[i, 1] = optimal_v[3*i+1]
        opt_radii[i] = optimal_v[3*i+2]
        
    # Ensure radii are non-negative (numerical safety)
    opt_radii = np.maximum(opt_radii, 0)
    
    sum_r = np.sum(opt_radii)
    
    return opt_centers, opt_radii, sum_r
