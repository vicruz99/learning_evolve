import numpy as np
from scipy.optimize import minimize

def compute_objective(var, n, mu):
    """
    Computes the objective function: negative sum of radii plus penalty terms
    for constraint violations (boundary and overlaps).
    """
    centers = var[:2 * n].reshape(n, 2)
    radii = var[2 * n:]

    # Primary objective: maximize sum of radii (minimize negative sum)
    obj = -np.sum(radii)

    # Penalty for boundary violations
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x < r:
            obj += mu * (r - x) ** 2
        if x > 1.0 - r:
            obj += mu * (x - (1.0 - r)) ** 2
        if y < r:
            obj += mu * (r - y) ** 2
        if y > 1.0 - r:
            obj += mu * (y - (1.0 - r)) ** 2

    # Penalty for circle overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            req = radii[i] + radii[j]
            if dist < req:
                obj += mu * (req - dist) ** 2

    return obj

def run_packing():
    """
    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    Returns (centers, radii, sum_radii).
    """
    n = 26
    best_var = None
    best_sum = -1.0
    # Bounds: x,y in [0,1], radii in [0, 0.5]
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n

    # Run multiple random restarts to avoid poor local optima
    for seed in range(20):
        np.random.seed(seed + 2024)
        # Initialize centers clustered in the middle, small radii to allow growth
        centers = np.random.uniform(0.15, 0.85, size=(n, 2))
        radii = np.ones(n) * 0.04
        var = np.concatenate([centers.flatten(), radii])

        current_var = var
        # Gradually increase penalty weight to enforce constraints strictly
        for mu in [10, 50, 200, 800, 1500]:
            res = minimize(
                compute_objective, 
                current_var, 
                args=(n, mu),
                method='L-BFGS-B', 
                bounds=bounds,
                options={'maxiter': 600, 'ftol': 1e-12}
            )
            current_var = res.x

        # Extract candidate configuration
        centers_cand = current_var[:2 * n].reshape(n, 2)
        radii_cand = current_var[2 * n:]

        # Strict validation check
        valid = True
        for i in range(n):
            x, y = centers_cand[i]
            r = radii_cand[i]
            if x < r - 1e-9 or x > 1.0 - r + 1e-9 or y < r - 1e-9 or y > 1.0 - r + 1e-9:
                valid = False
                break
        if valid:
            for i in range(n):
                for j in range(i + 1, n):
                    d = np.sqrt(np.sum((centers_cand[i] - centers_cand[j]) ** 2))
                    if d < radii_cand[i] + radii_cand[j] - 1e-9:
                        valid = False
                        break
                if not valid:
                    break

        if valid:
            current_sum = np.sum(radii_cand)
            if current_sum > best_sum:
                best_sum = current_sum
                best_var = current_var.copy()

    # Fallback to a deterministic valid packing if optimization fails to find one
    if best_var is None:
        centers_fb = np.zeros((n, 2))
        radii_fb = np.ones(n) * 0.09
        idx = 0
        for r in range(5):
            y = 0.15 + r * 0.16
            ncols = 6 if r % 2 == 0 else 5
            for c in range(ncols):
                if idx >= n: break
                x = 0.15 + c * 0.16 + (0.08 if r % 2 == 1 else 0)
                centers_fb[idx] = [x, y]
                idx += 1
        best_var = np.concatenate([centers_fb.flatten(), radii_fb])
        best_sum = np.sum(radii_fb)

    centers = best_var[:2 * n].reshape(n, 2)
    radii = best_var[2 * n:]
    return centers, radii, float(np.sum(radii))
