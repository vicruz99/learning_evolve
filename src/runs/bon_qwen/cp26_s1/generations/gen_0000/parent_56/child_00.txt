import numpy as np
from scipy.optimize import minimize

def compute_constraints(vars):
    """Compute all inequality constraints: >= 0 means feasible."""
    N = len(vars) // 3
    centers = vars[:2*N].reshape(N, 2)
    radii = vars[2*N:]
    
    cons = []
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    cons.append(centers[:, 0] - radii)
    cons.append(1.0 - centers[:, 0] - radii)
    cons.append(centers[:, 1] - radii)
    cons.append(1.0 - centers[:, 1] - radii)
    
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist_sq = np.sum(diff**2, axis=2)
    r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    overlap = dist_sq - r_sum**2
    mask = np.triu(np.ones((N, N), dtype=bool), k=1)
    cons.append(overlap[mask])
    
    return np.concatenate(cons)

def objective(vars):
    """Objective function: minimize negative sum of radii."""
    N = len(vars) // 3
    return -np.sum(vars[2*N:])

def run_packing():
    N = 26
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    np.random.seed(42)
    
    for trial in range(8):
        # Initialize with a 5x5 grid + 1 central circle, perturbed
        centers = np.zeros((N, 2))
        radii = np.ones(N) * 0.098
        idx = 0
        for i in range(5):
            for j in range(5):
                centers[idx] = [0.1 + j*0.2, 0.1 + i*0.2]
                idx += 1
        centers[idx] = [0.5, 0.5]
        radii[idx] = 0.04
        
        # Add random perturbation to explore configuration space
        centers += np.random.uniform(-0.015, 0.015, (N, 2))
        radii += np.random.uniform(-0.008, 0.008, N)
        radii = np.maximum(radii, 0.01)
        
        x0 = np.hstack([centers.flatten(), radii])
        bounds = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N
        cons = {'type': 'ineq', 'fun': compute_constraints}
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, 
                           constraints=[cons], options={'maxiter': 2000, 'ftol': 1e-9})
            
            if np.isfinite(res.fun):
                cur_sum = -res.fun
                if cur_sum > best_sum:
                    c = res.x[:2*N].reshape(N, 2)
                    r = res.x[2*N:]
                    # Strict feasibility enforcement
                    r = np.maximum(r, 1e-7)
                    c = np.clip(c, r, 1.0 - r)
                    
                    best_sum = cur_sum
                    best_centers = c
                    best_radii = r
        except Exception:
            continue
            
    if best_centers is None:
        # Fallback should not be reached given constraints, but provided for safety
        best_centers = np.zeros((N, 2))
        best_radii = np.ones(N) * 0.01
        best_sum = np.sum(best_radii)
        
    return best_centers, best_radii, best_sum
