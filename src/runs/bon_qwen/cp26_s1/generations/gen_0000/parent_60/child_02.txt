To maximize the sum of radii for 26 circles in a unit square, we utilize a two-stage optimization approach. 

First, for any fixed set of circle centers, the problem of finding the maximum possible radii is a Linear Programming (LP) problem. The objective is to maximize $\sum r_i$ subject to pairwise distance constraints ($r_i + r_j \le d_{ij}$) and boundary constraints ($r_i \le \text{dist\_to\_wall}$). 

Second, we use the sensitivity analysis (dual variables/marginals) from the LP solution to guide the movement of the circle centers. The marginal of a constraint represents how much the objective (sum of radii) improves if the constraint is relaxed. For pairwise constraints, this translates to a repulsive force between circles; for boundary constraints, it translates to a force pushing circles away from the walls.

1. **Initialize** 26 centers in a staggered hexagonal grid pattern, which is a known efficient packing structure.
2. **Iterative Optimization Loop**:
   - Solve the LP to find optimal radii and their associated dual variables (marginals).
   - Calculate the gradient of the sum of radii with respect to the centers using these marginals.
   - Update centers in the direction of the gradient (Gradient Ascent) and project them back into the unit square.

import numpy as np
from scipy.optimize import linprog

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # 1. Initialization: Hexagonal Grid
    # A 5x5 hexagonal grid can fit 26 circles comfortably if optimized.
    # We start with a regular grid and perturb it slightly.
    centers = np.zeros((n, 2))
    row, col = 0, 0
    for i in range(n):
        # Stagger every other row
        offset = 0.5 * (2 / 5.0) if row % 2 == 1 else 0
        x = (col + 0.5) * (2 / 5.0) + offset
        y = (row + 0.5) * (2 / 6.0) # slightly taller spacing for hex
        centers[i] = [np.clip(x, 0.05, 0.95), np.clip(y, 0.05, 0.95)]
        
        # Advance to next grid position
        col += 1
        if col > 4: # 5 items per row
            col = 0
            row += 1

    # 2. Optimization Loop
    # We perform gradient ascent on the centers to maximize sum(radii)
    # using the dual variables from the LP as sensitivity factors.
    
    for _ in range(1000):
        # --- Setup LP ---
        # Variables: r_0 ... r_25 (size 26)
        # Objective: Minimize -sum(r) => c = -1 for all
        c = np.ones(n) * -1.0
        
        # Constraints Matrix A_ub and b_ub
        # r_i + r_j <= d_ij
        # r_i <= dist_to_boundary
        
        num_pairwise = n * (n - 1) // 2
        num_boundary = 4 * n
        num_constraints = num_pairwise + num_boundary
        
        A_ub = np.zeros((num_constraints, n))
        b_ub = np.zeros(num_constraints)
        
        idx = 0
        
        # Pairwise Constraints
        # Store mapping of index -> (i, j) for later gradient calculation
        pair_indices = []
        
        for i in range(n):
            for j in range(i + 1, n):
                d = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                A_ub[idx, i] = 1.0
                A_ub[idx, j] = 1.0
                b_ub[idx] = d
                pair_indices.append((i, j))
                idx += 1
                
        # Boundary Constraints
        # r_i <= x, r_i <= 1-x, r_i <= y, r_i <= 1-y
        for i in range(n):
            x, y = centers[i]
            # 4 constraints for each circle
            A_ub[idx, i] = 1.0; b_ub[idx] = x; idx += 1      # r <= x
            A_ub[idx, i] = 1.0; b_ub[idx] = 1 - x; idx += 1  # r <= 1-x
            A_ub[idx, i] = 1.0; b_ub[idx] = y; idx += 1      # r <= y
            A_ub[idx, i] = 1.0; b_ub[idx] = 1 - y; idx += 1  # r <= 1-y

        # Solve LP
        bounds = [(0, None) for _ in range(n)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        
        if not res.success:
            continue
            
        radii = -res.x # Convert back from minimization
        
        # 3. Gradient Calculation using Dual Variables
        # The marginal of the objective w.r.t b_ub[k] is the sensitivity.
        # For min(-sum r), marginals are negative or positive?
        # linprog minimizes. Increasing RHS (b) relaxes constraint, allowing lower min value (more negative).
        # So marginal should be negative. 
        # However, we want to maximize sum(r).
        # If constraint r_i + r_j <= d_ij is active (marginal != 0), increasing d_ij allows larger r.
        # So we treat -marginal as the gradient for sum(r).
        
        marginals = res.ineqlin.marginals # Shape (num_constraints,)
        
        # Initialize forces
        forces = np.zeros((n, 2))
        
        # Pairwise Forces
        # Marginals for pairwise constraints start at index 0
        for k, (i, j) in enumerate(pair_indices):
            m = -marginals[k] # Sensitivity of max sum(r) w.r.t distance d_ij
            if m > 1e-7: # Only apply force if constraint is active
                d = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if d > 1e-9:
                    vec = (centers[i] - centers[j]) / d
                    forces[i] += m * vec
                    forces[j] -= m * vec
        
        # Boundary Forces
        # Marginals for boundary constraints start after pairwise
        base_idx = num_pairwise
        for i in range(n):
            x, y = centers[i]
            
            # r <= x (Left wall) -> idx base_idx + 4*i
            m_left = -marginals[base_idx + 4*i]
            if m_left > 1e-7:
                forces[i, 0] += m_left # Push right (increase x)
                
            # r <= 1-x (Right wall) -> idx base_idx + 4*i + 1
            m_right = -marginals[base_idx + 4*i + 1]
            if m_right > 1e-7:
                forces[i, 0] -= m_right # Push left (decrease x)
                
            # r <= y (Bottom wall) -> idx base_idx + 4*i + 2
            m_bottom = -marginals[base_idx + 4*i + 2]
            if m_bottom > 1e-7:
                forces[i, 1] += m_bottom # Push up (increase y)
                
            # r <= 1-y (Top wall) -> idx base_idx + 4*i + 3
            m_top = -marginals[base_idx + 4*i + 3]
            if m_top > 1e-7:
                forces[i, 1] -= m_top # Push down (decrease y)

        # 4. Update Centers
        # Use a decaying step size for stability
        step_size = 0.05 * (1.0 - 0.8 * (_ / 1000.0))
        centers += step_size * forces
        
        # Project back to [0, 1]
        np.clip(centers, 0.0, 1.0, out=centers)

    # Final radii calculation
    # Re-run LP to get exact final radii for the final centers
    # (Similar setup as inside loop)
    c = np.ones(n) * -1.0
    num_pairwise = n * (n - 1) // 2
    num_boundary = 4 * n
    num_constraints = num_pairwise + num_boundary
    A_ub = np.zeros((num_constraints, n))
    b_ub = np.zeros(num_constraints)
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            d = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = d
            idx += 1
    for i in range(n):
        x, y = centers[i]
        A_ub[idx, i] = 1.0; b_ub[idx] = x; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1 - x; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = y; idx += 1
        A_ub[idx, i] = 1.0; b_ub[idx] = 1 - y; idx += 1
    
    bounds = [(0, None) for _ in range(n)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    final_radii = -res.x
    total_sum = np.sum(final_radii)
    
    return centers, final_radii, total_sum
