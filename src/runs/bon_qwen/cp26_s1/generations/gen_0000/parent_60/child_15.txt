import numpy as np
from scipy.optimize import minimize

def compute_loss(centers, radii, lam):
    """Compute the penalty-augmented objective function."""
    n = centers.shape[0]
    loss = -np.sum(radii)
    
    # Boundary penalties: ensure circles stay within [0,1]x[0,1]
    loss += lam * np.sum(np.maximum(0, radii - centers[:, 0])**2)
    loss += lam * np.sum(np.maximum(0, centers[:, 0] + radii - 1)**2)
    loss += lam * np.sum(np.maximum(0, radii - centers[:, 1])**2)
    loss += lam * np.sum(np.maximum(0, centers[:, 1] + radii - 1)**2)
    
    # Pairwise non-overlap penalties
    for i in range(n):
        for j in range(i + 1, n):
            dij = np.sqrt(np.sum((centers[i] - centers[j])**2))
            overlap = radii[i] + radii[j] - dij
            if overlap > 0:
                loss += lam * overlap**2
                
    return loss

def run_packing():
    n = 26
    best_centers = np.zeros((n, 2))
    best_radii = np.zeros(n)
    best_sum = 0.0
    
    # Multiple restarts to avoid local minima
    for seed in range(30):
        rng = np.random.default_rng(seed)
        
        # Initialize with a dense grid pattern
        xg = np.linspace(0.05, 0.95, 6)
        yg = np.linspace(0.05, 0.95, 5)
        cx, cy = np.meshgrid(xg, yg)
        grid = np.column_stack([cx.ravel(), cy.ravel()])
        
        c0 = grid[:n].copy() + rng.uniform(-0.02, 0.02, grid[:n].shape)
        c0 = np.clip(c0, 0.02, 0.98)
        r0 = np.full(n, 0.09)
        
        x0 = np.concatenate([c0.ravel(), r0])
        bounds = [(0.0, 1.0)]*(2*n) + [(0.0, 0.5)]*n
        
        # Sequential penalty method
        lam = 200.0
        for _ in range(8):
            def objective(z):
                c = z[:2*n].reshape(n, 2)
                r = z[2*n:]
                return compute_loss(c, r, lam)
                
            res = minimize(objective, x0, method='L-BFGS-B', bounds=bounds, 
                           options={'maxiter': 400, 'ftol': 1e-12, 'gtol': 1e-12})
            x0 = res.x
            lam *= 2.5
            
        c_opt = x0[:2*n].reshape(n, 2)
        r_opt = x0[2*n:]
        
        # Strict validation check
        valid = True
        if np.any(r_opt < 0) or np.any(c_opt[:, 0] < r_opt) or np.any(c_opt[:, 0] > 1 - r_opt) or \
           np.any(c_opt[:, 1] < r_opt) or np.any(c_opt[:, 1] > 1 - r_opt):
            valid = False
        if valid:

                for j in range(i+1, n):
                    d = np.sqrt(np.sum((c_opt[i] - c_opt[j])**2))
                    if d < r_opt[i] + r_opt[j] - 1e-11:
                        valid = False
                        break
                if not valid: break
                
        if valid:
            s = np.sum(r_opt)
            if s > best_sum:
                best_sum = s
                best_centers = c_opt.copy()
                best_radii = r_opt.copy()
        else:
            # Fallback: scale radii down until strictly valid
            lo, hi = 0.5, 1.0
            for _ in range(20):
                mid = (lo + hi) / 2
                r_test = r_opt * mid
                v = True
                if np.any(c_opt[:, 0] < r_test) or np.any(c_opt[:, 0] > 1 - r_test) or \
                   np.any(c_opt[:, 1] < r_test) or np.any(c_opt[:, 1] > 1 - r_test):
                    v = False
                else:
                    for i in range(n):
                        for j in range(i+1, n):

                        best_centers = c_opt.copy()

    return best_centers, best_radii, best_sum
