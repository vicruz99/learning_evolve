    if not validate_pack(best_centers, best_radii):
        scale = 0.9999
        while not validate_pack(best_centers, best_radii * scale) and scale > 0.1:
            scale *= 0.9999
        best_radii *= scale
        best_sum = np.sum(best_radii)
    
    return (best_centers, best_radii, float(best_sum))
