   - **Non-overlap constraints:** The distance between any two centers must be at least the sum of their radii.

import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None

    def objective(params):
        # We want to maximize sum of radii, so we minimize the negative sum
        return -np.sum(params[2::3])

    def constraints(params):
        constraints_list = []
        for i in range(n):
            x = params[i * 3]
            y = params[i * 3 + 1]
            r = params[i * 3 + 2]
            # Boundary constraints
            constraints_list.append({'type': 'ineq', 'fun': lambda p, idx=i: p[idx*3+2] - p[idx*3]})      # r - x <= 0 => x >= r
            constraints_list.append({'type': 'ineq', 'fun': lambda p, idx=i: 1 - p[idx*3+2] - p[idx*3]})  # 1 - r - x >= 0 => x <= 1-r
            constraints_list.append({'type': 'ineq', 'fun': lambda p, idx=i: p[idx*3+2] - p[idx*3+1]})    # r - y <= 0 => y >= r
            constraints_list.append({'type': 'ineq', 'fun': lambda p, idx=i: 1 - p[idx*3+2] - p[idx*3+1]}) # 1 - r - y >= 0 => y <= 1-r

            for j in range(i + 1, n):
                # Overlap constraint: dist >= r_i + r_j
                dist_sq = (p[j*3] - p[i*3])**2 + (p[j*3+1] - p[i*3+1])**2
                constraints_list.append({
                    'type': 'ineq',
                    'fun': lambda p, i=i, j=j: dist_sq - (p[i*3+2] + p[j*3+2])**2
                })
        return constraints_list

    def get_bounds():
        bounds = []
        for _ in range(n):
            bounds.append((0, 1)) # x
            bounds.append((0, 1)) # y
            bounds.append((0, 0.5)) # r
        return bounds

    def generate_hex_init():
        params = np.zeros(3 * n)
        r = 0.09
        d = 2 * r
        dy = d * np.sqrt(3) / 2
        row_idx = 0
        col_idx = 0
        
        for i in range(n):
            # Hexagonal pattern: shift every odd row
            x_offset = (row_idx % 2) * (d / 2)
            x = r + col_idx * d + x_offset
            y = r + row_idx * dy
            
            params[i * 3] = x
            params[i * 3 + 1] = y
            params[i * 3 + 2] = r
            
            col_idx += 1
            if col_idx > 4: # 5 columns per row
                col_idx = 0
                row_idx += 1
        return params

    def generate_random_init():
        params = np.zeros(3 * n)
        # Place small circles randomly to ensure valid start
        for i in range(n):
            r = 0.02
            params[i * 3] = np.random.uniform(r, 1 - r)
            params[i * 3 + 1] = np.random.uniform(r, 1 - r)
            params[i * 3 + 2] = r
        return params

    bounds = get_bounds()
    
    # Run optimization from multiple starting points
    starts = [generate_hex_init()]
    for _ in range(3):
        starts.append(generate_random_init())

    for x0 in starts:
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, 
                           constraints=constraints(x0), options={'maxiter': 1000, 'ftol': 1e-8})
            
            if res.success and -res.fun > best_sum:
                # Verify constraints strictly
                centers = res.x.reshape(n, 3)[:, :2]
                radii = res.x.reshape(n, 3)[:, 2]
                
                # Quick validation check
                is_valid = True
                for i in range(n):
                    if radii[i] < 1e-9 or radii[i] > 0.5:
                        is_valid = False; break
                    if centers[i, 0] < radii[i] - 1e-9 or centers[i, 0] > 1 - radii[i] + 1e-9:
                        is_valid = False; break
                    if centers[i, 1] < radii[i] - 1e-9 or centers[i, 1] > 1 - radii[i] + 1e-9:
                        is_valid = False; break
                
                if is_valid:
                    best_sum = -res.fun
                    best_centers = centers
                    best_radii = radii
        except Exception:
            continue

    # Fallback to hexagonal if optimization fails to find a better solution
    if best_centers is None or best_sum < 2.0:
        init_params = generate_hex_init()
        best_centers = init_params.reshape(n, 3)[:, :2]
        best_radii = init_params.reshape(n, 3)[:, 2]
        best_sum = np.sum(best_radii)

    return best_centers, best_radii, float(best_sum)

if __name__ == "__main__":
    centers, radii, total = run_packing()
    print(f"Sum of radii: {total:.5f}")
