import numpy as np
from scipy.optimize import minimize

def run_packing():
    """
    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    """
    n = 26
    num_vars = 3 * n  # x, y, r for each circle
    
    # Objective: Minimize negative sum of radii
    def objective(v):
        r = v[2*n:]
        return -np.sum(r)

    # Vectorized Constraint Function
    def constraint_func(v):
        x = v[:n]
        y = v[n:2*n]
        r = v[2*n:]
        
        # Boundary constraints: 4 per circle
        # x - r >= 0
        c_x_left = x - r
        # 1 - x - r >= 0
        c_x_right = 1.0 - x - r
        # y - r >= 0
        c_y_bottom = y - r
        # 1 - y - r >= 0
        c_y_top = 1.0 - y - r
        
        # Overlap constraints: (x_i - x_j)^2 + (y_i - y_j)^2 >= (r_i + r_j)^2
        # Compute pairwise squared distances
        # x_diff matrix: shape (n, n)
        x_diff = x[:, np.newaxis] - x[np.newaxis, :]
        y_diff = y[:, np.newaxis] - y[np.newaxis, :]
        dist_sq = x_diff**2 + y_diff**2
        
        # Sum of radii squared matrix
        r_sum = r[:, np.newaxis] + r[np.newaxis, :]
        r_sum_sq = r_sum**2
        
        # Constraint value for all pairs
        overlap_constraints = dist_sq - r_sum_sq
        
        # We only need upper triangle (i < j) to avoid duplicates and self-interaction
        # np.triu_indices(n, k=1) gives indices for i < j
        indices = np.triu_indices(n, k=1)
        upper_tri_constraints = overlap_constraints[indices]
        
        # Combine all constraints
        return np.concatenate([
            c_x_left, c_x_right, c_y_bottom, c_y_top,
            upper_tri_constraints
        ])

    # Define constraint object for SLSQP

    constraints = [{'type': 'ineq', 'fun': constraint_func}]

                        min_dist = d
                if min_dist > 0.15: # Heuristic
                    grid_coords.append((px, py))
                    break
            else:
                # Fallback random
                grid_coords.append((np.random.uniform(0.1, 0.9), np.random.uniform(0.1, 0.9)))
        
        return grid_coords[:n_circles]

                method='SLSQP',
