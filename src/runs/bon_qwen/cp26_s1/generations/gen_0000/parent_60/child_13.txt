from scipy.optimize import minimize

def objective(z):
    """Objective function: maximize sum of radii -> minimize negative sum."""
    n = 26
    return -np.sum(z[2*n:])

def constraints(z):
    """Inequality constraints: boundary and non-overlap."""
    n = 26
    cx = z[:n]
    cy = z[n:2*n]
    r = z[2*n:]
    c = np.column_stack((cx, cy))

    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    bnd = np.zeros(4 * n)
    bnd[0::4] = cx - r
    bnd[1::4] = 1.0 - cx - r
    bnd[2::4] = cy - r
    bnd[3::4] = 1.0 - cy - r

    # Non-overlap constraints: dist^2 >= (r_i + r_j)^2
    diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
    dist_sq = np.sum(diff**2, axis=2)
    rad_sum = r[:, np.newaxis] + r[np.newaxis, :]

    i_idx, j_idx = np.triu_indices(n, k=1)
    overlap = dist_sq[i_idx, j_idx] - rad_sum[i_idx, j_idx]**2

    return np.concatenate([bnd, overlap])

def run_packing():

    best_obj = np.inf
    best_x = None
