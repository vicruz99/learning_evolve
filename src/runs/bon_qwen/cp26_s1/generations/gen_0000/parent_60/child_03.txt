def objective_and_constraints(vars_flat, n):

    Objective: -sum(radii)
    Constraints:
    1. x_i - r_i >= 0
    2. 1 - x_i - r_i >= 0
    3. y_i - r_i >= 0
    4. 1 - y_i - r_i >= 0
    5. dist(i,j) - (r_i + r_j) >= 0
    """

        cons.append({'type': 'ineq', 'fun': lambda v, i=i: v[2*i+1] - v[2*n+i]}) # y - r
        cons.append({'type': 'ineq', 'fun': lambda v, i=i: 1.0 - v[2*i+1] - v[2*n+i]}) # 1 - y - r
        
    # Overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            def dist_constraint(v, i=i, j=j):

                return d - (ri + rj)
            cons.append({'type': 'ineq', 'fun': dist_constraint})
            
    return obj, cons
