import numpy as np
from scipy.optimize import linprog, differential_evolution

def get_pairwise_constraints(n):
    """Precompute the sparse structure of the pairwise distance constraints."""
    num_pairs = n * (n - 1) // 2
    A_pairs = np.zeros((num_pairs, n))
    row_idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_pairs[row_idx, i] = 1.0
            A_pairs[row_idx, j] = 1.0
            row_idx += 1
    return A_pairs

def solve_radii(centers, n, A_pairs):
    """Solve the LP to maximize sum of radii for a given set of centers."""
    # 1. Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    # We use the minimum of the 4 distances to the walls
    boundaries = np.minimum.reduce([centers[:, 0], 1 - centers[:, 0], 
                                    centers[:, 1], 1 - centers[:, 1]])
    
    # 2. Pairwise distance constraints: r_i + r_j <= dist(c_i, c_j)
    # Calculate Euclidean distances efficiently
    # (centers[:, np.newaxis, :] - centers[np.newaxis, :, :]) -> (n, n, 2)
    # Squared distances -> sqrt
    dist_matrix = np.linalg.norm(centers[:, np.newaxis, :] - centers[np.newaxis, :, :], axis=2)
    
    # Extract upper triangle distances for r_i + r_j <= d_ij (i < j)
    dists = dist_matrix[np.triu_indices(n, k=1)]
    
    # Construct Linear Programming problem
    # Maximize sum(r) -> Minimize -sum(r)
    c_obj = -np.ones(n)
    
    # Constraints: A_ub @ r <= b_ub
    # A_ub is a vertical stack of Identity (for boundaries) and A_pairs (for distances)
    # We preallocate the full matrix for performance, but for n=26 it's small enough to just stack
    A_ub = np.vstack([np.eye(n), A_pairs])
    b_ub = np.concatenate([boundaries, dists])
    
    # Bounds: 0 <= r_i
    bounds = [(0, None)] * n
    
    # Solve using HiGHS solver (fast and robust)
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return -res.fun, -res.x
    else:
        # Fallback if LP fails (unlikely)
        return 0.0, np.zeros(n)

def objective(centers_flat, n, A_pairs):
    """Objective function for optimization: negative sum of radii."""
    centers = centers_flat.reshape(n, 2)
    # Clip centers slightly to ensure they are strictly inside [0, 1] for numerical stability
    centers = np.clip(centers, 1e-6, 1.0 - 1e-6)
    
    sum_radii, radii = solve_radii(centers, n, A_pairs)
    return -sum_radii

def generate_hexagonal_lattice(n, padding=0.05):
    """Generate initial centers based on a hexagonal lattice."""
    points = []
    # Adjust density to fit n points
    # A hexagonal grid with spacing s has approx 2/(sqrt(3)*s^2) points per unit area
    # We try a few rows/cols
    for y in np.linspace(padding, 1 - padding, 6):
        for x in np.linspace(padding, 1 - padding, 5):
            if len(points) < n:
                points.append([x, y])
    
    # If we need more, add random or grid points
    if len(points) < n:
        # Fill remaining with random spread
        rng = np.random.default_rng(42)
        while len(points) < n:
            p = rng.uniform(padding, 1 - padding, 2)
            points.append(p)
            
    return np.array(points[:n])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    A_pairs = get_pairwise_constraints(n)
    
    # Initial guess: Hexagonal lattice
    initial_guess = generate_hexagonal_lattice(n)
    
    # Optimization bounds for centers: [0, 1] x [0, 1]
    bounds = [(0.0, 1.0)] * (n * 2)
    
    # Run Differential Evolution
    # seed for reproducibility, maxiter high enough for convergence
    result = differential_evolution(
        objective, 
        bounds, 
        args=(n, A_pairs),
        x0=initial_guess.flatten(),
        maxiter=200,
        popsize=20,
        mutation=(0.5, 1.5),
        recombination=0.9,
        seed=42,
        tol=1e-6
    )
    
    # Extract best centers
    best_centers = result.x.reshape(n, 2)
    
    # Solve final LP to get precise radii for the optimized centers
    # Ensure centers are valid before final solve
    best_centers = np.clip(best_centers, 1e-7, 1.0 - 1e-7)
    
    sum_radii, radii = solve_radii(best_centers, n, A_pairs)
    
    return best_centers, radii, float(sum_radii)
