import numpy as np
from scipy.optimize import minimize

def generate_initial_config(n, seed):
    rng = np.random.RandomState(seed)
    # Hexagonal-ish lattice initialization
    cols = int(np.ceil(np.sqrt(n)))
    rows = int(np.ceil(n / cols))
    
    xs = np.zeros(n)
    ys = np.zeros(n)
    idx = 0
    for r in range(rows):
        offset = (r % 2) * 0.5
        for c in range(cols):
            if idx < n:
                xs[idx] = (c + offset) / (cols + 0.5)
                ys[idx] = r / (rows + 0.5)
                idx += 1
                
    # Add controlled noise to break symmetry
    xs += rng.uniform(-0.08, 0.08, n)
    ys += rng.uniform(-0.08, 0.08, n)
    
    # Clip to feasible region roughly
    xs = np.clip(xs, 0.1, 0.9)
    ys = np.clip(ys, 0.1, 0.9)
    
    # Initial radius slightly less than theoretical max for 25 circles
    rs = np.full(n, 0.075)
    return np.concatenate([xs, ys, rs])

def wall_constraints(v, n):
    res = np.empty(4 * n)
    for i in range(n):
        x, y, r = v[i], v[n + i], v[2 * n + i]
        res[4 * i] = x - r
        res[4 * i + 1] = 1 - x - r
        res[4 * i + 2] = y - r
        res[4 * i + 3] = 1 - y - r
    return res

def overlap_constraints(v, n):
    num_pairs = n * (n - 1) // 2
    res = np.empty(num_pairs)
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(v[i] - v[j], v[n + i] - v[n + j])
            res[idx] = dist - (v[2 * n + i] + v[2 * n + j])
            idx += 1
    return res

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    best_sum = 0.0
    best_x0 = None
    best_res = None
    
    # Run multiple restarts to avoid local minima
    for seed in range(50):
        x0 = generate_initial_config(n, seed)
        
        bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)] * n + [(0.0, 0.5)] * n
        
        constraints = [
            {'type': 'ineq', 'fun': wall_constraints, 'args': (n,)},
            {'type': 'ineq', 'fun': overlap_constraints, 'args': (n,)}
        ]
        
        try:
            res = minimize(
                fun=lambda v: -np.sum(v[2*n:]),
                x0=x0,
                bounds=bounds,
                constraints=constraints,
                method='SLSQP',
                options={'maxiter': 3000, 'ftol': 1e-10, 'disp': False}
            )
            
            current_sum = -res.fun
            # Verify constraints are satisfied (accounting for numerical tolerance)
            if np.all(wall_constraints(res.x, n) >= -1e-7) and \
               np.all(overlap_constraints(res.x, n) >= -1e-7):
                if current_sum > best_sum:
                    best_sum = current_sum
                    best_res = res
        except Exception:
            continue
            
    if best_res is None:
        # Fallback to simple grid if optimization fails
        xs, ys, rs = generate_initial_config(n, 0)
        return np.column_stack((xs, ys)), rs, np.sum(rs)
        
    # Extract results
    centers = np.column_stack((best_res.x[:n], best_res.x[n:2*n]))
    radii = best_res.x[2*n:]
    
    # Slight shrinkage to guarantee strict feasibility under validation tolerance
    # This handles any residual numerical slack near boundaries
    min_margin = np.min([
        np.min(centers[:,0] - radii),
        np.min(1 - centers[:,0] - radii),
        np.min(centers[:,1] - radii),
        np.min(1 - centers[:,1] - radii)
    ])
    if min_margin < 1e-8:
        radii -= 1e-9
        radii = np.maximum(radii, 0.0)
        
    return centers, radii, np.sum(radii)
