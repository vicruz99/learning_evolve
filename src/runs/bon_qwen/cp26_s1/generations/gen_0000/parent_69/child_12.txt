import numpy as np
from scipy.optimize import minimize, linprog

def get_sum_radii(centers):
    """
    Given centers, solve the LP to find max sum of radii.
    centers: np.array of shape (26, 2)
    Returns: (sum_radii, radii_array)
    """
    n = 26
    c = -np.ones(n)  # Maximize sum r_i <=> Minimize -sum r_i
    
    # Constraints A_ub r <= b_ub
    # 1. Pairwise: r_i + r_j <= dist(i, j)
    # 2. Boundaries: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    
    # Prepare matrices
    # Number of constraints: n*(n-1)/2 + 4*n
    n_pairs = n * (n - 1) // 2
    n_bound = 4 * n
    n_constraints = n_pairs + n_bound
    
    A_ub = np.zeros((n_constraints, n))
    b_ub = np.zeros(n_constraints)
    
    idx = 0
    
    # Pairwise constraints
    # Optimization: compute distances and fill A_ub
    # A_ub row for (i, j) has 1 at i and 1 at j
    # We can construct this efficiently
    
    # Precompute distances
    # centers shape (n, 2)
    # diff: (n, n, 2)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1
            
    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        
        # r_i <= x
        A_ub[idx, i] = 1.0
        b_ub[idx] = x
        idx += 1
        
        # r_i <= 1 - x
        A_ub[idx, i] = 1.0
        b_ub[idx] = 1.0 - x
        idx += 1
        
        # r_i <= y
        A_ub[idx, i] = 1.0
        b_ub[idx] = y
        idx += 1
        
        # r_i <= 1 - y
        A_ub[idx, i] = 1.0
        b_ub[idx] = 1.0 - y
        idx += 1
        
    # Bounds for r: r >= 0
    bounds_r = [(0, None) for _ in range(n)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method='highs')
        if res.success:
            return -res.fun, res.x
        else:
            # Infeasible (should not happen if centers in [0,1])
            return -1.0, None
    except Exception:
        return -1.0, None

def objective(centers_flat):
    """
    Objective function for optimizer.
    Minimize -sum_radii.
    """
    centers = centers_flat.reshape(26, 2)
    # Clip centers to [0, 1] just in case, though bounds should handle it.
    # Clipping helps if numerical error pushes slightly out.
    centers = np.clip(centers, 0.0, 1.0)
    
    s, _ = get_sum_radii(centers)
    if s < 0:
        return 1e6 # Penalty for infeasible
    return -s
