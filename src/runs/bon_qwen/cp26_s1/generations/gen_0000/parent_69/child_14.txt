The radii variables appear linearly, and all constraints are linear in $r$.

import numpy as np
from scipy.optimize import linprog, minimize

def compute_packing(centers):
    """Solve LP to maximize sum of radii for fixed centers."""
    n = centers.shape[0]
    c = -np.ones(n)  # Maximize sum => minimize -sum
    
    # Constraints: r_i + r_j <= dist_ij, r_i <= boundary, r_i >= 0
    m = n * (n - 1) // 2 + 4 * n
    A = np.zeros((m, n))
    b = np.zeros(m)
    
    idx = 0
    # Pairwise distance constraints
    for i in range(n):
        for j in range(i + 1, n):
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            idx += 1
            
    # Boundary constraints
    for i in range(n):
        row = np.zeros(n)
        row[i] = 1.0
        A[idx] = row; b[idx] = centers[i, 0]; idx += 1
        A[idx] = row; b[idx] = 1.0 - centers[i, 0]; idx += 1
        A[idx] = row; b[idx] = centers[i, 1]; idx += 1
        A[idx] = row; b[idx] = 1.0 - centers[i, 1]; idx += 1
        
    bounds = [(0, None) for _ in range(n)]
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    
    if res.success:
        return res.x, -res.fun
    return np.zeros(n), 0.0

def objective(centers_flat):
    centers = centers_flat.reshape(26, 2)
    _, s = compute_packing(centers)
    return -s  # Minimize negative sum

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    best_val = 0.0
    best_centers = np.random.rand(26, 2)
    best_radii = np.zeros(26)
    
    n_circles = 26
    dims = 52
    bounds = [(0.0, 1.0) for _ in range(dims)]
    
    # Generate diverse initial configurations
    x0_candidates = []
    
    # 1. Random starts
    for _ in range(10):
        x0_candidates.append(np.random.rand(dims))
        
    # 2. Perturbed grid
    grid_x = np.linspace(0.05, 0.95, 6)
    grid_y = np.linspace(0.05, 0.95, 5)
    gx, gy = np.meshgrid(grid_x, grid_y)
    pts = np.vstack([gx.flatten(), gy.flatten()]).T
    idx = np.random.choice(pts.shape[0], 26, replace=False)
    x0_candidates.append((pts[idx] + np.random.randn(26, 2) * 0.03).flatten())
    
    # 3. Perturbed hexagonal lattice
    hex_pts = []
    y = 0.08
    while len(hex_pts) < 30:
        x = 0.05
        while x < 0.95:
            hex_pts.append([x, y])
            x += 0.28 * np.sqrt(3)
        y += 0.28 * 1.5
    hex_arr = np.array(hex_pts[:26]) + np.random.randn(26, 2) * 0.03
    x0_candidates.append(np.clip(hex_arr, 0.05, 0.95).flatten())
    
    # Optimize from each starting point
    for x0 in x0_candidates:
        # Powell handles the non-smooth objective landscape robustly
        res_p = minimize(objective, x0, method='Powell', 
                         options={'maxiter': 150, 'ftol': 1e-10})
        # Refine with gradient-based method for precision
        res_f = minimize(objective, res_p.x, method='L-BFGS-B', 
                         bounds=bounds, options={'maxiter': 300, 'ftol': 1e-10})
        
        curr_val = -res_f.fun
        if curr_val > best_val:
            best_val = curr_val
            best_centers = res_f.x.reshape(n_circles, 2)
            best_radii, _ = compute_packing(best_centers)
            
    # Strictly enforce interior bounds to pass validation tolerances safely
    best_centers = np.clip(best_centers, 1e-8, 1.0 - 1e-8)
    best_radii, best_val = compute_packing(best_centers)
    
    return best_centers, best_radii, best_val
