import numpy as np
from scipy.optimize import minimize

# Global constant for number of circles
N_CIRCLES = 26

def run_packing():
    """
    Main function to generate the circle packing.
    Returns centers, radii, and sum of radii.
    """
    # 1. Generate Initial Guess (Hexagonal Packing)
    # Layout: 5 rows. Pattern 6-4-6-4-6 circles.
    # This fits 26 circles.
    # We start with a slightly smaller radius to ensure strict validity, 
    # then let the optimizer expand them.
    r_init = 0.09
    
    centers_init = []
    y = r_init
    
    # Row 0: 6 circles
    x_start = r_init
    x_step = 2 * r_init
    row_0_x = [x_start + k * x_step for k in range(6)]
    centers_init.extend([(x, y) for x in row_0_x])
    
    # Row 1: 4 circles (shifted)
    y += r_init * np.sqrt(3)
    x_start = r_init + r_init # Shift by radius
    # We need 4 circles. 
    # With r=0.09, step=0.18. 
    # 0.18, 0.36, 0.54, 0.72. Max center 0.72 + 0.09 = 0.81 <= 1. OK.
    row_1_x = [x_start + k * x_step for k in range(4)]
    centers_init.extend([(x, y) for x in row_1_x])
    
    # Row 2: 6 circles
    y += r_init * np.sqrt(3)
    x_start = r_init
    row_2_x = [x_start + k * x_step for k in range(6)]
    centers_init.extend([(x, y) for x in row_2_x])
    
    # Row 3: 4 circles
    y += r_init * np.sqrt(3)
    x_start = r_init + r_init
    row_3_x = [x_start + k * x_step for k in range(4)]
    centers_init.extend([(x, y) for x in row_3_x])
    
    # Row 4: 6 circles
    y += r_init * np.sqrt(3)
    x_start = r_init
    row_4_x = [x_start + k * x_step for k in range(6)]
    centers_init.extend([(x, y) for x in row_4_x])
    
    centers_init = np.array(centers_init)
    radii_init = np.full(N_CIRCLES, r_init)
    
    # 2. Prepare for Optimization
    # Parameters vector: [x1, y1, r1, x2, y2, r2, ...]
    # Size: 3 * 26 = 78
    x0 = np.zeros(3 * N_CIRCLES)
    for i in range(N_CIRCLES):
        x0[3*i] = centers_init[i, 0]
        x0[3*i+1] = centers_init[i, 1]
        x0[3*i+2] = radii_init[i]
        
    # Bounds: x in [0,1], y in [0,1], r >= 0
    # Ideally x, y should be in [r, 1-r], but bounds are simple [0,1]
    # and constraints enforce the rest.
    bounds = [(0, 1)] * (2 * N_CIRCLES) + [(0, 0.5)] * N_CIRCLES
    
    # 3. Run Optimization
    # We use SLSQP which supports inequality constraints
    result = minimize(
        fun=objective_function,
        x0=x0,
        args=(N_CIRCLES,),
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'ineq', 'fun': constraint_function, 'args': (N_CIRCLES,)},
        options={'maxiter': 1000, 'ftol': 1e-12}
    )
    
    # 4. Extract Results
    if result.success:
        final_params = result.x
    else:
        # Fallback to initial if optimization fails (unlikely)
        final_params = x0
        
    final_centers = np.zeros((N_CIRCLES, 2))
    final_radii = np.zeros(N_CIRCLES)
    
    for i in range(N_CIRCLES):
        final_centers[i, 0] = final_params[3*i]
        final_centers[i, 1] = final_params[3*i+1]
        final_radii[i] = final_params[3*i+2]
        
    sum_radii = np.sum(final_radii)
    
    return final_centers, final_radii, sum_radii

def objective_function(params, n):
    """
    Objective: Maximize sum of radii.
    We minimize negative sum.
    """
    radii = params[2::3]
    return -np.sum(radii)

def constraint_function(params, n):
    """
    Returns a vector of constraint values. All must be >= 0.
    Constraints:
    1. Boundary: x >= r, x <= 1-r, y >= r, y <= 1-r
    2. Separation: dist^2 >= (r_i + r_j)^2
    """
    constraints = []
    
    # Extract arrays
    xs = params[0::3]
    ys = params[1::3]
    rs = params[2::3]
    
    # Boundary Constraints
    # x - r >= 0
    constraints.extend(xs - rs)
    # 1 - r - x >= 0  => 1 - (x + r) >= 0
    constraints.extend(1.0 - (xs + rs))
    # y - r >= 0
    constraints.extend(ys - rs)
    # 1 - r - y >= 0
    constraints.extend(1.0 - (ys + rs))
    
    # Separation Constraints
    # (xi - xj)^2 + (yi - yj)^2 - (ri + rj)^2 >= 0
    # Iterate over pairs
    for i in range(n):
        xi, yi, ri = xs[i], ys[i], rs[i]
        for j in range(i + 1, n):
            xj, yj, rj = xs[j], ys[j], rs[j]
            
            dx = xi - xj
            dy = yi - yj
            dist_sq = dx*dx + dy*dy
            
            r_sum = ri + rj
            r_sum_sq = r_sum * r_sum
            
            constraints.append(dist_sq - r_sum_sq)
            
    return np.array(constraints)
