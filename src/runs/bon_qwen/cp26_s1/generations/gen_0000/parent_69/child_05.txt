1. **LP-based radius optimization**: Given fixed centers, the problem of finding optimal radii is a Linear Program (maximize ∑r_i subject to r_i + r_j ≤ ‖c_i - c_j‖ and r_i ≤ boundary distance). This gives the exact optimal radii for any configuration.

2. **Center optimization**: Use a force-directed approach combined with coordinate descent to find centers that allow larger radii. Circles are pushed apart when too close, and pulled toward boundaries to use corner/edge space efficiently.

   - Corner-biased initialization (large circles in corners)
   - Random restarts with simulated annealing

4. **Key insight**: For maximizing SUM of radii (not equal radii), unequal circles can outperform equal ones. Circles near corners can be larger, and smaller circles fill gaps efficiently.

import numpy as np
from scipy.optimize import linprog


def solve_radii_lp(centers, n):
    """Solve LP to find optimal radii given centers."""
    c_obj = -np.ones(n)
    
    A_ub_list = []
    b_ub_list = []
    
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub_list.append(row)
            b_ub_list.append(dist)
    
    for i in range(n):
        b = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
        row = np.zeros(n)
        row[i] = 1.0
        A_ub_list.append(row)
        b_ub_list.append(b)
    
    A_ub = np.array(A_ub_list)
    b_ub = np.array(b_ub_list)
    
    bounds = [(1e-12, None)] * n
    
    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            return result.x
    except Exception:
        pass
    
    return np.ones(n) * 0.02


def force_optimize_centers(centers, n, n_iter=300):
    """Optimize centers using force-directed method."""
    centers = centers.copy()
    
    for it in range(n_iter):
        radii = solve_radii_lp(centers, n)
        forces = np.zeros_like(centers)
        
        # Repulsive forces between circles
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist_sq = dx * dx + dy * dy
                dist = np.sqrt(dist_sq)
                min_dist = radii[i] + radii[j]
                
                if dist < min_dist and dist > 1e-10:
                    overlap = (min_dist - dist) / dist
                    fx = dx * overlap
                    fy = dy * overlap
                    forces[i, 0] += fx
                    forces[i, 1] += fy
                    forces[j, 0] -= fx
                    forces[j, 1] -= fy
        
        # Boundary attraction forces (encourage circles to touch boundaries)
        for i in range(n):
            r = radii[i]
            if centers[i, 0] < r + 0.02:
                forces[i, 0] += 0.5
            elif centers[i, 0] > 1 - r - 0.02:
                forces[i, 0] -= 0.5
            if centers[i, 1] < r + 0.02:
                forces[i, 1] += 0.5
            elif centers[i, 1] > 1 - r - 0.02:
                forces[i, 1] -= 0.5
        
        step = 0.008 * (1.0 - it / n_iter)
        centers += forces * step
        centers[:, 0] = np.clip(centers[:, 0], 1e-6, 1 - 1e-6)
        centers[:, 1] = np.clip(centers[:, 1], 1e-6, 1 - 1e-6)
    
    radii = solve_radii_lp(centers, n)
    return centers, radii


def init_hexagonal(n, spacing, offset_x, offset_y, perturb=0.0):
    """Initialize centers in hexagonal pattern."""
    centers = np.zeros((n, 2))
    idx = 0
    
    for row in range(7):
        y = offset_y + row * spacing * np.sqrt(3) / 2.0
        nx_in_row = 5 if row % 2 == 0 else 4
        x_start = offset_x + (row % 2) * spacing / 2.0
        
        for col in range(nx_in_row):
            if idx >= n:
                break
            x = x_start + col * spacing
            centers[idx, 0] = x + (np.random.rand() - 0.5) * perturb
            centers[idx, 1] = y + (np.random.rand() - 0.5) * perturb
            idx += 1
    
    centers[:, 0] = np.clip(centers[:, 0], 0.05, 0.95)
    centers[:, 1] = np.clip(centers[:, 1], 0.05, 0.95)
    return centers[:n]


def init_corner_biased(n, perturb=0.02):
    """Initialize with large circles in corners, filling rest."""
    centers = np.zeros((n, 2))
    
    # 4 corner circles
    corner_r = 0.22
    centers[0] = [corner_r + 0.02, corner_r + 0.02]
    centers[1] = [1 - corner_r - 0.02, corner_r + 0.02]
    centers[2] = [corner_r + 0.02, 1 - corner_r - 0.02]
    centers[3] = [1 - corner_r - 0.02, 1 - corner_r - 0.02]
    
    # Edge circles
    edge_r = 0.12
    edge_pts = [
        [0.25, edge_r + 0.02], [0.5, edge_r + 0.02], [0.75, edge_r + 0.02],
        [0.25, 1 - edge_r - 0.02], [0.5, 1 - edge_r - 0.02], [0.75, 1 - edge_r - 0.02],
        [edge_r + 0.02, 0.25], [edge_r + 0.02, 0.5], [edge_r + 0.02, 0.75],
        [1 - edge_r - 0.02, 0.25], [1 - edge_r - 0.02, 0.5], [1 - edge_r - 0.02, 0.75],
    ]
    for i in range(4, 16):
        centers[i] = edge_pts[i - 4]
    
    # Middle circles
    mid_idx = 16
    spacing = 0.12
    for row in range(4):
        y = 0.35 + row * spacing * np.sqrt(3) / 2
        nx = 3 if row % 2 == 0 else 2
        x_start = 0.35 + (row % 2) * spacing / 2
        for col in range(nx):
            if mid_idx < n:
                centers[mid_idx] = [x_start + col * spacing, y]
                mid_idx += 1
    
    if mid_idx < n:
        centers[mid_idx] = [0.5, 0.5]
    
    centers += (np.random.rand(n, 2) - 0.5) * perturb
    centers[:, 0] = np.clip(centers[:, 0], 0.05, 0.95)
    centers[:, 1] = np.clip(centers[:, 1], 0.05, 0.95)
    return centers[:n]


def init_random_grid(n):
    """Random initialization with some structure."""
    centers = np.zeros((n, 2))
    
    # Create a rough grid
    rows = 5
    cols = 5
    spacing_x = 0.8 / (cols - 1)
    spacing_y = 0.8 / (rows - 1)
    
    idx = 0
    for r in range(rows):
        for c in range(cols):
            if idx >= n:
                break
            x = 0.1 + c * spacing_x + (r % 2) * spacing_x / 2
            y = 0.1 + r * spacing_y
            centers[idx] = [x + np.random.rand() * 0.02, y + np.random.rand() * 0.02]
            idx += 1
    
    centers[:, 0] = np.clip(centers[:, 0], 0.05, 0.95)
    centers[:, 1] = np.clip(centers[:, 1], 0.05, 0.95)
    return centers[:n]


def coordinate_descent(centers, n, n_iter=200):
    """Coordinate descent: try moving each circle to improve sum of radii."""
    centers = centers.copy()
    
    for it in range(n_iter):
        radii = solve_radii_lp(centers, n)
        current_sum = np.sum(radii)
        step = 0.02 * (1.0 - it / n_iter)
        
        for i in range(n):
            best_move = None
            best_sum = current_sum
            
            # Try moving in 8 directions + random
            for angle in np.linspace(0, 2 * np.pi, 9):
                dx = step * np.cos(angle)
                dy = step * np.sin(angle)
                
                new_c = centers[i].copy()
                new_c[0] = np.clip(centers[i, 0] + dx, 0.01, 0.99)
                new_c[1] = np.clip(centers[i, 1] + dy, 0.01, 0.99)
                
                temp_centers = centers.copy()
                temp_centers[i] = new_c
                
                temp_radii = solve_radii_lp(temp_centers, n)
                temp_sum = np.sum(temp_radii)
                
                if temp_sum > best_sum:
                    best_sum = temp_sum
                    best_move = new_c
            
            # Try random moves
            for _ in range(3):
                dx = np.random.uniform(-step, step)
                dy = np.random.uniform(-step, step)
                new_c = centers[i].copy()
                new_c[0] = np.clip(centers[i, 0] + dx, 0.01, 0.99)
                new_c[1] = np.clip(centers[i, 1] + dy, 0.01, 0.99)
                
                temp_centers = centers.copy()
                temp_centers[i] = new_c
                
                temp_radii = solve_radii_lp(temp_centers, n)
                temp_sum = np.sum(temp_radii)
                
                if temp_sum > best_sum:
                    best_sum = temp_sum
                    best_move = new_c
            
            if best_move is not None:
                centers[i] = best_move
                current_sum = best_sum
    
    radii = solve_radii_lp(centers, n)
    return centers, radii
