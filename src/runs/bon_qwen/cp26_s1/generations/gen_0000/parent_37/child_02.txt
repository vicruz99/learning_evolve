from scipy.optimize import minimize
import math

    centers, radii = init_hexagonal(n)
    
    # Pack into optimization vector
    params = np.zeros(3 * n)

        params[3 * i + 1] = centers[i, 1]
        params[3 * i + 2] = radii[i]

                x0,

def init_hexagonal(n):
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    r = 0.09
    idx = 0
    y = r
    row = 0
    
    while idx < n:
        x = r
        if row % 2 == 1:
            x += r * math.sqrt(3) / 2
        
        while x <= 1 - r and idx < n:
            centers[idx] = [x, y]
            radii[idx] = r
            idx += 1
            x += 2 * r
        
        y += r * math.sqrt(3)
        row += 1
    
    return centers, radii


def get_bounds(n):
    bounds = []
    for i in range(n):
        bounds.extend([(0.001, 0.999), (0.001, 0.999)])
        bounds.append((0.001, 0.45))
    return bounds


def compute_objective(params, n):
    centers = np.column_stack((params[0::3], params[1::3]))
    radii = np.maximum(params[2::3], 1e-10)
    
    penalty_weight = 50000.0
    
    penalty = 0.0
    
    # Boundary penalty
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        for v in [x - r, 1.0 - x - r, y - r, 1.0 - y - r]:
            if v < 0:
                penalty += penalty_weight * v * v
    
    # Overlap penalty
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = math.sqrt(dx * dx + dy * dy)
            min_dist = radii[i] + radii[j]
            if dist < min_dist:
                overlap = min_dist - dist
                penalty += penalty_weight * overlap * overlap
    
    return -np.sum(radii) + penalty


def fix_constraints(centers, radii, n):
    for iteration in range(500):
        changed = False
        for i in range(n):
            r = radii[i]
            x, y = centers[i]
            
            # Enforce boundary constraints
            r = min(r, x - 1e-10, 1.0 - x - 1e-10, y - 1e-10, 1.0 - y - 1e-10)
            
            # Enforce non-overlap constraints
            for j in range(n):
                if i == j:
                    continue
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = math.sqrt(dx * dx + dy * dy)
                required = dist - radii[j] - 1e-10
                if required < r:
                    r = required
            
            if abs(r - radii[i]) > 1e-12:
                radii[i] = max(r, 1e-10)
                changed = True
        
        if not changed:
            break
    
    return centers, radii
