import numpy as np
from scipy.optimize import linprog, minimize

# Global constants
N_CIRCLES = 26

def get_max_sum_radii(centers):
    """
    Solves the LP to find the maximum sum of radii for fixed centers.
    Returns (sum_radii, radii_array).
    """
    n = centers.shape[0]
    
    # Objective: Maximize sum(r_i) -> Minimize -sum(r_i)
    c_obj = -np.ones(n)
    
    # Constraints: A_ub * r <= b_ub
    A_ub = []
    b_ub = []
    
    # Precompute pairwise distances using broadcasting
    # dists[i, j] = distance between center i and center j
    # Shape (n, n)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    # 1. Pairwise Distance Constraints: r_i + r_j <= dist_ij
    # We only need to add constraints for unique pairs (i, j) where i < j
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    # 2. Boundary Constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        x, y = centers[i]
        # Distance to nearest wall
        d_boundary = min(x, 1 - x, y, 1 - y)
        
        row = np.zeros(n)
        row[i] = 1.0
        A_ub.append(row)
        b_ub.append(d_boundary)
        
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    # Bounds: 0 <= r_i < infinity
    bounds = [(0, None) for _ in range(n)]
    
    # Solve LP using HiGHS method (fast and robust)
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return -res.fun, res.x
        else:
            # Fallback if LP fails (e.g. numerical issues)
            # Return a small valid packing or 0
            # To be safe, we can try to return 0 sum
            return 0.0, np.zeros(n)
    except Exception:
        return 0.0, np.zeros(n)

def objective(centers_flat):
    """
    Objective function for the optimizer.
    Minimizes the negative sum of radii.
    """
    centers = centers_flat.reshape((N_CIRCLES, 2))
    
    # Clip centers to [0, 1] to handle optimizer steps that might go out of bounds
    # Note: L-BFGS-B handles bounds, but clipping adds safety for the LP
    centers = np.clip(centers, 0.0, 1.0)
    
    sum_r, radii = get_max_sum_radii(centers)
    
    # We want to maximize sum_r, so we minimize -sum_r
    return -sum_r

def generate_hexagonal_grid():
    """
    Generates an initial configuration of 26 centers in a hexagonal pattern.
    """
    centers = []
    
    # Hexagonal grid parameters
    # We want to fill the square [0,1]x[0,1]
    # Let's create a dense grid and pick points.
    # A simple way to generate hex lattice points:
    # Row y spacing: sqrt(3)/2 * s
    # Col x spacing: s
    # Odd rows shifted by s/2
    
    # Let's guess a spacing s. 
    # If we fit ~5 circles horizontally, s ~ 1/5 = 0.2.
    # If we fit ~5 rows vertically, s*sqrt(3)/2 ~ 1/5 => s ~ 0.23.
    # Let's use a slightly larger grid to ensure coverage and let optimizer shrink/shift.
    
    s = 0.22 
    y_spacing = s * np.sqrt(3) / 2
    
    row_idx = 0
    y = 0.15 # Initial y offset
    
    while len(centers) < N_CIRCLES:
        x = 0.1 # Initial x offset
        # Shift odd rows
        if row_idx % 2 == 1:
            x += s / 2
        
        while x <= 0.9 and len(centers) < N_CIRCLES:
            centers.append([x, y])
            x += s
        
        y += y_spacing
        row_idx += 1
        
    # Convert to numpy array and ensure shape (26, 2)
    centers = np.array(centers[:N_CIRCLES])
    
    # Randomly perturb slightly to break symmetry and avoid local minima on grid
    centers += np.random.uniform(-0.01, 0.01, size=centers.shape)
    
    # Clip to bounds
    centers = np.clip(centers, 0, 1)
    
    return centers

def run_packing():
    """
    Main function to run the packing optimization.
    Returns (centers, radii, sum_radii).
    """
    # Set random seed for reproducibility (optional, but good for debugging)
    np.random.seed(42)
    
    # 1. Generate Initial Centers
    centers_init = generate_hexagonal_grid()
    
    # Flatten for optimizer
    x0 = centers_init.flatten()
    
    # 2. Define Bounds for Optimizer
    # Each coordinate must be in [0, 1]
    bounds = [(0.0, 1.0)] * (N_CIRCLES * 2)
    
    # 3. Run Optimization
    # Using L-BFGS-B which supports bounds
    # We might run it a few times or with high maxiter
    res = minimize(objective, x0, method='L-BFGS-B', bounds=bounds, 
                   options={'maxiter': 500, 'ftol': 1e-9})
    
    centers_opt = res.x.reshape((N_CIRCLES, 2))
    
    # 4. Compute Final Radii
    sum_radii, radii = get_max_sum_radii(centers_opt)
    
    # 5. Validation Check (Self-check)
    # Just to be sure, though the problem statement implies we just return
    # We rely on the LP constraints ensuring validity.
    
    return centers_opt, radii, sum_radii

if __name__ == "__main__":
    centers, radii, total_r = run_packing()
    print(f"Sum of radii: {total_r}")
    print(f"Number of circles: {len(radii)}")
    
    # Quick validation logic inline
    valid = True
    if np.isnan(centers).any() or np.isnan(radii).any():
        print("NaN detected")
        valid = False
    
    for i in range(len(radii)):
        x, y = centers[i]
        r = radii[i]
        if r < 0 or x < r or x > 1-r or y < r or y > 1-r:
            print(f"Circle {i} invalid")
            valid = False
            
    for i in range(len(radii)):
        for j in range(i + 1, len(radii)):
            d = np.sqrt(np.sum((centers[i] - centers[j])**2))
            if d < radii[i] + radii[j] - 1e-9:
                print(f"Overlap {i} {j}")
                valid = False
                
    if valid:
        print("Packing is valid.")
    else:
        print("Packing validation failed.")
