

To solve the problem of packing 26 circles in a unit square to maximize the sum of their radii, we use a multi-start local optimization strategy. Since the problem is non-convex and has many variables (78 dimensions for 26 circles), a single local optimizer might get stuck in a poor local optimum. Therefore, we:

1.  **Generate diverse initial configurations**:
    *   **Hexagonal Lattice**: A high-density packing arrangement that serves as a strong starting point.
    *   **Perturbed Grid**: A 5x5 grid configuration with the 26th circle added in a gap, which is then optimized.
    *   **Random Valid Packing**: Randomly placed circles with radii set to ensure an initially valid (non-overlapping) configuration.
2.  **Optimization**: We use the SLSQP (Sequential Least Squares Programming) method from `scipy.optimize`. This method is well-suited for problems with non-linear constraints.
3.  **Variable Scaling**: We define a penalty-based objective function and handle constraints to ensure that the optimization converges to a solution that maximizes $\sum r_i$ while satisfying non-overlap and boundary constraints.

The code implements these strategies, running multiple iterations and returning the best result found.

```python
import numpy as np
from scipy.optimize import minimize

def run_packing():
    """
    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    Uses a multi-start approach with local optimization.
    """
    N = 26
    
    # --- Optimization Setup ---
    
    def get_obj_and_con():
        """
        Returns the objective function and constraints for the optimizer.
        Variables are flattened as [x1, y1, r1, x2, y2, r2, ...]
        """
        def obj_func(vars_flat):
            # Radii are at indices 2, 5, 8, ...
            radii = vars_flat[2::3]
            return -np.sum(radii)

        constraints = []
        
        # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
        for i in range(N):
            idx_x = 3 * i
            idx_y = 3 * i + 1
            idx_r = 3 * i + 2
            
            constraints.append({'type': 'ineq', 'fun': lambda v, i=i: v[idx_x] - v[idx_r]}) # x >= r
            constraints.append({'type': 'ineq', 'fun': lambda v, i=i: 1.0 - v[idx_x] - v[idx_r]}) # x + r <= 1
            constraints.append({'type': 'ineq', 'fun': lambda v, i=i: v[idx_y] - v[idx_r]}) # y >= r
            constraints.append({'type': 'ineq', 'fun': lambda v, i=i: 1.0 - v[idx_y] - v[idx_r]}) # y + r <= 1

        # Overlap constraints: dist(i, j) >= r_i + r_j
        for i in range(N):
            for j in range(i + 1, N):
                idx_xi, idx_yi, idx_ri = 3 * i, 3 * i + 1, 3 * i + 2
                idx_xj, idx_yj, idx_rj = 3 * j, 3 * j + 1, 3 * j + 2
                
                def overlap_constraint(v, i=i, j=j):
                    xi, yi, ri = v[idx_xi], v[idx_yi], v[idx_ri]
                    xj, yj, rj = v[idx_xj], v[idx_yj], v[idx_rj]
                    return (xi - xj)**2 + (yi - yj)**2 - (ri + rj)**2
                
                constraints.append({'type': 'ineq', 'fun': overlap_constraint})
        
        return obj_func, constraints

    obj_func, constraints = get_obj_and_con()

    def flat_to_arrays(v):
        centers = np.column_stack((v[0::3], v[1::3]))
        radii = v[2::3]
        return centers, radii

    def arrays_to_flat(centers, radii):
        v = np.zeros(3 * N)
        v[0::3] = centers[:, 0]
        v[1::3] = centers[:, 1]
        v[2::3] = radii
        return v

    # --- Initialization Strategies ---

    def init_hexagonal():
        centers = np.zeros((N, 2))
        # Hexagonal lattice spacing approximation
        s = 0.22 
        y = 0
        row = 0
        count = 0
        while count < N and y <= 1.0:
            x = 0.0
            if row % 2 == 1:
                x = s / 2.0
            while x <= 1.0 and count < N:
                if count < N:
                    centers[count] = [x, y]
                x += s
                count += 1
            y += s * np.sqrt(3) / 2.0
            row += 1
        
        # If not enough points, fill rest randomly or just return
        if count < N:
            centers[count:] = np.random.rand(N - count, 2)
        
        # Center the configuration in the square
        centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0) + 1e-8) * 0.8 + 0.1
        radii = np.full(N, 0.05)
        return centers, radii

    def init_grid():
        centers = np.zeros((N, 2))
        # 5x5 grid
        coords = np.array([0.1, 0.3, 0.5, 0.7, 0.9])
        xx, yy = np.meshgrid(coords, coords)
        pts = np.c_[xx.ravel(), yy.ravel()]
        centers[:25] = pts
        # 26th point in a gap
        centers[25] = [0.5, 0.5] # Center is already there, so we'll perturb it
        centers[25] = [0.5, 0.6] 
        
        # Small perturbation to avoid exact symmetry issues
        centers += np.random.uniform(-0.02, 0.02, centers.shape)
        centers = np.clip(centers, 0.05, 0.95)
        
        radii = np.full(N, 0.04)
        return centers, radii

    def init_random_valid():
        centers = np.random.rand(N, 2)
        radii = np.full(N, 0.001)
        
        # Iteratively grow radii until they touch or hit boundary
        # This is a simple heuristic to get a valid, decent starting point
        for _ in range(100):
            for i in range(N):
                max_r = 1.0
                # Boundary
                max_r = min(max_r, centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])
                # Neighbors
                for j in range(N):
                    if i != j:
                        dist = np.linalg.norm(centers[i] - centers[j])
                        max_r = min(max_r, dist - radii[j])
                radii[i] = max(0.001, max_r * 0.5) # Grow to 50% of available space
        return centers, radii

    # --- Optimization Loop ---
    best_sum = -1.0
    best_centers = None
    best_radii = None
    best_vars = None

    strategies = [init_hexagonal, init_grid, init_random_valid]
    
    # Run optimization multiple times
    for _ in range(5): # Repeat each strategy a few times
        for strategy in strategies:
            try:
                centers, radii = strategy()
                v0 = arrays_to_flat(centers, radii)
                
                res = minimize(obj_func, v0, method='SLSQP', constraints=constraints,
                               options={'maxiter': 1000, 'disp': False})
                
                if res.success or res.fun < best_sum:
                    v_opt = res.x
                    c_opt, r_opt = flat_to_arrays(v_opt)
                    current_sum = np.sum(r_opt)
                    
                    # Check validity manually to be safe
                    valid = True
                    for i in range(N):
                        x, y, r = c_opt[i, 0], c_opt[i, 1], r_opt[i]
                        if r < 0 or x - r < -1e-9 or x + r > 1 + 1e-9 or y - r < -1e-9 or y + r > 1 + 1e-9:
                            valid = False
                            break
                    for i in range(N):
                        for j in range(i + 1, N):
                            dist = np.linalg.norm(c_opt[i] - c_opt[j])
                            if dist < r_opt[i] + r_opt[j] - 1e-9:
                                valid = False
                                break
                        if not valid: break
                    
                    if valid:
                        if current_sum > best_sum:
                            best_sum = current_sum
                            best_centers = c_opt.copy()
                            best_radii = r_opt.copy()
                            best_vars = v_opt.copy()
            except Exception as e:
                continue

    if best_centers is None:
        # Fallback to random valid if all optimizations failed
        centers, radii = init_random_valid()
        best_centers, best_radii = centers, radii
        best_sum = np.sum(radii)

    return best_centers, best_radii, best_sum

def run_packing_wrapper():
    centers, radii, sum_radii = run_packing()
```
