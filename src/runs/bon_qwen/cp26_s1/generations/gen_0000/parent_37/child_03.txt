            x_start = r_init
        else:
            x_start = 2 * r_init
        
        x = x_start

            centers.append([x, y])
            x += 2 * r_init
            row_circles += 1
        
        row_idx += 1
        
    centers = np.array(centers[:n])
    radii = np.full(n, r_init)

            # y - r >= 0

            # 1 - (y + r) >= 0
