import numpy as np
from scipy.optimize import minimize

def _objective(vars, n):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(vars[2*n:])

def _constraints(vars, n):
    """Constraint function: returns array of inequality constraints >= 0."""
    c = vars[:2*n].reshape(n, 2)
    r = vars[2*n:]
    con = []
    
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    con.append(c[:, 0] - r)
    con.append(1.0 - c[:, 0] - r)
    con.append(c[:, 1] - r)
    con.append(1.0 - c[:, 1] - r)
    
    # Non-overlap constraints: distance >= r_i + r_j
    diffs = c[:, None, :] - c[None, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    con.append(dists[mask] - (r[:, None] + r[None, :])[mask])
    
    return np.concatenate(con)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # Initial configuration: Hexagonal lattice perturbed for symmetry breaking
    centers = []
    s = 0.25
    for i in range(10):
        y = i * s * np.sqrt(3) / 2
        if y > 0.9: break
        offset = (i % 2) * s / 2
        for j in range(10):
            x = offset + j * s
            if x > 0.9: break
            centers.append([x, y])
        if len(centers) >= n: break
        
    centers = np.array(centers[:n])
    centers += np.random.normal(0, 0.01, centers.shape)
    centers = np.clip(centers, 0.05, 0.95)
    radii = np.ones(n) * 0.05
    x0 = np.concatenate([centers.ravel(), radii])
    
    cons = {'type': 'ineq', 'fun': _constraints, 'args': (n,)}
    
    best_vars = None
    best_val = -np.inf
    
    # Multi-start optimization to escape local minima
    for _ in range(5):
        try:
            res = minimize(_objective, x0, args=(n,), method='SLSQP', 
                           constraints=cons, options={'maxiter': 500, 'ftol': 1e-10})
            if res.success:
                val = -res.fun
                if val > best_val:
                    best_val = val
                    best_vars = res.x
            # Perturb for next restart
            x0 = best_vars + np.random.normal(0, 0.002, len(x0)) if best_vars is not None else x0
        except Exception:
            pass
            
    if best_vars is not None:
        centers = best_vars[:2*n].reshape(n, 2)
        radii = best_vars[2*n:]
    else:
        # Fallback grid packing if optimization fails
        grid = np.array([(i/5, j/5) for i in range(6) for j in range(6)])[0:n]
        centers = grid + np.random.normal(0, 0.02, grid.shape)
        centers = np.clip(centers, 0.1, 0.9)
        radii = np.ones(n) * 0.08
        
    # Post-optimization validation and clamping to ensure strict validity
    for _ in range(20):
        min_margin = 1.0
        min_margin = min(min_margin, np.min(centers[:, 0] - radii), np.min(1 - centers[:, 0] - radii))
        min_margin = min(min_margin, np.min(centers[:, 1] - radii), np.min(1 - centers[:, 1] - radii))
        
        diffs = centers[:, None, :] - centers[None, :, :]
        dists = np.sqrt(np.sum(diffs**2, axis=2))
        ri_rj = radii[:, None] + radii[None, :]
        margins = dists - ri_rj
        np.fill_diagonal(margins, np.inf)
        min_margin = min(min_margin, np.min(margins))
        
        if min_margin < -1e-9:
            scale = 1.0 + min_margin * 0.95
            radii *= max(scale, 0.99)
        else:
            break
            
    radii = np.maximum(radii, 0.0)
    return centers, radii, float(np.sum(radii))
