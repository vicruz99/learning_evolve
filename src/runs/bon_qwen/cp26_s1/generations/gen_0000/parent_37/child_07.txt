To maximize the sum of radii for 26 circles in a unit square, we treat this as a geometric optimization problem. The goal is to find a configuration where the minimum distance between any two circles (and between circles and the square boundaries) is maximized. If the minimum pairwise distance is $d_{\min}$ and the minimum distance to the wall is $w_{\min}$, the maximum possible equal radius is $r = \min(d_{\min}/2, w_{\min})$. Maximizing this $r$ effectively maximizes the sum of radii (assuming equal radii, which is a strong heuristic for this objective).

We will use a force-directed simulation approach (similar to molecular dynamics) combined with a gradual increase of the target radius.
1.  **Initialization**: Place 26 circles in a compact hexagonal-like grid pattern inside the square with a small initial radius (e.g., 0.08) to ensure a valid starting configuration.
2.  **Iterative Expansion**: Gradually increase the target radius $r$.
3.  **Repulsion Forces**: At each step, compute repulsive forces between overlapping circles (distance $< 2r$) and between circles and the walls (distance to wall $< r$).
    -   Circle-Circle Force: Push centers apart proportional to the overlap amount.
    -   Wall Force: Push centers away from boundaries proportional to the boundary violation.
4.  **Relaxation**: Apply these forces to update circle positions using a small learning rate. Repeat this relaxation multiple times for each radius increment to resolve overlaps.
5.  **Termination**: Continue increasing $r$ until the system cannot resolve overlaps or a maximum iteration limit is reached. The final positions and radius will yield a high sum of radii.

This method avoids local minima by starting small and "growing" the circles, allowing the system to find a dense packing configuration naturally. We will use `numpy` for vectorized calculations to ensure efficiency.

import numpy as np

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Packs 26 circles in a unit square [0,1]x[0,1] to maximize sum of radii.
    Returns (centers, radii, sum_radii).
    """
    n = 26
    
    # Initial radius
    # 26 circles area ~ 26 * pi * r^2 <= 1 => r <= 0.11 approx
    # Start slightly lower to allow room for packing
    r = 0.08
    
    # Initialize centers in a grid pattern
    # Spacing 2r to avoid initial overlaps
    centers = np.zeros((n, 2))
    idx = 0
    
    # Fill grid
    # We can fit approx (1/(2r)) circles per row
    step = 2 * r + 1e-6 # tiny buffer
    x = r
    while x <= 1 - r:
        y = r
        while y <= 1 - r and idx < n:
            centers[idx, 0] = x
            centers[idx, 1] = y
            idx += 1
            y += step
        x += step
        
    # If grid didn't fill enough (unlikely for small r), fill remaining randomly
    while idx < n:
        cx, cy = np.random.rand(2)
        # Simple validity check not strictly needed for init, but good practice
        centers[idx] = [cx, cy]
        idx += 1

    # Center the configuration to maximize symmetry
    # Calculate center of mass
    com = np.mean(centers, axis=0)
    # Shift to 0.5, 0.5
    centers += (0.5 - com)
    
    # Ensure we are inside bounds after shift (clipping if necessary)
    # Though with small r and centering, it should be fine.
    # Just to be safe, push back if outside
    centers[:, 0] = np.clip(centers[:, 0], r, 1 - r)
    centers[:, 1] = np.clip(centers[:, 1], r, 1 - r)

    # Optimization parameters
    dr = 0.0002  # Step size for increasing radius
    max_iter = 3000 # Total iterations for radius increase
    relax_steps = 100 # Relaxation steps per radius increment
    k = 200.0      # Stiffness of repulsion
    lr = 0.005     # Learning rate for position updates
    
    # We will perform a loop to increase r and relax
    for _ in range(max_iter):
        # Increase radius
        r += dr
        
        # Relaxation loop
        for _ in range(relax_steps):
            forces = np.zeros((n, 2))
            
            # 1. Circle-Circle Repulsion
            # Compute pairwise distances and overlaps
            # Vectorized diff: (n, 1, 2) - (1, n, 2) -> (n, n, 2)
            diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
            dists = np.linalg.norm(diffs, axis=2)
            
            # Diagonal is 0, ignore
            np.fill_diagonal(dists, np.inf)
            
            # Overlap amount: 2r - dist. If negative, no overlap.
            overlaps = 2 * r - dists
            overlaps = np.maximum(overlaps, 0)
            
            # Direction vectors (normalized)
            # Avoid division by zero
            safe_dists = np.where(dists > 1e-9, dists, 1e-9)
            directions = diffs / safe_dists[:, :, np.newaxis]
            
            # Force magnitude proportional to overlap
            # F = k * overlap * direction
            # Since diffs[i,j] = -diffs[j,i], forces will be equal and opposite
            force_tensor = k * overlaps[:, :, np.newaxis] * directions
            
            # Sum forces on each circle
            forces += np.sum(force_tensor, axis=1)
            
            # 2. Wall Repulsion
            # Left wall (x < r)
            viol_left = r - centers[:, 0]
            viol_left = np.maximum(viol_left, 0)
            forces[:, 0] += k * viol_left
            
            # Right wall (x > 1-r)
            viol_right = centers[:, 0] - (1 - r)
            viol_right = np.maximum(viol_right, 0)
            forces[:, 0] -= k * viol_right
            
            # Bottom wall (y < r)
            viol_bottom = r - centers[:, 1]
            viol_bottom = np.maximum(viol_bottom, 0)
            forces[:, 1] += k * viol_bottom
            
            # Top wall (y > 1-r)
            viol_top = centers[:, 1] - (1 - r)
            viol_top = np.maximum(viol_top, 0)
            forces[:, 1] -= k * viol_top
            
            # Update positions
            centers += lr * forces
            
            # Hard constraint clamping to prevent numerical escape
            centers[:, 0] = np.clip(centers[:, 0], 0, 1)
            centers[:, 1] = np.clip(centers[:, 1], 0, 1)
            
            # Optional: If we are deep inside, we can allow small violations for better packing?
            # But strict constraints are safer for validation.
            # However, forces should handle it.

    # After optimization, the radius might be slightly too large for the exact positions
    # We need to compute the valid radius for the final positions
    # r_valid = min(min_dist / 2, min_dist_to_wall)
    
    # Compute pairwise distances
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.linalg.norm(diffs, axis=2)
    np.fill_diagonal(dists, np.inf)
    min_dist = np.min(dists)
    
    # Compute distance to walls
    dist_left = centers[:, 0]
    dist_right = 1 - centers[:, 0]
    dist_bottom = centers[:, 1]
    dist_top = 1 - centers[:, 1]
    
    dist_to_wall = np.min(np.array([dist_left, dist_right, dist_bottom, dist_top]), axis=0)
    min_dist_to_wall = np.min(dist_to_wall)
    
    # The valid radius is constrained by both
    final_r = min(min_dist / 2, min_dist_to_wall)
    
    # To be safe against numerical errors, shrink slightly
    final_r = final_r * 0.9999
    
    # If final_r is very small (optimization failed), fallback
    if final_r < 0.01:
        final_r = 0.05
        # Re-center or random? Just fallback to valid small circles
        centers = np.random.rand(n, 2)
        # Ensure valid for r=0.05
        centers[:, 0] = np.clip(centers[:, 0], 0.05, 0.95)
        centers[:, 1] = np.clip(centers[:, 1], 0.05, 0.95)
        
    radii = np.full(n, final_r)
    sum_radii = np.sum(radii)
    
    # Final validation check (internal)
    # Ensure strictly valid
    # If any violation, reduce r further
    valid = True
    # Check walls
    for i in range(n):
        if centers[i, 0] < radii[i] or centers[i, 0] > 1 - radii[i] or \
           centers[i, 1] < radii[i] or centers[i, 1] > 1 - radii[i]:
            valid = False
            break
    
    if valid:
        # Check overlaps
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                if dist < radii[i] + radii[j] - 1e-12:
                    # Overlap detected, reduce r globally
                    valid = False
                    break
            if not valid:
                break
    
    if not valid:
        # Fallback: reduce r until valid
        # This is a safety net
        step = 0.001
        while not valid and radii[0] > 0.01:
            radii -= step
            final_r -= step # update scalar if needed
            
            valid = True
            for i in range(n):
                if centers[i, 0] < radii[i] or centers[i, 0] > 1 - radii[i] or \
                   centers[i, 1] < radii[i] or centers[i, 1] > 1 - radii[i]:
                    valid = False
                    break
            
            if valid:
                for i in range(n):
                    for j in range(i + 1, n):
                        dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                        if dist < radii[i] + radii[j] - 1e-12:
                            valid = False
                            break
                    if not valid:
                        break
    
    sum_radii = np.sum(radii)
    
    return centers, radii, float(sum_radii)

def validate_internal(centers, radii):
    n = centers.shape[0]
    if np.isnan(centers).any() or np.isnan(radii).any():
        return False
    for i in range(n):
        if radii[i] < 0: return False
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            return False
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False
    return True
