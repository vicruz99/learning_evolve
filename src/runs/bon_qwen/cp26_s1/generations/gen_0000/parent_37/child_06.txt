import numpy as np
from scipy.optimize import minimize
import math

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    
    Returns:
        tuple: (centers, radii, sum_radii)
    """
    n = 26
    np.random.seed(42) # For reproducibility

    # 1. Initialization: Hexagonal-like grid
    # We arrange circles in rows to provide a good starting point for the optimizer.
    # Row configuration: 6, 5, 5, 5, 5 (Total 26)
    # We start with a small radius to ensure validity, then let the optimizer expand them.
    
    r_init = 0.04 # Initial small radius
    centers = []
    
    # Generate hexagonal coordinates
    # Vertical spacing for hex packing
    dy = r_init * math.sqrt(3)
    
    row_counts = [6, 5, 5, 5, 5]
    current_row = 0
    
    for count in row_counts:
        y = (current_row + 0.5) * dy + r_init
        
        # Horizontal centering for the row
        # Width occupied by 'count' circles in hex packing is approx count * 2*r, 
        # but with staggering, effective width is (count-1)*r + 2r? 
        # Actually, centers are spaced by 2r.
        # We center the row in [0, 1].
        
        if count == 1:
            x_start = 0.5 - r_init
            x_positions = [x_start]
        else:
            # For hex packing, rows are often shifted. 
            # Here we just place them centered to start.
            total_width = (count - 1) * 2 * r_init
            x_start = (1 - total_width) / 2
            x_positions = [x_start + i * 2 * r_init for i in range(count)]
        
        # Stagger odd rows slightly? 
        # Let's keep it simple for initialization, optimizer will fix geometry.
        if current_row % 2 == 1:
            # Shift by r_init to create hex pattern
            x_positions = [x + r_init for x in x_positions]
            # Re-center if needed, but simple shift is usually fine for start
            # Actually, let's just ensure they stay in [0,1]
            # If shifted, check bounds. If out, clamp or don't shift?
            # With r_init=0.04, shift is small.
        
        for x in x_positions:
            # Ensure inside bounds [r, 1-r]
            x = max(r_init, min(1 - r_init, x))
            y = max(r_init, min(1 - r_init, y))
            centers.append([x, y])
        
        current_row += 1
        
    centers = np.array(centers)
    radii = np.full(n, r_init)
    
    # 2. Optimization
    # Variables: [x0, y0, r0, x1, y1, r1, ..., x25, y25, r25]
    x0 = np.concatenate([centers.flatten(), radii])
    
    def objective(vars):
        # Maximize sum of radii -> Minimize negative sum
        r_vals = vars[5::3] # Extract radii (every 3rd var starting from index 2, stride 3)
        return -np.sum(r_vals)

    def constraint_boundary(vars):
        c = vars.reshape(-1, 3)
        # c[:, 0] is x, c[:, 1] is y, c[:, 2] is r
        # x - r >= 0
        # 1 - (x + r) >= 0
        # y - r >= 0
        # 1 - (y + r) >= 0
        # r >= 0.001 (lower bound)
        
        cons = []
        for i in range(n):
            x, y, r = c[i]
            cons.append(x - r)
            cons.append(1.0 - (x + r))
            cons.append(y - r)
            cons.append(1.0 - (y + r))
            cons.append(r - 0.001)
        return np.array(cons)

    def constraint_overlap(vars):
        c = vars.reshape(-1, 3)
        cons = []
        for i in range(n):
            for j in range(i + 1, n):
                x1, y1, r1 = c[i]
                x2, y2, r2 = c[j]
                
                dist = math.sqrt((x1 - x2)**2 + (y1 - y2)**2)
                # dist >= r1 + r2  => dist - (r1 + r2) >= 0
                cons.append(dist - (r1 + r2))
        return np.array(cons)

    # Combine constraints
    # SLSQP requires constraints in a specific format or a list of dictionaries.
    # We can return a single array for inequality constraints (g(x) >= 0).
    
    def all_constraints(vars):
        return np.concatenate([constraint_boundary(vars), constraint_overlap(vars)])

    cons = {'type': 'ineq', 'fun': all_constraints}
    
    # Bounds for x, y, r
    bounds = [(0, 1)] * (2 * n) + [(0.001, 0.5)] * n

    # Run optimization
    # SLSQP is robust but can be slow. We limit maxiter.
    res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, 
                   options={'maxiter': 1000, 'ftol': 1e-9, 'disp': False})

    if res.success:
        final_vars = res.x
    else:
        # Fallback to initial if optimization fails (unlikely)
        final_vars = x0

    final_centers = final_vars[:2*n].reshape(n, 2)
    final_radii = final_vars[2*n:]
    
    # Validation check (internal)
    # The provided validate_packing function uses a tolerance. 
    # We rely on the optimizer constraints.
    
    sum_radii = np.sum(final_radii)
    
    return final_centers, final_radii, sum_radii
