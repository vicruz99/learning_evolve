import numpy as np
from scipy.optimize import minimize

def calculate_penalties(centers, radii):
    """
    Calculate a smooth penalty score for a configuration.
    The score is 0 if the configuration is valid, and positive otherwise.
    """
    penalty = 0.0
    n = centers.shape[0]
    
    # Pairwise overlap penalties
    # Using vectorized operations for efficiency
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.linalg.norm(diff, axis=2)
    min_dists = radii[:, np.newaxis] + radii[np.newaxis, :]
    
    # Calculate violation where distance < sum of radii
    violations = min_dists - dists
    # Only consider upper triangle to avoid double counting and self
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    overlaps = violations[mask]
    if len(overlaps) > 0:
        penalty += np.sum(np.maximum(0, overlaps) ** 2)

    # Boundary penalties
    # x - r >= 0 => r - x <= 0
    # x + r <= 1 => x + r - 1 <= 0
    # Same for y
    
    # Left boundary (x - r < 0)
    pen_x_min = radii - centers[:, 0]
    penalty += np.sum(np.maximum(0, pen_x_min) ** 2)
    
    # Right boundary (x + r > 1)
    pen_x_max = centers[:, 0] + radii - 1.0
    penalty += np.sum(np.maximum(0, pen_x_max) ** 2)
    
    # Bottom boundary (y - r < 0)
    pen_y_min = radii - centers[:, 1]
    penalty += np.sum(np.maximum(0, pen_y_min) ** 2)
    
    # Top boundary (y + r > 1)
    pen_y_max = centers[:, 1] + radii - 1.0
    penalty += np.sum(np.maximum(0, pen_y_max) ** 2)
    
    return penalty

def optimize_positions(centers, radii):
    """
    Optimize circle centers for fixed radii to minimize overlaps and boundary violations.
    """
    n = centers.shape[0]
    
    def objective(x_flat):
        x = x_flat.reshape(n, 2)
        return calculate_penalties(x, radii)

    # Bounds for centers: [0, 1]
    bounds = [(0, 1)] * (n * 2)
    
    # Initial guess
    x0 = centers.flatten()
    
    # Use L-BFGS-B for bound-constrained optimization
    result = minimize(objective, x0, method='L-BFGS-B', bounds=bounds, options={'ftol': 1e-12, 'maxiter': 2000})
    
    return result.x.reshape(n, 2)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    # 1. Initialization: Hexagonal-like Grid
    # We want to distribute 26 circles. A 5x5 grid has 25.

    centers = np.zeros((n_circles, 2))

    grid_coords = np.linspace(0.1, 0.9, 5)
    count = 0

            centers[count, 0] = grid_coords[i]
            centers[count, 1] = grid_coords[j]
            count += 1
            
    # Place the 26th circle in the center
    centers[count, 0] = 0.5
    centers[count, 1] = 0.5
    
    # Initial small radius
    radii = np.full(n_circles, 0.02)
    
    # 2. Iterative Expansion and Relaxation
    # We gradually increase the radius and re-optimize positions
    current_radius = 0.02
    target_radius = 0.12 # We aim higher, solver will find the limit
    
    # Number of expansion steps
    steps = 100
    
    # Adaptive step size
    radius_step = (target_radius - current_radius) / steps
    
    for step in range(steps):
        # Increase radii slightly
        radii += radius_step
        
        # Optimize positions to resolve overlaps caused by expansion
        centers = optimize_positions(centers, radii)
        
        # If penalty is still high, we might be pushing too hard, 
        # but the solver tries its best. 
        # For robustness, we could check penalty and adjust step, 
        # but simple expansion usually works for this scale.
        
    # 3. Final Refinement
    # Run a few more optimization steps with fixed radii to settle any jitter
    for _ in range(5):
        centers = optimize_positions(centers, radii)

    # Though optimizer with bounds should handle centers.
