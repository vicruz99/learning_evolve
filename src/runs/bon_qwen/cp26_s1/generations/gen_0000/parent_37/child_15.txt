def make_init(n, seed):
    np.random.seed(seed)

    configs = []

    # Config type 1: Hexagonal staggered rows 5+6+5+5+5
    centers = []
    row_spacing = 0.185
    for row in range(6):
        if row == 0 or row == 2 or row == 3 or row == 4:
            n_row = 5
        elif row == 1:
            n_row = 6
        else:
            n_row = max(0, n - len(centers))
        offset = (row % 2) * 0.088
        for col in range(n_row):
            x = 0.055 + col * 0.176 + offset
            y = 0.055 + row * row_spacing
            centers.append([x, y])
        if len(centers) >= n:
            break
    centers = np.array(centers[:n])
    radii = np.ones(n) * 0.06
    configs.append((centers, radii))

    radii = np.ones(n) * 0.06
    configs.append((centers, radii))

def opt_penalty(centers, radii, n, penalty=100):
    params = np.concatenate([centers.flatten(), radii])

    def objective(p):
        c = p[:2 * n].reshape(n, 2)
        r = p[2 * n:]

        obj = -np.sum(r)

        for i in range(n):
            x = c[i, 0]
            y = c[i, 1]
            ri = r[i]

            v = x - ri
            if v < 0:
                obj += penalty * v * v
            v = 1.0 - x - ri
            if v < 0:
                obj += penalty * v * v
            v = y - ri
            if v < 0:
                obj += penalty * v * v
            v = 1.0 - y - ri
            if v < 0:
                obj += penalty * v * v

        for i in range(n):
            for j in range(i + 1, n):
                dx = c[i, 0] - c[j, 0]
                dy = c[i, 1] - c[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                overlap = r[i] + r[j] - dist
                if overlap > 0:
                    obj += penalty * overlap * overlap

        for i in range(n):
            if r[i] < 0:
                obj += penalty * r[i] * r[i]

        return obj

    bounds = [(0.01, 0.99)] * (2 * n) + [(0.001, 0.45)] * n

    result = minimize(
        objective, params, method="L-BFGS-B",
        bounds=bounds,
        options={"maxiter": 1500, "ftol": 1e-16, "gtol": 1e-10}
    )

    centers = result.x[:2 * n].reshape(n, 2)
    radii = np.maximum(result.x[2 * n:], 1e-8)

    return centers, radii


def opt_slsqp(centers, radii, n):
    params = np.concatenate([centers.flatten(), radii])

    def objective(p):
        return -np.sum(p[2 * n:])

    def constraint_func(p):
        c = p[:2 * n].reshape(n, 2)
        r = p[2 * n:]
        cons = []

        for i in range(n):
            x = c[i, 0]
            y = c[i, 1]
            ri = r[i]
            cons.append(x - ri)
            cons.append(1.0 - x - ri)
            cons.append(y - ri)
            cons.append(1.0 - y - ri)

        for i in range(n):
            for j in range(i + 1, n):
                dx = c[i, 0] - c[j, 0]
                dy = c[i, 1] - c[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                cons.append(dist - r[i] - r[j])

        for i in range(n):
            cons.append(r[i])

        return np.array(cons)

    cons = {"type": "ineq", "fun": constraint_func}
    bounds = [(0, 1)] * (2 * n) + [(0, 0.5)] * n

    result = minimize(
        objective, params, method="SLSQP",
        constraints=cons,
        bounds=bounds,
        options={"maxiter": 2000, "ftol": 1e-15}
    )

    centers = result.x[:2 * n].reshape(n, 2)
    radii = result.x[2 * n:]

    return centers, radii
