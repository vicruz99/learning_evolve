                idx += 1
        centers[idx] = [0.5, 0.5]
        
        # Add random noise
        centers += np.random.normal(0, 0.05, centers.shape)
        centers = np.clip(centers, 0.01, 0.99)
        
        radii = np.full(n_circles, 0.09) # Start small
        
        # Optimization
        # We want to maximize sum(radii) - penalty(overlap)
        # We can use a simple gradient free optimizer or SLSQP.
        # Let's use SLSQP.
        
        # Objective: minimize -sum(radii) + lambda * overlap
        # But lambda needs to be large.
        
        def obj(p):
            c = p[:2*n_circles].reshape(-1, 2)
            r = p[2*n_circles:]
            # Penalty
            ov = calculate_overlap(c, r)
            return -np.sum(r) + 1000.0 * ov
        
        # Bounds
        b = [(0.0, 1.0)] * (2*n_circles) + [(0.0, 0.5)] * n_circles
        
        # Initial point
        x0 = np.concatenate([centers.flatten(), radii])
        
        res = minimize(obj, x0, method='SLSQP', bounds=b, options={'maxiter': 200, 'ftol': 1e-10})
        
        if res.success or res.nit > 190:
            c_res = res.x[:2*n_circles].reshape(-1, 2)
            r_res = res.x[2*n_circles:]
            
            # Check overlap
            ov = calculate_overlap(c_res, r_res)
            if ov < 1e-4: # Valid packing
                s = np.sum(r_res)
                if s > best_overall_sum:
                    best_overall_sum = s
                    best_centers_overall = c_res
                    best_radii_overall = r_res
    
    # If we didn't find a valid packing with high sum, fallback to a known valid one.
    # 5x5 grid with r=0.1 and one small circle?
    # Actually, 5x5 grid sum 2.5.
    # Let's construct a valid packing with 26 circles.
    # 5x5 grid (25 circles) r=0.1.
    # 26th circle: place at (0.5, 0.5) is occupied.
    # Place at (0.2, 0.2)? Occupied by circle (0.1, 0.1) ... no (0.2, 0.2) is hole.
    # Hole at (0.2, 0.2).
    # Neighbors: (0.1, 0.1), (0.3, 0.1), (0.1, 0.3), (0.3, 0.3).
    # Dist to (0.1, 0.1) is sqrt(0.02) ≈ 0.1414.
    # Radius of neighbors 0.1.
    # Space available: 0.1414 - 0.1 = 0.0414.
    # So r_26 = 0.0414.
    # Sum = 25*0.1 + 0.0414 = 2.5414.
    # This is better than 2.5.
    # Can we do better?
    # Maybe move the 25 circles apart to make space for a larger 26th?
    # If we move them, their radii might decrease.
    
    # Let's try to construct this "hole" solution explicitly if optimization fails.
    # But optimization should find it.
    
    if best_centers_overall is None:
        # Fallback: 5x5 grid + 1 small circle in hole
        centers_fb = np.zeros((n_circles, 2))
        idx = 0
        for i in range(5):
            for j in range(5):
                centers_fb[idx] = [0.1 + i*0.2, 0.1 + j*0.2]
                idx += 1
        # Place 26th at (0.2, 0.2) - but wait, (0.2, 0.2) is equidistant to 4 corners.
        # Actually, the hole center is (0.2, 0.2) ?
        # Grid points: (0.1, 0.1), (0.3, 0.1), (0.1, 0.3), (0.3, 0.3).
        # Center of square formed by these is (0.2, 0.2).
        # Distance to any corner is sqrt((0.1)^2 + (0.1)^2) = sqrt(0.02) ≈ 0.14142.
        # Radius of corner circles is 0.1.
        # Max radius for center circle = 0.14142 - 0.1 = 0.04142.
        centers_fb[25] = [0.2, 0.2]
        radii_fb = np.full(n_circles, 0.1)
        radii_fb[25] = 0.04142
        
        # But wait, circle at (0.2, 0.2) with r=0.04142.
        # Does it overlap with boundary?
        # x-r = 0.158 > 0. x+r = 0.241 < 1. OK.
        # So this is valid.
        # Sum = 2.54142.
        
        # Can we fit more holes?
        # There are 16 holes in 5x5 grid.
        # But circles in holes would overlap each other.
        # Hole at (0.2, 0.2) and (0.2, 0.4)?
        # Dist = 0.2.
        # If both have r=0.041, sum of radii 0.082 < 0.2. No overlap.
        # But we only have 1 extra circle.
        
        # Maybe we can fit 26 circles with better sum?
        # 5x5 grid is not optimal for 26.
        
        best_centers_overall = centers_fb
        best_radii_overall = radii_fb
        best_overall_sum = np.sum(radii_fb)

    # Final validation check
    centers_final = best_centers_overall
    radii_final = best_radii_overall
    
    # Ensure shapes
    centers_final = np.array(centers_final).reshape(26, 2)
    radii_final = np.array(radii_final).reshape(26)
    
    sum_radii = np.sum(radii_final)
    
    return centers_final, radii_final, sum_radii
