    """

    for constraint violations.
    
    Args:
        x: np.array of shape (3*N,) containing [x1, y1, r1, x2, y2, r2, ...]
        
    Returns:
        loss: float, value to minimize
    """

    x_coords = x[0::3]
    y_coords = x[1::3]
    radii = x[2::3]

    loss = -np.sum(radii)

    # Boundary violations

        loss += P * np.sum((viol1[mask1])**2)

    viol2 = x_coords + radii - 1.0

    # Overlap violations
