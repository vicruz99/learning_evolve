def get_hexagonal_lattice_points(num_points, spacing):
    """
    Generate points on a hexagonal lattice with a given spacing.
    """
    points = []
    # Vertical spacing between rows
    row_height = spacing * np.sqrt(3) / 2
    
    y = spacing / 2
    row_idx = 0
    
    while y < 1 + spacing / 2:
        # Horizontal offset for alternating rows
        x_start = spacing / 2 if row_idx % 2 == 0 else 0
        
        x = x_start
        while x < 1 + spacing / 2:
            if 0 <= x <= 1 and 0 <= y <= 1:
                points.append((x, y))
            x += spacing
        
        y += row_height
        row_idx += 1
        
    return np.array(points)

def get_max_separated_points(points, n):
    """
    Greedily select n points from a set of points such that the minimum distance 
    between any pair of selected points is maximized.
    """
    selected = []
    if len(points) <= n:
        return points

    # Start with the point farthest from the origin (or just the first one)
    # A heuristic: start with the center point if available, or random.
    # Let's just pick the first one.
    current = points[0]
    selected.append(current)
    
    while len(selected) < n:
        # Find the point in 'points' that is farthest from the nearest selected point
        max_min_dist = -1
        best_point = None
        
        for p in points:
            if np.any(p == selected, axis=0).all():
                continue
            
            # Calculate min distance to any selected point
            dists = np.linalg.norm(selected - p, axis=1)
            min_dist = np.min(dists)
            
            if min_dist > max_min_dist:
                max_min_dist = min_dist
                best_point = p
        
        if best_point is not None:
            selected.append(best_point)
        else:
            break
            
    return np.array(selected)
