import numpy as np

def force_optimize(centers, radii, steps=400):
    """
    Optimizes circle positions to minimize overlaps and boundary violations.
    """
    n = centers.shape[0]
    lr = 0.01
    
    # Precompute pair indices
    i_indices, j_indices = np.triu_indices(n, k=1)
    
    for _ in range(steps):
        forces = np.zeros((n, 2))
        
        # Vectorized distance calculation
        diff = centers[i_indices] - centers[j_indices]
        dists = np.linalg.norm(diff, axis=1)
        min_dists = radii[i_indices] + radii[j_indices]
        
        # Overlap mask
        overlap_mask = dists < min_dists
        
        if np.any(overlap_mask):
            d = dists[overlap_mask]
            min_d = min_dists[overlap_mask]
            d_vec = diff[overlap_mask]
            
            safe_d = np.where(d > 1e-9, d, 1e-9)
            overlap = min_d - d
            f_mag = overlap * 100.0 
            
            ux = d_vec[:, 0] / safe_d
            uy = d_vec[:, 1] / safe_d
            
            fx = ux * f_mag
            fy = uy * f_mag
            
            i_ov = i_indices[overlap_mask]
            j_ov = j_indices[overlap_mask]
            
            # Scatter add
            np.add.at(forces, (i_ov, 0), fx)
            np.add.at(forces, (i_ov, 1), fy)
            np.add.at(forces, (j_ov, 0), -fx)
            np.add.at(forces, (j_ov, 1), -fy)
        
        # Boundary forces
        mask_left = centers[:, 0] < radii
        forces[mask_left, 0] += (radii[mask_left] - centers[mask_left, 0]) * 1000.0
        
        mask_right = centers[:, 0] > 1 - radii
        forces[mask_right, 0] -= (centers[mask_right, 0] - (1 - radii[mask_right])) * 1000.0
        
        mask_bottom = centers[:, 1] < radii
        forces[mask_bottom, 1] += (radii[mask_bottom] - centers[mask_bottom, 1]) * 1000.0
        
        mask_top = centers[:, 1] > 1 - radii
        forces[mask_top, 1] -= (centers[mask_top, 1] - (1 - radii[mask_top])) * 1000.0
        
        centers += forces * lr
        np.clip(centers, 0.0, 1.0, out=centers)

    return centers
