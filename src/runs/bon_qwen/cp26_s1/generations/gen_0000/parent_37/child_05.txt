        return -np.sum(r)

        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: v[i*3+1] - v[i*3+2]})

    result = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints,

    return final_centers, final_radii, float(sum_radii)
