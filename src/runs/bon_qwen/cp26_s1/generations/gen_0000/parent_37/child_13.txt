def calculate_sum_of_radii(params, n_circles):
    """Helper to calculate sum of radii from flattened parameter array."""
    radii = params[n_circles * 2:]
    return np.sum(radii)

def constraints_function(params, n_circles):
    """
    Defines the constraints for the optimization problem.
    1. Non-overlap: dist(C_i, C_j) >= r_i + r_j
    2. Inside Square: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    """
    centers = params[:n_circles * 2].reshape(n_circles, 2)
    radii = params[n_circles * 2:]
    
    constraints = []
    
    # Boundary constraints
    for i in range(n_circles):
        x, y = centers[i]
        r = radii[i]
        # x - r >= 0
        constraints.append({'type': 'ineq', 'fun': lambda p, i=i: p[i*2] - p[n_circles*2+i]})
        # x + r <= 1 => 1 - x - r >= 0
        constraints.append({'type': 'ineq', 'fun': lambda p, i=i: 1.0 - p[i*2] - p[n_circles*2+i]})
        # y - r >= 0
        constraints.append({'type': 'ineq', 'fun': lambda p, i=i: p[i*2+1] - p[n_circles*2+i]})
        # y + r <= 1 => 1 - y - r >= 0
        constraints.append({'type': 'ineq', 'fun': lambda p, i=i: 1.0 - p[i*2+1] - p[n_circles*2+i]})
        
        # Non-negative radii (strictly positive for valid packing)
        constraints.append({'type': 'ineq', 'fun': lambda p, i=i: p[n_circles*2+i]})

    # Non-overlap constraints
    for i in range(n_circles):
        for j in range(i + 1, n_circles):
            def dist_constraint(p, i=i, j=j):

                r2 = p[n_circles*2+j]
                dist = np.linalg.norm(c1 - c2)
                return dist - (r1 + r2)
            constraints.append({'type': 'ineq', 'fun': dist_constraint})

    return constraints
