import numpy as np
from scipy.optimize import minimize


class ConstraintFunc:
    """Wrapper to bind arguments to a top-level function without closures."""
    def __init__(self, fun, *args):
        self.fun = fun
        self.args = args

    def __call__(self, params):
        return self.fun(params, *self.args)


def obj_func(params, n):
    """Negative sum of radii (minimizing this maximizes the sum)."""
    return -np.sum(params[n * 2:])


def constraint_boundary_left(params, n, i):
    return params[2 * i] - params[n * 2 + i]


def constraint_boundary_right(params, n, i):
    return 1.0 - params[2 * i] - params[n * 2 + i]


def constraint_boundary_bottom(params, n, i):
    return params[2 * i + 1] - params[n * 2 + i]


def constraint_boundary_top(params, n, i):
    return 1.0 - params[2 * i + 1] - params[n * 2 + i]


def constraint_nonoverlap(params, n, i, j):
    dx = params[2 * i] - params[2 * j]
    dy = params[2 * i + 1] - params[2 * j + 1]
    dist = np.sqrt(dx * dx + dy * dy)
    return dist - params[n * 2 + i] - params[n * 2 + j]


def constraint_radius_nonneg(params, n, i):
    return params[n * 2 + i]


def build_constraints(n):
    """Build all constraint dictionaries for n circles."""
    constraints = []
    for i in range(n):
        constraints.append({'type': 'ineq', 'fun': ConstraintFunc(constraint_boundary_left, n, i)})
        constraints.append({'type': 'ineq', 'fun': ConstraintFunc(constraint_boundary_right, n, i)})
        constraints.append({'type': 'ineq', 'fun': ConstraintFunc(constraint_boundary_bottom, n, i)})
        constraints.append({'type': 'ineq', 'fun': ConstraintFunc(constraint_boundary_top, n, i)})
    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({'type': 'ineq', 'fun': ConstraintFunc(constraint_nonoverlap, n, i, j)})
    for i in range(n):
        constraints.append({'type': 'ineq', 'fun': ConstraintFunc(constraint_radius_nonneg, n, i)})
    return constraints


def compute_max_radius_for_circle(centers, radii, i):
    """Compute maximum feasible radius for circle i given all other circles' positions and radii."""
    n = len(radii)
    max_r = min(
        centers[i, 0],
        1.0 - centers[i, 0],
        centers[i, 1],
        1.0 - centers[i, 1]
    )
    for j in range(n):
        if i != j:
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            max_r = min(max_r, dist - radii[j])
    return max(max_r, 1e-10)


def create_hexagonal_init(n, row_configs, y_positions):
    """Create initial positions in a hexagonal pattern."""
    centers = []
    for row_i, count in enumerate(row_configs):
        y = y_positions[row_i]
        offset = (row_i % 2) * 0.04
        if count > 1:
            x_spacing = (0.8 - 2 * offset) / (count - 1)
            x_start = 0.1 + offset
        else:
            x_start = 0.5
            x_spacing = 1.0
        for col_i in range(count):
            x = x_start + col_i * x_spacing
            centers.append([x, y])
    return np.array(centers)


def optimize_packing(n, centers_init, radii_init, constraints, seed_val):
    """Run SLSQP optimization from given initial conditions."""
    np.random.seed(seed_val)
    params_init = np.concatenate([centers_init.flatten(), radii_init])

    result = minimize(
        ConstraintFunc(obj_func, n),
        params_init,
        method='SLSQP',
        constraints=constraints,
        options={'maxiter': 5000, 'ftol': 1e-14, 'disp': False}
    )

    centers = result.x[:n * 2].reshape(n, 2)
    radii = result.x[n * 2:]
    return centers, radii, result


def refine_radii(centers, radii, n, iterations=50):
    """Iteratively refine radii by computing exact maximum feasible radius."""
    for iteration in range(iterations):
        for i in range(n):
            new_r = compute_max_radius_for_circle(centers, radii, i)
            radii[i] = new_r
    return radii


def try_perturbation(centers, radii, n, constraints, base_seed):
    """Try optimizing with a random perturbation of initial positions."""
    np.random.seed(base_seed)
    perturbation = np.random.randn(n, 2) * 0.02
    centers_pert = np.clip(centers + perturbation, 0.05, 0.95)
    radii_pert = np.ones(n) * np.min(radii) * 0.9
    return optimize_packing(n, centers_pert, radii_pert, constraints, base_seed)


def run_packing():
    n = 26

    # Initial hexagonal configuration: 5 rows of 5 circles + 1 extra = 26
    row_configs = [5, 5, 5, 5, 5, 1]
    y_positions = np.linspace(0.12, 0.88, len(row_configs))
    centers_init = create_hexagonal_init(n, row_configs, y_positions)
    radii_init = np.ones(n) * 0.075

    # Build constraints (reusable across runs)
    constraints = build_constraints(n)

    # First optimization run
    best_centers, best_radii, best_result = optimize_packing(
        n, centers_init, radii_init, constraints, seed_val=42
    )
    best_sum = np.sum(best_radii)
    print(f"Initial optimization sum: {best_sum}")

    # Refine radii
    best_radii = refine_radii(best_centers, best_radii, n)
    best_sum = np.sum(best_radii)
    print(f"After refinement: {best_sum}")

    # Try multiple perturbations to escape local optima
    for seed_val in range(100):
        trial_centers, trial_radii, _ = try_perturbation(
            best_centers, best_radii, n, constraints, base_seed=seed_val
        )
        trial_radii = refine_radii(trial_centers, trial_radii, n)
        trial_sum = np.sum(trial_radii)
        if trial_sum > best_sum:
            best_sum = trial_sum
            best_centers = trial_centers.copy()
            best_radii = trial_radii.copy()
            if seed_val < 20:
                print(f"Improvement at seed {seed_val}: {best_sum}")

    # Final refinement
    best_radii = refine_radii(best_centers, best_radii, n, iterations=100)

    # Ensure strict feasibility by slightly reducing radii if needed
    for i in range(n):
        r = best_radii[i]
        best_centers[i, 0] = np.clip(best_centers[i, 0], r + 1e-8, 1.0 - r - 1e-8)
        best_centers[i, 1] = np.clip(best_centers[i, 1], r + 1e-8, 1.0 - r - 1e-8)

    # Recompute radii one final time
    best_radii = refine_radii(best_centers, best_radii, n, iterations=20)

    return best_centers, best_radii, np.sum(best_radii)
