import numpy as np
import scipy.optimize as opt

def solve_radii_lp(centers):
    """
    Given fixed centers, solve the LP to find radii maximizing sum(r_i).
    Constraints:
      r_i >= 0
      r_i <= dist(centers[i], boundary)
      r_i + r_j <= dist(centers[i], centers[j]) for all i < j
    """
    n = centers.shape[0]
    c = -np.ones(n) # Minimize -sum(r_i) -> Maximize sum(r_i)

    # Bounds for r_i: [0, 1] (upper bound 1 is loose, but safe)
    # Actually we can set tighter bounds using boundary distances
    bounds = []
    b_ub = []
    for i in range(n):
        x, y = centers[i]
        dist_to_boundary = min(x, 1-x, y, 1-y)
        bounds.append((0, max(0, dist_to_boundary + 1e-9))) # Allow slight slack for solver
        b_ub.append(dist_to_boundary)

    # Pairwise constraints: r_i + r_j <= dist_ij
    # We construct the matrix A_ub such that A_ub @ r <= b_ub
    # But b_ub in linprog is for inequalities.
    # We need to append pairwise constraints to the inequality matrix.
    
    # Initial inequalities: r_i <= b_ub[i]
    # These are diagonal entries.
    
    n_pairs = n * (n - 1) // 2
    A_ub = np.zeros((n + n_pairs, n))
    b_ub_full = np.zeros(n + n_pairs)
    
    # Boundary constraints
    for i in range(n):
        A_ub[i, i] = 1.0
        b_ub_full[i] = b_ub[i]
        
    # Pairwise constraints
    idx = n
    dists = np.linalg.norm(centers[:, np.newaxis] - centers, axis=2)
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub_full[idx] = dists[i, j]
            idx += 1
            
    # Solve LP
    # Using 'highs' method is generally robust and fast
    try:
        res = opt.linprog(c, A_ub=A_ub, b_ub=b_ub_full, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
        else:
            # Fallback to small radii if LP fails
            return np.full(n, 0.01), 0.26
    except Exception:
        return np.full(n, 0.01), 0.26

def repulsion_force(centers, radii):
    """
    Compute repulsive forces based on active constraints.
    """
    n = centers.shape[0]
    forces = np.zeros_like(centers)
    k_repulsion = 100.0 # Strong repulsion
    k_boundary = 100.0
    epsilon = 1e-6
    
    # Pairwise repulsion
    # We only apply force if circles are "touching" or overlapping (r_i + r_j >= dist)
    # Or if they are very close.
    for i in range(n):
        for j in range(i + 1, n):
            diff = centers[i] - centers[j]
            dist = np.sqrt(np.sum(diff**2))
            if dist < epsilon:
                # Prevent singularity
                dist = epsilon
                diff = np.random.randn(2) * epsilon
            
            sum_radii = radii[i] + radii[j]
            
            # If touching or overlapping, push apart
            if sum_radii >= dist - 1e-4:
                # Force magnitude proportional to penetration or just constant repulsion
                # To maximize space, we push them apart if they constrain each other
                strength = k_repulsion
                forces[i] += strength * diff / dist
                forces[j] -= strength * diff / dist

    # Boundary repulsion
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        
        # Left wall
        if x - r < 1e-4:
            forces[i, 0] += k_boundary
        # Right wall
        if 1 - x - r < 1e-4:
            forces[i, 0] -= k_boundary
        # Bottom wall
        if y - r < 1e-4:
            forces[i, 1] += k_boundary
        # Top wall
        if 1 - y - r < 1e-4:
            forces[i, 1] -= k_boundary
            
    return forces
