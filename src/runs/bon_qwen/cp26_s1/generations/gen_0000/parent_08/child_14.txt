import numpy as np
from scipy.optimize import minimize

def compute_penalty_objective(v, n, penalty_weight):
    """Compute the objective function with smooth penalty terms."""
    xs = v[:n]
    ys = v[n:2*n]
    rs = v[2*n:]
    
    pen = 0.0
    
    # Boundary penalties: ensure circles are inside [0,1]x[0,1]
    # max(0, r - x)^2 + max(0, r + x - 1)^2
    pen += np.sum(np.maximum(rs - xs, 0.0)**2)
    pen += np.sum(np.maximum(xs + rs - 1.0, 0.0)**2)
    pen += np.sum(np.maximum(rs - ys, 0.0)**2)
    pen += np.sum(np.maximum(ys + rs - 1.0, 0.0)**2)
    
    # Overlap penalties
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dists = np.sqrt(dx**2 + dy**2)
    
    r_sum = rs[:, None] + rs[None, :]
    violations = r_sum - dists
    
    # Sum over upper triangle to avoid double counting
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    pen += np.sum(np.maximum(violations[mask], 0.0)**2)
    
    # Objective: maximize sum of radii => minimize negative sum
    return -np.sum(rs) + penalty_weight * pen

def evaluate_true_sum(v, n):
    """Extract and return the sum of radii from the variable vector."""
    return np.sum(v[2*n:])

def run_packing():
    n = 26
    best_score = -np.inf
    best_v = None
    
    # Bounds for optimization: coordinates in [0,1], radii in [0, 0.5]
    bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n
    
    def generate_initial_guess(pattern):
        centers = np.zeros((n, 2))
        r0 = 0.06
        
        if pattern == 'hex':
            idx = 0
            rows = [6, 5, 6, 5, 4]
            dy = np.sqrt(3) * 0.08
            for r_idx, count in enumerate(rows):
                y = 0.08 + r_idx * dy
                x_start = (1 - (count - 1) * 2 * 0.08) / 2
                for c in range(count):
                    if idx < n:
                        centers[idx] = [x_start + c * 2 * 0.08, y]
                        idx += 1
        elif pattern == 'grid':
            xs = np.linspace(0.1, 0.9, 5)
            ys = np.linspace(0.1, 0.9, 5)
            idx = 0
            for y in ys:
                for x in xs:
                    if idx < n:
                        centers[idx] = [x, y]
                        idx += 1
            if idx < n:
                centers[idx] = [0.5, 0.5]
        else: # random
            centers = np.random.rand(n, 2) * 0.8 + 0.1
            
        v0 = np.concatenate([centers[:, 0], centers[:, 1], np.full(n, r0)])
        return v0

    # Phase 1: Fast global exploration using penalty method
    np.random.seed(42)
    patterns = ['hex', 'grid', 'random']
    
    for pat in patterns:
        v0 = generate_initial_guess(pat)
        # High penalty weight to enforce constraints during ascent
        res = minimize(
            lambda v: compute_penalty_objective(v, n, 1000.0),
            v0,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 4000, 'ftol': 1e-15, 'gtol': 1e-13}
        )
        
        score = evaluate_true_sum(res.x, n)
        if score > best_score:
            best_score = score
            best_v = res.x.copy()

    # Phase 2: Local refinement with strict constraints using SLSQP
    # This ensures the solution is mathematically valid without penalty artifacts
    def objective_slsqp(v):
        return -np.sum(v[2*n:])

    constraints = []
    # Boundary constraints
    for i in range(n):
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: v[i] - v[2*n + i]})
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: 1.0 - v[i] - v[2*n + i]})
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: v[n + i] - v[2*n + i]})
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: 1.0 - v[n + i] - v[2*n + i]})
        
    # Overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({
                'type': 'ineq',
                'fun': lambda v, i=i, j=j: np.sqrt((v[i]-v[j])**2 + (v[n+i]-v[n+j])**2) - v[2*n+i] - v[2*n+j]
            })

    res_slsqp = minimize(
        objective_slsqp,
        best_v,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 2000, 'ftol': 1e-12}
    )
    
    v_final = res_slsqp.x
    xs = v_final[:n]
    ys = v_final[n:2*n]
    rs = v_final[2*n:]
    
    # Shrink radii minimally to guarantee strict inequality for validation
    # This handles floating point tolerance issues
    rs = rs * (1.0 - 1e-7)
    
    centers = np.column_stack((xs, ys))
    radii = rs
    
    return centers, radii, float(np.sum(radii))
