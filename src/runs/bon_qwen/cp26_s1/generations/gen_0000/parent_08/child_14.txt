from scipy.optimize import minimize
import math

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    Pack 26 circles in a unit square [0,1]x[0,1] to maximize the sum of radii.

    best_radii = None

        # Hexagonal packing parameters
        # Spacing in x is 2r, vertical spacing is sqrt(3)r
        # We start with a theoretical r=0.1 to place centers
        r_init = 0.105

        row = 0
        # Estimate number of columns per row for hex packing
        # Width 1.0, approx 1/(2r) columns

        while idx < n and row < 10:
            # Offset for hexagonal rows
            x_start = r_init * (1 + (row % 2))
            y_pos = r_init + row * r_init * math.sqrt(3)
            
            if y_pos + r_init > 1.0:
                row += 1
                continue
            
            col = 0
            x_pos = x_start
            while x_pos + r_init <= 1.0 and idx < n:
                centers[idx] = [x_pos, y_pos]
                idx += 1
                x_pos += 2 * r_init
                col += 1
            row += 1
            
        return centers

    def calculate_penalty(centers, radii):

        n_c = centers.shape[0]

            x, y = centers[i]
            r = radii[i]
            
            # Left wall
            if x < r:
                penalty += (r - x) ** 2
            # Right wall
            if x > 1.0 - r:
                penalty += (x - (1.0 - r)) ** 2
            # Bottom wall
            if y < r:
                penalty += (r - y) ** 2
            # Top wall
            if y > 1.0 - r:
                penalty += (y - (1.0 - r)) ** 2
                
        # 2. Overlap penalties
        for i in range(n_c):

                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                min_dist = radii[i] + radii[j]
                if dist < min_dist:
                    penalty += (min_dist - dist) ** 2
                    
        return penalty
