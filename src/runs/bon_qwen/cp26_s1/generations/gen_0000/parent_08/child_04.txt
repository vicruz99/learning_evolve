import numpy as np
from scipy.optimize import minimize

def objective(vars):
    """Objective function: minimize negative sum of radii"""
    return -np.sum(vars[2::3])

def compute_constraints(vars):
    """
    Compute all inequality constraints. Returns an array of constraint values.
    For 'ineq' type, scipy expects fun(x) >= 0.
    """
    N = len(vars) // 3
    con_size = 4 * N + N * (N - 1) // 2
    con = np.empty(con_size)
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    con[:N] = vars[0::3] - vars[2::3]
    con[N:2*N] = 1.0 - vars[0::3] - vars[2::3]
    con[2*N:3*N] = vars[1::3] - vars[2::3]
    con[3*N:4*N] = 1.0 - vars[1::3] - vars[2::3]
    
    # Pairwise non-overlap constraints: dist^2 - (r_i + r_j)^2 >= 0
    idx = 4 * N
    for i in range(N):
        for j in range(i + 1, N):
            dx = vars[3*i] - vars[3*j]
            dy = vars[3*i+1] - vars[3*j+1]
            dr = vars[3*i+2] + vars[3*j+2]
            con[idx] = dx*dx + dy*dy - dr*dr
            idx += 1
    return con

def run_packing():
    N = 26
    r_init = 0.05
    
    # Initialize centers with a hexagonal lattice pattern
    centers_init = np.zeros((N, 2))
    idx = 0
    for row in range(6):
        y = r_init + row * np.sqrt(3) * r_init
        for col in range(5):
            x = r_init + col * 2 * r_init + (row % 2) * r_init
            if idx < N:
                centers_init[idx] = [x, y]
                idx += 1
                
    # Flatten initial configuration: [x1, y1, r1, x2, y2, r2, ...]
    x0 = np.zeros(3 * N)
    for i in range(N):
        x0[3*i] = centers_init[i, 0]
        x0[3*i+1] = centers_init[i, 1]
        x0[3*i+2] = r_init
        
    # Variable bounds: coordinates in [0,1], radii in [0, 0.5]
    bounds = [(0.0, 1.0) for _ in range(2 * N)] + [(0.0, 0.5) for _ in range(N)]
    
    # Define constraints dictionary
    cons = ({'type': 'ineq', 'fun': compute_constraints})
    
    # Run optimization
    res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, 
                   options={'ftol': 1e-10, 'maxiter': 1500, 'disp': False})
                   
    final_vars = res.x
    
    # Extract results
    centers = np.column_stack((final_vars[0::3], final_vars[1::3]))
    radii = np.maximum(final_vars[2::3], 0.0)
    
    return centers, radii, np.sum(radii)
