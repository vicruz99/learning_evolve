import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # 1. Initial Configuration: Hexagonal Lattice
    centers = []
    k = 0
    # Generate a sufficiently large hexagonal grid and pick n points
    for row in range(6):
        for col in range(7):
            x = col * 2 + (row % 2) * 1.0
            y = row * np.sqrt(3)
            centers.append([x, y])
            k += 1
            if k == n:
                break
        if k == n:
            break
    centers = np.array(centers[:n])
    
    # Center and scale the points to fit within [0, 1] x [0, 1] with a small buffer
    centers -= centers.mean(axis=0)
    max_extent = np.max(np.abs(centers))
    centers /= (max_extent * 1.1) 
    centers += 0.5
    
    # Initial radii estimate (uniform)
    r_init = 0.08
    
    # Variable vector: [x1, y1, r1, x2, y2, r2, ...]
    x0 = np.zeros(3 * n)
    for i in range(n):
        x0[3*i] = centers[i, 0]
        x0[3*i+1] = centers[i, 1]
        x0[3*i+2] = r_init
        
    # 2. Optimization Setup
    # Bounds: 0 <= x,y <= 1, 0 <= r <= 0.5
    bounds = []
    for _ in range(n):
        bounds.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])

    def objective(z):
        # Negative sum of radii
        return -sum(z[3*i+2] for i in range(n))

    # Inequality constraints: c(z) >= 0
    constraints = []
    
    # Boundary constraints: r <= x <= 1-r => x-r >= 0, 1-x-r >= 0
    for i in range(n):
        idx_x = 3*i
        idx_y = 3*i+1
        idx_r = 3*i+2
        constraints.append({'type': 'ineq', 'fun': lambda z, i=i: z[idx_x] - z[idx_r]})
        constraints.append({'type': 'ineq', 'fun': lambda z, i=i: 1.0 - z[idx_x] - z[idx_r]})
        constraints.append({'type': 'ineq', 'fun': lambda z, i=i: z[idx_y] - z[idx_r]})
        constraints.append({'type': 'ineq', 'fun': lambda z, i=i: 1.0 - z[idx_y] - z[idx_r]})
        
    # Non-overlap constraints: dist >= r_i + r_j => dist^2 - (r_i+r_j)^2 >= 0
    for i in range(n):
        for j in range(i + 1, n):
            idx_i = 3*i
            idx_j = 3*j
            constraints.append({
                'type': 'ineq', 
                'fun': lambda z, i=i, j=j: 
                    (z[idx_i] - z[idx_j])**2 + (z[idx_i+1] - z[idx_j+1])**2 - (z[idx_i+2] + z[idx_j+2])**2
            })

    # Run optimization
    res = minimize(
        objective, 
        x0, 
        method='SLSQP', 
        bounds=bounds, 
        constraints=constraints, 
        options={'maxiter': 2000, 'ftol': 1e-9}
    )
    
    # 3. Extract and Format Results
    final_centers = np.zeros((n, 2))
    final_radii = np.zeros(n)
    
    for i in range(n):
        final_centers[i, 0] = res.x[3*i]
        final_centers[i, 1] = res.x[3*i+1]
        final_radii[i] = res.x[3*i+2]
        
    sum_radii = float(np.sum(final_radii))
    
    return final_centers, final_radii, sum_radii

if __name__ == "__main__":
    centers, radii, total_sum = run_packing()
    print(f"Total Sum of Radii: {total_sum:.6f}")
    print(f"Min Radius: {np.min(radii):.6f}, Max Radius: {np.max(radii):.6f}")
