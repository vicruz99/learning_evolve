def objective_function(params, n):
    """
    Objective function to minimize.
    We want to maximize the sum of radii, so we minimize -sum(radii).
    We add penalty terms for boundary violations and overlaps.
    """
    # Convert params to centers and radii arrays
    # params shape: (n * 3,) -> [x1, y1, r1, x2, y2, r2, ...]
    p_array = np.asarray(params)
    centers = p_array.reshape(n, 3)[:, :2]
    radii = p_array.reshape(n, 3)[:, 2]
    
    # Check for negative radii (though bounds handle it, penalty helps)
    if np.any(radii < 0):
        return 1e9
    
    penalty = 0.0
    
    # 1. Boundary penalties
    # Circle i must be inside [0,1]x[0,1].
    # Constraints: r_i <= x_i, r_i <= 1 - x_i, r_i <= y_i, r_i <= 1 - y_i
    # Violations are calculated as max(0, violation_amount)
    
    # x - r < 0 => r > x
    v1 = np.maximum(0, radii - centers[:, 0])
    # x + r > 1 => r > 1 - x
    v2 = np.maximum(0, radii - (1.0 - centers[:, 0]))
    # y - r < 0 => r > y
    v3 = np.maximum(0, radii - centers[:, 1])
    # y + r > 1 => r > 1 - y
    v4 = np.maximum(0, radii - (1.0 - centers[:, 1]))
    
    penalty += np.sum(v1**2 + v2**2 + v3**2 + v4**2)
    
    # 2. Overlap penalties
    # Distance between centers must be >= r_i + r_j
    # Violation: max(0, r_i + r_j - dist)
    # We iterate over all unique pairs
    for i in range(n):
        ri = radii[i]
        xi, yi = centers[i]
        for j in range(i + 1, n):
            rj = radii[j]
            dx = xi - centers[j, 0]
            dy = yi - centers[j, 1]
            dist = math.sqrt(dx*dx + dy*dy)

                penalty += viol**2
