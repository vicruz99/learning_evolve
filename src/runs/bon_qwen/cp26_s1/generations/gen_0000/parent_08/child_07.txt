import numpy as np
from scipy.optimize import linprog, differential_evolution

def solve_radii(centers):
    """
    Given fixed centers, solve the Linear Programming problem to maximize sum of radii.
    Constraints:
    1. r_i >= 0
    2. r_i <= min(x_i, 1-x_i, y_i, 1-y_i) (Boundary containment)
    3. r_i + r_j <= distance(center_i, center_j) (Non-overlap)
    """
    n = centers.shape[0]
    
    # 1. Bounds for radii based on distance to square boundaries
    bounds_r = []
    for i in range(n):
        x, y = centers[i]
        # Distance to nearest wall
        dist_wall = min(x, 1 - x, y, 1 - y)
        # Ensure non-negative upper bound
        ub = max(0.0, dist_wall)
        bounds_r.append((0.0, ub))

    # 2. Pairwise non-overlap constraints: r_i + r_j <= dist_ij
    # We construct the inequality matrix A_ub * r <= b_ub
    n_pairs = n * (n - 1) // 2
    A_ub = np.zeros((n_pairs, n))
    b_ub = np.zeros(n_pairs)
    
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dist
            idx += 1
            
    # Objective: Maximize sum(r_i) <=> Minimize -sum(r_i)
    c_obj = -np.ones(n)
    
    # Solve LP
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method='highs')
        if res.success:
            radii = res.x
            sum_radii = np.sum(radii)
            return radii, sum_radii
        else:
            # Fallback if LP fails (should be rare)
            return np.zeros(n), 0.0
    except Exception:
        return np.zeros(n), 0.0

def fitness(x_flat):
    """
    Objective function for optimization.
    x_flat is a 1D array of coordinates [x1, y1, x2, y2, ...]
    Returns negative sum of radii (since optimizers minimize).
    """
    n = 26
    centers = x_flat.reshape(n, 2)
    radii, sum_r = solve_radii(centers)
    return -sum_r

    )

        # vars_flat: [x1, y1, ..., xn, yn, r1, ..., rn]

        return -np.sum(radii)

        radii = vars_flat[2*n:]

        # Boundary constraints
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            cons.append(x - r)      # x - r >= 0
            cons.append(1 - x - r)  # 1 - x - r >= 0
            cons.append(y - r)      # y - r >= 0
            cons.append(1 - y - r)  # 1 - y - r >= 0
            
        # Non-overlap constraints
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.linalg.norm(centers[i] - centers[j])
                cons.append(dist - radii[i] - radii[j])
                
        return cons

    x0 = np.concatenate([centers_de.flatten(), radii_de])
    
    # Bounds for all variables
    bounds_full = []
    for _ in range(2 * n):
        bounds_full.append((0.0, 1.0)) # Centers

        bounds_full.append((0.0, 1.0)) # Radii (loose upper bound)

    try:
        res_nl = minimize(
            full_objective,
            x0,
            method='SLSQP',
            bounds=bounds_full,
