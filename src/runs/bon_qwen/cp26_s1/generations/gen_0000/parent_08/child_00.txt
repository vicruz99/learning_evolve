import numpy as np
from scipy.optimize import minimize, NonlinearConstraint

N = 26

def compute_objective(params):
    """Objective function: maximize sum of radii (minimize negative sum)"""
    radii = params[2 * N:]
    return -np.sum(radii)

def compute_constraints(params):
    """
    Returns a vector of constraint values that must be >= 0.
    First 4*N elements: boundary constraints.
    Remaining elements: pairwise non-overlap constraints.
    """
    centers = params[:2 * N].reshape(N, 2)
    radii = params[2 * N:]
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    c_x_left = centers[:, 0] - radii
    c_x_right = 1.0 - centers[:, 0] - radii
    c_y_bottom = centers[:, 1] - radii
    c_y_top = 1.0 - centers[:, 1] - radii
    bc = np.hstack([c_x_left, c_x_right, c_y_bottom, c_y_top])
    
    # Overlap constraints: dist^2 - (r1 + r2)^2 >= 0
    # Vectorized pairwise distance calculation
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist_sq = np.sum(diff ** 2, axis=2)
    r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    oc = dist_sq - r_sum ** 2
    
    # Extract upper triangle to avoid duplicates
    tri_idx = np.triu_indices(N, k=1)
    oc_vec = oc[tri_idx]
    
    return np.hstack([bc, oc_vec])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    best_obj = -np.inf
    best_params = None
    
    # Multiple random restarts to escape local optima
    for seed in range(7):
        np.random.seed(seed)
        # Initial guess: random positions with safe margins, moderate radius
        centers = np.random.rand(N, 2) * 0.8 + 0.1
        radii = np.full(N, 0.06)
        params = np.hstack([centers.flatten(), radii])
        
        # Bounds for variables
        bounds = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N
        
        # Define constraint object
        cons = NonlinearConstraint(compute_constraints, lb=0.0, ub=np.inf)
        
        # Run optimizer
        res = minimize(
            compute_objective, 
            params, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons, 
            options={'maxiter': 3000, 'ftol': 1e-14, 'disp': False}
        )
        
        # Track best result
        if -res.fun > best_obj:
            best_obj = -res.fun
            best_params = res.x.copy()
            
    # Fallback if optimization fails completely
    if best_params is None:
        best_params = np.hstack([np.random.rand(N, 2).flatten(), np.ones(N) * 0.01])
        
    centers = best_params[:2 * N].reshape(N, 2)
    radii = best_params[2 * N:]
    
    # Post-processing: Ensure strict validity within numerical tolerance
    # If any constraint is slightly violated, uniformly shrink radii
    min_constraint = np.min(compute_constraints(best_params))
    if min_constraint < -1e-9:
        shrink_factor = 1.0 + min_constraint / (np.max(radii) + 1e-12)
        radii *= shrink_factor
        
    return centers, radii, float(np.sum(radii))
