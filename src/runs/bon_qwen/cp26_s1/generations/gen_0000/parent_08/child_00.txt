- Phase 5: Multiple restarts from different initial configurations

def compute_pairwise_distances(centers):
    n = centers.shape[0]
    dists = np.zeros((n, n))
    for i in range(n):
        for j in range(i+1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dists[i, j] = np.sqrt(dx*dx + dy*dy)
            dists[j, i] = dists[i, j]
    return dists

def validate_quick(centers, radii):
    n = centers.shape[0]
    if np.isnan(centers).any() or np.isnan(radii).any():
        return False
    for i in range(n):
        if radii[i] < 0:
            return False
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-10 or x + r > 1 + 1e-10 or y - r < -1e-10 or y + r > 1 + 1e-10:
            return False
    for i in range(n):

            dy = centers[i, 1] - centers[j, 1]

                return False
    return True

def force_optimize(centers, radii, n_iters=20000, growth_target=0.105):
    """Force-based optimization with gradual radius growth"""
    centers = centers.copy()
    radii = radii.copy()
    n = len(radii)
    
    best_centers = centers.copy()
    best_radii = radii.copy()
    best_sum = np.sum(radii)
    
    dt = 0.008
    repulsion_strength = 5.0
    boundary_strength = 8.0
    pressure_strength = 0.3
    
    for iteration in range(n_iters):
        progress = iteration / n_iters
        targets = 0.08 + progress * (growth_target - 0.08)
        
        forces = np.zeros((n, 2))
        max_force = 0.0
        
        for i in range(n):
            for j in range(i+1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy) + 1e-12
                min_dist = radii[i] + radii[j]
                
                if dist < min_dist:
                    overlap = min_dist - dist
                    fx = dx / dist * overlap * repulsion_strength
                    fy = dy / dist * overlap * repulsion_strength
                    forces[i] += [fx, fy]
                    forces[j] -= [fx, fy]
                    max_force = max(max_force, overlap)
        
        for i in range(n):
            for d in range(2):
                min_bound = radii[i]
                max_bound = 1.0 - radii[i]
                if centers[i, d] < min_bound:
                    forces[i, d] += (min_bound - centers[i, d]) * boundary_strength
                elif centers[i, d] > max_bound:
                    forces[i, d] -= (centers[i, d] - max_bound) * boundary_strength
            
            target_r = targets
            if radii[i] < target_r:
                forces[i, 0] += pressure_strength * 0.1
                forces[i, 1] += pressure_strength * 0.1
        
        centers += forces * dt
        
        for i in range(n):
            centers[i, 0] = max(radii[i], min(1.0 - radii[i], centers[i, 0]))
            centers[i, 1] = max(radii[i], min(1.0 - radii[i], centers[i, 1]))
        
        if iteration % 5 == 0 and iteration > 0:
            radii *= 1.0002
        
        temp = 0.002 * (1.0 - progress * 0.95)
        centers += np.random.randn(n, 2) * temp

        if current_sum > best_sum:

                best_centers = centers.copy()
                best_radii = radii.copy()
                best_sum = current_sum

                max_r = min(centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])
                radii[i] = min(radii[i], max_r)
    
    return best_centers, best_radii, best_sum

def scipy_refine(centers, radii, n=26):
    """Use scipy to refine the packing"""
    x0 = np.concatenate([centers.flatten(), radii])
    
    def objective(x):
        return -np.sum(x[2*n:])
    
    def constraint(i, j, x):
        ci = x[2*i:2*i+2]
        cj = x[2*j:2*j+2]
        ri = x[2*n + i]
        rj = x[2*n + j]
        dist = np.sqrt(np.sum((ci - cj)**2))
        return dist - ri - rj
    
    def boundary_constraint(i, d, x):
        ci = x[2*i + d]
        ri = x[2*n + i]
        return ci - ri
    
    def boundary_constraint2(i, d, x):
        ci = x[2*i + d]
        ri = x[2*n + i]
        return 1 - ci - ri
    
    constraints = []
    for i in range(n):
        for j in range(i+1, n):
            constraints.append({
                'type': 'ineq',
                'fun': lambda x, i=i, j=j: constraint(i, j, x)
            })
        for d in range(2):
            constraints.append({
                'type': 'ineq',
                'fun': lambda x, i=i, d=d: boundary_constraint(i, d, x)
            })
            constraints.append({
                'type': 'ineq',
                'fun': lambda x, i=i, d=d: boundary_constraint2(i, d, x)
            })
    
    bounds = [(0, 1)] * (2*n) + [(0, 0.5)] * n
    
    try:
        result = minimize(objective, x0, method='SLSQP', bounds=bounds, 
                         constraints=constraints, options={'maxiter': 500, 'ftol': 1e-12})
        if result.success or result.fun < -np.sum(radii):
            new_centers = result.x[:2*n].reshape(n, 2)
            new_radii = result.x[2*n:]
            new_radii = np.maximum(new_radii, 0)
            return new_centers, new_radii
    except Exception:
        pass
    
    return centers, radii
