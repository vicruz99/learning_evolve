    Returns 0 if valid, positive value if overlaps/out of bounds exist.

    n = centers.shape[0]
    energy = 0.0
    
    # Boundary penalties
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x < r: energy += (r - x)**2
        if x > 1 - r: energy += (x - (1 - r))**2
        if y < r: energy += (r - y)**2
        if y > 1 - r: energy += (y - (1 - r))**2

            r = current_r + (target_r - current_r) * (inc / num_increases)

                    # Right
