def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    np.random.seed(42)
    
    n_circles = 26
    # Initialize centers in a hexagonal grid pattern
    # This provides a dense initial configuration
    centers = np.zeros((n_circles, 2))
    count = 0
    row = 0
    while count < n_circles:
        col = 0
        shift = 0.5 if row % 2 == 1 else 0.0
        while True:
            x = 0.05 + col * 0.15 + shift * 0.075
            y = 0.05 + row * 0.13
            if x <= 0.95 and y <= 0.95:
                centers[count] = [x, y]
                count += 1
                if count >= n_circles:
                    break
            col += 1
        row += 1
        
    # Truncate to 26
    centers = centers[:n_circles]
    
    # Initial radii
    radii = np.full(n_circles, 0.05)
    
    # Optimization parameters
    learning_rate = 0.01
    lambda_sum = 1.0  # Weight for maximizing sum of radii
    penalty_weight = 100.0 # Weight for constraints

    n_steps = 5000
    
    for step in range(n_steps):
        # Adjust parameters over time (annealing)
        # Increase lambda to push for larger radii
        current_lambda = lambda_sum * (1 + 0.001 * step)
        # Decrease learning rate
        current_lr = learning_rate * (1 - 0.9 * (step / n_steps))

        grad_centers = np.zeros_like(centers)
        grad_radii = np.zeros_like(radii)

        for i in range(n_circles):
            x, y = centers[i]
            r = radii[i]

            # Penalty max(0, r - x)^2
            err_left = r - x
            if err_left > 0:
                grad_radii[i] += 2 * penalty_weight * err_left
                grad_centers[i, 0] -= 2 * penalty_weight * err_left
            
            # Right boundary: x + r <= 1 => x + r - 1 <= 0
            err_right = x + r - 1
            if err_right > 0:
                grad_radii[i] += 2 * penalty_weight * err_right
                grad_centers[i, 0] += 2 * penalty_weight * err_right
                
            # Bottom boundary: y - r >= 0
            err_bottom = r - y
            if err_bottom > 0:
                grad_radii[i] += 2 * penalty_weight * err_bottom
                grad_centers[i, 1] -= 2 * penalty_weight * err_bottom

            if err_top > 0:
                grad_radii[i] += 2 * penalty_weight * err_top
                grad_centers[i, 1] += 2 * penalty_weight * err_top

            for j in range(i + 1, n_circles):
                vec = centers[j] - centers[i]
                dist = np.sqrt(np.sum(vec**2))
                sum_r = radii[i] + radii[j]

                    grad_radii[i] += 2 * penalty_weight * overlap
                    grad_radii[j] += 2 * penalty_weight * overlap
                    
                    # d/dc_i (overlap^2) = 2 * overlap * (-d(dist)/dc_i)
                    # d(dist)/dc_i = -vec / dist
                    # So term is 2 * overlap * vec / dist

                    grad_centers[i] -= force
                    grad_centers[j] += force

        grad_radii -= current_lambda

        radii -= current_lr * grad_radii
