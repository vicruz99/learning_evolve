            dx = c[j][0] - c[i][0]
            dy = c[j][1] - c[i][1]
            dist = np.sqrt(dx * dx + dy * dy)
            min_dist = r[i] + r[j]
            if dist < min_dist:
                penalty += (min_dist - dist) ** 2

    # Boundary and radius penalties
    for i in range(n):
        x_pos, y_pos = c[i]
        ri = r[i]
        penalty += max(0, ri - x_pos) ** 2
        penalty += max(0, x_pos - (1 - ri)) ** 2
        penalty += max(0, ri - y_pos) ** 2
        penalty += max(0, y_pos - (1 - ri)) ** 2
        penalty += max(0, -ri) ** 2

    return obj + 5000 * penalty


def clean_packing(centers, radii, n):
    centers = centers.copy()
    radii = radii.copy()

    for iteration in range(2000):
        overlap_found = False

        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[j][0] - centers[i][0]
                dy = centers[j][1] - centers[i][1]
                dist = np.sqrt(dx * dx + dy * dy)
                if dist < radii[i] + radii[j] - 1e-10:
                    overlap_found = True
                    if radii[i] <= radii[j]:
                        radii[i] = max(1e-6, (dist - radii[j] - 1e-10) / 2)
                    else:
                        radii[j] = max(1e-6, (dist - radii[i] - 1e-10) / 2)

        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            centers[i][0] = np.clip(x, r + 1e-12, 1 - r - 1e-12)
            centers[i][1] = np.clip(y, r + 1e-12, 1 - r - 1e-12)

        if not overlap_found:
            break

    valid = True
    for i in range(n):
        if radii[i] <= 0:
            valid = False
            break
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-10 or x + r > 1 + 1e-10 or y - r < -1e-10 or y + r > 1 + 1e-10:
            valid = False
            break

    if valid:
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[j][0] - centers[i][0]
                dy = centers[j][1] - centers[i][1]
                dist = np.sqrt(dx * dx + dy * dy)
                if dist < radii[i] + radii[j] - 1e-10:
                    valid = False
                    break
            if not valid:
                break

    return valid, centers, radii
