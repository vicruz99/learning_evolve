

Looking at this problem, I need to pack 26 circles in a unit square to maximize sum of radii, targeting ~2.636.

<strategy>
My strategy:
1. Start with multiple grid configurations (5×6, 6×5, etc.) as initialization points
2. Use force-based relaxation to spread circles apart and reduce overlaps
3. Iteratively optimize: fix radii → optimize positions, fix positions → compute max radii
4. Use joint optimization with scipy's L-BFGS-B to refine both positions and radii simultaneously
5. Apply a radius growth schedule to gradually push toward optimal packing
6. Run multiple restarts from different configurations to escape local optima
7. Final refinement with additional iterations

Key insight: The problem is a non-convex optimization with ~325 pairwise distance constraints. A combination of force-based methods (good at escaping local optima) and gradient-based optimization (good at precision) works well.
</strategy>

import numpy as np
from scipy.optimize import minimize

def force_relaxation(centers, radii, n=26, n_iter=1000, dt=0.005):
    """Force-based relaxation to spread circles apart"""
    centers = centers.copy()
    radii = radii.copy()
    
    for iteration in range(n_iter):
        for i in range(n):
            force = np.array([0.0, 0.0])
            # Repulsive forces from other circles
            for j in range(n):
                if i != j:
                    diff = centers[i] - centers[j]
                    dist_sq = np.sum(diff ** 2)
                    dist = np.sqrt(dist_sq)
                    if dist < 0.01:
                        dist = 0.01
                        diff = np.random.randn(2) * 0.01
                    desired_dist = radii[i] + radii[j] + 0.02
                    if dist < desired_dist:
                        repulsion = (desired_dist - dist) / dist
                        force += diff * repulsion * 10.0
            
            # Boundary repulsion
            r = radii[i]
            if centers[i, 0] < r + 0.01:
                force[0] += 5.0
            if centers[i, 0] > 1 - r - 0.01:
                force[0] -= 5.0
            if centers[i, 1] < r + 0.01:
                force[1] += 5.0
            if centers[i, 1] > 1 - r - 0.01:
                force[1] -= 5.0
            
            # Centering force (weak)
            force += (np.array([0.5, 0.5]) - centers[i]) * 0.3
            
            centers[i] += force * dt
            centers[i, 0] = np.clip(centers[i, 0], radii[i] + 1e-8, 1 - radii[i] - 1e-8)
            centers[i, 1] = np.clip(centers[i, 1], radii[i] + 1e-8, 1 - radii[i] - 1e-8)
    
    return centers

def compute_max_radii(centers, n=26):
    """Given centers, compute maximum valid radii"""
    radii = np.zeros(n)
    for i in range(n):
        # Boundary constraints
        r = min(centers[i, 0], 1 - centers[i, 0],
               centers[i, 1], 1 - centers[i, 1])
        # Pairwise constraints
        for j in range(n):
            if i != j:
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                r = min(r, dist / 2 - 1e-10)
        radii[i] = max(0, r)
    return radii

def position_optimization(centers, radii, n=26):
    """Optimize positions given fixed radii"""
    def objective(x):
        c = x.reshape((n, 2))
        penalty = 0.0
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((c[i] - c[j]) ** 2))
                overlap = radii[i] + radii[j] - dist
                if overlap > 0:
                    penalty += overlap ** 2
            # Boundary penalties
            r = radii[i]
            if c[i, 0] < r:
                penalty += (c[i, 0] - r) ** 2 * 100
            if c[i, 0] > 1 - r:
                penalty += (c[i, 0] - (1 - r)) ** 2 * 100
            if c[i, 1] < r:
                penalty += (c[i, 1] - r) ** 2 * 100
            if c[i, 1] > 1 - r:
                penalty += (c[i, 1] - (1 - r)) ** 2 * 100
        return penalty
    
    bounds = []
    for i in range(n):
        bounds.extend([(radii[i] + 1e-6, 1 - radii[i] - 1e-6),
                       (radii[i] + 1e-6, 1 - radii[i] - 1e-6)])
    
    x0 = centers.flatten()
    result = minimize(objective, x0, method='L-BFGS-B', bounds=bounds,
                     options={'maxiter': 500, 'ftol': 1e-15})
    return result.x.reshape((n, 2))

def full_optimization(centers, radii, n=26):
    """Full iterative optimization"""
    for phase in range(3):
        for iteration in range(40):
            radii = compute_max_radii(centers, n)
            centers = position_optimization(centers, radii, n)
        radii = compute_max_radii(centers, n)
    
    return centers, radii

def run_packing():
    n = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Try multiple grid configurations
    configs = [(5, 6), (6, 5), (7, 4), (4, 7), (5, 5)]
    
    for rows, cols in configs:
        if rows * cols < n:
            continue
        
        # Initialize grid
        centers = np.zeros((n, 2))
        x_spacing = 1.0 / (cols + 1)
        y_spacing = 1.0 / (rows + 1)
        
        for idx in range(n):
            r = idx // cols
            c = idx % cols
            centers[idx, 0] = (c + 1) * x_spacing
            centers[idx, 1] = (r + 1) * y_spacing
        
        # Initial radii estimate
        init_r = min(x_spacing, y_spacing) / 2.5
        radii = np.full(n, init_r)
        
        # Force relaxation
        centers = force_relaxation(centers, radii, n, n_iter=2000, dt=0.003)
        radii = compute_max_radii(centers, n)
        
        # Full optimization
        centers, radii = full_optimization(centers, radii, n)
        
        current_sum = np.sum(radii)
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers.copy()
            best_radii = radii.copy()
    
    # Additional refinement from best solution
    centers = best_centers.copy()
    radii = best_radii.copy()
    
    # Extra optimization rounds
    for _ in range(5):
        centers = force_relaxation(centers, radii, n, n_iter=500, dt=0.002)
        radii = compute_max_radii(centers, n)
        centers, radii = full_optimization(centers, radii, n)
    
    # Final radius computation
    radii = compute_max_radii(centers, n)
    final_sum = np.sum(radii)
    
    return centers, radii, final_sum
