import numpy as np

def get_violations(centers, radii):
    """
    Calculate the sum of squared violations for boundary and overlap constraints.
    
    Args:
        centers: np.array of shape (n, 2)
        radii: np.array of shape (n)
        
    Returns:
        float: Sum of squared violations. 0 if valid.
    """
    n = centers.shape[0]
    violations = 0.0
    
    # Boundary violations
    # Constraints: r <= x, r <= 1-x, r <= y, r <= 1-y
    # Violations occur if r > limit. We sum (r - limit)^2 for violations.
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        
        if r > x:
            violations += (r - x)**2
        if r > 1.0 - x:
            violations += (r - (1.0 - x))**2
        if r > y:
            violations += (r - y)**2
        if r > 1.0 - y:
            violations += (r - (1.0 - y))**2
            
    # Overlap violations
    # Constraint: dist >= r_i + r_j
    # Violation if r_i + r_j > dist
    # We sum (r_i + r_j - dist)^2
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt((centers[i, 0] - centers[j, 0])**2 + (centers[i, 1] - centers[j, 1])**2)
            overlap = radii[i] + radii[j] - dist
            if overlap > 0:
                violations += overlap**2
                
    return violations

def objective(params, n, lambda_p):
    """
    Objective function for optimization: Minimize -(Sum of Radii) + lambda * Violations

    sum_r = np.sum(radii)
    
    # We want to maximize sum_r, so we minimize -sum_r
    # Add penalty for violations
    return -sum_r + lambda_p * viol
