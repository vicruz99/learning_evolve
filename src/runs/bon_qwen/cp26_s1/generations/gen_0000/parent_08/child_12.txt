import numpy as np

def compute_max_radii(centers):
    """
    Computes the maximum feasible radii for each circle given fixed centers.
    Uses iterative relaxation to satisfy boundary and non-overlap constraints.
    """
    n = centers.shape[0]
    x, y = centers[:, 0], centers[:, 1]
    
    # Initialize radii with distance to square boundaries
    radii = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
    
    # Precompute pairwise distances efficiently
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    # Iterative shrinkage: r_i <= dist_ij - r_j for all j != i
    # Converges rapidly to the maximal fixed point
    for _ in range(20):
        new_bounds = np.min(dists - radii[np.newaxis, :], axis=1)
        radii = np.minimum(radii, new_bounds)
        radii = np.maximum(radii, 0.0)
        
    return radii
