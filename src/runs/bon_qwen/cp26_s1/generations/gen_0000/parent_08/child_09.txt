import numpy as np

def compute_radii(centers):
    """Compute maximum feasible radii for a given set of centers."""
    n = centers.shape[0]
    radii = np.full(n, 0.5)
    
    # Boundary constraints
    radii = np.minimum(radii, centers[:, 0])
    radii = np.minimum(radii, 1 - centers[:, 0])
    radii = np.minimum(radii, centers[:, 1])
    radii = np.minimum(radii, 1 - centers[:, 1])
    
    # Pairwise constraints
    for i in range(n):
        for j in range(i + 1, n):
            d = np.sqrt(np.sum((centers[i] - centers[j])**2))
            r_val = d * 0.5
            if r_val < radii[i]:
                radii[i] = r_val
            if r_val < radii[j]:
                radii[j] = r_val
    return radii
