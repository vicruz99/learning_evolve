import numpy as np
from scipy.optimize import minimize


def boundary_constraint_left(x, n, i):
    return x[2 * i] - x[2 * n + i]


def boundary_constraint_right(x, n, i):
    return 1.0 - x[2 * i] - x[2 * n + i]


def boundary_constraint_bottom(x, n, i):
    return x[2 * i + 1] - x[2 * n + i]


def boundary_constraint_top(x, n, i):
    return 1.0 - x[2 * i + 1] - x[2 * n + i]


def radius_nonneg(x, n, i):
    return x[2 * n + i]


def overlap_constraint(x, n, i, j):
    dx = x[2 * i] - x[2 * j]
    dy = x[2 * i + 1] - x[2 * j + 1]
    dist = np.sqrt(dx * dx + dy * dy)
    return dist - x[2 * n + i] - x[2 * n + j]


def objective_neg_sum(x, n):
    return -np.sum(x[2 * n:])


def init_hexagonal(n):
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    r_init = 0.09
    row_counts = [6, 5, 6, 5, 4]
    idx = 0
    for row_idx, count in enumerate(row_counts):
        y = 0.1 + row_idx * np.sqrt(3) * r_init
        x_offset = r_init if row_idx % 2 == 1 else 0
        for col_idx in range(count):
            x = x_offset + col_idx * 2 * r_init
            centers[idx] = [x, y]
            radii[idx] = r_init
            idx += 1
            if idx >= n:
                break
        if idx >= n:
            break
    return centers, radii


def init_grid_plus_one(n):
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    grid_r = 0.099
    for i in range(5):
        for j in range(5):
            idx = i * 5 + j
            centers[idx] = [0.05 + grid_r + i * 2 * grid_r, 0.05 + grid_r + j * 2 * grid_r]
            radii[idx] = grid_r
    centers[25] = [0.5, 0.5]
    radii[25] = 0.03
    return centers, radii


def expand_and_resolve(centers, radii, n, expansion_steps=800, resolution_steps=150):
    centers = centers.copy()
    radii = radii.copy()
    expansion_rate = 0.00008
    for step in range(expansion_steps):
        radii += expansion_rate
        for res_step in range(resolution_steps):
            max_overlap = 0.0
            max_i, max_j = -1, -1
            for i in range(n):
                for j in range(i + 1, n):
                    dx = centers[i, 0] - centers[j, 0]
                    dy = centers[i, 1] - centers[j, 1]
                    dist = np.sqrt(dx * dx + dy * dy)
                    overlap = radii[i] + radii[j] - dist
                    if overlap > max_overlap:
                        max_overlap = overlap
                        max_i, max_j = i, j
            if max_overlap < 1e-12:
                break
            i, j = max_i, max_j
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            if dist > 1e-12:
                push = (max_overlap / 2.0) + 1e-7
                centers[i, 0] += push * dx / dist
                centers[i, 1] += push * dy / dist
                centers[j, 0] -= push * dx / dist
                centers[j, 1] -= push * dy / dist
        for k in range(n):
            for d in range(2):
                if centers[k, d] < radii[k]:
                    centers[k, d] = radii[k]
                if centers[k, d] > 1.0 - radii[k]:
                    centers[k, d] = 1.0 - radii[k]
    return centers, radii


def build_constraints(n):
    constraints = []
    for i in range(n):
        constraints.append({
            'type': 'ineq',
            'fun': boundary_constraint_left,
            'args': (n, i)
        })
        constraints.append({
            'type': 'ineq',
            'fun': boundary_constraint_right,
            'args': (n, i)
        })
        constraints.append({
            'type': 'ineq',
            'fun': boundary_constraint_bottom,
            'args': (n, i)
        })
        constraints.append({
            'type': 'ineq',
            'fun': boundary_constraint_top,
            'args': (n, i)
        })
        constraints.append({
            'type': 'ineq',
            'fun': radius_nonneg,
            'args': (n, i)
        })
    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({
                'type': 'ineq',
                'fun': overlap_constraint,
                'args': (n, i, j)
            })
    return constraints


def build_bounds(n):
    bounds = []
    for _ in range(2 * n):
        bounds.append((0.0, 1.0))
    for _ in range(n):
        bounds.append((0.0, 0.5))
    return bounds


def slsqp_optimize(centers, radii, n, max_iter=3000):
    x0 = np.concatenate([centers.flatten(), radii])
    constraints = build_constraints(n)
    bounds = build_bounds(n)
    result = minimize(
        objective_neg_sum,
        x0,
        args=(n,),
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': max_iter, 'ftol': 1e-15}
    )
    opt_centers = result.x[:2 * n].reshape(n, 2)
    opt_radii = result.x[2 * n:]
    return opt_centers, opt_radii


def try_alternative_row_patterns(n):

        [6, 5, 6, 5, 4],
        [5, 6, 5, 6, 4],
        [4, 6, 6, 6, 4],
        [5, 5, 5, 5, 6],

        [4, 7, 6, 6, 3],
    ]
    best_centers = None
    best_radii = None
    best_sum = -1.0
    for pattern in patterns:
        if sum(pattern) != n:
            continue
        centers = np.zeros((n, 2))
        radii = np.zeros(n)
        r_init = 0.095
        idx = 0
        for row_idx, count in enumerate(pattern):
            y = 0.08 + row_idx * np.sqrt(3) * r_init
            x_offset = r_init if row_idx % 2 == 1 else 0
            for col_idx in range(count):
                if idx >= n:
                    break
                x = x_offset + col_idx * 2 * r_init
                centers[idx] = [x, y]
                radii[idx] = r_init
                idx += 1
        if idx < n:
            continue
        centers, radii = expand_and_resolve(centers, radii, n, expansion_steps=600, resolution_steps=120)
        cur_sum = np.sum(radii)
        if cur_sum > best_sum:
            best_sum = cur_sum
            best_centers = centers.copy()
            best_radii = radii.copy()
    return best_centers, best_radii, best_sum
