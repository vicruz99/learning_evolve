import numpy as np
import scipy.optimize as opt

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    n = centers.shape[0]

    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True

def compute_objective_and_penalty(params, n):
    """
    Computes the objective (negative sum of radii) and a penalty for constraint violations.
    params: flat array of [x1, y1, r1, x2, y2, r2, ...]
    """
    centers = params[:2*n].reshape(n, 2)
    radii = params[2*n:]
    
    # Objective: maximize sum of radii -> minimize negative sum
    objective = -np.sum(radii)
    
    penalty = 0.0
    penalty_weight = 1000.0
    
    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        # Left
        if x - r < 0:
            penalty += penalty_weight * (x - r)**2
        # Right
        if x + r > 1:
            penalty += penalty_weight * (x + r - 1)**2
        # Bottom
        if y - r < 0:
            penalty += penalty_weight * (y - r)**2
        # Top
        if y + r > 1:
            penalty += penalty_weight * (y + r - 1)**2
            
    # Overlap constraints
    # To improve performance, we only compute pairwise for nearby circles or all if n is small.
    # n=26 is small enough for O(n^2).
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx*dx + dy*dy)
            sum_r = radii[i] + radii[j]
            if dist < sum_r:
                overlap = sum_r - dist
                penalty += penalty_weight * overlap**2
                
    return objective + penalty, radii

def run_local_optimization(initial_centers, initial_radii):
    n = len(initial_radii)
    initial_params = np.concatenate([initial_centers.flatten(), initial_radii])
    
    # Bounds: x, y in [0, 1], r >= 0
    bounds = []
    for i in range(n):
        bounds.extend([(0, 1), (0, 1), (0, None)]) # x, y, r
        
    # We use L-BFGS-B. It handles bounds well.
    # We define a wrapper for the objective that takes the flat array.
    def objective_func(params):
        obj, _ = compute_objective_and_penalty(params, n)
        return obj

    try:
        result = opt.minimize(objective_func, initial_params, method='L-BFGS-B', bounds=bounds, options={'maxiter': 1000, 'ftol': 1e-9})
        return result.x
    except Exception:
        return initial_params

def generate_5x5_plus_1():
    """
    Generates a 5x5 grid with radius 0.1 and an extra circle in a gap.
    """
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    idx = 0
    # 5x5 grid
    coords = [0.1, 0.3, 0.5, 0.7, 0.9]
    for x in coords:
        for y in coords:
            centers[idx] = [x, y]
            radii[idx] = 0.1
            idx += 1
    
    # Place 26th circle in a gap, e.g., at (0.2, 0.2)
    # Gap between (0.1, 0.1), (0.3, 0.1), (0.1, 0.3), (0.3, 0.3)
    # Center at (0.2, 0.2). Dist to (0.1, 0.1) is sqrt(0.02) approx 0.1414.
    # r_gap approx 0.1414 - 0.1 = 0.0414.
    centers[idx] = [0.2, 0.2]
    radii[idx] = 0.04
    
    return centers, radii

def generate_hexagonal_pattern():

    idx = 0

        # For count=4 (shifted): 0.2, 0.4, 0.6, 0.8 (OK)

            if shift == 0.0:
                x = r + k * (2 * r)
            else:
                x = 2 * r + k * (2 * r) # Shifted by r from start?

                # Yes.
            
            centers[idx] = [x, y_pos]
            radii[idx] = r
            idx += 1
            
        # Update y for next row
        y_pos += r * np.sqrt(3)
        
        # Toggle shift.

        # But we need to be careful with width.

        # Shifted row of 4 circles fits.
        # Shifted row of 5 circles would need width 1.1 (fail).
        # So alternating 5 and 4 works perfectly with shift.
        
        if shift == 0.0:
            shift = r
        else:
            shift = 0.0
            
    return centers, radii
