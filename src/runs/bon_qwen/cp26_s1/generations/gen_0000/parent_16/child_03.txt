import numpy as np
from scipy.optimize import minimize

def objective(vars):
    """Objective function: negative sum of radii (to be minimized)"""
    return -np.sum(vars[2::3])

def con_x_min(vars, i):
    """Constraint: x_i - r_i >= 0"""
    return vars[3*i] - vars[3*i+2]

def con_x_max(vars, i):
    """Constraint: 1 - x_i - r_i >= 0"""
    return 1.0 - vars[3*i] - vars[3*i+2]

def con_y_min(vars, i):
    """Constraint: y_i - r_i >= 0"""
    return vars[3*i+1] - vars[3*i+2]

def con_y_max(vars, i):
    """Constraint: 1 - y_i - r_i >= 0"""
    return 1.0 - vars[3*i+1] - vars[3*i+2]

def con_overlap(vars, pair):
    """Constraint: dist(i,j) - r_i - r_j >= 0"""
    i, j = pair
    xi, yi, ri = vars[3*i], vars[3*i+1], vars[3*i+2]
    xj, yj, rj = vars[3*j], vars[3*j+1], vars[3*j+2]
    return np.sqrt((xi-xj)**2 + (yi-yj)**2) - ri - rj

def run_packing():
    n = 26
    
    # Generate initial configuration using a hexagonal pattern
    centers_init = []
    y = 0.1
    row_idx = 0
    # Pattern to generate ~26 circles
    while len(centers_init) < 30:
        count = 6 if row_idx % 2 == 0 else 5
        xs = np.linspace(0.1, 0.9, count)
        if row_idx % 2 != 0:
            spacing = xs[1] - xs[0]
            xs = xs + spacing / 2.0
            xs = np.clip(xs, 0.1, 0.9)
        for x in xs:
            centers_init.append([x, y])
        y += 0.2
        row_idx += 1
        
    centers_init = np.array(centers_init[:n])
    radii_init = np.full(n, 0.08)
    x0 = np.concatenate([centers_init.ravel(), radii_init])
    
    # Setup constraints
    constraints = []
    for i in range(n):
        constraints.append({'type': 'ineq', 'fun': con_x_min, 'args': (i,)})
        constraints.append({'type': 'ineq', 'fun': con_x_max, 'args': (i,)})
        constraints.append({'type': 'ineq', 'fun': con_y_min, 'args': (i,)})
        constraints.append({'type': 'ineq', 'fun': con_y_max, 'args': (i,)})
        
    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({'type': 'ineq', 'fun': con_overlap, 'args': ((i, j),)})
            
    # Bounds: x, y in [0,1], r in [0, 0.5]
    bounds = [(0, 1) for _ in range(n)]
    bounds += [(0, 1) for _ in range(n)]
    bounds += [(0, 0.5) for _ in range(n)]
    
    # Optimization
    try:
        res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints, 
                       options={'maxiter': 3000, 'disp': False})
        best_vars = res.x
    except Exception:
        best_vars = x0
        
    centers = best_vars[:2*n].reshape(n, 2)
    radii = best_vars[2*n:]
    
    # Ensure validity
    radii = np.maximum(radii, 0.0)
    
    return centers, radii, np.sum(radii)
