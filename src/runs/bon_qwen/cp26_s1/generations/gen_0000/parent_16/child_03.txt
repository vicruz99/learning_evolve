import numpy as np
from scipy.optimize import linprog, minimize

def get_boundary_limits(centers):
    """Calculates the maximum possible radius for each circle based on square boundaries."""
    n = centers.shape[0]
    limits = np.full(n, 1.0)
    for i in range(n):
        x, y = centers[i]
        limits[i] = min(x, 1.0 - x, y, 1.0 - y)
    return limits

def solve_radii_lp(centers, boundary_limits):
    """Solves for optimal radii using Linear Programming."""
    n = centers.shape[0]
    
    # Objective: maximize sum(r) -> minimize sum(-r)
    c_obj = -np.ones(n)
    
    # Constraints: A_ub @ r <= b_ub
    # 1. Boundary constraints: r_i <= limit_i
    A_ub = np.zeros((n, n))
    for i in range(n):
        A_ub[i, i] = 1.0
    b_ub = boundary_limits
    
    # 2. Non-overlap constraints: r_i + r_j <= dist_ij
    # Add these to the matrix
    # We flatten the pairs
    pairs = []
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            pairs.append((i, j, dist))
    
    if pairs:
        n_pairs = len(pairs)
        A_overlap = np.zeros((n_pairs, n))
        b_overlap = np.zeros(n_pairs)
        for idx, (i, j, d) in enumerate(pairs):
            A_overlap[idx, i] = 1.0
            A_overlap[idx, j] = 1.0
            b_overlap[idx] = d
        
        # Combine constraints
        A_ub = np.vstack([A_ub, A_overlap])
        b_ub = np.concatenate([b_ub, b_overlap])
    
    # Bounds for radii: r_i >= 0
    bounds = [(0, None) for _ in range(n)]
    
    # Solve LP
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return -res.fun, -res.x
        else:
            # Fallback to small radii if LP fails
            return 0, np.zeros(n) * 0.01
    except Exception:
        return 0, np.zeros(n) * 0.01

def optimize_centers(initial_centers):
    """Optimizes centers to maximize sum of radii."""
    n = initial_centers.shape[0]
    
    def objective(vars):
        centers = vars.reshape((n, 2))
        # Soft constraint to keep centers inside [0,1]
        # We clamp for calculation but optimization handles bounds
        limits = get_boundary_limits(centers)
        obj, radii = solve_radii_lp(centers, limits)
        # Return negative because we want to maximize
        return -obj

    def gradient(vars):
        centers = vars.reshape((n, 2))
        limits = get_boundary_limits(centers)
        base_obj, _ = solve_radii_lp(centers, limits)
        
        grad = np.zeros_like(vars)
        epsilon = 1e-4
        
        for i in range(n):
            for k in range(2):
                # Perturb center
                center_plus = centers[i].copy()
                center_plus[k] += epsilon
                
                centers_pert = centers.copy()
                centers_pert[i] = center_plus
                
                limits_pert = get_boundary_limits(centers_pert)
                pert_obj, _ = solve_radii_lp(centers_pert, limits_pert)
                
                # Numerical derivative
                # Note: we are maximizing, so derivative of -obj
                grad[2*i + k] = (-(pert_obj) - (-base_obj)) / epsilon
                
        return grad

    # Bounds for centers [0, 1]
    bounds = [(0, 1) for _ in range(n * 2)]
    
    try:
        result = minimize(objective, initial_centers.flatten(), 
                         jac=gradient, 
                         bounds=bounds, 
                         method='L-BFGS-B',
                         options={'maxiter': 500, 'ftol': 1e-8, 'gtol': 1e-8})
        if result.success:
            return result.x.reshape((n, 2))
    except Exception:
        pass
        
    return initial_centers

def run_packing():
    n = 26
    best_centers = None
    best_radii = None
    max_sum_radii = 0
    
    # Multiple restarts to find global optimum
    restarts = 20
    for restart in range(restarts):
        # Initialize with hexagonal packing
        centers = np.zeros((n, 2))
        
        # Use a radius slightly smaller than 0.1 to ensure initial feasibility
        r_init = 0.08
        x, y = 0, 0
        count = 0
        
        # Generate hexagonal grid
        while count < n:
            row = int(y / (r_init * np.sqrt(3)))
            col_x = x
            if row % 2 == 1:
                col_x = r_init # Offset for hexagonal packing
            
            while col_x <= 1 and count < n:
                centers[count] = [col_x, y]
                count += 1
                col_x += 2 * r_init
            y += r_init * np.sqrt(3)
        
        # Add random perturbation to escape local minima
        if restart > 0:
            centers += np.random.uniform(-0.01, 0.01, size=centers.shape)
            # Clip to valid range
            centers = np.clip(centers, 0.05, 0.95)
            
        # Optimize centers
        opt_centers = optimize_centers(centers)
        
        # Solve for optimal radii
        limits = get_boundary_limits(opt_centers)
        sum_r, radii = solve_radii_lp(opt_centers, limits)
        
        if sum_r > max_sum_radii:
            max_sum_radii = sum_r
            best_centers = opt_centers.copy()
            best_radii = radii.copy()
            
    return best_centers, best_radii, max_sum_radii
