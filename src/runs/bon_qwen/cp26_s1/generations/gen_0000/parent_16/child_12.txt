So the optimizer feels no force from the constraint at the exact boundary.

```python
import numpy as np
import scipy.optimize as opt

def packing_objective(params, n, mask, penalty_weight):
    """
    Objective function for circle packing optimization.
    Minimizes negative sum of radii plus penalties for constraint violations.
    
    Args:
        params: Array containing [x1, y1, ..., xn, yn, r1, ..., rn]
        n: Number of circles
        mask: Boolean mask for unique pairs (upper triangle)
        penalty_weight: Weight for the penalty term
    
    Returns:
        Scalar value to minimize
    """
    centers = params[:2*n].reshape(n, 2)
    radii = params[2*n:]
    
    # Objective: Maximize sum of radii (minimize negative sum)
    val = -np.sum(radii)
    
    penalty = 0.0
    
    # --- Boundary Constraints ---
    # Circle i must satisfy: r_i <= x_i <= 1 - r_i and r_i <= y_i <= 1 - r_i
    
    # Left boundary: x_i >= r_i  => violation if r_i - x_i > 0
    viol = np.maximum(0, radii - centers[:, 0])
    penalty += np.sum(viol**2)
    
    # Right boundary: x_i <= 1 - r_i  => x_i + r_i <= 1 => violation if x_i + r_i - 1 > 0
    viol = np.maximum(0, centers[:, 0] + radii - 1.0)
    penalty += np.sum(viol**2)
    
    # Bottom boundary: y_i >= r_i
    viol = np.maximum(0, radii - centers[:, 1])
    penalty += np.sum(viol**2)
    
    # Top boundary: y_i <= 1 - r_i
    viol = np.maximum(0, centers[:, 1] + radii - 1.0)
    penalty += np.sum(viol**2)
    
    # --- Overlap Constraints ---
    # Distance between centers i and j must be >= r_i + r_j
    # Violation if r_i + r_j - dist > 0
    
    # Vectorized distance calculation
    # centers shape: (n, 2)
    # diff shape: (n, n, 2)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    # Sum of radii matrix
    radii_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    
    # Calculate overlap violation
    ov = np.maximum(0, radii_sum - dists)
    
    # Apply mask to consider only unique pairs (i < j)
    ov = ov * mask
    
    penalty += np.sum(ov**2)
    
    return val + penalty_weight * penalty

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Pack 26 circles in a unit square [0,1]x[0,1] to maximize the sum of radii.
    
    Strategy:
    1. Initialize circles in a hexagonal lattice pattern with a small feasible radius.
    2. Use an iterative penalty method with L-BFGS-B optimization.
    3. The objective function minimizes -sum(radii) + weight * (boundary_violations + overlap_violations).
    4. The penalty weight is increased iteratively to enforce constraints strictly while allowing radii to grow.
    
    Returns:
        (centers, radii, sum_radii)
    """
    n = 26
    
    # Precompute the upper triangle mask for overlap checks (i < j)
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    
    # --- Initialization ---
    # Start with a hexagonal packing of radius 0.08.
    # This is safe (diameter 0.16) and ensures a valid starting point.
    r_init = 0.08
    centers = []
    
    # Generate hexagonal points
    y = r_init
    row = 0
    while len(centers) < n:
        x = r_init
        # Shift odd rows by r_init to create hexagonal structure
        # Actually, standard hex shift is r_init horizontally relative to previous row?
        # In hex packing, horizontal spacing is 2r, vertical is r*sqrt(3).
        # Row 0 centers at x = r, 3r, 5r...
        # Row 1 centers at x = 2r, 4r, 6r... (shifted by r)
        if row % 2 == 1:
            x = 2 * r_init
        
        while x + r_init <= 1.0 + 1e-9:
            centers.append([x, y])
            if len(centers) >= n:
                break
            x += 2 * r_init
        
        y += r_init * np.sqrt(3)
        row += 1
    
    centers = np.array(centers[:n])
    radii = np.ones(n) * r_init
    
    # Flatten variables: [x1, y1, ..., xn, yn, r1, ..., rn]
    x0 = np.concatenate([centers.flatten(), radii])
    
    # Bounds: x, y in [0, 1], r in [0, 0.5]
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
    
    # --- Optimization ---
    # We iteratively increase the penalty weight to force the solution to respect constraints
    # while the optimizer tries to maximize the sum of radii.
    current_params = x0
    weight = 100.0
    
    # Run optimization for a fixed number of steps
    for _ in range(40):
        res = opt.minimize(
            packing_objective,
            current_params,
            args=(n, mask, weight),
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 5000, 'ftol': 1e-16}
        )
        current_params = res.x
        # Increase penalty weight to tighten constraints
        weight *= 2.0
        
    final_centers = current_params[:2*n].reshape(n, 2)
    final_radii = current_params[2*n:]
    
    sum_radii = float(np.sum(final_radii))
    
    return final_centers, final_radii, sum_radii
```
