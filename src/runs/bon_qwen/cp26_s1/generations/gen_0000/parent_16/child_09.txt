def get_constraints(centers, radii):
    """
    Computes the constraint violations for a given packing.
    Returns a list of constraint values. 
    For equality constraints, val = 0. For inequality, val >= 0.
    Here we return g(x) >= 0.
    """
    n = centers.shape[0]
    constraints = []
    
    # Boundary constraints: r <= x <= 1-r  =>  x - r >= 0, 1 - x - r >= 0
    # Same for y
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        
        # Left boundary
        constraints.append(x - r)
        # Right boundary
        constraints.append((1.0 - x) - r)
        # Bottom boundary
        constraints.append(y - r)
        # Top boundary
        constraints.append((1.0 - y) - r)
        
    # Non-overlap constraints: dist >= r_i + r_j => dist^2 - (r_i+r_j)^2 >= 0
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist_sq = dx*dx + dy*dy
            r_sum = radii[i] + radii[j]
            constraints.append(dist_sq - r_sum**2)
            
    return np.array(constraints)
