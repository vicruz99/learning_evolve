def get_distance_matrix(centers):
    """Compute pairwise distance matrix."""
    # centers shape (n, 2)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    return dists

def compute_penalty(centers, radii, alpha):
    """
    Compute penalty for overlaps and boundary violations.
    Returns a scalar penalty value.
    """
    n = len(radii)
    penalty = 0.0
    
    # Boundary violations
    # Circle i must be inside [0,1]x[0,1]
    # x - r >= 0 => r - x <= 0
    # x + r <= 1 => x + r - 1 <= 0
    # Same for y

    # Left: r - x
    # Right: x + r - 1
    # Bottom: r - y
    # Top: y + r - 1
    
    left_viol = np.maximum(0, radii - centers[:, 0])
    right_viol = np.maximum(0, centers[:, 0] + radii - 1)
    bottom_viol = np.maximum(0, radii - centers[:, 1])
    top_viol = np.maximum(0, centers[:, 1] + radii - 1)
    
    boundary_violations = left_viol**2 + right_viol**2 + bottom_viol**2 + top_viol**2
    penalty += np.sum(boundary_violations)
    
    # Overlap violations
    # dist(i,j) >= r_i + r_j
    # violation = max(0, r_i + r_j - dist(i,j))
    
    # Vectorized overlap check
    # radii shape (n,), centers (n, 2)

    dists = get_distance_matrix(centers)

    overlap = np.maximum(0, sum_radii_matrix - dists)
    
    # Sum of squared overlaps (upper triangle to avoid double counting)
    penalty += np.sum(np.triu(overlap**2))
    
    return alpha * penalty
