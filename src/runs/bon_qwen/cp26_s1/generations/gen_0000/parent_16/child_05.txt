import numpy as np
from scipy.optimize import minimize


def make_constraints_and_objective(n):
    """Create objective and constraint functions for circle packing optimization."""

    def objective(p):
        r = p[2 * n:]
        return -np.sum(r)

    def boundary_constraints(p):
        c = p[:2 * n].reshape(n, 2)
        r = p[2 * n:]
        vals = []
        for i in range(n):
            vals.append(c[i, 0] - r[i])
            vals.append(1.0 - c[i, 0] - r[i])
            vals.append(c[i, 1] - r[i])
            vals.append(1.0 - c[i, 1] - r[i])
        return np.array(vals)

    def overlap_constraints(p):
        c = p[:2 * n].reshape(n, 2)
        r = p[2 * n:]
        vals = []
        for i in range(n):
            for j in range(i + 1, n):
                dx = c[i, 0] - c[j, 0]
                dy = c[i, 1] - c[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                vals.append(dist - r[i] - r[j])
        return np.array(vals)

    cons = [
        {"type": "ineq", "fun": boundary_constraints},
        {"type": "ineq", "fun": overlap_constraints},
    ]
    bounds = [(0.0, 1.0)] * (2 * n) + [(1e-8, 0.5)] * n
    return objective, cons, bounds


def create_hexagonal_config(n, row_counts, radii_init):
    """Create a hexagonal-like initial configuration."""
    centers = np.zeros((n, 2))
    radii = np.full(n, radii_init)
    idx = 0
    num_rows = len(row_counts)
    for row in range(num_rows):
        count = row_counts[row]
        y = (row + 0.5) / num_rows
        if row % 2 == 0:
            for col in range(count):
                x = (col + 0.5) / count
                centers[idx] = [x, y]
                idx += 1
        else:
            for col in range(count):
                x = (col + 1) / (count + 1)
                centers[idx] = [x, y]
                idx += 1
    return centers, radii


def create_staggered_config(n, row_counts, radii_init):
    """Create a staggered initial configuration with better spacing."""
    centers = np.zeros((n, 2))
    radii = np.full(n, radii_init)
    idx = 0
    num_rows = len(row_counts)
    for row in range(num_rows):
        count = row_counts[row]
        y = (row + 0.5) / num_rows
        margin = radii_init * (1.0 if row % 2 == 0 else 0.5)
        width = 1.0 - 2 * margin
        for col in range(count):
            x = margin + col * (width / max(count - 1, 1)) if count > 1 else 0.5
            centers[idx] = [x, y]
            idx += 1
    return centers, radii


def validate_and_fix(centers, radii, n):
    """Validate packing and fix any numerical violations."""
    centers = centers.copy()
    radii = radii.copy()

    # Fix boundary violations by moving centers and shrinking radii
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < 0:
            r = x
            radii[i] = r
        if x + r > 1:
            r = 1 - x
            radii[i] = r
        if y - r < 0:
            r = y
            radii[i] = r
        if y + r > 1:
            r = 1 - y
            radii[i] = r
        centers[i] = [x, y]

    # Fix overlap violations by shrinking radii
    for iteration in range(10):
        max_violation = 0
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                overlap = radii[i] + radii[j] - dist
                if overlap > 1e-10:
                    max_violation = max(max_violation, overlap)
                    shrink = overlap / 2 + 1e-12
                    radii[i] = max(1e-8, radii[i] - shrink)
                    radii[j] = max(1e-8, radii[j] - shrink)
        if max_violation < 1e-12:
            break

    return centers, radii


def run_optimization(n, centers, radii, objective, cons, bounds, max_iter=2000):
    """Run a single optimization from given initial conditions."""
    params = np.concatenate([centers.flatten(), radii])
    try:
        result = minimize(
            objective,
            params,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": max_iter, "ftol": 1e-14, "iprint": -1},
        )
        return result
    except Exception:
        return None
