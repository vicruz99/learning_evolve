import numpy as np
from scipy.optimize import minimize

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True

def get_initial_guess(n_circles):

def objective(vars, n_circles):
    """
    Objective function: minimize negative sum of radii.
    vars: [x1, y1, r1, x2, y2, r2, ...]
    """
    radii = vars[2::3]
    return -np.sum(radii)

def get_constraints(n_circles):
    """
    Returns a list of constraint dictionaries for scipy.optimize.minimize.
    Constraints:
    1. r_i >= 0
    2. x_i - r_i >= 0
    3. x_i + r_i <= 1
    4. y_i - r_i >= 0
    5. y_i + r_i <= 1
    6. (x_i - x_j)^2 + (y_i - y_j)^2 >= (r_i + r_j)^2  =>  (x_i-x_j)^2 + ... - (r_i+r_j)^2 >= 0
    """
    constraints = []
    
    # Variable mapping: 
    # index i: x -> 3*i, y -> 3*i+1, r -> 3*i+2
    
    # Boundary and non-negativity constraints (linear)
    for i in range(n_circles):
        idx_x = 3 * i
        idx_y = 3 * i + 1
        idx_r = 3 * i + 2
        
        # r_i >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: v[idx_r],
            'jac': lambda v, i=i: np.zeros(3 * n_circles).__setitem__(idx_r, 1) or np.zeros(3 * n_circles) # This is tricky with lambda closure
            # Better to define jacobian explicitly or let scipy approximate. 
            # For simplicity and robustness, we might omit jac or define it carefully.
            # But defining a generic jacobian function is hard with loop indices.
            # Let's rely on finite difference for jacobian or define simple callbacks.
        })
        
        # x_i - r_i >= 0
        def con_x_minus_r(v, i=i):
            return v[3*i] - v[3*i+2]
        constraints.append({'type': 'ineq', 'fun': con_x_minus_r})
        
        # x_i + r_i <= 1  =>  1 - x_i - r_i >= 0
        def con_1_minus_x_minus_r(v, i=i):
            return 1.0 - v[3*i] - v[3*i+2]
        constraints.append({'type': 'ineq', 'fun': con_1_minus_x_minus_r})
        
        # y_i - r_i >= 0
        def con_y_minus_r(v, i=i):
            return v[3*i+1] - v[3*i+2]
        constraints.append({'type': 'ineq', 'fun': con_y_minus_r})
        
        # y_i + r_i <= 1  =>  1 - y_i - r_i >= 0
        def con_1_minus_y_minus_r(v, i=i):
            return 1.0 - v[3*i+1] - v[3*i+2]
        constraints.append({'type': 'ineq', 'fun': con_1_minus_y_minus_r})

    # Non-overlap constraints (nonlinear)
    for i in range(n_circles):
        for j in range(i + 1, n_circles):
            idx_xi, idx_yi, idx_ri = 3*i, 3*i+1, 3*i+2
            idx_xj, idx_yj, idx_rj = 3*j, 3*j+1, 3*j+2
            
            def con_overlap(v, i=i, j=j, xi=idx_xi, yi=idx_yi, ri=idx_ri, xj=idx_xj, yj=idx_yj, rj=idx_rj):
                dx = v[xi] - v[xj]
                dy = v[yi] - v[yj]
                dist_sq = dx*dx + dy*dy
                sum_r = v[ri] + v[rj]
                return dist_sq - sum_r * sum_r
            
            constraints.append({'type': 'ineq', 'fun': con_overlap})
            
    return constraints
