import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def objective_func(t):
    """Maximize sum of radii (minimize negative sum)"""
    return -np.sum(t[:N_CIRCLES])

def constraint_func(t):
    """
    Returns array of constraint values. All must be >= 0.
    Includes boundary, non-negative radius, and pairwise non-overlap constraints.
    """
    N = N_CIRCLES
    r = t[:N]
    x = t[N:2*N]
    y = t[2*N:3*N]
    
    c = []
    # Non-negative radii
    c.append(r)
    # Left & Right boundaries
    c.append(x - r)
    c.append(1.0 - x - r)
    # Bottom & Top boundaries
    c.append(y - r)
    c.append(1.0 - y - r)
    
    # Pairwise distance constraints
    # Vectorized computation of all pairwise distances
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dists = np.sqrt(dx**2 + dy**2)
    
    # Extract strictly lower triangle indices
    i, j = np.tril_indices(N, -1)
    c.append(dists[i, j] - r[i] - r[j])
    
    return np.concatenate(c)

def get_bounds():
    """Variable bounds: r in [0, 0.5], x,y in [0, 1]"""
    return [(0, 0.5)]*N_CIRCLES + [(0, 1.0)]*N_CIRCLES + [(0, 1.0)]*N_CIRCLES

def run_packing():
    N = N_CIRCLES
    best_val = -1.0
    best_t = None
    
    # Multiple restarts to escape local minima
    for seed in range(12):
        np.random.seed(seed)
        
        # Generate hexagonal lattice initial positions
        pts = []
        y = 0.1
        while y <= 0.9 and len(pts) < N:
            x = 0.1
            if len(pts) % 2 == 1:
                x = 0.15  # Shift odd rows for hex pattern
            while x <= 0.9 and len(pts) < N:
                pts.append([x, y])
                x += 0.15
            y += 0.13
        pts = np.array(pts[:N])
        
        # Assemble initial parameter vector
        t0 = np.zeros(3*N)
        t0[:N] = 0.06  # Initial small radii
        t0[N:2*N] = pts[:, 0]
        t0[2*N:3*N] = pts[:, 1]
        
        # Add controlled noise to break symmetry
        t0[N:] += np.random.randn(2*N) * 0.008
        t0[N:] = np.clip(t0[N:], 0.06, 0.94)
        
        bnds = get_bounds()
        cons = {'type': 'ineq', 'fun': constraint_func}
        
        # Optimize
        res = minimize(
            objective_func, 
            t0, 
            method='SLSQP', 
            bounds=bnds, 
            constraints=cons, 
            options={'maxiter': 5000, 'ftol': 1e-13, 'disp': False}
        )
        
        if -res.fun > best_val:
            best_val = -res.fun
            best_t = res.x.copy()
            
    # Extract best solution
    centers = best_t[N:3*N].reshape((N, 2))
    radii = best_t[:N]
    
    # Final safety adjustments to guarantee validation passes
    radii = np.maximum(radii, 1e-9)
    centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
    centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)
    
    # Resolve any residual microscopic overlaps from numerical optimization
    for _ in range(5):
        for i in range(N):
            for j in range(i+1, N):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                sum_r = radii[i] + radii[j]
                if dist < sum_r - 1e-12:
                    # Separate centers proportionally
                    if dist > 1e-12:
                        shift = (sum_r - dist + 1e-10) / 2.0
                        nx, ny = dx/dist, dy/dist
                        centers[i, 0] += nx * shift
                        centers[i, 1] += ny * shift
                        centers[j, 0] -= nx * shift
                        centers[j, 1] -= ny * shift
    
    # Re-clip after shifting
    centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
    centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)
    
    return centers, radii, np.sum(radii)
