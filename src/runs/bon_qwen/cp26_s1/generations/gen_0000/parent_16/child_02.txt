import numpy as np
from scipy.optimize import linprog, differential_evolution

def compute_radii(centers):
    """
    Solves the LP to find maximum radii for a fixed set of centers.
    Maximize sum(r) subject to r_i + r_j <= dist(i,j) and boundary constraints.
    """
    n = centers.shape[0]
    
    # Boundary constraints: r_i <= distance to nearest wall
    x = centers[:, 0]
    y = centers[:, 1]
    b = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
    b = np.maximum(b, 0.0)
    
    n_pairs = n * (n - 1) // 2
    n_constraints = n_pairs + n
    A = np.zeros((n_constraints, n))
    ub = np.zeros(n_constraints)
    
    idx = 0
    # Pairwise distance constraints
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            ub[idx] = dist
            idx += 1
            
    # Boundary constraints
    for i in range(n):
        A[idx, i] = 1.0
        ub[idx] = b[i]
        idx += 1
        
    # Solve LP: minimize -sum(r) equivalent to maximize sum(r)
    c_obj = -np.ones(n)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=ub, bounds=bounds, method='highs')
    
    if res.success:
        return -res.fun, res.x
    return 0.0, np.zeros(n)

def packing_objective(centers_flat):
    """Objective function for the optimizer: returns negative sum of radii."""
    centers = centers_flat.reshape(26, 2)
    s, _ = compute_radii(centers)
    return -s

def run_packing():
    np.random.seed(42)
    n_circles = 26
    dim = n_circles * 2
    bounds = [(0.0, 1.0)] * dim
    
    # Differential Evolution handles the non-convex center placement robustly.
    # polish=True performs a local refinement at the end.
    try:
        de_res = differential_evolution(
            packing_objective, 
            bounds, 
            maxiter=2500, 
            popsize=15, 
            seed=42, 
            tol=1e-9, 
            polish=True,
            mutation=(0.5, 1.0),
            recombination=0.9
        )
        best_centers = de_res.x.reshape(n_circles, 2)
    except Exception:
        # Fallback to a dense grid if optimization fails
        best_centers = np.random.uniform(0.1, 0.9, (n_circles, 2))
        
    # Compute optimal radii for the best centers found
    sum_r, radii = compute_radii(best_centers)
    
    # Apply a tiny shrinkage to guarantee strict feasibility against 1e-12 tolerance
    radii *= 0.99999
    
    return best_centers, radii, np.sum(radii)
