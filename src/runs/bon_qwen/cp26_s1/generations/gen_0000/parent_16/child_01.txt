import numpy as np
from scipy.optimize import minimize
import math

def compute_penalty(centers, radii, pair_indices):
    """Compute the sum of squared constraint violations."""
    n = len(radii)
    pen = 0.0
    
    # Boundary constraints: circles must stay inside [0,1]x[0,1]
    left = radii - centers[:, 0]
    bottom = radii - centers[:, 1]
    right = radii - (1.0 - centers[:, 0])
    top = radii - (1.0 - centers[:, 1])
    
    pen += np.sum(np.maximum(0.0, left)**2)
    pen += np.sum(np.maximum(0.0, bottom)**2)
    pen += np.sum(np.maximum(0.0, right)**2)
    pen += np.sum(np.maximum(0.0, top)**2)
    
    # Overlap constraints: distance >= sum of radii
    for i, j in pair_indices:
        d = math.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
        pen += max(0.0, radii[i] + radii[j] - d)**2
    return pen

def objective(params, n, w, pair_indices):
    """Objective function: maximize sum of radii with penalty for violations."""
    centers = params[:n*2].reshape((n, 2))
    radii = params[n*2:]
    return -np.sum(radii) + w * compute_penalty(centers, radii, pair_indices)

def init_hex_grid(n=26):
    """Generate an initial hexagonal grid configuration."""
    pts = []
    r_start = 0.15
    for row in range(8):
        for col in range(6):
            if len(pts) >= n:
                break
            x = col * r_start * 2.0 + (row % 2) * r_start
            y = row * r_start * 1.73205
            pts.append([x, y])
        if len(pts) >= n:
            break
            
    pts = np.array(pts[:n])
    min_p = pts.min(axis=0)
    max_p = pts.max(axis=0)
    span = max_p - min_p
    # Scale to fit comfortably inside [0,1] with padding
    pts = (pts - min_p) / span * 0.9 + 0.05
    r_init = np.full(n, 0.08)
    return np.concatenate([pts.ravel(), r_init])

def run_packing():
    np.random.seed(42)
    n = 26
    pair_indices = [(i, j) for i in range(n) for j in range(i + 1, n)]
    bounds = [(0.0, 1.0)] * (n*2) + [(1e-6, 0.45)] * n
    
    best_params = init_hex_grid(n)
    
    # Homotopy optimization: gradually increase penalty weight
    w = 2.0
    for step in range(60):
        w *= 1.25
        
        # Add small stochastic perturbation to escape local minima
        best_params += np.random.randn(len(best_params)) * 0.003
        
        res = minimize(objective, best_params, args=(n, w, pair_indices), 
                       method='L-BFGS-B', bounds=bounds, 
                       options={'maxiter': 400, 'ftol': 1e-12})
        best_params = res.x
        
        # Periodic random restart to explore different basins
        if step % 15 == 0:
            rand_p = np.random.uniform(0.1, 0.9, n*2)
            rand_r = np.random.uniform(0.05, 0.15, n)
            cand = np.concatenate([rand_p, rand_r])
            res_r = minimize(objective, cand, args=(n, w, pair_indices), 
                             method='L-BFGS-B', bounds=bounds, options={'maxiter': 400})
            if res_r.fun < objective(best_params, n, w, pair_indices):
                best_params = res_r.x

    centers = best_params[:n*2].reshape((n, 2))
    radii = best_params[n*2:]
    
    # Final strict validation and correction
    max_v = 0.0
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        v = max(r - x, r - y, r - (1-x), r - (1-y))
        if v > max_v:
            max_v = v
    for i, j in pair_indices:
        d = math.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
        v = radii[i] + radii[j] - d
        if v > max_v:
            max_v = v
            
    # If any violation remains, shrink radii uniformly to guarantee validity
    if max_v > 1e-9:
        shrink = 1.0 - max_v * 1.5
        radii *= max(0.95, shrink)
        
    return centers, radii, float(np.sum(radii))
