import numpy as np
from scipy.optimize import linprog, minimize
import itertools

def compute_max_radii(centers, n):
    """Solve LP to find maximum radii for given centers."""
    c = -np.ones(n)
    
    A_ub_list = []
    b_ub_list = []
    
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        xi, yi = centers[i]
        for bound_val in [xi, 1.0 - xi, yi, 1.0 - yi]:
            row = np.zeros(n)
            row[i] = 1.0
            A_ub_list.append(row)
            b_ub_list.append(max(bound_val, 1e-12))
    
    # Pairwise constraints: r_i + r_j <= dist(i,j)
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub_list.append(row)
            b_ub_list.append(dist)
    
    A_ub = np.array(A_ub_list)
    b_ub = np.array(b_ub_list)
    bounds = [(0.0, None)] * n
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        radii = -result.x
        return radii, np.sum(radii)
    else:
        return np.zeros(n), 0.0


def create_hexagonal_packing(n, r, pattern=None, random_offset=0.0):
    """Create hexagonal packing with given pattern."""
    if pattern is None:
        pattern = [5, 5, 5, 5, 6]
    
    centers = []
    y = r + 0.0001
    for row_idx, n_in_row in enumerate(pattern):
        row_width = (n_in_row - 1) * 2.0 * r
        x_start = (1.0 - row_width) / 2.0
        if row_idx % 2 == 1:
            x_start += r
        
        for i in range(n_in_row):
            x = x_start + i * 2.0 * r + random_offset * 0.01
            y_val = y + random_offset * 0.01
            centers.append([x, y_val])
        
        y += np.sqrt(3) * r
    
    return np.array(centers)


def force_relax(centers, n, num_iters=2000, dt=0.005):
    """Force-based relaxation to improve packing."""
    centers = centers.copy()
    radii, _ = compute_max_radii(centers, n)
    
    for it in range(num_iters):
        forces = np.zeros_like(centers)
        
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist_sq = dx * dx + dy * dy
                dist = np.sqrt(dist_sq)
                
                min_dist = radii[i] + radii[j]
                
                if dist < min_dist and dist > 1e-10:
                    overlap = min_dist - dist
                    # Force proportional to overlap, inversely proportional to distance
                    force_mag = overlap * 10.0 / (dist + 0.01)
                    fx = force_mag * dx / dist
                    fy = force_mag * dy / dist
                    forces[i] += [fx, fy]
                    forces[j] -= [fx, fy]
        
        centers += dt * forces
        centers = np.clip(centers, 0.0001, 0.9999)
        
        if it % 100 == 0:
            radii, _ = compute_max_radii(centers, n)
        
        radii, _ = compute_max_radii(centers, n)
    
    return centers


def objective_function(centers_flat, n):
    """Objective: negative sum of radii."""
    centers = centers_flat.reshape(n, 2)
    radii, total = compute_max_radii(centers, n)
    return -total
