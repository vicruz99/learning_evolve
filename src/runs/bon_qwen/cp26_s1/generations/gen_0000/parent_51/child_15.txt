import numpy as np
from scipy.optimize import minimize

def compute_loss_and_grad(v, beta, n_circles=26):
    loss = 0.0
    grad = np.zeros_like(v)
    
    centers = v[:2*n_circles].reshape(n_circles, 2)
    radii = v[2*n_circles:]
    
    # Objective: maximize sum(r) => minimize -sum(r)
    loss -= np.sum(radii)
    
    grad_radii = np.full(n_circles, -1.0)
    grad_centers = np.zeros_like(centers)
    
    # Boundary penalties: r <= x, r <= 1-x, r <= y, r <= 1-y
    diffs = np.zeros((n_circles, 4))
    diffs[:, 0] = radii - centers[:, 0]          # left
    diffs[:, 1] = radii - (1.0 - centers[:, 0])  # right
    diffs[:, 2] = radii - centers[:, 1]          # bottom
    diffs[:, 3] = radii - (1.0 - centers[:, 1])  # top
    
    violations = np.maximum(0.0, diffs)
    loss += beta * np.sum(violations**2)
    
    active = violations > 1e-12
    mask = np.where(active, 2.0 * violations * beta, 0.0)
    
    grad_radii += np.sum(mask, axis=1)
    grad_centers[:, 0] += np.sum(mask * np.array([-1.0, 1.0, 0.0, 0.0]), axis=1)
    grad_centers[:, 1] += np.sum(mask * np.array([0.0, 0.0, -1.0, 1.0]), axis=1)
    
    # Pairwise penalties: d >= r_i + r_j
    for i in range(n_circles):
        for j in range(i + 1, n_circles):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx*dx + dy*dy)
            if dist < 1e-12:
                dist = 1e-12
                
            delta = radii[i] + radii[j] - dist
            if delta > 1e-12:
                loss += beta * delta**2
                factor = 2.0 * delta * beta
                grad_radii[i] += factor
                grad_radii[j] += factor
                
                spatial_factor = factor / dist
                grad_centers[i, 0] += spatial_factor * dx
                grad_centers[i, 1] += spatial_factor * dy
                grad_centers[j, 0] += spatial_factor * (-dx)
                grad_centers[j, 1] += spatial_factor * (-dy)
                
    grad[:2*n_circles] = grad_centers.flatten()
    grad[2*n_circles:] = grad_radii
    
    return loss, grad

def run_packing():
    np.random.seed(42)
    n_circles = 26
    bounds = [(0.0, 1.0) for _ in range(2 * n_circles)] + [(0.0001, 0.5) for _ in range(n_circles)]
    
    # Initialize with 5x5 grid + 1 extra circle
    v0 = np.zeros(3 * n_circles)
    coords = [0.1, 0.3, 0.5, 0.7, 0.9]
    c_idx = 0
    r_idx = 52
    for x in coords:
        for y in coords:
            v0[c_idx] = x
            v0[c_idx+1] = y
            v0[r_idx] = 0.04
            c_idx += 2
            r_idx += 1
    v0[c_idx] = 0.05
    v0[c_idx+1] = 0.05
    v0[r_idx] = 0.04
    
    # Add small noise to break symmetry
    v0[:52] += np.random.uniform(-0.005, 0.005, size=52)
    
    # Annealing optimization
    beta = 1.0
    for step in range(25):
        res = minimize(compute_loss_and_grad, v0, args=(beta, n_circles), method='L-BFGS-B', jac=True,
                       bounds=bounds, options={'maxiter': 400, 'ftol': 1e-12})
        v0 = res.x
        beta *= 2.5
        
    centers = v0[:2*n_circles].reshape(n_circles, 2)
    radii = v0[2*n_circles:]
    
    # Post-processing to guarantee strict validity
    safe_margin = 1e-7
    for _ in range(300):
        violated = False
        # Boundary constraints
        for i in range(n_circles):
            x, y = centers[i]
            r = radii[i]
            max_r = min(x, 1.0 - x, y, 1.0 - y)
            if r > max_r + safe_margin:
                radii[i] = max_r
                violated = True
                
        # Overlap constraints
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                if dist < radii[i] + radii[j] + safe_margin:
                    reduction = (radii[i] + radii[j] - dist) / 2.0
                    if reduction > 0:
                        radii[i] -= reduction
                        radii[j] -= reduction
                        violated = True
        if not violated:
            break
            
    return centers, radii, np.sum(radii)
