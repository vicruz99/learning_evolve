import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    """
    n = 26
    
    # 1. Initialization: Hexagonal Lattice
    # Start with a safe radius to ensure initial validity
    r_init = 0.09
    centers_init = []
    row = 0
    
    # Generate points on a hexagonal grid
    while len(centers_init) < n:
        y = r_init + row * r_init * np.sqrt(3)
        # Check if the row fits vertically within [0, 1]
        if y + r_init > 1.0 + 1e-9:
            break
            
        # Determine x-offset for the row (staggered for hexagonal)
        if row % 2 == 0:
            start_x = r_init
        else:
            start_x = 2 * r_init
            
        x = start_x
        while x + r_init <= 1.0 + 1e-9:
            centers_init.append([x, y])
            x += 2 * r_init
        row += 1
        
    # Ensure we have exactly n centers (should be sufficient with r=0.09)
    # If for some reason we have fewer, fill with center points (fallback)
    while len(centers_init) < n:
        centers_init.append([0.5, 0.5])
        
    centers_init = np.array(centers_init[:n])
    radii_init = np.full(n, r_init)
    
    # 2. Flatten variables for scipy
    # Order: [x1, y1, r1, x2, y2, r2, ..., xn, yn, rn]
    x0 = []
    for i in range(n):
        x0.append(centers_init[i, 0])
        x0.append(centers_init[i, 1])
        x0.append(radii_init[i])
    x0 = np.array(x0)
    
    # 3. Bounds
    # x, y in [0, 1], r in [0, 0.5]
    bounds = [
        (0.0, 1.0), (0.0, 1.0), (0.0, 0.5) 
        for _ in range(n)
    ]
    
    # 4. Constraints (Vectorized for performance)
    
    def cons_boundary(vars_flat):
        """
        Boundary constraints:
        x_i - r_i >= 0
        1 - x_i - r_i >= 0
        y_i - r_i >= 0
        1 - y_i - r_i >= 0
        """
        pts = vars_flat.reshape(n, 3)
        xs = pts[:, 0]
        ys = pts[:, 1]
        rs = pts[:, 2]
        
        return np.concatenate([
            xs - rs,
            1.0 - xs - rs,
            ys - rs,
            1.0 - ys - rs
        ])

    def cons_distance(vars_flat):
        """
        Non-overlap constraints:
        dist_ij^2 - (r_i + r_j)^2 >= 0 for all i < j
        """
        pts = vars_flat.reshape(n, 3)
        xs = pts[:, 0]
        ys = pts[:, 1]
        rs = pts[:, 2]
        
        # Compute squared Euclidean distance matrix using broadcasting
        # (n, 1) - (1, n) -> (n, n)
        dx = xs[:, None] - xs[None, :]
        dy = ys[:, None] - ys[None, :]
        dist_sq = dx**2 + dy**2
        
        # Compute squared sum of radii matrix
        r_sum = rs[:, None] + rs[None, :]
        r_sum_sq = r_sum**2
        
        # Constraint violation margin
        margin = dist_sq - r_sum_sq
        
        # Extract upper triangle (i < j) to avoid duplicates and self-comparison
        mask = np.triu(np.ones((n, n), dtype=bool), k=1)
        return margin[mask]

    constraints = [
        {'type': 'ineq', 'fun': cons_boundary},
        {'type': 'ineq', 'fun': cons_distance}
    ]
    
    # 5. Objective Function
    def objective(vars_flat):
        # Maximize sum(r) => Minimize -sum(r)
        return -np.sum(vars_flat[2::3]) # Every 3rd element starting from index 2 is r
    
    # 6. Run Optimization
    # SLSQP is suitable for constrained nonlinear optimization
    try:
        res = minimize(
            objective, 
            x0, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=constraints, 
            options={
                'ftol': 1e-12, 
                'maxiter': 2000,
                'disp': False
            }
        )
        
        if res.success:
            x_opt = res.x
        else:
            # If optimization fails, we might still want to return the result 
            # if it's valid, or fall back. SLSQP often converges locally.
            x_opt = res.x 
    except Exception:
        # Fallback to initial valid configuration if optimization crashes
        x_opt = x0

    # 7. Reshape results
    centers_opt = x_opt.reshape(n, 3)[:, :2]
    radii_opt = x_opt.reshape(n, 3)[:, 2]
    
    # 8. Compute Sum
    sum_radii = float(np.sum(radii_opt))
    
    return centers_opt, radii_opt, sum_radii
