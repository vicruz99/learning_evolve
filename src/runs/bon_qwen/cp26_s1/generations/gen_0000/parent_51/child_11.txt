import numpy as np
from scipy.optimize import minimize

N = 26
# Precompute upper triangle indices for overlap constraints
UPPER_INDICES = np.triu_indices(N, k=1)

def objective(vars_flat):
    V = vars_flat.reshape((N, 3))
    return -np.sum(V[:, 2])

def constraint_func(vars_flat):
    V = vars_flat.reshape((N, 3))
    c = V[:, :2]
    r = V[:, 2]
    
    # Boundary constraints: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    con = np.concatenate([c[:, 0] - r, 
                          1.0 - c[:, 0] - r, 
                          c[:, 1] - r, 
                          1.0 - c[:, 1] - r])
                          
    # Non-overlap constraints: ||c_i - c_j||^2 >= (r_i + r_j)^2
    diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
    dist_sq = np.sum(diff**2, axis=2)
    np.fill_diagonal(dist_sq, np.inf)
    
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    overlap = dist_sq - r_sum**2
    con = np.concatenate([con, overlap[UPPER_INDICES]])
    return con

def get_initial_config():
    r0 = 0.095
    centers = []
    row = 0
    while len(centers) < N:
        col = 0
        while len(centers) < N:
            x = r0 + col * 2 * r0 + (row % 2) * r0
            y = r0 + row * np.sqrt(3) * r0
            if x + r0 <= 1.0 and y + r0 <= 1.0:
                centers.append([x, y])
            col += 1
            if col > 15: break
        row += 1
        if row > 15: break
    return np.array(centers[:N])

def run_packing():
    centers0 = get_initial_config()
    radii0 = np.ones(N) * 0.095
    vars0 = np.column_stack([centers0, radii0]).flatten()
    
    cons = {'type': 'ineq', 'fun': constraint_func}
    bounds = [(0, 1) for _ in range(2*N)] + [(0, 0.5) for _ in range(N)]
    
    # First pass: coarse optimization to approach dense packing
    res = minimize(objective, vars0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': 2000, 'ftol': 1e-12, 'disp': False})
                     
    # Second pass: fine-tuning for maximum precision
    res2 = minimize(objective, res.x, method='SLSQP', bounds=bounds, constraints=cons,
                    options={'maxiter': 2000, 'ftol': 1e-12, 'disp': False})
                      
    V_opt = res2.x.reshape((N, 3))
    centers_opt = V_opt[:, :2]
    radii_opt = V_opt[:, 2]
    sum_r = np.sum(radii_opt)
    
    return centers_opt, radii_opt, sum_r
