import numpy as np

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True

def objective(params, n):

    Negative sum of radii (to minimize)

def constraints_builder(params, n):
    """
    Build constraints for the optimizer.
    Constraints:
    1. Boundary: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    2. Non-overlap: dist(i,j) >= r_i + r_j  =>  dist(i,j)^2 >= (r_i + r_j)^2
       We use the linearized form for radii: r_i + r_j <= dist(i,j)
       However, dist depends on params (x,y).
       So constraints are non-linear.
    """
    centers = params[:2*n].reshape((n, 2))
    radii = params[2*n:]
    
    cons = []
    
    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        cons.append({'type': 'ineq', 'fun': lambda p, i=i: p[2*i] - p[2*n+i]}) # x - r >= 0
        cons.append({'type': 'ineq', 'fun': lambda p, i=i: 1 - p[2*i] - p[2*n+i]}) # 1 - x - r >= 0
        cons.append({'type': 'ineq', 'fun': lambda p, i=i: p[2*i+1] - p[2*n+i]}) # y - r >= 0
        cons.append({'type': 'ineq', 'fun': lambda p, i=i: 1 - p[2*i+1] - p[2*n+i]}) # 1 - y - r >= 0
        
    # Non-overlap constraints
    # r_i + r_j - dist(i,j) <= 0
    for i in range(n):
        for j in range(i + 1, n):
            def non_overlap(p, i=i, j=j):
                xi, yi = p[2*i], p[2*i+1]
                xj, yj = p[2*j], p[2*j+1]
                ri, rj = p[2*n+i], p[2*n+j]
                dist = np.sqrt((xi - xj)**2 + (yi - yj)**2)
                return dist - (ri + rj)
            cons.append({'type': 'ineq', 'fun': non_overlap})
            
    return cons
