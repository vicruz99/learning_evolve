import numpy as np
from scipy.optimize import minimize

def compute_objective(params, n, weight):
    """
    Computes the objective function: negative sum of radii + penalty for constraint violations.
    """
    centers = params[:2 * n].reshape(n, 2)
    radii = params[2 * n:]
    
    val = -np.sum(radii)
    penalty = 0.0
    
    # Boundary penalties: circles must be within [0, 1]x[0, 1]
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x < r:
            penalty += weight * (r - x) ** 2
        if x > 1 - r:
            penalty += weight * (x - (1 - r)) ** 2
        if y < r:
            penalty += weight * (r - y) ** 2
        if y > 1 - r:
            penalty += weight * (y - (1 - r)) ** 2
            
    # Overlap penalties: distance between centers must be >= sum of radii
    for i in range(n):
        xi, yi = centers[i]
        ri = radii[i]
        for j in range(i + 1, n):
            xj, yj = centers[j]
            rj = radii[j]
            dist = np.sqrt((xi - xj) ** 2 + (yi - yj) ** 2)
            if dist < ri + rj:
                penalty += weight * (ri + rj - dist) ** 2
                
    return val + penalty

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    Uses a penalty-based optimization with continuation method and L-BFGS-B.
    """
    n = 26
    
    # Deterministic hexagonal initialization
    centers = []
    y = 0.12
    while len(centers) < n:
        x = 0.12
        row_idx = int((y - 0.12) / 0.18)
        shift = 0.09 if row_idx % 2 == 1 else 0.0
        while x <= 0.88 and len(centers) < n:
            centers.append([x + shift, y])
            x += 0.18
        y += 0.18
    centers = np.array(centers[:n])
    radii = np.full(n, 0.05)
    
    x_curr = np.concatenate([centers.flatten(), radii])
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
    
    # Continuation method: gradually increase penalty weight to escape local minima and enforce constraints
    weights = [50.0, 200.0, 1000.0, 3000.0]
    for w in weights:
        res = minimize(compute_objective, x_curr, args=(n, w), 
                       method='L-BFGS-B', bounds=bounds, 
                       options={'maxiter': 800, 'ftol': 1e-12})
        x_curr = res.x
        
    best_params = x_curr
    centers_opt = best_params[:2 * n].reshape(n, 2)
    radii_opt = best_params[2 * n:]
    
    # Post-processing: clamp radii to guarantee strict feasibility within numerical tolerance
    eps = 1e-7
    for i in range(n):
        r = radii_opt[i]
        x, y = centers_opt[i]
        r = min(r, x, 1 - x, y, 1 - y) - eps
        for j in range(n):
            if i == j: 
                continue
            dist = np.sqrt((x - centers_opt[j, 0]) ** 2 + (y - centers_opt[j, 1]) ** 2)
            r = min(r, dist - radii_opt[j] - eps)
        radii_opt[i] = max(0.0, r)
        
    return centers_opt, radii_opt, float(np.sum(radii_opt))
