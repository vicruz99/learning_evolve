import numpy as np
from scipy.optimize import minimize

def compute_energy_and_grad(centers_flat, r, n):
    """
    Compute the penalty energy and its gradient with respect to flattened centers.
    Energy is sum of squared violations of pairwise distance and boundary constraints.
    """
    centers = centers_flat.reshape(n, 2)
    energy = 0.0
    grad = np.zeros_like(centers_flat)
    
    # Pairwise constraints: distance >= 2*r
    for i in range(n):
        for j in range(i + 1, n):
            diff = centers[i] - centers[j]
            d = np.sqrt(np.sum(diff ** 2))
            if d < 2 * r and d > 1e-9:
                pen = 2 * r - d
                energy += pen ** 2
                # Gradient direction: repulsive
                g_dir = -pen / d * diff
                grad[i * 2 : i * 2 + 2] += 2 * pen * g_dir
                grad[j * 2 : j * 2 + 2] -= 2 * pen * g_dir
                
    # Boundary constraints: r <= x,y <= 1-r
    for i in range(n):
        for dim in range(2):
            val = centers[i, dim]
            if val < r:
                pen = r - val
                energy += pen ** 2
                grad[i * 2 + dim] -= 2 * pen
            elif val > 1 - r:
                pen = val - (1 - r)
                energy += pen ** 2
                grad[i * 2 + dim] += 2 * pen
                
    return energy, grad

def run_packing():
    np.random.seed(42)
    n = 26
    
    # Initialize with a hexagonal lattice pattern
    # This provides a near-optimal starting configuration
    centers = []
    row = 0
    while len(centers) < n:
        for col in range(10):
            x = col * 0.21
            y = row * 0.21 * np.sqrt(3) / 2
            if row % 2 == 1:
                x += 0.105
            centers.append([x, y])
        row += 1
        
    centers = np.array(centers[:n])
    # Normalize to fit comfortably inside the square
    centers = centers * 0.7 + 0.15
    # Add tiny perturbation to break symmetry and aid optimization
    centers += np.random.randn(*centers.shape) * 0.001
    
    # Binary search for the maximum feasible radius
    low, high = 0.05, 0.12
    best_c = centers.copy()
    
    for _ in range(45):
        r = (low + high) / 2
        res = minimize(compute_energy_and_grad, best_c.flatten(), args=(r, n),
                       method='L-BFGS-B', jac=True, bounds=[(0, 1)] * (2 * n),
                       options={'maxiter': 1000, 'ftol': 1e-14})
                       
        if res.fun < 1e-9:
            best_c = res.x.reshape(n, 2)
            low = r
        else:
            high = r
            
    final_r = low
    radii = np.full(n, final_r)
    return best_c, radii, float(np.sum(radii))
