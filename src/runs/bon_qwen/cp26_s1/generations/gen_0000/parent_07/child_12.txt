import numpy as np
from scipy.optimize import minimize
import itertools

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    n = centers.shape[0]

    if np.isnan(centers).any():
        return False
    if np.isnan(radii).any():
        return False

    for i in range(n):
        if radii[i] < 0:
            return False
        elif np.isnan(radii[i]):
            return False

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            return False

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False

    return True

def objective_function(vars, n):
    """Objective function to minimize (negative sum of radii)"""
    radii = vars[n * 2:]
    return -np.sum(radii)

def boundary_constraints(vars, n):
    """Constraints for circles to be inside the unit square"""
    constraints_list = []
    centers = vars[:n * 2].reshape(n, 2)
    radii = vars[n * 2:]
    
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        # x - r >= 0  => r - x <= 0
        constraints_list.append(r - x)
        # x + r <= 1  => x + r - 1 <= 0
        constraints_list.append(x + r - 1)
        # y - r >= 0  => r - y <= 0
        constraints_list.append(r - y)
        # y + r <= 1  => y + r - 1 <= 0
        constraints_list.append(y + r - 1)
    return constraints_list

def overlap_constraints(vars, n):
    """Constraints for non-overlapping circles"""
    constraints_list = []
    centers = vars[:n * 2].reshape(n, 2)
    radii = vars[n * 2:]
    
    for i in range(n):
        for j in range(i + 1, n):
            dist_sq = np.sum((centers[i] - centers[j]) ** 2)
            sum_radii = radii[i] + radii[j]
            # dist >= sum_radii  => dist^2 >= sum_radii^2  => sum_radii^2 - dist^2 <= 0
            constraints_list.append(sum_radii**2 - dist_sq)
    return constraints_list

def get_constraints(n):
    """Setup constraints for scipy optimizer"""
    cons = []
    # Boundary constraints (inequality <= 0)
    for i in range(n):
        # x - r >= 0 => -x + r <= 0
        cons.append({'type': 'ineq', 'fun': lambda v, idx=i: v[idx*2] - v[n*2+idx]}) # Wait, vars structure: [x0, y0, x1, y1, ..., r0, r1...]
        # Let's re-define vars structure for consistency in lambdas or just pass indices
        # Structure: [x0, y0, x1, y1, ... x25, y25, r0, r1, ... r25]
        # Indices: x_i at 2*i, y_i at 2*i+1, r_i at 2*n + i
        
        # x_i - r_i >= 0
        cons.append({'type': 'ineq', 'fun': lambda v, idx=i: v[2*idx] - v[2*n+idx]})
        # x_i + r_i <= 1
        cons.append({'type': 'ineq', 'fun': lambda v, idx=i: 1.0 - v[2*idx] - v[2*n+idx]})
        # y_i - r_i >= 0
        cons.append({'type': 'ineq', 'fun': lambda v, idx=i: v[2*idx+1] - v[2*n+idx]})
        # y_i + r_i <= 1
        cons.append({'type': 'ineq', 'fun': lambda v, idx=i: 1.0 - v[2*idx+1] - v[2*n+idx]})
        
    # Overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            # dist^2 - (r_i + r_j)^2 >= 0
            cons.append({
                'type': 'ineq',
                'fun': lambda v, i=i, j=j: np.sum((v[2*i:2*i+2] - v[2*j:2*j+2])**2) - (v[2*n+i] + v[2*n+j])**2
            })
    return cons

def run_packing():
    n = 26
    best_sum_radii = -np.inf
    best_centers = None
    best_radii = None
    
    # We will try multiple random starts to find a good local optimum
    num_trials = 10
    
    for trial in range(num_trials):
        # Initialization: Hexagonal-like grid
        # 5 rows, roughly 5-6 circles per row
        # We'll just generate random points inside a slightly smaller square and expand
        np.random.seed(42 + trial)
        
        # Start with a grid that fits
        # 6x5 grid is 30 circles, too many. 5x5 is 25.
        # Let's place 26 circles in a perturbed 6x5 grid scaled down?
        # Or just random initialization with small radii.
        
        centers_init = np.random.rand(n, 2)
        # Scale down to center a bit to avoid boundary issues initially
        centers_init = 0.5 + 0.4 * (2 * centers_init - 1)
        
        # Initial radius: small enough to not overlap
        # With 26 circles, average area 1/26 ~ 0.04. r ~ 0.1
        # But to be safe for start, let's use 0.05
        radii_init = np.full(n, 0.05)
        
        # Flatten for optimizer: [x0, y0, ..., x25, y25, r0, ..., r25]
        x0 = np.concatenate([centers_init.flatten(), radii_init])
        
        # Bounds: x,y in [0,1], r >= 0
        bounds = []
        for _ in range(n):
            bounds.append((0.0, 1.0)) # x
            bounds.append((0.0, 1.0)) # y
        for _ in range(n):
            bounds.append((0.0, 1.0)) # r (upper bound 1 is loose but safe)

        constraints = get_constraints(n)
        
        # Optimize
        # SLSQP is good for this
        res = minimize(
            fun=objective_function,
            x0=x0,
            args=(n,),
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 1000, 'ftol': 1e-9}
        )
        
        if res.success:
            centers_opt = res.x[:2*n].reshape(n, 2)
            radii_opt = res.x[2*n:]
            sum_r = np.sum(radii_opt)
            
            # Validate just in case of numerical issues
            # The constraints handle it, but let's double check
            if validate_packing(centers_opt, radii_opt):
                if sum_r > best_sum_radii:
                    best_sum_radii = sum_r
                    best_centers = centers_opt.copy()
                    best_radii = radii_opt.copy()

    # If no valid solution found (unlikely), return a fallback
    if best_centers is None:
        # Fallback: simple grid
        best_centers = np.zeros((n, 2))
        best_radii = np.zeros(n)

        k = 5
        count = 0

            for j in range(k):
                if count < n:
                    best_centers[count] = [ (2*i+1)/(2*k), (2*j+1)/(2*k) ]
                    best_radii[count] = 1/(2*k) - 0.01 # Slightly smaller
                    count += 1
        if count < n:
            best_centers[count] = [0.5, 0.5]
            best_radii[count] = 0.01
            count += 1
        best_sum_radii = np.sum(best_radii)

    return best_centers, best_radii, best_sum_radii
