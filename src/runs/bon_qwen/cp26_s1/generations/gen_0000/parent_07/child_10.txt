import numpy as np
from scipy.optimize import minimize


def compute_violation_sum(centers, radii):
    """Compute sum of squared constraint violations"""
    n = centers.shape[0]
    total = 0.0

    for i in range(n):
        x = centers[i, 0]
        y = centers[i, 1]
        r = radii[i]

        if x < r:
            total += (r - x) ** 2
        if x + r > 1:
            total += (x + r - 1) ** 2
        if y < r:
            total += (r - y) ** 2
        if y + r > 1:
            total += (y + r - 1) ** 2

    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist_sq = dx * dx + dy * dy
            dist = np.sqrt(dist_sq)
            overlap = radii[i] + radii[j] - dist
            if overlap > 0:
                total += overlap ** 2

    return total


def objective(params, n, penalty_weight):
    """Objective: maximize sum of radii with penalty for violations"""
    centers = params[:2 * n].reshape(n, 2)
    radii = params[2 * n:]

    penalty = compute_violation_sum(centers, radii)

    return -np.sum(radii) + penalty_weight * penalty


def create_initial_params(n, r, row_sizes):
    """Create initial hexagonal arrangement"""
    centers = np.zeros((n, 2))
    radii = np.ones(n) * r

    idx = 0
    for row_idx, size in enumerate(row_sizes):
        for col_idx in range(size):
            if idx >= n:
                break
            x = r + col_idx * 2 * r + (row_idx % 2) * r
            y = r + row_idx * r * np.sqrt(3)
            centers[idx] = np.array([x, y])
            idx += 1

    return np.concatenate([centers.flatten(), radii])


def optimize_from_initial(params, n):
    """Optimize starting from given initial parameters"""
    bounds = []
    for _ in range(n):
        bounds.append((0.0, 1.0))
        bounds.append((0.0, 1.0))
    for _ in range(n):
        bounds.append((1e-8, 0.5))

    params_opt = params.copy()

    for weight in [100, 1000, 10000, 50000, 100000]:
        result = minimize(
            objective,
            params_opt,
            args=(n, weight),
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 10000, 'ftol': 1e-18, 'gtol': 1e-15}
        )
        params_opt = result.x.copy()

    return params_opt
