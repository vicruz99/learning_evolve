import numpy as np
from scipy.optimize import minimize

def compute_initial_layout(n_circles):
    """
    Generates an initial hexagonal grid layout for n_circles.
    """
    points = []
    # Estimated parameters for layout generation
    r_est = 0.1
    h = r_est * np.sqrt(3)
    
    # Hexagonal rows configuration: (count, offset)
    # This pattern yields a dense packing
    rows = [
        (5, 0), (4, 1), (5, 0), (4, 1), (5, 0), (4, 1)
    ]
    
    y = r_est
    for count, offset in rows:
        for i in range(count):
            # Calculate x with horizontal spacing 2*r_est and shift
            x = r_est + 2 * r_est * i + offset * r_est
            points.append((x, y))
        y += h
        if len(points) >= n_circles:
            break
            
    return np.array(points[:n_circles])

def evaluate_cost_and_jac(vars, n_circles):
    """
    Calculates the objective function (negative sum of radii) and its gradient.
    vars: array of shape (3*n_circles,) [x1...xn, y1...yn, r1...rn]
    """
    # Extract radii (last n_circles variables)
    r = vars[2 * n_circles:]
    
    # Objective: minimize -sum(r)
    cost = -np.sum(r)
    
    # Jacobian of objective
    jac = np.zeros_like(vars)
    jac[2 * n_circles:] = -1.0
    
    return cost, jac

def evaluate_constraints(vars, n_circles):
    """
    Calculates constraint values for boundary and non-overlap.
    Returns an array of constraint values (>= 0 required).
    """
    x = vars[:n_circles]
    y = vars[n_circles:2 * n_circles]
    r = vars[2 * n_circles:]
    
    constraints = []
    
    # 1. Boundary Constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    # Flattened to a single array
    constraints.append(x - r)           # x - r >= 0
    constraints.append(1 - x - r)       # 1 - x - r >= 0
    constraints.append(y - r)           # y - r >= 0
    constraints.append(1 - y - r)       # 1 - y - r >= 0
    
    # 2. Non-overlap Constraints: dist^2 >= (r_i + r_j)^2
    # Vectorized calculation for efficiency
    for i in range(n_circles):
        for j in range(i + 1, n_circles):
            dx = x[i] - x[j]
            dy = y[i] - y[j]
            dist_sq = dx**2 + dy**2
            r_sum = r[i] + r[j]
            constraints.append(dist_sq - r_sum**2)
            
    return np.concatenate(constraints)

def evaluate_constraints_jac(vars, n_circles):
    """
    Calculates the Jacobian matrix of the constraints.
    Returns a matrix of shape (num_constraints, 3*n_circles).
    """
    x = vars[:n_circles]
    y = vars[n_circles:2 * n_circles]
    r = vars[2 * n_circles:]
    
    # Dimensions
    num_boundary = 4 * n_circles
    num_overlap = n_circles * (n_circles - 1) // 2
    num_total = num_boundary + num_overlap
    jac = np.zeros((num_total, 3 * n_circles))
    
    # 1. Boundary Constraints Jacobian
    # x_i - r_i >= 0
    jac[0:n_circles, 0:n_circles] = np.eye(n_circles) # dx
    jac[0:n_circles, 2*n_circles:] = -np.eye(n_circles) # dr
    
    # 1 - x_i - r_i >= 0
    jac[n_circles:2*n_circles, 0:n_circles] = -np.eye(n_circles) # dx
    jac[n_circles:2*n_circles, 2*n_circles:] = -np.eye(n_circles) # dr
    
    # y_i - r_i >= 0
    jac[2*n_circles:3*n_circles, n_circles:2*n_circles] = np.eye(n_circles) # dy
    jac[2*n_circles:3*n_circles, 2*n_circles:] = -np.eye(n_circles) # dr
    
    # 1 - y_i - r_i >= 0
    jac[3*n_circles:4*n_circles, n_circles:2*n_circles] = -np.eye(n_circles) # dy
    jac[3*n_circles:4*n_circles, 2*n_circles:] = -np.eye(n_circles) # dr
    
    # 2. Non-overlap Constraints Jacobian
    # Constraint: (xi-xj)^2 + (yi-yj)^2 - (ri+rj)^2 >= 0
    # Derivatives w.r.t xi, xj, yi, yj, ri, rj
    
    row = num_boundary
    for i in range(n_circles):
        for j in range(i + 1, n_circles):
            dx = x[i] - x[j]
            dy = y[i] - y[j]
            r_sum = r[i] + r[j]
            
            # x coordinates
            jac[row, i] = 2 * dx
            jac[row, j] = -2 * dx
            
            # y coordinates
            jac[row, n_circles + i] = 2 * dy
            jac[row, n_circles + j] = -2 * dy
            
            # radii
            jac[row, 2 * n_circles + i] = -2 * r_sum
            jac[row, 2 * n_circles + j] = -2 * r_sum
            
            row += 1
            
    return jac

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    Packs 26 circles in a unit square to maximize the sum of radii.
    """
    n_circles = 26
    
    # 1. Initialize Layout
    centers_init = compute_initial_layout(n_circles)
    radii_init = np.full(n_circles, 0.01) # Small feasible start

    x0 = np.concatenate([centers_init[:, 0], centers_init[:, 1], radii_init])

        'type': 'ineq',
        'fun': lambda v: evaluate_constraints(v, n_circles),
        'jac': lambda v: evaluate_constraints_jac(v, n_circles)
    }
    
    # 4. Optimize
    # Using SLSQP for non-linear constraints and gradients
    result = minimize(
        fun=lambda v: evaluate_cost_and_jac(v, n_circles)[0],

# Validation function (Read-Only)
def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True
