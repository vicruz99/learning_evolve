def calculate_max_uniform_radius(centers):
    """
    Given a set of centers, calculate the maximum radius r such that 
    all circles of radius r are inside the unit square and do not overlap.
    """
    n = centers.shape[0]
    min_dist = 1.0

    # Distance to walls
    for i in range(n):
        x, y = centers[i]
        dist_to_wall = min(x, 1-x, y, 1-y)
        if dist_to_wall < min_dist:
            min_dist = dist_to_wall

    # Distance between circles
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = math.sqrt(dx*dx + dy*dy)
            # The distance between centers must be at least 2*r
            # So 2*r <= dist  => r <= dist/2
            half_dist = dist / 2.0
            if half_dist < min_dist:
                min_dist = half_dist

    return min_dist
