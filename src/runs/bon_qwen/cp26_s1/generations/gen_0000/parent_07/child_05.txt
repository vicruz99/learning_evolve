import numpy as np
from scipy.optimize import minimize
import itertools

# Helper functions (Top level, no closures)

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square.
    Returns True if valid, False otherwise.
    """
    n = centers.shape[0]
    if np.isnan(centers).any() or np.isnan(radii).any():
        return False

    for i in range(n):
        if radii[i] < 0:
            return False
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            return False

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False
    return True

def compute_overlap_penalty(centers, radii):
    """
    Calculates a penalty value based on overlaps and boundary violations.
    Returns 0.0 if no violations, positive value otherwise.
    """
    n = centers.shape[0]
    penalty = 0.0
    
    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        # Left wall
        if x - r < 0:
            penalty += max(0, -(x - r)) ** 2
        # Right wall
        if x + r > 1:
            penalty += max(0, (x + r) - 1) ** 2
        # Bottom wall
        if y - r < 0:
            penalty += max(0, -(y - r)) ** 2
        # Top wall
        if y + r > 1:
            penalty += max(0, (y + r) - 1) ** 2
            
    # Circle overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            min_dist = radii[i] + radii[j]
            if dist < min_dist:
                penalty += (min_dist - dist) ** 2
                
    return penalty

def objective_function(params, n):
    """
    Objective function to minimize. 
    We want to maximize sum(radii). So we minimize -sum(radii) + penalty.
    """
    centers = params[:2*n].reshape(n, 2)
    radii = params[2*n:]
    
    # Ensure radii are positive (soft constraint, heavy penalty if negative)
    # Although physical radii must be positive, optimizer might dip.
    # We clamp or penalize. Let's penalize large negative values.
    radii_clamped = np.maximum(radii, 1e-6)
    
    sum_radii = np.sum(radii_clamped)
    penalty = compute_overlap_penalty(centers, radii_clamped)
    
    # We want to maximize sum_radii, so we minimize -sum_radii.
    # We add a large weight to penalty to enforce constraints.
    return -sum_radii + 1000.0 * penalty

def optimize_packing(n, initial_params):
    """
    Runs the optimizer on a given set of initial parameters.
    """
    try:
        result = minimize(
            objective_function, 
            initial_params, 
            args=(n,),
            method='Nelder-Mead', 
            options={'maxiter': 5000, 'xatol': 1e-7, 'fatol': 1e-7}
        )
        return result.x
    except Exception:
        return None

    return best_centers, best_radii, best_sum
