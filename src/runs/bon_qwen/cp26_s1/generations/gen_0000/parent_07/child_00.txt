import numpy as np
from scipy.optimize import minimize


def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    best_centers = None
    best_radii = None
    best_sum = 0.0
    
    for seed in range(15):
        centers, radii = init_hex(n, seed)
        
        x0 = make_flat(centers, radii, n)
        
        def objective(x):
            return -np.sum(x[2::3])
        
        cons = []
        for i in range(n):
            cons.append({'type': 'ineq', 'fun': lambda x, i=i: x[3*i] - x[3*i+2]})
            cons.append({'type': 'ineq', 'fun': lambda x, i=i: 1.0 - x[3*i] - x[3*i+2]})
            cons.append({'type': 'ineq', 'fun': lambda x, i=i: x[3*i+1] - x[3*i+2]})
            cons.append({'type': 'ineq', 'fun': lambda x, i=i: 1.0 - x[3*i+1] - x[3*i+2]})
            cons.append({'type': 'ineq', 'fun': lambda x, i=i: x[3*i+2]})
        
        for i in range(n):
            for j in range(i+1, n):
                cons.append({
                    'type': 'ineq',
                    'fun': lambda x, i=i, j=j: 
                        (x[3*i] - x[3*j])**2 + (x[3*i+1] - x[3*j+1])**2 - (x[3*i+2] + x[3*j+2])**2
                })
        
        bounds = []
        for i in range(n):
            bounds.extend([(0.0001, 0.9999), (0.0001, 0.9999), (1e-9, 0.5)])
        
        try:
            res = minimize(
                objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                options={'maxiter': 3000, 'ftol': 1e-13, 'disp': False}
            )
            
            c_opt, r_opt = make_uf(res.x, n)
            c_valid, r_valid = enforce_valid(c_opt, r_opt, n)
            
            s = np.sum(r_valid)
            if s > best_sum:
                best_sum = s
                best_centers = c_valid.copy()
                best_radii = r_valid.copy()
        except Exception:
            pass
    
    return best_centers, best_radii, best_sum


def make_flat(centers, radii, n):
    x = np.zeros(3 * n)
    for i in range(n):
        x[3 * i] = centers[i, 0]
        x[3 * i + 1] = centers[i, 1]
        x[3 * i + 2] = radii[i]
    return x


def make_uf(x, n):
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        centers[i, 0] = x[3 * i]
        centers[i, 1] = x[3 * i + 1]
        radii[i] = x[3 * i + 2]
    return centers, radii


def init_hex(n, seed):
    rng = np.random.RandomState(seed)
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.09
    
    idx = 0
    row = 0
    while idx < n:
        col = 0
        x_offset = 0.08 if row % 2 == 1 else 0.0
        while idx < n:
            x = 0.1 + col * 0.168 + x_offset + rng.randn() * 0.003
            y = 0.1 + row * 0.145 + rng.randn() * 0.003
            x = np.clip(x, 0.02, 0.98)
            y = np.clip(y, 0.02, 0.98)
            centers[idx] = [x, y]
            idx += 1
            col += 1
        row += 1
    
    while idx < n:
        centers[idx] = [0.3 + rng.randn() * 0.3, 0.3 + rng.randn() * 0.3]
        idx += 1
    
    return centers, radii


def enforce_valid(centers, radii, n):
    for _ in range(500):
        changed = False
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            
            r = min(r, x, 1.0 - x, y, 1.0 - y)
            
            for j in range(n):
                if i != j:
                    dx = centers[i, 0] - centers[j, 0]
                    dy = centers[i, 1] - centers[j, 1]
                    d = np.sqrt(dx * dx + dy * dy)
                    r = min(r, d - radii[j])
            
            r = max(r, 0.0)
            if abs(r - radii[i]) > 1e-12:
                radii[i] = r
                changed = True
        if not changed:
            break
    
    return centers, radii
