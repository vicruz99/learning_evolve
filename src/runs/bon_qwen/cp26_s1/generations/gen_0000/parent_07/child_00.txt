from scipy.optimize import minimize

def compute_obj_grad(vars, alpha):
    """
    Computes the objective function and its gradient for the circle packing problem.
    Objective: Maximize sum of radii <=> Minimize -sum(radii) + penalty(overlaps)
    """
    n = 26
    centers = vars[:n*2].reshape((n, 2))
    radii = vars[n*2:]
    
    obj = -np.sum(radii)
    grad = np.zeros_like(vars)
    grad[n*2:] -= 1.0  # Gradient of -sum(radii)
    
    for i in range(n):
        for j in range(i + 1, n):
            diff = centers[i] - centers[j]
            d_sq = np.sum(diff**2)
            d = np.sqrt(d_sq) if d_sq > 1e-15 else 1e-9
            violation = radii[i] + radii[j] - d
            
            if violation > 0:
                obj += alpha * violation**2
                g_r = 2 * alpha * violation
                grad[n*2 + i] += g_r
                grad[n*2 + j] += g_r
                
                g_vec = -2 * alpha * violation * diff / d
                grad[2*i] += g_vec[0]
                grad[2*i + 1] += g_vec[1]
                grad[2*j] -= g_vec[0]
                grad[2*j + 1] -= g_vec[1]
                
    return obj, grad

def get_hex_config(n=26):
    """Generates a hexagonal lattice initialization for circle centers."""
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.08)
    
    cols = int(np.ceil(np.sqrt(n * 2 / np.sqrt(3)))) + 1
    rows = int(np.ceil(n / cols))
    
    x_step = 0.8 / (cols - 1) if cols > 1 else 1.0
    y_step = 0.8 / (rows - 1) if rows > 1 else 1.0
    
    idx = 0
    for r in range(rows):
        for c in range(cols):
            if idx >= n: break

            y = 0.1 + r * y_step
            if r % 2 == 1:
                x += x_step / 2
            centers[idx] = [x, y]
            idx += 1
    return centers, radii
