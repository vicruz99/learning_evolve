import numpy as np
from scipy.optimize import linprog

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    """
    n = 26
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Configuration for the optimizer
    num_restarts = 10
    iterations_per_restart = 3000
    step_size = 0.005  # Step size for center movement
    active_tol = 1e-4  # Tolerance for considering a constraint active
    
    for restart_idx in range(num_restarts):
        # Initialize centers randomly within the square
        # Using a seed based on restart index for reproducibility
        rng = np.random.RandomState(42 + restart_idx)
        centers = rng.rand(n, 2)
        
        # Optimization loop
        for iteration in range(iterations_per_restart):
            # 1. Compute wall bounds (max possible radius for each center based on walls)
            x = centers[:, 0]
            y = centers[:, 1]
            # Distance to closest wall: min(x, 1-x, y, 1-y)
            W = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
            
            # 2. Solve LP to find optimal radii for current centers
            # Maximize sum(r_i)  <=> Minimize -sum(r_i)
            # Subject to:
            #   r_i >= 0
            #   r_i <= W_i
            #   r_i + r_j <= dist(c_i, c_j) for all i < j
            
            c_obj = -np.ones(n)  # Minimize -sum(r)
            
            # Inequality constraints: A_ub @ r <= b_ub
            # We need r_i + r_j <= D_ij
            # Constructing the matrix explicitly might be memory heavy if n is large, 
            # but for n=26, it's fine. Number of pairs = 325.

            # but 325 constraints is trivial for modern LP solvers.
            
            A_ub = []
            b_ub = []

            # centers shape: (n, 2)
            # diff shape: (n, n, 2)
            diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
            dist_matrix = np.sqrt(np.sum(diff**2, axis=2))

            # But constructing the full matrix and filtering is easier code-wise
            # Or just iterate.

            # There are n*(n-1)/2 such constraints.

            bounds = []

                    d = dist_matrix[i, j]
                    # r_i + r_j <= d
                    row = np.zeros(n)
                    row[i] = 1.0
                    row[j] = 1.0
                    rows.append(row)
                    bounds.append(d)

                A_ub = np.array(rows)
                b_ub = np.array(bounds)

                A_ub = np.empty((0, n))
                b_ub = np.empty(0)
