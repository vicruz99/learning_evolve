To maximize the sum of radii for 26 circles in a unit square, we treat this as a constrained optimization problem. The objective is to maximize $\sum r_i$ subject to boundary constraints ($r_i \le x_i \le 1-r_i$, etc.) and pairwise non-overlap constraints ($(x_i-x_j)^2 + (y_i-y_j)^2 \ge (r_i+r_j)^2$).

Key strategy points:
1. **Penalty Method**: We use an interior penalty approach with `L-BFGS-B`. The penalty term squares constraint violations, making the objective smooth and differentiable almost everywhere. This avoids the overhead of explicit constraint handling in SLSQP and is highly robust for packing problems.
2. **Multi-Stage Annealing**: We start with a moderate penalty coefficient $\mu$ and gradually increase it (by a factor of 5 each stage). This allows the optimizer to first find a roughly feasible configuration with larger radii, then progressively tighten constraints to reach the true boundary of the feasible region.
3. **Diverse Initializations**: We start from a structured 5×5 grid (which naturally holds 25 circles) plus an extra circle in a gap, and a perturbed version of it. Multiple restarts help escape poor local minima.
4. **High-Precision Refinement**: After the annealing stages, we run a final optimization with a very high penalty to ensure all constraints are satisfied within $10^{-12}$ tolerance, as required by the validator.
5. **Vectorization**: Pairwise distance and overlap calculations are fully vectorized using NumPy broadcasting for speed and numerical stability.

import numpy as np
from scipy.optimize import minimize

def compute_loss(vars, n, mu):
    """
    Computes the negative sum of radii plus penalty terms for constraint violations.
    vars: flattened array [x1, y1, r1, x2, y2, r2, ...]
    """
    cx = vars[0::3]
    cy = vars[1::3]
    r = vars[2::3]
    
    # Objective: maximize sum of radii => minimize negative sum
    loss = -np.sum(r)
    
    # Boundary penalties: circles must stay within [0,1]x[0,1]
    loss += mu * np.sum(np.maximum(0.0, r - cx)**2)
    loss += mu * np.sum(np.maximum(0.0, cx + r - 1.0)**2)
    loss += mu * np.sum(np.maximum(0.0, r - cy)**2)
    loss += mu * np.sum(np.maximum(0.0, cy + r - 1.0)**2)
    
    # Pairwise non-overlap penalties
    diff_x = cx[:, None] - cx[None, :]
    diff_y = cy[:, None] - cy[None, :]
    dist = np.sqrt(diff_x**2 + diff_y**2)
    
    sum_r = r[:, None] + r[None, :]
    overlap = sum_r - dist
    
    # Only consider upper triangle to avoid double counting
    tri_u, tri_v = np.triu_indices(n, k=1)
    overlaps = overlap[tri_u, tri_v]
    loss += mu * np.sum(np.maximum(0.0, overlaps)**2)
    
    return loss

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    n = 26
    best_loss = np.inf
    best_vars = None
    
    # Prepare initial configurations
    configs = []
    
    # Config 1: 5x5 grid for first 25 circles, 26th in a gap
    c1 = np.zeros(3 * n)
    for i in range(5):
        for j in range(5):
            idx = i * 5 + j
            c1[idx * 3] = 0.1 + 0.2 * i
            c1[idx * 3 + 1] = 0.1 + 0.2 * j
            c1[idx * 3 + 2] = 0.09
    c1[25 * 3] = 0.2
    c1[25 * 3 + 1] = 0.2
    c1[25 * 3 + 2] = 0.01
    configs.append(c1)
    
    # Config 2: Randomly perturbed grid to break symmetry
    c2 = c1.copy()
    c2[0::3] += np.random.normal(0, 0.02, n)
    c2[1::3] += np.random.normal(0, 0.02, n)
    np.clip(c2[0::3], 0.05, 0.95, out=c2[0::3])
    np.clip(c2[1::3], 0.05, 0.95, out=c2[1::3])
    configs.append(c2)
    
    # Bounds for variables: x,y in [0,1], r in [0, 0.25]
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.25)] * n
    
    # Optimization loop with penalty annealing
    for start in configs:
        curr_vars = start.copy()
        mu = 100.0
        
        for stage in range(12):
            res = minimize(
                compute_loss, 
                curr_vars, 
                args=(n, mu), 
                method='L-BFGS-B', 
                bounds=bounds, 
                options={'maxiter': 800, 'ftol': 1e-12}
            )
            curr_vars = res.x
            mu *= 5
            
            # Inject small noise in later stages to escape shallow local minima
            if stage > 6:
                curr_vars += np.random.normal(0, 1e-5, size=curr_vars.shape)
                
        # Evaluate true objective without penalty
        loss0 = compute_loss(curr_vars, n, 0.0)
        if loss0 < best_loss:
            best_loss = loss0
            best_vars = curr_vars.copy()
            
    # Final high-precision refinement step
    res_final = minimize(
        compute_loss, 
        best_vars, 
        args=(n, 50000.0), 
        method='L-BFGS-B', 
        bounds=bounds, 
        options={'maxiter': 2000, 'ftol': 1e-14}
    )
    best_vars = res_final.x
    
    # Extract results
    cx = best_vars[0::3]
    cy = best_vars[1::3]
    r = best_vars[2::3]
    
    centers = np.column_stack((cx, cy))
    return centers, r, float(np.sum(r))
