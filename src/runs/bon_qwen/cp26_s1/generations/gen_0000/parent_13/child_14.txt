import numpy as np
from scipy.optimize import linprog

def run_packing():

    N = 26
    centers = np.zeros((N, 2))
    radii = np.zeros(N)

    # If 6 rows don't fit, use 5 rows with perturbations.

    initial_r = 0.08

        row_counts = []

                centers[idx, 0] = x_start + k * 2 * initial_r
                centers[idx, 1] = y

        #      r_i <= dist(i, boundary)
        #      r_i >= 0

        # r_i <= B_i

        constraints_b = []

                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                D[i, j] = dist
                D[j, i] = dist
                
                # Constraint: r_i + r_j <= dist
                row = np.zeros(N)
                row[i] = 1.0
                row[j] = 1.0
                constraints_A.append(row)
                constraints_b.append(dist)

            row = np.zeros(N)
            row[i] = 1.0
            constraints_A.append(row)
            constraints_b.append(B[i])

        b_ub = np.array(constraints_b)

        bounds = [(0, None) for _ in range(N)]
        
        # Solve LP

                dx = centers[j, 0] - xi
                dy = centers[j, 1] - yi
                dist = np.sqrt(dx*dx + dy*dy)
                
                if dist < 1e-9: dist = 1e-9 # Avoid div by zero
                
                # Repulsive force proportional to sum of radii and inverse distance squared

                mag = 0.5 * (ri + rj) / (dist**2 + 1e-6)
                
                fx = mag * dx / dist
                fy = mag * dy / dist
                
                forces[i, 0] -= fx
                forces[i, 1] -= fy
                forces[j, 0] += fx
                forces[j, 1] += fy

        # Update centers
        centers += alpha * forces
        
        # Clamp to square [0, 1]
        centers = np.clip(centers, 0.0, 1.0)

    return centers, radii, sum_radii
