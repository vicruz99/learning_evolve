import numpy as np
from scipy.optimize import minimize

# Small epsilon to enforce strict separation and boundary distance
# This ensures the solution comfortably passes the 1e-12 validation tolerance
EPS = 1e-8

def objective(x, n):
    """Objective function: maximize sum of radii (minimize negative sum)"""
    return -np.sum(x[2*n:])

def boundary_constraints(x, n):
    """Constraint: circles must be strictly inside the unit square"""
    coords = x[:2*n].reshape(n, 2)
    radii = x[2*n:]
    cons = np.empty(4 * n)
    for i in range(n):
        xi, yi = coords[i]
        ri = radii[i]
        cons[4*i] = xi - ri - EPS
        cons[4*i+1] = 1 - xi - ri - EPS
        cons[4*i+2] = yi - ri - EPS
        cons[4*i+3] = 1 - yi - ri - EPS
    return cons

def overlap_constraints(x, n):
    """Constraint: circles must not overlap"""
    coords = x[:2*n].reshape(n, 2)
    radii = x[2*n:]
    
    # Vectorized pairwise distance calculation
    diff = coords[:, np.newaxis, :] - coords[np.newaxis, :, :]
    dist_sq = np.sum(diff**2, axis=2)
    
    # Vectorized radii sum calculation
    rad_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    rad_sum_sq = rad_sum**2
    
    # Extract upper triangle indices for unique pairs
    triu_idx = np.triu_indices(n, k=1)
    return dist_sq[triu_idx] - rad_sum_sq[triu_idx] - EPS

def generate_initial_guess(n, seed):
    """Generate a valid initial guess using a hexagonal lattice with random noise"""
    rng = np.random.default_rng(seed)
    x0 = np.zeros(3 * n)
    
    # Hexagonal grid parameters
    r_init = 0.12
    idx = 0
    y = r_init
    shift = False
    
    while idx < n:
        x = r_init
        if shift:
            x += r_init
        while idx < n and x <= 1 - r_init:
            i = idx
            x0[3*i] = x
            x0[3*i+1] = y
            x0[3*i+2] = r_init
            idx += 1
            x += 2 * r_init
        y += np.sqrt(3) * r_init
        shift = not shift
        
    # Add small random perturbation to break symmetry and help convergence
    x0 += rng.standard_normal(3*n) * 0.005
    
    # Project to ensure initial validity
    for i in range(n):
        r = max(0.0, x0[3*i+2])
        x0[3*i] = np.clip(x0[3*i], r, 1-r)
        x0[3*i+1] = np.clip(x0[3*i+1], r, 1-r)
        x0[3*i+2] = r
        
    return x0

def run_packing():
    n = 26
    best_res = None
    best_obj = -np.inf
    
    # Define bounds for variables: x in [0,1], y in [0,1], r in [0, 0.5]
    bounds = [(0, 1)] * (2*n) + [(0, 0.5)] * n
    
    # Define constraint dictionaries
    cons = [
        {'type': 'ineq', 'fun': boundary_constraints, 'args': (n,)},
        {'type': 'ineq', 'fun': overlap_constraints, 'args': (n,)}
    ]
    
    # Multi-start optimization to find global optimum
    for seed in range(10):
        x0 = generate_initial_guess(n, seed)
        try:
            res = minimize(objective, x0, args=(n,), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 5000, 'ftol': 1e-10, 'disp': False})
            
            # Track best result
            if -res.fun > best_obj:
                best_obj = -res.fun
                best_res = res
        except Exception:
            continue
            
    if best_res is not None:
        x_opt = best_res.x
        centers = x_opt[:2*n].reshape(n, 2)
        radii = x_opt[2*n:]
        
        # Safety scaling to ensure strict validity against 1e-12 tolerance
        # The optimizer uses 1e-8 buffer, but float precision can vary.
        # A tiny scale factor guarantees no overlaps without significantly impacting sum.
        radii *= 0.999995
        
        # Final projection to boundaries
        for i in range(n):
            r = radii[i]
            centers[i, 0] = np.clip(centers[i, 0], r, 1 - r)
            centers[i, 1] = np.clip(centers[i, 1], r, 1 - r)
            
        return centers, radii, np.sum(radii)
    
    # Fallback solution if optimization fails
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.1
    for i in range(n):
        centers[i] = [(i % 5) * 0.2 + 0.1, (i // 5) * 0.2 + 0.1]
    return centers, radii, 2.6
