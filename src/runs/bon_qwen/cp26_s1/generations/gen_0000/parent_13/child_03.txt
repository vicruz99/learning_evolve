5. **Final validation**: Ensure all circles satisfy constraints within numerical tolerance.

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.spatial.distance import cdist

def estimate_radii_fast(centers):
    """Fast O(n^2) estimate of maximum radii given centers"""
    n = len(centers)
    radii = np.zeros(n)
    for i in range(n):
        x, y = centers[i]
        r = min(x, 1-x, y, 1-y)
        for j in range(n):
            if i != j:
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = np.sqrt(dx*dx + dy*dy)
                r = min(r, dist / 2)
        radii[i] = max(r, 0)
    return radii

def solve_radii_lp(centers):
    """Exact LP solution for optimal radii given fixed centers"""
    n = len(centers)
    c_obj = -np.ones(n)
    
    A_rows = []
    b_vals = []
    
    for i in range(n):
        x, y = centers[i]
        max_r = max(0, min(x, 1-x, y, 1-y))
        row = np.zeros(n)
        row[i] = 1
        A_rows.append(row)
        b_vals.append(max_r)
    
    for i in range(n):
        for j in range(i+1, n):
            dx = centers[i][0] - centers[j][0]
            dy = centers[i][1] - centers[j][1]
            dist = np.sqrt(dx*dx + dy*dy)
            row = np.zeros(n)
            row[i] = 1
            row[j] = 1
            A_rows.append(row)
            b_vals.append(dist)
    
    A_ub = np.array(A_rows)
    b_ub = np.array(b_vals)
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return np.maximum(res.x, 0)
    return np.full(n, 0.01)

def smooth_objective(params, n):
    """Smooth objective: maximize sum of radii with penalties for violations"""
    centers = params[:2*n].reshape(n, 2)
    radii = params[2*n:]
    
    obj = -np.sum(radii)
    pen = 0.0
    
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        margin = 1e-6
        for c, bound in [(x-r, margin), (x+r, 1-margin), (y-r, margin), (y+r, 1-margin)]:
            if c < bound:
                pen += 500 * (bound - c)**2
    
    for i in range(n):
        for j in range(i+1, n):
            dx = centers[i][0] - centers[j][0]
            dy = centers[i][1] - centers[j][1]
            dist = np.sqrt(dx*dx + dy*dy)
            overlap = radii[i] + radii[j] - dist
            if overlap > 0:
                pen += 500 * overlap**2
    
    return obj + pen

def hexagonal_init(n, seed=0):
    """Initialize with hexagonal pattern"""
    np.random.seed(seed)
    centers_list = []
    
    best_arrangement = None
    
    for spacing in np.arange(0.15, 0.26, 0.005):
        y_step = spacing * np.sqrt(3) / 2
        temp_centers = []
        row = 0
        cy = y_step / 2
        
        while cy + spacing < 1.0 and len(temp_centers) < n:
            cx_start = spacing / 2 + (row % 2) * spacing / 2
            cx = cx_start
            while cx + spacing < 1.0 and len(temp_centers) < n:
                if 0 < cx < 1 and 0 < cy < 1:
                    temp_centers.append([cx, cy])
                cx += spacing
            cy += y_step
            row += 1
        
        if len(temp_centers) == n:
            if best_arrangement is None:
                best_arrangement = temp_centers
            break
    
    if best_arrangement is None:
        g = int(np.ceil(np.sqrt(n)))
        for i in range(1, g+1):
            for j in range(1, g+1):
                if len(centers_list) < n:
                    centers_list.append([i/(g+1), j/(g+1)])
    else:
        centers_list = best_arrangement
    
    centers = np.array(centers_list[:n])
    centers += np.random.randn(n, 2) * 0.008
    centers = np.clip(centers, 0.05, 0.95)
    return centers

def run_packing():
    n = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    for seed in range(80):
        init_centers = hexagonal_init(n, seed)
        
        # Initialize radii based on initial centers
        init_radii = estimate_radii_fast(init_centers)
        init_radii = np.clip(init_radii, 0.01, 0.2)
        
        # Combine into parameter vector
        init_params = np.concatenate([init_centers.flatten(), init_radii])
        
        bounds = [(0.01, 0.99)] * (2*n) + [(0.001, 0.25)] * n
        
        result = minimize(
            smooth_objective,
            init_params,
            args=(n,),
            method='L-BFGS-B',
            bounds=bounds,
            options={
                'maxiter': 5000,
                'ftol': 1e-18,
                'gtol': 1e-12,
                'disp': False
            }
        )
        
        opt_centers = result.x[:2*n].reshape(n, 2)
        opt_radii = result.x[2*n:]
        
        # Clip to valid range
        opt_centers = np.clip(opt_centers, 0.01, 0.99)
        opt_radii = np.clip(opt_radii, 0.001, 0.25)
        
        # Refine radii using LP (exact solution given these centers)
        lp_radii = solve_radii_lp(opt_centers)
        
        current_sum = np.sum(lp_radii)
        
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = opt_centers.copy()
            best_radii = lp_radii.copy()
    
    # Final validation and adjustment
    final_centers = best_centers
    final_radii = best_radii
    
    # Ensure no numerical violations
    for i in range(n):
        x, y = final_centers[i]
        r = final_radii[i]
        if x - r < 1e-10:
            final_radii[i] = max(final_radii[i], x - 1e-10)
        if x + r > 1 - 1e-10:
            final_radii[i] = max(0, 1 - x - 1e-10)
        if y - r < 1e-10:
            final_radii[i] = max(final_radii[i], y - 1e-10)
        if y + r > 1 - 1e-10:
            final_radii[i] = max(0, 1 - y - 1e-10)
    
    final_sum = np.sum(final_radii)
    
    return final_centers, final_radii, final_sum
