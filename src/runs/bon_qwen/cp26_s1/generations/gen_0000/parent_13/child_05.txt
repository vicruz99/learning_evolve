import numpy as np
from scipy.optimize import minimize
import math


def compute_objective(params, n, penalty_weight):
    centers = params[:2 * n].reshape(n, 2)
    radii = params[2 * n:]

    obj = -np.sum(radii)

    for i in range(n):
        x = centers[i, 0]
        y = centers[i, 1]
        r = radii[i]

        if r < 0:
            obj += penalty_weight * r ** 2

        if x - r < 0:
            obj += penalty_weight * (x - r) ** 2
        if x + r > 1:
            obj += penalty_weight * (x + r - 1) ** 2
        if y - r < 0:
            obj += penalty_weight * (y - r) ** 2
        if y + r > 1:
            obj += penalty_weight * (y + r - 1) ** 2

    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = math.sqrt(dx * dx + dy * dy)
            sum_r = radii[i] + radii[j]

            if dist < sum_r:
                obj += penalty_weight * (sum_r - dist) ** 2

    return obj


def make_hex_centers(row_configs, r_init):
    n = sum(row_configs)
    centers = np.zeros((n, 2))
    idx = 0
    for row in range(len(row_configs)):
        y = r_init + row * r_init * math.sqrt(3)
        x_offset = r_init + (row % 2) * r_init
        for col in range(row_configs[row]):
            x = x_offset + col * 2 * r_init
            centers[idx, 0] = x
            centers[idx, 1] = y
            idx += 1
    return centers
