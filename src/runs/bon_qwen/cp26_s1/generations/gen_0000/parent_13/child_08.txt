import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def _objective(vars):
    """Objective function: minimize negative sum of radii"""
    return -np.sum(vars[2::3])

def _constr(vars):
    """Constraint function: returns array of constraint values >= 0"""
    n = N_CIRCLES
    num_constraints = n * 4 + n * (n - 1) // 2
    c = np.empty(num_constraints)
    idx = 0
    
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    for i in range(n):
        xi, yi, ri = vars[3*i], vars[3*i+1], vars[3*i+2]
        c[idx] = xi - ri; idx += 1
        c[idx] = 1.0 - xi - ri; idx += 1
        c[idx] = yi - ri; idx += 1
        c[idx] = 1.0 - yi - ri; idx += 1
        
    # Non-overlap constraints: dist^2 >= (r1 + r2)^2
    for i in range(n):
        xi, yi, ri = vars[3*i], vars[3*i+1], vars[3*i+2]
        for j in range(i + 1, n):
            xj, yj, rj = vars[3*j], vars[3*j+1], vars[3*j+2]
            dx, dy = xi - xj, yi - yj
            c[idx] = dx*dx + dy*dy - (ri + rj)**2
            idx += 1
    return c

def run_packing():
    np.random.seed(42)
    n = N_CIRCLES
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * n
    
    best_res = None
    best_val = np.inf
    
    # Initial configuration 1: Hexagonal lattice
    x0_1 = np.zeros(3 * n)
    r_start = 0.09
    idx = 0
    for row in range(5):
        y = r_start * (1 + row * np.sqrt(3))
        num = 6 if row % 2 == 1 else 5
        for col in range(num):
            x = r_start * (1 + 2 * col) + (r_start if row % 2 == 1 else 0)
            x0_1[3*idx] = x
            x0_1[3*idx+1] = y
            x0_1[3*idx+2] = r_start
            idx += 1
            if idx == n: break
        if idx == n: break
        
    # Initial configuration 2: Perturbed hexagonal to escape symmetric minima
    x0_2 = x0_1.copy()
    x0_2 += np.random.uniform(-0.015, 0.015, size=x0_2.shape)
    x0_2[2::3] = 0.09  # Reset radii to ensure feasibility start
    
    configs = [x0_1, x0_2]
    
    for x_init in configs:
        try:
            res = minimize(_objective, x_init, method='SLSQP', bounds=bounds,
                           constraints={'type': 'ineq', 'fun': _constr},
                           options={'maxiter': 1500, 'ftol': 1e-10})
            if res.fun < best_val:
                best_val = res.fun
                best_res = res
        except Exception:
            continue
            
    # Fallback if optimization fails
    if best_res is not None:
        final_vars = best_res.x
    else:
        final_vars = x0_1
        
    centers = np.column_stack((final_vars[0::3], final_vars[1::3]))
    radii = final_vars[2::3]
    
    return centers, radii, np.sum(radii)
