import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26
I_IDX, J_IDX = np.triu_indices(N_CIRCLES, k=1)

def get_objective(vars):
    """Minimize negative sum of radii (equivalent to maximizing sum)"""
    return -np.sum(vars[2*N_CIRCLES:])

def get_constraints(vars):
    """Compute constraint values. All must be >= 0 for validity."""
    x = vars[:N_CIRCLES]
    y = vars[N_CIRCLES:2*N_CIRCLES]
    r = vars[2*N_CIRCLES:]
    
    con = np.empty(4*N_CIRCLES + len(I_IDX))
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    con[:N_CIRCLES] = x - r
    con[N_CIRCLES:2*N_CIRCLES] = 1 - x - r
    con[2*N_CIRCLES:3*N_CIRCLES] = y - r
    con[3*N_CIRCLES:4*N_CIRCLES] = 1 - y - r
    
    # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    dx = x[I_IDX] - x[J_IDX]
    dy = y[I_IDX] - y[J_IDX]
    dr = r[I_IDX] + r[J_IDX]
    con[4*N_CIRCLES:] = dx**2 + dy**2 - dr**2
    return con

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    bounds = [(0, 1)]*N_CIRCLES + [(0, 1)]*N_CIRCLES + [(0, 0.5)]*N_CIRCLES
    
    inits = []
    
    # 1. Hexagonal lattice initialization
    hex_pts = []
    r0 = 0.1
    rows = [5, 6, 5, 6, 4]
    for r_idx, cnt in enumerate(rows):
        y = r_idx * np.sqrt(3) * r0
        x_start = (0 if r_idx % 2 == 0 else 1) * r0
        for k in range(cnt):
            hex_pts.append((x_start + k * 2*r0, y))
            
    hex_arr = np.array(hex_pts)
    min_p = hex_arr.min(axis=0)
    max_p = hex_arr.max(axis=0)
    range_p = max_p - min_p
    # Scale to fit comfortably within [0.1, 0.9]
    target_range = 0.8
    scale = target_range / range_p
    hex_arr = (hex_arr - min_p) * scale + 0.1
    
    inits.append(np.concatenate([hex_arr[:, 0], hex_arr[:, 1], np.full(N_CIRCLES, 0.1)]))
    
    # 2. Random initializations
    for seed in range(5):
        np.random.seed(seed)
        rx = np.random.uniform(0.15, 0.85, N_CIRCLES)
        ry = np.random.uniform(0.15, 0.85, N_CIRCLES)
        inits.append(np.concatenate([rx, ry, np.full(N_CIRCLES, 0.08)]))
        
    best_sum_r = 0.0
    best_vars = None
    
    for init in inits:
        try:
            res = minimize(get_objective, init, method='SLSQP', bounds=bounds, 
                           constraints={'type': 'ineq', 'fun': get_constraints},
                           options={'maxiter': 5000, 'ftol': 1e-12, 'disp': False})
            if not np.isnan(res.fun) and -res.fun > best_sum_r:
                best_sum_r = -res.fun
                best_vars = res.x
        except Exception:
            continue
            
    if best_vars is None:
        best_vars = inits[0]
        
    cx = best_vars[:N_CIRCLES]
    cy = best_vars[N_CIRCLES:2*N_CIRCLES]
    cr = best_vars[2*N_CIRCLES:]
    
    centers = np.column_stack((cx, cy))
    radii = cr
    return centers, radii, np.sum(radii)
