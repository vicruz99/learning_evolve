import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def compute_penalized_objective(vars, lam):
    """
    Objective function with penalty for overlaps.
    Minimizes -sum(radii) + lambda * sum(violations^2)
    """
    x = vars[:N_CIRCLES]
    y = vars[N_CIRCLES:2*N_CIRCLES]
    r = vars[2*N_CIRCLES:]
    
    obj = -np.sum(r)
    penalty = 0.0
    
    # Check overlaps between all pairs
    for i in range(N_CIRCLES):
        for j in range(i + 1, N_CIRCLES):
            dx = x[i] - x[j]
            dy = y[i] - y[j]
            dist = np.sqrt(dx*dx + dy*dy)
            v = r[i] + r[j] - dist
            if v > 0:
                penalty += v**2
                
    return obj + lam * penalty

def run_packing():
    # 1. Initialization: 5x5 grid + 1 extra circle
    # This provides a dense, structured starting point
    x_grid = np.array([0.1, 0.3, 0.5, 0.7, 0.9])
    y_grid = np.array([0.1, 0.3, 0.5, 0.7, 0.9])
    
    x = np.tile(x_grid, 5)
    y = np.repeat(y_grid, 5)
    
    # Place the 26th circle in a gap to avoid immediate massive overlap
    x = np.append(x, 0.55)
    y = np.append(y, 0.55)
    
    r = np.full(N_CIRCLES, 0.09)
    
    # Concatenate variables: [x_0..x_25, y_0..y_25, r_0..r_25]
    vars0 = np.concatenate([x, y, r])
    
    # Bounds: coordinates in [0, 1], radii in [0, 0.5]
    bounds = [(0.0, 1.0)] * (2 * N_CIRCLES) + [(0.0, 0.5)] * N_CIRCLES
    
    # Optimization
    # Use a large penalty lambda to enforce non-overlap strictly
    lam = 10000.0
    res = minimize(compute_penalized_objective, vars0, args=(lam,), 
                   method='L-BFGS-B', bounds=bounds, 
                   options={'maxiter': 5000, 'ftol': 1e-15})
    
    # Extract results
    x_opt = res.x[:N_CIRCLES]
    y_opt = res.x[N_CIRCLES:2*N_CIRCLES]
    r_opt = res.x[2*N_CIRCLES:]
    
    # Ensure radii are strictly positive
    r_opt = np.maximum(r_opt, 1e-9)
    
    centers = np.column_stack([x_opt, y_opt])
    total_sum = np.sum(r_opt)
    
    return centers, r_opt, total_sum
