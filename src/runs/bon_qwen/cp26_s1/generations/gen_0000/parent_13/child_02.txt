import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist

def objective(v):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def constraint(v, triu_idx):
    """
    Constraint function: returns array of values >= 0.
    Includes boundary constraints and pairwise non-overlap constraints.
    """
    n = len(v) // 3
    centers = v[:2*n].reshape(n, 2)
    radii = v[2*n:]
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    c_bound = np.empty(4*n)
    c_bound[0::4] = centers[:, 0] - radii
    c_bound[1::4] = 1.0 - centers[:, 0] - radii
    c_bound[2::4] = centers[:, 1] - radii
    c_bound[3::4] = 1.0 - centers[:, 1] - radii
    
    # Pairwise constraints: dist >= r_i + r_j
    dists = pdist(centers)
    r_sum = radii[triu_idx[0]] + radii[triu_idx[1]]
    
    return np.concatenate((c_bound, dists - r_sum))

def run_packing():
    n = 26
    triu_idx = np.triu_indices(n, k=1)

    bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 1.0)] * n
    cons = {'type': 'ineq', 'fun': constraint, 'args': (triu_idx,)}

    for seed in range(25):

    return best_centers, best_radii, best_sum
