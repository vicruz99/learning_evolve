from scipy.optimize import minimize, differential_evolution

    centers = np.array(centers[:n])

    return centers, radii

def objective(params, n):
    """Objective function: Minimize negative sum of radii plus penalty for constraints."""
    r = params[2*n:]
    x = params[0:n]
    y = params[n:2*n]
    
    # Penalty parameters
    p_boundary = 1e5
    p_overlap = 1e6
    
    cost = -np.sum(r)
    
    # Boundary penalties
    for i in range(n):
        if x[i] - r[i] < 0:
            cost -= p_boundary * (x[i] - r[i])**2
        if x[i] + r[i] > 1:
            cost -= p_boundary * (x[i] + r[i] - 1)**2
        if y[i] - r[i] < 0:
            cost -= p_boundary * (y[i] - r[i])**2
        if y[i] + r[i] > 1:
            cost -= p_boundary * (y[i] + r[i] - 1)**2
        if r[i] < 0:
            cost -= p_boundary * r[i]**2

            min_dist = r[i] + r[j]

def refine_packing(centers, radii):
    """Refines the packing using SLSQP to ensure strict validity and optimality."""
    n = len(radii)
    params = np.concatenate([centers[:, 0], centers[:, 1], radii])

    bounds = [(0, 1)] * n + [(0, 1)] * n + [(0, 0.5)] * n
    
    # Constraints for SLSQP
    cons = []
    
    # Boundary constraints
    for i in range(n):
        # x - r >= 0
        cons.append({'type': 'ineq', 'fun': lambda p, i=i: p[i] - p[2*n+i]})
        # x + r <= 1
        cons.append({'type': 'ineq', 'fun': lambda p, i=i: 1 - p[i] - p[2*n+i]})
        # y - r >= 0
        cons.append({'type': 'ineq', 'fun': lambda p, i=i: p[n+i] - p[2*n+i]})
        # y + r <= 1
        cons.append({'type': 'ineq', 'fun': lambda p, i=i: 1 - p[n+i] - p[2*n+i]})

    # Non-overlap constraints
    for i in range(n):
        for j in range(i + 1, n):

                xj, yj = p[j], p[n+j]
                ri, rj = p[2*n+i], p[2*n+j]
                dist_sq = (xi - xj)**2 + (yi - yj)**2
                return dist_sq - (ri + rj)**2
            cons.append({'type': 'ineq', 'fun': dist_constraint})

    result = minimize(
        lambda p: -np.sum(p[2*n:]), 
        params, 
        method='SLSQP', 
        bounds=bounds, 
        constraints=cons,
        options={'maxiter': 1000, 'ftol': 1e-12}
    )

        return np.column_stack([opt_x, opt_y]), opt_r
