import numpy as np
from scipy.optimize import minimize
from scipy.optimize import linprog

def run_packing():
    """
    Pack 26 circles in a unit square [0,1]x[0,1] to maximize sum of radii.
    Returns (centers, radii, sum_radii).
    """
    n = 26
    
    # --- Helper Functions ---
    
    def generate_hex_initial(r_est):
        """Generates a hexagonal grid of points within the unit square."""
        points = []
        d = 2 * r_est
        h = d * np.sqrt(3) / 2
        row = 0
        
        while True:
            y = r_est + row * h
            if y + r_est > 1.0:
                break
            offset = (d / 2.0) if row % 2 == 1 else 0.0
            col = 0
            while True:
                x = r_est + col * d + offset
                if x + r_est > 1.0:
                    break
                points.append((x, y))
                col += 1
            row += 1
        
        # Ensure we have exactly n points
        if len(points) > n:
            points = points[:n]
        return np.array(points)

    def objective(vars_arr):
        """Objective function: minimize negative sum of radii + penalty for constraints."""
        c_x = vars_arr[0::3]
        c_y = vars_arr[1::3]
        r = vars_arr[2::3]
        
        obj_val = -np.sum(r)
        penalty = 0.0
        
        # Overlap penalty
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt((c_x[i] - c_x[j])**2 + (c_y[i] - c_y[j])**2 + 1e-15)
                overlap = (r[i] + r[j]) - dist
                if overlap > 0:
                    penalty += overlap**2
        
        # Boundary penalty
        for i in range(n):
            vals = [r[i] - c_x[i], r[i] - (1.0 - c_x[i]), r[i] - c_y[i], r[i] - (1.0 - c_y[i])]
            penalty += sum(v**2 for v in vals if v > 0)
            
        return obj_val + 1000.0 * penalty

    def get_valid_radii(centers):
        """
        Given fixed centers, solve LP to find max sum of radii.
        Maximize sum(r_i)
        s.t. r_i + r_j <= dist(i, j)
             r_i <= dist_to_boundary(i)
             r_i >= 0
        """
        # dist matrix
        dists = np.zeros((n, n))
        for i in range(n):
            for j in range(i+1, n):
                d = np.sqrt(np.sum((centers[i] - centers[j])**2))
                dists[i, j] = d
                dists[j, i] = d
        
        # Boundary constraints
        b = np.min([centers[:, 0], 1 - centers[:, 0], centers[:, 1], 1 - centers[:, 1]], axis=0)
        
        # LP Setup: Maximize c^T x => Minimize -c^T x
        c_obj = np.ones(n) * -1
        
        # Inequality constraints: A_ub @ x <= b_ub
        A_rows = []
        b_rows = []
        
        # 1. r_i + r_j <= dist(i, j)
        for i in range(n):
            for j in range(i + 1, n):
                row = np.zeros(n)
                row[i] = 1
                row[j] = 1
                A_rows.append(row)
                b_rows.append(dists[i, j])
                
        # 2. r_i <= b_i
        A_rows.append(np.eye(n))
        b_rows.extend(b)
        
        A_ub = np.array(A_rows)
        b_ub = np.array(b_rows)
        
        # Bounds: r_i >= 0
        bounds = [(0, None) for _ in range(n)]
        
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return -res.fun, res.x # Return sum of radii, and radii array
        else:
            return 0, np.zeros(n)

    # --- Main Logic ---
    best_sum = 0
    best_centers = None
    best_radii = None
    
    # Try a few different initial densities
    initial_r_estimates = [0.11, 0.10, 0.09]
    
    for r_est in initial_r_estimates:
        centers_init = generate_hex_initial(r_est)
        
        # Flatten to optimization variables
        x0 = np.zeros(3 * n)
        for i in range(n):
            x0[3*i] = centers_init[i, 0]
            x0[3*i+1] = centers_init[i, 1]
            x0[3*i+2] = r_est
            
        bounds_opt = [(0, 1)] * (2*n) + [(0, 0.5)] * n
        
        try:
            res = minimize(objective, x0, method='L-BFGS-B', bounds=bounds_opt, 
                           options={'ftol': 1e-9, 'maxiter': 2000})
            
            # Extract centers from optimization result
            opt_centers = np.column_stack((res.x[0::3], res.x[1::3]))
            
            # Solve LP for exact valid radii
            valid_sum, radii = get_valid_radii(opt_centers)
            
            if valid_sum > best_sum:
                best_sum = valid_sum
                best_centers = opt_centers
                best_radii = radii
                
        except Exception:
            continue
            
    # Fallback if optimization fails
    if best_centers is None:
        best_centers = generate_hex_initial(0.09)
        _, best_radii = get_valid_radii(best_centers)
        best_sum = np.sum(best_radii)

    return best_centers, best_radii, float(best_sum)
