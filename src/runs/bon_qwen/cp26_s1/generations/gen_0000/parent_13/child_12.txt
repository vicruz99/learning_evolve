import numpy as np
from scipy.optimize import minimize


def create_hexagonal_initial(n, r, seed=0):
    """Create hexagonal grid initial placement."""
    centers = np.zeros((n, 2))
    idx = 0
    np.random.seed(seed)
    
    for row in range(7):
        for col in range(7):
            if idx >= n:
                break
            x = r + col * 2 * r
            y = r + row * r * np.sqrt(3)
            if row % 2 == 1:
                x += r
            if x + r <= 1 + 1e-10 and y + r <= 1 + 1e-10:
                centers[idx, 0] = x
                centers[idx, 1] = y
                idx += 1
        if idx >= n:
            break
    
    # If not enough placed, fill remaining with random valid positions
    while idx < n:
        for _ in range(1000):
            x = np.random.uniform(r, 1 - r)
            y = np.random.uniform(r, 1 - r)
            valid = True
            for k in range(idx):
                dist = np.sqrt((centers[k, 0] - x)**2 + (centers[k, 1] - y)**2)
                if dist < 2 * r - 1e-6:
                    valid = False
                    break
            if valid:
                centers[idx] = [x, y]
                idx += 1
                break
    
    return centers


def objective_function(vars_arr, n):
    """Objective: negative sum of radii."""
    return -np.sum(vars_arr[2*n:])


def boundary_constraint(vars_arr, n):
    """Circles inside unit square."""
    centers = vars_arr[:2*n].reshape(n, 2)
    radii = vars_arr[2*n:]
    con = []
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        con.append(x - r)
        con.append(1.0 - x - r)
        con.append(y - r)
        con.append(1.0 - y - r)
    return np.array(con)


def overlap_constraint(vars_arr, n):
    """No circle overlaps."""
    centers = vars_arr[:2*n].reshape(n, 2)
    radii = vars_arr[2*n:]
    con = []
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy + 1e-14)
            con.append(dist - radii[i] - radii[j])
    return np.array(con)


def force_simulation(centers, radii, n, steps=5000):
    """Force-directed simulation to improve packing."""
    centers = centers.copy()
    radii = radii.copy()
    
    dt = 0.005
    for step in range(steps):
        forces = np.zeros((n, 2))
        cooling = 1.0 - step / steps
        expand_rate = 0.0003 * cooling
        
        # Expand radii gradually
        radii += expand_rate
        
        # Repulsion forces between circles
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                if dist < 1e-8:
                    dist = 1e-8
                    dx = 1e-4
                    dy = 0
                required = radii[i] + radii[j]
                if dist < required:
                    force_mag = (required - dist) / dist * 10.0
                    fx = dx * force_mag
                    fy = dy * force_mag
                    forces[i, 0] += fx
                    forces[i, 1] += fy
                    forces[j, 0] -= fx
                    forces[j, 1] -= fy
        
        # Apply forces
        centers += forces * dt * cooling
        
        # Clamp to boundaries and reduce radii if needed
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            # Boundary clamping
            min_x = r
            max_x = 1.0 - r
            min_y = r
            max_y = 1.0 - r
            
            if x < min_x:
                x = min_x
                radii[i] = max(radii[i] - 0.001, 1e-6)
            if x > max_x:
                x = max_x
                radii[i] = max(radii[i] - 0.001, 1e-6)
            if y < min_y:
                y = min_y
                radii[i] = max(radii[i] - 0.001, 1e-6)
            if y > max_y:
                y = max_y
                radii[i] = max(radii[i] - 0.001, 1e-6)
            
            centers[i, 0] = x
            centers[i, 1] = y
        
        # Reduce radii if overlapping after clamping
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                if dist < radii[i] + radii[j] and dist > 0:
                    excess = (radii[i] + radii[j] - dist) / 2.0
                    radii[i] = max(radii[i] - excess, 1e-6)
                    radii[j] = max(radii[j] - excess, 1e-6)
    
    return centers, radii


def run_packing():

    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Try multiple initial configurations
    configs_to_try = []

    # Config 2: Hexagonal with r=0.09
    configs_to_try.append((0.09, 1))
    # Config 3: Hexagonal with r=0.1
    configs_to_try.append((0.10, 2))
    # Config 4: Slightly perturbed
    configs_to_try.append((0.092, 3))
    # Config 5: Another perturbation
    configs_to_try.append((0.088, 4))
    
    for r_init, seed in configs_to_try:
        centers = create_hexagonal_initial(n, r_init, seed=seed)
        radii = np.ones(n) * r_init * 0.92
        
        # Phase 1: Force simulation
        centers_sim, radii_sim = force_simulation(centers, radii, n, steps=3000)
        
        # Phase 2: SLSQP optimization
        x0 = np.concatenate([centers_sim.flatten(), radii_sim])
        
        bounds = [(0.0, 1.0)] * (2 * n) + [(1e-8, 0.5)] * n
        
        constraints = [
            {'type': 'ineq', 'fun': lambda v: boundary_constraint(v, n)},
            {'type': 'ineq', 'fun': lambda v: overlap_constraint(v, n)}
        ]
        
        try:
            result = minimize(
                lambda v: objective_function(v, n),
                x0,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 2000, 'ftol': 1e-13, 'disp': False}
            )
            
            c_opt = result.x[:2*n].reshape(n, 2)
            r_opt = result.x[2*n:]
            r_opt = np.maximum(r_opt, 1e-8)
            
            current_sum = np.sum(r_opt)
        except Exception:
            c_opt = centers_sim
            r_opt = radii_sim
            current_sum = np.sum(r_opt)
        
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = c_opt.copy()
            best_radii = r_opt.copy()
    
    # Post-processing: ensure strict feasibility
    # Slightly shrink radii to account for numerical tolerance
    margin = 1e-10
    for i in range(n):
        x, y = best_centers[i]
        r = best_radii[i]
        best_radii[i] = min(r, x + margin, 1.0 - x + margin, y + margin, 1.0 - y + margin)
    
    for i in range(n):
        for j in range(i + 1, n):
            dx = best_centers[i, 0] - best_centers[j, 0]
            dy = best_centers[i, 1] - best_centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            max_allowed = dist - margin

                avg = max_allowed / 2.0
                best_radii[i] = max(avg, 1e-8)
                best_radii[j] = max(avg, 1e-8)
    
    best_sum = np.sum(best_radii)
    best_radii = np.maximum(best_radii, 1e-8)
    
    return best_centers, best_radii, best_sum
