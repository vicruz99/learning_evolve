

<strategy>
I'll use an alternating optimization approach combining Linear Programming (LP) for radius optimization with force-directed relaxation for position optimization.

Key insights:
1. For fixed positions, finding optimal radii is a LINEAR PROGRAM - maximize sum(r_i) subject to r_i <= boundary distances and r_i + r_j <= distance(i,j)
2. For fixed radii, positions can be optimized using repulsive forces when circles overlap
3. This alternating approach converges to a locally optimal packing
4. Multiple random restarts help escape bad local optima

Strategy details:
- Precompute the LP constraint matrix (it only changes structure, not values between iterations)
- Initialize with a hexagonal grid pattern (densest packing in the plane)
- Use the HiGHS LP solver for fast, robust solutions
- Apply simulated annealing-like behavior: add occasional random perturbations to escape local minima
- Gradually decrease the force stiffness to allow fine-tuning
- Run multiple restarts and keep the best result

import numpy as np
from scipy.optimize import linprog


def build_lp_constraint_matrix(n):
    A_ub = []
    for i in range(n):
        row = np.zeros(n)
        row[i] = 1
        A_ub.append(row)
        A_ub.append(row.copy())
        A_ub.append(row.copy())
        A_ub.append(row.copy())
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1
            row[j] = 1
            A_ub.append(row)
    return np.array(A_ub)


def compute_lp_rhs(centers, n):
    n_boundary = 4 * n
    n_overlap = n * (n - 1) // 2
    b_ub = np.zeros(n_boundary + n_overlap)
    idx = 0
    for i in range(n):
        x, y = centers[i]
        b_ub[idx] = x
        b_ub[idx + 1] = 1.0 - x
        b_ub[idx + 2] = y
        b_ub[idx + 3] = 1.0 - y
        idx += 4
    for i in range(n):
        for j in range(i + 1, n):
            diff = centers[i] - centers[j]
            dist = np.sqrt(np.sum(diff * diff))
            b_ub[idx] = dist
            idx += 1
    return b_ub


def solve_lp_for_radii(centers, n, A_ub):
    b_ub = compute_lp_rhs(centers, n)
    c = -np.ones(n)
    bounds = [(0.0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x
    return np.zeros(n)


def initialize_hexagonal_grid(n, spacing=0.12):
    centers = np.zeros((n, 2))
    idx = 0
    row = 0
    col = 0
    while idx < n:
        x = col * spacing + 0.5
        y = row * spacing * np.sqrt(3) / 2.0 + 0.5
        if row % 2 == 1:
            x += spacing / 2.0
        centers[idx] = [x, y]
        idx += 1
        col += 1
        if col > 8 or x > 1.0:
            col = 0
            row += 1
    return centers[:n]


def initialize_square_grid(n):
    centers = np.zeros((n, 2))
    positions = []
    for i in range(6):
        for j in range(6):
            positions.append([0.1 + j * 0.16 + 0.08, 0.1 + i * 0.16 + 0.08])
    for idx in range(min(n, len(positions))):
        centers[idx] = positions[idx]
    return centers[:n]


def initialize_random(n, seed):
    rng = np.random.RandomState(seed)
    centers = rng.uniform(0.15, 0.85, (n, 2))
    return centers


def compute_forces(centers, radii, n, stiffness=10.0):
    forces = np.zeros((n, 2))
    for i in range(n):
        for j in range(i + 1, n):
            diff = centers[i] - centers[j]
            dist_sq = np.sum(diff * diff)
            dist = np.sqrt(dist_sq)
            min_dist = radii[i] + radii[j]
            if dist < min_dist and dist > 1e-10:
                overlap = min_dist - dist
                factor = overlap / dist * stiffness
                forces[i] += factor * diff
                forces[j] -= factor * diff
        for d in range(2):
            if centers[i, d] - radii[i] < -1e-12:
                forces[i, d] += (radii[i] - centers[i, d]) * stiffness
            if centers[i, d] + radii[i] > 1.0 + 1e-12:
                forces[i, d] -= (centers[i, d] + radii[i] - 1.0) * stiffness
    return forces


def clamp_positions(centers, radii, n):
    for i in range(n):
        centers[i, 0] = max(radii[i], min(1.0 - radii[i], centers[i, 0]))
        centers[i, 1] = max(radii[i], min(1.0 - radii[i], centers[i, 1]))
    return centers


def run_packing():
    n = 26
    A_ub = build_lp_constraint_matrix(n)
    
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    initializations = [
        ("hex1", initialize_hexagonal_grid(n, spacing=0.11)),
        ("hex2", initialize_hexagonal_grid(n, spacing=0.13)),
        ("hex3", initialize_hexagonal_grid(n, spacing=0.10)),
        ("sq1", initialize_square_grid(n)),
        ("rand1", initialize_random(n, 42)),
        ("rand2", initialize_random(n, 123)),
        ("rand3", initialize_random(n, 456)),
        ("rand4", initialize_random(n, 789)),
        ("rand5", initialize_random(n, 1001)),
    ]
    
    for init_name, centers in initializations:
        centers = centers.copy()
        centers = np.clip(centers, 0.02, 0.98)
        
        radii = solve_lp_for_radii(centers, n, A_ub)
        
        current_best_sum = np.sum(radii)
        current_best_centers = centers.copy()
        current_best_radii = radii.copy()
        
        for iteration in range(3000):
            radii = solve_lp_for_radii(centers, n, A_ub)
            current_sum = np.sum(radii)
            
            if current_sum > current_best_sum:
                current_best_sum = current_sum
                current_best_centers = centers.copy()
                current_best_radii = radii.copy()
            
            forces = compute_forces(centers, radii, n, stiffness=8.0)
            
            lr = 0.015 / (1.0 + iteration * 0.0008)
            centers += forces * lr
            
            centers = clamp_positions(centers, radii, n)
            
            # Add small random perturbation occasionally to escape local optima
            if iteration % 100 == 0 and iteration > 0:
                perturbation_scale = 0.002 * (1.0 - iteration / 3000.0)
                centers += np.random.randn(n, 2) * perturbation_scale
                centers = clamp_positions(centers, radii, n)
        
        if current_best_sum > best_sum:
            best_sum = current_best_sum
            best_centers = current_best_centers.copy()
            best_radii = current_best_radii.copy()
    
    # Final refinement with tighter constraints
    if best_centers is not None:
        centers = best_centers.copy()
        for iteration in range(2000):
            radii = solve_lp_for_radii(centers, n, A_ub)
            current_sum = np.sum(radii)
            if current_sum > best_sum:
                best_sum = current_sum
                best_centers = centers.copy()
                best_radii = radii.copy()
            forces = compute_forces(centers, radii, n, stiffness=5.0)
            lr = 0.005 / (1.0 + iteration * 0.001)
            centers += forces * lr
            centers = clamp_positions(centers, radii, n)
            if iteration % 50 == 0:
                centers += np.random.randn(n, 2) * 0.001
                centers = clamp_positions(centers, radii, n)
    
    return best_centers, best_radii, best_sum
