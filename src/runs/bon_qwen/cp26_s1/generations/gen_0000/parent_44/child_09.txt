import numpy as np
import math

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii
    using a force-directed simulation.
    """
    n_circles = 26
    centers = np.zeros((n_circles, 2))
    radii = np.zeros(n_circles)
    
    # 1. Initialization: Staggered Grid (Hexagonal Lattice)
    # Distribute 26 circles roughly into 5 rows
    cols = 5
    rows = 6  # 5x5=25, 5x6=30, so 6 rows works well for staggered
    row_height = 1.0 / (rows - 1) if rows > 1 else 1.0
    col_width = 1.0 / (cols - 1) if cols > 1 else 1.0
    
    idx = 0
    for r in range(rows):
        # Stagger even/odd rows
        offset = col_width / 2 if r % 2 == 1 else 0
        # Number of circles in this row
        n_in_row = cols if r % 2 == 0 else cols - 1
        
        for c in range(n_in_row):
            if idx >= n_circles:
                break
            # Scale positions to fit within [0, 1] with some initial padding
            x = (c + 0.5) * col_width + offset
            y = r * row_height
            # Keep within bounds initially
            x = np.clip(x, 0.05, 0.95)
            y = np.clip(y, 0.05, 0.95)
            
            centers[idx] = [x, y]
            radii[idx] = 0.05 # Small initial radius
            idx += 1
        if idx >= n_circles:
            break

    # 2. Force-Directed Simulation
    dt = 0.01
    k_repulsion = 1.0
    k_boundary = 1.5
    radius_growth = 0.0005
    
    # Run simulation for a number of steps
    for step in range(4000):
        forces = np.zeros_like(centers)
        
        # Apply repulsion between circles
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                diff = centers[i] - centers[j]
                dist = np.linalg.norm(diff)
                
                # Prevent division by zero
                if dist < 1e-6:
                    dist = 1e-6
                    diff = np.array([1e-6, 0.0])
                
                # Required distance is sum of radii
                min_dist = radii[i] + radii[j]
                
                if dist < min_dist:
                    # Repulsive force proportional to overlap
                    overlap = min_dist - dist
                    # Normalize direction
                    direction = diff / dist
                    # Force magnitude
                    force_mag = k_repulsion * overlap
                    forces[i] += direction * force_mag
                    forces[j] -= direction * force_mag

        # Apply boundary repulsion
        for i in range(n_circles):
            x, y = centers[i]
            r = radii[i]
            
            # Left wall
            if x - r < 0:
                forces[i, 0] += k_boundary * (r - x)
            # Right wall
            if x + r > 1:
                forces[i, 0] -= k_boundary * (x + r - 1)
            # Bottom wall
            if y - r < 0:
                forces[i, 1] += k_boundary * (r - y)
            # Top wall
            if y + r > 1:
                forces[i, 1] -= k_boundary * (y + r - 1)

        # Update positions
        centers += forces * dt
        
        # Grow radii
        radii += radius_growth
        
        # Clamp radii to be positive
        radii = np.maximum(radii, 1e-6)
        
        # Periodically dampen forces to stabilize
        if step % 1000 == 0 and step > 0:
            dt *= 0.8
            radius_growth *= 0.8

    # 3. Final Adjustment and Validation
    # Ensure strict validity by slightly shrinking if necessary
    for i in range(n_circles):
        x, y = centers[i]
        r = radii[i]
        
        # Boundary fix
        r = min(r, x, 1 - x, y, 1 - y)
        
        # Overlap fix (reduce radius)
        for j in range(n_circles):
            if i != j:
                dist = np.linalg.norm(centers[i] - centers[j])
                min_r_sum = dist
                if radii[i] + radii[j] > min_r_sum:
                    # Reduce radii proportionally or just clip
                    radii[i] = min(radii[i], (min_r_sum - radii[j]) * 0.999)

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
