import numpy as np
from scipy.optimize import minimize

def compute_penalty(centers, radii, n):
    """
    Computes the sum of squared constraint violations.
    Violations occur if circles overlap or extend outside the unit square.
    """
    # Boundary violations: x >= r, x <= 1-r, y >= r, y <= 1-r
    vx = np.maximum(0, radii - centers[:, 0]) + np.maximum(0, radii + centers[:, 0] - 1.0)
    vy = np.maximum(0, radii - centers[:, 1]) + np.maximum(0, radii + centers[:, 1] - 1.0)
    bound_pen = np.sum(vx**2 + vy**2)
    
    # Overlap violations: dist(i,j) >= r_i + r_j
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    overlaps = np.maximum(0, radii[:, np.newaxis] + radii[np.newaxis, :] - dists)
    # Sum only upper triangle to avoid double counting
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    overlap_pen = np.sum(overlaps[mask]**2)
    
    return bound_pen + overlap_pen

def objective(vars, n, pen_weight):
    """
    Objective function to minimize: -sum(radii) + penalty.
    Minimizing this is equivalent to maximizing sum(radii) while penalizing constraint violations.
    """
    centers = vars[:2*n].reshape((n, 2))
    radii = vars[2*n:]
    pen = compute_penalty(centers, radii, n)
    return -np.sum(radii) + pen_weight * pen

def run_packing():
    np.random.seed(42)
    n = 26
    best_centers = None
    best_radii = None
    best_sum = -np.inf
    
    # Generate diverse initial configurations
    configs = []
    
    # 1. Hexagonal lattice (high density baseline)
    hex_pts = []
    s = 0.11
    for j in range(10):
        y = j * s * np.sqrt(3) + s
        if y > 1 - s: break
        for i in range(10):
            x = i * s + s
            if j % 2 == 1: x += s / 2
            if x <= 1 - s: hex_pts.append([x, y])
    if len(hex_pts) < n:
        extra = np.random.rand(n - len(hex_pts), 2) * 0.8 + 0.1
        hex_pts = np.vstack([np.array(hex_pts), extra])
    configs.append(np.array(hex_pts[:n]))
    
    # 2. Random initialization
    configs.append(np.random.rand(n, 2) * 0.8 + 0.1)
    
    # 3. Perturbed 5x5 grid
    grid = np.array([[i/5 + 0.1, j/5 + 0.1] for i in range(5) for j in range(5)])
    grid = np.vstack([grid, np.array([0.5, 0.5])])
    configs.append(grid[:n])
    
    bnds = [(0, 1)] * (2*n) + [(0, 0.5)] * n
    pen_weight = 3000.0
    
    # Phase 1: Local Optimization with Multiple Restarts
    for cfg in configs:
        for _ in range(5):
            centers = cfg + np.random.randn(n, 2) * 0.005
            radii = np.full(n, 0.06)
            vars0 = np.concatenate([centers.flatten(), radii])
            
            res = minimize(objective, vars0, args=(n, pen_weight), method='L-BFGS-B',
                          bounds=bnds, options={'maxiter': 1000, 'ftol': 1e-14})
            
            c = res.x[:2*n].reshape((n, 2))
            r = res.x[2*n:]
            pen = compute_penalty(c, r, n)
            
            if pen < 1e-8:
                s = np.sum(r)
                if s > best_sum:
                    best_sum = s
                    best_centers = c.copy()
                    best_radii = r.copy()
                    
    # Phase 2: Radial Expansion to push boundaries
    if best_centers is not None:
        curr_c, curr_r = best_centers, best_radii
        for _ in range(40):
            expansion = 1.0015
            curr_r *= expansion
            
            vars0 = np.concatenate([curr_c.flatten(), curr_r])
            res = minimize(objective, vars0, args=(n, pen_weight * 2), method='L-BFGS-B',
                          bounds=bnds, options={'maxiter': 600, 'ftol': 1e-14})
            
            new_c = res.x[:2*n].reshape((n, 2))
            new_r = res.x[2*n:]
            pen = compute_penalty(new_c, new_r, n)
            
            if pen < 1e-8:
                curr_c, curr_r = new_c, new_r
                best_sum = np.sum(curr_r)
            else:
                break
                
        best_centers, best_radii = curr_c, curr_r
        best_sum = np.sum(best_radii)

    return best_centers, best_radii, best_sum
