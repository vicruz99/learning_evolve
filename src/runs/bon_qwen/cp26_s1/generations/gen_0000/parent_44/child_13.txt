        r = np.ones(n) * 0.01
        best_valid_c = c
        best_valid_r = r

    # Final validation
    if get_sum_radii(best_valid_c, best_valid_r) > 0:
        return best_valid_c, best_valid_r, best_valid_sum
    else:
        # Return a safe small packing
        c = np.random.rand(n, 2)
        r = np.ones(n) * 0.001
        return c, r, np.sum(r)
