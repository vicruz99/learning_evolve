import numpy as np
from scipy.optimize import minimize


def create_hexagonal_packing(n, r_init, shift_x=0.0, shift_y=0.0):
    """Create an initial hexagonal grid packing."""
    centers = []
    y = r_init + shift_y
    row = 0
    while len(centers) < n:
        x = r_init + (row % 2) * r_init + shift_x
        while x <= 1 - r_init and len(centers) < n:
            centers.append([x, y])
            x += 2 * r_init
        y += r_init * np.sqrt(3)
        row += 1
    return np.array(centers[:n])


def penalized_objective(v, n, penalty):
    """Compute objective with penalty terms for constraint violations."""
    obj = -np.sum(v[2 * n: 2 * n + n])
    
    for i in range(n):
        x = v[2 * i]
        y = v[2 * i + 1]
        r = v[2 * n + i]
        
        # Boundary violations
        if r > x:
            obj += penalty * (r - x) ** 2
        if r + x > 1:
            obj += penalty * (r + x - 1) ** 2
        if r > y:
            obj += penalty * (r - y) ** 2
        if r + y > 1:
            obj += penalty * (r + y - 1) ** 2
    
    # Pairwise non-overlap violations
    for i in range(n):
        xi = v[2 * i]
        yi = v[2 * i + 1]
        ri = v[2 * n + i]
        for j in range(i + 1, n):
            xj = v[2 * j]
            yj = v[2 * j + 1]
            rj = v[2 * n + j]
            dx = xi - xj
            dy = yi - yj
            dist_sq = dx * dx + dy * dy
            min_dist = ri + rj
            min_dist_sq = min_dist * min_dist
            if dist_sq < min_dist_sq:
                dist = np.sqrt(dist_sq)
                overlap = min_dist - dist
                obj += penalty * overlap ** 2
    
    return obj


def fix_packing_validity(centers, radii, n):
    """Post-process to ensure all constraints are satisfied."""
    radii = np.maximum(radii, 1e-10)
    
    # Clip centers to valid range
    for i in range(n):
        r = radii[i]
        centers[i, 0] = max(r, min(1 - r, centers[i, 0]))
        centers[i, 1] = max(r, min(1 - r, centers[i, 1]))
    
    # Iteratively reduce radii to eliminate overlaps
    for _ in range(2000):
        max_overlap = 0.0
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                overlap = radii[i] + radii[j] - dist
                if overlap > max_overlap:
                    max_overlap = overlap
        
        if max_overlap < 1e-12:
            break
        
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                overlap = radii[i] + radii[j] - dist
                if overlap > 1e-12:
                    sum_r = radii[i] + radii[j]
                    if sum_r > 0:
                        ratio_i = radii[i] / sum_r
                        ratio_j = radii[j] / sum_r
                        radii[i] -= overlap * ratio_i
                        radii[j] -= overlap * ratio_j
    
    radii = np.maximum(radii, 1e-10)
    return centers, radii


def run_packing():
    n = 26
    
    best_centers = None
    best_radii = None
    best_sum = 0.0
    
    # Try multiple initial configurations
    r_values = [0.085, 0.09, 0.095, 0.10]
    shift_values = [0.0, 0.015, 0.03]
    
    all_configs = []
    for r_init in r_values:
        for sx in shift_values:
            for sy in shift_values:
                centers_init = create_hexagonal_packing(n, r_init, sx, sy)
                if len(centers_init) < n:
                    continue
                radii_init = np.full(n, r_init)
                x0 = np.concatenate([centers_init.flatten(), radii_init])
                all_configs.append(x0)
    
    # Also add some random perturbations of the best-looking config
    if len(all_configs) > 0:
        base_config = all_configs[0]
        for _ in range(10):
            perturbed = base_config.copy()
            perturbed[:2 * n] += np.random.randn(2 * n) * 0.02
            all_configs.append(perturbed)
    
    penalty = 50000.0
    bounds = [(0.0, 1.0)] * (2 * n) + [(1e-8, 0.5)] * n
    
    for x0 in all_configs:
        try:
            result = minimize(
                penalized_objective,
                x0,
                args=(n, penalty),
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 15000, 'ftol': 1e-16, 'gtol': 1e-12}
            )
            
            centers_opt = result.x[:2 * n].reshape((n, 2))
            radii_opt = result.x[2 * n: 2 * n + n]
            
            centers_opt, radii_opt = fix_packing_validity(centers_opt, radii_opt, n)
            
            current_sum = np.sum(radii_opt)
            if current_sum > best_sum:
                best_sum = current_sum
                best_centers = centers_opt.copy()
                best_radii = radii_opt.copy()
        except Exception:
            continue
    
    # If no valid packing found, return a safe default
    if best_centers is None:
        r_safe = 0.08
        centers_init = create_hexagonal_packing(n, r_safe)
        radii_init = np.full(n, r_safe)
        best_centers = centers_init
        best_radii = radii_init
        best_sum = np.sum(radii_init)
    
    return best_centers, best_radii, best_sum
