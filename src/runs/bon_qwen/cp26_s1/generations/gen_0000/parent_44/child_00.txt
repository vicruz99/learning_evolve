import numpy as np
from scipy.optimize import minimize, NonlinearConstraint

def compute_constraints(v, n):
    """Compute all boundary and non-overlap constraints for n circles."""
    n_constraints = 4 * n + n * (n - 1) // 2
    vals = np.zeros(n_constraints)
    idx = 0
    
    # Boundary constraints: 4 per circle
    for i in range(n):
        xi = v[3*i]
        yi = v[3*i+1]
        ri = v[3*i+2]
        vals[idx] = xi - ri
        vals[idx+1] = 1.0 - xi - ri
        vals[idx+2] = yi - ri
        vals[idx+3] = 1.0 - yi - ri
        idx += 4
        
    # Non-overlap constraints: pairwise distance squared >= (r_i + r_j)^2
    for i in range(n):
        xi = v[3*i]
        yi = v[3*i+1]
        ri = v[3*i+2]
        for j in range(i + 1, n):
            xj = v[3*j]
            yj = v[3*j+1]
            rj = v[3*j+2]
            dx = xi - xj
            dy = yi - yj
            dr = ri + rj
            vals[idx] = dx*dx + dy*dy - dr*dr
            idx += 1
    return vals

def get_constraints_26(v):
    """Wrapper for 26 circles to avoid closures."""
    return compute_constraints(v, 26)

def objective_26(v):
    """Objective: minimize negative sum of radii."""
    return -np.sum(v[2::3])

def run_packing():
    n = 26
    
    # Initialization: Hexagonal lattice pattern
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.065
    
    y = 0.08
    row = 0
    for i in range(n):
        # 5 circles per row, alternating horizontal shift for hex packing
        x = 0.08 + (i % 5) * 0.17 + (row % 2) * 0.085
        centers[i] = [x, y]
        if (i + 1) % 5 == 0:
            y += 0.147 # sqrt(3)/2 * 0.17
            row += 1
            
    # Flatten to optimization variable array: [x1, y1, r1, x2, y2, r2, ...]
    x0 = np.zeros(3 * n)
    for i in range(n):
        x0[3*i] = centers[i, 0]
        x0[3*i+1] = centers[i, 1]
        x0[3*i+2] = radii[i]

    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)] * n + [(0.0, 0.5)] * n
    
    # Define constraint object
    cons = NonlinearConstraint(get_constraints_26, 0.0, np.inf)
    
    # Run optimization
    res = minimize(
        objective_26, 
        x0, 
        method='trust-constr', 
        bounds=bounds, 
        constraints=cons, 
        options={'maxiter': 2000, 'verbose': 0, 'finite_diff_rel_step': 'internal'}
    )

        for i in range(n):
            r = opt_radii[i]
            x = np.clip(opt_centers[i, 0], r, 1.0 - r)
            y = np.clip(opt_centers[i, 1], r, 1.0 - r)
            opt_centers[i] = [x, y]
            
        return opt_centers, opt_radii, float(np.sum(opt_radii))
    else:
        # Fallback to initial configuration if optimizer fails
        return centers, radii, float(np.sum(radii))
