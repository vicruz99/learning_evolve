For packing 26 circles to maximize sum of radii in a unit square, I need a sophisticated approach since equal circles would only achieve ~1.91 sum (radius ~0.073 each), but our target is 2.636 (average radius ~0.101).

Key insights:
1. Variable-sized circles are essential - larger circles near corners can touch 2 boundaries, edge circles touch 1 boundary
2. The optimal configuration has circles "locked" together touching neighbors/boundaries
3. Hexagonal packing patterns are efficient for circle arrangements

Strategy:
1. Initialize with a hexagonal grid pattern for good starting geometry
2. Use force-based repulsion to handle overlaps while simultaneously growing radii
3. Use multiple random perturbations to escape local optima
4. Apply a final correction pass to ensure feasibility
5. Try different initialization patterns (hexagonal, corner-focused, random) and pick the best

The force-based optimizer:
- Repulsion forces push overlapping circles apart
- Boundary forces keep circles inside the square
- Radii grow when there's available space (not touching any other circle or boundary)
- Learning rate and growth rate decrease over time for convergence

import numpy as np


def ensure_feasibility(centers, radii, n):
    """Ensure all constraints are satisfied by reducing radii if needed."""
    for _ in range(1000):
        changed = False
        for i in range(n):
            max_r = min(centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])
            for j in range(n):
                if i != j:
                    dx = centers[j, 0] - centers[i, 0]
                    dy = centers[j, 1] - centers[i, 1]
                    dist = np.sqrt(dx * dx + dy * dy)
                    max_r = min(max_r, dist - radii[j])
            max_r = max(0, max_r)
            if max_r < radii[i] - 1e-12:
                radii[i] = max_r
                changed = True
        if not changed:
            break
    return radii


def init_hexagonal(n):
    """Initialize circles in a hexagonal grid pattern."""
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    idx = 0
    row = 0
    while idx < n:
        for col in range(7):
            x = 0.06 + col * 0.145
            y = 0.06 + row * 0.125 + (col % 2) * 0.0625
            if x < 0.94 and y < 0.94:
                centers[idx] = [x, y]
                radii[idx] = 0.06
                idx += 1
                if idx >= n:
                    break
        row += 1
    
    return centers, radii


def init_corner_focused(n):
    """Initialize with larger circles in corners."""
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Place 4 corner circles
    centers[0] = [0.15, 0.15]
    centers[1] = [0.85, 0.15]
    centers[2] = [0.15, 0.85]
    centers[3] = [0.85, 0.85]
    radii[:4] = 0.15
    
    # Fill rest with hexagonal pattern
    idx = 4
    row = 0
    while idx < n:
        for col in range(7):
            x = 0.25 + col * 0.12
            y = 0.25 + row * 0.10 + (col % 2) * 0.05
            if x < 0.75 and y < 0.75:
                centers[idx] = [x, y]
                radii[idx] = 0.05
                idx += 1
                if idx >= n:
                    break
        row += 1
    
    return centers, radii


def init_random(n):
    """Initialize with random positions."""
    np.random.seed(42)
    centers = np.random.uniform(0.1, 0.9, (n, 2))
    radii = np.ones(n) * 0.04
    return centers, radii


def optimize_force_based(centers, radii, n, iterations=8000, seed=0):
    """Optimize using force-based repulsion with growing radii."""
    centers = centers.copy()
    radii = radii.copy()
    
    # Ensure initial feasibility
    radii = ensure_feasibility(centers, radii, n)
    
    repulsion_k = 500
    dt = 0.005
    growth_rate = 0.002
    
    for iteration in range(iterations):
        forces = np.zeros((n, 2))
        
        # Circle-circle repulsion
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[j, 0] - centers[i, 0]
                dy = centers[j, 1] - centers[i, 1]
                dist_sq = dx * dx + dy * dy
                dist = np.sqrt(dist_sq)
                if dist < 1e-10:
                    dist = 1e-10
                    dx, dy = 0.001, 0.001
                
                target_dist = radii[i] + radii[j]
                if dist < target_dist:
                    overlap = target_dist - dist
                    force_mag = repulsion_k * overlap
                    fx = force_mag * dx / dist
                    fy = force_mag * dy / dist
                    forces[i, 0] -= fx
                    forces[i, 1] -= fy
                    forces[j, 0] += fx
                    forces[j, 1] += fy
        
        # Boundary repulsion
        for i in range(n):
            for dim in range(2):
                if centers[i, dim] - radii[i] < 0:
                    forces[i, dim] += repulsion_k * (radii[i] - centers[i, dim])
                if centers[i, dim] + radii[i] > 1:
                    forces[i, dim] -= repulsion_k * (centers[i, dim] + radii[i] - 1)
        
        # Update positions
        centers += dt * forces
        
        # Clamp positions
        for i in range(n):
            centers[i, 0] = max(0.001, min(0.999, centers[i, 0]))
            centers[i, 1] = max(0.001, min(0.999, centers[i, 1]))
        
        # Grow radii
        current_growth = growth_rate * max(0.1, 1 - iteration / iterations)
        for i in range(n):
            max_r = min(centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])
            for j in range(n):
                if i != j:
                    dx = centers[j, 0] - centers[i, 0]
                    dy = centers[j, 1] - centers[i, 1]
                    dist = np.sqrt(dx * dx + dy * dy)
                    max_r = min(max_r, dist - radii[j])
            
            room = max(0, max_r - radii[i])
            radii[i] += current_growth * room
            radii[i] = max(1e-6, min(radii[i], 0.49))
    
    # Final feasibility correction
    radii = ensure_feasibility(centers, radii, n)
    
    return centers, radii, np.sum(radii)


def run_packing():
    n = 26
    best_sum = 0
    best_centers = None
    best_radii = None
    
    # Try multiple initialization strategies
    strategies = [init_hexagonal, init_corner_focused, init_random]
    
    for strat_idx, strategy in enumerate(strategies):
        for restart in range(5):
            np.random.seed(strat_idx * 100 + restart)
            
            centers, radii = strategy(n)
            
            # Random perturbation for diversity
            if restart > 0:
                centers += np.random.normal(0, 0.02, centers.shape)
                centers = np.clip(centers, 0.05, 0.95)
            
            # Optimize
            centers_opt, radii_opt, sum_r = optimize_force_based(centers, radii, n, iterations=6000, seed=strat_idx * 100 + restart)
            
            if sum_r > best_sum:
                best_sum = sum_r
                best_centers = centers_opt.copy()
                best_radii = radii_opt.copy()
    
    # Additional refinement: try perturbing the best solution and re-optimizing
    for refine_round in range(3):
        for restart in range(5):
            np.random.seed(1000 + refine_round * 10 + restart)
            centers = best_centers.copy() + np.random.normal(0, 0.01, best_centers.shape)
            centers = np.clip(centers, 0.05, 0.95)
            radii = best_radii.copy()
            
            centers_opt, radii_opt, sum_r = optimize_force_based(centers, radii, n, iterations=4000, seed=1000 + refine_round * 10 + restart)
            
            if sum_r > best_sum:
                best_sum = sum_r
                best_centers = centers_opt.copy()
                best_radii = radii_opt.copy()
    
    return best_centers, best_radii, best_sum
