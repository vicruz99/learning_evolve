import numpy as np
from scipy.optimize import linprog, minimize
import math

def solve_radii(centers):
    """
    Given a set of centers, solve the LP to find radii that maximize sum of radii.
    Returns (sum_radii, radii_array).
    """
    n = centers.shape[0]
    
    # Objective: maximize sum(r_i) => minimize -sum(r_i)
    c = -np.ones(n)
    
    # Constraints: A_ub @ r <= b_ub
    # 1. Boundary constraints: r_i <= dist(center_i, boundary)
    #    r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    # 2. Pairwise constraints: r_i + r_j <= dist(center_i, center_j)
    
    # Total constraints: 4*n (boundaries) + n*(n-1)/2 (pairs)
    num_pair_constraints = n * (n - 1) // 2
    A_ub = np.zeros((4 * n + num_pair_constraints, n))
    b_ub = np.zeros(4 * n + num_pair_constraints)
    
    idx = 0
    
    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        
        # r_i <= x
        A_ub[idx, i] = 1.0
        b_ub[idx] = x
        idx += 1
        
        # r_i <= 1 - x
        A_ub[idx, i] = 1.0
        b_ub[idx] = 1.0 - x
        idx += 1
        
        # r_i <= y
        A_ub[idx, i] = 1.0
        b_ub[idx] = y
        idx += 1
        
        # r_i <= 1 - y
        A_ub[idx, i] = 1.0
        b_ub[idx] = 1.0 - y
        idx += 1
    
    # Pairwise constraints
    # To save memory and time, we compute distances on the fly or precompute.
    # Precomputing distances matrix
    dists = np.zeros((n, n))
    for i in range(n):
        for j in range(i + 1, n):
            d = math.sqrt((centers[i, 0] - centers[j, 0])**2 + 
                          (centers[i, 1] - centers[j, 1])**2)
            dists[i, j] = d
            dists[j, i] = d
            
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = d
            idx += 1
            
    bounds = [(0, None) for _ in range(n)]
    
    # Solve LP
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            radii = res.x
            sum_r = np.sum(radii)
            return sum_r, radii
        else:
            # If LP fails (unlikely with valid centers), return 0
            return 0.0, np.zeros(n)
    except Exception:
        return 0.0, np.zeros(n)
