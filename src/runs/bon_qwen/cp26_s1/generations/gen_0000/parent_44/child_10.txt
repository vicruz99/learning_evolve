import numpy as np
import scipy.optimize

def compute_obj_and_penalty(vars_arr, lam, n=26):
    """
    Computes the penalized objective function and the raw penalty value.
    vars_arr: flattened array of [x1, y1, ..., xn, yn, r1, ..., rn]
    lam: penalty weight
    n: number of circles
    """
    centers = vars_arr[:2*n].reshape(n, 2)
    radii = vars_arr[2*n:]
    
    penalty = 0.0
    
    # Boundary penalties: circles must be inside [0,1]x[0,1]
    penalty += np.sum(np.maximum(0, radii - centers[:,0])**2)
    penalty += np.sum(np.maximum(0, radii - centers[:,1])**2)
    penalty += np.sum(np.maximum(0, centers[:,0] + radii - 1)**2)
    penalty += np.sum(np.maximum(0, centers[:,1] + radii - 1)**2)
    
    # Overlap penalties: circles must not intersect
    for i in range(n):
        for j in range(i+1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            overlap = radii[i] + radii[j] - dist
            if overlap > 0:
                penalty += overlap**2
                
    obj_val = -np.sum(radii) + lam * penalty
    return obj_val, penalty

                return val
                
            res = scipy.optimize.minimize(
