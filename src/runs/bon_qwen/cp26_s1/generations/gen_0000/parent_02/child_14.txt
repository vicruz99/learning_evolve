import numpy as np

def run_packing():
    np.random.seed(42)
    n = 26
    
    # 1. Initialization: Hexagonal lattice for high-density starting configuration
    pts = []
    r_init = 0.10
    for row in range(7):
        for col in range(7):
            x = col * 2 * r_init + (row % 2) * r_init
            y = row * np.sqrt(3) * r_init
            pts.append([x, y])
    pts = np.array(pts)
    
    # Scale and center the lattice within [0, 1]^2
    min_p = pts.min(axis=0)
    max_p = pts.max(axis=0)
    span = max_p - min_p
    scale = 0.85 / span.max()  # Leave margin for optimization
    centers = (pts - min_p) * scale + 0.5 - 0.5 * scale * span
    centers = centers[:n]
    
    radii = np.full(n, 0.06)  # Start with small radii to allow growth
    vel = np.zeros((n, 2))
    
    # Simulation parameters
    stiffness = 80.0
    damping = 0.88
    dt = 0.01
    growth_rate = 0.0012
    max_iter = 25000
    
    for step in range(max_iter):
        # 2a. Pairwise repulsion forces
        diffs = centers[:, None, :] - centers[None, :, :]
        dists = np.sqrt(np.sum(diffs**2, axis=2))
        np.fill_diagonal(diffs, 0.0)
        np.fill_diagonal(dists, 1.0)  # Prevent division by zero
        
        r_sum = radii[:, None] + radii[None, :]
        overlaps = np.maximum(r_sum - dists, 0.0)
        np.fill_diagonal(overlaps, 0.0)
        
        dirs = diffs / dists[..., None]
        f_mag = overlaps * overlaps * stiffness
        forces = np.sum(f_mag[:, :, None] * dirs, axis=1)
        
        # 2b. Wall repulsion forces
        forces[:, 0] += np.maximum(0.0, radii - centers[:, 0]) * stiffness
        forces[:, 0] -= np.maximum(0.0, centers[:, 0] - (1.0 - radii)) * stiffness
        forces[:, 1] += np.maximum(0.0, radii - centers[:, 1]) * stiffness
        forces[:, 1] -= np.maximum(0.0, centers[:, 1] - (1.0 - radii)) * stiffness
        
        # 2c. Integration
        vel += forces * dt
        vel *= damping
        centers += vel * dt
        
        # Project to valid bounds to prevent escape
        bounds_low = radii[:, None]
        bounds_high = 1.0 - radii[:, None]
        centers = np.clip(centers, bounds_low, bounds_high)
        
        # 2d. Grow radii
        radii += growth_rate
        
        # Anneal growth rate to stabilize
        if step % 2000 == 0 and step > 0:
            growth_rate *= 0.95
            stiffness *= 0.98

    # 3. Final validation and minor correction for numerical precision
    for _ in range(100):
        valid = True
        # Check overlaps
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                if dist < radii[i] + radii[j] - 1e-9:
                    valid = False
                    break
            if not valid: break
            
            # Check boundaries
            if centers[i, 0] < radii[i] - 1e-9 or centers[i, 0] > 1.0 - radii[i] + 1e-9:
                valid = False
                break
            if centers[i, 1] < radii[i] - 1e-9 or centers[i, 1] > 1.0 - radii[i] + 1e-9:
                valid = False
                break
        if valid:
            break
        radii *= 0.9999  # Slight shrink if invalid
        
    sum_radii = float(np.sum(radii))
    return centers, radii, sum_radii
