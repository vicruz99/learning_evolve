N_CIRCLES = 26


def compute_penalty(params):
    n = N_CIRCLES
    centers = params[:2 * n].reshape((n, 2))
    radii = params[2 * n:]
    penalty = 0.0

    for i in range(n):
        x = centers[i, 0]
        y = centers[i, 1]
        r = radii[i]
        margin_left = x - r
        margin_right = 1 - x - r
        margin_bottom = y - r
        margin_top = 1 - y - r
        r_margin = r
        margin = min(margin_left, margin_right, margin_bottom, margin_top, r_margin)
        if margin < 0:
            penalty += 50000.0 * margin * margin

    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            overlap = radii[i] + radii[j] - dist
            if overlap > 0:
                penalty += 50000.0 * overlap * overlap

    return penalty


def objective_func(params):
    n = N_CIRCLES

def fix_feasibility(centers, radii):
    n = N_CIRCLES
    # Ensure non-negative radii
    radii = np.maximum(radii, 1e-12)

    # Clip to boundaries
    for i in range(n):
        r = radii[i]
        centers[i, 0] = np.clip(centers[i, 0], r, 1 - r)
        centers[i, 1] = np.clip(centers[i, 1], r, 1 - r)

    # Fix overlaps iteratively
    for _ in range(200):
        max_overlap = 0
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                overlap = radii[i] + radii[j] - dist
                if overlap > max_overlap:
                    max_overlap = overlap

        if max_overlap < 1e-12:
            break

        # Reduce radii to eliminate overlaps
        reduction = max_overlap / (2 * n) + 1e-14
        radii = np.maximum(radii - reduction, 1e-12)

    # Re-clip to boundaries
    for i in range(n):
        r = radii[i]
        centers[i, 0] = np.clip(centers[i, 0], r, 1 - r)
        centers[i, 1] = np.clip(centers[i, 1], r, 1 - r)
