def compute_objective(v):
    """
    Objective function to minimize: -sum(radii)
    v is a 1D array of shape (3*N_CIRCLES,)
    Ordering: x_1, y_1, r_1, x_2, y_2, r_2, ...
    """
    # Extract radii. They are at indices 2, 5, 8, ... (stride 3, offset 2)
    radii = v[2::3]
    return -np.sum(radii)

def compute_constraints(v):
    """
    Computes constraints for the packing problem.
    Returns a 1D array of constraint values.
    Constraints must be >= 0.
    
    Constraints:
    1. Boundary: x >= r, 1-x >= r, y >= r, 1-y >= r  (4 per circle)
    2. Pairwise: dist(i,j) >= r_i + r_j               (N*(N-1)/2 constraints)
    """
    n = N_CIRCLES
    
    # Reshape v into (N, 3) matrix where columns are x, y, r
    # v is [x1, y1, r1, x2, y2, r2, ...]
    # We can reshape directly if we reorder, but slicing is easier
    # Actually, v is 1D.
    # x's are v[0::3], y's are v[1::3], r's are v[2::3]
    
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    
    centers = np.column_stack((x, y))
    
    constraints_list = []
    
    # 1. Boundary constraints
    # x >= r  => x - r >= 0
    # 1 - x >= r => 1 - x - r >= 0
    # y >= r => y - r >= 0
    # 1 - y >= r => 1 - y - r >= 0
    
    c_bound_x_left = x - r
    c_bound_x_right = 1.0 - x - r
    c_bound_y_bottom = y - r
    c_bound_y_top = 1.0 - y - r
    
    constraints_list.append(c_bound_x_left)
    constraints_list.append(c_bound_x_right)
    constraints_list.append(c_bound_y_bottom)
    constraints_list.append(c_bound_y_top)
    
    # 2. Pairwise constraints
    # dist(i, j) >= r_i + r_j
    # Vectorized distance calculation
    # centers shape (N, 2)
    # diff shape (N, N, 2)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist_sq = np.sum(diff**2, axis=2)
    dist = np.sqrt(dist_sq)
    
    # sum of radii matrix
    sum_r = r[:, np.newaxis] + r[np.newaxis, :]
    
    # We need lower triangle (i < j)
    tri_idx = np.tril_indices(n, k=-1)
    pairwise_constraints = dist[tri_idx] - sum_r[tri_idx]
    
    constraints_list.append(pairwise_constraints)
    
    return np.concatenate(constraints_list)
