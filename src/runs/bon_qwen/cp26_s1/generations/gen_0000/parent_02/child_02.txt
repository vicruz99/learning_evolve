import numpy as np
from scipy.optimize import minimize

def compute_violation(params, n):
    """Computes the sum of squared violations for boundary and overlap constraints."""
    centers = params[:2*n].reshape(n, 2)
    radii = params[2*n:]
    violation = 0.0
    
    # Boundary violations
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x < r:
            violation += (r - x)**2
        if y < r:
            violation += (r - y)**2
        if x > 1 - r:
            violation += (x - (1 - r))**2
        if y > 1 - r:
            violation += (y - (1 - r))**2
            
    # Overlap violations
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.hypot(dx, dy)
            req = radii[i] + radii[j]
            if dist < req:
                violation += (req - dist)**2
                
    return violation

def objective(params, n, penalty):
    """Objective: minimize negative sum of radii + penalty * violations."""
    radii = params[2*n:]
    val = -np.sum(radii)
    if penalty > 0:
        val += penalty * compute_violation(params, n)
    return val

def run_packing():
    n = 26
