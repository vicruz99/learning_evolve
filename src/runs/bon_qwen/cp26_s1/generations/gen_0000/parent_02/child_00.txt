import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes the packing of 26 circles in a unit square to maximize sum of radii.
    """
    np.random.seed(42)
    n_circles = 26
    
    def compute_overlap_penalty(centers, radii):
        # Calculate pairwise distances
        dists = np.sqrt(np.sum((centers[:, np.newaxis, :] - centers[np.newaxis, :, :])**2, axis=2))
        # Overlap constraint: dist >= r1 + r2 => dist - (r1 + r2) >= 0
        # We want to penalize violations where dist < r1 + r2
        sum_radii_matrix = radii[:, np.newaxis] + radii[np.newaxis, :]
        overlaps = np.maximum(0, sum_radii_matrix - dists)
        return np.sum(overlaps**2)

    def compute_boundary_penalty(centers, radii):
        # Constraint: r <= x <= 1-r, r <= y <= 1-r
        # Violation 1: x < r => r - x > 0
        # Violation 2: x > 1-r => x + r - 1 > 0
        # Same for y
        
        left_viol = np.maximum(0, radii - centers[:, 0])**2
        right_viol = np.maximum(0, centers[:, 0] + radii - 1)**2
        bottom_viol = np.maximum(0, radii - centers[:, 1])**2
        top_viol = np.maximum(0, centers[:, 1] + radii - 1)**2
        
        return np.sum(left_viol + right_viol + bottom_viol + top_viol)

    def objective(params, n):
        # params structure: [x1, y1, r1, x2, y2, r2, ...]
        centers = params[:2*n].reshape(n, 2)
        radii = params[2*n:]
        
        # We want to maximize sum of radii, so minimize -sum(radii)
        # Add penalties for constraint violations
        overlap_pen = compute_overlap_penalty(centers, radii)
        boundary_pen = compute_boundary_penalty(centers, radii)
        
        # Weight penalties heavily
        return -np.sum(radii) + 1000 * overlap_pen + 1000 * boundary_pen

    def get_initial_guess():
        # Create a hexagonal-ish grid
        centers = np.zeros((n_circles, 2))
        radii = np.ones(n_circles) * 0.05 # Start small

                centers[idx] = [x_current, y_current]
                radii[idx] = 0.09 # Start near target
                x_current += x_step
                idx += 1
            y_current += y_step

        radii_init += np.random.normal(0, 0.005, radii_init.shape)

        radii_init = np.clip(radii_init, 0.01, 0.2)
        # Clip centers

        initial_params = np.concatenate([centers_init, radii_init])
        
        # Bounds: centers in [0, 1], radii in [0, 0.5]
        bounds = []

            bounds.extend([(0, 1), (0, 1)]) # x, y
            bounds.append((0, 0.5))         # r

        res = minimize(objective, initial_params, args=(n_circles,),
