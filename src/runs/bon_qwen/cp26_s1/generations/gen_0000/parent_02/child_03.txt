import numpy as np
from scipy.optimize import linprog

def solve_packing_lp(centers):
    """
    Solves the linear program to find optimal radii for fixed centers.
    Returns radii, a mask of active constraints, and pair information.
    """
    n = centers.shape[0]
    pairs = []
    b_ub = []
    
    # Build pairwise constraints: r_i + r_j <= distance(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            diff = centers[i] - centers[j]
            d = np.sqrt(np.sum(diff**2))
            pairs.append((i, j, diff, d))
            b_ub.append(d)
            
    b_ub = np.array(b_ub)
    n_pairs = len(pairs)
    
    # Constraint matrix A_ub where each row has 1 at i and j
    A_ub = np.zeros((n_pairs, n))
    for idx, (i, j, _, _) in enumerate(pairs):
        A_ub[idx, i] = 1.0
        A_ub[idx, j] = 1.0
        
    # Boundary constraints handled as variable bounds
    bounds = []
    for i in range(n):
        x, y = centers[i]
        max_r = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max_r))
        
    # Objective: maximize sum(r) -> minimize -sum(r)
    c = np.ones(n) * -1.0
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs', options={'disp': False})
    
    if not res.success:
        return np.zeros(n), np.zeros(n_pairs, dtype=bool), pairs
        
    radii = res.x
    
    # Identify active constraints (touching circles)
    slack = b_ub - np.dot(A_ub, radii)
    active_mask = slack < 1e-4
    
    return radii, active_mask, pairs

def run_packing():

    n = 26

    grid_x = np.linspace(0.12, 0.88, 5)
    grid_y = np.linspace(0.12, 0.88, 5)
    centers = np.array([[x, y] for x in grid_x for y in grid_y])
    centers = np.vstack([centers, [0.5, 0.5]])  # 26th circle
    centers += np.random.uniform(-0.04, 0.04, size=centers.shape)
    centers = np.clip(centers, 0.05, 0.95)
    
    best_centers = centers.copy()
    best_radii = np.zeros(n)
    best_sum = 0.0
    
    for step in range(600):
        radii, active_mask, pairs = solve_packing_lp(centers)
        current_sum = np.sum(radii)
        
        if current_sum > best_sum + 1e-8:
            best_sum = current_sum
            best_centers = centers.copy()
            best_radii = radii.copy()
            
        # Periodic jitter to escape local minima
        if step % 100 == 0 and step > 0:
            centers += np.random.normal(0, 0.01, size=centers.shape)
            centers = np.clip(centers, 0.05, 0.95)
            
        # Compute repulsive forces from active constraints
        forces = np.zeros_like(centers)
        for k in range(len(pairs)):
            if active_mask[k]:
                i, j, diff, d = pairs[k]
                if d > 1e-6:
                    dir_vec = diff / d
                    forces[i] += dir_vec
                    forces[j] -= dir_vec
                    
        # Wall repulsion forces
        for i in range(n):
            x, y = centers[i]
            if x < 0.1: forces[i, 0] += (0.1 - x) * 5.0
            elif x > 0.9: forces[i, 0] -= (x - 0.9) * 5.0
            if y < 0.1: forces[i, 1] += (0.1 - y) * 5.0
            elif y > 0.9: forces[i, 1] -= (y - 0.9) * 5.0
            
        # Normalize forces to ensure stable updates
        norms = np.linalg.norm(forces, axis=1, keepdims=True)
        norms = np.where(norms > 1e-6, norms, 1.0)
        forces_normalized = forces / norms

        step_size = 0.04 / (1.0 + step / 40.0)
        centers += step_size * forces_normalized
        centers = np.clip(centers, 0.05, 0.95)
        
    # Final LP pass to get precise radii for the best configuration
    final_radii, _, _ = solve_packing_lp(best_centers)
    
    return best_centers, final_radii, np.sum(final_radii)
