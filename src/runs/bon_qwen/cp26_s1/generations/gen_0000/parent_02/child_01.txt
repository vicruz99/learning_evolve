import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    """
    n_circles = 26
    
    # 1. Initialization: Hexagonal Lattice
    # We generate a hexagonal grid and select 26 points.
    # Hexagonal spacing factor
    k = 5.5  # Approximate number of circles per side
    rows = 6
    cols = 5
    
    positions = []
    r_init = 0.04 # Initial radius small enough to avoid overlap
    
    # Generate hexagonal points
    # y spacing is sqrt(3)/2 * diameter approx sqrt(3)/2 * 2r = sqrt(3)r
    # But here we just generate coordinates
    for r_idx in range(rows):
        for c_idx in range(cols):
            if len(positions) >= n_circles:
                break
            # Hexagonal offset
            x = (c_idx + 0.5) + (r_idx % 2) * 0.5
            y = r_idx * (np.sqrt(3) / 2)
            
            # Normalize to [0, 1] roughly
            # We'll scale these later, just need relative positions
            positions.append([x, y])
        
        if len(positions) >= n_circles:
            break
            
    positions = np.array(positions[:n_circles])
    
    # Normalize positions to fit well within [0,1]
    # Shift and scale
    min_pos = positions.min(axis=0)
    max_pos = positions.max(axis=0)
    spread = max_pos - min_pos
    # Add padding
    positions = (positions - min_pos) / spread * 0.8 + 0.1
    
    # 2. Optimization Variables
    # We pack all variables into a single array: [x1, y1, r1, x2, y2, r2, ...]
    # Total variables: 3 * 26 = 78
    x0 = np.zeros(3 * n_circles)
    for i in range(n_circles):
        x0[3*i] = positions[i, 0]
        x0[3*i + 1] = positions[i, 1]
        x0[3*i + 2] = r_init

    # 3. Objective Function: Maximize sum of radii -> Minimize negative sum
    def objective(vars):
        radii = vars[2::3]
        return -np.sum(radii)

    # 4. Constraints
    constraints = []
    
    # Boundary constraints for each circle:
    # r <= x <= 1-r  =>  x - r >= 0  and  x + r <= 1
    # r <= y <= 1-r  =>  y - r >= 0  and  y + r <= 1
    
    for i in range(n_circles):
        idx_x = 3*i
        idx_y = 3*i + 1
        idx_r = 3*i + 2
        
        # x - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: v[idx_x] - v[idx_r]
        })
        # 1 - x - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: 1 - v[idx_x] - v[idx_r]
        })
        # y - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: v[idx_y] - v[idx_r]
        })
        # 1 - y - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: 1 - v[idx_y] - v[idx_r]
        })
        
    # Non-overlap constraints:
    # dist(i, j) >= r_i + r_j
    # sqrt((xi-xj)^2 + (yi-yj)^2) - (ri + rj) >= 0
    # To avoid sqrt issues, we can use squared form but it's not convex.
    # Using sqrt is fine for local optimization.
    
    for i in range(n_circles):
        for j in range(i + 1, n_circles):
            idx_xi, idx_yi, idx_ri = 3*i, 3*i+1, 3*i+2
            idx_xj, idx_yj, idx_rj = 3*j, 3*j+1, 3*j+2
            
            def distance_constraint(v, i=i, j=j):
                xi, yi, ri = v[idx_xi], v[idx_yi], v[idx_ri]
                xj, yj, rj = v[idx_xj], v[idx_yj], v[idx_rj]
                dist = np.sqrt((xi - xj)**2 + (yi - yj)**2)
                return dist - (ri + rj)
            
            constraints.append({
                'type': 'ineq',
                'fun': distance_constraint
            })

    # 5. Bounds
    # x, y in [0, 1], r >= 0
    bounds = []
    for _ in range(n_circles):
        bounds.extend([
            (0.0, 1.0), # x
            (0.0, 1.0), # y
            (0.0, 0.5)  # r (max possible radius)
        ])

    # 6. Run Optimization
    # Using SLSQP
    result = minimize(
        objective, 
        x0, 
        method='SLSQP', 
        bounds=bounds, 
        constraints=constraints,
        options={'maxiter': 1000, 'ftol': 1e-9}
    )
    
    # Check if optimization was successful or at least improved
    if result.success or result.fun < -0.01: # Improved from random small radii
        final_vars = result.x
    else:
        # Fallback to initial if it failed (unlikely with valid init)
        final_vars = x0

    # Extract results
    centers = np.array([[final_vars[3*i], final_vars[3*i+1]] for i in range(n_circles)])
    radii = np.array([final_vars[3*i+2] for i in range(n_circles)])
    
    # Ensure non-negative radii (just in case)
    radii = np.maximum(radii, 0.0)
    
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
