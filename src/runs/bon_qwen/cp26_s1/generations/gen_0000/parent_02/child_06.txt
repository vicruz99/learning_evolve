import numpy as np
from scipy.optimize import minimize

def compute_radii(centers):
    """Compute maximal valid radii for each circle given centers."""
    n = centers.shape[0]
    # Distance to walls
    d_walls = np.min([centers[:, 0], 1 - centers[:, 0], centers[:, 1], 1 - centers[:, 1]], axis=0)
    
    # Pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    d_pairs = dists.min(axis=1)
    
    return np.minimum(d_walls, d_pairs / 2.0)

def objective(centers_flat):
    """Objective function: negative sum of radii."""
    centers = centers_flat.reshape(-1, 2)
    r = compute_radii(centers)
    return -np.sum(r)

def get_hex_initial_guess(n):
    """Generate a hexagonal grid initial guess."""
    centers = []
    y = 0.12
    row_sizes = [5, 5, 6, 5, 5]  # Total 26
    for i, size in enumerate(row_sizes):
        x_start = 0.16 if i % 2 == 1 else 0.1
        step = 0.17
        for j in range(size):
            centers.append([x_start + j * step, y])
        y += 0.17 * np.sqrt(3) / 2
    return np.array(centers)

def run_packing():
    n = 26
    best_val = -np.inf
    best_centers = None
    
    # Multiple starting configurations
    starts = [get_hex_initial_guess(n)]
    np.random.seed(42)
    for _ in range(3):
        rand_c = np.random.rand(n, 2) * 0.8 + 0.1
        starts.append(rand_c)
        
    for start in starts:
        centers = start.copy()
        step = 0.015
        
        # Force-based relaxation
        for _ in range(3000):
            diff = centers[np.newaxis, :, :] - centers[:, np.newaxis, :]
            dists = np.sqrt(np.sum(diff**2, axis=2))
            np.fill_diagonal(dists, 1.0)  # Avoid division by zero
            inv_d3 = 1.0 / dists**3
            
            # Pairwise repulsion forces
            forces = np.sum(diff * inv_d3[:, :, np.newaxis], axis=1)
            
            # Boundary repulsion forces
            fx = 1.0 / (centers[:, 0] + 1e-6) - 1.0 / (1 - centers[:, 0] + 1e-6)
            fy = 1.0 / (centers[:, 1] + 1e-6) - 1.0 / (1 - centers[:, 1] + 1e-6)
            forces[:, 0] += fx
            forces[:, 1] += fy
            
            centers += step * forces
            centers = np.clip(centers, 0.005, 0.995)  # Keep strictly inside
            step *= 0.997  # Cooling schedule
            
        # Local optimization refinement
        res = minimize(objective, centers.flatten(), method='Nelder-Mead',

        if curr_val > best_val:
            best_val = curr_val
            best_centers = res.x.reshape(-1, 2)
            
    final_radii = compute_radii(best_centers)
    final_radii = np.maximum(final_radii, 0.0)  # Ensure non-negative
    return best_centers, final_radii, np.sum(final_radii)
