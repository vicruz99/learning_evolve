    *   **Objective**: Minimize the negative sum of radii (equivalent to maximizing sum).

        *   Boundary constraints: $r_i \le x_i \le 1-r_i$ and $r_i \le y_i \le 1-r_i$.
        *   Non-overlap constraints: For every pair of circles $i, j$, the distance between centers must be at least $r_i + r_j$. i.e., $\sqrt{(x_i-x_j)^2 + (y_i-y_j)^2} - (r_i + r_j) \ge 0$.
    *   **Bounds**: $x, y \in [0, 1]$, $r \in [0, 1]$.

import numpy as np
from scipy.optimize import minimize
import math

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    N = 26
    
    # --- 1. Initialization ---
    # Create a hexagonal grid of centers to start close to a dense packing.
    # We aim for a radius of roughly 0.1, so diameter 0.2.
    # Spacing 0.2 horizontally, 0.173 vertically.
    
    centers = np.zeros((N, 2))
    idx = 0
    
    # Hexagonal packing parameters
    # Estimate a reasonable spacing based on 26 circles in 1x1
    # Area per circle ~ 1/26 ~ 0.038. Radius ~ sqrt(0.038/pi) ~ 0.11.
    # Diameter ~ 0.22.
    spacing = 0.22 
    row_height = spacing * math.sqrt(3) / 2
    
    # Start positions with a margin
    margin = 0.1
    y = margin
    row = 0
    
    # Generate points in a hexagonal pattern
    points = []
    while idx < N:
        # Offset every other row
        if row % 2 == 0:
            x_start = margin
        else:
            x_start = margin + spacing / 2
        
        x = x_start
        while x <= 1.0 - margin:
            if idx < N:
                centers[idx] = [x, y]
                points.append([x, y])
                idx += 1
            x += spacing
        
        y += row_height
        row += 1
    
    # If the grid didn't fill N (unlikely with these params), fill with random
    while idx < N:
        centers[idx] = [np.random.uniform(0.2, 0.8), np.random.uniform(0.2, 0.8)]
        idx += 1

    # Initial radii small to ensure feasibility
    radii = np.full(N, 0.01)
    
    # Flatten variables: [x0, y0, r0, x1, y1, r1, ...]
    x0 = np.hstack([centers, radii.reshape(-1, 1)]).flatten()
    
    # --- 2. Optimization Setup ---
    
    # Objective function: maximize sum(radii) -> minimize -sum(radii)
    def objective(vars):
        # Radii are at indices 2, 5, 8, ... i.e., 3*i + 2
        r = vars[2::3]
        return -np.sum(r)
    
    # Constraints
    cons = []
    
    # Helper to create constraint functions to avoid closure variable issues
    def create_boundary_constraints():
        boundary_cons = []
        for i in range(N):
            idx_x = 3 * i
            idx_y = 3 * i + 1
            idx_r = 3 * i + 2
            
            # x_i - r_i >= 0
            boundary_cons.append({'type': 'ineq', 'fun': lambda v, i_x=idx_x, i_r=idx_r: v[i_x] - v[i_r]})
            # 1 - x_i - r_i >= 0
            boundary_cons.append({'type': 'ineq', 'fun': lambda v, i_x=idx_x, i_r=idx_r: 1.0 - v[i_x] - v[i_r]})
            # y_i - r_i >= 0
            boundary_cons.append({'type': 'ineq', 'fun': lambda v, i_y=idx_y, i_r=idx_r: v[i_y] - v[i_r]})
            # 1 - y_i - r_i >= 0
            boundary_cons.append({'type': 'ineq', 'fun': lambda v, i_y=idx_y, i_r=idx_r: 1.0 - v[i_y] - v[i_r]})
        return boundary_cons

    cons.extend(create_boundary_constraints())
    
    # Overlap constraints: dist(i, j) >= r_i + r_j
    # dist - (r_i + r_j) >= 0
    def create_overlap_constraints():
        overlap_cons = []
        for i in range(N):
            for j in range(i + 1, N):
                idx_xi = 3 * i
                idx_yi = 3 * i + 1
                idx_ri = 3 * i + 2
                idx_xj = 3 * j
                idx_yj = 3 * j + 1
                idx_rj = 3 * j + 2
                
                # Capture indices in closure
                def make_overlap(i_xi, i_yi, i_ri, i_xj, i_yj, i_rj):
                    def constraint(v):
                        dist = math.hypot(v[i_xi] - v[i_xj], v[i_yi] - v[i_yj])
                        return dist - (v[i_ri] + v[i_rj])
                    return constraint
                
                overlap_cons.append({'type': 'ineq', 'fun': make_overlap(idx_xi, idx_yi, idx_ri, idx_xj, idx_yj, idx_rj)})
        return overlap_cons

    cons.extend(create_overlap_constraints())
    
    # Bounds for variables
    # x, y in [0, 1], r in [0, 1] (r max 0.5 effectively due to constraints)
    bounds = []
    for _ in range(N):
        bounds.append((0, 1)) # x
        bounds.append((0, 1)) # y
        bounds.append((0, 1)) # r

    # --- 3. Run Optimizer ---
    
    # Use SLSQP which handles non-linear constraints
    # maxiter increased for convergence
    res = minimize(
        objective, 
        x0, 
        method='SLSQP', 
        bounds=bounds, 
        constraints=cons, 
        options={'maxiter': 2000, 'ftol': 1e-12, 'disp': False}
    )
    
    # --- 4. Extract Results ---
    final_vars = res.x
    centers_opt = np.zeros((N, 2))
    radii_opt = np.zeros(N)
    
    for i in range(N):
        centers_opt[i] = [final_vars[3*i], final_vars[3*i+1]]
        radii_opt[i] = final_vars[3*i+2]
        
    sum_radii = np.sum(radii_opt)
    
    return centers_opt, radii_opt, sum_radii
