import numpy as np
from scipy.optimize import minimize

def compute_cost(Z, n, alpha, beta):
    centers = Z[:2*n].reshape((n, 2))
    radii = Z[2*n:]
    cost = -np.sum(radii)
    
    # Boundary constraint violations: x >= r, x <= 1-r, y >= r, y <= 1-r
    v1 = radii - centers[:, 0]
    v2 = radii + centers[:, 0] - 1.0
    v3 = radii - centers[:, 1]
    v4 = radii + centers[:, 1] - 1.0
    boundary_viol = np.maximum(0, np.maximum(v1, v2))
    boundary_viol = np.maximum(boundary_viol, np.maximum(v3, v4))
    cost += beta * np.sum(boundary_viol**2)
    
    # Overlap constraint violations: dist >= r_i + r_j
    c_diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(c_diff**2, axis=2))
    r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    overlaps = r_sum - dists
    overlaps[mask == False] = 0.0
    overlaps = np.maximum(0, overlaps)
    
    cost += alpha * np.sum(overlaps**2)
    return cost

def run_packing():
    n = 26
    best_Z = None
    best_sum = -np.inf
    
    inits = []
    # 1. Structured 5x5 grid + center
    cx, cy = np.meshgrid(np.linspace(0.15, 0.85, 5), np.linspace(0.15, 0.85, 5))

    init1[:25] = pts[:25]
    init1[25] = [0.5, 0.5]
    inits.append(init1)

    np.random.seed(123)
    inits.append(np.random.rand(n, 2) * 0.7 + 0.15)
    np.random.seed(456)
    inits.append(np.random.rand(n, 2) * 0.7 + 0.15)

    bounds = [(0, 1)] * (2*n) + [(0, 0.5)] * n

    for init_centers in inits:
        r0 = np.full(n, 0.04)
        Z = np.concatenate([init_centers.flatten(), r0])
        
        alpha = 500.0
        beta = 500.0
        # Iteratively increase penalty weights to enforce constraints strictly
        for _ in range(4):
            res = minimize(compute_cost, Z, args=(n, alpha, beta), method='L-BFGS-B', 
                           bounds=bounds, options={'maxiter': 1000, 'ftol': 1e-12})
            Z = res.x
            alpha *= 3.0
            beta *= 3.0
            
        centers = Z[:2*n].reshape((n, 2))
        radii = Z[2*n:]
        s_r = np.sum(radii)
        
        if s_r > best_sum:
            best_sum = s_r
            best_Z = Z.copy()

    centers = best_Z[:2*n].reshape((n, 2))
    radii = best_Z[2*n:]
    
    # Compute exact max violations for final safety scaling
    c_diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(c_diff**2, axis=2))
    r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    max_ov = np.max(np.triu(r_sum - dists, k=1))
    
    v1 = radii - centers[:, 0]
    v2 = radii + centers[:, 0] - 1.0
    v3 = radii - centers[:, 1]
    v4 = radii + centers[:, 1] - 1.0
    max_bv = np.max(np.maximum(0, np.maximum(v1, v2)))
    max_bv = np.max(np.maximum(max_bv, np.maximum(v3, v4)))
    
    violation = max(0, max_ov, max_bv)
    if violation > 0:
        scale = 1.0 - violation - 1e-6
        radii = radii * scale
        best_sum = np.sum(radii)
        
    return centers, radii, best_sum
