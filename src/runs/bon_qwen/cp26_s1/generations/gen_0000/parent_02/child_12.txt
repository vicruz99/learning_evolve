import numpy as np
from scipy.optimize import minimize

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    n = centers.shape[0]
    if np.isnan(centers).any() or np.isnan(radii).any():
        return False
    for i in range(n):
        if radii[i] < 0:
            return False
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            return False
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False
    return True

def get_constraints(centers, radii):
    """
    Returns a list of constraint dictionaries for scipy.optimize
    """
    n = len(radii)
    constraints = []
    
    # Boundary constraints
    for i in range(n):
        # x - r >= 0
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: v[i*3] - v[i*3 + 2]})
        # 1 - x - r >= 0
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: 1.0 - v[i*3] - v[i*3 + 2]})
        # y - r >= 0
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: v[i*3 + 1] - v[i*3 + 2]})
        # 1 - y - r >= 0
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: 1.0 - v[i*3 + 1] - v[i*3 + 2]})
        
    # Overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            # dist^2 - (r_i + r_j)^2 >= 0
            constraints.append({
                'type': 'ineq',
                'fun': lambda v, i=i, j=j: (v[i*3] - v[j*3])**2 + (v[i*3 + 1] - v[j*3 + 1])**2 - (v[i*3 + 2] + v[j*3 + 2])**2
            })
    return constraints

def run_packing():
    np.random.seed(42)
    n = 26
    best_centers = None
    best_radii = None
    best_sum = -np.inf
    constraints = get_constraints(np.zeros(n), np.zeros(n))
    
    # Define objective function
    def objective(v):
        radii = v[2::3]
        return -np.sum(radii)

    initial_guesses = []

    centers_grid = np.zeros((n, 2))

    for r in range(5):

            if idx < n:
                centers_grid[idx] = [0.1 + c * 0.2, 0.1 + r * 0.2]
                idx += 1
    if idx < n:
        centers_grid[idx] = [0.5, 0.5] # Center
    initial_guesses.append(np.hstack([centers_grid.T, radii_grid]))

    # 2. Hexagonal packing
    centers_hex = np.zeros((n, 2))
    radii_hex = np.full(n, 0.09)
    idx = 0
    row = 0
    while idx < n:
        for c in range(5):
            if idx >= n: break
            x = 0.1 + c * 0.2
            y = 0.1 + row * (0.2 * np.sqrt(3) / 2)
            if row % 2 == 1:
                x += 0.1
            centers_hex[idx] = [x, y]
            idx += 1
        row += 1
    initial_guesses.append(np.hstack([centers_hex.T, radii_hex]))

    # 3. Random packing
    for _ in range(5):
        centers_rand = np.random.rand(n, 2) * 0.8 + 0.1
        radii_rand = np.full(n, 0.05)
        initial_guesses.append(np.hstack([centers_rand.T, radii_rand]))
        
    # 4. Corner focused
    centers_corner = np.zeros((n, 2))
    radii_corner = np.full(n, 0.08)

    centers_corner[1] = [0.9, 0.1]
    centers_corner[2] = [0.1, 0.9]
