from scipy.optimize import minimize
import math


def objective(x):

    radii = x[2::3]
    return -np.sum(radii)


def all_constraints(x):

    c = []

    for i in range(n):

        c.append(1.0 - x[3*i+1] - x[3*i+2])

            dy = x[3*i+1] - x[3*j+1]

            c.append(dist - x[3*i+2] - x[3*j+2])
    return np.array(c)

    n = 26

        else:
            n_in_row = 1
