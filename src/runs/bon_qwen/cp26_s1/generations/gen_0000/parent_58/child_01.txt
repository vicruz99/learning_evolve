def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    points = np.array(points)

    # Normalize to fit in [0,1] with margin
    # Scale so that bounding box is roughly [0.1, 0.9]
    min_xy = np.min(centers, axis=0)
    max_xy = np.max(centers, axis=0)
    center_of_points = (min_xy + max_xy) / 2
    # Shift to 0.5
    centers = centers - center_of_points + center
    
    # Scale down to ensure initial radius 0.05 fits

    max_dist = np.max(np.linalg.norm(centers - center, axis=1))
    # We want max_dist + r <= 1. With r=0.05, max_dist should be <= 0.95
    if max_dist > 0.8: # safety margin
        scale = 0.8 / max_dist
        centers = (centers - center) * scale + center
        
    r_init = 0.08 # Start a bit larger to push optimizer
    
    # 2. Optimization setup
    # Variables: x_0, y_0, ..., x_n-1, y_n-1, r
    # We will use a simpler approach: iterative expansion with repulsion
    # This is often more robust for packing than one-shot NLP with many constraints.
    
    # Current state
    curr_centers = centers.copy()
    curr_r = r_init
    
    # Repulsion parameters
    k_repulse = 10.0 # Force constant
    dt = 0.001 # Time step for simulation
    
    # We try to increase r slowly
    for step in range(500):
        target_r = curr_r * 1.001 # Increase by 0.1%
        
        # Check if we can expand. If overlap, push apart.
        # Compute forces
        forces = np.zeros_like(curr_centers)
        penalty = 0.0
        
        # Circle-circle repulsion
        for i in range(n):
            for j in range(i + 1, n):
                dist_vec = curr_centers[i] - curr_centers[j]
                dist = np.linalg.norm(dist_vec)
                min_dist = 2 * target_r
                if dist < min_dist:
                    overlap = min_dist - dist
                    # Force proportional to overlap
                    # Normalize vector
                    if dist > 1e-9:
                        dir_vec = dist_vec / dist
                        force_mag = k_repulse * overlap
                        forces[i] += dir_vec * force_mag
                        forces[j] -= dir_vec * force_mag
                    penalty += overlap**2
        
        # Wall repulsion
        for i in range(n):
            x, y = curr_centers[i]
            r = target_r
            # Left wall
            if x < r:
                forces[i, 0] += k_repulse * (r - x)
                penalty += (r - x)**2
            # Right wall
            if x > 1 - r:
                forces[i, 0] -= k_repulse * (x - (1 - r))
                penalty += (x - (1 - r))**2
            # Bottom wall
            if y < r:
                forces[i, 1] += k_repulse * (r - y)
                penalty += (r - y)**2
            # Top wall
            if y > 1 - r:
                forces[i, 1] -= k_repulse * (y - (1 - r))
                penalty += (y - (1 - r))**2
        
        # Update positions

        curr_centers += forces * dt
        
        # Project back to square if needed (hard constraint)

    # And maybe slightly reduce r if numerical errors occurred.

    # min_dist / 2 and distance to walls.
    min_wall_dist = min(np.min(curr_centers[:, 0]), np.min(curr_centers[:, 1]), 
                        np.min(1 - curr_centers[:, 0]), np.min(1 - curr_centers[:, 1]))

    final_radii = np.full(n, final_r)
    
    # Sum of radii
    sum_radii = np.sum(final_radii)
    
    return curr_centers, final_radii, sum_radii
