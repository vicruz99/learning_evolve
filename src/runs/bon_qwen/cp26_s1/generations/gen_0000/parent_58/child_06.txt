import numpy as np
import math

def run_packing():
    """
    Packs 26 circles in a unit square [0,1]x[0,1] to maximize sum of radii.
    Uses a force-directed simulation with radius expansion.
    """
    n = 26

    np.random.seed(123)

    # 1. Initialization
    # Random positions inside the square
    centers = np.random.rand(n, 2)
    # Small initial radius to prevent immediate overlaps
    radii = np.full(n, 0.02)

    # Simulation parameters
    iterations = 50000
    growth_rate = 2e-6  # How much to increase radii per step
    repulsion_strength = 1.0  # Force multiplier for overlaps
    boundary_strength = 5.0   # Force multiplier for walls
    damping = 0.1             # Learning rate for position updates
    noise_scale = 0.001       # Initial random jitter scale
    
    # Helper to compute pairwise distances and forces
    # We use loops since n=26 is small enough for Python loops to be fast enough
    
    for step in range(iterations):
        # Decay noise to settle down
        current_noise = noise_scale * max(0.0, 1.0 - step / (iterations * 0.8))
        
        # 1. Expand Radii
        # Increasing radii forces the system to find new equilibrium
        radii += growth_rate
        
        # 2. Compute Forces
        forces = np.zeros_like(centers)
        
        # Pairwise Repulsion
        # Check all pairs (i, j) with i < j
        # To speed up, we can vectorize, but for n=26 loops are fine
        for i in range(n):
            for j in range(i + 1, n):
                # Vector from j to i
                diff = centers[i] - centers[j]
                dist_sq = np.dot(diff, diff)
                
                # Avoid sqrt if possible, but we need dist for normalization
                # If dist is very small, handle carefully
                if dist_sq < 1e-12:
                    dist = 1e-6
                    # If centers coincide, push in random direction
                    diff = np.random.rand(2) - 0.5
                    diff /= np.linalg.norm(diff)
                else:
                    dist = math.sqrt(dist_sq)
                
                required_dist = radii[i] + radii[j]

                    overlap = required_dist - dist
                    # Direction vector (unit vector from j to i)
                    direction = diff / dist
                    
                    # Force magnitude proportional to overlap
                    # We split the correction between the two circles
                    # Push i away from j, push j away from i
                    force_vec = direction * overlap * repulsion_strength
                    
                    forces[i] += force_vec
                    forces[j] -= force_vec

        for i in range(n):

            x, y = centers[i]
            
            # Left Wall
            if x < r:
                forces[i, 0] += (r - x) * boundary_strength
            # Right Wall
            elif x > 1.0 - r:
                forces[i, 0] -= (x - (1.0 - r)) * boundary_strength
            
            # Bottom Wall
            if y < r:
                forces[i, 1] += (r - y) * boundary_strength
            # Top Wall
            elif y > 1.0 - r:
                forces[i, 1] -= (y - (1.0 - r)) * boundary_strength

        # 3. Update Positions

        noise = np.random.randn(n, 2) * current_noise
        centers += (forces * damping) + noise

        # This is important because forces might overshoot
        for i in range(n):
            r = radii[i]
            centers[i, 0] = np.clip(centers[i, 0], r, 1.0 - r)
            centers[i, 1] = np.clip(centers[i, 1], r, 1.0 - r)

    # Final Cleanup
