

<strategy>
My approach is to maximize the sum of radii for 26 circles in a unit square using a hybrid optimization strategy:

1. **LP for radii given fixed centers**: For any fixed configuration of centers, the problem of finding radii that maximize the sum becomes a linear program. The constraints are:
   - r_i ≤ x_i, 1-x_i, y_i, 1-y_i (boundary constraints)
   - r_i + r_j ≤ distance(c_i, c_j) for all pairs (non-overlap)
   - r_i ≥ 0

2. **Derivative-free optimization for centers**: Since the radii as a function of centers is piecewise linear (from the LP), I use Nelder-Mead (derivative-free) to optimize center positions.

3. **Multiple restarts**: I try different initial configurations (hexagonal grids, random perturbations) to avoid local minima.

4. **Additional refinement**: After getting a good solution from LP+optimization, I also try a direct SLSQP approach that jointly optimizes centers and radii.

5. **Return the best valid solution** from all attempts.

Key insights:
- The LP formulation efficiently computes optimal radii for any center configuration
- Nelder-Mead handles the non-convexity of center optimization well
- Multiple restarts help find better global solutions
- The constraint-based SLSQP approach can refine solutions further

import numpy as np
from scipy.optimize import linprog, minimize


def solve_radii_lp(centers, n):
    """Given centers, solve LP to find optimal radii maximizing sum."""
    c_obj = -np.ones(n)

    n_constraints = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((n_constraints, n))
    b_ub = np.zeros(n_constraints)

    idx = 0
    for i in range(n):
        # r_i <= x_i
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 0]
        idx += 1
        # r_i <= 1 - x_i
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 0]
        idx += 1
        # r_i <= y_i
        A_ub[idx, i] = 1
        b_ub[idx] = centers[i, 1]
        idx += 1
        # r_i <= 1 - y_i
        A_ub[idx, i] = 1
        b_ub[idx] = 1 - centers[i, 1]
        idx += 1

    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = dist
            idx += 1

    bounds = [(0, None)] * n
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')

    if result.success:
        return result.x
    else:
        return np.zeros(n)


def evaluate_sum_radii(centers, n):
    """Compute sum of radii for given centers using LP."""
    radii = solve_radii_lp(centers, n)
    return np.sum(radii)


def get_hexagonal_init(n, row_shift, row_scale, col_scale):
    """Generate hexagonal-ish initial placement."""
    centers = np.zeros((n, 2))
    idx = 0
    for row in range(8):
        for col in range(6):
            if idx < n:
                x = (col + 0.5) * col_scale + (row % 2) * row_shift
                y = row * row_scale + row_scale * 0.4
                if x < 0.99 and y < 0.99 and x > 0.01 and y > 0.01:
                    centers[idx] = [x, y]
                    idx += 1
    while idx < n:
        centers[idx] = [np.random.rand() * 0.8 + 0.1, np.random.rand() * 0.8 + 0.1]
        idx += 1
    return centers


def run_packing():
    n = 26
    best_sum = 0
    best_centers = None
    best_radii = None

    # Strategy 1: LP-based optimization with Nelder-Mead, multiple restarts
    for seed in range(30):
        np.random.seed(seed)

        # Generate initial centers with various patterns
        if seed < 5:
            centers = get_hexagonal_init(n, 0.2, 0.18, 0.18)
        elif seed < 10:
            centers = get_hexagonal_init(n, 0.15, 0.15, 0.2)
        elif seed < 15:
            centers = get_hexagonal_init(n, 0.25, 0.2, 0.16)
        elif seed < 20:
            centers = get_hexagonal_init(n, 0.1, 0.16, 0.22)
        else:
            # Random with grid structure
            idx = 0
            for i in range(6):
                for j in range(5):
                    if idx < n:
                        centers[idx] = [(i + 0.5 + np.random.normal(0, 0.05)) / 6,
                                       (j + 0.5 + np.random.normal(0, 0.05)) / 5]
                        idx += 1
            while idx < n:
                centers[idx] = np.random.rand(2) * 0.8 + 0.1
                idx += 1

        centers = np.clip(centers, 0.02, 0.98)

        def objective(centers_flat):
            c = centers_flat.reshape((n, 2))
            return -evaluate_sum_radii(c, n)

        # Nelder-Mead optimization
        result = minimize(
            objective,
            centers.flatten(),
            method='Nelder-Mead',
            options={'maxiter': 15000, 'xatol': 1e-9, 'fatol': 1e-12}
        )

        opt_centers = result.x.reshape((n, 2))
        opt_centers = np.clip(opt_centers, 0.01, 0.99)

        # Solve for radii
        radii = solve_radii_lp(opt_centers, n)
        total = np.sum(radii)

        if total > best_sum:
            best_sum = total
            best_centers = opt_centers.copy()
            best_radii = radii.copy()

    # Strategy 2: SLSQP joint optimization from best LP solution
    if best_centers is not None:
        x0 = np.concatenate([best_centers.flatten(), best_radii])

        def obj_slsqp(x):
            return -np.sum(x[2 * n:])

        def constr_boundary(x):
            c = x[:2 * n].reshape((n, 2))
            r = x[2 * n:]
            cons = []
            for i in range(n):
                cons.append(c[i, 0] - r[i])
                cons.append(1 - c[i, 0] - r[i])
                cons.append(c[i, 1] - r[i])
                cons.append(1 - c[i, 1] - r[i])
            return np.array(cons)

        def constr_overlap(x):
            c = x[:2 * n].reshape((n, 2))
            r = x[2 * n:]
            cons = []
            for i in range(n):
                for j in range(i + 1, n):
                    dx = c[i, 0] - c[j, 0]
                    dy = c[i, 1] - c[j, 1]
                    dist = np.sqrt(dx * dx + dy * dy)
                    cons.append(dist - r[i] - r[j] + 1e-10)
            return np.array(cons)

        constraints = [
            {'type': 'ineq', 'fun': constr_boundary},
            {'type': 'ineq', 'fun': constr_overlap}
        ]
        bounds_slsqp = [(0, 1)] * (2 * n) + [(0, 0.5)] * n

        result = minimize(
            obj_slsqp, x0, method='SLSQP',
            bounds=bounds_slsqp, constraints=constraints,
            options={'maxiter': 5000, 'ftol': 1e-15}
        )

        if result.fun < 0:
            opt_x = result.x
            opt_centers = opt_x[:2 * n].reshape((n, 2))
            opt_radii = opt_x[2 * n:]
            opt_total = np.sum(opt_radii)

            if opt_total > best_sum:
                best_sum = opt_total
                best_centers = opt_centers.copy()
                best_radii = opt_radii.copy()

    # Strategy 3: Simulated annealing refinement
    if best_centers is not None:
        centers = best_centers.copy()
        current_sum = best_sum

        for temp_idx in range(500):
            temp = 0.05 * (1 - temp_idx / 500) + 0.001
            for _ in range(100):
                new_centers = centers.copy()
                new_centers += np.random.normal(0, temp, size=new_centers.shape)
                new_centers = np.clip(new_centers, 0.01, 0.99)

                new_radii = solve_radii_lp(new_centers, n)
                new_sum = np.sum(new_radii)

                if new_sum > current_sum or np.random.rand() < np.exp((new_sum - current_sum) / (temp + 1e-10)):
                    centers = new_centers.copy()
                    current_sum = new_sum

                if current_sum > best_sum:
                    best_sum = current_sum
                    best_centers = centers.copy()
                    best_radii = new_radii.copy()

    # Final validation and cleanup
    if best_centers is not None:
        # Re-solve radii LP for final centers to ensure optimality
        final_radii = solve_radii_lp(best_centers, n)
        final_sum = np.sum(final_radii)

        # Clip radii to avoid negative values
        final_radii = np.maximum(final_radii, 0)

        # Ensure centers are within bounds given radii
        for i in range(n):
            r = final_radii[i]
            best_centers[i, 0] = np.clip(best_centers[i, 0], r + 1e-12, 1 - r - 1e-12)
            best_centers[i, 1] = np.clip(best_centers[i, 1], r + 1e-12, 1 - r - 1e-12)

        return best_centers, final_radii, final_sum

    # Fallback: uniform grid
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.05
    idx = 0
    for i in range(6):
        for j in range(5):
            if idx < n:
                centers[idx] = [(i + 0.5) / 6, (j + 0.5) / 5]
                idx += 1
    while idx < n:
        centers[idx] = [np.random.rand(), np.random.rand()]
        idx += 1
    return centers, radii, np.sum(radii)
