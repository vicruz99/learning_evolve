def objective(vars, n):
    # vars: [x1, y1, r1, x2, y2, r2, ...]
    radii = vars[2::3]
    return -np.sum(radii) # Minimize negative sum

def constraints_builder(n):
    constraints = []
    
    # 1. Boundary constraints: r <= x <= 1-r  =>  x - r >= 0, x + r <= 1
    # 2. r <= y <= 1-r  =>  y - r >= 0, y + r <= 1
    # 3. r >= 0 (handled by bounds)

            return func

            def func(vars): return 1 - vars[3*i] - vars[3*i + 2]
            return func
        constraints.append({'type': 'ineq', 'fun': make_x_upper(i)})
        
        # y - r >= 0

            def func(vars): return vars[3*i + 1] - vars[3*i + 2]
            return func
        constraints.append({'type': 'ineq', 'fun': make_y_lower(i)})

            def func(vars): return 1 - vars[3*i + 1] - vars[3*i + 2]
            return func

    # Non-overlap constraints
    # Number of pairs is n*(n-1)/2. For n=26, ~325 constraints.
    # This might be heavy for some solvers, but SLSQP should handle it.
    # We can define a function that returns a vector of constraint violations?
    # SLSQP accepts vector functions.

    def non_overlap(vars):
        xs = vars[0::3]
        ys = vars[1::3]

                res.append(dist_sq - r_sum*r_sum)
        return np.array(res)

    constraints.append({'type': 'ineq', 'fun': non_overlap})
