import numpy as np
import scipy.optimize as opt

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Pack 26 circles in a unit square to maximize the sum of radii.
    """
    n = 26
    
    # 1. Initialize a hexagonal-like lattice packing
    # This provides a good starting point for the optimizer
    rows = 6
    cols = 5
    centers = []
    radii = []
    
    r_init = 0.1 # Initial guess radius
    
    for i in range(rows):
        # Shift odd rows to create a hexagonal pattern
        if i % 2 == 1:
            x_offset = r_init * 0.5
            row_width = 1 - 2 * r_init - r_init * 0.5
        else:
            x_offset = r_init
            row_width = 1 - 2 * r_init
            
        # Calculate how many circles fit in this row with current spacing
        # We aim to distribute 26 circles roughly evenly
        # For 6 rows, we can have 5, 4, 5, 4, 5, 3 -> sum 26
        if i == 0: count = 5
        elif i == 1: count = 4
        elif i == 2: count = 5
        elif i == 3: count = 4
        elif i == 4: count = 5
        else: count = 3
        
        y_pos = r_init + i * (r_init * np.sqrt(3))
        
        # Space circles evenly within the row width
        if count > 1:
            x_spacing = row_width / (count - 1)
            for j in range(count):
                x_pos = x_offset + j * x_spacing
                centers.append([x_pos, y_pos])
                radii.append(r_init)
        else:
            centers.append([0.5, y_pos]) # Center the single circle
            radii.append(r_init)

    centers = np.array(centers[:n])
    radii = np.array(radii[:n])
    
    # Ensure shapes
    if len(centers) < n:
        # Fallback if initialization failed to generate enough
        # Fill remaining with small circles in gaps or corners
        for k in range(n - len(centers)):
            centers = np.vstack([centers, [0.5, 0.5]])
            radii = np.append(radii, 0.01)
    else:
        centers = centers[:n]
        radii = radii[:n]

    # 2. Optimization Setup
    # We optimize a combined vector of [x1, y1, r1, x2, y2, r2, ...]
    # To handle constraints effectively, we use a penalty method or constrained optimization.
    # scipy.optimize.minimize with SLSQP is suitable for bound constraints.
    
    # Flattened variable vector
    x0 = np.hstack([centers, radii])
    
    # Bounds: x, y in [0, 1], r in [0, 0.5]
    # Actually r <= 0.5 is loose, but safe. 
    # Better bounds: r <= min(x, 1-x, y, 1-y) is handled by constraints.
    # We can bound x,y in [0,1] and r in [1e-4, 0.5]
    bounds = [(0, 1) for _ in range(2 * n)] + [(1e-4, 0.5) for _ in range(n)]
    
    def objective(vars_vec):
        # We want to maximize sum of radii -> minimize negative sum
        r = vars_vec[2*n:]
        return -np.sum(r)
    
    def constraint_overlap(vars_vec):
        # dist_ij >= r_i + r_j  => dist_ij - r_i - r_j >= 0
        c = vars_vec[0:2*n]
        r = vars_vec[2*n:]
        cx, cy = c[0::2], c[1::2]
        
        # Vectorized distance calculation
        # Compute pairwise distances
        diff_x = cx[:, None] - cx[None, :]
        diff_y = cy[:, None] - cy[None, :]
        dists = np.sqrt(diff_x**2 + diff_y**2)
        
        # Compute sum of radii matrix
        r_sum = r[:, None] + r[None, :]
        
        # Constraint matrix (lower triangle to avoid duplicates and self)
        mask = np.tril(np.ones((n, n), dtype=bool), k=-1)
        vals = dists[mask] - r_sum[mask]
        return vals

    def constraint_boundary(vars_vec):
        # x - r >= 0, 1 - x - r >= 0, etc.
        c = vars_vec[0:2*n]
        r = vars_vec[2*n:]
        cx, cy = c[0::2], c[1::2]
        
        # 4 constraints per circle
        # x >= r  => x - r >= 0
        # 1 - x >= r => 1 - x - r >= 0
        # y >= r  => y - r >= 0
        # 1 - y >= r => 1 - y - r >= 0
        
        con1 = cx - r
        con2 = 1 - cx - r
        con3 = cy - r
        con4 = 1 - cy - r
        
        return np.concatenate([con1, con2, con3, con4])

    constraints = [
        {'type': 'ineq', 'fun': constraint_overlap},
        {'type': 'ineq', 'fun': constraint_boundary}
    ]

    # Run optimization
    # Using SLSQP which handles bounds and constraints
    result = opt.minimize(
        objective, 
        x0, 
        method='SLSQP', 
        bounds=bounds, 
        constraints=constraints,
        options={'maxiter': 1000, 'ftol': 1e-9}
    )
    
    # Extract solution
    final_centers = result.x[0:2*n].reshape((n, 2))
    final_radii = result.x[2*n:]
    
    # Validation and cleanup
    # Ensure radii are non-negative
    final_radii = np.maximum(final_radii, 1e-9)
    
    # Final sum of radii
    sum_radii = np.sum(final_radii)
    
    return final_centers, final_radii, sum_radii
