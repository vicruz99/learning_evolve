import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def penalty_objective(vars, lam):
    """
    Objective function: minimize -sum(radii) + lambda * sum(violations^2)
    """
    n = N_CIRCLES
    centers = vars[:n * 2].reshape(n, 2)
    radii = vars[n * 2:]
    
    obj = -np.sum(radii)
    pen = 0.0
    
    # Boundary constraints: circle must be inside [0,1]x[0,1]
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        
        # Left & Right
        v_left = x - r
        if v_left < 0.0:
            pen += v_left ** 2
        v_right = 1.0 - x - r
        if v_right < 0.0:
            pen += v_right ** 2
            
        # Bottom & Top
        v_bottom = y - r
        if v_bottom < 0.0:
            pen += v_bottom ** 2
        v_top = 1.0 - y - r
        if v_top < 0.0:
            pen += v_top ** 2
            
    # Inter-circle constraints: distance >= r_i + r_j
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            v_dist = dist - radii[i] - radii[j]
            if v_dist < 0.0:
                pen += v_dist ** 2
                
    return obj + lam * pen

def generate_initial_config():
    """
    Generate a hexagonal-ish initial arrangement of centers.
    """
    pts = []
    # 5 rows with alternating column counts to approximate hex packing
    y_vals = np.linspace(0.12, 0.88, 5)
    for idx, y in enumerate(y_vals):
        n_cols = 6 if idx % 2 == 0 else 5
        x_vals = np.linspace(0.1, 0.9, n_cols)
        for x in x_vals:
            pts.append([x, y])
        if len(pts) >= N_CIRCLES:
            break
    return np.array(pts[:N_CIRCLES])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    
    best_sum = -np.inf
    best_centers = None
    best_radii = None
    
    # Bounds: x,y in [0,1], r in [0, 0.5]
    bounds = [(0.0, 1.0)] * (N_CIRCLES * 2) + [(0.0, 0.5)] * N_CIRCLES
    
    # Base configuration
    c_base = generate_initial_config()
    
    # Run multiple trials with perturbations
    for trial in range(6):
        centers = c_base.copy()
        if trial > 0:
            # Add noise to escape local minima
            centers += np.random.randn(N_CIRCLES, 2) * 0.04
            centers = np.clip(centers, 0.02, 0.98)
            
        radii = np.full(N_CIRCLES, 0.03)
        x0 = np.concatenate([centers.ravel(), radii])
        
        # Progressive penalty annealing
        lam = 500.0
        x_curr = x0.copy()
        
        for step in range(25):
            res = minimize(
                penalty_objective, 
                x_curr, 
                args=(lam,), 
                method='L-BFGS-B', 
                bounds=bounds, 
                options={'maxiter': 800, 'ftol': 1e-12, 'gtol': 1e-8}
            )
            x_curr = res.x
            lam *= 1.4  # Increase penalty weight gradually
            
        # Extract optimized parameters
        c_opt = x_curr[:N_CIRCLES * 2].reshape(N_CIRCLES, 2)
        r_opt = x_curr[N_CIRCLES * 2:]
        
        # Ensure physical validity
        r_opt = np.maximum(r_opt, 1e-7)
        c_opt = np.clip(c_opt, r_opt[:, None], 1.0 - r_opt[:, None])
        
        current_sum = np.sum(r_opt)
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = c_opt
            best_radii = r_opt
            
    return best_centers, best_radii, best_sum
