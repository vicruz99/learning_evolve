import numpy as np
from scipy.optimize import minimize

def compute_penalty(centers, radii, n):
    penalty = 0.0
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x < r:
            penalty += 10000.0 * (x - r) ** 2
        if x > 1 - r:
            penalty += 10000.0 * (x - (1 - r)) ** 2
        if y < r:
            penalty += 10000.0 * (y - r) ** 2
        if y > 1 - r:
            penalty += 10000.0 * (y - (1 - r)) ** 2
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i][0] - centers[j][0]
            dy = centers[i][1] - centers[j][1]
            dist = np.sqrt(dx * dx + dy * dy)
            overlap = radii[i] + radii[j] - dist
            if overlap > 0:
                penalty += 10000.0 * overlap ** 2
    return penalty

def objective(params, n):
    centers = params[:2 * n].reshape(n, 2)
    radii = params[2 * n:]
    return -np.sum(radii) + compute_penalty(centers, radii, n)

def force_optimize(centers, radii, n, steps=20000):
    for step in range(steps):
        force = np.zeros_like(centers)
        r_force = np.zeros(n)
        cooling = max(0.00001, 1.0 - step / steps)
        
        for i in range(n):
            # Expansion tendency
            r_force[i] = cooling * 0.008
            
            # Boundary repulsion
            if centers[i][0] < radii[i]:
                force[i][0] += cooling * 0.08
            if centers[i][0] > 1 - radii[i]:
                force[i][0] -= cooling * 0.08
            if centers[i][1] < radii[i]:
                force[i][1] += cooling * 0.08
            if centers[i][1] > 1 - radii[i]:
                force[i][1] -= cooling * 0.08
            
            # Circle-circle interactions
            for j in range(i + 1, n):
                dx = centers[j][0] - centers[i][0]
                dy = centers[j][1] - centers[i][1]
                dist_sq = dx * dx + dy * dy
                dist = np.sqrt(dist_sq) + 1e-12
                required = radii[i] + radii[j]
                gap = dist - required
                
                if gap < 0:
                    f = -gap * cooling * 0.15 / dist
                    force[i][0] -= dx * f
                    force[i][1] -= dy * f
                    force[j][0] += dx * f
                    force[j][1] += dy * f
                    r_force[i] -= cooling * 0.004
                    r_force[j] -= cooling * 0.004
        
        centers += force * cooling * 0.5
        radii += r_force * cooling
        radii = np.maximum(radii, 1e-8)
        centers = np.clip(centers, 1e-8, 1 - 1e-8)
    
    return centers, radii

def initialize_hexagonal(n, row_counts, r_init):
    centers = np.zeros((n, 2))
    radii = np.ones(n) * r_init
    
    idx = 0
    for row_idx, count in enumerate(row_counts):
        offset = r_init if count % 2 == 0 else 0
        y = r_init + row_idx * np.sqrt(3) * r_init
        for col in range(count):
            x = r_init + offset + col * 2 * r_init
            if idx < n:
                centers[idx] = [x, y]
                idx += 1
    
    return centers, radii

def gradient_optimize(centers, radii, n):
    params = np.concatenate([centers.flatten(), radii])
    bounds = [(0, 1)] * (2 * n) + [(1e-10, 0.5)] * n
    
    for _ in range(3):
        result = minimize(objective, params, method='L-BFGS-B', bounds=bounds,
                         options={'maxiter': 10000, 'ftol': 1e-15, 'gtol': 1e-10})
        params = result.x
    
    centers = params[:2 * n].reshape(n, 2)
    radii = params[2 * n:]
    return centers, radii

def enforce_validity(centers, radii, n):
    # Enforce boundary constraints strictly
    for i in range(n):
        max_r = min(centers[i][0], 1 - centers[i][0], centers[i][1], 1 - centers[i][1])
        if radii[i] > max_r:
            radii[i] = max(max_r, 1e-10)
    
    # Resolve overlaps by shrinking the smaller circle
    changed = True
    iterations = 0
    while changed and iterations < 1000:
        changed = False
        iterations += 1

            for j in range(i + 1, n):
                dx = centers[i][0] - centers[j][0]

                    # Shrink the smaller circle
                    if radii[i] <= radii[j]:
                        radii[i] -= overlap / 2
                    else:
                        radii[j] -= overlap / 2
                    radii[i] = max(radii[i], 1e-10)
                    radii[j] = max(radii[j], 1e-10)

        max_r = min(centers[i][0], 1 - centers[i][0], centers[i][1], 1 - centers[i][1])

            radii[i] = max(max_r, 1e-10)
    
    return centers, radii
