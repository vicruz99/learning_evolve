import numpy as np
import math

def resolve_collisions(centers, radii, n):
    """
    Iteratively resolves collisions between circles and with boundaries.
    """
    # Convert to numpy if needed
    centers = np.array(centers)
    radii = np.array(radii)
    
    for _ in range(100): # Resolve iterations
        moved = False
        for i in range(n):
            # Boundary checks
            for dim in range(2):
                if centers[i, dim] - radii[i] < 0:
                    centers[i, dim] = radii[i]
                    moved = True
                elif centers[i, dim] + radii[i] > 1:
                    centers[i, dim] = 1 - radii[i]
                    moved = True
            
            # Pairwise checks
            for j in range(i + 1, n):
                diff = centers[i] - centers[j]
                dist = np.sqrt(np.sum(diff**2))
                min_dist = radii[i] + radii[j]
                if dist < min_dist and dist > 1e-9:
                    # Collision
                    overlap = min_dist - dist
                    direction = diff / dist
                    # Push apart
                    centers[i] += direction * (overlap * 0.5)
                    centers[j] -= direction * (overlap * 0.5)
                    moved = True
        if not moved:
            break
            
    return centers
