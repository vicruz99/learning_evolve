import numpy as np
import scipy.optimize as opt

def get_duals_and_radii(centers):
    """
    Solves the LP to find max sum of radii for fixed centers.
    Returns radii and the dual variables (marginals) for the distance constraints.
    """
    n = centers.shape[0]
    
    # Objective: maximize sum(r_i) -> minimize -sum(r_i)
    c_obj = -np.ones(n)
    
    # Constraints matrix A_ub * r <= b_ub
    # 1. Pairwise distance constraints: r_i + r_j <= dist_ij
    # 2. Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    
    # We will construct the constraints manually to keep track of duals
    # Order of constraints:
    # 0 to n*(n-1)/2 - 1 : Pairwise constraints
    # n*(n-1)/2 to ... : Boundary constraints
    
    # Preallocate
    num_pairs = n * (n - 1) // 2
    num_boundary = 4 * n
    total_constraints = num_pairs + num_boundary
    
    # Using sparse matrix for efficiency
    rows = []
    cols = []
    data = []
    b_ub = []
    
    # Pairwise constraints
    # Map (i, j) to constraint index
    pair_idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            rows.extend([i, j])
            cols.extend([pair_idx, pair_idx]) # This is for A_ub, wait.
            # A_ub is (m, n). We need rows for each constraint.
            # Let's build A_ub directly.
            pass
            
    # Let's build A_ub explicitly
    # A_ub shape: (total_constraints, n)
    A_ub = np.zeros((total_constraints, n))
    
    # 1. Pairwise
    pair_idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            A_ub[pair_idx, i] = 1.0
            A_ub[pair_idx, j] = 1.0
            b_ub_val = dist
            if np.isfinite(b_ub_val):
                b_ub.append(b_ub_val)
            else:
                b_ub.append(1000.0) # Large value if overlap (dist 0)
            pair_idx += 1
            
    # 2. Boundary constraints
    # r_i <= x_i
    # r_i <= 1 - x_i
    # r_i <= y_i
    # r_i <= 1 - y_i
    
    for i in range(n):
        x, y = centers[i]
        
        # r_i <= x_i
        A_ub[pair_idx + i, i] = 1.0
        b_ub.append(x)
        
        # r_i <= 1 - x_i  => r_i + x_i <= 1? No, r_i is variable, x_i is param.
        # Constraint is r_i <= 1 - x_i.
        A_ub[pair_idx + n + i, i] = 1.0
        b_ub.append(1.0 - x)
        
        # r_i <= y_i
        A_ub[pair_idx + 2*n + i, i] = 1.0
        b_ub.append(y)
        
        # r_i <= 1 - y_i
        A_ub[pair_idx + 3*n + i, i] = 1.0
        b_ub.append(1.0 - y)
        
    b_ub = np.array(b_ub)
    
    # Bounds for r_i: 0 <= r_i
    bounds = [(0, None) for _ in range(n)]
    
    # Solve LP
    # method='highs' is usually robust
    try:
        res = opt.linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            radii = -res.fun # Since we minimized -sum
            # Actually res.x is the radii
            radii = res.x
            
            # Extract duals (marginals)
            # res.marginals contains duals for constraints (inequalities) and bounds
            # res.marginals.ineq corresponds to A_ub constraints
            # res.marginals.ub corresponds to upper bounds of variables
            # res.marginals.lb corresponds to lower bounds of variables
            
            # The duals for A_ub constraints are in res.marginals.ineq
            # Note: highs might return a dict or an array depending on version.
            # In recent scipy, it's an array or dict. Let's check.
            # Usually res.marginals is a Bunch.
            # marginals.ineq is the vector of duals for A_ub x <= b_ub.
            
            if hasattr(res, 'marginals'):
                # Access duals
                try:
                    duals_ineq = res.marginals.ineq
                except AttributeError:
                    # Fallback for older versions or different return types
                    # If marginals is not available, we might need to solve dual or estimate.
                    # But let's assume modern scipy.
                    duals_ineq = np.zeros(total_constraints)
            else:
                duals_ineq = np.zeros(total_constraints)
                
            return radii, duals_ineq
        else:
            # If LP fails, return small radii and zero duals
            return np.zeros(n), np.zeros(total_constraints)
    except Exception:
        return np.zeros(n), np.zeros(total_constraints)
