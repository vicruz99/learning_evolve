import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    N = 26
    np.random.seed(42)

    # 1. Hybrid Initialization: Grid-like distribution with random perturbation
    init_centers = []
    for i in range(N):
        # Place centers roughly on a grid, avoiding edges
        row = i // 6
        col = i % 6
        # Use slightly wider spacing than 1/7 to allow for expansion
        x = 0.1 + col * 0.15 + np.random.uniform(-0.02, 0.02)
        y = 0.1 + row * 0.15 + np.random.uniform(-0.02, 0.02)
        init_centers.append([x, y])
    init_centers = np.array(init_centers)
    
    # Initial radii: Small but non-zero to allow growth
    init_radii = np.ones(N) * 0.01

    # 2. Variable Transformation (Optimize squares of coords/radii)
    # vars: [x1^2, y1^2, r1^2, x2^2, ...]
    # Dimensions: 3 * N
    x0 = np.zeros(3 * N)
    for i in range(N):
        x0[3*i] = init_centers[i, 0] ** 2
        x0[3*i+1] = init_centers[i, 1] ** 2
        x0[3*i+2] = init_radii[i] ** 2

    # Bounds for squared variables
    bounds = [(0, 1.0)] * (2 * N) + [(0, 0.25)] * N

    def objective(vars):
        # Decode variables
        sq_vars = np.array(vars)
        cx = np.sqrt(sq_vars[0::3])
        cy = np.sqrt(sq_vars[1::3])
        cr = np.sqrt(sq_vars[2::3])
        
        # Objective: Maximize sum of radii -> Minimize negative sum
        obj_val = -np.sum(cr)
        
        penalty = 0.0
        
        # Boundary Constraints
        for i in range(N):
            x, y, r = cx[i], cy[i], cr[i]
            # x - r >= 0
            val = r - x
            if val > 0: penalty += val**2
            # 1 - x - r >= 0
            val = r + x - 1
            if val > 0: penalty += val**2
            # y - r >= 0
            val = r - y
            if val > 0: penalty += val**2
            # 1 - y - r >= 0
            val = r + y - 1
            if val > 0: penalty += val**2
            
        # Overlap Constraints (Squared Distance)
        # dist^2 >= (r_i + r_j)^2
        for i in range(N):
            for j in range(i + 1, N):
                dist_sq = (cx[i] - cx[j])**2 + (cy[i] - cy[j])**2
                sum_r = cr[i] + cr[j]
                viol = sum_r**2 - dist_sq
                if viol > 0:
                    penalty += viol**2
                    
        # Soft repulsion to distribute circles (prevent clustering)
        for i in range(N):
            for j in range(i + 1, N):
                d = np.sqrt((cx[i] - cx[j])**2 + (cy[i] - cy[j])**2)
                if d < 0.15:
                    penalty += (0.15 - d)**2

        return obj_val + 100.0 * penalty

    # 3. Sequential Optimization with increasing penalty weights
    current_x = x0
    for lam_mult in [10.0, 50.0, 200.0, 1000.0]:
        def obj_scaled(vars):
            sq_vars = np.array(vars)
            cx = np.sqrt(sq_vars[0::3])
            cy = np.sqrt(sq_vars[1::3])
            cr = np.sqrt(sq_vars[2::3])
            
            obj_val = -np.sum(cr)
            penalty = 0.0
            
            # Hard constraints via penalty
            for i in range(N):
                x, y, r = cx[i], cy[i], cr[i]
                if r > x: penalty += (r - x)**2
                if r + x > 1: penalty += (r + x - 1)**2
                if r > y: penalty += (r - y)**2
                if r + y > 1: penalty += (r + y - 1)**2
                
            for i in range(N):
                for j in range(i + 1, N):
                    dist_sq = (cx[i] - cx[j])**2 + (cy[i] - cy[j])**2
                    sum_r = cr[i] + cr[j]
                    viol = sum_r**2 - dist_sq
                    if viol > 0:
                        penalty += viol**2

            return obj_val + lam_mult * penalty

        res = minimize(obj_scaled, current_x, method='SLSQP', bounds=bounds, 
                       options={'maxiter': 1000, 'ftol': 1e-12})
        current_x = res.x

    # 4. Final Extraction and Verification
    final_sq = np.array(current_x)
    centers = np.zeros((N, 2))
    radii = np.zeros(N)
    
    for i in range(N):
        centers[i, 0] = np.sqrt(final_sq[3*i])
        centers[i, 1] = np.sqrt(final_sq[3*i+1])
        radii[i] = np.sqrt(final_sq[3*i+2])
        
    # Clamp to [0,1] and ensure non-negative
    centers = np.clip(centers, 1e-9, 1 - 1e-9)
    radii = np.abs(radii)
    
    # Re-validate and adjust if necessary
    valid = validate_packing(centers, radii)
    if not valid:
        # Fallback to simple grid if optimizer failed
        radii[:] = 0.09
        for i in range(N):
            row = i // 6
            col = i % 6
            centers[i] = [0.1 + col * 0.14, 0.1 + row * 0.14]
            if col == 0: centers[i, 0] = 0.095
            if row == 0: centers[i, 1] = 0.095
            
    return centers, radii, np.sum(radii)

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    n = centers.shape[0]
    if np.isnan(centers).any() or np.isnan(radii).any():
        return False
    for i in range(n):
        if radii[i] < 0: return False
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            return False
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False
    return True
