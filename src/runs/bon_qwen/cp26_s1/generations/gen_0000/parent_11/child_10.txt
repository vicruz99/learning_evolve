import numpy as np
from scipy.optimize import minimize

def get_hex_centers(n):
    """Generate initial centers using a hexagonal lattice pattern."""
    centers = []
    y = 0.15
    # Row counts that sum to 26 and approximate a hexagonal packing
    row_counts = [6, 5, 6, 5, 4]
    for cnt in row_counts:
        if len(centers) >= n:
            break
        # Space circles evenly along x for the current row
        xs = np.linspace(0.15, 0.85, cnt)
        for x in xs:
            centers.append([x, y])
        y += 0.18
    return np.array(centers[:n])

def resolve_overlaps_force(centers, radii, steps=30, k=0.5):
    """Resolve overlaps using a simple force-directed relaxation."""
    n = centers.shape[0]
    c = centers.copy()
    r = radii.copy()
    
    for _ in range(steps):
        forces = np.zeros_like(c)
        
        # Pairwise repulsion for overlaps
        for i in range(n):
            for j in range(i + 1, n):
                diff = c[i] - c[j]
                dist = np.sqrt(np.sum(diff ** 2))
                min_dist = r[i] + r[j]
                if dist < min_dist and dist > 1e-7:
                    rep = k * (min_dist - dist) / dist
                    forces[i] += diff * rep
                    forces[j] -= diff * rep
                    
        # Boundary repulsion
        for i in range(n):
            for dim in range(2):
                if c[i, dim] - r[i] < 0:
                    forces[i, dim] += k * (r[i] - c[i, dim])
                if c[i, dim] + r[i] > 1.0:
                    forces[i, dim] -= k * (c[i, dim] + r[i] - 1.0)
                    
        c += forces
        # Project back to valid range to prevent divergence
        c = np.clip(c, 1e-6, 1.0 - 1e-6)
        
    return c

def packing_objective(v):
    """Objective: maximize sum of radii (minimize negative sum)."""
    # v[0:52] are centers, v[52:78] are radii
    return -np.sum(v[52:])

def packing_constraints(v):
    """Inequality constraints: boundary and non-overlap."""
    n = 26
    c = v[:52].reshape(26, 2)
    r = v[52:]
    
    vals = []
    # Boundary constraints: x - r >= 0, 1 - x - r >= 0, same for y
    vals.append(c[:, 0] - r)
    vals.append(1.0 - c[:, 0] - r)
    vals.append(c[:, 1] - r)
    vals.append(1.0 - c[:, 1] - r)
    
    # Distance constraints: ||c_i - c_j||^2 >= (r_i + r_j)^2
    for i in range(n):
        diff = c[i] - c[i + 1:]
        d2 = np.sum(diff ** 2, axis=1)
        r2 = (r[i] + r[i + 1:]) ** 2
        vals.append(d2 - r2)
        
    return np.concatenate(vals)

def run_packing():
    n = 26
    centers = get_hex_centers(n)
    radii = np.full(n, 0.06)
    
    # Phase 1: Iterative expansion with force relaxation
    # Grow radii while pushing circles apart to maintain feasibility
    for _ in range(350):
        radii += 0.00012
        centers = resolve_overlaps_force(centers, radii, steps=20, k=0.6)
        
    # Phase 2: Precise optimization
    v0 = np.concatenate([centers.flatten(), radii])
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
    constraint = {'type': 'ineq', 'fun': packing_constraints}
    
    res = minimize(packing_objective, v0, method='SLSQP',
                   bounds=bounds, constraints=constraint,
                   options={'maxiter': 300, 'ftol': 1e-12, 'disp': False})
                   
    final_centers = res.x[:52].reshape(n, 2)
    final_radii = res.x[52:]
    
    # Ensure strict compliance with bounds (numerical safety)
    final_centers = np.clip(final_centers, final_radii[:, None], 1.0 - final_radii[:, None])
    
    return final_centers, final_radii, float(np.sum(final_radii))
