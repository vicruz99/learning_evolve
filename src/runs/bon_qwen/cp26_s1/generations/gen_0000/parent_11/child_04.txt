        y = points[:, 1]
        r = points[:, 2]

        # x - r >= 0
        constraints.append(x - r)
        # 1 - (x + r) >= 0  => x + r <= 1
        constraints.append(1.0 - (x + r))
        # y - r >= 0
        constraints.append(y - r)
        # 1 - (y + r) >= 0
        constraints.append(1.0 - (y + r))

        # We compute squared distance matrix and subtract squared sum of radii

        dy = y[:, None] - y[None, :]
