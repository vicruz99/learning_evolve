import numpy as np
from scipy.optimize import minimize

def get_jacobian_and_val(x_vars, n):
    """
    Computes the constraint values and the Jacobian matrix for all constraints.
    
    Args:
        x_vars: 1D numpy array of shape (3*n,). Structure: [x1, y1, r1, x2, y2, r2, ...]
        n: Number of circles.
        
    Returns:
        vals: 1D numpy array of constraint values (must be >= 0).
        jac: 2D numpy array (Jacobian matrix) of shape (num_constraints, 3*n).
    """
    # Reshape variables for easier access
    # Centers: (n, 2), Radii: (n,)
    centers = x_vars[:2*n].reshape(n, 2)
    radii = x_vars[2*n:]
    
    # Pre-calculate pairwise differences
    # centers[:, np.newaxis, :] is (n, 1, 2)
    # centers[np.newaxis, :, :] is (1, n, 2)
    # dx, dy: (n, n)
    dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
    dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
    
    # Distance matrix (n, n)
    dist_sq = dx**2 + dy**2
    
    # Radii sums matrix (n, n)
    r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    
    # Number of constraints
    # 4 boundaries per circle -> 4*n
    # n*(n-1)/2 overlap constraints
    n_bounds = 4 * n
    n_pairs = n * (n - 1) // 2
    total_constraints = n_bounds + n_pairs
    
    # Initialize Jacobian
    # Shape: (total_constraints, 3*n)
    jac = np.zeros((total_constraints, 3 * n))
    vals = np.zeros(total_constraints)
    
    # --- Boundary Constraints ---
    # 1. x_i - r_i >= 0
    # 2. 1 - x_i - r_i >= 0
    # 3. y_i - r_i >= 0
    # 4. 1 - y_i - r_i >= 0
    
    for i in range(n):
        # Indices in vars
        idx_x = 3 * i
        idx_y = 3 * i + 1
        idx_r = 3 * i + 2
        
        # Constraint 1: x_i - r_i
        c_idx = i
        vals[c_idx] = centers[i, 0] - radii[i]
        jac[c_idx, idx_x] = 1.0
        jac[c_idx, idx_r] = -1.0
        
        # Constraint 2: 1 - x_i - r_i
        c_idx = n + i
        vals[c_idx] = 1.0 - centers[i, 0] - radii[i]
        jac[c_idx, idx_x] = -1.0
        jac[c_idx, idx_r] = -1.0
        
        # Constraint 3: y_i - r_i
        c_idx = 2 * n + i
        vals[c_idx] = centers[i, 1] - radii[i]
        jac[c_idx, idx_y] = 1.0
        jac[c_idx, idx_r] = -1.0
        
        # Constraint 4: 1 - y_i - r_i
        c_idx = 3 * n + i
        vals[c_idx] = 1.0 - centers[i, 1] - radii[i]
        jac[c_idx, idx_y] = -1.0
        jac[c_idx, idx_r] = -1.0
        
    # --- Overlap Constraints ---
    # dist(i, j) >= r_i + r_j
    # sqrt(dx^2 + dy^2) - (r_i + r_j) >= 0
    # We use a small epsilon to avoid division by zero if centers coincide
    EPS = 1e-12
    
    pair_idx = n_bounds
    for i in range(n):
        for j in range(i + 1, n):
            d_val = np.sqrt(dist_sq[i, j] + EPS) # distance
            r_sum_val = radii[i] + radii[j]
            
            # Constraint value
            vals[pair_idx] = d_val - r_sum_val
            
            # Gradients
            # d/dx_i = dx / dist
            # d/dy_i = dy / dist
            # d/dr_i = -1
            # d/dx_j = -dx / dist
            # d/dy_j = -dy / dist
            # d/dr_j = -1
            
            # Indices
            idx_xi = 3 * i
            idx_yi = 3 * i + 1
            idx_ri = 3 * i + 2
            idx_xj = 3 * j
            idx_yj = 3 * j + 1
            idx_rj = 3 * j + 2
            
            inv_dist = 1.0 / d_val
            
            jac[pair_idx, idx_xi] = dx[i, j] * inv_dist
            jac[pair_idx, idx_yi] = dy[i, j] * inv_dist
            jac[pair_idx, idx_ri] = -1.0
            
            jac[pair_idx, idx_xj] = -dx[i, j] * inv_dist
            jac[pair_idx, idx_yj] = -dy[i, j] * inv_dist
            jac[pair_idx, idx_rj] = -1.0
            
            pair_idx += 1
            
    return vals, jac

def objective_func(x_vars, n):
    """
    Objective function: Minimize -sum(radii).
    """
    radii = x_vars[2*n:]
    return -np.sum(radii)

def objective_jac(x_vars, n):
    """
    Jacobian of the objective function.
    """
    jac = np.zeros(3 * n)
    # Derivative wrt r_i is -1
    for i in range(n):
        jac[3 * i + 2] = -1.0
    return jac

def generate_initial_guess(n, seed=None):
    """
    Generates a valid initial configuration using a hexagonal lattice 
    with random perturbations.
    """
    if seed is not None:
        np.random.seed(seed)
        
    # Create a hexagonal grid points
    # We want to cover the unit square densely
    points = []
    # Spacing roughly 1/sqrt(n) * constant
    # n=26, sqrt(26)~5.1. Spacing ~0.2-0.25
    spacing = 0.25 
    
    # Generate points in a slightly larger area to ensure coverage, then filter
    # Or just generate enough to pick n
    y_curr = 0
    row = 0
    while len(points) < n * 2: # Generate surplus
        x_curr = 0
        # Offset every other row
        offset = (spacing / 2) if (row % 2 == 1) else 0
        
        while x_curr < 1.5: # Generate slightly outside
            points.append([x_curr + offset, y_curr])
            x_curr += spacing
        y_curr += spacing * np.sqrt(3) / 2
        row += 1
        
    points = np.array(points)
    
    # Filter points inside [0, 1]x[0, 1]
    mask = (points[:, 0] >= 0) & (points[:, 0] <= 1) & \
           (points[:, 1] >= 0) & (points[:, 1] <= 1)
    valid_points = points[mask]
    
    # If we don't have enough, fill with random
    if len(valid_points) < n:
        random_pts = np.random.rand(n, 2)
        valid_points = np.vstack([valid_points, random_pts])
        
    # Shuffle and pick first n
    np.random.shuffle(valid_points)
    centers = valid_points[:n]
    
    # Add small random noise to avoid symmetry issues
    noise = np.random.uniform(-0.02, 0.02, size=centers.shape)
    centers += noise
    centers = np.clip(centers, 0.05, 0.95) # Ensure inside bounds with margin
    
    # Initial radii small to ensure no overlap initially
    radii = np.full(n, 0.01)
    
    # Construct x_vars
    x_vars = np.zeros(3 * n)
    for i in range(n):
        x_vars[3*i] = centers[i, 0]
        x_vars[3*i+1] = centers[i, 1]
        x_vars[3*i+2] = radii[i]
        
    return x_vars
