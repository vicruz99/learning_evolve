import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    np.random.seed(42)
    
    def objective(z):
        r = z[2::3]
        return -np.sum(r)

    def boundary_constraints(z):
        cons = []
        for i in range(n):
            x, y, r = z[3*i], z[3*i+1], z[3*i+2]
            cons.extend([x - r, 1 - (x + r), y - r, 1 - (y + r)])
        return np.array(cons)

    def nonoverlap_constraints(z):
        cons = []
        for i in range(n):
            xi, yi, ri = z[3*i], z[3*i+1], z[3*i+2]
            for j in range(i + 1, n):
                xj, yj, rj = z[3*j], z[3*j+1], z[3*j+2]
                dist_sq = (xi - xj)**2 + (yi - yj)**2
                cons.append(dist_sq - (ri + rj)**2)
        return np.array(cons)

    # Initial hexagonal packing
    centers = []
    r_init = 0.04
    j = 0
    while len(centers) < n:
        y = r_init + j * np.sqrt(3) * r_init
        if y + r_init > 1:
            break
        i = 0
        while True:
            x = r_init + i * (2 * r_init) + (j % 2) * r_init
            if x + r_init > 1:
                break
            centers.append([x, y])
            i += 1
        j += 1
    
    z0 = np.zeros(3 * n)
    for i in range(min(n, len(centers))):
        z0[3*i] = centers[i][0]
        z0[3*i+1] = centers[i][1]
        z0[3*i+2] = r_init
    
    # Fill remaining circles if any
    for i in range(len(centers), n):
        z0[3*i] = np.random.uniform(0.1, 0.9)
        z0[3*i+1] = np.random.uniform(0.1, 0.9)
        z0[3*i+2] = r_init

    best_z = None
    best_val = -np.inf

    for restart in range(10):
        pert = np.random.normal(0, 0.01, size=3*n)
        z_start = z0 + pert
        z_start[2::3] = np.clip(z_start[2::3], 0.01, 0.45)
        
        bounds = [(0, 1) for _ in range(2*n)] + [(0.001, 0.5) for _ in range(n)]
        
        res = minimize(
            objective,
            z_start,
            method='SLSQP',
            bounds=bounds,
            constraints=[
                {'type': 'ineq', 'fun': boundary_constraints},
                {'type': 'ineq', 'fun': nonoverlap_constraints}
            ],
            options={'maxiter': 500}
        )
        
        if not res.success and not res.nit == 500:
            continue
            
        curr_val = -res.fun
        if curr_val > best_val:
            best_val = curr_val
            best_z = res.x.copy()

    centers = best_z[:2*n].reshape(n, 2)
    radii = best_z[2::3]
    
    # Final clamp to satisfy validation tolerance
    radii = np.clip(radii, 1e-4, 0.499)
    centers = np.clip(centers, radii[:, np.newaxis], 1 - radii[:, np.newaxis])

    return centers, radii, np.sum(radii)
