import numpy as np
from scipy.optimize import minimize

# Precompute indices for pairwise distance calculations
INDICES = np.tril_indices(26, -1)

def compute_penalty(centers_flat, r, n=26):
    """Compute constraint violation penalty for a given radius r and center positions."""
    centers = centers_flat.reshape(n, 2)
    penalty = 0.0
    
    # Boundary constraints: centers must lie within [r, 1-r]
    dx = np.maximum(0, r - centers[:, 0]) + np.maximum(0, centers[:, 0] - (1 - r))
    dy = np.maximum(0, r - centers[:, 1]) + np.maximum(0, centers[:, 1] - (1 - r))
    penalty += np.sum(dx**2 + dy**2)
    
    # Inter-circle constraints: Euclidean distance between any pair must be >= 2r
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist_sq = np.sum(diff**2, axis=2)
    dists = np.sqrt(dist_sq[INDICES] + 1e-12)
    violations = 2.0 * r - dists
    penalty += np.sum(np.maximum(0, violations)**2)
    
    return penalty

def run_packing():
    n = 26
    low, high = 0.095, 0.105
    best_r = low
    best_centers = np.zeros((n, 2))
    
    # 1. Generate hexagonal initial layout
    r_init = 0.095
    y_step = r_init * np.sqrt(3)
    init_centers = []
    row = 0
    while len(init_centers) < n:
        x_start = r_init if row % 2 == 0 else 2 * r_init
        col = 0
        while x_start + col * 2 * r_init + r_init <= 1.0 and len(init_centers) < n:
            init_centers.append([x_start + col * 2 * r_init, r_init + row * y_step])
            col += 1
        row += 1
    init_centers = np.array(init_centers)
    
    # 2. Binary search for the maximum feasible radius
    for step in range(25):
        mid = (low + high) / 2
        best_pen = 1e9
        best_cen_mid = None
        
        # Try multiple perturbations to escape local minima
        for seed in range(3):
            rng = np.random.default_rng(seed + step * 3)
            c = init_centers.copy() + rng.normal(0, 0.005, init_centers.shape)
            c = np.clip(c, 0.05, 0.95)
            
            res = minimize(compute_penalty, c.flatten(), args=(mid, n),
                           method='L-BFGS-B', bounds=[(0, 1)] * (2 * n),
                           options={'maxiter': 1000, 'ftol': 1e-12, 'gtol': 1e-10})
            if res.fun < best_pen:
                best_pen = res.fun
                best_cen_mid = res.x.reshape(n, 2)
                
        # Feasibility check
        if best_pen < 1e-4:
            best_r = mid
            best_centers = best_cen_mid
            low = mid
        else:
            high = mid

    c = np.clip(c, 0.01, 0.99)

                         method='L-BFGS-B', bounds=[(0, 1)] * (2 * n),
                         options={'maxiter': 3000, 'ftol': 1e-15, 'gtol': 1e-12})
                         
    final_centers = final_res.x.reshape(n, 2)
    final_r = best_r
    final_radii = np.full(n, final_r)
    
    return final_centers, final_radii, final_radii.sum()
