

To maximize the sum of radii for 26 circles in a unit square, we treat the problem as a Non-Linear Programming (NLP) task. The objective is to maximize the sum of radii $\sum r_i$ subject to non-overlap constraints ($(x_i - x_j)^2 + (y_i - y_j)^2 \ge (r_i + r_j)^2$) and boundary constraints ($r_i \le x_i \le 1 - r_i$, etc.).

### Strategy
1.  **Initialization**: Since $N=26$ is near the capacity of a 5x5 grid, we initialize centers in a dense 5x5 pattern with one additional point. This provides a valid starting point where circles can "grow" without immediate overlaps.
2.  **Optimization**: We use the `scipy.optimize.minimize` function with the SLSQP method. This method is effective for handling the non-convex distance constraints and the linear boundary constraints.
3.  **Robustness**: To avoid local minima (where the optimizer gets stuck in a suboptimal grid), we run the optimization multiple times with different random perturbations of the initial centers. The best valid result across all runs is returned.
4.  **Constraint Handling**: The distance constraint is implemented in its squared form to avoid potential numerical issues with square roots, though both are mathematically equivalent for non-negative radii.

```python
import numpy as np
from scipy.optimize import minimize
import warnings

warnings.filterwarnings('ignore')

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # Objective function: Minimize negative sum of radii
    def objective(x):
        return -np.sum(x[2::3])

    # Inequality constraints: g(x) >= 0
    # 1. Boundary constraints: x_i >= r_i, 1 - x_i >= r_i, y_i >= r_i, 1 - y_i >= r_i
    # 2. Non-overlap: (x_i - x_j)^2 + (y_i - y_j)^2 >= (r_i + r_j)^2

    constraints = []
    for i in range(n):
        # Boundary constraints for circle i
        # x - r >= 0
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: x[3*i] - x[3*i+2]})
        # 1 - x - r >= 0
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: 1.0 - x[3*i] - x[3*i+2]})
        # y - r >= 0
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: x[3*i+1] - x[3*i+2]})
        # 1 - y - r >= 0
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: 1.0 - x[3*i+1] - x[3*i+2]})

    # Overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            def cb_overlap(x, i=i, j=j):
                xi, yi, ri = x[3*i], x[3*i+1], x[3*i+2]
                xj, yj, rj = x[3*j], x[3*j+1], x[3*j+2]
                return (xi - xj)**2 + (yi - yj)**2 - (ri + rj)**2
            constraints.append({'type': 'ineq', 'fun': cb_overlap})

    # Bounds: x, y in [0, 1], r in [0, 0.5]
    bounds = [(0, 1)] * (2 * n) + [(0, 0.5)] * n

    best_result = None
    best_sum_r = -1.0

    # Run multiple times with different seeds to avoid local minima
    for seed in range(10):
        np.random.seed(seed)
        
        # Initialize centers: 5x5 grid plus one random point
        centers_init = np.linspace(0.1, 0.9, 5).reshape(1, -1)
        y_grid = np.linspace(0.1, 0.9, 5).reshape(-1, 1)
        grid_centers = np.repeat(centers_init, 5, axis=0) + np.repeat(y_grid, 5, axis=1)
        
        # Add one extra center for the 26th circle
        extra_center = np.array([np.random.uniform(0.2, 0.8), np.random.uniform(0.2, 0.8)])
        x0 = np.concatenate([grid_centers.ravel(), extra_center.ravel(), np.zeros(n * 2)])
        
        # Initial small radii to ensure feasibility
        x0[2::3] = 0.04

        # Optimization
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints,
                           options={'maxiter': 500, 'ftol': 1e-12})
            
            if res.success or -res.fun > best_sum_r:
                # Extract valid radii and centers
                r_vals = res.x[2::3]
                centers = np.column_stack((res.x[0::3], res.x[1::3]))
                
                # Enforce non-negative radii (fix numerical issues)
                r_vals = np.maximum(r_vals, 0.0)
                
                sum_r = np.sum(r_vals)
                if sum_r > best_sum_r:
                    best_sum_r = sum_r
                    best_result = (centers, r_vals)
        except Exception:
            continue

    if best_result is not None:
        return best_result[0], best_result[1], best_sum_r
    
    # Fallback: 26 circles of small radius in a grid
    fallback_centers = np.random.rand(n, 2) * 0.8 + 0.1
    fallback_radii = np.full(n, 0.05)
    return fallback_centers, fallback_radii, np.sum(fallback_radii)
```
