import numpy as np
from scipy.optimize import minimize

def compute_valid_radii(centers):
    """Compute maximum valid radii for a given set of centers."""
    n = centers.shape[0]
    # Pairwise distances
    D = np.sqrt(np.sum((centers[:, None, :] - centers[None, :, :])**2, axis=2))
    np.fill_diagonal(D, np.inf)
    min_D = np.min(D, axis=1) / 2.0
    
    # Boundary constraints
    bounds = np.minimum(np.minimum(centers[:, 0], 1 - centers[:, 0]), 
                        np.minimum(centers[:, 1], 1 - centers[:, 1]))
    return np.minimum(min_D, bounds)

def neg_sum_radii(x):
    """Objective function to minimize (negative sum of radii)."""
    centers = np.clip(x.reshape(-1, 2), 1e-6, 1 - 1e-6)
    radii = compute_valid_radii(centers)
    return -np.sum(radii)
