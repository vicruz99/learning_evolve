import numpy as np
from scipy.optimize import minimize


def packing_objective(vars, n, penalty_weight):
    """Objective: negative sum of radii plus penalty for constraint violations."""
    centers = vars[:2 * n].reshape(n, 2)
    radii = vars[2 * n:3 * n]

    obj = -np.sum(radii)
    penalty = 0.0

    # Boundary violations
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        v = x - r
        if v < 0:
            penalty += penalty_weight * v * v
        v = x + r - 1.0
        if v > 0:
            penalty += penalty_weight * v * v
        v = y - r
        if v < 0:
            penalty += penalty_weight * v * v
        v = y + r - 1.0
        if v > 0:
            penalty += penalty_weight * v * v

    # Overlap violations
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            min_dist = radii[i] + radii[j]
            v = min_dist - dist
            if v > 0:
                penalty += penalty_weight * v * v

    return obj + penalty


def initialize_grid(n):
    """Initialize circles in a grid pattern."""
    centers = np.zeros((n, 2))
    rows = int(np.ceil(np.sqrt(n)))
    cols = int(np.ceil(n / rows))
    idx = 0
    for i in range(rows):
        for j in range(cols):
            if idx >= n:
                break
            x = (j + 0.5) / cols
            y = (i + 0.5) / rows
            centers[idx] = [x, y]
            idx += 1
    return centers


def initialize_hexagonal(n):

                idx += 1
        row += 1
    while idx < n:
        centers[idx] = [0.5, 0.5]
        idx += 1
    return centers

def optimize_packing(centers, radii, n):
    """Optimize packing with gradually increasing penalty."""
    x0 = np.concatenate([centers.flatten(), radii])

    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n

    penalty_weight = 50
    for iteration in range(20):
        result = minimize(
            packing_objective,
            x0,
            method='L-BFGS-B',
            args=(n, penalty_weight),
            bounds=bounds,
            options={
                'maxiter': 5000,
                'ftol': 1e-15,
                'gtol': 1e-10,
                'eps': 1e-12
            }
        )
        x0 = result.x.copy()
        penalty_weight *= 3
        if penalty_weight > 500000:
            break

    centers_opt = x0[:2 * n].reshape(n, 2)
    radii_opt = x0[2 * n:3 * n]
    radii_opt = np.maximum(radii_opt, 1e-10)

    return centers_opt, radii_opt


def is_valid(centers, radii, n):
    """Check if packing is valid."""
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-10 or x + r > 1.0 + 1e-10:
            return False
        if y - r < -1e-10 or y + r > 1.0 + 1e-10:
            return False
        if r < 0:
            return False

    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            if dist < radii[i] + radii[j] - 1e-10:
                return False
    return True


def fix_solution(centers, radii, n):
    """Fix any minor constraint violations by reducing radii."""
    radii = np.copy(radii)

    # Fix boundary violations
    for i in range(n):
        x, y = centers[i]
        max_r = min(x, 1 - x, y, 1 - y)
        if max_r < radii[i]:
            radii[i] = max(0, max_r - 1e-8)

    # Fix overlap violations
    for _ in range(100):
        max_violation = 0
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                if dist < 1e-15:
                    dist = 1e-15
                min_dist = radii[i] + radii[j]
                if dist < min_dist - 1e-12:
                    excess = min_dist - dist
                    # Reduce both radii proportionally
                    factor = dist / min_dist if min_dist > 0 else 0
                    radii[i] *= factor
                    radii[j] *= factor
                    max_violation = excess
        if max_violation < 1e-12:
            break

    return radii
