import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    Pack 26 circles in a unit square to maximize the sum of radii.
    Uses SLSQP optimization on a hexagonal initialization.
    """

    # Spacing parameters for the lattice generation
    dx = 1.0
    dy = np.sqrt(3) / 2.0 * 1.0

    # Generate a grid larger than needed and pick 26 points
    # Approximate layout: 6 rows
    for row in range(8):
        for col in range(8):
            x = col * dx + (row % 2) * (dx / 2)
            y = row * dy
            points.append([x, y])
            
    points = np.array(points)
    
    # Select 26 points closest to the center of the generated cluster
    # or just the first 26 if they fit well. 
    # Let's pick the first 26, which form a compact block.
    # Then we will scale and center them.
    initial_centers = points[:n]
    
    # Normalize centers to fit in [0, 1] x [0, 1] with some margin
    min_coords = initial_centers.min(axis=0)
    max_coords = initial_centers.max(axis=0)
    center = (min_coords + max_coords) / 2.0
    span = max_coords - min_coords
    span = np.max(span) # Scale based on larger dimension
    
    # Scale to 0.8 size to leave room for optimizer to expand
    scale = 0.8 / span
    initial_centers = (initial_centers - center) * scale + 0.5
    
    # Initial radii: small, to ensure feasibility
    initial_radii = np.full(n, 0.02)
    
    # Flatten variables: [x1, y1, r1, x2, y2, r2, ...]
    x0 = np.zeros(3 * n)
    for i in range(n):
        x0[3*i] = initial_centers[i, 0]
        x0[3*i+1] = initial_centers[i, 1]
        x0[3*i+2] = initial_radii[i]
    
    # --- 2. Optimization ---
    
    # Objective function: Minimize negative sum of radii
    def objective(x):
        return -np.sum(x[2::3])
    
    # Constraints
    constraints = []
    
    # Boundary constraints: x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
    # We can add these as bounds or explicit constraints. 
    # SLSQP handles bounds well, but let's keep r >= 0.
    # Bounds: x in [0,1], y in [0,1], r in [0, 0.5]
    bounds = []
    for i in range(n):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 0.5)) # r
        
    # Non-overlap constraints: (xi-xj)^2 + (yi-yj)^2 >= (ri+rj)^2
    # Implemented as (xi-xj)^2 + (yi-yj)^2 - (ri+rj)^2 >= 0
    # To speed up, we only check constraints. SLSQP evaluates them.
    
    # We define a single constraint function that returns an array of values
    # This is more efficient than defining many individual constraint dicts
    
    def constraint_func(x):
        vals = []
        centers = np.zeros((n, 2))
        radii = np.zeros(n)
        for i in range(n):
            centers[i, 0] = x[3*i]
            centers[i, 1] = x[3*i+1]
            radii[i] = x[3*i+2]
            
        # Boundary constraints (optional if bounds are used, but explicit is safer for some solvers)
        # Actually SLSQP bounds are strict, so we don't need explicit boundary constraints
        # for x, y, r limits. But we need to ensure circles are inside.
        # The bounds [0,1] for centers and [0, 0.5] for radii ensure x >= 0.
        # But we need x - r >= 0.
        # So we must add explicit constraints for boundaries.
        
        for i in range(n):
            xi, yi, ri = x[3*i], x[3*i+1], x[3*i+2]
            # Left, Right, Bottom, Top
            vals.append(xi - ri)
            vals.append(1.0 - xi - ri)
            vals.append(yi - ri)

                r_sum_sq = (x[3*i+2] + x[3*j+2])**2
                vals.append(dist_sq - r_sum_sq)
                
        return np.array(vals)

    # Define constraint dictionary
    cons = {'type': 'ineq', 'fun': constraint_func}

    # Using SLSQP. Method is robust for this type of problem.
    # We might need to run it a couple of times or with different starts if it fails, 
    # but a good hexagonal start usually works well.
    
    try:

        # Fallback to simple grid if optimizer fails
        x_opt = x0

    radii = np.zeros(n)
    for i in range(n):
        centers[i, 0] = x_opt[3*i]

        radii[i] = x_opt[3*i+2]
        
    # Clip small negative values due to numerical errors
    radii = np.clip(radii, 0, None)
    
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
