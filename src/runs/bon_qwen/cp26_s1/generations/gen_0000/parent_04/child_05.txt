import numpy as np

def compute_radii(centers):
    """
    Computes the maximum feasible radius for each circle given their centers.
    Constraints:
    1. Circle must be inside [0, 1]x[0, 1] -> r <= min(x, 1-x, y, 1-y)
    2. Circles must not overlap -> r_i + r_j <= dist(i, j) -> r_i <= 0.5 * dist(i, j)
    """
    n = centers.shape[0]
    # Initialize radii with a large value
    radii = np.full(n, np.inf)
    
    # 1. Wall constraints
    # Distance to left wall (x=0)
    radii = np.minimum(radii, centers[:, 0])
    # Distance to right wall (x=1)
    radii = np.minimum(radii, 1.0 - centers[:, 0])
    # Distance to bottom wall (y=0)
    radii = np.minimum(radii, centers[:, 1])
    # Distance to top wall (y=1)
    radii = np.minimum(radii, 1.0 - centers[:, 1])
    
    # 2. Neighbor constraints
    # Compute pairwise distances
    # diff shape: (n, n, 2)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    # Set diagonal to infinity (distance to self)
    np.fill_diagonal(dists, np.inf)
    
    # For each circle, the max radius is limited by half the distance to the nearest neighbor
    min_dists = np.min(dists, axis=1)
    radii = np.minimum(radii, min_dists / 2.0)
    
    return radii

def compute_forces(centers, radii):
    """
    Computes a force vector for each center to help increase the sum of radii.
    Heuristic: Move away from the constraint that limits the radius.
    """
    n = centers.shape[0]
    forces = np.zeros_like(centers)
    
    # We need to identify which constraint is active for each circle.
    # A constraint is active if dist to constraint is approx equal to current radius.
    # Tolerance for checking active constraints
    tol = 1e-4 
    
    # Precompute distances for efficiency
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    # Check wall constraints
    # Left wall (x=0)
    mask_left = np.abs(centers[:, 0] - radii) < tol
    forces[mask_left, 0] += 1.0
    
    # Right wall (x=1)
    mask_right = np.abs((1.0 - centers[:, 0]) - radii) < tol
    forces[mask_right, 0] -= 1.0
    
    # Bottom wall (y=0)
    mask_bottom = np.abs(centers[:, 1] - radii) < tol
    forces[mask_bottom, 1] += 1.0
    
    # Top wall (y=1)
    mask_top = np.abs((1.0 - centers[:, 1]) - radii) < tol
    forces[mask_top, 1] -= 1.0
    
    # Check neighbor constraints
    # If r_i + r_j approx dist(i, j), they are touching.
    # We want to push i away from j.
    # Vector from j to i is (centers[i] - centers[j])
    # Normalized vector is direction to move i away from j.
    
    # We can vectorize this check
    # Condition: dists[i, j] <= radii[i] + radii[j] + tol
    # Actually, with our radius calculation, dists[i,j] >= radii[i] + radii[j] is guaranteed.
    # Equality holds if constraint is active.
    
    sum_radii_ij = radii[:, np.newaxis] + radii[np.newaxis, :]
    # Identify touching pairs
    touching = dists < sum_radii_ij + tol
    
    # For each pair (i, j) that is touching, add repulsive force
    # Force on i: direction (centers[i] - centers[j]) normalized
    # Force on j: direction (centers[j] - centers[i]) normalized
    
    # To avoid division by zero, add small epsilon to dists where 0 (though diagonal is inf)
    safe_dists = np.where(dists < 1e-9, 1e-9, dists)
    unit_vecs = diff / safe_dists[:, :, np.newaxis] # (n, n, 2)
    
    # Accumulate forces
    # If touching, push away.
    # We can sum the unit vectors for all touching neighbors.
    for i in range(n):
        # Find neighbors j touching i
        neighbors = np.where(touching[i, :])[0]
        if len(neighbors) > 0:
            # Sum of unit vectors pointing away from neighbors
            # unit_vecs[i, j] is vector from j to i.
            # So adding this pushes i away from j.
            f_sum = np.sum(unit_vecs[i, neighbors, :], axis=0)
            forces[i] += f_sum

    return forces
