import numpy as np
from scipy.optimize import minimize


def run_packing():
    n = 26
    
    best_centers = None
    best_radii = None
    best_sum = 0.0
    
    for seed in range(8):
        centers, radii = initialize_hexagonal(n, seed)
        centers, radii = expand_and_resolve(centers, radii, n)
        centers, radii = local_optimize_scipy(centers, radii, n)
        
        # Final cleanup
        centers, radii = enforce_constraints(centers, radii, n)
        
        s = np.sum(radii)
        if s > best_sum:
            best_sum = s
            best_centers = centers.copy()
            best_radii = radii.copy()
    
    return best_centers, best_radii, best_sum


def initialize_hexagonal(n, seed=0):
    rng = np.random.RandomState(seed)
    row_counts = [6, 5, 6, 5, 4]
    r = 0.075 + rng.uniform(-0.005, 0.005)
    centers = []
    
    for row, count in enumerate(row_counts):
        y = r + row * np.sqrt(3) * r
        x_start = r if row % 2 == 0 else 2 * r
        for i in range(count):
            x = x_start + i * 2 * r
            centers.append([x, y])
    
    centers = np.array(centers)
    centers += rng.randn(n, 2) * 0.002
    radii = np.ones(n) * r
    return centers, radii


def expand_and_resolve(centers, radii, n, num_steps=30000):
    dt = 0.005
    expansion_per_step = 2e-7
    repulsion_strength = 50.0
    boundary_strength = 100.0
    
    for step in range(num_steps):
        radii += expansion_per_step
        
        forces = np.zeros_like(centers)
        
        for i in range(n):
            xi, yi = centers[i]
            ri = radii[i]
            
            for j in range(i + 1, n):
                dx = centers[j][0] - xi
                dy = centers[j][1] - yi
                dist_sq = dx * dx + dy * dy
                dist = np.sqrt(dist_sq) if dist_sq > 1e-15 else 1e-8
                min_dist = ri + radii[j]
                
                if dist < min_dist:
                    overlap = min_dist - dist
                    fx = dx / dist * overlap * repulsion_strength
                    fy = dy / dist * overlap * repulsion_strength
                    forces[i][0] -= fx
                    forces[i][1] -= fy
                    forces[j][0] += fx
                    forces[j][1] += fy
            
            for d in range(2):
                coord = centers[i][d]
                if coord - ri < 0:
                    forces[i][d] += boundary_strength * (ri - coord)
                elif coord + ri > 1:
                    forces[i][d] -= boundary_strength * (coord + ri - 1)
        
        centers += dt * forces
        dt *= 0.9999
        
        for i in range(n):
            centers[i][0] = max(radii[i], min(1 - radii[i], centers[i][0]))
            centers[i][1] = max(radii[i], min(1 - radii[i], centers[i][1]))
    
    return centers, radii


def compute_overlap_penalty(centers, radii, n):
    penalty = 0.0
    for i in range(n):
        for d in range(2):
            if centers[i][d] - radii[i] < 0:
                penalty += (centers[i][d] - radii[i]) ** 2
            if centers[i][d] + radii[i] > 1:
                penalty += (centers[i][d] + radii[i] - 1) ** 2
    
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i][0] - centers[j][0]
            dy = centers[i][1] - centers[j][1]
            dist = np.sqrt(dx * dx + dy * dy)
            min_dist = radii[i] + radii[j]
            if dist < min_dist:
                penalty += (min_dist - dist) ** 2
    
    return penalty


def objective_func(params, n):
    centers = params[:2 * n].reshape(n, 2)
    radii = params[2 * n:]
    
    return -np.sum(radii) + 500.0 * compute_overlap_penalty(centers, radii, n)


def local_optimize_scipy(centers, radii, n, num_restarts=3):
    best_centers = centers.copy()
    best_radii = radii.copy()
    best_obj = objective_func(np.concatenate([centers.flatten(), radii]), n)
    
    for restart in range(num_restarts):
        rng = np.random.RandomState(restart * 7 + 3)
        params = np.concatenate([best_centers.flatten(), best_radii])
        params += rng.randn(len(params)) * 0.001
        
        bounds = []
        for i in range(n):
            bounds.extend([(1e-6, 1 - 1e-6), (1e-6, 1 - 1e-6)])
            bounds.append((1e-6, 0.5))
        
        result = minimize(
            objective_func,
            params,
            args=(n,),
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 2000, 'ftol': 1e-15, 'gtol': 1e-10}
        )
        
        obj_val = result.fun
        if obj_val < best_obj:
            best_obj = obj_val
            best_centers = result.x[:2 * n].reshape(n, 2)
            best_radii = result.x[2 * n:]
    
    return best_centers, best_radii


def enforce_constraints(centers, radii, n):
    for i in range(n):
        radii[i] = max(1e-8, radii[i])
        centers[i][0] = max(radii[i], min(1 - radii[i], centers[i][0]))
        centers[i][1] = max(radii[i], min(1 - radii[i], centers[i][1]))
    
    for _ in range(50):
        changed = False
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = np.sqrt(dx * dx + dy * dy)
                if dist < radii[i] + radii[j] - 1e-12 and dist > 1e-12:
                    excess = radii[i] + radii[j] - dist
                    if radii[i] > radii[j]:
                        radii[i] = max(1e-8, radii[i] - excess * 0.5)
                    else:
                        radii[j] = max(1e-8, radii[j] - excess * 0.5)
                    changed = True
        
        if not changed:
            break
    
    for i in range(n):
        centers[i][0] = max(radii[i], min(1 - radii[i], centers[i][0]))
        centers[i][1] = max(radii[i], min(1 - radii[i], centers[i][1]))
    
    return centers, radii
