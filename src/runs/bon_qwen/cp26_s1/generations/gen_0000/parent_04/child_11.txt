import numpy as np
from scipy.optimize import minimize

def compute_penalty(v, n):
    """Compute differentiable penalty for boundary and overlap violations."""
    pen = 0.0
    for i in range(n):
        xi, yi, ri = v[3*i], v[3*i+1], v[3*i+2]
        # Boundary penalties (squared hinge loss)
        pen += np.maximum(0.0, ri - xi)**2
        pen += np.maximum(0.0, xi + ri - 1.0)**2
        pen += np.maximum(0.0, ri - yi)**2
        pen += np.maximum(0.0, yi + ri - 1.0)**2
    # Overlap penalties
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(v[3*i] - v[3*j], v[3*i + 1] - v[3*j + 1])
            sum_r = v[3*i + 2] + v[3*j + 2]
            overlap = sum_r - dist
            pen += np.maximum(0.0, overlap)**2
    return pen

def objective_fn(v, n, penalty_weight):
    """Objective function: minimize negative sum of radii + penalty."""
    base = -np.sum(v[2::3])
    pen = compute_penalty(v, n)
    return base + penalty_weight * pen

def get_initial_guess(n):
    """Generate a hexagonal grid initial guess."""
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.08)
    row_y = np.linspace(0.12, 0.88, 5)
    idx = 0
    for r_idx in range(5):
        y = row_y[r_idx]
        if r_idx % 2 == 0:
            xs = np.linspace(0.08, 0.92, 6)
        else:
            xs = np.linspace(0.16, 0.84, 5)
        for x in xs:
            if idx < n:
                centers[idx] = [x, y]
                idx += 1
    v0 = np.zeros(3 * n)
    for i in range(n):
        v0[3*i] = centers[i, 0]
        v0[3*i + 1] = centers[i, 1]
        v0[3*i + 2] = radii[i]
    return v0

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    v0 = get_initial_guess(n)
    bounds = [(0, 1), (0, 1), (0, 0.5)] * n
    
    # Iterative penalty increase to converge to feasible region
    current_v = v0.copy()
    weights = [100, 1000, 5000]
    for lw in weights:
        res = minimize(objective_fn, current_v, args=(n, lw), 
                       method='L-BFGS-B', bounds=bounds, 
                       options={'maxiter': 3000, 'ftol': 1e-12, 'disp': False})
        current_v = res.x
        
    centers_out = current_v[:3*n].reshape(n, 3)[:, :2]
    radii_out = current_v[2::3]
    radii_out = np.maximum(radii_out, 0.0)
    
    # Safety scaling to strictly satisfy validator tolerances
    scale = 1.0
    for i in range(n):
        r = radii_out[i]
        if r > 0:
            min_d = min(centers_out[i, 0], 1.0 - centers_out[i, 0], 
                        centers_out[i, 1], 1.0 - centers_out[i, 1])
            if min_d < r:
                scale = min(scale, min_d / r)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(centers_out[i, 0] - centers_out[j, 0], 
                         centers_out[i, 1] - centers_out[j, 1])
            sum_r = radii_out[i] + radii_out[j]
            if d < sum_r:
                scale = min(scale, d / sum_r)
                
    radii_out *= scale * 0.999
    return centers_out, radii_out, float(np.sum(radii_out))
