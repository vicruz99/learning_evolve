def get_max_sum_radii(centers):
    """
    Solves the LP to find optimal radii for fixed centers.
    Maximizes sum(r_i) subject to non-overlap and boundary constraints.
    """
    n = len(centers)
    c = np.ones(n) * -1  # Coefficients for objective: max sum(r) -> min -sum(r)
    
    # Constraints: r_i + r_j <= distance(centers[i], centers[j])
    # And r_i <= distance to walls
    
    # Prepare inequality matrix A_ub and vector b_ub
    # r_i + r_j <= dist_ij
    
    constraints_A = []
    constraints_b = []
    
    # Wall constraints: r_i <= min(x, 1-x, y, 1-y)
    for i in range(n):
        x, y = centers[i]

        # Create a row with 1 at index i
        row = np.zeros(n)
        row[i] = 1.0
        constraints_A.append(row)
        constraints_b.append(dist_wall)
        
    # Pairwise constraints: r_i + r_j <= dist_ij
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            constraints_A.append(row)
            constraints_b.append(dist)
            
    constraints_A = np.array(constraints_A)
    constraints_b = np.array(constraints_b)
    
    # Bounds: r_i >= 0
    bounds = [(0, None) for _ in range(n)]
    
    try:
        res = linprog(c, A_ub=constraints_A, b_ub=constraints_b, bounds=bounds, method='highs')

            return 0.0, np.zeros(n)
