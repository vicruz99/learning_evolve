import numpy as np
from scipy.optimize import minimize


def compute_penalty_and_objective(x, n, big_M):
    """Compute negative sum of radii plus penalty for constraint violations."""
    centers = x[:2 * n].reshape((n, 2))
    radii = x[2 * n:]

    obj = -np.sum(radii)
    penalty = 0.0

    # Boundary constraint penalties
    for i in range(n):
        if radii[i] > centers[i, 0]:
            penalty += big_M * (radii[i] - centers[i, 0]) ** 2
        if centers[i, 0] + radii[i] > 1.0:
            penalty += big_M * (centers[i, 0] + radii[i] - 1.0) ** 2
        if radii[i] > centers[i, 1]:
            penalty += big_M * (radii[i] - centers[i, 1]) ** 2
        if centers[i, 1] + radii[i] > 1.0:
            penalty += big_M * (centers[i, 1] + radii[i] - 1.0) ** 2
        if radii[i] < 0:
            penalty += big_M * radii[i] ** 2

    # Overlap penalties
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist_sq = dx * dx + dy * dy
            dist = np.sqrt(max(0, dist_sq))
            overlap = radii[i] + radii[j] - dist
            if overlap > 0:
                penalty += big_M * overlap ** 2

    return obj + penalty


def is_valid_packing(centers, radii, n):
    """Check if packing is valid."""
    for i in range(n):
        if radii[i] < -1e-10:
            return False
        if centers[i, 0] - radii[i] < -1e-10:
            return False
        if centers[i, 0] + radii[i] > 1.0 + 1e-10:
            return False
        if centers[i, 1] - radii[i] < -1e-10:
            return False
        if centers[i, 1] + radii[i] > 1.0 + 1e-10:
            return False

    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            if dist < radii[i] + radii[j] - 1e-10:
                return False

    return True


def push_apart_algorithm(n, centers_init, radii_init, max_iter=2000):
    """Iteratively grow circles and push apart overlapping ones."""
    centers = centers_init.copy()
    radii = radii_init.copy()

    for iteration in range(max_iter):
        # Grow all circles slightly
        growth = 1e-5
        radii = radii + growth

        # Push apart overlapping circles multiple times per iteration
        for _ in range(50):
            any_moved = False
            for i in range(n):
                for j in range(i + 1, n):
                    dx = centers[i, 0] - centers[j, 0]
                    dy = centers[i, 1] - centers[j, 1]
                    dist_sq = dx * dx + dy * dy
                    dist = np.sqrt(dist_sq)
                    min_dist = radii[i] + radii[j]

                    if dist < min_dist - 1e-12:
                        if dist < 1e-12:
                            dir_x = np.random.randn()
                            dir_y = np.random.randn()
                            norm = np.sqrt(dir_x * dir_x + dir_y * dir_y)
                            dir_x = dir_x / norm
                            dir_y = dir_y / norm
                        else:
                            dir_x = dx / dist
                            dir_y = dy / dist

                        push = (min_dist - dist) / 2.0 + 1e-7
                        centers[i, 0] += dir_x * push
                        centers[i, 1] += dir_y * push
                        centers[j, 0] -= dir_x * push
                        centers[j, 1] -= dir_y * push
                        any_moved = True

            if not any_moved:
                break

        # Project centers to stay within feasible region
        for i in range(n):
            centers[i, 0] = max(radii[i], min(1.0 - radii[i], centers[i, 0]))
            centers[i, 1] = max(radii[i], min(1.0 - radii[i], centers[i, 1]))

    return centers, radii


def create_hexagonal_init(n, pattern_idx=0):
    """Create an initial hexagonal-ish arrangement of circles."""
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    if pattern_idx == 0:
        # Pattern 0: 5-6-5-6-4 rows
        rows_config = [5, 6, 5, 6, 4]
    elif pattern_idx == 1:
        # Pattern 1: 6-5-6-5-4 rows  
        rows_config = [6, 5, 6, 5, 4]
    elif pattern_idx == 2:
        # Pattern 2: 5-5-5-5-6 rows
        rows_config = [5, 5, 5, 5, 6]
    elif pattern_idx == 3:
        # Pattern 3: 6-5-5-5-5 rows
        rows_config = [6, 5, 5, 5, 5]
    else:
        rows_config = [5, 6, 5, 6, 4]

    y_vals = np.linspace(0.12, 0.88, 5)

    idx = 0
    for row in range(5):
        ncols = rows_config[row]
        x_start = 0.1
        x_end = 0.9
        if ncols == 1:
            x_vals = np.array([0.5])
        elif ncols == 2:
            x_vals = np.array([0.3, 0.7])
        elif ncols == 3:
            x_vals = np.array([0.2, 0.5, 0.8])
        elif ncols == 4:
            x_vals = np.linspace(x_start, x_end, ncols)
        elif ncols == 5:
            x_vals = np.linspace(x_start, x_end, ncols)
        elif ncols == 6:
            x_vals = np.linspace(x_start, x_end, ncols)
        else:
            x_vals = np.linspace(x_start, x_end, ncols)

        # Offset odd rows for hexagonal packing
        if row % 2 == 1 and ncols > 1:
            spacing = (x_end - x_start) / (ncols - 1)
            x_vals = x_vals + spacing * 0.5
            x_vals = np.clip(x_vals, 0.15, 0.85)

        for xv in x_vals:
            if idx < n:
                centers[idx] = [xv, y_vals[row]]
                radii[idx] = 0.06
                idx += 1

    return centers, radii


def run_packing():
    n = 26
    big_M = 1e7

    best_sum = 0.0
    best_centers = None
    best_radii = None

    # Try multiple initial patterns and random seeds
    for pattern_idx in range(4):
        for seed in range(8):
            np.random.seed(seed)

            # Create initial configuration
            centers_init, radii_init = create_hexagonal_init(n, pattern_idx)

            # Add small random perturbations
            centers_init = centers_init + np.random.randn(n, 2) * 0.02
            centers_init = np.clip(centers_init, 0.1, 0.9)

            # Run push-apart algorithm
            centers_pa, radii_pa = push_apart_algorithm(n, centers_init, radii_init, max_iter=1500)

            if is_valid_packing(centers_pa, radii_pa, n):
                current_sum = np.sum(radii_pa)
                if current_sum > best_sum:
                    best_sum = current_sum
                    best_centers = centers_pa.copy()
                    best_radii = radii_pa.copy()

            # Further refine with gradient-based optimization
            x0 = np.concatenate([centers_pa.flatten(), radii_pa])

            bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.25)] * n

            # Use L-BFGS-B with penalty method
            result = minimize(
                compute_penalty_and_objective,
                x0,
                method='L-BFGS-B',
                bounds=bounds,
                args=(n, big_M),
                options={
                    'maxiter': 50000,
                    'ftol': 1e-18,
                    'gtol': 1e-12,
                    'maxfun': 100000
                }
            )

            centers_opt = result.x[:2 * n].reshape((n, 2))
            radii_opt = result.x[2 * n:]

            # Clip radii to be non-negative
            radii_opt = np.maximum(radii_opt, 0)

            # Adjust centers to be feasible if needed
            for i in range(n):
                centers_opt[i, 0] = max(radii_opt[i], min(1.0 - radii_opt[i], centers_opt[i, 0]))
                centers_opt[i, 1] = max(radii_opt[i], min(1.0 - radii_opt[i], centers_opt[i, 1]))

            if is_valid_packing(centers_opt, radii_opt, n):
                current_sum = np.sum(radii_opt)
                if current_sum > best_sum:
                    best_sum = current_sum
                    best_centers = centers_opt.copy()
                    best_radii = radii_opt.copy()

    # If we still have a valid result, return it
    if best_centers is not None:
        return best_centers, best_radii, np.sum(best_radii)

    # Fallback: return the push-apart result from the first pattern
    centers_init, radii_init = create_hexagonal_init(n, 0)
    centers_pa, radii_pa = push_apart_algorithm(n, centers_init, radii_init, max_iter=2000)
    return centers_pa, radii_pa, np.sum(radii_pa)
