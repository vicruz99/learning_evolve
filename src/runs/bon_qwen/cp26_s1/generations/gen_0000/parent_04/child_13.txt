import numpy as np
from scipy.optimize import minimize


def obj_func(vars_flat, n):
    """Objective: minimize negative sum of radii."""
    radii = vars_flat[2 * n:]
    return -np.sum(radii)


def make_boundary_x_lower(i, n):
    def constraint(vars_flat):
        return vars_flat[2 * i] - vars_flat[2 * n + i]
    return constraint


def make_boundary_x_upper(i, n):
    def constraint(vars_flat):
        return 1.0 - vars_flat[2 * i] - vars_flat[2 * n + i]
    return constraint


def make_boundary_y_lower(i, n):
    def constraint(vars_flat):
        return vars_flat[2 * i + 1] - vars_flat[2 * n + i]
    return constraint


def make_boundary_y_upper(i, n):
    def constraint(vars_flat):
        return 1.0 - vars_flat[2 * i + 1] - vars_flat[2 * n + i]
    return constraint


def make_overlap_constraint(i, j, n):
    def constraint(vars_flat):
        dx = vars_flat[2 * i] - vars_flat[2 * j]
        dy = vars_flat[2 * i + 1] - vars_flat[2 * j + 1]
        dist_sq = dx * dx + dy * dy
        r_sum = vars_flat[2 * n + i] + vars_flat[2 * n + j]
        return dist_sq - r_sum * r_sum
    return constraint


def generate_hexagonal_init(n, r_init):
    """Generate hexagonal packing initial configuration."""
    centers_list = []
    radii_list = []
    
    row_configs = [5, 5, 5, 5, 6]
    y = r_init
    
    for i, ncols in enumerate(row_configs):
        if i % 2 == 0:
            x_start = r_init
        else:
            x_start = 2.0 * r_init
        for j in range(ncols):
            x = x_start + j * 2.0 * r_init
            centers_list.append([x, y])
            radii_list.append(r_init)
        y += r_init * np.sqrt(3.0)
    
    return np.array(centers_list), np.array(radii_list)


def generate_grid_init(n, r_init):
    """Generate grid packing initial configuration."""
    centers_list = []
    radii_list = []
    
    # 5x5 grid = 25, plus 1
    cols = 5
    rows = 5
    count = 0

        for j in range(cols):
            if count >= n:
                break
            x = r_init + j * 2.0 * r_init
            y = r_init + i * 2.0 * r_init
            centers_list.append([x, y])
            radii_list.append(r_init)
            count += 1
        if count >= n:
            break
    
    # Add remaining circles in gaps
    while len(centers_list) < n:
        # Place in center of a gap
        centers_list.append([0.5, 0.5])
        radii_list.append(r_init * 0.5)
    
    return np.array(centers_list), np.array(radii_list)


def optimize_packing(centers_init, radii_init, n):
    """Run optimization from given initial configuration."""
    x0 = np.concatenate([centers_init.flatten(), radii_init])
    
    # Build constraints list
    constraints = []
    
    for i in range(n):
        constraints.append({'type': 'ineq', 'fun': make_boundary_x_lower(i, n)})
        constraints.append({'type': 'ineq', 'fun': make_boundary_x_upper(i, n)})
        constraints.append({'type': 'ineq', 'fun': make_boundary_y_lower(i, n)})
        constraints.append({'type': 'ineq', 'fun': make_boundary_y_upper(i, n)})
    
    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({'type': 'ineq', 'fun': make_overlap_constraint(i, j, n)})
    
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 1.0)] * n
    
    try:
        result = minimize(
            obj_func,
            x0,
            method='SLSQP',
            args=(n,),
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 2000, 'ftol': 1e-15, 'disp': False}
        )
        return result.x, result.success
    except Exception:
        return x0, False

def project_to_feasible(centers, radii, n):
    """Project solution to satisfy basic bounds."""
    centers = np.clip(centers, 0, 1)
    radii = np.clip(radii, 0, 1)
    
    # Ensure circles are inside square
    for i in range(n):
        r = radii[i]
        centers[i, 0] = np.clip(centers[i, 0], r, 1 - r)
        centers[i, 1] = np.clip(centers[i, 1], r, 1 - r)
    
    return centers, radii


def resolve_overlaps(centers, radii, n, max_iter=100):
    """Resolve overlaps by reducing radii."""
    for iteration in range(max_iter):
        max_violation = 0
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                r_sum = radii[i] + radii[j]
                if r_sum > dist:
                    violation = (r_sum - dist) / 2.0 + 1e-10
                    radii[i] -= violation / 2.0
                    radii[j] -= violation / 2.0
                    radii[i] = max(0, radii[i])
                    radii[j] = max(0, radii[j])
                    max_violation = max(max_violation, violation)
        if max_violation < 1e-10:
            break
    
    # Re-enforce boundary
    centers, radii = project_to_feasible(centers, radii, n)
    return centers, radii
