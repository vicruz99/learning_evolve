import numpy as np
from scipy.optimize import minimize

def objective(v):
    """Maximize sum of radii by minimizing negative sum."""
    return -np.sum(v[2::3])

def constraints_fun(v):
    """Evaluate all inequality constraints: boundary and pairwise non-overlap."""
    n = len(v) // 3
    X = v[0::3]
    Y = v[1::3]
    R = v[2::3]
    
    # Boundary constraints: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    c_bound = np.concatenate([X - R, 1.0 - X - R, Y - R, 1.0 - Y - R])
    
    # Pairwise constraints: dist_sq - (r_i + r_j)^2 >= 0
    iu, ju = np.triu_indices(n, k=1)
    dx = X[iu] - X[ju]
    dy = Y[iu] - Y[ju]
    dr = R[iu] + R[ju]
    c_pair = dx**2 + dy**2 - dr**2
    
    return np.concatenate([c_bound, c_pair])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    bounds = [(0, 1)]*n + [(0, 1)]*n + [(1e-6, 0.5)]*n
    cons = [{'type': 'ineq', 'fun': constraints_fun}]
    
    best_val = -np.inf
    best_v = None
    
    # Run multiple restarts with different initial perturbations
    for seed in range(5):
        # Hexagonal grid initialization
        rows, cols = 5, 6
        x_base = np.linspace(0.12, 0.88, cols)
        y_base = np.linspace(0.12, 0.88, rows)
        pts = []
        for r in range(rows):
            for c in range(cols):
                x = x_base[c] + (r % 2) * (x_base[1] - x_base[0]) / 2
                y = y_base[r]
                pts.append([x, y])
                if len(pts) == n:
                    break
            if len(pts) == n:
                break
        pts = np.array(pts)
        
        # Add controlled randomness
        rng = np.random.default_rng(seed)
        pts = np.clip(pts + rng.uniform(-0.02, 0.02, pts.shape), 0.05, 0.95)
        
        x0 = np.zeros(3 * n)
        x0[0::3] = pts[:, 0]
        x0[1::3] = pts[:, 1]
        x0[2::3] = 0.06 + rng.uniform(0, 0.02, n)
        
        try:
            res = minimize(objective, x0, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 5000, 'ftol': 1e-12})
            if -res.fun > best_val:
                best_val = -res.fun
                best_v = res.x
        except Exception:
            continue
            
    if best_v is not None:
        centers = np.column_stack((best_v[0::3], best_v[1::3]))
        radii = best_v[2::3]
        # Ensure strict validity within numerical tolerances
        radii = np.maximum(radii, 0.0)
        centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
        centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)
        return centers, radii, float(np.sum(radii))
    
    # Fallback initialization
    x0 = np.zeros(3 * n)
    x0[0::3] = np.tile(np.linspace(0.1, 0.9, 5), 6)[:n]
    x0[1::3] = np.repeat(np.linspace(0.1, 0.9, 5), 6)[:n]
    x0[2::3] = 0.05
    centers = np.column_stack((x0[0::3], x0[1::3]))
    radii = x0[2::3]
    return centers, radii, float(np.sum(radii))
