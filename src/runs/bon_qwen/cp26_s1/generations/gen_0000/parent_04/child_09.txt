import numpy as np
from scipy.optimize import minimize

def get_hexagonal_layout(n):
    """
    Generates an initial hexagonal packing layout for n circles in a unit square.
    """
    positions = []
    # Try to fit circles in rows. Hexagonal packing vertical spacing is r * sqrt(3).
    # Assuming r ~ 0.1, spacing ~ 0.173. 1 / 0.173 ~ 5.7 rows.
    # We can fit 6 rows.
    
    # Estimate radius for layout
    r_est = 0.1
    
    row_counts = [5, 5, 5, 5, 4, 2] # Sums to 26
    
    current_n = 0
    for i, count in enumerate(row_counts):
        if current_n >= n:
            break
        y = r_est + i * r_est * np.sqrt(3)
        if y + r_est > 1:
            # If row is too low, compress or stop
            break
            
        # Horizontal shift for odd rows
        x_start = r_est
        if i % 2 == 1:
            x_start += r_est # Shift right by radius (diameter/2)
            
        x = x_start
        for _ in range(count):
            if current_n >= n:
                break
            if x + r_est <= 1:
                positions.append((x, y))
                current_n += 1
                x += 2 * r_est
            else:
                break
    return positions

        bounds.extend([(0, 1), (0, 1), (0, 0.5)]) # r <= 0.5 is safe upper bound
        
    # 2. Objective Function
    def objective(vars):
        # Maximize sum of radii => Minimize negative sum
        radii = vars[2::3]
        return -np.sum(radii)

    # 3. Constraints
    constraints = []
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    for i in range(n):
        idx = 3 * i
        x, y, r = vars[idx], vars[idx+1], vars[idx+2]
        
        # x - r >= 0
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: v[3*i] - v[3*i+2]})
        # 1 - x - r >= 0
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: 1.0 - v[3*i] - v[3*i+2]})
        # y - r >= 0
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: v[3*i+1] - v[3*i+2]})
        # 1 - y - r >= 0
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: 1.0 - v[3*i+1] - v[3*i+2]})
        
    # Pairwise non-overlap constraints: dist >= r_i + r_j
    # dist^2 >= (r_i + r_j)^2
    for i in range(n):
        for j in range(i + 1, n):
            idx_i = 3 * i
            idx_j = 3 * j
            
            # Constraint function: dist_sq - (r_i + r_j)^2 >= 0
            def make_constraint(i, j):
                def constr(v):
                    xi, yi, ri = v[3*i], v[3*i+1], v[3*i+2]
                    xj, yj, rj = v[3*j], v[3*j+1], v[3*j+2]

    return centers, radii, sum_radii
