import numpy as np
from scipy.optimize import minimize

def run_packing():
    """
    Pack 26 circles in a unit square to maximize the sum of radii.
    Returns centers (26, 2), radii (26,), and sum_radii.
    """
    n = 26
    
    # 1. Generate Initial Configuration
    # We use a hexagonal lattice pattern to distribute points densely.
    # This provides a better starting point than a random or square grid.
    
    # Estimate a reasonable radius for initialization. 
    # For n=26, r is roughly around 0.1.
    # We start with a smaller radius to ensure feasibility.
    r_init = 0.06
    
    centers = np.zeros((n, 2))
    radii = np.full(n, r_init)
    
    count = 0
    row = 0
    # Hexagonal packing parameters
    # Vertical spacing between rows
    dy = r_init * np.sqrt(3)
    # Horizontal spacing between columns
    dx = 2 * r_init
    
    while count < n:
        y = r_init + row * dy
        if y + r_init > 1.0:
            # If row is too high, stop (shouldn't happen with our logic, but safety)
            break
        
        # Determine x start for this row
        # Even rows start at r_init, Odd rows shifted by r_init (which is dx/2)
        x_start = r_init
        if row % 2 == 1:
            x_start += r_init # Shift by r_init = dx/2
        
        col = 0
        x = x_start
        while x + r_init <= 1.0 and count < n:
            centers[count, 0] = x
            centers[count, 1] = y
            count += 1
            x += dx
            col += 1
            
        row += 1
        
    # In case we generated fewer than n (unlikely with these params), fill remaining randomly
    while count < n:
        centers[count] = np.random.uniform(0.1, 0.9, 2)
        count += 1
        
    # 2. Define Optimization Problem
    # Variables: [x_1, y_1, r_1, x_2, y_2, r_2, ..., x_n, y_n, r_n]
    # Size: 3 * n
    initial_vars = np.zeros(3 * n)
    for i in range(n):
        initial_vars[3 * i] = centers[i, 0]
        initial_vars[3 * i + 1] = centers[i, 1]
        initial_vars[3 * i + 2] = radii[i]
        
    # Bounds for variables
    # x, y in [0, 1], r in [0, 0.5]
    bounds = []
    for i in range(n):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 0.5)) # r

    # Objective function: Minimize -sum(radii)
    def objective(vars):
        r = vars[2::3] # Extract radii
        return -np.sum(r)

    # Constraints
    # We need to define constraints for:
    # 1. Circle i does not overlap Circle j: dist^2 >= (r_i + r_j)^2
    # 2. Circle i inside boundary: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    
    constraints = []
    
    # Non-overlap constraints (n choose 2)
    # To speed up, we only add constraints for pairs that are likely to be close?
    # Actually, for correctness, we should check all pairs, but SLSQP might be slow with 325 constraints.
    # However, 325 is manageable.
    for i in range(n):
        for j in range(i + 1, n):
            def make_constraint(i, j):
                def constraint(vars):
                    xi, yi = vars[3*i], vars[3*i+1]
                    xj, yj = vars[3*j], vars[3*j+1]
                    ri, rj = vars[3*i+2], vars[3*j+2]
                    dist_sq = (xi - xj)**2 + (yi - yj)**2
                    sum_r_sq = (ri + rj)**2
                    return dist_sq - sum_r_sq
                return {'type': 'ineq', 'fun': constraint}
            
            constraints.append(make_constraint(i, j))

    # Boundary constraints
    for i in range(n):
        # x - r >= 0
        def make_boundary_x_min(i):
            def constraint(vars):
                return vars[3*i] - vars[3*i+2]
            return {'type': 'ineq', 'fun': constraint}
        constraints.append(make_boundary_x_min(i))
        
        # 1 - x - r >= 0
        def make_boundary_x_max(i):
            def constraint(vars):
                return 1.0 - vars[3*i] - vars[3*i+2]
            return {'type': 'ineq', 'fun': constraint}
        constraints.append(make_boundary_x_max(i))
        
        # y - r >= 0
        def make_boundary_y_min(i):
            def constraint(vars):
                return vars[3*i+1] - vars[3*i+2]
            return {'type': 'ineq', 'fun': constraint}
        constraints.append(make_boundary_y_min(i))
        
        # 1 - y - r >= 0
        def make_boundary_y_max(i):
            def constraint(vars):
                return 1.0 - vars[3*i+1] - vars[3*i+2]
            return {'type': 'ineq', 'fun': constraint}
        constraints.append(make_boundary_y_max(i))

    # 3. Run Optimization
    # Using SLSQP method
    result = minimize(objective, initial_vars, method='SLSQP', bounds=bounds, 
                      constraints=constraints, options={'maxiter': 500, 'ftol': 1e-9, 'disp': False})
    
    # 4. Extract Result
    final_vars = result.x
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    for i in range(n):
        centers[i, 0] = final_vars[3 * i]
        centers[i, 1] = final_vars[3 * i + 1]
        radii[i] = final_vars[3 * i + 2]
        
    # Ensure non-negative radii (numerical safety)
    radii = np.maximum(radii, 0.0)
    
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
