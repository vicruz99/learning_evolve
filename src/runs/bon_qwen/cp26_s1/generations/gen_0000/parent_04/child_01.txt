import numpy as np
from scipy.optimize import minimize

# Global constants
N_CIRCLES = 26
SQ_DIM = 1.0

def objective(x, n):
    """
    Objective function to minimize.
    We want to maximize sum of radii, so we minimize negative sum.
    x structure: [x1, y1, r1, x2, y2, r2, ..., xn, yn, rn]
    """
    radii = x[2::3]
    return -np.sum(radii)

def constraints_fun(x, n):
    """
    Returns a flattened array of constraint values.
    SLSQP requires inequality constraints to be >= 0.
    Constraints included:
    1. x_i - r_i >= 0
    2. 1 - x_i - r_i >= 0
    3. y_i - r_i >= 0
    4. 1 - y_i - r_i >= 0
    5. dist_sq(i,j) - (r_i + r_j)^2 >= 0 for i < j
    """
    # Extract centers and radii
    # x is flattened: x0, y0, r0, x1, y1, r1, ...
    centers = x[:2*n].reshape(n, 2)
    radii = x[2*n:]
    
    constraints = []
    
    # Boundary constraints
    # x >= r  => x - r >= 0
    x_coords = centers[:, 0]
    y_coords = centers[:, 1]
    
    # Left boundary: x >= r
    constraints.append(x_coords - radii)
    # Right boundary: x + r <= 1 => 1 - x - r >= 0
    constraints.append(SQ_DIM - x_coords - radii)
    # Bottom boundary: y >= r
    constraints.append(y_coords - radii)
    # Top boundary: y + r <= 1 => 1 - y - r >= 0
    constraints.append(SQ_DIM - y_coords - radii)
    
    # Overlap constraints
    # dist^2 >= (r_i + r_j)^2
    # dist^2 - (r_i + r_j)^2 >= 0
    
    # Compute pairwise squared distances efficiently
    # centers shape: (n, 2)
    # diff shape: (n, n, 2)
    # Using broadcasting: centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    # This computes c_i - c_j for all i, j
    
    # Optimization: only need upper triangle, but computing full matrix is fast for n=26
    # diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :] 
    # This creates a 26x26x2 array. Memory is negligible.
    
    # However, to save a bit of memory/time, we can compute diffs directly
    # Or just use the full matrix approach as it's cleaner.
    
    # Compute squared Euclidean distance matrix
    # dist_sq[i, j] = (xi-xj)^2 + (yi-yj)^2
    # Using numpy broadcasting
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist_sq = np.sum(diff**2, axis=2)
    
    # Compute (r_i + r_j)^2
    # rad_sum[i, j] = r_i + r_j
    rad_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    rad_sum_sq = rad_sum**2
    
    # Constraint value matrix
    # val[i, j] = dist_sq[i, j] - rad_sum_sq[i, j]
    # We need val[i, j] >= 0 for all i != j
    # For i=j, dist=0, rad_sum=2r, val = -4r^2 <= 0. This is expected (self-overlap constraint is trivial/invalid).
    # We only enforce for i < j.
    
    overlap_constraints = dist_sq - rad_sum_sq
    
    # Extract upper triangle indices (i < j)
    triu_idx = np.triu_indices(n, k=1)
    constraints.append(overlap_constraints[triu_idx])
    
    # Concatenate all constraints into a single array
    return np.concatenate(constraints)

def get_bounds(n):
    """
    Returns box constraints for variables.
    x, y in [0, 1]
    r in [0, 0.5] (max possible radius in unit square is 0.5)
    """
    bounds = []
    for _ in range(n):
        bounds.append((0.0, 1.0))      # x
        bounds.append((0.0, 1.0))      # y
        bounds.append((0.0, 0.5))      # r
    return bounds

def initialize_grid(n):
    """
    Initializes centers and radii in a dense grid pattern.
    """
    # We need n points. 
    # A 6x5 grid has 30 points. We take first n.
    # We want points somewhat distributed in [0,1]x[0,1]
    # To allow room for expansion, we start them slightly inset or evenly spaced.
    
    # Let's create a grid that covers the space well.
    # 6 columns, 5 rows.
    # x range: 0.1 to 0.9 (leaving 0.1 margin) -> 6 points
    # y range: 0.1 to 0.9 (leaving 0.1 margin) -> 5 points
    
    # Actually, linspace(0.1, 0.9, 6) gives points at 0.1, 0.26, ... 0.9
    # Spacing is 0.18.
    # If we start with radius 0.04 (diameter 0.08), they are well separated.
    
    cols = 6
    rows = 5
    
    # Generate grid points
    # Using linspace to distribute evenly
    # To be safer near boundaries, maybe start a bit in? 
    # But optimizer can move them.
    # Let's use a spacing that fits n roughly.
    # sqrt(26) approx 5.1. So 5x5 or 6x5.
    
    x_coords = np.linspace(0.15, 0.85, cols)
    y_coords = np.linspace(0.15, 0.85, rows)
    
    points = []
    for y in y_coords:
        for x in x_coords:
            points.append([x, y])
            if len(points) >= n:
                break
        if len(points) >= n:
            break
            
    centers = np.array(points[:n])
    radii = np.full(n, 0.04) # Initial small radius
    
    return centers, radii

def run_packing():
    n = N_CIRCLES
    
    # 1. Initialization
    centers, radii = initialize_grid(n)
    
    # Flatten initial guess: [x1, y1, r1, ...]
    x0 = np.concatenate([centers.flatten(), radii])
    
    # 2. Setup Optimization
    # Bounds
    bnds = get_bounds(n)
    
    # Constraints
    # SLSQP expects a list of constraint dicts.
    # We combine all inequality constraints into one function returning a vector.
    cons = {'type': 'ineq', 'fun': lambda x: constraints_fun(x, n)}
    
    # 3. Run Optimizer
    # SLSQP is a good choice for constrained optimization with gradients (approximated)
    # maxiter limit to prevent infinite loops
    # ftol for convergence tolerance
    res = minimize(
        objective, 
        x0, 
        args=(n,), 
        method='SLSQP', 
        bounds=bnds, 
        constraints=cons, 
        options={'maxiter': 2000, 'ftol': 1e-10, 'disp': False}
    )
    
    # 4. Extract Results
    if res.success:
        x_opt = res.x
    else:
        # If optimization didn't report success, we still use the result 
        # provided it's better or valid. 
        # Sometimes success is False but result is good.
        x_opt = res.x
        
    centers = x_opt[:2*n].reshape(n, 2)
    radii = x_opt[2*n:]
    
    # Ensure radii are non-negative (solver bounds should handle this, but clip for safety)
    radii = np.maximum(radii, 0.0)
    
    sum_radii = np.sum(radii)
    
    # Basic validation/repair if needed (though solver should satisfy constraints)
    # The validation function is strict, so we return what we have.
    # If the solver failed to satisfy constraints strictly due to numerics,
    # the validation function allows 1e-12 error.
    
    return centers, radii, sum_radii
