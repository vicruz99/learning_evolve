import numpy as np
from scipy.optimize import minimize


def obj(params, n):
    return -np.sum(params[2*n:])


def bc_xmin(params, i, n):
    return params[2*i] - params[2*n+i]


def bc_xmax(params, i, n):
    return 1.0 - params[2*i] - params[2*n+i]


def bc_ymin(params, i, n):
    return params[2*i+1] - params[2*n+i]


def bc_ymax(params, i, n):
    return 1.0 - params[2*i+1] - params[2*n+i]


def bc_rmin(params, i, n):
    return params[2*n+i]


def noc(params, i, j, n):
    ci = params[2*i:2*i+2]
    cj = params[2*j:2*j+2]
    ri = params[2*n+i]
    rj = params[2*n+j]
    d2 = np.sum((ci - cj)**2)
    if d2 < 1e-20:
        return -ri - rj + 1e-10
    return np.sqrt(d2) - ri - rj


def gen_initial(n, rows):
    centers = np.zeros((n, 2))
    idx = 0
    nr = len(rows)
    for ri, cnt in enumerate(rows):
        y = (ri + 0.5) / nr
        for ci in range(cnt):
            x = (ci + 0.5) / cnt
            if ri % 2 == 1:
                x += 0.5 / cnt
            if idx < n:
                centers[idx] = [x, y]
                idx += 1
    return centers


def expand_circles(centers, radii, n, iters=1500, step=0.0003):
    """Iteratively expand circles and resolve overlaps."""
    for _ in range(iters):
        # Phase 1: Try to expand each circle
        for i in range(n):
            # Max radius from boundary constraints
            mr = min(centers[i, 0], 1.0 - centers[i, 0],
                      centers[i, 1], 1.0 - centers[i, 1])
            # Max radius from non-overlap constraints
            for j in range(n):
                if i != j:
                    d = np.sqrt(np.sum((centers[i] - centers[j])**2))
                    candidate = d - radii[j]
                    if candidate < mr:
                        mr = candidate
            mr = max(0.0, mr)
            radii[i] = min(radii[i] + step, mr)

        # Phase 2: Push apart overlapping circles
        for i in range(n):
            for j in range(i + 1, n):
                d2 = np.sum((centers[i] - centers[j])**2)
                d = np.sqrt(d2)
                md = radii[i] + radii[j]
                if d < md and d > 1e-14:
                    ov = (md - d) * 0.51
                    dir_vec = (centers[i] - centers[j]) / d
                    centers[i] += dir_vec * ov
                    centers[j] -= dir_vec * ov

        # Phase 3: Clamp centers to valid region
        for i in range(n):
            centers[i, 0] = max(radii[i], min(1.0 - radii[i], centers[i, 0]))
            centers[i, 1] = max(radii[i], min(1.0 - radii[i], centers[i, 1]))

    return centers, radii


def run_scipy_opt(centers, radii, n, max_iter=3000):
    """Run SLSQP optimization for local refinement."""
    params = np.concatenate([centers.flatten(), radii])

    cons = []
    for i in range(n):
        cons.append({'type': 'ineq', 'fun': bc_xmin, 'args': (i, n)})
        cons.append({'type': 'ineq', 'fun': bc_xmax, 'args': (i, n)})
        cons.append({'type': 'ineq', 'fun': bc_ymin, 'args': (i, n)})
        cons.append({'type': 'ineq', 'fun': bc_ymax, 'args': (i, n)})
        cons.append({'type': 'ineq', 'fun': bc_rmin, 'args': (i, n)})
    for i in range(n):
        for j in range(i + 1, n):
            cons.append({'type': 'ineq', 'fun': noc, 'args': (i, j, n)})

    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n

    try:
        result = minimize(obj, params, args=(n,), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': max_iter, 'ftol': 1e-15, 'disp': False})
        if result.success and np.isfinite(result.fun):
            c = result.x[:2*n].reshape(n, 2)
            r = result.x[2*n:]
            # Clamp radii to be non-negative
            r = np.maximum(r, 0.0)
            # Re-clamp centers
            for i in range(n):
                c[i, 0] = max(r[i], min(1.0 - r[i], c[i, 0]))
                c[i, 1] = max(r[i], min(1.0 - r[i], c[i, 1]))
            return c, r, np.sum(r)
    except Exception:
        pass
    return None, None, 0.0
