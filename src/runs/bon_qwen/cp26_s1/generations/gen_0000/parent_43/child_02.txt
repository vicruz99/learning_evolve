import numpy as np
from scipy.optimize import minimize
import math

def compute_penalty_and_objective(centers, radii, n):
    """Compute objective (negative sum of radii) and constraint penalties."""
    obj = -np.sum(radii)
    penalty = 0.0
    penalty_coeff = 10000.0
    
    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        # Left boundary
        if x - r < 0:
            penalty += penalty_coeff * (x - r) ** 2
        # Right boundary
        if x + r > 1:
            penalty += penalty_coeff * (x + r - 1) ** 2
        # Bottom boundary
        if y - r < 0:
            penalty += penalty_coeff * (y - r) ** 2
        # Top boundary
        if y + r > 1:
            penalty += penalty_coeff * (y + r - 1) ** 2
        
        # Radius must be positive
        if r < 0:
            penalty += penalty_coeff * r ** 2
    
    # Non-overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = math.sqrt(dx * dx + dy * dy)
            sum_r = radii[i] + radii[j]
            if dist < sum_r:
                penalty += penalty_coeff * (sum_r - dist) ** 2
    
    return obj + penalty


def objective_wrapper(x, n):
    """Wrapper for scipy optimizer."""
    centers = x[:2 * n].reshape(n, 2)
    radii = x[2 * n:]
    return compute_penalty_and_objective(centers, radii, n)


def generate_hexagonal_initial(n, density=0.08):
    """Generate hexagonal lattice initial configuration."""
    centers = np.zeros((n, 2))
    radii = np.full(n, density)
    
    # Hexagonal spacing
    h = density * 2 * np.sqrt(3) / 2
    w = density * 2
    
    idx = 0
    row = 0
    y = density + 0.02
    while idx < n:
        x_offset = (row % 2) * w / 2
        x = density + 0.02 + x_offset
        col = 0
        while idx < n and x < 1 - density:
            centers[idx, 0] = x
            centers[idx, 1] = y
            x += w
            idx += 1
            col += 1
        y += h
        row += 1
    
    # Center the configuration
    min_x = centers[:, 0].min()
    min_y = centers[:, 1].min()
    max_x = centers[:, 0].max()
    max_y = centers[:, 1].max()
    
    range_x = max_x - min_x
    range_y = max_y - min_y
    
    if range_x > 0:
        centers[:, 0] = (centers[:, 0] - min_x) / range_x * 0.85 + 0.075
    if range_y > 0:
        centers[:, 1] = (centers[:, 1] - min_y) / range_y * 0.85 + 0.075
    
    return centers, radii


def generate_grid_initial(n):
    """Generate grid-based initial configuration."""
    # Find approximately sqrt(n) rows and columns
    cols = int(math.ceil(math.sqrt(n)))
    rows = int(math.ceil(n / cols))
    
    centers = np.zeros((n, 2))
    idx = 0
    for r in range(rows):
        for c in range(cols):
            if idx >= n:
                break
            centers[idx, 0] = (c + 0.5) / cols
            centers[idx, 1] = (r + 0.5) / rows
            idx += 1
    
    # Add small random perturbation
    centers += np.random.uniform(-0.02, 0.02, centers.shape)
    centers = np.clip(centers, 0.05, 0.95)
    
    radii = np.full(n, 0.04)
    return centers, radii


def generate_random_initial(n):
    """Generate random initial configuration."""
    centers = np.random.uniform(0.1, 0.9, (n, 2))
    radii = np.full(n, 0.03)
    return centers, radii


def run_single_optimization(centers_init, radii_init, n, seed=None):
    """Run a single optimization from given initial conditions."""
    if seed is not None:
        np.random.seed(seed)
    
    x0 = np.concatenate([centers_init.flatten(), radii_init])
    
    # Use Nelder-Mead for robustness
    result = minimize(
        objective_wrapper,
        x0,
        args=(n,),
        method='Nelder-Mead',
        options={
            'maxiter': 50000,
            'xatol': 1e-10,
            'fatol': 1e-10,
            'adaptive': True
        }
    )
    
    centers_opt = result.x[:2 * n].reshape(n, 2)
    radii_opt = result.x[2 * n:]
    
    # Ensure radii are non-negative
    radii_opt = np.maximum(radii_opt, 1e-8)
    
    # Project centers to valid range
    for i in range(n):
        centers_opt[i, 0] = np.clip(centers_opt[i, 0], radii_opt[i], 1 - radii_opt[i])
        centers_opt[i, 1] = np.clip(centers_opt[i, 1], radii_opt[i], 1 - radii_opt[i])
    
    return centers_opt, radii_opt, -result.fun
