The problem asks to pack 26 circles in a unit square to maximize the sum of radii. The target sum is approximately 2.636, which implies an average radius of about 0.101. Since a simple 5x5 grid of equal circles has radius 0.1 (sum 2.5), a denser packing like a hexagonal lattice is required to exceed the target.

The strategy involves two main steps:
1.  **Radius Calculation (LP Solver):** For a fixed set of centers, finding the optimal radii that maximize the sum is a Linear Programming (LP) problem. The objective is to maximize $\sum r_i$ subject to $r_i + r_j \le \text{dist}(i, j)$ for all pairs and boundary constraints. This ensures that for any given configuration, we extract the maximum possible sum of radii.
2.  **Center Optimization (Simulated Annealing):** We will initialize the centers using a hexagonal lattice pattern, which provides a high-density starting point. Then, we will use a simulated annealing approach to perturb the centers. In each step, we calculate the new maximum sum of radii. If it improves (or with some probability if it degrades), we accept the new configuration. This allows escaping local optima and finding a globally better packing.

import numpy as np
from scipy.optimize import linprog
import math

def get_boundaries(centers):
    """
    Computes the upper bounds for radii based on the unit square boundaries.
    For a center (x, y), the max radius is min(x, 1-x, y, 1-y).
    """
    n = centers.shape[0]
    bounds = np.zeros(n)
    for i in range(n):
        x, y = centers[i]
        bounds[i] = min(x, 1 - x, y, 1 - y)
    return bounds

def solve_radii_lp(centers):
    """
    Solves the Linear Programming problem to find radii that maximize sum(r)
    given fixed centers.
    
    Maximize: sum(r_i)
    Subject to:
      r_i >= 0
      r_i <= boundary_i
      r_i + r_j <= distance(i, j)
    """
    n = centers.shape[0]
    
    # Objective: maximize sum(r_i) => minimize -sum(r_i)
    c = np.ones(n) * -1
    
    # Inequality constraints A_ub @ x <= b_ub
    # 1. Boundary constraints: r_i <= b_i  => r_i - b_i <= 0 (handled by bounds)
    #    Actually linprog handles bounds directly, but pairwise constraints need matrix.
    # 2. Pairwise constraints: r_i + r_j <= d_ij
    
    constraints = []
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            constraints.append((row, dist))
    
    A_ub = np.array([c_row for c_row, _ in constraints])
    b_ub = np.array([b_val for _, b_val in constraints])
    
    # Bounds for variables r_i: [0, boundary_i]
    bounds_list = [(0, b) for b in get_boundaries(centers)]
    
    # Solve LP
    # Using high bound for boundaries that might be negative if center is outside (though shouldn't happen)
    # Ensure bounds are non-negative
    bounds_list = [(0, max(0, b)) for b in get_boundaries(centers)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds_list, method='highs')
        if res.success:
            return res.x
        else:
            # Fallback if LP fails (e.g. infeasible due to numerical issues)
            # Return a valid but suboptimal solution
            radii = get_boundaries(centers)
            for i in range(n):
                for j in range(i + 1, n):
                    dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                    if radii[i] + radii[j] > dist:
                        # Shrink proportionally or just half
                        excess = radii[i] + radii[j] - dist
                        radii[i] -= excess / 2
                        radii[j] -= excess / 2
            return np.maximum(radii, 0)
    except Exception:
        # Fallback
        return get_boundaries(centers) * 0.5

def initial_hex_packing(n):
    """
    Generates an initial set of centers in a hexagonal pattern.
    """
    centers = []
    # Estimate radius for hex packing to fit n circles
    # Area ~ n * pi * r^2 / density. Density ~ 0.9.
    # r ~ sqrt(1 / (n * pi / 0.9))
    r_est = math.sqrt(1 / (n * math.pi / 0.9069))
    
    row_height = r_est * math.sqrt(3)
    cols = int(1 / (2 * r_est)) + 2
    
    y = r_est
    while y + r_est <= 1:
        x = r_est
        shift = r_est if (int((y - r_est) / row_height) % 2 == 1) else 0
        x += shift
        while x + r_est <= 1:
            centers.append([x, y])
            if len(centers) >= n:
                break
            x += 2 * r_est
        if len(centers) >= n:
            break
        y += row_height
        
    # If we didn't get enough, fill randomly or adjust
    if len(centers) < n:
        # Simple grid fill
        grid_side = int(math.ceil(math.sqrt(n)))
        step = 1.0 / (grid_side + 1)
        count = 0
        for r in range(1, grid_side + 1):
            for c in range(1, grid_side + 1):
                if count < n:
                    centers.append([c * step, r * step])
                    count += 1
        
    centers = np.array(centers[:n])
    return centers

def run_packing():
    """
    Optimizes the packing of 26 circles in a unit square to maximize sum of radii.
    """
    n = 26
    
    # Initialize centers
    centers = initial_hex_packing(n)
    
    # Calculate initial radii
    radii = solve_radii_lp(centers)
    best_sum = np.sum(radii)
    best_centers = centers.copy()
    best_radii = radii.copy()
    
    # Simulated Annealing parameters
    temperature = 0.05
    cooling_rate = 0.995
    min_temperature = 1e-6
    max_iterations = 5000 # Reasonable limit
    
    iteration = 0
    
    while temperature > min_temperature and iteration < max_iterations:
        iteration += 1
        
        # Perturb centers
        # Randomly select a circle and move it slightly
        idx = np.random.randint(0, n)
        delta_x = np.random.uniform(-temperature, temperature)
        delta_y = np.random.uniform(-temperature, temperature)
        
        new_centers = centers.copy()
        new_centers[idx, 0] += delta_x
        new_centers[idx, 1] += delta_y
        
        # Check boundary validity for center (radius will be 0 if outside, handled by LP bounds)
        # But let's keep centers strictly inside for better performance
        if new_centers[idx, 0] < 0 or new_centers[idx, 0] > 1 or \
           new_centers[idx, 1] < 0 or new_centers[idx, 1] > 1:
            continue
            
        # Solve for radii
        new_radii = solve_radii_lp(new_centers)
        new_sum = np.sum(new_radii)
        
        # Acceptance criterion
        delta_sum = new_sum - best_sum
        if delta_sum > 0:
            centers = new_centers
            radii = new_radii
            best_sum = new_sum
            best_centers = centers.copy()
            best_radii = radii.copy()
        else:
            # Accept with probability exp(delta/T)
            if np.random.random() < math.exp(delta_sum / temperature):
                centers = new_centers
                radii = new_radii
        
        temperature *= cooling_rate

    return best_centers, best_radii, best_sum

# Execute to generate the solution
centers, radii, sum_radii = run_packing()
