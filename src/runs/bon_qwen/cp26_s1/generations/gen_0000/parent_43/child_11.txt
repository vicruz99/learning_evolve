import numpy as np
from scipy.optimize import minimize

def objective(v):
    """Minimize negative sum of radii (equivalent to maximizing sum of radii)"""
    return -np.sum(v[2::3])

def constraints_func(v):
    """
    Compute all inequality constraints:
    - 4 boundary constraints per circle (x>=r, 1-x>=r, y>=r, 1-y>=r)
    - 1 distance constraint per pair (squared distance >= squared sum of radii)
    Returns array where each element must be >= 0
    """
    n = 26
    n_constr = 4*n + n*(n-1)//2
    c = np.empty(n_constr)
    k = 0
    
    # Boundary constraints
    for i in range(n):
        xi, yi, ri = 3*i, 3*i+1, 3*i+2
        c[k] = v[xi] - v[ri]
        c[k+1] = 1.0 - v[xi] - v[ri]
        c[k+2] = v[yi] - v[ri]
        c[k+3] = 1.0 - v[yi] - v[ri]
        k += 4
        
    # Distance constraints
    for i in range(n):
        xi, yi, ri = 3*i, 3*i+1, 3*i+2
        for j in range(i+1, n):
            xj, yj, rj = 3*j, 3*j+1, 3*j+2
            dx = v[xi] - v[xj]
            dy = v[yi] - v[yj]
            dr = v[ri] + v[rj]
            c[k] = dx*dx + dy*dy - dr*dr
            k += 1
    return c

def run_packing():
    n = 26
    
    # Initial configuration: structured grid with small feasible radius
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        centers[i] = [(i % 6 + 0.5) / 6.0, (i // 6 + 0.5) / 5.0]
        radii[i] = 0.04
        
    x0 = np.concatenate([centers.ravel(), radii])
    bounds = [(0, 1)] * n + [(0, 1)] * n + [(0, 0.5)] * n
    cons = {'type': 'ineq', 'fun': constraints_func}
    
    # Run constrained optimization
    res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, 
                   options={'maxiter': 5000, 'ftol': 1e-10, 'disp': False})
                   
    centers_opt = res.x[:2*n].reshape(n, 2)
    radii_opt = res.x[2*n:]
    
    # Post-processing: enforce strict boundary constraints
    for i in range(n):
        x, y = centers_opt[i]
        mr = min(x, 1-x, y, 1-y)
        radii_opt[i] = min(radii_opt[i], max(mr, 0.0))
        
    # Post-processing: resolve any tiny numerical overlaps
    for _ in range(5):
        overlap_found = False
        for i in range(n):
            for j in range(i+1, n):
                d = np.sqrt((centers_opt[i,0]-centers_opt[j,0])**2 + **(centers_opt[i,1]-centers_opt[j,1])2)
                s = radii_opt[i] + radii_opt[j]
                if d < s:
                    f = d / s
                    radii_opt[i] *= f
                    radii_opt[j] *= f
                    overlap_found = True
        if not overlap_found:
            break
            
    # Final boundary check after overlap resolution
    for i in range(n):
        x, y = centers_opt[i]
        mr = min(x, 1-x, y, 1-y)
        radii_opt[i] = min(radii_opt[i], max(mr, 0.0))
        
    return centers_opt, radii_opt, float(np.sum(radii_opt))
