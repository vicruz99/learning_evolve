import numpy as np
from scipy.optimize import linprog, minimize


def compute_optimal_radii(centers):
    """
    Compute optimal radii given fixed centers using LP.
    
    Maximizes sum(r_i) subject to:
    - r_i <= distance to each boundary
    - r_i + r_j <= distance between centers i and j
    - r_i >= 0
    """
    n = centers.shape[0]
    c = -np.ones(n)  # Maximize sum of radii = minimize negative sum
    
    A_ub_rows = []
    b_ub_rows = []
    
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        x, y = centers[i]
        for bound in [x, 1 - x, y, 1 - y]:
            row = np.zeros(n)
            row[i] = 1
            A_ub_rows.append(row)
            b_ub_rows.append(max(0, bound))
    
    # Non-overlap constraints: r_i + r_j <= ||c_i - c_j||
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            row = np.zeros(n)
            row[i] = 1
            row[j] = 1
            A_ub_rows.append(row)
            b_ub_rows.append(max(0, dist))
    
    A_ub = np.array(A_ub_rows)
    b_ub = np.array(b_ub_rows)
    bounds = [(0, None) for _ in range(n)]
    
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            return np.maximum(result.x, 0)
    except Exception:
        pass
    
    # Fallback: simple greedy approach
    radii = np.zeros(n)
    for i in range(n):
        r = min(centers[i, 0], 1 - centers[i, 0],
                centers[i, 1], 1 - centers[i, 1])
        for j in range(n):
            if i != j:
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                r = min(r, dist / 2)
        radii[i] = r
    return radii


def objective(centers_flat):
    """Objective function for optimizing centers: negative sum of optimal radii."""
    n = 26
    centers = centers_flat.reshape(n, 2)
    radii = compute_optimal_radii(centers)
    return -np.sum(radii)


def make_5x5_plus_center(n=26):
    """5x5 grid with extra circle in center."""
    centers = []
    for i in range(5):
        for j in range(5):
            x = (i + 0.5) / 5.0
            y = (j + 0.5) / 5.0
            centers.append([x, y])
    centers.append([0.5, 0.5])
    return np.array(centers[:n])


def make_hexagonal_grid(n=26):
    """Hexagonal packing arrangement."""
    centers = []
    spacing = 0.22
    for row in range(7):
        y = row * spacing * np.sqrt(3) / 2
        x_offset = spacing / 2 if row % 2 == 1 else 0
        # Calculate how many fit in [0, 1]
        num = int((1 - 2 * x_offset) / spacing) + 1
        for col in range(num):
            x = x_offset + col * spacing
            if x <= 1:
                centers.append([x, y])
        if len(centers) >= n:
            break
    
    result = np.array(centers[:n])
    # Normalize to fit in [0.05, 0.95]
    mn = result.min(axis=0)
    mx = result.max(axis=0)
    result = (result - mn) / (mx - mn + 1e-10)
    result = result * 0.9 + 0.05
    return result


def make_6x5_grid(n=26):
    """6 columns x 5 rows grid."""
    centers = []
    for i in range(6):
        for j in range(5):
            x = (i + 0.5) / 6.0
            y = (j + 0.5) / 5.0
            centers.append([x, y])
    return np.array(centers[:n])


def make_staggered_grid(n=26):
    """Staggered/brick-like arrangement."""
    centers = []
    cols = 6
    rows = 5
    for j in range(rows):
        y = (j + 0.5) / rows
        num_in_row = cols if j % 2 == 0 else cols - 1
        x_start = 0.5 / cols if j % 2 == 1 else 0
        for i in range(num_in_row):
            x = (x_start + i + 0.5) / cols
            centers.append([x, y])
        if len(centers) >= n:
            break
    return np.array(centers[:n])


def make_random_config(n=26, seed=42):
    """Random configuration."""
    rng = np.random.RandomState(seed)
    centers = rng.uniform(0.08, 0.92, (n, 2))
    return centers


def make_corner_heavy(n=26):
    """Place more weight near corners."""
    centers = []
    # 4 corners
    corner_offset = 0.15
    for x in [corner_offset, 1 - corner_offset]:
        for y in [corner_offset, 1 - corner_offset]:
            centers.append([x, y])
    
    # Edge midpoints
    for x in [0.35, 0.65]:
        for y in [corner_offset, 1 - corner_offset]:
            centers.append([x, y])
    for x in [corner_offset, 1 - corner_offset]:
        for y in [0.35, 0.65]:
            centers.append([x, y])
    
    # Interior points
    interior = []
    for i in range(4):
        for j in range(4):
            x = 0.25 + i * 0.2
            y = 0.25 + j * 0.2
            interior.append([x, y])
    
    centers.extend(interior)
    return np.array(centers[:n])


def make_dense_centered(n=26):
    """Dense packing with more circles near center."""
    centers = []
    # Start with a base grid
    for i in range(5):
        for j in range(5):
            x = (i + 0.5) / 5.0
            y = (j + 0.5) / 5.0
            centers.append([x, y])
    
    # Add extra circles in gaps
    extra_candidates = [
        [0.125, 0.125], [0.875, 0.125], [0.125, 0.875], [0.875, 0.875],
        [0.5, 0.125], [0.5, 0.875], [0.125, 0.5], [0.875, 0.5],
        [0.5, 0.5],
        [0.3, 0.3], [0.7, 0.3], [0.3, 0.7], [0.7, 0.7],
    ]
    for ec in extra_candidates:
        if len(centers) >= n:
            break
        # Check if this position is far enough from existing centers
        too_close = False
        for c in centers:
            dist = np.sqrt(np.sum((np.array(ec) - np.array(c)) ** 2))
            if dist < 0.08:
                too_close = True
                break
        if not too_close:
            centers.append(ec)
    
    return np.array(centers[:n])


def perturb_config(centers, scale=0.02, seed=0):
    """Add small random perturbation to configuration."""
    rng = np.random.RandomState(seed)
    perturbed = centers + rng.normal(0, scale, centers.shape)
    perturbed = np.clip(perturbed, 0.02, 0.98)
    return perturbed


def optimize_from_config(config, method='Nelder-Mead'):
    """Run optimization from a given configuration."""
    n = 26
    try:
        if method == 'Nelder-Mead':
            result = minimize(
                objective, config.flatten(),
                method='Nelder-Mead',
                options={'maxiter': 100000, 'xatol': 1e-10, 'fatol': 1e-10, 'adaptive': True}
            )
        elif method == 'Powell':
            result = minimize(
                objective, config.flatten(),
                method='Powell',
                options={'maxiter': 50000, 'xtol': 1e-10, 'ftol': 1e-10}
            )
        else:
            result = minimize(
                objective, config.flatten(),
                method='Nelder-Mead',
                options={'maxiter': 100000, 'xatol': 1e-10, 'fatol': 1e-10}
            )
        
        centers = result.x.reshape(n, 2)
        radii = compute_optimal_radii(centers)
        return centers, radii, np.sum(radii)
    except Exception as e:
        # Fallback: just compute radii for the config
        radii = compute_optimal_radii(config)
        return config, radii, np.sum(radii)


def run_packing():
    n = 26
    
    best_centers = None

    best_sum = -np.inf

    base_configs = [
