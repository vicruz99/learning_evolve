

Looking at this circle packing problem, I need to maximize the sum of radii for 26 circles in a unit square.

<strategy>
**Strategy:**

1. **Hexagonal Initialization**: Start with circles arranged in a hexagonal lattice pattern (5 rows with distribution 6-5-5-5-5), which is the densest packing for equal circles.

2. **Multi-stage Penalty Optimization**: Use scipy's L-BFGS-B optimizer with penalty terms for constraint violations. The penalty weight increases across stages (100 → 1000 → 5000 → 10000) to first explore the landscape, then enforce constraints tightly.

3. **Objective Function**: Minimize `-sum(radii) + penalty`. Penalties handle:
   - Boundary violations (circle extending outside [0,1]²)
   - Overlap violations (distance between centers < sum of radii)
   - Non-negative radii

4. **Multiple Restarts**: Run 5 trials with slight random perturbations to escape local optima.

5. **Post-processing**: Clip centers to ensure boundary validity, ensure positive radii.

6. **Key Insight**: Unequal radii allow ~4% improvement over equal-circle packing (known optimum r≈0.0973 for n=26, sum≈2.53). By allowing some circles to be larger while others fill gaps with smaller radii, we can exceed the target of 2.636.
</strategy>

import numpy as np
from scipy.optimize import minimize


def compute_penalty(params, n, pw):
    """Compute smooth penalty for constraint violations."""
    centers = params[:2 * n].reshape(n, 2)
    radii = params[2 * n:]
    
    penalty = 0.0
    
    # Boundary violations
    for i in range(n):
        x = centers[i, 0]
        y = centers[i, 1]
        r = radii[i]
        
        # x - r >= 0
        val = r - x
        if val > 0:
            penalty += pw * val * val
        
        # y - r >= 0
        val = r - y
        if val > 0:
            penalty += pw * val * val
        
        # x + r <= 1
        val = x + r - 1.0
        if val > 0:
            penalty += pw * val * val
        
        # y + r <= 1
        val = y + r - 1.0
        if val > 0:
            penalty += pw * val * val
        
        # r >= 0
        if r < 0:
            penalty += pw * r * r
    
    # Overlap violations
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            min_dist = radii[i] + radii[j]
            overlap = min_dist - dist
            if overlap > 0:
                penalty += pw * overlap * overlap
    
    return penalty


def objective(params, n, pw):
    """Objective: minimize negative sum of radii + penalty."""
    radii = params[2 * n:]
    return -np.sum(radii) + compute_penalty(params, n, pw)


def hexagonal_init(n):
    """Initialize circles in hexagonal lattice pattern."""
    # 5 rows with 6-5-5-5-5 distribution for 26 circles
    row_counts = [6, 5, 5, 5, 5]
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.085
    
    idx = 0
    num_rows = len(row_counts)
    y_spacing = 0.82 / num_rows
    base_y = 0.09
    
    for row_i, count in enumerate(row_counts):
        y = base_y + row_i * y_spacing
        # Offset odd rows for hexagonal pattern
        x_offset = 0.045 if row_i % 2 == 1 else 0.0
        x_start = 0.06 + x_offset
        
        if count > 1:
            x_spacing = (0.88 - 2 * x_offset) / (count - 1)
        else:
            x_spacing = 0.5
        
        for col in range(count):
            x = x_start + col * x_spacing
            centers[idx, 0] = x
            centers[idx, 1] = y
            idx += 1
    
    return centers, radii


def ensure_validity(centers, radii, n):
    """Post-process to ensure all constraints are satisfied."""
    radii = np.maximum(radii, 1e-9)
    
    for i in range(n):
        r = radii[i]
        centers[i, 0] = np.clip(centers[i, 0], r, 1.0 - r)
        centers[i, 1] = np.clip(centers[i, 1], r, 1.0 - r)
    
    # Iteratively reduce radii to resolve any remaining overlaps
    for iteration in range(50):
        max_violation = 0
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                required = radii[i] + radii[j]
                if dist < required:
                    violation = required - dist
                    max_violation = max(max_violation, violation)
                    # Reduce both radii slightly
                    reduction = violation * 0.5
                    radii[i] -= reduction
                    radii[j] -= reduction
        
        if max_violation < 1e-10:
            break
        
        radii = np.maximum(radii, 1e-9)
        # Re-clip centers
        for i in range(n):
            r = radii[i]
            centers[i, 0] = np.clip(centers[i, 0], r, 1.0 - r)
            centers[i, 1] = np.clip(centers[i, 1], r, 1.0 - r)
    
    return centers, radii


def run_packing():
    """Pack 26 circles in unit square to maximize sum of radii."""
    n = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    for trial in range(6):
        # Initialize
        if trial == 0:
            centers, radii = hexagonal_init(n)
        else:
            np.random.seed(42 + trial)
            centers, radii = hexagonal_init(n)
            # Add controlled random perturbation
            centers += np.random.randn(n, 2) * 0.015
            radii += np.random.randn(n) * 0.008
        
        # Ensure valid starting point
        centers = np.clip(centers, 0.08, 0.92)
        radii = np.maximum(radii, 0.02)
        
        params = np.concatenate([centers.flatten(), radii])
        
        # Multi-stage optimization with increasing penalty weights
        penalty_weights = [50, 300, 1500, 5000, 15000]
        
        for pw in penalty_weights:
            result = minimize(
                objective,
                params,
                args=(n, pw),
                method='L-BFGS-B',
                options={'maxiter': 5000, 'ftol': 1e-15, 'gtol': 1e-12}
            )
            params = result.x.copy()
        
        centers_opt = params[:2 * n].reshape(n, 2)
        radii_opt = params[2 * n:]
        
        # Ensure validity
        centers_opt, radii_opt = ensure_validity(centers_opt, radii_opt, n)
        
        current_sum = np.sum(radii_opt)
        
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers_opt.copy()
            best_radii = radii_opt.copy()
    
    # Final refinement pass on best solution
    params = np.concatenate([best_centers.flatten(), best_radii])
    
    for pw in [5000, 20000, 50000]:
        result = minimize(
            objective,
            params,
            args=(n, pw),
            method='L-BFGS-B',
            options={'maxiter': 10000, 'ftol': 1e-15, 'gtol': 1e-12}
        )
        params = result.x.copy()
    
    best_centers = params[:2 * n].reshape(n, 2)
    best_radii = params[2 * n:]
    best_centers, best_radii = ensure_validity(best_centers, best_radii, n)
    
    best_sum = np.sum(best_radii)
    
    return best_centers, best_radii, best_sum
