import numpy as np
import scipy.optimize as opt

def compute_objective(vars, N, penalty):
    centers = vars[:2*N].reshape(N, 2)
    radii = vars[2*N:]
    
    # Objective: maximize sum of radii
    loss = -np.sum(radii)
    
    # Pairwise distance matrix
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    
    # Radius sum matrix
    r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    
    # Penalty for overlapping circles
    overlap = r_sum - dists
    mask = np.triu(np.ones((N, N)), k=1)
    loss += penalty * np.sum(np.maximum(0, overlap)**2 * mask)
    
    # Penalty for circles crossing boundaries
    margins = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                         np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    boundary_v = np.maximum(0, radii - margins)
    loss += penalty * np.sum(boundary_v**2)
    
    return loss

def optimize_single(N, seed):
    rng = np.random.RandomState(seed)
    
    # Initialization: hexagonal lattice for even seeds, random for odd
    centers = np.zeros((N, 2))
    if seed % 2 == 0:
        d = 0.22
        idx = 0
        for i in range(10):
            for j in range(10):
                if idx >= N: break
                cx = 0.12 + j * d + (i % 2) * d / 2
                cy = 0.12 + i * d * np.sqrt(3) / 2
                if cx < 0.88 and cy < 0.88:
                    centers[idx] = [cx, cy]
                    idx += 1
            if idx >= N: break
        for i in range(idx, N):
            centers[i] = rng.uniform(0.2, 0.8, 2)
    else:
        centers = rng.uniform(0.15, 0.85, (N, 2))
        
    radii = rng.uniform(0.05, 0.12, N)
    vars = np.concatenate([centers.flatten(), radii])
    
    bounds = [(0.0, 1.0)] * (2*N) + [(0.001, 0.45)] * N
    
    # Annealing penalty schedule with perturbation
    current_penalty = 200.0
    for step in range(7):
        current_penalty *= 2.5
        res = opt.minimize(compute_objective, vars, args=(N, current_penalty), method='L-BFGS-B',
                           bounds=bounds, options={'maxiter': 4000, 'ftol': 1e-15, 'gtol': 1e-14})
        vars = res.x
        # Random perturbation to escape local minima
        if step < 5:
            noise = rng.normal(0, 0.008, vars.shape)
            vars = np.clip(vars + noise, 0.0, 1.0)
            
    co = vars[:2*N].reshape(N, 2)
    ro = vars[2*N:]
    
    # Post-processing: strictly enforce constraints by shrinking radii if necessary
    min_ratio = 1.0
    for i in range(N):
        r = ro[i]
        x, y = co[i]
        m = min(x, 1.0 - x, y, 1.0 - y)
        if m < r:
            min_ratio = min(min_ratio, m / r)
        for j in range(i + 1, N):
            d = np.sqrt(np.sum((co[i] - co[j])**2))
            if d < ro[i] + ro[j]:
                min_ratio = min(min_ratio, d / (ro[i] + ro[j]))
                
    if min_ratio < 1.0:
        ro *= min_ratio * 0.999999
        
    return co, ro, np.sum(ro)

def run_packing():
    N = 26
    best_sum = -1.0
    best_c = None
    best_r = None
    
    # Multiple restarts to find global optimum
    for s in range(30):
        c, r, s_val = optimize_single(N, s)
        if s_val > best_sum:
            best_sum = s_val
            best_c = c
            best_r = r
            
    return best_c, best_r, best_sum
