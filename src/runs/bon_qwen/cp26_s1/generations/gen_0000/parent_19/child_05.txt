import numpy as np
from scipy.optimize import minimize

def objective_func(vars):
    """Minimize negative sum of radii to maximize sum of radii."""
    return -np.sum(vars[2::3])

def constraint_func(vars):
    """
    Compute all inequality constraints.
    Returns an array where all elements must be >= 0.
    """
    N = len(vars) // 3
    c = np.empty(N * 4 + N * (N - 1) // 2)
    idx = 0
    
    # Boundary constraints: r <= x <= 1-r  and  r <= y <= 1-r
    for i in range(N):
        c[idx] = vars[3*i] - vars[3*i+2]          # x - r >= 0
        idx += 1
        c[idx] = 1.0 - vars[3*i] - vars[3*i+2]    # 1 - x - r >= 0
        idx += 1
        c[idx] = vars[3*i+1] - vars[3*i+2]        # y - r >= 0
        idx += 1
        c[idx] = 1.0 - vars[3*i+1] - vars[3*i+2]  # 1 - y - r >= 0
        idx += 1
        
    # Non-overlap constraints: dist^2 >= (r_i + r_j)^2
    for i in range(N):
        for j in range(i+1, N):
            dx = vars[3*i] - vars[3*j]
            dy = vars[3*i+1] - vars[3*j+1]
            dr = vars[3*i+2] + vars[3*j+2]
            c[idx] = dx*dx + dy*dy - dr*dr
            idx += 1
            
    return c

def generate_initial(N, method, seed):
    """Generate diverse initial configurations for multi-start optimization."""
    rng = np.random.RandomState(seed)
    centers = np.zeros((N, 2))
    radii = np.full(N, 0.025)  # Start small to ensure initial feasibility
    
    if method == 0:
        # Perturbed 5x5 grid + 1 extra
        pos = []
        for i in range(5):
            for j in range(5):
                pos.append((0.1 + i*0.18, 0.1 + j*0.18))
        pos.append((0.5, 0.5))
        rng.shuffle(pos)
        centers[:N] = pos[:N]
    elif method == 1:
        # Hexagonal-like pattern
        r_init = 0.055
        centers_list = []
        y = r_init
        pattern = [6, 5, 6, 5, 4]
        for count in pattern:
            shift = r_init if count % 2 == 1 else 0.0
            for i in range(count):
                x = r_init + shift + i * 2 * r_init
                centers_list.append([x, y])
                if len(centers_list) >= N:
                    break
            if len(centers_list) >= N:
                break
            y += np.sqrt(3) * r_init
        centers[:N] = centers_list[:N]
    else:
        # Random scatter
        centers = rng.uniform(0.15, 0.85, size=(N, 2))
        
    return centers, radii

def run_packing():
    N = 26
    best_obj = -np.inf
    best_vars = None
    
    # Define bounds for each variable: x in [0,1], y in [0,1], r in [1e-7, 0.5]
    single_bounds = [(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)]
    bounds = single_bounds * N
    
    # Inequality constraint
    cons = {'type': 'ineq', 'fun': constraint_func}
    
    # Multi-start optimization
    methods = [0, 1, 2]
    seeds = range(12)
    
    for method in methods:
        for seed in seeds:
            centers, radii = generate_initial(N, method, seed)
            x0 = np.zeros(3*N)
            for i in range(N):
                x0[3*i] = centers[i, 0]
                x0[3*i+1] = centers[i, 1]
                x0[3*i+2] = radii[i]
                
            try:
                res = minimize(objective_func, x0, method='SLSQP', 
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 3000, 'ftol': 1e-12})
                
                # Track best solution
                if -res.fun > best_obj:
                    best_obj = -res.fun
                    best_vars = res.x
            except Exception:
                continue
                
    # Extract solution
    sol = best_vars
    centers = sol.reshape((N, 3))[:, :2]
    radii = sol.reshape((N, 3))[:, 2]
    
    # Post-processing: strictly enforce boundary constraints to satisfy validator tolerance
    radii = np.maximum(radii, 1e-7)
    centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
    centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)
    
    return centers, radii, np.sum(radii)
