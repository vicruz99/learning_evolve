import numpy as np
import scipy.optimize
import math

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # --- 1. Initial Configuration (Heuristic) ---
    # Start with 5x5 grid of radius 0.1
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.1
    
    idx = 0
    for r in range(5):
        for c in range(5):
            centers[idx] = [0.1 + c * 0.2, 0.1 + r * 0.2]
            idx += 1
            
    # Place 26th circle in a gap. 
    # The center of the square of circles at (0.1,0.1), (0.3,0.1), (0.1,0.3), (0.3,0.3) is (0.2, 0.2)? 
    # No, center of square formed by (0.1,0.1) and (0.3,0.3) is (0.2, 0.2).
    # But (0.2, 0.2) is distance 0.1414 from corners. 
    # Gap radius ~ 0.0414.
    # However, (0.2, 0.2) is not a center of any circle in the grid?
    # Grid centers: x in {0.1, 0.3, 0.5, 0.7, 0.9}. 
    # 0.2 is not in grid. So (0.2, 0.2) is a valid gap location.
    centers[n-1] = [0.2, 0.2]
    radii[n-1] = 0.04 # Initial small radius
    
    # --- 2. Force-Directed Relaxation & Inflation ---
    # This helps to "jiggle" circles out of the rigid grid and allow them to grow/rearrange
    
    dt = 0.05
    inflation_rate = 0.005
    
    for step in range(1000):
        # Inflate radii
        radii += inflation_rate / (1 + step * 0.01) # Slow down inflation
        
        # Compute forces
        forces = np.zeros_like(centers)
        
        # Overlap repulsion
        for i in range(n):
            for j in range(i + 1, n):
                diff = centers[i] - centers[j]
                dist = np.linalg.norm(diff)
                min_dist = radii[i] + radii[j]
                if dist < min_dist and dist > 1e-9:
                    # Repulsive force proportional to overlap, inversely proportional to dist
                    overlap = min_dist - dist
                    force_mag = overlap * 10.0 
                    force_vec = (diff / dist) * force_mag
                    forces[i] += force_vec
                    forces[j] -= force_vec
                elif dist < 1e-9:
                    # Prevent division by zero, push randomly
                    forces[i] += np.random.randn(2) * 0.1
                    forces[j] -= np.random.randn(2) * 0.1

        # Boundary repulsion
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            # Left
            if x - r < 0:
                forces[i, 0] += (r - x) * 20.0
            # Right
            if x + r > 1:
                forces[i, 0] -= (x + r - 1) * 20.0
            # Bottom
            if y - r < 0:
                forces[i, 1] += (r - y) * 20.0
            # Top
            if y + r > 1:
                forces[i, 1] -= (y + r - 1) * 20.0
                
        # Apply forces
        centers += forces * dt
        
        # Clamp centers to valid range [0, 1] roughly (though forces push back)
        # Actually, let's just clamp to [0,1] to prevent escape
        centers = np.clip(centers, 0, 1)
        
        # If valid, keep; if invalid, the forces will correct it in next step
        
    # --- 3. Gradient-Based Optimization (SLSQP) ---
    # Refine the solution found by simulation
    
    def objective(z):
        # z is flattened [x0, y0, r0, x1, y1, r1, ...]
        radii_opt = z[2::3]
        return -np.sum(radii_opt)
    
    def constraint_overlap(z):
        # Constraint: dist >= r_i + r_j  =>  dist^2 - (r_i+r_j)^2 >= 0
        # We return the value, must be >= 0
        cons = []
        centers_opt = np.zeros((n, 2))
        radii_opt = np.zeros(n)
        
        for i in range(n):
            centers_opt[i] = z[3*i], z[3*i+1]
            radii_opt[i] = z[3*i+2]
            
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers_opt[i, 0] - centers_opt[j, 0]
                dy = centers_opt[i, 1] - centers_opt[j, 1]
                dist_sq = dx*dx + dy*dy
                r_sum = radii_opt[i] + radii_opt[j]
                # Avoid numerical issues with very small dist
                val = dist_sq - r_sum*r_sum
                cons.append(val)
        return np.array(cons)

    def constraint_boundary_x_min(z):
        # x >= r
        cons = []
        for i in range(n):
            cons.append(z[3*i] - z[3*i+2])
        return np.array(cons)

    def constraint_boundary_x_max(z):
        # 1 - x >= r  =>  1 - x - r >= 0
        cons = []
        for i in range(n):
            cons.append(1.0 - z[3*i] - z[3*i+2])
        return np.array(cons)
        
    def constraint_boundary_y_min(z):
        # y >= r
        cons = []
        for i in range(n):
            cons.append(z[3*i+1] - z[3*i+2])
        return np.array(cons)

    def constraint_boundary_y_max(z):
        # 1 - y >= r
        cons = []
        for i in range(n):
            cons.append(1.0 - z[3*i+1] - z[3*i+2])
        return np.array(cons)

    # Prepare initial guess for optimizer
    z0 = np.zeros(n * 3)
    for i in range(n):
        z0[3*i] = centers[i, 0]
        z0[3*i+1] = centers[i, 1]
        z0[3*i+2] = radii[i]
        
    # Bounds for variables
    bounds = []
    for i in range(n):
        bounds.append((0, 1)) # x
        bounds.append((0, 1)) # y
        bounds.append((0, 0.5)) # r
        
    # Constraints
    cons = [
        {'type': 'ineq', 'fun': constraint_overlap},
        {'type': 'ineq', 'fun': constraint_boundary_x_min},
        {'type': 'ineq', 'fun': constraint_boundary_x_max},
        {'type': 'ineq', 'fun': constraint_boundary_y_min},
        {'type': 'ineq', 'fun': constraint_boundary_y_max}
    ]
    
    # Run optimizer
    # We might need to run it a few times or with different settings if it gets stuck
    # SLSQP is good for this
    
    best_z = z0.copy()
    best_val = objective(z0)
    
    try:
        res = scipy.optimize.minimize(objective, z0, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 500})
        if res.fun < best_val:
            best_z = res.x
            best_val = res.fun
    except Exception:
        pass
        
    # Extract results
    final_centers = np.zeros((n, 2))
    final_radii = np.zeros(n)
    for i in range(n):
        final_centers[i] = best_z[3*i], best_z[3*i+1]
        final_radii[i] = best_z[3*i+2]
        
    # Clamp radii to non-negative just in case
    final_radii = np.maximum(final_radii, 0)
    
    sum_r = np.sum(final_radii)
    
    return final_centers, final_radii, sum_r
