    cy = params[2 * n:3 * n]

        return 1e10

    P = 1e6

        x, y = cx[i], cy[i]
        if r < 0:

import numpy as np

def loss(params, n):
    radii = params[:n]
    cx = params[n:2*n]
    cy = params[2*n:3*n]
    obj = -np.sum(radii)
    pen = 0.0
    W = 1e8
    for i in range(n):
        r = radii[i]
        x, y = cx[i], cy[i]
        if r < 0:
            pen += W * r * r
        for v in [x - r, y - r, r - (1 - x), r - (1 - y)]:
            if v > 0:
                pen += W * v * v
    for i in range(n):
        for j in range(i + 1, n):
            d = np.sqrt((cx[i] - cx[j])**2 + (cy[i] - cy[j])**2)
            o = radii[i] + radii[j] - d
            if o > 0:
                pen += W * o * o
    return obj + pen


def project(centers, radii, n):
    for _ in range(10):
        for i in range(n):
            r = radii[i]
            x, y = centers[i]
            r = min(r, x, 1 - x, y, 1 - y)
            for j in range(n):
                if i != j:
                    d = np.sqrt((x - centers[j,0])**2 + (y - centers[j,1])**2)
                    r = min(r, d - radii[j])
            radii[i] = max(r, 0.0)
    return centers, radii

    """Refine a configuration using both L-BFGS-B and Nelder-Mead."""

    res = minimize(loss, p, args=(n,), method='L-BFGS-B',
