import numpy as np
from scipy.optimize import minimize
import math

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    n = centers.shape[0]

    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True

def compute_loss(params, n_circles):
    """
    Computes the loss function: -(sum of radii) + penalties.
    """
    x = params[0:n_circles]
    y = params[n_circles:2*n_circles]
    r = params[2*n_circles:3*n_circles]
    
    loss = -np.sum(r)
    
    # Boundary penalties
    # Penalty for going outside [0, 1]
    for i in range(n_circles):
        if x[i] < r[i]: loss += 100.0 * (r[i] - x[i])**2
        if x[i] > 1 - r[i]: loss += 100.0 * (x[i] - (1 - r[i]))**2
        if y[i] < r[i]: loss += 100.0 * (r[i] - y[i])**2
        if y[i] > 1 - r[i]: loss += 100.0 * (y[i] - (1 - r[i]))**2
        
    # Overlap penalties
    for i in range(n_circles):
        for j in range(i + 1, n_circles):
            dx = x[i] - x[j]
            dy = y[i] - y[j]
            dist = math.sqrt(dx*dx + dy*dy)
            min_dist = r[i] + r[j]
            if dist < min_dist:
                loss += 100.0 * (min_dist - dist)**2
                
    return loss

def run_packing():
    n_circles = 26
    best_loss = float('inf')
    best_params = None
    
    # Number of restarts to find a good global optimum
    n_restarts = 10
    
    for k in range(n_restarts):
        # --- Initialization ---
        # Strategy: Start with a 5x5 grid (25 circles) and 1 in the center gap
        # Then add some noise to escape local optima
        centers = []
        radii = []
        
        for i in range(5):
            for j in range(5):
                cx = 0.1 + i * 0.2
                cy = 0.1 + j * 0.2
                centers.append([cx, cy])
                radii.append(0.1)
        
        # 26th circle in the center gap
        centers.append([0.5, 0.5])
        radii.append(0.04)
        
        centers = np.array(centers)
        radii = np.array(radii)
        
        # Add random perturbation
        np.random.seed(42 + k)
        centers += np.random.uniform(-0.005, 0.005, centers.shape)
        radii += np.random.uniform(-0.001, 0.001, len(radii))
        
        # Ensure radii are positive
        radii = np.maximum(radii, 1e-4)
        
        # Flatten for optimizer
        # Order: x's, y's, r's
        x = centers[:, 0]
        y = centers[:, 1]
        params = np.concatenate([x, y, radii])
        
        # Bounds for x, y (0 to 1), r (0 to 1)
        bounds = []
        for _ in range(n_circles):
            bounds.append((0, 1)) # x
            bounds.append((0, 1)) # y
            bounds.append((1e-5, 1)) # r
            
        # Optimization
        res = minimize(
            compute_loss, 
            params, 
            args=(n_circles,), 
            method='L-BFGS-B', 
            bounds=bounds,
            options={'ftol': 1e-10, 'gtol': 1e-10, 'maxiter': 1000}
        )
        
        if res.fun < best_loss:
            best_loss = res.fun
            best_params = res.x
            
    # --- Extract and Fix Results ---
    if best_params is None:
        # Fallback if optimization failed
        return np.zeros((26, 2)), np.zeros(26), 0.0

    y = best_params[n_circles:2*n_circles]
    r = best_params[2*n_circles:3*n_circles]

        r[i] = min(r[i], min(x[i], 1-x[i], y[i], 1-y[i]))
        r[i] = max(r[i], 0)
        
    # One final check and slight adjustment if needed
    # If any overlap remains due to discretization, reduce radii slightly
    # This is a safety net.
    # But with L-BFGS-B and high penalty, it should be valid.

    if not validate_packing(centers, r):

        centers = np.zeros((n_circles, 2))
        r = np.zeros(n_circles)
        idx = 0
        for i in range(5):

                idx += 1

        r[idx] = 0.0414
    else:

        for i in range(n_circles):
            x, y = centers[i]
            rad = r[i]
            centers[i][0] = np.clip(x, rad, 1-rad)
            centers[i][1] = np.clip(y, rad, 1-rad)

    sum_radii = np.sum(r)
    return centers, r, sum_radii
