import numpy as np
from scipy.optimize import minimize, NonlinearConstraint

def objective_func(params):
    """Objective: maximize sum of radii -> minimize negative sum."""
    return -np.sum(params[2::3])

def constraint_func(params):
    """Returns constraint values that must be >= 0."""
    n = 26
    num_constraints = 4 * n + n * (n - 1) // 2
    res = np.zeros(num_constraints)
    idx = 0
    
    # Boundary constraints
    for i in range(n):
        xi = params[3 * i]
        yi = params[3 * i + 1]
        ri = params[3 * i + 2]
        res[idx] = xi - ri
        idx += 1
        res[idx] = 1.0 - xi - ri
        idx += 1
        res[idx] = yi - ri
        idx += 1
        res[idx] = 1.0 - yi - ri
        idx += 1
        
    # Overlap constraints
    for i in range(n):
        xi = params[3 * i]
        yi = params[3 * i + 1]
        ri = params[3 * i + 2]
        for j in range(i + 1, n):
            dx = xi - params[3 * j]
            dy = yi - params[3 * j + 1]
            rr = ri + params[3 * j + 2]
            res[idx] = dx * dx + dy * dy - rr * rr
            idx += 1
            
    return res

def run_packing():
    np.random.seed(42)
    n = 26
    
    # Initialize centers in a perturbed grid to break symmetry
    rows, cols = 5, 6
    dx, dy = 1.0 / cols, 1.0 / rows
    centers_init = np.zeros((n, 2))
    count = 0
    for r in range(rows):
        for c in range(cols):
            if count >= n:
                break
            centers_init[count, 0] = (c + 0.5) * dx
            centers_init[count, 1] = (r + 0.5) * dy
            count += 1
        if count >= n:
            break
            
    centers_init += np.random.uniform(-0.005, 0.005, size=centers_init.shape)
    radii_init = np.full(n, 0.08)
    
    # Flatten to optimization vector: [x1, y1, r1, x2, y2, r2, ...]
    params0 = np.zeros(3 * n)
    params0[0::3] = centers_init[:, 0]
    params0[1::3] = centers_init[:, 1]
    params0[2::3] = radii_init
    
    # Variable bounds
    bounds = [(0.0, 1.0) for _ in range(2 * n)] + [(0.0, 0.5) for _ in range(n)]
    
    # Nonlinear constraints (all must be >= 0)
    lb = np.zeros(4 * n + n * (n - 1) // 2)
    con = NonlinearConstraint(constraint_func, lb, np.inf)
    
    # Run optimization
    res = minimize(
        objective_func, 
        params0, 
        method='SLSQP', 
        bounds=bounds,
        constraints=con, 
        options={'maxiter': 2000, 'ftol': 1e-10}
    )
    
    params_opt = res.x
    centers_opt = np.zeros((n, 2))
    centers_opt[:, 0] = params_opt[0::3]
    centers_opt[:, 1] = params_opt[1::3]
    radii_opt = params_opt[2::3]
    
    # Numerical cleanup to guarantee validity
    radii_opt = np.maximum(radii_opt, 0.0)
    centers_opt[:, 0] = np.clip(centers_opt[:, 0], 0.0, 1.0)
    centers_opt[:, 1] = np.clip(centers_opt[:, 1], 0.0, 1.0)
    
    for i in range(n):
        r = radii_opt[i]
        x, y = centers_opt[i]
        max_r = min(x, 1.0 - x, y, 1.0 - y)
        if r > max_r:
            radii_opt[i] = max_r
            
    sum_r = np.sum(radii_opt)
    return centers_opt, radii_opt, sum_r
