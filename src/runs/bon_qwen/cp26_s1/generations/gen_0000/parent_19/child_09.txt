import numpy as np
from scipy.optimize import minimize

def _make_constraints(n):
    """Create boundary and non-overlap inequality constraints for SLSQP."""
    constraints = []
    for i in range(n):
        idx = 3 * i
        # Boundary: x >= r, 1-x >= r, y >= r, 1-y >= r
        constraints.append({'type': 'ineq', 'fun': lambda v, idx=idx: v[idx] - v[idx+2]})
        constraints.append({'type': 'ineq', 'fun': lambda v, idx=idx: 1.0 - v[idx] - v[idx+2]})
        constraints.append({'type': 'ineq', 'fun': lambda v, idx=idx: v[idx+1] - v[idx+2]})
        constraints.append({'type': 'ineq', 'fun': lambda v, idx=idx: 1.0 - v[idx+1] - v[idx+2]})

    # Non-overlap: dist^2 >= (r_i + r_j)^2
    for i in range(n):
        for j in range(i+1, n):
            ii, jj = i, j
            constraints.append({
                'type': 'ineq',
                'fun': lambda v, ii=ii, jj=jj: 
                    (v[3*ii] - v[3*jj])**2 + (v[3*ii+1] - v[3*jj+1])**2 - (v[3*ii+2] + v[3*jj+2])**2
            })
    return constraints

def _get_initial(n, seed):
    """Generate a feasible initial configuration on a perturbed grid."""
    rng = np.random.default_rng(seed)
    pts = []
    # Dense grid to place n circles
    xs = np.linspace(0.1, 0.9, 6)
    ys = np.linspace(0.1, 0.9, 5)
    for y in ys:
        for x in xs:
            if len(pts) >= n:
                break
            pts.append([x, y])
        if len(pts) >= n:
            break
            
    pts = np.array(pts[:n])
    # Add controlled random perturbation to break symmetry and explore space
    pts += rng.uniform(-0.04, 0.04, size=pts.shape)
    pts = np.clip(pts, 0.05, 0.95)
    
    r_init = 0.06 * np.ones(n)
    vars = np.empty(3 * n)
    vars[0::3] = pts[:, 0]
    vars[1::3] = pts[:, 1]
    vars[2::3] = r_init
    return vars

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # Bounds for [x, y, r]
    bounds = []
    for _ in range(n):
        bounds.extend([(0.0, 1.0), (0.0, 1.0), (1e-7, 0.5)])
        
    constraints = _make_constraints(n)
    
    best_sum = -1.0
    best_vars = None
    
    # Multi-start optimization
    for seed in range(20):
        x0 = _get_initial(n, seed)
        try:
            res = minimize(
                lambda v: -np.sum(v[2::3]),
                x0,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,

                best_sum = current_sum

    # Extract results
    centers = np.zeros((n, 2))
    radii = best_vars[2::3].copy()
    centers[:, 0] = best_vars[0::3]
    centers[:, 1] = best_vars[1::3]
    
    # Numerical safety: ensure non-negative radii
    radii = np.maximum(radii, 1e-9)
    
    return centers, radii, float(np.sum(radii))
