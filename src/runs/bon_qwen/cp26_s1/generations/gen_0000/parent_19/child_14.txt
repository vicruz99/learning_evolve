2. **Objective Function**: Maximize $\sum r_i$ by minimizing a composite loss: $-\sum r_i + \lambda \cdot (\text{boundary violations} + \text{overlap violations})$. Quadratic penalties with a high coefficient ($\lambda=10000$) strongly discourage constraint violations.

import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

# Precompute unique pair indices for efficient overlap penalty calculation
_PAIR_INDICES = []
for i in range(N_CIRCLES):
    for j in range(i + 1, N_CIRCLES):
        _PAIR_INDICES.append((i, j))
_PAIR_INDICES = np.array(_PAIR_INDICES)

def _compute_objective(vars):
    """
    Objective: Maximize sum of radii with heavy quadratic penalties for violations.
    vars layout: [x0, y0, r0, x1, y1, r1, ..., x25, y25, r25]
    """
    pos = vars[:2 * N_CIRCLES].reshape(N_CIRCLES, 2)
    radii = vars[2 * N_CIRCLES:]
    
    # Boundary penalties: ensure r <= x <= 1-r and r <= y <= 1-r
    # Equivalent to: r - x <= 0, r + x - 1 <= 0, etc.
    p_bound = np.sum(np.maximum(0.0, radii - pos[:, 0])**2)
    p_bound += np.sum(np.maximum(0.0, radii + pos[:, 0] - 1.0)**2)
    p_bound += np.sum(np.maximum(0.0, radii - pos[:, 1])**2)
    p_bound += np.sum(np.maximum(0.0, radii + pos[:, 1] - 1.0)**2)
    
    # Overlap penalties: ensure dist(i,j) >= r_i + r_j
    xi, yi = _PAIR_INDICES[:, 0], _PAIR_INDICES[:, 1]
    dx = pos[xi, 0] - pos[yi, 0]
    dy = pos[xi, 1] - pos[yi, 1]
    dists = np.sqrt(dx * dx + dy * dy)
    sums_r = radii[xi] + radii[yi]
    overlap = np.maximum(0.0, sums_r - dists)
    p_overlap = np.sum(overlap**2)
    
    # High penalty weight to enforce strict feasibility
    lam = 10000.0
    return -np.sum(radii) + lam * (p_bound + p_overlap)

def _get_initial_guess():
    """Hexagonal-ish grid initialization to guide optimization."""
    rows = [6, 5, 5, 5, 5]
    init = np.zeros(3 * N_CIRCLES)
    idx = 0
    for r_idx, count in enumerate(rows):
        y = 0.15 + r_idx * 0.17
        width = (count - 1) * 0.14
        sx = (1.0 - width) / 2.0
        for c in range(count):
            init[idx * 3] = sx + c * 0.14
            init[idx * 3 + 1] = y
            init[idx * 3 + 2] = 0.09  # Initial radius guess
            idx += 1
    return init

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N_CIRCLES
    x0 = _get_initial_guess()
    
    # Run L-BFGS-B optimization
    res = minimize(_compute_objective, x0, method='L-BFGS-B', bounds=bounds, 
                   options={'maxiter': 5000, 'ftol': 1e-15})
    
    pos = res.x[:2 * N_CIRCLES].reshape(N_CIRCLES, 2)
    radii = res.x[2 * N_CIRCLES:]
    
    # Strict feasibility check and corrective scaling
    min_clearance = 1.0
    
    # Check boundaries
    for i in range(N_CIRCLES):
        min_clearance = min(min_clearance, 
                            pos[i, 0] - radii[i], 1.0 - pos[i, 0] - radii[i],
                            pos[i, 1] - radii[i], 1.0 - pos[i, 1] - radii[i])
        
    # Check pairwise overlaps
    for i in range(N_CIRCLES):
        for j in range(i + 1, N_CIRCLES):
            d = np.sqrt(np.sum((pos[i] - pos[j])**2))
            min_clearance = min(min_clearance, d - radii[i] - radii[j])
            
    # If any constraint is violated (even by epsilon), scale radii down safely
    if min_clearance < 1e-7:
        scale_factor = 1.0 - max(0.0, -min_clearance) * 1.001
        radii *= scale_factor
        
    return pos, radii, float(np.sum(radii))
