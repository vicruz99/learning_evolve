import numpy as np
from scipy.optimize import minimize

def compute_loss(params, penalty_factor):
    """
    Computes the objective function with penalty terms for boundary and overlap constraints.
    """
    centers = params[:52].reshape(26, 2)
    radii = params[52:]

    # Primary objective: maximize sum of radii -> minimize negative sum
    obj = -np.sum(radii)

    # Boundary penalties: ensure r <= x <= 1-r and r <= y <= 1-r
    vx = centers[:, 0]
    vy = centers[:, 1]
    bound_pen = np.sum(
        np.maximum(0, radii - vx)**2 + 
        np.maximum(0, radii - (1.0 - vx))**2 +
        np.maximum(0, radii - vy)**2 + 
        np.maximum(0, radii - (1.0 - vy))**2
    )

    # Overlap penalties: ensure r_i + r_j <= dist_ij
    diffs = centers[:, None, :] - centers[None, :, :]
    dists = np.linalg.norm(diffs, axis=2)

    sum_r = radii[:, None] + radii[None, :]

    # Only consider upper triangle to avoid double counting
    mask = np.triu(np.ones((26, 26), dtype=bool), k=1)
    violations = np.maximum(0, sum_r[mask] - dists[mask])
    overlap_pen = np.sum(violations**2)

    return obj + penalty_factor * (bound_pen + overlap_pen)

def run_packing():
    """
    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    Returns (centers, radii, sum_radii).
    """
    # 1. Initialize positions on a hexagonal lattice
    centers = np.zeros((26, 2))
    radii = np.full(26, 0.06)  # Start small to avoid initial overlaps
    
    idx = 0
    for row in range(6):
        for col in range(5):
            if idx >= 26:
                break
            # Hexagonal spacing
            x = 0.15 + col * 0.22
            y = 0.15 + row * 0.19
            if row % 2 == 1:
                x += 0.11
            centers[idx] = [x, y]
            idx += 1
        if idx >= 26:
            break

    # Flatten to 1D array for optimizer: [x1, y1, ..., x26, y26, r1, ..., r26]
    params = np.concatenate([centers.ravel(), radii])
    
    # Define bounds: centers in [0, 1], radii in [1e-6, 0.5]
    bnds = [(0.0, 1.0)] * 52 + [(1e-6, 0.5)] * 26

    # 2. Iterative optimization with increasing penalty weight
    current_penalty = 100.0
    for _ in range(5):
        res = minimize(
            compute_loss, 
            params, 
            args=(current_penalty,),
            method='L-BFGS-B', 
            bounds=bnds,
            options={'maxiter': 2500, 'ftol': 1e-12}
        )
        params = res.x
        current_penalty *= 4.0  # Increase penalty to tighten constraints

    # 3. Extract final results
    centers_opt = params[:52].reshape(26, 2)
    radii_opt = params[52:]
    sum_r = np.sum(radii_opt)
    
    return centers_opt, radii_opt, sum_r
