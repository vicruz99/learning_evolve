import numpy as np
from scipy.optimize import minimize

# Global constant for number of circles
N_CIRCLES = 26

def calculate_penalty(centers_flat, r):
    """
    Calculates the penalty for a given configuration of centers and radius r.
    Penalty is non-zero if circles overlap or are outside the unit square.
    
    Args:
        centers_flat: np.array of shape (2*N_CIRCLES,), flattened centers (x1, y1, x2, y2, ...)
        r: float, radius of circles
        
    Returns:
        float, total penalty
    """
    centers = centers_flat.reshape(-1, 2)
    penalty = 0.0
    
    # Boundary penalties
    # Circle i must be within [r, 1-r] for both x and y
    # Penalty for being too small
    lower_violation = np.maximum(0, r - centers)
    penalty += np.sum(lower_violation**2)
    
    # Penalty for being too large
    upper_violation = np.maximum(0, centers - (1 - r))
    penalty += np.sum(upper_violation**2)
    
    # Overlap penalties
    # Distance between any pair must be >= 2r
    # We sum (2r - dist)^2 if dist < 2r
    for i in range(N_CIRCLES):
        for j in range(i + 1, N_CIRCLES):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            min_dist = 2 * r
            if dist < min_dist:
                overlap = min_dist - dist
                penalty += overlap**2
                
    return penalty

def check_feasibility(r, num_restarts=20, seed=42):
    """
    Checks if N_CIRCLES of radius r can fit in the unit square.
    Uses optimization to find centers.
    
    Args:
        r: float, radius to test
        num_restarts: int, number of random initializations
        seed: int, random seed
        
    Returns:
        tuple (bool, np.array): (is_feasible, best_centers)
    """
    best_penalty = float('inf')
    best_centers = None
    
    # Bounds for optimization: [r, 1-r] for each coordinate
    bounds = [(r, 1 - r)] * (2 * N_CIRCLES)
    
    rng = np.random.RandomState(seed)
    
    for _ in range(num_restarts):
        # Initialize centers randomly within bounds
        # Try to spread them out initially
        init_centers = np.zeros((N_CIRCLES, 2))
        for i in range(N_CIRCLES):
            # Simple grid-like initialization with noise
            row = i // 6
            col = i % 6
            # Map to unit square roughly
            x = r + (col + 0.5) * (1 - 2*r) / 6
            y = r + (row + 0.5) * (1 - 2*r) / 6
            # Add small noise
            x += rng.uniform(-0.01, 0.01)
            y += rng.uniform(-0.01, 0.01)
            # Clamp to bounds
            x = np.clip(x, r, 1-r)
            y = np.clip(y, r, 1-r)
            init_centers[i] = [x, y]
            
        x0 = init_centers.flatten()
        
        # Optimization
        try:
            res = minimize(
                calculate_penalty,
                x0,
                args=(r,),
                method='L-BFGS-B',
                bounds=bounds,
                options={'ftol': 1e-10, 'gtol': 1e-8, 'maxiter': 1000}
            )
            
            if res.fun < best_penalty:
                best_penalty = res.fun
                best_centers = res.x.reshape(-1, 2)
                
            # If we found a valid packing (penalty approx 0), return immediately
            if best_penalty < 1e-6:
                return True, best_centers
                
        except Exception:
            continue
            
    return best_penalty < 1e-6, best_centers

def find_optimal_packing():
    """
    Finds the optimal radius and centers for 26 circles.
    Uses binary search on radius.
    """
    # Initial search range
    # r=0.1 is easy (5x5 grid fits 25, 26 might be tight but let's check lower)
    # r=0.05 is trivial
    low = 0.05
    high = 0.12 
    best_r = low
    best_centers = None
    
    # Binary search steps
    for _ in range(20):
        mid = (low + high) / 2.0
        
        # We increase restarts as we get tighter to ensure robustness
        # But keep it fast. 
        feasible, centers = check_feasibility(mid, num_restarts=15)
        
        if feasible:
            best_r = mid
            best_centers = centers
            low = mid
        else:
            high = mid
            
    # Return result
    radii = np.full(N_CIRCLES, best_r)
    sum_radii = np.sum(radii)
    
    return best_centers, radii, sum_radii

def run_packing():
    """
    Main function to run the packing optimization.
    Returns centers, radii, sum_radii.
    """
    centers, radii, sum_radii = find_optimal_packing()
    return centers, radii, sum_radii
