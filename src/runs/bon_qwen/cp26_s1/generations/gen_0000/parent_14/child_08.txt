from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    n_circles = 26
    box_size = 1.0
    epsilon = 1e-6

    centers = np.zeros((n_circles, 2))
    radii = np.zeros(n_circles)
    
    r_est = 0.058  # Initial estimated radius
    centers_list = []

    rows = [(9, 0.0), (8, r_est), (9, 0.0)]
    y_step = np.sqrt(3) * r_est
    
    current_idx = 0
    for i, (count, offset) in enumerate(rows):
        y_pos = r_est + i * y_step
        # Center the row horizontally
        row_width = 2 * r_est * count
        x_start = (1 - row_width) / 2 + offset
        
        for j in range(count):
            x_pos = x_start + 2 * r_est * j
            centers_list.append([x_pos, y_pos])
            current_idx += 1
            
    centers = np.array(centers_list[:n_circles])
    radii[:] = r_est

    # 2. Stage 1: Force-directed simulation to expand and relax
    # This handles the non-convexity and finds a good local basin
    temp = 1.0
    k_repulse = 1.0
    k_expand = 0.5
    wall_penalty = 10.0
    
    for step in range(1500):
        forces = np.zeros_like(centers)
        
        # Repulsion between circles
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                diff = centers[i] - centers[j]
                dist = np.linalg.norm(diff)
                min_dist = radii[i] + radii[j]
                
                if dist < min_dist + 0.05 and dist > 1e-5:
                    overlap = min_dist - dist
                    # Force is proportional to overlap, inversely proportional to distance
                    force_vec = diff / dist * overlap * k_repulse
                    forces[i] += force_vec
                    forces[j] -= force_vec
        
        # Expansion pressure (grow radii)
        # Calculate max possible radius for each circle based on neighbors and walls
        current_radii = radii.copy()
        for i in range(n_circles):
            # Distance to walls
            r_wall = min(centers[i, 0], box_size - centers[i, 0], 
                         centers[i, 1], box_size - centers[i, 1])
            
            # Distance to other circles
            r_neighbors = np.inf
            for j in range(n_circles):
                if i == j: continue
                d = np.linalg.norm(centers[i] - centers[j])
                r_neighbors = min(r_neighbors, (d - current_radii[j]))
            
            # Target radius
            target_r = min(r_wall, r_neighbors / 2) # Rough estimate for equal radii
            
            # Apply pressure to grow if below target
            if target_r > current_radii[i]:
                forces[i] += (target_r - current_radii[i]) * k_expand * np.random.uniform(0.9, 1.1, 2)
                radii[i] = min(current_radii[i] + 1e-5, target_r)
        
        # Update positions
        centers += forces * 0.1 * (0.5 + 0.5 * np.random.random())
        
        # Wall constraints (reflect)
        for i in range(n_circles):
            if centers[i, 0] < radii[i]: centers[i, 0] = radii[i]
            if centers[i, 0] > box_size - radii[i]: centers[i, 0] = box_size - radii[i]
            if centers[i, 1] < radii[i]: centers[i, 1] = radii[i]
            if centers[i, 1] > box_size - radii[i]: centers[i, 1] = box_size - radii[i]

    # 3. Stage 2: Refined Optimization with SLSQP
    # Define objective and constraints for scipy
    
    def objective(vars):
        # vars layout: x0, y0, r0, x1, y1, r1, ...
        return -np.sum(vars[2::3]) # Maximize sum of radii

    def constraint_wall(vars):
        c = np.zeros(n_circles * 4)
        idx = 0
        for i in range(n_circles):
            x, y, r = vars[i*3:i*3+3]
            c[idx] = r - x # x - r >= 0
            c[idx+1] = r - (1 - x) # 1 - x - r >= 0
            c[idx+2] = r - y # y - r >= 0
            c[idx+3] = r - (1 - y) # 1 - y - r >= 0
            idx += 4
        return c

    def constraint_overlap(vars):
        c = np.zeros((n_circles * (n_circles - 1)) // 2)
        idx = 0
        for i in range(n_circles):
            xi, yi, ri = vars[i*3:i*3+3]
            for j in range(i + 1, n_circles):

                dist_sq = (xi - xj)**2 + (yi - yj)**2
                dist = np.sqrt(dist_sq + 1e-12) # numerical stability
                c[idx] = (ri + rj) - dist # dist - (ri+rj) >= 0
                idx += 1
        return c

    # Flatten variables
    initial_vars = np.zeros(n_circles * 3)
    for i in range(n_circles):
        initial_vars[i*3] = centers[i, 0]
        initial_vars[i*3+1] = centers[i, 1]
        initial_vars[i*3+2] = radii[i]

    # Bounds: 0 <= x,y <= 1, 0 <= r
    bounds = []
    for i in range(n_circles):
        bounds.extend([(0, 1), (0, 1), (0, 1)]) # x, y, r bounds

    # Setup constraints
    cons = []
    cons.append({'type': 'ineq', 'fun': constraint_wall})
    cons.append({'type': 'ineq', 'fun': constraint_overlap})

    # Optimize
    res = minimize(objective, initial_vars, method='SLSQP', bounds=bounds, 
                   constraints=cons, options={'maxiter': 1000, 'ftol': 1e-12})

    # Extract results
    final_vars = res.x
    final_centers = np.zeros((n_circles, 2))
    final_radii = np.zeros(n_circles)
    
    for i in range(n_circles):
        final_centers[i] = final_vars[i*3:i*3+2]
        final_radii[i] = final_vars[i*3+2]

    # Calculate sum
    sum_radii = np.sum(final_radii)
    
    return final_centers, final_radii, sum_radii
