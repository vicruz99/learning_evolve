import numpy as np
from scipy.optimize import minimize

def obj(x, n):
    """Objective function: negative sum of radii (to be minimized)"""
    r = x[2::3]
    return -np.sum(r)

def constr(x, n):
    """
    Constraint function: returns array of constraint values.
    Inequality constraints are satisfied when all values >= 0.
    """
    cx = x[0::3]
    cy = x[1::3]
    r = x[2::3]
    
    # Boundary constraints: circle i must be inside [0,1]x[0,1]
    c1 = cx - r          # x - r >= 0
    c2 = 1.0 - cx - r    # 1 - x - r >= 0
    c3 = cy - r          # y - r >= 0
    c4 = 1.0 - cy - r    # 1 - y - r >= 0
    
    # Pairwise non-overlap constraints: dist^2 >= (r_i + r_j)^2
    dx = cx[:, None] - cx[None, :]
    dy = cy[:, None] - cy[None, :]
    dist2 = dx**2 + dy**2
    
    r_sum = r[:, None] + r[None, :]
    r_sum2 = r_sum**2
    
    # Extract strict upper triangle to avoid duplicates and self-constraints
    idx = np.triu_indices(n, k=1)
    c_pair = dist2[idx] - r_sum2[idx]
    
    return np.concatenate([c1, c2, c3, c4, c_pair])

def init_x(n, seed):
    """Generate an initial feasible configuration using a perturbed grid"""
    np.random.seed(seed)
    cols = int(np.ceil(np.sqrt(n)))
    rows = int(np.ceil(n / cols))
    
    x = np.zeros(3 * n)
    for i in range(n):
        r_idx = i // cols
        c_idx = i % cols
        # Grid position with controlled jitter to break symmetry
        px = (c_idx + 1) / (cols + 1) + np.random.uniform(-0.015, 0.015)
        py = (r_idx + 1) / (rows + 1) + np.random.uniform(-0.015, 0.015)
        
        # Clamp strictly inside to guarantee initial feasibility
        px = np.clip(px, 0.05, 0.95)
        py = np.clip(py, 0.05, 0.95)
        
        idx = 3 * i
        x[idx] = px
        x[idx+1] = py
        x[idx+2] = 0.02  # Small initial radius ensures no overlaps initially
    return x

def run_packing():
    n = 26
    best_sum = -np.inf
    best_centers = None
    best_radii = None
    
    # Box bounds: x, y in [0, 1], r in [0, 1]
    bounds = [(0, 1) for _ in range(2*n)] + [(0, 1) for _ in range(n)]
    cons = {'type': 'ineq', 'fun': constr, 'args': (n,)}
    
    # Multiple restarts to escape local optima
    for seed in range(12):
        x0 = init_x(n, seed)
        res = minimize(obj, x0, args=(n,), method='SLSQP', 
                       constraints=cons, bounds=bounds,
                       options={'maxiter': 1500, 'ftol': 1e-10, 'disp': False})
        
        # Verify feasibility with numerical tolerance
        cons_val = constr(res.x, n)
        if np.all(cons_val >= -1e-7):
            current_sum = -res.fun
            if current_sum > best_sum:
                best_sum = current_sum
                best_centers = res.x[:2*n].reshape(n, 2)
                best_radii = res.x[2*n:]
                
    # Fallback to last result if somehow no feasible solution was recorded
    if best_centers is None:
        best_centers = res.x[:2*n].reshape(n, 2)
        best_radii = res.x[2*n:]
        best_sum = np.sum(best_radii)
        
    return best_centers, best_radii, best_sum
