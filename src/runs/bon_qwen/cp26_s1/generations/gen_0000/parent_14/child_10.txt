To maximize the sum of radii for 26 circles in a unit square, we treat this as a nested optimization problem:
1. **Inner Problem (Fixed Centers)**: For a given set of circle centers, finding the optimal radii that maximize $\sum r_i$ subject to non-overlap ($r_i + r_j \le \|c_i - c_j\|$) and boundary constraints ($r_i \le \text{dist}(c_i, \partial[0,1]^2)$) is a **Linear Programming (LP)** problem. This is exact and computationally cheap.
2. **Outer Problem (Optimal Centers)**: We optimize the 52 center coordinates to maximize the objective value returned by the inner LP. This is a non-convex, high-dimensional problem.
3. **Optimization Strategy**: 
   - Initialize centers using a hexagonal lattice pattern, which is known to be dense.
   - Use `scipy.optimize.differential_evolution` for robust global search over the center space.
   - Refine the result using `scipy.optimize.minimize` with the Nelder-Mead method for local convergence.
   - The LP solver efficiently computes the best radii for any center configuration, allowing the outer optimizer to focus purely on geometry.

This approach naturally handles unequal radii and exploits boundary gaps, typically yielding higher sums than equal-radius assumptions.

import numpy as np
from scipy.optimize import linprog, differential_evolution, minimize

def compute_max_radii(centers):
    """
    Solves the LP to maximize sum of radii for fixed centers.
    Returns (sum_radii, radii_array).
    """
    n = centers.shape[0]
    c_obj = -np.ones(n)  # Minimize negative sum = Maximize sum
    
    # Pairwise non-overlap constraints: r_i + r_j <= distance(i, j)
    i_idx, j_idx = np.triu_indices(n, k=1)
    n_pairs = len(i_idx)
    A_ub = np.zeros((n_pairs, n))
    A_ub[np.arange(n_pairs), i_idx] = 1.0
    A_ub[np.arange(n_pairs), j_idx] = 1.0
    
    diffs = centers[i_idx] - centers[j_idx]
    b_ub = np.sqrt(np.sum(diffs**2, axis=1))
    
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    dist_to_border = np.minimum(
        np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
        np.minimum(centers[:, 1], 1.0 - centers[:, 1])
    )
    bounds = [(0.0, float(d)) for d in dist_to_border]
    
    # Solve LP
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds)
        if res.success:
            radii = np.maximum(res.x, 0.0)  # Clip numerical negatives
            return np.sum(radii), radii
    except Exception:
        pass
        
    # Fallback: uniform small radii if LP fails
    r_fallback = np.maximum(dist_to_border * 0.8, 0.0)
    return np.sum(r_fallback), r_fallback

def objective_1d(x_flat):
    """Objective function for center optimization (minimize negative sum of radii)."""
    centers = x_flat.reshape(-1, 2)
    # Keep centers strictly inside to avoid degenerate LP bounds
    centers = np.clip(centers, 1e-4, 1.0 - 1e-4)
    s, _ = compute_max_radii(centers)
    return -s

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    n = 26
    
    # 1. Initialize with a hexagonal lattice pattern
    init_centers = []
    row = 0
    while len(init_centers) < n:
        shift = 0.5 if row % 2 == 1 else 0.0
        col = 0
        while len(init_centers) < n:
            init_centers.append([shift + col, row * np.sqrt(3)/2])
            col += 1
        row += 1
        
    init_centers = np.array(init_centers[:n])
    # Normalize to [0,1]
    cx, cy = init_centers[:, 0], init_centers[:, 1]
    init_centers[:, 0] = (cx - cx.min()) / (cx.max() - cx.min() + 1e-9)
    init_centers[:, 1] = (cy - cy.min()) / (cy.max() - cy.min() + 1e-9)
    
    # Add perturbation to break symmetry and allow exploration
    init_centers += np.random.uniform(-0.02, 0.02, init_centers.shape)
    init_centers = np.clip(init_centers, 0.05, 0.95)
    
    # 2. Global Optimization (Differential Evolution)
    bounds = [(0.0, 1.0)] * (2 * n)
    res_de = differential_evolution(
        objective_1d, bounds, popsize=10, maxiter=50, seed=42, tol=1e-4, 
        mutation=(0.5, 1.5), recombination=0.7
    )
    
    # 3. Local Refinement (Nelder-Mead)
    res_local = minimize(
        objective_1d, res_de.x, method='Nelder-Mead', 
        options={'maxiter': 500, 'xatol': 1e-7, 'fatol': 1e-7}
    )
    
    best_centers = np.clip(res_local.x.reshape(-1, 2), 0, 1)
    total_r, radii = compute_max_radii(best_centers)
    
    return best_centers, radii, float(total_r)
