import numpy as np
from scipy.optimize import minimize

def compute_loss(vars, n, w):
    """
    Computes the objective function with penalty terms for overlaps and boundary violations.
    vars: flattened array of [x0, y0, r0, x1, y1, r1, ...]
    n: number of circles
    w: penalty weight
    """
    vars = np.array(vars).reshape(n, 3)
    centers = vars[:, :2]
    radii = vars[:, 2]
    
    # Primary objective: maximize sum of radii (minimize negative sum)
    loss = -np.sum(radii)
    
    # Boundary penalty: circles must stay within [0,1]x[0,1]
    dist_b = np.minimum(np.minimum(centers[:, 0], 1 - centers[:, 0]),
                        np.minimum(centers[:, 1], 1 - centers[:, 1]))
    loss += w * np.sum(np.maximum(0, radii - dist_b)**2)
    
    # Pairwise overlap penalty
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist = np.linalg.norm(diff, axis=2)
    np.fill_diagonal(dist, np.inf)
    
    r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    overlap = np.maximum(0, r_sum - dist)
    loss += w * np.sum(overlap**2)
    
    return loss

def run_packing():
    best_loss = np.inf
    best_vars = None
    n = 26
    
    # Multiple restarts to explore configuration space and avoid local minima
    for seed in range(20):
        rng = np.random.RandomState(seed)
        
        # Hexagonal grid initialization perturbed by small noise
        pts = []
        dy = 0.175
        dx = 0.2
        for i in range(6):
            for j in range(6):
                if len(pts) < n:
                    y = 0.1 + i * dy
                    x = 0.1 + j * dx + (i % 2) * dx / 2
                    if x <= 0.9:
                        pts.append([x, y])
                        
        centers = np.array(pts[:n]) + rng.randn(n, 2) * 0.03
        centers = np.clip(centers, 0.1, 0.9)
        radii = np.ones(n) * 0.06
        
        x0 = np.hstack([centers, radii.reshape(n, 1)]).flatten()
        bnds = [(0, 1) for _ in range(2 * n)] + [(1e-4, 0.5) for _ in range(n)]
        
        # Stage 1: Moderate penalty for fast convergence to feasible region
        res1 = minimize(compute_loss, x0, args=(n, 2000.0), method='L-BFGS-B', bounds=bnds,
                        options={'maxiter': 2000, 'ftol': 1e-9})
        
        # Stage 2: High penalty to strictly satisfy constraints and polish solution
        res2 = minimize(compute_loss, res1.x, args=(n, 10000.0), method='L-BFGS-B', bounds=bnds,
                        options={'maxiter': 2000, 'ftol': 1e-10})
        
        if res2.fun < best_loss:
            best_loss = res2.fun
            best_vars = res2.x
            
    # Extract best configuration
    best_vars = np.array(best_vars).reshape(n, 3)
    centers = best_vars[:, :2]
    radii = best_vars[:, 2]
    
    # Deterministic repair pass to guarantee strict validity within numerical tolerance
    for _ in range(10):
        changed = False
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            m = min(x, 1-x, y, 1-y)
            if r > m + 1e-10:
                radii[i] = m
                changed = True
                
        for i in range(n):
            for j in range(i+1, n):
                d = np.linalg.norm(centers[i] - centers[j])
                if radii[i] + radii[j] > d + 1e-10:
                    diff = radii[i] + radii[j] - d
                    radii[i] -= diff / 2
                    radii[j] -= diff / 2
                    radii[i] = max(0.0, radii[i])
                    radii[j] = max(0.0, radii[j])
                    changed = True
        if not changed:
            break
            
    sum_r = np.sum(radii)
    return centers, radii, sum_r
