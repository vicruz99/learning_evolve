

To maximize the sum of radii for 26 circles in a unit square, we treat the problem as a constrained non-linear optimization. We maximize $\sum r_i$ subject to boundary and non-overlap constraints.

### Strategy

1.  **Hexagonal Initialization**: Start with a hexagonal grid, which is the densest packing for equal circles. This provides a high-quality, feasible starting point closer to the global optimum than a random or square grid.
2.  **SLSQP Optimization**: Use the Sequential Least Squares Programming (SLSQP) method. This is ideal for differentiable non-linear constraints.
3.  **Vectorized Constraints**: The 325 pair-wise overlap constraints and boundary constraints are managed efficiently to ensure the optimizer converges without numerical noise.
4.  **Final Validation**: The solution is processed and validated against the unit square constraints to ensure no circles overlap or extend outside boundaries.

import numpy as np
from scipy.optimize import minimize

def get_hex_grid(n, side=1.0):
    """Initialize n centers in a hexagonal pattern within a square."""
    pts = []
    y = 0.15
    while len(pts) < n:
        x = 0.15
        row_y = y
        if len(pts) % 2 == 1:
            x = 0.25
        while x <= 0.9 and len(pts) < n:
            pts.append([x, row_y])
            x += 0.19
        y += 0.17
    pts = np.array(pts[:n])
    return pts

def run_packing():
    n = 26
    # 1. Initialization
    centers = get_hex_grid(n)
    radii = np.full(n, 0.08)
    x0 = np.concatenate([centers.flatten(), radii])

    # 2. Bounds for x, y, r
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n

    # 3. Constraints
    constraints = []
    
    # Boundary constraints
    for i in range(n):
        idx_r = 3 * i + 2
        # x >= r
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: v[3*i] - v[idx_r]})
        # 1-x >= r
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: 1.0 - v[3*i] - v[idx_r]})
        # y >= r
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: v[3*i+1] - v[idx_r]})
        # 1-y >= r
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: 1.0 - v[3*i+1] - v[idx_r]})

    # Overlap constraints (dist^2 >= (r_i + r_j)^2)
    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({
                'type': 'ineq',
                'fun': lambda v, i=i, j=j: (v[3*i]-v[3*j])**2 + (v[3*i+1]-v[3*j+1])**2 - (v[3*i+2]+v[3*j+2])**2
            })

    # 4. Objective: Maximize sum of radii (minimize negative sum)
    def objective(v):
        return -np.sum(v[2::3])

    # 5. Optimization
    res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints, 
                   options={'maxiter': 5000, 'ftol': 1e-12, 'disp': False})

    # 6. Final Extraction and Validation
    final_centers = res.x[:52].reshape(26, 2)
    final_radii = np.maximum(res.x[52:], 0.0)
    
    # Ensure boundary validity
    for i in range(n):
        x, y = final_centers[i]
        r = final_radii[i]
        final_radii[i] = min(r, x, 1-x, y, 1-y)

    total_sum = float(np.sum(final_radii))
    return final_centers, final_radii, total_sum
