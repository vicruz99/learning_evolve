from scipy.optimize import minimize

def run_packing():

    # Create a hexagonal pattern that can support n=26 circles
    points = []
    row_spacing = 0.15  # Initial spacing
    col_spacing = 0.15
    
    # Generate a sufficiently large grid to pick 26 points
    # Rows: y = 0, row_spacing, 2*row_spacing...
    # Columns: x = 0, col_spacing, 2*col_spacing... (shifted by 0.5*col_spacing for even rows)
    rows = 6
    cols = 5
    
    for r_idx in range(rows):
        y = r_idx * row_spacing
        for c_idx in range(cols):
            x = c_idx * col_spacing
            if r_idx % 2 == 1:
                x += col_spacing / 2.0
            points.append([x, y])
            if len(points) >= n:
                break
        if len(points) >= n:
            break
            
    # Scale points to center around 0.5, 0.5 and fit roughly in [0,1]
    points = np.array(points[:n])
    center = np.mean(points, axis=0)
    points -= center
    points += [0.5, 0.5]
    
    initial_positions = points.flatten()

    # 2. Penalty Function for Fixed Radius r
    def objective_function(variables, r):

        penalty = 0.0

        for i in range(n):
            x, y = centers[i]
            if x < r: penalty += (r - x)**2
            elif x > 1 - r: penalty += (x - (1 - r))**2
            if y < r: penalty += (r - y)**2
            elif y > 1 - r: penalty += (y - (1 - r))**2

        for i in range(n):

                if dist < min_dist:
                    penalty += (min_dist - dist)**2
                    
        return penalty

    final_centers = best_positions.reshape(-1, 2)
