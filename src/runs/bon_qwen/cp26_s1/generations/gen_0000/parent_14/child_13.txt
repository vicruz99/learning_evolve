            x = x_start + i * 2 * r_start
            # Stagger rows by r_start (half diameter)
            if row_counts.index(r_count) % 2 == 1:
                x += r_start
            
            centers[idx] = [x, y]
            radii[idx] = r_start
            idx += 1

        penalty = 0.0

        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x < r: penalty += 1000 * (r - x)**2
            if x > 1 - r: penalty += 1000 * (x - (1 - r))**2
            if y < r: penalty += 1000 * (r - y)**2
            if y > 1 - r: penalty += 1000 * (y - (1 - r))**2

            for j in range(i + 1, n):
                dist = np.linalg.norm(centers[i] - centers[j])
                min_dist = radii[i] + radii[j]

                    penalty += 1000 * (min_dist - dist)**2
                    
        return score - penalty

    # Initial score

    step_size_radius = 0.005

        return True
