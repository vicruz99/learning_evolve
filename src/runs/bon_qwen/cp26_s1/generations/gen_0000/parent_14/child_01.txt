from scipy.optimize import minimize

def run_packing():

    best_sum = 0.0
    best_centers = None
    best_radii = None

    def objective(x_vars):
        # Variables: [x0, y0, ..., xn-1, yn-1, r0, ..., rn-1]
        # x_vars shape: 2n + n
        radii = x_vars[2 * n:]
        return -np.sum(radii)

    def boundary_constraints(x_vars):
        c = x_vars[:2 * n].reshape((n, 2))
        r = x_vars[2 * n:]
        constraints = []
        for i in range(n):
            constraints.append(c[i, 0] - r[i])       # x - r >= 0
            constraints.append(1 - c[i, 0] - r[i])   # 1 - x - r >= 0
            constraints.append(c[i, 1] - r[i])       # y - r >= 0
            constraints.append(1 - c[i, 1] - r[i])   # 1 - y - r >= 0
            constraints.append(r[i])                 # r >= 0
        return np.array(constraints)

    def overlap_constraints(x_vars):
        c = x_vars[:2 * n].reshape((n, 2))
        r = x_vars[2 * n:]
        constraints = []
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((c[i] - c[j]) ** 2))
                constraints.append(dist - r[i] - r[j])
        return np.array(constraints)

    cons = [
        {'type': 'ineq', 'fun': boundary_constraints},
        {'type': 'ineq', 'fun': overlap_constraints}
    ]

    bounds = []
    for _ in range(n):
        bounds.extend([(0.0, 1.0), (0.0, 1.0), (1e-5, 0.5)])

    def create_hex_grid():
        # Hexagonal lattice with 6 rows
        centers = []
        for row in range(6):
            if row % 2 == 0:
                count = 5
                start_x = 0.1
            else:
                count = 5
                start_x = 0.2
            y = 0.1 + row * 0.17
            for i in range(count):
                x = start_x + i * 0.2
                if x <= 0.95:
                    centers.append([x, y])
        return np.array(centers)

    def create_square_grid():
        centers = []
        for i in range(5):
            for j in range(5):
                centers.append([0.1 + i * 0.2, 0.1 + j * 0.2])
        centers.append([0.2, 0.2])
        return np.array(centers)

                    best_centers = result.x[:2 * n].reshape((n, 2))
                    best_radii = result.x[2 * n:]

        except Exception:
            continue

    # Ensure no negative radii due to numerical drift
    if best_radii is not None:
        best_radii = np.maximum(best_radii, 0.0)

    return best_centers, best_radii, best_sum
