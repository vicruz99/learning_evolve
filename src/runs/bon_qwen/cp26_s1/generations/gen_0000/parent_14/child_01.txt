

Looking at this circle packing problem, I need to maximize the sum of radii of 26 circles in a unit square. Let me think through the strategy systematically.

<strategy>
1. **Problem Analysis**: We're maximizing sum of radii (not minimizing radius of equal circles). This allows unequal circles, which is key. For 26 circles, a good hexagonal arrangement should give sum ≈ 5.0+.

2. **Initialization**: Use hexagonal packing pattern as starting point - rows offset by r, spaced by √3·r vertically. This is near-optimal for dense packings.

3. **Optimization**: Use scipy.optimize.minimize with L-BFGS-B (supports bounds). The objective maximizes sum of radii with heavy quadratic penalties for overlaps and boundary violations.

4. **Multiple Restarts**: Try different seeds and initialization patterns (hexagonal grid, random perturbations) to escape local optima.

5. **Two-Phase Approach**: 
   - Phase 1: Optimize positions and radii together with penalty method
   - Phase 2: For fixed centers, solve LP to get optimal radii (linear constraints)

6. **Post-processing**: After optimization, iteratively reduce any overlapping radii to ensure validity, then optionally grow circles in available gaps.

7. **Key insight for post-optimization**: After getting a valid packing, circles in gaps can be grown. Use a "grow" phase where each circle expands to its maximum possible radius given all constraints.
</strategy>

import numpy as np
from scipy.optimize import minimize, linprog


def compute_hexagonal_centers(n, r):
    """Generate hexagonal packing centers for n circles with initial radius r."""
    centers = []
    row = 0
    while len(centers) < n:
        y = r + row * r * np.sqrt(3)
        if y + r > 1 + 1e-10:
            y = min(y, 1 - r)
        offset = r if row % 2 == 1 else 0
        col = 0
        while len(centers) < n:
            x = r + col * 2 * r + offset
            if x + r > 1 + 1e-10:
                col += 1
                continue
            centers.append([x, y])
            col += 1
        row += 1
    return np.array(centers[:n])


def optimize_radii_given_centers(centers, n):
    """Given fixed centers, solve LP to find optimal radii maximizing sum."""
    c_obj = -np.ones(n)
    A_ub = []
    b_ub = []

    for i in range(n):
        for j in range(i + 1, n):
            d_ij = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            row = np.zeros(n)
            row[i] = 1
            row[j] = 1
            A_ub.append(row)
            b_ub.append(d_ij)

    for i in range(n):
        b_i = min(centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])
        row = np.zeros(n)
        row[i] = 1
        A_ub.append(row)
        b_ub.append(max(b_i, 0))

    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0, None)] * n

    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x
    return np.zeros(n)


def penalty_objective(params, n, penalty_weight=5000):
    """Objective: maximize sum of radii with penalty for constraint violations."""
    c = params[:n * 2].reshape(n, 2)
    r = params[n * 2:]

    obj = -np.sum(r)

    for i in range(n):
        x, y = c[i]
        ri = r[i]
        if ri < 0:
            ri = 0
        obj += penalty_weight * max(0, ri - x) ** 2
        obj += penalty_weight * max(0, x + ri - 1) ** 2
        obj += penalty_weight * max(0, ri - y) ** 2
        obj += penalty_weight * max(0, y + ri - 1) ** 2

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((c[i] - c[j]) ** 2) + 1e-20)
            ri, rj = r[i], r[j]
            overlap = ri + rj - dist
            if overlap > 0:
                obj += penalty_weight * overlap ** 2

    return obj


def postprocess_valid(centers, radii, n):
    """Ensure packing is valid by reducing radii if needed, then grow in gaps."""
    for iteration in range(50):
        changed = False
        for i in range(n):
            r = radii[i]
            x, y = centers[i]
            r = min(r, x, 1 - x, y, 1 - y)
            for j in range(n):
                if i != j:
                    dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                    r = min(r, dist - radii[j])
            if r < 0:
                r = 0
            if abs(r - radii[i]) > 1e-12:
                radii[i] = r
                changed = True
        if not changed:
            break
    return radii


def try_optimization(centers_init, radii_init, n, seed=0):
    """Run optimization from given initial configuration."""
    np.random.seed(seed)
    params_init = np.concatenate([centers_init.flatten(), radii_init])

    bounds = [(0, 1)] * (2 * n) + [(0, 0.5)] * n

    result = minimize(
        penalty_objective,
        params_init,
        args=(n, 5000),
        method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': 50000, 'ftol': 1e-15, 'gtol': 1e-10}
    )

    centers_opt = result.x[:n * 2].reshape(n, 2)
    radii_opt = np.maximum(result.x[n * 2:], 0)

    # Clip centers to valid range
    centers_opt = np.clip(centers_opt, 0.001, 0.999)

    # Post-process to ensure validity
    radii_opt = postprocess_valid(centers_opt, radii_opt, n)

    # Grow radii in available gaps
    for iteration in range(20):
        improved = False
        for i in range(n):
            x, y = centers_opt[i]
            max_r = min(x, 1 - x, y, 1 - y)
            for j in range(n):
                if i != j:
                    dist = np.sqrt(np.sum((centers_opt[i] - centers_opt[j]) ** 2))
                    max_r = min(max_r, dist - radii_opt[j])
            new_r = max(0, max_r)
            if new_r > radii_opt[i] + 1e-12:
                radii_opt[i] = new_r
                improved = True
        if not improved:
            break

    return centers_opt, radii_opt, np.sum(radii_opt)


def run_packing():

    best_sum = -1

    best_radii = None

    configs = []

    for r_try in [0.15, 0.17, 0.18, 0.19, 0.20, 0.22]:

        radii_hex = np.ones(n) * 0.1
        configs.append((centers_hex.copy(), radii_hex.copy()))

    # Random initializations
    for seed in range(10):
        np.random.seed(seed)
        centers_rand = np.random.uniform(0.15, 0.85, (n, 2))
        radii_rand = np.random.uniform(0.05, 0.15, n)
        configs.append((centers_rand, radii_rand))

    # Grid-based initializations
    for spacing in [0.20, 0.22, 0.25]:
        centers = []
        idx = 0
        for i in range(6):
            for j in range(6):
                if idx >= n:
                    break
                x = spacing + i * spacing
                y = spacing + j * spacing
                if x <= 0.95 and y <= 0.95:
                    centers.append([x, y])
                    idx += 1
            if idx >= n:
                break
        while len(centers) < n:
            centers.append([0.5, 0.5])
        centers_arr = np.array(centers[:n])
        radii_arr = np.ones(n) * 0.1
        configs.append((centers_arr, radii_arr))

    # Run optimization for each configuration
    for cfg_idx, (c_init, r_init) in enumerate(configs):
        centers_opt, radii_opt, current_sum = try_optimization(
            c_init, r_init, n, seed=cfg_idx * 137

            best_sum = current_sum
            best_centers = centers_opt.copy()
            best_radii = radii_opt.copy()

    for perturbation in range(10):
        np.random.seed(42 + perturbation * 73)
        c_pert = best_centers + np.random.uniform(-0.005, 0.005, (n, 2))
        c_pert = np.clip(c_pert, 0.01, 0.99)

            c_pert, r_pert, n, seed=1000 + perturbation * 31

        if current_sum > best_sum:

            best_centers = centers_opt.copy()
            best_radii = radii_opt.copy()

    best_radii = postprocess_valid(best_centers, best_radii, n)

    for iteration in range(100):
        improved = False

            x, y = best_centers[i]
            max_r = min(x, 1 - x, y, 1 - y)

            if new_r > best_radii[i] + 1e-12:
                best_radii[i] = new_r
                improved = True

    final_sum = np.sum(best_radii)
    return best_centers, best_radii, final_sum
