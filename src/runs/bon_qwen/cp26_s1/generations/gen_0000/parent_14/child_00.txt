    while y <= 1.0 - r_init:
        x = r_init
        if row % 2 == 1:
            x += d_init / 2.0
        while x <= 1.0 - r_init:
            centers.append([x, y])
            x += d_init
        y += d_init * math.sqrt(3) / 2.0
        row += 1
        
    # If grid doesn't generate enough points, fill with random points
    if len(centers) < N:
        random_pts = np.random.rand(N - len(centers), 2) * 0.8 + 0.1
        centers.extend(random_pts)
    
    centers = np.array(centers[:N])
    
    # 2. Simulated Annealing to maximize min distance (score)
    # This optimizes the positions to maximize the minimum clearance,
    # which corresponds to maximizing the radius of equal circles.
    current_centers = centers.copy()
    
    T = 0.05 # Initial temperature
    T_min = 1e-7
    decay = 0.9995
    
    for step in range(40000):
        if T < T_min:
            break
            
        # Pick random circle to move
        i = np.random.randint(0, N)
        
        # Propose move
        step_size = T * 0.3
        move = np.random.randn(2) * step_size
        new_centers = current_centers.copy()
        new_centers[i] += move
        new_centers[i] = np.clip(new_centers[i], 0.0, 1.0)
        
        score_new = get_score(new_centers, N)
        score_curr = get_score(current_centers, N)
        
        delta = score_new - score_curr
        
        # Accept if better or with probability based on temperature
        if delta > 0 or np.random.rand() < math.exp(delta / (T + 1e-9)):
            current_centers = new_centers
    
    # 3. Local Optimization (Hill Climbing) to polish
    # This phase greedily pushes circles apart to resolve tight constraints.
    for _ in range(10000):
        # Identify the most critical constraint (closest pair or boundary)
        min_d_val = 2.0
        i_crit, j_crit = -1, -1
        is_bound = False
        
        # Check boundary constraints
        for i in range(N):
            x, y = current_centers[i]
            d = 2.0 * min(x, 1-x, y, 1-y)
            if d < min_d_val:
                min_d_val = d
                i_crit = i
                is_bound = True
        
        # Check inter-circle constraints
        for i in range(N):
            for j in range(i + 1, N):
                dist = np.sqrt(np.sum((current_centers[i] - current_centers[j]) ** 2))
                if dist < min_d_val:
                    min_d_val = dist
                    i_crit = i
                    j_crit = j
                    is_bound = False
        
        # Attempt to improve the critical constraint
        if is_bound:
            i = i_crit
            x, y = current_centers[i]
            # Push away from the closest boundary
            dists = [x, 1-x, y, 1-y]
            idx = np.argmin(dists)
            move = 0.0002
            dx, dy = 0, 0
            if idx == 0: dx = move      # Move Right
            elif idx == 1: dx = -move   # Move Left
            elif idx == 2: dy = move    # Move Up
            else: dy = -move            # Move Down
            
            current_centers[i] += [dx, dy]
            current_centers[i] = np.clip(current_centers[i], 1e-9, 1-1e-9)
        else:
            i, j = i_crit, j_crit
            vec = current_centers[i] - current_centers[j]
            dist = np.linalg.norm(vec)
            if dist > 1e-9:
                vec /= dist
                move = 0.0002
                current_centers[i] += vec * move
                current_centers[j] -= vec * move
                current_centers[i] = np.clip(current_centers[i], 1e-9, 1-1e-9)
                current_centers[j] = np.clip(current_centers[j], 1e-9, 1-1e-9)
            else:
                # If centers are coincident, random push to escape singularity
                current_centers[i] += (np.random.rand(2) - 0.5) * 0.001
                current_centers[j] += (np.random.rand(2) - 0.5) * 0.001

    centers = current_centers
    
    # 4. Compute Radii based on final positions
    # Each circle's radius is limited by the nearest neighbor and boundaries.
    radii = np.zeros(N)
    for i in range(N):
        x, y = centers[i]
        # Constraint from boundaries
        r = min(x, 1-x, y, 1-y)
        # Constraint from other circles
        for j in range(N):
            if i == j: continue
            d = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            r = min(r, d / 2.0)
        radii[i] = r
        
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
