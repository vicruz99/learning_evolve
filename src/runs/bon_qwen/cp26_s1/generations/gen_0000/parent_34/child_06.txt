import numpy as np
from scipy.optimize import minimize
import itertools

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Pack 26 circles in a unit square to maximize the sum of radii.
    
    Returns:
        centers: np.array of shape (26, 2)
        radii: np.array of shape (26,)
        sum_radii: float
    """
    n = 26
    
    # Objective function: minimize negative sum of radii
    def objective(v):
        r = v[2::3]
        return -np.sum(r)

    # Constraints function: returns a vector of constraint values (must be >= 0)
    def constraints_func(v):
        x = v[0::3]
        y = v[1::3]
        r = v[2::3]
        
        # 1. Boundary Constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
        # Reshape to (n,)
        cons_boundary = np.concatenate([
            x - r,
            1.0 - x - r,
            y - r,
            1.0 - y - r
        ])
        
        # 2. Pairwise Non-overlap Constraints: dist^2 >= (r_i + r_j)^2
        # Using squared distance to avoid sqrt and maintain smoothness where possible
        # dist^2 - (r_i + r_j)^2 >= 0
        # Compute pairwise squared distances
        # x_diff = x[:, None] - x[None, :] ... but let's do it efficiently
        # Using broadcasting
        xx = x[:, np.newaxis]
        yy = y[:, np.newaxis]
        dist_sq = (xx - xx.T)**2 + (yy - yy.T)**2
        
        # Compute (r_i + r_j)^2
        rr = r[:, np.newaxis]
        rad_sum_sq = (rr + rr.T)**2
        
        # Extract upper triangle (strictly upper) to avoid duplicate and diagonal constraints
        tri_mask = np.triu(np.ones((n, n), dtype=bool), k=1)
        pairwise_cons = dist_sq[tri_mask] - rad_sum_sq[tri_mask]
        
        # Combine all constraints
        return np.concatenate([cons_boundary, pairwise_cons])

    # Bounds for variables: x, y in [0, 1], r in [0, 0.5]
    # x, y are implicitly constrained by boundary inequalities, but [0,1] is a safe box
    bounds = [(0, 1), (0, 1), (0, 0.5)] * n

    # Initial Guess Generation (Hexagonal Grid)
    def get_initial_guess(scale_factor=1.0):
        # Hexagonal packing pattern
        centers = []
        r_init = 0.08 * scale_factor
        
        row_y = 0.1 * scale_factor
        while row_y <= 0.9 * scale_factor:
            # Determine row offset (alternating)
            if int((row_y - 0.1 * scale_factor) / (0.1 * np.sqrt(3))) % 2 == 1:
                offset = 0.1 * scale_factor
            else:
                offset = 0.0
            
            row_x = 0.1 * scale_factor + offset
            while row_x <= 0.9 * scale_factor:
                centers.append((row_x, row_y, r_init))
                row_x += 0.2 * scale_factor
            row_y += 0.1 * np.sqrt(3) * scale_factor
            
            if len(centers) >= 26:
                break
        
        # If we don't have enough, fill with random
        if len(centers) < 26:
            while len(centers) < 26:
                cx = np.random.uniform(0.1, 0.9) * scale_factor
                cy = np.random.uniform(0.1, 0.9) * scale_factor
                centers.append((cx, cy, r_init))
        
        # Take first 26
        centers = centers[:26]
        # Construct vector
        v0 = np.zeros(26 * 3)
        for i, (cx, cy, cr) in enumerate(centers):
            v0[3*i] = cx
            v0[3*i+1] = cy
            v0[3*i+2] = cr
        return v0

    best_result = None
    best_score = -np.inf

    # Run optimization multiple times with different initial perturbations
    # to escape local minima
    num_tries = 20
    
    for i in range(num_tries):
        # Perturb initial guess slightly
        v0 = get_initial_guess()
        noise = np.random.normal(0, 0.02, size=v0.shape)
        # Apply noise but keep r positive and reasonable
        v0 += noise
        v0[2::3] = np.clip(v0[2::3], 0.01, 0.5) # Keep radii reasonable
        v0[0::3] = np.clip(v0[0::3], 0.01, 0.99)
        v0[1::3] = np.clip(v0[1::3], 0.01, 0.99)

        cons = {'type': 'ineq', 'fun': constraints_func}
        
        try:
            res = minimize(objective, v0, method='SLSQP', bounds=bounds, 
                           constraints=cons, options={'maxiter': 2000, 'ftol': 1e-9})
            
            if res.success:
                sum_r = -res.fun
                if sum_r > best_score:
                    best_score = sum_r
                    best_result = res.x
        except Exception:
            continue

    if best_result is None:
        # Fallback if optimization failed (should not happen often)
        v0 = get_initial_guess()
        best_result = v0
        best_score = np.sum(best_result[2::3])

    # Extract centers and radii
    centers = best_result.reshape(n, 3)[:, :2]
    radii = best_result.reshape(n, 3)[:, 2]
    sum_radii = np.sum(radii)

    return centers, radii, sum_radii
