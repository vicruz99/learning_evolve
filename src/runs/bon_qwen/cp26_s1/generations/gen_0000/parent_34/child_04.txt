            {'type': 'ineq', 'fun': lambda v: (1 - v[n:2*n]) - v[:n]}, # 1 - x_i - r_i >= 0
            {'type': 'ineq', 'fun': lambda v: v[2*n:3*n] - v[:n]}, # y_i - r_i >= 0

                    dy = y[i] - y[j]

                    rad_sum = r[i] + r[j]
