def compute_min_radius(centers):
    """Compute maximum equal radius that fits given centers."""
    n = len(centers)
    min_r = 1.0
    
    # Boundary constraints
    for i in range(n):
        min_r = min(min_r, centers[i, 0], 1 - centers[i, 0],
                     centers[i, 1], 1 - centers[i, 1])
    
    # Overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            min_r = min(min_r, dist / 2)
    
    return min_r


def obj_min_radius(x, n):
    """Objective for equal circles: maximize minimum radius."""
    centers = x.reshape(n, 2)
    return -compute_min_radius(centers)


def make_hex_init(n, seed, row_pattern=None):

            centers[idx] = [x + np.random.uniform(-0.005, 0.005),
                            y + np.random.uniform(-0.005, 0.005)]
            idx += 1
    
    return centers


def validate(centers, radii, tol=1e-9):

            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
