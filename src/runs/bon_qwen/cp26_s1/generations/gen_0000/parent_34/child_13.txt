import numpy as np
import scipy.optimize as opt

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    """
    n_circles = 26
    
    # 1. Initialization: Hexagonal Lattice
    # We arrange points in a hexagonal grid pattern scaled to fit in [0,1]x[0,1]
    # This provides a much better starting point than random noise.
    
    # Estimate initial radius based on density for 26 circles
    # Density ~ 0.9, Area = 1. 26 * pi * r^2 approx 0.9 => r approx 0.105
    # We start slightly smaller to ensure valid initialization
    init_r = 0.05 
    
    # Generate hexagonal grid points
    # Rows and columns based on spacing
    # Spacing in hex grid is 2*r for horizontal, sqrt(3)*r for vertical
    # But we don't know r yet. Let's just generate a fixed shape grid and scale later?
    # Or just generate points for a specific r and scale to fit.
    
    # Let's try to fit a 5x5 grid roughly, but perturbed.
    # 5x5 = 25. We need 26.
    # Maybe 4 rows of 7? No, too wide.
    # Let's generate a rectangular grid of 6x5 = 30 points, remove 4, or just place 26.
    # Better: Generate points for a hexagonal lattice with unit spacing, 
    # then fit into the box.
    
    # Generate a large hexagonal grid
    pts = []
    # Approximate number of points needed
    # If we use spacing 1.0, we can fit ~1x1. 
    # Let's just use a dense grid and trim.
    
    # Let's create a 6x5 grid (30 points)
    # Shift every other row
    rows = 6
    cols = 5
    pts_x = []
    pts_y = []
    
    # Spacing parameters for initial layout
    sx = 1.0 / 4.5 # approx spacing
    sy = 1.0 / 4.5 
    
    for r_idx in range(rows):
        for c_idx in range(cols):
            x = c_idx * sx + (r_idx % 2) * (sx / 2)
            y = r_idx * sy
            pts_x.append(x)
            pts_y.append(y)
            
    pts_x = np.array(pts_x)
    pts_y = np.array(pts_y)
    
    # We have 30 points. We need 26.
    # Keep the first 26.
    pts_x = pts_x[:n_circles]
    pts_y = pts_y[:n_circles]
    
    # Normalize to fit inside [0,1] with some margin
    # Actually, the optimizer will move them. 
    # Just ensure they are within [0,1].
    pts_x = np.clip(pts_x, 0.05, 0.95)
    pts_y = np.clip(pts_y, 0.05, 0.95)
    
    # Initial radii
    radii_init = np.full(n_circles, init_r)
    
    # Combine variables: x1, y1, r1, x2, y2, r2, ...
    x0 = np.zeros(3 * n_circles)
    for i in range(n_circles):
        x0[3*i] = pts_x[i]
        x0[3*i+1] = pts_y[i]
        x0[3*i+2] = radii_init[i]
        
    # 2. Optimization Setup
    
    # Penalty weight for constraint violations
    # High penalty ensures constraints are respected
    penalty_weight = 1000.0
    
    def objective(variables):
        """
        Minimizes negative sum of radii + penalty for overlaps/boundary violations.
        """
        vars_3d = variables.reshape(-1, 3) # Each row is [x, y, r]
        sum_radii = np.sum(vars_3d[:, 2])
        
        penalty = 0.0
        
        # Boundary penalties
        # x - r >= 0  => r - x <= 0
        # 1 - x - r >= 0 => x + r - 1 <= 0
        # Same for y
        # We penalize if constraint is violated (value > 0)
        
        # Left wall: r - x
        viol_left = np.maximum(0, vars_3d[:, 2] - vars_3d[:, 0])
        # Right wall: x + r - 1
        viol_right = np.maximum(0, vars_3d[:, 0] + vars_3d[:, 2] - 1.0)
        # Bottom wall: r - y
        viol_bottom = np.maximum(0, vars_3d[:, 2] - vars_3d[:, 1])
        # Top wall: y + r - 1
        viol_top = np.maximum(0, vars_3d[:, 1] + vars_3d[:, 2] - 1.0)
        
        wall_violations = viol_left + viol_right + viol_bottom + viol_top
        penalty += penalty_weight * np.sum(wall_violations ** 2)
        
        # Inter-circle overlap penalties
        # dist(i, j) >= r_i + r_j
        # (x_i - x_j)^2 + (y_i - y_j)^2 >= (r_i + r_j)^2
        # Violation if (r_i + r_j)^2 - dist^2 > 0
        
        # Vectorized distance calculation
        # Centers: (N, 2), Radii: (N, 1)
        centers = vars_3d[:, :2]
        radii = vars_3d[:, 2:3]
        
        # Pairwise differences
        # This is O(N^2), N=26 is small (676 pairs)
        diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
        dist_sq = np.sum(diff**2, axis=2)
        
        # Sum of radii matrix
        sum_radii_mat = radii + radii.T
        
        # Avoid self-interaction (diagonal)
        np.fill_diagonal(dist_sq, np.inf) 
        np.fill_diagonal(sum_radii_mat, -np.inf)
        
        # Violation condition: (r_i + r_j)^2 > dist_sq
        # We want to penalize (r_i + r_j - dist)^2 if r_i + r_j > dist
        # Or simpler: max(0, (r_i + r_j)^2 - dist_sq)
        
        # Calculate squared radii sum
        r_sum_sq = sum_radii_mat ** 2
        
        # Violation amount
        violations = r_sum_sq - dist_sq
        
        # Only positive values are violations
        violations = np.maximum(0, violations)
        
        # Add to penalty (upper triangle only to avoid double counting)
        triu_indices = np.triu_indices(n_circles, k=1)
        penalty += penalty_weight * np.sum(violations[triu_indices])
        
        return -sum_radii + penalty

    # Bounds for variables: x in [0,1], y in [0,1], r in [0, 0.5]
    bounds = []
    for _ in range(n_circles):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 0.5)) # r

    # Run optimization
    # L-BFGS-B is good for bound constrained problems
    # We might need multiple restarts or a good initial guess.
    # The hexagonal start should be good.
    
    res = opt.minimize(
        objective, 
        x0, 
        method='L-BFGS-B', 
        bounds=bounds,
        options={'maxiter': 5000, 'ftol': 1e-9, 'gtol': 1e-6}
    )
    
    # Extract results
    optimal_vars = res.x.reshape(-1, 3)
    centers = optimal_vars[:, :2]
    radii = optimal_vars[:, 2]
    
    # Post-processing: Clean up tiny negative radii or NaNs if any
    radii = np.maximum(radii, 1e-9)
    
    # Calculate final sum
    total_sum = np.sum(radii)
    
    return centers, radii, total_sum
