import numpy as np
from scipy.optimize import minimize

def objective(v, n, mu):
    """
    Computes the objective value with penalty terms for constraint violations.
    Objective: Maximize sum of radii (minimize negative sum) + penalty for overlaps/boundary exits.
    """
    c = v[:2*n].reshape(n, 2)
    r = v[2*n:]
    penalty = 0.0
    
    # Boundary penalties
    for i in range(n):
        cx, cy = c[i]
        ri = r[i]
        # Check left boundary
        if cx < ri: penalty += mu * (cx - ri)**2
        # Check right boundary
        if 1.0 - cx < ri: penalty += mu * (1.0 - cx - ri)**2
        # Check bottom boundary
        if cy < ri: penalty += mu * (cy - ri)**2
        # Check top boundary
        if 1.0 - cy < ri: penalty += mu * (1.0 - cy - ri)**2
        
    # Overlap penalties
    for i in range(n):
        for j in range(i + 1, n):
            d = np.sqrt(np.sum((c[i] - c[j])**2))
            min_d = r[i] + r[j]
            if d < min_d:
                penalty += mu * (d - min_d)**2
                
    return -np.sum(r) + penalty

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    best_centers = None
    best_radii = None
    best_sum = 0.0

    # Run multiple trials with different seeds to improve global optimum chances
    for trial in range(4):
        rng = np.random.default_rng(trial + 100)
        
        # Smart initialization: Start with a grid-like configuration perturbed slightly
        # This places circles reasonably apart to help the optimizer scale up radii efficiently
        centers = np.zeros((n, 2))
        radii = np.full(n, 0.03)
        
        idx = 0
        # Fill 25 circles in a 5x5 grid
        for i in range(5):
            for j in range(5):
                centers[idx] = [0.1 + 0.2*i, 0.1 + 0.2*j]
                idx += 1
        
        # Place the 26th circle randomly
        if idx < n:
            centers[idx] = rng.uniform(0.05, 0.95, 2)
            
        x0 = np.concatenate([centers.ravel(), radii])
        bounds = [(0, 1)] * (2*n) + [(0, 0.5)] * n
        
        x = x0
        mu = 100.0
        
        # Iteratively increase penalty weight to enforce constraints strictly
        for step in range(15):
            res = minimize(objective, x, args=(n, mu), method='L-BFGS-B', bounds=bounds, 
                           options={'maxiter': 400, 'ftol': 1e-12})
            x = res.x
            mu *= 3.0
            
        final_c = x[:2*n].reshape(n, 2)
        final_r = x[2*n:]
        current_sum = np.sum(final_r)
        
        # Validate configuration strictly
        valid = True
        for i in range(n):
            ri = final_r[i]
            if ri < 0: valid = False; break
            cx, cy = final_c[i]
            if cx - ri < -1e-12 or cx + ri > 1 + 1e-12 or cy - ri < -1e-12 or cy + ri > 1 + 1e-12:
                valid = False; break
        
        if valid:
            for i in range(n):
                for j in range(i + 1, n):
                    d = np.sqrt(np.sum((final_c[i] - final_c[j])**2))
                    if d < final_r[i] + final_r[j] - 1e-12:
                        valid = False; break
                if not valid: break
                
        if valid and current_sum > best_sum:
            best_centers = final_c.copy()
            best_radii = final_r.copy()
            best_sum = current_sum
            
    return best_centers, best_radii, float(best_sum)
