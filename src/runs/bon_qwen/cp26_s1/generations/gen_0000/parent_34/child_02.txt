                    overlap = min_dist - dist
                    if dist > 1e-9:

                        forces[i, 0] += diff[0] * factor
                        forces[i, 1] += diff[1] * factor
                        forces[j, 0] -= diff[0] * factor
                        forces[j, 1] -= diff[1] * factor

                        forces[i, 1] += np.random.randn() * 0.05

        final_radii[i] = max_r

    return centers, final_radii, np.sum(final_radii)
