import numpy as np
from scipy.optimize import minimize


def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26

    def penalty_objective(x):
        centers = x[:2 * n].reshape(n, 2)
        radii = x[2 * n:]
        radii = np.maximum(radii, 1e-12)

        obj = -np.sum(radii)
        penalty_weight = 500.0

        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                min_dist = radii[i] + radii[j]
                if dist < min_dist:
                    overlap = min_dist - dist
                    obj += penalty_weight * overlap * overlap

        for i in range(n):
            x_i, y_i = centers[i]
            r_i = radii[i]
            if x_i - r_i < 0:
                obj += penalty_weight * (x_i - r_i) ** 2
            if x_i + r_i > 1:
                obj += penalty_weight * (x_i + r_i - 1) ** 2
            if y_i - r_i < 0:
                obj += penalty_weight * (y_i - r_i) ** 2
            if y_i + r_i > 1:
                obj += penalty_weight * (y_i + r_i - 1) ** 2

        return obj

    def force_based_optimize(centers_init, radii_init, n, max_iter=15000):
        centers = centers_init.copy()
        radii = radii_init.copy()

        for iteration in range(max_iter):
            forces = np.zeros_like(centers)

            for i in range(n):
                for j in range(i + 1, n):
                    dx = centers[i, 0] - centers[j, 0]
                    dy = centers[i, 1] - centers[j, 1]
                    dist_sq = dx * dx + dy * dy
                    dist = np.sqrt(dist_sq) if dist_sq > 1e-20 else 1e-10
                    min_dist = radii[i] + radii[j]

                    if dist < min_dist:
                        overlap = min_dist - dist
                        strength = overlap * 200.0
                        fx = dx / dist * strength
                        fy = dy / dist * strength
                        forces[i, 0] += fx
                        forces[i, 1] += fy
                        forces[j, 0] -= fx
                        forces[j, 1] -= fy

            for i in range(n):
                x_i, y_i = centers[i]
                r_i = radii[i]

                if x_i - r_i < 0:
                    forces[i, 0] += (r_i - x_i) * 500.0
                elif x_i + r_i > 1:
                    forces[i, 0] -= (x_i + r_i - 1) * 500.0

                if y_i - r_i < 0:
                    forces[i, 1] += (r_i - y_i) * 500.0
                elif y_i + r_i > 1:
                    forces[i, 1] -= (y_i + r_i - 1) * 500.0

            dt = 0.005 / (1.0 + iteration * 0.0001)
            centers += forces * dt

            r_vec = radii.reshape(-1, 1)
            centers[:, 0] = np.clip(centers[:, 0], r_vec[:, 0], 1.0 - r_vec[:, 0])
            centers[:, 1] = np.clip(centers[:, 1], r_vec[:, 0], 1.0 - r_vec[:, 0])

            growth = 0.00004 / (1.0 + iteration * 0.0002)
            radii += growth

        return centers, radii

    def make_grid_init(n, rows, cols, extra_x=0.05, extra_y=0.05):
        centers = []
        if rows * cols >= n:
            for r in range(rows):
                for c in range(cols):
                    if len(centers) >= n:
                        break
                    x = 1.0 / (cols + 1) * (c + 1)
                    y = 1.0 / (rows + 1) * (r + 1)
                    centers.append([x, y])
        else:
            for r in range(rows):
                for c in range(cols):
                    x = 1.0 / (cols + 1) * (c + 1)
                    y = 1.0 / (rows + 1) * (r + 1)
                    centers.append([x, y])
            while len(centers) < n:
                centers.append([extra_x, extra_y])
        return np.array(centers[:n])

    def make_hex_init(n, base_r=0.06):
        centers = []
        for row in range(8):
            y = base_r + row * np.sqrt(3) * base_r
            if row % 2 == 0:
                x_start = base_r
                n_in_row = 5
            else:
                x_start = base_r * 2.0
                n_in_row = 4
            for i in range(n_in_row):
                x = x_start + 2.0 * base_r * i
                if x + base_r <= 1.0 and len(centers) < n:
                    centers.append([x, y])
        while len(centers) < n:
            centers.append([0.1, 0.1])
        return np.array(centers[:n])

    def refine_with_scipy(centers, radii, n):
        x0 = np.concatenate([centers.flatten(), radii])
        try:
            result = minimize(
                penalty_objective, x0, method="BFGS",
                options={"maxiter": 5000, "gtol": 1e-14, "eps": 1e-8}
            )
            co = result.x[:2 * n].reshape(n, 2)
            ro = result.x[2 * n:]
            ro = np.maximum(ro, 0)
            return co, ro
        except Exception:
            return centers, radii

    best_sum = -1.0
    best_centers = None
    best_radii = None

    init_configs = [
        ("grid_5x6", make_grid_init(n, 6, 5), np.ones(n) * 0.08),
        ("grid_5x5_extra", make_grid_init(n, 5, 5, 0.05, 0.05), np.ones(n) * 0.08),
        ("grid_5x5_extra2", make_grid_init(n, 5, 5, 0.5, 0.5), np.ones(n) * 0.08),
        ("hex1", make_hex_init(n, 0.06), np.ones(n) * 0.06),
        ("hex2", make_hex_init(n, 0.07), np.ones(n) * 0.05),
        ("hex3", make_hex_init(n, 0.05), np.ones(n) * 0.07),
    ]

    for name, c_init, r_init in init_configs:
        centers_f, radii_f = force_based_optimize(c_init, r_init, n, max_iter=12000)
        centers_r, radii_r = refine_with_scipy(centers_f, radii_f, n)
        s = np.sum(radii_r)
        if s > best_sum:
            best_sum = s
            best_centers = centers_r.copy()
            best_radii = radii_r.copy()
            print(f"Config {name}: sum = {s:.6f}")

    # Additional refinement passes on the best solution
    for pass_num in range(5):
        centers_ref, radii_ref = refine_with_scipy(best_centers, best_radii, n)
        s = np.sum(radii_ref)
        if s > best_sum:
            best_sum = s
            best_centers = centers_ref.copy()
            best_radii = radii_ref.copy()
            print(f"Refinement pass {pass_num}: sum = {s:.6f}")

    # Final force-based polish
    centers_final, radii_final = force_based_optimize(
        best_centers, best_radii, n, max_iter=8000
    )
    centers_final, radii_final = refine_with_scipy(centers_final, radii_final, n)

    s_final = np.sum(radii_final)
    if s_final > best_sum:
        best_sum = s_final
        best_centers = centers_final.copy()
        best_radii = radii_final.copy()

    print(f"Final sum of radii: {best_sum:.6f}")
    return best_centers, best_radii, float(best_sum)
