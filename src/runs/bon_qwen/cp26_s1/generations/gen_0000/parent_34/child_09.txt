import numpy as np
from scipy.optimize import minimize

def compute_penalty(centers_flat, r, n):
    """Compute squared overlap penalty for N circles of radius r."""
    centers = centers_flat.reshape((n, 2))
    # Vectorized pairwise distance computation
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    d = np.sqrt(np.sum(diff**2, axis=2))
    
    # Only consider upper triangle pairs (i < j)
    triu_mask = np.triu(np.ones((n, n)), k=1)
    
    # Penalty is non-zero only when circles overlap
    overlaps = 2 * r - d
    overlaps = np.where(overlaps > 0, overlaps, 0.0)
    
    return np.sum((overlaps * triu_mask)**2)
