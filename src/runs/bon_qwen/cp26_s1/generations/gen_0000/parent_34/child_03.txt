def get_constraints(centers, radii, n):

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    # Constraints
    constraints = get_constraints(centers_init, radii_init, n)
    
    # Optimization settings
    # trust-constr is robust for constrained nonlinear optimization
    # However, it might be slow or fail if not initialized well.
    # Let's try SLSQP as well, often faster for medium size.
    
    best_result = None
    best_val = np.inf
    
    # Multi-start: try a few times
    for _ in range(3):
        # Add some noise to initial guess if not first iteration
        if _ > 0:
            x0_noisy = x0 + np.random.normal(0, 0.01, size=x0.shape)
            # Clip radii to be positive
            for k in range(2, len(x0_noisy), 3):

                x0_noisy[k+1] = np.clip(x0_noisy[k+1], 0, 1)
            x0_current = x0_noisy
        else:
            x0_current = x0
            
        try:
            res = minimize(objective, x0_current, method='trust-constr', 
                           bounds=bounds, constraints=constraints,
                           options={'verbose': 0, 'maxiter': 1000})

                    best_val = res.fun

                pass

        radii[i] = x_sol[3*i+2]
        
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
