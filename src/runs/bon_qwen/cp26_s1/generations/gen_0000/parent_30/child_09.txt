import numpy as np
from scipy.optimize import minimize

def compute_constraints(centers_radii, n):
    """
    Computes the constraint values for the optimization problem.
    Returns an array where each element must be >= 0.
    Constraints include:
    1. Non-overlap between circles: dist^2 >= (r1 + r2)^2
    2. Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    """
    constraints_list = []
    
    # Boundary constraints
    # x_i >= r_i  => x_i - r_i >= 0
    # x_i <= 1 - r_i => 1 - x_i - r_i >= 0
    # y_i >= r_i => y_i - r_i >= 0
    # y_i <= 1 - r_i => 1 - y_i - r_i >= 0
    for i in range(n):
        idx = 3 * i
        x = centers_radii[idx]
        y = centers_radii[idx + 1]
        r = centers_radii[idx + 2]
        
        constraints_list.append(x - r)
        constraints_list.append(1 - x - r)
        constraints_list.append(y - r)
        constraints_list.append(1 - y - r)
        
    # Non-overlap constraints
    # (x_i - x_j)^2 + (y_i - y_j)^2 >= (r_i + r_j)^2
    # dist_sq - (r_i + r_j)^2 >= 0
    for i in range(n):
        for j in range(i + 1, n):
            idx_i = 3 * i
            idx_j = 3 * j
            
            dx = centers_radii[idx_i] - centers_radii[idx_j]
            dy = centers_radii[idx_i + 1] - centers_radii[idx_j + 1]
            dist_sq = dx**2 + dy**2
            
            r_i = centers_radii[idx_i + 2]
            r_j = centers_radii[idx_j + 2]
            sum_r = r_i + r_j
            
            constraints_list.append(dist_sq - sum_r**2)
            
    return np.array(constraints_list)

    """
    Objective function to minimize: -sum(radii)

    radii = centers_radii[2::3]
    return -np.sum(radii)
