def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    n = 26
    
    # 1. Initial hexagonal lattice configuration
    centers = []
    row = 0
    while len(centers) < n:
        col = 0
        while len(centers) < n:
            x = 0.1 + col * 0.2 + (row % 2) * 0.1
            y = 0.1 + row * 0.17320508
            if x < 0.95 and y < 0.95:
                centers.append([x, y])
            col += 1
        row += 1
    centers = np.array(centers[:n])
    centers += np.random.randn(n, 2) * 0.005
    radii = np.ones(n) * 0.06
    
    # 2. Force-directed optimization with radius growth
    max_iter = 5000
    r_growth = 0.0007
    pos_step = 0.05
    rep_scale = 300.0
    
    for _ in range(max_iter):
        radii += r_growth
        
        forces = np.zeros_like(centers)
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                min_dist = radii[i] + radii[j]
                
                if dist < min_dist and dist > 1e-9:
                    overlap = min_dist - dist
                    f = overlap * overlap * rep_scale / dist
                    forces[i, 0] += dx * f
                    forces[i, 1] += dy * f
                    forces[j, 0] -= dx * f
                    forces[j, 1] -= dy * f
                    
            # Boundary repulsion
            for dim in range(2):
                if centers[i, dim] - radii[i] < 0:
                    forces[i, dim] += (radii[i] - centers[i, dim]) * rep_scale * 100
                if centers[i, dim] + radii[i] > 1:
                    forces[i, dim] -= (centers[i, dim] + radii[i] - 1) * rep_scale * 100
                    
        centers += forces * pos_step
        
        # Cooling schedule
        r_growth *= 0.9995
        pos_step *= 0.9998
        rep_scale *= 0.9999
        
    # 3. Strict projection to guarantee validity
    for _ in range(1000):
        for i in range(n):
            for dim in range(2):
                if centers[i, dim] < radii[i]: centers[i, dim] = radii[i]
                if centers[i, dim] > 1 - radii[i]: centers[i, dim] = 1 - radii[i]
                
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii[i] + radii[j] and dist > 1e-12:
                    overlap = (radii[i] + radii[j] - dist) / 2.0
                    nx, ny = dx / dist, dy / dist
                    centers[i, 0] += nx * overlap
                    centers[i, 1] += ny * overlap
                    centers[j, 0] -= nx * overlap
                    centers[j, 1] -= ny * overlap
                    
    return centers, radii, float(np.sum(radii))
