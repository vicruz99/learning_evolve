To maximize the sum of radii for 26 circles in a unit square, we treat the problem as an optimization of both circle centers and radii. Since the constraints are non-convex, a direct solver might struggle, so we employ a two-stage approach:

1.  **LP for Radii**: For any fixed set of centers, the problem of maximizing the sum of radii subject to non-overlap and boundary constraints is a Linear Programming (LP) problem. This allows us to efficiently evaluate the quality of a configuration.
2.  **Heuristic Optimization for Centers**: We initialize 26 centers using a perturbed hexagonal lattice (which is denser than a square grid). We then use a local search (Hill Climbing with Simulated Annealing) to perturb the centers. If a perturbation increases the sum of radii (computed via the LP), we accept it. This helps escape local optima and fine-tune the positions.
3.  **Final Polish**: After the center optimization, we run the LP one last time to determine the optimal radii and calculate the final sum.

This hybrid approach leverages the speed of LP for radii evaluation and the flexibility of stochastic search for geometry.

import numpy as np
import scipy.optimize as opt

def solve_radii_lp(centers):
    """
    Solves for the radii that maximize the sum of radii for fixed centers using Linear Programming.
    
    Parameters:
    centers : np.array of shape (n, 2)
    
    Returns:
    radii : np.array of shape (n)
    sum_radii : float
    """
    n = centers.shape[0]
    
    # Objective: Maximize sum(r_i) -> Minimize sum(-r_i)
    c = -np.ones(n)
    
    # Constraints matrix A_ub * r <= b_ub
    # 1. Pairwise non-overlap: r_i + r_j <= dist_ij
    # 2. Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    
    constraints = []
    bounds_values = []
    
    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        dists_to_boundary = [x, 1 - x, y, 1 - y]
        min_dist = min(dists_to_boundary)
        
        # r_i <= min_dist
        # In standard form A_ub * x <= b_ub
        # row i has 1 at index i, 0 elsewhere
        row = np.zeros(n)
        row[i] = 1.0
        constraints.append(row)
        bounds_values.append(min_dist)
        
    # Pairwise constraints
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            constraints.append(row)
            bounds_values.append(dist)
            
    A_ub = np.array(constraints)
    b_ub = np.array(bounds_values)
    
    # Bounds for variables r_i >= 0
    bounds = [(0, None) for _ in range(n)]
    
    try:
        res = opt.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            radii = res.x
            return radii, -res.fun
        else:
            # Fallback: small radii
            return np.zeros(n), 0.0
    except Exception:
        return np.zeros(n), 0.0

def evaluate_packing(centers):
    """
    Evaluates the sum of radii for a given set of centers.
    """
    radii, sum_r = solve_radii_lp(centers)
    return sum_r, radii

def run_packing():
    """
    Runs the packing optimization.
    Returns centers, radii, sum_radii.
    """
    n = 26
    
    # 1. Initialize centers in a perturbed hexagonal lattice
    # We try to fit 26 points. A 5x5 grid has 25 points.
    # Hexagonal packing allows more density.
    # Let's create a grid of points and then optimize.
    
    # Initialize with a grid structure but slightly randomized to break symmetry
    # 5 columns, 5 rows is 25. We need 26.
    # Let's try a hexagonal arrangement.
    # Rows of 5, 5, 5, 5, 4? Or 5, 6, 5, 5, 5?
    # Let's just place them in a grid and let the optimizer move them.
    
    centers = np.zeros((n, 2))
    
    # Hexagonal grid parameters
    # Spacing roughly 0.2
    rows = 6
    cols = 5 # 6*5 = 30 > 26
    
    idx = 0
    for r in range(rows):
        for c in range(cols):
            if idx >= n:
                break
            # Hexagonal offset
            x = (c + 0.5 + (r % 2) * 0.5) * 0.19 + 0.02
            y = r * 0.165 + 0.05
            centers[idx] = [x, y]
            idx += 1
        if idx >= n:
            break
            
    # Ensure we have exactly n points
    centers = centers[:n]
    
    # Add some noise
    np.random.seed(42)
    centers += np.random.normal(0, 0.01, centers.shape)
    
    # Clip to [0, 1]
    centers = np.clip(centers, 0, 1)
    
    # 2. Optimize centers using Simulated Annealing / Hill Climbing
    current_sum, current_radii = evaluate_packing(centers)
    best_sum = current_sum
    best_centers = centers.copy()
    
    temperature = 0.1
    decay = 0.995
    
    # Iterate optimization
    for step in range(2000): # Number of iterations
        # Pick a random circle to move
        i = np.random.randint(0, n)
        
        # Generate a perturbation
        # Step size decreases over time
        step_size = 0.02 * (0.995 ** (step // 10))
        move = np.random.normal(0, step_size, 2)
        
        new_center = centers[i] + move
        
        # Check bounds
        if 0 <= new_center[0] <= 1 and 0 <= new_center[1] <= 1:
            centers[i] = new_center
            
            # Evaluate new sum
            new_sum, _ = evaluate_packing(centers)
            
            # Acceptance criteria (Simulated Annealing)
            delta = new_sum - current_sum
            if delta > 0 or np.random.rand() < np.exp(delta / temperature):
                current_sum = new_sum
                if current_sum > best_sum:
                    best_sum = current_sum
                    best_centers = centers.copy()
            else:
                # Revert if not accepted? 
                # Actually in SA we usually keep the state, but for simple hill climbing with noise, 
                # reverting on rejection is safer for convergence to local optima.
                # Let's stick to simple Hill Climbing for stability, or SA with state tracking.
                # For simplicity in this script, we'll just revert if not accepted to keep valid packing.
                centers[i] -= move 
        else:
            # Out of bounds, reject
            pass
            
        temperature *= decay
        
    # 3. Final optimization of radii for the best centers found
    best_radii, final_sum = evaluate_packing(best_centers)
    
    # Validation check (internal)
    # We trust the LP, but let's ensure no NaNs
    if np.isnan(best_centers).any() or np.isnan(best_radii).any():
        # Fallback to grid
        centers = np.zeros((n, 2))
        idx = 0
        for r in range(6):
            for c in range(5):
                if idx < n:
                    centers[idx] = [0.1 + c * 0.2, 0.1 + r * 0.16]
                    idx += 1
        best_radii, final_sum = evaluate_packing(centers)
        best_centers = centers

    return best_centers, best_radii, final_sum
