import numpy as np
from scipy.optimize import minimize

def run_packing():
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    Uses L-BFGS-B with a penalty method for constraints.
    """
    n = 26
    
    # 1. Generate Initial Configuration
    # We place circles in a grid pattern to ensure a valid starting point (no overlaps).
    # A hexagonal arrangement is generally better, but a simple grid is easier to generate
    # and valid with small radii.
    
    centers = []
    
    # Let's try to fit 26 circles.
    # A 6x5 grid has 30 slots. We can pick 26.
    # To make it slightly hexagonal, we can offset rows, but simple grid is safe.
    # Let's use a spacing that allows radius 0.05 (diameter 0.1).
    # Margins of 0.05 on each side. Available space 0.9.
    # 5 intervals for 6 points? No, 6 points need 5 gaps.
    # Gap size = 0.9 / 5 = 0.18.
    # Points: 0.05, 0.23, 0.41, 0.59, 0.77, 0.95.
    
    x_spacing = 0.18
    y_spacing = 0.18
    margin = 0.05
    
    # Generate a grid of potential centers
    grid_x = np.arange(margin, 1.0, x_spacing)
    grid_y = np.arange(margin, 1.0, y_spacing)
    
    # Collect points
    points = []
    for y in grid_y:
        for x in grid_x:
            points.append([x, y])
            if len(points) >= n:
                break
        if len(points) >= n:
            break
            
    # If we didn't get enough points (unlikely with these params), fill with random valid
    if len(points) < n:
        while len(points) < n:
            px = np.random.uniform(0.1, 0.9)
            py = np.random.uniform(0.1, 0.9)
            # Check simple distance to existing points (approx)
            # Just append for now, optimizer will fix
            points.append([px, py])
            
    initial_centers = np.array(points[:n])
    initial_radii = np.full(n, 0.05) # Small initial radius
    
    # 2. Setup Optimization Variables
    # Variables vector: [x1, y1, r1, x2, y2, r2, ..., x26, y26, r26]
    x0 = np.zeros(n * 3)
    for i in range(n):
        x0[3*i] = initial_centers[i, 0]
        x0[3*i+1] = initial_centers[i, 1]
        x0[3*i+2] = initial_radii[i]
        
    # Bounds: x, y in [0, 1], r in [0, 0.5]
    bounds = []
    for _ in range(n):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 0.5)) # r
        
    # 3. Define Objective Function with Penalty
    # We minimize -sum(r) + mu * penalty
    # Penalty includes overlap and boundary violations.
    
    # Precompute indices for vectorization if needed, but inside function is fine for n=26
    # To speed up, we can precompute the mask for upper triangle
    triu_indices = np.triu_indices(n, k=1)
    
    def objective(vars):
        # Unpack variables
        cx = vars[0::3]
        cy = vars[1::3]
        r = vars[2::3]
        
        # Objective: minimize negative sum of radii
        obj_val = -np.sum(r)
        
        # Penalty Parameter
        mu = 5000.0 
        
        penalty = 0.0
        
        # --- Boundary Penalties ---
        # Constraints: r <= x, r <= 1-x, r <= y, r <= 1-y
        # Violations: r - x, r - (1-x), etc.
        
        # Vectorized
        v1 = r - cx
        v2 = r - (1.0 - cx)
        v3 = r - cy
        v4 = r - (1.0 - cy)
        
        # Only count positive violations
        # Using np.maximum to handle vectorized clipping
        p1 = np.maximum(0.0, v1)
        p2 = np.maximum(0.0, v2)
        p3 = np.maximum(0.0, v3)
        p4 = np.maximum(0.0, v4)
        
        penalty += np.sum(p1**2) + np.sum(p2**2) + np.sum(p3**2) + np.sum(p4**2)
        
        # --- Overlap Penalties ---
        # Constraint: (r_i + r_j)^2 <= (x_i - x_j)^2 + (y_i - y_j)^2
        # Violation: (r_i + r_j)^2 - dist_sq > 0
        
        # Use broadcasting to compute all pairwise distances and radius sums
        # cx shape (n,), cx[:, None] shape (n, 1)
        dx = cx[:, None] - cx[None, :] # (n, n)
        dy = cy[:, None] - cy[None, :] # (n, n)
        dist_sq = dx**2 + dy**2        # (n, n)
        
        sum_r = r[:, None] + r[None, :] # (n, n)
        sum_r_sq = sum_r**2             # (n, n)
        
        # Violation matrix
        violation = sum_r_sq - dist_sq
        
        # We only need upper triangle (i < j)
        # Extract upper triangle elements
        # violation is symmetric, diagonal is 2r^2 - 0 = 4r^2 (self overlap? No, i!=j)
        # Actually diagonal i=j gives 4r_i^2 - 0. But we ignore diagonal.
        
        # Extract upper triangle
        viol_upper = violation[triu_indices]
        
        # Apply penalty for positive violations
        # Squared penalty for smoothness
        pos_viol = np.maximum(0.0, viol_upper)
        penalty += np.sum(pos_viol**2)
        
        return obj_val + mu * penalty

    # 4. Run Optimization
    # Use L-BFGS-B as it handles bounds well.
    # We might need to run it a few times or rely on a good start.
    # With a valid start, it should converge to a local maximum of sum of radii.
    
    options = {
        'maxiter': 5000,
        'ftol': 1e-12,
        'gtol': 1e-10
    }
    
    try:
        res = minimize(objective, x0, method='L-BFGS-B', bounds=bounds, options=options)
        final_vars = res.x
    except Exception:
        # Fallback or use the initial if something crashes (unlikely)
        final_vars = x0

    # 5. Extract Results
    final_centers = np.zeros((n, 2))
    final_radii = np.zeros(n)
    
    for i in range(n):
        final_centers[i, 0] = final_vars[3*i]
        final_centers[i, 1] = final_vars[3*i+1]
        final_radii[i] = final_vars[3*i+2]
        
    sum_radii = np.sum(final_radii)
    
    return final_centers, final_radii, sum_radii
