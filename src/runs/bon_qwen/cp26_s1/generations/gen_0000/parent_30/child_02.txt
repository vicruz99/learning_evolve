def compute_penalty(centers, radii):
    """
    Calculates the penalty for circle overlaps and boundary violations.
    Lower is better. 0 means valid packing.
    """
    n = centers.shape[0]
    penalty = 0.0
    
    # Pairwise overlaps
    # Compute distance matrix
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2) + 1e-15)
    r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    
    # Overlap amount
    overlap = np.maximum(0, r_sum - dists)
    np.fill_diagonal(overlap, 0)
    penalty += np.sum(overlap**2)
    
    # Boundary violations
    x = centers[:, 0]
    y = centers[:, 1]
    r = radii
    
    # Left: x < r => r - x > 0
    penalty += np.sum(np.maximum(0, r - x)**2)
    # Right: x > 1-r => x - (1-r) > 0
    penalty += np.sum(np.maximum(0, x - (1 - r))**2)
    # Bottom: y < r
    penalty += np.sum(np.maximum(0, r - y)**2)
    # Top: y > 1-r
    penalty += np.sum(np.maximum(0, y - (1 - r))**2)
