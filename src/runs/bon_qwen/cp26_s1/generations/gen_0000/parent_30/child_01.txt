from scipy.optimize import minimize

def compute_violations(centers, radii):
    """
    Computes the sum of squared violations for overlaps and boundaries.
    """
    n = centers.shape[0]
    violations = 0.0
    
    # Boundary violations
    # Circle i must satisfy: r_i <= x_i <= 1 - r_i and r_i <= y_i <= 1 - r_i
    # This implies x_i - r_i >= 0 => r_i - x_i <= 0. Violation is max(0, r_i - x_i)^2
    # Similarly for other sides.
    
    # x_min
    viol_x_min = np.maximum(0, radii - centers[:, 0])
    violations += np.sum(viol_x_min**2)
    
    # x_max
    viol_x_max = np.maximum(0, centers[:, 0] - (1 - radii))
    violations += np.sum(viol_x_max**2)
    
    # y_min
    viol_y_min = np.maximum(0, radii - centers[:, 1])
    violations += np.sum(viol_y_min**2)
    
    # y_max
    viol_y_max = np.maximum(0, centers[:, 1] - (1 - radii))
    violations += np.sum(viol_y_max**2)
    
    # Overlap violations
    # dist >= r_i + r_j
    # Violation: max(0, r_i + r_j - dist)^2
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            sum_r = radii[i] + radii[j]
            if sum_r > dist:
                violations += (sum_r - dist)**2
                
    return violations

def objective_func(params, n):
    """
    Objective function to minimize: -sum(radii) + lambda * violations

    centers = params[:2*n].reshape(n, 2)

    lambda_val = 1000.0
    
    violations = compute_violations(centers, radii)
    
    obj = -np.sum(radii) + lambda_val * violations
