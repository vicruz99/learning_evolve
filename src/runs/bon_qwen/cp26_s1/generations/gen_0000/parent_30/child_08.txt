

Looking at this circle packing problem, I need to pack 26 circles in a unit square to maximize the sum of radii, targeting 2.636.

<strategy>
**Key Insights:**

1. **For fixed centers, computing optimal radii is a Linear Program**: The problem becomes: maximize Σr_i subject to boundary constraints (r_i ≤ distance to each edge) and pairwise constraints (r_i + r_j ≤ ||c_i - c_j||).

2. **Optimizing centers is non-convex**: I'll use Nelder-Mead from multiple starting points, with the LP solved at each evaluation.

3. **Good initial arrangements matter**: I'll use hexagonal-like patterns with different row counts (4-6 rows) since hexagonal packing is the densest in 2D.

4. **Varying radii helps**: The LP naturally assigns larger radii to circles with more space, and smaller to cramped ones. Corner circles can be larger.

5. **Multi-start optimization**: Run Nelder-Mead from ~10 different initial configurations to avoid bad local optima.

**Implementation Plan:**
- `compute_radii()`: Solve LP using scipy's linprog (fast for this size)
- `hex_init()`: Generate hexagonal initial arrangements  
- `run_packing()`: Multi-start Nelder-Mead optimization with fallback
- Use scipy.optimize.linprog with 'highs' method for fast LP solving
- Clip centers to valid range to avoid infeasible LPs
</strategy>

```python
import numpy as np
from scipy.optimize import minimize, linprog


def compute_radii(centers):
    """Compute optimal radii for given centers using LP.
    
    Maximizes sum of radii subject to:
    - Each circle fits in unit square
    - No two circles overlap
    """
    n = centers.shape[0]
    
    # Objective: minimize -sum(r_i)  [equivalent to maximize sum(r_i)]
    c_obj = -np.ones(n)
    
    # Build constraint matrix A_ub @ r <= b_ub
    n_boundary = 4 * n
    n_pairwise = n * (n - 1) // 2
    n_total = n_boundary + n_pairwise
    
    A = np.zeros((n_total, n))
    b = np.zeros(n_total)
    
    idx = 0
    for i in range(n):
        x, y = centers[i]
        # r_i <= x (left boundary)
        A[idx, i] = 1.0
        b[idx] = x
        idx += 1
        # r_i <= 1 - x (right boundary)
        A[idx, i] = 1.0
        b[idx] = 1.0 - x
        idx += 1
        # r_i <= y (bottom boundary)
        A[idx, i] = 1.0
        b[idx] = y
        idx += 1
        # r_i <= 1 - y (top boundary)
        A[idx, i] = 1.0
        b[idx] = 1.0 - y
        idx += 1
    
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = dist
            idx += 1
    
    bounds = [(0.0, None)] * n
    
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    
    if res.success:
        return np.maximum(res.x, 0.0)
    else:
        # Fallback: use boundary distances only
        radii = np.array([
            min(x, 1.0 - x, y, 1.0 - y) for x, y in centers
        ])
        return np.maximum(radii, 0.0)


def neg_sum_radii(params, n=26):
    """Objective function: negative sum of radii (to be minimized)."""
    centers = np.array(params, dtype=np.float64).reshape((n, 2))
    centers = np.clip(centers, 1e-6, 1.0 - 1e-6)
    radii = compute_radii(centers)
    return -np.sum(radii)


def hex_init(n, num_rows):
    """Create a hexagonal-like initial arrangement with given number of rows."""
    # Distribute n circles across num_rows rows
    base_count = n // num_rows
    remainder = n % num_rows
    
    counts = [base_count] * num_rows
    for i in range(remainder):
        counts[i] += 1
    
    positions = []
    
    if num_rows == 1:
        y = 0.5
        x_spacing = 1.0 / (counts[0] + 1)
        for j in range(counts[0]):
            x = x_spacing * (j + 1)
            positions.append([x, y])
    else:
        y_spacing = 0.85 / (num_rows - 1)
        
        for i, count in enumerate(counts):
            if count == 0:
                continue
            y = 0.075 + i * y_spacing
            x_spacing = 0.85 / max(count - 1, 1)
            x_start = (1.0 - (count - 1) * x_spacing) / 2.0
            
            # Stagger odd rows for hexagonal pattern
            if i % 2 == 1:
                x_start += x_spacing / 2.0
            
            for j in range(count):
                x = x_start + j * x_spacing
                positions.append([x, y])
    
    return np.array(positions[:n]).flatten()


def grid_init(n, nx, ny):
    """Create a grid-like initial arrangement."""
    x_vals = np.linspace(0.12, 0.88, nx)
    y_vals = np.linspace(0.12, 0.88, ny)
    
    positions = []
    for y in y_vals:
        for x in x_vals:
            if len(positions) < n:
                positions.append([x, y])
    
    while len(positions) < n:
        positions.append([0.5, 0.5])
    
    return np.array(positions[:n]).flatten()


def random_init(n, seed):
    """Create a random initial arrangement."""
    rng = np.random.RandomState(seed)
    x = rng.uniform(0.15, 0.85, n)
    y = rng.uniform(0.15, 0.85, n)
    return np.column_stack([x, y]).flatten()


def run_packing():
    """Pack 26 circles in unit square to maximize sum of radii."""
    n = 26
    
    best_obj = float('inf')
    best_params = None
    
    # Generate diverse initial configurations
    initial_configs = []
    
    # Hexagonal patterns with different row counts
    for num_rows in [4, 5, 6, 7]:
        try:
            initial_configs.append(hex_init(n, num_rows))
        except Exception:
            pass
    
    # Grid patterns
    for nx, ny in [(5, 6), (6, 5), (4, 7), (7, 4)]:
        try:
            initial_configs.append(grid_init(n, nx, ny))
        except Exception:
            pass
    
    # Random patterns
    for seed in [42, 123, 456, 789, 101]:
        try:
            initial_configs.append(random_init(n, seed))
        except Exception:
            pass
    
    # Perturb hex initial configs
    for i in range(0, len(initial_configs), 2):
        try:
            perturbed = initial_configs[i].copy()
            rng = np.random.RandomState(i * 100)
            perturbed += rng.normal(0, 0.02, perturbed.shape)
            perturbed = np.clip(perturbed, 0.05, 0.95)
            initial_configs.append(perturbed)
        except Exception:
            pass
    
    # Optimize from each initial configuration
    for init in initial_configs:
        if len(init) != 2 * n:
            continue
        
        try:
            result = minimize(
                neg_sum_radii,
                init,
                args=(n,),
                method='Nelder-Mead',
                options={
                    'maxiter': 20000,
                    'xatol': 1e-9,
                    'fatol': 1e-12,
                    'adaptive': True
                }
            )
            
            if result.success or result.fun < best_obj:
                if result.fun < best_obj:
                    best_obj = result.fun
                    best_params = result.x.copy()
        except Exception:
            continue
    
    # If no optimization succeeded, use best initial config
    if best_params is None:
        best_obj = float('inf')
        for init in initial_configs:
            if len(init) == 2 * n:
                obj = neg_sum_radii(init, n)
                if obj < best_obj:
                    best_obj = obj
                    best_params = init.copy()
    
    # Compute final centers and radii
    if best_params is not None:
        centers = np.clip(best_params.reshape((n, 2)), 1e-6, 1.0 - 1e-6)
    else:
        centers = hex_init(n, 5).reshape((n, 2))
        centers = np.clip(centers, 1e-6, 1.0 - 1e-6)
    
    radii = compute_radii(centers)
    total = np.sum(radii)
    
    return centers, radii, total
```
