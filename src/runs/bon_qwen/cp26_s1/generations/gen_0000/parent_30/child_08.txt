import numpy as np
from scipy.optimize import minimize

def run_packing():
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    
    Returns:
        tuple: (centers, radii, sum_radii)
    """
    n = 26
    
    # --- Phase 1: Initialization and Force-Directed Layout ---
    
    # Initialize centers in a grid-like pattern with some randomness
    # Using a 6x6 grid points to get 36 potential spots, we pick 26.
    # Actually, let's just place them in a 5x5 grid plus one extra or similar.
    # A simple dense grid is a good starting point for repulsion.
    
    # Create a dense grid of points
    grid_size = 6
    xs = np.linspace(0.05, 0.95, grid_size)
    ys = np.linspace(0.05, 0.95, grid_size)
    grid_points = []
    for x in xs:
        for y in ys:
            grid_points.append([x, y])
    
    # We need 26 points. Let's take the first 26 from the 36 grid points.
    # This gives a fairly uniform distribution.
    centers = np.array(grid_points[:n])
    
    # Initialize radii to a small valid value
    radii = np.ones(n) * 0.05
    
    # Force-directed simulation to spread centers
    # This helps the optimizer start from a configuration where circles are not overlapping
    # and have room to grow.
    n_iters = 500
    dt = 0.005
    repulsion_strength = 0.05
    boundary_stiffness = 0.1
    
    current_centers = centers.copy()
    
    for _ in range(n_iters):
        forces = np.zeros_like(current_centers)
        
        # Repulsion between circles
        for i in range(n):
            for j in range(i + 1, n):
                diff = current_centers[i] - current_centers[j]
                dist_sq = np.sum(diff**2)
                dist = np.sqrt(dist_sq)
                
                # Avoid division by zero
                if dist < 1e-6:
                    dist = 1e-6
                    diff = np.random.rand(2) * 1e-6 # Random nudge if exactly same
                
                # Repulsive force: F ~ 1/d^2
                # We want to push them apart if they are too close
                # "Too close" is subjective, let's say if dist < 0.2
                if dist < 0.25:
                    force_mag = repulsion_strength / (dist * dist)
                    force_vec = (diff / dist) * force_mag
                    forces[i] += force_vec
                    forces[j] -= force_vec
        
        # Boundary repulsion (keep centers inside [0,1])
        # If a center is close to boundary, push it in.
        # We treat boundaries as repulsive walls.
        margin = 0.05
        for i in range(n):
            c = current_centers[i]
            # Left wall
            if c[0] < margin:
                forces[i, 0] += boundary_stiffness * (margin - c[0])
            # Right wall
            if c[0] > 1.0 - margin:
                forces[i, 0] -= boundary_stiffness * (c[0] - (1.0 - margin))
            # Bottom wall
            if c[1] < margin:
                forces[i, 1] += boundary_stiffness * (margin - c[1])
            # Top wall
            if c[1] > 1.0 - margin:
                forces[i, 1] -= boundary_stiffness * (c[1] - (1.0 - margin))
        
        # Update positions
        current_centers += dt * forces
        
        # Clamp to valid range [0, 1] just in case
        current_centers = np.clip(current_centers, 0.0, 1.0)
        
    # Update centers with the result of simulation
    centers = current_centers
    # Reset radii to small value for the optimizer to scale up
    radii = np.ones(n) * 0.01
    
    # --- Phase 2: Optimization ---
    
    # Prepare variables: [x1, y1, r1, x2, y2, r2, ...]
    # Flattened vector of size 3*n
    x0 = np.concatenate([centers.flatten(), radii])
    
    # Bounds for variables
    # x, y in [0, 1]
    # r in [1e-6, 0.5] (radius must be positive, max 0.5)
    bounds = []
    for _ in range(n):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((1e-6, 0.5)) # r
    
    # Objective function: Minimize -sum(radii)
    def objective(vars_flat):
        r = vars_flat[2::3] # radii are at indices 2, 5, 8...
        return -np.sum(r)
    
    # Constraints
    # 1. Boundary constraints: 
    #    x - r >= 0  => r - x <= 0  (Standard form: g(x) <= 0 for inequality in scipy? 
    #    Wait, scipy minimize constraints:
    #    type 'ineq' means fun(x) >= 0.
    #    So we need fun(x) >= 0.
    #    x - r >= 0
    #    1 - (x + r) >= 0
    #    y - r >= 0
    #    1 - (y + r) >= 0
    
    # 2. Overlap constraints:
    #    dist^2 - (r_i + r_j)^2 >= 0
    
    cons = []
    
    # Add boundary constraints
    for i in range(n):
        idx_x = 3 * i
        idx_y = 3 * i + 1
        idx_r = 3 * i + 2
        
        # x - r >= 0
        cons.append({
            'type': 'ineq',
            'fun': lambda v, i=i: v[idx_x] - v[idx_r]
        })
        # 1 - x - r >= 0
        cons.append({
            'type': 'ineq',
            'fun': lambda v, i=i: 1.0 - v[idx_x] - v[idx_r]
        })
        # y - r >= 0
        cons.append({
            'type': 'ineq',
            'fun': lambda v, i=i: v[idx_y] - v[idx_r]
        })
        # 1 - y - r >= 0
        cons.append({
            'type': 'ineq',
            'fun': lambda v, i=i: 1.0 - v[idx_y] - v[idx_r]
        })
        
    # Add overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            idx_xi = 3 * i
            idx_yi = 3 * i + 1
            idx_ri = 3 * i + 2
            
            idx_xj = 3 * j
            idx_yj = 3 * j + 1
            idx_rj = 3 * j + 2
            
            # Constraint: (xi - xj)^2 + (yi - yj)^2 - (ri + rj)^2 >= 0
            cons.append({
                'type': 'ineq',
                'fun': lambda v, i=i, j=j: \
                    (v[idx_xi] - v[idx_xj])**2 + (v[idx_yi] - v[idx_yj])**2 - (v[idx_ri] + v[idx_rj])**2
            })

    # Run optimization
    # Method SLSQP is good for constrained non-linear problems
    result = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, 
                      options={'maxiter': 1000, 'ftol': 1e-9, 'disp': False})
    
    if not result.success:
        # If optimization fails, try to recover or return best found
        # But usually it converges to a local optimum
        pass
        
    # Extract results
    final_vars = result.x
    final_centers = final_vars[:2*n].reshape((n, 2))
    final_radii = final_vars[2*n:]
    
    # Ensure radii are non-negative (clipping)
    final_radii = np.maximum(final_radii, 0.0)
    
    # Ensure centers are within bounds (clipping, though constraints should handle it)
    final_centers = np.clip(final_centers, 0.0, 1.0)
    
    # Calculate sum of radii
    sum_radii = np.sum(final_radii)
    
    return final_centers, final_radii, sum_radii
