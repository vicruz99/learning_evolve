def calculate_overlaps(centers, radii):
    """
    Calculates the overlap violations for all pairs of circles.
    Returns an array of violations (negative means valid, positive means overlap).
    """
    n = centers.shape[0]
    violations = []
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            violations.append(radii[i] + radii[j] - dist)
    return np.array(violations)
