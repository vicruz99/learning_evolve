import numpy as np
from scipy.optimize import linprog, minimize

def compute_max_sum_radii(centers):
    """
    Given fixed centers, solve the LP to find radii that maximize their sum
    while satisfying non-overlap and boundary constraints.
    """
    n = centers.shape[0]
    # Objective: minimize -sum(r_i)  =>  c = -1 for each variable
    c = -np.ones(n)
    
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        x, y = centers[i]
        row = np.zeros(n)
        row[i] = 1
        A_ub.extend([row, row, row, row])
        b_ub.extend([x, 1.0 - x, y, 1.0 - y])
        
    # Pairwise constraints: r_i + r_j <= dist_{ij}
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            row = np.zeros(n)
            row[i] = 1
            row[j] = 1
            A_ub.append(row)
            b_ub.append(dist)
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0.0, 1.0) for _ in range(n)]
    
    # Solve LP using HiGHS solver (efficient and robust)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return -res.fun, res.x
    # Fallback if LP fails (should not happen with valid centers)
    return 0.0, np.zeros(n)

def objective(centers_flat):
    """
    Objective function for center optimization.
    Returns negative sum of radii to be minimized.
    """
    centers = centers_flat.reshape(-1, 2)
    # Clip centers to valid domain to keep LP feasible
    centers = np.clip(centers, 0.0, 1.0)
    s, _ = compute_max_sum_radii(centers)
    return -s

def run_packing():
    np.random.seed(42)
    n = 26
    
    # Initialization: Dense hexagonal-like grid covering the square
    pts = []
    for i in range(9):
        for j in range(9):
            # Offset every other row for hexagonal packing
            px = 0.1 + i * 0.10 + (j % 2) * 0.05
            py = 0.1 + j * 0.10
            pts.append([px, py])
    pts = np.array(pts)
    np.random.shuffle(pts)
    x0 = pts[:n].flatten()
    
    # Optimize centers to maximize the LP-derived sum of radii
    # Nelder-Mead is robust for non-smooth objectives like this LP oracle
    res = minimize(objective, x0, method='Nelder-Mead', 
                   options={'maxiter': 25000, 'xatol': 1e-8, 'fatol': 1e-9})
                   
    centers = res.x.reshape(n, 2)
    centers = np.clip(centers, 0.0, 1.0)
    
    # Compute final optimal radii for the best centers
    _, radii = compute_max_sum_radii(centers)
    
    return centers, radii, float(np.sum(radii))
