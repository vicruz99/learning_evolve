def solve_packing_seed(seed: int) -> tuple[np.ndarray, np.ndarray, float]:
    """
    Solves the packing problem for a specific seed.
    Uses force-directed growth followed by numerical optimization.
    """
    n = 26
    np.random.seed(seed)

    centers = generate_hex_grid(n, jitter=0.05)

    # We assume equal circles for the main optimization to keep variables low.
    # We maximize r (or minimize -r) subject to constraints.
    
    # Variables: [x1, y1, ..., x26, y26, r]
    x0 = np.concatenate([centers.flatten(), [r_init]])
    
    # Define bounds: 0 <= x,y <= 1, 0 <= r
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 1.0)]

    cons = []

    # We add constraints for each circle.
    for i in range(n):
        idx_x = 2 * i
        idx_y = 2 * i + 1
        idx_r = 2 * n
        
        # x - r >= 0
        cons.append({'type': 'ineq', 'fun': lambda v, i=i: v[idx_x] - v[idx_r]})
        # 1 - x - r >= 0
        cons.append({'type': 'ineq', 'fun': lambda v, i=i: 1.0 - v[idx_x] - v[idx_r]})

        # 1 - y - r >= 0

        pass

    # Fallback if optimization fails
    # Return the result from force growth
    radii = np.full(n, r_init)

def generate_hex_grid(n: int, jitter: float) -> np.ndarray:
    """
    Generates n centers on a hexagonal grid within [0,1]^2.
    """
    centers = []
    spacing = 0.15 # Dense enough to pick from
    # Generate grid points
    for i in range(int(1 / spacing) + 2):
        for j in range(int(1 / spacing) + 2):
            x = i * spacing
            y = j * spacing * np.sqrt(3) / 2
            if j % 2 == 1:
                x += spacing / 2
            
            if 0 <= x <= 1 and 0 <= y <= 1:
                centers.append([x, y])
                
    # If we don't have enough points, add random ones
    if len(centers) < n:
        while len(centers) < n:
            centers.append(np.random.uniform(0, 1, 2))
            
    # Select n points. 
    # Strategy: Pick points that are well spread (Farthest Point Sampling)
    # Or just take the first n if grid is dense?
    # Let's take a subset that is well distributed.
    # Simple heuristic: Take points from the list, ensuring we don't pick too many from one cluster.
    # But actually, just taking first n from a dense grid is a decent start for force simulation.
    # Let's shuffle to avoid bias.
    np.random.shuffle(centers)
    centers = np.array(centers[:n])
    
    # Add jitter
    centers += np.random.uniform(-jitter, jitter, size=centers.shape)
    centers = np.clip(centers, 0.01, 0.99)
    
    return centers

def force_directed_growth(centers: np.ndarray, n: int, r_start: float) -> tuple[float, np.ndarray]:
    """
    Iteratively increases radius and pushes circles apart to find a dense packing.
    """
    r = r_start
    dt = 0.005
    k_repulsion = 100.0
    k_boundary = 100.0
    r_step = 0.0002
    
    # Precompute indices for pairs
    pairs = []
    for i in range(n):
        for j in range(i + 1, n):
            pairs.append((i, j))
            
    for step in range(2000):
        forces = np.zeros_like(centers)
        
        # Calculate forces
        for i, j in pairs:
            c_i = centers[i]
            c_j = centers[j]
            diff = c_j - c_i
            dist = np.linalg.norm(diff)
            
            if dist < 2 * r:
                if dist < 1e-7:
                    # Random push if overlapping exactly
                    diff = np.random.rand(2) * 2 - 1
                    dist = np.linalg.norm(diff)
                
                overlap = 2 * r - dist
                # Force vector
                f_vec = (diff / dist) * overlap * k_repulsion
                forces[i] -= f_vec
                forces[j] += f_vec
        
        # Boundary forces
        for i in range(n):
            x, y = centers[i]
            # Left
            if x < r:
                forces[i, 0] += (r - x) * k_boundary
            # Right
            if x > 1 - r:
                forces[i, 0] -= (x - (1 - r)) * k_boundary
            # Bottom
            if y < r:
                forces[i, 1] += (r - y) * k_boundary
            # Top
            if y > 1 - r:
                forces[i, 1] -= (y - (1 - r)) * k_boundary
                
        # Update centers
        centers += forces * dt
        
        # Clamp to [0, 1]
        centers = np.clip(centers, 0, 1)
        
        # Check validity to decide if we can increase r
        # If max violation is low, increase r
        max_violation = 0.0
        
        for i in range(n):
            x, y = centers[i]
            if x < r: max_violation = max(max_violation, r - x)
            if x > 1 - r: max_violation = max(max_violation, x - (1 - r))
            if y < r: max_violation = max(max_violation, r - y)
            if y > 1 - r: max_violation = max(max_violation, y - (1 - r))
            
            for j in range(i + 1, n):
                d = np.linalg.norm(centers[i] - centers[j])
                if d < 2 * r:
                    max_violation = max(max_violation, 2 * r - d)
        
        # If valid (within tolerance), increase r
        if max_violation < 1e-4:
            r += r_step
            # Slightly damp r increase if violation is close to limit
        else:
            # If stuck, apply a small shake to escape local minima
            if step % 100 == 0:
                shake = np.random.uniform(-0.01, 0.01, size=centers.shape)
                centers += shake
                centers = np.clip(centers, r, 1 - r)

    return r, centers

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square

    Args:
        centers: np.array of shape (n, 2) with (x, y) coordinates
        radii: np.array of shape (n) with radius of each circle

    Returns:
        True if valid, False otherwise
    """
    # Note: np is assumed to be imported
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True
