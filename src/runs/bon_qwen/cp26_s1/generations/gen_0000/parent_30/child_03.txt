import numpy as np
import scipy.optimize as opt

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Number of circles
    N = 26

    # Define objective function: minimize negative sum of radii
    def objective(variables):
        # variables: [x0, y0, r0, x1, y1, r1, ...]
        radii = variables[2::3]
        return -np.sum(radii)

    # Define constraints
    def create_constraints(N):
        constraints = []

        # Bounds for variables: x in [0, 1], y in [0, 1], r >= 0
        # We handle r >= 0 in bounds, x,y bounds in constraints to keep it simple or use bounds dict
        # Actually SLSQP supports bounds, but let's use inequality constraints for clarity if needed.
        # Using bounds is more efficient.

        # Constraint: x_i >= r_i  => x_i - r_i >= 0
        # Constraint: x_i <= 1 - r_i => 1 - x_i - r_i >= 0
        # Constraint: y_i >= r_i  => y_i - r_i >= 0
        # Constraint: y_i <= 1 - r_i => 1 - y_i - r_i >= 0
        
        # These are linear constraints.
        # Let's construct them as function constraints for SLSQP type 'ineq'
        
        # We can vectorize these or add them one by one.
        # 4 constraints per circle.
        
        def constraint_x_lower(v):
            x = v[0::3]
            r = v[2::3]
            return x - r
        
        def constraint_x_upper(v):
            x = v[0::3]
            r = v[2::3]
            return 1.0 - x - r
            
        def constraint_y_lower(v):
            y = v[1::3]
            r = v[2::3]
            return y - r
            
        def constraint_y_upper(v):
            y = v[1::3]
            r = v[2::3]
            return 1.0 - y - r

        # Add linear bounds constraints
        # SLSQP might handle simple bounds better via `bounds` argument, 
        # but we have coupling r <= x.
        # Let's stick to inequality constraints for x, y, r relations.
        
        constraints.append({'type': 'ineq', 'fun': constraint_x_lower})
        constraints.append({'type': 'ineq', 'fun': constraint_x_upper})
        constraints.append({'type': 'ineq', 'fun': constraint_y_lower})
        constraints.append({'type': 'ineq', 'fun': constraint_y_upper})

        # Non-overlap constraints
        # dist(i, j) >= r_i + r_j
        # sqrt((xi-xj)^2 + (yi-yj)^2) - (ri + rj) >= 0
        
        def constraint_non_overlap(v):
            # Compute all pairwise distances and sums of radii
            # Centers
            centers = v.reshape(-1, 3)[:, :2]
            radii = v[2::3]
            
            # Vectorized distance computation
            # dist_matrix[i, j] = ||c_i - c_j||
            # We only need upper triangle
            n = N
            diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :] # (N, N, 2)
            dists = np.sqrt(np.sum(diffs**2, axis=2)) # (N, N)
            
            # Sum of radii
            rad_sums = radii[:, np.newaxis] + radii[np.newaxis, :]
            
            # Margins
            margins = dists - rad_sums
            
            # Extract upper triangle (strictly upper)
            triu_indices = np.triu_indices(n, k=1)
            return margins[triu_indices]

        constraints.append({'type': 'ineq', 'fun': constraint_non_overlap})
        
        return constraints

    constraints = create_constraints(N)

    # Bounds
    # x, y in [0, 1], r in [0, 0.5] (max possible radius)
    bounds = []
    for i in range(N):
        bounds.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])

    # Helper to generate initial guess
    def get_initial_guess(method='hex'):
        centers = np.zeros((N, 2))
        radii = np.full(N, 0.01) # Small initial radius
        
        if method == 'hex':
            # Generate hexagonal lattice
            # We want to pack 26 circles. 
            # Try fitting in a pattern.
            # Let's try a rectangular hex grid.
            # Rows and cols logic to fit 26.
            # 5 rows: counts 5, 5, 5, 5, 6? No.
            # Let's just generate points and scale.
            
            # Generate a hex grid with spacing 1
            points = []
            # Approximate number of points needed in grid area
            # Area ~ 1. Density ~ 1.15 points per unit area (if r=0.5, area pi/4 ~ 0.78, but here points)
            # Let's just generate a 10x10 grid of hex points and pick 26?
            # Or generate specifically.
            
            # Let's try to fit 26 in a square box using hex packing logic.
            # Width W, Height H.
            # If we use 5 rows.
            # Row 0: 6 points.
            # Row 1: 5 points.
            # Row 2: 6 points.
            # Row 3: 5 points.
            # Row 4: 4 points.
            # Total 26.
            # Coordinates (unscaled):
            # Row k (0-indexed): y = k * sqrt(3)/2
            # x coords: if k even: 0, 1, 2, 3, 4, 5 (6 pts). if k odd: 0.5, 1.5, 2.5, 3.5, 4.5 (5 pts).
            # Last row (4, even) should have 4 points? 0, 1, 2, 3.
            
            # Let's adjust counts to sum to 26.
            # 6, 5, 6, 5, 4 -> 26.
            # Widths:
            # Row 0 (6 pts): x in 0..5. Span 5. (Actually centers 0 to 5, so width 5).
            # Wait, if points are at integers, span is 5.
            # Row 1 (5 pts): x in 0.5..4.5. Span 4.
            # Row 2 (6 pts): x in 0..5.
            # Row 3 (5 pts): x in 0.5..4.5.
            # Row 4 (4 pts): x in 0..3. Span 3.
            
            # Max X span: 5. Max Y span: 4 * sqrt(3)/2 = 2*sqrt(3) approx 3.46.
            
            pts = []
            # Row 0
            for x in range(6): pts.append([float(x), 0.0])
            # Row 1
            for x in range(5): pts.append([0.5 + float(x), np.sqrt(3)/2])
            # Row 2
            for x in range(6): pts.append([float(x), np.sqrt(3)])
            # Row 3
            for x in range(5): pts.append([0.5 + float(x), 1.5 * np.sqrt(3)])
            # Row 4
            for x in range(4): pts.append([float(x), 2.0 * np.sqrt(3)])
            
            # Scale to fit in [0,1]x[0,1]
            # Center and scale
            xs = [p[0] for p in pts]
            ys = [p[1] for p in pts]
            min_x, max_x = min(xs), max(xs)
            min_y, max_y = min(ys), max(ys)
            width = max_x - min_x
            height = max_y - min_y

                cy = (p[1] - min_y) / height
                # Wait, if we normalize to [0,1] separately, we might distort hex packing.
                # But it's just an initial guess.
                centers_list.append([cx, cy])
            
            centers = np.array(centers_list)
            
        elif method == 'random':
            centers = np.random.rand(N, 2)
            radii = np.full(N, 0.001)
        elif method == 'grid':
            # 5x5 grid is 25. Add one.
            coords = []
            for i in range(5):
                for j in range(5):
                    coords.append([i/4.0, j/4.0]) # 5x5 grid points?

                    # Boundaries are 0 and 1.
                    # Better: 0.1, 0.3, 0.5, 0.7, 0.9
            # Let's use linspace
            # 5x5 grid

            pts = []
            for i in range(5):
                for j in range(5):

            # 25 points.
            # Add 26th at (0.5, 0.5)? That's occupied.
            # Add at (0.2, 0.2)?
            pts.append([0.2, 0.2]) # Might overlap, but optimizer will move it.
            centers = np.array(pts)
            
        return centers, radii

    best_sum = -1.0
    best_centers = None
    best_radii = None

    # Try a few initial configurations
    methods = ['hex', 'grid'] # Random is less deterministic
    for method in methods:
        for _ in range(2): # 2 restarts per method
            centers, radii = get_initial_guess(method)
            
            # Construct initial variable vector
            x0 = np.zeros(3 * N)
            x0[0::3] = centers[:, 0]
            x0[1::3] = centers[:, 1]
            x0[2::3] = radii
            
            try:
                res = opt.minimize(objective, x0, method='SLSQP', 
                                   bounds=bounds, 
                                   constraints=constraints,
                                   options={'maxiter': 200, 'ftol': 1e-12, 'disp': False})
                
                if res.success or res.fun < best_sum * -1: # fun is negative sum
                    current_sum = -res.fun
                    if current_sum > best_sum:
                        best_sum = current_sum
                        best_centers = res.x[:2*N].reshape(N, 2)
                        best_radii = res.x[2::3]
            except Exception as e:
                # print(f"Optimization failed for {method}: {e}")
                pass

    # Final validation and cleanup
    # Check if valid
    # Note: The validation function checks for overlaps with 1e-12 tolerance.
    # The optimizer might have small violations due to numerical precision.
    # We can try to slightly reduce radii if needed, but usually SLSQP is good.
    
    # Just to be safe, clip radii to ensure non-negative
    if best_radii is not None:
        best_radii = np.maximum(best_radii, 0.0)

    if best_centers is not None:
        best_centers = np.clip(best_centers, 0.0, 1.0)
        
    # If we found nothing (should not happen), return dummy
    if best_centers is None:

        best_centers = np.random.rand(N, 2) * 0.8 + 0.1
        best_radii = np.full(N, 0.01)
        best_sum = np.sum(best_radii)

    return (best_centers, best_radii, float(best_sum))
