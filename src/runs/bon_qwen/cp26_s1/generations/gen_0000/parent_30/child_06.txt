from scipy.optimize import minimize
import math

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    Pack 26 circles in a unit square to maximize the sum of radii.

        centers: np.array of shape (26, 2)
        radii: np.array of shape (26,)

        # vars_flat: [x1, y1, r1, x2, y2, r2, ...]

        return -np.sum(radii) # Minimize negative sum to maximize sum

        constraints = []
        for i in range(n):
            x = vars_flat[3*i]
            y = vars_flat[3*i + 1]
            r = vars_flat[3*i + 2]
            # x - r >= 0 => r - x <= 0
            constraints.append(r - x)
            # x + r <= 1 => x + r - 1 <= 0
            constraints.append(x + r - 1)
            # y - r >= 0 => r - y <= 0
            constraints.append(r - y)
            # y + r <= 1 => y + r - 1 <= 0
            constraints.append(y + r - 1)
        return np.array(constraints)

    def distance_constraints(vars_flat):
        constraints = []
        for i in range(n):
            for j in range(i + 1, n):
                x1, y1 = vars_flat[3*i], vars_flat[3*i + 1]
                x2, y2 = vars_flat[3*j], vars_flat[3*j + 1]
                r1, r2 = vars_flat[3*i + 2], vars_flat[3*j + 2]
                
                dist_sq = (x1 - x2)**2 + (y1 - y2)**2
                # dist >= r1 + r2  =>  (r1+r2)^2 <= dist_sq  =>  (r1+r2)^2 - dist_sq <= 0
                # Note: This constraint is non-convex, but SLSQP can handle it locally.
                val = (r1 + r2)**2 - dist_sq
                constraints.append(val)
        return np.array(constraints)

        c1 = boundary_constraints(vars_flat)
        c2 = distance_constraints(vars_flat)

            b.append((0, 1))
