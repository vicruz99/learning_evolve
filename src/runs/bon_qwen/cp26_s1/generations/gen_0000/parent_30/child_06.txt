import numpy as np
from scipy.optimize import minimize

def objective(x):
    """Objective: minimize negative sum of radii."""
    return -np.sum(x[2::3])

def constraints_val(x):
    """Evaluate all inequality constraints. Returns array >= 0 for feasibility."""
    n = 26
    c = []
    # Boundary constraints
    for i in range(n):
        xi, yi, ri = x[3*i], x[3*i+1], x[3*i+2]
        c.extend([
            ri,                   # r >= 0
            xi - ri,              # x - r >= 0
            1 - xi - ri,          # 1 - x - r >= 0  => x + r <= 1
            yi - ri,              # y - r >= 0
            1 - yi - ri           # 1 - y - r >= 0  => y + r <= 1
        ])
    # Overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            xi, yi, ri = x[3*i], x[3*i+1], x[3*i+2]
            xj, yj, rj = x[3*j], x[3*j+1], x[3*j+2]
            dist_sq = (xi - xj)**2 + (yi - yj)**2
            sum_r_sq = (ri + rj)**2
            c.append(dist_sq - sum_r_sq)
    return np.array(c)

def constraints_jac(x):
    """Compute Jacobian matrix of constraints."""
    n = 26
    m_b = 5 * n
    m_o = n * (n - 1) // 2
    jac = np.zeros((m_b + m_o, 3 * n))
    
    # Boundary constraints jacobian
    for i in range(n):
        base = 5 * i
        jac[base, 3*i+2] = 1.0
        jac[base+1, 3*i] = 1.0
        jac[base+1, 3*i+2] = -1.0
        jac[base+2, 3*i] = -1.0
        jac[base+2, 3*i+2] = -1.0
        jac[base+3, 3*i+1] = 1.0
        jac[base+3, 3*i+2] = -1.0
        jac[base+4, 3*i+1] = -1.0
        jac[base+4, 3*i+2] = -1.0
        
    # Overlap constraints jacobian
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            xi, yi, ri = x[3*i], x[3*i+1], x[3*i+2]

            dx = xi - xj
            dy = yi - yj
            sr = ri + rj
            
            row = m_b + idx
            jac[row, 3*i] = 2 * dx
            jac[row, 3*i+1] = 2 * dy
            jac[row, 3*i+2] = -2 * sr
            jac[row, 3*j] = -2 * dx
            jac[row, 3*j+1] = -2 * dy
            jac[row, 3*j+2] = -2 * sr
            idx += 1
    return jac
