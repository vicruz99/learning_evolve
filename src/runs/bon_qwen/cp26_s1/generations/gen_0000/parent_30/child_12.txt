import numpy as np
from scipy.optimize import minimize

# Precompute indices for pairwise overlap constraints (i < j)
N_CIRCLES = 26
TRI_IDX = np.triu_indices(N_CIRCLES, k=1)

def objective(v):
    """Minimize negative sum of radii"""
    return -np.sum(v[2::3])

def constraint_bounds(v):
    """Boundary constraints: circles must stay inside [0,1]^2"""
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    # x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
    return np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])

def constraint_overlaps(v):
    """Non-overlap constraints: squared distance >= squared sum of radii"""
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    
    # Vectorized pairwise differences
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dist_sq = dx**2 + dy**2
    
    r_sum = r[:, None] + r[None, :]
    return dist_sq[TRI_IDX] - r_sum[TRI_IDX]**2

def run_packing():
    n = N_CIRCLES
    np.random.seed(42)  # For reproducibility
    
    # 1. Hexagonal lattice initialization
    centers = []
    y = 0.5
    shift = 0.0
    while len(centers) < n:
        x = 0.5
        while x <= 1.5 and len(centers) < n:
            if 0 <= x + shift <= 1:
                centers.append([x + shift, y])
            x += 0.2
        y += 0.17320508075688772  # sqrt(3)/2 * spacing
        shift = 0.1 if shift == 0.0 else 0.0
        
    centers = np.array(centers[:n])
    # Add small perturbation to break symmetry and aid optimization
    centers += np.random.randn(n, 2) * 0.005
    centers = np.clip(centers, 0.1, 0.9)
    
    # Initial feasible radii
    r_init = 0.05
    
    # Flatten variables: [x1, y1, r1, x2, y2, r2, ...]
    x0 = np.zeros(3*n)
    x0[0::3] = centers[:, 0]
    x0[1::3] = centers[:, 1]
    x0[2::3] = r_init
    
    bounds = [(0.0, 1.0)]*n + [(0.0, 1.0)]*n + [(0.0, 0.5)]*n
    
    constraints = [
        {'type': 'ineq', 'fun': constraint_bounds},
        {'type': 'ineq', 'fun': constraint_overlaps}
    ]
    
    # 2. Optimize
    res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints,
                   options={'maxiter': 5000, 'ftol': 1e-12, 'disp': False})
                   
    x_final = res.x
    
    # 3. Extract results
    c = np.column_stack((x_final[0::3], x_final[1::3]))
    r = np.maximum(x_final[2::3], 0.0)  # Ensure non-negative
    
    return c, r, np.sum(r)
