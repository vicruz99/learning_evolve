import numpy as np

def generate_hexagonal_grid(n, r_guess):
    """Generate an initial hexagonal grid layout for n circles."""
    # Approximate spacing for hexagonal packing
    # Horizontal spacing: 2r, Vertical spacing: r * sqrt(3)
    # We want to fit n circles.
    # Estimate dimensions.
    # Area per circle in hex packing is approx 2r * r*sqrt(3) = r^2 * sqrt(3) * 2? 
    # Actually area of unit cell is 2 * sqrt(3) * r^2.
    # n * 2 * sqrt(3) * r^2 approx 1 => r approx sqrt(1 / (2*sqrt(3)*n))
    
    # Let's just create a grid dense enough to hold n points
    # 5 rows of 5 or 6 points usually works for n=26
    cols = 6
    rows = 5
    
    centers = []
    x_spacing = 1.0 / cols
    y_spacing = 1.0 / rows
    
    # Generate points
    for row in range(rows):
        for col in range(cols):
            if len(centers) >= n:
                break
            # Hexagonal shift for odd rows
            shift = 0.5 * x_spacing if row % 2 == 1 else 0
            x = (col + 0.5) * x_spacing + shift
            y = (row + 0.5) * y_spacing
            
            # Clamp to safe interior
            x = np.clip(x, 0.1, 0.9)
            y = np.clip(y, 0.1, 0.9)
            
            centers.append([x, y])
        if len(centers) >= n:
            break
            
    return np.array(centers)

def check_overlap(centers, r):
    """Calculate total overlap energy for circles of radius r."""
    n = centers.shape[0]
    energy = 0.0
    
    # Pairwise overlaps
    # Vectorized distance calculation
    # diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :] # (n, n, 2)
    # dist = np.linalg.norm(diff, axis=2)
    # To save memory and time for n=26, simple loops are fine, or optimized logic.
    
    # Boundary overlaps
    x = centers[:, 0]
    y = centers[:, 1]
    
    # Left wall
    dist_left = x - r
    mask_left = dist_left < 0
    if np.any(mask_left):
        energy += np.sum(dist_left[mask_left]**2)
        
    # Right wall
    dist_right = (1.0 - x) - r
    mask_right = dist_right < 0
    if np.any(mask_right):
        energy += np.sum(dist_right[mask_right]**2)
        
    # Bottom wall
    dist_bottom = y - r
    mask_bottom = dist_bottom < 0
    if np.any(mask_bottom):
        energy += np.sum(dist_bottom[mask_bottom]**2)
        
    # Top wall
    dist_top = (1.0 - y) - r
    mask_top = dist_top < 0
    if np.any(mask_top):
        energy += np.sum(dist_top[mask_top]**2)

    # Pairwise
    # We iterate only upper triangle
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            d_sq = dx*dx + dy*dy
            d = np.sqrt(d_sq)
            min_dist = 2.0 * r
            if d < min_dist:
                overlap = min_dist - d
                energy += overlap**2
                
    return energy

def optimize_positions(centers, r, steps=1000):
    """Optimize positions to minimize overlap for fixed radius r."""
    n = centers.shape[0]
    # Copy centers
    pos = centers.copy()
    
    # Learning rate / step size
    # Adaptive step size could be better, but constant is simpler
    alpha = 0.05 
    
    for step in range(steps):
        forces = np.zeros_like(pos)
        
        # 1. Boundary forces
        # Left
        penetration = r - pos[:, 0]
        mask = penetration > 0
        if np.any(mask):
            forces[mask, 0] += penetration[mask]
            
        # Right
        penetration = r - (1.0 - pos[:, 0])
        mask = penetration > 0
        if np.any(mask):
            forces[mask, 0] -= penetration[mask]
            
        # Bottom
        penetration = r - pos[:, 1]
        mask = penetration > 0
        if np.any(mask):
            forces[mask, 1] += penetration[mask]
            
        # Top
        penetration = r - (1.0 - pos[:, 1])
        mask = penetration > 0
        if np.any(mask):
            forces[mask, 1] -= penetration[mask]
            
        # 2. Pairwise repulsion
        for i in range(n):
            for j in range(i + 1, n):
                dx = pos[i, 0] - pos[j, 0]
                dy = pos[i, 1] - pos[j, 1]
                d_sq = dx*dx + dy*dy
                
                # Avoid division by zero
                if d_sq < 1e-10:
                    d_sq = 1e-10
                    dx = 0.01
                    dy = 0.01
                
                d = np.sqrt(d_sq)
                min_dist = 2.0 * r
                
                if d < min_dist:
                    overlap = min_dist - d
                    # Force magnitude proportional to overlap
                    # Normalize direction
                    fx = (dx / d) * overlap
                    fy = (dy / d) * overlap
                    
                    forces[i, 0] += fx
                    forces[i, 1] += fy
                    forces[j, 0] -= fx
                    forces[j, 1] -= fy
        
        # Update positions
        pos += alpha * forces
        
        # Clamp to keep inside square (strictly speaking, optimization handles it, 
        # but clamping prevents numerical instability)
        pos = np.clip(pos, r, 1.0 - r)
        
    return pos

def is_valid_packing(centers, r):
    """Check if packing is valid (overlap < tolerance)."""
    # Re-calculate energy to check
    energy = check_overlap(centers, r)
    # Tolerance for numerical errors
    return energy < 1e-8
