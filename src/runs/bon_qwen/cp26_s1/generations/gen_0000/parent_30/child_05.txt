import numpy as np

def compute_overlap_penalty(centers, radii, n):
    """Compute total squared overlap and boundary violation."""
    penalty = 0.0
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < 0:
            penalty += (x - r) ** 2
        if x + r > 1:
            penalty += (x + r - 1) ** 2
        if y - r < 0:
            penalty += (y - r) ** 2
        if y + r > 1:
            penalty += (y + r - 1) ** 2
    
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i][0] - centers[j][0]
            dy = centers[i][1] - centers[j][1]
            dist = math.sqrt(dx * dx + dy * dy)
            overlap = radii[i] + radii[j] - dist
            if overlap > 0:
                penalty += overlap ** 2
    
    return penalty

def make_hex_grid(n, seed=0):
    """Create a hexagonal-like initial arrangement."""
    np.random.seed(seed)
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.02
    
    rows = [6, 5, 6, 5, 4]
    idx = 0
    for i, count in enumerate(rows):
        y = (i + 1) * (1.0 / (len(rows) + 1))
        for j in range(count):
            if idx >= n:
                break
            if i % 2 == 0:
                x = (j + 1) * (1.0 / (count + 1))
            else:
                x = (j + 1.5) * (1.0 / (count + 2))
            centers[idx][0] = max(0.05, min(0.95, x))
            centers[idx][1] = max(0.05, min(0.95, y))
            idx += 1
    
    while idx < n:
        centers[idx][0] = np.random.uniform(0.1, 0.9)
        centers[idx][1] = np.random.uniform(0.1, 0.9)
        idx += 1
    
    return centers, radii

def expand_and_optimize(centers, radii, n, max_iter=50000):

def refine_positions(centers, radii, n, iterations=2000):

                improved = True

            dist = math.sqrt(dx * dx + dy * dy)
            if dist < radii[i] + radii[j] - 1e-9:
