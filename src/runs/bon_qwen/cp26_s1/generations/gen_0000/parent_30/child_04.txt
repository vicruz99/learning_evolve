import numpy as np

def _pack_simulation(init_centers, n, seed=0):
    """
    Iteratively expands radii and repels overlapping circles to maximize packing density.
    Uses a force-directed layout with expanding circles heuristic.
    """
    rng = np.random.RandomState(seed)
    centers = init_centers.copy()
    radii = np.full(n, 0.02)
    dt = 0.1
    decay = 0.997
    growth = 1.001
    
    for _ in range(6000):
        forces = np.zeros_like(centers)
        
        # Boundary forces: push circles inside [0,1]
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x < r: 
                forces[i, 0] += (r - x)
            elif x > 1 - r: 
                forces[i, 0] -= (x - (1 - r))
            if y < r: 
                forces[i, 1] += (r - y)
            elif y > 1 - r: 
                forces[i, 1] -= (y - (1 - r))
                
        # Pairwise repulsion: push overlapping circles apart
        for i in range(n):
            for j in range(i+1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                d2 = dx*dx + dy*dy
                r_sum_sq = (radii[i] + radii[j])**2
                
                if d2 < r_sum_sq:
                    dist = np.sqrt(d2)
                    overlap = radii[i] + radii[j] - dist
                    # Force proportional to overlap, inversely proportional to distance
                    f = overlap / (2.0 * (dist + 1e-9))
                    if dist > 1e-9:
                        fx, fy = dx/dist, dy/dist
                        forces[i, 0] += f * fx
                        forces[i, 1] += f * fy
                        forces[j, 0] -= f * fx
                        forces[j, 1] -= f * fy
                    else:
                        # Fallback for coincident centers
                        forces[i, 0] += rng.normal(0, 0.01)
                        forces[j, 0] -= rng.normal(0, 0.01)
                        
        # Update positions
        centers += forces * dt
        centers = np.clip(centers, 0, 1)
        
        # Gradually increase radii
        radii *= growth
        
        # Reduce step size for convergence
        dt *= decay
        
        # Add jitter to escape local minima during early phases
        if dt > 0.01:
            centers += rng.normal(0, 0.005, centers.shape)
            centers = np.clip(centers, 0, 1)
            
        # Stop if step size is negligible
        if dt < 1e-5:
            break
            
    return centers, radii
