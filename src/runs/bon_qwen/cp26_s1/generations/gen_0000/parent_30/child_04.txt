def create_constraints(n):
    """
    Creates a list of constraint dictionaries for scipy.optimize.minimize.
    Constraints:
    1. x_i - r_i >= 0
    2. 1 - x_i - r_i >= 0
    3. y_i - r_i >= 0
    4. 1 - y_i - r_i >= 0
    5. r_i >= 0
    6. (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
    """
    constraints = []
    
    # Boundary and non-negativity constraints
    for i in range(n):
        x_idx = 3 * i
        y_idx = 3 * i + 1
        r_idx = 3 * i + 2
        
        # x >= r
        constraints.append({'type': 'ineq', 'fun': lambda v, ix=x_idx, ir=r_idx: v[ix] - v[ir]})
        # 1 - x >= r  => 1 - x - r >= 0
        constraints.append({'type': 'ineq', 'fun': lambda v, ix=x_idx, ir=r_idx: 1.0 - v[ix] - v[ir]})
        # y >= r
        constraints.append({'type': 'ineq', 'fun': lambda v, iy=y_idx, ir=r_idx: v[iy] - v[ir]})

        # r >= 0
        constraints.append({'type': 'ineq', 'fun': lambda v, ir=r_idx: v[ir]})
