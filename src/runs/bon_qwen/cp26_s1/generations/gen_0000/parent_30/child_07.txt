import numpy as np

def run_packing():
    n = 26
    best_centers = None
    best_radii = None
    best_sum = 0.0

    # Try multiple random initializations to find a good local optimum
    for trial in range(20):
        np.random.seed(trial * 42 + 7)
        centers = np.random.rand(n, 2)
        radii = np.full(n, 0.005)

        step = 0.002
        for grow_step in range(5000):
            # Attempt to grow all radii
            radii += step

            # Relax positions to resolve overlaps and boundary violations
            for _ in range(50):
                # Vectorized pairwise differences and distances
                diff = centers[:, None, :] - centers[None, :, :]
                dists_sq = np.sum(diff**2, axis=2)
                dists = np.sqrt(np.maximum(dists_sq, 1e-18))
                
                # Required distances (sum of radii)
                req_dists = radii[:, None] + radii[None, :]
                overlaps = np.maximum(0.0, req_dists - dists)
                
                # If no significant overlaps, break early
                if np.max(overlaps) < 1e-9:
                    break

                # Compute repulsion forces
                # Direction: diff / dists
                # Magnitude: overlaps
                forces = diff * (overlaps / dists)[:, :, None]
                grad = np.sum(forces, axis=1)

                # Boundary repulsion forces
                bound_grad = np.zeros_like(centers)
                # Left & Right boundaries
                mask_left = centers[:, 0] < radii
                mask_right = centers[:, 0] > 1.0 - radii
                bound_grad[mask_left, 0] += radii[mask_left] - centers[mask_left, 0]
                bound_grad[mask_right, 0] -= centers[mask_right, 0] - (1.0 - radii[mask_right])
                
                # Bottom & Top boundaries
                mask_bottom = centers[:, 1] < radii
                mask_top = centers[:, 1] > 1.0 - radii
                bound_grad[mask_bottom, 1] += radii[mask_bottom] - centers[mask_bottom, 1]
                bound_grad[mask_top, 1] -= centers[mask_top, 1] - (1.0 - radii[mask_top])

                grad += bound_grad

                # Update positions
                centers -= 0.15 * grad
                # Strictly enforce boundaries
                centers = np.clip(centers, radii[:, None], 1.0 - radii[:, None])

            # Verify feasibility after relaxation
            dists = np.sqrt(np.sum((centers[:, None, :] - centers[None, :, :])**2, axis=2))
            max_ov = np.max(np.maximum(0.0, radii[:, None] + radii[None, :] - dists))
            
            min_dist_to_wall = np.minimum(
                np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                np.minimum(centers[:, 1], 1.0 - centers[:, 1])
            )
            max_wall_ov = np.max(np.maximum(0.0, radii - min_dist_to_wall))

            if max_ov > 1e-7 or max_wall_ov > 1e-7:
                # Failed to resolve, revert growth and reduce step
                radii -= step
                step *= 0.92
                if step < 1e-10:
                    break
            else:
                # Success, slightly increase step for efficiency
                step = min(step * 1.03, 0.005)

        current_sum = np.sum(radii)
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers.copy()
            best_radii = radii.copy()

    return best_centers, best_radii, best_sum
