import numpy as np
from scipy.optimize import minimize


def compute_max_radii(centers, max_iter=200, smooth_factor=0.5):
    """Compute maximum possible radii for given centers using iterative relaxation."""
    n = centers.shape[0]
    radii = np.ones(n) * 1e-6

    for iteration in range(max_iter):
        new_radii = np.zeros(n)
        for i in range(n):
            max_r = min(centers[i, 0], 1 - centers[i, 0],
                        centers[i, 1], 1 - centers[i, 1])
            for j in range(n):
                if i != j:
                    dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                    max_r = min(max_r, dist - radii[j])
            new_radii[i] = max(max_r, 1e-8)
        if iteration > max_iter // 2:
            radii = new_radii
        else:
            radii = radii * smooth_factor + new_radii * (1 - smooth_factor)

    return radii


def generate_hexagonal_pattern(n, rows, cols, x_start, y_start, x_spacing, y_spacing, offset_even):
    """Generate a hexagonal-like packing pattern."""
    centers = np.zeros((n, 2))
    idx = 0
    for row in range(rows):
        n_cols = cols if row % 2 == 0 else cols
        for col in range(n_cols):
            if idx < n:
                x = x_start + col * x_spacing + (row % 2) * offset_even
                y = y_start + row * y_spacing
                centers[idx] = [x, y]
                idx += 1
    while idx < n:
        centers[idx] = [0.5, 0.5 + 0.05 * (idx - n + 1)]
        idx += 1
    return centers


def generate_grid_pattern(n, rows, cols, x_start, y_start, x_spacing, y_spacing):
    """Generate a rectangular grid pattern."""
    centers = np.zeros((n, 2))
    idx = 0
    for row in range(rows):
        for col in range(cols):
            if idx < n:
                centers[idx] = [x_start + col * x_spacing, y_start + row * y_spacing]
                idx += 1
    while idx < n:
        centers[idx] = [0.5, 0.5 + 0.05 * (idx - n + 1)]
        idx += 1
    return centers


def generate_candidates(n):
    """Generate multiple initial candidate configurations."""
    candidates = []

    # Hexagonal patterns with different spacings
    candidates.append(generate_hexagonal_pattern(n, 6, 5, 0.08, 0.08, 0.18, 0.18 * np.sqrt(3) / 2, 0.09))
    candidates.append(generate_hexagonal_pattern(n, 6, 5, 0.06, 0.06, 0.20, 0.20 * np.sqrt(3) / 2, 0.10))
    candidates.append(generate_hexagonal_pattern(n, 5, 6, 0.06, 0.06, 0.16, 0.20, 0.08))
    candidates.append(generate_hexagonal_pattern(n, 6, 5, 0.05, 0.05, 0.22, 0.22 * np.sqrt(3) / 2, 0.11))

    # Grid patterns
    candidates.append(generate_grid_pattern(n, 5, 5, 0.08, 0.08, 0.18, 0.18))
    candidates.append(generate_grid_pattern(n, 6, 5, 0.06, 0.06, 0.16, 0.16))
    candidates.append(generate_grid_pattern(n, 5, 6, 0.06, 0.06, 0.18, 0.18))

    # Staggered grid
    centers = np.zeros((n, 2))
    idx = 0
    for row in range(5):
        for col in range(5):
            if idx < n:
                centers[idx] = [0.07 + col * 0.18 + (row % 2) * 0.09, 0.07 + row * 0.19]
                idx += 1
    while idx < n:
        centers[idx] = [0.5, 0.5 + 0.05 * (idx - n + 1)]
        idx += 1
    candidates.append(centers)

    # Random perturbed versions of best patterns
    for base in candidates[:3]:
        perturbed = base + np.random.normal(0, 0.02, size=base.shape)
        perturbed = np.clip(perturbed, 0.01, 0.99)
        candidates.append(perturbed)

    return candidates


def evaluate_configuration(centers, n):
    """Evaluate a configuration by computing sum of radii."""
    radii = compute_max_radii(centers, 150)
    return np.sum(radii), radii


def optimize_centers(centers, n):
    """Optimize centers using L-BFGS-B to maximize sum of radii."""
    def objective(x):
        c = x.reshape(n, 2)
        r = compute_max_radii(c, 80)
        return -np.sum(r)

    x0 = centers.flatten()
    bounds = [(0.001, 0.999)] * (2 * n)

    result = minimize(objective, x0, method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': 8000, 'ftol': 1e-15, 'gtol': 1e-12})
    return result.x.reshape(n, 2)


def run_packing():
    n = 26
    np.random.seed(123)

    # Generate multiple candidate configurations
    candidates = generate_candidates(n)

    # Evaluate and select best initial configuration
    best_centers = None
    best_sum = 0
    for c in candidates:
        s, r = evaluate_configuration(c, n)
        if s > best_sum:
            best_sum = s
            best_centers = c.copy()

    centers = best_centers

    # Phase 1: First optimization pass
    centers = optimize_centers(centers, n)
    radii = compute_max_radii(centers, 300)
    sum_radii = np.sum(radii)

    # Phase 2: Multiple restarts from perturbed versions
    best_result = (centers.copy(), radii.copy(), sum_radii)

    for restart in range(5):
        # Perturb the best solution
        perturbed = centers + np.random.normal(0, 0.01, size=centers.shape)
        perturbed = np.clip(perturbed, 0.01, 0.99)

        # Optimize from perturbed position
        opt_centers = optimize_centers(perturbed, n)
        opt_radii = compute_max_radii(opt_centers, 300)
        opt_sum = np.sum(opt_radii)

        if opt_sum > best_result[2]:
            best_result = (opt_centers.copy(), opt_radii.copy(), opt_sum)

    centers, radii, sum_radii = best_result

    # Phase 3: Final refinement with more careful radius computation
    radii = compute_max_radii(centers, 500)

    # Ensure no negative radii
    radii = np.maximum(radii, 1e-8)

    return centers, radii, np.sum(radii)
