import numpy as np
from scipy.optimize import minimize

def constraint_boundary_low(z, i, coord):
    """Constraint: coord_i - r_i >= 0"""
    idx = 3 * i + coord
    return z[idx] - z[idx + 2]

def constraint_boundary_high(z, i, coord):
    """Constraint: 1 - coord_i - r_i >= 0"""
    idx = 3 * i + coord
    return 1.0 - z[idx] - z[idx + 2]

def constraint_overlap(z, i, j):
    """Constraint: dist_ij^2 - (r_i + r_j)^2 >= 0"""
    xi, yi, ri = z[3*i], z[3*i+1], z[3*i+2]
    xj, yj, rj = z[3*j], z[3*j+1], z[3*j+2]
    return (xi - xj)**2 + (yi - yj)**2 - (ri + rj)**2

def objective_function(z, n):
    """Minimize negative sum of radii"""
    radii = z[2::3]
    return -np.sum(radii)

def get_initial_configuration(n=26, perturbation=0.0):
    """Generate initial hexagonal grid positions"""
    rows_config = [5, 5, 5, 5, 5, 1]
    centers = []
    y_spacing = np.sqrt(3)/2 * 0.18
    x_spacing = 0.18
    
    y_curr = 0.5
    for i, count in enumerate(rows_config):
        # Horizontal offset for staggered rows
        shift = x_spacing / 2.0 if i % 2 == 1 else 0.0
        x_start = 0.5 - (count - 1) * x_spacing / 2.0 + shift
        
        for j in range(count):
            cx = x_start + j * x_spacing
            cy = y_curr
            if perturbation != 0.0:
                cx += np.random.uniform(-perturbation, perturbation)
                cy += np.random.uniform(-perturbation, perturbation)
            centers.append([cx, cy])
        y_curr += y_spacing if i < len(rows_config) - 1 else 0.0
        
    centers = np.array(centers)
    # Normalize to fit roughly inside [0,1]x[0,1]
    min_xy = centers.min(axis=0)
    max_xy = centers.max(axis=0)
    span = max_xy - min_xy
    if span[0] > 0: centers[:, 0] = (centers[:, 0] - min_xy[0]) / span[0]
    if span[1] > 0: centers[:, 1] = (centers[:, 1] - min_xy[1]) / span[1]
    
    radii = np.ones(n) * 0.075

def setup_optimization(z0, n):
    """Prepare bounds and constraints for scipy.optimize"""
    bounds = [(0.0, 1.0)] * (2 * n) + [(1e-9, 0.5)] * n

        cons.append({'type': 'ineq', 'fun': constraint_boundary_low, 'args': (i, 0)})
        cons.append({'type': 'ineq', 'fun': constraint_boundary_high, 'args': (i, 0)})
        cons.append({'type': 'ineq', 'fun': constraint_boundary_low, 'args': (i, 1)})
        cons.append({'type': 'ineq', 'fun': constraint_boundary_high, 'args': (i, 1)})
        
    # Overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            cons.append({'type': 'ineq', 'fun': constraint_overlap, 'args': (i, j)})
            
    return bounds, cons

def run_packing():
    np.random.seed(42) # For reproducibility
    n = 26
    
    best_z = None
    best_val = -np.inf

    for trial in range(2):
