                break

    radii = np.ones(n) * 0.03

    expansion_rate = 0.008

            max_r = min(
                centers[i, 0],
                1.0 - centers[i, 0],
                centers[i, 1],
                1.0 - centers[i, 1]
            )

                if i != j:
                    dx = centers[i, 0] - centers[j, 0]
                    dy = centers[i, 1] - centers[j, 1]
                    dist = np.sqrt(dx * dx + dy * dy)
                    max_r = min(max_r, dist - radii[j])

                radii[i] = min(max_r, radii[i] * (1.0 + expansion_rate))

        for i in range(n):
            for j in range(i + 1, n):

                    centers[i, 0] += scale * dx
                    centers[i, 1] += scale * dy

                    centers[j, 1] -= scale * dy
