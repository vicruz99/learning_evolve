To pack 26 circles in a unit square maximizing sum of radii (target: 2.636), I'll use a multi-stage optimization approach:

1. **Initial Configuration**: Start with a hexagonal-like arrangement, which is more space-efficient than square grids. For 26 circles, I'll try multiple row configurations (e.g., 6+5+6+5+4=26, 7+6+6+7=26, etc.)

2. **Optimization Method**: Use scipy.optimize with SLSQP method. The key insight is to encode all constraints (boundary + non-overlap) into a single constraint function returning a vector, avoiding lambda/closure issues.

3. **Multiple Restarts**: Run optimization from different initial configurations to escape local optima.

4. **Refinement**: After initial optimization, run additional refinement passes to push radii higher.

5. **Physics-based Pre-optimization**: Before formal optimization, use a force-directed simulation to spread circles apart and inflate radii, providing a better starting point.

The constraint formulation uses squared distances for non-overlap to avoid square root numerical issues: dist² >= (r₁ + r₂)².

import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26


def compute_constraints(x):
    """Compute all constraint values. Positive means satisfied."""
    n = N_CIRCLES
    constraints = []
    
    for i in range(n):
        xi = x[2 * i]
        yi = x[2 * i + 1]
        ri = x[2 * n + i]
        
        constraints.append(xi - ri)
        constraints.append(1.0 - xi - ri)
        constraints.append(yi - ri)
        constraints.append(1.0 - yi - ri)
        constraints.append(ri)
    
    for i in range(n):
        for j in range(i + 1, n):
            xi, yi = x[2 * i], x[2 * i + 1]
            xj, yj = x[2 * j], x[2 * j + 1]
            ri, rj = x[2 * n + i], x[2 * n + j]
            
            dist_sq = (xi - xj) ** 2 + (yi - yj) ** 2
            r_sum_sq = (ri + rj) ** 2
            constraints.append(dist_sq - r_sum_sq)
    
    return np.array(constraints)


def objective(x):
    """Minimize negative sum of radii."""
    n = N_CIRCLES
    return -np.sum(x[2 * n:])


def make_initial_config_hex(row_counts):
    """Create hexagonal-like initial configuration."""
    n = N_CIRCLES
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.085)
    
    idx = 0
    num_rows = len(row_counts)
    
    for row_i, count in enumerate(row_counts):
        y = 0.15 + row_i * 0.155
        for col_j in range(count):
            spacing = 1.0 / (count + 1)
            x = spacing * (col_j + 1)
            if row_i % 2 == 1:
                x += spacing * 0.5
            centers[idx] = [x, y]
            idx += 1
    
    return centers, radii


def make_initial_config_grid():
    """Create 5x5 grid + 1 corner configuration."""
    n = N_CIRCLES
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.09)
    
    for i in range(5):
        for j in range(5):
            centers[i * 5 + j] = [0.1 + j * 0.2, 0.1 + i * 0.2]
    
    centers[25] = [0.05, 0.95]
    radii[25] = 0.05
    
    return centers, radii


def make_initial_config_dense():
    """Create a dense packing starting configuration."""
    n = N_CIRCLES
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.08)
    
    row_counts = [7, 6, 7, 6]
    idx = 0
    
    for row_i, count in enumerate(row_counts):
        y = 0.12 + row_i * 0.18
        for col_j in range(count):
            spacing = 0.9 / (count + 1)
            x = spacing * (col_j + 1)
            if row_i % 2 == 1:
                x += spacing * 0.5
            centers[idx] = [x, y]
            idx += 1
    
    return centers, radii


def physics_based_inflation(centers, radii, num_iterations=5000):
    """Use force-directed simulation to improve packing."""
    n = N_CIRCLES
    centers = centers.copy()
    radii = radii.copy()
    
    dt = 0.005
    
    for iteration in range(num_iterations):
        forces = np.zeros((n, 2))
        
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[j, 0] - centers[i, 0]
                dy = centers[j, 1] - centers[i, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                
                min_dist = radii[i] + radii[j] + 0.005
                
                if dist < min_dist and dist > 1e-10:
                    overlap = min_dist - dist
                    fx = dx / dist * overlap * 50
                    fy = dy / dist * overlap * 50
                    forces[i, 0] -= fx
                    forces[i, 1] -= fy
                    forces[j, 0] += fx
                    forces[j, 1] += fy
        
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            
            if x - r < 0:
                forces[i, 0] += (r - x + 0.01) * 200
            if x + r > 1:
                forces[i, 0] -= (x + r - 1 + 0.01) * 200
            if y - r < 0:
                forces[i, 1] += (r - y + 0.01) * 200
            if y + r > 1:
                forces[i, 1] -= (y + r - 1 + 0.01) * 200
        
        centers += forces * dt
        
        centers[:, 0] = np.clip(centers[:, 0], 0.001, 0.999)
        centers[:, 1] = np.clip(centers[:, 1], 0.001, 0.999)
        
        if iteration % 50 == 0:
            radii *= 1.0005
    
    return centers, radii


def optimize_from_initial(centers, radii, max_iter=5000):
    """Run SLSQP optimization from given initial configuration."""
    n = N_CIRCLES
    x0 = np.concatenate([centers.flatten(), radii])
    
    bounds = [(0.0, 1.0)] * (2 * n) + [(1e-10, 0.5)] * n
    
    constraint = {
        'type': 'ineq',
        'fun': compute_constraints
    }
    
    try:
        result = minimize(
            objective, x0, method='SLSQP', bounds=bounds,
            constraints=[constraint],
            options={'maxiter': max_iter, 'ftol': 1e-15, 'disp': False}
        )
        
        centers_opt = result.x[:2 * n].reshape((n, 2))
        radii_opt = np.maximum(result.x[2 * n:], 0)
        
        return centers_opt, radii_opt, np.sum(radii_opt)
    except Exception:
        return centers, radii, np.sum(radii)


def refine_packing(centers, radii, num_refinements=3):
    """Iteratively refine packing by re-optimizing."""
    best_sum = np.sum(radii)
    best_centers = centers.copy()
    best_radii = radii.copy()
    
    for ref in range(num_refinements):
        centers_phys, radii_phys = physics_based_inflation(
            best_centers, best_radii, num_iterations=2000
        )
        
        centers_opt, radii_opt, current_sum = optimize_from_initial(
            centers_phys, radii_phys, max_iter=3000
        )
        
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers_opt.copy()
            best_radii = radii_opt.copy()
    
    return best_centers, best_radii, best_sum


def run_packing():
    """Main function to pack 26 circles in unit square."""
    best_sum = 0
    best_centers = None
    best_radii = None
    
    # Strategy 1: Hexagonal with 6+5+6+5+4
    row_configs = [

        [6, 6, 6, 5, 3],
        [7, 7, 6, 6],
        [5, 6, 5, 6, 4],
    ]
    
    for row_counts in row_configs:
        if sum(row_counts) != N_CIRCLES:
            continue
        
        centers, radii = make_initial_config_hex(row_counts)
        centers_phys, radii_phys = physics_based_inflation(centers, radii, 3000)
        centers_opt, radii_opt, current_sum = optimize_from_initial(
            centers_phys, radii_phys, max_iter=4000
        )
        
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers_opt.copy()
            best_radii = radii_opt.copy()
    
    # Strategy 2: Grid-based
    centers, radii = make_initial_config_grid()
    centers_phys, radii_phys = physics_based_inflation(centers, radii, 3000)
    centers_opt, radii_opt, current_sum = optimize_from_initial(
        centers_phys, radii_phys, max_iter=4000
    )
    
    if current_sum > best_sum:
        best_sum = current_sum
        best_centers = centers_opt.copy()
        best_radii = radii_opt.copy()
    
    # Strategy 3: Dense packing
    centers, radii = make_initial_config_dense()
    centers_phys, radii_phys = physics_based_inflation(centers, radii, 3000)
    centers_opt, radii_opt, current_sum = optimize_from_initial(
        centers_phys, radii_phys, max_iter=4000
    )
    
    if current_sum > best_sum:
        best_sum = current_sum
        best_centers = centers_opt.copy()
        best_radii = radii_opt.copy()
    
    # Final refinement
    if best_centers is not None:
        best_centers, best_radii, best_sum = refine_packing(
            best_centers, best_radii, num_refinements=2
        )
    
    return best_centers, best_radii, best_sum
