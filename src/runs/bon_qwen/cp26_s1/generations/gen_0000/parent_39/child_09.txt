import numpy as np
import scipy.optimize as opt
import itertools

# Constants
N_CIRCLES = 26
PENALTY_WEIGHT = 10000.0  # High weight to enforce constraints strictly
EPSILON = 1e-9            # Small value to avoid division by zero

def get_dist_and_grad(vec, i, j):
    """
    Calculate distance and gradient components for circle pair (i, j).
    vec is the flattened vector [x0, y0, r0, x1, y1, r1, ...]
    Returns (dist, grad_dist_i, grad_dist_j)
    grad_dist_i is a tuple (dx, dy, dr) for circle i
    """
    xi, yi, ri = vec[3*i], vec[3*i+1], vec[3*i+2]
    xj, yj, rj = vec[3*j], vec[3*j+1], vec[3*j+2]
    
    dx = xi - xj
    dy = yi - yj
    dist_sq = dx*dx + dy*dy
    dist = np.sqrt(dist_sq)
    
    # Avoid division by zero
    if dist < EPSILON:
        dist = EPSILON
        inv_dist = 1.0 / EPSILON
    else:
        inv_dist = 1.0 / dist
        
    # Gradient of distance w.r.t coordinates
    # d(dist)/d(xi) = (xi - xj) / dist
    # d(dist)/d(yi) = (yi - yj) / dist
    # d(dist)/d(ri) = 0
    
    gxi = dx * inv_dist
    gyi = dy * inv_dist
    gri = 0.0
    
    gxj = -gxi
    gyj = -gyi
    grj = 0.0
    
    return dist, (gxi, gyi, gri), (gxj, gyj, grj)

def objective_and_grad(vec):
    """
    Computes the objective function (negative sum of radii + penalties)
    and its gradient.
    
    vec: array of shape (78,) -> [x0, y0, r0, x1, y1, r1, ...]
    """
    n = len(vec)
    sum_radii = 0.0
    penalty = 0.0
    grad = np.zeros(n)
    
    # 1. Objective: Maximize sum of radii -> Minimize -sum_radii
    # Grad contribution: -1 for each r component
    for i in range(N_CIRCLES):
        r = vec[3*i + 2]
        sum_radii += r
        grad[3*i + 2] -= 1.0  # Derivative of -r is -1

    # 2. Boundary Penalties
    # Constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    # i.e., r - x <= 0, r - (1-x) <= 0, etc.
    # Penalty: max(0, constraint)^2
    
    for i in range(N_CIRCLES):
        xi = vec[3*i]
        yi = vec[3*i+1]
        ri = vec[3*i+2]
        
        # Left boundary: x >= r => r - x <= 0
        val_l = ri - xi
        if val_l > 0:
            penalty += val_l**2
            # d/d(ri) = 2*val_l
            # d/d(xi) = -2*val_l
            grad[3*i + 2] -= 2.0 * val_l * PENALTY_WEIGHT
            grad[3*i] += 2.0 * val_l * PENALTY_WEIGHT
            
        # Right boundary: 1-x >= r => r - (1-x) <= 0 => r + x - 1 <= 0
        val_r = ri + xi - 1.0
        if val_r > 0:
            penalty += val_r**2
            grad[3*i + 2] -= 2.0 * val_r * PENALTY_WEIGHT
            grad[3*i] -= 2.0 * val_r * PENALTY_WEIGHT
            
        # Bottom boundary: y >= r => r - y <= 0
        val_b = ri - yi
        if val_b > 0:
            penalty += val_b**2
            grad[3*i + 2] -= 2.0 * val_b * PENALTY_WEIGHT
            grad[3*i + 1] += 2.0 * val_b * PENALTY_WEIGHT
            
        # Top boundary: 1-y >= r => r + y - 1 <= 0
        val_t = ri + yi - 1.0
        if val_t > 0:
            penalty += val_t**2
            grad[3*i + 2] -= 2.0 * val_t * PENALTY_WEIGHT
            grad[3*i + 1] -= 2.0 * val_t * PENALTY_WEIGHT

    # 3. Overlap Penalties
    # Constraint: dist(i,j) >= ri + rj => ri + rj - dist <= 0
    # Penalty: max(0, ri + rj - dist)^2
    
    # We iterate over all pairs
    # To optimize, we only compute if potentially overlapping? 
    # But checking distance is cheap.
    
    for i in range(N_CIRCLES):
        for j in range(i + 1, N_CIRCLES):
            dist, (gxi, gyi, gri), (gxj, gyj, grj) = get_dist_and_grad(vec, i, j)
            ri = vec[3*i + 2]
            rj = vec[3*j + 2]
            
            val = ri + rj - dist
            if val > 0:
                penalty += val**2
                factor = 2.0 * val * PENALTY_WEIGHT
                
                # Gradient of val w.r.t variables
                # d(val)/d(ri) = 1
                # d(val)/d(rj) = 1
                # d(val)/d(xi) = -gxi  (since val = ... - dist, and d(dist)/dxi = gxi)
                # Wait, val = ri + rj - dist.
                # d(val)/dxi = - d(dist)/dxi = -gxi.
                
                # Gradient of -PENALTY_WEIGHT * val^2 (since we are minimizing obj + penalty)
                # Actually obj = -sum_r + Penalty.
                # We want gradient of Obj.
                # d(Obj)/d(ri) = -1 + d(Penalty)/d(ri)
                # d(Penalty)/d(ri) = sum(2*val * d(val)/d(ri))
                # d(val)/d(ri) = 1.
                
                # So we ADD to gradient (since we are computing gradient of function to minimize)
                # grad[ri_idx] += factor * 1
                
                grad[3*i + 2] += factor * 1.0
                grad[3*j + 2] += factor * 1.0
                
                grad[3*i] += factor * (-gxi)
                grad[3*i + 1] += factor * (-gyi)
                
                grad[3*j] += factor * (-gxj)
                grad[3*j + 1] += factor * (-gyj)

    return -sum_radii + PENALTY_WEIGHT * penalty, grad

def run_packing():
    # 1. Initialization
    # Generate a hexagonal lattice configuration
    # We need 26 circles.
    # Let's estimate a radius. If sum ~ 2.6, avg r ~ 0.1.
    # Let's start with r = 0.08.
    
    r_init = 0.08
    centers = []
    
    # Hexagonal packing logic
    # Rows shifted
    # y spacing = r * sqrt(3)
    # x spacing = 2r
    
    dy = r_init * np.sqrt(3)
    dx = 2 * r_init
    
    y = r_init
    row = 0
    while len(centers) < N_CIRCLES:
        x = r_init
        if row % 2 == 1:
            x = r_init + dx / 2.0  # Shifted row
        
        while x + r_init <= 1.0 and len(centers) < N_CIRCLES:
            centers.append([x, y, r_init])
            x += dx
        
        y += dy
        row += 1
    
    # Flatten to vector
    vec_init = np.array([])
    for c in centers:
        vec_init = np.append(vec_init, c)
    
    # Bounds for L-BFGS-B
    # x, y in [0, 1], r in [0, 0.5]
    bounds = []
    for _ in range(N_CIRCLES):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 0.5)) # r
        
    # 2. Optimization
    # Use L-BFGS-B
    res = opt.minimize(
        objective_and_grad,
        vec_init,
        method='L-BFGS-B',
        bounds=bounds,
        jac=True, # We provide gradient
        options={'maxiter': 2000, 'ftol': 1e-12}
    )
    
    best_vec = res.x
    
    # 3. Extract results
    centers_out = np.zeros((N_CIRCLES, 2))
    radii_out = np.zeros(N_CIRCLES)
    
    for i in range(N_CIRCLES):
        centers_out[i, 0] = best_vec[3*i]
        centers_out[i, 1] = best_vec[3*i+1]
        radii_out[i] = best_vec[3*i+2]
        
    # Ensure non-negative radii (should be handled by bounds, but clamp just in case)
    radii_out = np.maximum(radii_out, 0.0)
    
    # Shrink radii slightly to satisfy strict constraints with tolerance
    # The validation allows 1e-12 error, but numerical optimization might be on the edge.
    # Shrinking by a tiny epsilon ensures validity.
    shrink_factor = 1.0 - 1e-6
    radii_out *= shrink_factor
    
    sum_radii = np.sum(radii_out)
    
    return centers_out, radii_out, sum_radii

if __name__ == "__main__":
    centers, radii, total = run_packing()
    print(f"Sum of radii: {total}")
