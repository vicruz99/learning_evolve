1. **Hexagonal packing** is the densest arrangement for circles. I'll initialize with a staggered hexagonal grid.

2. **Force-based optimization**: Start with small circles and gradually grow them. When circles overlap or hit boundaries, repulsive forces push them apart. This naturally leads to an optimal packing where:
   - Corner circles grow larger (less constrained)
   - Edge circles are medium-sized
   - Interior circles are smaller (more constrained by neighbors)

3. **Multiple restarts** with different random seeds and perturbations to escape local minima.

4. **Adaptive parameters**: Learning rate and growth rate decrease over time for fine-tuning near the optimum.

5. **Cleanup**: After optimization, slightly shrink radii to ensure strict validity.

The force method works because:
- The "growth pressure" acts like a uniform pressure trying to expand all circles
- Repulsion from overlaps acts like hard-sphere interactions
- Boundary repulsion keeps circles inside the square
- The system converges to a configuration where circles are maximally packed

import numpy as np
import math


def create_hexagonal_init(n, spacing=None, offset_x=0, offset_y=0):
    """Create initial hexagonal arrangement of circles."""
    centers = np.zeros((n, 2))
    count = 0
    row = 0
    if spacing is None:
        spacing = math.sqrt((1.0 - 2 * 0.1) ** 2 / (math.sqrt(3) * n))
    
    y = 0.1 + offset_y
    while count < n:
        x = 0.1 + (row % 2) * spacing / 2 + offset_x
        while x <= 0.9 and count < n:
            centers[count, 0] = x
            centers[count, 1] = y
            count += 1
            x += spacing
        y += spacing * math.sqrt(3) / 2
        row += 1
    
    while count < n:
        centers[count, 0] = 0.5 + np.random.randn() * 0.05
        centers[count, 1] = 0.5 + np.random.randn() * 0.05
        count += 1
    
    return centers


def compute_forces(centers, radii):
    """Compute repulsion forces between circles and with boundaries."""
    n = len(radii)
    forces = np.zeros((n, 2))
    
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[j, 0] - centers[i, 0]
            dy = centers[j, 1] - centers[i, 1]
            dist = math.sqrt(dx * dx + dy * dy)
            min_dist = radii[i] + radii[j]
            
            if dist < min_dist and dist > 1e-12:
                overlap = min_dist - dist
                force_mag = 200.0 * overlap
                fx = force_mag * dx / dist
                fy = force_mag * dy / dist
                forces[i, 0] -= fx
                forces[i, 1] -= fy
                forces[j, 0] += fx
                forces[j, 1] += fy
        
        for dim in range(2):
            if centers[i, dim] - radii[i] < 0:
                forces[i, dim] += 500.0 * (radii[i] - centers[i, dim])
            if centers[i, dim] + radii[i] > 1.0:
                forces[i, dim] -= 500.0 * (centers[i, dim] + radii[i] - 1.0)
    
    return forces


def clamp_to_boundary(centers, radii):
    """Clamp centers to valid region."""
    n = len(radii)
    for i in range(n):
        for dim in range(2):
            centers[i, dim] = max(radii[i], min(1.0 - radii[i], centers[i, dim]))
    return centers


def check_validity(centers, radii, tol=1e-9):
    """Check if packing is valid."""
    n = len(radii)
    for i in range(n):
        if radii[i] < -tol:
            return False
        for dim in range(2):
            if centers[i, dim] - radii[i] < -tol:
                return False
            if centers[i, dim] + radii[i] > 1.0 + tol:
                return False
    for i in range(n):
        for j in range(i + 1, n):
            dist = math.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - tol:
                return False
    return True


def cleanup_overlaps(centers, radii, max_iterations=200):
    """Slightly shrink radii to eliminate small overlaps."""
    for _ in range(max_iterations):
        max_overlap = 0.0
        for i in range(len(radii)):
            for j in range(i + 1, len(radii)):
                dist = math.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                min_dist = radii[i] + radii[j]
                if dist < min_dist:
                    max_overlap = max(max_overlap, min_dist - dist)
        if max_overlap < 1e-12:
            break
        radii *= 0.9995
    return radii


def optimize_trial(centers_init, radii_init, steps=30000, seed=0):
    """Run force-based optimization for one trial."""
    np.random.seed(seed)
    centers = centers_init.copy()
    radii = radii_init.copy()
    
    lr = 0.002
    base_growth = 0.00012
    
    for step in range(steps):
        # Adaptive growth rate - high at start, tapering off
        if step < 8000:
            growth = base_growth
        elif step < 16000:
            growth = base_growth * 0.5
        elif step < 24000:
            growth = base_growth * 0.2
        else:
            growth = base_growth * 0.05
        
        radii += growth
        
        # Compute repulsion forces
        forces = compute_forces(centers, radii)
        
        # Apply forces with damping
        centers += forces * lr * 0.005
        
        # Clamp to valid region
        clamp_to_boundary(centers, radii)
        
        # Random perturbation to escape local minima
        if step % 3000 == 0 and step > 0:
            perturbation = np.random.normal(0, 0.008, centers.shape)
            centers += perturbation
            clamp_to_boundary(centers, radii)
    
    return centers, radii


def run_packing():
    """Pack 26 circles in unit square to maximize sum of radii."""
    n = 26
    
    best_centers = None
    best_radii = None
    best_sum = 0.0
    
    # Try multiple initial configurations and seeds
    for trial_idx in range(25):
        # Different initialization strategies
        if trial_idx < 8:
            # Hexagonal with different spacing
            spacing = 0.15 + trial_idx * 0.02
            centers_init = create_hexagonal_init(n, spacing=spacing)
        elif trial_idx < 16:
            # Grid-based initialization
            centers_init = np.zeros((n, 2))
            for i in range(n):
                row = i // 6
                col = i % 6
                centers_init[i, 0] = 0.08 + col * 0.15 + (row % 2) * 0.075
                centers_init[i, 1] = 0.08 + row * 0.18
        else:
            # Random initialization
            centers_init = np.random.uniform(0.1, 0.9, (n, 2))
        
        radii_init = np.full(n, 0.04)
        
        # Run optimization
        seed = trial_idx * 100 + 42
        centers, radii = optimize_trial(centers_init, radii_init, steps=35000, seed=seed)
        
        # Cleanup small overlaps
        radii = cleanup_overlaps(centers, radii)
        
        # Check validity and track best
        if check_validity(centers, radii):
            current_sum = np.sum(radii)
            if current_sum > best_sum:
                best_sum = current_sum
                best_centers = centers.copy()
                best_radii = radii.copy()
    
    # Fallback if no valid solution found (shouldn't happen)
    if best_centers is None:
        best_centers = np.zeros((n, 2))
        best_radii = np.full(n, 0.04)
        for i in range(n):
            row = i // 6
            col = i % 6
            best_centers[i, 0] = 0.05 + col * 0.15
            best_centers[i, 1] = 0.05 + row * 0.18
        best_sum = np.sum(best_radii)
    
    return best_centers, best_radii, best_sum
