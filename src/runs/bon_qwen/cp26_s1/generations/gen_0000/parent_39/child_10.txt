def compute_penalty(centers, radii, w_boundary=5000, w_overlap=5000):
    """Compute penalty for constraint violations."""
    n = len(radii)
    penalty = 0.0

        x, y = centers[i]
        r = radii[i]
        # Boundary violations
        penalty += w_boundary * (max(0, r - x)**2 + max(0, r - (1 - x))**2 +

            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            overlap = radii[i] + radii[j] - dist
            if overlap > 0:
                penalty += w_overlap * overlap**2

    return penalty


def objective(params, n):

    return obj


def initialize_hexagonal(n):

            break

            x_start = r
        else:

            centers.append([x, y])
