import numpy as np
from scipy.optimize import minimize

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    n = centers.shape[0]
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False
    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False
    return True

def objective(vars, n, penalty_factor):
    r = vars[:n]
    ux = vars[n:2*n]
    uy = vars[2*n:3*n]
    
    # Reparametrization: x = r + (1-2r)ux
    x = r + (1 - 2 * r) * ux
    y = r + (1 - 2 * r) * uy
    
    sum_r = np.sum(r)
    penalty = 0.0
    
    # Overlap penalties
    for i in range(n):
        for j in range(i + 1, n):
            dx = x[i] - x[j]
            dy = y[i] - y[j]
            dist = np.sqrt(dx*dx + dy*dy)
            sum_r_ij = r[i] + r[j]
            
            violation = sum_r_ij - dist
            if violation > 0:
                penalty += violation ** 2
                
    return -sum_r + penalty_factor * penalty

def jacobian(vars, n, penalty_factor):
    r = vars[:n]
    ux = vars[n:2*n]
    uy = vars[2*n:3*n]
    
    x = r + (1 - 2 * r) * ux
    y = r + (1 - 2 * r) * uy

    grad = np.zeros(3 * n)

    # Jacobian of reparametrization w.r.t r, ux, uy
    # dx/dr = 1 - 2ux, dx/dux = 1 - 2r
    # dy/dr = 1 - 2uy, dy/duy = 1 - 2r
    
    for i in range(n):
        for j in range(i + 1, n):
            dx = x[i] - x[j]
            dy = y[i] - y[j]
            dist = np.sqrt(dx*dx + dy*dy)
            sum_r_ij = r[i] + r[j]
            violation = sum_r_ij - dist
            
            if violation > 0:
                # Gradient of penalty term: 2 * violation * (d_violation/d_vars)
                # d_violation/d_r_i = 1 - dx/dx_i * (dx/dist) ... 
                # Actually, d_violation/dr = 1 - (x_i-x_j)/dist * dx_i/dr
                
                # Components for circle i
                dx_dr_i = 1 - 2 * ux[i]
                dx_dux_i = 1 - 2 * r[i]
                dy_dr_i = 1 - 2 * uy[i]
                dy_duy_i = 1 - 2 * r[i]
                
                # Components for circle j
                dx_dr_j = 1 - 2 * ux[j]
                dx_dux_j = 1 - 2 * r[j]
                dy_dr_j = 1 - 2 * uy[j]
                dy_duy_j = 1 - 2 * r[j]
                
                # Distance derivative parts
                d_dist_dx_i = dx / dist
                d_dist_dy_i = dy / dist
                d_dist_dx_j = -dx / dist
                d_dist_dy_j = -dy / dist
                
                factor = 2.0 * violation * penalty_factor
                
                # Circle i
                d_violation_dr_i = 1 - d_dist_dx_i * dx_dr_i - d_dist_dy_i * dy_dr_i
                d_violation_dux_i = - d_dist_dx_i * dx_dux_i
                d_violation_duy_i = - d_dist_dy_i * dy_duy_i
                
                grad[i] += factor * d_violation_dr_i
                grad[n + i] += factor * d_violation_dux_i
                grad[2*n + i] += factor * d_violation_duy_i
                
                # Circle j
                d_violation_dr_j = 1 - d_dist_dx_j * dx_dr_j - d_dist_dy_j * dy_dr_j
                d_violation_dux_j = - d_dist_dx_j * dx_dux_j

                grad[j] += factor * d_violation_dr_j
                grad[n + j] += factor * d_violation_dux_j
                grad[2*n + j] += factor * d_violation_duy_j

    return grad
