import numpy as np
from scipy.optimize import minimize

def objective(v):
    """Objective: minimize negative sum of radii (equivalent to maximizing sum of radii)"""
    return -np.sum(v[2::3])

def objective_jac(v):
    """Analytical Jacobian for the objective function"""
    jac = np.zeros_like(v)
    jac[2::3] = -1.0
    return jac

def constraints_fun(v):
    """
    Computes all inequality constraints: fun(v) >= 0
    Returns a 1D numpy array of constraint values.
    """
    n = 26
    c = []
    
    # Boundary and non-negativity constraints
    for i in range(n):
        c.append(v[3*i] - v[3*i+2])           # x >= r
        c.append(1.0 - v[3*i] - v[3*i+2])     # x <= 1 - r
        c.append(v[3*i+1] - v[3*i+2])         # y >= r
        c.append(1.0 - v[3*i+1] - v[3*i+2])   # y <= 1 - r
        c.append(v[3*i+2])                    # r >= 0
        
    # Non-overlap constraints using vectorized operations
    centers = v[:2*n].reshape(n, 2)
    radii = v[2*n:]
    
    # Compute squared distance matrix
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist_sq = np.sum(diff**2, axis=2)
    
    # Compute squared sum of radii matrix
    r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    
    # Extract strictly upper triangular elements to avoid duplicates/self
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    c.extend(dist_sq[mask] - r_sum[mask]**2)
    
    return np.array(c)

def run_packing():
    """
    Optimizes circle packing in a unit square to maximize sum of radii.
    Returns (centers, radii, sum_radii)
    """
    n = 26
    best_res = None
    best_val = -np.inf
    
    # Variable bounds: x, y in [0, 1], r in [0, 1]
    bounds = [(0.0, 1.0) for _ in range(2*n)] + [(0.0, 1.0) for _ in range(n)]
    cons = [{'type': 'ineq', 'fun': constraints_fun}]
    
    np.random.seed(42)
    
    # Multi-start strategy with different initial patterns
    for trial in range(10):
        pattern_type = trial % 3
        
        if pattern_type == 0:
            # Hexagonal lattice pattern
            pts = []
            row, col = 0, 0
            while len(pts) < n:
                x = 0.12 + col * 0.16
                y = 0.12 + row * 0.16

                    y += 0.08
                pts.append((x, y))
                col += 1
                if col > 5:
                    col = 0
                    row += 1
            init_centers = np.array(pts[:n]) + np.random.uniform(-0.015, 0.015, (n, 2))

            # Perturbed square grid

            pts = [(x, y) for x in gs for y in gs]
            np.random.shuffle(pts)
            init_centers = np.array(pts[:n]) + np.random.uniform(-0.02, 0.02, (n, 2))
            
        else:
            # Random initialization
            init_centers = np.random.uniform(0.15, 0.85, (n, 2))
            
        # Ensure initial points are safely inside
        init_centers = np.clip(init_centers, 0.05, 0.95)
        init_radii = np.full(n, 0.035)
        
        # Flatten to optimization variable vector: [x1, y1, r1, x2, y2, r2, ...]
        x0 = np.concatenate([init_centers.flatten(), init_radii])
        
        try:
            res = minimize(
                objective, x0, method='SLSQP', jac=objective_jac,
                bounds=bounds, constraints=cons,
                options={'maxiter': 2500, 'ftol': 1e-12, 'disp': False}

    radii = np.maximum(radii, 0.0)
    
    return centers, radii, float(np.sum(radii))
