import numpy as np
from scipy.optimize import minimize

def objective_function(vars, n, penalty):
    """
    Objective function: minimize -sum(radii) + penalty * constraint_violations
    """
    centers = vars[:2*n].reshape(n, 2)
    radii = vars[2*n:]
    
    sum_r = np.sum(radii)
    violation = 0.0
    
    # Boundary violations: circles must stay within [0,1]^2
    x, y = centers[:, 0], centers[:, 1]
    r = radii
    v_x0 = np.maximum(0.0, r - x)
    v_x1 = np.maximum(0.0, r - (1.0 - x))
    v_y0 = np.maximum(0.0, r - y)
    v_y1 = np.maximum(0.0, r - (1.0 - y))
    violation += np.sum(v_x0**2 + v_x1**2 + v_y0**2 + v_y1**2)
    
    # Overlap violations: distance between centers >= sum of radii
    dx = centers[:, 0, None] - centers[None, :, 0]
    dy = centers[:, 1, None] - centers[None, :, 1]
    dists = np.sqrt(dx**2 + dy**2)
    np.fill_diagonal(dists, np.inf)  # Ignore self-distance
    
    r_sums = radii[:, None] + radii[None, :]
    overlaps = np.maximum(0.0, r_sums - dists)
    violation += np.sum(overlaps**2)
    
    return -sum_r + penalty * violation

def check_validity(centers, radii, n):
    """Helper to strictly validate constraints without tolerance slack"""
    x, y = centers[:, 0], centers[:, 1]
    r = radii
    for i in range(n):
        if r[i] > x[i] or r[i] > 1.0 - x[i] or r[i] > y[i] or r[i] > 1.0 - y[i]:
            return False
    for i in range(n):
        for j in range(i + 1, n):
            d = np.sqrt((x[i]-x[j])**2 + (y[i]-y[j])**2)
            if d < r[i] + r[j] - 1e-10:
                return False
    return True

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    best_centers = None
    best_radii = None
    best_sum = 0.0
    
    # Hexagonal grid initialization for good starting density
    init_centers = np.zeros((n, 2))
    idx = 0
    row = 0
    while idx < n:
        for col in range(6):
            if idx >= n: break
            x = 0.14 + col * 0.145 + (0.0725 if row % 2 else 0.0)
            y = 0.14 + row * 0.115
            init_centers[idx] = [x, y]
            idx += 1
        row += 1
        
    init_radii = np.full(n, 0.09)
    
    # Bounds: centers in [0,1], radii in [epsilon, 0.5]
    bounds = [(0.0, 1.0)] * (2*n) + [(1e-5, 0.5)] * n
    
    # Multi-start optimization to escape local minima
    for seed in range(20):
        np.random.seed(seed)
        current_vars = np.concatenate([init_centers.flatten(), init_radii])
        current_vars += np.random.randn(3*n) * 0.03
        
        # Enforce bounds explicitly before optimization
        current_vars[:2*n] = np.clip(current_vars[:2*n], 0.0, 1.0)
        current_vars[2*n:] = np.clip(current_vars[2*n:], 1e-5, 0.5)
        
        res = minimize(objective_function, current_vars, args=(n, 10000.0),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': 15000, 'ftol': 1e-15, 'gtol': 1e-12})
        
        cur_centers = res.x[:2*n].reshape(n, 2)
        cur_radii = res.x[2*n:]
        
        # Strict validation to filter out numerically drifted solutions
        if check_validity(cur_centers, cur_radii, n):
            cur_sum = np.sum(cur_radii)
            if cur_sum > best_sum:
                best_sum = cur_sum
                best_centers = cur_centers.copy()
                best_radii = cur_radii.copy()
                
    # Fallback if optimization unexpectedly fails all validations
    if best_centers is None:
        best_centers = init_centers
        best_radii = init_radii
        best_sum = np.sum(init_radii)
        
    return best_centers, best_radii, best_sum
