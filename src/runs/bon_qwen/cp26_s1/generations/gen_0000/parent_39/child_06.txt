

Looking at this problem, I need to pack 26 circles in a unit square to maximize the sum of radii, targeting 2.636.

**Key observations:**
1. For equal circles, 26 circles would give sum ≈ 2.1-2.2, well below target
2. Different-sized circles (larger near corners/edges) can significantly improve the sum
3. Need aggressive optimization with good initialization

<strategy>
1. Start with a hexagonal lattice arrangement (optimal for equal circles) as initial configuration
2. Use L-BFGS-B with penalty method for fast optimization of large constraint set
3. Progressive penalty increase to ensure constraint satisfaction
4. Multiple random restarts to escape local optima
5. Clip final results to ensure validity
6. The penalty approach handles 429+ constraints efficiently compared to SLSQP
</strategy>

```python
import numpy as np
from scipy.optimize import minimize


def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26

    def create_hexagonal_init(radius_scale=1.0):
        centers = np.zeros((n, 2))
        radii = np.ones(n) * 0.08 * radius_scale
        rows = [5, 4, 5, 4, 5, 3]
        idx = 0
        for row, n_circles in enumerate(rows):
            y = 0.08 + row * 0.15
            for col in range(n_circles):
                x = 0.08 + col * 0.18 + (0.09 if row % 2 == 1 else 0)
                centers[idx] = [x, y]
                idx += 1
        return centers, radii

    def penalty_objective(x, penalty, n):
        centers = x[:2 * n].reshape(n, 2)
        radii = x[2 * n:]

        obj = -np.sum(radii)
        violation = 0.0

        # Boundary constraints
        for i in range(n):
            for d in range(2):
                if centers[i, d] < radii[i]:
                    violation += (radii[i] - centers[i, d]) ** 2
                if centers[i, d] > 1 - radii[i]:
                    violation += (centers[i, d] - (1 - radii[i])) ** 2

        # Non-overlap constraints
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                min_dist = radii[i] + radii[j]
                if dist < min_dist:
                    violation += (min_dist - dist) ** 2

        return obj + penalty * violation

    def ensure_validity(centers, radii, n):
        for i in range(n):
            radii[i] = max(0.001, min(0.5, radii[i]))
            for d in range(2):
                centers[i, d] = max(radii[i], min(1 - radii[i], centers[i, d]))
        return centers, radii

    def check_validity(centers, radii, n):
        for i in range(n):
            if radii[i] < 0:
                return False
            for d in range(2):
                if centers[i, d] - radii[i] < -1e-10 or centers[i, d] + radii[i] > 1 + 1e-10:
                    return False
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist < radii[i] + radii[j] - 1e-10:
                    return False
        return True

    best_sum = 0
    best_centers = None
    best_radii = None

    # Try multiple configurations and restarts
    rng = np.random.RandomState(42)

    for attempt in range(10):
        centers, radii = create_hexagonal_init()

        # Add perturbation for diversity
        if attempt > 0:
            centers += rng.randn(n, 2) * 0.02 * attempt
            radii += rng.randn(n) * 0.005 * attempt
            radii = np.maximum(radii, 0.01)
        
        # Ensure initial validity
        centers, radii = ensure_validity(centers, radii, n)

        x0 = np.concatenate([centers.flatten(), radii])

        bounds = [(0, 1)] * (2 * n) + [(0.001, 0.5)] * n

        # Progressive penalty optimization
        x = x0.copy()
        penalties = [50, 200, 1000, 5000, 20000, 100000]
        for penalty in penalties:
            result = minimize(
                penalty_objective,
                x,
                args=(penalty, n),
                method="L-BFGS-B",
                bounds=bounds,
                options={"maxiter": 3000, "ftol": 1e-15, "gtol": 1e-12}
            )
            x = result.x

        final_centers = x[:2 * n].reshape(n, 2)
        final_radii = x[2 * n:]

        final_centers, final_radii = ensure_validity(final_centers, final_radii, n)

        if check_validity(final_centers, final_radii, n):
            current_sum = np.sum(final_radii)
            if current_sum > best_sum:
                best_sum = current_sum
                best_centers = final_centers.copy()
                best_radii = final_radii.copy()

    if best_centers is None:
        # Fallback
        centers, radii = create_hexagonal_init()
        best_centers = centers
        best_radii = radii
        best_sum = np.sum(radii)

    return best_centers, best_radii, best_sum
```
