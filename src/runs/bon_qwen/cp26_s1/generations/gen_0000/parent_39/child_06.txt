import numpy as np
from scipy.optimize import minimize

def run_packing():
    n = 26
    
    # Define the objective function to maximize sum of radii (minimize negative sum)
    def objective(vars):
        r = vars[2::3]
        return -np.sum(r)
    
    # List to hold constraint dictionaries
    constraints = []
    
    # Define bounds for variables: x in [0, 1], y in [0, 1], r in [0, 0.5]
    bounds = []
    for i in range(n):
        bounds.append((0, 1)) # x
        bounds.append((0, 1)) # y
        bounds.append((0, 0.5)) # r
    
    # Add boundary constraints for each circle
    # x >= r, 1-x >= r, y >= r, 1-y >= r
    for i in range(n):
        idx = 3 * i
        
        # x - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: v[3*i] - v[3*i+2]
        })
        # 1 - x - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: 1.0 - v[3*i] - v[3*i+2]
        })
        # y - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: v[3*i+1] - v[3*i+2]
        })
        # 1 - y - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: 1.0 - v[3*i+1] - v[3*i+2]
        })

    # Add non-overlap constraints
    # dist^2 >= (r_i + r_j)^2
    for i in range(n):
        for j in range(i + 1, n):
            idx_i = 3 * i
            idx_j = 3 * j
            
            def cons_func(v, i=i, j=j):
                xi, yi, ri = v[idx_i], v[idx_i+1], v[idx_i+2]
                xj, yj, rj = v[idx_j], v[idx_j+1], v[idx_j+2]
                
                dx = xi - xj
                dy = yi - yj
                dr = ri + rj
                
                return dx*dx + dy*dy - dr*dr
            
            constraints.append({
                'type': 'ineq',
                'fun': cons_func
            })

    # Helper to generate hexagonal grid initialization
    def get_hex_init(radius, shift_x=0, shift_y=0):
        centers = []
        # Hexagonal packing parameters
        # Vertical spacing sqrt(3) * r
        # Horizontal spacing 2 * r
        # Staggered rows
        
        r = radius
        y = r
        row_idx = 0
        
        # Estimate max rows and cols to fit approx 26 circles
        # We will just generate points until we have enough and clip to 26
        
        while len(centers) < n + 10: # Generate slightly more to ensure coverage
            x = r + (row_idx % 2) * r # Stagger by r
            while x <= 1 - r and len(centers) < n + 10:
                if r <= 1 - r: # Check bounds roughly
                    centers.append([x + shift_x, y + shift_y, r])
                x += 2 * r
            y += np.sqrt(3) * r
            row_idx += 1
            
        # Return first n centers
        return np.array(centers[:n]).flatten()

    best_vars = None
    best_val = -np.inf
    
    # Multi-start optimization
    # Attempt 1: Hexagonal grid centered
    # Attempt 2-5: Perturbed hexagonal grids
    
    attempts = [
        {'radius': 0.09, 'shift_x': 0.0, 'shift_y': 0.0},
        {'radius': 0.09, 'shift_x': 0.05, 'shift_y': 0.0},
        {'radius': 0.09, 'shift_x': -0.05, 'shift_y': 0.0},
        {'radius': 0.09, 'shift_x': 0.0, 'shift_y': 0.05},
        {'radius': 0.09, 'shift_x': 0.0, 'shift_y': -0.05},
    ]
    
    # Also try a random initialization
    np.random.seed(42)
    random_vars = np.zeros(3 * n)
    for i in range(n):
        random_vars[3*i] = 0.1 + np.random.random() * 0.8
        random_vars[3*i+1] = 0.1 + np.random.random() * 0.8
        random_vars[3*i+2] = 0.05
    attempts.append({'vars': random_vars})

    for i, att in enumerate(attempts):
        if 'vars' in att:
            x0 = att['vars']
        else:
            x0 = get_hex_init(att['radius'], att['shift_x'], att['shift_y'])
        
        # Ensure initial guess is valid-ish (radii positive)
        # SLSQP might struggle if initial point is heavily infeasible, but should recover.
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints, 
                           options={'maxiter': 2000, 'ftol': 1e-10, 'disp': False})
            
            current_sum = -res.fun
            if current_sum > best_val:
                best_val = current_sum
                best_vars = res.x

    # Fallback if optimization failed

        best_vars = get_hex_init(0.09)

    # Extract results

    radii = best_vars.reshape((n, 3))[:, 2]
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
