import numpy as np
from scipy.optimize import minimize


def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26

    best_centers = None
    best_radii = None
    best_sum = 0.0

    # Try many seeds for robustness
    for seed in range(200):
        centers, radii = attempt_optimization(n, seed)
        current_sum = np.sum(radii)
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers.copy()
            best_radii = radii.copy()

    # Final cleanup: ensure all circles are strictly inside
    for i in range(n):
        r = best_radii[i]
        best_centers[i, 0] = np.clip(best_centers[i, 0], r + 1e-9, 1 - r - 1e-9)
        best_centers[i, 1] = np.clip(best_centers[i, 1], r + 1e-9, 1 - r - 1e-9)

    # Trim any invalid circles (shouldn't happen but just in case)
    valid_mask = best_radii > 0
    if not np.all(valid_mask):
        best_radii = best_radii[valid_mask]
        best_centers = best_centers[valid_mask]

    return best_centers, best_radii, np.sum(best_radii)


def attempt_optimization(n, seed):
    np.random.seed(seed)

    # Initialize with hexagonal packing
    centers, radii = hexagonal_init(n)

    # Add random perturbation to escape symmetry
    centers += np.random.randn(n, 2) * 0.008
    centers = np.clip(centers, 0.03, 0.97)

    # Slightly vary initial radii to break symmetry
    radii += np.random.randn(n) * 0.003
    radii = np.maximum(radii, 0.05)

    params = np.concatenate([centers.flatten(), radii])

    # Multi-phase optimization with increasing penalty
    for phase in range(6):
        if phase < 2:
            pen = 5e4 * (10 ** phase)
        else:
            pen = 5e5 * (10 ** (phase - 2))

        params = local_optimize_penalty(params, n, pen)

    centers = params[:2 * n].reshape(n, 2)
    radii = params[2 * n:]

    # Ensure positivity and validity
    radii = np.maximum(radii, 0.0001)
    for i in range(n):
        r = radii[i]
        centers[i, 0] = np.clip(centers[i, 0], r, 1 - r)
        centers[i, 1] = np.clip(centers[i, 1], r, 1 - r)

    return centers, radii


def hexagonal_init(n):
    """Initialize circles in hexagonal pattern."""
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.095)

    idx = 0
    r = 0.095
    row_height = r * np.sqrt(3)

    row = 0
    while idx < n:
        offset = (row % 2) * r
        col = 0
        row_done = False
        while idx < n and not row_done:
            x = r + offset + col * 2 * r
            y = r + row * row_height

            if x + r <= 1.0 and y + r <= 1.0:
                centers[idx] = [x, y]
                idx += 1
            col += 1
            if col > 12:
                row_done = True
        row += 1
        if row > 12:
            break

    return centers, radii


def local_optimize_penalty(params, n, penalty):
    """Local optimization with penalty method."""
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0001, 0.5)] * n

    def objective(p):
        c = p[:2 * n].reshape(n, 2)
        r = p[2 * n:]

        # Objective: maximize sum of radii => minimize negative
        loss = -np.sum(r)

        # Boundary constraint penalties
        for i in range(n):
            loss += penalty * (
                max(0.0, r[i] - c[i, 0]) ** 2 +
                max(0.0, r[i] - (1.0 - c[i, 0])) ** 2 +
                max(0.0, r[i] - c[i, 1]) ** 2 +
                max(0.0, r[i] - (1.0 - c[i, 1])) ** 2
            )

        # Overlap constraint penalties
        for i in range(n):
            for j in range(i + 1, n):
                dx = c[i, 0] - c[j, 0]
                dy = c[i, 1] - c[j, 1]
                d = np.sqrt(dx * dx + dy * dy)
                overlap = max(0.0, r[i] + r[j] - d)
                loss += penalty * overlap ** 2

        return loss

    result = minimize(
        objective,
        params,
        method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': 4000, 'ftol': 1e-16, 'gtol': 1e-12}
    )

    return result.x


def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square

    Args:
        centers: np.array of shape (n, 2) with (x, y) coordinates
        radii: np.array of shape (n) with radius of each circle

    Returns:
        True if valid, False otherwise
    """
    import numpy as np
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True
