import numpy as np
from scipy.optimize import minimize


def compute_penalty_and_objective(x, n, penalty_weight):
    """Compute negative sum of radii plus penalty for constraint violations."""
    centers = x[:2 * n].reshape((n, 2))
    radii = x[2 * n:]
    
    if np.isnan(centers).any() or np.isnan(radii).any():
        return 1e10
    
    penalty = 0.0
    
    for i in range(n):
        x_i, y_i = centers[i]
        r_i = radii[i]
        
        if x_i - r_i < 0:
            penalty += penalty_weight * (x_i - r_i) ** 2
        if x_i + r_i > 1:
            penalty += penalty_weight * (x_i + r_i - 1) ** 2
        if y_i - r_i < 0:
            penalty += penalty_weight * (y_i - r_i) ** 2
        if y_i + r_i > 1:
            penalty += penalty_weight * (y_i + r_i - 1) ** 2
    
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            min_dist = radii[i] + radii[j]
            if dist < min_dist:
                penalty += penalty_weight * (dist - min_dist) ** 2
    
    obj = -np.sum(radii) + penalty
    return obj


def initialize_grid_5x5(n):
    """Initialize with 5x5 grid plus one extra circle."""
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.085)
    
    for i in range(5):

            centers[idx] = [0.09 + j * 0.2, 0.09 + i * 0.2]

    radii[25] = 0.035
    
    return centers, radii


def initialize_hexagonal(n):

    centers = np.zeros((n, 2))
    radii = np.full(n, 0.085)
    
    r_init = 0.085
    idx = 0
    
    for row in range(5):
        y = r_init + row * np.sqrt(3) * r_init
        if row % 2 == 0:
            ncols = 5
            x_start = r_init
        else:
            ncols = 6
            x_start = r_init + r_init
        
        for col in range(ncols):
            if idx >= n:
                break
            x = x_start + col * 2 * r_init
            centers[idx] = [x, y]
            idx += 1
    
    while idx < n:
        centers[idx] = [0.5, 0.5]
        radii[idx] = 0.03
        idx += 1
    
    return centers, radii


def initialize_staggered(n):

    centers = np.zeros((n, 2))
    radii = np.full(n, 0.08)
    
    r_init = 0.08
    spacing_x = 2 * r_init
    spacing_y = np.sqrt(3) * r_init

            ncols = 5

        centers[idx] = [0.5, 0.5]
        radii[idx] = 0.02
        idx += 1
    
    return centers, radii


def force_based_optimize(centers, radii, n, steps=5000):
    """Force-based simulation to optimize packing."""
    learning_rate = 0.02
    repulsion_strength = 1.0
    expansion_rate = 0.00002
    
    for step in range(steps):
        forces = np.zeros((n, 2))
        radius_forces = np.zeros(n)
        
        current_lr = learning_rate * (0.5 + 0.5 * (1 - step / steps))
        
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                
                if dist < 1e-12:
                    dist = 1e-12
                    dx = 1.0
                    dy = 0.0
                
                min_dist = radii[i] + radii[j]
                if dist < min_dist * 1.01:
                    overlap = max(0, min_dist - dist)
                    force_magnitude = repulsion_strength * overlap / dist
                    fx = force_magnitude * dx
                    fy = force_magnitude * dy
                    forces[i, 0] += fx
                    forces[i, 1] += fy
                    forces[j, 0] -= fx
                    forces[j, 1] -= fy
                    radius_forces[i] += 0.01
                    radius_forces[j] += 0.01
        
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            
            if x - r < 0.001:
                forces[i, 0] += repulsion_strength * (0.001 - (x - r))
            if x + r > 0.999:
                forces[i, 0] -= repulsion_strength * (x + r - 0.999)
            if y - r < 0.001:
                forces[i, 1] += repulsion_strength * (0.001 - (y - r))
            if y + r > 0.999:
                forces[i, 1] -= repulsion_strength * (y + r - 0.999)
            
            # Attraction to boundaries for better packing
            boundary_margin = 0.02
            if x < boundary_margin:
                forces[i, 0] += 0.01 * (boundary_margin - x)
            if x > 1 - boundary_margin:
                forces[i, 0] -= 0.01 * (x - (1 - boundary_margin))
            if y < boundary_margin:
                forces[i, 1] += 0.01 * (boundary_margin - y)
            if y > 1 - boundary_margin:
                forces[i, 1] -= 0.01 * (y - (1 - boundary_margin))
        
        centers += current_lr * forces
        radii += expansion_rate
        
        # Clip and project
        for i in range(n):
            r = radii[i]
            r = max(1e-6, min(r, 0.5))
            radii[i] = r
            centers[i, 0] = np.clip(centers[i, 0], r + 1e-8, 1 - r - 1e-8)
            centers[i, 1] = np.clip(centers[i, 1], r + 1e-8, 1 - r - 1e-8)
    
    return centers, radii


def gradient_optimize(centers, radii, n, penalty_weight=5000.0, maxiter=10000):
    """Gradient-based optimization using L-BFGS-B."""
    x0 = np.concatenate([centers.flatten(), radii])
    
    bounds = [(1e-6, 1 - 1e-6)] * (2 * n) + [(1e-6, 0.5)] * n
    
    def obj_func(x):
        return compute_penalty_and_objective(x, n, penalty_weight)
    
    result = minimize(obj_func, x0, bounds=bounds, method='L-BFGS-B',
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-15})
    
    centers_opt = result.x[:2 * n].reshape((n, 2))
    radii_opt = result.x[2 * n:]
    
    # Final projection to ensure feasibility
    for i in range(n):
        r = radii_opt[i]
        centers_opt[i, 0] = np.clip(centers_opt[i, 0], r + 1e-10, 1 - r - 1e-10)
        centers_opt[i, 1] = np.clip(centers_opt[i, 1], r + 1e-10, 1 - r - 1e-10)
    
    return centers_opt, radii_opt
