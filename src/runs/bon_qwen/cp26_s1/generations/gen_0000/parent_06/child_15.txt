

<strategy>
1. **Initialization**: Start with a structured grid (5x5 + 1 center) to ensure good initial spacing, then add random perturbations to break symmetry and allow the optimizer to find better local configurations. Initial radii are set to a conservative value (0.075) to avoid immediate overlaps.
2. **Penalty-based Optimization**: Formulate the problem as an unconstrained minimization of $-\sum r_i + \lambda \cdot \text{Penalty}$, where the penalty term penalizes boundary violations and inter-circle overlaps using squared hinge losses ($\max(0, \text{violation})^2$). This makes the objective differentiable almost everywhere and compatible with gradient-based methods.
3. **Continuation Method**: Run `scipy.optimize.minimize` (L-BFGS-B) in multiple phases with increasing penalty weights ($\lambda = 100 \to 500 \to 2000 \to 10000$). This helps the optimizer first move circles into feasible regions and then tightly pack them to maximize radii. Multiple random restarts at each phase prevent trapping in poor local minima.
4. **Strict Feasibility Correction**: After optimization, apply a deterministic projection step that iteratively shrinks radii to strictly satisfy all boundary and non-overlap constraints, compensating for any numerical tolerance issues from the optimizer.
5. **Validation**: The final configuration is checked against the problem constraints. The vectorized penalty computation ensures efficiency despite the $O(N^2)$ pairwise checks.
</strategy>

import numpy as np
from scipy.optimize import minimize

def obj_func(v, n, lam):
    centers = v.reshape(-1, 3)[:, :2]
    radii = v.reshape(-1, 3)[:, 2]
    
    # Objective: maximize sum of radii -> minimize negative sum
    obj = -np.sum(radii)
    
    # Boundary penalty: circle must be inside [0,1]^2
    # Violation occurs if r > x, r > 1-x, r > y, or r > 1-y
    p = np.maximum(0, radii - centers[:, 0])**2
    p += np.maximum(0, radii - (1 - centers[:, 0]))**2
    p += np.maximum(0, radii - centers[:, 1])**2
    p += np.maximum(0, radii - (1 - centers[:, 1]))**2
    penalty = np.sum(p)
    
    # Overlap penalty: distance between centers must be >= sum of radii
    # Compute pairwise distances efficiently
    c = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(c**2, axis=2))
    r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    
    # Only consider upper triangle to avoid double counting
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    viol = np.maximum(0, r_sum[mask] - dists[mask])**2
    penalty += np.sum(viol)
    
    return obj + lam * penalty

def run_packing():
    n = 26
    rng = np.random.RandomState(42)
    
    # 1. Initialization: Grid pattern + center point, shuffled
    xs = np.linspace(0.12, 0.88, 5)
    ys = np.linspace(0.12, 0.88, 5)
    pts = np.array([(x, y) for x in xs for y in ys])
    pts = np.vstack([pts, [0.5, 0.5]]) # 26 points total
    pts = rng.permutation(pts)[:n]
    
    r0 = 0.075
    v0 = np.zeros(3 * n)
    for i in range(n):
        v0[3*i] = pts[i, 0]
        v0[3*i+1] = pts[i, 1]
        v0[3*i+2] = r0
        
    # Bounds for L-BFGS-B
    bounds = [(0.0, 1.0) for _ in range(3*n)]
    for i in range(n):
        bounds[3*i+2] = (1e-5, 0.5)
        
    best_v = v0.copy()
    best_fun = obj_func(v0, n, 1000)
    
    # 2. Continuation optimization with multiple restarts
    lams = [100, 500, 2000]
    for lam in lams:
        for _ in range(6):
            # Perturb current best to explore landscape
            vp = best_v + rng.normal(0, 0.008, size=v0.shape)
            # Enforce bounds before optimization
            vp[::3] = np.clip(vp[::3], 0.05, 0.95)
            vp[1::3] = np.clip(vp[1::3], 0.05, 0.95)
            vp[2::3] = np.clip(vp[2::3], 1e-4, 0.25)
            
            res = minimize(obj_func, vp, args=(n, lam), bounds=bounds, 
                           method='L-BFGS-B', options={'maxiter': 3000, 'ftol': 1e-12, 'gtol': 1e-8})
            
            if res.fun < best_fun:
                best_fun = res.fun
                best_v = res.x.copy()
                
    # Final refinement with very high penalty
    res_final = minimize(obj_func, best_v, args=(n, 10000), bounds=bounds, 
                         method='L-BFGS-B', options={'maxiter': 5000, 'ftol': 1e-14})
    best_v = res_final.x
    
    # Extract results
    centers = best_v.reshape(-1, 3)[:, :2]
    radii = best_v.reshape(-1, 3)[:, 2]
    
    # 3. Strict feasibility correction
    # Iteratively shrink radii to guarantee non-overlap and boundary constraints
    for _ in range(100):
        changed = False
        
        # Boundary constraints
        for i in range(n):
            x, y, r = centers[i, 0], centers[i, 1], radii[i]
            new_r = r
            if x - r < 0: new_r = x
            if x + r > 1: new_r = min(new_r, 1 - x)
            if y - r < 0: new_r = min(new_r, y)
            if y + r > 1: new_r = min(new_r, 1 - y)
            if new_r < r - 1e-15:
                radii[i] = new_r
                changed = True
                
        # Overlap constraints
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                d = np.sqrt(dx*dx + dy*dy)
                req = radii[i] + radii[j]
                
                if d < req - 1e-14:
                    # Shrink proportionally to distance
                    scale = d / (req + 1e-15)
                    radii[i] *= scale
                    radii[j] *= scale
                    changed = True
                    
        if not changed:
            break
            
    return centers, radii, float(np.sum(radii))
