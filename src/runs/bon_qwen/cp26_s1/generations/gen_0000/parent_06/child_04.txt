import numpy as np
from scipy.optimize import minimize


def compute_objective(x, n):
    """Compute objective with penalties for constraint violations.
    
    Objective: maximize sum of radii = minimize negative sum of radii
    Penalties added for boundary violations and overlaps.
    """
    radii = x[:n]
    centers = x[n:].reshape(n, 2)
    
    obj = -np.sum(radii)
    
    # Boundary penalties (vectorized over circles)
    r = radii
    cx = centers[:, 0]
    cy = centers[:, 1]
    
    # Left boundary: cx >= r
    viol_left = np.maximum(0, r - cx)
    obj += 10000 * np.sum(viol_left**2)
    
    # Right boundary: cx <= 1 - r
    viol_right = np.maximum(0, cx + r - 1)
    obj += 10000 * np.sum(viol_right**2)
    
    # Bottom boundary: cy >= r
    viol_bottom = np.maximum(0, r - cy)
    obj += 10000 * np.sum(viol_bottom**2)
    
    # Top boundary: cy <= 1 - r
    viol_top = np.maximum(0, cy + r - 1)
    obj += 10000 * np.sum(viol_top**2)
    
    # Overlap penalties (vectorized pairwise)
    # Compute all pairwise squared distances
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2) + 1e-30)  # Add epsilon to avoid div by zero
    np.fill_diagonal(dists, np.inf)  # Ignore self-pairs
    
    # Minimum required separation
    min_dists = radii[:, np.newaxis] + radii[np.newaxis, :]
    
    # Amount of overlap (positive means violation)
    overlaps = np.maximum(0, min_dists - dists)
    
    # Sum squared overlaps for upper triangle only (avoid double counting)
    upper_tri = np.triu(overlaps, k=1)
    obj += 10000 * np.sum(upper_tri**2)
    
    return obj


def hex_init(n, pattern, noise_scale=0.02):
    """Initialize centers in a hexagonal grid pattern.
    
    Args:
        n: number of circles
        pattern: list of row lengths
        noise_scale: standard deviation of random perturbation
    
    Returns:
        centers: array of shape (n, 2)
    """
    centers = np.zeros((n, 2))
    idx = 0
    
    max_cols = max(pattern)
    num_rows = len(pattern)
    
    # Spacing to fit within unit square with margin
    h_step = 1.0 / (max_cols + 1)
    v_step = 1.0 / (num_rows + 1)
    
    for row_idx, count in enumerate(pattern):
        y = v_step * (row_idx + 1)
        for col_idx in range(count):
            # Offset every other row for hexagonal packing
            offset = h_step / 2 if row_idx % 2 == 1 else 0
            x = h_step * (col_idx + 1) + offset
            centers[idx, 0] = x + np.random.randn() * noise_scale
            centers[idx, 1] = y + np.random.randn() * noise_scale
            idx += 1
    
    return centers


def check_validity(centers, radii, n):
    """Check if a packing is valid (no violations within tolerance)."""
    tol = 1e-10
    
    # Boundary checks
    for i in range(n):
        r = radii[i]
        cx, cy = centers[i]
        if cx - r < -tol or cx + r > 1 + tol:
            return False
        if cy - r < -tol or cy + r > 1 + tol:
            return False
    
    # Overlap checks
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            if dist < radii[i] + radii[j] - tol:
                return False
    
    return True


def run_packing():
    """Pack 26 circles in unit square to maximize sum of radii.
    
    Returns:
        centers: np.array of shape (26, 2)
        radii: np.array of shape (26,)
        sum_radii: float
    """
    np.random.seed(42)
    n = 26
    
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Various hexagonal row patterns that sum to 26
    patterns = [
        [6, 5, 6, 5, 4],
        [5, 6, 5, 6, 4],
        [6, 6, 6, 5, 3],
        [5, 5, 5, 5, 6],
        [7, 5, 5, 5, 4],
        [4, 6, 6, 6, 4],
        [6, 4, 6, 6, 4],
        [5, 4, 6, 5, 6],
        [7, 6, 5, 5, 3],
        [4, 5, 6, 5, 6],
    ]
    
    num_trials = 30
    
    for trial in range(num_trials):
        # Select pattern
        pattern = patterns[trial % len(patterns)]
        if sum(pattern) != n:
            continue
        
        # Initialize with hexagonal grid + noise
        noise = 0.02 + trial * 0.005
        centers = hex_init(n, pattern, noise_scale=noise)
        centers = np.clip(centers, 0.05, 0.95)
        
        # Initial radii with variation
        radii = np.ones(n) * 0.08 + np.random.randn(n) * 0.025
        radii = np.clip(radii, 0.02, 0.3)
        
        # Combine into optimization variable
        x0 = np.concatenate([radii, centers.flatten()])
        
        # Bounds: radii in [0.001, 0.4], coordinates in [0.001, 0.999]
        bounds = [(0.001, 0.4)] * n + [(0.001, 0.999)] * (2 * n)
        
        # Optimize
        result = minimize(
            compute_objective,
            x0,
            args=(n,),
            method='L-BFGS-B',
            bounds=bounds,
            options={
                'maxiter': 15000,
                'ftol': 1e-16,
                'gtol': 1e-13,
                'disp': False
            }
        )
        
        radii_opt = result.x[:n]
        centers_opt = result.x[n:].reshape(n, 2)
        
        # Validate solution
        if check_validity(centers_opt, radii_opt, n):
            s = np.sum(radii_opt)
            if s > best_sum:
                best_sum = s
                best_centers = centers_opt.copy()
                best_radii = radii_opt.copy()
    
    # If no valid solution found from optimization, create a safe fallback
    if best_centers is None:
        # Use a conservative hexagonal packing

        centers = hex_init(n, pattern, noise_scale=0.0)

        best_centers = centers
        best_radii = radii
        best_sum = np.sum(radii)
    
    return best_centers, best_radii, best_sum
