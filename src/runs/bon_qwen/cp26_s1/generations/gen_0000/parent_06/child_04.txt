import numpy as np
from scipy.optimize import linprog

def get_dist_matrix(centers):
    """Compute pairwise Euclidean distances."""
    # centers shape (N, 2)
    # Use broadcasting to compute distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    return dists

def solve_radii_lp(centers):
    """
    Solve LP to maximize sum of radii for fixed centers.
    Maximize sum(r_i)
    Subject to:
      r_i + r_j <= dist(i, j)
      r_i <= x_i
      r_i <= 1 - x_i
      r_i <= y_i
      r_i <= 1 - y_i
      r_i >= 0
    """
    n = centers.shape[0]
    
    # Objective: maximize sum(r) -> minimize -sum(r)
    c_obj = -np.ones(n)
    
    # Inequality constraints A_ub @ r <= b_ub
    # We will build A_ub and b_ub
    A_rows = []
    b_rows = []
    
    # 1. Pairwise non-overlap: r_i + r_j <= dist(i, j)
    # Only for i < j
    dists = get_dist_matrix(centers)
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_rows.append(row)
            b_rows.append(dists[i, j])
            
    # 2. Boundary constraints
    for i in range(n):
        x, y = centers[i]
        # r_i <= x_i
        row = np.zeros(n)
        row[i] = 1.0
        A_rows.append(row)
        b_rows.append(x)
        
        # r_i <= 1 - x_i
        row = np.zeros(n)
        row[i] = 1.0
        A_rows.append(row)
        b_rows.append(1.0 - x)
        
        # r_i <= y_i
        row = np.zeros(n)
        row[i] = 1.0
        A_rows.append(row)
        b_rows.append(y)
        
        # r_i <= 1 - y_i
        row = np.zeros(n)
        row[i] = 1.0
        A_rows.append(row)
        b_rows.append(1.0 - y)
        
    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)
    
    # Bounds for r_i >= 0
    bounds = [(0, None) for _ in range(n)]
    
    # Solve
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return -res.fun, -res.x # Return max sum, radii
        else:
            return 0.0, np.zeros(n)
    except Exception:
        return 0.0, np.zeros(n)

def resolve_overlaps(centers, radii, steps=50, strength=0.1):
    """
    Iteratively push overlapping circles apart.
    """
    n = len(centers)
    for _ in range(steps):
        forces = np.zeros_like(centers)
        
        # Check circle-circle overlaps
        for i in range(n):
            for j in range(i + 1, n):
                diff = centers[i] - centers[j]
                dist = np.linalg.norm(diff)
                min_dist = radii[i] + radii[j]
                
                if dist < min_dist and dist > 1e-9:
                    # Overlap amount
                    overlap = min_dist - dist
                    # Force direction
                    force_vec = diff / dist
                    # Apply repulsive force proportional to overlap
                    # Stronger force for larger overlaps
                    f_mag = strength * overlap
                    forces[i] += force_vec * f_mag
                    forces[j] -= force_vec * f_mag
                elif dist == 0:
                    # Push apart randomly if centers coincide
                    rand_dir = np.random.randn(2)
                    rand_dir /= np.linalg.norm(rand_dir)
                    forces[i] += rand_dir * 0.1
                    forces[j] -= rand_dir * 0.1

        # Check circle-wall overlaps
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            
            # Left wall
            if x - r < 0:
                forces[i, 0] += strength * (r - x)
            # Right wall
            if x + r > 1:
                forces[i, 0] -= strength * (x + r - 1)
            # Bottom wall
            if y - r < 0:
                forces[i, 1] += strength * (r - y)
            # Top wall
            if y + r > 1:
                forces[i, 1] -= strength * (y + r - 1)

        # Update centers
        centers += forces
        
        # Clamp centers to valid range [0, 1] (loosely, to avoid extreme escapes)
        # Actually, let's clamp to [r, 1-r] roughly, but careful with changing r.
        # Just clamp to [0,1] for safety during simulation
        centers = np.clip(centers, 1e-5, 1.0 - 1e-5)

    return centers
