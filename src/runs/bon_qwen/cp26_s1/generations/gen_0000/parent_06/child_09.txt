import numpy as np
from scipy.optimize import linprog

def get_max_radii(centers):
    """
    Given fixed centers, solve the LP to find optimal radii.
    Maximize sum(r_i) subject to:
    r_i + r_j <= distance(c_i, c_j)
    r_i <= distance(c_i, boundary)
    r_i >= 0
    """
    n = centers.shape[0]
    
    # Objective: minimize -sum(r)
    c_obj = -np.ones(n)
    
    # Constraints: A_ub @ r <= b_ub
    # We collect constraints in lists for speed, then build matrix
    rows = []
    rhs = []
    
    # Precompute distances
    # dists[i, j]
    # Since n=26, O(n^2) is fine
    dists = np.zeros((n, n))
    for i in range(n):
        for j in range(i + 1, n):
            d = np.linalg.norm(centers[i] - centers[j])
            dists[i, j] = d
            dists[j, i] = d

    # 1. Pairwise constraints: r_i + r_j <= dist
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            rows.append(row)
            rhs.append(dists[i, j])

    # 2. Boundary constraints: r_i <= x, r_i <= 1-x, r_i <= y, r_i <= 1-y
    for i in range(n):
        x, y = centers[i]
        row = np.zeros(n)
        row[i] = 1.0
        
        # r <= x
        rows.append(row)
        rhs.append(x)
        
        # r <= 1-x
        rows.append(row)
        rhs.append(1.0 - x)
        
        # r <= y
        rows.append(row)
        rhs.append(y)
        
        # r <= 1-y
        rows.append(row)
        rhs.append(1.0 - y)

    # Convert to numpy arrays
    A_ub = np.array(rows)
    b_ub = np.array(rhs)
    
    # Bounds: r_i >= 0
    bounds = [(0, None)] * n
    
    # Solve LP
    # 'highs' is efficient
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return -res.fun, res.x
    else:
        # Fallback, should not happen as r=0 is always feasible
        return 0.0, np.zeros(n)
