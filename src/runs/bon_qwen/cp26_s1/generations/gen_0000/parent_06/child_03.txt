import numpy as np
import scipy.optimize as opt

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    """
    n_circles = 26
    
    # 1. Initialization: Hexagonal Lattice
    # We try to fit 26 circles in a hexagonal pattern.
    # A common configuration for N=26 is 6 rows.
    # Pattern: 5, 4, 5, 4, 5, 4 = 27 circles. We need 26.
    # We can use 5, 4, 5, 4, 5, 3 = 26 circles.
    # Or 4, 5, 4, 5, 4, 4 = 26.
    # Let's try to fit a dense pack. 
    # Vertical spacing dy = sqrt(3)/2 * diameter = sqrt(3) * r
    # Horizontal spacing dx = diameter = 2r
    
    # Let's estimate a starting radius. 
    # For N=26, if we assume equal circles, max radius is likely around 0.09-0.10.
    # Let's start with r = 0.09.
    
    r_init = 0.09
    centers = []
    
    # Configuration: 5 rows of 5? No, 25.
    # Let's try a 6-row hexagonal packing.
    # Row 0 (y=r): 5 circles
    # Row 1 (y=r+sqrt(3)r): 4 circles (shifted by r)
    # Row 2 (y=r+2sqrt(3)r): 5 circles
    # Row 3 (y=r+3sqrt(3)r): 4 circles
    # Row 4 (y=r+4sqrt(3)r): 5 circles
    # Row 5 (y=r+5sqrt(3)r): 3 circles (to make 26: 5+4+5+4+5+3 = 26)
    # Wait, 5+4+5+4+5+3 = 26.
    
    rows_config = [5, 4, 5, 4, 5, 3]
    
    # Check if this fits vertically with r=0.09
    # Height needed = 2r + (num_rows-1)*sqrt(3)*r
    # 2*0.09 + 5*1.732*0.09 = 0.18 + 0.779 = 0.959 < 1. OK.
    # Width check for row of 5: 2r*5 = 0.9 <= 1. OK.
    # Width check for row of 3: 2r*3 = 0.54 <= 1. OK.
    
    current_circle_idx = 0
    for i, count in enumerate(rows_config):
        y = r_init + i * np.sqrt(3) * r_init
        # Determine x-offset. Even rows (0, 2, 4) start at r, odd rows (1, 3, 5) start at 2r (shifted)
        # Actually, standard hex packing:
        # Row 0: centers at r, 3r, 5r...
        # Row 1: centers at 2r, 4r, 6r...
        
        x_start = r_init if i % 2 == 0 else 2 * r_init
        
        # Center the row horizontally if possible to utilize space better?
        # For now, left-aligned is fine for initialization.
        
        for j in range(count):
            x = x_start + j * 2 * r_init
            centers.append([x, y])
            current_circle_idx += 1
            
    # If we didn't reach 26 or exceeded, adjust. 
    # The config 5,4,5,4,5,3 sums to 26.
    if len(centers) != n_circles:
        # Fallback to simple grid if logic fails, but 26 is specific.
        # Re-fill to 26
        centers = []
        grid_n = int(np.ceil(np.sqrt(n_circles))) # 6
        step_x = 1.0 / (grid_n + 1)
        step_y = 1.0 / (grid_n + 1)
        idx = 0
        for i in range(grid_n):
            for j in range(grid_n):
                if idx < n_circles:
                    centers.append([step_x * (i + 1), step_y * (j + 1)])
                    idx += 1
                else:
                    break
            if idx >= n_circles: break
        r_init = 0.08

    centers = np.array(centers)
    radii = np.full(n_circles, r_init)
    
    # 2. Optimization Setup
    # Variables: [x1, y1, r1, x2, y2, r2, ..., x26, y26, r26]
    # Total 3 * 26 = 78 variables.
    
    x0 = np.hstack([centers.ravel(), radii])
    
    def objective(x_vars):
        # Minimize negative sum of radii
        r = x_vars[2::3]
        return -np.sum(r)
    
    def get_constraints():
        constraints = []
        
        # Boundary constraints:
        # r <= x <= 1-r  =>  x - r >= 0, 1 - x - r >= 0
        # r <= y <= 1-r  =>  y - r >= 0, 1 - y - r >= 0
        
        for i in range(n_circles):
            # Index in x_vars: x=3i, y=3i+1, r=3i+2
            idx_x = 3 * i
            idx_y = 3 * i + 1
            idx_r = 3 * i + 2
            
            # x - r >= 0
            constraints.append({
                'type': 'ineq',
                'fun': lambda x, i=i: x[idx_x] - x[idx_r]
            })
            # 1 - x - r >= 0
            constraints.append({
                'type': 'ineq',
                'fun': lambda x, i=i: 1.0 - x[idx_x] - x[idx_r]
            })
            # y - r >= 0
            constraints.append({
                'type': 'ineq',
                'fun': lambda x, i=i: x[idx_y] - x[idx_r]
            })
            # 1 - y - r >= 0
            constraints.append({
                'type': 'ineq',
                'fun': lambda x, i=i: 1.0 - x[idx_y] - x[idx_r]
            })
            
            # r >= 0 (strictly positive is better, but >= 0 is constraint)
            constraints.append({
                'type': 'ineq',
                'fun': lambda x, i=i: x[idx_r]
            })
        
        # Non-overlap constraints:
        # dist(i, j)^2 >= (r_i + r_j)^2
        # (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
        
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                idx_x_i = 3 * i
                idx_y_i = 3 * i + 1
                idx_r_i = 3 * i + 2
                
                idx_x_j = 3 * j
                idx_y_j = 3 * j + 1
                idx_r_j = 3 * j + 2
                
                def dist_constraint(x, i=i, j=j, 
                                    idx_x_i=idx_x_i, idx_y_i=idx_y_i, idx_r_i=idx_r_i,
                                    idx_x_j=idx_x_j, idx_y_j=idx_y_j, idx_r_j=idx_r_j):
                    dx = x[idx_x_i] - x[idx_x_j]
                    dy = x[idx_y_i] - x[idx_y_j]
                    dist_sq = dx*dx + dy*dy
                    sum_r = x[idx_r_i] + x[idx_r_j]
                    return dist_sq - sum_r*sum_r
                
                constraints.append({
                    'type': 'ineq',
                    'fun': dist_constraint
                })
        
        return constraints

    cons = get_constraints()
    
    # 3. Run Optimizer
    # SLSQP is a good choice for constrained optimization
    try:
        res = opt.minimize(objective, x0, method='SLSQP', constraints=cons, options={'maxiter': 1000, 'ftol': 1e-9})
        x_opt = res.x
    except Exception as e:
        print(f"Optimization failed: {e}")
        # Fallback to initial guess
        x_opt = x0
        
    centers_opt = x_opt[:2*n_circles].reshape((n_circles, 2))
    radii_opt = x_opt[2*n_circles:]
    
    # Ensure radii are non-negative and valid
    radii_opt = np.maximum(radii_opt, 1e-9)
    
    # Project centers to valid range just in case (though constraints should handle it)
    for i in range(n_circles):
        r = radii_opt[i]
        x = centers_opt[i, 0]
        y = centers_opt[i, 1]
        x = np.clip(x, r, 1.0 - r)
        y = np.clip(y, r, 1.0 - r)
        centers_opt[i, 0] = x
        centers_opt[i, 1] = y

    sum_radii = np.sum(radii_opt)
    
    return centers_opt, radii_opt, sum_radii

import numpy as np # Ensure numpy is imported

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    n = centers.shape[0]
    if np.isnan(centers).any() or np.isnan(radii).any():
        return False
    for i in range(n):
        if radii[i] < 0: return False
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            return False
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False
    return True
