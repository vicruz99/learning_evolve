import numpy as np
from scipy.optimize import minimize, linprog


def compute_optimal_radii(centers):
    """Given centers, solve LP to find radii maximizing sum of radii."""
    n = centers.shape[0]
    c = -np.ones(n)
    
    n_boundary = 4 * n
    n_pairwise = n * (n - 1) // 2
    n_constraints = n_boundary + n_pairwise
    
    A_ub = np.zeros((n_constraints, n))
    b_ub = np.zeros(n_constraints)
    
    idx = 0
    for i in range(n):
        x, y = centers[i]
        A_ub[idx, i] = 1
        b_ub[idx] = x
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1.0 - x
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = y
        idx += 1
        A_ub[idx, i] = 1
        b_ub[idx] = 1.0 - y
        idx += 1
    
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            A_ub[idx, i] = 1
            A_ub[idx, j] = 1
            b_ub[idx] = dist
            idx += 1
    
    bounds = [(0, None) for _ in range(n)]
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        return -result.fun, result.x
    else:
        return 0.0, np.full(n, 0.01)


def objective_centers(centers_flat):
    """Objective: negative sum of optimal radii for given centers."""
    centers = centers_flat.reshape(-1, 2)
    _, radii = compute_optimal_radii(centers)
    return -np.sum(radii)


def initialize_hexagonal(n, spacing=0.22, offset=0.11):
    """Initialize circles in a hexagonal grid pattern."""
    centers = np.zeros((n, 2))
    idx = 0
    row = 0
    while idx < n:
        col = 0
        while idx < n and col < 10:
            x = col * spacing + offset + (spacing / 2.0 if row % 2 == 1 else 0.0)
            y = row * spacing * np.sqrt(3.0) / 2.0 + offset
            if x >= 0 and x <= 1 and y >= 0 and y <= 1:
                centers[idx] = [x, y]
                idx += 1
            col += 1
        row += 1
    return centers


def initialize_square_grid(n):
    """Initialize circles in a square grid pattern."""
    centers = np.zeros((n, 2))
    idx = 0
    for row in range(7):
        for col in range(7):
            if idx >= n:
                break
            x = (col + 0.5) / 7.0
            y = (row + 0.5) / 7.0
            centers[idx] = [x, y]
            idx += 1
    return centers


def initialize_random(n, seed=0):
    """Initialize circles at random positions."""
    rng = np.random.RandomState(seed)
    centers = np.zeros((n, 2))
    for i in range(n):
        x = rng.uniform(0.1, 0.9)
        y = rng.uniform(0.1, 0.9)
        centers[i] = [x, y]
    return centers


def initialize_distributed(n):
    """Initialize with well-distributed points using Poisson-disc-like approach."""
    centers = np.zeros((n, 2))
    rng = np.random.RandomState(42)
    
    # Create a grid of candidate points
    grid_size = 20
    candidates = []
    for i in range(grid_size):
        for j in range(grid_size):
            candidates.append([i / grid_size, j / grid_size])
    
    # Select n points maximizing minimum distance
    selected = []
    remaining = list(range(len(candidates)))
    
    for _ in range(n):
        if not remaining:
            break
        best_idx = -1
        best_min_dist = -1
        for idx in remaining:
            cand = candidates[idx]
            min_dist = min(np.linalg.norm(np.array(cand) - np.array(s)) for s in selected) if selected else 1.0
            if min_dist > best_min_dist:
                best_min_dist = min_dist
                best_idx = idx
        if best_idx >= 0:
            selected.append(candidates[best_idx])
            remaining.remove(best_idx)
    
    for i, s in enumerate(selected):
        centers[i] = s
    
    return centers


def try_optimization(centers, maxiter=15000):
    """Run optimization from given centers."""
    n = centers.shape[0]
    bounds = [(0.01, 0.99) for _ in range(2 * n)]
    
    result = minimize(
        objective_centers,
        centers.flatten(),
        bounds=bounds,
        method='Nelder-Mead',
        options={'maxiter': maxiter, 'xatol': 1e-9, 'fatol': 1e-10}
    )
    
    optimal_centers = result.x.reshape(-1, 2)
    sum_r, radii = compute_optimal_radii(optimal_centers)
    return optimal_centers, radii, sum_r


def grow_radii_locally(centers, radii, n_iterations=100):
    """Try to grow radii by pushing circles apart from each other and boundaries."""
    n = len(radii)
    centers = centers.copy()
    radii = radii.copy()
    
    for iteration in range(n_iterations):
        moved = False
        for i in range(n):
            # Compute max possible radius for circle i
            max_r = min(centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])
            
            for j in range(n):
                if i != j:
                    dist = np.linalg.norm(centers[i] - centers[j])
                    max_r = min(max_r, dist - radii[j])
            
            max_r = max(0, max_r)
            
            if max_r > radii[i] + 1e-10:
                # Can grow - try to move center to allow more growth
                radii[i] = max_r
                moved = True
        
        if not moved:
            break
    
    return centers, radii
