        Returns a numpy array of constraint values.
        For inequality constraints g(x) >= 0, we return the value.
        For equality constraints h(x) == 0, we return the value.
        Here we have:
        1. x_i - r_i >= 0
        2. 1 - (x_i + r_i) >= 0
        3. y_i - r_i >= 0
        4. 1 - (y_i + r_i) >= 0
        5. dist(i,j) - (r_i + r_j) >= 0

        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            constraints.append(x - r)
            constraints.append(1 - (x + r))
            constraints.append(y - r)
            constraints.append(1 - (y + r))

                dist = np.sqrt(dx*dx + dy*dy)
                constraints.append(dist - (radii[i] + radii[j]))
