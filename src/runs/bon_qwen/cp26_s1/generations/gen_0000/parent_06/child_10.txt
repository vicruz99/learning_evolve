import numpy as np
from scipy.optimize import minimize

N = 26


def objective(vars_flat):
    radii = vars_flat[2::3]
    return -np.sum(radii)


def constraint_function(vars_flat):
    vals = []
    for i in range(N):
        x = vars_flat[3 * i]
        y = vars_flat[3 * i + 1]
        r = vars_flat[3 * i + 2]
        vals.append(x - r)
        vals.append(1.0 - x - r)
        vals.append(y - r)
        vals.append(1.0 - y - r)
        vals.append(r)
    for i in range(N):
        for j in range(i + 1, N):
            dx = vars_flat[3 * i] - vars_flat[3 * j]
            dy = vars_flat[3 * i + 1] - vars_flat[3 * j + 1]
            ri = vars_flat[3 * i + 2]
            rj = vars_flat[3 * j + 2]
            vals.append(dx * dx + dy * dy - (ri + rj) * (ri + rj))
    return np.array(vals)


def constraint_wrapper(vars_flat):
    return constraint_function(vars_flat)


def make_hex_grid():
    rows = [5, 4, 5, 4, 5, 3]
    r_init = 0.04
    centers = []
    y = r_init
    sx = (1.0 - 2.0 * r_init) / 4.0
    for idx, nc in enumerate(rows):
        if idx % 2 == 0:
            for k in range(nc):
                centers.append([r_init + k * sx, y])
        else:
            for k in range(nc):
                centers.append([r_init + 0.5 * sx + k * sx, y])
        y += r_init * np.sqrt(3.0)
    return np.array(centers), np.ones(N) * r_init


def make_random_config(r0=0.02):
    centers = np.random.rand(N, 2)
    radii = np.ones(N) * r0
    return centers, radii


def make_grid_config():
    rows = [5, 5, 5, 5, 6]
    r_init = 0.035
    centers = []
    y = r_init
    sy = (1.0 - 2.0 * r_init) / (len(rows) - 1)
    for idx, nc in enumerate(rows):
        if nc > 1:
            sx = (1.0 - 2.0 * r_init) / (nc - 1)
        else:
            sx = 0.0
        for k in range(nc):
            x = r_init + k * sx
            centers.append([x, y])
        y += sy
    return np.array(centers), np.ones(N) * r_init


def make_hex_dense():
    rows = [5, 4, 5, 4, 5, 3]
    r_init = 0.08
    centers = []
    y = r_init
    sx = (1.0 - 2.0 * r_init) / 4.0
    for idx, nc in enumerate(rows):
        if idx % 2 == 0:
            for k in range(nc):
                centers.append([r_init + k * sx, y])
        else:
            for k in range(nc):
                centers.append([r_init + 0.5 * sx + k * sx, y])
        y += r_init * np.sqrt(3.0)
    return np.array(centers), np.ones(N) * r_init


def run_optimization(centers, radii, max_iter=15000):
    v = np.zeros(3 * N)
    for i in range(N):
        v[3 * i] = centers[i, 0]
        v[3 * i + 1] = centers[i, 1]
        v[3 * i + 2] = radii[i]
    
    cons = {'type': 'ineq', 'fun': constraint_wrapper}
    bnds = [(0.0, 1.0)] * N + [(0.0, 1.0)] * N + [(0.0, 0.5)] * N
    
    res = minimize(
        objective, v, method='SLSQP',
        constraints=cons, bounds=bnds,
        options={'maxiter': max_iter, 'ftol': 1e-15, 'disp': False}
    )
    
    c = np.array([[res.x[3 * i], res.x[3 * i + 1]] for i in range(N)])
    r = res.x[2::3]
    return c, r, np.sum(r)


def post_process(centers, radii):
    c = centers.copy()
    r = radii.copy()
    n = len(c)
    
    # Clamp to square
    for i in range(n):
        x, y = c[i]
        mr = min(x, 1.0 - x, y, 1.0 - y, r[i])
        r[i] = max(0.0, mr)
    
    # Resolve overlaps iteratively
    for _ in range(500):
        changed = False
        for i in range(n):
            if r[i] <= 0:
                continue
            for j in range(i + 1, n):
                if r[j] <= 0:
                    continue
                dx = c[i, 0] - c[j, 0]
                dy = c[i, 1] - c[j, 1]
                d = np.sqrt(dx * dx + dy * dy)
                md = r[i] + r[j]
                if d < md - 1e-12:
                    ov = md - d
                    # Reduce proportionally to current radius
                    total = r[i] + r[j]
                    if total > 0:
                        r[i] -= ov * (r[i] / total) * 0.5
                        r[j] -= ov * (r[j] / total) * 0.5
                    else:
                        r[i] -= ov * 0.5
                        r[j] -= ov * 0.5
                    r[i] = max(0.0, r[i])
                    r[j] = max(0.0, r[j])
                    changed = True
        if not changed:
            break
    
    return c, r


def try_optimization(centers, radii):
    c, r, s = run_optimization(centers, radii)
    c, r = post_process(c, r)
    return c, r, np.sum(r)
