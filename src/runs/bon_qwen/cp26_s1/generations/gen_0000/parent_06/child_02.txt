        yi = x[3 * circle_idx + 1]
        ri = x[3 * circle_idx + 2]
        # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
        return np.array([xi - ri, 1 - (xi + ri), yi - ri, 1 - (yi + ri)])
