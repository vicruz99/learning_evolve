    radii = params[n * 2:]
    return -np.sum(radii)

def make_overlap_constraints(n):
    """Create non-overlap constraints for all circle pairs."""
    constraints = []
    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({

    return constraints

    overlap_con = make_overlap_constraints(n)

    best_sum = np.sum(params0[n * 2:])

            current_params = params0.copy()

            current_params = np.clip(current_params[:n * 2], 0.01, 0.99)

        try:
            result = minimize(
                lambda p: objective(p, n),
                current_params,
                method='SLSQP',
                constraints=all_constraints,
                bounds=bounds,
                options={'maxiter': 5000, 'ftol': 1e-14, 'disp': False}
            )
            current_sum = np.sum(result.x[n * 2:])
            if current_sum > best_sum:
                best_sum = current_sum
                best_params = result.x.copy()

            continue

def penalty_optimize(params0, n):
    """Optimize using penalty method with L-BFGS-B."""
    best_params = params0.copy()
    bounds = [(0, 1)] * (n * 2) + [(1e-8, 0.5)] * n

    penalty_weights = [50, 200, 1000, 5000, 10000, 50000]

    for pw in penalty_weights:
        def obj_with_penalty(p):
            centers = p[:n * 2].reshape(n, 2)
            radii = p[n * 2:]
            obj = -np.sum(radii)
            penalty = 0.0

            # Boundary penalties
            for i in range(n):
                x, y = centers[i]
                r = radii[i]
                v = r - x
                if v > 0:
                    penalty += pw * v * v
                v = x + r - 1
                if v > 0:
                    penalty += pw * v * v
                v = r - y
                if v > 0:
                    penalty += pw * v * v
                v = y + r - 1
                if v > 0:
                    penalty += pw * v * v

            # Overlap penalties
            for i in range(n):
                for j in range(i + 1, n):
                    dx = centers[i, 0] - centers[j, 0]
                    dy = centers[i, 1] - centers[j, 1]
                    dist = np.sqrt(dx * dx + dy * dy)
                    v = radii[i] + radii[j] - dist
                    if v > 0:
                        penalty += pw * v * v

            return obj + penalty

        try:
            result = minimize(
                obj_with_penalty, best_params, method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 3000, 'ftol': 1e-15}
            )
            best_params = result.x.copy()
        except Exception:
            continue

    return best_params


def grow_circles_heuristic(params, n, max_iter=500):
    """Try to grow circles while maintaining feasibility."""
    best_params = params.copy()

    for iteration in range(max_iter):
        improved = False
        for i in range(n):
            idx_r = n * 2 + i
            current_r = best_params[idx_r]
            # Try slightly larger radius
            for delta in [0.002, 0.001, 0.0005, 0.0001]:
                new_r = current_r + delta
                if new_r > 0.5:
                    continue

                # Check boundary
                x = best_params[i * 2]
                y = best_params[i * 2 + 1]
                if x - new_r < -1e-12 or x + new_r > 1 + 1e-12:
                    continue
                if y - new_r < -1e-12 or y + new_r > 1 + 1e-12:
                    continue

                # Check overlaps
                feasible = True
                for j in range(n):
                    if j == i:
                        continue
                    jx = best_params[j * 2]
                    jy = best_params[j * 2 + 1]
                    jr = best_params[n * 2 + j]
                    dist = np.sqrt((x - jx) ** 2 + (y - jy) ** 2)
                    if dist < new_r + jr - 1e-12:
                        feasible = False
                        break

                if feasible:
                    best_params[idx_r] = new_r
                    current_r = new_r
                    improved = True
                    break

        if not improved:
            break

    return best_params


def try_boundary_optimization(params, n):
    """Try pushing circles toward boundaries to create space."""
    best_params = params.copy()

    # For each circle, try moving it toward each boundary
    for i in range(n):
        idx_x = i * 2
        idx_y = i * 2 + 1
        idx_r = n * 2 + i
        r = best_params[idx_r]

        # Try pushing toward each corner/boundary
        directions = [
            (best_params[idx_x] - r - 0.01, best_params[idx_y]),  # left
            (1 - r - 0.01, best_params[idx_y]),  # right
            (best_params[idx_x], best_params[idx_y] - r - 0.01),  # bottom
            (best_params[idx_x], 1 - r - 0.01),  # top
            (best_params[idx_x] - r - 0.01, best_params[idx_y] - r - 0.01),  # bottom-left
            (1 - r - 0.01, best_params[idx_y] - r - 0.01),  # bottom-right
            (best_params[idx_x] - r - 0.01, 1 - r - 0.01),  # top-left
            (1 - r - 0.01, 1 - r - 0.01),  # top-right
        ]

        best_pos = None
        best_improvement = 0

        for new_x, new_y in directions:
            new_x = max(r, min(1 - r, new_x))
            new_y = max(r, min(1 - r, new_y))

            # Check overlaps at new position
            feasible = True
            for j in range(n):
                if j == i:
                    continue
                jx = best_params[j * 2]
                jy = best_params[j * 2 + 1]
                jr = best_params[n * 2 + j]
                dist = np.sqrt((new_x - jx) ** 2 + (new_y - jy) ** 2)
                if dist < r + jr - 1e-12:
                    feasible = False
                    break

            if feasible:
                # Check if this position allows more room for neighbors
                # by computing minimum distance to other circles
                min_gap = float('inf')
                for j in range(n):
                    if j == i:
                        continue
                    jx = best_params[j * 2]
                    jy = best_params[j * 2 + 1]
                    jr = best_params[n * 2 + j]
                    dist = np.sqrt((new_x - jx) ** 2 + (new_y - jy) ** 2)
                    gap = dist - r - jr
                    min_gap = min(min_gap, gap)

                if min_gap > best_improvement:
                    best_improvement = min_gap
                    best_pos = (new_x, new_y)

        if best_pos is not None:
            best_params[idx_x] = best_pos[0]
            best_params[idx_y] = best_pos[1]

    return best_params

    sum_grown = np.sum(params_grown[n * 2:])

    if sum_grown > sum_slsqp2:

    # Step 7: Try boundary optimization and re-optimize
    params_boundary = try_boundary_optimization(params_best.copy(), n)
    params_after_boundary = optimize_with_slsqp(params_boundary.copy(), n, num_restarts=3)
    sum_after_boundary = np.sum(params_after_boundary[n * 2:])

    if sum_after_boundary > sum_grown:
        params_best = params_after_boundary

    # Step 8: Final penalty optimization pass

    sum_final = np.sum(params_final[n * 2:])

    sum_radii = np.sum(radii_opt)
    return centers_opt, radii_opt, float(sum_radii)
