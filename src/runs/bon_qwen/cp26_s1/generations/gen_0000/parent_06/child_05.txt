def objective(params, n):
    """Objective: Maximize sum of radii => Minimize -sum(radii)"""
    radii = params[2*n:]
    return -np.sum(radii)

def constraint_boundary(params, n):
    """Constraints: circles inside square"""
    centers = params[:2*n].reshape(n, 2)
    radii = params[2*n:]
    
    # x - r >= 0
    cons_x1 = centers[:, 0] - radii
    # 1 - x - r >= 0  => x + r <= 1
    cons_x2 = 1 - centers[:, 0] - radii
    # y - r >= 0
    cons_y1 = centers[:, 1] - radii
    # 1 - y - r >= 0
    cons_y2 = 1 - centers[:, 1] - radii
    
    return np.concatenate([cons_x1, cons_x2, cons_y1, cons_y2])

def constraint_non_overlap(params, n):
    """Constraints: non-overlap between circles"""

    radii = params[2*n:]
