import numpy as np
from scipy.optimize import minimize


def obj_func(x, n):
    """Negative sum of radii (minimize this = maximize sum of radii)."""
    s = 0.0
    for i in range(n):
        s += x[3 * i + 2]
    return -s


def con_bound_x_lo(x, idx):
    """x_i - r_i >= 0"""
    return x[3 * idx] - x[3 * idx + 2]


def con_bound_x_hi(x, idx):
    """1 - x_i - r_i >= 0"""
    return 1.0 - x[3 * idx] - x[3 * idx + 2]


def con_bound_y_lo(x, idx):
    """y_i - r_i >= 0"""
    return x[3 * idx + 1] - x[3 * idx + 2]


def con_bound_y_hi(x, idx):
    """1 - y_i - r_i >= 0"""
    return 1.0 - x[3 * idx + 1] - x[3 * idx + 2]


def con_overlap(x, i, j):
    """(x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0"""
    xi = x[3 * i]
    yi = x[3 * i + 1]
    ri = x[3 * i + 2]
    xj = x[3 * j]
    yj = x[3 * j + 1]
    rj = x[3 * j + 2]
    return (xi - xj) ** 2 + (yi - yj) ** 2 - (ri + rj) ** 2


def make_hex_init(n):
    """Hexagonal grid initialization."""
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.07
    idx = 0
    for row in range(7):
        for col in range(5):
            if idx >= n:
                break

            y = 0.07 + row * 0.155
            centers[idx] = [x, y]
            radii[idx] = 0.07
            idx += 1
        if idx >= n:
            break
    return centers, radii


def make_random_init(n):

def project_and_fix(centers, radii, n):
    """Project to feasible region and resolve overlaps."""
    centers = centers.copy()
    radii = radii.copy()

    # Ensure non-negative radii
    radii = np.maximum(radii, 0)

    # Project to boundary constraints
    for i in range(n):
        r = radii[i]
        centers[i][0] = max(r, min(1 - r, centers[i][0]))
        centers[i][1] = max(r, min(1 - r, centers[i][1]))

    # Iteratively resolve overlaps
    for _ in range(200):
        fixed = True
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[j][0] - centers[i][0]
                dy = centers[j][1] - centers[i][1]
                dist = np.sqrt(dx ** 2 + dy ** 2)
                min_dist = radii[i] + radii[j]
                if dist < min_dist - 1e-10 and dist > 1e-14:
                    fixed = False
                    reduction = (min_dist - dist) / 2 + 1e-6
                    # Reduce the larger radius more
                    if radii[i] >= radii[j]:
                        radii[i] = max(0, radii[i] - reduction)
                    else:
                        radii[j] = max(0, radii[j] - reduction)
                    # Reproject centers
                    for k in range(n):
                        r = radii[k]
                        centers[k][0] = max(r, min(1 - r, centers[k][0]))
                        centers[k][1] = max(r, min(1 - r, centers[k][1]))

    return centers, radii

            constraints.append({'type': 'ineq', 'fun': con_bound_y_lo, 'args': (i,)})
