import numpy as np
import math
import scipy.optimize

def calculate_loss(params, n, num_cols, num_rows, lambda_penalty):
    """
    Calculate the objective function (negative sum of radii) plus penalty terms.
    """
    # Reshape params into centers and radii
    centers = params[:2 * n].reshape(n, 2)
    radii = params[2 * n:]
    
    # 1. Objective: Maximize sum of radii (minimize negative sum)
    objective = -np.sum(radii)
    
    # 2. Penalty for overlaps
    overlap_penalty = 0.0
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            min_dist = radii[i] + radii[j]
            if dist < min_dist:
                diff = min_dist - dist
                overlap_penalty += (diff * lambda_penalty) ** 2
                
    # 3. Penalty for boundary violations
    boundary_penalty = 0.0
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        
        # Check left and right walls
        if x - r < 0:
            boundary_penalty += (lambda_penalty * (r - x)) ** 2
        if x + r > 1:
            boundary_penalty += (lambda_penalty * (x + r - 1)) ** 2
            
        # Check bottom and top walls
        if y - r < 0:
            boundary_penalty += (lambda_penalty * (r - y)) ** 2
        if y + r > 1:
            boundary_penalty += (lambda_penalty * (y + r - 1)) ** 2
            
    return objective + overlap_penalty + boundary_penalty

def generate_hexagonal_init(n, scale=0.12):
    """Generate a hexagonal lattice initialization."""
    centers = []
    radii = []
    y = scale
    row = 0
    while len(centers) < n:
        offset = scale / 2 if row % 2 == 1 else 0
        x = scale + offset
        while x <= 1 - scale and len(centers) < n:
            centers.append([x, y])
            radii.append(scale)
            x += 2 * scale
        centers.append([x - scale, y]) # Check last one
        if len(centers) >= n:
            break
        y += scale * math.sqrt(3)
        row += 1
        
    centers = np.array(centers[:n])
    radii = np.array(radii[:n])
    return centers, radii

def generate_grid_init(n):
    """Generate a grid initialization."""
    centers = []
    radii = []
    # Try to fit n circles in a grid
    rows = int(math.ceil(math.sqrt(n)))
    cols = int(math.ceil(n / rows))
    
    x_step = 1.0 / (cols + 1)
    y_step = 1.0 / (rows + 1)
    r = min(x_step, y_step) / 2 * 0.9
    
    for i in range(rows):
        for j in range(cols):
            if len(centers) < n:
                centers.append([x_step * (j + 1), y_step * (i + 1)])
                radii.append(r)
                
    centers = np.array(centers[:n])
    radii = np.array(radii[:n])
    return centers, radii

        params = np.concatenate([centers.flatten(), radii])
        
        # 2. Optimization
