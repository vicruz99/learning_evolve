import numpy as np
from scipy.optimize import minimize


def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Try multiple initial configurations with different patterns and seeds
    configs = generate_initial_configs(n)
    
    for config_idx, (centers, radii) in enumerate(configs):
        x0 = np.concatenate([centers.ravel(), radii])
        
        # Bounds for optimization
        bounds = [(0.002, 0.998)] * (2 * n) + [(0.002, 0.35)] * n
        
        # Run optimization with multiple penalty strengths
        for penalty_mult in [500, 1000, 2000]:
            try:
                result = minimize(
                    obj_func,
                    x0,
                    args=(n, penalty_mult),
                    bounds=bounds,
                    method='L-BFGS-B',
                    options={'maxiter': 50000, 'ftol': 1e-15, 'gtol': 1e-10}
                )
                
                centers_opt = result.x[:2 * n].reshape(n, 2)
                radii_opt = result.x[2 * n:]
                
                # Ensure validity
                centers_opt, radii_opt = make_valid(centers_opt, radii_opt, n)
                
                current_sum = np.sum(radii_opt)
                if current_sum > best_sum:
                    best_sum = current_sum
                    best_centers = centers_opt.copy()
                    best_radii = radii_opt.copy()
            except Exception:
                continue
        
        # Also try iterative growth from this config
        centers_iter, radii_iter = iterative_optimize(centers, radii, n)
        centers_iter, radii_iter = make_valid(centers_iter, radii_iter, n)
        current_sum = np.sum(radii_iter)
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers_iter.copy()

    return best_centers, best_radii, best_sum


def generate_initial_configs(n):

    configs = []

def hex_pattern(n, row_counts, r_init, seed):
    """Create hexagonal packing with given row pattern"""
    np.random.seed(seed)
    
    r = r_init
    centers = []
    radii = []
    
    row_spacing = r * np.sqrt(3) * 0.95
    col_spacing = r * np.sqrt(3) * 0.95
    
    # Add small random perturbation
    y_offset = np.random.uniform(-0.005, 0.005)
    x_offset = np.random.uniform(-0.005, 0.005)
    
    y = 0.1 + y_offset
    for row_idx, ncols in enumerate(row_counts):
        x_start = 0.5 - (ncols - 1) * col_spacing / 2 + x_offset
        if row_idx % 2 == 1:
            x_start += col_spacing / 2
        
        for col in range(ncols):
            if len(centers) >= n:
                break
            centers.append([x_start + col * col_spacing, y])
            radii.append(r)
        y += row_spacing
    
    centers = np.array(centers[:n])
    radii = np.array(radii[:n])
    
    # Scale to fit nicely in [0.05, 0.95]
    cmin = centers.min(axis=0)
    cmax = centers.max(axis=0)
    margin = 0.08
    scale_x = (1 - 2 * margin) / (cmax[0] - cmin[0] + 1e-10)
    scale_y = (1 - 2 * margin) / (cmax[1] - cmin[1] + 1e-10)
    scale = min(scale_x, scale_y)
    
    centers[:, 0] = (centers[:, 0] - cmin[0]) * scale + margin
    centers[:, 1] = (centers[:, 1] - cmin[1]) * scale + margin
    radii *= scale
    
    return centers, radii


def obj_func(x, n, penalty_mult=1000):
    """Objective function with smooth penalties"""
    centers = x[:2 * n].reshape(n, 2)
    radii = x[2 * n:]
    
    # Primary objective: maximize sum of radii
    obj = -np.sum(radii)
    
    penalty = 0.0
    w = penalty_mult
    
    # Boundary constraints: r <= x, r <= y, r <= 1-x, r <= 1-y
    for i in range(n):
        xi, yi, ri = centers[i, 0], centers[i, 1], radii[i]
        violation = ri - xi
        if violation > 0:
            penalty += w * violation ** 2
        violation = ri - yi
        if violation > 0:
            penalty += w * violation ** 2
        violation = ri - (1 - xi)
        if violation > 0:
            penalty += w * violation ** 2
        violation = ri - (1 - yi)
        if violation > 0:
            penalty += w * violation ** 2
    
    # Non-overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            diff = centers[i] - centers[j]
            dist = np.sqrt(np.sum(diff ** 2))
            min_dist = radii[i] + radii[j]
            violation = min_dist - dist
            if violation > 0:
                penalty += w * violation ** 2
    
    # Penalty for negative radii
    for i in range(n):
        if radii[i] < 0:
            penalty += w * radii[i] ** 2
    
    return obj + penalty


def make_valid(centers, radii, n):
    """Ensure the solution satisfies all constraints"""
    # Clip radii to boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = min(radii[i], x, y, 1 - x, 1 - y)
        radii[i] = max(r, 0.0001)
    
    # Resolve overlaps by iteratively reducing radii
    for _ in range(20):
        max_overlap = 0
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                overlap = radii[i] + radii[j] - dist
                if overlap > 1e-10:
                    max_overlap = max(max_overlap, overlap)
                    # Reduce proportionally
                    total_r = radii[i] + radii[j]
                    radii[i] *= dist / total_r
                    radii[j] *= dist / total_r
        
        if max_overlap < 1e-12:
            break
    
    return centers, radii


def iterative_optimize(centers, radii, n):
    """Iteratively grow radii and adjust positions"""
    for iteration in range(100):
        # Step 1: Compute maximum possible radii given current positions
        new_radii = compute_max_radii(centers, radii, n)
        
        # Step 2: Reposition circles using gradient-based repulsion
        centers = reposition_circles(centers, new_radii, n, step_size=0.0005)
        
        radii = new_radii
    
    return centers, radii


def compute_max_radii(centers, radii, n):
    """Compute the maximum radius for each circle given positions and other radii"""
    new_radii = np.copy(radii)
    
    for iteration in range(10):
        for i in range(n):
            x, y = centers[i]
            r = min(x, y, 1 - x, 1 - y)  # Boundary constraint
            
            for j in range(n):
                if i != j:
                    dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                    r = min(r, dist - new_radii[j])  # Non-overlap
            
            new_radii[i] = max(r, 0.0001)
    
    return new_radii


def reposition_circles(centers, radii, n, step_size=0.001):
    """Reposition circles to maximize minimum clearance"""
    for step in range(200):
        grad = np.zeros_like(centers)
        
        for i in range(n):
            for j in range(i + 1, n):
                diff = centers[j] - centers[i]
                dist = np.sqrt(np.sum(diff ** 2))
                if dist < 1e-10:
                    continue
                
                target = radii[i] + radii[j]
                # Repulsion force when circles are too close
                if dist < target + 0.005:
                    force = (target + 0.005 - dist) / (dist + 1e-10)
                    grad[i] -= diff * force
                    grad[j] += diff * force
            
            # Push away from boundaries
            xi, yi = centers[i]
            ri = radii[i]
            if xi < ri + 0.01:
                grad[i, 0] += 0.1
            if xi > 1 - ri - 0.01:
                grad[i, 0] -= 0.1
            if yi < ri + 0.01:
                grad[i, 1] += 0.1
            if yi > 1 - ri - 0.01:
                grad[i, 1] -= 0.1
        
        centers += step_size * grad
        centers = np.clip(centers, 0.005, 0.995)
    
    return centers
