def constraint_boundaries(vars):
    n = len(vars) // 3

    for i in range(n):
        x = vars[3*i]
        y = vars[3*i + 1]
        r = vars[3*i + 2]
        result.append(x - r)
        result.append(1 - x - r)
        result.append(y - r)
        result.append(1 - y - r)
        result.append(r - 1e-10)

def constraint_overlap(vars):
    n = len(vars) // 3

    # Start with circles in a grid pattern
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.05
    
    # Place in a rough hexagonal pattern
    idx = 0
    row = 0
    while idx < n:
        y = 0.1 + row * 0.16
        count = 5 if row % 2 == 0 else 4
        count = min(count, n - idx)
        for col in range(count):
            x_offset = 0.08 if row % 2 == 1 else 0
            x = x_offset + col * 0.2
            centers[idx] = [x, y]
            idx += 1
        row += 1
    
    # Force-directed simulation
    for it in range(iterations):
        forces = np.zeros((n, 2))
        
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[j, 0] - centers[i, 0]
                dy = centers[j, 1] - centers[i, 1]
                dist = np.sqrt(dx**2 + dy**2)

                    # Repulsive force
                    if dist > 0:
                        force = (radii[i] + radii[j] + 0.01 - dist) / dist
                        fx = dx * force
                        fy = dy * force
                        forces[i] += [-fx, -fy]
                        forces[j] += [fx, fy]

        # Keep within bounds
        for i in range(n):
            centers[i, 0] = np.clip(centers[i, 0], radii[i], 1 - radii[i])
            centers[i, 1] = np.clip(centers[i, 1], radii[i], 1 - radii[i])
        
        # Gradually increase radii
        radii += 0.001
        radii = np.clip(radii, 0, 0.5)
