import numpy as np
from scipy.optimize import minimize, NonlinearConstraint

N_CIRCLES = 26


def objective(x):
    radii = x[2 * N_CIRCLES:]
    return -np.sum(radii)


def constraint_boundary(x):
    centers = x[:2 * N_CIRCLES].reshape(N_CIRCLES, 2)
    radii = x[2 * N_CIRCLES:]
    c = []
    for i in range(N_CIRCLES):
        c.append(centers[i, 0] - radii[i])
        c.append(centers[i, 1] - radii[i])
        c.append(1.0 - centers[i, 0] - radii[i])
        c.append(1.0 - centers[i, 1] - radii[i])
        c.append(radii[i])
    return np.array(c)


def constraint_overlap(x):
    centers = x[:2 * N_CIRCLES].reshape(N_CIRCLES, 2)
    radii = x[2 * N_CIRCLES:]
    c = []
    for i in range(N_CIRCLES):
        for j in range(i + 1, N_CIRCLES):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            c.append(dist - radii[i] - radii[j])
    return np.array(c)


def init_smart(n):
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    idx = 0

    corners = [[0.10, 0.10], [0.90, 0.10], [0.10, 0.90], [0.90, 0.90]]
    for corner in corners:
        if idx < n:
            centers[idx] = corner
            radii[idx] = 0.10
            idx += 1

    edges = [
        [0.50, 0.06], [0.06, 0.50], [0.94, 0.50], [0.50, 0.94],
        [0.30, 0.06], [0.70, 0.06], [0.06, 0.30], [0.06, 0.70],
        [0.94, 0.30], [0.94, 0.70], [0.30, 0.94], [0.70, 0.94]
    ]
    for edge in edges:
        if idx < n:
            centers[idx] = edge
            radii[idx] = 0.06
            idx += 1

    interior = [
        [0.50, 0.50], [0.30, 0.30], [0.70, 0.30], [0.30, 0.70], [0.70, 0.70],
        [0.50, 0.30], [0.50, 0.70], [0.30, 0.50], [0.70, 0.50],
        [0.20, 0.20], [0.80, 0.20], [0.20, 0.80], [0.80, 0.80]
    ]
    for point in interior:
        if idx < n:
            centers[idx] = point
            radii[idx] = 0.04
            idx += 1

    return centers, radii


def init_hexagonal(n):
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.035
    idx = 0
    row = 0
    y = 0.08
    while idx < n:
        offset = 0.08 if row % 2 == 1 else 0.0
        x = 0.05 + offset
        while idx < n and x <= 0.95:
            if y <= 0.95:
                centers[idx] = [x, y]
                idx += 1
            x += 0.16
        y += 0.14
        row += 1
    return centers, radii


def run_packing():
    np.random.seed(123)
    best_sum = -np.inf
    best_centers = None
    best_radii = None

    initializations = [
        init_smart(N_CIRCLES),
        init_hexagonal(N_CIRCLES),
    ]

    for init_func_result in initializations:
        for restart in range(3):
            centers, radii = init_func_result.copy() if isinstance(init_func_result, tuple) else init_func_result
            if restart > 0:
                centers_pert = centers.copy()
                centers_pert += np.random.randn(*centers.shape) * 0.01
                radii_pert = radii.copy()
                radii_pert += np.random.randn(N_CIRCLES) * 0.005
                radii_pert = np.maximum(radii_pert, 0.01)
                centers_pert = np.clip(centers_pert, 0.02, 0.98)
                centers = centers_pert
                radii = radii_pert
            else:
                centers = centers.copy()
                radii = radii.copy()

            x0 = np.concatenate([centers.flatten(), radii])

            bounds = [(0, 1)] * (2 * N_CIRCLES) + [(0, 0.5)] * N_CIRCLES

            bc = NonlinearConstraint(constraint_boundary, 0, np.inf)
            oc = NonlinearConstraint(constraint_overlap, 0, np.inf)

            result = minimize(
                objective,
                x0,
                method='trust-constr',
                bounds=bounds,
                constraints=[bc, oc],
                options={'maxiter': 3000, 'verbose': 0, 'xtol': 1e-12, 'gtol': 1e-10}
            )

            final_radii = result.x[2 * N_CIRCLES:]
            sum_r = np.sum(final_radii)

            if sum_r > best_sum:
                best_sum = sum_r
                best_centers = result.x[:2 * N_CIRCLES].reshape(N_CIRCLES, 2).copy()
                best_radii = final_radii.copy()

    return best_centers, best_radii, float(best_sum)
