import numpy as np
import scipy.optimize as opt

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    Returns (centers, radii, sum_radii) for a packing of 26 circles in a unit square.
    """
    n = 26
    
    # Helper to compute penalty for a given state
    # State is flattened array: [x1, y1, r1, x2, y2, r2, ...]
    def evaluate_packing(state):
        centers = state[:2*n].reshape(n, 2)
        radii = state[2*n:]
        
        penalty = 0.0
        sum_radii = np.sum(radii)
        
        # Boundary constraints penalty
        # x - r >= 0 => r - x <= 0
        # x + r <= 1 => x + r - 1 <= 0
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            
            # Check boundaries
            if x - r < 0: penalty += (r - x)**2
            if x + r > 1: penalty += (x + r - 1)**2
            if y - r < 0: penalty += (r - y)**2
            if y + r > 1: penalty += (y + r - 1)**2
            
            # Radii non-negative
            if r < 0: penalty += r**2
            
        # Overlap constraints penalty
        # dist >= r_i + r_j => r_i + r_j - dist <= 0
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                sum_r = radii[i] + radii[j]
                if sum_r > dist:
                    penalty += (sum_r - dist)**2
                    
        # We want to maximize sum_radii, so minimize -sum_radii + penalty
        # We scale penalty to ensure it dominates when constraints are violated
        return -sum_radii + 1000.0 * penalty

    best_state = None

    rows_counts = [5, 4, 5, 4, 5, 3]

    centers_hex = []
    radii_hex = []

    if best_state is not None:

    centers = np.clip(centers, 0, 1)

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    try:
        import numpy as np
    except ImportError:
        return False

    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        return False

    if np.isnan(radii).any():
        return False

    # Check if radii are nonnegative
    for i in range(n):
        if radii[i] < 0:
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False

    return True
