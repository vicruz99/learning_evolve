import numpy as np
import scipy.optimize as opt
import math

def hexagonal_grid(n, width=1.0, height=1.0, margin=0.02):
    """
    Generates a hexagonal grid of points.
    """
    points = []
    # Estimate spacing for a hexagonal grid
    # Area per point approx sqrt(3)/2 * s^2
    # n * sqrt(3)/2 * s^2 approx (width * height)
    # s approx sqrt( 2 * area / (n * sqrt(3)) )
    area = width * height
    if n == 0:
        return np.array([]).reshape(0, 2)
    s = math.sqrt(2.0 * area / (n * math.sqrt(3)))
    
    y = margin + s * math.sqrt(3) / 2
    row = 0
    while y <= height - margin:
        x = margin
        if row % 2 == 1:
            x += s / 2
        while x <= width - margin:
            points.append([x, y])
            if len(points) >= n:
                break
            x += s
        if len(points) >= n:
            break
        y += s * math.sqrt(3) / 2
        row += 1
        
    if len(points) < n:
        # Fill remaining with random or simple grid if needed, though unlikely for 26
        while len(points) < n:
            points.append([np.random.rand() * width, np.random.rand() * height])
            
    return np.array(points[:n])

def get_initial_guess(n):
    """
    Create a diverse set of initial guesses.
    """
    guesses = []
    
    # 1. Hexagonal grid centered
    pts = hexagonal_grid(n)
    # Normalize to fit strictly inside with some margin
    pts = pts * 0.9 + 0.05 
    guesses.append(pts)
    
    # 2. Random points with some spread
    pts_rand = np.random.rand(n, 2) * 0.8 + 0.1
    guesses.append(pts_rand)
    
    # 3. Perturbed hex grid
    pts_pert = pts + np.random.randn(n, 2) * 0.02
    pts_pert = np.clip(pts_pert, 0.05, 0.95)
    guesses.append(pts_pert)
    
    return guesses

def objective(vars, n):
    """
    Objective function: Minimize -(sum of radii) + penalty for violations.
    vars structure: [x0, y0, r0, x1, y1, r1, ...]
    """
    centers = vars[:2*n].reshape(n, 2)
    radii = vars[2*n:]
    
    # Soft penalty for radii going outside bounds or negative
    # We want r >= 0, x-r >= 0, x+r <= 1, y-r >= 0, y+r <= 1
    # And dist(i,j) >= r_i + r_j
    
    sum_radii = np.sum(radii)
    penalty = 0.0
    
    # Boundary penalties (soft constraints)
    # If circle is outside, add large penalty
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        
        # Radius must be non-negative
        if r < 0:
            penalty += 1000 * abs(r)
            
        # Inside square
        violations = []
        if x - r < 0: violations.append(x - r)
        if x + r > 1: violations.append(x + r - 1)
        if y - r < 0: violations.append(y - r)
        if y + r > 1: violations.append(y + r - 1)
        
        for v in violations:
            if v < 0:
                penalty += 100 * v * v # Quadratic penalty for being outside

    # Overlap penalties
    # dist >= r_i + r_j  =>  dist - r_i - r_j >= 0
    # If negative, penalty
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            overlap = radii[i] + radii[j] - dist
            if overlap > 0:
                penalty += 500 * overlap * overlap
                
    return -sum_radii + penalty

def run_packing():
    n = 26
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    guesses = get_initial_guess(n)
    
    # Optimization bounds
    # x, y in [0, 1]
    # r in [0, 0.5]
    bounds = []
    for i in range(n):
        bounds.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
        
    for guess_centers in guesses:
        # Initial radii: small value to start feasible
        initial_radii = np.ones(n) * 0.01
        initial_vars = np.concatenate([guess_centers.flatten(), initial_radii])
        
        try:
            result = opt.minimize(
                objective,
                initial_vars,
                args=(n,),
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 1000, 'ftol': 1e-9}
            )
            
            if result.success or result.fun < -best_sum: # Since we minimize negative sum
                current_centers = result.x[:2*n].reshape(n, 2)
                current_radii = result.x[2*n:]
                
                # Validate and compute actual sum
                # The optimizer might have allowed slight overlaps if penalty wasn't high enough,
                # but with high penalty it should be close.

                # No, the problem asks for a valid packing.

                is_valid = True
                # Check bounds

                    x, y = current_centers[i]
                    r = current_radii[i]
                    if r < -1e-7: is_valid = False
                    if x - r < -1e-7 or x + r > 1 + 1e-7 or y - r < -1e-7 or y + r > 1 + 1e-7:
                        is_valid = False
                
                # Check overlaps
                if is_valid:
                    for i in range(n):
                        for j in range(i+1, n):
                            dist = np.sqrt(np.sum((current_centers[i] - current_centers[j])**2))
                            if dist < current_radii[i] + current_radii[j] - 1e-7:
                                is_valid = False
                                break
                        if not is_valid: break
                
                if is_valid:
                    current_sum = np.sum(current_radii)
                    if current_sum > best_sum:
                        best_sum = current_sum
                        best_centers = current_centers
                        best_radii = current_radii
                        
        except Exception as e:

    # If we didn't find a valid one or want to improve, we can do a clean-up step
    # But the optimizer with penalties should find a local optimum.
    
    # Fallback: if best_centers is None, return a simple grid
    if best_centers is None:
        pts = hexagonal_grid(n)
        pts = pts * 0.9 + 0.05
        r = 0.02
        best_centers = pts
        best_radii = np.ones(n) * r
        best_sum = n * r

    return best_centers, best_radii, best_sum
