import numpy as np
from scipy.optimize import minimize
import warnings

warnings.filterwarnings('ignore')

def calculate_loss(centers, radii):
    """
    Calculate a loss function based on constraint violations and sum of radii.
    We want to maximize sum(radii), so we minimize negative sum.
    Violations are penalized heavily.
    """
    n = len(radii)
    penalty = 0.0
    target_penalty_scale = 100.0
    
    # 1. Boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        
        # Distance to walls
        dist_left = x - r
        dist_right = 1.0 - (x + r)
        dist_bottom = y - r
        dist_top = 1.0 - (y + r)
        
        min_dist = min(dist_left, dist_right, dist_bottom, dist_top)
        if min_dist < 0:
            penalty += target_penalty_scale * (min_dist ** 2)

    # 2. Overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            required_dist = radii[i] + radii[j]
            
            # Violation is how much they overlap
            overlap = required_dist - dist
            if overlap > 1e-6: # Small tolerance
                penalty += target_penalty_scale * (overlap ** 2)

    # Objective: Maximize sum of radii -> Minimize negative sum
    return -np.sum(radii) + penalty

def get_initial_guess():

    Generate a valid initial configuration of 26 circles.
    Uses a hexagonal lattice pattern.

    centers = []
    radii = []

    # Pattern: 5, 4, 5, 4, 5, 3 circles in each row.
    # Total: 26.
    
    rows = [5, 4, 5, 4, 5, 3]
    r_guess = 0.08 # Start with a safe radius
    
    # Vertical spacing for hex packing is r * sqrt(3)
    # But we don't know r exactly yet, so we place them and scale later.
    # Let's just place centers in a bounding box [0.1, 0.9]
    
    y_step = 1.0 / 6.0
    x_step = 1.0 / 6.0
    
    current_y = 0.1
    
    count = 0
    for i, num_circles in enumerate(rows):
        # Shift x for odd rows to create hex pattern
        shift = x_step / 2.0 if i % 2 != 0 else 0.0
        
        # Calculate start x to center the row
        row_width = (num_circles - 1) * x_step
        start_x = 0.1 + (0.8 - row_width) / 2.0 # Center in [0.1, 0.9]
        
        if i % 2 != 0:
            start_x += shift # Apply shift
            
        for j in range(num_circles):
            x = start_x + j * x_step
            y = current_y
            centers.append([x, y])
            radii.append(r_guess)
            count += 1
            
        current_y += y_step

    # Normalize coordinates to be safely inside [0, 1]
    centers = np.array(centers)
    # Scale down slightly to ensure no boundary violation with initial r
    # Current max coord is around 0.9. r=0.08. 0.9+0.08 = 0.98. Safe.
    
    return centers, np.array(radii)

    """

    # Initial guess
    centers, radii = get_initial_guess()

    # Bounds:

    # r in [0, 0.5]
    bounds = []

        'ftol': 1e-9,
        'disp': False
