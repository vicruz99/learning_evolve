import numpy as np
import scipy.optimize as opt

def get_constraints(centers, radii):
    n = centers.shape[0]
    cons = []
    
    # Boundary constraints: r <= x <= 1-r  =>  r - x <= 0, x + r - 1 <= 0
    # Same for y
    for i in range(n):
        cons.append({'type': 'ineq', 'fun': lambda v, i=i: v[2*i] - v[2*n + i]}) # x_i - r_i >= 0 => r_i - x_i <= 0 (ineq form: fun >= 0)
        # Wait, scipy ineq constraint is fun(x) >= 0.
        # We need x_i >= r_i => x_i - r_i >= 0.
        # And 1 - x_i >= r_i => 1 - x_i - r_i >= 0.
        
        # Let's map variables:
        # vars = [x0, y0, x1, y1, ..., r0, r1, ...]
        # x_i is at 2*i, y_i at 2*i+1, r_i at 2*n + i
        
        # Constraint 1: x_i - r_i >= 0
        cons.append({'type': 'ineq', 'fun': lambda v, i=i: v[2*i] - v[2*n + i]})
        # Constraint 2: 1 - x_i - r_i >= 0
        cons.append({'type': 'ineq', 'fun': lambda v, i=i: 1 - v[2*i] - v[2*n + i]})
        # Constraint 3: y_i - r_i >= 0
        cons.append({'type': 'ineq', 'fun': lambda v, i=i: v[2*i+1] - v[2*n + i]})
        # Constraint 4: 1 - y_i - r_i >= 0
        cons.append({'type': 'ineq', 'fun': lambda v, i=i: 1 - v[2*i+1] - v[2*n + i]})

    # Non-overlap constraints: dist(i, j) >= r_i + r_j
    # dist^2 >= (r_i + r_j)^2 is non-convex? 
    # dist >= r_i + r_j is standard.
    # Constraint: dist(i, j) - r_i - r_j >= 0
    for i in range(n):
        for j in range(i + 1, n):
            # dist = sqrt((xi-xj)^2 + (yi-yj)^2)
            # fun = sqrt(...) - ri - rj
            def fun(v, i=i, j=j):
                xi, yi = v[2*i], v[2*i+1]
                xj, yj = v[2*j], v[2*j+1]
                ri, rj = v[2*n + i], v[2*n + j]
                dist = np.sqrt((xi - xj)**2 + (yi - yj)**2)
                return dist - ri - rj
            cons.append({'type': 'ineq', 'fun': fun})
            
    return cons
