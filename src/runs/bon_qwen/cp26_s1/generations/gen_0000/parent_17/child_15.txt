def check_constraints(centers, radii):
    """
    Check constraints and return violation metrics.
    """
    n = len(radii)
    violations = 0
    
    # Boundary violations
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x < r or x > 1 - r:
            violations += 1
        if y < r or y > 1 - r:
            violations += 1
            
    # Overlap violations
    for i in range(n):
        for j in range(i + 1, n):
            dist_sq = (centers[i] - centers[j])**2
            dist = np.sqrt(np.sum(dist_sq))
            if dist < radii[i] + radii[j]:
                violations += 1
                
    return violations
