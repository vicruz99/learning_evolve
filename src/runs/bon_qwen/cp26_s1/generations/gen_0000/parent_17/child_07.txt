def compute_overlap_penalty(centers, radii):
    """
    Computes the sum of squared penetration depths for all pairs of circles.
    penetration = max(0, r_i + r_j - distance)
    """
    n = centers.shape[0]
    penalty = 0.0
    
    # Vectorized distance calculation
    # Centers shape (n, 2)
    # diffs shape (n, n, 2)
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    
    # Radii sum shape (n, n)
    rad_sums = radii[:, np.newaxis] + radii[np.newaxis, :]
    
    # Penetration
    penetration = rad_sums - dists
    
    # We only care about positive penetration (overlap)
    # Use np.maximum for vectorized max(0, x)
    overlap = np.maximum(0.0, penetration)
    
    # Sum of squared overlaps, avoiding double counting (i, j) and (j, i) and (i, i)
    # Triangle mask
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    penalty = np.sum(overlap[mask]**2)
    
    return penalty

def compute_boundary_penalty(centers, radii):
    """
    Computes penalty for circles going outside [0,1]x[0,1] or having radius > coordinate constraints.
    Constraints: r <= x <= 1-r, r <= y <= 1-r
    """
    n = centers.shape[0]
    penalty = 0.0
    
    # x constraints
    # x >= r  => r - x <= 0
    viol_x_low = np.maximum(0.0, radii - centers[:, 0])
    # x <= 1-r => x + r - 1 <= 0
    viol_x_high = np.maximum(0.0, centers[:, 0] + radii - 1.0)
    
    # y constraints
    viol_y_low = np.maximum(0.0, radii - centers[:, 1])
    viol_y_high = np.maximum(0.0, centers[:, 1] + radii - 1.0)
    
    # Sum of squares
    penalty = np.sum(viol_x_low**2) + np.sum(viol_x_high**2) + \
              np.sum(viol_y_low**2) + np.sum(viol_y_high**2)
              
    return penalty
