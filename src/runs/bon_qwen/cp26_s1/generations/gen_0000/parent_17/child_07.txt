import numpy as np
from scipy.optimize import minimize

def get_hex_grid(n):
    """Generates a hexagonal grid of n points to be used as initial centers."""
    # Estimate grid spacing for 26 circles in unit square
    r_est = np.sqrt(0.9 / (n * np.pi))
    s = 2 * r_est
    
    points = []
    row = 0
    while len(points) < n:
        y = row * s * np.sqrt(3) / 2
        offset = (row % 2) * s / 2
        col = 0
        while True:
            x = offset + col * s
            if x > 1 or y > 1: 
                break
            points.append([x, y])
            col += 1
            if len(points) >= n: break
        row += 1
        
    # If we didn't get enough points, fallback to random
    if len(points) < n:
        return np.random.rand(n, 2)
        
    points = np.array(points[:n])
    
    # Center the grid in the unit square
    min_c = np.min(points, axis=0)
    max_c = np.max(points, axis=0)
    center_c = (max_c + min_c) / 2
    points -= center_c
    points += 0.5
    
    return points

def objective_function(vars, centers_shape):
    """
    Objective function: Minimize negative sum of radii plus constraint penalties.
    vars: flattened array of [x0, y0, r0, x1, y1, r1, ...]
    """
    n = centers_shape[0]
    centers = vars[0:2*n].reshape((n, 2))
    radii = vars[2*n:3*n]
    
    # 1. Objective term: -Sum(radii)
    obj_val = -np.sum(radii)
    
    # 2. Boundary Penalties: Circle must be inside [0,1]x[0,1]
    # r <= x, r <= 1-x, r <= y, r <= 1-y => 0 <= x-r, x+r <= 1, etc.
    # Violation = max(0, lower_bound - val)
    x, y = centers[:, 0], centers[:, 1]
    
    violations_boundary = np.zeros(4 * n)
    violations_boundary[0:n] = np.maximum(0, radii - x)           # x - r < 0
    violations_boundary[n:2*n] = np.maximum(0, radii - (1 - x))   # x + r > 1
    violations_boundary[2*n:3*n] = np.maximum(0, radii - y)       # y - r < 0
    violations_boundary[3*n:] = np.maximum(0, radii - (1 - y))    # y + r > 1
    
    obj_val += 100.0 * np.sum(violations_boundary ** 2)
    
    # 3. Overlap Penalties: dist(i, j) >= r_i + r_j
    # Violation = max(0, r_i + r_j - dist)
    # Vectorized distance computation
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    # We only need upper triangle indices (i < j)
    i, j = np.triu_indices(n, k=1)
    overlap_violations = np.maximum(0, radii[i] + radii[j] - dists[i, j])
    
    obj_val += 100.0 * np.sum(overlap_violations ** 2)
    
    return obj_val

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Constants
    n = 26
    bounds_list = []
    
    # Bounds for x, y, r: x,y in [0, 1], r in [0, 0.5]
    for _ in range(n):
        bounds_list.extend([
            (0.0, 1.0), (0.0, 1.0), (0.0, 0.5)
        ])
        
    # Initial Guess
    centers_init = get_hex_grid(n)
    # Initial radii: small but non-zero to help convergence
    radii_init = np.ones(n) * 0.01
    
    # Flatten initial guess
    x0 = np.concatenate([centers_init.flatten(), radii_init])
    
    # Optimization
    result = minimize(
        objective_function,
        x0,
        args=(centers_init.shape,),
        method='L-BFGS-B',
        bounds=bounds_list,
        options={'maxiter': 2000, 'ftol': 1e-15}
    )
    
    # Extract optimized variables
    centers = result.x[0:2*n].reshape((n, 2))
    radii = result.x[2*n:3*n]
    
    # Projection to ensure strict boundary validity (clipping)
    # Clamp centers to [r, 1-r] effectively by adjusting position
    # If center is at 0 and radius is 0.1, move center to 0.1
    for i in range(n):
        r = radii[i]
        x, y = centers[i]
        
        # Boundary correction
        if x < r: x = r
        if x > 1 - r: x = 1 - r
        if y < r: y = r
        if y > 1 - r: y = 1 - r
        
        centers[i] = [x, y]
        
    return centers, radii, np.sum(radii)
