def make_initial(n, trial):

def hexagonal_init(n, seed):

    radii = np.zeros(n)

    idx = 0
    row = 0

    while idx < n:
        col = 0
        while idx < n:

            y = r + row * r * np.sqrt(3)
            if row % 2 == 1:
                x += r

                centers[idx] = [x, y]
                radii[idx] = r
                idx += 1

            if x + 2 * r > 1:
                break
            col += 1

            break
        row += 1

def random_init(n, seed):
    """Random initialization with small circles."""
    np.random.seed(seed)
    centers = np.random.rand(n, 2)
    radii = np.ones(n) * 0.03

    # Ensure boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = min(x, 1 - x, y, 1 - y) * 0.5
        radii[i] = r

    return centers, radii


def packing_objective(x):
    """Objective: maximize sum of radii (minimize negative)."""
    n = len(x) // 3
    return -np.sum(x[2 * n :])


def make_bounds(n):
    """Variable bounds."""
    bounds = []
    for i in range(n):
        bounds.append((0.0, 1.0))  # x
        bounds.append((0.0, 1.0))  # y
        bounds.append((1e-12, 0.5))  # radius
    return bounds


def make_constraints(n):
    """Inequality constraints: g(x) >= 0."""

    def boundary(x):
        c = x[: 2 * n].reshape(n, 2)
        r = x[2 * n :]
        out = []
        for i in range(n):
            out.append(c[i, 0] - r[i])
            out.append(1.0 - c[i, 0] - r[i])
            out.append(c[i, 1] - r[i])
            out.append(1.0 - c[i, 1] - r[i])
        return np.array(out)

    def overlaps(x):
        c = x[: 2 * n].reshape(n, 2)
        r = x[2 * n :]
        out = []
        for i in range(n):
            for j in range(i + 1, n):
                dx = c[i, 0] - c[j, 0]
                dy = c[i, 1] - c[j, 1]
                d2 = dx * dx + dy * dy
                rs = r[i] + r[j]
                out.append(d2 - rs * rs)
        return np.array(out)

    return [
        {"type": "ineq", "fun": boundary},
        {"type": "ineq", "fun": overlaps},
    ]


def is_feasible(centers, radii, n):
    """Check if configuration satisfies all constraints."""
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if r < 0:
            return False
        if x - r < -1e-10 or x + r > 1 + 1e-10:
            return False
        if y - r < -1e-10 or y + r > 1 + 1e-10:
            return False

    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            d = np.sqrt(dx * dx + dy * dy)
            if d < radii[i] + radii[j] - 1e-10:
                return False

    return True
