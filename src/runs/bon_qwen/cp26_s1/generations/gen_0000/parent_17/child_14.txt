import numpy as np
from scipy.optimize import minimize

def objective(x, n):
    """Objective function: negative sum of radii (to be minimized)."""
    radii = x[2 * n:]
    return -np.sum(radii)

def eval_constraints(x, n):
    """
    Computes all inequality constraints as a single array.
    Returns values that must be >= 0.
    """
    centers = x[:2 * n].reshape(n, 2)
    radii = x[2 * n:]
    
    # We will collect constraint values in a list then convert to array
    cons = []
    
    # Boundary constraints: x - r >= 0, 1 - x - r >= 0, etc.
    for i in range(n):
        xi, yi, ri = centers[i, 0], centers[i, 1], radii[i]
        cons.append(xi - ri)
        cons.append(1.0 - xi - ri)
        cons.append(yi - ri)
        cons.append(1.0 - yi - ri)
        
    # Pairwise non-overlap constraints: dist^2 - (r_i + r_j)^2 >= 0
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist_sq = dx * dx + dy * dy
            r_sum = radii[i] + radii[j]
            cons.append(dist_sq - r_sum * r_sum)
            
    return np.array(cons)

def get_bounds(n):
    """Creates bounds for all variables: x, y in [0,1], r in [0, 0.25]."""
    b = []
    for _ in range(n):
        b.append((0.0, 1.0))
        b.append((0.0, 1.0))
        b.append((0.0, 0.25))
    return b

def generate_initial_configs(n):
    """Generates multiple valid initial configurations."""
    configs = []
    r_start = 0.085
    
    # 1. Hexagonal packing
    cx, cy = [], []
    for row in range(6):
        y = r_start + row * r_start * 1.732
        for col in range(6):
            x = r_start + col * 2 * r_start + (row % 2) * r_start
            if x < 1 - r_start and y < 1 - r_start:
                cx.append(x)
                cy.append(y)
            if len(cx) >= n:
                break
        if len(cx) >= n:
            break
            
    cx, cy = np.array(cx[:n]), np.array(cy[:n])
    # Add slight perturbation
    c1 = np.column_stack((cx + np.random.uniform(-0.02, 0.02, n), 
                          cy + np.random.uniform(-0.02, 0.02, n)))
    c1 = np.clip(c1, r_start, 1 - r_start)
    configs.append(np.hstack([c1.ravel(), np.ones(n) * r_start]))
    
    # 2. 5x5 Grid + 1
    grid_pts = []
    for i in range(5):
        for j in range(5):
            grid_pts.append([0.1 + i * 0.2, 0.1 + j * 0.2])
    grid_pts.append([0.5, 0.5])
    c2 = np.array(grid_pts[:n])
    c2 = c2 + np.random.uniform(-0.01, 0.01, c2.shape)
    c2 = np.clip(c2, r_start, 1 - r_start)
    configs.append(np.hstack([c2.ravel(), np.ones(n) * r_start]))
    
    # 3. Structured Random (Lloyd-like initialization)
    np.random.seed(42)
    pts = np.random.uniform(0.2, 0.8, (n, 2))
    # Simple repulsion to avoid initial overlaps
    for _ in range(100):
        for i in range(n):
            for j in range(i + 1, n):
                diff = pts[i] - pts[j]
                dist = np.linalg.norm(diff)
                if dist < 0.15 and dist > 1e-6:
                    force = (0.15 - dist) * diff / dist
                    pts[i] += 0.5 * force
                    pts[j] -= 0.5 * force
    c3 = np.clip(pts, r_start, 1 - r_start)
    configs.append(np.hstack([c3.ravel(), np.ones(n) * (r_start - 0.01)]))
    
    return configs

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    np.random.seed(123)  # For reproducibility of perturbations
    
    bnds = get_bounds(n)
    cons = {'type': 'ineq', 'fun': eval_constraints, 'args': (n,)}
    
    best_sum = 0.0
    best_x = None
    
    initial_configs = generate_initial_configs(n)
    
    for x0 in initial_configs:
        try:
            res = minimize(
                objective, 
                x0, 
                args=(n,), 
                method='SLSQP', 
                bounds=bnds, 
                constraints=cons, 
                options={'maxiter': 3000, 'ftol': 1e-12, 'disp': False}
            )
            
            current_sum = -res.fun
            if current_sum > best_sum:
                best_sum = current_sum
                best_x = res.x.copy()
        except Exception:
            continue
            
    # Fallback if optimization fails completely (should not happen)
    if best_x is None:
        best_x = initial_configs[0]
        
    centers = best_x[:2 * n].reshape(n, 2)
    radii = best_x[2 * n:]
    
    return centers, radii, float(np.sum(radii))
