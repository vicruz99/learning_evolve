import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    Strategy:
    1. Initialization: Start with a grid-like configuration.
    2. Iterative Expansion: Physically grow the circles and resolve overlaps to find a feasible, dense packing.
    3. Gradient Optimization: Use SLSQP to fine-tune centers and radii for maximum sum.
    """
    n = 26
    
    # --- Stage 1: Iterative Expansion ---
    # Initialize centers in a dense grid pattern
    # 6x5 grid has 30 spots, we pick 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    grid_x = np.linspace(0.1, 0.9, 5) # 5 points
    grid_y = np.linspace(0.1, 0.9, 6) # 6 points
    
    idx = 0
    for y in grid_y:
        for x in grid_x:
            if idx < n:
                centers[idx] = [x, y]
                radii[idx] = 0.05 # Initial radius
                idx += 1
    
    # Expand circles iteratively
    growth_rate = 1.001
    max_radius = 0.12 # Safety cap
    for step in range(500):
        radii *= growth_rate
        radii = np.clip(radii, 0, max_radius)
        
        # Resolve overlaps with simulated annealing-like forces
        velocities = np.zeros_like(centers)
        for _ in range(100): # Sub-steps for physics resolution
            forces = np.zeros_like(centers)
            
            # Pairwise repulsion
            for i in range(n):
                for j in range(i + 1, n):
                    dx = centers[i, 0] - centers[j, 0]
                    dy = centers[i, 1] - centers[j, 1]
                    dist = np.sqrt(dx*dx + dy*dy)
                    min_dist = radii[i] + radii[j]
                    
                    if dist < min_dist and dist > 1e-9:
                        overlap = min_dist - dist
                        force_mag = overlap * 50.0 # Strong repulsion
                        forces[i, 0] += (dx / dist) * force_mag
                        forces[i, 1] += (dy / dist) * force_mag
                        forces[j, 0] -= (dx / dist) * force_mag
                        forces[j, 1] -= (dy / dist) * force_mag
            
            # Boundary repulsion
            for i in range(n):
                r = radii[i]
                x, y = centers[i]
                
                # Left wall
                if x < r:
                    forces[i, 0] += (r - x) * 50.0
                # Right wall
                if x > 1 - r:
                    forces[i, 0] -= (x - (1 - r)) * 50.0
                # Bottom wall
                if y < r:
                    forces[i, 1] += (r - y) * 50.0
                # Top wall
                if y > 1 - r:
                    forces[i, 1] -= (y - (1 - r)) * 50.0

            # Update velocities and positions
            velocities += forces * 0.01
            velocities *= 0.8 # Damping
            centers += velocities
            
            # Clamp to square
            np.clip(centers, radii, 1 - radii, out=centers)

    # --- Stage 2: Gradient Optimization ---
    # Combine centers and radii into one vector for optimization
    x0 = np.concatenate([centers.flatten(), radii])
    
    # Objective: Maximize sum of radii -> Minimize negative sum
    def objective(vars):
        return -np.sum(vars[n*2:])

    # Constraints
    cons = []
    
    # Boundary constraints
    for i in range(n):
        r_idx = n * 2 + i
        x_idx = i * 2
        y_idx = i * 2 + 1
        
        # x - r >= 0  =>  r_idx - x_idx <= 0
        cons.append({'type': 'ineq', 'fun': lambda v, i=i: v[r_idx] - v[x_idx]})
        # x + r <= 1  =>  x_idx + r_idx - 1 <= 0
        cons.append({'type': 'ineq', 'fun': lambda v, i=i: 1 - (v[x_idx] + v[r_idx])})
        # y - r >= 0
        cons.append({'type': 'ineq', 'fun': lambda v, i=i: v[r_idx] - v[y_idx]})
        # y + r <= 1
        cons.append({'type': 'ineq', 'fun': lambda v, i=i: 1 - (v[y_idx] + v[r_idx])})

    # Pairwise non-overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            r_i = n * 2 + i
            r_j = n * 2 + j
            x_i, y_i = i * 2, i * 2 + 1
            x_j, y_j = j * 2, j * 2 + 1
            
            def dist_constraint(v, i=i, j=j):
                xi, yi = v[x_i], v[y_i]
                xj, yj = v[x_j], v[y_j]
                ri, rj = v[r_i], v[r_j]
                return (xi - xj)**2 + (yi - yj)**2 - (ri + rj)**2

            cons.append({'type': 'ineq', 'fun': dist_constraint})

    # Run optimization
    try:
        result = minimize(objective, x0, method='SLSQP', constraints=cons, 
                         options={'maxiter': 1000, 'ftol': 1e-9})
        final_vars = result.x
    except Exception:
        final_vars = x0

    # Extract final results
    final_centers = final_vars[:n*2].reshape(n, 2)
    final_radii = final_vars[n*2:]
    
    # Final cleanup to ensure strict validity
    final_radii = np.maximum(final_radii, 0)
    np.clip(final_centers, final_radii, 1 - final_radii, out=final_centers)

    sum_radii = np.sum(final_radii)
    return final_centers, final_radii, sum_radii
