import numpy as np
import scipy.optimize as opt
import math

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    Uses L-BFGS-B optimization with a penalty method for constraints.
    """
    n = 26
    num_vars = n * 3 # x, y, r for each circle
    
    # 1. Initialization
    # Create a hexagonal lattice initialization
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Estimate a starting radius. 
    # For n=26, equal circles radius is approx 0.09-0.10.
    # We start smaller to ensure feasibility.
    r_init = 0.06
    
    idx = 0
    # Hexagonal grid parameters
    # Spacing dy = r * sqrt(3), dx = 2*r
    # We scale the grid to fit roughly in the square
    # But since we don't know final r, we just spread points out.
    # Let's place them in a 5x5 grid + extras or just a dense scatter.
    # A simple dense grid is robust.
    
    # Let's try a hexagonal arrangement for 26 circles
    # Rows with y spacing sqrt(3)/2 * scale
    scale = 0.2 # approximate width per circle
    dy = scale * math.sqrt(3) / 2
    dx = scale
    
    # Generate points
    points = []
    # We want to cover the square [0,1]x[0,1]
    # Let's generate a lattice and pick the first 26 that fit in [0,1]
    # Or just generate a lattice that fits and select points.
    
    y = r_init
    row = 0
    while y <= 1 - r_init and len(points) < n:
        x = r_init if row % 2 == 0 else r_init + dx/2
        while x <= 1 - r_init and len(points) < n:
            points.append([x, y])
            x += dx
        y += dy
        row += 1
        
    if len(points) < n:
        # Fallback to random or grid if lattice didn't yield enough
        # (Should not happen with small r_init)
        grid_x = np.linspace(0.1, 0.9, 6)
        grid_y = np.linspace(0.1, 0.9, 6)
        for yy in grid_y:
            for xx in grid_x:
                if len(points) >= n: break
                points.append([xx, yy])
            if len(points) >= n: break
            
    points = np.array(points[:n])
    centers = points
    radii[:] = r_init
    
    # Flatten variables for optimizer
    # Order: x0, y0, r0, x1, y1, r1, ...
    x0 = np.zeros(num_vars)
    for i in range(n):
        x0[i*3] = centers[i, 0]
        x0[i*3 + 1] = centers[i, 1]
        x0[i*3 + 2] = radii[i]
        
    # Bounds: x, y in [0, 1], r in [0, 0.5]
    bounds = []
    for _ in range(n):
        bounds.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])

    # We maximize sum(r), so minimize -sum(r)
    # Penalty for overlaps and boundary violations
    
    penalty_weight = 1000.0 # Large weight to enforce constraints
    
    def objective(vars_flat):
        # Extract centers and radii
        c_x = vars_flat[0::3]
        c_y = vars_flat[1::3]
        r = vars_flat[2::3]
        
        # Objective: negative sum of radii
        obj = -np.sum(r)
        
        # Penalty for boundary violations
        # x - r >= 0  => violation if r > x
        # 1 - x - r >= 0 => violation if x + r > 1
        # y - r >= 0
        # 1 - y - r >= 0
        
        # Violation amounts
        v1 = np.maximum(0, r - c_x) # r > x
        v2 = np.maximum(0, c_x + r - 1) # x+r > 1
        v3 = np.maximum(0, r - c_y) # r > y
        v4 = np.maximum(0, c_y + r - 1) # y+r > 1
        
        boundary_penalty = np.sum(v1**2 + v2**2 + v3**2 + v4**2)
        
        # Penalty for overlaps
        # dist >= r_i + r_j  => violation if dist < r_i + r_j
        # We check all pairs. O(N^2) but N=26 is small (325 pairs).
        overlap_penalty = 0.0
        # Vectorized distance computation might be faster but loop is fine for N=26
        # To avoid full O(N^2) in Python loop, we can use broadcasting
        # c_x shape (n,), c_y shape (n,)
        # Diff shape (n, n)
        diff_x = c_x[:, None] - c_x[None, :]
        diff_y = c_y[:, None] - c_y[None, :]
        dist_sq = diff_x**2 + diff_y**2
        dist = np.sqrt(dist_sq)
        
        # r_sum matrix
        r_sum = r[:, None] + r[None, :]
        
        # Overlap condition: dist < r_sum
        # We want to penalize (r_sum - dist) when positive

        # Mask upper triangle to avoid double counting and self
        mask = np.triu(np.ones((n, n), dtype=bool), k=1)
        
        violation = np.maximum(0, diff_r)
        # Apply mask
        masked_violation = violation * mask
        overlap_penalty = np.sum(masked_violation**2)
        
        return obj + penalty_weight * (boundary_penalty + overlap_penalty)

    # We run optimization. Might need multiple restarts or careful tuning.
    # Given the penalty method, the landscape might be rugged, but starting close helps.

        objective, 
        x0, 
        method='L-BFGS-B', 
        bounds=bounds,
        options={'maxiter': 5000, 'ftol': 1e-12, 'gtol': 1e-8}
    )

        vars_opt = result.x
    else:

    radii_opt = np.zeros(n)

        centers_opt[i, 0] = vars_opt[i*3]
        centers_opt[i, 1] = vars_opt[i*3 + 1]
        radii_opt[i] = vars_opt[i*3 + 2]
        
    # Final validation and clamping to be safe
    # Ensure centers and radii are valid
    # If optimization pushed radius to negative (unlikely with bounds), clamp
    radii_opt = np.maximum(radii_opt, 0)
    centers_opt = np.clip(centers_opt, 0, 1)

    # A post-processing shrink step.

    max_attempts = 10
    attempt = 0
    
    while not valid and attempt < max_attempts:
        # Check constraints
        is_valid = True
        
        # Boundary
        for i in range(n):
            x, y = centers_opt[i]
            r = radii_opt[i]
            if x - r < -1e-9 or x + r > 1 + 1e-9 or y - r < -1e-9 or y + r > 1 + 1e-9:
                # Shrink radius to fit boundary
                max_r_x = min(x, 1-x)
                max_r_y = min(y, 1-y)
                radii_opt[i] = max(0, min(radii_opt[i], max_r_x, max_r_y) - 1e-7)
                is_valid = False

                dist = np.sqrt((centers_opt[i,0]-centers_opt[j,0])**2 + (centers_opt[i,1]-centers_opt[j,1])**2)
                min_dist = radii_opt[i] + radii_opt[j]
                if dist < min_dist - 1e-9:
                    # Overlap detected. Reduce radii.
                    # Simple heuristic: reduce both radii proportionally or just enough to separate?
                    # Reducing radii is safe for validity but lowers objective.
                    # Better to push centers apart?
                    # For safety, let's just reduce radii slightly.
                    # Or push centers apart.
                    
                    # Push centers apart along the line connecting them
                    dx = centers_opt[j,0] - centers_opt[i,0]
                    dy = centers_opt[j,1] - centers_opt[i,1]
                    if dist > 1e-9:
                        nx = dx / dist
                        ny = dy / dist
                        # We need dist to be at least min_dist
                        # Current dist is D. We need D' >= min_dist.
                        # Shift centers by (min_dist - D)/2 in opposite directions
                        shift = (min_dist - dist) / 2 + 1e-7
                        centers_opt[i,0] -= nx * shift
                        centers_opt[i,1] -= ny * shift
                        centers_opt[j,0] += nx * shift
                        centers_opt[j,1] += ny * shift
                        
                        # Clamp centers to square
                        centers_opt = np.clip(centers_opt, 0, 1)
                    else:
                        # Centers are same, random perturbation

                        centers_opt[j,0] -= 0.001

                    is_valid = False
        
        if is_valid:
            valid = True
        attempt += 1
            
    # Final sum
    sum_radii = np.sum(radii_opt)
    
    return centers_opt, radii_opt, sum_radii
