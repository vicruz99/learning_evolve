

To maximize the sum of radii for 26 circles in a unit square, we must achieve a dense packing where circles are as large as possible without overlapping. This is a non-convex optimization problem. We will use a multi-stage strategy:

1.  **Initialization**: Place the 26 centers in a hexagonal lattice pattern, which is the densest way to pack equal circles. This provides a geometrically sound starting point.
2.  **Expansion**: Iteratively increase the radii of all circles while using a "repulsion" force to resolve overlaps and boundary violations.
3.  **Jiggling**: To escape local optima where circles are "stuck" in a suboptimal arrangement, we introduce random perturbations (jiggling) to the centers.
4.  **Refinement**: Finally, we solve a Linear Program (LP) for the optimal radii given the final centers to maximize the sum precisely.

2. Use a force-directed expansion loop:

   e. Occasionally apply random "jiggling" to centers to prevent getting stuck in local optima.
3. After the expansion phase converges, use `scipy.optimize.linprog` to find the absolute maximum radii for the fixed center positions, respecting all pairwise and boundary constraints. This ensures the sum is maximized for the achieved configuration.

import numpy as np

def generate_hexagonal_centers(n):
    """
    Generates n centers in a hexagonal lattice pattern inside a unit square.
    """
    centers = []
    # Parameters for hexagonal packing
    # We want to fit n circles. Let's estimate a side length.
    # Area of square = 1. Density of hex packing ~ 0.9069.
    # n * pi * r^2 ~ 0.9069 => r ~ sqrt(0.9069 / (n*pi))
    # But we need spacing s = 2r.
    # Let's just try to fit them in a grid-like structure.
    
    # Heuristic: Try to fit in a roughly square shape.
    # Number of columns approx sqrt(n)
    cols = int(math.ceil(math.sqrt(n)))
    rows = int(math.ceil(n / cols))
    
    # Spacing
    # If we have 'cols' circles in a row, width used is (cols-1)*s + 2r = (cols-1)*2r + 2r = 2*cols*r?
    # Actually, if centers are spaced by s=2r, width from first to last center is (cols-1)*s.
    # Plus radius on both sides: (cols-1)*2r + 2r = 2*cols*r <= 1 => r <= 1/(2*cols).
    # But hexagonal packing has vertical spacing s*sqrt(3)/2 = r*sqrt(3).
    # Height: (rows-1)*r*sqrt(3) + 2r <= 1.
    
    # Let's pick a target r and adjust.
    # Estimate r
    r_est = 1.0 / (2.0 * max(cols, rows * math.sqrt(3)/2 + 1)) * 0.8 # loose estimate
    # Better estimate:
    # Width constraint: 2*r*cols <= 1 => r <= 1/(2*cols)
    # Height constraint: r + (rows-1)*r*sqrt(3) + r <= 1 => r(2 + (rows-1)sqrt(3)) <= 1
    denom_w = 2.0 * cols
    denom_h = 2.0 + (rows - 1) * math.sqrt(3)
    r_start = min(1.0/denom_w, 1.0/denom_h) * 0.5 # Start small

            x += s
            row_count += 1
            count += 1
        
        y += h
        row_idx += 1

        if y > 1.0:

             pass

    # Ensure we have exactly n centers. If we generated fewer due to bounds, add some random ones?
    # Or better, just return what we have and let the optimizer work.

    while len(centers) < n:

    return np.array(centers[:n])

def resolve_conflicts(centers, radii, steps=100):
    """
    Iteratively pushes circles apart to resolve overlaps and boundary violations.
    """
    n = len(centers)
    for _ in range(steps):
        moved = False
        forces = np.zeros_like(centers)
        
        # Pairwise repulsion
        for i in range(n):
            for j in range(i + 1, n):
                c1 = centers[i]
                c2 = centers[j]
                r1 = radii[i]
                r2 = radii[j]
                
                diff = c1 - c2
                dist = np.linalg.norm(diff)
                
                if dist < 1e-12:
                    dist = 1e-12
                    diff = np.array([np.random.rand()-0.5, np.random.rand()-0.5])
                
                if dist < r1 + r2:
                    # Overlap
                    overlap = r1 + r2 - dist
                    # Normalize direction
                    dir_vec = diff / dist
                    # Push apart proportional to radii or equally? Equally is fine.
                    push = overlap / 2.0
                    forces[i] += dir_vec * push
                    forces[j] -= dir_vec * push
                    moved = True
        
        # Boundary repulsion
        for i in range(n):
            c = centers[i]
            r = radii[i]
            
            # Left wall
            if c[0] < r:
                forces[i][0] += (r - c[0])
                moved = True
            # Right wall
            if c[0] > 1 - r:
                forces[i][0] -= (c[0] - (1 - r))
                moved = True
            # Bottom wall
            if c[1] < r:
                forces[i][1] += (r - c[1])
                moved = True
            # Top wall
            if c[1] > 1 - r:
                forces[i][1] -= (c[1] - (1 - r))
                moved = True
        
        centers += forces
        
        if not moved:
            break
            
    return centers

def get_max_radii_lp(centers):
    """
    Given fixed centers, solve LP to maximize sum of radii.
    """
    n = centers.shape[0]
    
    # Variables: r_0 ... r_{n-1}
    # Objective: Maximize sum(r_i) => Minimize -sum(r_i)
    c_obj = np.ones(n) * -1
    
    # Constraints:
    # 1. Boundary constraints:
    #    r_i <= x_i
    #    r_i <= 1 - x_i
    #    r_i <= y_i
    #    r_i <= 1 - y_i
    #    r_i >= 0 (handled by bounds)
    
    # 2. Pairwise constraints:
    #    r_i + r_j <= dist(i, j)
    
    A_ub = []
    b_ub = []
    
    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        
        # r_i <= x
        row = np.zeros(n)
        row[i] = 1.0
        A_ub.append(row)
        b_ub.append(x)
        
        # r_i <= 1 - x
        row = np.zeros(n)
        row[i] = 1.0
        A_ub.append(row)
        b_ub.append(1.0 - x)
        
        # r_i <= y
        row = np.zeros(n)
        row[i] = 1.0
        A_ub.append(row)
        b_ub.append(y)
        
        # r_i <= 1 - y
        row = np.zeros(n)
        row[i] = 1.0
        A_ub.append(row)
        b_ub.append(1.0 - y)
        
    # Pairwise constraints
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dist)
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    bounds = [(0, None) for _ in range(n)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return -res.fun, -res.x
    else:
        # Fallback if LP fails (shouldn't happen with valid centers)
        return 0.0, np.zeros(n)
