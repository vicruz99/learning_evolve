def solve_radii_lp(centers):
    """
    Given fixed centers, solve for radii that maximize sum of radii.
    Constraints:
    r_i >= 0
    r_i <= x_i
    r_i <= 1 - x_i
    r_i <= y_i
    r_i <= 1 - y_i
    r_i + r_j <= dist(centers[i], centers[j])
    """
    n = centers.shape[0]
    
    # Objective: maximize sum(r_i) => minimize -sum(r_i)
    c = -np.ones(n)
    
    # Inequality constraints: A_ub * r <= b_ub
    # We will collect all constraints in a list and stack them
    
    A_ub_list = []
    b_ub_list = []
    
    # 1. Wall constraints
    # r_i <= x_i  => 1*r_i <= x_i
    # r_i <= 1-x_i => 1*r_i <= 1-x_i
    # r_i <= y_i
    # r_i <= 1-y_i
    
    for i in range(n):
        x, y = centers[i]
        
        # r_i <= x
        row = np.zeros(n)
        row[i] = 1.0
        A_ub_list.append(row)
        b_ub_list.append(x)
        
        # r_i <= 1 - x
        A_ub_list.append(row)
        b_ub_list.append(1.0 - x)
        
        # r_i <= y
        row_y = np.zeros(n)
        row_y[i] = 1.0
        A_ub_list.append(row_y)
        b_ub_list.append(y)
        
        # r_i <= 1 - y
        A_ub_list.append(row_y)
        b_ub_list.append(1.0 - y)
        
    # 2. Pairwise constraints
    # r_i + r_j <= d_ij
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub_list.append(row)
            b_ub_list.append(dist)
            
    A_ub = np.array(A_ub_list)
    b_ub = np.array(b_ub_list)
    
    # Bounds for r_i: r_i >= 0
    bounds = [(0, None) for _ in range(n)]
    
    try:
        res = opt.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            radii = res.x
            return radii, np.sum(radii)
        else:
            # Fallback if LP fails (should not happen with valid centers)
            # Return minimal valid radii (0) or handle error
            return np.zeros(n), 0.0
    except Exception:
        return np.zeros(n), 0.0
