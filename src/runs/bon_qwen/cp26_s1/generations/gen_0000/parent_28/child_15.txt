def run_packing():

    centers = np.zeros((n, 2))

    row_counts = [6, 5, 6, 5, 4]
    idx = 0
    for i, count in enumerate(row_counts):
        y = 0.1 + i * 0.2
        # Center each row horizontally
        start_x = 0.5 - (count - 1) * 0.09
        # Shift odd rows by half spacing for hexagonal alignment
        if i % 2 == 1:
            start_x += 0.09
        for j in range(count):
            centers[idx] = [start_x + j * 0.18, y]
            idx += 1
            
    r = 0.08
    alpha = 0.04
    dr = 0.0002
    
    # Force-directed optimization loop
    for step in range(60000):
        forces = np.zeros_like(centers)
        max_violation = 0.0
        
        # Pairwise repulsion
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                d = np.hypot(dx, dy)
                if d < 2 * r:
                    viol = 2 * r - d
                    if viol > max_violation:
                        max_violation = viol
                    f = viol * 8.0
                    fx = f * dx / (d + 1e-9)
                    fy = f * dy / (d + 1e-9)
                    forces[i] += [fx, fy]
                    forces[j] -= [fx, fy]
                    
        # Boundary repulsion
        for i in range(n):
            for dim in range(2):
                if centers[i, dim] < r:
                    viol = r - centers[i, dim]
                    if viol > max_violation:
                        max_violation = viol
                    forces[i, dim] += viol * 8.0

                    viol = centers[i, dim] - (1 - r)
                    if viol > max_violation:
                        max_violation = viol
                    forces[i, dim] -= viol * 8.0
                    
        # Update positions
        centers += alpha * forces
        centers = np.clip(centers, r, 1 - r)
        
        # Expand radius if system is stable
        if max_violation < 1e-7:
            r += dr
            alpha = max(0.005, alpha * 0.97)
        else:
            alpha *= 0.999

    min_pair = 1.0

    radii = np.full(n, r_final)
