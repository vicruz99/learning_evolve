import numpy as np
from scipy.optimize import minimize

def run_packing():

    Packs 26 circles in a unit square to maximize the sum of radii.
    
    Returns:
        centers: np.array of shape (26, 2) with (x, y) coordinates
        radii: np.array of shape (26) with radius of each circle
        sum_radii: float, the sum of all radii
    """
    n_circles = 26

    # We arrange circles in rows. A hexagonal pattern suggests alternating row lengths.
    # For 26 circles, a 5-row configuration like [6, 5, 6, 5, 4] sums to 26.
    # This mimics the densest packing.
    
    rows_counts = [6, 5, 6, 5, 4]
    
    # Estimate an initial radius. 
    # In a hexagonal packing, vertical spacing is sqrt(3)*r.
    # Height constraint: r + (num_rows - 1)*sqrt(3)*r + r <= 1
    # r * (2 + 4*sqrt(3)) <= 1  => r <= 1 / (2 + 4*1.732) approx 1/8.928 approx 0.112
    # Width constraint for row with 6 circles: 6 circles span 5 gaps of 2r + 2r (diameters at ends?)
    # Actually centers span (count-1)*2r. Plus radius margins.
    # (6-1)*2r + 2r = 12r <= 1 => r <= 0.0833.
    # So width is the bottleneck for the 6-circle rows.

    r_init = 0.04 
    
    centers_init = []
    y_current = r_init

    dy = np.sqrt(3) * r_init * 1.01 # Slightly larger to avoid initial overlap
    
    for count in rows_counts:
        # Determine x-start for this row
        # If it's a shifted row (odd index in 0-based list, i.e., 2nd, 4th), shift by r
        # However, our counts are [6, 5, 6, 5, 4].

        # But we have variable counts. Let's try to align them nicely.
        # If row has 6 circles, it's "full". If 5, it might fit in gaps.
        # Let's assume standard shift pattern relative to a theoretical infinite grid.
        
        is_shifted = False
        # Determine shift based on row index to maintain hexagonal structure
        # Row 0: 6 circles. Unshifted (starts at r)
        # Row 1: 5 circles. Shifted (starts at 2r? or centered?)
        # Let's just distribute them evenly in the row for the initial guess.
        
        # To approximate hex packing, shifted rows should be offset by r (diameter/2).

            # Even row index (0, 2, 4): Unshifted relative to left wall?

            x_start = r_init
        else:
            # Odd row index (1, 3): Shifted
            # Start at 2r (so it sits in the valley of previous row)
            x_start = 2 * r_init
            
        # Spacing between centers in a row is 2r
        dx = 2 * r_init
        
        x_coords = []
        for k in range(count):
            x_coords.append(x_start + k * dx)
            
        for x in x_coords:
            centers_init.append([x, y_current])
            
        y_current += dy

    centers_init = np.array(centers_init)
    radii_init = np.full(n_circles, r_init)
    
    # 2. Optimization Setup
    # Variables: [x1, y1, r1, x2, y2, r2, ..., x26, y26, r26]
    # Total 78 variables.
    
    x0 = np.zeros(3 * n_circles)
    for i in range(n_circles):
        x0[3*i] = centers_init[i, 0]
        x0[3*i+1] = centers_init[i, 1]
        x0[3*i+2] = radii_init[i]

        idx_r = 3*i + 2

        constraints.append({

            'fun': lambda v, i=i: v[idx_x] - v[idx_r]
        })
        # 1 - x - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: 1.0 - v[idx_x] - v[idx_r]
        })
        # y >= r
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: v[idx_y] - v[idx_r]
        })

        constraints.append({
            'type': 'ineq',

            idx_y2 = 3*j + 1

                dist_sq = (x1 - x2)**2 + (y1 - y2)**2

            })

        return -total_r

            final_centers = centers_init
            final_radii = radii_init

        final_centers = centers_init
        final_radii = radii_init

    sum_radii = np.sum(final_radii)
    
    return final_centers, final_radii, sum_radii
