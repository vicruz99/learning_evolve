import numpy as np
from scipy.optimize import minimize

# Top-level helper functions for constraints and objective

def objective(vars):
    """
    Objective function: Maximize sum of radii.
    Since minimize seeks to minimize, we return negative sum of radii.
    Radii are stored at indices 2, 5, 8, ... (3*i + 2).
    """
    return -np.sum(vars[2::3])

def boundary_x_ge_r(v, i):
    """Constraint: x_i >= r_i"""
    return v[3*i] - v[3*i+2]

def boundary_x_le_1_minus_r(v, i):
    """Constraint: x_i <= 1 - r_i"""
    return 1.0 - v[3*i] - v[3*i+2]

def boundary_y_ge_r(v, i):
    """Constraint: y_i >= r_i"""
    return v[3*i+1] - v[3*i+2]

def boundary_y_le_1_minus_r(v, i):
    """Constraint: y_i <= 1 - r_i"""
    return 1.0 - v[3*i+1] - v[3*i+2]

def non_overlap(v, i, j):
    """Constraint: Distance between centers >= sum of radii"""
    xi, yi, ri = v[3*i], v[3*i+1], v[3*i+2]
    xj, yj, rj = v[3*j], v[3*j+1], v[3*j+2]
    # Squared distance - squared sum of radii >= 0
    return (xi - xj)**2 + (yi - yj)**2 - (ri + rj)**2

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # 1. Initialization: Hexagonal lattice pattern
    # Starting with a small radius ensures a feasible initial point.
    r_init = 0.05 
    centers = []
    y = r_init
    row_idx = 0
    circles_placed = 0
    
    # Generate points in a hexagonal pattern to approximate dense packing
    while circles_placed < n:
        # Shift for hexagonal packing: even rows start at x=r, odd rows shifted by r
        if row_idx % 2 == 0:
            x_start = r_init
        else:
            x_start = 2 * r_init
            
        # Check vertical boundary
        if y + r_init > 1.0:
            break
            
        # Calculate max circles in this row based on horizontal width
        width_available = 1.0 - r_init - x_start
        if width_available < 0:
            max_k = 0
        else:
            max_k = int(width_available / (2 * r_init)) + 1
        
        needed = n - circles_placed
        k = min(max_k, needed)
        
        for i in range(k):
            x = x_start + i * (2 * r_init)
            centers.append([x, y])
            circles_placed += 1
            
        y += np.sqrt(3) * r_init
        row_idx += 1

    centers = np.array(centers)
    
    # Fallback initialization if hex grid didn't fit 26 circles
    if len(centers) < n:
        grid_x = np.linspace(0.1, 0.9, 6)
        grid_y = np.linspace(0.1, 0.9, 5)
        pts = []
        for gy in grid_y:
            for gx in grid_x:
                if len(pts) < n:
                    pts.append([gx, gy])
        centers = np.array(pts[:n])
        
    radii = np.full(n, r_init)
    
    # 2. Optimization Setup
    # Variables vector: [x1, y1, r1, x2, y2, r2, ...]
    x0 = []
    for i in range(n):
        x0.append(centers[i, 0])
        x0.append(centers[i, 1])
        x0.append(radii[i])
    x0 = np.array(x0)
    
    constraints = []
    
    # Add boundary constraints for each circle
    for i in range(n):
        constraints.append({'type': 'ineq', 'fun': boundary_x_ge_r, 'args': (i,)})
        constraints.append({'type': 'ineq', 'fun': boundary_x_le_1_minus_r, 'args': (i,)})
        constraints.append({'type': 'ineq', 'fun': boundary_y_ge_r, 'args': (i,)})
        constraints.append({'type': 'ineq', 'fun': boundary_y_le_1_minus_r, 'args': (i,)})
        
    # Add pairwise non-overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({'type': 'ineq', 'fun': non_overlap, 'args': (i, j)})
            
    # Bounds for variables: x, y in [0, 1], r in [0, 0.5]
    bounds = []
    for _ in range(n):
        bounds.append((0.0, 1.0))
        bounds.append((0.0, 1.0))
        bounds.append((0.0, 0.5))

    # 3. Run Optimization
    try:
        result = minimize(
            objective, 
            x0, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=constraints, 
            options={'maxiter': 5000, 'ftol': 1e-12}
        )
        
        if result.success:
            x_opt = result.x
            final_centers = np.array([[x_opt[3*i], x_opt[3*i+1]] for i in range(n)])
            final_radii = np.array([x_opt[3*i+2] for i in range(n)])
            final_radii = np.maximum(final_radii, 0.0) # Numerical safety
            return final_centers, final_radii, float(np.sum(final_radii))
    except Exception:
        pass

    # Fallback return if optimization fails
    return centers, radii, float(np.sum(radii))
