import numpy as np
from scipy.optimize import linprog, differential_evolution, minimize
import math

def compute_lp_bounds(centers):
    """
    Computes the maximum sum of radii for a given set of centers
    by solving a Linear Programming problem.
    """
    n = centers.shape[0]
    
    # Compute pairwise distances
    # dists[i, j] = Euclidean distance between center i and center j
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.linalg.norm(diff, axis=2)
    
    # We need to solve:
    # Maximize sum(r_i)
    # Subject to:
    # r_i + r_j <= dists[i, j] for all i < j
    # r_i <= min(x_i, 1-x_i, y_i, 1-y_i)
    # r_i >= 0
    
    # Objective: Maximize sum(r) => Minimize -sum(r)
    c = -np.ones(n)
    
    # Inequality constraints: A_ub @ r <= b_ub
    # 1. Pairwise constraints: r_i + r_j <= d_ij
    pairs_i = []
    pairs_j = []
    for i in range(n):
        for j in range(i + 1, n):
            pairs_i.append(i)
            pairs_j.append(j)
    
    num_pairs = len(pairs_i)
    A_pair = np.zeros((num_pairs, n))
    for k in range(num_pairs):
        A_pair[k, pairs_i[k]] = 1
        A_pair[k, pairs_j[k]] = 1
    
    b_pair = dists[pairs_i, pairs_j]
    
    # 2. Boundary constraints: r_i <= dist_to_boundary
    # dist_to_boundary = min(x, 1-x, y, 1-y)
    x, y = centers[:, 0], centers[:, 1]
    boundary_dists = np.minimum(np.minimum(x, 1 - x), np.minimum(y, 1 - y))
    
    # A_bound is identity matrix (since r_i <= b_i)
    # Actually, we can just add bounds in linprog or include in A_ub
    # linprog supports bounds directly.
    
    bounds = [(0, None) for _ in range(n)]
    
    # Combine constraints
    A_ub = A_pair
    b_ub = b_pair
    
    # Solve LP
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return -res.fun, res.x
        else:
            # Fallback or error handling
            return 0, np.zeros(n)
    except Exception:
        return 0, np.zeros(n)

def objective_function(coords_flat):
    """
    Objective function for the optimizer.
    Minimizes the negative sum of radii.
    """
    n = 26
    centers = coords_flat.reshape(n, 2)
    # Ensure centers are within [0, 1] (bounds handle this, but good for safety)
    # However, linprog handles boundary constraints via r_i <= dist_to_boundary.
    # If a center is outside [0,1], dist_to_boundary might be negative, 
    # making r_i <= negative number => r_i = 0 (since r_i >= 0).
    # So the objective would be low, penalizing outside positions.
    
    sum_r, _ = compute_lp_bounds(centers)
    return -sum_r
