import numpy as np
from scipy.optimize import minimize
import math

# Global constants
N_CIRCLES = 26

def calculate_objective(vars_flat):
    """
    Calculates the loss function for the packing problem.
    Loss = -Sum(Radii) + Penalty(Overlaps) + Penalty(Boundaries)
    """
    centers = vars_flat[:2 * N_CIRCLES].reshape(N_CIRCLES, 2)
    radii = vars_flat[2 * N_CIRCLES:]
    
    sum_radii = np.sum(radii)
    loss = -sum_radii
    
    # 1. Boundary Penalty
    # r <= x, r <= 1-x, r <= y, r <= 1-y
    # Violation = max(0, r - dist_to_boundary)
    
    # x bounds
    viol_x_left = np.maximum(0, radii - centers[:, 0])
    viol_x_right = np.maximum(0, radii - (1.0 - centers[:, 0]))
    
    # y bounds
    viol_y_bottom = np.maximum(0, radii - centers[:, 1])
    viol_y_top = np.maximum(0, radii - (1.0 - centers[:, 1]))
    
    boundary_penalty = np.sum(viol_x_left**2) + np.sum(viol_x_right**2) + \
                       np.sum(viol_y_bottom**2) + np.sum(viol_y_top**2)
    
    # 2. Overlap Penalty
    # dist(i, j) >= r_i + r_j
    # Violation = max(0, r_i + r_j - dist(i, j))
    
    c1 = centers[:, np.newaxis, :]
    c2 = centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum((c1 - c2)**2, axis=2))
    
    r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    
    # Upper triangle indices for i < j
    triu_indices = np.triu_indices(N_CIRCLES, k=1)
    viol = r_sum[triu_indices] - dists[triu_indices]
    viol = np.maximum(0, viol)
    
    overlap_penalty = np.sum(viol**2)
    
    # Penalty weight
    penalty_weight = 1000.0
    loss += penalty_weight * (boundary_penalty + overlap_penalty)
    
    return loss

def run_optimization(initial_centers, initial_radii):
    """
    Runs L-BFGS-B optimization starting from given initial state.
    """
    vars_init = np.concatenate([initial_centers.flatten(), initial_radii])
    
    bounds = []
    for _ in range(N_CIRCLES):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 0.5)) # r

    options = {
        'maxiter': 1500,
        'ftol': 1e-9,
        'gtol': 1e-8
    }
    
    try:
        res = minimize(
            calculate_objective,
            vars_init,
            method='L-BFGS-B',
            bounds=bounds,
            options=options
        )
        return res.x
    except Exception:
        return None

            current_x += dx
