import numpy as np
from scipy.optimize import linprog

def solve_lp(centers):
    """
    Solve Linear Program to maximize sum of radii for fixed centers.
    Constraints: r_i + r_j <= dist(i,j), r_i <= dist_to_boundary(i), r_i >= 0
    """
    n = centers.shape[0]
    c = -np.ones(n)  # Maximize sum(r) -> Minimize -sum(r)
    A_ub = []
    b_ub = []
    
    # Pairwise distances
    dists = np.sqrt(np.sum((centers[:, None, :] - centers[None, :, :])**2, axis=2))
    np.fill_diagonal(dists, 0.0)
    
    # Non-overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])
            
    # Boundary constraints
    bounds_val = np.min(np.stack([centers, 1.0 - centers], axis=1), axis=1)
    for i in range(n):
        row = np.zeros(n)
        row[i] = 1.0
        A_ub.append(row)
        b_ub.append(bounds_val[i])
        
    res = linprog(c, A_ub=np.array(A_ub), b_ub=np.array(b_ub), bounds=[(0, None)] * n, method='highs')
    if res.success:
        return -res.fun, res.x
    return 0.0, np.zeros(n)
