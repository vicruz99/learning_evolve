import numpy as np
from scipy.optimize import minimize

def compute_overlap(x, radii):
    """
    Computes the sum of squared overlaps for a given set of centers and radii.
    x: 1D array of shape (2*n) representing flattened centers.
    radii: 1D array of shape (n).
    """
    n = len(radii)
    centers = x.reshape(n, 2)
    
    # Vectorized pairwise distance calculation
    # centers shape: (n, 2)
    # c1: (n, 1, 2), c2: (1, n, 2)
    c1 = centers[:, np.newaxis, :]
    c2 = centers[np.newaxis, :, :]
    dist = np.sqrt(np.sum((c1 - c2)**2, axis=2))
    
    # Pairwise sum of radii
    r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    
    # Overlaps are positive when r_i + r_j > dist
    overlaps = r_sum - dist
    
    # Ignore diagonal (self-distance)
    np.fill_diagonal(overlaps, 0.0)
    
    # Only positive values contribute to penalty
    overlaps = np.maximum(0.0, overlaps)
    return np.sum(overlaps**2)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Try multiple random initializations to avoid poor local minima
    for seed in range(5):
        np.random.seed(seed)
        # Initialize centers randomly within [0.1, 0.9]
        centers = np.random.rand(n, 2) * 0.8 + 0.1
        radii = np.full(n, 0.02)
        
        # Iteratively grow radii
        step = 0.0002
        for _ in range(6000):
            radii += step
            
            # Bounds for L-BFGS-B: each coordinate must be in [r, 1-r]
            bounds = [(r, 1.0 - r) for r in radii for _ in range(2)]
            x0 = centers.flatten()
            
            # Minimize overlap for fixed radii
            res = minimize(
                compute_overlap, 
                x0, 
                args=(radii,), 
                method='L-BFGS-B', 
                bounds=bounds, 
                options={'maxiter': 80, 'ftol': 1e-14}
            )
            centers = res.x.reshape(n, 2)
            
            # Check if configuration is valid (overlap near zero)
            if compute_overlap(centers.flatten(), radii) > 1e-6:
                radii -= step  # Backtrack
                break
                
        curr_sum = np.sum(radii)
        if curr_sum > best_sum:
            best_sum = curr_sum
            best_centers = centers.copy()
            best_radii = radii.copy()
            
        # Early exit if we significantly beat the target
        if best_sum > 2.68:
            break
            
    # Final polishing step to ensure strict validity and optimal arrangement
    radii = best_radii.copy()
    bounds = [(r, 1.0 - r) for r in radii for _ in range(2)]
    x0 = best_centers.flatten()
    res = minimize(
        compute_overlap, 
        x0, 
        args=(radii,), 
        method='L-BFGS-B', 
        bounds=bounds, 
        options={'maxiter': 500, 'ftol': 1e-15}
    )
    best_centers = res.x.reshape(n, 2)
    
    return best_centers, best_radii, np.sum(best_radii)
