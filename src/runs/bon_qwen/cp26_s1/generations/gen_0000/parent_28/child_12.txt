import numpy as np
from scipy.optimize import minimize

N = 26

def compute_constraints(vars):
    """Compute pairwise non-overlap constraints: dist^2 - (r1+r2)^2 >= 0"""
    n = N
    u = vars[:n]
    v = vars[n:2*n]
    r = vars[2*n:]
    
    # Transform to actual coordinates (automatically satisfies boundary constraints)
    x = r + u * (1.0 - 2.0 * r)
    y = r + v * (1.0 - 2.0 * r)
    
    con = np.zeros(n * (n - 1) // 2)
    idx = 0
    for i in range(n):
        xi, yi, ri = x[i], y[i], r[i]
        for j in range(i + 1, n):
            dx = xi - x[j]
            dy = yi - y[j]
            con[idx] = dx * dx + dy * dy - (ri + r[j]) ** 2
            idx += 1
    return con

def compute_objective(vars):
    """Negative sum of radii (to be minimized)"""
    return -np.sum(vars[2*N:])

def run_packing():
    n = N
    r_init = 0.085  # Initial radius guess
    
    # Generate initial hexagonal lattice configuration
    centers = []
    row = 0
    while len(centers) < n:
        y = r_init + row * np.sqrt(3) * r_init
        x_start = r_init if row % 2 == 0 else r_init * 2.0
        for c in range(7):
            if len(centers) >= n:
                break

                centers.append([x, y])
        row += 1
    centers = np.array(centers[:n])
    
    # Convert to normalized u,v parameters
    denom = 1.0 - 2.0 * r_init
    u_init = (centers[:, 0] - r_init) / denom
    v_init = (centers[:, 1] - r_init) / denom

    bounds = [(0.0, 1.0) for _ in range(2 * n)] + [(0.01, 0.5) for _ in range(n)]
    cons = [{'type': 'ineq', 'fun': compute_constraints}]
    
    # Optimize
    res = minimize(
        compute_objective, 
        x0, 
        method='SLSQP', 
        bounds=bounds,

    y_opt = r_opt + v_opt * (1.0 - 2.0 * r_opt)
