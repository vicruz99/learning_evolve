import numpy as np
from scipy.optimize import minimize

def run_packing():

    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    
    Returns:
        tuple: (centers, radii, sum_radii)

    def objective(vars):
        radii = vars[2::3]
        return -np.sum(radii)

    def get_geometry(vars):
        centers = np.zeros((n_circles, 2))
        radii = np.zeros(n_circles)
        for i in range(n_circles):
            centers[i, 0] = vars[3 * i]
            centers[i, 1] = vars[3 * i + 1]
            radii[i] = vars[3 * i + 2]
        return centers, radii

    # Generate constraints
    # SLSQP expects constraints in format {'type': 'ineq', 'fun': func}
    # where fun(vars) >= 0
    
    def generate_constraints():
        constraints = []
        
        # Boundary constraints for each circle
        # x >= r  => x - r >= 0
        # x <= 1-r => 1 - x - r >= 0
        # y >= r  => y - r >= 0
        # y <= 1-r => 1 - y - r >= 0
        # r >= 0   => r >= 0
        
        for i in range(n_circles):
            idx_x = 3 * i
            idx_y = 3 * i + 1
            idx_r = 3 * i + 2
            
            # x - r >= 0
            constraints.append({
                'type': 'ineq',
                'fun': lambda v, i=i: v[idx_x] - v[idx_r]
            })
            # 1 - x - r >= 0
            constraints.append({
                'type': 'ineq',
                'fun': lambda v, i=i: 1.0 - v[idx_x] - v[idx_r]
            })
            # y - r >= 0
            constraints.append({
                'type': 'ineq',
                'fun': lambda v, i=i: v[idx_y] - v[idx_r]
            })
            # 1 - y - r >= 0
            constraints.append({
                'type': 'ineq',
                'fun': lambda v, i=i: 1.0 - v[idx_y] - v[idx_r]
            })
            # r >= 0 (though handled by bounds, added for safety in formulation logic)
            constraints.append({
                'type': 'ineq',
                'fun': lambda v, i=i: v[idx_r]
            })

                idx_i_r = 3 * i + 2

                idx_j_y = 3 * j + 1
                idx_j_r = 3 * j + 2

                    yi = v[idx_i_y]
                    ri = v[idx_i_r]
                    xj = v[idx_j_x]
                    yj = v[idx_j_y]
                    rj = v[idx_j_r]
                    
                    dist_sq = (xi - xj)**2 + (yi - yj)**2
                    sum_r = ri + rj
                    return dist_sq - sum_r**2

                constraints.append({
                    'type': 'ineq',
                    'fun': circle_constraint
                })
        
        return constraints

        try:
