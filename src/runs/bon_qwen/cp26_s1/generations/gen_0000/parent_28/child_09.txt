import numpy as np
from scipy.optimize import minimize

# Constants
N_CIRCLES = 26
SQUARE_SIZE = 1.0
NUM_RESTARTS = 15
PERTURBATION_FACTOR = 0.05
INITIAL_RADIUS = 0.09

def compute_constraints(vars):
    """
    Computes the constraint functions for the optimizer.
    vars: [x_0, y_0, r_0, x_1, y_1, r_1, ..., x_N, y_N, r_N]
    """
    centers = vars[:N_CIRCLES * 2].reshape((N_CIRCLES, 2))
    radii = vars[N_CIRCLES * 2:]
    
    constraints = []
    
    # 1. Boundary Constraints
    # x - r >= 0  => x >= r
    # 1 - x - r >= 0 => x <= 1 - r
    for i in range(N_CIRCLES):
        x, y = centers[i]
        r = radii[i]
        # Boundary lower
        constraints.append(x - r)
        # Boundary upper
        constraints.append((1.0 - x) - r)
        # Boundary lower Y
        constraints.append(y - r)
        # Boundary upper Y
        constraints.append((1.0 - y) - r)

    # 2. Non-overlap Constraints
    # dist(i, j) >= r_i + r_j  => dist - (r_i + r_j) >= 0
    for i in range(N_CIRCLES):
        for j in range(i + 1, N_CIRCLES):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx*dx + dy*dy)
            constraints.append(dist - (radii[i] + radii[j]))
            
    return np.array(constraints)

def get_jacobian(vars):
    """
    Computes the Jacobian of the constraint functions.
    Essential for SLSQP performance.
    Shape: (num_constraints, 3 * N_CIRCLES)
    """
    n_vars = 3 * N_CIRCLES
    # We need to calculate number of constraints first to initialize matrix
    # Boundary: 4 per circle. Overlap: N*(N-1)/2
    n_constraints = 4 * N_CIRCLES + (N_CIRCLES * (N_CIRCLES - 1)) // 2
    J = np.zeros((n_constraints, n_vars))
    
    centers = vars[:N_CIRCLES * 2].reshape((N_CIRCLES, 2))
    radii = vars[N_CIRCLES * 2:]
    
    idx = 0
    
    # 1. Boundary Constraints Jacobian
    for i in range(N_CIRCLES):
        x, y = centers[i]
        r = radii[i]
        
        # Variable indices: x -> 3*i, y -> 3*i+1, r -> 3*i+2
        
        # Constraint: x - r >= 0
        # d/dx = 1, d/dr = -1
        J[idx, 3*i] = 1.0
        J[idx, 3*i+2] = -1.0
        idx += 1
        
        # Constraint: (1-x) - r >= 0
        # d/dx = -1, d/dr = -1
        J[idx, 3*i] = -1.0
        J[idx, 3*i+2] = -1.0
        idx += 1
        
        # Constraint: y - r >= 0
        # d/dy = 1, d/dr = -1
        J[idx, 3*i+1] = 1.0
        J[idx, 3*i+2] = -1.0
        idx += 1
        
        # Constraint: (1-y) - r >= 0
        # d/dy = -1, d/dr = -1
        J[idx, 3*i+1] = -1.0
        J[idx, 3*i+2] = -1.0
        idx += 1

    # 2. Overlap Constraints Jacobian
    # Constraint: sqrt((xi-xj)^2 + (yi-yj)^2) - ri - rj >= 0
    for i in range(N_CIRCLES):
        for j in range(i + 1, N_CIRCLES):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx*dx + dy*dy)
            
            # Avoid division by zero if centers overlap perfectly
            if dist < 1e-9:
                dist = 1e-9
            
            # Partial derivatives of dist w.r.t coordinates
            # d(dist)/d(xi) = dx / dist
            # d(dist)/d(xj) = -dx / dist
            
            # For circle i
            J[idx, 3*i] = dx / dist
            J[idx, 3*i+1] = dy / dist
            J[idx, 3*i+2] = -1.0
            
            # For circle j
            J[idx, 3*j] = -dx / dist
            J[idx, 3*j+1] = -dy / dist
            J[idx, 3*j+2] = -1.0
            
            idx += 1
            
    return J

def run_packing():
    best_sum_radii = -1.0
    best_centers = None
    best_radii = None
    
    # Helper to initialize a hexagonal grid
    def init_hex_grid(perturbation=0.0):
        # Try to fit 26 circles. 
        # Hexagonal packing rows: 5, 5, 5, 5, 4, 2 is roughly 26? 
        # Let's just use a dense grid logic or random grid perturbation
        # A 6x5 grid gives 30 points, we pick 26.
        
        # Create a 6x5 grid points
        cols = 6
        rows = 5
        xs = np.linspace(0.1, 0.9, cols)
        ys = np.linspace(0.1, 0.9, rows)
        
        # Offset odd rows for hexagonal feel
        x_grid = []
        for r_idx, y in enumerate(ys):
            row_xs = list(xs)
            if r_idx % 2 == 1:
                # Shift x slightly
                row_xs = [x + 0.05 for x in xs]
                # Clip
                row_xs = [max(0.05, min(0.95, x)) for x in row_xs]
            x_grid.extend(row_xs)
            
        y_grid = []
        for y in ys:
            y_grid.extend([y] * cols)
            
        # We have 30 points, need 26. 
        # Just take the first 26
        selected_x = x_grid[:N_CIRCLES]
        selected_y = y_grid[:N_CIRCLES]
        
        centers = np.column_stack([selected_x, selected_y])
        radii = np.full(N_CIRCLES, INITIAL_RADIUS)
        
        # Add perturbation
        if perturbation > 0:
            centers += np.random.uniform(-perturbation, perturbation, centers.shape)
            # Clip to valid range for centers (loose clip, radii handle constraints)
            centers = np.clip(centers, 0.01, 0.99)
            
        return centers, radii

    for k in range(NUM_RESTARTS):
        # Perturbation amount increases with restart index to explore more
        p_factor = PERTURBATION_FACTOR * (1 + k * 0.2)
        centers_init, radii_init = init_hex_grid(p_factor)
        
        # Combine into single vector for optimizer
        # Order: x0, y0, r0, x1, y1, r1 ...
        x0 = np.concatenate([centers_init.flatten(), radii_init])
        
        # Bounds: x, y in [0, 1], r in [0, 0.5] (max possible radius)
        bounds = []
        for _ in range(N_CIRCLES):
            bounds.append((0.0, 1.0)) # x
            bounds.append((0.0, 1.0)) # y
            bounds.append((1e-5, 0.5)) # r (small positive lower bound to avoid degeneracy)

        # Objective: minimize -sum(radii)
        def objective(vars):
            radii_vec = vars[N_CIRCLES * 2:]
            return -np.sum(radii_vec)

        # Constraint function returning values that must be >= 0
        # We use a dictionary format for constraints in scipy
        # But SLSQP accepts a dict with 'type': 'ineq' and 'fun'
        # However, providing jacobian is efficient
        
        # Define constraint dict
        cons = {
            'type': 'ineq',
            'fun': compute_constraints,
            'jac': get_jacobian
        }

        # Run optimization
        try:
            res = minimize(
                objective, 
                x0, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 1000, 'ftol': 1e-9}
            )
            
            if res.success:
                vars_opt = res.x
                centers_opt = vars_opt[:N_CIRCLES * 2].reshape((N_CIRCLES, 2))
                radii_opt = vars_opt[N_CIRCLES * 2:]
                sum_r = np.sum(radii_opt)
                
                if sum_r > best_sum_radii:
                    best_sum_radii = sum_r
                    best_centers = centers_opt.copy()
                    best_radii = radii_opt.copy()
        except Exception as e:
            pass # Skip failed optimizations

    # Final validation check and clipping just in case
    if best_centers is not None:
        # Ensure strict validity for the return
        # Clip radii to be valid with respect to centers (though optimizer should have done this)
        # But numerical noise might leave tiny violations
        # We can do a final pass to shrink radii slightly if needed, 
        # but SLSQP usually respects constraints well.
        
        # Just return the best found
        return best_centers, best_radii, best_sum_radii
    else:
        # Fallback
        centers = np.random.rand(N_CIRCLES, 2) * 0.8 + 0.1
        radii = np.full(N_CIRCLES, 0.05)
        return centers, radii, np.sum(radii)
