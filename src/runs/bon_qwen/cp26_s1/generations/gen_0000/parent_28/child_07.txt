

Looking at this circle packing problem, I need to pack 26 circles in a unit square to maximize the sum of radii (target: 2.636).

<strategy>
My strategy combines multiple approaches:

1. **Initial Configuration**: Generate multiple initial packings using hexagonal grid patterns (the densest 2D packing) with different seeds and offsets to explore diverse starting points.

2. **Iterative Expansion Algorithm**: Gradually increase all radii by a small factor each iteration, then resolve conflicts using force-based repulsion. This "growing circles" approach naturally drives toward the boundary of feasibility.

3. **Force-Based Conflict Resolution**: When circles overlap, apply repulsive forces proportional to overlap amount. Boundary violations get strong repulsive forces too. The damping coefficient decreases over time for convergence.

4. **Local Optimization Pass**: After the expansion phase, use scipy's SLSQP optimizer with explicit constraints to fine-tune the configuration. This handles the non-convex optimization with proper constraint handling.

5. **Multiple Restarts**: Run the expansion from different initial configurations and keep the best result.

Key insights:
- For equal circles, n=26 in a unit square should give radius ≈0.102, sum ≈2.65
- Hexagonal packing is the densest arrangement
- The iterative expansion naturally finds good local optima
- Multiple restarts help escape poor local optima
</strategy>

import numpy as np
from scipy.optimize import minimize


def hexagonal_initial(n, r_init=0.05, seed=0):
    """Generate initial hexagonal packing."""
    np.random.seed(seed)
    centers = np.zeros((n, 2))
    radii = np.ones(n) * r_init
    
    idx = 0
    for row in range(10):
        for col in range(10):
            if idx >= n:
                break
            x = r_init + col * r_init * np.sqrt(3) + (row % 2) * r_init * np.sqrt(3) / 2
            y = r_init + row * r_init * 1.5
            if x - r_init >= 0 and x + r_init <= 1 and y - r_init >= 0 and y + r_init <= 1:
                centers[idx] = [x, y]
                idx += 1
    
    # Fill remaining randomly
    while idx < n:
        placed = False
        for _ in range(10000):
            x = np.random.uniform(r_init + 0.01, 1 - r_init - 0.01)
            y = np.random.uniform(r_init + 0.01, 1 - r_init - 0.01)
            ok = True
            for j in range(idx):
                dx = x - centers[j, 0]
                dy = y - centers[j, 1]
                if dx * dx + dy * dy < 4 * r_init * r_init:
                    ok = False
                    break
            if ok:
                centers[idx] = [x, y]
                idx += 1
                placed = True
                break
        if not placed:
            break
    
    return centers, radii


def expand_and_resolve(centers, radii, n, iterations=50000):
    """Iteratively expand radii and resolve conflicts."""
    centers = centers.copy()
    radii = radii.copy()
    
    for it in range(iterations):
        # Growth rate decreases over time
        growth = 1.0 + 5e-7 * (1.0 + 2.0 * np.exp(-it / 10000.0))
        
        # Increase all radii
        for i in range(n):
            radii[i] *= growth
        
        # Resolve conflicts with force-based approach
        forces = np.zeros((n, 2))
        
        for i in range(n):
            # Boundary repulsion
            for axis in range(2):
                if centers[i, axis] - radii[i] < 0:
                    forces[i, axis] += (radii[i] - centers[i, axis]) * 50
                if centers[i, axis] + radii[i] > 1:
                    forces[i, axis] -= (centers[i, axis] + radii[i] - 1) * 50
            
            # Circle-circle repulsion
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist_sq = dx * dx + dy * dy
                
                if dist_sq < 1e-20:
                    # Coincident circles - push in random direction
                    forces[i, 0] += np.random.uniform(-1, 1) * 0.01
                    forces[i, 1] += np.random.uniform(-1, 1) * 0.01
                    forces[j, 0] -= forces[i, 0]
                    forces[j, 1] -= forces[i, 1]
                    continue
                
                dist = np.sqrt(dist_sq)
                min_dist = radii[i] + radii[j]
                
                if dist < min_dist * 1.15:
                    if dist < min_dist:
                        overlap = min_dist - dist
                        force_mag = overlap * 80
                    else:
                        # Small repulsion for near-touching to maintain separation
                        force_mag = (min_dist * 1.15 - dist) * 3
                    
                    nx = dx / dist
                    ny = dy / dist
                    forces[i, 0] += nx * force_mag
                    forces[i, 1] += ny * force_mag
                    forces[j, 0] -= nx * force_mag
                    forces[j, 1] -= ny * force_mag
        
        # Apply forces with damping
        damping = 0.08 * (1.0 + 3.0 * np.exp(-it / 15000.0))
        damping = min(damping, 0.5)
        
        centers += forces * damping
        
        # Clamp to valid range
        for i in range(n):
            centers[i, 0] = max(radii[i] + 1e-12, min(1 - radii[i] - 1e-12, centers[i, 0]))
            centers[i, 1] = max(radii[i] + 1e-12, min(1 - radii[i] - 1e-12, centers[i, 1]))
        
        # Periodically reduce radii slightly if stuck in infeasible region
        if it % 2000 == 0 and it > 0:
            feasible = True
            for i in range(n):
                for j in range(i + 1, n):
                    dx = centers[i, 0] - centers[j, 0]
                    dy = centers[i, 1] - centers[j, 1]
                    dist = np.sqrt(dx * dx + dy * dy)
                    if dist < radii[i] + radii[j] - 1e-10:
                        feasible = False
                        break
                if not feasible:
                    break
            
            if not feasible:
                radii *= 0.995
    
    return centers, radii


def local_optimize(centers, radii, n):
    """Local optimization using scipy SLSQP."""
    
    def obj(params):
        r = params[2 * n:]
        return -np.sum(r)
    
    # Build constraints list
    constraints = []
    
    # Boundary constraints
    for i in range(n):
        def make_bc(p, idx=i, axis=0, sign=1):
            if sign > 0:
                return p[2 * idx + axis] - p[2 * n + idx]
            else:
                return 1 - p[2 * idx + axis] - p[2 * n + idx]
        
        constraints.append({'type': 'ineq', 'fun': lambda p, i=i, a=0: p[2*i] - p[2*n+i]})
        constraints.append({'type': 'ineq', 'fun': lambda p, i=i, a=0: 1 - p[2*i] - p[2*n+i]})
        constraints.append({'type': 'ineq', 'fun': lambda p, i=i, a=1: p[2*i+1] - p[2*n+i]})
        constraints.append({'type': 'ineq', 'fun': lambda p, i=i, a=1: 1 - p[2*i+1] - p[2*n+i]})
        constraints.append({'type': 'ineq', 'fun': lambda p, i=i: p[2*n+i]})
    
    # Overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            def make_overlap(p, ii=i, jj=j):
                dx = p[2*ii] - p[2*jj]
                dy = p[2*ii+1] - p[2*jj+1]
                ri = p[2*n+ii]
                rj = p[2*n+jj]
                return dx*dx + dy*dy - (ri + rj)**2
            constraints.append({'type': 'ineq', 'fun': make_overlap})
    
    params = np.concatenate([centers.flatten(), radii])
    bounds = [(0, 1)] * (2 * n) + [(0, 0.5)] * n
    
    try:
        result = minimize(obj, params, method='SLSQP', bounds=bounds,
                         constraints=constraints, options={'maxiter': 200, 'ftol': 1e-12})
        
        if result.fun < -np.sum(radii) + 1e-10:
            new_centers = result.x[:2*n].reshape(n, 2)
            new_radii = np.maximum(result.x[2*n:], 0)
            return new_centers, new_radii
    except Exception:
        pass
    
    return centers, radii


def run_packing():
    n = 26
    
    best_centers = None
    best_radii = None
    best_sum = -np.inf
    
    # Try multiple initial configurations
    configs = []
    
    # Hexagonal packings with different starting radii
    for r_init in [0.03, 0.04, 0.05, 0.06, 0.07]:
        for seed in range(3):
            c, r = hexagonal_initial(n, r_init=r_init, seed=seed)
            configs.append((c, r))
    
    # Random initial configurations
    for seed in range(5):
        np.random.seed(seed + 100)
        c = np.random.uniform(0.15, 0.85, (n, 2))
        r = np.ones(n) * 0.03
        configs.append((c, r))
    
    for c_init, r_init in configs:
        # Phase 1: Expand and resolve
        centers, radii = expand_and_resolve(c_init, r_init, n, iterations=40000)
        
        # Phase 2: Local optimization
        centers, radii = local_optimize(centers, radii, n)
        
        # Validate and record
        current_sum = np.sum(radii)
        
        # Check feasibility
        feasible = True
        for i in range(n):
            if radii[i] < 0:
                feasible = False
                break
            x, y = centers[i]
            if x - radii[i] < -1e-10 or x + radii[i] > 1 + 1e-10:
                feasible = False
                break
            if y - radii[i] < -1e-10 or y + radii[i] > 1 + 1e-10:
                feasible = False
                break
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii[i] + radii[j] - 1e-10:
                    feasible = False
                    break
            if not feasible:
                break
        
        if feasible and current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers.copy()
            best_radii = radii.copy()
    
    if best_centers is None:
        # Fallback
        best_centers, best_radii = hexagonal_initial(n, r_init=0.05, seed=0)
        best_sum = np.sum(best_radii)
    
    return best_centers, best_radii, best_sum
