import numpy as np

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    Uses a force-directed simulation starting from a hexagonal lattice.
    """
    n = 26
    # Initial configuration: Hexagonal packing in 5 rows
    # Rows: 5, 5, 5, 5, 6 circles (Total 26)
    # This arrangement is denser than a square grid and allows for a higher radius.
    
    centers = np.zeros((n, 2))
    idx = 0
    
    # Hexagonal parameters
    # We aim for an initial radius that fits, then grow it.
    # 5 rows with vertical spacing approx sqrt(3)/2 * diameter
    # Horizontal spacing approx diameter
    # Shift alternate rows by half diameter
    
    row_counts = [5, 5, 5, 5, 6]
    current_r = 0.09  # Start with a safe radius
    
    # Calculate vertical positions for rows to fit roughly in [0, 1]
    # We will center them later.
    y_step = current_r * np.sqrt(3)
    
    for r_idx, count in enumerate(row_counts):
        y = current_r + r_idx * y_step
        
        # Shift even/odd rows to create hexagonal structure
        # To fit better, we center the row horizontally
        # Total width for 'count' circles with diameter 2r is 2r + (count-1)*2r = 2r*count
        # We want this centered in [0, 1]
        # x_start = (1 - 2 * r * count) / 2 + r
        # But we just place them and let the optimizer center them
        
        x_step = 2 * current_r
        if r_idx % 2 == 1:
            x_offset = current_r + x_step / 2 # Shift by r
        else:
            x_offset = current_r
            
        for c_idx in range(count):
            x = x_offset + c_idx * x_step
            centers[idx] = [x, y]
            idx += 1
            
    # Radii array - initially equal
    radii = np.full(n, current_r)
    
    # Simulation parameters
    dt = 0.05
    repulsion_k = 50.0
    pressure = 0.0005 # Force to grow circles
    max_iter = 1500
    
    # Simulation loop
    for step in range(max_iter):
        forces = np.zeros_like(centers)
        
        # 1. Calculate repulsion forces between circles
        for i in range(n):
            for j in range(i + 1, n):
                vec = centers[i] - centers[j]
                dist = np.linalg.norm(vec)
                min_dist = radii[i] + radii[j]
                
                if dist < min_dist:
                    # Overlap detected
                    overlap = min_dist - dist
                    # Repulsive force proportional to overlap
                    if dist > 1e-9:
                        force_vec = (vec / dist) * repulsion_k * overlap
                        forces[i] += force_vec
                        forces[j] -= force_vec
                    else:
                        # Prevent division by zero, push randomly apart
                        forces[i] += np.random.normal(0, 1, 2) * 0.1
                        forces[j] -= np.random.normal(0, 1, 2) * 0.1

        # 2. Calculate boundary forces (keep circles inside [0,1])
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            
            # Left wall
            if x - r < 0:
                forces[i, 0] += repulsion_k * (r - x)
            # Right wall
            if x + r > 1:
                forces[i, 0] -= repulsion_k * (x + r - 1)
            # Bottom wall
            if y - r < 0:
                forces[i, 1] += repulsion_k * (r - y)
            # Top wall
            if y + r > 1:
                forces[i, 1] -= repulsion_k * (y + r - 1)

        # 3. Apply forces to move centers
        centers += dt * forces
        
        # 4. Clamp centers to valid range to prevent runaway
        # Centers must be in [r, 1-r]
        for i in range(n):
            r = radii[i]
            centers[i, 0] = np.clip(centers[i, 0], r, 1 - r)
            centers[i, 1] = np.clip(centers[i, 1], r, 1 - r)
            
        # 5. Grow radii (Pressure)
        # We try to increase radii, but check for validity roughly
        # If step is high enough, we can safely grow
        if step > 100: # Let it settle first
            radii += pressure
            
            # If radii grow too much causing massive overlaps, the repulsion will handle it.
            # However, if a circle gets stuck in a corner, it might not grow.
            # We rely on the repulsion to push it out to make space.
            
            # Cap radii to prevent explosion if logic fails
            radii = np.clip(radii, 0, 0.5)

    # Post-processing: Gradient ascent to fine-tune radii
    # We perform a few steps of "pushing boundaries"
    # This helps to squeeze out more radius where possible
    
    for _ in range(200):
        for i in range(n):
            # Try to increase radius of circle i
            current_r = radii[i]

            max_r = 1.0 # Upper bound
            
            # Distance to walls
            d_left = centers[i, 0]
            d_right = 1.0 - centers[i, 0]
            d_bottom = centers[i, 1]
            d_top = 1.0 - centers[i, 1]
            max_r = min(max_r, d_left, d_right, d_bottom, d_top)
            
            # Distance to other circles
            for j in range(n):
                if i == j: continue
                dist = np.linalg.norm(centers[i] - centers[j])
                # max_r + radii[j] <= dist => max_r <= dist - radii[j]

    # Re-verify constraints and fix minor violations

        x, y = centers[i]
        r = radii[i]
        # Clamp radius to boundary limits

        radii[i] = r
        
        # Clamp center to be safe

        centers[i, 1] = np.clip(centers[i, 1], radii[i], 1 - radii[i])

    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
