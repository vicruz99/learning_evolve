from scipy.optimize import minimize

def compute_penalty(params, n, lam):
    """Objective function: negative sum of radii + penalty for constraint violations."""
    x = params[:n]
    y = params[n:2*n]
    r = params[2*n:]
    penalty = 0.0
    
    for i in range(n):
        # Boundary penalties
        if x[i] < r[i]:
            penalty += lam * (r[i] - x[i])**2
        if x[i] > 1.0 - r[i]:
            penalty += lam * (r[i] - (1.0 - x[i]))**2
        if y[i] < r[i]:
            penalty += lam * (r[i] - y[i])**2
        if y[i] > 1.0 - r[i]:
            penalty += lam * (r[i] - (1.0 - y[i]))**2
            
        for j in range(i + 1, n):
            dx = x[i] - x[j]
            dy = y[i] - y[j]
            dist = np.hypot(dx, dy)
            overlap = r[i] + r[j] - dist
            if overlap > 0:
                penalty += lam * overlap**2
    return -np.sum(r) + penalty

def resolve_overlaps(centers, radii, n):

    for _ in range(30):

            centers[i, 0] = np.clip(centers[i, 0], r, 1.0 - r)
            centers[i, 1] = np.clip(centers[i, 1], r, 1.0 - r)

                dist = np.hypot(dx, dy)

                    nx, ny = dx / dist, dy / dist
                    centers[i, 0] += push * nx
                    centers[i, 1] += push * ny

                    centers[j, 1] -= push * ny
