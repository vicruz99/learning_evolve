import numpy as np
import scipy.optimize as opt

def get_initial_config():
    """
    Generates a valid initial configuration: 25 circles in a 5x5 grid (r=0.1)
    and 1 circle in the center gap.
    """
    centers = []
    radii = []

    # 5x5 Grid
    step = 0.2
    offset = 0.1
    for i in range(5):
        for j in range(5):
            x = offset + i * step
            y = offset + j * step
            centers.append([x, y])
            radii.append(0.1)

    # 26th circle in the center gap (e.g., at 0.2, 0.2)
    # Max radius approx 0.0414
    centers.append([0.2, 0.2])
    radii.append(0.041)

    centers = np.array(centers)
    radii = np.array(radii)
    return centers, radii

def objective(params, n_circles):
    """
    Objective function to minimize: -sum(radii) + penalties for constraint violations.
    """
    # Reshape parameters
    # params layout: [x1, y1, r1, x2, y2, r2, ..., xn, yn, rn]
    centers = params.reshape(-1, 3)[:, :2]
    radii = params.reshape(-1, 3)[:, 2]

    # Soft penalty factor
    penalty = 1000.0
    
    obj = -np.sum(radii)

    # Boundary penalties
    # Ensure x - r >= 0 => r - x <= 0
    # Ensure x + r <= 1 => x + r - 1 <= 0
    # Similarly for y
    
    # Violations
    v_x_low = np.maximum(0, radii - centers[:, 0])
    v_x_high = np.maximum(0, centers[:, 0] + radii - 1)
    v_y_low = np.maximum(0, radii - centers[:, 1])
    v_y_high = np.maximum(0, centers[:, 1] + radii - 1)
    
    boundary_viol = np.sum(v_x_low**2 + v_x_high**2 + v_y_low**2 + v_y_high**2)
    obj += penalty * boundary_viol

    # Overlap penalties
    # dist >= r1 + r2 => r1 + r2 - dist <= 0
    # We use a smooth approximation for distance to avoid sqrt issues, 
    # but for precision we can use direct distance with a small epsilon if needed.
    # Given n=26, O(N^2) is fine.
    
    for i in range(n_circles):
        for j in range(i + 1, n_circles):
            dist = np.sqrt((centers[i, 0] - centers[j, 0])**2 + 
                           (centers[i, 1] - centers[j, 1])**2)
            min_dist = radii[i] + radii[j]
            overlap = min_dist - dist
            if overlap > 1e-12:
                obj += penalty * overlap**2
                
    return obj

    n = 26
    centers, radii = get_initial_config()

    # Optimization
    # L-BFGS-B is efficient for bound-constrained problems
    result = opt.minimize(
        objective, 
        x0, 
        args=(n,), 
        method='L-BFGS-B', 
        bounds=bounds,
        options={'maxiter': 1000, 'ftol': 1e-9}
