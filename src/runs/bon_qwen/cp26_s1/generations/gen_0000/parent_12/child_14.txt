import numpy as np
from scipy.optimize import minimize


def create_initial_hexagonal(n=26):

    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.08

    # Option: 5-5-5-5-5-1 or 5-5-5-5-4-2 etc.
    row_counts = [5, 5, 5, 5, 5, 1]
    
    idx = 0
    spacing = 0.18

        for col in range(count):
            if idx < n:
                x = 0.08 + col * spacing + (spacing / 2 if row % 2 == 1 else 0)
                y = 0.08 + row * spacing * np.sqrt(3) / 2
                centers[idx] = [x, y]
                idx += 1
    
    return centers, radii


def penalty_objective(x, n, penalty):
    """Compute objective with penalty for constraint violations."""
    c = x[:2 * n].reshape(n, 2)
    r = x[2 * n:]
    
    # Objective: maximize sum of radii (negate for minimization)
    obj = -np.sum(r)
    
    # Boundary penalty: x >= r, 1-x >= r, y >= r, 1-y >= r
    for i in range(n):
        if r[i] > c[i, 0]:
            obj += penalty * (r[i] - c[i, 0]) ** 2
        if c[i, 0] > 1 - r[i]:
            obj += penalty * (c[i, 0] - 1 + r[i]) ** 2
        if r[i] > c[i, 1]:
            obj += penalty * (r[i] - c[i, 1]) ** 2
        if c[i, 1] > 1 - r[i]:
            obj += penalty * (c[i, 1] - 1 + r[i]) ** 2
    
    # Non-overlap penalty: dist(i,j) >= r_i + r_j
    for i in range(n):
        for j in range(i + 1, n):
            dx = c[i, 0] - c[j, 0]
            dy = c[i, 1] - c[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            if dist < r[i] + r[j]:
                obj += penalty * (r[i] + r[j] - dist) ** 2
    
    return obj


def penalty_optimize(centers, radii, n, n_iterations=5):
    """Run penalty-based optimization with increasing penalty."""
    x0 = np.concatenate([centers.flatten(), radii])
    
    penalty = 50
    x = x0.copy()
    
    bounds = [(0.001, 0.999)] * (2 * n) + [(0.001, 0.49)] * n
    
    for iteration in range(n_iterations):
        result = minimize(
            penalty_objective,
            x,
            args=(n, penalty),
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 2000, 'ftol': 1e-15}
        )
        x = result.x
        penalty *= 20
    
    return x[:2 * n].reshape(n, 2), x[2 * n:]


def force_refine(centers, radii, n, iterations=3000):
    """Force-directed refinement to improve packing."""
    centers = centers.copy()
    radii = radii.copy()
    
    lr = 0.003
    repulsion_strength = 80.0
    boundary_force = 5.0
    
    for it in range(iterations):
        forces = np.zeros((n, 2))
        
        # Repulsive forces between circles
        for i in range(n):
            for j in range(i + 1, n):
                diff = centers[j] - centers[i]
                dist_sq = diff[0] ** 2 + diff[1] ** 2
                dist = np.sqrt(dist_sq)
                min_dist = radii[i] + radii[j] + 1e-10
                
                if dist < min_dist:
                    overlap = min_dist - dist
                    force_mag = overlap * repulsion_strength
                    if dist > 1e-10:
                        fx = force_mag * diff[0] / dist
                        fy = force_mag * diff[1] / dist
                        forces[i] -= np.array([fx, fy])
                        forces[j] += np.array([fx, fy])
        
        # Apply forces
        centers = centers + forces * lr
        
        # Project to feasible region (boundary constraints)
        for i in range(n):
            centers[i, 0] = max(radii[i], min(1 - radii[i], centers[i, 0]))
            centers[i, 1] = max(radii[i], min(1 - radii[i], centers[i, 1]))
        
        # Gradually increase radii if no overlaps

            radii = radii + max_radius_increase
            
            # Enforce boundary after radius increase

                centers[i, 1] = max(radii[i], min(1 - radii[i], centers[i, 1]))
        
        # Decay learning rate
        if it % 500 == 0 and it > 0:
            lr *= 0.95
    
    return centers, radii


def validate_and_fix(centers, radii, n):
    """Ensure validity and fix minor violations."""
    centers = centers.copy()
    radii = radii.copy()
    
    # Ensure boundaries
    for i in range(n):
        radii[i] = min(radii[i], centers[i, 0], 1 - centers[i, 0],

        for j in range(i + 1, n):

                radii[j] -= excess / 2
                radii[i] = max(0, radii[i])
                radii[j] = max(0, radii[j])
    
    return centers, radii


def try_configuration(n, perturbation=0.0, seed=0):

    # Final validation
    centers, radii = validate_and_fix(centers, radii, n)
    
    return centers, radii, np.sum(radii)
