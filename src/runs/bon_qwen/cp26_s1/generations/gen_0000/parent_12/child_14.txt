import numpy as np
from scipy.optimize import linprog, minimize

def solve_radii(centers):
    """
    Given centers, solve the LP to find optimal radii maximizing sum(r).
    Returns (radii, sum_radii).
    """
    n = centers.shape[0]
    
    # Precompute pairwise distances
    # dx[i, j] = x_i - x_j
    dx = centers[:, 0][:, np.newaxis] - centers[:, 0][np.newaxis, :]
    dy = centers[:, 1][:, np.newaxis] - centers[:, 1][np.newaxis, :]
    dists = np.sqrt(dx**2 + dy**2)
    
    # LP Constraints: A_ub @ r <= b_ub
    # Variables: r_0, ..., r_{n-1}
    
    # 1. Pairwise constraints: r_i + r_j <= dist(i, j)
    # There are n*(n-1)/2 such constraints.
    # We will construct A_ub and b_ub.
    
    n_pairs = n * (n - 1) // 2
    n_boundary = 4 * n
    total_constraints = n_pairs + n_boundary
    
    A_ub = np.zeros((total_constraints, n))
    b_ub = np.zeros(total_constraints)
    
    row = 0
    
    # Pairwise constraints
    # To speed up, we can iterate
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[row, i] = 1.0
            A_ub[row, j] = 1.0
            b_ub[row] = dists[i, j]
            row += 1
            
    # Boundary constraints
    # r_i <= x_i
    # r_i <= 1 - x_i
    # r_i <= y_i
    # r_i <= 1 - y_i
    
    for i in range(n):
        x, y = centers[i]
        
        # r_i <= x
        A_ub[row, i] = 1.0
        b_ub[row] = x
        row += 1
        
        # r_i <= 1 - x
        A_ub[row, i] = 1.0
        b_ub[row] = 1.0 - x
        row += 1
        
        # r_i <= y
        A_ub[row, i] = 1.0
        b_ub[row] = y
        row += 1
        
        # r_i <= 1 - y
        A_ub[row, i] = 1.0
        b_ub[row] = 1.0 - y
        row += 1
        
    # Objective: Maximize sum(r) -> Minimize -sum(r)
    c = -np.ones(n)
    
    # Bounds: r_i >= 0
    bounds = [(0, None) for _ in range(n)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            radii = res.x
            sum_radii = -res.fun
            return radii, sum_radii
        else:
            # Fallback if LP fails (should not happen for valid centers)
            # Return small radii
            return np.full(n, 0.001), 0.026
    except Exception:
        return np.full(n, 0.001), 0.026

def objective_function(centers_flat):
    """
    Objective function for optimization.
    Minimizes negative sum of radii.
    """
    centers = centers_flat.reshape((26, 2))
    # Clip centers to [0, 1] to handle boundary issues during optimization steps
    # Although bounds are set, numerical noise might occur.
    centers = np.clip(centers, 0.0, 1.0)
    
    radii, sum_r = solve_radii(centers)
    return -sum_r

    # r ~ 1/sqrt(2*26*1.732) ~ 1/sqrt(90) ~ 0.105

    # Row height = r * sqrt(3)
    # Horizontal spacing = 2 * r
