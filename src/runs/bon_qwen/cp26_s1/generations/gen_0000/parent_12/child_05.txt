2. **Alternating Optimization**: Use an iterative two-phase approach:
   - **Phase A (Radius Optimization)**: Fix circle centers and solve a Linear Programming (LP) problem to find the optimal radii that maximize $\sum r_i$ subject to $r_i + r_j \le d_{ij}$ and $r_i \le \text{dist\_to\_wall}$. This is exact, convex, and computationally cheap for $n=26$.
   - **Phase B (Center Optimization)**: Fix radii and compute repulsive forces. Circles that are "tight" (where $r_i + r_j \approx d_{ij}$) or overlapping will experience repulsive forces pushing them apart. This creates more spatial freedom, allowing radii to grow in the next Phase A.
3. **Stochastic Exploration**: Incorporate decaying random jitter (simulated annealing style) to escape local minima where circles get stuck in suboptimal configurations.
4. **Convergence Control**: Use an exponentially decaying step size for center updates to ensure stability as the packing tightens.
5. **Validation Compliance**: The LP constraints directly enforce non-overlap and boundary conditions, guaranteeing the output satisfies the validation function.

import numpy as np
from scipy.optimize import linprog

def generate_initial_centers(n):
    """Generates an initial hexagonal grid of centers."""
    centers = np.zeros((n, 2))
    idx = 0
    # Hexagonal pattern rows tailored for ~26 circles
    rows = [5, 4, 5, 4, 5, 3]
    for r_idx, count in enumerate(rows):
        if idx >= n:
            break
        for c in range(count):
            x = (c + 0.5) * 0.2
            if r_idx % 2 == 1:
                x += 0.1
            y = r_idx * 0.1732 + 0.08
            centers[idx] = [x, y]
            idx += 1
    # Add small random noise to break symmetry and avoid degenerate gradients
    centers += np.random.randn(n, 2) * 0.01
    centers = np.clip(centers, 0.05, 0.95)
    return centers

def compute_forces(centers, radii):
    """Computes repulsive forces to resolve overlaps and push tight constraints apart."""
    n = len(centers)
    forces = np.zeros_like(centers)
    eps = 1e-5
    
    for i in range(n):
        for j in range(i + 1, n):
            diff = centers[i] - centers[j]
            dist = np.linalg.norm(diff)
            limit = radii[i] + radii[j]
            
            # Repulsion proportional to how tight the constraint is.
            # We push apart not just overlapping circles, but also those near the limit,
            # to create space for radius growth in the next LP step.
            slack = limit - dist
            if slack > -0.02:
                force_mag = max(0.0, slack) * 5.0 + 0.005
                if dist > eps:
                    direction = diff / dist
                    forces[i] += force_mag * direction
                    forces[j] -= force_mag * direction
                    
    # Wall repulsion: push circles away if they are too close to boundaries
    for i in range(n):
        for d in range(2):
            if centers[i, d] < radii[i] + 1e-4:
                forces[i, d] += 0.1
            elif centers[i, d] > 1.0 - radii[i] - 1e-4:
                forces[i, d] -= 0.1
                
    return forces

def run_packing():
    np.random.seed(42)
    n = 26
    centers = generate_initial_centers(n)
    radii = np.full(n, 0.02)
    
    # Optimization loop
    for it in range(3000):
        # 1. Solve LP to maximize sum of radii for current centers
        # Objective: maximize sum(r_i) <=> minimize -sum(r_i)
        c_obj = -np.ones(n)
        A_ub = []
        b_ub = []
        
        # Boundary constraints: r_i <= distance to nearest wall
        for i in range(n):
            d_wall = min(centers[i, 0], 1.0 - centers[i, 0], 
                         centers[i, 1], 1.0 - centers[i, 1])
            A_ub.append(np.eye(n)[i])
            b_ub.append(d_wall)
            
        # Non-overlap constraints: r_i + r_j <= Euclidean distance
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.linalg.norm(centers[i] - centers[j])
                row = np.zeros(n)
                row[i] = 1.0
                row[j] = 1.0
                A_ub.append(row)
                b_ub.append(dist)
                
        A_ub = np.array(A_ub)
        b_ub = np.array(b_ub)
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=(0, 0.5), method='highs')
            if res.success:
                radii = res.x
            else:
                # Fallback if LP fails (rare): shrink radii slightly to ensure feasibility
                radii *= 0.99
        except Exception:
            pass
            
        # 2. Update centers based on forces to create more space
        forces = compute_forces(centers, radii)
        
        # Adaptive step size with exponential decay for stability
        step = 0.05 * np.exp(-it / 800.0)
        centers += step * forces
        
        # Simulated annealing jitter to escape local minima
        if it % 50 == 0:
            temp = 0.02 * np.exp(-it / 1000.0)
            centers += np.random.randn(n, 2) * temp
            
        # Ensure centers stay within valid bounds
        centers = np.clip(centers, 0.0, 1.0)
        
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
