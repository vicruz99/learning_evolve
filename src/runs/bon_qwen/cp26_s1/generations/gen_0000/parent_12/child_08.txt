import numpy as np

def run_packing():
    np.random.seed(42)
    N = 26
    
    # Initialize centers in a dense hexagonal-like grid
    centers = np.zeros((N, 2))
    idx = 0
    row = 0
    while idx < N:
        cols_in_row = 6 if row % 2 == 0 else 5
        for c in range(cols_in_row):
            if idx >= N: break
            # Staggered coordinates for hexagonal packing
            x = 0.15 + c * 0.16
            y = 0.12 + row * 0.21
            centers[idx] = [x, y]
            idx += 1
        row += 1
        
    radii = np.full(N, 0.05)
    
    # Optimization parameters
    max_iters = 10000
    growth_rate = 1.00010
    push_strength = 0.7
    
    for it in range(max_iters):
        # Grow radii
        radii *= growth_rate
        
        # Project to boundary constraints
        radii_vec = radii[:, np.newaxis]
        centers = np.clip(centers, radii_vec, 1.0 - radii_vec)
        
        # Resolve overlaps iteratively
        for i in range(N):
            for j in range(i + 1, N):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                min_d = radii[i] + radii[j]
                
                if dist < min_d and dist > 1e-8:
                    overlap = min_d - dist
                    nx, ny = dx / dist, dy / dist
                    # Apply repulsive force, scaled by overlap and damped
                    factor = push_strength * min(1.5, overlap / 0.05)
                    centers[i, 0] += nx * overlap * factor * 0.5
                    centers[i, 1] += ny * overlap * factor * 0.5
                    centers[j, 0] -= nx * overlap * factor * 0.5
                    centers[j, 1] -= ny * overlap * factor * 0.5
                elif dist < 1e-8:
                    # Separate coincident circles
                    centers[i, 0] += 0.005
                    centers[j, 1] += 0.005
                    
        # Re-clamp after push to respect boundaries strictly
        centers = np.clip(centers, radii_vec, 1.0 - radii_vec)
        
        # Annealing noise to escape local minima
        noise_scale = 0.002 * max(0.0, 1.0 - it / max_iters)
        centers += np.random.uniform(-noise_scale, noise_scale, (N, 2))
        centers = np.clip(centers, radii_vec, 1.0 - radii_vec)

    # Final precision clamp to guarantee compliance with validation tolerance
    margin = 1e-9
    radii_final = np.minimum(radii, np.minimum(np.minimum(centers[:,0], 1-centers[:,0]), 
                                               np.minimum(centers[:,1], 1-centers[:,1])))
    radii_final = np.maximum(radii_final, 0.0)
    
    return centers, radii_final, np.sum(radii_final)
