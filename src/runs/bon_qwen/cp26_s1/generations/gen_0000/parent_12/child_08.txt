from scipy.optimize import minimize

    # 2. Optimization Function
    def objective(params):

        x = params[:N * 2].reshape(N, 2)
        r = params[N * 2:]

        obj = -np.sum(r)
        
        # Penalty terms for constraints
        penalty = 0

        for i in range(N):

            if x[i, 0] - r[i] < 0: penalty += 100 * (x[i, 0] - r[i])**2
