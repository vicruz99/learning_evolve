1. Each circle stays within the unit square boundaries.
2. No two circles overlap (Euclidean distance between centers $\ge$ sum of radii).

We use `scipy.optimize.minimize` with the `SLSQP` method, which handles non-linear constraints efficiently. The variable vector contains all x-coordinates, y-coordinates, and radii. Constraints are formulated as inequalities $\ge 0$.

1. Initialize circles in a valid 5x5 grid plus one extra circle with small radii.
2. Run SLSQP to optimize.
3. Perturb the best solution slightly and repeat optimization several times.

import numpy as np
from scipy.optimize import minimize, NonlinearConstraint

N_CIRCLES = 26

def _objective(vars):
    """Objective function: minimize negative sum of radii."""
    return -np.sum(vars[2 * N_CIRCLES:])

def _constraints(vars):
    """
    Constraint function: returns array of constraint values.
    All values must be >= 0 for a feasible solution.
    """
    centers = vars[:2 * N_CIRCLES].reshape(N_CIRCLES, 2)
    radii = vars[2 * N_CIRCLES:]
    
    c = []
    # Boundary constraints: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    for i in range(N_CIRCLES):
        c.append(centers[i, 0] - radii[i])
        c.append(1.0 - centers[i, 0] - radii[i])
        c.append(centers[i, 1] - radii[i])
        c.append(1.0 - centers[i, 1] - radii[i])
        
    # Pairwise separation constraints: dist^2 >= (r_i + r_j)^2
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist_sq = np.sum(diff**2, axis=2)
    sum_r = radii[:, np.newaxis] + radii[np.newaxis, :]
    idx = np.triu_indices(N_CIRCLES, k=1)
    c.extend(dist_sq[idx] - sum_r[idx]**2)
    
    return np.array(c)

def run_packing():
    # Bounds: centers in [0, 1], radii in [1e-5, 0.4]
    bounds = [(0.0, 1.0)] * (2 * N_CIRCLES) + [(1e-5, 0.4)] * N_CIRCLES
    
    nl_con = NonlinearConstraint(_constraints, 0, np.inf)
    
    # Initial valid configuration: 5x5 grid + 1 extra circle
    centers_init = np.zeros((N_CIRCLES, 2))
    radii_init = np.full(N_CIRCLES, 0.03)
    idx = 0
    for i in range(5):
        for j in range(5):
            centers_init[idx] = [0.1 + i * 0.2, 0.1 + j * 0.2]
            idx += 1
    centers_init[idx] = [0.5, 0.9]
    
    x0 = np.hstack([centers_init.flatten(), radii_init])
    
    best_x = x0.copy()
    best_val = -np.inf
    options = {'maxiter': 2000, 'ftol': 1e-12, 'disp': False}
    
    # Multi-start optimization with perturbation
    for _ in range(12):
        # Perturb current best to explore landscape
        perturbation = np.random.uniform(-0.02, 0.02, size=x0.shape)
        x_curr = np.clip(best_x + perturbation, 1e-3, 1 - 1e-3)
        x_curr[2 * N_CIRCLES:] = np.clip(x_curr[2 * N_CIRCLES:], 1e-4, 0.35)
        
        try:
            res = minimize(_objective, x_curr, method='SLSQP', bounds=bounds,
                           constraints=nl_con, options=options)
            val = -res.fun
            if val > best_val:
                best_val = val
                best_x = res.x
        except Exception:
            continue
            
    centers_opt = best_x[:2 * N_CIRCLES].reshape(N_CIRCLES, 2)
    radii_opt = best_x[2 * N_CIRCLES:]
    
    # Apply tiny safety margin to satisfy 1e-12 tolerance strictly
    radii_opt = np.maximum(radii_opt - 1e-7, 1e-8)
    
    return centers_opt, radii_opt, float(np.sum(radii_opt))
