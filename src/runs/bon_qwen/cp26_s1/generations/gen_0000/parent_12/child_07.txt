import numpy as np
import scipy.optimize as opt
import warnings

# Suppress warnings for cleaner output
warnings.filterwarnings('ignore')

def generate_hexagonal_initialization(n, square_size=1.0, init_radius=0.05):
    """
    Generates an initial valid packing of n circles in a hexagonal lattice pattern.
    Starts with a small radius to ensure validity, then we rely on optimizer to expand.
    """
    centers = []
    # Approximate rows needed
    # Area estimate: n * pi * r^2 ~ square_size^2 * density
    # But we just want to place them.
    # Estimate side length of hex grid roughly sqrt(n)
    
    # We can just place them in a grid and then shift to hex
    # Or just iterate rows.
    
    # Heuristic: estimate rows based on area
    # r = 0.05, diameter 0.1. 
    # 1.0 / 0.1 = 10 spots per row? 
    # Let's just try to fill the square with a hex pattern until we have n circles.
    
    # To ensure we fit, let's scale the pattern down if needed, 
    # but init_radius is small enough.
    
    x = init_radius
    y = init_radius
    row_height = init_radius * np.sqrt(3)
    col_width = 2 * init_radius
    
    count = 0
    row_idx = 0
    
    # We need to collect centers. 
    # We'll iterate y, then x.
    
    # Estimate how many rows/cols to generate buffer
    max_iter = n * 2
    
    while count < n and row_idx < max_iter:
        # x offset for hexagonal packing
        x_offset = init_radius if row_idx % 2 == 1 else 0
        curr_x = init_radius + x_offset
        
        while curr_x + init_radius <= square_size + 1e-9: # Allow tiny overlap for init, or strict?
             # Strictly inside
             if curr_x + init_radius <= 1.0 + 1e-12 and curr_x - init_radius >= -1e-12:
                 centers.append([curr_x, y + row_idx * row_height])
                 count += 1
                 if count >= n:
                     break
             curr_x += col_width
        row_idx += 1
        
    centers = np.array(centers[:n])
    # If we failed to generate enough (unlikely with small radius), pad with random
    if len(centers) < n:
        extra = np.random.uniform(0, 1, (n - len(centers), 2))
        centers = np.vstack([centers, extra])
        
    # Randomize slightly to break symmetry and avoid local optima
    centers += np.random.uniform(-0.01, 0.01, centers.shape)
    centers = np.clip(centers, init_radius + 0.01, 1.0 - init_radius - 0.01)
    
    radii = np.full(n, init_radius)
    return centers, radii

def run_packing():
    n = 26
    
    # 1. Generate initial configuration
    # We try a few different seeds/perturbations to find a good start
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Function to run one optimization trial
    def optimize_trial(seed):
        np.random.seed(seed)
        centers, radii = generate_hexagonal_initialization(n)
        
        # Variables: [x1, y1, r1, x2, y2, r2, ...]
        # Total 3 * n variables
        x0 = np.concatenate([centers.flatten(), radii])
        
        # Bounds for variables
        # x, y in [0, 1]
        # r in [0, 0.5] (max possible radius in unit square)
        bounds = []
        for i in range(n):
            bounds.append((0.0, 1.0)) # x
            bounds.append((0.0, 1.0)) # y
            bounds.append((1e-6, 0.5)) # r (small epsilon to avoid 0)
            
        # Define constraints
        # 1. Non-overlap: dist(c_i, c_j) >= r_i + r_j
        # Equivalent to: (r_i + r_j)^2 <= ||c_i - c_j||^2
        # Or simply: r_i + r_j - dist <= 0
        
        def constraint_overlap(params, i, j):
            xi, yi, ri = params[3*i], params[3*i+1], params[3*i+2]
            xj, yj, rj = params[3*j], params[3*j+1], params[3*j+2]
            
            dist = np.sqrt((xi - xj)**2 + (yi - yj)**2)
            # Constraint: dist - (ri + rj) >= 0
            return dist - (ri + rj)
            
        # 2. Boundary constraints
        # x - r >= 0 => x >= r
        # 1 - x - r >= 0 => x + r <= 1
        # Same for y
        
        def constraint_left(params, i):
            return params[3*i] - params[3*i+2] # x - r >= 0
        
        def constraint_right(params, i):
            return 1.0 - params[3*i] - params[3*i+2] # 1 - x - r >= 0
            
        def constraint_bottom(params, i):
            return params[3*i+1] - params[3*i+2] # y - r >= 0
            
        def constraint_top(params, i):
            return 1.0 - params[3*i+1] - params[3*i+2] # 1 - y - r >= 0

        constraints = []
        
        # Add overlap constraints for all pairs
        # This is O(N^2), for N=26 it's ~325 constraints. Manageable.
        for i in range(n):
            for j in range(i + 1, n):
                constraints.append({
                    'type': 'ineq',
                    'fun': lambda p, i=i, j=j: constraint_overlap(p, i, j)
                })
        
        # Add boundary constraints
        for i in range(n):
            constraints.append({'type': 'ineq', 'fun': lambda p, i=i: constraint_left(p, i)})
            constraints.append({'type': 'ineq', 'fun': lambda p, i=i: constraint_right(p, i)})
            constraints.append({'type': 'ineq', 'fun': lambda p, i=i: constraint_bottom(p, i)})
            constraints.append({'type': 'ineq', 'fun': lambda p, i=i: constraint_top(p, i)})
            
        # Objective: Maximize sum(r) => Minimize -sum(r)
        def objective(params):
            return -np.sum(params[2::3]) # r is at indices 2, 5, 8...
            
        # Optimization
        try:
            res = opt.minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints,
                               options={'maxiter': 1000, 'ftol': 1e-9})
            
            if res.success:
                final_r_sum = -res.fun
                return res.x, final_r_sum
            else:
                # Even if not successful, take the result
                final_r_sum = -objective(res.x)
                return res.x, final_r_sum
        except Exception as e:
            return None, -1.0

    # Run multiple trials with different random seeds to find global optimum
    # 5-10 trials should be sufficient
    for seed in range(10):
        params, s = optimize_trial(seed)
        if params is not None and s > best_sum:
            best_sum = s
            best_params = params
            
    # Extract results
    if best_params is None:
        # Fallback if everything fails
        centers, radii = generate_hexagonal_initialization(n)
        best_sum = np.sum(radii)
    else:
        best_centers = np.array([[best_params[3*i], best_params[3*i+1]] for i in range(n)])
        best_radii = np.array([best_params[3*i+2] for i in range(n)])

    # Post-processing: Validate and fix small numerical issues
    # Ensure radii are positive and centers valid
    if np.any(best_radii < 1e-9):
         best_radii[best_radii < 1e-9] = 1e-9

    # Re-clip centers if they drifted slightly outside
    best_centers = np.clip(best_centers, 1e-9, 1.0 - 1e-9)
    
    # Final Check: If any overlap due to precision, shrink radii slightly
    # This is a safety step
    valid = True
    for i in range(n):
        for j in range(i+1, n):
            dist = np.linalg.norm(best_centers[i] - best_centers[j])
            req_dist = best_radii[i] + best_radii[j]
            if dist < req_dist - 1e-9:
                # Overlap detected. Shrink.
                # Simple fix: reduce both radii proportionally
                scale = dist / req_dist
                best_radii[i] *= scale
                best_radii[j] *= scale
                valid = False # Might need to re-check, but usually fixes locally
                
    # Check boundaries
    for i in range(n):
        x, y = best_centers[i]
        r = best_radii[i]
        if x - r < 0: best_radii[i] = x
        if x + r > 1: best_radii[i] = 1 - x
        if y - r < 0: best_radii[i] = y
        if y + r > 1: best_radii[i] = 1 - y

    final_sum = np.sum(best_radii)
    return best_centers, best_radii, final_sum
