import numpy as np
import math

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    Uses a physics-based repulsion and growth algorithm.
    """
    n = 26
    np.random.seed(42)

    # 1. Initialization: Staggered Hexagonal Grid
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Approximate radius for 26 circles in hex packing is ~0.1
    # Start with a safe small radius to build up
    current_r = 0.02
    
    # Determine grid dimensions to fit 26 circles
    # Rows roughly sqrt(26) ~ 5-6
    # Hexagonal packing: width of row with k circles is 2*r + (k-1)*2*r = 2*k*r
    # But we shift rows.
    
    # Let's just place them in a dense pattern
    # 5 rows
    # Counts: 5, 6, 5, 6, 4 -> 26
    counts = [5, 6, 5, 6, 4]
    
    idx = 0
    for row_idx, count in enumerate(counts):
        y = 0.1 + row_idx * 0.15 # Initial vertical spacing
        # Stagger x for even/odd rows
        x_start = 0.1 if row_idx % 2 == 0 else 0.2
        
        for col_idx in range(count):
            x = x_start + col_idx * 0.15
            centers[idx] = [x, y]
            radii[idx] = current_r
            idx += 1

    # 2. Optimization Loop
    # We want to maximize sum(radii).
    # We will iteratively increase radii and resolve collisions.
    
    num_iterations = 5000
    growth_rate = 0.0005
    repulsion_strength = 5.0
    damping = 0.95
    
    velocities = np.zeros((n, 2))

    for step in range(num_iterations):
        # 1. Grow radii
        # Decay growth rate to converge
        current_growth = growth_rate * (0.9995 ** step)
        radii += current_growth
        
        # 2. Check constraints and apply forces
        forces = np.zeros((n, 2))
        
        # Boundary forces
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            
            # Left
            if x - r < 0:
                forces[i, 0] += repulsion_strength * (r - x)
            # Right
            if x + r > 1:
                forces[i, 0] -= repulsion_strength * (x + r - 1)
            # Bottom
            if y - r < 0:
                forces[i, 1] += repulsion_strength * (r - y)
            # Top
            if y + r > 1:
                forces[i, 1] -= repulsion_strength * (y + r - 1)
                
        # Overlap forces
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist_sq = dx*dx + dy*dy
                dist = math.sqrt(dist_sq)
                min_dist = radii[i] + radii[j]
                
                if dist < min_dist and dist > 1e-9:
                    overlap = min_dist - dist
                    # Normalized direction
                    nx = dx / dist
                    ny = dy / dist
                    
                    # Apply repulsion proportional to overlap
                    force_mag = repulsion_strength * overlap
                    
                    forces[i, 0] += nx * force_mag
                    forces[i, 1] += ny * force_mag
                    forces[j, 0] -= nx * force_mag
                    forces[j, 1] -= ny * force_mag

        # Update velocities and positions
        velocities += forces * (1.0 / (radii.reshape(-1, 1) + 0.01)) # Inverse mass roughly
        velocities *= damping
        centers += velocities * 0.05 # Step size

        # Clamp positions to square to prevent numerical drift out of bounds
        centers = np.clip(centers, radii.reshape(-1, 1), 1 - radii.reshape(-1, 1))

    # 3. Final Validation and Adjustment
    # Ensure strict validity by shrinking radii slightly if needed
    # But the simulation should keep them valid.
    # Let's do a final pass to fix any tiny overlaps due to floating point
    for i in range(n):
        r = radii[i]
        x, y = centers[i]
        # Boundary clamp
        if x - r < -1e-12: radii[i] = x
        if x + r > 1 + 1e-12: radii[i] = 1 - x
        if y - r < -1e-12: radii[i] = y
        if y + r > 1 + 1e-12: radii[i] = 1 - y

    # Sort circles by radius for consistent output (optional)
    # Not required but good practice
    
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
