

```python
import numpy as np
from scipy.optimize import minimize, linprog
import warnings

warnings.filterwarnings('ignore')

N_CIRCLES = 26

def get_pairwise_dists(centers):
    """Compute Euclidean distances between all pairs of centers."""
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    return np.linalg.norm(diffs, axis=2)

def solve_radii_lp(centers):
    """Solve LP to maximize sum of radii for fixed centers."""
    N = N_CIRCLES
    dists = get_pairwise_dists(centers)
    i_idx, j_idx = np.tril_indices(N, -1)
    
    bounds = []
    for idx in range(N):
        x, y = centers[idx]
        max_r = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max(0.0, max_r)))
        
    A_ub = []
    b_ub = []
    for k in range(len(i_idx)):
        row = np.zeros(N)
        row[i_idx[k]] = 1.0
        row[j_idx[k]] = 1.0
        A_ub.append(row)
        b_ub.append(dists[i_idx[k], j_idx[k]])
        
    c_obj = np.ones(N)
    res = linprog(-c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success and res.x is not None:
        return np.maximum(res.x, 0.0)
    return np.full(N, 0.05)

def objective(v):
    """Objective to maximize sum of radii."""
    return -np.sum(v[2*N_CIRCLES:])

def constraints_overlap(v):
    """Non-overlap constraints: dist(i,j) - r_i - r_j >= 0."""
    centers = v[:2*N_CIRCLES].reshape(N_CIRCLES, 2)
    radii = v[2*N_CIRCLES:]
    dists = get_pairwise_dists(centers)
    i, j = np.tril_indices(N_CIRCLES, -1)
    return dists[i, j] - radii[i] - radii[j]

def constraints_boundary(v):
    """Boundary constraints: center +/- radius in [0, 1]."""
    centers = v[:2*N_CIRCLES].reshape(N_CIRCLES, 2)
    radii = v[2*N_CIRCLES:]
    con = np.empty(4*N_CIRCLES)
    con[0::4] = centers[:, 0] - radii
    con[1::4] = 1.0 - centers[:, 0] - radii
    con[2::4] = centers[:, 1] - radii
    con[3::4] = 1.0 - centers[:, 1] - radii
    return con

def run_packing():
    bounds = [(0.0, 1.0)] * 2*N_CIRCLES + [(0.0, 0.5)] * N_CIRCLES
    cons = [
        {'type': 'ineq', 'fun': constraints_overlap},
        {'type': 'ineq', 'fun': constraints_boundary}
    ]
    
    best_v = None
    best_sum_r = -1.0
    
    # Prepare diverse initial configurations
    init_configs = []
    
    # 1. Hexagonal lattice approximation
    cfg_hex = []
    y = 0.12
    for r_idx in range(7):
        n_in_row = 6 if r_idx % 2 == 0 else 5
        x_start = 0.06 if r_idx % 2 == 1 else 0.01
        for c_idx in range(n_in_row):
            if len(cfg_hex) < N_CIRCLES:
                cfg_hex.append([x_start + c_idx * 0.17, y + r_idx * 0.14])
    init_configs.append(np.array(cfg_hex))
    
    # 2. Structured grid
    cx, cy = np.meshgrid(np.linspace(0.2, 0.8, 5), np.linspace(0.2, 0.8, 5))
    cfg_grid = np.column_stack([cx.ravel(), cy.ravel()])[:N_CIRCLES]
    init_configs.append(cfg_grid)
    
    # 3. Randomized positions
    np.random.seed(42)
    cfg_rand = np.random.uniform(0.15, 0.85, (N_CIRCLES, 2))
    init_configs.append(cfg_rand)

    for cfg in init_configs:
        centers_init = cfg.copy()
        centers_init = np.clip(centers_init, 0.02, 0.98)
        r_init = np.full(N_CIRCLES, 0.04)
        v0 = np.concatenate([centers_init.flatten(), r_init])
        
        # Joint optimization of centers and radii
        res = minimize(objective, v0, method='SLSQP', bounds=bounds, constraints=cons, 
                       options={'maxiter': 3000, 'ftol': 1e-10})
        
        curr_sum_r = -res.fun
        if curr_sum_r > best_sum_r:
            best_sum_r = curr_sum_r
            best_v = res.x
            
        # LP refinement for fixed centers
        centers_lp = best_v[:2*N_CIRCLES].reshape(N_CIRCLES, 2)
        radii_lp = solve_radii_lp(centers_lp)
        sum_lp = np.sum(radii_lp)
        
        if sum_lp > best_sum_r:
            best_sum_r = sum_lp
            best_v = np.concatenate([centers_lp.flatten(), radii_lp])
            
        # Second SLSQP pass to adjust centers to LP radii
        res2 = minimize(objective, best_v, method='SLSQP', bounds=bounds, constraints=cons, 
                        options={'maxiter': 2000, 'ftol': 1e-10})
        curr_sum_r2 = -res2.fun
        if curr_sum_r2 > best_sum_r:
            best_sum_r = curr_sum_r2
            best_v = res2.x
            
    centers = best_v[:2*N_CIRCLES].reshape(N_CIRCLES, 2)
    radii = solve_radii_lp(centers)
    
    return centers, radii, np.sum(radii)
```
