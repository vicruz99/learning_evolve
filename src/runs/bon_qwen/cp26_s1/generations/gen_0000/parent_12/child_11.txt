import numpy as np
from scipy.optimize import minimize

def generate_hex_lattice(n):
    """
    Generates an initial configuration of n circles arranged in a hexagonal lattice.
    Returns centers (n, 2) and radii (n,).
    """
    centers = []
    r_init = 0.05 # Small initial radius to ensure feasibility
    row = 0
    col = 0
    
    # Approximate spacing based on density
    # Area per circle in hex packing is sqrt(3)/2 * (2r)^2 = 2*sqrt(3)*r^2
    # Total area 1. n * 2*sqrt(3)*r^2 approx 1 => r approx sqrt(1/(n*2*sqrt(3)))
    # For n=26, r approx 0.1. 
    # Spacing between centers should be approx 2r.
    # We use a spacing that fits n circles roughly in 1x1.
    spacing = 0.25 
    
    # We will generate more points than needed and pick the first n, 
    # or just generate exactly n in rows.
    # Let's try to fit them in a block.
    
    # Estimate number of rows
    # Hex packing: rows have length n_row.
    # Width ~ n_row * spacing. Height ~ num_rows * spacing * sqrt(3)/2.
    # Let's try 6 rows.
    
    points = []
    y = 0
    while len(points) < n:
        x = 0
        # Offset for odd rows
        if y % 2 == 1:
            x_start = spacing / 2
        else:
            x_start = 0
            
        while x + x_start < 1.2 and len(points) < n: # Generate slightly more to fill
            points.append([x_start + x, y * spacing * np.sqrt(3)/2])
            x += spacing
        y += 1
    
    centers = np.array(points[:n])
    
    # Normalize centers to be well within [0,1]
    # Find min/max x, y
    min_c = np.min(centers, axis=0)
    max_c = np.max(centers, axis=0)
    
    # Scale to fit inside with margin
    # We want centers to be in [margin, 1-margin]
    margin = 0.1
    if max_c[0] - min_c[0] > 0:
        centers[:, 0] = margin + (centers[:, 0] - min_c[0]) / (max_c[0] - min_c[0]) * (1 - 2*margin)
    else:
        centers[:, 0] = 0.5
        
    if max_c[1] - min_c[1] > 0:
        centers[:, 1] = margin + (centers[:, 1] - min_c[1]) / (max_c[1] - min_c[1]) * (1 - 2*margin)
    else:
        centers[:, 1] = 0.5
        
    radii = np.full(n, r_init)
    return centers, radii

def objective_func(vars):
    """Objective: Maximize sum of radii => Minimize negative sum."""
    n = len(vars) // 3
    radii = vars[2::3] # Every 3rd element starting from index 2
    return -np.sum(radii)

def constraint_func(vars):
    """
    Returns a vector of constraint values. All must be >= 0.
    Constraints:
    1. x_i - r_i >= 0
    2. 1 - x_i - r_i >= 0
    3. y_i - r_i >= 0
    4. 1 - y_i - r_i >= 0
    5. dist(i, j) - (r_i + r_j) >= 0
    """
    n = len(vars) // 3
    centers = np.zeros((n, 2))
    centers[:, 0] = vars[0::3]
    centers[:, 1] = vars[1::3]
    radii = vars[2::3]
    
    constraints = []
    
    # Boundary constraints
    for i in range(n):
        constraints.append(centers[i, 0] - radii[i])      # x >= r
        constraints.append(1 - centers[i, 0] - radii[i])  # x <= 1-r
        constraints.append(centers[i, 1] - radii[i])      # y >= r
        constraints.append(1 - centers[i, 1] - radii[i])  # y <= 1-r
        
    # Pairwise constraints
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            constraints.append(dist - (radii[i] + radii[j]))
            
    return np.array(constraints)

        try:
