import numpy as np

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    n = centers.shape[0]

    if np.isnan(centers).any():
        return False
    if np.isnan(radii).any():
        return False

    for i in range(n):
        if radii[i] < 0:
            return False
        if np.isnan(radii[i]):
            return False

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            return False

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False
    return True

def init_hexagonal(n):
    centers = np.zeros((n, 2))
    x, y = 0.05, 0.05
    row_len = 5
    row = 0
    count = 0
    while count < n:
        cols = row_len if row % 2 == 0 else row_len - 1
        for c in range(cols):
            if count >= n: break
            centers[count, 0] = x + c * 0.19
            centers[count, 1] = y + row * 0.16
            count += 1
        row += 1
    return centers

def init_grid(n):
    centers = np.zeros((n, 2))
    cols = int(np.ceil(np.sqrt(n)))
    rows = int(np.ceil(n / cols))
    count = 0
    for r in range(rows):
        for c in range(cols):
            if count >= n: break
            centers[count, 0] = (c + 0.5) / cols
            centers[count, 1] = (r + 0.5) / rows
            count += 1
    return centers

def run_packing():
    n = 26
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Inflation parameters
    base_r = 0.01
    delta_r = 0.0003
    max_steps = 1500
    res_passes = 15
    
    # Test multiple initial layouts
    init_funcs = [init_hexagonal, init_grid]
    seeds = [np.random.seed(i) for i in range(5)]

    for init_func in init_funcs:
        for seed in range(5):
            np.random.seed(seed)
            centers = init_func(n).copy()
            # Add jitter
            centers += np.random.uniform(-0.02, 0.02, size=(n, 2))
            centers = np.clip(centers, 0.05, 0.95)
            
            radii = np.full(n, base_r)
            
            current_delta = delta_r
            
            for step in range(max_steps):
                radii += current_delta
                
                for _ in range(res_passes):
                    moved = False
                    # Boundary resolution
                    for i in range(n):
                        r = radii[i]
                        if centers[i, 0] < r: centers[i, 0] = r; moved = True
                        if centers[i, 0] > 1 - r: centers[i, 0] = 1 - r; moved = True
                        if centers[i, 1] < r: centers[i, 1] = r; moved = True
                        if centers[i, 1] > 1 - r: centers[i, 1] = 1 - r; moved = True
                    
                    # Overlap resolution
                    for i in range(n):
                        for j in range(i + 1, n):
                            d_vec = centers[j] - centers[i]
                            d = np.linalg.norm(d_vec)
                            if d == 0:
                                d_vec = np.array([0.01, 0.01])
                                d = np.linalg.norm(d_vec)
                            
                            req_dist = radii[i] + radii[j]

                                shift = (req_dist - d) / 2.0
                                direction = d_vec / d
                                centers[i] -= shift * direction
                                centers[j] += shift * direction
                                moved = True
                    
                    if not moved:
                        break
            
            # Final clamp
            for i in range(n):
                centers[i, 0] = np.clip(centers[i, 0], radii[i], 1 - radii[i])

                    best_centers = centers.copy()
                    best_radii = radii.copy()

    if best_centers is None:
        best_centers = np.random.rand(n, 2)
        best_radii = np.ones(n) * 0.01
        
    return best_centers, best_radii, np.sum(best_radii)
