import numpy as np
from scipy.optimize import minimize
import math

def initialize_hexagonal(n):
    """Initialize circles in a hexagonal lattice pattern"""
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.08
    
    row_counts = [6, 5, 6, 5, 4]
    x_spacing = 0.17
    y_spacing = x_spacing * math.sqrt(3) / 2
    
    x_center = 0.5
    y_start = 0.1
    
    idx = 0
    for row_idx, n_cols in enumerate(row_counts):
        y = y_start + row_idx * y_spacing
        x_start = x_center - (n_cols - 1) * x_spacing / 2
        
        for col in range(n_cols):
            if idx >= n:
                break
            x = x_start + col * x_spacing
            if row_idx % 2 == 1:
                x += x_spacing / 2
            
            centers[idx] = [x, y]
            idx += 1
    
    return centers, radii


def initialize_random(n, seed=42):
    """Initialize circles randomly in the unit square"""
    rng = np.random.RandomState(seed)
    centers = rng.rand(n, 2) * 0.6 + 0.2  # Concentrated in center
    radii = np.ones(n) * 0.03
    return centers, radii


def expand_and_resolve(centers, radii, n, iterations=3000):
    """Iteratively expand circles and resolve overlaps"""
    centers = centers.copy()
    radii = radii.copy()
    
    for iteration in range(iterations):
        # Expand each circle
        for i in range(n):
            max_r = min(
                centers[i, 0],
                1.0 - centers[i, 0],
                centers[i, 1],
                1.0 - centers[i, 1]
            )
            
            for j in range(n):
                if i != j:
                    dx = centers[i, 0] - centers[j, 0]
                    dy = centers[i, 1] - centers[j, 1]
                    dist = math.sqrt(dx * dx + dy * dy)
                    max_r = min(max_r, dist - radii[j])
            
            max_r = max(0.0001, max_r)
            # Gradual expansion
            radii[i] = radii[i] + 0.5 * (max_r - radii[i])
        
        # Resolve overlaps
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = math.sqrt(dx * dx + dy * dy)
                required = radii[i] + radii[j]
                
                if dist < required and dist > 1e-12:
                    direction_x = dx / dist
                    direction_y = dy / dist
                    shift = (required - dist) / 2.0
                    centers[i, 0] += shift * direction_x
                    centers[i, 1] += shift * direction_y
                    centers[j, 0] -= shift * direction_x
                    centers[j, 1] -= shift * direction_y
        
        # Clip to bounds
        for i in range(n):
            centers[i, 0] = max(radii[i], min(1.0 - radii[i], centers[i, 0]))
            centers[i, 1] = max(radii[i], min(1.0 - radii[i], centers[i, 1]))
    
    return centers, radii


def optimize_with_perturbations(centers, radii, n, num_restarts=5):
    """Try multiple perturbed initializations and keep best"""
    best_centers = centers.copy()
    best_radii = radii.copy()
    best_sum = np.sum(radii)
    
    for restart in range(num_restarts):
        c = centers.copy()
        r = radii.copy()
        
        # Add perturbation
        if restart > 0:
            scale = 0.02 * (restart + 1)
            for i in range(n):
                c[i, 0] += np.random.randn() * scale
                c[i, 1] += np.random.randn() * scale
                # Clamp
                c[i, 0] = max(r[i], min(1.0 - r[i], c[i, 0]))
                c[i, 1] = max(r[i], min(1.0 - r[i], c[i, 1]))
        
        c, r = expand_and_resolve(c, r, n, iterations=2000)
        
        if np.sum(r) > best_sum:
            best_sum = np.sum(r)
            best_centers = c.copy()
            best_radii = r.copy()
    
    return best_centers, best_radii


def constraint_func(v, n):
    """Compute all constraint values (must be >= 0)"""
    constraints = []
    
    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    
    # Boundary constraints
    constraints.extend(x - r)
    constraints.extend(1.0 - x - r)
    constraints.extend(y - r)
    constraints.extend(1.0 - y - r)
    
    # Non-overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            dx = v[3 * i] - v[3 * j]
            dy = v[3 * i + 1] - v[3 * j + 1]
            dist = math.sqrt(dx * dx + dy * dy + 1e-20)
            constraints.append(dist - v[3 * i + 2] - v[3 * j + 2])
    
    return np.array(constraints)


def objective_func(v):
    """Negative sum of radii (minimize this)"""
    return -np.sum(v[2::3])


def constraint_func_26(v):
    """Constraint function wrapper for n=26"""
    return constraint_func(v, 26)


def scipy_fine_tune(centers, radii, n):
    """Use scipy SLSQP to fine-tune the packing"""
    x0 = np.concatenate([centers.ravel(), radii])
    
    bounds = []
    for i in range(n):
        bounds.append((0.0, 1.0))
        bounds.append((0.0, 1.0))
        bounds.append((0.0, 0.5))
    
    cons = {
        'type': 'ineq',
        'fun': constraint_func_26
    }
    
    try:
        result = minimize(
            objective_func,
            x0,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 3000, 'ftol': 1e-15, 'disp': False}
        )
        
        if result.success or result.fun < 0:
            centers_opt = result.x[:2 * n].reshape(n, 2)
            radii_opt = result.x[2 * n:]
            # Ensure non-negative radii
            radii_opt = np.maximum(radii_opt, 0)
            # Ensure circles are in bounds
            for i in range(n):
                radii_opt[i] = min(radii_opt[i], centers_opt[i, 0], 1 - centers_opt[i, 0],
                                   centers_opt[i, 1], 1 - centers_opt[i, 1])
                radii_opt[i] = max(0, radii_opt[i])
                centers_opt[i, 0] = max(radii_opt[i], min(1 - radii_opt[i], centers_opt[i, 0]))
                centers_opt[i, 1] = max(radii_opt[i], min(1 - radii_opt[i], centers_opt[i, 1]))
            
            return centers_opt, radii_opt
    except Exception:
        pass
    
    return centers, radii


def equalize_radii_pass(centers, radii, n, iterations=100):
    """Pass to make radii more equal, which can improve packing density"""
    centers = centers.copy()
    radii = radii.copy()
    
    for _ in range(iterations):
        avg_r = np.mean(radii)
        # Gradually move radii toward average
        for i in range(n):
            radii[i] = radii[i] + 0.1 * (avg_r - radii[i])
        
        # Resolve overlaps after radius changes
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = math.sqrt(dx * dx + dy * dy)
                required = radii[i] + radii[j]
                
                if dist < required and dist > 1e-12:
                    direction_x = dx / dist
                    direction_y = dy / dist
                    shift = (required - dist) / 2.0
                    centers[i, 0] += shift * direction_x
                    centers[i, 1] += shift * direction_y
                    centers[j, 0] -= shift * direction_x
                    centers[j, 1] -= shift * direction_y
        
        # Clip to bounds
        for i in range(n):
            centers[i, 0] = max(radii[i], min(1.0 - radii[i], centers[i, 0]))
            centers[i, 1] = max(radii[i], min(1.0 - radii[i], centers[i, 1]))
            
            # Ensure radius fits in bounds
            max_possible = min(
                centers[i, 0],
                1.0 - centers[i, 0],
                centers[i, 1],
                1.0 - centers[i, 1]
            )
            radii[i] = min(radii[i], max_possible)
    
    return centers, radii


def run_packing():

    n = 26
