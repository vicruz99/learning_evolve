import numpy as np
from scipy.optimize import minimize
from functools import partial


def penalty_objective(x, n, penalty):
    """Objective with penalty terms for constraint violations.
    Maximizes sum of radii (negated) with penalties for overlaps and boundary violations.
    """
    obj = -np.sum(x[2::3])

    # Non-overlap penalty: penalize when dist < r_i + r_j
    for i in range(n):
        xi, yi, ri = x[3*i], x[3*i+1], x[3*i+2]
        for j in range(i + 1, n):
            xj, yj, rj = x[3*j], x[3*j+1], x[3*j+2]
            dist = np.sqrt((xi - xj) ** 2 + (yi - yj) ** 2 + 1e-12)
            overlap = ri + rj - dist
            if overlap > 0:
                obj -= penalty * overlap * overlap

    # Boundary penalties
    for i in range(n):
        xi, yi, ri = x[3*i], x[3*i+1], x[3*i+2]

        # Left boundary: xi - ri >= 0
        v = -(xi - ri)
        if v > 0:
            obj -= penalty * v * v

        # Right boundary: xi + ri <= 1
        v = xi + ri - 1
        if v > 0:
            obj -= penalty * v * v

        # Bottom boundary: yi - ri >= 0
        v = -(yi - ri)
        if v > 0:
            obj -= penalty * v * v

        # Top boundary: yi + ri <= 1
        v = yi + ri - 1
        if v > 0:
            obj -= penalty * v * v

        # Radius non-negativity
        if ri < 0:
            obj -= penalty * ri * ri

    return obj


def initialize_hexagonal(n, seed):
    """Initialize circles in a hexagonal grid pattern with given seed."""
    np.random.seed(seed)

    row_options = [
        [6, 5, 6, 5, 4],
        [5, 5, 5, 5, 6],
        [4, 6, 5, 6, 5],
        [5, 6, 5, 6, 4],
        [6, 5, 5, 5, 5],
        [5, 6, 5, 5, 5],
        [5, 5, 6, 5, 5],
        [5, 5, 5, 6, 5],
    ]
    row_counts = row_options[seed % len(row_options)]

    centers = []
    num_rows = len(row_counts)
    y_spacing = 0.8 / (num_rows - 1) if num_rows > 1 else 0

    for row_idx, count in enumerate(row_counts):
        y = 0.1 + row_idx * y_spacing
        if row_idx % 2 == 0:
            x_positions = np.linspace(0.1, 0.9, count)
        else:
            x_positions = np.linspace(0.15, 0.85, count)

        for x in x_positions:
            centers.append([x, y])

    centers = np.array(centers)
    radii = np.ones(n) * (0.065 + 0.015 * np.random.random())

    return centers, radii


def initialize_random(n, seed):
    """Initialize circles with random positions, avoiding overlaps."""
    np.random.seed(seed + 1000)
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.05

    for i in range(n):
        placed = False
        for attempt in range(1000):
            x = np.random.uniform(0.15, 0.85)
            y = np.random.uniform(0.15, 0.85)
            ok = True
            for j in range(i):
                dist = np.sqrt((x - centers[j, 0]) ** 2 + (y - centers[j, 1]) ** 2)
                if dist < radii[j] + radii[i] + 0.01:
                    ok = False
                    break
            if ok:
                centers[i] = [x, y]
                placed = True
                break
        if not placed:
            centers[i] = [np.random.uniform(0.1, 0.9), np.random.uniform(0.1, 0.9)]

    return centers, radii


def flatten(centers, radii):
    """Flatten centers and radii into a single 1D array."""
    n = len(radii)
    x = np.zeros(3 * n)
    for i in range(n):
        x[3 * i] = centers[i, 0]
        x[3 * i + 1] = centers[i, 1]
        x[3 * i + 2] = radii[i]
    return x


def unflatten(x, n):
    """Unflatten a 1D array into centers and radii arrays."""
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        centers[i, 0] = x[3 * i]
        centers[i, 1] = x[3 * i + 1]
        radii[i] = max(0.0, x[3 * i + 2])
    return centers, radii


def optimize_from(x0, n):
    """Run multi-phase penalty optimization from initial point x0."""
    x = x0.copy()
    penalties = [30, 100, 500, 2000, 10000, 50000, 100000, 200000]

    for penalty in penalties:
        result = minimize(
            partial(penalty_objective, n=n, penalty=penalty),
            x,
            method='L-BFGS-B',
            bounds=[(0, 1)] * (2 * n) + [(0, 0.5)] * n,
            options={
                'maxiter': 800,
                'ftol': 1e-15,
                'gtol': 1e-12,
            }
        )
        x = result.x

    return x


def verify_and_correct(centers, radii, n):
    """Verify packing validity and correct minor violations."""
    # Check and fix any small violations by reducing radii slightly
    max_iterations = 50
    for _ in range(max_iterations):
        max_violation = 0
        for i in range(n):
            x, y, r = centers[i, 0], centers[i, 1], radii[i]
            # Boundary checks
            if x - r < 0:
                max_violation = max(max_violation, r - x)
            if x + r > 1:
                max_violation = max(max_violation, x + r - 1)
            if y - r < 0:
                max_violation = max(max_violation, r - y)
            if y + r > 1:
                max_violation = max(max_violation, y + r - 1)
            for j in range(i + 1, n):
                dist = np.sqrt((x - centers[j, 0]) ** 2 + (y - centers[j, 1]) ** 2)
                overlap = radii[i] + radii[j] - dist
                if overlap > 0:
                    max_violation = max(max_violation, overlap)

        if max_violation < 1e-10:
            break

        # Reduce all radii slightly
        reduction = max_violation * 0.5 + 1e-8
        radii = np.maximum(radii - reduction * 0.01, 1e-10)

    return centers, radii


def run_packing():
