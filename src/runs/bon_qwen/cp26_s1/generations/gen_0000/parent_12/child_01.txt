import numpy as np
from scipy.optimize import minimize
import math


def generate_hexagonal_initial(n, trial):
    """Generate initial hexagonal packing configuration."""
    np.random.seed(trial * 13 + 7)
    
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.085
    
    row_configs = [
        [6, 5, 5, 5, 4, 1],
        [5, 5, 5, 5, 4, 2],
        [5, 5, 5, 4, 5, 2],
        [4, 5, 6, 5, 4, 2],
        [5, 6, 5, 5, 4, 1],
        [6, 6, 5, 5, 4],
        [5, 5, 6, 5, 5],
        [6, 5, 5, 5, 3, 2],
        [5, 5, 5, 5, 3, 3],
        [7, 5, 5, 5, 4],
    ]
    
    config = row_configs[trial % len(row_configs)]
    
    idx = 0
    y_spacing = 0.155 + (trial % 5) * 0.01
    
    for row_idx, count in enumerate(config):
        if idx >= n:
            break
        
        y = 0.09 + row_idx * y_spacing
        
        if row_idx % 2 == 1:
            offset = 0.075
        else:
            offset = 0.0
        
        x_spacing = 0.145
        x_start = (1.0 - (count - 1) * x_spacing) / 2.0
        
        for i in range(count):
            x = x_start + i * x_spacing + offset
            if idx < n:
                centers[idx, 0] = np.clip(x + np.random.randn() * 0.008, 0.05, 0.95)
                centers[idx, 1] = np.clip(y + np.random.randn() * 0.008, 0.05, 0.95)
                idx += 1
    
    return centers, radii


def compute_all_constraints(params, n):
    """Compute all inequality constraint values for the packing problem."""
    centers = params[:2 * n].reshape(n, 2)
    radii = params[2 * n:]
    
    num_boundary = 5 * n  # 4 boundary + 1 non-negativity per circle
    num_overlap = n * (n - 1) // 2
    all_constraints = np.zeros(num_boundary + num_overlap)
    
    idx = 0
    for i in range(n):
        x = centers[i, 0]
        y = centers[i, 1]
        r = radii[i]
        all_constraints[idx] = x - r
        idx += 1
        all_constraints[idx] = 1.0 - x - r
        idx += 1
        all_constraints[idx] = y - r
        idx += 1
        all_constraints[idx] = 1.0 - y - r
        idx += 1
        all_constraints[idx] = r
        idx += 1
    
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = math.sqrt(dx * dx + dy * dy + 1e-30)
            all_constraints[idx] = dist - radii[i] - radii[j]
            idx += 1
    
    return all_constraints


def objective_function(params, n):
    """Objective: minimize negative sum of radii."""
    radii = params[2 * n:]
    return -np.sum(radii)


def is_valid_packing(centers, radii, n):
    """Check if packing is valid."""
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if r < -1e-12:
            return False
        if x - r < -1e-10 or x + r > 1.0 + 1e-10:
            return False
        if y - r < -1e-10 or y + r > 1.0 + 1e-10:
            return False
    
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = math.sqrt(dx * dx + dy * dy)
            if dist < radii[i] + radii[j] - 1e-10:
                return False
    
    return True


def optimize_from_initial(centers_init, radii_init, n):
    """Run SLSQP optimization from given initial point."""
    params_init = np.concatenate([centers_init.flatten(), radii_init])
    
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
    
    constraint_dict = {'type': 'ineq', 'fun': lambda p: compute_all_constraints(p, n)}
    
    try:
        result = minimize(
            objective_function,
            params_init,
            args=(n,),
            method='SLSQP',
            bounds=bounds,
            constraints=constraint_dict,
            options={'maxiter': 2000, 'ftol': 1e-14, 'disp': False}
        )
        return result.x, result.success
    except Exception:
        return params_init, False


def refine_solution(centers, radii, n, max_iter=500):
    """Refine solution by iteratively scaling up radii and repositioning."""
    centers = centers.copy()
    radii = radii.copy()
    
    for iteration in range(max_iter):
        # Compute minimum margin
        min_margin = float('inf')
        
        # Check boundary margins
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            margin = min(x - r, 1.0 - x - r, y - r, 1.0 - y - r)
            min_margin = min(min_margin, margin)
        
        # Check overlap margins
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = math.sqrt(dx * dx + dy * dy + 1e-30)
                margin = dist - radii[i] - radii[j]
                min_margin = min(min_margin, margin)
        
        if min_margin < 1e-12:
            break
        
        # Scale up all radii slightly
        scale = 1.0 + min_margin * 0.1
        radii = radii * scale
        
        # Adjust centers to maintain validity
        for i in range(n):
            centers[i, 0] = np.clip(centers[i, 0], radii[i], 1.0 - radii[i])
            centers[i, 1] = np.clip(centers[i, 1], radii[i], 1.0 - radii[i])
    
    return centers, radii
