import numpy as np
from scipy.optimize import minimize, differential_evolution

def objective(params, n=26):
    # params contains x, y, and r for each circle
    # We want to maximize sum of radii, so minimize negative sum
    return -np.sum(params[2*n:3*n])

def boundary_constraints(params, n=26):
    x = params[0:n]
    y = params[n:2*n]
    r = params[2*n:3*n]
    
    # x - r >= 0
    c1 = x - r
    # x + r <= 1
    c2 = 1 - (x + r)
    # y - r >= 0
    c3 = y - r
    # y + r <= 1
    c4 = 1 - (y + r)
    
    return np.concatenate([c1, c2, c3, c4])

def circle_constraints(params, n=26):
    x = params[0:n]
    y = params[n:2*n]
    r = params[2*n:3*n]
    
    # Distance squared between centers >= (r_i + r_j)^2
    # We return (dist_sq - (r_i+r_j)^2) which should be >= 0
    # But scipy expects constraints to be >= 0 for 'ineq'
    
    # To improve performance, we only compute necessary pairs or use a mask if n is large, 
    # but for 26, n^2 is small enough.
    constraints = []
    for i in range(n):
        for j in range(i + 1, n):
            dist_sq = (x[i] - x[j])**2 + (y[i] - y[j])**2
            r_sum = r[i] + r[j]
            constraints.append(dist_sq - r_sum**2)
            
    return np.array(constraints)

def initial_layout(n=26):

    radii = np.zeros(n)

    r_est = 0.09

    row = 0
    while idx < n:

        # Staggered rows: odd rows have 1 less or shifted
        if row % 2 == 0:
            count = 6
        else:
            count = 5
            
        # Adjust count if we exceed n
        if idx + count > n:
            count = n - idx
            
        y_pos = r_est + row * (r_est * np.sqrt(3))

            centers *= scale

            # Re-calculate remaining circles if any? 
            # This is just a heuristic init, we break and optimize.
            break

    # Flatten for optimizer
    params = np.concatenate([centers[:, 0], centers[:, 1], radii])
    return params

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    # 1. Generate initial layout
    params0 = initial_layout(n)
    
    # 2. Define constraints for scipy

    bnd_constraints = {
        'type': 'ineq',
        'fun': boundary_constraints,
        'args': (n,)

    # Circle non-overlap constraints
    cir_constraints = {
        'type': 'ineq',
        'fun': circle_constraints,
        'args': (n,)
