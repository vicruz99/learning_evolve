3. Use penalty annealing (gradually increase penalty weight) to converge to feasible solutions
4. Multiple random restarts to escape local minima
5. Post-process to ensure validity (clip centers to valid ranges)
6. Vectorize the penalty computation for speed

The penalty function penalizes:
- Boundary violations (circle going outside [0,1])
- Overlap violations (distance between centers < sum of radii)

I'll use scipy.optimize.minimize with bounds on parameters and a carefully tuned penalty schedule.

import numpy as np
from scipy.optimize import minimize

def compute_penalty(centers, radii, penalty_weight):
    """Compute constraint violation penalty using vectorized operations"""
    n = len(radii)
    penalty = 0.0
    
    # Boundary constraints
    left = np.maximum(0, radii - centers[:, 0])
    right = np.maximum(0, centers[:, 0] + radii - 1)
    bottom = np.maximum(0, radii - centers[:, 1])
    top = np.maximum(0, centers[:, 1] + radii - 1)
    
    penalty += penalty_weight * np.sum(left**2 + right**2 + bottom**2 + top**2)
    
    # Overlap constraints - pairwise
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy + 1e-20)
            min_dist = radii[i] + radii[j]
            if dist < min_dist:
                penalty += penalty_weight * (min_dist - dist) ** 2
    
    return penalty

def objective(params, n=26, penalty_weight=10000):
    """Objective: minimize negative sum of radii + penalty for violations"""
    centers = params[:2 * n].reshape(n, 2)
    radii = np.maximum(params[2 * n:], 1e-10)
    
    penalty = compute_penalty(centers, radii, penalty_weight)
    
    return -np.sum(radii) + penalty

def hexagonal_init(n=26, jitter=0.02):
    """Initialize circles in hexagonal pattern with optional jitter"""
    rows = [5, 4, 5, 4, 5, 3]
    centers = []
    radii_list = []
    
    r = 0.09
    y_spacing = r * np.sqrt(3)
    
    for row_idx, count in enumerate(rows):
        x_start = r + (row_idx % 2) * r
        for col_idx in range(count):
            if len(centers) >= n:
                break
            x = x_start + col_idx * 2 * r
            y = r + row_idx * y_spacing
            centers.append([x, y])
            radii_list.append(r)
    
    centers = np.array(centers[:n])
    radii = np.array(radii_list[:n])
    
    if jitter > 0:
        centers += np.random.randn(n, 2) * jitter
        radii += np.random.randn(n) * jitter * 0.5
        radii = np.maximum(radii, 0.02)
    
    return centers, radii

def is_valid(centers, radii):
    """Validate packing constraints"""
    n = len(radii)
    for i in range(n):
        if radii[i] < 0:
            return False
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-9 or x + r > 1 + 1e-9 or y - r < -1e-9 or y + r > 1 + 1e-9:
            return False
    
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-9:
                return False
    
    return True

def optimize_packing(n=26, seed=0, num_restarts=30):
    """Main optimization routine with multiple restarts"""
    best_centers = None
    best_radii = None
    best_sum = 0
    
    for restart_idx in range(num_restarts):
        np.random.seed(seed * 1000 + restart_idx)
        
        centers, radii = hexagonal_init(n, jitter=0.015)
        
        x0 = np.concatenate([centers.flatten(), radii])
        
        # Penalty annealing schedule
        penalty_schedule = [500, 5000, 50000, 200000]
        
        for pw in penalty_schedule:
            result = minimize(
                objective,
                x0,
                args=(n, pw),
                method='L-BFGS-B',
                bounds=[(0.001, 0.999)] * (2 * n) + [(1e-8, 0.5)] * n,
                options={'maxiter': 30000, 'ftol': 1e-15, 'gtol': 1e-12}
            )
            x0 = result.x.copy()
        
        centers_opt = x0[:2 * n].reshape(n, 2)
        radii_opt = np.maximum(x0[2 * n:], 1e-10)
        
        # Post-process: ensure circles are within bounds
        radii_opt = np.clip(radii_opt, 1e-10, 0.5)
        centers_opt[:, 0] = np.clip(centers_opt[:, 0], radii_opt, 1 - radii_opt)
        centers_opt[:, 1] = np.clip(centers_opt[:, 1], radii_opt, 1 - radii_opt)
        
        # Validate
        if is_valid(centers_opt, radii_opt):
            s = np.sum(radii_opt)
            if s > best_sum:
                best_sum = s
                best_centers = centers_opt.copy()
                best_radii = radii_opt.copy()
    
    return best_centers, best_radii, best_sum

def run_packing():
    """Run the circle packing optimization and return results"""
    n = 26
    
    # Try multiple seeds for diversity
    global_best_centers = None
    global_best_radii = None
    global_best_sum = 0
    
    for base_seed in range(10):
        centers, radii, s = optimize_packing(n, seed=base_seed, num_restarts=15)
        if s > global_best_sum:
            global_best_sum = s
            global_best_centers = centers.copy()
            global_best_radii = radii.copy()
    
    return global_best_centers, global_best_radii, global_best_sum
