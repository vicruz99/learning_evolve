import numpy as np
from scipy.optimize import minimize


def boundary_x_lower(x, n, i):
    """Constraint: x_i >= r_i"""
    return x[2 * i] - x[2 * n + i]


def boundary_x_upper(x, n, i):
    """Constraint: 1 - x_i >= r_i"""
    return 1 - x[2 * i] - x[2 * n + i]


def boundary_y_lower(x, n, i):
    """Constraint: y_i >= r_i"""
    return x[2 * i + 1] - x[2 * n + i]


def boundary_y_upper(x, n, i):
    """Constraint: 1 - y_i >= r_i"""
    return 1 - x[2 * i + 1] - x[2 * n + i]


def overlap_constraint(x, n, i, j):
    """Constraint: ||c_i - c_j||^2 >= (r_i + r_j)^2"""
    dx = x[2 * i] - x[2 * j]
    dy = x[2 * i + 1] - x[2 * j + 1]
    dist_sq = dx * dx + dy * dy
    sum_r = x[2 * n + i] + x[2 * n + j]
    return dist_sq - sum_r * sum_r


def objective(x, n):
    """Negative sum of radii (minimization)"""
    return -np.sum(x[2 * n:])


def hexagonal_init(n, spacing_x, spacing_y, offset_x, offset_y):
    """Generate hexagonal grid initialization"""
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.09)
    idx = 0

    for row in range(8):
        for col in range(6):
            if idx >= n:
                break
            x = offset_x + col * spacing_x + (spacing_x / 2 if row % 2 == 1 else 0)
            y = offset_y + row * spacing_y
            if 0 <= x <= 1 and 0 <= y <= 1:
                centers[idx] = [x, y]
                idx += 1

    return centers[:n], radii[:n]


def random_init(n, seed):
    """Generate random initialization"""
    rng = np.random.RandomState(seed)
    centers = rng.uniform(0.12, 0.88, (n, 2))
    radii = rng.uniform(0.04, 0.10, n)
    return centers, radii


def force_relax(centers, radii, n, iterations=500):
    """Apply repulsive forces to resolve overlaps"""
    centers = centers.copy()
    radii = radii.copy()

    for it in range(iterations):
        forces = np.zeros((n, 2))

        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                if dist > 1e-10:
                    overlap = radii[i] + radii[j] - dist
                    if overlap > 0:
                        force_mag = overlap * 10.0 / (dist + 1e-10)
                        fx = force_mag * dx / (dist + 1e-10)
                        fy = force_mag * dy / (dist + 1e-10)
                        forces[i, 0] += fx
                        forces[i, 1] += fy
                        forces[j, 0] -= fx
                        forces[j, 1] -= fy

        # Boundary repulsion
        for i in range(n):
            if centers[i, 0] - radii[i] < 0:
                forces[i, 0] += (radii[i] - centers[i, 0]) * 20.0
            if centers[i, 0] + radii[i] > 1:
                forces[i, 0] -= (centers[i, 0] + radii[i] - 1) * 20.0
            if centers[i, 1] - radii[i] < 0:
                forces[i, 1] += (radii[i] - centers[i, 1]) * 20.0
            if centers[i, 1] + radii[i] > 1:
                forces[i, 1] -= (centers[i, 1] + radii[i] - 1) * 20.0

        step = 0.02 * (1.0 / (1.0 + it * 0.01))
        centers = centers + forces * step

        for i in range(n):
            centers[i, 0] = np.clip(centers[i, 0], radii[i] + 1e-6, 1 - radii[i] - 1e-6)
            centers[i, 1] = np.clip(centers[i, 1], radii[i] + 1e-6, 1 - radii[i] - 1e-6)

    return centers


def try_optimization(x0, n, constraints, bounds):
    """Run SLSQP optimization and return result"""
    try:
        result = minimize(
            objective,
            x0,
            args=(n,),
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 15000, 'ftol': 1e-15, 'disp': False}
        )
        return result
    except Exception:
        return None


def run_packing():
    n = 26

    # Build constraints
    constraints = []
    for i in range(n):
        constraints.append({'type': 'ineq', 'fun': boundary_x_lower, 'args': (n, i)})
        constraints.append({'type': 'ineq', 'fun': boundary_x_upper, 'args': (n, i)})
        constraints.append({'type': 'ineq', 'fun': boundary_y_lower, 'args': (n, i)})
        constraints.append({'type': 'ineq', 'fun': boundary_y_upper, 'args': (n, i)})

    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({'type': 'ineq', 'fun': overlap_constraint, 'args': (n, i, j)})

    bounds = [(0, 1)] * (2 * n) + [(0, 0.5)] * n

    best_state = None
    best_sum = -float('inf')

    # Strategy 1: Hexagonal grids with various spacings
    base_spacing = 0.185
    for mult in [0.9, 1.0, 1.05, 1.1, 1.15, 1.2]:
        spacing_x = base_spacing * mult
        spacing_y = spacing_x * np.sqrt(3) / 2

        # Try different offsets
        for ox_mult in [0.0, 0.1, 0.2]:
            for oy_mult in [0.0, 0.1, 0.2]:
                offset_x = 0.05 + ox_mult * 0.1
                offset_y = 0.05 + oy_mult * 0.1

                centers, radii = hexagonal_init(n, spacing_x, spacing_y, offset_x, offset_y)

                # Force relax
                centers = force_relax(centers, radii, n, iterations=200)

                x0 = np.concatenate([centers.flatten(), radii])
                result = try_optimization(x0, n, constraints, bounds)

                if result is not None and result.success:
                    sum_r = np.sum(result.x[2 * n:])
                    if sum_r > best_sum:
                        best_sum = sum_r
                        best_state = result.x.copy()

    # Strategy 2: Random initializations with force relaxation

        centers, radii = random_init(n, seed)

        x0 = np.concatenate([centers.flatten(), radii])
        result = try_optimization(x0, n, constraints, bounds)

        idx = 0
        spacing_x = 2 * target_r
        spacing_y = spacing_x * np.sqrt(3) / 2

        for row in range(7):
            for col in range(6):
                if idx >= n:

                y = (row + 0.5) * spacing_y

                    idx += 1

        radii = np.full(n, target_r)
        centers = force_relax(centers, radii, n, iterations=400)
        x0 = np.concatenate([centers.flatten(), radii])
        result = try_optimization(x0, n, constraints, bounds)

        if result is not None and result.success:

            if sum_r > best_sum:
                best_sum = sum_r
                best_state = result.x.copy()

    # Extract final solution
    centers = best_state[:2 * n].reshape(n, 2)
    radii = best_state[2 * n:]
    radii = np.maximum(radii, 0)

    # Final validation fix: ensure all constraints satisfied
    for i in range(n):
        centers[i, 0] = np.clip(centers[i, 0], radii[i], 1 - radii[i])
        centers[i, 1] = np.clip(centers[i, 1], radii[i], 1 - radii[i])

    return centers, radii, np.sum(radii)
