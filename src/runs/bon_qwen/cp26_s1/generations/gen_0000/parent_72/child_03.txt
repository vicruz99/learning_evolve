import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def objective(vars):
    # We want to maximize sum of radii, so minimize negative sum
    r = vars[2::3]
    return -np.sum(r)

def constraints(vars):
    n = N_CIRCLES
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    
    c = []
    # Boundary constraints: circles must be inside [0,1]x[0,1]
    for i in range(n):
        c.append(x[i] - r[i])           # left edge >= 0
        c.append(1.0 - x[i] - r[i])     # right edge <= 1
        c.append(y[i] - r[i])           # bottom edge >= 0
        c.append(1.0 - y[i] - r[i])     # top edge <= 1
        
    # Overlap constraints: distance between centers >= sum of radii
    for i in range(n):
        for j in range(i + 1, n):
            dx = x[i] - x[j]
            dy = y[i] - y[j]
            c.append(dx*dx + dy*dy - (r[i] + r[j])**2)
            
    return np.array(c)

def run_packing():
    n = N_CIRCLES
    
    # Generate a good initial configuration using a hexagonal lattice pattern
    init_vars = np.zeros(3 * n)
    pts = []
    
    # Create a roughly hexagonal arrangement
    for row in range(6):
        y = 0.1 + row * 0.16
        num_cols = 5 if row % 2 == 0 else 4
        start_x = 0.1 if row % 2 == 0 else 0.2
        for col in range(num_cols):
            if len(pts) >= n:
                break
            pts.append([start_x + col * 0.2, y])
            
    # Fill remaining spots if any
    while len(pts) < n:
        pts.append([0.5, 0.5])
        
    for i in range(n):
        init_vars[3*i] = pts[i][0]
        init_vars[3*i+1] = pts[i][1]
        init_vars[3*i+2] = 0.095  # Start slightly below expected optimal to ensure feasibility
        
    bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n
    cons = {'type': 'ineq', 'fun': constraints}
    
    best_vars = init_vars.copy()
    best_sum = np.sum(init_vars[2::3])
    
    # Primary optimization
    res = minimize(objective, init_vars, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': 3000, 'ftol': 1e-9, 'disp': False})
    
    if not np.isnan(res.fun) and -res.fun > best_sum:
        best_vars = res.x
        best_sum = -res.fun
        
    # Secondary optimization with perturbation to escape local minima
    np.random.seed(42)
    for _ in range(3):
        pert = best_vars.copy()
        pert[:2*n] += np.random.uniform(-0.02, 0.02, 2*n)
        # Clip positions to stay within bounds
        pert[:2*n] = np.clip(pert[:2*n], 0.01, 0.99)
        pert[2*n:] = best_sum / n  # reset radii to average to allow re-expansion
        
        res2 = minimize(objective, pert, method='SLSQP', bounds=bounds, constraints=cons,
                        options={'maxiter': 2000, 'ftol': 1e-9, 'disp': False})
        
        if not np.isnan(res2.fun) and -res2.fun > best_sum:
            best_vars = res2.x
            best_sum = -res2.fun
            
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        centers[i] = [best_vars[3*i], best_vars[3*i+1]]
        radii[i] = best_vars[3*i+2]
        
    # Ensure strict non-negativity and validity against numerical noise
    radii = np.maximum(radii, 0.0)
    
    return centers, radii, float(best_sum)
