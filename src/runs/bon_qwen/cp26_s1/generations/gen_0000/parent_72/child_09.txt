def run_packing():

    Packs 26 circles in a unit square to maximize the sum of radii.

    centers = np.zeros((n, 2))

            # Slightly compressed grid to allow initial movement

            idx += 1

    radii = np.full(n, 0.04)
    
    # 2. Simulation Parameters

    k_wall = 60.0       # Repulsion strength from walls
    dt = 0.004          # Time step for position updates
    growth_rate = 0.00006 # Rate of radius expansion
    max_iter = 40000    # Total simulation steps
    relax_threshold = 0.008 # Max total overlap allowed before growing
    
    # Precompute pairwise indices for efficiency
    i_idx, j_idx = np.triu_indices(n, 1)
    
    # 3. Main Simulation Loop
    for step in range(max_iter):
        # Calculate pairwise distances and overlaps
        diff = centers[i_idx] - centers[j_idx]
        dists = np.linalg.norm(diff, axis=1)
        sum_r = radii[i_idx] + radii[j_idx]
        overlaps = sum_r - dists
        
        total_overlap = np.sum(np.maximum(overlaps, 0.0))

        if total_overlap < relax_threshold:
            radii += growth_rate
            
        # Compute Forces
        forces = np.zeros_like(centers)
        
        # Pairwise repulsion forces
        mask_overlap = overlaps > 0
        if np.any(mask_overlap):
            # Avoid division by zero
            dists_safe = np.maximum(dists, 1e-9)
            dirs = diff / dists_safe[:, None]
            force_mag = overlaps * k_repulse
            
            # Accumulate forces using np.add.at for correct handling of repeated indices
            np.add.at(forces, i_idx[mask_overlap], dirs[mask_overlap] * force_mag[mask_overlap][:, None])
            np.add.at(forces, j_idx[mask_overlap], -dirs[mask_overlap] * force_mag[mask_overlap][:, None])
            
        # Wall constraint forces
        # Left boundary
        ol = radii - centers[:, 0]
        mask_l = ol > 0
        forces[mask_l, 0] += ol[mask_l] * k_wall
        
        # Right boundary
        or_ = centers[:, 0] + radii - 1.0
        mask_r = or_ > 0
        forces[mask_r, 0] -= or_[mask_r] * k_wall
        
        # Bottom boundary
        ob = radii - centers[:, 1]
        mask_b = ob > 0
        forces[mask_b, 1] += ob[mask_b] * k_wall

        ot = centers[:, 1] + radii - 1.0
        mask_t = ot > 0
        forces[mask_t, 1] -= ot[mask_t] * k_wall
        
        # Update positions
        centers += forces * dt
        
        # Strictly enforce boundaries to prevent numerical drift
        centers = np.clip(centers, 0.0, 1.0)
        
        # Gradually reduce growth rate to fine-tune packing near convergence
        if step > 0 and step % 10000 == 0:
            growth_rate *= 0.85
            
    # 4. Return Results
    return centers, radii, np.sum(radii)
