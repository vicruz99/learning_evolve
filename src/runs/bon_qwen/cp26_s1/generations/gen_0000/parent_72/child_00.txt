    *   **Gradients**: Provide analytical gradients (Jacobian) for the objective and constraints to improve solver efficiency and convergence.

import numpy as np
from scipy.optimize import minimize

def run_packing():
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    Uses SLSQP optimization with a hexagonal lattice initialization.
    """
    n = 26
    num_vars = n * 3  # x, y, r for each circle

    # --- Initialization ---
    # Create a hexagonal grid pattern for centers
    # Pattern: 6, 5, 6, 5, 4 circles in rows
    row_counts = [6, 5, 6, 5, 4]
    centers_init = []
    
    # Define grid parameters
    # We want to fit these in [0,1]x[0,1]
    # Let's space rows vertically with spacing dy
    # And columns horizontally with spacing dx
    # Hexagonal packing suggests dx = 2r, dy = sqrt(3)r. 
    # Since r is variable, we just pick a geometric layout.
    
    # Let's distribute rows evenly in y
    # 5 rows -> 4 intervals. 
    # To allow margin, let's place centers in [0.1, 0.9] roughly, or just [0,1]
    # The optimizer will handle boundary constraints.
    
    y_coords = np.linspace(0.1, 0.9, 5) 
    
    # For each row, place x coordinates
    for idx, count in enumerate(row_counts):
        y = y_coords[idx]
        # Stagger rows for hexagonal packing
        # Row 0 (even index 0): start at 1/12? 
        # Let's use a spacing that fits count circles.
        # If count=6, we need 5 gaps.
        # If we want them spread from 0 to 1, gap = 1/5? 
        # But we need margins for radius.
        # Let's just place them linearly.
        
        # To create hexagonal shift:
        # Even rows (0, 2) aligned one way, Odd rows (1, 3) shifted.
        # Let's define a base spacing.
        
        # Simple heuristic: spread them in [0, 1]
        # Shift odd rows by half a unit of spacing?
        
        # Let's just generate points.
        # Row 0: 6 points
        if count == 6:
            xs = np.linspace(1/12, 11/12, 6) # 1/12, 3/12, ..., 11/12
        elif count == 5:
            # Shifted relative to 6-point row?
            # 6-point row has points at 1, 3, 5, 7, 9, 11 (twelfths)
            # 5-point row should be at 2, 4, 6, 8, 10?
            xs = np.linspace(2/12, 10/12, 5) # 2/12, 4/12, ..., 10/12
        elif count == 4:
            # 4 points, maybe centered?
            xs = np.linspace(3/12, 9/12, 4) # 3/12, 5/12, 7/12, 9/12
            
        for x in xs:
            centers_init.append([x, y])
            
    centers_init = np.array(centers_init)
    
    # Initial radii: small value to ensure feasibility
    # Distance between closest neighbors in this grid is approx 1/6 ~ 0.166
    # So r <= 0.08 is safe.
    radii_init = np.full(n, 0.01)
    
    # Flatten to vector [x0, y0, r0, x1, y1, r1, ...]
    x0 = np.zeros(num_vars)
    for i in range(n):
        x0[3*i] = centers_init[i, 0]
        x0[3*i+1] = centers_init[i, 1]
        x0[3*i+2] = radii_init[i]

    # --- Optimization Setup ---
    
    def objective(vars_vec):
        # Maximize sum of radii => Minimize negative sum
        radii = vars_vec[2::3]
        return -np.sum(radii)

    def objective_jac(vars_vec):
        jac = np.zeros(num_vars)
        jac[2::3] = -1.0
        return jac

    # Constraints list
    constraints = []
    
    # 1. Boundary Constraints
    # For each circle i:
    # x_i - r_i >= 0
    # 1 - x_i - r_i >= 0
    # y_i - r_i >= 0
    # 1 - y_i - r_i >= 0
    
    for i in range(n):
        idx = 3 * i
        
        # x - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: v[idx] - v[idx+2],
            'jac': lambda v, i=i: np.array([1.0 if k == idx else -1.0 if k == idx+2 else 0.0 for k in range(num_vars)])
        })
        
        # 1 - x - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: 1.0 - v[idx] - v[idx+2],
            'jac': lambda v, i=i: np.array([-1.0 if k == idx else -1.0 if k == idx+2 else 0.0 for k in range(num_vars)])
        })
        
        # y - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: v[idx+1] - v[idx+2],
            'jac': lambda v, i=i: np.array([1.0 if k == idx+1 else -1.0 if k == idx+2 else 0.0 for k in range(num_vars)])
        })
        
        # 1 - y - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: 1.0 - v[idx+1] - v[idx+2],
            'jac': lambda v, i=i: np.array([-1.0 if k == idx+1 else -1.0 if k == idx+2 else 0.0 for k in range(num_vars)])
        })

    # 2. Non-overlap Constraints
    # For each pair (i, j) with i < j:
    # sqrt((xi-xj)^2 + (yi-yj)^2) - (ri + rj) >= 0
    
    # Precompute indices for efficiency if needed, but n=26 is small enough.
    # Number of pairs = 26*25/2 = 325.
    
    pairs = []
    for i in range(n):
        for j in range(i + 1, n):
            pairs.append((i, j))
            
    def non_overlap_fun(vars_vec, i, j):
        xi, yi, ri = vars_vec[3*i], vars_vec[3*i+1], vars_vec[3*i+2]
        xj, yj, rj = vars_vec[3*j], vars_vec[3*j+1], vars_vec[3*j+2]
        dist = np.sqrt((xi - xj)**2 + (yi - yj)**2)
        return dist - (ri + rj)

    def non_overlap_jac(vars_vec, i, j):
        xi, yi = vars_vec[3*i], vars_vec[3*i+1]
        xj, yj = vars_vec[3*j], vars_vec[3*j+1]
        
        diff_x = xi - xj
        diff_y = yi - yj
        dist = np.sqrt(diff_x**2 + diff_y**2)
        
        # Avoid division by zero
        if dist < 1e-12:
            dist = 1e-12
            
        dx_dist = diff_x / dist
        dy_dist = diff_y / dist
        
        jac = np.zeros(num_vars)
        
        # Gradient w.r.t xi
        jac[3*i] = dx_dist
        # Gradient w.r.t yi
        jac[3*i+1] = dy_dist
        # Gradient w.r.t ri
        jac[3*i+2] = -1.0
        
        # Gradient w.r.t xj
        jac[3*j] = -dx_dist
        # Gradient w.r.t yj
        jac[3*j+1] = -dy_dist
        # Gradient w.r.t rj
        jac[3*j+2] = -1.0
        
        return jac

    for i, j in pairs:
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i, j=j: non_overlap_fun(v, i, j),
            'jac': lambda v, i=i, j=j: non_overlap_jac(v, i, j)
        })

    # Bounds: x, y in [0, 1], r in [0, 0.5]
    bounds = [(0, 1)] * num_vars
    # Radii bounds
    for i in range(n):
        bounds[3*i+2] = (0, 0.5)

    # Run optimization
    # SLSQP is good for constrained problems
    res = minimize(
        objective, 
        x0, 
        method='SLSQP', 
        jac=objective_jac, 
        bounds=bounds, 
        constraints=constraints, 
        options={'ftol': 1e-9, 'maxiter': 1000, 'disp': False}
    )

    # Extract results
    best_vars = res.x
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    for i in range(n):
        centers[i, 0] = best_vars[3*i]
        centers[i, 1] = best_vars[3*i+1]
        radii[i] = best_vars[3*i+2]
        
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii

if __name__ == "__main__":
    centers, radii, sum_radii = run_packing()
    print(f"Sum of radii: {sum_radii}")
