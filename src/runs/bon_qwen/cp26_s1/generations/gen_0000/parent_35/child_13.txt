def generate_hexagonal_init(n):
    """Generate initial positions for n circles in a hexagonal pattern inside [0,1]x[0,1]."""
    points = []
    # Approximate spacing to fit n circles. 
    # Density of hex packing is ~0.9069. Area per circle ~ 1/n.
    # pi * r^2 ~ 1/(n * density) => r ~ sqrt(1 / (n * pi * density))
    # Diameter d = 2r. Spacing s = d.
    # Let's estimate a safe spacing.
    # For n=26, r ~ 0.1, d ~ 0.2.
    s = 0.2 
    
    rows = 0
    while len(points) < n:
        for col in range(int(np.ceil(1/s)) + 1):
            x = col * s + 0.1 # Start a bit inside to respect boundary
            y = rows * s * np.sqrt(3)/2 + 0.1
            
            # Offset every other row
            if rows % 2 == 1:
                x += s / 2.0
            
            if 0 <= x <= 1 and 0 <= y <= 1:
                points.append([x, y])
            
            if len(points) >= n:
                break
        rows += 1
    
    # Return first n points
    return np.array(points[:n])
