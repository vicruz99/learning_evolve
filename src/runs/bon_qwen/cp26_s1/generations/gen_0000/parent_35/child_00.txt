import numpy as np
import scipy.optimize as opt

def generate_hexagonal_layout(n_circles):
    """
    Generates an initial hexagonal packing layout for n_circles in a unit square.
    """
    centers = []
    radii = []
    
    # Estimate a reasonable radius for hex packing
    # For n=26, we can try rows like 5, 6, 5, 6, 4 or similar
    # Let's try to fit as many as possible in a dense hex grid
    
    # Approximate radius for 26 circles in hex packing
    # Density argument: Area ~ 1, sum of areas ~ 26 * pi * r^2 ~ 0.85
    # r ~ sqrt(0.85 / (26 * pi)) ~ 0.101
    # But boundary effects reduce this. Let's start with r=0.09
    
    r_init = 0.085 
    
    # Hexagonal spacing
    dx = 2 * r_init
    dy = np.sqrt(3) * r_init
    
    row = 0
    count = 0
    
    # We need to fit 26 circles. 
    # Let's build rows.
    # Width constraint: 1.0. Height constraint: 1.0.
    
    while count < n_circles:
        y = r_init + row * dy
        
        # If we exceed height, break (shouldn't happen if r is small enough)
        if y + r_init > 1.0 + 1e-9:
            break
            
        # Determine shift for this row
        # Even rows: 0, 2r, 4r...
        # Odd rows: r, 3r, 5r... (shifted by r)

        else:

            # Center to center distance 2r.
            # Row 0: x = r, 3r, 5r...
            # Row 1: x = 2r, 4r, 6r...
            # So shift is r.
            # Wait, if Row 0 starts at r, Row 1 should start at 2r?
            # Distance between (r, 0) and (2r, sqrt(3)r) is sqrt(r^2 + 3r^2) = 2r. Correct.

                radii.append(r_init)
                count += 1
                x += dx
            else:
                break
        row += 1
        
    return np.array(centers), np.array(radii)

def get_constraints(centers, radii, n):
    """
    Returns the constraint functions for scipy.optimize.
    Constraints:
    1. Circle i inside square: x_i - r_i >= 0, x_i + r_i <= 1, etc.
    2. Non-overlap: dist(i,j) >= r_i + r_j
    """
    constraints = []
    
    # Boundary constraints
    for i in range(n):
        # x_i >= r_i  => x_i - r_i >= 0
        # Variable indices: 3*i (x), 3*i+1 (y), 3*i+2 (r)
        def con_x_min(x_all, i=i):
            return x_all[3*i] - x_all[3*i+2]
        
        def con_x_max(x_all, i=i):
            return 1.0 - (x_all[3*i] + x_all[3*i+2])
            
        def con_y_min(x_all, i=i):
            return x_all[3*i+1] - x_all[3*i+2]
            
        def con_y_max(x_all, i=i):
            return 1.0 - (x_all[3*i+1] + x_all[3*i+2])
            
        constraints.append({'type': 'ineq', 'fun': con_x_min})
        constraints.append({'type': 'ineq', 'fun': con_x_max})
        constraints.append({'type': 'ineq', 'fun': con_y_min})
        constraints.append({'type': 'ineq', 'fun': con_y_max})

    # Non-overlap constraints
    # dist(i,j)^2 >= (r_i + r_j)^2
    for i in range(n):
        for j in range(i + 1, n):
            def con_dist(x_all, i=i, j=j):
                xi, yi, ri = x_all[3*i], x_all[3*i+1], x_all[3*i+2]
                xj, yj, rj = x_all[3*j], x_all[3*j+1], x_all[3*j+2]
                dist_sq = (xi - xj)**2 + (yi - yj)**2
                sum_r = ri + rj
                return dist_sq - sum_r**2
            
            constraints.append({'type': 'ineq', 'fun': con_dist})
            
    return constraints

def objective_func(x_all):
    """
    Objective: Maximize sum of radii => Minimize negative sum of radii.
    """
    # x_all is a flattened array [x1, y1, r1, x2, y2, r2, ...]
    n = len(x_all) // 3
    radii_sum = 0
    for i in range(n):
        radii_sum += x_all[3*i + 2]
    return -radii_sum
