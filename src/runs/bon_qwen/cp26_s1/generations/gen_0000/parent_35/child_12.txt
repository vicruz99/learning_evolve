import numpy as np
from scipy.optimize import minimize
import warnings
warnings.filterwarnings('ignore')

N_CIRCLES = 26

def eval_constraints(vars):
    """
    Evaluates all inequality constraints >= 0
    vars: array of shape (3*N,) ordered as [x0, y0, r0, x1, y1, r1, ...]
    """
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    cons = []
    
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    cons.extend(x - r)
    cons.extend(1.0 - x - r)
    cons.extend(y - r)
    cons.extend(1.0 - y - r)
    
    # Positive radius constraint
    cons.extend(r - 1e-5)
    
    # Non-overlap constraints: dist^2 >= (r_i + r_j)^2
    for i in range(N_CIRCLES):
        xi, yi, ri = x[i], y[i], r[i]
        for j in range(i + 1, N_CIRCLES):
            dx = xi - x[j]
            dy = yi - y[j]
            dr = ri + r[j]
            cons.append(dx*dx + dy*dy - dr*dr)
            
    return np.array(cons)

def objective(vars):
    """Maximize sum of radii -> minimize negative sum"""
    return -np.sum(vars[2::3])

def get_grid_init(r_init):
    """Structured 5x5 grid + 1 offset circle initialization"""
    pts = []
    y = r_init
    for i in range(5):
        x = r_init
        for j in range(5):
            pts.append([x, y, r_init])
            x += 2.0 * r_init
        y += 2.0 * r_init
    # 26th circle placed slightly off-center to avoid exact overlap
    pts.append([0.4, 0.6, r_init])
    return np.array(pts).flatten()

def get_rand_init(seed, r_init):
    """Random initialization within safe bounds"""
    np.random.seed(seed)
    pts = []
    for _ in range(N_CIRCLES):
        pts.append([
            np.random.uniform(r_init, 1.0 - r_init),
            np.random.uniform(r_init, 1.0 - r_init),
            r_init
        ])
    return np.array(pts).flatten()

def run_packing():
    """
    Optimizes circle packing and returns (centers, radii, sum_radii)
    """
    # Bounds for x, y in [0,1] and r in [1e-5, 0.5]
    bounds = [(0.0, 1.0)] * N_CIRCLES + [(0.0, 1.0)] * N_CIRCLES + [(1e-5, 0.5)] * N_CIRCLES
    constraint = {'type': 'ineq', 'fun': eval_constraints}
    
    best_sum = -np.inf
    best_vars = None
    
    # Multiple restarts to find global optimum
    initial_configs = [
        get_grid_init(0.09),
        get_grid_init(0.10),
        get_rand_init(42, 0.09),
        get_rand_init(123, 0.085),
        get_rand_init(2024, 0.095)
    ]
    
    for init_vars in initial_configs:
        # Ensure strict feasibility at start
        init_vars[:N_CIRCLES] = np.clip(init_vars[:N_CIRCLES], 1e-5, 1.0 - 1e-5)
        init_vars[N_CIRCLES:2*N_CIRCLES] = np.clip(init_vars[N_CIRCLES:2*N_CIRCLES], 1e-5, 1.0 - 1e-5)
        init_vars[2*N_CIRCLES:] = np.clip(init_vars[2*N_CIRCLES:], 1e-5, 0.5)
        
        res = minimize(
            objective, 
            init_vars, 
            method='SLSQP', 
            bounds=bounds,
            constraints=constraint, 
            options={'maxiter': 2500, 'ftol': 1e-12}
        )
        
        curr_sum = -res.fun
        if curr_sum > best_sum:
            best_sum = curr_sum
            best_vars = res.x
            
    # Extract and format results
    centers = np.column_stack((best_vars[0::3], best_vars[1::3]))
    radii = best_vars[2::3]
    total_sum = float(np.sum(radii))
    
    return centers, radii, total_sum
