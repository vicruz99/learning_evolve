import numpy as np
from scipy.optimize import minimize


def objective(params, n):
    """Negative sum of radii (minimize negative to maximize sum)"""
    return -np.sum(params[2 * n:])


def constraint_boundary(params, n):
    """Boundary constraints: circles must be inside unit square"""
    centers = params[:2 * n].reshape(n, 2)
    radii = params[2 * n:]
    c = []
    for i in range(n):
        c.append(centers[i, 0] - radii[i])
        c.append(1.0 - centers[i, 0] - radii[i])
        c.append(centers[i, 1] - radii[i])
        c.append(1.0 - centers[i, 1] - radii[i])
    return np.array(c)


def constraint_overlap(params, n):
    """Non-overlap constraints: distance >= sum of radii"""
    centers = params[:2 * n].reshape(n, 2)
    radii = params[2 * n:]
    c = []
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy + 1e-12)
            c.append(dist - radii[i] - radii[j])
    return np.array(c)


def hex_grid(n, scale=1.0):
    """Generate hexagonal lattice positions"""
    centers = np.zeros((n, 2))
    idx = 0
    y_start = 0.07 * scale
    x_start = 0.07 * scale
    y_step = 0.14 * scale
    x_step = 0.16 * scale
    
    y = y_start
    row = 0
    while idx < n and y < 1 - y_start:
        x = x_start
        offset = (row % 2) * x_step * 0.866
        while x < 1 - x_start and idx < n:
            centers[idx] = [x + offset, y]
            x += x_step
            idx += 1
        y += y_step
        row += 1
    
    while idx < n:
        centers[idx] = [0.5, 0.5]
        idx += 1
    return centers


def grid_layout(n):
    """Generate rectangular grid positions"""
    centers = np.zeros((n, 2))
    idx = 0
    rows = 6
    cols = 5
    for i in range(rows):
        for j in range(cols):
            if idx < n:
                centers[idx] = [0.1 + j * 0.2, 0.08 + i * 0.16]
                idx += 1
    while idx < n:
        centers[idx] = [0.5, 0.5]
        idx += 1
    return centers


def run_packing():
    n = 26
    best_sum = 0.0
    best_centers = None

    for trial in range(60):

        if trial < 25:

            centers = hex_grid(n, scale)
        elif trial < 40:
            centers = grid_layout(n)
        else:

        # Add small random perturbation

        radii = np.full(n, 0.03 + 0.02 * np.random.random())
        
        # Flatten parameters
        params = np.concatenate([centers.flatten(), radii])
        
        # Bounds
        bounds = [(0.01, 0.99)] * (2 * n) + [(0.01, 0.5)] * n
        
        # Constraints
        cons = [
            {'type': 'ineq', 'fun': lambda p, nn=n: constraint_boundary(p, nn)},
            {'type': 'ineq', 'fun': lambda p, nn=n: constraint_overlap(p, nn)}
        ]
        
        try:
            result = minimize(
                objective, params, args=(n,),
                method='SLSQP', bounds=bounds,
                constraints=cons,
                options={'maxiter': 8000, 'ftol': 1e-13, 'disp': False}
