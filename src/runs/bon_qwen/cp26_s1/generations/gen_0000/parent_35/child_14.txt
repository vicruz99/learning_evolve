def hexagonal_init(n, row_counts=None):
    """Initialize n circles in a hexagonal pattern"""
    if row_counts is None:
        row_counts = [5, 5, 5, 5, 5, 1]
    
    r = 0.09
    centers = []
    
    num_rows = len(row_counts)
    y_spacing = (1 - 2 * r) / max(num_rows - 1, 1)
    
    for row_idx, ncols in enumerate(row_counts):
        y = r + row_idx * y_spacing
        x_start = r + (row_idx % 2) * r
        
        for col_idx in range(ncols):
            if len(centers) >= n:
                break
            x = x_start + col_idx * 2 * r
            if x + r <= 1 + 1e-10:
                centers.append([x, y])
    
    centers = np.array(centers[:n])
    radii = np.ones(n) * r
    return centers, radii


def row_config_init(n, row_counts):

def compute_constraints_for_slsqp(n):
    """Precompute constraint functions for SLSQP"""
    cons = []
    
    for i in range(n):
        cons.append({'type': 'ineq', 'fun': lambda x, i=i: x[2*i] - x[2*n+i]})
        cons.append({'type': 'ineq', 'fun': lambda x, i=i: 1.0 - x[2*i] - x[2*n+i]})
        cons.append({'type': 'ineq', 'fun': lambda x, i=i: x[2*i+1] - x[2*n+i]})
        cons.append({'type': 'ineq', 'fun': lambda x, i=i: 1.0 - x[2*i+1] - x[2*n+i]})
    
    for i in range(n):
        for j in range(i + 1, n):
            cons.append({
                'type': 'ineq',
                'fun': lambda x, i=i, j=j: math.sqrt((x[2*i]-x[2*j])**2 + (x[2*i+1]-x[2*j+1])**2) - x[2*n+i] - x[2*n+j]
            })
    
    return cons


def slsqp_optimize(initial_centers, initial_radii, n, maxiter=2000):
    """Optimize using SLSQP"""
    x0 = np.concatenate([initial_centers.flatten(), initial_radii])
    
    def objective(x):
        return -np.sum(x[2*n:])
    
    cons = compute_constraints_for_slsqp(n)
    bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n
    
    result = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-12, 'disp': False})
    
    centers_opt = result.x[:2*n].reshape(n, 2)
    radii_opt = result.x[2*n:]
    return centers_opt, radii_opt


def penalty_optimize(initial_centers, initial_radii, n, max_iter=500):
    """Optimize using penalty method with Nelder-Mead"""
    x0 = np.concatenate([initial_centers.flatten(), initial_radii])
    penalty = 1e6
    
    def objective(x):
        centers = x[:2*n].reshape(n, 2)
        radii = x[2*n:]
        
        obj = -np.sum(radii)
        
        for i in range(n):
            obj += penalty * max(0, radii[i] - centers[i, 0])
            obj += penalty * max(0, centers[i, 0] + radii[i] - 1.0)
            obj += penalty * max(0, radii[i] - centers[i, 1])
            obj += penalty * max(0, centers[i, 1] + radii[i] - 1.0)
        
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt((centers[i,0]-centers[j,0])**2 + (centers[i,1]-centers[j,1])**2)
                overlap = radii[i] + radii[j] - dist
                if overlap > 0:
                    obj += penalty * overlap
        
        return obj
    
    result = minimize(objective, x0, method='Nelder-Mead', 
                      options={'maxiter': max_iter, 'xatol': 1e-10, 'fatol': 1e-10})
    
    centers_opt = result.x[:2*n].reshape(n, 2)
    radii_opt = result.x[2*n:]
    return centers_opt, radii_opt

def validate_packing(centers, radii):
    """Validate packing (simplified version)"""
    n = centers.shape[0]
    
    if np.isnan(centers).any() or np.isnan(radii).any():
        return False
    
    for i in range(n):
        if radii[i] < 0:
            return False
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12:
            return False
        if y - r < -1e-12 or y + r > 1 + 1e-12:
            return False
    
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False
    
    return True


def fix_packing(centers, radii, n):

        r = min(radii[i], centers[i, 0], 1 - centers[i, 0],

        # Reduce radii slightly to fix overlaps
