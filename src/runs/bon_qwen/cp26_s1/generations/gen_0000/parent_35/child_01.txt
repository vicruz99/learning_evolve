from scipy.optimize import minimize

def constraint_func(params):
    n = 26
    c = params.reshape(n, 3)[:, :2]
    r = params.reshape(n, 3)[:, 2]
    
    # Boundary and non-negativity constraints
    vals = np.concatenate([
        c[:, 0] - r,
        1.0 - c[:, 0] - r,
        c[:, 1] - r,
        1.0 - c[:, 1] - r,
        r
    ])
    
    # Pairwise non-overlap constraints using broadcasting
    diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
    dist_sq = np.sum(diff**2, axis=2)
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    overlap_vals = dist_sq - r_sum**2
    
    idx = np.triu_indices(n, k=1)
    vals = np.concatenate([vals, overlap_vals[idx]])
    return vals

def objective_func(params):

    r = params.reshape(n, 3)[:, 2]
    return -np.sum(r)
