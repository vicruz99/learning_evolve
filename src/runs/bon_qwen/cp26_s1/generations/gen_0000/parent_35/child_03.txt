import numpy as np
from scipy.optimize import minimize

def compute_penalty(centers, radii):
    """Compute squared constraint violations for packing."""
    n = centers.shape[0]
    p = 0.0
    
    # Boundary constraints: circle i must be within [0,1]
    # x - r >= 0  => r - x <= 0
    # x + r <= 1  => x - (1-r) <= 0
    p += np.sum(np.maximum(0.0, radii - centers[:,0])**2)
    p += np.sum(np.maximum(0.0, radii - (1.0 - centers[:,0]))**2)
    p += np.sum(np.maximum(0.0, radii - centers[:,1])**2)
    p += np.sum(np.maximum(0.0, radii - (1.0 - centers[:,1]))**2)
    
    # Overlap constraints: dist(i,j) >= r_i + r_j
    idx_i, idx_j = np.triu_indices(n, k=1)
    diffs = centers[idx_i] - centers[idx_j]
    dists = np.sqrt(np.sum(diffs**2, axis=1))
    p += np.sum(np.maximum(0.0, radii[idx_i] + radii[idx_j] - dists)**2)
    
    return p

def objective_centers(x, n, radii):
    """Objective for center optimization during growth phase."""
    c = x.reshape(n, 2)
    return compute_penalty(c, radii)

def objective_final(x, n):
    """Objective for final phase: maximize sum radii while minimizing penalty."""
    c = x[:2*n].reshape(n, 2)
    r = np.maximum(x[2*n:], 0.0)
    p = compute_penalty(c, r)
    # Weight chosen to prioritize validity but encourage radius growth
    return p - 6.0 * np.sum(r)

def run_packing():
    n = 26
    np.random.seed(42)
    
    # 1. Initialization: Hexagonal lattice
    centers = np.zeros((n, 2))
    idx = 0
    sp = 0.22  # Initial spacing
    y_sp = sp * np.sqrt(3) / 2
    
    for r_idx in range(7):
        if idx >= n: break
        ncols = 5 if r_idx % 2 == 0 else 4
        for c_idx in range(ncols):
            if idx >= n: break
            x = 0.5 + (c_idx - ncols/2 + 0.5) * sp
            y = 0.5 + (r_idx - 3.0) * y_sp
            if r_idx % 2 == 1:
                x += sp / 2
            centers[idx] = [np.clip(x, 0.1, 0.9), np.clip(y, 0.1, 0.9)]
            idx += 1
            
    radii = np.full(n, 0.05)
    dt = 0.001  # Growth step size
    
    # 2. Growth Phase
    for step in range(200):
        res = minimize(objective_centers, centers.flatten(), args=(n, radii), 
                       method='L-BFGS-B', 
                       bounds=[(0.0, 1.0)] * (2*n),
                       options={'maxiter': 300, 'ftol': 1e-9, 'gtol': 1e-6})
        centers = res.x.reshape(n, 2)
        
        if res.fun < 1e-5:
            radii += dt
        else:
            # Escape local minima periodically
            if step % 25 == 0:
                centers += np.random.normal(0, 0.015, centers.shape)
                centers = np.clip(centers, 0.05, 0.95)
                
    # 3. Fine-Tuning Phase
    x0 = np.concatenate([centers.flatten(), radii])
    bounds_all = [(0.0, 1.0)]*(2*n) + [(0.0, 0.5)]*n
    res_final = minimize(objective_final, x0, args=(n,), method='L-BFGS-B', 
                         bounds=bounds_all, options={'maxiter': 1500, 'ftol': 1e-9})
                         
    centers_final = res_final.x[:2*n].reshape(n, 2)
    radii_final = np.maximum(res_final.x[2*n:], 0.0)
    
    # 4. Validation & Clamping
    for i in range(n):
        x, y = centers_final[i]
        r = radii_final[i]
        centers_final[i, 0] = np.clip(x, r, 1.0 - r)
        centers_final[i, 1] = np.clip(y, r, 1.0 - r)
        
    return centers_final, radii_final, np.sum(radii_final)
