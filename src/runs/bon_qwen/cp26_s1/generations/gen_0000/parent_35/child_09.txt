1. **Parameterization**: Variables are flattened to $v = [x_1, y_1, r_1, \dots, x_{26}, y_{26}, r_{26}]$.
2. **Objective Function**: Minimize $-\sum r_i + \lambda \cdot P$, where $P$ is the sum of squared violations of boundary and overlap constraints. A large $\lambda$ (5000) strongly penalizes infeasible configurations.
3. **Initialization**: Multiple trials starting from structured hexagonal grids (dense packing heuristic) and random layouts, perturbed with small noise to escape symmetries.
4. **Solver**: `scipy.optimize.minimize` with `L-BFGS-B`, which efficiently handles box bounds and uses gradient-free finite differences robustly for piecewise-smooth penalty functions.
5. **Post-processing**: A deterministic projection step iteratively clips radii to satisfy boundary and separation constraints strictly, adding a tiny margin ($10^{-8}$) to guarantee numerical validity per the checker's tolerance.
6. **Polishing**: The best result undergoes a final optimization run with higher precision settings to squeeze out extra radius.

import numpy as np
from scipy.optimize import minimize
import warnings

warnings.filterwarnings('ignore')

def compute_objective(v, n):
    """
    Computes the negative sum of radii plus penalty terms for constraint violations.
    """
    pos = v[:2*n].reshape(n, 2)
    rad = v[2*n:]
    
    obj = -np.sum(rad)
    pen = 0.0
    lam = 5000.0
    
    # Boundary constraints: r <= x <= 1-r and r <= y <= 1-r
    for i in range(n):
        x, y = pos[i]
        r = rad[i]
        if x < r:
            pen += lam * (x - r)**2
        if x > 1 - r:
            pen += lam * (x - (1 - r))**2
        if y < r:
            pen += lam * (y - r)**2
        if y > 1 - r:
            pen += lam * (y - (1 - r))**2
            
    # Overlap constraints: distance >= r_i + r_j
    for i in range(n):
        xi, yi, ri = pos[i][0], pos[i][1], rad[i]
        for j in range(i + 1, n):
            xj, yj, rj = pos[j][0], pos[j][1], rad[j]
            dx = xi - xj
            dy = yi - yj
            dist = np.sqrt(dx*dx + dy*dy)
            min_dist = ri + rj
            if dist < min_dist:
                pen += lam * (dist - min_dist)**2
                
    return obj + pen

def ensure_validity(pos, rad, n):
    """
    Iteratively adjusts radii to strictly satisfy all constraints.
    """
    for _ in range(300):
        changed = False
        
        # Enforce boundary constraints
        for i in range(n):
            x, y = pos[i]
            r = rad[i]
            new_r = min(r, x, 1-x, y, 1-y)
            if new_r < r:
                rad[i] = new_r
                changed = True
                
        # Enforce non-overlap constraints
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.linalg.norm(pos[i] - pos[j])
                req = rad[i] + rad[j]
                if dist < req:
                    deficit = req - dist
                    red = deficit / 2 + 1e-8
                    rad[i] -= red
                    rad[j] -= red
                    changed = True
                    
        if not changed:
            break
            
    rad = np.maximum(rad, 1e-9)
    return rad

def run_packing():
    np.random.seed(42)
    n = 26
    
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Bounds: coordinates in [0, 1], radii in [0, 0.5]
    bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n
    
    # Multiple restarts with diverse initializations
    for trial in range(18):
        if trial < 12:
            # Hexagonal grid pattern (dense packing heuristic)
            rows = [5, 6, 5, 6, 4]
            positions = []
            dx = 0.185 + np.random.uniform(-0.01, 0.01)
            dy = 0.165 + np.random.uniform(-0.01, 0.01)
            for i, cnt in enumerate(rows):
                y = 0.1 + i * dy
                x_start = (1 - (cnt - 1) * dx) / 2
                if i % 2 == 1:
                    x_start += dx / 2
                for j in range(cnt):
                    positions.append([x_start + j * dx, y])
        else:
            # Random initialization
            positions = np.random.uniform(0.15, 0.85, (n, 2))
            
        positions = np.array(positions)
        positions += np.random.normal(0, 0.008, positions.shape)
        positions = np.clip(positions, 0.05, 0.95)
        
        r0 = 0.07 * np.ones(n)
        v0 = np.concatenate([positions.flatten(), r0])
        
        # Optimization phase
        res = minimize(compute_objective, v0, args=(n,), method='L-BFGS-B', 
                       bounds=bounds, options={'maxiter': 8000, 'ftol': 1e-12})
        
        pos_opt = res.x[:2*n].reshape(n, 2)
        rad_opt = res.x[2*n:]
        rad_final = ensure_validity(pos_opt.copy(), rad_opt.copy(), n)
        current_sum = np.sum(rad_final)
        
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = pos_opt
            best_radii = rad_final
            
    # Final polishing step with tighter tolerances
    v_polish = np.concatenate([best_centers.flatten(), best_radii])
    res_polish = minimize(compute_objective, v_polish, args=(n,), method='L-BFGS-B',
                          bounds=bounds, options={'maxiter': 15000, 'ftol': 1e-14})
    best_centers = res_polish.x[:2*n].reshape(n, 2)
    best_radii = res_polish.x[2*n:]
    best_radii = ensure_validity(best_centers.copy(), best_radii.copy(), n)
    best_sum = np.sum(best_radii)
    
    return best_centers, best_radii, float(best_sum)
