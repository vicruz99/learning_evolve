import scipy.optimize

def run_packing():

    # Or simply create a pattern.

    # 5 rows of 5 is 25. We need 1 more.
    # Maybe 6 rows? 5, 5, 5, 5, 4, 2?

    # A 5x5 grid at 0.1, 0.3, 0.5, 0.7, 0.9 fits 25 circles of r=0.1.
    # Let's use that as base and add one.
    
    # Base grid points
    xs = np.array([0.1, 0.3, 0.5, 0.7, 0.9])
    ys = np.array([0.1, 0.3, 0.5, 0.7, 0.9])
    
    grid_points = []
    for y in ys:
        for x in xs:
            grid_points.append([x, y])
            
    # This gives 25 points. We need 26.
    # Add a point in the center of the square? (0.5, 0.5) is already there.
    # Add a point in a gap?
    # Gaps are at 0.2, 0.4, 0.6, 0.8.
    # Let's add at (0.2, 0.2)
    grid_points.append([0.2, 0.2])
    
    centers = np.array(grid_points)
    
    # Initial radii
    # With r=0.1, the 25 grid circles touch. The new one at (0.2, 0.2)
    # is dist sqrt(0.1^2+0.1^2) = 0.141 from (0.1, 0.1).
    # r + r <= 0.141 => 2r <= 0.141 => r <= 0.07.
    # So initial radius 0.05 is safe.
    radii = np.ones(n) * 0.05
    
    # Stack variables: [x1, y1, r1, x2, y2, r2, ...]
    # Or [x1...xn, y1...yn, r1...rn]
    # Let's use order: x1, y1, r1, x2, y2, r2 ...
    x0 = []
    for i in range(n):
        x0.append(centers[i, 0])
        x0.append(centers[i, 1])
        x0.append(radii[i])
    x0 = np.array(x0)
    
    # Bounds: x, y in [0, 1], r in [0, 0.5]
    bounds = []
    for _ in range(n):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 0.5)) # r
    bounds = tuple(bounds)
    
    # Constraints
    constraints = []
    
    # Boundary constraints: x >= r, x <= 1-r, y >= r, y <= 1-r
    # x - r >= 0
    # 1 - x - r >= 0
    # y - r >= 0
    # 1 - y - r >= 0
    def make_boundary_constraint(i, dim_idx, sign):
        # dim_idx 0 for x, 1 for y. 
        # But variables are grouped (x, y, r).
        # Index of x_i is 3*i, y_i is 3*i+1, r_i is 3*i+2
        idx_xy = 3*i + dim_idx
        idx_r = 3*i + 2
        # Constraint: val >= 0
        # if sign=1: x - r >= 0 => x + (-1)r >= 0? No, linear.
        # We need to return a dict for SLSQP
        def fun(v):
            return v[idx_xy] - v[idx_r]
        def jac(v):
            grad = np.zeros_like(v)
            grad[idx_xy] = 1.0
            grad[idx_r] = -1.0
            return grad
        return {'type': 'ineq', 'fun': fun, 'jac': jac}

    def make_upper_boundary_constraint(i, dim_idx):
        idx_xy = 3*i + dim_idx
        idx_r = 3*i + 2
        def fun(v):
            return 1.0 - v[idx_xy] - v[idx_r]
        def jac(v):
            grad = np.zeros_like(v)
            grad[idx_xy] = -1.0
            grad[idx_r] = -1.0
            return grad
        return {'type': 'ineq', 'fun': fun, 'jac': jac}

        constraints.append(make_boundary_constraint(i, 0)) # x >= r
        constraints.append(make_upper_boundary_constraint(i, 0)) # x <= 1-r
        constraints.append(make_boundary_constraint(i, 1)) # y >= r
        constraints.append(make_upper_boundary_constraint(i, 1)) # y <= 1-r

    def make_overlap_constraint(i, j):
        idx_xi = 3*i
        idx_yi = 3*i+1
        idx_ri = 3*i+2
        
        idx_xj = 3*j
        idx_yj = 3*j+1
        idx_rj = 3*j+2
        
        def fun(v):
            dx = v[idx_xi] - v[idx_xj]
            dy = v[idx_yi] - v[idx_yj]
            dr = v[idx_ri] + v[idx_rj]
            return dx*dx + dy*dy - dr*dr
        
        def jac(v):
            dx = v[idx_xi] - v[idx_xj]
            dy = v[idx_yi] - v[idx_yj]
            dr = v[idx_ri] + v[idx_rj]
            
            grad = np.zeros_like(v)
            
            # d/dxi: 2dx
            grad[idx_xi] = 2.0 * dx
            # d/dxj: -2dx
            grad[idx_xj] = -2.0 * dx
            # d/dyi: 2dy
            grad[idx_yi] = 2.0 * dy
            # d/dyj: -2dy
            grad[idx_yj] = -2.0 * dy
            
            # d/dri: -2(dr)
            grad[idx_ri] = -2.0 * dr
            # d/drj: -2(dr)
            grad[idx_rj] = -2.0 * dr
            
            return grad
        
        return {'type': 'ineq', 'fun': fun, 'jac': jac}

    for i in range(n):
        for j in range(i + 1, n):
            constraints.append(make_overlap_constraint(i, j))

        sum_r = 0.0
        for i in range(n):
            sum_r += v[3*i + 2]
        return -sum_r

    def objective_jac(v):
        grad = np.zeros_like(v)
        for i in range(n):
            grad[3*i + 2] = -1.0
        return grad

    # Using SLSQP
    try:
        res = scipy.optimize.minimize(objective, x0, method='SLSQP', jac=objective_jac,

        if res.success:
            v_opt = res.x
        else:
            # Fallback to initial guess if optimization fails (though unlikely)
            # Or try to return the best found so far?
            # Let's just use the result
            v_opt = res.x
            
    except Exception as e:
        # In case of error, return initial configuration
        v_opt = x0

    # Extract results
    final_centers = np.zeros((n, 2))
    final_radii = np.zeros(n)
    
    for i in range(n):
        final_centers[i, 0] = v_opt[3*i]
        final_centers[i, 1] = v_opt[3*i+1]
        final_radii[i] = v_opt[3*i+2]

    # If a circle is out of bounds, shrink radius
    for i in range(n):
        x, y = final_centers[i]
        r = final_radii[i]
        min_dist_wall = min(x, 1-x, y, 1-y)
        if min_dist_wall < r:
            final_radii[i] = max(0.0, min_dist_wall)
            
    sum_radii = np.sum(final_radii)
    
    return final_centers, final_radii, sum_radii
