    num_iterations = 25000 # Number of optimization steps
    r_increment = 0.000015 # Rate at which radius grows per step
    
    # Force constants
    k_repulsion = 10.0 # Strength of repulsion between overlapping circles
    k_boundary = 20.0 # Strength of repulsion from boundaries
    
    for it in range(num_iterations):
        # Slowly increase the target radius
        r += r_increment
        
        # Vectorized calculation of differences and distances
        # diffs[i, j] = centers[i] - centers[j]
        diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
        dists = np.sqrt(np.sum(diffs**2, axis=2))
        
        # Avoid division by zero for diagonal elements (distance to self)
        safe_dists = np.where(dists < 1e-9, 1e-9, dists)
        
        # Unit vectors pointing from j to i (direction of repulsion for i)
        dirs = diffs / safe_dists[..., np.newaxis]
        
        # Calculate overlaps: positive if distance < 2*r
        min_dist = 2 * r
        overlaps = np.maximum(0, min_dist - dists)
        
        # Force magnitudes proportional to overlap
        force_mags = overlaps * k_repulsion
        
        # Pairwise forces: force on i from j is in direction of (i-j) with magnitude force_mags
        # pair_forces[i, j] is force exerted by j on i
        pair_forces = force_mags[..., np.newaxis] * dirs
        
        # Sum forces from all j to get total repulsive force on i
        # Note: Self-interaction (i==j) is zero because diffs[i,i] is zero vector.
        forces = np.sum(pair_forces, axis=1)
        
        # Boundary forces
        x = centers[:, 0]
        y = centers[:, 1]
        
        # X-axis boundaries
        # Left wall (x < r): push right (+ direction)
        mask_left = x < r
        forces[mask_left, 0] += (r - x[mask_left]) * k_boundary
        # Right wall (x > 1-r): push left (- direction)
        mask_right = x > 1 - r
        forces[mask_right, 0] -= (x[mask_right] - (1 - r)) * k_boundary
        
        # Y-axis boundaries
        # Bottom wall (y < r): push up (+ direction)
        mask_bottom = y < r
        forces[mask_bottom, 1] += (r - y[mask_bottom]) * k_boundary
        # Top wall (y > 1-r): push down (- direction)
        mask_top = y > 1 - r
        forces[mask_top, 1] -= (y[mask_top] - (1 - r)) * k_boundary
        
        # Update centers using Euler integration
        centers += forces * dt
        
        # Strict clipping to keep centers inside [0, 1] to prevent divergence
        centers[:, 0] = np.clip(centers[:, 0], 0.0, 1.0)
        centers[:, 1] = np.clip(centers[:, 1], 0.0, 1.0)
        
    # 3. Calculate final valid radii
    # We find the largest radius r_max such that all circles of radius r_max
    # fit within the square and do not overlap with each other.
    
    min_r = 1.0
    
    # Constraint 1: Distance to boundaries
    # For each circle, r <= min(x, 1-x, y, 1-y)
    bounds_r = np.minimum(np.minimum(centers[:, 0], 1 - centers[:, 0]), 
                          np.minimum(centers[:, 1], 1 - centers[:, 1]))
    min_r = np.min(bounds_r)
    
    # Constraint 2: Distance between circles
    # For each pair, 2r <= dist(i, j) => r <= dist(i, j) / 2
    # Recompute distances for final centers
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    # Minimum distance between distinct circles (lower triangle excluding diagonal)
    min_pair_dist = np.min(np.tril(dists, -1))
    min_r = min(min_r, min_pair_dist / 2)
    
    # Apply a small safety margin to ensure strict inequality for the validator
    min_r -= 1e-9
    
    # Assign radii (assuming equal radii is optimal or near-optimal for this heuristic)
    radii = np.full(n, min_r)
    sum_radii = float(np.sum(radii))
    
    return centers, radii, sum_radii
