import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26


def objective(x):
    """Minimize negative sum of radii (equivalent to maximizing sum of radii)."""
    return -np.sum(x[2 * N_CIRCLES:])


def constraints(x):
    """Compute all inequality constraints (must be >= 0 for feasibility).
    
    Constraints include:
    - Each circle stays within [0,1] x [0,1]
    - Each radius is positive
    - No two circles overlap
    """
    n = N_CIRCLES
    pts = x[:2 * n].reshape(n, 2)
    rs = x[2 * n:]
    c = []
    
    # Boundary constraints for each circle
    for i in range(n):
        # x_i >= r_i (left boundary)
        c.append(pts[i, 0] - rs[i])
        # x_i + r_i <= 1 (right boundary)
        c.append(1.0 - pts[i, 0] - rs[i])
        # y_i >= r_i (bottom boundary)
        c.append(pts[i, 1] - rs[i])
        # y_i + r_i <= 1 (top boundary)
        c.append(1.0 - pts[i, 1] - rs[i])
        # r_i > 0 (positive radius)
        c.append(rs[i] - 1e-14)
    
    # Non-overlap constraints for each pair
    for i in range(n):
        for j in range(i + 1, n):
            dx = pts[i, 0] - pts[j, 0]
            dy = pts[i, 1] - pts[j, 1]
            d = np.sqrt(dx * dx + dy * dy)
            c.append(d - rs[i] - rs[j])
    
    return np.array(c)


def hexagonal_init(n):
    """Initialize circles in a hexagonal lattice arrangement.
    
    Tries multiple grid configurations and picks the one allowing largest circles.
    Odd rows are offset horizontally for hexagonal packing efficiency.
    """
    best_r = 0.0
    best_cfg = None
    
    for cols in range(4, 10):
        rows = int(np.ceil(n / cols))
        # Width constraint: 2*r*cols <= 1
        r_w = 1.0 / (2.0 * cols)
        # Height constraint for hexagonal: 2*r + (rows-1)*r*sqrt(3) <= 1
        r_h = 1.0 / (2.0 + (rows - 1) * np.sqrt(3))
        r = min(r_w, r_h) * 0.82  # Start slightly smaller for safety
        if r > best_r:
            best_r = r
            best_cfg = (cols, rows, r)
    
    cols, rows, r = best_cfg
    pts = np.zeros((n, 2))
    rs = np.full(n, r)
    
    idx = 0
    for row in range(rows):
        nc = min(cols, n - idx)
        # Offset odd rows for hexagonal packing
        x_off = r if row % 2 == 1 else 0.0
        for col in range(nc):
            if idx >= n:
                break
            x = r + x_off + 2.0 * r * col
            y = r + row * r * np.sqrt(3)
            # Clamp to valid range
            x = max(r, min(1.0 - r, x))
            y = max(r, min(1.0 - r, y))
            pts[idx] = [x, y]
            idx += 1
        if idx >= n:
            break
    
    return pts, rs


def run_packing():
    """Pack 26 circles in unit square to maximize sum of radii.
    
    Returns:
        centers: (26, 2) array of circle centers
        radii: (26,) array of circle radii
        sum_radii: sum of all radii
    """
    # Generate initial hexagonal configuration
    pts, rs = hexagonal_init(N_CIRCLES)
    x0 = np.concatenate([pts.flatten(), rs])
    
    # Define bounds for all variables
    bounds = [(0.0, 1.0) for _ in range(2 * N_CIRCLES)] + \
             [(1e-14, 0.5) for _ in range(N_CIRCLES)]
    
    # Define constraint dictionary
    constr_dict = {'type': 'ineq', 'fun': constraints}
    
    best_val = float('inf')
    best_x = x0.copy()
    
    # Multiple restarts with different perturbations
    for seed in range(15):
        rng = np.random.RandomState(seed)
        
        # Perturb initial configuration
        x0_pert = x0 + rng.normal(0, 0.012, size=x0.shape)
        x0_pert[:2 * N_CIRCLES] = np.clip(x0_pert[:2 * N_CIRCLES], 0.02, 0.98)
        x0_pert[2 * N_CIRCLES:] = np.clip(x0_pert[2 * N_CIRCLES:], 0.005, 0.5)
        
        # Run optimization
        result = minimize(
            objective,
            x0_pert,
            method='SLSQP',
            constraints=constr_dict,
            bounds=bounds,
            options={
                'maxiter': 4000,
                'ftol': 1e-15,
                'disp': False
            }
        )
        
        if -result.fun < best_val:
            best_val = -result.fun
            best_x = result.x.copy()
    
    # Extract final configuration
    centers = best_x[:2 * N_CIRCLES].reshape(N_CIRCLES, 2)
    radii = best_x[2 * N_CIRCLES:]
    radii = np.maximum(radii, 0.0)
    
    return centers, radii, np.sum(radii)
