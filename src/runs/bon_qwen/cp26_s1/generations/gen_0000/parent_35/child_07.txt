

<strategy>
I'll use a block coordinate descent approach combining:
1. Linear Programming (LP) to compute optimal radii for fixed centers
2. Force-directed simulation to update centers and create more space
3. Multiple random restarts to escape local optima

Key insights:
- For fixed centers, the optimal radii can be found via LP with constraints: r_i ≤ boundary distances and r_i + r_j ≤ distance between centers
- After computing radii, apply repulsive forces between nearly-touching circles to create space for growth
- Use gradient information from the LP binding constraints to guide position updates
- Multiple restarts from perturbed hexagonal grids help find global optimum
- Use scipy.optimize.minimize with L-BFGS-B for final refinement
</strategy>

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.spatial.distance import cdist


def solve_radii_lp(centers, n):
    """Solve LP to maximize sum of radii given centers."""
    c = -np.ones(n)
    
    A_ub_list = []
    b_ub_list = []
    
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        x, y = centers[i]
        for val in [x, 1-x, y, 1-y]:
            row = np.zeros(n)
            row[i] = 1
            A_ub_list.append(row)
            b_ub_list.append(val)
    
    # Non-overlap constraints: r_i + r_j <= dist(i,j)
    for i in range(n):
        for j in range(i+1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            row = np.zeros(n)
            row[i] = 1
            row[j] = 1
            A_ub_list.append(row)
            b_ub_list.append(dist)
    
    A_ub = np.array(A_ub_list)
    b_ub = np.array(b_ub_list)
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        return -result.fun, result.x
    else:
        return None, None


def compute_forces(centers, radii, n):
    """Compute repulsive forces to separate circles and push away from boundaries."""
    forces = np.zeros((n, 2))
    
    for i in range(n):
        for j in range(i+1, n):
            dist_vec = centers[i] - centers[j]
            dist = np.sqrt(np.sum(dist_vec**2))
            
            if dist > 1e-10:
                desired_dist = radii[i] + radii[j]
                if dist < desired_dist + 0.005:
                    # Repulsive force proportional to overlap
                    overlap = max(0, desired_dist - dist + 0.003)
                    force_mag = overlap * 50.0
                    forces[i] += (dist_vec / dist) * force_mag
                    forces[j] -= (dist_vec / dist) * force_mag
        
        # Boundary forces
        x, y = centers[i]
        r = radii[i]
        
        # Left boundary
        if x - r < 0.015:
            forces[i, 0] += (0.015 - (x - r)) * 30.0
        # Right boundary
        if x + r > 0.985:
            forces[i, 0] -= (x + r - 0.985) * 30.0
        # Bottom boundary
        if y - r < 0.015:
            forces[i, 1] += (0.015 - (y - r)) * 30.0
        # Top boundary
        if y + r > 0.985:
            forces[i, 1] -= (y + r - 0.985) * 30.0
    
    return forces


def initialize_hexagonal(n, seed=0):
    """Initialize centers in a hexagonal pattern with optional perturbation."""
    rng = np.random.RandomState(seed)
    
    centers = np.zeros((n, 2))
    idx = 0
    
    # Try different row configurations
    row_configs = [
        [5, 5, 5, 5, 4, 2],  # 26 circles
        [5, 5, 5, 5, 5, 1],  # 26 circles
        [5, 5, 5, 6, 5],     # 26 circles
        [6, 5, 5, 5, 5],     # 26 circles
        [5, 6, 5, 6, 4],     # 26 circles
        [6, 6, 4, 6, 4],     # 26 circles
        [5, 5, 6, 5, 5],     # 26 circles
        [4, 6, 6, 6, 4],     # 26 circles
    ]
    
    config_idx = seed % len(row_configs)
    rows = row_configs[config_idx]
    
    y_start = 0.1 + rng.uniform(0, 0.03)
    y_spacing = 0.155 + rng.uniform(-0.01, 0.02)
    x_spacing = 0.185 + rng.uniform(-0.01, 0.02)
    
    for row_idx, count in enumerate(rows):
        y = y_start + row_idx * y_spacing
        shift = x_spacing / 2.0 if row_idx % 2 == 1 else 0
        x_start = 0.09 - shift + rng.uniform(-0.02, 0.02)
        
        for col_idx in range(count):
            if idx < n:
                x = x_start + col_idx * x_spacing + rng.uniform(-0.005, 0.005)
                centers[idx] = [x, y]
                idx += 1
    
    # Clip to safe range
    centers[:, 0] = np.clip(centers[:, 0], 0.05, 0.95)
    centers[:, 1] = np.clip(centers[:, 1], 0.05, 0.95)
    
    return centers


def optimize_positions_with_lbfgs(centers, radii, n):
    """Use L-BFGS-B to optimize positions given current radii as target."""
    def objective(params):
        c = params.reshape(n, 2)
        # Penalize constraint violations
        penalty = 0.0
        
        # Boundary penalties
        for i in range(n):
            x, y = c[i]
            r = radii[i]
            if x - r < 0:
                penalty += 100 * (x - r)**2
            if x + r > 1:
                penalty += 100 * (x + r - 1)**2
            if y - r < 0:
                penalty += 100 * (y - r)**2
            if y + r > 1:
                penalty += 100 * (y + r - 1)**2
        
        # Overlap penalties
        for i in range(n):
            for j in range(i+1, n):
                dist = np.sqrt(np.sum((c[i] - c[j])**2))
                if dist < radii[i] + radii[j]:
                    penalty += 100 * (radii[i] + radii[j] - dist)**2
        
        # Also penalize being too far from desired spacing
        # This helps maintain the hexagonal structure
        return penalty
    
    bounds = [(0.01, 0.99)] * (2 * n)
    x0 = centers.flatten()
    
    result = minimize(objective, x0, method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': 100, 'ftol': 1e-10})
    
    return result.x.reshape(n, 2)


def run_packing():
    n = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Try multiple restarts
    for restart in range(15):
        centers = initialize_hexagonal(n, seed=restart)
        
        # Phase 1: Block coordinate descent with force-directed updates
        for iteration in range(300):
            # Solve LP for radii
            obj, radii = solve_radii_lp(centers, n)
            
            if obj is None:
                break
            
            # Compute and apply forces
            forces = compute_forces(centers, radii, n)
            
            # Adaptive step size
            step_size = 0.0005 / (1 + iteration * 0.002)
            
            # Add small random perturbation for exploration
            if iteration % 50 == 0 and iteration > 0:
                perturbation = np.random.randn(n, 2) * 0.002
                centers += perturbation
            
            centers += forces * step_size
            
            # Project to feasible region
            obj2, radii2 = solve_radii_lp(centers, n)
            if obj2 is not None:
                for i in range(n):
                    centers[i, 0] = max(radii2[i] + 1e-8, min(1 - radii2[i] - 1e-8, centers[i, 0]))
                    centers[i, 1] = max(radii2[i] + 1e-8, min(1 - radii2[i] - 1e-8, centers[i, 1]))
            
            if obj > best_sum:
                best_sum = obj
                best_centers = centers.copy()
                best_radii = radii.copy()
        
        # Phase 2: Refine with L-BFGS-B
        obj, radii = solve_radii_lp(best_centers, n)
        if obj is not None:
            refined_centers = optimize_positions_with_lbfgs(best_centers, radii, n)
            obj2, radii2 = solve_radii_lp(refined_centers, n)
            
            if obj2 is not None and obj2 > best_sum:
                best_sum = obj2
                best_centers = refined_centers.copy()
                best_radii = radii2.copy()
    
    # Final cleanup: ensure all constraints are satisfied
    # Small adjustments if needed
    if best_centers is not None:
        for i in range(n):
            x, y = best_centers[i]
            r = best_radii[i]
            best_centers[i, 0] = max(r + 1e-10, min(1 - r - 1e-10, x))
            best_centers[i, 1] = max(r + 1e-10, min(1 - r - 1e-10, y))
        
        # Verify and fix any remaining overlaps
        for _ in range(10):
            changed = False
            for i in range(n):
                for j in range(i+1, n):
                    dist = np.sqrt(np.sum((best_centers[i] - best_centers[j])**2))
                    required = best_radii[i] + best_radii[j]
                    if dist < required:
                        # Reduce the smaller radius slightly
                        overlap = required - dist
                        if best_radii[i] < best_radii[j]:
                            best_radii[i] -= overlap / 2 + 1e-12
                        else:
                            best_radii[j] -= overlap / 2 + 1e-12
                        changed = True
            
            if not changed:
                break
        
        # Recompute sum after cleanup
        final_sum = np.sum(best_radii)
        best_sum = final_sum
    
    return best_centers, best_radii, best_sum
