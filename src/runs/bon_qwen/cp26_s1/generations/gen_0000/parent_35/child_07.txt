import numpy as np
from scipy.optimize import linprog, minimize

def solve_radii_for_centers(centers):
    """
    Given fixed centers, solve the LP to find optimal radii maximizing their sum.
    Returns (sum_radii, radii_array).
    """
    n = centers.shape[0]
    # Pairwise Euclidean distances
    dists = np.sqrt(np.sum((centers[:, None] - centers[None, :])**2, axis=2))
    
    c = -np.ones(n)
    n_pairs = n * (n - 1) // 2
    n_bounds = 4 * n
    A_ub = np.zeros((n_pairs + n_bounds, n))
    b_ub = np.zeros(n_pairs + n_bounds)
    
    idx = 0
    # Circle-circle non-overlap constraints: r_i + r_j <= d_ij
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dists[i, j]
            idx += 1
            
    # Boundary constraints: r_i <= x, r_i <= 1-x, r_i <= y, r_i <= 1-y
    for i in range(n):
        bounds_val = [centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1]]
        for v in bounds_val:
            A_ub[idx, i] = 1.0
            b_ub[idx] = v
            idx += 1
            
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return -res.fun, -res.x
    return 0.0, np.zeros(n)

def optimization_obj(x, n):
    """Objective function for scipy.optimize.minimize"""
    centers = np.clip(x.reshape(n, 2), 1e-5, 1.0 - 1e-5)
    s, _ = solve_radii_for_centers(centers)
    return -s

def run_packing():
    n = 26
    best_sum = 0.0
    best_centers = None
    
    inits = []
    
    # 1. Dense Grid initialization
    gs = np.linspace(0.08, 0.92, 6)
    c1 = np.array([(x, y) for y in gs for x in gs])[:n]
    inits.append(c1)
    
    # 2. Hexagonal lattice initialization
    c2 = np.zeros((n, 2))
    idx = 0
    y_pos = 0.08
    row = 0
    while idx < n:
        x_start = 0.08 + (0.065 if row % 2 == 1 else 0.0)
        x_pos = x_start
        while idx < n and x_pos < 0.95:
            c2[idx] = [x_pos, y_pos]
            idx += 1
            x_pos += 0.13
        y_pos += 0.113
        row += 1
    inits.append(c2)
    
    # 3. Multiple random initializations to escape local minima
    np.random.seed(42)
    for _ in range(8):
        inits.append(np.random.uniform(0.05, 0.95, (n, 2)))
        
    # Run local optimization from each start
    for ic in inits:
        res = minimize(optimization_obj, ic.flatten(), args=(n,), method='Nelder-Mead', 
                       options={'maxiter': 6000, 'xatol': 1e-7, 'fatol': 1e-9})
        
        if -res.fun > best_sum:
            best_sum = -res.fun
            best_centers = res.x.reshape(n, 2)
            
    # Final safety clipping and radius recomputation
    best_centers = np.clip(best_centers, 1e-6, 1.0 - 1e-6)
    final_sum, radii = solve_radii_for_centers(best_centers)
    
    return best_centers, radii, final_sum
