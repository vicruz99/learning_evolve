import numpy as np
from scipy.optimize import minimize

def compute_loss_and_grad(params, n, lam, lam_b):
    """
    Compute loss and analytic gradient for circle packing optimization.
    Variables order: [x0, y0, x1, y1, ..., r0, r1, ..., r_{n-1}]
    """
    centers = params[:2 * n].reshape(n, 2)
    radii = params[2 * n:]
    
    loss = -np.sum(radii)
    grad = np.zeros_like(params)
    grad[2 * n:] -= 1.0  # Derivative of -sum(r) wrt radii
    
    # Pairwise non-overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            
            # Avoid division by zero if centers coincide
            if dist < 1e-9:
                dist = 1e-9
                dx, dy = 0.0, 0.0
                
            overlap = radii[i] + radii[j] - dist
            if overlap > 0.0:
                pen = lam * overlap * overlap
                loss += pen
                
                f = 2.0 * lam * overlap
                grad[2 * n + i] += f
                grad[2 * n + j] += f
                
                fac = f / dist
                grad[2 * i] += fac * dx
                grad[2 * i + 1] += fac * dy
                grad[2 * j] -= fac * dx
                grad[2 * j + 1] -= fac * dy
                
    # Boundary constraints (circles inside [0,1]x[0,1])
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        
        # Left boundary: x - r >= 0  =>  r - x <= 0
        v = r - x
        if v > 0.0:
            loss += lam_b * v * v
            grad[2 * n + i] += 2.0 * lam_b * v
            grad[2 * i] -= 2.0 * lam_b * v
            
        # Right boundary: 1 - x - r >= 0  =>  r - (1-x) <= 0
        v = r - (1.0 - x)
        if v > 0.0:
            loss += lam_b * v * v
            grad[2 * n + i] += 2.0 * lam_b * v
            grad[2 * i] += 2.0 * lam_b * v
            
        # Bottom boundary: y - r >= 0
        v = r - y
        if v > 0.0:
            loss += lam_b * v * v
            grad[2 * n + i] += 2.0 * lam_b * v
            grad[2 * i + 1] -= 2.0 * lam_b * v
            
        # Top boundary: 1 - y - r >= 0
        v = r - (1.0 - y)
        if v > 0.0:
            loss += lam_b * v * v
            grad[2 * n + i] += 2.0 * lam_b * v
            grad[2 * i + 1] += 2.0 * lam_b * v
            
    return loss, grad

            centers[idx] = [x_pos + c * dx, y_pos]
            idx += 1
        y_pos += dy
        
    # Ensure initial placement is safely inside the square
    centers = np.clip(centers, 0.05, 0.95)
    
    # Flatten to optimization vector: [x0,y0, x1,y1, ..., r0,r1,...]
    params0 = np.concatenate([centers.flatten(), radii])
    
    # Bounds for variables
    bounds = [(0.0, 1.0)] * (2 * n) + [(1e-5, 0.5)] * n
    
    params = params0.copy()

    ]

            compute_loss_and_grad,

            bounds=bounds,
            options={'maxiter': 1500, 'ftol': 1e-15, 'gtol': 1e-8}
        )
        params = res.x

    radii_opt = params[2 * n:]

        radii_opt -= max_v * 0.5

    return centers_opt, radii_opt, sum_r
