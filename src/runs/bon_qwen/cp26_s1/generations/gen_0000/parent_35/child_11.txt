import numpy as np
from scipy.optimize import minimize

def compute_penalty(centers, radii):
    """Computes squared penalty for overlaps and boundary violations."""
    n = centers.shape[0]
    penalty = 0.0
    
    # Pairwise distances and overlaps
    diffs = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    
    sum_radii = radii[:, None] + radii[None, :]
    overlaps = np.maximum(0.0, sum_radii - dists)
    penalty += np.sum(overlaps**2)
    
    # Boundary violations
    penalty += np.sum(np.maximum(0.0, radii - centers[:, 0])**2)
    penalty += np.sum(np.maximum(0.0, radii - (1.0 - centers[:, 0]))**2)
    penalty += np.sum(np.maximum(0.0, radii - centers[:, 1])**2)
    penalty += np.sum(np.maximum(0.0, radii - (1.0 - centers[:, 1]))**2)
    
    return penalty

def objective(x, lam):
    """Objective function: minimize penalty - sum(radii)."""
    n = 26
    centers = x[:2*n].reshape(n, 2)
    radii = x[2*n:]
    return lam * compute_penalty(centers, radii) - np.sum(radii)

                    idx += 1
            row += 1
            
        # Add controlled random perturbation
        centers += rng.uniform(-0.04, 0.04, size=centers.shape)

            forces = np.sum((overlaps / safe_dists)[:, :, None] * (diffs / safe_dists[:, :, None]), axis=1)

                dr *= 0.92
