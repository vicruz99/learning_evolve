import numpy as np
from scipy.optimize import minimize


def obj_func(x):
    """Negative sum of radii (to minimize)."""
    n = 26
    return -np.sum(x[2 * n:])


def con_func(x):
    """Compute all constraint violations (must be >= 0 for feasible)."""
    n = 26
    centers = x[:2 * n].reshape(n, 2)
    radii = x[2 * n:]

    c = []

    # Boundary constraints: circle inside unit square
    for i in range(n):
        c.append(centers[i, 0] - radii[i])
        c.append(1.0 - centers[i, 0] - radii[i])
        c.append(centers[i, 1] - radii[i])
        c.append(1.0 - centers[i, 1] - radii[i])

    # Non-overlap constraints using squared distance
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist_sq = dx * dx + dy * dy
            c.append(dist_sq - (radii[i] + radii[j]) ** 2)

    return np.array(c)


def get_bounds():
    """Create variable bounds: centers in [0,1], radii in [0, 0.5]."""
    n = 26
    b = []
    for _ in range(n):
        b.append((0.0, 1.0))
        b.append((0.0, 1.0))
        b.append((0.0, 0.5))
    return b


def init_hexagonal(n, row_counts=None, r0=None, jitter=0.0):
    """Initialize circles in a hexagonal pattern."""
    if row_counts is None:
        row_counts = [6, 5, 6, 5, 4]
    if r0 is None:
        r0 = 0.07

    centers = np.zeros((n, 2))
    radii = np.full(n, r0)

    idx = 0
    num_rows = len(row_counts)
    for row in range(num_rows):
        count = row_counts[row]
        # Vertical position with hexagonal spacing
        y = r0 + row * r0 * np.sqrt(3)
        # Horizontal start, with offset for alternating rows
        x_start = r0 + (row % 2) * r0

        for col in range(count):
            x = x_start + col * 2 * r0
            centers[idx, 0] = x + np.random.uniform(-jitter, jitter)
            centers[idx, 1] = y + np.random.uniform(-jitter, jitter)
            radii[idx] = r0 + np.random.uniform(-jitter * 0.2, jitter * 0.2)
            idx += 1

    return centers, radii


def init_grid(n, rows=5, cols=5, jitter=0.0):

    centers = np.zeros((n, 2))

    radii = np.full(n, r0)

    idx = 0
    for row in range(rows):
        for col in range(cols):

                break

            centers[idx, 1] = (row + 1) * r0 + np.random.uniform(-jitter, jitter)
            idx += 1

    return centers, radii


def init_random(n):

    return centers, radii


def fix_packing(centers, radii, n):
    """Post-process to ensure all constraints are satisfied."""
    # Fix boundary violations
    for i in range(n):
        max_r = min(centers[i, 0], 1.0 - centers[i, 0],
                    centers[i, 1], 1.0 - centers[i, 1])
        radii[i] = min(radii[i], max_r)
        radii[i] = max(radii[i], 0.0)

    # Fix overlap violations iteratively
    for _ in range(5):
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist < radii[i] + radii[j] - 1e-12:
                    excess = radii[i] + radii[j] - dist
                    if excess > 0:
                        radii[i] -= excess / 2.0
                        radii[j] -= excess / 2.0

    radii[:] = np.maximum(radii, 0.0)


def grow_circles(n, num_steps=5000):
    """Growing circles algorithm: gradually increase radii while resolving overlaps."""

    for step in range(num_steps):
        # Grow all circles slightly
        growth = 5e-6
        radii[:] += growth

            for i in range(n):
                for j in range(i + 1, n):
                    dx = centers[i, 0] - centers[j, 0]
                    dy = centers[i, 1] - centers[j, 1]

                        centers[i, 1] += ny * excess
                        centers[j, 0] -= nx * excess
                        centers[j, 1] -= ny * excess

        # Push circles inside boundary
        for i in range(n):
            if centers[i, 0] - radii[i] < 0:
                centers[i, 0] = radii[i]
            if centers[i, 0] + radii[i] > 1:
                centers[i, 0] = 1.0 - radii[i]
            if centers[i, 1] - radii[i] < 0:
                centers[i, 1] = radii[i]
            if centers[i, 1] + radii[i] > 1:
                centers[i, 1] = 1.0 - radii[i]

    return centers, radii
