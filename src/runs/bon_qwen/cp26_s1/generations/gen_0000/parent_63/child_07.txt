import numpy as np

def compute_overlaps_and_gradients(centers, radii, penalty_weight=100.0, lr_c=0.02, lr_r=0.02):
    """
    Computes overlap penalties, boundary penalties, and their gradients 
    for centers and radii to guide optimization.
    """
    n = centers.shape[0]
    
    # --- Pairwise Overlaps ---
    # centers: (n, 2)
    # diffs: (n, n, 2)
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    
    # Handle self-distances (set to 1 to avoid div by zero later, as overlap is 0 anyway)
    np.fill_diagonal(dists, 1.0)
    
    # sum_radii: (n, n)
    sum_radii = radii[:, np.newaxis] + radii[np.newaxis, :]
    
    # Overlap amount: (r_i + r_j) - dist. If positive, circles overlap.
    overlaps = np.maximum(0.0, sum_radii - dists)
    
    # Penalty term for overlaps (squared for smooth gradient)
    # We only care about i < j for energy, but gradients are symmetric.
    overlap_energy = np.sum(overlaps**2)
    
    # --- Boundary Penalties ---
    # x constraints: r <= x <= 1-r
    # Violation: max(0, r - x) + max(0, r - (1 - x))
    # r - x > 0 => x < r (too close to 0)
    # r - (1 - x) = x + r - 1 > 0 => x > 1 - r (too close to 1)
    
    x_viol_0 = np.maximum(0.0, radii - centers[:, 0])
    x_viol_1 = np.maximum(0.0, centers[:, 0] + radii - 1.0)
    y_viol_0 = np.maximum(0.0, radii - centers[:, 1])
    y_viol_1 = np.maximum(0.0, centers[:, 1] + radii - 1.0)
    
    boundary_energy = np.sum(x_viol_0**2 + x_viol_1**2 + y_viol_0**2 + y_viol_1**2)
    
    total_energy = overlap_energy + boundary_energy
    
    # --- Gradients ---
    # Gradient of overlap_energy w.r.t centers
    # d/dr((r_i+r_j-d)^2) w.r.t x_i: 2(r_i+r_j-d) * (-d/dx_i(d))
    # d/dx_i(d) = (x_i - x_j) / d
    # Force on i from j is in direction of (x_i - x_j)
    
    # Unit vectors (direction from j to i)
    # Normalize diffs
    unit_vecs = diffs / dists[:, :, np.newaxis] # (n, n, 2)
    
    # Gradient contribution from overlaps
    # overlap_grad_C[i] = sum_j ( -2 * overlap_ij * unit_vec_ij ) 
    # Note: We want to minimize energy, so we move opposite to gradient.
    # Overlap increases energy. Gradient points into the overlap.
    # So we want force pointing away.
    # The term -2 * overlap * unit_vec is correct for descent direction.
    # But standard gradient of (r-r_j-d)^2 w.r.t C_i is -2(r-r_j-d) * (C_i-C_j)/d
    # If overlap > 0, (r-r_j-d) > 0. (C_i-C_j)/d is unit vector from j to i.
    # Gradient points towards i (increasing distance reduces d, increasing overlap? No.)
    # Wait: E = (R - d)^2. dE/dd = -2(R-d).
    # dE/dC_i = dE/dd * dd/dC_i = -2(R-d) * (C_i-C_j)/d.
    # If R > d (overlap), term is negative.
    # (C_i-C_j)/d is vector from j to i.
    # So gradient points from i towards j (into the overlap).
    # To minimize, we move opposite: from j to i (away).
    # So Force_i = + 2 * overlap * unit_vec_ij.
    
    overlap_force = 2.0 * overlaps[:, :, np.newaxis] * unit_vecs
    grad_C_overlap = np.sum(overlap_force, axis=1) # Sum over j
    
    # Gradient of boundary_energy w.r.t centers
    # x_viol_0 = max(0, r-x). dE/dx = 2(r-x)*(-1) = -2(r-x).
    # If x < r, gradient is negative (pushes x up). Correct.
    # x_viol_1 = max(0, x+r-1). dE/dx = 2(x+r-1)*(1).
    # If x > 1-r, gradient is positive (pushes x down). Correct.
    
    grad_C_x_0 = -2.0 * x_viol_0
    grad_C_x_1 = 2.0 * x_viol_1
    grad_C_y_0 = -2.0 * y_viol_0
    grad_C_y_1 = 2.0 * y_viol_1
    
    grad_C = np.zeros_like(centers)
    grad_C[:, 0] += grad_C_x_0 + grad_C_x_1
    grad_C[:, 1] += grad_C_y_0 + grad_C_y_1
    
    # Combine center gradients (weighted by penalty weight to balance with radius growth)
    # We use the energy gradient directly for descent: C -= lr * grad
    # So effective update is -lr * grad.
    
    # --- Gradients w.r.t Radii ---
    # Objective: Maximize sum(r). So we want gradient of (sum(r) - weight * Energy).
    # Grad sum(r) = 1.
    # Grad Energy w.r.r r_i:
    
    # From overlaps:
    # d/dr_i ((r_i+r_j-d)^2) = 2(r_i+r_j-d)
    grad_r_overlap = 2.0 * np.sum(overlaps, axis=1)
    
    # From boundaries:
    # x_viol_0: 2(r-x). d/dr = 2(r-x).
    # x_viol_1: 2(x+r-1). d/dr = 2(x+r-1).
    grad_r_bound = 2.0 * (x_viol_0 + x_viol_1 + y_viol_0 + y_viol_1)
    
    grad_r_total = grad_r_overlap + grad_r_bound
    
    # We want to move r in direction of (1 - weight * grad_r_total)
    # This is equivalent to gradient ascent on (sum r - weight * E)
    grad_r_descent = grad_r_total # We will subtract this from 1 later or use logic
    
    # Let's define the update directions explicitly to be clear
    # Center update: C += step * ( - grad_C_overlap - grad_C_boundary )
    # But wait, standard gradient descent on E is C -= lr * dE/dC.
    # dE/dC_overlap = -2 * overlap * unit_vec? 
    # Let's re-verify sign.
    # E = (R-d)^2. dE/dC_i = -2(R-d) * u_ij.
    # If overlap > 0, R-d > 0. dE/dC_i is negative of unit vector.
    # So gradient points towards j.
    # To minimize, C -= lr * (-u) => C += lr * u (away).
    # My previous `grad_C_overlap` was +2*overlap*u.
    # So that variable actually holds the *Force* (negative gradient).
    # Let's stick to Forces (push away).
    
    force_C = grad_C_overlap + grad_C # Wait, grad_C boundary calculation above:
    # x_viol_0 = r-x. dE/dx = -2(r-x). Force should be +2(r-x) (push right).
    # My grad_C_x_0 was -2(r-x). That is the gradient.
    # So Force = -Gradient.
    force_C_boundary = -np.array([grad_C_x_0 + grad_C_x_1, grad_C_y_0 + grad_C_y_1]).T
    force_C_overlap = grad_C_overlap # This was defined as +2*overlap*u (Away).
    
    total_force_C = force_C_overlap + force_C_boundary
    
    # Radius update:
    # We want to increase r if no penalty.
    # Gradient of (Sum r - Weight * E) w.r.t r is 1 - Weight * dE/dr.
    # dE/dr_overlap = 2*overlap.
    # dE/dr_bound = 2*violations.
    # So ascent direction = 1 - Weight * (2*overlap + 2*viol).
    
    grad_E_r = grad_r_overlap + grad_r_bound
    ascent_dir_r = 1.0 - penalty_weight * grad_E_r
    
    return total_energy, total_force_C, ascent_dir_r, (overlap_energy, boundary_energy)

def run_packing():
    # Number of circles
    n = 26
    
    # --- Initialization ---
    # Hexagonal packing initialization
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.08) # Start with reasonable radius
    
    # Fill hex grid
    count = 0
    row = 0
    while count < n:
        # Number of circles in this row
        # Alternating 5 and 6 or similar to fit in 1x1
        # With r=0.08, spacing ~0.16 (x) and ~0.14 (y).
        # We can fit ~5-6 cols and ~7-8 rows.
        
        # Shift for hexagonal
        x_offset = 0.0 if row % 2 == 0 else 0.1 
        
        # Approximate number of columns
        # Width 1. Range of x roughly [0.1, 0.9]. Span 0.8. Step 0.16. ~5 cols.
        cols = 6 if row % 2 == 0 else 5
        
        for col in range(cols):
            if count >= n:
                break
            x = 0.1 + x_offset + col * 0.16
            y = 0.1 + row * 0.14
            
            # Check if inside [0,1]
            if 0 <= x <= 1 and 0 <= y <= 1:
                centers[count] = [x, y]
                count += 1
        row += 1
        
    # If grid didn't fill enough (unlikely with these params), fill remaining randomly
    while count < n:
        cx, cy = np.random.uniform(0.1, 0.9, 2)
        centers[count] = [cx, cy]
        count += 1

    # --- Optimization Loop ---
    # We will run gradient ascent/descent
    # Maximize sum(r) subject to constraints
    
    penalty_weight = 50.0 # Weight for constraints
    lr_c = 0.01           # Learning rate for centers
    lr_r = 0.005          # Learning rate for radii
    decay = 0.995         # Learning rate decay
    
    max_iters = 3000
    
    for i in range(max_iters):
        # Compute gradients/forces
        energy, force_C, ascent_dir_r, (e_overlap, e_bound) = \
            compute_overlaps_and_gradients(centers, radii, penalty_weight)
            
        # Update Centers: Move in direction of force (repulsion/boundary push)
        # Force is already computed as "push away" direction
        centers += lr_c * force_C
        
        # Update Radii: Move in direction of ascent_dir_r
        # If valid, ascent_dir_r ~ 1. If overlapping, ascent_dir_r < 0.
        radii += lr_r * ascent_dir_r
        
        # Clamp radii to be positive
        radii = np.maximum(radii, 1e-6)
        
        # Clamp centers to be somewhat inside to help stability, 
        # though boundary forces handle this.
        # Hard clamp can interfere with smooth gradient, but prevents explosion.
        # We rely on boundary forces primarily.
        
        # Decay learning rates slightly for fine-tuning
        lr_c *= decay
        lr_r *= decay
        
        # If energy is very low and stable, we can increase penalty weight 
        # to enforce stricter constraints, but constant weight is usually fine 
        # with proper learning rates.
        
        # Debug check (optional)
        if i % 500 == 0:
            # print(f"Iter {i}: Energy {energy:.4f}, Sum R {np.sum(radii):.4f}")
            pass

    # Final cleanup: ensure no NaNs or negative radii
    radii = np.maximum(radii, 0.0)
    
    # Calculate sum of radii
    sum_radii = float(np.sum(radii))
    
    # Convert to standard types if necessary
    centers = np.array(centers, dtype=np.float64)
    radii = np.array(radii, dtype=np.float64)
    
    return centers, radii, sum_radii
