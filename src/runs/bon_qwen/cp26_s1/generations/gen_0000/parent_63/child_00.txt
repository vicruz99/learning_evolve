import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    """
    n = 26
    best_sum_radii = -1.0
    best_centers = None
    best_radii = None
    
    # Penalty weight for overlaps and boundary violations
    penalty_weight = 5000.0
    
    def get_objective_and_gradient(z):
        """
        Calculates the objective function value and its gradient.
        z: [x1, y1, x2, y2, ..., x26, y26, r1, r2, ..., r26]
        """
        xs = z[:2 * n].reshape(n, 2)
        rs = z[2 * n:]
        
        obj_val = -np.sum(rs)
        grad = np.zeros_like(z)
        
        # Gradient contribution from objective: d/dr (-r) = -1
        grad[2 * n:] = -1.0
        
        # Wall constraints penalties and gradients
        for i in range(n):
            x, y = xs[i]
            r = rs[i]
            
            # Left wall: x - r >= 0  -> violation if x - r < 0
            if x - r < 0:
                error = x - r
                penalty = error ** 2
                obj_val += penalty_weight * penalty
                # d/dx: 2*error*1, d/dr: 2*error*(-1)
                grad[i * 2] += penalty_weight * 2 * error
                grad[i * 2 + 1] += 0.0 # y not involved
                grad[2 * n + i] += penalty_weight * 2 * error * (-1)
            
            # Right wall: 1 - (x + r) >= 0 -> violation if x + r > 1
            if x + r > 1:
                error = x + r - 1
                penalty = error ** 2
                obj_val += penalty_weight * penalty
                # d/dx: 2*error*1, d/dr: 2*error*1
                grad[i * 2] += penalty_weight * 2 * error
                grad[2 * n + i] += penalty_weight * 2 * error
                
            # Bottom wall: y - r >= 0
            if y - r < 0:
                error = y - r
                penalty = error ** 2
                obj_val += penalty_weight * penalty
                grad[i * 2 + 1] += penalty_weight * 2 * error
                grad[2 * n + i] += penalty_weight * 2 * error * (-1)
            
            # Top wall: 1 - (y + r) >= 0
            if y + r > 1:
                error = y + r - 1
                penalty = error ** 2
                obj_val += penalty_weight * penalty
                grad[i * 2 + 1] += penalty_weight * 2 * error
                grad[2 * n + i] += penalty_weight * 2 * error

        # Pairwise constraints penalties and gradients
        for i in range(n):
            xi, yi = xs[i]
            ri = rs[i]
            
            for j in range(i + 1, n):
                xj, yj = xs[j]
                rj = rs[j]
                
                dx = xi - xj
                dy = yi - yj
                dist = np.sqrt(dx * dx + dy * dy)
                
                # Avoid division by zero
                if dist < 1e-12:
                    dist = 1e-12
                    
                overlap = ri + rj - dist
                
                if overlap > 0:
                    penalty = overlap ** 2
                    obj_val += penalty_weight * penalty
                    
                    # Gradients
                    # d(overlap)/dri = 1, d(overlap)/drj = 1
                    # d(overlap)/dxi = -dx/dist, d(overlap)/dyi = -dy/dist
                    # d(overlap)/dxj = dx/dist, d(overlap)/dyj = dy/dist
                    
                    term = penalty_weight * 2 * overlap
                    
                    # Radius gradients
                    grad[2 * n + i] += term * 1.0
                    grad[2 * n + j] += term * 1.0
                    
                    # Position gradients
                    grad[i * 2] += term * (-dx / dist)
                    grad[i * 2 + 1] += term * (-dy / dist)
                    
                    grad[j * 2] += term * (dx / dist)
                    grad[j * 2 + 1] += term * (dy / dist)
        
        return obj_val, grad

    def optimize_from_init(init_centers, init_radii):
        z0 = np.concatenate([init_centers.flatten(), init_radii])
        bounds = [(0, 1)] * (2 * n) + [(0, 0.5)] * n
        
        res = minimize(
            fun=lambda z: get_objective_and_gradient(z)[0],
            x0=z0,
            jac=lambda z: get_objective_and_gradient(z)[1],
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 2000, 'ftol': 1e-15, 'gtol': 1e-10}
        )
        
        final_centers = res.x[:2 * n].reshape(n, 2)
        final_radii = res.x[2 * n:]
        
        return final_centers, final_radii, np.sum(final_radii)

    # Helper to check validity strictly
    def is_valid(centers, radii):
        # Boundary check
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-9 or x + r > 1 + 1e-9 or y - r < -1e-9 or y + r > 1 + 1e-9:
                return False
        # Overlap check
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist < radii[i] + radii[j] - 1e-9:
                    return False
        return True

    # Strategy 1: Hexagonal Grid Packing
    # Estimate radius for 26 circles in hex packing
    # Area approx: 26 * pi * r^2 approx 0.9 * 1 => r approx 0.105
    # But boundaries reduce this. Let's start with r=0.1
    r_est = 0.095
    centers_hex = []
    # Rows
    row_height = r_est * np.sqrt(3)
    # Try to fit rows
    y = r_est
    row_idx = 0
    while len(centers_hex) < n:
        if row_idx % 2 == 0:
            x_start = r_est
            count_in_row = int((1 - 2 * r_est) / (2 * r_est)) + 1
        else:
            x_start = 2 * r_est
            count_in_row = int((1 - 2 * r_est - r_est) / (2 * r_est)) + 1 # Offset by r_est effectively
        
        # Ensure we don't exceed width
        if count_in_row < 1:
            count_in_row = 1 # Fallback
            
        for k in range(count_in_row):
            if len(centers_hex) >= n:
                break
            x = x_start + k * 2 * r_est
            if x + r_est <= 1:
                centers_hex.append([x, y])
        
        y += row_height
        row_idx += 1
        if y + r_est > 1 and len(centers_hex) < n:
            # If we run out of vertical space but need more circles, squeeze slightly or shift
            # Just continue adding if we can fit even with overlap in init, optimizer will fix
            pass

    # Pad if needed (should not happen with logic above, but safety)
    while len(centers_hex) < n:
        centers_hex.append([0.5, 0.5])

    radii_hex = np.full(n, r_est)
    c1, r1, s1 = optimize_from_init(np.array(centers_hex[:n]), radii_hex[:n])
    if is_valid(c1, r1) and s1 > best_sum_radii:
        best_sum_radii = s1
        best_centers = c1
        best_radii = r1

    # Strategy 2: Perturbed Grid (5x5 + 1)
    centers_grid = []
    # 5x5 grid
    step = 0.18 # 1/5.5 approx
    for i in range(5):
        for j in range(5):
            if len(centers_grid) < n:
                centers_grid.append([0.1 + i * step, 0.1 + j * step])
    
    # Add 26th circle
    if len(centers_grid) < n:
        centers_grid.append([0.5, 0.5]) # Center
        
    radii_grid = np.full(n, 0.08)
    c2, r2, s2 = optimize_from_init(np.array(centers_grid[:n]), radii_grid[:n])
    if is_valid(c2, r2) and s2 > best_sum_radii:
        best_sum_radii = s2
        best_centers = c2
        best_radii = r2

    # Strategy 3: Random Initialization (for diversity)
    np.random.seed(42) # For reproducibility
    for _ in range(5):
        # Random centers in [0.2, 0.8] to start away from walls
        rand_centers = np.random.uniform(0.2, 0.8, (n, 2))
        # Random small radii
        rand_radii = np.full(n, 0.05)
        
        c3, r3, s3 = optimize_from_init(rand_centers, rand_radii)
        if is_valid(c3, r3) and s3 > best_sum_radii:
            best_sum_radii = s3
            best_centers = c3
            best_radii = r3

    # Final cleanup: clamp values to [0,1] strictly just in case of float errors
    if best_centers is not None:
        best_centers = np.clip(best_centers, 0, 1)
        best_radii = np.clip(best_radii, 0, 0.5)
        
    # If no valid solution found (unlikely), return a trivial one
    if best_centers is None:
        trivial_centers = np.random.uniform(0.1, 0.9, (n, 2))
        trivial_radii = np.full(n, 0.04)
        return trivial_centers, trivial_radii, np.sum(trivial_radii)

    return best_centers, best_radii, float(best_sum_radii)

# Run the packing optimization
if __name__ == "__main__":
    centers, radii, total_r = run_packing()
    print(f"Packed {len(centers)} circles.")
    print(f"Sum of radii: {total_r}")
