import numpy as np
from scipy.optimize import minimize

def run_packing():
    """
    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    
    Returns:
        centers: np.array of shape (26, 2)
        radii: np.array of shape (26,)
        sum_radii: float
    """
    n = 26
    
    # 1. Initialization Strategy
    # Start with a 5x5 grid (25 circles) with radius 0.1.
    # Add a 26th circle in a gap.
    
    grid_coords = np.array([0.1, 0.3, 0.5, 0.7, 0.9])
    centers_init = []
    for y in grid_coords:
        for x in grid_coords:
            centers_init.append([x, y])
            
    # Add 26th circle in a gap (0.2, 0.2)
    # Nearest neighbor is at (0.1, 0.1), dist ~ 0.1414. 
    # With r=0.1, gap allows r ~ 0.0414.
    centers_init.append([0.2, 0.2])
    
    centers_init = np.array(centers_init)
    radii_init = np.full(n, 0.1)
    radii_init[-1] = 0.04  # Radius for the 26th circle
    
    # Variables vector: [x0, y0, r0, x1, y1, r1, ...]
    # Size: 3 * n = 78
    x0 = centers_init[:, 0]
    y0 = centers_init[:, 1]
    r0 = radii_init
    
    v0 = np.array([x0[i], y0[i], r0[i] for i in range(n)])
    
    # Bounds: x, y in [0, 1], r in [0, 0.5]
    bounds = [(0, 1)] * n + [(0, 1)] * n + [(0, 0.5)] * n
    # Re-structuring bounds to match v0 layout:
    # v0 is [x0..xn, y0..yn, r0..rn] ? No, the construction above creates [x0, y0, r0, x1, y1, r1...]?
    # Let's check v0 construction:
    # v0 = [x0[0], y0[0], r0[0], x0[1], y0[1], r0[1] ...]
    # So bounds should be interleaved.
    bounds = []
    for _ in range(n):
        bounds.extend([(0, 1), (0, 1), (0, 0.5)])
        
    # 2. Objective Function
    def objective(v):
        r = v[2::3]
        return -np.sum(r) # Minimize negative sum = Maximize sum

    # 3. Constraints
    constraints = []
    
    # Wall constraints: r <= x <= 1-r, r <= y <= 1-r
    # This implies x - r >= 0, 1 - x - r >= 0, etc.
    # Inequality constraint form: func(v) >= 0
    for i in range(n):
        idx_x = 3 * i
        idx_y = 3 * i + 1
        idx_r = 3 * i + 2
        
        # x - r >= 0
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: v[idx_x] - v[idx_r]})
        # 1 - x - r >= 0
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: 1.0 - v[idx_x] - v[idx_r]})
        # y - r >= 0
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: v[idx_y] - v[idx_r]})
        # 1 - y - r >= 0
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: 1.0 - v[idx_y] - v[idx_r]})

    # Non-overlap constraints: dist >= r_i + r_j
    # (x_i - x_j)^2 + (y_i - y_j)^2 >= (r_i + r_j)^2
    # => (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
    for i in range(n):
        for j in range(i + 1, n):
            idx_xi, idx_yi, idx_ri = 3 * i, 3 * i + 1, 3 * i + 2
            idx_xj, idx_yj, idx_rj = 3 * j, 3 * j + 1, 3 * j + 2
            
            def constraint_pair(v, i=i, j=j):
                xi, yi, ri = v[idx_xi], v[idx_yi], v[idx_ri]
                xj, yj, rj = v[idx_xj], v[idx_yj], v[idx_rj]
                
                dist_sq = (xi - xj)**2 + (yi - yj)**2
                sum_r_sq = (ri + rj)**2
                return dist_sq - sum_r_sq

            constraints.append({'type': 'ineq', 'fun': constraint_pair})

    # 4. Optimization
    # Using SLSQP which supports bounds and constraints
    result = minimize(objective, v0, method='SLSQP', bounds=bounds, constraints=constraints, 
                      options={'ftol': 1e-9, 'maxiter': 1000, 'disp': False})
    
    # Extract results
    v_opt = result.x
    centers_opt = np.zeros((n, 2))
    radii_opt = np.zeros(n)
    
    for i in range(n):
        centers_opt[i, 0] = v_opt[3 * i]
        centers_opt[i, 1] = v_opt[3 * i + 1]
        radii_opt[i] = v_opt[3 * i + 2]
        
    sum_radii = np.sum(radii_opt)
    
    return centers_opt, radii_opt, sum_radii
