import numpy as np

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    n = centers.shape[0]
    if np.isnan(centers).any() or np.isnan(radii).any():
        return False
    for i in range(n):
        if radii[i] < 0: return False
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            return False
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False
    return True

def get_penalty(centers, radii):
    n = centers.shape[0]
    penalty = 0.0
    # Boundary penalties
    # We want r <= x, r <= 1-x, r <= y, r <= 1-y
    # Violation amount: max(0, r - margin)
    # To make it smooth, we can use a soft penalty or just hard constraint in solver.
    # But for penalty method:
    
    # Boundary: r - x < 0 => x - r >= 0. Violation if x - r < 0.
    # Violation amount = max(0, r - x)
    
    # Let's use a large multiplier for penalties
    P_MULT = 1000.0
    
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        
        # Boundary checks
        violations = []
        violations.append(max(0.0, r - x))
        violations.append(max(0.0, r - (1.0 - x)))
        violations.append(max(0.0, r - y))
        violations.append(max(0.0, r - (1.0 - y)))
        
        for v in violations:
            penalty += P_MULT * (v ** 2)
            
    # Overlap checks
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            sum_r = radii[i] + radii[j]
            # We need dist >= sum_r. Violation if dist < sum_r.
            # Violation amount = max(0, sum_r - dist)
            v = max(0.0, sum_r - dist)
            if v > 0:
                penalty += P_MULT * (v ** 2)
                
    return penalty

def objective_function(params, n):
    # params contains [x1, y1, r1, x2, y2, r2, ...]
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    for i in range(n):
        centers[i, 0] = params[3*i]
        centers[i, 1] = params[3*i + 1]
        radii[i] = params[3*i + 2]
        
    # Objective: Maximize sum of radii => Minimize -sum(radii)
    obj = -np.sum(radii)
    
    # Add penalty
    pen = get_penalty(centers, radii)
    
    return obj + pen
