from scipy.optimize import minimize

# Constant for number of circles
N_CIRCLES = 26

def objective(vars, n):
    """
    Objective function to minimize: negative sum of radii.
    vars layout: [x0, y0, x1, y1, ..., x(n-1), y(n-1), r0, r1, ..., r(n-1)]

    Constraints:
    1. x_i - r_i >= 0
    2. 1 - x_i - r_i >= 0
    3. y_i - r_i >= 0
