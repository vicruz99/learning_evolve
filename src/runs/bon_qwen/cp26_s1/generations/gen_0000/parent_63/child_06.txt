import numpy as np

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    Packs 26 circles in a unit square [0,1]x[0,1] to maximize sum of radii.
    Uses a force-directed growing circles algorithm with multiple restarts.
    """
    
    def validate_internal(centers, radii):

        Helper to check validity quickly (without printing)
        """
        n = centers.shape[0]
        # Check boundaries
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-9 or x + r > 1 + 1e-9 or y - r < -1e-9 or y + r > 1 + 1e-9:
                return False
        # Check overlaps
        # Vectorized distance check
        # dist_matrix[i, j] = distance between i and j
        # We need dist >= r_i + r_j
        # dist^2 >= (r_i + r_j)^2
        
        # Compute pairwise squared distances
        # diff = centers[:, None, :] - centers[None, :, :] # Shape (26, 26, 2)
        # dist_sq = np.sum(diff**2, axis=2)
        
        # Compute required squared distances
        # sum_r = radii[:, None] + radii[None, :]
        # req_sq = sum_r**2
        
        # Check if dist_sq < req_sq - tolerance
        # We use a mask for i < j to check upper triangle
        
        # Optimization: Loop is fine for N=26
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii[i] + radii[j] - 1e-9:
                    return False
        return True

    def resolve_conflicts(centers, radii, iterations=200):
        """
        Pushes circles apart to resolve overlaps and boundary violations.
        """
        n = centers.shape[0]
        
        for _ in range(iterations):
            moved = False
            
            # 1. Resolve Pairwise Overlaps
            # We iterate through all pairs. If overlap, push apart.
            # To avoid bias, we can process in random order or fixed order.
            # Fixed order is deterministic.
            for i in range(n):
                for j in range(i + 1, n):
                    dx = centers[i, 0] - centers[j, 0]
                    dy = centers[i, 1] - centers[j, 1]
                    dist_sq = dx*dx + dy*dy
                    
                    min_dist = radii[i] + radii[j]
                    
                    if dist_sq < min_dist * min_dist:
                        # Overlap detected
                        # Calculate vector direction
                        dist = np.sqrt(dist_sq)
                        if dist > 1e-12:
                            nx = dx / dist
                            ny = dy / dist
                        else:
                            # Centers are coincident, push randomly
                            nx = np.random.uniform(-1, 1)
                            ny = np.random.uniform(-1, 1)
                            norm = np.sqrt(nx*nx + ny*ny)
                            if norm > 0:
                                nx /= norm
                                ny /= norm
                            else:
                                nx, ny = 1.0, 0.0
                        
                        overlap = min_dist - dist
                        # Push apart by half the overlap (assuming equal stiffness)
                        # Adding a small factor to ensure separation
                        push = overlap * 0.5 + 1e-7 
                        
                        centers[i, 0] -= nx * push
                        centers[i, 1] -= ny * push
                        centers[j, 0] += nx * push
                        centers[j, 1] += ny * push
                        moved = True

            # 2. Resolve Boundary Violations
            for i in range(n):
                x, y = centers[i]
                r = radii[i]
                
                # X boundaries
                if x - r < 0:
                    centers[i, 0] = r
                    moved = True
                elif x + r > 1:
                    centers[i, 0] = 1 - r
                    moved = True
                
                # Y boundaries
                if y - r < 0:
                    centers[i, 1] = r
                    moved = True
                elif y + r > 1:
                    centers[i, 1] = 1 - r
                    moved = True
            
            if not moved:
                break
                
        return centers

    def init_random_packing(n_circles=26, r_init=0.04, max_tries=10000):
        """
        Generates an initial valid packing using rejection sampling.
        """
        centers = []
        radii = []
        
        # Try to place circles one by one
        attempts = 0
        while len(centers) < n_circles and attempts < max_tries:
            attempts += 1
            # Random position within valid bounds for radius r_init
            x = np.random.uniform(r_init, 1 - r_init)
            y = np.random.uniform(r_init, 1 - r_init)
            
            valid = True
            for k in range(len(centers)):
                dx = x - centers[k][0]
                dy = y - centers[k][1]
                dist = np.sqrt(dx*dx + dy*dy)
                # Check against existing radius + new radius
                if dist < radii[k] + r_init:
                    valid = False
                    break
            
            if valid:
                centers.append([x, y])
                radii.append(r_init)
        
        if len(centers) < n_circles:
            return None, None
        
        return np.array(centers), np.array(radii)

    def optimize(centers, radii):
        """
        Grows circles and resolves conflicts to maximize sum of radii.

        min_delta_r = 1e-6

        #    No, resolve_conflicts pushes centers. It maintains radii.
        #    But if circles are trapped against walls, they can't grow.

        # But resolve_conflicts should handle it by pushing.
        
        # We just keep increasing and pushing.
        
        for step in range(2000): # Max steps
            radii += delta_r
            
            # Resolve conflicts
            centers = resolve_conflicts(centers, radii, iterations=100)
            
            # Check validity (should be valid after resolve)
            # However, resolve_conflicts might not be perfect if pushed against walls?
            # It clamps centers, which might cause overlaps.
            # Let's run resolve_conflicts again after clamping?

            # If radii are too large, centers might be pushed out of bounds?

            # If we can't resolve overlaps (e.g. 3 circles jammed), we might need to reduce radii.
            # But simple push usually works for convex shapes.
            
            # Decay step size to settle
            delta_r *= decay_factor
            if delta_r < min_delta_r:
                delta_r = min_delta_r
                
        return centers, radii

    best_sum = -1.0
    best_centers = None
    best_radii = None

    # Run multiple restarts
    num_restarts = 15
    
    for restart in range(num_restarts):
        # Random seed variation implicitly via np.random state
        # Or explicitly set? No, let it be random.
        
        # Try to initialize
        init_r = 0.04
        # Sometimes init_r might be too large for rejection sampling if we are unlucky

        centers, radii = init_random_packing(26, init_r)

            continue
            
        # Optimize
        opt_centers, opt_radii = optimize(centers.copy(), radii.copy())
        
        # Final cleanup pass with tight resolution
        opt_centers = resolve_conflicts(opt_centers, opt_radii, iterations=500)
        
        # Calculate sum
        current_sum = np.sum(opt_radii)
        
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = opt_centers.copy()
            best_radii = opt_radii.copy()
            
        # Optional: If we found a very good sum, maybe stop?

        if best_sum >= 2.636:
            break
            
    # Final validation check and minor correction
    # Ensure strict validity
    # Sometimes numerical noise leaves tiny overlaps.
    # resolve_conflicts with high iterations should fix it.
    best_centers = resolve_conflicts(best_centers, best_radii, iterations=1000)

    best_radii = np.maximum(best_radii, 0.0)

    for i in range(26):

        x = best_centers[i, 0]
        y = best_centers[i, 1]
        
        # Clamp center to be inside [r, 1-r]
        x = max(r, min(1.0 - r, x))
        y = max(r, min(1.0 - r, y))
        best_centers[i, 0] = x
        best_centers[i, 1] = y

    return best_centers, best_radii, float(np.sum(best_radii))
