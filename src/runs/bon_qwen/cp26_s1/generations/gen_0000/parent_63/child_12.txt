import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def compute_constraints(v):
    """
    Computes all inequality constraints for the circle packing problem.
    Returns an array where each element must be >= 0.
    """
    c = []
    for i in range(N_CIRCLES):
        xi, yi, ri = v[3*i], v[3*i+1], v[3*i+2]
        # Pairwise non-overlap constraints
        for j in range(i+1, N_CIRCLES):
            xj, yj, rj = v[3*j], v[3*j+1], v[3*j+2]
            dist = np.hypot(xi - xj, yi - yj)
            c.append(dist - ri - rj)
        # Boundary constraints
        c.append(xi - ri)
        c.append(1.0 - xi - ri)
        c.append(yi - ri)
        c.append(1.0 - yi - ri)
    return np.array(c)

def compute_objective(v):
    """
    Objective function: negative sum of radii (to be minimized).
    """
    return -np.sum(v[2::3])

def run_packing():
    # Generate initial hexagonal grid placement
    centers = []
    r_init = 0.08
    dy = 2 * r_init * 1.1
    dx = np.sqrt(3) * r_init * 1.1
    
    y = 0.12
    shift = 0.0
    while len(centers) < N_CIRCLES:
        x = 0.12 + shift
        while x <= 0.88 and len(centers) < N_CIRCLES:
            centers.append([x, y])
            x += dx
        y += dy
        shift = dx / 2.0 if shift == 0.0 else 0.0
        
    # Fill remaining spots if any (should be exactly 26 with above loop)
    while len(centers) < N_CIRCLES:
        centers.append([0.5, 0.5])
        
    # Flatten initial guess into optimization vector [x1, y1, r1, x2, y2, r2, ...]
    v0 = np.zeros(N_CIRCLES * 3)
    for i in range(N_CIRCLES):
        v0[3*i] = centers[i][0]
        v0[3*i+1] = centers[i][1]
        v0[3*i+2] = 0.05  # Start with small valid radii
        
    # Bounds for variables: x,y in [0,1], r in [0, 0.5]
    bounds = [(0.0, 1.0)] * (2*N_CIRCLES) + [(0.0, 0.5)] * N_CIRCLES
    
    # Constraint definition for scipy
    cons = {'type': 'ineq', 'fun': compute_constraints}
    
    # Run optimization
    res = minimize(
        compute_objective, 
        v0, 
        method='SLSQP', 
        bounds=bounds, 
        constraints=cons, 
        options={'maxiter': 5000, 'ftol': 1e-12}
    )
    
    v_best = res.x
    best_centers = np.array([[v_best[3*i], v_best[3*i+1]] for i in range(N_CIRCLES)])
    best_radii = v_best[2::3]
    
    # Ensure radii are strictly non-negative
    best_radii = np.maximum(best_radii, 0.0)
    
    return best_centers, best_radii, np.sum(best_radii)
