def initialize_hexagonal(n):
    """Place circles in a hexagonal grid pattern with small random perturbation."""
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.008)

    idx = 0
    spacing = 0.165
    for row in range(7):
        for col in range(6):
            if idx >= n:
                break
            x = 0.06 + col * spacing + (row % 2) * spacing / 2
            y = 0.06 + row * spacing * np.sqrt(3) / 2
            if x < 0.96 and y < 0.96:
                centers[idx] = [x, y] + np.random.uniform(-0.005, 0.005, 2)
                centers[idx] = np.clip(centers[idx], 0.02, 0.98)
                idx += 1

    return centers, radii


def simulate_forces(centers, radii, n, iters=50000):
    """
    Physics-based optimization: circles repel when overlapping, expand over time.
    """
    for t in range(iters):
        fx = np.zeros(n)
        fy = np.zeros(n)
        fr = np.zeros(n)

        # Constant expansion drive
        expansion = 6e-5

        for i in range(n):
            fr[i] = expansion

            x, y = centers[i]
            r = radii[i]

            # Boundary spring forces
            margin = 1e-6
            kb = 3000
            if x - r < margin:
                fx[i] -= kb * (margin - (x - r))
            if x + r > 1.0 - margin:
                fx[i] += kb * (x + r - (1.0 - margin))
            if y - r < margin:
                fy[i] -= kb * (margin - (y - r))
            if y + r > 1.0 - margin:
                fy[i] += kb * (y + r - (1.0 - margin))

            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist_sq = dx * dx + dy * dy
                dist = np.sqrt(dist_sq)

                if dist < 1e-12:
                    dist = 1e-12
                    dx, dy = 1.0, 0.0

                required = radii[i] + radii[j]
                overlap = required - dist

                if overlap > -0.012:
                    if overlap > 0:
                        # Strong repulsion proportional to overlap
                        k = 2500
                        f = k * overlap / dist
                        fx[i] += f * dx
                        fy[i] += f * dy
                        fx[j] -= f * dx
                        fy[j] -= f * dy

                        # Grow radii to compensate for repulsion
                        fr[i] += 2.5 * overlap
                        fr[j] += 2.5 * overlap
                    else:
                        # Gentle repulsion to maintain spacing
                        f = 40 * (0.012 + overlap) / dist
                        fx[i] += f * dx
                        fy[i] += f * dy
                        fx[j] -= f * dx
                        fy[j] -= f * dy

        # Cooling schedule: step size decreases over time
        step = 0.015 / (1.0 + t * 0.00015)

        centers[:, 0] += fx * step
        centers[:, 1] += fy * step
        radii += fr * step

        # Clip to valid ranges
        radii = np.maximum(radii, 1e-9)
        centers = np.clip(centers, 1e-8, 1.0 - 1e-8)

        # Enforce circle-inside-square constraint
        for i in range(n):
            max_r = min(
                centers[i, 0],
                1.0 - centers[i, 0],
                centers[i, 1],
                1.0 - centers[i, 1]
            )
            radii[i] = min(radii[i], max_r - 1e-10)

    return centers, radii


def enforce_all_constraints(centers, radii, n):
    """
    Iteratively reduce radii to eliminate any remaining overlaps or boundary violations.
    """
    for iteration in range(500):
        changed = False

        # Fix overlaps
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)

                min_dist = radii[i] + radii[j]
                if dist < min_dist:
                    reduction = (min_dist - dist) / 2.0 + 1e-12
                    radii[i] -= reduction
                    radii[j] -= reduction
                    radii[i] = max(radii[i], 1e-12)
                    radii[j] = max(radii[j], 1e-12)
                    changed = True

        # Fix boundary violations
        for i in range(n):
            max_r = min(
                centers[i, 0],
                1.0 - centers[i, 0],
                centers[i, 1],
                1.0 - centers[i, 1]
            )
            if radii[i] > max_r - 1e-12:
                radii[i] = max(max_r - 1e-12, 1e-12)
                changed = True

        if not changed:
            break

    return centers, radii
