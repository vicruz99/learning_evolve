   - Non-negativity: $r_i \ge 0$
2. **Solver:** We use `scipy.optimize.minimize` with the `SLSQP` method, which handles nonlinear inequality constraints efficiently.

```python
import numpy as np
from scipy.optimize import minimize

N = 26
# Precompute indices for pairwise constraints (upper triangle of distance matrix)
PAIR_IDX = np.triu_indices(N, k=1)

def objective(v):
    """Minimize negative sum of radii to maximize total radius length."""
    return -np.sum(v[2*N:])

def constraints(v):
    """
    Vectorized inequality constraints:
    - Boundary: x_i >= r_i, x_i <= 1-r_i, y_i >= r_i, y_i <= 1-r_i
    - Pairwise: dist_ij^2 >= (r_i + r_j)^2
    All constraints are formulated to return values >= 0.
    """
    xc = v[:2*N].reshape(N, 2)
    r = v[2*N:]
    
    c = []
    # Boundary constraints
    c.extend(xc - r[:, None])          # x >= r
    c.extend(1.0 - xc - r[:, None])    # x <= 1 - r
    
    # Pairwise non-overlap constraints
    diff = xc[:, None, :] - xc[None, :, :]  # Shape (N, N, 2)
    dist_sq = np.sum(diff**2, axis=2)        # Shape (N, N)
    r_sum = r[:, None] + r[None, :]          # Shape (N, N)
    
    val = dist_sq - r_sum**2                 # Must be >= 0
    c.extend(val[PAIR_IDX])                  # Extract unique pairs
    
    return np.concatenate(c)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    
    # Structured initialization: 5x5 grid + 1 central circle
    xs = np.linspace(0.15, 0.85, 5)
    ys = np.linspace(0.15, 0.85, 5)
    cx, cy = np.meshgrid(xs, ys)
    centers = np.vstack([cx.flatten(), cy.flatten()]).T
    centers = np.vstack([centers, [0.5, 0.5]])  # 26th circle
    
    # Initial radii slightly smaller than half the grid spacing to ensure feasibility
    radii = np.full(N, 0.085)
    
    x0 = np.concatenate([centers.flatten(), radii])
    
    bounds = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N
    cons = {'type': 'ineq', 'fun': constraints}
    
    # Optimize
    res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': 3000, 'ftol': 1e-12, 'disp': False})
                   
    xc = res.x[:2*N].reshape(N, 2)
    r = res.x[2*N:]
    
    # Post-processing for strict numerical validity
    r = np.maximum(r, 1e-9)
    xc = np.clip(xc, r[:, None], 1.0 - r[:, None])
    
    return xc, r, float(np.sum(r))
```
