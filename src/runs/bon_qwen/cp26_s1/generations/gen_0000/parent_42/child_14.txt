import numpy as np
from scipy.optimize import minimize
import math

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # 1. Initialize with a hexagonal lattice pattern
    # Estimate radius to fit roughly in square
    r_init = 0.06
    centers = []
    
    # Hexagonal grid generation
    # Vertical spacing: r * sqrt(3)
    # Horizontal spacing: 2 * r
    # Shift odd rows by r horizontally
    
    y = r_init
    row_idx = 0
    while y < 1.0 - r_init:
        if row_idx % 2 == 0:
            x_start = r_init
        else:
            x_start = r_init + r_init # shift by r
        
        x = x_start
        while x < 1.0 - r_init:
            if len(centers) < n:
                centers.append([x, y])
            x += 2 * r_init
        y += r_init * math.sqrt(3)
        row_idx += 1
        
    # If we didn't get enough points (unlikely with r=0.06), fill with random
    while len(centers) < n:
        centers.append([np.random.rand(), np.random.rand()])
        
    centers = np.array(centers[:n])
    radii = np.ones(n) * r_init
    
    # Flatten variables for optimizer: [x1, y1, r1, x2, y2, r2, ...]
    x0 = np.hstack([centers.flatten(), radii])
    
    # Bounds: x, y in [0, 1], r in [0, 0.5]
    bounds = []
    for _ in range(n):
        bounds.extend([(0, 1), (0, 1), (0, 0.5)])
        
    # Penalty weight
    penalty_weight = 10000.0
    
    def objective(x):
        # Unpack
        centers = x[:2*n].reshape(n, 2)
        radii = x[2*n:]
        
        # Objective: Minimize negative sum of radii
        obj = -np.sum(radii)
        
        # Penalty term
        pen = 0.0
        
        # Boundary constraints
        # Circle i must be inside [0, 1]x[0, 1]
        # 0 <= x - r  => r - x <= 0
        # x + r <= 1  => x + r - 1 <= 0
        for i in range(n):
            cx, cy = centers[i]
            r = radii[i]
            
            # Boundary violations
            v1 = r - cx
            if v1 > 0: pen += v1**2
            
            v2 = r - (1.0 - cx)
            if v2 > 0: pen += v2**2
            
            v3 = r - cy
            if v3 > 0: pen += v3**2
            
            v4 = r - (1.0 - cy)
            if v4 > 0: pen += v4**2
            
        # Overlap constraints
        # dist(c_i, c_j) >= r_i + r_j
        # r_i + r_j - dist <= 0
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                min_dist = radii[i] + radii[j]
                v = min_dist - dist
                if v > 0:
                    pen += v**2
                    
        return obj + penalty_weight * pen

    # Optimization
    best_res = None
    best_sum_radii = -np.inf
    
    # Run optimization a few times with slight perturbations to avoid local minima
    # First run from initialized hex grid
    for restart in range(5):
        if restart > 0:
            # Perturb the best found solution so far
            perturbation = np.random.normal(0, 0.01, size=x0.shape)
            # Clip bounds
            x_start = best_res.x.copy()
            # x, y bounds [0,1], r bounds [0, 0.5]
            for k in range(0, 2*n, 2):
                x_start[k] = np.clip(x_start[k] + perturbation[k], 0, 1)
                x_start[k+1] = np.clip(x_start[k+1] + perturbation[k+1], 0, 1)
            for k in range(2*n, 2*n + n):
                x_start[k] = np.clip(x_start[k] + perturbation[k], 0, 0.5)
        else:
            x_start = x0.copy()

        res = minimize(objective, x_start, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': 2000, 'ftol': 1e-10, 'gtol': 1e-6})
        
        # Calculate actual sum of radii for the result
        res_centers = res.x[:2*n].reshape(n, 2)
        res_radii = res.x[2*n:]
        current_sum = np.sum(res_radii)
        
        # Check validity roughly (penalty should be small)
        # We check the objective value which includes penalty
        # If penalty is small, the sum is trusted
        
        if current_sum > best_sum_radii:
            # Verify if penalty is low enough to be considered valid
            # Re-evaluate penalty to be sure
            centers_check = res_centers
            radii_check = res_radii
            pen_val = 0.0

                r = radii_check[i]
                if r - cx > 1e-6: valid = False

                    dist = np.sqrt(np.sum((centers_check[i] - centers_check[j])**2))
                    if dist < radii_check[i] + radii_check[j] - 1e-6:
                        valid = False
            
            if valid:
                best_sum_radii = current_sum
                best_res = res
    
    # Extract final solution
    final_centers = best_res.x[:2*n].reshape(n, 2)

    final_radii = np.maximum(final_radii, 0.0)
    
    return final_centers, final_radii, np.sum(final_radii)
