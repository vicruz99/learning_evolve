import numpy as np
from scipy.optimize import linprog, minimize


def optimize_radii(centers, n):
    """Given centers, find optimal radii using LP."""
    A_ub = []
    b_ub = []
    
    # Pairwise non-overlap constraints: r_i + r_j <= distance(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            b_ub.append(dist)
    
    # Boundary constraints: r_i <= min(x_i, 1-x_i, y_i, 1-y_i)
    for i in range(n):
        row = np.zeros(n)
        row[i] = 1.0
        A_ub.append(row)
        max_r = min(centers[i, 0], 1.0 - centers[i, 0],
                    centers[i, 1], 1.0 - centers[i, 1])
        b_ub.append(max(0.0, max_r))
    
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    c = -np.ones(n)  # Minimize -sum(r_i) = maximize sum(r_i)
    bounds = [(0.0, None)] * n
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        return np.maximum(result.x, 0.0)
    return np.zeros(n)


def optimize_centers(centers, radii, n):
    """Given radii, optimize centers to minimize overlaps."""
    def objective(x):
        c = x.reshape(n, 2)
        penalty = 0.0
        
        # Overlap penalty
        for i in range(n):
            for j in range(i + 1, n):
                dx = c[i, 0] - c[j, 0]
                dy = c[i, 1] - c[j, 1]
                dist = np.sqrt(dx * dx + dy * dy + 1e-12)
                overlap = radii[i] + radii[j] - dist
                if overlap > 0:
                    penalty += overlap * overlap
        
        # Boundary penalty
        for i in range(n):
            x_i, y_i = c[i]
            r_i = radii[i]
            if r_i - x_i > 0:
                penalty += (r_i - x_i) ** 2
            if x_i + r_i - 1.0 > 0:
                penalty += (x_i + r_i - 1.0) ** 2
            if r_i - y_i > 0:
                penalty += (r_i - y_i) ** 2
            if y_i + r_i - 1.0 > 0:
                penalty += (y_i + r_i - 1.0) ** 2
        
        return penalty
    
    x0 = centers.flatten()
    bounds_opt = [(0.0, 1.0)] * (2 * n)
    
    result = minimize(objective, x0, bounds=bounds_opt, method='L-BFGS-B',
                      options={'maxiter': 5000, 'ftol': 1e-16})
    
    return result.x.reshape(n, 2)


def init_hexagonal(n):
    """Initialize centers in hexagonal pattern."""
    centers = np.zeros((n, 2))
    idx = 0
    row = 0
    while idx < n:
        cols = 6 if row % 2 == 0 else 5
        for col in range(cols):
            if idx >= n:
                break
            x = 0.10 + col * 0.15 + (0.075 if row % 2 == 1 else 0)
            y = 0.10 + row * 0.13
            centers[idx] = [x, y]
            idx += 1
        row += 1
    return centers


def init_structured(n):
    """Initialize with structured placement favoring corners and edges."""
    centers = np.zeros((n, 2))
    idx = 0
    
    # Corners
    corners = [(0.10, 0.10), (0.90, 0.10), (0.10, 0.90), (0.90, 0.90)]
    for c in corners:
        centers[idx] = c
        idx += 1
    
    # Edge points
    edges = [(0.50, 0.10), (0.50, 0.90), (0.10, 0.50), (0.90, 0.50),
             (0.30, 0.10), (0.70, 0.10), (0.30, 0.90), (0.70, 0.90),
             (0.10, 0.30), (0.10, 0.70), (0.90, 0.30), (0.90, 0.70)]
    for e in edges:
        if idx < n:
            centers[idx] = e
            idx += 1
    
    # Interior points in grid
    for i in range(3):
        for j in range(4):
            if idx < n:
                centers[idx] = [0.30 + i * 0.20, 0.25 + j * 0.17]
                idx += 1
    
    return centers


def init_grid(n):
    """Initialize with regular grid."""
    centers = np.zeros((n, 2))
    idx = 0
    # Try 5x6 grid, take first 26
    for row in range(6):
        for col in range(5):
            if idx >= n:
                break
            x = 0.1 + col * 0.2
            y = 0.1 + row * 0.16
            centers[idx] = [x, y]
            idx += 1
        if idx >= n:
            break
    return centers


def run_packing():
    n = 26
    
    best_sum = 0.0
    best_centers = np.zeros((n, 2))
    best_radii = np.zeros(n)
    
    initializers = [init_hexagonal, init_structured, init_grid]
    
    for restart in range(30):

        # Choose initialization strategy

            centers = initializers[restart](n)

            centers = np.random.uniform(0.15, 0.85, (n, 2))
        
        # Run alternating optimization
        for iteration in range(150):
            radii = optimize_radii(centers, n)
            
            if np.sum(radii) < 1e-12:
                break

            if iteration > 0 and np.max(np.abs(radii - prev_radii)) < 1e-10:
                pass  # Could break early, but continue for safety
            
            prev_radii = radii.copy()
            centers = optimize_centers(centers, radii, n)
            
            # Perturbation to escape local minima
            if iteration > 0 and iteration % 25 == 20:
                noise_scale = 0.015 * (1.0 - iteration / 150.0)
                perturbation = np.random.normal(0, noise_scale, centers.shape)
                centers = np.clip(centers + perturbation, 0.01, 0.99)
        
        # Final radius optimization
        radii = optimize_radii(centers, n)
        current_sum = np.sum(radii)
        
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers.copy()

    # Ensure non-negative radii
    best_radii = np.maximum(best_radii, 0.0)
    
    # Clip centers to valid range
    best_centers = np.clip(best_centers, 0.0, 1.0)
    
    return best_centers, best_radii, best_sum
