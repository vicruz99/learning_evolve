   - Objective: Maximize sum of radii
   - Constraints: Boundary containment and non-overlap

3. **Multi-start strategy**: Run optimization from multiple initial configurations (different hexagonal spacings and random perturbations) to escape local optima.

   - distance(c_i, c_j) >= r_i + r_j (non-overlap)

import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def objective(x):
    n = N_CIRCLES
    radii = x[2 * n:]
    return -np.sum(radii)

def constraint_x_min(x, i):
    return x[2 * i] - x[2 * N_CIRCLES + i]

def constraint_x_max(x, i):
    return 1.0 - x[2 * i] - x[2 * N_CIRCLES + i]

def constraint_y_min(x, i):
    return x[2 * i + 1] - x[2 * N_CIRCLES + i]

def constraint_y_max(x, i):
    return 1.0 - x[2 * i + 1] - x[2 * N_CIRCLES + i]

def constraint_no_overlap(x, i, j):
    c1x = x[2 * i]
    c1y = x[2 * i + 1]
    c2x = x[2 * j]
    c2y = x[2 * j + 1]
    r1 = x[2 * N_CIRCLES + i]
    r2 = x[2 * N_CIRCLES + j]
    dx = c1x - c2x
    dy = c1y - c2y
    dist = np.sqrt(dx * dx + dy * dy)
    return dist - r1 - r2

def build_constraints(n):
    cons = []
    for i in range(n):
        cons.append({'type': 'ineq', 'fun': constraint_x_min, 'args': (i,)})
        cons.append({'type': 'ineq', 'fun': constraint_x_max, 'args': (i,)})
        cons.append({'type': 'ineq', 'fun': constraint_y_min, 'args': (i,)})
        cons.append({'type': 'ineq', 'fun': constraint_y_max, 'args': (i,)})
    for i in range(n):
        for j in range(i + 1, n):
            cons.append({'type': 'ineq', 'fun': constraint_no_overlap, 'args': (i, j)})
    return cons

def build_bounds(n):
    center_bounds = [(0.001, 0.999) for _ in range(2 * n)]
    radius_bounds = [(0.0001, 0.5) for _ in range(n)]
    return center_bounds + radius_bounds

def init_hexagonal(n, spacing):
    centers = []
    row = 0
    while len(centers) < n and row < 10:
        col = 0
        while len(centers) < n and col < 10:
            x = col * spacing + (row % 2) * spacing / 2.0
            y = row * spacing * np.sqrt(3) / 2.0
            if x < 0.95 and y < 0.95:
                centers.append([x, y])
            col += 1
        row += 1
    while len(centers) < n:
        centers.append([0.1 + np.random.random() * 0.8, 0.1 + np.random.random() * 0.8])
    return np.array(centers[:n])

def optimize_from_start(x0, constraints, bounds):
    result = minimize(objective, x0, method='SLSQP',
                     bounds=bounds, constraints=constraints,
                     options={'maxiter': 3000, 'ftol': 1e-14, 'disp': False})
    return result

def run_packing():
    n = N_CIRCLES
    constraints = build_constraints(n)
    bounds = build_bounds(n)
    
    best_sum = 0.0
    best_centers = np.zeros((n, 2))
    best_radii = np.ones(n) * 0.01
    
    # Multiple trials with different initial configurations
    for trial in range(15):
        np.random.seed(trial * 42)
        
        spacing = 0.10 + trial * 0.008
        centers_init = init_hexagonal(n, spacing)
        
        # Add small random perturbations
        centers_perturbed = centers_init + np.random.randn(n, 2) * 0.01
        centers_perturbed = np.clip(centers_perturbed, 0.05, 0.95)
        
        radii_init = np.ones(n) * (0.04 + np.random.random() * 0.02)
        
        x0 = np.concatenate([centers_perturbed.flatten(), radii_init])
        
        result = optimize_from_start(x0, constraints, bounds)
        
        current_sum = -result.fun
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = result.x[:2 * n].reshape(n, 2)
            best_radii = result.x[2 * n:]
    
    # Refinement: run optimization from the best solution with perturbation
    for trial in range(10):
        np.random.seed(trial * 17 + 100)
        
        centers_perturbed = best_centers + np.random.randn(n, 2) * 0.005
        centers_perturbed = np.clip(centers_perturbed, 0.05, 0.95)
        
        radii_perturbed = best_radii + np.random.randn(n) * 0.005
        radii_perturbed = np.clip(radii_perturbed, 0.001, 0.5)
        
        x0 = np.concatenate([centers_perturbed.flatten(), radii_perturbed])
        
        result = optimize_from_start(x0, constraints, bounds)
        
        current_sum = -result.fun
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = result.x[:2 * n].reshape(n, 2)
            best_radii = result.x[2 * n:]
    
    # Ensure all radii are positive and valid
    best_radii = np.maximum(best_radii, 0.0001)
    
    # Verify centers are within bounds
    best_centers = np.clip(best_centers, best_radii.reshape(-1, 1), 
                           1.0 - best_radii.reshape(-1, 1))
    
    return best_centers, best_radii, best_sum
