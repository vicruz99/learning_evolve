import numpy as np
from scipy.optimize import minimize

def penalty_objective(vars, N, lam):
    """
    Objective function for circle packing optimization.
    Minimizes negative sum of radii plus penalty for constraint violations.
    """
    centers = vars[:2*N].reshape(N, 2)
    radii = vars[2*N:]
    
    obj = -np.sum(radii)
    penalty = 0.0
    
    # Boundary penalties
    for i in range(N):
        x, y = centers[i]
        r = radii[i]
        penalty += max(0.0, r - x)**2 + max(0.0, x + r - 1.0)**2
        penalty += max(0.0, r - y)**2 + max(0.0, y + r - 1.0)**2
        
    # Overlap penalties
    for i in range(N):
        for j in range(i+1, N):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx*dx + dy*dy)
            overlap = radii[i] + radii[j] - dist
            if overlap > 0.0:
                penalty += overlap**2
                
    return obj + lam * penalty

def run_packing():
    N = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Configuration for restarts
    num_restarts = 12
    
    for seed in range(num_restarts):
        rng = np.random.default_rng(seed)
        
        # Initialize on a staggered hexagonal grid
        r_init = 0.082
        pts = []
        y = r_init + 0.035
        row = 0
        while len(pts) < N and y < 1.0 - r_init - 0.035:
            x_start = r_init + 0.035 + (0.5 * (r_init + 0.035) if row % 2 == 1 else 0)
            x = x_start
            while x < 1.0 - r_init - 0.035 and len(pts) < N:
                pts.append([x, y])
                x += np.sqrt(3) * (r_init + 0.015)
            y += 1.5 * (r_init + 0.015)
            row += 1
            
        # Fill remaining slots randomly if grid doesn't produce exactly N
        while len(pts) < N:
            pts.append([rng.uniform(0.15, 0.85), rng.uniform(0.15, 0.85)])
            
        centers_init = np.array(pts[:N])
        # Add controlled random perturbation to escape symmetry
        centers_init += rng.normal(0, 0.008, centers_init.shape)
        centers_init = np.clip(centers_init, 0.05, 0.95)
        
        radii_init = np.full(N, r_init)
        x0 = np.concatenate([centers_init.flatten(), radii_init])
        
        bounds = [(0.0, 1.0) for _ in range(2*N)] + [(1e-6, 0.5) for _ in range(N)]
        
        # Stage 1: Find feasible region
        res1 = minimize(penalty_objective, x0, args=(N, 30.0), method='L-BFGS-B', 
                        bounds=bounds, options={'maxiter': 800, 'ftol': 1e-11})
        
        # Stage 2: Tighten constraints and maximize radii
        res2 = minimize(penalty_objective, res1.x, args=(N, 150.0), method='L-BFGS-B', 
                        bounds=bounds, options={'maxiter': 800, 'ftol': 1e-12})
                        
        centers_cand = res2.x[:2*N].reshape(N, 2)
        radii_cand = res2.x[2*N:]
        
        # Validation
        valid = True
        for i in range(N):
            if radii_cand[i] < 0.0:
                valid = False; break
            x, y = centers_cand[i]
            r = radii_cand[i]
            if x - r < -1e-9 or x + r > 1.0 + 1e-9 or y - r < -1e-9 or y + r > 1.0 + 1e-9:
                valid = False; break
        if valid:
            for i in range(N):
                for j in range(i+1, N):
                    d = np.linalg.norm(centers_cand[i] - centers_cand[j])
                    if d < radii_cand[i] + radii_cand[j] - 1e-9:
                        valid = False; break
                if not valid: break
                
        if valid:
            s = np.sum(radii_cand)
            if s > best_sum:
                best_sum = s
                best_centers = centers_cand.copy()
                best_radii = radii_cand.copy()
                
    # Fallback initialization if optimization fails (extremely unlikely)
    if best_centers is None:
        best_centers = np.tile([0.5, 0.5], (N, 1))
        best_radii = np.full(N, 0.01)
        best_sum = np.sum(best_radii)
        
    return best_centers, best_radii, best_sum
