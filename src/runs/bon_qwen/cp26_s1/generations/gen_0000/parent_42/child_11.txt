import numpy as np
from scipy.optimize import minimize


def objective_func(x):
    """Objective: maximize sum of radii (minimize negative sum)."""
    radii = x[2::3]
    return -np.sum(radii)


def compute_all_constraints(x, n):
    """Compute all constraint values (should be >= 0 for feasibility)."""
    vals = []
    for i in range(n):
        xi, yi, ri = x[3 * i], x[3 * i + 1], x[3 * i + 2]
        vals.append(ri)
        vals.append(xi - ri)
        vals.append(1.0 - xi - ri)
        vals.append(yi - ri)
        vals.append(1.0 - yi - ri)

    for i in range(n):
        for j in range(i + 1, n):
            xi, yi, ri = x[3 * i], x[3 * i + 1], x[3 * i + 2]
            xj, yj, rj = x[3 * j], x[3 * j + 1], x[3 * j + 2]
            dx = xi - xj
            dy = yi - yj
            dist = np.sqrt(dx * dx + dy * dy)
            vals.append(dist - ri - rj)

    return np.array(vals)


def init_hexagonal(n, r_base=0.07):
    """Initialize circles in a hexagonal pattern."""
    x0 = np.zeros(3 * n)
    r = r_base

    # Pattern: 6, 5, 6, 5, 4 = 26
    pattern = [6, 5, 6, 5, 4]
    idx = 0
    row = 0
    for count in pattern:
        y = r + row * np.sqrt(3) * r
        for col in range(count):
            x_off = (row % 2) * r
            x = r + col * 2 * r + x_off
            x0[3 * idx] = x
            x0[3 * idx + 1] = y
            x0[3 * idx + 2] = r
            idx += 1
        row += 1

    return x0


def init_grid(n, r_base=0.06):
    """Initialize circles in a regular grid pattern."""
    x0 = np.zeros(3 * n)
    r = r_base

    # Approximate sqrt for grid
    cols = int(np.ceil(np.sqrt(n * 1.5)))
    rows = int(np.ceil(n / cols))

    # Adjust to fit
    x_step = 1.0 / (cols + 1)
    y_step = 1.0 / (rows + 1)

    idx = 0
    for i in range(rows):
        for j in range(cols):
            if idx >= n:
                break
            x0[3 * idx] = (j + 1) * x_step
            x0[3 * idx + 1] = (i + 1) * y_step
            x0[3 * idx + 2] = r
            idx += 1
        if idx >= n:
            break

    return x0


def init_random(n, r_base=0.05):
    """Initialize circles at random positions."""
    x0 = np.zeros(3 * n)
    for i in range(n):
        x0[3 * i] = np.random.uniform(2 * r_base, 1 - 2 * r_base)
        x0[3 * i + 1] = np.random.uniform(2 * r_base, 1 - 2 * r_base)
        x0[3 * i + 2] = r_base
    return x0

def init_staggered(n, r_base=0.07):

def run_optimization(x0, n, maxiter=3000):
    """Run SLSQP optimization from a given starting point."""
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n

    cons = [{'type': 'ineq', 'fun': lambda x: compute_all_constraints(x, n)}]

    try:
        result = minimize(
            objective_func,
            x0,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={
                'maxiter': maxiter,
                'ftol': 1e-15,
                'disp': False
            }
        )
        return result
    except Exception:
        return None


def extract_solution(result, n):
    """Extract centers, radii, and sum from optimization result."""
    centers = result.x[:2 * n].reshape((n, 2))
    radii = result.x[2 * n:]
    return centers, radii, np.sum(radii)


def fix_and_reoptimize(centers, radii, n):
    """Fix any constraint violations and reoptimize."""
    x = np.concatenate([centers.ravel(), radii])

    # Clamp values
    for i in range(n):
        ri = x[3 * i + 2]
        xi = x[3 * i]
        yi = x[3 * i + 1]

        ri = max(0, ri)
        xi = max(ri, min(1 - ri, xi))
        yi = max(ri, min(1 - ri, yi))

        x[3 * i] = xi
        x[3 * i + 1] = yi
        x[3 * i + 2] = ri

    return run_optimization(x, n, maxiter=2000)
