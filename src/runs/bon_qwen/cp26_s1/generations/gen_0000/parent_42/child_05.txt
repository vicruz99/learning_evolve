import numpy as np

def calculate_overlap_forces(centers, radii):
    """Calculates repulsive forces between overlapping circles."""
    n = centers.shape[0]
    forces = np.zeros_like(centers)
    
    # Vectorized distance calculation
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist_sq = np.sum(diff**2, axis=2)
    dist_sq = np.maximum(dist_sq, 1e-12) # Avoid division by zero
    dist = np.sqrt(dist_sq)
    
    # Pairwise radius sums
    rad_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    
    # Overlap amount: positive if circles overlap
    overlap = rad_sum - dist
    # Only consider actual overlaps
    is_overlapping = overlap > 0
    overlap[~is_overlapping] = 0
    
    # Normalize direction vectors (avoid division by zero for identical centers)
    # diff / dist
    # We need to handle dist=0 case, but dist_sq is clamped
    # However, diff can be 0.
    safe_dist = np.where(dist < 1e-10, 1e-10, dist)
    directions = diff / safe_dist[:, :, np.newaxis]
    
    # Force magnitude: proportional to overlap
    # F = k * overlap * direction
    # k is stiffness, let's use 1.0
    force_mag = overlap
    
    # Accumulate forces
    # For each i, sum forces from all j
    # directions[i, j] points from j to i
    # Force on i from j is in direction of (i - j)
    # So directions[i, j] is correct
    
    # Sum over axis 1 (j)
    forces = np.sum(force_mag[:, :, np.newaxis] * directions, axis=1)
    
    return forces

def calculate_boundary_forces(centers, radii):
    """Calculates forces pushing circles inside the unit square."""
    n = centers.shape[0]
    forces = np.zeros_like(centers)
    
    # Left wall: x < r => force pushes right

    left_active = left_dist > 0
    forces[left_active, 0] += left_dist[left_active]
    
    # Right wall: x > 1 - r => force pushes left
    right_dist = centers[:, 0] - (1 - radii)
    right_active = right_dist > 0
    forces[right_active, 0] -= right_dist[right_active]
    
    # Bottom wall: y < r => force pushes up

    top_dist = centers[:, 1] - (1 - radii)
    top_active = top_dist > 0
    forces[top_active, 1] -= top_dist[top_active]
    
    return forces
