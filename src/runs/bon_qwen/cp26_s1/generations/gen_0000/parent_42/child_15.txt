```python
import numpy as np
from scipy.optimize import minimize
import math

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    Uses a multi-start strategy with hexagonal and grid initializations, followed by
    local optimization with SLSQP.
    """
    n = 26
    
    def create_constraints():
        """Creates a list of constraint dictionaries for the optimizer."""
        cons = []
        for i in range(n):
            # Boundary constraints: circle i must be inside [0,1]x[0,1]
            # x_i - r_i >= 0
            cons.append({'type': 'ineq', 'fun': lambda v, i=i: v[3*i] - v[3*i+2]})
            # 1 - x_i - r_i >= 0
            cons.append({'type': 'ineq', 'fun': lambda v, i=i: 1.0 - v[3*i] - v[3*i+2]})
            # y_i - r_i >= 0
            cons.append({'type': 'ineq', 'fun': lambda v, i=i: v[3*i+1] - v[3*i+2]})
            # 1 - y_i - r_i >= 0
            cons.append({'type': 'ineq', 'fun': lambda v, i=i: 1.0 - v[3*i+1] - v[3*i+2]})
            
            # Non-overlap constraints: distance between centers >= sum of radii
            for j in range(i + 1, n):
                def dist_cn(v, i=i, j=j):
                    x1, y1, r1 = v[3*i], v[3*i+1], v[3*i+2]
                    x2, y2, r2 = v[3*j], v[3*j+1], v[3*j+2]
                    # dist^2 >= (r1 + r2)^2
                    return (x1 - x2)**2 + (y1 - y2)**2 - (r1 + r2)**2
                cons.append({'type': 'ineq', 'fun': dist_cn})
        return cons

    def optimize(xs, ys, rs):
        """Runs the optimization solver from a given starting configuration."""
        # Variable vector layout: [x0, y0, r0, x1, y1, r1, ...]
        x0 = np.concatenate([xs, ys, rs])
        
        # Bounds for variables
        bounds = []
        for _ in range(n):
            bounds.append((0.0, 1.0)) # x in [0, 1]
            bounds.append((0.0, 1.0)) # y in [0, 1]
            bounds.append((0.0, 0.5)) # r >= 0 (max radius 0.5)
        
        cons = create_constraints()
        
        def objective(v):
            # Maximize sum of radii => Minimize negative sum
            return -np.sum(v[2::3])
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 2000, 'ftol': 1e-10})
            if res.success:
                return res.x, -res.fun
        except Exception:
            pass
        return None, -1.0

    def hex_init():
        """Generates a hexagonal lattice initialization."""
        r = 0.1
        pts = []
        y = r
        row = 0
        while len(pts) < n:
            # Shift odd rows by r to create hexagonal pattern
            x_start = r if row % 2 == 0 else 2 * r
            x = x_start
            while x + r <= 1.0 and len(pts) < n:
                pts.append((x, y))
                x += 2 * r
            y += r * math.sqrt(3)
            row += 1
        
        # Fill remaining slots if any (unlikely with proper spacing)
        while len(pts) < n:
            pts.append((np.random.rand(), np.random.rand()))
        
        pts = np.array(pts[:n])
        return pts[:, 0], pts[:, 1], np.full(n, r)

    best_sum = -1.0
    best_c = None
    best_r = None

    # Strategy 1: Hexagonal Packing Initialization
    xs, ys, rs = hex_init()
    vec, s = optimize(xs, ys, rs)
    if vec is not None:
        best_c = np.column_stack((vec[::3], vec[1::3]))
        best_r = vec[2::3]
        best_sum = s

    # Strategy 2: Grid Initialization (5x5 + 1)
    # A 5x5 grid fits 25 circles of radius 0.1. We add one more.
    grid_pts = []
    for x in np.linspace(0.1, 0.9, 5):
        for y in np.linspace(0.1, 0.9, 5):
            grid_pts.append((x, y))
    grid_pts.append((0.5, 0.5)) # 26th circle in center
    
    pts = np.array(grid_pts)
    # Initial radii: smaller for safety, larger for grid ones
    rs = np.full(n, 0.05)
    rs[:25] = 0.09
    
    vec, s = optimize(pts[:, 0], pts[:, 1], rs)
    if vec is not None and s > best_sum:
        best_c = np.column_stack((vec[::3], vec[1::3]))
        best_r = vec[2::3]
        best_sum = s

    # Strategy 3: Random Restarts to escape local minima
    for _ in range(5):
        xs = np.random.rand(n)
        ys = np.random.rand(n)
        rs = np.full(n, 0.05)
        vec, s = optimize(xs, ys, rs)
        if vec is not None and s > best_sum:
            best_c = np.column_stack((vec[::3], vec[1::3]))
            best_r = vec[2::3]
            best_sum = s

    # Fallback if optimization failed completely
    if best_c is None:
        best_c = np.random.rand(n, 2)
        best_r = np.full(n, 0.01)
        best_sum = np.sum(best_r)

    return best_c, best_r, best_sum
```
