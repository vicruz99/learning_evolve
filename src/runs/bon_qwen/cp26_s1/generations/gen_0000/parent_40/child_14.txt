import numpy as np
from scipy.optimize import minimize


def compute_violations(centers, radii):
    """Compute sum of squared constraint violations."""
    n = len(radii)
    violations = 0.0

    # Boundary violations
    for i in range(n):
        x, y = centers[i]
        r = radii[i]

        v = max(0, r - x)
        violations += v * v
        v = max(0, x + r - 1)
        violations += v * v
        v = max(0, r - y)
        violations += v * v
        v = max(0, y + r - 1)
        violations += v * v

    # Overlap violations
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist_sq = dx * dx + dy * dy
            dist = np.sqrt(dist_sq) if dist_sq > 0 else 0
            v = radii[i] + radii[j] - dist
            if v > 0:
                violations += v * v

    return violations


def objective_with_penalty(x, penalty_weight):
    """Objective: minimize negative sum of radii plus penalty for violations."""
    n = 26
    centers = x[:2 * n].reshape(n, 2)
    radii = x[2 * n:]
    violations = compute_violations(centers, radii)
    return -np.sum(radii) + penalty_weight * violations


def make_objective(weight):
    """Create objective function with fixed penalty weight."""
    def obj(x):
        return objective_with_penalty(x, weight)
    return obj


def init_hexagonal_pattern():
    """Initialize circles in a hexagonal lattice pattern: 6,4,6,4,6 = 26."""
    centers = []
    radii = []
    rows = [6, 4, 6, 4, 6]
    r_init = 0.075
    y = r_init + 0.04

    for i, count in enumerate(rows):
        offset = (i % 2) * 0.06
        for j in range(count):
            if count > 1:
                x = offset + (1 - 2 * offset) * j / (count - 1)
            else:
                x = 0.5
            centers.append([x, y])
            radii.append(r_init)
        y += 0.16

    return np.array(centers), np.array(radii)


def init_grid_pattern():
    """Initialize circles in a 5x5 grid plus 1 extra."""
    centers = []
    radii = []

    for i in range(5):
        for j in range(5):
            x = 0.1 + j * 0.2
            y = 0.1 + i * 0.2
            centers.append([x, y])
            radii.append(0.08)

    # Add extra circle
    centers.append([0.5, 0.9])
    radii.append(0.08)

    return np.array(centers), np.array(radii)


def init_compact_hexagonal():

    centers = []
    radii = []
    rows = [5, 5, 5, 5, 5, 1]

    y = r_init + 0.02

    return np.array(centers), np.array(radii)


def init_random_pattern(seed):

    rng = np.random.RandomState(seed)
    centers = []
    radii = []

    # Place circles in roughly 5 rows
    for i in range(26):
        row = i // 6

        base_x = 0.1 + col * 0.16

        y = base_y + rng.uniform(-0.03, 0.03)
        centers.append([np.clip(x, 0.05, 0.95), np.clip(y, 0.05, 0.95)])
        radii.append(0.06)

    return np.array(centers), np.array(radii)


def optimize_from_init(centers, radii):
    """Run optimization from given initial configuration."""
    n = 26
    x0 = np.concatenate([centers.flatten(), radii])

    lower = [0.0001] * (2 * n) + [0.0001] * n
    upper = [0.9999] * (2 * n) + [0.5] * n
    bounds = list(zip(lower, upper))

    # Progressive penalty optimization
    current_x = x0.copy()
    weights = [500, 2000, 10000, 50000, 200000, 500000]

    for w in weights:
        obj_func = make_objective(w)
        result = minimize(obj_func, current_x, method='SLSQP', bounds=bounds,
                         options={'maxiter': 3000, 'ftol': 1e-14})
        current_x = result.x

    # Refine with L-BFGS-B
    obj_final = make_objective(1000000)
    result = minimize(obj_final, current_x, method='L-BFGS-B', bounds=bounds,
                     options={'maxiter': 5000, 'ftol': 1e-15})

    radii_opt = result.x[2 * n:]

    radii_opt = np.maximum(radii_opt, 1e-6)

    # Project centers to valid range given radii
    centers_opt = np.clip(centers_opt, radii_opt[:, np.newaxis], 1 - radii_opt[:, np.newaxis])

    return centers_opt, radii_opt


def alternating_optimize(centers, radii, n_iter=20):
    """Alternating optimization: positions then radii."""
    n = 26

    for iteration in range(n_iter):
        # Optimize positions with fixed radii
        def pos_obj(x):
            c = x.reshape(n, 2)
            return compute_violations(c, radii)

        x0 = centers.flatten()
        bounds_pos = [(0, 1)] * (2 * n)
        result = minimize(pos_obj, x0, method='L-BFGS-B', bounds=bounds_pos,
                         options={'maxiter': 2000})
        centers = result.x.reshape(n, 2)

        # Optimize radii with fixed positions
        def rad_obj(r):
            return -np.sum(r) + 100000 * compute_violations(centers, r)

        bounds_rad = [(0.0001, 0.5)] * n
        result = minimize(rad_obj, radii, method='L-BFGS-B', bounds=bounds_rad,
                         options={'maxiter': 2000})
        radii = result.x
        radii = np.maximum(radii, 1e-6)

        # Project centers
        centers = np.clip(centers, radii[:, np.newaxis], 1 - radii[:, np.newaxis])

    return centers, radii
