def compute_objective_and_constraints(vars, n):
    """
    Computes the negative sum of radii (to minimize) and the constraint values.
    vars layout: [x0, y0, r0, x1, y1, r1, ...]
    """
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    for i in range(n):
        centers[i, 0] = vars[3*i]
        centers[i, 1] = vars[3*i + 1]
        radii[i] = vars[3*i + 2]
        
    # Objective: maximize sum(radii) => minimize -sum(radii)
    obj = -np.sum(radii)
    
    # Constraints
    # 1. Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    # Equivalent to: x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
    # And r >= 0
    
    constraints_list = []
    
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        
        # r >= 0
        # We can add this as a bound or constraint. SLSQP supports bounds.
        
        # Boundary constraints
        constraints_list.append(x - r)       # x >= r
        constraints_list.append(1.0 - x - r) # x <= 1 - r => 1 - x - r >= 0
        constraints_list.append(y - r)       # y >= r
        constraints_list.append(1.0 - y - r) # y <= 1 - r
        
    # 2. Overlap constraints: dist >= r_i + r_j
    # dist^2 >= (r_i + r_j)^2
    # dist^2 - (r_i + r_j)^2 >= 0
    
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist_sq = dx*dx + dy*dy
            sum_r = radii[i] + radii[j]
            
            constraints_list.append(dist_sq - sum_r**2)
            
    return obj, constraints_list

def run_optimization(n, initial_centers, initial_radii, seed=None):
    if seed is not None:
        np.random.seed(seed)
        
    # Initial variables
    x0 = []
    for i in range(n):
        x0.append(initial_centers[i, 0])
        x0.append(initial_centers[i, 1])
        x0.append(initial_radii[i])
        
    x0 = np.array(x0)
    
    # Bounds: 
    # x in [0, 1], y in [0, 1], r in [0, 0.5] (r cannot be > 0.5)

    bounds = []
    for _ in range(n):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 0.5)) # r
        
    # Callback to track progress (optional, kept simple for performance)
    
    def fun(vars):
        obj, _ = compute_objective_and_constraints(vars, n)
        return obj
        
    def cons(vars):
        _, constraints = compute_objective_and_constraints(vars, n)
        return constraints
        
    # Define constraints for scipy
    # type 'ineq' means func(x) >= 0
    constraint_def = {'type': 'ineq', 'fun': cons}
    
    try:
        res = minimize(fun, x0, method='SLSQP', bounds=bounds, constraints=constraint_def, 
                       options={'maxiter': 1000, 'ftol': 1e-9, 'disp': False})
        
        if res.success or (not np.isnan(res.fun)):
            # Extract result
