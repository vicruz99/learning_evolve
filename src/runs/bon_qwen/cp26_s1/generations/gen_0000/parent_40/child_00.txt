import numpy as np
from scipy.optimize import minimize

def compute_cost_and_grad(params, n, mu):
    """
    Computes the cost function and its analytical gradient.
    Cost = -sum(radii) + mu * (boundary_violations^2 + overlap_violations^2)
    """
    centers = params[:2*n].reshape(n, 2)
    radii = params[2*n:]
    
    cost = -np.sum(radii)
    grad = np.zeros_like(params)
    grad[2*n:] = -1.0  # Gradient of -sum(radii) w.r.t radii
    
    # Boundary penalties and gradients
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        
        # Left (x=0) and Right (x=1) walls
        for wall_x, sign_x in [(0, -1), (1, 1)]:
            if r > abs(x - wall_x):
                val = r - abs(x - wall_x)
                cost += mu * val**2
                grad[3*i + 2] += 2 * mu * val
                grad[3*i] += sign_x * 2 * mu * val
                
        # Bottom (y=0) and Top (y=1) walls
        for wall_y, sign_y in [(0, -1), (1, 1)]:
            if r > abs(y - wall_y):
                val = r - abs(y - wall_y)
                cost += mu * val**2
                grad[3*i + 2] += 2 * mu * val
                grad[3*i + 1] += sign_y * 2 * mu * val
                
    # Overlap penalties and gradients
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            d = np.sqrt(dx*dx + dy*dy)
            
            if d < 1e-12: 
                d = 1e-12
                dx, dy = 0.0, 0.0
            
            r_sum = radii[i] + radii[j]
            if r_sum > d:
                val = r_sum - d
                cost += mu * val**2
                
                # Gradient w.r.t radii
                grad[3*i + 2] += 2 * mu * val
                grad[3*j + 2] += 2 * mu * val
                
                # Gradient w.r.t centers (repulsion force)
                fx = 2 * mu * val * (dx / d)
                grad[3*i] += fx
                grad[3*j] -= fx
                
                fy = 2 * mu * val * (dy / d)
                grad[3*i + 1] += fy
                grad[3*j + 1] -= fy
                
    return cost, grad

def run_packing():
    n = 26
    best_cost = -np.inf
    best_params = None
    rng = np.random.default_rng(2024)
    
    obj_func = compute_cost_and_grad
    
    # Multiple random restarts to find a robust local/global optimum
    for _ in range(50):
        # Initialize centers randomly inside the square and small radii
        cx = rng.uniform(0.15, 0.85, n)
        cy = rng.uniform(0.15, 0.85, n)
        cr = np.ones(n) * 0.01
        params = np.concatenate([cx, cy, cr])
        
        cur_params = params.copy()
        
        # Annealing penalty weight to ensure feasibility while maximizing radii
        for mu in [10, 100, 1000, 1e4, 1e5]:
            bounds = [(0, 1)] * (2 * n) + [(0, 0.5)] * n
            res = minimize(obj_func, cur_params, args=(n, mu), method='L-BFGS-B',
                           jac=True, bounds=bounds,
                           options={'maxiter': 2000, 'ftol': 1e-15, 'gtol': 1e-15})
            cur_params = res.x
            
        cur_radii = cur_params[2*n:]
        cur_sum = np.sum(cur_radii)
        
        if cur_sum > best_cost:
            best_cost = cur_sum
            best_params = cur_params.copy()
            
    centers = best_params[:2*n].reshape(n, 2)
    radii = best_params[2*n:]
    
    # Final feasibility adjustment to guarantee validator passes
    max_viol = 1e-12
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        v = max(0, r - x, r - (1-x), r - y, r - (1-y))
        if v > max_viol: max_viol = v
        
    for i in range(n):
        for j in range(i+1, n):
            d = np.sqrt(np.sum((centers[i] - centers[j])**2))
            v = radii[i] + radii[j] - d
            if v > max_viol: max_viol = v
            
    if max_viol > 1e-12:
        factor = max(0.0, 1.0 - max_viol - 1e-10)
        radii *= factor
        
    return centers, radii, np.sum(radii)
