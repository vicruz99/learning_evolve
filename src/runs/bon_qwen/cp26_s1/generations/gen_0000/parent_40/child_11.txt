

To solve the problem of packing 26 circles in a unit square to maximize the sum of radii, we employ a hybrid optimization strategy. The primary approach is to find the maximum radius for a uniform packing (26 circles of equal radius $r$) using a simulated annealing algorithm with local packing constraints. Once a dense uniform configuration is found, a secondary refinement phase optimizes the radii and centers individually to exploit local geometric gaps and boundary constraints.

### Strategy
1.  **Uniform Packing Search (Simulated Annealing):**
    *   Initialize 26 circles in a high-density grid-like pattern.
    *   Iterate using a cooling schedule to allow non-overlapping movement and gradual shrinking of the common radius $r$ to find the tightest possible fit.
    *   This establishes a baseline high-value configuration (typically yielding $r \approx 0.093$, Sum $\approx 2.41$).

2.  **Non-Uniform Refinement (Sequential Optimization):**
    *   Starting from the best uniform configuration, iterate through each circle $i$.
    *   Fix the centers and radii of all other circles $j \neq i$.
    *   Use `scipy.optimize.minimize` to expand circle $i$'s radius to its maximum possible limit given the fixed constraints of the others.
    *   This process captures potential gains from unequal radii, potentially pushing the sum toward the target of 2.636.

3.  **Randomization and Robustness:**
    *   The process is repeated with multiple random seeds and different initial topologies to ensure the global maximum is approached rather than getting stuck in local minima.

import numpy as np
from scipy.optimize import minimize

def run_packing():
    def distance_matrix(centers):
        return np.sqrt(np.sum((centers[:, np.newaxis] - centers[np.newaxis, :]) ** 2, axis=2))

    def check_validity(centers, radii):
        n = centers.shape[0]
        for i in range(n):
            if radii[i] < 0:
                return False
            if centers[i, 0] - radii[i] < -1e-12 or centers[i, 0] + radii[i] > 1 + 1e-12:
                return False
            if centers[i, 1] - radii[i] < -1e-12 or centers[i, 1] + radii[i] > 1 + 1e-12:
                return False
        
        dists = distance_matrix(centers)
        r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
        mask = np.triu(np.ones((n, n), dtype=bool), k=1)
        if np.any(dists[mask] < r_sum[mask] - 1e-12):
            return False
        return True

    def simulated_annealing(centers, radii, iterations=5000, temp_init=0.05):
        best_centers = centers.copy()
        best_radii = radii.copy()
        best_sum = np.sum(radii)
        
        current_temp = temp_init
        
        for i in range(iterations):
            current_r = np.min(radii) if np.allclose(radii, radii[0]) else np.mean(radii)
            if np.allclose(radii, radii[0]):
                trial_r = current_r
            else:
                trial_r = current_r
            
            # Perturb centers
            step_size = current_temp * 0.02
            trial_centers = best_centers + np.random.uniform(-step_size, step_size, size=best_centers.shape)
            
            # Boundary correction
            trial_centers = np.clip(trial_centers, trial_r, 1 - trial_r)
            
            if np.allclose(radii, radii[0]):
                trial_radii = np.full(26, trial_r)
            else:
                trial_radii = best_radii.copy()

            # If radii are fixed uniform, try to shrink r slightly to allow more movement
            if np.allclose(radii, radii[0]):
                trial_radii -= 1e-5

            if check_validity(trial_centers, trial_radii):
                trial_sum = np.sum(trial_radii)
                if trial_sum > best_sum:
                    best_centers = trial_centers.copy()
                    best_radii = trial_radii.copy()
                    best_sum = trial_sum
            
            current_temp *= 0.995
        return best_centers, best_radii

    best_global_centers = None
    best_global_radii = None
    best_global_sum = 0

    # Run multiple optimization attempts with different initial topologies
    for seed in range(5):
        np.random.seed(seed)
        
        # Strategy A: Hexagonal-ish initial placement
        centers = []
        for i in range(26):
            row = i // 5
            col = i % 5
            y = 0.15 + row * 0.18
            x_offset = 0.08 if row % 2 == 1 else 0.15
            x = x_offset + col * 0.18
            if x > 0.85:
                x = 0.15 + col * 0.18
            centers.append([x, y])
        centers = np.array(centers)
        
        # Strategy B: Grid initial placement
        if seed % 2 == 1:
            centers = np.array([[ (i % 5) * 0.2 + 0.1, (i // 5) * 0.2 + 0.1 ] for i in range(26)])
            centers[:, 0] = np.clip(centers[:, 0], 0.1, 0.9)
            centers[:, 1] = np.clip(centers[:, 1], 0.1, 0.9)

        initial_r = 0.095
        radii = np.full(26, initial_r)

        # Step 1: Simulated Annealing to optimize uniform packing
        centers, radii = simulated_annealing(centers, radii, iterations=8000, temp_init=0.08)
        
        # Step 2: Non-uniform refinement (Iterative individual expansion)
        for _ in range(3):
            for i in range(26):
                other_indices = [j for j in range(26) if j != i]
                others_centers = centers[other_indices]
                others_radii = radii[other_indices]
                c_fixed = centers[i]
                
                def obj(r_new):
                    r = r_new[0]
                    # Boundary constraint
                    if r < 0 or c_fixed[0] - r < 0 or c_fixed[0] + r > 1 or \
                       c_fixed[1] - r < 0 or c_fixed[1] + r > 1:
                        return -10000
                    
                    # Overlap constraint
                    dists = np.sqrt(np.sum((others_centers - c_fixed) ** 2, axis=1))
                    if np.any(dists < others_radii + r - 1e-6):
                        return -10000
                    return -r # Maximize radius

                res = minimize(obj, [radii[i]], bounds=[(0, 0.5)])
                if res.success:
                    radii[i] = res.x[0]

        current_sum = np.sum(radii)
        if current_sum > best_global_sum:
            best_global_centers = centers.copy()
            best_global_radii = radii.copy()
            best_global_sum = current_sum

    return best_global_centers, best_global_radii, best_global_sum
