from scipy.optimize import minimize

def compute_penalty(vars, n):

    radii = vars[2*n:]

    pen += np.sum(np.maximum(0, radii - centers[:, 1])**2)
    pen += np.sum(np.maximum(0, centers[:, 1] + radii - 1)**2)

    pen += np.sum((rad_sums[i_j, j_i] - dists[i_j, j_i])**2)
        
    return pen

def objective(vars, n, w):

    return -np.sum(radii) + w * compute_penalty(vars, n)
