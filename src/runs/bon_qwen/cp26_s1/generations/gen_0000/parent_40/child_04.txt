import numpy as np

def get_overlap_penalty(centers, radii):
    """Calculate penalty for overlapping circles."""
    n = len(radii)
    penalty = 0.0
    # Vectorized distance calculation for all pairs
    # Centers shape: (n, 2)
    # diffs shape: (n, n, 2)
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    
    # Sum of radii matrix (n, n)
    radii_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    
    # Overlap amount: positive if overlapping
    overlap = radii_sum - dists
    
    # We only care about pairs i < j to avoid double counting and self
    # Mask for upper triangle (i < j)
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    overlaps = overlap[mask]
    
    # Penalize positive overlaps (where dist < r1 + r2)
    # Using a smooth approximation or direct max. 
    # Direct max(0, overlap)^2 is non-smooth but usually works or we can use softplus.
    # For L-BFGS-B, smoothness helps, but max(0,x) is okay. 
    # Let's use max(0, overlap) ** 2.
    
    pos_overlap = np.maximum(0, overlaps)
    penalty = np.sum(pos_overlap**2)
    return penalty

def get_boundary_penalty(centers, radii):
    """Calculate penalty for circles outside the unit square."""
    penalty = 0.0
    n = len(radii)
    
    # Constraints: x - r >= 0 => r - x <= 0
    # x + r <= 1 => x + r - 1 <= 0
    # y - r >= 0 => r - y <= 0
    # y + r <= 1 => y + r - 1 <= 0
    
    # Calculate violations
    # Violations are positive if constraint is violated
    v1 = np.maximum(0, radii - centers[:, 0])      # Left wall
    v2 = np.maximum(0, centers[:, 0] + radii - 1)  # Right wall
    v3 = np.maximum(0, radii - centers[:, 1])      # Bottom wall
    v4 = np.maximum(0, centers[:, 1] + radii - 1)  # Top wall
    
    penalty = np.sum(v1**2) + np.sum(v2**2) + np.sum(v3**2) + np.sum(v4**2)
    return penalty
