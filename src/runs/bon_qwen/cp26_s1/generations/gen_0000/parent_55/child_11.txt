import numpy as np
from scipy.optimize import minimize
import math

N_CIRCLES = 26

def constraints_func(v):
    """
    Compute all geometric constraints as a single array.
    Returns g(v) >= 0 for valid configurations.
    """
    n = N_CIRCLES
    C = v[:2*n].reshape(n, 2)
    R = v[2*n:]
    cons = []
    
    # Pairwise non-overlap: dist(i,j) >= r_i + r_j
    for i in range(n):
        diff = C[i] - C[i+1:]
        dist = np.sqrt(np.sum(diff**2, axis=1))
        r_sum = R[i] + R[i+1:]
        cons.extend(dist - r_sum)
        
    # Boundary constraints: r <= x <= 1-r and r <= y <= 1-r
    cons.extend(C[:,0] - R)
    cons.extend(1.0 - C[:,0] - R)
    cons.extend(C[:,1] - R)
    cons.extend(1.0 - C[:,1] - R)
    
    return np.array(cons)

def objective_func(v):
    """
    Objective: minimize negative sum of radii (equivalent to maximizing sum of radii).
    """
    return -np.sum(v[2::3])

def run_packing():
    n = N_CIRCLES
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Multiple trials to escape local minima
    for trial in range(5):
        # 1. Initialization: Perturbed hexagonal grid
        centers = np.zeros((n, 2))
        idx = 0
        for row in range(6):
            y = 0.15 + row * 0.14 + trial * 0.01
            for col in range(5):
                if idx >= n: break
                x = 0.15 + col * 0.15 + (0.075 if row % 2 == 1 else 0) + trial * 0.02
                centers[idx] = [x, y]
                idx += 1
        # Fill any remaining slots (though 6x5=30 > 26)
        for i in range(idx, n):
            centers[i] = [0.5 + 0.05*(2*(i%2)-1), 0.5 + 0.05*(2*(i//2%2)-1)]
        centers = np.clip(centers, 0.05, 0.95)
        
        radii = np.full(n, 0.06)
        
        # 2. Simulation: Repulsion + gradual expansion
        for step in range(4000):
            forces = np.zeros_like(centers)
            
            # Pairwise repulsion
            for i in range(n):
                for j in range(i+1, n):
                    dx = centers[i,0] - centers[j,0]
                    dy = centers[i,1] - centers[j,1]
                    dist = math.sqrt(dx*dx + dy*dy)
                    r_sum = radii[i] + radii[j]
                    if dist < r_sum and dist > 1e-9:
                        overlap = r_sum - dist
                        f = overlap * overlap / dist
                        forces[i,0] += f * dx
                        forces[i,1] += f * dy
                        forces[j,0] -= f * dx
                        forces[j,1] -= f * dy
                        
            # Boundary repulsion
            for i in range(n):
                x, y = centers[i]
                r = radii[i]
                if x < r:
                    forces[i,0] += (r - x)**2
                elif x > 1 - r:
                    forces[i,0] -= (x - (1-r))**2
                if y < r:
                    forces[i,1] += (r - y)**2
                elif y > 1 - r:
                    forces[i,1] -= (y - (1-r))**2
                    
            centers += 1e-4 * forces
            centers = np.clip(centers, 1e-6, 1-1e-6)
            radii *= 1.0004
            
        # 3. Refinement: SLSQP optimization
        x0 = np.concatenate([centers.ravel(), radii])
        bounds = [(0.0, 1.0) for _ in range(2*n)] + [(0.0, 0.5) for _ in range(n)]
        cons = {'type': 'ineq', 'fun': constraints_func}
        
        try:
            res = minimize(objective_func, x0, method='SLSQP', bounds=bounds, constraints=cons, 
                           options={'maxiter': 1500, 'ftol': 1e-10, 'disp': False})
            cur_sum = -res.fun
            if cur_sum > best_sum:
                best_sum = cur_sum
                best_centers = res.x[:2*n].reshape(n, 2)
                best_radii = res.x[2*n:]
        except Exception:
            pass
            
    # Fallback if optimization never succeeded
    if best_centers is None:
        best_centers, best_radii = centers, radii
        best_sum = np.sum(radii)
        
    # Final validation adjustments
    best_radii = np.maximum(best_radii, 1e-9)
    best_sum = np.sum(best_radii)
    
    return best_centers, best_radii, float(best_sum)
