import numpy as np
import scipy.optimize

def obj_unequal(vars_flat):
    # Maximize sum of radii -> Minimize negative sum
    radii = vars_flat[2::3]
    return -np.sum(radii)

def constr_boundary_unequal(vars_flat):
    # Variables: [x1, y1, r1, x2, y2, r2, ...]
    n = len(vars_flat) // 3
    x = vars_flat[0::3]
    y = vars_flat[1::3]
    r = vars_flat[2::3]
    
    cons = []
    # x - r >= 0
    cons.extend((x - r).tolist())
    # 1 - x - r >= 0  => x + r <= 1
    cons.extend((1.0 - x - r).tolist())
    # y - r >= 0
    cons.extend((y - r).tolist())
    # 1 - y - r >= 0 => y + r <= 1
    cons.extend((1.0 - y - r).tolist())
    # r >= 0 (covered by bounds, but good for constraint logic)
    # cons.extend(r.tolist()) 
    # Bounds handle r >= 0.
    
    return np.array(cons)

def constr_overlap_unequal(vars_flat):
    n = len(vars_flat) // 3
    centers = vars_flat.reshape((n, 3))[:, :2]
    radii = vars_flat.reshape((n, 3))[:, 2]
    
    cons = []
    for i in range(n):
        for j in range(i + 1, n):
            # dist^2 >= (ri + rj)^2
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist_sq = dx*dx + dy*dy
            sum_r = radii[i] + radii[j]
            cons.append(dist_sq - sum_r * sum_r)
    return np.array(cons)

def penalty_obj(vars_flat, penalty_weight=1000.0):
    n = len(vars_flat) // 3
    x = vars_flat[0::3]
    y = vars_flat[1::3]
    r = vars_flat[2::3]
    
    obj = -np.sum(r)
    
    # Boundary penalties
    # x - r >= 0 -> penalty if r > x
    # val = r - x
    val_x_left = r - x
    mask_x_left = val_x_left > 0
    obj += penalty_weight * np.sum(val_x_left[mask_x_left]**2)
    
    # x + r <= 1 -> penalty if x + r > 1
    val_x_right = x + r - 1.0
    mask_x_right = val_x_right > 0
    obj += penalty_weight * np.sum(val_x_right[mask_x_right]**2)
    
    # y - r >= 0
    val_y_bottom = r - y
    mask_y_bottom = val_y_bottom > 0
    obj += penalty_weight * np.sum(val_y_bottom[mask_y_bottom]**2)
    
    # y + r <= 1
    val_y_top = y + r - 1.0
    mask_y_top = val_y_top > 0
    obj += penalty_weight * np.sum(val_y_top[mask_y_top]**2)
    
    # Overlap penalties
    # dist >= r_i + r_j -> penalty if r_i + r_j > dist
    # dist = sqrt(dx^2 + dy^2)
    # To avoid sqrt, maybe check (r_i + r_j)^2 > dist^2?
    # But ReLU argument is r_i + r_j - dist.
    # Computing dist is needed.
    
    # Vectorized pairwise distance
    # centers (n, 2)
    centers = vars_flat.reshape((n, 3))[:, :2]
    # diffs (n, n, 2)
    # This creates large array. n=26 -> 26x26x2. Small.
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    
    # radii sum matrix (n, n)
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    
    # Overlap amount: r_sum - dists
    overlap = r_sum - dists
    
    # Only consider upper triangle to avoid double counting and self
    # triu_indices
    triu = np.triu_indices(n, k=1)
    overlaps_upper = overlap[triu]
    
    mask_overlap = overlaps_upper > 0
    obj += penalty_weight * np.sum(overlaps_upper[mask_overlap]**2)
    
    return obj
