import scipy.optimize

# Constants
N_CIRCLES = 26
PENALTY = 5000.0

def objective(params):
    """
    Objective function to minimize.
    We minimize (Penalty * Violations - Sum of Radii).
    Minimizing this encourages valid packing (violations ~ 0) and large radii.
    """
    centers = params[:2 * N_CIRCLES].reshape(N_CIRCLES, 2)
    radii = params[2 * N_CIRCLES:]
    
    xs = centers[:, 0]
    ys = centers[:, 1]
    rs = radii
    
    # Boundary violations
    # Constraints: r <= x <= 1-r  =>  r-x <= 0, x+r-1 <= 0
    # Constraints: r <= y <= 1-y  =>  r-y <= 0, y+r-1 <= 0
    # Violation is squared max(0, constraint_violation)
    
    v_left = np.maximum(0, rs - xs)
    v_right = np.maximum(0, xs + rs - 1)
    v_bottom = np.maximum(0, rs - ys)
    v_top = np.maximum(0, ys + rs - 1)
    
    boundary_viol = np.sum(v_left**2 + v_right**2 + v_bottom**2 + v_top**2)
    
    # Overlap violations
    # Constraint: dist(i, j) >= r_i + r_j
    # Violation: max(0, r_i + r_j - dist)^2

    # diff[i, j, k] = centers[i, k] - centers[j, k]
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
