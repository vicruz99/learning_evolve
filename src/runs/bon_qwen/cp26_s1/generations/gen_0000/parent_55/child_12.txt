import numpy as np
from scipy.optimize import minimize

def compute_violations(centers, radii):
    """Compute sum of squared violations for overlaps and boundaries."""
    n = len(radii)
    v = 0.0
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        # Boundary constraints
        if x - r < 0: v += (r - x)**2
        if x + r > 1: v += (x + r - 1)**2
        if y - r < 0: v += (r - y)**2
        if y + r > 1: v += (y + r - 1)**2
        # Pairwise constraints
        for j in range(i + 1, n):
            dx = x - centers[j, 0]
            dy = y - centers[j, 1]
            dist = np.sqrt(dx*dx + dy*dy)
            r_sum = r + radii[j]
            if dist < r_sum:
                v += (r_sum - dist)**2
    return v

def objective(vars_, n_circles):
    """Objective function: maximize sum of radii with penalty for violations."""
    c = vars_[:2*n_circles].reshape(n_circles, 2)
    r = vars_[2*n_circles:]
    return -np.sum(r) + 5000.0 * compute_violations(c, r)

def run_packing():
    np.random.seed(42)
    N = 26
    
    # 1. Initialization: Staggered grid layout
    centers = np.zeros((N, 2))
    idx = 0
    rows = [6, 5, 6, 5, 4]  # Sums to 26
    for r_idx, cnt in enumerate(rows):
        y = 0.10 + r_idx * 0.20
        x_start = 0.10 if r_idx % 2 == 0 else 0.18
        for _ in range(cnt):
            centers[idx, 0] = x_start
            centers[idx, 1] = y
            x_start += 0.17
            idx += 1
            
    radii = np.full(N, 0.05)
    
    # 2. Iterative Relaxation Phase
    lr = 0.015
    dr = 0.00006
    for step in range(5000):
        radii += dr
        for _ in range(2):
            for i in range(N):
                # Boundary pushes
                for dim in range(2):
                    if centers[i, dim] - radii[i] < 0:
                        centers[i, dim] += lr * (radii[i] - centers[i, dim])
                    if centers[i, dim] + radii[i] > 1:
                        centers[i, dim] -= lr * (centers[i, dim] + radii[i] - 1)
                # Pairwise repulsion
                for j in range(i + 1, N):
                    dx = centers[i, 0] - centers[j, 0]
                    dy = centers[i, 1] - centers[j, 1]
                    dist = np.sqrt(dx*dx + dy*dy)
                    min_dist = radii[i] + radii[j]
                    if dist < min_dist and dist > 1e-8:
                        push = lr * (min_dist - dist)
                        ddx = dx / dist
                        ddy = dy / dist
                        centers[i, 0] += push * ddx
                        centers[i, 1] += push * ddy
                        centers[j, 0] -= push * ddx
                        centers[j, 1] -= push * ddy
        centers = np.clip(centers, 0.001, 0.999)
        
    # 3. Gradient-Based Refinement with L-BFGS-B
    x0 = np.concatenate([centers.flatten(), radii])
    bounds = [(0.001, 0.999) for _ in range(2*N)] + [(1e-4, 0.5) for _ in range(N)]
    
    best_res = None
    best_obj = np.inf
    
    for restart in range(4):
        x_init = x0.copy()
        if restart > 0:
            x_init[:2*N] += np.random.uniform(-0.02, 0.02, 2*N)
            x_init[:2*N] = np.clip(x_init[:2*N], 0.05, 0.95)
            x_init[2*N:] = np.full(N, radii[0] * 0.98)
            
        res = minimize(objective, x_init, args=(N,), method='L-BFGS-B', 
                       bounds=bounds, options={'maxiter': 4000, 'ftol': 1e-10})

            best_obj = res.fun
            best_res = res

                hi = mid
        radii_opt *= lo
        
    return centers_opt, radii_opt, np.sum(radii_opt)
