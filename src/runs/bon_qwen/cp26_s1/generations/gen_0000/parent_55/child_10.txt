import numpy as np

def compute_max_radius_for_circle(centers, radii, i):
    """Compute maximum possible radius for circle i given current positions of all circles."""
    n = len(radii)
    max_r = min(centers[i, 0], 1.0 - centers[i, 0], centers[i, 1], 1.0 - centers[i, 1])
    for j in range(n):
        if i != j:
            dist = np.sqrt((centers[i, 0] - centers[j, 0]) ** 2 + (centers[i, 1] - centers[j, 1]) ** 2)
            max_r = min(max_r, dist - radii[j])
    return max(max_r, 1e-10)


def initialize_hexagonal(n, r0):
    """Initialize circles in a hexagonal pattern."""
    centers = np.zeros((n, 2))
    radii = np.full(n, r0)
    idx = 0
    dx = 2.0 * r0
    dy = r0 * np.sqrt(3.0)
    for row in range(50):
        offset = (row % 2) * r0
        for col in range(50):
            x = r0 + col * dx + offset
            y = r0 + row * dy
            if x + r0 <= 1.0 and y + r0 <= 1.0 and idx < n:
                centers[idx] = [x, y]
                idx += 1
            if x > 1.0 - r0:
                break
        if idx >= n:
            break
    return centers, radii


def force_directed_optimize(centers, radii, n, iterations, seed=42):
    """Run force-directed optimization to expand circles."""
    np.random.seed(seed)
    velocities = np.random.randn(n, 2) * 0.0001
    
    for iteration in range(iterations):
        forces = np.zeros((n, 2))
        
        for i in range(n):
            # Wall repulsion forces
            for c in range(2):
                dist_left = centers[i, c] - radii[i]
                dist_right = 1.0 - centers[i, c] - radii[i]
                if dist_left < 0.001:
                    forces[i, c] += 200.0 * dist_left
                if dist_right < 0.001:
                    forces[i, c] -= 200.0 * dist_right
            
            # Circle-circle repulsion
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist_sq = dx * dx + dy * dy
                dist = np.sqrt(dist_sq)
                min_dist = radii[i] + radii[j]
                
                if dist < min_dist and dist > 1e-12:
                    repulsion = (min_dist - dist) * 300.0
                    fx = (dx / dist) * repulsion
                    fy = (dy / dist) * repulsion
                    forces[i, 0] += fx
                    forces[i, 1] += fy
                    forces[j, 0] -= fx
                    forces[j, 1] -= fy
        
        # Update velocities with damping
        velocities = 0.85 * velocities + forces * 0.00002
        centers += velocities
        
        # Clamp centers to valid region
        for i in range(n):
            for c in range(2):
                centers[i, c] = max(radii[i] + 1e-10, min(1.0 - radii[i] - 1e-10, centers[i, c]))
        
        # Gradually expand radii
        if iteration % 5 == 0:
            for i in range(n):
                max_r = compute_max_radius_for_circle(centers, radii, i)
                if max_r > radii[i] + 1e-12:
                    radii[i] = 0.92 * radii[i] + 0.08 * max_r
    
    return centers, radii


def refine_packing(centers, radii, n, num_iterations=5000):
    """Refine packing using alternating position and radius optimization."""
    for iteration in range(num_iterations):
        # Compute max radii
        max_radii = np.zeros(n)
        for i in range(n):
            max_radii[i] = compute_max_radius_for_circle(centers, radii, i)
        
        # Update radii toward max
        alpha = 0.15
        radii = (1.0 - alpha) * radii + alpha * max_radii
        
        # Compute forces to move circles apart
        forces = np.zeros((n, 2))
        for i in range(n):
            # Find the most constraining neighbor/wall and move away
            best_gap = 1.0
            best_grad = np.zeros(2)
            
            for j in range(n):
                if i != j:
                    dx = centers[i, 0] - centers[j, 0]
                    dy = centers[i, 1] - centers[j, 1]
                    dist = np.sqrt(dx * dx + dy * dy)
                    gap = dist - radii[i] - radii[j]
                    if gap < best_gap:
                        best_gap = gap
                        if dist > 1e-10:
                            best_grad = np.array([dx / dist, dy / dist])
            
            # Check walls
            for c in range(2):
                gap_left = centers[i, c] - radii[i]
                gap_right = 1.0 - centers[i, c] - radii[i]
                gap_wall = min(gap_left, gap_right)
                if gap_wall < best_gap:
                    best_gap = gap_wall
                    grad_c = np.zeros(2)
                    if gap_left <= gap_right:
                        grad_c[c] = 1.0
                    else:
                        grad_c[c] = -1.0
                    best_grad = grad_c
            
            # Move in direction of increasing gap
            lr = 0.005 / (1.0 + iteration * 0.001)
            forces[i] = best_grad * lr * max(0, -best_gap + 0.01)
        
        centers += forces
        for i in range(n):
            for c in range(2):
                centers[i, c] = max(radii[i] + 1e-10, min(1.0 - radii[i] - 1e-10, centers[i, c]))
    
    return centers, radii


def run_packing():
