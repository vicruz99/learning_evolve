    *   **Non-overlap**: For every pair of circles $(i, j)$, the distance between centers must be at least the sum of radii: $\sqrt{(x_i - x_j)^2 + (y_i - y_j)^2} \ge r_i + r_j$.

import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # 1. Initialization
    # Place circles in a grid pattern to start with a valid configuration.
    # 5x5 grid covers 25 spots. We'll place 26th slightly offset or in a 6x5 grid.
    # A 6x5 grid (30 spots) allows us to pick 26 well-spaced points.
    # Or simply random uniform with small radii.
    
    # Let's use a deterministic grid initialization for better convergence.
    # 6 columns, 5 rows = 30 points. We will pick 26.
    # Actually, just random uniform with small radii is robust enough for SLSQP 
    # if we provide a valid start.
    
    np.random.seed(123)
    centers = np.random.uniform(0.2, 0.8, (n, 2))
    # Add some jitter to avoid symmetry issues
    centers += np.random.uniform(-0.05, 0.05, (n, 2))
    centers = np.clip(centers, 0.1, 0.9)
    
    # Initial radii small enough to not overlap
    # With centers in [0.1, 0.9], min dist is small, but 0.05 might be safe?
    # Let's start with very small radii to ensure feasibility.
    radii = np.full(n, 0.02)
    
    # Flatten variables: [x1, y1, r1, x2, y2, r2, ...]
    x0 = np.zeros(3 * n)
    for i in range(n):
        x0[3*i] = centers[i, 0]
        x0[3*i+1] = centers[i, 1]
        x0[3*i+2] = radii[i]
    
    # 2. Objective Function
    def objective(vars_flat):
        # Sum of radii is the sum of every 3rd element starting from index 2
        radii = vars_flat[2::3]
        return -np.sum(radii)

    # 3. Constraints
    constraints = []
    
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    # Form: g(vars) >= 0
    # x_i - r_i >= 0
    # 1 - x_i - r_i >= 0
    # y_i - r_i >= 0
    # 1 - y_i - r_i >= 0
    
    for i in range(n):
        idx_x = 3*i
        idx_y = 3*i + 1
        idx_r = 3*i + 2
        
        # x >= r
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: v[idx_x] - v[idx_r],
            'jac': lambda v, i=i: np.zeros(3*n) # Jacobian optional but good for speed
        })
        # Actually, defining jacobians for 78 vars manually is tedious. 
        # SLSQP can approximate, but it's slower. 
        # Let's just define functions.
        
        # To make it faster, we can use vectorized operations if we bundle constraints,
        # but SLSQP handles list of dicts.
        # Let's stick to simple dicts.
        
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: 1.0 - v[idx_x] - v[idx_r]
        })
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: v[idx_y] - v[idx_r]
        })
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: 1.0 - v[idx_y] - v[idx_r]
        })

    # Non-overlap constraints: dist(i, j) >= r_i + r_j
    # sqrt((xi-xj)^2 + (yi-yj)^2) - ri - rj >= 0
    for i in range(n):
        for j in range(i + 1, n):
            idx_xi, idx_yi, idx_ri = 3*i, 3*i + 1, 3*i + 2
            idx_xj, idx_yj, idx_rj = 3*j, 3*j + 1, 3*j + 2
            
            # Capture i, j in closure
            def make_constraint(ii, jj):
                def fun(v):
                    xi, yi, ri = v[3*ii], v[3*ii+1], v[3*ii+2]
                    xj, yj, rj = v[3*jj], v[3*jj+1], v[3*jj+2]
                    dist = np.sqrt((xi - xj)**2 + (yi - yj)**2)
                    return dist - ri - rj
                return fun

            constraints.append({
                'type': 'ineq',
                'fun': make_constraint(i, j)
            })

    # 4. Bounds
    # x, y in [0, 1], r >= 0
    bounds = []
    for i in range(n):
        bounds.extend([
            (0.0, 1.0), # x
            (0.0, 1.0), # y
            (0.0, None) # r (no upper bound theoretically, but constrained by geometry)
        ])

    # 5. Optimization
    # SLSQP is suitable for constrained optimization
    # Max iterations might need to be high for convergence
    res = minimize(
        objective,
        x0,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 2000, 'ftol': 1e-9, 'disp': False}
    )
    
    # 6. Extract results
    if res.success or res.fun > -10: # If optimization failed but gave reasonable value
        final_vars = res.x
    else:
        # Fallback to initial if optimization completely failed (unlikely)
        final_vars = x0

    final_centers = np.zeros((n, 2))
    final_radii = np.zeros(n)
    
    for i in range(n):
        final_centers[i, 0] = final_vars[3*i]
        final_centers[i, 1] = final_vars[3*i+1]
        final_radii[i] = final_vars[3*i+2]
    
    # Ensure radii are non-negative (numerical safety)
    final_radii = np.maximum(final_radii, 0.0)
    
    # Ensure centers are within [0, 1] (numerical safety)
    final_centers = np.clip(final_centers, 0.0, 1.0)
    
    # Re-validate and fix potential tiny overlaps due to precision
    # If overlaps exist, slightly reduce radii
    for i in range(n):
        # Check walls
        margin_x = min(final_centers[i, 0], 1.0 - final_centers[i, 0])
        margin_y = min(final_centers[i, 1], 1.0 - final_centers[i, 1])
        max_r_wall = min(margin_x, margin_y)
        if final_radii[i] > max_r_wall + 1e-12:
            final_radii[i] = max(0, max_r_wall)
            
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(final_centers[i] - final_centers[j])
            sum_r = final_radii[i] + final_radii[j]
            if sum_r > dist + 1e-12:
                # Reduce radii equally to fix overlap
                # We want r_i' + r_j' = dist
                # Scale factor
                if sum_r > 0:
                    scale = dist / sum_r
                    final_radii[i] *= scale
                    final_radii[j] *= scale

    sum_radii = np.sum(final_radii)
    
    return final_centers, final_radii, sum_radii
