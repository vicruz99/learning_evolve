import numpy as np
from scipy.optimize import minimize
import math

def objective(vars_):
    """Objective function: minimize negative sum of radii."""
    n = len(vars_) // 3
    return -np.sum(vars_[2*n:])

def compute_constraints(vars_, n):
    """
    Computes inequality constraints for circle packing.
    Returns an array of values that should be >= 0.
    """
    xs = vars_[::3]
    ys = vars_[1::3]
    rs = vars_[2::3]
    
    con = []
    # Boundary constraints: r <= x <= 1-r and r <= y <= 1-r
    con.extend(xs - rs)
    con.extend(1.0 - xs - rs)
    con.extend(ys - rs)
    con.extend(1.0 - ys - rs)
    
    # Pairwise non-overlap constraints: dist(i,j) >= r_i + r_j
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dr = rs[:, None] + rs[None, :]
    
    dist_sq = dx**2 + dy**2
    dist_sq = np.maximum(dist_sq, 0.0)  # Numerical safety
    dist_mat = np.sqrt(dist_sq)
    
    # Only consider i < j to avoid duplicates and self-comparison
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    con.extend((dist_mat - dr)[mask].ravel())
    
    return np.array(con)

def run_packing():
    n = 26
    best_sum = 0.0
    best_centers = np.zeros((n, 2))
    best_radii = np.zeros(n)
    
    initial_guesses = []
    
    # 1. Grid initialization (5x5 + 1)
    grid_pts = np.linspace(0.15, 0.85, 5)
    centers = np.array([[x, y] for y in grid_pts for x in grid_pts])
    centers = np.vstack([centers, [0.5, 0.5]])
    radii = np.ones(n) * 0.04
    initial_guesses.append(np.concatenate([centers.flatten(), radii]))
    
    # 2. Hexagonal initialization
    centers_hex = []
    r_est = 0.08
    rows = [6, 5, 6, 5, 4]
    y_curr = 0.2
    for idx, cnt in enumerate(rows):
        if y_curr > 0.9: break
        x_start = (1.0 - (cnt - 1) * r_est * np.sqrt(3)) / 2.0
        if idx % 2 == 1:
            x_start += r_est * np.sqrt(3) / 2.0
        for c in range(cnt):
            centers_hex.append([x_start + c * r_est * np.sqrt(3), y_curr])
        y_curr += r_est * 1.5
        
    if len(centers_hex) == n:
        radii_hex = np.ones(n) * 0.03
        initial_guesses.append(np.concatenate([np.array(centers_hex).flatten(), radii_hex]))
        
    # 3. Random initializations
    for seed in range(10):
        rng = np.random.default_rng(seed + 100)
        c = rng.uniform(0.2, 0.8, (n, 2))
        r = np.ones(n) * 0.02 + rng.uniform(0, 0.01, n)
        initial_guesses.append(np.concatenate([c.flatten(), r]))
        
    bounds = [(0, 1)] * n + [(0, 1)] * n + [(0, 0.5)] * n
    
    for x0 in initial_guesses:
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                           args=(n,), constraints={'type': 'ineq', 'fun': compute_constraints},
                           options={'ftol': 1e-9, 'maxiter': 5000, 'disp': False})
            
            current_sum = -res.fun
            if current_sum <= best_sum:
                continue
                
            c_res = np.array([res.x[3*i:3*i+2] for i in range(n)])
            r_res = res.x[2*n:]
            
            # Strict validation to filter numerical artifacts
            valid = True
            if np.any(r_res < 1e-10) or np.any(c_res[:,0] < r_res - 1e-9) or \
               np.any(c_res[:,0] > 1 - r_res + 1e-9) or \
               np.any(c_res[:,1] < r_res - 1e-9) or \
               np.any(c_res[:,1] > 1 - r_res + 1e-9):
                valid = False
                
            if valid:
                for i in range(n):
                    for j in range(i+1, n):
                        d = math.sqrt((c_res[i,0]-c_res[j,0])**2 + (c_res[i,1]-c_res[j,1])**2)
                        if d < r_res[i] + r_res[j] - 1e-9:
                            valid = False
                            break
                    if not valid: break
                    
            if valid:
                best_sum = current_sum
                best_centers = c_res
                best_radii = r_res
                
        except Exception:
            continue
            
    if best_sum == 0:
        return initial_guesses[0][:2*n].reshape(n, 2), initial_guesses[0][2*n:], 0.0
        
    return best_centers, best_radii, best_sum
