import numpy as np
from scipy.optimize import minimize

def run_packing():
    np.random.seed(42)
    n = 26
    
    # Initialize positions using a hexagonal lattice pattern
    # This arrangement naturally approximates optimal packing density
    centers = []
    dy = 0.25 * np.sqrt(3) / 2
    for r_idx in range(7):
        dx = 0.125 if r_idx % 2 == 1 else 0.0
        for c_idx in range(6):
            x = dx + c_idx * 0.25
            y = 0.05 + r_idx * dy
            if x <= 0.95 and y <= 0.95:
                centers.append([x, y])
            if len(centers) >= n:
                break
        if len(centers) >= n:
            break
            
    centers = np.array(centers[:n])
    # Start with slightly larger radii to encourage dense packing via overlap resolution
    radii = np.full(n, 0.11)
    
    x0 = np.concatenate([centers.flatten(), radii])
    
    def objective(x):
        c = x[:52].reshape(n, 2)
        r = x[52:]
        loss = 0.0
        # Boundary penalties
        for i in range(n):
            for ax in range(2):
                val = c[i, ax]
                if val < r[i]:
                    loss += 5000 * (r[i] - val)**2
                elif val > 1 - r[i]:
                    loss += 5000 * (val - (1 - r[i]))**2
        # Overlap penalties
        for i in range(n):
            for j in range(i + 1, n):
                d = np.sqrt(np.sum((c[i] - c[j])**2))
                if d < r[i] + r[j]:
                    loss += 5000 * (r[i] + r[j] - d)**2
        return -np.sum(r) + loss

    bounds = [(0.0, 1.0)] * 52 + [(0.0, 0.5)] * 26
    res = minimize(objective, x0, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': 8000, 'ftol': 1e-12})
    
    c_opt = res.x[:52].reshape(n, 2)
    r_opt = res.x[52:]
    
    # Projection step to guarantee strict feasibility according to validator
    # Shrinks radii slightly to account for the 1e-12 tolerance margin
    for i in range(n):
        max_r = min(c_opt[i, 0], 1.0 - c_opt[i, 0], c_opt[i, 1], 1.0 - c_opt[i, 1])
        for j in range(n):
            if i != j:
                d = np.sqrt(np.sum((c_opt[i] - c_opt[j])**2))
                max_r = min(max_r, d / 2.0)
        r_opt[i] = max(0.0, max_r - 1e-9)
        
    return c_opt, r_opt, float(np.sum(r_opt))
