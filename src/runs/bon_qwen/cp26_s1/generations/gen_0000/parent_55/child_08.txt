import numpy as np
from scipy.optimize import minimize

# Global constants
N_CIRCLES = 26

def obj_func(vars):
    """Objective: minimize -r"""
    return -vars[-1]

def jac_func(vars):
    """Gradient of objective"""
    grad = np.zeros(len(vars))
    grad[-1] = -1.0
    return grad

def constraint_boundary(vars):
    """Boundary constraints: r <= x <= 1-r, r <= y <= 1-r"""
    r = vars[-1]
    c = vars[:N_CIRCLES*2].reshape(-1, 2)
    cons = []
    for i in range(N_CIRCLES):
        x, y = c[i]
        cons.append(x - r)
        cons.append(1 - x - r)
        cons.append(y - r)
        cons.append(1 - y - r)
    return np.array(cons)

def constraint_distance(vars):
    """Non-overlap constraints: dist^2 >= 4r^2"""
    r = vars[-1]
    c = vars[:N_CIRCLES*2].reshape(-1, 2)
    cons = []
    for i in range(N_CIRCLES):
        for j in range(i + 1, N_CIRCLES):
            dx = c[i, 0] - c[j, 0]
            dy = c[i, 1] - c[j, 1]
            cons.append(dx*dx + dy*dy - 4*r*r)
    return np.array(cons)

def get_initial_guess():
    """Generate hexagonal lattice initial guess"""
    centers = np.zeros((N_CIRCLES, 2))
    row_counts = [5, 6, 5, 6, 4] # Sum = 26
    target_spacing = 0.2
    dy = target_spacing * np.sqrt(3) / 2
    margin = target_spacing / 2
    y_positions = np.linspace(margin, 1 - margin, len(row_counts))
    
    idx = 0
    for r_idx, count in enumerate(row_counts):

        if r_idx % 2 == 1:
            start_x += target_spacing / 2
        
        for c in range(count):
            x = start_x + c * target_spacing
            centers[idx] = [x, y]
            idx += 1
    return centers

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    centers_init = get_initial_guess()
    x0 = np.concatenate([centers_init.flatten(), [0.05]]) # Start with r=0.05
