I'll tackle this circle packing problem using a multi-phase optimization approach:

1. **Initial Configuration**: Start with a hexagonal grid arrangement, which is the densest lattice packing. This gives a solid starting point.

2. **Optimization Method**: Use scipy's differential_evolution for global optimization combined with local refinement. The key challenge is that circle packing is non-convex, so I need global search capabilities.

3. **Objective Function**: Maximize sum of radii with heavy penalties for constraint violations (overlaps and boundary violations). The penalty approach converts constrained optimization to unconstrained.

4. **Key Insights for Better Packing**:
   - Corner circles can be larger (touching 2 boundaries leaves more room)
   - Edge circles are medium-sized (touching 1 boundary)
   - Interior circles are smaller (no boundary contact)
   - A hexagonal arrangement naturally creates space for size variation

5. **Implementation Strategy**:
   - Phase 1: Generate hexagonal initial positions
   - Phase 2: Use differential evolution with penalty constraints to find global optimum
   - Phase 3: Fine-tune with local optimization (Nelder-Mead or SLSQP)
   - Use multiple random restarts to avoid local optima
   - Gradually tighten penalty parameters to enforce constraints

The penalty function will be:
- For each pair (i,j): penalty = max(0, r_i + r_j - distance(i,j))²
- For boundaries: penalty = max(0, r_i - x_i)² + max(0, r_i - (1-x_i))² + similar for y
- Objective: sum(radii) - λ × total_penalty

def compute_constraint_violations(centers, radii):
    """Compute constraint violations for circle packing."""
    n = centers.shape[0]
    violations = []
    
    # Boundary violations
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        # Left boundary
        if x - r < 0:
            violations.append(r - x)
        # Right boundary
        if x + r > 1:
            violations.append(x + r - 1)
        # Bottom boundary
        if y - r < 0:
            violations.append(r - y)
        # Top boundary
        if y + r > 1:
            violations.append(y + r - 1)
    
    # Overlap violations
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            r_sum = radii[i] + radii[j]
            if dist < r_sum:
                violations.append(r_sum - dist)
    
    return np.array(violations) if violations else np.array([0.0])

def objective_function(params, n=26):
    """Objective function: maximize sum of radii (minimize negative sum)."""
    centers = params[:2 * n].reshape(n, 2)
    radii = params[2 * n:]
    
    # Ensure radii are non-negative
    radii = np.maximum(radii, 0)
    
    # Compute constraint violations
    violations = compute_constraint_violations(centers, radii)
    
    # Penalty term
    penalty = np.sum(violations ** 2)
    
    # Objective: maximize sum of radii, penalize violations
    obj = -np.sum(radii) + 1000 * penalty
    
    return obj

def objective_with_tight_penalty(params, n=26, penalty_weight=10000):
    """Objective with tight penalty for final optimization."""
    centers = params[:2 * n].reshape(n, 2)
    radii = params[2 * n:]
    
    radii = np.maximum(radii, 0)
    
    violations = compute_constraint_violations(centers, radii)
    penalty = np.sum(violations ** 2)
    
    obj = -np.sum(radii) + penalty_weight * penalty
    return obj

def local_refinement(centers, radii, n=26):
    """Perform local refinement using SLSQP with box constraints."""
    params0 = np.concatenate([centers.flatten(), radii])
    
    # Box constraints
    bounds = []
    for i in range(n):
        # x in [0, 1]
        bounds.append((0, 1))
        # y in [0, 1]
        bounds.append((0, 1))
        # r in [0, 0.5]
        bounds.append((0, 0.5))
    
    # Nonlinear constraints
    constraints = []
    
    # Boundary constraints: x - r >= 0, x + r <= 1, y - r >= 0, y + r <= 1
    for i in range(n):
        x_idx = 2 * i
        y_idx = 2 * i + 1
        r_idx = 2 * n + i
        
        # x - r >= 0
        cons = {
            'type': 'ineq',
            'fun': lambda p, x=x_idx, r=r_idx: p[x] - p[r],
            'jac': lambda p, x=x_idx, r=r_idx: np.zeros_like(p)
        }
        constraints.append(cons)
        
        # x + r <= 1
        cons = {
            'type': 'ineq',
            'fun': lambda p, x=x_idx, r=r_idx: 1 - p[x] - p[r],
            'jac': lambda p, x=x_idx, r=r_idx: np.zeros_like(p)
        }
        constraints.append(cons)
        
        # y - r >= 0
        cons = {
            'type': 'ineq',
            'fun': lambda p, y=y_idx, r=r_idx: p[y] - p[r],
            'jac': lambda p, y=y_idx, r=r_idx: np.zeros_like(p)
        }
        constraints.append(cons)
        
        # y + r <= 1
        cons = {
            'type': 'ineq',
            'fun': lambda p, y=y_idx, r=r_idx: 1 - p[y] - p[r],
            'jac': lambda p, y=y_idx, r=r_idx: np.zeros_like(p)
        }
        constraints.append(cons)
    
    # Overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            xi_idx = 2 * i
            yi_idx = 2 * i + 1
            xj_idx = 2 * j
            yj_idx = 2 * j + 1
            ri_idx = 2 * n + i
            rj_idx = 2 * n + j
            
            cons = {
                'type': 'ineq',
                'fun': lambda p, xi=xi_idx, yi=yi_idx, xj=xj_idx, yj=yj_idx, ri=ri_idx, rj=rj_idx:
                    (p[xi] - p[xj])**2 + (p[yi] - p[yj])**2 - (p[ri] + p[rj])**2 + 1e-6,
            }
            constraints.append(cons)
    
    try:
        result = minimize(
            lambda p: -np.sum(p[2*n:]),
            params0,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 1000, 'ftol': 1e-12}
        )
        
        if result.success or result.fun < -np.sum(radii):
            new_centers = result.x[:2 * n].reshape(n, 2)
            new_radii = np.maximum(result.x[2 * n:], 0)
            return new_centers, new_radii
    except:
        pass
    
    return centers, radii

def iterative_growth_optimization(n=26, n_iterations=100):
    """Iteratively grow circles while pushing them apart."""
    centers, radii = get_smart_initial_positions(n)
    
    for iteration in range(n_iterations):
        # Compute forces between circles
        forces = np.zeros_like(centers)
        
        for i in range(n):
            for j in range(i + 1, n):
                diff = centers[i] - centers[j]
                dist = np.sqrt(np.sum(diff**2))
                if dist < 1e-10:
                    dist = 1e-10
                    diff = np.array([0.01, 0.01])
                
                r_sum = radii[i] + radii[j]
                if dist < r_sum * 1.05:  # If too close
                    repulsion = (r_sum - dist) / dist * 10
                    forces[i] += diff * repulsion
                    forces[j] -= diff * repulsion
        
        # Apply boundary repulsion
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            
            # Push away from boundaries
            if x - r < 0.01:
                forces[i, 0] += (0.01 - (x - r)) * 20
            if x + r > 0.99:
                forces[i, 0] -= ((x + r) - 0.99) * 20
            if y - r < 0.01:
                forces[i, 1] += (0.01 - (y - r)) * 20
            if y + r > 0.99:
                forces[i, 1] -= ((y + r) - 0.99) * 20
        
        # Update positions
        learning_rate = 0.005 * (1 - iteration / n_iterations)
        centers += forces * learning_rate
        
        # Clip to [0, 1]
        centers = np.clip(centers, 0.01, 0.99)
        
        # Grow radii based on available space
        for i in range(n):
            min_dist = np.inf
            
            # Distance to boundaries
            min_dist = min(min_dist, centers[i, 0], 1 - centers[i, 0],
                          centers[i, 1], 1 - centers[i, 1])
            
            # Distance to other circles
            for j in range(n):
                if i != j:
                    d = np.sqrt(np.sum((centers[i] - centers[j])**2))
                    min_dist = min(min_dist, d - radii[j])
            
            # Set radius to available space
            new_r = min(min_dist, 0.25)
            radii[i] = max(new_r * 0.98, radii[i] * 0.99)  # Conservative growth
    
    return centers, radii

def validate_and_fix(centers, radii, n=26):
    """Validate and fix any constraint violations."""
    # Ensure all inside square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        
        # Fix boundary violations
        if x - r < 0:
            r = x
            radii[i] = r
        if x + r > 1:
            r = 1 - x
            radii[i] = r
        if y - r < 0:
            r = y
            radii[i] = r
        if y + r > 1:
            r = 1 - y
            radii[i] = r
        
        radii[i] = max(r, 0)
    
    # Fix overlaps by reducing radii
    for iteration in range(50):
        max_violation = 0
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                r_sum = radii[i] + radii[j]
                if dist < r_sum - 1e-10:
                    violation = r_sum - dist
                    max_violation = max(max_violation, violation)
                    # Reduce both radii
                    reduction = violation * 0.51
                    radii[i] = max(radii[i] - reduction / 2, 0)
                    radii[j] = max(radii[j] - reduction / 2, 0)
        
        if max_violation < 1e-8:
            break
    
    return centers, radii

def iterative_growth_optimization_from(centers, radii, n=26, n_iterations=100):
    """Iteratively grow circles from given initial configuration."""
    centers = centers.copy()
    radii = radii.copy()
    
    for iteration in range(n_iterations):
        forces = np.zeros_like(centers)
        
        for i in range(n):
            for j in range(i + 1, n):
                diff = centers[i] - centers[j]
                dist = np.sqrt(np.sum(diff**2))
                if dist < 1e-10:
                    dist = 1e-10
                    diff = np.array([np.random.randn(), np.random.randn()]) * 0.01
                
                r_sum = radii[i] + radii[j]
                if dist < r_sum * 1.1:
                    repulsion = max(0, (r_sum - dist)) / dist * 15
                    forces[i] += diff * repulsion
                    forces[j] -= diff * repulsion
        
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            
            margin = 0.02
            if x - r < margin:
                forces[i, 0] += (margin - (x - r)) * 30
            if x + r > 1 - margin:
                forces[i, 0] -= ((x + r) - (1 - margin)) * 30
            if y - r < margin:
                forces[i, 1] += (margin - (y - r)) * 30
            if y + r > 1 - margin:
                forces[i, 1] -= ((y + r) - (1 - margin)) * 30
        
        lr = 0.003 * (1 - iteration / n_iterations)
        centers += forces * lr
        centers = np.clip(centers, 0.01, 0.99)
        
        for i in range(n):
            min_dist = np.inf
            min_dist = min(min_dist, centers[i, 0], 1 - centers[i, 0],
                          centers[i, 1], 1 - centers[i, 1])
            
            for j in range(n):
                if i != j:
                    d = np.sqrt(np.sum((centers[i] - centers[j])**2))
                    min_dist = min(min_dist, d - radii[j])
            
            new_r = min(min_dist, 0.25)
            radii[i] = min(radii[i] * 1.002, max(new_r * 0.97, 0.01))
    
    return centers, radii
