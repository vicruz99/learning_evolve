To maximize the sum of radii for 26 circles in a unit square, we formulate this as a constrained nonlinear optimization problem. The objective is to maximize $\sum r_i$, which is equivalent to minimizing $-\sum r_i$. The constraints ensure that:
1. Each circle stays within $[0,1] \times [0,1]$ (handled via variable bounds).
2. Circles do not overlap: $\sqrt{(x_i-x_j)^2 + (y_i-y_j)^2} \ge r_i + r_j$ for all $i < j$.
3. Radii are non-negative.

We use `scipy.optimize.minimize` with the SLSQP algorithm, which handles inequality constraints and bounds efficiently. The problem is non-convex, so we mitigate local minima by running the optimizer from multiple diverse initial configurations:
- A hexagonal packing layout (optimal density for infinite planes)
- A regular grid layout
- Random perturbed layouts
Each trial starts with a feasible configuration (small radii, spaced centers) to guide the optimizer. The best result across trials is returned. Explicit Jacobians are provided to accelerate convergence and improve numerical stability.

import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26
# Precompute circle pairs for overlap constraints
PAIRS = [(i, j) for i in range(N_CIRCLES) for j in range(i+1, N_CIRCLES)]

def objective(vars_):
    """Objective: maximize sum of radii -> minimize negative sum."""
    return -np.sum(vars_[2::3])

def constraints_fun(vars_):
    """Inequality constraints: dist(i,j) - ri - rj >= 0"""
    con = np.empty(len(PAIRS))
    for k, (i, j) in enumerate(PAIRS):
        xi, yi, ri = vars_[3*i], vars_[3*i+1], vars_[3*i+2]
        xj, yj, rj = vars_[3*j], vars_[3*j+1], vars_[3*j+2]
        dx, dy = xi - xj, yi - yj
        con[k] = np.sqrt(dx*dx + dy*dy) - ri - rj
    return con

def constraints_jac(vars_):
    """Jacobian of inequality constraints."""
    jac_mat = np.zeros((len(PAIRS), 3*N_CIRCLES))
    for k, (i, j) in enumerate(PAIRS):
        xi, yi, ri = vars_[3*i], vars_[3*i+1], vars_[3*i+2]
        xj, yj, rj = vars_[3*j], vars_[3*j+1], vars_[3*j+2]
        dx, dy = xi - xj, yi - yj
        dist = np.sqrt(dx*dx + dy*dy)
        if dist < 1e-12:
            dist = 1e-12
        inv_dist = 1.0 / dist
        
        jac_mat[k, 3*i] = dx * inv_dist
        jac_mat[k, 3*i+1] = dy * inv_dist
        jac_mat[k, 3*i+2] = -1.0
        
        jac_mat[k, 3*j] = -dx * inv_dist
        jac_mat[k, 3*j+1] = -dy * inv_dist
        jac_mat[k, 3*j+2] = -1.0
    return jac_mat

def generate_initial_guess(strategy, seed=42):
    """Create feasible initial configuration."""
    n = N_CIRCLES
    rng = np.random.RandomState(seed)
    x0 = np.zeros(3*n)
    
    if strategy == 'hex':
        # Hexagonal-like layout: 5 rows with counts 6,5,6,5,4
        counts = [6, 5, 6, 5, 4]
        idx = 0
        h = 1.0 / 6.0
        for r_idx, cnt in enumerate(counts):
            y = (r_idx + 1) * h
            w = 1.0 / (cnt + 1)
            for c in range(cnt):
                x = (c + 1) * w
                x0[3*idx] = x
                x0[3*idx+1] = y
                x0[3*idx+2] = 0.07
                idx += 1
    elif strategy == 'grid':
        # 5x5 grid + center circle
        idx = 0
        for r in range(5):
            for c in range(5):
                x0[3*idx] = (c + 0.5) * 0.2
                x0[3*idx+1] = (r + 0.5) * 0.2
                x0[3*idx+2] = 0.06
                idx += 1
        x0[3*25] = 0.5
        x0[3*25+1] = 0.5
        x0[3*25+2] = 0.06
    else:
        # Random feasible start
        for i in range(n):
            x0[3*i] = 0.1 + 0.8 * rng.rand()
            x0[3*i+1] = 0.1 + 0.8 * rng.rand()
            x0[3*i+2] = 0.04
            
    # Small perturbation to break symmetry
    x0[:2*n] += 0.008 * rng.randn(2*n)
    x0[:2*n] = np.clip(x0[:2*n], 0.05, 0.95)
    return x0

def run_packing():
    constraints = {'type': 'ineq', 'fun': constraints_fun, 'jac': constraints_jac}
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N_CIRCLES
    
    best_sum_r = 0.0
    best_centers = None
    best_radii = None
    
    # Try multiple strategies to escape local minima
    configs = [('hex', 42), ('grid', 123), ('rand', 456), ('rand', 789), ('hex', 999)]
    
    for strat, seed in configs:
        x0 = generate_initial_guess(strat, seed)
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, 
                           constraints=constraints, 
                           options={'maxiter': 2000, 'ftol': 1e-10, 'disp': False})
            current_sum_r = -res.fun
            if current_sum_r > best_sum_r:
                best_sum_r = current_sum_r
                best_centers = res.x[:2*N_CIRCLES].reshape(N_CIRCLES, 2)
                best_radii = res.x[2::3]
        except Exception:
            continue
            
    # Fallback if optimization completely fails
    if best_centers is None:
        best_centers = np.tile([0.5, 0.5], (N_CIRCLES, 1))
        best_radii = np.ones(N_CIRCLES) * 0.01
        best_sum_r = np.sum(best_radii)
        
    return best_centers, best_radii, best_sum_r
