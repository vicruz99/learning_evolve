import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes the positions and radii of 26 circles in a unit square 
    to maximize the sum of radii.
    """
    n = 26
    num_vars = 3 * n  # x, y, r for each circle
    
    def objective(vars_):
        # Variables layout: [x1, y1, r1, x2, y2, r2, ...]
        # We want to maximize sum(r), so minimize -sum(r)
        return -np.sum(vars_[2::3])

    def constraints(vars_):
        cons = []
        for i in range(n):
            x_idx = 3 * i
            y_idx = 3 * i + 1
            r_idx = 3 * i + 2
            
            # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
            # Transformed to: r - x <= 0, r - (1-x) <= 0, etc.
            cons.append({'type': 'ineq', 'fun': lambda v, idx=r_idx, x=x_idx: v[x_idx] - v[idx]})
            cons.append({'type': 'ineq', 'fun': lambda v, idx=r_idx, x=x_idx: 1 - v[x_idx] - v[idx]})
            cons.append({'type': 'ineq', 'fun': lambda v, idx=r_idx, y=y_idx: v[y_idx] - v[idx]})
            cons.append({'type': 'ineq', 'fun': lambda v, idx=r_idx, y=y_idx: 1 - v[y_idx] - v[idx]})
            cons.append({'type': 'ineq', 'fun': lambda v, idx=r_idx: v[idx]}) # r >= 0
            
        # Non-overlap constraints: ||c_i - c_j|| >= r_i + r_j
        # Transformed to: ||c_i - c_j||^2 - (r_i + r_j)^2 >= 0
        for i in range(n):
            for j in range(i + 1, n):
                i_x, i_y, i_r = 3*i, 3*i+1, 3*i+2
                j_x, j_y, j_r = 3*j, 3*j+1, 3*j+2
                
                # Using a lambda with fixed arguments to avoid closure issues
                def cons_func(v, ix=i_x, iy=i_y, ir=i_r, jx=j_x, jy=j_y, jr=j_r):
                    dx = v[ix] - v[jx]
                    dy = v[iy] - v[jy]
                    dr = v[ir] + v[jr]
                    return (dx**2 + dy**2) - dr**2
                
                cons.append({'type': 'ineq', 'fun': cons_func})
                
        return cons

    def get_constraints():
        # Return constraints fresh to avoid state issues in some optimizers
        return constraints([0.0] * num_vars)

    best_result = None
    best_sum_r = -np.inf
    
    # We try multiple initializations to find a global optimum
    # Strategy 1: Hexagonal Grid Approximation
    # Strategy 2: Random Perturbations
    
    n_trials = 5
    
    for trial in range(n_trials):
        # Initial guess generation
        x = np.zeros(n)
        y = np.zeros(n)
        r = np.ones(n) * 0.05 # Small initial radius
        
        if trial == 0:
            # Hexagonal packing initialization
            # 5 rows, approx 5-6 circles per row
            # Row 0: 5 circles
            # Row 1: 6 circles (shifted)
            # Row 2: 5 circles
            # Row 3: 6 circles (shifted)
            # Row 4: 4 circles
            # Total 26
            
            row_counts = [5, 6, 5, 6, 4]
            idx = 0
            for row_i, count in enumerate(row_counts):
                y_base = 0.1 + row_i * 0.175 # Vertical spacing approx sqrt(3)/2 * 0.2
                x_start = 0.15 if row_i % 2 == 1 else 0.1
                spacing = 0.2 # Approx horizontal spacing
                
                for k in range(count):
                    if idx < n:
                        x[idx] = x_start + k * spacing
                        y[idx] = y_base
                        idx += 1
        else:
            # Random initialization inside square with safe padding
            pad = 0.1
            x = np.random.uniform(pad, 1-pad, n)
            y = np.random.uniform(pad, 1-pad, n)
            
        # Construct initial variable vector
        x0 = np.zeros(num_vars)
        x0[0::3] = x
        x0[1::3] = y
        x0[2::3] = r
        
        # Run optimization
        try:
            res = minimize(objective, x0, method='SLSQP', constraints=get_constraints(), 
                           options={'maxiter': 1000, 'ftol': 1e-9})
            
            if res.success:
                current_sum_r = -res.fun
                if current_sum_r > best_sum_r:
                    best_sum_r = current_sum_r
                    best_result = res
        except Exception:
            continue
            
    if best_result is None:
        # Fallback if optimization fails (unlikely)
        centers = np.random.rand(n, 2) * 0.5 + 0.25
        radii = np.ones(n) * 0.01
        return centers, radii, np.sum(radii)

    x_opt = best_result.x[0::3]
    y_opt = best_result.x[1::3]
    r_opt = best_result.x[2::3]

    radii = np.maximum(r_opt, 1e-9)

    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
