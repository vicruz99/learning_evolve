def compute_overlap_penalty(centers, radii, epsilon=1e-9):
    """
    Helper to check overlaps manually if needed, though constraints handle it.
    """
    n = centers.shape[0]
    penalty = 0.0
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            min_dist = radii[i] + radii[j]
            if dist < min_dist:
                penalty += (min_dist - dist)
    return penalty

def boundary_constraint(centers, radii):
    """
    Ensure circles are inside [0,1]x[0,1].
    Constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    Returns array of values that must be >= 0.

    constraints = []

        constraints.append(x - r)
        constraints.append(1 - x - r)
        constraints.append(y - r)
        constraints.append(1 - y - r)
    return np.array(constraints)

def non_overlap_constraints(centers, radii):

    Let's use the direct distance constraint but be careful with 0 distance.
    Actually, SLSQP works better with smooth constraints.
    dist - (r_i + r_j) >= 0 is smooth except at 0.
    We assume initial centers are distinct.

    n = centers.shape[0]

            # But physically centers shouldn't coincide in optimal packing.

            # Let's use the squared form to avoid sqrt issues?
            # dist^2 >= (r_i + r_j)^2

            val = (centers[i][0] - centers[j][0])**2 + \

    return np.array(constraints)

    return -np.sum(radii)

def get_constraints(n):
    """
    Factory function to return constraint dictionary for scipy.
    """
    constraints = []
    
    # Boundary constraints for each circle
    # r_i <= x_i => x_i - r_i >= 0
    # r_i <= 1-x_i => 1 - x_i - r_i >= 0
    # r_i <= y_i => y_i - r_i >= 0
    # r_i <= 1-y_i => 1 - y_i - r_i >= 0
    for i in range(n):
        idx = i * 3
        x_idx, y_idx, r_idx = idx, idx + 1, idx + 2
        
        # x - r >= 0
        def make_func_x_lb(j):
            def f(v):
                return v[j] - v[j+2]
            return f
        constraints.append({'type': 'ineq', 'fun': make_func_x_lb(idx)})
        
        # 1 - x - r >= 0
        def make_func_x_ub(j):
            def f(v):
                return 1 - v[j] - v[j+2]
            return f
        constraints.append({'type': 'ineq', 'fun': make_func_x_ub(idx)})
        
        # y - r >= 0
        def make_func_y_lb(j):
            def f(v):
                return v[j+1] - v[j+2]
            return f
        constraints.append({'type': 'ineq', 'fun': make_func_y_lb(idx)})
        
        # 1 - y - r >= 0
        def make_func_y_ub(j):
            def f(v):
                return 1 - v[j+1] - v[j+2]
            return f
        constraints.append({'type': 'ineq', 'fun': make_func_y_ub(idx)})

    # Non-overlap constraints: (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
    for i in range(n):
        for j in range(i + 1, n):
            idx_i = i * 3
            idx_j = j * 3
            
            def make_overlap_func(ii, jj):
                def f(v):
                    x_diff = v[ii] - v[jj]
                    y_diff = v[ii+1] - v[jj+1]
                    r_sum = v[ii+2] + v[jj+2]
                    return (x_diff**2 + y_diff**2) - (r_sum**2)
                return f
            
            constraints.append({'type': 'ineq', 'fun': make_overlap_func(idx_i, idx_j)})
            
    return constraints
