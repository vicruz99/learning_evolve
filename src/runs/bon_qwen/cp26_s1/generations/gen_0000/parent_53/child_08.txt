import numpy as np
from scipy.optimize import minimize
import math

def run_packing():
    # Number of circles
    n = 26
    
    # Objective function to minimize: sum of squared overlaps
    # We assume equal radii r for the optimization of positions
    # The radius r is passed as a parameter
    def objective(x, r):
        centers = x.reshape(-1, 2)
        penalty = 0.0
        # Compute pairwise distances and check for overlap
        # Overlap condition: dist < 2*r
        # Penalty: (2*r - dist)^2
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = math.sqrt(dx*dx + dy*dy)
                if dist < 2 * r:
                    penalty += (2 * r - dist) ** 2
        return penalty

    # Initial configuration
    # We start with a small radius to ensure feasibility
    # and a random placement within safe bounds.
    r_start = 0.05
    rng = np.random.default_rng(42)
    # Place centers randomly in [r_start, 1-r_start]
    # Actually, just [0.1, 0.9] is safe for r=0.05
    x0 = rng.uniform(0.1, 0.9, size=(2 * n,))
    
    # Ensure x0 is valid for r_start (it is, since 0.1 >= 0.05 and 0.9 <= 0.95)
    
    # We will perform a binary search (or linear search) to find the maximum r.
    # Range [0.05, 0.15]
    low = r_start
    high = 0.15
    
    # Current best valid configuration
    best_r = low
    best_centers_flat = x0.copy()
    
    # Pre-calculate bounds generator
    # Bounds for coordinates are [r, 1-r]
    
    # Optimization parameters
    opt_options = {'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-12}
    
    # Binary search iterations
    # 40 iterations gives high precision
    for _ in range(40):
        mid = (low + high) / 2
        
        # Define bounds for this radius
        # x coordinates must be in [mid, 1-mid]
        # y coordinates must be in [mid, 1-mid]
        bounds = [(mid, 1 - mid)] * (2 * n)
        
        # Prepare initial guess: project current best centers to new bounds
        # Because increasing r shrinks the valid region
        x_init = best_centers_flat.copy()
        for i in range(2 * n):
            val = x_init[i]
            if val < mid:
                val = mid
            elif val > 1 - mid:
                val = 1 - mid
            x_init[i] = val
            
        # Run optimization
        # We use L-BFGS-B which handles bounds
        res = minimize(objective, x_init, args=(mid,), method='L-BFGS-B', bounds=bounds, options=opt_options)
        
        # Check if solution is valid (penalty ~ 0)
        final_penalty = objective(res.x, mid)
        
        if final_penalty < 1e-8:
            # Feasible, try larger radius
            low = mid
            best_r = mid
            best_centers_flat = res.x.copy()
        else:
            # Not feasible, reduce radius
            high = mid

    radii = np.full(n, best_r)
    sum_radii = np.sum(radii)
    
    # Validation check (optional but good for debugging)
    # In a real scenario, we trust the optimizer, but let's just return.
    
    return centers, radii, sum_radii

import numpy as np
from scipy.optimize import minimize

def objective_function(x, r, n):
    """
    Computes the penalty for overlapping circles.
    x: 1D array of size 2*n representing centers (x1, y1, x2, y2, ...)
    r: radius of circles (assumed equal for this objective)
    n: number of circles
    """
    centers = x.reshape(-1, 2)
    # Compute pairwise distances using broadcasting
    # diff shape: (n, n, 2)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    # distances shape: (n, n)
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    # Extract upper triangle elements (excluding diagonal) to check unique pairs
    i_idx, j_idx = np.triu_indices(n, k=1)
    upper_dists = dists[i_idx, j_idx]
    
    # Minimum allowed distance is 2*r
    min_dist = 2 * r
    overlaps = min_dist - upper_dists
    
    # Penalty is sum of squared positive overlaps
    pos_overlaps = overlaps[overlaps > 0]
    return np.sum(pos_overlaps**2)

def run_packing():
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    Uses a binary search approach to find the maximum equal radius r
    such that 26 circles of radius r can be packed without overlap.
    """
    n = 26
    
    # Initial configuration: random placement within a safe margin
    # Using a fixed seed for reproducibility
    rng = np.random.default_rng(42)
    
    # Start with a small radius to ensure feasibility
    r_start = 0.05
    # Place centers in [0.1, 0.9] to keep away from boundaries initially
    x0 = rng.uniform(0.1, 0.9, size=(2 * n,))
    
    # Binary search range for radius
    low = r_start
    high = 0.15  # Theoretical upper bound is lower, but safe start
    
    best_r = low
    best_centers_flat = x0.copy()
    
    # Optimization options for L-BFGS-B
    # maxiter: max iterations per optimization step
    # ftol, gtol: tolerance for convergence
    opt_options = {'maxiter': 150, 'ftol': 1e-12, 'gtol': 1e-12}
    
    # Number of binary search steps
    # 35 steps gives high precision: 0.1 / 2^35 is negligible
    for _ in range(35):
        mid = (low + high) / 2
        
        # Define bounds for the optimizer
        # Center coordinates must be within [mid, 1-mid] to satisfy boundary constraints
        # This ensures circle i is inside [0, 1]
        bounds = [(mid, 1 - mid)] * (2 * n)
        
        # Prepare initial guess for the optimizer
        # We use the best configuration found so far (valid for 'low')
        # But we must project it to the new valid region [mid, 1-mid]
        # because 'mid' > 'low' implies a smaller valid region.
        x_init = best_centers_flat.copy()
        x_init = np.clip(x_init, mid, 1 - mid)
        
        # Run optimization to minimize overlap penalty for radius 'mid'
        res = minimize(objective_function, x_init, args=(mid, n), 
                       method='L-BFGS-B', bounds=bounds, options=opt_options)
        
        # Evaluate the penalty at the result
        penalty = objective_function(res.x, mid, n)
        
        # If penalty is near zero, the configuration is valid for radius 'mid'
        # We can try to increase the radius further.
        if penalty < 1e-8:
            low = mid
            best_r = mid
            best_centers_flat = res.x.copy()
        else:
            # If not valid, the radius 'mid' is too large.
            high = mid
            
    # Reshape centers and create radii array
    centers = best_centers_flat.reshape(-1, 2)
    radii = np.full(n, best_r)
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
