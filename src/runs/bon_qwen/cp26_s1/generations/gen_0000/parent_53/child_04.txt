def get_min_radius(centers):
    """
    Calculates the maximum equal radius 'r' such that circles at 'centers'
    do not overlap and stay within [0,1]x[0,1].
    Returns r.
    """

    # r <= x, r <= 1-x, r <= y, r <= 1-y
    # So r <= min(x, 1-x, y, 1-y)

    # We need r to satisfy this for ALL i.
    # So r <= min over i of (min(x_i, 1-x_i, y_i, 1-y_i))
    
    # Calculate boundary constraints
    # x coords
    x = centers[:, 0]
    y = centers[:, 1]
    
    # Distance to vertical walls
    dist_v = np.minimum(x, 1.0 - x)
    # Distance to horizontal walls
    dist_h = np.minimum(y, 1.0 - y)
    
    # Min distance to any wall for each circle
    dist_wall = np.minimum(dist_v, dist_h)
    
    # The radius must be <= dist_wall[i] for all i
    min_r_wall = np.min(dist_wall)
    
    # Distance between circles
    # r_i + r_j <= dist(c_i, c_j) => 2r <= dist => r <= dist/2
    # We need to check all pairs
    # Optimization: use broadcasting
    # centers shape (n, 2)
    # diff shape (n, n, 2)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    # Mask out self-distances and lower triangle
    np.fill_diagonal(dists, np.inf)
    # We only need min of upper triangle or just min of all (symmetric)
    min_r_pair = np.min(dists) / 2.0
    
    # The feasible radius is the minimum of these constraints
    r = min(min_r_wall, min_r_pair)
    return r
