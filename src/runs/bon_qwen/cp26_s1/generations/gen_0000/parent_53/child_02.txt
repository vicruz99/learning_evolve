import numpy as np
from scipy.optimize import minimize

def run_packing():

    Packs 26 circles in a unit square to maximize the sum of radii.
    """
    n_circles = 26
    
    # Strategy: Optimize for equal radii first, as it is often near-optimal for sum of radii
    # due to area efficiency. We can relax this if needed, but equal radii is robust.
    # Variables: x1, y1, ..., x26, y26, r
    # Total 26*2 + 1 = 53 variables

        # We want to maximize sum of radii. 
        # For equal radii case, vars[-1] is r. 
        # Objective is -r (to minimize)
        return -vars[-1]

    def objective_general(vars):
        # General case: vars are x1, y1, r1, x2, y2, r2 ...

        radii = vars[2::3]
        return -np.sum(radii)

    rows = 5 # 6*5 = 30 > 26

    grid_y = np.linspace(1/6, 1 - 1/6, rows)   # Spacing 1/5 approx
    
    initial_centers = []
    idx = 0
    for r_idx in range(rows):
        for c_idx in range(cols):
            if idx < n_circles:

                idx += 1
            else:
                break

            break
            
    initial_centers = np.array(initial_centers)

    bounds = []
    for _ in range(n_circles):

    # Equivalent to: dist^2 - (r_i + r_j)^2 >= 0

    # Let's use dist - r_i - r_j >= 0 for better conditioning near 0?
    # Actually, squared is standard.
