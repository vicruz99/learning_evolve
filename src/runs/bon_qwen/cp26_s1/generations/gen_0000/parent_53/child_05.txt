import numpy as np
import scipy.optimize

def calculate_radii(centers):
    """
    Calculates the maximum possible radius for each circle given the centers.
    Radius is limited by boundaries and distance to other circles.
    """
    n = centers.shape[0]
    radii = np.full(n, 0.5)

    # Boundary constraints
    radii = np.minimum(radii, centers[:, 0])
    radii = np.minimum(radii, 1.0 - centers[:, 0])
    radii = np.minimum(radii, centers[:, 1])
    radii = np.minimum(radii, 1.0 - centers[:, 1])

    # Pairwise distance constraints
    # Efficiently calculate pairwise distances
    # diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    # dists = np.sqrt(np.sum(diff**2, axis=2))
    
    # Manual loop to save memory and avoid NaN issues if needed, 
    # though vectorized is faster. Given n=26, vectorized is fine.
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    # Radius constraint: r_i + r_j <= dist_ij
    # This implies r_i <= dist_ij - r_j. 
    # In a simultaneous equilibrium, r_i = dist_ij / 2 is a safe bound if we assume equal influence,
    # but strictly, we want to find r such that r_i + r_j <= dist_ij.
    # A robust way to set radii for a given configuration is:
    # r_i = min( boundary_dist_i, min_j( dist_ij / 2 ) )
    # This ensures r_i + r_j <= dist_ij / 2 + dist_ij / 2 = dist_ij.
    
    # Calculate min distance to any other circle for each circle
    # Set diagonal to infinity to ignore self-distance
    dists_min = np.min(np.where(dists < 1e-9, np.inf, dists), axis=1)
    radii = np.minimum(radii, dists_min / 2.0)
    
    return radii

def objective_function(coords, n_circles):
    """
    Objective function for optimization. Minimizes negative sum of radii.
    """
    centers = coords.reshape((n_circles, 2))
    radii = calculate_radii(centers)
    return -np.sum(radii)

def run_packing():
    n_circles = 26
    num_vars = n_circles * 2
    
    # Bounds for variables: x in [0, 1], y in [0, 1]
    # We use a small margin to avoid exact boundary issues, though calculate_radii handles it.
    bounds = [(0.0, 1.0)] * num_vars
    
    # Initial population size and iteration count for Differential Evolution
    # Larger population helps avoid local minima
    pop_size = 50
    max_iter = 200
    
    # Seed for reproducibility if desired, but random is okay
    # Using multiple restarts manually might be better but DE is robust
    
    # We can define a custom mutation strategy or use defaults.
    # strategy='best1bin' is default.
    
    # To improve performance, we can provide a set of initial guesses if DE supports it, 
    # but standard DE generates its own. 
    # Let's just run DE.
    
    result = scipy.optimize.differential_evolution(
        objective_function,
        bounds,
        args=(n_circles,),
        popsize=pop_size,
        maxiter=max_iter,
        tol=1e-7,
        mutation=(0.5, 1.0),
        recombination=0.7,
        seed=42,
        polish=True, # Perform local optimization after DE
        updating='deferred'
    )
    
    best_coords = result.x
    centers = best_coords.reshape((n_circles, 2))
    radii = calculate_radii(centers)
    sum_radii = np.sum(radii)
    
    # Final refinement step using a local search (Nelder-Mead or similar) 
    # just in case DE stopped slightly off or to polish the radii calculation logic.
    # Actually, calculate_radii is non-differentiable at contacts, so gradient methods struggle.
    # But DE with polish=True (which uses L-BFGS-B or similar) might get stuck.
    # Let's try a simple local perturbation to ensure we are optimal.
    
    # Perturbation search
    current_sum = sum_radii

        improved = False
        for i in range(n_circles):
            best_move = None
            best_val = current_sum

                        best_val = test_sum
                        best_move = test_centers
            
            if best_move is not None:
                centers = best_move
                radii = calculate_radii(centers)
                current_sum = best_val
                improved = True
        
        if not improved:
            break
            
    sum_radii = current_sum
    
    return centers, radii, sum_radii
