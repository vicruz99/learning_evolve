import numpy as np
from scipy.optimize import minimize

def objective_func(v):
    """Objective: maximize sum of radii (minimize negative sum)."""
    radii = v[2::3]
    return -np.sum(radii)

def constraints_func(v):
    """Compute all inequality constraints (must be >= 0 for SLSQP 'ineq')."""
    n = 26
    n_boundary = 4 * n
    n_overlap = n * (n - 1) // 2
    con = np.zeros(n_boundary + n_overlap)
    
    idx = 0
    # Boundary constraints
    for i in range(n):
        base = 3 * i
        x, y, r = v[base], v[base+1], v[base+2]
        con[idx]   = x - r          # x >= r
        con[idx+1] = 1.0 - x - r    # x <= 1-r
        con[idx+2] = y - r          # y >= r
        con[idx+3] = 1.0 - y - r    # y <= 1-r
        idx += 4
        
    # Overlap constraints
    for i in range(n):
        ci = 3 * i
        xi, yi, ri = v[ci], v[ci+1], v[ci+2]
        for j in range(i + 1, n):
            cj = 3 * j
            xj, yj, rj = v[cj], v[cj+1], v[cj+2]
            con[idx] = (xi - xj)**2 + (yi - yj)**2 - (ri + rj)**2
            idx += 1
            
    return con

def run_packing() -> tuple:
    n = 26
    
    # Initialization 1: Hexagonal grid
    centers1 = np.zeros((n, 2))
    radii1 = np.full(n, 0.04)
    idx = 0
    y = 0.08
    offset = 0.0
    while idx < n:
        x = 0.08 + offset
        while x < 0.92 and idx < n:
            centers1[idx] = [x, y]
            idx += 1
            x += 0.18
        y += 0.15
        offset = 0.09 if offset == 0.0 else 0.0
        
    # Initialization 2: Randomized layout
    rng = np.random.RandomState(42)
    centers2 = rng.uniform(0.1, 0.9, (n, 2))
    radii2 = np.full(n, 0.03)
    
    best_sum = -1.0
    best_centers = centers1
    best_radii = radii1
    
    # Run optimization from both starts
    for centers_init, radii_init in [(centers1, radii1), (centers2, radii2)]:
        x0 = np.concatenate([centers_init.ravel(), radii_init])
        bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n
        cons = {'type': 'ineq', 'fun': constraints_func}
        
        res = minimize(objective_func, x0, method='SLSQP', bounds=bounds, 
                       constraints=cons, options={'maxiter': 8000, 'ftol': 1e-12})
                       
        curr_centers = res.x[:2*n].reshape((n, 2))
        curr_radii = res.x[2*n:]
        
        # Post-process to strictly satisfy bounds and non-negativity
        curr_radii = np.maximum(curr_radii, 0.0)
        curr_centers = np.clip(curr_centers, curr_radii[:, None], 1.0 - curr_radii[:, None])
        
        curr_sum = np.sum(curr_radii)
        if curr_sum > best_sum:
            best_sum = curr_sum
            best_centers = curr_centers
            best_radii = curr_radii
            
    return best_centers, best_radii, best_sum
