

<strategy>
To maximize the sum of radii for 26 circles in a unit square, we formulate it as a constrained nonlinear optimization problem. While equal-radius packings are often near-optimal, allowing radii to vary gives the optimizer more freedom to exploit local gaps, potentially yielding a higher sum. We use `scipy.optimize.minimize` with the SLSQP method, which handles nonlinear inequality constraints effectively.

Key design choices:
1. **Variable Layout**: Concatenate x-coordinates, y-coordinates, and radii into a single vector of length $3N = 78$.
2. **Objective**: Minimize the negative sum of radii.
3. **Constraints**: 
   - Boundary: $0 \le x_i - r_i$ and $x_i + r_i \le 1$ (similarly for $y$).
   - Non-overlap: $\sqrt{(x_i-x_j)^2 + (y_i-y_j)^2} \ge r_i + r_j$ for all $i < j$.
   All constraints are vectorized for efficiency.
4. **Initialization & Robustness**: Run the optimizer from 40 different random starting configurations to escape local minima. Initial radii are set to ~0.05-0.10 to give the solver room to expand.
5. **Post-processing**: Compute the minimum clearance to boundaries and between circles. If numerical optimization leaves tiny violations ($< 10^{-12}$), uniformly scale radii down by a negligible factor to guarantee strict validity per the checker's tolerance.

```python
import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def objective_unequal(v, N):
    """Objective: maximize sum of radii -> minimize negative sum"""
    return -np.sum(v[2*N:])

def constraints_unequal(v, N):
    """Constraints: boundaries and non-overlap"""
    cx = v[:N]
    cy = v[N:2*N]
    r = v[2*N:]
    centers = np.column_stack((cx, cy))
    
    # Boundary constraints: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
    c1 = cx - r
    c2 = 1.0 - cx - r
    c3 = cy - r
    c4 = 1.0 - cy - r
    
    # Pairwise non-overlap: dist >= r_i + r_j
    dists = cdist(centers, centers)
    idx = np.triu_indices(N, k=1)
    c5 = dists[idx] - r[idx[0]] - r[idx[1]]
    
    return np.concatenate([c1, c2, c3, c4, c5])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    N = 26
    best_sum = 0.0
    best_v = None
    
    # Multiple restarts to escape local minima
    for seed in range(40):
        rng = np.random.RandomState(seed)
        x0_cx = rng.uniform(0.2, 0.8, N)
        x0_cy = rng.uniform(0.2, 0.8, N)
        x0_r = rng.uniform(0.05, 0.10, N)
        x0 = np.concatenate([x0_cx, x0_cy, x0_r])
        
        bounds = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N
        
        try:
            res = minimize(
                fun=objective_unequal,
                x0=x0,
                bounds=bounds,
                args=(N,),
                constraints={'type': 'ineq', 'fun': constraints_unequal, 'args': (N,)},
                method='SLSQP',
                options={'maxiter': 2000, 'ftol': 1e-12, 'disp': False}
            )
            if -res.fun > best_sum:
                best_sum = -res.fun
                best_v = res.x.copy()
        except Exception:
            continue
            
    if best_v is None:
        # Fallback configuration
        best_v = np.zeros(3*N)
        best_v[:N] = 0.5
        best_v[N:2*N] = 0.5
        best_v[2*N:] = 0.04
        
    centers = np.column_stack((best_v[:N], best_v[N:2*N]))
    radii = best_v[2*N:]
    
    # Post-processing to guarantee validity within numerical tolerance
    # Compute minimum clearance to boundaries and between circles
    min_clear = 1.0
    for i in range(N):
        min_clear = min(min_clear, 
                        centers[i,0]-radii[i], 1.0-centers[i,0]-radii[i],
                        centers[i,1]-radii[i], 1.0-centers[i,1]-radii[i])
                        
    dists = cdist(centers, centers)
    np.fill_diagonal(dists, np.inf)
    # Check pairwise clearances
    pairwise_clear = dists - radii[:, None] - radii[None, :]
    min_clear = min(min_clear, np.min(pairwise_clear))
    
    # If any overlap or boundary violation (even tiny numerical error), scale radii down slightly
    if min_clear < 1e-12:
        scale_factor = 1.0 + min_clear / (np.max(radii) + 1e-9)
        radii *= max(scale_factor, 0.9999999)
        
    return centers, radii, float(np.sum(radii))
```
