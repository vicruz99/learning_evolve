import numpy as np
from scipy.optimize import linprog
from scipy.spatial.distance import pdist, squareform

def solve_radii(centers):
    """
    Given fixed centers, solves the LP to find radii that maximize sum of radii.
    Returns (radii, sum_radii).
    """
    n = centers.shape[0]
    
    # Distance matrix
    dist_matrix = squareform(pdist(centers))
    
    # Boundary distances for each center
    x = centers[:, 0]
    y = centers[:, 1]
    boundary_dists = np.minimum(np.minimum(x, 1 - x), np.minimum(y, 1 - y))
    
    # LP Formulation:
    # Minimize -sum(radii) => c = -1
    c = -np.ones(n)
    
    # Constraints: A_ub * r <= b_ub
    # 1. Pairwise constraints: r_i + r_j <= dist(i, j)
    # 2. Boundary constraints: r_i <= boundary_dist(i)
    
    num_pair_constraints = n * (n - 1) // 2
    num_boundary_constraints = n
    A_ub = np.zeros((num_pair_constraints + num_boundary_constraints, n))
    b_ub = np.zeros(num_pair_constraints + num_boundary_constraints)
    
    # Fill pairwise constraints
    row_idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[row_idx, i] = 1
            A_ub[row_idx, j] = 1
            b_ub[row_idx] = dist_matrix[i, j]
            row_idx += 1
            
    # Fill boundary constraints
    for i in range(n):
        A_ub[row_idx, i] = 1
        b_ub[row_idx] = boundary_dists[i]
        row_idx += 1
        
    # Bounds for radii: r_i >= 0
    bounds = [(0, None) for _ in range(n)]
    
    # Solve LP
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        radii = res.x
        sum_radii = np.sum(radii)
        return radii, sum_radii
    else:
        # Fallback if LP fails (should not happen with valid inputs)
        return np.zeros(n), 0.0
