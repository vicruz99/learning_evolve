import numpy as np
import scipy.optimize as opt

def run_packing():
    """
    Pack 26 circles in a unit square to maximize the sum of radii.
    """
    N = 26
    best_sum = -1.0
    best_centers = None
    best_radii = None

    # Helper function to define constraints for the optimizer
    # We optimize variables: x_0, y_0, r_0, x_1, y_1, r_1, ...
    # Or simpler: centers (Nx2) and radii (N).
    # Let's flatten to 1D array: [x0, y0, r0, x1, y1, r1, ...]
    
    def objective(vars):
        # vars shape: (3*N,)
        # Maximize sum of radii -> Minimize negative sum
        radii = vars[2::3]
        return -np.sum(radii)

    def constraints(vars):
        centers = np.column_stack((vars[0::3], vars[1::3]))
        radii = vars[2::3]
        
        cons = []
        
        # Boundary constraints: x >= r, x <= 1-r, y >= r, y <= 1-r
        # x - r >= 0
        # 1 - x - r >= 0
        # y - r >= 0
        # 1 - y - r >= 0
        for i in range(N):
            x, y = centers[i]
            r = radii[i]
            cons.append(x - r)          # x >= r
            cons.append(1.0 - x - r)    # x <= 1-r
            cons.append(y - r)          # y >= r
            cons.append(1.0 - y - r)    # y <= 1-r
            cons.append(r)              # r >= 0 (though usually implied or handled by bounds)

        # Non-overlap constraints: dist(i, j) >= r_i + r_j
        # dist^2 >= (r_i + r_j)^2
        # (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
        for i in range(N):
            for j in range(i + 1, N):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist_sq = dx*dx + dy*dy
                r_sum = radii[i] + radii[j]
                cons.append(dist_sq - r_sum*r_sum)
                
        return np.array(cons)

    # x, y in [0, 1]
    # r in [0, 0.5] (max possible radius is 0.5)

        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 0.5)) # r

        r_est = 0.1
        centers = []

        # Hex spacing y = r * sqrt(3)

        num_rows = 5
        row_y_spacing = r_est * np.sqrt(3)

        # Let's just fill rows

            y = r_est + k * row_y_spacing
            
            # Shift odd rows

                x = r_est + shift + m * (2 * r_est)
                centers.append([x, y])
                current_count += 1
                if current_count == N:
                    break
            if current_count == N:
                break
        
        return np.array(centers)

    # Just another random seed
