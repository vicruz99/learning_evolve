import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    
    Returns:
        centers: np.array of shape (26, 2)
        radii: np.array of shape (26,)
        sum_radii: float
    """
    n_circles = 26
    
    def hexagonal_initial_guess(scale_factor=0.8):
        """
        Generates an initial guess based on a hexagonal packing pattern.
        scale_factor < 1 ensures no overlaps initially.
        """
        r_init = 0.04 * scale_factor # Small enough to fit
        centers = []
        
        # Hexagonal grid parameters
        # Vertical spacing: r * sqrt(3)
        # Horizontal spacing: 2 * r
        # Shift alternating rows by r
        
        row_height = r_init * np.sqrt(3)
        row_shift = r_init
        y_start = r_init
        
        row_idx = 0
        while len(centers) < n_circles:
            y = y_start + row_idx * row_height
            
            # Determine x start based on row parity (shift)
            if row_idx % 2 == 0:
                x_start = r_init
            else:
                x_start = r_init + row_shift # Shifted by r (diameter is 2r, so shift is half diameter? No, shift is r for tangent)
                # Wait, in hex packing with diameter 2r, horizontal shift is r.
                # If row 0 centers at r, 3r, 5r...
                # Row 1 centers at 2r, 4r, 6r... (Shift is r)
                # But 2r is distance r from r. Correct.
            
            # Place circles in this row
            x = x_start
            while x <= 1 - r_init:
                centers.append([x, y])
                x += 2 * r_init
                if len(centers) >= n_circles:
                    break
            row_idx += 1
            
        # Trim to exactly n_circles
        centers = centers[:n_circles]
        
        # Construct variable vector: [x0, y0, r0, x1, y1, r1, ...]
        # We set initial r to r_init
        vars = []
        for x, y in centers:
            vars.extend([x, y, r_init])
            
        return np.array(vars)

    def objective(vars):
        # Minimize negative sum of radii -> Maximize sum of radii
        # Radii are at indices 2, 5, 8, ... (3*i + 2)
        radii = vars[2::3]
        return -np.sum(radii)

    def constraint_bound_left(vars, i):
        # x_i - r_i >= 0
        return vars[3*i] - vars[3*i + 2]

    def constraint_bound_right(vars, i):
        # 1 - x_i - r_i >= 0
        return 1 - vars[3*i] - vars[3*i + 2]

    def constraint_bound_bottom(vars, i):
        # y_i - r_i >= 0
        return vars[3*i + 1] - vars[3*i + 2]

    def constraint_bound_top(vars, i):
        # 1 - y_i - r_i >= 0
        return 1 - vars[3*i + 1] - vars[3*i + 2]

    def constraint_non_overlap(vars, i, j):
        # dist^2 - (r_i + r_j)^2 >= 0
        x_i, y_i, r_i = vars[3*i], vars[3*i+1], vars[3*i+2]
        x_j, y_j, r_j = vars[3*j], vars[3*j+1], vars[3*j+2]
        dx = x_i - x_j
        dy = y_i - y_j
        return dx*dx + dy*dy - (r_i + r_j)**2

    def solve_packing(initial_vars):
        n_vars = len(initial_vars)
        
        # Define bounds for each variable
        # x in [0, 1], y in [0, 1], r in [0, 0.5]
        bounds = []
        for _ in range(n_circles):
            bounds.extend([(0, 1), (0, 1), (0, 0.5)])
            
        # Define inequality constraints (func >= 0)
        constraints = []
        
        # Boundary constraints
        for i in range(n_circles):
            constraints.append({'type': 'ineq', 'fun': lambda v, idx=i: constraint_bound_left(v, idx)})
            constraints.append({'type': 'ineq', 'fun': lambda v, idx=i: constraint_bound_right(v, idx)})
            constraints.append({'type': 'ineq', 'fun': lambda v, idx=i: constraint_bound_bottom(v, idx)})
            constraints.append({'type': 'ineq', 'fun': lambda v, idx=i: constraint_bound_top(v, idx)})
            
        # Non-overlap constraints
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                constraints.append({'type': 'ineq', 'fun': lambda v, idx_i=i, idx_j=j: constraint_non_overlap(v, idx_i, idx_j)})
        
        # Run optimizer
        try:
            res = minimize(
                objective,
                initial_vars,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 1000, 'ftol': 1e-9, 'disp': False}
            )
            return res
        except Exception:
            return None

    # Strategy: Multiple random restarts with perturbations
    best_sum_radii = -1.0
    best_centers = None
    best_radii = None
    
    # Number of restarts
    n_restarts = 10
    
    for k in range(n_restarts):
        # Generate initial guess
        # Base hexagonal guess
        base_vars = hexagonal_initial_guess(scale_factor=0.9)
        
        # Add noise to escape local minima
        noise_scale = 0.02
        noisy_vars = base_vars + np.random.normal(0, noise_scale, base_vars.shape)
        
        # Clip noisy variables to valid bounds roughly to ensure feasibility
        # x, y in [0,1], r > 0
        for i in range(n_circles):
            noisy_vars[3*i] = np.clip(noisy_vars[3*i], 0, 1)
            noisy_vars[3*i+1] = np.clip(noisy_vars[3*i+1], 0, 1)
            noisy_vars[3*i+2] = max(0.01, noisy_vars[3*i+2]) # Ensure positive radius
            
        res = solve_packing(noisy_vars)
        
        if res is not None and res.success:
            current_sum = -res.fun # Negative of objective
            if current_sum > best_sum_radii:
                best_sum_radii = current_sum
                # Extract centers and radii
                best_vars = res.x
                best_centers = np.column_stack((best_vars[0::3], best_vars[1::3]))
                best_radii = best_vars[2::3]

    # Final validation check (internal)
    # We should return a valid packing. The optimizer constraints should ensure this.
    # However, numerical precision might be an issue.
    # We trust the optimizer with tight tolerances.
    
    if best_centers is None:
        # Fallback to grid if optimization failed
        best_centers = np.zeros((n_circles, 2))
        best_radii = np.zeros(n_circles)
        for i in range(n_circles):
            best_centers[i, 0] = (i % 6 + 0.5) / 6
            best_centers[i, 1] = (i // 6 + 0.5) / 5
            best_radii[i] = 0.04
        best_sum_radii = np.sum(best_radii)

    return best_centers, best_radii, best_sum_radii
