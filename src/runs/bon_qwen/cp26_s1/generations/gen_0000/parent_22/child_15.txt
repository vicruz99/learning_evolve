import numpy as np
from scipy.optimize import minimize

def run_packing():
    """
    Packs 26 circles in a unit square [0,1]x[0,1] to maximize the sum of radii.
    Uses a penalty-based optimization method starting from a hexagonal lattice.
    """
    n_circles = 26
    
    # 1. Initialization: Hexagonal Lattice
    centers = np.zeros((n_circles, 2))
    radii = np.zeros(n_circles)
    
    # Estimate a safe initial radius
    # Area argument: 26 * pi * r^2 ~ 0.8 => r ~ 0.1
    # Hex packing allows slightly higher density, so 0.08 is a safe start
    initial_r = 0.08
    
    # Generate hexagonal grid coordinates
    # Vertical spacing for hex packing is sqrt(3)/2 * diameter = sqrt(3) * r
    # Horizontal spacing is diameter = 2 * r
    # We'll use a slightly larger spacing to ensure initial validity
    spacing_x = 2 * initial_r * 1.1
    spacing_y = np.sqrt(3) * initial_r * 1.1
    
    idx = 0
    y = initial_r + spacing_y / 2 # Start slightly above bottom
    row_idx = 0
    
    while idx < n_circles:
        # Determine x offset for zigzag
        x_offset = 0.0 if row_idx % 2 == 0 else spacing_x / 2
        
        x = initial_r + x_offset
        while x <= 1 - initial_r and idx < n_circles:
            centers[idx, 0] = x
            centers[idx, 1] = y
            radii[idx] = initial_r
            idx += 1
            x += spacing_x
        
        y += spacing_y
        row_idx += 1
        
    # If we didn't fit enough, extend or adjust (though logic above should fit 26 easily)
    # Fallback random fill if needed, but hex grid is robust.
    if idx < n_circles:
        for i in range(idx, n_circles):
            centers[i] = [0.5, 0.5]
            radii[i] = 0.01

    # 2. Optimization Loop with Homotopy (increasing penalty weights)
    # We optimize the flattened array of [x1, y1, r1, x2, y2, r2, ...]
    # Or better, keep structure in objective
    
    # Initial state vector
    # Shape (n_circles, 3): [x, y, r]
    state = np.zeros((n_circles, 3))
    state[:, 0] = centers[:, 0]
    state[:, 1] = centers[:, 1]
    state[:, 2] = radii
    
    def objective(state_flat, lam_overlap, lam_bound):
        state = state_flat.reshape(n_circles, 3)
        c = state[:, :2] # centers (N, 2)
        r = state[:, 2]  # radii (N,)
        
        # Objective: Maximize sum of radii => Minimize -sum(r)
        val = -np.sum(r)
        
        # --- Overlap Penalty ---
        # Vectorized pairwise distance
        # diff[i, j] = c[i] - c[j]
        diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
        dist_sq = np.sum(diff**2, axis=2)
        dist = np.sqrt(np.maximum(dist_sq, 0))
        
        # r_i + r_j - dist_ij
        r_sum = r[:, np.newaxis] + r[np.newaxis, :]
        overlap = np.maximum(0, r_sum - dist)
        
        # Sum of squared overlaps for upper triangle
        triu_indices = np.triu_indices(n_circles, k=1)
        penalty_overlap = lam_overlap * np.sum(overlap[triu_indices]**2)
        val += penalty_overlap
        
        # --- Boundary Penalty ---
        # Constraints: 
        # r <= x <= 1-r  => x-r >= 0, 1-x-r >= 0
        # r <= y <= 1-r  => y-r >= 0, 1-y-r >= 0
        
        # Violations (positive means violation)
        v1 = np.maximum(0, r - c[:, 0])       # r > x
        v2 = np.maximum(0, r - (1 - c[:, 0])) # r > 1-x
        v3 = np.maximum(0, r - c[:, 1])       # r > y
        v4 = np.maximum(0, r - (1 - c[:, 1])) # r > 1-y
        
        penalty_bound = lam_bound * np.sum([v1**2, v2**2, v3**2, v4**2])
        val += penalty_bound
        
        return val

    # Bounds for variables
    # x, y in [0, 1], r in [0, 0.5]
    bounds = []
    for _ in range(n_circles):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 0.5)) # r

    current_state = state.flatten()
    
    # Homotopy parameters
    # Start with moderate penalties and increase them
    # This helps the optimizer escape local minima with overlaps and find valid high-radii configs
    lambdas_overlap = [100, 1000, 10000, 50000, 100000]
    lambdas_bound = [1000, 10000, 100000, 500000, 1000000]

    for lo, lb in zip(lambdas_overlap, lambdas_bound):
        # Minimize
        res = minimize(
            objective,
            current_state,
            args=(lo, lb),
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 500, 'ftol': 1e-9, 'gtol': 1e-6}
        )
        current_state = res.x

    # 3. Extract and Validate/Correct Final Solution
    final_state = current_state.reshape(n_circles, 3)
    final_centers = final_state[:, :2]
    final_radii = final_state[:, 2]
    
    # Clamp tiny negative radii to 0 (numerical noise)
    final_radii = np.maximum(final_radii, 0.0)
    
    # Ensure centers are within [0,1] (clamping just in case)
    final_centers = np.clip(final_centers, 0.0, 1.0)
    
    sum_radii = np.sum(final_radii)
    
    return final_centers, final_radii, sum_radii

# Helper to run and print result

    centers, radii, s_r = run_packing()
