import numpy as np
from scipy.optimize import linprog, minimize

def compute_optimal_radii(centers):
    """
    Given fixed centers, compute the maximum possible radii using Linear Programming.
    Solves: maximize sum(r_i) s.t. r_i + r_j <= dist(i,j), 0 <= r_i <= dist_to_boundary(i)
    """
    n = centers.shape[0]
    c = -np.ones(n)  # Minimize negative sum => maximize sum
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= min(x_i, 1-x_i, y_i, 1-y_i)
    for i in range(n):
        x, y = centers[i]
        max_r = min(x, 1.0 - x, y, 1.0 - y)
        if max_r < 0:
            max_r = 0.0
        row = np.zeros(n)
        row[i] = 1.0
        A_ub.append(row)
        b_ub.append(max_r)
        
    # Pairwise non-overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dist)
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0, 1.0)] * n
    
    # Use HiGHS solver for speed and reliability
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return -res.fun, -res.x
    return 0.0, np.zeros(n)

def objective_func(centers_flat):
    """Objective function for center optimization: negative of max sum of radii."""
    centers = centers_flat.reshape(-1, 2)
    # Keep centers strictly inside to avoid LP degeneracy at exact boundaries
    centers = np.clip(centers, 1e-6, 1.0 - 1e-6)
    val, _ = compute_optimal_radii(centers)
    return -val

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # Generate a hexagonal-like initial arrangement
    init_centers = []
    y_levels = np.linspace(0.12, 0.88, 5)
    row_counts = [6, 5, 6, 5, 4]  # Total = 26
    
    for idx, (y, cnt) in enumerate(zip(y_levels, row_counts)):
        x_coords = np.linspace(0.05, 0.95, cnt)
        # Shift odd rows to create hexagonal packing pattern
        if idx % 2 == 1:
            x_coords += (x_coords[1] - x_coords[0]) / 2.0
        for x in x_coords:
            init_centers.append([x, y])
            
    init_centers = np.array(init_centers)
    
    # Add small random perturbation to break symmetries
    np.random.seed(42)
    init_centers += np.random.uniform(-0.01, 0.01, init_centers.shape)
    
    # Optimize centers using Nelder-Mead
    res = minimize(objective_func, init_centers.flatten(), method='Nelder-Mead',
                   options={'maxiter': 20000, 'xatol': 1e-7, 'fatol': 1e-9})
                   
    best_centers = res.x.reshape(-1, 2)
    best_sum, best_radii = compute_optimal_radii(best_centers)
    
    # Ensure radii are strictly non-negative
    best_radii = np.clip(best_radii, 0, None)
    
    return best_centers, best_radii, best_sum
