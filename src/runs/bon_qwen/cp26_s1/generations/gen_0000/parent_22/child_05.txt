import numpy as np
from scipy.optimize import minimize

N = 26
VAR_DIM = 3 * N

IDX_X = np.arange(N)
IDX_Y = np.arange(N, 2*N)
IDX_R = np.arange(2*N, 3*N)

NON_OVERLAP_PAIRS = [(i, j) for i in range(N) for j in range(i+1, N)]
NUM_CONSTRAINTS = 4 * N + len(NON_OVERLAP_PAIRS)

def compute_constraints(v):
    x = v[:N]
    y = v[N:2*N]
    r = v[2*N:3*N]
    
    vals = np.empty(NUM_CONSTRAINTS)
    jacs = np.empty((NUM_CONSTRAINTS, VAR_DIM))
    
    idx = 0
    
    # x - r >= 0
    vals[idx:idx+N] = x - r
    jacs[idx:idx+N, IDX_X] = 1.0
    jacs[idx:idx+N, IDX_R] = -1.0
    idx += N
    
    # 1 - x - r >= 0
    vals[idx:idx+N] = 1.0 - x - r
    jacs[idx:idx+N, IDX_X] = -1.0
    jacs[idx:idx+N, IDX_R] = -1.0
    idx += N
    
    # y - r >= 0
    vals[idx:idx+N] = y - r
    jacs[idx:idx+N, IDX_Y] = 1.0
    jacs[idx:idx+N, IDX_R] = -1.0
    idx += N
    
    # 1 - y - r >= 0
    vals[idx:idx+N] = 1.0 - y - r
    jacs[idx:idx+N, IDX_Y] = -1.0
    jacs[idx:idx+N, IDX_R] = -1.0
    idx += N
    
    # Non-overlap constraints
    for k, (i, j) in enumerate(NON_OVERLAP_PAIRS):
        dx = x[i] - x[j]
        dy = y[i] - y[j]
        dr = r[i] + r[j]
        val = dx*dx + dy*dy - dr*dr
        vals[idx] = val
        
        jac_row = jacs[idx]
        jac_row[i] = 2.0 * dx
        jac_row[j] = -2.0 * dx
        jac_row[N+i] = 2.0 * dy
        jac_row[N+j] = -2.0 * dy
        jac_row[2*N+i] = -2.0 * dr
        jac_row[2*N+j] = -2.0 * dr
        idx += 1
        
    return vals, jacs

def constraint_fun(v):
    return compute_constraints(v)[0]

def constraint_jac(v):
    return compute_constraints(v)[1]

def objective(v):
    return -np.sum(v[2*N:3*N])

def objective_jac(v):
    jac = np.zeros(VAR_DIM)
    jac[2*N:3*N] = -1.0
    return jac

def generate_initial_guess(seed):
    np.random.seed(seed)
    centers = np.zeros((N, 2))
    radii = np.full(N, 0.08)
    idx = 0
    rows_counts = [6, 5, 6, 5, 4]
    dy = 0.18
    dx = 0.15
    for row, count in enumerate(rows_counts):
        y_pos = 0.1 + row * dy
        x_start = 0.1 + (6.0 - count) * dx / 2.0
        for c in range(count):
            centers[idx, 0] = x_start + c * dx
            centers[idx, 1] = y_pos
            idx += 1
            
    centers += np.random.randn(N, 2) * 0.01
    v0 = np.concatenate([centers[:, 0], centers[:, 1], radii])
    return v0

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    bounds = [(0.0, 1.0)] * N + [(0.0, 1.0)] * N + [(0.0, 0.5)] * N
    constraint = {'type': 'ineq', 'fun': constraint_fun, 'jac': constraint_jac}
    
    best_v = None
    best_sum = -1.0
    
    for trial in range(5):
        v0 = generate_initial_guess(trial)
        res = minimize(objective, v0, method='SLSQP', jac=objective_jac,
                       bounds=bounds, constraints=constraint,
                       options={'maxiter': 1500, 'ftol': 1e-12, 'disp': False})
        
        if res.success:
            s = -res.fun
            if s > best_sum:
                best_sum = s
                best_v = res.x.copy()
                
    if best_v is None:
        best_v = generate_initial_guess(0)
        
    centers_out = np.zeros((N, 2))
    centers_out[:, 0] = best_v[:N]
    centers_out[:, 1] = best_v[N:2*N]
    radii_out = best_v[2*N:3*N]
    
    radii_out = np.maximum(radii_out, 0.0)
    
    return centers_out, radii_out, float(np.sum(radii_out))
