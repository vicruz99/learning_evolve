import numpy as np
from scipy.optimize import minimize

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    n = centers.shape[0]
    if np.isnan(centers).any() or np.isnan(radii).any():
        return False
    for i in range(n):
        if radii[i] < 0: return False
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            return False
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False
    return True

def get_initial_positions(n=26):
    """Generate a hexagonal grid initialization."""
    r_init = 0.1
    dx = 2 * r_init
    dy = np.sqrt(3) * r_init
    points = []
    y = r_init
    while y + r_init <= 1.0:
        x = r_init
        offset = 0
        if len(points) % 2 != 0:
            offset = r_init
        x += offset
        while x + r_init <= 1.0:
            points.append([x, y])
            x += dx
        y += dy
        if len(points) >= n:
            break
    if len(points) < n:
        for i in range(n - len(points)):
            points.append([0.5, 0.5])
    return np.array(points[:n])

def optimize_packing(n=26):
    centers = get_initial_positions(n)
    radii = np.ones(n) * 0.09
    
    def objective(vars):
        c = vars[:2*n].reshape((n, 2))
        r = vars[2*n:]
        
        # 1. Negative sum of radii
        obj = -np.sum(r)
        
        # 2. Boundary penalties
        penalty = 0.0
        for i in range(n):
            x, y = c[i]
            ri = r[i]
            # r <= x, r <= 1-x, r <= y, r <= 1-y
            violation = max(0, ri - x) + max(0, ri - (1-x)) + max(0, ri - y) + max(0, ri - (1-y))
            penalty += violation ** 2
            
        # 3. Inter-circle distance penalties
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((c[i] - c[j]) ** 2))
                min_dist = r[i] + r[j]
                if dist < min_dist:
                    penalty += (min_dist - dist) ** 2
                    
        return obj + 1000.0 * penalty

    def objective_jac(vars):
        c = vars[:2*n].reshape((n, 2))
        r = vars[2*n:]
        
        grad = np.zeros_like(vars)
        
        # Gradient of -sum(r) w.r.t r is -1
        grad[2*n:] = -1.0
        
        # Gradients for boundary penalties
        for i in range(n):
            x, y = c[i]
            ri = r[i]
            # d/dx (max(0, ri - x)^2) = -2(ri-x) if ri>x
            # d/dr (max(0, ri - x)^2) = 2(ri-x) if ri>x
            for val, idx in [(x, 0), (1-x, 1)]:
                if ri > val:
                    diff = ri - val
                    grad[2*i + idx] -= 1000.0 * 2 * diff
                    grad[2*n + i] += 1000.0 * 2 * diff
            for val, idx in [(y, 1), (1-y, 0)]:
                if ri > val:
                    diff = ri - val
                    grad[2*i + idx] -= 1000.0 * 2 * diff
                    grad[2*n + i] += 1000.0 * 2 * diff

        # Gradients for inter-circle penalties
        for i in range(n):
            for j in range(i + 1, n):
                diff = c[i] - c[j]
                dist = np.sqrt(np.sum(diff ** 2))
                if dist < 1e-10: dist = 1e-10
                sum_r = r[i] + r[j]
                
                if sum_r > dist:
                    violation = sum_r - dist
                    # Normal vector pointing from j to i
                    n_vec = diff / dist
                    
                    # Gradient w.r.t c[i] and c[j]
                    grad[2*i:2*i+2] -= 1000.0 * 2 * violation * n_vec
                    grad[2*j:2*j+2] += 1000.0 * 2 * violation * n_vec
                    
                    # Gradient w.r.t r[i] and r[j]
                    grad[2*n + i] += 1000.0 * 2 * violation
                    grad[2*n + j] += 1000.0 * 2 * violation

        return grad

    x0 = np.concatenate([centers.flatten(), radii])
    bounds = [(0, 1) for _ in range(2*n)] + [(0, 0.5) for _ in range(n)]

    result = minimize(objective, x0, jac=objective_jac, method='L-BFGS-B', bounds=bounds, 
                      options={'maxiter': 1000, 'ftol': 1e-9})
    
    c_opt = result.x[:2*n].reshape((n, 2))
    r_opt = result.x[2*n:]
    
    # Final cleanup to ensure strict validity (clip tiny violations)
    # In case of numerical drift
    for i in range(n):
        r_opt[i] = min(r_opt[i], c_opt[i, 0], 1-c_opt[i, 0], c_opt[i, 1], 1-c_opt[i, 1])
        
    sum_r = np.sum(r_opt)
    return c_opt, r_opt, sum_r

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    centers, radii, sum_radii = optimize_packing(26)
    
    if not validate_packing(centers, radii):
        # Fallback if optimization fails validation (e.g. numerical drift)
        # Scale down slightly to ensure validity
        radii = radii * 0.99
        sum_radii = np.sum(radii)
        
    return centers, radii, sum_radii
