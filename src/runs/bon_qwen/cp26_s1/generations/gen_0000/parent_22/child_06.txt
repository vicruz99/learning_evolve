import numpy as np

def compute_forces(centers, radii):
    n = centers.shape[0]
    fx = np.zeros(n)
    fy = np.zeros(n)
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.linalg.norm(diff, axis=2)
    np.fill_diagonal(dists, np.inf)
    
    min_dist = radii[:, np.newaxis] + radii[np.newaxis, :]
    overlap = min_dist - dists
    
    mask = overlap > 0
    if not np.any(mask):
        return fx, fy, 0.0
        
    safe_dist = np.where(dists > 1e-8, dists, 1e-8)
    force_mag = np.where(mask, overlap / safe_dist, 0.0)
    
    dir_x = np.where(dists > 1e-8, diff[:, :, 0], 0.0)
    dir_y = np.where(dists > 1e-8, diff[:, :, 1], 0.0)
    
    fx -= np.sum(force_mag * dir_x, axis=1)
    fy -= np.sum(force_mag * dir_y, axis=1)
    
    for i in range(n):
        if centers[i, 0] - radii[i] < 0: fx[i] += 1.0
        if centers[i, 0] + radii[i] > 1: fx[i] -= 1.0
        if centers[i, 1] - radii[i] < 0: fy[i] += 1.0
        if centers[i, 1] + radii[i] > 1: fy[i] -= 1.0
        
    return fx, fy, np.max(overlap) if np.any(mask) else 0.0

def run_packing():
    np.random.seed(42)
    n = 26
    centers = np.zeros((n, 2))
    
    # Hexagonal lattice initialization for better starting density
    r_init = 0.08
    idx = 0
    y = r_init
    row = 0
    while y < 1.0 - r_init and idx < n:
        x = r_init
        if row % 2 == 1:
            x += r_init * 0.5
        while x < 1.0 - r_init and idx < n:
            centers[idx] = [x, y]
            idx += 1
            x += r_init
        y += r_init * np.sqrt(3)
        row += 1
        
    # Fill remaining positions if lattice didn't cover exactly 26
    if idx < n:
        for _ in range(n - idx):
            centers[idx] = np.random.uniform(r_init, 1-r_init, 2)
            idx += 1
            
    radii = np.ones(n) * r_init
    
    step = 0.15
    grow_step = 0.0005
    max_outer = 8000
    
    for outer in range(max_outer):
        # Inner loop: repel overlapping circles and push to boundaries
        for _ in range(400):
            fx, fy, max_ov = compute_forces(centers, radii)
            if max_ov < 1e-8:
                break
            centers[:, 0] += step * fx
            centers[:, 1] += step * fy
            centers[:, 0] = np.clip(centers[:, 0], 1e-5, 1-1e-5)
            centers[:, 1] = np.clip(centers[:, 1], 1e-5, 1-1e-5)
            
        # Calculate available space for uniform radius growth
        diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
        dists = np.linalg.norm(diff, axis=2)
        np.fill_diagonal(dists, np.inf)
        min_dist = radii[:, np.newaxis] + radii[np.newaxis, :]
        gaps = dists - min_dist
        
        gaps_bound = np.minimum(np.minimum(centers[:, 0], 1-centers[:, 0]),
                                np.minimum(centers[:, 1], 1-centers[:, 1])) - radii
        min_gap = min(np.min(gaps), np.min(gaps_bound))
        
        if min_gap > 1e-6:
            # Grow radii uniformly, capped by a fraction of the minimum gap
            radii += min(grow_step, min_gap * 0.4)
        else:
            # Reduce step size to settle into tight configuration
            step *= 0.95
            if step < 0.001:
                step = 0.001
                
    # Final boundary clamping to ensure strict validity
    centers = np.clip(centers, 1e-5, 1-1e-5)
    
    return centers, radii, np.sum(radii)
