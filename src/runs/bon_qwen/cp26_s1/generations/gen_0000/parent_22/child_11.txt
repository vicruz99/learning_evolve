import numpy as np
from scipy.optimize import minimize


def init_hexagonal(n, base_r=0.08):
    """Initialize circles in hexagonal packing arrangement."""
    centers = np.zeros((n, 2))
    radii = np.full(n, base_r)
    
    row_counts = [6, 5, 6, 5, 4]
    idx = 0
    r = base_r
    h_spacing = 2 * r * 1.05
    v_spacing = h_spacing * np.sqrt(3) / 2
    
    for row_idx, count in enumerate(row_counts):
        y = r + row_idx * v_spacing
        offset = h_spacing / 2 if row_idx % 2 == 1 else 0
        for col_idx in range(count):
            x = r + col_idx * h_spacing + offset
            centers[idx] = [x, y]
            idx += 1
    
    return centers, radii


def init_grid(n, base_r=0.08):
    """Initialize circles in grid arrangement."""
    centers = np.zeros((n, 2))
    radii = np.full(n, base_r)
    
    # 5x5 grid = 25, plus 1 extra
    idx = 0
    for row in range(5):
        for col in range(5):
            if idx < n:
                centers[idx] = [0.1 + col * 0.2, 0.1 + row * 0.2]
                idx += 1
    
    # Add extra circle in center if needed
    if n == 26:
        centers[25] = [0.5, 0.5]
        radii[25] = base_r * 0.5
    
    return centers, radii


def compute_objective_and_penalties(x, n, penalty):
    """Compute objective and penalties."""
    centers = x[:2 * n].reshape((n, 2))
    radii = x[2 * n:]
    
    obj = -np.sum(radii)
    penalty_term = 0.0
    
    # Boundary penalties
    for i in range(n):
        dx1 = min(0, centers[i][0] - radii[i])
        dx2 = min(0, 1.0 - centers[i][0] - radii[i])
        dy1 = min(0, centers[i][1] - radii[i])
        dy2 = min(0, 1.0 - centers[i][1] - radii[i])
        penalty_term += dx1 ** 2 + dx2 ** 2 + dy1 ** 2 + dy2 ** 2
    
    # Overlap penalties
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i][0] - centers[j][0]
            dy = centers[i][1] - centers[j][1]
            dist = np.sqrt(dx * dx + dy * dy)
            overlap = radii[i] + radii[j] - dist
            if overlap > 0:
                penalty_term += overlap ** 2
    
    return obj + penalty * penalty_term, penalty_term


def run_packing():
    """Pack 26 circles in unit square to maximize sum of radii."""
    n = 26
    
    best_centers = None
    best_radii = None
    best_sum = 0.0
    
    # Set seed for reproducibility
    rng = np.random.RandomState(42)
    
    # Try multiple initializations
    for restart in range(15):
        # Choose initialization method
        if restart < 5:
            base_r = 0.08 + rng.random() * 0.03
            centers, radii = init_hexagonal(n, base_r)
        elif restart < 10:
            base_r = 0.08 + rng.random() * 0.03
            centers, radii = init_grid(n, base_r)
        else:
            # Random initialization with some structure
            centers = np.zeros((n, 2))
            radii = np.full(n, 0.07)
            for i in range(n):
                centers[i][0] = rng.uniform(0.1, 0.9)
                centers[i][1] = rng.uniform(0.1, 0.9)
                radii[i] = 0.05 + rng.random() * 0.05
        
        # Add small perturbation
        centers += rng.randn(n, 2) * 0.005
        radii += rng.randn(n) * 0.003
        radii = np.clip(radii, 0.01, 0.3)
        
        x0 = np.concatenate([centers.flatten(), radii])
        
        # Bounds for optimization
        bounds = [(0.0, 1.0)] * (2 * n) + [(0.001, 0.5)] * n
        
        # Continuation: gradually increase penalty
        x = x0.copy()
        penalties = [50, 200, 1000, 5000, 20000]
        
        for p in penalties:
            x = minimize(
                compute_objective_and_penalties,
                x,
                args=(n, p),
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 2000, 'ftol': 1e-15, 'gtol': 1e-10}
            )
        
        centers_opt = x.x[:2 * n].reshape((n, 2))
        radii_opt = x.x[2 * n:]
        
        # Clip radii to be non-negative
        radii_opt = np.maximum(radii_opt, 0.0001)
        
        # Verify and compute sum
        valid = True
        for i in range(n):
            if centers_opt[i][0] < 0 or centers_opt[i][0] > 1:
                valid = False
                break

                valid = False
                break

                valid = False
                break

            if centers_opt[i][0] - radii_opt[i] < -1e-10:
                valid = False
                break
            if centers_opt[i][0] + radii_opt[i] > 1 + 1e-10:
                valid = False
                break
            if centers_opt[i][1] - radii_opt[i] < -1e-10:

            )
        
        best_centers = x0.x[:2 * n].reshape((n, 2))
        best_radii = x0.x[2 * n:]

        best_sum = np.sum(best_radii)
    
    return best_centers, best_radii, best_sum
