import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    
    Returns:
        centers: np.array of shape (26, 2)
        radii: np.array of shape (26,)
        sum_radii: float
    """
    n = 26
    
    # Helper to create initial hexagonal packing configuration
    def get_initial_config(seed=0):
        np.random.seed(seed)
        # Start with a small radius that definitely fits in a dense grid
        r_init = 0.06
        
        centers = []
        # Hexagonal packing generation
        # Row height is sqrt(3)/2 * diameter = sqrt(3) * r
        # Row width spacing is 2 * r
        # Shift alternate rows by r
        
        r = r_init
        dy = np.sqrt(3) * r
        dx = 2 * r
        
        row = 0
        while len(centers) < n:
            y = r + row * dy
            if y + r > 1.0:
                # If row doesn't fit vertically, try to squeeze or break?
                # For initialization, just stop if y is too large, 
                # but we need 26 points. 
                # With r=0.06, dy ~ 0.104. 
                # Max rows ~ (1 - 0.12)/0.104 + 1 ~ 9. 
                # Plenty of space.
                pass
            
            # x positions for this row
            # Even rows start at r, odd rows start at r + r = 2r (shifted by r? No, shift is r to align in valleys)
            # Actually, standard hex: row 0 at x=r, x=3r, x=5r...
            # row 1 at x=2r, x=4r... (shifted by r)
            start_x = r + (row % 2) * r 
            
            col = 0
            while True:
                x = start_x + col * dx
                if x + r > 1.0:
                    break
                
                centers.append([x, y])
                col += 1
                if len(centers) >= n:
                    break
            row += 1
            
        # Take exactly n centers
        centers = centers[:n]
        
        # Add some random noise to break symmetry
        noise = np.random.uniform(-0.005, 0.005, size=(n, 2))
        centers = np.array(centers) + noise
        
        # Ensure inside bounds [r, 1-r]
        centers = np.clip(centers, r, 1 - r)
        
        return centers, np.full(n, r)

    # Objective function: minimize -sum(radii)
    def objective(vars):
        # vars: [x1, y1, x2, y2, ..., r1, r2, ...]
        # Actually, let's structure vars as [x_coords (2n), radii (n)]
        # Or [x1, y1, r1, x2, y2, r2...] for easier indexing?
        # Let's use: x_flat (2n), radii_flat (n)
        # Total size 3n
        
        xs = vars[:2*n]
        rs = vars[2*n:]
        
        return -np.sum(rs)

    # Constraints
    def boundary_constraints(vars):
        xs = vars[:2*n]
        rs = vars[2*n:]
        
        constraints = []
        for i in range(n):
            x = xs[2*i]
            y = xs[2*i+1]
            r = rs[i]
            
            # x - r >= 0  =>  x - r >= 0
            constraints.append(x - r)
            # 1 - x - r >= 0
            constraints.append(1 - x - r)
            # y - r >= 0
            constraints.append(y - r)
            # 1 - y - r >= 0
            constraints.append(1 - y - r)
            
        return np.array(constraints)

    def overlap_constraints(vars):
        xs = vars[:2*n]
        rs = vars[2*n:]
        
        constraints = []
        for i in range(n):
            for j in range(i + 1, n):
                x1, y1 = xs[2*i], xs[2*i+1]
                x2, y2 = xs[2*j], xs[2*j+1]
                r1, r2 = rs[i], rs[j]
                
                dist_sq = (x1 - x2)**2 + (y1 - y2)**2
                min_dist_sq = (r1 + r2)**2
                
                # dist_sq >= min_dist_sq  =>  dist_sq - min_dist_sq >= 0
                constraints.append(dist_sq - min_dist_sq)
                
        return np.array(constraints)

    # Run optimization
    best_sum = -np.inf
    best_centers = None
    best_radii = None
    
    # Try multiple random seeds for robustness
    num_trials = 5
    
    for seed in range(num_trials):
        # 1. Initialize
        c_init, r_init = get_initial_config(seed)
        
        # Flatten variables: [x1, y1, x2, y2, ..., r1, r2, ...]
        x_flat = c_init.flatten()
        r_flat = r_init
        x0 = np.concatenate([x_flat, r_flat])
        
        # 2. Define constraints for scipy
        # Boundary constraints (linear)
        # Overlap constraints (nonlinear)
        
        # We can use 'ineq' constraint type: fun(x) >= 0
        
        cons = []
        
        # Boundary
        cons.append({'type': 'ineq', 'fun': boundary_constraints})
        
        # Overlap
        cons.append({'type': 'ineq', 'fun': overlap_constraints})
        
        # 3. Bounds
        # x, y in [0, 1], r in [0, 0.5]
        bounds = []
        for _ in range(2*n):
            bounds.append((0.0, 1.0)) # coords
        for _ in range(n):
            bounds.append((1e-6, 0.5)) # radii must be positive
            
        # 4. Optimize
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, 
                           options={'maxiter': 1000, 'ftol': 1e-9})
            
            if res.success:
                current_sum = -res.fun
                if current_sum > best_sum:
                    best_sum = current_sum
                    best_centers = res.x[:2*n].reshape((n, 2))
                    best_radii = res.x[2*n:]
        except Exception as e:
            print(f"Optimization failed for seed {seed}: {e}")
            continue

    # Fallback if optimization fails (should not happen with valid init)
    if best_centers is None:
        c_init, r_init = get_initial_config(0)
        best_centers = c_init
        best_radii = r_init
        best_sum = np.sum(best_radii)

    return best_centers, best_radii, best_sum
