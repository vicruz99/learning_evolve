import numpy as np
from scipy.optimize import minimize

def compute_overlap_energy(centers, R):
    """Compute squared penalty for overlapping circles."""
    n = centers.shape[0]
    energy = 0.0
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx*dx + dy*dy)
            if dist < 2.0 * R:
                diff = 2.0 * R - dist
                energy += diff * diff
    return energy

def get_hex_init(n, R):
    """Generate a hexagonal lattice initialization scaled to fit in [R, 1-R]."""
    pts = []
    s = 2.0 * R
    for row in range(15):
        for col in range(15):
            x = col * s + (0.5 * s if row % 2 == 1 else 0.0)
            y = row * s * np.sqrt(3.0) / 2.0
            pts.append([x, y])
    pts = np.array(pts)
    if pts.shape[0] == 0:
        return np.random.uniform(R, 1.0 - R, (n, 2))
    
    pts -= pts.min(axis=0)
    scale = pts.max(axis=0)
    valid_scale = scale > 1e-9
    pts[:, valid_scale] /= scale[valid_scale]
    pts = pts * (1.0 - 2.0 * R) + R
    return pts[:n]

def run_packing():
    np.random.seed(42)
    n = 26
    
    R = 0.06
    step = 0.001
    current_centers = get_hex_init(n, R)
    
    def obj(x, R_val):
        c = x.reshape(n, 2)
        return compute_overlap_energy(c, R_val)
        
    # Iteratively increase radius until packing becomes infeasible
    for _ in range(200):
        R += step
        bounds = [(R, 1.0 - R)] * (2 * n)
        x0 = current_centers.ravel()
        
        res = minimize(obj, x0, args=(R,), method='L-BFGS-B', 
                       bounds=bounds, options={'maxiter': 500, 'ftol': 1e-13})
        
        if res.fun < 1e-10:
            current_centers = res.x.reshape(n, 2)
        else:
            # Fallback to random restarts if warm start gets stuck
            found = False
            for _ in range(2):
                rand_c = np.random.uniform(R, 1.0 - R, (n, 2))
                res2 = minimize(obj, rand_c.ravel(), args=(R,), method='L-BFGS-B', 
                                bounds=bounds, options={'maxiter': 500})
                if res2.fun < 1e-10:
                    current_centers = res2.x.reshape(n, 2)
                    found = True
                    break
            if not found:
                R -= step
                if step < 1e-7:
                    break
                step /= 2.0
                
    # Final polish at the largest feasible R
    bounds = [(R, 1.0 - R)] * (2 * n)
    res_final = minimize(obj, current_centers.ravel(), args=(R,), method='L-BFGS-B', 
                         bounds=bounds, options={'maxiter': 1000})
    if res_final.fun < 1e-10:
        current_centers = res_final.x.reshape(n, 2)
        
    radii = np.full(n, R)
    sum_radii = float(np.sum(radii))
    
    return current_centers, radii, sum_radii
