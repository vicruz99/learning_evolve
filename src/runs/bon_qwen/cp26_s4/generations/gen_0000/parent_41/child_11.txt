def run_packing():

    N = 26

    centers = np.zeros((N, 2))
    idx = 0
    for y in np.linspace(0.15, 0.85, 5):
        for x in np.linspace(0.15, 0.85, 5):
            if idx < N:
                centers[idx] = [x, y]
                idx += 1
    # Add slight random perturbation to break symmetry
    centers += np.random.uniform(-0.02, 0.02, (N, 2))
    centers = np.clip(centers, 0.05, 0.95)
    
    radii = np.full(N, 0.03)
    
    # Optimization parameters
    lr = 0.6
    grow_rate = 1.0005
    shrink_rate = 0.9993
    noise_scale = 0.04
    threshold = 1e-9  # Strict threshold for valid packing
    
    best_sum = np.sum(radii)
    best_centers = centers.copy()
    best_radii = radii.copy()
    
    for step in range(60000):
        forces = np.zeros_like(centers)
        radii_penalties = np.zeros(N)
        total_penalty = 0.0
        
        # Pairwise repulsion forces
        for i in range(N):
            for j in range(i + 1, N):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist_sq = dx*dx + dy*dy
                dist = np.sqrt(dist_sq)
                if dist < 1e-8:
                    dist = 1e-8
                    dx, dy = 0.001, 0.001  # Avoid division by zero
                req = radii[i] + radii[j]
                if dist < req:
                    overlap = req - dist
                    # Direction vector normalized by distance
                    fx = dx / dist * overlap
                    fy = dy / dist * overlap
                    forces[i, 0] += fx
                    forces[i, 1] += fy
                    forces[j, 0] -= fx
                    forces[j, 1] -= fy
                    p = overlap ** 2
                    radii_penalties[i] += p
                    radii_penalties[j] += p
                    total_penalty += p

        # Boundary repulsion forces
        for i in range(N):
            x, y = centers[i]
            r = radii[i]
            if x < r:
                forces[i, 0] += (r - x)
                p = (r - x)**2
                radii_penalties[i] += p
                total_penalty += p

                forces[i, 0] -= (x - (1.0 - r))
                p = (x - (1.0 - r))**2
                radii_penalties[i] += p
                total_penalty += p
            if y < r:
                forces[i, 1] += (r - y)
                p = (r - y)**2
                radii_penalties[i] += p
                total_penalty += p
            if y > 1.0 - r:
                forces[i, 1] -= (y - (1.0 - r))
                p = (y - (1.0 - r))**2
                radii_penalties[i] += p
                total_penalty += p

        centers += lr * forces
        centers = np.clip(centers, 0.0, 1.0)

        noise = noise_scale * np.random.randn(N, 2)
        centers += noise
        centers = np.clip(centers, 0.0, 1.0)

        for i in range(N):
            if radii_penalties[i] < threshold:
                radii[i] *= grow_rate
            else:
                radii[i] *= shrink_rate
        radii = np.clip(radii, 1e-5, 0.5)

        noise_scale *= 0.9998
        lr *= 0.9997
        
        current_sum = np.sum(radii)
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers.copy()
            best_radii = radii.copy()
            
    return best_centers, best_radii, best_sum
