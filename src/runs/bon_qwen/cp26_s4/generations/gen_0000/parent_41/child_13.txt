def solve_lp_radii(centers):
    """
    Solve the Linear Programming problem to maximize sum of radii 
    for a fixed set of centers.
    
    Maximize sum(r)
    Subject to:
      r_i + r_j <= dist(c_i, c_j) for all i < j
      r_i <= min(x_i, 1-x_i, y_i, 1-y_i) for all i
      r_i >= 0
    """
    n = centers.shape[0]
    
    # Objective: minimize -sum(r)
    c_obj = -np.ones(n)
    
    # Inequality constraints: A_ub @ r <= b_ub
    # 1. Pairwise distance constraints: r_i + r_j <= d_ij
    # 2. Boundary constraints: r_i <= dist_to_boundary
    
    # We will build the A_ub matrix and b_ub vector
    # Number of constraints: n(n-1)/2 (pairs) + 4n (boundaries)
    num_pairs = n * (n - 1) // 2
    num_boundaries = 4 * n
    total_constraints = num_pairs + num_boundaries
    
    A_ub = np.zeros((total_constraints, n))
    b_ub = np.zeros(total_constraints)
    
    row_idx = 0
    
    # Pairwise constraints
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            A_ub[row_idx, i] = 1.0
            A_ub[row_idx, j] = 1.0
            b_ub[row_idx] = dist
            row_idx += 1
            
    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        # r_i <= x
        A_ub[row_idx, i] = 1.0
        b_ub[row_idx] = x
        row_idx += 1
        # r_i <= 1 - x  => r_i + x <= 1
        A_ub[row_idx, i] = 1.0
        b_ub[row_idx] = 1.0 - x
        row_idx += 1
        # r_i <= y
        A_ub[row_idx, i] = 1.0
        b_ub[row_idx] = y
        row_idx += 1
        # r_i <= 1 - y
        A_ub[row_idx, i] = 1.0
        b_ub[row_idx] = 1.0 - y
        row_idx += 1
        
    # Bounds for r: r_i >= 0
    bounds = [(0, None) for _ in range(n)]
    
    try:
        res = opt.linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
        else:
            # Fallback if LP fails, though rare with valid centers
            return np.zeros(n)
    except Exception:
        return np.zeros(n)

    """Helper to get sum of radii for a given set of centers."""
    radii = solve_lp_radii(centers)
