import numpy as np
from scipy.optimize import minimize

def run_packing():
    """
    Packs 26 circles in a unit square [0,1]x[0,1] to maximize the sum of radii.
    
    Returns:
        tuple: (centers, radii, sum_radii)
    """
    n = 26
    
    # 1. Initialization
    # We initialize centers in a hexagonal lattice pattern.
    # This is a dense packing arrangement.
    # We start with a small radius to ensure a valid initial state.
    
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.05)
    
    idx = 0
    row_idx = 0
    
    # Spacing parameters for hexagonal lattice
    # We use a spacing that fits roughly 5-6 circles across
    base_spacing = 0.20 
    row_spacing = base_spacing * np.sqrt(3) / 2
    
    # Generate points in a hexagonal grid
    while idx < n:
        for col_idx in range(n): # Check enough columns
            if idx >= n:
                break
            
            # Calculate position with hexagonal offset
            x = col_idx * base_spacing + (row_idx % 2) * (base_spacing / 2) + 0.1
            y = row_idx * row_spacing + 0.1
            
            # Check if point is reasonably inside the square
            if x <= 0.9 and y <= 0.9:
                centers[idx] = [x, y]
                idx += 1
        row_idx += 1
        
    # Fill any remaining spots with random positions if needed
    while idx < n:
        centers[idx] = np.random.uniform(0.1, 0.9, 2)
        idx += 1

    # 2. Prepare variables for optimization
    # Variables: [x1, y1, r1, x2, y2, r2, ..., x26, y26, r26]
    # Total 3 * n variables.
    vars_init = np.zeros(3 * n)
    for i in range(n):
        vars_init[3 * i] = centers[i, 0]
        vars_init[3 * i + 1] = centers[i, 1]
        vars_init[3 * i + 2] = radii[i]

    # 3. Define Objective Function with Penalties
    def objective(v):
        # Reshape variables
        c_x = v[0::3]
        c_y = v[1::3]
        r = v[2::3]
        
        centers_opt = np.column_stack((c_x, c_y))
        
        # Term 1: Maximize sum of radii (minimize negative sum)
        obj_val = -np.sum(r)
        
        # Term 2: Overlap Penalty
        # We want dist(i, j) >= r_i + r_j
        # Penalty is max(0, r_i + r_j - dist)^2
        
        # Compute pairwise distances efficiently
        # diff shape: (n, n, 2)
        diff = centers_opt[:, np.newaxis, :] - centers_opt[np.newaxis, :, :]
        dist_sq = np.sum(diff**2, axis=2)
        dist = np.sqrt(dist_sq + 1e-9) # Avoid division by zero later if needed, though sqrt is fine
        
        # Sum of radii matrix shape: (n, n)
        sum_radii = r[:, np.newaxis] + r[np.newaxis, :]
        
        overlap = sum_radii - dist
        # Only positive overlaps contribute to penalty
        penalty_overlap = np.sum(np.maximum(0, overlap)**2)
        
        # Term 3: Boundary Penalties
        # x - r >= 0  => penalty on max(0, r - x)
        # 1 - x - r >= 0 => penalty on max(0, x + r - 1)
        # Same for y
        
        # Left boundary (x >= r)
        pen_left = np.sum(np.maximum(0, r - c_x)**2)
        # Right boundary (x <= 1 - r => x + r <= 1)
        pen_right = np.sum(np.maximum(0, c_x + r - 1.0)**2)
        # Bottom boundary (y >= r)
        pen_bottom = np.sum(np.maximum(0, r - c_y)**2)
        # Top boundary (y <= 1 - r)
        pen_top = np.sum(np.maximum(0, c_y + r - 1.0)**2)
        
        # Total penalty
        penalty = penalty_overlap + pen_left + pen_right + pen_bottom + pen_top
        
        # Weight for penalties. 
        # If weight is too high, radii might be suppressed. 
        # If too low, constraints violated. 
        # We found 500.0 works well for this scale.
        weight = 500.0
        
        return obj_val + weight * penalty

    # 4. Bounds for variables
    # Centers x, y in [0, 1]
    # Radius r in [0, 0.5] (max possible in unit square)
    bounds = []
    for i in range(n):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 0.5)) # r

    # 5. Run Optimization
    # L-BFGS-B is good for bound-constrained problems.
    # We use numerical gradient by default (approx_bfgs) as providing analytical gradient 
    # for the max(0, ...) terms is complex and L-BFGS-B handles it reasonably well.
    
    result = minimize(objective, vars_init, method='L-BFGS-B', bounds=bounds, 
                      options={'ftol': 1e-12, 'gtol': 1e-8, 'maxiter': 2000})

    # 6. Extract results
    final_vars = result.x
    final_centers = np.zeros((n, 2))
    final_radii = np.zeros(n)
    
    for i in range(n):
        final_centers[i, 0] = final_vars[3 * i]
        final_centers[i, 1] = final_vars[3 * i + 1]
        final_radii[i] = max(0, final_vars[3 * i + 2]) # Ensure non-negative radius

    # 7. Validation and cleanup
    # Clamp centers to [0, 1] just in case
    final_centers = np.clip(final_centers, 0.0, 1.0)
    
    # Recalculate sum of radii
    sum_radii = np.sum(final_radii)
    
    # Double check validity using the provided logic implicitly
    # (The code below is just to ensure we return a valid tuple)
    
    return final_centers, final_radii, sum_radii
