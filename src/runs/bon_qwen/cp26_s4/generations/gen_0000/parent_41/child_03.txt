import numpy as np
from scipy.optimize import linprog, differential_evolution
import math

def get_optimal_radii_and_sum(centers):
    """
    Given centers, solve the LP to find radii that maximize sum of radii.
    Returns (sum_radii, radii_array)
    """
    n = centers.shape[0]
    
    # Objective: maximize sum(r_i) => minimize -sum(r_i)
    c = -np.ones(n)
    
    # Constraints:
    # 1. r_i + r_j <= distance(i, j)
    # 2. r_i <= x_i
    # 3. r_i <= 1 - x_i
    # 4. r_i <= y_i
    # 5. r_i <= 1 - y_i
    # r_i >= 0 (handled by bounds)
    
    # Number of constraints
    # Pairwise: n*(n-1)/2
    # Boundary: 4*n
    n_pairwise = n * (n - 1) // 2
    n_boundary = 4 * n
    num_constraints = n_pairwise + n_boundary
    
    A_ub = np.zeros((num_constraints, n))
    b_ub = np.zeros(num_constraints)
    
    row = 0
    
    # Pairwise constraints
    # To speed up, we can use vectorized distance calculation
    # But loops are fine for n=26
    for i in range(n):
        for j in range(i + 1, n):
            # Distance between center i and j
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = math.sqrt(dx*dx + dy*dy)
            
            A_ub[row, i] = 1.0
            A_ub[row, j] = 1.0
            b_ub[row] = dist
            row += 1
            
    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        
        # r_i <= x
        A_ub[row, i] = 1.0
        b_ub[row] = x
        row += 1
        
        # r_i <= 1 - x
        A_ub[row, i] = 1.0
        b_ub[row] = 1.0 - x
        row += 1
        
        # r_i <= y
        A_ub[row, i] = 1.0
        b_ub[row] = y
        row += 1
        
        # r_i <= 1 - y
        A_ub[row, i] = 1.0
        b_ub[row] = 1.0 - y
        row += 1
        
    bounds = [(0, None) for _ in range(n)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            radii = res.x
            return -res.fun, radii
        else:
            # Fallback if LP fails (should not happen for valid centers)
            # Return 0 radii
            return 0.0, np.zeros(n)
    except Exception:
        return 0.0, np.zeros(n)

def objective_func(centers_flat):
    """
    Objective function for optimization: minimize negative sum of radii.
    """
    centers = centers_flat.reshape(-1, 2)
    s, _ = get_optimal_radii_and_sum(centers)
    return -s

    Generate a set of initial centers based on a hexagonal/grid pattern.
    """
    n = 26
    centers = np.zeros((n, 2))
    
    # Try to arrange in a roughly hexagonal pattern
    # 5 rows might fit well.
    # Let's try a grid first as a baseline
    # 5x5 grid has 25 points. We need 26.
    # Maybe 5 rows with varying counts.
    
    # Approximate radius 0.101.
    # Grid spacing ~0.2
    # Let's use a perturbed grid.
    
    # Simple grid placement
    # 5 columns, 6 rows -> 30 points. Select 26?
    # Or just random initialization for DE.
    
    # Let's create a structured initial guess
    # Rows: 0, 1, 2, 3, 4
    # x spacing: 1.0 / 5 = 0.2
    # y spacing: 1.0 / 5 = 0.2
    
    idx = 0
    for r in range(5):
        # 5 points per row roughly
        # shift odd rows
        shift = 0.1 if r % 2 == 1 else 0
        for c in range(5):
            if idx < n:
                cx = (c + 0.5) * 0.2 + shift
                cy = (r + 0.5) * 0.2
                centers[idx] = [cx, cy]
                idx += 1
    
    # Fill remaining
    while idx < n:
        centers[idx] = [np.random.rand(), np.random.rand()]
        idx += 1
        
    return centers.flatten()

def run_packing():
    n = 26
    
    # Define bounds for centers: [0, 1] x [0, 1]
    # Actually, to allow non-zero radius, centers should be strictly inside,
    # but optimizer can explore boundaries.
    # However, if center is at 0, radius must be 0.
    # DE handles bounds well.
    bounds = [(0, 1) for _ in range(2 * n)]
    
    # Use Differential Evolution for global optimization
    # popsize controls the number of trial vectors maintained during optimization.
    # Default is 10 * dim = 520. This might be slow.
    # Let's reduce it or use multiple runs.
    
    # Strategy: Run DE with moderate population and iterations.
    # Or use a few local optimizations from random starts.
    
    best_centers_flat = None
    best_sum = -1.0
    
    # Run 3 times with different seeds or strategies
    for seed in range(3):
        # Initial guess
        if seed == 0:
            x0 = generate_initial_centers()
        else:
            x0 = np.random.uniform(0.1, 0.9, 2*n)
            
        # Optimize
        # Using Nelder-Mead for local refinement from a good start
        # Or BasinHopping?
        # Let's use differential_evolution with reduced popsize
        
        res = differential_evolution(
            objective_func, 
            bounds, 
            seed=seed,
            maxiter=100,

            tol=1e-7,
            polish=True

    # Finalize
    final_centers = best_centers_flat.reshape(-1, 2)
    final_sum, final_radii = get_optimal_radii_and_sum(final_centers)
    
    # Just in case the optimizer returned a slightly invalid state (unlikely with LP)
    # But let's verify and re-optimize radii exactly
    
    # Return
    return final_centers, final_radii, final_sum
