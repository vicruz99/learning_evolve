import scipy.optimize as opt
import math

def run_packing():

    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    
    Returns:
        tuple: (centers, radii, sum_radii)
    """
    n_circles = 26
    
    # Function to generate initial centers based on a hexagonal grid
    def generate_initial_centers(n, seed=None):
        if seed is not None:
            np.random.seed(seed)
        
        centers = []
        # Approximate radius for 26 circles ~ 0.1
        r_est = 0.1
        # Hexagonal spacing
        dy = r_est * math.sqrt(3)
        dx = 2 * r_est
        
        # Try to fill the square
        y = r_est
        while y <= 1 - r_est:
            x = r_est
            # Offset every other row
            offset = 0 if int(y / dy) % 2 == 0 else r_est
            x = r_est + offset
            
            while x <= 1 - r_est:
                centers.append([x, y])
                x += dx
            y += dy
            
        # If we have fewer circles, add some randomly or extend grid
        # If we have more, trim
        if len(centers) < n:
            # Add random centers within bounds
            while len(centers) < n:
                cx = np.random.uniform(r_est, 1 - r_est)
                cy = np.random.uniform(r_est, 1 - r_est)
                centers.append([cx, cy])
        elif len(centers) > n:
            centers = centers[:n]
            
        return np.array(centers)

    # Objective function: Maximize sum of radii -> Minimize negative sum
    def objective(vars):
        # vars contains [x1, y1, r1, x2, y2, r2, ...]
        radii = vars[2::3]
        return -np.sum(radii)

    # Constraints
    # 1. Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    #    Equivalent to: r - x <= 0, x + r - 1 <= 0, r - y <= 0, y + r - 1 <= 0
    def boundary_constraints(vars):
        con = []
        for i in range(n_circles):
            x = vars[3*i]
            y = vars[3*i + 1]
            r = vars[3*i + 2]
            con.append(r - x)       # x >= r
            con.append(x + r - 1)   # x <= 1 - r
            con.append(r - y)       # y >= r
            con.append(y + r - 1)   # y <= 1 - r
        return con

    # 2. Non-overlap constraints: dist^2 >= (r_i + r_j)^2
    #    (r_i + r_j)^2 - dist^2 <= 0
    def overlap_constraints(vars):
        con = []
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                xi, yi, ri = vars[3*i], vars[3*i+1], vars[3*i+2]
                xj, yj, rj = vars[3*j], vars[3*j+1], vars[3*j+2]
                
                dist_sq = (xi - xj)**2 + (yi - yj)**2
                r_sum_sq = (ri + rj)**2
                
                # Constraint: r_sum_sq - dist_sq <= 0
                # To avoid numerical issues with sqrt, keep squared form
                # However, for optimization, it's better to use the form that is differentiable everywhere
                # The squared form is differentiable.
                con.append(r_sum_sq - dist_sq)
        return con

    # Combine constraints for scipy (format: dict with 'type' and 'fun')
    # SLSQP expects constraints as a list of dicts or a single dict with vector function

        c = []
        # Boundary
        for i in range(n_circles):
            x = vars[3*i]
            y = vars[3*i + 1]
            r = vars[3*i + 2]
            c.append(r - x)       
            c.append(x + r - 1)   
            c.append(r - y)       
            c.append(y + r - 1)   
        
        # Overlap

            for j in range(i + 1, n_circles):
                xi, yi, ri = vars[3*i], vars[3*i+1], vars[3*i+2]
                xj, yj, rj = vars[3*j], vars[3*j+1], vars[3*j+2]
                dist_sq = (xi - xj)**2 + (yi - yj)**2
                r_sum_sq = (ri + rj)**2
                c.append(r_sum_sq - dist_sq)
        return np.array(c)

    constraint = {'type': 'ineq', 'fun': lambda vars: -all_constraints(vars)} 
    # SLSQP minimizes f subject to c >= 0.

    # Wait, standard form for 'ineq' in scipy.optimize.minimize is fun(x) >= 0.
    # So we need - (r_sum_sq - dist_sq) >= 0 => dist_sq - r_sum_sq >= 0.
    # And for boundary: x - r >= 0 => -(r - x) >= 0.

            y = vars[3*i + 1]
            r = vars[3*i + 2]
            c.append(x - r)       
            c.append(1 - (x + r)) # 1 - x - r >= 0 => x + r <= 1
            c.append(y - r)       
            c.append(1 - (y + r))

                xi, yi, ri = vars[3*i], vars[3*i+1], vars[3*i+2]

                dist_sq = (xi - xj)**2 + (yi - yj)**2
                r_sum_sq = (ri + rj)**2
                c.append(dist_sq - r_sum_sq)
        return np.array(c)

                x0, 
                method='SLSQP', 
                bounds=bounds,

                options={'maxiter': 1000, 'ftol': 1e-9}
