def get_constraints(centers, radii, n):
    """
    Helper function to calculate constraint violations.
    Returns a list of values that should be >= 0.
    """
    constraints = []
    
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    # Equivalent to: x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
    for i in range(n):

        r = radii[i]
        constraints.append(x - r)
        constraints.append(1 - x - r)
        constraints.append(y - r)
        constraints.append(1 - y - r)
        
    # Non-overlap constraints: dist >= r1 + r2
    # (x1-x2)^2 + (y1-y2)^2 >= (r1+r2)^2
    # We use the squared distance constraint to avoid sqrt, but need to be careful with signs.
    # Actually, standard form for SLSQP is ceq(x)=0 or c(x)>=0.
    # dist^2 - (r1+r2)^2 >= 0
    for i in range(n):
        for j in range(i + 1, n):

            dy = centers[i, 1] - centers[j, 1]
            dist_sq = dx*dx + dy*dy
            r_sum = radii[i] + radii[j]
            constraints.append(dist_sq - r_sum*r_sum)
            
    return np.array(constraints)
