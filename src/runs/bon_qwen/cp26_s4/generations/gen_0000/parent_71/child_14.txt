import numpy as np
import scipy.optimize as opt

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26

    def objective(vars):
        # Variables: [x1, y1, r1, x2, y2, r2, ...]
        # Maximize sum of radii => minimize negative sum of radii
        radii = vars[2::3]
        return -np.sum(radii)

    def gradient(vars):
        # Gradient of -sum(radii) with respect to vars
        grad = np.zeros_like(vars)
        # r variables are at indices 2, 5, 8, ... (every 3rd starting at 2)
        grad[2::3] = -1.0
        return grad

    def get_constraints(vars):
        constraints = []
        
        # 1. Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
        # Equivalent to: x - r >= 0, 1 - x - r >= 0, etc.
        # r - x <= 0, r + x - 1 <= 0
        # Using fun(x) >= 0 format for scipy:
        
        for i in range(n):
            idx = i * 3
            x, y, r = vars[idx], vars[idx+1], vars[idx+2]
            
            # x - r >= 0
            constraints.append({
                'type': 'ineq',
                'fun': lambda v, i=i: v[i*3] - v[i*3+2],
                'jac': lambda v, i=i: np.eye(3*n).flatten()[i*3] - np.eye(3*n).flatten()[i*3+2]
            })
            # 1 - x - r >= 0
            constraints.append({
                'type': 'ineq',
                'fun': lambda v, i=i: 1.0 - v[i*3] - v[i*3+2],
                'jac': lambda v, i=i: -np.eye(3*n).flatten()[i*3] - np.eye(3*n).flatten()[i*3+2]
            })
            # y - r >= 0
            constraints.append({
                'type': 'ineq',
                'fun': lambda v, i=i: v[i*3+1] - v[i*3+2],
                'jac': lambda v, i=i: np.eye(3*n).flatten()[i*3+1] - np.eye(3*n).flatten()[i*3+2]
            })
            # 1 - y - r >= 0
            constraints.append({
                'type': 'ineq',
                'fun': lambda v, i=i: 1.0 - v[i*3+1] - v[i*3+2],
                'jac': lambda v, i=i: -np.eye(3*n).flatten()[i*3+1] - np.eye(3*n).flatten()[i*3+2]
            })
            # r >= 0
            constraints.append({
                'type': 'ineq',
                'fun': lambda v, i=i: v[i*3+2],
                'jac': lambda v, i=i: np.eye(3*n).flatten()[i*3+2]
            })

        # 2. Non-overlap constraints: dist >= r1 + r2
        # (x1-x2)^2 + (y1-y2)^2 - (r1+r2)^2 >= 0
        for i in range(n):
            for j in range(i + 1, n):
                idx_i = i * 3
                idx_j = j * 3
                
                def fun_overlap(v, i=i, j=j):
                    xi, yi, ri = v[i*3], v[i*3+1], v[i*3+2]
                    xj, yj, rj = v[j*3], v[j*3+1], v[j*3+2]
                    dist_sq = (xi - xj)**2 + (yi - yj)**2
                    sum_r_sq = (ri + rj)**2
                    return dist_sq - sum_r_sq
                
                def jac_overlap(v, i=i, j=j):
                    jac = np.zeros(3*n)
                    xi, yi, ri = v[i*3], v[i*3+1], v[i*3+2]
                    xj, yj, rj = v[j*3], v[j*3+1], v[j*3+2]
                    
                    # Derivatives w.r.t x_i, y_i, r_i
                    jac[i*3] = 2 * (xi - xj)
                    jac[i*3+1] = 2 * (yi - yj)
                    jac[i*3+2] = -2 * (ri + rj)
                    
                    # Derivatives w.r.t x_j, y_j, r_j
                    jac[j*3] = -2 * (xi - xj)
                    jac[j*3+1] = -2 * (yi - yj)
                    jac[j*3+2] = -2 * (ri + rj)
                    
                    return jac

                constraints.append({'type': 'ineq', 'fun': fun_overlap, 'jac': jac_overlap})
        
        return constraints

    # Strategy: Hexagonal packing initialization with variable radii
    def get_initial_guess():
        vars = np.zeros(3 * n)
        # 5 rows, staggered
        # Approximate layout to maximize density
        centers = []
        r_init = 0.1
        # Row 0: 5 circles
        for i in range(5):
            centers.append((0.1 + i * 0.2, 0.1))
        # Row 1: 5 circles
        for i in range(5):
            centers.append((0.2 + i * 0.2, 0.2 + 0.1 * np.sqrt(3)))
        # Row 2: 5 circles
        for i in range(5):
            centers.append((0.1 + i * 0.2, 0.2 + 2 * 0.1 * np.sqrt(3)))

        for i in range(5):
            centers.append((0.2 + i * 0.2, 0.2 + 3 * 0.1 * np.sqrt(3)))

            vars[i*3] = cx
            vars[i*3+1] = cy
            vars[i*3+2] = r_init # Initial radius
            
        return vars

    best_vars = None
    best_val = -np.inf

    # Run optimization from structured start
    x0 = get_initial_guess()
    bounds = [(0, 1)] * (2*n) + [(0, 0.5)] * n
    cons = get_constraints(x0) # Precompute constraint structure

    try:
        res = opt.minimize(objective, x0, method='SLSQP', jac=gradient,

                           options={'maxiter': 1000, 'ftol': 1e-9})
        if res.success and -res.fun > best_val:
            best_val = -res.fun
            best_vars = res.x.copy()
    except Exception:
        pass

    # Run from random start to avoid local optima
    rng = np.random.default_rng(42)
    x0_rand = np.zeros(3*n)
    for i in range(n):

        x0_rand[i*3+2] = 0.05 
        
    try:
        res2 = opt.minimize(objective, x0_rand, method='SLSQP', jac=gradient, 
                            bounds=bounds, constraints=cons, 
                            options={'maxiter': 1000, 'ftol': 1e-9})
        if res2.success and -res2.fun > best_val:
            best_val = -res2.fun
            best_vars = res2.x.copy()
    except Exception:
        pass

    if best_vars is None:
        # Fallback
        best_vars = x0
        best_val = -objective(x0)

    centers = np.array([[best_vars[i*3], best_vars[i*3+1]] for i in range(n)])
    radii = np.array([best_vars[i*3+2] for i in range(n)])
    
    return centers, radii, best_val
