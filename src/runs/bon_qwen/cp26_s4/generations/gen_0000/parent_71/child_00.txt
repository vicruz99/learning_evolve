import numpy as np
from scipy.optimize import minimize
import math

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    n = centers.shape[0]
    if np.isnan(centers).any() or np.isnan(radii).any():
        return False
    for i in range(n):
        if radii[i] < 0:
            return False
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            return False
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False
    return True

def get_objective_and_constraints(params, n):
    # Unpack parameters
    centers = params[:2*n].reshape((n, 2))
    radii = params[2*n:]
    
    obj_val = -np.sum(radii)
    
    constraints = []
    
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    for i in range(n):
        constraints.append({'type': 'ineq', 'fun': lambda p, i=i: p[2*n + i] - p[2*i]})       # r - x <= 0 => x >= r
        constraints.append({'type': 'ineq', 'fun': lambda p, i=i: p[2*i + 1] - p[2*n + i]})    # y - r >= 0 => y >= r
        constraints.append({'type': 'ineq', 'fun': lambda p, i=i: 1 - p[2*i] - p[2*n + i]})    # 1 - x - r >= 0 => x <= 1-r
        constraints.append({'type': 'ineq', 'fun': lambda p, i=i: 1 - p[2*i + 1] - p[2*n + i]})# 1 - y - r >= 0 => y <= 1-r
        
        # Non-negative radius
        constraints.append({'type': 'ineq', 'fun': lambda p, i=i: p[2*n + i]})
        
    # Overlap constraints: dist >= r_i + r_j
    for i in range(n):
        for j in range(i + 1, n):
            def overlap_constraint(p, i=i, j=j):
                ci = p[2*i : 2*i+2]
                cj = p[2*j : 2*j+2]
                ri = p[2*n + i]
                rj = p[2*n + j]
                dist = np.linalg.norm(ci - cj)
                return dist - (ri + rj)
            constraints.append({'type': 'ineq', 'fun': overlap_constraint})
            
    return obj_val, constraints

def run_packing():
    n = 26
    best_sum = 0
    best_centers = None
    best_radii = None
    
    # Helper to generate initial guess
    def generate_guess(random_seed):
        np.random.seed(random_seed)
        # Start with 5x5 grid for 25 circles
        centers = np.zeros((n, 2))
        radii = np.zeros(n)
        
        idx = 0
        for r in range(5):
            for c in range(5):
                centers[idx] = [0.1 + c * 0.2, 0.1 + r * 0.2]
                radii[idx] = 0.1
                idx += 1
        
        # Place 26th circle in a gap (e.g., between (0.1,0.1) and (0.3,0.1))
        # Center of gap at (0.2, 0.2), max radius approx 0.041
        centers[25] = [0.2, 0.2]
        radii[25] = 0.04
        
        # Add small random perturbation
        centers += np.random.uniform(-0.02, 0.02, (n, 2))
        radii += np.random.uniform(-0.01, 0.01, n)
        radii = np.maximum(radii, 0.01)
        
        # Clip centers to valid range roughly
        centers = np.clip(centers, 0.01, 0.99)
        
        return np.concatenate([centers.flatten(), radii])

    # Run optimization with multiple seeds
    for seed in range(50):
        initial_params = generate_guess(seed)
        
        # Define objective function for optimizer
        def objective(params):
            obj_val, _ = get_objective_and_constraints(params, n)
            return obj_val
            
        # Constraints are computed inside the solver loop via get_objective_and_constraints
        # But scipy needs a static list of constraints. 
        # We will define them dynamically or use a callback, but static list is better for SLSQP.
        # Since n is fixed, we can generate the list once.
        
        # Actually, generating the list of dicts with closures is tricky. 
        # Let's just use a penalty method or a simpler constraint handling if needed.
        # But SLSQP handles lists of dicts well.
        
        # Re-generate constraints list carefully
        constraints = []
        for i in range(n):
            constraints.append({'type': 'ineq', 'fun': lambda p, i=i: p[2*n + i] - p[2*i]})
            constraints.append({'type': 'ineq', 'fun': lambda p, i=i: p[2*i + 1] - p[2*n + i]})
            constraints.append({'type': 'ineq', 'fun': lambda p, i=i: 1 - p[2*i] - p[2*n + i]})
            constraints.append({'type': 'ineq', 'fun': lambda p, i=i: 1 - p[2*i + 1] - p[2*n + i]})
            constraints.append({'type': 'ineq', 'fun': lambda p, i=i: p[2*n + i]})

            for j in range(i + 1, n):
                # Capture i, j in closure
                def make_constraint(i, j):
                    def con(p):
                        ci = p[2*i : 2*i+2]
                        cj = p[2*j : 2*j+2]
                        ri = p[2*n + i]
                        rj = p[2*n + j]
                        return np.linalg.norm(ci - cj) - (ri + rj)
                    return con
                constraints.append({'type': 'ineq', 'fun': make_constraint(i, j)})

        bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n

            res = minimize(objective, initial_params, method='SLSQP', bounds=bounds, constraints=constraints,

                sum_r = -res.fun
                if sum_r > best_sum:

                    best_centers = res.x[:2*n].reshape((n, 2))
                    best_radii = res.x[2*n:]

    # Final validation and fallback if optimization failed or produced invalid results

    centers_fallback = np.zeros((n, 2))
    radii_fallback = np.zeros(n)
    idx = 0
    for r in range(5):

            centers_fallback[idx] = [0.1 + c * 0.2, 0.1 + r * 0.2]
            radii_fallback[idx] = 0.1
            idx += 1
    centers_fallback[25] = [0.2, 0.2]
    radii_fallback[25] = 0.04
    return centers_fallback, radii_fallback, np.sum(radii_fallback)

if __name__ == "__main__":
    centers, radii, s_r = run_packing()
    print(f"Sum of radii: {s_r}")
    print(f"Valid: {validate_packing(centers, radii)}")
