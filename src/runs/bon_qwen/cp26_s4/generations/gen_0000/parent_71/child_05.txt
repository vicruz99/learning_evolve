def solve_radii_lp(centers):

    Iteratively solves for the maximum radii of circles at fixed centers
    using a relaxation method.
    """
    n = centers.shape[0]
    radii = np.zeros(n)

    for i in range(n):
        x, y = centers[i]
        radii[i] = min(x, 1 - x, y, 1 - y)

        radii = np.where(radii <= 0, 1e-9, radii)

        dist_matrix = np.sqrt(np.sum((centers[:, np.newaxis] - centers[np.newaxis, :]) ** 2, axis=2))
        dist_matrix = np.maximum(dist_matrix, 0)
        np.fill_diagonal(dist_matrix, np.inf) # Self distance is inf

        new_radii = radii.copy()

            # Max allowed radius based on neighbors

                new_radii[i] = r

            if dist < min_dist:
                penalty += (min_dist - dist) ** 2
                
    return penalty
