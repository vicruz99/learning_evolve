    def compute_loss(params, lambdas):

        Compute the loss function: negative sum of radii + penalties.
        params: array of shape (N*3,) [x1, y1, r1, x2, y2, r2, ...]
        """

        overlap_pen, boundary_pen = lambdas

        # Circle i must be inside [0, 1] x [0, 1]
        # x - r >= 0  => violation: max(0, r - x)^2
        # x + r <= 1  => violation: max(0, x + r - 1)^2

        violations_left = np.maximum(0, r - x)
        penalty_boundary += np.sum(violations_left**2)

        penalty_boundary += np.sum(violations_right**2)

        penalty_boundary += np.sum(violations_bottom**2)

        penalty_overlap = 0.0
