import numpy as np
import scipy.optimize as opt

def get_valid_radii(centers):
    """
    Given centers, compute the maximum valid radius for each circle
    such that no overlaps occur and circles stay inside [0,1]x[0,1].
    """
    n = centers.shape[0]
    radii = np.ones(n) * 1.0  # Initialize with large value

    # Constraint 1: Boundaries
    # r <= x, r <= 1-x, r <= y, r <= 1-y
    radii = np.minimum(radii, centers[:, 0])
    radii = np.minimum(radii, 1.0 - centers[:, 0])
    radii = np.minimum(radii, centers[:, 1])
    radii = np.minimum(radii, 1.0 - centers[:, 1])

    # Constraint 2: Non-overlap
    # dist(i, j) >= r_i + r_j
    # This is harder to solve directly for all r_i simultaneously if we just take min dist/2
    # because r_i depends on r_j.
    # However, for a fixed configuration, the maximum radii are the solution to:
    # r_i <= d_ij - r_j for all j != i
    # r_i <= dist_to_wall_i
    # This is a system of linear inequalities. We can solve it iteratively or just approximate.
    # A simple approximation is r_i = min(dist_to_wall_i, 0.5 * min_dist_to_other_centers)
    # But this might be loose.
    # Better approach: Iterative relaxation.
    
    # Initialize radii based on walls
    r_wall = np.minimum(
        np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
        np.minimum(centers[:, 1], 1.0 - centers[:, 1])
    )
    radii = r_wall.copy()

    # Iteratively enforce pairwise constraints
    # r_i + r_j <= dist_ij  =>  r_i <= dist_ij - r_j
    # We can perform a few passes of relaxation.
    for _ in range(50):
        new_radii = radii.copy()
        for i in range(n):
            # Check against all other circles
            # r_i <= dist(i, j) - r_j
            # So r_i <= min_j (dist(i, j) - r_j)
            # Also r_i <= r_wall[i]
            
            # Vectorized distance calculation
            dists = np.linalg.norm(centers[i] - centers, axis=1)
            # For self, dist is 0, but we ignore i==i
            # We want min over j != i of (dists[j] - radii[j])
            
            # Mask out self
            mask = np.ones(n, dtype=bool)
            mask[i] = False
            
            # Calculate upper bounds imposed by neighbors
            bounds_from_neighbors = dists[mask] - radii[mask]
            
            # r_i must be <= min(bounds_from_neighbors)
            if len(bounds_from_neighbors) > 0:
                max_r = np.min(bounds_from_neighbors)
            else:
                max_r = r_wall[i]
            
            # Also bounded by wall
            max_r = np.min(max_r, r_wall[i])
            
            # Ensure non-negative
            max_r = np.max(max_r, 0.0)
            
            new_radii[i] = max_r
        radii = new_radii
        
    return radii

def objective_function(centers_flat):
    """
    Objective to maximize: sum of radii.
    Since scipy minimizes, we return negative sum.
    """
    n = 26
    centers = centers_flat.reshape(n, 2)
    
    # Ensure centers are strictly inside to avoid issues, though get_valid_radii handles boundaries
    # But if center is at 0, radius is 0.
    
    radii = get_valid_radii(centers)
    total_sum = np.sum(radii)
    
    # Return negative for minimization
    return -total_sum
