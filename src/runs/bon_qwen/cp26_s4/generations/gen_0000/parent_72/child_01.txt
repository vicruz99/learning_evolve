import numpy as np
from scipy.optimize import minimize
import math

def run_packing():
    """
    Packs 26 circles in a unit square to maximize sum of radii.
    Returns (centers, radii, sum_radii).
    """
    n = 26
    best_sum_radii = -1.0
    best_centers = None
    best_radii = None

    # Helper to run optimization from a specific initial guess
    def optimize_from(initial_guess):
        nonlocal best_sum_radii, best_centers, best_radii
        
        # Flatten variables: [x1, y1, r1, x2, y2, r2, ...]
        x0 = initial_guess.flatten()
        
        # Bounds: x in [0, 1], y in [0, 1], r in [0, 0.5]
        bounds = []
        for _ in range(n):
            bounds.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
        
        # Constraints
        constraints = []
        
        # 1. Non-overlap constraints: dist(ci, cj) >= ri + rj
        # dist^2 >= (ri + rj)^2  =>  (xi-xj)^2 + (yi-yj)^2 - (ri+rj)^2 >= 0
        # Using squared distance avoids sqrt, but (ri+rj)^2 is quadratic.
        # SLSQP handles this.
        for i in range(n):
            for j in range(i + 1, n):
                def con_overlap(idx_i=i, idx_j=j):
                    def func(x):
                        xi, yi, ri = x[3*idx_i], x[3*idx_i+1], x[3*idx_i+2]
                        xj, yj, rj = x[3*idx_j], x[3*idx_j+1], x[3*idx_j+2]
                        # Return value >= 0 for validity
                        # dist_sq - (ri + rj)^2
                        dx = xi - xj
                        dy = yi - yj
                        return (dx*dx + dy*dy) - (ri + rj)**2
                    return {'type': 'ineq', 'fun': func}
                constraints.append(con_overlap())
        
        # 2. Boundary constraints
        for i in range(n):
            # x >= r  => x - r >= 0
            def con_x_min(idx_i=i):
                def func(x):
                    xi, ri = x[3*idx_i], x[3*idx_i+2]
                    return xi - ri
                return {'type': 'ineq', 'fun': func}
            constraints.append(con_x_min())
            
            # 1 - x >= r => 1 - x - r >= 0
            def con_x_max(idx_i=i):
                def func(x):
                    xi, ri = x[3*idx_i], x[3*idx_i+2]
                    return 1.0 - xi - ri
                return {'type': 'ineq', 'fun': func}
            constraints.append(con_x_max())
            
            # y >= r => y - r >= 0
            def con_y_min(idx_i=i):
                def func(x):
                    yi, ri = x[3*idx_i+1], x[3*idx_i+2]
                    return yi - ri
                return {'type': 'ineq', 'fun': func}
            constraints.append(con_y_min())
            
            # 1 - y >= r => 1 - y - r >= 0
            def con_y_max(idx_i=i):
                def func(x):
                    yi, ri = x[3*idx_i+1], x[3*idx_i+2]
                    return 1.0 - yi - ri
                return {'type': 'ineq', 'fun': func}
            constraints.append(con_y_max())

        # Objective: Maximize sum(r_i) -> Minimize -sum(r_i)
        def objective(x):
            return -np.sum(x[2::3]) # radii are at indices 2, 5, 8...

        # Run optimization
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints, 
                           options={'maxiter': 1000, 'ftol': 1e-9})
            
            if res.success:
                sum_r = -res.fun
                if sum_r > best_sum_radii:
                    # Extract results
                    centers = res.x.reshape((n, 3))[:, :2]
                    radii = res.x.reshape((n, 3))[:, 2]
                    # Ensure non-negative radii (though bounds handle it, numerical noise might occur)
                    radii = np.maximum(radii, 0.0)
                    
                    # Validate quickly
                    # Check simple bounds
                    valid = True
                    for k in range(n):
                        cx, cy = centers[k]
                        cr = radii[k]
                        if cx < -1e-5 or cx > 1+1e-5 or cy < -1e-5 or cy > 1+1e-5:
                            valid = False
                            break
                        if cr < -1e-5:
                            valid = False
                            break
                    
                    if valid:
                        best_sum_radii = sum_r
                        best_centers = centers.copy()
                        best_radii = radii.copy()
                        
        except Exception:
            pass

    # --- Strategy 1: Perturbed Grid ---
    # Start with 25 circles in grid (r=0.1) and 1 in hole (r=0.04)
    # But optimizer might start with small radii and grow them.
    # Let's initialize centers for a dense packing.
    
    # Grid 5x5
    pts = []
    for i in range(5):
        for j in range(5):
            pts.append([0.1 + i*0.2, 0.1 + j*0.2])
    # Add one in hole
    pts.append([0.2, 0.2])
    
    # Initial radii small to satisfy constraints easily, then grow
    radii_init = [0.01] * 26
    
    x0_grid = []
    for p, r in zip(pts, radii_init):
        x0_grid.extend([p[0], p[1], r])
    
    optimize_from(np.array(x0_grid))

    # --- Strategy 2: Random Search (Multiple restarts) ---
    np.random.seed(42)
    for _ in range(5):
        centers_rand = np.random.rand(n, 2)
        radii_rand = np.full(n, 0.01)
        x0_rand = []
        for p, r in zip(centers_rand, radii_rand):
            x0_rand.extend([p[0], p[1], r])
        optimize_from(np.array(x0_rand))

    # --- Strategy 3: Hexagonal packing initialization ---
    # Try to pack 26 points in hexagonal lattice
    pts_hex = []
    r_approx = 0.08
    # Rows
    row_count = 0
    y = r_approx
    while y < 1.0 - r_approx:
        x = r_approx
        # Even rows
        while x < 1.0 - r_approx:
            pts_hex.append([x, y])
            if len(pts_hex) >= 26:
                break
            x += 2 * r_approx
        if len(pts_hex) >= 26:
            break
        row_count += 1
        
        y += r_approx * math.sqrt(3)
        x = r_approx + r_approx # Shifted
        row_count += 1
        while x < 1.0 - r_approx:
            pts_hex.append([x, y])
            if len(pts_hex) >= 26:
                break
            x += 2 * r_approx
        if len(pts_hex) >= 26:
            break
        y += r_approx * math.sqrt(3)
    
    # If we have fewer than 26, pad; if more, truncate
    while len(pts_hex) < 26:
        pts_hex.append([0.5, 0.5]) # Center
    
    pts_hex = pts_hex[:26]
    radii_hex_init = [0.01] * 26
    
    x0_hex = []
    for p, r in zip(pts_hex, radii_hex_init):
        x0_hex.extend([p[0], p[1], r])
    
    optimize_from(np.array(x0_hex))

    # If we found a solution
    if best_centers is not None:
        # Final check and formatting
        # The optimizer might have found radii slightly violating due to numerical issues, 
        # but we enforced constraints.
        # Just return the best found.
        return best_centers, best_radii, float(best_sum_radii)
    
    # Fallback (should not happen)
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    return centers, radii, 0.0
