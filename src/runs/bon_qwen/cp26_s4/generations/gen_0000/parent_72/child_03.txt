def compute_optimal_radii(centers):
    """
    Given fixed centers, compute radii that maximize sum of radii 
    subject to non-overlap and boundary constraints.
    This is solved as a Linear Programming problem.
    Maximize sum(r_i)
    Subject to:
      r_i + r_j <= d_ij  for all i < j
      r_i <= x_i
      r_i <= 1 - x_i
      r_i <= y_i
      r_i <= 1 - y_i
      r_i >= 0
    """
    n = centers.shape[0]
    if n == 0:
        return np.array([])

    # Calculate pairwise distances
    # dist[i, j] is distance between center i and center j
    dists = cdist(centers, centers, metric='euclidean')
    np.fill_diagonal(dists, np.inf) # Ignore self

    # Calculate boundary constraints for each circle
    # r_i <= min(x_i, 1-x_i, y_i, 1-y_i)
    boundaries = np.zeros(n)
    for i in range(n):
        x, y = centers[i]
        boundaries[i] = min(x, 1 - x, y, 1 - y)

    # We want to solve LP: max c^T r
    # c = [1, 1, ..., 1]
    # Constraints: A_ub r <= b_ub
    # r_i + r_j <= dists[i, j]
    # r_i <= boundaries[i]
    
    # Number of variables = n
    # Number of constraints = n*(n-1)/2 (pairwise) + 4*n (boundary)
    
    # Construct A_ub and b_ub
    # To save memory/time, we can just use the heuristic if LP is too slow,
    # but for N=26, LP is fast.
    
    # Heuristic approach (Voronoi) is faster and often near-optimal for sum.
    # r_i = 0.5 * min(boundaries[i] * 2, min_j dists[i, j])
    # Let's stick to the heuristic for speed in the optimization loop, 
    # as solving LP inside the objective function of a global optimizer is too slow.
    # However, the heuristic is valid.
    
    # Heuristic:
    # r_i is limited by half the distance to the nearest neighbor and distance to wall.
    min_dists = np.min(dists, axis=1)
    radii = 0.5 * np.minimum(min_dists, 2 * boundaries)
