def get_constraints(centers_init, radii_init):
    """
    Define constraints for the optimization problem.
    We need to minimize -sum(radii) subject to:
    1. 0 <= x_i - r_i
    2. x_i + r_i <= 1
    3. 0 <= y_i - r_i
    4. y_i + r_i <= 1
    5. (x_i - x_j)^2 + (y_i - y_j)^2 >= (r_i + r_j)^2

    n = len(radii_init)

    # Boundary constraints: r_i >= 0 is handled by bounds? 
    # Actually scipy bounds can handle lower bound 0 for r.
    # But x-r >= 0 => x >= r.

        # x_i - r_i >= 0
        cons.append({'type': 'ineq', 'fun': lambda v, i=i: v[3*i] - v[3*i+2]})
        # 1 - (x_i + r_i) >= 0
        cons.append({'type': 'ineq', 'fun': lambda v, i=i: 1.0 - (v[3*i] + v[3*i+2])})
        # y_i - r_i >= 0
        cons.append({'type': 'ineq', 'fun': lambda v, i=i: v[3*i+1] - v[3*i+2]})
        # 1 - (y_i + r_i) >= 0
        cons.append({'type': 'ineq', 'fun': lambda v, i=i: 1.0 - (v[3*i+1] + v[3*i+2])})

    # Overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            # dist^2 - (r_i + r_j)^2 >= 0
            # dist^2 = (x_i - x_j)^2 + (y_i - y_j)^2
            def overlap_constraint(v, i=i, j=j):
                xi, yi, ri = v[3*i], v[3*i+1], v[3*i+2]
                xj, yj, rj = v[3*j], v[3*j+1], v[3*j+2]
                dist_sq = (xi - xj)**2 + (yi - yj)**2
                sum_r_sq = (ri + rj)**2
                return dist_sq - sum_r_sq
            cons.append({'type': 'ineq', 'fun': overlap_constraint})
            
    return cons
