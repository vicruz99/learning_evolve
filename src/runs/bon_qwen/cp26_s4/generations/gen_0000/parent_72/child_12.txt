from scipy.optimize import minimize

    rows = 5

            if len(points) >= n:
                break

            x = (c_idx + 0.5) / cols
            y = (r_idx + 0.5) / rows
            
            # Shift odd rows horizontally by half a cell width for hexagonal packing
            if r_idx % 2 == 1:
                x += 0.5 / cols
                
            # Ensure points are within bounds [0, 1]

            if 0 <= x <= 1 and 0 <= y <= 1:

            r[i] = params[3*i + 2]

        params = np.zeros(3 * n)
