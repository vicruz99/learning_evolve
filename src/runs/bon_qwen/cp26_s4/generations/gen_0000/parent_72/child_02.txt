        radii: np.array of shape (26) with radius of each circle
        sum_radii: float sum of all radii

             # Simple check: if current row index (in terms of rows) is odd, offset by r

             # Let's just use a standard staggered grid logic
             pass

        # We center the row in the square [0, 1]

        y_pos += np.sqrt(3) * radii_init # Vertical spacing in hex packing

    centers_init = np.array(centers_init[:n])
    radii_init = np.full(n, radii_init)
