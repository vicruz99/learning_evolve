import numpy as np

def get_max_sum_radii(centers):
    """
    Solves the LP to find the maximum sum of radii for fixed centers.
    """
    n = centers.shape[0]
    
    # Variables: r_1, ..., r_n
    # Objective: minimize -sum(r_i)
    c = -np.ones(n)
    
    # Constraints matrix A_ub * x <= b_ub
    # Constraints:
    # 1. Boundary constraints for each circle i:
    #    r_i <= x_i  => -r_i <= -x_i
    #    r_i <= 1-x_i => -r_i <= -(1-x_i)
    #    r_i <= y_i  => -r_i <= -y_i
    #    r_i <= 1-y_i => -r_i <= -(1-y_i)
    #    Also r_i >= 0 is handled by bounds or implicit? 
    #    linprog handles bounds [0, None] by default? No, default is (-inf, inf).
    #    We must set bounds or add constraints.
    #    Let's add r_i >= 0 explicitly or use bounds.
    
    # We will construct A_ub and b_ub for the upper bounds of r_i.
    # Lower bounds r_i >= 0 can be set via `bounds` argument in linprog.
    
    # Number of upper bound constraints: 4*n + n*(n-1)/2
    num_boundary_constraints = 4 * n
    num_pair_constraints = n * (n - 1) // 2
    total_constraints = num_boundary_constraints + num_pair_constraints
    
    A_ub = np.zeros((total_constraints, n))
    b_ub = np.zeros(total_constraints)
    
    idx = 0
    
    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        # r_i <= x
        A_ub[idx, i] = -1.0
        b_ub[idx] = -x
        idx += 1
        
        # r_i <= 1 - x
        A_ub[idx, i] = -1.0
        b_ub[idx] = -(1.0 - x)
        idx += 1
        
        # r_i <= y
        A_ub[idx, i] = -1.0
        b_ub[idx] = -y
        idx += 1
        
        # r_i <= 1 - y
        A_ub[idx, i] = -1.0
        b_ub[idx] = -(1.0 - y)
        idx += 1
        
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dist
            idx += 1
            
    # Bounds for r_i: [0, infinity)
    bounds = [(0, None) for _ in range(n)]
    
    # Solve LP
    try:
        res = opt.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return -res.fun, res.x
        else:
            # Fallback: return 0 radii
            return 0.0, np.zeros(n)
    except Exception:
        return 0.0, np.zeros(n)

    centers = np.clip(centers, 0.0, 1.0)
    
    sum_r, _ = get_max_sum_radii(centers)
    return -sum_r
