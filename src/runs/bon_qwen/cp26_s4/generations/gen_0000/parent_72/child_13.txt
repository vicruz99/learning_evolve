import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def get_objective(x):
    """Objective function: minimize negative sum of radii."""
    radii = x[2::3]
    return -np.sum(radii)

def get_constraints(x):
    """Inequality constraints: boundary and non-overlap."""
    n = len(x) // 3
    c = np.empty(4 * n + n * (n - 1) // 2)
    
    X = x[:2*n].reshape(n, 2)
    R = x[2*n:]
    
    # Boundary constraints: x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
    c[:4*n] = np.concatenate([
        X[:, 0] - R,
        1.0 - X[:, 0] - R,
        X[:, 1] - R,
        1.0 - X[:, 1] - R
    ])
    
    # Overlap constraints: ||Ci - Cj||^2 >= (ri + rj)^2
    dx = X[:, 0, np.newaxis] - X[:, 0]
    dy = X[:, 1, np.newaxis] - X[:, 1]
    dist_sq = dx**2 + dy**2
    
    r_sum = R[:, np.newaxis] + R
    r_sum_sq = r_sum**2
    
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    c[4*n:] = (dist_sq - r_sum_sq)[mask]
    
    return c

def run_packing():
    # Initial guess: Hexagonal packing
    r_init = 0.095
    centers = []
    radii = []
    
    y = r_init
    row = 0
    while len(centers) < N_CIRCLES:
        x = r_init if row % 2 == 0 else 2 * r_init
        while x + r_init <= 1.0 and len(centers) < N_CIRCLES:
            centers.append([x, y])
            radii.append(r_init)
            x += 2 * r_init
        y += r_init * np.sqrt(3)
        row += 1
        
    centers = centers[:N_CIRCLES]
    radii = radii[:N_CIRCLES]
    
    x0 = np.zeros(3 * N_CIRCLES)
    for i in range(N_CIRCLES):
        x0[3*i] = centers[i][0]
        x0[3*i+1] = centers[i][1]
        x0[3*i+2] = radii[i]
        
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N_CIRCLES
    cons = {'type': 'ineq', 'fun': get_constraints}
    
    best_x = x0
    best_val = get_objective(x0)
    
    # Multiple restarts to escape local minima
    for attempt in range(20):
        x_try = x0.copy()
        if attempt > 0:
            noise = np.random.normal(0, 0.005, size=3 * N_CIRCLES)
            x_try += noise
            for i in range(0, 3*N_CIRCLES, 3):
                x_try[i] = np.clip(x_try[i], 0, 1)
                x_try[i+1] = np.clip(x_try[i+1], 0, 1)
                x_try[i+2] = np.clip(x_try[i+2], 0, 0.5)
                
        try:
            res = minimize(get_objective, x_try, method='SLSQP', bounds=bounds, 
                           constraints=cons, options={'maxiter': 2000, 'ftol': 1e-10})
            if res.success and res.fun < best_val:
                best_val = res.fun
                best_x = res.x
        except Exception:
            continue
            
    res_centers = best_x[:2*N_CIRCLES].reshape(N_CIRCLES, 2)
    res_radii = best_x[2*N_CIRCLES:]
    
    return res_centers, res_radii, -best_val
