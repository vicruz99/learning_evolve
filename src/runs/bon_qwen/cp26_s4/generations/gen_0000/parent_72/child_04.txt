import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    Uses SLSQP optimization starting from a hexagonal lattice configuration.
    """
    n = 26
    
    # 1. Generate initial hexagonal configuration
    # Pattern: 5, 6, 5, 6, 4 circles per row to sum to 26
    row_counts = [5, 6, 5, 6, 4]
    centers_init = []
    radii_init = []
    
    # Estimated radius for initialization (slightly less than target avg to ensure feasibility)
    # Target avg ~ 0.1014. Let's start with 0.09 to be safe and let optimizer expand.
    r_est = 0.09
    dy = r_est * np.sqrt(3)  # Vertical spacing for hex packing
    dx = 2 * r_est           # Horizontal spacing
    
    current_y = r_est + 0.05 # Start with some margin
    
    for i, count in enumerate(row_counts):
        # Shift even rows (1, 3, ...) by dx/2 to create hexagonal stagger
        x_offset = r_est
        if i % 2 == 1:
            x_offset += dx / 2
        
        # Center the row horizontally within the square
        # Total width of 'count' circles in this row is roughly count * dx
        # But we need to place them such that they fit in [0, 1]
        # A simple heuristic: distribute centers evenly with margin
        # Actually, let's just place them starting from x_offset and check bounds later
        # Better: calculate positions relative to row center
        
        row_width = (count - 1) * dx
        start_x = (1.0 - row_width) / 2
        
        for j in range(count):
            cx = start_x + j * dx
            cy = current_y
            centers_init.append([cx, cy])
            radii_init.append(r_est)
        
        current_y += dy
        
    centers_init = np.array(centers_init)
    radii_init = np.array(radii_init)
    
    # Ensure we have exactly 26
    assert len(centers_init) == n
    
    # 2. Define objective and constraints for optimization
    # Variables: [x1, y1, r1, x2, y2, r2, ...]
    # Total variables: 26 * 3 = 78
    
    def objective(vars):
        # Maximize sum of radii -> Minimize -sum(radii)
        r_part = vars[2::3]
        return -np.sum(r_part)

    def constraint_boundary(vars):
        # Constraints: x >= r, x <= 1-r, y >= r, y <= 1-r
        # Reshaped to (26, 3) -> columns 0:x, 1:y, 2:r
        pts = vars.reshape(-1, 3)
        x = pts[:, 0]
        y = pts[:, 1]
        r = pts[:, 2]
        
        # x - r >= 0
        c1 = x - r
        # 1 - x - r >= 0
        c2 = 1 - x - r
        # y - r >= 0
        c3 = y - r
        # 1 - y - r >= 0
        c4 = 1 - y - r
        
        return np.concatenate([c1, c2, c3, c4])

    def constraint_overlap(vars):
        # Constraints: dist(i, j) >= r_i + r_j
        # dist^2 >= (r_i + r_j)^2
        pts = vars.reshape(-1, 3)
        n_pts = pts.shape[0]
        constraints = []
        
        for i in range(n_pts):
            for j in range(i + 1, n_pts):
                dx = pts[i, 0] - pts[j, 0]
                dy = pts[i, 1] - pts[j, 1]
                dist_sq = dx*dx + dy*dy
                r_sum = pts[i, 2] + pts[j, 2]
                # dist_sq - r_sum^2 >= 0
                constraints.append(dist_sq - r_sum*r_sum)
        
        return np.array(constraints)

    # 3. Setup bounds and initial guess
    # Bounds: 0 <= x <= 1, 0 <= y <= 1, 0 <= r <= 0.5
    bounds = []
    for _ in range(n):
        bounds.extend([
            (0.0, 1.0), # x
            (0.0, 1.0), # y
            (0.0, 0.5)  # r
        ])
    
    # Flatten initial guess
    x0 = np.zeros(n * 3)
    for i in range(n):
        x0[3*i] = centers_init[i, 0]
        x0[3*i+1] = centers_init[i, 1]
        x0[3*i+2] = radii_init[i]

    # 4. Run Optimization
    # SLSQP is suitable for constrained non-linear optimization
    res = minimize(
        objective,
        x0,
        method='SLSQP',
        bounds=bounds,
        constraints=[
            {'type': 'ineq', 'fun': constraint_boundary},
            {'type': 'ineq', 'fun': constraint_overlap}
        ],
        options={'maxiter': 1000, 'ftol': 1e-9}
    )
    
    # 5. Extract results
    if res.success or not res.success: # Even if not strictly successful, best effort
        best_vars = res.x
    else:
        # Fallback to initial if optimization failed completely
        best_vars = x0

    centers_res = np.zeros((n, 2))
    radii_res = np.zeros(n)
    for i in range(n):
        centers_res[i, 0] = best_vars[3*i]
        centers_res[i, 1] = best_vars[3*i+1]
        radii_res[i] = best_vars[3*i+2]
        
    sum_radii = np.sum(radii_res)
    
    return centers_res, radii_res, sum_radii
