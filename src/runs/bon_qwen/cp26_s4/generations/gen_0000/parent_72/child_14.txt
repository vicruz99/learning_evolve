To maximize the sum of radii for 26 circles in a unit square, we need to find a configuration where the circles are packed as densely as possible. Since the objective is to maximize the sum of radii, and the constraints are non-convex, a numerical optimization approach is appropriate.

**Strategy:**
1.  **Optimization Formulation**: We formulate the problem as a non-linear programming (NLP) problem. We optimize the variables $(x_i, y_i, r_i)$ for $i=1 \dots 26$. The objective is to maximize $\sum r_i$. The constraints ensure that circles remain within the square $[0,1]^2$ and do not overlap.
2.  **Solver**: We use `scipy.optimize.minimize` with the `SLSQP` method. This solver is capable of handling the quadratic inequality constraints arising from circle-circle and circle-boundary interactions.
3.  **Gradients**: To improve convergence speed and robustness, we provide analytical gradients (Jacobian) for both the objective function and the constraint functions. The constraint functions are differentiable everywhere except at points where distance is zero (which is excluded by the constraints themselves).
4.  **Initialization**: We start with a well-distributed initial configuration (a hexagonal lattice) to give the optimizer a head start. Since the landscape has many local minima, we perform multiple optimization runs with slightly perturbed initial positions and select the best result.
5.  **Robustness**: We include a fallback mechanism. If the optimizer fails or returns an invalid result, we rely on the best valid solution found.

import numpy as np
from scipy.optimize import minimize

# Global constants
N_CIRCLES = 26

def objective(vars):
    """Objective function: Minimize negative sum of radii."""
    # vars structure: [x0..x25, y0..y25, r0..r25]
    # radii are in the last N_CIRCLES elements
    radii = vars[2 * N_CIRCLES:]
    return -np.sum(radii)

def obj_jac(vars):
    """Gradient of the objective function."""
    grad = np.zeros_like(vars)
    # Derivative wrt radii is -1
    grad[2 * N_CIRCLES:] = -1.0
    return grad

def constraints_fun(vars):
    """
    Computes all inequality constraints (must be >= 0).
    1. Boundary constraints: x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
    2. Pairwise distance constraints: dist^2 - (r_i + r_j)^2 >= 0
    """
    centers_x = vars[:N_CIRCLES]
    centers_y = vars[N_CIRCLES:2 * N_CIRCLES]
    radii = vars[2 * N_CIRCLES:]

    # 1. Boundary constraints
    # x_i - r_i >= 0
    c1 = centers_x - radii
    # 1 - x_i - r_i >= 0
    c2 = 1.0 - centers_x - radii
    # y_i - r_i >= 0
    c3 = centers_y - radii
    # 1 - y_i - r_i >= 0
    c4 = 1.0 - centers_y - radii
    
    boundary_cons = np.concatenate([c1, c2, c3, c4])

    # 2. Pairwise constraints
    # (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
    # We only need to check i < j to avoid duplicates and i == j (which is trivial)
    
    # Vectorized calculation
    # Create column vectors for broadcasting
    cx = centers_x[:, np.newaxis] # (N, 1)
    cy = centers_y[:, np.newaxis]
    cr = radii[:, np.newaxis]

    # Compute squared distances for all pairs
    # diff_x shape (N, N)
    diff_x = cx - cx.T
    diff_y = cy - cy.T
    dist_sq = diff_x**2 + diff_y**2
    
    # Compute (r_i + r_j)^2
    r_sum_sq = (cr + cr.T)**2
    
    # Constraint violation
    pairwise_cons_mat = dist_sq - r_sum_sq
    
    # Extract upper triangle (excluding diagonal)
    # np.triu_indices with k=1 gives indices for i < j
    triu_idx = np.triu_indices(N_CIRCLES, k=1)
    pairwise_cons = pairwise_cons_mat[triu_idx]

    return np.concatenate([boundary_cons, pairwise_cons])

def constraints_jac(vars):
    """
    Jacobian of the constraints function.
    Shape: (num_constraints, 3 * N_CIRCLES)
    """
    n = N_CIRCLES
    num_boundary = 4 * n
    num_pairwise = n * (n - 1) // 2
    total_constraints = num_boundary + num_pairwise
    
    jac = np.zeros((total_constraints, 3 * n))
    
    centers_x = vars[:n]
    centers_y = vars[n:2 * n]
    radii = vars[2 * n:]
    
    # --- Boundary Constraints Jacobian ---
    # c1: x_i - r_i >= 0  -> dx_i = 1, dr_i = -1
    # c2: 1 - x_i - r_i >= 0 -> dx_i = -1, dr_i = -1
    # c3: y_i - r_i >= 0 -> dy_i = 1, dr_i = -1
    # c4: 1 - y_i - r_i >= 0 -> dy_i = -1, dr_i = -1
    
    # Indices in vars:
    # x: 0..n-1
    # y: n..2n-1
    # r: 2n..3n-1
    
    # c1
    for i in range(n):
        jac[i, i] = 1.0                 # dx_i
        jac[i, 2 * n + i] = -1.0        # dr_i
        
    # c2
    for i in range(n):
        jac[n + i, i] = -1.0            # dx_i
        jac[n + i, 2 * n + i] = -1.0    # dr_i
        
    # c3
    for i in range(n):
        jac[2 * n + i, n + i] = 1.0     # dy_i
        jac[2 * n + i, 2 * n + i] = -1.0 # dr_i
        
    # c4
    for i in range(n):
        jac[3 * n + i, n + i] = -1.0    # dy_i
        jac[3 * n + i, 2 * n + i] = -1.0 # dr_i
        
    # --- Pairwise Constraints Jacobian ---
    # Constraint g_ij = (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
    # Indices in pairwise_cons correspond to pairs (i, j) with i < j
    
    # Map (i, j) to linear index in pairwise_cons
    # Number of pairs before row i is sum_{k=0}^{i-1} (n - 1 - k)
    # Actually, easier to just iterate and assign.
    
    # Precompute distances for efficiency if needed, but loop is fine for N=26
    # 325 constraints, 78 vars. 325*78 ~ 25k operations. Fast.
    
    row_idx = num_boundary
    for i in range(n):
        for j in range(i + 1, n):
            # Terms
            dx = centers_x[i] - centers_x[j]
            dy = centers_y[i] - centers_y[j]
            r_sum = radii[i] + radii[j]
            
            # Partial derivatives
            # wrt x_i: 2 * dx
            jac[row_idx, i] = 2.0 * dx
            # wrt x_j: -2 * dx
            jac[row_idx, j] = -2.0 * dx
            
            # wrt y_i: 2 * dy
            jac[row_idx, n + i] = 2.0 * dy
            # wrt y_j: -2 * dy
            jac[row_idx, n + j] = -2.0 * dy
            
            # wrt r_i: -2 * (r_i + r_j)
            jac[row_idx, 2 * n + i] = -2.0 * r_sum
            # wrt r_j: -2 * (r_i + r_j)
            jac[row_idx, 2 * n + j] = -2.0 * r_sum
            
            row_idx += 1
            
    return jac

def generate_hexagonal_centers(n):
    """Generates a hexagonal packing initial guess."""
    # Estimate radius. For 26 circles, roughly 0.1
    r_est = 0.1
    s = 2 * r_est
    
    centers = []
    row = 0
    while len(centers) < n:
        y = r_est + row * s * np.sqrt(3) / 2
        if y + r_est > 1.0:
            # If row is too high, just stop or shrink? 
            # Let's just keep adding, optimizer will fix it.
            pass
        
        # Offset for odd rows
        offset = 0 if row % 2 == 0 else s / 2
        
        col = 0
        while True:
            x = r_est + offset + col * s
            if x + r_est > 1.0:
                break
            centers.append([x, y])
            col += 1
        row += 1
        
        # Safety break
        if row > 20:
            break
            
    # If we have more than n, truncate
    # If fewer, fill with random?
    if len(centers) < n:
        # Fill remaining with random positions near boundaries
        for _ in range(n - len(centers)):
            centers.append([np.random.uniform(0.1, 0.9), np.random.uniform(0.1, 0.9)])
            
    return centers[:n]

def run_packing():
    # Bounds: x, y in [0, 1], r in [0, 0.5]
    bounds = []
    for _ in range(N_CIRCLES):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 0.5)) # r
    
    constraints = {
        'type': 'ineq',
        'fun': constraints_fun,
        'jac': constraints_jac
    }
    
    best_sum_r = -1.0
    best_vars = None
    
    # Run multiple optimizations from different starting points
    # 1. Hexagonal grid
    # 2. Perturbed hexagonal grid
    # 3. Random
    
    initial_guesses = []
    
    # 1. Hexagonal
    centers_hex = generate_hexagonal_centers(N_CIRCLES)
    r_init = 0.05 # Start small to be feasible
    vars_hex = np.array([])
    for c in centers_hex:
        vars_hex = np.append(vars_hex, [c[0], c[1], r_init])
    initial_guesses.append(vars_hex)
    
    # 2. Perturbed Hexagonal
    vars_hex_pert = vars_hex.copy()
    # Add small noise
    noise = np.random.normal(0, 0.01, size=vars_hex_pert.shape)
    # Clip to bounds roughly
    vars_hex_pert += noise
    # Ensure feasibility roughly? Optimizer handles bounds.
    initial_guesses.append(vars_hex_pert)
    
    # 3. Grid (5x5 approx)
    vars_grid = np.array([])
    count = 0
    # 5 rows, approx 5 cols
    for r in range(5):
        for c in range(6):
            if count >= N_CIRCLES:
                break
            x = 0.1 + c * 0.18
            y = 0.1 + r * 0.2
            if x > 0.9: x = 0.9
            if y > 0.9: y = 0.9
            vars_grid = np.append(vars_grid, [x, y, 0.05])
            count += 1
    initial_guesses.append(vars_grid)
    
    # 4. Random
    vars_rand = np.array([])
    for _ in range(N_CIRCLES):
        x = np.random.uniform(0.2, 0.8)
        y = np.random.uniform(0.2, 0.8)
        vars_rand = np.append(vars_rand, [x, y, 0.02])
    initial_guesses.append(vars_rand)
    
    for i, x0 in enumerate(initial_guesses):
        try:
            res = minimize(objective, x0, method='SLSQP', jac=obj_jac, 
                           bounds=bounds, constraints=constraints,
                           options={'maxiter': 500, 'ftol': 1e-9})
            
            if res.success:
                current_sum = -res.fun # Objective was negative sum
                if current_sum > best_sum_r:
                    best_sum_r = current_sum
                    best_vars = res.x
            else:
                # Even if not successful, check if result is valid and better
                # Sometimes 'success' is False but we found a local max
                current_sum = -res.fun
                # Validate manually just in case
                centers = np.column_stack((res.x[:N_CIRCLES], res.x[N_CIRCLES:2*N_CIRCLES]))
                radii = res.x[2*N_CIRCLES:]
                # Quick check for validity (simplified)
                # We rely on the validation function provided later, but for comparison:
                if np.all(radii >= 0) and np.all(centers >= 0) and np.all(centers <= 1):
                     # Check overlaps roughly
                     valid = True
                     for k in range(N_CIRCLES):
                         for l in range(k+1, N_CIRCLES):
                             dist = np.sqrt(np.sum((centers[k] - centers[l])**2))
                             if dist < radii[k] + radii[l] - 1e-7:
                                 valid = False
                                 break
                         if not valid: break
                     
                     if valid and current_sum > best_sum_r:
                         best_sum_r = current_sum
                         best_vars = res.x
                        
        except Exception as e:
            print(f"Optimization run {i} failed: {e}")
            continue

    if best_vars is None:
        # Fallback to last result or initial guess
        best_vars = vars_hex # Just a placeholder
        
    centers = np.column_stack((best_vars[:N_CIRCLES], best_vars[N_CIRCLES:2*N_CIRCLES]))
    radii = best_vars[2*N_CIRCLES:]
    
    return centers, radii, np.sum(radii)
