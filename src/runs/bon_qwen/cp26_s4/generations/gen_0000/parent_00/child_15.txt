import numpy as np
from scipy.optimize import minimize

def objective_func(v):
    """Objective function: maximize sum of radii (minimize negative sum)"""
    n = len(v) // 3
    return -np.sum(v[2*n:])

def constraint_func(v):
    """Constraint function: ensures circles are inside square and do not overlap"""
    n = len(v) // 3
    x = v[:n]
    y = v[n:2*n]
    r = v[2*n:]
    
    # Boundary constraints: r <= x, r <= 1-x, r <= y, r <= 1-y
    # Formulated as inequalities >= 0
    c = [x - r, 1.0 - x - r, y - r, 1.0 - y - r]
    
    # Pairwise non-overlap constraints: r_i + r_j <= dist(i, j)
    # Formulated as dist - (r_i + r_j) >= 0
    ix, jx = np.triu_indices(n, k=1)
    dx = x[ix] - x[jx]
    dy = y[ix] - y[jx]
    dist = np.sqrt(dx*dx + dy*dy)
    c.append(dist - (r[ix] + r[jx]))
    
    return np.concatenate(c)

def generate_initial_configs(n):
    """Generate diverse initial center configurations for multiple restarts"""
    configs = []
    
    # Config 1: 5x5 square grid + center point
    x = np.linspace(0.2, 0.8, 5)
    y = np.linspace(0.2, 0.8, 5)
    c1 = np.array([[a, b] for a in x for b in y])
    c1 = np.vstack([c1, [0.5, 0.5]])
    configs.append(c1)
    
    # Config 2: Hexagonal lattice approximation
    pts = []
    row = 0
    while len(pts) < n:
        for col in range(6):
            if len(pts) >= n: break
            cx = 0.1 + col * 0.17

            pts.append([cx, cy])
        row += 1
    configs.append(np.array(pts))
    
    # Config 3: Structured random initialization
    np.random.seed(42)
    c3 = np.random.uniform(0.1, 0.9, size=(n, 2))
    configs.append(c3)
    
    return configs

def optimize_single(init_centers):
    """Run SLSQP optimization from a given initial configuration"""
    n = init_centers.shape[0]
    x0 = init_centers.flatten()
    r0 = np.full(n, 0.05)
    vars0 = np.concatenate([x0, r0])
    
    bounds = [(0.0, 1.0)]*(2*n) + [(0.0, None)]*n
    cons = {'type': 'ineq', 'fun': constraint_func}
    
    res = minimize(objective_func, vars0, method='SLSQP', bounds=bounds, constraints=cons, 
                   options={'maxiter': 3000, 'ftol': 1e-10})
    return res.x, -res.fun

def run_packing():
    """Main entry point to pack 26 circles and maximize sum of radii"""
    n = 26
    configs = generate_initial_configs(n)
    best_sum = -np.inf
    best_vars = None
    
    for c in configs:
        vars_res, s = optimize_single(c)
        if s > best_sum:
            best_sum = s
            best_vars = vars_res
            
    x = best_vars[:n]
    y = best_vars[n:2*n]
    r = best_vars[2*n:]
    centers = np.column_stack([x, y])
    
    # Ensure centers are strictly within [0,1]
    centers = np.clip(centers, 0.0, 1.0)
    
    return centers, r, np.sum(r)
