import numpy as np
from scipy.optimize import minimize


def neg_sum_radii(params):
    """Objective function: minimize negative sum of radii (maximize sum)."""
    n = len(params) // 3
    return -np.sum(params[2::3])


def constr_left(params, i):
    """Circle i must not cross left boundary: x_i - r_i >= 0."""
    return params[3 * i] - params[3 * i + 2]


def constr_right(params, i):
    """Circle i must not cross right boundary: 1 - x_i - r_i >= 0."""
    return 1.0 - params[3 * i] - params[3 * i + 2]


def constr_bottom(params, i):
    """Circle i must not cross bottom boundary: y_i - r_i >= 0."""
    return params[3 * i + 1] - params[3 * i + 2]


def constr_top(params, i):
    """Circle i must not cross top boundary: 1 - y_i - r_i >= 0."""
    return 1.0 - params[3 * i + 1] - params[3 * i + 2]


def constr_overlap(params, i, j):
    """Circles i and j must not overlap: dist^2 - (r_i + r_j)^2 >= 0."""
    xi = params[3 * i]
    yi = params[3 * i + 1]
    ri = params[3 * i + 2]
    xj = params[3 * j]
    yj = params[3 * j + 1]
    rj = params[3 * j + 2]
    dx = xi - xj
    dy = yi - yj
    return dx * dx + dy * dy - (ri + rj) ** 2

    for i in range(n):
        for j in range(i + 1, n):
            cons.append({'type': 'ineq', 'fun': constr_overlap, 'args': (i, j)})
    return cons
