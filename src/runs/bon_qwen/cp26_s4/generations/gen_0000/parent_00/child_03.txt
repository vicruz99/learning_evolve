import numpy as np
from scipy.optimize import linprog, basinhopping

def calculate_max_sum_radii(centers):
    """
    For a fixed set of centers, solve the LP to find the maximum sum of radii
    and the corresponding radius for each circle.
    """
    n = centers.shape[0]
    
    # Variables: r_0, ..., r_{n-1}
    # Objective: Maximize sum(r_i) -> Minimize -sum(r_i)
    c_obj = -np.ones(n)
    
    # Constraints: A_ub @ r <= b_ub
    # 1. Boundary constraints (r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i)
    # 2. Pairwise constraints (r_i + r_j <= dist(i, j))
    
    num_constraints = 4 * n + n * (n - 1) // 2
    A_ub = np.zeros((num_constraints, n))
    b_ub = np.zeros(num_constraints)
    
    idx = 0
    
    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        
        # r_i <= x
        A_ub[idx, i] = 1.0
        b_ub[idx] = x
        idx += 1
        
        # r_i <= 1-x
        A_ub[idx, i] = 1.0
        b_ub[idx] = 1.0 - x
        idx += 1
        
        # r_i <= y
        A_ub[idx, i] = 1.0
        b_ub[idx] = y
        idx += 1
        
        # r_i <= 1-y
        A_ub[idx, i] = 1.0
        b_ub[idx] = 1.0 - y
        idx += 1
        
    # Pairwise constraints
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt((centers[i,0] - centers[j,0])**2 + (centers[i,1] - centers[j,1])**2)
            if dist < 1e-9: # Prevent numerical instability
                dist = 1e-9
            
            # r_i + r_j <= dist
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dist
            idx += 1
            
    # Bounds: r_i >= 0
    bounds = [(0, None) for _ in range(n)]
    
    # Solve LP
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return res.x, -res.fun # radii, sum_radii
    else:
        # Fallback if infeasible (shouldn't happen if centers in [0,1])
        return np.zeros(n), 0.0

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # 1. Initialization: Hexagonal Lattice
    # Row spacing and x-offsets for hex packing
    x_even = [0.1, 0.3, 0.5, 0.7, 0.9]
    x_odd = [0.2, 0.4, 0.6, 0.8]
    y_start = 0.1
    dy = 0.1732
    
    centers_init = []
    count = 0
    row_idx = 0
    current_y = y_start
    
    while count < n:
        xs = x_even if row_idx % 2 == 0 else x_odd
        for x in xs:
            if count < n:
                centers_init.append([x, current_y])
                count += 1
            else:
                break
        current_y += dy
        row_idx += 1
        
    centers_init = np.array(centers_init[:n])
    
    # 2. Define Objective for Optimizer (Minimize Negative Sum)
    def objective(centers_flat):
        centers = centers_flat.reshape(-1, 2)
        # Clip to [0, 1] to ensure LP feasibility
        centers = np.clip(centers, 0.0, 1.0)
        _, s = calculate_max_sum_radii(centers)
        return -s # Minimize negative sum -> Maximize sum
    
    # 3. Optimization using Basin Hopping
    bounds = [(0.0, 1.0) for _ in range(n * 2)]
    minres = basinhopping(
        objective, 
        centers_init.flatten(), 
        niter=100, 
        T=0.5, 
        stepsize=0.05,
        minimizer_kwargs={"method": "Nelder-Mead"}
    )
    
    # 4. Extract and Finalize
    best_centers = np.clip(minres.x.reshape(n, 2), 0.0, 1.0)
    best_radii, sum_radii = calculate_max_sum_radii(best_centers)
    
    # Ensure no NaNs and valid values
    best_radii = np.nan_to_num(best_radii, nan=0.0, posinf=0.0, neginf=0.0)
    best_radii = np.maximum(best_radii, 0.0)
    
    return best_centers, best_radii, sum_radii
