def repulsion_and_boundary_forces(centers, radii):

    Calculates repulsive forces between overlapping circles and 
    boundary constraints.
    """

                force_mag = (min_dist - dist) / dist
                forces[i] += diff * force_mag
                forces[j] -= diff * force_mag

    for i in range(n):

            forces[i, 0] += (r - x) * 2.0
