import numpy as np
from scipy.optimize import minimize, NonlinearConstraint

def boundary_constraints(v):
    """Ensures circles remain within the unit square."""
    n = len(v) // 3
    c = np.zeros(4 * n)
    for i in range(n):
        idx = 3 * i
        # x >= r
        c[4 * i] = v[idx] - v[idx + 2]
        # x + r <= 1  => 1 - x - r >= 0
        c[4 * i + 1] = 1.0 - v[idx] - v[idx + 2]
        # y >= r
        c[4 * i + 2] = v[idx + 1] - v[idx + 2]
        # y + r <= 1  => 1 - y - r >= 0
        c[4 * i + 3] = 1.0 - v[idx + 1] - v[idx + 2]
    return c

def pairwise_constraints(v):
    """Ensures no two circles overlap."""
    n = len(v) // 3
    X = v[:2 * n].reshape((n, 2))
    R = v[2 * n:]
    # Compute squared Euclidean distances between all pairs
    diff = X[:, np.newaxis, :] - X[np.newaxis, :, :]
    dist2 = np.sum(diff**2, axis=2)
    # Compute squared sum of radii
    radii_sum = R[:, np.newaxis] + R[np.newaxis, :]
    # Mask for upper triangle (i < j)
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    # Constraint: dist^2 - (r_i + r_j)^2 >= 0
    return dist2[mask] - radii_sum[mask]**2

def run_packing():
    n = 26
    r_init = 0.04
    
    # Generate a hexagonal lattice initial configuration
    points = []
    y = r_init
    row_parity = 0
    while y + r_init <= 1.0:
        x = r_init
        shift = 0.0 if row_parity == 0 else r_init
        while x + r_init <= 1.0:
            points.append([x + shift, y])
            x += 2 * r_init
        y += np.sqrt(3) * r_init
        row_parity = 1 - row_parity
        
    # Pad if necessary (should not happen with r_init=0.04)
    while len(points) < n:
        points.append([0.5, 0.5])
        
    points = np.array(points[:n])
    radii = np.full(n, r_init)
    # Add tiny perturbation to break symmetry and aid optimization
    radii += np.arange(n) * 1e-6
    
    x0 = np.hstack([points.ravel(), radii])
    
    def objective(v):
        return -np.sum(v[2*n:])
        
    cons_boundary = NonlinearConstraint(boundary_constraints, 0, np.inf)
    cons_pairwise = NonlinearConstraint(pairwise_constraints, 0, np.inf)
    
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
    
    res = minimize(objective, x0, method='SLSQP', bounds=bounds, 
                   constraints=[cons_boundary, cons_pairwise], 
                   options={'maxiter': 3000, 'ftol': 1e-12, 'disp': False})
                   
    centers = res.x[:2*n].reshape((n, 2))
    radii_opt = res.x[2*n:3*n]
    
    return centers, radii_opt, float(np.sum(radii_opt))
