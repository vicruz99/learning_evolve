def objective(v):
    """
    Objective function to minimize.
    We want to maximize sum of radii, so we minimize negative sum.
    v is a flattened array [x1, y1, r1, x2, y2, r2, ...]
    """
    # Radii are at indices 2, 5, 8, ... (step 3, starting at 2)
    radii = v[2::3]
    return -np.sum(radii)

def constraint_boundary(v):
    """
    Boundary constraints:
    Circles must be inside [0, 1] x [0, 1].
    This implies:
    x_i >= r_i  => x_i - r_i >= 0
    x_i <= 1 - r_i => 1 - x_i - r_i >= 0
    y_i >= r_i  => y_i - r_i >= 0
    y_i <= 1 - r_i => 1 - y_i - r_i >= 0
    """
    # Reshape v to (N, 3) where columns are x, y, r
    V = v.reshape(-1, 3)
    x = V[:, 0]
    y = V[:, 1]
    r = V[:, 2]
    
    # Compute constraints
    c1 = x - r
    c2 = 1.0 - x - r
    c3 = y - r
    c4 = 1.0 - y - r
    
    # Concatenate all constraints into a single array
    return np.concatenate([c1, c2, c3, c4])

    """
    Overlap constraints:
    Distance between centers >= sum of radii.
    Equivalent to: dist^2 >= (r_i + r_j)^2
    dist^2 - (r_i + r_j)^2 >= 0

    radii = V[:, 2]   # (N,)

    diff = coords[:, np.newaxis, :] - coords[np.newaxis, :, :]
