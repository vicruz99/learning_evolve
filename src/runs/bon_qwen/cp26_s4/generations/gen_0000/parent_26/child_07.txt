    def get_constraints(vars, n):
        constraints = []
        
        # Extract r
        r = vars[-1]
        
        # Extract centers
        centers = vars[:-1].reshape(n, 2)
        x = centers[:, 0]
        y = centers[:, 1]
        
        # Boundary constraints
        # x - r >= 0
        for i in range(n):
            constraints.append({'type': 'ineq', 'fun': lambda v, idx=i: v[2*idx] - v[-1]})
            # 1 - r - x >= 0  => x <= 1 - r
            constraints.append({'type': 'ineq', 'fun': lambda v, idx=i: 1.0 - v[-1] - v[2*idx]})
            # y - r >= 0
            constraints.append({'type': 'ineq', 'fun': lambda v, idx=i: v[2*idx + 1] - v[-1]})
            # 1 - r - y >= 0
            constraints.append({'type': 'ineq', 'fun': lambda v, idx=i: 1.0 - v[-1] - v[2*idx + 1]})
            
        # Distance constraints
        # (xi - xj)^2 + (yi - yj)^2 - 4r^2 >= 0
        # We only need to add active constraints or all? All is safer but slow.
        # 26 circles -> 325 pairs. It's manageable.
        for i in range(n):
            for j in range(i + 1, n):
                # Indices in vars: 2*i, 2*i+1, 2*j, 2*j+1, -1
                xi_idx, yi_idx = 2 * i, 2 * i + 1
                xj_idx, yj_idx = 2 * j, 2 * j + 1
                r_idx = -1
                
                def make_dist_constraint(i_idx, j_idx, r_idx):
                    def dist_con(v):
                        xi, yi = v[i_idx], v[i_idx+1]
                        xj, yj = v[j_idx], v[j_idx+1]
                        r = v[r_idx]
                        dist_sq = (xi - xj)**2 + (yi - yj)**2
                        return dist_sq - 4 * r**2
                    return dist_con

                constraints.append({'type': 'ineq', 'fun': make_dist_constraint(xi_idx, xj_idx, r_idx)})

        return constraints
