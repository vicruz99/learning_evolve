import numpy as np
from scipy.optimize import minimize
import math

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    Uses a penalty-based optimization with multiple random restarts.
    """
    n_circles = 26
    best_centers = None
    best_radii = None
    max_sum = 0.0

    # Parameters for the optimization
    penalty_scale = 1e6
    
    def evaluate_packing(vars):
        """Calculates the objective (Sum of radii) minus penalties for violations."""
        r = vars[n_circles:]
        x = vars[:n_circles]
        y = vars[n_circles:2*n_circles]
        
        # Base objective: Sum of radii
        sum_radii = np.sum(r)
        
        # Penalty accumulator
        penalty = 0.0
        
        # 1. Boundary Constraints Penalties
        # Circle must be inside [0, 1] x [0, 1]
        # If x - r < 0, penalty; if x + r > 1, penalty
        boundary_violation = np.maximum(0, r - x) + np.maximum(0, r - (1 - x)) + \
                             np.maximum(0, r - y) + np.maximum(0, r - (1 - y))
        penalty += np.sum(boundary_violation)
        
        # 2. Overlap Penalties
        # For each pair, distance < r_i + r_j implies overlap
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                dist = math.sqrt((x[i] - x[j])**2 + (y[i] - y[j])**2)
                min_dist = r[i] + r[j]
                if dist < min_dist:
                    overlap = min_dist - dist
                    penalty += overlap
        
        return sum_radii - penalty_scale * penalty

    # Run multiple restarts to find a better global maximum
    num_restarts = 15
    
    for _ in range(num_restarts):
        # 1. Initialization: Staggered Grid (Hexagonal Pattern)
        # Arrange circles in 5 rows (6, 5, 6, 5, 4 pattern) for 26 circles
        x_init = np.zeros(n_circles)
        y_init = np.zeros(n_circles)
        r_init = np.full(n_circles, 0.05)
        
        idx = 0
        row_counts = [6, 5, 6, 5, 4]
        spacing_x = 0.18
        spacing_y = 0.18 * math.sqrt(3)
        
        for row_idx, count in enumerate(row_counts):
            # Shift odd rows to create staggered pattern
            shift = (spacing_x / 2) if row_idx % 2 == 1 else 0
            
            # Center the row horizontally
            row_width = (count - 1) * spacing_x
            start_x = (1.0 - row_width) / 2 + shift
            
            for col_idx in range(count):
                if idx < n_circles:
                    x_init[idx] = start_x + col_idx * spacing_x
                    y_init[idx] = 0.1 + row_idx * spacing_y
                    idx += 1
        
        # Add small random perturbation to avoid symmetries
        x_init += np.random.uniform(-0.02, 0.02, n_circles)
        y_init += np.random.uniform(-0.02, 0.02, n_circles)
        
        # Combine into a single vector for optimization
        # Variables: [x_0, ..., x_25, y_0, ..., y_25, r_0, ..., r_25]
        initial_vars = np.concatenate([x_init, y_init, r_init])
        
        # Bounds for the variables
        # x, y in [0, 1]
        # r >= 0
        bounds = [(0, 1)] * (2 * n_circles) + [(0, 1)] * n_circles
        
        # 2. Optimization
        # Use 'Nelder-Mead' as it doesn't require gradients and handles non-smooth penalty functions well
        res = minimize(evaluate_packing, initial_vars, method='Nelder-Mead', 
                       bounds=bounds, options={'maxiter': 10000, 'xatol': 1e-7, 'fatol': 1e-7})
        
        # Extract results
        opt_x = res.x[:n_circles]
        opt_y = res.x[n_circles:2*n_circles]
        opt_r = res.x[2*n_circles:]
        
        # Validate and update best
        # Check for NaN or invalid
        if np.isnan(opt_x).any() or np.isnan(opt_r).any() or np.any(opt_r < 0):
            continue
            
        current_sum = np.sum(opt_r)
        
        # Perform a final strict validation check (optional but good for sanity)
        # We rely on the penalty scale to ensure constraints are met
        if current_sum > max_sum:
            max_sum = current_sum
            best_centers = np.column_stack((opt_x, opt_y))
            best_radii = opt_r

    # Final cleanup: ensure no radii are negative or NaN (safety clamp)
    if best_radii is not None:
        best_radii = np.clip(best_radii, 0, None)
        
    return best_centers, best_radii, max_sum
