import numpy as np
from scipy.optimize import minimize

def run_packing():
    """
    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    """
    n = 26
    
    # Objective function: Minimize negative sum of radii (equivalent to maximizing sum)
    def objective(vars):
        # vars layout: [x_0...x_n, y_0...y_n, r_0...r_n]
        r = vars[2*n:]
        return -np.sum(r)
    
    # Precompute indices for pairwise distance constraints
    # Pairs (i, j) with 0 <= i < j < n
    pair_i = []
    pair_j = []
    for i in range(n):
        for j in range(i + 1, n):
            pair_i.append(i)
            pair_j.append(j)
    pair_i = np.array(pair_i)
    pair_j = np.array(pair_j)
    
    # Constraint functions
    # 1. Boundary constraints: 
    #    x_i - r_i >= 0
    #    1 - x_i - r_i >= 0
    #    y_i - r_i >= 0
    #    1 - y_i - r_i >= 0
    def boundary_constraints(vars):
        x = vars[:n]
        y = vars[n:2*n]
        r = vars[2*n:]
        
        # Concatenate all 4n constraints
        c1 = x - r           # x >= r
        c2 = 1.0 - x - r     # x <= 1 - r
        c3 = y - r           # y >= r
        c4 = 1.0 - y - r     # y <= 1 - r
        
        return np.concatenate([c1, c2, c3, c4])

    # 2. Distance constraints:
    #    (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
    def distance_constraints(vars):
        x = vars[:n]
        y = vars[n:2*n]
        r = vars[2*n:]
        
        # Vectorized calculation for all pairs
        dx = x[pair_i] - x[pair_j]
        dy = y[pair_i] - y[pair_j]
        dr = r[pair_i] + r[pair_j]
        
        return dx*dx + dy*dy - dr*dr

    # Define constraints list
    constraints = [
        {'type': 'ineq', 'fun': boundary_constraints},
        {'type': 'ineq', 'fun': distance_constraints}
    ]
    
    # Bounds for variables: x, y in [0, 1], r in [0, 0.5]
    # (r cannot exceed 0.5 as diameter would be > 1)
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
    
    best_sum = -np.inf
    best_solution = None
    
    # Generate initial guesses
    initial_guesses = []
    
    # 1. Random initializations with small radii
    # Using a fixed seed for reproducibility, but trying multiple
    rng = np.random.default_rng(42)
    for _ in range(15):
        centers = rng.random((n, 2))
        # Start with small radii to ensure no initial overlap
        # With random points in unit square, min dist is roughly 0.15, so 0.04 is safe
        radii = np.full(n, 0.04)
        initial_guesses.append(np.concatenate([centers[:, 0], centers[:, 1], radii]))
        
    # 2. Grid-like initialization
    # Try to place circles in a somewhat regular grid to escape random local minima
    # 5x5 grid covers 25, add one
    # Using linspace to distribute evenly
    x_grid = np.linspace(0.1, 0.9, 5)
    y_grid = np.linspace(0.1, 0.9, 6) # 6 rows to fit 26+
    
    grid_centers = []
    idx = 0
    for y in y_grid:
        for x in x_grid:
            if idx < n:
                grid_centers.append([x, y])
                idx += 1
    # Fill remaining if any (though 5x6=30 > 26)
    # If we have fewer, pad with random
    while len(grid_centers) < n:
        grid_centers.append(rng.random(2))
        
    grid_centers = np.array(grid_centers[:n])
    grid_radii = np.full(n, 0.08) # Slightly larger initial guess
    initial_guesses.append(np.concatenate([grid_centers[:, 0], grid_centers[:, 1], grid_radii]))

    # Run optimization for each guess
    for k, vars0 in enumerate(initial_guesses):
        try:
            res = minimize(objective, vars0, method='SLSQP', 
                          constraints=constraints,
                          bounds=bounds,
                          options={'maxiter': 2000, 'ftol': 1e-12})
            
            # Check if the solution is valid and better
            # The optimizer might return a point where constraints are slightly violated due to tolerance
            # We check the objective value (sum of radii)
            current_sum = -res.fun
            
            # Optional: Verify constraints strictly for safety, though SLSQP is usually good
            # We rely on the optimizer's convergence.
            
            if current_sum > best_sum:
                best_sum = current_sum
                best_solution = res.x
        except Exception as e:
            # In case of optimization failure, skip
            pass

    # Fallback solution if optimization fails completely
    if best_solution is None:
        centers = np.array([[i*0.2 + 0.1, j*0.2 + 0.1] for i in range(5) for j in range(5)][:26])
        radii = np.full(26, 0.05)
        best_solution = np.concatenate([centers[:, 0], centers[:, 1], radii])
        best_sum = 26 * 0.05

    # Extract results
    x = best_solution[:n]
    y = best_solution[n:2*n]
    r = best_solution[2*n:]
    
    centers = np.column_stack((x, y))
    
    # Ensure radii are non-negative (should be guaranteed by bounds)
    r = np.maximum(r, 0.0)
    
    return centers, r, best_sum
