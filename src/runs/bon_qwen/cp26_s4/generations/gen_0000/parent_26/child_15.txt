import numpy as np
from scipy.optimize import minimize

def run_packing():
    """
    Solves the circle packing problem for 26 circles in a unit square.
    Maximizes the sum of radii.
    """
    N = 26
    
    # --- Helper functions ---
    
    def unpack_variables(v):
        """Converts the 1D optimization vector to centers and radii arrays."""
        # v is shape (78,)
        # Layout: x0, y0, r0, x1, y1, r1, ...
        centers = np.zeros((N, 2))
        radii = np.zeros(N)
        for i in range(N):
            centers[i] = v[3*i], v[3*i+1]
            radii[i] = v[3*i+2]
        return centers, radii

    def objective(v):
        """Objective function: minimize negative sum of radii."""
        centers, radii = unpack_variables(v)
        return -np.sum(radii)

    def get_constraints():
        """Generates the list of constraint dictionaries for SLSQP."""
        constraints = []
        
        # 1. Boundary Constraints
        # x - r >= 0
        # 1 - x - r >= 0
        # y - r >= 0
        # 1 - y - r >= 0
        
        def boundary_constraint_x_min(v, i):
            c, r = unpack_variables(v)
            return c[i, 0] - r[i]
            
        def boundary_constraint_x_max(v, i):
            c, r = unpack_variables(v)
            return 1.0 - c[i, 0] - r[i]
            
        def boundary_constraint_y_min(v, i):
            c, r = unpack_variables(v)
            return c[i, 1] - r[i]
            
        def boundary_constraint_y_max(v, i):
            c, r = unpack_variables(v)
            return 1.0 - c[i, 1] - r[i]
            
        for i in range(N):
            constraints.append({'type': 'ineq', 'fun': lambda v, i=i: boundary_constraint_x_min(v, i)})
            constraints.append({'type': 'ineq', 'fun': lambda v, i=i: boundary_constraint_x_max(v, i)})
            constraints.append({'type': 'ineq', 'fun': lambda v, i=i: boundary_constraint_y_min(v, i)})
            constraints.append({'type': 'ineq', 'fun': lambda v, i=i: boundary_constraint_y_max(v, i)})
            
        # 2. Non-overlap Constraints
        # dist(i, j) - (r_i + r_j) >= 0
        
        def non_overlap_constraint(v, i, j):
            c, r = unpack_variables(v)
            dist = np.sqrt(np.sum((c[i] - c[j])**2))
            return dist - (r[i] + r[j])
            
        for i in range(N):
            for j in range(i + 1, N):
                constraints.append({'type': 'ineq', 'fun': lambda v, i=i, j=j: non_overlap_constraint(v, i, j)})
                
        return constraints

    # --- Initialization ---
    # Start with a 5x5 grid for 25 circles and place the 26th in a gap.
    # Grid centers at 0.1, 0.3, 0.5, 0.7, 0.9 with radius 0.1 fits perfectly.
    
    initial_v = np.zeros(3 * N)
    
    # Fill 25 circles in grid
    grid_coords = np.array([0.1, 0.3, 0.5, 0.7, 0.9])
    idx = 0
    for i in range(5):
        for j in range(5):
            x = grid_coords[i]
            y = grid_coords[j]
            r = 0.1
            initial_v[3*idx] = x
            initial_v[3*idx+1] = y
            initial_v[3*idx+2] = r
            idx += 1
            
    # Place 26th circle in a gap, e.g., (0.2, 0.2)
    # Distance to nearest grid circle (0.1, 0.1) is sqrt(0.02) approx 0.1414.
    # With grid r=0.1, max r for 26th is 0.0414. Start with small valid r.
    initial_v[3*idx] = 0.2
    initial_v[3*idx+1] = 0.2
    initial_v[3*idx+2] = 0.01 # Small valid radius
    
    # --- Optimization ---
    
    cons = get_constraints()
    
    # Bounds for variables: x, y in [0, 1], r >= 0
    # Actually bounds can be tighter: r <= 0.5, x in [0,1]
    bounds = []
    for i in range(N):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 0.5)) # r (max possible radius is 0.5)
        
    # Run optimizer
    # SLSQP is good for constrained non-linear optimization
    result = minimize(
        objective,
        initial_v,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'ftol': 1e-9, 'maxiter': 1000, 'disp': False}
    )

        centers, radii = unpack_variables(result.x)

        # Fallback to initial valid config if optimization fails
        centers, radii = unpack_variables(initial_v)

    radii = np.maximum(radii, 0.0)
    
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
