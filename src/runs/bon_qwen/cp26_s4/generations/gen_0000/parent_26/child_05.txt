import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Solves the circle packing problem for 26 circles in a unit square.
    Maximizes the sum of radii.
    """
    n = 26
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Precompute indices for vectorized distance calculation
    # We need to compute distance matrix and sum of radii matrix
    # Indices for upper triangle of adjacency matrix to avoid double counting
    # and self-interaction
    rows, cols = np.triu_indices(n, k=1)
    
    def objective(params):
        """
        Objective function: Minimize negative sum of radii + penalties.
        params: array of shape (3*n,)
        Structure: [x1, y1, r1, x2, y2, r2, ...]
        """
        # Reshape parameters
        centers = np.column_stack((params[0::3], params[1::3]))
        radii = params[2::3]
        
        # Main objective: Maximize sum of radii (minimize negative)
        obj_val = -np.sum(radii)
        
        # Penalty term scaling factor
        # High value ensures constraints are respected
        lam = 500.0 
        penalty = 0.0
        
        # 1. Boundary Constraints
        # Circle i must satisfy: r_i <= x_i <= 1-r_i and r_i <= y_i <= 1-r_i
        # Violations: r_i - x_i > 0, x_i + r_i - 1 > 0, etc.
        
        # Left boundary: x >= r  => r - x <= 0
        penalty += np.sum(np.maximum(0.0, radii - centers[:, 0])**2)
        # Right boundary: x <= 1-r => x + r - 1 <= 0
        penalty += np.sum(np.maximum(0.0, centers[:, 0] + radii - 1.0)**2)
        # Bottom boundary: y >= r => r - y <= 0
        penalty += np.sum(np.maximum(0.0, radii - centers[:, 1])**2)
        # Top boundary: y <= 1-r => y + r - 1 <= 0
        penalty += np.sum(np.maximum(0.0, centers[:, 1] + radii - 1.0)**2)
        
        # 2. Overlap Constraints
        # Distance between i and j >= r_i + r_j
        # Violation: (r_i + r_j) - dist > 0
        
        # Extract centers and radii for pairs
        c_i = centers[rows]
        c_j = centers[cols]
        r_i = radii[rows]
        r_j = radii[cols]
        
        # Euclidean distance
        diff = c_i - c_j
        dists = np.sqrt(np.sum(diff**2, axis=1))
        
        # Sum of radii
        sum_r = r_i + r_j
        
        # Overlap violation
        overlap = sum_r - dists
        penalty += np.sum(np.maximum(0.0, overlap)**2)
        
        return obj_val + lam * penalty

    # Bounds for variables
    # x, y in [0, 1]
    # r in [0, 0.5] (radius cannot exceed 0.5 in unit square)
    bounds = []
    for _ in range(n):
        bounds.extend([
            (0.0, 1.0), # x
            (0.0, 1.0), # y
            (0.0, 0.5)  # r
        ])

    # Number of restarts
    num_restarts = 15
    
    for restart in range(num_restarts):
        # Generate initial guess
        if restart < 10:
            # Hexagonal-like initialization
            # Attempt to place circles in a dense grid
            # Estimate optimal radius roughly 0.1
            r_est = 0.10 
            
            # Create a grid of positions
            # We want roughly 5x5 = 25, plus 1
            # Hexagonal packing allows slightly denser placement
            # Row spacing: sqrt(3)/2 * diameter = sqrt(3)/2 * 2r = r*sqrt(3)
            
            centers_init = np.zeros((n, 2))
            count = 0
            
            # Try to fit in a pattern
            # 5 rows
            y_step = r_est * np.sqrt(3)
            for row in range(6):
                y = r_est + row * y_step
                if y + r_est > 1.0:
                    break
                
                # X spacing
                x_step = 2 * r_est
                offset = r_est if row % 2 == 1 else r_est # Hexagonal offset logic usually shifts by r
                # Actually standard hex: row 0 at r, row 1 at r+r...
                # Let's just use simple offset r
                x_offset = r_est if row % 2 == 1 else r_est
                
                # Number of circles in this row
                # Width available approx 1.0
                # Count = floor((1.0 - 2*r_est) / (2*r_est)) + 1 ?
                # With offset, maybe more?
                
                # Let's just fill x
                x = x_offset
                while x <= 1.0 - r_est and count < n:
                    centers_init[count, 0] = x
                    centers_init[count, 1] = y
                    count += 1
                    x += x_step
                
                # If we didn't fill enough, maybe adjust offset or spacing?
                # But for init, simple is fine.
                
            # If we haven't placed all, fill remaining randomly or in gaps
            while count < n:
                # Place randomly in available space?
                # Just random
                centers_init[count, 0] = np.random.uniform(0, 1)
                centers_init[count, 1] = np.random.uniform(0, 1)
                count += 1

            radii_init = np.full(n, r_est * 0.9) # Start slightly smaller
            
        else:
            # Random initialization
            centers_init = np.random.rand(n, 2)
            radii_init = np.full(n, 0.08) # Small radius

        # Flatten to params
        params = np.zeros(n * 3)
        for i in range(n):
            params[3*i] = centers_init[i, 0]
            params[3*i+1] = centers_init[i, 1]
            params[3*i+2] = radii_init[i]
            
        # Optimize
        try:
            res = minimize(objective, params, method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 2000, 'ftol': 1e-12, 'gtol': 1e-12})
            
            # Extract results
            centers_res = res.x[0::3].reshape(n, 2)
            radii_res = res.x[2::3]
            
            # Check validity strictly
            is_valid = True
            
            # Check radii non-negative
            if np.any(radii_res < -1e-9):
                is_valid = False
            
            # Check bounds
            # x - r >= 0
            if np.any(centers_res[:, 0] - radii_res < -1e-9):
                is_valid = False
            # x + r <= 1
            if np.any(centers_res[:, 0] + radii_res > 1.0 + 1e-9):
                is_valid = False
            # y - r >= 0
            if np.any(centers_res[:, 1] - radii_res < -1e-9):
                is_valid = False
            # y + r <= 1
            if np.any(centers_res[:, 1] + radii_res > 1.0 + 1e-9):
                is_valid = False
            
            # Check overlaps
            # dist >= r_i + r_j
            # dist^2 >= (r_i + r_j)^2
            diff = centers_res[:, np.newaxis, :] - centers_res[np.newaxis, :, :]
            dists_sq = np.sum(diff**2, axis=2)
            sum_radii = radii_res[:, np.newaxis] + radii_res[np.newaxis, :]
            sum_radii_sq = sum_radii**2
            
            # Diagonal is 0, but sum_radii_sq > 0, so check off-diagonal
            # Mask diagonal
            mask = ~np.eye(n, dtype=bool)
            violations = (sum_radii_sq - dists_sq)[mask]
            if np.any(violations > 1e-9): # Allow tiny numerical error
                is_valid = False
            
            if is_valid:
                current_sum = np.sum(radii_res)
                if current_sum > best_sum:
                    best_sum = current_sum
                    best_centers = centers_res.copy()
                    best_radii = radii_res.copy()
                    
        except Exception as e:
            continue

    # If no valid solution found (unlikely), fallback to simple grid
    if best_centers is None:
        best_centers = np.zeros((n, 2))
        best_radii = np.full(n, 0.0)
        best_sum = 0.0
        # Fallback: small circles
        for i in range(n):
            r = 0.02
            x = (i % 5) * 0.2 + 0.1
            y = (i // 5) * 0.2 + 0.1
            best_centers[i] = [x, y]
            best_radii[i] = r
        best_sum = np.sum(best_radii)

    return best_centers, best_radii, float(best_sum)
