import numpy as np
from scipy.optimize import minimize
import math

N_CIRCLES = 26

def objective(params):
    radii = params[2 * N_CIRCLES:]
    return -np.sum(radii)


def constraint_boundary(params):
    centers = params[:2 * N_CIRCLES].reshape(N_CIRCLES, 2)
    radii = params[2 * N_CIRCLES:]

    result = np.zeros(4 * N_CIRCLES)
    for i in range(N_CIRCLES):
        x, y = centers[i]
        r = radii[i]
        result[4 * i] = x - r
        result[4 * i + 1] = 1 - x - r
        result[4 * i + 2] = y - r
        result[4 * i + 3] = 1 - y - r
    return result


def constraint_overlap(params):
    centers = params[:2 * N_CIRCLES].reshape(N_CIRCLES, 2)
    radii = params[2 * N_CIRCLES:]

    n_pairs = N_CIRCLES * (N_CIRCLES - 1) // 2
    result = np.zeros(n_pairs)
    idx = 0
    for i in range(N_CIRCLES):
        for j in range(i + 1, N_CIRCLES):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist_sq = dx * dx + dy * dy
            r_sum = radii[i] + radii[j]
            result[idx] = dist_sq - r_sum * r_sum
            idx += 1
    return result


def generate_init(n, seed=42):
    np.random.seed(seed)
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.04)

    idx = 0
    spacing = 0.18
    for row in range(6):
        y = 0.08 + row * spacing * math.sqrt(3) / 2
        for col in range(6):
            if idx >= n:
                break
            x = 0.08 + col * spacing
            if row % 2 == 1:
                x += spacing / 2
            if x < 0.92 and y < 0.92:
                centers[idx] = [x, y]
                idx += 1

    centers += np.random.randn(n, 2) * 0.01
    centers = np.clip(centers, 0.05, 0.95)

    return centers, radii


def run_packing():
    n = N_CIRCLES

    best_centers = None
    best_radii = None
    best_sum = 0.0

    bounds = [(0.01, 0.99) for _ in range(2 * n)]
    bounds += [(0.01, 0.5) for _ in range(n)]

    cons = [
        {'type': 'ineq', 'fun': constraint_boundary},
        {'type': 'ineq', 'fun': constraint_overlap}
    ]

    for seed in range(30):
        centers, radii = generate_init(n, seed=seed)
        params = np.concatenate([centers.flatten(), radii])

        try:
            result = minimize(
                objective, params, method='SLSQP',
                bounds=bounds, constraints=cons,
                options={'maxiter': 15000, 'ftol': 1e-15}
            )

            current_sum = -result.fun
            if current_sum > best_sum:
                best_sum = current_sum
                best_centers = result.x[:2 * n].reshape(n, 2).copy()
                best_radii = result.x[2 * n:].copy()
        except Exception:
            continue

    if best_centers is None:
        centers, radii = generate_init(n, seed=0)
        best_centers = centers
        best_radii = radii
        best_sum = np.sum(radii)

    best_radii = np.maximum(best_radii, 0.001)

    return best_centers, best_radii, float(np.sum(best_radii))
