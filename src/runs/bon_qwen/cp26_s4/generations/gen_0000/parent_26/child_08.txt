import numpy as np
from scipy.optimize import minimize

def compute_loss(z, mu, n=26):
    """Computes the penalized objective function."""
    centers = z[:2 * n].reshape(n, 2)
    radii = z[2 * n:]
    
    # Primary objective: maximize sum of radii (minimize negative sum)
    loss = -np.sum(radii)
    
    # Boundary penalties: circles must stay within [0, 1]x[0, 1]
    loss += mu * np.sum(np.maximum(0, radii - centers[:, 0])**2)
    loss += mu * np.sum(np.maximum(0, centers[:, 0] + radii - 1)**2)
    loss += mu * np.sum(np.maximum(0, radii - centers[:, 1])**2)
    loss += mu * np.sum(np.maximum(0, centers[:, 1] + radii - 1)**2)
    
    # Overlap penalties: circles must not intersect
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.linalg.norm(diff, axis=2)
    rad_sums = radii[:, np.newaxis] + radii[np.newaxis, :]
    
    overlaps = np.maximum(0, rad_sums - dists)
    np.fill_diagonal(overlaps, 0.0)
    
    loss += mu * np.sum(overlaps**2)
    return loss

def get_initial_config(n=26):
    """Generates a structured initial placement (5x5 grid + center) with noise."""
    x = np.linspace(0.125, 0.875, 5)
    grid_x, grid_y = np.meshgrid(x, x)
    pts = np.vstack([grid_x.ravel(), grid_y.ravel()]).T
    pts = np.vstack([pts, [[0.5, 0.5]]])  # 26th circle in the center
    
    # Add controlled random perturbation
    np.random.seed(42)
    pts += np.random.randn(*pts.shape) * 0.015
    
    radii = np.ones(n) * 0.085
    return np.concatenate([pts.flatten(), radii])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """Optimizes circle packing and returns the best valid configuration."""
    n = 26
    best_z = None
    best_sum_r = -1.0
    
    # Variable bounds: x,y in [0,1], r in [0, 0.5]
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
    
    # Continuation schedule for penalty parameter
    mus = [200, 2000, 20000, 200000]
    
    # Run multiple optimizations with different seeds
    for seed in range(5):
        np.random.seed(seed)
        z0 = get_initial_config(n)
        # Additional perturbation per run
        z0 += np.random.randn(*z0.shape) * 0.008
        
        z_curr = z0.copy()
        
        # Continuation optimization
        for mu in mus:
            res = minimize(
                compute_loss, 
                z_curr, 
                args=(mu, n), 
                method='L-BFGS-B', 
                bounds=bounds, 
                options={'maxiter': 3000, 'ftol': 1e-12}
            )
            z_curr = res.x
            
        radii = z_curr[2 * n:]
        s_r = np.sum(radii)
        
        # Post-processing: ensure strict validity by shrinking if necessary
        centers = z_curr[:2 * n].reshape(n, 2)
        diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
        dists = np.linalg.norm(diff, axis=2)
        rad_sums = radii[:, np.newaxis] + radii[np.newaxis, :]
        
        overlaps = rad_sums - dists
        np.fill_diagonal(overlaps, 0.0)
        max_ov = np.max(overlaps)
        
        if max_ov > 1e-10:
            shrink_amount = max_ov + 1e-12
            radii = np.maximum(0.0, radii - shrink_amount)
            s_r = np.sum(radii)
            
        if s_r > best_sum_r:
            best_sum_r = s_r
            best_z = z_curr.copy()
            
    # Extract best result
    centers = best_z[:2 * n].reshape(n, 2)
    radii = best_z[2 * n:]
    
    return centers, radii, np.sum(radii)
