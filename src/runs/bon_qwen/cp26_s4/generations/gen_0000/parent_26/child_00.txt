import numpy as np
from scipy.optimize import minimize


def generate_hexagonal_initial(n, r_init, perturb=0.0):
    """Generate initial centers in hexagonal pattern."""
    centers = np.zeros((n, 2))
    idx = 0
    
    y_spacing = r_init * np.sqrt(3)
    
    for row in range(8):
        y = r_init + row * y_spacing
        if y + r_init > 1.0:
            break
        
        x_offset = r_init + (r_init if row % 2 == 1 else 0)
        x = x_offset
        
        while x + r_init <= 1.0 and idx < n:
            centers[idx] = [
                x + (np.random.rand() - 0.5) * perturb,
                y + (np.random.rand() - 0.5) * perturb
            ]
            idx += 1
            x += 2 * r_init
    
    # Fill remaining with grid
    while idx < n:
        row = idx // 5
        col = idx % 5
        centers[idx] = [
            0.1 + col * 0.18 + (np.random.rand() - 0.5) * perturb,
            0.1 + row * 0.18 + (np.random.rand() - 0.5) * perturb
        ]
        idx += 1
    
    return centers


def generate_random_initial(n, r_init, seed):
    """Generate random initial configuration."""
    np.random.seed(seed)
    centers = np.random.uniform(0.15, 0.85, (n, 2))
    radii = np.full(n, r_init)
    return centers, radii


def objective_slsqp(x, n):
    """Objective: minimize negative sum of radii."""
    radii = x[2*n:]
    return -np.sum(radii)


def create_constraints(n):
    """Create boundary and non-overlap constraints."""
    constraints = []
    
    # Boundary constraints for each circle
    for i in range(n):
        xi = 2 * i
        yi = 2 * i + 1
        ri = 2 * n + i
        
        # x_i >= r_i
        constraints.append({
            'type': 'ineq',
            'fun': lambda x, i=i: x[2 * i] - x[2 * n + i]
        })
        # 1 - x_i >= r_i
        constraints.append({
            'type': 'ineq',
            'fun': lambda x, i=i: 1.0 - x[2 * i] - x[2 * n + i]
        })
        # y_i >= r_i
        constraints.append({
            'type': 'ineq',
            'fun': lambda x, i=i: x[2 * i + 1] - x[2 * n + i]
        })
        # 1 - y_i >= r_i
        constraints.append({
            'type': 'ineq',
            'fun': lambda x, i=i: 1.0 - x[2 * i + 1] - x[2 * n + i]
        })
    
    # Non-overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({
                'type': 'ineq',
                'fun': lambda x, i=i, j=j: 
                    np.sqrt((x[2*i] - x[2*j])**2 + (x[2*i+1] - x[2*j+1])**2) - x[2*n+i] - x[2*n+j]
            })
    
    return constraints


def penalty_objective(x, n, weight=1e6):
    """Penalty-based objective for L-BFGS-B."""
    centers = x[:2*n].reshape((n, 2))
    radii = x[2*n:]
    
    obj = -np.sum(radii)
    penalty = 0.0
    
    # Boundary violations
    for i in range(n):
        cx, cy = centers[i]
        r = radii[i]
        
        # Left
        v = r - cx
        if v > 0:
            penalty += v * v
        # Right
        v = cx + r - 1.0
        if v > 0:
            penalty += v * v
        # Bottom
        v = r - cy
        if v > 0:
            penalty += v * v
        # Top
        v = cy + r - 1.0
        if v > 0:
            penalty += v * v
    
    # Overlap violations
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            d = np.sqrt(dx * dx + dy * dy)
            v = radii[i] + radii[j] - d
            if v > 0:
                penalty += v * v
    
    return obj + weight * penalty


def refine_with_lbfgsb(centers, radii, n, weight=1e6, maxiter=50000):
    """Refine solution using L-BFGS-B with penalty."""
    x0 = np.concatenate([centers.flatten(), radii])
    
    bounds = []
    for i in range(n):
        bounds.append((0.0001, 0.9999))  # x
        bounds.append((0.0001, 0.9999))  # y
    for i in range(n):
        bounds.append((0.0001, 0.3))  # radius
    
    result = minimize(
        penalty_objective,
        x0,
        method='L-BFGS-B',
        args=(n, weight),
        bounds=bounds,
        options={
            'maxiter': maxiter,
            'ftol': 1e-15,
            'gtol': 1e-10
        }
    )
    
    centers_opt = result.x[:2*n].reshape((n, 2))
    radii_opt = np.maximum(result.x[2*n:], 0.0001)
    
    return centers_opt, radii_opt
