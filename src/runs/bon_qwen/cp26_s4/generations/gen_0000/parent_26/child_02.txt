import numpy as np
from scipy.optimize import minimize

def run_packing():
    """
    Solves the circle packing problem for 26 circles in a unit square.
    Maximizes the sum of radii.
    """
    N = 26
    
    # Helper to flatten/unflatten variables
    def unpack(v):
        centers = np.zeros((N, 2))
        radii = np.zeros(N)
        for i in range(N):
            centers[i, 0] = v[3*i]
            centers[i, 1] = v[3*i + 1]
            radii[i] = v[3*i + 2]
        return centers, radii

    def pack(centers, radii):
        v = np.zeros(3 * N)
        for i in range(N):
            v[3*i] = centers[i, 0]
            v[3*i + 1] = centers[i, 1]
            v[3*i + 2] = radii[i]
        return v

    # Initialization: Hexagonal grid
    def initialize_hex_grid(n_circles):
        # Estimate radius to fit roughly
        # Area density approx pi/sqrt(12) ~ 0.9069
        # n * pi * r^2 <= 1 => r <= sqrt(1 / (n * pi))
        r_est = np.sqrt(1 / (n_circles * np.pi))
        # Hexagonal spacing
        dx = 2 * r_est
        dy = np.sqrt(3) * r_est
        
        centers = []
        y = r_est
        row_idx = 0
        
        while len(centers) < n_circles:
            x_offset = 0 if row_idx % 2 == 0 else r_est
            x = r_est + x_offset
            
            while x < 1 - r_est and len(centers) < n_circles:
                # Check bounds
                if x - r_est >= 0 and x + r_est <= 1 and y - r_est >= 0 and y + r_est <= 1:
                    centers.append([x, y])
                x += dx
            
            y += dy
            row_idx += 1
            
        return np.array(centers), r_est * 0.9 # Slightly shrink to be safe

    centers_init, r_init = initialize_hex_grid(N)
    
    # If we didn't get enough circles (unlikely with logic above, but safe fallback), pad with random
    if len(centers_init) < N:
        pad_centers = np.random.uniform(0.2, 0.8, (N - len(centers_init), 2))
        centers_init = np.vstack([centers_init, pad_centers])
        r_init = 0.05

    x0 = pack(centers_init, np.full(N, r_init))

    # Objective function
    def objective(v):
        _, radii = unpack(v)
        return -np.sum(radii)

    # Constraints
    # 1. Boundary constraints: r_i <= x_i, r_i <= 1-x_i, etc.
    # Format: func(v) >= 0
    def constraint_boundary(v):
        centers, radii = unpack(v)
        con = []
        for i in range(N):
            x, y = centers[i]
            r = radii[i]
            con.append(x - r)       # x - r >= 0
            con.append((1 - x) - r) # 1 - x - r >= 0
            con.append(y - r)       # y - r >= 0
            con.append((1 - y) - r) # 1 - y - r >= 0
        return np.array(con)

    # 2. Non-overlap constraints: dist(i, j) >= r_i + r_j
    # dist - (r_i + r_j) >= 0
    def constraint_overlap(v):
        centers, radii = unpack(v)
        con = []
        for i in range(N):
            for j in range(i + 1, N):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                con.append(dist - (radii[i] + radii[j]))
        return np.array(con)

    # Bounds
    bounds = []
    for i in range(N):
        bounds.extend([(0, 1), (0, 1), (0, 0.5)]) # x, y in [0,1], r in [0, 0.5]

    # Define constraints for optimizer
    cons = [
        {'type': 'ineq', 'fun': constraint_boundary},
        {'type': 'ineq', 'fun': constraint_overlap}
    ]

    # Optimization
    # Run 1: Coarse
    res1 = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, 
                    options={'maxiter': 200, 'ftol': 1e-6, 'disp': False})
    
    # Run 2: Refine
    res2 = minimize(objective, res1.x, method='SLSQP', bounds=bounds, constraints=cons, 
                    options={'maxiter': 500, 'ftol': 1e-9, 'disp': False})

    # Final extraction
    centers_opt, radii_opt = unpack(res2.x)
    sum_radii = np.sum(radii_opt)

    # Post-processing to ensure strict validity (handle numerical precision)
    # Shrink radii slightly if any overlap or boundary violation detected
    # This is a safety measure. Ideally, optimizer satisfies constraints.
    valid = False
    safety_shrink = 1.0
    while not valid and safety_shrink > 0.99:
        radii_temp = radii_opt * safety_shrink
        # Quick validation check logic
        is_valid = True
        # Boundary
        for i in range(N):
            x, y = centers_opt[i]
            r = radii_temp[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                is_valid = False
                break
        # Overlap
        if is_valid:
            for i in range(N):
                for j in range(i + 1, N):
                    dist = np.sqrt(np.sum((centers_opt[i] - centers_opt[j])**2))
                    if dist < radii_temp[i] + radii_temp[j] - 1e-12:
                        is_valid = False
                        break
                if not is_valid: break
            
        if is_valid:
            valid = True
            radii_opt = radii_temp
            sum_radii = np.sum(radii_opt)
        else:
            safety_shrink -= 0.0001 # Shrink by 0.01%

    return centers_opt, radii_opt, sum_radii
