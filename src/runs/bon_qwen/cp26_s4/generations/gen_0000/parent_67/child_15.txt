import numpy as np
from scipy.optimize import minimize

def compute_obj(vars_, N, lam):
    """Objective function: minimize -sum(radii) + penalty for constraint violations"""
    c = vars_[:2*N].reshape(N, 2)
    r = vars_[2*N:]
    pen = 0.0
    
    for i in range(N):
        x, y, ri = c[i,0], c[i,1], r[i]
        # Boundary penalties
        for b in [x, 1-x, y, 1-y]:
            ov = ri - b
            if ov > 0:
                pen += ov**2
                
        # Pairwise overlap penalties
        for j in range(i+1, N):
            d = np.sqrt((c[i,0]-c[j,0])**2 + (c[i,1]-c[j,1])**2)
            ov = ri + r[j] - d
            if ov > 0:
                pen += ov**2
                
    return -np.sum(r) + lam * pen

def compute_grad(vars_, N, lam):
    """Analytical gradient of the objective function"""
    c = vars_[:2*N].reshape(N, 2)
    r = vars_[2*N:]
    grad = np.zeros_like(vars_)

    # Pairwise gradients
    for i in range(N):
        for j in range(i+1, N):
            dx = c[i,0] - c[j,0]
            dy = c[i,1] - c[j,1]
            d = np.sqrt(dx*dx + dy*dy)
            ov = r[i] + r[j] - d
            if ov > 0:
                term = 2.0 * ov * lam
                inv_d = 1.0 / (d + 1e-12)
                # Gradients w.r.t centers
                grad[2*i]   -= term * dx * inv_d
                grad[2*i+1] -= term * dy * inv_d
                grad[2*j]   += term * dx * inv_d
                grad[2*j+1] += term * dy * inv_d
                # Gradients w.r.t radii
                grad[2*N+i] += 2.0 * ov * lam
                grad[2*N+j] += 2.0 * ov * lam

        x, y, ri = c[i,0], c[i,1], r[i]
        # Objective gradient: d(-sum(r))/dr = -1
        grad[2*N+i] -= 1.0

        # Boundary gradients
        # Left: x >= r  => penalty (r-x)^2  => d/dr=+2(r-x), d/dx=-2(r-x)
        ov = ri - x
        if ov > 0:
            term = 2.0 * ov * lam
            grad[2*i] -= term
            grad[2*N+i] += term

        # Right: 1-x >= r => penalty (r-(1-x))^2 => d/dr=+2ov, d/dx=+2ov
        ov = ri - (1-x)
        if ov > 0:
            term = 2.0 * ov * lam
            grad[2*i] += term
            grad[2*N+i] += term

        # Bottom: y >= r => penalty (r-y)^2 => d/dr=+2ov, d/dy=-2ov
        ov = ri - y
        if ov > 0:
            term = 2.0 * ov * lam
            grad[2*i+1] -= term
            grad[2*N+i] += term

        # Top: 1-y >= r => penalty (r-(1-y))^2 => d/dr=+2ov, d/dy=+2ov
        ov = ri - (1-y)
        if ov > 0:
            term = 2.0 * ov * lam
            grad[2*i+1] += term
            grad[2*N+i] += term
            
    return grad

def run_packing():
    N = 26
    lam = 500.0  # Penalty weight

    # 1. Initialize on a hexagonal grid
    centers = np.zeros((N, 2))
    idx = 0
    s = 0.14  # Grid spacing
    for i in range(6):
        for j in range(5):
            if idx < N:
                x = 0.08 + j * s + (i % 2) * s / 2
                y = 0.08 + i * s * np.sqrt(3) / 2
                centers[idx] = [x, y]
                idx += 1

    radii = np.full(N, 0.04)
    x0 = np.concatenate([centers.ravel(), radii])

    # Bounds: x,y in [0,1], r in [0, 0.5]
    bounds = [(0.0, 1.0) if k % 3 != 2 else (0.0, 0.5) for k in range(3*N)]

    # 2. Optimize positions and radii
    res = minimize(compute_obj, x0, jac=compute_grad, args=(N, lam), method='L-BFGS-B',

    centers_opt = res.x[:2*N].reshape(N, 2)
    radii_opt = res.x[2*N:]

    # 3. Repair step: Project radii to strict feasibility based on final centers
    for i in range(N):
        max_r = min(centers_opt[i,0], 1-centers_opt[i,0], centers_opt[i,1], 1-centers_opt[i,1])

                dist = np.sqrt(np.sum((centers_opt[i] - centers_opt[j])**2))
                if dist / 2.0 < max_r:
                    max_r = dist / 2.0
        radii_opt[i] = max_r

    # Ensure centers are strictly inside bounds
    centers_opt = np.clip(centers_opt, 0.0, 1.0)
    
    return centers_opt, radii_opt, float(np.sum(radii_opt))
