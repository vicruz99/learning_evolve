import numpy as np
from scipy.optimize import minimize


def neg_sum_radii(params):
    n = len(params) // 3
    radii = params[2 * n:]
    return -np.sum(radii)


def boundary_constraint(params, n, i, bt):
    centers = params[:2 * n].reshape(n, 2)
    radii = params[2 * n:]
    if bt == 0:
        return centers[i, 0] - radii[i]
    elif bt == 1:
        return 1.0 - centers[i, 0] - radii[i]
    elif bt == 2:
        return centers[i, 1] - radii[i]
    else:
        return 1.0 - centers[i, 1] - radii[i]


def overlap_constraint(params, n, i, j):
    centers = params[:2 * n].reshape(n, 2)
    radii = params[2 * n:]
    dx = centers[i, 0] - centers[j, 0]
    dy = centers[i, 1] - centers[j, 1]
    dist = np.sqrt(dx * dx + dy * dy)
    return dist - radii[i] - radii[j]


def make_initial_hexagonal(n, row_counts, r_init):
    centers = np.zeros((n, 2))
    radii = np.ones(n) * r_init

    idx = 0
    num_rows = len(row_counts)
    h = r_init * np.sqrt(3.0)
    w = 2.0 * r_init

    total_height = (num_rows - 1) * h + 2 * r_init
    y_start = (1.0 - total_height) / 2.0

    for row in range(num_rows):
        count = row_counts[row]
        total_width = (count - 1) * w + 2 * r_init
        x_start = (1.0 - total_width) / 2.0

        for col in range(count):
            x = x_start + col * w
            y = y_start + row * h
            centers[idx] = [x, y]
            idx += 1

    return centers, radii


def optimize_with_slsqp(centers, radii, n):
    params = np.concatenate([centers.flatten(), radii])

    constraints = []

    for i in range(n):
        for bt in range(4):
            constraints.append({
                'type': 'ineq',
                'fun': boundary_constraint,
                'args': (n, i, bt)
            })

    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({
                'type': 'ineq',
                'fun': overlap_constraint,
                'args': (n, i, j)
            })

    bounds = []
    for i in range(2 * n):
        bounds.append((0.0, 1.0))
    for i in range(n):
        bounds.append((0.0, 0.5))

    result = minimize(
        neg_sum_radii,
        params,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 5000, 'ftol': 1e-15, 'disp': False}
    )

    centers_opt = result.x[:2 * n].reshape(n, 2)
    radii_opt = result.x[2 * n:]
    radii_opt = np.maximum(radii_opt, 0)

    for i in range(n):
        r = radii_opt[i]
        centers_opt[i, 0] = np.clip(centers_opt[i, 0], r, 1.0 - r)
        centers_opt[i, 1] = np.clip(centers_opt[i, 1], r, 1.0 - r)

    return centers_opt, radii_opt


def jiggle_optimize(centers, radii, n, iterations=5000):
    centers = centers.copy()
    radii = radii.copy()

    for iteration in range(iterations):
        improved = False
        grow_step = 0.0005

        for i in range(n):
            current_r = radii[i]
            new_r = current_r + grow_step

            test_centers = centers.copy()
            test_radii = radii.copy()
            test_radii[i] = new_r

            # Check boundary
            valid = True
            x, y = test_centers[i]
            if x - new_r < -1e-10 or x + new_r > 1.0 + 1e-10:
                valid = False
            if y - new_r < -1e-10 or y + new_r > 1.0 + 1e-10:
                valid = False

            if not valid:
                continue

            # Resolve overlaps
            for push_iter in range(50):
                max_violation = 0
                for j in range(n):
                    if i == j:
                        continue
                    dx = test_centers[i, 0] - test_centers[j, 0]
                    dy = test_centers[i, 1] - test_centers[j, 1]
                    dist = np.sqrt(dx * dx + dy * dy)
                    needed = test_radii[i] + test_radii[j]
                    if dist < needed - 1e-12:
                        violation = needed - dist
                        if violation > max_violation:
                            max_violation = violation

                if max_violation < 1e-12:
                    break

                # Push circles apart
                for j in range(n):
                    if i == j:
                        continue
                    dx = test_centers[i, 0] - test_centers[j, 0]
                    dy = test_centers[i, 1] - test_centers[j, 1]
                    dist = np.sqrt(dx * dx + dy * dy)
                    needed = test_radii[i] + test_radii[j]
                    if dist < needed - 1e-12 and dist > 1e-15:
                        push_frac = (needed - dist) / dist * 0.5
                        test_centers[i, 0] += push_frac * dx
                        test_centers[i, 1] += push_frac * dy
                        test_centers[j, 0] -= push_frac * dx
                        test_centers[j, 1] -= push_frac * dy

                # Project to boundary
                for j in range(n):
                    r = test_radii[j]
                    test_centers[j, 0] = np.clip(test_centers[j, 0], r, 1.0 - r)
                    test_centers[j, 1] = np.clip(test_centers[j, 1], r, 1.0 - r)

            # Final check
            valid = True
            for j in range(n):
                if i == j:
                    continue
                dx = test_centers[i, 0] - test_centers[j, 0]
                dy = test_centers[i, 1] - test_centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                if dist < test_radii[i] + test_radii[j] - 1e-10:
                    valid = False
                    break

            if valid:
                radii[i] = new_r
                centers = test_centers.copy()
                improved = True

        if not improved and iteration > 100:
            break

    return centers, radii
