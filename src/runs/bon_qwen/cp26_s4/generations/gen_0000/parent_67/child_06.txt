def run_packing():

    Pack 26 circles in a unit square to maximize the sum of radii.
    Uses a repulsion-based physical simulation with gradual radius expansion.

    n = 26

    x = np.linspace(0.2, 0.8, 5)
    y = np.linspace(0.2, 0.8, 5)

    centers += np.random.uniform(-0.05, 0.05, centers.shape)
    centers = np.clip(centers, 0.05, 0.95)
    
    radii = np.full(n, 0.02)
    
    # 2. Simulation parameters
    dt = 0.003
    rep_k = 150.0
    max_iter = 25000
    
    # 3. Repulsion dynamics loop
    for it in range(max_iter):
        # Gradually expand radii
        radii += 0.000025
        
        forces = np.zeros((n, 2))
        
        # Boundary repulsion
        left_ov = np.maximum(0, radii - centers[:, 0])
        forces[:, 0] += rep_k * left_ov
        right_ov = np.maximum(0, radii - (1 - centers[:, 0]))
        forces[:, 0] -= rep_k * right_ov
        bottom_ov = np.maximum(0, radii - centers[:, 1])
        forces[:, 1] += rep_k * bottom_ov
        top_ov = np.maximum(0, radii - (1 - centers[:, 1]))
        forces[:, 1] -= rep_k * top_ov
        
        # Pairwise repulsion (vectorized)
        diff = centers[:, None, :] - centers[None, :, :]  # Shape: (n, n, 2)
        dist = np.sqrt(np.sum(diff**2, axis=2)) + 1e-8    # Shape: (n, n)
        
        min_dist = radii[:, None] + radii[None, :]        # Shape: (n, n)
        overlap = np.maximum(0, min_dist - dist)          # Shape: (n, n)
        
        # Direction vectors (avoid division by zero)
        norm = diff / dist[..., None]                     # Shape: (n, n, 2)
        force_mag = overlap * rep_k                       # Shape: (n, n)
        forces_vec = norm * force_mag[..., None]          # Shape: (n, n, 2)
        
        # Accumulate forces
        forces += np.sum(forces_vec, axis=1)               # Shape: (n, 2)
        
        # Update positions
        centers += forces * dt
        centers = np.clip(centers, 0, 1)
        
    # 4. Final projection: compute exact maximum valid radius
    min_clearance = 1.0
    for i in range(n):
        # Distance to boundaries
        min_clearance = min(min_clearance,

        for j in range(i + 1, n):

            min_clearance = min(min_clearance, d / 2)
            
    # Apply safety margin for floating point tolerance in validator
    final_radii = np.full(n, min_clearance * 0.9995)
    
    return centers, final_radii, np.sum(final_radii)
