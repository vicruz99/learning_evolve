    growth_factor = 1.002 # Multiplicative increase

    def calculate_energy(x_flat, radii, n):
        centers = x_flat.reshape((n, 2))
        energy = 0.0

            if x < r:
                energy += (r - x)**2
            elif x > 1 - r:
                energy += (x - (1 - r))**2
            
            # y-direction
            if y < r:
                energy += (r - y)**2
            elif y > 1 - r:
                energy += (y - (1 - r))**2
