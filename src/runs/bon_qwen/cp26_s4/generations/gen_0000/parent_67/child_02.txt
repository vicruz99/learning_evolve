def check_validity(x_vars):
    """Helper to check if variable vector is valid"""
    n = 26
    centers = x_vars[0::3].reshape(-1, 2)
    radii = x_vars[2::3]
    
    if np.isnan(centers).any() or np.isnan(radii).any():
        return False
    if np.any(radii < 0):
        return False

        r = radii[i]
        if x - r < -1e-5 or x + r > 1 + 1e-5 or y - r < -1e-5 or y + r > 1 + 1e-5:
            return False

            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-5:
                return False
    return True
