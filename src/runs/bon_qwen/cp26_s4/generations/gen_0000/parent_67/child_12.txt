import numpy as np


def run_packing():
    n = 26
    best_sum = -1.0
    best_centers = None
    best_radii = None

    # Define multiple row configurations to try
    config_options = [
        [5, 5, 5, 5, 6],
        [6, 5, 5, 5, 5],
        [5, 6, 5, 5, 5],
        [5, 5, 6, 5, 5],
        [5, 5, 5, 6, 5],
        [4, 6, 6, 6, 4],
        [6, 6, 6, 6, 2],
        [3, 6, 7, 6, 4],
    ]

    for config_idx, row_counts in enumerate(config_options):
        if sum(row_counts) != n:
            continue

        for trial in range(8):
            centers = np.zeros((n, 2))
            radii = np.full(n, 0.04)

            # Hexagonal initialization with variation
            base_spacing_x = 0.18 + trial * 0.008
            spacing_x = base_spacing_x
            spacing_y = spacing_x * np.sqrt(3) / 2 * (0.95 + trial * 0.01)

            idx = 0
            for row, count in enumerate(row_counts):
                y = 0.08 + row * spacing_y
                x_offset = spacing_x / 2 if row % 2 == 1 else 0.02 + trial * 0.003
                for col in range(count):
                    x = x_offset + col * spacing_x
                    x = np.clip(x, 0.05, 0.95)
                    y_val = np.clip(y, 0.05, 0.95)
                    centers[idx] = [x, y_val]
                    idx += 1

            # Force-directed optimization
            lr = 0.008
            for step in range(2500):
                forces = np.zeros((n, 2))

                for i in range(n):
                    # Boundary repulsion
                    for d in range(2):
                        if centers[i, d] < radii[i] + 1e-8:
                            forces[i, d] += (radii[i] + 1e-8 - centers[i, d]) * 3000.0
                        if centers[i, d] > 1.0 - radii[i] - 1e-8:
                            forces[i, d] -= (centers[i, d] - (1.0 - radii[i] - 1e-8)) * 3000.0

                    # Circle-circle repulsion
                    for j in range(i + 1, n):
                        dx = centers[i, 0] - centers[j, 0]
                        dy = centers[i, 1] - centers[j, 1]
                        dist_sq = dx * dx + dy * dy
                        if dist_sq < 1e-20:
                            dist = 1e-10
                            dx, dy = 1e-10, 0.0
                        else:
                            dist = np.sqrt(dist_sq)

                        min_dist = radii[i] + radii[j]
                        if dist < min_dist + 0.02:
                            overlap = max(0.0, min_dist - dist)
                            if overlap > 0:
                                strength = overlap * 8000.0 / (dist + 0.001)
                                forces[i, 0] += dx * strength
                                forces[i, 1] += dy * strength
                                forces[j, 0] -= dx * strength
                                forces[j, 1] -= dy * strength

                # Update centers
                centers = centers + forces * lr

                # Clip to valid region
                for i in range(n):
                    centers[i, 0] = np.clip(centers[i, 0], radii[i] + 1e-10, 1.0 - radii[i] - 1e-10)
                    centers[i, 1] = np.clip(centers[i, 1], radii[i] + 1e-10, 1.0 - radii[i] - 1e-10)

                # Adaptive radius growth
                for i in range(n):
                    # Compute max possible radius for this circle
                    max_r = min(
                        centers[i, 0],
                        1.0 - centers[i, 0],
                        centers[i, 1],
                        1.0 - centers[i, 1]
                    )
                    for j in range(n):
                        if i != j:
                            d = np.sqrt(
                                (centers[i, 0] - centers[j, 0]) ** 2 +
                                (centers[i, 1] - centers[j, 1]) ** 2
                            )
                            max_r = min(max_r, d - radii[j])
                    max_r = max(0.0, max_r)

                    # Grow towards max with exponential moving average
                    radii[i] = radii[i] * 0.96 + max_r * 0.04

            current_sum = np.sum(radii)
            if current_sum > best_sum:
                best_sum = current_sum
                best_centers = centers.copy()
                best_radii = radii.copy()

    # Final safety clip
    for i in range(n):
        best_centers[i, 0] = np.clip(best_centers[i, 0], best_radii[i], 1.0 - best_radii[i])
        best_centers[i, 1] = np.clip(best_centers[i, 1], best_radii[i], 1.0 - best_radii[i])

    return best_centers, best_radii, np.sum(best_radii)
