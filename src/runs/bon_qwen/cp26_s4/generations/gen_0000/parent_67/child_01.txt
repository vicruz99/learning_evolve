import numpy as np
from scipy.optimize import minimize


def hexagonal_init(n):
    """Initialize circles in hexagonal pattern"""
    centers = np.zeros((n, 2))
    idx = 0
    y = 0.10
    row = 0
    while idx < n:
        if row % 2 == 0:
            x_positions = np.linspace(0.10, 0.90, 6)
        else:
            x_positions = np.linspace(0.15, 0.85, 5)
        for x in x_positions:
            if idx < n:
                centers[idx] = [x, y]
                idx += 1
        y += 0.18
        row += 1
    return centers[:n]


def force_optimize(centers, radii, n, iterations=50000, random_jitter=0.0):
    """Force-based simulation to optimize circle positions and grow radii"""
    centers = centers.copy()
    radii = radii.copy()
    
    if random_jitter > 0:
        jitter = np.random.uniform(-random_jitter, random_jitter, centers.shape)
        centers = np.clip(centers + jitter, 0.05, 0.95)
    
    for iteration in range(iterations):
        forces = np.zeros((n, 2))
        
        # Circle-circle repulsion
        for i in range(n):
            for j in range(i + 1, n):
                diff = centers[i] - centers[j]
                dist_sq = np.sum(diff ** 2)
                dist = np.sqrt(dist_sq)
                min_dist = radii[i] + radii[j]
                
                if dist < min_dist and dist > 1e-10:
                    overlap = min_dist - dist
                    force = overlap * diff / dist
                    forces[i] += force
                    forces[j] -= force
        
        # Boundary repulsion
        for i in range(n):
            for dim in [0, 1]:
                if centers[i, dim] < radii[i] + 1e-9:
                    forces[i, dim] += (radii[i] + 1e-9 - centers[i, dim]) * 3.0
                if centers[i, dim] > 1 - radii[i] - 1e-9:
                    forces[i, dim] -= (centers[i, dim] - (1 - radii[i] - 1e-9)) * 3.0
        
        # Apply forces with cooling
        step = 0.015 * max(0.05, 1 - iteration / iterations)
        centers += step * forces
        
        # Clamp to valid region
        for i in range(n):
            centers[i, 0] = np.clip(centers[i, 0], radii[i] + 1e-9, 1 - radii[i] - 1e-9)
            centers[i, 1] = np.clip(centers[i, 1], radii[i] + 1e-9, 1 - radii[i] - 1e-9)

        if iteration % 50 == 0:

                    d = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                    clearance = d - radii[i] - radii[j]

                        min_clearance = clearance

                    clearance = min(centers[i, dim], 1 - centers[i, dim]) - radii[i]

                        min_clearance = clearance
            
            if min_clearance > 0.002:
                radii *= (1 + 0.0008)
            elif min_clearance > 0.0005:
                radii *= (1 + 0.0002)
    
    return centers, radii


def sliqp_finetune(centers, radii, n, max_iter=500):
    """Fine-tune using SLSQP optimization"""
    params = np.concatenate([centers.flatten(), radii])
    
    def obj(p):
        c = p[:2*n].reshape(n, 2)
        r = p[2*n:]
        return -np.sum(r)
    
    def con(p):
        c = p[:2*n].reshape(n, 2)
        r = p[2*n:]
        cons = []
        # Circle-circle
        for i in range(n):
            for j in range(i + 1, n):
                d = np.sqrt(np.sum((c[i] - c[j]) ** 2))
                cons.append(d - r[i] - r[j])
        # Boundary
        for i in range(n):
            cons.append(c[i, 0] - r[i])
            cons.append(1 - c[i, 0] - r[i])
            cons.append(c[i, 1] - r[i])
            cons.append(1 - c[i, 1] - r[i])
        # Non-negative radii
        for i in range(n):
            cons.append(r[i])
        return np.array(cons)
    
    bounds = [(0, 1)] * (2*n) + [(0, 1)] * n
    constraint = {'type': 'ineq', 'fun': con}
    
    result = minimize(obj, params, method='SLSQP', bounds=bounds,
                     constraints=constraint, options={'maxiter': max_iter, 'ftol': 1e-14})
    
    centers_opt = result.x[:2*n].reshape(n, 2)
    radii_opt = result.x[2*n:]
    
    return centers_opt, radii_opt


def penalty_finetune(centers, radii, n, max_outer=15):
    """Fine-tune using penalty method with L-BFGS-B"""
    params = np.concatenate([centers.flatten(), radii])
    
    def penalty_obj(p, lam):
        c = p[:2*n].reshape(n, 2)
        r = p[2*n:]
        pen = 0.0
        for i in range(n):
            for j in range(i + 1, n):
                d = np.sqrt(np.sum((c[i] - c[j]) ** 2))
                overlap = r[i] + r[j] - d
                if overlap > 0:
                    pen += overlap ** 2
            for dim in [0, 1]:
                if c[i, dim] < r[i]:
                    pen += (r[i] - c[i, dim]) ** 2
                if c[i, dim] > 1 - r[i]:
                    pen += (c[i, dim] - (1 - r[i])) ** 2
            if r[i] < 0:
                pen += r[i] ** 2
        return -np.sum(r) + lam * pen
    
    bounds = [(0, 1)] * (2*n) + [(0, 0.5)] * n
    lam = 50.0
    for _ in range(max_outer):
        result = minimize(penalty_obj, params, args=(lam,),
                         method='L-BFGS-B', bounds=bounds,
                         options={'maxiter': 2000, 'ftol': 1e-16})
        params = result.x
        lam *= 3
    
    centers_opt = params[:2*n].reshape(n, 2)
    radii_opt = params[2*n:]
    
    return centers_opt, radii_opt


def validate_local(centers, radii, n):
    """Check validity of packing and return sum of radii if valid"""
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if r < 0 or x - r < -1e-10 or x + r > 1 + 1e-10 or y - r < -1e-10 or y + r > 1 + 1e-10:
            return False, 0
    for i in range(n):
        for j in range(i + 1, n):
            d = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if d < radii[i] + radii[j] - 1e-10:
                return False, 0
    return True, np.sum(radii)
