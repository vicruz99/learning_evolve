                return 1.0 - vars_flat[idx] - vars_flat[idx + 2]
            constraints.append({'type': 'ineq', 'fun': cons_x_max})

            def cons_y_min(vars_flat, i=i, idx=idx):
                return vars_flat[idx + 1] - vars_flat[idx + 2]
