import numpy as np
from scipy.optimize import minimize

def compute_objective(vars, n, alpha):
    """
    Objective function for circle packing.
    Maximizes sum of radii while penalizing constraint violations.
    """
    r = vars[2::3]
    c = vars[:2*n].reshape(n, 2)
    
    # Boundary penalties: ensure circles are inside [0,1]x[0,1]
    pen = np.sum(np.maximum(0, r - c[:, 0])**2)
    pen += np.sum(np.maximum(0, r - c[:, 1])**2)
    pen += np.sum(np.maximum(0, r - (1.0 - c[:, 0]))**2)
    pen += np.sum(np.maximum(0, r - (1.0 - c[:, 1]))**2)
    
    # Overlap penalties: ensure circles don't intersect
    diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
    dist = np.sqrt(np.sum(diff**2, axis=2))
    overlap = r[:, np.newaxis] + r[np.newaxis, :] - dist
    np.fill_diagonal(overlap, 0.0)
    pen += np.sum(np.maximum(0, overlap)**2)
    
    return -np.sum(r) + alpha * pen

    n = 26
