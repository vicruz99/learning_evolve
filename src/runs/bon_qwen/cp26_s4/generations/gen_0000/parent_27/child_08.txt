import numpy as np
from scipy.optimize import minimize

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    n = centers.shape[0]

    if np.isnan(centers).any():
        return False

    if np.isnan(radii).any():
        return False

    for i in range(n):
        if radii[i] < 0:
            return False
        elif np.isnan(radii[i]):
            return False

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            return False

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False

    return True

def initialize_hex_grid(n):
    """
    Initialize centers in a hexagonal packing pattern to start close to a dense state.

    # Better: 5, 5, 5, 5, 5, 1 is sparse. Let's do 5, 5, 5, 5, 6 (tilted)?

    # A 5x6 grid (30 circles) with some removed or just dense packing.
    # Let's just use a 5x5 grid plus some points in the middle to start.
    
    centers = []
    # 5x5 grid
    x_vals = np.linspace(0.1, 0.9, 5)
    y_vals = np.linspace(0.1, 0.9, 5)
    for y in y_vals:
        for x in x_vals:
            centers.append([x, y])
    
    # Add 1 more circle in the center to make 26
    centers.append([0.5, 0.5])
    
    centers = np.array(centers)
    radii = np.ones(n) * 0.05 # Start small
    return centers, radii

def objective_and_constraints(centers_flat, radii_flat):
    """
    Objective: Maximize sum of radii (minimize negative sum).
    We handle constraints via a heavy penalty in the objective function 
    for simplicity in this unconstrained minimization wrapper, or use SLSQP.
    However, SLSQP with many inequality constraints is slow.
    Let's use a penalty method within the objective.
    """
    n = len(radii_flat)
    centers = centers_flat.reshape((n, 2))
    radii = radii_flat
    
    # Objective: Negative sum of radii
    obj = -np.sum(radii)
    
    penalty = 0.0
    penalty_weight = 1000.0
    
    # Boundary constraints: 0 <= x-r, x+r <= 1, etc.
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        
        # Boundary penalties
        if x - r < 0: penalty += penalty_weight * (x - r) ** 2
        if x + r > 1: penalty += penalty_weight * (x + r - 1) ** 2
        if y - r < 0: penalty += penalty_weight * (y - r) ** 2
        if y + r > 1: penalty += penalty_weight * (y + r - 1) ** 2
        
        # Radius non-negativity
        if r < 0: penalty += penalty_weight * r ** 2

    # Overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            min_dist = radii[i] + radii[j]
            if dist < min_dist:
                overlap = min_dist - dist
                penalty += penalty_weight * overlap ** 2
                
    return obj + penalty

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    n = 26
    centers, radii = initialize_hex_grid(n)
    
    # Flatten for optimizer
    x0 = np.concatenate([centers.flatten(), radii])
    
    # 2. Optimize
    # We use L-BFGS-B or SLSQP. Given the non-convex nature, we might need multiple restarts.
    # However, starting from a good grid should help.
    
    best_result = None
    best_score = float('inf')
    
    # Run multiple times with slight perturbations to escape local minima
    for _ in range(5):
        # Perturb initial positions slightly
        np.random.seed(42) # Reproducibility

        # Bounds: x,y in [0,1], r >= 0
        bounds = []
        for i in range(n):
            bounds.append((0, 1)) # x

            bounds.append((0, 0.5)) # r (cannot exceed 0.5)

            objective_and_constraints,
            x_perturbed,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 2000, 'ftol': 1e-12}
        )

            best_score = res.fun
            best_result = res.x

        if x + r > 1: centers_opt[i, 0] = 1 - r
        if y - r < 0: centers_opt[i, 1] = r
        if y + r > 1: centers_opt[i, 1] = 1 - r

    # Check overlaps and reduce radii slightly if needed (simple fix)

    for _ in range(10): # Iterative relaxation

                    changed = True
        if not changed:

    sum_radii = np.sum(radii_opt)
    
    return centers_opt, radii_opt, sum_radii

if __name__ == "__main__":
