For packing 26 circles to maximize sum of radii in a unit square:

1. **Initial configuration**: Start with hexagonal packing arrangements which are known to be dense for equal circles. The 5x5 grid gives r=0.1 for 25 circles, but with 26 circles and unequal sizes, we can potentially exceed sum=2.6.

2. **Optimization approach**: Use scipy's SLSQP optimizer to jointly optimize centers and radii. The problem is formulated as:
   - Maximize: sum(radii)
   - Subject to: boundary constraints and non-overlap constraints (squared distances >= sum of radii squared)

3. **Multiple restarts**: Since the problem is non-convex, use multiple random seeds with perturbed initial positions to escape local optima.

4. **Iterative refinement**: After initial optimization, perform multiple rounds where we:
   - Expand radii to their maximum allowed values
   - Use the expanded configuration as a new starting point for optimization

5. **Helper functions**: All defined at top level with no closures, using a global variable for n.

import numpy as np
from scipy.optimize import minimize

_n = 26

def objective_func(x):
    radii = x[2 * _n:]
    return -np.sum(radii)

def constraints_func(x):
    n = _n
    centers = x[:2 * n].reshape((n, 2))
    radii = x[2 * n:]
    
    cons = []
    
    for i in range(n):
        cons.append(centers[i, 0] - radii[i])
        cons.append(1 - centers[i, 0] - radii[i])
        cons.append(centers[i, 1] - radii[i])
        cons.append(1 - centers[i, 1] - radii[i])
    
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist_sq = dx * dx + dy * dy
            r_sum = radii[i] + radii[j]
            cons.append(dist_sq - r_sum * r_sum)
    
    for i in range(n):
        cons.append(radii[i])
    
    return np.array(cons)

def compute_max_radii(centers, radii, n):
    new_radii = np.zeros(n)
    for i in range(n):
        r = min(centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])
        for j in range(n):
            if i != j:
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                r = min(r, dist - radii[j])
        new_radii[i] = max(0.001, r)
    return new_radii

def generate_hexagonal_packing(n, spacing, row_offset_factor=0.866, seed=42):
    np.random.seed(seed)
    centers = np.zeros((n, 2))
    idx = 0
    for row in range(8):
        for col in range(6):
            if idx >= n:
                break
            x = spacing / 2 + col * spacing
            y = spacing / 2 + row * spacing * row_offset_factor
            if row % 2 == 1:
                x += spacing / 2
            x += np.random.randn() * 0.005
            y += np.random.randn() * 0.005
            x = np.clip(x, 0.02, 0.98)
            y = np.clip(y, 0.02, 0.98)
            centers[idx] = [x, y]
            idx += 1
        if idx >= n:
            break
    radii = np.ones(n) * 0.06
    return centers, radii

def generate_grid_packing(n, seed=42):
    np.random.seed(seed)
    centers = np.zeros((n, 2))
    idx = 0
    for row in range(6):
        for col in range(5):
            if idx >= n:
                break
            x = 0.05 + col * 0.19
            y = 0.05 + row * 0.17
            x += np.random.randn() * 0.005
            y += np.random.randn() * 0.005
            x = np.clip(x, 0.02, 0.98)
            y = np.clip(y, 0.02, 0.98)
            centers[idx] = [x, y]
            idx += 1
        if idx >= n:
            break
    radii = np.ones(n) * 0.06
    return centers, radii

def optimize_from_initial(x0, maxiter=2000, tol=1e-14):
    result = minimize(
        objective_func,
        x0,
        method='SLSQP',
        constraints={'type': 'ineq', 'fun': constraints_func},
        options={'maxiter': maxiter, 'ftol': tol}
    )
    return result.x

def run_packing():
    global _n
    _n = 26
    
    best_sum = -1.0
    best_centers = np.zeros((_n, 2))
    best_radii = np.zeros(_n)
    
    # Generate multiple initial configurations
    all_configs = []
    
    # Hexagonal packings with different spacings and seeds
    for spacing in [0.18, 0.19, 0.20]:
        for seed in range(10):
            centers, radii = generate_hexagonal_packing(26, spacing, seed=seed)
            all_configs.append((centers, radii))
    
    # Grid packings with different seeds
    for seed in range(10):
        centers, radii = generate_grid_packing(26, seed=seed)
        all_configs.append((centers, radii))
    
    # First pass: optimize each initial configuration
    all_results = []
    for centers_init, radii_init in all_configs:
        x0 = np.concatenate([centers_init.flatten(), radii_init])
        x_opt = optimize_from_initial(x0, maxiter=3000, tol=1e-14)
        
        centers = x_opt[:2 * _n].reshape((_n, 2))
        radii = x_opt[2 * _n:]
        
        # Clip radii to be non-negative
        radii = np.maximum(radii, 0.0001)
        
        current_sum = np.sum(radii)
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers.copy()
            best_radii = radii.copy()
        
        all_results.append((centers.copy(), radii.copy()))
    
    # Second pass: expand radii and re-optimize from best configurations
    for _ in range(3):
        new_configs = []
        for centers, radii in all_results:
            # Expand radii to maximum allowed
            expanded_radii = compute_max_radii(centers, radii, _n)
            new_configs.append((centers, expanded_radii))
        
        for centers_init, radii_init in new_configs:
            # Add small perturbation
            np.random.seed(np.random.randint(0, 10000))
            centers_pert = centers_init + np.random.randn(*centers_init.shape) * 0.002
            centers_pert = np.clip(centers_pert, 0.02, 0.98)
            
            x0 = np.concatenate([centers_pert.flatten(), radii_init])
            x_opt = optimize_from_initial(x0, maxiter=2000, tol=1e-14)
            
            centers = x_opt[:2 * _n].reshape((_n, 2))
            radii = x_opt[2 * _n:]
            radii = np.maximum(radii, 0.0001)
            
            current_sum = np.sum(radii)
            if current_sum > best_sum:
                best_sum = current_sum
                best_centers = centers.copy()
                best_radii = radii.copy()
            
            all_results.append((centers.copy(), radii.copy()))
    
    # Final expansion of radii for best result
    final_radii = compute_max_radii(best_centers, best_radii, _n)
    final_sum = np.sum(final_radii)
    
    return best_centers, final_radii, final_sum
