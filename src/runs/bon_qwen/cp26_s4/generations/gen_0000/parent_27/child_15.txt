import numpy as np
from scipy.optimize import minimize

def compute_penalty(centers, radii, weight):
    """Vectorized penalty computation for overlaps and boundaries."""
    n = len(radii)
    # Pairwise distances
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    min_dists = radii[:, None] + radii[None, :]
    
    # Ignore self-interaction on diagonal
    np.fill_diagonal(dists, np.inf)
    np.fill_diagonal(min_dists, np.inf)
    
    overlap = np.maximum(0, min_dists - dists)
    pair_penalty = np.sum(overlap**2)
    
    # Boundary penalties
    pen_x = np.sum(np.maximum(0, radii - centers[:, 0])**2)
    pen_x += np.sum(np.maximum(0, centers[:, 0] + radii - 1)**2)
    pen_y = np.sum(np.maximum(0, radii - centers[:, 1])**2)
    pen_y += np.sum(np.maximum(0, centers[:, 1] + radii - 1)**2)
    
    return (pair_penalty + pen_x + pen_y) * weight

def objective(params, n, weight):
    """Objective: maximize sum of radii (minimize negative) + penalties."""
    centers = params[:2*n].reshape(n, 2)
    radii = params[2*n:]
    obj = -np.sum(radii)
    obj += compute_penalty(centers, radii, weight)
    return obj

def repulsion_precondition(centers, radii, steps=1500):
    """Fast force-directed layout to resolve overlaps and grow radii."""
    n = len(radii)
    push_strength = 0.8
    
    for step in range(steps):
        # Resolve overlaps
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.hypot(dx, dy)
                req = radii[i] + radii[j]
                if dist < req and dist > 1e-9:
                    overlap = req - dist
                    nx, ny = dx / dist, dy / dist
                    push = overlap * 0.5 * push_strength
                    centers[i, 0] += nx * push
                    centers[i, 1] += ny * push
                    centers[j, 0] -= nx * push
                    centers[j, 1] -= ny * push
                    
        # Boundary constraints
        for i in range(n):
            if centers[i, 0] < radii[i]: centers[i, 0] = radii[i]
            if centers[i, 0] > 1 - radii[i]: centers[i, 0] = 1 - radii[i]
            if centers[i, 1] < radii[i]: centers[i, 1] = radii[i]
            if centers[i, 1] > 1 - radii[i]: centers[i, 1] = 1 - radii[i]
            
        # Gradual radius growth
        radii += 0.00015
        radii[radii > 0.45] = 0.45
        
    return centers, radii

def run_packing():
    N = 26
    
    # Initial placement: 5x5 grid + 1 circle in the center
    centers = np.zeros((N, 2))
    idx = 0
    for r_idx in range(5):

            if idx < N - 1:
                centers[idx, 0] = 0.1 + c_idx * 0.2
                centers[idx, 1] = 0.1 + r_idx * 0.2
                idx += 1
    centers[N-1, 0], centers[N-1, 1] = 0.5, 0.5
    
    radii = np.full(N, 0.08)
    radii[N-1] = 0.03  # Extra circle starts smaller
    
    # Pre-condition to find a feasible, tight packing
    centers, radii = repulsion_precondition(centers.copy(), radii.copy())
    
    # Prepare for optimization
    params = np.concatenate([centers.ravel(), radii])
    bounds = [(0.0, 1.0)] * (2*N) + [(1e-5, 0.5)] * N
    
    # Gradient-based refinement
    res = minimize(objective, params, args=(N, 8000.0), method='L-BFGS-B', 
                   bounds=bounds, options={'maxiter': 3000, 'ftol': 1e-10})
    
    centers_opt = res.x[:2*N].reshape(N, 2)
    radii_opt = res.x[2*N:]
    
    # Final safety clamp (optimizer should respect bounds, but numerical noise can occur)
    radii_opt = np.clip(radii_opt, 1e-9, None)
    centers_opt[:, 0] = np.clip(centers_opt[:, 0], radii_opt, 1 - radii_opt)
    centers_opt[:, 1] = np.clip(centers_opt[:, 1], radii_opt, 1 - radii_opt)
    
    sum_r = np.sum(radii_opt)
    return centers_opt, radii_opt, sum_r
