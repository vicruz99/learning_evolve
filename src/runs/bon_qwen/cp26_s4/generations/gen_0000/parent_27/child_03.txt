import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def objective(vars):
    """Objective: minimize negative sum of radii (equivalent to maximizing sum)"""
    return -np.sum(vars[2::3])

def constraint_x_min(vars, i):
    return vars[3*i] - vars[3*i+2]

def constraint_x_max(vars, i):
    return 1.0 - vars[3*i] - vars[3*i+2]

def constraint_y_min(vars, i):
    return vars[3*i+1] - vars[3*i+2]

def constraint_y_max(vars, i):
    return 1.0 - vars[3*i+1] - vars[3*i+2]

def constraint_overlap(vars, i, j):
    dx = vars[3*i] - vars[3*j]
    dy = vars[3*i+1] - vars[3*j+1]
    dr = vars[3*i+2] + vars[3*j+2]
    return dx*dx + dy*dy - dr*dr

def run_packing():
    # Prepare bounds: x in [0,1], y in [0,1], r in [0.01, 0.5]
    bounds = [(0.0, 1.0)] * (2 * N_CIRCLES) + [(0.01, 0.5)] * N_CIRCLES
    
    # Prepare constraints
    cons = []
    for i in range(N_CIRCLES):
        cons.append({'type': 'ineq', 'fun': constraint_x_min, 'args': (i,)})
        cons.append({'type': 'ineq', 'fun': constraint_x_max, 'args': (i,)})
        cons.append({'type': 'ineq', 'fun': constraint_y_min, 'args': (i,)})
        cons.append({'type': 'ineq', 'fun': constraint_y_max, 'args': (i,)})
        
    for i in range(N_CIRCLES):
        for j in range(i + 1, N_CIRCLES):
            cons.append({'type': 'ineq', 'fun': constraint_overlap, 'args': (i, j)})
            
    best_sum = -1.0
    best_result = None
    
    # Try multiple initial layouts to escape local optima
    initial_layouts = []
    
    # Layout 1: Staggered hexagonal packing
    centers_hex = np.zeros((N_CIRCLES, 2))
    idx = 0
    row_counts = [5, 6, 5, 6, 4]
    for r_idx, cnt in enumerate(row_counts):
        y = 0.15 + r_idx * 0.16
        offset = 0.08 if r_idx % 2 == 1 else 0.0
        for c in range(cnt):
            x = 0.15 + c * 0.16 + offset
            centers_hex[idx] = [x, y]
            idx += 1
    initial_layouts.append(centers_hex)
    
    # Layout 2: Dense grid with one extra
    centers_grid = np.zeros((N_CIRCLES, 2))
    idx = 0
    for r in range(5):
        cnt = 5 if r < 4 else 1
        y = 0.1 + r * 0.2
        for c in range(cnt):
            x = 0.1 + c * 0.2
            centers_grid[idx] = [x, y]
            idx += 1
    initial_layouts.append(centers_grid)
    
    opt_options = {'maxiter': 4000, 'ftol': 1e-11, 'disp': False}
    
    for init_centers in initial_layouts:
        x0 = np.zeros(3 * N_CIRCLES)
        for i in range(N_CIRCLES):
            x0[3*i] = init_centers[i, 0]
            x0[3*i+1] = init_centers[i, 1]
            x0[3*i+2] = 0.06  # Initial radius
            
        res = minimize(objective, x0, method='SLSQP', bounds=bounds, 
                       constraints=cons, options=opt_options)
                       
        if res.success:
            current_sum = np.sum(res.x[2::3])
            if current_sum > best_sum:
                best_sum = current_sum
                best_result = (res.x.reshape((N_CIRCLES, 3))[:, :2], 
                               res.x.reshape((N_CIRCLES, 3))[:, 2])
                               
    if best_result is None:
        # Fallback to initial hex layout if optimization fails unexpectedly
        return initial_layouts[0], np.full(N_CIRCLES, 0.06), 0.06 * N_CIRCLES
        
    centers, radii = best_result
    
    # Post-processing for strict validity
    radii = np.maximum(radii, 1e-6)
    for i in range(N_CIRCLES):
        r = radii[i]
        centers[i, 0] = np.clip(centers[i, 0], r, 1.0 - r)
        centers[i, 1] = np.clip(centers[i, 1], r, 1.0 - r)
        
    # Safeguard: shrink radii slightly if numerical overlaps persist
    for _ in range(20):
        valid = True
        for i in range(N_CIRCLES):
            for j in range(i + 1, N_CIRCLES):
                dist_sq = np.sum((centers[i] - centers[j]) ** 2)
                if dist_sq < (radii[i] + radii[j]) ** 2 - 1e-12:
                    valid = False
                    break
            if not valid:
                break
        if valid:
            break
        radii *= 0.998
        
    return centers, radii, np.sum(radii)
