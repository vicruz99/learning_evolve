    for i in range(n):

                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True

def get_overlap_energy(centers, radii):
    """
    Computes the total overlap energy and boundary violation energy.
    Energy = sum of squared overlaps + sum of squared boundary violations.
    """
    n = centers.shape[0]
    energy = 0.0
    
    # Pairwise overlaps
    # Vectorized computation for efficiency
    # centers shape: (n, 2)
    # radii shape: (n,)
    
    # Compute pairwise distances
    # diff shape: (n, n, 2)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    # Sum of radii matrix
    radii_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    
    # Overlap amount (positive if overlapping)
    overlap = radii_sum - dists
    
    # We only care about positive overlaps
    overlap = np.maximum(0, overlap)
    
    # Add squared overlaps to energy
    # Divide by 2 to avoid double counting pairs in sum
    energy += np.sum(overlap**2) / 2.0
    
    # Boundary violations
    # x - r < 0  => violation r - x
    # x + r > 1  => violation x + r - 1
    # y - r < 0  => violation r - y
    # y + r > 1  => violation y + r - 1
    
    x = centers[:, 0]
    y = centers[:, 1]
    r = radii
    
    v1 = np.maximum(0, r - x)
    v2 = np.maximum(0, x + r - 1)
    v3 = np.maximum(0, r - y)
    v4 = np.maximum(0, y + r - 1)
    
    energy += np.sum(v1**2 + v2**2 + v3**2 + v4**2)
    
    return energy

def optimize_centers(centers, radii, max_iters=200):
    """
    Optimizes centers to minimize overlap energy for fixed radii.
    Uses L-BFGS-B.
    """
    n = centers.shape[0]
    
    # Flatten centers for optimization
    x0 = centers.flatten()
    
    def objective(x):
        c = x.reshape((n, 2))
        return get_overlap_energy(c, radii)
    
    # Bounds for centers: [0, 1]
    bounds = [(0, 1) for _ in range(2 * n)]
    
    # Use L-BFGS-B
    res = opt.minimize(objective, x0, method='L-BFGS-B', bounds=bounds, 
                       options={'maxiter': max_iters, 'ftol': 1e-15})
    
    return res.x.reshape((n, 2))
