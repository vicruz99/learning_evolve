import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    """
    n = 26
    best_centers = None
    best_radii = None
    best_sum = -1.0

    def objective(x):
        # x is a flattened array of [x0, y0, r0, x1, y1, r1, ...]
        # Maximize sum of radii => Minimize -sum(r)
        radii_part = x[2::3]
        return -np.sum(radii_part)

    def constraint_boundary(x):
        # r_i <= x_i <= 1 - r_i  => x_i - r_i >= 0, 1 - x_i - r_i >= 0
        # Same for y
        # Returns array of values that must be >= 0
        c = []
        for i in range(n):
            xi = x[3*i]
            yi = x[3*i+1]
            ri = x[3*i+2]
            # x - r >= 0
            c.append(xi - ri)
            # 1 - x - r >= 0
            c.append(1.0 - xi - ri)
            # y - r >= 0
            c.append(yi - ri)
            # 1 - y - r >= 0
            c.append(1.0 - yi - ri)
        return np.array(c)

    def constraint_overlap(x):
        # dist^2 >= (r_i + r_j)^2 => dist^2 - (r_i + r_j)^2 >= 0
        c = []
        centers = x.reshape(n, 3)[:, :2]
        radii = x.reshape(n, 3)[:, 2]
        
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist_sq = dx*dx + dy*dy
                r_sum = radii[i] + radii[j]
                c.append(dist_sq - r_sum*r_sum)
        return np.array(c)

    def create_initial_config(strategy='grid'):
        x0 = np.zeros(3 * n)
        if strategy == 'grid':
            # 5x5 grid plus 1
            # 25 circles in 5x5, r=0.09
            idx = 0
            positions = []
            for r in range(5):
                for c in range(5):
                    x = 0.1 + c * 0.2
                    y = 0.1 + r * 0.2
                    positions.append((x, y))
            
            # Place 26th circle in a gap, e.g., center of a hole
            # Hole at (0.2, 0.2) relative to grid? 
            # Let's place it at (0.15, 0.15) or random
            # Better: (0.5, 0.5) is occupied. 
            # Let's put it at (0.95, 0.95) - corner? 
            # Or just a small radius random pos.
            positions.append((0.5, 0.05)) # Try near edge
            
            for k, (px, py) in enumerate(positions):
                x0[3*k] = px
                x0[3*k+1] = py
                # Start with radius 0.09, shrink last one
                x0[3*k+2] = 0.085 if k < 25 else 0.02
        
        elif strategy == 'hex':
            # Hexagonal packing attempt
            # Rows of circles
            # Approximate layout for 26 circles
            # 5, 6, 5, 6, 4 ? Total 26
            # Or 5, 5, 5, 5, 5, 1
            # Let's try 5 rows of 5, 1 extra
            # Actually let's just generate points in a dense hex pattern
            row_y = 0.05
            r_approx = 0.09
            dy = r_approx * np.sqrt(3)
            dx = 2 * r_approx
            idx = 0
            row_count = 0
            while idx < n:
                cols_in_row = 5 if row_count % 2 == 0 else 4 # Zigzag
                for c in range(cols_in_row):
                    if idx < n:
                        x_pos = 0.05 + c * dx + (0.5 * dx if row_count % 2 != 0 else 0)
                        y_pos = row_y + row_count * dy
                        x0[3*idx] = x_pos
                        x0[3*idx+1] = y_pos
                        x0[3*idx+2] = 0.08
                        idx += 1
                row_count += 1
        
        # Add small random noise to break symmetry
        x0 += np.random.normal(0, 0.005, size=x0.shape)
        # Clip radii to be positive
        x0[2::3] = np.abs(x0[2::3]) + 1e-5
        return x0

    # Optimization settings
    # SLSQP handles constraints well
    # We run a few iterations to find the best local optimum
    
    num_attempts = 10
    for attempt in range(num_attempts):
        # Alternate strategies
        strat = 'grid' if attempt % 2 == 0 else 'hex'
        x0 = create_initial_config(strat)
        
        # Bounds: x, y in [0, 1], r >= 0
        bounds = [(0, 1)] * (2 * n) + [(0, 1)] * n
        
        cons = [
            {'type': 'ineq', 'fun': constraint_boundary},
            {'type': 'ineq', 'fun': constraint_overlap}
        ]
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, 
                           options={'maxiter': 100, 'ftol': 1e-9})
            
            if res.success or res.fun < best_sum * -1: # Since objective is -sum, smaller is better sum

                radii = res.x[2*n:]

                # Boundary

                # Overlap

                        for j in range(i+1, n):
                            dist = np.linalg.norm(centers[i] - centers[j])
                            if dist < radii[i] + radii[j] - 1e-9:
                                valid = False
                                break
                        if not valid: break
                
                if valid:
                    current_sum = np.sum(radii)
                    if current_sum > best_sum:
                        best_sum = current_sum
                        best_centers = centers
                        best_radii = radii
        except Exception as e:
            pass

    # Fallback if optimization failed completely (unlikely)
    if best_centers is None:

        best_centers = np.zeros((26, 2))
        best_radii = np.zeros(26)

        # Last circle

        best_sum = np.sum(best_radii)

    return best_centers, best_radii, best_sum
