import numpy as np
import scipy.optimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Pack 26 circles in a unit square to maximize sum of radii.
    """
    n = 26
    np.random.seed(42) # For reproducibility

    # Helper to initialize a hexagonal packing
    def initialize_hex_packing(n_circles):
        centers = np.zeros((n_circles, 2))
        radii = np.zeros(n_circles)
        
        # Try to fit circles in rows
        # Based on analysis, 6 rows with counts [5, 4, 5, 4, 5, 3] fits well
        # This is a starting point.
        row_counts = [5, 4, 5, 4, 5, 3]
        # If n_circles doesn't match sum, adjust (though we use 26)
        
        idx = 0
        r_init = 0.09 # A safe initial radius
        
        for row_idx, count in enumerate(row_counts):
            if idx >= n_circles:
                break
            
            # y coordinate for this row
            # Hexagonal vertical spacing: r * sqrt(3)
            # First row at y = r, next at y = r + r*sqrt(3), etc.
            y = r_init + row_idx * r_init * np.sqrt(3)
            
            # x coordinates
            # Even rows (0, 2, ...) start at x = r
            # Odd rows (1, 3, ...) start at x = 2r (shifted by r)
            if row_idx % 2 == 0:
                x_start = r_init
            else:
                x_start = 2 * r_init
            
            for col_idx in range(count):
                if idx >= n_circles:
                    break
                x = x_start + col_idx * 2 * r_init
                centers[idx] = [x, y]
                radii[idx] = r_init
                idx += 1
        
        # Pad if fewer circles than requested (should not happen with correct counts)
        while idx < n_circles:
            centers[idx] = [0.5, 0.5]
            radii[idx] = 0.001
            idx += 1
            
        return centers, radii

    # Initialization
    centers, radii = initialize_hex_packing(n)
    
    # Flatten variables for optimizer
    # Order: x1, y1, r1, x2, y2, r2, ...
    x0 = np.zeros(3 * n)
    for i in range(n):
        x0[3*i] = centers[i, 0]
        x0[3*i+1] = centers[i, 1]
        x0[3*i+2] = radii[i]

    # Objective function: Maximize sum of radii -> Minimize -sum(radii)
    def objective(x):
        radii_opt = x[2::3]
        return -np.sum(radii_opt)

        constraints = []
        radii_opt = x[2::3]
        centers_x = x[0::3]
        centers_y = x[1::3]

        # Same for y
        for i in range(n):
            # Left
            constraints.append({'type': 'ineq', 'fun': lambda x, i=i: x[3*i] - x[3*i+2]})
            # Right
            constraints.append({'type': 'ineq', 'fun': lambda x, i=i: 1 - x[3*i+2] - x[3*i]})
            # Bottom
            constraints.append({'type': 'ineq', 'fun': lambda x, i=i: x[3*i+1] - x[3*i+2]})
            # Top
            constraints.append({'type': 'ineq', 'fun': lambda x, i=i: 1 - x[3*i+2] - x[3*i+1]})
            
            # Radius non-negativity (weakly handled by boundaries usually, but good to be safe)
            # constraints.append({'type': 'ineq', 'fun': lambda x, i=i: x[3*i+2]})

        # Non-overlap constraints
        # dist^2 >= (r_i + r_j)^2
        # (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
        for i in range(n):
            for j in range(i + 1, n):
                def dist_con(x, i=i, j=j):
                    dx = x[3*i] - x[3*j]
                    dy = x[3*i+1] - x[3*j+1]
                    dr = x[3*i+2] + x[3*j+2]
                    return dx*dx + dy*dy - dr*dr
                constraints.append({'type': 'ineq', 'fun': dist_con})
        
        return constraints

    # Simple bounds: x,y in [0, 1], r in [0, 0.5]
    bounds = [(0, 1)] * (2 * n) + [(0, 0.5)] * n

    res = scipy.optimize.minimize(
