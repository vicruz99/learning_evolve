import numpy as np
from scipy.optimize import linprog
import math

def solve_radii(centers):
    """
    Solves the LP to find the maximum sum of radii for fixed centers.
    Maximize sum(r_i) subject to boundary and pairwise constraints.
    """
    n = centers.shape[0]
    # Objective: Maximize sum(r_i) => Minimize -sum(r_i)
    c_obj = -np.ones(n)
    
    # Constraints: A_ub @ r <= b_ub
    A_ub = []
    b_ub = []
    
    # 1. Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        x, y = centers[i]
        row = np.zeros(n)
        row[i] = 1.0
        
        A_ub.append(row); b_ub.append(x)
        A_ub.append(row); b_ub.append(1.0 - x)
        A_ub.append(row); b_ub.append(y)
        A_ub.append(row); b_ub.append(1.0 - y)
        
    # 2. Pairwise constraints: r_i + r_j <= dist_ij
    for i in range(n):
        for j in range(i + 1, n):
            dist = math.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dist)
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    bounds = [(0, None) for _ in range(n)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, -res.fun
        else:
            # Fallback to small radii if LP fails
            return np.full(n, 1e-9), 26e-9
    except Exception:
        return np.full(n, 1e-9), 26e-9

def get_sum_radii(centers_flat):
    """
    Objective function wrapper for the optimizer.
    Centers are flattened for scipy optimization.
    """
    centers = np.reshape(centers_flat, (-1, 2))
    # Clip centers to valid range to avoid numerical issues in LP
    centers = np.clip(centers, 1e-6, 1.0 - 1e-6)
    _, sum_r = solve_radii(centers)
    return -sum_r

def run_packing():
    """
    Generates and optimizes a packing of 26 circles in a unit square.
    """
    n_circles = 26
    
    # 1. Initialize centers in a Hexagonal Grid
    centers = []
    spacing = 0.15
    row = 0
    while len(centers) < n_circles:
        y = row * spacing * (math.sqrt(3) / 2) + 0.1
        col = 0
        while len(centers) < n_circles:
            x = col * spacing + (0.075 if row % 2 == 1 else 0.1)
            if 0 <= x <= 1 and 0 <= y <= 1:
                centers.append([x, y])
            col += 1
        row += 1
    centers = np.array(centers[:n_circles])

    # 2. Coordinate Descent Optimization
    # We perform multiple passes to refine the positions
    centers = np.clip(centers, 0.05, 0.95) # Initial safety margin
    
    for _ in range(3): # 3 passes of coordinate descent
        for i in range(n_circles):
            # Optimize X coordinate
            def opt_x(d):
                c = centers.copy()
                c[i, 0] = d
                return get_sum_radii(c.flatten())
            
            from scipy.optimize import minimize_scalar
            res_x = minimize_scalar(opt_x, bounds=(0.05, 0.95), method='bounded')
            centers[i, 0] = res_x.x
            
            # Optimize Y coordinate
            def opt_y(d):
                c = centers.copy()
                c[i, 1] = d
                return get_sum_radii(c.flatten())
            
            res_y = minimize_scalar(opt_y, bounds=(0.05, 0.95), method='bounded')
            centers[i, 1] = res_y.y

    # 3. Final Radii Calculation
    radii, sum_radii = solve_radii(centers)
    
    # Ensure strict compliance with boundaries for the final output
    for i in range(n_circles):
        r = radii[i]
        x, y = centers[i]
        # Shrink radius if it touches boundary slightly
        r = min(r, x - 1e-12, 1 - x - 1e-12, y - 1e-12, 1 - y - 1e-12)
        radii[i] = max(0, r)

    return centers, radii, float(sum_radii)
