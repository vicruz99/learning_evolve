import numpy as np
from scipy.optimize import minimize

def run_packing():
    """
    Optimizes the positions and radii of 26 circles in a unit square
    to maximize the sum of radii.
    """
    n_circles = 26
    best_result = None
    best_sum = -1.0

    # Optimization configuration
    n_runs = 10
    
    for run in range(n_runs):
        # 1. Initialization: Hexagonal Grid
        centers_init = []
        # Try to fit into rows. 5 rows is a good heuristic for n=26.
        # Row lengths: 6, 5, 5, 5, 5 (total 26)
        row_lengths = [6, 5, 5, 5, 5]
        radius_est = 0.09  # Initial guess smaller than 0.1 to ensure feasibility
        
        r = radius_est
        x_start = r
        y_start = r
        
        # Vertical spacing for hexagonal packing
        dy = r * np.sqrt(3)
        
        count = 0
        for i, length in enumerate(row_lengths):
            # Offset alternating rows
            offset = r if i % 2 == 1 else 0
            current_y = y_start + i * dy
            
            # Distribute 'length' circles in the row within [0, 1]
            if length > 1:
                # Space them evenly
                x_coords = np.linspace(r, 1 - r, length)
            else:
                x_coords = [0.5]
            
            # Apply offset
            x_coords = x_coords + offset
            
            # If offset pushes out of bounds, clamp (simple fix)
            x_coords = np.clip(x_coords, r, 1 - r)
            
            for x in x_coords:
                centers_init.append([x, current_y])
                count += 1
        
        # Pad if necessary (though logic above sums to 26)
        while len(centers_init) < n_circles:
            centers_init.append([0.5, 0.5])
            
        centers_init = np.array(centers_init[:n_circles])
        
        # Perturb centers slightly to break symmetry and help optimization
        np.random.seed(run)
        perturbation = np.random.uniform(-0.02, 0.02, centers_init.shape)
        centers_init = centers_init + perturbation
        centers_init = np.clip(centers_init, r, 1 - r)
        
        # Flatten variables: [x1, y1, r1, x2, y2, r2, ...]
        x0 = np.zeros(3 * n_circles)
        for i in range(n_circles):
            x0[3 * i] = centers_init[i, 0]
            x0[3 * i + 1] = centers_init[i, 1]
            x0[3 * i + 2] = r # Initial radius
            
        # 2. Define Objective and Constraints
        
        def objective(vars_flat):
            # Maximize sum of radii => Minimize negative sum
            radii = vars_flat[2::3]
            return -np.sum(radii)
        
        def get_boundaries(vars_flat):
            # Returns 4 constraints per circle: x-r>=0, 1-(x+r)>=0, y-r>=0, 1-(y+r)>=0
            cons = []
            for i in range(n_circles):
                x = vars_flat[3 * i]
                y = vars_flat[3 * i + 1]
                r = vars_flat[3 * i + 2]
                cons.append(x - r)
                cons.append(1 - (x + r))
                cons.append(y - r)
                cons.append(1 - (y + r))
            return cons
        
        def get_overlap_constraints(vars_flat):
            # Returns constraints: dist_ij - (ri + rj) >= 0
            cons = []
            for i in range(n_circles):
                xi = vars_flat[3 * i]
                yi = vars_flat[3 * i + 1]
                ri = vars_flat[3 * i + 2]
                for j in range(i + 1, n_circles):
                    xj = vars_flat[3 * j]
                    yj = vars_flat[3 * j + 1]
                    rj = vars_flat[3 * j + 2]
                    
                    dist = np.sqrt((xi - xj)**2 + (yi - yj)**2)
                    cons.append(dist - (ri + rj))
            return cons

        # Construct constraints list for SLSQP
        # We define a single constraint function that returns an array of values >= 0
        def constraints_func(vars_flat):
            bounds_cons = get_boundaries(vars_flat)
            overlap_cons = get_overlap_constraints(vars_flat)
            return bounds_cons + overlap_cons

        # Bounds for variables
        # x, y in [0, 1], r in [0, 1]
        bounds = [(0, 1)] * (3 * n_circles)
        
        # 3. Run Optimizer
        try:
            result = minimize(
                objective,
                x0,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'ineq', 'fun': constraints_func},
                options={'maxiter': 2000, 'ftol': 1e-9}
            )
            
            if result.success or result.fun < 0: # Valid solution found
                final_sum = -result.fun
                if final_sum > best_sum:
                    best_sum = final_sum
                    best_result = result.x
                    
        except Exception as e:
            continue

    # 4. Extract and Validate Result
    if best_result is None:
        # Fallback to simple grid if optimization fails
