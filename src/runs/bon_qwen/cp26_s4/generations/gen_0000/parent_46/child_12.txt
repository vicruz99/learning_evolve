import numpy as np

def compute_repulsion(centers, radii):
    """Compute pairwise repulsive forces based on overlap depth."""
    n = centers.shape[0]
    forces = np.zeros_like(centers)
    for i in range(n):
        for j in range(i + 1, n):
            diff = centers[j] - centers[i]
            dist = np.linalg.norm(diff)
            if dist < 1e-9:
                dist = 1e-9
                diff = np.array([1.0, 0.0])
            
            overlap = radii[i] + radii[j] - dist
            if overlap > 0:
                # Force proportional to overlap, directed along separation vector
                f = overlap * diff / dist
                forces[i] -= f
                forces[j] += f
    return forces

def apply_boundary_forces(centers, radii, forces):

    n = centers.shape[0]
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        # Left boundary
        if x < r:
            forces[i, 0] += (r - x) * 5.0
        # Right boundary
        if x > 1 - r:
            forces[i, 0] -= (x - (1 - r)) * 5.0
        # Bottom boundary

            forces[i, 1] += (r - y) * 5.0
        # Top boundary
        if y > 1 - r:
            forces[i, 1] -= (y - (1 - r)) * 5.0
    return forces
