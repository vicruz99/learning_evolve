import numpy as np
from scipy.optimize import minimize

def compute_loss(vars, n, penalty):
    """
    Computes the objective: negative sum of radii plus penalties for constraint violations.
    vars: flattened array [x1, y1, x2, y2, ..., r1, r2, ...]
    n: number of circles
    penalty: weight for constraint violations
    """
    centers = vars[:2*n].reshape(n, 2)
    radii = vars[2*n:]
    
    # Overlap penalty
    diff = centers[:, None, :] - centers[None, :, :]
    dist = np.sqrt(np.sum(diff**2, axis=2))
    idx = np.triu_indices(n, k=1)
    violations = np.maximum(0, radii[idx[0]] + radii[idx[1]] - dist[idx])
    pen_overlap = np.sum(violations**2)
    
    # Boundary penalty
    pen_bound = np.sum(np.maximum(0, radii - centers[:, 0])**2)
    pen_bound += np.sum(np.maximum(0, radii - (1 - centers[:, 0]))**2)
    pen_bound += np.sum(np.maximum(0, radii - centers[:, 1])**2)
    pen_bound += np.sum(np.maximum(0, radii - (1 - centers[:, 1]))**2)
    
    return -np.sum(radii) + penalty * (pen_overlap + pen_bound)

def inflate_radii(centers, radii, n):
    """
    Uniformly increases radii as long as constraints are satisfied.
    """
    radii = radii.copy()
    for _ in range(200):
        diff = centers[:, None, :] - centers[None, :, :]
        dist = np.sqrt(np.sum(diff**2, axis=2))
        idx = np.triu_indices(n, k=1)
        slack_overlap = dist[idx] - (radii[idx[0]] + radii[idx[1]])
        
        slack_bound = np.minimum(
            np.minimum(centers[:, 0] - radii, 1 - centers[:, 0] - radii),
            np.minimum(centers[:, 1] - radii, 1 - centers[:, 1] - radii)
        )
        
        min_slack = np.min(np.concatenate([slack_overlap, slack_bound]))
        if min_slack < 1e-8:
            break
        radii += min_slack
    return radii

def run_optimizer(n, x0, penalty, max_iter):
    """Runs L-BFGS-B optimization without closures."""
    bounds = [(0, 1)] * (2*n) + [(0, 0.5)] * n
    res = minimize(compute_loss, x0, args=(n, penalty), method='L-BFGS-B', 
                   bounds=bounds, options={'maxiter': max_iter, 'ftol': 1e-15})
    return res.x

def run_packing():
    n = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    penalty = 8000.0
    max_iter = 2500
    n_seeds = 40
    
    for seed in range(n_seeds):
        np.random.seed(seed)
        
        # Generate diverse initial configurations
        if seed % 4 == 0:
            # Random
            c = np.random.rand(n, 2) * 0.7 + 0.15
        elif seed % 4 == 1:
            # Hexagonal lattice
            c = np.zeros((n, 2))
            cnt = 0
            y = 0.0
            row = 0
            while cnt < n:
                x = 0.0
                while cnt < n:
                    px = x + (0.06 if row % 2 else 0)
                    py = y
                    if px <= 0.94 and py <= 0.94:
                        c[cnt] = [px, py]
                        cnt += 1
                    x += 0.13
                y += 0.113
                row += 1
            c += np.random.randn(n, 2) * 0.005
        elif seed % 4 == 2:
            # Grid-like
            cols = int(np.ceil(np.sqrt(n)))
            rows = int(np.ceil(n / cols))
            c = np.zeros((n, 2))
            idx = 0
            for r in range(rows):
                for cc in range(cols):
                    if idx < n:
                        c[idx] = [cc * 0.12 + 0.06, r * 0.12 + 0.06]
                        idx += 1
            c += np.random.randn(n, 2) * 0.01
        else:
            # Random dense cluster
            c = np.random.rand(n, 2) * 0.6 + 0.2
            c += np.random.randn(n, 2) * 0.05
            c = np.clip(c, 0.05, 0.95)
            
        r_init = np.ones(n) * 0.045
        x0 = np.concatenate([c.flatten(), r_init])
        
        res_x = run_optimizer(n, x0, penalty, max_iter)
        centers = res_x[:2*n].reshape(n, 2)
        radii = res_x[2*n:]
        
        # Post-optimization inflation
        radii = inflate_radii(centers, radii, n)
        
        # Strict validation
        valid = True
        diff = centers[:, None, :] - centers[None, :, :]
        dist = np.sqrt(np.sum(diff**2, axis=2))
        
        # Check overlaps
        idx = np.triu_indices(n, k=1)
        if np.any(dist[idx] < radii[idx[0]] + radii[idx[1]] - 1e-9):
            valid = False
            
        # Check boundaries
        if np.any(centers[:, 0] < radii - 1e-9) or np.any(centers[:, 0] > 1 - radii + 1e-9) or \
           np.any(centers[:, 1] < radii - 1e-9) or np.any(centers[:, 1] > 1 - radii + 1e-9):
            valid = False
            
        if valid:
            current_sum = np.sum(radii)
            if current_sum > best_sum:
                best_sum = current_sum
                best_centers = centers.copy()
                best_radii = radii.copy()
                
    return best_centers, best_radii, best_sum
