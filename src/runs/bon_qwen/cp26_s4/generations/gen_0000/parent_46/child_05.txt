import numpy as np
from scipy.optimize import minimize
from functools import partial

def eval_objective(x, n, lam, idx_pairs):
    """
    Computes the objective function: negative sum of radii + penalty terms.
    """
    c = x[:2*n].reshape(n, 2)
    r = x[2*n:]
    
    val = -np.sum(r)
    
    i_idx = idx_pairs[0]
    j_idx = idx_pairs[1]
    
    # Overlap penalty
    c_diff = c[i_idx] - c[j_idx]
    d = np.sqrt(np.sum(c_diff**2, axis=1))
    sums_r = r[i_idx] + r[j_idx]
    
    diff = sums_r - d
    val += lam * np.sum(np.maximum(diff, 0.0)**2)
    
    # Boundary penalty
    val += lam * np.sum(np.maximum(r - c[:, 0], 0.0)**2)
    val += lam * np.sum(np.maximum(r - (1.0 - c[:, 0]), 0.0)**2)
    val += lam * np.sum(np.maximum(r - c[:, 1], 0.0)**2)
    val += lam * np.sum(np.maximum(r - (1.0 - c[:, 1]), 0.0)**2)
    
    return val

def run_packing():
    np.random.seed(42)
    n = 26
    idx_pairs = np.triu_indices(n, k=1)
    
    # 1. Initialize on a perturbed hexagonal lattice
    centers = []
    x_step = 0.23
    y_step = 0.2
    for j in range(6):
        y = 0.15 + j * y_step
        offset = (x_step / 2.0) if j % 2 == 1 else 0.0
        i = 0
        while True:
            x = 0.1 + i * x_step + offset
            if x > 0.9:
                break
            if len(centers) < n:
                centers.append([x, y])
            i += 1
    centers = np.array(centers[:n])
    centers += np.random.uniform(-0.02, 0.02, centers.shape)
    centers = np.clip(centers, 0.05, 0.95)
    
    radii = np.full(n, 0.06)
    
    # Flatten to 1D array for scipy: [x1, y1, r1, x2, y2, r2, ...]
    x0 = np.zeros(3 * n)
    for i in range(n):
        x0[3*i] = centers[i, 0]
        x0[3*i+1] = centers[i, 1]
        x0[3*i+2] = radii[i]
        
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
    
    # 2. Iterative penalty optimization
    lam = 100.0
    for _ in range(15):
        obj_func = partial(eval_objective, n=n, lam=lam, idx_pairs=idx_pairs)
        res = minimize(obj_func, x0, method='L-BFGS-B', bounds=bounds, options={'maxiter': 2000, 'ftol': 1e-10})
        x0 = res.x
        lam *= 1.8
        
    centers_out = x0[:2*n].reshape(n, 2)
    radii_out = x0[2*n:]
    
    # 3. Post-processing to guarantee strict validity within 1e-12 tolerance
    max_viol = 0.0
    for i in range(n):
        r = radii_out[i]
        x, y = centers_out[i]
        max_viol = max(max_viol, r - x, r - (1.0 - x), r - y, r - (1.0 - y))
        
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(centers_out[i] - centers_out[j])
            max_viol = max(max_viol, radii_out[i] + radii_out[j] - dist)
            
    if max_viol > 1e-7:
        shrink = max_viol + 1e-7
        radii_out = np.maximum(radii_out - shrink, 0.0)
        
    sum_r = np.sum(radii_out)
    return centers_out, radii_out, sum_r
