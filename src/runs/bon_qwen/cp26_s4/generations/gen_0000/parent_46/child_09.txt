import numpy as np
from scipy.optimize import minimize
import math

def compute_penalty(c, r, n):
    """Compute squared penalty for overlaps and boundary violations."""
    pen = 0.0
    # Pairwise overlap penalties
    for i in range(n):
        for j in range(i + 1, n):
            dx = c[i, 0] - c[j, 0]
            dy = c[i, 1] - c[j, 1]
            d = math.sqrt(dx * dx + dy * dy + 1e-12)
            diff = r[i] + r[j] - d
            if diff > 0:
                pen += diff ** 2
    # Boundary penalties
    for i in range(n):
        d = min(c[i, 0], 1.0 - c[i, 0], c[i, 1], 1.0 - c[i, 1])
        diff = r[i] - d
        if diff > 0:
            pen += diff ** 2
    return pen

def objective(x, n, w):
    """Objective: maximize sum of radii (minimize negative sum) + penalty."""
    c = x[:2 * n].reshape(n, 2)
    r = x[2 * n:]
    return -np.sum(r) + w * compute_penalty(c, r, n)

def fix_violations(c, r, n):
    """Iteratively shrink radii to guarantee strict feasibility."""
    for _ in range(100):
        changed = False
        # Boundary correction
        for i in range(n):
            d = min(c[i, 0], 1.0 - c[i, 0], c[i, 1], 1.0 - c[i, 1])
            if r[i] > d + 1e-10:
                r[i] = d
                changed = True
        # Overlap correction
        for i in range(n):
            for j in range(i + 1, n):
                dx = c[i, 0] - c[j, 0]
                dy = c[i, 1] - c[j, 1]
                d = math.sqrt(dx * dx + dy * dy + 1e-12)
                if r[i] + r[j] > d + 1e-10:
                    scale = d / (r[i] + r[j])
                    r[i] *= scale
                    r[j] *= scale
                    changed = True
        if not changed:
            break
    return c, r

def run_packing():
    n = 26
    np.random.seed(42)
    
    best_c = None
    best_r = None
    best_sum = -1.0
    
    # Generate diverse initial configurations
    inits = []
    
    # 1. Hexagonal lattice (high packing density)
    pts = []
    for i in range(6):
        for j in range(5):
            if len(pts) >= n: break
            x = j * 0.18 + (i % 2) * 0.09 + 0.05
            y = i * 0.156 + 0.05
            pts.append([x, y])
    pts = np.array(pts[:n])
    pts -= pts.min(axis=0)
    pts /= pts.max(axis=0)
    pts *= 0.8
    pts += 0.1
    inits.append((pts, np.full(n, 0.05)))
    
    # 2. Grid layout with center circle
    pts = []
    for i in range(5):
        for j in range(5):
            pts.append([0.1 + j * 0.18, 0.1 + i * 0.18])

    pts = np.array(pts[:n])
    inits.append((pts, np.full(n, 0.04)))

            for i in range(n):
                for j in range(i + 1, n):
                    diff = pts[i] - pts[j]
                    d = np.sqrt(np.sum(diff ** 2) + 1e-6)
                    f = 0.01 / (d ** 2 + 0.01)
                    forces[i] += diff * f
                    forces[j] -= diff * f
            pts += forces * 0.1
            pts = np.clip(pts, 0.1, 0.9)
        inits.append((pts, np.full(n, 0.04)))
        
    # Adaptive penalty weights
    penalty_weights = [20, 100, 500, 2000]
    
    for pts, r0 in inits:

        bounds = [(0, 1)] * (2 * n) + [(1e-4, 0.5)] * n
        current_x = x0.copy()

        for w in penalty_weights:
            res = minimize(lambda x, w=w: objective(x, n, w), current_x,

            best_c = c.copy()
