import numpy as np
from scipy.optimize import minimize
import math

def run_packing():
    """
    Pack 26 circles in a unit square [0,1]x[0,1] to maximize the sum of radii.
    
    Returns:
        tuple: (centers, radii, sum_radii)
            centers: np.array of shape (26, 2)
            radii: np.array of shape (26,)
            sum_radii: float
    """
    np.random.seed(42)
    n = 26

    # Helper function to calculate the objective and constraints
    def objective_and_penalty(variables):
        # Unpack variables: 26 circles * (x, y, r) = 78 variables
        # We reshape to (26, 3) where columns are x, y, r
        points = variables.reshape((n, 3))
        x = points[:, 0]
        y = points[:, 1]
        r = points[:, 2]

        sum_r = np.sum(r)
        penalty = 0.0

        # Boundary constraints: 0 <= x-r, x+r <= 1, same for y
        # Penalty if violated
        # Using a squared penalty for smoothness
        # Lower bound: x - r >= 0 => r - x <= 0
        # Upper bound: x + r <= 1 => x + r - 1 <= 0
        
        # Violations
        v_x_lo = np.maximum(0, r - x)
        v_x_hi = np.maximum(0, x + r - 1.0)
        v_y_lo = np.maximum(0, r - y)
        v_y_hi = np.maximum(0, y + r - 1.0)
        
        boundary_violation = np.sum(v_x_lo**2) + np.sum(v_x_hi**2) + \
                             np.sum(v_y_lo**2) + np.sum(v_y_hi**2)
        
        # Overlap constraints: dist(i, j) >= r_i + r_j
        # dist^2 >= (r_i + r_j)^2
        # We check for all pairs
        # To speed up, we can vectorize, but n=26 is small enough for loops or simple broadcasting
        # Broadcasting for pairwise distances
        # x_diff shape (26, 26)
        x_diff = x[:, None] - x[None, :]
        y_diff = y[:, None] - y[None, :]
        r_sum = r[:, None] + r[None, :]
        
        dist_sq = x_diff**2 + y_diff**2
        min_dist_sq = r_sum**2
        
        # Overlap amount squared if dist < r_sum
        # violation = max(0, r_sum - dist)
        # We use dist to handle sqrt, or just check squares.
        # If dist_sq < min_dist_sq, there is overlap.
        # Overlap distance = r_sum - sqrt(dist_sq)
        # But to avoid sqrt in penalty, we can use:
        # violation = max(0, r_sum^2 - dist_sq) ? No, units don't match nicely for gradient.
        # Let's use the linear overlap for penalty
        
        # Compute distance matrix
        dist = np.sqrt(np.maximum(dist_sq, 1e-12)) # Avoid nan
        overlap = r_sum - dist
        
        # Only consider upper triangle to avoid double counting
        # Mask for i < j
        mask = np.triu(np.ones((n, n), dtype=bool), k=1)
        overlaps = overlap[mask]
        
        overlap_violation = np.sum(np.maximum(0, overlaps)**2)

        # Total penalty
        # We need to maximize sum_r, so we minimize -sum_r + penalty
        # Penalty weight should be high enough to enforce constraints
        # But if penalty is too high, gradients might explode.
        # Adaptive or static weight.
        penalty_weight = 100.0
        penalty = penalty_weight * (boundary_violation + overlap_violation)
        
        return -sum_r + penalty

    # Initialization
    # Try to place circles in a dense pattern first
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.09) # Start slightly smaller than 0.1
    
    # Hexagonal grid initialization for better start
    # 5 rows, varying columns
    # Row heights: r + k * sqrt(3)*r
    r_init = 0.09
    h = math.sqrt(3) * r_init
    
    idx = 0
    rows = [5, 5, 5, 5, 6] # Total 26
    # Adjust rows to fit in width 1
    # Width needed for m circles: 2r + (m-1)2r = 2mr ? No, centers span (m-1)2r

    # Let's just place them and let optimizer fix positions.
    
    y_pos = r_init
    for i, count in enumerate(rows):
        # Shift odd rows
        shift = 0
        if i % 2 == 1:
            shift = r_init # Shift by radius (half diameter)
        
        x_start = r_init + shift
        # We need to fit 'count' circles.
        # If count is large, spacing might be tight.
        # Let's distribute them evenly in [shift, 1-shift] roughly
        # But standard hex spacing is 2r.
        # If 2r*count > 1, we compress.
        
        # Calculate available width
        if shift > 0:
            # Left margin is shift (center at shift + r? No, center at shift+r if wall is 0)
            # Actually if we shift by r, first center is at r+r = 2r?
            # Let's just place centers at r + j*2r + shift
            # Clamp to [r, 1-r]
            pass

        # Simple linear distribution for initialization
        margin = r_init
        width_available = 1.0 - 2 * margin
        if count > 1:
            spacing = width_available / (count - 1)
        else:
            spacing = 0
            
        for j in range(count):
            if idx < n:
                cx = margin + j * spacing
                cy = y_pos
                centers[idx] = [cx, cy]
                radii[idx] = r_init
                idx += 1

    # r in [0, 1] (loose upper bound)

    for i in range(n):

    # L-BFGS-B is usually robust.

    return centers_opt, radii_opt, sum_radii
