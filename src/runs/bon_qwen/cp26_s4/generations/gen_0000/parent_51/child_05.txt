import numpy as np
from scipy.optimize import minimize

def objective(vars):
    """Objective function: minimize negative sum of radii"""
    return -np.sum(vars[2::3])

def boundary_con(v, i, t):
    """Boundary constraint: ensures circle i stays inside [0,1]x[0,1]
    t=0: x >= r
    t=1: x <= 1-r
    t=2: y >= r
    t=3: y <= 1-r
    """
    idx = i * 3
    if t == 0: return v[idx] - v[idx+2]
    if t == 1: return 1.0 - v[idx] - v[idx+2]
    if t == 2: return v[idx+1] - v[idx+2]
    if t == 3: return 1.0 - v[idx+1] - v[idx+2]
    return 0.0

def pair_con(v, i, j):
    """Non-overlap constraint: distance^2 >= (r_i + r_j)^2"""
    idx_i = i * 3
    idx_j = j * 3
    dx = v[idx_i] - v[idx_j]
    dy = v[idx_i+1] - v[idx_j+1]
    dr = v[idx_i+2] + v[idx_j+2]
    return dx*dx + dy*dy - dr*dr

def run_packing():
    n = 26
    cons = []
    
    # Add boundary constraints for each circle
    for i in range(n):
        for t in range(4):
            cons.append({'type': 'ineq', 'fun': boundary_con, 'args': (i, t)})
            
    # Add pairwise non-overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            cons.append({'type': 'ineq', 'fun': pair_con, 'args': (i, j)})
            
    # Variable bounds: x, y in [0,1], r in [1e-6, 0.5]
    bnds = [(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)] * n
    
    best_res = None
    best_val = -np.inf
    
    # Prepare clipping bounds for initialization
    low = np.array([0.0, 0.0, 0.0] * n)
    high = np.array([1.0, 1.0, 0.5] * n)
    
    # Run optimization with multiple restarts from hexagonal grids
    for seed in range(10):
        np.random.seed(seed)
        pts = []
        r_init = 0.07
        y = 0.1 + r_init
        while len(pts) < n:
            x = 0.1 + r_init
            while len(pts) < n:
                pts.append([x, y, r_init])
                x += 0.17
            y += 0.15
            
        x0 = np.array(pts[:n]).flatten()
        
        # Add random perturbation to break symmetry
        x0 += np.random.uniform(-0.01, 0.01, size=x0.shape)
        x0 = np.clip(x0, low, high)
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bnds, constraints=cons, 
                           options={'maxiter': 5000, 'ftol': 1e-12})
            if -res.fun > best_val:
                best_val = -res.fun
                best_res = res
        except Exception:
            pass
            
    if best_res is None:
        best_res = minimize(objective, x0, method='SLSQP', bounds=bnds, constraints=cons)
        
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        centers[i] = best_res.x[i*3:i*3+2]
        radii[i] = best_res.x[i*3+2]
        
    # Strict post-processing to guarantee validation passes
    for i in range(n):
        r = radii[i]
        cx, cy = centers[i]
        # Enforce boundary constraints
        r = min(r, cx, 1.0 - cx, cy, 1.0 - cy)
        if r < 1e-6: r = 1e-6
        radii[i] = r
        centers[i] = [np.clip(cx, r, 1.0 - r), np.clip(cy, r, 1.0 - r)]
        
    return centers, radii, np.sum(radii)
