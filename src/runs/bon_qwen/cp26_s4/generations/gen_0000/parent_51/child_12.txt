import numpy as np
from scipy.optimize import minimize

def get_initial_guess():
    # Initialize centers and radii for 26 circles using a hexagonal lattice
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # We aim for a hexagonal pattern. 
    # Let's try to fit rows of 5 and 4 circles.
    # Pattern: 5, 4, 5, 4, 5, 4 -> 27 circles. We will drop one to get 26.
    # Initial radius guess (small enough to fit, close to target ~0.101)
    r_guess = 0.095
    
    dy = np.sqrt(3) * r_guess
    dx = 2 * r_guess
    
    idx = 0
    rows = [5, 4, 5, 4, 5, 4] # 6 rows
    
    for r_idx, count in enumerate(rows):
        y_coord = r_guess + r_idx * dy
        
        # Shift x for even rows (1, 3, 5) to create hexagonal stagger
        if r_idx % 2 == 1:
            x_start = 2 * r_guess
        else:
            x_start = r_guess
            
        for c_idx in range(count):
            if idx >= n:
                break
            x_coord = x_start + c_idx * dx
            
            # Safety check for boundary (though with r_guess=0.095 this should be fine)
            if x_coord + r_guess > 1.0 or x_coord - r_guess < 0.0:
                continue
                
            centers[idx] = [x_coord, y_coord]
            radii[idx] = r_guess
            idx += 1
            
        if idx >= n:
            break
            
    # Ensure we have exactly 26 circles
    centers = centers[:n]
    radii = radii[:n]
    
    # Flatten parameters for optimizer: [x1, y1, r1, x2, y2, r2, ...]
    # Order: x0, x1... x25, y0, y1... y25, r0, r1... r25
    # Actually, simpler to pack as (x0, y0, r0), (x1, y1, r1)...
    # But scipy expects a 1D array.
    # Let's use order: [x0, y0, r0, x1, y1, r1, ...]
    
    init_params = np.zeros(3 * n)
    for i in range(n):
        init_params[3*i] = centers[i, 0]
        init_params[3*i+1] = centers[i, 1]
        init_params[3*i+2] = radii[i]
        
    return init_params, n

def unpack_params(params, n):
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        centers[i, 0] = params[3*i]
        centers[i, 1] = params[3*i+1]
        radii[i] = params[3*i+2]
    return centers, radii

def objective(params, n):
    # We want to maximize sum of radii, so we minimize negative sum
    _, radii = unpack_params(params, n)
    return -np.sum(radii)

def get_constraints(n):
    constraints = []
    
    # 1. Boundary constraints: 0 <= x-r, x+r <= 1, 0 <= y-r, y+r <= 1
    # Equivalent to: r <= x <= 1-r and r <= y <= 1-r
    # Which implies: x - r >= 0, 1 - (x + r) >= 0, y - r >= 0, 1 - (y + r) >= 0
    
    def con_x_lower(params):
        centers, radii = unpack_params(params, n)
        return centers[:, 0] - radii
    def con_x_upper(params):
        centers, radii = unpack_params(params, n)
        return 1.0 - (centers[:, 0] + radii)
    def con_y_lower(params):
        centers, radii = unpack_params(params, n)
        return centers[:, 1] - radii
    def con_y_upper(params):
        centers, radii = unpack_params(params, n)
        return 1.0 - (centers[:, 1] + radii)
    def con_r_nonneg(params):
        _, radii = unpack_params(params, n)
        return radii

    constraints.append({'type': 'ineq', 'fun': con_x_lower})
    constraints.append({'type': 'ineq', 'fun': con_x_upper})
    constraints.append({'type': 'ineq', 'fun': con_y_lower})
    constraints.append({'type': 'ineq', 'fun': con_y_upper})
    constraints.append({'type': 'ineq', 'fun': con_r_nonneg})
    
    # 2. Non-overlap constraints: dist >= r_i + r_j
    # dist^2 >= (r_i + r_j)^2
    # (xi - xj)^2 + (yi - yj)^2 - (ri + rj)^2 >= 0
    
    for i in range(n):
        for j in range(i + 1, n):
            def con_overlap(idx1=i, idx2=j):
                def fun(params):
                    centers, radii = unpack_params(params, n)
                    c1, c2 = centers[idx1], centers[idx2]
                    r1, r2 = radii[idx1], radii[idx2]
                    dist_sq = (c1[0] - c2[0])**2 + (c1[1] - c2[1])**2
                    sum_r = r1 + r2
                    return dist_sq - sum_r**2
                return fun
            constraints.append({'type': 'ineq', 'fun': con_overlap()})
            
    return constraints

def run_packing():
    init_params, n = get_initial_guess()
    
    # Bounds for parameters: 
    # x in [0, 1], y in [0, 1], r in [0, 0.5] (r cannot exceed 0.5)
    bounds = []
    for i in range(n):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 0.5)) # r

    cons = get_constraints(n)
    
    # Optimization options
    options = {'maxiter': 1000, 'ftol': 1e-10, 'disp': False}
    
    # Run optimizer
    res = minimize(
        objective, 
        init_params, 
        method='SLSQP', 
        bounds=bounds, 
        constraints=cons, 
        options=options
    )
    
    if res.success:
        final_centers, final_radii = unpack_params(res.x, n)
        sum_radii = np.sum(final_radii)
        
        # Apply a small epsilon to ensure strict validity against float errors if needed, 
        # but usually the solver respects constraints. 
        # The validation function allows 1e-12 tolerance.
        
        return final_centers, final_radii, sum_radii
    else:
        # Fallback to initial guess if optimization fails
        centers, radii = unpack_params(init_params, n)
        return centers, radii, np.sum(radii)
