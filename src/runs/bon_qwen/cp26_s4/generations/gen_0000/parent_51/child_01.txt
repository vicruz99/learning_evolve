import numpy as np
from scipy.optimize import minimize

def compute_violations(params, n):
    """
    Computes the sum of squared violations for boundary and overlap constraints.
    Violations are positive when constraints are broken, zero otherwise.
    """
    centers = params[:2*n].reshape(n, 2)
    radii = params[2*n:]
    
    # Boundary violations: circle i must satisfy r_i <= x_i <= 1-r_i and r_i <= y_i <= 1-r_i
    # We penalize r_i - x_i < 0, x_i + r_i - 1 > 0, etc.
    viol_x_left = np.maximum(0, radii - centers[:, 0])**2
    viol_x_right = np.maximum(0, radii + centers[:, 0] - 1)**2
    viol_y_bottom = np.maximum(0, radii - centers[:, 1])**2
    viol_y_top = np.maximum(0, radii + centers[:, 1] - 1)**2
    
    bound_viol = np.sum(viol_x_left + viol_x_right + viol_y_bottom + viol_y_top)
    
    # Overlap violations: dist(i,j) >= r_i + r_j
    # Vectorized distance calculation
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.linalg.norm(diff, axis=2)
    np.fill_diagonal(dists, np.inf)  # Ignore self-distance
    
    min_dists = radii[:, np.newaxis] + radii[np.newaxis, :]
    # Penalize when min_dists > dists
    overlap_viol = np.sum(np.maximum(0, min_dists - dists)**2)
    
    return bound_viol + overlap_viol

def objective_function(params, n, penalty_weight):
    """
    Objective to minimize: -sum(radii) + penalty * violations
    """
    sum_radii = np.sum(params[2*n:])
    violations = compute_violations(params, n)
    return -sum_radii + penalty_weight * violations

    n = 26
    rng = np.random.default_rng(42)
    
    # Initialize centers randomly within the central region to avoid immediate boundary conflicts
    # Radii start small to allow optimization to expand them freely
    centers = rng.uniform(0.15, 0.85, size=(n, 2))
    radii = np.ones(n) * 0.05
    params = np.concatenate([centers.flatten(), radii])
    
    # Bounds for coordinates and radii
    bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n
    
    current_penalty = 50.0
    best_params = params.copy()
    best_score = -np.inf
    
    # Iterative optimization with increasing penalty weight
    # This anneals the constraints, allowing the optimizer to first find a valid layout, then expand circles
    for epoch in range(40):
        res = minimize(objective_function, params, args=(n, current_penalty), method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': 200, 'ftol': 1e-12})
        params = res.x
        score = np.sum(params[2*n:])
        if score > best_score:
            best_score = score
            best_params = params.copy()
            
        # Increase penalty to enforce constraints more strictly
        current_penalty *= 1.5
        
        # Inject small noise to help escape local minima
        noise = rng.normal(0, 1e-5, size=params.shape)
        params += noise
        params[:2*n] = np.clip(params[:2*n], 0, 1)
        params[2*n:] = np.clip(params[2*n:], 0, 0.5)
        
    # Start final refinement from the best found configuration
    params = best_params
    
    # Use SLSQP for final high-precision refinement with a large penalty
    res = minimize(objective_function, params, args=(n, current_penalty*2), method='SLSQP', bounds=bounds,
                   options={'maxiter': 1000, 'ftol': 1e-14})
    params = res.x
    
    centers = params[:2*n].reshape(n, 2)
    radii = params[2*n:]
    
    # Post-processing to strictly enforce validity against numerical precision limits
    # This guarantees the validate_packing function will return True
    for _ in range(100):

        # Resolve overlaps

            for j in range(i+1, n):

                dist = np.sqrt(dx*dx + dy*dy)

                    radii[j] -= shrink
