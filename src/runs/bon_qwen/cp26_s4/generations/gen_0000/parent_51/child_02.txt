from scipy.optimize import minimize

def run_packing():

    points = []
    spacing_x = 1.0 / 6.0

    for row in range(5):
        y = 0.15 + row * spacing_y

        while len(points) < n_circles:

            points.append([px, py])
            
    # Initialize variables: [x1, y1, r1, x2, y2, r2, ...]
    # Start with small radius to ensure feasibility
    r_initial = 0.03
    initial_vars = []

        initial_vars.extend([p[0], p[1], r_initial])
        
    initial_vars = np.array(initial_vars)
    
    def objective(vars):
        total_r = 0
        barrier = 0
        penalty_factor = 1e5 # Large penalty for constraint violation
        
        radii = vars[2::3]
        total_r = np.sum(radii)
        
        # Boundary barriers: x >= r, 1-x >= r, etc.
        # Slack = x - r. If x < r, slack < 0 -> log undefined (handled by penalty)
        # We use max(0, slack) to avoid NaN if it goes negative, but ideally we stay feasible.
        
        for i in range(n_circles):
            idx = i * 3
            x, y, r = vars[idx], vars[idx+1], vars[idx+2]
            
            # Boundary slacks
            slacks = [x - r, 1.0 - x - r, y - r, 1.0 - y - r]
            for s in slacks:
                if s > 1e-9:
                    barrier += np.log(s)
                else:
                    barrier -= penalty_factor * (1e-9 - s) # Quadratic-ish penalty for severe violation

        # d^2 >= (r_i + r_j)^2

                if slack > 1e-12:
                    barrier += np.log(slack)
                else:
                    barrier -= penalty_factor * (1e-12 - slack)
                    
        # We want to maximize sum of radii, so we minimize -sum - barrier
        return -total_r - 0.01 * barrier

    best_result = None

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        return False

    if np.isnan(radii).any():
        return False

    # Check if radii are nonnegative
    for i in range(n):
        if radii[i] < 0:
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False

    return True
