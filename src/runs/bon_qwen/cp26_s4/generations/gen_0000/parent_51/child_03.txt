import numpy as np
from scipy.optimize import minimize

def get_constraints(centers_radii, n):
    """
    Computes all constraints for the circle packing problem.
    Returns a 1D numpy array of constraint values (must be >= 0).
    
    Constraints:
    1. Boundary: x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
    2. Overlap: dist_sq - (r_i + r_j)^2 >= 0 for all i < j
    """
    x = centers_radii[:, 0]
    y = centers_radii[:, 1]
    r = centers_radii[:, 2]
    
    # Boundary constraints (4 per circle)
    # x >= r  => x - r >= 0
    # x <= 1 - r => 1 - x - r >= 0
    # y >= r  => y - r >= 0
    # y <= 1 - r => 1 - y - r >= 0
    bounds_constraints = np.concatenate([
        x - r,
        1.0 - x - r,
        y - r,
        1.0 - y - r
    ])
    
    # Overlap constraints
    # dist^2 >= (r_i + r_j)^2
    # dist^2 = (x_i - x_j)^2 + (y_i - y_j)^2
    
    # Compute pairwise differences using broadcasting
    # Shape (n, n, 2)
    diff_xy = centers_radii[:, :2, np.newaxis] - centers_radii[:, :2]
    # Sum of squares over the last axis (x and y dims)
    # Shape (n, n)
    dist_sq = np.sum(diff_xy**2, axis=2)
    
    # Sum of radii squared
    # r shape (n,), r_sum shape (n, n)
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    r_sum_sq = r_sum**2
    
    # Constraint: dist_sq - r_sum_sq >= 0
    # We only need upper triangle (i < j) to avoid duplicates and self-checks
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    overlap_constraints = dist_sq[mask] - r_sum_sq[mask]
    
    return np.concatenate([bounds_constraints, overlap_constraints])
