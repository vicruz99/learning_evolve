1.  **Initialization**: A hexagonal lattice pattern is used to initialize the centers. This allows for a higher density of circles compared to a square grid. We fit as many circles as possible in rows and columns, with alternating shifts.
2.  **Optimization**: We formulate the problem as minimizing a penalty function. The objective is to maximize the sum of radii ($-\sum r_i$). We add penalties for:

    *   **Overlap Violations**: Circles intersecting with each other.
3.  **Solver**: We use `scipy.optimize.minimize` with the `L-BFGS-B` method. This optimizer is robust for bound-constrained problems. We initialize with small radii to ensure a feasible starting point, then the optimizer expands them until constraints are hit.

import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    def objective(params):
        # Unpack parameters
        centers = params[:2*n].reshape(n, 2)
        radii = params[2*n:]
        
        # 1. Objective: Maximize sum of radii (minimize negative sum)
        loss = -np.sum(radii)
        
        # 2. Boundary Penalties
        # Circles must be inside [0, 1] x [0, 1]
        # Constraints: r <= x, r <= 1-x, r <= y, r <= 1-y
        x, y = centers[:, 0], centers[:, 1]
        r = radii
        
        # Calculate violations (clamped at 0)
        v_left = np.maximum(0, r - x)
        v_right = np.maximum(0, r - (1 - x))
        v_bottom = np.maximum(0, r - y)
        v_top = np.maximum(0, r - (1 - y))
        
        boundary_penalty = np.sum(v_left**2 + v_right**2 + v_bottom**2 + v_top**2)
        
        # 3. Overlap Penalties
        # Circles must not overlap: dist >= r_i + r_j
        # Penalty: max(0, r_i + r_j - dist)^2
        
        # Vectorized distance calculation for all pairs
        # diff shape (n, n, 2)
        diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        
        # Sum of radii matrix shape (n, n)
        r_sums = r[:, np.newaxis] + r[np.newaxis, :]
        
        # Violations matrix
        # We only care about i < j to avoid double counting and self-check
        # But for penalty sum, full matrix with mask is fine
        # dists[i, i] = 0, r_sums[i, i] = 2r_i -> violation 2r_i (always penalized)
        # We must exclude diagonal.
        
        mask = np.triu(np.ones((n, n), dtype=bool), k=1)
        violations = np.maximum(0, r_sums - dists)
        overlap_penalty = np.sum(violations[mask]**2)
        
        # Weights for penalties
        # Large weights enforce constraints strictly
        penalty_weight = 10000.0
        
        total_penalty = boundary_penalty + overlap_penalty
        loss += penalty_weight * total_penalty
        
        return loss

    best_result = None
    best_score = -np.inf
    
    # Try multiple restarts to find global optimum
    # Strategy 1: Hexagonal Grid Initialization
    # Strategy 2: Random Perturbations
    
    rng = np.random.RandomState(123)
    
    # Helper to run optimization
    def optimize_run(init_centers, init_radii):
        params = np.hstack([init_centers.flatten(), init_radii])
        bounds = [(0, 1)] * (2 * n) + [(0, 0.5)] * n
        
        try:
            res = minimize(objective, params, method='L-BFGS-B', bounds=bounds, 
                           options={'ftol': 1e-12, 'gtol': 1e-12, 'maxiter': 50000, 'iprint': -1})
            return res
        except Exception:
            return None

    # --- Initialization 1: Hexagonal Lattice ---
    centers_list = []
    y_step = 0.175  # Approx spacing for r ~ 0.1
    x_step = 0.200
    
    row_idx = 0
    y = 0.1
    while y + 0.1 <= 1.0: # Keep some margin
        # Determine number of circles in this row
        # Even rows (0, 2...) start at 0.1, Odd rows start at 0.2
        start_x = 0.1 + (0.1 if row_idx % 2 != 0 else 0)
        x = start_x
        while x + 0.1 <= 1.0:
            if len(centers_list) < n:
                centers_list.append([x, y])
            x += x_step
        y += y_step
        row_idx += 1
    
    # If we have fewer than n, fill with random or specific spots
    # If more, truncate
    centers_arr = np.array(centers_list[:n])
    
    # If we don't have enough, add random points
    while len(centers_arr) < n:
        centers_arr = np.vstack([centers_arr, rng.rand(1, 2) * 0.8 + 0.1])
    centers_arr = centers_arr[:n]
    
    # Initial radii small
    init_radii = np.full(n, 0.01)
    
    res1 = optimize_run(centers_arr, init_radii)
    if res1 is not None:
        s1 = -objective(res1.x[:2*n].reshape(n,2), res1.x[2*n:]) + 10000 * 0 # Just evaluate sum
        # Actually extract sum directly
        r1 = res1.x[2*n:]
        s1 = np.sum(r1)
        if s1 > best_score:
            best_score = s1
            best_result = res1

    # --- Initialization 2: Random with Repulsion (Simulated Annealing style start) ---
    for _ in range(5): # 5 random restarts
        # Generate random points
        c_rand = rng.rand(n, 2)
        # Simple repulsion to spread them out
        for _ in range(100):
            forces = np.zeros_like(c_rand)
            for i in range(n):
                for j in range(i+1, n):
                    d = c_rand[i] - c_rand[j]
                    dist = np.sqrt(np.sum(d**2))
                    if dist < 0.2 and dist > 1e-5:
                        f = (0.2 - dist) / dist * d # Push apart
                        forces[i] += f
                        forces[j] -= f
            c_rand += 0.01 * forces
            c_rand = np.clip(c_rand, 0, 1)
            
        res_rand = optimize_run(c_rand, np.full(n, 0.01))
        if res_rand is not None:
            r_rand = res_rand.x[2*n:]
            s_rand = np.sum(r_rand)
            if s_rand > best_score:
                best_score = s_rand
                best_result = res_rand

    # Extract final solution
    if best_result is None:
        # Fallback
        return np.zeros((n, 2)), np.zeros(n), 0.0

    radii_opt = best_result.x[2*n:]
    
    # Clip radii to be non-negative just in case
    radii_opt = np.maximum(radii_opt, 0)
    
    # Final sum
    final_sum = np.sum(radii_opt)
    
    # Sort by radius for consistent output? Not required but nice.
    # Actually, order doesn't matter.
    
    return centers_opt, radii_opt, final_sum
