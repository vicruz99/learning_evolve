    *   Iteratively call `scipy.optimize.minimize` on the objective function $F$.

import numpy as np
from scipy.optimize import linprog, minimize

def solve_radii_lp(centers):
    """
    Solves for optimal radii given fixed centers using Linear Programming.
    Maximizes sum of radii subject to non-overlap and boundary constraints.
    
    Args:
        centers: np.array of shape (n, 2)
        
    Returns:
        tuple: (radii array, sum of radii)
    """
    n = centers.shape[0]
    # Objective: Maximize sum(r_i) => Minimize -sum(r_i)
    c = -np.ones(n)
    
    # We will construct inequality constraints A_ub @ r <= b_ub
    # Constraints:
    # 1. Pairwise: r_i + r_j <= distance(i, j)
    # 2. Boundary: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    # 3. Non-negativity: r_i >= 0 (handled by bounds)
    
    rows = []
    bounds_vals = []
    
    # Pairwise constraints
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.hypot(centers[i, 0] - centers[j, 0], 
                            centers[i, 1] - centers[j, 1])
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            rows.append(row)
            bounds_vals.append(dist)
            
    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        
        # r_i <= x
        row = np.zeros(n); row[i] = 1.0
        rows.append(row)
        bounds_vals.append(x)
        
        # r_i <= 1-x
        row = np.zeros(n); row[i] = 1.0
        rows.append(row)
        bounds_vals.append(1.0 - x)
        
        # r_i <= y
        row = np.zeros(n); row[i] = 1.0
        rows.append(row)
        bounds_vals.append(y)
        
        # r_i <= 1-y
        row = np.zeros(n); row[i] = 1.0
        rows.append(row)
        bounds_vals.append(1.0 - y)
        
    A_ub = np.array(rows)
    b_ub = np.array(bounds_vals)
    bounds_r = [(0, None) for _ in range(n)]
    
    try:
        # Use 'highs' solver for efficiency
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method='highs')
        if res.success:
            return res.x, np.sum(res.x)
        else:
            # Fallback if LP fails (should not happen as 0 is feasible)
            return np.zeros(n), 0.0
    except Exception:
        return np.zeros(n), 0.0

def objective_func(centers_flat):

    Objective function for center optimization.
    Returns negative sum of radii (to be minimized).
    """
    centers = centers_flat.reshape(-1, 2)
    _, s = solve_radii_lp(centers)
    return -s
