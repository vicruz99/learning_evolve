import numpy as np

def get_optimal_radii_and_sum(centers):
    """
    Given centers, solve the LP to find optimal radii and return the sum of radii.
    Returns (sum_radii, radii).
    """
    n = centers.shape[0]
    
    # Compute pairwise distances
    # Using vectorized operations for speed
    # centers shape (n, 2)
    # diff shape (n, n, 2)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    # Compute boundary distances: min(x, 1-x, y, 1-y)
    # centers[:, 0] is x, centers[:, 1] is y
    x = centers[:, 0]
    y = centers[:, 1]
    b_vals = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
    
    # LP Problem: Maximize sum(r_i)
    # Equivalent to Minimize -sum(r_i)
    # Variables: r_0, ..., r_{n-1}
    c_obj = -np.ones(n)
    
    # Constraints:
    # 1. r_i + r_j <= dists[i, j] for all i < j
    # 2. r_i <= b_vals[i] for all i
    # 3. r_i >= 0 (handled by bounds)
    
    # Construct A_ub and b_ub
    # Number of constraints: n*(n-1)/2 pairs + n boundary constraints
    num_pairs = n * (n - 1) // 2
    num_bounds = n
    num_constraints = num_pairs + num_bounds
    
    A_ub = np.zeros((num_constraints, n))
    b_ub = np.zeros(num_constraints)
    
    row = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[row, i] = 1.0
            A_ub[row, j] = 1.0
            b_ub[row] = dists[i, j]
            row += 1
            
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = b_vals[i]
        row += 1
        
    # Bounds for radii: r_i >= 0
    bounds = [(0.0, None) for _ in range(n)]
    
    # Solve LP
    # method='highs' is efficient
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        radii = res.x
        # Ensure non-negative (numerical safety)
        radii = np.maximum(radii, 0.0)
        return -res.fun, radii
    else:
        # If LP fails, return 0 sum and small radii to avoid crash
        return 0.0, np.zeros(n)

    """

    centers = np.clip(centers, 0.0, 1.0)
    
    sum_r, _ = get_optimal_radii_and_sum(centers)
    return -sum_r
