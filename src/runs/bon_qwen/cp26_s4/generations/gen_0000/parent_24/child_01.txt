import numpy as np
from scipy.optimize import minimize

def get_initial_guess():
    """
    Returns initial centers and radii based on a 5x5 grid + 1 small circle strategy.
    This provides a strong starting point with sum of radii approx 2.54.
    """
    # 5x5 grid parameters
    rows = 5
    cols = 5
    r_base = 0.1
    x_coords = np.linspace(r_base, 1 - r_base, cols)
    y_coords = np.linspace(r_base, 1 - r_base, rows)
    
    centers = []
    radii = []
    
    # Add 25 circles in a grid
    for y in y_coords:
        for x in x_coords:
            centers.append([x, y])
            radii.append(r_base)
            
    # Add 26th circle in a gap (e.g., between 0.1 and 0.3)
    # Position at 0.2, 0.2. Distance to nearest is sqrt(0.1^2+0.1^2) approx 0.141.
    # Max radius approx 0.041. Start small to be safe.
    centers.append([0.2, 0.2])
    radii.append(0.01)
    
    return np.array(centers), np.array(radii)

def compute_loss_and_constraints(variables, n=26):
    """
    Computes the negative sum of radii (to maximize) and a penalty for constraint violations.
    Variables are flattened: [x1, y1, r1, x2, y2, r2, ...]
    """
    centers = variables[0:2*n].reshape((n, 2))
    radii = variables[2*n:]
    
    # 1. Objective: Minimize -sum(radii)
    obj = -np.sum(radii)
    
    # 2. Constraint Penalties
    penalty = 0.0
    weight = 1000.0
    
    # Boundary constraints: 0 <= center - radius and center + radius <= 1
    # Equivalent to: radius <= center <= 1-radius
    # Penalty if violated
    lower_bound_violation = np.maximum(0, radii - centers[:, 0]) + np.maximum(0, radii - centers[:, 1])
    upper_bound_violation = np.maximum(0, (centers[:, 0] + radii) - 1) + np.maximum(0, (centers[:, 1] + radii) - 1)
    radius_negative_violation = np.maximum(0, -radii)
    
    boundary_violation = np.sum(lower_bound_violation) + np.sum(upper_bound_violation) + np.sum(radius_negative_violation)
    penalty += weight * boundary_violation
    
    # Overlap constraints: dist >= r_i + r_j
    # dist^2 >= (r_i + r_j)^2
    # Violation if dist < r_i + r_j
    # We use a soft penalty: max(0, (r_i + r_j - dist))^2
    # Vectorized overlap check
    # Centers shape (n, 2)
    # diff shape (n, n, 2)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    radii_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    
    # Only check upper triangle to avoid double counting and self-check
    indices = np.triu_indices(n, k=1)
    overlap_violations = radii_sum[indices] - dists[indices]
    overlap_violations = np.maximum(0, overlap_violations)
    
    # Squared penalty for smoother gradient
    penalty += weight * np.sum(overlap_violations**2)
    
    return obj + penalty, radii

def optimize_packing(centers, radii):
    """
    Optimizes the packing configuration.
    """
    n = len(radii)
    initial_vars = np.concatenate([centers.flatten(), radii])
    
    # Bounds: 
    # x, y in [0, 1]
    # r >= 0 (lower bound 0, upper bound 0.5 roughly)
    bounds = []
    for _ in range(n):
        bounds.extend([(0, 1), (0, 1)]) # x, y
        bounds.extend([(1e-6, 0.5)])     # r
    
    options = {'maxiter': 2000, 'ftol': 1e-12, 'disp': False}
    
    try:
        res = minimize(
            compute_loss_and_constraints,
            initial_vars,
            args=(n,),
            method='SLSQP',
            bounds=bounds,
            options=options
        )
        best_centers = res.x[0:2*n].reshape((n, 2))
        best_radii = res.x[2*n:]
        best_sum = np.sum(best_radii)
        return best_centers, best_radii, best_sum
    except Exception:
        return centers, radii, np.sum(radii)

def run_packing():
    """
    Main function to run the packing optimization.
    Returns (centers, radii, sum_radii).
    """
    np.random.seed(42)
    
    # 1. Get robust initial guess
    centers, radii = get_initial_guess()
    
    # 2. Primary Optimization
    centers, radii, total_sum = optimize_packing(centers, radii)
    
    # 3. Perturbation and Secondary Optimization (to escape local minima)
    # Try a few random restarts based on the best found so far
    best_centers = centers
    best_radii = radii
    best_sum = total_sum
    
    for _ in range(3):
        perturbed_centers = centers + np.random.normal(0, 0.005, centers.shape)
        perturbed_radii = radii + np.random.normal(0, 0.001, radii.shape)
        # Clip radii to be positive
        perturbed_radii = np.maximum(perturbed_radii, 1e-6)
        
        c, r, s = optimize_packing(perturbed_centers, perturbed_radii)
        if s > best_sum:
            best_centers = c
            best_radii = r
            best_sum = s
            
    return best_centers, best_radii, best_sum
