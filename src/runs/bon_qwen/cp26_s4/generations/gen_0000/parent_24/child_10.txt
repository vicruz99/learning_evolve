def run_packing():

    Pack 26 circles in a unit square to maximize sum of radii.
    Returns (centers, radii, sum_radii).

    n_circles = 26

    def compute_violations(centers, r):
        """
        Computes the sum of squared violations for a given configuration and radius.
        Violations include:
        1. Overlap between circles (distance < 2r)
        2. Circles crossing boundaries (x < r, x > 1-r, etc.)
        """
        violations = 0.0
        
        # Inter-circle violations
        # Vectorized distance calculation
        # centers shape: (N, 2)
        diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :] # (N, N, 2)
        dists_sq = np.sum(diff**2, axis=2) # (N, N)
        
        # Avoid self-distance issues
        np.fill_diagonal(dists_sq, 2 * r) # Distance to self is effectively infinite or ignored, 
                                          # but here we just want to avoid sqrt(0) if used in gradient logic, 
                                          # though we don't compute gradient here.
                                          # For violation check, self distance doesn't matter.
                                          # Actually, we only care about i < j.
        
        dists = np.sqrt(dists_sq)
        
        # Lower triangle (i < j)
        mask = np.tril(np.ones((n_circles, n_circles), dtype=bool), k=-1)
        relevant_dists = dists[mask]
        
        # Overlap condition: dist < 2r
        # Violation amount: 2r - dist
        overlap_violations = 2 * r - relevant_dists
        positive_overlaps = np.maximum(0, overlap_violations)
        violations += np.sum(positive_overlaps**2)
        
        # Boundary violations
        # x < r -> violation r - x
        # x > 1-r -> violation x - (1-r)
        # same for y
        
        # x boundaries
        x_viol_low = np.maximum(0, r - centers[:, 0])
        x_viol_high = np.maximum(0, centers[:, 0] - (1 - r))
        violations += np.sum(x_viol_low**2) + np.sum(x_viol_high**2)
        
        # y boundaries
        y_viol_low = np.maximum(0, r - centers[:, 1])
        y_viol_high = np.maximum(0, centers[:, 1] - (1 - r))
        violations += np.sum(y_viol_low**2) + np.sum(y_viol_high**2)
        
        return violations

                objective, 
                x0, 
                method='L-BFGS-B',

                    if y < r_target - 1e-9 or y > 1 - r_target + 1e-9:
                        valid = False; break

                                break

        except Exception:

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    (Copied from prompt to ensure self-contained code if needed, 
     though problem statement says it's read-only, we use it for check)
    """
    import numpy as np
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True
