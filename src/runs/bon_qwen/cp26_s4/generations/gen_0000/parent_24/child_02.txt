import numpy as np
from scipy.optimize import minimize

def constraint_violation(centers, radii):
    """Calculate total constraint violation."""
    n = len(radii)
    violation = 0.0
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < 0:
            violation += (x - r) ** 2
        if x + r > 1:
            violation += (x + r - 1) ** 2
        if y - r < 0:
            violation += (y - r) ** 2
        if y + r > 1:
            violation += (y + r - 1) ** 2
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            overlap = radii[i] + radii[j] - dist
            if overlap > 0:
                violation += overlap ** 2
    return violation

def objective_with_penalty(params, n, penalty_weight):
    """Objective function with penalty for constraints."""
    centers = params[:2 * n].reshape(n, 2)
    radii = params[2 * n:]
    radii = np.maximum(radii, 0)
    violation = constraint_violation(centers, radii)
    return -np.sum(radii) + penalty_weight * violation

def force_optimization(initial_centers, initial_radii, num_iterations, n):
    """Run force-based optimization with simulated annealing."""
    centers = initial_centers.copy()
    radii = initial_radii.copy()
    
    best_sum = 0.0
    best_centers = centers.copy()
    best_radii = radii.copy()
    
    for iteration in range(num_iterations):
        forces = np.zeros((n, 2))
        temp = 0.3 * (1 - iteration / num_iterations)
        
        # Compute forces
        for i in range(n):
            margin = radii[i]
            # Boundary forces
            if centers[i, 0] - margin < 0:
                forces[i, 0] += 200 * (margin - centers[i, 0])
            if centers[i, 0] + margin > 1:
                forces[i, 0] -= 200 * (centers[i, 0] + margin - 1)
            if centers[i, 1] - margin < 0:
                forces[i, 1] += 200 * (margin - centers[i, 1])
            if centers[i, 1] + margin > 1:
                forces[i, 1] -= 200 * (centers[i, 1] + margin - 1)
            
            # Overlap repulsion forces
            for j in range(i + 1, n):
                dx = centers[j, 0] - centers[i, 0]
                dy = centers[j, 1] - centers[i, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                min_dist = radii[i] + radii[j]
                if dist < min_dist and dist > 1e-12:
                    force_strength = 300 * (min_dist - dist)
                    direction_x = dx / dist
                    direction_y = dy / dist
                    forces[i, 0] -= direction_x * force_strength
                    forces[i, 1] -= direction_y * force_strength
                    forces[j, 0] += direction_x * force_strength
                    forces[j, 1] += direction_y * force_strength
        
        # Apply forces with perturbation
        step_size = 0.02 * max(0.1, 1 - iteration / num_iterations)
        centers[:, 0] += forces[:, 0] * step_size + np.random.randn(n) * temp * 0.005
        centers[:, 1] += forces[:, 1] * step_size + np.random.randn(n) * temp * 0.005
        
        # Project to valid region
        for i in range(n):
            r = radii[i]
            centers[i, 0] = max(r + 1e-9, min(1 - r - 1e-9, centers[i, 0]))
            centers[i, 1] = max(r + 1e-9, min(1 - r - 1e-9, centers[i, 1]))
        
        # Expand radii towards maximum
        for i in range(n):
            max_r = min(centers[i, 0] - 1e-9, 1 - centers[i, 0] - 1e-9,
                       centers[i, 1] - 1e-9, 1 - centers[i, 1] - 1e-9)
            for j in range(n):
                if i != j:
                    dx = centers[i, 0] - centers[j, 0]
                    dy = centers[i, 1] - centers[j, 1]
                    dist = np.sqrt(dx * dx + dy * dy)
                    max_r = min(max_r, dist - radii[j] - 1e-9)
            
            if max_r > 0:
                radii[i] = max(0.0001, radii[i] * 0.85 + max_r * 0.15)
            else:
                radii[i] = max(0.0001, radii[i] * 0.98)
        
        current_sum = np.sum(radii)
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers.copy()
            best_radii = radii.copy()
    
    return best_centers, best_radii, best_sum
