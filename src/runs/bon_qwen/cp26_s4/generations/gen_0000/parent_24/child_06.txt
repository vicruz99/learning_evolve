import numpy as np
import scipy.optimize as opt

def get_max_radii(centers):
    """
    Solves the LP to find maximum radii for given centers.
    Maximize sum(r) subject to r_i <= wall_dist_i and r_i + r_j <= dist_ij.
    """
    n = centers.shape[0]
    if n == 0:
        return np.array([])
    
    # Objective: maximize sum(r) -> minimize -sum(r)
    c = -np.ones(n)
    
    # Constraints A_ub @ r <= b_ub
    # 1. Wall constraints: r_i <= min(x, 1-x, y, 1-y)
    # 2. Pairwise constraints: r_i + r_j <= distance
    
    A_ub = []
    b_ub = []
    
    for i in range(n):
        x, y = centers[i]
        # Wall distances
        d_wall = min(x, 1-x, y, 1-y)
        
        # Constraint: 1*r_i <= d_wall
        row = np.zeros(n)
        row[i] = 1.0
        A_ub.append(row)
        b_ub.append(d_wall)
        
        # Pairwise constraints with j > i
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.hypot(dx, dy)
            
            # Constraint: 1*r_i + 1*r_j <= dist
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dist)
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    # Bounds r_i >= 0
    bounds = [(0, None) for _ in range(n)]
    
    try:
        # Try highs method first (fast), fallback to simplex
        try:
            res = opt.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        except ValueError:
            res = opt.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='simplex')
        
        if res.success:
            return res.x
        else:
            # Fallback if LP fails
            return np.zeros(n)
    except Exception:
        return np.zeros(n)
