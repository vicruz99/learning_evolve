import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    Uses a hexagonal lattice initialization followed by numerical optimization.
    """
    n = 26
    
    # 1. Initialization: Hexagonal Lattice
    # Generate points in a hexagonal pattern to provide a good initial dense packing.
    centers = []
    x = 0.1
    y = 0.1
    dx = 0.2  # Horizontal spacing
    dy = 0.1732  # Vertical spacing (approx sqrt(3)/2 * dx)
    
    # Add points in a hexagonal arrangement
    row = 0
    while len(centers) < n:
        for col in range(n): # Sufficient range
            cx = x + col * dx + (0.1 if row % 2 == 1 else 0)
            cy = y + row * dy
            if 0 <= cx <= 1 and 0 <= cy <= 1:
                centers.append([cx, cy])
                if len(centers) == n:
                    break
        if len(centers) == n:
            break
        row += 1
        
    centers = np.array(centers[:n])

    # 2. Optimization: Maximize Minimum Distance
    # We optimize the positions to maximize the minimum pairwise distance.
    # This is equivalent to maximizing the radius r = min_dist / 2.
    
    def objective(vars):
        """Objective function: returns negative of minimum distance (to be minimized)."""
        c = vars.reshape((n, 2))
        # Calculate all pairwise distances
        diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        # Set diagonal to infinity to ignore self-distance
        np.fill_diagonal(dists, np.inf)
        min_dist = np.min(dists)
        return -min_dist # Minimize negative max_dist => Maximize max_dist

    def constraint_bound(vars):
        """Constraint: Centers must be within [0, 1]."""
        c = vars.reshape((n, 2))
        # We want 0 <= x <= 1 and 0 <= y <= 1.
        # scipy.optimize constraints are g(vars) >= 0.
        # So we use c >= 0 and 1 - c >= 0.
        # Flatten constraints: x_i, 1-x_i, y_i, 1-y_i >= 0
        bounds = np.concatenate([c, 1 - c])
        return bounds

    # Use SLSQP which supports constraints
    # Initial guess is the flattened centers array
    x0 = centers.flatten()
    
    # Define constraints
    cons = ({'type': 'ineq', 'fun': constraint_bound})

    # Run optimization
    # We use a method that handles constraints. SLSQP is a good choice.
    result = minimize(objective, x0, method='SLSQP', constraints=cons, 
                      options={'maxiter': 1000, 'ftol': 1e-9})
    
    optimal_centers = result.x.reshape((n, 2))

    # 3. Calculate Radii
    # The maximum equal radius is half the minimum distance between centers.
    diff = optimal_centers[:, np.newaxis, :] - optimal_centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    min_dist = np.min(dists)
    r = min_dist / 2.0
    
    radii = np.full(n, r)

    # 4. Final Validation and Adjustment
    # Ensure strict validity within numerical tolerance
    # The optimizer might push points slightly outside or overlap due to float precision.
    # We clamp centers and reduce radius slightly if needed to be safe.
    
    # Clamp centers to [0, 1]
    optimal_centers = np.clip(optimal_centers, 1e-9, 1 - 1e-9)
    
    # Recalculate min distance after clamping
    diff = optimal_centers[:, np.newaxis, :] - optimal_centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    min_dist = np.min(dists)
    
    # Set radius slightly smaller than half min distance to avoid overlap
    r_final = (min_dist / 2.0) * (1.0 - 1e-9)
    radii = np.full(n, r_final)
    
    # Ensure circles are inside square (adjust radius if center is too close to edge)
    # Distance to closest wall for each circle
    dist_to_walls = np.minimum(np.minimum(optimal_centers[:, 0], 1 - optimal_centers[:, 0]),
                               np.minimum(optimal_centers[:, 1], 1 - optimal_centers[:, 1]))
    
    # Radius cannot exceed distance to wall
    max_r_by_wall = np.min(dist_to_walls)
    r_final = min(r_final, max_r_by_wall)
    
    radii = np.full(n, r_final)
    
    sum_radii = np.sum(radii)
    
    return optimal_centers, radii, sum_radii
