import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes the packing of 26 circles in a unit square to maximize sum of radii.
    """
    n = 26
    np.random.seed(42)
    
    # 1. Initialization
    # Initialize centers in a grid to spread them out
    # 5x5 grid gives 25 points, add 1 more
    x_coords = np.linspace(0.1, 0.9, 5)
    y_coords = np.linspace(0.1, 0.9, 5)
    centers = []
    for y in y_coords:
        for x in x_coords:
            centers.append([x, y])
    # Add 26th center in a gap
    centers.append([0.15, 0.15])
    centers = np.array(centers)
    
    # Initial radii small to avoid overlap initially
    radii = np.full(n, 0.04)
    
    # Flatten variables: [x0, y0, r0, x1, y1, r1, ...]
    x0 = np.concatenate([centers[:, 0], centers[:, 1], radii])
    
    # Bounds for variables: x, y in [0, 1], r in [0, 0.5]
    bounds = []
    for _ in range(n):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 0.5)) # r
        
    # Penalty weight
    P = 5000.0
    
    def objective(vars):
        # Unpack variables
        x = vars[0::3]
        y = vars[1::3]
        r = vars[2::3]
        
        # Objective: Maximize sum(r) => Minimize -sum(r)
        cost = -np.sum(r)
        
        # Penalties for boundary violations
        # Circle must be within [0, 1]
        # x - r >= 0  => r - x <= 0
        # x + r <= 1  => x + r - 1 <= 0
        # Same for y
        
        # Left boundary: r - x
        pen_left = np.maximum(0, r - x)
        cost += P * np.sum(pen_left**2)
        
        # Right boundary: x + r - 1
        pen_right = np.maximum(0, x + r - 1.0)
        cost += P * np.sum(pen_right**2)
        
        # Bottom boundary: r - y
        pen_bottom = np.maximum(0, r - y)
        cost += P * np.sum(pen_bottom**2)
        
        # Top boundary: y + r - 1
        pen_top = np.maximum(0, y + r - 1.0)
        cost += P * np.sum(pen_top**2)
        
        # Penalties for overlaps
        # Distance between centers must be >= r_i + r_j
        # violation = max(0, r_i + r_j - dist)
        
        # Vectorized distance calculation for speed
        # x shape (n,), we need (n, n) matrix of distances
        # Using broadcasting
        x_col = x[:, np.newaxis]
        y_col = y[:, np.newaxis]
        
        # Calculate squared distances
        # diff_x = x[i] - x[j]
        diff_x = x_col - x_col.T
        diff_y = y_col - y_col.T
        
        # Distance matrix
        dist_matrix = np.sqrt(np.maximum(diff_x**2 + diff_y**2, 1e-15)) # avoid 0 div later if needed
        
        # Radius sum matrix
        r_col = r[:, np.newaxis]
        r_sum_matrix = r_col + r_col.T
        
        # Identify overlapping pairs (upper triangle)
        # We only need to check i < j
        # Create a mask for upper triangle
        mask = np.triu(np.ones((n, n), dtype=bool), k=1)
        
        violations = r_sum_matrix[mask] - dist_matrix[mask]
        cost += P * np.sum(np.maximum(0, violations)**2)
        
        return cost

    # 2. Optimization
    # Use L-BFGS-B for bound handling
    res = minimize(objective, x0, method='L-BFGS-B', bounds=bounds, 
                   options={'maxiter': 2000, 'ftol': 1e-12, 'gtol': 1e-8})
    
    # 3. Extract results
    x_opt = res.x[0::3]
    y_opt = res.x[1::3]
    r_opt = res.x[2::3]
    
    centers_opt = np.column_stack([x_opt, y_opt])
    
    # 4. Post-processing to ensure strict validity
    # The optimizer might leave tiny violations due to numerical precision or penalty trade-off.
    # We clamp radii to ensure they fit strictly inside and don't overlap.
    
    # Ensure inside square
    for i in range(n):
        x, y = centers_opt[i]
        r = r_opt[i]
        # Max radius allowed by boundaries
        r_max = min(x, 1 - x, y, 1 - y)
        if r > r_max:
            r_opt[i] = r_max
            
    # Ensure no overlaps (simple iterative shrinking if needed, though usually not needed if P is high)
    # Check all pairs
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt((centers_opt[i, 0] - centers_opt[j, 0])**2 + 
                           (centers_opt[i, 1] - centers_opt[j, 1])**2)
            sum_r = r_opt[i] + r_opt[j]
            if sum_r > dist:
                # Shrink both radii equally to fit
                diff = sum_r - dist
                r_opt[i] -= diff / 2
                r_opt[j] -= diff / 2
                # Ensure non-negative
                r_opt[i] = max(0.0, r_opt[i])
                r_opt[j] = max(0.0, r_opt[j])

    # Recalculate sum
    sum_radii = np.sum(r_opt)
    
    return centers_opt, r_opt, sum_radii
