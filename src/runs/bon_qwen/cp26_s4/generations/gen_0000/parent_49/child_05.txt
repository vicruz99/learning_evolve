import numpy as np
from scipy.optimize import minimize

def run_packing():
    n_circles = 26
    
    # 1. Initialize centers on a perturbed hexagonal-style grid
    # This gives a denser initial packing than a simple square grid
    centers = np.zeros((n_circles, 2))
    r = 0.095 # Initial small radius to ensure feasibility
    i = 0
    row = 0
    col = 0
    while i < n_circles:
        x = 0.1 + col * 0.15
        y = 0.1 + row * 0.13
        if row % 2 == 1:
            x += 0.075 # Shift odd rows
        
        # Wrap or limit to square
        if x + r <= 1.0 and y + r <= 1.0:
            centers[i] = [x, y]
            i += 1
        
        col += 1
        if col * 0.15 > 0.85:
            col = 0
            row += 1
            
    # Ensure we have exactly n_circles, fill rest randomly if needed
    if i < n_circles:
        np.random.seed(42)
        random_centers = np.random.rand(n_circles - i, 2) * 0.8 + 0.1
        centers[i:] = random_centers

    # 2. Define the optimization variables: [x1, y1, r1, x2, y2, r2, ...]
    initial_radii = np.full(n_circles, 0.095)
    x0 = np.zeros(3 * n_circles)
    for i in range(n_circles):
        x0[3 * i] = centers[i, 0]
        x0[3 * i + 1] = centers[i, 1]
        x0[3 * i + 2] = initial_radii[i]
        
    # Bounds for variables: x, y in [0, 1], r >= 0
    bounds = []
    for i in range(n_circles):
        bounds.extend([(0.0, 1.0), (0.0, 1.0), (1e-6, 1.0)])

    def objective(vars):
        # Objective: Maximize sum of radii -> Minimize negative sum
        return -sum(vars[3 * i + 2] for i in range(n_circles))

    def penalty(vars):
        penalty_val = 0.0
        for i in range(n_circles):
            xi = vars[3 * i]
            yi = vars[3 * i + 1]
            ri = vars[3 * i + 2]
            
            # Boundary penalties (if circle goes out of [0,1])
            if xi - ri < 0: penalty_val += (xi - ri)**2
            if xi + ri > 1: penalty_val += (xi + ri - 1)**2
            if yi - ri < 0: penalty_val += (yi - ri)**2
            if yi + ri > 1: penalty_val += (yi + ri - 1)**2
            
            # Overlap penalties
            for j in range(i + 1, n_circles):
                xj = vars[3 * j]
                yj = vars[3 * j + 1]
                rj = vars[3 * j + 2]
                
                dist = np.sqrt((xi - xj)**2 + (yi - yj)**2)
                overlap = (ri + rj) - dist
                if overlap > 0:
                    penalty_val += overlap**2
        return penalty_val

    def constrained_objective(vars):
        return objective(vars) + 1e4 * penalty(vars)

    # 3. Optimization
    # Run multiple times with slight noise to find global optimum
    best_result = None
    best_obj = float('inf')

    for run in range(5):
        # Add small noise to x0 for each run
        noise = np.random.randn(3 * n_circles) * 0.001
        current_x0 = x0 + noise
        
        result = minimize(constrained_objective, current_x0, method='SLSQP', bounds=bounds,
                          options={'ftol': 1e-8, 'maxiter': 500})
        
        if result.fun < best_obj:
            best_obj = result.fun
            best_result = result.x

    # 4. Extract and format the best solution
    final_centers = np.zeros((n_circles, 2))
    final_radii = np.zeros(n_circles)
    
    for i in range(n_circles):
        final_centers[i, 0] = best_result[3 * i]
        final_centers[i, 1] = best_result[3 * i + 1]
        final_radii[i] = best_result[3 * i + 2]
        
    sum_radii = np.sum(final_radii)
    
    return final_centers, final_radii, sum_radii

# Helper to run and validate if needed
if __name__ == "__main__":
    centers, radii, s = run_packing()
