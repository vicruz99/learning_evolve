def compute_clearance(centers, radii):
    """
    Calculates the maximum possible radius for each circle based on current positions 
    and other circles' radii, and boundaries.
    """
    n = centers.shape[0]
    clearance = np.zeros(n)
    
    for i in range(n):
        # Distance to boundaries
        dist_boundary = min(
            centers[i, 0],
            1.0 - centers[i, 0],
            centers[i, 1],
            1.0 - centers[i, 1]
        )
        clearance[i] = dist_boundary

        # Distance to other circles
        for j in range(n):
            if i == j:
                continue
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            # The available space for circle i is distance minus radius of j
            # We want r_i + r_j <= dist => r_i <= dist - r_j
            available = dist - radii[j]
            if available < clearance[i]:
                clearance[i] = available
                
    return np.maximum(clearance, 0.0)
