import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Packs 26 circles in a unit square [0,1]x[0,1] to maximize the sum of radii.
    
    Returns:
        centers: np.array of shape (26, 2)
        radii: np.array of shape (26,)
        sum_radii: float
    """
    n_circles = 26

    # Create a hexagonal grid of points
    pts = []
    y_coord = 0

        x_coord = 0
        row_points = 0
        # In a hex grid, odd rows are shifted by 0.5 units horizontally
        while len(pts) < n_circles and row_points < n_circles:
            pts.append((x_coord, y_coord))
            x_coord += 1.0
            row_points += 1
        y_coord += np.sqrt(3) / 2
        
    pts = pts[:n_circles]
    pts = np.array(pts)
    
    # Scale points to fit inside the unit square with some margin
    # The scaling ensures the initial configuration is valid for the optimizer
    min_x, min_y = pts.min(axis=0)
    max_x, max_y = pts.max(axis=0)
    width = max_x - min_x
    height = max_y - min_y
    
    # Normalize to [0.1, 0.9] range initially to allow room for radius expansion
    scale = 0.8 / max(width, height)
    centers_init = (pts - min_x) * scale + 0.1

    radii_init = np.full(n_circles, 0.02)
    
    # Initial parameter vector
    x0 = np.concatenate([centers_init.ravel(), radii_init])
    
    # --- Phase 2: Optimization ---
    
    def objective(params):
        # We want to maximize sum of radii, so minimize negative sum
        r = params[2*n_circles:]
        return -np.sum(r)

    def boundary_constraints(params):
        constraints_list = []
        for i in range(n_circles):
            x = params[i*2]
            y = params[i*2 + 1]
            r = params[2*n_circles + i]
            # x - r >= 0  => x - r >= 0
            constraints_list.append(x - r)
            # 1 - r - x >= 0
            constraints_list.append(1.0 - r - x)
            # y - r >= 0
            constraints_list.append(y - r)
            # 1 - r - y >= 0
            constraints_list.append(1.0 - r - y)
        return np.array(constraints_list)

    def non_overlap_constraints(params):
        constraints_list = []
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                xi, yi = params[i*2], params[i*2 + 1]
                xj, yj = params[j*2], params[j*2 + 1]
                ri = params[2*n_circles + i]
                rj = params[2*n_circles + j]
                
                # dist^2 >= (ri + rj)^2
                dist_sq = (xi - xj)**2 + (yi - yj)**2
                min_dist = ri + rj
                
                # Constraint: dist_sq - min_dist^2 >= 0
                # We add a tiny epsilon to avoid numerical singularity at exact contact
                constraints_list.append(dist_sq - min_dist**2 + 1e-9)
        return np.array(constraints_list)

    # Define constraints for SLSQP (must be >= 0)
    bnd_constraints = {'type': 'ineq', 'fun': boundary_constraints}
    overlap_constraints = {'type': 'ineq', 'fun': non_overlap_constraints}
    
    # Bounds: x, y in [0, 1], r >= 0
    bounds = [(0, 1)] * (2 * n_circles) + [(0, 1)] * n_circles
    
    # Optimization
    result = minimize(
        objective,
        x0,
        method='SLSQP',
        bounds=bounds,
        constraints=[bnd_constraints, overlap_constraints],
        options={'ftol': 1e-12, 'maxiter': 1000, 'disp': False}
    )
    
    if result.success:
        final_params = result.x
    else:
        # Fallback to initial guess if optimization fails
        final_params = x0
        
    centers = final_params[:2*n_circles].reshape(n_circles, 2)
    radii = final_params[2*n_circles:]
    
    # Clip tiny negative radii due to numerical noise
    radii = np.maximum(radii, 0.0)
    
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
