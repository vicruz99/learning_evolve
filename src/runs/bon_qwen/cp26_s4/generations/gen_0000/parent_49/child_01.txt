import numpy as np
from scipy.optimize import minimize
import math

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # 1. Initial Guess: Hexagonal Lattice
    # We try to fit 26 circles in a hexagonal pattern.
    # A good approximation is to arrange them in rows.
    # Let's try 5 rows. 26/5 = 5.2 circles per row.
    # Rows could have counts: 5, 6, 5, 6, 4? Or 5, 5, 5, 5, 6.
    # Let's generate a grid and perturb it slightly to a hexagonal layout.
    
    centers = np.zeros((n, 2))
    
    # Heuristic placement
    # Try to fit in 5 rows
    # Rows at y = r, r + sqrt(3)r, ...
    # Since r is unknown, let's assume r ~ 0.1
    # y coords: 0.1, 0.273, 0.446, 0.620, 0.793
    # We can just place them roughly.
    
    # Let's use a simple grid first, then the optimizer will move them.
    # Actually, a slightly shifted grid is better.
    # 6 columns, 5 rows? 30 slots.
    # Let's pick 26 slots.
    
    xs = np.linspace(0.1, 0.9, 6) # 6 points
    ys = np.linspace(0.1, 0.9, 5) # 5 points
    
    count = 0
    idx = 0
    for j in range(5):
        for i in range(6):
            if idx < n:
                centers[idx] = [xs[i], ys[j]]
                # Shift odd rows for hexagonal feel
                if j % 2 == 1:
                    centers[idx][0] += 0.05 
                idx += 1
    
    # Initial radius guess
    # Distance between closest points in grid is 0.18 (0.9-0.1)/5
    # So r approx 0.09
    r_init = 0.08
    
    # Variables: x1, y1, ..., x26, y26, r
    # Total 53 variables
    x0 = np.zeros(2 * n + 1)
    for i in range(n):
        x0[2*i] = centers[i, 0]
        x0[2*i+1] = centers[i, 1]
    x0[-1] = r_init
    
    # Bounds
    # x, y in [0, 1]
    # r in [0, 0.5]
    bounds = []
    for i in range(n):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
    bounds.append((1e-6, 0.5))    # r
    
    # Constraints
    # 1. Boundary constraints:
    # x_i >= r  => x_i - r >= 0
    # x_i <= 1-r => 1 - x_i - r >= 0
    # same for y
    # 2. Pairwise distance constraints:
    # (x_i - x_j)^2 + (y_i - y_j)^2 >= (2r)^2
    
    # To reduce constraint count, we can define a function that computes
    # the minimum violation. But SLSQP requires explicit constraints or a penalty.
    # With 26 circles, 325 pairwise constraints is a lot.
    # However, SLSQP can handle it. Let's try.
    
    constraints = []
    
    # Helper to get center and radius from x vector
    def get_vars(x):
        cx = x[0::2]
        cy = x[1::2]
        r = x[-1]
        return cx, cy, r

    # Boundary constraints
    for i in range(n):
        # x_i - r >= 0
        def c_x_min(i=i):
            return lambda x: get_vars(x)[0][i] - get_vars(x)[2]
        constraints.append({'type': 'ineq', 'fun': c_x_min()})
        
        # 1 - x_i - r >= 0
        def c_x_max(i=i):
            return lambda x: 1.0 - get_vars(x)[0][i] - get_vars(x)[2]
        constraints.append({'type': 'ineq', 'fun': c_x_max()})
        
        # y_i - r >= 0
        def c_y_min(i=i):
            return lambda x: get_vars(x)[1][i] - get_vars(x)[2]
        constraints.append({'type': 'ineq', 'fun': c_y_min()})
        
        # 1 - y_i - r >= 0
        def c_y_max(i=i):
            return lambda x: 1.0 - get_vars(x)[1][i] - get_vars(x)[2]
        constraints.append({'type': 'ineq', 'fun': c_y_max()})

    # Pairwise constraints
    # We can add them in batches or all at once.
    # Let's add all.
    for i in range(n):
        for j in range(i + 1, n):
            def c_dist(i=i, j=j):
                def func(x):
                    cx, cy, r = get_vars(x)
                    dx = cx[i] - cx[j]
                    dy = cy[i] - cy[j]
                    dist_sq = dx*dx + dy*dy
                    return dist_sq - 4.0 * r * r
                return func
            constraints.append({'type': 'ineq', 'fun': c_dist()})

    # Objective: Maximize r => Minimize -r
    def objective(x):
        return -get_vars(x)[2]
    
    # Run optimization
    # method 'SLSQP'
    # options: maxiter
    try:
        res = minimize(objective, x0, method='SLSQP', bounds=bounds, 
                       constraints=constraints, options={'maxiter': 1000, 'ftol': 1e-9})
        
        if res.success:
            cx, cy, r = get_vars(res.x)
        else:
            # Fallback to initial or best found
            cx, cy, r = get_vars(x0) # Might be invalid, but we try to fix later
            # Try to find valid r for the centers if optimizer failed
            min_dist = np.inf
            for i in range(n):
                d_bound = min(cx[i], 1-cx[i], cy[i], 1-cy[i])
                min_dist = min(min_dist, d_bound)
                for j in range(i+1, n):
                    d = np.sqrt((cx[i]-cx[j])**2 + (cy[i]-cy[j])**2)
                    min_dist = min(min_dist, d/2.0)
            r = max(1e-6, min_dist)
            
    except Exception:
        cx, cy, r = get_vars(x0)

    # Final validation and cleanup
    # Ensure non-negative radius
    r = max(0.0, r)
    
    # Construct arrays
    centers_out = np.column_stack((cx, cy))
    radii_out = np.full(n, r)
    
    sum_radii = np.sum(radii_out)
    
    return centers_out, radii_out, float(sum_radii)
