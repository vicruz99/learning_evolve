import math
from scipy.optimize import minimize

def get_max_radius(centers):
    """
    Given centers, find max possible radii for each circle satisfying
    non-overlap and boundary constraints.
    """
    n = centers.shape[0]

    boundaries = np.array([min(c[0], 1-c[0], c[1], 1-c[1]) for c in centers])

        # Vectorized check for violations
