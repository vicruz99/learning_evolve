import numpy as np

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Solves the 26-circle packing problem in a unit square.
    Returns (centers, radii, sum_radii).
    """
    n = 26
    
    # 1. Initialization: Hexagonal Lattice Pattern
    # We target a radius of ~0.1014. 
    # A 5-row pattern (5-6-5-6-4) is dense and fits the boundary well.
    
    # Base parameters for initial hexagonal layout
    r_init = 0.101
    dx = 2 * r_init
    dy = np.sqrt(3) * r_init
    
    centers = []
    row_counts = [5, 6, 5, 6, 4]
    
    # Calculate vertical offset to center the cluster
    # Total height of 5 rows (centers) is 4 * dy
    # We want the whole cluster (including radius margins) to be in [0, 1]
    # But we will just place them and let the optimizer push them out.
    
    current_y = r_init
    
    for i, count in enumerate(row_counts):
        # Horizontal centering for this row
        # Row width is approx (count - 1) * 2r
        # We place them centered in [0, 1]
        x_start = 0.5 - ((count - 1) * r_init)
        
        # Offset even/odd rows for hexagonal packing (shift by radius)
        # Row indices 0, 2, 4 are "even", 1, 3 are "odd"
        if i % 2 == 1:
            x_start += r_init
            
        for j in range(count):
            x = x_start + j * (2 * r_init)
            centers.append([x, current_y])
            
        current_y += dy
        
    centers = np.array(centers)
    radii = np.full(n, r_init)
    
    # 2. Optimization: Relaxation to maximize sum of radii
    # We use a force-directed relaxation / simulated annealing approach.
    # We try to increase radii and resolve overlaps.
    
    # Precompute pairs for efficient overlap checking
    pairs = []
    for i in range(n):
        for j in range(i + 1, n):
            pairs.append((i, j))
            
    # Parameters
    temp = 0.05  # Initial "temperature" for jitter
    cooling = 0.99
    steps = 5000
    
    # Initial expansion
    radii *= 1.01 

    for step in range(steps):
        # Try to increase radii slightly
        radii *= 1.00005
        
        # Jitter centers to escape local minima
        if temp > 1e-5:
            jitter = np.random.normal(0, temp, size=centers.shape)
            centers += jitter
            # Clamp to boundaries with radius margin
            centers[:, 0] = np.clip(centers[:, 0], radii, 1 - radii)
            centers[:, 1] = np.clip(centers[:, 1], radii, 1 - radii)
        
        # Resolve overlaps (Force-directed repulsion)
        for _ in range(3): # Multiple passes per step for better resolution
            for i, j in pairs:
                c1, c2 = centers[i], centers[j]
                r1, r2 = radii[i], radii[j]
                
                dist_vec = c1 - c2
                dist = np.linalg.norm(dist_vec)
                min_dist = r1 + r2
                
                if dist < min_dist and dist > 1e-9:
                    # Repulsion: push centers apart
                    overlap = min_dist - dist
                    # Push factor (soft constraint)
                    push = overlap / 2.0 
                    direction = dist_vec / dist
                    centers[i] += direction * push
                    centers[j] -= direction * push
            
            # Boundary enforcement
            centers[:, 0] = np.clip(centers[:, 0], radii, 1 - radii)
            centers[:, 1] = np.clip(centers[:, 1], radii, 1 - radii)
            
            # If we hit a boundary, push back inward to maintain radius
            # (The clip handles the center, but if radius > distance to wall, 
            # the circle is cut. We assume centers stay valid.)
            
            # Cooling
            temp *= cooling

    # Final validation check (self-correction if any tiny overlaps remain)
    # We can scale down radii uniformly if validation fails, but with 
    # repulsion above, it should be valid.
    
    # Calculate sum of radii
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
