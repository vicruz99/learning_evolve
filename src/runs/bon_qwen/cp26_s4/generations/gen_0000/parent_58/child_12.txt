import numpy as np
from scipy.optimize import minimize

def compute_objective(v, N, W):
    """Compute the negative sum of radii plus penalty for constraint violations."""
    centers = v[:2*N].reshape(N, 2)
    radii = v[2*N:]
    
    # Pairwise distances
    diff = centers[:, None, :] - centers[None, :, :]
    dist = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dist, np.inf)
    
    # Overlap penalty: max(0, r_i + r_j - d_ij)^2
    overlap = radii[:, None] + radii[None, :] - dist
    penalty_overlap = np.sum(np.maximum(0, overlap)**2)
    
    # Boundary penalty: max(0, r_i - dist_to_boundary)^2
    penalty_bound = np.sum(np.maximum(0, radii - centers[:, 0])**2)
    penalty_bound += np.sum(np.maximum(0, radii - (1.0 - centers[:, 0]))**2)
    penalty_bound += np.sum(np.maximum(0, radii - centers[:, 1])**2)
    penalty_bound += np.sum(np.maximum(0, radii - (1.0 - centers[:, 1]))**2)
    
    return -np.sum(radii) + W * (penalty_overlap + penalty_bound)

def run_packing():

    Pack 26 circles in a unit square to maximize the sum of radii.
    Returns (centers, radii, sum_radii).

    N = 26
    best_sum = -np.inf
    best_v = None
    bounds = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N
    
    # Generate diverse initial configurations
    inits = []
    
    # 1. Hexagonal-like lattice
    hex_pts = []
    ry = 0.12
    while ry < 0.92:
        rx = 0.12 if len(hex_pts) % 2 == 1 else 0.09

            hex_pts.append([rx, ry])
            rx += 0.18
        ry += 0.16
    inits.append(np.array(hex_pts[:N]))
    
    # 2. Staggered grid
    grid_pts = []
    for i in range(6):
        for j in range(5):
            if len(grid_pts) >= N: break
            grid_pts.append([0.10 + j*0.18, 0.10 + i*0.17])
        if len(grid_pts) >= N: break
    inits.append(np.array(grid_pts))
    
    # 3. Random perturbed starts
    np.random.seed(42)
    for _ in range(4):
        rand_pts = np.random.rand(N, 2) * 0.7 + 0.15
        inits.append(rand_pts)
        
    # Continuation weights
    weights = [100.0, 1000.0, 5000.0]
    
    for ic, init_centers in enumerate(inits):
        # Initialize state vector
        v0 = np.zeros(3*N)
        v0[:2*N] = init_centers.flatten()
        v0[2*N:] = 0.06  # Initial radius guess
        
        # Add small deterministic perturbation to break symmetry
        v0 += np.arange(3*N) * 0.0005
        
        # Continuation optimization
        current_v = v0
        for w in weights:
            res = minimize(compute_objective, current_v, args=(N, w),
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 1500, 'ftol': 1e-10, 'gtol': 1e-8})
            current_v = res.x
            
        # Evaluate true objective without penalty to compare
        temp_centers = current_v[:2*N].reshape(N, 2)
        temp_radii = current_v[2*N:]
        current_sum = np.sum(temp_radii)
        
        if current_sum > best_sum:
            best_sum = current_sum
            best_v = current_v.copy()
            
    # Post-processing: strictly enforce constraints without heavy radius loss
    centers = best_v[:2*N].reshape(N, 2)
    radii = best_v[2*N:]
    
    # Iterative projection to fix boundary/overlap violations
    for _ in range(200):
        improved = False
        # Boundary clipping
        for i in range(N):
            x, y = centers[i]
            max_r = min(x, 1.0 - x, y, 1.0 - y)
            if radii[i] > max_r + 1e-12:
                radii[i] = max_r
                improved = True
                
        # Overlap resolution
        for i in range(N):

                dy = centers[i, 1] - centers[j, 1]
                d = np.sqrt(dx*dx + dy*dy)
                sum_r = radii[i] + radii[j]
                if sum_r > d + 1e-12:
                    # Shrink both proportionally
                    diff = (sum_r - d) / 2.0
                    radii[i] -= diff
                    radii[j] -= diff
                    radii[i] = max(0.0, radii[i])
                    radii[j] = max(0.0, radii[j])
                    improved = True
                    
        if not improved:
            break
            
    # Final clamp to guarantee validity
    radii = np.maximum(radii, 0.0)
    for i in range(N):
        x, y = centers[i]
        radii[i] = min(radii[i], x, 1.0-x, y, 1.0-y)
        
    final_sum = np.sum(radii)
    return centers, radii, final_sum
