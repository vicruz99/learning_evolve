            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False
    return True

def run_packing():
    np.random.seed(42)
    n = 26
    
    # Step 1: Heuristic Initialization
    # Create a 5x5 grid (25 circles)
    # Grid points at 0.1, 0.3, 0.5, 0.7, 0.9
    grid_points = []
    for i in range(5):
        for j in range(5):
            grid_points.append([0.1 + i*0.2, 0.1 + j*0.2])
    
    # Add the 26th circle in a gap to ensure dense packing
    # The center of the square (0.5, 0.5) is occupied by a grid point.
    # We pick a point in a gap, e.g., (0.25, 0.25) is somewhat central but safe distance from grid points.
    # Distance to (0.1, 0.1) is sqrt(0.15^2 + 0.15^2) approx 0.21.
    # Distance to (0.3, 0.1) is sqrt(0.05^2 + 0.15^2) approx 0.16.
    # With initial radius 0.05, these are safe.
    grid_points.append([0.25, 0.25])
    
    centers = np.array(grid_points)
    radii = np.full(n, 0.05) # Start with small valid radii
    
    # Step 2: Repulsion Simulation
    # Iteratively grow radii and resolve overlaps using repulsive forces
    dt = 0.05
    growth_rate = 0.0002
    max_iters = 2000
    
    for _ in range(max_iters):
        # Grow radii
        radii += growth_rate
        
        # Resolve overlaps with multiple sub-steps
        for _ in range(50):
            forces = np.zeros_like(centers)
            
            # Pairwise repulsion
            for i in range(n):
                for j in range(i + 1, n):
                    diff = centers[i] - centers[j]
                    dist = np.linalg.norm(diff)
                    min_dist = radii[i] + radii[j]
                    
                    if dist < min_dist and dist > 1e-9:
                        # Repulsive force proportional to overlap
                        overlap = min_dist - dist
                        f_mag = overlap * 5.0 
                        f_vec = (diff / dist) * f_mag
                        forces[i] += f_vec
                        forces[j] -= f_vec
                    elif dist <= 1e-9:
                        # Push apart randomly if centers coincide
                        forces[i] += np.random.randn(2) * 0.01
                        forces[j] -= np.random.randn(2) * 0.01
            
            # Boundary repulsion
            for i in range(n):
                r = radii[i]
                c = centers[i]
                # X boundaries
                if c[0] < r: 
                    forces[i, 0] += (r - c[0]) * 5.0
                elif c[0] > 1 - r: 
                    forces[i, 0] -= (c[0] - (1 - r)) * 5.0
                # Y boundaries
                if c[1] < r: 
                    forces[i, 1] += (r - c[1]) * 5.0
                elif c[1] > 1 - r: 
                    forces[i, 1] -= (c[1] - (1 - r)) * 5.0
            
            # Update positions
            centers += forces * dt
            
            # Clamp to [0, 1] to prevent numerical explosion, though forces should handle it
            np.clip(centers, 0, 1, out=centers)

    # Step 3: Optimization with scipy.optimize
    # Maximize sum of radii allowing them to be unequal
    # Variables: [x1, y1, r1, x2, y2, r2, ..., x26, y26, r26]
    x0 = np.zeros(3 * n)
    for i in range(n):
        x0[3*i] = centers[i, 0]
        x0[3*i+1] = centers[i, 1]
        x0[3*i+2] = radii[i]
    
    def objective(vars):
        # Maximize sum of radii -> Minimize negative sum
        return -np.sum(vars[2::3])
    
    def constraints(vars):
        con = []
        # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
        for i in range(n):
            x = vars[3*i]
            y = vars[3*i+1]
            r = vars[3*i+2]
            con.append(x - r)          # x >= r
            con.append(1 - r - x)      # x <= 1-r
            con.append(y - r)          # y >= r
            con.append(1 - r - y)      # y <= 1-r
        
        # Non-overlap constraints: dist^2 >= (r1 + r2)^2
        for i in range(n):
            for j in range(i + 1, n):
                x1, y1 = vars[3*i], vars[3*i+1]
                x2, y2 = vars[3*j], vars[3*j+1]
                r1, r2 = vars[3*i+2], vars[3*j+2]
                dist_sq = (x1 - x2)**2 + (y1 - y2)**2
                min_dist_sq = (r1 + r2)**2
                con.append(dist_sq - min_dist_sq)
        return np.array(con)
    
    bounds = [(0, 1) for _ in range(2 * n)] + [(0, 1) for _ in range(n)]
    
    # Try to optimize
    # SLSQP is suitable for this type of problem
    try:
        res = minimize(objective, x0, method='SLSQP', bounds=bounds, 
                       constraints={'type': 'ineq', 'fun': constraints},
                       options={'maxiter': 200, 'ftol': 1e-10})
        
        if res.success:
            centers_opt = res.x[:2*n].reshape(n, 2)
            radii_opt = res.x[2*n:]
            
            # Final strict validation before returning
            if check_validity(centers_opt, radii_opt):
                return centers_opt, radii_opt, np.sum(radii_opt)
    except Exception:
        pass

    # Fallback to simulation result if optimization fails or returns invalid
    # Ensure validity of simulation result by shrinking radii if necessary
    # (Though simulation should be valid with proper parameters)
    if not check_validity(centers, radii):
        # Simple shrink to ensure validity
        scale = 1.0
        while not check_validity(centers, radii * scale):
            scale *= 0.99
            if scale < 0.1: break # Safety break
        radii *= scale
    
    return centers, radii, np.sum(radii)
