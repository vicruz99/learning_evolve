import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    """
    n = 26
    
    # 1. Generate initial configuration (Hexagonal Lattice)
    # We place circles in a hexagonal grid pattern.
    # Rows and columns are determined to fit n=26.
    # A 6x5 grid (30 circles) is too many, so we take a subset or adjust.
    # Let's try a pattern that fits 26 well. 
    # 5 rows of 5 circles + 1 extra? Or 5, 5, 5, 5, 6?
    # Let's just generate a grid and pick the first 26 points, 
    # ensuring they are within bounds and spaced out.
    
    centers = []
    y = 0.1
    while len(centers) < n:
        x = 0.1
        row_len = 0
        while x <= 0.9 and len(centers) < n:
            centers.append([x, y])
            x += 0.15 # Initial spacing
            row_len += 1
        
        # Stagger next row
        y += 0.13
        offset = 0.075
        x = 0.1 + offset
        while x <= 0.9 and len(centers) < n:
            centers.append([x, y])
            x += 0.15
    
    # Trim to exactly 26
    centers = centers[:n]
    centers = np.array(centers)
    
    # Initial radii: small to ensure validity
    radii_init = np.full(n, 0.04)
    
    # 2. Define Objective Function
    # We want to maximize sum of radii, so minimize negative sum.
    # We add a small penalty for variance in radii to encourage equal sizes, 
    # which is generally better for packing density, but mainly to help convergence.
    def objective(vars):
        r = vars[2*n:]
        # Negative sum of radii
        obj = -np.sum(r)
        # Regularization: penalize deviation from mean (optional but helpful)
        # mean_r = np.mean(r)
        # obj += 0.01 * np.sum((r - mean_r)**2)
        return obj

    # 3. Define Constraints
    # Boundary constraints: 0 <= x - r, x + r <= 1, 0 <= y - r, y + r <= 1
    # Non-overlap constraints: dist^2 >= (r_i + r_j)^2
    
    def boundary_constraints(vars):
        x = vars[:n]
        y = vars[n:2*n]
        r = vars[2*n:]
        
        c_list = []
        for i in range(n):
            c_list.append(x[i] - r[i])      # x - r >= 0
            c_list.append(1.0 - (x[i] + r[i])) # x + r <= 1
            c_list.append(y[i] - r[i])      # y - r >= 0
            c_list.append(1.0 - (y[i] + r[i])) # y + r <= 1
        return c_list

    def non_overlap_constraints(vars):
        x = vars[:n]
        y = vars[n:2*n]
        r = vars[2*n:]
        
        c_list = []
        for i in range(n):
            for j in range(i + 1, n):
                dx = x[i] - x[j]
                dy = y[i] - y[j]
                dist_sq = dx*dx + dy*dy
                min_dist_sq = (r[i] + r[j])**2
                # Constraint: dist_sq - min_dist_sq >= 0
                c_list.append(dist_sq - min_dist_sq)
        return c_list

    # Combine constraints for scipy
    # SciPy minimize with SLSQP takes a list of constraint dictionaries
    constraints = []
    
    # Boundary
    b_cons = {'type': 'ineq', 'fun': boundary_constraints}
    constraints.append(b_cons)
    
    # Non-overlap
    # Note: Passing a function that returns an array is supported
    n_cons = {'type': 'ineq', 'fun': non_overlap_constraints}
    constraints.append(n_cons)

    # 4. Initial State
    x0 = np.concatenate([centers[:, 0], centers[:, 1], radii_init])
    
    # Bounds for variables
    # x, y in [0, 1]
    # r in [0, 1] (loose upper bound)
    bounds = []
    for _ in range(n):
        bounds.append((0, 1)) # x
        bounds.append((0, 1)) # y
    for _ in range(n):
        bounds.append((0, 0.5)) # r (max radius is 0.5)

    # 5. Run Optimization
    # Using SLSQP
    try:
        res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints, 
                       options={'maxiter': 1000, 'ftol': 1e-9})
        
        # Extract results
        x_opt = res.x[:n]
        y_opt = res.x[n:2*n]
        r_opt = res.x[2*n:]
        
        # Ensure non-negative radii (should be handled by bounds/constraints but safe guard)
        r_opt = np.maximum(r_opt, 1e-9)
        
        centers_opt = np.column_stack((x_opt, y_opt))
        
        sum_radii = np.sum(r_opt)
        
        # Validation check (internal)
        # if not validate_packing(centers_opt, r_opt):
        #     print("Warning: Optimization result failed validation")
        
        return centers_opt, r_opt, sum_radii

    except Exception as e:
        # Fallback to initial config if optimization fails
        print(f"Optimization failed: {e}")
        return centers, radii_init, np.sum(radii_init)
