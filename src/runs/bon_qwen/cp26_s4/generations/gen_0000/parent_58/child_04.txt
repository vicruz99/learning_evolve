import numpy as np
from scipy.optimize import differential_evolution

def compute_min_gap(centers):
    """
    Compute the maximum possible equal radius for a given configuration of centers.
    This is the minimum distance to any boundary or to any other circle.
    """
    x, y = centers[:, 0], centers[:, 1]
    
    # Distance to boundaries: min(x, 1-x, y, 1-y)
    dists_bound = np.minimum(np.minimum(x, 1.0 - x), np.minimum(y, 1.0 - y))
    min_bound = np.min(dists_bound)
    
    # Pairwise distances: min(||c_i - c_j||) / 2
    dx = x[:, None] - x[None, :]
    dy = y[:, None] - y[None, :]
    dists_pair = np.sqrt(dx**2 + dy**2)
    np.fill_diagonal(dists_pair, np.inf)
    min_pair = np.min(dists_pair) / 2.0
    
    return min(min_bound, min_pair)

def objective(centers_flat):
    """
    Objective function for differential evolution.
    We minimize the negative of the minimum gap to maximize it.
    """
    centers = centers_flat.reshape(26, 2)
    return -compute_min_gap(centers)

def run_packing():
    n_circles = 26
    # Each circle has 2 coordinates (x, y)
    bounds = [(0.0, 1.0) for _ in range(n_circles * 2)]
    
    # Differential Evolution is robust for non-convex packing problems.
    # We use a large population and iterations to navigate the rugged landscape.
    result = differential_evolution(
        objective, 
        bounds, 
        popsize=50, 
        maxiter=3000, 
        seed=42, 
        tol=1e-9, 
        polish=False,  # Avoids local optimizer issues with non-smooth min function
        workers=1
    )
    
    centers = result.x.reshape(n_circles, 2)
    
    # Re-compute radius explicitly to ensure strict validity within numerical tolerance
    r = compute_min_gap(centers)
    radii = np.full(n_circles, r)
    
    return centers, radii, np.sum(radii)
