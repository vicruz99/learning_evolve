        cons.append({'type': 'ineq', 'fun': lambda p, i=i: boundary_x_lower(p, n, i)})
        cons.append({'type': 'ineq', 'fun': lambda p, i=i: boundary_x_upper(p, n, i)})
        cons.append({'type': 'ineq', 'fun': lambda p, i=i: boundary_y_lower(p, n, i)})
        cons.append({'type': 'ineq', 'fun': lambda p, i=i: boundary_y_upper(p, n, i)})
        cons.append({'type': 'ineq', 'fun': lambda p, i=i: radius_nonneg(p, n, i)})
    
    for i in range(n):
        for j in range(i + 1, n):
            cons.append({'type': 'ineq', 'fun': lambda p, i=i, j=j: non_overlap_constraint(p, n, i, j)})
    
    bounds = [(0, 1)] * (2*n) + [(0, 0.5)] * n
    
    try:
        result = minimize(
            objective, x0, args=(n,),
            method='SLSQP', bounds=bounds, constraints=cons,
            options={'maxiter': 3000, 'ftol': 1e-14}
        )
        c, r = extract_solution(result.x, n)
        return c, r, -result.fun
    except Exception:
        return centers, radii, np.sum(radii)


def optimize_penalty(centers, radii, n, pw):
    """Optimize using penalty method with L-BFGS-B."""
    x0 = build_params(centers, radii, n)
    bounds = [(0, 1)] * (2*n) + [(0, 0.5)] * n
    
    try:
        result = minimize(
            penalty_objective, x0, args=(n, pw),
            method='L-BFGS-B', bounds=bounds,
            options={'maxiter': 5000, 'ftol': 1e-14}
        )
        c, r = extract_solution(result.x, n)
        return c, r, np.sum(r)
    except Exception:
        return centers, radii, np.sum(radii)


def perturb_optimize(centers, radii, n, num_trials, scale):
    """Perturb and optimize multiple times."""
    best_sum = np.sum(radii)
    best_c = centers.copy()
    best_r = radii.copy()
    
    for _ in range(num_trials):
        pc = centers + np.random.uniform(-scale, scale, (n, 2))
        pr = radii * np.random.uniform(0.85, 1.15, n)
        pc = np.clip(pc, 0.01, 0.99)
        pr = np.maximum(pr, 0.005)
        
        oc, or_, os = optimize_penalty(pc, pr, n, 1e6)
        if os > best_sum:
            best_sum = os
            best_c = oc
            best_r = or_
    
    return best_c, best_r, best_sum


def run_packing():
    """Pack 26 circles in unit square to maximize sum of radii."""
    n = 26
    
    np.random.seed(42)
    
    best_sum = -1
    best_centers = None
    best_radii = None
    
    # Try multiple initial configurations
    for init_func in [generate_grid_initial, generate_hexagonal_initial]:
        c, r = init_func(n)
        
        # SLSQP optimization
        oc, or_, os = optimize_slsqp(c, r, n)
        if os > best_sum:
            best_sum = os
            best_centers = oc
            best_radii = or_
        
        # Penalty optimization
        oc2, or2, os2 = optimize_penalty(c, r, n, 1e6)
        if os2 > best_sum:
            best_sum = os2
            best_centers = oc2
            best_radii = or2
    
    # Perturbation search from best so far
    if best_centers is not None:
        for seed in [1, 2, 3, 5, 7]:
            np.random.seed(seed)
            pc, pr, ps = perturb_optimize(best_centers, best_radii, n, num_trials=5, scale=0.008)
            if ps > best_sum:
                best_sum = ps
                best_centers = pc
                best_radii = pr
    
    # Final SLSQP refinement
    if best_centers is not None:
        oc, or_, os = optimize_slsqp(best_centers, best_radii, n)
        if os > best_sum:
            best_sum = os
            best_centers = oc
            best_radii = or_
    
    # Fallback
    if best_centers is None:
        best_centers, best_radii = generate_grid_initial(n)
        best_sum = np.sum(best_radii)
    
    best_radii = np.maximum(best_radii, 0)
    best_centers = np.clip(best_centers, 0, 1)
    
    return best_centers, best_radii, best_sum
