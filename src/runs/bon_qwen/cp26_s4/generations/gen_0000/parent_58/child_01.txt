

import numpy as np

def init_centers(n):
    """Initialize centers in a hexagonal pattern."""
    centers = np.zeros((n, 2))
    rows, cols = 5, 5
    idx = 0
    dx, dy = 0.2, 0.1732
    cx, cy = 0.5, 0.5
    for r in range(rows):
        for c in range(cols):
            if idx >= n:
                break
            x = cx + (c - (cols - 1) / 2) * dx
            y = cy + (r - (rows - 1) / 2) * dy
            if r % 2 == 1:
                x += dx / 2
            centers[idx] = [x, y]
            idx += 1
    for i in range(idx, n):
        centers[i] = [0.5, 0.5]
    return centers

def compute_repulsion(centers, radii, repulsion_k):
    """Compute repulsive forces between overlapping circles."""
    n = len(radii)
    diff = centers[:, None, :] - centers[None, :, :]
    dist = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dist, np.inf)
    
    min_dist = radii[:, None] + radii[None, :]
    overlap = np.maximum(min_dist - dist, 0.0)
    
    dir_vec = diff / (dist[:, :, None] + 1e-7)
    force = np.sum(overlap[:, :, None] * dir_vec, axis=1) * repulsion_k
    return force, np.max(overlap)

def apply_boundary_forces(centers, radii, force, repulsion_k):
    """Apply repulsive forces from the unit square boundaries."""
    n = len(radii)
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < 0:
            force[i, 0] += repulsion_k * (r - x)
        elif x + r > 1:
            force[i, 0] -= repulsion_k * (x + r - 1)
        if y - r < 0:
            force[i, 1] += repulsion_k * (r - y)
        elif y + r > 1:
            force[i, 1] -= repulsion_k * (y + r - 1)
    return force

def run_packing():
    """Run the circle packing optimization and return results."""
    np.random.seed(42)
    n = 26
    centers = init_centers(n)
    radii = np.full(n, 0.05)
    velocities = np.zeros((n, 2))
    
    dt = 2e-4
    grow_rate = 1e-5
    repulsion_k = 100.0
    damping = 0.9
    max_iters = 80000
    
    # Main optimization loop: expand circles while resolving overlaps via repulsion
    for it in range(max_iters):
        radii += grow_rate
        radii = np.maximum(radii, 1e-6)
        
        force, max_ov = compute_repulsion(centers, radii, repulsion_k)
        force = apply_boundary_forces(centers, radii, force, repulsion_k)
        
        velocities = velocities * damping + force * dt
        centers += velocities * dt
        
        # Adaptively adjust growth rate based on overlap severity
        if max_ov < 1e-5:
            grow_rate = min(grow_rate * 1.03, 1e-3)
        else:
            grow_rate = max(grow_rate * 0.95, 1e-7)
            
        # Occasional random velocity kick to escape local minima
        if it % 15000 == 0 and it > 0:
            velocities += np.random.randn(n, 2) * 1e-4
            
    # Final stabilization pass to eliminate residual overlaps
    radii = np.maximum(radii, 0)
    for _ in range(5000):
        force, _ = compute_repulsion(centers, radii, repulsion_k * 3.0)
        force = apply_boundary_forces(centers, radii, force, repulsion_k * 3.0)
        centers += force * 1e-3
        
    # Slight margin shrink to guarantee validity under numerical tolerance
    radii = np.maximum(radii - 1e-6, 0)
    
    return centers, radii, float(np.sum(radii))
