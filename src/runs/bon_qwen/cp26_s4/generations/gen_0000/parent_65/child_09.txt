import numpy as np

def check_validity(centers, radii, tol=1e-9):
    n = centers.shape[0]
    # Check boundaries
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x < r - tol or x > 1 - r + tol or y < r - tol or y > 1 - r + tol:
            return False, "boundary"
    # Check overlaps
    for i in range(n):
        for j in range(i+1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = math.sqrt(dx*dx + dy*dy)
            if dist < radii[i] + radii[j] - tol:
                return False, f"overlap {i}-{j}"
    return True, "valid"
