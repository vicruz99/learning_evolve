import numpy as np

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    Uses a hexagonal initialization and iterative expansion with repulsion forces.
    """
    n = 26
    
    # 1. Initialization: Hexagonal Grid Perturbation
    # This is much more effective than random placement for packing problems.
    centers = np.empty((n, 2))
    
    # Create a hexagonal pattern (triangular lattice)
    # We generate slightly more points than needed and select the best 26
    # to fill the square uniformly.
    points = []
    s = 0.15  # Approximate spacing for initial grid
    
    # Generate points on a triangular grid
    for row in range(10):
        for col in range(10):
            x = col * s + (row % 2) * s / 2
            y = row * s * np.sqrt(3) / 2
            # Keep points roughly inside the square (with margin for radius)
            if 0.05 <= x <= 0.95 and 0.05 <= y <= 0.95:
                points.append([x, y])
    
    # Select 26 points. If we have more, pick them to spread out.
    # A simple heuristic is to just take the first 26 from the grid generation,
    # but to ensure better coverage, we can just take the generated list if it has >= 26
    if len(points) >= n:
        # We can use a simple greedy selection or just take the first n.
        # Taking the first n from a grid scan usually covers the space well.
        centers = np.array(points[:n])
    else:
        # Fallback to random if grid didn't yield enough (unlikely with these params)
        np.random.seed(42)
        centers = np.random.rand(n, 2) * 0.8 + 0.1

    # 2. Simulation Parameters
    r = 0.02  # Initial radius
    steps = 8000  # Number of optimization steps
    dt = 5e-4    # Time step for integration
    growth_rate = 2e-5  # Radius increase per step
    
    # Forces constants
    k_repulse = 500.0   # Strength of circle-circle repulsion
    k_boundary = 500.0  # Strength of boundary repulsion
    damping = 0.85      # Velocity damping to prevent oscillation
    
    velocities = np.zeros((n, 2))

    # 3. Optimization Loop
    for _ in range(steps):
        # A. Increase Radius
        r += growth_rate
        
        # B. Compute Pairwise Forces (Circle-Circle)
        # Vectorized distance calculation
        # diff[i, j] = center[i] - center[j]
        diff = centers[:, None, :] - centers[None, :, :] # Shape (n, n, 2)
        dists = np.linalg.norm(diff, axis=2) # Shape (n, n)
        
        # Set diagonal to large value to ignore self-interaction
        np.fill_diagonal(dists, 2 * r + 1.0)
        
        # Overlap amount: positive if circles overlap
        # We use a soft constraint: force proportional to overlap amount
        overlap = np.maximum(0, 2 * r - dists)
        
        # Direction of repulsion (from j to i)
        # Avoid division by zero
        safe_dists = np.where(dists > 0, dists, 1.0)
        directions = diff / safe_dists[:, :, None]
        
        # Force magnitude
        forces = k_repulse * overlap[:, :, None] * directions
        
        # Sum forces acting on each circle
        pairwise_forces = np.sum(forces, axis=1) # Shape (n, 2)
        
        # C. Compute Boundary Forces
        boundary_forces = np.zeros_like(centers)
        
        # Left (x=0) and Right (x=1) boundaries
        # If center x < r, push right. If center x > 1-r, push left.
        overlap_left = np.maximum(0, r - centers[:, 0])
        overlap_right = np.maximum(0, centers[:, 0] - (1 - r))
        
        boundary_forces[:, 0] += k_boundary * overlap_left  # Push right (+x)
        boundary_forces[:, 0] -= k_boundary * overlap_right # Push left (-x)
        
        # Bottom (y=0) and Top (y=1) boundaries
        overlap_bottom = np.maximum(0, r - centers[:, 1])
        overlap_top = np.maximum(0, centers[:, 1] - (1 - r))
        
        boundary_forces[:, 1] += k_boundary * overlap_bottom # Push up (+y)
        boundary_forces[:, 1] -= k_boundary * overlap_top    # Push down (-y)
        
        # D. Update Velocities and Positions
        total_forces = pairwise_forces + boundary_forces
        
        # Simple Euler integration with damping
        velocities = damping * velocities + dt * total_forces
        centers += dt * velocities
        
        # E. Hard Constraint Correction (Clipping)
        # Ensure circles don't get pushed outside the square due to numerical errors
        centers[:, 0] = np.clip(centers[:, 0], r, 1 - r)
        centers[:, 1] = np.clip(centers[:, 1], r, 1 - r)
        
    # 4. Final Validation and Return
    # Create radii array
    radii = np.full(n, r)
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
