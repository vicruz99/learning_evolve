

<strategy>
The goal is to maximize the sum of radii of 26 circles in a unit square. This is a non-linear, non-convex optimization problem.
A robust approach to solve this involves two stages:
1.  **Geometric Initialization**: Start with a structured arrangement (like a hexagonal grid) that is known to be efficient for packing. This provides a good baseline sum of radii and prevents the optimizer from getting stuck in poor local minima (like clusters).
2.  **Constrained Optimization**: Treat the circle centers as variables. For any fixed configuration of centers, the maximum non-overlapping radii are determined by the nearest neighbors and boundaries. This sub-problem can be solved exactly using Linear Programming (LP). By wrapping this LP solver inside an objective function, we can use a global optimization algorithm (Differential Evolution) to search for the optimal centers. This method is superior to gradient-based methods because the "max radius" function is non-smooth.

The strategy focuses on:
-   Using a hexagonal grid perturbation to initialize 26 points.
-   Formulating the radius determination as an LP for precision.
-   Using `scipy.optimize.differential_evolution` for robust global search over the 52-dimensional space of circle centers.

import numpy as np
from scipy.optimize import minimize, differential_evolution
from scipy.optimize import linprog
import math

# Problem constants
N_CIRCLES = 26

def solve_max_radii_for_centers(centers):
    """
    Given fixed centers, solve the LP to find the maximum possible radii
    such that circles do not overlap and stay inside the unit square.
    
    Returns:
        sum_radii: float
        radii: np.array of shape (N_CIRCLES,)
    """
    n = centers.shape[0]
    
    # Objective: minimize -sum(r) -> c = -1 for each variable
    c = np.ones(n) * -1.0
    
    # Constraints matrix A_ub and vector b_ub such that A_ub @ r <= b_ub
    # 1. Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    # 2. Pairwise constraints: r_i + r_j <= dist(centers[i], centers[j])
    
    num_bound_constraints = 4 * n
    num_pair_constraints = n * (n - 1) // 2
    num_constraints = num_bound_constraints + num_pair_constraints
    
    A_ub = np.zeros((num_constraints, n))
    b_ub = np.zeros(num_constraints)
    
    idx = 0
    
    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        # r_i <= x
        A_ub[idx, i] = 1.0
        b_ub[idx] = x
        idx += 1
        # r_i <= 1 - x
        A_ub[idx, i] = 1.0
        b_ub[idx] = 1.0 - x
        idx += 1
        # r_i <= y
        A_ub[idx, i] = 1.0
        b_ub[idx] = y
        idx += 1
        # r_i <= 1 - y
        A_ub[idx, i] = 1.0
        b_ub[idx] = 1.0 - y
        idx += 1
        
    # Pairwise constraints
    # Precompute distances to speed up? Not strictly necessary for N=26
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = math.hypot(dx, dy)
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dist
            idx += 1
            
    # Bounds for radii: r_i >= 0
    bounds = [(0, None) for _ in range(n)]
    
    # Solve LP
    try:
        # method='highs' is efficient
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        
        if res.success:
            # linprog minimizes c^T x, so min(-sum r) = -max(sum r)
            # res.fun is the minimum value, which is negative of max sum
            max_sum = -res.fun
            optimal_radii = res.x
            return max_sum, optimal_radii
        else:
            # Fallback if LP fails (should not happen for valid geometry)
            return 0.0, np.zeros(n)
    except Exception:
        return 0.0, np.zeros(n)

def objective_function(centers_flat):
    """
    Objective function for the optimizer.
    We want to maximize sum of radii, so we minimize negative sum.
    """
    centers = centers_flat.reshape(N_CIRCLES, 2)
    sum_radii, _ = solve_max_radii_for_centers(centers)
    return -sum_radii

def generate_hexagonal_init():
    """
    Generate an initial configuration based on a hexagonal lattice.
    """
    points = []
    # Hexagonal lattice parameters
    # We want to fit 26 points. 
    # Approximate density suggests a grid.
    # Let's try to generate points in a lattice and pick best or just arrange them.
    
    # Simple approach: Fill a grid with some offset
    # 6 rows, varying columns
    # Hex spacing: vertical = sqrt(3)/2 * spacing, horizontal = spacing
    # To fit in 1x1, let's estimate spacing.
    # 26 points. Area ~ 1. 26 * pi * r^2 ~ 0.9. r ~ 0.1. Diameter 0.2.
    # Spacing ~ 0.2.
    
    # Let's try a perturbed grid to ensure distinctness and good coverage
    # 5x5 grid has 25 points. We need 26.
    # Let's place 26 points randomly but somewhat distributed, 
    # or use a lattice subset.
    
    # Lattice generation
    pts = []
    # Try to fill 6 rows
    # Row 0: 5 pts
    # Row 1: 5 pts (shifted)
    # Row 2: 5 pts
    # Row 3: 5 pts (shifted)
    # Row 4: 4 pts
    # Row 5: 2 pts
    # Total 26? 5+5+5+5+4+2 = 26.
    
    # Coordinates
    # x spacing dx, y spacing dy = dx * sqrt(3)/2
    # Let dx = 0.2, dy = 0.1732
    # Width needed for 5 pts: 4*dx = 0.8. Margin 0.1 each side. OK.
    # Height needed for 6 rows: 5*dy = 0.866. Margin ~0.067. OK.
    
    dx = 0.2
    dy = dx * math.sqrt(3) / 2.0
    
    # Center the grid
    # X centers: 0.1, 0.3, 0.5, 0.7, 0.9
    x_centers = np.linspace(0.1, 0.9, 5)
    
    # Y centers: 6 rows
    # 0, dy, 2dy, ...
    # Let's align y to center in [0, 1]
    # Range of y: 0 to 5*dy = 0.866.
    # Shift to center: (1 - 0.866)/2 = 0.067
    y_start = 0.067
    y_rows = [y_start + k * dy for k in range(6)]
    
    # Construct points
    # Row 0 (5 pts)
    for x in x_centers:
        pts.append([x, y_rows[0]])
    # Row 1 (5 pts, shifted by dx/2? No, hex shift is dx/2)
    # Actually standard hex shift is dx/2.
    # But if we shift, x might go out of bounds?
    # If we shift x_centers by dx/2 = 0.1, range becomes 0.2 to 1.0.
    # 1.0 is boundary. Center at 1.0 with radius > 0 is bad.
    # Let's keep x aligned for stability, or shift carefully.
    # Let's just use a perturbed 5x5 grid + 1 point for robustness.
    
    # Revert to 5x5 grid + 1 point
    pts = []
    grid_x = np.linspace(0.15, 0.85, 5) # Slightly inset to allow movement
    grid_y = np.linspace(0.15, 0.85, 5)
    for y in grid_y:
        for x in grid_x:
            pts.append([x, y])
    # 25 points. Add one in center
    pts.append([0.5, 0.5])
    
    return np.array(pts)

def run_packing():
    """
    Main function to run the packing optimization.
    """
    # 1. Initialization
    np.random.seed(123) # Reproducibility
    init_centers = generate_hexagonal_init()
    
    # Add small random perturbation to avoid symmetry degeneracy
    init_centers += np.random.normal(0, 0.01, init_centers.shape)
    
    # Clip to [0, 1]
    init_centers = np.clip(init_centers, 0.0, 1.0)
    
    x0 = init_centers.flatten()
    
    # 2. Optimization
    # Bounds for centers: [0, 1] for each x, y
    bounds = [(0.0, 1.0)] * (2 * N_CIRCLES)
    
    # Use Differential Evolution for global optimization
    # It handles the non-smooth nature of the problem better than local methods
    # population size: 'I' * nparams. Default 15.
    # maxiter: limit iterations.
    # tol: convergence tolerance.
    
    # Note: DE can be slow for 52 dimensions. 
    # We use a smaller population and iterations to keep runtime reasonable,
    # but it should find a good local optimum from the grid start.
    # Alternatively, use a local optimizer on the grid start.
    
    # Let's try Nelder-Mead first as it's fast and often good for this.
    # But Nelder-Mead is local.
    # Let's use a multi-start strategy or BasinHopping?
    # Actually, let's just run DE with limited steps.
    
    try:
        result = differential_evolution(
            objective_function,
            bounds,
            seed=42,
            maxiter=100,
            popsize=15,
            tol=1e-7,
            mutation=(0.5, 1),
            recombination=0.7,
            workers=1
        )
        
        best_centers = result.x.reshape(N_CIRCLES, 2)
        best_val = -result.fun
        
        # Refine with local optimization (Powell)
        # Start from DE result
        local_res = minimize(
            objective_function,
            best_centers.flatten(),
            method='Powell',
            options={'maxiter': 200, 'ftol': 1e-8, 'xtol': 1e-8}
        )
        
        if local_res.fun < result.fun:
            best_centers = local_res.x.reshape(N_CIRCLES, 2)
            best_val = -local_res.fun
            
    except Exception as e:
        print(f"Optimization error: {e}")
        # Fallback to initial
        best_centers = init_centers
        best_val, _ = solve_max_radii_for_centers(best_centers)

    # 3. Final Calculation
    sum_radii, radii = solve_max_radii_for_centers(best_centers)
    
    # Ensure radii are valid (non-negative)
    radii = np.maximum(radii, 0.0)
    
    return best_centers, radii, float(sum_radii)
