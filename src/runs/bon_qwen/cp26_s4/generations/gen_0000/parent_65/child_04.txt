import numpy as np
import math
from scipy.optimize import differential_evolution


def compute_fitness(params, n=26):
    """Compute negative sum of radii plus constraint violation penalties."""
    BOUNDARY_PENALTY = 1e6
    OVERLAP_PENALTY = 1e8
    
    total_radius = 0.0
    penalty = 0.0
    
    for i in range(n):
        cx = params[3 * i]
        cy = params[3 * i + 1]
        r = params[3 * i + 2]
        
        total_radius += r
        
        # Boundary penalties
        if cx - r < 0:
            penalty += (r - cx) ** 2 * BOUNDARY_PENALTY
        if cx + r > 1:
            penalty += (cx + r - 1) ** 2 * BOUNDARY_PENALTY
        if cy - r < 0:
            penalty += (r - cy) ** 2 * BOUNDARY_PENALTY
        if cy + r > 1:
            penalty += (cy + r - 1) ** 2 * BOUNDARY_PENALTY
    
    # Overlap penalties
    for i in range(n):
        for j in range(i + 1, n):
            dx = params[3 * i] - params[3 * j]
            dy = params[3 * i + 1] - params[3 * j + 1]
            dist = math.sqrt(dx * dx + dy * dy)
            min_dist = params[3 * i + 2] + params[3 * j + 2]
            if dist < min_dist:
                overlap = min_dist - dist
                penalty += overlap ** 2 * OVERLAP_PENALTY
    
    return -total_radius + penalty


def create_hexagonal_start(n, seed=0):
    """Create hexagonal lattice starting configuration."""
    rng = np.random.RandomState(seed)
    
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.1)
    
    # Determine grid dimensions
    approx_side = math.sqrt(n)
    n_cols = int(approx_side) + 1
    spacing_x = 0.18
    spacing_y = spacing_x * math.sqrt(3) / 2
    
    offset_x = 0.15 + rng.uniform(-0.02, 0.02)
    offset_y = 0.15 + rng.uniform(-0.02, 0.02)
    
    idx = 0
    for row in range(n_cols + 3):
        for col in range(n_cols + 3):
            if idx >= n:
                break
            x = offset_x + col * spacing_x
            y = offset_y + row * spacing_y
            if row % 2 == 1:
                x += spacing_x / 2
            if x + 0.1 < 1.0 and y + 0.1 < 1.0 and x - 0.1 > 0 and y - 0.1 > 0:
                centers[idx] = [x, y]
                idx += 1
        if idx >= n:
            break
    
    return centers, radii


def create_spiral_start(n, seed=0):
    """Create spiral starting configuration."""
    rng = np.random.RandomState(seed)
    
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.09)
    
    cx, cy = 0.5 + rng.uniform(-0.1, 0.1), 0.5 + rng.uniform(-0.1, 0.1)
    
    for i in range(n):
        angle = i * 2.39996  # Golden angle approximation
        radius = 0.12 * math.sqrt(i + 1)
        x = cx + radius * math.cos(angle)
        y = cy + radius * math.sin(angle)
        
        # Clamp to valid region
        x = max(0.1, min(0.9, x))
        y = max(0.1, min(0.9, y))
        
        centers[i] = [x, y]
    
    return centers, radii


def create_grid_start(n, seed=0):
    """Create grid-based starting configuration."""
    rng = np.random.RandomState(seed)
    
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.1)
    
    grid_size = int(math.ceil(math.sqrt(n)))
    spacing = 0.8 / (grid_size - 1) if grid_size > 1 else 0.5
    
    offset_x = rng.uniform(0.05, 0.15)
    offset_y = rng.uniform(0.05, 0.15)
    
    idx = 0
    for row in range(grid_size + 1):
        for col in range(grid_size + 1):
            if idx >= n:
                break
            x = offset_x + col * spacing
            y = offset_y + row * spacing
            if 0.1 <= x <= 0.9 and 0.1 <= y <= 0.9:
                centers[idx] = [x, y]
                idx += 1
        if idx >= n:
            break
    
    return centers, radii


def run_packing():
    """Pack 26 circles in a unit square to maximize sum of radii."""
    n = 26
    
    # Define optimization bounds

    bounds = [(0.0, 1.0)] * (2 * n) + [(0.01, 0.2)] * n
    
    best_result = None
    best_score = float('inf')
    
    # Try multiple random seeds for diversity
    seeds = [42, 123, 456, 789, 101, 202, 303]
    
    for seed_idx, seed in enumerate(seeds):
        # Create diverse initial population

                compute_fitness,
