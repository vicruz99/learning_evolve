import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Initial hexagonal grid packing
    row = 0
    col = 0
    idx = 0
    r = 0.05
    while idx < n:
        x = r + col * 2 * r + (0.5 * r if row % 2 == 1 else 0)
        y = r + row * np.sqrt(3) * r
        
        if x <= 1 - r:
            centers[idx] = [x, y]
            radii[idx] = r
            idx += 1
            col += 1
        else:
            row += 1
            col = 0

    # Random perturbation for diversity
    centers += np.random.uniform(-0.01, 0.01, size=centers.shape)
    radii += np.random.uniform(-0.005, 0.005, size=radii.shape)
    radii = np.maximum(radii, 1e-5)

    # Define constraints
    def boundary_constraints(params):
        cons = []
        for i in range(n):
            xi, yi, ri = params[3*i], params[3*i+1], params[3*i+2]
            cons.append(xi - ri)
            cons.append(1 - xi - ri)
            cons.append(yi - ri)
            cons.append(1 - yi - ri)
        return np.array(cons)

    def overlap_constraints(params):
        cons = []
        for i in range(n):
            for j in range(i + 1, n):
                xi, yi, ri = params[3*i], params[3*i+1], params[3*i+2]
                xj, yj, rj = params[3*j], params[3*j+1], params[3*j+2]
                dist = np.sqrt((xi - xj)**2 + (yi - yj)**2)
                cons.append(dist - (ri + rj))
        return np.array(cons)

    def objective(params):
        return -np.sum(params[2::3])

    initial_params = np.hstack([centers, radii])
    bounds = [(0, 1)] * (2 * n) + [(0, None)] * n
    
    cons = [
        {'type': 'ineq', 'fun': boundary_constraints},
        {'type': 'ineq', 'fun': overlap_constraints}
    ]

    result = minimize(objective, initial_params, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 5000, 'ftol': 1e-8})

    if result.success:
        opt_params = result.x
        opt_centers = np.column_stack((opt_params[:n], opt_params[n:2*n]))
        opt_radii = opt_params[2*n:]
    else:
        opt_centers, opt_radii = centers, radii

    # Final validation and adjustment
    opt_centers = np.clip(opt_centers, opt_radii[:, None], 1 - opt_radii[:, None])
    
    return opt_centers, opt_radii, np.sum(opt_radii)
