from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    Pack 26 circles in a unit square to maximize sum of radii.
    """
    n = 26
    
    # 1. Initialization: Hexagonal grid pattern
    # We want to pack 26 circles. 
    # A 5x5 grid fits 25 circles with radius 0.1.
    # We can add one more or use a denser pattern.
    # Let's generate a grid of points and pick the first 26.

    # Actually, just generating a structured set of 26 points is better.

    # but arranged in a hexagonal pattern.
    
    cols = 5
    rows = 6 # 5*6 = 30 > 26
    
    # Grid parameters
    dx = 0.205 # slightly larger than 0.2 to start loose
    dy = dx * np.sqrt(3) / 2
    
    # Start offset to center the pattern roughly
    start_x = 0.1
    start_y = 0.1
    
    points = []
    for r_idx in range(rows):
        # Offset every other row for hexagonal packing
        x_offset = (dx / 2) if r_idx % 2 == 1 else 0
        for c_idx in range(cols):
            x = start_x + c_idx * dx + x_offset
            y = start_y + r_idx * dy
            # Keep within [0,1] roughly
            if 0 <= x <= 1 and 0 <= y <= 1:
                points.append([x, y])
    
    # Take first 26 points
    points = points[:n]
    
    # Convert to numpy array
    centers_init = np.array(points)
    
    # Initial radii: small enough to not overlap
    r_init = 0.04
    radii_init = np.full(n, r_init)

    x0 = centers_init[:, 0]
    y0 = centers_init[:, 1]
    r0 = radii_init

    # We will structure params as [x0, y0, r0, x1, y1, r1 ...] for easy unpacking
    # Or separate arrays? Scipy takes 1D array.
    # Let's use order: all x, all y, all r?

    # Interleaved is easier for index arithmetic.

        bounds.append((0.0, 0.5)) # r

    def objective(params):
        # Unpack
        xs = params[0::3]
        ys = params[1::3]
        rs = params[2::3]
        
        # 1. Objective: Minimize -sum(rs)
        score = -np.sum(rs)
        
        # 2. Boundary Penalties
        # Constraints: r <= x, r <= 1-x, r <= y, r <= 1-y

        return score
