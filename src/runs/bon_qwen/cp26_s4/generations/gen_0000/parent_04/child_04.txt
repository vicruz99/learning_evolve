import numpy as np
from scipy.optimize import minimize
import math

class CircleBoundaryConstraint:
    def __init__(self, n, i, ctype):
        self.n = n
        self.i = i
        self.ctype = ctype
    
    def __call__(self, x):
        xi = x[2 * self.i]
        yi = x[2 * self.i + 1]
        ri = x[2 * self.n + self.i]
        if self.ctype == 0:
            return xi - ri
        elif self.ctype == 1:
            return 1.0 - xi - ri
        elif self.ctype == 2:
            return yi - ri
        else:
            return 1.0 - yi - ri


class CircleOverlapConstraint:
    def __init__(self, n, i, j):
        self.n = n
        self.i = i
        self.j = j
    
    def __call__(self, x):
        dx = x[2 * self.i] - x[2 * self.j]
        dy = x[2 * self.i + 1] - x[2 * self.j + 1]
        dist = math.sqrt(dx * dx + dy * dy)
        return dist - x[2 * self.n + self.i] - x[2 * self.n + self.j]


def initialize_hexagonal(n, base_r):
    """Initialize circles in a hexagonal pattern."""
    centers = np.zeros((n, 2))
    radii = np.full(n, base_r)
    
    idx = 0
    row_height = base_r * math.sqrt(3)
    col_width = 2 * base_r
    x_start = base_r
    y_start = base_r
    
    for row in range(8):
        for col in range(6):
            if idx >= n:
                break
            offset = col_width / 2 if row % 2 == 1 else 0
            cx = x_start + col * col_width + offset
            cy = y_start + row * row_height
            if cx + base_r <= 1 and cy + base_r <= 1:
                centers[idx, 0] = cx
                centers[idx, 1] = cy
                idx += 1
        if idx >= n:
            break
    
    return centers, radii


def force_directed_optimize(centers, radii, n, num_steps=5000):
    """Use force-directed relaxation to find good positions."""
    centers = centers.copy()
    radii = radii.copy()
    
    lr = 0.005
    best_sum = np.sum(radii)
    best_centers = centers.copy()
    best_radii = radii.copy()
    
    for step in range(num_steps):
        forces = np.zeros_like(centers)
        
        # Vectorized inter-circle repulsion
        for i in range(n):
            diff = centers - centers[i]  # shape (n, 2)
            diff[i, :] = 0  # zero out self
            dist_sq = np.sum(diff ** 2, axis=1)
            dist_sq[dist_sq < 1e-16] = 1e-16
            dist = np.sqrt(dist_sq)
            
            desired = radii + radii[i]
            overlap = desired - dist
            overlap = np.maximum(overlap, 0)
            
            # Force magnitude: proportional to overlap
            f_mag = overlap * 100.0 / (dist + 1e-10)
            
            fx = diff[:, 0] / (dist + 1e-10) * f_mag
            fy = diff[:, 1] / (dist + 1e-10) * f_mag
            
            forces[i, 0] = np.sum(fx)
            forces[i, 1] = np.sum(fy)
        
        # Wall repulsion
        for i in range(n):
            ri = radii[i]
            xi, yi = centers[i]
            if xi < ri:
                forces[i, 0] += (ri - xi) * 200.0
            if xi > 1 - ri:
                forces[i, 0] -= (xi - (1 - ri)) * 200.0
            if yi < ri:
                forces[i, 1] += (ri - yi) * 200.0
            if yi > 1 - ri:
                forces[i, 1] -= (yi - (1 - ri)) * 200.0
        
        # Apply forces with decaying learning rate
        effective_lr = lr * max(0.1, 1 - step / num_steps)
        centers += forces * effective_lr
        
        # Clip to valid range
        for i in range(n):
            radii[i] = np.maximum(radii[i], 0.001)
            centers[i, 0] = np.clip(centers[i, 0], 0.001, 0.999)
            centers[i, 1] = np.clip(centers[i, 1], 0.001, 0.999)
        
        # Gradually expand radii
        radii *= 1.00008
        
        current_sum = np.sum(radii)
        if current_sum > best_sum and step % 100 == 0:
            best_sum = current_sum
            best_centers = centers.copy()
            best_radii = radii.copy()
    
    return best_centers, best_radii


def check_validity(centers, radii, n):
    """Check if packing is valid."""
    for i in range(n):
        x, y, r = centers[i, 0], centers[i, 1], radii[i]
        if x - r < -1e-8 or x + r > 1 + 1e-8:
            return False
        if y - r < -1e-8 or y + r > 1 + 1e-8:
            return False
        if r < -1e-8:
            return False
    
    for i in range(n):
        for j in range(i + 1, n):
            dist = math.sqrt((centers[i, 0] - centers[j, 0]) ** 2 + (centers[i, 1] - centers[j, 1]) ** 2)
            if dist < radii[i] + radii[j] - 1e-8:
                return False
    
    return True


def local_optimize_slsqp(centers, radii, n):
    """Fine-tune using SLSQP optimization."""
    x0 = np.concatenate([centers.flatten(), radii])
    
    bounds = []
    for i in range(n):
        bounds.append((0.0, 1.0))
        bounds.append((0.0, 1.0))
    for i in range(n):
        bounds.append((0.001, 0.5))
    
    constraints = []
    
    # Boundary constraints
    for i in range(n):
        for ctype in range(4):
            c = CircleBoundaryConstraint(n, i, ctype)
            constraints.append({'type': 'ineq', 'fun': c})
    
    # Overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            c = CircleOverlapConstraint(n, i, j)
            constraints.append({'type': 'ineq', 'fun': c})
    
    def objective(x):
        return -np.sum(x[2 * n:])
    
    result = minimize(objective, x0, method='SLSQP', bounds=bounds,
                      constraints=constraints,
                      options={'maxiter': 200, 'ftol': 1e-12, 'disp': False})
    
    centers_opt = result.x[:2 * n].reshape(n, 2)
    radii_opt = result.x[2 * n:]
    
    return centers_opt, radii_opt, np.sum(radii_opt)


def run_packing():

            test_radii = best_radii * scale
            if check_validity(best_centers, test_radii, n):
                best_radii = test_radii
                break
            scale -= 0.001
    
    return best_centers, best_radii, np.sum(best_radii)
