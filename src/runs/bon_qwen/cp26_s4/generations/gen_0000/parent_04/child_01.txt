import numpy as np
from scipy.optimize import minimize

def run_packing():
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    """
    n = 26
    best_sum_radii = 0.0
    best_centers = None
    best_radii = None

    # Helper to flatten/unflatten variables
    def flatten(vars_list):
        return np.array([val for item in vars_list for val in item]).flatten()
    
    def unflatten(arr):
        # Returns list of [x, y, r] for each circle
        return [arr[3*i : 3*(i+1)] for i in range(n)]

    # Constraints
    def constraints_fun(v):
        vars_list = unflatten(v)
        c_list = []
        
        # Boundary constraints
        for i in range(n):
            x, y, r = vars_list[i]
            c_list.append(x - r)           # x >= r
            c_list.append(1 - x - r)       # x <= 1 - r
            c_list.append(y - r)           # y >= r
            c_list.append(1 - y - r)       # y <= 1 - r
            
            # Non-overlap constraints (lower triangular to avoid duplicates)
            for j in range(i + 1, n):
                x_j, y_j, r_j = vars_list[j]
                dist_sq = (x - x_j)**2 + (y - y_j)**2
                sum_r = r + r_j
                # Constraint: dist >= sum_r  => dist^2 >= sum_r^2
                c_list.append(dist_sq - sum_r**2)
                
        return np.array(c_list)

    # Objective: Maximize sum of radii -> Minimize -sum(radii)
    def objective(v):
        vars_list = unflatten(v)
        return -sum(r for _, _, r in vars_list)

    # Initial configuration generator (Hexagonal packing)
    def get_initial_guess(seed=0):

                if x < 0.95: # keep within bounds roughly
                    points.append([x, y])
                    col += 1
                else:
                    break
            row += 1
        
        # If we generated too many, trim. If too few, add random.
        # Actually the loop above might generate more or less depending on bounds.
        # Let's just ensure we have n points.
        if len(points) > n:
            points = points[:n]
        while len(points) < n:
            points.append([rng.uniform(0.2, 0.8), rng.uniform(0.2, 0.8)])
            
        # Set initial radii small to ensure feasibility
        radii_init = [0.01] * n
        
        vars_list = [[p[0], p[1], r] for p, r in zip(points, radii_init)]
        return flatten(vars_list)

    # Optimization parameters
    bounds = []

        # x in [0, 1]
        bounds.append((0.0, 1.0))
        # y in [0, 1]
        bounds.append((0.0, 1.0))
        # r in [0, 0.5]
        bounds.append((0.0, 0.5))

        try:

            if res.success:
                current_sum = -res.fun
                if current_sum > best_sum_radii:

        except Exception:
            continue

    # Final validation and return
    if best_centers is None:
        # Fallback to a simple valid packing if optimization failed
        # 5x5 grid with one extra small circle
