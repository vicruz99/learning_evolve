import numpy as np
from scipy.optimize import linprog

def get_active_forces(centers, radii, constraints_active):
    """
    Calculate forces on centers based on active constraints to increase clearance.
    """
    n = centers.shape[0]
    forces = np.zeros_like(centers)
    pair_slacks = constraints_active[0]
    wall_slacks = constraints_active[1]
    
    # Process pair constraints (circle-circle overlaps)
    for idx in range(len(pair_slacks)):
        i, j = pair_slacks[idx]['indices']
        slack = pair_slacks[idx]['slack']
        if slack < 1e-4:
            # Circle i and j are touching or overlapping
            # Push them apart
            diff = centers[j] - centers[i]
            dist = np.linalg.norm(diff)
            if dist > 1e-9:
                direction = diff / dist
                # Force magnitude inversely proportional to distance squared (stronger when closer)
                # or proportional to radius sum to ensure large circles move appropriately.
                force_mag = 0.1 * (radii[i] + radii[j]) 
                forces[i] -= direction * force_mag
                forces[j] += direction * force_mag
                
    # Process wall constraints
    for idx in range(len(wall_slacks)):
        i = wall_slacks[idx]['circle_idx']
        wall_type = wall_slacks[idx]['type']
        slack = wall_slacks[idx]['slack']
        
        if slack < 1e-4:
            force_vec = np.zeros(2)
            strength = 0.1 * radii[i]
            if wall_type == 'left':
                force_vec = np.array([strength, 0])
            elif wall_type == 'right':
                force_vec = np.array([-strength, 0])
            elif wall_type == 'bottom':
                force_vec = np.array([0, strength])
            elif wall_type == 'top':
                force_vec = np.array([0, -strength])
            forces[i] += force_vec

    return forces

def solve_packing_lp(centers):
    """
    Solves the Linear Programming problem to maximize sum of radii 
    given fixed centers.
    """
    n = centers.shape[0]
    
    # Objective: maximize sum(r) => minimize -sum(r)
    c_obj = np.ones(n)
    
    # Inequality constraints A_ub @ r <= b_ub
    A_ub = []
    b_ub = []
    
    # 1. Circle-Circle constraints: r_i + r_j <= distance(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dist)
            
    # 2. Boundary constraints
    # r_i <= x_i
    # r_i <= 1 - x_i
    # r_i <= y_i
    # r_i <= 1 - y_i
    for i in range(n):
        # Left wall
        row = np.zeros(n); row[i] = 1.0; A_ub.append(row); b_ub.append(centers[i, 0])
        # Right wall
        row = np.zeros(n); row[i] = 1.0; A_ub.append(row); b_ub.append(1.0 - centers[i, 0])
        # Bottom wall
        row = np.zeros(n); row[i] = 1.0; A_ub.append(row); b_ub.append(centers[i, 1])
        # Top wall
        row = np.zeros(n); row[i] = 1.0; A_ub.append(row); b_ub.append(1.0 - centers[i, 1])

    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    # Bounds: 0 <= r_i
    bounds = [(0, None) for _ in range(n)]
    
    # Solve
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return res.x, A_ub, b_ub
    else:
        return np.zeros(n), A_ub, b_ub

def check_constraints(centers, radii, A_ub, b_ub):
    """
    Check which constraints are active (slack near zero).
    """
    n = centers.shape[0]
    active_pairs = []
    active_walls = []
    
    # Pair constraints start at index 0
    pair_idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            slack = b_ub[pair_idx] - (radii[i] + radii[j])
            active_pairs.append({'indices': (i, j), 'slack': slack})
            pair_idx += 1
            
    # Wall constraints start after pairs
    wall_idx = pair_idx
    for i in range(n):
        # Left
        slack = b_ub[wall_idx] - radii[i]
        active_walls.append({'circle_idx': i, 'type': 'left', 'slack': slack})
        wall_idx += 1
        # Right
        slack = b_ub[wall_idx] - radii[i]
        active_walls.append({'circle_idx': i, 'type': 'right', 'slack': slack})
        wall_idx += 1
        # Bottom
        slack = b_ub[wall_idx] - radii[i]
        active_walls.append({'circle_idx': i, 'type': 'bottom', 'slack': slack})
        wall_idx += 1
        # Top
        slack = b_ub[wall_idx] - radii[i]
        active_walls.append({'circle_idx': i, 'type': 'top', 'slack': slack})
        wall_idx += 1

    return active_pairs, active_walls
