from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    Optimizes the positions and radii of 26 circles in a unit square
    to maximize the sum of radii.
    """
    n = 26

    # --- 1. Initial Configuration ---
    # Start with a 5x5 grid for 25 circles, plus 1 extra in the center
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.095)  # Slightly smaller than 0.1 to allow space

    # Fill 5x5 grid
    grid_size = 5
    spacing = 0.2
    idx = 0
    for r in range(grid_size):
        for c in range(grid_size):
            if idx < n:
                centers[idx, 0] = 0.1 + c * spacing
                centers[idx, 1] = 0.1 + r * spacing
                idx += 1
    
    # Place the 26th circle in the center
    centers[-1] = [0.5, 0.5]

    # --- 2. Optimization Setup ---
    def objective(x):
        # x contains radii followed by centers (r1..r26, x1..x26, y1..y26)
        r = x[:n]
        cx = x[n:2*n]
        cy = x[2*n:3*n]
        return -np.sum(r)  # Negate for minimization

    def constraints_factory(n_circles):
        constraints = []

        # Boundary constraints: r <= x <= 1-r and r <= y <= 1-r
        for i in range(n_circles):
            constraints.append({'type': 'ineq', 'fun': lambda x, i=i: x[i] - (x[n_circles + i] - x[i])}) # r <= x - r -> 2r <= x
            constraints.append({'type': 'ineq', 'fun': lambda x, i=i: (x[n_circles + i] + x[i]) - 1}) # x + r <= 1
            constraints.append({'type': 'ineq', 'fun': lambda x, i=i: x[i] - (x[2*n_circles + i] - x[i])}) # r <= y - r
            constraints.append({'type': 'ineq', 'fun': lambda x, i=i: (x[2*n_circles + i] + x[i]) - 1}) # y + r <= 1

        # Non-overlap constraints: dist >= r_i + r_j
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                def overlap_dist(x, i=i, j=j):
                    dx = x[n_circles + i] - x[n_circles + j]
                    dy = x[2*n_circles + i] - x[2*n_circles + j]
                    dist = np.sqrt(dx**2 + dy**2)
                    r_sum = x[i] + x[j]
                    return dist - r_sum
                
                constraints.append({'type': 'ineq', 'fun': overlap_dist})
        
        # Non-negative radii
        for i in range(n_circles):
            constraints.append({'type': 'ineq', 'fun': lambda x, i=i: x[i]})

        return constraints

    initial_r = radii
    initial_cx = centers[:, 0]
    initial_cy = centers[:, 1]
    x0 = np.concatenate([initial_r, initial_cx, initial_cy])

    constraints = constraints_factory(n)

    # --- 3. Run Optimizer ---
    # Using SLSQP as it handles constraints well
    res = minimize(objective, x0, method='SLSQP', constraints=constraints,

        optimal_cx = res.x[n:2*n]

    else:

        optimal_cx = initial_cx
        optimal_cy = initial_cy

    final_centers = np.column_stack((optimal_cx, optimal_cy))
    
    # Clean up tiny negative values due to numerical precision
    optimal_r = np.maximum(optimal_r, 0.0)

    sum_radii = np.sum(optimal_r)
    
    return final_centers, optimal_r, sum_radii
