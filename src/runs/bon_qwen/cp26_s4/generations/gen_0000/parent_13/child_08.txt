import numpy as np
from scipy.optimize import minimize

def run_packing():
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    
    Returns:
        centers: np.array of shape (26, 2)
        radii: np.array of shape (26)
        sum_radii: float
    """
    n = 26
    best_centers = None
    best_radii = None
    best_sum = -np.inf
    
    # Number of restarts to try
    num_restarts = 20
    
    # Penalty weight for constraints
    penalty_weight = 5000.0
    
    for k in range(num_restarts):
        # --- 1. Initialization ---
        # Create a hexagonal grid initialization
        spacing = 0.2 + (k % 5) * 0.01  # Vary density slightly
        pts = []
        row_y = 0.05
        row_idx = 0
        
        while row_y < 1.0:
            col_x = 0.05 + (row_idx % 2) * (spacing / 2)
            while col_x < 1.0:
                pts.append([col_x, row_y])
                col_x += spacing
            row_y += spacing * np.sqrt(3) / 2
            row_idx += 1
            
        pts = np.array(pts)
        # Take first 26 points
        if len(pts) < n:
            # Fallback if grid is too sparse (unlikely with 0.2 spacing)
            centers_init = np.random.uniform(0.1, 0.9, (n, 2))
        else:
            centers_init = pts[:n]
        
        # Add small random perturbation to break symmetry
        if k % 2 == 1:
            centers_init += np.random.normal(0, 0.01, centers_init.shape)
            # Clip to valid range
            centers_init = np.clip(centers_init, 0.01, 0.99)

        radii_init = np.full(n, 0.02)

        # Variables: [x0, y0, ..., x25, y25, r0, ..., r25]
        x0 = np.concatenate([centers_init.flatten(), radii_init])
        
        # Bounds: x, y in [0, 1], r in [0, 0.5]
        bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
        
        def objective(vars):
            centers = vars[:2*n].reshape(n, 2)
            radii = vars[2*n:]
            
            # Objective: Maximize sum of radii (minimize negative sum)
            obj = -np.sum(radii)

            # x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
            x, y = centers[:, 0], centers[:, 1]
            r = radii

            viol = np.maximum(0, -(x - r)) + \
                   np.maximum(0, -(1 - x - r)) + \
                   np.maximum(0, -(y - r)) + \
                   np.maximum(0, -(1 - y - r))
            
            obj += penalty_weight * np.sum(viol**2)
            
            # Penalty for overlaps between circles
            # dist >= r_i + r_j  =>  r_i + r_j - dist <= 0
            # Vectorized distance calculation
            # Compute upper triangle distances
            diff = centers[:, None, :] - centers[None, :, :] # (n, n, 2)
            dists = np.sqrt(np.sum(diff**2, axis=2) + 1e-12) # (n, n)
            
            # Sum of radii matrix
            r_sum = radii[:, None] + radii[None, :]
            
            # Violations: max(0, r_sum - dists)
            violations = np.maximum(0, r_sum - dists)
            
            # We only need to sum upper triangle to avoid double counting
            triu_indices = np.triu_indices(n, k=1)
            obj += penalty_weight * np.sum(violations[triu_indices]**2)
            
            return obj

        # Run optimizer
        try:
            res = minimize(objective, x0, method='L-BFGS-B', bounds=bounds, 
                           options={'maxiter': 1000, 'ftol': 1e-12})

    return best_centers, best_radii, best_sum


def is_valid_packing(centers, radii):
    """Helper function to validate the packing constraints"""
    n = centers.shape[0]
    
    # Boundary check
    x, y = centers[:, 0], centers[:, 1]
    r = radii
    
    if np.any(r < -1e-12): return False
    if np.any(x - r < -1e-12) or np.any(x + r > 1 + 1e-12): return False
    if np.any(y - r < -1e-12) or np.any(y + r > 1 + 1e-12): return False
    
    # Overlap check
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    r_sum = radii[:, None] + radii[None, :]
    
    # Check lower triangle (dist < r_i + r_j)
    # Using a small tolerance
    min_dist = np.min(dists - r_sum)
    return min_dist >= -1e-12
