import numpy as np
from scipy.optimize import minimize, linprog

N = 26

def penalty_objective(v, P):
    """
    Computes the objective: -sum(radii) + penalty for constraint violations.
    v: array of shape (3*N,) containing [x1, y1, r1, x2, y2, r2, ...]
    P: penalty multiplier
    """
    loss = -np.sum(v[2::3])
    
    # Boundary penalties
    for i in range(N):
        idx = 3 * i
        x, y, r = v[idx], v[idx + 1], v[idx + 2]
        for val in (x, 1.0 - x, y, 1.0 - y):
            if val < r:
                loss += P * (r - val) ** 2
                
        xi, yi, ri = x, y, r
        # Overlap penalties
        for j in range(i + 1, N):
            d = np.sqrt((xi - v[3 * j]) ** 2 + (yi - v[3 * j + 1]) ** 2)
            s = ri + v[3 * j + 2]
            if d < s:
                loss += P * (s - d) ** 2
    return loss

def run_packing():
    # Variable bounds: x in [0,1], y in [0,1], r in [0, 0.5]
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N
    
    best_v = None
    best_sum_r = -np.inf
    
    # Run multiple restarts with different seeds
    for seed in range(6):
        rng = np.random.RandomState(seed)
        v0 = np.zeros(3 * N)
        idx = 0
        cnt = 0
        
        # Initialize centers in a perturbed grid pattern
        for i in range(10):
            for j in range(10):
                if cnt >= N:
                    break
                x = (i + 0.5) / 10.0
                y = (j + 0.5) / 10.0
                # Hexagonal offset for odd rows
                if j % 2 == 1:
                    x += 0.05
                x = np.clip(x, 0.1, 0.9)
                y = np.clip(y, 0.1, 0.9)
                
                v0[idx] = x
                v0[idx + 1] = y
                v0[idx + 2] = 0.08  # Initial radius guess
                idx += 3
                cnt += 1
                
        # Add random perturbation
        v0 += rng.normal(0, 0.02, size=v0.shape)
        v0 = np.clip(v0, 0.0, 1.0)
        v0[2::3] = np.clip(v0[2::3], 0.01, 0.49)  # Ensure positive radii
        
        # Anneal penalty parameter P
        curr_v = v0.copy()
        P = 50.0
        for step in range(12):
            P *= 2.0
            res = minimize(
                penalty_objective, 
                curr_v, 
                args=(P,), 
                method='L-BFGS-B', 
                bounds=bounds, 
                options={'maxiter': 2000, 'ftol': 1e-12}
            )
            curr_v = res.x
            
        curr_sum = np.sum(curr_v[2::3])
        if curr_sum > best_sum_r:
            best_sum_r = curr_sum
            best_v = curr_v
            
    # Extract optimized centers
    centers = best_v[:2 * N].reshape((N, 2))
    
    # Exact radius refinement using Linear Programming
    # Maximize sum(r_i) s.t. r_i >= 0, r_i <= dist_to_boundary, r_i + r_j <= dist(i,j)
    c = -np.ones(N)
    A_ub = []
    b_ub = []
    
    for i in range(N):
        x, y = centers[i]
        r_max = min(x, 1.0 - x, y, 1.0 - y)
        row = np.zeros(N)
        row[i] = 1.0
        A_ub.append(row)
        b_ub.append(max(0.0, r_max))
        
    for i in range(N):
        for j in range(i + 1, N):
            d = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            row = np.zeros(N)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(max(0.0, d))
            
    res_lp = linprog(
        c, 
        A_ub=np.array(A_ub), 
        b_ub=np.array(b_ub), 
        bounds=[(0, None)] * N, 
        method='highs'
    )
    
    if res_lp.success:
        final_radii = res_lp.x
    else:
        # Fallback to optimizer radii if LP fails (shouldn't happen)
        final_radii = best_v[2::3]
        
    return centers, final_radii, float(np.sum(final_radii))
