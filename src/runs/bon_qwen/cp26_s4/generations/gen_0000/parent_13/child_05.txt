To solve the problem of packing 26 circles in a unit square to maximize the sum of radii, I will employ a two-stage computational strategy.

### Stage 1: Geometric Heuristics (Expansion Simulation)
A purely gradient-based approach often struggles with the hard inequality constraints (non-overlap) of circle packing. Therefore, the first stage will use a robust "inflation" simulation:
1.  **Initialization**: 26 centers are initialized in a dense hexagonal grid pattern, ensuring a uniform starting distribution.
2.  **Iterative Growth**: All circle radii are incremented simultaneously.
3.  **Constraint Resolution**: At each step, the simulation detects overlaps between circles and collisions with square boundaries. To resolve these, overlapping circles are pushed apart along the vector connecting their centers, and boundary collisions are clamped. This physical repulsion drives the circles toward a high-density, valid configuration.

### Stage 2: Continuous Optimization (SLSQP)
The result from Stage 1 provides a high-quality "warm start." However, it enforces equal radii and uses discrete steps. The second stage uses a local optimization algorithm to refine the solution:
1.  **Variable Relaxation**: We optimize both the centers $(x, y)$ and individual radii $r$ to maximize the objective function $\sum r_i$.
2.  **Solver**: The Sequential Least Squares Quadratic Programming (SLSQP) algorithm is used to handle the non-linear distance and boundary constraints.
3.  **Multi-Start**: To mitigate the risk of local minima, the optimizer is run multiple times from slightly perturbed versions of the initial solution. The best result across all runs is selected as the final output.

This approach leverages the global stability of the physical simulation to escape poor initial basins of attraction, followed by the precision of numerical optimization to squeeze out the final gains in the sum of radii.

import numpy as np
from scipy.optimize import minimize

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # --- STAGE 1: Expansion Simulation (Warm Start) ---
    
    # 1. Initialize centers in a hexagonal grid pattern
    # We need to pack 26 circles. A 5x5 grid is 25. We add one more.
    # Hexagonal spacing allows for denser packing.
    # Approximate radius is ~0.1. Diameter ~0.2.
    # Horizontal spacing 2r, vertical r*sqrt(3).
    
    centers = np.zeros((n, 2))
    idx = 0
    
    # Generate a grid that covers the square
    # We'll iterate rows.
    row_y = 0
    row_idx = 0
    while idx < n:
        # Determine x-start based on row parity for hexagonal packing
        x_start = 0.0 if row_idx % 2 == 0 else 0.1 # Offset by approx 1 diameter unit
        
        x = x_start
        while x <= 1.0 and idx < n:
            centers[idx, 0] = x
            centers[idx, 1] = row_y
            idx += 1
            x += 0.2 # Horizontal spacing
            
        row_y += 0.1732 # Vertical spacing (approx sqrt(3)/10)
        row_idx += 1
        
    # Clamp initial centers to be well within bounds
    centers = np.clip(centers, 0.05, 0.95)
    
    # 2. Iterative Growth and Repulsion
    radii = np.full(n, 0.01)
    
    # Simulation parameters
    growth_rate = 0.0005
    max_outer_steps = 2000
    inner_resolution_steps = 5
    
    for step in range(max_outer_steps):
        # Grow circles
        radii += growth_rate
        
        # Resolve overlaps and boundary collisions
        for _ in range(inner_resolution_steps):
            for i in range(n):
                # Boundary checks
                if centers[i, 0] < radii[i]: centers[i, 0] = radii[i]
                if centers[i, 0] > 1.0 - radii[i]: centers[i, 0] = 1.0 - radii[i]
                if centers[i, 1] < radii[i]: centers[i, 1] = radii[i]
                if centers[i, 1] > 1.0 - radii[i]: centers[i, 1] = 1.0 - radii[i]

            # Pairwise overlap resolution
            for i in range(n):
                for j in range(i + 1, n):
                    diff = centers[i] - centers[j]
                    dist = np.linalg.norm(diff)
                    min_dist = radii[i] + radii[j]
                    
                    if dist < min_dist and dist > 1e-9:
                        overlap = min_dist - dist
                        # Normalize vector
                        norm_diff = diff / dist
                        # Push apart (split overlap equally)
                        push_vec = norm_diff * (overlap / 2.0)
                        centers[i] += push_vec
                        centers[j] -= push_vec
                    elif dist == 0:
                        # Random perturbation to avoid singularity
                        centers[i] += np.random.randn(2) * 0.001

    # --- STAGE 2: Continuous Optimization (SLSQP) ---
    
    # We optimize x, y, r for all 26 circles
    # Variables: [x1, y1, r1, x2, y2, r2, ..., x26, y26, r26]
    # Total 78 variables.
    
    # Initial guess from simulation
    x0 = np.zeros(3 * n)
    for i in range(n):
        x0[3*i] = centers[i, 0]
        x0[3*i + 1] = centers[i, 1]
        x0[3*i + 2] = radii[i]

    def objective(vars):
        # Maximize sum of radii -> Minimize negative sum
        s = 0.0
        for i in range(n):
            s += vars[3*i + 2]
        return -s

    def constraint_wall(vars, wall_type, circle_idx, dir_sign):
        # dir_sign: 1 for positive direction (x+r <= 1 or y+r <= 1)
        #          -1 for negative direction (x-r >= 0 or y-r >= 0)
        # We need constraint >= 0 for SLSQP inequality
        x = vars[3*circle_idx]
        y = vars[3*circle_idx + 1]
        r = vars[3*circle_idx + 2]
        
        if wall_type == 'x':
            if dir_sign == -1: return x - r # x >= r
            else: return 1.0 - x - r # x+r <= 1 -> 1-x-r >= 0
        else: # y
            if dir_sign == -1: return y - r # y >= r
            else: return 1.0 - y - r # y+r <= 1 -> 1-y-r >= 0

    def constraint_overlap(vars, i, j):
        # dist^2 >= (ri + rj)^2
        # vars[3*i] is xi, vars[3*i+1] is yi, vars[3*i+2] is ri
        x1, y1, r1 = vars[3*i], vars[3*i+1], vars[3*i+2]
        x2, y2, r2 = vars[3*j], vars[3*j+1], vars[3*j+2]
        
        dist_sq = (x1 - x2)**2 + (y1 - y2)**2
        sum_r_sq = (r1 + r2)**2
        
        return dist_sq - sum_r_sq

    constraints = []
    
    # Wall constraints for all circles
    for i in range(n):
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: constraint_wall(v, 'x', i, -1)})
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: constraint_wall(v, 'x', i, 1)})
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: constraint_wall(v, 'y', i, -1)})
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: constraint_wall(v, 'y', i, 1)})
        
    # Overlap constraints for all pairs
    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({'type': 'ineq', 'fun': lambda v, i=i, j=j: constraint_overlap(v, i, j)})

    # Non-negative radii constraint
    bounds = [(None, None) for _ in range(3*n)]
    for i in range(n):
        bounds[3*i + 2] = (0, None) # Radius >= 0

    best_result = None
    best_sum = -np.inf

    # Run optimization multiple times with slight perturbations to avoid local minima
    for trial in range(3):
        # Perturb initial guess
        current_x0 = x0.copy()
        if trial > 0:
            noise = np.random.uniform(-0.02, 0.02, size=current_x0.shape)
            # Keep radii relatively stable during perturbation to maintain feasibility roughly
            noise[::3] *= 0.1 # Less noise in radius
            current_x0 += noise
            
        try:
            res = minimize(objective, current_x0, method='SLSQP', bounds=bounds, constraints=constraints, options={'ftol': 1e-12, 'maxiter': 1000})
            
            if res.success or res.fun < best_sum: # Lower fun is better (more negative sum)
                # Check validity manually just in case solver drifted
                c = np.array([[res.x[3*i], res.x[3*i+1]] for i in range(n)])
                r = np.array([res.x[3*i+2] for i in range(n)])
                
                # Clean up small negatives due to float errors
                r = np.maximum(r, 0)
                
                if validate_packing(c, r):
                    current_sum = np.sum(r)
                    if current_sum > best_sum:
                        best_sum = current_sum
                        best_result = (c, r)
        except Exception as e:
            pass

    if best_result is None:
        # Fallback to simulation result if optimizer failed
        best_result = (centers, radii)

    final_centers, final_radii = best_result
    final_sum = np.sum(final_radii)
    
    return final_centers, final_radii, final_sum

if __name__ == "__main__":
    centers, radii, s = run_packing()
    print(f"Sum of radii: {s}")
