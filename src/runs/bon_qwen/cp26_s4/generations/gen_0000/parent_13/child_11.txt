from scipy.optimize import minimize

def init_hexagonal(row_counts):
    n = sum(row_counts)
    centers = np.zeros((n, 2))
    idx = 0

    n_rows = len(row_counts)
    r_init = 0.1

    for row_i, count in enumerate(row_counts):
        y = (row_i + 0.5) * (1.0 / (n_rows + 1)) * n_rows / (n_rows - 1 + 1e-10) if n_rows > 1 else 0.5
        if n_rows == 1:
            y = 0.5
        else:
            y = (row_i + 0.5) / (n_rows + 1) * (n_rows + 1) / n_rows * 2 * r_init * np.sqrt(3) / 2 * 0.5

        row_width = (count - 1) * 2 * r_init
        x_start = (1.0 - row_width) / 2.0

        offset = (row_i % 2) * r_init

        for col in range(count):
            x = x_start + col * 2 * r_init + offset
            centers[idx] = [x, y]
            idx += 1

    # Normalize y to be in [0, 1]
    if centers[:, 1].max() > centers[:, 1].min() + 1e-10:
        y_range = centers[:, 1].max() - centers[:, 1].min()
        centers[:, 1] = (centers[:, 1] - centers[:, 1].min()) / y_range

    return centers, np.ones(n) * r_init


def init_grid(n):
    centers = np.zeros((n, 2))
    cols = int(np.ceil(np.sqrt(n)))
    rows = int(np.ceil(n / cols))

    idx = 0
    for i in range(rows):
        for j in range(cols):
            if idx >= n:
                break
            x = (j + 0.5) / cols
            y = (i + 0.5) / rows
            centers[idx] = [x, y]
            idx += 1

    return centers, np.ones(n) * 0.1


def objective(x, n):
    return -np.sum(x[2 * n:])


def c_x_left(x, i, n):
    return x[2 * i] - x[2 * n + i]


def c_x_right(x, i, n):
    return 1.0 - x[2 * i] - x[2 * n + i]


def c_y_bottom(x, i, n):
    return x[2 * i + 1] - x[2 * n + i]


def c_y_top(x, i, n):
    return 1.0 - x[2 * i + 1] - x[2 * n + i]


def c_overlap(x, i, j, n):
    dx = x[2 * i] - x[2 * j]
    dy = x[2 * i + 1] - x[2 * j + 1]
    return np.sqrt(dx * dx + dy * dy) - x[2 * n + i] - x[2 * n + j]


def c_radius(x, i, n):
    return x[2 * n + i]


def optimize_packing(centers, radii, n):
    x0 = np.concatenate([centers.flatten(), radii])

    bounds = []
    for i in range(n):
        bounds.append((0.0, 1.0))
        bounds.append((0.0, 1.0))
    for i in range(n):
        bounds.append((1e-8, 0.5))

    constraints = []
    for i in range(n):
        constraints.append({'type': 'ineq', 'fun': c_x_left, 'args': (i, n)})
        constraints.append({'type': 'ineq', 'fun': c_x_right, 'args': (i, n)})
        constraints.append({'type': 'ineq', 'fun': c_y_bottom, 'args': (i, n)})
        constraints.append({'type': 'ineq', 'fun': c_y_top, 'args': (i, n)})

    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({'type': 'ineq', 'fun': c_overlap, 'args': (i, j, n)})

    for i in range(n):
        constraints.append({'type': 'ineq', 'fun': c_radius, 'args': (i, n)})

    try:
        result = minimize(
            objective,
            x0,
            bounds=bounds,
            constraints=constraints,
            method='SLSQP',
            options={'maxiter': 20000, 'ftol': 1e-15, 'disp': False}
        )

        if result.success or result.fun < 0:
            centers_opt = result.x[:2 * n].reshape(n, 2)
            radii_opt = result.x[2 * n:]
            # Clip tiny radii
            radii_opt = np.maximum(radii_opt, 1e-8)
            return centers_opt, radii_opt
    except Exception:
        pass

    return None, None
