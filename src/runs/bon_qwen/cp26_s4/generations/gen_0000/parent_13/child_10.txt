import numpy as np
from scipy.optimize import minimize

def run_packing():
    """
    Optimizes the positions and radii of 26 circles in a unit square to maximize the sum of radii.
    Returns (centers, radii, sum_radii).
    """
    n_circles = 26
    best_centers = None
    best_radii = None
    max_sum = 0.0

    # Run optimization from multiple starting configurations to avoid local minima
    for trial in range(5):
        # 1. Initialization: Hexagonal Lattice
        centers = np.zeros((n_circles, 2))
        radii = np.zeros(n_circles)
        
        # Hexagonal parameters
        cols = 6
        rows = 5
        r_est = 0.09  # Initial estimate for radius
        dx = 2 * r_est
        dy = r_est * np.sqrt(3)
        
        idx = 0
        for r in range(rows):
            for c in range(cols):
                if idx >= n_circles:
                    break
                # Stagger every other row
                x_offset = 0 if r % 2 == 0 else r_est
                x = 0.05 + c * dx + x_offset
                y = 0.05 + r * dy
                
                # Wrap or shift if out of bounds to keep initial guess valid-ish
                x = np.clip(x, 0, 1)
                y = np.clip(y, 0, 1)
                
                centers[idx] = [x, y]
                radii[idx] = r_est
                idx += 1
        
        # 2. Optimization Function
        def objective_and_gradient(params):
            # Unpack parameters: [x1, y1, r1, ..., x26, y26, r26]
            # Grouped as (n, 3) for easier handling
            data = params.reshape(n_circles, 3)
            
            sum_radii = 0.0
            penalty = 0.0
            grad = np.zeros_like(data)
            
            # Weights for the penalty term to balance maximization and constraint satisfaction
            alpha = 1.0 
            beta = 100.0  # High weight for constraints
            
            # Calculate sum of radii (objective)
            sum_radii = np.sum(data[:, 2])
            
            # Constraint Violations
            for i in range(n_circles):
                x, y, r = data[i]
                
                # Boundary violations
                if x < r:
                    penalty += beta * (x - r) ** 2
                    grad[i, 0] -= alpha + 2 * beta * (x - r) * 1
                    grad[i, 2] -= alpha + 2 * beta * (x - r) * (-1)
                if x + r > 1:
                    penalty += beta * (x + r - 1) ** 2
                    grad[i, 0] -= alpha + 2 * beta * (x + r - 1) * 1
                    grad[i, 2] -= alpha + 2 * beta * (x + r - 1) * 1
                if y < r:
                    penalty += beta * (y - r) ** 2
                    grad[i, 1] -= alpha + 2 * beta * (y - r) * 1
                    grad[i, 2] -= alpha + 2 * beta * (y - r) * (-1)
                if y + r > 1:
                    penalty += beta * (y + r - 1) ** 2
                    grad[i, 1] -= alpha + 2 * beta * (y + r - 1) * 1
                    grad[i, 2] -= alpha + 2 * beta * (y + r - 1) * 1
                    
                # Objective gradient for radii
                grad[i, 2] += 1.0
            
            # Pairwise non-overlap violations
            for i in range(n_circles):
                for j in range(i + 1, n_circles):
                    dx = data[i, 0] - data[j, 0]
                    dy = data[i, 1] - data[j, 1]
                    dr = data[i, 2] + data[j, 2]
                    
                    dist_sq = dx**2 + dy**2
                    dist = np.sqrt(dist_sq)
                    
                    if dist < dr and dist > 1e-6:
                        # Violation: dist < dr
                        overlap = dr - dist
                        penalty += beta * overlap ** 2
                        
                        # Gradient calculation for overlap penalty
                        d_pen_d_dist = 2 * beta * overlap * (-1) # derivative of (dr-dist)^2 wrt dist
                        # Chain rule: d_dist/d_xi = -dx/dist, d_dist/d_yi = -dy/dist
                        d_dist_dx = dx / dist
                        d_dist_dy = dy / dist
                        
                        # Update gradients
                        # For circle i
                        grad[i, 0] += d_pen_d_dist * d_dist_dx
                        grad[i, 1] += d_pen_d_dist * d_dist_dy
                        grad[i, 2] += d_pen_d_dist * 1.0 # derivative of (ri+rj-dist) wrt ri is 1
                        # For circle j
                        grad[j, 0] += d_pen_d_dist * (-d_dist_dx)
                        grad[j, 1] += d_pen_d_dist * (-d_dist_dy)
                        grad[j, 2] += d_pen_d_dist * 1.0 # derivative of (ri+rj-dist) wrt rj is 1
            
            # We want to minimize (penalty - sum_radii)
            # But scipy minimizes. So we return (penalty - sum_radii).
            # However, standard minimizers might struggle with the huge penalty scale.
            # Let's return just the negative sum plus penalty.
            return penalty - sum_radii, grad.reshape(-1)

        params0 = np.hstack([centers, radii.reshape(-1, 1)]).flatten()
        
        # Optimization
        # Using SLSQP as it handles bounds and gradients well
        bounds = []
        for i in range(n_circles):
            bounds.append((0.0, 1.0)) # x
            bounds.append((0.0, 1.0)) # y
            bounds.append((0.0, 0.5)) # r (max radius in unit square is 0.5)
            
        options = {'ftol': 1e-9, 'gtol': 1e-8, 'maxiter': 500}
        res = minimize(objective_and_gradient, params0, method='L-BFGS-B', 
                       jac=True, bounds=bounds, options=options)
        
        # Extract results
        data_opt = res.x.reshape(n_circles, 3)
        centers_opt = data_opt[:, :2]
        radii_opt = data_opt[:, 2]
        
        current_sum = np.sum(radii_opt)
        
        # Verify validity (strictly, though optimizer should handle it)
        valid = True
        # Check radii non-negative (handled by bounds)
        # Check overlaps and bounds with a small tolerance
        for i in range(n_circles):
            x, y, r = centers_opt[i], centers_opt[i+1] if i+1 < n_circles else (0,0), radii_opt[i]
            if x - r < -1e-5 or x + r > 1 + 1e-5 or radii_opt[i] < -1e-5:
                valid = False
                break
        
        if valid and current_sum > max_sum:
            max_sum = current_sum
            best_centers = centers_opt.copy()
            best_radii = radii_opt.copy()

    return best_centers, best_radii, float(max_sum)
