import numpy as np
from scipy.optimize import minimize

def run_packing():
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    Returns (centers, radii, sum_radii).
    """
    n = 26
    
    # 1. Initial Configuration: Hexagonal Lattice
    # We generate points in a hexagonal pattern.
    # Row spacing is approx sqrt(3)/2 * diameter. 
    # We start with a reasonable radius estimate to place centers.
    # For n=26, r is around 0.1. Let's start with centers based on r=0.1.
    
    centers = []
    r_init = 0.1
    # Vertical spacing for hex packing
    h = r_init * np.sqrt(3)
    
    # Generate rows
    y = r_init
    row_idx = 0
    while y <= 1 - r_init:
        # Horizontal spacing is 2*r
        # Offset for odd rows
        x_start = r_init + (row_idx % 2) * r_init
        x = x_start
        while x <= 1 - r_init:
            centers.append([x, y])
            x += 2 * r_init
        y += h
        row_idx += 1
    
    # We might have more or fewer than 26 points.
    # If more, we keep the first 26 (or select best).
    # If fewer, we need to adjust.
    # With r=0.1, we expect around 25-27 points.
    
    if len(centers) > n:
        centers = centers[:n]
    elif len(centers) < n:
        # Fill remaining with random valid points or perturbation
        # This case is unlikely with r=0.1, but just in case.
        while len(centers) < n:
            centers.append([np.random.rand(), np.random.rand()])
            
    centers = np.array(centers)
    
    # Initialize radii slightly smaller to ensure feasibility
    radii = np.full(n, 0.05)
    
    # 2. Optimization Setup
    # Variables: [x1, y1, r1, x2, y2, r2, ..., x26, y26, r26]
    # Total 3 * n variables.
    
    def objective(vars):
        # vars is 1D array of length 3*n
        # We want to maximize sum of radii, so minimize negative sum
        r = vars[2::3]
        return -np.sum(r)
    
    # Constraints
    constraints = []
    
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    # x - r >= 0  => r - x <= 0
    # 1 - x - r >= 0 => x + r - 1 <= 0
    # Same for y
    
    # We can define these as linear constraints or nonlinear.
    # Using nonlinear for generality with the solver.
    
    def constraint_x_lower(vars):
        x = vars[0::3]
        r = vars[2::3]
        return x - r # >= 0
    
    def constraint_x_upper(vars):
        x = vars[0::3]
        r = vars[2::3]
        return 1 - x - r # >= 0
        
    def constraint_y_lower(vars):
        y = vars[1::3]
        r = vars[2::3]
        return y - r # >= 0
        
    def constraint_y_upper(vars):
        y = vars[1::3]
        r = vars[2::3]
        return 1 - y - r # >= 0

    def constraint_r_nonneg(vars):
        r = vars[2::3]
        return r # >= 0

    # Non-overlap constraints: dist^2 >= (r_i + r_j)^2
    # dist^2 - (r_i + r_j)^2 >= 0
    
    def constraint_no_overlap(vars):
        # This function returns a vector of constraint values for all pairs
        # However, passing a huge vector to constraints list might be inefficient.
        # Better to define individual constraints or use a loop in the solver options?
        # SLSQP can handle vectorized constraints if 'type' is 'ineq'.
        
        x = vars[0::3]
        y = vars[1::3]
        r = vars[2::3]
        
        vals = []
        for i in range(n):
            for j in range(i + 1, n):
                dx = x[i] - x[j]
                dy = y[i] - y[j]
                dr = r[i] + r[j]
                vals.append(dx*dx + dy*dy - dr*dr)
        return np.array(vals)

    # Add constraints
    constraints.append({'type': 'ineq', 'fun': constraint_x_lower})
    constraints.append({'type': 'ineq', 'fun': constraint_x_upper})
    constraints.append({'type': 'ineq', 'fun': constraint_y_lower})
    constraints.append({'type': 'ineq', 'fun': constraint_y_upper})
    constraints.append({'type': 'ineq', 'fun': constraint_r_nonneg})
    constraints.append({'type': 'ineq', 'fun': constraint_no_overlap})
    
    # Bounds
    # x, y in [0, 1], r in [0, 0.5] (max possible radius is 0.5)
    bounds = []
    for i in range(n):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 0.5)) # r
        
    # Initial guess
    x0 = np.zeros(3 * n)
    x0[0::3] = centers[:, 0]
    x0[1::3] = centers[:, 1]
    x0[2::3] = radii
    
    # 3. Run Optimizer
    # Using SLSQP
    try:
        res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints, 
                       options={'maxiter': 1000, 'ftol': 1e-9, 'disp': False})
        
        if res.success:
            final_centers = np.column_stack((res.x[0::3], res.x[1::3]))
            final_radii = res.x[2::3]
        else:
            # Fallback to initial if optimization fails (unlikely)
            final_centers = centers
            final_radii = radii
            
    except Exception as e:
        # Fallback
        final_centers = centers
        final_radii = radii

    sum_radii = np.sum(final_radii)
    return final_centers, final_radii, sum_radii
