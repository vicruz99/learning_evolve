import numpy as np
from scipy.optimize import minimize


def compute_objective(x, n):
    """Compute objective: negative sum of radii plus penalty for constraint violations."""
    c = x[:2 * n].reshape(n, 2)
    r = np.maximum(x[2 * n:], 1e-10)

    # Objective: maximize sum of radii (minimize negative)
    obj = -np.sum(r)

    # Boundary penalties: circles must be inside [0,1]x[0,1]
    pen = 10000.0 * (
        np.sum(np.maximum(r - c[:, 0], 0) ** 2) +
        np.sum(np.maximum(c[:, 0] + r - 1, 0) ** 2) +
        np.sum(np.maximum(r - c[:, 1], 0) ** 2) +
        np.sum(np.maximum(c[:, 1] + r - 1, 0) ** 2)
    )

    # Overlap penalties: circles must not overlap
    for i in range(n):
        for j in range(i + 1, n):
            d = np.sqrt(np.sum((c[i] - c[j]) ** 2))
            overlap = r[i] + r[j] - d
            if overlap > 0:
                pen += 10000.0 * overlap ** 2

    return obj + pen


def is_valid_packing(centers, radii, n):
    """Check if the packing satisfies all constraints."""
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if r < -1e-12:
            return False
        if x - r < -1e-12 or x + r > 1 + 1e-12:
            return False
        if y - r < -1e-12 or y + r > 1 + 1e-12:
            return False

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False

    return True


def init_hexagonal(n, r0, noise_scale=0.001):
    """Initialize circles in a hexagonal grid pattern."""
    centers = []
    radii = []
    cnt = 0
    row = 0
    y = r0

    while cnt < n:
        shift = (row % 2) * r0
        x = r0 + shift
        while x <= 1 - r0 and cnt < n:
            centers.append([x, y])
            radii.append(r0)
            x += 2 * r0
            cnt += 1
        y += r0 * np.sqrt(3)
        row += 1

    centers = np.array(centers[:n])
    radii = np.array(radii[:n])

    if noise_scale > 0:
        centers = centers + np.random.randn(n, 2) * noise_scale

    return centers, radii


def run_packing():
    """Pack 26 circles in unit square to maximize sum of radii."""
    n = 26

    best_sum = 0.0
    best_centers = np.zeros((n, 2))
    best_radii = np.zeros(n)

    # Try multiple initial configurations
    for seed in range(50):
        np.random.seed(seed)

        # Vary initial radius to explore different configurations
        r0 = 0.080 + seed * 0.0015
        if r0 > 0.13:
            r0 = 0.13

        centers, radii = init_hexagonal(n, r0, noise_scale=0.002)

        # Flatten for optimization
        x0 = np.concatenate([centers.ravel(), radii])

        # Optimize using BFGS with penalty method
        result = minimize(
            compute_objective,
            x0,
            args=(n,),
            method='BFGS',
            options={'maxiter': 10000, 'gtol': 1e-13, 'ftol': 1e-15}
        )

        c_opt = result.x[:2 * n].reshape(n, 2)
        r_opt = np.maximum(result.x[2 * n:], 0)

        # Validate the solution
        if is_valid_packing(c_opt, r_opt, n):
            s = np.sum(r_opt)
            if s > best_sum:
                best_sum = s
                best_centers = c_opt.copy()
                best_radii = r_opt.copy()

    return best_centers, best_radii, best_sum
