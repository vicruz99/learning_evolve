import numpy as np

def run_packing():
    n = 26
    np.random.seed(42)
    
    # 1. Initialization on a perturbed grid
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.02
    
    idx = 0
    for i in range(5):
        for j in range(5):
            centers[idx] = [i * 0.2 + 0.1, j * 0.2 + 0.1]
            idx += 1
    centers[25] = [0.5, 0.5]
    centers += np.random.uniform(-0.02, 0.02, size=centers.shape)
    
    # Ensure initial validity
    for i in range(n):
        centers[i, 0] = np.clip(centers[i, 0], radii[i], 1-radii[i])
        centers[i, 1] = np.clip(centers[i, 1], radii[i], 1-radii[i])
        
    max_sum = 0.0
    best_c = centers.copy()
    best_r = radii.copy()
    
    dt = 0.00005
    
    # 2. Main optimization loop
    for step in range(6000):
        # Grow radii with slight random variation to allow inequality
        growth = 1.0 + 0.1 * (np.random.rand(n) - 0.5)
        radii += dt * growth
        
        # 3. Inner relaxation loop to resolve overlaps
        for _ in range(40):
            # Boundary constraints
            centers[:, 0] = np.clip(centers[:, 0], radii, 1 - radii)
            centers[:, 1] = np.clip(centers[:, 1], radii, 1 - radii)
            
            # Vectorized pairwise distance and overlap calculation
            dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
            dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
            dist = np.sqrt(dx**2 + dy**2 + 1e-12)
            min_dist = radii[:, np.newaxis] + radii[np.newaxis, :]
            
            overlap = min_dist - dist
            np.fill_diagonal(overlap, 0)
            mask = overlap > 0
            
            if np.any(mask):
                inv_dist = 1.0 / dist
                push_mag = np.where(mask, overlap / 2.0, 0.0) * 0.5
                push_x = dx * inv_dist * push_mag
                push_y = dy * inv_dist * push_mag
                
                # Apply aggregated pushes
                centers[:, 0] += np.sum(push_x, axis=1)
                centers[:, 1] += np.sum(push_y, axis=1)
                
                # Re-clip after push
                centers[:, 0] = np.clip(centers[:, 0], radii, 1 - radii)
                centers[:, 1] = np.clip(centers[:, 1], radii, 1 - radii)
                
        # 4. Validate current state
        dx = centers[:, 0, np.newaxis] - centers[np.newaxis, :, 0]
        dy = centers[:, 1, np.newaxis] - centers[np.newaxis, :, 1]
        dist = np.sqrt(dx**2 + dy**2 + 1e-12)
        min_dist = radii[:, np.newaxis] + radii[np.newaxis, :]
        overlap = np.maximum(min_dist - dist, 0)
        np.fill_diagonal(overlap, 0)
        
        valid = (np.max(overlap) < 1e-4 and 
                 np.all(centers[:, 0] >= radii) and 
                 np.all(centers[:, 1] >= radii) and
                 np.all(centers[:, 0] <= 1 - radii) and 
                 np.all(centers[:, 1] <= 1 - radii))
                 
        if valid:
            current_sum = np.sum(radii)
            if current_sum > max_sum:
                max_sum = current_sum
                best_c = centers.copy()
                best_r = radii.copy()
                
        # 5 & 6. Cooling and periodic shaking to escape local minima
        if step % 200 == 0:
            dt *= 0.96
            
        if step % 500 == 0 and step > 0:
            centers += np.random.uniform(-0.005, 0.005, size=centers.shape)
            centers[:, 0] = np.clip(centers[:, 0], radii, 1 - radii)
            centers[:, 1] = np.clip(centers[:, 1], radii, 1 - radii)
            
    return best_c, best_r, max_sum
