The problem of maximizing the sum of radii of $n$ non-overlapping circles in a unit square is a non-convex optimization problem. Since $n=26$ is relatively small, we can formulate it directly as a constrained nonlinear optimization problem and solve it using the Sequential Least Squares Programming (SLSQP) algorithm from `scipy.optimize`.

Key steps in the strategy:
1. **Decision Variables**: We optimize over $3n$ variables: $(x_i, y_i, r_i)$ for each circle $i=1,\dots,26$.
2. **Objective Function**: Maximize $\sum r_i$, which is equivalent to minimizing $-\sum r_i$.
3. **Constraints**:
   - Boundary: $r_i \le x_i \le 1-r_i$ and $r_i \le y_i \le 1-r_i$ for all $i$.
   - Non-overlap: $(x_i-x_j)^2 + (y_i-y_j)^2 \ge (r_i+r_j)^2$ for all $i<j$. Using squared distances avoids expensive square roots and preserves inequality direction.
4. **Initialization**: SLSQP is a local optimizer. To increase the chance of finding the global optimum, we run multiple optimizations from different feasible starting points (structured grids and randomized spreads). All starts use small initial radii to guarantee feasibility.
5. **Robustness**: We collect the best result across all runs, apply minor numerical clamping to satisfy strict validation tolerances, and return the configuration.

This approach directly targets the objective function while strictly enforcing geometric constraints, typically yielding packings near or above the target sum of radii.

import numpy as np
from scipy.optimize import minimize

def objective(z):
    """Objective function: minimize negative sum of radii."""
    radii = z[2::3]
    return -np.sum(radii)

def constraints(z):
    """Constraint function: returns array of constraint values that must be >= 0."""
    n = len(z) // 3
    x = z[0::3]
    y = z[1::3]
    r = z[2::3]
    
    # Boundary constraints: x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
    c_boundary = np.concatenate([x - r, 1.0 - x - r, y - r, 1.0 - y - r])
    
    # Overlap constraints: dist^2 - (r_i + r_j)^2 >= 0
    X = x[:, None]
    Y = y[:, None]
    R = r[:, None]
    ii, jj = np.triu_indices(n, k=1)
    c_overlap = (X[ii] - X[jj])**2 + (Y[ii] - Y[jj])**2 - (R[ii] + R[jj])**2
    
    return np.concatenate([c_boundary, c_overlap])

def get_bounds(n):
    """Variable bounds: x,y in [0,1], r in [0, 0.5]."""
    b = [(0.0, 1.0)] * n + [(0.0, 1.0)] * n + [(0.0, 0.5)] * n
    return b

def solve_packing(n, z0):
    """Run SLSQP optimization from initial point z0."""
    bounds = get_bounds(n)
    cons = {'type': 'ineq', 'fun': constraints}
    try:
        res = minimize(objective, z0, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'ftol': 1e-12, 'maxiter': 3000, 'disp': False})
        return res
    except Exception:
        return None

def run_packing():
    n = 26
    best_val = -np.inf
    best_z = None
    
    np.random.seed(42)
    initial_configs = []
    
    # 1. Structured grid starts with slight shifts
    for s in np.linspace(0.0, 0.05, 3):
        gx = np.linspace(0.15 + s, 0.85 + s, 6)
        gy = np.linspace(0.15 + s, 0.85 + s, 6)
        xx, yy = np.meshgrid(gx, gy)
        pts = np.vstack([xx.ravel()[:n], yy.ravel()[:n]]).T
        z = np.zeros(3*n)
        z[0::3] = pts[:,0]
        z[1::3] = pts[:,1]
        z[2::3] = 0.01  # Small feasible radii
        initial_configs.append(z)
        
    # 2. Random starts
    for _ in range(10):
        cx = np.random.rand(n) * 0.8 + 0.1
        cy = np.random.rand(n) * 0.8 + 0.1
        z = np.zeros(3*n)
        z[0::3] = cx
        z[1::3] = cy
        z[2::3] = 0.005
        initial_configs.append(z)
        
    # Optimize from each start
    for z0 in initial_configs:
        res = solve_packing(n, z0)
        if res is not None:
            val = -res.fun
            if val > best_val:
                best_val = val
                best_z = res.x
                
    if best_z is not None:
        centers = np.column_stack((best_z[0::3], best_z[1::3]))
        radii = best_z[2::3]
        
        # Numerical safety clamping
        centers = np.clip(centers, 0.0, 1.0)
        radii = np.maximum(radii, 0.0)
        
        return centers, radii, np.sum(radii)
    
    # Fallback (should not be reached)
    centers = np.random.rand(n, 2) * 0.8 + 0.1
    radii = np.full(n, 0.01)
    return centers, radii, np.sum(radii)
