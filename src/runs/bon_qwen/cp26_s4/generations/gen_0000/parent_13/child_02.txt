import numpy as np
from scipy.optimize import minimize

N = 26

def get_initial_guess():
    centers = np.zeros((N, 2))
    radii = np.zeros(N)
    r_init = 0.08
    row_counts = [6, 5, 6, 5, 4]
    idx = 0
    for row, count in enumerate(row_counts):
        y = r_init + row * np.sqrt(3) * r_init
        for col in range(count):
            x_offset = 0.0 if row % 2 == 0 else r_init
            x = r_init + col * 2 * r_init + x_offset
            centers[idx] = [x, y]
            radii[idx] = r_init
            idx += 1
    return centers, radii

def objective(v):
    return -np.sum(v[2::3])

def objective_grad(v):
    grad = np.zeros_like(v)
    grad[2::3] = -1.0
    return grad

def constraints_func(v):
    n_con = N*4 + N*(N-1)//2
    c = np.zeros(n_con)
    idx = 0
    for i in range(N):
        xi, yi, ri = v[3*i], v[3*i+1], v[3*i+2]
        c[idx] = xi - ri
        c[idx+1] = 1.0 - xi - ri
        c[idx+2] = yi - ri
        c[idx+3] = 1.0 - yi - ri
        idx += 4
    for i in range(N):
        for j in range(i+1, N):
            dx = v[3*i] - v[3*j]
            dy = v[3*i+1] - v[3*j+1]
            dist2 = dx*dx + dy*dy
            sum_r = v[3*i+2] + v[3*j+2]
            c[idx] = dist2 - sum_r*sum_r
            idx += 1
    return c

def run_packing():
    np.random.seed(42)
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * N
    
    best_val = -1e9
    best_v = None
    
    for trial in range(3):
        centers0, radii0 = get_initial_guess()
        v0 = np.zeros(3*N)
        v0[0::3] = centers0[:, 0]
        v0[1::3] = centers0[:, 1]
        v0[2::3] = radii0
        
        if trial > 0:
            v0[0::3] += np.random.uniform(-0.02, 0.02, N)
            v0[1::3] += np.random.uniform(-0.02, 0.02, N)
            v0 = np.clip(v0, 0.0, 1.0)
            v0[2::3] = np.clip(v0[2::3], 0.0, 0.5)
            
        res = minimize(objective, v0, method='SLSQP', jac=objective_grad,
                       bounds=bounds, constraints={'type': 'ineq', 'fun': constraints_func},
                       options={'maxiter': 3000, 'ftol': 1e-10})
                       
        if not np.isnan(res.fun):
            val = -res.fun
            if val > best_val:
                best_val = val
                best_v = res.x.copy()
                
    centers = np.zeros((N, 2))
    radii = np.zeros(N)
    for i in range(N):
        centers[i] = [best_v[3*i], best_v[3*i+1]]
        radii[i] = best_v[3*i+2]
        
    return centers, radii, np.sum(radii)
