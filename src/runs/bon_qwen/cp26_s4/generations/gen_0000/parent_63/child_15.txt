def run_packing():
    """
    Pack 26 circles in a unit square to maximize the sum of radii.
    Uses optimization to find the maximum radius for equal circles,
    starting from a hexagonal lattice initialization.
    """
    n = 26
    
    # Helper to generate initial centers in a hexagonal pattern
    def get_initial_centers():
        centers_list = []
        # Base x coordinates for 5 circles
        x_base = [0.1, 0.3, 0.5, 0.7, 0.9]
        # Shifted x coordinates for 4 circles
        x_shifted = [0.2, 0.4, 0.6, 0.8]
        
        y_curr = 0.1
        row_idx = 0
        
        while len(centers_list) < 26:
            if row_idx % 2 == 0:
                xs = x_base
            else:
                xs = x_shifted
            
            for x in xs:
                centers_list.append([x, y_curr])
                if len(centers_list) >= 26:
                    break
            
            # Vertical spacing for hexagonal packing (sqrt(3)/2 * spacing)
            # Assuming spacing of 0.2 (diameter 0.2 -> r=0.1) for layout
            y_curr += 0.173205 
            row_idx += 1
            
        return np.array(centers_list[:26])

    # Optimization function
    def solve_packing(centers_init):
        # Start with a small feasible radius
        r_init = 0.01
        x0 = np.concatenate([centers_init.flatten(), [r_init]])
        
        # Bounds: x, y in [0, 1], r in [0, 0.5]
        bounds = [(0.0, 1.0)] * 52 + [(0.0, 0.5)]

        def objective(vars):
            # Maximize r (minimize -r)
            return -vars[-1]

        def constraints(vars):
            centers = vars[:52].reshape(26, 2)
            r = vars[52]
            con = []
            
            # Boundary constraints: circle must be inside [0,1]x[0,1]
            for i in range(26):
                x, y = centers[i]
                con.append(x - r)       # x - r >= 0
                con.append(1.0 - x - r) # 1 - x - r >= 0
                con.append(y - r)       # y - r >= 0
                con.append(1.0 - y - r) # 1 - y - r >= 0
            
            # Overlap constraints: distance >= 2r
            for i in range(26):
                for j in range(i + 1, 26):
                    dx = centers[i, 0] - centers[j, 0]
                    dy = centers[i, 1] - centers[j, 1]
                    dist = np.hypot(dx, dy)
                    con.append(dist - 2.0 * r)
            
            return np.array(con)

        try:
            # SLSQP is suitable for constrained non-linear optimization
            res = opt.minimize(objective, x0, method='SLSQP', bounds=bounds,
                               constraints={'type': 'ineq', 'fun': constraints},
                               options={'maxiter': 3000, 'ftol': 1e-12, 'disp': False})
            if res.success:
                return res.x, res.x[-1]
        except Exception:
            pass
        return None, 0.0

    # 1. Try optimization from hexagonal initialization
    centers_hex = get_initial_centers()
    res_x, best_r = solve_packing(centers_hex)
    
    if res_x is not None:
        best_centers = res_x[:52].reshape(26, 2)
    else:
        best_centers = centers_hex
        best_r = 0.05 # Fallback radius

    # 2. Try a few random perturbations to escape local minima

    # np.random.seed(42) 
    
    for _ in range(3):
        # Perturb centers slightly
        perturbation = np.random.uniform(-0.02, 0.02, centers_hex.shape)

        if res_x is not None and r_val > best_r:

            best_r = r_val

    # Ensure radii are valid

        best_r = 0.0
    best_radii = np.full(26, best_r)

    return (best_centers, best_radii, total_sum)
