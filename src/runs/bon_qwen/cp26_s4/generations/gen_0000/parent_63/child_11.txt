import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def objective(vars):
    radii = vars[2::3]
    return -np.sum(radii)

def objective_jac(vars):
    jac = np.zeros_like(vars)
    jac[2::3] = -1.0
    return jac

def constraint_func(vars):
    n = N_CIRCLES
    m = 4 * n + n * (n - 1) // 2
    vals = np.zeros(m)
    idx = 0
    for i in range(n):
        xi, yi, ri = vars[3*i], vars[3*i+1], vars[3*i+2]
        vals[idx] = xi - ri
        vals[idx+1] = 1.0 - ri - xi
        vals[idx+2] = yi - ri
        vals[idx+3] = 1.0 - ri - yi
        idx += 4
        for j in range(i + 1, n):
            xj, yj, rj = vars[3*j], vars[3*j+1], vars[3*j+2]
            vals[idx] = (xi - xj)**2 + (yi - yj)**2 - (ri + rj)**2
            idx += 1
    return vals

def run_packing():
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Multiple restarts to avoid local optima
    for seed in range(8):
        np.random.seed(seed)
        
        x0 = np.zeros(3 * N_CIRCLES)
        idx = 0
        # Hexagonal-ish initialization for good starting geometry
        for i in range(5):
            offset = (i % 2) * 0.5
            for j in range(6):
                if idx >= N_CIRCLES:
                    break
                x0[idx*3] = (j + 0.5 + offset) / 6.0
                x0[idx*3 + 1] = (i + 0.5) / 5.0
                x0[idx*3 + 2] = 0.08
                idx += 1
                
        # Perturb centers to break symmetry
        noise = np.random.uniform(-0.03, 0.03, 52)
        x0[:52] = np.clip(x0[:52] + noise, 0.05, 0.95)
        
        bounds = [(0.0, 1.0)] * 52 + [(0.0, 0.5)] * 26
        cons = {'type': 'ineq', 'fun': constraint_func}
        
        try:
            res = minimize(objective, x0, method='SLSQP', jac=objective_jac, 
                           bounds=bounds, constraints=cons, 
                           options={'maxiter': 2500, 'ftol': 1e-9, 'disp': False})
            current_sum = -res.fun
            if current_sum > best_sum:
                best_sum = current_sum
                best_centers = res.x[:52].reshape(-1, 2)
                best_radii = res.x[52:]
        except Exception:
            pass

    if best_centers is None:
        best_centers = x0[:52].reshape(-1, 2)
        best_radii = x0[52:]
        best_sum = np.sum(best_radii)
        
    best_radii = np.maximum(best_radii, 0.0)
    
    return best_centers, best_radii, best_sum
