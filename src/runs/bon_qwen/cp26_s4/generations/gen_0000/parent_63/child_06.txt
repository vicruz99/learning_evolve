import numpy as np
from scipy.optimize import minimize


def obj_fun(x, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(x[:n])


def bc_left(x, i, n):
    """Boundary: center_x >= radius."""
    return x[n + 2 * i] - x[i]


def bc_right(x, i, n):
    """Boundary: center_x + radius <= 1."""
    return 1.0 - x[n + 2 * i] - x[i]


def bc_bottom(x, i, n):
    """Boundary: center_y >= radius."""
    return x[n + 2 * i + 1] - x[i]


def bc_top(x, i, n):
    """Boundary: center_y + radius <= 1."""
    return 1.0 - x[n + 2 * i + 1] - x[i]


def non_overlap(x, i, j, n):
    """Non-overlap: dist^2 >= (r_i + r_j)^2."""
    ri = x[i]
    rj = x[j]
    cxi = x[n + 2 * i]
    cyi = x[n + 2 * i + 1]
    cxj = x[n + 2 * j]
    cyj = x[n + 2 * j + 1]
    dx = cxi - cxj
    dy = cyi - cyj
    return dx * dx + dy * dy - (ri + rj) ** 2


def hex_init(rows, r):
    """Create hexagonal lattice initialization."""
    centers = []
    y = r
    for ri, cnt in enumerate(rows):
        xs = r if ri % 2 == 0 else 1.5 * r
        for ci in range(cnt):
            centers.append([xs + ci * 2 * r, y])
        y += r * np.sqrt(3)
    return np.array(centers)


def grid_init(n):
    """Create a grid-like initialization."""
    centers = []
    for i in range(5):
        for j in range(5):
            centers.append([0.1 + i * 0.2, 0.1 + j * 0.2])
    centers.append([0.5, 0.5])
    return np.array(centers)


def run_packing():
    """Pack 26 circles in unit square to maximize sum of radii."""
    n = 26
    best_s = -1.0
    best_x = None

    # Generate multiple initial configurations
    configs = []

    # Hexagonal patterns with different row distributions
    configs.append(hex_init([7, 6, 7, 6], 0.1))
    configs.append(hex_init([6, 5, 6, 5, 4], 0.11))
    configs.append(hex_init([5, 6, 5, 6, 4], 0.11))
    configs.append(hex_init([4, 5, 6, 5, 6], 0.11))
    configs.append(hex_init([7, 5, 7, 5, 2], 0.12))
    configs.append(hex_init([5, 5, 5, 5, 6], 0.12))

    # Grid-based
    configs.append(grid_init(n))

    # Slightly perturbed hexagonal
    base = hex_init([7, 6, 7, 6], 0.1)
    for scale in [0.95, 1.05]:
        configs.append(base * scale + (1 - scale) * np.array([0.5, 0.5]))

    for ic in configs:
        if len(ic) != n:
            continue

        # Build variable vector: [r1,...,r26, x1,y1,...,x26,y26]
        x0 = np.zeros(n + 2 * n)
        x0[:n] = 0.07  # Start with small radii
        x0[n:] = ic.flatten()

        # Build constraints
        con = []
        for i in range(n):
            con.append({"type": "ineq", "fun": bc_left, "args": (i, n)})
            con.append({"type": "ineq", "fun": bc_right, "args": (i, n)})
            con.append({"type": "ineq", "fun": bc_bottom, "args": (i, n)})
            con.append({"type": "ineq", "fun": bc_top, "args": (i, n)})
        for i in range(n):
            for j in range(i + 1, n):
                con.append({"type": "ineq", "fun": non_overlap, "args": (i, j, n)})

        # Bounds
        bnd = [(0, 0.5)] * n + [(0, 1)] * (2 * n)

        try:
            res = minimize(
                obj_fun, x0, method="SLSQP", bounds=bnd, constraints=con,
                args=(n,), options={"maxiter": 15000, "ftol": 1e-16, "disp": False}
            )
            s = -res.fun
            if s > best_s:
                best_s = s
                best_x = res.x.copy()
        except Exception:
            pass

    if best_x is None:
        # Fallback: small circles in grid
        centers = np.tile(np.arange(5) * 0.2 + 0.1, (5, 1))
        centers = centers.reshape(25, 2)
        y_vals = np.tile(np.arange(5) * 0.2 + 0.1, (5, 1)).T
        centers[:, 1] = y_vals
        centers = np.vstack([centers, [[0.5, 0.5]]])
        radii = np.ones(26) * 0.05
        return centers, radii, np.sum(radii)

    radii = best_x[:n]
    centers = best_x[n:].reshape(n, 2)

    # Post-optimization: fix any tiny constraint violations
    for _ in range(50):
        max_viol = 0.0
        for i in range(n):
            r = radii[i]
            cx, cy = centers[i]
            for m in [cx - r, 1 - cx - r, cy - r, 1 - cy - r]:
                if m < -1e-12:
                    max_viol = max(max_viol, -m)
        for i in range(n):
            for j in range(i + 1, n):
                d = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                ov = radii[i] + radii[j] - d
                if ov > 1e-12:
                    max_viol = max(max_viol, ov)
        if max_viol < 1e-12:
            break
        # Shrink radii slightly to resolve violations
        radii *= 0.999

    return centers, radii, float(np.sum(radii))
