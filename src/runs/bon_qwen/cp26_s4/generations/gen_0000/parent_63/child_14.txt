2. **Optimization with SLSQP**: Use scipy's SLSQP method which handles inequality constraints natively. This is better than penalty methods because it maintains feasibility throughout optimization.

3. **Sparse constraints**: Only include overlap constraints for circle pairs that are initially close (distance < threshold), reducing from 325 to ~100 constraints.

4. **Multiple restarts**: Run optimization from several different initial configurations to escape local minima.

5. **Multi-stage refinement**: After initial SLSQP optimization, add all pair constraints and re-optimize to polish the solution.

6. **Smart initialization variations**: Try different grid spacings, perturbations, and even grid-like vs hexagonal patterns to find the best starting point.

Key insight: The sum of radii objective favors having some larger circles (corners/edges) and filling gaps with smaller ones, unlike the standard equal-circle packing problem.

import numpy as np
from scipy.optimize import minimize
from functools import partial


class CirclePackingOpt:
    def __init__(self, n, active_pairs=None):
        self.n = n
        self.active_pairs = active_pairs if active_pairs is not None else []

    def objective(self, x):
        radii = x[2 * self.n:]
        return -np.sum(radii)

    def get_centers(self, x):
        return x[:2 * self.n].reshape((self.n, 2))

    def get_radii(self, x):
        return x[2 * self.n:]

    def overlap_constraint(self, x, i, j):
        centers = self.get_centers(x)
        radii = self.get_radii(x)
        dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
        return dist - radii[i] - radii[j]

    def boundary_xmin(self, x, i):
        return x[2 * i] - x[2 * self.n + i]

    def boundary_xmax(self, x, i):
        return 1.0 - x[2 * i] - x[2 * self.n + i]

    def boundary_ymin(self, x, i):
        return x[2 * i + 1] - x[2 * self.n + i]

    def boundary_ymax(self, x, i):
        return 1.0 - x[2 * i + 1] - x[2 * self.n + i]

    def radius_nonneg(self, x, i):
        return x[2 * self.n + i]


def hexagonal_init(n, rows, start_r=0.09):
    centers = np.zeros((n, 2))
    radii = np.full(n, start_r)
    idx = 0
    for row in range(rows):
        if row % 2 == 0:
            ncols = (n + rows - 1) // rows
        else:
            ncols = n // rows
        # Adjust to get exactly n circles
        remaining = n - idx
        if row == rows - 1:
            ncols = remaining
        for col in range(min(ncols, remaining)):
            if idx >= n:
                break
            spacing_x = 1.0 / (ncols + 1)
            spacing_y = 1.0 / (rows + 1)
            x = spacing_x * (col + 1) + (row % 2) * spacing_x * 0.5
            y = spacing_y * (row + 1)
            # Clamp to valid range
            x = min(max(x, 0.05), 0.95)
            y = min(max(y, 0.05), 0.95)
            centers[idx] = [x, y]
            idx += 1
    return centers, radii


def grid_hex_init(n, cols, rows, start_r=0.09):
    centers = np.zeros((n, 2))
    radii = np.full(n, start_r)
    idx = 0
    for row in range(rows):
        shift = (row % 2) * 0.5 * (1.0 / (cols + 1))
        for col in range(cols):
            if idx >= n:
                break
            x = (col + 1) * (1.0 / (cols + 1)) + shift
            y = (row + 1) * (1.0 / (rows + 1))
            x = min(max(x, 0.05), 0.95)
            y = min(max(y, 0.05), 0.95)
            centers[idx] = [x, y]
            idx += 1
    return centers, radii


def smart_init(n):
    """Create a smart initial arrangement with larger circles near corners/edges"""
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    # Place 4 corner circles first (slightly larger)
    corner_r = 0.12
    corners = [(corner_r, corner_r), (1 - corner_r, corner_r),
               (corner_r, 1 - corner_r), (1 - corner_r, 1 - corner_r)]
    for i, (x, y) in enumerate(corners):
        centers[i] = [x, y]
        radii[i] = corner_r

    idx = 4
    # Place circles along edges
    edge_r = 0.095
    # Bottom edge
    for k in range(4):
        if idx >= n:
            break
        x = corner_r + (k + 1) * (1 - 2 * corner_r) / 5
        y = edge_r
        centers[idx] = [x, y]
        radii[idx] = edge_r
        idx += 1
    # Top edge
    for k in range(4):
        if idx >= n:
            break
        x = corner_r + (k + 1) * (1 - 2 * corner_r) / 5
        y = 1 - edge_r
        centers[idx] = [x, y]
        radii[idx] = edge_r
        idx += 1
    # Left edge
    for k in range(3):
        if idx >= n:
            break
        x = edge_r
        y = corner_r + (k + 1) * (1 - 2 * corner_r) / 4
        centers[idx] = [x, y]
        radii[idx] = edge_r
        idx += 1
    # Right edge
    for k in range(3):
        if idx >= n:
            break
        x = 1 - edge_r
        y = corner_r + (k + 1) * (1 - 2 * corner_r) / 4
        centers[idx] = [x, y]
        radii[idx] = edge_r
        idx += 1

    # Fill interior with hexagonal pattern
    interior_r = 0.07
    if idx < n:
        for row in range(6):
            for col in range(6):
                if idx >= n:
                    break
                x = 0.2 + col * 0.12 + (row % 2) * 0.06
                y = 0.2 + row * 0.13
                if x < 0.9 and y < 0.9:
                    centers[idx] = [x, y]
                    radii[idx] = interior_r
                    idx += 1
            if idx >= n:
                break

    return centers, radii


def build_constraints(packer):
    constraints = []
    for i, j in packer.active_pairs:
        constraints.append({'type': 'ineq', 'fun': partial(packer.overlap_constraint, i=i, j=j)})
    for i in range(packer.n):
        constraints.append({'type': 'ineq', 'fun': partial(packer.boundary_xmin, i=i)})
        constraints.append({'type': 'ineq', 'fun': partial(packer.boundary_xmax, i=i)})
        constraints.append({'type': 'ineq', 'fun': partial(packer.boundary_ymin, i=i)})
        constraints.append({'type': 'ineq', 'fun': partial(packer.boundary_ymax, i=i)})
        constraints.append({'type': 'ineq', 'fun': partial(packer.radius_nonneg, i=i)})
    return constraints


def get_active_pairs(centers, threshold=0.35):
    pairs = []
    n = centers.shape[0]
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < threshold:
                pairs.append((i, j))
    return pairs


def optimize_from_init(centers, radii, threshold=0.35):
    n = centers.shape[0]
    active_pairs = get_active_pairs(centers, threshold)
    packer = CirclePackingOpt(n, active_pairs)

    x0 = np.concatenate([centers.flatten(), radii])
    bounds = [(0, 1)] * (2 * n) + [(1e-6, 0.5)] * n
    constraints = build_constraints(packer)

    try:
        result = minimize(packer.objective, x0, method='SLSQP', bounds=bounds,
                          constraints=constraints,
                          options={'maxiter': 3000, 'ftol': 1e-14, 'disp': False})
        return result
    except Exception:
        return None


def optimize_full(centers, radii):
    """Optimize with all pair constraints for final polish"""
    n = centers.shape[0]
    active_pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
    packer = CirclePackingOpt(n, active_pairs)

    x0 = np.concatenate([centers.flatten(), radii])
    bounds = [(0, 1)] * (2 * n) + [(1e-6, 0.5)] * n
    constraints = build_constraints(packer)

    try:
        result = minimize(packer.objective, x0, method='SLSQP', bounds=bounds,
                          constraints=constraints,
                          options={'maxiter': 2000, 'ftol': 1e-14, 'disp': False})
        return result
    except Exception:
        return None


def perturb_solution(centers, radii, scale=0.02):
    n = centers.shape[0]
    new_centers = centers.copy()
    new_radii = radii.copy()

    for i in range(n):
        dx = np.random.uniform(-scale, scale)
        dy = np.random.uniform(-scale, scale)
        new_centers[i, 0] = min(max(new_centers[i, 0] + dx, new_radii[i]), 1 - new_radii[i])
        new_centers[i, 1] = min(max(new_centers[i, 1] + dy, new_radii[i]), 1 - new_radii[i])
        new_radii[i] = max(new_radii[i] + np.random.uniform(-scale * 0.1, scale * 0.1), 1e-6)

    return new_centers, new_radii
