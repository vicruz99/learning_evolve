import numpy as np
from scipy.optimize import minimize

def objective(p):
    """Objective function: maximize sum of radii (minimize negative sum)."""
    return -np.sum(p[2::3])

def constraint_func(p):
    """
    Computes non-overlap constraints: dist^2 - (r1+r2)^2 >= 0
    Returns a 1D numpy array of constraint values for all i < j.
    """
    x = p[0::3]
    y = p[1::3]
    r = p[2::3]
    
    # Get indices for all pairs (i, j) with i < j
    i_idx, j_idx = np.triu_indices(26, k=1)
    
    dx = x[i_idx] - x[j_idx]
    dy = y[i_idx] - y[j_idx]
    dr = r[i_idx] + r[j_idx]
    
    return dx**2 + dy**2 - dr**2

def get_bounds():
    """Defines box bounds for x, y, and r variables."""
    bounds = []
    # x coordinates in [0, 1]
    for _ in range(26):
        bounds.append((0.0, 1.0))
    # y coordinates in [0, 1]
    for _ in range(26):
        bounds.append((0.0, 1.0))
    # radii in [1e-5, 0.5]
    for _ in range(26):
        bounds.append((1e-5, 0.5))
    return bounds

def init_grid(rng):
    """Creates an initial configuration based on a perturbed 5x5 grid + 1 center."""
    p0 = np.zeros(78)
    coords = []
    for i in range(5):
        for j in range(5):
            coords.append((0.1 + i * 0.2, 0.1 + j * 0.2))
    coords.append((0.5, 0.5))
    
    for idx, (cx, cy) in enumerate(coords):
        p0[3 * idx] = cx
        p0[3 * idx + 1] = cy
        p0[3 * idx + 2] = 0.05
        
    # Add noise to break symmetry and help optimizer escape saddle points
    p0 += rng.normal(0, 0.008, 78)
    
    # Clip to safe initial bounds
    p0[:26] = np.clip(p0[:26], 0.05, 0.95)
    p0[26:52] = np.clip(p0[26:52], 0.05, 0.95)
    p0[52:] = np.clip(p0[52:], 0.03, 0.09)
    return p0

def init_random(rng):
    """Creates a random initial configuration."""
    p0 = np.zeros(78)
    p0[:26] = rng.uniform(0.1, 0.9, 26)
    p0[26:52] = rng.uniform(0.1, 0.9, 26)
    p0[52:] = rng.uniform(0.04, 0.08, 26)
    return p0

def run_packing():
    """Runs the optimization to pack 26 circles in a unit square."""
    rng = np.random.default_rng(42)
    bounds = get_bounds()
    con = {'type': 'ineq', 'fun': constraint_func}
    opts = {'maxiter': 4000, 'ftol': 1e-12, 'disp': False}
    
    best_sum = 0.0
    best_p = None
    
    # Prepare multiple starting points to avoid local minima
    starts = [init_grid(rng)]
    for _ in range(8):
        starts.append(init_random(rng))
        
    for p0 in starts:
        res = minimize(objective, p0, method='SLSQP', bounds=bounds, 
                       constraints=con, options=opts)
        
        # Accept result even if not strictly successful, as it often still improves
        p = res.x
        current_sum = np.sum(p[52:])
        
        if current_sum > best_sum:
            best_sum = current_sum
            best_p = p.copy()
            
    # Final safety adjustments to guarantee validity against numerical drift
    r = best_p[52:]
    x = best_p[:26]
    y = best_p[26:52]
    
    for i in range(26):
        # Enforce boundary constraints strictly
        if x[i] - r[i] < 0.0:
            best_p[3 * i] = r[i]
        elif x[i] + r[i] > 1.0:
            best_p[3 * i] = 1.0 - r[i]
            
        if y[i] - r[i] < 0.0:
            best_p[3 * i + 1] = r[i]
        elif y[i] + r[i] > 1.0:
            best_p[3 * i + 1] = 1.0 - r[i]
            
    centers = np.column_stack((best_p[:26], best_p[26:52]))
    radii = best_p[52:]
    return centers, radii, np.sum(radii)
