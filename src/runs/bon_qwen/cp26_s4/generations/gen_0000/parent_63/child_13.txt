import numpy as np

def generate_initial_positions(n):
    """Generate initial positions on a hexagonal-like grid."""
    positions = []
    # Row pattern to sum to 26: 5 + 4 + 5 + 4 + 5 + 3 = 26
    row_counts = [5, 4, 5, 4, 5, 3]
    y = 0.0
    for idx, count in enumerate(row_counts):
        # Shift every other row by 0.5 to create hexagonal packing
        x_start = 0.5 if idx % 2 == 1 else 0.0
        for c in range(count):
            positions.append([x_start + c, y])
        y += 1.0
        
    pos_arr = np.array(positions)
    
    # Scale and center the grid inside [0,1] with margin
    min_vals = pos_arr.min(axis=0)
    max_vals = pos_arr.max(axis=0)
    dims = max_vals - min_vals
    scale = 0.8 / np.max(dims)
    
    pos_arr = (pos_arr - min_vals) * scale
    pos_arr += 0.1  # Center roughly in the square
    
    # Add tiny noise to break exact symmetries
    pos_arr += np.random.uniform(-0.001, 0.001, pos_arr.shape)
    return pos_arr

        violations = np.maximum(0.0, 2.0 * r - dists)
