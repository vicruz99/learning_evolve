import numpy as np
from scipy.optimize import minimize

def compute_loss_wrapper(centers_flat, r):
    """
    Computes the sum of squared violations for boundary and overlap constraints.
    """
    centers = centers_flat.reshape(-1, 2)
    x, y = centers[:, 0], centers[:, 1]
    
    # Boundary violations
    loss = 0.0
    loss += np.sum(np.maximum(0.0, r - x)**2)
    loss += np.sum(np.maximum(0.0, x - (1.0 - r))**2)
    loss += np.sum(np.maximum(0.0, r - y)**2)
    loss += np.sum(np.maximum(0.0, y - (1.0 - r))**2)
    
    # Pairwise overlap violations
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dist, np.inf)
    violations = np.maximum(0.0, 2.0 * r - dist)
    loss += np.sum(violations**2)
    
    return loss

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    n = 26
    
    # Initialize centers on a 5x6 grid with jitter
    centers = np.zeros((n, 2))
    idx = 0
    for i in range(5):
        for j in range(6):
            if idx < n:
                centers[idx] = [j * 0.2, i * 0.2]
                idx += 1
    centers += np.random.uniform(-0.05, 0.05, size=(n, 2))
    centers = np.clip(centers, 0.05, 0.95)
    
    r_low, r_high = 0.05, 0.11
    best_r = r_low
    best_centers = centers.copy()
    
    # Binary search for the maximum feasible radius
    for _ in range(20):
        r_mid = (r_low + r_high) / 2.0
        res = minimize(compute_loss_wrapper,
                       best_centers.flatten(),
                       args=(r_mid,),
                       method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * (2 * n),
                       options={'maxiter': 5000, 'ftol': 1e-12})
        
        if res.fun < 1e-6:
            best_r = r_mid
            best_centers = res.x.reshape(-1, 2)
            r_low = r_mid
        else:
            r_high = r_mid
            
    return best_centers, np.full(n, best_r), best_r * n
