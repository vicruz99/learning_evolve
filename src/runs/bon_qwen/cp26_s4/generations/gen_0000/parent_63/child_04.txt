def get_initial_centers(n=26):

    centers = []

    counts = [6, 5, 6, 5, 4]

    dx = 0.18

        x = 0.12 + (0.09 if i % 2 == 1 else 0)

            centers.append([x, y])
            x += dx
        y += dy
        
    return np.array(centers[:n])
