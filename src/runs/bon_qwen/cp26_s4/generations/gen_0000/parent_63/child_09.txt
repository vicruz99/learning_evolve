import numpy as np
from scipy.optimize import minimize

def compute_objective(v, mu, n):
    """
    Computes the penalized objective: -sum(radii) + mu * sum(violations^2)
    Violations include boundary crossings and circle overlaps.
    """
    centers = v[:2 * n].reshape(n, 2)
    radii = v[2 * n:]
    
    # Boundary violations: circle must be within [0,1]x[0,1]
    p = np.maximum(0.0, radii - centers[:, 0])
    p += np.maximum(0.0, radii - (1.0 - centers[:, 0]))
    p += np.maximum(0.0, radii - centers[:, 1])
    p += np.maximum(0.0, radii - (1.0 - centers[:, 1]))
    bound_pen = np.sum(p ** 2)
    
    # Overlap violations: distance between centers >= r_i + r_j
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff ** 2, axis=2))
    np.fill_diagonal(dists, np.inf)  # Ignore self-distances
    
    overlap = np.maximum(0.0, radii[:, np.newaxis] + radii[np.newaxis, :] - dists)
    overlap_pen = np.sum(overlap ** 2)
    
    return -np.sum(radii) + mu * (bound_pen + overlap_pen)

def run_packing() -> tuple:

    n = 26

    row = 0

            pts.append([x, y])

        row += 1
        
    centers = np.array(pts[:n])

    centers = (centers - 0.5) * 0.8 + 0.5
    # Add small random perturbation to break symmetry
    centers += np.random.uniform(-0.005, 0.005, centers.shape)
    
    radii = np.full(n, 0.08)  # Feasible initial radius
    
    x0 = np.concatenate([centers.flatten(), radii])
    bounds = [(0.0, 1.0) for _ in range(2 * n)] + [(0.0, 0.5) for _ in range(n)]

    current_x = x0
    for mu in [10.0, 100.0, 1000.0, 10000.0]:
        res = minimize(
            compute_objective,
            current_x,
            args=(mu, n),
            method='L-BFGS-B',

    final_radii = np.maximum(final_radii, 0.0)
    
    total_sum = float(np.sum(final_radii))
    return final_centers, final_radii, total_sum
