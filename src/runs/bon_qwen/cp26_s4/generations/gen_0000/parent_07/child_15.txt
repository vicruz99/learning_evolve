I'll use a two-phase approach:

**Phase 1: Optimize center positions using simulated annealing**
- Start with hexagonal packing as initial configuration
- Randomly perturb circle centers and accept/reject based on improvement
- The key insight: for any fixed center positions, the optimal radii can be found via Linear Programming

**Phase 2: Solve LP for radii given fixed centers**
- For fixed centers, maximizing sum of radii subject to:
  - r_i ≤ min(x_i, 1-x_i, y_i, 1-y_i) [boundary constraints]
  - r_i + r_j ≤ ||c_i - c_j|| [non-overlap constraints]
  - r_i ≥ 0
- This is a standard LP problem solved efficiently with scipy.optimize.linprog

**Phase 3: Iterative refinement**
- Use gradient-based local optimization (SLSQP) on the full problem as a final polish
- The SLSQP phase uses explicit constraint classes to avoid closures/lambdas

This approach leverages the fact that radii optimization for fixed centers is convex, while center optimization benefits from the annealing heuristic.

```python
import numpy as np
from scipy.optimize import linprog, minimize

N_CIRCLES = 26


def objective(x):
    """Maximize sum of radii (minimize negative sum)"""
    total = 0.0
    for i in range(N_CIRCLES):
        total += x[3 * i + 2]
    return -total


class NonOverlapConstraint:
    """Distance between circles i and j >= r_i + r_j"""
    def __init__(self, i, j):
        self.i = i
        self.j = j

    def __call__(self, x):
        xi = x[3 * self.i]
        yi = x[3 * self.i + 1]
        ri = x[3 * self.i + 2]
        xj = x[3 * self.j]
        yj = x[3 * self.j + 1]
        rj = x[3 * self.j + 2]
        dist = np.sqrt((xi - xj) ** 2 + (yi - yj) ** 2)
        return dist - ri - rj


class LeftBoundaryConstraint:
    """x_i - r_i >= 0"""
    def __init__(self, i):
        self.i = i

    def __call__(self, x):
        return x[3 * self.i] - x[3 * self.i + 2]


class RightBoundaryConstraint:
    """1 - x_i - r_i >= 0"""
    def __init__(self, i):
        self.i = i

    def __call__(self, x):
        return 1.0 - x[3 * self.i] - x[3 * self.i + 2]


class BottomBoundaryConstraint:
    """y_i - r_i >= 0"""
    def __init__(self, i):
        self.i = i

    def __call__(self, x):
        return x[3 * self.i + 1] - x[3 * self.i + 2]


class TopBoundaryConstraint:
    """1 - y_i - r_i >= 0"""
    def __init__(self, i):
        self.i = i

    def __call__(self, x):
        return 1.0 - x[3 * self.i + 1] - x[3 * self.i + 2]


class RadiusNonnegConstraint:
    """r_i >= 0"""
    def __init__(self, i):
        self.i = i

    def __call__(self, x):
        return x[3 * self.i + 2]


def solve_radii_lp(centers):
    """Given fixed centers, solve LP to find optimal radii maximizing sum"""
    n = centers.shape[0]

    # Objective: minimize -sum(r_i)
    c = -np.ones(n)

    # Inequality constraints: A_ub @ r <= b_ub
    A_ub = []
    b_ub = []

    # Boundary constraints: r_i <= min(x_i, 1-x_i, y_i, 1-y_i)
    for i in range(n):
        max_r = min(centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])
        max_r = max(max_r, 1e-10)  # Ensure positive upper bound
        row = np.zeros(n)
        row[i] = 1.0
        A_ub.append(row)
        b_ub.append(max_r)

    # Non-overlap constraints: r_i + r_j <= dist(i, j)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dist)

    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)

    # Bounds: 0 <= r_i
    bounds = [(0, None)] * n

    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')

    if result.success:
        return result.x
    else:
        # Fallback: small equal radii
        return np.full(n, 0.01)


def evaluate_centers(centers):
    """Evaluate the sum of radii for given centers using LP"""
    radii = solve_radii_lp(centers)
    return np.sum(radii), radii


def create_hexagonal_initial(n, r=0.09):
    """Create initial hexagonal packing configuration"""
    centers = np.zeros((n, 2))
    radii = np.full(n, r)

    idx = 0
    for row in range(20):
        for col in range(20):
            if idx >= n:
                return centers, radii

            x = r + col * 2 * r
            y = r + row * r * np.sqrt(3)
            if row % 2 == 1:
                x += r

            if x + r <= 1.0 + 1e-10 and y + r <= 1.0 + 1e-10:
                centers[idx] = [x, y]
                radii[idx] = r
                idx += 1

    return centers, radii


def create_staggered_initial(n):
    """Create staggered initial configuration with better coverage"""
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    idx = 0
    # Try multiple row patterns
    for row in range(20):
        for col in range(20):
            if idx >= n:
                return centers, radii

            # Adaptive spacing based on available space
            n_per_row_approx = int(np.sqrt(n))
            spacing_x = 1.0 / (n_per_row_approx + 1)
            spacing_y = 1.0 / (n_per_row_approx + 1)

            x = spacing_x + col * 2 * spacing_x
            y = spacing_y + row * spacing_y * np.sqrt(3) / 2
            if row % 2 == 1:
                x += spacing_x

            if x <= 1.0 and y <= 1.0:
                centers[idx] = [x, y]
                idx += 1

    # Solve for radii
    radii = solve_radii_lp(centers)
    return centers, radii


def create_random_initial(n):
    """Create random initial configuration"""
    centers = np.random.uniform(0.1, 0.9, (n, 2))
    radii = solve_radii_lp(centers)
    return centers, radii


def simulated_annealing_optimize(centers_init, n_iterations=50000, temp_initial=0.1, temp_final=1e-6):
    """Optimize center positions using simulated annealing with LP-based evaluation"""
    n = centers_init.shape[0]

    centers = centers_init.copy()
    best_sum, best_radii = evaluate_centers(centers)
    best_centers = centers.copy()

    current_sum = best_sum

    temp = temp_initial
    decay = (temp_final / temp_initial) ** (1.0 / n_iterations)

    for iteration in range(n_iterations):
        # Pick random circle to perturb
        i = np.random.randint(n)

        # Perturbation scale decreases with temperature
        scale = 0.02 * temp

        new_centers = centers.copy()
        new_centers[i, 0] += np.random.normal(0, scale)
        new_centers[i, 1] += np.random.normal(0, scale)

        # Keep within bounds
        new_centers[i, 0] = np.clip(new_centers[i, 0], 0.001, 0.999)
        new_centers[i, 1] = np.clip(new_centers[i, 1], 0.001, 0.999)

        new_sum, new_radii = evaluate_centers(new_centers)

        # Accept or reject
        delta = new_sum - current_sum
        if delta > 0 or np.random.random() < np.exp(delta / max(temp, 1e-10)):
            centers = new_centers
            current_sum = new_sum
            if new_sum > best_sum:
                best_sum = new_sum
                best_centers = centers.copy()
                best_radii = new_radii.copy()

        temp *= decay

    return best_centers, best_radii, best_sum


def build_constraints(n):
    """Build constraint list for SLSQP optimization"""
    constraints = []

    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({'type': 'ineq', 'fun': NonOverlapConstraint(i, j)})
        constraints.append({'type': 'ineq', 'fun': LeftBoundaryConstraint(i)})
        constraints.append({'type': 'ineq', 'fun': RightBoundaryConstraint(i)})
        constraints.append({'type': 'ineq', 'fun': BottomBoundaryConstraint(i)})
        constraints.append({'type': 'ineq', 'fun': TopBoundaryConstraint(i)})
        constraints.append({'type': 'ineq', 'fun': RadiusNonnegConstraint(i)})

    return constraints


def run_packing():
    n = N_CIRCLES

    best_sum = -1
    best_centers = None
    best_radii = None

    # Strategy 1: Simulated annealing from hexagonal initial
    print("Starting simulated annealing from hexagonal initial...")
    centers_init, radii_init = create_hexagonal_initial(n, 0.095)
    centers, radii, s = simulated_annealing_optimize(centers_init, n_iterations=80000, temp_initial=0.08, temp_final=1e-8)
    if s > best_sum:
        best_sum = s
        best_centers = centers.copy()
        best_radii = radii.copy()
        print(f"SA hexagonal: sum = {s:.6f}")

    # Strategy 2: Simulated annealing from staggered initial
    print("Starting simulated annealing from staggered initial...")
    centers_init, radii_init = create_staggered_initial(n)
    centers, radii, s = simulated_annealing_optimize(centers_init, n_iterations=80000, temp_initial=0.06, temp_final=1e-8)
    if s > best_sum:
        best_sum = s
        best_centers = centers.copy()
        best_radii = radii.copy()
        print(f"SA staggered: sum = {s:.6f}")

    # Strategy 3: Simulated annealing from random initial (multiple restarts)
    print("Starting simulated annealing from random initial...")
    for restart in range(5):
        centers_init, radii_init = create_random_initial(n)
        centers, radii, s = simulated_annealing_optimize(centers_init, n_iterations=60000, temp_initial=0.05, temp_final=1e-8)
        if s > best_sum:
            best_sum = s
            best_centers = centers.copy()
            best_radii = radii.copy()
            print(f"SA random restart {restart}: sum = {s:.6f}")

    # Strategy 4: SLSQP refinement from best SA result
    print("Starting SLSQP refinement...")
    if best_centers is not None:
        x0 = np.zeros(3 * n)
        for i in range(n):
            x0[3 * i] = best_centers[i, 0]
            x0[3 * i + 1] = best_centers[i, 1]
            x0[3 * i + 2] = best_radii[i]

        constraints = build_constraints(n)
        bounds = [(0.0001, 0.9999)] * (2 * n) + [(1e-8, 0.5)] * n

        result = minimize(
            objective, x0, method='SLSQP', bounds=bounds,
            constraints=constraints,
            options={'maxiter': 5000, 'ftol': 1e-15, 'disp': False}
        )

        s = -result.fun
        if s > best_sum:
            best_sum = s
            best_centers = np.zeros((n, 2))
            best_radii = np.zeros(n)
            for i in range(n):
                best_centers[i] = [result.x[3 * i], result.x[3 * i + 1]]
                best_radii[i] = result.x[3 * i + 2]
            print(f"SLSQP refined: sum = {s:.6f}")

    # Strategy 5: Further annealing refinement
    print("Starting final annealing refinement...")
    centers, radii, s = simulated_annealing_optimize(
        best_centers, n_iterations=100000, temp_initial=0.03, temp_final=1e-10
    )
    if s > best_sum:
        best_sum = s
        best_centers = centers.copy()
        best_radii = radii.copy()
        print(f"Final SA: sum = {s:.6f}")

    return best_centers, best_radii, best_sum
```
