import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Pairs 26 circles in a unit square [0,1]x[0,1] to maximize sum of radii.
    
    Strategy:
    1. Initialize centers in a hexagonal lattice pattern.
    2. Use scipy.optimize.minimize with L-BFGS-B to maximize sum of radii.
    3. Objective function includes penalties for boundary violations and circle overlaps.
    """
    n = 26
    
    # --- Initialization ---
    # Hexagonal lattice generation
    centers_list = []
    x_step = 0.21 # Slightly larger than 0.2 to ensure initial separation
    y_step = x_step * np.sqrt(3) / 2
    
    y_current = 0.12 
    row_idx = 0
    
    # Generate points
    while y_current <= 0.90:
        if row_idx % 2 == 0:
            x_start = 0.12
            x_end = 0.88
        else:
            x_start = 0.23
            x_end = 0.77
        
        x_current = x_start
        while x_current <= x_end:
            centers_list.append([x_current, y_current])
            x_current += x_step
        y_current += y_step
        row_idx += 1
        
    # Ensure we have exactly 26 points
    if len(centers_list) > n:
        centers_list = centers_list[:n]
    else:
        # Fill remaining with random perturbations around center if needed (unlikely)
        while len(centers_list) < n:
            centers_list.append([0.5, 0.5]) 
            
    centers = np.array(centers_list)
    radii = np.ones(n) * 0.09 # Start with radius 0.09
    
    # Flatten variables: [x1, y1, r1, x2, y2, r2, ...]
    x0 = np.zeros(3 * n)
    for i in range(n):
        x0[3*i] = centers[i, 0]
        x0[3*i+1] = centers[i, 1]
        x0[3*i+2] = radii[i]
        
    # Bounds for optimizer
    # x, y in [0, 1], r in [0, 0.5]
    bnds = [(0, 1)] * n + [(0, 1)] * n + [(0, 0.5)] * n
    
    # Objective function: Maximize sum(radii) => Minimize -sum(radii)
    # With penalty for constraints
    def obj(v):
        val = 0.0
        penalty = 0.0
        
        # Reconstruct centers and radii
        # x indices: 0, 3, 6...
        # y indices: 1, 4, 7...
        # r indices: 2, 5, 8...
        
        # Extract arrays for speed
        xs = v[0::3]
        ys = v[1::3]
        rs = v[2::3]
        
        # Sum of radii (we want to maximize, so negative in minimization)
        val = -np.sum(rs)
        
        # Boundary penalties
        # r <= x <= 1-r  =>  x-r >= 0  and  x+r <= 1
        # r <= y <= 1-r  =>  y-r >= 0  and  y+r <= 1
        # Violation amount
        viol_x_left = rs - xs
        viol_y_left = rs - ys
        viol_x_right = xs + rs - 1.0
        viol_y_right = ys + rs - 1.0
        
        # Sum of squared violations (only positive ones matter)
        # Using maximum with 0 is not differentiable, but smooth enough or we can assume small violations
        # For L-BFGS-B, we can use smooth approximation or just rely on bounds + penalty.
        # However, simple max(0, x)^2 is C1 continuous.
        
        # Vectorized penalty calculation
        penalty += np.sum(np.maximum(viol_x_left, 0)**2)
        penalty += np.sum(np.maximum(viol_y_left, 0)**2)
        penalty += np.sum(np.maximum(viol_x_right, 0)**2)
        penalty += np.maximum(viol_y_right, 0)**2
        
        # Overlap penalties
        # dist(i,j) >= r_i + r_j
        # dist^2 >= (r_i + r_j)^2
        # Violation: r_i + r_j - dist
        
        # Compute pairwise distances
        # O(N^2) loop is fine for N=26
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt((xs[i] - xs[j])**2 + (ys[i] - ys[j])**2)
                sum_r = rs[i] + rs[j]
                viol = sum_r - dist
                if viol > 0:
                    penalty += viol**2
                    
        return val + 2000 * penalty # Weight 2000 for constraints

    # Run optimization
    # L-BFGS-B is good for bound-constrained problems
    res = minimize(obj, x0, method='L-BFGS-B', bounds=bnds, 
                   options={'maxiter': 5000, 'ftol': 1e-15, 'gtol': 1e-15})
    
    # Extract results
    final_xs = res.x[0::3]
    final_ys = res.x[1::3]
    final_rs = res.x[2::3]
    
    final_centers = np.column_stack((final_xs, final_ys))
    final_radii = final_rs
    
    # Clean up tiny violations due to numerical precision
    # Enforce strict boundary constraints
    for i in range(n):
        x, y = final_centers[i]
        r = final_radii[i]
        
        # Ensure inside [0,1] with radius clearance
        # If x-r < 0, set x=r (and potentially reduce r? No, x can move)
        # But optimizer should have handled this.
        # Just clamp to valid range for safety
        if x - r < 1e-10:
            final_centers[i, 0] = r + 1e-10
        if x + r > 1.0 - 1e-10:
            final_centers[i, 0] = 1.0 - r - 1e-10
        if y - r < 1e-10:
            final_centers[i, 1] = r + 1e-10
        if y + r > 1.0 - 1e-10:
            final_centers[i, 1] = 1.0 - r - 1e-10
            
    sum_radii = np.sum(final_radii)
    
    return final_centers, final_radii, float(sum_radii)
