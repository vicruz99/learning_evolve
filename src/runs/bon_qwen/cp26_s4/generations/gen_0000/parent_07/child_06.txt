import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    """
    n_circles = 26
    
    # --- 1. Initialization Strategy (Hexagonal Grid) ---
    # Generate a hexagonal lattice of points
    initial_points = []
    row_height = np.sqrt(3) / 2
    # Spacing estimate: to fit 26 circles, we might need ~5 rows with ~5 circles each.
    # Spacing approx 1/6 = 0.16.
    spacing = 0.16
    
    # Generate enough points to pick from
    max_y = 1.0
    while spacing * np.floor(max_y / (spacing * row_height)) > 0:
        for row_idx in range(15): # Arbitrary limit to ensure we get enough points
            y = 0.05 + row_idx * spacing * row_height
            if y > 0.95: break
            
            # Determine x-start based on row parity for hex packing
            x_offset = 0.05 if row_idx % 2 == 0 else 0.05 + spacing / 2
            
            for col_idx in range(8): # Arbitrary limit
                x = x_offset + col_idx * spacing
                if x > 0.95: break
                
                # Check if a small circle fits (strictly inside)
                if 0.01 <= x <= 0.99 and 0.01 <= y <= 0.99:
                    initial_points.append([x, y])
        
        # If we found enough, break outer loop
        if len(initial_points) >= n_circles + 10:
            break
            
    # If we need more points, just use a fallback grid
    if len(initial_points) < n_circles:
        for r in range(7):
            for c in range(7):
                x = 0.1 + c * 0.13
                y = 0.1 + r * 0.13
                if 0.05 <= x <= 0.95 and 0.05 <= y <= 0.95:
                    initial_points.append([x, y])

    # Shuffle and pick n_circles
    np.random.shuffle(initial_points)
    selected_centers = np.array(initial_points[:n_circles])
    initial_radii = np.full(n_circles, 0.05) # Start with small valid radii
    
    # Flatten for optimization: [x0, y0, r0, x1, y1, r1, ...]
    # Or better: [x_all, y_all, r_all] for easier constraint handling logic
    x0 = selected_centers[:, 0]
    y0 = selected_centers[:, 1]
    r0 = initial_radii
    x_init = np.concatenate([x0, y0, r0])

    # --- 2. Optimization ---
    
    # Objective: Minimize negative sum of radii (Maximize sum of radii)
    def objective(x):
        r = x[2::3]
        return -np.sum(r)

    # Gradient of objective
    def objective_grad(x):
        grad = np.zeros_like(x)
        grad[2::3] = -1.0
        return grad

    # Constraints
    constraints = []

    # 1. Boundary Constraints: r <= x, r <= 1-x, r <= y, r <= 1-y
    # We can add these as simple bounds or explicit constraints.
    # Explicit constraints are safer for SLSQP with varying r.
    # Constraint: x - r >= 0
    def bound_x_low(x):
        return x[0::3] - x[2::3]
    constraints.append({'type': 'ineq', 'fun': bound_x_low})

    # Constraint: 1 - x - r >= 0
    def bound_x_high(x):
        return 1.0 - x[0::3] - x[2::3]
    constraints.append({'type': 'ineq', 'fun': bound_x_high})

    # Constraint: y - r >= 0
    def bound_y_low(x):
        return x[1::3] - x[2::3]
    constraints.append({'type': 'ineq', 'fun': bound_y_low})

    # Constraint: 1 - y - r >= 0
    def bound_y_high(x):
        return 1.0 - x[1::3] - x[2::3]
    constraints.append({'type': 'ineq', 'fun': bound_y_high})

    # 2. Non-overlap Constraints: dist^2 >= (r_i + r_j)^2
    # dist^2 - (r_i + r_j)^2 >= 0
    # We only need to add these for pairs. 
    # To keep it efficient, we rely on SLSQP's handling.
    # There are 325 pairs.
    for i in range(n_circles):
        for j in range(i + 1, n_circles):
            idx_i_x = i * 3
            idx_j_x = j * 3
            
            def make_constraint(i, j):
                def con(x):
                    # x[i] is at 3*i, y at 3*i+1, r at 3*i+2
                    xi, yi, ri = x[3*i], x[3*i+1], x[3*i+2]
                    xj, yj, rj = x[3*j], x[3*j+1], x[3*j+2]
                    
                    dist_sq = (xi - xj)**2 + (yi - yj)**2
                    sum_r_sq = (ri + rj)**2
                    return dist_sq - sum_r_sq
                return con
            
            constraints.append({'type': 'ineq', 'fun': make_constraint(i, j)})

    # Bounds for variables
    # x, y in [0, 1], r in [0, 0.5]
    bnds = []
    for _ in range(n_circles):
        bnds.append((0.0, 1.0)) # x
        bnds.append((0.0, 1.0)) # y
        bnds.append((0.0, 0.5)) # r

    # Run Optimization
    result = minimize(objective, x_init, method='SLSQP', jac=objective_grad, 
                      bounds=bnds, constraints=constraints, 
                      options={'maxiter': 1000, 'ftol': 1e-9})

    # Extract results
    final_x = result.x[0::3]
    final_y = result.x[1::3]
    final_r = result.x[2::3]
    
    centers = np.column_stack((final_x, final_y))
    radii = final_r
    total_sum = np.sum(radii)

    # --- 3. Post-Processing / Cleanup ---
    # Ensure no negative radii (due to numerical noise)
    radii = np.maximum(radii, 1e-9)
    
    # If optimization failed to find valid packing (rare with good init), 
    # fallback to just returning the initial (though sum will be low).
    if not result.success and result.fun > 0: # fun is -sum, so >0 means sum < 0? No.
        pass 

    return centers, radii, total_sum
