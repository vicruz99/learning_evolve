import numpy as np
from scipy.optimize import minimize

def calculate_penalty_and_objective(params, weights):
    """
    Calculates the loss function for optimization.
    Loss = -Sum of Radii + Weight * (Overlap Penalty + Boundary Penalty)
    
    Args:
        params: Flattened array of [x1, y1, r1, x2, y2, r2, ...]
        weights: Multiplier for the penalty term
    Returns:
        Loss value, and individual penalty components for debugging
    """
    n = 26
    centers = params[:n*2].reshape(n, 2)
    radii = params[n*2:]
    
    overlap_penalty = 0.0
    boundary_penalty = 0.0
    
    # Boundary Penalty
    # Circle must be within [r, 1-r] for both x and y
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        # Check lower bounds (x >= r, y >= r)
        if x < r:
            boundary_penalty += (r - x)**2
        if y < r:
            boundary_penalty += (r - y)**2
        # Check upper bounds (x <= 1-r, y <= 1-r)
        if x > 1 - r:
            boundary_penalty += (x - (1 - r))**2
        if y > 1 - r:
            boundary_penalty += (y - (1 - r))**2

    # Overlap Penalty
    # Distance between centers must be >= r_i + r_j
    # Vectorized calculation for pairs
    # diff_centers = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    # dists = np.linalg.norm(diff_centers, axis=2)
    # radii_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    # diff = radii_sum - dists
    # mask = diff > 0
    # overlap_penalty = np.sum(np.where(mask, diff**2, 0))
    
    # Loop for clarity and lower memory overhead for N=26
    for i in range(n):
        xi, yi = centers[i]
        ri = radii[i]
        for j in range(i + 1, n):
            xj, yj = centers[j]
            rj = radii[j]
            
            dist = np.sqrt((xi - xj)**2 + (yi - yj)**2)
            min_dist = ri + rj
            
            if dist < min_dist:
                overlap_penalty += (min_dist - dist)**2

    objective = -np.sum(radii) + weights * (overlap_penalty + boundary_penalty)
    return objective, overlap_penalty, boundary_penalty
