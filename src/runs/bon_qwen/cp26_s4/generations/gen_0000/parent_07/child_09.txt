import numpy as np
from scipy.optimize import minimize
import math

def run_packing():
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    Strategy: Minimize the side length S of a square containing 26 unit circles,
    then scale the result to the unit square.
    """
    n = 26
    
    # 1. Initialization: Hexagonal packing of 26 unit circles
    # We estimate the required side length S. 
    # Area of 26 unit circles is 26*pi approx 81.
    # Square area S^2. Density approx 0.9. S^2 approx 90 => S approx 9.5.
    # We start with a safe S_init = 12.0.
    S_init = 12.0
    
    # Generate centers for a hexagonal pattern
    # Rows configuration to sum to 26: 6, 5, 6, 5, 4
    rows_config = [6, 5, 6, 5, 4]
    
    # Box for unit circles is [1, S-1]. Width is S-2.
    # With S=12, box is [1, 11], width 10.
    # Unit circle diameter is 2.
    # 6 circles span 2 + 5*2 = 12 width? No.
    # Centers at 1, 3, 5, 7, 9, 11. Distance between 1 and 11 is 10.
    # Extent of circles: [0, 12]. Fits in [0, 12] if box is [1, 11]?
    # Wait, center at 1 implies left edge at 0. Center at 11 implies right edge at 12.
    # So if box is [1, S-1], centers must be in [1, S-1].
    # If centers are in [1, 11], edges are [0, 12].
    # The square containing them would be [0, 12], side 12.
    # But our variable S represents the side of the square containing the circles.
    # If circles are radius 1, they fit in [0, S] if centers in [1, S-1].
    # So yes, S=12 fits centers in [1, 11].
    
    centers = []
    
    # Vertical spacing for hexagonal packing is sqrt(3)
    h = math.sqrt(3)
    
    # We have 5 rows. Height occupied by centers is 4*h approx 6.928.
    # Plus radius margin 1 on top and bottom => total height 8.928.
    # S=12 is plenty.
    
    # Let's center the packing in the vertical space of the box [1, S-1]
    # Box vertical range: [1, 11]. Available height for centers: 10.
    # Packing height (centers): 4*h.
    # Offset y_start = 1 + (10 - 4*h)/2
    
    y_start = 1.0 + (10.0 - 4*h) / 2.0
    
    for r_idx, count in enumerate(rows_config):
        current_y = y_start + r_idx * h
        
        # Horizontal spacing is 2.
        # Width occupied by 'count' circles centers: (count-1)*2.
        # Center them in [1, 11].
        # Range width 10.
        x_start = 1.0 + (10.0 - (count - 1) * 2.0) / 2.0
        
        # Shift odd rows by 1 unit to the right for hexagonal packing
        if r_idx % 2 == 1:
            x_start += 1.0
            
        for c in range(count):
            x = x_start + c * 2.0
            centers.append([x, current_y])
            
    centers = np.array(centers)
    
    # Flatten centers and append S
    # Variables: x_0, y_0, ..., x_25, y_25, S
    x0 = np.concatenate([centers.flatten(), [S_init]])
    
    # 2. Define Optimization Problem
    # Minimize S (last variable)
    # Subject to:
    # 1. x_i >= 1
    # 2. x_i <= S - 1
    # 3. y_i >= 1
    # 4. y_i <= S - 1
    # 5. dist(i, j) >= 2  => (x_i-x_j)^2 + (y_i-y_j)^2 >= 4
    
    # We will use SLSQP which handles inequalities well.
    
    def objective(vars):
        return vars[-1] # Minimize S

    # Constraint helper
    # In scipy, constraints are 'ineq' meaning fun(x) >= 0
    
    constraints = []
    
    # Boundary constraints
    # For each circle i (0 to 25)
    # x_i is at index 2*i, y_i at 2*i+1
    # S is at index 52
    
    for i in range(n):
        idx_x = 2 * i
        idx_y = 2 * i + 1
        idx_S = 52
        
        # x_i >= 1
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: v[idx_x] - 1.0})
        # x_i <= S - 1  =>  S - 1 - x_i >= 0
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: v[idx_S] - 1.0 - v[idx_x]})
        # y_i >= 1
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: v[idx_y] - 1.0})
        # y_i <= S - 1
        constraints.append({'type': 'ineq', 'fun': lambda v, i=i: v[idx_S] - 1.0 - v[idx_y]})
        
    # Distance constraints
    # For every pair i < j
    for i in range(n):
        for j in range(i + 1, n):
            idx_xi = 2 * i
            idx_yi = 2 * i + 1
            idx_xj = 2 * j
            idx_yj = 2 * j + 1
            
            def dist_con(v, i=i, j=j):
                dx = v[idx_xi] - v[idx_xj]
                dy = v[idx_yi] - v[idx_yj]
                return dx*dx + dy*dy - 4.0 # dist^2 >= 4
            
            constraints.append({'type': 'ineq', 'fun': dist_con})

    # 3. Run Optimization
    # maxiter might need to be high.
    res = minimize(objective, x0, method='SLSQP', constraints=constraints, 
                   options={'maxiter': 2000, 'ftol': 1e-10})
    
    if not res.success:
        # Fallback or error handling? 
        # Just use result anyway if it's decent.
        pass
        
    optimal_S = res.x[-1]
    optimal_centers_scaled = res.x[:-1].reshape(n, 2)
    
    # 4. Scale back to unit square
    # Current config: circles radius 1 in square of side optimal_S.
    # We want circles in square [0, 1].
    # Scale factor = 1 / optimal_S.
    # But wait, our coordinates were in [1, S-1] range for centers?
    # Actually, the square containing the circles (including radius) was [0, S] 
    # because centers were in [1, S-1] and radius is 1.
    # Let's verify.
    # If center is 1, left edge is 0. If center is S-1, right edge is S.
    # So the square is [0, S] x [0, S]. Side length S.
    # To map [0, S] to [0, 1], we divide coordinates by S.
    # Radius 1 becomes 1/S.
    
    scale_factor = 1.0 / optimal_S
    final_centers = optimal_centers_scaled * scale_factor
    final_radii = np.ones(n) * scale_factor
    
    # Verify sum
    sum_radii = np.sum(final_radii)
    
    # Optional: Local refinement with unequal radii?
    # The problem asks to maximize sum of radii.
    # With equal radii, we found a good packing.
    # Unequal radii might squeeze more, but equal is a strong baseline.
    # Let's stick to this robust solution.
    
    return final_centers, final_radii, float(sum_radii)
