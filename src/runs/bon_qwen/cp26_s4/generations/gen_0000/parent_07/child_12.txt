import numpy as np
from scipy.optimize import minimize
import math

def run_packing():
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    """
    n = 26
    np.random.seed(42) # For reproducibility if random parts used

    # --- Initialization ---
    # We start with a hexagonal packing arrangement.
    # This provides a good structural basis for the optimizer to work from.
    # We aim to place circles in rows.
    # Pattern: 5, 4, 5, 4, 5, 3 circles per row sums to 26.
    
    initial_radii = 0.05 # Start small to ensure no overlaps initially
    
    centers = []
    current_y = initial_radii
    row_idx = 0
    
    # Row configurations: list of number of circles in each row
    # 5 + 4 + 5 + 4 + 5 + 3 = 26
    row_counts = [5, 4, 5, 4, 5, 3]
    
    # Vertical spacing for hex packing is sqrt(3) * radius
    # But since we start with small radius, we use a fixed small vertical step
    # to spread them out initially, or just use the hex geometry relative to r_init
    vertical_step = math.sqrt(3) * initial_radii
    
    for count in row_counts:
        # Horizontal shift for alternating rows
        if row_idx % 2 == 0:
            shift = 0.0
        else:
            shift = initial_radii # Shift by radius (half of 2r spacing)
            
        # Calculate x positions
        # Total width available is 1.
        # We want to center the row of 'count' circles.
        # Spacing between centers is 2 * initial_radii
        total_width_circles = (count - 1) * (2 * initial_radii)
        start_x = (1.0 - total_width_circles) / 2.0 + shift
        
        # Adjust shift logic: if we shift by r, the effective span changes?
        # In hex packing, the centers are at x, x+2r, x+4r...
        # If shifted, they are at x+r, x+3r...
        # The bounding box of the row is [min_x - r, max_x + r].
        # We need max_x + r <= 1 and min_x - r >= 0.
        
        # Let's just place them simply and let optimizer fix positions.
        # Simple placement:
        x_coords = []
        for k in range(count):
            x = start_x + k * (2 * initial_radii)
            # Check bounds, if out, clamp or adjust. 
            # With r=0.05, 2r=0.1. 
            # Row 5 circles: width ~ 0.4. Fits easily.
            # Row 3 circles: width ~ 0.2. Fits easily.
            # Shift might push them out if not careful, but with r=0.05 and width 1, 
            # even shift of 0.05 is fine.
            x_coords.append(x)
        
        for x in x_coords:
            centers.append([x, current_y])
            
        current_y += vertical_step
        row_idx += 1
        
    centers = np.array(centers[:n]) # Ensure we have exactly n
    radii_init = np.full(n, initial_radii)
    
    # Flatten variables: [x1, y1, r1, x2, y2, r2, ...]
    x0 = np.zeros(n * 3)
    for i in range(n):
        x0[3*i] = centers[i, 0]
        x0[3*i+1] = centers[i, 1]
        x0[3*i+2] = radii_init[i]
        
    # Bounds: x, y in [0, 1], r in [0, 0.5]
    bounds = []
    for _ in range(n):
        bounds.append((0, 1)) # x
        bounds.append((0, 1)) # y
        bounds.append((0, 0.5)) # r

    # --- Optimization ---
    
    def objective(vars_flat):
        # Minimize negative sum of radii
        radii = vars_flat[2::3]
        return -np.sum(radii)

    def constraints_factory(vars_flat, n):
        constraints = []
        
        # Extract variables
        xs = vars_flat[0::3]
        ys = vars_flat[1::3]
        rs = vars_flat[2::3]
        
        # 1. Boundary constraints
        # r <= x, r <= 1-x, r <= y, r <= 1-y
        # i.e., x - r >= 0, 1 - x - r >= 0, etc.
        for i in range(n):
            # x - r >= 0
            constraints.append({
                'type': 'ineq',
                'fun': lambda v, i=i: v[3*i] - v[3*i+2]
            })
            # 1 - x - r >= 0
            constraints.append({
                'type': 'ineq',
                'fun': lambda v, i=i: 1.0 - v[3*i] - v[3*i+2]
            })
            # y - r >= 0
            constraints.append({
                'type': 'ineq',
                'fun': lambda v, i=i: v[3*i+1] - v[3*i+2]
            })
            # 1 - y - r >= 0
            constraints.append({
                'type': 'ineq',
                'fun': lambda v, i=i: 1.0 - v[3*i+1] - v[3*i+2]
            })

        # dist >= r_i + r_j
        # dist^2 - (r_i + r_j)^2 >= 0

        # (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0

            for j in range(i + 1, n):
                constraints.append({
                    'type': 'ineq',
                    'fun': lambda v, i=i, j=j: \
                        (v[3*i] - v[3*j])**2 + (v[3*i+1] - v[3*j+1])**2 - (v[3*i+2] + v[3*j+2])**2
                })
                
        return constraints

    cons = constraints_factory(x0, n)
    
    # Run optimizer
    # SLSQP is generally good for this
    result = minimize(
        objective, 
        x0, 
        method='SLSQP', 
        bounds=bounds, 
        constraints=cons,

    sum_radii = np.sum(final_radii)
    
    # Validation check (internal)
    # If validation fails (due to numerical issues), we might need to shrink slightly
    # But SLSQP should respect constraints. 
    # However, floating point errors might cause tiny overlaps.
    # The validate function allows 1e-12 error, so it should be fine.
    
    return final_centers, final_radii, sum_radii
