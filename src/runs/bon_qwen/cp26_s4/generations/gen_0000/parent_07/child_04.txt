    y_spacing = 1.0 / (total_rows + 1) # Rough spacing
    x_spacing = 1.0 / (max(rows) + 1)  # Rough spacing
    
    for r_idx, count in enumerate(rows):
        y = (r_idx + 1) * y_spacing
        # Shift every other row for hexagonal effect
        x_offset = 0.5 * x_spacing if r_idx % 2 == 1 else 0.0

            x = (c_idx + 1) * x_spacing + x_offset
            centers[current_idx, 0] = x
            centers[current_idx, 1] = y
            current_idx += 1

        radii += expansion_step

                        forces[j] -= direction * force

                    forces[i, 0] -= repulsion_strength * (x + r - 1)

                    forces[i, 1] -= repulsion_strength * (y + r - 1)
