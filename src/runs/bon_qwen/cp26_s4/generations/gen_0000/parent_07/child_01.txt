def generate_initial_hex_packing(n_circles):
    """Generates an initial hexagonal packing configuration for n_circles."""
    # Estimate radius for n_circles in a hexagonal packing
    # Approximation based on density
    # Area of square = 1. Density of hex packing ~ 0.9069
    # n * pi * r^2 ~ 0.9069 => r ~ sqrt(0.9069 / (n * pi))
    # However, boundary effects reduce this. Start with a safe r.
    r_init = 0.08 
    
    # Create a hexagonal grid of points
    points = []
    row_height = r_init * np.sqrt(3)
    
    # We need to fit n_circles. Let's estimate rows needed.
    # Width ~ 1, so ~ 1/(2*r) circles per row.
    # Let's just generate a dense grid and pick first n_circles
    x = 0.0
    y = 0.0
    
    # Add a small margin for safety
    margin = 0.0
    
    row_idx = 0
    while len(points) < n_circles:
        # Determine number of circles in this row to fit in width 1
        # Centers at x, x+2r, x+4r...
        # First center at r + margin, last at 1 - r - margin
        # But let's just place them with spacing 2*r_init
        
        x_pos = r_init + margin
        while x_pos <= 1 - r_init - margin:
            points.append([x_pos, y])
            x_pos += 2 * r_init
        
        # Shift next row
        if len(points) < n_circles:
            if row_idx % 2 == 0:
                x_start = r_init + margin + r_init # shift by r
            else:
                x_start = r_init + margin
            
            x_pos = x_start
            while x_pos <= 1 - r_init - margin:
                points.append([x_pos, y + row_height])
                x_pos += 2 * r_init
            
            y += row_height
            row_idx += 1
            
    # Select first n_circles
    points = points[:n_circles]
    
    centers = np.array(points)
    radii = np.full(n_circles, r_init)
    
    return centers, radii
