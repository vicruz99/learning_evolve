    n = 26

    radii = x[2 * n:]

    constraints = np.empty(4 * n)
    constraints[0::4] = cx - radii
    constraints[1::4] = 1.0 - cx - radii

    constraints[3::4] = 1.0 - cy - radii
    return constraints
