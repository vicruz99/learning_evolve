import numpy as np
from scipy.optimize import minimize

def check_overlap_constraint(vars_flat, i, j, n_circles):
    """Helper to extract constraint for pair (i, j)"""
    # vars_flat structure: [x0, y0, r0, x1, y1, r1, ...]
    # Index for circle k: 3*k, 3*k+1, 3*k+2
    idx_i = 3 * i
    idx_j = 3 * j
    
    x_i, y_i, r_i = vars_flat[idx_i], vars_flat[idx_i+1], vars_flat[idx_i+2]
    x_j, y_j, r_j = vars_flat[idx_j], vars_flat[idx_j+1], vars_flat[idx_j+2]
    
    # Constraint: dist >= r_i + r_j  => dist^2 >= (r_i + r_j)^2
    # We want constraint >= 0.
    # dist^2 - (r_i + r_j)^2 >= 0
    
    dx = x_i - x_j
    dy = y_i - y_j
    dist_sq = dx*dx + dy*dy
    r_sum_sq = (r_i + r_j)**2
    
    return dist_sq - r_sum_sq

def check_boundary_constraint(vars_flat, i, side_type, n_circles):
    """Helper to extract boundary constraint.
    side_type: 'left' (x >= r), 'right' (x <= 1-r => 1-x-r >= 0), 
                'bottom' (y >= r), 'top' (y <= 1-r => 1-y-r >= 0)
    """
    idx = 3 * i
    x, y, r = vars_flat[idx], vars_flat[idx+1], vars_flat[idx+2]
    
    if side_type == 'left':
        return x - r
    elif side_type == 'right':
        return 1 - x - r
    elif side_type == 'bottom':
        return y - r
    elif side_type == 'top':
        return 1 - y - r
    return 0
