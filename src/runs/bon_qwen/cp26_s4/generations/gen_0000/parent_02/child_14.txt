The problem is to maximize $\sum r_i$ subject to non-overlap and boundary constraints. This is a non-convex optimization problem due to the Euclidean distance constraints. However, for fixed centers, the radii subproblem is a linear program (LP). We can leverage this structure:

1. **Initialization**: Generate multiple random center configurations spread within $[0.15, 0.85]^2$ to avoid boundary clustering initially.
2. **Feasible Radii via LP**: For each center configuration, solve an LP to find the maximum possible radii that satisfy all constraints. This guarantees a feasible starting point for the joint optimizer.
3. **Joint Optimization**: Use `scipy.optimize.minimize` with the `SLSQP` method to jointly optimize centers and radii. SLSQP handles inequality constraints efficiently. The constraints enforce $r_i + r_j \le \|c_i - c_j\|$ and $r_i \le \text{dist}(c_i, \partial)$.
4. **Multiple Restarts**: Run the optimizer from several random seeds to escape poor local minima and select the best result.
5. **Efficiency**: Vectorize constraint evaluation using NumPy broadcasting and `np.triu_indices` to handle pairwise constraints efficiently. Keep all helper functions top-level as requested.

This approach combines the robustness of LP initialization with the flexibility of continuous constrained optimization, effectively navigating the non-convex landscape to find a high-quality packing.

import numpy as np
from scipy.optimize import minimize, linprog

N_CIRCLES = 26

def solve_radii_lp(centers):
    """Solve LP to find maximum feasible radii for fixed centers."""
    n = centers.shape[0]
    c = -np.ones(n)  # Maximize sum r_i => minimize -sum r_i
    A_ub = []
    b_ub = []
    
    # Wall constraints: r_i <= x_i, 1-x_i, y_i, 1-y_i
    for i in range(n):
        x, y = centers[i]
        row = np.zeros(n)
        row[i] = 1.0
        for val in [x, 1.0-x, y, 1.0-y]:
            A_ub.append(row)
            b_ub.append(val)
            
    # Pairwise constraints: r_i + r_j <= ||c_i - c_j||
    for i in range(n):
        for j in range(i+1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dist)
            
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    return res.x if res.success else np.full(n, 1e-6)

def objective_func(x):
    """Objective: maximize sum of radii (minimize negative sum)."""
    return -np.sum(x[2*N_CIRCLES:])

def constraint_func(x):
    """Inequality constraints: g(x) >= 0."""
    n = N_CIRCLES
    cx = x[:2*n]
    r = x[2*n:]
    
    # Wall constraints: dist_to_boundary - r_i >= 0
    w = np.empty(4*n)
    w[0::4] = cx[0::2] - r
    w[1::4] = 1.0 - cx[0::2] - r
    w[2::4] = cx[1::2] - r
    w[3::4] = 1.0 - cx[1::2] - r
    
    # Pairwise constraints: ||c_i - c_j|| - r_i - r_j >= 0
    cx_2d = cx.reshape(n, 2)
    diff = cx_2d[:, np.newaxis, :] - cx_2d[np.newaxis, :, :]
    dists = np.sqrt(diff[:,:,0]**2 + diff[:,:,1]**2)
    np.fill_diagonal(dists, 0.0)
    
    i_idx, j_idx = np.triu_indices(n, k=1)
    pair_vals = dists[i_idx, j_idx] - r[i_idx] - r[j_idx]
    
    return np.concatenate([w, pair_vals])

def run_packing():
    n = N_CIRCLES
    best_sum_r = 0.0
    best_centers = np.zeros((n, 2))
    best_radii = np.zeros(n)
    
    bounds = [(0.0, 1.0)] * (2*n) + [(0.0, None)] * n
    cons = {'type': 'ineq', 'fun': constraint_func}
    
    # Multiple random restarts to find global optimum
    for seed in range(20):
        rng = np.random.default_rng(seed)
        centers = rng.uniform(0.15, 0.85, size=(n, 2))
        
        # Initialize radii via LP to ensure feasibility
        radii = solve_radii_lp(centers)
        x0 = np.concatenate([centers.flatten(), radii])
        
        res = minimize(objective_func, x0, method='SLSQP', 
                       bounds=bounds, constraints=cons, 
                       options={'maxiter': 1000, 'ftol': 1e-9, 'disp': False})
                       
        if res.success:
            s_r = np.sum(res.x[2*n:])
            if s_r > best_sum_r:
                best_sum_r = s_r
                best_centers = res.x[:2*n].reshape(n, 2)
                best_radii = res.x[2*n:]
                
    return best_centers, best_radii, best_sum_r
