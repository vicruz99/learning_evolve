

<strategy>
My approach combines several strategies:

1. **Multiple Initial Configurations**: I'll try hexagonal grids, random placements, and structured grids (5×5 + extras, 5×6 with extras removed) to find diverse starting points.

2. **LP-based Radius Optimization**: For fixed centers, finding optimal radii is a linear program (maximize sum(r_i) subject to r_i + r_j ≤ dist(i,j) and boundary constraints). I'll use scipy's linprog with HiGHS solver.

3. **Iterative Position Optimization**: I'll use gradient ascent with numerical gradients to improve center positions, alternating between:
   - LP-based precise radius computation (periodically)
   - Fast iterative radius updates (for speed)

4. **Force-based refinement**: After gradient optimization, I'll apply a repulsive force method to spread circles apart and allow them to grow.

5. **Corner exploitation**: Circles near corners can be larger. I'll ensure my initial configurations explore corner regions.

6. **Multiple restarts**: I'll run several independent optimizations from different starting points and keep the best result.

The key insight is that for maximizing sum of radii, circles don't need to be equal - corner and boundary circles can be larger than interior ones, so the LP naturally handles this.

import numpy as np
from scipy.optimize import linprog


def solve_radii_lp(centers):
    """Solve LP to find optimal radii for fixed centers."""
    n = centers.shape[0]
    c = -np.ones(n)
    
    constraints = []
    rhs = []
    
    for i in range(n):
        x, y = centers[i]
        for bv in [x, 1 - x, y, 1 - y]:
            row = np.zeros(n)
            row[i] = 1
            constraints.append(row)
            rhs.append(bv)
    
    for i in range(n):
        for j in range(i + 1, n):
            d = np.linalg.norm(centers[i] - centers[j])
            row = np.zeros(n)
            row[i] = 1
            row[j] = 1
            constraints.append(row)
            rhs.append(d)
    
    A = np.array(constraints)
    b = np.array(rhs)
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    
    if result.success:
        return result.x
    return np.zeros(n)


def compute_radii_fast(centers):
    """Fast iterative radius computation without LP."""
    n = centers.shape[0]
    radii = np.array([min(c[0], 1 - c[0], c[1], 1 - c[1]) for c in centers])
    
    for _ in range(150):
        for i in range(n):
            for j in range(i + 1, n):
                d = np.linalg.norm(centers[i] - centers[j])
                if radii[i] + radii[j] > d + 1e-10:
                    r_sum = radii[i] + radii[j]
                    if r_sum > 0:
                        ratio_i = radii[i] / r_sum
                    else:
                        ratio_i = 0.5
                    excess = radii[i] + radii[j] - d
                    radii[i] -= excess * ratio_i
                    radii[j] -= excess * (1 - ratio_i)
    
    return np.maximum(radii, 0)


def generate_hex_grid(n, margin=0.1):
    """Generate hexagonal grid of n points."""
    centers = []
    row = 0
    col = 0
    sx = 0.16
    sy = 0.14
    while len(centers) < n:
        x = margin + col * sx
        y = margin + row * sy
        if row % 2 == 1:
            x += sx / 2
        if x <= 1 - margin and y <= 1 - margin:
            centers.append([x, y])
            col += 1
        else:
            col = 0
            row += 1
    return np.array(centers[:n])


def generate_structured_grid(n):
    """Generate structured grid with some circles removed from 5x6."""
    centers = []
    for i in range(5):
        for j in range(6):
            if len(centers) < n:
                x = 0.1 + i * 0.18
                y = 0.0833 + j * 0.15
                if i % 2 == 1:
                    x += 0.09
                centers.append([x, y])
    return np.array(centers[:n])


def generate_corner_exploiting(n):
    """Generate configuration that exploits corners."""
    centers = []
    # Four corners
    corners = [[0.1, 0.1], [0.9, 0.1], [0.1, 0.9], [0.9, 0.9]]
    for c in corners:
        if len(centers) < n:
            centers.append(c)
    
    # Edge midpoints
    edges = [[0.5, 0.1], [0.5, 0.9], [0.1, 0.5], [0.9, 0.5]]
    for c in edges:
        if len(centers) < n:
            centers.append(c)
    
    # Fill remaining with hex grid
    row = 0
    col = 0
    while len(centers) < n:
        x = 0.25 + col * 0.18
        y = 0.25 + row * 0.16
        if row % 2 == 1:
            x += 0.09
        if 0.2 <= x <= 0.8 and 0.2 <= y <= 0.8:
            centers.append([x, y])
            col += 1
        else:
            col = 0
            row += 1
    
    return np.array(centers[:n])


def optimize_gradient(centers_init, n_iter=120, use_lp_every=15):
    """Optimize centers using gradient ascent on sum of radii."""
    n = centers_init.shape[0]
    centers = centers_init.copy()
    
    for it in range(n_iter):
        if it % use_lp_every == 0 or it == n_iter - 1:
            radii = solve_radii_lp(centers)
        else:
            radii = compute_radii_fast(centers)
        
        s0 = np.sum(radii)
        
        grad = np.zeros((n, 2))
        h = 1e-4
        for i in range(n):
            for d in range(2):
                cp = centers.copy()
                cp[i, d] += h
                if it % use_lp_every == 0 or it == n_iter - 1:
                    rp = solve_radii_lp(cp)
                else:
                    rp = compute_radii_fast(cp)
                grad[i, d] = (np.sum(rp) - s0) / h
        
        lr = 0.005 / (1 + it * 0.02)
        centers += lr * grad
        centers = np.clip(centers, 0.03, 0.97)
    
    radii = solve_radii_lp(centers)
    return centers, radii


def optimize_force_based(centers_init, n_steps=500):
    """Optimize using repulsive forces to spread circles."""
    n = centers_init.shape[0]
    centers = centers_init.copy()
    radii = compute_radii_fast(centers)
    
    for step in range(n_steps):
        forces = np.zeros((n, 2))
        
        # Repulsive forces between circles
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[j, 0] - centers[i, 0]
                dy = centers[j, 1] - centers[i, 1]
                d = np.sqrt(dx * dx + dy * dy)
                if d < 0.001:
                    d = 0.001
                    dx, dy = 0.001, 0
                
                target_dist = radii[i] + radii[j] + 0.02
                if d < target_dist:
                    force_mag = (target_dist - d) * 0.1 / d
                    forces[i, 0] -= dx * force_mag
                    forces[i, 1] -= dy * force_mag
                    forces[j, 0] += dx * force_mag
                    forces[j, 1] += dy * force_mag
        
        # Boundary forces (push inward if too close)
        for i in range(n):
            if centers[i, 0] < radii[i] + 0.01:
                forces[i, 0] += 0.05
            if centers[i, 0] > 1 - radii[i] - 0.01:
                forces[i, 0] -= 0.05
            if centers[i, 1] < radii[i] + 0.01:
                forces[i, 1] += 0.05
            if centers[i, 1] > 1 - radii[i] - 0.01:
                forces[i, 1] -= 0.05
        
        # Apply forces
        lr = 0.01 / (1 + step * 0.005)
        centers += lr * forces
        centers = np.clip(centers, 0.02, 0.98)
        
        # Update radii periodically
        if step % 20 == 0:
            radii = compute_radii_fast(centers)
    
    radii = solve_radii_lp(centers)
    return centers, radii


def run_packing():
    """Main function to pack 26 circles in unit square."""
    n = 26
    best_centers = None
    best_radii = None
    best_sum = -1.0
    
    # Generate diverse initial configurations
    configs = []
    
    # Hexagonal grids with different parameters
    configs.append(generate_hex_grid(n, margin=0.08))
    configs.append(generate_hex_grid(n, margin=0.12))
    configs.append(generate_hex_grid(n, margin=0.15))
    
    # Structured grids
    configs.append(generate_structured_grid(n))
    
    # Corner-exploiting
    configs.append(generate_corner_exploiting(n))
    
    # Random configurations
    np.random.seed(42)
    for seed in range(1, 8):
        np.random.seed(seed)
        c = np.random.uniform(0.15, 0.85, (n, 2))
        configs.append(c)
    
    # Grid-based: 5x5 + 1
    c55 = []
    for i in range(5):
        for j in range(5):
            c55.append([0.1 + i * 0.2, 0.1 + j * 0.2])
    c55.append([0.5, 0.5])
    configs.append(np.array(c55[:n]))
    
    # Grid-based: 6x5 with 4 removed
    c65 = []
    idx = 0
    for i in range(6):
        for j in range(5):
            if idx < n:
                x = 0.0833 + i * 0.15
                y = 0.1 + j * 0.18
                c65.append([x, y])
                idx += 1
    configs.append(np.array(c65))
    
    # Optimize each configuration
    for cfg in configs:
        # Gradient-based optimization
        try:
            c1, r1 = optimize_gradient(cfg.copy(), n_iter=80, use_lp_every=12)
            s1 = np.sum(r1)
            if s1 > best_sum:
                best_sum = s1
                best_centers = c1.copy()
                best_radii = r1.copy()
        except Exception:
            pass
        
        # Force-based optimization
        try:
            c2, r2 = optimize_force_based(cfg.copy(), n_steps=300)
            s2 = np.sum(r2)
            if s2 > best_sum:
                best_sum = s2
                best_centers = c2.copy()
                best_radii = r2.copy()
        except Exception:
            pass
        
        # Combined: gradient then force
        try:
            c3, r3 = optimize_gradient(cfg.copy(), n_iter=60, use_lp_every=15)
            c4, r4 = optimize_force_based(c3.copy(), n_steps=200)
            s4 = np.sum(r4)
            if s4 > best_sum:
                best_sum = s4
                best_centers = c4.copy()
                best_radii = r4.copy()
        except Exception:
            pass
    
    # Final LP refinement
    if best_centers is not None:
        best_radii = solve_radii_lp(best_centers)
        best_sum = np.sum(best_radii)
    
    # Validate and ensure no NaN
    if np.isnan(best_centers).any() or np.isnan(best_radii).any():
        # Fallback to simple configuration
        best_centers = generate_hex_grid(n)
        best_radii = solve_radii_lp(best_centers)
        best_sum = np.sum(best_radii)
    
    return best_centers, best_radii, best_sum
