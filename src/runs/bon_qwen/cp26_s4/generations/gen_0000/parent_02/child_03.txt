import numpy as np
from scipy.optimize import minimize

# Read-only validation function provided in the prompt
def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square

    Args:
        centers: np.array of shape (n, 2) with (x, y) coordinates
        radii: np.array of shape (n) with radius of each circle

    Returns:
        True if valid, False otherwise
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True


def compute_loss(x, n, penalty_weight):
    """
    Computes the objective function value: negative sum of radii + penalty for violations.
    
    Args:
        x: 1D array containing [x0, y0, r0, x1, y1, r1, ...]
        n: Number of circles
        penalty_weight: Weight for the penalty term
    """
    centers = x[:2*n].reshape(n, 2)
    radii = x[2*n:]
    
    # 1. Objective: minimize -sum(radii) => maximize sum(radii)
    loss = -np.sum(radii)
    
    # 2. Boundary penalties
    # Constraints: r <= x <= 1-r  =>  x - r >= 0  and  x + r <= 1
    # Same for y
    x_c = centers[:, 0]
    y_c = centers[:, 1]
    r = radii
    
    # Violation magnitudes (clamped to 0)
    v_left   = np.maximum(0, r - x_c)
    v_right  = np.maximum(0, x_c + r - 1.0)
    v_bottom = np.maximum(0, r - y_c)
    v_top    = np.maximum(0, y_c + r - 1.0)
    
    boundary_penalty = np.sum(v_left**2 + v_right**2 + v_bottom**2 + v_top**2)
    
    # 3. Overlap penalties
    # Compute pairwise distances
    # (N, 1, 2) - (1, N, 2) -> (N, N, 2)
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    
    # Set diagonal to large value to ignore self-distance
    np.fill_diagonal(dists, 10.0)
    
    # Sum of radii matrix
    r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    
    # Violation: sum_radii - dist > 0
    violation = np.maximum(0, r_sum - dists)
    # Sum of squared violations (upper triangle is sufficient, but full sum is constant factor)
    overlap_penalty = np.sum(violation**2)
    
    total_penalty = boundary_penalty + overlap_penalty
    
    return loss + penalty_weight * total_penalty


def generate_hex_init(n):
    """
    Generates an initial configuration based on a hexagonal lattice.
    """
    # Create a dense grid of points
    # We want to roughly cover the square
    pts = []
    # Spacing approx 0.15
    for y in np.arange(0.15, 0.95 + 1e-9, 0.173): # sqrt(3)/2 * 0.2 approx 0.173
        offset = 0.1 if int(y / 0.173) % 2 == 1 else 0.0
        for x in np.arange(0.15, 1.0 - offset + 1e-9, 0.2):
            pts.append([x + offset, y])
    
    # Shuffle and pick n points
    np.random.shuffle(pts)
    return np.array(pts[:n])


def run_packing():
    n = 26
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    num_attempts = 40
    rng = np.random.RandomState(42)
    
    for attempt in range(num_attempts):
        # 1. Initialization
        if attempt < 15:
            # Structured Hexagonal Init
            centers = generate_hex_init(n)
            # Ensure within bounds slightly
            centers = np.clip(centers, 0.05, 0.95)
        else:
            # Random Init
            centers = rng.uniform(0.05, 0.95, (n, 2))
        
        # Initial radii small enough to avoid overlap in most random cases
        # but large enough to give gradient signal
        radii = np.full(n, 0.05)
        
        # Flatten variables: [x0, y0, r0, ...]
        x0 = np.concatenate([centers.flatten(), radii])
        
        # 2. Optimization Loop with increasing penalty
        # This helps the optimizer to first find a feasible region (if starting invalid)
        # or simply enforce constraints tightly as radii grow.
        penalty_weights = [50, 200, 1000, 5000]
        
        curr_x = x0.copy()
        
        for w in penalty_weights:
            res = minimize(
                lambda z: compute_loss(z, n, w),
                curr_x,
                method='BFGS',
                options={'maxiter': 1000, 'gtol': 1e-6}
            )
            curr_x = res.x
            
            # If loss is NaN or huge, break
            if np.isnan(res.fun) or res.fun > 1e6:
                break
        
        # 3. Extract and Validate
        res_centers = curr_x[:2*n].reshape(n, 2)
        res_radii = curr_x[2*n:]
        
        # Clamp radii to be non-negative just in case
        res_radii = np.maximum(res_radii, 0.0)
        
        if validate_packing(res_centers, res_radii):
            current_sum = np.sum(res_radii)
            if current_sum > best_sum:
                best_sum = current_sum
                best_centers = res_centers.copy()
                best_radii = res_radii.copy()
                
                # Early exit if we reach target (optional, but good for performance)
                if best_sum >= 2.635:
                    break
    
    if best_centers is None:
        # Fallback if nothing worked (unlikely)
        centers = np.random.uniform(0.1, 0.9, (n, 2))
        radii = np.full(n, 0.01)
        return centers, radii, np.sum(radii)

    return best_centers, best_radii, best_sum

# Execution wrapper
if __name__ == "__main__":
    centers, radii, s = run_packing()
    print(f"Sum of radii: {s}")
