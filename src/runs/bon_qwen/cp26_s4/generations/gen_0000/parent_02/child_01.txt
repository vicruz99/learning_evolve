import numpy as np
from scipy.optimize import minimize

def compute_penalty(centers, radii, alpha):
    # Vectorized distance computation
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.linalg.norm(diffs, axis=2)
    np.fill_diagonal(dists, np.inf)
    
    radii_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    overlaps = np.maximum(0, radii_sum - dists)
    overlap_pen = np.sum(overlaps**2)
    
    # Boundary distances
    bounds = np.minimum(np.minimum(centers[:,0], 1-centers[:,0]), 
                        np.minimum(centers[:,1], 1-centers[:,1]))
    bound_viol = np.maximum(0, radii - bounds)
    bound_pen = np.sum(bound_viol**2)
    
    # Objective: maximize sum of radii (minimize negative sum) + penalties
    return -np.sum(radii) + alpha * (overlap_pen + bound_pen)

def objective(params, N, alpha):
    c = params[:2*N].reshape(N, 2)
    r = params[2*N:]
    return compute_penalty(c, r, alpha)

def run_packing():
    N = 26
    # Bounds for variables: x,y in [0,1], r in [0, 0.5]
    bounds = [(0.0, 1.0)] * 2*N + [(0.0, 0.5)] * N
    
    best_res = None
    best_val = np.inf
    
    # Try multiple initializations to escape local minima
    for seed in range(5):
        np.random.seed(seed)
        
        # Create a structured grid with slight jitter
        cx = np.linspace(0.12, 0.88, 6)
        cy = np.linspace(0.12, 0.88, 5)
        pts = []
        for y in cy:
            for x in cx:
                if len(pts) < N:
                    pts.append([x, y])
        
        pts = np.array(pts) + np.random.randn(*pts.shape) * 0.015
        pts = np.clip(pts, 0.05, 0.95)
        
        r_init = 0.04 + np.random.rand(N) * 0.02
        x0 = np.concatenate([pts.ravel(), r_init])
        
        # Phase 1: Lower penalty to quickly resolve overlaps and find feasible layout
        res1 = minimize(objective, x0, args=(N, 1000.0), method='L-BFGS-B', bounds=bounds,
                        options={'maxiter': 2000, 'ftol': 1e-9, 'gtol': 1e-6})
        
        # Phase 2: Higher penalty to aggressively maximize radii while maintaining validity
        res2 = minimize(objective, res1.x, args=(N, 20000.0), method='L-BFGS-B', bounds=bounds,
                        options={'maxiter': 3000, 'ftol': 1e-10, 'gtol': 1e-8})
                        
        if res2.fun < best_val:
            best_val = res2.fun
            best_res = res2
            
    c_opt = best_res.x[:2*N].reshape(N, 2)
    r_opt = best_res.x[2*N:]
    
    # Strict boundary enforcement
    for i in range(N):
        max_r = min(c_opt[i,0], 1-c_opt[i,0], c_opt[i,1], 1-c_opt[i,1], 0.5)
        r_opt[i] = min(r_opt[i], max_r)
        
    # Iterative overlap resolution to guarantee validity within tolerance
    changed = True
    while changed:
        changed = False
        for i in range(N):
            for j in range(i+1, N):
                dist = np.linalg.norm(c_opt[i] - c_opt[j])
                if dist < r_opt[i] + r_opt[j] - 1e-9:
                    overlap = r_opt[i] + r_opt[j] - dist
                    r_opt[i] -= overlap/2
                    r_opt[j] -= overlap/2
                    changed = True
        # Re-clip after shrinking
        for i in range(N):
             max_r = min(c_opt[i,0], 1-c_opt[i,0], c_opt[i,1], 1-c_opt[i,1], 0.5)
             r_opt[i] = min(r_opt[i], max_r)

    return c_opt, r_opt, float(np.sum(r_opt))
