import numpy as np
from scipy.optimize import minimize

def get_constraints(params, n):
    """
    Compute inequality constraints: g(x) >= 0
    params: [x1, y1, ..., xn, yn, r1, ..., rn]
    """
    centers = params[:2*n].reshape(n, 2)
    radii = params[2*n:]
    
    constraints = []
    
    # Boundary constraints: x - r >= 0, 1 - x - r >= 0, etc.
    constraints.append(centers[:, 0] - radii)
    constraints.append(1.0 - centers[:, 0] - radii)
    constraints.append(centers[:, 1] - radii)
    constraints.append(1.0 - centers[:, 1] - radii)
    
    # Pairwise distance constraints: dist(i,j) - r_i - r_j >= 0
    # Compute pairwise differences
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    # Extract upper triangle to avoid duplicates and self-pairs
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    
    constraints.append(dists[mask] - r_sum[mask])
    
    return np.concatenate(constraints)

def run_packing():

    # --- Initialization: Hexagonal Lattice ---
    # We arrange circles in rows to approximate hexagonal packing
    rows = [5, 5, 6, 5, 5]  # Total 26 circles
    h = np.sqrt(3)/2.0
    pts = []
    y = 0
    for r_idx, count in enumerate(rows):
        x_start = 0.5 - (count * 1.0 + (1 if r_idx%2==1 else 0)) / 2.0
        for c_idx in range(count):
            x = x_start + c_idx + (0.5 if r_idx%2==1 else 0.0)
            pts.append([x, y])
        y += h
    pts = np.array(pts)
    
    # Scale and translate to fit within [0,1]^2 with some margin
    # We want to leave room for the optimizer to expand radii
    margin = 0.1
    pts = (pts - pts.min(axis=0)) / (pts.max(axis=0) - pts.min(axis=0))
    pts = pts * (1.0 - 2*margin) + margin
    
    # Initial radii
    r_init = np.full(n, 0.05)
    
    # Combine into params vector: [x1, y1, ..., x26, y26, r1, ..., r26]
    x0 = np.concatenate([pts.flatten(), r_init])
    
    # Bounds: 0 <= x,y <= 1, 0 <= r <= 0.5
    bounds = [(0, 1)] * (2*n) + [(0, 0.5)] * n
    
    # Constraints for SLSQP
    cons = {'type': 'ineq', 'fun': lambda p: get_constraints(p, n)}
    
    # Objective: minimize negative sum of radii
    def objective(p):
        return -np.sum(p[2*n:])
    
    # Run optimization
    try:
        res = minimize(objective, x0, method='SLSQP', bounds=bounds, 
                       constraints=cons, options={'maxiter': 2000, 'ftol': 1e-12, 'disp': False})
        best_params = res.x
    except Exception:
        # Fallback if optimization fails
        best_params = x0
        
    # Extract results

    radii_opt = best_params[2*n:]

    # Slightly shrink radii if any constraint is barely violated due to numerical noise
    # Check pairwise
    diff = centers_opt[:, np.newaxis, :] - centers_opt[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    r_sum = radii_opt[:, np.newaxis] + radii_opt[np.newaxis, :]
    
    # Find minimum slack
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    slacks = dists[mask] - r_sum[mask]
    min_slack = np.min(slacks)
    
    # Check boundaries

        np.minimum(centers_opt[:, 0] - radii_opt, 1.0 - centers_opt[:, 0] - radii_opt),

        radii_opt = radii_opt * shrink_factor

    sum_radii = np.sum(radii_opt)
    return centers_opt, radii_opt, sum_radii
