import numpy as np
from scipy.optimize import minimize


def objective_func(x, n):
    """Maximize sum of radii by minimizing negative sum."""
    return -np.sum(x[2 * n:])


def constraint_func(x, n):
    """All constraints: boundary and non-overlap."""
    centers = x[:2 * n].reshape(n, 2)
    radii = x[2 * n:]
    result = []

    # Boundary constraints: each circle must be inside [0,1]^2
    for i in range(n):
        result.append(centers[i, 0] - radii[i])
        result.append(1.0 - centers[i, 0] - radii[i])
        result.append(centers[i, 1] - radii[i])
        result.append(1.0 - centers[i, 1] - radii[i])

    # Non-overlap constraints: distance squared >= (r1 + r2)^2
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist_sq = dx * dx + dy * dy
            r_sum = radii[i] + radii[j]
            result.append(dist_sq - r_sum * r_sum)

    return np.array(result)


def make_hex_initial(row_sizes, r0):
    """Create hexagonal grid initial configuration."""
    centers = []
    sqrt3 = np.sqrt(3)
    for ri, sz in enumerate(row_sizes):
        y = r0 + ri * r0 * sqrt3
        for ci in range(sz):
            offset = r0 if ri % 2 == 1 else 0
            x = r0 + ci * 2.0 * r0 + offset
            centers.append([x, y])
    return np.array(centers)


def center_config(centers):
    """Center and scale configuration to fit nicely in unit square."""
    centers = centers.copy()
    for dim in range(2):
        min_val = centers[:, dim].min()
        max_val = centers[:, dim].max()
        span = max_val - min_val
        if span > 1e-10:
            centers[:, dim] = (centers[:, dim] - min_val) / span * 0.82 + 0.09
    return centers


def try_optimization(centers, radii, n, maxiter=10000):
    """Run one SLSQP optimization attempt."""
    x0 = np.concatenate([centers.flatten(), radii])

    bounds = []
    for i in range(n):
        bounds.append((1e-6, 0.99))
        bounds.append((1e-6, 0.99))
        bounds.append((1e-9, 0.49))

    cons = {'type': 'ineq', 'fun': constraint_func, 'args': (n,)}

    try:
        result = minimize(
            objective_func,
            x0,
            args=(n,),
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={
                'maxiter': maxiter,
                'ftol': 1e-15,
                'disp': False,
                'iprint': -1,
            },
        )
        if result.success or result.fun < -2.0:
            co = result.x[:2 * n].reshape(n, 2)
            ro = result.x[2 * n:]
            return co, ro, np.sum(ro)
    except Exception:
        pass

    return centers, radii, np.sum(radii)


def perturb_config(centers, radii, n, scale=0.015):
    """Add random perturbation to configuration."""
    rng = np.random.default_rng(42)
    c2 = centers + rng.normal(size=(n, 2)) * scale
    r2 = radii + rng.normal(size=n) * scale * 0.3
    c2 = np.clip(c2, 0.01, 0.99)
    r2 = np.maximum(r2, 0.001)
    return c2, r2
