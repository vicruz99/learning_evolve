import numpy as np
import math

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    Solves the 26-circle packing problem in a unit square to maximize the sum of radii.
    Uses a hexagonal initial layout and Simulated Annealing for optimization.

    radii = np.zeros(n)

    # Row configuration to sum to 26: 4 + 5 + 5 + 6 + 5 + 5
    rows = [4, 5, 5, 6, 5, 5]

        y_coord = 0.08 + r_idx * 0.16  # Staggered vertical spacing
        
        # Horizontal spacing adjusted for row width
        x_start = (1.0 - (count - 1) * 0.18) / 2.0
        
        # Stagger even/odd rows slightly
        offset = 0.09 if r_idx % 2 == 1 else 0.0
        
        for c_idx in range(count):
            if idx < n:
                centers[idx, 0] = x_start + c_idx * 0.18 + offset
                centers[idx, 1] = y_coord
                radii[idx] = 0.09
                idx += 1

    # --- 2. Simulated Annealing Optimization ---
    current_sum = np.sum(radii)
    best_centers = centers.copy()
    best_radii = radii.copy()
    best_sum = current_sum
    
    temperature = 0.05
    alpha = 0.995
    iterations = 25000
    
    for it in range(iterations):
        # Select random circle to perturb
        i = np.random.randint(n)
        
        # 2a. Positional Perturbation (Jiggle)
        centers[i] += np.random.normal(0, temperature * 0.5, 2)
        centers[i] = np.clip(centers[i], 0.001, 0.999)
        
        # 2b. Radius Expansion Attempt
        if np.random.random() < 0.6:
            radii[i] *= 1.002

        # Ensure no overlaps with other circles
        for j in range(n):
            if i == j: continue
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, , 1]
            dist = math.sqrt(dx*dx + dy*dy)
            min_dist = radii[i] + radii[j]
            
            if dist < min_dist and dist > 1e-9:
                # Push circles apart
                overlap = min_dist - dist
                factor = overlap / dist
                push = np.array([dx * factor, dy * factor]) * 0.5
                centers[i] += push
                centers[j] -= push
        
        # Ensure no overlap with boundaries
        if centers[i, 0] < radii[i]: centers[i, 0] = radii[i]
        if centers[i, 0] > 1.0 - radii[i]: centers[i, 0] = 1.0 - radii[i]
        if centers[i, 1] < radii[i]: centers[i, 1] = radii[i]
        if centers[i, 1] > 1.0 - radii[i]: centers[i, 1] = 1.0 - radii[i]

        # --- 4. Acceptance Criteria ---
        new_sum = np.sum(radii)
        delta = new_sum - current_sum

        if delta > 0 or np.random.random() < math.exp(delta / temperature):
            current_sum = new_sum
            if current_sum > best_sum:
                best_sum = current_sum
                best_centers = centers.copy()
                best_radii = radii.copy()
        else:
            # Revert changes if rejected
            centers[i] = best_centers[i]
            radii[i] = best_radii[i]
            current_sum = best_sum
            
        temperature *= alpha

    return best_centers, best_radii, best_sum
