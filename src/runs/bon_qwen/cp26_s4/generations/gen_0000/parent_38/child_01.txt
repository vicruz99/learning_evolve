    r_i = min(dist(c_i, c_j) - r_j for all j != i, dist(c_i, boundary))

    )

    If we set all r_i based on nearest neighbor, it's consistent.
    """
    n = centers.shape[0]
    radii = np.full(n, 1.0)

    for i in range(n):
        x, y = centers[i]

        d_bound = min(x, 1 - x, y, 1 - y)

        d_centers = np.inf

        radii[i] = min(d_bound, d_centers / 2.0)
        
    return radii
