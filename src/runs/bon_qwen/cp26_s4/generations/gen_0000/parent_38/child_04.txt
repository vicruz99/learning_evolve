import numpy as np
from scipy.optimize import minimize

def objective_func(v, n):
    """Objective: minimize negative sum of radii (equivalent to maximizing sum of radii)."""
    return -np.sum(v[2 * n:])

def constraint_func(v, n, triu_indices):
    """
    Inequality constraints: all values must be >= 0.
    1. Circle-Circle: dist_sq - (r_i + r_j)^2 >= 0
    2. Circle-Wall: x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
    """
    centers = v[:2 * n].reshape(n, 2)
    radii = np.maximum(v[2 * n:], 0.0)  # Ensure non-negative radii for stability

    # Circle-Circle constraints
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist_sq = np.sum(diff ** 2, axis=2)
    rad_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    cc_constraints = dist_sq - rad_sum ** 2
    
    # Extract upper triangle (i < j) to avoid duplicates and self-constraints
    cc_vals = cc_constraints[triu_indices]

    # Circle-Wall constraints
    cw_vals = np.concatenate([
        centers[:, 0] - radii,
        1.0 - centers[:, 0] - radii,
        centers[:, 1] - radii,
        1.0 - centers[:, 1] - radii
    ])

    return np.concatenate([cc_vals, cw_vals])

def generate_initial_guess(n, seed=42):
    """Generate a feasible initial guess using a hexagonal lattice pattern."""
    rng = np.random.default_rng(seed)
    points = []
    r_init = 0.04
    spacing = 0.18
    dy = spacing * np.sqrt(3) / 2
    
    # Generate hexagonal grid points within the square
    for i in range(10):
        for j in range(10):
            x = 0.08 + j * spacing + (i % 2) * spacing / 2
            y = 0.08 + i * dy
            if 0 <= x <= 1 and 0 <= y <= 1:
                points.append([x, y])
                
    points = np.array(points)
    # Shuffle and select n points
    indices = rng.permutation(len(points))[:n]
    centers = points[indices]
    radii = np.full(n, r_init)
    
    v = np.concatenate([centers.flatten(), radii])
    return v

def run_packing():
    n = 26
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * n
    triu_indices = np.triu_indices(n, k=1)
    
    constraints = {
        'type': 'ineq',
        'fun': constraint_func,
        'args': (n, triu_indices)
    }
    
    best_v = None
    best_sum_r = -1.0
    
    # Multi-start optimization
    num_trials = 5
    for trial in range(num_trials):
        v0 = generate_initial_guess(n, seed=42 + trial)
        # Add small perturbation to break symmetry and explore locally
        v0 += np.random.randn(v0.shape[0]) * 0.002
        v0 = np.clip(v0[:2 * n], 0.0, 1.0)
        v0[2 * n:] = np.clip(v0[2 * n:], 0.0, 0.5)
        
        res = minimize(
            objective_func, 
            v0, 
            args=(n,), 
            method='SLSQP', 
            bounds=bounds, 
            constraints=constraints,
            options={'maxiter': 800, 'ftol': 1e-10, 'disp': False}
        )
        
        if res.success:
            current_sum_r = -res.fun
            if current_sum_r > best_sum_r:
                best_sum_r = current_sum_r
                best_v = res.x.copy()
                
    # Fallback if all trials fail
    if best_v is None:
        v0 = generate_initial_guess(n)
        res = minimize(objective_func, v0, args=(n,), method='SLSQP', bounds=bounds, constraints=constraints, options={'maxiter': 1000})
        best_v = res.x
        
    centers = best_v[:2 * n].reshape(n, 2)
    radii = np.maximum(best_v[2 * n:], 0.0)
    
    # Final clamping to strictly satisfy boundaries
    centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
    centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)
    
    return centers, radii, float(np.sum(radii))
