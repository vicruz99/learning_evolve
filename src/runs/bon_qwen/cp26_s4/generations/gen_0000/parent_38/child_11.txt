import numpy as np
import scipy.optimize as opt

def run_packing():
    """
    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    """
    n_circles = 26
    rng = np.random.default_rng(42)
    
    # 1. Initialization: Perturbed 5x5 grid
    centers = []
    radii = []
    
    # Grid for 25 circles
    r_init = 0.09
    for i in range(5):
        for j in range(5):
            centers.append([r_init + i * 0.18 + rng.uniform(-0.01, 0.01), 
                            r_init + j * 0.18 + rng.uniform(-0.01, 0.01)])
            radii.append(r_init)
    
    # 26th circle in the center
    centers.append([0.5, 0.5])
    radii.append(0.05)
    
    centers = np.array(centers)
    radii = np.array(radii)

    def objective(x):
        # Maximize sum of radii = Minimize -sum(radii)
        return -np.sum(x[52:])

    def constraints(x):
        c_centers = x[:52].reshape(-1, 2)
        c_radii = x[52:]
        con_list = []
        
        # Boundary constraints (center +/- radius within [0,1])
        for i in range(n_circles):
            con_list.append({'type': 'ineq', 'fun': lambda x, i=i: x[i*2] - x[52+i]})
            con_list.append({'type': 'ineq', 'fun': lambda x, i=i: 1.0 - x[i*2] - x[52+i]})
            con_list.append({'type': 'ineq', 'fun': lambda x, i=i: x[i*2+1] - x[52+i]})
            con_list.append({'type': 'ineq', 'fun': lambda x, i=i: 1.0 - x[i*2+1] - x[52+i]})
            
        # Non-overlap constraints (distance >= r1 + r2)
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                con_list.append({
                    'type': 'ineq',
                    'fun': lambda x, i=i, j=j: np.linalg.norm(c_centers[i] - c_centers[j]) - (c_radii[i] + c_radii[j])
                })
        return con_list

    # Combine into flat vector
    x0 = np.concatenate([centers.flatten(), radii])
    cons = constraints(x0)
    
    # Bounds: coordinates [0,1], radii [0, 0.5]
    bounds = [(0, 1) for _ in range(52)] + [(0, 0.5) for _ in range(n_circles)]

    # Solve
    res = opt.minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, 
                       options={'maxiter': 1000, 'ftol': 1e-9})

    final_radii = res.x[52:]
    
    # Final validation
    validate_packing(final_centers, final_radii)
    return final_centers, final_radii, np.sum(final_radii)

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True
