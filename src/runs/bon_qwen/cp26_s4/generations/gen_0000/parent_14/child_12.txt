# Global constant for penalty weight
PENALTY_WEIGHT = 10000.0

def compute_objective(params, n):
    """
    Computes the objective function: -sum(radii) + penalty for violations.
    """
    radii = params[:n]
    centers_x = params[n:2*n]
    centers_y = params[2*n:3*n]
    
    # Objective term: negative sum of radii (since we minimize)
    obj = -np.sum(radii)
    
    # Penalty accumulation
    penalty = 0.0
    
    # 1. Boundary Constraints
    # Circle i must satisfy: r_i <= x_i <= 1 - r_i  and  r_i <= y_i <= 1 - r_i
    # Violations: x_i < r_i  =>  r_i - x_i > 0
    #             x_i > 1 - r_i => x_i + r_i - 1 > 0
    # Similarly for y.
    
    # Vectorized boundary penalties
    # r - x
    viol_x_left = radii - centers_x
    penalty += np.sum(np.maximum(0.0, viol_x_left)**2)
    
    # r + x - 1
    viol_x_right = radii + centers_x - 1.0
    penalty += np.sum(np.maximum(0.0, viol_x_right)**2)
    
    viol_y_bottom = radii - centers_y
    penalty += np.sum(np.maximum(0.0, viol_y_bottom)**2)
    
    viol_y_top = radii + centers_y - 1.0
    penalty += np.sum(np.maximum(0.0, viol_y_top)**2)
    
    # 2. Overlap Constraints
    # Distance between centers >= sum of radii
    # dist_ij >= r_i + r_j  =>  r_i + r_j - dist_ij <= 0
    # Violation: r_i + r_j - dist_ij > 0
    
    centers = np.column_stack((centers_x, centers_y))
    
    # Compute pairwise distances
    # Using broadcasting for O(N^2) calculation
    # diff shape: (n, n, 2)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist_matrix = np.sqrt(np.sum(diff**2, axis=2))
    
    # Set diagonal to infinity to avoid self-comparison issues (though max(0, -inf) is 0)
    np.fill_diagonal(dist_matrix, np.inf)
    
    # Sum of radii matrix
    r_sum_matrix = radii[:, np.newaxis] + radii[np.newaxis, :]
    
    # Violation amount: positive if overlapping
    violation = r_sum_matrix - dist_matrix
    
    # We only care about the upper triangle to avoid double counting, 
    # but summing all is just a constant factor (2x) in penalty, which is fine.
    # However, for correctness of gradient magnitude, we can take upper triangle.
    # But for simplicity and since weight is tunable, full sum is acceptable.
    # Let's just sum all positive violations.
    
    # Mask for positive violations
    pos_violation = np.maximum(0.0, violation)
    penalty += np.sum(pos_violation**2)
    
    return obj + PENALTY_WEIGHT * penalty
