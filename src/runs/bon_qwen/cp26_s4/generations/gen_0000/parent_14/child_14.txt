def calculate_overlap_penalty(centers, radii):
    """
    Calculates a smooth penalty for circle-circle overlaps.
    Uses squared distance constraint: d^2 >= (r1 + r2)^2.
    """
    n = centers.shape[0]
    penalty = 0.0
    
    # Vectorized calculation for efficiency
    # centers: (n, 2)
    # radii: (n,)
    
    # Compute all pairwise squared distances
    # diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :] # (n, n, 2)
    # dist_sq = np.sum(diff**2, axis=2) # (n, n)
    
    # This memory heavy approach for N=26 is fine, but let's do loops or triangular mask to be safe/clean
    # Actually for N=26, n^2 is 676, very small.
    
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist_sq = np.sum(diff**2, axis=2)
    
    # Compute (r_i + r_j)^2
    r_sum_sq = (radii[:, np.newaxis] + radii[np.newaxis, :])**2
    
    # Violation: if (r_i + r_j)^2 > dist_sq, we have overlap
    # We want to penalize this.
    # Violation amount = (r_sum_sq - dist_sq)
    # Only if positive.
    
    violations = r_sum_sq - dist_sq
    # Mask for lower triangle to avoid double counting and self
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    
    relevant_violations = violations[mask]
    
    # Penalty is sum of squared violations
    # Using a hinge function: max(0, v)^2
    penalty = np.sum(np.maximum(0, relevant_violations)**2)
    
    return penalty

    penalty += np.sum(np.maximum(0, v3)**2)
    penalty += np.sum(np.maximum(0, v4)**2)
    
    return penalty
