import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def objective(x):
    """Objective function: maximize radius (minimize negative radius)"""
    return -x[-1]

def compute_constraints(x):
    """
    Compute inequality constraints:
    - Boundary constraints: circles must be inside [0,1]x[0,1]
    - Non-overlap constraints: distance between any pair >= 2*r
    Returns an array of constraint values, all should be >= 0.
    """
    r = x[-1]
    c = x[:-1].reshape(N_CIRCLES, 2)
    
    # Boundary constraints (4 per circle)
    bc = np.concatenate([
        c[:, 0] - r,            # x - r >= 0
        1 - c[:, 0] - r,       # 1 - x - r >= 0
        c[:, 1] - r,            # y - r >= 0
        1 - c[:, 1] - r        # 1 - y - r >= 0
    ])
    
    # Pairwise non-overlap constraints
    # Compute all pairwise distances efficiently
    diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]  # Shape (N, N, 2)
    dists = np.sqrt(np.sum(diff**2, axis=2))           # Shape (N, N)
    # Extract upper triangle indices (unique pairs)
    upper = dists[np.triu_indices(N_CIRCLES, k=1)]
    pc = upper - 2 * r
    
    return np.concatenate([bc, pc])

def run_packing():
    # Initial configuration: 5x5 grid + 1 circle in the center gap
    centers = np.zeros((N_CIRCLES, 2))
    idx = 0
    for i in range(5):
        for j in range(5):
            centers[idx] = [0.1 + 0.2 * i, 0.1 + 0.2 * j]
            idx += 1
    # 26th circle placed in the largest gap of the grid
    centers[25] = [0.4, 0.4]
    
    # Initial radius safe for this configuration
    r0 = 0.07
    x0 = np.concatenate([centers.flatten(), [r0]])
    
    # Bounds: centers in [0,1], radius in [0, 0.5]
    bounds = [(0.0, 1.0) for _ in range(2 * N_CIRCLES)] + [(0.0, 0.5)]
    cons = {'type': 'ineq', 'fun': compute_constraints}
    
    # Optimize
    res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': 3000, 'ftol': 1e-12, 'disp': False})
                   
    final_r = res.x[-1]
    final_c = res.x[:-1].reshape(N_CIRCLES, 2)
    radii = np.full(N_CIRCLES, final_r)
    
    return final_c, radii, np.sum(radii)
