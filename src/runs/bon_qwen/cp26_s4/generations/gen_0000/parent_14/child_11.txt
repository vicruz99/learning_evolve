import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # 1. Initialization: Grid-based starting point
    # We create a grid of points and select the first 26
    x_vals = np.linspace(0.1, 0.9, 6)
    y_vals = np.linspace(0.1, 0.9, 5)
    
    init_centers = []
    idx = 0
    for y in y_vals:
        for x in x_vals:
            if idx < n:
                init_centers.append([x, y, 0.04]) # Small initial radius
                idx += 1
    
    init_centers = np.array(init_centers)
    x0 = init_centers.flatten()
    
    # 2. Objective Function
    def objective(v):
        # We want to maximize sum of radii, so we minimize negative sum
        radii = v[2::3]
        return -np.sum(radii)

    # 3. Constraints
    # Inequality constraints: g(v) >= 0
    
    def boundary_constraints(v):
        # x - r >= 0, 1 - x - r >= 0
        # y - r >= 0, 1 - y - r >= 0
        centers = v.reshape(-1, 3)
        c = []
        for i in range(n):
            x, y, r = centers[i]
            c.append(x - r)
            c.append(1 - x - r)
            c.append(y - r)
            c.append(1 - y - r)
        return np.array(c)

    def overlap_constraints(v):
        centers = v.reshape(-1, 3)
        c = []
        for i in range(n):
            for j in range(i + 1, n):
                xi, yi, ri = centers[i]
                xj, yj, rj = centers[j]
                dist_sq = (xi - xj)**2 + (yi - yj)**2
                min_dist_sq = (ri + rj)**2
                c.append(dist_sq - min_dist_sq)
        return np.array(c)

    # Combine constraints
    def all_constraints(v):
        bc = boundary_constraints(v)
        oc = overlap_constraints(v)
        return np.concatenate((bc, oc))

    # 4. Bounds
    # x, y in [0, 1], r in [0, 1]
    bounds = []
    for i in range(n):
        bounds.append((0, 1)) # x
        bounds.append((0, 1)) # y
        bounds.append((0, 1)) # r

    # 5. Run Optimization
    cons = {'type': 'ineq', 'fun': all_constraints}
    
    # We run the optimizer. 
    # Note: SLSQP is a local optimizer, so the result depends on x0.
    # We use a robust method and multiple restarts internally if needed, 
    # but here we rely on the grid start being close enough.
    
    res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, 
                   options={'ftol': 1e-9, 'maxiter': 1000})
    
    if res.success:
        centers = res.x.reshape(-1, 3)[:, :2]
        radii = res.x.reshape(-1, 3)[:, 2]

        # Better: Use a slightly smaller radius for all to be safe.
        r_fallback = 0.09
        radii_fallback = np.full(n, r_fallback)
        
        # Adjust center of 26th circle to (0.5, 0.9) ? No, occupied.
        # Let's just use the grid result from res.x if it's not terrible, 
        # otherwise use the fallback.
        centers_fallback = centers_fallback[:n]
        # To ensure validity of fallback:
        # 5x5 grid points: x in {0.1, 0.3, 0.5, 0.7, 0.9}, y same.

        radii = radii_fallback

# Validation function provided in prompt
def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square

    Args:
        centers: np.array of shape (n, 2) with (x, y) coordinates
        radii: np.array of shape (n) with radius of each circle

    Returns:
        True if valid, False otherwise
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True
