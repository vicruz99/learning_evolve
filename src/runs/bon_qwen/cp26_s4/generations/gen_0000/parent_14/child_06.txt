def compute_energy_and_grad(centers, radii):
    """
    Computes the overlap energy and its gradient with respect to circle centers.
    Energy penalizes pairwise overlaps and boundary violations.
    """
    n = centers.shape[0]
    energy = 0.0
    grad = np.zeros_like(centers)
    
    # Boundary penalties
    for i in range(n):
        for dim in range(2):
            val = centers[i, dim]
            r = radii[i]
            if val < r:
                pen = r - val
                energy += pen**2
                grad[i, dim] += 2.0 * pen
            if val > 1.0 - r:
                pen = r - (1.0 - val)
                energy += pen**2
                grad[i, dim] -= 2.0 * pen

    # Pairwise overlap penalties
    for i in range(n):
        for j in range(i + 1, n):
            diff = centers[i] - centers[j]
            dist = np.hypot(diff[0], diff[1])
            if dist < 1e-12:
                dist = 1e-12
                diff = np.array([1e-12, 0.0])
            overlap = radii[i] + radii[j] - dist
            if overlap > 0:
                energy += overlap**2
                force = 2.0 * overlap * diff / dist
                grad[i] += force
                grad[j] -= force
                
    return energy, grad
