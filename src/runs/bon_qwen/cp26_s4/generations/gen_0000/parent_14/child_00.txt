

<strategy>
For packing 26 circles in a unit square to maximize sum of radii, I'll use a multi-stage approach:

1. **Initial Configuration**: Create several hexagonal-grid-based initial arrangements since hexagonal packing is the densest circle packing in 2D. I'll try different row configurations (5,4,5,4,4,4 = 26 circles, 6,5,6,5,4 = 26, etc.)

2. **Force-Based Pre-optimization**: Use a force-directed simulation where:
   - Overlapping circles repel each other
   - Boundary forces push circles inward
   - A growth pressure pushes radii to expand
   This quickly finds a reasonable jammed configuration.

3. **Refinement with scipy.optimize**: Use L-BFGS-B with a smooth penalty function. The penalty function penalizes overlaps and boundary violations. I'll use a continuation method (gradually increasing penalty weight) for better convergence.

4. **Multiple Restarts**: Try different initial configurations and pick the best result.

5. **Validation & Post-processing**: Ensure the final packing is valid by slightly reducing radii if needed.

Key insight: By allowing variable radii, we can fill gaps between larger circles with smaller ones, achieving higher total sum than equal-circle packing.
</strategy>

import numpy as np

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    best_centers = None
    best_radii = None
    best_sum = 0.0
    
    # Try multiple initial configurations
    init_configs = [
        make_hex_config_1(n),
        make_hex_config_2(n),
        make_hex_config_3(n),
        make_grid_config(n),
        make_random_config(n),
    ]
    
    for centers, radii in init_configs:
        result = force_optimize(centers.copy(), radii.copy(), n, steps=30000)
        centers_f = result['centers']
        radii_f = result['radii']
        
        # Refine with scipy
        result2 = scipy_refine(centers_f, radii_f, n)
        centers_r = result2['centers']
        radii_r = result2['radii']
        
        # Validate and fix
        valid, scale = check_validity(centers_r, radii_r, n)
        if valid:
            cur_sum = np.sum(radii_r)
        else:
            radii_r = radii_r * scale
            cur_sum = np.sum(radii_r)
        
        if cur_sum > best_sum:
            best_sum = cur_sum
            best_centers = centers_r.copy()
            best_radii = radii_r.copy()
    
    # Final validation pass
    valid, scale = check_validity(best_centers, best_radii, n)
    if not valid:
        best_radii *= scale
    
    best_sum = np.sum(best_radii)
    
    return best_centers, best_radii, best_sum


def make_hex_config_1(n):
    """5,4,5,4,4,4 arrangement"""
    row_counts = [5, 4, 5, 4, 4, 4]
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.08)
    idx = 0
    for row in range(6):
        cnt = row_counts[row]
        y = 0.12 + row * 0.15
        offset = 0.07 if row % 2 == 1 else 0.0
        x_start = 0.1 + offset
        x_end = 0.9 - offset
        for col in range(cnt):
            x = x_start + (x_end - x_start) * col / (cnt - 1) if cnt > 1 else 0.5
            centers[idx] = [x, y]
            idx += 1
    return centers, radii

def make_grid_config(n):
    """Grid-based initial config"""
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.08)
    cols = 5
    idx = 0
    for row in range(6):
        for col in range(cols):
            if idx >= n:
                break
            x = 0.1 + col * 0.2
            y = 0.12 + row * 0.15
            if row % 2 == 1:
                x += 0.1
            centers[idx] = [x, y]
            idx += 1
    return centers, radii


def make_random_config(n):
    """Random initial config"""
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.05)
    for i in range(n):
        centers[i, 0] = 0.1 + np.random.random() * 0.8
        centers[i, 1] = 0.1 + np.random.random() * 0.8
    return centers, radii


def force_optimize(centers, radii, n, steps=30000):
    """Force-based optimization for circle packing."""
    lr = 0.003
    lr_decay = 0.9998
    
    for step in range(steps):
        forces = np.zeros_like(centers)
        radius_forces = np.zeros(n)
        
        current_lr = lr * (lr_decay ** step)
        
        # Forces between circles
        for i in range(n):
            for j in range(i + 1, n):
                diff = centers[i] - centers[j]
                dist_sq = np.sum(diff ** 2)
                dist = np.sqrt(dist_sq)
                if dist < 1e-12:
                    dist = 1e-12
                    diff = np.array([1e-6, 0.0])
                    dist_sq = 1e-12
                
                min_dist = radii[i] + radii[j]
                
                if dist < min_dist * 1.5:
                    if dist < min_dist:
                        overlap = min_dist - dist
                        force_mag = overlap * 200.0 / (dist + 0.01)
                    else:
                        force_mag = 0.05 * (min_dist * 1.5 - dist) / (dist + 0.01)
                    
                    direction = diff / dist
                    forces[i] += direction * force_mag * current_lr
                    forces[j] -= direction * force_mag * current_lr
                    
                    if dist < min_dist:
                        shrink = overlap * 10.0 * current_lr
                        radius_forces[i] -= shrink
                        radius_forces[j] -= shrink
        
        # Boundary forces
        for i in range(n):
            for k in range(2):
                d_left = centers[i, k] - radii[i]
                if d_left < 0:
                    forces[i, k] += d_left * 100.0 * current_lr
                    radius_forces[i] -= (-d_left) * 30.0 * current_lr
                
                d_right = 1.0 - centers[i, k] - radii[i]
                if d_right < 0:
                    forces[i, k] -= d_right * 100.0 * current_lr
                    radius_forces[i] -= (-d_right) * 30.0 * current_lr
        
        # Growth pressure
        for i in range(n):
            radius_forces[i] += 2.0 * current_lr
        
        # Update positions and radii
        centers += forces
        radii += radius_forces
        
        # Enforce constraints
        radii = np.maximum(radii, 0.002)
        for i in range(n):
            for k in range(2):
                centers[i, k] = max(radii[i] + 0.001, min(1.0 - radii[i] - 0.001, centers[i, k]))
    
    return {'centers': centers, 'radii': radii}


def scipy_refine(centers, radii, n):
    """Refine packing using scipy L-BFGS-B with penalty method."""
    
    def penalty_obj(vars, lam):
        c = vars[:2 * n].reshape(n, 2)
        r = vars[2 * n:]
        
        pen = 0.0
        # Overlap penalty
        for i in range(n):
            for j in range(i + 1, n):
                d = np.sqrt(np.sum((c[i] - c[j]) ** 2))
                ov = r[i] + r[j] - d
                if ov > 0:
                    pen += lam * ov * ov
        # Boundary penalty
        for i in range(n):
            for k in range(2):
                v1 = c[i, k] - r[i]
                v2 = c[i, k] + r[i] - 1.0
                if v1 < 0:
                    pen += lam * v1 * v1
                if v2 > 0:
                    pen += lam * v2 * v2
        
        return -np.sum(r) + pen
    
    x0 = np.concatenate([centers.flatten(), radii])
    
    # Continuation method
    for lam in [50, 200, 1000, 5000, 20000, 100000]:
        bounds = [(0.001, 0.999)] * (2 * n) + [(0.001, 0.5)] * n
        
        try:
            res = minimize(
                penalty_obj,
                x0,
                args=(lam,),
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 2000, 'ftol': 1e-15, 'gtol': 1e-10}
            )
            x0 = res.x
        except Exception:
            pass
    
    centers_opt = x0[:2 * n].reshape(n, 2)
    radii_opt = x0[2 * n:]
    
    return {'centers': centers_opt, 'radii': radii_opt}


def check_validity(centers, radii, n):
    """Check if packing is valid, return (valid, scale_factor)."""
    eps = 1e-12
    
    # Check NaN
    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, 0.5
    
    # Check non-negative radii
    if np.any(radii < 0):
        return False, 0.5
    
    scale = 1.0
    
    # Check boundaries
    for i in range(n):
        for k in range(2):
            if centers[i, k] - radii[i] < -eps:
                needed = (eps - (centers[i, k] - radii[i])) / radii[i] if radii[i] > 0 else 1.0
                scale = min(scale, max(0.5, 1.0 - needed))
            if centers[i, k] + radii[i] > 1.0 + eps:
                needed = (centers[i, k] + radii[i] - 1.0 - eps) / radii[i] if radii[i] > 0 else 1.0
                scale = min(scale, max(0.5, 1.0 - needed))
    
    # Check overlaps
    for i in range(n):
        for j in range(i + 1, n):
            d = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            sum_r = radii[i] + radii[j]
            if d < sum_r - eps:
                needed = (sum_r - eps - d) / sum_r if sum_r > 0 else 1.0
                scale = min(scale, max(0.5, 1.0 - needed))
    
    valid = scale >= 1.0 - eps
    return valid, scale
