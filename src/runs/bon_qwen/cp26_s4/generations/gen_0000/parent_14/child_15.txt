import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    Pack 26 circles in a unit square to maximize the sum of radii.
    """
    n_circles = 26

    # We aim for an average radius around 0.1.
    # A hexagonal lattice is denser than a square grid.
    # Let's try to fit 26 circles in a staggered arrangement.
    # We can arrange them in 5 rows.
    # Pattern: 5, 5, 5, 5, 6 circles per row?
    # However, fitting 6 circles in a row requires r <= 1/12 approx 0.083, which is small.
    # Let's try a pattern that balances row lengths: 6, 5, 6, 5, 4? Or 5, 6, 5, 6, 4?
    # Actually, let's just generate a hex grid and pick the first 26 valid points.

    # Hexagonal spacing
    dx = 2 * r_init
    dy = np.sqrt(3) * r_init
    
    centers_init = []
    
    # Generate a grid of potential centers
    # We'll iterate through rows and columns
    for row in range(10): # Enough rows
        y = r_init + row * dy
        if y + r_init > 1.0 + 1e-6:
            break
        
        # Determine starting x offset for the row
        # Even rows (0, 2...) start at r_init
        # Odd rows (1, 3...) shifted by r_init
        x_start = r_init + (row % 2) * r_init
        
        for col in range(10): # Enough cols
            x = x_start + col * dx
            if x + r_init > 1.0 + 1e-6:
                break
            
            centers_init.append([x, y])
            
            if len(centers_init) >= n_circles:
                break
        if len(centers_init) >= n_circles:
            break
            
    # If we didn't get enough (unlikely with this logic), pad or resize
    # But with r=0.1, we should get plenty.
    # Let's verify we have 26.
    # If we have more, take the first 26.
    # If fewer (due to boundary), we might need to adjust r_init, but 0.1 is safe for generating points.
    
    centers_init = np.array(centers_init[:n_circles])
    
    # If we have fewer than 26, we have a problem with generation, but let's assume it works.
    # Just in case, fill remaining with random valid points if needed (not expected here)
    while len(centers_init) < n_circles:
        centers_init = np.vstack([centers_init, [0.5, 0.5]]) # Fallback
        
    # Initial radii
    radii_init = np.full(n_circles, r_init)
    
    # 2. Optimization
    # Variables: [x1, y1, r1, x2, y2, r2, ...]
    # Total variables: 3 * n_circles
    
    def objective(vars):
        # We want to maximize sum of radii, so minimize negative sum
        radii = vars[2::3]
        return -np.sum(radii)
    
    def constraints_boundary(vars):
        # Returns values that must be >= 0
        cons = []
        centers = vars[:2*n_circles].reshape(n_circles, 2)
        radii = vars[2::3]
        
        for i in range(n_circles):
            x, y = centers[i]
            r = radii[i]
            # x - r >= 0  => x - r
            # x + r <= 1  => 1 - (x + r)
            # y - r >= 0  => y - r
            # y + r <= 1  => 1 - (y + r)
            cons.append(x - r)
            cons.append(1.0 - (x + r))
            cons.append(y - r)
            cons.append(1.0 - (y + r))
            # r >= 0
            cons.append(r)
        return cons

    def constraints_overlap(vars):
        # Returns values that must be >= 0
        # dist^2 >= (r1 + r2)^2  => dist^2 - (r1+r2)^2 >= 0
        cons = []
        centers = vars[:2*n_circles].reshape(n_circles, 2)
        radii = vars[2::3]
        
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                c1 = centers[i]
                c2 = centers[j]
                r1 = radii[i]
                r2 = radii[j]
                
                dist_sq = (c1[0] - c2[0])**2 + (c1[1] - c2[1])**2

                # Constraint: dist_sq - sum_r^2 >= 0
                cons.append(dist_sq - sum_r**2)
        return cons

    # Combine constraints
    # scipy.optimize requires constraints to be a list of dictionaries

    bounds = [(None, None)] * (2 * n_circles) + [(0, None)] * n_circles

        c_overlap = constraints_overlap(vars)
        return np.concatenate([c_boundary, c_overlap])

    constraint = {
        'type': 'ineq',
        'fun': all_constraints,
        'jac': '2-point' # Use numerical Jacobian for simplicity, or provide analytical if needed. 
                         # For 78 vars, 700+ constraints, analytical is complex. '2-point' is safe.
    }

    # Run optimization
    # Method 'SLSQP' is suitable for constrained optimization
    res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraint, 
                   options={'ftol': 1e-9, 'maxiter': 500, 'disp': False})
    
    # Extract results
    final_vars = res.x
    final_centers = final_vars[:2*n_circles].reshape(n_circles, 2)
    final_radii = final_vars[2::3]
    
    sum_radii = np.sum(final_radii)
    
    return final_centers, final_radii, sum_radii
