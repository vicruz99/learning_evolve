from scipy.optimize import minimize

def run_packing():

        # Objective: maximize sum of radii -> minimize negative sum
        r = params[2::3]
        return -np.sum(r)

        g = np.zeros_like(params)
        g[2::3] = -1.0
        return g

    def constraints_overlap(params, n):
        cons = []
        centers = np.column_stack((params[0::3], params[1::3]))
        radii = params[2::3]
        
        for i in range(n):

                c2 = centers[j]

                r2 = radii[j]

                    ci = p[0::3]
                    cj = p[1::3]
                    ri = p[2::3]
                    # Squared distance - (sum of radii)^2 >= 0
                    # We use a soft margin to avoid 0 gradient issues when not touching, 
                    # but SLSQP handles active constraints better if we stick to strict geometry.
                    # d^2 - (r1+r2)^2
                    # To be safe with numerical noise, we just use the exact distance constraint.
                    dist_sq = (ci[i] - ci[j])**2 + (cj[i] - cj[j])**2
                    return dist_sq - (ri[i] + ri[j])**2

                def jac(p):
                    ci = p[0::3]
                    cj = p[1::3]
                    ri = p[2::3]
                    g = np.zeros_like(p)
                    # Gradient w.r.t xi: 2(xi - xj)
                    g[3*i] = 2.0 * (ci[i] - ci[j])
                    g[3*i + 1] = 2.0 * (cj[i] - cj[j])
                    g[3*i + 2] = -2.0 * (ri[i] + ri[j])
                    
                    # Gradient w.r.t xj: 2(xj - xi)
                    g[3*j] = 2.0 * (ci[j] - ci[i])
                    g[3*j + 1] = 2.0 * (cj[j] - cj[i])
                    g[3*j + 2] = -2.0 * (ri[i] + ri[j])
                    return g

                cons.append({'type': 'ineq', 'fun': fun, 'jac': jac})
        return cons

    def constraints_boundary(params, n):

        # x - r >= 0
        # x + r <= 1 => 1 - x - r >= 0
        # y - r >= 0
        # y + r <= 1 => 1 - y - r >= 0

            xi = idx
            yi = idx + 1
            ri = idx + 2

                return p[3*i] - p[3*i + 2]
            def jac_xr(p, i=i, idx=idx):
                g = np.zeros_like(p)
                g[3*i] = 1.0
                g[3*i + 2] = -1.0
                return g
            cons.append({'type': 'ineq', 'fun': fun_xr, 'jac': jac_xr})

            def fun_1xr(p, i=i, idx=idx):
                return 1.0 - p[3*i] - p[3*i + 2]
            def jac_1xr(p, i=i, idx=idx):
                g = np.zeros_like(p)
                g[3*i] = -1.0
                g[3*i + 2] = -1.0
                return g
            cons.append({'type': 'ineq', 'fun': fun_1xr, 'jac': jac_1xr})

            def fun_yr(p, i=i, idx=idx):
                return p[3*i + 1] - p[3*i + 2]
            def jac_yr(p, i=i, idx=idx):
                g = np.zeros_like(p)
                g[3*i + 1] = 1.0
                g[3*i + 2] = -1.0
                return g
            cons.append({'type': 'ineq', 'fun': fun_yr, 'jac': jac_yr})

                g = np.zeros_like(p)
                g[3*i + 1] = -1.0
                g[3*i + 2] = -1.0

            y += 1.5 * r
            # Check if next row fits vertically

        return np.array(points)
