import numpy as np
from scipy.optimize import minimize

def _compute_loss_grad(params, n, w_ov, w_bd):
    """
    Compute objective (negative sum of radii) and gradient for the packing problem.
    Uses quadratic penalty terms for boundary and overlap constraints.
    """
    pts = params.reshape(n, 3)
    obj = -np.sum(pts[:, 2])
    grad = np.zeros_like(params)
    grad[:, 2] = -1.0  # Derivative of -sum(r) w.r.t r

    # Boundary constraints: r <= x, r <= 1-x, r <= y, r <= 1-y
    for i in range(n):
        x, y, r = pts[i]
        for val, idx_v, idx_c, sign in [(r - x, 2, 0, -1), 
                                         (r + x - 1, 2, 0, 1), 
                                         (r - y, 2, 1, -1), 
                                         (r + y - 1, 2, 1, 1)]:
            if val > 0:
                obj += w_bd * val**2
                term = 2 * w_bd * val
                grad[i, idx_v] += term
                grad[i, idx_c] -= term * sign

    # Overlap constraints: dist >= r_i + r_j
    for i in range(n):
        for j in range(i + 1, n):
            dx = pts[i, 0] - pts[j, 0]
            dy = pts[i, 1] - pts[j, 1]
            d = np.hypot(dx, dy)
            if d < 1e-12:
                d = 1e-12
            ov = pts[i, 2] + pts[j, 2] - d
            if ov > 0:
                obj += w_ov * ov**2
                term = 2 * w_ov * ov
                grad[i, 2] += term
                grad[j, 2] += term
                fx = dx / d
                fy = dy / d
                grad[i, 0] -= term * fx
                grad[i, 1] -= term * fy
                grad[j, 0] += term * fx
                grad[j, 1] += term * fy

    return obj, grad.flatten()

def _optimize_packing(n, init_params):
    """Run optimization with increasing penalty weights."""
    bounds = []
    for _ in range(n):
        bounds.extend([(0.0, 1.0), (0.0, 1.0), (1e-6, 0.5)])

    w_ov, w_bd = 100.0, 100.0
    params = init_params.copy()

    for _ in range(20):
        res = minimize(_compute_loss_grad, params, args=(n, w_ov, w_bd),
                       method='L-BFGS-B', bounds=bounds, jac=True,
                       options={'maxiter': 500, 'ftol': 1e-12})
        params = res.x
        w_ov *= 1.3
        w_bd *= 1.3

    return params

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    best_params = None
    best_sum = -np.inf

    # Initialization 1: 5x5 Grid + Center circle
    x = np.linspace(0.1, 0.9, 5)
    y = np.linspace(0.1, 0.9, 5)
    cx, cy = np.meshgrid(x, y)
    c1 = np.column_stack([cx.ravel(), cy.ravel(), np.full(25, 0.09)])
    c1 = np.vstack([c1, [0.5, 0.5, 0.02]])
    p1 = _optimize_packing(n, c1.flatten())
    s1 = -p1.reshape(n, 3)[:, 2].sum()
    if s1 > best_sum:
        best_sum = s1
        best_params = p1

    # Initialization 2: Hexagonal Lattice
    c2 = np.full((n, 3), 0.0)
    idx = 0
    r_est = 0.085
    for row in range(7):
        y_c = 0.08 + row * r_est * np.sqrt(3)
        if y_c > 0.95 or idx >= n:
            break
        n_cols = 6 if row % 2 == 0 else 5
        for col in range(n_cols):
            if idx >= n:
                break
            x_c = 0.08 + col * 2*r_est + (0.5 if row%2==1 else 0) * 2*r_est
            if x_c > 0.95:
                break
            c2[idx] = [x_c, y_c, r_est]
            idx += 1
    # Fill any remaining slots
    for k in range(idx, n):
        c2[k] = [0.5, 0.5, 0.02]

    p2 = _optimize_packing(n, c2.flatten())
    s2 = -p2.reshape(n, 3)[:, 2].sum()
    if s2 > best_sum:
        best_sum = s2
        best_params = p2

    final_c = best_params.reshape(n, 3)
    centers = final_c[:, :2]
    radii = final_c[:, 2]
    return centers, radii, np.sum(radii)
