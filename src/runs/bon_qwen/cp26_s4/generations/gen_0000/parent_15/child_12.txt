import numpy as np
from scipy.optimize import minimize

N = 26

def _get_vars(centers, radii):
    v = np.zeros(3*N)
    for i in range(N):
        v[3*i] = centers[i, 0]
        v[3*i+1] = centers[i, 1]
        v[3*i+2] = radii[i]
    return v

def _set_vars(v):
    centers = np.zeros((N, 2))
    radii = np.zeros(N)
    for i in range(N):
        centers[i, 0] = v[3*i]
        centers[i, 1] = v[3*i+1]
        radii[i] = v[3*i+2]
    return centers, radii

def _obj(v):
    return -v[2::3].sum()

def _cons_bound(v):
    c = np.empty(4*N)
    for i in range(N):
        c[4*i] = v[3*i] - v[3*i+2]
        c[4*i+1] = 1.0 - v[3*i] - v[3*i+2]
        c[4*i+2] = v[3*i+1] - v[3*i+2]
        c[4*i+3] = 1.0 - v[3*i+1] - v[3*i+2]
    return c

def _cons_pair(v):
    c = np.empty(N*(N-1)//2)
    k = 0
    for i in range(N):
        for j in range(i+1, N):
            dx = v[3*i] - v[3*j]
            dy = v[3*i+1] - v[3*j+1]
            dr = v[3*i+2] + v[3*j+2]
            c[k] = dx*dx + dy*dy - dr*dr
            k += 1
    return c

def run_packing():
    # Initial feasible configuration: 5x5 grid + 1 center point
    centers = np.zeros((N, 2))
    idx = 0
    for i in range(5):
        for j in range(5):
            if idx >= N: break
            centers[idx, 0] = 0.1 + j * 0.2
            centers[idx, 1] = 0.1 + i * 0.2
            idx += 1
    if N > 25:
        centers[25, 0] = 0.5
        centers[25, 1] = 0.5
        
    radii = np.full(N, 0.06)
    v0 = _get_vars(centers, radii)
    
    bounds = [(0.0, 1.0)] * (3*N)
    for i in range(N):
        bounds[3*i+2] = (0.0, 0.5)
        
    cons = [
        {'type': 'ineq', 'fun': _cons_bound},
        {'type': 'ineq', 'fun': _cons_pair}
    ]
    
    best_v = v0.copy()
    best_val = -v0[2::3].sum()
    
    # Multiple restarts to escape local minima
    for seed in range(8):
        np.random.seed(42 + seed)
        v_cur = best_v.copy()
        if seed > 0:
            pos_idx = np.concatenate([np.arange(0, 3*N, 3), np.arange(1, 3*N, 3)])
            v_cur[pos_idx] += np.random.uniform(-0.03, 0.03, size=len(pos_idx))
            v_cur[pos_idx] = np.clip(v_cur[pos_idx], 0.05, 0.95)
            
        res = minimize(_obj, v_cur, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 5000, 'ftol': 1e-9, 'disp': False})
        
        if -res.fun > best_val:
            best_val = -res.fun
            best_v = res.x.copy()
            
    centers_opt, radii_opt = _set_vars(best_v)
    return centers_opt, radii_opt, np.sum(radii_opt)
