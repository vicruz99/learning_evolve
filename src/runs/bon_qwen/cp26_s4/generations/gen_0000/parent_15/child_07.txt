import numpy as np
from scipy.optimize import minimize

def compute_constraints(vars_flat):
    """
    Computes constraint violations for circle packing.
    Returns an array where all values should be >= 0 for a valid packing.
    Structure of vars_flat: [x1, y1, r1, x2, y2, r2, ...]
    """
    n = 26
    pts = vars_flat.reshape(n, 3)
    x = pts[:, 0]
    y = pts[:, 1]
    r = pts[:, 2]
    
    # Preallocate constraint array
    # 4 walls * n + n*(n-1)/2 pairs
    n_pairs = n * (n - 1) // 2
    cons = np.empty(4 * n + n_pairs)
    
    # Wall constraints: x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
    cons[:n] = x - r
    cons[n:2*n] = 1.0 - x - r
    cons[2*n:3*n] = y - r
    cons[3*n:4*n] = 1.0 - y - r
    
    # Pairwise non-overlap: dist^2 - (r_i + r_j)^2 >= 0
    idx = 4 * n
    for i in range(n):
        for j in range(i + 1, n):
            dx = x[i] - x[j]
            dy = y[i] - y[j]
            min_d = r[i] + r[j]
            cons[idx] = dx * dx + dy * dy - min_d * min_d
            idx += 1
            
    return cons

def objective_fn(vars_flat):
    """Objective: minimize negative sum of radii (equivalent to maximizing sum of radii)"""
    # Radii are at indices 2, 5, 8, ...
    radii = vars_flat[2::3]
    return -np.sum(radii)

def run_packing():
    n = 26
    np.random.seed(42)
    
    # 1. Initialize with a dense hexagonal-like grid
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.08)
    idx = 0
    row = 0
    col = 0
    while idx < n:
        cx = 0.05 + col * 0.18
        cy = 0.05 + row * 0.155
        if cx <= 0.95 and cy <= 0.95:
            centers[idx] = [cx, cy]
            idx += 1
        col += 1
        if col > 7:
            col = 0 if row % 2 == 0 else 1
            row += 1
            
    # 2. Force-directed simulation to find a high-density configuration
    dt = 0.06
    dr = 0.0004
    force_k = 18.0
    
    for step in range(2500):
        forces = np.zeros_like(centers)
        overlap = False
        
        # Pairwise repulsive forces
        for i in range(n):
            for j in range(i + 1, n):
                diff = centers[i] - centers[j]
                dist = np.linalg.norm(diff)
                md = radii[i] + radii[j]
                if dist < md and dist > 1e-6:
                    f = (md - dist) / dist * diff * force_k
                    forces[i] += f
                    forces[j] -= f
                    overlap = True
                    
        # Wall repulsive forces
        for i in range(n):
            cx, cy = centers[i]
            r = radii[i]
            if cx < r: forces[i, 0] += (r - cx) * force_k * 2
            elif cx > 1 - r: forces[i, 0] -= (cx - (1 - r)) * force_k * 2
            if cy < r: forces[i, 1] += (r - cy) * force_k * 2
            elif cy > 1 - r: forces[i, 1] -= (cy - (1 - r)) * force_k * 2
            
        centers += dt * forces
        centers = np.clip(centers, 0.0, 1.0)
        
        # Adaptive radius growth
        if not overlap:
            radii += dr
            if step % 40 == 0:
                centers += np.random.normal(0, 0.002, centers.shape)
                centers = np.clip(centers, 0.0, 1.0)
        else:
            dr *= 0.95
            dt *= 0.95
            
        if dr < 1e-6:
            break
            
    # 3. Prepare variables for the local optimizer
    vars_init = np.empty(n * 3)
    for i in range(n):
        vars_init[3*i] = centers[i, 0]
        vars_init[3*i+1] = centers[i, 1]
        vars_init[3*i+2] = radii[i]
        
    # Quick feasibility fix if simulation left slight violations
    cons_val = compute_constraints(vars_init)
    if np.min(cons_val) < -1e-4:
        radii_arr = vars_init[2::3]
        # Scale down radii just enough to clear worst violation
        scale = 1.0 + np.min(cons_val) / 0.01
        vars_init[2::3] = radii_arr * max(0.5, scale)
        
    # 4. Local optimization with SLSQP
    bounds = [(0, 1)] * (2*n) + [(0, 0.5)] * n
    constraints = {'type': 'ineq', 'fun': compute_constraints}
    
    res = minimize(objective_fn, vars_init, method='SLSQP', bounds=bounds, 
                   constraints=constraints, options={'ftol': 1e-9, 'maxiter': 1500})
                   
    final_vars = res.x
    final_centers = np.column_stack((final_vars[0::3], final_vars[1::3]))
    final_radii = final_vars[2::3]
    
    return final_centers, final_radii, np.sum(final_radii)
