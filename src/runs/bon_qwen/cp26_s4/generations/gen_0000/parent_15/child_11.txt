import numpy as np
from scipy.optimize import minimize

def run_packing():
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    Returns (centers, radii, sum_radii).
    """
    n = 26
    
    # 1. Initialization: Hexagonal-like grid
    # We place circles in rows. Odd rows are shifted to create a staggered pattern.
    # This mimics hexagonal packing, which is denser than square packing.
    centers = []
    # Parameters for initialization
    start_r = 0.05
    # Grid dimensions
    # We want to cover the square. Let's try 6 rows.
    # Vertical spacing roughly sqrt(3) * 2 * start_r ~ 0.17, but let's use 0.15 to spread out initially
    y_spacing = 0.15
    x_spacing = 0.20
    
    row_y = 0.05 + start_r
    row_idx = 0
    
    while len(centers) < n:
        # Shift for odd rows to stagger
        shift = 0.10 if row_idx % 2 == 1 else 0.0
        
        col_x = 0.05 + start_r + shift
        while col_x <= 1.0 - (0.05 + start_r):
            # Check if adding this point keeps us within reasonable bounds
            # We just collect points, bounds are handled by optimizer later, 
            # but let's keep them in [0,1]
            if 0 <= col_x <= 1:
                centers.append([col_x, row_y])
            
            if len(centers) >= n:
                break
            
            col_x += x_spacing
        
        if len(centers) >= n:
            break
            
        row_y += y_spacing
        row_idx += 1
    
    # Ensure we have exactly n centers
    centers = np.array(centers[:n])
    
    # Initialize radii to a small valid value
    radii = np.full(n, 0.05)
    
    # Prepare the initial vector for optimization: [x0, y0, r0, x1, y1, r1, ...]
    x0 = np.zeros(3 * n)
    for i in range(n):
        x0[3*i] = centers[i, 0]
        x0[3*i+1] = centers[i, 1]
        x0[3*i+2] = radii[i]
    
    # Define bounds for variables
    # x, y in [0, 1], r in [0, 0.5] (r cannot be > 0.5 to fit in square)
    bounds = []
    for i in range(n):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 0.5)) # r
    
    # Define constraints
    constraints = []
    
    # 1. Boundary Constraints
    # For each circle i:
    # x_i >= r_i  => x_i - r_i >= 0
    # x_i <= 1 - r_i => 1 - x_i - r_i >= 0
    # y_i >= r_i => y_i - r_i >= 0
    # y_i <= 1 - r_i => 1 - y_i - r_i >= 0
    
    for i in range(n):
        idx = 3 * i
        # x - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: v[idx] - v[idx+2]
        })
        # 1 - x - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: 1.0 - v[idx] - v[idx+2]
        })
        # y - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: v[idx+1] - v[idx+2]
        })
        # 1 - y - r >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, i=i: 1.0 - v[idx+1] - v[idx+2]
        })
    
    # 2. Non-overlap Constraints
    # For each pair (i, j) with i < j:
    # dist(i, j) >= r_i + r_j
    # dist^2 >= (r_i + r_j)^2
    # (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
    
    for i in range(n):
        for j in range(i + 1, n):
            idx_i = 3 * i
            idx_j = 3 * j
            
            def dist_constraint(v, i=i, j=j):
                xi, yi, ri = v[idx_i], v[idx_i+1], v[idx_i+2]
                xj, yj, rj = v[idx_j], v[idx_j+1], v[idx_j+2]
                dist_sq = (xi - xj)**2 + (yi - yj)**2
                return dist_sq - (ri + rj)**2
            
            constraints.append({
                'type': 'ineq',
                'fun': dist_constraint
            })
    
    # Objective function: Maximize sum(r) => Minimize -sum(r)
    def objective(v):
        return -np.sum(v[2::3])
    
    # Run the optimization
    # SLSQP is suitable for this type of constrained problem
    try:
        res = minimize(
            objective, 
            x0, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=constraints,
            options={'maxiter': 2000, 'ftol': 1e-10, 'disp': False}
        )

        final_radii = np.zeros(n)

            final_radii[i] = res.x[3*i+2]
            
        # Ensure radii are non-negative (clip small negative values due to numerical error)
        final_radii = np.maximum(final_radii, 0.0)
        
        sum_radii = np.sum(final_radii)
        
    except Exception as e:
        # Fallback in case optimization fails
        # Return the initial valid configuration (though suboptimal)
        final_centers = centers
        final_radii = radii
        sum_radii = np.sum(radii)

    return final_centers, final_radii, sum_radii
