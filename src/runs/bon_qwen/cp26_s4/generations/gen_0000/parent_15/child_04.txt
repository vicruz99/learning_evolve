import numpy as np
from scipy.optimize import minimize

def get_objective_and_gradient(params, n_circles):
    """
    Computes the objective function (negative sum of radii) and gradient.
    Includes penalties for boundary and overlap violations.
    """
    params = params.reshape(-1, 3)
    x = params[:, 0]
    y = params[:, 1]
    r = params[:, 2]
    
    sum_r = np.sum(r)
    
    penalty = 0.0
    grad_x = np.zeros_like(x)
    grad_y = np.zeros_like(y)
    grad_r = np.zeros_like(r)
    
    # Negative gradient for objective (maximizing sum_r)
    grad_r -= 1.0
    
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    # If x < r, penalty. If x > 1-r, penalty.
    for i in range(n_circles):
        # Left boundary
        if x[i] < r[i]:
            val = r[i] - x[i]
            penalty += val**2
            grad_r[i] += 2 * val
            grad_x[i] -= 2 * val
        
        # Right boundary
        if x[i] > 1 - r[i]:
            val = x[i] - (1 - r[i])
            penalty += val**2
            grad_r[i] += 2 * val
            grad_x[i] += 2 * val
            
        # Bottom boundary
        if y[i] < r[i]:
            val = r[i] - y[i]
            penalty += val**2
            grad_r[i] += 2 * val
            grad_y[i] -= 2 * val
            
        # Top boundary
        if y[i] > 1 - r[i]:
            val = y[i] - (1 - r[i])
            penalty += val**2
            grad_r[i] += 2 * val
            grad_y[i] += 2 * val

    # Overlap constraints: dist >= r_i + r_j
    # Using squared distance to avoid sqrt
    for i in range(n_circles):
        for j in range(i + 1, n_circles):
            dx = x[i] - x[j]
            dy = y[i] - y[j]
            dist_sq = dx*dx + dy*dy
            sum_r_ij = r[i] + r[j]
            
            # Constraint: dist_sq >= sum_r_ij^2
            # Violation if sum_r_ij^2 > dist_sq
            target_sq = sum_r_ij * sum_r_ij
            
            if target_sq > dist_sq:
                diff = target_sq - dist_sq
                penalty += diff**2
                
                # Gradient w.r.t positions
                # d/dx (target^2 - dist^2)^2 = 2 * diff * (-2 * dx) = -4 * diff * dx
                force_factor = 4 * diff
                
                grad_x[i] += force_factor * dx
                grad_x[j] -= force_factor * dx
                
                grad_y[i] += force_factor * dy
                grad_y[j] -= force_factor * dy
                
                # Gradient w.r.t radii
                # d/dr_i (r_i + r_j)^4 ... wait, we used diff = (r_i+r_j)^2 - d^2
                # diff^2 = ((r_i+r_j)^2 - d^2)^2
                # d/dr_i = 2 * diff * 2 * (r_i + r_j) = 4 * diff * (r_i + r_j)
                force_r = 4 * diff * sum_r_ij
                grad_r[i] += force_r
                grad_r[j] += force_r

    objective = -sum_r + 1000.0 * penalty
    grad = np.concatenate([grad_x, grad_y, grad_r]).flatten()
    
    return objective, grad

def run_packing():
    """
    Pack 26 circles in a unit square to maximize the sum of radii.
    """
    n_circles = 26
    
    # Initialization: 5x5 grid + 1 extra point
    # Base grid for 25 circles
    coords = []
    grid_size = 5
    # Start a bit inset to allow radius growth
    offset = 0.05
    step = 0.9 / (grid_size - 1)
    
    for i in range(grid_size):
        for j in range(grid_size):
            coords.append([offset + i * step, offset + j * step])
            
    # Add 26th circle slightly perturbed in center
    coords.append([0.5 + 0.01, 0.5 + 0.01])
    
    # Add small random noise to break symmetry
    np.random.seed(42)
    centers = np.array(coords) + np.random.normal(0, 0.001, size=(n_circles, 2))
    
    # Initialize radii slightly below 0.1
    initial_radii = np.full(n_circles, 0.095)
    
    # Combine into single parameter vector [x1, y1, r1, x2, y2, r2, ...]
    initial_params = np.zeros((n_circles, 3))
    initial_params[:, 0] = centers[:, 0]
    initial_params[:, 1] = centers[:, 1]
    initial_params[:, 2] = initial_radii
    
    # Bounds: 0 <= x, y <= 1, 0 <= r <= 0.5
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * n_circles
    
    # Stage 1: Coarse optimization with Powell method (derivative-free)
    def objective_powell(p):
        obj, _ = get_objective_and_gradient(p, n_circles)
        return obj

    res1 = minimize(objective_powell, initial_params.flatten(), method='Powell', 
                    bounds=bounds, options={'maxiter': 500})
    
    # Stage 2: Fine optimization with L-BFGS-B (gradient-based)
    # We use the result from Stage 1 as the starting point.
    # The penalty weight in the function is fixed high (1000), which should be sufficient.
    res2 = minimize(lambda p: get_objective_and_gradient(p, n_circles)[0], 
                    res1.x, method='L-BFGS-B', jac=lambda p: get_objective_and_gradient(p, n_circles)[1],
                    bounds=bounds, options={'maxiter': 1000, 'ftol': 1e-12})
    
    # Extract results
    final_params = res2.x.reshape(-1, 3)
    centers_final = final_params[:, :2]
    radii_final = final_params[:, 2]
    
    # Validation and clamping to ensure strict validity against numerical noise
    # If any radius is negative, set to 0
    radii_final = np.maximum(radii_final, 0.0)
    
    # Re-check boundaries and clamp if necessary
    for i in range(n_circles):
        x, y, r = centers_final[i][0], centers_final[i][1], radii_final[i]
        # Clamp radius to fit in boundary
        r = min(r, x, 1-x, y, 1-y)
        radii_final[i] = max(r, 0.0)
        
    sum_radii = np.sum(radii_final)
    
    return centers_final, radii_final, sum_radii
