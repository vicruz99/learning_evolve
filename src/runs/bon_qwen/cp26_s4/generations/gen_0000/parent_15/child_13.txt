    if max_cols < 1: max_cols = 1
    
    # We'll try to fill rows

    row_number = 0
    
    while count < n_circles:
        # Offset for hexagonal rows
        offset = row_number % 2 * r_est

        # x coords: r + offset, r + offset + 2r, ...

             num_in_row = max_k + 1
        else:
             # Non-shifted row
             # x starts at r_est

             max_k = int((1.0 - 2*r_est) / h_spacing)

            radii[count] = r_est

        row_number += 1
