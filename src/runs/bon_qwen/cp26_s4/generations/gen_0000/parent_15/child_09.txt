import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def compute_objective(vars, lam):
    N = N_CIRCLES
    c = vars[:N*2].reshape(N, 2)
    r = vars[N*2:]
    
    obj = -np.sum(r)
    
    # Boundary penalties: ensure x ± r ∈ [0, 1] and y ± r ∈ [0, 1]
    obj += lam * np.sum(np.maximum(0.0, r - c[:, 0])**2)
    obj += lam * np.sum(np.maximum(0.0, c[:, 0] + r - 1.0)**2)
    obj += lam * np.sum(np.maximum(0.0, r - c[:, 1])**2)
    obj += lam * np.sum(np.maximum(0.0, c[:, 1] + r - 1.0)**2)
    
    # Overlap penalties: ensure dist(i, j) ≥ r_i + r_j
    diffs = c[:, None, :] - c[None, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    r_sum = r[:, None] + r[None, :]
    ov = r_sum - dists
    mask = np.triu(np.ones((N, N), dtype=bool), k=1)
    obj += lam * np.sum(np.maximum(0.0, ov[mask])**2)
                
    return obj

def run_packing():
    # Initial hexagonal-ish configuration for 26 circles
    centers = np.zeros((N_CIRCLES, 2))
    idx = 0
    for i in range(6):

                break
            x = 0.15 + j * 0.15

                x += 0.075
            centers[idx] = [x, y]
            idx += 1
            
    radii = np.full(N_CIRCLES, 0.08)
    x0 = np.concatenate([centers.flatten(), radii])
    
    # Bounds for L-BFGS-B
    bounds = [(0.0, 1.0)] * (2 * N_CIRCLES) + [(0.0, 0.5)] * N_CIRCLES
    
    lam = 1000.0
    best_x = x0
    
    # Continuation optimization: gradually increase penalty weight
    for _ in range(10):
        res = minimize(compute_objective, best_x, args=(lam,), method='L-BFGS-B', 
                       bounds=bounds, options={'maxiter': 2000, 'ftol': 1e-12, 'gtol': 1e-10})
        best_x = res.x
        lam *= 3.0
        
    centers_opt = best_x[:N_CIRCLES*2].reshape(N_CIRCLES, 2)
    radii_opt = best_x[N_CIRCLES*2:]
    
    # Ensure strict positivity
    radii_opt = np.maximum(radii_opt, 1e-9)
    
    return centers_opt, radii_opt, float(np.sum(radii_opt))
