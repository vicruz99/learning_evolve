import numpy as np
from scipy.optimize import minimize

def compute_penalty(centers, r):
    """Compute smooth penalty for boundary and overlap violations."""
    n = centers.shape[0]
    p = 0.0
    
    # Boundary violations: distance to nearest wall must be >= r
    dx = np.minimum(centers[:,0], 1.0 - centers[:,0])
    dy = np.minimum(centers[:,1], 1.0 - centers[:,1])
    d_wall = np.minimum(dx, dy)
    p += np.sum(np.maximum(r - d_wall, 0.0)**2)
    
    # Pairwise violations: distance between centers must be >= 2r
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)  # Ignore self-distance
    p += np.sum(np.maximum(2.0*r - dists, 0.0)**2)
    
    return p

def objective(c_flat, r, n):
    """Objective function wrapper for scipy optimizer."""
    return compute_penalty(c_flat.reshape(n, 2), r)

def run_packing() -> tuple:
    n = 26
    
    # 1. Initialize with a hexagonal lattice pattern
    pts = []
    for i in range(7):
        for j in range(7):
            x = i * 0.14 + (0.07 if j % 2 == 1 else 0.0)
            y = j * 0.14 * np.sqrt(3) / 2
            pts.append([x, y])
    pts = np.array(pts)
    
    # Normalize to fit in unit square with initial margin
    pts = pts - pts.min(axis=0)
    pts = pts / pts.max(axis=0)
    centers = pts[:n].copy() * 0.8 + 0.1
    
    # 2. Homotopy optimization: gradually increase target radius
    r = 0.05
    best_r = r
    best_centers = centers.copy()
    
    # Schedule defines the step size for radius increase over iterations
    schedule = [0.001] * 20 + [0.0005] * 30 + [0.0002] * 50
    
    for step, dr in enumerate(schedule):
        # Bounds enforce circles staying inside [0,1] relative to current r
        bounds = [(r, 1.0 - r)] * (2 * n)
        
        try:
            res = minimize(objective, centers.flatten(), args=(r, n), method='L-BFGS-B', 
                           bounds=bounds, options={'ftol': 1e-12, 'gtol': 1e-10, 'maxiter': 500})
            centers = res.x.reshape(n, 2)
        except Exception:
            pass
            
        pen = compute_penalty(centers, r)
        
        # Adapt radius based on feasibility
        if pen < 1e-9:
            r += dr
            if r > best_r:
                best_r = r
                best_centers = centers.copy()
        else:
            r -= dr * 0.5
            
        if r < 0.01:
            r = 0.01
            
    # 3. Final refinement pass at the best found radius
    bounds = [(best_r, 1.0 - best_r)] * (2 * n)
    try:
        res = minimize(objective, best_centers.flatten(), args=(best_r, n), method='L-BFGS-B', 
                       bounds=bounds, options={'ftol': 1e-14, 'gtol': 1e-12, 'maxiter': 1000})
        best_centers = res.x.reshape(n, 2)
    except Exception:
        pass
        
    # 4. Apply safety margin to guarantee strict validator compliance
    best_r *= 0.9998
    radii = np.full(n, best_r)
    sum_radii = float(np.sum(radii))
    
    return best_centers, radii, sum_radii
