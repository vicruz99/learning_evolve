import numpy as np
from scipy.optimize import minimize

def compute_loss(params, n):
    """
    Computes the objective function: negative sum of radii + penalty for constraint violations.
    params: 1D array of length 3*n (x1, y1, r1, x2, y2, r2, ...)
    """
    coords = params[:2*n].reshape(n, 2)
    radii = params[2*n:]
    
    # Objective: maximize sum of radii => minimize negative sum
    loss = -np.sum(radii)
    
    # Boundary penalties: ensure circles are inside [0,1]x[0,1]
    # x >= r, x <= 1-r, y >= r, y <= 1-r
    v1 = np.maximum(0.0, radii - coords[:,0])
    v2 = np.maximum(0.0, coords[:,0] + radii - 1.0)
    v3 = np.maximum(0.0, radii - coords[:,1])
    v4 = np.maximum(0.0, coords[:,1] + radii - 1.0)
    loss += 1000.0 * (np.sum(v1**2) + np.sum(v2**2) + np.sum(v3**2) + np.sum(v4**2))
    
    # Overlap penalties: ensure dist(i,j) >= r_i + r_j
    # Vectorized pairwise distance calculation
    diffs = coords[:, np.newaxis, :] - coords[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    
    r_sums = radii[:, np.newaxis] + radii[np.newaxis, :]
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    violations = np.maximum(0.0, r_sums - dists)
    violations[~mask] = 0.0
    loss += 1000.0 * np.sum(violations**2)
    
    return loss

def run_packing():
    n = 26
    np.random.seed(42)
    
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Prepare initial configurations
    configs = []
    
    # 1. 5x5 Grid + 1 center circle
    grid = np.linspace(0.125, 0.875, 5)
    X, Y = np.meshgrid(grid, grid)
    c1 = np.column_stack((X.ravel(), Y.ravel()))
    c1 = np.vstack([c1, [0.5, 0.5]])
    configs.append((c1[:n], np.ones(n) * 0.13))
    
    # 2. Random perturbed grid
    c2 = c1.copy() + np.random.randn(n, 2) * 0.05
    # Clip to bounds for safety
    c2 = np.clip(c2, 0.05, 0.95)
    configs.append((c2, np.ones(n) * 0.12))
    
    # 3. Pure random
    c3 = np.random.rand(n, 2) * 0.8 + 0.1
    configs.append((c3, np.ones(n) * 0.1))
    
    # Optimization bounds
    bounds = [(0.0, 1.0)] * (2*n) + [(1e-6, 0.5)] * n
    
    for c_init, r_init in configs:
        x0 = np.hstack([c_init.ravel(), r_init])
        
        # Run optimization
        res = minimize(compute_loss, x0, args=(n,), method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': 2000, 'ftol': 1e-12, 'gtol': 1e-10})
        
        c_opt = res.x[:2*n].reshape(n, 2)
        r_opt = res.x[2*n:]
        
        # Validate packing strictly
        valid = True
        for i in range(n):
            x, y, r = c_opt[i,0], c_opt[i,1], r_opt[i]
            if x - r < -1e-13 or x + r > 1 + 1e-13 or y - r < -1e-13 or y + r > 1 + 1e-13:
                valid = False
                break
        if valid:
            for i in range(n):
                for j in range(i + 1, n):
                    d = np.sqrt((c_opt[i,0]-c_opt[j,0])**2 + (c_opt[i,1]-c_opt[j,1])**2)
                    if d < r_opt[i] + r_opt[j] - 1e-13:
                        valid = False
                        break
                if not valid:
                    break
                    
        if valid:
            current_sum = np.sum(r_opt)
            if current_sum > best_sum:
                best_sum = current_sum
                best_centers = c_opt.copy()
                best_radii = r_opt.copy()
                
    # Fallback to a known valid grid packing if optimization fails or yields poor results
    if best_centers is None or best_sum < 2.5:
        grid = np.linspace(0.125, 0.875, 5)
        X, Y = np.meshgrid(grid, grid)
        best_centers = np.column_stack((X.ravel(), Y.ravel()))[:n]
        best_radii = np.ones(n) * 0.125 * 0.99  # Slight shrink for safety margin
        best_sum = np.sum(best_radii)
        
    return best_centers, best_radii, best_sum
