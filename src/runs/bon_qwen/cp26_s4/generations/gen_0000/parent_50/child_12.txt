import numpy as np
import math

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    
    # Constants
    n_circles = 26
    square_size = 1.0
    
    # Initialize positions: Hexagonal packing
    # We try to fit a hexagonal grid. 
    # Rows offset by r horizontally, spaced by sqrt(3)*r vertically.
    # Initial guess for r
    r_current = 0.09
    
    # Generate initial centers
    centers = np.zeros((n_circles, 2))
    idx = 0

    # We will generate a grid and pick the first 26 valid points
    row_y = r_current
    row_idx = 0
    grid_points = []

    while len(grid_points) < n_circles + 10:
        row_y = r_current + row_idx * r_current * math.sqrt(3)
        if row_y + r_current > square_size:
            break
            
        # Determine x start
        if row_idx % 2 == 0:
            start_x = r_current
        else:
            start_x = 2 * r_current
            
        row_x = start_x
        while row_x + r_current <= square_size:
            if idx < n_circles:
                grid_points.append([row_x, row_y])
                idx += 1
            row_x += 2 * r_current
        row_idx += 1
        
    if len(grid_points) < n_circles:
        # Fallback to random if grid failed (unlikely)
        for i in range(n_circles):
            centers[i] = [np.random.rand(), np.random.rand()]
    else:
        centers[:n_circles] = grid_points[:n_circles]

    # Optimization loop
    # We will try to maximize r by expanding it and relaxing positions
    r = 0.09
    step_size_r = 0.0001
    max_iterations = 50000
    cooling_rate = 0.9995

    force_repulsion = 0.5
    force_boundary = 2.0
    max_move = 0.05

        # Increase radius slowly
        if iteration % 50 == 0:
            r += step_size_r

        # We treat it as an energy minimization problem
        # Energy = sum of overlap^2 + sum of boundary penetration^2
        
        forces = np.zeros_like(centers)
        violations = 0
        max_violation = 0.0

                dist_vec = centers[i] - centers[j]
                dist = np.linalg.norm(dist_vec)
                min_dist = r + r  # 2r
                
                if dist < min_dist:
                    overlap = min_dist - dist
                    violations += overlap**2
                    if overlap > max_violation:
                        max_violation = overlap
                    
                    # Normalize force
                    if dist > 1e-9:
                        force_vec = (dist_vec / dist) * overlap * force_repulsion
                        forces[i] += force_vec
                        forces[j] -= force_vec
                    else:
                        # Random push if on top of each other
                        forces[i] += np.random.randn(2) * 0.1
                        forces[j] -= np.random.randn(2) * 0.1

            x, y = centers[i]
            # Left

                pen = r - x
                violations += pen**2
                if pen > max_violation: max_violation = pen
                forces[i, 0] += pen * force_boundary
            # Right

                pen = x + r - square_size
                violations += pen**2
                if pen > max_violation: max_violation = pen
                forces[i, 0] -= pen * force_boundary
            # Bottom
            if y - r < 0:
                pen = r - y
                violations += pen**2
                if pen > max_violation: max_violation = pen
                forces[i, 1] += pen * force_boundary
            # Top

                pen = y + r - square_size
                violations += pen**2
                if pen > max_violation: max_violation = pen
                forces[i, 1] -= pen * force_boundary

        # Update positions
        # Cap the movement to ensure stability
        move = np.clip(forces, -max_move, max_move)
        centers += move

        if iteration % 100 == 0 and max_violation < 1e-6:
            # Valid configuration found for current r

            pass

            pass

        if max_move < 1e-8:
            max_move = 1e-8

    # The simulation might have slight numerical errors
    # Let's check constraints and shrink r if needed
    # But the validator allows 1e-12 error.
    
    # To be safe, let's run a quick check and adjust
    # If any overlap, shrink r
    # Since we want to maximize sum, we want r as large as possible.
    # The simulation naturally finds the max r where forces balance.

    # The radius cannot exceed half the minimum distance between centers
    # and cannot exceed the minimum distance to boundary
    r_final = min(min_dist / 2.0, min_boundary)
    
    # Apply this radius
    radii = np.full(n_circles, r_final)
    
    # Sum of radii
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
