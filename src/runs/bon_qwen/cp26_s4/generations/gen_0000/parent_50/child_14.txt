import numpy as np
from scipy.optimize import linprog


def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26

    best_sum = 0.0
    best_centers = None
    best_radii = None

    for trial in range(80):
        centers = init_centers(n, trial)

        for iteration in range(200):
            radii = compute_optimal_radii(centers, n)

            forces = compute_forces(centers, radii, n)

            lr = 0.004 / (1 + iteration * 0.008)
            centers += lr * forces

            for i in range(n):
                centers[i, 0] = np.clip(centers[i, 0], radii[i] + 1e-10, 1 - radii[i] - 1e-10)
                centers[i, 1] = np.clip(centers[i, 1], radii[i] + 1e-10, 1 - radii[i] - 1e-10)

        radii = compute_optimal_radii(centers, n)
        s = float(np.sum(radii))

        if s > best_sum:
            best_sum = s
            best_centers = centers.copy()
            best_radii = radii.copy()

    # Final refinement with smaller steps
    for iteration in range(300):
        radii = compute_optimal_radii(best_centers, n)
        forces = compute_forces(best_centers, radii, n)
        lr = 0.001 / (1 + iteration * 0.01)
        best_centers += lr * forces
        for i in range(n):
            best_centers[i, 0] = np.clip(best_centers[i, 0], radii[i] + 1e-10, 1 - radii[i] - 1e-10)
            best_centers[i, 1] = np.clip(best_centers[i, 1], radii[i] + 1e-10, 1 - radii[i] - 1e-10)

    best_radii = compute_optimal_radii(best_centers, n)
    best_sum = float(np.sum(best_radii))

    return best_centers, best_radii, best_sum


def init_centers(n, seed):
    np.random.seed(seed)
    centers = np.zeros((n, 2))
    idx = 0

    spacing = 0.17
    for row in range(8):
        for col in range(7):
            if idx >= n:
                break
            x = 0.08 + col * spacing
            y = 0.08 + row * spacing * 0.866
            if row % 2 == 1:
                x += spacing * 0.5
            x = x % 0.85 + 0.05
            y = y % 0.85 + 0.05
            centers[idx] = [x, y]
            idx += 1
        if idx >= n:
            break

    centers += np.random.randn(n, 2) * 0.015
    return centers


def compute_optimal_radii(centers, n):
    c = -np.ones(n)

    A_ub = []
    b_ub = []

    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dist)

    for i in range(n):
        for dim in range(2):
            row = np.zeros(n)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(min(centers[i, dim], 1.0 - centers[i, dim]))

    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)

    bounds = [(0.0, None) for _ in range(n)]

    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')

    if result.success:
        return np.maximum(result.x, 1e-12)
    else:
        return np.ones(n) * 0.01


def compute_forces(centers, radii, n):
    forces = np.zeros_like(centers)

    for i in range(n):
        for j in range(i + 1, n):
            diff = centers[i] - centers[j]
            dx = diff[0]
            dy = diff[1]
            dist = np.sqrt(dx * dx + dy * dy)
            if dist > 1e-15:
                desired = radii[i] + radii[j]
                if dist < desired + 1e-8:
                    overlap = desired - dist + 1e-8
                    force_mag = overlap * 80.0 / dist
                    forces[i, 0] += dx * force_mag
                    forces[i, 1] += dy * force_mag
                    forces[j, 0] -= dx * force_mag
                    forces[j, 1] -= dy * force_mag

    for i in range(n):
        for dim in range(2):
            if centers[i, dim] < radii[i] + 1e-9:
                forces[i, dim] += (radii[i] + 1e-9 - centers[i, dim]) * 200.0
            elif centers[i, dim] > 1.0 - radii[i] - 1e-9:
                forces[i, dim] -= (centers[i, dim] - (1.0 - radii[i] - 1e-9)) * 200.0

    return forces
