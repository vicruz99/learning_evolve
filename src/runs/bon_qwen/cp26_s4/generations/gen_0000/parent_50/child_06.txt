import numpy as np
import scipy.optimize as opt

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes the positions and radii of 26 circles in a unit square to maximize the sum of radii.
    
    Returns:
        centers: np.array of shape (26, 2)
        radii: np.array of shape (26,)
        sum_radii: float
    """
    N = 26

    # --- 1. Initialization: Hexagonal Lattice ---
    # A 5x5 square grid is a good baseline, but we have 26 circles.
    # A hexagonal arrangement of rows (e.g., 6, 5, 5, 5, 5) is denser.
    centers_init = []
    radii_init = []
    
    # Define row lengths to sum to 26
    rows = [6, 5, 5, 5, 5] 
    
    # Approximate diameter for initialization
    # 5 rows of height approx 0.2 + 4*0.1732 ~ 0.89 fits in 1.0
    d = 0.205 
    r_approx = d / 2.0
    
    y_pos = r_approx # Start at bottom margin
    
    for i, count in enumerate(rows):
        # Calculate row height (vertical spacing)
        if i > 0:
            y_pos += r_approx * np.sqrt(3)
        
        # Horizontal spacing
        x_start = r_approx
        # Shift for even rows in hexagonal packing
        if i % 2 == 1:
            x_start += r_approx 
        
        x_pos = x_start
        for j in range(count):
            centers_init.append([x_pos, y_pos])
            radii_init.append(r_approx)
            x_pos += d
            
    centers_init = np.array(centers_init)
    radii_init = np.array(radii_init)
    
    # Flatten variables for the optimizer
    # Format: [x1, y1, r1, x2, y2, r2, ...]
    x0 = np.concatenate([centers_init.ravel(), radii_init])

    # --- 2. Define Constraints ---
    def constraints(vars):
        centers = vars[:N*2].reshape(N, 2)
        radii = vars[N*2:]
        cons = []
        
        # Boundary constraints
        for i in range(N):
            x, y = centers[i]
            r = radii[i]
            # x - r >= 0
            cons.append({'type': 'ineq', 'fun': lambda v, i=i: v[i*3] - v[N*2 + i]}) 
            # 1 - x - r >= 0  =>  x + r <= 1
            cons.append({'type': 'ineq', 'fun': lambda v, i=i: 1.0 - v[i*3] - v[N*2 + i]})
            # y - r >= 0
            cons.append({'type': 'ineq', 'fun': lambda v, i=i: v[i*3 + 1] - v[N*2 + i]})
            # 1 - y - r >= 0 => y + r <= 1
            cons.append({'type': 'ineq', 'fun': lambda v, i=i: 1.0 - v[i*3 + 1] - v[N*2 + i]})

        # Non-overlap constraints: dist(i,j) >= r_i + r_j
        # dist^2 >= (r_i + r_j)^2
        for i in range(N):
            for j in range(i + 1, N):
                c1 = centers[i]
                c2 = centers[j]
                r1, r2 = radii[i], radii[j]
                
                # Define a closure-free helper or use a fixed function reference
                # To avoid closures capturing variables improperly in scipy callbacks, 
                # we pass indices or use the flat vector directly.
                def make_dist_constraint(i_idx, j_idx, n_vars):
                    def dist_func(v):
                        c1 = np.array([v[i_idx*3], v[i_idx*3+1]])
                        c2 = np.array([v[j_idx*3], v[j_idx*3+1]])
                        r1 = v[n_vars + i_idx]
                        r2 = v[n_vars + j_idx]
                        dist_sq = np.sum((c1 - c2)**2)
                        return dist_sq - (r1 + r2)**2
                    return dist_func

                cons.append({'type': 'ineq', 'fun': make_dist_constraint(i, j, N*2)})
                
        return cons

    # --- 3. Optimization ---
    
    # Objective: Maximize sum(radii) => Minimize -sum(radii)
    def objective(vars):
        radii = vars[N*2:]
        return -np.sum(radii)

    # Bounds for variables: x, y in [0, 1], r in [0, 0.5]
    bounds = []
    for _ in range(N):
        bounds.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])

    # Run optimization
    # We run it twice: once to get a good solution, once to refine
    res = opt.minimize(
        objective, 
        x0, 
        method='SLSQP', 
        bounds=bounds, 
        constraints=constraints,
        options={'maxiter': 500, 'ftol': 1e-12}
    )

    # Second pass for refinement using the result of the first pass
    if res.success:
        res2 = opt.minimize(
            objective, 
            res.x, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=constraints,
            options={'maxiter': 1000, 'ftol': 1e-14}
        )
        final_vars = res2.x
    else:
        final_vars = res.x

    # --- 4. Formatting Output ---
    final_centers = final_vars[:N*2].reshape(N, 2)
    final_radii = final_vars[N*2:]
    
    # Clamp small negative radii due to numerical noise if any (though bounds should prevent)
    final_radii = np.maximum(final_radii, 0)

    sum_radii = np.sum(final_radii)
    
    # Sort by radii for cleaner output (optional but good practice)
    # Reorder centers and radii together
    sort_idx = np.argsort(final_radii)[::-1]
    final_centers = final_centers[sort_idx]
    final_radii = final_radii[sort_idx]

    return final_centers, final_radii, sum_radii
