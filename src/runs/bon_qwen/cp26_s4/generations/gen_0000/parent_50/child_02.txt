import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    n = 26
    r_init = 0.08
    
    # 1. Construct initial hexagonal packing (6-5-6-5-4 pattern)
    centers = np.zeros((n, 2))
    idx = 0
    for i in range(5):
        n_circles = 6 if i % 2 == 0 else 5
        if i == 4:
            n_circles = 4
        y = r_init + i * (np.sqrt(3) * r_init)
        x_start = r_init + (i % 2) * r_init
        for j in range(n_circles):
            x = x_start + j * (2 * r_init)
            centers[idx] = [x, y]
            idx += 1
            
    # 2. Break symmetry to avoid degenerate optimization paths
    centers += np.random.uniform(-0.002, 0.002, centers.shape)
    centers = np.clip(centers, 0.01, 0.99)
    
    radii = np.full(n, r_init)
    
    # 3. Prepare optimizer input
    x0 = np.concatenate([centers.flatten(), radii])
    bounds = [(0.0, 1.0)] * 52 + [(0.0, 0.25)] * 26
    
    def objective(params):
        c = params[:52].reshape(n, 2)
        r = params[52:]
        
        # Negative sum of radii (we minimize this)
        obj = -np.sum(r)
        P = 3000.0
        
        # Boundary penalties
        for i in range(n):
            x, y = c[i]
            ri = r[i]
            obj += P * np.maximum(0, ri - x)**2
            obj += P * np.maximum(0, x + ri - 1)**2
            obj += P * np.maximum(0, ri - y)**2
            obj += P * np.maximum(0, y + ri - 1)**2
            
        # Overlap penalties
        for i in range(n):
            for j in range(i+1, n):
                dx = c[i,0] - c[j,0]
                dy = c[i,1] - c[j,1]
                dist = np.sqrt(dx*dx + dy*dy)
                min_d = r[i] + r[j]
                if dist < min_d:
                    obj += P * (min_d - dist)**2
        return obj
        
    # 4. Run optimization
    res = minimize(objective, x0, method='L-BFGS-B', bounds=bounds, 
                   options={'maxiter': 2000, 'ftol': 1e-12, 'gtol': 1e-8})
    
    opt_centers = res.x[:52].reshape(n, 2)
    opt_radii = res.x[52:]
    
    # 5. Ensure strict numerical validity
    opt_radii = np.maximum(opt_radii, 1e-9)
    
    return opt_centers, opt_radii, float(np.sum(opt_radii))
