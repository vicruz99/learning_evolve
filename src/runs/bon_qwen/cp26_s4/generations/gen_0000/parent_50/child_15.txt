import numpy as np
from math import sqrt

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    
    Returns:
        tuple: (centers, radii, sum_radii) where centers is (26, 2), radii is (26,), 
               and sum_radii is the total sum of radii.
    """

    n = 26

    # Start with a 5x5 grid of circles with radius 0.1, plus a small central circle.
    # This provides a valid starting configuration with a high sum of radii.
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Initialize 25 circles in a 5x5 grid
    grid_radius = 0.1
    for i in range(5):
        for j in range(5):
            idx = i * 5 + j
            centers[idx] = [grid_radius + j * 2 * grid_radius, 
                            grid_radius + i * 2 * grid_radius]
            radii[idx] = grid_radius
            
    # Initialize the 26th circle in the center
    idx = 25
    centers[idx] = [0.5, 0.5]
    radii[idx] = 0.05  # Small initial radius to avoid overlap
    
    current_sum_radii = np.sum(radii)
    
    # --- Phase 2: Iterative Optimization ---
    # We use a simple Simulated Annealing-like approach to maximize sum of radii.
    
    num_iterations = 200000
    initial_temp = 0.01
    final_temp = 1e-9
    cooling_rate = (final_temp / initial_temp) ** (1.0 / num_iterations)
    temp = initial_temp
    
    best_centers = centers.copy()
    best_radii = radii.copy()
    best_sum = current_sum_radii
    
    # Define bounds for perturbation
    center_move_scale = 0.05
    radius_change_scale = 0.01

        # Select a random circle to move or resize
        idx = np.random.randint(0, n)
        
        # Store current state
        old_center = centers[idx].copy()
        old_radius = radii[idx]
        old_sum = current_sum_radii
        
        # Propose a move
        # 50% chance to move center, 50% chance to change radius
        if np.random.random() < 0.5:
            # Move center
            move = np.random.normal(0, center_move_scale, 2)
            new_center = old_center + move
            # Clip to valid bounds (must stay within [r, 1-r])
            # We'll check validity strictly in the validation step
            centers[idx] = new_center
        else:
            # Change radius
            radius_delta = np.random.normal(0, radius_change_scale)
            new_radius = old_radius + radius_delta

                current_sum_radii = new_sum

                    best_sum = current_sum_radii
                    best_centers = centers.copy()
                    best_radii = radii.copy()

                # Revert move
                centers[idx] = old_center
                radii[idx] = old_radius
        else:
            # Invalid configuration, revert move
            centers[idx] = old_center
            radii[idx] = old_radius
            
        # Cool down
        temp *= cooling_rate
        
    return best_centers, best_radii, best_sum


def validate_packing_step(centers, radii, modified_idx):
    """
    Efficiently validates only the constraints involving the modified circle.
    This is much faster than checking all pairs for every iteration.
    """
    n = centers.shape[0]
    c_mod = centers[modified_idx]
    r_mod = radii[modified_idx]
    
    # Check boundary constraints for modified circle
    if (r_mod < 0 or 
        c_mod[0] - r_mod < -1e-9 or c_mod[0] + r_mod > 1 + 1e-9 or
        c_mod[1] - r_mod < -1e-9 or c_mod[1] + r_mod > 1 + 1e-9):
        return False
        
    # Check overlap with other circles
    for i in range(n):
        if i == modified_idx:
            continue
            
        c_other = centers[i]
        r_other = radii[i]
        
        dist_sq = np.sum((c_mod - c_other) ** 2)
        min_dist = r_mod + r_other
        
        # Check if distance is sufficient
        if dist_sq < (min_dist - 1e-9) ** 2:
            return False
            
    return True

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    (Provided in prompt, used for final validation or reference)
    """
    n = centers.shape[0]

    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True
