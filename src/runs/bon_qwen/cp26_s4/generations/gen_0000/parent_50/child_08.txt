import numpy as np
from scipy.optimize import minimize, NonlinearConstraint
import math

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square

    Args:
        centers: np.array of shape (n, 2) with (x, y) coordinates
        radii: np.array of shape (n) with radius of each circle

    Returns:
        True if valid, False otherwise
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True

def get_initial_guess(n):
    """
    Generates a hexagonal lattice initial guess for n circles.
    """
    # Estimate radius based on area to get a feasible starting point
    # Area of square = 1. Area of n circles = n * pi * r^2.
    # Density approx 0.8. 1 = 0.8 * n * pi * r^2 => r ~ sqrt(1 / (0.8 * n * pi))
    r_est = math.sqrt(1.0 / (0.85 * n * math.pi))
    # Clamp to a safe small value if estimation is too aggressive for lattice
    r_est = min(r_est, 0.1) 
    
    # Try to fit in hexagonal pattern
    centers = []
    radii = []
    
    # Hexagonal packing parameters
    # Horizontal spacing 2r, Vertical spacing r*sqrt(3)
    # We try to fit rows
    r = 0.09 # Start with a safe radius
    
    row_y = r
    row_idx = 0
    
    while len(centers) < n:
        # Determine x start for this row (staggered)
        if row_idx % 2 == 0:
            x_start = r
        else:
            x_start = 2 * r # Shifted
        
        x = x_start
        count_in_row = 0
        while x + r <= 1.0 and len(centers) < n:
            centers.append([x, row_y])
            radii.append(r)
            x += 2 * r
            count_in_row += 1
        
        row_y += r * math.sqrt(3)
        row_idx += 1
        
        # If we can't fit more rows because of height, break
        if row_y + r > 1.0:
            break
            
    # If we still don't have n circles, fill randomly with small radius
    while len(centers) < n:
        # Find a spot
        found = False
        for _ in range(100):
            rx, ry = np.random.rand(2)
            rr = 0.05
            # Check simple bounds
            if rx - rr >= 0 and rx + rr <= 1 and ry - rr >= 0 and ry + rr <= 1:
                centers.append([rx, ry])
                radii.append(rr)
                found = True
                break
        if not found:
            centers.append([0.5, 0.5])
            radii.append(0.01)

    return np.array(centers), np.array(radii)

    bounds = []

        bounds.append((0.0, 1.0))

                g = np.zeros_like(vars)
