import numpy as np
from scipy.optimize import minimize

def compute_loss(x, n, lam):
    """
    Computes the objective function: maximize sum of radii subject to packing constraints.
    Uses a penalty method for constraints.
    """
    c = x[:2*n].reshape(n, 2)
    r = x[2*n:]
    
    # Boundary penalties: circles must be inside [0,1]x[0,1]
    # Conditions: x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
    pen = np.sum(np.maximum(0, r - c[:, 0])**2)
    pen += np.sum(np.maximum(0, r - (1.0 - c[:, 0]))**2)
    pen += np.sum(np.maximum(0, r - c[:, 1])**2)
    pen += np.sum(np.maximum(0, r - (1.0 - c[:, 1]))**2)
    
    # Overlap penalties: circles must not overlap
    # Condition: dist(i,j) >= r_i + r_j
    diffs = c[:, np.newaxis, :] - c[np.newaxis, :, :]
    dist = np.sqrt(np.sum(diffs**2, axis=2))
    idx = np.tril_indices(n, -1)
    pen += np.sum(np.maximum(0, r[idx[0]] + r[idx[1]] - dist[idx])**2)
    
    # Objective: maximize sum(r) <=> minimize -sum(r)
    return -np.sum(r) + lam * pen

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

            x += 0.1
        y += 0.1 * np.sqrt(3)
        row += 1
    centers = np.array(centers[:n])
    # Add small random perturbation to break symmetry and avoid local minima
    centers += np.random.normal(0, 0.005, centers.shape)
    
    # Initialize radii
    radii = np.full(n, 0.03)
    
    # Combine variables: [x1, y1, x2, y2, ..., r1, r2, ...]

    lams = [100, 1000, 5000, 20000, 50000]
    x_curr = x0

            return compute_loss(x, n, lam)
        
        res = minimize(objective, x_curr, method='L-BFGS-B', bounds=bounds,

    best_radii = np.maximum(best_radii, 0.0)
    
    return best_centers, best_radii, float(np.sum(best_radii))
