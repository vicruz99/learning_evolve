import numpy as np
from scipy.optimize import linprog

def run_packing():

    n = 26

    rows = 6

        # We have M constraints.

            b_ub[idx] = b_i[i]
            idx += 1

            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
