To maximize the sum of radii for 26 circles in a unit square, we formulate this as a constrained optimization problem. Directly optimizing centers and radii with inequality constraints is non-convex and prone to getting stuck. We employ two key techniques:

1. **Boundary-Enforcing Parameterization**: We map each circle's position $(x_i, y_i)$ to variables $(u_i, v_i) \in [0, 1]$ and radius $r_i$ via:
   $x_i = r_i + (1 - 2r_i)u_i$
   $y_i = r_i + (1 - 2r_i)v_i$
   This transformation guarantees $r_i \le x_i \le 1-r_i$ and $r_i \le y_i \le 1-r_i$ automatically, eliminating boundary constraints from the optimizer's burden.

2. **Penalty Method with Continuation**: We optimize an unconstrained objective:
   $\text{minimize } -\sum r_i + \lambda \sum_{i<j} \max(0, r_i + r_j - d_{ij})^2$
   Starting with a small $\lambda$ and gradually increasing it allows the optimizer to first arrange circles roughly, then tighten them to resolve overlaps while expanding radii. This smooth penalty function is compatible with L-BFGS-B.

3. **Multi-Start Initialization**: We seed the optimizer with diverse configurations (hexagonal lattice, uniform grid, and random placements) to escape local minima. Each run undergoes the continuation schedule. The best valid configuration is selected.

This approach efficiently handles non-overlap constraints, automatically respects boundaries, and pushes radii to their maximal feasible values.

import numpy as np
from scipy.optimize import minimize

def run_packing():
    n = 26
    # Precompute indices for upper triangle to vectorize pairwise overlap computation
    upper_idx = np.triu_indices(n, k=1)
    
    def objective_and_penalty(z, lam):
        # Extract parameters
        u = z[0::3]
        v = z[1::3]
        r = z[2::3]
        
        # Boundary-enforcing transformation
        # Maps u,v in [0,1] to x,y in [r, 1-r]
        x = r + (1.0 - 2.0 * r) * u
        y = r + (1.0 - 2.0 * r) * v
        
        # Pairwise distances
        cx, cy = x, y
        diff_x = cx[:, None] - cx[None, :]
        diff_y = cy[:, None] - cy[None, :]
        dist = np.sqrt(diff_x**2 + diff_y**2)
        
        # Overlap calculation: positive means circles intersect
        r_sum = r[:, None] + r[None, :]
        overlap = r_sum[upper_idx] - dist[upper_idx]
        
        # Smooth penalty for overlaps
        penalty = np.sum(np.maximum(0.0, overlap)**2)
        
        # We want to maximize sum(r), so minimize -sum(r) + penalty
        return -np.sum(r) + lam * penalty

    # Bounds: u,v in [0,1], r in [epsilon, 0.45]
    bounds = [(0.0, 1.0) if i % 3 != 2 else (1e-5, 0.45) for i in range(3 * n)]
    
    best_val = np.inf
    best_z = None
    
    def make_z(pts, r_val):
        z = np.zeros(3 * n)
        denom = 1.0 - 2.0 * r_val
        for i in range(n):
            x, y = pts[i]
            u = np.clip((x - r_val) / denom, 0.0, 1.0)
            v = np.clip((y - r_val) / denom, 0.0, 1.0)
            z[3 * i] = u
            z[3 * i + 1] = v
            z[3 * i + 2] = r_val
        return z

    # Generate diverse initial configurations
    starts = []
    
    # 1. Hexagonal lattice pattern
    r_init = 0.09
    hex_pts = []
    ys = np.sqrt(3.0) * r_init
    xs = 2.0 * r_init
    y0 = (1.0 - 4.0 * ys) / 2.0
    x0 = (1.0 - 4.0 * xs) / 2.0
    for i in range(6):
        for j in range(6):
            if len(hex_pts) >= n:
                break
            x = x0 + j * xs + (i % 2) * (xs / 2.0)
            y = y0 + i * ys
            hex_pts.append([x, y])
    starts.append(hex_pts)
    
    # 2. Random placements
    np.random.seed(42)
    for _ in range(8):
        starts.append((np.random.rand(n, 2)).tolist())
        
    # 3. Uniform grid
    grid_pts = []
    for i in range(5):
        for j in range(5):
            if len(grid_pts) >= n:
                break
            grid_pts.append([0.1 + j * 0.2, 0.1 + i * 0.2])
    starts.append(grid_pts)

    # Optimization loop with continuation method
    for pts in starts:
        z0 = make_z(pts, 0.085)
        curr = z0.copy()
        
        # Gradually increase penalty weight to tighten constraints
        for lam in [50.0, 200.0, 1000.0, 5000.0, 20000.0]:
            res = minimize(lambda z, l=lam: objective_and_penalty(z, l), 
                           curr, method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 3000, 'ftol': 1e-14, 'gtol': 1e-12})
            curr = res.x
            
        # Track best objective value (minimizing -sum(r))
        val = -np.sum(curr[2::3])
        if val < best_val:
            best_val = val
            best_z = curr.copy()
            
    # Extract and format results
    u = best_z[0::3]
    v = best_z[1::3]
    r = best_z[2::3]
    x = r + (1.0 - 2.0 * r) * u
    y = r + (1.0 - 2.0 * r) * v
    
    centers = np.column_stack((x, y))
    radii = r
    
    # Final safety clamp (transformation should guarantee this, but ensures numerical robustness)
    centers[:, 0] = np.clip(centers[:, 0], radii, 1.0 - radii)
    centers[:, 1] = np.clip(centers[:, 1], radii, 1.0 - radii)
    
    return centers, radii, float(np.sum(radii))
