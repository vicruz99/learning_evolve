    Returns (centers, radii, sum_radii).
    """

    best_sum = -1.0

    best_radii = None

    def solve_packing(init_centers, init_radii):

        # Shape: (26*2 + 26) = 78

        bounds = [(0, 1)] * (2 * N) + [(0, 0.5)] * N

            radii = x_vars[2*N:]

            score = -np.sum(radii)

            # Vectorized distance calculation
            # centers shape (N, 2)
            # diff shape (N, N, 2)
            diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
            dists = np.sqrt(np.sum(diff**2, axis=2))

            # We want to penalize when dist < r_i + r_j

            radii_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
            overlaps = radii_sum - dists

            overlaps = np.maximum(0, overlaps)
            # Ignore self-overlaps and double counting (upper triangle)
            # Using a mask or just summing all and dividing by 2 is fine
            # But we must exclude diagonal (self)
            np.fill_diagonal(overlaps, 0)
            penalty_overlap = np.sum(overlaps**2)
            
            # Penalty for wall violations
            # x - r >= 0, 1 - x - r >= 0, same for y
            # Violations: r - x, r - (1-x)
            v1 = np.maximum(0, radii - centers[:, 0])
            v2 = np.maximum(0, radii - (1 - centers[:, 0]))
            v3 = np.maximum(0, radii - centers[:, 1])
            v4 = np.maximum(0, radii - (1 - centers[:, 1]))
            penalty_wall = np.sum(v1**2) + np.sum(v2**2) + np.sum(v3**2) + np.sum(v4**2)
            
            # Total objective
            # Large multiplier for penalties to enforce constraints
            lam = 1000.0
            return score + lam * (penalty_overlap + penalty_wall)

        x0 = np.concatenate([init_centers.flatten(), init_radii])
        
        # Use L-BFGS-B as it handles bounds well
        res = minimize(objective, x0, method='L-BFGS-B', bounds=bounds, 
                       options={'maxiter': 2000, 'ftol': 1e-9})
        
        res_centers = res.x[:2*N].reshape(N, 2)
        res_radii = res.x[2*N:]
        
        # Clean up small negative radii due to numerical errors
        res_radii = np.maximum(res_radii, 0)

        # Though penalty should have handled it, let's ensure strict validity
        # Project centers to valid range based on radii
        for i in range(N):
            r = res_radii[i]
            # Ensure center is within [r, 1-r]
            res_centers[i, 0] = np.clip(res_centers[i, 0], r, 1-r)
            res_centers[i, 1] = np.clip(res_centers[i, 1], r, 1-r)
            
        return res_centers, res_radii, np.sum(res_radii)
