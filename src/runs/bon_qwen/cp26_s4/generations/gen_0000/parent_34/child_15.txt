import numpy as np
from scipy.optimize import minimize

def compute_objective(x, lam, N):
    centers = x[:2*N].reshape(N, 2)
    radii = x[2*N:]
    
    obj = -np.sum(radii)
    penalty = 0.0
    
    for i in range(N):
        cx, cy = centers[i]
        r = radii[i]
        
        # Boundary constraints: circle must be inside [0,1]x[0,1]
        if cx - r < 0:
            penalty += (cx - r)**2
        if cx + r > 1:
            penalty += (cx + r - 1)**2
        if cy - r < 0:
            penalty += (cy - r)**2
        if cy + r > 1:
            penalty += (cy + r - 1)**2
            
        # Pairwise non-overlap constraints
        for j in range(i + 1, N):
            rj = radii[j]
            dx = cx - centers[j, 0]
            dy = cy - centers[j, 1]
            dist = np.sqrt(dx*dx + dy*dy)
            overlap = r + rj - dist
            if overlap > 0.0:
                penalty += overlap * overlap
                
    return obj + lam * penalty

def get_initial_guess(N):
    centers = []
    row = 0
    while len(centers) < N:
        col = 0
        while len(centers) < N:
            x = 0.05 + col * 0.14

                x += 0.07
            if 0 <= x <= 1 and 0 <= y <= 1:
                centers.append([x, y])
            col += 1
        row += 1
    return np.array(centers[:N])

def run_packing():

    centers_init = get_initial_guess(N)

    bounds = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N

    x_current = x0
    
    # Continuation method: progressively tighten constraint enforcement
    for stage in range(5):

        res = minimize(compute_objective, x_current, method='L-BFGS-B',

        x_current = res.x

    radii = x_current[2*N:]

    radii = np.maximum(radii, 1e-6)
    
    return centers, radii, np.sum(radii)
