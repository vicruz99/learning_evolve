import numpy as np
from scipy.optimize import minimize

def compute_energy(centers, radii):
    """
    Compute the energy of the configuration based on constraint violations.
    Energy is sum of squared violations of non-overlap and boundary constraints.
    """
    N = centers.shape[0]
    E = 0.0
    
    # Boundary violations: circle must be inside [0,1]x[0,1]
    # x - r >= 0  => violation max(0, r - x)
    # 1 - x - r >= 0 => violation max(0, r + x - 1)
    dx1 = np.maximum(0.0, radii - centers[:, 0])
    dx2 = np.maximum(0.0, radii + centers[:, 0] - 1.0)
    dy1 = np.maximum(0.0, radii - centers[:, 1])
    dy2 = np.maximum(0.0, radii + centers[:, 1] - 1.0)
    
    E = np.sum(dx1**2 + dx2**2 + dy1**2 + dy2**2)
    
    # Pairwise overlap violations
    # distance between centers >= sum of radii
    # Compute pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    
    r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    
    overlaps = np.maximum(0.0, r_sum - dists)
    
    # Sum over unique pairs (upper triangle) to avoid double counting
    mask = np.triu(np.ones((N, N), dtype=bool), k=1)
    E += np.sum(overlaps[mask]**2)
    
    return E

def optimization_objective(pos_flat, radii):
    """
    Wrapper for scipy optimizer to match required signature.
    """
    N = len(radii)
    centers = pos_flat.reshape(N, 2)
    return compute_energy(centers, radii)

        x = margin + (r % 2) * (col_spacing * 0.5)
        for c in range(7): # Enough cols to cover N
            if idx < N:
                centers[idx] = [x, y]
                x += col_spacing
                idx += 1
        y += row_spacing
        
    # Initial radii
    radii = np.full(N, 0.06)
    
    best_centers = centers.copy()
    best_radii = radii.copy()
    
    # Optimization parameters
    growth_factor = 1.005
    min_growth = 1.0001
    energy_threshold = 1e-9
    
    # Iterative optimization loop
    for _ in range(300):
        pos_flat = centers.flatten()
        bounds = [(0.0, 1.0)] * (2 * N)
        
        # Optimize positions to minimize energy for fixed radii
        res = minimize(optimization_objective, pos_flat, args=(radii,), 
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': 100, 'ftol': 1e-12, 'gtol': 1e-10})

        current_energy = res.fun

            # Configuration is valid, try to grow radii
            radii = radii * growth_factor
            best_centers = centers.copy()
            best_radii = radii.copy()
        else:
            # Configuration invalid, reduce growth factor to refine
            growth_factor *= 0.95
            if growth_factor < min_growth:
                growth_factor = min_growth
                
    return best_centers, best_radii, float(np.sum(best_radii))
