import numpy as np
from scipy.optimize import minimize
from functools import partial

def validate_helper(centers, radii):
    """Top-level helper to strictly validate packing configuration."""
    n = len(radii)
    for i in range(n):
        if centers[i, 0] - radii[i] < -1e-9 or centers[i, 0] + radii[i] > 1.0 + 1e-9:
            return False
        if centers[i, 1] - radii[i] < -1e-9 or centers[i, 1] + radii[i] > 1.0 + 1e-9:
            return False
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt((centers[i, 0] - centers[j, 0])**2 + (centers[i, 1] - centers[j, 1])**2)
            if dist < radii[i] + radii[j] - 1e-9:
                return False
    return True

def obj_func(vars):
    """Top-level objective: minimize negative sum of radii."""
    return -np.sum(vars[2::3])

def cons_func(vars, n):
    """Top-level constraint function: returns array of constraint values >= 0."""
    c = []
    # Boundary constraints
    for i in range(n):
        x, y, r = vars[3*i], vars[3*i+1], vars[3*i+2]
        c.append(x - r)
        c.append(1.0 - (x + r))
        c.append(y - r)
        c.append(1.0 - (y + r))
        
    # Pairwise non-overlap constraints
    for i in range(n):
        xi, yi, ri = vars[3*i], vars[3*i+1], vars[3*i+2]
        for j in range(i + 1, n):
            xj, yj, rj = vars[3*j], vars[3*j+1], vars[3*j+2]
            dx = xi - xj
            dy = yi - yj
            dr = ri + rj
            c.append(dx*dx + dy*dy - dr*dr)
            
    return np.array(c)

def run_packing():

    n = 26
    np.random.seed(42)

    rows_config = [5, 5, 6, 5, 5]

        shift = 0.06 if i % 2 == 1 else 0.0
        x_coords = np.linspace(0.08 + shift, 0.92 + shift, n_cols)

            base_centers.append([x, y_coords[i]])
    base_centers = np.array(base_centers)
    
    r_init = 0.06
    bounds = [(0.0, 1.0) for _ in range(n * 3)]

        bounds[i + 2] = (0.0, 0.5)

    best_radii = None

        for i in range(n):
            cur_x0[3*i] = cur_centers[i, 0]
            cur_x0[3*i+1] = cur_centers[i, 1]
            cur_x0[3*i+2] = r_init

                break
                
    return best_centers, best_radii, np.sum(best_radii)
