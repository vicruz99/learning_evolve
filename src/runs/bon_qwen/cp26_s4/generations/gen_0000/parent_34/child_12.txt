from scipy.optimize import minimize

    # ...

        for x in x_coords:
            points.append([x, y])
            if len(points) >= 26:
                break
        if len(points) >= 26:
            break
            
    centers = np.array(points[:26])
    
    # Initial radii: small enough to not overlap

    radii = np.full(n, 0.05)

    x0 = np.zeros(3 * n)
    for i in range(n):
        x0[3*i] = centers[i, 0]
        x0[3*i+1] = centers[i, 1]
        x0[3*i+2] = radii[i]

        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 0.5)) # r

    # We need to sum over i < j
    triu_mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    
    def objective(v, lam):
        # Extract variables
        xs = v[0::3]
        ys = v[1::3]
        rs = v[2::3]
        
        # Objective: maximize sum of radii => minimize -sum
        sum_r = np.sum(rs)
        cost = -sum_r
        
        # Boundary penalties
        # x - r >= 0  => penalty if x < r
        # 1 - x - r >= 0 => penalty if 1-x < r
        # Same for y
        
        # Vectorized calculation
        # Use np.maximum(0, violation)**2
        
        # Left wall violation: r - x (if positive)
        pen_x1 = np.maximum(0, rs - xs)**2
        # Right wall violation: r - (1-x)
        pen_x2 = np.maximum(0, rs - (1.0 - xs))**2
        # Bottom wall
        pen_y1 = np.maximum(0, rs - ys)**2
        # Top wall
        pen_y2 = np.maximum(0, rs - (1.0 - ys))**2
        
        cost += lam * np.sum(pen_x1 + pen_x2 + pen_y1 + pen_y2)
        
        # Overlap penalties
        # Constraint: dist >= r_i + r_j
        # Equivalent to: dist^2 >= (r_i + r_j)^2
        # Violation: (r_i + r_j)^2 - dist^2 > 0

        dx = xs[:, None] - xs[None, :]
        dy = ys[:, None] - ys[None, :]

        # Compute squared sum of radii
        r_sum = rs[:, None] + rs[None, :]

        gap = gap * triu_mask
        
        # Penalty for positive gap

        cost += lam * np.sum(overlap_pen)
        
        return cost

        # maxiter can be increased if needed, but 1000 is usually enough for local refinement
        res = minimize(
            objective, 
            current_v, 
            args=(lam,),
            method='L-BFGS-B', 
            bounds=bounds,
            options={'maxiter': 2000, 'ftol': 1e-15, 'gtol': 1e-12}
        )
        current_v = res.x
        
        # Check convergence or validity?

    xs_opt = current_v[0::3]
    ys_opt = current_v[1::3]
    rs_opt = current_v[2::3]
    
    centers_opt = np.column_stack([xs_opt, ys_opt])
    sum_radii = np.sum(rs_opt)

    # Although optimizer should handle it, float precision might cause tiny overlaps.

    sum_radii = np.sum(rs_opt)
    
    return centers_opt, rs_opt, sum_radii
