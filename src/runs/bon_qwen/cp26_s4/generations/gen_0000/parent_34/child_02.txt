**Mathematical Formulation:**
Maximize $\sum_{i=1}^{26} r_i$ subject to:
1.  **Boundary Constraints:** $r_i \le x_i \le 1-r_i$ and $r_i \le y_i \le 1-r_i$ (circles must be inside $[0,1]^2$).
2.  **Non-Overlap Constraints:** $(x_i-x_j)^2 + (y_i-y_j)^2 \ge (r_i+r_j)^2$ for all $i \neq j$.
3.  **Non-negativity:** $r_i \ge 0$.

1.  **Initialization:** Generate valid starting configurations using geometric patterns (hexagonal and square grids) and random sampling. Hexagonal patterns are chosen because they offer higher packing density ($\approx 0.907$) than square grids ($\approx 0.785$).

3.  **Multi-Start Strategy:** Since the objective landscape is non-convex, run the optimization from multiple diverse starting points to avoid getting stuck in local minima (e.g., a $5 \times 5$ grid with sum 2.5).
4.  **Constraint Smoothing:** The non-overlap constraints are differentiable. The solver will naturally push circles apart to increase their radii.

import numpy as np
from scipy.optimize import minimize
import math

def run_packing():
    """
    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    
    Returns:
        tuple: (centers, radii, sum_radii)
    """
    n = 26
    best_sum_r = 0.0
    best_centers = None
    best_radii = None

    # Objective: Maximize sum of radii -> Minimize negative sum
    def objective(vars):
        # vars = [x0, y0, r0, x1, y1, r1, ...]
        r_sum = 0
        for i in range(n):
            r_sum += vars[3 * i + 2]
        return -r_sum

    # Constraints
    # 1. Boundary constraints: r <= x <= 1-r  =>  x - r >= 0 and 1 - x - r >= 0
    # 2. Non-overlap: dist^2 >= (r1 + r2)^2
    
    def make_constraints():
        cons = []
        
        # Boundary constraints
        for i in range(n):
            # x_i >= r_i
            cons.append({
                'type': 'ineq',
                'fun': lambda vars, i=i: vars[3 * i] - vars[3 * i + 2]
            })
            # 1 - x_i >= r_i
            cons.append({
                'type': 'ineq',
                'fun': lambda vars, i=i: 1.0 - vars[3 * i] - vars[3 * i + 2]
            })
            # y_i >= r_i
            cons.append({
                'type': 'ineq',
                'fun': lambda vars, i=i: vars[3 * i + 1] - vars[3 * i + 2]
            })
            # 1 - y_i >= r_i
            cons.append({
                'type': 'ineq',
                'fun': lambda vars, i=i: 1.0 - vars[3 * i + 1] - vars[3 * i + 2]
            })
            
        # Non-overlap constraints
        for i in range(n):
            for j in range(i + 1, n):
                def constraint_overlap(vars, i=i, j=j):
                    x_i, y_i = vars[3 * i], vars[3 * i + 1]
                    x_j, y_j = vars[3 * j], vars[3 * j + 1]
                    r_i, r_j = vars[3 * i + 2], vars[3 * j + 2]
                    dist_sq = (x_i - x_j)**2 + (y_i - y_j)**2
                    return dist_sq - (r_i + r_j)**2
                
                cons.append({'type': 'ineq', 'fun': constraint_overlap})
                
        return cons

    constraints = make_constraints()

    # Variable bounds: x, y in [0, 1], r in [0, 0.5]
    bounds = []
    for _ in range(n):
        bounds.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])

    def generate_hexagonal_init():
        centers = np.zeros((n, 2))
        idx = 0
        # Try to fit a hexagonal lattice
        # Vertical spacing sqrt(3)*r, Horizontal spacing 2r
        # Offset odd rows by r
        r_est = 0.105 # Estimated optimal radius
        row = 0
        while idx < n:
            y = r_est * (1 + row * math.sqrt(3))
            if y + r_est > 1.0:
                row += 1
                continue
            
            col = 0
            while idx < n:
                x = r_est * (1 + 2 * col)
                if row % 2 == 1:
                    x += r_est # Shift odd rows
                
                if x + r_est <= 1.0 + 1e-9:
                    centers[idx] = [x, y]
                    idx += 1
                    col += 1
                else:
                    break
            row += 1
        
        # Pad or truncate if necessary (though loop ensures n)
        radii = np.ones(n) * r_est * 0.5 # Start with smaller radii to be valid
        return centers, radii

    def generate_random_init():
        centers = np.random.uniform(0.1, 0.9, (n, 2))
        radii = np.ones(n) * 0.05
        return centers, radii

    def run_optimization(initial_centers, initial_radii):
        vars0 = []
        for i in range(n):
            vars0.extend([initial_centers[i, 0], initial_centers[i, 1], initial_radii[i]])
        
        # Run SLSQP
        try:
            res = minimize(
                objective,
                vars0,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 1000, 'ftol': 1e-12}
            )
            
            if res.success:
                sum_r = -res.fun
                # Extract solution
                centers = np.zeros((n, 2))
                radii = np.zeros(n)
                for i in range(n):
                    centers[i, 0] = res.x[3 * i]
                    centers[i, 1] = res.x[3 * i + 1]
                    radii[i] = res.x[3 * i + 2]
                return centers, radii, sum_r
        except Exception:
            pass
        return None, None, 0

    # Try Hexagonal Initialization (likely closest to optimal structure)
    # Run a few variations of radius estimate for initialization
    for r_start in [0.1, 0.105, 0.11]:
        centers_hex, radii_hex = generate_hexagonal_init()
        # Scale radii for init to ensure validity
        radii_hex[:] = r_start * 0.6 
        # Adjust centers slightly if needed, but hex grid logic is robust
        c, r, s = run_optimization(centers_hex, radii_hex)
        if s > best_sum_r:
            best_sum_r = s
            best_centers = c
            best_radii = r

    # Try Random Initializations
    for _ in range(5):
        centers_rand, radii_rand = generate_random_init()
        c, r, s = run_optimization(centers_rand, radii_rand)
        if s > best_sum_r:
            best_sum_r = s
            best_centers = c
            best_radii = r

    # Fallback/Refinement: If we have a solution, try to optimize again from that solution
    if best_centers is not None:
        # Perturb slightly and re-optimize
        centers_pert = best_centers + np.random.normal(0, 0.001, best_centers.shape)
        radii_pert = best_radii * (1 + np.random.normal(0, 0.01, best_radii.shape))
        # Clip to ensure validity roughly
        radii_pert = np.maximum(0, radii_pert)
        c, r, s = run_optimization(centers_pert, radii_pert)
        if s > best_sum_r:
            best_sum_r = s
            best_centers = c
            best_radii = r

    return best_centers, best_radii, best_sum_r
