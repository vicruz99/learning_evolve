from scipy.optimize import minimize

def compute_loss(p, n, mu):
    """Computes objective + penalty for circle packing configuration."""
    cx = p[:n]
    cy = p[n:2*n]
    cr = p[2*n:]
    
    # Objective: maximize sum of radii -> minimize negative sum
    obj = -np.sum(cr)
    pen = 0.0
    
    # Circle-circle overlap penalty
    for i in range(n):
        for j in range(i + 1, n):
            dx = cx[i] - cx[j]
            dy = cy[i] - cy[j]
            dist = np.sqrt(dx*dx + dy*dy)
            overlap = cr[i] + cr[j] - dist
            if overlap > 0:
                pen += mu * overlap * overlap
                
    # Boundary constraint penalties
    for i in range(n):
        if cx[i] < cr[i]: pen += mu * (cx[i] - cr[i])**2
        if cx[i] > 1 - cr[i]: pen += mu * (cx[i] + cr[i] - 1)**2
        if cy[i] < cr[i]: pen += mu * (cy[i] - cr[i])**2
        if cy[i] > 1 - cr[i]: pen += mu * (cy[i] + cr[i] - 1)**2
        if cr[i] < 0: pen += mu * cr[i]**2
        
    return obj + pen
