    max_outer_iter = 1000 # Number of growth steps
    max_inner_iter = 50   # Number of relaxation steps per growth
    
    for _ in range(max_outer_iter):
        # Attempt to grow
        radii += growth_rate
        
        # Relaxation: resolve overlaps and boundary violations
        for _ in range(max_inner_iter):

            has_conflict = False
            
            # Circle-Circle repulsion
            for i in range(n):
                for j in range(i + 1, n):
                    diff = centers[i] - centers[j]
                    dist = np.sqrt(np.sum(diff**2))
                    min_dist = radii[i] + radii[j]
                    
                    if dist < min_dist and dist > 1e-9:
                        has_conflict = True
                        # Force proportional to overlap
                        force_mag = (min_dist - dist) * 0.5 # 0.5 factor for stability

                        forces[i] += force_vec
                        forces[j] -= force_vec

                    has_conflict = True
                # Bottom
                if centers[i, 1] < radii[i]:
                    forces[i, 1] += (radii[i] - centers[i, 1]) * 2.0
                    has_conflict = True
                # Top
                if centers[i, 1] > 1 - radii[i]:
                    forces[i, 1] -= (centers[i, 1] - (1 - radii[i])) * 2.0
                    has_conflict = True
            
            # Apply forces
            if has_conflict:
                centers += forces * 0.5 # Damping factor
            else:
                break # No conflicts, stable
        
        # Check validity after relaxation
        valid = True
        
        # Check boundaries
        for i in range(n):
            if centers[i, 0] < radii[i] - 1e-7 or centers[i, 0] > 1 - radii[i] + 1e-7 or \
               centers[i, 1] < radii[i] - 1e-7 or centers[i, 1] > 1 - radii[i] + 1e-7:
                valid = False
                break
        
        if valid:
            # Check overlaps
            for i in range(n):
                for j in range(i + 1, n):
                    dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                    if dist < radii[i] + radii[j] - 1e-7:
                        valid = False
                        break
                if not valid:
                    break
        
        if not valid:
            # If invalid, revert growth and continue (try to relax more? or stop growing)
            # Reverting radius allows us to stay valid but stop increasing.
            # However, we might just be stuck.
            # To be safe, reduce radius slightly to ensure validity.
            radii -= growth_rate
            
        # Optional: If we are very stable, we can increase growth rate?
        # But for safety, keep constant.
        
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
