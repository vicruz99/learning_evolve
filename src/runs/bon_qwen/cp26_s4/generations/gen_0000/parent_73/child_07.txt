import numpy as np
from scipy.optimize import minimize

def evaluate_constraints(v, N):
    """Evaluate all inequality constraints for the packing problem."""
    con = []
    # Boundary constraints: x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
    for i in range(N):
        con.append(v[3*i] - v[3*i+2])
        con.append(1.0 - v[3*i] - v[3*i+2])
        con.append(v[3*i+1] - v[3*i+2])
        con.append(1.0 - v[3*i+1] - v[3*i+2])
        
    # Overlap constraints: distance(r_i, r_j) >= r_i + r_j
    for i in range(N):
        for j in range(i + 1, N):
            dx = v[3*i] - v[3*j]
            dy = v[3*i+1] - v[3*j+1]
            dist = np.sqrt(dx*dx + dy*dy)
            con.append(dist - v[3*i+2] - v[3*j+2])
            
    return np.array(con)

def run_packing():
    """Pack 26 circles in a unit square to maximize the sum of radii."""
    N = 26
    # Hexagonal row configuration summing to 26
    row_counts = [5, 6, 5, 6, 4]
    
    # Initial spacing based on hexagonal packing geometry
    dx = 0.2
    dy = dx * np.sqrt(3) / 2
    r_init = 0.09
    
    # Generate initial centers
    centers = []
    for j, count in enumerate(row_counts):
        y = r_init + j * dy
        row_width = (count - 1) * dx
        x_start = (1.0 - row_width) / 2.0
        for i in range(count):
            x = x_start + i * dx
            centers.append([x, y])
            
    centers = np.array(centers)
    radii = np.full(N, r_init)
    
    # Flatten initial state for optimizer: [x0, y0, r0, x1, y1, r1, ...]
    x0 = np.zeros(3 * N)
    for i in range(N):
        x0[3*i] = centers[i, 0]
        x0[3*i+1] = centers[i, 1]
        x0[3*i+2] = radii[i]
        
    # Variable bounds: x,y in [0,1], r in [1e-5, 0.5]
    bounds = [(0.0, 1.0)] * N + [(0.0, 1.0)] * N + [(1e-5, 0.5)] * N
    
    # Constraint wrapper (avoids lambda/closure restrictions)
    def con_fun(v):
        return evaluate_constraints(v, N)
        
    cons = [{'type': 'ineq', 'fun': con_fun}]
        
    # Objective: maximize sum of radii -> minimize negative sum
    def obj(v):
        return -np.sum(v[2::3])
        
    # Run constrained optimization
    res = minimize(obj, x0, method='SLSQP', bounds=bounds, constraints=cons, 
                   options={'maxiter': 2000, 'ftol': 1e-9, 'disp': False})
                   
    best_v = res.x
        
    # Unpack results
    final_x = best_v[:N]
    final_y = best_v[N:2*N]
    final_r = best_v[2*N:]
    
    return np.column_stack((final_x, final_y)), final_r, np.sum(final_r)
