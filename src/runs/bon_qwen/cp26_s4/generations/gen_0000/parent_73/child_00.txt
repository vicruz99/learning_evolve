import numpy as np
from scipy.optimize import minimize

def compute_loss(z, N, mu):
    """
    Computes the negative sum of radii plus penalty terms for constraint violations.
    z: flattened array of [x0, y0, r0, x1, y1, r1, ...]
    N: number of circles
    mu: penalty weight
    """
    cs = z[:2*N].reshape(N, 2)
    rs = z[2*N:]
    val = -np.sum(rs)

    # Boundary penalties: circles must stay within [0,1]x[0,1]
    pen = np.maximum(0, rs - cs[:,0])**2
    pen += np.maximum(0, rs - cs[:,1])**2
    pen += np.maximum(0, rs + cs[:,0] - 1)**2
    pen += np.maximum(0, rs + cs[:,1] - 1)**2
    val += mu * np.sum(pen)

    # Overlap penalties: circles must not intersect
    diffs = cs[:, None, :] - cs[None, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2) + 1e-12)
    np.fill_diagonal(dists, 2.0)  # Ensure diagonal is large so self-overlap is ignored
    overlaps = rs[:, None] + rs[None, :] - dists
    val += mu * np.sum(np.maximum(0, overlaps)**2)
    return val

def validate_config(centers, radii, N):
    """Quick validation to filter out severely invalid configurations during search."""
    for i in range(N):
        if centers[i,0] < radii[i] or centers[i,0] > 1 - radii[i]: return False
        if centers[i,1] < radii[i] or centers[i,1] > 1 - radii[i]: return False
    for i in range(N):
        for j in range(i+1, N):
            if np.linalg.norm(centers[i] - centers[j]) < radii[i] + radii[j] - 1e-7:
                return False
    return True

def run_packing():
    N = 26
    best_sum = -np.inf
    best_centers = None
    best_radii = None

    # Bounds for optimization: x,y in (0.005, 0.995), r in (0.005, 0.5)
    bounds = [(0.005, 0.995)] * 2*N + [(0.005, 0.5)] * N

    for seed in range(10):
        np.random.seed(seed * 137 + 7)
        # Hexagonal arrangement: rows of 6, 5, 6, 5, 4 circles
        rows = [6, 5, 6, 5, 4]
        c_list = []
        y_pos = 0.15
        dy = 0.17
        for cnt in rows:
            spacing = 0.175
            x_start = (1.0 - (cnt - 1) * spacing) / 2.0
            xs = x_start + np.arange(cnt) * spacing
            xs += np.random.randn(cnt) * 0.015
            ys = np.full(cnt, y_pos) + np.random.randn(cnt) * 0.015
            c_list.extend(np.column_stack([xs, ys]))
            y_pos += dy * np.sqrt(3) / 2

        centers = np.clip(np.array(c_list), 0.1, 0.9)
        radii = np.ones(N) * 0.08
        x0 = np.concatenate([centers.ravel(), radii])

        z = x0.copy()
        mu = 50.0
        # Anneal penalty weight to strictly enforce constraints
        for _ in range(25):
            res = minimize(compute_loss, z, args=(N, mu), method='L-BFGS-B', bounds=bounds, options={'maxiter': 800})
            z = res.x
            mu *= 1.8
            if res.success:
                base_loss = compute_loss(z, N, 0.0)
                penalized_loss = compute_loss(z, N, mu)
                if penalized_loss - base_loss < 1e-5:
                    break

        cs = z[:2*N].reshape(N, 2)
        rs = z[2*N:]
        curr_sum = np.sum(rs)

        if validate_config(cs, rs, N) and curr_sum > best_sum:
            best_sum = curr_sum
            best_centers = cs.copy()
            best_radii = rs.copy()

    if best_centers is not None:
        # Slight shrink to guarantee strict validity against numerical tolerances
        best_radii *= 0.9999
        final_sum = np.sum(best_radii)
        return best_centers, best_radii, final_sum
    else:
        centers = np.tile([[0.5, 0.5]], (N, 1))
        radii = np.zeros(N)
        return centers, radii, 0.0
