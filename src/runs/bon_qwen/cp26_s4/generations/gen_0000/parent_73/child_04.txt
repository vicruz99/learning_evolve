        best_r = 0.05
        best_centers = np.zeros((n, 2))
        idx = 0
        # Generate a valid grid packing
        for r in range(10):
            for c in range(10):
                if idx >= n: break
                best_centers[idx] = [0.05 + c*0.1, 0.05 + r*0.1]
                idx += 1
            if idx >= n: break

    radii = np.full(26, best_r)
    return best_centers, radii, 26 * best_r
