import numpy as np

def run_packing():
    np.random.seed(42)
    N = 26
    
    # 1. Initialize centers in a staggered grid pattern
    centers = np.zeros((N, 2))
    idx = 0
    for r in range(6):
        for c in range(6):
            if idx >= N: break
            # Hexagonal-ish spacing initialization
            x = 0.15 + c * 0.17 + (r % 2) * 0.085
            y = 0.15 + r * 0.17
            centers[idx] = [x, y]
            idx += 1
            
    # Perturb to break symmetry and avoid degenerate configurations
    centers += np.random.uniform(-0.005, 0.005, (N, 2))
    centers = np.clip(centers, 0.05, 0.95)
    
    # Start with moderate radius, will be expanded during simulation
    radii = np.full(N, 0.065)
    velocities = np.zeros((N, 2))
    
    # Simulation parameters
    dt = 0.001
    damping = 0.93
    k_repulse = 100.0
    k_bound = 200.0
    
    num_steps = 70000
    radius_increment = 1.0e-6
    
    # 2. Force-directed simulation with gradual expansion
    for step in range(num_steps):
        forces = np.zeros((N, 2))
        
        # Circle-circle repulsion
        for i in range(N):
            for j in range(i + 1, N):
                diff = centers[i] - centers[j]
                dist = np.linalg.norm(diff)
                min_dist = radii[i] + radii[j]
                if dist < min_dist and dist > 1e-9:
                    overlap = min_dist - dist
                    force_mag = k_repulse * overlap
                    forces[i] += (diff / dist) * force_mag
                    forces[j] -= (diff / dist) * force_mag
                    
        # Boundary repulsion
        for i in range(N):
            x, y = centers[i]
            r = radii[i]
            if x - r < 0:
                forces[i, 0] += k_bound * (r - x)
            if x + r > 1:
                forces[i, 0] -= k_bound * (x + r - 1)
            if y - r < 0:
                forces[i, 1] += k_bound * (r - y)
            if y + r > 1:
                forces[i, 1] -= k_bound * (y + r - 1)
                
        # Update dynamics
        velocities = damping * velocities + forces * dt
        centers += velocities * dt
        
        # Expand radii over time
        radii += radius_increment
        
        # Keep centers safely within bounds during simulation
        centers = np.clip(centers, 1e-4, 1 - 1e-4)
        
    # 3. Final adjustment to guarantee strict validity
    # Shrink radii minimally to resolve any remaining numerical overlaps
    for _ in range(100):
        overlap_found = False
        for i in range(N):
            # Boundary constraints
            for d in range(2):
                if centers[i, d] - radii[i] < 0:
                    radii[i] = centers[i, d]
                    overlap_found = True
                if centers[i, d] + radii[i] > 1:
                    radii[i] = 1 - centers[i, d]
                    overlap_found = True
                    
            # Pairwise constraints
            for j in range(i + 1, N):
                dist = np.linalg.norm(centers[i] - centers[j])
                if dist < radii[i] + radii[j]:
                    shrink = (radii[i] + radii[j] - dist) / 2 + 1e-7
                    radii[i] -= shrink
                    radii[j] -= shrink
                    overlap_found = True
        if not overlap_found:
            break
            
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
