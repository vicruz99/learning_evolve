2. Boundary containment: $r_i \le x_i, y_i \le 1 - r_i$

import numpy as np
from scipy.optimize import minimize

# Precompute pair indices for the 26 circles to avoid repeated allocation
_N = 26
_PAIR_I, _PAIR_J = np.tril_indices(_N, -1)

def _evaluate_objective(v, w):
    """
    Compute the objective: negative sum of radii + penalty for constraint violations.
    v: flattened array [x1, y1, r1, x2, y2, r2, ...]
    w: penalty weight
    """
    c = v[:2 * _N].reshape(_N, 2)
    r = v[2 * _N:]
    
    # 1. Overlap penalty: max(0, r_i + r_j - d_ij)^2
    dx = c[_PAIR_I, 0] - c[_PAIR_J, 0]
    dy = c[_PAIR_I, 1] - c[_PAIR_J, 1]
    d = np.sqrt(dx**2 + dy**2)
    overlaps = r[_PAIR_I] + r[_PAIR_J] - d
    penalty = np.sum(np.maximum(overlaps, 0.0)**2)
    
    # 2. Boundary penalty: max(0, r_i - x_i)^2 + max(0, x_i + r_i - 1)^2 (and y)
    penalty += np.sum(np.maximum(r - c[:, 0], 0.0)**2)
    penalty += np.sum(np.maximum(c[:, 0] + r - 1.0, 0.0)**2)
    penalty += np.sum(np.maximum(r - c[:, 1], 0.0)**2)
    penalty += np.sum(np.maximum(c[:, 1] + r - 1.0, 0.0)**2)
    
    return -np.sum(r) + w * penalty

def _strictify(c, r):
    """
    Post-optimization cleanup to guarantee strict validity within 1e-12 tolerance.
    """
    # Fix boundary violations
    for i in range(_N):
        # Left
        if c[i, 0] < r[i]:
            r[i] = c[i, 0]
        # Right
        if c[i, 0] > 1.0 - r[i]:
            r[i] = 1.0 - c[i, 0]
        # Bottom
        if c[i, 1] < r[i]:
            r[i] = c[i, 1]
        # Top
        if c[i, 1] > 1.0 - r[i]:
            r[i] = 1.0 - c[i, 1]
            
        if r[i] < 1e-9:
            r[i] = 1e-9
            
    # Fix overlap violations by proportional scaling
    # Iterate a few times to resolve chain reactions
    for _ in range(5):
        dx = c[_PAIR_I, 0] - c[_PAIR_J, 0]
        dy = c[_PAIR_I, 1] - c[_PAIR_J, 1]
        d = np.sqrt(dx**2 + dy**2)
        required = r[_PAIR_I] + r[_PAIR_J]
        violation = required - d
        
        idxs = np.where(violation > 1e-11)[0]
        if len(idxs) == 0:
            break
            
        for k in idxs:
            i, j = _PAIR_I[k], _PAIR_J[k]
            scale = d[k] / required[k]
            # Distribute shrinkage proportionally to current radii
            total = r[i] + r[j]
            r[i] = r[i] * (scale * 0.99999) * (r[i]/total + r[j]/total) # Simplified: just scale both slightly
            r[j] = r[j] * (scale * 0.99999)
            
    return r

def run_packing():
    best_sum = 0.0
    best_c = None
    best_r = None
    
    # Optimization bounds
    bounds = [(0.0, 1.0)] * (2 * _N) + [(0.0, 0.5)] * _N
    
    # Run multiple restarts to escape local minima
    for restart in range(20):
        rng = np.random.default_rng(restart * 17 + 42)
        
        # Initialize with a perturbed hexagonal grid
        pts = []
        for row in range(6):
            for col in range(5):
                x = 0.15 + col * 0.175
                y = 0.15 + row * 0.158
                if col % 2 == 1:
                    y += 0.08
                if len(pts) < _N:
                    pts.append([x, y])
        pts = np.array(pts[:_N])
        
        # Add controlled noise
        pts += rng.uniform(-0.025, 0.025, size=pts.shape)
        pts = np.clip(pts, 0.05, 0.95)
        
        r0 = np.full(_N, 0.045)
        v0 = np.concatenate([pts.ravel(), r0])
        
        # Sequential penalty annealing
        v = v0.copy()
        weights = [30.0, 150.0, 800.0, 4000.0]
        
        for w in weights:
            res = minimize(_evaluate_objective, v, args=(w,), 
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 4000, 'ftol': 1e-12, 'gtol': 1e-9})
            v = res.x
            
        c = v[:2 * _N].reshape(_N, 2)
        r = v[2 * _N:]
        
        # Ensure strict validity
        r = _strictify(c, r)
        
        current_sum = np.sum(r)
        if current_sum > best_sum:
            best_sum = current_sum
            best_c = c.copy()
            best_r = r.copy()
            
    return best_c, best_r, best_sum
