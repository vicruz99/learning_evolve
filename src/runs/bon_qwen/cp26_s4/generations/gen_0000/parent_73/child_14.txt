import numpy as np
import scipy.optimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # 1. Initialization: Hexagonal Packing
    # We arrange circles in rows with alternating counts to mimic hex packing.
    # 6 + 5 + 6 + 5 + 4 = 26 circles.
    # Initial radius small enough to fit easily.
    r_init = 0.08
    
    centers = []
    radii = []
    
    # Row configurations: (count, shift_x)
    # Standard hex shift is r, but we will center them manually.
    row_counts = [6, 5, 6, 5, 4]
    row_y = 0
    y_spacing = r_init * np.sqrt(3)
    
    for k, count in enumerate(row_counts):
        # Calculate x coordinates for this row
        # Width required for 'count' circles with radius r is 2*r*count
        # But in hex packing, horizontal spacing is 2r.
        # We center the row in the square.
        row_width = 2 * r_init * count
        start_x = (1.0 - row_width) / 2
        
        for i in range(count):
            cx = start_x + i * 2 * r_init
            # Shift odd rows by r to create hex pattern
            if k % 2 == 1:
                cx += r_init
            cy = r_init + k * y_spacing
            centers.append([cx, cy])
            radii.append(r_init)
    
    centers = np.array(centers)
    radii = np.array(radii)
    
    # 2. Define Constraints for Optimization
    # Variables: [x1, y1, r1, x2, y2, r2, ...]
    # Total 3 * n variables.
    
    def get_params(centers, radii):
        # Interleave to create vector: [x1, y1, r1, x2, y2, r2, ...]
        # Actually, scipy likes flat arrays.
        # Let's map indices:
        # x_i -> 3*i, y_i -> 3*i+1, r_i -> 3*i+2
        params = np.zeros(3 * n)
        for i in range(n):
            params[3*i] = centers[i, 0]
            params[3*i+1] = centers[i, 1]
            params[3*i+2] = radii[i]
        return params

    def get_centers_radii(params):
        centers = np.zeros((n, 2))
        radii = np.zeros(n)
        for i in range(n):
            centers[i, 0] = params[3*i]
            centers[i, 1] = params[3*i+1]
            radii[i] = params[3*i+2]
        return centers, radii

    # Objective: Minimize negative sum of radii
    def objective(params):
        _, radii = get_centers_radii(params)
        return -np.sum(radii)

    # Constraints
    # Boundary constraints: 0 <= x-r, x+r <= 1, same for y, r >= 0
    # We will handle these via bounds in minimize if possible, or constraints.
    # SLSQP supports bounds.
    
    # Bounds:
    # x in [0, 1], y in [0, 1], r in [0, 0.5] (loose upper bound)
    # But strictly x-r >= 0 => x >= r. This is coupled.
    # So we use explicit constraints for boundaries involving r.
    
    constraints = []
    
    # Boundary constraints
    for i in range(n):
        # x_i - r_i >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda p, i=i: p[3*i] - p[3*i+2]
        })
        # 1 - (x_i + r_i) >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda p, i=i: 1.0 - (p[3*i] + p[3*i+2])
        })
        # y_i - r_i >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda p, i=i: p[3*i+1] - p[3*i+2]
        })
        # 1 - (y_i + r_i) >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda p, i=i: 1.0 - (p[3*i+1] + p[3*i+2])
        })
        # r_i >= 0 (handled by bounds, but safe to include or just use bounds)
        
    # Overlap constraints: dist^2 >= (r_i + r_j)^2
    # dist^2 - (r_i + r_j)^2 >= 0
    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({
                'type': 'ineq',
                'fun': lambda p, i=i, j=j: 
                    (p[3*i] - p[3*j])**2 + (p[3*i+1] - p[3*j+1])**2 - (p[3*i+2] + p[3*j+2])**2
            })

    # Set bounds for parameters to help optimizer
    # x, y in [0, 1], r in [0, 0.5]
    bounds = []
    for _ in range(n):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((1e-6, 0.5)) # r (small epsilon to avoid 0 issues)

    # 3. Optimization
    initial_params = get_params(centers, radii)
    
    # Use SLSQP which handles non-linear constraints and bounds
    try:
        result = scipy.optimize.minimize(
            objective,
            initial_params,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 1000, 'ftol': 1e-12, 'disp': False}
        )
        
        if result.success:
            centers_opt, radii_opt = get_centers_radii(result.x)
        else:
            # Fallback to initial if optimization fails
            centers_opt, radii_opt = centers, radii
            print("Optimization did not succeed, using initial configuration.")
            
    except Exception as e:
        print(f"Optimization error: {e}")
        centers_opt, radii_opt = centers, radii

    # 4. Validation and Refinement
    # Sometimes numerical errors cause slight violations.
    # We can run a quick check and shrink slightly if needed, 
    # but the validator allows 1e-12 tolerance.
    
    # Let's ensure strict validity for safety
    # If any overlap, scale down radii slightly? 
    # Or just trust the optimizer with tight tolerance.
    
    sum_radii = np.sum(radii_opt)
    
    return centers_opt, radii_opt, sum_radii

# Helper to run and validate locally if needed
if __name__ == "__main__":
    centers, radii, s = run_packing()
