import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def objective(vars):
    """Objective function: negative sum of radii (to be minimized)"""
    return -np.sum(vars[2::3])

def constraints(vars):
    """Inequality constraints: boundary and non-overlap"""
    N = N_CIRCLES
    X = vars[0::3]
    Y = vars[1::3]
    R = vars[2::3]
    
    # Boundary constraints: x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
    con = np.concatenate([
        X - R,
        1.0 - X - R,
        Y - R,
        1.0 - Y - R
    ])
    
    # Non-overlap constraints: ||P_i - P_j||^2 - (r_i + r_j)^2 >= 0
    i_idx, j_idx = np.triu_indices(N, k=1)
    dx = X[i_idx] - X[j_idx]
    dy = Y[i_idx] - Y[j_idx]
    dist_sq = dx**2 + dy**2
    r_sum = R[i_idx] + R[j_idx]
    
    return np.concatenate([con, dist_sq - r_sum**2])

def run_packing():
    N = N_CIRCLES
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    cons = {'type': 'ineq', 'fun': constraints}
    bounds = [(0.0, 1.0)] * (2*N) + [(0.0, 0.5)] * N
    
    # Run multiple optimizations with perturbed initial positions to escape local minima
    np.random.seed(42)
    for seed in range(5):
        np.random.seed(seed)
        
        # Initialize on a hexagonal grid
        init_pos = []
        dx = 1.0 / 7.0
        dy = dx * np.sqrt(3) / 2
        count = 0
        for r in range(6):
            for c in range(6):
                if count >= N: 
                    break
                x = dx * (c + 1) + (r % 2) * dx * 0.5
                y = dy * r + dy * 0.5
                # Add controlled noise
                x += np.random.uniform(-0.03, 0.03)
                y += np.random.uniform(-0.03, 0.03)
                init_pos.append([x, y])
                count += 1
                
        vars0 = np.zeros(3 * N)
        for i in range(N):
            vars0[3*i] = np.clip(init_pos[i][0], 0.05, 0.95)
            vars0[3*i+1] = np.clip(init_pos[i][1], 0.05, 0.95)
            vars0[3*i+2] = 0.06  # Start with feasible small radii
            
        res = minimize(objective, vars0, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 4000, 'ftol': 1e-14, 'disp': False})
                       
        cur_sum = -res.fun
        if cur_sum > best_sum:
            best_sum = cur_sum
            best_centers = np.column_stack((res.x[0::3], res.x[1::3]))
            best_radii = res.x[2::3]
            
    return best_centers, best_radii, best_sum
