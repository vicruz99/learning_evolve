import numpy as np
from scipy.optimize import minimize

def compute_objective(vars, N, lam):
    """
    Computes the negative sum of radii plus penalty terms for constraint violations.
    """
    c = vars[:2*N].reshape(N, 2)
    r = vars[2*N:]
    obj = -np.sum(r)
    pen = 0.0
    
    # Boundary penalties
    for i in range(N):
        x, y = c[i]
        rad = r[i]
        if x - rad < 0:
            pen += (x - rad)**2
        if x + rad > 1:
            pen += (x + rad - 1)**2
        if y - rad < 0:
            pen += (y - rad)**2
        if y + rad > 1:
            pen += (y + rad - 1)**2
            
    # Overlap penalties
    for i in range(N):
        for j in range(i+1, N):
            dx = c[i,0] - c[j,0]
            dy = c[i,1] - c[j,1]
            d_sq = dx*dx + dy*dy
            d = np.sqrt(d_sq) if d_sq > 1e-16 else 1e-8
            sr = r[i] + r[j]
            if d < sr:
                pen += (sr - d)**2
                
    return obj + lam * pen

def project_and_validate(centers, radii, N):
    """
    Projects radii to strictly satisfy boundary and non-overlap constraints.
    """
    # Ensure strict feasibility by clipping radii to boundaries
    radii = np.minimum(radii, centers[:,0])
    radii = np.minimum(radii, 1 - centers[:,0])
    radii = np.minimum(radii, centers[:,1])
    radii = np.minimum(radii, 1 - centers[:,1])
    
    # Iteratively resolve overlaps by shrinking radii proportionally
    for _ in range(100):
        overlap_found = False
        for i in range(N):
            for j in range(i+1, N):
                dx = centers[i,0] - centers[j,0]
                dy = centers[i,1] - centers[j,1]
                d = np.sqrt(dx*dx + dy*dy)
                if d < radii[i] + radii[j] - 1e-9:
                    total_r = radii[i] + radii[j]
                    scale = d / total_r
                    radii[i] *= scale
                    radii[j] *= scale
                    overlap_found = True
        if not overlap_found:
            break
            
    return centers, radii

def run_packing():
    N = 26
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    lam = 20000.0
    bounds = [(0.0, 1.0)] * (2*N) + [(1e-5, 0.5)] * N

    for seed in range(6):
        np.random.seed(seed)
        # Initial positions: structured grid + small jitter to break symmetry
        cols = 6
        rows = 5
        x_init = np.tile(np.linspace(0.15, 0.85, cols), rows)[:N]

        res = minimize(compute_objective, x0, args=(N, lam), method='L-BFGS-B', bounds=bounds, 
                       options={'maxiter': 5000, 'ftol': 1e-12, 'gtol': 1e-8})
                       
        c_cand = res.x[:2*N].reshape(N, 2)

        # Project to ensure strict feasibility
        c_cand, r_cand = project_and_validate(c_cand, r_cand, N)

        valid = True

            r = r_cand[i]
