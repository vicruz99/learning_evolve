import numpy as np
from scipy.optimize import minimize
from functools import partial

def objective(vars):
    return -np.sum(vars[:26])

def con_r(vars, i):
    return vars[i]

def con_x1(vars, i):
    return vars[i+26] - vars[i]

def con_x2(vars, i):
    return 1.0 - vars[i+26] - vars[i]

def con_y1(vars, i):
    return vars[i+52] - vars[i]

def con_y2(vars, i):
    return 1.0 - vars[i+52] - vars[i]

def con_dist(vars, i, j):
    ri, rj = vars[i], vars[j]
    xi, yi = vars[i+26], vars[i+52]
    xj, yj = vars[j+26], vars[j+52]
    return (xi - xj)**2 + (yi - yj)**2 - (ri + rj)**2

def build_constraints():
    cons = []
    for i in range(26):
        cons.append({'type': 'ineq', 'fun': partial(con_r, i=i)})
        cons.append({'type': 'ineq', 'fun': partial(con_x1, i=i)})
        cons.append({'type': 'ineq', 'fun': partial(con_x2, i=i)})
        cons.append({'type': 'ineq', 'fun': partial(con_y1, i=i)})
        cons.append({'type': 'ineq', 'fun': partial(con_y2, i=i)})
    for i in range(26):
        for j in range(i+1, 26):
            cons.append({'type': 'ineq', 'fun': partial(con_dist, i=i, j=j)})
    return cons

def run_packing():
    cons = build_constraints()
    bounds = [(0, 0.5)]*26 + [(0, 1)]*52
    
    best_sum = -np.inf
    best_vars = None
    
    guesses = []
    # 1. Grid initialization
    x0, y0 = np.meshgrid(np.linspace(0.05, 0.95, 6), np.linspace(0.05, 0.95, 5))
    x0, y0 = x0.flatten()[:26], y0.flatten()[:26]
    guesses.append(np.concatenate([np.ones(26)*0.09, x0, y0]))
    
    # 2. Hexagonal-like initialization
    pts = []
    for row in range(5):
        nc = 6 if row == 0 else 5
        for col in range(nc):
            pts.append([col*0.2 + (0.1 if row%2 else 0), row*0.25 + 0.1])
    pts = np.array(pts[:26])
    pts[:,0] = (pts[:,0] - pts[:,0].min()) / (pts[:,0].max() - pts[:,0].min())
    pts[:,1] = (pts[:,1] - pts[:,1].min()) / (pts[:,1].max() - pts[:,1].min())
    guesses.append(np.concatenate([np.ones(26)*0.09, pts[:,0], pts[:,1]]))
    
    # 3. Random initialization
    np.random.seed(42)
    guesses.append(np.concatenate([np.ones(26)*0.08, np.random.uniform(0.05, 0.95, 26), np.random.uniform(0.05, 0.95, 26)]))

    for g in guesses:
        try:
            res = minimize(objective, g, method='SLSQP', bounds=bounds, constraints=cons, 
                           options={'ftol': 1e-12, 'maxiter': 3000, 'disp': False})
            if -res.fun > best_sum:
                best_sum = -res.fun
                best_vars = res.x
        except Exception:
            continue
            
    if best_vars is None:
        best_vars = guesses[0]
        best_sum = -np.sum(best_vars[:26])
        
    r_opt = best_vars[:26]
    x_opt = best_vars[26:52]
    y_opt = best_vars[52:78]
    
    centers = np.column_stack([x_opt, y_opt])
    return centers, r_opt, np.sum(r_opt)
