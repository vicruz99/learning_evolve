

Looking at this circle packing problem, I need to maximize the sum of radii for 26 circles in a unit square. The target is 2.636.

My strategy:
1. Initialize circles in hexagonal patterns (locally optimal for dense packing)
2. Use force-directed simulation to grow circles while maintaining non-overlap
3. Refine with gradient-based optimization allowing size variation
4. Try multiple initial configurations and keep the best

The key insight is that unequal circles can sometimes pack more densely than equal ones, by filling gaps between larger circles with smaller ones.

```python
import numpy as np
from scipy.optimize import minimize

def compute_penalty(centers, radii, n):
    """Compute constraint violation penalty"""
    penalty = 0.0
    pen_coeff = 1000.0
    
    # Boundary violations
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < 0:
            penalty += pen_coeff * (r - x) ** 2
        if x + r > 1:
            penalty += pen_coeff * (x + r - 1) ** 2
        if y - r < 0:
            penalty += pen_coeff * (r - y) ** 2
        if y + r > 1:
            penalty += pen_coeff * (y + r - 1) ** 2
    
    # Overlap violations
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            min_dist = radii[i] + radii[j]
            if min_dist > dist:
                penalty += pen_coeff * (min_dist - dist) ** 2
    
    return penalty

def objective(params, n):
    """Objective: maximize sum of radii with penalty for constraints"""
    centers = params[:2 * n].reshape(n, 2)
    radii = params[2 * n:]
    
    # Ensure non-negative radii
    radii = np.maximum(radii, 0)
    
    # Negative sum of radii (we maximize by minimizing negative)
    obj = -np.sum(radii)
    
    # Add penalty
    obj += compute_penalty(centers, radii, n)
    
    return obj

def force_directed_relaxation(centers, radii, n, iterations=5000):
    """Grow circles using force-directed method"""
    centers = centers.copy()
    radii = radii.copy()
    
    for it in range(iterations):
        forces = np.zeros((n, 2))
        max_overlap = 0
        
        # Compute repulsive forces
        for i in range(n):
            for j in range(i + 1, n):
                dist_vec = centers[i] - centers[j]
                dist = np.sqrt(np.sum(dist_vec ** 2))
                if dist < 1e-12:
                    dist = 1e-12
                    dist_vec = np.array([np.random.randn(), np.random.randn()]) * 1e-12
                
                repulsion = radii[i] + radii[j] - dist
                if repulsion > 0:
                    max_overlap = max(max_overlap, repulsion)
                    f = 500 * repulsion / dist
                    forces[i] += f * dist_vec
                    forces[j] -= f * dist_vec
        
        # Apply forces
        dt = 0.0005
        centers += dt * forces
        
        # Enforce boundary constraints
        for i in range(n):
            centers[i, 0] = np.clip(centers[i, 0], radii[i], 1 - radii[i])
            centers[i, 1] = np.clip(centers[i, 1], radii[i], 1 - radii[i])
        
        # Gradually grow radii
        growth_rate = 5e-7 if max_overlap < 1e-6 else 1e-7
        radii *= 1 + growth_rate
    
    return centers, radii

def hexagonal_init(n, r_init, noise_scale=0.005):
    """Initialize circles in hexagonal pattern"""
    centers = np.zeros((n, 2))
    idx = 0
    row_spacing = r_init * np.sqrt(3)
    
    row = 0
    while idx < n:
        col = 0
        while idx < n:
            x = r_init + col * 2 * r_init + (row % 2) * r_init
            y = r_init + row * row_spacing
            if x + r_init <= 1 and y + r_init <= 1:
                centers[idx, 0] = x + np.random.randn() * noise_scale
                centers[idx, 1] = y + np.random.randn() * noise_scale
                idx += 1
            col += 1
            if col > 10:
                break
        row += 1
        if row > 10:
            break
    
    return centers

def grid_init(n, n_rows, n_cols, noise_scale=0.005):
    """Initialize circles in rectangular grid pattern"""
    centers = np.zeros((n, 2))
    idx = 0
    
    cell_w = 1.0 / (n_cols + 1)
    cell_h = 1.0 / (n_rows + 1)
    
    for i in range(n_rows):
        for j in range(n_cols):
            if idx >= n:
                break
            x = (j + 1) * cell_w
            y = (i + 1) * cell_h
            centers[idx, 0] = x + np.random.randn() * noise_scale
            centers[idx, 1] = y + np.random.randn() * noise_scale
            idx += 1
    
    return centers

def run_packing():
    n = 26
    
    best_centers = None
    best_radii = None
    best_sum = 0
    
    # Try multiple initializations
    initializations = []
    
    # Hexagonal patterns with different starting radii
    for r_init in [0.08, 0.085, 0.09, 0.095]:
        centers = hexagonal_init(n, r_init, noise_scale=0.003)
        radii = np.full(n, r_init)
        initializations.append((centers, radii))
    
    # Grid patterns
    for n_rows, n_cols in [(5, 6), (6, 5), (4, 7), (7, 4), (5, 5)]:
        centers = grid_init(n, n_rows, n_cols, noise_scale=0.005)
        # Estimate radius from minimum distance
        min_d = float('inf')
        for i in range(n):
            for j in range(i + 1, n):
                d = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                min_d = min(min_d, d)
        r_est = min_d / 2 * 0.7
        radii = np.full(n, r_est)
        initializations.append((centers, radii))
    
    # Random initialization
    for seed in range(5):
        np.random.seed(seed + 100)
        centers = np.random.uniform(0.1, 0.9, (n, 2))
        radii = np.full(n, 0.06)
        initializations.append((centers, radii))
    
    # Process each initialization
    for idx, (centers, radii) in enumerate(initializations):
        np.random.seed(idx * 31)
        
        # Add small noise to radii for variation
        radii_noise = radii.copy() + np.random.randn(n) * 0.002
        radii_noise = np.maximum(radii_noise, 0.01)
        
        # Phase 1: Force-directed relaxation to grow circles
        centers_relaxed, radii_grown = force_directed_relaxation(
            centers, radii_noise, n, iterations=8000
        )
        
        # Phase 2: Gradient-based optimization with Nelder-Mead
        params = np.concatenate([centers_relaxed.flatten(), radii_grown])
        
        result = minimize(
            objective,
            params,
            args=(n,),
            method='Nelder-Mead',
            options={'maxiter': 80000, 'xatol': 1e-11, 'fatol': 1e-11}
        )
        
        final_centers = result.x[:2 * n].reshape(n, 2)
        final_radii = result.x[2 * n:]
        
        # Ensure validity
        final_radii = np.maximum(final_radii, 0)
        for i in range(n):
            final_centers[i, 0] = np.clip(final_centers[i, 0], final_radii[i], 1 - final_radii[i])
            final_centers[i, 1] = np.clip(final_centers[i, 1], final_radii[i], 1 - final_radii[i])
        
        current_sum = np.sum(final_radii)
        
        # Validate
        if current_sum > best_sum:
            # Quick validation check
            valid = True
            for i in range(n):
                x, y = final_centers[i]
                r = final_radii[i]
                if r < 0 or x - r < -1e-10 or x + r > 1 + 1e-10 or y - r < -1e-10 or y + r > 1 + 1e-10:
                    valid = False
                    break
            if valid:
                for i in range(n):
                    for j in range(i + 1, n):
                        d = np.sqrt(np.sum((final_centers[i] - final_centers[j]) ** 2))
                        if d < final_radii[i] + final_radii[j] - 1e-10:
                            valid = False
                            break
                    if not valid:
                        break
            
            if valid and current_sum > best_sum:
                best_sum = current_sum
                best_centers = final_centers.copy()
                best_radii = final_radii.copy()
        
        print(f"Init {idx}: sum_radii = {current_sum:.6f}, best = {best_sum:.6f}")
    
    # Final cleanup - ensure all constraints are satisfied
    for i in range(n):
        best_centers[i, 0] = np.clip(best_centers[i, 0], best_radii[i], 1 - best_radii[i])
        best_centers[i, 1] = np.clip(best_centers[i, 1], best_radii[i], 1 - best_radii[i])
    
    return best_centers, best_radii, np.sum(best_radii)
```
