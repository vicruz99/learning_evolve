import numpy as np
import scipy.optimize

def run_packing():

    Pack 26 circles in a unit square to maximize sum of radii.
    """

    def get_initial_config(seed):

        # Start with a grid-like distribution
        # 5x5 grid is 25 points. We need 26.
        # Let's place 25 in a 5x5 grid and 1 in center, then perturb.
        # Actually, a denser valid packing is better.
        # A 6x5 grid (30 spots) with r=0.08 fits easily.
        # We pick 26 spots.
        
        points = []
        # 6 columns, 5 rows
        cols = np.linspace(0.08, 1 - 0.08, 6)
        rows = np.linspace(0.08, 1 - 0.08, 5)
        
        # Generate all grid points
        grid_pts = []
        for y in rows:
            for x in cols:
                grid_pts.append([x, y])
        
        # We have 30 points. Shuffle and pick 26.
        rng.shuffle(grid_pts)
        selected_pts = grid_pts[:26]
        
        centers = np.array(selected_pts)
        radii = np.full(n, 0.08) # Safe initial radius
        
        # Add small random noise to break symmetry
        noise = rng.uniform(-0.01, 0.01, size=(n, 2))
        centers += noise
        
        # Clip to valid range [0.1, 0.9] to ensure boundary slack
        centers = np.clip(centers, 0.1, 0.9)
        
        return centers, radii

    best_radii = None

        bounds = []

            bounds.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])

        # 1. Boundary constraints: r <= x <= 1-r  => x - r >= 0, r - x <= 0 ?
        #    Actually: x - r >= 0  => -x + r <= 0
        #               x + r <= 1 => x + r - 1 <= 0
        #               y - r >= 0 => -y + r <= 0
        #               y + r <= 1 => y + r - 1 <= 0

        constraints = []
        
        # Boundary constraints for each circle
        for i in range(n):
            idx_x = 3 * i
            idx_y = 3 * i + 1
            idx_r = 3 * i + 2

                'fun': lambda v, i=i: v[idx_x] - v[idx_r], # x - r >= 0
                'jac': None # We can provide jac for speed but ineq fun is easier
            })
            # x + r <= 1  ->  x + r - 1 <= 0  ->  1 - x - r >= 0
            constraints.append({
                'type': 'ineq',
                'fun': lambda v, i=i: 1.0 - v[idx_x] - v[idx_r]
            })
            # y - r >= 0
            constraints.append({
                'type': 'ineq',
                'fun': lambda v, i=i: v[idx_y] - v[idx_r]
            })
