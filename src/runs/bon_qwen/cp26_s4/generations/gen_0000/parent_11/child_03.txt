import numpy as np
from scipy.optimize import minimize

def get_initial_guess(n_circles=26):
    """
    Generates an initial guess for circle centers and radii using a hexagonal packing pattern.
    Pattern: 6, 5, 5, 5, 5 circles in 5 rows.
    """
    # Define row lengths
    row_lengths = [6, 5, 5, 5, 5]
    
    # Estimate radius to fit in unit square roughly
    # Height for 5 rows: 2r + 4*r*sqrt(3) <= 1 => r(2 + 6.928) <= 1 => r <= 0.112
    # Width for 6 circles: 12r <= 1 => r <= 0.0833
    # Width for 5 circles: 10r <= 1 => r <= 0.1
    # Let's start with a safe radius, say 0.08, and let optimizer expand it.
    r_init = 0.08
    
    centers = []
    radii = []
    
    current_y = r_init
    row_idx = 0
    
    # We need to place circles. 
    # Hexagonal packing: rows shifted by r horizontally relative to each other?
    # Actually, usually shift is r (half diameter) to nestle.
    # Row 0 (6 circles): centers at x = r, 3r, 5r, 7r, 9r, 11r.
    # Row 1 (5 circles): shifted by r? centers at x = 2r, 4r, 6r, 8r, 10r.
    # This fits well.
    
    y_pos = r_init
    shift = 0.0
    
    for length in row_lengths:
        # Determine x positions
        # If shift is 0, start at r_init
        # If shift is r_init, start at 2*r_init
        start_x = r_init + shift
        
        # Generate x coordinates
        # Spacing is 2*r_init
        x_coords = [start_x + i * (2 * r_init) for i in range(length)]
        
        for x in x_coords:
            centers.append([x, y_pos])
            radii.append(r_init)
        
        # Move to next row
        # Vertical distance for hex packing is sqrt(3)*r
        y_pos += np.sqrt(3) * r_init
        
        # Toggle shift
        shift = r_init if shift == 0 else 0.0
        
    centers = np.array(centers)
    radii = np.array(radii)
    
    return centers, radii

    # x - r >= 0  => x - r >= 0
    # 1 - x - r >= 0
    # y - r >= 0
    # 1 - y - r >= 0
    # r >= 0
    
    # SLSQP constraints are inequalities: fun(x) >= 0
    
    for i in range(n_circles):
        idx = 3 * i
        # x_i - r_i >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda x, i=i: x[idx] - x[idx+2]
        })
        # 1 - x_i - r_i >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda x, i=i: 1.0 - x[idx] - x[idx+2]
        })
        # y_i - r_i >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda x, i=i: x[idx+1] - x[idx+2]
        })
        # 1 - y_i - r_i >= 0
        constraints.append({
            'type': 'ineq',
            'fun': lambda x, i=i: 1.0 - x[idx+1] - x[idx+2]
        })

        constraints.append({
            'type': 'ineq',
            'fun': lambda x, i=i: x[idx+2]
        })

    # (xi-xj)^2 + (yi-yj)^2 - (ri+rj)^2 >= 0

                return dist_sq - rad_sum_sq

                'type': 'ineq',

        x_opt = x0

    radii = np.maximum(radii, 0.0)

    # We can do a quick correction.

    sum_radii = np.sum(radii)
    
    return centers, radii, float(sum_radii)
