    -   **Variables**: We have $N=26$ circles. For each circle $i$, we optimize its center $(x_i, y_i)$ and radius $r_i$. This gives $3N = 78$ variables.
    -   **Objective**: Maximize $\sum_{i=1}^{26} r_i$. In the minimizer, we minimize $-\sum r_i$.
    -   **Constraints**:
        -   **Boundary**: Each circle must be inside $[0,1] \times [0,1]$. This implies $r_i \le x_i \le 1-r_i$ and $r_i \le y_i \le 1-r_i$.
        -   **Non-overlap**: The distance between any two centers must be at least the sum of their radii: $\sqrt{(x_i-x_j)^2 + (y_i-y_j)^2} \ge r_i + r_j$.

2.  **Optimization Strategy**:

    -   **Vectorized Constraints**: To improve efficiency, we will compute all constraint violations (boundary and overlap) in a vectorized manner within a single constraint function callback.

3.  **Refinement**:

import numpy as np
from scipy.optimize import minimize

def run_packing():
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    
    Returns:
        centers: np.array of shape (26, 2)
        radii: np.array of shape (26,)
        sum_radii: float
    """
    N = 26
    
    # Define bounds for variables: x, y in [0, 1], r in [0, 0.5]
    # Variable order: x0, y0, r0, x1, y1, r1, ...
    bounds = []
    for _ in range(N):
        bounds.extend([
            (0.0, 1.0),  # x
            (0.0, 1.0),  # y
            (0.0, 0.5)   # r
        ])

    def objective(params):
        # Maximize sum of radii -> Minimize negative sum
        radii = params[2::3]
        return -np.sum(radii)

    def constraints(params):
        # Returns a vector of constraint values.
        # We require all values >= 0.
        
        # Unpack parameters
        # params shape: (78,)
        # We can reshape to (26, 3) for easier indexing
        P = params.reshape(N, 3)
        x = P[:, 0]
        y = P[:, 1]
        r = P[:, 2]
        
        cons_list = []
        
        # 1. Boundary Constraints
        # x - r >= 0
        # 1 - x - r >= 0
        # y - r >= 0
        # 1 - y - r >= 0
        cons_list.append(x - r)
        cons_list.append(1.0 - x - r)
        cons_list.append(y - r)
        cons_list.append(1.0 - y - r)
        
        # 2. Overlap Constraints
        # (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
        # We only need to check pairs (i, j) with i < j
        # Vectorized computation for all pairs
        
        # Create matrices for x, y, r
        X = x[:, np.newaxis] # (N, 1)
        Y = y[:, np.newaxis]
        R = r[:, np.newaxis]
        
        # Differences
        dx = X - X.T  # (N, N)
        dy = Y - Y.T
        dr = R + R.T
        
        # Squared distance
        dist_sq = dx**2 + dy**2
        sum_r_sq = dr**2
        
        # Constraint matrix: dist_sq - sum_r_sq
        # We want elements >= 0.
        # We only take the upper triangle (i < j)
        mask = np.triu(np.ones((N, N), dtype=bool), k=1)
        overlap_constraints = dist_sq[mask] - sum_r_sq[mask]
        
        cons_list.append(overlap_constraints)
        
        return np.concatenate(cons_list)

    def get_initial_guess(strategy='grid', seed=None):
        """Generates initial parameters based on strategy."""
        if seed is not None:
            np.random.seed(seed)
            
        centers = np.zeros((N, 2))
        radii = np.zeros(N)
        
        if strategy == 'grid':
            # Hexagonal grid initialization
            # Estimate density. 26 circles in area 1. 
            # Approx radius 0.1. Spacing ~0.2
            # Let's try to fit them in a hexagonal pattern
            rows = 5
            # Distribute 26 circles into 5 rows
            # 6, 5, 5, 5, 5 -> 26? No 6+20=26.
            # 6, 5, 5, 5, 5
            
            row_counts = [6, 5, 5, 5, 5]
            idx = 0
            
            # Horizontal spacing
            # We have width 1. 
            # Let's place centers in [0.1, 0.9] roughly.
            
            for r_idx, count in enumerate(row_counts):
                # Vertical position
                # Offset every other row?
                # Hexagonal packing: row i is shifted by dx/2
                y_pos = 0.2 + r_idx * 0.16 # Approximate spacing
                
                # Horizontal positions
                x_spacing = 0.8 / (count + 1)
                x_positions = np.linspace(0.1 + x_spacing, 0.9 - x_spacing, count)
                
                # Shift for hexagonal effect?
                # If row index is odd, shift slightly?
                # Actually, simple grid is fine for init, optimizer will fix it.
                
                for c_idx in range(count):
                    centers[idx, 0] = x_positions[c_idx]
                    centers[idx, 1] = y_pos
                    idx += 1
        
        elif strategy == 'random':
            centers = np.random.uniform(0.2, 0.8, (N, 2))
            
        elif strategy == 'corner':
            # Place some in corners
            centers[0] = [0.2, 0.2]
            centers[1] = [0.8, 0.2]
            centers[2] = [0.2, 0.8]
            centers[3] = [0.8, 0.8]
            # Rest random
            centers[4:] = np.random.uniform(0.2, 0.8, (N-4, 2))

        # Initial radii: small value to ensure feasibility
        # Calculate max possible radius for each given positions
        # But simple small constant is safer to start
        radii[:] = 0.02 
        
        # Flatten to params array
        params = np.zeros(3 * N)
        for i in range(N):
            params[3*i] = centers[i, 0]
            params[3*i+1] = centers[i, 1]
            params[3*i+2] = radii[i]
            
        return params

    best_params = None
    best_val = -np.inf
    
    # Run optimization multiple times with different seeds/strategies
    num_restarts = 15
    strategies = ['grid', 'random', 'corner']
    
    for k in range(num_restarts):
        # Cycle through strategies
        strat = strategies[k % len(strategies)]
        init_params = get_initial_guess(strategy=strat, seed=k)
        
        # Optimize
        try:
            res = minimize(objective, 
                           init_params, 
                           method='SLSQP', 
                           bounds=bounds, 
                           constraints={'type': 'ineq', 'fun': constraints},
                           options={'ftol': 1e-9, 'maxiter': 500, 'disp': False})
            
            if res.success or res.fun < best_val: # Note: fun is negative sum
                 # Check constraints satisfaction roughly
                 # constraints function returns >= 0 for valid
                 c_vals = constraints(res.x)
                 if np.all(c_vals >= -1e-5): # Tolerance for numerical errors
                     current_sum = -res.fun
                     if current_sum > best_val:
                         best_val = current_sum
                         best_params = res.x
        except Exception:
            continue

    if best_params is None:
        # Fallback
        best_params = get_initial_guess(strategy='grid', seed=0)
        best_val = np.sum(best_params[2::3]) # Just sum radii

    # Extract centers and radii
    centers = np.zeros((N, 2))
    radii = np.zeros(N)
    
    for i in range(N):
        centers[i, 0] = best_params[3*i]
        centers[i, 1] = best_params[3*i+1]
        radii[i] = best_params[3*i+2]
        
    sum_radii = np.sum(radii)
    
    # Final validation check and slight adjustment if needed
    # Sometimes SLSQP finds a point on boundary, let's ensure strict validity
    # by shrinking radii slightly if any overlap detected
    # But the validator allows 1e-12 error.
    
    return centers, radii, sum_radii
