import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Solves the circle packing problem for 26 circles in a unit square
    to maximize the sum of radii.
    """
    np.random.seed(42)  # For reproducibility
    N = 26
    max_iter = 200
    binary_steps = 15
    
    # Helper to initialize centers in a hexagonal pattern
    def init_hex(n_circles):
        # Estimate radius for hex packing
        # Area approx n * pi * r^2 ~ 1 => r ~ sqrt(1/(pi*n))
        # But for hex, density is pi/sqrt(12), so r ~ 1/sqrt(sqrt(12)*n)
        r_est = 1.0 / np.sqrt(np.sqrt(12) * n_circles)
        # Start with slightly smaller radius to ensure initialization fits
        r = r_est * 0.9
        
        centers = []
        row = 0
        while len(centers) < n_circles:
            col = 0
            # Shift for hexagonal packing
            x_start = 0.1 if row % 2 == 0 else 0.1 + r 
            y = r + row * r * np.sqrt(3)
            
            while True:
                x = x_start + col * 2 * r
                if x + r > 1.0 or y + r > 1.0:
                    break
                centers.append([x, y])
                if len(centers) == n_circles:
                    break
                col += 1
            row += 1
        return np.array(centers[:n_circles])

    # Loss function for feasibility
    def calculate_loss(vars, r):
        # vars contains x and y coordinates for all N circles
        # Shape: (2*N,)
        loss = 0.0
        centers = vars.reshape(N, 2)
        
        # Boundary constraints: r <= x, y <= 1-r
        # Penalty for violating boundaries
        boundary_pen = np.maximum(0, r - centers[:, 0])**2 + \
                       np.maximum(0, centers[:, 0] - (1.0 - r))**2 + \
                       np.maximum(0, r - centers[:, 1])**2 + \
                       np.maximum(0, centers[:, 1] - (1.0 - r))**2
        loss += np.sum(boundary_pen)
        
        # Overlap constraints: distance >= 2r
        # Penalty for violating overlap
        for i in range(N):
            for j in range(i + 1, N):
                dist_sq = np.sum((centers[i] - centers[j])**2)
                min_dist = 2.0 * r
                if dist_sq < min_dist**2:
                    loss += (min_dist - np.sqrt(dist_sq))**2
        
        return loss

    # Solve for a specific radius r
    def solve_for_r(r):
        best_loss = float('inf')
        best_centers = None
        
        # Try multiple restarts to avoid local minima
        num_restarts = 5
        
        for k in range(num_restarts):
            # Initialization
            if k == 0:
                x0 = init_hex(N).flatten()
            else:
                # Random perturbation or random init
                x0 = np.random.uniform(r, 1-r, (N, 2)).flatten()

            # Bounds for variables [x_i, y_i]
            bounds = [(r, 1.0-r) for _ in range(2 * N)]
            
            # Optimization
            res = minimize(calculate_loss, x0, args=(r,), method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-12})
            
            if res.fun < best_loss:
                best_loss = res.fun
                best_centers = res.x.reshape(N, 2)
                
            # If we found a perfect packing, return it
            if best_loss < 1e-8:
                return best_centers, 0.0
                
        return best_centers, best_loss

    # Binary search for max radius
    low = 0.0
    high = 0.15 # Upper bound estimate (sqrt(1/pi/26) approx 0.11, but let's be safe)
    optimal_r = 0.1 # Fallback
    optimal_centers = init_hex(N)
    
    # Initial check
    centers, loss = solve_for_r(0.1)
    if loss < 1e-6:
        optimal_r = 0.1
        optimal_centers = centers
        low = 0.1
    else:
        high = 0.1

    for _ in range(binary_steps):
        mid = (low + high) / 2.0
        centers, loss = solve_for_r(mid)
        
        if loss < 1e-7:
            # Feasible
            optimal_r = mid
            optimal_centers = centers
            low = mid
        else:
            high = mid

    # Final validation and return
    # The optimizer might leave tiny violations, so we clamp radii slightly if needed,
    # but with loss < 1e-7 it should be valid.
    
    # Ensure no NaNs
    if np.isnan(optimal_centers).any():
        optimal_centers = init_hex(N)
        optimal_r = 0.1

    radii = np.full(N, optimal_r)
    sum_radii = np.sum(radii)

    # Double check validity manually just in case of numerical drift
    # If invalid, reduce radius slightly
    if not validate_packing(optimal_centers, radii):
        # Reduce radius to ensure validity
        while not validate_packing(optimal_centers, radii):
            optimal_r *= 0.99
            radii = np.full(N, optimal_r)
            sum_radii = np.sum(radii)

    return optimal_centers, radii, sum_radii

def validate_packing(centers, radii):
    """
    Validation function as provided in the prompt.
    """
    n = centers.shape[0]
    if np.isnan(centers).any():
        return False
    if np.isnan(radii).any():
        return False
    for i in range(n):
        if radii[i] < 0:
            return False
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            return False
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False
    return True
