import numpy as np
from scipy.optimize import minimize


def init_grid_plus(n):
    """Initialize with 5x5 grid + extra circles"""
    centers = []
    radii = []
    r = 0.1
    
    for i in range(5):
        for j in range(5):
            x = 0.1 + i * 0.2
            y = 0.1 + j * 0.2
            centers.append([x, y])
            radii.append(r)
    
    # Add extra circles in gaps
    extra_positions = [
        [0.3, 0.3], [0.7, 0.3], [0.3, 0.7], [0.7, 0.7],

        [0.5, 0.5]
    ]
    for pos in extra_positions:
        if len(centers) >= n:
            break
        centers.append(pos[:])
        radii.append(0.04)
    
    return np.array(centers[:n]), np.array(radii[:n])


def init_hexagonal(n):
    """Initialize with hexagonal packing"""
    centers = []
    radii = []
    r = 0.1
    
    # 5 columns, multiple rows
    row = 0
    while len(centers) < n:
        if row % 2 == 0:
            num_cols = 5
            x_start = r
        else:
            num_cols = 4
            x_start = 2 * r
        
        for col in range(num_cols):
            if len(centers) >= n:
                break
            x = x_start + col * 2 * r
            y = r + row * np.sqrt(3) * r
            
            if x + r <= 1 and y + r <= 1:
                centers.append([x, y])
                radii.append(r)
        row += 1
    
    # Fill remaining if needed
    while len(centers) < n:
        centers.append([0.5, 0.5])
        radii.append(0.05)
    
    return np.array(centers[:n]), np.array(radii[:n])


def init_random(n):
    """Random initialization with small radii"""
    centers = np.random.rand(n, 2) * 0.8 + 0.1
    radii = np.ones(n) * 0.05
    return centers, radii


def init_staggered(n):

    centers = []
    radii = []
    r = 0.095

                centers.append([x, y])

        radii.append(0.05)
    
    return np.array(centers[:n]), np.array(radii[:n])


def compute_penalty(centers, radii, n, penalty_weight=100):
    """Compute constraint violations"""
    penalty = 0.0
    
    # Boundary violations
    for i in range(n):
        cx, cy = centers[i]
        r = radii[i]
        if cx - r < 0:
            penalty += (-(cx - r)) ** 2
        if cx + r > 1:
            penalty += (cx + r - 1) ** 2
        if cy - r < 0:
            penalty += (-(cy - r)) ** 2
        if cy + r > 1:
            penalty += (cy + r - 1) ** 2
    
    # Overlap violations
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            overlap = radii[i] + radii[j] - dist
            if overlap > 0:
                penalty += overlap ** 2
    
    return penalty_weight * penalty


def objective_and_penalty(x, n, penalty_weight):
    """Combined objective and penalty"""
    centers = x[:2 * n].reshape(n, 2)
    radii = x[2 * n:]
    
    obj = -np.sum(radii)
    penalty = compute_penalty(centers, radii, n, penalty_weight)
    
    return obj + penalty, centers, radii, penalty


def optimize_packing(centers, radii, n, max_iters=2000):
    """Optimize packing using progressive penalty method"""
    x0 = np.concatenate([centers.flatten(), radii])
    x = x0.copy()
    
    bounds = [(0.0, 1.0)] * (2 * n) + [(1e-8, 0.5)] * n
    
    penalty_weight = 50
    for phase in range(6):
        pw = penalty_weight
        
        def obj_func(x):
            val, _, _, _ = objective_and_penalty(x, n, pw)
            return val
        
        result = minimize(
            obj_func, x,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': max_iters, 'ftol': 1e-15, 'gtol': 1e-10}
        )
        x = result.x
        penalty_weight *= 3
    
    centers_opt = x[:2 * n].reshape(n, 2)
    radii_opt = x[2 * n:]
    
    return centers_opt, radii_opt


def optimize_with_grad(centers, radii, n):

    bounds = [(0.0, 1.0)] * (2 * n) + [(1e-8, 0.5)] * n
    
    pw_schedule = [50, 150, 450, 1350, 4050, 12150]

            c = x[:2 * n].reshape(n, 2)
            r = x[2 * n:]
            obj_val = -np.sum(r)
            pen = 0.0
            
            for i in range(n):
                cx, cy = c[i]
                ri = r[i]
                for b in [-(cx - ri), -(1 - cx - ri), -(cy - ri), -(1 - cy - ri)]:
                    if b > 0:
                        pen += b * b
            
            for i in range(n):
                for j in range(i + 1, n):
                    dx = c[i, 0] - c[j, 0]
                    dy = c[i, 1] - c[j, 1]
                    d = np.sqrt(dx * dx + dy * dy + 1e-16)
                    ov = r[i] + r[j] - d
                    if ov > 0:
                        pen += ov * ov
            
            return obj_val + pw * pen
        
        result = minimize(obj, x, method='L-BFGS-B', bounds=bounds,
                         options={'maxiter': 3000, 'ftol': 1e-16})
        x = result.x
    
    centers_opt = x[:2 * n].reshape(n, 2)
    radii_opt = x[2 * n:]
    return centers_opt, radii_opt
