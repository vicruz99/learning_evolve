import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    def penalty_function(centers, radii, penalty_weight=1e5):
        """
        Calculates the penalty for overlapping circles and boundary violations.
        Uses a smooth approximation for better optimization.
        """
        penalty = 0.0
        # Pairwise overlap penalty
        for i in range(n):
            xi, yi = centers[i]
            ri = radii[i]
            
            # Boundary penalties (soft)
            # x boundaries
            if xi - ri < 0:
                diff = -(xi - ri)
                penalty += penalty_weight * diff**2
            if xi + ri > 1:
                diff = (xi + ri) - 1
                penalty += penalty_weight * diff**2
                
            # y boundaries
            if yi - ri < 0:
                diff = -(yi - ri)
                penalty += penalty_weight * diff**2
            if yi + ri > 1:
                diff = (yi + ri) - 1
                penalty += penalty_weight * diff**2

        # Overlap penalties between circles
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt((centers[i, 0] - centers[j, 0])**2 + 
                               (centers[i, 1] - centers[j, 1])**2)
                sum_r = radii[i] + radii[j]
                
                if dist < sum_r:
                    overlap = sum_r - dist
                    penalty += penalty_weight * overlap**2
                    
        return penalty

    def objective_flat(x_vec):
        """
        Flattened objective for scipy.
        x_vec contains [x1, y1, r1, x2, y2, r2, ...]
        We want to maximize sum(r), so minimize -sum(r) + penalty.
        """
        centers = x_vec[:2*n].reshape(n, 2)
        radii = x_vec[2*n:]

        # but bounds in optimizer handle this better)
        
        # We want to maximize sum(radii), so minimize negative sum
        obj = -np.sum(radii)
        
        # Add penalty for constraints
        obj += penalty_function(centers, radii, penalty_weight=1e6)
        
        return obj

                x_curr += x_step

            row_idx += 1
