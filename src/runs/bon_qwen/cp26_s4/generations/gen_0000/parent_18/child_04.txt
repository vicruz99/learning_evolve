import numpy as np
from scipy.optimize import minimize

N = 26


def objective(x):
    """Negative sum of radii (minimize = maximize sum)"""
    return -np.sum(x[2::3])


def constraint_x_min(x, i):
    """x_i - r_i >= 0"""
    return x[3 * i] - x[3 * i + 2]


def constraint_x_max(x, i):
    """x_i + r_i <= 1"""
    return 1 - x[3 * i] - x[3 * i + 2]


def constraint_y_min(x, i):
    """y_i - r_i >= 0"""
    return x[3 * i + 1] - x[3 * i + 2]


def constraint_y_max(x, i):
    """y_i + r_i <= 1"""
    return 1 - x[3 * i + 1] - x[3 * i + 2]


def constraint_radius(x, i):
    """r_i >= 0"""
    return x[3 * i + 2]


def constraint_non_overlap(x, i, j):
    """(x_i - x_j)^2 + (y_i - y_j)^2 >= (r_i + r_j)^2"""
    dx = x[3 * i] - x[3 * j]
    dy = x[3 * i + 1] - x[3 * j + 1]
    dr = x[3 * i + 2] + x[3 * j + 2]
    return dx * dx + dy * dy - dr * dr


def build_constraints():
    """Build all constraints for the optimization"""
    constraints = []
    for i in range(N):
        constraints.append({'type': 'ineq', 'fun': constraint_x_min, 'args': (i,)})
        constraints.append({'type': 'ineq', 'fun': constraint_x_max, 'args': (i,)})
        constraints.append({'type': 'ineq', 'fun': constraint_y_min, 'args': (i,)})
        constraints.append({'type': 'ineq', 'fun': constraint_y_max, 'args': (i,)})
        constraints.append({'type': 'ineq', 'fun': constraint_radius, 'args': (i,)})
    for i in range(N):
        for j in range(i + 1, N):
            constraints.append({'type': 'ineq', 'fun': constraint_non_overlap, 'args': (i, j)})
    return constraints


def hexagonal_init(cols, rows, x_spacing, y_spacing, start_x, start_y):
    """Create initial configuration with hexagonal grid"""
    x0 = np.zeros(3 * N)
    idx = 0
    for row in range(rows):
        for col in range(cols):
            if idx >= N:
                break
            x = start_x + col * x_spacing
            y = start_y + row * y_spacing
            if row % 2 == 1:
                x += x_spacing / 2
            x0[3 * idx] = np.clip(x, 0.05, 0.95)
            x0[3 * idx + 1] = np.clip(y, 0.05, 0.95)
            x0[3 * idx + 2] = 0.05
            idx += 1
    return x0


def force_based_optimize(centers, radii, iterations=2000):
    """Force-based pre-optimization to improve initial packing"""
    n = len(radii)
    lr = 0.005
    
    for it in range(iterations):
        # Grow radii slightly
        growth = 0.00008 * np.exp(-it / 500)
        radii += growth
        
        forces = np.zeros_like(centers)
        
        for i in range(n):
            # Boundary forces
            for axis in range(2):
                if centers[i, axis] < radii[i] + 0.01:
                    forces[i, axis] += 50 * (radii[i] + 0.01 - centers[i, axis])
                elif centers[i, axis] > 1 - radii[i] - 0.01:
                    forces[i, axis] -= 50 * (centers[i, axis] - (1 - radii[i] - 0.01))
            
            # Repulsion from other circles
            for j in range(n):
                if i == j:
                    continue
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                min_dist = radii[i] + radii[j]
                
                if dist < min_dist + 0.02 and dist > 1e-10:
                    overlap = max(0, min_dist - dist + 0.005)
                    force_mag = 100 * overlap / (dist + 1e-10)
                    forces[i, 0] += force_mag * dx
                    forces[i, 1] += force_mag * dy
        
        # Update positions
        decay = 1.0 / (1 + it * 0.002)
        centers += lr * decay * forces
        
        # Clip to valid range
        centers = np.clip(centers, 0.005, 0.995)
    
    return centers, radii


def make_config_v1():
    """5x6 hexagonal grid, vertical orientation"""
    return hexagonal_init(5, 6, 0.19, 0.165, 0.07, 0.06)


def make_config_v2():
    """6x5 hexagonal grid, horizontal orientation"""
    x0 = np.zeros(3 * N)
    idx = 0
    for row in range(5):
        for col in range(6):
            if idx >= N:
                break
            x = 0.06 + col * 0.165
            y = 0.07 + row * 0.19
            if col % 2 == 1:
                y += 0.095
            x0[3 * idx] = np.clip(x, 0.05, 0.95)
            x0[3 * idx + 1] = np.clip(y, 0.05, 0.95)
            x0[3 * idx + 2] = 0.05
            idx += 1
    return x0


def make_config_corners():
    """Grid with emphasis on corner placement"""
    x0 = np.zeros(3 * N)
    idx = 0
    # Place circles in a refined grid
    for row in range(6):
        for col in range(5):
            if idx >= N:
                break
            x = 0.09 + col * 0.175
            y = 0.09 + row * 0.15
            if row % 2 == 1:
                x += 0.0875
            x0[3 * idx] = x
            x0[3 * idx + 1] = y
            x0[3 * idx + 2] = 0.045
            idx += 1
    return x0


def run_packing():
    constraints = build_constraints()
    bounds = [(0.001, 0.999)] * (2 * N) + [(0.001, 0.5)] * N
    
    best_obj = np.inf
    best_x = None
    
    # Generate multiple initial configurations
    configs = [make_config_v1(), make_config_v2(), make_config_corners()]
    
    for trial in range(12):
        np.random.seed(trial)
        
        # Pick a base config and add perturbation
        base_idx = trial % 3
        x0 = configs[base_idx].copy()
        
        # Add random perturbation
        perturbation = np.random.uniform(-0.015, 0.015, 2 * N)
        x0[:2 * N] += perturbation
        x0[:2 * N] = np.clip(x0[:2 * N], 0.05, 0.95)
        
        # Run force-based pre-optimization
        centers = x0[:2 * N].reshape(N, 2)
        radii = x0[2 * N:].copy()
        centers, radii = force_based_optimize(centers, radii, iterations=800)
        
        # Build new initial point from force-optimized result
        x0_force = np.zeros(3 * N)
        x0_force[:2 * N] = centers.flatten()
        x0_force[2 * N:] = radii
        
        # Run constrained optimization
        result = minimize(
            objective,
            x0_force,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 15000, 'ftol': 1e-15, 'disp': False}
        )
        
        if result.fun < best_obj:
            best_obj = result.fun
            best_x = result.x.copy()
    
    # Post-optimization: try to improve with the best solution as starting point
    if best_x is not None:
        for trial in range(5):
            np.random.seed(100 + trial)
            x0_refine = best_x.copy()
            perturbation = np.random.uniform(-0.005, 0.005, 2 * N)
            x0_refine[:2 * N] += perturbation
            x0_refine[:2 * N] = np.clip(x0_refine[:2 * N], 0.05, 0.95)
            
            result = minimize(
                objective,
                x0_refine,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 20000, 'ftol': 1e-15, 'disp': False}
            )
            
            if result.fun < best_obj:
                best_obj = result.fun
                best_x = result.x.copy()
    
    centers = best_x[:2 * N].reshape(N, 2)
    radii = best_x[2 * N:]
    
    return centers, radii, np.sum(radii)
