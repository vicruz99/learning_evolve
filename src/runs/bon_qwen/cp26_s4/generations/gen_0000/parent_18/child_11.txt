import numpy as np
from scipy.optimize import minimize

def run_packing():
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    Returns (centers, radii, sum_radii).
    """
    n = 26
    
    # --- 1. Initialization: Hexagonal Lattice Pattern ---
    centers_list = []
    # Parameters for initial grid
    spacing_x = 0.20
    spacing_y = 0.20 * np.sqrt(3) / 2  # Hexagonal vertical spacing
    
    # Iterate to place points in a staggered grid
    y = 0.12
    row_parity = 0
    while y < 0.95 and len(centers_list) < n:
        x_start = 0.12 + (spacing_x / 2 * row_parity)
        x = x_start
        while x < 0.95 and len(centers_list) < n:
            centers_list.append([x, y])
            x += spacing_x
        y += spacing_y
        row_parity = 1 - row_parity
    
    # If we still need more points (unlikely with these params), fill from grid
    while len(centers_list) < n:
        centers_list.append([0.12, 0.12]) # Fallback
        
    centers = np.array(centers_list[:n])
    radii = np.ones(n) * 0.02 # Small initial radius to satisfy constraints
    
    # Flatten to optimization vector: [x1, y1, r1, x2, y2, r2, ...]
    x0 = np.concatenate([centers.flatten(), radii])
    
    # --- 2. Bounds ---
    # x, y in [0, 1]; r in [0, 0.5]
    bounds = []
    for _ in range(n):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 0.5)) # r
    
    # --- 3. Constraints ---
    def all_constraints(v):
        # Reshape vector
        c = v[0:2*n].reshape(n, 2)
        r = v[2*n:]
        
        # 1. Pairwise Non-overlap Constraints: dist^2 >= (r_i + r_j)^2
        # We use squared distance to keep it differentiable and avoid sqrt
        vals_overlap = []
        for i in range(n):
            for j in range(i + 1, n):
                dx = c[i, 0] - c[j, 0]
                dy = c[i, 1] - c[j, 1]
                dist_sq = dx*dx + dy*dy
                r_sum = r[i] + r[j]
                # Constraint: dist_sq - (r_sum)^2 >= 0
                vals_overlap.append(dist_sq - r_sum*r_sum)
        
        # 2. Boundary Constraints: 0 <= x-r, x+r <= 1, 0 <= y-r, y+r <= 1
        # Equivalent to: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0
        vals_boundary = []
        for i in range(n):
            vals_boundary.append(c[i, 0] - r[i])       # x >= r
            vals_boundary.append(1.0 - c[i, 0] - r[i]) # x + r <= 1
            vals_boundary.append(c[i, 1] - r[i])       # y >= r
            vals_boundary.append(1.0 - c[i, 1] - r[i]) # y + r <= 1
            
        return np.concatenate([vals_overlap, vals_boundary])
    
    # --- 4. Objective Function ---
    # Maximize sum of radii => Minimize negative sum
    def objective(v):
        return -np.sum(v[2*n:])
    
    # --- 5. Run Optimization ---
    # SLSQP handles non-linear constraints and bounds
    res = minimize(
        objective, 
        x0, 
        method='SLSQP', 
        bounds=bounds, 
        constraints={'type': 'ineq', 'fun': all_constraints},
        options={'maxiter': 2000, 'disp': False}
    )
    
    # --- 6. Extract Results ---
    centers_opt = res.x[0:2*n].reshape(n, 2)
    radii_opt = res.x[2*n:]
    
    # Ensure non-negative radii (solver should keep them positive, but just in case)
    radii_opt = np.maximum(radii_opt, 0.0)
    
    sum_radii = np.sum(radii_opt)
    
    return centers_opt, radii_opt, sum_radii
