import numpy as np
from scipy.optimize import minimize

def compute_penalty(params, n, w):
    """
    Computes the objective: maximize sum of radii - penalty for violations.
    params layout: [r1..rn, x1..xn, y1..yn]
    """
    r = params[:n]
    x = params[n:2*n]
    y = params[2*n:3*n]
    
    penalty = 0.0
    
    # Boundary penalties
    for i in range(n):
        dist_b = min(x[i], 1.0 - x[i], y[i], 1.0 - y[i])
        penalty += np.maximum(0.0, r[i] - dist_b)**2
        
    # Overlap penalties
    for i in range(n):
        for j in range(i + 1, n):
            d = np.hypot(x[i] - x[j], y[i] - y[j])
            penalty += np.maximum(0.0, r[i] + r[j] - d)**2
            
    return -np.sum(r) + w * penalty

def run_packing():
    n = 26
    best_sum = -1.0
    best_params = None
    
    # 1. Generate a hexagonal initial configuration
    xs, ys = [], []
    y_cur = 0.12
    row = 0
    while len(xs) < n:
        x_start = 0.12 if row % 2 == 0 else 0.18
        x_cur = x_start
        while x_cur <= 0.88 and len(xs) < n:
            xs.append(x_cur)
            ys.append(y_cur)
            x_cur += 0.14
        y_cur += 0.12
        row += 1
        
    init_x = np.array(xs[:n])
    init_y = np.array(ys[:n])
    
    # 2. Optimization with multiple restarts
    for seed in range(20):
        np.random.seed(seed)
        
        # Perturb initial guess to explore different local minima
        p_r = np.random.uniform(0.07, 0.12, size=n)
        p_x = np.clip(init_x + np.random.normal(0, 0.025, n), 0.05, 0.95)
        p_y = np.clip(init_y + np.random.normal(0, 0.025, n), 0.05, 0.95)
        
        x0 = np.concatenate([p_r, p_x, p_y])
        bounds = [(0.0, 0.5)]*n + [(0.0, 1.0)]*n + [(0.0, 1.0)]*n
        
        # L-BFGS-B with large penalty weight
        res = minimize(
            compute_penalty, x0, args=(n, 15000.0), method='L-BFGS-B',
            bounds=bounds, options={'maxiter': 3000, 'ftol': 1e-12, 'gtol': 1e-10}
        )
        
        curr_r = res.x[:n]
        curr_x = res.x[n:2*n]
        curr_y = res.x[2*n:3*n]
        
        # 3. Strict validation matching the prompt's tolerance
        valid = True
        for i in range(n):
            if curr_x[i] - curr_r[i] < -1e-12 or curr_x[i] + curr_r[i] > 1.0 + 1e-12:
                valid = False; break
            if curr_y[i] - curr_r[i] < -1e-12 or curr_y[i] + curr_r[i] > 1.0 + 1e-12:
                valid = False; break
                
        if valid:
            for i in range(n):
                for j in range(i + 1, n):
                    d = np.hypot(curr_x[i] - curr_x[j], curr_y[i] - curr_y[j])
                    if d < curr_r[i] + curr_r[j] - 1e-12:
                        valid = False; break
                if not valid: break
                
        if valid and np.sum(curr_r) > best_sum:
            best_sum = np.sum(curr_r)
            best_params = res.x.copy()
            
    # Fallback (should not be reached with proper tuning)
    if best_params is None:
        best_r = np.full(n, 0.08)
        best_x = init_x
        best_y = init_y
        best_params = np.concatenate([best_r, best_x, best_y])
        
    # Extract and format results
    centers = np.column_stack([best_params[n:2*n], best_params[2*n:3*n]])
    radii = best_params[:n]
    
    return centers, radii, np.sum(radii)
