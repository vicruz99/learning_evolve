import numpy as np
import scipy.optimize as opt

def obj_packing(vars, n):
    """Objective function: maximize the minimum distance between circles and boundaries."""
    c = vars.reshape(n, 2)
    min_d = 1.0
    # Distance to boundaries
    for i in range(n):
        min_d = min(min_d, c[i, 0], 1-c[i, 0], c[i, 1], 1-c[i, 1])
    # Distance to other circles
    for i in range(n):
        for j in range(i+1, n):
            d = np.sqrt(np.sum((c[i]-c[j])**2))
            min_d = min(min_d, d / 2)
    return -min_d

def run_packing():
    n = 26
    np.random.seed(42)
    
    # 1. Initialization: Hexagonal-like layout (6, 5, 6, 5, 4)
    centers = np.zeros((n, 2))
    idx = 0
    y_vals = np.linspace(0.12, 0.88, 5)
    for i, yi in enumerate(y_vals):
        cols = 6 if i % 2 == 0 else 5
        if i == 4:
            cols = 4
        x_vals = np.linspace(0.12, 0.88, cols)
        for xi in x_vals:
            centers[idx] = [xi, yi]
            idx += 1
            
    # Add small perturbation to break symmetry
    centers += np.random.uniform(-0.01, 0.01, centers.shape)
    
    # 2. Force-Directed Simulation
    lr = 0.008
    decay = 0.997
    target_r = 0.04
    max_iter = 12000
    
    for step in range(max_iter):
        if step % 50 == 0:
            target_r = min(0.105, target_r + 0.00006)
            
        forces = np.zeros_like(centers)
        for i in range(n):
            # Boundary repulsion
            for d in range(2):
                if centers[i, d] - target_r < 0:
                    forces[i, d] += (target_r - centers[i, d]) * 25.0
                if centers[i, d] + target_r > 1:
                    forces[i, d] -= (centers[i, d] + target_r - 1) * 25.0
                    
            # Circle repulsion
            for j in range(i + 1, n):
                diff = centers[i] - centers[j]
                dist_sq = np.sum(diff**2)
                dist = np.sqrt(dist_sq)
                min_dist = target_r * 2
                if dist < min_dist and dist > 1e-9:
                    overlap = min_dist - dist
                    rep = overlap * 20.0
                    forces[i] += diff / dist * rep
                    forces[j] -= diff / dist * rep
                    
        centers += lr * forces
        lr *= decay
        centers = np.clip(centers, 1e-5, 1 - 1e-5)
        
    # 3. Local Refinement
    x0 = centers.flatten()
    best_res = None
    best_val = np.inf
    
    # Multiple restarts to avoid local minima
    for _ in range(5):
        x0_pert = x0 + np.random.uniform(-0.002, 0.002, x0.shape)
        res = opt.minimize(obj_packing, x0_pert, args=(n,), method='Nelder-Mead', 
                           options={'maxiter': 30000, 'xatol': 1e-9, 'fatol': 1e-10})
        if res.fun < best_val:
            best_val = res.fun
            best_res = res
            
    centers_opt = best_res.x.reshape(n, 2)
    
    # 4. Radius Extraction
    final_radii = np.zeros(n)
    for i in range(n):
        min_d = min(centers_opt[i, 0], 1-centers_opt[i, 0], centers_opt[i, 1], 1-centers_opt[i, 1])
        for j in range(n):
            if i != j:
                d = np.linalg.norm(centers_opt[i] - centers_opt[j])
                min_d = min(min_d, d / 2.0)
        final_radii[i] = max(1e-7, min_d - 1e-10)
        
    return centers_opt, final_radii, np.sum(final_radii)
