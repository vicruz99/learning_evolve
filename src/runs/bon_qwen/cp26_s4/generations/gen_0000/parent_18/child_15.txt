    best_radii = None

                    break

                        if overlap > max_overlap:
                            max_overlap = overlap

                            dir = diff / dist

                            forces[j] -= f

                            forces[i] += np.random.randn(2) * 0.05
                            forces[j] -= np.random.randn(2) * 0.05

                            break

            else:
                high = mid

        if s > best_sum:

            best_centers = centers.copy()
