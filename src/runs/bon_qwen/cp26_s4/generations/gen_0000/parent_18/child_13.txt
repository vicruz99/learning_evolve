    pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]

        for i in range(n):

            x, y = centers[i]

            if x < r:
                centers[i, 0] = r
            # Right wall
            elif x > 1 - r:
                centers[i, 0] = 1 - r

            if y < r:
                centers[i, 1] = r
            # Top wall
