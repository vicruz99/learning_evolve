import scipy.optimize as opt

N_CIRCLES = 26

def constraint_overlap(vars, i, j):
    """
    Constraint: Distance between centers >= sum of radii
    Returns value >= 0 if satisfied.
    """
    n = N_CIRCLES
    x = vars[0:n]

    r = vars[2*n:3*n]

    dy = y[i] - y[j]
    dr = r[i] + r[j]

    """
    Constraint: Circle inside [0, 1] in x.
    side=0: left (x - r >= 0)
    side=1: right (1 - x - r >= 0)

    """

    r = vars[2*n:3*n]

        return y[i] - r[i]
    else:
        return 1.0 - y[i] - r[i]
