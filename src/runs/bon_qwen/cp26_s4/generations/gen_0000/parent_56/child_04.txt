import numpy as np
from scipy.optimize import minimize, differential_evolution


def hexagonal_init(n):
    """Initialize circles in a hexagonal pattern for n circles."""
    centers = []
    r = 0.08
    
    # 6 rows: 5, 4, 5, 4, 5, 3 = 26 circles
    row_counts = [5, 4, 5, 4, 5, 3]
    
    for row, count in enumerate(row_counts):
        y = r + row * 1.5 * r
        total_width = (count - 1) * np.sqrt(3) * r + 2 * r
        x_start = (1 - total_width) / 2
        for col in range(count):
            x = x_start + col * np.sqrt(3) * r
            centers.append([x, y])
    
    return np.array(centers), np.full(n, r)


def grid_init(n):
    """Initialize circles in a grid pattern."""
    centers = []
    r = 0.085
    
    # 5x5 grid + 1 extra
    positions = []
    for i in range(5):
        for j in range(5):
            x = (j + 0.5) * 0.2 + 0.1
            y = (i + 0.5) * 0.2 + 0.1
            positions.append([x, y])
    positions.append([0.5, 0.5])
    
    centers = positions[:n]
    return np.array(centers), np.full(n, r)


def expand_circles(centers, radii, n, iterations=5000):
    """Expand circles iteratively while resolving overlaps."""
    centers = centers.copy()
    radii = radii.copy()
    
    for iteration in range(iterations):
        # Expand all circles slightly
        expansion = max(0.00001, 0.001 / (1 + iteration * 0.001))
        radii += expansion
        
        # Resolve overlaps iteratively
        for _ in range(100):
            for i in range(n):
                for j in range(i + 1, n):
                    dx = centers[i, 0] - centers[j, 0]
                    dy = centers[i, 1] - centers[j, 1]
                    dist_sq = dx * dx + dy * dy
                    dist = np.sqrt(dist_sq)
                    min_dist = radii[i] + radii[j]
                    
                    if dist < min_dist and dist > 1e-12:
                        overlap = min_dist - dist
                        total_r = radii[i] + radii[j]
                        if total_r > 0:
                            factor_i = radii[j] / total_r
                            factor_j = radii[i] / total_r
                            nx = dx / dist
                            ny = dy / dist
                            centers[i, 0] += factor_i * overlap * nx
                            centers[i, 1] += factor_i * overlap * ny
                            centers[j, 0] -= factor_j * overlap * nx
                            centers[j, 1] -= factor_j * overlap * ny
        
        # Enforce boundary constraints by reducing radii if needed
        for i in range(n):
            max_r = min(centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])
            if max_r < 0:
                max_r = 0
            if radii[i] > max_r:
                radii[i] = max_r
        
        # Clip centers to valid range
        for i in range(n):
            centers[i, 0] = np.clip(centers[i, 0], radii[i], 1 - radii[i])
            centers[i, 1] = np.clip(centers[i, 1], radii[i], 1 - radii[i])
    
    return centers, radii


def objective_penalty(x, n):
    """Objective function with penalty for violated constraints."""
    centers = x[:2 * n].reshape(n, 2)
    radii = np.maximum(x[2 * n:], 1e-10)
    
    obj = -np.sum(radii)
    penalty = 0.0
    pw = 5000.0
    
    # Boundary penalties
    for i in range(n):
        cx, cy = centers[i]
        r = radii[i]
        penalty += pw * max(0, r - cx) ** 2
        penalty += pw * max(0, r - (1 - cx)) ** 2
        penalty += pw * max(0, r - cy) ** 2
        penalty += pw * max(0, r - (1 - cy)) ** 2
    
    # Non-overlap penalties
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            min_dist = radii[i] + radii[j]
            if dist < min_dist:
                penalty += pw * (min_dist - dist) ** 2
    
    return obj + penalty


def objective_local(x, n):
    """Local objective for SLSQP."""
    return -np.sum(x[2 * n:])


def make_constraints(n):
    """Create constraints for SLSQP optimizer."""
    cons = []
    
    for i in range(n):
        cons.append({'type': 'ineq', 'fun': lambda x, i=i: x[2 * i] - x[2 * n + i]})
        cons.append({'type': 'ineq', 'fun': lambda x, i=i: 1 - x[2 * i] - x[2 * n + i]})
        cons.append({'type': 'ineq', 'fun': lambda x, i=i: x[2 * i + 1] - x[2 * n + i]})
        cons.append({'type': 'ineq', 'fun': lambda x, i=i: 1 - x[2 * i + 1] - x[2 * n + i]})
    
    for i in range(n):
        for j in range(i + 1, n):
            cons.append({
                'type': 'ineq',

            })
    
    return cons
