import numpy as np
from scipy.optimize import minimize

def compute_radii(centers):
    """Compute maximum valid radii for a given set of centers."""
    n = centers.shape[0]
    # Distance to nearest boundary
    xb = np.minimum(centers[:, 0], 1.0 - centers[:, 0])
    yb = np.minimum(centers[:, 1], 1.0 - centers[:, 1])
    dist_bound = np.minimum(xb, yb)
    
    # Distance to nearest neighbor
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist_mat = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dist_mat, np.inf)
    min_dist = np.min(dist_mat, axis=1)
    
    # Radius is limited by boundary or neighbor
    return np.minimum(dist_bound, min_dist / 2.0)

def compute_updates(centers, alpha):
    """Compute gradient-like updates to maximize sum of radii."""
    n = centers.shape[0]
    
    # --- Boundary contribution ---
    d0 = centers[:, 0]
    d1 = 1.0 - centers[:, 0]
    d2 = centers[:, 1]
    d3 = 1.0 - centers[:, 1]
    b_dists = np.stack((d0, d1, d2, d3), axis=1)
    d_b = np.min(b_dists, axis=1)
    b_args = np.argmin(b_dists, axis=1)
    
    g_b = np.zeros_like(centers)
    g_b[b_args == 0, 0] = 1.0   # Push right from x=0
    g_b[b_args == 1, 0] = -1.0  # Push left from x=1
    g_b[b_args == 2, 1] = 1.0   # Push up from y=0
    g_b[b_args == 3, 1] = -1.0  # Push down from y=1
    
    # --- Neighbor contribution ---
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist_mat = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dist_mat, np.inf)
    min_dist = np.min(dist_mat, axis=1)
    min_idx = np.argmin(dist_mat, axis=1)
    
    vecs = diff[np.arange(n), min_idx]
    norms = np.linalg.norm(vecs, axis=1)
    norms = np.maximum(norms, 1e-9)
    g_n = vecs / norms[:, np.newaxis] * 0.5  # Factor 0.5 because r = d/2
    
    # --- Combine based on active constraint ---
    is_boundary_active = (d_b < min_dist / 2.0 - 1e-9)
    g = np.where(is_boundary_active[:, np.newaxis], g_b, g_n)
    
    return alpha * g

def objective_function(x, n):
    """Objective for scipy optimizer: negative sum of radii."""
    c = np.clip(x.reshape(-1, 2), 0.0, 1.0)
    return -np.sum(compute_radii(c))

def local_optimize(centers, n):
    """Refine centers using derivative-free local optimization."""
    res = minimize(objective_function, centers.flatten(), args=(n,), 
                   method='Nelder-Mead', 
                   options={'maxiter': 3000, 'xatol': 1e-8, 'fatol': 1e-8})
    return res.x.reshape(-1, 2), -res.fun

def run_packing():

    best_centers = None
