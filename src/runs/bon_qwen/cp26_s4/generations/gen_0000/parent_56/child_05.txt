import numpy as np
from scipy.optimize import linprog, minimize

def get_optimal_radii(centers):
    """
    Solves the LP to find optimal radii for fixed centers.
    Maximize sum(r_i)
    Subject to:
      r_i >= 0
      r_i <= boundary_dist_i
      r_i + r_j <= dist(c_i, c_j)
    """
    n = centers.shape[0]
    
    # Objective: minimize -sum(r_i)
    c_obj = np.ones(n) * -1.0
    
    # Inequality constraints A_ub @ r <= b_ub
    # We will build these lists
    A_ub = []
    b_ub = []
    
    # Precompute pairwise distances and boundary distances
    dists = np.zeros((n, n))
    for i in range(n):
        for j in range(i + 1, n):
            d = np.linalg.norm(centers[i] - centers[j])
            dists[i, j] = d
            dists[j, i] = d
            
    # Boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    # This can be handled by bounds in linprog (r_i <= max_r_i)
    max_r = np.ones(n)
    for i in range(n):
        x, y = centers[i]
        max_r[i] = min(x, 1.0 - x, y, 1.0 - y)
        if max_r[i] < 0:
            max_r[i] = 0 # Should not happen if centers are in [0,1]

    # Pairwise constraints: r_i + r_j <= d_ij
    # We only need to add constraints for pairs that are "close" or all of them.
    # For n=26, 325 constraints is manageable.
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(dists[i, j])

    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)

    # Bounds for r_i: 0 <= r_i <= max_r[i]
    bounds = [(0.0, max_r[i]) for i in range(n)]

    # Solve LP
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return -res.fun, res.x
        else:
            # Fallback if LP fails (should not happen)
            return 0.0, np.zeros(n)
    except Exception:
        return 0.0, np.zeros(n)

def objective(centers_flat):
    centers = centers_flat.reshape((26, 2))
    # Ensure centers are within bounds for safety, though optimizer handles bounds
    # But linear interpolation in Powell might go out of bounds temporarily if not careful
    # However, minimize with bounds handles this.
    
    sum_r, radii = get_optimal_radii(centers)
    # We want to maximize sum_r, so minimize -sum_r
    return -sum_r

def generate_initial_centers(n=26):
    # Generate hexagonal lattice
    # Estimate spacing
    # Area 1, n=26. s approx 0.21
    s = 0.20 
    points = []
    
    # Generate enough points to cover the square
    # y step is s * sqrt(3)/2 approx 0.173
    max_j = int(1.0 / (s * np.sqrt(3)/2)) + 2
    max_i = int(1.0 / s) + 2
    
    for j in range(max_j):
        y = j * s * np.sqrt(3)/2
        for i in range(max_i):
            x = i * s + (j % 2) * s / 2
            if 0 <= x <= 1 and 0 <= y <= 1:
                points.append([x, y])
    
    points = np.array(points)
    
    # If we have more than n, pick n that are well spaced?
    # Or just take the first n.
    # To make it robust, let's just take a subset if we have too many.
    # A simple heuristic: keep points that are far from existing selected points.
    
    selected = []
    if len(points) > 0:
        # Start with center-most point or corner?
        # Let's just take a random shuffle and pick first n?
        # No, structured is better.
        # Let's just take the first n.
        # But the order depends on loop.
        # Let's try to space them out.
        
        # Simple approach: just take first n.

        if len(points) >= n:
            # To improve, we can try to select points that maximize min distance
            # But for simplicity, let's just take the first n.
            # Actually, the first points are near (0,0).

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # 1. Initialize centers
    centers = generate_initial_centers(n)
    
    # 2. Define bounds for centers: [0, 1] for each coord
    bounds = [(0.0, 1.0) for _ in range(2 * n)]
    
    # 3. Optimize centers
    # Using Powell method which is derivative-free and handles non-smooth objectives
    # We minimize the negative sum of radii
    result = minimize(objective, centers.flatten(), method='Powell', bounds=bounds, 
                      options={'maxiter': 1000, 'ftol': 1e-12, 'xtol': 1e-12})
    
    final_centers = result.x.reshape((n, 2))
    
    # 4. Get final optimal radii
    sum_r, radii = get_optimal_radii(final_centers)
    
    # Validation check (internal)
    # The LP ensures constraints are satisfied, but numerical precision might be an issue.
    # The validate_packing function uses 1e-12 tolerance.
    # Our LP solution should be consistent.
    
    return final_centers, radii, float(sum_r)
