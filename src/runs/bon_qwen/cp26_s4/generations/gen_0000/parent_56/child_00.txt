import numpy as np
import scipy.optimize as opt
import random

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes the packing of 26 circles in a unit square to maximize sum of radii.
    """
    n = 26
    
    # Helper to generate hexagonal grid initialization
    def get_initial_guess(n_circles, width=1.0, height=1.0):
        # Estimate radius for hex packing to fit n circles
        # Area approx n * pi * r^2 ~ width*height * density
        # Density of hex packing ~ 0.9069
        # But boundary effects reduce this. 
        # A safe heuristic: r ~ 0.1
        r_guess = 0.1
        
        # Try to fit circles in rows
        # Vertical spacing in hex grid is sqrt(3)/2 * diameter = sqrt(3)*r
        # Let's try to place them
        points = []
        
        # We can iterate to find a good layout
        # Or just place them in a grid and let optimizer fix it
        # Let's try a dense random grid first, then hex
        
        # Hexagonal lattice generation
        # Spacing s = 2*r
        # Row spacing h = s * sqrt(3)/2
        
        # We don't know exact r, let's assume r=0.1 for layout
        s = 2 * r_guess
        h = s * np.sqrt(3) / 2
        
        y = r_guess
        row = 0
        while y + r_guess <= height and len(points) < n_circles:
            # Determine x start based on row parity
            x_start = r_guess if row % 2 == 0 else r_guess + s/2
            x = x_start
            
            while x + r_guess <= width and len(points) < n_circles:
                points.append([x, y])
                x += s
            y += h
            row += 1
            
        if len(points) < n_circles:
            # Fill remaining with random points in free space? 
            # Or just random in square
            while len(points) < n_circles:
                points.append([random.uniform(0, 1), random.uniform(0, 1)])
        
        return np.array(points)

    best_sum_radii = -1.0
    best_centers = None
    best_radii = None
    
    # Number of restarts to avoid local minima
    n_restarts = 10
    
    for k in range(n_restarts):
        # 1. Initialization
        # Use hex grid
        centers_init = get_initial_guess(n)
        
        # Add small random noise to break symmetry
        centers_init += np.random.normal(0, 0.005, centers_init.shape)
        # Clip to valid range (leaving room for radius)
        centers_init = np.clip(centers_init, 0.05, 0.95)
        
        initial_radii = np.full(n, 0.1) # Start with reasonable radius
        if k == 0:
             # For first run, maybe tighter radii to ensure validity
             pass
        else:
             # For others, vary radii slightly
             initial_radii *= np.random.uniform(0.9, 1.1, n)

        # Initial vector: x1, y1, r1, x2, y2, r2, ...
        x0 = np.concatenate([centers_init.flatten(), initial_radii])
        
        # 2. Optimization
        # Bounds for variables
        # x, y in [0, 1], r in [0, 0.5]
        bounds = [(0, 1)] * (2 * n) + [(0, 0.5)] * n
        
        # Constraints
        constraints = []
        
        # Boundary constraints:
        # x_i - r_i >= 0
        # 1 - x_i - r_i >= 0
        # y_i - r_i >= 0
        # 1 - y_i - r_i >= 0
        # These are linear.
        
        def constr_boundary_x_min(x_vars):
            # x_i - r_i >= 0
            # indices: 3i, 3i+1, 3i+2
            vals = []
            for i in range(n):
                xi = x_vars[3*i]
                ri = x_vars[3*i + 2]
                vals.append(xi - ri)
            return np.array(vals)

        def constr_boundary_x_max(x_vars):
            # 1 - x_i - r_i >= 0
            vals = []
            for i in range(n):
                xi = x_vars[3*i]
                ri = x_vars[3*i + 2]
                vals.append(1.0 - xi - ri)
            return np.array(vals)

        def constr_boundary_y_min(x_vars):
            # y_i - r_i >= 0
            vals = []
            for i in range(n):
                yi = x_vars[3*i + 1]
                ri = x_vars[3*i + 2]
                vals.append(yi - ri)
            return np.array(vals)

        def constr_boundary_y_max(x_vars):
            # 1 - y_i - r_i >= 0
            vals = []
            for i in range(n):
                yi = x_vars[3*i + 1]
                ri = x_vars[3*i + 2]
                vals.append(1.0 - yi - ri)
            return np.array(vals)

        constraints.append({'type': 'ineq', 'fun': constr_boundary_x_min})
        constraints.append({'type': 'ineq', 'fun': constr_boundary_x_max})
        constraints.append({'type': 'ineq', 'fun': constr_boundary_y_min})
        constraints.append({'type': 'ineq', 'fun': constr_boundary_y_max})

        # Overlap constraints
        # dist(i, j) >= r_i + r_j
        # sqrt((xi-xj)^2 + (yi-yj)^2) - ri - rj >= 0
        
        # To reduce overhead, we might only enforce this for "close" pairs?
        # But in dense packing, all might be close.
        # Let's implement a vectorized constraint function for all pairs.
        
        def constr_overlap(x_vars):
            centers = x_vars[:2*n].reshape(n, 2)
            radii = x_vars[2*n:]
            
            # Compute pairwise distances
            # centers shape (n, 2)
            # diff shape (n, n, 2)
            diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
            dists = np.sqrt(np.sum(diff**2, axis=2))
            
            # Radii sum matrix
            radii_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
            
            # We only care about i < j
            mask = np.triu(np.ones((n, n), dtype=bool), k=1)
            active_dists = dists[mask]
            active_radii_sums = radii_sum[mask]
            
            # Constraint: dist - (r_i + r_j) >= 0
            return active_dists - active_radii_sums

        constraints.append({'type': 'ineq', 'fun': constr_overlap})

        # Objective: maximize sum(r_i) -> minimize -sum(r_i)
        def objective(x_vars):
            radii = x_vars[2*n:]
            return -np.sum(radii)

        try:
            res = opt.minimize(
                objective, 
                x0, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=constraints,
                options={'maxiter': 1000, 'ftol': 1e-9, 'disp': False}
            )
            
            if res.success:
                current_sum = -res.fun
                # Validate manually to be sure
                c_res = res.x[:2*n].reshape(n, 2)
                r_res = res.x[2*n:]
                
                # Quick check for validity with tolerance
                valid = True
                
                # Boundary check
                for i in range(n):
                    x, y = c_res[i]
                    r = r_res[i]
                    if x - r < -1e-7 or x + r > 1 + 1e-7 or y - r < -1e-7 or y + r > 1 + 1e-7:
                        valid = False
                        break
                
                if valid:
                    # Overlap check
                    for i in range(n):
                        for j in range(i + 1, n):
                            d = np.sqrt(np.sum((c_res[i] - c_res[j])**2))
                            if d < r_res[i] + r_res[j] - 1e-7:
                                valid = False
                                break
                        if not valid: break
                
                if valid and current_sum > best_sum_radii:
                    best_sum_radii = current_sum
                    best_centers = c_res.copy()
                    best_radii = r_res.copy()
                    
        except Exception as e:
            # print(f"Optimization failed: {e}")
            pass

    # If no valid solution found (unlikely), return a fallback
    if best_centers is None:
        # Fallback to simple grid
        best_centers = np.zeros((n, 2))
        best_radii = np.zeros(n)
        # Just place them
        idx = 0
        for r in range(5):
            for c in range(5):
                if idx < n:
                    best_centers[idx] = [0.1 + c*0.2, 0.1 + r*0.2]
                    best_radii[idx] = 0.1
                    idx += 1
        best_sum_radii = np.sum(best_radii)

    return best_centers, best_radii, best_sum_radii

# For testing purposes within the script
if __name__ == "__main__":
    centers, radii, s_r = run_packing()
    print(f"Sum of radii: {s_r}")
    print(f"Min radius: {np.min(radii)}, Max radius: {np.max(radii)}")
