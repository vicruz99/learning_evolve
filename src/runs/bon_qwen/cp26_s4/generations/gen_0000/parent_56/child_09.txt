            r_opt = safe_r
            
        # Construct radii array
        radii = np.full(n, r_opt)
        sum_radii = 26 * r_opt
        
        return c_opt, radii, sum_radii
    else:
        # Fallback to initial configuration if optimization fails
        return pts, np.full(n, r_init), 26 * r_init
