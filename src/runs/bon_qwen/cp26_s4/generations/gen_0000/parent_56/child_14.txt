import numpy as np
from scipy.optimize import minimize
import warnings

# Suppress warnings from scipy optimization steps
warnings.filterwarnings('ignore')

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    n = centers.shape[0]
    if np.isnan(centers).any() or np.isnan(radii).any():
        return False
    for i in range(n):
        if radii[i] < 0:
            return False
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            return False
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False
    return True

def get_constraints(params, n):
    """
    Returns the constraint functions for the optimization problem.
    Constraints are of the form fun(x) >= 0.
    """
    constraints = []
    
    # Extract variables
    x = params[:n]
    y = params[n:2*n]
    r = params[2*n:]
    
    # Boundary constraints: x - r >= 0, 1 - x - r >= 0, etc.
    # We can implement these as simple bounds, but for clarity in constraints list:
    # x_i >= r_i  => x_i - r_i >= 0
    for i in range(n):
        constraints.append({'type': 'ineq', 'fun': lambda p, i=i: p[i] - p[2*n+i]}) # x_i - r_i >= 0
        constraints.append({'type': 'ineq', 'fun': lambda p, i=i: 1.0 - p[i] - p[2*n+i]}) # 1 - x_i - r_i >= 0
        constraints.append({'type': 'ineq', 'fun': lambda p, i=i: p[n+i] - p[2*n+i]}) # y_i - r_i >= 0
        constraints.append({'type': 'ineq', 'fun': lambda p, i=i: 1.0 - p[n+i] - p[2*n+i]}) # 1 - y_i - r_i >= 0

    # Pairwise non-overlap constraints: dist^2 >= (r_i + r_j)^2
    # dist^2 = (x_i - x_j)^2 + (y_i - y_j)^2
    # constraint: (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
    for i in range(n):
        for j in range(i + 1, n):
            def constraint_fn(p, i=i, j=j):
                xi, xj = p[i], p[j]
                yi, yj = p[n+i], p[n+j]
                ri, rj = p[2*n+i], p[2*n+j]
                dist_sq = (xi - xj)**2 + (yi - yj)**2
                rad_sum_sq = (ri + rj)**2
                return dist_sq - rad_sum_sq
            constraints.append({'type': 'ineq', 'fun': constraint_fn})
            
    return constraints

def objective(params, n):
    """
    Objective function to maximize sum of radii.
    Since minimize minimizes, we return negative sum.
    """
    r = params[2*n:]
    return -np.sum(r)

def generate_hex_grid_initial(n, seed=None):

    Generates an initial configuration based on a hexagonal lattice.
    """
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.05) # Start with small radii

    # Try to fit a pattern of rows
    # Rows of 5 and 4 circles usually fit well.
    # 5, 4, 5, 4, 5, 3 sums to 26.
    
    row_counts = [5, 4, 5, 4, 5, 3]
    current_idx = 0
    
    # Adjust spacing to fit in 1x1
    # Horizontal spacing (dx) and Vertical spacing (dy)
    # In hex grid, dx = 2*r, dy = sqrt(3)*r
    # But we don't know r yet. Let's place centers first.
    # Let's assume a diameter of approx 0.2 (r=0.1) for spacing.
    spacing_x = 0.22
    spacing_y = 0.22 * np.sqrt(3) / 2

    # Estimate total width and height
    # Max width approx 5 * spacing_x
    # Max height approx 6 * spacing_y
    
    x_start = 0.5 - (max(row_counts) - 1) * spacing_x / 2
    y_start = 0.5 - (len(row_counts) - 1) * spacing_y / 2
    
    y_pos = y_start

        # Shift odd rows (index 1, 3, 5...) by spacing_x / 2

                centers[current_idx, 1] = y_pos
                current_idx += 1
            else:
                break
        y_pos += spacing_y
        
    return centers, radii

def generate_random_initial(n, seed=None):
    """
    Generates a random initial configuration.

    if seed is not None:
        np.random.seed(seed)
    
    # Place centers randomly, but try to avoid immediate overlaps by jittering a grid
    # Or just simple random with small radius
    centers = np.random.uniform(0.1, 0.9, size=(n, 2))
    radii = np.full(n, 0.02)
    return centers, radii
