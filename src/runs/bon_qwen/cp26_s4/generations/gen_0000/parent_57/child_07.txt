import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def objective_func(x):
    # Maximize sum of radii -> minimize negative sum
    return -np.sum(x[2 * N_CIRCLES:])

def constraint_func(x):
    n = N_CIRCLES
    cx = x[:n]
    cy = x[n:2 * n]
    r = x[2 * n:]
    
    # Total constraints: 4 per circle for boundaries + n*(n-1)/2 for overlaps
    n_constraints = 4 * n + n * (n - 1) // 2
    cons = np.empty(n_constraints)
    idx = 0
    
    # Left boundary: x - r >= 0
    cons[idx:idx + n] = cx - r
    idx += n
    # Right boundary: 1 - x - r >= 0
    cons[idx:idx + n] = 1.0 - cx - r
    idx += n
    # Bottom boundary: y - r >= 0
    cons[idx:idx + n] = cy - r
    idx += n
    # Top boundary: 1 - y - r >= 0
    cons[idx:idx + n] = 1.0 - cy - r
    idx += n
    
    # Non-overlap: dist^2 >= (r_i + r_j)^2  =>  dist^2 - (r_i + r_j)^2 >= 0
    for i in range(n):
        for j in range(i + 1, n):
            dx = cx[i] - cx[j]
            dy = cy[i] - cy[j]
            dr = r[i] + r[j]
            cons[idx] = dx * dx + dy * dy - dr * dr
            idx += 1
            
    return cons

def run_packing():
    np.random.seed(42)
    n = N_CIRCLES
    
    # Initialize with a hexagonal-like grid pattern
    # This provides a geometrically sound starting point closer to optimal packing
    cx = np.zeros(n)
    cy = np.zeros(n)
    k = 0
    for row in range(6):
        y = 0.12 + row * 0.17
        offset = 0.085 if row % 2 == 1 else 0.0
        for col in range(5):
            if k < n:
                cx[k] = 0.1 + col * 0.2 + offset
                cy[k] = y
                k += 1
                
    # Start with small valid radii to ensure initial feasibility
    r = np.ones(n) * 0.03
    x0 = np.concatenate([cx, cy, r])
    
    # Bounds: centers in [0,1], radii in [0, 0.5]
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
    
    cons = {'type': 'ineq', 'fun': constraint_func}
    
    # SLSQP is robust for this scale of non-linear constraints
    res = minimize(objective_func, x0, method='SLSQP', 
                   bounds=bounds, constraints=cons,
                   options={'maxiter': 2000, 'ftol': 1e-12, 'disp': False})
                   
    centers = res.x[:2 * n].reshape(n, 2)
    radii = res.x[2 * n:]
    
    return centers, radii, np.sum(radii)
