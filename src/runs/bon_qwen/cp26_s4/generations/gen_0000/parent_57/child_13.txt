import numpy as np
from scipy.optimize import minimize


def compute_distances(centers):
    """Compute pairwise distances between circle centers."""
    n = centers.shape[0]
    dists = np.zeros(n * (n - 1) // 2)
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            diff = centers[i] - centers[j]
            dists[idx] = np.sqrt(np.sum(diff ** 2))
            idx += 1
    return dists


def get_overlap_pairs(n):
    """Generate list of (i, j) pairs for overlap checks."""
    pairs = []
    for i in range(n):
        for j in range(i + 1, n):
            pairs.append((i, j))
    return pairs


def objective_penalty(x, n, penalty, overlap_pairs):
    """Penalty-based objective function."""
    centers = x[:2 * n].reshape(n, 2)
    radii = x[2 * n:]

    obj = -np.sum(radii)

    # Boundary penalties (using ReLU-style penalties for smoothness)
    for i in range(n):
        x_i, y_i = centers[i]
        r_i = radii[i]
        
        # Left boundary
        if x_i - r_i < 0:
            obj += penalty * (x_i - r_i) ** 2
        # Right boundary
        if x_i + r_i > 1:
            obj += penalty * (x_i + r_i - 1) ** 2
        # Bottom boundary
        if y_i - r_i < 0:
            obj += penalty * (y_i - r_i) ** 2
        # Top boundary
        if y_i + r_i > 1:
            obj += penalty * (y_i + r_i - 1) ** 2

    # Overlap penalties
    for i, j in overlap_pairs:
        diff = centers[i] - centers[j]
        dist = np.sqrt(np.sum(diff ** 2))
        min_dist = radii[i] + radii[j]
        if dist < min_dist:
            obj += penalty * (min_dist - dist) ** 2

    return obj


def objective_smooth_penalty(x, n, penalty, overlap_pairs):

    obj = -np.sum(radii)

        dist = np.sqrt(np.sum(diff ** 2))
        gap = dist - radii[i] - radii[j]
        if gap < 0:
            obj += penalty * gap ** 2

    return obj


def boundary_constraints_slsqp(x, n):
    """Inequality constraints for SLSQP: all must be >= 0."""
    centers = x[:2 * n].reshape(n, 2)
    radii = x[2 * n:]
    constraints = []
    for i in range(n):
        constraints.append(centers[i, 0] - radii[i])
        constraints.append(1 - centers[i, 0] - radii[i])
        constraints.append(centers[i, 1] - radii[i])
        constraints.append(1 - centers[i, 1] - radii[i])
        constraints.append(radii[i])
    return np.array(constraints)


def overlap_constraints_slsqp(x, n, overlap_pairs):
    """Overlap inequality constraints for SLSQP."""
    centers = x[:2 * n].reshape(n, 2)
    radii = x[2 * n:]
    constraints = []
    for i, j in overlap_pairs:
        diff = centers[i] - centers[j]
        dist = np.sqrt(np.sum(diff ** 2))
        constraints.append(dist - radii[i] - radii[j])
    return np.array(constraints)


def force_relaxation(centers, radii, n, overlap_pairs, n_steps=200, step_size=0.001):
    """Apply force-based relaxation to improve packing."""
    centers = centers.copy()
    radii = radii.copy()
    
    for step in range(n_steps):
        forces = np.zeros((n, 2))
        
        # Repulsive forces between close circles

            diff = centers[i] - centers[j]
            dist_sq = np.sum(diff ** 2)

            if dist < required + 0.03:
                strength = (required + 0.03 - dist) * 50.0
                force = strength * diff / dist
                forces[i] += force
                forces[j] -= force
        
        # Boundary forces

            if centers[i, 0] - radii[i] < 0.005:
                forces[i, 0] += 2.0
            if centers[i, 0] + radii[i] > 0.995:
                forces[i, 0] -= 2.0
            if centers[i, 1] - radii[i] < 0.005:
                forces[i, 1] += 2.0
            if centers[i, 1] + radii[i] > 0.995:
                forces[i, 1] -= 2.0
        
        centers += forces * step_size
        centers = np.clip(centers, 0.005, 0.995)
        
        step_size *= 0.995
    
    return centers


def grow_radii(centers, radii, n, overlap_pairs, iterations=100):
    """Iteratively grow radii while adjusting positions."""

    radii = radii.copy()
    
    for it in range(iterations):
        # Compute maximum possible radius for each circle
        new_radii = np.zeros(n)
        for i in range(n):
            # Distance to boundary
            d_boundary = min(
                centers[i, 0],
                1 - centers[i, 0],
                centers[i, 1],
                1 - centers[i, 1]
            )
            
            # Distance to other circles minus their radii
            d_others = np.inf
            for j in range(n):
                if i == j:
                    continue
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                d_others = min(d_others, dist - radii[j])
            
            new_radii[i] = min(d_boundary, d_others, radii[i] * 1.1)
        
        # Update radii gradually
        growth = 0.3
        radii = radii * (1 - growth) + new_radii * growth
        radii = np.maximum(radii, 0.01)
        
        # Apply force relaxation to make room
        centers = force_relaxation(centers, radii, n, overlap_pairs, n_steps=50, step_size=0.0005)
    
    return centers, radii
