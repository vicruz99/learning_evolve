import numpy as np
from scipy.optimize import minimize

def boundary_constraints(vars_arr, n):
    xs = vars_arr[:n]
    ys = vars_arr[n:2*n]
    rs = vars_arr[2*n:3*n]
    c1 = xs - rs
    c2 = (1.0 - rs) - xs
    c3 = ys - rs
    c4 = (1.0 - rs) - ys
    return np.concatenate([c1, c2, c3, c4])

def overlap_constraints(vars_arr, n):
    xs = vars_arr[:n]
    ys = vars_arr[n:2*n]
    rs = vars_arr[2*n:3*n]
    con_vals = []
    for i in range(n):
        for j in range(i+1, n):
            dx = xs[i] - xs[j]
            dy = ys[i] - ys[j]
            dr = rs[i] + rs[j]
            con_vals.append(dx*dx + dy*dy - dr*dr)
    return np.array(con_vals)

def _bc(v):
    return boundary_constraints(v, 26)

def _oc(v):
    return overlap_constraints(v, 26)

def objective(vars_arr):
    n = 26
    rs = vars_arr[2*n:3*n]
    return -np.sum(rs)

def run_packing():
    n = 26
    # Initialize with a hexagonal packing pattern
    centers = []
    rows = [6, 5, 6, 5, 4]
    x_step = 0.18
    y_step = x_step * np.sqrt(3) / 2.0
    
    y_curr = 0.12
    x_start = 0.1
    
    for idx, count in enumerate(rows):
        offset = 0.0 if idx % 2 == 0 else x_step / 2.0
        x_curr = x_start + offset
        for _ in range(count):
            centers.append([x_curr, y_curr])
            x_curr += x_step
        y_curr += y_step
        
    centers = np.array(centers)
    radii = np.full(n, 0.085)
    x0 = np.concatenate([centers.ravel(), radii])
    
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 1.0)] * n
    
    constraints = [
        {'type': 'ineq', 'fun': _bc},
        {'type': 'ineq', 'fun': _oc}
    ]
    
    res = minimize(objective, x0, method='SLSQP', bounds=bounds, 
                   constraints=constraints, options={'maxiter': 3000, 'ftol': 1e-10, 'disp': False})
    
    if res.success:
        final_centers = res.x[:2*n].reshape(n, 2)
        final_radii = res.x[2*n:3*n]
    else:
        final_centers = centers
        final_radii = radii
        
    return final_centers, final_radii, np.sum(final_radii)
