import numpy as np
from scipy.optimize import minimize, differential_evolution, linprog


def compute_optimal_radii(centers):
    """Given fixed centers, solve LP to find optimal radii maximizing sum."""
    n = centers.shape[0]
    
    # Pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    d = np.sqrt(np.sum(diff ** 2, axis=2))
    np.fill_diagonal(d, np.inf)
    
    # Boundary limits for each circle
    b = np.minimum(
        np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
        np.minimum(centers[:, 1], 1.0 - centers[:, 1])
    )
    
    # LP: maximize sum(r_i) subject to r_i + r_j <= d[i,j], 0 <= r_i <= b_i
    c_obj = -np.ones(n)
    
    num_pairs = n * (n - 1) // 2
    A_ub = np.zeros((num_pairs, n))
    b_ub = np.zeros(num_pairs)
    
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = d[i, j]
            idx += 1
    
    bounds = [(0.0, b[i]) for i in range(n)]
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        return result.x
    else:
        # Fallback: greedy approach
        radii = np.copy(b)
        for _ in range(n * 10):
            for i in range(n):
                for j in range(n):
                    if i != j:
                        radii[i] = min(radii[i], max(0, d[i, j] - radii[j]))
        return radii


def objective(params, n=26):
    """Objective: negative sum of optimal radii for given centers."""
    centers = params.reshape(n, 2)
    centers = np.clip(centers, 1e-8, 1.0 - 1e-8)
    radii = compute_optimal_radii(centers)
    return -np.sum(radii)


def create_hexagonal_grid(n=26):
    """Create initial hexagonal grid arrangement."""
    centers = []
    
    # Try different patterns that sum to 26
    patterns = [
        [6, 5, 6, 5, 4],
        [5, 6, 5, 6, 4],
        [5, 5, 6, 5, 5],
        [4, 6, 6, 6, 4],
        [6, 6, 5, 5, 4],
        [5, 5, 5, 5, 6],
    ]
    
    best_sum = -np.inf
    best_centers = None
    
    for pattern in patterns:
        if sum(pattern) != n:
            continue
        
        num_rows = len(pattern)
        row_centers = []
        
        for row_idx, num in enumerate(pattern):
            y = 0.5 + (row_idx - (num_rows - 1) / 2.0) * 0.15
            y = np.clip(y, 0.1, 0.9)
            
            if row_idx % 2 == 0:
                x_min = 0.08
                x_max = 0.92
            else:
                x_min = 0.15
                x_max = 0.85
            
            if num > 1:
                xs = np.linspace(x_min, x_max, num)
            else:
                xs = [0.5]
            
            for x in xs:
                row_centers.append([x, y])
        
        candidate = np.array(row_centers)
        candidate = np.clip(candidate, 0.05, 0.95)
        
        radii = compute_optimal_radii(candidate)
        s = np.sum(radii)
        
        if s > best_sum:
            best_sum = s
            best_centers = candidate
    
    return best_centers


def run_packing():
    n = 26
    
    # Stage 1: Create initial hexagonal grid
    initial_centers = create_hexagonal_grid(n)
    initial_radii = compute_optimal_radii(initial_centers)
    best_sum = np.sum(initial_radii)
    best_centers = initial_centers.copy()
    best_radii = initial_radii.copy()
    
    print(f"Initial sum of radii: {best_sum:.6f}")
    
    # Stage 2: Differential evolution for global optimization
    bounds = [(0.01, 0.99)] * (2 * n)
    
    for seed in range(3):
        try:
            result = differential_evolution(
                objective,
                bounds,
                args=(n,),
                initial_guess=best_centers.flatten(),
                seed=seed,
                maxiter=500,
                popsize=15,
                tol=1e-10,
                mutation=(0.5, 1.5),
                recombination=0.7,
                polish=True,
                updating='deferred',
                init='sobol'
            )
            
            candidate_centers = result.x.reshape(n, 2)
            candidate_radii = compute_optimal_radii(candidate_centers)
            candidate_sum = np.sum(candidate_radii)
            
            if candidate_sum > best_sum:
                best_sum = candidate_sum
                best_centers = candidate_centers.copy()
                best_radii = candidate_radii.copy()
                print(f"After DE (seed={seed}): {best_sum:.6f}")

            print(f"DE failed with seed {seed}: {e}")

            best_centers[i] = old_center
    
    print(f"After hill-climbing: {best_sum:.6f}")

    # Final verification
    final_radii = compute_optimal_radii(best_centers)
    final_sum = np.sum(final_radii)
    
    return best_centers, final_radii, final_sum
