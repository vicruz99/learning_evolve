import numpy as np
from scipy.optimize import minimize

def run_packing():
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    Uses a hexagonal lattice initialization and scipy optimization.
    """
    n_circles = 26
    
    # 1. Initialization: Generate centers in a hexagonal pattern
    # This provides a dense, physically plausible starting configuration.
    centers = []
    y = 0
    row_count = 0
    
    # Generate points in a hexagonal grid covering slightly more than [0,1]^2
    # We'll filter and select the first n_circles points inside the square
    while len(centers) < n_circles:
        row_count += 1
        x = 0 if row_count % 2 == 0 else 0.5  # Shift every other row
        while x < 1.5:
            centers.append([x, y])
            x += 1.0
        y += np.sqrt(3) / 2
    
    # Select the first n_circles points
    centers = np.array(centers[:n_circles])
    
    # Normalize centers to be roughly within [0,1] if they aren't already
    # The generation above puts them in [0, ~1.5] x [0, ~1.5]
    # We can simply map them to [0, 1] or keep them and let optimizer handle it.
    # Better to scale them to fit tightly.
    min_c = np.min(centers, axis=0)
    max_c = np.max(centers, axis=0)
    range_c = max_c - min_c
    # Add a small margin
    centers = (centers - min_c) / range_c * 0.9 + 0.05
    
    # Initialize radii based on initial distances (small but non-zero)
    radii = np.full(n_circles, 0.05)

    # 2. Optimization Setup
    # Variables: [x1, y1, r1, x2, y2, r2, ..., x26, y26, r26]
    initial_params = np.zeros(3 * n_circles)
    for i in range(n_circles):
        initial_params[3 * i] = centers[i, 0]
        initial_params[3 * i + 1] = centers[i, 1]
        initial_params[3 * i + 2] = radii[i]

    def objective(params):
        # Maximize sum of radii -> Minimize negative sum
        radii = params[2::3]
        return -np.sum(radii)

    def constraints_builder():
        cons = []
        
        # Boundary Constraints: x - r >= 0, x + r <= 1, y - r >= 0, y + r <= 1
        for i in range(n_circles):
            idx_x = 3 * i
            idx_y = 3 * i + 1
            idx_r = 3 * i + 2
            
            # x >= r  => x - r >= 0
            cons.append({
                'type': 'ineq',
                'fun': lambda p, i=i: p[idx_x] - p[idx_r]
            })
            # x + r <= 1 => 1 - x - r >= 0
            cons.append({
                'type': 'ineq',
                'fun': lambda p, i=i: 1.0 - p[idx_x] - p[idx_r]
            })
            # y >= r
            cons.append({
                'type': 'ineq',
                'fun': lambda p, i=i: p[idx_y] - p[idx_r]
            })
            # y + r <= 1
            cons.append({
                'type': 'ineq',
                'fun': lambda p, i=i: 1.0 - p[idx_y] - p[idx_r]
            })
            
        # Non-overlap Constraints: dist(i, j) >= r_i + r_j
        # dist^2 >= (r_i + r_j)^2
        # (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                idx_xi, idx_yi, idx_ri = 3 * i, 3 * i + 1, 3 * i + 2
                idx_xj, idx_yj, idx_rj = 3 * j, 3 * j + 1, 3 * j + 2
                
                def non_overlap(p, i=i, j=j):
                    xi, yi, ri = p[idx_xi], p[idx_yi], p[idx_ri]
                    xj, yj, rj = p[idx_xj], p[idx_yj], p[idx_rj]
                    dist_sq = (xi - xj)**2 + (yi - yj)**2
                    sum_r_sq = (ri + rj)**2
                    return dist_sq - sum_r_sq

                cons.append({
                    'type': 'ineq',
                    'fun': non_overlap
                })
        return cons

    cons = constraints_builder()

    # Bounds for variables
    # x, y in [0, 1], r >= 0
    bounds = [(0, 1) for _ in range(2 * n_circles)] # x and y
    bounds += [(0, None) for _ in range(n_circles)]  # r

    # 3. Run Optimization
    # Using SLSQP which handles non-linear constraints well
    try:
        result = minimize(
            objective,
            initial_params,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 1000, 'ftol': 1e-9, 'disp': False}
        )
        
        if result.success:
            optimal_params = result.x
        else:
            # If optimization fails, fallback to best feasible found or initial
            # We can try to extract a valid solution from the result even if not fully successful
            optimal_params = result.x
    except Exception:
        optimal_params = initial_params

    # 4. Extract and Validate Results
    final_centers = np.zeros((n_circles, 2))
    final_radii = np.zeros(n_circles)
    
    for i in range(n_circles):
        final_centers[i, 0] = optimal_params[3 * i]
        final_centers[i, 1] = optimal_params[3 * i + 1]
        final_radii[i] = optimal_params[3 * i + 2]

    # Simple validation check
    # If any radius is negative (unlikely with bounds), clamp it
    final_radii = np.maximum(final_radii, 0.0)
    
    # Ensure centers are within bounds if optimizer went slightly out
    # (Though constraints should prevent this)
    final_centers[:, 0] = np.clip(final_centers[:, 0], 0, 1)
    final_centers[:, 1] = np.clip(final_centers[:, 1], 0, 1)
    
    sum_radii = np.sum(final_radii)
    
    return final_centers, final_radii, sum_radii
