        try:
            res = opt.minimize(
                lambda x: get_penalty(x.reshape(n, 2), best_r), 
                best_centers.flatten(), 
                method='BFGS', 
                options={'maxiter': 1000}
            )
            best_centers = res.x.reshape(n, 2)
            best_r = get_valid_radius(best_centers)
        except Exception:
            pass

    # Final refinement: ensure radii are consistent with centers
    # The best_r computed is the max possible for these centers.
    # We set all radii to this value.
    radii = np.full(n, best_r)
    
    return best_centers, radii, np.sum(radii)
