import numpy as np
from scipy.optimize import minimize

def packing_objective(c_flat, n, idx_i, idx_j, r):
    """
    Computes the penalty for boundary violations and circle overlaps.
    c_flat: flattened center coordinates
    n: number of circles
    idx_i, idx_j: indices for pairwise distance extraction
    r: current radius
    """
    c = c_flat.reshape(-1, 2)
    cost = 0.0
    
    # Boundary penalties: circles must stay within [r, 1-r]
    cost += np.sum(np.maximum(0, r - c[:,0])**2)
    cost += np.sum(np.maximum(0, c[:,0] - (1-r))**2)
    cost += np.sum(np.maximum(0, r - c[:,1])**2)
    cost += np.sum(np.maximum(0, c[:,1] - (1-r))**2)
    
    # Overlap penalties: distance between centers must be >= 2*r
    diffs = c[:, np.newaxis, :] - c[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    overlaps = 2*r - dists[idx_i, idx_j]
    cost += np.sum(np.maximum(0, overlaps)**2)
    
    return cost

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    np.random.seed(42)
    
    # 1. Initial placement: slightly perturbed grid
    centers = np.zeros((n, 2))
    idx = 0
    for i in range(5):
        for j in range(5):
            if idx < n:
                centers[idx] = [0.1 + 0.2 * i, 0.1 + 0.2 * j]
                idx += 1
    while idx < n:
        centers[idx] = [0.1 + 0.2 * (idx % 5), 0.1 + 0.2 * (idx // 5)]
        idx += 1
        
    # Add small noise to break symmetry
    centers += np.random.randn(n, 2) * 0.005
    centers = np.clip(centers, 0.05, 0.95)
    
    radii = np.full(n, 0.05)
    r_base = 0.05
    
    # Precompute indices for pairwise distances
    idx_i, idx_j = np.tril_indices(n, -1)
    
    r_step = 0.0003
    max_iters = 600
    
    # 2. Iterative expansion loop
    for it in range(max_iters):
        r_base += r_step
        radii[:] = r_base
        
        # Bounds ensure centers stay within valid range during optimization
        bounds = [(r_base, 1-r_base) for _ in range(2 * n)]
        
        # Local optimization to resolve overlaps
        res = minimize(packing_objective, centers.flatten(), method='L-BFGS-B', 
                       bounds=bounds, args=(n, idx_i, idx_j, r_base),
                       options={'maxiter': 100, 'ftol': 1e-14})
        centers = res.x.reshape(-1, 2)
        
        # 3. Escape local minima if stuck
        if res.fun > 1e-4:
            centers += np.random.randn(n, 2) * 0.002
            centers = np.clip(centers, r_base, 1-r_base)
            res = minimize(packing_objective, centers.flatten(), method='L-BFGS-B',
                           bounds=bounds, args=(n, idx_i, idx_j, r_base),
                           options={'maxiter': 50, 'ftol': 1e-14})
            centers = res.x.reshape(-1, 2)
            
    # 4. Final strict clamping to guarantee boundary compliance
    r_final = radii[0]
    centers = np.clip(centers, r_final, 1 - r_final)
    
    return centers, radii, float(np.sum(radii))
