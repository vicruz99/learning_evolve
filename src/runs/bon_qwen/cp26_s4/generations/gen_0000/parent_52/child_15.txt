    1. Boundary: x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
    2. Distance: dist(i,j) - r_i - r_j >= 0

    x = v[0::3]
    y = v[1::3]
    r = v[2::3]
    
    # Boundary constraints
    c_x_min = x - r
    c_x_max = 1.0 - x - r
    c_y_min = y - r
    c_y_max = 1.0 - y - r
    
    constraints = np.concatenate([c_x_min, c_x_max, c_y_min, c_y_max])
    
    # Distance constraints
    # To avoid O(N^2) loop overhead in python, we can vectorize or use loops.
    # For N=26, N^2 = 676 pairs, loop is acceptable inside optimization step.
    # However, creating a large array is better for SLSQP if possible, 
    # but constructing it efficiently is key.
    
    dist_constraints = []
    for i in range(n):
        for j in range(i + 1, n):

            dy = y[i] - y[j]
            dist = np.sqrt(dx*dx + dy*dy)
            dist_constraints.append(dist - r[i] - r[j])
            
    if dist_constraints:
        constraints = np.concatenate([constraints, dist_constraints])
        
    return constraints
