def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    n = centers.shape[0]
    # Check for NaN values
    if np.isnan(centers).any():
        return False
    if np.isnan(radii).any():
        return False
    # Check if radii are nonnegative
    for i in range(n):
        if radii[i] < 0:
            return False
    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-9 or x + r > 1 + 1e-9 or y - r < -1e-9 or y + r > 1 + 1e-9:
            return False

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-9:
                return False
    return True

def get_violations(centers, radii):
    """
    Calculate sum of squared violations for overlaps and boundaries.
    Returns a scalar penalty.
    """
    n = centers.shape[0]
    penalty = 0.0
    
    # Boundary violations
    for i in range(n):

        r = radii[i]
        # Constraints: r <= x <= 1-r  =>  x-r >= 0, 1-x-r >= 0
        # r <= y <= 1-r  =>  y-r >= 0, 1-y-r >= 0

            r - x,       # x < r
            x + r - 1,   # x > 1-r
            r - y,       # y < r
            y + r - 1    # y > 1-r
        ]

            if v > 0:
                penalty += v**2

            req_dist = radii[i] + radii[j]
            if dist < req_dist:
                v = req_dist - dist
                penalty += v**2
                
    return penalty
