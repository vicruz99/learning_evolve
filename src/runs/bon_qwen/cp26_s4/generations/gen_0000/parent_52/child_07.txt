import numpy as np
from scipy.optimize import minimize

def compute_objective(vars_flat, alpha, n):
    """
    Computes the penalized objective for circle packing.
    vars_flat: array of length 3*n (x1,y1,r1, x2,y2,r2, ...)
    alpha: penalty weight
    n: number of circles
    """
    pos = vars_flat[:2*n].reshape(n, 2)
    r = vars_flat[2*n:]
    
    # Primary objective: maximize sum of radii => minimize negative sum
    obj = -np.sum(r)
    
    # Boundary penalties
    # x >= r  => r - x <= 0
    # 1-x >= r => r - (1-x) <= 0
    # same for y
    pen_bound = np.sum(np.maximum(0, r - pos[:, 0])**2)
    pen_bound += np.sum(np.maximum(0, r - (1.0 - pos[:, 0]))**2)
    pen_bound += np.sum(np.maximum(0, r - pos[:, 1])**2)
    pen_bound += np.sum(np.maximum(0, r - (1.0 - pos[:, 1]))**2)
    
    # Overlap penalties
    # dist(i,j) >= r_i + r_j  => r_i + r_j - dist(i,j) <= 0
    diff = pos[:, np.newaxis, :] - pos[np.newaxis, :, :]
    dist = np.sqrt(np.sum(diff**2, axis=2)).copy()
    np.fill_diagonal(dist, np.inf)
    
    r_sum = r[:, np.newaxis] + r[np.newaxis, :]
    pen_overlap = np.sum(np.maximum(0, r_sum - dist)**2)
    
    return obj + alpha * (pen_bound + pen_overlap)

def run_packing():
    n = 26
    best_sum = -1.0
    best_pos = None
    best_r = None
    
    # Run multiple starts to find a robust local optimum
    for seed in range(10):
        np.random.seed(seed + 1000)
        # Initialize centers in the middle region to avoid immediate boundary trapping
        init_pos = np.random.rand(n, 2) * 0.5 + 0.25
        init_r = np.full(n, 0.07)
        
        vars0 = np.concatenate([init_pos.ravel(), init_r])
        bounds = [(0.0, 1.0) for _ in range(2*n)] + [(0.0, 0.5) for _ in range(n)]
        
        # High penalty weight to enforce constraints tightly
        alpha = 8000.0
        
        res = minimize(
            compute_objective,
            vars0,
            args=(alpha, n),
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 4000, 'ftol': 1e-12, 'gtol': 1e-8}
        )
        
        cur_pos = res.x[:2*n].reshape(n, 2)
        cur_r = res.x[2*n:]
        cur_sum = np.sum(cur_r)
        
        if cur_sum > best_sum:
            best_sum = cur_sum
            best_pos = cur_pos.copy()
            best_r = cur_r.copy()
            
    # Post-processing: ensure strict validity within tolerance
    # Scale radii down slightly to guarantee dist > r1+r2 - 1e-12
    # This loses negligible sum but guarantees validator passes
    best_r *= 0.99995
    
    # Clip radii to be strictly inside boundaries
    best_r = np.minimum(best_r, np.minimum(best_pos[:, 0], 1.0 - best_pos[:, 0]))
    best_r = np.minimum(best_r, np.minimum(best_pos[:, 1], 1.0 - best_pos[:, 1]))
    best_r = np.maximum(best_r, 0.0)
    
    final_sum = float(np.sum(best_r))
    return best_pos, best_r, final_sum
