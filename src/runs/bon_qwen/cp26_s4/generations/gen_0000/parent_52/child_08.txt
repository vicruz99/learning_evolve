from scipy.optimize import minimize, NonlinearConstraint

def compute_objective(vars):
    """
    Objective function to maximize sum of radii.
    vars: array of shape (3*N,) containing [x1, y1, r1, x2, y2, r2, ...]
    """
    return -np.sum(vars[2::3])

def compute_constraints(vars, N):
    """
    Computes all constraint values.
    Returns an array of constraint values >= 0.
    Constraints:
    1. x_i >= r_i  => x_i - r_i >= 0
    2. x_i <= 1-r_i => 1 - x_i - r_i >= 0
    3. y_i >= r_i
    4. y_i <= 1-r_i
    5. dist(i,j) >= r_i + r_j => dist^2 - (r_i+r_j)^2 >= 0
    """
    num_bc = 4 * N
    num_oc = N * (N - 1) // 2
    total = num_bc + num_oc
    c = np.empty(total)
    idx = 0
    
    # Boundary constraints
    for i in range(N):
        ix = 3 * i
        c[idx] = vars[ix] - vars[ix+2]
        idx += 1
        c[idx] = 1.0 - vars[ix] - vars[ix+2]
        idx += 1
        c[idx] = vars[ix+1] - vars[ix+2]
        idx += 1
        c[idx] = 1.0 - vars[ix+1] - vars[ix+2]
        idx += 1
        
    # Overlap constraints
    for i in range(N):
        for j in range(i+1, N):

            ix_j = 3 * j

            dy = vars[ix_i+1] - vars[ix_j+1]
            dr = vars[ix_i+2] + vars[ix_j+2]
            c[idx] = dx*dx + dy*dy - dr*dr
            idx += 1
    return c

def generate_initial_guess(N):
    """
    Generates a feasible initial configuration on a grid.
    """
    # Grid layout to place N circles
    rows = 5
    cols = 6
    dx = 1.0 / (cols + 1)
    dy = 1.0 / (rows + 1)
    
    centers = []
    for r in range(rows):
        for c in range(cols):
            if len(centers) < N:
                cx = dx * (c + 1)
                cy = dy * (r + 1)
                centers.append([cx, cy])
            else:
                break
        if len(centers) >= N:
            break
            
    centers = np.array(centers[:N])
    r_init = 0.05
    x0 = np.zeros(3 * N)
    x0[0::3] = centers[:, 0]
    x0[1::3] = centers[:, 1]
    x0[2::3] = r_init
    return x0

    best_x = None
    
    num_constraints = 4 * N + N * (N - 1) // 2
    cons = NonlinearConstraint(compute_constraints, np.zeros(num_constraints), np.inf, args=(N,))
    
    # Try multiple initial guesses to avoid local minima
    for trial in range(3):
        x0 = generate_initial_guess(N)
        if trial > 0:

            noise_y = np.random.uniform(-0.03, 0.03, size=N)

                best_x = res.x.copy()

            continue
            
    if best_x is None:

                radii[i] -= shrink
                radii[j] -= shrink
                
    return centers, radii, np.sum(radii)
