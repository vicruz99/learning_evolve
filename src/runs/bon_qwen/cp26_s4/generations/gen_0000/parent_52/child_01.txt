import numpy as np
from scipy.optimize import minimize

def compute_cost(v, n, lam):
    """
    Computes the objective function: negative sum of radii + penalty for violations.
    """
    centers = v[:2*n].reshape((n, 2))
    radii = v[2*n:]
    
    # Objective: maximize sum(radii) <=> minimize -sum(radii)
    cost = -np.sum(radii)
    
    # Penalty for boundary violations
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < 0:
            cost += lam * (x - r)**2
        if x + r > 1:
            cost += lam * (x + r - 1)**2
        if y - r < 0:
            cost += lam * (y - r)**2
        if y + r > 1:
            cost += lam * (y + r - 1)**2
            
    # Penalty for inter-circle overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.hypot(dx, dy)
            if dist < radii[i] + radii[j]:
                cost += lam * (radii[i] + radii[j] - dist)**2
                
    return cost
