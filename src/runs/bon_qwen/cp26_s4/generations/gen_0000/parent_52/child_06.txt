import numpy as np
from scipy.optimize import linprog

def generate_greedy_centers(n, seed=None):
    """
    Generates n centers using a Greedy Max-Min approach.
    Places points sequentially at the location furthest from existing points.
    """
    rng = np.random.RandomState(seed)
    centers = np.array([[0.5, 0.5]])
    
    for _ in range(n - 1):
        # Sample candidate points
        candidates = rng.rand(2000, 2)
        
        # Calculate min distance from candidates to existing centers
        # Use broadcasting for efficiency
        dists = np.linalg.norm(candidates[:, np.newaxis, :] - centers[np.newaxis, :, :], axis=2)
        min_dists = np.min(dists, axis=1)
        
        # Find the candidate with the maximum minimum distance
        best_idx = np.argmax(min_dists)
        best_point = candidates[best_idx]
        
        centers = np.vstack([centers, best_point])
        
    return centers

def get_optimal_radii(centers):
    """
    Solves an LP to find the maximum sum of radii for fixed centers.
    
    Variables: r_1, ..., r_n
    Maximize: sum(r_i)
    Subject to:
        r_i >= 0
        r_i <= min(x_i, 1-x_i, y_i, 1-y_i)  (Boundary constraints)
        r_i + r_j <= dist(c_i, c_j)         (Non-overlap constraints)
    """
    n = centers.shape[0]
    
    # 1. Bounds from walls
    ub = np.full(n, 1.0)
    for i in range(n):
        x, y = centers[i]
        ub[i] = min(x, 1 - x, y, 1 - y)
    
    # 2. Distance constraints (r_i + r_j <= dist)
    # Matrix A_ub * r <= b_ub
    # Each row corresponds to a pair (i, j)
    n_pairs = n * (n - 1) // 2
    A_ub = np.zeros((n_pairs, n))
    b_ub = np.zeros(n_pairs)
    
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dist
            idx += 1
            
    # Objective: Maximize sum(r) -> Minimize -sum(r)
    c_obj = -np.ones(n)
    
    # Bounds for variables: 0 <= r_i <= ub[i]
    bounds = [(0, b) for b in ub]
    
    # Solve LP
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return res.x, np.sum(res.x)
    else:
        # Fallback: small radii if LP fails (should not happen)
        return np.full(n, 0.01), 0.26

def compute_forces(centers, radii, threshold=1e-6):
    """
    Computes repulsive forces to move centers apart from tight constraints.
    """
    n = centers.shape[0]
    forces = np.zeros_like(centers)
    
    # Pairwise repulsion
    for i in range(n):
        for j in range(i + 1, n):
            diff = centers[j] - centers[i]
            dist = np.linalg.norm(diff)
            
            if dist < 1e-9:
                continue
                
            r_sum = radii[i] + radii[j]
            slack = dist - r_sum
            
            # If circles are touching or overlapping (within threshold), push apart
            if slack < threshold:
                dir_vec = diff / dist
                # Force magnitude constant for active constraints
                f_mag = 1.0
                forces[i] -= dir_vec * f_mag
                forces[j] += dir_vec * f_mag

    # Wall repulsion
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        
        # Left wall: x - r <= 0
        if x - r < threshold:
            forces[i, 0] += 1.0
        # Right wall: 1 - x - r <= 0
        if (1 - x) - r < threshold:
            forces[i, 0] -= 1.0
        # Bottom wall: y - r <= 0
        if y - r < threshold:
            forces[i, 1] += 1.0
        # Top wall: 1 - y - r <= 0
        if (1 - y) - r < threshold:
            forces[i, 1] -= 1.0
            
    return forces

def optimize_packing(centers, max_iter=2000):
    """
    Iteratively optimizes centers and radii.
    """
    radii, current_sum = get_optimal_radii(centers)
    if radii is None:
        return centers, np.full(centers.shape[0], 0.01), 0.26
        
    best_centers = centers.copy()
    best_radii = radii.copy()
    best_sum = current_sum
    
    step_size = 0.05
    decay = 0.999
    
    for iteration in range(max_iter):
        # Solve LP for current centers
        radii, current_sum = get_optimal_radii(centers)
        if radii is None:
            break
            
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers.copy()
            best_radii = radii.copy()
            
        # Compute forces
        forces = compute_forces(centers, radii)
        
        # Update centers
        centers += step_size * forces
        
        # Clip centers to stay within square (with small margin to avoid numerical issues at exact boundary)
        centers = np.clip(centers, 1e-8, 1.0 - 1e-8)
        
        # Decay step size
        step_size *= decay
        
    return best_centers, best_radii, best_sum

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Main function to run the packing optimization.
    Returns the best packing found.
    """
    n = 26
    best_global_centers = None
    best_global_radii = None
    best_global_sum = 0.0
    
    # Try multiple random restarts to avoid local optima
    for seed in range(10):
        # Generate initial centers
        centers = generate_greedy_centers(n, seed=seed)
        
        # Optimize
        opt_centers, opt_radii, opt_sum = optimize_packing(centers)
        
        if opt_sum > best_global_sum:
            best_global_sum = opt_sum
            best_global_centers = opt_centers
            best_global_radii = opt_radii
            
    # Final verification and return
    return best_global_centers, best_global_radii, best_global_sum
