2. **Penalty Formulation**: We convert constraints into quadratic penalty terms added to the objective: $L = -\sum r_i + \lambda \sum \text{violations}^2$. The gradient is computed analytically for efficient optimization.
3. **Continuation Method**: We start with a small penalty weight $\lambda$ to allow the optimizer to move circles freely, then gradually increase $\lambda$ to strictly enforce constraints. This avoids getting stuck early.
4. **Optimization**: `scipy.optimize.minimize` with `L-BFGS-B` is used. It handles bound constraints ($0 \le x,y \le 1$, $r \ge 0$) efficiently and uses our analytical gradient.

import numpy as np
from scipy.optimize import minimize
import math

def compute_loss_and_grad(vars, N, lam):
    """
    Computes the loss and gradient for the circle packing problem.
    Loss = -sum(radii) + penalty for overlaps and boundary violations.
    """
    centers = vars[:2*N].reshape(N, 2)
    radii = vars[2*N:]
    
    loss = -np.sum(radii)
    grad = np.zeros_like(vars)
    grad[2*N:] = -1.0  # Gradient of -sum(radii)
    
    # Pairwise overlap penalties
    for i in range(N):
        for j in range(i+1, N):
            diff = centers[i] - centers[j]
            dist = np.linalg.norm(diff)
            r_sum = radii[i] + radii[j]
            gap = r_sum - dist
            
            if gap > 0:
                loss += lam * gap**2
                gf = 2 * lam * gap
                
                # Gradient w.r.t radii
                grad[2*N + i] += gf
                grad[2*N + j] += gf
                
                # Gradient w.r.t centers
                if dist > 1e-9:
                    dir = diff / dist
                    grad[2*i]     -= gf * dir[0]
                    grad[2*i + 1] -= gf * dir[1]
                    grad[2*j]     += gf * dir[0]
                    grad[2*j + 1] += gf * dir[1]
                else:
                    # Fallback direction if centers coincide
                    grad[2*i] -= gf
                    grad[2*j] += gf
                    
    # Boundary penalties
    for i in range(N):
        x, y = centers[i]
        r = radii[i]
        
        # Left boundary
        if x < r:
            viol = r - x
            loss += lam * viol**2
            g = 2 * lam * viol
            grad[2*N + i] += g
            grad[2*i]     -= g
        # Right boundary
        if x > 1 - r:
            viol = x + r - 1
            loss += lam * viol**2
            g = 2 * lam * viol
            grad[2*N + i] -= g
            grad[2*i]     += g
        # Bottom boundary
        if y < r:
            viol = r - y
            loss += lam * viol**2
            g = 2 * lam * viol
            grad[2*N + i] += g
            grad[2*i + 1] -= g
        # Top boundary
        if y > 1 - r:
            viol = y + r - 1
            loss += lam * viol**2
            g = 2 * lam * viol
            grad[2*N + i] -= g
            grad[2*i + 1] += g
            
    return loss, grad

def run_packing():
    N = 26
    best_sum = -1.0
    best_res = None
    
    # Bounds: x,y in [0,1], r in [1e-5, 0.5]
    bounds = [(0.0, 1.0)] * (2*N) + [(1e-5, 0.5)] * N
    
    starts = []
    rng = np.random.RandomState(2024)
    
    # Generate random starting configurations
    for _ in range(5):
        cx = rng.uniform(0.15, 0.85, N)
        cy = rng.uniform(0.15, 0.85, N)
        r = rng.uniform(0.03, 0.07, N)
        starts.append(np.concatenate([cx, cy, r]))
        
    # Generate hexagonal-like starting configuration
    pts = []
    rows = [6, 5, 6, 5, 4] # Sums to 26
    y = 0.18
    for idx, cnt in enumerate(rows):
        # Offset alternate rows for hex packing
        x_start = 0.15 + (0.5 if idx % 2 else 0) * 0.14
        for k in range(cnt):
            pts.append([x_start + k*0.125, y])
        y += 0.108 * math.sqrt(3)/2
    pts = np.array(pts)
    starts.append(np.concatenate([pts.ravel(), np.full(N, 0.055)]))
    
    # Optimization loop with continuation on penalty weight
    for x0 in starts:
        current_x = x0
        # Gradually increase penalty to enforce constraints strictly
        for lam in [50, 200, 800, 3000]:
            res = minimize(compute_loss_and_grad, current_x, args=(N, lam), 
                           method='L-BFGS-B', jac=True, bounds=bounds,
                           options={'ftol': 1e-10, 'gtol': 1e-7, 'maxiter': 3000})
            current_x = res.x
            
        centers = current_x[:2*N].reshape(N, 2)
        radii = current_x[2*N:]
        
        # Ensure numerical validity
        radii = np.maximum(radii, 1e-6)
        centers = np.clip(centers, radii[:, None], 1 - radii[:, None])
        
        s = np.sum(radii)
        if s > best_sum:
            best_sum = s
            best_res = (centers, radii)
            
    return best_res[0], best_res[1], best_sum
