import numpy as np
from scipy.optimize import minimize
import random

def objective_func(x):
    return -np.sum(x[2::3])

def constraint_x_min_i(x, i):
    return x[3*i] - x[3*i + 2]

def constraint_x_max_i(x, i):
    return 1.0 - x[3*i] - x[3*i + 2]

def constraint_y_min_i(x, i):
    return x[3*i + 1] - x[3*i + 2]

def constraint_y_max_i(x, i):
    return 1.0 - x[3*i + 1] - x[3*i + 2]

def constraint_r_min_i(x, i):
    return x[3*i + 2]

def constraint_overlap_ij(x, i, j):
    dx = x[3*i] - x[3*j]
    dy = x[3*i + 1] - x[3*j + 1]
    dist = np.sqrt(dx**2 + dy**2)
    return dist - x[3*i + 2] - x[3*j + 2]

def build_constraints(n):
    constraints = []
    for i in range(n):
        constraints.append({'type': 'ineq', 'fun': constraint_x_min_i, 'args': (i,)})
        constraints.append({'type': 'ineq', 'fun': constraint_x_max_i, 'args': (i,)})
        constraints.append({'type': 'ineq', 'fun': constraint_y_min_i, 'args': (i,)})
        constraints.append({'type': 'ineq', 'fun': constraint_y_max_i, 'args': (i,)})
        constraints.append({'type': 'ineq', 'fun': constraint_r_min_i, 'args': (i,)})
    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({'type': 'ineq', 'fun': constraint_overlap_ij, 'args': (i, j)})
    return constraints

def make_hexagonal_init(n, radius):
    centers = np.zeros((n, 2))
    radii = np.full(n, radius)
    idx = 0
    row = 0
    s = 2.0 * radius
    vertical_spacing = s * np.sqrt(3) / 2.0
    margin = radius + 0.001
    while idx < n:
        num_cols = 6 if row % 2 == 0 else 5
        for col in range(num_cols):
            if idx >= n:
                break
            x_base = col * s + margin
            if row % 2 == 1:
                x_base += s / 2.0
            y = row * vertical_spacing + margin
            if x_base + radius < 1.0 - 0.001 and y + radius < 1.0 - 0.001:
                centers[idx] = [x_base, y]
                idx += 1
        row += 1
    return centers, radii

def make_random_init(n, radius, seed):
    rng = random.Random(seed)
    centers = np.zeros((n, 2))
    radii = np.full(n, radius)
    spacing = 2.0 * radius * 1.05
    idx = 0
    row = 0
    while idx < n:
        col = 0
        while col < 7 and idx < n:
            x = col * spacing + radius + 0.02 + rng.uniform(-0.01, 0.01)
            y = row * spacing * np.sqrt(3) / 2.0 + radius + 0.02 + rng.uniform(-0.01, 0.01)
            if row % 2 == 1:
                x += spacing / 2.0
            x = max(radius + 0.001, min(1.0 - radius - 0.001, x))
            y = max(radius + 0.001, min(1.0 - radius - 0.001, y))
            centers[idx] = [x, y]
            idx += 1
            col += 1
        row += 1
    return centers, radii

def flatten(centers, radii):
    n = centers.shape[0]
    x0 = np.zeros(3 * n)
    for i in range(n):
        x0[3*i] = centers[i, 0]
        x0[3*i + 1] = centers[i, 1]
        x0[3*i + 2] = radii[i]
    return x0

def unflatten(x, n):
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        centers[i] = [x[3*i], x[3*i + 1]]
        radii[i] = x[3*i + 2]
    return centers, radii

def clamp_radii(centers, radii, tol=1e-10):
    n = len(radii)
    for _ in range(10):
        changed = False
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            max_r = min(x, 1.0 - x, y, 1.0 - y)
            if max_r < r - tol:
                radii[i] = max(0.0, max_r)
                changed = True
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] + tol:
                    sum_r = radii[i] + radii[j]
                    scale = dist / max(sum_r, 1e-15)
                    r1_new = radii[i] * scale
                    r2_new = radii[j] * scale
                    radii[i] = r1_new
                    radii[j] = r2_new
                    changed = True
        if not changed:
            break
    return centers, radii

def optimize_packing(n, x0, constraints, bounds, maxiter=3000):
    result = minimize(
        objective_func,
        x0,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={
            'maxiter': maxiter,
            'ftol': 1e-12,
            'disp': False
        }
    )
    return result
