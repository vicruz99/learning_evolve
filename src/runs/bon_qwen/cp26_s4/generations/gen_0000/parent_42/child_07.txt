def compute_repulsion(centers, radii, stiffness):

    forces = np.zeros_like(centers)
    for i in range(n):

            dist = np.sqrt(dist_sq)
            if dist < 1e-8:
                dist = 1e-8

            overlap = radii[i] + radii[j] - dist
            if overlap > 0:
                f = stiffness * diff / dist * overlap
                forces[i] += f
                forces[j] -= f
    return forces

def apply_boundary_forces(centers, radii, stiffness):

            forces[i, 1] += stiffness * (r - y)

            forces[i, 1] -= stiffness * (y + r - 1)
    return forces
