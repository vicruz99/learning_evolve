def constraints_func(vars_flat, n):
    """Returns constraint values for the optimization problem."""
    vars_ = vars_flat.reshape((n, 3))
    centers = vars_[:, :2]
    radii = vars_[:, 2]
    
    constraints = []
    
    # Boundary constraints: x - r >= 0, 1 - x - r >= 0, etc.
    constraints.append(centers[:, 0] - radii)
    constraints.append(1 - centers[:, 0] - radii)
    constraints.append(centers[:, 1] - radii)
    constraints.append(1 - centers[:, 1] - radii)
    
    # Non-overlap constraints: dist >= r_i + r_j

            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            constraints.append(dist - (radii[i] + radii[j]))
            
    return np.concatenate(constraints)
