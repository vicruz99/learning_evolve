2. Use penalty-based optimization with L-BFGS-B (handles bounds natively)

import numpy as np
from scipy.optimize import minimize


def compute_violations(centers, radii):
    n = centers.shape[0]
    violations = 0.0
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            d = np.sqrt(dx * dx + dy * dy)
            gap = d - radii[i] - radii[j]
            if gap < 0:
                violations += gap * gap
        for coord in range(2):
            gap1 = centers[i, coord] - radii[i]
            gap2 = 1.0 - centers[i, coord] - radii[i]
            if gap1 < 0:
                violations += gap1 * gap1
            if gap2 < 0:
                violations += gap2 * gap2
    return violations


def objective_with_penalty(x, penalty):
    n = 26
    centers = x[:52].reshape(n, 2)
    radii = x[52:]
    return -np.sum(radii) + penalty * compute_violations(centers, radii)


def get_bounds(n):
    bounds = []
    for _ in range(2 * n):
        bounds.append((0.001, 0.999))
    for _ in range(n):
        bounds.append((0.001, 0.49))
    return bounds


def initialize_hexagonal(n, perturb_scale=0.0):
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.085

    idx = 0
    row_counts = [6, 5, 6, 5, 4]
    spacing_x = 0.155
    spacing_y = spacing_x * np.sqrt(3) / 2

    for row in range(5):
        y = 0.085 + row * spacing_y
        count = row_counts[row]
        for col in range(count):
            if row % 2 == 0:
                x = 0.085 + col * spacing_x
            else:
                x = 0.085 + spacing_x / 2 + col * spacing_x
            centers[idx] = [x, y]
            idx += 1

    if perturb_scale > 0:
        rng = np.random.default_rng(0)
        centers += rng.uniform(-perturb_scale, perturb_scale, (n, 2))
        radii += rng.uniform(-perturb_scale * 0.5, perturb_scale * 0.5, n)
        radii = np.maximum(radii, 0.01)

    return centers, radii


def shrink_to_valid(centers, radii):
    n = centers.shape[0]
    radii = radii.copy()

    # Iteratively shrink overlapping circles
    for iteration in range(100):
        max_violation = 0.0
        worst_i, worst_j = 0, 1

        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                d = np.sqrt(dx * dx + dy * dy)
                gap = d - radii[i] - radii[j]
                if gap < max_violation:
                    max_violation = gap
                    worst_i, worst_j = i, j

        if max_violation > 1e-8:
            shrink = -max_violation / 2
            radii[worst_i] += shrink
            radii[worst_j] += shrink
        else:
            break

    # Handle boundary violations
    for iteration in range(100):
        max_violation = 0.0
        worst_i = 0
        for i in range(n):
            for coord in range(2):
                if centers[i, coord] - radii[i] < 0:
                    v = centers[i, coord] - radii[i]
                    if v < max_violation:
                        max_violation = v
                        worst_i = i
                if 1.0 - centers[i, coord] - radii[i] < 0:
                    v = 1.0 - centers[i, coord] - radii[i]
                    if v < max_violation:
                        max_violation = v
                        worst_i = i

        if max_violation > 1e-8:
            radii[worst_i] += max_violation
        else:
            break

    radii = np.maximum(radii, 1e-10)
    return centers, radii


def run_packing():
    n = 26
    best_centers = None
    best_radii = None
    best_sum = -np.inf

    # Try multiple seeds
    for seed in range(30):
        np.random.seed(seed)

        # Initialize with hexagonal packing
        centers, radii = initialize_hexagonal(n, perturb_scale=0.008)

        x0 = np.concatenate([centers.flatten(), radii])
        bounds = get_bounds(n)

        # Progressive penalty optimization
        for penalty in [5, 50, 500, 5000]:
            result = minimize(
                objective_with_penalty,
                x0,
                method="L-BFGS-B",
                bounds=bounds,
                args=(penalty,),
                options={"maxiter": 5000, "ftol": 1e-15}
            )
            x0 = result.x.copy()

        centers_opt = x0[:52].reshape(n, 2)
        radii_opt = np.maximum(x0[52:], 1e-10)

        # Ensure validity
        centers_opt, radii_opt = shrink_to_valid(centers_opt, radii_opt)

        # Clamp centers to valid range
        for i in range(n):
            for coord in range(2):
                centers_opt[i, coord] = max(radii_opt[i], min(1.0 - radii_opt[i], centers_opt[i, coord]))

        current_sum = np.sum(radii_opt)

        # Verify no NaN
        if not np.isnan(current_sum) and not np.any(np.isnan(centers_opt)) and not np.any(np.isnan(radii_opt)):
            if current_sum > best_sum:
                best_sum = current_sum
                best_centers = centers_opt.copy()
                best_radii = radii_opt.copy()

    # Final refinement on best solution
    if best_centers is not None:
        x0 = np.concatenate([best_centers.flatten(), best_radii])
        bounds = get_bounds(n)

        for penalty in [5000, 20000, 50000]:
            result = minimize(
                objective_with_penalty,
                x0,
                method="L-BFGS-B",
                bounds=bounds,
                args=(penalty,),
                options={"maxiter": 10000, "ftol": 1e-16}
            )
            x0 = result.x.copy()

        best_centers = x0[:52].reshape(n, 2)
        best_radii = np.maximum(x0[52:], 1e-10)
        best_centers, best_radii = shrink_to_valid(best_centers, best_radii)

        for i in range(n):
            for coord in range(2):
                best_centers[i, coord] = max(best_radii[i], min(1.0 - best_radii[i], best_centers[i, coord]))

        best_sum = np.sum(best_radii)

    return best_centers, best_radii, best_sum
