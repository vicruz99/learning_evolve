import numpy as np
from scipy.optimize import minimize

def calculate_violations(centers, radii):
    """
    Calculate the total constraint violations (penalty) for a packing.
    Violations occur if circles are outside the square or if they overlap.
    """
    n = len(radii)
    penalty = 0.0
    
    # Boundary violations
    # x - r < 0, x + r > 1, y - r < 0, y + r > 1
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        
        # Left and Right walls
        penalty += max(0, r - x)**2
        penalty += max(0, x + r - 1)**2
        
        # Bottom and Top walls
        penalty += max(0, r - y)**2
        penalty += max(0, y + r - 1)**2

    # Overlap violations
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            sum_radii = radii[i] + radii[j]
            if dist < sum_radii:
                penalty += (sum_radii - dist)**2
                
    return penalty

def objective_function(vars_flat, centers_init, radii_init, mu):

    Objective function to minimize.

    return -np.sum(radii) + mu * penalty
