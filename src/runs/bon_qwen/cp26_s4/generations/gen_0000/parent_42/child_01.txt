        y_pos += 0.2 # Vertical spacing

        for y in grid_y:
            for x in grid_x:
                temp_centers.append([x, y])
        centers = temp_centers[:n]
    elif len(centers) > n:
        centers = centers[:n]
    
    centers = np.array(centers)

        x0[3*i + 1] = centers[i, 1]
        x0[3*i + 2] = initial_radii[i]

    def objective(x_vars):
        # Objective is to maximize sum of radii -> minimize negative sum
        radii = x_vars[2::3]
        return -np.sum(radii)

    def gradient(x_vars):
        # Analytical gradient for the objective
        grad = np.zeros_like(x_vars)
        grad[2::3] = -1.0
        return grad

    cons = []

        xi = x_vars[idx]
        yi = x_vars[idx+1]
        ri = x_vars[idx+2]

        cons.append({'type': 'ineq', 'fun': lambda v, i=i: 1.0 - v[3*i] - v[3*i+2]})
        # Bottom boundary: y - r >= 0

        # Top boundary: 1 - y - r >= 0

        # Non-negative radius: r >= 0
