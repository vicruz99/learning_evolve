import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n_circles = 26
    n_vars = 3 * n_circles
    best_centers = None
    best_radii = None
    best_sum_r = -1.0

    # High penalty weights to strictly enforce constraints
    w_boundary = 100000.0
    w_overlap = 100000.0

    def objective(vars):
        x = vars[0::3]
        y = vars[1::3]
        r = vars[2::3]
        
        sum_r = np.sum(r)
        
        # 1. Boundary Penalties
        # r <= x, r <= 1-x, r <= y, r <= 1-y
        v_x = np.maximum(0, r - x)
        v_x_high = np.maximum(0, r - (1.0 - x))
        v_y = np.maximum(0, r - y)
        v_y_high = np.maximum(0, r - (1.0 - y))
        pen_boundary = np.sum(v_x**2 + v_x_high**2 + v_y**2 + v_y_high**2)
        
        # 2. Overlap Penalties
        # dist >= r_i + r_j
        diff_x = x[:, None] - x[None, :]
        diff_y = y[:, None] - y[None, :]
        dist = np.sqrt(diff_x**2 + diff_y**2)
        
        r_sum = r[:, None] + r[None, :]
        overlap = np.maximum(0, r_sum - dist)
        
        # Sum over upper triangle (i < j)
        # Mask for upper triangle
        mask = np.triu(np.ones((n_circles, n_circles), dtype=bool), k=1)
        pen_overlap = np.sum(overlap[mask]**2)
        
        # Objective: Minimize -sum_radii + penalties
        return -sum_r + w_overlap * pen_overlap + w_boundary * pen_boundary

    # Run multiple restarts
    n_restarts = 10
    for seed in range(n_restarts):
        rng = np.random.default_rng(seed)
        
        # Initialization: Random points with some margin
        centers_init = rng.uniform(0.15, 0.85, size=(n_circles, 2))
        radii_init = np.full(n_circles, 0.05)
        
        # Pack into vector
        x0 = np.zeros(n_vars)
        x0[0::3] = centers_init[:, 0]
        x0[1::3] = centers_init[:, 1]
        x0[2::3] = radii_init
        
        # Bounds for L-BFGS-B
        bounds = []
        for i in range(n_circles):
            bounds.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
            
        try:
            res = minimize(objective, x0, method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 2000, 'ftol': 1e-14})
            
            x_opt = res.x[0::3]
            y_opt = res.x[1::3]
            r_opt = res.x[2::3]
            
            # Post-processing: Check validity and shrink radii if necessary
            # Check overlaps
            diff_x = x_opt[:, None] - x_opt[None, :]
            diff_y = y_opt[:, None] - y_opt[None, :]
            dist = np.sqrt(diff_x**2 + diff_y**2)
            r_sum = r_opt[:, None] + r_opt[None, :]
            max_overlap = np.max(np.maximum(0, r_sum - dist))
            
            # Check boundaries
            max_bound_violation = np.max([
                np.maximum(0, r_opt - x_opt),
                np.maximum(0, r_opt - (1.0 - x_opt)),
                np.maximum(0, r_opt - y_opt),
                np.maximum(0, r_opt - (1.0 - y_opt))
            ])
            
            # Determine shrink factor to ensure strict validity
            # Overlap requires reducing sum of radii by overlap amount. 
            # If we reduce all radii by delta, r_sum reduces by 2*delta.
            # So 2*delta >= max_overlap.
            # Boundary requires delta >= max_bound_violation.
            
            delta_overlap = max_overlap / 2.0
            delta_bound = max_bound_violation
            
            delta = max(0.0, max(delta_overlap, delta_bound) + 1e-12)
            
            if delta > 0:
                r_opt = r_opt - delta
            
            sum_r = np.sum(r_opt)
            if sum_r > best_sum_r:
                best_sum_r = sum_r
                best_centers = np.column_stack((x_opt, y_opt))
                best_radii = r_opt

        except Exception as e:
            pass

    if best_centers is None:
        # Fallback initialization
        centers = np.random.rand(n_circles, 2)
        radii = np.full(n_circles, 0.0)
        return centers, radii, 0.0

    return best_centers, best_radii, best_sum_r
