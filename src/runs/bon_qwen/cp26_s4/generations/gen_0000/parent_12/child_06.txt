import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def objective(x):
    """Minimize negative sum of radii (i.e., maximize sum of radii)"""
    return -np.sum(x[2 * N_CIRCLES:])


def constraints(x):
    """Return all inequality constraints (must be >= 0)"""
    centers = x[:2 * N_CIRCLES].reshape(N_CIRCLES, 2)
    radii = x[2 * N_CIRCLES:]
    
    cons = []
    
    # Boundary constraints: each circle must be inside [0,1]x[0,1]
    for i in range(N_CIRCLES):
        # x - r >= 0
        cons.append(centers[i, 0] - radii[i])
        # 1 - x - r >= 0  (i.e., x + r <= 1)
        cons.append(1 - centers[i, 0] - radii[i])
        # y - r >= 0
        cons.append(centers[i, 1] - radii[i])
        # 1 - y - r >= 0  (i.e., y + r <= 1)
        cons.append(1 - centers[i, 1] - radii[i])
    
    # Non-overlap constraints: distance between centers >= sum of radii
    for i in range(N_CIRCLES):
        for j in range(i + 1, N_CIRCLES):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            cons.append(dist - radii[i] - radii[j])
    
    return np.array(cons)


def create_hexagonal_config():
    """Create hexagonal grid initial configuration with pattern 5-6-5-6-5"""
    pattern = [5, 6, 5, 6, 5]
    r0 = 0.095
    
    centers = np.zeros((N_CIRCLES, 2))
    radii = np.ones(N_CIRCLES) * r0
    
    idx = 0
    for row, count in enumerate(pattern):
        y = r0 + row * r0 * np.sqrt(3)
        offset = (row % 2) * r0
        for col in range(count):
            x = r0 + col * 2 * r0 + offset
            centers[idx] = [x, y]
            idx += 1
    
    return centers, radii


def create_grid_config():
    """Create a 5x5 grid plus one extra circle"""
    centers = np.zeros((N_CIRCLES, 2))
    radii = np.ones(N_CIRCLES) * 0.08
    
    idx = 0
    for i in range(5):
        for j in range(5):
            centers[idx] = [0.1 + i * 0.2, 0.1 + j * 0.2]
            idx += 1
    # Place 26th circle in center gap
    centers[25] = [0.5, 0.5]
    
    return centers, radii


def create_random_config(seed):
    """Create random initial configuration"""
    np.random.seed(seed)
    centers = np.random.rand(N_CIRCLES, 2) * 0.6 + 0.2
    radii = np.ones(N_CIRCLES) * 0.06
    return centers, radii


def optimize_from_initial(centers, radii):
    """Run optimization from given initial configuration"""
    x0 = np.concatenate([centers.flatten(), radii])
    
    bounds = [(0, 1)] * (2 * N_CIRCLES) + [(1e-6, 0.5)] * N_CIRCLES
    
    cons_dict = {'type': 'ineq', 'fun': constraints}
    
    result = minimize(
        objective,
        x0,
        method='SLSQP',
        bounds=bounds,
        constraints=cons_dict,
        options={
            'maxiter': 3000,
            'ftol': 1e-15,
            'disp': False
        }
    )
    
    return result


def refine_solution(centers, radii):
    """Apply a simple refinement pass to improve radii given fixed positions"""
    n = len(radii)
    improved_radii = radii.copy()
    
    for i in range(n):
        max_r = min(
            centers[i, 0],
            1 - centers[i, 0],
            centers[i, 1],
            1 - centers[i, 1]
        )
        for j in range(n):
            if i != j:
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                max_r = min(max_r, dist - radii[j])
        improved_radii[i] = max(1e-8, max_r * 0.9999)
    
    return improved_radii


def run_packing():
    """Pack 26 circles in unit square to maximize sum of radii"""
    best_sum = -1
    best_centers = None
    best_radii = None
    
    # Collect multiple initial configurations to try
    initial_configs = [
        create_hexagonal_config(),
        create_grid_config(),
        create_random_config(42),
        create_random_config(123),
        create_random_config(456),
        create_random_config(789),
        create_random_config(101),
    ]
    
    for idx, (centers, radii) in enumerate(initial_configs):
        result = optimize_from_initial(centers, radii)
        
        current_sum = np.sum(result.x[2 * N_CIRCLES:])
        
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = result.x[:2 * N_CIRCLES].reshape(N_CIRCLES, 2)
            best_radii = result.x[2 * N_CIRCLES:]
    
    # Additional refinement: re-optimize radii given best positions
    best_radii = refine_solution(best_centers, best_radii)
    best_sum = np.sum(best_radii)
    
    # Ensure no NaN or negative values
    best_radii = np.maximum(best_radii, 0)
    best_centers = np.clip(best_centers, 0, 1)
    
    # Final optimization from refined solution
    x0 = np.concatenate([best_centers.flatten(), best_radii])
    bounds = [(0, 1)] * (2 * N_CIRCLES) + [(1e-6, 0.5)] * N_CIRCLES
    
    result2 = minimize(
        objective,
        x0,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'ineq', 'fun': constraints},
        options={'maxiter': 5000, 'ftol': 1e-15, 'disp': False}
    )
    
    final_sum = np.sum(result2.x[2 * N_CIRCLES:])
    if final_sum > best_sum:
        best_centers = result2.x[:2 * N_CIRCLES].reshape(N_CIRCLES, 2)
        best_radii = result2.x[2 * N_CIRCLES:]
        best_sum = final_sum
    
    best_radii = np.maximum(best_radii, 0)
    best_centers = np.clip(best_centers, 0, 1)
    
    return best_centers, best_radii, np.sum(best_radii)
