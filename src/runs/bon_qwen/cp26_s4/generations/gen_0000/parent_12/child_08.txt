from scipy.optimize import linprog

def run_packing():

    centers = []
    r_init = 0.095

    row_idx = 0

        x = r_init
        while x + r_init <= 1.0:
            if row_idx % 2 == 1:
                x += r_init
            if len(centers) >= 26:
                break
            centers.append([x, r_init + row_idx * y_offset])
            x += 2 * r_init
        row_idx += 1
        if row_idx > 6: break
        
    centers = np.array(centers[:26])

    # 2. Force-Directed Relaxation
    num_circles = 26
    for iteration in range(300):

        forces = np.zeros((num_circles, 2))

        dist_matrix = np.zeros((num_circles, num_circles))
        for i in range(num_circles):
            dist_matrix[i, i] = 1000
            for j in range(i + 1, num_circles):
                d = np.sqrt(np.sum((centers[i] - centers[j])**2))
                dist_matrix[i, j] = d
                dist_matrix[j, i] = d

        c_obj = -np.ones(num_circles)
        A_ub = []
        b_ub = []
        
        for i in range(num_circles):
            # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, etc.
            limits = [centers[i, 0], 1 - centers[i, 0],

                row = np.zeros(num_circles)
                row[i] = 1.0
                A_ub.append(row)
                b_ub.append(limit)

            for j in range(i + 1, num_circles):
                row = np.zeros(num_circles)
                row[i] = 1.0
                row[j] = 1.0
                A_ub.append(row)
                b_ub.append(dist_matrix[i, j])

        bounds = [(0, None) for _ in range(num_circles)]
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')

            radii = res.x
        else:
            break
            
        # Compute forces based on tight constraints
        for i in range(num_circles):
            fx, fy = 0.0, 0.0
            for j in range(num_circles):
                if i == j: continue
                d = dist_matrix[i, j]
                sum_r = radii[i] + radii[j]

                if sum_r >= d * 0.95:

                    dx = centers[j, 0] - centers[i, 0]
                    dy = centers[j, 1] - centers[i, 1]
                    fx -= (dx / d) * force_mag
                    fy -= (dy / d) * force_mag

                    if coord == 0: fx += direction * repel
                    else: fy += direction * repel
                    
            forces[i, 0] = fx
            forces[i, 1] = fy
            
        centers += forces * 0.005

    # 3. Final High-Precision LP Solve
    c_obj = -np.ones(num_circles)
    A_ub = []
    b_ub = []
    
    for i in range(num_circles):
        limits = [centers[i, 0], 1 - centers[i, 0],

        for limit in limits:
            row = np.zeros(num_circles)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(limit)

            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)

    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')

        radii = res.x
    else:

    # 4. Final cleanup for numerical stability
    radii = np.clip(radii, 0, None)
    
    sum_radii = float(np.sum(radii))
    return centers, radii, sum_radii
