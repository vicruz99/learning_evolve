_OPT_PENALTY_WEIGHT = 100

        yi = x[3 * i + 1]

        penalty += max(0, yi + ri - 1) ** 2  # Top boundary
        penalty += max(0, -ri) ** 2  # Non-negative radius

def objective_with_penalty(x):

    n = _OPT_N
