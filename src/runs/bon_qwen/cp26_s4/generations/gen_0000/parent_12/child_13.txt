import numpy as np
from scipy.optimize import minimize

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    n = centers.shape[0]
    if np.isnan(centers).any() or np.isnan(radii).any():
        return False
    for i in range(n):
        if radii[i] < 0: return False
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            return False
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False
    return True

def generate_initial_hex_layout(n=26):
    """Generates a hexagonal packing layout for n circles."""
    centers = []
    # Strategy: 6 rows, alternating 5 and 4 circles. 
    # Rows: 5, 4, 5, 4, 5, 4 (Total 27, we'll drop 1 if needed)
    row_counts = [5, 4, 5, 4, 5, 4]
    
    # We want to fit these in a 1x1 box initially for scaling
    # Vertical spacing for hex: sqrt(3)/2 * 2r = sqrt(3)r. 
    # But since r is variable, let's just define a normalized lattice.
    # Lattice constant a = 1.
    # y_k = k * sqrt(3)/2
    # x_even = 0, x_odd = 0.5
    
    rows = 6
    h_step = np.sqrt(3)/2
    w_step = 1.0 # Relative width step
    
    current_row = 0
    for r_idx, count in enumerate(row_counts):
        if len(centers) >= n:
            break
        
        # y position
        y_pos = r_idx * h_step
        
        # x positions
        # If even row (0, 2, 4), shift is 0. If odd (1, 3, 5), shift is 0.5
        shift = 0.5 if r_idx % 2 != 0 else 0.0
        
        # Determine x positions to center the row
        # Width of row with 'count' circles is (count-1) * w_step
        # We want this centered in [0, 1]
        # Let's just map 0..count-1 to normalized x
        
        # Actually, for initialization, let's just space them evenly 
        # in a way that respects hex packing geometry.
        
        # Let's create points.
        for c in range(count):
            x_pos = shift + c * w_step
            centers.append([x_pos, y_pos])
            
    # We might have more than n points. Trim to n.
    centers = centers[:n]
    centers = np.array(centers)
    
    # Center the cluster in [0,1]
    min_c = centers.min(axis=0)
    max_c = centers.max(axis=0)
    width = max_c[0] - min_c[0]
    height = max_c[1] - min_c[1]
    
    # Scale to fit with some margin, e.g., 0.8 of the box
    scale = 0.8 / max(width, height)
    centers = (centers - min_c) * scale
    # Center in square
    centers += 0.5 - (centers.min(axis=0) + centers.max(axis=0)) / 2
    
    return centers

def run_force_expansion(centers, radii, iterations=2000, dt=0.001):
    """
    Iteratively expands radii and moves centers to resolve overlaps.
    Uses a simple repulsion force model.
    """
    n = len(radii)
    centers = centers.copy().astype(np.float64)
    radii = radii.copy().astype(np.float64)
    
    for _ in range(iterations):
        # Try to expand radii slightly
        # Expansion rate proportional to current radius to keep relative sizes, 
        # or constant to increase smaller ones faster? 
        # Constant growth pushes towards equal radii, which is usually good for sum.
        radii += 1e-5 
        
        # Apply forces
        forces = np.zeros_like(centers)
        
        # Circle-Circle repulsion
        for i in range(n):
            for j in range(i + 1, n):
                diff = centers[i] - centers[j]
                dist = np.linalg.norm(diff)
                min_dist = radii[i] + radii[j]
                
                if dist < min_dist and dist > 1e-9:
                    # Overlap detected
                    overlap = min_dist - dist
                    # Force vector
                    dir_vec = diff / dist
                    # Push apart. Weighted by inverse radius? Or equal?
                    # Equal force is simple.
                    f_mag = overlap * 0.1 # Spring constant
                    forces[i] += dir_vec * f_mag
                    forces[j] -= dir_vec * f_mag
        
        # Boundary forces
        for i in range(n):
            # Left wall
            if centers[i, 0] < radii[i]:
                forces[i, 0] += (radii[i] - centers[i, 0]) * 0.5
            # Right wall
            if centers[i, 0] > 1.0 - radii[i]:
                forces[i, 0] -= (centers[i, 0] - (1.0 - radii[i])) * 0.5
            # Bottom wall
            if centers[i, 1] < radii[i]:
                forces[i, 1] += (radii[i] - centers[i, 1]) * 0.5
            # Top wall
            if centers[i, 1] > 1.0 - radii[i]:
                forces[i, 1] -= (centers[i, 1] - (1.0 - radii[i])) * 0.5

        # Update centers
        centers += forces * dt
        
        # Clamp centers strictly inside bounds defined by radii to ensure validity
        # This acts as a hard constraint enforcement
        for i in range(n):
            r = radii[i]
            centers[i, 0] = np.clip(centers[i, 0], r, 1.0 - r)
            centers[i, 1] = np.clip(centers[i, 1], r, 1.0 - r)
            
        # If radii became too large causing impossible constraints, shrink back slightly
        # Check for any extreme overlaps that force can't resolve immediately
        # (Handled by clamping and next iter)
        
    return centers, radii
