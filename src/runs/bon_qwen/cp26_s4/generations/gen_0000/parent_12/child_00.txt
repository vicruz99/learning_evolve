import numpy as np
from scipy.optimize import minimize


N = 26


def create_hexagonal_packing():
    """Create initial hexagonal grid packing."""
    rows_config = [5, 5, 5, 5, 5, 1]
    centers = []
    r_est = 0.094

    for row_idx, num in enumerate(rows_config):
        y = r_est + row_idx * np.sqrt(3) * r_est
        if row_idx % 2 == 0:
            x_start = r_est
        else:
            x_start = 2 * r_est
        for col in range(num):
            x = x_start + col * 2 * r_est
            centers.append([x, y])

    centers = np.array(centers[:N])
    radii = np.full(N, r_est)
    return centers, radii


def create_alternating_packing():
    """Create alternating row packing for variety."""
    rows_config = [5, 4, 5, 4, 5, 3]
    centers = []
    r_est = 0.1

    for row_idx, num in enumerate(rows_config):
        y = r_est + row_idx * np.sqrt(3) * r_est
        if row_idx % 2 == 0:
            x_start = r_est
        else:
            x_start = 2 * r_est
        for col in range(num):
            x = x_start + col * 2 * r_est
            centers.append([x, y])

    centers = np.array(centers[:N])
    radii = np.full(N, r_est)
    return centers, radii


def create_random_packing(seed=42):
    """Create a random initial packing."""
    rng = np.random.RandomState(seed)
    centers = rng.uniform(0.2, 0.8, (N, 2))
    radii = np.full(N, 0.05)
    return centers, radii


def force_based_optimization(centers, radii, n, num_iterations=5000):
    """Optimize using force-directed approach with simultaneous radius growth."""
    centers = centers.copy()
    radii = radii.copy()

    for iteration in range(num_iterations):
        forces = np.zeros_like(centers)
        cooling = 1.0 / (1.0 + iteration * 0.0005)

        # Compute repulsion forces between overlapping circles
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist_sq = dx * dx + dy * dy
                dist = np.sqrt(dist_sq + 1e-16)
                min_dist = radii[i] + radii[j]

                if dist < min_dist and dist > 1e-10:
                    overlap = min_dist - dist
                    fx = overlap * dx / dist
                    fy = overlap * dy / dist
                    forces[i, 0] += fx
                    forces[i, 1] += fy
                    forces[j, 0] -= fx
                    forces[j, 1] -= fy

                # Add slight repulsion even when not overlapping to maintain separation
                elif dist < min_dist * 1.1 and dist > 1e-10:
                    repel = (min_dist * 1.1 - dist) * 0.1
                    fx = repel * dx / dist
                    fy = repel * dy / dist
                    forces[i, 0] += fx
                    forces[i, 1] += fy
                    forces[j, 0] -= fx
                    forces[j, 1] -= fy

        # Apply boundary repulsion forces
        for i in range(n):
            r = radii[i]
            margin = 0.001
            if centers[i, 0] < r + margin:
                forces[i, 0] += (r + margin - centers[i, 0]) * 2.0
            if centers[i, 0] > 1 - r - margin:
                forces[i, 0] -= (centers[i, 0] - (1 - r - margin)) * 2.0
            if centers[i, 1] < r + margin:
                forces[i, 1] += (r + margin - centers[i, 1]) * 2.0
            if centers[i, 1] > 1 - r - margin:
                forces[i, 1] -= (centers[i, 1] - (1 - r - margin)) * 2.0

        # Update positions
        step_size = 0.05 * cooling
        centers += step_size * forces

        # Clamp to valid range
        for i in range(n):
            r = radii[i]
            centers[i, 0] = max(r, min(1 - r, centers[i, 0]))
            centers[i, 1] = max(r, min(1 - r, centers[i, 1]))

        # Try to grow radii based on available space
        for i in range(n):
            max_r = min(centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])
            for j in range(n):
                if i != j:
                    dx = centers[i, 0] - centers[j, 0]
                    dy = centers[i, 1] - centers[j, 1]
                    dist = np.sqrt(dx * dx + dy * dy)
                    max_r = min(max_r, dist - radii[j])
            growth_factor = 1.0 + 0.02 * cooling
            radii[i] = min(max(0, max_r), radii[i] * growth_factor)

    return centers, radii


def smooth_penalized_objective(x, penalty_weight):
    """Objective with smooth penalty for constraints."""
    centers = x[:2 * N].reshape((N, 2))
    radii = x[2 * N:]

    obj = -np.sum(radii)
    penalty = 0.0

    # Boundary penalties (smooth)
    for i in range(N):
        cx, cy = centers[i]
        r = radii[i]

        # x - r >= 0
        val = r - cx
        if val > 0:
            penalty += penalty_weight * val * val
        # x + r <= 1
        val = cx + r - 1
        if val > 0:
            penalty += penalty_weight * val * val
        # y - r >= 0
        val = r - cy
        if val > 0:
            penalty += penalty_weight * val * val
        # y + r <= 1
        val = cy + r - 1
        if val > 0:
            penalty += penalty_weight * val * val

    # Overlap penalties (smooth)
    for i in range(N):
        for j in range(i + 1, N):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy + 1e-16)
            min_dist = radii[i] + radii[j]
            val = min_dist - dist
            if val > 0:
                penalty += penalty_weight * val * val

    return obj + penalty


def objective_function(x):
    """Pure objective: negative sum of radii."""
    radii = x[2 * N:]
    return -np.sum(radii)


def constraint_x_min(x, i):
    return x[2 * i] - x[2 * N + i]


def constraint_x_max(x, i):
    return 1 - x[2 * i] - x[2 * N + i]


def constraint_y_min(x, i):
    return x[2 * i + 1] - x[2 * N + i]


def constraint_y_max(x, i):
    return 1 - x[2 * i + 1] - x[2 * N + i]


def constraint_radius_pos(x, i):
    return x[2 * N + i]


def constraint_overlap(x, i, j):
    dx = x[2 * i] - x[2 * j]
    dy = x[2 * i + 1] - x[2 * j + 1]
    dist = np.sqrt(dx * dx + dy * dy)
    return dist - x[2 * N + i] - x[2 * N + j]


def build_constraints():
    """Build constraint list for SLSQP."""
    constraints = []

    for i in range(N):
        constraints.append({'type': 'ineq', 'fun': constraint_x_min, 'args': (i,)})
        constraints.append({'type': 'ineq', 'fun': constraint_x_max, 'args': (i,)})
        constraints.append({'type': 'ineq', 'fun': constraint_y_min, 'args': (i,)})
        constraints.append({'type': 'ineq', 'fun': constraint_y_max, 'args': (i,)})
        constraints.append({'type': 'ineq', 'fun': constraint_radius_pos, 'args': (i,)})

    for i in range(N):
        for j in range(i + 1, N):
            constraints.append({'type': 'ineq', 'fun': constraint_overlap, 'args': (i, j)})

    return constraints


def build_bounds():
    """Build bounds for optimizer."""
    bounds = []
    for i in range(N):
        bounds.append((0.0, 1.0))
        bounds.append((0.0, 1.0))
        bounds.append((0.0, 0.5))
    return bounds


def project_to_feasible(centers, radii):
    """Project solution to ensure feasibility."""
    centers = centers.copy()
    radii = radii.copy()

    # Ensure non-negative radii
    for i in range(N):
        radii[i] = max(0.0, radii[i])

    # Iterate to resolve overlaps
    for iteration in range(100):
        max_violation = 0.0

        for i in range(N):
            # Project center to boundary-feasible region
            r = radii[i]
            centers[i, 0] = max(r, min(1 - r, centers[i, 0]))
            centers[i, 1] = max(r, min(1 - r, centers[i, 1]))

        # Check and resolve overlaps
        for i in range(N):
            for j in range(i + 1, N):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                min_dist = radii[i] + radii[j]

                if dist < min_dist and dist > 1e-12:
                    overlap = min_dist - dist
                    # Shrink the smaller circle more
                    total_r = radii[i] + radii[j]
                    if total_r > 0:
                        scale_i = radii[j] / total_r
                        scale_j = radii[i] / total_r
                    else:
                        scale_i = 0.5
                        scale_j = 0.5

                    radii[i] = max(0, radii[i] - overlap * scale_i * 0.5)
                    radii[j] = max(0, radii[j] - overlap * scale_j * 0.5)
                    max_violation = max(max_violation, overlap)

        if max_violation < 1e-12:
            break

    return centers, radii


def penalty_optimization(x0, num_phases=8):
    """Run penalty method with increasing penalty weights."""
    x_current = x0.copy()
    bounds = build_bounds()

    for phase in range(num_phases):
        penalty_weight = 100 * (10 ** phase)

        result = minimize(
            smooth_penalized_objective,
            x_current,
            method='L-BFGS-B',
            bounds=bounds,
            args=(penalty_weight,),
            options={'maxiter': 3000, 'ftol': 1e-15, 'gtol': 1e-10}
        )
        x_current = result.x.copy()

    return x_current


def slsqp_optimization(x0):
    """Run SLSQP optimization with explicit constraints."""
    bounds = build_bounds()
    constraints = build_constraints()

    result = minimize(
        objective_function,
        x0,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 3000, 'ftol': 1e-12}
    )

    return result.x


def optimize_from_init(centers_init, radii_init):
    """Full optimization pipeline from an initial configuration."""
    centers = centers_init.copy()
    radii = radii_init.copy()

    # Phase 1: Force-based optimization
    centers, radii = force_based_optimization(centers, radii, N, num_iterations=3000)

    x0 = np.concatenate([centers.flatten(), radii])

    # Phase 2: Penalty method optimization
    x_opt = penalty_optimization(x0, num_phases=6)

    centers = x_opt[:2 * N].reshape((N, 2))
    radii = x_opt[2 * N:]

    # Phase 3: SLSQP fine-tuning
    try:
        x_opt2 = slsqp_optimization(x_opt)
        centers = x_opt2[:2 * N].reshape((N, 2))
        radii = x_opt2[2 * N:]
    except Exception:
        pass

    # Project to feasible
    centers, radii = project_to_feasible(centers, radii)

    return centers, radii


def run_packing():
    best_centers = None
    best_radii = None
    best_sum = -1.0

    # Try multiple initial configurations
    initial_configs = [
        create_hexagonal_packing(),
        create_alternating_packing(),
        create_random_packing(42),
        create_random_packing(123),
        create_random_packing(456),
    ]

    for centers_init, radii_init in initial_configs:
        try:
            centers, radii = optimize_from_init(centers_init, radii_init)
            current_sum = np.sum(radii)

            if current_sum > best_sum:
                best_sum = current_sum
                best_centers = centers.copy()
                best_radii = radii.copy()
        except Exception:
            continue

    # Sort circles by radius (optional, for readability)
    # Keep as is for consistency

    return best_centers, best_radii, float(best_sum)
