class BoundaryConstraint:
    def __init__(self, circle_idx, dim, direction):
        # dim: 0 for x, 1 for y
        # direction: 1 for lower bound (x-r >= 0), -1 for upper bound (x+r <= 1)
        self.idx = circle_idx
        self.dim = dim
        self.direction = direction
        
    def __call__(self, v):
        x = v[self.idx * 3 + self.dim]
        r = v[self.idx * 3 + 2]
        if self.direction == 1:
            return x - r
        else:
            return 1.0 - (x + r)

class RadiusConstraint:
    def __init__(self, circle_idx):
        self.idx = circle_idx
        
    def __call__(self, v):
        return v[self.idx * 3 + 2]

class OverlapConstraint:
    def __init__(self, i, j):
        self.i = i
        self.j = j
        
    def __call__(self, v):
        # dist^2 - (r1 + r2)^2 >= 0
        xi, yi, ri = v[self.i*3], v[self.i*3+1], v[self.i*3+2]
        xj, yj, rj = v[self.j*3], v[self.j*3+1], v[self.j*3+2]
        dist_sq = (xi - xj)**2 + (yi - yj)**2
        sum_r = ri + rj
        return dist_sq - sum_r**2
