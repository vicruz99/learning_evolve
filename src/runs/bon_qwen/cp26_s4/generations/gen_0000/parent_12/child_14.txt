import numpy as np

N_CIRCLES = 26

def objective_function(vars_array):
    """Objective: Maximize sum of radii. Here we optimize equal radius r."""
    # vars_array: [x1, y1, ..., xN, yN, r]
    r = vars_array[-1]
    return -r

def boundary_constraint_x_min(vars_array, i):
    """Constraint: x_i - r >= 0"""
    return vars_array[2*i] - vars_array[-1]

def boundary_constraint_x_max(vars_array, i):
    """Constraint: 1 - x_i - r >= 0"""
    return 1.0 - vars_array[2*i] - vars_array[-1]

def boundary_constraint_y_min(vars_array, i):
    """Constraint: y_i - r >= 0"""
    return vars_array[2*i+1] - vars_array[-1]

def boundary_constraint_y_max(vars_array, i):
    """Constraint: 1 - y_i - r >= 0"""
    return 1.0 - vars_array[2*i+1] - vars_array[-1]

def overlap_constraint(vars_array, i, j):
    """Constraint: dist^2 - 4r^2 >= 0"""
    xi = vars_array[2*i]
    yi = vars_array[2*i+1]
    xj = vars_array[2*j]
    yj = vars_array[2*j+1]
    r = vars_array[-1]
    dist_sq = (xi - xj)**2 + (yi - yj)**2
    return dist_sq - 4 * r**2

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Define bounds
    bounds = []
    for _ in range(N_CIRCLES):
        bounds.append((0.0, 1.0)) # x

            constraints.append({'type': 'ineq', 'fun': lambda v, i=i, j=j: overlap_constraint(v, i, j)})

    best_r = 0.0

            r_init = 0.05

    radii = np.full(N_CIRCLES, final_r)
    sum_radii = np.sum(radii)
    
    return final_centers, radii, sum_radii
