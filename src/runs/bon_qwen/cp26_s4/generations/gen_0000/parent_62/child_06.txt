import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def objective(params, n):
    """
    Objective function to minimize: -sum(radii)
    """
    radii = params[2::3]
    return -np.sum(radii)

def constraints(params, n, i_idx, j_idx):
    """
    Constraint functions: all must be >= 0
    1. Boundary: x - r >= 0, 1 - x - r >= 0, y - r >= 0, 1 - y - r >= 0
    2. Non-overlap: dist(i,j) - (r_i + r_j) >= 0
    """
    centers = params.reshape(n, 3)
    X = centers[:, 0]
    Y = centers[:, 1]
    R = centers[:, 2]
    
    cons = []
    
    # Boundary constraints
    cons.append(X - R)
    cons.append(1.0 - X - R)
    cons.append(Y - R)
    cons.append(1.0 - Y - R)
    
    # Overlap constraints
    # Vectorized computation of pairwise distances
    dx = X[:, np.newaxis] - X[np.newaxis, :]
    dy = Y[:, np.newaxis] - Y[np.newaxis, :]
    dist_mat = np.sqrt(dx*dx + dy*dy)
    
    r_sum_mat = R[:, np.newaxis] + R[np.newaxis, :]
    
    # Apply precomputed indices for i < j
    overlap_constr = dist_mat[i_idx, j_idx] - r_sum_mat[i_idx, j_idx]
    cons.append(overlap_constr)
    
    return np.concatenate(cons)

def run_packing():
    N = N_CIRCLES
    
    # Bounds for variables: x, y in [0, 1], r in [0, 0.5]
    bounds = [(0, 1)] * (2 * N) + [(0, 0.5)] * N
    
    # Precompute indices for upper triangle of distance matrix
    i_idx, j_idx = np.triu_indices(N, k=1)
    
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Run multiple optimization trials with different initializations
    for trial in range(40):
        # Generate initial centers using a perturbed grid
        x_vals = np.linspace(0.1, 0.9, 6)
        y_vals = np.linspace(0.1, 0.9, 5)
        pts = []
        for yi in y_vals:
            for xi in x_vals:
                pts.append([xi, yi])
                if len(pts) >= N:
                    break
            if len(pts) >= N:
                break
        
        centers_init = np.array(pts[:N])
        # Add random perturbation to escape symmetric local minima
        centers_init += np.random.normal(0, 0.05, (N, 2))
        centers_init = np.clip(centers_init, 0.05, 0.95)
        
        # Start with small valid radii
        radii_init = np.full(N, 0.02)
        
        # Flatten parameters
        params0 = np.zeros(3 * N)
        params0[0::3] = centers_init[:, 0]
        params0[1::3] = centers_init[:, 1]
        params0[2::3] = radii_init
        
        try:
            # Optimize
            res = minimize(objective, params0, args=(N, i_idx, j_idx),
                          method='SLSQP', 
                          constraints={'type': 'ineq', 'fun': constraints, 'args': (N, i_idx, j_idx)},
                          bounds=bounds,
                          options={'maxiter': 1000, 'ftol': 1e-10, 'disp': False})
            
            # Extract results
            centers_res = res.x.reshape(N, 3)[:, :2]
            radii_res = res.x.reshape(N, 3)[:, 2]
            
            # Validate result strictly
            valid = True
            
            # Boundary check
            if np.any(centers_res[:, 0] - radii_res < -1e-9) or \
               np.any(centers_res[:, 0] + radii_res > 1 + 1e-9) or \
               np.any(centers_res[:, 1] - radii_res < -1e-9) or \
               np.any(centers_res[:, 1] + radii_res > 1 + 1e-9):
                valid = False
            
            # Overlap check
            if valid:
                dx = centers_res[:, 0][:, None] - centers_res[:, 0][None, :]
                dy = centers_res[:, 1][:, None] - centers_res[:, 1][None, :]
                dists = np.sqrt(dx*dx + dy*dy)
                r_sums = radii_res[:, None] + radii_res[None, :]
                
                # Check upper triangle
                if np.any(dists[i_idx, j_idx] < r_sums[i_idx, j_idx] - 1e-9):
                    valid = False
            
            if valid:
                s = np.sum(radii_res)
                if s > best_sum:
                    best_sum = s
                    best_centers = centers_res.copy()
                    best_radii = radii_res.copy()
        except Exception:
            continue
            
    # Fallback if no valid solution found (should not happen with this setup)
    if best_centers is None:
        best_centers = np.random.uniform(0.2, 0.8, (N, 2))
        best_radii = np.full(N, 0.05)
        best_sum = np.sum(best_radii)
        
    return best_centers, best_radii, best_sum
