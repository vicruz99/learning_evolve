from scipy.optimize import minimize

N_CIRCLES = 26

def objective(v):

    return -np.sum(v[2::3])

def constraints_func(v):
    """Compute all inequality constraints: boundaries and pairwise non-overlap."""
    n = N_CIRCLES
    cx = v[0::3]
    cy = v[1::3]
    cr = v[2::3]
    
    # Total constraints: 4 per circle (boundaries) + N*(N-1)/2 (pairs)
    cons = np.empty(4*n + n*(n-1)//2)

    cons[0::4] = cx - cr
    cons[1::4] = 1.0 - cx - cr
    cons[2::4] = cy - cr
    cons[3::4] = 1.0 - cy - cr

    idx = 4*n
    for i in range(n):

        for j in range(i+1, n):

            dy = yi - cy[j]

            cons[idx] = dx*dx + dy*dy - dr*dr
            idx += 1
    return cons

def get_initial_guess(n, seed=0):
