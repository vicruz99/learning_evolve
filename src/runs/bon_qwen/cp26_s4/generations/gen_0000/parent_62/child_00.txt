import numpy as np
from scipy.optimize import minimize
import math

def objective(vars, n):
    """Objective function: Minimize negative sum of radii."""
    r_sum = 0.0
    for i in range(n):
        r_sum += vars[3 * i + 2]
    return -r_sum

def get_boundary_con(vars, n, i):
    """Check boundary constraints for circle i."""
    x = vars[3 * i]
    y = vars[3 * i + 1]
    r = vars[3 * i + 2]
    
    # Return the minimum slack; must be >= 0
    c1 = x - r
    c2 = 1.0 - x - r
    c3 = y - r
    c4 = 1.0 - y - r
    
    return min(c1, c2, c3, c4)

def get_distance_con(vars, n, i, j):
    """Check non-overlap constraints for pair (i, j)."""
    xi, yi, ri = vars[3 * i], vars[3 * i + 1], vars[3 * i + 2]
    xj, yj, rj = vars[3 * j], vars[3 * j + 1], vars[3 * j + 2]
    
    dist_sq = (xi - xj)**2 + (yi - yj)**2
    sum_r = ri + rj
    
    return dist_sq - sum_r**2

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Number of optimization attempts
    num_attempts = 15
    
    for attempt in range(num_attempts):
        # 1. Initialization
        centers = []
        radii = []
        
        if attempt == 0:
            # Start with a 5x5 grid + 1 center
            grid_size = 5
            spacing = 1.0 / (grid_size + 1)
            
            idx = 0
            for r_idx in range(grid_size):
                for c_idx in range(grid_size):
                    x = spacing * (c_idx + 1)
                    y = spacing * (r_idx + 1)
                    centers.append([x, y])
                    radii.append(0.08) # Start slightly smaller than optimal
                    idx += 1
            
            centers.append([0.5, 0.5])
            radii.append(0.08)
        else:
            # Perturb a previous good result or use random grid
            if best_centers is not None:
                # Add random noise to best known solution
                centers = best_centers + np.random.uniform(-0.02, 0.02, size=best_centers.shape)
                radii = best_radii * np.random.uniform(0.9, 1.1, size=n)
            else:
                # Random initialization
                centers = np.random.uniform(0.1, 0.9, size=(n, 2))
                radii = np.full(n, 0.05)

        # Ensure valid bounds for initial guess
        centers = np.clip(centers, 0.0, 1.0)
        radii = np.clip(radii, 0.0, 0.5)
        
        # Flatten variables
        x0 = np.zeros(3 * n)
        for i in range(n):
            x0[3 * i] = centers[i, 0]
            x0[3 * i + 1] = centers[i, 1]
            x0[3 * i + 2] = radii[i]
            
        # Bounds
        bounds = []
        for i in range(n):
            bounds.append((0.0, 1.0)) # x
            bounds.append((0.0, 1.0)) # y
            bounds.append((0.0, 0.5)) # r
            
        # Constraints
        cons = []
        
        # Boundary constraints (one per circle to reduce constraint list size)
        for i in range(n):
            cons.append({'type': 'ineq', 'fun': lambda v, n=n, i=i: get_boundary_con(v, n, i)})
            
        # Distance constraints
        for i in range(n):
            for j in range(i + 1, n):
                cons.append({'type': 'ineq', 'fun': lambda v, n=n, i=i, j=j: get_distance_con(v, n, i, j)})
                
        # Optimization
        res = minimize(
            objective, 
            x0, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            args=(n,),
            options={'maxiter': 500, 'ftol': 1e-9}
        )
        
        if res.success and -res.fun > best_sum:
            # Validate and store if it's the best valid result so far
            # (Simple validation check inside the loop to ensure no overlaps)
            valid = True
            for k in range(n):
                xk, yk, rk = res.x[3*k], res.x[3*k+1], res.x[3*k+2]
                if xk < 0 or xk > 1 or yk < 0 or yk > 1 or rk < 0:
                    valid = False
                    break
            
            if valid:
                # Check overlaps for the new candidate
                for k in range(n):
                    xk, yk, rk = res.x[3*k], res.x[3*k+1], res.x[3*k+2]
                    if xk - rk < -1e-12 or xk + rk > 1 + 1e-12 or yk - rk < -1e-12 or yk + rk > 1 + 1e-12:
                        valid = False
                        break
                    if not valid: break
                
                for k in range(n):
                    for m in range(k + 1, n):
                        xk, yk, rk = res.x[3*k], res.x[3*k+1], res.x[3*k+2]
                        xm, ym, rm = res.x[3*m], res.x[3*m+1], res.x[3*m+2]
                        dist = math.sqrt((xk-xm)**2 + (yk-ym)**2)
                        if dist < rk + rm - 1e-12:
                            valid = False
                            break
                    if not valid: break
                
                if valid:
                    best_sum = -res.fun
                    best_centers = np.array([[res.x[3*i], res.x[3*i+1]] for i in range(n)])
                    best_radii = np.array([res.x[3*i+2] for i in range(n)])

    return best_centers, best_radii, best_sum
