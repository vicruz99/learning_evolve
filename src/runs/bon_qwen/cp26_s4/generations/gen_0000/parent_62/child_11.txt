   - Boundary penalties: circles must stay within [0,1]²
   - Overlap penalties: circles must not overlap

3. **Multi-start**: Try multiple initial configurations (hexagonal + perturbed) to avoid local minima.

import numpy as np
from scipy.optimize import minimize

N = 26


def compute_objective_and_penalty(vars):
    centers = vars[:2 * N].reshape(N, 2)
    radii = vars[2 * N:]

    penalty = 0.0

    # Boundary penalties
    for i in range(N):
        x, y = centers[i]
        r = radii[i]
        violation = r - x
        if violation > 0:
            penalty += 5000 * violation ** 2
        violation = r - (1 - x)
        if violation > 0:
            penalty += 5000 * violation ** 2
        violation = r - y
        if violation > 0:
            penalty += 5000 * violation ** 2
        violation = r - (1 - y)
        if violation > 0:
            penalty += 5000 * violation ** 2

    # Overlap penalties
    for i in range(N):
        for j in range(i + 1, N):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            overlap = radii[i] + radii[j] - dist
            if overlap > 0:
                penalty += 10000 * overlap ** 2

    obj = -np.sum(radii)
    return obj + penalty


def compute_gradient_numerical(vars, eps=1e-7):
    n_vars = len(vars)
    grad = np.zeros(n_vars)
    f0 = compute_objective_and_penalty(vars)
    for k in range(n_vars):
        vars[k] += eps
        fk = compute_objective_and_penalty(vars)
        grad[k] = (fk - f0) / eps
        vars[k] -= eps
    return grad


def make_hexagonal_init(r_init):
    centers = np.zeros((N, 2))
    radii = np.full(N, r_init)
    
    idx = 0
    y_offset = r_init * np.sqrt(3)
    
    for row in range(7):
        for col in range(6):
            if idx >= N:
                break
            if row % 2 == 0:
                x = col * 2 * r_init + r_init
            else:
                x = col * 2 * r_init + 2 * r_init
            
            y = row * y_offset + r_init
            
            if x + r_init <= 1.0 and y + r_init <= 1.0:
                centers[idx] = [x, y]
                idx += 1
        if idx >= N:
            break
    
    while idx < N:
        centers[idx] = [0.5, 0.5]
        idx += 1
    
    return centers, radii


def make_perturbed_init(centers, radii, seed):
    rng = np.random.RandomState(seed)
    new_centers = centers.copy()
    new_radii = radii.copy()
    
    for i in range(N):
        new_centers[i, 0] += rng.uniform(-0.02, 0.02)
        new_centers[i, 1] += rng.uniform(-0.02, 0.02)
        new_centers[i, 0] = np.clip(new_centers[i, 0], 0.05, 0.95)
        new_centers[i, 1] = np.clip(new_centers[i, 1], 0.05, 0.95)
        new_radii[i] = radii[i] + rng.uniform(-0.005, 0.005)
        new_radii[i] = max(new_radii[i], 0.01)
    
    return new_centers, new_radii


def optimize_from_start(centers, radii, maxiter=15000):
    x0 = np.concatenate([centers.flatten(), radii])
    
    bounds = []
    for i in range(2 * N):
        bounds.append((0.0, 1.0))
    for i in range(N):
        bounds.append((0.0, 0.5))
    
    result = minimize(
        compute_objective_and_penalty,
        x0,
        method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10, 'disp': False}
    )
    
    centers_opt = result.x[:2 * N].reshape(N, 2)
    radii_opt = result.x[2 * N:]
    radii_opt = np.maximum(radii_opt, 0)
    
    return centers_opt, radii_opt


def shrink_to_valid(centers, radii, shrink_factor=1.0):
    centers = centers.copy()
    radii = radii.copy()
    
    for iteration in range(50):
        max_violation = 0
        
        for i in range(N):
            x, y = centers[i]
            r = radii[i]
            min_dist = min(x, 1 - x, y, 1 - y)
            if r > min_dist:
                shrink_needed = (r - min_dist) / r + 0.001
                max_violation = max(max_violation, shrink_needed)
        
        for i in range(N):
            for j in range(i + 1, N):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                sum_r = radii[i] + radii[j]
                if sum_r > dist:
                    shrink_needed = (sum_r - dist) / sum_r + 0.001
                    max_violation = max(max_violation, shrink_needed)
        
        if max_violation < 1e-10:
            break
        
        radii *= (1 - max_violation * 0.5)
    
    return centers, radii


def run_packing():
    best_centers = None
    best_radii = None
    best_sum = 0.0
    
    # Try multiple starting radii and perturbations
    starting_radii = [0.09, 0.092, 0.095, 0.1, 0.105]
    
    for r_init in starting_radii:
        centers, radii = make_hexagonal_init(r_init)
        
        # Try unperturbed
        c_opt, r_opt = optimize_from_start(centers, radii, maxiter=10000)
        s = np.sum(r_opt)
        if s > best_sum:
            best_sum = s
            best_centers = c_opt.copy()
            best_radii = r_opt.copy()
        
        # Try perturbed versions
        for seed in range(5):
            c_pert, r_pert = make_perturbed_init(centers, radii, seed)
            c_opt, r_opt = optimize_from_start(c_pert, r_pert, maxiter=10000)
            s = np.sum(r_opt)
            if s > best_sum:
                best_sum = s
                best_centers = c_opt.copy()
                best_radii = r_opt.copy()
    
    # Fine-tune the best result
    best_centers, best_radii = optimize_from_start(best_centers, best_radii, maxiter=30000)
    
    # Ensure validity by slight shrink if needed
    best_centers, best_radii = shrink_to_valid(best_centers, best_radii)
    
    final_sum = float(np.sum(best_radii))
    
    return best_centers, best_radii, final_sum
