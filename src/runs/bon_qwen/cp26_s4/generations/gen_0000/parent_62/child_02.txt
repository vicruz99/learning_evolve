import numpy as np
from scipy.optimize import linprog, minimize
import itertools

def solve_radii(centers):
    """
    Solves the LP to find the maximum sum of radii for fixed centers.
    Returns (sum_radii, radii_vector).
    """
    n = centers.shape[0]
    # Distance to walls
    x, y = centers[:, 0], centers[:, 1]
    dist_to_walls = np.minimum(np.minimum(x, 1 - x), np.minimum(y, 1 - y))
    
    # Pairwise distances
    dist_matrix = np.sqrt(np.sum((centers[:, np.newaxis, :] - centers[np.newaxis, :, :]) ** 2, axis=2))
    
    # LP Setup: Maximize sum(r_i) -> Minimize -sum(r_i)
    # Variables: r_0, ..., r_n-1
    # Constraints: r_i + r_j <= d_ij for all i < j
    # Bounds: 0 <= r_i <= dist_to_walls[i]
    
    c_obj = -np.ones(n)
    
    # Create constraint matrix for pairwise distances
    # We need to form A_ub * r <= b_ub
    # There are n*(n-1)/2 constraints
    pairs = list(itertools.combinations(range(n), 2))
    num_constraints = len(pairs)
    A_ub = np.zeros((num_constraints, n))
    b_ub = np.zeros(num_constraints)
    
    for k, (i, j) in enumerate(pairs):
        A_ub[k, i] = 1
        A_ub[k, j] = 1
        b_ub[k] = dist_matrix[i, j]
        
    # Bounds for variables
    bounds = [(0, d) for d in dist_to_walls]
    
    # Solve LP
    # Use 'highs' method if available, otherwise 'simplex' or 'interior-point'
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    except ValueError:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='interior-point')
    
    if res.success:
        radii = res.x
        return -res.fun, radii
    else:
        # Fallback if LP fails, return small radii
        return 0.0, np.zeros(n)

def objective_func(centers_flat):
    """
    Objective function for optimization. 
    Centers are flattened to 1D array.
    Returns negative sum of radii (to minimize).
    """
    n = 26
    centers = centers_flat.reshape(n, 2)
    
    # Project centers to [0, 1] to ensure validity
    centers = np.clip(centers, 0, 1)
    
    # Calculate sum of radii
    try:
        s_r, _ = solve_radii(centers)
        return -s_r
    except Exception:
        return 1e6

def generate_hexagonal_centers(n, spacing=0.15):
    """Generate centers based on a hexagonal lattice."""
    centers = []
    y = spacing
    row = 0
    while len(centers) < n:
        x = spacing
        if row % 2 == 1:
            x += spacing / 2 # Offset for hexagonal
        while x <= 1 - spacing and len(centers) < n:
            centers.append([x, y])
            x += spacing
        y += spacing * np.sqrt(3) / 2
        row += 1
    return np.array(centers[:n])

def generate_grid_centers(n):
    """Generate centers on a grid."""
    cols = int(np.ceil(np.sqrt(n)))
    rows = int(np.ceil(n / cols))
    step_x = 1.0 / (cols + 1)
    step_y = 1.0 / (rows + 1)
    
    centers = []
    for r in range(rows):
        for c in range(cols):
            if len(centers) < n:
                centers.append([step_x * (c + 1), step_y * (r + 1)])
    return np.array(centers[:n])

def run_packing():
    n = 26
    best_sum = 0
    best_centers = None
    best_radii = None

    # 1. Generate multiple starting points
    init_configs = []
    
    # Hexagonal lattices with different spacings
    for sp in [0.12, 0.15, 0.18, 0.2]:
        try:
            init_configs.append(generate_hexagonal_centers(n, sp))
        except:
            pass
            
    # Grid initialization
    init_configs.append(generate_grid_centers(n))
    
    # Random initializations
    for _ in range(10):
        init_configs.append(np.random.rand(n, 2))

    # 2. Optimize each configuration
    for centers in init_configs:
        centers = np.clip(centers, 0.01, 0.99) # Keep slightly away from walls
        
        # Try Nelder-Mead
        try:
            res = minimize(objective_func, centers.flatten(), method='Nelder-Mead', 
                           options={'maxiter': 2000, 'xatol': 1e-5, 'fatol': 1e-5})
            if res.success or (-res.fun > best_sum):
                s = -res.fun
                if s > best_sum:
                    c = res.x.reshape(n, 2)
                    _, r = solve_radii(c)
                    best_sum = s
                    best_centers = c
                    best_radii = r
        except Exception:
            pass

        # Try L-BFGS-B with bounds
        try:
            bounds_opt = [(0.01, 0.99)] * (n * 2)
            res2 = minimize(objective_func, centers.flatten(), method='L-BFGS-B', bounds=bounds_opt, 
                            options={'maxiter': 500})
            if -res2.fun > best_sum:
                s = -res2.fun
                c = res2.x.reshape(n, 2)
                _, r = solve_radii(c)
                best_sum = s
                best_centers = c
                best_radii = r
        except Exception:
            pass

    # 3. Final validation and return
    if best_centers is not None:
        # Re-solve LP on final centers to ensure radii match exactly
        s, r = solve_radii(best_centers)
        return best_centers, r, s
    else:
        # Fallback
        centers = np.random.rand(n, 2)
        _, r = solve_radii(centers)
        return centers, r, np.sum(r)
