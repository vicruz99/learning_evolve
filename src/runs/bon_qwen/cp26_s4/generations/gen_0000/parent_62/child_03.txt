import numpy as np
from scipy.optimize import minimize

def compute_loss(params, N=26, penalty=1e7):
    """Computes the objective value: negative sum of radii plus constraint penalties."""
    centers = params[:2*N].reshape(N, 2)
    radii = params[2*N:]
    radii = np.maximum(radii, 1e-6)
    
    violation = 0.0
    
    # Boundary violations
    for i in range(N):
        x, y = centers[i]
        r = radii[i]
        if x < r: violation += (x - r)**2
        if y < r: violation += (y - r)**2
        if x + r > 1: violation += (x + r - 1)**2
        if y + r > 1: violation += (y + r - 1)**2
        
    # Overlap violations
    for i in range(N):
        for j in range(i+1, N):
            dist = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
            req = radii[i] + radii[j]
            if dist < req:
                violation += (dist - req)**2
                
    return -np.sum(radii) + penalty * violation

def expand_radii(centers, radii):
    """Finds maximum uniform scaling factor to expand radii while maintaining validity."""
    N = len(radii)
    factor = 1.0
    
    # Check boundaries
    for i in range(N):
        x, y = centers[i]
        r = radii[i]
        if r > 1e-9:
            factor = min(factor, x/r, (1-x)/r, y/r, (1-y)/r)
            
    # Check overlaps
    for i in range(N):
        for j in range(i+1, N):
            dist = np.hypot(centers[i,0]-centers[j,0], centers[i,1]-centers[j,1])
            r_sum = radii[i] + radii[j]
            if r_sum > 1e-9:
                factor = min(factor, dist / r_sum)
                
    # Apply factor with a small safety margin to avoid floating-point constraint violations
    return radii * min(factor * 0.999, 1.0)

def run_packing():
    N = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None
    
    # Generate diverse initial configurations
    inits = []
    for seed in range(8):
        np.random.seed(seed)
        centers = np.random.rand(N, 2)
        radii = np.random.uniform(0.05, 0.12, N)
        inits.append(np.concatenate([centers.ravel(), radii]))
        
    # Hexagonal lattice initialization
    centers_hex = np.zeros((N, 2))
    radii_hex = np.full(N, 0.09)
    idx = 0
    for row in range(5):
        n_in_row = 6 if row % 2 == 0 else 5
        if row == 4: n_in_row = 4  # Adjust last row to total 26
        for col in range(n_in_row):
            if idx >= N: break
            x = (col + 0.5) * (1.0 / n_in_row) + (0.5 * (row % 2)) * (1.0 / n_in_row)
            y = (row + 0.5) * (1.0 / 5)
            centers_hex[idx] = [x, y]
            idx += 1
    inits.append(np.concatenate([centers_hex.ravel(), radii_hex]))
    
    bounds = [(0, 1)] * 2*N + [(1e-4, 0.5)] * N
    
    for params0 in inits:
        try:
            res = minimize(compute_loss, params0, args=(N, 1e7), method='L-BFGS-B', 
                          bounds=bounds, options={'maxiter': 5000, 'ftol': 1e-12})
            
            c = res.x[:2*N].reshape(N, 2)
            r = res.x[2*N:]
            
            # Recompute violation strictly to check validity
            viol = 0.0
            for i in range(N):
                x, y = c[i]
                rr = r[i]
                if x < rr: viol += (x - rr)**2
                if y < rr: viol += (y - rr)**2
                if x + rr > 1: viol += (x + rr - 1)**2
                if y + rr > 1: viol += (y + rr - 1)**2
            for i in range(N):
                for j in range(i+1, N):
                    dist = np.hypot(c[i,0]-c[j,0], c[i,1]-c[j,1])
                    req = r[i] + r[j]
                    if dist < req: viol += (dist - req)**2
                    
            if viol < 1e-6:
                r_exp = expand_radii(c, r)
                s = np.sum(r_exp)
                if s > best_sum:
                    best_sum = s
                    best_centers = c.copy()
                    best_radii = r_exp.copy()
        except Exception:
            continue
            
    # Fallback initialization if optimization somehow fails (highly unlikely)
    if best_centers is None:
        best_centers = np.tile([0.5, 0.5], (N, 1))
        best_radii = np.full(N, 0.03)
        best_sum = np.sum(best_radii)
        
    return best_centers, best_radii, best_sum
