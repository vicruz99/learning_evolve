import numpy as np
from scipy.optimize import minimize

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    n = centers.shape[0]
    
    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False
    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False
        
    # Check if radii are nonnegative
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
            
    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # --- 1. Initialization: Hexagonal Grid Pattern ---
    # We create a hexagonal arrangement to start with a dense packing structure.
    # This is generally better than a square grid for packing efficiency.
    
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Parameters for hex grid
    # We try to fit circles with an initial radius that fits, then optimize.
    # A rough estimate: 5x5 grid fits radius 0.1. Hexagonal might allow slightly larger or same.
    # Let's start with r=0.05 to be safe and let optimizer grow them.
    r_init = 0.05
    spacing_x = 2 * r_init
    spacing_y = spacing_x * np.sqrt(3) / 2
    
    idx = 0
    # Generate points in a hexagonal pattern
    # We iterate rows and columns
    row = 0
    col = 0
    
    # We need to cover the square [0,1]x[0,1]
    # Let's define a range for rows and cols to ensure we generate enough points
    max_rows = 10
    max_cols = 10
    
    # Center the grid roughly in the square
    offset_x = 0.05
    offset_y = 0.05
    
    while idx < n:
        x = offset_x + col * spacing_x
        y = offset_y + row * spacing_y
        
        # Shift odd rows to create hexagonal lattice
        if row % 2 == 1:
            x += spacing_x / 2
            
        if 0 <= x <= 1 and 0 <= y <= 1:
            centers[idx] = [x, y]
            radii[idx] = r_init
            idx += 1
        
        col += 1
        if col > max_cols or x > 1: # Move to next row if out of bounds horizontally
            col = 0
            row += 1
            if row > max_rows:
                break
    
    # If we didn't fill 26 (unlikely with these params), fill with random valid points
    if idx < n:
        for k in range(idx, n):
            # Find a random spot far from others

                cx, cy = np.random.uniform(0.1, 0.9, 2)
                min_dist = 1.0
                for j in range(k):
                    d = np.sqrt((centers[j,0]-cx)**2 + (centers[j,1]-cy)**2)
                    if d < min_dist:
                        min_dist = d
                if min_dist > 2 * r_init: # Ensure some separation
                    centers[k] = [cx, cy]
                    radii[k] = r_init
                    break
    
    # --- 2. Optimization Setup ---
    # Variables: [x1, y1, r1, x2, y2, r2, ..., x26, y26, r26]
    x0 = np.zeros(3 * n)
    for i in range(n):
        x0[3*i] = centers[i, 0]
        x0[3*i+1] = centers[i, 1]
        x0[3*i+2] = radii[i]
        
    # Bounds: x, y in [0, 1], r in [0, 0.5]
    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n
    
    # --- 3. Constraints ---
    # We need to ensure:
    # 1. Circle inside square: x-r >= 0, x+r <= 1, y-r >= 0, y+r <= 1
    # 2. No overlap: dist_ij >= ri + rj  =>  dist_ij^2 >= (ri + rj)^2
    
    cons = []
    
    # Helper to access variables
    def get_x(v, i): return v[3*i]
    def get_y(v, i): return v[3*i+1]
    def get_r(v, i): return v[3*i+2]
    
    # Boundary Constraints
    for i in range(n):
        # x - r >= 0
        cons.append({'type': 'ineq', 'fun': lambda v, i=i: get_x(v, i) - get_r(v, i)})
        # 1 - x - r >= 0
        cons.append({'type': 'ineq', 'fun': lambda v, i=i: 1.0 - get_x(v, i) - get_r(v, i)})
        # y - r >= 0
        cons.append({'type': 'ineq', 'fun': lambda v, i=i: get_y(v, i) - get_r(v, i)})
        # 1 - y - r >= 0
        cons.append({'type': 'ineq', 'fun': lambda v, i=i: 1.0 - get_y(v, i) - get_r(v, i)})

        for j in range(i + 1, n):
            cons.append({
                'type': 'ineq',
                'fun': lambda v, i=i, j=j: 
                    (get_x(v, i) - get_x(v, j))**2 + 
                    (get_y(v, i) - get_y(v, j))**2 - 
                    (get_r(v, i) + get_r(v, j))**2
            })
            
    # --- 4. Objective Function ---
    # Maximize sum of radii => Minimize negative sum of radii
    def objective(v):
        return -sum(get_r(v, i) for i in range(n))

    # Using SLSQP (Sequential Least Squares Programming)
    # It handles non-linear constraints well.
    res = minimize(
        objective, 
        x0, 
        method='SLSQP', 
        bounds=bounds, 
        constraints=cons,

    )

    final_radii = np.array([res.x[3*i+2] for i in range(n)])
    
    # Ensure strict non-negativity and small adjustments for numerical stability
    final_radii = np.maximum(final_radii, 0.0)

    for i in range(n):
        r = final_radii[i]
        final_centers[i, 0] = np.clip(final_centers[i, 0], r, 1 - r)
        final_centers[i, 1] = np.clip(final_centers[i, 1], r, 1 - r)

    sum_radii = np.sum(final_radii)
    
    # Final Validation check
    if not validate_packing(final_centers, final_radii):
        # If still invalid, return a safe fallback (e.g. small circles)

        sum_radii = np.sum(final_radii)
