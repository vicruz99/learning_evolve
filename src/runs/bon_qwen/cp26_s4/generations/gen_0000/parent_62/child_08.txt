import numpy as np
from scipy.optimize import minimize

def run_packing():
    """
    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    """
    n_circles = 26
    
    # 1. Initial Configuration: Perturbed 5x5 grid + 1 extra circle
    # A standard 5x5 grid has radius 0.1. We start slightly smaller to allow optimization room.
    # We place 25 circles in a grid and 1 circle in a gap.
    initial_r = 0.095
    
    centers = []
    # 5x5 grid centers
    grid_points = [0.1, 0.3, 0.5, 0.7, 0.9]
    for x in grid_points:
        for y in grid_points:
            centers.append([x, y])
    
    # 26th circle placed in a gap (e.g., between centers (0.3, 0.3) and (0.5, 0.5))
    # Gap center approx (0.4, 0.4)
    centers.append([0.4, 0.4])
    
    initial_centers = np.array(centers)
    initial_radii = np.full(n_circles, initial_r)
    
    # Flatten variables: [x0, y0, r0, x1, y1, r1, ...]
    x0 = np.concatenate([initial_centers, initial_radii.reshape(-1, 1)], axis=1).flatten()
    
    # 2. Define Objective Function (Maximize sum of radii -> Minimize negative sum)
    def objective(vars):
        radii = vars[2::3]
        return -np.sum(radii)
    
    # 3. Define Constraints
    def constraint_boundary(vars):
        """
        Ensures circles are inside [0,1]x[0,1].
        Constraints: r <= x <= 1-r  =>  x - r >= 0  AND  x + r <= 1
                      r <= y <= 1-r  =>  y - r >= 0  AND  y + r <= 1
        """
        constraints = []
        for i in range(n_circles):
            idx = i * 3
            x = vars[idx]
            y = vars[idx+1]
            r = vars[idx+2]
            
            # x - r >= 0
            constraints.append(x - r)
            # 1 - (x + r) >= 0
            constraints.append(1 - (x + r))
            # y - r >= 0
            constraints.append(y - r)
            # 1 - (y + r) >= 0
            constraints.append(1 - (y + r))
        return np.array(constraints)

    def constraint_non_overlap(vars):
        """
        Ensures circles do not overlap.
        Distance between centers >= sum of radii.
        dist^2 >= (r1 + r2)^2  =>  dist^2 - (r1 + r2)^2 >= 0
        """
        constraints = []
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                idx_i = i * 3
                idx_j = j * 3
                
                xi, yi, ri = vars[idx_i], vars[idx_i+1], vars[idx_i+2]
                xj, yj, rj = vars[idx_j], vars[idx_j+1], vars[idx_j+2]
                
                dist_sq = (xi - xj)**2 + (yi - yj)**2
                r_sum = ri + rj
                
                constraints.append(dist_sq - r_sum**2)
        return np.array(constraints)
    
    # Combine constraints for scipy.optimize
    # We can pass them as a list of dictionaries or a single function returning array
    # SLSQP supports nonlinear constraints defined as functions returning arrays >= 0
    
    cons_boundary = {'type': 'ineq', 'fun': constraint_boundary}
    cons_overlap = {'type': 'ineq', 'fun': constraint_non_overlap}
    
    # 4. Run Optimization
    # Bounds: x, y in [0, 1], r in [0, 0.5]
    bounds = []
    for _ in range(n_circles):
        bounds.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
        
    result = minimize(
        objective, 
        x0, 
        method='SLSQP', 
        bounds=bounds, 
        constraints=[cons_boundary, cons_overlap],
        options={'maxiter': 1000, 'ftol': 1e-12}
    )
    
    # 5. Extract Results
    final_vars = result.x
    final_centers = np.array([[final_vars[i*3], final_vars[i*3+1]] for i in range(n_circles)])
    final_radii = np.array([final_vars[i*3+2] for i in range(n_circles)])
    
    # Small correction to ensure strict validity if optimization is borderline
    # Shrink radii slightly if needed to fix numerical noise
    # Check overlaps one last time
    for i in range(n_circles):
        for j in range(i + 1, n_circles):
            dist = np.sqrt(np.sum((final_centers[i] - final_centers[j])**2))
            r_sum = final_radii[i] + final_radii[j]
            if dist < r_sum - 1e-12:
                # Violation detected, scale down radii slightly
                scale = (dist + 1e-12) / r_sum
                final_radii[i] *= scale
                final_radii[j] *= scale

    sum_radii = np.sum(final_radii)
    
    return final_centers, final_radii, sum_radii
