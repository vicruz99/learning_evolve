import numpy as np
from scipy.optimize import minimize
import math


def make_initial_config(n=26):
    """Create hexagonal initial configuration with 5 rows."""
    centers = []
    radii = []
    idx = 0
    row_counts = [6, 5, 6, 5, 4]
    for row in range(5):
        ncols = row_counts[row]
        for col in range(ncols):
            if idx >= n:
                break
            x = 0.06 + col * 0.16
            if row % 2 == 1:
                x += 0.08
            y = 0.06 + row * 0.22
            centers.append([x, y])
            radii.append(0.075)
            idx += 1
    return np.array(centers), np.array(radii)


def penalized_objective(params, n, pw):
    """Objective: maximize sum of radii with penalty for constraint violations."""
    centers = params[:2 * n].reshape(n, 2)
    radii = np.maximum(params[2 * n:], 1e-10)

    obj = -np.sum(radii)
    penalty = 0.0

    x = centers[:, 0]
    y = centers[:, 1]

    # Boundary penalties (vectorized)
    below_x = np.maximum(0.0, radii - x)
    above_x = np.maximum(0.0, x + radii - 1.0)
    penalty += pw * (np.sum(below_x ** 2) + np.sum(above_x ** 2))

    below_y = np.maximum(0.0, radii - y)
    above_y = np.maximum(0.0, y + radii - 1.0)
    penalty += pw * (np.sum(below_y ** 2) + np.sum(above_y ** 2))

    # Non-overlap penalties
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = math.sqrt(dx * dx + dy * dy)
            min_dist = radii[i] + radii[j]
            if dist < min_dist:
                overlap = min_dist - dist
                penalty += pw * overlap * overlap

    return obj + penalty


def validate_solution(centers, radii, n):
    """Check if solution is valid."""
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-9 or x + r > 1 + 1e-9:
            return False
        if y - r < -1e-9 or y + r > 1 + 1e-9:
            return False
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = math.sqrt(dx * dx + dy * dy)
            if dist < radii[i] + radii[j] - 1e-9:
                return False
    return True


def run_packing():
    n = 26

    centers_init, radii_init = make_initial_config(n)
    params = np.concatenate([centers_init.flatten(), radii_init])

    bounds = [(0.001, 0.999)] * (2 * n) + [(0.001, 0.3)] * n

    best_sum = 0.0
    best_centers = None
    best_radii = None

    # Try multiple random seeds / perturbations
    for trial in range(40):
        np.random.seed(trial * 37 + 5)

        # Create perturbed starting point
        if trial == 0:
            c_start = centers_init.copy()
            r_start = radii_init.copy()
        else:
            c_start = best_centers + np.random.randn(n, 2) * 0.008
            r_start = best_radii + np.random.randn(n) * 0.004

        r_start = np.maximum(r_start, 0.02)
        c_start = np.clip(c_start, 0.02, 0.98)

        current_params = np.concatenate([c_start.flatten(), r_start])

        # Progressive penalty weight annealing
        for pw in [200, 1000, 5000, 20000, 80000, 200000]:
            result = minimize(
                penalized_objective,
                current_params,
                args=(n, pw),
                method="L-BFGS-B",
                bounds=bounds,
                options={"maxiter": 20000, "ftol": 1e-15, "gtol": 1e-12},
            )
            current_params = result.x

        c = current_params[:2 * n].reshape(n, 2)
        r = np.maximum(current_params[2 * n:], 0)

        # Validate
        if validate_solution(c, r, n):
            s = np.sum(r)
            if s > best_sum:
                best_sum = s
                best_centers = c.copy()
                best_radii = r.copy()

    # Fallback
    if best_centers is None:
        best_centers, best_radii = make_initial_config(n)
        best_sum = np.sum(best_radii)

    return best_centers, best_radii, float(best_sum)
