def get_dist_matrix(centers):
    """Compute pairwise distance matrix."""

def compute_overlap_energy(centers, radii):
    """
    Computes a penalty energy for overlaps and boundary violations.
    Lower is better. 0 means valid (approximately).
    """
    n = len(radii)
    energy = 0.0
    
    # Boundary violations
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        # Left
        if x < r:
            energy += (r - x)**2
        # Right
        if x > 1 - r:
            energy += (x - (1 - r))**2
        # Bottom
        if y < r:
            energy += (r - y)**2
        # Top
        if y > 1 - r:
            energy += (y - (1 - r))**2

    # Pairwise overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt((centers[i,0] - centers[j,0])**2 + (centers[i,1] - centers[j,1])**2)
            sum_r = radii[i] + radii[j]
            if dist < sum_r:
                energy += (sum_r - dist)**2
                
    return energy
