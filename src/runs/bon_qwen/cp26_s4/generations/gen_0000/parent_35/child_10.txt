def run_packing():

    def simulate(centers, radii, n, steps=100000):

        centers = centers.copy()
        radii = radii.copy()
        
        dt = 0.003
        growth = 0.000015
        cool = 0.99996

            radii += growth

            # Pairwise repulsion for overlapping circles
            for i in range(n):
                for j in range(i + 1, n):

                        scale = (need - d) / max(d, 1e-10)

                        fy = scale * dy

                        f[i, 1] -= fy
                        f[j, 0] += fx
                        f[j, 1] += fy
