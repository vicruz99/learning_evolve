                # If overlap (dist < sum_r)
                if dist < sum_r - 1e-9:
                    if sum_r > 1e-12:
                        # Scale radii down to fit exactly
                        scale = dist / sum_r
                        radii_opt[i] *= np.sqrt(scale)
                        radii_opt[j] *= np.sqrt(scale)
                    else:
                        radii_opt[i] = 0.0
                        radii_opt[j] = 0.0
                    changed = True
        
        # Resolve boundary violations
        for i in range(n):
            x, y = centers_opt[i]
            r = radii_opt[i]
            min_dist_bound = min(x, 1.0 - x, y, 1.0 - y)
            if r > min_dist_bound - 1e-9:
                radii_opt[i] = max(0.0, min_dist_bound)
                changed = True
        
        if not changed:
            break

    sum_radii = float(np.sum(radii_opt))
    
    return centers_opt, radii_opt, sum_radii
