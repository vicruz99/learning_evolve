        ux = vars[n:2*n]
        uy = vars[2*n:3*n]

        # This keeps x in [r, 1-r] if r <= 0.5 and ux in [0,1]

        # but bounds on r will prevent this.
        
        # Compute centers
        # To avoid numerical issues with 1-2r being negative if r>0.5, 
        # we clamp r effectively by bounds, but let's assume valid r.
        
        # Optimization variables are bounded, so r should be in [0, 0.5]
        # ux, uy in [0, 1]
        
        x = r + (1.0 - 2.0 * r) * ux
        y = r + (1.0 - 2.0 * r) * uy
        
        centers = np.column_stack((x, y))
        
        # Objective: Maximize sum of radii -> Minimize -sum(r)
        obj = -np.sum(r)
        
        # Penalty for overlaps
        # dist(i, j) >= r_i + r_j

        penalty = 0.0

            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                overlap = r[i] + r[j] - dist
                if overlap > 0:
                    penalty += overlap**2
        
        # We can also add a small penalty for r being too close to 0 to avoid trivial solution
        # But constraints usually handle it if we start with reasonable radii.
        
        return obj + lambda_pen * penalty

    # Bounds for variables

        bounds.append((0.0, 0.5)) # r_i
