import scipy.optimize as opt

def run_packing():

    Returns:
        tuple: (centers, radii, sum_radii)
    """

    # We try a layout with rows having different numbers of circles.

    radii_init = []

    dx = 2 * r_start
    dy = np.sqrt(3) * r_start
    
    row_counts = [6, 5, 6, 5, 4] # Sum = 26
    
    y = 0.0
    count = 0

        x = 0.0
        for j in range(n_circles):
            # Offset every other row
            if i % 2 == 1:
                x_offset = dx / 2
            else:
                x_offset = 0
            
            cx = x + x_offset + r_start
            cy = y + r_start
            
            centers_init.append([cx, cy])
            radii_init.append(r_start)
            count += 1
            x += dx
        y += dy

    radii_init = np.array(radii_init[:n])

    # The initial generation might exceed bounds, so we'll rely on the optimizer

    centers_init = np.clip(centers_init, [r_start, r_start], [1-r_start, 1-r_start])

    # Initial guess for optimizer: [x1, y1, r1, x2, y2, r2, ...]

    # 2. 1 - x_i - r_i >= 0
    # 3. y_i - r_i >= 0
    # 4. 1 - y_i - r_i >= 0
    
    def bound_constraint_x_lower(vars, i):
        x = vars[3*i]
        r = vars[3*i+2]
        return x - r

        x = vars[3*i]
        r = vars[3*i+2]
        return 1 - x - r
        
    def bound_constraint_y_lower(vars, i):
        y = vars[3*i+1]
        r = vars[3*i+2]
        return y - r
        
    def bound_constraint_y_upper(vars, i):
        y = vars[3*i+1]
        r = vars[3*i+2]
        return 1 - y - r

            dy = yi - yj
            d = np.sqrt(dx*dx + dy*dy)
            return d - ri - rj
        return dist_con

        else:

    # Ensure no NaNs or negatives

    return centers_opt, radii_opt, sum_radii
