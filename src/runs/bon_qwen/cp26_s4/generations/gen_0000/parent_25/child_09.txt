import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Pack 26 circles in a unit square [0,1]x[0,1] to maximize sum of radii.
    Uses numerical optimization (SLSQP) starting from a grid initialization.
    """
    n = 26
    
    def objective(vars):
        # vars: [x1, y1, r1, x2, y2, r2, ...]
        # We want to maximize sum(r), so minimize -sum(r)
        radii = vars[2::3]
        return -np.sum(radii)

    def cons_func(vars):
        # Returns array of constraint values >= 0
        x = vars[0::3]
        y = vars[1::3]
        r = vars[2::3]
        
        # Boundary constraints
        # x >= r  => x - r >= 0
        c1 = x - r
        # x + r <= 1 => 1 - x - r >= 0
        c2 = 1.0 - x - r
        # y >= r  => y - r >= 0
        c3 = y - r
        # y + r <= 1 => 1 - y - r >= 0
        c4 = 1.0 - y - r
        
        # Overlap constraints
        # dist^2 >= (r_i + r_j)^2
        c_overlaps = []
        for i in range(n):
            for j in range(i + 1, n):
                dx = x[i] - x[j]
                dy = y[i] - y[j]
                dr = r[i] + r[j]
                # (dx^2 + dy^2) - dr^2 >= 0
                c_overlaps.append(dx*dx + dy*dy - dr*dr)
        
        return np.concatenate([c1, c2, c3, c4, c_overlaps])

    def get_initial_guess(seed):

        # Initialize with a dense grid pattern

                    break

                break
