import numpy as np
from scipy.optimize import linprog, minimize
from itertools import combinations
import math


def compute_optimal_radii(centers):
    """Given centers, compute optimal radii using LP to maximize sum."""
    n = centers.shape[0]
    c_obj = -np.ones(n)
    
    A_ub = []
    b_ub = []
    
    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        x, y = centers[i]
        for val in [x, 1 - x, y, 1 - y]:
            row = np.zeros(n)
            row[i] = 1
            A_ub.append(row)
            b_ub.append(max(0, val))
    
    # Non-overlap constraints: r_i + r_j <= dist(i,j)
    for i, j in combinations(range(n), 2):
        dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
        row = np.zeros(n)
        row[i] = 1
        row[j] = 1
        A_ub.append(row)
        b_ub.append(dist)
    
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0, None) for _ in range(n)]
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        return -result.fun, result.x
    else:
        return 0.0, np.zeros(n)


def objective_func(x, n=26):
    """Objective: negative sum of radii for given centers."""
    centers = x.reshape((n, 2))
    sum_r, _ = compute_optimal_radii(centers)
    return -sum_r


def init_hexagonal(n, r_est=0.095):
    """Initialize centers in hexagonal grid pattern."""
    centers = np.zeros((n, 2))
    row_spacing = r_est * np.sqrt(3)
    col_spacing = 2 * r_est
    
    row = 0
    col = 0
    idx = 0
    while idx < n:
        x = col * col_spacing + r_est
        y = row * row_spacing + r_est
        if col % 2 == 1:
            y += row_spacing * 0.5
        
        if x + r_est <= 1.0 and y + r_est <= 1.0:
            centers[idx] = [x, y]
            idx += 1
        
        col += 1
        if col * col_spacing + 2 * r_est > 1.0:
            col = 0
            row += 1
    
    return centers


def init_grid_random(n, seed=0):
    """Initialize with perturbed grid."""
    rng = np.random.RandomState(seed)
    centers = np.zeros((n, 2))
    
    # Arrange in roughly sqrt(n) x sqrt(n) grid
    cols = int(np.ceil(np.sqrt(n)))
    rows = int(np.ceil(n / cols))
    
    x_spacing = 1.0 / (cols + 1)
    y_spacing = 1.0 / (rows + 1)
    
    idx = 0
    for r in range(rows):
        for c in range(cols):
            if idx >= n:
                break
            x = (c + 1) * x_spacing + rng.uniform(-x_spacing * 0.2, x_spacing * 0.2)
            y = (r + 1) * y_spacing + rng.uniform(-y_spacing * 0.2, y_spacing * 0.2)
            x = np.clip(x, 0.05, 0.95)
            y = np.clip(y, 0.05, 0.95)
            centers[idx] = [x, y]
            idx += 1
        if idx >= n:
            break
    
    return centers


def init_hexagonal_dense(n):
    """Try different hexagonal densities."""
    best_centers = None
    best_sum = 0
    
    for r_try in [0.08, 0.085, 0.09, 0.095, 0.10, 0.105]:
        centers = init_hexagonal(n, r_try)
        s, _ = compute_optimal_radii(centers)
        if s > best_sum:
            best_sum = s
            best_centers = centers.copy()
    
    return best_centers
