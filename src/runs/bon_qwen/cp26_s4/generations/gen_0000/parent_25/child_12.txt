import numpy as np
from scipy.optimize import minimize, linear_sum_assignment

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    # Check if radii are nonnegative and not nan
    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:  # Allow for tiny numerical errors
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True

def compute_optimal_radii(centers):
    """
    Given fixed centers, solve an LP to maximize sum of radii.
    Variables: r_i >= 0
    Maximize: sum(r_i)
    Subject to:
        r_i <= min(x_i, 1-x_i, y_i, 1-y_i)  (Boundary constraints)
        r_i + r_j <= ||c_i - c_j||           (Distance constraints)
    """
    n = centers.shape[0]
    
    # 1. Compute boundary limits for each circle
    # r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    # So r_i <= min(x, 1-x, y, 1-y)
    bounds_x = np.minimum(centers[:, 0], 1 - centers[:, 0])
    bounds_y = np.minimum(centers[:, 1], 1 - centers[:, 1])
    boundary_limits = np.minimum(bounds_x, bounds_y)
    
    # 2. Compute pairwise distances
    # d_ij = ||c_i - c_j||
    dist_matrix = np.linalg.norm(centers[:, np.newaxis, :] - centers[np.newaxis, :, :], axis=2)
    
    # 3. Setup Linear Programming
    # Maximize sum(r_i) <=> Minimize -sum(r_i)
    c_obj = -np.ones(n)
    
    # Inequality constraints: A_ub @ r <= b_ub
    # We need to construct the matrix A_ub and vector b_ub
    
    # Rows for boundary constraints: 1*r_i <= limit_i
    A_boundary = np.eye(n)
    b_boundary = boundary_limits
    
    # Rows for distance constraints: r_i + r_j <= d_ij
    # We need to flatten the upper triangular part of the distance matrix
    constraints_rows = []
    constraints_vals = []
    
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            constraints_rows.append(row)
            constraints_vals.append(dist_matrix[i, j])
            
    if constraints_rows:
        A_dist = np.array(constraints_rows)
        b_dist = np.array(constraints_vals)
        
        A_ub = np.vstack([A_boundary, A_dist])
        b_ub = np.concatenate([b_boundary, b_dist])
    else:
        A_ub = A_boundary
        b_ub = b_boundary

    # Bounds for r_i >= 0
    bounds_r = [(0, None) for _ in range(n)]
    
    from scipy.optimize import linprog
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_r, method='highs')
    
    if res.success:
        return res.x
    else:
        # Fallback if LP fails (unlikely with valid inputs)
        # Return valid small radii
        return np.full(n, 1e-5)

def pack_circles_force_repulsion(n_circles, seed=42):
    """
    Use force-directed repulsion to spread centers apart (max-min distance).
    """
    np.random.seed(seed)
    
    # Initialize centers in a grid or random
    # A slightly perturbed grid is usually better
    k = int(np.ceil(np.sqrt(n_circles)))
    x = np.linspace(0.1, 0.9, k)
    y = np.linspace(0.1, 0.9, k)
    grid = []
    for xi in x:
        for yi in y:
            grid.append([xi, yi])
            if len(grid) >= n_circles:
                break
        if len(grid) >= n_circles:
            break
            
    centers = np.array(grid[:n_circles])
    
    # Add random noise to break symmetry
    centers += np.random.uniform(-0.02, 0.02, centers.shape)
    centers = np.clip(centers, 0.01, 0.99)
    
    # Optimization to maximize minimum distance
    # We minimize negative of the minimum distance
    # However, min function is non-smooth. 
    # We can use a smooth approximation or just optimize a potential energy function.
    # Potential energy: Sum of 1/distance^2 or similar repulsive forces.
    
    def energy_func(pos):
        pos = pos.reshape(n_circles, 2)
        # Keep within bounds (soft penalty)
        penalty = 0
        for i in range(n_circles):
            if pos[i,0] < 0.01: penalty += 100 * (0.01 - pos[i,0])**2
            if pos[i,0] > 0.99: penalty += 100 * (pos[i,0] - 0.99)**2
            if pos[i,1] < 0.01: penalty += 100 * (0.01 - pos[i,1])**2
            if pos[i,1] > 0.99: penalty += 100 * (pos[i,1] - 0.99)**2
            
        # Repulsion
        repulsion = 0
        for i in range(n_circles):
            for j in range(i+1, n_circles):
                d = np.sqrt(np.sum((pos[i] - pos[j])**2))
                if d < 0.001: d = 0.001 # Avoid singularity
                repulsion += 1.0 / (d**2)
                
        return penalty + repulsion

    res = minimize(energy_func, centers.flatten(), method='BFGS', options={'maxiter': 500})
    optimal_centers = res.x.reshape(n_circles, 2)
    
    return optimal_centers

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # Step 1: Find good centers by maximizing separation
    # We run this with a few seeds and pick the best configuration based on LP sum
    best_sum = 0
