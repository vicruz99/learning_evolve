import scipy.optimize as opt

    idx = 0

    h_spacing = init_r * np.sqrt(3)

                    centers[idx] = [x, y]
                    idx += 1
        else:

        radii_opt = np.zeros(n)

            centers_opt[i] = variables[i*2 : i*2+2]
            radii_opt[i] = variables[2*n + i]
            
        # Objective: Maximize sum of radii -> Minimize -sum(radii)
        obj_val = -np.sum(radii_opt)

        penalty = 0.0

        for i in range(n):

            # Right

            # Bottom

            # Top
            if y + r > 1: penalty += weight * (y + r - 1)**2

                    penalty += weight * overlap**2
