def compute_objective_and_penalty(vars, n_circles=26):
    """
    Computes the objective (negative sum of radii) and penalty for constraints.
    """
    # vars shape: (n_circles, 3) -> x, y, r
    centers = vars[:, :2]
    radii = vars[:, 2]
    
    obj = -np.sum(radii)
    penalty = 0.0
    penalty_factor = 1000.0
    
    # Boundary constraints
    # Circle i must be inside [0,1]x[0,1]
    # x - r >= 0  => r - x <= 0
    # x + r <= 1  => x + r - 1 <= 0
    # Same for y
    
    # Vectorized boundary penalty
    # Violation: max(0, r - x), max(0, x + r - 1), etc.
    left_viol = np.maximum(0, radii - centers[:, 0])
    right_viol = np.maximum(0, centers[:, 0] + radii - 1)
    bottom_viol = np.maximum(0, radii - centers[:, 1])
    top_viol = np.maximum(0, centers[:, 1] + radii - 1)
    
    boundary_violations = left_viol + right_viol + bottom_viol + top_viol
    penalty += penalty_factor * np.sum(boundary_violations**2)
    
    # Overlap constraints
    # dist(i, j) >= r_i + r_j
    # Violation: max(0, r_i + r_j - dist)
    
    # Calculate pairwise distances
    # Using broadcasting for efficiency
    # centers shape (N, 2)
    # diff shape (N, N, 2)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.linalg.norm(diff, axis=2)
    
    # radii shape (N,)
    # sum_radii_ij shape (N, N)
    sum_radii_ij = radii[:, np.newaxis] + radii[np.newaxis, :]
    
    # We only care about i < j
    # Triangular mask
    mask = np.triu(np.ones((n_circles, n_circles), dtype=bool), k=1)
    
    # Violation amount
    violation = sum_radii_ij - dists
    
    # Clip negative violations (no overlap) to 0
    violation = np.maximum(0, violation)
    
    # Sum squared violations
    overlap_violations = violation[mask]
    penalty += penalty_factor * np.sum(overlap_violations**2)
    
    # Negative radii penalty
    neg_r_viol = np.maximum(0, -radii)
    penalty += penalty_factor * np.sum(neg_r_viol**2)
    
    return obj + penalty
