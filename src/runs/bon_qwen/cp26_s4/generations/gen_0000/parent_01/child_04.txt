import numpy as np
from scipy.optimize import minimize


def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    n = 26

    best_sum = 0.0
    best_centers = None
    best_radii = None

    for restart in range(40):
        centers_init, radii_init = initialize_hex(n, restart)
        params = np.concatenate([centers_init.flatten(), radii_init])

        bounds = [(0.001, 0.999)] * (2 * n) + [(0.001, 0.4)] * n

        def objective(p):
            return -np.sum(p[2 * n:])

        def constraints(p):
            centers = p[: 2 * n].reshape(n, 2)
            radii = p[2 * n :]
            c = []
            for i in range(n):
                c.append(centers[i, 0] - radii[i] - 1e-7)
                c.append(1.0 - centers[i, 0] - radii[i] - 1e-7)
                c.append(centers[i, 1] - radii[i] - 1e-7)
                c.append(1.0 - centers[i, 1] - radii[i] - 1e-7)
            for i in range(n):
                for j in range(i + 1, n):
                    dx = centers[i, 0] - centers[j, 0]
                    dy = centers[i, 1] - centers[j, 1]
                    dist_sq = dx * dx + dy * dy
                    r_sum = radii[i] + radii[j]
                    c.append(dist_sq - r_sum * r_sum - 1e-10)
            return np.array(c)

        cons = {"type": "ineq", "fun": constraints}

        result = minimize(
            objective,
            params,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": 8000, "ftol": 1e-15, "disp": False},
        )

        current_sum = -result.fun
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = result.x[: 2 * n].reshape(n, 2).copy()
            best_radii = result.x[2 * n :].copy()

    if best_centers is None:
        centers = np.random.rand(n, 2) * 0.8 + 0.1
        radii = np.ones(n) * 0.01
        return centers, radii, np.sum(radii)

    # Post-process: ensure strict validity
    for i in range(n):
        r = best_radii[i]
        best_centers[i, 0] = np.clip(best_centers[i, 0], r + 1e-6, 1.0 - r - 1e-6)
        best_centers[i, 1] = np.clip(best_centers[i, 1], r + 1e-6, 1.0 - r - 1e-6)

    return best_centers, best_radii, float(best_sum)


def initialize_hex(n, seed):
    np.random.seed(seed)
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.06

    # Hexagonal arrangement: rows of 6,5,6,5,4 = 26 circles
    rows = [6, 5, 6, 5, 4]
    idx = 0
    for row_i, count in enumerate(rows):
        for col_j in range(count):
            x = 0.06 + (col_j + (row_i % 2) * 0.5) * 0.155
            y = 0.06 + row_i * 0.21
            centers[idx] = [x, y]
            idx += 1

    # Add small random perturbation
    centers += np.random.randn(n, 2) * 0.002
    centers = np.clip(centers, 0.02, 0.98)

    return centers, radii
