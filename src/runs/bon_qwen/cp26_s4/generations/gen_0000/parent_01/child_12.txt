import numpy as np
from scipy.optimize import minimize, linprog

def get_optimal_radii(centers):
    """
    Given a set of centers, solve the LP to find radii that maximize sum(r_i)
    subject to non-overlap and boundary constraints.
    """
    n = centers.shape[0]
    if n == 0:
        return 0.0, np.array([])
    
    # Objective: maximize sum(r) => minimize -sum(r)
    c_obj = np.ones(n) * -1
    
    # Constraints:
    # 1. r_i + r_j <= dist(i, j) for all pairs i < j
    # 2. 0 <= r_i <= dist_to_wall(i)
    
    x = centers[:, 0]
    y = centers[:, 1]
    
    # Calculate distance to walls for each circle
    # r_i <= min(x_i, 1-x_i, y_i, 1-y_i)
    wall_dists = np.minimum(np.minimum(x, 1 - x), np.minimum(y, 1 - y))
    wall_dists = np.maximum(wall_dists, 0)
    
    # Bounds for each variable r_i: [0, wall_dist_i]
    bounds = [(0, wd) for wd in wall_dists]
    
    # Pairwise constraints: r_i + r_j <= dist(i, j)
    # This forms a matrix A_ub @ r <= b_ub
    num_constraints = n * (n - 1) // 2
    A_ub = np.zeros((num_constraints, n))
    b_ub = np.zeros(num_constraints)
    
    row_idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt((x[i] - x[j])**2 + (y[i] - y[j])**2)
            A_ub[row_idx, i] = 1.0
            A_ub[row_idx, j] = 1.0
            b_ub[row_idx] = dist
            row_idx += 1
            
    try:
        # Solve LP
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            # res.x contains the radii
            # The objective value res.fun is -sum(r), so sum(r) = -res.fun
            # However, sum(res.x) is more direct and robust
            return sum(res.x), res.x
        else:
            # If LP fails, return 0 radii
            return 0.0, np.zeros(n)
    except Exception:
        return 0.0, np.zeros(n)

    s, _ = get_optimal_radii(centers)
    return -s  # We want to maximize s, so minimize -s
