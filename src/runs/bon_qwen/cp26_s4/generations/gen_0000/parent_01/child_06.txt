import numpy as np
from scipy.optimize import minimize


def hexagonal_init(n):
    """Initialize circles in a hexagonal lattice pattern."""
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.04
    
    spacing = 0.13
    row_height = spacing * np.sqrt(3) / 2
    offset = spacing / 2
    
    idx = 0
    row = 0
    while idx < n:
        col = 0
        while idx < n:
            x = 0.1 + col * spacing + (row % 2) * offset
            y = 0.1 + row * row_height
            if 0 <= x <= 1 and 0 <= y <= 1:
                centers[idx] = [x, y]
                idx += 1
            col += 1
            if col > 8:
                break
        row += 1
        if row > 8:
            break
    
    return centers, radii


def grid_init(n):
    """Initialize circles in a perturbed grid pattern."""
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.04
    
    # Place in roughly 5x5 with some perturbation
    grid_n = int(np.ceil(np.sqrt(n)))
    spacing = 0.8 / (grid_n - 1) if grid_n > 1 else 1.0
    start = 0.1
    
    idx = 0
    for i in range(grid_n):
        for j in range(grid_n):
            if idx >= n:
                break
            centers[idx] = [start + j * spacing, start + i * spacing]
            idx += 1
        if idx >= n:
            break
    
    # Add small random perturbation
    for i in range(n):
        centers[i, 0] += np.random.uniform(-0.01, 0.01)
        centers[i, 1] += np.random.uniform(-0.01, 0.01)
    
    return centers, radii


def random_init(n):
    """Initialize circles randomly with small radii."""
    np.random.seed(42)
    centers = np.random.uniform(0.1, 0.9, (n, 2))
    radii = np.ones(n) * 0.02
    return centers, radii


def compute_overlap_forces(centers, radii, n):
    """Compute repulsive forces for overlapping circles."""
    forces = np.zeros((n, 2))
    
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[j, 0] - centers[i, 0]
            dy = centers[j, 1] - centers[i, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            
            min_dist = radii[i] + radii[j]
            
            if dist < min_dist and dist > 1e-10:
                overlap = min_dist - dist
                strength = overlap * 10.0
                fx = dx / dist * strength
                fy = dy / dist * strength
                forces[i, 0] -= fx
                forces[i, 1] -= fy
                forces[j, 0] += fx
                forces[j, 1] += fy
    
    return forces


def compute_boundary_forces(centers, radii, n):
    """Compute forces to keep circles inside boundary."""
    forces = np.zeros((n, 2))
    strength = 50.0
    
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        
        if x < r + 0.001:
            forces[i, 0] += strength * (r + 0.001 - x)
        if x > 1 - r - 0.001:
            forces[i, 0] -= strength * (x - (1 - r - 0.001))
        if y < r + 0.001:
            forces[i, 1] += strength * (r + 0.001 - y)
        if y > 1 - r - 0.001:
            forces[i, 1] -= strength * (y - (1 - r - 0.001))
    
    return forces


def expand_radii(centers, radii, n):
    """Expand radii as much as possible given current centers."""
    new_radii = radii.copy()
    
    for iteration in range(20):
        for i in range(n):
            max_r = min(
                centers[i, 0],
                1.0 - centers[i, 0],
                centers[i, 1],
                1.0 - centers[i, 1]
            )
            
            for j in range(n):
                if i != j:
                    dx = centers[i, 0] - centers[j, 0]
                    dy = centers[i, 1] - centers[j, 1]
                    dist = np.sqrt(dx * dx + dy * dy)
                    max_r = min(max_r, dist - new_radii[j])
            
            new_radii[i] = min(new_radii[i], max(0, max_r))
    
    return new_radii


def optimize_packing(centers, radii, n, iterations=5000):
    """Iteratively optimize circle positions and radii."""
    centers = centers.copy()
    radii = radii.copy()
    
    for it in range(iterations):
        # Decay step size
        step = max(0.001, 0.1 * (1.0 - it / iterations))
        
        # Compute forces
        overlap_forces = compute_overlap_forces(centers, radii, n)
        boundary_forces = compute_boundary_forces(centers, radii, n)
        
        # Update centers
        centers += step * (overlap_forces + boundary_forces)
        
        # Project centers to valid region
        for i in range(n):
            centers[i, 0] = np.clip(centers[i, 0], radii[i] + 1e-8, 1.0 - radii[i] - 1e-8)
            centers[i, 1] = np.clip(centers[i, 1], radii[i] + 1e-8, 1.0 - radii[i] - 1e-8)
        
        # Expand radii
        if it % 5 == 0:
            radii = expand_radii(centers, radii, n)
        
        # Ensure radii don't exceed boundary
        for i in range(n):
            max_r = min(
                centers[i, 0],
                1.0 - centers[i, 0],
                centers[i, 1],
                1.0 - centers[i, 1]
            )
            radii[i] = min(radii[i], max_r)
    
    return centers, radii


def final_projection(centers, radii, n):
    """Final projection to ensure all constraints are satisfied."""
    radii = radii.copy()
    
    # Iteratively reduce radii to satisfy overlap constraints
    for iteration in range(100):
        for i in range(n):
            max_r = min(
                centers[i, 0],
                1.0 - centers[i, 0],
                centers[i, 1],
                1.0 - centers[i, 1]
            )
            
            for j in range(n):
                if i != j:
                    dx = centers[i, 0] - centers[j, 0]
                    dy = centers[i, 1] - centers[j, 1]
                    dist = np.sqrt(dx * dx + dy * dy)
                    max_r = min(max_r, dist - radii[j])
            
            radii[i] = min(radii[i], max(1e-10, max_r))
    
    # Final projection of centers
    for i in range(n):
        centers[i, 0] = np.clip(centers[i, 0], radii[i], 1.0 - radii[i])
        centers[i, 1] = np.clip(centers[i, 1], radii[i], 1.0 - radii[i])
    
    return centers, radii


def run_packing():

    def penalty_objective(p):
        c = p[:2 * n].reshape(n, 2)
        r = p[2 * n:]
        
        obj = -np.sum(r)
        pen = 0.0
        
        for i in range(n):
            x, y = c[i]
            rad = r[i]
            v1 = max(0, rad - x)
            v2 = max(0, x - (1 - rad))
            v3 = max(0, rad - y)
            v4 = max(0, y - (1 - rad))

                dist = np.sqrt(dx * dx + dy * dy)
