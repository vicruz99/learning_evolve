import numpy as np

def generate_hex_grid(n, s):
    """Generate n points in a hexagonal grid with spacing s."""
    points = []
    # Hexagonal lattice vectors
    v1 = np.array([1.0, 0.0]) * s
    v2 = np.array([0.5, np.sqrt(3)/2.0]) * s
    
    i, j = 0, 0
    while len(points) < n:
        # Add row j
        # Shift odd rows by s/2
        x_offset = 0
        if j % 2 == 1:
            x_offset = s / 2.0
        
        # Determine range of i for this row
        # We want to cover the square, so we iterate a wide range
        # and filter/sort later, but simpler is just to fill a block
        row_points = []
        for k in range(i, i + 10): # Check a generous range
            p = k * v1 + x_offset * v1 + j * v2
            if 0 <= p[0] <= 1 and 0 <= p[1] <= 1:
                row_points.append(p)
        
        points.extend(row_points)
        if len(points) >= n:
            break
            
        # Adjust i for next row logic if needed, but simpler to just scan
        # A better way for n points is to just generate a dense grid and pick
        pass
    
    # If the simple row-by-row didn't work well, fallback to dense grid scan
    if len(points) < n:
        points = []
        for j_idx in range(30):
            y = j_idx * s * np.sqrt(3) / 2
            if y > 1: break
            x_off = (s/2) if j_idx % 2 == 1 else 0
            for i_idx in range(30):
                x = x_off + i_idx * s
                if x <= 1:
                    points.append([x, y])
                else:
                    break
    return np.array(points[:n])

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    best_centers = None
    best_radii = None
    best_sum = 0.0
    
    # Try multiple initial configurations
    initial_configs = []
    
    # 1. Hexagonal Grid (Spacing ~0.21)
    hex_centers = generate_hex_grid(n, 0.21)

        initial_configs.append(hex_centers)

    sq_centers.append([0.5, 0.5]) # Already occupied, let's try 0.6, 0.5
    sq_centers[-1] = [0.6, 0.5]
    initial_configs.append(np.array(sq_centers))

    # 3. Random Perturbation of Hex
    if len(initial_configs) > 0:
        perturbed = initial_configs[0].copy()
        perturbed += np.random.uniform(-0.01, 0.01, perturbed.shape)
        # Clip to bounds roughly
        perturbed = np.clip(perturbed, 0.01, 0.99)
        initial_configs.append(perturbed)

    # Force-directed optimization parameters
    for init_centers in initial_configs:
        centers = init_centers.copy()
        r = 0.05 # Start with small radius
        
        # Grow radius
        for step in range(2000):
            r += 0.00005
            
            # Check validity and compute forces
            # We want to minimize overlaps.
            # If valid, we just keep increasing r.
            # If invalid, we move centers to reduce overlap.
            
            # Calculate overlaps
            overlaps = []
            total_overlap = 0.0
            
            # Boundary overlaps
            for i in range(n):
                x, y = centers[i]
                if x < r: 
                    overlaps.append((i, 'x', 'left'))
                    total_overlap += (r - x)**2
                if x > 1 - r:
                    overlaps.append((i, 'x', 'right'))
                    total_overlap += (x - (1-r))**2
                if y < r:
                    overlaps.append((i, 'y', 'down'))
                    total_overlap += (r - y)**2
                if y > 1 - r:
                    overlaps.append((i, 'y', 'up'))
                    total_overlap += (y - (1-r))**2
            
            # Pair overlaps
            for i in range(n):
                for j in range(i + 1, n):
                    dx = centers[i, 0] - centers[j, 0]
                    dy = centers[i, 1] - centers[j, 1]
                    dist = np.sqrt(dx*dx + dy*dy)
                    min_dist = 2 * r
                    if dist < min_dist:
                        overlaps.append((i, j, min_dist - dist))
                        total_overlap += (min_dist - dist)**2
            
            if total_overlap == 0:
                continue # Valid, next step will increase r
            
            # Move centers to reduce overlap
            forces = np.zeros_like(centers)
            
            # Apply boundary forces
            for i in range(n):
                x, y = centers[i]
                if x < r: forces[i, 0] += (r - x) * 10.0
                elif x > 1 - r: forces[i, 0] -= (x - (1-r)) * 10.0
                if y < r: forces[i, 1] += (r - y) * 10.0
                elif y > 1 - r: forces[i, 1] -= (y - (1-r)) * 10.0
            
            # Apply repulsive forces for overlaps
            for i in range(n):
                for j in range(i + 1, n):
                    dx = centers[i, 0] - centers[j, 0]
                    dy = centers[i, 1] - centers[j, 1]
                    dist = np.sqrt(dx*dx + dy*dy)
                    if dist < 2 * r and dist > 1e-9:
                        # Repulsive force proportional to penetration depth
                        penetration = 2 * r - dist
                        fx = (dx / dist) * penetration
                        fy = (dy / dist) * penetration
                        
                        # Apply to both
                        forces[i, 0] += fx * 0.5 # Dampen force
                        forces[i, 1] += fy * 0.5
                        forces[j, 0] -= fx * 0.5
                        forces[j, 1] -= fy * 0.5
            # Also apply a weak repulsion for non-overlapping close circles to prevent clumping
            # This helps spread them out to allow larger r
            for i in range(n):
                for j in range(i + 1, n):
                    dx = centers[i, 0] - centers[j, 0]
                    dy = centers[i, 1] - centers[j, 1]
                    dist = np.sqrt(dx*dx + dy*dy)
                    if dist < 3 * r and dist > 2 * r:
                        # Weak repulsion
                        rep = 1 / (dist * dist)
                        fx = (dx / dist) * rep
                        fy = (dy / dist) * rep
                        forces[i, 0] += fx * 0.01
                        forces[i, 1] += fy * 0.01
                        forces[j, 0] -= fx * 0.01
                        forces[j, 1] -= fy * 0.01

            # Apply forces
            centers += forces
            
            # Clamp to valid area [0, 1]
            centers = np.clip(centers, 0.001, 0.999)

        # After growing, verify and update best
        # Re-calculate r based on tightest constraint
        min_dist = 1.0
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                d = np.sqrt(dx*dx + dy*dy)
                if d < min_dist: min_dist = d
        
        min_wall = 1.0
        for i in range(n):
            x, y = centers[i]
            min_wall = min(min_wall, x, 1-x, y, 1-y)
            
        final_r = min(min_dist / 2, min_wall)
        final_sum = 26 * final_r
        
        if final_sum > best_sum:
            best_sum = final_sum
            best_centers = centers.copy()
            best_radii = np.full(n, final_r)
            
    return best_centers, best_radii, best_sum
