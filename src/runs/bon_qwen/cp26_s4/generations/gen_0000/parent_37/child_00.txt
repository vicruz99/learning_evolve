import numpy as np
from scipy.optimize import minimize

def get_initial_guess(n, seed=0):
    """
    Generates an initial guess for centers and radii.
    Uses a hexagonal packing pattern perturbed by noise.
    """
    rng = np.random.default_rng(seed)
    
    # Hexagonal packing parameters
    # Estimate a starting radius. 0.08 is safe and fits well.
    r_est = 0.08
    
    centers = np.zeros((n, 2))
    radii = np.full(n, r_est)
    
    # Try to place circles in a hexagonal grid
    # Row height is sqrt(3)/2 * 2r = sqrt(3)*r
    # Spacing x is 2r
    row_height = np.sqrt(3) * r_est
    col_width = 2 * r_est
    
    count = 0
    row = 0
    y_pos = r_est
    
    while count < n:
        col = 0
        x_pos = r_est
        # Offset every other row
        if row % 2 == 1:
            x_pos = r_est + col_width / 2
        
        while x_pos + r_est <= 1.0 and count < n:
            centers[count] = [x_pos + rng.uniform(-0.005, 0.005), 
                              y_pos + rng.uniform(-0.005, 0.005)]
            count += 1
            x_pos += col_width
        
        y_pos += row_height
        row += 1
        
    # If for some reason we didn't fill (unlikely with these params), fill remaining
    while count < n:
        centers[count] = [rng.uniform(0.1, 0.9), rng.uniform(0.1, 0.9)]
        count += 1
        
    return centers, radii

def objective_function(variables, n):
    """
    Objective function: minimize negative sum of radii.
    Variables layout: [x1, y1, r1, x2, y2, r2, ..., xn, yn, rn]
    """
    radii = variables[2::3]
    return -np.sum(radii)

def get_constraints(n):
    """
    Returns a list of constraint dictionaries for scipy.optimize.
    Constraints must be non-negative (>= 0).
    """
    constraints = []
    
    # 1. Boundary constraints
    # For each circle i:
    # x_i - r_i >= 0
    # y_i - r_i >= 0
    # 1 - (x_i + r_i) >= 0  => -x_i - r_i + 1 >= 0
    # 1 - (y_i + r_i) >= 0  => -y_i - r_i + 1 >= 0
    
    for i in range(n):
        # x_i >= r_i
        idx_x = 3 * i
        idx_r = 3 * i + 2
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, idx_x=idx_x, idx_r=idx_r: v[idx_x] - v[idx_r]
        })
        
        # y_i >= r_i
        idx_y = 3 * i + 1
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, idx_y=idx_y, idx_r=idx_r: v[idx_y] - v[idx_r]
        })
        
        # x_i + r_i <= 1
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, idx_x=idx_x, idx_r=idx_r: 1.0 - v[idx_x] - v[idx_r]
        })
        
        # y_i + r_i <= 1
        constraints.append({
            'type': 'ineq',
            'fun': lambda v, idx_y=idx_y, idx_r=idx_r: 1.0 - v[idx_y] - v[idx_r]
        })

    # 2. Non-overlap constraints
    # dist(i, j) >= r_i + r_j
    # dist^2 >= (r_i + r_j)^2
    # (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
    
    for i in range(n):
        for j in range(i + 1, n):
            idx_xi = 3 * i
            idx_yi = 3 * i + 1
            idx_ri = 3 * i + 2
            
            idx_xj = 3 * j
            idx_yj = 3 * j + 1
            idx_rj = 3 * j + 2
            
            # Using squared distance constraint to avoid sqrt, but careful with gradients?
            # SLSQP handles smooth functions. sqrt is smooth for dist > 0.
            # Since we start with non-overlapping, dist > 0 usually.
            # Let's use distance directly.
            
            def dist_constraint(v, i=i, j=j):
                xi, yi, ri = v[3*i], v[3*i+1], v[3*i+2]
                xj, yj, rj = v[3*j], v[3*j+1], v[3*j+2]
                dist = np.sqrt((xi - xj)**2 + (yi - yj)**2)
                return dist - (ri + rj)
            
            constraints.append({
                'type': 'ineq',
                'fun': dist_constraint
            })
            
    return constraints

def run_packing():
    """
    Solves the circle packing problem.
    """
    n = 26
    
    best_sum_radii = -np.inf
    best_centers = None
    best_radii = None
    
    # Try multiple random seeds to find a good packing
    num_restarts = 10
    
    constraints = get_constraints(n)
    
    for seed in range(num_restarts):
        centers, radii = get_initial_guess(n, seed=seed)
        
        # Flatten variables: [x1, y1, r1, ..., xn, yn, rn]
        x0 = np.empty(n * 3)
        for i in range(n):
            x0[3*i] = centers[i, 0]
            x0[3*i+1] = centers[i, 1]
            x0[3*i+2] = radii[i]
        
        # Optimization
        # Bounds for variables:
        # x, y in [0, 1]
        # r in [0, 0.5] (max radius in unit square is 0.5)
        bounds = []
        for i in range(n):
            bounds.extend([
                (0, 1), # x
                (0, 1), # y
                (1e-6, 0.5) # r (small positive lower bound to avoid 0)
            ])
            
        try:
            result = minimize(
                objective_function,
                x0,
                args=(n,),
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 1000, 'ftol': 1e-9}
            )
            
            if result.success:
                current_sum = -result.fun
                if current_sum > best_sum_radii:
                    best_sum_radii = current_sum
                    # Extract best solution

                    best_radii = np.zeros(n)
                    for i in range(n):
                        best_centers[i] = [result.x[3*i], result.x[3*i+1]]
                        best_radii[i] = result.x[3*i+2]
        except Exception:
            continue
            
    return best_centers, best_radii, best_sum_radii
