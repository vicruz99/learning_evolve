import numpy as np
import math
from scipy.optimize import minimize

def calculate_distance(p1, p2):
    return math.sqrt((p1[0] - p2[0])**2 + (p1[1] - p2[1])**2)

def objective_function(params, n_circles):
    # Parameters include centers (x, y) for each circle and a base radius scaling
    # We will optimize centers and radii directly.
    # Shape: (n_circles * 3) -> [x1, y1, r1, x2, y2, r2, ...]
    
    centers = params[:n_circles * 2].reshape((n_circles, 2))
    radii = params[n_circles * 2:]
    
    # Ensure non-negative radii
    if np.any(radii < 0):
        return 1e9
    
    cost = -np.sum(radii)
    
    # Boundary constraints: center - r >= 0 and center + r <= 1
    # Penalty for violation
    for i in range(n_circles):
        x, y = centers[i]
        r = radii[i]
        if x - r < 0:
            cost += 100 * (r - x)**2
        if x + r > 1:
            cost += 100 * (x + r - 1)**2
        if y - r < 0:
            cost += 100 * (r - y)**2
        if y + r > 1:
            cost += 100 * (y + r - 1)**2
            
    # Overlap constraints: dist >= r1 + r2
    for i in range(n_circles):
        for j in range(i + 1, n_circles):
            dist = calculate_distance(centers[i], centers[j])
            min_dist = radii[i] + radii[j]
            if dist < min_dist:
                overlap = min_dist - dist
                cost += 100 * overlap**2
                
    return cost

def run_packing():
    n = 26
    best_sum_radii = -1.0
    best_centers = None
    best_radii = None
    
    # Function to generate initial hexagonal positions
    def get_initial_config(seed):
        np.random.seed(seed)
        # Try a hexagonal packing layout
        # We want to fit n circles. 
        # Approximate radius for 26 circles in hex packing is slightly > 0.1
        # Let's try to arrange them in rows.
        # 5 rows: 6, 5, 6, 5, 4? No, 26.
        # 6, 5, 6, 5, 4 is 26.
        # Or 5, 6, 5, 6, 4?
        
        # Let's just scatter them in a hex grid pattern and let optimizer refine
        # Hex grid spacing approx 0.2 (since r~0.1)
        
        centers = np.zeros((n, 2))
        count = 0
        row = 0
        col = 0
        while count < n:
            # Stagger odd rows
            offset = 0.1 * (row % 2)
            x = 0.1 + col * 0.2 + offset
            y = 0.1 + row * 0.1732 # sqrt(3)/2 * 0.2
            
            if x < 1.0 and y < 1.0:
                centers[count] = [x, y]
                count += 1
                col += 1
            else:
                col = 0
                row += 1
        
        # If we didn't get 26 (should not happen with this logic), fill randomly
        while count < n:
            centers[count] = [np.random.rand(), np.random.rand()]
            count += 1
            
        # Add random jitter
        centers += np.random.randn(n, 2) * 0.01
        
        # Initial radii estimate
        # Inverse distance transform or just constant
        radii = np.ones(n) * 0.09 
        
        params = np.concatenate([centers.flatten(), radii])
        return params

    # Run optimization multiple times with different seeds
    for seed in range(50):
        initial_params = get_initial_config(seed)
        
        # Bounds: x, y in [0, 1], r in [0, 0.5]
        bounds = [(0, 1)] * (2 * n) + [(0, 0.5)] * n
        
        # We use L-BFGS-B because of bounds
        res = minimize(objective_function, initial_params, args=(n,), method='L-BFGS-B', bounds=bounds, 
                       options={'maxiter': 1000, 'ftol': 1e-12})
        
        centers_opt = res.x[:2 * n].reshape((n, 2))
        radii_opt = res.x[2 * n:]
        
        # Check validity manually with a tolerance
        valid = True
        for i in range(n):
            x, y = centers_opt[i]
            r = radii_opt[i]
            if x - r < -1e-6 or x + r > 1 + 1e-6 or y - r < -1e-6 or y + r > 1 + 1e-6:
                valid = False
                break
        
        if valid:
            for i in range(n):
                for j in range(i + 1, n):
                    dist = np.sqrt(np.sum((centers_opt[i] - centers_opt[j])**2))
                    if dist < radii_opt[i] + radii_opt[j] - 1e-6:
                        valid = False
                        break
                if not valid:
                    break
        
        if valid:
            current_sum = np.sum(radii_opt)
            if current_sum > best_sum_radii:
                best_sum_radii = current_sum
                best_centers = centers_opt.copy()

    # Ensure radii are non-negative
    if best_radii is not None:
        best_radii = np.maximum(best_radii, 0)
        
    return best_centers, best_radii, best_sum_radii
