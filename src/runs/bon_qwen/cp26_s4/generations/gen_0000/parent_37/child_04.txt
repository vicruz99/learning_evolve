    Pack 26 circles in a unit square to maximize the sum of radii.

    """
    n = 26

    np.random.seed(42)
    
    # Objective function with penalty method
    # We minimize: -sum(radii) + lam * penalty
    # Penalty enforces non-overlap and boundary constraints
    def objective(x):
        centers = x[:2*n].reshape(n, 2)
        radii = x[2*n:]
        
        # High penalty weight to enforce constraints.
        # Tuned to balance between maximizing radii and satisfying constraints.
        lam = 5000.0
        
        # 1. Overlap Penalty
        # Compute pairwise distances
        # diffs shape (n, n, 2)
        diffs = centers[:, None, :] - centers[None, :, :]
        dists = np.sqrt(np.sum(diffs**2, axis=2)) # (n, n)
        
        # Set diagonal to infinity to ignore self-distance
        np.fill_diagonal(dists, np.inf)
        
        # Sum of radii matrix
        rad_sum = radii[:, None] + radii[None, :]
        
        # Overlap amount: positive if circles overlap
        overlap = np.maximum(rad_sum - dists, 0.0)
        penalty_overlap = np.sum(overlap**2)

        x = centers[:, 0]
