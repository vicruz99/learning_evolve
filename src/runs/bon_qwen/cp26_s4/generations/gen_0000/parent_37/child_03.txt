def objective_func(x_vars):

def constraint_non_overlap(args):
    """
    Factory for non-overlap constraint functions.
    Returns a function that calculates distance - (r1 + r2).
    Constraint is satisfied if value >= 0.
    """
    i, j = args
    
    def constraint_func(x_vars):
        n = len(x_vars) // 3
        # Extract centers and radii
        # x_vars layout: [x0, y0, r0, x1, y1, r1, ...]
        idx_i = i * 3
        idx_j = j * 3
        
        x1, y1, r1 = x_vars[idx_i], x_vars[idx_i+1], x_vars[idx_i+2]
        x2, y2, r2 = x_vars[idx_j], x_vars[idx_j+1], x_vars[idx_j+2]
        
        dist = np.sqrt((x1 - x2)**2 + (y1 - y2)**2)
        return dist - (r1 + r2)
        
    return constraint_func

def constraint_boundary(args):
    """
    Factory for boundary constraint functions.
    Returns function for x - r >= 0 (or 1 - x - r >= 0, etc.)
    """
    i, boundary_type = args
    # boundary_type: 0 -> x - r >= 0
    #                1 -> 1 - x - r >= 0
    #                2 -> y - r >= 0
    #                3 -> 1 - y - r >= 0

    def constraint_func(x_vars):
        idx = i * 3
        x, y, r = x_vars[idx], x_vars[idx+1], x_vars[idx+2]

            return x - r
        elif boundary_type == 1:
            return 1.0 - x - r
        elif boundary_type == 2:
            return y - r
        elif boundary_type == 3:
            return 1.0 - y - r
        return 0.0

    return constraint_func

def generate_hexagonal_packing(n_circles):

    radii = []

    # Density ~ 0.907. Area ~ 1.

    count = 0

        y = r_start + row * (np.sqrt(3) * r_start)

            break

            if count < n_circles:
                centers.append([x, y])
                radii.append(r_start)
                count += 1
            x += 2 * r_start
        row += 1
        
    # If we didn't fit enough, just append some random ones or duplicate
    while count < n_circles:
        centers.append([0.5, 0.5])
        radii.append(0.05)
        count += 1
        
    return np.array(centers), np.array(radii)

def generate_grid_packing(n_circles):
    """Generate a grid packing initial guess."""
    centers = []
    radii = []
    # 5x5 grid fits 25 circles with r=0.1.
    # For 26, we can try to fit in 5x5 and one extra.
    r_start = 0.09
    
    # Simple grid placement
    k = int(np.ceil(np.sqrt(n_circles)))
    step = 1.0 / (k + 1) # Approx spacing
    
    # Better: just place them in a grid
    count = 0
    for i in range(k + 1):
        for j in range(k + 1):
            if count >= n_circles:
                break
            x = (i + 1) * step
            y = (j + 1) * step
            centers.append([x, y])
            radii.append(r_start)
            count += 1
        if count >= n_circles:
            break
            
    return np.array(centers), np.array(radii)
