import numpy as np
import scipy.optimize as opt

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Packs 26 circles in a unit square to maximize the sum of radii.
    Uses numerical optimization with multiple initial seeds.
    """
    n_circles = 26
    best_sum = -1.0
    best_centers = None
    best_radii = None

    # Helper to convert flat vector to centers and radii
    def get_params(params):
        # params is flat array of [x1, y1, r1, x2, y2, r2, ...]
        # But SLSQP handles bounds better if we separate or map correctly.
        # Let's use order: x, y, r for each circle.
        # Shape (n, 3)
        arr = params.reshape(-1, 3)
        centers = arr[:, :2]
        radii = arr[:, 2]
        return centers, radii

    # Objective: Maximize sum of radii => Minimize negative sum
    def objective(params):
        _, radii = get_params(params)
        return -np.sum(radii)

    # Constraints
    # 1. Boundary constraints: r <= x, r <= 1-x, r <= y, r <= 1-y
    # These can be handled by bounds if we shift variables, but let's use explicit constraints for clarity
    # or better, map variables.
    # Let's stick to constraints for flexibility.
    
    # Actually, bounds in SLSQP are (lb, ub).
    # We can set bounds for x, y in [0, 1] and r in [0, 0.5].
    # But we need r <= x, etc.
    # It's easier to use inequality constraints.
    
    constraints = []

    # Boundary constraints for each circle
    for i in range(n_circles):
        idx_x = i * 3
        idx_y = i * 3 + 1
        idx_r = i * 3 + 2
        
        # x - r >= 0
        def make_bound_const(i, type_):
            def con(params):
                c, r = get_params(params)
                if type_ == 'left': return c[i, 0] - r[i]
                if type_ == 'right': return 1.0 - c[i, 0] - r[i]
                if type_ == 'bottom': return c[i, 1] - r[i]
                if type_ == 'top': return 1.0 - c[i, 1] - r[i]
            return con

        constraints.append({'type': 'ineq', 'fun': make_bound_const(i, 'left')})
        constraints.append({'type': 'ineq', 'fun': make_bound_const(i, 'right')})
        constraints.append({'type': 'ineq', 'fun': make_bound_const(i, 'bottom')})
        constraints.append({'type': 'ineq', 'fun': make_bound_const(i, 'top')})

    # Non-overlap constraints
    # dist(i, j) - r_i - r_j >= 0
    # There are n*(n-1)/2 constraints. ~325 constraints.
    # This might be slow. We can batch them or just list them.
    # For 26 circles, 325 constraints is manageable for SLSQP? 
    # It might be on the edge. Let's try to limit or optimize.
    # Alternatively, we can rely on the fact that only nearby circles matter,
    # but the optimizer doesn't know that.
    
    # Let's define a function that returns all constraint violations for efficiency?
    # SLSQP accepts a dict with 'fun' returning array.
    
    def overlap_constraints(params):
        centers, radii = get_params(params)
        n = centers.shape[0]
        violations = []
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.linalg.norm(centers[i] - centers[j])
                violations.append(dist - radii[i] - radii[j])
        return np.array(violations)

    constraints.append({'type': 'ineq', 'fun': overlap_constraints})

    # Bounds: x in [0, 1], y in [0, 1], r in [0, 0.5]
    # Tighter bound for r: max possible radius is 0.5.
    bounds = []
    for _ in range(n_circles):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
        bounds.append((0.0, 0.5)) # r

    # --- Initialization Strategies ---
    
    def generate_seeds():
        seeds = []
        
        # Seed 1: Grid 5x5 + 1 in gap
        # 25 circles at 0.1, 0.3, 0.5, 0.7, 0.9 with r=0.1
        # 1 circle at (0.2, 0.2) with r=0.04
        centers = []
        radii = []
        grid_step = 0.2
        for i in range(5):
            for j in range(5):
                centers.append([0.1 + i * grid_step, 0.1 + j * grid_step])
                radii.append(0.09) # Slightly less to allow movement
        
        # Add 26th circle in a gap, e.g., (0.2, 0.2) is dist sqrt(0.02)=0.141 from (0.1,0.1)
        # r approx 0.04. 
        centers.append([0.2, 0.2])
        radii.append(0.04)
        
        seeds.append(np.hstack([np.array(centers), np.array(radii)[:, None]]).flatten())

        # Try to fit 26 circles in hex pattern
        # Rows: 5, 4, 5, 4, 5, 3? Sum = 26.

        # Let's try 5 rows of 5, 5, 5, 5, 5 (25) + 1.

        centers = []
        radii = []
        r_est = 0.09
        y_curr = r_est
        count = 0

            # Row 1 (shifted) centers at 2r, 4r...

                x = r_est + shift + k * (2 * r_est)
                centers.append([x, y_curr])
                radii.append(r_est)
                count += 1
                if count == 26: break
            if count == 26: break
            y_curr += r_est * np.sqrt(3)
            
        # Normalize to fit in [0,1] if needed, but r_est=0.09 should fit.
        # Just to be safe, scale if out of bounds?
        # For now, just use.
        seeds.append(np.hstack([np.array(centers), np.array(radii)[:, None]]).flatten())

        rng = np.random.RandomState(42)

        radii = rng.uniform(0.01, 0.1, 26)

        # Add noise
        centers += rng.normal(0, 0.01, centers.shape)
        centers = np.clip(centers, 0.05, 0.95)
        seeds.append(np.hstack([centers, radii[:, None]]).flatten())
        
        return seeds

    seeds = generate_seeds()
    
    for seed in seeds:

                if r[i] > max_r:
                    r[i] = max_r * 0.9
            seed[2::3] = r # Update r in seed
            
            res = opt.minimize(objective, seed, method='SLSQP', bounds=bounds, 
                               constraints=constraints, options={'ftol': 1e-9, 'maxiter': 200})
            
            if res.success:
                final_centers, final_radii = get_params(res.x)
                current_sum = np.sum(final_radii)

                # (The validation function checks constraints. We can run a quick check)
                # But running the full validation might be slow inside loop? 
                # Let's trust the optimizer but check basic validity.
                
                # Quick check
                valid = True
                for i in range(n_circles):
                    if final_radii[i] < 0 or final_centers[i,0] < 0 or final_centers[i,1] < 0:
                        valid = False; break
                    if final_centers[i,0] + final_radii[i] > 1 + 1e-9 or final_centers[i,1] + final_radii[i] > 1 + 1e-9:
                        valid = False; break
                    if final_centers[i,0] - final_radii[i] < -1e-9 or final_centers[i,1] - final_radii[i] < -1e-9:
                        valid = False; break
                
                if valid:
                    # Check overlaps quickly
                    for i in range(n_circles):
                        for j in range(i+1, n_circles):
                            dist = np.linalg.norm(final_centers[i] - final_centers[j])
                            if dist < final_radii[i] + final_radii[j] - 1e-9:
                                valid = False; break
                        if not valid: break

                if valid and current_sum > best_sum:
                    best_sum = current_sum
                    best_centers = final_centers.copy()
                    best_radii = final_radii.copy()
                    
        except Exception as e:
            # Continue with next seed
            pass

    # If optimization failed to find better, return a safe grid solution
    if best_centers is None or best_sum < 2.5:
        # Fallback to 5x5 grid + 1 small circle
        centers = []
        radii = []
        for i in range(5):
            for j in range(5):
                centers.append([0.1 + 0.2*i, 0.1 + 0.2*j])
                radii.append(0.1)
        centers.append([0.2, 0.2]) # In a gap
        radii.append(0.0414) # Max possible in that gap
        best_centers = np.array(centers)
        best_radii = np.array(radii)
        best_sum = np.sum(best_radii)
        # Adjust if overlap
        # (0.2, 0.2) dist to (0.1, 0.1) is 0.1414. r=0.1+0.0414=0.1414. Touches.
        # But (0.2, 0.2) dist to (0.3, 0.1) is sqrt(0.01+0.01)=0.1414.
        # So it's tight. Maybe reduce slightly.
        best_radii[-1] = 0.041

    return best_centers, best_radii, best_sum
