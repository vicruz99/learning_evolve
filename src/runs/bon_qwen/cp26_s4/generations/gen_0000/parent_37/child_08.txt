import numpy as np
import scipy.optimize as opt
import math

def run_packing():
    """
    Packs 26 circles in a unit square [0,1]x[0,1] to maximize the sum of radii.
    Uses hexagonal initialization and numerical optimization.
    """
    n = 26
    
    # --- 1. Initialization: Hexagonal Packing ---
    # We try to generate a hexagonal grid and pick 26 points.
    # A hexagonal lattice has vertical spacing r*sqrt(3) and horizontal spacing 2r.
    # We estimate an initial radius. For 25 circles (5x5), r=0.1. 
    # For 26, slightly less, maybe 0.095. Let's try to fit with r=0.095.
    
    # Function to get valid centers for a given radius on a hex grid
    def get_hex_centers(r, count):
        centers = []
        y = r
        row = 0
        # Iterate rows until we have enough points or exceed square
        while y + r <= 1.0 + 1e-9:
            if row % 2 == 0:
                x_start = r
            else:
                x_start = 2 * r # Shifted by r relative to even rows (center at 2r vs r)
            
            x = x_start
            while x + r <= 1.0 + 1e-9:
                centers.append((x, y))
                x += 2 * r
            y += r * math.sqrt(3)
            row += 1
        
        # If we don't have enough, we might need smaller r, but let's just take what we have
        # and scale or adjust later. 
        # However, optimization will handle resizing.
        return centers

    # Try to generate enough centers with a reasonable r
    # We want 26 centers.
    # Let's generate a grid with r=0.1 and see.
    # If too few, reduce r.
    r_init = 0.1
    while True:
        init_centers = get_hex_centers(r_init, 1000)
        if len(init_centers) >= n:
            break
        r_init -= 0.005
    
    # Select the first 26 centers
    init_centers = init_centers[:n]
    
    # Convert to arrays
    centers_init = np.array(init_centers)
    radii_init = np.full(n, r_init)
    
    # --- 2. Optimization ---
    # We want to maximize sum(radii)
    # Variables: x0, y0, r0, x1, y1, r1, ...
    # Total 3*n variables.
    
    # Initial guess vector
    x0 = np.zeros(3 * n)
    for i in range(n):
        x0[3*i] = centers_init[i, 0]
        x0[3*i+1] = centers_init[i, 1]
        x0[3*i+2] = radii_init[i]
        
    # Bounds: x, y in [0, 1], r in [0, 0.5]
    bounds = []
    for _ in range(n):
        bounds.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])
        
    # Constraints
    constraints = []
    
    # Boundary constraints:
    # x_i >= r_i  =>  x_i - r_i >= 0
    # x_i <= 1-r_i => x_i + r_i - 1 <= 0 => 1 - x_i - r_i >= 0
    # y_i >= r_i
    # y_i <= 1-r_i
    
    def con_left(val):
        # x_i - r_i >= 0
        res = []
        for i in range(n):
            res.append(val[3*i] - val[3*i+2])
        return np.array(res)

    def con_right(val):
        # 1 - x_i - r_i >= 0
        res = []
        for i in range(n):
            res.append(1.0 - val[3*i] - val[3*i+2])
        return np.array(res)

    def con_bottom(val):
        # y_i - r_i >= 0
        res = []
        for i in range(n):
            res.append(val[3*i+1] - val[3*i+2])
        return np.array(res)

    def con_top(val):
        # 1 - y_i - r_i >= 0
        res = []
        for i in range(n):
            res.append(1.0 - val[3*i+1] - val[3*i+2])
        return np.array(res)

    constraints.append({'type': 'ineq', 'fun': con_left})
    constraints.append({'type': 'ineq', 'fun': con_right})
    constraints.append({'type': 'ineq', 'fun': con_bottom})
    constraints.append({'type': 'ineq', 'fun': con_top})
    
    # Overlap constraints: dist(i,j) >= r_i + r_j
    # dist^2 >= (r_i + r_j)^2
    # (x_i-x_j)^2 + (y_i-y_j)^2 - (r_i+r_j)^2 >= 0
    
    # To reduce computational cost, we can pass a function that computes all pairwise constraints
    # But SLSQP expects a vector or scalar. Vector is fine.
    
    def con_overlap(val):
        res = []
        for i in range(n):
            xi = val[3*i]
            yi = val[3*i+1]
            ri = val[3*i+2]
            for j in range(i + 1, n):
                xj = val[3*j]
                yj = val[3*j+1]
                rj = val[3*j+2]
                
                dx = xi - xj
                dy = yi - yj
                dist_sq = dx*dx + dy*dy
                r_sum = ri + rj
                
                # We want dist_sq - r_sum^2 >= 0
                res.append(dist_sq - r_sum*r_sum)
        return np.array(res)

    constraints.append({'type': 'ineq', 'fun': con_overlap})
    
    # Objective function to minimize (negative sum of radii)
    def objective(val):
        s = 0.0
        for i in range(n):
            s += val[3*i+2]
        return -s # Minimize negative sum => Maximize sum

    # Run optimization
    # SLSQP is a good choice for constrained problems
    try:
        result = opt.minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints,

        x_opt = result.x
    except Exception:
        # Fallback to initialization if optimization fails
        x_opt = x0

    # Extract results
    centers_opt = np.zeros((n, 2))
    radii_opt = np.zeros(n)
    
    sum_radii = 0.0
    for i in range(n):
        centers_opt[i, 0] = x_opt[3*i]
        centers_opt[i, 1] = x_opt[3*i+1]
        radii_opt[i] = x_opt[3*i+2]
        sum_radii += radii_opt[i]
        
    # Clamp small negative radii due to numerical errors
    radii_opt = np.maximum(radii_opt, 0.0)
    
    return centers_opt, radii_opt, sum_radii
