        return False

    radii = params[2*n_circles:]
    
    # Penalty for constraints
    penalty = 0.0

        x, y = centers[i]
        r = radii[i]
        if x < r: penalty += 1000 * (r - x)**2
        if x > 1 - r: penalty += 1000 * (x - (1 - r))**2
