import numpy as np
from scipy.optimize import minimize


def objective_with_penalty(x, n, penalty):
    """Objective: maximize sum of radii minus penalty for constraint violations."""
    centers = x[:2 * n].reshape(n, 2)
    radii = x[2 * n:]
    
    obj = np.sum(radii)
    
    # Boundary penalties: each circle must satisfy r <= x <= 1-r and r <= y <= 1-r
    for i in range(n):
        xi = centers[i, 0]
        yi = centers[i, 1]
        ri = radii[i]
        
        if ri - xi > 0:
            obj -= penalty * (ri - xi) ** 2
        if xi + ri - 1 > 0:
            obj -= penalty * (xi + ri - 1) ** 2
        if ri - yi > 0:
            obj -= penalty * (ri - yi) ** 2
        if yi + ri - 1 > 0:
            obj -= penalty * (yi + ri - 1) ** 2
    
    # Overlap penalties (vectorized pairwise)
    # Use squared distance form for numerical stability
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist_sq = dx * dx + dy * dy
            min_dist = radii[i] + radii[j]
            min_dist_sq = min_dist * min_dist
            
            if dist_sq < min_dist_sq:
                # Penalty based on squared violation
                obj -= penalty * (min_dist_sq - dist_sq) ** 2
    
    return obj


def negative_objective(x, n, penalty):
    """Negated objective for minimization."""
    return -objective_with_penalty(x, n, penalty)


def initialize_hexagonal(n):
    """Initialize circles in a hexagonal pattern."""
    centers = np.zeros((n, 2))
    idx = 0
    
    # Pattern: 6-5-6-5-4 = 26 circles
    rows = [6, 5, 6, 5, 4]
    for row_idx, count in enumerate(rows):
        y = 0.12 + row_idx * 0.17
        for col_idx in range(count):
            if idx >= n:
                break
            x = 0.1 + col_idx * 0.14 + (row_idx % 2) * 0.07
            centers[idx] = [x, y]
            idx += 1
    
    return centers


def initialize_random(n, seed):
    """Initialize circles randomly with some structure."""
    np.random.seed(seed)
    
    # Place in rough grid then perturb
    centers = np.zeros((n, 2))
    idx = 0
    
    rows = [6, 5, 6, 5, 4]
    for row_idx, count in enumerate(rows):
        y = 0.12 + row_idx * 0.17
        for col_idx in range(count):
            if idx >= n:
                break
            x = 0.1 + col_idx * 0.14 + (row_idx % 2) * 0.07
            centers[idx] = [x + np.random.randn() * 0.01, y + np.random.randn() * 0.01]
            idx += 1
    
    return centers


def run_packing():
    n = 26
    
    best_centers = None
    best_radii = None
    best_sum = 0.0
    
    # Try multiple initial configurations
    for seed in range(15):
        # Initialize with perturbed hexagonal pattern
        centers = initialize_random(n, seed)
        radii = np.full(n, 0.08)
        
        x0 = np.concatenate([centers.flatten(), radii])
        
        # Penalty method: gradually increase penalty
        penalty = 100.0
        current_x = x0.copy()
        
        for phase in range(8):
            # Optimize with current penalty using BFGS
            result = minimize(
                negative_objective,
                current_x,
                args=(n, penalty),
                method='BFGS',
                options={
                    'maxiter': 800,
                    'gtol': 1e-10,
                    'disp': False
                }
            )
            
            current_x = result.x
            penalty *= 2.0
        
        centers_opt = current_x[:2 * n].reshape(n, 2)
        radii_opt = current_x[2 * n:]
        
        # Clip radii to be non-negative
        radii_opt = np.maximum(radii_opt, 0)
        
        # Clip centers to valid range
        for i in range(n):
            centers_opt[i, 0] = np.clip(centers_opt[i, 0], radii_opt[i], 1 - radii_opt[i])
            centers_opt[i, 1] = np.clip(centers_opt[i, 1], radii_opt[i], 1 - radii_opt[i])
        
        # Compute sum of radii
        total = np.sum(radii_opt)
        
        if total > best_sum:
            best_sum = total
            best_centers = centers_opt.copy()
            best_radii = radii_opt.copy()
    
    return best_centers, best_radii, best_sum
