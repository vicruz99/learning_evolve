import numpy as np
import scipy.optimize as opt

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    Pack 26 circles in a unit square to maximize the sum of radii.
    """
    n_circles = 26

    # A hexagonal packing has higher density than a square grid.
    # We rotate the lattice to align better with the square boundaries.
    
    centers = np.zeros((n_circles, 2))
    radii = np.zeros(n_circles)

    angle = np.radians(14) # Rotation angle to fit boundaries better
    cos_a, sin_a = np.cos(angle), np.sin(angle)

    for i in range(20):

            y = i * r_init * np.sqrt(3)
            
            # Rotate
            x_rot = x * cos_a - y * sin_a
            y_rot = x * sin_a + y * cos_a
            
            # Shift to center in [0,1]
            # We can just collect points and scale later, but let's try to fit directly
            # Simple bounding box centering
            if count < n_circles:
                centers[count] = [x_rot, y_rot]
                radii[count] = r_init
                count += 1
    
    # Normalize the initial configuration to fit within [0,1]x[0,1]
    if count > 0:

        # For now, just scale to fit in [0,1] and let optimizer handle radius expansion
        scale = 1.0 / max(current_width, current_height)

        centers[:, 1] *= scale
        radii *= scale

        centers[:, 0] -= (np.min(centers[:, 0]) + np.max(centers[:, 0]) - 1.0) / 2.0

        # The optimizer will expand them.

        avg_r = np.mean(radii)
        radii[:] = avg_r

        r = vars_flat[2::3]
        return -np.sum(r)

        constraints = []
        for i in range(n_circles):
            x = vars_flat[3*i]
            y = vars_flat[3*i+1]
            r = vars_flat[3*i+2]
            
            # x - r >= 0  => x - r >= 0
            constraints.append(x - r - 1e-6)
            # x + r <= 1  => 1 - (x + r) >= 0
            constraints.append(1 - (x + r) - 1e-6)
            # y - r >= 0
            constraints.append(y - r - 1e-6)
            # y + r <= 1
            constraints.append(1 - (y + r) - 1e-6)
            
            # r >= 0
            constraints.append(r - 1e-6)
        return constraints

        constraints = []
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                x1, y1, r1 = vars_flat[3*i], vars_flat[3*i+1], vars_flat[3*i+2]

                dx = x1 - x2
                dy = y1 - y2
                dist = np.sqrt(dx**2 + dy**2)
                
                # dist >= r1 + r2  => dist - (r1 + r2) >= 0
                constraints.append(dist - (r1 + r2) - 1e-6)
        return constraints

    # Define constraints for scipy

    # Type 'ineq' means func(x) >= 0
    
    cons = []
    
    # Boundary constraints
    for i in range(n_circles):
        cons.append({'type': 'ineq', 'fun': lambda v, idx=i: v[3*idx] - v[3*idx+2] - 1e-6})       # x - r >= 0

    # Using SLSQP
    result = opt.minimize(objective, x0, method='SLSQP', bounds=bnds, constraints=cons,

        final_centers = centers

    sum_radii = np.sum(final_radii)
    
    return final_centers, final_radii, sum_radii
