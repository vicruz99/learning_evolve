import numpy as np
from scipy.optimize import minimize, linprog

def get_hexagonal_centers(n_circles):
    """
    Generates initial centers on a hexagonal lattice to promote dense packing.
    """
    centers = []
    row = 0
    cols_per_row = [6, 5, 6, 5, 4] # 6+5+6+5+4 = 26 circles
    y_spacing = 0.25
    x_spacing = 0.25
    
    # Start a bit away from walls
    y_start = 0.15
    x_start = 0.1
    
    for c in cols_per_row:
        if len(centers) >= n_circles:
            break
        # Shift odd rows for hexagonal packing
        x_offset = (x_spacing / 2) if (row % 2 == 1) else 0
        
        for i in range(c):
            if len(centers) >= n_circles:
                break
            x = x_start + i * x_spacing + x_offset
            y = y_start + row * y_spacing
            centers.append([x, y])
        row += 1
        
    return np.array(centers[:n_circles])

def solve_radii_lp(centers):
    """
    Given fixed centers, solves LP to maximize sum of radii.
    """
    n = centers.shape[0]
    
    # 1. Calculate boundary limits (distance to nearest wall)
    # r_i <= min(x, 1-x, y, 1-y)
    x = centers[:, 0]
    y = centers[:, 1]
    b = np.minimum(np.minimum(x, 1 - x), np.minimum(y, 1 - y))
    
    # 2. Prepare constraints for linprog
    # We want to maximize sum(r), so we minimize -sum(r)
    c_obj = -np.ones(n)
    
    # Constraints matrix A_ub * r <= b_ub
    # Constraints types:
    # a) r_i + r_j <= dist_ij  (Pairwise non-overlap)
    # b) r_i <= b_i            (Boundary)
    
    n_constraints = 0
    # Count pairwise constraints
    n_constraints += n * (n - 1) // 2
    # Count boundary constraints
    n_constraints += n
    
    A_ub = np.zeros((n_constraints, n))
    b_ub = np.zeros(n_constraints)
    
    idx = 0
    
    # Pairwise constraints
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt((centers[i, 0] - centers[j, 0])**2 + 
                           (centers[i, 1] - centers[j, 1])**2)
            A_ub[idx, i] = 1.0
            A_ub[idx, j] = 1.0
            b_ub[idx] = dist
            idx += 1
            
    # Boundary constraints
    for i in range(n):
        A_ub[idx, i] = 1.0
        b_ub[idx] = b[i]
        idx += 1
        
    # Bounds for radii: r_i >= 0
    bounds = [(0, None) for _ in range(n)]
    
    # Solve LP
    # method='highs' is efficient and default in modern scipy
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        max_sum = -res.fun
        radii = res.x
        return max_sum, radii
    else:
        # Fallback if LP fails (should not happen with valid geometry)
        return 0.0, np.zeros(n)

def objective_function(centers_flat):
    """
    Objective function for optimizing centers.
    Minimizes negative sum of radii.
    """
    centers = centers_flat.reshape((26, 2))
    # Ensure centers are within bounds (soft constraint handled by optimizer bounds)
    # But clip just in case to prevent numerical issues in distance calc
    centers = np.clip(centers, 1e-6, 1 - 1e-6)
    
    sum_radii, _ = solve_radii_lp(centers)
    return -sum_radii

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes the packing of 26 circles in a unit square.
    """
    n_circles = 26
    
    # Generate initial hexagonal grid
    initial_centers = get_hexagonal_centers(n_circles)
    initial_centers_flat = initial_centers.flatten()
    
    # Bounds for centers: [0, 1]
    bounds = [(0.0, 1.0) for _ in range(26 * 2)]
    
    best_sum = 0.0
    best_centers = initial_centers
    best_radii = np.zeros(n_circles)
    
    # Multi-start optimization to avoid local minima
    # We run Nelder-Mead a few times with small perturbations
    n_restarts = 5
    for k in range(n_restarts):
        # Perturb initial centers slightly
        if k > 0:
            noise = np.random.uniform(-0.02, 0.02, size=initial_centers.shape)
            current_flat = initial_centers_flat + noise.flatten()
            current_flat = np.clip(current_flat, 0, 1)
        else:
            current_flat = initial_centers_flat.copy()
            
        # Run optimization
        # Nelder-Mead is good for non-smooth functions (like LP value)
        res = minimize(objective_function, current_flat, method='Nelder-Mead', 
                       bounds=bounds, 
                       options={'maxiter': 2000, 'xatol': 1e-6, 'fatol': 1e-6})

            current_sum = -res.fun
            if current_sum > best_sum:
                best_sum = current_sum
                best_centers = res.x.reshape((n_circles, 2))
                # Re-solve LP to get precise radii for the best centers
                _, best_radii = solve_radii_lp(best_centers)

    # Validation check (internal sanity check, though validation function is provided externally)
    # The solve_radii_lp guarantees valid radii for the centers
    
    return best_centers, best_radii, best_sum
