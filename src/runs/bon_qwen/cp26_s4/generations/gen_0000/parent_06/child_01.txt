2. **Force-Based Repulsion Optimization**: Use a molecular-dynamics-like approach where:
   - Overlapping circles experience strong repulsive forces
   - Circles near boundaries are pushed inward
   - Radii are gradually expanded as space becomes available

3. **Multiple Random Restarts**: Since this is a non-convex problem, use multiple random initializations to escape local minima.

4. **Post-processing Cleanup**: Ensure numerical validity by clipping boundaries and resolving any tiny overlaps.

5. **Key Insight**: The force-based expansion allows circles to find non-uniform sizes naturally - corner circles can grow larger while interior ones adjust, which may yield a higher sum than equal radii.

def hexagonal_init(n, row_counts):
    num_rows = len(row_counts)
    r = 0.09
    
    centers = []
    radii = []
    
    for row_idx, count in enumerate(row_counts):
        y = r + row_idx * r * np.sqrt(3.0)
        
        x_offset = r * (row_idx % 2)
        row_width = (count - 1) * 2.0 * r
        x_start = (1.0 - row_width) / 2.0 + x_offset
        
        for col_idx in range(count):
            x = x_start + col_idx * 2.0 * r
            centers.append([x, y])
            radii.append(r)
    
    return np.array(centers), np.array(radii)


def random_init(n, seed):
    rng = np.random.RandomState(seed)
    
    centers = rng.rand(n, 2)
    radii = np.ones(n) * 0.02
    
    return centers, radii


def corner_init(n):
    centers = []
    radii = []
    
    # Place 4 corner circles
    r_corner = 0.15
    for cx in [r_corner, 1.0 - r_corner]:
        for cy in [r_corner, 1.0 - r_corner]:
            centers.append([cx, cy])
            radii.append(r_corner)
    
    # Place edge circles
    r_edge = 0.08
    edge_positions = []
    for i in range(3):
        x = (i + 1) / 4.0
        edge_positions.append([x, r_edge])
        edge_positions.append([x, 1.0 - r_edge])
        edge_positions.append([r_edge, x])
        edge_positions.append([1.0 - r_edge, x])
    
    placed = 4
    for pos in edge_positions:
        if placed >= n:
            break
        centers.append(pos)
        radii.append(r_edge)
        placed += 1
    
    # Fill remaining with random small circles
    rng = np.random.RandomState(42)
    while placed < n:
        x = rng.uniform(0.2, 0.8)
        y = rng.uniform(0.2, 0.8)
        centers.append([x, y])
        radii.append(0.03)
        placed += 1
    
    return np.array(centers[:n]), np.array(radii[:n])


def force_optimize(centers, radii, n, max_iter=8000):
    centers = centers.astype(np.float64).copy()
    radii = radii.astype(np.float64).copy()
    
    for iteration in range(max_iter):
        forces = np.zeros_like(centers)
        
        # Circle-circle repulsion (vectorized where possible)
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist_sq = dx * dx + dy * dy
                dist = np.sqrt(dist_sq + 1e-12)
                min_dist = radii[i] + radii[j]
                
                if dist < min_dist:
                    overlap = min_dist - dist
                    force_mag = overlap * 50.0 / (dist + 1e-12)
                    fx = force_mag * dx / (dist + 1e-12)
                    fy = force_mag * dy / (dist + 1e-12)
                    forces[i, 0] += fx
                    forces[i, 1] += fy
                    forces[j, 0] -= fx
                    forces[j, 1] -= fy
        
        # Wall repulsion
        for i in range(n):
            for axis in range(2):
                if centers[i, axis] < radii[i] + 1e-10:
                    forces[i, axis] += (radii[i] - centers[i, axis]) * 200.0
                if centers[i, axis] > 1.0 - radii[i] - 1e-10:
                    forces[i, axis] -= (centers[i, axis] - (1.0 - radii[i])) * 200.0
        
        # Update positions with adaptive learning rate
        progress = iteration / max_iter
        lr = 0.15 * (1.0 - progress * 0.7)
        centers = centers + lr * forces
        centers = np.clip(centers, 1e-8, 1.0 - 1e-8)
        
        # Expand radii gradually
        if iteration % 2 == 0:
            for i in range(n):
                min_wall = min(centers[i, 0], 1.0 - centers[i, 0],
                              centers[i, 1], 1.0 - centers[i, 1])
                min_neighbor = 2.0
                for j in range(n):
                    if i != j:
                        dist = np.sqrt((centers[i, 0] - centers[j, 0])**2 + 
                                     (centers[i, 1] - centers[j, 1])**2)
                        min_neighbor = min(min_neighbor, dist - radii[j])
                
                room = min(min_wall - radii[i], min_neighbor - radii[i])
                if room > 1e-10:
                    radii[i] += room * 0.08
    
    return centers, radii


def cleanup(centers, radii, n):
    centers = centers.astype(np.float64).copy()
    radii = radii.astype(np.float64).copy()
    
    # Ensure circles are within bounds
    for i in range(n):
        max_r = min(centers[i, 0], 1.0 - centers[i, 0],
                    centers[i, 1], 1.0 - centers[i, 1])
        radii[i] = min(radii[i], max_r - 1e-10)
        radii[i] = max(radii[i], 0.0)
    
    # Iteratively resolve overlaps by reducing the smaller circle
    for iteration in range(50):
        overlap_found = False
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy + 1e-12)
                
                if dist < radii[i] + radii[j] - 1e-10:
                    overlap = radii[i] + radii[j] - dist
                    # Reduce proportionally
                    total_r = radii[i] + radii[j]
                    if total_r > 1e-12:
                        scale_i = radii[i] / total_r
                        scale_j = radii[j] / total_r
                    else:
                        scale_i = 0.5
                        scale_j = 0.5
                    
                    reduction = overlap + 1e-10
                    radii[i] -= reduction * scale_i
                    radii[j] -= reduction * scale_j
                    
                    # Recheck boundary
                    for circle_idx in [i, j]:
                        max_r = min(centers[circle_idx, 0], 1.0 - centers[circle_idx, 0],
                                   centers[circle_idx, 1], 1.0 - centers[circle_idx, 1])
                        radii[circle_idx] = min(radii[circle_idx], max_r - 1e-10)
                        radii[circle_idx] = max(radii[circle_idx], 0.0)
                    
                    overlap_found = True
        
        if not overlap_found:
            break
    
    return centers, radii
