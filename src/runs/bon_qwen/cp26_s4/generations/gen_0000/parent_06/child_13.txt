

Looking at this circle packing problem, I need to maximize the sum of 26 circle radii in a unit square. Let me reason about the strategy.

<strategy>
1. **Initial Placement**: Use a hexagonal grid pattern which is the densest packing arrangement for circles. This gives a good starting configuration with circles evenly distributed.

2. **Force-Directed Optimization**: Implement an iterative method where:
   - Circles repel each other when overlapping (pushing centers apart)
   - Strong boundary repulsion keeps circles inside the square
   - All circles grow gradually each iteration
   - This naturally maximizes radii as circles are pushed to their limits

3. **Multi-phase approach**:
   - Phase 1: Large steps to quickly resolve overlaps and grow radii
   - Phase 2: Medium steps for refinement
   - Phase 3: Small steps for fine-tuning the final arrangement

4. **Multiple restarts**: Try different initial configurations (hexagonal, random, grid) and keep the best result to avoid local optima.

5. **Post-processing**: After the force-directed method converges, attempt to further grow radii by redistributing space - shrinking circles in tight spots slightly to allow others to grow more.

The key insight is that uniform growth with repulsion naturally finds a configuration where circles are "jammed" - touching neighbors and boundaries - which is exactly what maximizes the sum of radii.
</strategy>

import numpy as np


def compute_validation(centers, radii, n):
    """Check if packing is valid."""
    for i in range(n):
        x, y, r = centers[i, 0], centers[i, 1], radii[i]
        if r < 0:
            return False
        if x - r < -1e-12 or x + r > 1 + 1e-12:
            return False
        if y - r < -1e-12 or y + r > 1 + 1e-12:
            return False
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False
    return True


def force_optimize(centers, radii, n, iterations, step_size, growth_rate):
    """Run force-directed optimization."""
    step = step_size
    grow = growth_rate
    
    for it in range(iterations):
        F = np.zeros((n, 2))
        
        # Repulsive forces between circles
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[j, 0] - centers[i, 0]
                dy = centers[j, 1] - centers[i, 1]
                d2 = dx * dx + dy * dy
                d = np.sqrt(d2) if d2 > 1e-20 else 1e-10
                rsum = radii[i] + radii[j]
                
                if d < rsum:
                    # Hard overlap: strong push
                    push = (rsum - d) / d
                    F[i, 0] -= push * dx
                    F[i, 1] -= push * dy
                    F[j, 0] += push * dx
                    F[j, 1] += push * dy
                elif d < rsum * 1.15:
                    # Soft repulsion to prevent future overlaps
                    push = (rsum * 1.15 - d) / (d * 20.0)
                    F[i, 0] -= push * dx
                    F[i, 1] -= push * dy
                    F[j, 0] += push * dx
                    F[j, 1] += push * dy
        
        # Boundary forces
        for i in range(n):
            if centers[i, 0] < radii[i]:
                F[i, 0] += (radii[i] - centers[i, 0]) * 100.0
            elif centers[i, 0] > 1.0 - radii[i]:
                F[i, 0] -= (centers[i, 0] - 1.0 + radii[i]) * 100.0
            if centers[i, 1] < radii[i]:
                F[i, 1] += (radii[i] - centers[i, 1]) * 100.0
            elif centers[i, 1] > 1.0 - radii[i]:
                F[i, 1] -= (centers[i, 1] - 1.0 + radii[i]) * 100.0
        
        # Apply forces
        centers = centers + F * step
        
        # Grow radii
        radii = radii * grow
        
        # Clamp positions to valid range
        for i in range(n):
            centers[i, 0] = max(radii[i], min(1.0 - radii[i], centers[i, 0]))
            centers[i, 1] = max(radii[i], min(1.0 - radii[i], centers[i, 1]))
        
        # Decay step size
        if it % 20000 == 0 and it > 0:
            step *= 0.85
    
    return centers, radii


def hexagonal_init(n, spacing_x, spacing_y, offset_x, offset_y):
    """Create hexagonal grid initialization."""
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.001
    
    idx = 0
    for row in range(10):
        for col in range(10):
            if idx >= n:
                break
            x = offset_x + col * spacing_x + (row % 2) * (spacing_x / 2.0)
            y = offset_y + row * spacing_y
            if x <= 0.99 and y <= 0.99 and x >= 0.01 and y >= 0.01:
                centers[idx] = [x, y]
                idx += 1
        if idx >= n:
            break
    
    return centers, radii


def post_optimize(centers, radii, n):
    """Fine-tune by trying to grow individual circles."""
    improved = True
    while improved:
        improved = False
        for i in range(n):
            # Find max possible radius for circle i given current positions of others
            max_r = min(centers[i, 0], 1.0 - centers[i, 0],
                        centers[i, 1], 1.0 - centers[i, 1])
            for j in range(n):
                if i == j:
                    continue
                dist = np.sqrt((centers[i, 0] - centers[j, 0]) ** 2 +
                               (centers[i, 1] - centers[j, 1]) ** 2)
                max_r = min(max_r, dist - radii[j])
            max_r = max(max_r, 0)
            if max_r > radii[i] + 1e-8:
                radii[i] = max_r
                improved = True
    
    return centers, radii


def redistribute_radii(centers, radii, n, num_passes=50):
    """Try to redistribute radii to increase total sum."""
    for _ in range(num_passes):
        # Find the circle with smallest "wiggle room"
        # and the circle with most "wiggle room"
        wiggle = np.zeros(n)
        for i in range(n):
            w = min(centers[i, 0] - radii[i], 1.0 - centers[i, 0] - radii[i],
                    centers[i, 1] - radii[i], 1.0 - centers[i, 1] - radii[i])
            for j in range(n):
                if i == j:
                    continue
                dist = np.sqrt((centers[i, 0] - centers[j, 0]) ** 2 +
                               (centers[i, 1] - centers[j, 1]) ** 2)
                w = min(w, dist - radii[i] - radii[j])
            wiggle[i] = max(w, 0)
        
        # Try to transfer from tight circles to loose ones
        for i in range(n):
            if wiggle[i] < 0:
                continue
            delta = wiggle[i] * 0.1
            if delta > 1e-8:
                new_r = radii[i] + delta
                valid = True
                for j in range(n):
                    if i == j:
                        continue
                    dist = np.sqrt((centers[i, 0] - centers[j, 0]) ** 2 +
                                   (centers[i, 1] - centers[j, 1]) ** 2)
                    if dist < new_r + radii[j] - 1e-10:
                        valid = False
                        break
                if new_r > centers[i, 0] or new_r > 1.0 - centers[i, 0]:
                    valid = False
                if new_r > centers[i, 1] or new_r > 1.0 - centers[i, 1]:
                    valid = False
                if valid:
                    radii[i] = new_r
    
    return centers, radii


def run_packing():
    n = 26
    best_centers = None
    best_radii = None
    best_sum = 0.0
    
    # Try multiple configurations
    configs = [
        # (spacing_x, spacing_y, offset_x, offset_y)
        (0.17, 0.15, 0.06, 0.06),
        (0.18, 0.14, 0.05, 0.05),
        (0.16, 0.16, 0.07, 0.07),
        (0.19, 0.13, 0.04, 0.04),
        (0.15, 0.17, 0.08, 0.08),
    ]
    
    for cfg in configs:
        centers, radii = hexagonal_init(n, cfg[0], cfg[1], cfg[2], cfg[3])
        
        # Phase 1: Quick growth with large steps
        centers, radii = force_optimize(centers, radii, n, 100000, 0.03, 1.00005)
        
        # Phase 2: Medium refinement
        centers, radii = force_optimize(centers, radii, n, 100000, 0.01, 1.00002)
        
        # Phase 3: Fine tuning
        centers, radii = force_optimize(centers, radii, n, 100000, 0.003, 1.00001)
        
        # Post-process: try to maximize each circle's radius
        centers, radii = post_optimize(centers, radii, n)
        
        # Redistribute for better sum
        centers, radii = redistribute_radii(centers, radii, n, 100)
        
        s = np.sum(radii)
        if s > best_sum:
            best_sum = s
            best_centers = centers.copy()
            best_radii = radii.copy()
    
    # Final force optimization on best result
    best_centers, best_radii = force_optimize(best_centers, best_radii, n, 150000, 0.005, 1.00001)
    best_centers, best_radii = post_optimize(best_centers, best_radii, n)
    best_centers, best_radii = redistribute_radii(best_centers, best_radii, n, 200)
    best_centers, best_radii = force_optimize(best_centers, best_radii, n, 100000, 0.002, 1.000005)
    best_centers, best_radii = post_optimize(best_centers, best_radii, n)
    
    # Validate and fix any numerical issues
    for i in range(n):
        best_centers[i, 0] = max(best_radii[i], min(1.0 - best_radii[i], best_centers[i, 0]))
        best_centers[i, 1] = max(best_radii[i], min(1.0 - best_radii[i], best_centers[i, 1]))
        best_radii[i] = max(best_radii[i], 0)
    
    return best_centers, best_radii, np.sum(best_radii)
