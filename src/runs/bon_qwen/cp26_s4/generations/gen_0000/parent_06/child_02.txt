def calculate_penalty(centers, radii):
    """
    Calculates the sum of squared violations for boundary and overlap constraints.
    Uses vectorized operations for performance.
    """
    n = centers.shape[0]
    penalty = 0.0
    
    # Boundary violations
    # Left wall: x >= r  => r - x <= 0. Violation is max(0, r - x)
    viol_left = np.maximum(0, radii - centers[:, 0])
    # Right wall: x + r <= 1 => x + r - 1 <= 0. Violation is max(0, x + r - 1)
    viol_right = np.maximum(0, centers[:, 0] + radii - 1.0)
    # Bottom wall: y >= r => r - y <= 0
    viol_bottom = np.maximum(0, radii - centers[:, 1])
    # Top wall: y + r <= 1 => y + r - 1 <= 0
    viol_top = np.maximum(0, centers[:, 1] + radii - 1.0)
    
    penalty += np.sum(viol_left**2 + viol_right**2 + viol_bottom**2 + viol_top**2)
    
    # Pairwise overlap violations
    # Distance matrix calculation using broadcasting
    # diff shape: (n, n, 2)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    # dist_sq shape: (n, n)
    dist_sq = np.sum(diff**2, axis=2)
    # Add small epsilon to avoid sqrt(0) issues, though dist > 0 usually
    dist = np.sqrt(dist_sq + 1e-12)
    
    # Sum of radii matrix shape: (n, n)
    sum_radii_matrix = radii[:, np.newaxis] + radii[np.newaxis, :]
    
    # Overlap violation: r_i + r_j - dist_ij. If > 0, there is overlap.
    violation = sum_radii_matrix - dist
    
    # We only need to sum the upper triangle to count each pair once
    # Set lower triangle and diagonal to 0 before squaring/summing?
    # Actually, violation is symmetric. Summing full matrix counts pairs twice.
    # Diagonal is r_i + r_i - 0 = 2r_i, which is not a constraint (distance to self is 0).
    # We should ignore diagonal.
    
    # Create a mask for upper triangle (excluding diagonal)
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    upper_violations = violation[mask]
    
    # Squared violations
    penalty += np.sum(np.maximum(0, upper_violations)**2)
    
    return penalty

def objective_function(variables, n, penalty_coeff):
    """
    Objective function to minimize: -sum(radii) + penalty_coeff * penalty.

    radii = np.maximum(radii, 0.0)
    
    pen = calculate_penalty(centers, radii)
    obj = -np.sum(radii) + penalty_coeff * pen
    
    return obj
