import numpy as np
from scipy.optimize import minimize

def compute_radii_and_sum(centers):
    """
    Computes the maximum valid radius for each circle given their centers,
    and returns the radii array and their sum.
    """
    n = centers.shape[0]
    
    # Distance to boundaries [0,1]x[0,1]
    r_bound = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                         np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    
    # Pairwise Euclidean distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    np.fill_diagonal(dists, np.inf)
    r_circles = np.min(dists, axis=1) / 2.0
    
    # Feasible radius is limited by both boundaries and neighbors
    radii = np.minimum(r_bound, r_circles)
    return radii, np.sum(radii)

def objective(x):
    """
    Objective function for optimization. Maximizes sum of radii.
    x is a flattened array of centers (n*2,).
    """
    centers = np.clip(x.reshape(-1, 2), 1e-9, 1.0 - 1e-9)
    _, total = compute_radii_and_sum(centers)
    return -total  # Minimize negative sum
