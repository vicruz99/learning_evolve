                return False
    return True

def get_overlap_penalty(centers, radii):
    n = centers.shape[0]
    penalty = 0.0
    # Boundary
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < 0: penalty += (x - r)**2
        if x + r > 1: penalty += (x + r - 1)**2
        if y - r < 0: penalty += (y - r)**2
        if y + r > 1: penalty += (y + r - 1)**2
    # Overlap
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            min_dist = radii[i] + radii[j]
            if dist < min_dist:
                overlap = min_dist - dist
                penalty += overlap**2
    return penalty

    x = vars_array[0::3]
    y = vars_array[1::3]
    r = vars_array[2::3]
    centers = np.column_stack((x, y))

    # Add heavy penalty for constraints
    obj = -np.sum(r)
    penalty = 10000.0 * get_overlap_penalty(centers, r)
    return obj + penalty

def relax_packing(centers, radii, n_steps=500):
    """
    Iteratively move circles to reduce overlap and increase radii.
    """
    n = len(radii)
    # Convert to float
    centers = centers.astype(float)
    radii = radii.astype(float)
    
    lr = 0.01 # Learning rate
    
    for step in range(n_steps):
        forces = np.zeros_like(centers)
        
        # 1. Overlap resolution (Repulsion)
        for i in range(n):
            for j in range(i + 1, n):
                vec = centers[i] - centers[j]
                dist = np.linalg.norm(vec)
                if dist == 0: dist = 1e-9
                dir_vec = vec / dist
                
                required_dist = radii[i] + radii[j]
                if dist < required_dist:
                    # Repulsive force proportional to overlap
                    # Push apart
                    overlap = required_dist - dist
                    force_mag = overlap * 10.0 # Strong repulsion
                    forces[i] += dir_vec * force_mag * 0.5
                    forces[j] -= dir_vec * force_mag * 0.5
        
        # 2. Boundary repulsion
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            
            # Push away from boundaries if too close
            # Left
            if x - r < 0:
                forces[i, 0] += (r - x) * 10.0
            # Right
            if x + r > 1:
                forces[i, 0] -= (x + r - 1) * 10.0
            # Bottom
            if y - r < 0:
                forces[i, 1] += (r - y) * 10.0
            # Top
            if y + r > 1:
                forces[i, 1] -= (y + r - 1) * 10.0
                
        # Apply forces
        centers += forces * lr
        
        # Ensure centers stay within [0, 1] roughly (project back)
        centers = np.clip(centers, 0, 1)
        
        # Decay learning rate
        if step % 100 == 0 and step > 0:
            lr *= 0.9

    return centers, radii
