import numpy as np
from scipy.optimize import minimize
import math

def compute_objective(vars, n, penalty_weight):
    centers = vars[:2*n].reshape(n, 2)
    radii = vars[2*n:]
    
    obj = -np.sum(radii)
    penalty = 0.0
    
    # Boundary constraint penalties
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if r > x: penalty += penalty_weight * (r - x)**2
        if x + r > 1: penalty += penalty_weight * (x + r - 1)**2
        if r > y: penalty += penalty_weight * (r - y)**2
        if y + r > 1: penalty += penalty_weight * (y + r - 1)**2
        
    # Overlap constraint penalties
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = math.hypot(dx, dy)
            min_dist = radii[i] + radii[j]
            if min_dist > dist:
                penalty += penalty_weight * (min_dist - dist)**2
                
    return obj + penalty

def force_relax(centers, radii, n, steps=1000):
    centers = centers.copy()
    radii = radii.copy()
    dt = 0.002
    for step in range(steps):
        forces = np.zeros_like(centers)
        decay = 1.0 - step / steps
        for i in range(n):
            r = radii[i]
            # Boundary repulsion
            if centers[i,0] < r: forces[i,0] += (r - centers[i,0])
            if centers[i,0] > 1-r: forces[i,0] -= (centers[i,0] - (1-r))
            if centers[i,1] < r: forces[i,1] += (r - centers[i,1])
            if centers[i,1] > 1-r: forces[i,1] -= (centers[i,1] - (1-r))
            
            for j in range(i+1, n):
                dx = centers[i,0] - centers[j,0]
                dy = centers[i,1] - centers[j,1]
                dist_sq = dx*dx + dy*dy
                dist = math.sqrt(dist_sq)
                min_dist = radii[i] + radii[j]
                if dist < min_dist and dist > 1e-6:
                    overlap = min_dist - dist
                    fx = overlap * dx / dist
                    fy = overlap * dy / dist
                    forces[i,0] += fx
                    forces[i,1] += fy
                    forces[j,0] -= fx
                    forces[j,1] -= fy
        centers += dt * decay * forces
        centers[:,0] = np.clip(centers[:,0], radii, 1-radii)
        centers[:,1] = np.clip(centers[:,1], radii, 1-radii)
    return centers

def run_optimization(centers, radii, n):
    vars0 = np.concatenate([centers.flatten(), radii])
    bounds = [(0, 1), (0, 1), (0, 0.45)] * n
    
    # Phased optimization with increasing penalty weights
    weights = [500.0, 5000.0, 20000.0]
    for w in weights:
        res = minimize(compute_objective, vars0, args=(n, w), method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': 2000, 'ftol': 1e-12})
        vars0 = res.x
        centers = vars0[:2*n].reshape(n, 2)
        radii = vars0[2*n:]
                   
    return centers, radii, np.sum(radii)
