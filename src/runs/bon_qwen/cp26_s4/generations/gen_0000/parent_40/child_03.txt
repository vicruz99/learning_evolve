

To maximize the sum of radii for 26 circles in a unit square, we use a two-phase strategy: first, constructing a high-density hexagonal initial layout, and second, refining that layout using a constrained non-linear optimization.

### Strategy

1.  **Hexagonal Initialization:** We place 26 circles in a staggered hexagonal grid. This is the densest known arrangement in a plane. We calculate a baseline radius that fits 26 circles within the unit square, accounting for row shifts.
2.  **Non-Linear Optimization (SLSQP):** Using `scipy.optimize.minimize`, we treat the sum of radii as the objective to maximize. We enforce boundary constraints (circles must remain within $[0, 1] \times [0, 1]$) and pairwise distance constraints (circles must not overlap).
3.  **Refinement:** To ensure the optimizer escapes local minima, we perform multiple optimization runs with perturbed initial states and select the configuration with the highest total radius.

import numpy as np
from scipy.optimize import minimize

def run_packing():
    np.random.seed(42)
    n_circles = 26
    best_sum_r = 0.0
    best_centers = None
    best_radii = None

    def objective(vars):
        return -np.sum(vars[2::3])

    def constraints(vars, n):
        c = []
        for i in range(n):
            x, y, r = vars[3*i], vars[3*i+1], vars[3*i+2]
            c.append({'type': 'ineq', 'fun': lambda v, idx=i: v[3*idx] - v[3*idx+2]})
            c.append({'type': 'ineq', 'fun': lambda v, idx=i: 1.0 - v[3*idx] - v[3*idx+2]})
            c.append({'type': 'ineq', 'fun': lambda v, idx=i: v[3*idx+1] - v[3*idx+2]})
            c.append({'type': 'ineq', 'fun': lambda v, idx=i: 1.0 - v[3*idx+1] - v[3*idx+2]})
        
        for i in range(n):
            for j in range(i + 1, n):
                c.append({
                    'type': 'ineq',
                    'fun': lambda v, i=i, j=j: ((v[3*i]-v[3*j])**2 + (v[3*i+1]-v[3*j+1])**2) - (v[3*i+2]+v[3*j+2])**2
                })
        return c

    def get_hex_init(radius):
        centers = []
        row_x = radius
        row_y = radius
        rows = 0
        count = 0
        
        while count < n_circles and row_y + radius <= 1.0:
            curr_x = row_x
            is_even = (rows % 2 == 0)
            
            while curr_x + radius <= 1.0 and count < n_circles:
                centers.append([curr_x, row_y])
                curr_x += 2 * radius
                count += 1
            
            row_y += radius * np.sqrt(3)
            rows += 1
        
        radii = [radius] * n_circles
        return np.array(centers), np.array(radii)

    # Phase 1: Find a valid hexagonal radius
    r_guess = 0.09
    for _ in range(10):
        r_guess += 0.002
        centers, radii = get_hex_init(r_guess)
        if len(centers) >= n_circles:
            break
            
    # Phase 2: Optimization
    for trial in range(5):
        centers, radii = get_hex_init(r_guess)
        if len(centers) < n_circles:
            continue
            
        centers = centers[:n_circles]
        radii = radii[:n_circles]
        
        if trial > 0:
            centers += np.random.uniform(-0.01, 0.01, centers.shape)
            centers = np.clip(centers, 0.01, 0.99)

        vars0 = np.zeros(3 * n_circles)
        for i in range(n_circles):
            vars0[3*i] = centers[i, 0]
            vars0[3*i+1] = centers[i, 1]
            vars0[3*i+2] = radii[i]

        bounds = [(0, 1) for _ in range(3 * n_circles)]
        # Better bounds for r
        for i in range(n_circles):
            bounds[3*i+2] = (0, 0.5)

        try:
            res = minimize(
                objective, 
                vars0, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=constraints(vars0, n_circles),
                options={'maxiter': 500, 'disp': False}

                curr_sum_r = -res.fun
                if curr_sum_r > best_sum_r:
                    best_sum_r = curr_sum_r
                    best_centers = res.x[:3*n_circles:3].reshape(-1, 2)

        except Exception as e:
            continue

    if best_centers is None:
        best_centers, best_radii = get_hex_init(r_guess)
        best_sum_r = np.sum(best_radii)

    return best_centers, best_radii, best_sum_r
