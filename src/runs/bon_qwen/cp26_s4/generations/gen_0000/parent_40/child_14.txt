import numpy as np
from scipy.optimize import minimize
import math

def compute_loss_equal(params, n):
    """
    Loss function for equal radius optimization.
    Minimizes: - (n * r) + penalty
    """
    # params: [x1, y1, ..., xn, yn, r]
    centers = params[:2*n].reshape(n, 2)
    r = params[2*n]
    
    # Objective: maximize sum of radii => minimize -n*r
    obj = -n * r
    
    penalty = 0.0
    
    # Boundary penalties
    for i in range(n):
        x, y = centers[i]
        
        # Left boundary: x >= r
        diff = r - x
        if diff > 0: penalty += diff**2
        
        # Right boundary: x + r <= 1
        diff = x + r - 1
        if diff > 0: penalty += diff**2
        
        # Bottom boundary: y >= r
        diff = r - y
        if diff > 0: penalty += diff**2
        
        # Top boundary: y + r <= 1
        diff = y + r - 1
        if diff > 0: penalty += diff**2
        
    # Overlap penalties
    # Distance between centers >= 2r
    for i in range(n):
        xi, yi = centers[i]
        for j in range(i+1, n):
            xj, yj = centers[j]
            dx = xi - xj
            dy = yi - yj
            dist = math.sqrt(dx*dx + dy*dy)
            min_d = 2 * r
            if dist < min_d:
                diff = min_d - dist
                penalty += diff**2
                
    # Weight for penalty term
    penalty_weight = 1000.0
    return obj + penalty_weight * penalty

def compute_loss_variable(params, n):
    """
    Loss function for variable radius optimization.
    Minimizes: - sum(radii) + penalty
    """
    # params: [x1, y1, r1, ..., xn, yn, rn]
    centers = params[:2*n].reshape(n, 2)
    radii = params[2*n:]
    
    obj = -np.sum(radii)
    penalty = 0.0
    
    # Boundary penalties
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        
        # Left boundary: x >= r
        diff = r - x
        if diff > 0: penalty += diff**2
        
        # Right boundary: x + r <= 1
        diff = x + r - 1
        if diff > 0: penalty += diff**2
        
        # Bottom boundary: y >= r
        diff = r - y
        if diff > 0: penalty += diff**2
        
        # Top boundary: y + r <= 1
        diff = y + r - 1
        if diff > 0: penalty += diff**2
        
    # Overlap penalties
    # Distance >= r_i + r_j
    for i in range(n):
        xi, yi = centers[i]
        ri = radii[i]
        for j in range(i+1, n):
            xj, yj = centers[j]
            rj = radii[j]
            dx = xi - xj
            dy = yi - yj
            dist = math.sqrt(dx*dx + dy*dy)
            min_d = ri + rj
            if dist < min_d:
                diff = min_d - dist
                penalty += diff**2
                
    penalty_weight = 1000.0
    return obj + penalty_weight * penalty
