import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26
TRIU_IDX = np.triu_indices(N_CIRCLES, k=1)

def compute_loss(vars_flat, mu):
    centers = vars_flat[:2*N_CIRCLES].reshape(N_CIRCLES, 2)
    radii = vars_flat[2*N_CIRCLES:]
    
    # Pairwise distances and overlap penalty
    diff = centers[:, None, :] - centers[None, :, :]
    dist = np.sqrt(np.sum(diff**2, axis=2) + 1e-12)
    r_sum = radii[:, None] + radii[None, :]
    overlap = np.maximum(0.0, r_sum - dist)
    p_overlap = np.sum(overlap[TRIU_IDX]**2)
    
    # Boundary penalties (left, right, bottom, top)
    x, y = centers[:, 0], centers[:, 1]
    p_bound = np.sum(
        np.maximum(0.0, radii - x)**2 + 
        np.maximum(0.0, radii - (1.0 - x))**2 + 
        np.maximum(0.0, radii - y)**2 + 
        np.maximum(0.0, radii - (1.0 - y))**2
    )
                     
    return -np.sum(radii) + mu * (p_overlap + p_bound)

def run_packing():
    bounds = [(0.0, 1.0)] * (2*N_CIRCLES) + [(0.0, 0.5)] * N_CIRCLES
    best_sum = 0.0
    best_res = (np.zeros((N_CIRCLES, 2)), np.zeros(N_CIRCLES))
    
    rng = np.random.default_rng(42)
    
    for trial in range(50):
        # Initialize on a perturbed hexagonal grid
        centers = []
        for i in range(6):
            row_n = 5 if i % 2 == 0 else 4
            for j in range(row_n):
                cx = 0.05 + j * 0.20
                cy = 0.05 + i * 0.16
                if i % 2 != 0: 
                    cx += 0.10
                centers.append([cx, cy])
        centers = np.array(centers[:N_CIRCLES])
        centers += rng.uniform(-0.015, 0.015, size=(N_CIRCLES, 2))
        centers = np.clip(centers, 0.05, 0.95)
        
        radii = np.ones(N_CIRCLES) * 0.08
        x0 = np.concatenate([centers.flatten(), radii])
        
        curr = x0
        mu = 20.0
        for step in range(9):
            res = minimize(compute_loss, curr, args=(mu,), method='L-BFGS-B', 
                           bounds=bounds, options={'maxiter': 200, 'ftol': 1e-10})
            curr = res.x
            mu *= 5.0
            
        c = curr[:2*N_CIRCLES].reshape(N_CIRCLES, 2)
        r = curr[2*N_CIRCLES:]
        s = np.sum(r)
        
        if s > best_sum:
            best_sum = s
            best_res = (c.copy(), r.copy())
            
    return best_res[0], best_res[1], best_sum
