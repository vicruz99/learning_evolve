import numpy as np
from scipy.optimize import minimize
import math

def calculate_overlap_energy(centers, radii, n):
    """
    Calculates a penalty energy based on overlaps and boundary violations.
    Lower energy is better. 0 means valid packing.
    """
    energy = 0.0
    
    # Boundary violations
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        
        # Left
        if x - r < 0:
            energy += (x - r) ** 2
        # Right
        if x + r > 1:
            energy += (x + r - 1) ** 2
        # Bottom
        if y - r < 0:
            energy += (y - r) ** 2
        # Top
        if y + r > 1:
            energy += (y + r - 1) ** 2

    # Pairwise overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = math.sqrt((centers[i, 0] - centers[j, 0])**2 + 
                             (centers[i, 1] - centers[j, 1])**2)
            min_dist = radii[i] + radii[j]
            if dist < min_dist:
                energy += (min_dist - dist) ** 2
                
    return energy

def relaxation_step(centers, radii, n, max_iter=500, tol=1e-9):
    """
    Adjusts centers to minimize overlap energy for fixed radii.
    """
    # Flatten centers for optimizer
    x0 = centers.flatten()
    
    def objective(x):
        c = x.reshape((n, 2))
        return calculate_overlap_energy(c, radii, n)

    # Use L-BFGS-B as it handles bound constraints well, though we rely on penalty
    # Actually, since we have a penalty function, standard BFGS or L-BFGS-B is fine.
    # We don't have hard bounds in the objective, but centers should stay reasonable.
    res = minimize(objective, x0, method='L-BFGS-B', options={'maxiter': max_iter, 'ftol': tol})
    
    optimal_centers = res.x.reshape((n, 2))
    return optimal_centers, res.fun
