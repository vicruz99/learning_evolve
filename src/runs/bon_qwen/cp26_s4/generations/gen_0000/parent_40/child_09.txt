

Looking at this circle packing problem, I need to maximize the sum of radii of 26 circles in a unit square.

   - Objective: minimize(-sum_radii + penalty_coeff * constraint_violations)

import numpy as np
from scipy.optimize import minimize
import math


def compute_penalty_and_gradient(centers, radii, penalty_coeff):
    n = len(radii)
    
    penalty = -np.sum(radii)
    grad = np.zeros(2 * n + n)
    grad_r = -np.ones(n)
    
    # Boundary penalties
    for i in range(n):
        r = radii[i]
        x, y = centers[i]
        
        # Left
        if x < r:
            violation = r - x
            penalty += penalty_coeff * violation ** 2
            grad[2 * i] += -2 * penalty_coeff * violation
            grad[2 * n + i] += 2 * penalty_coeff * violation
        # Right
        if x + r > 1:
            violation = x + r - 1
            penalty += penalty_coeff * violation ** 2
            grad[2 * i] += 2 * penalty_coeff * violation
            grad[2 * n + i] += 2 * penalty_coeff * violation
        # Bottom
        if y < r:
            violation = r - y
            penalty += penalty_coeff * violation ** 2
            grad[2 * i + 1] += -2 * penalty_coeff * violation
            grad[2 * n + i] += 2 * penalty_coeff * violation
        # Top
        if y + r > 1:
            violation = y + r - 1
            penalty += penalty_coeff * violation ** 2
            grad[2 * i + 1] += 2 * penalty_coeff * violation
            grad[2 * n + i] += 2 * penalty_coeff * violation
    
    # Overlap penalties
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = math.sqrt(dx * dx + dy * dy)
            if dist < 1e-12:
                dist = 1e-12
                dx, dy = 1e-12, 0
            
            r_sum = radii[i] + radii[j]
            if r_sum > dist:
                violation = r_sum - dist
                penalty += penalty_coeff * violation ** 2
                
                factor = 2 * penalty_coeff * violation / dist
                
                grad[2 * i] += factor * dx
                grad[2 * i + 1] += factor * dy
                grad[2 * j] -= factor * dx
                grad[2 * j + 1] -= factor * dy
                
                grad_r[i] += 2 * penalty_coeff * violation
                grad_r[j] += 2 * penalty_coeff * violation
    
    grad[2 * n:] = grad_r
    
    return penalty, grad


def hex_init(n):
    centers = []
    radii = []
    r = 0.1
    patterns = [6, 5, 6, 5, 4]
    
    for row, count in enumerate(patterns):
        y = r + row * r * math.sqrt(3)
        for col in range(count):
            x = r + col * 2 * r
            if row % 2 == 1:
                x += r
            centers.append([x, y])
            radii.append(r)
    
    return np.array(centers), np.array(radii)


def random_init(n):
    centers = np.random.rand(n, 2) * 0.8 + 0.1
    radii = np.random.rand(n) * 0.08 + 0.02
    return centers, radii


def grid_init(n):
    centers = []
    radii = []
    r = 0.09
    rows = 5
    cols = 6
    
    circle_count = 0
    for row in range(rows):
        for col in range(cols):
            if circle_count >= n:
                break
            x = r + col * 2 * r
            y = r + row * 2 * r
            if x + r <= 1 and y + r <= 1:
                centers.append([x, y])
                radii.append(r)
                circle_count += 1
        if circle_count >= n:
            break
    
    while len(centers) < n:
        r_new = 0.05
        centers.append([r_new + np.random.rand() * (1 - 2 * r_new), r_new + np.random.rand() * (1 - 2 * r_new)])
        radii.append(r_new)
    
    return np.array(centers[:n]), np.array(radii[:n])


def run_packing():
    n = 26
    best_centers = None
    best_radii = None
    best_sum = -1
    
    np.random.seed(42)
    
    init_funcs = [hex_init, random_init, grid_init]
    
    for restart in range(40):
        init_idx = restart % 3
        centers, radii = init_funcs[init_idx](n)
        
        centers += np.random.randn(*centers.shape) * 0.005
        radii += np.abs(np.random.randn(*radii.shape)) * 0.003
        radii = np.clip(radii, 0.01, 0.5)
        centers = np.clip(centers, 0.01, 0.99)
        
        x0 = np.concatenate([centers.flatten(), radii])
        
        coeffs = [50, 200, 1000, 5000, 20000, 80000]
        current_x = x0.copy()
        
        for pc in coeffs:
            def obj(x):
                c = x[:2 * n].reshape((n, 2))
                r = x[2 * n:]
                return compute_penalty_and_gradient(c, r, pc)[0]
            
            def grad(x):
                c = x[:2 * n].reshape((n, 2))
                r = x[2 * n:]
                return compute_penalty_and_gradient(c, r, pc)[1]
            
            result = minimize(
                obj,
                current_x,
                jac=grad,
                method='L-BFGS-B',
                bounds=[(0.01, 0.99)] * (2 * n) + [(0.01, 0.5)] * n,
                options={'maxiter': 1500, 'ftol': 1e-14, 'gtol': 1e-10}
            )
            current_x = result.x.copy()
        
        opt_centers = current_x[:2 * n].reshape((n, 2))
        opt_radii = current_x[2 * n:]
        
        opt_radii = np.maximum(opt_radii, 0.0)
        opt_centers = np.clip(opt_centers, 0.0, 1.0)
        
        if validate_packing(opt_centers, opt_radii):
            current_sum = np.sum(opt_radii)
            if current_sum > best_sum:
                best_centers = opt_centers.copy()
                best_radii = opt_radii.copy()
                best_sum = current_sum
        else:
            # Try to repair by shrinking slightly
            for shrink_factor in [0.99, 0.98, 0.95, 0.9]:
                repaired_radii = opt_radii * shrink_factor
                repaired_centers = opt_centers * shrink_factor + 0.5 * (1 - shrink_factor)
                if validate_packing(repaired_centers, repaired_radii):
                    current_sum = np.sum(repaired_radii)
                    if current_sum > best_sum:
                        best_centers = repaired_centers.copy()
                        best_radii = repaired_radii.copy()
                        best_sum = current_sum
                    break
    
    # Final refinement on best solution
    if best_centers is not None:
        x0 = np.concatenate([best_centers.flatten(), best_radii])
        for pc in [1000, 5000, 20000, 100000]:
            def obj(x):
                c = x[:2 * n].reshape((n, 2))
                r = x[2 * n:]
                return compute_penalty_and_gradient(c, r, pc)[0]
            
            def grad(x):
                c = x[:2 * n].reshape((n, 2))
                r = x[2 * n:]
                return compute_penalty_and_gradient(c, r, pc)[1]
            
            result = minimize(
                obj,
                x0,
                jac=grad,
                method='L-BFGS-B',
                bounds=[(0.01, 0.99)] * (2 * n) + [(0.01, 0.5)] * n,
                options={'maxiter': 2000, 'ftol': 1e-15}
            )
            x0 = result.x.copy()
        
        opt_centers = x0[:2 * n].reshape((n, 2))
        opt_radii = x0[2 * n:]
        opt_radii = np.maximum(opt_radii, 0.0)
        
        if validate_packing(opt_centers, opt_radii):
            best_centers = opt_centers
            best_radii = opt_radii
            best_sum = np.sum(opt_radii)
    
    return best_centers, best_radii, best_sum


def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    n = centers.shape[0]

    if np.isnan(centers).any():
        print("NaN values detected in circle centers")
        return False

    if np.isnan(radii).any():
        print("NaN values detected in circle radii")
        return False

    for i in range(n):
        if radii[i] < 0:
            print(f"Circle {i} has negative radius {radii[i]}")
            return False
        elif np.isnan(radii[i]):
            print(f"Circle {i} has nan radius")
            return False

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            print(f"Circle {i} at ({x}, {y}) with radius {r} is outside the unit square")
            return False

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                print(f"Circles {i} and {j} overlap: dist={dist}, r1+r2={radii[i]+radii[j]}")
                return False

    return True
