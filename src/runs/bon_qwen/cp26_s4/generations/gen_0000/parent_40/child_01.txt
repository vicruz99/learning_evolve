import numpy as np
from scipy.optimize import minimize

# Constants
N_CIRCLES = 26
LAMBDA_P = 10000.0

def compute_cost_and_grad(v):
    """
    Computes the cost and gradient for the circle packing optimization.
    Cost = -sum(radii) + Penalty(overlaps) + Penalty(boundaries)
    
    Args:
        v: 1D array of size 3*N containing [x1, y1, r1, x2, y2, r2, ...]
        
    Returns:
        cost: scalar float
        grad: 1D array of size 3*N
    """
    n = v.shape[0] // 3
    xs = v[0::3]
    ys = v[1::3]
    rs = v[2::3]
    
    # Base objective: maximize sum of radii -> minimize -sum(radii)
    cost = -np.sum(rs)
    grad = np.zeros_like(v)
    grad[2::3] -= 1.0  # Gradient of -r is -1
    
    # --- Pairwise Overlap Constraints ---
    # dx[i, j] = x_i - x_j
    dx = xs[:, None] - xs[None, :] 
    dy = ys[:, None] - ys[None, :]
    d2 = dx**2 + dy**2
    
    # rs_sum[i, j] = r_i + r_j
    rs_sum = rs[:, None] + rs[None, :] 
    rs_sum2 = rs_sum**2
    
    # Gap = (r_i + r_j)^2 - dist^2
    # Positive gap indicates overlap
    gap = rs_sum2 - d2
    
    # Ignore self-interaction on diagonal
    np.fill_diagonal(gap, 0.0)
    
    # Only consider overlaps (gap > 0)
    overlap = gap > 0
    g = np.where(overlap, gap, 0.0)
    
    # Add penalty cost: sum(gap^2)
    penalty_cost = np.sum(g**2)
    cost += LAMBDA_P * penalty_cost
    
    # --- Gradients for Pairwise Constraints ---
    # d(Penalty)/d(x_i) = sum_j 2 * lambda * g_ij * (-2 * dx_ij) = -4 * lambda * sum(g_ij * dx_ij)
    grad_x = -4.0 * LAMBDA_P * np.sum(g * dx, axis=1)
    grad_y = -4.0 * LAMBDA_P * np.sum(g * dy, axis=1)
    
    # d(Penalty)/d(r_i) = sum_j 2 * lambda * g_ij * (2 * (r_i + r_j)) = 4 * lambda * sum(g_ij * rs_sum_ij)
    grad_r_pair = 4.0 * LAMBDA_P * np.sum(g * rs_sum, axis=1)
    
    grad[0::3] += grad_x
    grad[1::3] += grad_y
    grad[2::3] += grad_r_pair
    
    # --- Boundary Constraints ---
    # 1. r - x > 0 (Left wall)
    v1 = rs - xs
    mask1 = v1 > 0
    if np.any(mask1):
        term1 = 2.0 * LAMBDA_P * v1
        grad[0::3] -= term1 
        grad[2::3] += term1 
        cost += LAMBDA_P * np.sum(np.where(mask1, v1**2, 0.0))

    # 2. r + x - 1 > 0 (Right wall)
    v2 = rs + xs - 1.0
    mask2 = v2 > 0
    if np.any(mask2):
        term2 = 2.0 * LAMBDA_P * v2
        grad[0::3] += term2 
        grad[2::3] += term2
        cost += LAMBDA_P * np.sum(np.where(mask2, v2**2, 0.0))

    # 3. r - y > 0 (Bottom wall)
    v3 = rs - ys
    mask3 = v3 > 0
    if np.any(mask3):
        term3 = 2.0 * LAMBDA_P * v3
        grad[1::3] -= term3
        grad[2::3] += term3
        cost += LAMBDA_P * np.sum(np.where(mask3, v3**2, 0.0))

    # 4. r + y - 1 > 0 (Top wall)
    v4 = rs + ys - 1.0
    mask4 = v4 > 0
    if np.any(mask4):
        term4 = 2.0 * LAMBDA_P * v4
        grad[1::3] += term4
        grad[2::3] += term4
        cost += LAMBDA_P * np.sum(np.where(mask4, v4**2, 0.0))
            
    return cost, grad

def init_random(n):
    """Initialize with random positions and small radii."""
    xs = np.random.rand(n) * 0.6 + 0.2
    ys = np.random.rand(n) * 0.6 + 0.2
    rs = np.full(n, 0.05)
    
    v = np.zeros(3 * n)
    v[0::3] = xs
    v[1::3] = ys
    v[2::3] = rs
    return v

def init_hexagonal(n):
    """Initialize with a hexagonal grid pattern."""
    r = 0.08
    dx = 2 * r
    dy = np.sqrt(3) * r
    pts = []
    y = r
    row = 0
    while y < 1 - r:
        x = r
        if row % 2 == 1:
            x += dx / 2
        while x < 1 - r:
            pts.append([x, y])
            x += dx
        y += dy
        row += 1
    
    pts = np.array(pts)
    if len(pts) < n:
        while len(pts) < n:
            pts = np.vstack([pts, [np.random.rand() * 0.6 + 0.2, np.random.rand() * 0.6 + 0.2]])
    else:
        np.random.shuffle(pts)
        pts = pts[:n]
        
    xs = pts[:, 0]
    ys = pts[:, 1]
    rs = np.full(n, r)
    
    v = np.zeros(3 * n)
    v[0::3] = xs
    v[1::3] = ys
    v[2::3] = rs
    return v

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Runs the optimization to pack 26 circles.
    Returns centers, radii, and sum of radii.
    """
    best_cost = float('inf')
    best_v = None
    
    # Bounds for variables: x, y in [0, 1], r in [0, 0.5]
    bounds = [(0, 1)] * (2 * N_CIRCLES) + [(0, 0.5)] * N_CIRCLES
    
    # Run multiple restarts to find global optimum
    # Mix of random and hexagonal initializations
    num_restarts = 60
    
    for i in range(num_restarts):
        if i % 2 == 0:
            v0 = init_hexagonal(N_CIRCLES)
        else:
            v0 = init_random(N_CIRCLES)
            
        try:
            res = minimize(
                compute_cost_and_grad, 
                v0, 
                jac=True, 
                method='L-BFGS-B', 
                bounds=bounds, 
                options={'maxiter': 2000, 'ftol': 1e-12}
            )
            
            # Track best result based on optimization cost
            if res.fun < best_cost:
                best_cost = res.fun
                best_v = res.x.copy()
                
        except Exception:
            continue
            
    if best_v is not None:
        centers = np.zeros((N_CIRCLES, 2))
        centers[:, 0] = best_v[0::3]
        centers[:, 1] = best_v[1::3]
        radii = best_v[2::3]
        return centers, radii, np.sum(radii)
    else:
        # Fallback if optimization fails completely
        return np.zeros((N_CIRCLES, 2)), np.zeros(N_CIRCLES), 0.0
