def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    n = centers.shape[0]

    # Check for NaN values
    if np.isnan(centers).any():
        return False
    if np.isnan(radii).any():
        return False

    # Check if radii are nonnegative
    for i in range(n):
        if radii[i] < 0:
            return False
        elif np.isnan(radii[i]):
            return False

    # Check if circles are inside the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            return False

    # Check for overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False

    return True

def packing_penalty_and_neg_sum(x, n):
    """
    Calculates the objective function: -sum(radii) + penalty.
    x contains flattened centers (2*n) and radii (n).
    """

    obj = -np.sum(radii)
    
    penalty = 0.0

    # diff shape (n, n, 2)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]

    r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    
    # Overlap amount (positive if overlapping)
    overlap = r_sum - dists
    # Only consider upper triangle to avoid double counting and self
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    overlaps_upper = overlap[mask]
    
    # Quadratic penalty for overlaps
    if np.any(overlaps_upper > 0):
        penalty += np.sum(overlaps_upper[overlaps_upper > 0]**2) * 1000.0

    violations += np.maximum(0, centers[:, 1] + radii - 1)
    
    if np.any(violations > 0):
        penalty += np.sum(violations**2) * 1000.0
        
    return obj + penalty
