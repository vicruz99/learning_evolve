     - Boundary: $r_i \le x_i \le 1-r_i$, $r_i \le y_i \le 1-r_i$

2. **Constraint Handling**: Use squared distances for overlap constraints ($\|c_i-c_j\|^2 \ge (r_i+r_j)^2$) to avoid square roots, improving numerical stability and gradient computation. Vectorize constraints for efficiency.

import numpy as np
from scipy.optimize import minimize

def obj_func(x, n):
    # Objective: minimize negative sum of radii -> maximize sum of radii
    return -np.sum(x.reshape(n, 3)[:, 2])

def con_bounds_func(x, n):
    # Boundary constraints: circles must be inside [0,1]x[0,1]
    pts = x.reshape(n, 3)
    return np.concatenate([
        pts[:, 0] - pts[:, 2],          # x - r >= 0
        1 - pts[:, 0] - pts[:, 2],      # x + r <= 1
        pts[:, 1] - pts[:, 2],          # y - r >= 0
        1 - pts[:, 1] - pts[:, 2]       # y + r <= 1
    ])

def con_overlap_func(x, n):
    # Overlap constraints: squared distance >= (r_i + r_j)^2
    pts = x.reshape(n, 3)
    c = pts[:, :2]
    r = pts[:, 2]
    diff = c[:, None, :] - c[None, :, :]
    dists_sq = np.sum(diff**2, axis=2)
    idx = np.triu_indices(n, k=1)
    return dists_sq[idx] - (r[idx[0]] + r[idx[1]])**2

def generate_initial_guess(n):
    # Generate a grid-based initial configuration
    rows = 5
    cols = 6
    ys = np.linspace(0.2, 0.8, rows)
    xs = np.linspace(0.2, 0.8, cols)
    centers = []
    for y in ys:
        for x in xs:
            centers.append([x, y])
            if len(centers) >= n:
                break
        if len(centers) >= n:
            break
    centers = np.array(centers[:n])
    # Add small random perturbation to break symmetry and help optimization
    centers += np.random.uniform(-0.03, 0.03, centers.shape)
    centers = np.clip(centers, 0.1, 0.9)
    
    x0 = np.zeros(n * 3)
    x0[:, 0] = centers[:, 0]
    x0[:, 1] = centers[:, 1]
    x0[:, 2] = 0.02  # Small initial radius to ensure feasibility
    return x0

def run_packing():
    np.random.seed(42)  # For reproducibility
    n = 26
    bounds = [(0, 1), (0, 1), (0, 0.5)] * n
    
    constraints = [
        {'type': 'ineq', 'fun': con_bounds_func, 'args': (n,)},
        {'type': 'ineq', 'fun': con_overlap_func, 'args': (n,)}
    ]
    
    best_x = None
    best_obj = np.inf
    
    # Multiple restarts to avoid local minima
    for trial in range(10):
        x0 = generate_initial_guess(n)
        res = minimize(obj_func, x0, method='SLSQP', args=(n,), bounds=bounds,
                       constraints=constraints, options={'maxiter': 1500, 'ftol': 1e-9})
        if res.fun < best_obj:
            best_obj = res.fun
            best_x = res.x
            
    centers = best_x.reshape(n, 3)[:, :2]
    radii = best_x.reshape(n, 3)[:, 2]
    
    # Post-processing: ensure strict feasibility within validator tolerance
    # The optimizer might produce values slightly violating constraints due to numerical precision
    try:
        if not validate_packing(centers, radii):
            scale = 0.999
            while scale > 0.5:
                if validate_packing(centers, radii * scale):
                    radii *= scale
                    break
                scale -= 0.001
    except NameError:
        # Fallback if validate_packing is not in scope during local testing
        radii *= 0.995
        
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
