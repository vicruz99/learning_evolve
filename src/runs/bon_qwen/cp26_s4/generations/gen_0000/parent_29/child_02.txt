import numpy as np
from scipy.optimize import minimize, NonlinearConstraint

N = 26

def objective_func(x):
    """Objective: minimize negative sum of radii (equivalent to maximizing sum of radii)"""
    return -np.sum(x[2::3])

def constraint_func(x):
    """
    Returns constraint values that must be <= 0.
    Includes boundary constraints and pairwise non-overlap constraints.
    """
    r = x[2::3]
    cx = x[0::3]
    cy = x[1::3]
    
    c_list = []
    # Boundary constraints: r - x <= 0, r + x - 1 <= 0, same for y
    c_list.append(r - cx)
    c_list.append(r + cx - 1.0)
    c_list.append(r - cy)
    c_list.append(r + cy - 1.0)
    
    # Overlap constraints: (ri + rj)^2 - dist^2 <= 0
    for i in range(N):
        for j in range(i + 1, N):
            dx = cx[i] - cx[j]
            dy = cy[i] - cy[j]
            c_list.append((r[i] + r[j])**2 - (dx*dx + dy*dy))
            
    return np.concatenate(c_list)

def run_packing():
    np.random.seed(42)
    
    # Initial guess: 5x5 grid + 1 center circle
    cx0, cy0, r0 = [], [], []
    for i in range(5):
        for j in range(5):
            cx0.append(0.125 + i * 0.25)
            cy0.append(0.125 + j * 0.25)
            r0.append(0.09)
    # 26th circle
    cx0.append(0.5)
    cy0.append(0.5)
    r0.append(0.04)
    
    x0 = np.zeros(3 * N)
    x0[0::3] = cx0
    x0[1::3] = cy0
    x0[2::3] = r0
    
    # Small perturbation to break symmetry and help optimizer escape local minima
    perturbation = np.random.uniform(-0.002, 0.002, size=len(x0))
    perturbation[2::3] = np.abs(perturbation[2::3])  # Keep radii positive
    x0 += perturbation
    
    # Ensure initial boundary feasibility
    x0[0::3] = np.clip(x0[0::3], x0[2::3], 1.0 - x0[2::3])
    x0[1::3] = np.clip(x0[1::3], x0[2::3], 1.0 - x0[2::3])
    
    bounds = [(0.0, 1.0)] * (2 * N) + [(0.0, 1.0)] * N
    
    cons = NonlinearConstraint(constraint_func, -np.inf, 0.0)
    
    res = minimize(
        objective_func,
        x0,
        method='trust-constr',
        bounds=bounds,
        constraints=[cons],
        options={'maxiter': 2000, 'verbose': 0}
    )
    
    cx_opt = res.x[0::3]
    cy_opt = res.x[1::3]
    r_opt = res.x[2::3]
    
    # Safety clamp for numerical stability to guarantee validity
    r_opt = np.maximum(r_opt, 0.0)
    cx_opt = np.clip(cx_opt, r_opt, 1.0 - r_opt)
    cy_opt = np.clip(cy_opt, r_opt, 1.0 - r_opt)
    
    centers = np.column_stack((cx_opt, cy_opt))
    radii = r_opt
    
    return centers, radii, np.sum(radii)
