import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def compute_objective(vars):
    """Objective function: minimize negative sum of radii"""
    r = vars[2::3]
    return -np.sum(r)

def compute_constraints(vars):
    """Inequality constraints: boundary containment and non-overlap"""
    x = vars[0::3]
    y = vars[1::3]
    r = vars[2::3]
    n = N_CIRCLES
    
    con_list = []
    # Boundary constraints: x >= r, 1-x >= r, y >= r, 1-y >= r
    con_list.append(x - r)
    con_list.append(1.0 - x - r)
    con_list.append(y - r)
    con_list.append(1.0 - y - r)
    
    # Non-overlap constraints: dist^2 >= (r_i + r_j)^2
    for i in range(n):
        dx = x[i] - x[i+1:]
        dy = y[i] - y[i+1:]
        dr = r[i] + r[i+1:]
        con_list.append(dx**2 + dy**2 - dr**2)
        
    return np.concatenate(con_list)

def run_packing():
    n = N_CIRCLES
    rng = np.random.default_rng(42)
    
    # Initialize with a hexagonal-like grid pattern
    centers = np.zeros((n, 2))
    idx = 0
    for row in range(5):
        for col in range(6):
            if idx >= n: 
                break
            x = col * 0.18 + 0.1 + (0.09 if row % 2 == 1 else 0.0)
            y = row * 0.22 + 0.1
            centers[idx] = [x, y]
            idx += 1
            
    # Add small perturbation to break symmetry and avoid flat gradients
    centers += rng.uniform(-0.01, 0.01, size=centers.shape)
    # Ensure initial feasibility
    centers = np.clip(centers, 0.05, 0.95)
    
    # Flatten initial guess: [x1, y1, r1, x2, y2, r2, ...]
    x0 = np.zeros(3 * n)
    for i in range(n):
        x0[3*i] = centers[i, 0]
        x0[3*i+1] = centers[i, 1]
        x0[3*i+2] = 0.08  # Initial radius guess
        
    # Variable bounds: x,y in [0,1], r in [0, 0.5]
    bounds = [(0.0, 1.0)] * (2*n) + [(0.0, 0.5)] * n
    cons = {'type': 'ineq', 'fun': compute_constraints}
    
    # Run SLSQP optimizer
    res = minimize(compute_objective, x0, method='SLSQP', bounds=bounds, constraints=cons, 
                   options={'maxiter': 5000, 'ftol': 1e-12})
                   
    x_opt = res.x
    
    # Reshape results
    centers_opt = np.column_stack((x_opt[0::3], x_opt[1::3]))
    radii_opt = x_opt[2::3]
    
    return centers_opt, radii_opt, float(np.sum(radii_opt))
