from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    Returns (centers, radii, sum_radii).

    best_radii = None

    # Rows: 5, 4, 5, 4, 5, 3 (Total 26)
    initial_positions = []
    r_init = 0.09

        # Shift alternate rows for hexagonal packing

        # Adjust x_offset to center the row in the unit square
        row_width = (count - 1) * 2 * r_init + 2 * r_init
        x_start = (1.0 - row_width) / 2.0 + r_init
        
        for j in range(count):
            x = x_start + j * 2 * r_init
            initial_positions.append([x, y_curr])
        
        y_curr += r_init * np.sqrt(3)
        
    centers = np.array(initial_positions)
    radii = np.full(n, r_init)
    best_sum = np.sum(radii)

        """Objective to minimize: negative sum of radii."""
        r = theta[2*n:]
        return -np.sum(r)

        Returns constraints:

        2. Boundary constraints

        c = theta[:2*n].reshape(n, 2)

                dist_sq = np.sum((c[i] - c[j])**2)

                constraints_list.append(dist_sq - rad_sum_sq)
