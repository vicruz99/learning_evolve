from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    Optimizes the packing of 26 circles in a unit square to maximize the sum of radii.
    """

    def init_square_grid():

        coords = np.linspace(0.2, 0.8, 5)
        idx = 0

            for c in range(5):
                centers[idx] = [coords[c], coords[r]]
                idx += 1
        # Place 26th circle in the center
        centers[25] = [0.5, 0.5]
        radii = np.ones(n_circles) * 0.05 # Small initial radius
        return centers, radii

    def init_hexagonal_grid():
        centers = np.zeros((n_circles, 2))
        radii = np.ones(n_circles) * 0.05
        r_est = 0.09
        row = 0
        col = 0
        idx = 0
        # Try to fit a hexagonal pattern
        # Rows with 5 and 4 circles
        pattern = [5, 4, 5, 4, 5, 3] # Sum = 26
        y = r_est + 0.01 # Start slightly inside
        for i, count in enumerate(pattern):
            # Shift odd rows
            x_start = r_est + 0.01 + (0.5 * r_est * (i % 2))
            # Adjust x_start to fit within bounds better
            x_start = r_est + 0.01
            if i % 2 == 1:
                x_start += r_est
            
            for j in range(count):
                if idx < n_circles:
                    x = x_start + j * (2 * r_est)
                    centers[idx] = [x, y]
                    idx += 1
            y += np.sqrt(3) * r_est
        return centers, radii

    def init_random():
        centers = np.random.rand(n_circles, 2)
        # Ensure radii are feasible initially
        radii = np.zeros(n_circles)
        for i in range(n_circles):
            dist_boundary = min(centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])
            dist_others = np.min(np.linalg.norm(centers[i] - centers[:i], axis=1)) if i > 0 else 1.0
            radii[i] = min(dist_boundary, dist_others / 2) * 0.5 # Small initial
        return centers, radii

    def objective(x):
        # x contains [x1, y1, r1, x2, y2, r2, ...]
        # We want to maximize sum of radii, so minimize negative sum
        radii = x[2::3]
        return -np.sum(radii)

        radii = x[2::3]
        cons = []

        # y - r >= 0
        cons.append({'type': 'ineq', 'fun': lambda x: x[1::3] - x[2::3]})
        # 1 - y - r >= 0

        for i in range(n_circles):

    best_radii[best_radii < 1e-12] = 0

    return best_centers, best_radii, np.sum(best_radii)
