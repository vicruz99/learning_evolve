import numpy as np
from scipy.optimize import minimize

def get_violations(centers, r):
    """Calculate sum of squared constraint violations for a given radius r."""
    n = centers.shape[0]
    v = 0.0
    # Boundary violations
    for i in range(n):
        x, y = centers[i]
        v += max(0.0, r - x)**2 + max(0.0, x - (1.0 - r))**2
        v += max(0.0, r - y)**2 + max(0.0, y - (1.0 - r))**2
    # Overlap violations
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            v += max(0.0, 2.0 * r - dist)**2
    return v

def objective_func(x, r):
    """Objective function for scipy optimizer."""
    centers = x.reshape(-1, 2)
    return get_violations(centers, r)

def optimize_centers(r, n, centers_init):
    """Optimize centers for a fixed radius r."""
    x0 = centers_init.flatten()
    bounds = [(0.0, 1.0)] * (2 * n)
    res = minimize(objective_func, x0, args=(r,), method='L-BFGS-B', 
                   bounds=bounds, options={'maxiter': 4000, 'ftol': 1e-12})
    return res.x.reshape(n, 2), res.fun

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Pack 26 circles in a unit square to maximize sum of radii.
    Uses binary search on radius combined with local optimization of centers.
    """
    np.random.seed(42)
    n = 26
    
    # Initial configurations to try
    configs = []
    # 1. 5x5 grid + 1 extra circle
    c1 = []
    for i in range(5):
        for j in range(5):
            c1.append([0.1 + 0.2*i, 0.1 + 0.2*j])
    c1.append([0.5, 0.5])
    configs.append(np.array(c1))
    
    # 2. Random configuration
    c2 = np.random.rand(n, 2) * 0.8 + 0.1
    configs.append(c2)
    
    best_r = 0.09
    best_centers = configs[0]
    
    # Binary search for the maximum feasible radius
    low, high = 0.09, 0.105
    for _ in range(25):
        mid = (low + high) / 2.0
        feasible = False
        
        for cfg in configs:
            # Add small noise to escape local minima
            noisy_cfg = cfg + np.random.normal(0, 5e-5, cfg.shape)
            opt_centers, viol = optimize_centers(mid, n, noisy_cfg)
            
            if viol < 1e-7:
                feasible = True
                best_r = mid
                best_centers = opt_centers
                break
                
        if feasible:
            low = mid
        else:
            high = mid
            
    centers = best_centers
    
    # Post-processing: compute the exact maximum equal radius for the optimized centers
    # This guarantees strict validity and handles numerical tolerances
    min_r = 1.0
    for i in range(n):
        x, y = centers[i]
        min_r = min(min_r, x, 1.0 - x, y, 1.0 - y)
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            min_r = min(min_r, dist / 2.0)
            
    radii = np.full(n, min_r)
    total_sum = float(np.sum(radii))
    
    return centers, radii, total_sum
