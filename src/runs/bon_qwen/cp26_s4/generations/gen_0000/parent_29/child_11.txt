import numpy as np
from scipy.optimize import minimize, NonlinearConstraint

N_CIRCLES = 26

def objective_func(x):
    """Maximize sum of radii (minimize negative sum)"""
    return -np.sum(x[2 * N_CIRCLES:])

def constraint_func(x):
    """Nonlinear constraints: boundaries and non-overlap"""
    c = x[:2 * N_CIRCLES].reshape((N_CIRCLES, 2))
    r = x[2 * N_CIRCLES:]
    
    cons = []
    # Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
    cons.extend(c[:, 0] - r)
    cons.extend(1.0 - c[:, 0] - r)
    cons.extend(c[:, 1] - r)
    cons.extend(1.0 - c[:, 1] - r)
    
    # Overlap constraints: dist(i, j) >= r_i + r_j
    # Vectorized pairwise distances
    diff = c[:, np.newaxis, :] - c[np.newaxis, :, :]
    dist = np.sqrt(np.sum(diff**2, axis=2))
    
    # Lower triangular indices for unique pairs (i > j)
    idxs = np.tril_indices(N_CIRCLES, k=-1)
    overlaps = dist[idxs] - (r[idxs[0]] + r[idxs[1]])
    cons.extend(overlaps)
    
    return np.array(cons)

def run_packing() -> tuple:
    # 1. Initialize with a feasible grid configuration
    centers = np.zeros((N_CIRCLES, 2))
    radii = np.zeros(N_CIRCLES)
    idx = 0
    for i in range(5):
        for j in range(5):
            if idx < N_CIRCLES:
                centers[idx] = [0.1 + j * 0.2, 0.1 + i * 0.2]
                radii[idx] = 0.08  # Strictly feasible initially
                idx += 1
    if idx < N_CIRCLES:
        centers[idx] = [0.5, 0.5]
        radii[idx] = 0.05
        
    x0 = np.concatenate([centers.flatten(), radii])
    
    # 2. Setup optimization bounds and constraints
    bounds = [(0.0, 1.0)] * (2 * N_CIRCLES) + [(0.0, 0.5)] * N_CIRCLES
    constraint = NonlinearConstraint(constraint_func, 0.0, np.inf)
    
    # 3. Run SLSQP optimizer
    try:
        res = minimize(objective_func, x0, method='SLSQP', bounds=bounds, 
                       constraints=[constraint], options={'maxiter': 2000, 'ftol': 1e-9})
        best_x = res.x if res.success else x0
    except Exception:
        best_x = x0
        
    final_centers = best_x[:2 * N_CIRCLES].reshape((N_CIRCLES, 2))
    final_radii = best_x[2 * N_CIRCLES:]
    
    # 4. Safety scaling to guarantee strict feasibility against numerical tolerance
    for _ in range(300):
        valid = True
        for i in range(N_CIRCLES):
            x, y = final_centers[i]
            r = final_radii[i]

                valid = False
                break

                d = np.sqrt(np.sum((final_centers[i] - final_centers[j])**2))
                if d < final_radii[i] + final_radii[j] - 1e-7:
                    valid = False
                    break
            if not valid:
                break
        if valid:
            break
        final_radii *= 0.998  # Slightly reduce radii until valid
        
    return final_centers, final_radii, np.sum(final_radii)
