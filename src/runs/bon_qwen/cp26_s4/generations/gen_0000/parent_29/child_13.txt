import numpy as np
from scipy.optimize import linprog, minimize


def solve_radii_lp(centers, n):
    """Solve LP to find optimal radii given fixed centers."""
    A_ub = []
    b_ub = []

    # Pairwise non-overlap: ri + rj <= dist(ci, cj)
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            dist = max(dist, 1e-12)
            row = np.zeros(n)
            row[i] = 1
            row[j] = 1
            A_ub.append(row)
            b_ub.append(dist)

    # Boundary: ri <= min(xi, 1-xi, yi, 1-yi)
    for i in range(n):
        max_r = min(centers[i, 0], 1 - centers[i, 0],
                    centers[i, 1], 1 - centers[i, 1])
        max_r = max(max_r, 1e-12)
        row = np.zeros(n)
        row[i] = 1
        A_ub.append(row)
        b_ub.append(max_r)

    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    c = -np.ones(n)
    bounds = [(0, None)] * n

    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')

    if result.success:
        return result.x
    else:
        return np.ones(n) * 0.01


def compute_energy_and_grad(x, radii, n):
    """Compute repulsion energy and its gradient for position optimization."""
    c = x.reshape(n, 2)
    e = 0.0
    g = np.zeros(2 * n)

    for i in range(n):
        for j in range(i + 1, n):
            dx = c[j, 0] - c[i, 0]
            dy = c[j, 1] - c[i, 1]
            d2 = dx * dx + dy * dy
            d = np.sqrt(d2) if d2 > 1e-16 else 1e-8
            overlap = radii[i] + radii[j] - d
            if overlap > 0:
                e += overlap ** 2 * 1000.0
                force = 2.0 * overlap * 1000.0 / d
                g[2 * i] -= force * dx
                g[2 * i + 1] -= force * dy
                g[2 * j] += force * dx
                g[2 * j + 1] += force * dy

    for i in range(n):
        for coord in range(2):
            if c[i, coord] - radii[i] < 0:
                e += (c[i, coord] - radii[i]) ** 2 * 10000.0
                g[2 * i + coord] += 2.0 * 10000.0 * (c[i, coord] - radii[i])
            if c[i, coord] + radii[i] - 1 > 0:
                e += (c[i, coord] + radii[i] - 1) ** 2 * 10000.0
                g[2 * i + coord] += 2.0 * 10000.0 * (c[i, coord] + radii[i] - 1)

    return e, g


def optimize_positions(centers, radii, n):
    """Optimize positions given fixed radii using L-BFGS-B with analytical gradient."""
    def energy_wrapper(x):
        e, _ = compute_energy_and_grad(x, radii, n)
        return e

    def grad_wrapper(x):
        _, g = compute_energy_and_grad(x, radii, n)
        return g

    x0 = centers.flatten()
    bounds_pos = [(0.0, 1.0)] * (2 * n)

    best_x = x0
    best_e = float('inf')

    for restart in range(8):
        if restart > 0:
            x0_perturbed = x0 + np.random.randn(2 * n) * 0.02
            x0_perturbed = np.clip(x0_perturbed, 0.0, 1.0)
        else:
            x0_perturbed = x0

        result = minimize(energy_wrapper, x0_perturbed, jac=grad_wrapper,
                          bounds=bounds_pos, method='L-BFGS-B',
                          options={'maxiter': 2000, 'ftol': 1e-18, 'gtol': 1e-10})

        e, _ = compute_energy_and_grad(result.x, radii, n)
        if e < best_e:
            best_e = e
            best_x = result.x

    return best_x.reshape(n, 2)


def create_initial_packing(n):
    """Create initial hexagonal packing arrangement."""
    centers = np.zeros((n, 2))
    idx = 0

    row_counts = [5, 4, 5, 4, 5, 3]
    dy = 0.158
    dx = 0.192

    for row, count in enumerate(row_counts):
        x_start = 0.07 + (dx / 2 if row % 2 == 1 else 0)
        for col in range(count):
            if idx >= n:
                break
            centers[idx, 0] = x_start + col * dx
            centers[idx, 1] = 0.07 + row * dy * np.sqrt(3) / 2
            idx += 1

    return centers


def run_packing():
    n = 26

    best_sum = 0.0
    best_centers = None
    best_radii = None

    for seed in range(15):
        np.random.seed(seed)

        centers = create_initial_packing(n)

        if seed > 0:
            centers += np.random.randn(n, 2) * 0.03
            centers = np.clip(centers, 0.05, 0.95)

        radii = np.ones(n) * 0.05

        for outer_iter in range(60):
            # Step 1: Optimize radii given positions (LP)
            radii = solve_radii_lp(centers, n)

            # Step 2: Optimize positions given radii
            centers = optimize_positions(centers, radii, n)

        # Final LP to ensure optimal radii for final positions
        radii = solve_radii_lp(centers, n)

        current_sum = np.sum(radii)
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers.copy()
            best_radii = radii.copy()

    # Final validation and cleanup
    best_centers = np.clip(best_centers, 0, 1)
    best_radii = np.maximum(best_radii, 0)

    # Ensure no circle exceeds boundaries
    for i in range(n):
        max_r = min(best_centers[i, 0], 1 - best_centers[i, 0],
                    best_centers[i, 1], 1 - best_centers[i, 1])
        best_radii[i] = min(best_radii[i], max(0, max_r - 1e-12))

    return best_centers, best_radii, float(np.sum(best_radii))
