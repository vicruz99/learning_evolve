4. **Multiple restarts**: Run the optimization from several initial configurations and select the best result. This helps avoid poor local optima.

import numpy as np
from scipy.optimize import minimize


def boundary_lower_x(x, i):
    n = x.shape[0] // 3
    return x[n + 2 * i] - x[i]


def boundary_upper_x(x, i):
    n = x.shape[0] // 3
    return 1 - x[n + 2 * i] - x[i]


def boundary_lower_y(x, i):
    n = x.shape[0] // 3
    return x[n + 2 * i + 1] - x[i]


def boundary_upper_y(x, i):
    n = x.shape[0] // 3
    return 1 - x[n + 2 * i + 1] - x[i]


def nonneg_radius(x, i):
    return x[i]


def overlap_constraint(x, ij):
    i, j = ij
    n = x.shape[0] // 3
    r_i = x[i]
    r_j = x[j]
    dx = x[n + 2 * i] - x[n + 2 * j]
    dy = x[n + 2 * i + 1] - x[n + 2 * j + 1]
    dist_sq = dx * dx + dy * dy
    if dist_sq < 1e-24:
        return -1e-6
    dist = np.sqrt(dist_sq)
    return dist - r_i - r_j


def objective(x):
    n = x.shape[0] // 3
    return -np.sum(x[:n])


def expand_circles(centers, radii, n, iterations=2000):
    centers = centers.copy()
    radii = radii.copy()

    for it in range(iterations):
        forces = np.zeros((n, 2))
        max_radii = np.full(n, 1.0)

        for i in range(n):
            max_r = min(centers[i, 0], 1 - centers[i, 0],
                        centers[i, 1], 1 - centers[i, 1])

            for j in range(n):
                if i == j:
                    continue
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)

                if dist > 1e-10:
                    available = dist - radii[j]
                    max_r = min(max_r, available)

                    overlap_amt = radii[i] + radii[j] - dist
                    if overlap_amt > 0:
                        forces[i, 0] += overlap_amt * dx / dist
                        forces[i, 1] += overlap_amt * dy / dist

            max_radii[i] = max(0, max_r)

        alpha = 0.5 / (1 + it * 0.1)
        radii = radii + alpha * np.maximum(0, max_radii - radii)

        step = 0.15 / (1 + it * 0.08)
        centers = centers + step * forces

        for i in range(n):
            centers[i, 0] = max(radii[i] + 1e-8, min(1 - radii[i] - 1e-8, centers[i, 0]))
            centers[i, 1] = max(radii[i] + 1e-8, min(1 - radii[i] - 1e-8, centers[i, 1]))

    return centers, radii


def make_hex_grid(rows_config, base_x, base_y, x_spacing, y_spacing, n):
    centers = np.zeros((n, 2))
    idx = 0
    for row_idx, (n_cols, x_offset) in enumerate(rows_config):
        for col in range(n_cols):
            if idx >= n:
                break
            x = base_x + col * x_spacing + x_offset
            y = base_y + row_idx * y_spacing
            centers[idx] = [x, y]
            idx += 1
    return centers


def run_packing():
    n = 26
    best_sum = -1
    best_centers = None
    best_radii = None

    configs = []

    centers1 = make_hex_grid([(6, 0.0), (5, 0.08), (6, 0.0), (5, 0.08), (4, 0.0)],
                            0.08, 0.08, 0.15, 0.18, n)
    configs.append((centers1.copy(), np.full(n, 0.02)))

    centers2 = make_hex_grid([(5, 0.0), (5, 0.075), (5, 0.0), (5, 0.075), (5, 0.0), (1, 0.0)],
                            0.1, 0.1, 0.16, 0.17, n)
    configs.append((centers2.copy(), np.full(n, 0.02)))

    centers3 = make_hex_grid([(6, 0.0), (5, 0.09), (6, 0.0), (5, 0.09), (4, 0.0)],
                            0.06, 0.06, 0.16, 0.19, n)
    configs.append((centers3.copy(), np.full(n, 0.018)))

    centers4 = make_hex_grid([(5, 0.0), (5, 0.08), (5, 0.0), (5, 0.08), (5, 0.0), (1, 0.0)],
                            0.12, 0.12, 0.15, 0.16, n)
    configs.append((centers4.copy(), np.full(n, 0.025)))

    centers5 = make_hex_grid([(7, 0.0), (6, 0.067), (7, 0.0), (6, 0.067)],
                            0.05, 0.08, 0.13, 0.22, n)
    configs.append((centers5.copy(), np.full(n, 0.015)))

    for centers_init, radii_init in configs:
        centers_exp, radii_exp = expand_circles(centers_init.copy(), radii_init.copy(), n, iterations=2000)

        x0 = np.concatenate([radii_exp, centers_exp.flatten()])

        constraints = []
        for i in range(n):
            constraints.append({'type': 'ineq', 'fun': boundary_lower_x, 'args': (i,)})
            constraints.append({'type': 'ineq', 'fun': boundary_upper_x, 'args': (i,)})
            constraints.append({'type': 'ineq', 'fun': boundary_lower_y, 'args': (i,)})
            constraints.append({'type': 'ineq', 'fun': boundary_upper_y, 'args': (i,)})
            constraints.append({'type': 'ineq', 'fun': nonneg_radius, 'args': (i,)})

        for i in range(n):
            for j in range(i + 1, n):
                constraints.append({'type': 'ineq', 'fun': overlap_constraint, 'args': ((i, j),)})

        bounds = [(0, 0.5) for _ in range(n)] + [(0, 1) for _ in range(2 * n)]

        try:
            result = minimize(objective, x0, method='SLSQP',
                             bounds=bounds, constraints=constraints,
                             options={'maxiter': 15000, 'ftol': 1e-15})

            current_sum = np.sum(result.x[:n])
            if current_sum > best_sum:
                best_sum = current_sum
                best_centers = result.x[n:].reshape((n, 2)).copy()
                best_radii = result.x[:n].copy()
        except Exception:
            pass

    return best_centers, best_radii, best_sum
