import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def boundary_constraints(params):
    """Boundary constraints: circles must stay inside unit square"""
    n = N_CIRCLES
    c = params[:2 * n].reshape(n, 2)
    r = params[2 * n:]
    v = np.empty(4 * n)
    for i in range(n):
        v[4 * i] = c[i, 0] - r[i]
        v[4 * i + 1] = 1 - c[i, 0] - r[i]
        v[4 * i + 2] = c[i, 1] - r[i]
        v[4 * i + 3] = 1 - c[i, 1] - r[i]
    return v

def overlap_constraints(params):
    """Overlap constraints: circles must not overlap"""
    n = N_CIRCLES
    c = params[:2 * n].reshape(n, 2)
    r = params[2 * n:]
    num_pairs = n * (n - 1) // 2
    v = np.empty(num_pairs)
    k = 0
    for i in range(n):
        for j in range(i + 1, n):
            dx = c[i, 0] - c[j, 0]
            dy = c[i, 1] - c[j, 1]
            v[k] = dx * dx + dy * dy - (r[i] + r[j]) ** 2
            k += 1
    return v

def objective(params):
    """Negated sum of radii (minimize negative = maximize sum)"""
    n = N_CIRCLES
    return -np.sum(params[2 * n:])

def hexagonal_initial(n, r):
    """Create hexagonal packing initial configuration"""
    c = np.zeros((n, 2))
    k = 0
    for row in range(7):
        ncols = 5 if row % 2 == 0 else 4
        for col in range(ncols):
            if k >= n:
                break
            x = r + col * 2 * r
            y = r + row * np.sqrt(3) * r
            if row % 2 == 1:
                x += r
            c[k, 0] = x
            c[k, 1] = y
            k += 1
        if k >= n:
            break
    return c, np.ones(n) * r

def run_packing():
    n = N_CIRCLES
    
    # Create initial hexagonal packing configuration
    c0, r0 = hexagonal_initial(n, 0.07)
    x0 = np.concatenate([c0.ravel(), r0])
    
    # Bounds: centers in [0,1], radii in [epsilon, 0.5]
    bnds = [(0.0, 1.0)] * (2 * n) + [(1e-8, 0.5)] * n
    
    # Constraints
    cons = [
        {'type': 'ineq', 'fun': boundary_constraints},
        {'type': 'ineq', 'fun': overlap_constraints}
    ]
    
    # Track best solution
    best_x = x0.copy()
    best_f = objective(x0)
    
    # Multiple restarts to escape local minima
    for seed in range(10):
        rng = np.random.RandomState(seed)
        x_init = x0.copy()
        x_init += rng.uniform(-0.005, 0.005, size=x_init.shape)
        
        # Clamp initial values to bounds
        for i in range(2 * n):
            x_init[i] = np.clip(x_init[i], 0.0, 1.0)
        for i in range(n):
            x_init[2 * n + i] = np.clip(x_init[2 * n + i], 1e-8, 0.5)
        
        try:
            res = minimize(
                objective,
                x_init,
                method='SLSQP',
                bounds=bnds,
                constraints=cons,
                options={
                    'maxiter': 5000,
                    'ftol': 1e-14,
                    'disp': False
                }
            )
            if res.fun < best_f:
                best_f = res.fun
                best_x = res.x.copy()
        except Exception:
            continue
    
    centers = best_x[:2 * n].reshape(n, 2)
    radii = best_x[2 * n:]
    sum_radii = float(np.sum(radii))
    
    return centers, radii, sum_radii
