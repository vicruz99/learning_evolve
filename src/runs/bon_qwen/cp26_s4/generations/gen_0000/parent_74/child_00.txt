import numpy as np
from scipy.optimize import minimize

def compute_penalty(v, n):
    """
    Computes the quadratic penalty for boundary violations and overlaps.
    v: flattened array of shape (3*n,) containing [x_0, y_0, r_0, ..., x_{n-1}, y_{n-1}, r_{n-1}]
    n: number of circles
    """
    pts = v.reshape((n, 3))
    centers = pts[:, :2]
    radii = pts[:, 2]
    p = 0.0
    
    # Boundary penalties
    for i in range(n):
        x, y, r = centers[i, 0], centers[i, 1], radii[i]
        if r > x:
            p += 0.5 * (r - x)**2
        if x + r > 1:
            p += 0.5 * (x + r - 1)**2
        if r > y:
            p += 0.5 * (r - y)**2
        if y + r > 1:
            p += 0.5 * (y + r - 1)**2
            
    # Overlap penalties
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx*dx + dy*dy)
            if radii[i] + radii[j] > dist:
                p += 0.5 * (radii[i] + radii[j] - dist)**2
    return p

def compute_penalty_grad(v, n):
    """
    Computes the gradient of the penalty function with respect to v.
    """
    pts = v.reshape((n, 3))
    centers = pts[:, :2]
    radii = pts[:, 2]
    gp = np.zeros_like(v)
    
    # Boundary gradient contributions
    for i in range(n):
        x, y, r = centers[i, 0], centers[i, 1], radii[i]
        # Left wall
        if r > x:
            gp[3*i]   -= (r - x)
            gp[3*i+2] += (r - x)
        # Right wall
        if x + r > 1:
            gp[3*i]   += (x + r - 1)
            gp[3*i+2] += (x + r - 1)
        # Bottom wall
        if r > y:
            gp[3*i+1] -= (r - y)
            gp[3*i+2] += (r - y)
        # Top wall
        if y + r > 1:
            gp[3*i+1] += (y + r - 1)
            gp[3*i+2] += (y + r - 1)
            
    # Overlap gradient contributions
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx*dx + dy*dy)
            rs = radii[i] + radii[j]
            if rs > dist:
                v_ij = rs - dist
                if dist > 1e-9:
                    fx = -v_ij * dx / dist
                    fy = -v_ij * dy / dist
                else:
                    fx, fy = 0.0, 0.0
                gp[3*i]   += fx
                gp[3*i+1] += fy
                gp[3*i+2] += v_ij
                gp[3*j]   -= fx
                gp[3*j+1] -= fy
                gp[3*j+2] += v_ij
    return gp

def obj_func(v, n, mu):
    """Objective function: minimize -sum(radii) + mu * penalty"""
    return -np.sum(v[2::3]) + mu * compute_penalty(v, n)

def grad_func(v, n, mu):
    """Gradient of the objective function"""
    g = np.zeros_like(v)
    g[2::3] = -1.0  # Gradient of -sum(r_i)
    g += mu * compute_penalty_grad(v, n)
    return g

def run_packing():
    n = 26
    # Initialize with a hexagonal grid pattern
    centers_init = np.zeros((n, 2))
    idx = 0
    # 5 rows: first 4 rows have 5 circles, last row has 6 circles
    for row in range(5):
        # Stagger rows
        shift = 0.075 if row % 2 == 1 else 0.0
        x_start = 0.05 + shift
        count = 5 if row < 4 else 6
        for col in range(count):
            if idx >= n:
                break
            centers_init[idx, 0] = x_start + col * 0.15
            centers_init[idx, 1] = 0.08 + row * 0.16
            idx += 1
            
    radii_init = np.full(n, 0.05)
    v0 = np.zeros(3 * n)
    for i in range(n):
        v0[3*i] = centers_init[i, 0]
        v0[3*i+1] = centers_init[i, 1]
        v0[3*i+2] = radii_init[i]
        
    # Bounds: 0 <= x, y, r <= 1
    bounds = [(0.0, 1.0)] * (3 * n)
    
    mu = 10.0
    v = v0.copy()
    
    # Iteratively increase penalty weight to tighten constraints
    for step in range(50):
        res = minimize(
            fun=obj_func,
            x0=v,
            args=(n, mu),
            jac=grad_func,
            bounds=bounds,
            method='L-BFGS-B',
            options={'maxiter': 300, 'ftol': 1e-12, 'gtol': 1e-8}
        )
        v = res.x
        mu *= 3.0
        # Early stopping if penalty is negligible and mu is high
        if res.success and compute_penalty(v, n) < 1e-8 and mu > 1e5:
            break
            
    centers = v.reshape((n, 3))[:, :2]
    radii = v.reshape((n, 3))[:, 2]
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
