1. $r_i \ge 0$

import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimizes the positions and radii of 26 circles in a unit square
    to maximize the sum of radii.
    
    Returns:
        centers: np.array of shape (26, 2)
        radii: np.array of shape (26,)
        sum_radii: float
    """
    n = 26
    
    # Function to compute constraint violations (must be >= 0)
    def get_constraints(v):
        # v is shape (3*n,)
        # Layout: x0, y0, r0, x1, y1, r1, ...
        x = v[0::3]
        y = v[1::3]
        r = v[2::3]
        
        # 1. Boundary constraints: r <= x <= 1-r  =>  x-r >= 0, 1-x-r >= 0
        #    r <= y <= 1-r  =>  y-r >= 0, 1-y-r >= 0
        cons_boundary = np.concatenate([
            x - r,
            1.0 - x - r,
            y - r,
            1.0 - y - r
        ])
        
        # 2. Pairwise non-overlap constraints: dist^2 >= (r_i + r_j)^2
        #    (x_i - x_j)^2 + (y_i - y_j)^2 - (r_i + r_j)^2 >= 0
        #    Use broadcasting for efficiency
        
        # Compute differences matrix (n x n)
        # dx[i, j] = x_i - x_j
        dx = x[:, None] - x[None, :]
        dy = y[:, None] - y[None, :]
        
        # Squared Euclidean distance
        d2 = dx**2 + dy**2
        
        # Sum of radii matrix
        r_sum = r[:, None] + r[None, :]
        r_sum2 = r_sum**2
        
        # Difference
        diff = d2 - r_sum2
        
        # We only need constraints for i < j (upper triangle)
        # np.triu_indices(n, k=1) gives indices for upper triangle excluding diagonal
        row_idx, col_idx = np.triu_indices(n, k=1)
        cons_pairs = diff[row_idx, col_idx]
        
        return np.concatenate([cons_boundary, cons_pairs])

    def objective(v):
        r = v[2::3]
        # We want to maximize sum(r), so minimize -sum(r)
        return -np.sum(r)

    # Bounds for variables: x, y in [0, 1], r in [0, 1]
    # Tighter bounds for r could be [0, 0.5] but [0,1] is safe
    bounds = []
    for _ in range(n):
        bounds.extend([(0, 1), (0, 1), (0, 1)])

    constraints = {'type': 'ineq', 'fun': get_constraints}

    best_sum_radii = -np.inf
    best_v = None
    
    # Run multiple trials with different seeds to find a good local optimum
    num_trials = 15
    
    for trial in range(num_trials):
        # Random seed for this trial
        seed = trial * 137 + 42
        rng = np.random.RandomState(seed)
        
        # Initialize centers randomly in the interior of the square
        # Keep away from boundaries slightly to help initial feasibility
        x0_centers = rng.uniform(0.1, 0.9, size=(n, 2))
        
        # Initialize radii to a small valid value
        # With centers spread out, 0.05 should be safe initially, 
        # but let's use 0.01 to be absolutely sure of feasibility
        r0 = np.full(n, 0.01)
        
        # Construct initial vector v
        v0 = np.zeros(3 * n)
        v0[0::3] = x0_centers[:, 0]
        v0[1::3] = x0_centers[:, 1]
        v0[2::3] = r0
        
        # Run optimization
        try:
            res = minimize(
                objective,
                v0,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 500, 'ftol': 1e-9}
            )
            
            if res.success or res.fun < best_sum_radii: # Note: res.fun is negative sum
                # Check validity of result manually to be safe
                # (Though solver should respect constraints)
                v_res = res.x
                r_res = v_res[2::3]
                current_sum = np.sum(r_res)
                
                if current_sum > best_sum_radii:
                    best_sum_radii = current_sum
                    best_v = v_res.copy()
                    
        except Exception:
            continue

    # If no valid solution found (unlikely), fallback to a simple grid
    if best_v is None:
        best_v = np.zeros(3 * n)
        # Grid placement
        pts = np.linspace(0.1, 0.9, 6)
        centers = []
        count = 0
        for i in range(5):
            for j in range(6):
                if count < n:
                    centers.append([pts[j], pts[i]])
                    count += 1
        centers = np.array(centers)
        best_v[0::3] = centers[:, 0]
        best_v[1::3] = centers[:, 1]
        best_v[2::3] = 0.01 # Small radius
        best_sum_radii = np.sum(best_v[2::3])

    # Extract results
    centers = np.column_stack((best_v[0::3], best_v[1::3]))
    radii = best_v[2::3]
    sum_radii = np.sum(radii)
    
    # Final validation check (silent)
    # This helps ensure we didn't drift outside bounds due to numerical issues
    # But the constraints should handle it.
    
    return centers, radii, sum_radii
