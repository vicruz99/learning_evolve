def run_packing():
    n = 26
    np.random.seed(42)
    
    # Initialize centers
    # Using a grid-like initialization for better packing density start
    # 26 points. 5x5 grid is 25. Let's use 5x5 + 1 in center?
    # Or just random.

    centers = np.random.uniform(0.15, 0.85, size=(n, 2))
    radii = np.full(n, 0.01)

                dist = np.linalg.norm(d)
                min_dist = radii[i] + radii[j]
                if dist < min_dist and dist > 1e-9:
                    push = (d / dist) * (min_dist - dist) * 0.5
                    centers[i] += push
                    centers[j] -= push
        centers = np.clip(centers, 0.01, 0.99)

    # Main Optimization Loop
    repulsion_k = 20.0
    growth_rate = 0.0004
    num_steps = 4000
    inner_iters = 30
    
    # Precompute indices for loops if needed, but vectorized is better.
    
    for step in range(num_steps):
        # Grow radii
        radii += growth_rate
        radii = np.clip(radii, 0, 0.5)
        
        # Relaxation
        for _ in range(inner_iters):
            # Compute pairwise distances
            # diffs shape (n, n, 2)
            diffs = centers[:, None, :] - centers[None, :, :]
            dists = np.sqrt(np.sum(diffs**2, axis=2))
            np.fill_diagonal(dists, np.inf)
            
            # Sum of radii
            sum_radii = radii[:, None] + radii[None, :]
            
            # Overlaps
            overlaps = np.maximum(0, sum_radii - dists)
            
            # If system is relaxed enough, break
            if np.max(overlaps) < 1e-7:
                break
                
            # Compute forces
            safe_dists = np.where(dists > 1e-9, dists, 1.0)
            normals = diffs / safe_dists[:, :, None]
            
            # Force vector = overlap * k * normal
            forces = normals * overlaps[:, :, None] * repulsion_k
            
            # Net force on each circle
            net_forces = np.sum(forces, axis=1)
            
            # Move centers
            centers += net_forces * 0.5
            
            # Boundary Projection
            # Ensure x in [r, 1-r] and y in [r, 1-r]
            lower = radii
            upper = 1.0 - radii
            
            # Clip x
            centers[:, 0] = np.clip(centers[:, 0], lower, upper)
            # Clip y
            centers[:, 1] = np.clip(centers[:, 1], lower, upper)
            
        # Jitter to escape local minima
        if step % 100 == 0:
            centers += np.random.uniform(-0.005, 0.005, centers.shape)
            # Re-project after jitter
            centers[:, 0] = np.clip(centers[:, 0], lower, upper)
            centers[:, 1] = np.clip(centers[:, 1], lower, upper)
