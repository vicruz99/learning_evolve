import numpy as np

def compute_step(centers, radii, step_size):
    """
    Perform one iteration of expansion and force-based relaxation.
    Returns updated centers and radii.
    """
    n = len(radii)
    
    # Pairwise differences and distances
    diff = centers[:, None, :] - centers[None, :, :]
    dist = np.sqrt(np.sum(diff ** 2, axis=2))
    np.fill_diagonal(dist, np.inf)
    
    # 1. Compute maximum allowed radius for each circle
    # Boundary constraints
    r_lim = np.minimum(centers[:, 0], 1.0 - centers[:, 0])
    r_lim = np.minimum(r_lim, centers[:, 1])
    r_lim = np.minimum(r_lim, 1.0 - centers[:, 1])
    # Neighbor constraints: dist(i,j) >= r_i + r_j  =>  r_i <= dist(i,j) - r_j
    r_lim = np.minimum(r_lim, np.min(dist - radii[None, :], axis=1))
    
    # Monotonically increase radii
    radii = np.maximum(radii, r_lim)
    
    # 2. Compute repulsive forces to resolve tight constraints/overlaps
    overlap = (radii[:, None] + radii[None, :] - dist)
    overlap = np.maximum(0.0, overlap)
    
    safe_dist = np.where(dist > 1e-10, dist, 1e-10)
    # Force direction: push away from overlapping circles
    forces = np.sum((overlap[:, :, None] * diff) / safe_dist[:, :, None], axis=1)
    
    # Boundary forces: push inward if penetrating walls
    forces[:, 0] += np.maximum(0.0, radii - centers[:, 0]) * 5.0
    forces[:, 0] -= np.maximum(0.0, centers[:, 0] - (1.0 - radii)) * 5.0
    forces[:, 1] += np.maximum(0.0, radii - centers[:, 1]) * 5.0
    forces[:, 1] -= np.maximum(0.0, centers[:, 1] - (1.0 - radii)) * 5.0
    
    # Update positions
    centers += forces * step_size
    centers = np.clip(centers, 0.0, 1.0)
    
    return centers, radii
