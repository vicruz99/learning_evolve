import numpy as np
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # Helper to compute distance squared
    def dist_sq(c1, c2):
        return np.sum((c1 - c2)**2)

    def get_max_r(centers, radii):
        # Calculate maximum possible radius for each circle given current centers and other radii
        # This is used for initialization or radius update heuristic if needed, 
        # but here we rely on the optimizer.
        return radii

    def initialize_hexagonal(n):
        # Generate points in a hexagonal grid pattern
        # Estimate radius to fit n circles. 
        # For n=26, density approx 0.9. Area per circle ~ 1/26.
        # r ~ sqrt(1/(26*pi*0.9)) ~ 0.11
        # Let's use a slightly smaller radius to be safe, 0.09
        r_init = 0.09
        dy = np.sqrt(3) * r_init
        dx = 2 * r_init
        
        points = []
        y = r_init
        row = 0
        while len(points) < n:
            x = r_init + (row % 2) * (dx / 2)
            while x <= 1 - r_init and len(points) < n:
                points.append([x, y])
                x += dx
            y += dy
            row += 1
        
        centers = np.array(points[:n])
        radii = np.full(n, 0.05) # Start with small radius to avoid constraint violation
        return centers, radii

    def objective(vars):
        # vars: [x0, y0, r0, x1, y1, r1, ..., x25, y25, r25]
        # But we structured it as centers then radii in the logic below?
        # Let's stick to a flat array: x0, y0, r0, x1, y1, r1...
        # Actually, easier to unpack.
        
        # Re-structure: variables are (n, 3) -> [x, y, r]
        c = vars[:2*n].reshape((n, 2))
        r = vars[2*n:]
        
        return -np.sum(r)

    def get_constraints(n, c_init=None):
        constraints = []
        
        # Boundary constraints: x - r >= 0 => x - r >= 0
        # In scipy, constraints are fun(x) >= 0
        # x_i - r_i >= 0
        for i in range(n):
            idx_x = i
            idx_r = 2*n + i
            constraints.append({
                'type': 'ineq',
                'fun': lambda v, i=i, idx_x=idx_x, idx_r=idx_r: v[idx_x] - v[idx_r]
            })
            # 1 - x_i - r_i >= 0
            constraints.append({
                'type': 'ineq',
                'fun': lambda v, i=i, idx_x=idx_x, idx_r=idx_r: 1 - v[idx_x] - v[idx_r]
            })
            # y_i - r_i >= 0
            idx_y = n + i
            constraints.append({
                'type': 'ineq',
                'fun': lambda v, i=i, idx_y=idx_y, idx_r=idx_r: v[idx_y] - v[idx_r]
            })
            # 1 - y_i - r_i >= 0
            constraints.append({
                'type': 'ineq',
                'fun': lambda v, i=i, idx_y=idx_y, idx_r=idx_r: 1 - v[idx_y] - v[idx_r]
            })
            # r_i >= 0
            constraints.append({
                'type': 'ineq',
                'fun': lambda v, i=i, idx_r=idx_r: v[idx_r]
            })

        # Overlap constraints
        # (x_i - x_j)^2 + (y_i - y_j)^2 >= (r_i + r_j)^2
        # fun(v) = dist_sq - (r_i + r_j)^2
        for i in range(n):
            for j in range(i + 1, n):
                idx_xi = i
                idx_yi = n + i
                idx_ri = 2*n + i
                
                idx_xj = j
                idx_yj = n + j
                idx_rj = 2*n + j
                
                # Using lambda closure carefully
                def fun(v, i=i, j=j, 
                        xi=idx_xi, yi=idx_yi, ri=idx_ri,
                        xj=idx_xj, yj=idx_yj, rj=idx_rj):
                    dx = v[xi] - v[xj]
                    dy = v[yi] - v[yj]
                    dr = v[ri] + v[rj]
                    return (dx*dx + dy*dy) - (dr*dr)
                
                constraints.append({
                    'type': 'ineq',
                    'fun': fun
                })
        
        return constraints

    best_centers = None
    best_radii = None
    best_sum = -1.0

    # 1. Hexagonal
    # 2. Random perturbation of Hexagonal
    # 3. Random

    strategies.append(v2)

                    best_centers = c_opt
                    best_radii = r_opt
                    print(f"Strategy {idx} improved sum to {current_sum}")
        except Exception as e:
            print(f"Strategy {idx} failed: {e}")

    # If optimization didn't find anything valid or good, fallback to grid
    if best_centers is None:
        # Fallback to simple grid

        # Scale to fit
        max_c = np.max(centers)
        min_c = np.min(centers)
        # Just use a safe radius

        best_centers = centers
        best_sum = np.sum(best_radii)

    # Final validation and clipping just in case
    # Ensure radii are non-negative
    best_radii = np.maximum(best_radii, 1e-9)
    
    # Ensure centers are within bounds adjusted by radius
    # This might be violated if optimizer struggled, so we enforce it
    # But if we enforce it, we might reduce radii.
    # The constraints should have handled it.
    
    return best_centers, best_radii, best_sum
