import numpy as np
from scipy.optimize import minimize, NonlinearConstraint, Bounds

def obj(v):
    """Objective function: negative sum of radii"""
    return -np.sum(v[2::3])

def bound_constraints(v):
    """Boundary constraints: circles inside [0,1]^2"""
    n = len(v) // 3
    cons = np.empty(4 * n)
    idx = 0
    for i in range(n):
        x = v[3*i]
        y = v[3*i+1]
        r = v[3*i+2]
        cons[idx] = x - r
        cons[idx+1] = 1.0 - x - r
        cons[idx+2] = y - r
        cons[idx+3] = 1.0 - y - r
        idx += 4
    return cons

def pair_constraints(v):
    """Pairwise non-overlap constraints"""
    n = len(v) // 3
    n_pairs = n * (n - 1) // 2
    cons = np.empty(n_pairs)
    idx = 0
    for i in range(n):
        xi = v[3*i]
        yi = v[3*i+1]
        ri = v[3*i+2]
        for j in range(i + 1, n):
            xj = v[3*j]
            yj = v[3*j+1]
            rj = v[3*j+2]
            cons[idx] = (xi - xj)**2 + (yi - yj)**2 - (ri + rj)**2
            idx += 1
    return cons

def run_packing():
    n = 26
    
    # Initial guess: staggered hexagonal layout
    v0 = np.zeros(3 * n)
    idx = 0
    r_init = 0.04
    for i in range(n):
        col = i % 6
        row = i // 6
        x = 0.12 + col * 0.14
        y = 0.12 + row * 0.14
        if row % 2 == 1:
            x += 0.07
        v0[idx] = x
        v0[idx+1] = y
        v0[idx+2] = r_init
        idx += 3
        
    # Bounds for variables: x,y in [0,1], r in [0, 0.5]
    lb = np.zeros(3 * n)
    ub = np.ones(3 * n)
    ub[2::3] = 0.5
    bounds = Bounds(lb, ub)
    
    # Vectorized constraints for trust-constr
    con1 = NonlinearConstraint(
        bound_constraints, 
        lb=np.zeros(4 * n), 
        ub=np.full(4 * n, np.inf)
    )
    n_pairs = n * (n - 1) // 2
    con2 = NonlinearConstraint(
        pair_constraints, 
        lb=np.zeros(n_pairs), 
        ub=np.full(n_pairs, np.inf)
    )
    
    # Run optimization
    res = minimize(
        obj, v0, method='trust-constr', bounds=bounds, 
        constraints=[con1, con2], 
        options={'maxiter': 2000, 'verbose': 0}
    )
    
    centers = res.x[:2*n].reshape(n, 2)
    radii = res.x[2*n:]
    
    return centers, radii, np.sum(radii)
