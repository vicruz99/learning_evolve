import numpy as np
from scipy.optimize import minimize, differential_evolution


def run_packing():

    best_centers = None
    best_radii = None

    if s3 > best_sum:

def try_slqp_hexagonal(n):
    x0 = generate_hexagonal_config(n)
    bounds = get_bounds(n)
    constraints = get_all_constraints(n)
    
    best_sum = -1
    best_x = None
    
    for trial in range(6):
        x_trial = x0.copy()
        if trial > 0:
            np.random.seed(trial * 13 + 7)
            noise = np.random.randn(len(x0)) * 0.015
            x_trial = x_trial + noise
            x_trial = clip_to_feasible(x_trial, n)
        
        result = minimize(objective_func, x_trial, args=(n,),
                         method='SLSQP', bounds=bounds,
                         constraints=constraints,
                         options={'maxiter': 5000, 'ftol': 1e-14, 'disp': False})
        
        s = np.sum(result.x[2::3])
        if s > best_sum:
            best_sum = s
            best_x = result.x.copy()
    
    centers, radii = decode_solution(best_x, n)
    return centers, radii, best_sum


def try_de_then_slqp(n):
    bounds = get_bounds(n)
    constraints = get_all_constraints(n)
    
    # Coarse DE search
    try:
        de_result = differential_evolution(
            objective_func, bounds, args=(n,),
            constraints=constraints,
            maxiter=300, popsize=12,
            mutation=(0.5, 1.0), recombination=0.8,
            seed=42, polish=True, tol=1e-6
        )
        best_x = de_result.x
    except Exception:
        # Fallback to hexagonal
        best_x = generate_hexagonal_config(n)
    
    # Refine with SLSQP
    result = minimize(objective_func, best_x, args=(n,),
                     method='SLSQP', bounds=bounds,
                     constraints=constraints,
                     options={'maxiter': 5000, 'ftol': 1e-14, 'disp': False})
    
    centers, radii = decode_solution(result.x, n)
    return centers, radii, np.sum(radii)


def try_slqp_random_starts(n, num_starts=8):
    bounds = get_bounds(n)
    constraints = get_all_constraints(n)
    
    best_sum = -1
    best_x = None
    
    for trial in range(num_starts):
        np.random.seed(trial * 31 + 11)
        x_trial = generate_random_feasible(n)
        
        result = minimize(objective_func, x_trial, args=(n,),
                         method='SLSQP', bounds=bounds,
                         constraints=constraints,
                         options={'maxiter': 5000, 'ftol': 1e-14, 'disp': False})
        
        s = np.sum(result.x[2::3])
        if s > best_sum:
            best_sum = s
            best_x = result.x.copy()
    
    centers, radii = decode_solution(best_x, n)
    return centers, radii, best_sum


def generate_hexagonal_config(n):
    """Generate hexagonal packing with rows of 5,4,5,4,5,3 = 26 circles."""
    rows = [5, 4, 5, 4, 5, 3]
    r = 0.095
    
    x0 = []
    y = r
    for n_in_row in rows:
        if n_in_row == 0:
            y += np.sqrt(3) * r
            continue
        spacing = 2 * r
        total_width = (n_in_row - 1) * spacing
        x_start = (1 - total_width) / 2
        for j in range(n_in_row):
            x = x_start + j * spacing
            x0.extend([x, y, r])
        y += np.sqrt(3) * r
    
    return np.array(x0)


def generate_random_feasible(n):
    """Generate a random feasible configuration with small circles."""
    r_init = 0.04
    x0 = []
    for i in range(n):
        x = r_init + np.random.random() * (1 - 2 * r_init)
        y = r_init + np.random.random() * (1 - 2 * r_init)
        x0.extend([x, y, r_init])
    return np.array(x0)


def clip_to_feasible(x, n):
    """Clip solution to be feasible."""
    x = x.copy()
    for i in range(n):
        ri = max(x[3*i + 2], 1e-6)
        x[3*i + 2] = ri
        x[3*i] = np.clip(x[3*i], ri, 1 - ri)
        x[3*i + 1] = np.clip(x[3*i + 1], ri, 1 - ri)
    return x


def get_bounds(n):
    """Return bounds for all variables."""
    bounds = []
    for _ in range(n):
        bounds.append((0.0, 1.0))   # x
        bounds.append((0.0, 1.0))   # y
        bounds.append((1e-7, 0.5))  # r
    return bounds


def objective_func(x, n):
    """Objective: maximize sum of radii (minimize negative sum)."""
    return -np.sum(x[2::3])


# Constraint functions (no closures, no lambdas)
def constr_x_min(x, i):
    return x[3 * i] - x[3 * i + 2]


def constr_x_max(x, i):
    return 1.0 - x[3 * i] - x[3 * i + 2]


def constr_y_min(x, i):
    return x[3 * i + 1] - x[3 * i + 2]


def constr_y_max(x, i):
    return 1.0 - x[3 * i + 1] - x[3 * i + 2]


def constr_no_overlap(x, i, j):
    xi, yi = x[3 * i], x[3 * i + 1]
    xj, yj = x[3 * j], x[3 * j + 1]
    ri, rj = x[3 * i + 2], x[3 * j + 2]
    dx = xi - xj
    dy = yi - yj
    dist = np.sqrt(dx * dx + dy * dy)
    return dist - ri - rj


def get_all_constraints(n):
    """Build all constraint dicts."""
    constraints = []
    # Boundary constraints
    for i in range(n):
        constraints.append({'type': 'ineq', 'fun': constr_x_min, 'args': (i,)})
        constraints.append({'type': 'ineq', 'fun': constr_x_max, 'args': (i,)})
        constraints.append({'type': 'ineq', 'fun': constr_y_min, 'args': (i,)})
        constraints.append({'type': 'ineq', 'fun': constr_y_max, 'args': (i,)})
    # Non-overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({'type': 'ineq', 'fun': constr_no_overlap, 'args': (i, j)})
    return constraints


def decode_solution(x, n):
    """Decode optimization variables to centers and radii."""
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        centers[i, 0] = x[3 * i]
        centers[i, 1] = x[3 * i + 1]
        radii[i] = x[3 * i + 2]
    return centers, radii
