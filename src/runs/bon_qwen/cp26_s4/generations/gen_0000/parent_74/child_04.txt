import numpy as np
import math
from scipy.optimize import minimize

def calculate_penalty(centers, r):
    """
    Calculates the penalty for a given configuration of centers and a target radius r.
    Penalty is 0 if valid.
    """
    penalty = 0.0
    
    # Boundary penalties
    for i in range(len(centers)):
        x, y = centers[i]
        # Check x boundaries
        if x < r:
            penalty += (r - x) ** 2
        if x > 1 - r:
            penalty += (x - (1 - r)) ** 2
        # Check y boundaries
        if y < r:
            penalty += (r - y) ** 2
        if y > 1 - r:
            penalty += (y - (1 - r)) ** 2
            
    # Overlap penalties (only check upper triangle to save time)
    for i in range(len(centers)):
        for j in range(i + 1, len(centers)):
            dx = centers[i][0] - centers[j][0]
            dy = centers[i][1] - centers[j][1]
            dist_sq = dx*dx + dy*dy
            dist = math.sqrt(dist_sq)
            min_dist = 2 * r
            if dist < min_dist:
                overlap = min_dist - dist
                penalty += overlap ** 2
                
    return penalty
