import numpy as np

def resolve_overlaps(centers, radii, max_iters=50):
    """
    Iteratively pushes overlapping circles apart and enforces boundary constraints.
    """
    n = centers.shape[0]
    for _ in range(max_iters):
        # Vectorized pairwise distance calculation
        diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
        dists = np.linalg.norm(diffs, axis=2)
        req = radii[:, np.newaxis] + radii[np.newaxis, :]
        overlap = req - dists

        # Identify overlapping pairs (excluding self-interaction)
        mask = overlap > 1e-5
        np.fill_diagonal(mask, False)
        row, col = np.where(mask)
        
        # Keep only i < j to process each pair exactly once per iteration
        idx = row < col
        row, col = row[idx], col[idx]

        if len(row) == 0:
            break

        # Apply repulsion forces
        for i, j in zip(row, col):
            d = centers[j] - centers[i]
            dist = dists[i, j]
            if dist < 1e-9:
                d = np.random.uniform(-1, 1, 2)
                dist = 1.0
            dir_vec = d / dist
            # Push apart by half the overlap plus a small margin to prevent stalling
            sep = (overlap[i, j] * 0.5 + 1e-6)
            centers[i] -= dir_vec * sep
            centers[j] += dir_vec * sep

    # Enforce square boundary constraints
    for i in range(n):
        centers[i, 0] = np.clip(centers[i, 0], radii[i], 1.0 - radii[i])
        centers[i, 1] = np.clip(centers[i, 1], radii[i], 1.0 - radii[i])
    return centers

    np.random.seed(42)

    best_centers = None
    best_radii = None

    # Define diverse initial configurations to sample different topologies
    inits = []
    
    # 1. Uniform random scatter
    inits.append(np.random.uniform(0.15, 0.85, (n, 2)))
    
    # 2. 5x5 Grid + center point (good baseline)
    grid = np.zeros((25, 2))
    x = np.linspace(0.125, 0.875, 5)
    y = np.linspace(0.125, 0.875, 5)
    k = 0
    for xi in x:
        for yi in y:
            grid[k] = [xi, yi]
            k += 1
    inits.append(np.vstack([grid, [0.5, 0.5]]))
    
    # 3. Staggered hexagonal subset (higher density potential)
    hex_pts = []
    for r in range(5):
        for c in range(6):
            if len(hex_pts) < n:
                x = 0.15 + c * 0.15 + (r % 2) * 0.075
                y = 0.15 + r * 0.13
                hex_pts.append([x, y])
    inits.append(np.array(hex_pts))

    # Optimize each initialization
    for init_c in inits:
        centers = init_c.copy()
        radii = np.full(n, 0.02)
        
        # Iterative growth and relaxation
        for step in range(3000):
            # Constant small growth step
            radii += 0.00015
            
            # Resolve overlaps and boundary violations
            centers = resolve_overlaps(centers, radii, max_iters=20)
            
            # Periodic jitter to escape local minima/jammed states
            if step % 400 == 0 and step > 0:
                centers += np.random.uniform(-0.02, 0.02, centers.shape)
                centers = resolve_overlaps(centers, radii, max_iters=15)
                
        # Final thorough resolution to guarantee validity
        centers = resolve_overlaps(centers, radii, max_iters=100)

        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers.copy()
            best_radii = radii.copy()
            
    return best_centers, best_radii, float(best_sum)
