import numpy as np
from scipy.optimize import minimize

def run_packing():
    """
    Pack 26 circles in a unit square to maximize the sum of radii.
    """
    n_circles = 26
    
    # Helper to generate initial hexagonal grid positions
    def generate_initial_centers(n, seed=42):
        rng = np.random.RandomState(seed)
        # Approximate spacing for n circles in unit square
        # Area per circle approx 1/n. Radius approx sqrt(1/(pi*n))
        # Diameter approx 2*sqrt(1/(pi*n))
        # Spacing d ~ diameter.
        # For hex packing, density is higher.
        # Let's just place them on a grid first and perturb.
        
        # Try to fit on a roughly square grid
        cols = int(np.ceil(np.sqrt(n)))
        rows = int(np.ceil(n / cols))
        
        # Grid spacing
        # We want to leave some margin for radii, so let's pack centers in [0.1, 0.9] initially?
        # Or just [0,1] and let radii be small.
        # Better to spread them out.
        
        x_coords = np.linspace(0.1, 0.9, cols)
        y_coords = np.linspace(0.1, 0.9, rows)
        
        centers = []
        count = 0
        for y in y_coords:
            for x in x_coords:
                if count >= n:
                    break
                centers.append([x, y])
                count += 1
            if count >= n:
                break
        
        centers = np.array(centers)
        
        # Add small random noise to break symmetry
        noise = rng.randn(n, 2) * 0.01
        centers += noise
        
        # Clip to [0, 1]
        centers = np.clip(centers, 0.0, 1.0)
        
        return centers

    # Objective function: negative sum of radii
    def objective(vars):
        centers = vars[:2*n_circles].reshape(n_circles, 2)
        radii = vars[2*n_circles:]
        return -np.sum(radii)

    # Constraints
    def constraints(vars):
        centers = vars[:2*n_circles].reshape(n_circles, 2)
        radii = vars[2*n_circles:]
        
        cons = []
        
        # 1. Boundary constraints
        # x - r >= 0
        # x + r <= 1
        # y - r >= 0
        # y + r <= 1
        # We can vectorize this or loop. Loop is safer for list of dicts.
        
        # x - r >= 0  =>  x - r
        cons.append({'type': 'ineq', 'fun': lambda v: v[0::2] - v[2*n_circles:]}) # x_i - r_i >= 0
        # x + r <= 1  =>  1 - x - r
        cons.append({'type': 'ineq', 'fun': lambda v: 1.0 - v[0::2] - v[2*n_circles:]})
        # y - r >= 0  =>  y - r
        cons.append({'type': 'ineq', 'fun': lambda v: v[1::2] - v[2*n_circles:]})
        # y + r <= 1  =>  1 - y - r
        cons.append({'type': 'ineq', 'fun': lambda v: 1.0 - v[1::2] - v[2*n_circles:]})
        
        # 2. Non-overlap constraints
        # dist(c_i, c_j) >= r_i + r_j
        # dist^2 >= (r_i + r_j)^2
        # x_diff^2 + y_diff^2 - (r_i + r_j)^2 >= 0
        
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                # Indices in vars array
                # Center i: 2*i, 2*i+1
                # Center j: 2*j, 2*j+1
                # Radius i: 2*n + i
                # Radius j: 2*n + j
                
                def make_constraint(i_idx, j_idx):
                    def constraint_func(v):
                        xi, yi = v[2*i_idx], v[2*i_idx+1]
                        xj, yj = v[2*j_idx], v[2*j_idx+1]
                        ri = v[2*n_circles + i_idx]
                        rj = v[2*n_circles + j_idx]
                        dist_sq = (xi - xj)**2 + (yi - yj)**2
                        sum_r_sq = (ri + rj)**2
                        return dist_sq - sum_r_sq
                    return constraint_func
                
                cons.append({'type': 'ineq', 'fun': make_constraint(i, j)})
                
        return cons

    # Initial guess
    centers_init = generate_initial_centers(n_circles, seed=123)
    # Initialize radii small to satisfy constraints initially
    radii_init = np.full(n_circles, 0.01)
    
    vars_init = np.concatenate([centers_init.flatten(), radii_init])
    
    # Bounds
    # Centers in [0, 1]
    # Radii >= 0
    bounds = []
    for _ in range(n_circles):
        bounds.append((0.0, 1.0)) # x
        bounds.append((0.0, 1.0)) # y
    for _ in range(n_circles):
        bounds.append((0.0, 0.5)) # r (max possible radius 0.5)

    cons_list = constraints(vars_init)
    
    # Run optimization
    # SLSQP is good for constrained optimization
    res = minimize(objective, vars_init, method='SLSQP', bounds=bounds, constraints=cons_list, 
                   options={'ftol': 1e-8, 'maxiter': 1000})
    
    best_centers = res.x[:2*n_circles].reshape(n_circles, 2)
    best_radii = res.x[2*n_circles:]
    best_sum = np.sum(best_radii)
    
    # Try a few different seeds to find a better local optimum
    global_best_sum = best_sum
    global_centers = best_centers
    global_radii = best_radii
    
    for seed in [42, 100, 200, 777, 999]:
        centers_init = generate_initial_centers(n_circles, seed=seed)
        radii_init = np.full(n_circles, 0.01)
        vars_init = np.concatenate([centers_init.flatten(), radii_init])
        
        try:
            res = minimize(objective, vars_init, method='SLSQP', bounds=bounds, constraints=cons_list, 
                           options={'ftol': 1e-8, 'maxiter': 2000})
            curr_sum = -res.fun
            if curr_sum > global_best_sum:
                global_best_sum = curr_sum
                global_centers = res.x[:2*n_circles].reshape(n_circles, 2)
                global_radii = res.x[2*n_circles:]
        except:
            pass

    return global_centers, global_radii, global_best_sum
