import numpy as np
from scipy.optimize import minimize

def objective(params, n):
    """Objective: minimize negative sum of radii."""
    return -np.sum(params[2 * n:])

def boundary_constraint(params, n):
    """Boundary constraints: x-r >= 0, 1-x-r >= 0, y-r >= 0, 1-y-r >= 0."""
    centers = params[:2 * n].reshape(n, 2)
    radii = params[2 * n:]
    con = []
    for i in range(n):
        con.append(centers[i, 0] - radii[i])
        con.append(1.0 - centers[i, 0] - radii[i])
        con.append(centers[i, 1] - radii[i])
        con.append(1.0 - centers[i, 1] - radii[i])
    return np.array(con)

def overlap_constraint(params, n):
    """Overlap constraints: distance between centers >= sum of radii."""
    centers = params[:2 * n].reshape(n, 2)
    radii = params[2 * n:]
    con = []
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            d = np.sqrt(dx * dx + dy * dy)
            con.append(d - radii[i] - radii[j])
    return np.array(con)

def initialize_hexagonal(n):
    """Initialize circles in a hexagonal pattern."""
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.035)
    
    row_sizes = [7, 6, 7, 6]
    idx = 0
    row_spacing = 0.25
    col_spacing = 0.13
    y = 0.12
    for row_idx, size in enumerate(row_sizes):
        x_offset = 0.08 if row_idx % 2 == 1 else 0.07
        for col in range(size):
            centers[idx, 0] = x_offset + col * col_spacing
            centers[idx, 1] = y
            idx += 1
        y += row_spacing
    
    return centers, radii

def force_simulation(centers, radii, n, steps=800):
    """Run force-based simulation to spread circles and grow radii."""
    centers = centers.copy()
    radii = radii.copy()
    
    for step in range(steps):
        decay = max(0.001, 0.01 * (1.0 - step / steps))
        forces = np.zeros_like(centers)
        
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                d = np.sqrt(dx * dx + dy * dy)
                if d < 0.0001:
                    d = 0.0001
                    dx = 0.001 * (i % 3)
                    dy = 0.001 * (j % 3)
                
                required = radii[i] + radii[j] + 0.002
                if d < required:
                    strength = (required - d) / d * 2.0
                    forces[i, 0] += dx * strength
                    forces[i, 1] += dy * strength
                    forces[j, 0] -= dx * strength
                    forces[j, 1] -= dy * strength
        
        for i in range(n):
            margin = radii[i] + 0.015
            if centers[i, 0] < margin:
                forces[i, 0] += (margin - centers[i, 0]) * 3.0
            if centers[i, 0] > 1.0 - margin:
                forces[i, 0] -= (centers[i, 0] - (1.0 - margin)) * 3.0
            if centers[i, 1] < margin:
                forces[i, 1] += (margin - centers[i, 1]) * 3.0
            if centers[i, 1] > 1.0 - margin:
                forces[i, 1] -= (centers[i, 1] - (1.0 - margin)) * 3.0
        
        centers += forces * decay
        centers = np.clip(centers, 0.005, 0.995)
        
        if step % 5 == 0:
            for i in range(n):
                max_r = min(centers[i, 0], 1.0 - centers[i, 0],
                           centers[i, 1], 1.0 - centers[i, 1])
                for j in range(n):
                    if i != j:
                        dx = centers[i, 0] - centers[j, 0]
                        dy = centers[i, 1] - centers[j, 1]
                        d = np.sqrt(dx * dx + dy * dy)
                        max_r = min(max_r, d - radii[j])
                max_r = max(max_r, 0.001)
                radii[i] = min(radii[i] * 1.008, max_r)
    
    return centers, radii

def grid_init(n):
    """Initialize with a 5x5 grid plus 1 extra circle."""
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.075)
    
    idx = 0
    for row in range(5):
        for col in range(5):
            centers[idx, 0] = 0.08 + col * 0.18
            centers[idx, 1] = 0.08 + row * 0.18
            idx += 1
    
    centers[25, 0] = 0.5
    centers[25, 1] = 0.5
    
    return centers, radii

def optimize_with_scipy(centers, radii, n):
    """Optimize using scipy SLSQP."""
    params0 = np.concatenate([centers.flatten(), radii])
    
    constraints = []
    constraints.append({
        'type': 'ineq',
        'fun': lambda p: boundary_constraint(p, n)
    })
    constraints.append({
        'type': 'ineq',
        'fun': lambda p: overlap_constraint(p, n)
    })
    
    bounds = []
    for i in range(n):
        bounds.append((0.001, 0.999))
        bounds.append((0.001, 0.999))
    for i in range(n):
        bounds.append((0.001, 0.5))
    
    result = minimize(
        objective,
        params0,
        args=(n,),
        method='SLSQP',
        constraints=constraints,
        bounds=bounds,
        options={'maxiter': 2000, 'ftol': 1e-12, 'disp': False}
    )
    
    if result.success or result.fun < -1.5:
        centers = result.x[:2 * n].reshape(n, 2)
        radii = result.x[2 * n:]
    
    return centers, radii

def local_optimization(centers, radii, n, iterations=100):
    """Perform local optimization by trying to grow each circle."""
    centers = centers.copy()
    radii = radii.copy()
    
    for _ in range(iterations):
        for i in range(n):
            current_r = radii[i]
            
            max_r = min(centers[i, 0], 1.0 - centers[i, 0],
                       centers[i, 1], 1.0 - centers[i, 1])
            for j in range(n):
                if i != j:
                    dx = centers[i, 0] - centers[j, 0]
                    dy = centers[i, 1] - centers[j, 1]
                    d = np.sqrt(dx * dx + dy * dy)
                    max_r = min(max_r, d - radii[j])
            
            if max_r > current_r + 1e-8:
                radii[i] = max_r
                continue
            
            for j in range(n):
                if i != j:
                    dx = centers[i, 0] - centers[j, 0]
                    dy = centers[i, 1] - centers[j, 1]
                    d = np.sqrt(dx * dx + dy * dy)
                    if d < 0.0001:
                        continue
                    
                    target_dist = radii[i] + radii[j]
                    if d > target_dist:
                        ratio = (d - target_dist) / d * 0.3
                        centers[i, 0] += (centers[i, 0] - centers[j, 0]) * ratio
                        centers[i, 1] += (centers[i, 1] - centers[j, 1]) * ratio
            
            centers[i] = np.clip(centers[i], 0.005, 0.995)
            
            max_r = min(centers[i, 0], 1.0 - centers[i, 0],
                       centers[i, 1], 1.0 - centers[i, 1])
            for j in range(n):
                if i != j:
                    dx = centers[i, 0] - centers[j, 0]
                    dy = centers[i, 1] - centers[j, 1]
                    d = np.sqrt(dx * dx + dy * dy)
                    max_r = min(max_r, d - radii[j])
            
            radii[i] = max(max_r, 0.001)
    
    return centers, radii

def run_packing():
    n = 26
    
    best_sum = 0
    best_centers = None
    best_radii = None
    
    for init_method in [0, 1]:
        if init_method == 0:
            centers, radii = initialize_hexagonal(n)
        else:
            centers, radii = grid_init(n)
        
        centers, radii = force_simulation(centers, radii, n, steps=1000)
        
        centers, radii = optimize_with_scipy(centers, radii, n)
        
        centers, radii = local_optimization(centers, radii, n, iterations=50)
        
        current_sum = np.sum(radii)
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers.copy()
            best_radii = radii.copy()
    
    centers = best_centers
    radii = best_radii
    
    radii = np.maximum(radii, 0)
    centers = np.clip(centers, radii[:, np.newaxis], 1.0 - radii[:, np.newaxis])
    
    return centers, radii, np.sum(radii)
