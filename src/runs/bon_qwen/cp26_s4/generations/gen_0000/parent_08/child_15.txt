import numpy as np
from scipy.optimize import minimize
import math

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    
    # Helper to generate initial configurations
    def generate_starts():
        starts = []
        
        # 1. Grid initialization (6x5 grid, take first 26)
        centers_grid = np.zeros((n, 2))
        idx = 0
        for r in range(5):
            for c in range(6):
                if idx < n:
                    # Spread within [0.05, 0.95]
                    centers_grid[idx, 0] = 0.05 + c * (0.90 / 5)
                    centers_grid[idx, 1] = 0.05 + r * (0.90 / 4)
                    idx += 1
        starts.append(centers_grid)
        
        # 2. Hexagonal-ish initialization
        centers_hex = np.zeros((n, 2))
        idx = 0
        y = 0.1
        row = 0
        while idx < n and y <= 0.9:
            x = 0.1 if row % 2 == 0 else 0.2
            while idx < n and x <= 0.9:
                centers_hex[idx, 0] = x
                centers_hex[idx, 1] = y
                idx += 1
                x += 0.15 # slightly wider spacing
            y += 0.12
            row += 1
        # Fill remaining if any (should not happen)
        if idx < n:
             for k in range(idx, n):
                 centers_hex[k] = [0.5, 0.5]
        starts.append(centers_hex)
        
        # 3. Random initialization
        np.random.seed(42) # Reproducibility
        centers_rand = np.random.uniform(0.1, 0.9, (n, 2))
        starts.append(centers_rand)
        
        return starts

    def solve_packing(centers_init):
        # Initial radii guess
        r_init = np.full(n, 0.05)
        
        # Flatten variables: x (n), y (n), r (n)
        x0 = np.concatenate([centers_init.flatten(), r_init])
        
        # Bounds: x, y in [0, 1], r in [1e-6, 0.5]
        # Using 1e-6 to avoid log issues if any, though penalty handles 0.
        bounds = [(0.0, 1.0)] * (2 * n) + [(1e-6, 0.5)] * n
        
        # Penalty weight
        mu = 10000.0 
        
        def objective(vars):
            xc = vars[:n]
            yc = vars[n:2*n]
            rc = vars[2*n:]
            
            # Objective: maximize sum(r) -> minimize -sum(r)
            val = -np.sum(rc)
            
            # Penalty for boundary violations
            # Constraints: r <= x, r <= 1-x, r <= y, r <= 1-y
            # Violation is max(0, r - x), etc.
            # We sum squared violations
            for i in range(n):
                r = rc[i]
                x = xc[i]
                y = yc[i]
                
                # Left boundary
                diff = r - x
                if diff > 0:
                    val += mu * diff * diff
                # Right boundary
                diff = r - (1.0 - x)
                if diff > 0:
                    val += mu * diff * diff
                # Bottom boundary
                diff = r - y
                if diff > 0:
                    val += mu * diff * diff
                # Top boundary
                diff = r - (1.0 - y)
                if diff > 0:
                    val += mu * diff * diff

            # Penalty for overlaps
            # Constraint: dist >= r_i + r_j
            # Violation: max(0, r_i + r_j - dist)
            for i in range(n):
                for j in range(i + 1, n):
                    dx = xc[i] - xc[j]
                    dy = yc[i] - yc[j]
                    dist = math.sqrt(dx*dx + dy*dy)
                    overlap = rc[i] + rc[j] - dist
                    if overlap > 0:
                        val += mu * overlap * overlap
                        
            return val

        # Run optimization
        try:
            res = minimize(objective, x0, method='L-BFGS-B', bounds=bounds, 
                           options={'maxiter': 5000, 'ftol': 1e-15, 'gtol': 1e-10})
            return res
        except Exception:
            return None

    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    starts = generate_starts()
    
    for centers_init in starts:
        res = solve_packing(centers_init)
        if res is not None and res.success:
            xc = res.x[:n]
            yc = res.x[n:2*n]
            rc = res.x[2*n:]
            
            centers = np.column_stack((xc, yc))
            
            # Post-processing: Ensure strict validity by scaling down if necessary
            # Find minimum scaling factor alpha such that alpha * radii is valid
            min_alpha = 1.0
            
            # 1. Boundary checks
            # For each circle, alpha * r <= x => alpha <= x/r
            # alpha * r <= 1-x => alpha <= (1-x)/r
            for i in range(n):
                r = rc[i]
                x, y = centers[i]
                if r > 1e-9:
                    min_alpha = min(min_alpha, x / r, (1.0 - x) / r, y / r, (1.0 - y) / r)
            
            # 2. Overlap checks
            # alpha * (r_i + r_j) <= dist => alpha <= dist / (r_i + r_j)
            for i in range(n):
                for j in range(i + 1, n):
                    dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                    sum_r = rc[i] + rc[j]
                    if sum_r > 1e-9:
                        min_alpha = min(min_alpha, dist / sum_r)
            
            # Apply scaling
            # Note: We only scale radii, centers stay same. 
            # If centers were optimal for larger radii, scaling down just makes it loose.
            # But this guarantees validity.
            rc_valid = rc * min_alpha
            
            current_sum = np.sum(rc_valid)
            
            if current_sum > best_sum:
                best_sum = current_sum
                best_centers = centers.copy()
                best_radii = rc_valid.copy()
                
    # If optimization failed to find anything better than 0 (unlikely), fallback to small grid
    if best_sum <= 0:
        # Fallback
        best_centers = np.zeros((n, 2))
        best_radii = np.full(n, 0.01)
        for i in range(n):
            best_centers[i, 0] = (i % 6 + 1) / 7.0
            best_centers[i, 1] = (i // 6 + 1) / 5.0
        best_sum = np.sum(best_radii)

    return best_centers, best_radii, best_sum
