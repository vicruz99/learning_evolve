import numpy as np
from scipy.optimize import minimize
import itertools

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    N = 26
    
    # Define objective function: maximize sum of radii -> minimize negative sum
    def objective(v):
        # v contains [x0, y0, r0, x1, y1, r1, ...]
        radii = v[2::3]
        return -np.sum(radii)

    # Define constraints
    # 1. Boundary constraints: r <= x <= 1-r  =>  x - r >= 0  and  1 - x - r >= 0
    # 2. Non-overlap constraints: dist(i,j) >= r_i + r_j  =>  dist(i,j) - (r_i + r_j) >= 0
    
    def constraints_func(v):
        centers = v[:2*N].reshape(N, 2)
        radii = v[2*N:]
        
        # Boundary constraints (4 per circle)
        # x - r >= 0
        c1 = centers[:, 0] - radii
        # 1 - x - r >= 0
        c2 = 1 - centers[:, 0] - radii
        # y - r >= 0
        c3 = centers[:, 1] - radii
        # 1 - y - r >= 0
        c4 = 1 - centers[:, 1] - radii
        
        # Non-overlap constraints
        # Vectorized distance calculation
        # diff shape (N, N, 2)
        diff = centers[:, None, :] - centers[None, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        
        # radii_sum shape (N, N)
        radii_sum = radii[:, None] + radii[None, :]
        
        # We only need upper triangle (i < j)
        i_idx, j_idx = np.triu_indices(N, k=1)
        c5 = dists[i_idx, j_idx] - radii_sum[i_idx, j_idx]
        
        # Concatenate all constraints
        return np.concatenate([c1, c2, c3, c4, c5])

    # Bounds: x, y in [0, 1], r in [0, 1] (actually r max 0.5)
    # Flattened bounds
    bounds = [(0, 1)] * (2 * N) + [(0, 1)] * N

    # Constraint specification for SLSQP
    cons = {'type': 'ineq', 'fun': constraints_func}

    # Function to generate initial configurations
    def get_initial_configs():
        configs = []
        
        # Config 1: 5x5 Grid + 1 extra
        # 5x5 grid points
        xs = np.array([0.1, 0.3, 0.5, 0.7, 0.9])
        ys = np.array([0.1, 0.3, 0.5, 0.7, 0.9])
        c1 = np.zeros((N, 3))
        idx = 0
        for y in ys:
            for x in xs:
                c1[idx] = [x, y, 0.05]
                idx += 1
        # Add one more circle, try to place in a gap
        c1[idx] = [0.5, 0.2, 0.05] 
        configs.append(c1.flatten())
        
        # Config 2: Hexagonal-ish packing
        # 5 rows, counts [6, 5, 6, 5, 4] sum to 26
        c2 = np.zeros((N, 3))
        counts = [6, 5, 6, 5, 4]
        idx = 0
        # Approximate spacing for hex packing in [0,1]
        # We'll just distribute points uniformly in rows with offset
        for r in range(5):
            count = counts[r]
            # x coordinates
            # For even rows, start at 0.1, for odd shift right
            if r % 2 == 0:
                xs_row = np.linspace(0.05, 0.95, count)
            else:
                # Shifted row, narrower range to fit
                xs_row = np.linspace(0.15, 0.85, count)
            
            # y coordinate
            ys_row = np.full(count, 0.1 + r * 0.2)
            
            for k in range(count):
                c2[idx] = [xs_row[k], ys_row[k], 0.05]
                idx += 1
        configs.append(c2.flatten())
        
        # Config 3: Random perturbation of grid
        c3 = c1.copy()
        c3[0:2*N] += np.random.normal(0, 0.01, (N, 2))
        # Clip to bounds
        c3[0:2*N] = np.clip(c3[0:2*N], 0.01, 0.99)
        configs.append(c3.flatten())

        return configs

    # Run optimization
    best_result = None
    best_val = -np.inf
    
    configs = get_initial_configs()
    
    # Use a fixed seed for reproducibility of random parts if needed, 
    # but here we just run deterministic configs and one random.
    np.random.seed(42)

    for i, x0 in enumerate(configs):
        # Small perturbation to break symmetry if exact grid
        if i == 0:
            x0[0:2*N] += np.random.normal(0, 1e-4, (N, 2))
            x0[0:2*N] = np.clip(x0[0:2*N], 0.01, 0.99)

        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, 
                           constraints=cons, options={'maxiter': 2000, 'ftol': 1e-9})
            
            if res.success or res.fun < best_val: # We minimize -sum, so lower is better
                # Check validity explicitly
                centers = res.x[:2*N].reshape(N, 2)
                radii = res.x[2*N:]
                
                # Quick validation check
                valid = True
                # Boundary
                if np.any(radii < -1e-9) or np.any(centers < radii.reshape(-1,1) - 1e-9) or np.any(centers > 1 - radii.reshape(-1,1) + 1e-9):
                    valid = False
                # Overlap
                dists = np.linalg.norm(centers[:, None, :] - centers[None, :, :], axis=2)
                radii_sum = radii[:, None] + radii[None, :]
                # Check off-diagonal
                mask = np.ones((N, N), dtype=bool)
                np.fill_diagonal(mask, False)
                if np.any(dists[mask] < radii_sum[mask] - 1e-9):
                    valid = False
                
                if valid:
                    current_sum = np.sum(radii)
                    if current_sum > best_val:
                        best_val = current_sum
                        best_result = (centers, radii, current_sum)
        except Exception as e:
            print(f"Optimization failed: {e}")
            continue

    if best_result is None:
        # Fallback to a safe packing if optimization failed
        # 5x5 grid with r=0.1 is invalid (overlap).

        radii = np.full(N, 0.05)

        idx = 0
        xs = np.linspace(0.1, 0.9, 5)

            for x in xs:
                centers[idx] = [x, y]
                idx += 1
        centers[idx] = [0.5, 0.2]
        # Scale down radii to ensure no overlap? 
        # Grid spacing 0.2, r=0.1 touches. 
        # Set r=0.09
        radii[:] = 0.09
        best_result = (centers, radii, np.sum(radii))

    return best_result
