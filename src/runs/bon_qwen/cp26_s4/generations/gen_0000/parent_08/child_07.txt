from scipy.optimize import minimize, differential_evolution

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

        centers: np.array of shape (26, 2)
        radii: np.array of shape (26,)

    r_init = 0.09 # Conservative start

    # 5 rows of 6 circles? 30 circles. Too many.

    rows_config = [6, 5, 6, 5, 4] # Total 26

    # r <= 1 / (2 + 6.928) = 1/8.928 = 0.112
    # Let's use r=0.105 for initialization
    
    r_init = 0.105
    h_spacing = r_init * np.sqrt(3)
    v_offset = r_init # Start y at radius

    row_idx = 0

            break

        # Let's just place them evenly.
        
        # Number of circles to add in this row
        count = min(k, n_circles - current_circles)
        
        if count == 0:
            row_idx += 1
            continue

        # Distribute 'count' circles in [r_init, 1-r_init]

            xs = [0.5]
        else:
            # Evenly spaced

            if available_width < 0: available_width = 0
            step = available_width / (count - 1)
            xs = [r_init + i * step for i in range(count)]

        # Staggered: even rows at y, odd rows at y + h_spacing
        # But we need to fit in [0,1].
        # Let's calculate Y based on row index to fit 5 rows.
        # Total height needed for 5 rows with spacing h_spacing is 4 * h_spacing.
        # We can distribute them evenly or stick to hex.
        # Let's use hex Y.
        
        # To fit 5 rows nicely, we can scale the vertical position.
        # But let's just generate points and let optimizer fix it.
        y = r_init + row_idx * h_spacing

        # With r=0.105, h=0.18.
        # Row 0: 0.105
        # Row 1: 0.285
        # Row 2: 0.465
        # Row 3: 0.645
        # Row 4: 0.825

            current_circles += 1

        return -np.sum(radii)

            r = radii[i]
            # x >= r => x - r >= 0

            cons.append(1 - x - r)
            # y >= r => y - r >= 0
            cons.append(y - r)
            # y <= 1 - r => 1 - y - r >= 0

                    if current_sum > best_sum:

                        best_result = (centers, radii)
        except Exception as e:
            continue

    # If SLSQP didn't find a great solution, fallback to the initial valid one

        centers_final[i, 0] = np.clip(centers_final[i, 0], r, 1 - r)

        radii_final[i] = min(radii_final[i], centers_final[i,0], 1-centers_final[i,0], 
                             centers_final[i,1], 1-centers_final[i,1])
        
    sum_radii = np.sum(radii_final)
    
    return centers_final, radii_final, sum_radii
