import numpy as np
from scipy.optimize import minimize


def obj_func(x, n):
    radii = x[2 * n:]
    return -np.sum(radii)


def bound_con(x, n):
    centers = x[:2 * n].reshape(n, 2)
    radii = x[2 * n:]
    x_min = centers[:, 0] - radii
    x_max = 1.0 - centers[:, 0] - radii
    y_min = centers[:, 1] - radii
    y_max = 1.0 - centers[:, 1] - radii
    return np.concatenate([x_min, x_max, y_min, y_max])


def overlap_con(x, n):
    centers = x[:2 * n].reshape(n, 2)
    radii = x[2 * n:]
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist_sq = np.sum(diff ** 2, axis=2)
    r_sum = radii[:, np.newaxis] + radii[np.newaxis, :]
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    return dist_sq[mask] - r_sum[mask] ** 2


def init_hex_pattern(n):
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.09
    row_counts = []
    total = 0
    row = 0
    while total < n:
        cnt = 6 if row % 2 == 0 else 5
        row_counts.append(cnt)
        total += cnt
        row += 1
    row_counts[-1] = n - sum(row_counts[:-1])
    idx = 0
    row = 0
    sx = 0.23
    sy = sx * np.sqrt(3) / 2.0
    for cnt in row_counts:
        offset = sx / 2.0 if row % 2 == 1 else 0.0
        for col in range(cnt):
            xc = 0.12 + col * sx + offset
            yc = 0.12 + row * sy
            centers[idx] = [xc, yc]
            idx += 1
        row += 1
    return centers, radii


def run_packing():
    n = 26
    best_sum = 0.0
    best_centers = None
    best_radii = None

    for trial in range(15):
        centers_init, radii_init = init_hex_pattern(n)
        np.random.seed(trial * 13 + 3)
        centers_init = centers_init + np.random.uniform(-0.02, 0.02, (n, 2))
        centers_init = np.clip(centers_init, 0.08, 0.92)
        x0 = np.concatenate([centers_init.flatten(), radii_init])
        bounds = [(0.0, 1.0)] * (2 * n) + [(0.001, 0.5)] * n
        cons = [
            {"type": "ineq", "fun": bound_con, "args": (n,)},
            {"type": "ineq", "fun": overlap_con, "args": (n,)},
        ]
        result = minimize(
            obj_func,
            x0,
            args=(n,),
            bounds=bounds,
            constraints=cons,
            method="SLSQP",
            options={"maxiter": 25000, "ftol": 1e-16, "disp": False},
        )
        c_opt = result.x[:2 * n].reshape(n, 2)
        r_opt = result.x[2 * n:]
        r_opt = np.maximum(r_opt, 0.001)
        c_opt = np.clip(c_opt, 0.0, 1.0)
        current_sum = np.sum(r_opt)
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = c_opt.copy()
            best_radii = r_opt.copy()

    # Refinement pass with best solution as warm start
    x0_ref = np.concatenate([best_centers.flatten(), best_radii])
    result_ref = minimize(
        obj_func,
        x0_ref,
        args=(n,),
        bounds=bounds,
        constraints=cons,
        method="SLSQP",
        options={"maxiter": 30000, "ftol": 1e-16, "disp": False},
    )
    c_ref = result_ref.x[:2 * n].reshape(n, 2)
    r_ref = result_ref.x[2 * n:]
    r_ref = np.maximum(r_ref, 0.001)
    c_ref = np.clip(c_ref, 0.0, 1.0)
    sum_ref = np.sum(r_ref)
    if sum_ref > best_sum:
        best_sum = sum_ref
        best_centers = c_ref
        best_radii = r_ref

    # Final projection to ensure feasibility with tolerance
    best_radii = np.maximum(best_radii, 0.0)
    for i in range(n):
        best_centers[i, 0] = max(best_radii[i], min(1.0 - best_radii[i], best_centers[i, 0]))
        best_centers[i, 1] = max(best_radii[i], min(1.0 - best_radii[i], best_centers[i, 1]))

    return best_centers, best_radii, float(np.sum(best_radii))
