import numpy as np
from scipy.optimize import minimize

def _compute_obj_grad(vars, n, mu):
    c = vars[:2*n].reshape(n, 2)
    r = vars[2*n:]
    
    obj = -np.sum(r)
    grad = np.zeros_like(vars)
    grad[2*n:] = -1.0
    
    # Boundary constraints: r - x <= 0, r + x - 1 <= 0, same for y
    for i in range(n):
        for d in range(2):
            v1 = r[i] - c[i, d]
            if v1 > 0:
                obj += mu * v1**2
                grad[2*n+i] += 2*mu*v1
                grad[2*i+d] -= 2*mu*v1
                
            v2 = r[i] + c[i, d] - 1
            if v2 > 0:
                obj += mu * v2**2
                grad[2*n+i] += 2*mu*v2
                grad[2*i+d] += 2*mu*v2
                
    # Overlap constraints: r_i + r_j - dist <= 0
    for i in range(n):
        for j in range(i+1, n):
            diff = c[i] - c[j]
            dist = np.sqrt(np.sum(diff**2))
            if dist < 1e-12: 
                dist = 1e-12
            v = r[i] + r[j] - dist
            if v > 0:
                obj += mu * v**2
                grad[2*n+i] += 2*mu*v
                grad[2*n+j] += 2*mu*v
                
                factor = -2*mu*v / dist
                grad[2*i] += factor * diff[0]
                grad[2*i+1] += factor * diff[1]
                grad[2*j] -= factor * diff[0]
                grad[2*j+1] -= factor * diff[1]
                
    return obj, grad

def _obj(vars, n, mu):
    return _compute_obj_grad(vars, n, mu)[0]

def _jac(vars, n, mu):
    return _compute_obj_grad(vars, n, mu)[1]

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Multi-start optimization
    for seed in range(8):
        np.random.seed(seed)
        centers = np.zeros((n, 2))
        idx = 0
        y = 0.04 + np.random.uniform(0, 0.02)
        while idx < n:
            x = 0.04 + np.random.uniform(0, 0.02)
            row_parity = int((y - 0.04) / 0.12) % 2
            if row_parity == 1: 
                x += 0.065
            while x <= 0.96 and idx < n:
                centers[idx] = [x, y]
                idx += 1
                x += 0.135
            y += 0.12
            
        centers = np.clip(centers, 0.01, 0.99)
        radii_init = np.full(n, 0.075)
        vars_init = np.concatenate([centers.flatten(), radii_init])
        bounds = [(0, 1)] * (2*n) + [(0, 0.5)] * n
        
        mu = 100.0
        vars_cur = vars_init.copy()
        for _ in range(14):
            res = minimize(_obj, vars_cur, args=(n, mu), jac=_jac, 
                           method='L-BFGS-B', bounds=bounds, 
                           options={'maxiter': 600, 'ftol': 1e-13, 'gtol': 1e-10})
            vars_cur = res.x
            mu *= 2.2
            
        c = vars_cur[:2*n].reshape(n, 2)
        r = vars_cur[2*n:].copy()
        
        # Strict projection to feasible region
        for _ in range(40):
            for i in range(n):
                lim = min(c[i,0], 1-c[i,0], c[i,1], 1-c[i,1])
                for j in range(n):
                    if i != j:
                        d = np.sqrt(np.sum((c[i]-c[j])**2))
                        lim = min(lim, d - r[j])
                r[i] = min(r[i], max(0, lim))
                
        # Uniform expansion to maximize sum
        for _ in range(15):
            margin = 1.0
            for i in range(n):
                lim = min(c[i,0], 1-c[i,0], c[i,1], 1-c[i,1])
                for j in range(n):
                    if i != j:
                        d = np.sqrt(np.sum((c[i]-c[j])**2))
                        lim = min(lim, d - r[j])
                margin = min(margin, lim - r[i])
            if margin > 1e-9:
                r += margin * 0.9999
            else:
                break
                
        s = np.sum(r)
        if s > best_sum:
            best_sum = s
            best_centers = c.copy()
            best_radii = r.copy()
            
    return best_centers, best_radii, best_sum
