    Pack 26 circles in a unit square to maximize the sum of radii.
    
    Uses a variable transformation to handle boundary constraints implicitly
    and SLSQP to solve the non-convex optimization problem.
    """
    n = 26
    
    # 1. Initialization
    # Create a grid of points to initialize centers.
    # We use a 6x5 grid subset to place 26 points densely.
    # X coordinates: 6 points spaced roughly 1/6 apart, centered in [0,1]

    x_coords = np.linspace(1/12, 11/12, 6) 
    y_coords = np.linspace(1/10, 9/10, 5)
    
    points = []
    # Fill grid row by row
    for y in y_coords:
        for x in x_coords:
            points.append((x, y))
    
    # Select first 26 points
    centers_init = np.array(points[:n])
    
    # Initial radii: small enough to ensure no overlap
    r_init = np.full(n, 0.05)
    
    # 2. Variable Transformation
    # We optimize r, xi, eta where:
    # x = r + (1 - 2*r) * xi
    # y = r + (1 - 2*r) * eta
    # Constraints: 0 <= r <= 0.5, 0 <= xi, eta <= 1
    # This ensures circles are inside the square.
    
    # Invert transformation to find initial xi, eta
    # xi = (x - r) / (1 - 2*r)
    denom = 1 - 2 * r_init
    # Avoid division by zero (r=0.5), though r_init is 0.05
    denom = np.where(denom < 1e-12, 1e-12, denom)

    eta_init = (centers_init[:, 1] - r_init) / denom

    eta_init = np.clip(eta_init, 0, 1)

        r = vars[:n]
        return -np.sum(r)

        xi = vars[n:2*n]
        eta = vars[2*n:3*n]

        min_dist_sq = (r_col + r)**2
        
        # We need upper triangle (i < j)
        # dist_sq is symmetric, min_dist_sq is symmetric
        # Mask upper triangle

        return val

        ])
        radii_final = r_opt
