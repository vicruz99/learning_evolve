import numpy as np
from scipy.optimize import minimize

def run_sim(n, seed):
    """Run a force-directed packing simulation."""
    rng = np.random.RandomState(seed)
    centers = rng.rand(n, 2) * 0.6 + 0.2
    r = 0.04
    dt = 0.005
    stiff = 200.0
    
    for step in range(2500):
        # Vectorized pairwise distances
        diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
        dists = np.linalg.norm(diffs, axis=2)
        np.fill_diagonal(dists, np.inf)
        
        # Repulsive forces from overlaps
        overlap = np.maximum(2 * r - dists, 0)
        dir_vec = np.where(dists > 1e-7, diffs / dists[:, :, np.newaxis], 0.0)
        forces = np.sum(dir_vec * overlap[:, :, np.newaxis], axis=1)
        
        # Boundary repulsion
        for d in range(2):
            viol_low = np.maximum(r - centers[:, d], 0)
            viol_high = np.maximum(centers[:, d] - (1 - r), 0)
            forces[:, d] += stiff * viol_low - stiff * viol_high
            
        # Update positions
        centers += forces * dt
        centers = np.clip(centers, r, 1 - r)
        
        # Gradually expand radii
        r *= 1.0005
        
        # Inject noise to escape local minima
        if step % 200 == 0:
            noise_scale = 0.03 * np.exp(-step / 600)
            centers += rng.randn(n, 2) * noise_scale
            centers = np.clip(centers, r, 1 - r)
            
    # Validate packing
    dists = np.linalg.norm(centers[:, np.newaxis, :] - centers[np.newaxis, :, :], axis=2)
    np.fill_diagonal(dists, np.inf)
    if np.min(dists) >= 2 * r - 1e-5:
        return centers, r
    return None, None

def obj(vars, n):
    """Objective function: maximize radius (minimize negative radius)."""
    return -vars[-1]

def con(vars, n):
    """Constraints: boundaries and non-overlap."""
    cs = vars[:-1].reshape(-1, 2)
    r = vars[-1]
    constraints = []
    
    # Boundary constraints
    for i in range(n):
        constraints.append(cs[i, 0] - r)
        constraints.append(1 - cs[i, 0] - r)
        constraints.append(cs[i, 1] - r)
        constraints.append(1 - cs[i, 1] - r)
        
    # Pairwise distance constraints
    for i in range(n):
        for j in range(i + 1, n):
            d2 = np.sum((cs[i] - cs[j]) ** 2)
            constraints.append(d2 - 4 * r ** 2)
            
    return np.array(constraints)

def run_packing():
    n = 26
    best_centers = None
    best_radii = None
    best_sum = 0.0
    
    # Phase 1: Simulation with multiple seeds
    for s in range(40):
        c, r = run_sim(n, s)
        if c is not None and n * r > best_sum:
            best_centers = c.copy()
            best_radii = np.full(n, r)
            best_sum = n * r
            
    # Phase 2: SLSQP Polish
    if best_centers is not None:
        x0 = np.concatenate([best_centers.ravel(), [best_radii[0]]])
        bnds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)]
        
        try:
            res = minimize(
                obj, x0, args=(n,),
                method='SLSQP',
                bounds=bnds,
                constraints={'type': 'ineq', 'fun': lambda x: con(x, n)},
                options={'maxiter': 5000, 'ftol': 1e-11}
            )
            
            if res.success or res.fun < -best_sum / n:
                new_r = res.x[-1]
                new_c = res.x[:-1].reshape(-1, 2)
                
                # Final validation
                dists = np.linalg.norm(new_c[:, np.newaxis, :] - new_c[np.newaxis, :, :], axis=2)
                np.fill_diagonal(dists, np.inf)
                if np.min(dists) >= 2 * new_r - 1e-5:
                    best_centers = new_c
                    best_radii = np.full(n, new_r)
                    best_sum = n * new_r
        except Exception:
            pass
            
    return best_centers, best_radii, best_sum
