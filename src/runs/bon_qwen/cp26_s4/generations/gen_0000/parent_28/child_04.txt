import numpy as np
from scipy.optimize import minimize, NonlinearConstraint

def compute_constraints(centers, radii):
    """Compute all boundary and non-overlap constraint values. Must be >= 0."""
    n = len(radii)
    cons = []
    
    # Boundary constraints
    for i in range(n):
        cons.append(centers[i, 0] - radii[i])
        cons.append(1.0 - centers[i, 0] - radii[i])
        cons.append(centers[i, 1] - radii[i])
        cons.append(1.0 - centers[i, 1] - radii[i])
        
    # Non-overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist_sq = dx * dx + dy * dy
            sum_r = radii[i] + radii[j]
            cons.append(dist_sq - sum_r**2)
            
    return np.array(cons)

def objective(v):
    """Objective function: negative sum of radii (to be minimized)."""
    return -np.sum(v[52:])

def objective_jac(v):
    """Analytical gradient of the objective function."""
    grad = np.zeros_like(v)
    grad[52:] = -1.0
    return grad

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    best_val = -np.inf
    best_centers = None
    best_radii = None
    
    # Constraint wrapper matching scipy signature
    def constr_func(v):
        c = v[:52].reshape(n, 2)
        r = v[52:]
        return compute_constraints(c, r)
        
    constr = NonlinearConstraint(constr_func, 0, np.inf)
    bounds = [(0, 1)] * 52 + [(0, 0.5)] * n
    opts = {'maxiter': 1000, 'ftol': 1e-10, 'disp': False}
    
    def evaluate_start(init_c, init_r):
        x0 = np.concatenate([init_c.ravel(), init_r])
        res = minimize(objective, x0, method='SLSQP', jac=objective_jac,
                       bounds=bounds, constraints=constr, options=opts)
        return res if res.success else None

    # --- Initialization 1: Hexagonal Lattice ---
    c1 = []
    r_est = 0.065
    for i in range(5):
        for j in range(6):
            if len(c1) >= n: break
            x = 0.12 + j * 2 * r_est + (i % 2) * r_est
            y = 0.12 + i * np.sqrt(3) * r_est
            c1.append([x, y])
    res1 = evaluate_start(np.array(c1[:n]), np.full(n, 0.04))
    if res1 and -res1.fun > best_val:
        best_val = -res1.fun
        best_centers = res1.x[:52].reshape(n, 2)
        best_radii = res1.x[52:]

    # --- Initialization 2: Regular Grid ---
    c2 = np.array(np.meshgrid(np.linspace(0.2, 0.8, 5), np.linspace(0.2, 0.8, 5))).T.reshape(-1, 2)[:n]
    res2 = evaluate_start(c2, np.full(n, 0.04))
    if res2 and -res2.fun > best_val:
        best_val = -res2.fun
        best_centers = res2.x[:52].reshape(n, 2)
        best_radii = res2.x[52:]

    # --- Initialization 3: Random Perturbation ---
    np.random.seed(42)
    c3 = np.random.rand(n, 2) * 0.6 + 0.2
    res3 = evaluate_start(c3, np.full(n, 0.03))
    if res3 and -res3.fun > best_val:
        best_val = -res3.fun
        best_centers = res3.x[:52].reshape(n, 2)
        best_radii = res3.x[52:]

    # Fallback if optimization failed
    if best_centers is None:
        best_centers = np.array(c1[:n]) if len(c1) >= n else c2
        best_radii = np.full(n, 0.02)

    # Post-processing: strict projection to satisfy boundaries and handle numerical tolerance
    for i in range(n):
        r = best_radii[i]
        x, y = best_centers[i]
        # Enforce boundary constraints strictly
        best_radii[i] = min(r, x, 1.0 - x, y, 1.0 - y) - 1e-9
        best_centers[i, 0] = max(best_radii[i], min(1.0 - best_radii[i], x))
        best_centers[i, 1] = max(best_radii[i], min(1.0 - best_radii[i], y))

    return best_centers, best_radii, float(np.sum(best_radii))
