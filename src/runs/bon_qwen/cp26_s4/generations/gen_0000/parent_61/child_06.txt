import math
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    n = 26

    # 5 rows of 5 = 25, plus 1. Or 6, 5, 6, 5, 4.

    points = []
    y_offset = 0.0
    for i, count in enumerate(row_counts):
        x_offset = 0.0
        if i % 2 == 1:
            x_offset = 1.0 # Shift by 1 unit in lattice space (half diameter)
        
        for j in range(count):
            points.append((j * 2.0 + x_offset, y_offset))
        
        y_offset += math.sqrt(3)
    
    # Scale points to fit in [0, 1]

    xs = [p[0] for p in points]
    ys = [p[1] for p in points]
    
    min_x, max_x = min(xs), max(xs)
    min_y, max_y = min(ys), max(ys)
    
    width = max_x - min_x
    height = max_y - min_y
    
    # We want to scale such that the lattice fits with some margin for radii
    # If we set radii to r, the points must be at least r from boundary.

    # So r = 1 / width (approx).

    scale_x = 1.0 / width
    scale_y = 1.0 / height
    scale = min(scale_x, scale_y)
    
    centers = []
    for p in points:
        x = (p[0] - min_x) * scale
        y = (p[1] - min_y) * scale
        centers.append([x, y])
    
    centers = np.array(centers)

    def objective(vars):
        # Maximize sum of radii => Minimize -sum(radii)
        return -np.sum(vars[2::3])

    def penalty(vars):
        centers_opt = np.column_stack((vars[0::3], vars[1::3]))
        radii_opt = vars[2::3]
        
        penalty_val = 0.0
        
        # Boundary constraints
        for i in range(n):
            x, y = centers_opt[i]
            r = radii_opt[i]
            
            # Left
            if x - r < 0: penalty_val += (x - r) ** 2
            # Right
            if x + r > 1: penalty_val += (x + r - 1) ** 2
            # Bottom
            if y - r < 0: penalty_val += (y - r) ** 2
            # Top
            if y + r > 1: penalty_val += (y + r - 1) ** 2
            
        # Overlap constraints
        for i in range(n):
            for j in range(i + 1, n):
                dist_sq = np.sum((centers_opt[i] - centers_opt[j]) ** 2)
                min_dist = radii_opt[i] + radii_opt[j]
                if dist_sq < min_dist ** 2:
                    overlap = min_dist - math.sqrt(dist_sq)
                    penalty_val += overlap ** 2
                    
        return penalty_val

    LAMBDA = 1000.0
    
    def augmented_objective(vars):
        return objective(vars) + LAMBDA * penalty(vars)
