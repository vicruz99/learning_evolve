    # params: [x0, y0, r0, x1, y1, r1, ...]

    # P[i, 0] = x, P[i, 1] = y, P[i, 2] = r
    P = params.reshape((n, 3))
    x = P[:, 0]
    y = P[:, 1]
    r = P[:, 2]
    
    cons = []
    
    # Boundary constraints: x - r >= 0, 1 - x - r >= 0, etc.
    cons.append(x - r)
    cons.append(1.0 - x - r)
    cons.append(y - r)
    cons.append(1.0 - y - r)

    # Vectorized calculation for all pairs i < j

    # x_i - x_j

    dy = y[:, np.newaxis] - y[np.newaxis, :]
    dr = r[:, np.newaxis] + r[np.newaxis, :]

    # np.triu_indices returns indices for upper triangle
    i_idx, j_idx = np.triu_indices(n, k=1)
    sep_cons = diff[i_idx, j_idx]
