import numpy as np

def validate_packing(centers, radii):
    """
    Validate that circles don't overlap and are inside the unit square
    """
    n = centers.shape[0]
    if np.isnan(centers).any() or np.isnan(radii).any():
        return False
    for i in range(n):
        if radii[i] < 0:
            return False
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            return False
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False
    return True

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    np.random.seed(42)
    
    # Initialize centers in a 5x5 grid plus one extra circle
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.05)
    
    k = 0
    for i in range(5):
        for j in range(5):
            centers[k] = [0.1 + i * 0.2, 0.1 + j * 0.2]
            k += 1
    centers[n-1] = [0.5, 0.5] # Overlapping initially, will be resolved by relaxation
    
    growth_step = 0.0005
    max_iters = 8000
    max_relax_iters = 200
    
    for _ in range(max_iters):
        # 1. Grow radii slightly, respecting wall constraints
        for i in range(n):
            x, y = centers[i]
            max_r = min(x, 1-x, y, 1-y)
            current_max = max(radii[i] + growth_step, max_r - 1e-9)
            radii[i] = min(radii[i] + growth_step, current_max)
        
        # 2. Relax overlaps
        for _ in range(max_relax_iters):
            overlaps_found = False
            # Precompute center differences and distances
            # Vectorized pairwise distance
            diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :] # (n, n, 2)
            dists = np.linalg.norm(diff, axis=2)
            
            # Find overlaps
            # i < j to avoid duplicates and self-interaction
            for i in range(n):
                for j in range(i + 1, n):
                    r_sum = radii[i] + radii[j]
                    d = dists[i, j]
                    
                    if d < r_sum and d > 1e-12:
                        overlaps_found = True
                        overlap_amt = r_sum - d
                        # Repulsion vector
                        dx = diff[i, j, 0]
                        dy = diff[i, j, 1]
                        
                        # Push apart proportional to overlap
                        # Move i away from j
                        cx_i, cy_i = centers[i]
                        cx_i += (overlap_amt / 2.0) * (dx / d)
                        cy_i += (overlap_amt / 2.0) * (dy / d)
                        
                        # Move j away from i
                        cx_j, cy_j = centers[j]
                        cx_j -= (overlap_amt / 2.0) * (dx / d)
                        cy_j -= (overlap_amt / 2.0) * (dy / d)
                        
                        # Clamp to walls
                        centers[i, 0] = np.clip(cx_i, radii[i], 1 - radii[i])
                        centers[i, 1] = np.clip(cy_i, radii[i], 1 - radii[i])
                        centers[j, 0] = np.clip(cx_j, radii[j], 1 - radii[j])
                        centers[j, 1] = np.clip(cy_j, radii[j], 1 - radii[j])
            
            if not overlaps_found:
                break
    
    # Final clamp and validation fix for boundary precision
    for i in range(n):
        centers[i, 0] = np.clip(centers[i, 0], radii[i], 1 - radii[i])
        centers[i, 1] = np.clip(centers[i, 1], radii[i], 1 - radii[i])

    sum_radii = np.sum(radii)
    return centers, radii, float(sum_radii)

if __name__ == "__main__":
    c, r, s = run_packing()
