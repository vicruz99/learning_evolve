def _prepare_variables(centers, radii):
    """
    Flatten centers (N, 2) and radii (N,) into a single array of size 3N.
    Order: [x0, x1, ..., xN-1, y0, ..., yN-1, r0, ..., rN-1]
    """

    x_coords = centers[:, 0]
    y_coords = centers[:, 1]
    vars = np.concatenate([x_coords, y_coords, radii])
    return vars

def _extract_vars(vars_array, n=N):
    """
    Extract x, y, r from vars_array.
    """
    cx = vars_array[:n]
    cy = vars_array[n:2*n]
    cr = vars_array[2*n:3*n]
    return cx, cy, cr

def _objective(vars_array, n=N):
    """
    Objective: minimize -sum(radii)
    """
    _, _, cr = _extract_vars(vars_array, n)
    return -np.sum(cr)
