import numpy as np
from scipy.optimize import minimize

# Global storage for constraint evaluation (avoids closures)
_CONSTRAINT_SPECS = []
_CONSTRAINT_N = 0


def objective(vars):
    """Minimize negative sum of radii (equivalent to maximizing sum)."""
    n = _CONSTRAINT_N
    radii = vars[2 * n:]
    return -np.sum(radii)


def constraints_fun(vars):
    """Evaluate all constraints. Each constraint should be >= 0."""
    n = _CONSTRAINT_N
    centers = vars[:2 * n].reshape(n, 2)
    radii = vars[2 * n:]
    results = []

    for spec in _CONSTRAINT_SPECS:
        ctype = spec[0]
        if ctype == 'bxl':
            i = spec[1]
            results.append(centers[i, 0] - radii[i])
        elif ctype == 'bxu':
            i = spec[1]
            results.append(1.0 - centers[i, 0] - radii[i])
        elif ctype == 'byl':
            i = spec[1]
            results.append(centers[i, 1] - radii[i])
        elif ctype == 'byu':
            i = spec[1]
            results.append(1.0 - centers[i, 1] - radii[i])
        elif ctype == 'no':
            i, j = spec[1], spec[2]
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            results.append(dist - radii[i] - radii[j])

    return np.array(results)


def build_constraint_specs(n):
    """Build constraint specification list for n circles."""
    specs = []
    for i in range(n):
        specs.append(('bxl', i))
        specs.append(('bxu', i))
        specs.append(('byl', i))
        specs.append(('byu', i))
    for i in range(n):
        for j in range(i + 1, n):
            specs.append(('no', i, j))
    return specs


def init_hexagonal(n):
    """Initialize centers in a hexagonal-like grid pattern."""
    centers = np.zeros((n, 2))
    idx = 0

    # Try multiple row configurations and pick best fitting
    configs = [
        [5, 4, 5, 4, 5, 3],
        [5, 5, 4, 5, 4, 3],
        [4, 5, 5, 4, 5, 3],
        [5, 4, 5, 4, 4, 4],
        [4, 4, 5, 5, 4, 4],
        [5, 5, 4, 4, 5, 3],
        [3, 5, 5, 5, 5, 3],
        [4, 5, 4, 5, 4, 4],
        [5, 3, 5, 5, 5, 3],
        [4, 5, 4, 5, 4, 4],
    ]

    best_centers = None
    best_spread = -1

    for rows_config in configs:
        if sum(rows_config) != n:
            continue

        c = np.zeros((n, 2))
        idx = 0
        spacing_x = 0.185
        spacing_y = spacing_x * np.sqrt(3) / 2

        for i, count in enumerate(rows_config):
            x_offset = spacing_x / 2 if i % 2 == 1 else 0.06
            for j in range(count):
                if idx < n:
                    c[idx, 0] = x_offset + j * spacing_x
                    c[idx, 1] = 0.05 + i * spacing_y
                    idx += 1

        # Check if all centers are within bounds
        valid = np.all(c >= 0) and np.all(c <= 1)
        if valid:
            # Measure spread (larger is better for initial placement)
            spread = np.mean([np.sqrt(np.sum((c[i] - c[j]) ** 2))
                              for i in range(min(5, n)) for j in range(i + 1, min(5, n))])
            if spread > best_spread:
                best_spread = spread
                best_centers = c.copy()

    if best_centers is None:
        # Fallback to simple grid
        cols = int(np.ceil(np.sqrt(n)))
        rows = int(np.ceil(n / cols))
        dx = 1.0 / (cols + 1)
        dy = 1.0 / (rows + 1)
        idx = 0
        for i in range(rows):
            for j in range(cols):
                if idx < n:
                    best_centers[idx, 0] = (j + 1) * dx
                    best_centers[idx, 1] = (i + 1) * dy
                    idx += 1

    return best_centers


def force_directed_optimize(centers, radii, n, iterations=20000):
    """Optimize using force-directed relaxation."""
    centers = centers.copy()
    radii = radii.copy()

    for iteration in range(iterations):
        forces = np.zeros_like(centers)

        # Pairwise repulsive forces
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)

                if dist < 1e-10:
                    dx, dy = 0.001, 0.001
                    dist = 0.00141421

                r_sum = radii[i] + radii[j]
                ideal_dist = r_sum * 1.3  # Target distance is 30% larger than touching

                if dist < r_sum:
                    # Strong repulsion for overlaps
                    overlap = r_sum - dist
                    fx = overlap * dx / dist * 500
                    fy = overlap * dy / dist * 500
                elif dist < ideal_dist:
                    # Moderate repulsion when too close
                    force = (ideal_dist - dist) * 50
                    fx = force * dx / dist
                    fy = force * dy / dist
                else:
                    fx, fy = 0.0, 0.0

                forces[i, 0] += fx
                forces[i, 1] += fy
                forces[j, 0] -= fx
                forces[j, 1] -= fy

        # Boundary forces (push away from edges)
        for i in range(n):
            for dim in range(2):
                margin = radii[i] + 0.002
                if centers[i, dim] < margin:
                    forces[i, dim] += (margin - centers[i, dim]) * 800
                elif centers[i, dim] > 1.0 - margin:
                    forces[i, dim] -= (centers[i, dim] - (1.0 - margin)) * 800

        # Apply forces with adaptive learning rate
        lr = 0.002 / (1.0 + iteration * 0.0001)
        centers = centers + forces * lr

        # Project to valid range
        for i in range(n):
            for dim in range(2):
                centers[i, dim] = max(radii[i] + 1e-8, min(1.0 - radii[i] - 1e-8, centers[i, dim]))

        # Gradually grow radii
        for i in range(n):
            max_r = min(centers[i, 0], 1.0 - centers[i, 0],
                        centers[i, 1], 1.0 - centers[i, 1])
            for j in range(n):
                if i != j:
                    dx = centers[i, 0] - centers[j, 0]
                    dy = centers[i, 1] - centers[j, 1]
                    dist = np.sqrt(dx * dx + dy * dy)
                    max_r = min(max_r, dist - radii[j])

            target = max(max_r * 0.99, 1e-6)
            radii[i] = radii[i] + (target - radii[i]) * 0.005
            radii[i] = max(radii[i], 1e-6)

    return centers, radii


def cleanup(centers, radii, n):
    """Ensure all constraints are satisfied."""
    centers = centers.copy()
    radii = radii.copy()

    # Fix overlaps by reducing radii
    for _ in range(200):
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                if dist < radii[i] + radii[j] - 1e-12:
                    excess = radii[i] + radii[j] - dist
                    radii[i] -= excess * 0.51
                    radii[j] -= excess * 0.51
                    radii[i] = max(radii[i], 1e-10)
                    radii[j] = max(radii[j], 1e-10)

    # Fix boundary violations
    for i in range(n):
        for dim in range(2):
            if centers[i, dim] - radii[i] < -1e-12:
                radii[i] = centers[i, dim] - 1e-8
            if centers[i, dim] + radii[i] > 1.0 + 1e-12:
                radii[i] = 1.0 - centers[i, dim] - 1e-8

    radii = np.maximum(radii, 1e-10)

    return centers, radii


def run_packing():
    global _CONSTRAINT_SPECS, _CONSTRAINT_N

    n = 26
    _CONSTRAINT_N = n
    _CONSTRAINT_SPECS = build_constraint_specs(n)

    best_centers = None
    best_radii = None
    best_sum = -1.0

    # Try multiple random seeds and configurations
    for seed in range(20):
        np.random.seed(seed)

        # Initialize centers
        centers = init_hexagonal(n)

        # Add small random perturbation
        centers = centers + np.random.uniform(-0.02, 0.02, centers.shape)
        centers = np.clip(centers, 0.05, 0.95)

        # Initial radii
        radii = np.ones(n) * 0.035

        # Force-directed optimization
        centers, radii = force_directed_optimize(centers, radii, n, iterations=15000)

        # Clean up to ensure feasibility
        centers, radii = cleanup(centers, radii, n)

        # Check feasibility
        feasible = True
        for i in range(n):
            for dim in range(2):
                if centers[i, dim] - radii[i] < -1e-10 or centers[i, dim] + radii[i] > 1.0 + 1e-10:
                    feasible = False
                    break
            if not feasible:
                break

        if feasible:
            for i in range(n):
                for j in range(i + 1, n):
                    dx = centers[i, 0] - centers[j, 0]
                    dy = centers[i, 1] - centers[j, 1]
                    dist = np.sqrt(dx * dx + dy * dy)
                    if dist < radii[i] + radii[j] - 1e-10:
                        feasible = False
                        break
                if not feasible:
                    break

        if not feasible:
            continue

        # Gradient-based refinement using SLSQP
        x0 = np.concatenate([centers.flatten(), radii])

        bounds = []
        for i in range(n):
            bounds.extend([(0.0, 1.0), (0.0, 1.0)])
            bounds.append((0.0, 0.5))

        cons = {'type': 'ineq', 'fun': constraints_fun}

        try:
            result = minimize(objective, x0, method='SLSQP',
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 500, 'ftol': 1e-14, 'disp': False})
            x_opt = result.x
        except Exception:
            x_opt = x0

        c_opt = x_opt[:2 * n].reshape(n, 2)
        r_opt = x_opt[2 * n:]

        # Final cleanup
        c_opt, r_opt = cleanup(c_opt, r_opt, n)

        current_sum = np.sum(r_opt)
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = c_opt.copy()
            best_radii = r_opt.copy()

    # Additional refinement: try to equalize radii locally
    if best_centers is not None:
        for pass_num in range(5):
            improved = False
            for i in range(n):
                # Try swapping radii with neighbors to improve sum
                for j in range(i + 1, n):
                    dx = best_centers[i, 0] - best_centers[j, 0]
                    dy = best_centers[i, 1] - best_centers[j, 1]
                    dist = np.sqrt(dx * dx + dy * dy)
                    current_pair = best_radii[i] + best_radii[j]
                    # They should be touching or very close
                    if dist > current_pair + 0.001:
                        # There's gap - try to grow both
                        gap = (dist - current_pair) / 2
                        new_r_i = best_radii[i] + gap * 0.3
                        new_r_j = best_radii[j] + gap * 0.3
                        # Check if this doesn't violate other constraints
                        ok = True
                        for k in range(n):
                            if k == i or k == j:
                                continue
                            if k != i:
                                dxk = best_centers[i, 0] - best_centers[k, 0]
                                dyk = best_centers[i, 1] - best_centers[k, 1]
                                distk = np.sqrt(dxk * dxk + dyk * dyk)
                                if distk < new_r_i + best_radii[k] + 1e-10:
                                    ok = False
                                    break
                            if ok and k != j:
                                dxk = best_centers[j, 0] - best_centers[k, 0]
                                dyk = best_centers[j, 1] - best_centers[k, 1]
                                distk = np.sqrt(dxk * dxk + dyk * dyk)
                                if distk < new_r_j + best_radii[k] + 1e-10:
                                    ok = False
                                    break
                            if not ok:
                                break

                        if ok:
                            # Check boundary constraints
                            for circ, new_r in [(i, new_r_i), (j, new_r_j)]:
                                for dim in range(2):
                                    if best_centers[circ, dim] - new_r < -1e-10 or best_centers[circ, dim] + new_r > 1.0 + 1e-10:
                                        ok = False
                                if not ok:
                                    break

                        if ok:
                            best_radii[i] = new_r_i
                            best_radii[j] = new_r_j
                            improved = True

            if not improved:
                break

    if best_centers is None:
        # Absolute fallback
        best_centers = init_hexagonal(n)
        best_radii = np.ones(n) * 0.02
        best_sum = np.sum(best_radii)

    return best_centers, best_radii, np.sum(best_radii)
