    *   **Constraints**: We enforce boundary constraints and non-overlap constraints using inequality constraints in the optimizer. We also enforce non-negative radii.
    *   **Validation**: After optimization, we perform a strict numerical validation to ensure the solution satisfies all constraints within a tight tolerance (matching the validation function's requirements).

3.  **Fallback**: If the optimization fails to find a valid packing better than a baseline, we return a constructive solution (a $5 \times 5$ grid of radius 0.1 circles plus a small circle in a gap).

This approach balances the need for high density (favoring hexagonal arrangements) with the flexibility to adjust positions and radii to fit the specific square boundary.
</strategy>

```python
import numpy as np
import scipy.optimize as opt
import math

def run_packing():
    """
    Pack 26 circles in a unit square to maximize the sum of radii.
    
    Returns:
        centers: np.array of shape (26, 2)
        radii: np.array of shape (26,)
        sum_radii: float
    """
    N = 26
    best_sum = -1.0
    best_centers = None
    best_radii = None
    
    # Number of restarts to explore the solution space
    # Multiple restarts help escape local minima
    num_restarts = 30
    
    for restart in range(num_restarts):
        # Initialization Strategy:
        # Use a hexagonal grid pattern perturbed by noise.
        # Hexagonal packing is the densest circle packing in a plane.
        
        # Base radius for grid generation. 
        # We use a radius slightly larger than expected to space points out,
        # then the optimizer will shrink them if needed or keep them if they fit.
        # However, for feasibility, we start with small radii in the optimizer.
        # Here we just generate centers.
        r_base = 0.11 
        dx = 2 * r_base
        dy = math.sqrt(3) * r_base
        
        points = []
        # Generate a grid covering the unit square
        # We generate enough points to select N from
        for r in range(8):
            for c in range(8):
                x = c * dx + (r % 2) * (dx / 2)
                y = r * dy
                points.append([x, y])
        
        # Select points that are well inside the square to start safely
        valid_points = [p for p in points if 0.15 <= p[0] <= 0.85 and 0.15 <= p[1] <= 0.85]
        
        np.random.seed(restart)
        
        if len(valid_points) < N:
            # Fallback to random placement if grid is sparse (unlikely with these params)
            centers_init = np.random.uniform(0.2, 0.8, (N, 2))
        else:
            # Randomly select N points and add small noise to break symmetry
            indices = np.random.choice(len(valid_points), N, replace=False)
            centers_init = np.array([valid_points[i] for i in indices])
            centers_init += np.random.normal(0, 0.005, (N, 2))
            centers_init = np.clip(centers_init, 0.1, 0.9)
            
        # Start with small radii to ensure initial feasibility (no overlaps)
        radii_init = np.full(N, 0.04)
        
        # Variable vector: [x1, y1, ..., xN, yN, r1, ..., rN]
        # Total 3N variables
        x0 = np.zeros(3 * N)
        x0[:2*N] = centers_init.flatten()
        x0[2*N:] = radii_init
        
        # Objective: Maximize sum of radii => Minimize -sum(radii)
        def objective(v):
            return -np.sum(v[2*N:])
        
        # Constraints setup
        constraints = []
        
        # 1. Boundary constraints: r <= x <= 1-r, r <= y <= 1-r
        # Equivalent to: x - r >= 0, 1 - x - r >= 0, etc.
        for i in range(N):
            # x_i - r_i >= 0
            constraints.append({'type': 'ineq', 'fun': lambda v, i=i: v[2*i] - v[2*N + i]})
            # 1 - x_i - r_i >= 0
            constraints.append({'type': 'ineq', 'fun': lambda v, i=i: 1.0 - v[2*i] - v[2*N + i]})
            # y_i - r_i >= 0
            constraints.append({'type': 'ineq', 'fun': lambda v, i=i: v[2*i + 1] - v[2*N + i]})
            # 1 - y_i - r_i >= 0
            constraints.append({'type': 'ineq', 'fun': lambda v, i=i: 1.0 - v[2*i + 1] - v[2*N + i]})
        
        # 2. Non-overlap constraints: d_ij^2 >= (r_i + r_j)^2
        # (xi-xj)^2 + (yi-yj)^2 - (ri+rj)^2 >= 0
        for i in range(N):
            for j in range(i + 1, N):
                constraints.append({
                    'type': 'ineq',
                    'fun': lambda v, i=i, j=j: \
                        (v[2*i] - v[2*j])**2 + (v[2*i+1] - v[2*j+1])**2 - (v[2*N+i] + v[2*N+j])**2
                })
        
        # 3. Non-negative radii
        for i in range(N):
            constraints.append({'type': 'ineq', 'fun': lambda v, i=i: v[2*N + i]})
            
        # Bounds for variables
        bounds = []
        for i in range(N):
            bounds.append((0.0, 1.0)) # x in [0, 1]
            bounds.append((0.0, 1.0)) # y in [0, 1]
            bounds.append((0.0, 0.5)) # r in [0, 0.5] (radius cannot exceed 0.5)
            
        try:
            # Run optimization
            res = opt.minimize(
                objective,
                x0,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 3000, 'ftol': 1e-13, 'disp': False}
            )
            
            if res.success:
                current_sum = -res.fun
                # Only update if we found a better valid sum
                if current_sum > best_sum:
                    c = res.x[:2*N].reshape(N, 2)
                    r = res.x[2*N:]
                    
                    # Strict validation check to ensure numerical stability
                    # matching the validation function's tolerance
                    valid = True
                    if np.any(r < 0): 
                        valid = False
                    else:
                        for i in range(N):
                            # Check boundaries
                            if c[i,0] < r[i] - 1e-9 or c[i,0] > 1 - r[i] + 1e-9: valid = False; break
                            if c[i,1] < r[i] - 1e-9 or c[i,1] > 1 - r[i] + 1e-9: valid = False; break
                    
                    if valid:
                        # Check overlaps
                        for i in range(N):
                            for j in range(i+1, N):
                                d2 = np.sum((c[i]-c[j])**2)
                                # Check if distance is strictly greater than sum of radii (with tolerance)
                                if d2 < (r[i]+r[j])**2 - 1e-9:
                                    valid = False; break
                            if not valid: break
                    
                    if valid:
                        best_sum = current_sum
                        best_centers = c
                        best_radii = r
        except Exception:
            pass

    # Fallback solution if optimization fails to find a valid packing
    # This is a safe baseline: 5x5 grid of radius 0.1 circles + 1 small circle
    if best_centers is None:
        centers = []
        radii = []
        # 5x5 grid
        for i in range(5):
            for j in range(5):
                centers.append([0.1 + i*0.2, 0.1 + j*0.2])
                radii.append(0.1)
        # 26th circle in a gap (e.g., at 0.2, 0.2)
        centers.append([0.2, 0.2])
        radii.append(0.04)
        best_centers = np.array(centers)
        best_radii = np.array(radii)
        best_sum = np.sum(best_radii)
        
    if best_radii is not None:
        best_radii = np.maximum(best_radii, 0.0)
        
    return best_centers, best_radii, best_sum
```
