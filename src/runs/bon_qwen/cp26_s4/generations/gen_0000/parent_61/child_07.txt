import numpy as np
from scipy.optimize import minimize

def compute_objective_and_penalty(vars, K):
    """
    Computes the objective (-sum of radii) and the penalty for constraint violations.
    
    Args:
        vars: 1D numpy array of shape (3 * n) containing [x0, y0, x1, y1, ..., r0, r1, ...]
        K: Penalty weight scalar.
    
    Returns:
        objective_value: float, -sum(radii) + K * total_penalty
    """
    n = 26
    # Split variables into centers and radii
    centers = vars[:2 * n].reshape((n, 2))
    radii = vars[2 * n:]
    
    # Base objective: maximize sum of radii (so minimize negative sum)
    obj = -np.sum(radii)
    penalty = 0.0
    
    # 1. Boundary Penalties
    # Ensure circles are within [0, 1] x [0, 1]
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        
        # Left wall: x >= r  => violation if r - x > 0
        val = r - x
        if val > 0:
            penalty += val ** 2
            
        # Right wall: x <= 1 - r => violation if x + r - 1 > 0
        val = x + r - 1.0
        if val > 0:
            penalty += val ** 2
            
        # Bottom wall: y >= r => violation if r - y > 0
        val = r - y
        if val > 0:
            penalty += val ** 2
            
        # Top wall: y <= 1 - r => violation if y + r - 1 > 0
        val = y + r - 1.0
        if val > 0:
            penalty += val ** 2

    # 2. Overlap Penalties
    # Ensure distance >= r_i + r_j
    # We use a small epsilon to avoid division by zero or singularity in gradients
    eps = 1e-12
    for i in range(n):
        xi, yi, ri = centers[i, 0], centers[i, 1], radii[i]
        # Optimization: radii cannot be negative
        if ri < 0: ri = 0 
        
        for j in range(i + 1, n):
            xj, yj, rj = centers[j, 0], centers[j, 1], radii[j]
            if rj < 0: rj = 0
            
            dist = np.sqrt((xi - xj) ** 2 + (yi - yj) ** 2 + eps)
            required_dist = ri + rj
            
            if dist < required_dist:
                # Penalty for overlap
                diff = required_dist - dist
                penalty += diff ** 2
                
    return obj + K * penalty

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
