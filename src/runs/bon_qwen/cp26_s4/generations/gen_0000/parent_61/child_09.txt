

<strategy>
To maximize the sum of radii for 26 circles in a unit square, we leverage two key insights:
1. **Hexagonal Packing Optimality**: Dense circle packings in the plane converge to a hexagonal lattice. While boundary effects distort this near the square edges, starting from a perturbed hexagonal configuration provides an excellent initial guess close to the global optimum.
2. **Sequential Quadratic Programming (SLSQP)**: The problem is a non-convex constrained optimization. We formulate it as maximizing $\sum r_i$ subject to pairwise distance constraints $(x_i-x_j)^2 + (y_i-y_j)^2 \ge (r_i+r_j)^2$ and boundary constraints $r_i \le x_i, y_i \le 1-r_i$. SLSQP efficiently handles these inequality constraints.

The implementation strategy:
- Initialize 26 centers on a scaled hexagonal grid within $[0,1]^2$.
- Compute initial radii as a fraction of nearest-neighbor distances to guarantee a strictly feasible starting point.
- Flatten variables to interleaved format `[x1, y1, r1, x2, y2, r2, ...]` for vectorized constraint evaluation.
- Use `scipy.optimize.minimize` with SLSQP to drive radii upward until constraints bind.
- Apply a post-processing relaxation step to guarantee numerical compliance with the validation function's tolerance ($10^{-12}$).
- Define all helper functions at the top level as requested.

import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26

def objective(v):
    """Objective: maximize sum of radii -> minimize negative sum."""
    return -np.sum(v[2::3])

def constraints(v):
    """Returns inequality constraints >= 0: non-overlap and boundary containment."""
    n = N_CIRCLES
    c = v.reshape(-1, 3)[:, :2]
    r = v.reshape(-1, 3)[:, 2]
    
    # Pairwise non-overlap: dist^2 >= (r_i + r_j)^2
    diff = c[:, None, :] - c[None, :, :]
    dist_sq = np.sum(diff**2, axis=2)
    rad_sum_sq = (r[:, None] + r[None, :])**2
    con_overlap = dist_sq - rad_sum_sq
    
    # Extract upper triangle to avoid duplicates
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    c_overlap = con_overlap[mask]
    
    # Boundary constraints: r <= x, r <= y, r <= 1-x, r <= 1-y
    c_bound = np.concatenate([
        c[:, 0] - r,
        c[:, 1] - r,
        1 - c[:, 0] - r,
        1 - c[:, 1] - r
    ])
    
    return np.concatenate([c_overlap, c_bound])

def run_packing():
    n = N_CIRCLES
    
    # 1. Generate initial hexagonal lattice configuration
    dx = 0.2
    dy = dx * np.sqrt(3) / 2
    points = []
    y = 0.1
    row = 0
    while y <= 0.9 and len(points) < n:
        x_start = 0.1 + (0.5 if row % 2 else 0) * dx
        x = x_start
        while x <= 0.9 and len(points) < n:
            points.append([x, y])
            x += dx
        y += dy
        row += 1
        
    centers_init = np.array(points[:n])
    
    # 2. Compute valid initial radii based on nearest neighbors
    dists = np.linalg.norm(centers_init[:, None, :] - centers_init[None, :, :], axis=2)
    np.fill_diagonal(dists, np.inf)
    min_dists = np.min(dists, axis=1)
    radii_init = 0.4 * min_dists
    
    # Ensure boundary validity for initial radii
    bounds_init = np.minimum(np.minimum(centers_init[:, 0], 1 - centers_init[:, 0]),
                             np.minimum(centers_init[:, 1], 1 - centers_init[:, 1]))
    radii_init = np.minimum(radii_init, 0.9 * bounds_init)
    
    # 3. Flatten to interleaved format: [x1, y1, r1, x2, y2, r2, ...]
    v0 = np.zeros(3 * n)
    v0[0::3] = centers_init[:, 0]
    v0[1::3] = centers_init[:, 1]
    v0[2::3] = radii_init
    
    # Variable bounds: x,y in [0,1], r in [0, 0.5]
    bounds_opt = [(0, 1) if i % 3 != 2 else (0, 0.5) for i in range(3 * n)]
    
    # 4. Run optimization
    cons = {'type': 'ineq', 'fun': constraints}
    res = minimize(objective, v0, method='SLSQP', bounds=bounds_opt, constraints=cons, 
                   options={'maxiter': 2000, 'ftol': 1e-12, 'disp': False})
                   
    v_opt = res.x
    centers = v_opt.reshape(-1, 3)[:, :2].copy()
    radii = v_opt.reshape(-1, 3)[:, 2].copy()
    
    # 5. Post-processing to guarantee validity against numerical errors
    radii = np.maximum(radii, 0.0)
    centers[:, 0] = np.clip(centers[:, 0], radii, 1 - radii)
    centers[:, 1] = np.clip(centers[:, 1], radii, 1 - radii)
    
    # Final safety check for overlaps (resolve by slightly shrinking if necessary)
    for _ in range(5):
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.linalg.norm(centers[i] - centers[j])
                sum_r = radii[i] + radii[j]
                if dist < sum_r - 1e-14:
                    scale = (dist + 1e-14) / sum_r
                    radii[i] *= scale
                    radii[j] *= scale
                    
    return centers, radii, float(np.sum(radii))
