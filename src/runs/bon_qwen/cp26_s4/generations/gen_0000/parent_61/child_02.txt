import numpy as np
from scipy.optimize import minimize

def compute_loss_and_grad(vars, n, lambda_p):
    """
    Computes the loss (negative sum of radii) plus penalty for constraint violations,
    and its analytical gradient.
    """
    centers = vars[:2*n].reshape(n, 2)
    radii = vars[2*n:]

    # Objective: maximize sum of radii => minimize -sum(radii)
    loss = -np.sum(radii)
    grad = np.zeros(2*n + n)
    grad[2*n:] = -1.0

    # Penalty weight scaling factor
    scale = lambda_p

    # --- Boundary Constraints ---
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        
        # Left: x - r >= 0
        if x - r < 0:
            err = r - x
            loss += scale * err**2
            grad[2*i] -= 2 * scale * err      # d/dx
            grad[2*n + i] += 2 * scale * err  # d/dr
            
        # Right: x + r <= 1
        if x + r > 1:
            err = x + r - 1
            loss += scale * err**2
            grad[2*i] += 2 * scale * err
            grad[2*n + i] += 2 * scale * err
            
        # Bottom: y - r >= 0
        if y - r < 0:
            err = r - y
            loss += scale * err**2
            grad[2*i+1] -= 2 * scale * err
            grad[2*n + i] += 2 * scale * err
            
        # Top: y + r <= 1
        if y + r > 1:
            err = y + r - 1
            loss += scale * err**2
            grad[2*i+1] += 2 * scale * err
            grad[2*n + i] += 2 * scale * err

    # --- Overlap Constraints ---
    # Vectorized distance computation
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2) + 1e-12)
    sum_r = radii[:, None] + radii[None, :]
    
    # Only consider upper triangle pairs (i < j)
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    d = dists[mask]
    sr = sum_r[mask]
    
    violations = sr - d
    pos_viol = np.maximum(violations, 0.0)
    
    # Add penalty to loss
    loss += scale * np.sum(pos_viol**2)
    
    # Compute gradients for violations
    # For each violating pair, add gradient contributions
    idx_i, idx_j = np.where(mask)
    for k in range(len(idx_i)):
        if pos_viol[k] > 0:
            v = pos_viol[k]
            i, j = idx_i[k], idx_j[k]
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = d[k]
            
            # Direction vectors
            ux = dx / dist
            uy = dy / dist
            
            # Gradient contribution: 2 * v * (dr/dparam)
            # d/d(c_i) = 2 * v * (c_i - c_j) / dist
            # d/d(c_j) = 2 * v * (c_j - c_i) / dist
            # d/d(r_i) = 2 * v
            # d/d(r_j) = 2 * v
            
            factor = 2 * scale * v
            
            grad[2*i] += factor * ux
            grad[2*i+1] += factor * uy
            grad[2*j] -= factor * ux
            grad[2*j+1] -= factor * uy
            grad[2*n + i] += factor
            grad[2*n + j] += factor

    return loss, grad
