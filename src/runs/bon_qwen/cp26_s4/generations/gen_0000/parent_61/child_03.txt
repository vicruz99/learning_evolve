import numpy as np
from scipy.optimize import minimize

def objective_function(params, n):
    """
    Objective function: -sum(radii) + penalty for constraints.
    We minimize this function.
    """
    # Reshape params to centers and radii
    centers = params[:2*n].reshape(n, 2)
    radii = params[2*n:]
    
    # Sum of radii (to be maximized, so negative in minimization)
    sum_radii = -np.sum(radii)
    
    penalty = 0.0
    
    # 1. Boundary constraints
    # x - r >= 0 => r - x <= 0
    # 1 - x - r >= 0 => x + r - 1 <= 0
    # Same for y
    
    # Penalty for x < r
    violation_x_left = np.maximum(0, radii - centers[:, 0])
    penalty += np.sum(violation_x_left ** 2)
    
    # Penalty for x + r > 1
    violation_x_right = np.maximum(0, centers[:, 0] + radii - 1)
    penalty += np.sum(violation_x_right ** 2)
    
    # Penalty for y < r
    violation_y_bottom = np.maximum(0, radii - centers[:, 1])
    penalty += np.sum(violation_y_bottom ** 2)
    
    # Penalty for y + r > 1
    violation_y_top = np.maximum(0, centers[:, 1] + radii - 1)
    penalty += np.sum(violation_y_top ** 2)
    
    # 2. Overlap constraints
    # dist(i, j) >= r_i + r_j
    # dist^2 >= (r_i + r_j)^2
    # We check 1D distances or squared distance. 
    # Using distance directly is smoother for penalty usually, but squared is fine.
    # Let's use distance.
    
    # Vectorized calculation for overlaps
    # dist matrix
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff ** 2, axis=2))
    
    # Mask out diagonal and lower triangle to check each pair once
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    relevant_dists = dists[mask]
    relevant_r_sum = (radii[:, None] + radii[None, :])[mask]
    
    # Violation: r_i + r_j - dist
    violations = np.maximum(0, relevant_r_sum - relevant_dists)
    penalty += np.sum(violations ** 2)
    
    # Weight the penalty heavily
    penalty_weight = 1000.0
    
    return sum_radii + penalty_weight * penalty
