import numpy as np
from scipy.optimize import minimize

def compute_objective(vars):
    """Objective: minimize negative sum of radii."""
    radii = vars[2::3]
    return -np.sum(radii)

def compute_constraints(vars):
    """Returns a 1D array of all constraint values. All must be >= 0."""
    n = 26
    c = []
    
    # Boundary and non-negativity constraints
    for i in range(n):
        x = vars[3*i]
        y = vars[3*i + 1]
        r = vars[3*i + 2]
        c.append(x - r)
        c.append(1.0 - x - r)
        c.append(y - r)
        c.append(1.0 - y - r)
        c.append(r)
        
    # Pairwise non-overlap constraints: dist^2 >= (r_i + r_j)^2
    for i in range(n):
        xi, yi, ri = vars[3*i], vars[3*i + 1], vars[3*i + 2]
        for j in range(i + 1, n):
            xj, yj, rj = vars[3*j], vars[3*j + 1], vars[3*j + 2]
            dx = xi - xj
            dy = yi - yj
            dr = ri + rj
            c.append(dx*dx + dy*dy - dr*dr)
            
    return np.array(c)

def run_packing():
    n = 26
    
    # Hexagonal-ish initialization for better convergence
    centers = np.zeros((n, 2))
    counts = [6, 5, 6, 5, 4]
    row_y = [0.2, 0.4, 0.5, 0.6, 0.8]
    idx = 0
    for i, y in enumerate(row_y):
        ncols = counts[i]
        if ncols > 0:
            xs = np.linspace(0.1, 0.9, ncols)
        else:
            xs = []
        for x in xs:
            if idx < n:
                centers[idx] = [x, y]
                idx += 1
                
    # Start with small feasible radii
    radii = np.full(n, 0.05)
    x0 = np.concatenate([centers.ravel(), radii])
    
    # Setup constraints and bounds
    cons = {'type': 'ineq', 'fun': compute_constraints}
    bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)] * n
    
    # Run SLSQP optimizer
    res = minimize(compute_objective, x0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': 1500, 'ftol': 1e-11, 'disp': False})
                   
    final_centers = res.x[:2*n].reshape(n, 2)
    final_radii = res.x[2*n:]
    
    # Post-processing to guarantee strict validity against numerical solver tolerances
    # 1. Clamp to boundaries
    for i in range(n):
        x, y = final_centers[i]
        r = final_radii[i]
        r = min(r, x, 1.0 - x, y, 1.0 - y)
        r = max(r, 0.0)
        final_radii[i] = r
        final_centers[i] = [x, y]
        
    # 2. Resolve any residual overlaps by shrinking
    for _ in range(20):
        for i in range(n):
            for j in range(i + 1, n):
                dx = final_centers[i, 0] - final_centers[j, 0]
                dy = final_centers[i, 1] - final_centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                r_sum = final_radii[i] + final_radii[j]
                if dist < r_sum - 1e-12:
                    shrink = (r_sum - dist) / 2.0 + 1e-7
                    final_radii[i] = max(0.0, final_radii[i] - shrink)
                    final_radii[j] = max(0.0, final_radii[j] - shrink)
                    
        # Re-clamp boundaries after shrinking
        for i in range(n):
            x, y = final_centers[i]
            r = final_radii[i]
            r = min(r, x, 1.0 - x, y, 1.0 - y)
            final_radii[i] = r
            
    return final_centers, final_radii, np.sum(final_radii)
