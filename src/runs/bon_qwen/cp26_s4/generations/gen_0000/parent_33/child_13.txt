def pos_obj(pos, fixed_r, n):
    """Penalty function for position optimization with fixed radii."""
    x, y = pos[:n], pos[n:]
    loss = 0.0
    for i in range(n):
        ri = fixed_r[i]
        # Boundary penalties
        if x[i] < ri: loss += (ri - x[i])**2
        if x[i] > 1 - ri: loss += (x[i] - (1 - ri))**2
        if y[i] < ri: loss += (ri - y[i])**2
        if y[i] > 1 - ri: loss += (y[i] - (1 - ri))**2

            d = np.sqrt((x[i] - x[j])**2 + (y[i] - y[j])**2)
            overlap = fixed_r[i] + fixed_r[j] - d
            if overlap > 0:
                loss += overlap**2
    return loss
