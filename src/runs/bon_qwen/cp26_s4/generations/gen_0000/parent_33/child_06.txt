import numpy as np
from scipy.optimize import minimize

N_CIRCLES = 26
PENALTY_WEIGHT = 5000


def hexagonal_initial(n):
    """Generate initial hexagonal packing positions for n circles."""
    centers = []
    # Pattern: 5-4-5-4-5-3 = 26 circles in 6 rows
    row_counts = [5, 4, 5, 4, 5, 3]
    num_rows = len(row_counts)

    # Space rows evenly with some margin
    y_positions = np.linspace(0.1, 0.9, num_rows)

    for row_idx, count in enumerate(row_counts):
        y = y_positions[row_idx]

        if count == 1:
            x_positions = [0.5]
        else:
            margin = 0.08
            x_spacing = (1 - 2 * margin) / (count - 1)
            x_positions = [margin + k * x_spacing for k in range(count)]

        # Offset odd rows for hexagonal pattern
        if row_idx % 2 == 1 and count > 1:
            offset = x_spacing / 2
            x_positions = [x + offset for x in x_positions]
            x_positions = [max(0.05, min(0.95, x)) for x in x_positions]

        for x in x_positions:
            centers.append([x, y])

    return np.array(centers[:n])


def apply_forces(centers, radii, lr=0.005, n_iters=100):
    """Apply repulsive forces to resolve overlaps and boundary violations."""
    n = len(centers)
    centers = centers.copy().astype(np.float64)

    for _ in range(n_iters):
        forces = np.zeros((n, 2), dtype=np.float64)

        # Repulsive forces between overlapping circles
        for i in range(n):
            for j in range(i + 1, n):
                diff = centers[j] - centers[i]
                dist_sq = np.sum(diff ** 2)
                dist = np.sqrt(max(dist_sq, 1e-20))
                min_dist = radii[i] + radii[j]

                if dist < min_dist:
                    overlap = min_dist - dist
                    force_mag = overlap * 10.0 / dist
                    forces[i] -= force_mag * diff
                    forces[j] += force_mag * diff

        # Boundary repulsion forces
        for i in range(n):
            x, y = centers[i]
            r = radii[i]

            if x < r:
                forces[i, 0] += (r - x) * 50
            if x > 1 - r:
                forces[i, 0] -= (x - (1 - r)) * 50
            if y < r:
                forces[i, 1] += (r - y) * 50
            if y > 1 - r:
                forces[i, 1] -= (y - (1 - r)) * 50

        centers += lr * forces

        # Clip to valid range
        for i in range(n):
            r = radii[i]
            centers[i, 0] = np.clip(centers[i, 0], r, 1 - r)
            centers[i, 1] = np.clip(centers[i, 1], r, 1 - r)

    return centers


def check_validity(centers, radii, tol=1e-9):
    """Check if packing satisfies all constraints."""
    n = len(centers)

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x < r - tol or x > 1 - r + tol or y < r - tol or y > 1 - r + tol:
            return False

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - tol:
                return False

    return True


def compute_penalty(params):
    """Compute penalty for constraint violations."""
    n = N_CIRCLES
    centers = params[:2 * n].reshape(n, 2)
    radii = params[2 * n:]

    penalty = 0.0
    w = PENALTY_WEIGHT

    # Boundary penalties (squared violations)
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x < r:
            penalty += w * (r - x) ** 2
        if x > 1 - r:
            penalty += w * (x - (1 - r)) ** 2
        if y < r:
            penalty += w * (r - y) ** 2
        if y > 1 - r:
            penalty += w * (y - (1 - r)) ** 2

    # Overlap penalties
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            min_dist = radii[i] + radii[j]
            if dist < min_dist:
                penalty += w * (min_dist - dist) ** 2

    return penalty


def objective(params):
    """Objective function: negative sum of radii plus penalty."""
    n = N_CIRCLES
    radii = params[2 * n:]
    return -np.sum(radii) + compute_penalty(params)


def iterative_growth(centers, radii, max_steps=600, growth_rate=1.003):
    """Gradually grow circles while maintaining valid packing."""
    n = len(centers)

    for step in range(max_steps):
        # Grow all radii
        radii *= growth_rate

        # Optimize positions
        centers = apply_forces(centers, radii, lr=0.008, n_iters=150)

        # Check if still valid
        if not check_validity(centers, radii, tol=1e-8):
            # Roll back growth
            radii /= growth_rate
            break

    return centers, radii


def perturb_and_optimize(centers, radii, n_perturbations=20):
    """Try random perturbations to escape local optima."""
    n = N_CIRCLES
    best_centers = centers.copy()
    best_radii = radii.copy()
    best_sum = np.sum(radii)

    for _ in range(n_perturbations):
        # Random perturbation
        pert_centers = centers.copy()
        for i in range(n):
            pert_centers[i] += np.random.normal(0, 0.01, 2)
            pert_centers[i, 0] = np.clip(pert_centers[i, 0], radii[i], 1 - radii[i])
            pert_centers[i, 1] = np.clip(pert_centers[i, 1], radii[i], 1 - radii[i])

        # Try to grow from perturbed configuration
        test_radii = radii.copy() * 1.005
        test_centers = apply_forces(pert_centers, test_radii, lr=0.005, n_iters=200)

        if check_validity(test_centers, test_radii, tol=1e-8):
            if np.sum(test_radii) > best_sum:
                best_centers = test_centers.copy()
                best_radii = test_radii.copy()
                best_sum = np.sum(test_radii)

    return best_centers, best_radii


def final_optimization(centers, radii):
    """Final optimization using scipy Nelder-Mead."""
    n = N_CIRCLES
    params = np.concatenate([centers.flatten(), radii])

    result = minimize(
        objective,
        params,
        method="Nelder-Mead",
        options={
            "maxiter": 200000,
            "xatol": 1e-12,
            "fatol": 1e-12,
            "adaptive": True,
        },
    )

    opt_centers = result.x[:2 * n].reshape(n, 2)
    opt_radii = result.x[2 * n:]
    opt_radii = np.maximum(opt_radii, 0)

    # Clip centers to valid range
    for i in range(n):
        r = opt_radii[i]
        opt_centers[i, 0] = np.clip(opt_centers[i, 0], r, 1 - r)
        opt_centers[i, 1] = np.clip(opt_centers[i, 1], r, 1 - r)

    return opt_centers, opt_radii


def refine_radii(centers, radii):
    """Refine radii by checking how much each can grow."""
    n = N_CIRCLES

    for iteration in range(50):
        improved = False
        for i in range(n):
            # Calculate maximum possible radius for circle i
            max_r = min(centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])

            # Check overlap constraints
            for j in range(n):
                if i != j:
                    dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                    max_r = min(max_r, dist - radii[j])

            max_r = max(max_r, 0)

            if max_r > radii[i] + 1e-10:
                radii[i] = max_r
                improved = True

        if not improved:
            break

    # Re-optimize positions after radius refinement
    centers = apply_forces(centers, radii, lr=0.003, n_iters=500)

    return centers, radii


def run_packing():
    """Main function to pack 26 circles in unit square."""
    n = N_CIRCLES

    # Phase 1: Initial hexagonal configuration
    centers = hexagonal_initial(n)
    radii = np.full(n, 0.03)

    # Phase 2: Iterative growth
    centers, radii = iterative_growth(centers, radii, max_steps=500, growth_rate=1.003)

    # Phase 3: Perturb and try to improve
    centers, radii = perturb_and_optimize(centers, radii, n_perturbations=30)

    # Phase 4: More growth from improved configuration
    centers, radii = iterative_growth(centers, radii, max_steps=300, growth_rate=1.002)

    # Phase 5: Refine individual radii
    centers, radii = refine_radii(centers, radii)

    # Phase 6: Another round of growth
    centers, radii = iterative_growth(centers, radii, max_steps=200, growth_rate=1.001)

    # Phase 7: Final scipy optimization
    centers, radii = final_optimization(centers, radii)

    # Phase 8: Post-optimization refinement
    centers, radii = refine_radii(centers, radii)

    # Final cleanup - ensure no NaN and valid
    centers = np.nan_to_num(centers, nan=0.5)
    radii = np.nan_to_num(radii, nan=0.0)
    radii = np.maximum(radii, 0)

    # Final force application to ensure validity
    centers = apply_forces(centers, radii, lr=0.001, n_iters=1000)

    sum_radii = np.sum(radii)

    return centers, radii, sum_radii
