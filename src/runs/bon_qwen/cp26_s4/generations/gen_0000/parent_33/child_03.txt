from scipy.optimize import minimize

def compute_loss(v, n):
    """Objective function: maximize sum of radii with overlap/boundary penalties."""
    c = v[:2*n].reshape(n, 2)
    r = v[2*n:]
    loss = -np.sum(r)
    
    # Overlap penalties
    for i in range(n):
        for j in range(i + 1, n):
            dx = c[i, 0] - c[j, 0]
            dy = c[i, 1] - c[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            violation = r[i] + r[j] - dist
            if violation > 0:
                loss -= 5000.0 * violation ** 2
                
        # Boundary penalties
        x, y = c[i]
        rad = r[i]
        margins = [x - rad, 1.0 - x - rad, y - rad, 1.0 - y - rad]
        for m in margins:
            if m < 0:
                loss -= 5000.0 * m ** 2
                
    return loss
